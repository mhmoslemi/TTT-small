def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_one(h, k, dx):

    sh = (n - 1) - k

        g[:n - sh] += (1.0 - h[sh:])

            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc

def _binary_snap(h, rng):
    """Try projecting towards binary and see if it helps."""
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c5 = c5c
    for thresh in [0.3, 0.4, 0.45, 0.5, 0.55, 0.6, 0.7]:
        hb = np.where(h > thresh, 1.0, 0.0)
        hb = _repair(hb, n)
        c5b = _eval(hb)
        if c5b < best_c5:
            best_c5 = c5b
            best_h = hb.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 4

    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: SLSQP (0-100s)
    dl1 = min(deadline, t0 + 100)
    for alpha, tk, mi in [(2000, 40, 500), (8000, 20, 400), (40000, 10, 350)]:
        if time.time() > dl1:
            break
        h_s = _slsqp(h, alpha, tk, mi, dl1)
        c5v = _eval(h_s)
        if c5v < best_c5:
            best_c5 = c5v; best_h = h_s.copy()
        h = h_s

    # Phase 2: Nesterov multi-alpha (100-350s)
    dl2 = min(deadline, t0 + 350)
    h_w = best_h.copy()
    for alpha0, grow, tk, st, ns in [
        (3000, 2.0, 35, 0.005, 5000),
        (15000, 1.6, 20, 0.003, 4000),
        (80000, 1.3, 10, 0.002, 3500),
        (400000, 1.15, 6, 0.001, 2500),
    ]:
        if time.time() > dl2:
            break
        h_w = _nesterov(h_w, alpha0, grow, tk, ns, st, dl2)
        c5v = _eval(h_w)
        if c5v < best_c5:
            best_c5 = c5v; best_h = h_w.copy()

    # Phase 3: Lag attack (350-500s)
    dl3 = min(deadline, t0 + 500)
    h_atk, c5_atk = _lag_attack(best_h.copy(), 20000, rng, dl3)
    if c5_atk < best_c5:
        best_c5 = c5_atk; best_h = h_atk.copy()

    # Phase 4: Binary snap + re-optimize (500-560s)
    dl4 = min(deadline, t0 + 560)
    h_bin, c5_bin = _binary_snap(best_h, rng)
    if c5_bin < best_c5:
        best_c5 = c5_bin; best_h = h_bin.copy()
    if time.time() < dl4:
        h_n = _nesterov(best_h.copy(), 5000, 2.0, 25, 2000, 0.004, dl4)
        c5v = _eval(h_n)
        if c5v < best_c5:
            best_c5 = c5v; best_h = h_n.copy()

    # Phase 5: Annealing (560-750s)
    dl5 = min(deadline, t0 + 750)
    blocks = [1, 2, 3, 5, 7, 10, 15, 20, 35, 50]
    h_a, c5_a = _anneal(best_h.copy(), 150000, rng, dl5, 0.005, 1e-11, blocks)
    if c5_a < best_c5:
        best_c5 = c5_a; best_h = h_a.copy()

    # Phase 6: Pair descent (750-880s)
    dl6 = min(deadline, t0 + 880)
    h_pd, c5_pd = _pair_descent(best_h.copy(), dl6, rng)
    if c5_pd < best_c5:
        best_c5 = c5_pd; best_h = h_pd.copy()

    # Phase 7: High-alpha Nesterov polish (880-980s)
    dl7 = min(deadline, t0 + 980)
    h_n2 = _nesterov(best_h.copy(), 2000000, 1.1, 3, 3000, 0.0005, dl7)
    c5v = _eval(h_n2)
    if c5v < best_c5:
        best_c5 = c5v; best_h = h_n2.copy()

    # Phase 8: Final lag attack + pair (980-end)
    dl8 = deadline
    h_f, c5_f = _lag_attack(best_h.copy(), 8000, rng, dl8)
    if c5_f < best_c5:
        best_c5 = c5_f; best_h = h_f.copy()
    if time.time() < dl8 - 2:
        h_f2, c5_f2 = _pair_descent(best_h.copy(), dl8, rng)
        if c5_f2 < best_c5:
            best_c5 = c5_f2; best_h = h_f2.copy()

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
