def _repair(h, n):

        return np.full(n, 0.5)

            ct = c[idx]
        else:
            ct = c
        m = ct.max()
        s = np.clip(alpha * (ct - m), -400, 400)

        c = _correl(x)
        if topk < len(c):
            idx = np.argpartition(c, -topk)[-topk:].copy()
            ct = c[idx]

            ct = c

            i2 = j + off
            mask2 = (i2 >= 0) & (i2 < n)
            g[mask2] -= w[ii] * dx * x[i2[mask2]]
        return g

            break
        cor = _correl(c)
        idx_top = np.argpartition(cor, -m_top)[-m_top:]
        idx_top = idx_top[np.argsort(-cor[idx_top])]
        grads = np.zeros((m_top, n))
        for ii, k in enumerate(idx_top):
            off = k - (n - 1)
            j = np.arange(n)
            i1 = j - off
            m1 = (i1 >= 0) & (i1 < n)
            grads[ii][m1] += (1.0 - c[i1[m1]])
            i2 = j + off
            m2 = (i2 >= 0) & (i2 < n)
            grads[ii][m2] -= c[i2[m2]]
        c_obj = -grads[0]
        A_ub = grads[1:] if m_top > 1 else None
        b_ub = np.zeros(m_top - 1) if m_top > 1 else None
        A_eq = np.ones((1, n)); b_eq = np.zeros(1)
        bnds = [(-eps, eps)] * n
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bnds, method='highs', options={'presolve': True})
        except Exception:
            stall += 1
            if stall > 30:
                break
            continue
        if not res.success or res.x is None or np.max(np.abs(res.x)) < 1e-14:
            stall += 1
            if stall > 30:
                break
            continue
        stall = 0
        d = res.x
        improved = False
        for step in [1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.1]:
            h_new = _repair(c + step * d, n)
            c5_new = _eval(h_new)
            if c5_new < c5_cur - 1e-15:
                c, c5_cur = h_new, c5_new
                improved = True
                if c5_cur < best_c5:
                    best_c5, best_h = c5_cur, c.copy()
                break
        if not improved:
            rng_r = np.random.RandomState(it * 7 + 3)
            d2 = rng_r.randn(n)
            d2 -= d2.mean()
            d2 *= eps * 0.05 / max(np.max(np.abs(d2)), 1e-15)
            h_new = _repair(c + d2, n)
            c5_new = _eval(h_new)
            if c5_new < c5_cur:
                c, c5_cur = h_new, c5_new
                if c5_cur < best_c5:
                    best_c5, best_h = c5_cur, c.copy()
    return best_h, best_c5

def _slsqp_smooth(h, alpha, topk, maxit, deadline):
    n = len(h)
    def obj(x):
        c = _correl(x)
        idx = np.argpartition(c, -topk)[-topk:] if topk < len(c) else np.arange(len(c))
        ct = c[idx]
        m = ct.max()
        s = np.clip(alpha * (ct - m), -400, 400)
        return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha
    def jac(x):
        c = _correl(x)
        idx = np.argpartition(c, -topk)[-topk:].copy() if topk < len(c) else np.arange(len(c))
        ct = c[idx]
        m = ct.max()
        s = np.clip(alpha * (ct - m), -400, 400)
        w = np.exp(s - s.max()); w /= w.sum()
        g = np.zeros(n); dx = 2.0 / n
        for ii in range(len(idx)):
            if w[ii] < 1e-13: continue
            off = idx[ii] - (n - 1)
            j = np.arange(n)
            i1 = j - off; m1 = (i1 >= 0) & (i1 < n)
            g[m1] += w[ii] * dx * (1.0 - x[i1[m1]])
            i2 = j + off; m2 = (i2 >= 0) & (i2 < n)
            g[m2] -= w[ii] * dx * x[i2[m2]]
        return g
    try:
        r = minimize(obj, h, jac=jac, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
                                   'jac': lambda x: np.ones(n)}],
                     options={'maxiter': maxit, 'ftol': 1e-18, 'maxls': 50})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _anneal(h, niter, rng, deadline, T0, T1, blocks):
    n = len(h); c5c = _eval(h); bh = h.copy(); bc = c5c
    for it in range(niter):
        if time.time() > deadline: break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0 + bs].copy()
        d = 0.004 * max(T, 1e-9) ** 0.5 * rng.randn(bs)
        d -= d.mean()
        seg = np.clip(seg + d, 0, 1)
        hn = h.copy(); hn[i0:i0 + bs] = seg; hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h, c5c = hn, c5n
            if c5n < bc: bc, bh = c5n, h.copy()
    return bh, bc

def _polish(h, niter, rng, deadline):
    n = len(h); c5c = _eval(h); bh = h.copy(); bc = c5c
    steps = [5e-6, 2e-6, 1e-6, 5e-7, 2e-7, 1e-7]
    per = max(1, niter // len(steps))
    for ss in steps:
        for _ in range(per):
            if time.time() > deadline: return bh, bc
            i = rng.randint(n)
            off = rng.choice([-4, -3, -2, -1, 1, 2, 3, 4])
            j = (i + off) % n
            if i == j: j = (i + 1) % n
            d = ss * (2 * rng.random() - 1)
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-18:
                    c5c = c5n
                    if c5n < bc: bc, bh = c5n, h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d
    # Very fine single-pair moves
    for ss in [5e-8, 2e-8]:
        for _ in range(per):
            if time.time() > deadline: return bh, bc
            i = rng.randint(n); j = (i + 1 + rng.randint(0, 2)) % n
            d = ss * (2 * rng.random() - 1)
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-18:
                    c5c = c5n
                    if c5n < bc: bc, bh = c5n, h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h, best_c5 = h.copy(), _eval(h)

    # Phase 1: SLSQP smoothing (~70s)
    dl1 = min(deadline, t0 + 70)
    for a, tk, mi in [(3000, 30, 350), (15000, 12, 250), (60000, 6, 180)]:
        if time.time() > dl1: break
        hv = _slsqp_smooth(best_h.copy(), a, tk, mi, dl1)
        cv = _eval(hv)
        if cv < best_c5: best_c5, best_h = cv, hv.copy()

    # Phase 2: LP descent coarse (~200s)
    dl2 = min(deadline, t0 + 270)
    for eps, mk, ns in [(0.01, 10, 6000), (0.005, 7, 10000), (0.003, 5, 12000)]:
        if time.time() > dl2: break
        hv, cv = _lp_descent(best_h.copy(), ns, eps, mk, dl2)
        if cv < best_c5: best_c5, best_h = cv, hv.copy()

    # Phase 3: SLSQP fine (~60s)
    dl3 = min(deadline, t0 + 330)
    for a, tk, mi in [(50000, 8, 200), (200000, 4, 120)]:
        if time.time() > dl3: break
        hv = _slsqp_smooth(best_h.copy(), a, tk, mi, dl3)
        cv = _eval(hv)
        if cv < best_c5: best_c5, best_h = cv, hv.copy()

    # Phase 4: LP descent fine (~180s)
    dl4 = min(deadline, t0 + 510)
    for eps, mk, ns in [(0.0015, 4, 20000), (0.0008, 3, 25000)]:
        if time.time() > dl4: break
        hv, cv = _lp_descent(best_h.copy(), ns, eps, mk, dl4)
        if cv < best_c5: best_c5, best_h = cv, hv.copy()

    # Phase 5: Annealing + LP rescue (~120s)
    dl5 = min(deadline, t0 + 630)
    blocks = [1, 2, 3, 4, 6, 8, 10, 14]
    hv, cv = _anneal(best_h.copy(), 50000, rng, dl5, 0.005, 1e-11, blocks)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()
        dl5b = min(deadline, t0 + 680)
        hv2, cv2 = _lp_descent(best_h.copy(), 15000, 0.001, 4, dl5b)
        if cv2 < best_c5: best_c5, best_h = cv2, hv2.copy()

    # Phase 6: Ultra-fine LP (~80s)
    dl6 = min(deadline, t0 + 760)
    for eps, mk, ns in [(0.0004, 3, 30000), (0.0002, 2, 30000)]:
        if time.time() > dl6: break
        hv, cv = _lp_descent(best_h.copy(), ns, eps, mk, dl6)
        if cv < best_c5: best_c5, best_h = cv, hv.copy()

    # Phase 7: Polish (remaining)
    rem = max(1.0, deadline - time.time())
    npol = min(int(rem / 0.0006), 120000)
    best_h, best_c5 = _polish(best_h, npol, rng, deadline)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
