def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a

    return h

from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_m(h, m_idx, n):
    k = m_idx - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n-k])
        g[:n-k] -= h[k:]
    else:
        s = -k
        g[:n-s] += (1.0 - h[s:])
        g[s:] -= h[:n-s]
    return g * (2.0 / n)

def _smax_val(h, alpha, topk):
    c = _corr(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        idx = np.arange(len(c))
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    sm = s.max()
    return m + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha, idx, ct

def _smax_grad(h, alpha, topk):
    c = _corr(h)
    m = c.max()
    n = len(h)
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
        ct = c[idx]
    else:
        idx = np.arange(len(c))
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    sm = s.max()
    w = np.exp(s - sm)
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-15:
            continue
        g += w[ii] * _grad_m(h, idx[ii], n)
    return g

        if np.clip(h - mid, 0, 1).sum() > target:

    return np.clip(h - (lo + hi) * 0.5, 0, 1)

    try:

def _repair(h, n):

    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break

    dlC = min(deadline, t0 + 740)

    if ca < best_c5:
        best_c5, best_h = ca, ha.copy()

    # Phase D: Second L-BFGS-B (740-870s)
    dlD = min(deadline, t0 + 870)
    for alpha, tk, LAM, mi in [(50000, 12, 500.0, 400), (500000, 5, 2000.0, 300)]:
        if time.time() > dlD:
            break
        h2 = _lbf(best_h, alpha, tk, LAM, mi, dlD)
        c5v = _eval(h2)
        if c5v < best_c5:
            best_c5, best_h = c5v, h2

    # Phase E: Nesterov high-alpha polish (870-980s)
    dlE = min(deadline, t0 + 980)
    h3 = _nesterov(best_h.copy(), 2000000, 4, 3500, 0.0004, dlE)
    c5v = _eval(h3)
    if c5v < best_c5:
        best_c5, best_h = c5v, h3

    # Phase F: Very fine polish (980-end)
    best_h, best_c5 = _fine(best_h, rng, deadline)

    # Quick multi-resolution check
    if time.time() < deadline - 20:
        for dn in [-60, 60, 120]:
            if time.time() > deadline - 20:
                break
            n_try = n0 + dn
            if n_try < 200:
                continue
            hr = _resample(best_h, n_try)
            hr = _nesterov(hr, 100000, 8, 800, 0.002, min(deadline - 20, time.time() + 18))
            c5r = _eval(hr)
            if c5r < best_c5:
                best_c5, best_h = c5r, hr

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
