    step_sz = 1e-5
    while time.time() < t5:
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 4)) % n
        d = step_sz * rng.randn()
        ni, nj = h_c[i] + d, h_c[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h_c[i], h_c[j] = ni, nj
            f_new = _eval(h_c)
            if f_new < f_c:
                f_c = f_new
                step_sz = min(step_sz * 1.1, 2e-4)
                if f_c < best_c5:
                    best_c5 = f_c
                    best_h = h_c.copy()
            else:
                h_c[i], h_c[j] = ni - d, nj + d
                step_sz *= 0.7
                if step_sz < 1e-8:
                    step_sz = 5e-6

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
