            wk = w[ii] * ddx
            if sh >= 0:
                g[:nn - sh] += wk * (1.0 - x[sh:])
                if sh > 0:
                    g[sh:] -= wk * x[:-sh]
            else:
                sp = -sh
                g[sp:] += wk * (1.0 - x[:nn - sp])
                g[:nn - sp] -= wk * x[sp:]
        return g

    def _sum_fun(x):
        return x.sum() - n * 0.5

    def _sum_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac}]
    try:
        r = minimize(_smax_obj, h, jac=_grad_topk, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-18, 'maxls': 80})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _pair_swap_polish(h, niter, deadline, seed_rng):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for _ in range(niter):
        if time.time() > deadline:
            break
        i = seed_rng.randint(n)
        j = seed_rng.randint(n)
        if i == j:
            continue
        di = 0.002 * (2.0 * seed_rng.random() - 1.0)
        hi = h[i] + di
        hj = h[j] - di
        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
            h[i], h[j] = hi, hj
            c5n = _eval(h)
            if c5n < c5c - 1e-18:
                c5c = c5n
                if c5n < bc:
                    bc, bh = c5c, h.copy()
            else:
                h[i], h[j] = hi - di, hj + di
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 6
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h, best_c5 = h.copy(), _eval(h)

    # Phase 1: SLSQP refinement (0-120s)
    dl1 = min(deadline, t0 + 120)
    for a, tk, mi in [(2000, 30, 300), (10000, 15, 200), (50000, 8, 150)]:
        if time.time() > dl1:
            break
        hv = _slsqp_opt(best_h.copy(), a, tk, mi, dl1)
        cv = _eval(hv)
        if cv < best_c5:
            best_c5, best_h = cv, hv.copy()

    # Phase 2: Full LP descent moderate (120-380s)
    dl2 = min(deadline, t0 + 380)
    hv, cv = _lp_full(best_h.copy(), 12000, 0.004, 6, dl2, rng)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()

    # Phase 3: Reduced-dim LP descent (380-600s)
    dl3 = min(deadline, t0 + 600)
    hv, cv = _lp_reduced(best_h.copy(), 20000, 0.002, 4, dl3, rng, 0.02)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()

    # Phase 4: Very fine LP descent (600-820s)
    dl4 = min(deadline, t0 + 820)
    hv, cv = _lp_full(best_h.copy(), 15000, 0.001, 3, dl4, rng)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()

    # Phase 5: Ultra-fine reduced LP (820-980s)
    dl5 = min(deadline, t0 + 980)
    hv, cv = _lp_reduced(best_h.copy(), 30000, 0.0005, 2, dl5, rng, 0.01)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()

    # Phase 6: Pair swap polish (980-end)
    dl6 = deadline
    rem = max(0, dl6 - time.time())
    npol = min(int(rem / 0.0015), 40000)
    best_h, best_c5 = _pair_swap_polish(best_h, max(npol, 1000), dl6, rng)

    # Final micro-LP if time remains
    rem2 = max(0, deadline - time.time())
    if rem2 > 30:
        nf = min(int(rem2 / 0.05), 8000)
        hv, cv = _lp_full(best_h.copy(), nf, 0.0003, 2, deadline, rng)
        if cv < best_c5:
            best_c5, best_h = cv, hv.copy()

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
