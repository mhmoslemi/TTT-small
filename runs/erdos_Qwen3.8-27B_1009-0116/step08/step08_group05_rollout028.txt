import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12: break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_smooth(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    nc = len(c)
    if topk < nc:
        idx = np.argpartition(c, -topk)[-topk:].copy()
    else:
        idx = np.arange(nc)
    ct = c[idx]
    m = ct.max()
    s = alpha * (ct - m)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-14: continue
        k = idx[ii]
        sh = (n - 1) - k
        wk = w[ii] * dx
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n-sp] -= wk * h[sp:]
    return g

def _smax(h, alpha, topk):
    c = _correl(h)
    nc = len(c)
    if topk < nc:
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        ct = c
    m = ct.max()
    s = np.clip(alpha * (ct - m), -400, 400)
    s -= s.max()
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def _fista(h, alpha, topk, iters, step0, deadline):
    n = len(h)
    target = n * 0.5
    x = h.copy()
    y = h.copy()
    t = 1.0
    step = step0
    f_cur = _smax(x, alpha, topk)
    best_h, best_f = x.copy(), f_cur
    for it in range(iters):
        if time.time() > deadline: break
        g = _grad_smooth(y, alpha, topk)
        x_new = _proj(y - step * g, target)
        f_new = _smax(x_new, alpha, topk)
        if f_new <= f_cur + 1e-15:
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
            y = x_new + ((t - 1.0) / t_new) * (x_new - x)
            x = x_new
            t = t_new
            f_cur = f_new
            step = min(step * 1.02, step0 * 25.0)
            if f_new < best_f:
                best_f, best_h = f_new, x.copy()
        else:
            step *= 0.5
            if step < 1e-14: break
            x_new = _proj(y - step * g, target)
            f_new = _smax(x_new, alpha, topk)
            if f_new <= f_cur + 1e-15:
                t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
                y = x_new + ((t - 1.0) / t_new) * (x_new - x)
                x = x_new
                t = t_new
                f_cur = f_new
                if f_new < best_f:
                    best_f, best_h = f_new, x.copy()
    return _repair(best_h, n)

def _slsqp_run(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline: return h
    def obj(x): return _smax(x, alpha, topk)
    def jac(x): return _grad_smooth(x, alpha, topk)
    try:
        r = minimize(obj, h, jac=jac, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq',
                                   'fun': lambda x: x.sum() - n * 0.5,
                                   'jac': lambda x: np.ones(n)}],
                     options={'maxiter': maxit, 'ftol': 1e-18, 'maxls': 60})
        if r.x is not None:
            rx = _repair(r.x, n)
            if _eval(rx) <= _eval(h) + 1e-12:
                return rx
    except Exception:
        pass
    return h

def _pair_cd(h, niter, rng, deadline):

    c5c = _eval(h)
    best_h, best_c5 = h.copy(), c5c
    spans = [1, 2, 3, 5, 8, 12, 16]
    for it in range(niter):
        if time.time() > deadline: break

        sp = int(spans[rng.randint(len(spans))])
        j = (i + sp) % n
        bi, bj = h[i], h[j]
        best_d = 0.0
        best_c = c5c
        for d in [0.002, -0.002, 0.005, -0.005, 0.01, -0.01, 0.001, -0.001]:
            ni, nj = bi + d, bj - d
            if 0 <= ni and ni <= 1 and 0 <= nj and nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < best_c:
                    best_c, best_d = c5n, d
        if best_d != 0.0:
            h[i] = bi + best_d
            h[j] = bj - best_d
            c5c = best_c
            if c5c < best_c5:
                best_c5, best_h = c5c, h.copy()
        else:
            h[i], h[j] = bi, bj
    return best_h, best_c5

def _anneal(h, niter, rng, deadline):
    n = len(h)
    c5c = _eval(h)
    best_h, best_c5 = h.copy(), c5c
    T0, T1 = 0.005, 1e-13
    blocks = [2, 3, 4, 5, 6, 8, 10, 14, 20]
    for it in range(niter):
        if time.time() > deadline: break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = int(min(blocks[rng.randint(len(blocks))], n))
        i0 = rng.randint(0, max(1, n - bs + 1))
        seg = h[i0:i0+bs].copy()
        amp = 0.005 * max(T, 1e-8) ** 0.5
        d = amp * rng.randn(bs)
        d -= d.mean()
        seg_new = np.clip(seg + d, 0, 1)
        diff = seg.sum() - seg_new.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - seg_new) if diff > 0 else np.argsort(seg_new)

        hn[i0:i0+bs] = seg_new
