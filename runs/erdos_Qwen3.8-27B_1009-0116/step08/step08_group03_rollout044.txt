        mN = Nv[1:] >= mx - tol_abs
        if np.any(mN):
            pos = np.nonzero(mN)[0] + 1; wN[pos] = 1.0; cnt += pos.size
    if cnt == 0:
        return np.zeros(n)
    wP /= cnt; wN /= cnt
    u = np.zeros(n)
    if n > 1:
        u[1:] = wP[1:] + wN[1:]
    cp = np.zeros(n); cn = np.zeros(n)
    if n > 1:
        cp[1:] = np.cumsum(wP[1:]); cn[1:] = np.cumsum(wN[1:])
    linP = cp[n - 1::-1]; linN = cn
    q = np.zeros(n)
    if n > 1 and np.any(u[1:]):
        Nf = _fft_size(n)
        H = np.fft.rfft(h, Nf)
        R = np.fft.irfft(H * np.fft.rfft(u[::-1], Nf), Nf)
        L = np.fft.irfft(H * np.fft.rfft(u, Nf), Nf)
        q = -(R[n - 1:2 * n - 1].real + L[:n].real)
    return dx * (wP[0] * (1.0 - 2.0 * h) + linP + linN + q)

def _fista(h0, alpha, max_time, L0=30.0):
    if max_time <= 0:
        return h0.copy(), _c5(h0)
    n = len(h0); S = n * 0.5; t0 = time.time()
    h = _repair(h0, n); hp = h.copy(); y = h.copy(); t = 1.0
    fy = _smooth_val(y, alpha)
    best_h = h.copy(); best_c = _c5(h)
    Lcur = L0; stale = 0
    for it in range(400000):
        if time.time() - t0 > max_time:
            break
        if it > 0:
            tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
            y = h + ((t - 1.0) / tn) * (h - hp)
            t = tn
            if y.min() < -0.2 or y.max() > 1.2 or abs(y.sum() - S) > 1e-5:
                y = _proj(y, S)
        g = _smooth_grad(y, alpha)
        step = 1.0 / max(Lcur, 1e-12)
        ok = False
        for _ in range(12):
            xn = _proj(y - step * g, S)
            d = xn - y
            fx = _smooth_val(xn, alpha)
            if fx <= fy + float(np.dot(g, d)) + 0.5 / step * float(np.dot(d, d)) + 1e-10:
                ok = True
                break
            step *= 0.4
        if ok:
            hp = h; h = xn; fy = _smooth_val(h, alpha)
            Lcur = max(Lcur * 0.88, 1.0)
            stale = 0
            if it % 15 == 0:
                c = _c5(h)
                if c < best_c:
                    best_c = c; best_h = h.copy()
        else:
            stale += 1; hp = h; y = h; t = 1.0
            Lcur = max(Lcur * 0.5, 1.0)
            if stale > 12:
                break
    c = _c5(h)
    if c < best_c:
        best_c = c; best_h = h.copy()
    return best_h, best_c

def _sa_search(h0, rng, deadline, T0=8e-5, T1=5e-8):
    t0 = time.time()
    n = len(h0); S = n * 0.5
    h = h0.copy(); c = _c5(h)
    best_c = c; best_h = h.copy()
    P, Nv, mx = _pn(h)
    tol = max(1e-9, 4.0 * (2.0 / n))
    g = _grad_active(h, tol)
    order = np.argsort(g)
    q = max(4, n // 10)
    low = order[:q]; high = order[-q:]
    dur = max(1e-9, deadline - t0)
    recomp = 3000; it = 0
    while time.time() < deadline - 0.5:
        frac = (time.time() - t0) / dur
        T = T0 * (T1 / T0) ** frac
        if it >= recomp:
            c = _c5(h)
            if c < best_c:
                best_c = c; best_h = h.copy()
            P, Nv, mx = _pn(h)
            tol = max(1e-9, 4.0 * (2.0 / n) * (1.0 - 0.5 * frac))
            g = _grad_active(h, tol)
            order = np.argsort(g)
            q = max(4, n // 10)
            low = order[:q]; high = order[-q:]
            recomp = it + 3000
        base = max(1e-12, 1.5e-4 * (1.0 + frac) ** (-3.0))
        r = rng.random()
        if r < 0.65:
            i = int(low[rng.randint(len(low))])
            j = int(high[rng.randint(len(high))])
        elif r < 0.85:
            i = int(low[rng.randint(len(low))])
            j = int(rng.randint(n))
        elif r < 0.95:
            i = int(high[rng.randint(len(high))])
            j = int(rng.randint(n))
        else:
            i = int(rng.randint(n)); j = int(rng.randint(n))
        if i == j:
            it += 1; continue
        if g[i] <= g[j]:
            s = 1
        elif g[j] < g[i]:
            s = -1
        else:
            s = 1 if rng.random() < 0.5 else -1
        if s > 0:
            avail = min(1.0 - h[i], h[j])
        else:
            avail = min(h[i], 1.0 - h[j])
        if avail < 1e-14:
            it += 1; continue
        mult = 0.3 + 1.7 * rng.random()
        if rng.random() < 0.02:
            mult *= 6.0
        d = s * min(avail, base * mult)
        h[i] += d; h[j] -= d
        cn = _c5(h)
        diff = cn - c
        if diff < -1e-15:
            c = cn
            if c < best_c:
                best_c = c; best_h = h.copy()
        else:
            if rng.random() < np.exp(-diff / max(T, 1e-15)):
                c = cn
                if c < best_c:
                    best_c = c; best_h = h.copy()
            else:
                h[i] -= d; h[j] += d
        it += 1
    return best_h, best_c

def _resample(h, m):
    n = len(h)
    if n == m:
        return _repair(h, n)
    if n % m == 0:
        out = h.reshape(m, n // m).mean(axis=1)
    else:
        x0 = (np.arange(n) + 0.5) * (2.0 / n)
        x1 = (np.arange(m) + 0.5) * (2.0 / m)
        out = np.interp(x1, x0, h)
    return _proj(out, m * 0.5)

def _variants(h):
    n = len(h)
    v1 = _repair(1.0 - h, n)
    v2 = _repair(h[::-1].copy(), n)
    v3 = _proj(0.5 + 0.5 * (h - h[::-1]), n * 0.5)
    v4 = _proj(0.5 + 0.5 * (h + h[::-1] - 1.0), n * 0.5)
    return [v1, v2, v3, v4]

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(560, 0.5)
    n0 = len(h0)
    h0 = _repair(h0, n0)
    best = h0.copy(); best_c = _c5(best)
    for vv in _variants(h0):
        cv = _c5(vv)
        if cv < best_c:
            best = vv; best_c = cv
    # Phase 1: FISTA sweep (200s)
    end1 = t0 + min(200.0, 0.2 * budget_s)
    alphas1 = [50, 200, 800, 4000, 15000]
    per1 = max(0.0, end1 - time.time()) / len(alphas1)
    hcur = best.copy()
    for a in alphas1:
        if time.time() >= end1 - 1.0 or per1 < 1.0:
            break
        bh, bc = _fista(hcur, a, per1, L0=30.0)
        if bc < best_c:
            best = bh; best_c = bc
        hcur = bh
    # Phase 2: multi-resolution (250s)
    end2 = t0 + min(450.0, 0.55 * budget_s)
    hmain = best if len(best) == n0 else _resample(best, n0)
    for fac in [2, 4]:
        m = n0 // fac
        if m < 32 or time.time() >= end2 - 5.0:
            break
        slot = min(60.0, end2 - time.time() - 5.0)
        if slot < 5.0:
            break
        hm = _resample(hmain, m)
        for a in [100, 600, 4000]:
            if time.time() >= end2 - 3.0:
                break
            bh, bc = _fista(hm, a, min(18.0, end2 - time.time() - 3.0), L0=20.0)
            if bc < best_c:
                best = bh; best_c = bc
            hm = bh
        hu = _resample(hm, n0)
        cu = _c5(hu)
        if cu < best_c:
            best = hu; best_c = cu
        hmain = best if len(best) == n0 else _resample(best, n0)
    # Phase 3: FISTA at native resolution with high alpha (250s)
    end3 = t0 + budget_s - 380.0
    if end3 > time.time() + 10.0:
        hmain = best if len(best) == n0 else _resample(best, n0)
        alphas3 = [200, 1000, 5000, 20000, 80000, 300000]
        per3 = max(0.0, end3 - time.time()) / len(alphas3)
        for a in alphas3:
            if time.time() >= end3 - 2.0:
                break
            bh, bc = _fista(hmain, a, per3, L0=50.0)
            if bc < best_c:
                best = bh; best_c = bc
            hmain = bh
        if len(best) != n0:
            best = _resample(best, n0); best_c = _c5(best)
    # Phase 4: Simulated annealing local search (370s)
    end4 = t0 + budget_s - 2.0
    if end4 > time.time():
        T0val = 1.0e-4 * (best_c - 0.30)
        T0val = max(3e-5, min(T0val, 2e-4))
        best, best_c = _sa_search(best, rng, end4, T0=T0val, T1=2e-8)
    # Final polish
    best = _repair(best, len(best))
    best_c = _c5(best)
    for vv in _variants(best):
        cv = _c5(vv)
        if cv < best_c:
            best = vv; best_c = cv
    best = _repair(best, len(best))
    best_c = _c5(best)
    return best, best_c, len(best)
