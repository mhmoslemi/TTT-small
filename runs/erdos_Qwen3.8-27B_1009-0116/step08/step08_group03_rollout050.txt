import numpy as np
import time

def _fft_size(n):
    m = 2 * n - 1
    N = 1
    while N < m:
        N <<= 1
    return N

def _repair(h, n):
    t = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if s <= 1e-15:
        return np.full(n, 0.5)
    for _ in range(80):
        h *= t / s
        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
        if abs(s - t) < 1e-12:
            break
    d = t - h.sum()
    if abs(d) > 1e-12:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)
    x += (S - x.sum()) / x.size
    for _ in range(10):
        np.clip(x, 0.0, 1.0, out=x)
        d = S - x.sum()
        if abs(d) < 1e-11:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
    if abs(x.sum() - S) > 1e-8:
        lo, hi = float(np.min(v)) - 1.0, float(np.max(v)) + 1.0
        for _ in range(50):
            mid = 0.5 * (lo + hi)
            if np.clip(v - mid, 0.0, 1.0).sum() > S:
                lo = mid
            else:
                hi = mid
        x = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    return x

def _pn_arrays(h):
    n = len(h); N = _fft_size(n)
    H = np.fft.rfft(h, N)
    ac = np.fft.irfft(H * np.conj(H), N).real[:n]
    pref = np.empty(n + 1); pref[0] = 0.0; np.cumsum(h, out=pref[1:])
    total = pref[n]
    P = np.empty(n); P[0] = total - ac[0]
    Nv = np.zeros(n)
    if n > 1:
        idx = np.arange(1, n)
        P[idx] = pref[n - idx] - ac[idx]
        Nv[idx] = total - pref[idx] - ac[idx]
        mx = float(max(np.max(P), np.max(Nv)))
    else:
        mx = float(P[0])
    return P, Nv, mx

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _grad_from_weights(h, wP, wN):
    n = len(h); dx = 2.0 / n
    u = np.zeros(n)
    if n > 1:
        u[1:] = wP[1:] + wN[1:]
        cp = np.zeros(n); cn = np.zeros(n)
        cp[1:] = np.cumsum(wP[1:]); cn[1:] = np.cumsum(wN[1:])
    else:
        cp = np.zeros(n); cn = np.zeros(n)
    linP = cp[n - 1::-1]; linN = cn
    q = np.zeros(n)
    if n > 1 and np.any(u[1:]):
        Nf = _fft_size(n); Hf = np.fft.rfft(h, Nf)
        Ur = np.fft.rfft(u, Nf); Urev = np.fft.rfft(u[::-1], Nf)
        R = np.fft.irfft(Hf * Urev, Nf)[n - 1:2*n - 1].real
        L = np.fft.irfft(Hf * Ur, Nf)[:n].real
        q = -(R + L)
    return dx * (wP[0] * (1.0 - 2.0 * h) + linP + linN + q)

def _active_grad(h, P, Nv):
    n = len(h); mx = float(P[0])
    if n > 1:
        mx = max(mx, float(np.max(P[1:])), float(np.max(Nv[1:])))
    tol = max(1e-9, 10.0 * (2.0 / n))
    wP = np.zeros(n); wN = np.zeros(n)
    if P[0] >= mx - tol:
        wP[0] = 1.0
    if n > 1:
        mp = P[1:] >= mx - tol; mn = Nv[1:] >= mx - tol
        wP[1:] = mp.astype(float); wN[1:] = mn.astype(float)
    cnt = wP.sum() + wN.sum()
    if cnt == 0:
        return np.zeros(n)
    wP /= cnt; wN /= cnt
    return _grad_from_weights(h, wP, wN)

def _smooth_val(h, alpha):
    P, Nv, mx = _pn_arrays(h)
    n = len(h); dx = 2.0 / n
    e = np.exp(np.clip(alpha * (P[0] - mx), -400.0, 0.0))
    if n > 1:
        e += np.exp(np.clip(alpha * (P[1:] - mx), -400.0, 0.0)).sum()
        e += np.exp(np.clip(alpha * (Nv[1:] - mx), -400.0, 0.0)).sum()
    return dx * (mx + np.log(max(e, 1e-300)) / alpha)

def _smooth_grad(h, alpha):
    P, Nv, mx = _pn_arrays(h); n = len(h)
    wP = np.zeros(n); wN = np.zeros(n)
    e0 = np.exp(np.clip(alpha * (P[0] - mx), -400.0, 0.0))
    tot = e0
    if n > 1:
        ep = np.exp(np.clip(alpha * (P[1:] - mx), -400.0, 0.0))
        en = np.exp(np.clip(alpha * (Nv[1:] - mx), -400.0, 0.0))
        tot += ep.sum() + en.sum()
        if tot <= 0:
            tot = 1.0; e0 = 1.0
        wP[0] = e0 / tot; wP[1:] = ep / tot; wN[1:] = en / tot
    else:
        wP[0] = 1.0
    return _grad_from_weights(h, wP, wN)

            tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0*t*t))

                y = _proj(y, S)

        g = _smooth_grad(y, alpha)
        step = 1.0 / max(Lcur, 1e-12); accepted = False
        for _ in range(14):

            if fx <= rhs + 1e-10:
