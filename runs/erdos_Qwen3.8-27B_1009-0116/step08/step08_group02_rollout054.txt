    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        # soft max
        m = c.max()
        s = alpha * (c - m)
        s -= s.max()
        softmax_val = m + np.log(np.exp(s).sum()) / alpha

    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _binarize(h):
    n = len(h)
    target = n // 2
    idx = np.argpartition(h, n - target)[n - target:]
    out = np.zeros(n)
    out[idx] = 1.0
    return out

def _sa_binary(h, rng, t_end, T0=0.04, cool=0.93, reheat_T=0.025, reheat_stall=30000):
    n = len(h)
    c5 = _eval(h)
    best_h = h.copy()
    best_c5 = c5
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    np1 = len(ones)
    np0 = len(zeros)
    T = T0
    stall = 0
    it = 0
    while time.time() < t_end:
        ii = rng.randint(np1)
        jj = rng.randint(np0)
        i = ones[ii]
        j = zeros[jj]
        h[i] = 0.0
        h[j] = 1.0
        c5_new = _eval(h)
        delta = c5_new - c5
        accept = delta < 0 or (T > 1e-12 and rng.rand() < np.exp(-delta / T))
        if accept:
            c5 = c5_new
            ones[ii] = j
            zeros[jj] = i
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            stall = 0 if c5 < best_c5 else stall + 1
        else:
            h[i] = 1.0
            h[j] = 0.0
            stall += 1
        it += 1
        if it % 5000 == 0:
            T *= cool
            if stall > reheat_stall and T < reheat_T * 0.5:
                T = reheat_T
                stall = 0
    return best_h, best_c5
