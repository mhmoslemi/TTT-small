            return g
        try:
            res = minimize(obj_fn, h, jac=grad_fn, method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-9})
            h2 = _repair(res.x, n)
            f2 = _eval(h2)
            if f2 < best_f:
                best_f, best_h = f2, h2.copy()
        except Exception:
            pass
    return best_h, best_f

def _slsqp_smax(h, alpha, topk, maxiter, deadline):
    n = len(h)
    target = n * 0.5
    if time.time() > deadline:
        return h
    try:
        res = minimize(lambda x: _smax_obj(x, alpha, topk), h,
                      jac=lambda x: _smax_grad(x, alpha, topk)[0],
                      method='SLSQP', bounds=[(0.0, 1.0)] * n,
                      constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - target,
                                    'jac': lambda x: np.ones(n)}],
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(res.x, n)
    except Exception:
        return h

def _anneal(h, niter, rng, deadline, T0=0.006, T1=1e-11):
    n = len(h)
    c5c = _eval(h); bh = h.copy(); bc = c5c
    blocks = [2, 3, 4, 5, 7, 10, 15, 25]
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0 + bs].copy()
        d = 0.004 * max(T, 1e-9) ** 0.5 * rng.randn(bs)
        d -= d.mean(); seg += d; seg = np.clip(seg, 0, 1)
        hn = h.copy(); hn[i0:i0 + bs] = seg; hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n <= c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h, c5c = hn, c5n
            if c5n < bc:
                bc, bh = c5n, h.copy()
    return bh, bc

def _polish(h, niter, rng, deadline, scale=1e-5):
    nc = len(h)
    c5c = _eval(h); bh = h.copy(); bc = c5c
    for _ in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(nc)
        j = (i + 1 + rng.randint(0, 9)) % nc
        if i == j:
            j = (i + 2) % nc
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-18:
                c5c = c5n
                if c5n < bc:
                    bc, bh = c5n, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc

def _coord_polish(h, deadline):
    n = len(h)
    target = n / 2.0
    c5 = _eval(h)
    for i in range(n):
        if time.time() > deadline:
            break
        best_v = h[i]; best_c5 = c5
        for dv in [-0.008, -0.004, -0.002, -0.001, 0.001, 0.002, 0.004, 0.008]:
            v = h[i] + dv
            if 0.0 <= v <= 1.0:
                hn = h.copy(); hn[i] = v; hn = _repair(hn, n)
                c5t = _eval(hn)
                if c5t < best_c5:
                    best_c5, best_v = c5t, v
        if best_v != h[i]:
            h[i] = best_v; h = _repair(h, n); c5 = best_c5
    return h, c5

def _jitter_reopt(h, rng, step0, deadline, jscale=0.002):
    n = len(h)
    c5c = _eval(h)
    d = jscale * rng.randn(n); d -= d.mean()
    hn = _repair(np.clip(h + d, 0, 1), n)
    results = []
    h2, c2 = _nesterov(hn, 600, step0, deadline, mode='exact', tol=0.002, beta=0.88)
    results.append((c2, h2))
    h3, c3 = _nesterov(hn, 400, step0 * 0.5, deadline, mode='smax', alpha=30000, topk=10, beta=0.85)
    results.append((c3, h3))
    best = min(results, key=lambda x: x[0])
    if best[0] < c5c:
        return best[1], best[0]
    return h, c5c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy(); best_c5 = _eval(best_h)

    # Phase 1: SLSQP warm-up
    dl1 = min(deadline, t0 + 50)
    h1 = _slsqp_smax(h, 5000, 20, 200, dl1)
    c1 = _eval(h1)
    if c1 < best_c5:
        best_c5, best_h = c1, h1.copy()

    # Phase 2: LBFGS with penalty + annealed alpha (key new technique)
    dl2 = min(deadline, t0 + 180)
    for alpha, tk in [(1500, 30), (8000, 20), (40000, 12)]:
        if time.time() > dl2:
            break
        h2, c5 = _lbfgs_smax(best_h.copy(), alpha, tk, dl2, max_iter=200)
        if c5 < best_c5:
            best_c5, best_h = c5, h2.copy()

    # Phase 3: Nesterov exact (progressive tolerance tightening)
    dl3 = min(deadline, t0 + 380)
    for step0, tol, mom, ns in [(0.003, 0.005, 0.85, 4000),
                                (0.0015, 0.002, 0.88, 4000),
                                (0.0008, 0.0008, 0.91, 4000)]:
        if time.time() > dl3:
            break
        h2, c5 = _nesterov(best_h.copy(), ns, step0, dl3, mode='exact', tol=tol, beta=mom)
        if c5 < best_c5:
            best_c5, best_h = c5, h2.copy()

    # Phase 4: Smax Nesterov (different smoothing)
    dl4 = min(deadline, t0 + 520)
    for alpha, tk, s0, ns in [(40000, 8, 0.002, 3000), (150000, 5, 0.001, 3000)]:
        if time.time() > dl4:
            break
        h2, c5 = _nesterov(best_h.copy(), ns, s0, dl4, mode='smax', alpha=alpha, topk=tk, beta=0.87)
        if c5 < best_c5:
            best_c5, best_h = c5, h2.copy()

    # Phase 5: Jitter + re-optimize (multiple restarts)
    dl5 = min(deadline, t0 + 720)
    for _ in range(10):
        if time.time() > dl5:
            break
        hj, cj = _jitter_reopt(best_h.copy(), rng, 0.002, dl5, jscale=0.0015)
        if cj < best_c5:
            best_c5, best_h = cj, hj.copy()

    # Phase 5b: LBFGS from best
    if time.time() < dl5:
        h2, c5 = _lbfgs_smax(best_h.copy(), 30000, 10, dl5, max_iter=150)
        if c5 < best_c5:
            best_c5, best_h = c5, h2.copy()

    # Phase 6: Block annealing
    dl6 = min(deadline, t0 + 920)
    best_h, best_c5 = _anneal(best_h.copy(), 100000, rng, dl6)

    # Phase 7: Final Nesterov (very fine)
    dl7 = min(deadline, t0 + budget_s - 60)
    h2, c5 = _nesterov(best_h.copy(), 4000, 0.0003, dl7, mode='exact', tol=0.0004, beta=0.93)
    if c5 < best_c5:
        best_c5, best_h = c5, h2.copy()

    # Phase 8: Coordinate polish
    if time.time() < dl7 + 30:
        best_h, best_c5 = _coord_polish(best_h, min(dl7 + 30, deadline))

    # Phase 9: Fine 2-point polish
    rem = max(0, deadline - time.time())
    n_pol = min(int(rem / 0.0012), 50000)
    best_h, best_c5 = _polish(best_h, n_pol, rng, deadline, 5e-6)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
