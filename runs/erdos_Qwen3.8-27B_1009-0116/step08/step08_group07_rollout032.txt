def _eval(h):

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0

        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid

    target = n * 0.5
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()

            for idx in order:

                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:

def _grad_lags(h, lags):

            g[sp:] += (1.0 - h[:-sp])

        else:
            ct = c
        s = np.clip(alpha * (ct - m), -400, 400)
        sm = s.max()
        return m + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha
    def grad(x):
        c = _correl(x); m = c.max()
        if topk < len(c):
            idx = np.argpartition(c, -topk)[-topk:]; ct = c[idx]
        else:
            idx = np.arange(len(c)); ct = c
        s = np.clip(alpha * (ct - m), -400, 400)
        sm = s.max()
        w = np.exp(s - sm); w /= w.sum()
        g = np.zeros(n)
        for ii in range(len(idx)):
            if w[ii] < 1e-14: continue
            k = idx[ii]; sh = (n - 1) - k; wk = w[ii] * (2.0 / n)
            if sh >= 0:
                g[:n - sh] += wk * (1.0 - x[sh:])
                if sh > 0: g[sh:] -= wk * x[:-sh]
            else:
                sp = -sh
                g[sp:] += wk * (1.0 - x[:-sp])
                g[:n - sp] -= wk * x[sp:]
        return g
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5, 'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(obj, h, jac=grad, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-18})
        return _repair(r.x, n)
    except Exception:
        return h

def _anneal(h, niter, rng, deadline, blocks, T0, T1):
    n = len(h)
    c5c = _eval(h); bh = h.copy(); bc = c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0 + bs].copy()
        d = 0.002 * max(T, 1e-10) ** 0.5 * rng.randn(bs)
        d -= d.mean()
        seg = np.clip(seg + d, 0, 1)
        hn = h.copy(); hn[i0:i0 + bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h, c5c = hn, c5n
            if c5n < bc:
                bc, bh = c5n, h.copy()
    return bh, bc

def _coord_descent(h, niter, rng, deadline):
    nc = len(h); c5c = _eval(h); bh = h.copy(); bc = c5c
    for _ in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(nc)
        j = (i + 1 + rng.randint(0, 4)) % nc
        for scale in [2e-7, 5e-7, 1e-6]:
            d = scale * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c:
                    c5c = c5n
                    break
                h[i], h[j] = ni - d, nj + d
        if c5c < bc:
            bc, bh = c5c, h.copy()
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    h = _repair(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values))
    n = len(h)
    best_h, best_c5 = h.copy(), _eval(h)

    # Phase 1: SLSQP (40s)
    dl1 = min(deadline, t0 + 40)
    for alpha, tk, mi in [(4000, 15, 250), (15000, 8, 200)]:
        if time.time() > dl1: break
        h = _slsqp(best_h, alpha, tk, mi, dl1)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5, best_h = c5v, h.copy()

    # Phase 2: Nesterov exact (250s)
    dl2 = min(deadline, t0 + 290)
    for step0, tol, mom, ns in [(0.004, 0.003, 0.84, 4000),
                                (0.0015, 0.001, 0.89, 4000),
                                (0.0005, 0.0005, 0.92, 3000)]:
        if time.time() > dl2: break
        h2, c5v = _nesterov(best_h.copy(), ns, step0, dl2, tol, mom)
        if c5v < best_c5:
            best_c5, best_h = c5v, h2.copy()

    # Phase 3: Smax Nesterov (130s)
    dl3 = min(deadline, t0 + 420)
    for alpha, topk, step0, ns in [(40000, 5, 0.002, 2500), (150000, 3, 0.001, 2500)]:
        if time.time() > dl3: break
        h2, c5v = _smax_nest(best_h.copy(), alpha, topk, ns, dl3, step0)
        if c5v < best_c5:
            best_c5, best_h = c5v, h2.copy()

    # Phase 4: Multi-restart perturbation (180s)
    dl4 = min(deadline, t0 + 600)
    for attempt in range(6):
        if time.time() > dl4: break
        scale = 0.0015 * (1.0 - 0.1 * attempt)
        d = scale * rng.randn(n)
        d -= d.mean()
        hj = _repair(np.clip(best_h + d, 0, 1), n)
        h2, c5v = _nesterov(hj, 400, 0.0008, dl4, 0.001, 0.88)
        if c5v < best_c5:
            best_c5, best_h = c5v, h2.copy()

    # Phase 5: Annealing (130s)
    dl5 = min(deadline, t0 + 730)
    blocks = [2, 3, 4, 5, 7, 11]
    best_h, best_c5 = _anneal(best_h.copy(), 35000, rng, dl5, blocks, 0.0015, 1e-11)

    # Phase 6: Final fine Nesterov (200s)
    dl6 = min(deadline, t0 + budget_s - 50)
    h2, c5v = _nesterov(best_h.copy(), 6000, 0.0002, dl6, 0.0002, 0.93)
    if c5v < best_c5:
        best_c5, best_h = c5v, h2.copy()

    # Phase 7: Coordinate descent polish
    dl7 = deadline
    rem = max(1, int(dl7 - time.time()))
    n_pol = min(rem * 800, 120000)
    best_h, best_c5 = _coord_descent(best_h, n_pol, rng, dl7)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
