    idx = np.argsort(c)[-m:]
    s = alpha * (c[idx] - c.max())
    s = np.clip(s - s.max(), -500, 500)
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _sum_fn(x, target):
    return x.sum() - target

def _sum_jac(x, target):
    return np.ones(len(x))

def _slsqp(h, alpha, m_lags, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fn, 'jac': _sum_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_obj_fn, h, jac=_grad_fn, args=(alpha, m_lags, n),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return h

def _proj_grad_step(h, alpha, m_lags, v, step, mu):
    n = len(h)
    g = _grad_fn(h, alpha, m_lags, n)
    v = mu * v + (1.0 - mu) * g
    h_new = _proj(h - step * v, n * 0.5)
    return h_new, v

    m = n // f

import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode='full').max() * 2.0 / n)

def _eval_vec(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(300):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
    return h

def _proj(h, target):
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _obj_fn(x, alpha, m, n):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    idx = np.argsort(c)[-m:]
    s = alpha * (c[idx] - c.max())
    s = np.clip(s - s.max(), -500, 500)
    return c.max() + np.log(np.exp(s).sum()) / alpha

def _grad_fn(x, alpha, m, n):

    c = np.correlate(x, 1.0 - x, mode='full') * dx
    idx = np.argsort(c)[-m:]

        else:

            g[:n - M] += wk * (1.0 - x[M:])

    return x.sum() - target

    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fn, 'jac': _sum_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_obj_fn, h, jac=_grad_fn, args=(alpha, m_lags, n),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except Exception:
        return h

def _proj_grad(h, alpha, m_lags, nsteps, step0):
    n = len(h)
    v = np.zeros(n)
    mu = 0.6
    step = step0
    cur_c = _eval(h)
    for _ in range(nsteps):
        g = _grad_fn(h, alpha, m_lags, n)
        v = mu * v + (1.0 - mu) * g
        taken = False
        for _t in range(12):
            h_new = _proj(h - step * v, n * 0.5)
            c_new = _eval(h_new)
            if c_new < cur_c - 1e-15:
                h = h_new
                cur_c = c_new
                step = min(step * 1.05, step0 * 10.0)
                taken = True
                break
            step *= 0.4

            step = min(step * 1.4, step0 * 10.0)
    return _repair(h, n)

def _local_search(h, n_iter, rng, base_step):
