def _repair(h, n):
    # Clip and adjust sum
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # Binary search for shift
        lo, hi = -max(h.max(), 1-h.min()) - 0.1, max(h.max(), 1-h.min()) + 0.1
        # Actually simpler: uniform shift then clip
        for _ in range(60):
            mid = (lo + hi) / 2
            shifted = np.clip(h - mid, 0, 1)
            if shifted.sum() > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo + hi) / 2, 0, 1)
        # Fine adjustment
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                order = np.argsort(1.0 - h)
            else:
                order = np.argsort(h)
            for idx in order:

                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-14:

    return h

def _eval(h):

    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * 2.0 / n)

    n = len(h)

    idx = np.argsort(c)[-k_top:]

            f = f_new

            v *= 0.5

        cmax = c.max()
        gamma = gamma0 + (gamma1 - gamma0) * (i / max(n_iter, 1))
        w = np.exp(-gamma * (cmax - c))
        wsum = w.sum()
        if wsum < 1e-15:
            break
        w = w / wsum
        thresh = 1e-9
        active = np.where(w > thresh)[0]
        if len(active) == 0:
            break
        wa = w[active] * dx
        g = _active_grad(h, active, n, wa)
        v = beta * v + (1.0 - beta) * g
        h_new = _repair(h - lr * v, n)
        f_new = _eval(h_new)
        if f_new < f:
            h, f = h_new, f_new
            lr = min(lr * 1.004, 0.05)
        else:
            lr *= 0.5
            v *= 0.6
            if lr < 1e-15:
                break
    return h

def _sa_structured(h, n, t_end, T0=0.006, T1=0.00002, max_iter=10000):
    rng = np.random.RandomState(2024)
    f = _eval(h)
    best_h, best_f = h.copy(), f
    for i in range(max_iter):
        if time.time() > t_end:
            break
        T = T0 * (T1 / T0) ** (i / max_iter)
        rtype = rng.randint(3)
        h_new = h.copy()
        if rtype == 0:
            length = rng.randint(n // 10, n // 3)
            start = rng.randint(0, n - length)
            x = np.linspace(0, 1, length, endpoint=False)
            f1 = rng.uniform(0.5, 4.0)
            f2 = rng.uniform(0.5, 5.0)
            p1 = rng.uniform(0, 6.28)
            p2 = rng.uniform(0, 6.28)
            amp = rng.uniform(0.02, 0.2) * (min(T / T0, 1.0)) ** 0.2
            pert = amp * (np.sin(2 * np.pi * f1 * x + p1) + 0.5 * np.sin(2 * np.pi * f2 * x + p2))
            h_new[start:start + length] += pert
        elif rtype == 1:
            bs = max(n // 10, 20)
            nb = n // bs
            b1, b2 = int(rng.randint(nb)), int(rng.randint(nb))
            if b1 != b2:
                h_new[b1 * bs:(b1 + 1) * bs] = h[b2 * bs:(b2 + 1) * bs].copy()
                h_new[b2 * bs:(b2 + 1) * bs] = h[b1 * bs:(b1 + 1) * bs].copy()
            buf = min(n // 6, bs)
            x = np.linspace(0, 1, buf, endpoint=False)
            h_new[b1 * bs:b1 * bs + buf] += (min(T / T0, 1.0)) ** 0.3 * 0.08 * np.sin(2 * x + rng.uniform(0, 6.28))
        else:
            center = rng.randint(0, n)
            width = max(rng.randint(n // 20, n // 5), 5)
            amp = rng.uniform(-0.2, 0.2) * (min(T / T0, 1.0)) ** 0.25
            x = np.arange(-width, width + 1) / width
            bump = amp * np.exp(-x * x)
            s, e = max(0, center - width), min(n, center + width + 1)
            h_new[s:e] += bump[s - center:e - 1 - center]
        h_new = _repair(h_new, n)
        f_new = _eval(h_new)
        if f_new < f or rng.random() < np.exp(-(f_new - f) / max(T, 1e-10)):
            h, f = h_new, f_new
            if f < best_f:
                best_f, best_h = f, h.copy()
    return best_h

def _coarse_fine(h, n, ratio, t_end):
    nc = n // ratio
    if nc < 40:
        return h
    hc = np.zeros(nc)
    for i in range(nc):
        hc[i] = h[i * ratio:(i + 1) * ratio].mean()
    hc = _repair(hc, nc)
    hc = _sens_descent(hc, t_end, n_iter=6000, lr0=0.006, beta=0.85, gamma0=40.0, gamma1=120.0)
    t_idx = np.linspace(0, nc - 0.5, n)
    h_up = np.interp(t_idx, np.arange(nc), hc)
    return _repair(np.clip(h_up, 0, 1), n)

def _fine_search(h, n, t_end, max_iters=5000000):
    rng = np.random.RandomState(3141)
    f = _eval(h)
    best_h, best_f = h.copy(), f
    steps = [1e-4, 5e-5, 2.5e-5, 1e-5, 5e-6, 2e-6, 1e-6]
    si = 2
    for i in range(max_iters):
        if time.time() > t_end:
            break
        s = steps[si]
        i1 = rng.randint(n)
        i2 = (i1 + rng.randint(1, 5)) % n
        d = s * rng.randn()
        h[i1] += d
        h[i2] -= d
        if 0.0 <= h[i1] <= 1.0 and 0.0 <= h[i2] <= 1.0:
            fn = _eval(h)
            if fn < f:
                f = fn
                if f < best_f:
                    best_f, best_h = f, h.copy()
                si = max(0, si - 1)
            else:
                h[i1] -= d
                h[i2] += d
                si = min(len(steps) - 1, si + 1)
        else:
            h[i1] -= d
            h[i2] += d
    return best_h, best_f

def _try_update(hcand, bh, bc5, n):
    hcand = _repair(hcand, n)
    c = _eval(hcand)
    if c < bc5:
        return hcand.copy(), c
    return bh, bc5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    tend = t0 + budget_s
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h, best_c5 = h.copy(), _eval(h)

    # Phase 1: Coarse-to-fine multiresolution
    t1 = t0 + budget_s * 0.12
    if time.time() < t1:
        for ratio in [4, 7]:
            if time.time() > t1 - 3:
                break
            best_h, best_c5 = _try_update(_coarse_fine(best_h, n, ratio, min(t1, time.time() + (t1 - time.time()) / 2)), best_h, best_c5, n)

    # Phase 2: Sensitivity-weighted gradient descent (broad)
    t2 = t0 + budget_s * 0.35
    if time.time() < t2:
        best_h, best_c5 = _try_update(_sens_descent(best_h.copy(), t2, n_iter=50000, lr0=0.002, beta=0.88, gamma0=50, gamma1=180), best_h, best_c5, n)

    # Phase 3: Structured SA (exploration)
    t3 = t0 + budget_s * 0.55
    if time.time() < t3:
        best_h, best_c5 = _try_update(_sa_structured(best_h, n, t3, T0=0.005, T1=0.00002, max_iter=9000), best_h, best_c5, n)

    # Phase 4: Second sensitivity descent (sharper)
    t4 = t0 + budget_s * 0.72
    if time.time() < t4:
        best_h, best_c5 = _try_update(_sens_descent(best_h.copy(), t4, n_iter=50000, lr0=0.001, beta=0.90, gamma0=100, gamma1=350), best_h, best_c5, n)

    # Phase 5: Cooled SA (exploitation)
    t5 = t0 + budget_s * 0.82
    if time.time() < t5:
        best_h, best_c5 = _try_update(_sa_structured(best_h, n, t5, T0=0.0015, T1=0.000005, max_iter=5000), best_h, best_c5, n)

    # Phase 6: Ultra-fine local search
    t6 = tend - 2.0
    if time.time() < t6:
        best_h, best_c5 = _fine_search(best_h, n, t6, max_iters=5000000)
        best_h = _repair(best_h, n)
        best_c5 = _eval(best_h)

    final_h = _repair(best_h, n)
    final_c5 = _eval(final_h)
    return (final_h, final_c5, n)
