def _fft_size(n):
    m = 2 * n - 1
    N = 1
    while N < m:

    return N

    s = h.sum()
    if s <= 1e-15:
        return np.full(n, 0.5)
    for _ in range(80):
        h *= t / s
        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
        if abs(s - t) < 1e-12:
            break
    d = t - h.sum()
    if abs(d) > 1e-12:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            if d > 0:
                a = min(d, 1.0 - h[i])
                h[i] += a
                d -= a
            else:
                a = min(-d, h[i])
                h[i] -= a
                d += a
            if abs(d) < 1e-13:

        if abs(d) < 1e-11:

def _active_subgrad(h):
    """Compute subgradient of max C(k) using exact active set."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    mx = corr.max()
    tol = 1e-9
    active = corr >= mx - tol
    # For each active lag, compute contribution to gradient
    # lag k in [0, 2n-2] corresponds to shift d = k - (n-1)
    g = np.zeros(n)
    n_active = active.sum()
    if n_active == 0:
        return g
    for k in np.where(active)[0]:
        d = k - (n - 1)
        if d >= 0:
            # C(k) = dx * sum_{i=0}^{n-1-d} h[i] * (1-h[i+d])
            # dC/dh[i] = (1-h[i+d]) for i in [0, n-1-d]
            #              - h[i-d] for i in [d, n-1]
            upper = n - 1 - d
            for i in range(upper + 1):
                g[i] += (1.0 - h[i + d])
            for i in range(d, n):
                g[i] -= h[i - d]
        else:
            # d < 0, let d' = -d > 0

import numpy as np
import time

def _fft_size(n):
    m = 2 * n - 1
    N = 1
    while N < m:
        N <<= 1
    return N

def _repair(h, n):
    t = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if s <= 1e-15:
        return np.full(n, 0.5)
    for _ in range(80):
        h *= t / s
        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
        if abs(s - t) < 1e-12:
            break
    d = t - h.sum()
    if abs(d) > 1e-12:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)
    x += (S - x.sum()) / x.size
    for _ in range(10):
        np.clip(x, 0.0, 1.0, out=x)
        d = S - x.sum()
        if abs(d) < 1e-11:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
    if abs(x.sum() - S) > 1e-8:
        lo = float(np.min(v)) - 1.0
        hi = float(np.max(v)) + 1.0
        for _ in range(50):
            mid = 0.5 * (lo + hi)
            if np.clip(v - mid, 0.0, 1.0).sum() > S:
                lo = mid
            else:
                hi = mid
        x = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    return x

def _c5_direct(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _pn_arrays(h):
    n = len(h)
    N = _fft_size(n)
    H = np.fft.rfft(h, N)
    ac = np.fft.irfft(H * np.conj(H), N).real[:n]
    pref = np.empty(n + 1)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    total = pref[n]
    P = np.empty(n)
    P[0] = total - ac[0]
    Nv = np.zeros(n)
    if n > 1:
        idx = np.arange(1, n)
        P[idx] = pref[n - idx] - ac[idx]
        Nv[idx] = total - pref[idx] - ac[idx]
    mx = float(P[0])
    if n > 1:
        mx = max(mx, float(np.max(P[1:])), float(np.max(Nv[1:])))
    return P, Nv, mx

def _smooth_value(h, alpha):
    P, Nv, mx = _pn_arrays(h)
    n = len(h)
    dx = 2.0 / n
    e = np.exp(np.clip(alpha * (P[0] - mx), -400.0, 0.0))
    if n > 1:
        e += np.exp(np.clip(alpha * (P[1:] - mx), -400.0, 0.0)).sum()
        e += np.exp(np.clip(alpha * (Nv[1:] - mx), -400.0, 0.0)).sum()
    return dx * (mx + np.log(e) / alpha)

def _grad_from_weights(h, wP, wN):
    n = len(h)
    dx = 2.0 / n
    u = np.zeros(n)
    if n > 1:
        u[1:] = wP[1:] + wN[1:
        cp = np.zeros(n)
        cn = np.zeros(n)
        cp[1:] = np.cumsum(wP[1:])
        cn[1:] = np.cumsum(wN[1:])
    linP = cp[n - 1::-1]
    linN = cn
    q = np.zeros(n)
    if n > 1 and np.any(u[1:]):
        Nf = _fft_size(n)
        H = np.fft.rfft(h, Nf)
        R = np.fft.irfft(H * np.fft.rfft(u[::-1], Nf), Nf)[n-1:2*n-1].real
        L = np.fft.irfft(H * np.fft.rfft(u, Nf), Nf)[:n].real
        q = -(R + L)
    return dx * (wP[0] * (1.0 - 2.0 * h) + linP + linN + q)

def _smooth_grad(h, alpha):
    P, Nv, mx = _pn_arrays(h)
    n = len(h)
    wP = np.zeros(n); wN = np.zeros(n)
    e0 = np.exp(np.clip(alpha * (P[0] - mx), -400.0, 0.0))
    tot = e0
    if n > 1:
        ep = np.exp(np.clip(alpha * (P[1:] - mx), -400.0, 0.0))
        en = np.exp(np.clip(alpha * (Nv[1:] - mx), -400.0, 0.0))
        tot += ep.sum() + en.sum()
        if tot <= 0.0:
            tot = 1.0; e0 = 1.0
        wP[0] = e0 / tot; wP[1:] = ep / tot; wN[1:] = en / tot
    else:
        wP[0] = 1.0
    return _grad_from_weights(h, wP, wN)

            best_c = bc; best = bh.copy()
        if chf < best_c:
            best_c = chf; best = hf.copy()
        hcur = hf if chf <= chb else bh.copy()

    # Phase 2: Gradient descent polish (15% budget)
    end2 = t0 + 0.75 * (budget_s - 5)
    if time.time() < end2 - 2:
        b2, c2 = _grad_descent_polish(best, end2, 5e-6)
        if c2 < best_c:
            best_c = c2; best = b2

    # Phase 3: Noise restarts (10% budget)
    end3 = t0 + 0.85 * (budget_s - 5)
    if time.time() < end3 - 3:
        b3, c3 = _noise_restart(best, rng, end3, [0.02, 0.05, 0.1])
        if c3 < best_c:
            best_c = c3; best = b3

    # Phase 4: Pair search (remaining time)
    end4 = deadline
    if time.time() < end4 - 2:
        b4, c4 = _pair_search(best, rng, end4, 1e-4)
        if c4 < best_c:
            best_c = c4; best = b4

    # Final: try variants and repair
    variants = [best, 1.0 - best, best[::-1],
                0.5 + 0.5 * (best - best[::-1]),
                0.5 + 0.5 * (best + best[::-1] - 1.0)]
    for hh in variants:
        h2 = _proj(hh, n * 0.5)
        c2 = _c5_direct(h2)
        if c2 < best_c:
            best_c = c2; best = h2

    best = _repair(best, n)
    best_c = _c5_direct(best)
    return best, best_c, n
