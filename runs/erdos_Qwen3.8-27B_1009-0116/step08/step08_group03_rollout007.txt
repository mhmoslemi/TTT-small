import numpy as np

def _proj(v, S):
    x = np.asarray(v, dtype=float)
    np.clip(x, 0.0, 1.0, out=x)
    d = S - float(x.sum())
    for _ in range(25):
        if abs(d) < 1e-12: break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = int(m.sum())
        if c == 0: break
        x[m] += d / c
        d = S - float(x.sum())
    if abs(d) > 1e-8:
        lo = float(np.min(x)) - 1.0
        hi = float(np.max(x)) + 1.0
        for _ in range(45):
            mid = 0.5*(lo+hi)
            s = float(np.clip(v - mid, 0.0, 1.0).sum())
            if s > S: lo = mid
            else: hi = mid
        x = np.clip(v - 0.5*(lo+hi), 0.0, 1.0)
    return x

    if dv == 0.0:

        mn = float(corr.max())
        _pair_move(corr, h, u, i, j, -d)
        if mn < best_mx:
            best_mx = mn
            best_d = d
    if best_d > 0.0:
        _pair_move(corr, h, u, i, j, best_d)
    return best_d, best_mx


def _corr_grad(h, u, corr, mx, tol=0.15):
    n = len(h)
    m = n - 1
    sel = np.nonzero(corr >= mx - tol)[0]
    limit = 48
    if sel.size > limit:
        sub = corr[sel]
        top = np.argsort(sub)[-limit:]
        sel = sel[top]
    g = np.zeros(n, dtype=float)
    for idx in sel:
        q = int(idx) - m
        if q >= 0:
            g[:n - q] += u[q:n]
            g[q:n] -= h[:n - q]
        else:
            d = -q
            g[d:n] += u[:n - d]
            g[:n - d] -= h[d:n]
    return g


def _resample(h, m):
    n = len(h)
    m = int(m)
    if m <= 0:
        return _proj(h, 0.0)
    if n == m:
        return _proj(h, 0.5 * m)
    if n % m == 0:
        out = np.asarray(h, dtype=float).reshape(m, n // m).mean(axis=1)
    else:
        x0 = (np.arange(n) + 0.5) * (2.0 / n)
        x1 = (np.arange(m) + 0.5) * (2.0 / m)
        out = np.interp(x1, x0, np.asarray(h, dtype=float))
    return _proj(out, 0.5 * m)


def _anti_pal(h):
    n = len(h)
    return _proj(0.5 + 0.5 * (h - h[::-1]), 0.5 * n)


def _pal_dev(h):
    n = len(h)
    return _proj(0.5 + 0.5 * (h + h[::-1] - 1.0), 0.5 * n)


def _to_binary(h):
    n = len(h)
    k = n // 2
    b = np.zeros(n, dtype=float)
    if k > 0:
        idx = np.argsort(-h)[:k]
        b[idx] = 1.0
    return b


def _local_exact(h0, rng, deadline, base0=0.12, full_prob=0.05, anneal=1.0):
    t0 = time.time()
    deadline = float(deadline)
    h_in = np.asarray(h0, dtype=float)
    n = len(h_in)
    if n < 2 or deadline <= t0:
        h = _finalize(h_in)
        return h, _c5_direct(h)
    S = 0.5 * n
    dx = 2.0 / n
    h = _proj(h_in, S)
    u = 1.0 - h
    corr = _corr_from(h, u)
    mx = float(corr.max())
    best_h = h.copy()
    best_mx = mx
    g = np.zeros(n)
    order = np.argsort(g)
    q = max(8, n // 6)
    low = order[:q]
    high = order[-q:]
    it = 0
    stall = 0
    T0 = max(1e-7, anneal * dx * 0.08)

    while time.time() < deadline:
        if it == 0 or it % 96 == 0:
            u = 1.0 - h
            if it % 1024 == 0:
                if abs(float(h.sum()) - S) > 1e-7:
                    h = _proj(h, S)
                    u = 1.0 - h
                corr = _corr_from(h, u)
            mx = float(corr.max())
            if mx + 1e-12 < best_mx:
                best_mx = mx
                best_h = h.copy()
            g = _corr_grad(h, u, corr, mx, 0.15)
            order = np.argsort(g)
            q = max(8, n // 6)
            low = order[:q]
            high = order[-q:]

            if it > 1024 and it % 2048 == 0:
                amax = float(np.max(np.abs(g)))
                if amax > 1e-12:
                    hsave = h.copy()
                    corrsave = corr.copy()
                    mxsave = mx
                    h2 = _proj(h - 0.05 * (g / amax), S)
                    corr2 = _corr_from(h2, 1.0 - h2)
                    mx2 = float(corr2.max())
                    if mx2 + 1e-12 < mxsave:
                        h = h2
                        u = 1.0 - h
                        corr = corr2
                        mx = mx2
                        stall = 0
                        if mx + 1e-12 < best_mx:
                            best_mx = mx
                            best_h = h.copy()
                    else:
                        h = hsave
                        corr = corrsave
                        mx = mxsave
                        u = 1.0 - h

        tf = (time.time() - t0) / max(deadline - t0, 1e-9)
        if tf >= 1.0:
            break

        if stall > 2500 and stall % 500 == 0:
            rls = rng.random()
            i = int(low[int(rng.randint(len(low)))]) if rls < 0.8 else int(rng.randint(n))
            j = int(high[int(rng.randint(len(high)))]) if rls < 0.8 else int(rng.randint(n))
            if i != j:
                if g[i] > g[j]:
                    i, j = j, i
                bd, bmx = _line_search_pair(corr, h, u, i, j, mx, 0.15)
                if bd > 0.0 and bmx < mx:
                    mx = bmx
                    stall = 0
                    if mx + 1e-12 < best_mx:
                        best_mx = mx
                        best_h = h.copy()
                    it += 1
                    continue

        r = rng.random()
        if r < 0.72:
            i = int(low[int(rng.randint(len(low)))])
            j = int(high[int(rng.randint(len(high)))])
        elif r < 0.90:
            a = max(1, n // 4)
            b = max(a + 1, 3 * n // 4)
            i = int(order[int(rng.randint(a))])
            j = int(order[int(rng.randint(b, n))])
        else:
            i = int(rng.randint(n))
            j = int(rng.randint(n))

        if i == j:
            it += 1
            continue

        if g[i] > g[j]:
            i, j = j, i

        avail = min(1.0 - float(h[i]), float(h[j]))
        if avail <= 1e-14:
            it += 1
            continue

        if rng.random() < full_prob:
            d = avail * (0.5 + 0.5 * rng.random())
        else:
            base_cur = base0 * (0.25 + 0.75 * (1.0 - tf))
            d = base_cur * (0.1 + 1.9 * rng.random())
            if d > avail:
                d = avail

        _pair_move(corr, h, u, i, j, d)
        mxn = float(corr.max())
        dt = mxn - mx
        accept = dt < -1e-12

        if not accept and dt > 0.0 and anneal > 0.0:
            T = T0 * (1.0 - tf) ** 2 + 1e-10
            dc = dt * dx
            if dc < 4.0 * T and rng.random() < math.exp(-dc / T):
                accept = True

        if accept:
            mx = mxn
            stall = 0
            if mx + 1e-12 < best_mx:
                best_mx = mx
                best_h = h.copy()
        else:
            _pair_move(corr, h, u, i, j, -d)
            stall += 1

        if stall > 6000:
            h = best_h.copy()
            nscale = 0.05 if rng.random() < 0.7 else 0.25
            h = _proj(h + nscale * rng.randn(n), S)
            u = 1.0 - h
            corr = _corr_from(h, u)
            mx = float(corr.max())
            g = _corr_grad(h, u, corr, mx, 0.15)
            order = np.argsort(g)
            low = order[:q]
            high = order[-q:]
            stall = 0

        it += 1

    h_best = _finalize(best_h)
    c_best = _c5_direct(h_best)
    h_cur = _finalize(h)
    c_cur = _c5_direct(h_cur)
    if c_cur < c_best:
        return h_cur, c_cur
    return h_best, c_best


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s)
    rng = np.random.RandomState(seed)

    if "initial_h_values" in globals():
        h_init = np.array(initial_h_values, dtype=float, copy=True)
    else:
        h_init = np.full(560, 0.5)

    n_init = len(h_init)
    h_init = _proj(h_init, 0.5 * n_init)

    best = h_init.copy()
    best_c = _c5_direct(best)

    for v in (_anti_pal(h_init), _pal_dev(h_init), _to_binary(h_init)):
        vv = _finalize(v)
        c = _c5_direct(vv)
        if c < best_c:
            best = vv
            best_c = c

    if deadline - time.time() < 30.0:
        for v in (best, _anti_pal(best), _pal_dev(best), _to_binary(best)):
            vv = _finalize(v)
            c = _c5_direct(vv)
            if c < best_c:
                best = vv
                best_c = c
        best = _finalize(best)
        return best, _c5_direct(best), len(best)

    cand = [_anti_pal(best), _pal_dev(best), _to_binary(best)]
    quick_end = t0 + min(180.0, 0.2 * budget_s)
    quick_end = min(quick_end, deadline - 60.0)
    per = max(0.0, (quick_end - t0) / max(1, len(cand)))
    if per < 15.0 and len(cand) > 1:
        keep = max(1, int((quick_end - t0) / 20.0))
        cand = cand[:keep]

    for v in cand:
        if time.time() > quick_end - 5.0:
            break
        end = min(quick_end - 5.0, time.time() + max(5.0, per))
        hh, cc = _local_exact(v, rng, end, base0=0.12, anneal=1.5)
        if cc < best_c:
            best = hh
            best_c = cc

    main_end = t0 + min(450.0, 0.45 * budget_s)
    main_end = min(main_end, deadline - max(60.0, 0.15 * budget_s))
    if main_end > time.time() + 20.0:
        hh, cc = _local_exact(best, rng, main_end, base0=0.12, anneal=2.0)
        if cc < best_c:
            best = hh
            best_c = cc

    multi_end = deadline - max(45.0, 0.1 * budget_s)
    if multi_end > time.time() + 30.0:
        ns = []
        for bn in (len(best), n_init):
            for f in (1, 2, 4):
                m = bn // f
                if m < 32:
                    break
                if m % 2:
                    m -= 1
                if m >= 32 and m <= 1024:
                    ns.append(m)
        ns = sorted(set(ns))
        calls = []
        for m in ns:
            sb = _resample(best, m)
            calls.append(sb)
            if m != len(best):
                calls.append(_to_binary(sb))

        max_calls = max(1, int((multi_end - time.time()) / 45.0))
        calls = calls[:max_calls]
        rem = multi_end - time.time()
        per = rem / max(1, len(calls))
        for v in calls:
            if time.time() > multi_end - 20.0:
                break
            slot = min(max(10.0, per), multi_end - 20.0 - time.time())
            if slot < 10.0:
                break
            hh, cc = _local_exact(v, rng, time.time() + slot, base0=0.14, anneal=3.0)
            if cc < best_c:
                best = hh
                best_c = cc

    if deadline - time.time() > 15.0:
        hh, cc = _local_exact(best, rng, deadline - 2.0, base0=0.05, full_prob=0.03, anneal=0.5)
        if cc < best_c:
            best = hh
            best_c = cc

    for v in (best, _anti_pal(best), _pal_dev(best), _to_binary(best), best[::-1]):
        vv = _finalize(v)
        c = _c5_direct(vv)
        if c < best_c:
            best = vv
            best_c = c

    best = _finalize(best)
    best_c = _c5_direct(best)
    return best, best_c, len(best)
