                k = (i + rng.randint(-8, 8) + 100) % n
                d1 = rng.randn() * 1.5e-5
                ni, nj, nk = h[i] + d1, h[j] - d1, h[k]
                if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= nk <= 1:
                    h[i], h[j] = ni, nj
                    fn = _eval(h)
                    if fn < f - 1e-15:
                        f = fn
                        if fn < best_f:
                            best_f, best_h = fn, h.copy()
                    else:
                        h[i], h[j] = ni - d1, nj + d1
                    continue
            if j == i:
                continue
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                fn = _eval(h)
                if fn < f - 1e-15:
                    f = fn
                    if fn < best_f:
                        best_f, best_h = fn, h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d
    return best_h, best_f

def _local_search_coarse(h, n, t_end):
    f = _eval(h)
    best_h, best_f = h.copy(), f
    rng = np.random.RandomState(67890)
    while time.time() < t_end:
        for _ in range(300):
            if time.time() > t_end:
                break
            i = rng.randint(n)
            j = (i + rng.randint(1, 12)) % n
            d = rng.randn() * 1e-3
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                fn = _eval(h)
                if fn < f - 1e-14:
                    f = fn
                    if fn < best_f:
                        best_f, best_h = fn, h.copy()
                else:
                    h[i], h[j] = h[i] - d, h[j] + d
    return best_h, best_f

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    t_end_global = t0 + budget_s
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    target = n / 2.0
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    def _update(hc):
        nonlocal best_h, best_c5
        c = _eval(hc)
        if c < best_c5:
            best_c5, best_h = c, hc.copy()
        return c

    # Phase 1: Quick SLSQP to establish baseline (10%)
    t1 = t0 + budget_s * 0.10
    h = _slsqp_refine(best_h.copy(), n, 2000, 500, t1)
    _update(h)

    # Phase 2: Multi-start with perturbations (35%)
    t2 = t0 + budget_s * 0.45
    candidates = []
    for trial in range(5):
        if time.time() > t2 - 30:
            break
        if trial == 0:
            hc = best_h.copy()
        elif trial == 1:
            hc = _fourier_perturb(best_h, n, rng)
        elif trial == 2:
            hc = _block_perturb(best_h, n, rng)
        elif trial == 3:
            noise = rng.randn(n) * 0.015
            hc = _repair(best_h + noise, n)
        else:
            hc = _fourier_perturb(best_h, n, rng)
            hc = _fourier_perturb(hc, n, rng)
        # Nesterov optimization for this candidate
        sub_end = min(t2, t0 + budget_s * (0.15 + trial * 0.06))
        hc, fc = _nesterov_weighted(hc, n, 20, 0.5, 2000, 0.003, sub_end)
        h2 = _slsqp_refine(hc, n, 50000, 300, min(t2, time.time() + 60))
        _update(h2)
        candidates.append(h2)

    # Phase 3: Refine best with higher-alpha SLSQP (15%)
    t3 = t0 + budget_s * 0.60
    for alpha in [200000, 2000000]:
        if time.time() > t3 - 5:
            break
        h = _slsqp_refine(best_h.copy(), n, alpha, 800, t3)
        _update(h)

    # Phase 4: Nesterov with increasing k (15%)
    t4 = t0 + budget_s * 0.75
    for k, stp in [(15, 0.002), (50, 0.001), (100, 0.0005)]:
        sub_end = min(t4, time.time() + budget_s * 0.04)
        h, f = _nesterov_weighted(best_h.copy(), n, k, 0.8, 3000, stp, sub_end)
        _update(h)

    # Phase 5: Coarse local search (10%)
    t5 = t0 + budget_s * 0.85
    if time.time() < t5:
        best_h, best_c5 = _local_search_coarse(best_h.copy(), n, t5)
        _update(best_h)

    # Phase 6: Fine local search (10%)
    t6 = t0 + budget_s * 0.97
    if time.time() < t6:
        best_h, best_c5 = _local_search_fine(best_h.copy(), n, t6)
        _update(best_h)

    # Final: one more SLSQP polish + repair
    if time.time() < t_end_global - 5:
        h = _slsqp_refine(best_h.copy(), n, 50000000, 300, t_end_global - 3)
        _update(h)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
