                    diff -= a
                else:

                    diff += a

                 'jac': lambda x: np.ones(n)}]
        try:
            r = minimize(lambda x: _smax_obj(x, alpha, tk), h,
                         jac=lambda x: _grad_smax(x, alpha, tk),
                         method='SLSQP', bounds=[(0.0, 1.0)] * n,
                         constraints=cons, options={'maxiter': mi, 'ftol': 1e-16})
            h = _repair(r.x, n)
            if _eval(h) < best_c:
                best_c = _eval(h)
                best = h.copy()
        except Exception:
            pass
    return best, best_c

def _fourier_perturb(h, rng, n_trial=3):
    n = len(h)
    best_h = h.copy()
    best_c = _eval(h)
    for _ in range(n_trial):
        hp = h.copy()
        freq = rng.randint(2, max(3, n // 10))
        amp = 0.02 * rng.uniform(0.5, 1.5)
        phase = rng.uniform(0, 2 * np.pi)
        x = np.arange(n)
        hp += amp * np.cos(2 * np.pi * freq * x / n + phase)
        hp = _proj_bisect(hp, n * 0.5)
        c = _eval(hp)
        if c < best_c:
            best_c = c
            best_h = hp.copy()
    return best_h, best_c

def _struct_anneal(h, niter, rng, deadline, T0, T1):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        r = rng.random()
        if r < 0.3:
            bs = rng.choice([1, 2, 3, 5, 8, 13, 21])
            i0 = rng.randint(0, max(1, n - bs))
            seg = h[i0:i0+bs].copy()
            d = 0.003 * max(T, 1e-9)**0.5 * rng.randn(bs)
            d -= d.mean()
            seg = np.clip(seg + d, 0, 1)
            hn = h.copy()
            hn[i0:i0+bs] = seg
        elif r < 0.6:
            freq = rng.randint(1, max(2, n // 20))
            amp = 0.01 * max(T, 1e-9)**0.5
            x = np.arange(n)
            d = amp * np.cos(2*np.pi*freq*x/n + rng.uniform(0, 2*np.pi)) * rng.randn()
            d -= d.mean()
            hn = np.clip(h + d, 0, 1)
        else:
            i = rng.randint(n)
            j = (i + rng.randint(1, 20)) % n
            d = 0.005 * max(T, 1e-9)**0.5 * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                hn = h.copy()
                hn[i], hn[j] = ni, nj
            else:
                continue
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h = hn
            c5c = c5n
            if c5n < bc:
                bc = c5n
                bh = h.copy()
    return bh, bc

def _pair_opt(h, deadline):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    improvements = 0
    for sweep in range(500):
        if time.time() > deadline:
            break
        improved_this = False
        for i in range(n):
            if time.time() > deadline:
                break
            for off in [1, 2, 3, 5]:
                j = (i + off) % n
                if j == i:
                    continue
                for d in [2e-6, -2e-6, 1e-6, -1e-6]:
                    ni, nj = h[i] + d, h[j] - d
                    if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                        h[i], h[j] = ni, nj
                        c5n = _eval(h)
                        if c5n < c5c - 1e-20:
                            c5c = c5n
                            improved_this = True
                            if c5n < bc:
                                bc = c5n
                                bh = h.copy()
                            break
                        h[i], h[j] = ni - d, nj + d
        if not improved_this:
            break
    return bh, bc

def _finestep(h, niter, rng, deadline):
    nc = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    step_base = 5e-7
    for _ in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(nc)
        j = (i + 1 + rng.randint(0, 6)) % nc
        d = step_base * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
                if c5n < bc:
                    bc = c5n
                    bh = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
                step_base = max(step_base * 0.97, 1e-9)
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 8

    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: SLSQP (0-100s)
    dl1 = min(deadline, t0 + 100)
    h, c5v = _slsqp_runs(h, dl1)
    if c5v < best_c5:
        best_c5 = c5v
        best_h = h.copy()

    # Phase 2: Cyclical Nesterov (100-380s)
    dl2 = min(deadline, t0 + 380)
    alpha_bots = [8000, 40000, 200000, 1000000]
    h2 = best_h.copy()
    for a0, gr, tk, st, ns in [(8000, 1.8, 30, 0.006, 4000),
                                (50000, 1.5, 15, 0.003, 3000),
                                (300000, 1.3, 8, 0.0015, 2500)]:
        if time.time() > dl2:
            break
        h2 = _nesterov_cyc(h2, [a0], gr, tk, ns, st, dl2)
        c5v = _eval(h2)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h2.copy()

    # Phase 3: Fourier perturb + Nesterov (380-520s)
    dl3 = min(deadline, t0 + 520)
    h3, c5_3 = _fourier_perturb(best_h, rng, n_trial=8)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()
    h3b = _nesterov_cyc(best_h.copy(), [200000], 1.4, 12, 2000, 0.002, dl3)
    c5v = _eval(h3b)
    if c5v < best_c5:
        best_c5 = c5v
        best_h = h3b.copy()

    # Phase 4: Structured annealing (520-750s)
    dl4 = min(deadline, t0 + 750)
    h4, c5_4 = _struct_anneal(best_h.copy(), 80000, rng, dl4, 0.006, 1e-11)
    if c5_4 < best_c5:
        best_c5 = c5_4
        best_h = h4.copy()

    # Phase 5: Pair optimization (750-920s)
    dl5 = min(deadline, t0 + 920)
    h5, c5_5 = _pair_opt(best_h.copy(), dl5)
    if c5_5 < best_c5:
        best_c5 = c5_5
        best_h = h5.copy()

    # Phase 6: Final Nesterov + finestep (920-end)
    dl6 = deadline
    h6 = _nesterov_cyc(best_h.copy(), [5000000], 1.1, 5, 2000, 0.0005, dl6)
    c5v = _eval(h6)
    if c5v < best_c5:
        best_c5 = c5v
        best_h = h6.copy()
    rem = max(0, dl6 - time.time())
    n_pol = min(int(rem / 0.002), 50000)
    best_h, best_c5 = _finestep(best_h.copy(), n_pol, rng, dl6)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
