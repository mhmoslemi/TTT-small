    best_c = cur_c
    best_h = h.copy()
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 12)) % n
        d = step_hover * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c_new = _eval(h)
            accept = c_new < cur_c
            if not accept and sa_temp is not None and sa_temp > 1e-12:
                if rng.rand() < np.exp(-(c_new - cur_c) / sa_temp):
                    accept = True
            if accept:
                cur_c = c_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def _sweep_coord(h, n_passes, rng):
    n = len(h)
    cur_c = _eval(h)
    best_c = cur_c
    best_h = h.copy()
    for p in range(n_passes):
        order = rng.permutation(n)
        for i in order:
            j = (i + 1 + rng.randint(0, 8)) % n
            for scale in [5e-5, 2e-5, 5e-6, 1e-6]:
                d = scale * rng.randn()
                ni, nj = h[i] + d, h[j] - d
                if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                    h[i], h[j] = ni, nj
                    c_new = _eval(h)
                    if c_new < cur_c:
                        cur_c = c_new
                        if c_new < best_c:
                            best_c = c_new
                            best_h = h.copy()
                    else:
                        h[i], h[j] = ni - d, nj + d
                        break
    return best_h, best_c

def _downsample(h, f):
    n = len(h)
    m = n // f
    return h[:m*f].reshape(m, f).mean(axis=1).copy()

def _upsample(h, f):
    n = len(h)
    out = np.repeat(h, f)
    return out

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    r = np.random.RandomState(seed)
    def _el():
        return time.time() - t0

    h_parent = _repair(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values))
    n = len(h_parent)
    best_h = h_parent.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Fixed MGD (correct gradient direction!)
    mgd_configs = [
        (200, 50000, 2000, 8, 0.005),
        (1000, 100000, 2000, 6, 0.003),
        (5000, 200000, 1500, 5, 0.002),
        (20000, 500000, 1000, 4, 0.001),
    ]
    for a0, a1, steps, tk, s0 in mgd_configs:
        if _el() > 250:
            break
        h_t = best_h.copy()
        h_t, c = _mgd_fixed(h_t, steps, a0, a1, tk, s0, r)
        c = _eval(h_t)
        if c < best_c5:
            best_c5 = c
            best_h = h_t.copy()

    # Phase 2: Multi-start with MGD
    if _el() < budget_s - 400:
        for trial in range(5):
            if _el() > budget_s - 350:
                break
            scale = 0.015 * (trial + 1)
            noise = scale * r.randn(n)
            noise -= noise.mean()
            h_pert = _repair(best_h + noise, n)
            c_pert = _eval(h_pert)
            if c_pert < best_c5:
                best_c5 = c_pert
                best_h = h_pert.copy()
            h_t, c = _mgd_fixed(h_pert, 800, 500, 200000, 6, 0.003, r)
            c = _eval(h_t)
            if c < best_c5:
                best_c5 = c
                best_h = h_t.copy()

    # Phase 3: SLSQP with increasing alpha
    if _el() < budget_s - 350:
        slsqp_configs = [(500, 20, 800), (2000, 12, 700), (10000, 8, 600),
                         (50000, 6, 500), (200000, 5, 400), (1000000, 4, 300)]
        for alpha, tk, mxi in slsqp_configs:
            if _el() > budget_s - 300:
                break
            h_t = _slsqp_lse(best_h.copy(), alpha, mxi, tk)
            c = _eval(h_t)
            if c < best_c5:
                best_c5 = c
                best_h = h_t.copy()

    # Phase 4: Multi-resolution
    if _el() < budget_s - 350:
        for ratio in [2, 4]:
            if _el() > budget_s - 300:
                break
            nc = n // ratio
            hc = _repair(_downsample(best_h, ratio), nc)
            hc, _ = _mgd_fixed(hc, 300, 100, 50000, 5, 0.005, r)
            h_up = _repair(_upsample(hc, ratio), n)
            c_up = _eval(h_up)
            if c_up < best_c5:
                best_c5 = c_up
                best_h = h_up.copy()
            best_h, best_c5 = _local2(best_h, 3000, 1e-4, r)

    # Phase 5: SLSQP after multi-res
    if _el() < budget_s - 250:
        for alpha, tk, mxi in [(2000, 10, 600), (20000, 7, 500),
                              (200000, 5, 400), (2000000, 4, 300)]:
            if _el() > budget_s - 200:
                break
            h_t = _slsqp_lse(best_h.copy(), alpha, mxi, tk)
            c = _eval(h_t)
            if c < best_c5:
                best_c5 = c
                best_h = h_t.copy()

    # Phase 6: Simulated annealing
    if _el() < budget_s - 150:
        sa_temp = max(1e-6, (best_c5 - 0.3808) * 0.01)
        nsa = min(int((budget_s - 150 - _el()) * 2000), 120000)
        best_h, best_c5 = _local2(best_h, nsa, 4e-5, r, sa_temp)

    # Phase 7: Coordinate sweep
    if _el() < budget_s - 80:
        npass = min(int((budget_s - 80 - _el()) * 20), 1500)
        best_h, best_c5 = _sweep_coord(best_h, npass, r)

    # Phase 8: Final ultra-fine polish
    if _el() < budget_s - 10:
        nfin = min(int((budget_s - 10 - _el()) * 4000), 200000)
        best_h, best_c5 = _local2(best_h, nfin, 3e-6, r)

    # Phase 9: Final MGD + local
    if _el() < budget_s - 5:
        h_t, c = _mgd_fixed(best_h, 200, 1000000, 5000000, 4, 0.0003, r)
        c = _eval(h_t)
        if c < best_c5:
            best_c5 = c
            best_h = h_t.copy()
        nfin2 = min(int(max(0, budget_s - 5 - _el()) * 5000), 100000)
        best_h, best_c5 = _local2(best_h, nfin2, 1e-6, r)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
