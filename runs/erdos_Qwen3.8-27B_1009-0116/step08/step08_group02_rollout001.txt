import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(300):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            a = min(abs(diff), 1.0 - h[idx]) if diff > 0 else min(-diff, h[idx])
            h[idx] += a if diff > 0 else -a
            diff += -a if diff > 0 else a
            if abs(diff) < 1e-14:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_worst(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return float(c.max()), int(np.argmax(c))

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_lag(k, h, n):
    dx = 2.0 / n
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        i1 = np.arange(n - sh)
        g[i1] += (1.0 - h[i1 + sh])
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh]
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp])
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp]
    return g * dx

def _slsqp_run(h, alpha, maxiter, top_k):
    n = len(h)
    def obj_fn(x):
        c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
        idx = np.argsort(c)[-top_k:] if 0 < top_k < len(c) else np.arange(len(c))
        cs = c[idx]
        m = cs.max()
        s = np.clip(alpha * (cs - m), -400, 400)
        s -= s.max()
        return m + np.log(np.exp(s).sum()) / alpha
    def grad_fn(x):
        c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
        idx = np.argsort(c)[-top_k:] if 0 < top_k < len(c) else np.arange(len(c))
        cs = c[idx]
        m = cs.max()
        s = np.clip(alpha * (cs - m), -400, 400)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        g = np.zeros(n)
        for ii in range(len(idx)):
            if w[ii] > 1e-15:
                g += w[ii] * _grad_lag(idx[ii], x, n)
        return g
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj_fn, h, jac=grad_fn, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return h

def _nesterov_pg(h, alpha, steps, step0, top_k, momentum):
    n = len(h)
    y = h.copy()
    v = np.zeros(n)
    beta = momentum
    cur_c = _eval(h)
    for it in range(steps):
        c = np.correlate(y, 1.0 - y, mode='full') * (2.0 / n)
        active = np.argsort(c)[-top_k:]
        c_a = c[active]
        m_a = c_a.max()
        s = np.clip(alpha * (c_a - m_a), -400, 400)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        g = np.zeros(n)
        for ii in range(len(active)):
            if w[ii] > 1e-15:
                g += w[ii] * _grad_lag(active[ii], y, n)
        beta = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * beta * beta))
        v_new = (y - h) * beta + v
        h = y.copy()
        y = _proj(y - step0 * v_new, n * 0.5)
        v = v_new
        c_new = _eval(y)
        if c_new < cur_c:
            cur_c = c_new
        if it % 500 == 0:
            v *= 0.5
    return _repair(h.copy(), n), _eval(h)

def _lag_focused_local(h, n_iter, rng, base_step):
    n = len(h)
    best_h = h.copy()
    best_c = _eval(h)
    c5c = best_c
    k_w = 0

            sh = (n - 1) - k_w

            r = rng.rand()

            else:

                if j >= n:

        else:
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 6)) % n
        d = base_step * abs(rng.randn())
        if r < 0.5 or sh <= 0:
            if h[i] > h[j]:
                d = min(d, h[i] - h[j] + 0.1)
                h[i] -= d; h[j] += d
            else:
                d = min(d, h[j] - h[i] + 0.1)
                h[j] -= d; h[i] += d
        else:
            d = base_step * rng.randn()
            h[i] += d; h[j] -= d
        if h[i] < 0 or h[i] > 1 or h[j] < 0 or h[j] > 1:
            if h[i] < 0: h[j] -= h[i]; h[i] = 0
            elif h[i] > 1: h[j] += h[i] - 1; h[i] = 1
            if h[j] < 0: h[i] -= h[j]; h[j] = 0
            elif h[j] > 1: h[i] += h[j] - 1; h[j] = 1
        h = np.clip(h, 0, 1)
        c5n = _eval(h)
        if c5n < c5c:
            c5c = c5n
            if c5n < best_c:
                best_c = c5n

        else:
            h = _repair(best_h if it % 2000 == 0 else h, n)

            c5c = _eval(h)

                best_h = h.copy()
    return _repair(best_h, n), best_c

def _block_local(h, n_iter, rng, base_step, bsz):
    n = len(h)
    best_h = h.copy()
    best_c = _eval(h)
    cur_c = best_c
    for it in range(n_iter):
        i = rng.randint(max(1, n - bsz))
        j = (i + bsz + rng.randint(0, n // 4)) % max(1, n - bsz)
        if j + bsz > n:
            j = rng.randint(max(1, n - bsz))
        d = base_step * rng.randn()
        block_i = h[i:i+bsz].copy()
        block_j = h[j:j+bsz].copy()
        h[i:i+bsz] = block_i + d
        h[j:j+bsz] = block_j - d
        if np.any(h < -0.5) or np.any(h > 1.5):
            h[i:i+bsz] = block_i
            h[j:j+bsz] = block_j
            continue
        h = np.clip(h, 0, 1)
        h = _repair(h, n)
        c5n = _eval(h)
        if c5n < cur_c:
            cur_c = c5n
            if c5n < best_c:
                best_c = c5n
                best_h = h.copy()
        else:
            h = h.copy()
            cur_c = best_c
            h = best_h.copy()
    return best_h, best_c

def _fine_2pt(h, n_iter, rng, base_step):
    n = len(h)
    best_h = h.copy()
    best_c = _eval(h)
    cur_c = best_c
    for it in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 3)) % n
        d = base_step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < cur_c:
                cur_c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c
