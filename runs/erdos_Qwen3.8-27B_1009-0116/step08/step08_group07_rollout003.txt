            if f < best_f:
                best_f, best_h = f, h.copy()
    return _repair(best_h, n), best_f

def _slsqp(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    def obj(x):
        c = _correl(x)
        m = c.max()
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
        s = np.clip(alpha * (ct - m), -400, 400)
        return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha
    def jac(x):
        c = _correl(x)
        m = c.max()
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
        s = np.clip(alpha * (ct - m), -400, 400)
        w = np.exp(s - s.max()); w /= w.sum()
        dx = 2.0 / n
        g = np.zeros(n)
        for ii in range(len(idx)):
            if w[ii] < 1e-13:
                continue
            k = idx[ii]; sh = (n-1) - k; wk = w[ii] * dx
            if 0 <= sh < n:
                g[:n-sh] += wk * (1.0 - h[sh:])
                if sh > 0:
                    g[sh:] -= wk * h[:-sh]
            else:
                sp = -sh
                g[sp:] += wk * (1.0 - h[:-sp])
                g[:n-sp] -= wk * h[sp:]
        # fix: use x not h
        g2 = np.zeros(n)
        for ii in range(len(idx)):
            if w[ii] < 1e-13:
                continue
            k = idx[ii]; sh = (n-1) - k; wk = w[ii] * dx
            if 0 <= sh < n:
                g2[:n-sh] += wk * (1.0 - x[sh:])
                if sh > 0:
                    g2[sh:] -= wk * x[:-sh]
            else:
                sp = -sh
                g2[sp:] += wk * (1.0 - x[:-sp])
                g2[:n-sp] -= wk * x[sp:]
        return g2
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(obj, h, jac=jac, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-16})
        return _repair(r.x, n)
    except Exception:
        return h

def _paired_cd(h, n, max_iter, deadline, topk=8):
    """Paired coordinate descent targeting worst shifts."""
    c5 = _eval(h)
    best_h, best_c5 = h.copy(), c5
    tol_improve = 1e-12
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if it % 50 == 0:
            g, m = _grad_exact(h, 0.0005)
        else:
            g, m = _grad_exact(h, 0.002)
        # Find pairs that reduce the max
        i = np.argmax(g)
        j = np.argmin(g)
        if g[i] - g[j] < 1e-14:
            break
        # Try several step sizes
        improved = False
        for frac in [0.5, 0.3, 0.15, 0.07, 0.03]:
            d = frac * min(g[i], 1.0 - h[j]) * 0.1
            if d < 1e-10:
                continue
            hn = h.copy()
            hn[i] = min(1.0, h[i] + d)
            hn[j] = max(0.0, h[j] - d)
            hn = _repair(hn, n)
            fn = _eval(hn)
            if fn < c5 - tol_improve:
                h, c5 = hn, fn
                if f < best_c5:
                    pass
                if c5 < best_c5:
                    best_c5, best_h = c5, h.copy()
                improved = True
                break
        if not improved and it > 500:
            break
    return best_h, best_c5

def _anneal(h, niter, rng, deadline, blocks, T0, T1):
    n = len(h)
    c5c = _eval(h); bh = h.copy(); bc = c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0+bs].copy()
        d = 0.004 * max(T, 1e-9) ** 0.5 * rng.randn(bs)
        d -= d.mean(); seg += d; seg = np.clip(seg, 0, 1)
        hn = h.copy(); hn[i0:i0+bs] = seg; hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h, c5c = hn, c5n
            if c5n < bc:
                bc, bh = c5n, h.copy()
    return bh, bc

def _polish(h, niter, rng, deadline, scale):
    nc = len(h); c5c = _eval(h); bh = h.copy(); bc = c5c
    for _ in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(nc)
        j = (i + 1 + rng.randint(0, 7)) % nc
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < bc:
                    bc, bh = c5n, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc

def _jitter_reopt(h, rng, step0, tol, deadline):
    n = len(h)
    c5c = _eval(h)
    d = 0.002 * rng.randn(n)
    d -= d.mean()
    hn = _repair(np.clip(h + d, 0, 1), n)
    hn2, c5n = _nesterov(hn, 400, step0, deadline, tol=tol, beta=0.9)
    if c5n < c5c:
        return hn2, c5n
    return h, c5c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: SLSQP soft-max refinement
    dl1 = min(deadline, t0 + 150)
    for alpha, tk, mi in [(10000, 25, 300), (50000, 10, 250), (200000, 5, 200)]:
        if time.time() > dl1:
            break
        h2 = _slsqp(best_h.copy(), alpha, tk, mi, dl1)
        c5v = _eval(h2)
        if c5v < best_c5:
            best_c5, best_h = c5v, h2.copy()

    # Phase 2: Nesterov exact subgradient (multiple configs)
    dl2 = min(deadline, t0 + 380)
    for step0, tol, mom in [(0.003, 0.003, 0.85), (0.001, 0.001, 0.90), (0.0005, 0.0005, 0.93)]:
        if time.time() > dl2:
            break
        h2, c5v = _nesterov(best_h.copy(), 3000, step0, dl2, tol=tol, beta=mom)
        if c5v < best_c5:
            best_c5, best_h = c5v, h2.copy()

    # Phase 3: Worst-shift targeting (focus on single worst shift)
    dl3 = min(deadline, t0 + 520)
    for step0 in [0.004, 0.001, 0.0004]:
        if time.time() > dl3:
            break
        h2, c5v = _nesterov(best_h.copy(), 2000, step0, dl3, worst_only=True, beta=0.92)
        if c5v < best_c5:
            best_c5, best_h = c5v, h2.copy()

    # Phase 4: Paired coordinate descent
    dl4 = min(deadline, t0 + 650)
    h2, c5v = _paired_cd(best_h.copy(), n, 30000, dl4)
    if c5v < best_c5:
        best_c5, best_h = c5v, h2.copy()

    # Phase 5: Jitter + re-optimize
    dl5 = min(deadline, t0 + 780)
    for scale in [0.003, 0.001, 0.0005]:
        if time.time() > dl5:
            break
        hj, cj = _jitter_reopt(best_h.copy(), rng, scale, 0.002, dl5)
        if cj < best_c5:
            best_c5, best_h = cj, hj.copy()

    # Phase 6: Annealing
    dl6 = min(deadline, t0 + 920)
    blocks = [1, 2, 3, 4, 6, 8, 12]
    best_h, best_c5 = _anneal(best_h.copy(), 50000, rng, dl6, blocks, 0.005, 1e-11)

    # Phase 7: Final Nesterov (ultra-fine)
    dl7 = min(deadline, t0 + budget_s - 50)
    h2, c5v = _nesterov(best_h.copy(), 4000, 0.0003, dl7, tol=0.0003, beta=0.94)
    if c5v < best_c5:
        best_c5, best_h = c5v, h2.copy()

    # Phase 8: Paired CD again (fine)
    dl8 = min(deadline, t0 + budget_s - 25)
    h2, c5v = _paired_cd(best_h.copy(), n, 20000, dl8)
    if c5v < best_c5:
        best_c5, best_h = c5v, h2.copy()

    # Phase 9: Ultra-fine polish
    rem = max(0, deadline - time.time())
    n_pol = min(int(rem / 0.001), 30000)
    best_h, best_c5 = _polish(best_h, n_pol, rng, deadline, 5e-7)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
