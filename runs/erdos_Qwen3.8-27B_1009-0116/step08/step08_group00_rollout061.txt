import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_fft(h):
    n = len(h)
    N = 2 * n
    H = np.fft.rfft(h, N)
    H2 = np.fft.rfft(1.0 - h, N)
    return np.fft.irfft(H * H2, N)[:2*n-1]

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    # Project onto {sum = target, 0 <= h <= 1}
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

    h = np.clip(h, 0.0, 1.0).copy()
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _precompute(n):
    offsets = np.arange(2 * n - 1) - (n - 1)
    vk_plus = [np.where((m + offsets >= 0) & (m + offsets < n))[0] for m in range(n)]
    vk_minus = [np.where((m - offsets >= 0) & (m - offsets < n))[0] for m in range(n)]
    return offsets, vk_plus, vk_minus

def _grad(h, p, offsets, vk_plus, vk_minus):
    n = len(h)
    g = np.zeros(n)
    for m in range(n):
        kp = vk_plus[m]
        km = vk_minus[m]
        g[m] = np.dot(p[kp], 1.0 - h[m + offsets[kp]]) - np.dot(p[km], h[m - offsets[km]])
    return g

def _grad_step(h, alpha, offsets, vk_plus, vk_minus, step0):
    n = len(h)
    c = _corr(h)
    m_c = c.max()
    s = alpha * (c - m_c)
    s -= s.max()
    p = np.exp(s)
    p /= p.sum()
    g = _grad(h, p, offsets, vk_plus, vk_minus)
    
    # Backtracking line search
    step = step0
    cur = m_c * (2.0 / n)
    for _ in range(15):
        h_new = h - step * g
        h_new = _repair(h_new, n)
        c_new = _corr(h_new)
        val = c_new.max() * (2.0 / n)
        if val < cur:
            return h_new, val
        step *= 0.5
    return h, cur
