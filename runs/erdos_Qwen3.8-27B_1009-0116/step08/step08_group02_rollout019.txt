    sh = (n - 1) - k
    g = np.zeros(n)

    return float(c.max()), int(np.argmax(c))

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:

    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_lag(k, h, n):
    dx = 2.0 / n
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        i1 = np.arange(n - sh)
        g[i1] += (1.0 - h[i1 + sh])
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh]
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp])
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp]
    return g * dx

def _smooth_max_obj(x, alpha, top_k):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    if top_k > 0 and top_k < len(c):
        idx = np.argsort(c)[-top_k:]
        cs = c[idx]
    else:
        cs = c
    m = cs.max()
    s = np.clip(alpha * (cs - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _smooth_max_grad(x, alpha, top_k):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    if top_k > 0 and top_k < len(c):
        idx = np.argsort(c)[-top_k:]
    else:
        idx = np.arange(len(c))
    cs = c[idx]
    m = cs.max()
    s = np.clip(alpha * (cs - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()

    for ii in range(len(idx)):
        if w[ii] < 1e-15:
            continue
        g += w[ii] * _grad_lag(idx[ii], x, n)
    return g

def _lgbfgs(h, alpha, top_k, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_smooth_max_obj, h, jac=_smooth_max_grad,
                       args=(alpha, top_k), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, n)
    except:
        return h

def _proj_grad(h, alpha_sched, steps, step0, top_k, mom):
    n = len(h)
    v = np.zeros(n)
    step = step0
    cur_c = _eval(h)
    for it in range(steps):
        alpha = alpha_sched[it % len(alpha_sched)]
        c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        if top_k > 0 and top_k < len(c):
            active = np.argsort(c)[-top_k:]
        else:
            active = np.arange(len(c))
        c_a = c[active]
        m_a = c_a.max()
        s = np.clip(alpha * (c_a - m_a), -400, 400)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        g = np.zeros(n)
        for ii in range(len(active)):
            if w[ii] < 1e-15:
                continue
            g += w[ii] * _grad_lag(active[ii], h, n)
        v = mom * v + (1.0 - mom) * g
        nv = np.linalg.norm(v)
        if nv > 0:
            v /= nv
        taken = False

def _informed_local(h, n_iter, rng, base_step):
    n = len(h)
    c5c, k_w = _eval_worst(h)
    best_h = h.copy()
    best_c = c5c
    sh = (n - 1) - k_w
    # Precompute which positions contribute most to worst lag
    for _ in range(n_iter):
        # Directed move: reduce overlap at worst lag
        if sh >= 0 and sh < n:
            # positions i where h[i] * (1-h[i+sh]) is large
            contrib = np.zeros(n)
            for i in range(n - sh):
                contrib[i] = h[i] * (1 - h[i + sh])
            # Also consider the "other side": increasing h[i+sh] helps
            # Pick a source (high contrib) and a sink (low h at shifted pos)
            src_candidates = np.where(h > 0.1)[0]
            dst_candidates = np.where(h < 0.9)[0]
            if len(src_candidates) > 0 and len(dst_candidates) > 0:
                # Weight by contribution
                src_wit = contrib[src_candidates]
                if src_wit.max() > 0:
                    src_wit_norm = src_wit / src_wit.sum()
                    si = src_candidates[rng.choice(len(src_candidates), p=src_wit_norm)]
                else:
                    si = rng.choice(src_candidates)
                di = rng.choice(dst_candidates)
                if si != di and 0 <= si + sh < n:
                    d = base_step * rng.randn()
                    ni, nj = h[si] - d, h[di] + d
                    if 0 <= ni and 0 <= nj <= 1:
                        h[si], h[di] = ni, nj
                        c5n, k_w = _eval_worst(h)
                        if c5n < c5c:
                            c5c = c5n
                            sh = (n - 1) - k_w
                            if c5n < best_c:
                                best_c = c5n
                                best_h = h.copy()
                        else:
                            h[si], h[di] = ni + d, nj - d
                        continue
        # Fallback: random 2-point
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 5)) % n
        d = base_step * rng.randn()
        ni, nj = h[i] + d, h[j] - d

            else:

def _block_swap(h, n_iter, rng, min_blk, max_blk):
    n = len(h)
    c5c = _eval(h)
    best_c = c5c
    best_h = h.copy()
    for _ in range(n_iter):
        m = rng.randint(min_blk, max_blk + 1)
        p = rng.randint(0, n - m)
        q = rng.randint(0, n - m)

            delta = rng.uniform(-0.02, 0.02)
            h[p:p+m] -= delta
            h[q:q+m] += delta
            if h.min() >= -1e-10 and h.max() <= 1 + 1e-10:
                h = np.clip(h, 0, 1)
                c5n = _eval(h)
                if c5n < c5c:
                    c5c = c5n
                    if c5n < best_c:

                else:
                    h[p:p+m] += delta
                    h[q:q+m] -= delta
            else:
                h[p:p+m] += delta
                h[q:q+m] -= delta
        else:
            # overlapping blocks, try swap
            if rng.rand() < 0.5:
                h[p:p+m], h[q:q+m] = h[q:q+m].copy(), h[p:p+m].copy()
                c5n = _eval(h)

    best_h = h.copy()
    best_c = c5c
    sh = (n - 1) - k_w
    for it in range(n_iter):
        if sh >= 0 and sh < n and n - sh > 1:
            contrib = np.zeros(n)
            for i in range(n - sh):
                contrib[i] = h[i] * (1.0 - h[i + sh])
            cmax = contrib.max()
            if cmax > 1e-10:
                src_pool = np.where(h > 0.05)[0]
                dst_pool = np.where(h < 0.95)[0]
                if len(src_pool) > 0 and len(dst_pool) > 0:
                    sw = contrib[src_pool]
                    sw_pos = np.maximum(sw, 1e-15)
                    p_src = sw_pos / sw_pos.sum()
                    si = src_pool[rng.choice(len(src_pool), p=p_src)]
                    di = rng.choice(dst_pool)
                    if si != di and si + sh < n:
                        d = base_step * rng.randn()
                        ni, nj = h[si] - d, h[di] + d
                        if 0.0 <= ni and 0.0 <= nj <= 1.0:
                            h[si], h[di] = ni, nj
                            c5n, k_w = _eval_worst(h)
                            if c5n < c5c:
                                c5c = c5n
                                sh = (n - 1) - k_w
                                if c5n < best_c:
                                    best_c = c5n
                                    best_h = h.copy()
                            else:
                                h[si], h[di] = ni + d, nj - d
                            continue
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 4)) % n
        d = base_step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n, k_w = _eval_worst(h)
            if c5n < c5c:
                c5c = c5n
                sh = (n - 1) - k_w
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def _block_local(h, n_iter, rng, min_blk, max_blk, base_delta):
    n = len(h)
    c5c = _eval(h)
    best_c = c5c
    best_h = h.copy()
    for _ in range(n_iter):
        m = rng.randint(min_blk, max_blk + 1)
        p = rng.randint(0, n - m)
        q = rng.randint(0, n - m)
        if p + m <= q or q + m <= p:
            delta = base_delta * rng.uniform(-1.0, 1.0)
            h[p:p+m] -= delta
            h[q:q+m] += delta
            if h.min() >= -1e-9 and h.max() <= 1.0 + 1e-9:
                h = np.clip(h, 0.0, 1.0)
                c5n = _eval(h)
                if c5n < c5c:
                    c5c = c5n
                    if c5n < best_c:
                        best_c = c5n
                        best_h = h.copy()
                else:
                    h[p:p+m] += delta
                    h[q:q+m] -= delta
            else:
                h[p:p+m] += delta
                h[q:q+m] -= delta
        else:
            tmp = h[p:p+m].copy()
            h[p:p+m] = h[q:q+m]
            h[q:q+m] = tmp
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[p:p+m] = tmp
                h[q:q+m] = h[p:p+m].copy()
                h[p:p+m] = h[q:q+m]
                h[q:q+m] = tmp
    return best_h, best_c

def _fine_polish(h, n_iter, rng, base_step):
    n = len(h)
    c5c = _eval(h)
    best_c = c5c
    best_h = h.copy()
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 3)) % n
        d = base_step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def _downsample(h, factor):
    n = len(h)
    m = n // factor
    return h[:m * factor].reshape(m, factor).mean(axis=1).copy()

def _upsample(h, factor):
    n = len(h)
    out = np.empty(n * factor)
    for i in range(n):
        out[i * factor:(i + 1) * factor] = h[i]
    return out

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    best_h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_c5 = _eval(best_h)

    # Phase 1: Coarse-to-fine SLSQP (50s)
    for ratio in [8, 4, 2]:
        nc = max(20, n // ratio)
        hc = _repair(_downsample(best_h, n // nc), nc)
        for alpha in [3000, 30000]:
            if time.time() - t0 > 50:
                break
            hc = _slsqp(hc, alpha, 0, 150)
        h_up = _repair(_upsample(hc, n // nc), n)
        c_up = _eval(h_up)
        if c_up < best_c5:
            best_c5 = c_up
            best_h = h_up.copy()

    # Phase 2: SLSQP at full resolution (120s)
    for alpha, tk, mit in [(5000, 4, 500), (30000, 3, 350), (150000, 2, 250), (600000, 2, 150)]:
        if time.time() - t0 > 170:
            break
        h_new = _slsqp(best_h, alpha, tk, mit)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new

    # Phase 3: Projected gradient with momentum (200s)
    if time.time() - t0 < budget_s - 380:
        alphas = [2000, 8000, 30000, 100000, 400000]
        h_new = _proj_grad(best_h, alphas, 2500, 0.006, 6, 0.7)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new
    if time.time() - t0 < budget_s - 320:
        alphas2 = [50000, 200000, 800000]
        h_new = _proj_grad(best_h, alphas2, 1500, 0.003, 3, 0.82)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new

    # Phase 4: SLSQP round 2 (40s)
    if time.time() - t0 < budget_s - 250:
        for alpha, tk, mit in [(2000000, 2, 200), (8000000, 1, 120)]:
            if time.time() - t0 > budget_s - 210:
                break
            h_new = _slsqp(best_h, alpha, tk, mit)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new

    # Phase 5: Informed local search (200s)
    if time.time() - t0 < budget_s - 120:
        time_left = budget_s - 120 - (time.time() - t0)
        _t = time.time()
        for _ in range(40):
            _eval_worst(best_h)
        eps = max((time.time() - _t) / 40.0, 1e-7)
        ntries = min(int(time_left * (1.0 / eps) * 0.7), 3000000)
        best_h, best_c5 = _informed_local(best_h, ntries, rng, 6e-6)

    # Phase 6: Block swap local search (100s)
    if time.time() - t0 < budget_s - 60:
        time_left = budget_s - 60 - (time.time() - t0)
        _t = time.time()
        for _ in range(40):
            _eval(best_h)
        eps = max((time.time() - _t) / 40.0, 1e-7)
        ntries = min(int(time_left * (1.0 / eps) * 0.55), 400000)
        best_h, best_c5 = _block_local(best_h, ntries, rng, 2, 12, 0.015)

    # Phase 7: Final SLSQP polish (30s)
    if time.time() - t0 < budget_s - 30:
        h_new = _slsqp(best_h, 10000000, 1, 80)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new

    # Phase 8: Ultra-fine 2-point polish (remaining)
    if time.time() - t0 < budget_s - 1:
        time_left = budget_s - 1 - (time.time() - t0)
        _t = time.time()
        for _ in range(40):
            _eval(best_h)
        eps = max((time.time() - _t) / 40.0, 1e-7)
        ntries = min(int(time_left * (1.0 / eps) * 0.75), 8000000)
        best_h, best_c5 = _fine_polish(best_h, ntries, rng, 5e-8)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
