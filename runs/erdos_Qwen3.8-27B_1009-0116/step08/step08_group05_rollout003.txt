    c = _correl(h)
    m = c.max()
    idx = np.argpartition(c, -topk)[-topk:].copy()
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-13:
            continue
        g += w[ii] * (2.0 / n) * _sg(h, idx[ii], n)
    return g

def _lp_descent(h, max_iters, eps_start, eps_end, m_start, m_end, deadline, rng):
    n = len(h)
    best_h, best_c5 = h.copy(), _eval(h)
    c = h.copy()
    c5_cur = best_c5
    stagnant = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        frac = it / max(max_iters, 1)
        eps = eps_start + (eps_end - eps_start) * frac
        m_top = max(3, int(m_start + (m_end - m_start) * frac))
        cor = _correl(c)
        idx_top = np.argpartition(cor, -m_top)[-m_top:]
        idx_top = idx_top[np.argsort(-cor[idx_top])]
        grads = np.array([_sg(c, k, n) for k in idx_top])
        c_obj = -grads[0]
        A_ub = grads[1:] if m_top > 1 else None
        b_ub = np.zeros(m_top - 1) if m_top > 1 else None
        A_eq = np.ones((1, n))
        b_eq = np.zeros(1)
        bnds = [(-eps, eps)] * n
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bnds, method='highs', options={'presolve': True})
        except Exception:
            break
        if not res.success or res.x is None:
            break
        d = res.x
        dmax = np.max(np.abs(d))
        if dmax < 1e-15:
            stagnant += 1
            if stagnant > 80:
                break
            continue
        stagnant = 0
        step = 1.0
        improved = False
        for _ in range(10):
            h_new = _repair(c + step * d, n)
            c5_new = _eval(h_new)
            if c5_new < c5_cur - 1e-15:
                c = h_new; c5_cur = c5_new
                if c5_cur < best_c5:
                    best_c5, best_h = c5_cur, c.copy()
                improved = True
                break
            step *= 0.5
        if not improved:
            d2 = rng.randn(n)
            d2 -= d2.mean()
            d2 *= eps * 0.3 / max(np.max(np.abs(d2)), 1e-15)
            h_new = _repair(c + d2, n)
            c5_new = _eval(h_new)
            if c5_new < c5_cur:
                c, c5_cur = h_new, c5_new
                if c5_cur < best_c5:
                    best_c5, best_h = c5_cur, c.copy()
    return best_h, best_c5

def _nesterov_method(h, alpha, topk, steps, step0, deadline):
    n = len(h)
    f_cur = _smax_obj(h, alpha, topk)
    v = np.zeros(n)
    beta = 0.82
    step = step0
    no_improve = 0
    for _ in range(steps):
        if time.time() > deadline:
            break
        g = _smax_grad(h, alpha, topk)
        v = beta * v + (1.0 - beta) * g
        h_new = _proj(h - step * v, n * 0.5)
        f_new = _smax_obj(h_new, alpha, topk)
        if f_new < f_cur:
            h, f_cur = h_new, f_new
            step = min(step * 1.015, step0 * 5.0)
            no_improve = 0
        else:
            step *= 0.65
            if step < 1e-13:
                break
            h = _proj(h - step * v, n * 0.5)
            f_cur = _smax_obj(h, alpha, topk)
            no_improve += 1
            if no_improve > 200:
                break
    return _repair(h, n)

def _coord_descent(h, rng, niter, deadline):
    n = len(h)
    best_h, best_c5 = h.copy(), _eval(h)
    c5_cur = best_c5
    cor = _correl(h)
    top30 = np.argpartition(cor, -30)[-30:]
    sizes = [8e-4, 4e-4, 2e-4, 1e-4, 5e-5]
    for ss in sizes:
        ni = max(20, niter // len(sizes))
        for _ in range(ni):
            if time.time() > deadline:
                return best_h, best_c5
            k = top30[rng.randint(len(top30))]
            off = k - (n - 1)
            i1 = rng.randint(n)
            j1 = i1 - off
            g = 0.0
            if 0 <= j1 < n:
                g += (1.0 - h[j1])
            if 0 <= j1 + off < n:
                g -= h[j1 + off]
            elif 0 <= j1 - off < n:
                g -= h[j1 - off]
            dl = -ss * g
            j = (i1 + 1 + rng.randint(0, 4)) % n
            a, b = h[i1] + dl, h[j] - dl
            if 0.0 <= a <= 1.0 and 0.0 <= b <= 1.0:
                h[i1], h[j] = a, b
                c5n = _eval(h)
                if c5n < c5_cur - 1e-18:
                    c5_cur = c5n
                    if c5n < best_c5:
                        best_c5, best_h = c5n, h.copy()
                    if _ % 800 == 0:
                        cor = _correl(h)
                        top30 = np.argpartition(cor, -30)[-30:]
                else:
                    h[i1], h[j] = a - dl, b + dl
    return best_h, best_c5

def _block_anneal(h, rng, niter, deadline, T0, T1):
    n = len(h)
    best_h, best_c5 = h.copy(), _eval(h)
    c5_cur = best_c5
    blocks = [1, 2, 3, 4, 5, 6, 8, 10, 14, 18]
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0 + bs].copy()
        d = 0.006 * max(T, 1e-9) ** 0.5 * rng.randn(bs)
        d -= d.mean()
        seg = np.clip(seg + d, 0.0, 1.0)
        h_new = h.copy()
        h_new[i0:i0 + bs] = seg
        h_new = _repair(h_new, n)
        c5n = _eval(h_new)
        if c5n < c5_cur or rng.random() < np.exp(-(c5n - c5_cur) / max(T, 1e-14)):
            h, c5_cur = h_new, c5n
            if c5n < best_c5:
                best_c5, best_h = c5n, h.copy()
    return best_h, best_c5

def _pg_step(h, alpha, topk, n_step, step_size, deadline):
    n = len(h)
    for _ in range(n_step):
        if time.time() > deadline:
            break
        g = _smax_grad(h, alpha, topk)
        h_new = _proj(h - step_size * g, n * 0.5)
        f_old = _smax_obj(h, alpha, topk)
        f_new = _smax_obj(h_new, alpha, topk)
        if f_new < f_old:
            h = h_new
        else:
            step_size *= 0.7
            if step_size < 1e-13:
                break
            h = _proj(h - step_size * _smax_grad(h, alpha, topk), n * 0.5)
    return _repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    n = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h, best_c5 = h0.copy(), _eval(h0)

    # Phase 1: Quick projected gradient to enter basin (30s)
    dl1 = min(deadline, t0 + 30)
    hv = _pg_step(best_h.copy(), 10000, 20, 300, 0.004, dl1)
    cv = _eval(hv)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()

    # Phase 2: LP Descent coarse (200s)
    dl2 = min(deadline, t0 + 230)
    hv, cv = _lp_descent(best_h.copy(), 10000, 0.007, 0.003, 10, 6, dl2, rng)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()

    # Phase 3: Nesterov with adapts (120s)
    dl3 = min(deadline, t0 + 380)
    for a, tk, st, ns in [(30000, 20, 0.003, 1500), (150000, 10, 0.001, 1500)]:
        if time.time() > dl3:
            break
        hv = _nesterov_method(best_h.copy(), a, tk, ns, st, dl3)
        cv = _eval(hv)
        if cv < best_c5:
            best_c5, best_h = cv, hv.copy()

    # Phase 4: LP Descent fine (250s)
    dl4 = min(deadline, t0 + 660)
    hv, cv = _lp_descent(best_h.copy(), 14000, 0.0025, 0.0008, 6, 4, dl4, rng)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()

    # Phase 5: Block annealing (100s)
    dl5 = min(deadline, t0 + 800)
    hv, cv = _block_anneal(best_h.copy(), rng, 40000, dl5, 0.005, 1e-12)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()

    # Phase 6: Targeted coord descent (100s)
    dl6 = min(deadline, t0 + 920)
    hv, cv = _coord_descent(best_h.copy(), rng, 80000, dl6)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()

    # Phase 7: Final ultra-fine LP (80s)
    dl7 = deadline
    hv, cv = _lp_descent(best_h.copy(), 20000, 0.0008, 0.0002, 4, 3, dl7, rng)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()

    # Final coord polish if time
    if time.time() < deadline - 3:
        hv, cv = _coord_descent(best_h.copy(), rng, 30000, deadline)
        if cv < best_c5:
            best_c5, best_h = cv, hv.copy()

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
