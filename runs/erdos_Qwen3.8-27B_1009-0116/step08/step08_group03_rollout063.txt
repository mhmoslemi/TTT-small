import numpy as np
import time

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)
    x += (S - x.sum()) / x.size
    for _ in range(10):
        np.clip(x, 0.0, 1.0, out=x)
        d = S - x.sum()
        if abs(d) < 1e-12:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
    return x

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _pn_arrays(h):
    n = len(h)
    N = 1
    m = 2 * n - 1
    while N < m:
        N <<= 1
    H = np.fft.rfft(h, N)
    ac = np.fft.irfft(H * np.conj(H), N).real[:n]
    pref = np.empty(n + 1)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    total = pref[n]
    P = np.empty(n)
    P[0] = total - ac[0]
    Nv = np.zeros(n)
    if n > 1:
        idx = np.arange(1, n)
        P[idx] = pref[n - idx] - ac[idx]
        Nv[idx] = total - pref[idx] - ac[idx]
    mx = float(max(np.max(P), np.max(Nv)))
    return P, Nv, mx

def _smooth_val(h, alpha):
    P, Nv, mx = _pn_arrays(h)
    n = len(h)
    dx = 2.0 / n
    v = np.concatenate([[P[0]], P[1:], Nv[1:]]) if n > 1 else np.array([P[0]])
    sh = v - mx
    sh = np.clip(sh, -400.0, 0.0)
    e = np.exp(sh)
    return dx * (mx + np.log(e.sum()) / alpha)

    n = len(h)
    dx = 2.0 / n
    if n == 1:
        return np.array([dx * (1.0 - 2.0 * h[0])])

    wN = np.zeros(n)

    u = np.zeros(n)
    u[1:] = wP[1:] + wN[1:]

    cp[1:] = np.cumsum(wP[1:])
    cn[1:] = np.cumsum(wN[1:])
    linP = cp[n-1::-1]
    linN = cn
    
    Nf = 1
    while Nf < 2*n-1:

    Hf = np.fft.rfft(h, Nf)

    L = convL[:n].real

    e = np.exp(np.clip(alpha * (v - mx), -350.0, 0.0))
    return (2.0 / n) * (mx + np.log(e.sum()) / alpha)

def _sg(h, alpha):
    P, Nv, mx = _pn(h)
    n = len(h)
    dx = 2.0 / n
    v = np.concatenate((P, Nv[1:]))
    e = np.exp(np.clip(alpha * (v - mx), -350.0, 0.0))
    t = e.sum()
    if t < 1e-300:
        e = np.ones_like(v)
        t = e.size
    w = e / t
    wP = w[:n]
    wN = w[n:]
    u = np.zeros(n)
    if n > 1:
        u[1:] = wP[1:] + wN
    cp = np.zeros(n)
    cn = np.zeros(n)
    if n > 1:
        cp[1:] = np.cumsum(wP[1:])
        cn[1:] = np.cumsum(wN)
    linP = cp[n-1::-1]
    linN = cn
    q = np.zeros(n)
    if n > 1 and np.any(u[1:]):
        Nf = 1
        while Nf < 2 * n - 1:
            Nf <<= 1
        Hf = np.fft.rfft(h, Nf)
        Ur = np.fft.rfft(u, Nf)
        Urev = np.fft.rfft(u[::-1], Nf)
        R = np.fft.irfft(Hf * Urev, Nf)[n-1:2*n-1].real
        L = np.fft.irfft(Hf * Ur, Nf)[:n].real
        q = -(R + L)
    return dx * (wP[0] * (1.0 - 2.0 * h) + linP + linN + q)

def _fista(h0, alpha, deadline, L0=10.0):
    n = len(h0)
    S = n * 0.5
    h = _proj(h0, S)
    hp = h.copy()
    y = h.copy()
    tk = 1.0
    fy = _sv(y, alpha)
    fh = fy
    best_h = h.copy()
    best_c = _c5(h)
    L = L0
    stale = 0
    for it in range(200000):
        if time.time() > deadline:
            break
        if it:
            tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * tk * tk))
            y = h + ((tk - 1.0) / tn) * (h - hp)
            tk = tn
            y = _proj(y, S)
            fy = _sv(y, alpha)
        else:
            y = h
            fy = fh
        g = _sg(y, alpha)
        step = 1.0 / max(L, 1e-12)
        ok = False
        for _ in range(12):
            xn = _proj(y - step * g, S)
            fx = _sv(xn, alpha)
            d = xn - y
            if fx <= fy + np.dot(g, d) + 0.5 / step * np.dot(d, d) + 1e-10:
                ok = True
                break
            step *= 0.5
        L = 1.0 / max(step, 1e-12)
        if ok:
            hp, h, fh = h, xn, fx
            stale = 0
            L = max(L * 0.88, 1.0)
            if it % 8 == 0:
                c = _c5(h)
                if c < best_c:
                    best_c, best_h = c, h.copy()
        else:
            stale += 1
            hp, y, tk, fy, L = h, h, 1.0, fh, max(L * 0.5, 1.0)
            if stale > 12:
                break
    c = _c5(h)
    if c < best_c:
        best_c, best_h = c, h.copy()
    return h, best_h, best_c

def _resample(h, m):
    n = len(h)
    if n == m:
        return _proj(h, m * 0.5)
    x0 = (np.arange(n) + 0.5) * (2.0 / n)
    x1 = (np.arange(m) + 0.5) * (2.0 / m)
    return _proj(np.interp(x1, x0, h), m * 0.5)
