import numpy as np
from scipy.optimize import minimize, linprog
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _smax(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _smax_grad(h, alpha, topk):

    m = c.max()
    if topk < len(c):

    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n)

def _coord_descent_1d(h, n_passes, deadline):
    n = len(h)
    h = h.copy()
    c = _correl(h)
    best_h, best_c5 = h.copy(), _eval(h)
    
    for p in range(n_passes):
        if time.time() > deadline:
            break
        improved = False
        order = np.arange(n)
        # Randomize order each pass
        # (use a simple shuffle)
        for i in range(n):
            j = order[i]
            # Compute alpha for index j
            alpha = np.zeros(2*n - 1)
            # j as first index: shift from -j to n-1-j
            sh1_lo = max(-j, -(n-1))
            sh1_hi = min(n-1-j, n-1)
            for sh in range(sh1_lo, sh1_hi + 1):
                if 0 <= j + sh < n:
                    alpha[sh + n - 1] += (1.0 - h[j + sh])
            # j as second index: shift from j-(n-1) to j
            sh2_lo = max(j - (n-1), -(n-1))
            sh2_hi = min(j, n-1)
            for sh in range(sh2_lo, sh2_hi + 1):
                if 0 <= j - sh < n:
                    alpha[sh + n - 1] += -h[j - sh]
            
            # 1D minimax: min_d max_k (c[k] + alpha[k]*d) s.t. 0<=h[j]+d<=1
            lo = -h[j]
            hi = 1.0 - h[j]
            # Evaluate at boundaries and find interior minimum
            f_lo = np.max(c + alpha * lo)
            f_hi = np.max(c + alpha * hi)
            # Binary search / ternary search on the convex function
            # The function g(d) = max_k(c[k] + alpha[k]*d) is convex in d
            # Use ternary search
            a, b = lo, hi
            for _ in range(30):
                m1 = a + (b - a) / 3
                m2 = b - (b - a) / 3
                f1 = np.max(c + alpha * m1)
                f2 = np.max(c + alpha * m2)
                if f1 < f2:
                    b = m2
                else:
                    a = m1
            d_opt = (a + b) / 2
            f_opt = np.max(c + alpha * d_opt)
            
            if f_opt < c.max() - 1e-10:
                h[j] = np.clip(h[j] + d_opt, 0, 1)
                c[alpha != 0] += alpha[alpha != 0] * d_opt  # Only update affected
                # Actually need to recompute c properly
                # The shifts affected are those where alpha != 0
                nz = np.nonzero(alpha)[0]
                c[nz] = c[nz] + alpha[nz] * d_opt
                improved = True
                cur_best = np.max(c)
                if cur_best < best_c5:
                    best_c5, best_h = cur_best, h.copy()
        
        if not improved:
            break
        # Repair after each pass
        h = _repair(h, n)
        c = _correl(h)
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _smax(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _smax_grad(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
    else:
        idx = np.arange(len(c))
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    for ii in range(len(idx)):
        if w[ii] < 1e-13:
            continue
        k = idx[ii]
        sh = k - (n - 1)
        wk = w[ii] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _slsqp_phase(h, deadline):
    n = len(h)
    best_h, best_c5 = h.copy(), _eval(h)
    configs = [(2000, 60, 800), (8000, 30, 600), (30000, 15, 400)]
    for alpha, topk, maxit in configs:
        if time.time() > deadline:
            break
        def obj(x):
            return _smax(x, alpha, topk)
        def jac(x):
            return _smax_grad(x, alpha, topk)
        def con_fun(x):
            return x.sum() - n * 0.5
        def con_jac(x):
            return np.ones(n)
        cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
        try:
            r = minimize(obj, best_h, jac=jac, method='SLSQP',
                         bounds=[(0.0, 1.0)] * n, constraints=cons,
                         options={'maxiter': maxit, 'ftol': 1e-18, 'maxls': 50})
            hv = _repair(r.x, n)
            cv = _eval(hv)
            if cv < best_c5:
                best_h, best_c5 = hv.copy(), cv
        except Exception:
            pass
    return best_h, best_c5

def _coord_descent(h, n_passes, deadline, rng):

    full = 2 * n - 1
    best_h, best_c5 = h.copy(), float(c.max())

        for idx_i in range(n):
            j = order[idx_i]
            hj = h[j]

For np.correlate(h, 1-h, mode='full'):
The result has length 2n-1.
result[j] = sum_i h[i] * (1-h)[j-i] for valid i, j-i where 0≤i<n, 0≤j-i<n.

Let me define shift = j - (n-1). Then j = (n-1) + shift.
result[n-1+shift] = sum_i h[i] * (1-h[i+shift]) for 0≤i<n, 0≤i+shift<n.

For shift=0: sum_i h[i]*(1-h[i]) = sum_i h[i] - sum_i h[i]²
For shift=1: sum_i h[i]*(1-h[i+1]) for i=0 to n-2
For shift=-1: sum_i h[i]*(1-h[i-1]) for i=1 to n-1

import numpy as np
from scipy.optimize import minimize, linprog
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _smax(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _smax_grad(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
    else:
        idx = np.arange(len(c))
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    for ii in range(len(idx)):
        if w[ii] < 1e-13:
            continue
        k = idx[ii]
        sh = k - (n - 1)
        wk = w[ii] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _slsqp_phase(h, deadline):
    n = len(h)
    best_h, best_c5 = h.copy(), _eval(h)
    configs = [(5000, 50, 700), (20000, 25, 500), (80000, 12, 400)]
    for alpha, topk, maxit in configs:
        if time.time() > deadline:
            break
        def _obj(x):
            return _smax(x, alpha, topk)
        def _jac(x):
            return _smax_grad(x, alpha, topk)
        def _cfun(x):
            return x.sum() - n * 0.5
        def _cjac(x):
            return np.ones(n)
        cons = [{'type': 'eq', 'fun': _cfun, 'jac': _cjac}]
        try:
            r = minimize(_obj, best_h, jac=_jac, method='SLSQP',
                         bounds=[(0.0, 1.0)] * n, constraints=cons,
                         options={'maxiter': maxit, 'ftol': 1e-18, 'maxls': 50})
            hv = _repair(r.x, n)
            cv = _eval(hv)
            if cv < best_c5:
                best_h, best_c5 = hv.copy(), cv
        except Exception:
            pass
    return best_h, best_c5

def _coord_descent(h, n_passes, deadline, rng):
    n = len(h)
    h = h.copy()
    c = _correl(h).copy()
    full = 2 * n - 1
    best_h, best_c5 = h.copy(), float(c.max())
    for p in range(n_passes):
        if time.time() > deadline:
            break
        order = rng.permutation(n)
        any_imp = False
        for idx_i in range(n):
            j = order[idx_i]
            hj = h[j]
            if hj < 1e-14 or hj > 1.0 - 1e-14:
                continue
            alpha = np.zeros(full)
            sh_lo = max(-(n - 1), -j)
            sh_hi = min(n - 1, n - 1 - j)
            if sh_lo <= sh_hi:
                sh_arr = np.arange(sh_lo, sh_hi + 1)
                alpha[sh_arr + n - 1] += (1.0 - h[j + sh_arr])
            sh_lo2 = max(-(n - 1), -(n - 1 - j))
            sh_hi2 = min(n - 1, j)
            if sh_lo2 <= sh_hi2:
                sh_arr2 = np.arange(sh_lo2, sh_hi2 + 1)
                alpha[sh_arr2 + n - 1] -= h[j - sh_arr2]
            a, b = -hj, 1.0 - hj
            for _ in range(24):
                m1 = a + (b - a) / 3.0
                m2 = b - (b - a) / 3.0
                if np.max(c + alpha * m1) <= np.max(c + alpha * m2):
                    b = m2
                else:
                    a = m1
            d_opt = (a + b) / 2.0
            if np.max(c + alpha * d_opt) < c.max() - 1e-11:
                h[j] = np.clip(hj + d_opt, 0.0, 1.0)
                d_act = h[j] - hj
                nz = np.nonzero(alpha)[0]
                c[nz] += alpha[nz] * d_act
                any_imp = True
                if c.max() < best_c5:
                    best_c5 = float(c.max())
                    best_h = h.copy()
        if not any_imp:
            break
        if (p + 1) % 30 == 0:
            h = _repair(h, n)
            c = _correl(h).copy()
    h = _repair(h, n)
    return best_h, best_c5

            i = rng.randint(n)

            ni, nj = h[i] + d, h[j] - d
