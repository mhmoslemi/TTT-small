def _slsqp(h, alpha, topk, maxit, deadline):

        m = c.max()

        else:
            ct = c
        s = np.clip(alpha * (ct - m), -350, 350)
        s -= s.max()
        return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

        return _grad_smax(x, _correl(x), alpha, topk)

import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo = float(h.min()) - 1.0
    hi = float(h.max())
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = n * 0.5
    if abs(h.sum() - t) > 1e-10:
        h = _proj(h, t)
    diff = t - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _grad_smax(h, c, alpha, topk):
    n = len(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx].copy()
    else:
        idx = np.arange(len(c))
        ct = c
    s = np.clip(alpha * (ct - m), -350.0, 350.0)
    s -= s.max()
    w = np.exp(s)
    wsum = w.sum()
    if wsum < 1e-300:
        w = np.ones(len(idx), dtype=float)
        wsum = float(len(idx))
    else:
        w /= wsum
    dx = 2.0 / n
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-14:
            continue
        k = int(idx[ii])
        sh = (n - 1) - k
        wk = w[ii] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            if sp < n:
                g[:n - sp] -= wk * h[sp:]
    return g

def _grad_exact(h, c, tol):
    n = len(h)
    m = c.max()
    active = np.where(c >= m - tol)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(c))])
    dx = 2.0 / n
    g = np.zeros(n)
    wa = dx / len(active)
    for k in active:
        sh = (n - 1) - int(k)
        if sh >= 0:
            g[:n - sh] += wa * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wa * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wa * (1.0 - h[:-sp])
            if sp < n:
                g[:n - sp] -= wa * h[sp:]
    return g, m

def _sgd(h, deadline, mode, alpha0, alpha1, topk, tol, lr0, mom, nmax, cooldown):
    n = len(h)
    target = n * 0.5
    f = _eval(h)
    best_h, best_f = h.copy(), f
    v = np.zeros(n)
    lr = lr0
    c = _correl(h)
    stale = 0
    for step in range(nmax):
        if time.time() > deadline:
            break
        frac = step / max(nmax, 1)
        alpha = alpha0 * (alpha1 / alpha0) ** frac if mode == 'smax' else 0.0
        if step % 25 == 0:
            c = _correl(h)
        if mode == 'smax':
            g = _grad_smax(h, c, alpha, topk)
        else:
            g, _ = _grad_exact(h, c, tol)
        v = mom * v + (1.0 - mom) * g
        hn = _proj(h - lr * v, target)
        fn = _eval(hn)
        if fn < f - 1e-16:
            h, f = hn, fn
            lr = min(lr * 1.008, lr0 * 30.0)
            stale = 0
            if f < best_f:
                best_f, best_h = f, h.copy()
        else:
            lr *= 0.6
            if lr < 1e-12:
                lr = lr0 * 0.08
            stale += 1
            h = _proj(h - lr * v, target)
            f = _eval(h)
            if f < best_f:
                best_f, best_h = f, h.copy()
        if stale > cooldown:
            break
    return best_h, best_f

    dl7 = min(deadline, t0 + budget_s - 50)
    rem = max(0.0, dl7 - time.time())
    n_cd = min(int(rem / 0.002), 70000)
    best_h, best_c = _coord_descent(best_h, n_cd, rng, dl7, 5, 0.003)

    if deadline - time.time() > 25:
        h2, c5 = _sgd(best_h.copy(), min(deadline - 5, time.time() + 28), 'exact', 0, 0, 0, 0.0003, 0.0008, 0.94, 2500, 180)
        if c5 < best_c:
            best_c, best_h = c5, h2.copy()

    rem = max(0.0, deadline - time.time())
    n_pol = min(int(rem / 0.0005), 120000)
    best_h, best_c = _fine_polish(best_h, n_pol, rng, deadline, 1.5e-5)

    best_h = _repair(best_h, len(best_h))
    best_c = _eval(best_h)
    return (best_h, best_c, len(best_h))
