def _delta_corr(k, h, n, i, j, d):
    """Compute change in corr[k] when h[i]+=d, h[j]-=d. Cross terms ignored."""
    s = k - (n - 1)
    delta = 0.0
    # Effect of h[i] += d
    if s >= 0:
        if i <= n - 1 - s:  # i in first factor
            delta += d * (1.0 - h[i + s])
        if i >= s:  # i in second factor (as h in 1-h)
            delta -= h[i - s] * d
    else:
        sp = -s
        if i >= sp:  # i in first factor
            delta += d * (1.0 - h[i - sp])
        if i <= n - 1 - sp:  # i in second factor
            delta -= h[i + sp] * d
    # Effect of h[j] -= d (i.e., delta_j = -d)
    if s >= 0:
        if j <= n - 1 - s:
            delta += (-d) * (1.0 - h[j + s])
        if j >= s:
            delta -= h[j - s] * (-d)
    else:
        sp = -s
        if j >= sp:
            delta += (-d) * (1.0 - h[j - sp])
        if j <= n - 1 - sp:
            delta -= h[j + sp] * (-d)
    return delta

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-14:
                break
    return h

    else:

    return g * (2.0 / n)

        cs = c
    m = cs.max()
    s = np.clip(alpha * (cs - m), -400, 400); s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _lsep_jac(x, n, alpha, top_k):

                cur_c = float(c_arr.max())
                top_lags = np.argsort(c_arr)[-KL:]
                no_improve = 0
        # Adaptive step
        if total_n > 0 and it % 5000 == 0:
            rate = accept_n / max(total_n, 1)
            if rate > 0.75: step *= 1.3
            elif rate < 0.15: step *= 0.6
            step = max(step, 1e-9)
            accept_n = 0; total_n = 0

    best_h = best_local
    best_c5 = _eval(best_h)

    # Phase 4: Final ultra-fine polish (20s)
    if time.time() - t0 < budget_s - 2:
        c_arr = _eval_full(best_h)
        top_lags = np.argsort(c_arr)[-10:]
        cur_c = float(c_arr.max())
        cur_h = best_h.copy()
        for _ in range(200000):
            if time.time() - t0 > budget_s - 2: break
            k_w = int(np.argmax(c_arr))
            s_w = k_w - (n - 1)
            if s_w > 0:
                i = rng.randint(n - s_w); j = i + s_w
            elif s_w < 0:
                j = rng.randint(n + s_w); i = j - s_w
            else:
                i = rng.randint(n); j = (i + 1) % n
            d = 3e-8 * rng.randn()
            ni, nj = cur_h[i]+d, cur_h[j]-d
            if not (0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0): continue
            max_new = -1.0
            for kk in top_lags:
                dv = c_arr[int(kk)] + _delta_k(int(kk), cur_h, n, i, j, d)
                if dv > max_new: max_new = dv
            if max_new < cur_c:
                cur_h[i], cur_h[j] = ni, nj
                cur_c = max_new
                if cur_c < best_c5:
                    best_c5 = cur_c
                    best_h = cur_h.copy()

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
