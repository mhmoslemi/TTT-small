import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _ls_obj(x, alpha, M, n):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    top = np.argpartition(c, -M)[-M:]
    vals = c[top]
    m = vals.max()
    s = alpha * (vals - m)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _ls_grad(x, alpha, M, n):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    top = np.argpartition(c, -M)[-M:]
    vals = c[top]
    m = vals.max()
    s = alpha * (vals - m)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(top):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            Mg = -L
            g[:n - Mg] += wk * (1.0 - x[Mg:])
            g[Mg:] -= wk * x[:n - Mg]
    return g

def _sum_fn(x, n):
    return x.sum() - n * 0.5

def _sum_jac(x, n):
    return np.ones(len(x))

def _slsqp(h, alpha, M, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fn, 'jac': _sum_jac, 'args': (n,)}]
    try:
        res = minimize(_ls_obj, h, jac=_ls_grad, args=(alpha, M, n),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h

def _nesterov(h, alpha, M, n_steps, step0, t0, deadline):
    n = len(h)
    v = np.zeros(n)
    step = step0
    cur_c = _eval(h)
    best_h = h.copy()
    best_c = cur_c
    for it in range(n_steps):
        if time.time() > deadline:
            break
        g = _ls_grad(h, alpha, M, n)
        v = 0.85 * v + 0.15 * g
        taken = False
        for _t in range(15):
            h_new = _proj(h - step * v, n * 0.5)
            c_new = _eval(h_new)
            if c_new < cur_c:
                h = h_new
                cur_c = c_new
                step = min(step * 1.05, step0 * 30.0)
                taken = True
                break
            step *= 0.5
        if not taken:
            step = max(step * 1.2, step0 * 0.3)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c
