    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _grad_softmax(h, alpha, n):
    """Gradient of softmax-weighted correlation sum"""
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    m = c.max()
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    dx2 = 2.0 / n

        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]
    return g

    """Gradient of max (or sum of top-k) correlation"""
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    idx = np.argsort(c)[-top_k:]
    g = np.zeros(n)

            sp = -sh

    step = 0.003

        progress = min(1.0, (it + 1) / 2000)

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) * 0.5, 0, 1)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _grad_shift(h, sh, n):
    """Gradient contribution for a single shift"""
    g = np.zeros(n)
    dx2 = 2.0 / n
    if sh >= 0:
        g[:n-sh] += dx2 * (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= dx2 * h[:n-sh]
    else:
        sp = -sh
        g[sp:] += dx2 * (1.0 - h[:n-sp])
        g[:n-sp] -= dx2 * h[sp:]
    return g

def _grad_softmax(h, alpha, n):
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        sh = (n - 1) - k
        g += w[k] * _grad_shift(h, sh, n)
    return g

def _grad_topk(h, top_k, n):
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    idx = np.argsort(c)[-top_k:]
    g = np.zeros(n)
    wk = 1.0 / top_k
    for k in idx:
        sh = (n - 1) - k
        g += wk * _grad_shift(h, sh, n)
    return g

class _Obj:
    def __init__(self, alpha, n):
        self.alpha = alpha
        self.n = n
    def f(self, x):
        c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / self.n)
        m = c.max()
        s = np.clip(self.alpha * (c - m), -500, 500)
        return m + np.log(np.sum(np.exp(s - s.max()))) / self.alpha
    def g(self, x):
        return _grad_softmax(x, self.alpha, self.n)

def _lbfgs_opt(h, alpha, maxiter, t_end):
    n = len(h)
    obj = _Obj(alpha, n)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj.f, h.copy(), jac=obj.g, method='L-BFGS-B',
                       bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-9, 'maxls': 30})
        return _repair(res.x, n)
    except Exception:
        return h

def _softmax_sgd(h, n, t_end, alpha0, alpha1, n_iter):
    c5 = _eval(h)
    v = np.zeros(n)
    step = 0.004
    mu = 0.85
    for it in range(n_iter):
        if time.time() > t_end:
            break
        prog = min(1.0, (it + 1) / float(n_iter))
        alpha = alpha0 * (alpha1 / alpha0) ** (prog * prog)
        g = _grad_softmax(h, alpha, n)
        v = mu * v + (1.0 - mu) * g
        h_new = _repair(h - step * v, n)
        c5_new = _eval(h_new)
        if c5_new < c5:
            h, c5 = h_new, c5_new
            step = min(step * 1.015, 0.012)
        else:
            step *= 0.45
            if step < 1e-11:
                step = 0.004
                v *= 0.5
            h_new = _repair(h - step * v, n)
            h, c5 = h_new, _eval(h_new)
    return h, c5

    v = np.zeros(n)
    step = 0.003

                h = h_new
                if c5 < best_c5:
                    best_c5, best_h = c5, h.copy()
                break
    return best_h, best_c5

def _pair_search(h, n, n_trials, rng, t_end, scale=0.0015):
    c5 = _eval(h)
    best_h, best_c5 = h.copy(), c5
    for _ in range(n_trials):

        i = rng.randint(n)

                if c5 < best_c5:
                    best_c5, best_h = c5, h.copy()
