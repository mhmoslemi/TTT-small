    n = len(h)
    H = np.fft.rfft(h)
    noise = scale * (rng.randn(H.shape[0]) + 1j * rng.randn(H.shape[0]))
    H[0] = 0
    h_new = np.fft.irfft(H + noise, n)
    return _repair(h_new, n)

def _downsample(h, f):
    n = len(h)
    m = n // f
    return h[:m*f].reshape(m, f).mean(axis=1).copy()

def _upsample(h, f):
    n = len(h)
    out = np.empty(n * f)
    for i in range(n):
        out[i*f:(i+1)*f] = h[i]
    return out

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def _el():
        return time.time() - t0
    rng = np.random.RandomState(seed)

    h_parent = _repair(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values))
    n = len(h_parent)
    best_h = h_parent.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Adam gradient with annealing (multiple configs)
    gd_configs = [
        (500, 5000, 8, 0.05),
        (200, 20000, 6, 0.03),
        (100, 100000, 5, 0.01),
        (50, 500000, 4, 0.005),
    ]
    for a0, a1, tk, lr in gd_configs:
        if _el() > 200:
            break
        h_t = best_h.copy()
        h_t, c = _adam_gd(h_t, 800, a0, a1, tk, lr, rng)
        c = _eval(h_t)
        if c < best_c5:
            best_c5, best_h = c, h_t.copy()

    # Phase 2: Multi-start with different n
    if _el() < budget_s - 600:
        for trial in range(6):
            if _el() > budget_s - 550:
                break
            if trial == 0:
                h_s = _freq_perturb(best_h, rng, 0.03)
            elif trial == 1:
                h_s = _freq_perturb(best_h, rng, 0.01)
            elif trial == 2:
                # Binary starting point
                h_bin = np.zeros(n)
                ones = rng.choice(n, n // 2, replace=False)
                h_bin[ones] = 1.0
                h_s = h_bin + 0.1 * rng.randn(n)
                h_s = _repair(h_s, n)
            elif trial == 3:
                # Multi-res: downsample, optimize, upsample
                nc = n // 4
                hc = _repair(_downsample(best_h, 4), nc)
                hc, _ = _adam_gd(hc, 300, 300, 3000, 5, 0.08, rng)
                h_up = _repair(_upsample(hc, 4), n)
                h_s = h_up
            elif trial == 4:
                # Periodic perturbation
                h_s = best_h.copy()
                h_s += 0.02 * np.sin(2 * np.pi * rng.randint(3, 20) * np.arange(n) / n)
                h_s = _repair(h_s, n)
            else:
                h_s = _freq_perturb(best_h, rng, 0.005)
            c_s = _eval(h_s)
            if c_s < best_c5:
                best_c5, best_h = c_s, h_s.copy()
            h_t, c = _adam_gd(h_s, 600, 200, 50000, 5, 0.02, rng)
            c = _eval(h_t)
            if c < best_c5:
                best_c5, best_h = c, h_t.copy()

    # Phase 3: Targeted worst-shift fixing
    if _el() < budget_s - 400:
        h_t, c = _targeted_fix(best_h.copy(), rng, 3000)
        c = _eval(h_t)
        if c < best_c5:
            best_c5, best_h = c, h_t.copy()
        # Local polish after targeted fix
        h_t, c = _local2(best_h.copy(), 5000, 1e-4, rng)
        c = _eval(h_t)
        if c < best_c5:
            best_c5, best_h = c, h_t.copy()

    # Phase 4: Block local search
    if _el() < budget_s - 300:
        remaining = int((budget_s - 300 - _el()) * 0.6)
        h_t, c = _local_block(best_h.copy(), max(2000, remaining), 0.03, rng, 3)
        c = _eval(h_t)
        if c < best_c5:
            best_c5, best_h = c, h_t.copy()
        h_t, c = _local_block(best_h.copy(), max(2000, remaining // 2), 0.01, rng, 2)
        c = _eval(h_t)
        if c < best_c5:
            best_c5, best_h = c, h_t.copy()

    # Phase 5: Fine 2-point local search (multi-scale)
    for step_sz in [5e-4, 1e-4, 3e-5, 1e-5, 3e-6]:
        if _el() > budget_s - 80:
            break
        n_it = min(int((budget_s - 80 - _el()) * 800), 50000)
        if n_it < 1000:
            break
        h_t, c = _local2(best_h.copy(), n_it, step_sz, rng)
        c = _eval(h_t)
        if c < best_c5:
            best_c5, best_h = c, h_t.copy()

    # Phase 6: Final targeted + polish
    if _el() < budget_s - 30:
        h_t, c = _targeted_fix(best_h.copy(), rng, 2000)
        c = _eval(h_t)
        if c < best_c5:
            best_c5, best_h = c, h_t.copy()
        n_fin = min(int((budget_s - 10 - _el()) * 2000), 100000)
        h_t, c = _local2(best_h, max(5000, n_fin), 1e-6, rng)
        c = _eval(h_t)
        if c < best_c5:
            best_c5, best_h = c, h_t.copy()

    # Phase 7: Ultra-fine final
    if _el() < budget_s - 5:
        n_fin = min(int((budget_s - 5 - _el()) * 3000), 150000)
        h_t, c = _local2(best_h, max(5000, n_fin), 5e-7, rng)
        c = _eval(h_t)
        if c < best_c5:
            best_c5, best_h = c, h_t.copy()

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
