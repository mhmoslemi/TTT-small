import numpy as np
from scipy.optimize import minimize, linprog
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_shift(h, k, n):
    g = np.zeros(n)
    off = k - (n - 1)
    j = np.arange(n)
    i1 = j - off
    m1 = (i1 >= 0) & (i1 < n)
    g[m1] += (1.0 - h[i1[m1]])
    i2 = j + off
    m2 = (i2 >= 0) & (i2 < n)
    g[m2] -= h[i2[m2]]
    return g

def _smax_val(x, alpha, topk):
    c = _correl(x)
    m = c.max()
    idx = np.argpartition(c, -topk)[-topk:]
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _smax_grad(x, alpha, topk):
    n = len(x)
    c = _correl(x)
    m = c.max()
    idx = np.argpartition(c, -topk)[-topk:].copy()
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max()); w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-13:
            continue
        g += w[ii] * _grad_shift(x, idx[ii], n)
    return g * dx

    return x.sum() - target

def _slsqp_opt(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    try:
        cons = {'type': 'eq', 'fun': _con_fn, 'jac': _con_jac, 'args': (n * 0.5,)}
        r = minimize(_smax_val, h, args=(alpha, topk), jac=_smax_grad,
                     method='SLSQP', bounds=[(0.0, 1.0)] * n,
                     constraints=[cons],
                     options={'maxiter': maxit, 'ftol': 1e-18, 'maxls': 100})
        return _repair(r.x, n)
    except Exception:
        return h

def _lp_descent(h, n_steps, eps, m_top, deadline):
    n = len(h)
    best_h, best_c5 = h.copy(), _eval(h)
    c = h.copy()
    c5_cur = best_c5
    stall = 0

        A_ub = grads[1:] if m_top > 1 else None

        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h, c5c = hn, c5n
            if c5n < bc:
                bc, bh = c5n, h.copy()
    return bh, bc

def _polish(h, niter, rng, deadline):
    nc = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    scales = [2e-5, 1e-5, 5e-6, 2e-6, 1e-6, 5e-7, 2e-7, 1e-7, 5e-8, 2e-8]
    for ss in scales:
        ni2 = max(niter // len(scales), 200)
        for _ in range(ni2):
            if time.time() > deadline:
                return bh, bc
            i = rng.randint(nc)
            j = (i + 1 + rng.randint(0, 5)) % nc
            if i == j:
                continue
            d = ss * (2.0 * rng.random() - 1.0)
            ni, nj = h[i] + d, h[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-19:
                    c5c = c5n
                    if c5n < bc:
                        bc, bh = c5n, h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 10
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h, best_c5 = h.copy(), _eval(h)

    # Phase 1: SLSQP (0-140s)
    dl1 = min(deadline, t0 + 140)
    slsqp_cfg = [(3000, 50, 500), (10000, 30, 500), (50000, 20, 400), (200000, 12, 350)]
    for a, tk, mi in slsqp_cfg:
        if time.time() > dl1:
            break
        hv = _slsqp_opt(best_h.copy(), a, tk, mi, dl1)
        cv = _eval(hv)
        if cv < best_c5:
            best_c5, best_h = cv, hv.copy()

    # Phase 2: LP descent large steps (140-320s)
    dl2 = min(deadline, t0 + 320)
    hv, cv = _lp_descent(best_h.copy(), 25000, 0.01, 8, dl2)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()

    # Phase 3: Subgradient + averaging (320-520s)
    dl3 = min(deadline, t0 + 520)
    sg_cfg = [(5000, 25, 0.008), (20000, 18, 0.004), (80000, 12, 0.002), (300000, 8, 0.001)]
    for a, tk, st in sg_cfg:
        if time.time() > dl3:
            break
        hv, cv = _subgrad_avg(best_h.copy(), a, tk, 8000, st, dl3)
        if cv < best_c5:
            best_c5, best_h = cv, hv.copy()

    # Phase 4: LP descent small steps (520-720s)
    dl4 = min(deadline, t0 + 720)
    lp_cfg = [(0.004, 6, 20000), (0.0015, 5, 20000)]
    for eps_v, m_t, ns in lp_cfg:
        if time.time() > dl4:
            break
        hv, cv = _lp_descent(best_h.copy(), ns, eps_v, m_t, dl4)
        if cv < best_c5:
            best_c5, best_h = cv, hv.copy()
        # Quick subgradient pass after LP
        if time.time() < dl4 - 20:
            hv2, cv2 = _subgrad_avg(best_h.copy(), 50000, 10, 3000, 0.002, dl4)
            if cv2 < best_c5:
                best_c5, best_h = cv2, hv2.copy()

    # Phase 5: Annealing (720-900s)
    dl5 = min(deadline, t0 + 900)
    blocks = [1, 2, 3, 4, 5, 6, 8, 10, 14]
    hv, cv = _anneal(best_h.copy(), 50000, rng, dl5, 0.003, 1e-12, blocks)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()
        if time.time() < dl5 - 40:
            hv2, cv2 = _lp_descent(best_h.copy(), 8000, 0.001, 5, dl5)
            if cv2 < best_c5:
                best_c5, best_h = cv2, hv2.copy()
            if time.time() < dl5 - 10:
                hv3, cv3 = _subgrad_avg(best_h.copy(), 100000, 8, 2000, 0.001, dl5)
                if cv3 < best_c5:
                    best_c5, best_h = cv3, hv3.copy()

    # Phase 6: Final SLSQP with very high alpha (900-960s)
    dl6 = min(deadline, t0 + 960)
    if time.time() < dl6:
        hv = _slsqp_opt(best_h.copy(), 500000, 8, 300, dl6)
        cv = _eval(hv)
        if cv < best_c5:
            best_c5, best_h = cv, hv.copy()

    # Phase 7: Very fine polish (960s to deadline)
    rem = max(0, deadline - time.time())
    npol = min(int(rem / 0.0005), 120000)
    best_h, best_c5 = _polish(best_h.copy(), npol, rng, deadline)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
