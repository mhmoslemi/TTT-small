def _repair(h, target=None):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    if target is None:
        target = h.size * 0.5
    diff = target - h.sum()
    if abs(diff) <= 1e-15:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
    else:
        order = np.argsort(h)
    for idx in order:
        if diff > 0:
            add = min(diff, 1.0 - h[idx])
            h[idx] += add
            diff -= add
        else:
            sub = min(-diff, h[idx])
            h[idx] -= sub
            diff += sub
        if abs(diff) < 1e-15:
            break
    # clean tiny negatives
    h[h < 0.0] = 0.0
    h[h > 1.0] = 1.0
    return h

def _soft_grad(h, alpha):

    dx = 2.0 / n
    nm1 = n - 1
    c = np.correlate(h, 1.0 - h, 'full') * dx
    m = c.max()
    z = alpha * (c - m)
    z -= z.max()
    w = np.exp(z)
    w /= w.sum()
    g = np.zeros(n)
    eps = 1e-18
    w0 = w[nm1]
    if w0 > eps:
        g += dx * w0 * (1.0 - 2.0 * h)

        low = h[:L]

        g[r:] += dx * (wp * (1.0 - low) - wm * low)
    return g

def _fourier_start(n, rng):

def _pair_update(h, corr, p, q, dp, dq, nm1, dx):
    op = h[p]; oq = h[q]
    if p < q: a=p; b=q; da=dp; db=dq; oa=op; ob=oq
    else: a=q; b=p; da=dq; db=dp; oa=oq; ob=op

def _interp_to(h, m):
    h = np.asarray(h, dtype=float)
    n = h.size
    if m == n: return h.copy()
    x0 = np.linspace(0.0, 2.0, n, endpoint=False)
    x1 = np.linspace(0.0, 2.0, m, endpoint=False)
    out = np.interp(x1, x0, h, left=h[0], right=h[-1])
    return _repair(out, 0.5*m)

import time
import numpy as np
from scipy.optimize import minimize

_SLQ_ALPHA = 1.0

def _repair(h, target=None):
    h = np.clip(np.asarray(h, dtype=float).copy(), 0.0, 1.0)
    if target is None:
        target = 0.5 * h.size
    diff = target - h.sum()
    if abs(diff) <= 1e-15:
        return h
    order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
    for idx in order:
        if diff > 0:
            add = min(diff, 1.0 - h[idx])
            h[idx] += add
            diff -= add
        else:
            sub = min(-diff, h[idx])
            h[idx] -= sub
            diff += sub
        if abs(diff) < 1e-15:
            break
    return np.clip(h, 0.0, 1.0)

def _eval(h):
    n = int(h.size)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max()) * (2.0 / n), c

def _pair_update(h, corr, p, q, dp, dq, nm1, dx):
    op, oq = h[p], h[q]
    if p < q:
        a, b, da, db, oa, ob = p, q, dp, dq, op, oq
    else:
        a, b, da, db, oa, ob = q, p, dq, dp, oq, op
    c0 = corr[nm1]
    corr[nm1] = c0 + dx * ((1.0 - 2.0 * oa) * da - da * da +
                           (1.0 - 2.0 * ob) * db - db * db)
    changed = [nm1]
    oldv = [c0]
    if p != q:
        r = b - a
        ip, im = nm1 + r, nm1 - r
        v1, v2 = corr[ip], corr[im]
        corr[ip] = v1 + dx * (db * (1.0 - oa) - ob * da - db * da)
        corr[im] = v2 + dx * (da * (1.0 - ob) - oa * db - db * da)
        changed += [ip, im]
        oldv += [v1, v2]
    h[p] = op + dp
    h[q] = oq + dq
    return changed, oldv, op, oq

def _soft_val(h, alpha):
    n = int(h.size)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = float(c.max())
    z = alpha * (c - m)
    z -= z.max()
    return m + np.log(np.exp(z).sum()) / alpha

def _soft_grad(h, alpha):
    n = int(h.size)
    dx = 2.0 / n
    nm1 = n - 1
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = float(c.max())
    z = alpha * (c - m)
    z -= z.max()
    w = np.exp(z)
    w /= w.sum()
    g = np.zeros(n)
    eps = 1e-18
    if w[nm1] > eps:
        g += dx * w[nm1] * (1.0 - 2.0 * h)
    for r in range(1, n):
        wp = w[nm1 + r]
        wm = w[nm1 - r]
        if wp + wm <= eps:
            continue
        L = n - r
        low, up = h[:L], h[r:]
        g[:L] += dx * (wm * (1.0 - up) - wp * up)
        g[r:] += dx * (wp * (1.0 - low) - wm * low)
    return g

def _soft_fun(x):
    return _soft_val(x, _SLQ_ALPHA)

def _soft_jac(x):
    return _soft_grad(x, _SLQ_ALPHA)

def _sum_fun(x):
    return float(np.sum(x) - 0.5 * x.size)

def _sum_jac(x):
    return np.ones_like(x)

def _slsqp_full(h0, alpha, maxiter):
    h0 = _repair(h0, 0.5 * h0.size)
    if h0.size < 3:
        return h0
    global _SLQ_ALPHA
    _SLQ_ALPHA = float(alpha)
    opts = {'maxiter': int(maxiter), 'ftol': 1e-10, 'disp': False}
    try:
        res = minimize(_soft_fun, h0, jac=_soft_jac, method='SLSQP',
                       bounds=[(0.0, 1.0)] * h0.size,
                       constraints=[{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac}],
                       options=opts)
        if res is not None and getattr(res, 'x', None) is not None:
            return _repair(res.x, 0.5 * h0.size)
    except Exception:
        pass
    return h0

def _interp_to(h, m):
    h = np.asarray(h, dtype=float)
    n = h.size
    if m == n:
        return h.copy()
    x0 = np.linspace(0.0, 2.0, n, endpoint=False)
    x1 = np.linspace(0.0, 2.0, m, endpoint=False)
    out = np.interp(x1, x0, h, left=h[0], right=h[-1])
    return _repair(out, 0.5 * m)

def _fourier_start(n, rng):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    u = np.zeros(n)
    for m in range(1, 10):
        u += (1.0 / (m ** 0.8)) * np.sin(np.pi * m * x + rng.uniform(0.0, 6.283185307179586))
    u -= u.mean()
    s = np.abs(u).max()
    if s > 1e-12:
        u /= s
    return _repair(0.5 + 0.45 * u, 0.5 * n)

def _binarize(h, t):
    b = np.where(h > t, 1.0, 0.0)
    target = int(round(0.5 * b.size))
    cnt = int(b.sum())
    if cnt > target:
        idx = np.where(b == 1.0)[0]
        idx = idx[np.argsort(h[idx])]
        for i in idx[:cnt - target]:
            b[i] = 0.0
    elif cnt < target:
        idx = np.where(b == 0.0)[0]
        idx = idx[np.argsort(-h[idx])]
        for i in idx[:target - cnt]:
            b[i] = 1.0
    return b
