import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _grad_single(h, k, dx):
    """Gradient of overlap at single shift k (correlate index)."""
    n = len(h)
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        g[:n - sh] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:-sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:-sp])
        g[:n - sp] -= h[sp:]
    return g * dx

def _grad_topk(h, topk, dx):

    c = _correl(h)

def _polish_pair(h0, deadline, rng, scale=5e-7):

def _smax_obj(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    s -= s.max()
    return m + (s.max() + np.log(np.sum(np.exp(s)))) / alpha

def _smax_grad(h, alpha, topk):
    n = len(h)
    dx = 2.0 / n
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        idx = np.arange(len(c)); ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    s -= s.max()
    w = np.exp(s); w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-14:
            continue
        g += w[ii] * _grad_single(h, idx[ii], dx)
    return g, m

def _slsqp(h0, alpha, topk, maxit, deadline):
    n = len(h0)
    if time.time() > deadline:
        return h0
    try:
        r = minimize(_smax_obj, h0, args=(alpha, topk),
                     jac=lambda x: _smax_grad(x, alpha, topk)[0],

                     options={'maxiter': maxit, 'ftol': 1e-15})
        return _repair(r.x, n)
    except:
        return h0

import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _g_single(h, k, dx):
    n = len(h)
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        g[:n - sh] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:-sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:-sp])
        g[:n - sp] -= h[sp:]
    return g * dx

def _grad_topk(h, topk, dx, beta_w=80.0):
    n = len(h)
    c = _correl(h)
    m = c.max()
    idx = np.argpartition(c, -topk)[-topk:] if topk < len(c) else np.arange(len(c))
    ct = c[idx]
    s = beta_w * (ct - m); s -= s.max()
    w = np.exp(s); w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-13:
            continue
        g += w[ii] * _g_single(h, idx[ii], dx)
    return g, m

def _grad_active(h, tol, dx):
    n = len(h)
    c = _correl(h)
    m = c.max()
    active = np.where(c >= m - tol)[0]
    g = np.zeros(n)
    for k in active:
        g += _g_single(h, k, dx)
    g /= len(active)
    return g, m

def _nesterov(h0, deadline, mode, step0, tol, topk, beta):
    n = len(h0); dx = 2.0/n; target = n*0.5
    h = h0.copy(); f = _eval(h)
    best_h, best_f = h.copy(), f
    v = np.zeros(n); step = step0; bad = 0
    it = 0
    while time.time() < deadline:
        it += 1
        if mode == 'smax':

        else:
            g, _ = _grad_active(h, tol, dx)
        hp = h.copy()

    best_h, best_c5 = h.copy(), c5

        j = (i + 1 + rng.randint(0, 8)) % n
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:

                    best_c5, best_h = c5, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c5

def _smax_obj(x, alpha, topk):
    c = _correl(x); m = c.max()
    idx = np.argpartition(c, -topk)[-topk:] if topk < len(c) else np.arange(len(c))
    s = np.clip(alpha * (c[idx] - m), -400, 400); s -= s.max()
    return m + (s.max() + np.log(np.sum(np.exp(s)))) / alpha

def _smax_jac(x, alpha, topk):
    n = len(x); dx = 2.0/n
    c = _correl(x); m = c.max()
    idx = np.argpartition(c, -topk)[-topk:] if topk < len(c) else np.arange(len(c))
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400); s -= s.max()
    w = np.exp(s); w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-13: continue
        g += w[ii] * _g_single(x, idx[ii], dx)
    return g

def _slsqp(h0, alpha, topk, maxit, deadline):
    n = len(h0)
    if time.time() > deadline: return h0
    try:
        r = minimize(_smax_obj, h0, args=(alpha, topk),
                     jac=lambda x: _smax_jac(x, alpha, topk),
                     method='SLSQP', bounds=[(0.0,1.0)]*n,
                     constraints=[{'type':'eq','fun':lambda x: x.sum()-n*0.5,
                                   'jac':lambda x: np.ones(n)}],
                     options={'maxiter':maxit,'ftol':1e-16})
        return _repair(r.x, n)
    except:
        return h0
