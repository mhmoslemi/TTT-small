            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_k(h, k, dx):
    n = len(h)
    g = np.zeros(n)
    sh = k - (n - 1)
    if sh >= 0:
        g[:n - sh] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:n - sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:n - sp])
        g[:n - sp] -= h[sp:]
    return g * dx

def _smax_obj(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    idx = np.argpartition(c, -topk)[-topk:] if topk < len(c) else np.arange(len(c))
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    sm = s.max()
    return m + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha

def _smax_grad(h, alpha, topk):
    n = len(h)
    dx = 2.0 / n
    c = _correl(h)
    m = c.max()
    idx = np.argpartition(c, -topk)[-topk:].copy() if topk < len(c) else np.arange(len(c)).copy()
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        g += w[ii] * _grad_k(h, int(idx[ii]), dx)
    return g

def _nesterov(h, alpha, topk, steps, step0, deadline):
    n = len(h)
    f = _smax_obj(h, alpha, topk)
    v = np.zeros(n)
    beta = 0.7
    step = step0
    best_h, best_f = h.copy(), _eval(h)
    for it in range(steps):
        if time.time() > deadline:
            break
        g = _smax_grad(h, alpha, topk)
        hp = h.copy()
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, n * 0.5)
        fn = _smax_obj(hn, alpha, topk)
        if fn < f:
            h = hn
            f = fn
            step = min(step * 1.03, step0 * 8)
            c5 = _eval(h)
            if c5 < best_f:
                best_f = c5
                best_h = h.copy()
        else:
            step *= 0.5
            hn = _proj(hp - step * v, n * 0.5)
            fn = _smax_obj(hn, alpha, topk)
            if fn < f:
                h = hn
                f = fn
                c5 = _eval(h)
                if c5 < best_f:
                    best_f = c5
                    best_h = h.copy()
            else:
                step *= 0.5
    return _repair(best_h, n)

def _slsqp(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    def obj(x):
        return _smax_obj(x, alpha, topk)
    def jac(x):
        return _smax_grad(x, alpha, topk)
    def cons_fun(x):
        return x.sum() - n * 0.5
    def cons_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        r = minimize(obj, h, jac=jac, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-20})
        return _repair(r.x, n)
    except:
        return _repair(h, n)

def _coord2(h, deadline, iters):
    n = len(h)
    c5c = _eval(h)
    bh, bc = h.copy(), c5c
    rng = np.random.RandomState(123)
    for _ in range(iters):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 30)) % n
        if i == j:
            continue
        s = rng.uniform(1e-6, 2e-4) * (2.0 if rng.random() < 0.5 else -1.0)
        ni, nj = h[i] + s, h[j] - s
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < bc:
                    bc = c5n
                    bh = h.copy()
            else:
                h[i], h[j] = ni - s, nj + s
    return bh, bc

def _anneal(h, niter, rng, deadline, blocks):
    n = len(h)
    T0, T1 = 0.003, 1e-10
    c5c = _eval(h)
    bh, bc = h.copy(), c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0+bs].copy()
        d = 0.002 * max(T, 1e-9)**0.5 * rng.randn(bs)
        d -= d.mean()
        seg = np.clip(seg + d, 0, 1)
        hn = h.copy()

        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n <= c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h = hn
            c5c = c5n
            if c5n < bc:
                bc = c5c
                bh = h.copy()
    return bh, bc

def _smooth(h, sigma):
    n = len(h)
    if sigma < 0.3:
        return h
    r = max(1, int(np.ceil(sigma * 3)))
    k = np.exp(-0.5 * (np.arange(-r, r + 1) / sigma) ** 2)
    k /= k.sum()
    hp = np.concatenate([h[-r:], h, h[:r]])
    out = np.array([np.dot(hp[i:i + 2*r + 1], k) for i in range(n)])
    return out

    t = n / 2.0

        return np.full(n, 0.5)

def _eval(h):
    return float(np.max(np.correlate(h, 1.0 - h, 'full')) * 2.0 / len(h))

def _correl(h):
    return np.correlate(h, 1.0 - h, 'full') * (2.0 / len(h))

def _proj(h, t):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        m = (lo + hi) * 0.5
        if np.clip(h - m, 0, 1).sum() > t: lo = m
        else: hi = m
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_k(h, k, dx):

    if sh >= 0:
        g[:n - sh] += 1.0 - h[sh:]
        if sh: g[sh:] -= h[:n - sh]
    else:
        sp = -sh; g[sp:] += 1.0 - h[:n - sp]; g[:n - sp] -= h[sp:]
    return g * dx

def _so(h, a, tk):
    c = _correl(h); m = c.max()
    idx = np.argpartition(c, -tk)[-tk:] if tk < len(c) else np.arange(len(c))
    ct = c[idx]; s = np.clip(a * (ct - m), -400, 400)
    sm = s.max()
    return m + (sm + np.log(np.sum(np.exp(s - sm)))) / a

def _sg(h, a, tk):
    n = len(h); dx = 2.0 / n; c = _correl(h); m = c.max()
    idx = np.argpartition(c, -tk)[-tk:].copy() if tk < len(c) else np.arange(len(c)).copy()
    ct = c[idx]; s = np.clip(a * (ct - m), -400, 400)
    w = np.exp(s - s.max()); w /= w.sum(); g = np.zeros(n)
    for i in range(len(idx)): g += w[i] * _grad_k(h, int(idx[i]), dx)
    return g

def _nest(h, a, tk, st, ns, dl):

    bh, bf = h.copy(), _eval(h)
    for _ in range(ns):
        if time.time() > dl: break
        g = _sg(h, a, tk); hp = h.copy()
        v = b * v + (1 - b) * g
        hn = _proj(h - stp * v, n * 0.5); fn = _so(hn, a, tk)
        if fn < f:

            c = _eval(h)

                c = _eval(h)
                if c < bf: bf, bh = c, h.copy()
            else: stp *= 0.5

def _o_fn(x): return _so(x, _g['a'], _g['tk'])
def _j_fn(x): return _sg(x, _g['a'], _g['tk'])
def _cf(x): return x.sum() - _g['n'] * 0.5
def _cj(x): return np.ones(_g['n'])

def _slsqp(h, a, tk, mi, dl):
    n = len(h)
    if time.time() > dl: return h
    _g['a'], _g['tk'], _g['n'] = a, tk, n
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
    try:
        r = minimize(_o_fn, h, jac=_j_fn, method='SLSQP',
                     bounds=[(0.0, 1.0)]*n, constraints=cons,
                     options={'maxiter': mi, 'ftol': 1e-20})
        return _repair(r.x, n)
    except: return _repair(h, n)

import numpy as np
from scipy.optimize import minimize
import time

_G = {}

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - t) > 1e-9:
        h *= t / s
        h = np.clip(h, 0, 1)
        d = t - h.sum()
        if abs(d) > 1e-12:
            o = np.argsort(1 - h) if d > 0 else np.argsort(h)
            for i in o:
                if d > 0:
                    a = min(d, 1 - h[i]); h[i] += a; d -= a
                else:
                    a = min(-d, h[i]); h[i] -= a; d += a
                if abs(d) < 1e-12: break
    return h

def _eval(h):
    return float(np.max(np.correlate(h, 1.0 - h, 'full')) * 2.0 / len(h))

def _correl(h):
    return np.correlate(h, 1.0 - h, 'full') * (2.0 / len(h))

def _proj(h, t):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        m = (lo + hi) * 0.5
        lo, hi = (m, hi) if np.clip(h - m, 0, 1).sum() > t else (lo, m)
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_k(h, k, dx):
    n = len(h); s = k - (n - 1); g = np.zeros(n)
    if s >= 0:
        g[:n-s] += 1.0 - h[s:]
        g[s:] -= h[:n-s]
    else:
        p = -s; g[p:] += 1.0 - h[:n-p]; g[:n-p] -= h[p:]
    return g * dx

def _so(h, a, tk):
    c = _correl(h); m = c.max()
    idx = np.argpartition(c, -tk)[-tk:] if tk < len(c) else np.arange(len(c))
    ct = c[idx]; s = np.clip(a * (ct - m), -400, 400); sm = s.max()
    return m + (sm + np.log(np.sum(np.exp(s - sm)))) / a

def _sg(h, a, tk):
    n = len(h); dx = 2.0 / n; c = _correl(h); m = c.max()
    idx = np.argpartition(c, -tk)[-tk:].copy() if tk < len(c) else np.arange(len(c)).copy()
    ct = c[idx]; s = np.clip(a * (ct - m), -400, 400)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n)
    for i in range(len(idx)):
        g += w[i] * _grad_k(h, int(idx[i]), dx)
    return g

def _nest(h, a, tk, st, ns, dl):

    bh, bf = h.copy(), _eval(h)

        g = _sg(h, a, tk); hp = h.copy()
        v = 0.7 * v + 0.3 * g
        hn = _proj(h - stp * v, n * 0.5); fn = _so(hn, a, tk)
        if fn < f:
            h, f = hn, fn; stp = min(stp * 1.03, st * 8)
            c = _eval(h)
            if c < bf: bf, bh = c, h.copy()
        else:
            stp *= 0.5
            hn = _proj(hp - stp * v, n * 0.5); fn = _so(hn, a, tk)
            if fn < f:

                c = _eval(h)

def _o(x): return _so(x, _G['a'], _G['tk'])
def _j(x): return _sg(x, _G['a'], _G['tk'])
def _cf(x): return x.sum() - _G['n'] * 0.5
def _cj(x): return np.ones(_G['n'])

def _slsqp(h, a, tk, mi, dl):

    if time.time() > dl: return h

    try:

                     bounds=[(0.0, 1.0)] * n,

                     options={'maxiter': mi, 'ftol': 1e-20})

    except:

def _coord(h, dl, it, sd):

                if c < bc: bc, bh = c, h.copy()
                stp = min(stp * 1.04, 3e-3)
            else:
                h[i], h[j] = ni - s, nj + s
                stp = max(stp * 0.96, 5e-7)

def _anneal(h, ni, rng, dl, bl):

    T0 = 3e-3
    for it in range(ni):

        T = T0 * (1e-10 / T0) ** (it / max(ni, 1))

        sg = h[i0:i0 + bs].copy()

        hn = _repair(hn, n); c = _eval(hn)

            h, c5 = hn, c
            if c < bc: bc, bh = c5, h.copy()
    return bh, bc

def _smoo(h, sig):
    n = len(h)
    if sig < 0.3: return h
    r = max(1, int(np.ceil(sig * 3)))
    k = np.exp(-0.5 * (np.arange(-r, r + 1) / sig) ** 2); k /= k.sum()
    hp = np.concatenate([h[-r:], h, h[:r]])
    return np.array([np.dot(hp[i:i + 2 * r + 1], k) for i in range(n)])

    return _repair(np.clip(hs, 0, 1), n)
