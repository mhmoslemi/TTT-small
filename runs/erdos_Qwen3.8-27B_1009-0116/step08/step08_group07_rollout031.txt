def _fft_correl(h):
    """Compute full correlation h * (1-h) using FFT, equivalent to np.correlate(h, 1-h, 'full') * dx"""
    n = len(h)
    size = 2 * n
    H = fft(h, size)
    O = fft(1.0 - h, size)
    corr = np.real(ifft(np.conj(H) * O))
    dx = 2.0 / n
    return corr[:2*n-1] * dx

import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _grad_max(h, tol):
    """Corrected subgradient of max_k c_k."""
    n = len(h)
    c = _correl(h)
    m = c.max()
    active = np.where(c >= m - tol)[0]
    dx = 2.0 / n
    g = np.zeros(n)
    for k in active:
        s = k - (n - 1)
        if s >= 0:
            g[s:] += (1.0 - h[:n-s])
            g[:n-s] -= h[s:]
        else:
            s2 = -s
            g[:n-s2] += (1.0 - h[s2:])
            g[s2:] -= h[:n-s2]
    g *= dx / len(active)
    return g, m

def _grad_smax(h, alpha, topk):
    """Gradient of soft-max approximation."""
    n = len(h)
    c = _correl(h)
    m = c.max()
    nc = len(c)
    if topk < nc:
        idx = np.argpartition(c, -topk)[-topk:].copy()
        ct = c[idx]
    else:
        idx = np.arange(nc); ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max()); w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-13:
            continue
        k = idx[ii]; wk = w[ii] * dx; sh = k - (n - 1)
        if sh >= 0:
            g[sh:] += wk * (1.0 - h[:n-sh])
            g[:n-sh] -= wk * h[sh:]
        else:
            sp = -sh
            g[:n-sp] += wk * (1.0 - h[sp:])
            g[sp:] -= wk * h[:n-sp]
    return g, m

def _nesterov(h, mode, steps, step0, deadline, **kw):
    n = len(h)
    target = n * 0.5
    f = _eval(h)
    v = np.zeros(n)
    beta = kw.get('beta', 0.85)
    step = step0
    best_h, best_f = h.copy(), f
    for _ in range(steps):
        if time.time() > deadline:
            break
        if mode == 'exact':
            g, _ = _grad_max(h, kw.get('tol', 0.003))
        else:
            g, _ = _grad_smax(h, kw.get('alpha', 30000), kw.get('topk', 20))
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, target)
        fn = _eval(hn)
        if fn < f - 1e-15:
            h, f = hn, fn
            step = min(step * 1.015, step0 * 15.0)
            if f < best_f:
                best_f, best_h = f, h.copy()
        else:
            step *= 0.5
            if step < 1e-10:
                step = step0
            h = _proj(h - step * v, target)
            f = _eval(h)
            if f < best_f:
                best_f, best_h = f, h.copy()
    return _repair(best_h, n), best_f

def _smax_obj(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    nc = len(c)
    if topk < nc:
        idx = np.argpartition(c, -topk)[-topk:]; ct = c[idx]
    else:
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _slsqp_refine(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    try:
        r = minimize(lambda x: _smax_obj(x, alpha, topk), _repair(h, n),
                     jac=lambda x: _grad_smax(x, alpha, topk)[0],
                     method='SLSQP', bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n*0.5,
                                   'jac': lambda x: np.ones(n)}],
                     options={'maxiter': maxit, 'ftol': 1e-18})
        return _repair(r.x, n)
    except Exception:
        return h
