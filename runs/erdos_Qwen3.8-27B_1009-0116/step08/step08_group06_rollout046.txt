import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if h.sum() < 1e-12:
        return np.full(n, 0.5)
    if abs(h.sum() - target) > 1e-10:
        lo, hi = -3.0, 3.0
        for _ in range(60):
            mid = (lo + hi) * 0.5
            if np.clip(h - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo + hi) * 0.5, 0, 1)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _topk_grad(h, k_top):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    top_k = np.argsort(c)[-k_top:]
    g = np.zeros(n)
    dx2 = 2.0 / n
    wk = dx2 / k_top
    for k in top_k:
        sh = (n - 1) - k
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]
    return g, c

def _adap_tophk(h, k_top, step0, t_end, n_iter_max=10000):
    n = len(h)
    f = _eval(h)
    step = step0
    for _ in range(n_iter_max):
        if time.time() > t_end:
            break
        g, _ = _topk_grad(h, k_top)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g = g / gn
        improved = False
        for _bt in range(25):
            h_new = _repair(h - step * g, n)
            f_new = _eval(h_new)
            if f_new < f:
                h, f = h_new, f_new
                step = min(step * 1.4, step0 * 8.0)
                improved = True
                break
            step *= 0.5
        if not improved and step < 1e-14:
            break
    return h, f

def _nbfgb_step(h, alpha, maxiter, t_end):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    pen = 5000.0

    def obj_fn(x):
        c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        lse = m + np.log(np.sum(np.exp(s - s.max()))) / alpha
        return lse + pen * (x.sum() - n / 2.0) ** 2

    def grad_fn(x):
        c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(n)
        dx2 = 2.0 / n
        for k in range(2 * n - 1):
            if w[k] < 1e-14:
                continue
            sh = (n - 1) - k
            wk = w[k] * dx2
            if sh >= 0:
                g[:n-sh] += wk * (1.0 - x[sh:])
                if sh > 0:
                    g[sh:] -= wk * x[:n-sh]
            else:
                sp = -sh
                g[sp:] += wk * (1.0 - x[:n-sp])
                g[:n-sp] -= wk * x[sp:]
        g += 2.0 * pen * (x.sum() - n / 2.0)
        return g

    try:
        res = minimize(obj_fn, h, jac=grad_fn, method='L-BFGS-B',
                       bounds=bnds, options={'maxiter': maxiter, 'gtol': 1e-11, 'maxls': 20, 'maxfun': maxiter * 4})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _coord_search(h, n_tries, rng, t_end, scale):
    n = len(h)
    f = _eval(h)
    best_h, best_f = h.copy(), f
    for _it in range(n_tries):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        r = rng.random()
        if r < 0.6:
            j = (i + 1 + rng.randint(0, 3)) % n
            d = scale * rng.randn()
            hi, hj = h[i] + d, h[j] - d
            if 0 <= hi <= 1 and 0 <= hj <= 1:
                h[i], h[j] = hi, hj
                f_new = _eval(h)
                if f_new < f:
                    f = f_new
                    if f < best_f:
                        best_f, best_h = f, h.copy()
                else:
                    h[i], h[j] = hi - d, hj + d
        else:
            j = (i + rng.randint(2, 6)) % n
            kk = (i + rng.randint(7, 14)) % n
            d1 = scale * 0.7 * rng.randn()
            d2 = scale * 0.7 * rng.randn()
            orig = (h[i], h[j], h[kk])
            h[i] = np.clip(h[i] + d1, 0, 1)
            h[j] = np.clip(h[j] + d2, 0, 1)
            h[kk] = orig[0] + orig[1] + orig[2] - h[i] - h[j]
            if 0 <= h[kk] <= 1:
                f_new = _eval(h)
                if f_new < f:
                    f = f_new
                    if f < best_f:
                        best_f, best_h = f, h.copy()
                else:
                    h[i], h[j], h[kk] = orig
    return best_h, best_f

def _gen_random_init(n, rng):
    h = np.clip(0.5 + 0.3 * rng.randn(n), 0, 1)
    return _repair(h, n)

def _upsample(h_coarse, n_fine):
    n_coarse = len(h_coarse)
    idx = np.linspace(0, n_coarse, n_fine, endpoint=False)
    lo = np.floor(idx).astype(int) % n_coarse
    hi = (lo + 1) % n_coarse
    frac = idx - np.floor(idx)
    return h_coarse[lo] * (1 - frac) + h_coarse[hi] * frac
