import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _smax_hg(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    n_c = len(c)
    if topk < n_c:
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        idx = np.arange(n_c)
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    sm = s.max()
    e = np.exp(s - sm)
    ess = e.sum()
    w = e / ess
    obj = m + (sm + np.log(ess)) / alpha
    dx = 2.0 / n
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-14:
            continue
        k = int(idx[ii])
        sh = (n - 1) - k
        wk = w[ii] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n - sp] -= wk * h[sp:]
    return obj, g

def _lbfgsb_opt(h, alpha, topk, maxit, deadline):
    n = len(h)
    best = _eval(h)
    # We'll do the optimization in a space that respects the sum constraint
    # by using L-BFGS-B with bounds, then repairing
    def objfun(x):
        return _smax_hg(x, alpha, topk)[0]
    def gradfun(x):
        return _smax_hg(x, alpha, topk)[1]
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(objfun, h, jac=gradfun, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxit,
                       'ftol': 1e-20, 'gtol': 1e-12, 'maxfun': maxit * 5,
                       'iprint': -1})
        hx = _repair(res.x, n)
        fv = _eval(hx)
        if fv < best:
            return hx, fv
    except Exception:
        pass
    return h, best

def _mom_gradient(h, steps, step0, dl, alpha, topk, beta):
    """Momentum gradient descent with backtracking."""
    n = len(h)
    target = n * 0.5
    f = _eval(h)
    v = np.zeros(n)
    step = step0
    best_h, best_f = h.copy(), f
    for i in range(steps):
        if time.time() > dl:
            break
        _, g = _smax_hg(h, alpha, topk)
        v = beta * v - step * g
        hn = _proj(h + v, target)
        fn = _eval(hn)
        if fn < f - 1e-16:
            h, f = hn, fn
            step = min(step * 1.01, step0 * 30.0)
            if f < best_f:
                best_f, best_h = f, h.copy()
        else:
            step *= 0.4
            v = beta * v - step * g
            h = _proj(h + v, target)
            f = _eval(h)
            if f < best_f:
                best_f, best_h = f, h.copy()
    return best_h, best_f

def _pair_polish(h, rounds, rng, dl, scale):
    """Polish by making small pair moves."""
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for r in range(rounds):
        if time.time() > dl:
            break
        # Pick a few random pairs per round
        n_pairs = 200
        for _ in range(n_pairs):
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 10)) % n
            d = scale * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-20:
                    c5c = c5n
                    if c5n < bc:
                        bc = c5n
                        bh = h.copy()
                else:
                    h[i], h[j] = h[i] - d, h[j] + d
    return bh, bc

def _cycle_cd(h, n_iter, dl, step):
    """Cycle coordinate descent with pair moves."""
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for it in range(n_iter):
        if time.time() > dl:
            break
        i = it % n
        j = (i + 1) % n  # pair with next coordinate
        # Try moving mass from j to i and vice versa
        for direction in [1, -1]:
            d = direction * step
            ni, nj = h[i] + d, h[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-20:
                    c5c = c5n
                    if c5n < bc:
                        bc = c5n
                        bh = h.copy()
                else:
                    h[i], h[j] = h[i] - d, h[j] + d
    return bh, bc

import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _smax_hg(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    n_c = len(c)
    if topk < n_c:
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        idx = np.arange(n_c)
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    sm = s.max()
    e = np.exp(s - sm)
    ess = e.sum()
    w = e / ess
    obj = m + (sm + np.log(ess)) / alpha
    dx = 2.0 / n
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-14:
            continue
        k = int(idx[ii])
        sh = (n - 1) - k
        wk = w[ii] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n - sp] -= wk * h[sp:]
    return obj, g

def _lbfgsb_opt(h, alpha, topk, maxit, deadline):
    n = len(h)
    best = _eval(h)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_smax_hg0, h, jac=_smax_hg1, method='L-BFGS-B',
                       bounds=bounds, args=(alpha, topk),
                       options={'maxiter': maxit, 'ftol': 1e-20,
                                'gtol': 1e-12, 'maxfun': maxit * 5, 'iprint': -1})
        hx = _repair(res.x, n)
        fv = _eval(hx)
        if fv < best:
            return hx, fv
    except Exception:
        pass
    return h, best

def _smax_hg0(h, alpha, topk):
    return _smax_hg(h, alpha, topk)[0]

def _smax_hg1(h, alpha, topk):
    return _smax_hg(h, alpha, topk)[1]

def _mom_grad(h, maxit, step0, dl, alpha, topk, beta):
    n = len(h)
    target = n * 0.5
    f = _eval(h)
    v = np.zeros(n)
    step = step0
    best_h, best_f = h.copy(), f
    for i in range(maxit):
        if time.time() > dl:
            break
        _, g = _smax_hg(h, alpha, topk)
        v = beta * v - step * g
        hn = _proj(h + v, target)
        fn = _eval(hn)
        if fn < f - 1e-16:
            h, f = hn, fn
            step = min(step * 1.012, step0 * 40.0)
            if f < best_f:
                best_f, best_h = f, h.copy()
        else:
            step *= 0.4
            v = beta * v - step * g
            h = _proj(h + v, target)
            f = _eval(h)
            if f < best_f:
                best_f, best_h = f, h.copy()
    return best_h, best_f

def _grad_exchange(h, n_passes, dl, max_step):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for p in range(n_passes):
        if time.time() > dl:
            break
        step = max_step / (p + 1)
        _, g = _smax_hg(h, 50000, 5)
        order = np.argsort(g)
        n_swap = min(n // 2, 120)
        for s in range(n_swap):
            i = int(order[s])
            j = int(order[n - 1 - s])
            for d in [step, -step]:
                ni, nj = h[i] + d, h[j] - d
                if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                    h[i], h[j] = ni, nj
                    c5n = _eval(h)
                    if c5n < c5c - 1e-20:
                        c5c = c5n
                        if c5n < bc:
                            bc, bh = c5n, h.copy()
                    else:
                        h[i], h[j] = ni - d, nj + d
    return bh, bc

def _jitter_reopt(h, rng, dl, step0, alpha, topk, n_steps):
    n = len(h)
    c5c = _eval(h)
    d = 0.002 * rng.randn(n)
    d -= d.mean()
    hn = _repair(np.clip(h + d, 0, 1), n)
    hn2, c5n = _mom_grad(hn, n_steps, step0, dl, alpha, topk, 0.9)
    if c5n < c5c:
        return hn2, c5n
    d2 = 0.005 * rng.randn(n)
    d2 -= d2.mean()
    hn = _repair(np.clip(h + d2, 0, 1), n)
    hn2, c5n = _mom_grad(hn, n_steps, step0, dl, alpha, topk, 0.9)
    if c5n < c5c:
        return hn2, c5n
    return h, c5c

def _fine_polish(h, max_iter, rng, dl, scale):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for _ in range(max_iter):
        if time.time() > dl:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 8)) % n
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-20:
                c5c = c5n
                if c5n < bc:
                    bc, bh = c5n, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc
