        h += a * np.cos(2 * np.pi * k * x + phase)
    h += 5 * x + (1 - 5) * 0  # Add a linear ramp for asymmetry
    h = _repair(h, n)
    return h

def _ramp_init(n, frac):
    """Piecewise: ramp up then ramp down."""
    x = np.linspace(0, 2, n, endpoint=False) / 2.0  # [0,1)
    h = np.zeros(n)
    tr = 0.1  # transition width
    h[(x < frac - tr) | (x >= frac + tr)] = 1.0
    mask_lo = (x >= frac - tr) & (x < frac)
    mask_hi = (x >= frac) & (x < frac + tr)
    h[mask_lo] = (x[mask_lo] - (frac - tr)) / tr
    h[mask_hi] = 1.0 - (x[mask_hi] - frac) / tr
    return _repair(h, n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if h.sum() < 1e-12:
        return np.full(n, 0.5)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        lo, hi = 0.0, 1.0 - h.min()
        for _ in range(60):
            mid = (lo + hi) * 0.5
            if np.clip(h - (-mid), 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        shift = -(lo + hi) * 0.5
    else:
        lo, hi = h.max() - 1.0, 0.0
        for _ in range(60):
            mid = (lo + hi) * 0.5
            if np.clip(h - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        shift = (lo + hi) * 0.5
    h = np.clip(h - shift, 0, 1)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _eval(h):
    n = len(h)

def _csol(h):

def _grad_from_weights(h, w):

    g = np.zeros(n)
    dx2 = 2.0 / n
    for k in range(2 * n - 1):

            continue

def _nesterov_topk(h, k_top, steps, step0, t_end):

    f = _eval(h)
    v = np.zeros(n)

            else:

def _make_ramp_n(n, frac, tr):
