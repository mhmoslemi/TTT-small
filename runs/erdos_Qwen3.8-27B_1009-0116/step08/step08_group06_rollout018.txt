import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if h.sum() < 1e-12:
        return np.full(n, 0.5)
    if abs(h.sum() - target) > 1e-10:
        lo, hi = -3.0, 3.0
        for _ in range(70):
            mid = (lo + hi) * 0.5
            if np.clip(h - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo + hi) * 0.5, 0, 1)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _grad_topk(h, k_top):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    top_k = np.argsort(c)[-k_top:]
    g = np.zeros(n)
    wk = dx / k_top
    for k in top_k:
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

class _LSE:
    def __init__(self, alpha, n):
        self.alpha = alpha
        self.n = n
    def f(self, x):
        c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / self.n)
        m = c.max()
        s = np.clip(self.alpha * (c - m), -500, 500)
        return m + np.log(np.sum(np.exp(s - s.max()))) / self.alpha
    def g(self, x):
        c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / self.n)
        m = c.max()
        s = np.clip(self.alpha * (c - m), -500, 500)
        w = np.exp(s - s.max()); w /= w.sum()
        g = np.zeros(self.n); dx = 2.0 / self.n
        for k in range(2 * self.n - 1):
            if w[k] < 1e-15:
                continue
            sh = (self.n - 1) - k
            wk = w[k] * dx
            if sh >= 0:
                g[:self.n - sh] += wk * (1.0 - x[sh:])
                if sh > 0:
                    g[sh:] -= wk * x[:self.n - sh]
            else:
                sp = -sh
                g[sp:] += wk * (1.0 - x[:n - sp])
                g[:n - sp] -= wk * x[sp:]
        return g
    def cfun(self, x):
        return x.sum() - self.n / 2.0
    def cjac(self, x):
        return np.ones(self.n)

def _slsqp(h, alpha, maxiter, t_end):
    n = len(h)
    obj = _LSE(alpha, n)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': obj.cfun, 'jac': obj.cjac}]
    try:
        res = minimize(obj.f, h, jac=obj.g, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _nesterov(h, k_top, t_end, step0=0.008):
    n = len(h)
    f = _eval(h)
    v = np.zeros(n)
    mu = 0.75
    step = step0
    for _ in range(200000):
        if time.time() > t_end:
            break
        g = _grad_topk(h, k_top)
        gn = np.sqrt(np.sum(g * g))
        if gn < 1e-14:
            break
        h_prev = h.copy()

        v_norm = np.sqrt(np.sum(v * v))
        if v_norm < 1e-14:
            break
        # Backtracking line search
        best_step = 0.0
        for ls in range(12):
            s_try = step / (2 ** ls)
            h_try = _repair(h - s_try * v, n)
            f_try = _eval(h_try)
            if f_try < f - 1e-4 * s_try * v_norm * gn:
                best_step = s_try
                break
        if best_step > 0:
            h = h_try
            f = f_try
            step = min(step * 1.03, step0 * 5.0)
        else:
            step *= 0.4
            if step < 1e-12:
                v *= 0.5
                step = step0 * 0.3

def _local_search(h, t_end, rng, min_step=1e-7):

    h_p = h.copy()
    h_p += scale * rng.randn(n)
