                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_k(h, k, n):
    g = np.zeros(n)
    off = k - (n - 1)
    j = np.arange(n)
    i1 = j - off
    m1 = (i1 >= 0) & (i1 < n)
    g[m1] += (1.0 - h[i1[m1]])
    i2 = j + off
    m2 = (i2 >= 0) & (i2 < n)
    g[m2] -= h[i2[m2]]
    return g

def _lp_step(h, n, eps, ks):
    grads = np.array([_grad_k(h, k, n) for k in ks])
    c_obj = -grads[0]
    nc = len(ks) - 1
    A_ub = grads[1:] if nc > 0 else None
    b_ub = np.zeros(nc) if nc > 0 else None
    A_eq = np.ones((1, n))
    b_eq = np.zeros(1)
    bnds = [(-eps, eps)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bnds, method='highs', options={'presolve': True})
    if not res.success or res.x is None:
        return None
    d = res.x
    if np.max(np.abs(d)) < 1e-15:
        return None
    return d

def _active_ks(cor, thresh, m_max):
    m = cor.max()
    active = np.where(cor >= m - thresh)[0]
    if len(active) == 0:
        active = np.array([np.argmax(cor)])
    if len(active) > m_max:
        order = np.argsort(-cor[active])
        active = active[order[:m_max]]
    return active.tolist()

def _lp_descent(h, n_steps, eps, thresh, m_max, deadline, rng):
    n = len(h)
    best_h, best_c5 = h.copy(), _eval(h)
    c = h.copy()
    c5_cur = best_c5
    stall = 0
    for it in range(n_steps):
        if time.time() > deadline:
            break
        cor = _correl(c)
        ks = _active_ks(cor, thresh, m_max)
        if len(ks) < 2:
            ks = np.argsort(-cor)[:6].tolist()
        d = _lp_step(c, n, eps, ks)
        if d is None:
            stall += 1
            if stall > 40:
                break
            d = rng.randn(n)
            d -= d.mean()
            d *= eps * 0.05 / max(np.std(d), 1e-15)
        else:
            stall = 0
        step = 1.0
        improved = False
        for _ in range(16):
            h_new = _repair(c + step * d, n)
            c5_new = _eval(h_new)
            if c5_new < c5_cur - 1e-15:
                c, c5_cur = h_new, c5_new
                improved = True
                if c5_cur < best_c5:
                    best_c5, best_h = c5_cur, c.copy()
                break
            step *= 0.5
        if not improved:
            d2 = rng.randn(n)
            d2 -= d2.mean()
            sc = eps * 0.03 / max(np.std(d2), 1e-15)
            h_new = _repair(c + sc * d2, n)
            c5_new = _eval(h_new)
            if c5_new < c5_cur:
                c, c5_cur = h_new, c5_new
                if c5_cur < best_c5:
                    best_c5, best_h = c5_cur, h_new.copy()
    return best_h, best_c5

def _smax_obj(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        idx = np.arange(len(c))
        ct = c
    s = np.clip(alpha * (ct - m), -300, 300)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _smax_grad(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
        ct = c[idx]
    else:
        idx = np.arange(len(c))
        ct = c
    s = np.clip(alpha * (ct - m), -300, 300)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    for ii in range(len(idx)):
        if w[ii] < 1e-12:
            continue
        k = idx[ii]
        gk = _grad_k(h, k, n)
        g += w[ii] * gk * dx
    return g

def _nesterov(h, alpha, topk, steps, step0, deadline):
    n = len(h)
    f = _smax_obj(h, alpha, topk)
    v = np.zeros(n)
    beta = 0.85
    step = step0
    for _ in range(steps):
        if time.time() > deadline:
            break
        g = _smax_grad(h, alpha, topk)
        hp = h.copy()
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, n * 0.5)
        fn = _smax_obj(hn, alpha, topk)
        if fn < f:
            h, f = hn, fn
            step = min(step * 1.02, step0 * 8.0)
        else:
            step *= 0.5
            if step < 1e-13:
                break
            h = _proj(hp - step * v, n * 0.5)
            f = _smax_obj(h, alpha, topk)
    return _repair(h, n)

def _slsqp(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    def obj(x):
        return _smax_obj(x, alpha, topk)
    def jac(x):
        return _smax_grad(x, alpha, topk)
    def cons_fun(x):
        return x.sum() - n * 0.5
    def cons_jac(x):
        return np.ones(n)
    try:
        r = minimize(obj, h, jac=jac, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}],
                     options={'maxiter': maxit, 'ftol': 1e-20, 'maxls': 40})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _anneal(h, niter, rng, deadline, T0, T1, blocks):
    n = len(h)
    c5c = _eval(h)
    bh, bc = h.copy(), c5c
    for _ in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0)
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0 + bs].copy()
        d = 0.002 * max(T, 1e-10) * rng.randn(bs)
        d -= d.mean()
        seg = np.clip(seg + d, 0, 1)
        hn = h.copy()
        hn[i0:i0 + bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h, c5c = hn, c5n
            if c5n < bc:
                bc, bh = c5n, h.copy()
        T *= 0.9995
    return bh, bc

def _coord_scan(h, niter, rng, deadline):
    n = len(h)
    c5c = _eval(h)
    bh, bc = h.copy(), c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 4)) % n
        ss = 3e-6 if it < niter // 3 else 1e-6 if it < 2 * niter // 3 else 3e-7
        d = ss * (2 * rng.random() - 1)
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < bc:
                    bc, bh = c5n, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 4

    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _repair(h0, n)
    best_h, best_c5 = h.copy(), _eval(h)

    # Phase 1: SLSQP quick pass
    dl1 = min(deadline, t0 + 80)
    for a, tk, mi in [(2000, 30, 300), (10000, 15, 250)]:
        if time.time() > dl1:
            break
        hv = _slsqp(best_h.copy(), a, tk, mi, dl1)
        cv = _eval(hv)
        if cv < best_c5:
            best_c5, best_h = cv, hv.copy()

    # Phase 2: LP descent, medium eps
    dl2 = min(deadline, t0 + 350)
    hv, cv = _lp_descent(best_h.copy(), 50000, 0.003, 0.008, 12, dl2, rng)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()

    # Phase 3: Nesterov softmax
    dl3 = min(deadline, t0 + 500)
    for a, tk, st, ns in [(30000, 20, 0.001, 2000), (200000, 10, 0.0005, 2000)]:
        if time.time() > dl3:
            break
        hv = _nesterov(best_h.copy(), a, tk, ns, st, dl3)
        cv = _eval(hv)
        if cv < best_c5:
            best_c5, best_h = cv, hv.copy()

    # Phase 4: LP descent, tight eps, more constraints
    dl4 = min(deadline, t0 + 750)
    hv, cv = _lp_descent(best_h.copy(), 50000, 0.001, 0.004, 20, dl4, rng)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()

    # Phase 5: Very tight LP
    dl5 = min(deadline, t0 + 900)
    hv, cv = _lp_descent(best_h.copy(), 30000, 0.0005, 0.002, 25, dl5, rng)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()

    # Phase 6: Annealing
    dl6 = min(deadline, t0 + budget_s - 60)
    blocks = [1, 2, 3, 4, 5, 7, 10]
    hv, cv = _anneal(best_h.copy(), 40000, rng, dl6, 0.002, 1e-11, blocks)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()

    # Phase 7: Coordinate scan
    rem = max(0, deadline - time.time())
    npol = min(int(rem / 0.0006), 60000)
    best_h, best_c5 = _coord_scan(best_h, npol, rng, deadline)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
