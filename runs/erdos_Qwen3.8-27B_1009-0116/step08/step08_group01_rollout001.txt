import numpy as np

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _full_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _grad_lse(h, alpha, top_k):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    idx = np.argsort(c)[-top_k:]
    c_top = c[idx]
    m = c_top.max()
    s = np.clip(alpha * (c_top - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[ii] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= wk * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += wk * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= wk * h[i2 + sp]
    return g

def _proj_target(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _mgd(h, n_iter, alpha, top_k, step0, rng):
    n = len(h)
    target = n * 0.5
    v = np.zeros(n)
    mu = 0.75
    step = step0
    cur_c = _eval(h)
    for it in range(n_iter):
        g = _grad_lse(h, alpha, top_k)
        y = h + step * g
        v = mu * v + (1.0 - mu) * (y - h)
        h_try = _proj_target(y + mu * v, target)
        c_try = _eval(h_try)
        if c_try < cur_c:
            h = h_try
            cur_c = c_try
            step = min(step * 1.04, step0 * 4.0)
        else:
            step *= 0.6
            step = max(step, step0 * 0.1)
    return _repair(h, n), cur_c

    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _full_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _grad_lse(h, alpha, top_k):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    idx = np.argsort(c)[-top_k:]
    c_top = c[idx]
    m = c_top.max()
    s = np.clip(alpha * (c_top - m), -400, 400)
    s -= s.max()
    w = np.exp(s); w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[ii] * dx
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += wk * (1.0 - h[i1 + sh])
            if sh > 0:
                g[sh:] -= wk * h[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]
    return g

def _proj_target(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target: lo = mid
        else: hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _mgd(h, n_iter, alpha, top_k, step0, rng):
    n = len(h)
    target = n * 0.5
    v = np.zeros(n)
    mu = 0.7
    step = step0
    cur_c = _eval(h)
    for _ in range(n_iter):
        g = _grad_lse(h, alpha, top_k)
        y = h + step * g
        v = mu * v + (1.0 - mu) * (y - h)
        h_try = _proj_target(y + mu * v, target)
        c_try = _eval(h_try)
        if c_try < cur_c:
            h = h_try; cur_c = c_try
            step = min(step * 1.03, step0 * 5.0)
        else:
            step = max(step * 0.5, step0 * 0.05)
    return _repair(h, n), cur_c

def _fast_local(h, n_iter, step_init, rng, t_end):
    n = len(h)
    c = _full_corr(h).copy()
    cur_c = c.max()
    best_c = cur_c
    best_h = h.copy()
    step = step_init
    stagnant = 0
    count = 0
    while count < n_iter:
        if time.time() > t_end:
            break
        batch = min(50000, n_iter - count)
        for _ in range(batch):
            i = rng.randint(n)
            r = rng.rand()
            if r < 0.35:
                j = (i + 1 + rng.randint(0, 7)) % n
            elif r < 0.65:
                j = (i + 1 + rng.randint(0, 40)) % n
            else:
                j = rng.randint(n)
                if j == i: j = (j + 1) % n
            d = step * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                one_mh = 1.0 - h
                h_rev = h[::-1]
                c[n-1-i : 2*n-1-i] += d * one_mh
                c[i : n+i] -= d * h_rev
                c[n-1-j : 2*n-1-j] -= d * one_mh
                c[j : n+j] += d * h_rev
                new_max = c.max()
                if new_max < cur_c:
                    h[i], h[j] = ni, nj
                    cur_c = new_max
                    stagnant = 0
                    if cur_c < best_c:
                        best_c = cur_c
                        best_h = h.copy()
                    step = min(step * 1.005, step_init * 4.0)
                else:
                    c[n-1-i : 2*n-1-i] -= d * (1.0 - h)
                    c[i : n+i] += d * h[::-1]
                    c[n-1-j : 2*n-1-j] += d * (1.0 - h)
                    c[j : n+j] -= d * h[::-1]
                    stagnant += 1
                    if stagnant % 20000 == 0:
                        step *= 0.6
        count += batch
    return best_h, best_c, c

def _downsample(h, f):
    n = len(h); m = n // f
    return h[:m*f].reshape(m, f).mean(axis=1).copy()

def _upsample(h, f):
    n = len(h)
    out = np.empty(n * f)
    for i in range(n):
        out[i*f:(i+1)*f] = h[i]
    return out

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    def _el(): return time.time() - t0

    h_parent = _repair(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values))
    n = len(h_parent)
    best_h = h_parent.copy()
    best_c5 = _eval(best_h)

    # Phase 1: MGD from parent
    for alpha, steps, tk, s0 in [(800, 400, 8, 0.004), (5000, 600, 6, 0.003), (30000, 800, 5, 0.001)]:
        if _el() > 60: break
        h_t, c = _mgd(best_h.copy(), steps, alpha, tk, s0, rng)
        c = _eval(h_t)
        if c < best_c5:
            best_c5 = c; best_h = h_t.copy()

    # Phase 2: Coarse-to-fine
    if _el() < budget_s - 600:
        nc = n // 2
        hc = _repair(_downsample(best_h, 2), nc)
        hc, _ = _mgd(hc, 300, 5000, 6, 0.005, rng)
        h_up = _repair(_upsample(hc, 2), n)
        c_up = _eval(h_up)
        if c_up < best_c5:
            best_c5 = c_up; best_h = h_up.copy()

    # Phase 3: Main fast local search (bulk of time)
    t_end3 = t0 + budget_s - 150
    t_end3 = max(t_end3, t0 + 100)
    remaining = t_end3 - time.time()
    n_iter = int(remaining * 200000)
    n_iter = max(n_iter, 500000)
    best_h, best_c5, _ = _fast_local(best_h, n_iter, 3e-4, rng, t_end3)

    # Phase 4: MGD polish
    if _el() < budget_s - 80:
        for alpha, steps, tk, s0 in [(5000, 300, 6, 0.003), (50000, 400, 4, 0.001), (200000, 300, 3, 0.0005)]:
            if _el() > budget_s - 70: break
            h_t, c = _mgd(best_h.copy(), steps, alpha, tk, s0, rng)
            c = _eval(h_t)
            if c < best_c5:
                best_c5 = c; best_h = h_t.copy()

    # Phase 5: More fast local search with smaller step
    if _el() < budget_s - 40:
        t_end5 = t0 + budget_s - 30
        remaining5 = t_end5 - time.time()
        n_iter5 = int(remaining5 * 250000)
        n_iter5 = max(n_iter5, 200000)
        best_h, best_c5, _ = _fast_local(best_h, n_iter5, 1e-5, rng, t_end5)

    # Phase 6: Final fine-tune
    if _el() < budget_s - 3:
        t_end6 = t0 + budget_s - 2
        remaining6 = t_end6 - time.time()
        n_iter6 = int(remaining6 * 300000)
        n_iter6 = max(n_iter6, 100000)
        best_h, best_c5, _ = _fast_local(best_h, n_iter6, 3e-6, rng, t_end6)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
