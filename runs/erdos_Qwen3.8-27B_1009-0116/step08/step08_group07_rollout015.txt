import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _grad_max(h, tol):
    n = len(h)
    c = _correl(h)
    m = c.max()
    active = np.where(c >= m - tol)[0]
    dx = 2.0 / n
    g = np.zeros(n)
    for k in active:
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= h[:-sh]
        else:
            sp = -sh
            g[sp:] += (1.0 - h[:-sp])
            g[:n - sp] -= h[sp:]
    g *= dx / len(active)
    return g, m

def _nesterov(h, steps, step0, deadline, tol, beta):
    n = len(h)
    target = n * 0.5
    f = _eval(h)
    v = np.zeros(n)
    step = step0
    best_h, best_f = h.copy(), f
    for it in range(steps):
        if time.time() > deadline:
            break
        g, m = _grad_max(h, tol)
        hp = h.copy()
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, target)
        fn = _eval(hn)
        if fn < f - 1e-15:
            h, f = hn, fn
            step = min(step * 1.012, step0 * 25.0)
            if f < best_f:
                best_f, best_h = f, h.copy()
        else:
            step *= 0.45
            if step < step0 * 1e-5:
                break
            h = _proj(hp - step * v, target)
            f = _eval(h)
            if f < best_f:
                best_f, best_h = f, h.copy()
    return _repair(best_h, n), best_f

_sm_alpha = 30000.0
_sm_topk = 20

def _smax_fun(x):
    n = len(x)
    c = _correl(x)
    m = c.max()
    idx = np.argpartition(c, -_sm_topk)[-_sm_topk:]
    s = np.clip(_sm_alpha * (c[idx] - m), -400, 400)
    sm = s.max()
    return m + (sm + np.log(np.sum(np.exp(s - sm)))) / _sm_alpha

def _smax_jac(x):
    n = len(x)
    c = _correl(x)
    m = c.max()
    idx = np.argpartition(c, -_sm_topk)[-_sm_topk:].copy()
    ct = c[idx]
    s = np.clip(_sm_alpha * (ct - m), -400, 400)
    sm = s.max()
    w = np.exp(s - sm)
    ws = w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] / ws < 1e-14:
            continue
        k = idx[ii]
        sh = (n - 1) - k
        wk = w[ii] / ws * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - x[sh:])
            if sh > 0:
                g[sh:] -= wk * x[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - x[:-sp])
            g[:n - sp] -= wk * x[sp:]
    return g

def _lbfgs(h, alpha, topk, maxit, deadline):
    global _sm_alpha, _sm_topk
    _sm_alpha, _sm_topk = alpha, topk
    n = len(h)
    try:
        r = minimize(_smax_fun, h, jac=_smax_jac, method='L-BFGS-B',
                     bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': maxit, 'ftol': 1e-18, 'gtol': 1e-14,
                              'maxls': 40, 'maxfun': maxit * 50})
        return _repair(np.clip(r.x, 0, 1), n)
    except Exception:
        return _repair(h, n)

    bc = c5c
    g_cache = None

        i = top[rng.randint(len(top))]

def _anneal(h, n_iter, rng, deadline, blocks, T0, T1):

    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for it in range(n_iter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(n_iter, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0 + bs].copy()
        d = 0.005 * max(T, 1e-9) ** 0.5 * rng.randn(bs)

        seg += d
