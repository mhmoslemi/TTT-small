import numpy as np

import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_dx(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smooth_obj(x, alpha, m_lags):
    c = _corr_dx(x)
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _smooth_grad(x, alpha, m_lags):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _sum_fn(x, half_n):
    return x.sum() - half_n

def _sum_jac(x, half_n):
    return np.ones(len(x))

    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fn, 'jac': _sum_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad,
                       args=(alpha, m_lags), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h

def _nesterov_grad(h, alpha, m_lags, n_steps, step0):
    n = len(h)
    v = np.zeros(n)
    t = 1
    step = step0
    cur_c = _eval(h)
    for _ in range(n_steps):
        g = _smooth_grad(h, alpha, m_lags)
        t = t + 1
        mu = (t - 1) / (t + 2)
        v = mu * v + (1 - mu) * g  # Nesterov-like
        taken = False
        for _ts in range(8):

            c_new = _eval(h_new)
            if c_new < cur_c:
                h = h_new
                cur_c = c_new
                step = min(step * 1.05, step0 * 10.0)
                taken = True
                break
            step *= 0.5
        if not taken:
            step = min(step * 1.2, step0 * 10.0)
    return _repair(h, n)

def _grad_2point(h, n_iter, rng, alpha, m_lags):
    n = len(h)
    c5cur = _eval(h)
    best_h = h.copy()
    best_c = c5cur
    for _ in range(n_iter):
        g = _smooth_grad(h, alpha, m_lags)
        # Gradient-informed: find best (i,j) pair for paired move
        # Direction: h[i] += d, h[j] -= d, directional deriv = g[i] - g[j]
        # We want g[i] - g[j] < 0, so pick i with small g, j with large g
        i = np.argmin(g)
        j = np.argmax(g)
        improved = False
        for mult in [1e-6, 5e-7, 1e-7, 5e-8]:
            d = mult
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5cur:
                    c5cur = c5n
                    if c5n < best_c:
                        best_c = c5n
                        best_h = h.copy()
                    improved = True
                    break

                            best_h = h.copy()
                    else:

    return best_h, best_c

    n = len(h)
    m = n // f
    return _repair(h[:m * f].reshape(m, f).mean(axis=1).copy(), m)

    n = len(h)
    out = np.empty(n * f)
    for i in range(n):
        out[i * f:(i + 1) * f] = h[i]
    return _repair(out, n * f)

    best_h = h.copy()

    for _ in range(n_iter):
        g = _smooth_grad(h, alpha, m_lags)
        i = np.argmin(g)
        j = np.argmax(g)
        improved = False
        for mult in [1e-6, 5e-7, 1e-7, 5e-8, 1e-8]:
            d = mult
            if h[i] + d > 1 or h[j] - d < 0:
                continue
            h[i], h[j] = h[i] + d, h[j] - d
            c5n = _eval(h)
            if c5n < c5cur:
                c5cur = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
                improved = True
                break
            else:
                h[i], h[j] = h[i] - d, h[j] + d
        if not improved:
            i2, j2 = rng.randint(n), rng.randint(n)
            if i2 != j2:
                d = 1e-7 * rng.randn()
                ni, nj = h[i2] + d, h[j2] - d
                if 0 <= ni <= 1 and 0 <= nj <= 1:
                    h[i2], h[j2] = ni, nj
                    c5n = _eval(h)
                    if c5n < c5cur:
                        c5cur = c5n
                        if c5n < best_c:
                            best_c = c5n
                            best_h = h.copy()
                    else:
                        h[i2], h[j2] = ni - d, nj + d
    return best_h, best_c

def _downsample(h, f):
    n = len(h)
    m = n // f
    return _repair(h[:m * f].reshape(m, f).mean(axis=1).copy(), m)

def _upsample(h, f):
    n = len(h)
    out = np.empty(n * f)
    for i in range(n):
        out[i * f:(i + 1) * f] = h[i]
    return _repair(out, n * f)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    def _el():
        return time.time() - t0

    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _repair(h0, n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Coarse multi-resolution
    for ratio in [4, 8]:
        if _el() > 50:
            break
        nc = max(40, n // ratio)
        hc = _downsample(h, n // nc)
        for alpha in [500, 3000, 20000]:
            if _el() > 70:
                break
            hc = _slsqp(hc, alpha, 30, 400, 1e-14)
        h_up = _upsample(hc, n // nc)
        c5u = _eval(h_up)
        if c5u < best_c5:
            best_c5 = c5u
            best_h = h_up.copy()

    # Phase 2: SLSQP alpha dive
    for alpha in [3000, 10000, 50000, 200000, 1000000, 5000000, 50000000]:
        if _el() > 350:
            break
        lift = min(80, max(20, int(40 * np.log10(alpha / 3000) + 20)))
        h_new = _slsqp(best_h, alpha, lift, 1200, 1e-16)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new

    # Phase 3: Projected gradient with momentum
    if _el() < budget_s - 350:
        for alpha, ml, steps, s0 in [(5000, 50, 800, 0.005), (50000, 60, 600, 0.003), (500000, 80, 400, 0.001)]:
            if _el() > budget_s - 300:
                break
            h_new = _proj_grad(best_h.copy(), alpha, ml, steps, s0, 0.8)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new

    # Phase 4: Basin hopping with structured perturbations
    if _el() < budget_s - 200:
        x = np.linspace(0, 2, n)
        for freq, amp in [(3, 0.03), (7, 0.03), (13, 0.02), (5, 0.05)]:
            if _el() > budget_s - 150:
                break
            h_pert = best_h + amp * np.sin(2 * np.pi * freq * x + rng.rand() * 6.28)
            h_pert = _repair(h_pert, n)
            h_opt = _slsqp(h_pert, 100000, 50, 300, 1e-15)
            c5n = _eval(h_opt)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_opt
        # Also try random perturbation
        if _el() < budget_s - 100:
            h_pert = best_h + 0.04 * rng.randn(n)
            h_pert = _repair(h_pert, n)
            h_opt = _slsqp(h_pert, 100000, 50, 300, 1e-15)
            c5n = _eval(h_opt)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_opt

    # Phase 5: Very high alpha SLSQP
    if _el() < budget_s - 80:
        for alpha in [200000000, 1000000000]:
            if _el() > budget_s - 40:
                break
            h_new = _slsqp(best_h, alpha, 100, 500, 1e-17)
            c5n = _eval(h_new)
