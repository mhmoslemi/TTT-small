        c = _corr_full(h)
        active = np.argsort(c)[-top_k:]
        ca = c[active]; ma = ca.max()
        s = np.clip(alpha*(ca-ma),-400,400); s -= s.max()
        w = np.exp(s); w /= w.sum()
        g = np.zeros(n)
        for ii, wv in enumerate(w):
            if wv > 1e-14:
                g += wv * _g_lag(active[ii], h, n)
        v = mom*v + (1-mom)*g
        taken = False
        for _ in range(12):
            h2 = _proj(h - step*v, n*0.5)
            cn = _eval(h2)
            if cn < cur - 1e-16:
                h, cur, step = h2, cn, min(step*1.07, step0*5)
                taken = True; break
            step *= 0.4
        if not taken:
            step = min(step*1.3, step0*5)
        if it % 100 == 0: v *= 0.5
    return _repair(h, n)

def _local_worst(h, n_iter, rng, step):
    n = len(h)
    c5, kw = _eval_worst(h)
    best_h, best_c = h.copy(), c5
    for _ in range(n_iter):
        sh = (n-1) - kw
        if 0 < sh < n:
            i = rng.randint(n-sh); j = i+sh
            d = abs(step*rng.randn())
            ni, nj = h[i]-d, h[j]+d
            if 0<=ni<=1 and 0<=nj<=1:
                h[i],h[j]=ni,nj
                cn,kw=_eval_worst(h)
                if cn<c5: c5=cn
                if cn<best_c: best_c,best_h=cn,h.copy()
                else: h[i],h[j]=ni+d,nj-d
        else:
            i=rng.randint(n); j=(i+1+rng.randint(0,3))%n
            d=step*rng.randn()
            ni,nj=h[i]+d,h[j]-d
            if 0<=ni<=1 and 0<=nj<=1:
                h[i],h[j]=ni,nj
                cn,kw=_eval_worst(h)
                if cn<c5: c5=cn
                if cn<best_c: best_c,best_h=cn,h.copy()
                else: h[i],h[j]=ni-d,nj+d
    return best_h, best_c

def _local_fine(h, n_iter, rng, step):
    n = len(h)
    c5 = _eval(h)
    best_h, best_c = h.copy(), c5
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i+1+rng.randint(0,2))%n
        d = step*rng.randn()
        ni, nj = h[i]+d, h[j]-d
        if 0<=ni<=1 and 0<=nj<=1:
            h[i],h[j]=ni,nj
            cn=_eval(h)
            if cn<c5: c5=cn
            if cn<best_c: best_c,best_h=cn,h.copy()
            else: h[i],h[j]=ni-d,nj+d
    return best_h, best_c

def _binary_local(h, n_iter, rng):
    n = len(h)
    hb = (h > 0.5).astype(float)
    target = n//2
    for _ in range(200):
        ones = np.sum(hb)
        if ones > target:
            idx = np.where(hb==1)[0]
            hb[rng.choice(idx)] = 0
        elif ones < target:
            idx = np.where(hb==0)[0]
            hb[rng.choice(idx)] = 1
        else: break
    c5 = _eval(hb)
    best_h, best_c = hb.copy(), c5
    for _ in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j or hb[i] == hb[j]: continue
        hb[i], hb[j] = hb[j], hb[i]
        cn = _eval(hb)
        if cn < c5:
            c5 = cn
            if cn < best_c: best_c, best_h = cn, hb.copy()
        else:
            hb[i], hb[j] = hb[j], hb[i]
    return best_h, best_c

def _bench_evals(h, count):
    t = time.time()
    for _ in range(count): _eval(h)
    return (time.time()-t)/count

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    def _el(): return time.time()-t0

    n = len(initial_h_values)
    best_h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_c5 = _eval(best_h)

    # Phase 1: L-BFGS-B with increasing alpha (250s)
    alphas = [(400,12,300,80),(1500,10,300,80),(5000,8,250,80),
              (20000,6,200,100),(80000,4,200,100),(300000,3,150,150),
              (1000000,2,150,150)]
    for a,tk,mit,pen in alphas:
        if _el() > 250: break
        h_new = _lbfbs(best_h.copy(), n, a, tk, mit, pen)
        c5n = _eval(h_new)
        if c5n < best_c5: best_c5, best_h = c5n, h_new.copy()

    # Phase 2: Binary push + re-optimize (100s)
    if _el() < 380:
        hb = (best_h > 0.5).astype(float)
        for _ in range(300):
            ones = np.sum(hb)
            if ones > n//2:
                hb[rng.choice(np.where(hb==1)[0])] = 0
            elif ones < n//2:
                hb[rng.choice(np.where(hb==0)[0])] = 1
            else: break
        hb = _repair(hb, n)
        c5b = _eval(hb)
        if c5b < best_c5: best_c5, best_h = c5b, hb.copy()
        h_new = _lbfbs(best_h.copy(), n, 50000, 4, 200, 100)
        c5n = _eval(h_new)
        if c5n < best_c5: best_c5, best_h = c5n, h_new.copy()

    # Phase 3: Projected gradient with momentum (200s)
    pg_cfgs = [(3000,400,0.006,10,0.7),(10000,500,0.004,7,0.75),
               (40000,500,0.003,5,0.8),(100000,400,0.002,4,0.82),
               (300000,350,0.001,3,0.85),(1000000,300,0.0005,2,0.87)]
    for a,st,s0,tk,mo in pg_cfgs:
        if _el() > 580: break
        h_new = _proj_grad(best_h.copy(), n, a, st, s0, tk, mo)
        c5n = _eval(h_new)
        if c5n < best_c5: best_c5, best_h = c5n, h_new.copy()

    # Phase 4: L-BFGS-B final push (50s)
    if _el() < budget_s - 250:
        for a,tk in [(500000,2),(2000000,1)]:
            if _el() > budget_s - 220: break
            h_new = _lbfbs(best_h.copy(), n, a, tk, 100, 200)
            c5n = _eval(h_new)
            if c5n < best_c5: best_c5, best_h = c5n, h_new.copy()

    # Phase 5: Binary local search (if beneficial)
    if _el() < budget_s - 150:
        epc = _bench_evals(best_h, 20)
        ntr = min(int((budget_s - 150 - _el())*0.5/epc), 500000)
        hb, cb = _binary_local(best_h, max(ntr,500), rng)
        if cb < best_c5:
            best_c5, best_h = cb, hb.copy()
            h_new = _lbfbs(best_h.copy(), n, 20000, 5, 150, 100)
            c5n = _eval(h_new)
            if c5n < best_c5: best_c5, best_h = c5n, h_new.copy()

    # Phase 6: Informed local search (worst lag)
    if _el() < budget_s - 60:
        epc = _bench_evals(best_h, 20)
        ntr = min(int((budget_s - 60 - _el())*0.6/epc), 1500000)
        best_h, best_c5 = _local_worst(best_h, max(ntr,1000), rng, 4e-6)

    # Phase 7: Ultra-fine local search
    if _el() < budget_s - 5:
        epc = _bench_evals(best_h, 20)
        ntr = min(int((budget_s - 5 - _el())*0.7/epc), 5000000)
        best_h, best_c5 = _local_fine(best_h, max(ntr,200), rng, 8e-8)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
