```python
def _block_cd(h, n, block_size, n_rounds, alpha, top_k):
    n_blocks = n // block_size
    target = n / 2.0
    for round in range(n_rounds):
        for b in range(n_blocks):
            start = b * block_size
            end = start + block_size
            h_block = h[start:end].copy()
            fixed_sum = h.sum() - h_block.sum()
            block_target = target - fixed_sum
            
            # Define subproblem
            x0 = np.clip(h_block, 0, 1)
            
            def obj(x):
                x = np.clip(x, 0, 1)
                # Reconstruct full h
                h_full = h.copy()
                h_full[start:end] = x
                c = np.correlate(h_full, 1.0 - h_full, mode='full') * (2.0/n)
                if top_k > 0 and top_k < len(c):
                    idx = np.argsort(c)[-top_k:]
                    cs = c[idx]
                else:
                    cs = c
                m = cs.max()
                s = np.clip(alpha * (cs - m), -400, 400)
                s -= s.max()
                return m + np.log(np.exp(s).sum()) / alpha
            
            # Constraints for block
            cons = [{'type': 'eq', 'fun': lambda x: x.sum() - block_target}]
            bnds = [(0.0, 1.0)] * block_size
            
            res = minimize(obj, x0, method='SLSQP', bounds=bnds, 
                          constraints=cons, options={'maxiter': 50, 'ftol': 1e-14})
            if res.success or res.fun < obj(x0):
                h[start:end] = np.clip(res.x, 0, 1)
    return _repair(h, n)
```

            bt = target - (h.sum() - x0.sum())
            if bt < -0.5 or bt > block_size + 0.5:
                continue

            def _boj(x, _s=start, _e=end, _n=n, _h=h, _al=alpha, _tk=top_k):
                hf = _h.copy()
                hf[_s:_e] = np.clip(x, 0, 1)
                c = np.correlate(hf, 1.0 - hf, mode='full') * (2.0 / _n)
                if _tk > 0 and _tk < len(c):
                    idx = np.argsort(c)[- _tk:]
                    cs = c[idx]
                else:
                    cs = c
                m = cs.max()
                s2 = np.clip(_al * (cs - m), -400, 400)
                s2 -= s2.max()
                return m + np.log(np.exp(s2).sum()) / _al

            bnds2 = [(0.0, 1.0)] * block_size
            _con_target[0] = bt
            cons2 = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
            try:
                res = minimize(_boj, x0, method='SLSQP', bounds=bnds2,
                               constraints=cons2, options={'maxiter': 25, 'ftol': 1e-14})
                if res.fun < _boj(x0) - 1e-15:
                    h[start:end] = np.clip(res.x, 0, 1)
            except Exception:
                pass
    return _repair(h, n)

def _sa_local(h, n_iter, rng, base_step, t0_temp, t1_temp, deadline):
    n = len(h)
    c5c, k_w, _ = _eval_worst(h)
    best_h = h.copy()
    best_c = c5c
    for it in range(n_iter):
        if time.time() > deadline:
            break
        t = t0_temp * (t1_temp / t0_temp) ** (it / max(n_iter, 1))
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 4)) % n
        d = base_step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n, k_w2, _ = _eval_worst(h)
            dc = c5n - c5c
            if dc < 0:
                c5c, k_w = c5n, k_w2
                if c5n < best_c:
                    best_c, best_h = c5n, h.copy()
            elif rng.rand() < np.exp(-dc / max(t, 1e-12)):
                c5c, k_w = c5n, k_w2
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def _fine_2pt(h, n_iter, rng, base_step, deadline):
    n = len(h)
    c5c = _eval(h)
    best_h = h.copy()
    best_c = c5c
    step = base_step
    hot = 0
    for it in range(n_iter):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 4)) % n
        d = step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-16:
                c5c = c5n
                hot += 1
                if c5n < best_c:
                    best_c, best_h = c5n, h.copy()
                if hot > 80:
                    step *= 0.93
                    hot = 0
            else:
                h[i], h[j] = ni - d, nj + d
                step = max(step * 0.9995, 1e-11)
    return best_h, best_c

def _proj_grad(h, alpha, steps, step0, top_k, momentum, deadline):
    n = len(h)
    v = np.zeros(n)
    step = step0
    cur_c = _eval(h)
    for it in range(steps):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        active = np.argsort(c)[-top_k:]
        c_a = c[active]
        m_a = c_a.max()
        s = np.clip(alpha * (c_a - m_a), -400, 400)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        g = np.zeros(n)
        for ii in range(len(active)):
            if w[ii] < 1e-15:
                continue
            g += w[ii] * _grad_lag(active[ii], h, n)
        v = momentum * v + (1.0 - momentum) * g
        taken = False
        for _t in range(15):
            h_new = _proj(h - step * v, n * 0.5)
            c_new = _eval(h_new)
            if c_new < cur_c - 1e-16:
                h, cur_c = h_new, c_new
                step = min(step * 1.1, step0 * 10.0)
                taken = True
                break
            step *= 0.4
        if not taken:
            step = min(step * 1.4, step0 * 10.0)
        if it % 200 == 0:
            v *= 0.3
    return _repair(h, n), cur_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    best_h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_c5 = _eval(best_h)

    # Phase 1: SLSQP polish
    for alpha, tk, mit in [(5000, 0, 500), (20000, 8, 350), (80000, 5, 300)]:
        if time.time() - t0 > 55:
            break
        h_new = _slsqp_focus(best_h, alpha, mit, tk)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5, best_h = c5n, h_new.copy()

    # Phase 2: Block coordinate descent
    if time.time() - t0 < 55:
        deadline = t0 + 160
        n_rounds = max(1, int((deadline - time.time()) / 3.0))
        h_cd = _block_cd(best_h.copy(), n, 8, n_rounds, 30000, 5, deadline)
        c5n = _eval(h_cd)
        if c5n < best_c5:
            best_c5, best_h = c5n, h_cd.copy()

    # Phase 3: Projected gradient
    if time.time() - t0 < 160:
        deadline = t0 + 340
        for alpha, steps, s0, tk, mu in [(40000, 400, 0.004, 5, 0.75),
                                         (200000, 400, 0.002, 4, 0.8),
                                         (1000000, 300, 1e-4, 3, 0.85)]:
            h_new, c_new = _proj_grad(best_h.copy(), alpha, steps, s0, tk, mu, deadline)
            if c_new < best_c5:
                best_c5, best_h = c_new, h_new.copy()

    # Phase 4: SLSQP round 2
    if time.time() - t0 < 340:
        deadline = t0 + 420
        for alpha, tk, mit in [(300000, 3, 250), (3000000, 2, 200), (10000000, 1, 150)]:
            if time.time() > deadline:
                break
            h_new = _slsqp_focus(best_h, alpha, mit, tk)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5, best_h = c5n, h_new.copy()

    # Phase 5: Simulated annealing
    if time.time() - t0 < 420:
        deadline = t0 + budget_s - 180
        t1 = time.time()
        for _ in range(30):
            _eval(best_h)
        eps = max((time.time() - t1) / 30.0, 1e-6)
        time_left = deadline - time.time()
        ntries = min(int(time_left * (1.0 / eps) * 0.7), 5000000)
        best_h, best_c5 = _sa_local(best_h, ntries, rng, 6e-5, 4e-3, 5e-5, deadline)

    # Phase 6: Fine 2-point polish
    if time.time() < t0 + budget_s - 10:
        deadline = t0 + budget_s - 3
        t1 = time.time()
        for _ in range(30):
            _eval(best_h)
        eps = max((time.time() - t1) / 30.0, 1e-6)
        time_left = deadline - time.time()
        npol = min(int(time_left * (1.0 / eps) * 0.75), 8000000)
        best_h, best_c5 = _fine_2pt(best_h, npol, rng, 2e-6, deadline)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
