    # Find best t for h[i]+=t, h[j]-=t
    # t ∈ [max(-h[i], 0), min(1-h[i], h[j])]
    t_lo = max(-h[i], 0.0)  # h[i]+t >= 0 → t >= -h[i]
    t_hi = min(1.0 - h[i], h[j])  # h[i]+t <= 1 and h[j]-t >= 0
    if t_hi - t_lo < 1e-15:
        return 0.0
    # Golden section search for min of max_k c_k(t)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = _eval_with_shift(h, i, j, c)
    fd = _eval_with_shift(h, i, j, d)
    for _ in range(25):
        if fc < fd:
            b = d; d = c; fd = fc
            c = b - gr * (b - a)
            fc = _eval_with_shift(h, i, j, c)
        else:
            a = c; c = d; fc = fd
            d = a + gr * (b - a)
            fd = _eval_with_shift(h, i, j, d)
    return (a + b) / 2.0

def _eval_with_shift(h, i, j, t):
    h2 = h.copy()
    h2[i] += t
    h2[j] -= t
    return _eval(h2)

def _eval_shift(h, i, j, t):
    # Compute max correlation after shifting h[i]+=t, h[j]-=t
    # This is O(n) if we only update affected correlations
    # But for simplicity, just do the full correlate
    h[i] += t
    h[j] -= t
    c = _eval(h)
    h[i] -= t
    h[j] += t
    return c

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _c_all(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _eval(h):
    return float(_c_all(h).max())

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_lse(h, alpha, top_k):
    n = len(h)
    c = _c_all(h)
    idx = np.argpartition(c, -(top_k - 1))[-(top_k - 1):]
    ct = c[idx]
    m = ct.max()
    s = np.clip(alpha * (ct - m), -400, 400)
    s -= s.max()
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    for ii in range(top_k):
        if w[ii] < 1e-15:
            continue
        k = idx[ii]
        L = k - (n - 1)
        wk = w[ii]
        if L >= 0:
            g[:n - L] += wk * (1.0 - h[L:])
            if L > 0:
                g[L:] -= wk * h[:n - L]
        else:
            sp = -L
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _mgd(h, n_iter, alpha, top_k, step0, rng):
    n = len(h)
    target = n * 0.5
    v = np.zeros(n)
    mu = 0.7
    step = step0
    cur_c = _eval(h)
    for _ in range(n_iter):
        g = _grad_lse(h, alpha, top_k)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-12:
            break
        g = g / gnorm if gnorm > 1.0 else g
        y = h + step * g
        v = mu * v + (1.0 - mu) * (y - h)
        h_try = _proj(y + mu * v, target)
        c_try = _eval(h_try)
        if c_try < cur_c:
            h = h_try
            cur_c = c_try
            step = min(step * 1.04, step0 * 6.0)
        else:
            step *= 0.55
            step = max(step, step0 * 0.05)
    return _repair(h, n), cur_c

def _lag_cycle(h, n_cycle, top_k, step, rng):
    n = len(h)
    target = n * 0.5
    cur_c = _eval(h)
    s = step
    for _ in range(n_cycle):
        c = _c_all(h)
        idx = np.argpartition(c, -(top_k - 1))[-(top_k - 1):]
        idx_shuffled = idx.copy()
        rng.shuffle(idx_shuffled)
        for k in idx_shuffled:
            L = k - (n - 1)
            g = np.zeros(n)
            if L >= 0:
                g[:n - L] += (1.0 - h[L:])
                if L > 0:
                    g[L:] -= h[:n - L]
            else:
                sp = -L
                g[sp:] += (1.0 - h[:n - sp])
                g[:n - sp] -= h[sp:]
            gn = np.linalg.norm(g)
            if gn < 1e-10:
                continue
            h_try = _proj(h - s * g / gn, target)
            c_try = _eval(h_try)
            if c_try < cur_c:
                h = h_try
                cur_c = c_try
                s = min(s * 1.02, step * 8.0)
            else:
                s *= 0.5
                s = max(s, step * 0.1)
    return _repair(h, n), cur_c

def _eval_shift(h, i, j, t):
    h[i] += t
    h[j] -= t
    c = _eval(h)
    h[i] -= t
    h[j] += t
    return c

def _coord_descent(h, n_passes, rng, max_offset=6):
    n = len(h)
    cur_c = _eval(h)
    gr = (np.sqrt(5.0) + 1.0) / 2.0
    for p in range(n_passes):
        offset = 1 + p % max_offset
        order = rng.permutation(n)
        for i in order:
            j = (i + offset) % n
            if i == j:
                continue
            t_lo = max(-h[i], 0.0)
            t_hi = min(1.0 - h[i], h[j])
            if t_hi - t_lo < 1e-10:
                continue
            a, b = t_lo, t_hi
            c_pt = b - (1.0 / gr) * (b - a)
            d_pt = a + (1.0 / gr) * (b - a)
            fc = _eval_shift(h, i, j, c_pt)
            fd = _eval_shift(h, i, j, d_pt)
            for _ in range(20):
                if b - a < 1e-12:
                    break
                if fc < fd:
                    b = d_pt; d_pt = c_pt; fd = fc
                    c_pt = b - (1.0 / gr) * (b - a)
                    fc = _eval_shift(h, i, j, c_pt)
                else:
                    a = c_pt; c_pt = d_pt; fc = fd
                    d_pt = a + (1.0 / gr) * (b - a)
                    fd = _eval_shift(h, i, j, d_pt)
            t_best = (a + b) / 2.0
            if abs(t_best) > 1e-12:
                h[i] += t_best
                h[j] -= t_best
                c_new = _eval(h)
                if c_new < cur_c:
                    cur_c = c_new
                else:
                    h[i] -= t_best
                    h[j] += t_best
    return _repair(h, n), cur_c

def _local2(h, n_iter, step, rng):
    n = len(h)
    cur_c = _eval(h)
    best_c = cur_c
    best_h = h.copy()
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 12)) % n
        d = step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c_new = _eval(h)
            if c_new < cur_c:
                cur_c = c_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c
