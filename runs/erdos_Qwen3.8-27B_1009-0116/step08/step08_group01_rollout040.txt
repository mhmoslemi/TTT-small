                best_h = h_t.copy()

    # Phase 3: SLSQP with high alpha
    if _el() < budget_s - 400:
        n_pts = len(best_h)
        bnds = [(0.0, 1.0)] * n_pts
        target = n_pts * 0.5
        for alpha_val, tk, mxi in [(20000, 4, 300), (200000, 3, 250), (2000000, 2, 200)]:
            if _el() > budget_s - 350:
                break
            def _obj(x, a=alpha_val, t=top_k):
                c = _corr_vals(x)
                idx = np.argsort(c)[-t]
                ct = c[idx]
                m = ct.max()
                s = np.clip(a * (ct - m), -400, 400)
                s -= s.max()
                return m + np.log(np.exp(s).sum()) / a
            def _grad(x, a=alpha_val, t=top_k):
                return _grad_lse(x, a, t)
            cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
                     'jac': lambda x: np.ones(n_pts)}]
            try:
                res = minimize(_obj, best_h.copy(), jac=_grad, method='SLSQP',
                              bounds=bnds, constraints=cons,
                              options={'maxiter': mxi, 'ftol': 1e-16})
                h_t = _repair(res.x, n_pts)
                c = _eval(h_t)
                if c < best_c5:
                    best_c5 = c
                    best_h = h_t.copy()
            except Exception:
                pass

    # Phase 4: Perturb + re-optimize
    if _el() < budget_s - 400:
        for trial in range(3):
            if _el() > budget_s - 350:
                break
            scale = 0.015 * (trial + 1)
            noise = scale * rng.randn(n)
            noise -= noise.mean()
            h_pert = _repair(best_h + noise, n)
            h_t, c = _mgd(h_pert, 600, 50000, 3, 0.001, rng)
            c = _eval(h_t)
            if c < best_c5:
                best_c5 = c
                best_h = h_t.copy()

    # Phase 5: Downsample-upsample refinement
    if _el() < budget_s - 400:
        for ratio in [2, 4]:
            if _el() > budget_s - 350:
                break
            nc = n // ratio
            hc = _repair(best_h[:nc*ratio].reshape(nc, ratio).mean(1), nc)
            hc, _ = _mgd(hc, 200, 20000, 4, 0.008, rng)
            h_up = _repair(np.repeat(hc, ratio), n)
            h_up, c_up = _subgrad(h_up, 300, 0.001, rng)
            c_up = _eval(h_up)
            if c_up < best_c5:
                best_c5 = c_up
                best_h = h_up.copy()

    # Phase 6: Fine local search with SA
    if _el() < budget_s - 200:
        sa_temp = max(1e-6, best_c5 * 2e-5)
        nsa = min(int((budget_s - 200 - _el()) * 2000), 120000)
        best_h, best_c5 = _local2(best_h, nsa, 3e-5, rng, sa_temp)

    # Phase 7: Coordinate descent
    if _el() < budget_s - 100:
        npass = min(int((budget_s - 100 - _el()) * 40), 80)
        best_h, best_c5 = _local_coord(best_h, npass, rng)

    # Phase 8: Ultra-fine local search
    if _el() < budget_s - 50:
        nfin = min(int((budget_s - 50 - _el()) * 4000), 150000)
        best_h, best_c5 = _local2(best_h, nfin, 2e-5, rng)

    # Phase 9: Final subgradient + local
    if _el() < budget_s - 20:
        h_t, c = _subgrad(best_h.copy(), 500, 0.0003, rng)
        c = _eval(h_t)
        if c < best_c5:
            best_c5 = c
            best_h = h_t.copy()
        nfin2 = min(int((budget_s - 20 - _el()) * 5000), 100000)
        best_h, best_c5 = _local2(best_h, nfin2, 1e-5, rng)

    # Phase 10: Tiny final polish
    if _el() < budget_s - 3:
        nfin3 = min(int((budget_s - 3 - _el()) * 8000), 50000)
        best_h, best_c5 = _local2(best_h, nfin3, 5e-6, rng)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
