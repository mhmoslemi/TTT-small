        return np.full(n, 0.5)

        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _sum_fn(x, half_n):
    return x.sum() - half_n

def _sum_jac(x, half_n):
    return np.ones(len(x))

def _smooth_obj(x, alpha, m_lags):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _smooth_grad(x, alpha, m_lags):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-m_lags:]
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        wk = w[ii] * dx
        L = k - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - x[:n - L])
            g[:n - L] -= wk * x[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n - M]
    return g

def _slsqp(h, alpha, m_lags, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fn, 'jac': _sum_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_smooth_obj, h, jac=_smooth_grad,
                       args=(alpha, m_lags), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h

def _subgradient(h):
    """Compute subgradient of max correlation and return (grad, max_val, active_lags)"""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    mval = c.max()
    eps = 1e-6
    active = np.where(c >= mval - eps)[0]
    g = np.zeros(n)
    for k in active:
        L = k - (n - 1)
        if L >= 0:
            g[L:] += (1.0 - h[:n - L])
            g[:n - L] -= h[L:]
        else:
            M = -L
            g[:n - M] += (1.0 - h[M:])
            g[M:] -= h[:n - M]
    g /= len(active)
    return g, mval, active

def _grad_search(h, n_iter, rng, base_step, deadline):
    """Gradient-guided 2-point search with adaptive step"""
    n = len(h)
    best_h = h.copy()
    c5cur = _eval(best_h)
    best_c = c5cur
    step = base_step
    stall = 0
    for it in range(n_iter):
        if time.time() > deadline:
            break
        g, _, _ = _subgradient(best_h)
        improved = False
        for _ in range(12):
            i = rng.randint(n)
            # Pick j: sometimes near i (local), sometimes far
            if rng.rand() < 0.4:
                j = (i + rng.randint(1, 20) * (1 if rng.rand() < 0.5 else -1)) % n
            else:
                j = rng.randint(n)
            if i == j:
                continue
            gd = g[i] - g[j]
            if abs(gd) < 1e-12:
                continue
            d = -np.sign(gd) * step * (0.4 + 0.6 * rng.rand())
            hi = best_h[i] + d
            hj = best_h[j] - d
            if hi < 0 or hi > 1 or hj < 0 or hj > 1:
                continue
            best_h[i] = hi
            best_h[j] = hj
            c5n = _eval(best_h)
            if c5n < c5cur:
                c5cur = c5n
                step = min(step * 1.15, base_step * 5)
                improved = True
                stall = 0
            else:
                best_h[i] -= d
                best_h[j] += d
        if not improved:
            step *= 0.7
            stall += 1
            if stall > 300:
                # Random restart perturbation
                idx = rng.randint(0, n, size=5)
                d = 5e-6 * rng.randn(5)
                d -= d.mean()
                best_h[idx] += d
                best_h = _repair(best_h, n)
                c5cur = _eval(best_h)
                step = base_step * 0.5
                stall = 0
        if c5cur < best_c:
            best_c = c5cur
    return best_h, best_c
