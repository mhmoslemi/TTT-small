        else:

import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h = _proj(h, target)
    return h

def _lsm_grad(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
        ct = c[idx]
    else:
        idx = np.arange(len(c)).copy()
        ct = c
    m = ct.max()
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-14:
            continue
        k = idx[ii]
        sh = (n - 1) - k
        wk = w[ii] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _lsm_obj(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        ct = c
    m = ct.max()
    s = np.clip(alpha * (ct - m), -400, 400)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _lbfgs(h, alpha, topk, deadline):
    n = len(h)
    def obj(x):
        return _lsm_obj(x, alpha, topk)
    def grad(x):
        return _lsm_grad(x, alpha, topk)
    try:
        r = minimize(obj, h, jac=grad, method='L-BFGS-B',
                     bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': 800, 'ftol': 1e-18, 'gtol': 1e-12, 'maxfun': 3000})
        if time.time() < deadline:
            return _repair(r.x, n)
    except Exception:
        pass
    return h

def _nesterov(h, steps, step0, deadline, tol, beta, mode, alpha, topk):
    n = len(h)
    target = n * 0.5
    f = _eval(h)
    v = np.zeros(n)
    step = step0
    best_h, best_f = h.copy(), f
    for _ in range(steps):
        if time.time() > deadline:
            break
        if mode == 'exact':
            c = _correl(h)
            m = c.max()
            active = np.where(c >= m - tol)[0]
            dx = 2.0 / n
            g = np.zeros(n)
            for k in active:
                sh = (n - 1) - k
                if sh >= 0:
                    g[:n - sh] += (1.0 - h[sh:])
                    if sh > 0:
                        g[sh:] -= h[:-sh]
                else:
                    sp = -sh
                    g[sp:] += (1.0 - h[:-sp])
                    g[:n - sp] -= h[sp:]
            g *= dx / max(len(active), 1)
        else:
            g = _lsm_grad(h, alpha, topk)
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, target)
        fn = _eval(hn)
        if fn < f - 1e-15:
            h, f = hn, fn
            step = min(step * 1.03, step0 * 25.0)
            if f < best_f:
                best_f, best_h = f, h.copy()
        else:
            step *= 0.45
            if step < 1e-12:
                break
    return _repair(best_h, n), best_f

def _coord_descent(h, n_sweeps, deadline):
    n = len(h)
    c_raw = np.correlate(h, 1.0 - h, mode='full').astype(float).copy()
    best_max = c_raw.max()
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        changed = False
        for j in range(n):
            ak = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                sh = (n - 1) - k
                a = 0.0
                jp = j + sh
                jm = j - sh
                if 0 <= jp < n:
                    a += (1.0 - h[jp])
                if 0 <= jm < n:
                    a -= h[jm]
                ak[k] = a
            vold = h[j]
            best_v = vold
            best_mk = c_raw.max()
            cands = [0.0, 1.0, 0.5, vold + 0.08, vold - 0.08]
            for vt in cands:
                vt = min(1.0, max(0.0, vt))
                if abs(vt - vold) < 1e-10:
                    continue
                c_try = c_raw + (vt - vold) * ak
                mk = c_try.max()
                if mk < best_mk - 1e-12:
                    best_mk = mk
                    best_v = vt
            if best_v != vold:
                c_raw += (best_v - vold) * ak
                h[j] = best_v
                changed = True
                if c_raw.max() < best_max:
                    best_max = c_raw.max()
        if not changed:
            break
        h = _repair(h, n)
        c_raw = np.correlate(h, 1.0 - h, mode='full').astype(float).copy()
    return h, float(c_raw.max() * 2.0 / n)

def _anneal(h, niter, rng, deadline, blocks, T0, T1):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0 + bs].copy()
        d = 0.003 * max(T, 1e-9) ** 0.5 * rng.randn(bs)
        d -= d.mean()
        seg = np.clip(seg + d, 0, 1)
        hn = h.copy()
        hn[i0:i0 + bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h, c5c = hn, c5n
            if c5n < bc:
                bc, bh = c5n, h.copy()
    return bh, bc

def _polish(h, niter, rng, deadline, scale):
    nc = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for _ in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(nc)
        j = (i + 1 + rng.randint(0, 9)) % nc
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < bc:
                    bc, bh = c5n, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc
