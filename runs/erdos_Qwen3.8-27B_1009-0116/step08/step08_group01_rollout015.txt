def _downsample(h, f):
    n = len(h)
    m = n // f
    return h[:m*f].reshape(m, f).mean(axis=1).copy()

def _upsample(h, f):
    n = len(h)
    out = np.empty(n * f)
    for i in range(n):
        out[i*f:(i+1)*f] = h[i]
    return out

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_vec(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _grad_lse(h, alpha, top_k):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    idx = np.argsort(c)[-top_k:]
    c_top = c[idx]
    m = c_top.max()
    s = np.clip(alpha * (c_top - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-15:
            continue
        k = idx[ii]
        sh = (n - 1) - k
        wk = w[ii] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _proj_sum(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _mgd_bt(h, n_iter, alpha, top_k, step0, rng):
    n = len(h)
    target = n * 0.5
    v = np.zeros(n)
    y = h.copy()
    mu = 0.7
    step = step0
    cur_c = _eval(h)
    for it in range(n_iter):
        g = _grad_lse(h, alpha, top_k)
        y = h + step * g
        v = mu * v + (1.0 - mu) * (y - h)
        h_try = _proj_sum(y + mu * v, target)
        c_try = _eval(h_try)
        if c_try < cur_c - 1e-14:
            h = h_try
            cur_c = c_try
            step = min(step * 1.03, step0 * 10.0)
        else:
            step *= 0.5
            step = max(step, step0 * 0.05)
    return _repair(h, n), cur_c

def _slsqp_opt(h, alpha, maxiter, top_k):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    target = n * 0.5

    def _obj(x):
        c = _eval_vec(x)
        idx = np.argsort(c)[-top_k:]
        ct = c[idx]
        m = ct.max()
        s = np.clip(alpha * (ct - m), -400, 400)
        s -= s.max()
        return m + np.log(np.exp(s).sum()) / alpha

    def _grad(x):
        return _grad_lse(x, alpha, top_k)

    def _sum_c(x):
        return x.sum() - target

    def _sum_j(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j}]
    try:
        res = minimize(_obj, h, jac=_grad, method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return _repair(res.x, n)
    except Exception:
        return h

def _peak_local(h, n_iter, step_hover, rng):
    n = len(h)
    cur_c = _eval(h)
    best_c = cur_c
    best_h = h.copy()
    c_full = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(n_iter):
        peak_idx = int(np.argmax(c_full))
        sh = (n - 1) - peak_idx
        if sh == 0:
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 10)) % n
        elif sh > 0:
            i = rng.randint(max(1, n - sh - 1))
            j = i + sh
        else:
            sp = -sh
            i = rng.randint(sp, max(sp+1, n - 1))
            j = i - sp
        d = step_hover * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c_new = _eval(h)
            if c_new < cur_c:
                cur_c = c_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
                c_full = np.correlate(h, 1.0 - h, mode='full')
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def _rand_local(h, n_iter, step_hover, rng, sa_temp=None):

    cur_c = _eval(h)
    best_c = cur_c
    best_h = h.copy()

            h[i], h[j] = ni, nj
            c_new = _eval(h)
            accept = c_new < cur_c
            if not accept and sa_temp is not None and sa_temp > 1e-12:

                    best_c = c_new
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def _sweep_local(h, n_passes, step_hover, rng):

def _downsample(h, f):
    n = len(h)
    m = n // f
    return h[:m * f].reshape(m, f).mean(axis=1).copy()

def _upsample(h, f):
    n = len(h)
    out = np.empty(n * f)
    for i in range(n):
        out[i * f:(i + 1) * f] = h[i]
    return out
