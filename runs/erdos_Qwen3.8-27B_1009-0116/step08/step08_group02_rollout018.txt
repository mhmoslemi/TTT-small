def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_worst(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return float(c.max()), int(np.argmax(c))

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_lag(k, h, n):
    dx = 2.0 / n
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        m = n - sh
        g[:m] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:-sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:n-sp])
        g[:n-sp] -= h[sp:]
    return g * dx

def _slsqp(h, alpha, maxiter, top_k, n):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, 'full') * dx
        if 0 < top_k < len(c):
            cs = np.sort(c)[-top_k:]
        else:
            cs = c
        m = cs.max()
        s = np.clip(alpha * (cs - m), -400, 400)
        s -= s.max()
        return m + np.log(np.exp(s).sum()) / alpha
    def grad(x):
        c = np.correlate(x, 1.0 - x, 'full') * dx
        if 0 < top_k < len(c):
            idx = np.argsort(c)[-top_k:]
            cs = c[idx]
        else:
            idx = np.arange(len(c))
            cs = c
        m = cs.max()
        s = np.clip(alpha * (cs - m), -400, 400)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        g = np.zeros(n)
        for ii in range(len(idx)):
            if w[ii] > 1e-14:
                g += w[ii] * _grad_lag(int(idx[ii]), x, n)
        return g
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return _repair(res.x, n)
    except:
        return h

def _proj_grad(h, alpha, steps, step0, top_k, momentum, n):
    dx = 2.0 / n
    v = np.zeros(n)
    step = step0
    cur_c = _eval(h)
    for it in range(steps):
        c = np.correlate(h, 1.0 - h, 'full') * dx
        active = np.argsort(c)[-top_k:]
        ca = c[active]
        m = ca.max()
        s = np.clip(alpha * (ca - m), -400, 400)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        g = np.zeros(n)
        for ii in range(len(active)):
            if w[ii] > 1e-14:
                g += w[ii] * _grad_lag(int(active[ii]), h, n)
        v = momentum * v + (1.0 - momentum) * g
        taken = False
        for _ in range(12):
            h_new = _proj(h - step * v, n * 0.5)
            cn = _eval(h_new)
            if cn < cur_c - 1e-16:
                h = h_new
                cur_c = cn
                step = min(step * 1.15, step0 * 20.0)
                taken = True
                break
            step *= 0.4
        if not taken:
            step = min(step * 1.5, step0 * 20.0)
        if it % 300 == 0:
            v *= 0.3
    return _repair(h, n)

def _local_guided(h, n_iter, rng, base_step, n):

    c5c, k_w = _eval_worst(h)
    best_h = h.copy()

    # Find hot indices
    g_worst = _grad_lag(k_w, h, n)

            g_worst = _grad_lag(k_w, h, n)
            hot = np.argsort(np.abs(g_worst))[-120:]
        i = hot[rng.randint(len(hot))]
        # Pick j from hot set or random
        if rng.random() < 0.7:
            j = hot[rng.randint(len(hot))]
            if j == i:
                j = (i + 1 + rng.randint(0, 10)) % n
        else:
            j = rng.randint(n)
        if i == j:
            continue
        d = base_step * rng.randn() * (1.0 + 0.5 * rng.random())
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n, k_w = _eval_worst(h)
            if c5n < c5c:
                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def _local_fine(h, n_iter, rng, step, n):

        for frac, sf in [(0.45, 1e-5), (0.30, 4e-6), (0.25, 1.5e-6)]:
            nt = max(int(ntries * frac), 2000)
            best_h, best_c5 = _local_guided(best_h, nt, rng, sf, n)
            if _el() > budget_s - 200:
                break

    # Phase 5: SLSQP micro-polish
    if _el() < budget_s - 150:
        for alpha, tk, mi in [(2000000, 1, 100)]:
            h_new = _slsqp(best_h, alpha, mi, tk, n)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5, best_h = c5n, h_new

    # Phase 6: Ultra-fine 2-point polish
    if _el() < budget_s - 5:
        time_left = budget_s - 5 - _el()
        _t = time.time()
        for _ in range(40):
            _eval(best_h)
        eps = max((time.time() - _t) / 40.0, 1e-6)
        rate = 1.0 / eps
        npol = min(int(time_left * rate * 0.72), 10000000)
        best_h, best_c5 = _local_fine(best_h, npol, rng, 4e-8, n)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
