import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-13:
        return h
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() >= target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - 0.5 * (lo + hi), 0, 1)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _og(h, alpha, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -100, 100)
    s -= s.max()
    e = np.exp(s); Z = e.sum()
    obj = m + np.log(Z) / alpha
    w = e / Z
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14: continue
        wk = w[k] * dx
        sh = (n - 1) - k
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - h[sh:])
            if sh > 0: g[sh:] -= wk * h[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]
    return obj, g

class _SF:
    def __init__(self, alpha, n):
        self.a = alpha; self.n = n
    def __call__(self, x):
        return _og(x, self.a, self.n)[0]
    def grad(self, x):
        return _og(x, self.a, self.n)[1]

class _EC:
    def __init__(self, n, target):
        self.t = target; self.jv = np.ones(n)
    def __call__(self, x):
        return x.sum() - self.t
    def jac(self, x):
        return self.jv

def _slsqp(h, alpha, mi, ft, n, target):
    sf, ec = _SF(alpha, n), _EC(n, target)
    try:
        r = minimize(sf, h, jack=sf.grad, method='SLSQP',
                     bounds=[(0.0, 1.0)]*n,
                     constraints=[{'type': 'eq', 'fun': ec, 'jac': ec.jac}],
                     options={'maxiter': mi, 'ftol': ft})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _tk(h, kt, ns, s0, te, n):
    st, fo = s0, _eval(h)
    for _ in range(ns):
        if time.time() > te: break
        c = np.correlate(h, 1.0-h, mode='full') * (2.0/n)
        tk = np.argsort(c)[-kt:]

        g = np.zeros(n)
        for k in tk:

            if sh >= 0:

            else:
                sp = -sh
                g[sp:] += wk*(1.0-h[:n-sp])
                g[:n-sp] -= wk*h[sp:]
        g -= g.mean()
        ho = h.copy()
        ht = _repair(ho - st*g, n)
        fn = _eval(ht)
        if fn < fo:
            h, fo, st = ht, fn, min(st*1.03, s0*5.0)
        else:
            st *= 0.5
            if st < 1e-15: st = s0*0.1
    return _repair(h, n)

def _c5s(h, rng, te, n, sc, np_):
    bh, bc = h.copy(), _eval(h)
    cc, h = bc, h.copy()
    while time.time() < te:
        ix = [rng.randint(n) for _ in range(np_)]
        ix[1] = (ix[0] + rng.randint(1,4)) % n
        if np_ >= 3: ix[2] = (ix[0] + rng.randint(5,12)) % n
        ds = [sc*rng.randn() for _ in range(np_-1)]
        for i in range(np_-1): h[ix[i]] += ds[i]
        h[ix[np_-1]] -= sum(ds)
        if h.min() < -1e-10 or h.max() > 1+1e-10:
            for i in range(np_-1): h[ix[i]] -= ds[i]
            h[ix[np_-1]] += sum(ds)
            continue
        hn = _repair(h, n); cn = _eval(hn)
        if cn < cc:
            cc, h = cn, hn
            if cn < bc: bc, bh = cn, h.copy()
        else:
            for i in range(np_-1): h[ix[i]] -= ds[i]
            h[ix[np_-1]] += sum(ds)
    return bh, bc

def _upd(hn, bh, bc):
    c = _eval(hn)
    return (hn.copy(), c) if c < bc else (bh, bc)

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-13:
        return h
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() >= target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - 0.5 * (lo + hi), 0, 1)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _og(h, alpha, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.clip(alpha * (c - m), -100, 100)
    s -= s.max()
    e = np.exp(s); Z = e.sum()
    obj = m + np.log(Z) / alpha
    w = e / Z
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        wk = w[k] * dx
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return obj, g

class _SF:
    def __init__(self, alpha, n):
        self.a = alpha
        self.n = n
    def __call__(self, x):
        return _og(x, self.a, self.n)[0]
    def grad(self, x):
        return _og(x, self.a, self.n)[1]

class _EC:
    def __init__(self, n, target):
        self.t = target
        self.jv = np.ones(n)
    def __call__(self, x):
        return x.sum() - self.t
    def jac(self, x):
        return self.jv

def _slsqp(h, alpha, mi, ft, n, target):
    sf = _SF(alpha, n)
    ec = _EC(n, target)
    try:
        r = minimize(sf, h, jack=sf.grad, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': ec, 'jac': ec.jac}],
                     options={'maxiter': mi, 'ftol': ft})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _tk(h, kt, ns, s0, te, n):
    st = s0
    fo = _eval(h)
    for _ in range(ns):
        if time.time() > te:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        tk = np.argsort(c)[-kt:]
        dx = 2.0 / n
        g = np.zeros(n)
        for k in tk:
            sh = (n - 1) - k
            wk = dx / kt
            if sh >= 0:
                g[:n - sh] += wk * (1.0 - h[sh:])
                if sh > 0:
                    g[sh:] -= wk * h[:n - sh]
            else:
                sp = -sh
                g[sp:] += wk * (1.0 - h[:n - sp])
                g[:n - sp] -= wk * h[sp:]
        g -= g.mean()
        ho = h.copy()
        ht = _repair(ho - st * g, n)
        fn = _eval(ht)
        if fn < fo:
            h, fo, st = ht, fn, min(st * 1.03, s0 * 5.0)
        else:
            st *= 0.5
            if st < 1e-15:
                st = s0 * 0.1
    return _repair(h, n)

def _c5s(h, rng, te, n, sc, np_):
    bh = h.copy()
    bc = _eval(h)
    cc = bc
    h = h.copy()
    while time.time() < te:

        d2 = sc * rng.randn() if np_ >= 3 else 0.0
        if np_ >= 3:

            if 0 <= a <= 1 and 0 <= b <= 1 and 0 <= c2 <= 1:

                hn = _repair(h, n)
                cn = _eval(hn)
                if cn < cc:

                    if cn < bc:
                        bc, bh = cn, h.copy()
                else:
                    h[ix0], h[ix1], h[ix2] = a - d1 - d2, b - d2, c2 + d1 + d2
        else:

            if 0 <= a <= 1 and 0 <= b <= 1:

                if cn < cc:

                    if cn < bc:
                        bc, bh = cn, h.copy()
                else:
                    h[ix0], h[ix1] = a - d1, b + d1
    return bh, bc

    c = _eval(hn)
    if c < bc:
        return hn.copy(), c
    return bh, bc
