        if s < 1e-12:
            return np.full(n, 0.5)
        h *= target / s
        h = np.clip(h, 0.0, 1.0)

                diff -= a

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / h.sum()
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_worst(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return float(c.max()), int(np.argmax(c))

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_shift(sh, h, n):
    dx = 2.0 / n
    g = np.zeros(n)
    if sh >= 0:
        cnt = n - sh
        if cnt > 0:
            g[:cnt] += (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= h[:cnt]
    else:
        sp = -sh
        cnt = n - sp
        g[sp:] += (1.0 - h[:cnt])
        g[:cnt] -= h[sp:]
    return g * dx

def _pg_nesterov(h, alpha, steps, s0, top_k, mu):
    n = len(h)
    v = np.zeros(n)
    step = s0
    cur_c = _eval(h)
    for it in range(steps):
        c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        if top_k > 0 and top_k < len(c):
            active = np.argsort(c)[-top_k:]
        else:
            active = np.arange(len(c))
        c_a = c[active]
        m_a = c_a.max()
        s = np.clip(alpha * (c_a - m_a), -400, 400)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        g = np.zeros(n)
        for ii in range(len(active)):
            if w[ii] < 1e-14:
                continue
            sh = (n - 1) - active[ii]
            g += w[ii] * _grad_shift(sh, h, n)
        v = mu * v + (1.0 - mu) * g
        taken = False
        for _ in range(18):
            h_new = _proj(h - step * v, n * 0.5)
            c_new = _eval(h_new)
            if c_new < cur_c - 1e-17:
                h = h_new
                cur_c = c_new
                step = min(step * 1.06, s0 * 30.0)
                taken = True
                break
            step *= 0.45
        if not taken:
            step = min(step * 1.4, s0 * 30.0)
        if it % 400 == 399:
            v *= 0.3
    return _repair(h, n)

def _subgrad(h, steps, s0, rng):
    n = len(h)
    cur_c, k_w = _eval_worst(h)
    step = s0
    for it in range(steps):
        sh = (n - 1) - k_w
        g = _grad_shift(sh, h, n)
        taken = False
        for _ in range(14):
            h_new = _proj(h - step * g, n * 0.5)

def _down(h, f):
    n = len(h); m = n // f
    return h[:m*f].reshape(m, f).mean(axis=1).copy()

def _up(h, f):
    return np.repeat(h, f)

def _informed_ls(h, n_iter, rng, base_step):

    best_h = h.copy()
    best_c = _eval(h)
    c5c, k_w = _eval_worst(h)

                cnt = n + sh
                pairs = h[-sh:] * (1.0 - h[:cnt])

                i, j = worst - sh, worst
        else:
            i = rng.randint(n)

        d = base_step * rng.randn()
        ni, nj = h[i] - d, h[j] + d

            else:
                h[i], h[j] = ni + d, nj - d
        if it % 200 == 0:
