            hi = mid
    h = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _sgrad(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        idx = np.arange(len(c)); ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max()); w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-13:
            continue
        k = idx[ii]; sh = (n - 1) - k; wk = w[ii] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _sobj(h, alpha, topk, pen):
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]; ct = c[idx]
    else:
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    val = m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha
    if pen > 0:
        val += pen * (h.sum() - n * 0.5) ** 2
    return val

def _sgrad_pen(h, alpha, topk, pen):
    g = _sgrad(h, alpha, topk)
    if pen > 0:
        g += 2.0 * pen * (h.sum() - len(h) * 0.5)
    return g

def _lbfgs(h, alpha, topk, pen, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    try:
        r = minimize(_sobj, h, jac=lambda x: _sgrad_pen(x, alpha, topk, pen),
                     method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': maxit, 'ftol': 1e-18, 'gtol': 1e-12, 'maxls': 40})
        return _proj(r.x, n * 0.5)
    except Exception:
        return h

def _nesterov(h, steps, step0, deadline, alpha, topk, beta):
    n = len(h); target = n * 0.5
    f = _eval(h); v = np.zeros(n); step = step0
    best_h, best_f = h.copy(), f
    for _ in range(steps):
        if time.time() > deadline:
            break
        g = _sgrad(h, alpha, topk)
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, target)
        fn = _eval(hn)
        if fn < f - 1e-16:
            h, f = hn, fn; step = min(step * 1.02, step0 * 25.0)
            if fn < best_f:
                best_f, best_h = fn, h.copy()
        else:
            step *= 0.4
            h = _proj(h - step * v, target); f = _eval(h)
            if f < best_f:
                best_f, best_h = f, h.copy()
    return best_h, best_f

def _polish(h, niter, rng, deadline, scale):
    nc = len(h); c5c = _eval(h); bh = h.copy(); bc = c5c
    for _ in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(nc)
        j = (i + 1 + rng.randint(0, 8)) % nc
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < bc:
                    bc, bh = c5n, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc

def _anneal(h, niter, rng, deadline, T0, T1):
    n = len(h); c5c = _eval(h); bh = h.copy(); bc = c5c
    blocks = [2, 3, 4, 6, 8, 12, 16]
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0+bs].copy()
        d = 0.002 * max(T, 1e-9) ** 0.5 * rng.randn(bs)
        d -= d.mean(); seg += d; seg = np.clip(seg, 0, 1)
        hn = h.copy(); hn[i0:i0+bs] = seg; hn = _proj(hn, n * 0.5)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h, c5c = hn, c5n
            if c5n < bc:
                bc, bh = c5n, h.copy()
    return bh, bc

def _jitter(h, rng, deadline):
    n = len(h); c5c = _eval(h)
    for scale in [0.004, 0.002, 0.0008]:
        if time.time() > deadline:
            break
        d = scale * rng.randn(n); d -= d.mean()
        hn = _proj(np.clip(h + d, 0, 1), n * 0.5)
        hn2, c5n = _nesterov(hn, 600, 0.001, deadline, 100000, 4, 0.92)
        if c5n < c5c:
            return hn2, c5n
        hn3, c5n3 = _lbfgs(hn, 200000, 3, 5000, 200, deadline)
        c5n3 = _eval(hn3)
        if c5n3 < c5c:
            return hn3, c5n3
    return h, c5c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    n0 = len(initial_h_values)
    h = _proj(np.array(initial_h_values, dtype=float).copy(), n0 * 0.5)
    best_h = h.copy(); best_c5 = _eval(best_h)

    dl1 = min(deadline, t0 + 180)
    for alpha, tk, s0, bt, ns in [(8000, 25, 0.003, 0.84, 2500), (50000, 10, 0.0015, 0.88, 2500)]:
        if time.time() > dl1:
            break
        h2, c5v = _nesterov(best_h.copy(), ns, s0, dl1, alpha, tk, bt)
        if c5v < best_c5:
            best_c5, best_h = c5v, h2.copy()

    dl2 = min(deadline, t0 + 480)
    for alpha, tk, pen, mi in [(20000, 15, 200, 400), (80000, 6, 1000, 350), (300000, 3, 5000, 250)]:
        if time.time() > dl2:
            break
        h2 = _lbfgs(best_h.copy(), alpha, tk, pen, mi, dl2)
        c5v = _eval(h2)
        if c5v < best_c5:
            best_c5, best_h = c5v, h2.copy()

    dl3 = min(deadline, t0 + 640)
    for alpha, tk, s0, bt, ns in [(300000, 4, 0.0008, 0.91, 3500), (1000000, 2, 0.0004, 0.93, 3500)]:
        if time.time() > dl3:
            break
        h2, c5v = _nesterov(best_h.copy(), ns, s0, dl3, alpha, tk, bt)
        if c5v < best_c5:
            best_c5, best_h = c5v, h2.copy()

    dl4 = min(deadline, t0 + 780)
    for alpha, tk, pen, mi in [(500000, 3, 8000, 250), (2000000, 2, 20000, 200)]:
        if time.time() > dl4:
            break
        h2 = _lbfgs(best_h.copy(), alpha, tk, pen, mi, dl4)
        c5v = _eval(h2)
        if c5v < best_c5:
            best_c5, best_h = c5v, h2.copy()

    dl5 = min(deadline, t0 + 900)
    for _ in range(4):
        if time.time() > dl5:
            break
        hj, cj = _jitter(best_h.copy(), rng, dl5)
        if cj < best_c5:
            best_c5, best_h = cj, hj.copy()

    dl6 = min(deadline, t0 + 1010)
    best_h, best_c5 = _anneal(best_h.copy(), 30000, rng, dl6, 0.001, 1e-12)

    dl7 = min(deadline, t0 + budget_s - 15)
    h2 = _lbfgs(best_h.copy(), 1500000, 2, 30000, 300, dl7)
    c5v = _eval(h2)
    if c5v < best_c5:
        best_c5, best_h = c5v, h2.copy()

    rem = max(0, deadline - time.time())
    n_pol = min(int(rem / 0.002), 25000)
    best_h, best_c5 = _polish(best_h, n_pol, rng, deadline, 4e-7)

    best_h = _proj(best_h, n0 * 0.5)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n0)
