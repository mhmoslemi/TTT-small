    g = np.zeros(n)
    # C_lag = sum_i h[i] * (1-h[i-lag]) for valid i
    # dC/dh[i] = (1-h[i-lag]) [from h[i] term] - h[i+lag] [from (1-h[i]) term in h[i+lag]*(1-h[i])]

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    # Bisection projection
    lo, hi = -1.0, 1.0
    for _ in range(60):

            lo = mid
        else:
            hi = mid
    result = np.clip(h - (lo + hi) * 0.5, 0, 1)

    diff = target - result.sum()
    if abs(diff) > 1e-12:

                diff -= a

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - (lo + hi) * 0.5, 0, 1)
    diff = target - result.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - result) if diff > 0 else np.argsort(result)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - result[idx])
                result[idx] += a; diff -= a
            else:
                a = min(-diff, result[idx])
                result[idx] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return result

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _grad_smoothed(h, alpha, top_k):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(c)[-top_k:]
    c_top = c[idx]
    m = c_top.max()
    s = np.clip(alpha * (c_top - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        lag = k - (n - 1)
        wk = w[ii]
        if lag >= 0:
            if lag == 0:
                g[:] += wk * (1.0 - h)
                g[:] -= wk * h
            else:
                g[lag:] += wk * (1.0 - h[:n - lag])
                g[:n - lag] -= wk * h[lag:]
        else:
            ll = -lag
            g[:n - ll] += wk * (1.0 - h[ll:])
            g[ll:] -= wk * h[:n - ll]
    return g * dx

def _mgd(h_init, n_iter, alpha, top_k, step0, rng):
    n = len(h_init)
    h = h_init.copy()
    v = np.zeros(n)
    mu = 0.7
    step = step0
    cur_c = _eval(h)
    for _ in range(n_iter):
        g = _grad_smoothed(h, alpha, top_k)
        h_new = h - step * g
        v = mu * v - step * g
        h_try = _repair(h_new + 0.1 * v, n)
        c_try = _eval(h_try)
        if c_try < cur_c:
            h = h_try
            cur_c = c_try
            step = min(step * 1.03, step0 * 8.0)
        else:
            step *= 0.7
            step = max(step, step0 * 0.05)
    return h, cur_c

def _lag_grad(h, lag, n):
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        g[lag:] += (1.0 - h[:n - lag])
        g[:n - lag] -= h[lag:]
    else:
        ll = -lag
        g[:n - ll] += (1.0 - h[ll:])
        g[ll:] -= h[:n - ll]
    return g

def _local_search(h_init, n_iter, base_step, rng, sa_temp=0.0):
    n = len(h_init)
    h = h_init.copy()
    cur_c = _eval(h)
    best_c = cur_c
    best_h = h.copy()
    step = base_step
    no_improve = 0
    for it in range(n_iter):
        # Periodic annealing bump
        if sa_temp > 0 and it % 20000 == 0 and no_improve > 5000:
            step = base_step * 3.0
            no_improve = 0
        d = step * rng.randn()
        # Correlation-aware selection (25% of iterations)
        if rng.rand() < 0.25 and it > 500:
            c_vec = np.correlate(h, 1.0 - h, mode='full')
            top_idx = np.argsort(c_vec)[-4:]
            k = top_idx[rng.randint(len(top_idx))]
            lag = k - (n - 1)
            g_k = _lag_grad(h, lag, n)
            neg = np.where(g_k < -0.02)[0]
            pos = np.where(g_k > 0.02)[0]
            if len(neg) > 0 and len(pos) > 0:
                i = neg[rng.randint(len(neg))]
                j = pos[rng.randint(len(pos))]
            else:
                i = rng.randint(n)
                j = (i + 1 + rng.randint(0, n // 50)) % n
        else:
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, n // 40)) % n
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c_new = _eval(h)
            accept = c_new < cur_c
            if not accept and sa_temp > 1e-12:
                if rng.rand() < np.exp(-min(50.0, (c_new - cur_c) / sa_temp)):
                    accept = True
            if accept:
                if c_new < cur_c:
                    no_improve = 0
                else:
                    no_improve += 1
                cur_c = c_new
                step = min(step * 1.01, base_step * 5.0)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
                step = max(step * 0.98, base_step * 0.3)
                no_improve += 1
    return best_h, best_c

def _micro_polish(h_init, n_iter, base_step, rng):
    n = len(h_init)
    h = h_init.copy()
    cur_c = _eval(h)
    best_c = cur_c
    best_h = h.copy()
    offsets = np.array([1, -1, 2, -2, 3, -3, 5, -5])
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + offsets[rng.randint(len(offsets))]) % n
        d = base_step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c_new = _eval(h)
            if c_new < cur_c:
                cur_c = c_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def _sweep_local(h_init, n_passes, base_step, rng):
    n = len(h_init)
    h = h_init.copy()
    cur_c = _eval(h)
    best_c = cur_c
    best_h = h.copy()
    for p in range(n_passes):
        order = rng.permutation(n)
        for i in order:
            j = (i + 1 + rng.randint(0, 4)) % n
            d = base_step * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h[i], h[j] = ni, nj
                c_new = _eval(h)
                if c_new < cur_c:
                    cur_c = c_new
                    if c_new < best_c:
                        best_c = c_new
                        best_h = h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    def _el():
        return time.time() - t0

    n = len(initial_h_values)
