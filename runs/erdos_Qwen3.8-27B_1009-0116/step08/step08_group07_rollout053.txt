import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _grad_smax(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
        ct = c[idx]
    else:
        idx = np.arange(len(c)); ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max()); w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-13:
            continue
        k = idx[ii]; sh = (n - 1) - k; wk = w[ii] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:

        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n - sp] -= wk * h[sp:]
    return g, m

def _grad_exact(h, topk):

    m = c.max()
    tol = 0.001

        else:
            sp = -sh
            g[sp:] += (1.0 - h[:-sp])
            g[:n - sp] -= h[sp:]
    g *= dx / len(active)
    return g, m

def _nesterov_smax(h, alphas, steps_each, step0, deadline, topk=8, beta_start=0.82, beta_end=0.93):

def _anneal(h, niter, rng, deadline, blocks, T0, T1):

    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for it in range(niter):

            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))

        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0+bs].copy()
        d = 0.002 * max(T, 1e-9) ** 0.5 * rng.randn(bs)
        d -= d.mean()
        seg += d
        seg = np.clip(seg, 0, 1)
        hn = h.copy()
        hn[i0:i0+bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h, c5c = hn, c5n
            if c5n < bc:

def _polish(h, niter, rng, deadline, scale):
    nc = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for _ in range(niter):

            break
        i = rng.randint(nc)

        ni = h[i] + d
        nj = h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:

                if c5n < bc:
                    bc, bh = c5n, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc

def _slsqp(h, alpha, topk, maxit, deadline):

    if time.time() > deadline:
        return h
    def obj(x):
        c = _correl(x)
        m = c.max()
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
        s = np.clip(alpha * (ct - m), -400, 400)
        return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha
    def jac(x):
        return _grad_smax(x, alpha, topk)[0]
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n*0.5, 'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(obj, h, jac=jac, method='SLSQP', bounds=[(0.0,1.0)]*n,
                     constraints=cons, options={'maxiter': maxit, 'ftol': 1e-16})
        return _repair(r.x, n)
    except:
        return h

def _nesterov_smax(h, alphas, steps_each, step0, deadline, topk=10, b0=0.82, b1=0.94):

    for ia, alpha in enumerate(alphas):

def _cyclic_refine(h, deadline, n_top=6, n_cycles=40):
    n = len(h)
    tgt = 0.5 * n
    best_h, best_f = h.copy(), _eval(h)
    h = h.copy()
    for _cy in range(n_cycles):
        if time.time() > deadline:
            break
        c = _correl(h)
        top = np.argsort(c)[-n_top:]
        for k in top:
            sh = (n - 1) - int(k)
            d = np.zeros(n)
            if sh >= 0:
                d[:n - sh] -= (1.0 - h[sh:])
                if sh > 0:
                    d[sh:] += h[:n - sh]
            else:
                sp = -sh
                d[sp:] -= (1.0 - h[:n - sp])
                d[:n - sp] += h[sp:]
            d -= d.mean()
            nd = np.linalg.norm(d)
            if nd < 1e-10:

            f0 = _eval(h)

def _anneal(h, niter, rng, deadline, blocks, T0, T1):
    n = len(h)
    c5c = _eval(h)
    bh, bc = h.copy(), c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = int(blocks[rng.randint(len(blocks))])
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0 + bs].copy()
        d = 0.0025 * max(T, 1e-9) ** 0.5 * rng.randn(bs)
        d -= d.mean()
        seg = np.clip(seg + d, 0, 1)
        hn = h.copy(); hn[i0:i0 + bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h, c5c = hn, c5n
            if c5n < bc:
                bc, bh = c5n, h.copy()
    return bh, bc

def _polish(h, niter, rng, deadline, scale):
    nc = len(h)
    c5c = _eval(h)
    bh, bc = h.copy(), c5c
    for _ in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(nc)
        j = (i + 1 + rng.randint(0, 8)) % nc
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < bc:
                    bc, bh = c5n, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc

def _reoptimize_from(h, rng, deadline):
    n = len(h)
    d = 0.012 * rng.randn(n)
    d -= d.mean()
    hn = _repair(np.clip(h + d, 0, 1), n)
    h2, c2 = _nesterov_smax(hn, [1000, 5000, 20000], 200, 0.003, deadline, topk=8)
    h3, c3 = _nesterov_exact(h2, 800, 0.001, deadline, topk=10, beta=0.91)
    return h3, c3

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 4
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h, best_c5 = h.copy(), _eval(h)

    # Phase 1: Graduated softmax Nesterov (main basin-finding)
    dl1 = min(deadline, t0 + 300)
    h1, c1 = _nesterov_smax(best_h.copy(), [150, 600, 2500, 10000, 40000, 120000, 300000], 350, 0.004, dl1, topk=12, b0=0.80, b1=0.95)
    if c1 < best_c5:
        best_c5, best_h = c1, h1.copy()

    # Phase 2: Exact subgradient Nesterov (fine)
    dl2 = min(deadline, t0 + 500)
    for tk, b, s in [(14, 0.88, 0.003), (10, 0.91, 0.002), (8, 0.93, 0.001)]:
        if time.time() > dl2:
            break
        h2, c2 = _nesterov_exact(best_h.copy(), 2500, s, dl2, topk=tk, beta=b)
        if c2 < best_c5:
            best_c5, best_h = c2, h2.copy()

    # Phase 3: Cyclic worst-shift refinement
    dl3 = min(deadline, t0 + 640)
    h3, c3 = _cyclic_refine(best_h.copy(), dl3, n_top=8, n_cycles=50)
    if c3 < best_c5:
        best_c5, best_h = c3, h3.copy()

    # Phase 4: Perturb + re-optimize (escape local min)
    dl4 = min(deadline, t0 + 760)
    for _ in range(3):
        if time.time() > dl4:
            break
        hp, cp = _reoptimize_from(best_h.copy(), rng, dl4)
        if cp < best_c5:
            best_c5, best_h = cp, hp.copy()

    # Phase 5: Annealing (long, careful)
    dl5 = min(deadline, t0 + 880)
    blocks = np.array([2, 3, 4, 5, 7, 10, 14])
    best_h, best_c5 = _anneal(best_h.copy(), 60000, rng, dl5, blocks, 0.002, 2e-10)

    # Phase 6: Final exact Nesterov
    dl6 = min(deadline, t0 + budget_s - 25)
    h6, c6 = _nesterov_exact(best_h.copy(), 3000, 0.0004, dl6, topk=6, beta=0.94)
    if c6 < best_c5:
        best_c5, best_h = c6, h6.copy()

    # Phase 7: Cyclic + polish
    if time.time() < deadline - 10:
        h7, c7 = _cyclic_refine(best_h.copy(), min(deadline - 8, time.time() + 40), n_top=5, n_cycles=20)
        if c7 < best_c5:
            best_c5, best_h = c7, h7.copy()

    rem = max(0, deadline - time.time())
    n_pol = min(int(rem / 0.0008), 50000)
    best_h, best_c5 = _polish(best_h, n_pol, rng, deadline, 5e-7)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
