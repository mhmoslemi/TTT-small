        if np.max(np.abs(d)) < 1e-14:
            stall += 1
            if stall > 20:
                break
            d = np.random.randn(n)
            d -= d.mean()
            d *= eps * 0.1
        else:
            stall = 0
        step = 1.0
        improved = False
        for _ in range(14):
            h_new = _repair(c + step * d, n)
            c5_new = _eval(h_new)
            if c5_new < c5_cur - 1e-16:
                c, c5_cur = h_new, c5_new
                improved = True
                if c5_cur < best_c5:
                    best_c5, best_h = c5_cur, c.copy()
                break
            step *= 0.5
        if not improved:
            eps *= 0.7
            if eps < 1e-6:
                break
    return best_h, best_c5

def _steep_lp(h, n_steps, eps, m_top, deadline):
    n = len(h)
    best_h, best_c5 = h.copy(), _eval(h)
    c = h.copy()
    c5_cur = best_c5
    for _ in range(n_steps):
        if time.time() > deadline:
            break
        cor = _correl(c)
        idx_top = np.argpartition(cor, -m_top)[-m_top:]
        idx_top = idx_top[np.argsort(-cor[idx_top])]
        grads = np.array([_sg(c, k, n) for k in idx_top])
        c_obj = -grads[0]
        A_ub = grads[1:] if m_top > 1 else None
        b_ub = np.zeros(m_top - 1) if m_top > 1 else None
        A_eq = np.ones((1, n))
        b_eq = np.zeros(1)
        bnds = [(-eps, eps)] * n
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bnds, method='highs', options={'presolve': True})
        except Exception:
            break
        if not res.success or res.x is None or np.max(np.abs(res.x)) < 1e-14:
            break
        d = res.x
        step = 1.0
        improved = False
        for _ in range(12):
            h_new = _repair(c + step * d, n)
            c5_new = _eval(h_new)
            if c5_new < c5_cur - 1e-16:
                c, c5_cur = h_new, c5_new
                improved = True
                if c5_cur < best_c5:
                    best_c5, best_h = c5_cur, c.copy()
                break
            step *= 0.5
        if not improved:
            break
    return best_h, best_c5

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smax_obj(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        ct = np.sort(c)[-topk:]
    else:
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _grad_topk(h, alpha, topk):
    n = len(h)
    c = _correl(h); m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
    else:
        idx = np.arange(len(c))
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n); dx = 2.0 / n
    for ii in range(len(idx)):
        if w[ii] < 1e-13: continue
        g += w[ii] * dx * _sg(h, idx[ii], n)
    return g

def _nesterov(h, alpha, topk, steps, step0, deadline):
    n = len(h)
    f = _smax_obj(h, alpha, topk); v = np.zeros(n)
    beta, step = 0.8, step0
    for _ in range(steps):
        if time.time() > deadline: break
        g = _grad_topk(h, alpha, topk)
        hp = h.copy()
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, n * 0.5)
        fn = _smax_obj(hn, alpha, topk)
        if fn < f:
            h, f = hn, fn; step = min(step * 1.03, step0 * 5.0)
        else:
            step *= 0.5
            if step < 1e-11: break
            h = _proj(hp - step * v, n * 0.5)
            f = _smax_obj(h, alpha, topk)
    return _repair(h, n)

def _slsqp(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline: return h
    def obj(x): return _smax_obj(x, alpha, topk)
    def jac(x): return _grad_topk(x, alpha, topk)
    cons = [{'type':'eq','fun':lambda x: x.sum()-n*0.5,'jac':lambda x: np.ones(n)}]
    try:
        r = minimize(obj, h, jac=jac, method='SLSQP',
                     bounds=[(0.0,1.0)]*n, constraints=cons,
                     options={'maxiter':maxit,'ftol':1e-20,'maxls':60})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _block_anneal(h, niter, rng, deadline, T0, T1, blocks):
    n = len(h); c5c = _eval(h); bh = h.copy(); bc = c5c
    for it in range(niter):
        if time.time() > deadline: break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0+bs].copy()
        d = 0.004 * max(T, 1e-9) ** 0.5 * rng.randn(bs); d -= d.mean()
        seg = np.clip(seg + d, 0, 1)
        hn = h.copy(); hn[i0:i0+bs] = seg; hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h, c5c = hn, c5n
            if c5n < bc: bc, bh = c5n, h.copy()
    return bh, bc

def _pairwise_descent(h, niter, rng, deadline):
    n = len(h); c5c = _eval(h); bh = h.copy(); bc = c5c
    cor = _correl(h)
    for _ in range(niter):
        if time.time() > deadline: return bh, bc
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 5)) % n
        if i == j: continue
        s = h[i] + h[j]
        lo_t = max(0.0, s - 1.0)
        hi_t = min(1.0, s)
        if hi_t - lo_t < 1e-12: continue
        best_t = h[i]; best_f = c5c
        for frac in [0.0, 0.2, 0.4, 0.5, 0.6, 0.8, 1.0]:
            t = lo_t + frac * (hi_t - lo_t)
            h[i], h[j] = t, s - t
            f = _eval(h)
            if f < best_f:
                best_f = f; best_t = t
        h[i], h[j] = best_t, s - best_t
        if best_f < c5c - 1e-18:
            c5c = best_f
            if c5c < bc: bc, bh = c5c, h.copy()
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 10
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h, best_c5 = h.copy(), _eval(h)

    dl1 = min(deadline, t0 + 120)
    for a, tk, mi in [(4000, 50, 600), (20000, 25, 500), (80000, 12, 400)]:
        if time.time() > dl1: break
        hv = _slsqp(best_h.copy(), a, tk, mi, dl1)
        cv = _eval(hv)
        if cv < best_c5: best_c5, best_h = cv, hv.copy()

    dl2 = min(deadline, t0 + 400)
    for eps, mt, mg in [(0.005, 8, 0.001), (0.003, 12, 0.0005), (0.001, 20, 0.0002)]:
        if time.time() > dl2: break
        hv, cv = _balanced_lp(best_h.copy(), 8000, eps, mt, mg, dl2)
        if cv < best_c5: best_c5, best_h = cv, hv.copy()

    dl3 = min(deadline, t0 + 550)
    for a, tk, st, ns in [(80000, 20, 0.002, 3000), (500000, 10, 0.001, 2500)]:
        if time.time() > dl3: break
        hv = _nesterov(best_h.copy(), a, tk, ns, st, dl3)
        cv = _eval(hv)
        if cv < best_c5: best_c5, best_h = cv, hv.copy()
        if time.time() > dl3: break
        hv2, cv2 = _steep_lp(hv.copy(), 3000, 0.002, 6, dl3)
        if cv2 < best_c5: best_c5, best_h = cv2, hv2.copy()

    dl4 = min(deadline, t0 + 750)
    for eps in [0.003, 0.001, 0.0005]:
        if time.time() > dl4: break
        hv, cv = _balanced_lp(best_h.copy(), 6000, eps, 10, 0.0003, dl4)
        if cv < best_c5: best_c5, best_h = cv, hv.copy()

    dl5 = min(deadline, t0 + 870)
    blocks = [1, 2, 3, 4, 5, 6, 8, 10]
    hv, cv = _block_anneal(best_h.copy(), 40000, rng, dl5, 0.004, 1e-11, blocks)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()
        if time.time() < dl5:
            hv2, cv2 = _balanced_lp(best_h.copy(), 5000, 0.001, 8, 0.0002, dl5)
            if cv2 < best_c5: best_c5, best_h = cv2, hv2.copy()

    dl6 = deadline
    rem = max(0, dl6 - time.time())
    npol = max(1000, int(rem / 0.002))
    best_h, best_c5 = _pairwise_descent(best_h, npol, rng, dl6)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
