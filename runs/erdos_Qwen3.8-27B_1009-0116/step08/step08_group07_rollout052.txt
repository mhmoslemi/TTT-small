def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj_sum(h, target):
    """Project h onto {0<=x<=1, sum(x)=target}"""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _grad(h, tol=0.001):
    """Compute subgradient of max correlation"""
    n = len(h)
    c = _correl(h)
    m = c.max()
    active = np.where(c >= m - tol)[0]
    dx = 2.0 / n
    g = np.zeros(n)
    for k in active:
        sh = (n - 1) - k
        if sh >= 0:
            g[:n - sh] += (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= h[:-sh]
        else:
            sp = -sh
            g[sp:] += (1.0 - h[:-sp])
            g[:n - sp] -= h[sp:]
    g *= dx / len(active)
    return g, m

def _bb_nesterov(h, steps, deadline, tol=0.001, beta=0.85):
    """BB step + momentum hybrid"""
    n = len(h)
    target = n * 0.5
    f = _eval(h)
    g, _ = _grad(h, tol)
    v = np.zeros(n)
    step = 0.003
    best_h, best_f = h.copy(), f
    prev_g = g.copy()
    prev_h = h.copy()
    
    for it in range(steps):
        if time.time() > deadline:
            break
        # BB step update
        s = h - prev_h
        y = g - prev_g
        sy = np.dot(s, y)
        ss = np.dot(s, s)
        if ss > 1e-20 and abs(sy) > 1e-20:
            step_bb = ss / abs(sy) if sy > 0 else ss / sy  # BB1 or BB2
            step = 0.5 * step + 0.5 * max(1e-6, min(step_bb, 0.1))
        
        hn = _proj_sum(h - step * v, target)
        fn = _eval(hn)
        
        if fn < f:
            prev_h, prev_g = h.copy(), g.copy()
            h, f = hn, fn
            g, _ = _grad(h, tol)
            v = beta * v + (1 - beta) * g
            step = min(step * 1.05, 0.1)
            if f < best_f:

            h = _proj_sum(prev_h - step * v, target)
            f = _eval(h)
            if f < best_f:

    return _repair(best_h, n), best_f

def _coord_descent(h, niter, rng, deadline, alpha=0.01):
    """Pairwise coordinate descent"""
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for _ in range(niter):
        if time.time() > deadline:
            break
        g, _ = _grad(h, 0.002)
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = g[i] - g[j]
        t = alpha * d
        ni, nj = h[i] + t, h[j] - t
        if ni < 0:
            nj += ni; ni = 0
        elif ni > 1:
            nj -= (ni - 1); ni = 1
        if nj < 0:
            ni += nj; nj = 0
        elif nj > 1:
            ni -= (nj - 1); nj = 1
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5c - 1e-20:
            c5c = c5n
            if c5n < bc:
                bc, bh = c5n, h.copy()
        else:
            h[i], h[j] = h[j] + (nj - h[j]), h[i]  # undo

        a = np.clip(h - mid, 0, 1).sum()
        if a > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0, 1)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _grad(h, tol):
    n = len(h)
    c = _correl(h)
    m = c.max()
    active = np.where(c >= m - tol)[0]
    dx = 2.0 / n
    g = np.zeros(n)
    for k in active:
        sh = (n - 1) - k
        if sh >= 0:
            g[:n-sh] += (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= h[:-sh]
        else:
            sp = -sh
            g[sp:] += (1.0 - h[:-sp])
            g[:n-sp] -= h[sp:]
    g *= dx / len(active)
    return g, m

def _nesterov(h, steps, deadline, tol, beta, step0):

    target = n * 0.5
    f = _eval(h)
    g, _ = _grad(h, tol)
    v = np.zeros(n)
    step = step0
    best_h, best_f = h.copy(), f
    for _ in range(steps):
        if time.time() > deadline:
            break
        hp = h.copy()
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, target)
        fn = _eval(hn)
        if fn < f - 1e-15:
            h, f, g = hn, fn, _grad(hn, tol)[0]
            step = min(step * 1.02, step0 * 30.0)
            if f < best_f:
                best_f, best_h = f, h.copy()
        else:
            step *= 0.5
            h = _proj(hp - step * v, target)
            f = _eval(h)
            if f < best_f:
                best_f, best_h = f, h.copy()
    return _repair(best_h, n), best_f

def _coord(h, niter, rng, deadline, alpha):

        elif ni > 1: nj += (ni - 1); ni = 1
        if nj < 0: ni -= nj; nj = 0

        h[i], h[j] = ni, nj
        c5n = _eval(h)

            if c5n < bc:
                bc, bh = c5n, h.copy()
        else:

                best_c5, best_h = c5y, h2.copy()

    # Phase 4: Soft-max Nesterov (smooth landscape navigation)
    dl4 = t0 + 720
    alpha_sm = 30000
    topk_sm = 10
    n = len(best_h)
    target = n * 0.5
    h2 = best_h.copy()
    f2 = _eval(h2)
    v2 = np.zeros(n)
    stp2 = 0.002
    for _ in range(4000):
        if time.time() > dl4:
            break
        g2 = _smax_grad(h2, alpha_sm, topk_sm)
        hp2 = h2.copy()
        v2 = 0.88 * v2 + 0.12 * g2
        hn2 = _proj(h2 - stp2 * v2, target)
        fn2 = _eval(hn2)
        if fn2 < f2 - 1e-15:
            h2, f2 = hn2, fn2
            stp2 = min(stp2 * 1.02, 0.1)
            if f2 < best_c5:
                best_c5, best_h = f2, h2.copy()
        else:
            stp2 *= 0.5
            h2 = _proj(hp2 - stp2 * v2, target)
            f2 = _eval(h2)
            if f2 < best_c5:
                best_c5, best_h = f2, h2.copy()

    # Phase 5: Coordinate descent
    dl5 = t0 + 880
    n_cd = max(3000, int((dl5 - time.time()) / 0.003))
    best_h, best_c5 = _coord(best_h, n_cd, rng, dl5, 0.01)

    # Phase 6: Fine Nesterov
    dl6 = t0 + budget_s - 40
    h2, c5v = _nesterov(best_h.copy(), 5000, dl6, 0.0008, 0.93, 0.0008)
    if c5v < best_c5:
        best_c5, best_h = c5v, h2.copy()

    # Phase 7: Ultra-fine coordinate polish
    rem = max(1, deadline - time.time())
    n_pol = min(int(rem / 0.002), 50000)
    best_h, best_c5 = _coord(best_h, n_pol, rng, deadline, 3e-5)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
