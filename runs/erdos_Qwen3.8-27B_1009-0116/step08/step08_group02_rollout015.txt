    """Find t in [t_lo, t_hi] minimizing max_p (c[p] + a[p]*t)."""
    # Binary search on T
    # For given T, feasible if exists t with c[p]+a[p]*t <= T for all p
    # → for a[p]>0: t <= (T-c[p])/a[p]
    # → for a[p]<0: t >= (T-c[p])/a[p]
    # → for a[p]=0: need c[p] <= T
    
    # Initial bounds on T
    T_lo = max(c + a*np.maximum(0, t_lo), c + a*np.minimum(0, t_hi))  # not quite right
    # Actually T_lo = min possible max, T_hi = max possible max
    T_hi = max(max(c + a*t_lo), max(c + a*t_hi))
    T_lo = max(c)  # rough lower bound (not tight)
    
    # Better: use the current value as T_hi
    # And a lower bound: max(c) when a[p] for the max elements is 0
    
    for _ in range(60):
        T = (T_lo + T_hi) * 0.5
        # Feasibility check
        pos = a > 1e-15
        neg = a < -1e-15
        zero = ~(pos | neg)
        if c[zero].max() > T:
            T_lo = T
            continue
        t_upper = t_hi
        if pos.any():
            t_upper = min(t_upper, ((T - c[pos]) / a[pos]).min())
        t_lower = t_lo
        if neg.any():
            t_lower = max(t_lower, ((T - c[neg]) / a[neg]).max())
        if t_lower <= t_upper:
            T_hi = T
            t_opt = (t_lower + t_upper) * 0.5
        else:
            T_lo = T
            t_opt = t_lo  # fallback
    
    return t_opt, T_hi

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(50):
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        if s > 0:
            h *= target / s
        h = np.clip(h, 0.0, 1.0)
    # Fix remaining error
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            idx = np.argsort(1.0 - h)
        else:
            idx = np.argsort(h)
        for j in idx:
            if diff > 0:
                a = min(diff, 1.0 - h[j])
                h[j] += a
                diff -= a
            else:
                a = min(-diff, h[j])
                h[j] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h
