def _eval(h):

def _corr_vec(h):
    n = len(h)

def _gradient(h, shifts):
    """Compute gradient of sum of overlaps at given shifts."""
    n = len(h)
    g = np.zeros(n)
    for sh in shifts:
        s = abs(sh)
        if s >= n:
            continue
        g[:n-s] += (1.0 - h[s:])
        g[s:] -= h[:n-s]
    return g

def _lse_obj(h, alpha, top_k):
    n = len(h)
    c = _corr_vec(h)
    idx = np.argsort(c)[-top_k:]
    ct = c[idx]
    m = ct.max()
    s = np.clip(alpha * (ct - m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _lse_grad(h, alpha, top_k):

    c = _corr_vec(h)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()

    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

    return g
