    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a

                    break
    return h

import numpy as np
from scipy.optimize import minimize, linprog
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _sg(h, k, n):
    g = np.zeros(n)
    off = k - (n - 1)
    j = np.arange(n)
    i1 = j - off
    m1 = (i1 >= 0) & (i1 < n)
    g[m1] += (1.0 - h[i1[m1]])
    i2 = j + off
    m2 = (i2 >= 0) & (i2 < n)
    g[m2] -= h[i2[m2]]
    return g

def _lp_descent(h, n_steps, eps, m_top, deadline):
    n = len(h)
    best_h, best_c5 = h.copy(), _eval(h)
    c = h.copy()
    c5_cur = best_c5
    stag = 0
    for _ in range(n_steps):
        if time.time() > deadline:
            break
        cor = _correl(c)
        idx_t = np.argpartition(cor, -m_top)[-m_top:]
        idx_t = idx_t[np.argsort(-cor[idx_t])]
        grads = np.array([_sg(c, k, n) for k in idx_t])
        c_obj = -grads[0]
        A_ub = grads[1:] if m_top > 1 else None
        b_ub = np.zeros(m_top - 1) if m_top > 1 else None
        A_eq = np.ones((1, n))
        b_eq = np.zeros(1)
        bnds = [(-eps, eps)] * n
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bnds, method='highs', options={'presolve': True})
        except Exception:
            break
        if not res.success or res.x is None or np.max(np.abs(res.x)) < 1e-14:
            stag += 1
            if stag > 80:
                break
            continue
        d = res.x
        step = 1.0
        improved = False
        for _ in range(14):
            h_new = _repair(c + step * d, n)
            c5_new = _eval(h_new)
            if c5_new < c5_cur - 1e-17:
                c, c5_cur = h_new, c5_new
                improved = True
                if c5_cur < best_c5:
                    best_c5, best_h = c5_cur, c.copy()
                stag = 0
                break
            step *= 0.5
        if not improved:
            stag += 1
            if stag > 80:
                break
    return best_h, best_c5

def _slsqp(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    c = _correl(h)
    def obj(x):
        cx = _correl(x)
        mx = cx.max()
        idx = np.argpartition(cx, -topk)[-topk:]
        ct = cx[idx]
        s = np.clip(alpha * (ct - mx), -300, 300)
        return mx + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha
    def jac(x):
        cx = _correl(x); mx = cx.max()
        idx = np.argpartition(cx, -topk)[-topk:]
        ct = cx[idx]
        s = np.clip(alpha * (ct - mx), -300, 300)
        w = np.exp(s - s.max()); w /= w.sum()
        g = np.zeros(n); dx = 2.0 / n
        for ii in range(len(idx)):
            if w[ii] < 1e-14: continue
            sh = (n-1) - idx[ii]; wk = w[ii] * dx
            if sh >= 0:
                g[:n-sh] += wk * (1.0 - x[sh:])
                if sh > 0: g[sh:] -= wk * x[:-sh]
            else:
                sp = -sh
                g[sp:] += wk * (1.0 - x[:-sp])
                g[:n-sp] -= wk * x[sp:]
        return g
    def cfun(x): return x.sum() - n * 0.5
    def cjac(x): return np.ones(n)
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        r = minimize(obj, h, jac=jac, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-18, 'maxls': 80})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _grad_polish(h, n_steps, m_top, deadline):
    """Gradient-aware coordinate polish: find best 2-swap per top shift."""
    n = len(h)
    c5c = _eval(h)
    bh, bc = h.copy(), c5c
    fail_count = 0
    for step_i in range(n_steps):
        if time.time() > deadline:
            break
        cor = _correl(h)
        idx_t = np.argpartition(cor, -m_top)[-m_top:]
        idx_t = idx_t[np.argsort(-cor[idx_t])]
        best_hn = None
        best_c5n = c5c
        for k_s in idx_t:
            g = _sg(h, k_s, n)
            i_src = int(np.argmax(g))
            i_dst = int(np.argmin(g))
            if i_src == i_dst:
                continue
            d_max = min(h[i_src], 1.0 - h[i_dst]) * 0.8
            if d_max < 1e-12:
                continue
            for frac in [1.0, 0.5, 0.25, 0.1]:
                d = d_max * frac
                hn = h.copy()
                hn[i_src] -= d
                hn[i_dst] += d
                hn = np.clip(hn, 0.0, 1.0)
                c5n = _eval(hn)
                if c5n < best_c5n - 1e-18:
                    best_c5n = c5n
                    best_hn = hn.copy()
        if best_hn is not None:
            h = best_hn
            c5c = best_c5n
            fail_count = 0
            if c5c < bc:
                bc = c5c
                bh = h.copy()
        else:
            fail_count += 1
            # Random micro-perturbation
            i1 = np.random.randint(n)
            i2 = (i1 + 1 + np.random.randint(0, 10)) % n
            d = 1e-5 * (2 * np.random.random() - 1)
            hn = h.copy()
            hn[i1] += d
            hn[i2] -= d
            hn = _repair(hn, n)
            c5n = _eval(hn)
            if c5n < c5c - 1e-18:
                h = hn
                c5c = c5n
                if c5c < bc:
                    bc = c5c
                    bh = h.copy()
                fail_count = 0
            if fail_count > 200:
                break
    return bh, bc

def _sweep_polish(h, niter, rng, deadline):
    n = len(h)
    c5c = _eval(h)
    bh, bc = h.copy(), c5c
    steps = [5e-7, 2e-7, 1e-7, 5e-8, 2e-8, 1e-8]
    ni_per = max(niter // len(steps), 1)
    for ss in steps:
        for _ in range(ni_per):
            if time.time() > deadline:
                return bh, bc
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 5)) % n
            d = ss * (2 * rng.random() - 1)
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-19:
                    c5c = c5n
                    if c5n < bc:
                        bc, bh = c5n, h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d
    return bh, bc
