```python
import cvxpy as cp
import numpy as np

def oa_solve(h, S, dx, n):
    # S is set of active lag indices
    h_var = cp.Variable(n)
    t = cp.Variable()
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n/2]
    
    c_vals = np.correlate(h, 1-h, mode='full') * dx
    # For each k in S, add linear constraint
    for k_idx in S:
        # k_idx is index into the full correlation array
        # Lag = k_idx - (n-1)
        lag = k_idx - (n - 1)
        # Compute gradient of c_lag at h
        grad = np.zeros(n)
        for j in range(n):
            # h[j] appears as first factor if j+lag in [0, n)
            if 0 <= j + lag < n:
                grad[j] += (1 - h[j + lag])
            # h[j] appears as second factor (with -) if j-lag in [0, n)
            if 0 <= j - lag < n:
                grad[j] -= h[j - lag]
        grad *= dx
        c_val = c_vals[k_idx]
        constraints.append(c_val + grad @ (h_var - h) <= t)
    
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.ECOS)  # or CBC, GLPK
    
    return h_var.value, t.value
```

def _oa_step(h, tol_frac=0.01):
    """One outer approximation step"""
    n = len(h)
    dx = 2.0 / n
    h1 = 1.0 - h
    corr = np.correlate(h, h1, mode='full') * dx
    mx = float(np.max(corr))
    
    # Active set: lags within tol_frac of max
    threshold = mx * (1 - tol_frac)
    active = np.where(corr >= threshold)[0]
    
    # Build LP
    import cvxpy as cp
    h_var = cp.Variable(n)
    t = cp.Variable()
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n/2]
    
    # Add linearized constraints
    for idx in active:
        k = idx - (n-1)
        # Compute gradient of corr[idx] w.r.t. h
        g = np.zeros(n)
        if k >= 0:
            if k == 0:
                g = 1.0 - 2.0 * h
            else:
                g[:n-k] += (1.0 - h[k:])
                g[k:] += (-h[:n-k])
        else:
            m = -k
            g[m:] += (1.0 - h[:n-m])
            g[:n-m] += (-h[m:])
        g *= dx
        c_val = corr[idx]
        constraints.append(c_val + g @ (h_var - h) <= t)
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    prob.solve(solver=cp.SCS, max_iter=1000)
    
    if prob.status == 'optimal' and h_var.value is not None:
        return np.array(h_var.value)
    return None

            if s > S: lo = mid
            else: hi = mid
        x = np.clip(v - 0.5*(lo+hi), 0.0, 1.0)
    return x

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0-h, mode='full')) * (2.0/n))

def _grad_active(h, tol_frac=0.001):
    n = len(h)
    dx = 2.0/n
    h1 = 1.0 - h
    corr = np.correlate(h, h1, mode='full') * dx
    mx = float(np.max(corr))
    thresh = mx * (1.0 - tol_frac)
    active = np.where(corr >= thresh)[0]
    if len(active) == 0: active = np.array([np.argmax(corr)])
    grad = np.zeros(n)
    for idx in active:
        k = idx - (n-1)
        if k >= 0:
            if k == 0:
                grad += 1.0 - 2.0*h
            else:
                grad[:n-k] += (1.0 - h[k:])
                grad[k:] += (-h[:n-k])
        else:
            m = -k
            grad[m:] += (1.0 - h[:n-m])
            grad[:n-m] += (-h[m:])
    return dx * grad / len(active)

def _fista_nonsmooth(h0, max_time, L_init=5.0):
    """FISTA-like accelerated gradient for the non-smooth max objective"""
    n = len(h0)
    S = n * 0.5
    t0 = time.time()
    h = _repair(h0, n)
    y = h.copy()
    y_prev = h.copy()
    t_k = 1.0
    c_y = _c5(y)
    best_h = h.copy()
    best_c = c_y
    L = L_init
    for it in range(200000):
        if time.time() - t0 > max_time: break
        # Compute gradient at y
        g = _grad_active(y)
        # Gradient step
        step = 1.0 / L
        # Backtracking
        z = _proj(y - step * g, S)
        for _ in range(10):
            if _c5(z) <= c_y + float(np.dot(g, z - y)) + 0.5 * L * float(np.dot(z-y, z-y)) + 1e-12:
                break
            step *= 0.5
            z = _proj(y - step * g, S)
        else:
            # Fallback
            z = h
            step = 1.0 / L
        L = 1.0 / step
        # Update
        t_next = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k**2))
        y = z + ((t_k - 1.0) / t_next) * (z - h)
        y = _proj(y, S)
        h = z
        t_k = t_next
        c_y = _c5(y)
        c_h = _c5(h)
        if c_h < best_c:
            best_c = c_h
            best_h = h.copy()
    return best_h, best_c

Hmm, the issue with FISTA on a non-smooth function is that the descent condition doesn't hold in general. The max of affine functions is convex, but our c_k are bilinear (not affine), so the smoothness analysis breaks down.

Let me use a simpler approach: **projected gradient descent with momentum** (Nesterov momentum), which is more robust:

```python
def _grad_descent(h0, max_time, lr_init=0.01, momentum=0.9):
    n = len(h0)
    S = n * 0.5
    t0 = time.time()
    h = _repair(h0, n)
    v = np.zeros(n)
    c = _c5(h)
    best_h = h.copy()
    best_c = c
    lr = lr_init
    for it in range(500000):
        if time.time() - t0 > max_time: break
        g = _grad_active(h)
        v = momentum * v + g
        step = lr / (np.linalg.norm(v) + 1e-12)
        # Backtracking line search
        alpha = step
        for _ in range(8):
            h_new = _proj(h - alpha * v, S)
            c_new = _c5(h_new)
            if c_new < c:
                break
            alpha *= 0.5
        else:
            h_new = h
            c_new = c
        h = h_new
        c = c_new
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c
```

This might be slower due to many C5 evaluations. Let me think about the computation cost:
- _c5: O(n²) for correlate (actually O(n log n) with FFT, but np.correlate is O(n²))
  For n=560: 560² = 313600 operations. About 0.3ms in numpy.
- _grad_active: O(|active| * n) for the loop over active lags. If |active|~50: 50*560 = 28000. About 0.03ms.
- _proj: O(n * iterations) ~ O(n). Very fast.

So each iteration is maybe 0.5ms. In 100 seconds, we can do about 200,000 iterations. That's plenty.

But the backtracking line search might evaluate C5 multiple times per iteration. Let me limit it to 3 evaluations max.

Actually, for the line search, I can skip it and just use a fixed step with occasional adjustment. This is faster.

Let me also think about the **local search** more carefully. The previous code's local search does random pair swaps. I'll do a more systematic one:

1. Compute full gradient g
2. Find i = argmin(g) (where increasing h helps most)
3. Find j = argmax(g) (where decreasing h helps most)
4. Try swap: h[i] += δ, h[j] -= δ
5. Line search on δ (try δ = 0.001, 0.005, 0.01, 0.05, 0.1 scaled)
6. Accept best, recompute gradient

This is O(n) per step (gradient computation) + O(1) line search evaluations.

Let me now write the complete solution:

def _repair(h, n):
    t = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if s <= 1e-15: return np.full(n, 0.5)
    for _ in range(80):
        h *= t/s; np.clip(h, 0.0, 1.0, out=h); s = h.sum()
        if abs(s-t) < 1e-12: break
    d = t - h.sum()
    if abs(d) > 1e-12:
        order = np.argsort(1.0-h) if d > 0 else np.argsort(h)
        for i in order:
            if d > 0: a = min(d, 1.0-h[i]); h[i] += a; d -= a
            else: a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13: break
    return h

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)
    x += (S - x.sum()) / x.size
