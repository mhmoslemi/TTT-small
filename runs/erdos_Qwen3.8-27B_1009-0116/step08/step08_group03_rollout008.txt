    if end3 > time.time() + 5:

    # Phase 4: Coordinate descent refinement
    end4 = t0 + budget_s - 5
    if end4 > time.time() + 10:
        best, best_c = _coord_descent(best, end4)
    # Final variants check
    variants = [best, 1.0 - best, best[::-1]]
    for vv in variants:
        v2 = _proj(vv, len(vv) * 0.5)
        cv = _c5_direct(v2)
        if cv < best_c: best_c, best = cv, v2
    best = _repair(best, len(best))
    best_c = _c5_direct(best)
    return best, best_c, len(best)
