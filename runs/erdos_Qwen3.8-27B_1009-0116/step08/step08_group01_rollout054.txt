def _ls_worst(h, n_iter, step, rng, top_lags=3):
    n = len(h)
    cur_c = _eval(h)
    best_c = cur_c
    best_h = h.copy()
    worst_shifts = [0]
    for it in range(n_iter):
        if it % 4000 == 0:
            cv = _eval_vec(h)
            top_idx = np.argsort(cv)[-top_lags:]
            worst_shifts = list(abs(ii - (n-1)) for ii in top_idx)
            worst_shifts = [s if s > 0 else 1 for s in worst_shifts]
        if rng.rand() < 0.65 and len(worst_shifts) > 0:
            ws = worst_shifts[rng.randint(len(worst_shifts))]
            i = rng.randint(n)
            j = (i + ws) % n
        else:
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 23)) % n
        d = step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c_new = _eval(h)
            if c_new < cur_c:
                cur_c = c_new
                if c_new < best_c:
                    best_c = c_new; best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values))
    n = len(h)
    best_h = h.copy()
    best_c = _eval(best_h)

    def _el():
        return time.time() - t0

    # Phase 1: Gradient descent warmup
    if _el() < 20:
        h, c = _gd(best_h.copy(), 200, 0.004, 3000, 5)
        if c < best_c:
            best_c, best_h = c, h

    # Phase 2: SLSQP refinement
    if _el() < 60:
        h = _slsqp(best_h.copy(), 5000, 5, 500)
        c = _eval(h)
        if c < best_c:
            best_c, best_h = c, h
    if _el() < 100:
        h = _slsqp(best_h.copy(), 50000, 4, 400)
        c = _eval(h)
        if c < best_c:
            best_c, best_h = c, h

    # Phase 3: Targeted worst-lag local search
    if _el() < 350:
        best_h, best_c = _ls_worst(best_h, 200000, 4e-5, rng)

    # Phase 4: General 2-point with SA
    if _el() < 550:
        sa = max(5e-7, best_c * 3e-5)
        best_h, best_c = _ls2(best_h, 150000, 3e-5, rng, sa)
        best_h, best_c = _ls2(best_h, 100000, 2e-5, rng)

    # Phase 5: 3-point search
    if _el() < 650:
        best_h, best_c = _ls3(best_h, 80000, 2e-5, rng)

    # Phase 6: Second gradient pass + polish
    if _el() < 720:
        h, c = _gd(best_h.copy(), 100, 0.001, 20000, 3)
        if c < best_c:
            best_c, best_h = c, h
        best_h, best_c = _ls2(best_h, 50000, 1e-5, rng)

    # Phase 7: Extended worst-lag search at smaller step
    if _el() < 900:
        best_h, best_c = _ls_worst(best_h, 180000, 1.5e-5, rng)

    # Phase 8: 3-point fine
    if _el() < 980:
        best_h, best_c = _ls3(best_h, 60000, 8e-6, rng)

    # Phase 9: Ultra-fine 2-point
    if _el() < budget_s - 10:
        best_h, best_c = _ls2(best_h, 100000, 5e-6, rng)
        best_h, best_c = _ls2(best_h, 60000, 2e-6, rng)

    # Phase 10: Final micro-polish
    if _el() < budget_s - 2:
        rem = int((budget_s - 2 - _el()) * 2500)
        if rem > 500:
            best_h, best_c = _ls2(best_h, rem, 1e-6, rng)

    best_h = _repair(best_h, n)
    best_c = _eval(best_h)
    return (best_h, best_c, n)
