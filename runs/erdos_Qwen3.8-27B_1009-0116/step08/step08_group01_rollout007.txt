    for _ in range(200):

        h = np.clip(h, 0.0, 1.0)
        s2 = h.sum()
        if abs(s2 - target) < 1e-12:
            s = s2; break
        s = s2
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:

def _corr(h):

def _proj_target(h, target):
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_top(h, top_k, alpha=1.0):
    """Vectorized gradient of soft-max of top_k corr values"""
    n = len(h)
    dx = 2.0 / n
    c_raw = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(c_raw)[-top_k:]
    ct = c_raw[idx] * dx
    m = ct.max()
    s = np.clip(alpha * (ct - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        wk = w[ii] * dx
        sh = k - (n - 1)

            sp = -sh

            g[:n-sp] -= wk * h[sp:]

def _mgd(h, n_iter, alpha, top_k, step0, rng):
    """Nesterov momentum gradient descent with projection"""
    n = len(h)
    target = n * 0.5
    v = np.zeros(n)
    y = h.copy()
    mu = 0.9
    step = step0
    cur_c = _eval(h)
    for it in range(n_iter):
        g = _grad_top(y, top_k, alpha)
        y_new = h + step * g
        v = mu * v + (1.0 - mu) * (y_new - h)
        y = y_new
        h_try = _proj_target(y + mu * v, target)
        c_try = _eval(h_try)
        if c_try < cur_c:

            cur_c = c_try
            step = min(step * 1.03, step0 * 5.0)

    """L-BFGS-B optimization of soft-max objective"""
    n = len(h)
    target = n * 0.5
    bnds = [(0.0, 1.0)] * n
    
    def obj(x):
        c_raw = np.correlate(x, 1.0 - x, mode='full')
        idx = np.argsort(c_raw)[-top_k:]
        ct = c_raw[idx] * (2.0/n)
        m = ct.max()
        s = np.clip(alpha * (ct - m), -400, 400)
        s -= s.max()
        return m + np.log(np.exp(s).sum()) / alpha
    
    def grad(x):
        return _grad_top(x, top_k, alpha)
    
    def sum_eq(x):
        return x.sum() - target
    
    def sum_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_jac}]
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-10})
        return _repair(res.x, n)

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(300):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _proj(h, target):
    lo, hi = -5.0, 5.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def _grad_smooth(h, alpha, top_k):
    n = len(h)
    dx = 2.0 / n
    c = _corr_raw(h)
    idx = np.argsort(c)[-top_k:]
    ct = c[idx] * dx
    m = ct.max()
    s = np.clip(alpha * (ct - m), -400.0, 400.0)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        k = idx[ii]
        wk = w[ii] * dx
        sh = k - (n - 1)
        if sh == 0:
            g += wk * (1.0 - 2.0 * h)
        elif sh > 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _smgd(h, n_iter, alpha, top_k, step0, rng):
    n = len(h)
    target = n * 0.5
    v = np.zeros(n)
    y = h.copy()
    mu = 0.9
    step = step0
    cur_c = _eval(h)
    for it in range(n_iter):
        g = _grad_smooth(y, alpha, top_k)
        y_new = h + step * g
        v = mu * v + (1.0 - mu) * (y_new - h)
        y = y_new
        h_try = _proj(y + mu * v, target)
        c_try = _eval(h_try)
        if c_try < cur_c:
            h = h_try
            cur_c = c_try
            step = min(step * 1.02, step0 * 5.0)
        else:
            step *= 0.6
            step = max(step, step0 * 0.05)
    return _repair(h, n), cur_c

def _slsqp_opt(h, alpha, top_k, maxiter):
    n = len(h)
    target = n * 0.5
    dx = 2.0 / n
    bnds = [(0.0, 1.0)] * n
    def _obj(x):
        c = _corr_raw(x)
        idx = np.argsort(c)[-top_k:]
        ct = c[idx] * dx
        m = ct.max()
        s = np.clip(alpha * (ct - m), -400.0, 400.0)
        s -= s.max()
        return m + np.log(np.exp(s).sum()) / alpha
    def _jac(x):
        return _grad_smooth(x, alpha, top_k)
    def _c_fun(x):
        return x.sum() - target
    def _c_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': _c_fun, 'jac': _c_jac}]
    try:
        res = minimize(_obj, h, jac=_jac, method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(res.x, n)
    except Exception:
        return h

def _lbfgs_opt(h, alpha, top_k, maxiter):
    n = len(h)
    target = n * 0.5
    dx = 2.0 / n
    bnds = [(0.0, 1.0)] * n
    def _obj(x):
        c = _corr_raw(x)
        idx = np.argsort(c)[-top_k:]
        ct = c[idx] * dx
        m = ct.max()
        s = np.clip(alpha * (ct - m), -400.0, 400.0)
        s -= s.max()
        return m + np.log(np.exp(s).sum()) / alpha
    def _jac(x):
        return _grad_smooth(x, alpha, top_k)
    def _c_fun(x):
        return x.sum() - target
    def _c_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': _c_fun, 'jac': _c_jac}]
    try:
        res = minimize(_obj, h, jac=_jac, method='L-BFGS-B',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15, 'maxfun': 5000})
        return _repair(res.x, n)
    except Exception:
        return h

def _local_pairs(h, n_iter, step, rng, adapt=True):
    n = len(h)
    cur_c = _eval(h)
    best_c = cur_c
    best_h = h.copy()
    s = step
    stale = 0
    for it in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 8)) % n
        d = s * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c = _eval(h)
            if c < cur_c:
                cur_c = c
                stale = 0
                if adapt:
                    s = min(s * 1.01, step * 3.0)
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
                stale += 1
                if adapt and stale > 500:
                    s *= 0.9
                    stale = 0
    return best_h, best_c

def _grad_descent_focused(h, n_iter, alpha0, rng):
    """Focused gradient descent on the max lag"""
    n = len(h)
    dx = 2.0 / n
    target = n * 0.5
    alpha = alpha0
    cur_c = _eval(h)
    for it in range(n_iter):
        c = _corr_raw(h) * dx
        k_max = int(np.argmax(c))
        sh = k_max - (n - 1)
        g = np.zeros(n)
        if sh == 0:
            g = 1.0 - 2.0 * h
        elif sh > 0:
            g[:n - sh] = 1.0 - h[sh:]
            g[sh:] -= h[:n - sh]
        else:
            sp = -sh
            g[sp:] = 1.0 - h[:n - sp]
            g[:n - sp] -= h[sp:]
        g *= dx
        # Line search
        found = False
        a = alpha
        for _ in range(6):
            h_try = _proj(h - a * g, target)
            c_try = _eval(h_try)
            if c_try < cur_c - 1e-15:
                h = h_try
                cur_c = c_try
                alpha = min(a * 1.05, alpha0 * 30.0)
                found = True
                break
            a *= 0.5
        if not found:
            alpha *= 0.7
    return h, cur_c

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(300):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _proj(h, target):
    lo, hi = -5.0, 5.0
    for _ in range(64):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def _grad_smooth(h, alpha, top_k):
    n = len(h)
    dx = 2.0 / n
    c = _corr_raw(h)
    idx = np.argsort(c)[-top_k:]
    ct = c[idx] * dx
    m = ct.max()
    s = np.clip(alpha * (ct - m), -400.0, 400.0)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        k = idx[ii]
        wk = w[ii] * dx
        sh = k - (n - 1)
        if sh == 0:
            g += wk * (1.0 - 2.0 * h)
        elif sh > 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _smgd(h, n_iter, alpha, top_k, step0, rng):
    n = len(h)
    target = n * 0.5
    v = np.zeros(n)
    y = h.copy()
    mu = 0.9
    step = step0
    cur_c = _eval(h)
    for it in range(n_iter):
        g = _grad_smooth(y, alpha, top_k)
        y_new = h + step * g
        v = mu * v + (1.0 - mu) * (y_new - h)
        y = y_new
        h_try = _proj(y + mu * v, target)
        c_try = _eval(h_try)
        if c_try < cur_c:
            h = h_try
            cur_c = c_try
            step = min(step * 1.02, step0 * 5.0)
        else:
            step *= 0.6
            step = max(step, step0 * 0.05)
    return _repair(h, n), cur_c

def _slsqp_opt(h, alpha, top_k, maxiter):
    n = len(h)
    target = n * 0.5
    dx = 2.0 / n
    bnds = [(0.0, 1.0)] * n
    c = _corr_raw(h)
    idx_dbg = np.argsort(c)[-top_k:]
    def _obj(x):
        cc = _corr_raw(x)
        ii2 = np.argsort(cc)[-top_k:]
        ct2 = cc[ii2] * dx
        m2 = ct2.max()
        ss = np.clip(alpha * (ct2 - m2), -400.0, 400.0)
        ss -= ss.max()
        return m2 + np.log(np.exp(ss).sum()) / alpha
    def _jac(x):
        return _grad_smooth(x, alpha, top_k)
    def _c_fun(x):
        return x.sum() - target
    def _c_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': _c_fun, 'jac': _c_jac}]
