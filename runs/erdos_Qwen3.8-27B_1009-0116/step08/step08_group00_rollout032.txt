def _repair(h, n):
    """Repair h to satisfy [0,1] bounds and sum = n/2."""
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    # Iterative proportional fitting
    for _ in range(100):
        if s == 0:
            return np.full(n, 0.5)
        scale = target / s
        h *= scale
        h = np.clip(h, 0.0, 1.0)
        s_new = h.sum()
        if abs(s_new - target) < 1e-14:
            break
        s = s_new
    # Final adjustment
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a

    return h

    m = c.max()
    # Get top m_lags

    s = np.clip(alpha * (c[top_idx] - m), -500, 500)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    val = m + np.log(w.sum()) / alpha
    
    # Gradient
    g = np.zeros(n)
    for ii, k in enumerate(top_idx):
        if w[ii] < 1e-15:

            if _el() > budget_s - 40:
                break
            h_hop, c_hop = _try_perturb(best_h, rng, 15, 0.0005 * (attempt + 1))
            if c_hop >= best_c5:
                h_hop = best_h.copy()
                c_hop = best_c5
            h_r, c_r = _fine_2pt(h_hop, rng, 3000, 5e-5)
            if c_r < best_c5:
                best_c5 = c_r
                best_h = h_r

    # Phase 4: High-alpha refinement (different schedule than parent)
    if _el() < budget_s - 120:
        for alpha, ml, iters in [(100000000, 100, 500), (1000000000, 120, 400)]:
            if _el() > budget_s - 30:
                break
            h_new = _slsqp_pen(best_h, alpha, ml, iters, 1e-15, 100000)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new

    # Phase 5: Gradient-informed local search (medium steps)
    if _el() < budget_s - 100:
        ntries = min(int((budget_s - 100 - _el()) * 150), 150000)
        best_h, best_c5 = _grad_local(best_h, rng, ntries, 100000, 80, 2e-5)

    # Phase 6: Fine 2-point (small steps)
    if _el() < budget_s - 40:
        n_pol = min(int((budget_s - 40 - _el()) / 0.0015), 300000)
        best_h, best_c5 = _fine_2pt(best_h, rng, n_pol, 2e-6, span=15)

    # Phase 7: Ultra-fine polish
    if _el() < budget_s - 3:
        n_pol2 = min(int((budget_s - 3 - _el()) / 0.001), 400000)
        best_h, best_c5 = _fine_2pt(best_h, rng, n_pol2, 2e-7, span=8)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
