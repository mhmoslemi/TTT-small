        k_lo2 = max(0, nm1 - j)
        k_hi2 = min(2*n - 2, nm1 + j)
        ks2 = np.arange(k_lo2, k_hi2 + 1)
        i2 = j + ks2 - nm1
        slope[ks2] -= h[i2]
        
        # base[k] = c[k] - slope[k]*h[j]
        base = c - slope * h[j]
        
        # Minimize max_k(base[k] + slope[k]*x) over x in [0,1]
        lo_x, hi_x = 0.0, 1.0
        for _ in range(25):
            mid = (lo_x + hi_x) * 0.5
            fv = base + slope * mid
            fmax = fv.max()
            active = np.where(fv >= fmax - 1e-12)[0]
            sl = slope[active]
            if sl.min() > 1e-10:
                hi_x = mid
            elif sl.max() < -1e-10:
                lo_x = mid
            else:
                break
        x_opt = (lo_x + hi_x) * 0.5
        # Evaluate improvement
        fv_new = base + slope * x_opt
        fv_old = base + slope * h[j]
        if fv_new.max() < fv_old.max() - 1e-14:
            h[j] = x_opt
            # Update c for affected entries (approximate: recompute)
            c = np.correlate(h, 1.0 - h, mode='full')
    return h

def _small_anneal(h, n, rng, deadline, n_iter=8000):
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    blocks = np.array([2, 3, 4, 5, 7, 10, 15])
    for it in range(n_iter):
        if time.time() > deadline:
            break
        T = 0.005 * (1e-8) ** (it / n_iter)
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0+bs].copy()
        d = 0.002 * max(T, 1e-9) ** 0.5 * rng.randn(bs)
        d -= d.mean()
        seg = np.clip(seg + d, 0, 1)
        hn = h.copy()
        hn[i0:i0+bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h = hn
            c5c = c5n
            if c5n < bc:
                bc, bh = c5n, h.copy()
    return bh, bc

def _fine_polish(h, n, rng, deadline, scale=1e-5, max_iter=30000):
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 8)) % n
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-18:
                c5c = c5n
                if c5n < bc:
                    bc, bh = c5n, h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Smoothed gradient descent with LSE, increasing sharpness
    dl1 = min(deadline, t0 + 200)
    alphas = [500, 2000, 8000, 30000, 100000, 300000]
    for alpha in alphas:
        if time.time() > dl1:
            break
        topk = max(5, min(20, int(56000.0 / alpha)))
        h2, c5v = _grad_desc(best_h.copy(), alpha, topk, n, 800, dl1)
        if c5v < best_c5:
            best_c5, best_h = c5v, h2.copy()

    # Phase 2: Coordinate descent passes (exact 1D optimization)
    dl2 = min(deadline, t0 + 500)
    for _ in range(15):
        if time.time() > dl2:
            break
        best_h = _coord_pass(best_h.copy(), n, dl2)
        c5v = _eval(best_h)
        if c5v < best_c5:
            best_c5 = c5v

    # Phase 3: Annealing to escape local minima
    dl3 = min(deadline, t0 + 700)
    best_h, best_c5 = _small_anneal(best_h.copy(), n, rng, dl3, 20000)

    # Phase 4: More coordinate descent after annealing
    dl4 = min(deadline, t0 + 900)
    for _ in range(12):
        if time.time() > dl4:
            break
        best_h = _coord_pass(best_h.copy(), n, dl4)
        c5v = _eval(best_h)
        if c5v < best_c5:
            best_c5 = c5v

    # Phase 5: Jitter + re-optimize (multiple attempts)
    dl5 = min(deadline, t0 + 1000)
    for attempt in range(6):
        if time.time() > dl5:
            break
        d = 0.003 * rng.randn(n)
        d -= d.mean()
        hj = _repair(np.clip(best_h + d, 0, 1), n)
        hj = _coord_pass(hj, n, dl5)
        c5v = _eval(hj)
        if c5v < best_c5:
            best_c5, best_h = c5v, hj.copy()

    # Phase 6: Final polished coordinate descent
    dl6 = min(deadline, t0 + budget_s - 15)
    for _ in range(8):
        if time.time() > dl6:
            break
        best_h = _coord_pass(best_h.copy(), n, dl6)
        c5v = _eval(best_h)
        if c5v < best_c5:
            best_c5 = c5v

    # Phase 7: Fine polish (pairwise swaps)
    rem = max(0.0, deadline - time.time())
    n_pol = int(min(rem / 0.001, 50000))
    best_h, best_c5 = _fine_polish(best_h, n, rng, deadline, 1e-6, n_pol)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
