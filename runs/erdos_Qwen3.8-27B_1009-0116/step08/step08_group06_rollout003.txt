    v = np.zeros(n)
    mu = 0.75
    step = step0
    for _s in range(steps):
        if time.time() > t_end:
            break
        g = _pnorm_grad(h, p, n)
        h_prev = h.copy()
        v = mu * v + (1.0 - mu) * g
        h_new = _repair(h - step * v, n)
        c_new = _eval(h_new)
        if c_new < best_c:
            h, best_c = h_new, c_new
            best_h = h.copy()
            step = min(step * 1.02, 0.015)
        else:
            step *= 0.45
            if step < 1e-12:
                v = np.zeros(n)
                step = step0 * 0.4
            h = _repair(h_prev - step * v, n)
            c_chk = _eval(h)
            if c_chk < best_c:
                best_c, best_h = c_chk, h.copy()
    return best_h, best_c


class _SLSCtx:
    def __init__(self, alpha, n, target):
        self.alpha = alpha
        self.n = n
        self.target = target

    def obj(self, x):
        c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / self.n)
        m = c.max()
        s = np.clip(self.alpha * (c - m), -500, 500)
        return m + np.log(np.sum(np.exp(s - s.max()))) / self.alpha

    def grad(self, x):
        c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / self.n)
        m = c.max()
        s = np.clip(self.alpha * (c - m), -500, 500)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(self.n)
        dx2 = 2.0 / self.n
        for k in range(len(w)):
            if w[k] < 1e-15:
                continue
            sh = (self.n - 1) - k
            wk = w[k] * dx2
            if sh >= 0:
                g[:self.n - sh] += wk * (1.0 - x[sh:])
                if sh > 0:
                    g[sh:] -= wk * x[:self.n - sh]
            else:
                sp = -sh
                g[sp:] += wk * (1.0 - x[:self.n - sp])
                g[:self.n - sp] -= wk * x[sp:]
        return g

    def cf(self, x):
        return x.sum() - self.target

    def cj(self, x):
        return np.ones(self.n)


def _slsqp(h, alpha, maxiter, ftol, target):
    n = len(h)
    ctx = _SLSCtx(alpha, n, target)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': ctx.cf, 'jac': ctx.cj}]
    try:
        res = minimize(ctx.obj, h, jac=ctx.grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)


def _local_search(h, n, rng, t_end, scale, j_window):
    best_h, best_c = h.copy(), _eval(h)
    cur_c = best_c
    while time.time() < t_end:
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, j_window)) % n
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c_new = _eval(h)
            if c_new < cur_c:
                cur_c = c_new
                if c_new < best_c:
                    best_c, best_h = c_new, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c


def _local_3pt(h, n, rng, t_end, scale):
    best_h, best_c = h.copy(), _eval(h)
    cur_c = best_c
    while time.time() < t_end:
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 20)) % n
        k = (i + 10 + rng.randint(0, 30)) % n
        d1 = scale * rng.randn()
        d2 = scale * rng.randn()
        ni, nj, nk = h[i] + d1, h[j] + d2, h[k] - d1 - d2
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0 and 0.0 <= nk <= 1.0:
            h[i], h[j], h[k] = ni, nj, nk
            c_new = _eval(h)
            if c_new < cur_c:
                cur_c = c_new
                if c_new < best_c:
                    best_c, best_h = c_new, h.copy()
            else:
                h[i], h[j], h[k] = ni - d1, nj - d2, nk + d1 + d2
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    t_global = t0 + budget_s
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    target = n / 2.0
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: p-norm Nesterov (smooth, broad)
    t1 = t0 + budget_s * 0.12
    for p in [4.0, 16.0]:
        if time.time() > t1 - 5:
            break
        h_new, c_new = _nesterov_pnorm(best_h.copy(), p, n, 2500, 0.004, t1)
        if c_new < best_c5:
            best_c5, best_h = c_new, h_new.copy()

    # Phase 2: Active-set projected (top-1, top-2, top-4)
    t2 = t0 + budget_s * 0.30
    for k_act in [1, 2, 4]:
        if time.time() > t2 - 5:
            break
        h_new, c_new = _akt_proj(best_h.copy(), n, k_act, 4000, 0.004, t2)
        if c_new < best_c5:
            best_c5, best_h = c_new, h_new.copy()

    # Phase 3: SLSQP LSE medium alpha
    t3 = t0 + budget_s * 0.48
    for alpha in [5000, 50000]:
        if time.time() > t3 - 10:
            break
        h_new = _slsqp(best_h.copy(), alpha, 1000, 1e-14, target)
        c_new = _eval(h_new)
        if c_new < best_c5:
            best_c5, best_h = c_new, h_new.copy()

    # Phase 4: p-norm sharp (p=256)
    t4 = t0 + budget_s * 0.62
    h_new, c_new = _nesterov_pnorm(best_h.copy(), 256.0, n, 3000, 0.002, t4)
    if c_new < best_c5:
        best_c5, best_h = c_new, h_new.copy()

    # Phase 5: Active set with more lags
    t5 = t0 + budget_s * 0.74
    for k_act in [6, 10]:
        if time.time() > t5 - 5:
            break
        h_new, c_new = _akt_proj(best_h.copy(), n, k_act, 2500, 0.003, t5)
        if c_new < best_c5:
            best_c5, best_h = c_new, h_new.copy()

    # Phase 6: SLSQP high alpha
    t6 = t0 + budget_s * 0.84
    if time.time() < t6 - 15:
        h_new = _slsqp(best_h.copy(), 500000, 600, 1e-15, target)
        c_new = _eval(h_new)
        if c_new < best_c5:
            best_c5, best_h = c_new, h_new.copy()
    if time.time() < t6 - 10:
        h_new = _slsqp(best_h.copy(), 5000000, 400, 1e-16, target)
        c_new = _eval(h_new)
        if c_new < best_c5:
            best_c5, best_h = c_new, h_new.copy()

    # Phase 7: 3-point local search
    t7 = t_global - 25
    if time.time() < t7:
        best_h, best_c5 = _local_3pt(best_h, n, rng, t7, 4e-5)

    # Phase 8: Ultra-fine 2-point
    t8 = t_global - 2
    if time.time() < t8:
        best_h, best_c5 = _local_search(best_h, n, rng, t8, 1.5e-6, 5)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
