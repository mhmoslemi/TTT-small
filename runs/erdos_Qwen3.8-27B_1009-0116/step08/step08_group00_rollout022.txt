        h = np.clip(h, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-15:
                break
    return h

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _eval(h):
    return float(_corr(h).max())

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _somax(h, alpha, m):
    n = len(h)
    c = _corr(h)
    mx = c.max()
    idx = np.argsort(c)[-m:]
    s = np.clip(alpha * (c[idx] - mx), -350, 350)
    s -= s.max()
    w = np.exp(s)
    ws = w.sum()
    obj = mx + np.log(ws) / alpha
    g = np.zeros(n)
    for ii in range(len(idx)):
        wk = w[ii] / ws
        if wk < 1e-16:
            continue
        L = idx[ii] - (n - 1)
        if L >= 0:
            g[L:] += wk * (1.0 - h[:n - L])
            g[:n - L] -= wk * h[L:]
        else:
            M = -L
            g[:n - M] += wk * (1.0 - h[M:])
            g[M:] -= wk * h[:n - M]
    return obj, g

def _somax_val(h, alpha, m):
    return _somax(h, alpha, m)[0]

def _somax_grad(h, alpha, m):
    return _somax(h, alpha, m)[1]

def _sum_fn(x, hn):
    return x.sum() - hn

def _sum_jac(x, hn):
    return np.ones(len(x))

def _slsqp(h, alpha, m, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fn, 'jac': _sum_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_somax_val, h, jac=_somax_grad, args=(alpha, m),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return h

def _nesterov(h, alpha, m, steps, step0, mom):
    n = len(h)
    y = h.copy()
    v = np.zeros(n)
    best_y = y.copy()
    best_c = _eval(y)
    step = step0
    for _ in range(steps):
        f, g = _somax(y, alpha, m)
        v = mom * v + g
        accepted = False
        for t in [step, step * 0.5, step * 0.25, step * 0.1]:
            y_new = _proj(y - t * v, n * 0.5)
            c_new = _corr(y_new).max()
            if c_new < best_c - 1e-16:
                y = y_new
                best_y = y_new.copy()
                best_c = c_new
                step = min(t * 1.1, 0.03)
                accepted = True
                break
        if not accepted:
            step = min(step * 1.2, 0.03)
    return best_y, best_c

def _coord_descent(h, n_sweeps, rng):
    n = len(h)
    best_h = h.copy()
    best_c = _eval(best_h)
    for sw in range(n_sweeps):
        order = rng.permutation(n)
        improved = False
        for j in order:
            old_val = best_h[j]
            if old_val <= 0.001 or old_val >= 0.999:
                candidates = np.array([0.0, 1.0, old_val])
            else:
                candidates = np.array([0.0, old_val * 0.5, old_val, 1.0 - old_val, 1.0,
                                       old_val - 0.05, old_val + 0.05])
            candidates = np.clip(np.unique(candidates), 0, 1)
            best_trial_c = best_c
            best_trial_v = old_val
            for cv in candidates:
                if abs(cv - old_val) < 1e-9:
                    continue
                best_h[j] = cv
                best_h = _repair(best_h, n)
                c = _eval(best_h)
                if c < best_trial_c - 1e-14:
                    best_trial_c = c
                    best_trial_v = cv
            best_h[j] = best_trial_v
            best_h = _repair(best_h, n)
            if best_trial_c < best_c - 1e-14:
                best_c = best_trial_c
                improved = True
        if not improved:
            break
    return best_h, best_c

def _two_point_search(h, n_iter, rng, step):
    n = len(h)
    best_h = h.copy()
    best_c = _eval(best_h)
    cur_c = best_c
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 8)) % n
        d = step * rng.randn()
        ni, nj = best_h[i] + d, best_h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            best_h[i], best_h[j] = ni, nj
            c_new = _eval(best_h)
            if c_new < cur_c - 1e-16:
                cur_c = c_new
                if c_new < best_c:
                    best_c = c_new
            else:
                best_h[i], best_h[j] = ni - d, nj + d
    return best_h, best_c

def _one_point_search(h, n_iter, rng, step):
    n = len(h)
    best_h = h.copy()
    best_c = _eval(best_h)
    cur_c = best_c
    for _ in range(n_iter):
        i = rng.randint(n)
        d = step * rng.randn()
        nv = best_h[i] + d
        if 0 <= nv <= 1:
            best_h[i] = nv
            best_h = _repair(best_h, n)
            c_new = _eval(best_h)
            if c_new < cur_c - 1e-17:
                cur_c = c_new
                if c_new < best_c:
                    best_c = c_new
            else:
                best_h[i] = nv - d
    return best_h, best_c
