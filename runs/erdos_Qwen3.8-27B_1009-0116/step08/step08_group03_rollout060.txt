        if time.time() - t0 > max_time:
            break
        if accel and it:
            tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
            y = h + ((t - 1.0) / tn) * (h - hp)
            t = tn
            if y.min() < -0.5 or y.max() > 1.5 or abs(y.sum() - S) > 0.5:
                y = _proj(y, S)
            fy = _smooth_val(y, alpha)
        g = _smooth_grad(y, alpha)
        step = 1.0 / max(Lcur, 1e-12)
        accepted = False
        for _ in range(12):
            xn = _proj(y - step * g, S)
            fx = _smooth_val(xn, alpha)
            d = xn - y
            rhs = fy + float(np.dot(g, d)) + 0.5 * (1.0 / step) * float(np.dot(d, d))
            if fx <= rhs + 1e-10:
                accepted = True
                break
            step *= 0.5
        Lcur = 1.0 / max(step, 1e-12)
        if accepted:
            hp = h
            h = xn
            stale = 0
            if it % 8 == 0:
                c = _c5_direct(h)
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
        else:
            stale += 1
            hp = h
            y = h.copy()
            t = 1.0
            fy = _smooth_val(y, alpha)
            Lcur = max(Lcur * 0.5, 1.0)
            if stale > 12:
                break
    c = _c5_direct(h)
    if c < best_c:
        best_h = h.copy()
        best_c = c
    return h, best_h, best_c

def _active_grad(h, active):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for j in active:
        s = j - (n - 1)
        if s >= 0:
            g[:n-s] += (1.0 - h[s:])
            g[s:] -= h[:n-s]
        else:
            g[-s:] += (1.0 - h[:n+s])
            g[:n+s] -= h[-s:]
    g /= max(1, len(active))
    return dx * g

def _active_descent(h0, time_limit):
    n = len(h0)
    S = n * 0.5
    h = h0.copy()
    best_h = h.copy()
    best_c = _c5_direct(h)
    t0 = time.time()
    step = 0.02
    recompute = 0
    vals = None
    while time.time() - t0 < time_limit:
        if recompute > 50 or vals is None:
            vals = _c5_all(h)
            mx = vals.max()
            tol = max(1e-6, 4.0 * (2.0 / n))
            active = np.where(vals >= mx - tol)[0]
            vals = None
            recompute = 0
        g = _active_grad(h, active)
        xn = _proj(h - step * g, S)
        cn = _c5_direct(xn)
        if cn < _c5_direct(h) - 1e-14:
            h = xn
            step = min(step * 1.02, 0.5)
            if cn < best_c:
                best_c = cn
                best_h = h.copy()
        else:
            step *= 0.8
            recompute += 1
            if step < 1e-7:
                step = 0.01
    return best_h, best_c

def _local_search(h0, rng, deadline):
    n = len(h0)
    S = n * 0.5
    h = h0.copy()
    c = _c5_direct(h)
    best_c = c
    best_h = h.copy()
    tl0 = time.time()
    vals = _c5_all(h)
    recompute = 0
    phase = (time.time() - tl0) / max(1e-9, deadline - tl0)
    T0 = (vals.max()) * 0.005
    while time.time() < deadline:
        if recompute > 1500:
            c = _c5_direct(h)
            if c < best_c:
                best_c = c
                best_h = h.copy()
            vals = _c5_all(h)
            recompute = 0
        g = vals  # use values as proxy for direction (high value = needs reduction)
        rel = (time.time() - tl0) / max(1e-9, deadline - tl0)
        T = T0 * (1.0 - rel) ** 2.0
        step_base = max(1e-10, 0.002 * (0.001) ** rel)
        move_type = rng.random()
        if move_type < 0.5:
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                continue
            if vals[i] > vals[j]:
                s = 1
            else:
                s = -1
        elif move_type < 0.75:
            bl = int(rng.randint(2, min(15, n // 4)))
            i = int(rng.randint(0, n - bl))
            j_start = int(rng.randint(0, n - bl))
            if i + bl <= j_start or j_start + bl <= i:
                s = 1 if rng.random() < 0.5 else -1
            else:
                continue
            avail = min(np.min(1.0 - h[i:i+bl]) if s > 0 else np.min(h[i:i+bl]),
                        np.min(h[j_start:j_start+bl]) if s > 0 else np.min(1.0 - h[j_start:j_start+bl]))
            d = s * min(avail, step_base * bl * (0.5 + 1.5 * rng.random()))
            if abs(d) < 1e-14:
                continue
            h[i:i+bl] += d / bl
            h[j_start:j_start+bl] -= d / bl
            cn = _c5_direct(h)
            recompute += 1
            if cn < c - 1e-15 or (T > 1e-9 and rng.random() < np.exp(-(cn - c) / T)):
                c = cn
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
            else:
                h[i:i+bl] -= d / bl
                h[j_start:j_start+bl] += d / bl
            continue
        else:
            k = int(rng.randint(1, 5))
            idx = rng.randint(n, size=k)
            d = step_base * (0.5 + 1.5 * rng.random())
            if rng.random() < 0.5:
                h[idx] += d
            else:
                h[idx] -= d
            h = _proj(h, S)
            cn = _c5_direct(h)
            recompute += 1
            if cn < c - 1e-15 or (T > 1e-9 and rng.random() < np.exp(-(cn - c) / T)):
                c = cn
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
            else:
                h = best_h.copy()
                c = best_c
        if move_type < 0.5:
            if i != j:
                if s > 0:
                    avail = min(1.0 - h[i], h[j])
                else:
                    avail = min(h[i], 1.0 - h[j])
                if avail > 1e-14:
                    mult = 0.3 + 1.7 * rng.random()
                    if rng.random() < 0.02:
                        mult *= 10.0
                    dd = s * min(avail, step_base * mult)
                    h[i] += dd
                    h[j] -= dd
                    cn = _c5_direct(h)
                    recompute += 1
                    if cn < c - 1e-15 or (T > 1e-9 and rng.random() < np.exp(-(cn - c) / T)):
                        c = cn
                        if c < best_c:
                            best_c = c
                            best_h = h.copy()
                    else:
                        h[i] -= dd
                        h[j] += dd
    return best_h, best_c

def _resample(h, m):
    n = len(h)
    if n == m:
        return _repair(h, n)
    if n % m == 0:
        out = h.reshape(m, n // m).mean(axis=1)
    else:
        x0 = (np.arange(n) + 0.5) * (2.0 / n)
        x1 = (np.arange(m) + 0.5) * (2.0 / m)
        out = np.interp(x1, x0, h)
    return _proj(out, m * 0.5)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(560, 0.5)
    n0 = len(h0)
    h0 = _repair(h0, n0)

    best = h0.copy()
    best_c = _c5_direct(best)
    for hh in (1.0 - h0, h0[::-1], _proj(0.5 + 0.5*(h0 - h0[::-1]), n0*0.5)):
        c = _c5_direct(hh)
        if c < best_c:
            best = hh.copy()
            best_c = c

    # Phase 1: Quick FISTA with moderate alphas
    end1 = t0 + min(80.0, 0.07 * budget_s)
    alphas1 = [30, 100, 400, 1500]
    per1 = (end1 - time.time()) / len(alphas1)
    hcur = best.copy()
    for a in alphas1:
        if time.time() >= end1 - 1:
            break
        hf, bh, bc = _fista_smooth(hcur, a, per1, 15.0, True)
        if bc < best_c:
            best_c = bc
            best = bh
        chf = _c5_direct(hf)
        if chf < best_c:
            best_c = chf
            best = hf
        hcur = hf if chf <= bc else bh

    # Phase 2: FISTA with high alphas (sharper max approximation)
    end2 = t0 + min(500.0, 0.42 * budget_s)
    alphas2 = [5000, 20000, 80000, 300000, 1000000]
    per2 = (end2 - time.time()) / len(alphas2)
    hcur = best.copy()
    for a in alphas2:
        if time.time() >= end2 - 1:
            break
        hf, bh, bc = _fista_smooth(hcur, a, per2, 30.0, True)
        if bc < best_c:
            best_c = bc
            best = bh
        chf = _c5_direct(hf)
        if chf < best_c:
            best_c = chf
            best = hf
        hcur = hf if chf <= bc else bh

    # Phase 3: Active-set gradient descent
    end3 = t0 + min(800.0, 0.70 * budget_s)
    if time.time() < end3 - 5:
        b, c = _active_descent(best, end3 - time.time() - 2)
        if c < best_c:
            best = b
            best_c = c

    # Phase 4: Multi-start perturbations + FISTA refinement
    end4 = t0 + budget_s - 250.0
    if time.time() < end4 - 10:
        for trial in range(4):
            if time.time() >= end4 - 5:
                break
            hpert = best + 0.005 * rng.randn(n0)
            hpert = _proj(hpert, n0 * 0.5)
            hf, bh, bc = _fista_smooth(hpert, 50000, min(30.0, end4 - time.time() - 3), 30.0, True)
            chf = _c5_direct(hf)
            if min(bc, chf) < best_c:
                best = bh if bc < chf else hf
                best_c = min(bc, chf)

    # Phase 5: Final local search with SA and block moves
    end5 = t0 + budget_s - 2.0
    if time.time() < end5 - 1:
        best, best_c = _local_search(best, rng, end5)

    # Final variants
    variants = [best, 1.0 - best, best[::-1],
                _proj(0.5 + 0.5*(best - best[::-1]), n0*0.5),
                _proj(0.5 + 0.5*(best + best[::-1] - 1.0), n0*0.5)]
    for hh in variants:
        c = _c5_direct(hh)
        if c < best_c:
            best = hh
            best_c = c

    best = _repair(best, n0)
    best_c = _c5_direct(best)
    return best, best_c, n0
