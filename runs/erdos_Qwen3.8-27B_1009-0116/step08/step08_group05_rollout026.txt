def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_shift(h, k, n):
    """Gradient of correlation at shift k w.r.t. h"""
    g = np.zeros(n)
    off = k - (n - 1)
    j = np.arange(n)
    i1 = j - off
    mask1 = (i1 >= 0) & (i1 < n)
    g[mask1] += (1.0 - h[i1[mask1]])
    i2 = j + off
    mask2 = (i2 >= 0) & (i2 < n)
    g[mask2] -= h[i2[mask2]]
    return g

        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid

    n = len(h)

            break
        g = _sgrad(c, alpha, topk)
        hp = c.copy()
        v = beta * v + (1.0 - beta) * g
        hn = _proj(c - step * v, n * 0.5)
        fn = _sobj(hn, alpha, topk)
        if fn < f:
            c, f = hn, fn
            step = min(step * 1.02, step0 * 5.0)
            c5n = _eval(c)
            if c5n < best_c5:
                best_h, best_c5 = c.copy(), c5n
        else:
            step *= 0.5
            if step < 1e-12:
                break
            c = _proj(hp - step * v, n * 0.5)
            f = _sobj(c, alpha, topk)
    return best_h, best_c5

def _anneal(h, niter, rng, deadline, T0, T1, blocks):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0 + bs].copy()
        d = 0.001 * max(T, 1e-9) ** 0.5 * rng.randn(bs)
        d -= d.mean()
        seg = np.clip(seg + d, 0, 1)
        hn = h.copy()
        hn[i0:i0 + bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or (T > 1e-12 and rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12))):
            h, c5c = hn, c5n
            if c5n < bc:
                bc, bh = c5n, h.copy()
    return bh, bc

def _polish(h, niter, rng, deadline):
    nc = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    sizes = [3e-7, 1e-7, 5e-8, 2e-8, 1e-8, 5e-9, 2e-9]
    sps = max(1, niter // len(sizes))
    for ss in sizes:
        for _ in range(sps):
            if time.time() > deadline:
                return bh, bc
            i = rng.randint(nc)
            j = (i + 1 + rng.randint(0, 4)) % nc
            d = ss * (2.0 * rng.random() - 1.0)
            ni, nj = h[i] + d, h[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-18:
                    c5c = c5n
                    if c5n < bc:
                        bc, bh = c5n, h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d
    return bh, bc

def run(seed=42, budget_s=1100, **kw):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 4
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h, best_c5 = h.copy(), _eval(h)

    # Phase 1: Nesterov warmup
    dl1 = min(deadline, t0 + 50)
    hv, cv = _nest(best_h.copy(), 5000, 20, 2000, 0.002, dl1)
    if cv < best_c5:
        best_c5, best_h = cv, hv

    # Phase 2: LP coarse (geometrically decreasing eps)
    dl2 = min(deadline, t0 + 280)
    hv, cv = _lp_ph2(best_h.copy(), 15000, 0.005, 12, dl2)
    if cv < best_c5:
        best_c5, best_h = cv, hv

    # Phase 3: Nesterov mid
    dl3 = min(deadline, t0 + 420)
    hv, cv = _nest(best_h.copy(), 15000, 10, 3000, 0.001, dl3)
    if cv < best_c5:
        best_c5, best_h = cv, hv

    # Phase 4: LP mid-fine
    dl4 = min(deadline, t0 + 600)
    hv, cv = _lp_ph2(best_h.copy(), 20000, 0.002, 10, dl4)
    if cv < best_c5:
        best_c5, best_h = cv, hv

    # Phase 5: Nesterov ultra-fine
    dl5 = min(deadline, t0 + 720)
    hv, cv = _nest(best_h.copy(), 100000, 5, 3000, 0.0004, dl5)
    if cv < best_c5:
        best_c5, best_h = cv, hv

    # Phase 6: LP ultra-fine
    dl6 = min(deadline, t0 + 820)
    hv, cv = _lp_ph2(best_h.copy(), 20000, 0.0008, 8, dl6)
    if cv < best_c5:
        best_c5, best_h = cv, hv

    # Phase 7: Anneal
    dl7 = min(deadline, t0 + 920)
    hv, cv = _anneal(best_h.copy(), 20000, rng, dl7, 0.0005, 1e-12, [1, 2, 3, 4, 5, 8])
    if cv < best_c5:
        best_c5, best_h = cv, hv
        if time.time() < dl7:
            hv2, cv2 = _lp_ph2(best_h.copy(), 3000, 0.0005, 6, dl7)
            if cv2 < best_c5:
                best_c5, best_h = cv2, hv2

    # Phase 8: Final polish
    rem = max(0, deadline - time.time())
    npol = min(int(rem / 0.0004), 100000)
    best_h, best_c5 = _polish(best_h, npol, rng, deadline - 1)

    # Final micro-LP
    if time.time() < deadline - 1:
        hv, cv = _lp_tmin(best_h, 0.0003, 5)
        if cv < best_c5:
            best_c5, best_h = cv, hv

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
