    H_pert = H + (scale * rng.randn(len(H)) + 1j * scale * rng.randn(len(H))) * mask
    h_pert = np.fft.irfft(H_pert, n)
    return _repair(h_pert, n)

def _binary_snap(h, rng, frac):
    n = len(h)
    h_s = h.copy()
    idx = rng.choice(n, size=int(n * frac), replace=False)
    for i in idx:
        h_s[i] = 0.0 if h[i] < 0.5 else 1.0
    return _repair(h_s, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values))
    n = len(h0)
    best_h = h0.copy()
    best_c5 = _eval(h0)

    def _el():
        return time.time() - t0

    # Phase 1: L-BFGS-B sweep with increasing sharpness
    p1_configs = [
        (200, 60, 400), (500, 40, 500), (1000, 30, 600),
        (3000, 20, 600), (8000, 15, 600), (20000, 10, 600),
        (50000, 8, 500), (100000, 6, 400), (300000, 5, 300),
    ]
    for alpha, tk, mxi in p1_configs:
        if _el() > 280:
            break
        h_opt = _lbf_run(best_h.copy(), alpha, tk, mxi)
        c = _eval(h_opt)
        if c < best_c5:
            best_c5 = c
            best_h = h_opt.copy()

    # Phase 2: Spectral diversification + L-BFGS
    if _el() < budget_s - 400:
        for trial in range(8):
            if _el() > budget_s - 350:
                break
            scale = 0.02 + 0.015 * trial
            h_pert = _spectral_perturb(best_h, rng, scale)
            c_pert = _eval(h_pert)
            if c_pert < best_c5:
                best_c5 = c_pert
                best_h = h_pert.copy()
            for alpha, tk, mxi in [(5000, 12, 400), (20000, 8, 400)]:
                if _el() > budget_s - 300:
                    break
                h_opt = _lbf_run(h_pert, alpha, tk, mxi)
                c = _eval(h_opt)
                if c < best_c5:
                    best_c5 = c
                    best_h = h_opt.copy()

    # Phase 3: Binary snap + L-BFGS
    if _el() < budget_s - 300:
        for frac in [0.15, 0.3, 0.5]:
            if _el() > budget_s - 250:
                break
            h_bin = _binary_snap(best_h, rng, frac)
            c_bin = _eval(h_bin)
            if c_bin < best_c5:
                best_c5 = c_bin
                best_h = h_bin.copy()
            for alpha, tk, mxi in [(10000, 10, 500), (50000, 6, 400)]:
                if _el() > budget_s - 200:
                    break
                h_opt = _lbf_run(h_bin, alpha, tk, mxi)
                c = _eval(h_opt)
                if c < best_c5:
                    best_c5 = c
                    best_h = h_opt.copy()

    # Phase 4: Final L-BFGS polish
    if _el() < budget_s - 250:
        for alpha, tk, mxi in [(100000, 5, 300), (500000, 4, 250), (1000000, 3, 200)]:
            if _el() > budget_s - 200:
                break
            h_opt = _lbf_run(best_h.copy(), alpha, tk, mxi)
            c = _eval(h_opt)
            if c < best_c5:
                best_c5 = c
                best_h = h_opt.copy()

    # Phase 5: Shift-targeted local search
    if _el() < budget_s - 200:
        n_iter = min(int((budget_s - 200 - _el()) * 1200), 120000)
        best_h, best_c5 = _shift_local(best_h, n_iter, 3e-5, rng)

    # Phase 6: Random-pair local search
    if _el() < budget_s - 150:
        n_iter = min(int((budget_s - 150 - _el()) * 2500), 200000)
        best_h, best_c5 = _local2(best_h, n_iter, 4e-5, rng)

    # Phase 7: Ultra-fine polish
    if _el() < budget_s - 30:
        n_iter = min(int((budget_s - 30 - _el()) * 5000), 400000)
        best_h, best_c5 = _local2(best_h, n_iter, 2e-6, rng)

    # Phase 8: Final micro L-BFGS + polish
    if _el() < budget_s - 5:
        h_opt = _lbf_run(best_h.copy(), 1000000, 3, 150)
        c = _eval(h_opt)
        if c < best_c5:
            best_c5 = c
            best_h = h_opt.copy()
        n_iter = min(int((budget_s - 5 - _el()) * 6000), 300000)
        best_h, best_c5 = _local2(best_h, n_iter, 1e-6, rng)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
