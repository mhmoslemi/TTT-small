    dx = 2.0/n
    # Objective: -sum_k w_k * sum_i h[i]*h[i+k] * dx
    # (we want to maximize the weighted autocorrelation)
    val = 0.0
    for k, w in enumerate(weights):
        if w < 1e-12:
            continue
        sh = (n-1) - k  # actual shift
        if sh == 0:
            val -= w * np.sum(h*h) * dx
        elif sh > 0:
            val -= w * np.sum(h[:n-sh] * h[sh:]) * dx
        else:
            sp = -sh
            val -= w * np.sum(h[sp:] * h[:n-sp]) * dx
    return val

    """Compute sum_k w_k * grad c_k"""
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(len(weights)):
        w = weights[k]
        if w < 1e-14:
            continue
        sh = (n - 1) - k
        if sh >= 0:
            i1 = np.arange(n - sh)
            g[i1] += w * (1.0 - h[i1 + sh])
            if sh > 0:
                i2 = np.arange(sh, n)
                g[i2] -= w * h[i2 - sh]
        else:
            sp = -sh
            i1 = np.arange(sp, n)
            g[i1] += w * (1.0 - h[i1 - sp])
            i2 = np.arange(n - sp)
            g[i2] -= w * h[i2 + sp]
    return g * dx

def _nesterov_pg(h, alpha, steps, step0, momentum, top_k, time_limit):
    """Nesterov-accelerated projected gradient"""
    t0 = time.time()
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    v = np.zeros(n)
    step = step0
    cur_c = _eval(h)
    
    for it in range(steps):
        if time.time() - t0 > time_limit:
            break
        c = _corr(h)
        # Weight for soft-max
        c_shifted = c - c.max()
        s = np.clip(alpha * c_shifted, -400, 400)
        w = np.exp(s)
        w /= w.sum()
        # Focus on top_k
        if top_k > 0 and top_k < n:
            idx = np.argpartition(c, -top_k)[-top_k:]
            w_full = np.zeros_like(w)
            w_full[idx] = w[idx]
            w_full /= w_full.sum()
            w = w_full
        
        g = _weighted_grad(h, w, n)
        v = momentum * v + (1.0 - momentum) * g
        # Line search
        taken = False
        for _t in range(12):
            h_new = _proj_sum(h - step * v, n * 0.5)
            c_new = _eval(h_new)
            if c_new < cur_c - 1e-16:
                h = h_new
                cur_c = c_new
                step = min(step * 1.05, step0 * 5.0)
                taken = True
                break
            step *= 0.45
        if not taken:
            step = min(step * 1.3, step0 * 5.0)
    
    return _repair(h, n), cur_c

def _slsqp_smooth(h, alpha, maxiter, top_k, time_limit):
    """SLSQP with log-sum-exp objective"""
    t0 = time.time()
    n = len(h)
    dx = 2.0 / n
    
    def obj_fn(x):
        if time.time() - t0 > time_limit:
            return _eval(x)
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        if top_k > 0 and top_k < len(c):
            idx = np.argpartition(c, -top_k)[-top_k:]
            cs = c[idx]
        else:
            cs = c
        m = cs.max()
        s = np.clip(alpha * (cs - m), -400, 400)
        s -= s.max()
        return m + np.log(np.exp(s).sum()) / alpha
    
    def grad_fn(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        if top_k > 0 and top_k < len(c):
            idx = np.argpartition(c, -top_k)[-top_k:]
            cs = c[idx]
        else:
            all_idx = np.arange(len(c))
            cs = c
            idx = all_idx
        m = cs.max()
        s = np.clip(alpha * (cs - m), -400, 400)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        g = np.zeros(n)
        for ii in range(len(idx)):
            if w[ii] < 1e-14:
                continue
            k = idx[ii]
            sh = (n - 1) - k
            wv = w[ii]
            if sh >= 0:
                i1 = np.arange(n - sh)
                g[i1] += wv * (1.0 - x[i1 + sh])
                if sh > 0:
                    i2 = np.arange(sh, n)
                    g[i2] -= wv * x[i2 - sh]
            else:
                sp = -sh
                i1 = np.arange(sp, n)
                g[i1] += wv * (1.0 - x[i1 - sp])
                i2 = np.arange(n - sp)
                g[i2] -= wv * x[i2 + sp]
        return g * dx
    
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj_fn, h, jac=grad_fn,
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(res.x, n)
    except:
        return h

def _sim_anneal_binary(h, n_iter, rng, t0_temp, t1_temp, time_limit):
    """Simulated annealing on near-binary states"""
    n = len(h)
    t0 = time.time()
    h = np.where(h > 0.5, 1.0, 0.0).copy()
    # Fix sum
    n_ones = int(h.sum())
    target = n // 2
    if n_ones != target:
        if n_ones > target:
            idx = np.where(h == 1)[0]
            rng.shuffle(idx)
            for i in range(n_ones - target):
                h[idx[i]] = 0
        else:
            idx = np.where(h == 0)[0]
            rng.shuffle(idx)
            for i in range(target - n_ones):
                h[idx[i]] = 1
    
    cur_c = _eval(h)
    best_h = h.copy()
    best_c = cur_c
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(n_iter):
        if time.time() - t0 > time_limit:
            break
        T = t0_temp * (t1_temp / t0_temp) ** (it / max(n_iter, 1))
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0, 1
        new_c = _eval(h)
        delta = new_c - cur_c
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            cur_c = new_c
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            h[i], h[j] = 1, 0
    
    return best_h, best_c

def _local_fine(h, n_iter, rng, base_step, worst_lag=None):

def _downsample(h, factor):

def _downsample(h, factor):
    n = len(h)
    m = n // factor
    return h[:m * factor].reshape(m, factor).mean(axis=1).copy()

def _upsample(h, factor):
    return np.repeat(h, factor)

def _perturb_reopt(h, rng, scale, slsqp_alpha, slsqp_iter, time_limit):
    n = len(h)
    h_p = h + scale * rng.randn(n)
    h_p = _repair(h_p, n)
    h_new = _slsqp_smooth(h_p, slsqp_alpha, slsqp_iter, 3, time_limit)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    n = len(initial_h_values)
    best_h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_c5 = _eval(best_h)

    # Phase 1: Coarse-to-fine (25s)
    for ratio in [8, 4, 2]:
        if _el() > 25:
            break
        nc = max(24, n // ratio)
        hc = _repair(_downsample(best_h, n // nc), nc)
        hc = _slsqp_smooth(hc, 5000, 80, 0, 5)
        h_up = _repair(_upsample(hc, n // nc), n)
        c5n = _eval(h_up)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_up.copy()

    # Phase 2: SLSQP rounds (120s)
    for alpha, tk, mit in [(5000, 0, 200), (20000, 6, 150), (100000, 4, 120), (500000, 3, 80), (2000000, 2, 60)]:
        if _el() > 145:
            break
        h_new = _slsqp_smooth(best_h.copy(), alpha, mit, tk, 25)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new.copy()

    # Phase 3: Nesterov PG (200s)
    pg_cfgs = [(3000, 800, 0.005, 0.8, 8), (10000, 600, 0.003, 0.85, 6),
               (50000, 500, 0.002, 0.9, 4), (200000, 400, 0.001, 0.9, 3),
               (1000000, 300, 0.0005, 0.95, 2)]
    for alpha, steps, s0, mu, tk in pg_cfgs:
        if _el() > 350:
            break
        h_new, c5n = _nesterov_pg(best_h.copy(), alpha, steps, s0, mu, tk, 35)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new.copy()

    # Phase 4: Perturb-restart cycle (100s)
    if _el() < budget_s - 350:
        for scale in [0.01, 0.005, 0.002]:
            if _el() > budget_s - 300:
                break
            h_new = _perturb_reopt(best_h, rng, scale, 50000, 60, 20)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new.copy()
            else:
                h_new2 = _slsqp_smooth(best_h.copy(), 200000, 60, 2, 15)
                c5n2 = _eval(h_new2)
                if c5n2 < best_c5:
                    best_c5 = c5n2
                    best_h = h_new2.copy()

    # Phase 5: Alternating focused + free local search (300s)
    if _el() < budget_s - 80:
        steps_per_round = 80000
        base_steps = [2e-5, 1e-5, 5e-6, 2e-6, 1e-6, 5e-7]
        for si, bs in enumerate(base_steps):
            # Focused on worst lag
            c_arr = _corr(best_h)
            wk = int(np.argmax(c_arr))
            h_new, c5n = _local_focused(best_h.copy(), steps_per_round, rng, bs, wk)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new.copy()
            # Free search
            h_new, c5n = _local_free(best_h.copy(), steps_per_round, rng, bs)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new.copy()
            if _el() > budget_s - 80:
                break
            # Re-optimize with SLSQP
            if _el() < budget_s - 120:
                h_new = _slsqp_smooth(best_h.copy(), 500000, 50, 2, 15)
                c5n = _eval(h_new)
                if c5n < best_c5:
                    best_c5 = c5n
                    best_h = h_new.copy()

    # Phase 6: Ultra-fine polish
    if _el() < budget_s - 2:
        time_left = budget_s - 2 - _el()
        _t = time.time()
        for _ in range(30):
            _eval(best_h)
        eps = max((time.time() - _t) / 30.0, 1e-6)
        evals_per_s = 1.0 / eps
        npol = min(int(time_left * evals_per_s * 0.7), 8000000)
        # Alternate tiny focused and free
        c_arr = _corr(best_h)
        wk = int(np.argmax(c_arr))
        half = npol // 2
        h_new, c5n = _local_focused(best_h.copy(), half, rng, 3e-7, wk)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new.copy()
        h_new, c5n = _local_free(best_h.copy(), npol - half, rng, 3e-7)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new.copy()

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
