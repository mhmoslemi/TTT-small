                        best_c5, best_h = c5_new, c.copy()
                    break
            if not improved:
                break
    return best_h, best_c5

def _block_perturb(h, rng, deadline, n_iters=20000):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    blocks = [1, 2, 3, 4, 5, 6, 8, 12]
    for it in range(n_iters):
        if time.time() > deadline:
            break
        T = 0.004 * (1e-10 / 0.004) ** (it / max(n_iters, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0+bs].copy()
        d = 0.004 * max(T, 1e-9) ** 0.5 * rng.randn(bs)
        d -= d.mean()
        seg = np.clip(seg + d, 0, 1)
        hn = h.copy()
        hn[i0:i0+bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h, c5c = hn, c5n
            if c5n < bc:
                bc, bh = c5n, h.copy()
    return bh, bc

def _coord_polish(h, deadline, n_passes=None):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    steps = [1e-5, 5e-6, 2e-6, 1e-6, 5e-7, 2e-7, 1e-7, 5e-8]
    for ss in steps:
        if time.time() > deadline:
            break
        ni_pass = max(200, int(3000 * (5e-6 / ss) ** 0.3))
        for _ in range(ni_pass):
            if time.time() > deadline:
                return bh, bc
            i = np.random.RandomState(int(time.time() * 1e5) % 999999).randint(n)
            j = (i + 1 + np.random.RandomState(int(time.time() * 1e7) % 88888) * 4) % n
            d = ss * (2.0 * np.random.random() - 1.0)
            ni, nj = h[i] + d, h[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-19:
                    c5c = c5n
                    if c5n < bc:
                        bc, bh = c5n, h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d
    return bh, bc

def _fine_lp(h, deadline, max_iters=8000):
    n = len(h)
    best_h, best_c5 = h.copy(), _eval(h)
    c = h.copy()
    for it in range(max_iters):
        if time.time() > deadline:
            break
        cor = _correl(c)
        m_c5 = cor.max()
        m_top = 5
        idx_top = np.argpartition(cor, -m_top)[-m_top:]
        idx_top = idx_top[np.argsort(-cor[idx_top])]
        grads = np.array([_grad_k(c, k, n) for k in idx_top])
        c_obj = -grads[0]
        A_ub = (grads[1:] - grads[0][None, :])
        b_ub = np.zeros(m_top - 1)
        A_eq = np.ones((1, n))
        b_eq = np.zeros(1)
        eps = 0.0008
        bnds = [(-eps, eps)] * n
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bnds, method='highs', options={'presolve': True})
        except Exception:
            break
        if not res.success or res.x is None or np.max(np.abs(res.x)) < 1e-14:
            break
        d = res.x
        improved = False
        for step in [1.0, 0.7, 0.5, 0.3, 0.15]:
            h_new = _repair(c + step * d, n)
            c5_new = _eval(h_new)
            if c5_new < m_c5 - 3e-8:
                c, m_c5 = h_new, c5_new
                improved = True
                if c5_new < best_c5:
                    best_c5, best_h = c5_new, c.copy()
                break
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 10
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h, best_c5 = h.copy(), _eval(h)

    dl1 = min(deadline, t0 + 400)
    hv, cv = _lp_adaptive(best_h.copy(), dl1, max_iters=30000)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()

    dl2 = min(deadline, t0 + 520)
    hv, cv = _wide_lp(best_h.copy(), dl2, n_rounds=3)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()

    dl3 = min(deadline, t0 + 700)
    hv, cv = _block_perturb(best_h.copy(), rng, dl3, n_iters=25000)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()

    dl4 = min(deadline, t0 + 820)
    hv, cv = _lp_adaptive(best_h.copy(), dl4, max_iters=10000)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()

    dl5 = min(deadline, t0 + 950)
    hv, cv = _fine_lp(best_h.copy(), dl5, max_iters=8000)
    if cv < best_c5:
        best_c5, best_h = cv, hv.copy()

    best_h, best_c5 = _coord_polish(best_h.copy(), deadline)
    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
