def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:

                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a

                    a = min(-diff, h[idx]); h[idx] -= a; diff += a

    return h

    idx_top = np.argpartition(cor, -m_top)[-m_top:]
    idx_top = idx_top[np.argsort(-cor[idx_top])]
    grads = np.array([_sg(h, k, n) for k in idx_top])

    for attempt in range(n_try):

        b_ub = np.zeros(m_top - 1) if m_top > 1 else None
        A_eq = np.ones((1, n))
        b_eq = np.zeros(1)

                          bounds=bnds, method='highs', options={'presolve': True})
            if res.success and res.x is not None:

                break

    s = np.clip(alpha * (ct - m), -400, 400)

def _repair(h, n):

    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

            lo = mid
        else:
