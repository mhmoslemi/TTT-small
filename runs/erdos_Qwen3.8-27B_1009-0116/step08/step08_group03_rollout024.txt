    n = len(h)
    N = _fft_size(n)
    H = np.fft.rfft(h, N)
    ac = np.fft.irfft(H * np.conj(H), N).real[:n]
    pref = np.empty(n + 1)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    total = pref[n]
    P = np.empty(n)
    P[0] = total - ac[0]
    Nv = np.zeros(n)
    if n > 1:
        idx = np.arange(1, n)
        P[idx] = pref[n - idx] - ac[idx]
        Nv[idx] = total - pref[idx] - ac[idx]
        mx = float(max(np.max(P), np.max(Nv)))
    else:
        mx = float(P[0])
    return P, Nv, mx

def _c5_direct(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _smooth_value(h, alpha):
    P, Nv, mx = _pn_arrays(h)
    n = len(h)
    dx = 2.0 / n
    e = np.exp(np.clip(alpha * (P[0] - mx), -400.0, 0.0))
    if n > 1:
        e += np.exp(np.clip(alpha * (P[1:] - mx), -400.0, 0.0)).sum()
        e += np.exp(np.clip(alpha * (Nv[1:] - mx), -400.0, 0.0)).sum()
    return dx * (mx + np.log(e) / alpha)

def _grad_from_weights(h, wP, wN):
    n = len(h)
    dx = 2.0 / n
    u = np.zeros(n)
    if n > 1:
        u[1:] = wP[1:] + wN[1:]
    cp = np.zeros(n)
    cn = np.zeros(n)
    if n > 1:
        cp[1:] = np.cumsum(wP[1:])
        cn[1:] = np.cumsum(wN[1:])
    linP = cp[n - 1::-1]
    linN = cn
    q = np.zeros(n)
    if n > 1 and np.any(u[1:]):
        Nf = _fft_size(n)
        H = np.fft.rfft(h, Nf)
        Urev = np.fft.rfft(u[::-1], Nf)
        Ur = np.fft.rfft(u, Nf)
        R = np.fft.irfft(H * Urev, Nf)[n - 1:2 * n - 1].real
        L = np.fft.irfft(H * Ur, Nf)[:n].real
        q = -(R + L)
    return dx * (wP[0] * (1.0 - 2.0 * h) + linP + linN + q)

def _smooth_grad(h, alpha):
    P, Nv, mx = _pn_arrays(h)
    n = len(h)
    wP = np.zeros(n)
    wN = np.zeros(n)
    e0 = np.exp(np.clip(alpha * (P[0] - mx), -400.0, 0.0))
    tot = e0
    if n > 1:
        ep = np.exp(np.clip(alpha * (P[1:] - mx), -400.0, 0.0))
        en = np.exp(np.clip(alpha * (Nv[1:] - mx), -400.0, 0.0))
        tot += ep.sum() + en.sum()
        if tot <= 0.0:
            tot = 1.0
            e0 = 1.0
        wP[0] = e0 / tot
        wP[1:] = ep / tot
        wN[1:] = en / tot
    else:
        wP[0] = 1.0
    return _grad_from_weights(h, wP, wN)

def _active_grad_C(h, P, Nv):
    n = len(h)
    dx = 2.0 / n
    mx = float(P[0])
    if n > 1:
        mx = max(mx, float(np.max(P[1:])), float(np.max(Nv[1:])))
    tol = max(1e-9, 5.0 * dx)
    wP = np.zeros(n)
    wN = np.zeros(n)
    cnt = 0
    if P[0] >= mx - tol:
        wP[0] = 1.0
        cnt += 1
    if n > 1:
        mP = P[1:] >= mx - tol
        if np.any(mP):
            pos = np.nonzero(mP)[0] + 1
            wP[pos] = 1.0
            cnt += int(pos.size)
        mN = Nv[1:] >= mx - tol
        if np.any(mN):
            pos = np.nonzero(mN)[0] + 1
            wN[pos] = 1.0
            cnt += int(pos.size)
    if cnt == 0:
        return np.zeros(n)
    inv = 1.0 / cnt
    wP *= inv
    wN *= inv
    return _grad_from_weights(h, wP, wN)

def _fista_smooth(h0, alpha, max_time, L=8.0):
    if max_time <= 0.0:
        h = h0.copy()
        return h, h.copy(), _c5_direct(h)
    n = len(h0)
    S = n * 0.5
    t0 = time.time()
    h = _repair(h0, n)
    hp = h.copy()
    y = h.copy()
    t = 1.0
    fy = _smooth_value(h, alpha)
    fh = fy
    best_h = h.copy()
    best_c = _c5_direct(h)
    stale = 0
    Lcur = L
    for it in range(500000):
        if time.time() - t0 > max_time:
            break
        if it > 0:
            tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
            y = h + ((t - 1.0) / tn) * (h - hp)
            t = tn
            if y.min() < -1e-4 or y.max() > 1.0 + 1e-4 or abs(y.sum() - S) > 1e-7:
                y = _proj(y, S)
            fy = _smooth_value(y, alpha)
        else:
            y = h
            fy = fh
        g = _smooth_grad(y, alpha)
        step = 1.0 / max(Lcur, 1e-12)
        accepted = False
        for _ in range(16):
            xn = _proj(y - step * g, S)
            fx = _smooth_value(xn, alpha)
            d = xn - y
            rhs = fy + float(np.dot(g, d)) + 0.5 * (1.0 / step) * float(np.dot(d, d))
            if fx <= rhs + 1e-12:
                accepted = True
                break
            step *= 0.4
        Lcur = 1.0 / max(step, 1e-12)
        if accepted:
            hp = h
            h = xn
            fh = fx
            stale = 0
            Lcur = max(Lcur * 0.9, 1.0)
            if it % 5 == 0:
                c = _c5_direct(h)
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
        else:
            stale += 1
            hp = h
            y = h
            t = 1.0
            fy = fh
            Lcur = max(Lcur * 0.5, 1.0)
            if stale > 15:
                break
    c = _c5_direct(h)
    if c < best_c:
        best_c = c
        best_h = h.copy()
    return h, best_h, best_c

def _resample(h, m):
    n = len(h)
    if n == m:
        return _repair(h, n)
    x0 = (np.arange(n) + 0.5) * (2.0 / n)
    x1 = (np.arange(m) + 0.5) * (2.0 / m)
    out = np.interp(x1, x0, h)
    return _proj(out, m * 0.5)

def _run_alphas(h0, alphas, deadline, L=8.0):
    hcur = h0.copy()
    best_h = h0.copy()
    best_c = _c5_direct(best_h)
    total_time = max(0.0, deadline - time.time())
    per = total_time / max(1, len(alphas))
    for a in alphas:
        if deadline - time.time() < 3.0 or per <= 0.0:
            break
        hf, bh, bc = _fista_smooth(hcur, a, per, L)
        chf = _c5_direct(hf)
        chb = _c5_direct(bh)
        if bc < best_c:
            best_c = bc
            best_h = bh
        if chf < best_c:
            best_c = chf
            best_h = hf
        hcur = hf if chf <= chb else bh
    return best_h, best_c

def _subgrad_search(h0, rng, deadline):
    n = len(h0)
    S = n * 0.5
    h = h0.copy()
    c = _c5_direct(h)
    best_c = c
    best_h = h.copy()
    step0 = 3e-5
    step = step0
    eval_count = 0
    last_improve = 0
    t0 = time.time()

    while time.time() < deadline:
        P, Nv, _ = _pn_arrays(h)
        g = _active_grad_C(h, P, Nv)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break

        # Projected negative gradient step
        direction = _proj(h - step * g, S)
        dn = direction - h
        dnn = float(np.dot(dn, dn))
        if dnn < 1e-20:
            # Try random pair move
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                continue
            s = 1 if g[i] <= g[j] else -1
            avail = min(1.0 - h[i], h[j]) if s > 0 else min(h[i], 1.0 - h[j])
            if avail > 1e-14:
                d = s * min(avail, step * 3.0 * rng.random())
                h[i] += d
                h[j] -= d
                cn = _c5_direct(h)
                eval_count += 1
                if cn < c - 1e-15:
                    c = cn
                    if c < best_c - 1e-15:
                        best_c = c
                        best_h = h.copy()
                        last_improve = eval_count
                else:
                    h[i] -= d
                    h[j] += d
            continue

        cn = _c5_direct(direction)
        eval_count += 1
        if cn < c - 1e-14:
            c = cn
            h = direction
            if c < best_c - 1e-15:
                best_c = c
                best_h = h.copy()
                last_improve = eval_count
        else:
            # Random pair move as fallback
            order = np.argsort(g)
            q = max(4, n // 12)
            if rng.random() < 0.65:
                i = int(order[rng.randint(q)])
                j = int(order[n - 1 - rng.randint(q)])
            else:
                i = int(rng.randint(n))
                j = int(rng.randint(n))
            if i != j:
                s = 1 if g[i] <= g[j] else -1
                avail = min(1.0 - h[i], h[j]) if s > 0 else min(h[i], 1.0 - h[j])
                if avail > 1e-14:
                    d = s * min(avail, step * (0.5 + 2.0 * rng.random()))
                    if rng.random() < 0.03:
                        d *= 6.0
                    h[i] += d
                    h[j] -= d
                    cn2 = _c5_direct(h)
                    eval_count += 1
                    if cn2 < c - 1e-15:
                        c = cn2
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            last_improve = eval_count
                    else:
                        h[i] -= d
                        h[j] += d

        # Adapt step size
        if eval_count - last_improve > 20000:
            step *= 0.7
            last_improve = eval_count
        elif eval_count - last_improve > 5000:
            step *= 0.9
        if step < 1e-12:
            break
    return best_h, best_c

def _cosine_init(n, freq, phase, rng):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = 0.5 + 0.35 * np.cos(2.0 * np.pi * freq * x / 2.0 + phase)
    h = h + 0.05 * rng.standard_normal(n)
    return _repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(560, 0.5)
    n0 = len(h0)
    h0 = _repair(h0, n0)

    best = h0.copy()
    best_c = _c5_direct(best)

    # Try a few diverse initializations and pick the best after quick FISTA
    inits = [best]
    for freq, ph in [(1, 0), (2, 0.5), (3, 1.0)]:
        ci = _cosine_init(n0, freq, ph, rng)
        inits.append(ci)

    end_init = t0 + min(40.0, 0.05 * budget_s)
    if end_init > time.time() + 1.0:
        per_init = (end_init - time.time()) / max(1, len(inits) - 1)
        for ci in inits[1:]:
            if time.time() >= end_init:
                break
            hf, bh, bc = _fista_smooth(ci, 50.0, per_init, 8.0)
            chf = _c5_direct(hf)
            if bc < best_c:
                best_c = bc
                best = bh
            if chf < best_c:
                best_c = chf
                best = hf

    # Phase 2: Multi-resolution (downsample, optimize, upsample)
    end2 = t0 + min(250.0, 0.25 * budget_s)
    if end2 > time.time() + 10.0:
        for m in (280, 140):
            if time.time() >= end2 - 5.0:
                break
            hm = _resample(best, m)
            b2, c2 = _run_alphas(hm, [15, 80, 400, 2000], end2 - 3.0)
            if c2 < best_c:
                best = b2
                best_c = c2
            hu = _resample(best, n0)
            cu = _c5_direct(hu)
            if cu < best_c:
                best = hu
                best_c = cu

    # Phase 3: High-precision FISTA at full resolution with many alphas
    end3 = t0 + budget_s * 0.62
    if end3 > time.time() + 15.0:
        alphas3 = [15, 60, 250, 1000, 4000, 20000, 80000, 300000]
        b3, c3 = _run_alphas(best, alphas3, end3)
        if c3 < best_c:
            best = b3
            best_c = c3

    # Phase 4: Subgradient local search
    end4 = t0 + budget_s - 8.0
    if end4 > time.time() + 5.0:
        best, best_c = _subgrad_search(best, rng, end4)

    # Phase 5: Final variants and polish
    b5_end = t0 + budget_s - 0.5
    if b5_end > time.time():
        variants = [best, 1.0 - best, best[::-1],
                    0.5 + 0.5 * (best - best[::-1]),
                    0.5 + 0.5 * (best + best[::-1] - 1.0)]
        for hh in variants:
            h2 = _proj(hh, len(hh) * 0.5)
            c = _c5_direct(h2)
            if c < best_c:
                best = h2
                best_c = c

    best = _repair(best, len(best))
    best_c = _c5_direct(best)
    return best, best_c, len(best)
