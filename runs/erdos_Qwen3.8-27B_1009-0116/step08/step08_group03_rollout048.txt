        if zeros.size < 2 or ones.size < 2:
            break
        i = int(zeros[rng.randint(len(zeros))])
        j = int(ones[rng.randint(len(ones))])
        if g[i] > g[j]:
            i, j = j, i
        h[i] = 1.0 - h[i]
        h[j] = 1.0 - h[j]
        cn = _c5(h)
        if cn < c - 1e-15:
            c = cn
            if c < best_c:
                best_c = c
                best_h = h.copy()
        else:
            h[i] = 1.0 - h[i]
            h[j] = 1.0 - h[j]
        it += 1
    return best_h, best_c

def _resample(h, m):
    n = len(h)
    if n == m:
        return _repair(h, n)
    x0 = (np.arange(n) + 0.5) * (2.0 / n)
    x1 = (np.arange(m) + 0.5) * (2.0 / m)
    out = np.interp(x1, x0, h)
    return _proj(out, m * 0.5)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(560, 0.5)
    n0 = len(h0)
    h0 = _repair(h0, n0)
    best = h0.copy()
    best_c = _c5(best)
    for vv in (1.0 - h0, h0[::-1], 0.5 + 0.5*(h0 - h0[::-1]), 0.5 + 0.5*(h0 + h0[::-1] - 1.0)):
        hv = _proj(vv, n0 * 0.5)
        cv = _c5(hv)
        if cv < best_c:
            best = hv
            best_c = cv
    end1 = t0 + min(90.0, 0.09 * budget_s)
    if end1 > time.time() + 3.0:
        hcur = best.copy()
        for a in (30, 150, 800, 4000, 20000):
            rem = end1 - time.time() - 1.0
            if rem < 2.0:
                break
            hf, _ = _fista(hcur, a, rem * 0.2, 20.0)
            cf = _c5(hf)
            if cf < best_c:
                best = hf
                best_c = cf
            hcur = hf
    end2 = t0 + min(240.0, 0.28 * budget_s)
    if end2 > time.time() + 3.0:
        hcur = best.copy()
        for a in (60000, 300000, 1500000, 8000000, 40000000):
            rem = end2 - time.time() - 1.0
            if rem < 2.0:
                break
            hf, _ = _fista(hcur, a, rem * 0.2, 20.0)
            cf = _c5(hf)
            if cf < best_c:
                best = hf
                best_c = cf
            hcur = hf
    end3 = t0 + min(370.0, 0.42 * budget_s)
    if end3 > time.time() + 5.0:
        for m in (n0 // 2, n0 // 4, max(96, n0 * 3 // 2)):
            if time.time() >= end3 - 4.0:
                break
            slot = min(35.0, end3 - time.time() - 2.0)
            if slot < 4.0:
                break
            hm = _resample(best, m)
            hcur = hm.copy()
            per = slot / 3.0
            for a in (200, 5000, 200000):
                rem = end3 - time.time() - 2.0
                if rem < 2.0:
                    break
                hf, _ = _fista(hcur, a, min(per, rem * 0.35), 20.0)
                hcur = hf
            if m != n0:
                hu = _resample(hcur, n0)
                cu = _c5(hu)
            else:
                hu = hcur
                cu = _c5(hu)
            if cu < best_c:
                best = hu
                best_c = cu
    end4 = t0 + min(550.0, 0.60 * budget_s)
    if end4 > time.time() + 5.0:
        hb, cb = _binary_search(best, rng, t0 + min(450.0, 0.50 * budget_s))
        if cb < best_c:
            best = hb
            best_c = cb
        best, best_c = _local_search(best, rng, end4)
    end5 = t0 + budget_s - 3.0
    if end5 > time.time() + 5.0:
        best, best_c = _local_search(best, rng, end5)
    for vv in (1.0 - best, best[::-1]):
        hv = _proj(vv, len(best) * 0.5)
        cv = _c5(hv)
        if cv < best_c:
            best = hv
            best_c = cv
    best = _repair(best, len(best))
    best_c = _c5(best)
    return best, best_c, len(best)
