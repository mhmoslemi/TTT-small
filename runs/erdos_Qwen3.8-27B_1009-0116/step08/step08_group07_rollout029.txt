import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) < 1e-12:
        return h
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _proj2(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smax_weights(c, m, alpha, topk):
    n_c = len(c)
    if topk < n_c:
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        idx = np.arange(n_c)
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    return idx, w, ct

def _smax_obj(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    _, w, ct = _smax_weights(c, m, alpha, topk)
    s = np.clip(alpha * (ct - m), -400, 400)
    s -= s.max()
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smax_grad(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    idx, w, _ = _smax_weights(c, m, alpha, topk)
    dx = 2.0 / n
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-14:
            continue
        lag = idx[ii] - (n - 1)
        wk = w[ii] * dx
        if lag >= 0:
            g[:n - lag] += wk * (1.0 - h[lag:])
            if lag > 0:
                g[lag:] -= wk * h[:n - lag]
        else:
            l = -lag
            g[l:] += wk * (1.0 - h[:n - l])
            g[:n - l] -= wk * h[l:]
    return g

def _nest(h, alpha, topk, steps, dl, beta0=0.88, step0=0.003):
    n = len(h)
    target = n * 0.5
    f = _eval(h)
    best_h, best_f = h.copy(), f
    v = np.zeros(n)
    step = step0
    beta = beta0
    no_improve = 0
    for it in range(steps):
        if time.time() > dl:
            break
        g = _smax_grad(h, alpha, topk)
        v = beta * v + (1.0 - beta) * g
        hn = _proj2(h - step * v, target)
        fn = _eval(hn)
        if fn < f - 1e-16:
            h, f = hn, fn
            step = min(step * 1.025, 0.04)
            no_improve = 0
        else:
            step *= 0.5
            no_improve += 1
            if no_improve > 50:
                v *= 0.5
                no_improve = 0
            h = _proj2(h - step * v, target)
            f = _eval(h)
        if f < best_f:
            best_f, best_h = f, h.copy()
    return _repair(best_h, n), best_f

def _l1_fun(x, alpha, topk, target):
    return _smax_obj(x, alpha, topk) + 80.0 * (x.sum() - target) ** 2

def _l1_jac(x, alpha, topk, target):

    return g

                     args=(alpha, topk, target),

                     options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14})
        return _repair(r.x, n)
    except Exception:
        return h

def _polish(h, n_iter, dl):

        i = it % n
        j = (i + 1) % n
        di, dj = d, -d
        if (i // 2) % 2 == 1:

        hi, hj = h[i] + di, h[j] + dj
        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:

            else:
