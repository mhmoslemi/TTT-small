            if c5n < bc: bc, bh = c5n, h.copy()
    return bh, bc

def _jitter_reopt(h, rng, scale, n_iter, deadline, alpha, topk, step):
    n = len(h)
    c0 = _eval(h)
    d = scale * rng.randn(n)
    d -= d.mean()
    hj = _repair(np.clip(h + d, 0, 1), n)
    hj2, cj = _fista_run(hj, n_iter, deadline, alpha, topk, step)
    if cj < c0: return hj2, cj
    return h, c0

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h, best_c5 = h.copy(), _eval(h)

    # Phase 1: SLSQP with broad soft-max
    dl1 = min(deadline, t0 + 100)
    h1 = _slsqp_smooth(best_h.copy(), 5000, 30, 300, dl1)
    c5v = _eval(h1)
    if c5v < best_c5: best_c5, best_h = c5v, h1

    # Phase 2: FISTA broad soft-max (navigates basin)
    dl2 = min(deadline, t0 + 300)
    h2, c5v = _fista_run(best_h.copy(), 12000, dl2, 8000, 20, 1.0)
    if c5v < best_c5: best_c5, best_h = c5v, h2

    # Phase 3: FISTA tighter soft-max
    dl3 = min(deadline, t0 + 500)
    h3, c5v = _fista_run(best_h.copy(), 12000, dl3, 2000, 8, 1.5)
    if c5v < best_c5: best_c5, best_h = c5v, h3

    # Phase 4: FISTA with exact subgradient
    dl4 = min(deadline, t0 + 680)
    for tol_v in [0.002, 0.0008, 0.0003]:
        if time.time() > dl4: break
        h4, c5v = _fista_exact(best_h.copy(), 6000, dl4, tol_v, 1.0)
        if c5v < best_c5: best_c5, best_h = c5v, h4

    # Phase 5: Multiple jitter-reopt
    dl5 = min(deadline, t0 + 880)
    for j in range(12):
        if time.time() > dl5: break
        sc = 0.0003 + 0.0003 * (j % 4)
        ak = 3000 + 2000 * (j % 3)
        tk = 5 + 3 * (j % 3)
        hj, cj = _jitter_reopt(best_h.copy(), rng, sc, 2000, dl5, ak, tk, 1.0)
        if cj < best_c5: best_c5, best_h = cj, hj

    # Phase 6: Block annealing
    dl6 = min(deadline, t0 + budget_s - 80)
    blocks = [2, 3, 4, 5, 7]
    best_h, best_c5 = _block_anneal(best_h.copy(), 50000, rng, dl6, blocks, 0.002, 1e-11)

    # Phase 7: Final exact FISTA
    if deadline - time.time() > 15:
        h7, c5v = _fista_exact(best_h.copy(), 5000, deadline, 0.0002, 0.8)
        if c5v < best_c5: best_c5, best_h = c5v, h7

    # Phase 8: Ultra-fine polish
    rem = max(0, deadline - time.time())
    if rem > 3:
        n_pol = min(int(rem / 0.003), 40000)
        best_h, best_c5 = _pair_polish(best_h, n_pol, rng, deadline, 3e-7)

    best_h = _repair(best_h, n0)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n0)
