import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _grad_smax(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
        ct = c[idx]
    else:
        idx = np.arange(len(c)); ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max()); w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-14:
            continue
        k = idx[ii]; sh = (n - 1) - k; wk = w[ii] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n - sp] -= wk * h[sp:]
    return g, m

def _adam_opt(h, alpha, topk, steps, lr, deadline, beta1=0.85, beta2=0.999, eps=1e-8):
    n = len(h)
    target = n * 0.5
    f = _eval(h)
    m = np.zeros(n); v = np.zeros(n)
    best_h, best_f = h.copy(), f
    for t in range(1, steps + 1):
        if time.time() > deadline:
            break
        g, _ = _grad_smax(h, alpha, topk)
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g * g
        mh = m / (1 - beta1**t)
        vh = v / (1 - beta2**t)
        step = lr / (1.0 + 0.01 * t)
        h_new = _proj(h - step * mh / (np.sqrt(vh) + eps), target)
        f_new = _eval(h_new)
        if f_new < f + 1e-13:
            h = h_new; f = f_new
            if f < best_f:
                best_f, best_h = f, h.copy()
        else:
            h_new = _proj(h - step * 0.3 * mh / (np.sqrt(vh) + eps), target)
            f_new = _eval(h_new)
            if f_new < f + 1e-13:
                h = h_new; f = f_new
                if f < best_f:
                    best_f, best_h = f, h.copy()
    return _repair(best_h, n), best_f

def _fista_opt(h, alpha, topk, steps, lr0, deadline):
    n = len(h)
    target = n * 0.5
    f = _eval(h)
    x_prev = h.copy()
    k = 1.0
    best_h, best_f = h.copy(), f
    for t in range(steps):
        if time.time() > deadline:
            break
        y = h + (k - 1.0) / (k + 2.0) * (h - x_prev)
        g, _ = _grad_smax(y, alpha, topk)
        x_prev = h.copy()
        lr = lr0 / np.sqrt(1.0 + t)
        h_new = _proj(y - lr * g, target)
        f_new = _eval(h_new)
        if f_new <= f + 1e-14:
            h = h_new; f = f_new
            k = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * k * k))
        else:
            h = x_prev
            f = _eval(h)
            k = 1.0
            lr *= 0.4
            h_new = _proj(h - lr * g, target)
            f_new = _eval(h_new)
            if f_new <= f:
                h = h_new; f = f_new
        if f < best_f:
            best_f, best_h = f, h.copy()
    return _repair(best_h, n), best_f

def _nesterov_exact(h, steps, step0, deadline, tol=0.003, beta=0.88):
    n = len(h)
    target = n * 0.5
    c = _correl(h)
    m = c.max()
    active = np.where(c >= m - tol)[0]
    dx = 2.0 / n
    f = _eval(h)
    v = np.zeros(n)
    best_h, best_f = h.copy(), f
    step = step0
    for _ in range(steps):
        if time.time() > deadline:
            break
        c = _correl(h)
        m = c.max()
        active = np.where(c >= m - tol)[0]
        g = np.zeros(n)
        for kk in active:
            sh = (n - 1) - kk
            if sh >= 0:
                g[:n - sh] += (1.0 - h[sh:])
                if sh > 0:
                    g[sh:] -= h[:-sh]
            else:
                sp = -sh
                g[sp:] += (1.0 - h[:-sp])
                g[:n - sp] -= h[sp:]
        g *= dx / max(len(active), 1)
        hp = h.copy()
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, target)
        fn = _eval(hn)
        if fn < f - 1e-15:
            h, f = hn, fn
            step = min(step * 1.01, step0 * 12.0)
            if f < best_f:
                best_f, best_h = f, h.copy()
        else:
            step *= 0.5
            h = _proj(hp - step * v, target)
            f = _eval(h)
            if f < best_f:
                best_f, best_h = f, h.copy()
    return _repair(best_h, n), best_f

def _anneal(h, niter, rng, deadline, blocks, T0, T1):
    n = len(h)
    c5c = _eval(h); bh = h.copy(); bc = c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0+bs].copy()
        d = 0.004 * max(T, 1e-9) ** 0.5 * rng.randn(bs)
        d -= d.mean(); seg += d; seg = np.clip(seg, 0, 1)
        hn = h.copy(); hn[i0:i0+bs] = seg; hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h, c5c = hn, c5n
            if c5n < bc:
                bc, bh = c5n, h.copy()
    return bh, bc

def _polish(h, niter, rng, deadline, scale):
    nc = len(h); c5c = _eval(h); bh = h.copy(); bc = c5c
    for _ in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(nc)
        j = (i + 1 + rng.randint(0, 4)) % nc
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < bc:
                    bc, bh = c5n, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc
