    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_lag(h, k, dx):
    n = len(h)
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        g[:n - sh] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:-sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:-sp])
        g[:n - sp] -= h[sp:]
    return g * dx

def _smax_obj(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        idx = np.arange(len(c))
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _grad_smax(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
        ct = c[idx]
    else:
        idx = np.arange(len(c))
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-14:
            continue
        g += w[ii] * _grad_lag(h, idx[ii], dx)
    return g

def _lbfgs_run(h, alpha, topk, lam, iters, deadline):
    n = len(h)
    target = n * 0.5
    def obj(x):
        return _smax_obj(x, alpha, topk) + lam * (x.sum() - target) ** 2
    def jac(x):
        return _grad_smax(x, alpha, topk) + 2.0 * lam * (x.sum() - target)
    if time.time() > deadline:
        return h
    try:
        r = minimize(obj, h, jac=jac, method='L-BFGS-B',
                     bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': iters, 'ftol': 1e-17, 'gtol': 1e-11, 'maxls': 40})
        return _repair(r.x, n)
    except Exception:
        return h

def _nesterov(h, alpha, topk, steps, step0, deadline):
    n = len(h)
    f = _smax_obj(h, alpha, topk)
    v = np.zeros(n)
    beta = 0.8
    step = step0
    for it in range(steps):
        if time.time() > deadline:
            break
        g = _grad_smax(h, alpha, topk)
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, n * 0.5)
        fn = _smax_obj(hn, alpha, topk)
        if fn < f:
            h = hn
            f = fn
            step = min(step * 1.04, step0 * 25.0)
        else:
            step *= 0.5
            hn = _proj(h - step * v, n * 0.5)
            fn = _smax_obj(hn, alpha, topk)
            if fn < f:
                h = hn
                f = fn
            else:
                step *= 0.5
                hn = _proj(h - step * v, n * 0.5)
                fn = _smax_obj(hn, alpha, topk)
                if fn < f:
                    h = hn
                    f = fn
                else:
                    break
    return _repair(h, n)

def _fourier_perturb(h, freq, amp, rng):
    n = len(h)
    x = np.arange(n)
    k = max(1, freq)
    phase = rng.rand() * 2.0 * np.pi
    f = amp * (np.sin(2.0 * np.pi * k * x / n + phase) + 0.5 * np.cos(4.0 * np.pi * k * x / n))
    f -= f.mean()
    return _repair(h + f, n)

def _noise_perturb(h, amp, rng):
    n = len(h)
    f = amp * rng.randn(n)
    f -= f.mean()
    return _repair(h + f, n)

def _anneal(h, niter, rng, deadline, T0, T1, blocks):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0+bs].copy()
        d = 0.002 * max(T, 1e-9) ** 0.5 * rng.randn(bs)
        d -= d.mean()
        seg += d
        seg = np.clip(seg, 0, 1)
        hn = h.copy()
        hn[i0:i0+bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h = hn
            c5c = c5n
            if c5n < bc:
                bc = c5n
                bh = h.copy()
    return bh, bc

def _coord_polish(h, deadline):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    remaining = deadline - time.time()
    max_sweeps = max(1, int(remaining / 0.5))
    for sweep in range(min(max_sweeps, 200)):
        if time.time() > deadline:
            break
        for i in range(0, n, 2):
            j = (i + 1) % n
            for d in [3e-6, -3e-6, 1e-6, -1e-6]:
                ni, nj = h[i] + d, h[j] - d
                if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                    h[i], h[j] = ni, nj
                    c5n = _eval(h)
                    if c5n < c5c - 1e-20:
                        c5c = c5n
                        if c5n < bc:
                            bc = c5n
                            bh = h.copy()
                        break
                    h[i], h[j] = ni - d, nj + d
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 4
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: L-BFGS-B progressive sharpening (0-220s)
    dl1 = min(deadline, t0 + 220)
    configs = [(1500, 100, 80, 250), (5000, 50, 80, 300), (20000, 25, 80, 350),
               (80000, 12, 60, 400), (300000, 6, 60, 500), (1000000, 3, 40, 500)]
    for alpha, tk, lam, it in configs:
        if time.time() > dl1:
            break
        h = _lbfgs_run(h, alpha, tk, lam, it, dl1)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 2: Nesterov high-alpha (220-420s)
    dl2 = min(deadline, t0 + 420)
    for alpha, tk, st, ns in [(200000, 12, 0.002, 2500), (800000, 6, 0.001, 2500),
                              (3000000, 3, 0.0005, 2000)]:
        if time.time() > dl2:
            break
        h = _nesterov(best_h.copy(), alpha, tk, ns, st, dl2)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 3: Fourier/noise perturbation + re-optimize (420-680s)
    dl3 = min(deadline, t0 + 680)
    freqs = [1, 2, 3, 4, 5, 6, 7, 8, 12, 16]
    amps = [0.003, 0.006, 0.01, 0.002]
    tries = 0
    max_tries = 40
    while tries < max_tries and time.time() < dl3:
        tries += 1
        if tries % 3 == 0:
            hp = _noise_perturb(best_h, amps[tries % len(amps)], rng)
        else:
            hp = _fourier_perturb(best_h, freqs[tries % len(freqs)],
                                   amps[tries % len(amps)], rng)
        hp = _lbfgs_run(hp, 50000, 15, 60, 150, dl3)
        hp = _nesterov(hp, 300000, 8, 800, 0.0008, dl3)
        c5v = _eval(hp)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = hp.copy()

    # Phase 4: Second L-BFGS pass on best (680-780s)
    dl4 = min(deadline, t0 + 780)
    for alpha, tk, lam, it in [(500000, 5, 40, 600), (2000000, 2, 40, 600)]:
        if time.time() > dl4:
            break
        h = _lbfgs_run(best_h.copy(), alpha, tk, lam, it, dl4)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 5: Annealing (780-900s)
    dl5 = min(deadline, t0 + 900)
    blocks = [1, 2, 3, 5, 7, 11, 17]
    h_a, c5_a = _anneal(best_h.copy(), 35000, rng, dl5, 0.002, 1e-11, blocks)
    if c5_a < best_c5:
        best_c5 = c5_a
        best_h = h_a.copy()

    # Phase 6: Ultra-high alpha Nesterov (900-1000s)
    dl6 = min(deadline, t0 + 1000)
    h = _nesterov(best_h.copy(), 8000000, 2, 3000, 0.0002, dl6)
    c5v = _eval(h)
    if c5v < best_c5:
        best_c5 = c5v
        best_h = h.copy()

    # Phase 7: Coordinate polish (1000-end)
    best_h, best_c5 = _coord_polish(best_h.copy(), deadline)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
