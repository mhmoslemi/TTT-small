import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) > 1e-10:
        h *= target / max(s, 1e-12)
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _smax_grad(h, alpha, topk):
    n = len(h)
    c = _corr(h)
    m = c.max()
    idx = np.argpartition(c, -topk)[-topk:].copy() if topk < len(c) else np.arange(len(c))
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    for ii in range(len(idx)):
        wk = w[ii] * dx
        if wk < 1e-14: continue
        k = idx[ii]; sh = (n - 1) - k
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - h[sh:])
            if sh > 0: g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n-sp] -= wk * h[sp:]
    return g

    best_h, best_f = h.copy(), f

        g = _smax_grad(h, alpha, topk)
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, target)
        fn = _eval(hn)

def _pair_opt(h, i, j, c0, topk):
    """Optimize 2D subproblem: transfer d from j to i."""
    n = len(h); dmin = -min(h[i], 1.0 - h[j])
    dmax = min(1.0 - h[i], h[j])
    if dmax - dmin < 1e-8: return h.copy(), _eval(h)
    delta = 1e-5
    hp = h.copy(); hp[i] += delta; hp[j] -= delta
    hq = h.copy(); hq[i] -= delta; hq[j] += delta
    c1 = _corr(hp); c2 = _corr(hq)
    b = (c1 - c2) / (2 * delta)

    idx = np.argpartition(c0, -topk)[-topk:]

        else: a, c_, fc = d_, c_, fc
        c_ = bnd - gr * (bnd - a); d_ = a + gr * (bnd - a)
        fc = (a0 + bk * c_ + ck * c_ * c_).max()
        fd = (a0 + bk * d_ + ck * d_ * d_).max()
    d_best = (a + bnd) / 2

Let me verify: corr_k = sum h[a] * (1-h[b]) over appropriate pairs. When h[i] -> h[i]+d and h[j] -> h[j]-d:
- Terms involving h[i] as the first factor: (h[i]+d)*(1-h[b]) = h[i]*(1-h[b]) + d*(1-h[b])
- Terms involving h[i] as the second factor (in 1-h[b]): h[a]*(1-(h[i]+d)) = h[a]*(1-h[i]) - d*h[a]
- Terms involving h[j] as the first factor: (h[j]-d)*(1-h[b]) = h[j]*(1-h[b]) - d*(1-h[b])
- Terms involving h[j] as the second factor: h[a]*(1-(h[j]-d)) = h[a]*(1-h[j]) + d*h[a]

The d² term comes only if i and j are in the same product term, i.e., if h[i] and h[j] appear in the same h[a]*h[b] term. This happens when:
- a=i, b=j (or a=j, b=i) in the same correlation term

In that case, the term is (h[i]+d)*(1-(h[j]-d)) = h[i]*(1-h[j]) + d*(1-h[j]) - d*h[i] - d²
So there IS a -d² term! But it only appears for the specific lags where i and j are "connected" (i.e., j = i+sh or i = j+sh for some sh).

So the quadratic approximation IS exact (not just approximate)! The c_k coefficient is:
- c_k = -1 (times dx) for the lags where i and j are directly connected
- c_k = 0 otherwise

import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) > 1e-10:
        h *= target / max(s, 1e-12)
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _smax_grad(h, alpha, topk):
    n = len(h)
    c = _corr(h)
    m = c.max()
    idx = np.argpartition(c, -topk)[-topk:].copy() if topk < len(c) else np.arange(len(c))
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max()); w /= w.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    for ii in range(len(idx)):
        wk = w[ii] * dx
        if wk < 1e-14:
            continue
        k = idx[ii]; sh = (n - 1) - k
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n-sp] -= wk * h[sp:]
    return g

def _nest(h, alpha, topk, steps, step0, deadline, beta):
    n = len(h); target = n * 0.5
    f = _eval(h); v = np.zeros(n); step = step0
    best_h, best_f = h.copy(), f
    for _ in range(steps):
        if time.time() > deadline:
            break
        g = _smax_grad(h, alpha, topk)
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, target)
        fn = _eval(hn)
        if fn < f - 1e-16:
            h, f = hn, fn
            step = min(step * 1.02, step0 * 20.0)
        else:
            step *= 0.5
            h = _proj(h - step * v, target); f = _eval(h)
        if f < best_f:
            best_f, best_h = f, h.copy()
    return _repair(best_h, n), best_f

def _pair_opt(h, i, j, c0, topk):
    n = len(h)
    dmin = -min(h[i], 1.0 - h[j])
    dmax = min(1.0 - h[i], h[j])
    if dmax - dmin < 1e-8:
        return h.copy(), _eval(h)
    delta = 1e-3
    hp = h.copy(); hp[i] += delta; hp[j] -= delta
    hq = h.copy(); hq[i] -= delta; hq[j] += delta
    c1 = _corr(hp); c2 = _corr(hq)
    b_all = (c1 - c2) / (2.0 * delta)
    cq_all = (c1 + c2 - 2.0 * c0) / (2.0 * delta * delta)
    idx = np.argpartition(c0, -topk)[-topk:]
    a0 = c0[idx]; bk = b_all[idx]; ck = cq_all[idx]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    ag, bg = dmin, dmax
    c_ = bg - gr * (bg - ag); d_ = ag + gr * (bg - ag)
    fc = (a0 + bk * c_ + ck * c_ * c_).max()
    fd = (a0 + bk * d_ + ck * d_ * d_).max()
    for _ in range(45):
        if fc < fd:
            bg, d_, fd = c_, d_, fc
            c_ = bg - gr * (bg - ag)
            fc = (a0 + bk * c_ + ck * c_ * c_).max()
        else:
            ag, c_, fc = d_, c_, fc
            d_ = ag + gr * (bg - ag)
            fd = (a0 + bk * d_ + ck * d_ * d_).max()
    d_best = (ag + bg) * 0.5
    hn = h.copy(); hn[i] += d_best; hn[j] -= d_best
    return hn, _eval(hn)

def _pair_sweep(h, deadline, n_pairs, topk, targeted):
    n = len(h); c5 = _eval(h); best_h = h.copy()
    rng = np.random.RandomState(314)
    for it in range(n_pairs):
        if time.time() > deadline:
            break
        if targeted and it % 50 == 0:
            c0 = _corr(h)
            cm = c0.max()
            active = np.where(c0 > cm - 0.002)[0]
        while True:
            if targeted and it % 50 == 0 and len(active) > 0:
                kk = active[rng.randint(len(active))]
                sh = (n - 1) - kk
                if sh >= 0 and sh < n:
                    i = rng.randint(0, max(1, n - sh))
                    j = i + sh
                elif sh < 0:
                    j = rng.randint(0, max(1, n + sh))
                    i = j - sh
                else:
                    i = rng.randint(n); j = (i + 1) % n
            else:
                i = rng.randint(n); j = (i + 1 + rng.randint(0, 20)) % n
            if i != j:
                break
        c0 = _corr(h)
        hn, fn = _pair_opt(h, i, j, c0, topk)
        if fn < c5 - 1e-14:
            h, c5 = hn, fn
            if c5 < _eval(best_h):
                best_h = h.copy()
    return best_h, _eval(best_h)

def _anneal(h, niter, rng, deadline, T0):
    n = len(h); c5c = _eval(h); bh = h.copy(); bc = c5c
    blocks = [2, 3, 4, 5, 7, 10, 14]
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (1e-9) * (1.0 + it / max(niter, 1)) ** (-3)
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0+bs].copy()
        d = 0.005 * rng.randn(bs); d -= d.mean()
        seg = np.clip(seg + d, 0.0, 1.0)
        hn = h.copy(); hn[i0:i0+bs] = seg; hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-14)):
            h, c5c = hn, c5n
            if c5n < bc:
                bc, bh = c5n, h.copy()
    return bh, bc
