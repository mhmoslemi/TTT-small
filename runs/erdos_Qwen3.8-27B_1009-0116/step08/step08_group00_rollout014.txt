    best_c = c5cur
    step = 2e-5
    t_start = time.time()
    while time.time() - t_start < t_limit:
        np_ = rng.randint(1, 5)
        idx = rng.randint(0, n, size=np_)
        d = step * rng.randn(np_)
        d -= d.mean()
        h_new = h.copy()
        h_new[idx] += d
        if h_new.min() < 0 or h_new.max() > 1:
            step *= 0.8
            continue
        h_new = _repair(h_new, n)
        c5n = _eval(h_new)
        if c5n < c5cur:
            h = h_new
            c5cur = c5n
            step = min(step * 1.03, 1e-4)
            if c5n < best_c:
                best_c = c5n
                best_h = h.copy()
        else:
            step *= 0.92
    return best_h, best_c

def _polish(h, rng, t_limit):
    n = len(h)
    h2 = h.copy()
    c5c = _eval(h2)
    best_c = c5c
    best_h = h2.copy()
    step = 2e-7
    t_start = time.time()
    while time.time() - t_start < t_limit:
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 10)) % n
        d = step * rng.randn()
        ni, nj = h2[i] + d, h2[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h2[i], h2[j] = ni, nj
            c5n = _eval(h2)
            if c5n < c5c:
                c5c = c5n
                step = min(step * 1.01, 5e-6)
                if c5n < best_c:
                    best_c = c5n
                    best_h = h2.copy()
            else:
                h2[i], h2[j] = ni - d, nj + d
                step *= 0.99
    return best_h, best_c
