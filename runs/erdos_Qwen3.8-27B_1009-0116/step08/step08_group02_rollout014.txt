    c0 = np.correlate(h, 1.0 - h, mode='full') * dx

        c = np.correlate(x, 1.0 - x, mode='full') * dx
        cs = c[idx]
        m = cs.max()
        s = np.clip(alpha * (cs - m), -400, 400)
        s -= s.max()
        return m + np.log(np.exp(s).sum()) / alpha
    
    def grad(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        cs = c[idx]
        m = cs.max()
        s = np.clip(alpha * (cs - m), -400, 400)

        w = np.exp(s) / np.exp(s).sum()

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(64):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _slsqp(h, n, alpha, top_k, maxiter):
    dx = 2.0 / n
    c0 = np.correlate(h, 1.0 - h, mode='full') * dx
    idx = np.argsort(c0)[-top_k:] if 0 < top_k < 2*n-1 else np.arange(2*n-1)
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        cs = c[idx]
        m = cs.max()
        s = np.clip(alpha * (cs - m), -400, 400)
        s -= s.max()
        return m + np.log(np.exp(s).sum()) / alpha
    def grad(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        cs = c[idx]
        m = cs.max()
        s = np.clip(alpha * (cs - m), -400, 400)
        s -= s.max()
        w = np.exp(s) / np.exp(s).sum()
        g = np.zeros(n)
        for ii, k in enumerate(idx):
            if w[ii] < 1e-15:
                continue
            sh = (n - 1) - k
            if sh >= 0:
                i1 = np.arange(n - sh)
                g[i1] += w[ii] * (1.0 - x[i1 + sh])
                if sh > 0:
                    i2 = np.arange(sh, n)
                    g[i2] -= w[ii] * x[i2 - sh]
            else:
                sp = -sh
                i1 = np.arange(sp, n)
                g[i1] += w[ii] * (1.0 - x[i1 - sp])
                i2 = np.arange(n - sp)
                g[i2] -= w[ii] * x[i2 + sp]
        return g * dx
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return h

def _proj_grad(h, n, alpha, steps, step0, top_k, momentum):
    v = np.zeros(n)
    step = step0
    dx = 2.0 / n
    cur_c = _eval(h, n)
    for it in range(steps):
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        active = np.argsort(c)[-top_k:]
        c_a = c[active]
        m_a = c_a.max()
        s = np.clip(alpha * (c_a - m_a), -400, 400)
        s -= s.max()
        w = np.exp(s) / np.exp(s).sum()
        g = np.zeros(n)
        for ii, k in enumerate(active):
            if w[ii] < 1e-15:
                continue
            sh = (n - 1) - k
            if sh >= 0:
                i1 = np.arange(n - sh)
                g[i1] += w[ii] * (1.0 - h[i1 + sh])
                if sh > 0:
                    i2 = np.arange(sh, n)
                    g[i2] -= w[ii] * h[i2 - sh]
            else:
                sp = -sh
                i1 = np.arange(sp, n)
                g[i1] += w[ii] * (1.0 - h[i1 - sp])
                i2 = np.arange(n - sp)
                g[i2] -= w[ii] * h[i2 + sp]
        g *= dx
        v = momentum * v + (1.0 - momentum) * g
        taken = False
        for _t in range(20):
            h_new = _proj(h - step * v, n * 0.5)
            c_new = _eval(h_new, n)
            if c_new < cur_c - 1e-18:
                h = h_new
                cur_c = c_new
                step = min(step * 1.08, step0 * 30.0)
                taken = True
                break
            step *= 0.4
        if not taken:
            step = min(step * 1.3, step0 * 30.0)
        if it % 300 == 0:
            v *= 0.4
    return h

def _binary_search(h, n, time_budget, rng, top_k=3):
    best_h = h.copy()
    best_c = _eval(h, n)
    one_pos = np.where(h > 0.5)[0]
    zero_pos = np.where(h <= 0.5)[0]
    t_start = time.time()
    n_try = 800
    stall = 0
    while time.time() - t_start < time_budget:
        c = np.correlate(h, 1.0 - h, mode='full') * 2.0 / n
        cur_max = c.max()
        if cur_max < best_c:
            best_c = cur_max
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        idx = np.argsort(c)[-top_k:]
        worst_k = int(np.argmax(c))
        sh = (n - 1) - worst_k
        if sh >= 0:
            active_start, active_end = 0, n - sh
        else:
            active_start, active_end = -sh, n
        active_ones = [i for i in one_pos if active_start <= i < active_end and
                       (0 <= i+sh < n) and h[i+sh] <= 0.5]
        active_ones = np.array(active_ones) if len(active_ones) else one_pos[:1]
        best_delta = 0
        best_i, best_j = -1, -1
        n_ones = len(one_pos)
        n_zeros = len(zero_pos)
        for _ in range(n_try):
            if rng.rand() < 0.5 and len(active_ones) > 0:
                i = active_ones[rng.randint(len(active_ones))]
            else:
                i = one_pos[rng.randint(n_ones)]
            j = zero_pos[rng.randint(n_zeros)]
            if i == j:
                continue
            h[i], h[j] = 0.0, 1.0
            c_new = _eval(h, n)
            h[i], h[j] = 1.0, 0.0
            delta = cur_max - c_new
            if delta > best_delta:
                best_delta = delta
                best_i, best_j = i, j
        if best_i >= 0 and best_delta > 1e-15:
            h[best_i] = 0.0
            h[best_j] = 1.0
            one_pos = np.where(h > 0.5)[0]
            zero_pos = np.where(h <= 0.5)[0]
        elif stall > 50 and rng.rand() < 0.2:
            i = one_pos[rng.randint(n_ones)]
            j = zero_pos[rng.randint(n_zeros)]
            h[i], h[j] = 0.0, 1.0
            one_pos = np.where(h > 0.5)[0]
            zero_pos = np.where(h <= 0.5)[0]
            stall = 0
    return best_h, best_c

def _local_2pt(h, n, n_iter, rng, step):
    best_h = h.copy()
    best_c = _eval(h, n)
    cur_c = best_c
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 5)) % n
        d = step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c_new = _eval(h, n)
            if c_new < cur_c:
                cur_c = c_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def _spectral_start(n, seed_s):
    rng2 = np.random.RandomState(seed_s)
    alpha = np.sqrt(2.0) - 1.0
    h = (np.arange(n) * alpha % 1.0) < 0.5
    h = h.astype(float)
    for _ in range(50):
        i = rng2.randint(n)
        j = (i + 1 + rng2.randint(0, 10)) % n
        if h[i] > 0.5 and h[j] <= 0.5:
            h[i], h[j] = 0.0, 1.0
        elif h[i] <= 0.5 and h[j] > 0.5:
            h[i], h[j] = 1.0, 0.0
    return _repair(h, n)
