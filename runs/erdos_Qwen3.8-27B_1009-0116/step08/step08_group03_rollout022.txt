import numpy as np
import time


def _fix_sum(h, S):
    h = np.asarray(h, dtype=float).copy()
    n = h.size
    np.clip(h, 0.0, 1.0, out=h)
    if S <= 0.0:
        return np.zeros(n)
    if S >= n:
        return np.ones(n)
    for _ in range(60):
        s = float(h.sum())
        if abs(s - S) < 1e-12:
            break
        if s <= 1e-15:
            h.fill(S / float(n))
            break
        h *= S / s
        np.clip(h, 0.0, 1.0, out=h)
    d = S - float(h.sum())
    if abs(d) > 1e-13:
        if d > 0.0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for i in order:
            if d > 0.0:
                add = min(d, 1.0 - h[i])
                if add > 0.0:
                    h[i] += add
                    d -= add
            else:
                sub = min(-d, h[i])
                if sub > 0.0:
                    h[i] -= sub
                    d += sub
            if abs(d) < 1e-14:
                break
    return h


def _proj(v, S):
    v = np.asarray(v, dtype=float).copy()
    n = v.size
    if n == 0:
        return v
    if S <= 0.0:
        return np.zeros(n)
    if S >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(v - mid, 0.0, 1.0).sum())
        if s > S:
            lo = mid
        else:
            hi = mid
    out = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    d = S - float(out.sum())
    if abs(d) > 1e-11:
        if d > 0.0:
            order = np.argsort(1.0 - out)
        else:
            order = np.argsort(out)
        for i in order:
            if d > 0.0:
                add = min(d, 1.0 - out[i])
                if add > 0.0:
                    out[i] += add
                    d -= add
            else:
                sub = min(-d, out[i])
                if sub > 0.0:
                    out[i] -= sub
                    d += sub
            if abs(d) < 1e-13:
                break
    return out


def _c5(h):
    n = len(h)
    if n == 0:
        return float("inf")
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))


def _resample(h, m):
    n = len(h)
    m = int(m)
    if m <= 0:
        return h.copy()
    if n == m:
        return _fix_sum(h, 0.5 * m)
    if n % m == 0:
        out = h.reshape(m, n // m).mean(axis=1)
    else:
        x0 = (np.arange(n) + 0.5) * (2.0 / n)
        x1 = (np.arange(m) + 0.5) * (2.0 / m)
        out = np.interp(x1, x0, h)
    return _fix_sum(out, 0.5 * m)


def _to_binary(h):
    n = len(h)
    target = 0.5 * n
    h = _fix_sum(h, target)
    idx = np.argsort(h)[::-1]
    k = int(np.floor(target + 1e-9))
    out = np.zeros(n)
    if k > 0:
        out[idx[:k]] = 1.0
    rem = target - float(k)
    if rem > 1e-9 and k < n:
        out[idx[k]] = rem
    return out


def _single_move(h, a, full, idx, delta):
    if delta == 0.0:
        return
    n = len(h)
    base = n - 1
    hi = float(h[idx])

    if idx > 0:
        full[base + 1:base + idx + 1] += delta * a[:idx][::-1]
    if idx < base:
        full[idx:base] += delta * a[base::-1][:base - idx]
    full[base] += delta * (1.0 - hi - hi) - delta * delta

    if idx + 1 < n:
        full[base + 1:base + n - idx] += -delta * h[idx + 1:]
    if idx > 0:
        full[base - idx:base] += -delta * h[:idx]

    h[idx] += delta
    a[idx] -= delta


def _active_gradient(h, full, K):
    n = len(h)
    base = n - 1
    L = full.size
    K = int(min(max(1, K), L))
    if K < L:
        order = np.argpartition(full, L - K)[-K:]
    else:
        order = np.arange(L)
    w = 1.0 / float(K)
    g = np.zeros(n)

    for p in range(K):
        t = int(order[p])
        k = t - base

        ns = -k if k < 0 else 0
        ne = (n - 1 - k) if k >= 0 else (n - 1)
        if ns <= ne:
            ist = ns + k
            ien = ne + k + 1
            g[ist:ien] += w * (1.0 - h[ns:ne + 1])

        ss = k if k >= 0 else 0
        se = (n - 1 + k) if k >= 0 else (n - 1)
        if ss <= se:
            ist = ss - k
            ien = se - k + 1
            g[ist:ien] -= w * h[ss:se + 1]

    return g


def _gradient_trial(h, g, cur_c):
    n = len(h)
    d = -g
    d -= float(d.mean())
    norm = float(np.linalg.norm(d))
    if norm <= 1e-12:
        return None
    u = d / norm
    best_h = None
    best_c = cur_c
    for s in (0.02, 0.05, 0.12, 0.25):
        ht = _proj(h + s * u, 0.5 * n)
        c = float(_c5(ht))
        if c < best_c - 1e-12:
            best_c = c
            best_h = ht
    return best_h

        else:
            i = int(low[int(rng.randint(low.size))])
            j = int(high[int(rng.randint(high.size))])

        if i == j:
            continue

        gi = float(g[i])
        gj = float(g[j])
        if gi < gj - 1e-12:
            sign = 1
        elif gj < gi - 1e-12:
            sign = -1
        else:
            sign = 1 if rng.random() < 0.5 else -1

        if sign > 0:
            avail = min(1.0 - float(h[i]), float(h[j]))
        else:
            avail = min(float(h[i]), 1.0 - float(h[j]))
        if avail <= 1e-14:
            continue

        mult = 0.25 + 2.0 * rng.random()
        if rng.random() < 0.07:
            mult *= 4.0
        d = sign * min(avail, step * mult)
        if d == 0.0:
            continue

        _single_move(h, a, full, i, d)
        _single_move(h, a, full, j, -d)
        newm = float(full.max())

        if newm < m - 1e-12:
            m = newm
            if m < best_m:
                best_m = m
                best_h = h.copy()
            acc += 1
        else:
            _single_move(h, a, full, j, d)
            _single_move(h, a, full, i, -d)

        win += 1
        if win >= 256:
            pr = acc / float(win)
            if pr > 0.45:
                step = min(0.2, step * 1.1)
            elif pr < 0.10:
                step = max(min_step, step * 0.85)
            acc = 0
            win = 0

        if move % 2048 == 0:
            h = _fix_sum(h, 0.5 * n)
            a = 1.0 - h
            full = np.correlate(h, a, mode="full")
            m = float(full.max())
            if m < best_m:
                best_m = m
                best_h = h.copy()

    best_h = _fix_sum(best_h, 0.5 * n)
    return best_h, float(_c5(best_h))


def _binary_search(h, rng, deadline, T_start=1.0):
    n = len(h)
    h = _fix_sum(h, 0.5 * n)
    if n < 4 or (n % 2 != 0) or deadline <= time.time():
        return h, float(_c5(h))

    h = _to_binary(h)
    a = 1.0 - h
    full = np.correlate(h, a, mode="full")
    m = float(full.max())
    best_m = m
    best_h = h.copy()
    start = time.time()

    K = max(8, min(80, n // 8))
    q = max(8, n // 8)
    cap = max(2.0, 0.03 * n)

    g = _active_gradient(h, full, K)
    zeros = np.nonzero(h < 0.5)[0]
    ones = np.nonzero(h > 0.5)[0]
    zlow = zeros
    ohigh = ones
    move = 0
    recompute = 0

    while time.time() < deadline:
        if recompute >= 96:
            h = _fix_sum(h, 0.5 * n)
            a = 1.0 - h
            full = np.correlate(h, a, mode="full")
            m = float(full.max())
            if m < best_m:
                best_m = m
                best_h = h.copy()

            g = _active_gradient(h, full, K)
            zeros = np.nonzero(h < 0.5)[0]
            ones = np.nonzero(h > 0.5)[0]
            if zeros.size == 0 or ones.size == 0:
                break

            if zeros.size > q:
                zs = g[zeros]
                zo = np.argsort(zs)
                zlow = zeros[zo[:q]]
            else:
                zlow = zeros

            if ones.size > q:
                os = g[ones]
                oo = np.argsort(os)
                ohigh = ones[oo[-q:]]
            else:
                ohigh = ones

            recompute = 0

        recompute += 1
        move += 1

        if zeros.size == 0 or ones.size == 0 or zlow.size == 0 or ohigh.size == 0:
            break

        frac = (time.time() - start) / max(1e-9, deadline - start)
        T = max(0.0, float(T_start) * (1.0 - frac) ** 2)

        if rng.random() < 0.78:
            i = int(zlow[int(rng.randint(zlow.size))])
            j = int(ohigh[int(rng.randint(ohigh.size))])
        else:
            i = int(zeros[int(rng.randint(zeros.size))])
            j = int(ones[int(rng.randint(ones.size))])

        _single_move(h, a, full, i, 1.0)
        _single_move(h, a, full, j, -1.0)
        newm = float(full.max())
        delta = newm - m

        accept = newm < m - 1e-12
        if not accept:
            if T > 1e-9 and newm <= best_m + cap and delta <= cap:
                if rng.random() < np.exp(-delta / T):
                    accept = True
            elif delta <= 1e-12 and rng.random() < 0.05:
                accept = True

        if accept:
            m = newm
            if m < best_m:
                best_m = m
                best_h = h.copy()
        else:
            _single_move(h, a, full, j, 1.0)
            _single_move(h, a, full, i, -1.0)

        if m > best_m + 3.0 * cap:
            h = best_h.copy()
            a = 1.0 - h
            full = np.correlate(h, a, mode="full")
            m = best_m
            recompute = 96

        if move % 1024 == 0:
            h = _fix_sum(h, 0.5 * n)
            recompute = 96

    best_h = _fix_sum(best_h, 0.5 * n)
    return best_h, float(_c5(best_h))


def _perturb(h, rng, kind):
    n = len(h)
    out = h.copy()
    if kind == 0:
        sig = float(rng.uniform(0.02, 0.12))
        out += rng.normal(0.0, sig, size=n)
    elif kind == 1:
        f = int(rng.randint(1, max(2, n // 12)))
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        amp = float(rng.uniform(0.03, 0.18))
        idx = np.arange(n)
        out += amp * np.sin(2.0 * np.pi * f * idx / n + ph)
    elif kind == 2:
        llo = 3
        lhi = max(4, min(n - 1, n // 12))
        if lhi > llo:
            L = int(rng.randint(llo, lhi))
            a1 = int(rng.randint(0, n - L))
            a2 = int(rng.randint(0, n - L))
            if a1 == a2:
                a2 = (a2 + 1) % max(1, n - L)
            m1 = float(out[a1:a1 + L].mean())
            m2 = float(out[a2:a2 + L].mean())
            c = float(rng.uniform(0.05, 0.25)) * (m1 - m2)
            out[a1:a1 + L] += c
            out[a2:a2 + L] -= c
    return _proj(out, 0.5 * n)


def _add_cand(cands, h):
    if h is None or len(h) == 0:
        return
    h = _fix_sum(h, 0.5 * len(h))
    c = float(_c5(h))
    cands.append((c, h))


def _candidate_starts(h0, rng):
    cands = []
    n0 = len(h0)

    _add_cand(cands, h0)
    _add_cand(cands, h0[::-1])
    _add_cand(cands, 1.0 - h0)
    _add_cand(cands, 0.5 + 0.5 * (h0 - h0[::-1]))
    _add_cand(cands, 0.5 + 0.5 * (h0 + h0[::-1] - 1.0))
    _add_cand(cands, _to_binary(h0))

    for sig in (0.03, 0.08, 0.15):
        _add_cand(cands, _proj(h0 + rng.normal(0.0, sig, size=n0), 0.5 * n0))

    max_size = max(n0, min(840, 4 * n0))
    sizes = []
    raw = (max(96, n0 // 2), n0, max(96, min(max_size, 3 * n0 // 2)), max_size)
    for m in raw:
        m = int(m)
        if m < 64:
            m = 64
        if m > max_size:
            m = max_size
        if m % 2:
            m += 1
        if m > max_size:
            m = max_size
        if m not in sizes:
            sizes.append(m)

    for m in sizes:
        hr = _resample(h0, m)
        _add_cand(cands, hr)
        _add_cand(cands, _to_binary(hr))

    sin_sizes = []
    for m in (max(96, n0 // 2), min(n0, max_size)):
        m = int(m)
        if m < 64:
            continue
        if m not in sin_sizes:
            sin_sizes.append(m)

    for m in sin_sizes:
        x = (np.arange(m) + 0.5) * (2.0 / m)
        for f in (1, 2, 3, 5, 8):
            for amp in (0.5, 0.9):
                v = 0.5 + 0.5 * amp * np.cos(np.pi * f * x)
                _add_cand(cands, _proj(v, 0.5 * m))
            v = np.cos(np.pi * f * x)
            idx = np.argsort(v)[::-1]
            out = np.zeros(m)
            out[idx[:m // 2]] = 1.0
            _add_cand(cands, out)

    return cands


def _fallback_start():
    n = 560
    x = (np.arange(n) + 0.5) * (2.0 / n)
    best = None
    bc = 1e9
    for f in range(1, 9):
        for amp in (0.5, 0.8, 1.0):
            h = 0.5 * (1.0 + amp * np.cos(np.pi * f * x))
            h = _fix_sum(h, 0.5 * n)
            c = float(_c5(h))
            if c < bc:
                bc = c
                best = h.copy()
    if best is None:
        best = np.full(n, 0.5)
    return best


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    B = float(budget_s)
    if B < 5.0:
        B = 5.0
    deadline = t0 + B - 1.0
    rng = np.random.RandomState(seed)

    init = None
    if "initial_h_values" in globals():
        init = np.asarray(initial_h_values, dtype=float)
    if init is None:
        init = _fallback_start()

    h0 = _fix_sum(init, 0.5 * len(init))
    best_h = h0.copy()
    best_c = float(_c5(best_h))

    if B < 60.0:
        hh, cc = _local_search(best_h, rng, t0 + B - 1.0, 0.03)
        if cc < best_c:
            best_h, best_c = hh, cc
        best_h = _fix_sum(best_h, 0.5 * len(best_h))
        return best_h, float(_c5(best_h)), len(best_h)

    cands = _candidate_starts(h0, rng)
    vals = np.empty(len(cands), dtype=float)
    for ii in range(len(cands)):
        vals[ii] = cands[ii][0]
    order = np.argsort(vals)

    starts = []
    for idx in order:
        if len(starts) >= 8:
            break
        c, h = cands[idx]
        if c < best_c + 0.10:
            starts.append(h)
    if not starts:
        starts.append(h0)

    phase1_budget = min(160.0, max(30.0, 0.16 * B))
    p1_end = t0 + phase1_budget
    per = phase1_budget / max(1, len(starts))

    for h in starts:
        if time.time() >= p1_end:
            break
        slot = min(per, p1_end - time.time())
        if slot < 3.0:
            continue

        is_bin = bool(np.all((h == 0.0) | (h == 1.0))) and (len(h) % 2 == 0)
        if is_bin and slot > 10.0:
            b1 = time.time() + max(1.0, slot * 0.35)
            hb, cb = _binary_search(h.copy(), rng, b1, 1.0)
            if cb < best_c:
                best_h, best_c = hb, cb
            c0 = float(_c5(h))
            hcur = hb if cb < c0 else h.copy()
            slot = min(slot * 0.65, p1_end - time.time())
        else:
            hcur = h.copy()

        if slot > 2.0:
            hh, cc = _local_search(hcur, rng, time.time() + slot, 0.035)
            if cc < best_c:
                best_h, best_c = hh, cc

    final_budget = min(520.0, max(140.0, 0.45 * B))
    p2_end = t0 + B - final_budget - 3.0
    if p2_end > time.time() + 30.0:
        nslots = 5
        slot = (p2_end - time.time()) / float(nslots)
        for s in range(nslots):
            if time.time() >= p2_end:
                break
            if slot < 5.0:
                slot = 5.0
            kind = s % 4
            hp = _perturb(best_h.copy(), rng, kind)
            if kind == 3:
                hp = _to_binary(hp)
            hh, cc = _local_search(hp, rng, time.time() + slot, 0.05)
            if cc < best_c:
                best_h, best_c = hh, cc

    if deadline > time.time() + 5.0:
        base_step = 0.03 if best_c > 0.385 else 0.015
        hh, cc = _local_search(best_h.copy(), rng, deadline, base_step)
        if cc < best_c:
            best_h, best_c = hh, cc

    n = len(best_h)
    variants = (
        1.0 - best_h,
        best_h[::-1],
        0.5 + 0.5 * (best_h - best_h[::-1]),
        0.5 + 0.5 * (best_h + best_h[::-1] - 1.0),
        _to_binary(best_h),
    )
    for hv in variants:
        hv = _fix_sum(hv, 0.5 * n)
        c = float(_c5(hv))
        if c < best_c:
            best_h, best_c = hv, c

    best_h = _fix_sum(best_h, 0.5 * len(best_h))
    best_c = float(_c5(best_h))
    return best_h, best_c, len(best_h)
