import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad(h, alpha, top_k):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    if top_k < len(c):
        idx = np.argpartition(c, -top_k)[-top_k:]
    else:
        idx = np.arange(len(c))
    ct = c[idx]
    m = ct.max()
    s = np.clip(alpha * (ct - m), -500, 500)
    s -= s.max()
    w = np.exp(s)
    ws = w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        wi = w[ii] / ws
        if wi < 1e-14:
            continue
        sh = (n - 1) - k
        wk = wi * dx
        if sh >= 0:
            cnt = n - sh
            g[:cnt] += wk * (1.0 - h[sh:cnt + sh])
            if sh > 0:
                g[sh:] -= wk * h[:cnt]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g

def _adam(h, n_iter, alpha, top_k, lr0):
    n = len(h)
    tgt = n * 0.5
    mv = np.zeros(n)
    vv = np.zeros(n)
    b1, b2 = 0.85, 0.999
    t = 0
    cur = _eval(h)
    for _ in range(n_iter):
        t += 1
        g = _grad(h, alpha, top_k)
        mv = b1 * mv + (1 - b1) * g
        vv = b2 * vv + (1 - b2) * g * g
        mhat = mv / (1 - b1 ** t)
        vhat = vv / (1 - b2 ** t)
        lr = lr0 * min(1.0, 30.0 / t)
        h_new = _proj(h - lr * mhat / (np.sqrt(vhat) + 1e-8), tgt)
        c_new = _eval(h_new)
        if c_new < cur:
            h = h_new
            cur = c_new
    return h, cur

def _slsqp(h, alpha, top_k, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    tgt = n * 0.5
    ones = np.ones(n)
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * 2.0 / n
        if top_k < len(c):
            idx = np.argpartition(c, -top_k)[-top_k:]
            ct = c[idx]
        else:
            ct = c
        m = ct.max()
        s = np.clip(alpha * (ct - m), -400, 400)
        s -= s.max()
        return m + np.log(np.exp(s).sum()) / alpha
    def jac(x):
        return _grad(x, alpha, top_k)
    def eqf(x):
        return x.sum() - tgt
    def eqj(x):
        return ones
    cons = [{'type': 'eq', 'fun': eqf, 'jac': eqj}]
    try:
        res = minimize(obj, h, jac=jac, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return h

def _local2(h, n_iter, step, rng, temp=0.0):
    n = len(h)
    cur = _eval(h)
    best_c, best_h = cur, h.copy()
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 11)) % n
        d = step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            cn = _eval(h)
            acc = cn < cur
            if not acc and temp > 1e-12 and rng.rand() < np.exp(-max(0.0, cn - cur) / temp):
                acc = True
            if acc:
                cur = cn
                if cn < best_c:
                    best_c, best_h = cn, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def _local_block(h, n_iter, rng):
    n = len(h)
    cur = _eval(h)
    best_c, best_h = cur, h.copy()
    for _ in range(n_iter):
        i = rng.randint(0, n - 6)
        blk = 1 + rng.randint(0, 3)
        j = rng.randint(0, n - 6)
        if abs(i - j) < 8:
            continue
        amt = 0.004 * rng.randn()
        s0 = h[i:i+blk].copy()
        s1 = h[j:j+blk].copy()
        h[i:i+blk] = s0 + amt
        h[j:j+blk] = s1 - amt
        if h[i:i+blk].min() < -0.01 or h[i:i+blk].max() > 1.01 or h[j:j+blk].min() < -0.01 or h[j:j+blk].max() > 1.01:
            h[i:i+blk] = s0
            h[j:j+blk] = s1
            continue
        h = np.clip(h, 0, 1)
        h = _repair(h, n)
        cn = _eval(h)
        if cn < cur:
            cur = cn
            if cn < best_c:
                best_c, best_h = cn, h.copy()
        else:
            h[i:i+blk] = s0
            h[j:j+blk] = s1
    return best_h, best_c

def _downsample(h, f):
    n = len(h)
    m = n // f
    return h[:m*f].reshape(m, f).mean(axis=1).copy()

def _upsample(h, f):
    n = len(h)
    out = np.empty(n * f)
    for i in range(n):
        out[i*f:(i+1)*f] = h[i]
    return out

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    def _el():
        return time.time() - t0

    h_par = _repair(np.array(initial_h_values, dtype=float).copy(), len(initial_h_values))
    n = len(h_par)
    best_h = h_par.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Multi-resolution Adam (coarse to fine)
    for ratio in [8, 4, 2]:
        if _el() > 150:
            break
        nc = n // ratio
        if nc < 40:
            continue
        hc = _repair(_downsample(best_h, ratio), nc)
        h_new, c = _adam(hc, 400, 5000, 6, 0.02)
        h_up = _repair(_upsample(h_new, ratio), n)
        c_up = _eval(h_up)
        if c_up < best_c5:
            best_c5, best_h = c_up, h_up.copy()
        best_h, best_c5 = _local2(best_h, 2000, 1e-4, rng)

    # Phase 2: Adam at full resolution (multiple configs)
    if _el() < budget_s - 350:
        for alpha, tk, lr in [(1500, 10, 0.006), (5000, 6, 0.004), (20000, 4, 0.002), (100000, 3, 0.001)]:
            if _el() > budget_s - 300:
                break
            h_new, c = _adam(best_h.copy(), 500, alpha, tk, lr)
            c = _eval(h_new)
            if c < best_c5:
                best_c5, best_h = c, h_new.copy()

    # Phase 3: SLSQP (broad to narrow)
    if _el() < budget_s - 250:
        for alpha, tk, mi in [(2000, 12, 400), (8000, 8, 350), (40000, 5, 250)]:
            if _el() > budget_s - 200:
                break
            h_new = _slsqp(best_h.copy(), alpha, tk, mi)
            c = _eval(h_new)
            if c < best_c5:
                best_c5, best_h = c, h_new.copy()

    # Phase 4: Block mutations
    if _el() < budget_s - 180:
        nb = min(int((budget_s - 180 - _el()) * 150), 25000)
        best_h, best_c5 = _local_block(best_h, nb, rng)

    # Phase 5: Multi-start perturbations + Adam
    if _el() < budget_s - 180:
        for trial in range(5):
            if _el() > budget_s - 130:
                break
            scale = 0.008 * (trial + 1)
            noise = scale * rng.randn(n)
            noise -= noise.mean()
            h_p = _repair(best_h + noise, n)
            h_new, c = _adam(h_p, 300, 5000, 6, 0.003)
            c = _eval(h_new)
            if c < best_c5:
                best_c5, best_h = c, h_new.copy()

    # Phase 6: SLSQP fine-tuning
    if _el() < budget_s - 130:
        for alpha, tk, mi in [(20000, 6, 300), (200000, 4, 200), (2000000, 3, 150)]:
            if _el() > budget_s - 80:
                break
            h_new = _slsqp(best_h.copy(), alpha, tk, mi)
            c = _eval(h_new)
            if c < best_c5:
                best_c5, best_h = c, h_new.copy()

    # Phase 7: Warped SA (focus near active lags)
    if _el() < budget_s - 80:
        temp = max(5e-7, best_c5 * 8e-5)
        nsa = min(int((budget_s - 80 - _el()) * 1000), 50000)
        best_h, best_c5 = _local2(best_h, nsa, 3e-5, rng, temp)

    # Phase 8: Fine 2-point search
    if _el() < budget_s - 40:
        nf = min(int((budget_s - 40 - _el()) * 2500), 120000)
        best_h, best_c5 = _local2(best_h, nf, 1.5e-5, rng)

    # Phase 9: Ultra-fine polish
    if _el() < budget_s - 5:
        nu = min(int((budget_s - 5 - _el()) * 3500), 200000)
        best_h, best_c5 = _local2(best_h, nu, 2e-6, rng)

    # Phase 10: Final Adam + 2-point
    if _el() < budget_s - 8:
        h_new, c = _adam(best_h, 300, 1000000, 3, 0.0002)
        c = _eval(h_new)
        if c < best_c5:
            best_c5, best_h = c, h_new
        if _el() < budget_s - 3:
            nf2 = min(int((budget_s - 3 - _el()) * 4000), 100000)
            best_h, best_c5 = _local2(best_h, nf2, 1e-6, rng)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
