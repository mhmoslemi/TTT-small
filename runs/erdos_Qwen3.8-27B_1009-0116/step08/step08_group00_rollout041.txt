def _gd_descent(h, alpha, ktop, nsteps, step0, beta_mom):
    n = len(h)
    best_h = h.copy()
    best_c = _eval_val(best_h)
    v = np.zeros(n)
    step = step0
    for it in range(nsteps):
        g = _grad_smooth(h, alpha, ktop)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-16:
            break
        v = beta_mom * v + (1.0 - beta_mom) * g
        v_norm = np.linalg.norm(v)
        if v_norm < 1e-16:
            break
        v /= v_norm
        accepted = False
        cur_c = _eval_val(h)
        for _ls in range(14):
            h_new = _proj(h - step * v, n * 0.5)
            c_new = _eval_val(h_new)
            if c_new < cur_c - 1e-15:
                h = h_new
                cur_c = c_new
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
                step = min(step * 1.18, step0 * 25.0)
                accepted = True
                break
            step *= 0.38
        if not accepted:
            step = min(step * 1.4, step0 * 25.0)
    return best_h, best_c

def _swap_search(h, n_iter, rng, base_step, window):
    n = len(h)
    best_h = h.copy()
    best_c = _eval_val(best_h)
    cur_c = best_c
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, window)) % n
        d = base_step * rng.randn()
        ni = h[i] + d
        nj = h[j] - d
        if ni < 0.0 or ni > 1.0 or nj < 0.0 or nj > 1.0:
            continue
        h[i] = ni
        h[j] = nj
        c = _eval_val(h)
        if c < cur_c:
            cur_c = c
            if c < best_c:
                best_c = c
                best_h = h.copy()
        else:
            h[i] = ni - d
            h[j] = nj + d
    return best_h, best_c

def _multi_search(h, n_iter, rng, base_step, n_pert):
    n = len(h)
    best_h = h.copy()
    best_c = _eval_val(best_h)
    cur_c = best_c
    for _ in range(n_iter):
        step = base_step * (0.4 + 0.6 * rng.rand())
        idx = rng.randint(0, n, size=n_pert)
        d = step * rng.randn(n_pert)
        d -= d.mean()
        h_new = h.copy()
        h_new[idx] += d
        if h_new.min() < -0.4 or h_new.max() > 1.4:
            continue
        h_new = _repair(h_new, n)
        c = _eval_val(h_new)
        if c < cur_c:
            cur_c = c
            h = h_new
            if c < best_c:
                best_c = c
                best_h = h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _repair(h0, n)
    best_h = h.copy()
    best_c5 = _eval_val(best_h)

    def _el():
        return time.time() - t0

    # Phase 1: Multi-alpha gradient descent
    configs1 = [
        (2000, 15, 400, 0.004, 0.4),
        (8000, 20, 400, 0.003, 0.5),
        (30000, 30, 350, 0.002, 0.5),
        (100000, 40, 300, 0.001, 0.6),
        (400000, 55, 250, 0.0008, 0.6),
    ]
    for alpha, ktop, steps, s0, beta in configs1:
        if _el() > 180:
            break
        h_new, c_new = _gd_descent(best_h.copy(), alpha, ktop, steps, s0, beta)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new

    # Phase 2: Higher alpha gradient
    configs2 = [
        (2000000, 70, 200, 0.0005, 0.7),
        (10000000, 90, 150, 0.0003, 0.7),
        (50000000, 120, 100, 0.0002, 0.8),
    ]
    for alpha, ktop, steps, s0, beta in configs2:
        if _el() > 350:
            break
        h_new, c_new = _gd_descent(best_h.copy(), alpha, ktop, steps, s0, beta)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new

    # Phase 3: Multi-point local search, multi-scale
    if _el() < budget_s - 300:
        for bs in [2e-4, 1e-4, 5e-5, 2e-5, 1e-5]:
            if _el() > budget_s - 200:
                break
            n_iter = 30000
            h_new, c_new = _multi_search(best_h.copy(), n_iter, rng, bs, 3)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new

    # Phase 4: Swap-based search (preserves sum exactly)
    if _el() < budget_s - 150:
        for bs, win, niter in [(2e-5, 3, 60000), (1e-5, 5, 50000), (5e-6, 3, 60000)]:
            if _el() > budget_s - 60:
                break
            h_new, c_new = _swap_search(best_h.copy(), niter, rng, bs, win)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new

    # Phase 5: Ultra-fine swap polish
    if _el() < budget_s - 2:
        h2 = best_h.copy()
        c2 = _eval_val(h2)
        remaining = budget_s - 2 - _el()
        npol = int(min(remaining / 0.0012, 600000))
        for _ in range(npol):
            if _el() > budget_s - 2:
                break
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 2)) % n
            d = 8e-8 * rng.randn()
            ni = h2[i] + d
            nj = h2[j] - d
            if ni < 0.0 or ni > 1.0 or nj < 0.0 or nj > 1.0:
                continue
            h2[i] = ni
            h2[j] = nj
            c = _eval_val(h2)
            if c < c2:
                c2 = c
                if c < best_c5:
                    best_c5 = c
                    best_h = h2.copy()
            else:
                h2[i] = ni - d
                h2[j] = nj + d

    best_h = _repair(best_h, n)
    best_c5 = _eval_val(best_h)
    return (best_h, best_c5, n)
