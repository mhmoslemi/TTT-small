def _sum_jac(x, target):
    return np.ones(len(x))

def _slsqp(h, alpha, m_lags, maxiter):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_slsqp_obj, h, jac=_slsqp_grad,
                       args=(alpha, m_lags), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(res.x, n)
    except Exception:
        return h

def _local_search(h, n, rng, n_iter):
    c5c = _eval(h)
    best_h = h.copy()
    best_c5 = c5c
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 8)) % n
        d = 5e-6 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
                if c5n < best_c5:
                    best_c5 = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _repair(h0, n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Coordinate descent (primary algorithm)
    elapsed = time.time() - t0
    p1_budget = budget_s * 0.45
    while elapsed < p1_budget:
        h = _cd_pass(h, rng)
        h = _repair(h, n)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if (int((time.time() - t0) * 10) % 7) == 0:
            elapsed = time.time() - t0

    # Phase 2: SLSQP refinement with increasing alpha
    if time.time() - t0 < budget_s - 250:
        for alpha in [3000, 30000, 300000, 3000000]:
            if time.time() - t0 > budget_s - 200:
                break
            h_new = _slsqp(best_h.copy(), alpha, 50, 800)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new

    # Phase 3: More coordinate descent from best
    if time.time() - t0 < budget_s - 120:
        h2 = best_h.copy()
        while time.time() - t0 < budget_s - 80:
            h2 = _cd_pass(h2, rng)
            h2 = _repair(h2, n)
            c5 = _eval(h2)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h2.copy()

    # Phase 4: Another SLSQP pass at high alpha
    if time.time() - t0 < budget_s - 40:
        h_new = _slsqp(best_h.copy(), 10000000, 80, 500)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new

    # Phase 5: Final local search
    if time.time() - t0 < budget_s - 3:
        ntr = int((budget_s - 5 - (time.time() - t0)) / 0.004)
        ntr = min(ntr, 300000)
        best_h, best_c5 = _local_search(best_h, n, rng, ntr)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
