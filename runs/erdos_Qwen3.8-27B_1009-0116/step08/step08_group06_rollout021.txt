def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h + mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) * 0.5, 0, 1)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _c_full(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _c5(h):
    return float(_c_full(h).max())

def _grad_lags(h, k_indices):
    n = len(h)
    dx2 = 2.0 / n
    g = np.zeros(n)
    for k in k_indices:
        sh = (n - 1) - k
        if sh >= 0:
            g[:n-sh] += (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= h[:n-sh]
        else:
            sp = -sh
            g[sp:] += (1.0 - h[:n-sp])
            g[:n-sp] -= h[sp:]
    return g * dx2 / max(len(k_indices), 1)

class _LSECtx:
    def __init__(self, alpha, n):
        self.alpha = alpha
        self.n = n
    def f(self, x):
        c = _c_full(x)
        m = c.max()
        s = np.clip(self.alpha * (c - m), -400, 400)
        s -= s.max()
        return m + np.log(np.sum(np.exp(s))) / self.alpha
    def g(self, x):
        c = _c_full(x)
        m = c.max()
        s = np.clip(self.alpha * (c - m), -400, 400)
        w = np.exp(s - s.max())
        ws = w.sum()
        n = self.n
        dx2 = 2.0 / n
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] / ws < 1e-14:
                continue
            wk = w[k] / ws
            sh = (n - 1) - k
            if sh >= 0:
                g[:n-sh] += wk * (1.0 - x[sh:])
                if sh > 0:
                    g[sh:] -= wk * x[:n-sh]
            else:
                sp = -sh
                g[sp:] += wk * (1.0 - x[:n-sp])
                g[:n-sp] -= wk * x[sp:]
        return g * dx2
    def con_f(self, x):
        return x.sum() - self.n / 2.0
    def con_j(self, x):
        return np.ones(self.n)

def _slsqp(h, alpha, maxiter, ftol, t_end):
    n = len(h)
    ctx = _LSECtx(alpha, n)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': ctx.con_f, 'jac': ctx.con_j}]
    try:
        res = minimize(ctx.f, h, jac=ctx.g, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)
