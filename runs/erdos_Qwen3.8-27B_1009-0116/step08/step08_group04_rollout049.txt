```python
import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_lag(h, k, dx):
    n = len(h)
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        g[:n - sh] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:-sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:-sp])
        g[:n - sp] -= h[sp:]
    return g * dx

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smax_obj(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _grad_smax(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
        ct = c[idx]
    else:
        idx = np.arange(len(c))
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-14:
            continue
        g += w[ii] * _grad_lag(h, idx[ii], dx)
    return g

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _nesterov(h, alpha0, alpha_end, topk0, topk_end, steps, step0, deadline):
    n = len(h)
    f = _smax_obj(h, alpha0, topk0)
    v = np.zeros(n)
    beta = 0.85
    step = step0
    for it in range(steps):
        if time.time() > deadline:
            break
        t = it / max(steps - 1, 1)
        alpha = alpha0 * (alpha_end / alpha0) ** t
        topk = max(3, int(topk0 * (topk_end / topk0) ** t))
        g = _grad_smax(h, alpha, topk)
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, n * 0.5)
        fn = _smax_obj(hn, alpha, topk)
        if fn < f:
            h = hn
            f = fn
            step = min(step * 1.03, step0 * 20.0)
        else:
            step *= 0.5
            hn = _proj(h - step * v, n * 0.5)
            fn = _smax_obj(hn, alpha, topk)
            if fn < f:
                h = hn
                f = fn
            else:
                step *= 0.5
                hn = _proj(h - step * v, n * 0.5)
                fn = _smax_obj(hn, alpha, topk)

                    h = hn
                    f = fn
                else:
                    # Reset momentum on failure

                    if step < step0 * 0.01:

    return _repair(h, n)

def _block_anneal(h, niter, rng, deadline, T0, T1, blocks):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0+bs].copy()
        d = 0.003 * max(T, 1e-9)**0.5 * rng.randn(bs)
        d -= d.mean()
        seg += d
        seg = np.clip(seg, 0, 1)
        hn = h.copy()
        hn[i0:i0+bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h = hn
            c5c = c5n
            if c5n < bc:
                bc = c5n
                bh = h.copy()
    return bh, bc

def _coord_descent(h, deadline):
    """Focused coordinate descent on top active lags."""
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    c = _correl(h)
    topk = np.argsort(c)[-8:]  # top 8 active lags
    step0 = 1e-4
    stuck = 0
    while time.time() < deadline:
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            old = h[i]
            for d in iter([step0, -step0, step0*0.1, -step0*0.1]):
                if 0 <= old + d <= 1:
                    h[i] = old + d
                    # Adjust sum: find j to adjust
                    diff = -d
                    # distribute to j = (i+1) % n
                    j = (i + 1) % n
                    if 0 <= h[j] + diff <= 1:
                        h[j] += diff
                        c5n = _eval(h)
                        if c5n < c5c - 1e-18:
                            c5c = c5n
                            improved = True
                            if c5n < bc:
                                bc = c5n
                                bh = h.copy()
                            break
                        h[j] -= diff
                    h[i] = old
            if c5c < bc - 1e-15:
                bc = c5c
                bh = h.copy()
        if not improved:
            step0 *= 0.7
            if step0 < 1e-8:
                break
            # Periodically re-identify active lags
        else:
            stuck = 0
        stuck += 1
        if stuck > 5:
            break
    return bh, bc

def _fine_polish(h, niter, rng, deadline):
    nc = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    step_base = 5e-7
    for _ in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(nc)
        j = (i + 1 + rng.randint(0, 6)) % nc
        d = step_base * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < bc:
                    bc = c5n
                    bh = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
                step_base = max(step_base * 0.97, 1e-9)
    return bh, bc

def _lbfgs(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    def obj(x):
        return _smax_obj(x, alpha, topk)
    def jac(x):
        return _grad_smax(x, alpha, topk)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(obj, h, jac=jac, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-20, 'maxls': 20})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_lag(h, k, dx):
    n = len(h)
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        g[:n - sh] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:-sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:-sp])
        g[:n - sp] -= h[sp:]
    return g * dx

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _smax_obj(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _grad_smax(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
        ct = c[idx]
    else:
        idx = np.arange(len(c))
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-15:
            continue
        g += w[ii] * _grad_lag(h, idx[ii], dx)
    return g

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _nesterov_phase(h, alpha0, alpha1, topk0, topk1, steps, step0, deadline):
    n = len(h)
    f = _smax_obj(h, alpha0, topk0)
    v = np.zeros(n)
    beta = 0.84
    step = step0
    stale = 0
    for it in range(steps):
        if time.time() > deadline:
            break
        t = it / max(steps - 1, 1)
        alpha = alpha0 * (alpha1 / alpha0) ** t
        topk = max(2, int(round(topk0 * (topk1 / topk0) ** t)))
        g = _grad_smax(h, alpha, topk)
        v = beta * v + (1.0 - beta) * g
        accepted = False
        for trial in range(8):
            hn = _proj(h - step * v, n * 0.5)
            fn = _smax_obj(hn, alpha, topk)
            if fn < f - 1e-15:
                h = hn
                f = fn
                step = min(step * 1.04, step0 * 30.0)
                accepted = True
                break
            step *= 0.55
        if not accepted:
            v = 0.5 * g
            stale += 1
            if stale > 60:
                break
        else:
            stale = 0
            if accepted and it % 200 == 0:
                beta = min(beta + 0.005, 0.93)
    return _repair(h, n)

def _block_anneal(h, niter, rng, deadline, T0, T1, blocks):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0 + bs].copy()
        d = 0.003 * max(T, 1e-9) ** 0.5 * rng.randn(bs)
        d -= d.mean()
        seg += d
        seg = np.clip(seg, 0, 1)
        hn = h.copy()
        hn[i0:i0 + bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h = hn
            c5c = c5n
            if c5n < bc:
                bc = c5n
                bh = h.copy()
    return bh, bc

def _waterfill(h, deadline):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    dx = 2.0 / n
    steps_ = [5e-4, 2e-4, 1e-4, 5e-5, 2e-5, 1e-5, 5e-6]
    for _ in range(500):
        if time.time() > deadline:
            break
        c = _correl(h)
        top5 = np.argsort(c)[-5:]
        improved = False
        for k in top5:
            if time.time() > deadline:
                break
            g = _grad_lag(h, k, dx)
            for st in steps_:
                hn = _proj(h - st * g, n * 0.5)
                c5n = _eval(hn)
                if c5n < c5c - 1e-17:
                    h = hn
                    c5c = c5n
                    improved = True
                    break
            if c5n if improved else c5c < bc:
                bc = c5c
                bh = h.copy()
        if not improved:
            break
    return bh, bc

def _coord_descent(h, deadline):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    step0 = 2e-5
    no_improve = 0
    while time.time() < deadline and step0 > 1e-9:
        improved = False
        for i in range(0, n, 1):
            if time.time() > deadline:
                break
            old = h[i]
            j = (i + 2) % n
            for d in (step0, -step0, step0 * 0.2, -step0 * 0.2):
                ni, nj = old + d, h[j] - d
                if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                    h[i], h[j] = ni, nj
                    c5n = _eval(h)
                    if c5n < c5c - 1e-18:
                        c5c = c5n
                        improved = True
                        break
                    h[i], h[j] = old, nj + d
            if c5c < bc:
                bc = c5c
                bh = h.copy()
        if not improved:
            step0 *= 0.6
            no_improve += 1
            if no_improve > 8:
                break
        else:
            no_improve = 0
    return bh, bc

def _fine_polish(h, niter, rng, deadline):
    nc = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    step_base = 3e-7
    for _ in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(nc)
        j = (i + 1 + rng.randint(0, 5)) % nc
        d = step_base * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < bc:
                    bc = c5n
                    bh = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
                step_base = max(step_base * 0.97, 5e-10)
    return bh, bc

def _slsqp_phase(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    def obj(x):
        return _smax_obj(x, alpha, topk)
    def jac(x):
        return _grad_smax(x, alpha, topk)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(obj, h, jac=jac, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-18, 'maxls': 30})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _shake_reopt(h, rng, deadline):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    for attempt in range(3):
        if time.time() > deadline:
            break
        h_p = h.copy()
        h_p += 0.002 * rng.randn(n)
        h_p = _repair(h_p, n)
        h_r = _nesterov_phase(h_p, 50000, 500000, 6, 3, 500, 0.002, deadline)
        c5r = _eval(h_r)
        if c5r < c5c - 1e-7:
            h = h_r
            c5c = c5r
            bh = h_r.copy()
    return bh, c5c if c5c < _eval(bh) else _eval(bh)
```
