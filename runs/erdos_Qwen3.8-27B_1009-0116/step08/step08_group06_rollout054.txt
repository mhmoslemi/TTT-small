from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if h.sum() < 1e-12:
        return np.full(n, 0.5)
    if abs(h.sum() - target) > 1e-10:
        lo, hi = -3.0, 3.0
        for _ in range(80):
            mid = (lo + hi) * 0.5
            if np.clip(h - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo + hi) * 0.5, 0, 1)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * 2.0 / n)

def _grad_shift(h, s):
    """Gradient of correlation at shift s (0 <= s < n)"""
    n = len(h)
    g = np.zeros(n)
    # First term: (1-h[j-s]) for j >= s
    g[s:] += (1.0 - h[:n-s])
    # Second term: -h[j+s] for j+s < n, i.e., j < n-s
    if n - s > 0:

    return g

                step = step0 * 0.1

    return h, best_c

        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

        m = c.max()

        return g

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if h.sum() < 1e-12:
        return np.full(n, 0.5)
    if abs(h.sum() - target) > 1e-10:
        lo, hi = -3.0, 3.0
        for _ in range(80):
            mid = (lo + hi) * 0.5
            if np.clip(h - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo + hi) * 0.5, 0, 1)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * 2.0 / n)

def _grad_shift(h, s):
    n = len(h)
    g = np.zeros(n)
    if s > 0:
        g[s:] += (1.0 - h[:n - s])
    else:
        g[:] += (1.0 - h[:n])
    if n - s > 0:
        g[:n - s] -= h[s:]
    return g

def _subgrad_momentum(h, max_iter, step0, t_end):
    n = len(h)
    best_c = _eval(h)
    step = step0
    v = np.zeros(n)
    mu = 0.75
    stall = 0
    rng = np.random.RandomState(999)
    for _ in range(max_iter):
        if time.time() > t_end:
            break
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        k_worst = np.argmax(corr)
        s = k_worst if k_worst < n else k_worst - n + 1
        g = _grad_shift(h, s)
        v = mu * v + (1.0 - mu) * g
        h_new = _repair(h - step * v, n)
        c_new = _eval(h_new)
        if c_new < best_c - 1e-15:
            h, best_c = h_new, c_new
            step = min(step * 1.02, step0 * 4.0)
            stall = 0
        else:
            step *= 0.45
            stall += 1
            if stall > 40:
                h = _repair(h + 5e-5 * rng.randn(n), n)
                v *= 0.1
                step = step0 * 0.3
                stall = 0
            if step < 1e-12:
                step = step0 * 0.1
                v = np.zeros(n)
    return h, best_c

def _subgrad_topk(h, k_top, max_iter, step0, t_end):
    n = len(h)
    best_c = _eval(h)
    step = step0
    stale = 0
    for _ in range(max_iter):
        if time.time() > t_end:
            break
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        top_k = np.argsort(corr)[-k_top:]
        g = np.zeros(n)
        for ki in top_k:
            s = ki if ki < n else ki - n + 1
            g += _grad_shift(h, s)
        g /= k_top
        h_new = _repair(h - step * g, n)
        c_new = _eval(h_new)
        if c_new < best_c - 1e-15:
            h, best_c = h_new, c_new
            step = min(step * 1.015, step0 * 3.0)
            stale = 0
        else:
            step *= 0.4
            stale += 1
            if stale > 30:
                step = max(step0 * 0.05, 1e-6)
                stale = 0
    return h, best_c

def _alternating_subgrad(h, max_iter, t_end):
    n = len(h)
    best_c = _eval(h)
    step_s = 0.002
    step_k = 0.0015
    loop_s = 0
    for _ in range(max_iter):
        if time.time() > t_end:
            break
        if loop_s % 3 != 2:
            corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
            k_w = np.argmax(corr)
            s = k_w if k_w < n else k_w - n + 1
            g = _grad_shift(h, s)
            h_new = _repair(h - step_s * g, n)
            c_new = _eval(h_new)
            if c_new < best_c - 1e-15:
                h, best_c = h_new, c_new
                step_s = min(step_s * 1.02, 0.01)
            else:
                step_s *= 0.5
        else:
            corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
            top3 = np.argsort(corr)[-3:]
            g = np.zeros(n)
            for ki in top3:
                s = ki if ki < n else ki - n + 1
                g += _grad_shift(h, s)
            g /= 3.0
            h_new = _repair(h - step_k * g, n)
            c_new = _eval(h_new)
            if c_new < best_c - 1e-15:
                h, best_c = h_new, c_new
                step_k = min(step_k * 1.02, 0.008)
            else:
                step_k *= 0.5
        loop_s += 1
    return h, best_c

def _slsqp_run(h, alpha, maxiter, target):
    n = len(h)

    def _obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

    def _grad(x):
        c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-13:
                continue
            sh = k if k < n else k - n + 1
            wk = w[k] * (2.0 / n)
            if sh > 0:
                g[sh:] += wk * (1.0 - x[:n - sh])
            else:
                g[:] += wk * (1.0 - x[:n])
            if n - sh > 0:
                g[:n - sh] -= wk * x[sh:]
        return g

    def _cfun(x):
        return x.sum() - target

    def _cjac(x):
        return np.ones(n)

    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cfun, 'jac': _cjac}]
    try:
        res = minimize(_obj, h, jac=_grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _local_2pt(h, n_tries, rng, t_end):
    n = len(h)
    c_cur = _eval(h)
    best_h, best_c = h.copy(), c_cur
    step_sizes = [5e-5, 1e-5, 5e-6, 2e-6, 1e-6]
    si = 0
    accepted = 0
    for _ in range(n_tries):
        if time.time() > t_end:
            break
        if accepted > 5000 and si < len(step_sizes) - 1:
            si += 1
            accepted = 0
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 4)) % n
        d = step_sizes[si] * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c_new = _eval(h)
            if c_new < c_cur:
                c_cur = c_new
                accepted += 1
                if c_new < best_c:
                    best_c, best_h = c_new, h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return best_h, best_c

def _local_1pt(h, n_tries, rng, t_end):
    n = len(h)
    c_cur = _eval(h)
    best_h, best_c = h.copy(), c_cur
    scale = 2e-5
    for _ in range(n_tries):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        d = scale * rng.randn()
        ni = h[i] + d
        if 0 <= ni <= 1:

            c_new = _eval(h)
            if c_new < c_cur:
                c_cur = c_new
                if c_new < best_c:

                scale *= 0.98
            else:
                h[i] = h[i] - d
        else:
            h[i] = ni  # no change since we didn't assign
    return best_h, best_c

def _sweep_all_points(h, t_end):
    n = len(h)
    c_cur = _eval(h)
    best_h, best_c = h, c_cur
    for scale in [1e-4, 5e-5, 2e-5, 1e-5, 5e-6]:
