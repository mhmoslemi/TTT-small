import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if abs(h.sum() - target) < 1e-10:
        return h
    lo, hi = h.min() - 1.5, h.max() + 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    res = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - res.sum()
    if abs(diff) > 1e-12:
        n_adj = max(1, int(abs(diff) * n + 1))
        if diff > 0:
            idx = np.argsort(1.0 - res)[:n_adj]
        else:
            idx = np.argsort(res)[:n_adj]
        res[idx] += diff / n_adj
        res = np.clip(res, 0.0, 1.0)
    return res

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _proj_constraint(h, target):
    lo, hi = h.min() - 2.0, h.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def _subgrad_one(h, L, n):
    lag = L - (n - 1)
    g = np.zeros(n)
    if lag >= 0:
        g[:n - lag] += (1.0 - h[lag:])
        g[lag:] -= h[:n - lag]
    else:
        M = -lag
        g[:n - M] += (1.0 - h[M:])
        g[M:] -= h[:n - M]
    return g

def _smooth_grad(h, n, top_k, alpha):
    dx = 2.0 / n
    c = _corr_full(h) * dx
    idx = np.argsort(c)[-top_k:]
    s = alpha * (c[idx] - c.max())
    s = np.clip(s, -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] > 1e-12:
            g += w[ii] * _subgrad_one(h, k, n)
    return g, c

def _pgdescent(h, n, iters, alpha0, top_k0, t0, deadline, rng):
    target = n / 2.0
    best_h = h.copy()
    best_c = _eval(h)
    v = np.zeros(n)
    mom = 0.4
    step = 0.04
    alpha = alpha0
    top_k = top_k0
    n_egg = 0

    for it in range(iters):
        if time.time() - t0 > deadline:
            break
        g, c = _smooth_grad(h, n, top_k, alpha)
        v = mom * v + (1.0 - mom) * g
        mom = min(mom + 0.0003, 0.85)

        taken = False
        s = step
        for _ in range(12):
            h_new = _proj_constraint(h - s * v, target)
            c_new = _eval(h_new)
            if c_new < best_c - 1e-15:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                n_egg = 0
                step = min(s * 1.05, 0.3)
                taken = True
                break
            s *= 0.45
        if not taken:
            n_egg += 1
            step = max(s * 2.5, 0.02)
            if n_egg > 50:
                h = best_h + 0.005 * rng.randn(n)
                h = _repair(h, n)
                n_egg = 0
                step = 0.03

        if it % 300 == 0:
            alpha = min(alpha * 1.35, 80000)
            top_k = min(top_k + 3, 250)

    return best_h, best_c

def _coord_search(h, n, rng, t0, deadline):
    target = n / 2.0
    best_h = h.copy()
    best_c = _eval(h)
    step_sizes = [5e-4, 1e-4, 3e-5, 1e-5, 3e-6, 1e-6]

    for _ in range(20000000):
        if time.time() - t0 > deadline:
            break
        step = step_sizes[ss_idx]
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 15)) % n
        d = step * rng.randn()
        ni, nj = best_h[i] + d, best_h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h_tmp = best_h.copy()
            h_tmp[i], h_tmp[j] = ni, nj
            c_new = _eval(h_tmp)

                best_c = c_new

                    ss_idx = max(0, ss_idx - 1)
                _rng_set_count(rng, 0)
            else:
                _rng_incr_count(rng)
        else:
            continue
    return best_h, best_c

def _rng_improve_count(rng):
    return getattr(rng, '_improve_count', 0)

def _rng_set_count(rng, v):
    rng._improve_count = v

def _rng_incr_count(rng):
    rng._improve_count = getattr(rng, '_improve_count', 0) + 1

def _anneal_restart(h, n, rng, t0, deadline):
    best_h = h.copy()
    best_c = _eval(h)
    temp = 0.01
    for gen in range(5000):
        if time.time() - t0 > deadline:
            break
        temp *= 0.995
        h_new = h + temp * rng.randn(n)
        h_new = _repair(h_new, n)
        c_new = _eval(h_new)
        if c_new < best_c:
            best_h = h_new.copy()
            best_c = c_new
        elif rng.rand() < np.exp(-(c_new - best_c) / max(temp, 1e-6)):
            pass
        h = h_new if c_new < best_c + 0.01 else best_h
        if c_new < best_c:
            best_h = h_new.copy()
            best_c = c_new
    return best_h, best_c
