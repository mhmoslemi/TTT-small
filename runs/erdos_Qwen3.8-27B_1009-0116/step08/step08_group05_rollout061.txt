import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _coeff_vec(h, i, n):
    """Coefficient of h[i] in each correlation component."""
    n_corr = 2 * n - 1
    c = np.zeros(n_corr)
    m_lo = max(-(n-1), -i)
    m_hi = min(n-1, n-1-i)
    if m_lo <= m_hi:
        ms = np.arange(m_lo, m_hi+1)
        c[n-1+ms] += (1.0 - h[i+ms])
    m_lo2 = max(-(n-1), i-(n-1))
    m_hi2 = min(n-1, i)
    if m_lo2 <= m_hi2:
        ms2 = np.arange(m_lo2, m_hi2+1)
        c[n-1+ms2] -= h[i-ms2]
    return c

def _min_pwl(a, b):
    """Minimize max(a + b*t) over t in [0,1] using golden section."""
    lo, hi = 0.0, 1.0
    gr = (np.sqrt(5.0) - 1.0) / 2.0  # golden ratio conjugate
    t1 = lo + gr * (hi - lo)
    t2 = hi - gr * (hi - lo)
    f1 = (a + b * t1).max()
    f2 = (a + b * t2).max()
    for _ in range(30):
        if f1 < f2:
            hi = t2; t2 = t1; f2 = f1
            t1 = lo + gr * (hi - lo)
            f1 = (a + b * t1).max()
        else:
            lo = t1; t1 = t2; f1 = f2
            t2 = hi - gr * (hi - lo)
            f2 = (a + b * t2).max()
    return (lo + hi) * 0.5

def _coord_descent(h, n, n_sweeps, rng, deadline):
    """Coordinate descent on the exact max-of-lines objective."""
    best_h = h.copy()
    best_c5 = _eval(h)
    c_cur = _corr_raw(h).copy()
    n_corr = len(c_cur)
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        order = rng.permutation(n)
        for i in order:
            old = h[i]
            if old < 1e-15 or old > 1.0 - 1e-15:
                continue
            coeff = _coeff_vec(h, i, n)
            a_vec = c_cur - coeff * old
            t_new = _min_pwl(a_vec, coeff)
            if abs(t_new - old) < 1e-16:
                continue
            c_cur += coeff * (t_new - old)
            h[i] = t_new
            c5_new = c_cur.max() * 2.0 / n
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        if sweep % 100 == 99:
            c_cur = _corr_raw(h)
            h = _repair(h, n)
    return best_h, best_c5

def _pairwise_polish(h, n, n_iter, rng, deadline):
    """Optimize pairs of coordinates simultaneously."""
    best_h = h.copy()
    best_c5 = _eval(h)
    c_cur = _corr_raw(h).copy()
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        old_i, old_j = h[i], h[j]
        coeff_i = _coeff_vec(h, i, n)
        coeff_j = _coeff_vec(h, j, n)
        a_vec = c_cur - coeff_i * old_i - coeff_j * old_j
        # Now c = a + coeff_i * t_i + coeff_j * t_j
        # Minimize max_k (a[k] + coeff_i[k]*t_i + coeff_j[k]*t_j)
        # For fixed t_i, this is 1D in t_j (and vice versa)
        # Do alternating 1D:
        for _ in range(8):
            # Optimize t_i (fix t_j)
            a1 = a_vec + coeff_j * h[j]
            t_i_new = _min_pwl(a1, coeff_i)
            # Optimize t_j (fix t_i)
            a2 = a_vec + coeff_i * h[i]
            t_j_new = _min_pwl(a2, coeff_j)
            if abs(t_i_new - h[i]) < 1e-15 and abs(t_j_new - h[j]) < 1e-15:
                break
            h[i] = t_i_new
            h[j] = t_j_new
        c_cur = _corr_raw(h)
        h = _repair(h, n)
        c5_new = c_cur.max() * 2.0 / n
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
    return best_h, best_c5

def _slsqp_quick(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    def obj(x):
        c = _corr_raw(x) * (2.0 / n)
        m = c.max()
        if topk < len(c):
            idx = np.argpartition(c, -topk)[-topk:]
            ct = c[idx]
        else:
            ct = c
        s = np.clip(alpha * (ct - m), -300, 300)
        return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha
    try:
        r = minimize(obj, h, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5}],
                     options={'maxiter': maxit, 'ftol': 1e-18, 'maxls': 40})
        return _repair(r.x, n)
    except Exception:
        return h

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _coeff_vec(h, i, n):
    n_corr = 2 * n - 1
    c = np.zeros(n_corr)
    m_lo = max(-(n-1), -i); m_hi = min(n-1, n-1-i)
    if m_lo <= m_hi:
        ms = np.arange(m_lo, m_hi+1)
        c[n-1+ms] += (1.0 - h[i+ms])
    m_lo2 = max(-(n-1), i-(n-1)); m_hi2 = min(n-1, i)
    if m_lo2 <= m_hi2:
        ms2 = np.arange(m_lo2, m_hi2+1)
        c[n-1+ms2] -= h[i-ms2]
    return c

def _min_pwl(a, b):
    lo, hi = 0.0, 1.0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    t1 = lo + gr*(hi-lo); t2 = hi - gr*(hi-lo)
    f1 = (a+b*t1).max(); f2 = (a+b*t2).max()
    for _ in range(25):
        if f1 < f2:
            hi=t2; t2=t1; f2=f1; t1=lo+gr*(hi-lo); f1=(a+b*t1).max()
        else:
            lo=t1; t1=t2; f1=f2; t2=hi-gr*(hi-lo); f2=(a+b*t2).max()
    return (lo+hi)*0.5

def _coord_descent(h, n, n_sweeps, rng, deadline):
    best_h = h.copy(); best_c5 = _eval(h)
    c_cur = _corr_raw(h).copy()
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        order = rng.permutation(n)
        for i in order:
            old = h[i]
            if old < 1e-15 or old > 1.0-1e-15:
                continue
            coeff = _coeff_vec(h, i, n)
            a_vec = c_cur - coeff*old
            t_new = _min_pwl(a_vec, coeff)
            if abs(t_new - old) < 1e-16:
                continue
            c_cur += coeff*(t_new - old)
            h[i] = t_new
            if c_cur.max()*2.0/n < best_c5:
                best_c5 = c_cur.max()*2.0/n
                best_h = h.copy()
        if sweep % 200 == 199:
            c_cur = _corr_raw(h); h = _repair(h, n)
    return best_h, best_c5

def _pairwise_polish(h, n, n_iter, rng, deadline):
    best_h = h.copy(); best_c5 = _eval(h)
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        ci = _coeff_vec(h, i, n)
        cj = _coeff_vec(h, j, n)
        c_cur = _corr_raw(h)
        for _ in range(6):
            oi = h[i]; oj = h[j]
            a1 = c_cur - ci*oi
            h[i] = _min_pwl(a1 + cj*oj, ci)
            a2 = _corr_raw(h) - cj*h[j]
            h[j] = _min_pwl(a2, cj)
            if abs(h[i]-oi) < 1e-15 and abs(h[j]-oj) < 1e-15:
                break
        h = _repair(h, n)
        c5n = _eval(h)
        if c5n < best_c5:
            best_c5 = c5n; best_h = h.copy()
    return best_h, best_c5

def _slsqp_q(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    def obj(x):
        c = _corr_raw(x)*(2.0/n); m = c.max()
        idx = np.argpartition(c, -topk)[-topk:]; ct = c[idx]
        s = np.clip(alpha*(ct-m), -300, 300)
        return m + (s.max()+np.log(np.sum(np.exp(s-s.max()))))/alpha
    try:
        r = minimize(obj, h, method='SLSQP',
                     bounds=[(0.0,1.0)]*n,
                     constraints=[{'type':'eq','fun':lambda x: x.sum()-n*0.5}],
                     options={'maxiter':maxit,'ftol':1e-18,'maxls':30})
        return _repair(r.x, n)
    except Exception:
        return h

def _fine_sweep(h, n, n_iter, rng, deadline, step0):
    bh = h.copy(); c5c = _eval(bh)
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        i = rng.randint(n); j = (i+1+rng.randint(0,4))%n
        d = step0*(2*rng.random()-1)
        ni, nj = bh[i]+d, bh[j]-d
        if 0<=ni<=1 and 0<=nj<=1:
            bh[i],bh[j] = ni,nj
            c5n = _eval(bh)
            if c5n < c5c-1e-19:
                c5c = c5n
            else:
                bh[i],bh[j] = ni-d, nj+d
    return bh, c5c

    return (best_h, best_c5, n)
