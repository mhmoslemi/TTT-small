                cur_c = c_new
                step = min(step * 1.12, step0 * 20.0)
                taken = True
                stall = 0
                break
            step *= 0.35
        if not taken:
            step = min(step * 1.4, step0 * 20.0)
            stall += 1
            if stall > 8:
                h = _repair(h + 0.005 * rng.randn(n), n)
                v = np.zeros(n)
                cur_c = _eval(h)
                stall = 0
    return _repair(h, n)

def _targeted_local(h, n_iter, rng, base_step, n_pert):
    n = len(h)
    c5cur = _eval(h)
    best_h = h.copy()
    best_c = c5cur
    c_arr = _eval_arr(h)
    top_k = max(10, n // 30)
    top_lags = np.argsort(c_arr)[-top_k:]
    for _ in range(n_iter):
        k = top_lags[rng.randint(len(top_lags))]
        L = k - (n - 1)
        if L >= 0:
           overlap = h[L:] * (1.0 - h[:n - L])
            nrel = n - L
            pos1 = np.arange(L, n)
            pos2 = np.arange(0, n - L)
        else:
            M = -L
            overlap = h[:n - M] * (1.0 - h[M:])
            nrel = n - M
            pos1 = np.arange(0, n - M)
            pos2 = np.arange(M, n)
        wts = overlap + 1e-12
        p = wts / wts.sum()
        picks = rng.choice(nrel, size=n_pert, replace=False, p=p)
        idx = np.unique(np.concatenate([pos1[picks], pos2[picks]]))
        step = base_step * (0.4 + 1.2 * rng.rand())
        d = step * rng.randn(len(idx))
        d -= d.mean()
        h_new = h.copy()
        h_new[idx] += d
        if h_new.min() < -0.3 or h_new.max() > 1.3:
            continue
        h_new = _repair(h_new, n)
        c5n = _eval(h_new)
        if c5n < c5cur:
            c5cur = c5n
            h = h_new
            if c5n < best_c:
                best_c = c5n
                best_h = h.copy()
                c_arr = _eval_arr(h)
                top_lags = np.argsort(c_arr)[-top_k:]
    return best_h, best_c

def _random_local(h, n_iter, rng, n_pert, base_step):
    n = len(h)
    c5cur = _eval(h)
    best_h = h.copy()
    best_c = c5cur
    for _ in range(n_iter):
        step = base_step * (0.5 + rng.rand())
        idx = rng.randint(0, n, size=n_pert)
        d = step * rng.randn(n_pert)
        d -= d.mean()
        h_new = h.copy()
        h_new[idx] += d
        if h_new.min() < -0.5 or h_new.max() > 1.5:
            continue
        h_new = _repair(h_new, n)
        c5n = _eval(h_new)
        if c5n < c5cur:
            c5cur = c5n
            h = h_new
            if c5n < best_c:
                best_c = c5n
                best_h = h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _repair(h0, n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: SLSQP alpha-dive with increasing focus
    alpha_seq = [5000, 20000, 100000, 500000, 3000000, 20000000, 200000000, 2000000000]
    for alpha in alpha_seq:
        if _el() > 350:
            break
        lift = min(120, max(25, int(35 * np.log10(alpha / 5000) + 25)))
        h_new = _slsqp(best_h.copy(), alpha, lift, 600, 1e-15)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new

    # Phase 2: Projected gradient with multi-config sweep
    if _el() < budget_s - 350:
        configs = [
            (5000, 40, 800, 0.01),
            (50000, 50, 600, 0.006),
            (300000, 60, 500, 0.004),
            (2000000, 80, 400, 0.002),
            (10000000, 100, 300, 0.001),
            (50000000, 120, 200, 0.0005),
        ]
        for alpha, ml, steps, s0 in configs:
            if _el() > budget_s - 280:
                break
            h_new = _proj_grad(best_h.copy(), alpha, ml, steps, s0, rng)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new

    # Phase 3: More SLSQP at very high alpha
    if _el() < budget_s - 250:
        for alpha in [10000000000, 50000000000]:
            if _el() > budget_s - 200:
                break
            h_new = _slsqp(best_h.copy(), alpha, 150, 400, 1e-16)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new

    # Phase 4: Targeted local search - coarse to fine
    if _el() < budget_s - 80:
        phases = [
            (8, 5e-5),
            (5, 2e-5),
            (3, 8e-6),
            (2, 3e-6),
        ]
        for npert, bstep in phases:
            if _el() > budget_s - 30:
                break
            nit = min(int((budget_s - 30 - _el()) / 2.5), 25000)
            best_h, best_c5 = _targeted_local(best_h, max(nit, 500), rng, bstep, npert)

    # Phase 5: Random local search for final polish
    if _el() < budget_s - 15:
        nit = min(int((budget_s - 15 - _el()) / 2.0), 40000)
        best_h, best_c5 = _random_local(best_h, max(nit, 500), rng, 3, 1.5e-6)

    # Phase 6: Ultra-fine 2-point swap
    if _el() < budget_s - 1:
        h2 = best_h.copy()
        c5c = _eval(h2)
        nc2 = len(h2)
        npol = min(int((budget_s - 1 - _el()) / 0.6), 300000)
        for _ in range(npol):
            if _el() > budget_s - 1:
                break
            i = rng.randint(nc2)
            j = (i + 1 + rng.randint(0, 9)) % nc2
            if i == j:
                continue
            d = 1.5e-7 * rng.randn()
            ni, nj = h2[i] + d, h2[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h2[i], h2[j] = ni, nj
                c5n = _eval(h2)
                if c5n < c5c:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5 = c5n
                        best_h = h2.copy()
                else:
                    h2[i], h2[j] = ni - d, nj + d

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
