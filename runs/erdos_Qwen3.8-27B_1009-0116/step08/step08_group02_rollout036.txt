The objective is: f(x) = c_k(x) = Σᵢ x[i]·(1-x[i+k]) for a specific lag k.

For lag k > 0 (shift by k): f = Σ_{i=0}^{n-1-k} x[i]·(1-x[i+k])
∂f/∂x[i] = (1-x[i+k]) for i ∈ [0, n-1-k)  [from the x[i] factor]
∂f/∂x[i] -= x[i-k] for i ∈ [k, n)  [from the (1-x[i]) factor where i = j+k]

So:
- For i < n-k: ∂f/∂x[i] += (1-x[i+k])
- For i >= k: ∂f/∂x[i] -= x[i-k]

For lag k < 0 (shift by |k| to the left): f = Σ_{i=|k|}^{n-1} x[i]·(1-x[i-|k|])
∂f/∂x[i] = (1-x[i-|k|]) for i ∈ [|k|, n)
∂f/∂x[i] -= x[i+|k|] for i ∈ [0, n-|k|)

- If shift < 0: use the second formula

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        if s > 0:
            h *= target / s
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)

def _grad_lag(kw, x, n):
    k = kw - (n - 1)
    g = np.zeros(n)
    if k == 0:
        g[:] = 1.0 - 2.0 * x
    elif k > 0:
        g[:n-k] += (1.0 - x[k:])
        g[k:] -= x[:n-k]
    else:
        s = -k
        g[s:] += (1.0 - x[:n-s])
        g[:n-s] -= x[s:]
    return g

def _slsqp(h, alpha, top_k, maxiter):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0

    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        if top_k > 0 and top_k < len(c):

        else:
            idx = np.arange(len(c))
        cs = c[idx]
        m = cs.max()
        s = np.clip(alpha * (cs - m), -400, 400)
        s -= s.max()
        return m + np.log(np.exp(s).sum()) / alpha

    def grad(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        if top_k > 0 and top_k < len(c):

        m = cs.max()
        s = np.clip(alpha * (cs - m), -400, 400)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        g = np.zeros(n)
        for ii in range(len(idx)):
            if w[ii] < 1e-15:
                continue
            g += w[ii] * _grad_lag(int(idx[ii]), x, n)
        return g * dx

    bnds = [(0.0, 1.0)] * n
