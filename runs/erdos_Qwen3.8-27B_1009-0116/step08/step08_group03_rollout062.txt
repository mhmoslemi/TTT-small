    # Part 2: conv[j]
    part2 = conv
    
    bilinear = -(part1 + part2)
    
    # k=0 contribution: wP[0]*dP[0]/dh[j] = wP[0]*(1-2*h[j])
    # (wN[0] is 0 since Nv[0] doesn't exist)
    k0 = wP[0] * (1.0 - 2.0 * h)
    
    return dx * (linP + linN + bilinear + k0)

def _smooth_grad(h, alpha):
    P, Nv = _pn_arrays(h)
    n = len(h)
    mx = float(max(P[0], np.max(Nv)))
    wP = np.zeros(n)
    wN = np.zeros(n)
    e0 = np.exp(np.clip(alpha * (P[0] - mx), -400.0, 0.0))
    tot = e0
    if n > 1:
        ep = np.exp(np.clip(alpha * (P[1:] - mx), -400.0, 0.0))
        en = np.exp(np.clip(alpha * (Nv[1:] - mx), -400.0, 0.0))
        tot += ep.sum() + en.sum()
        if tot <= 0.0:
            tot = 1.0
            e0 = 1.0
        wP[0] = e0 / tot
        wP[1:] = ep / tot
        wN[1:] = en / tot
    else:
        wP[0] = 1.0
    return _grad_from_weights(h, wP, wN)

def _subgrad(h):
    """Subgradient of C5(h) = max_k overlap_k. Uses the active constraint(s)."""
    P, Nv = _pn_arrays(h)
    n = len(h)
    mx = float(max(P[0], np.max(Nv)))
    tol = 8.0 * (2.0 / n)
    wP = np.zeros(n)
    wN = np.zeros(n)
    cnt = 0
    if P[0] >= mx - tol:
        wP[0] = 1.0
        cnt += 1
    if n > 1:
        mP = P[1:] >= mx - tol
        if np.any(mP):
            pos = np.nonzero(mP)[0] + 1
            wP[pos] = 1.0
            cnt += pos.size
        mN = Nv[1:] >= mx - tol
        if np.any(mN):
            pos = np.nonzero(mN)[0] + 1
            wN[pos] = 1.0
            cnt += pos.size
    if cnt == 0:
        return np.zeros(n)
    inv = 1.0 / cnt
    wP *= inv
    wN *= inv
    return _grad_from_weights(h, wP, wN)

def _projected_grad(h, g, step):
    return _proj(h - step * g, len(h) * 0.5)

def _fista(h0, alpha, max_time, L=10.0):
    n = len(h0)
    S = n * 0.5
    t0 = time.time()
    h = _repair(h0, n)
    hp = h.copy()
    y = h.copy()
    t_ratio = 1.0
    fy = _smooth_value(h, alpha)
    best_h = h.copy()
    best_c = _c5_direct(h)
    stale = 0
    Lcur = L
    for it in range(500000):
        if time.time() - t0 > max_time:
            break
        if it:
            tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_ratio * t_ratio))
            y = h + ((t_ratio - 1.0) / tn) * (h - hp)
            t_ratio = tn
            if y.min() < -0.5 or y.max() > 1.5 or abs(y.sum() - S) > 0.1:
                y = _proj(y, S)
            fy = _smooth_value(y, alpha)
        else:
            y = h
            fy = _smooth_value(y, alpha)
        g = _smooth_grad(y, alpha)
        step = 1.0 / max(Lcur, 1e-12)
        accepted = False
        for _ in range(20):
            xn = _proj(y - step * g, S)
            fx = _smooth_value(xn, alpha)
            d = xn - y
            rhs = fy + float(np.dot(g, d)) + 0.5 / step * float(np.dot(d, d))
            if fx <= rhs + 1e-10:
                accepted = True
                break
            step *= 0.5
        Lcur = 1.0 / max(step, 1e-12)
        if accepted:
            hp = h
            h = xn
            stale = 0
            if it % 10 == 0:
                c = _c5_direct(h)
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
        else:
            stale += 1
            hp = h
            y = h
            t_ratio = 1.0
            fy = _smooth_value(y, alpha)
            Lcur = max(Lcur * 0.5, 1.0)
            if stale > 15:
                break
    c = _c5_direct(h)
    if c < best_c:
        best_c = c
        best_h = h.copy()
    return h, best_h, best_c

            recalc = it

        step = step0 * (0.5) ** frac
        
        # Try projected gradient step
        hn = _proj(h - step * g, n * 0.5)

        else:

            gi = int(np.argmin(g))

def _binarize(h, n, rng, n_iters=500):
    """Push h towards binary values via threshold + repair."""
    S = n * 0.5
    for _ in range(n_iters):
        # Sort and threshold
        h_sorted = np.sort(h)
        # The S/2 smallest become 0, S/2 largest become 1 (approximately)
        # Actually we want sum = S, so if binary: exactly S positions are 1
        # But S = n/2, so exactly n/2 positions are 1
        h2 = (h >= 0.5).astype(float)
        if h2.sum() > S:
            # Too many 1s, flip some
            idx = np.where(h2 == 1)[0]
            order = h[idx][::-1].argsort()  # smallest h among 1s
            n_flip = int(h2.sum() - S)
            for i in order[:n_flip]:
                h2[idx[i]] = 0
        elif h2.sum() < S:
            idx = np.where(h2 == 0)[0]
            order = h[idx].argsort()[::-1]  # largest h among 0s
            n_flip = int(S - h2.sum())
            for i in order[:n_flip]:
                h2[idx[i]] = 1
        h = h2
    return h

def _resample(h, m):
    n = len(h)
    if n == m:
        return _repair(h, n)
    x0 = (np.arange(n) + 0.5) * (2.0 / n)
    x1 = (np.arange(m) + 0.5) * (2.0 / m)
    out = np.interp(x1, x0, h)
    return _proj(out, m * 0.5)

import numpy as np
import time

def _fft_size(n):
    m = 2 * n - 1
    N = 1
    while N < m:
        N <<= 1
    return N

def _repair(h, n):
    t = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if s <= 1e-15:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= t / s
        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
        if abs(s - t) < 1e-12:
            break
    d = t - h.sum()
    if abs(d) > 1e-12:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            a = min(abs(d), 1.0 - h[i]) if d > 0 else min(abs(d), h[i])
            h[i] += a if d > 0 else -a
            d = t - h.sum()
            if abs(d) < 1e-13:
                break
    return h

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)
    x += (S - x.sum()) / x.size
    for _ in range(20):

        d = S - x.sum()
        if abs(d) < 1e-11:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
    if abs(x.sum() - S) > 1e-8:
        lo, hi = float(np.min(v)) - 1.0, float(np.max(v)) + 1.0
        for _ in range(50):
            mid = 0.5 * (lo + hi)
            if np.clip(v - mid, 0.0, 1.0).sum() > S:
                lo = mid
            else:
                hi = mid
        x = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    return x

def _c5_direct(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _pn_arrays(h):
    n = len(h)
    N = _fft_size(n)
    H = np.fft.rfft(h, N)
    ac = np.fft.irfft(H * np.conj(H), N).real[:n]
    pref = np.empty(n + 1)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    total = pref[n]
    P = np.empty(n)
    P[0] = total - ac[0]
    Nv = np.zeros(n)
    if n > 1:
        idx = np.arange(1, n)
        P[idx] = pref[n - idx] - ac[idx]
        Nv[idx] = total - pref[idx] - ac[idx]
    return P, Nv

        e += np.exp(np.clip(alpha * (P[1:] - mx), -400.0, 0.0)).sum()
        e += np.exp(np.clip(alpha * (Nv[1:] - mx), -400.0, 0.0)).sum()
    return dx * (mx + np.log(max(e, 1e-300)) / alpha)

    n = len(h)
    mx = float(max(P[0], np.max(Nv)))

        ep = np.exp(np.clip(alpha * (P[1:] - mx), -400.0, 0.0))
        en = np.exp(np.clip(alpha * (Nv[1:] - mx), -400.0, 0.0))
        tot += ep.sum() + en.sum()
        if tot <= 0:
            tot, e0 = 1.0, 1.0
        wP[0], wP[1:], wN[1:] = e0 / tot, ep / tot, en / tot
    else:

    return _grad_fw(h, wP, wN)

def _subgrad(h):
    P, Nv = _pn_arrays(h)
    n = len(h)
    mx = float(max(P[0], np.max(Nv)))
    tol = max(1e-9, 8.0 * (2.0 / n))
    wP, wN = np.zeros(n), np.zeros(n)
    cnt = 0
    if P[0] >= mx - tol:
        wP[0], cnt = 1.0, 1
    if n > 1:
        mP = P[1:] >= mx - tol
        if np.any(mP):
            wP[1:][mP] = 1.0
            cnt += int(mP.sum())
        mN = Nv[1:] >= mx - tol
        if np.any(mN):
            wN[1:][mN] = 1.0
            cnt += int(mN.sum())
    if cnt == 0:
        return np.zeros(n)
    wP /= cnt
    wN /= cnt
    return _grad_fw(h, wP, wN)

def _resample(h, m):
    n = len(h)
    if n == m:
        return _repair(h, n)
    x0 = (np.arange(n) + 0.5) * (2.0 / n)
    x1 = (np.arange(m) + 0.5) * (2.0 / m)
    return _proj(np.interp(x1, x0, h), m * 0.5)

    best_h, best_c = h.copy(), c_ref
    for _ in range(n_trials):
        h2 = np.array(h, dtype=float)
        for _ in range(3):
            h2[np.random.RandomState(rng.randint(2**31)).rand(n) < 0.5] = 0.0
            h2[h2 > 0.0] = 1.0
            if h2.sum() > S:
                idx = np.where(h2 == 1)[0]
                n_fl = int(h2.sum() - S)
                order = np.argsort(h2[idx])
                h2[idx[order[:n_fl]]] = 0
            elif h2.sum() < S:
                idx = np.where(h2 == 0)[0]
                n_fl = int(S - h2.sum())
                order = np.argsort(h2[idx])[::-1]
                h2[idx[order[:n_fl]]] = 1
        if h2.sum() != S:
            h2 = _repair(h2, n)
        c2 = _c5_direct(h2)
        if c2 < best_c:
            best_c, best_h = c2, h2.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 2.0
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(560, 0.5)
    n0 = len(h0)
    h0 = _repair(h0, n0)
    best, best_c = h0.copy(), _c5_direct(h0)
    # Quick variants
    for fn in (lambda x: 0.5 + 0.5 * (x - x[::-1]), lambda x: 0.5 * (x + x[::-1]), lambda x: 1.0 - x):
        hh = _proj(fn(h0), n0 * 0.5)
        c = _c5_direct(hh)
        if c < best_c:
            best, best_c = hh, c
    # Phase 1: FISA sweep
    end1 = t0 + min(50.0, 0.06 * budget_s)
    if end1 > time.time() + 5:
        hcur = best.copy()
        for a in [30, 100, 500, 2000, 8000]:
            if time.time() > end1 - 5:
                break
            slot = min(10.0, end1 - time.time() - 2)
            if slot < 3:
                break
            hf, bh, bc = _fista(hcur, a, slot)
            ch = _c5_direct(hf)
            if bc < best_c:
                best, best_c = bh, bc
            if ch < best_c:
                best, best_c = hf, ch
            hcur = hf if ch <= bc else bh
    # Phase 2: Main FISTA at full res
    end2 = t0 + min(350.0, 0.35 * budget_s)
    if end2 > time.time() + 10:
        hcur = best.copy()
        for a in [100, 400, 1500, 6000, 25000, 100000, 400000]:
            if time.time() > end2 - 15:
                break
            slot = min(35.0, end2 - time.time() - 5)
            if slot < 5:
                break
            hf, bh, bc = _fista(hcur, a, slot)
            ch = _c5_direct(hf)
            if bc < best_c:
                best, best_c = bh, bc
            if ch < best_c:
                best, best_c = hf, ch
            hcur = hf if ch <= bc else bh
    # Phase 3: Multi-resolution
    if budget_s > 500:
        end3 = min(t0 + 0.55 * budget_s, deadline - 200)
        for m in [n0 // 2, n0 // 4]:
            if m < 32 or time.time() > end3 - 30:
                break
            hm = _resample(best, m)
            for a in [200, 1000, 5000]:
                if time.time() > end3 - 10:
                    break
                slot = min(18.0, end3 - time.time() - 5)
                if slot < 4:
                    break
                hf, bh, bc = _fista(hm, a, slot)
                if bc < _c5_direct(hm):
                    hm = bh
            hu = _resample(hm, n0)
            cu = _c5_direct(hu)
            if cu < best_c:
                best, best_c = hu, cu
            if time.time() < end3 - 8:
                hf, bh, bc = _fista(hu, 5000, end3 - time.time() - 4)
                if bc < best_c:
                    best, best_c = bh, bc
    # Phase 4: Directed local search
    end4 = min(t0 + 0.82 * budget_s, deadline - 80)
    if end4 > time.time() + 10:
        b, c = _targeted_search(best, rng, end4)
        if c < best_c:
            best, best_c = b, c
    # Phase 5: Binary perturbation + final polish
    end5 = deadline
    if end5 > time.time() + 5:
        bh2, bc2 = _try_binary(best, n0, rng, 100)
        if bc2 < best_c:
            best, best_c = bh2, bc2
        b, c = _targeted_search(best, rng, end5)
        if c < best_c:
            best, best_c = b, c
    best = _repair(best, len(best))
    best_c = _c5_direct(best)
    return best, best_c, len(best)
