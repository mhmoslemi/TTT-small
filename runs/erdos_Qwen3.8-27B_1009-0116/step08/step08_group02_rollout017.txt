```python
def _delta_swap_vec(h, n, i, j):
    """Vectorized delta for swap h[i]:1→0, h[j]:0→1"""
    s_vals = np.arange(-(n-1), n)  # shape (2n-1,)
    # i-s: shape (2n-1,), valid where 0 <= i-s < n
    i_minus_s = i - s_vals
    i_plus_s = i + s_vals
    j_minus_s = j - s_vals
    j_plus_s = j + s_vals
    
    dc = np.zeros(2*n-1)
    # -(1-h[i-s]) where valid
    m1 = (i_minus_s >= 0) & (i_minus_s < n)
    dc[m1] -= (1.0 - h[i_minus_s[m1]])
    # +h[i+s] where valid
    m2 = (i_plus_s >= 0) & (i_plus_s < n)
    dc[m2] += h[i_plus_s[m2]]
    # +(1-h[j-s]) where valid
    m3 = (j_minus_s >= 0) & (j_minus_s < n)
    dc[m3] += (1.0 - h[j_minus_s[m3]])
    # -h[j+s] where valid
    m4 = (j_plus_s >= 0) & (j_plus_s < n)
    dc[m4] -= h[j_plus_s[m4]]
    return dc
```
