        h_new = _repair(h_new, n)
        c5n = _eval(h_new)
        if c5n < c5cur - 1e-15:
            c5cur = c5n
            h = h_new
            if c5n < best_c:
                best_c = c5n
                best_h = h.copy()
    return best_h, best_c

def _downsample(h, f):
    n = len(h)
    m = n // f
    return _repair(h[:m * f].reshape(m, f).mean(axis=1).copy(), m)

def _upsample(h, f):
    n = len(h)
    out = np.empty(n * f)
    for i in range(n):
        out[i * f:(i + 1) * f] = h[i]
    return _repair(out, n * f)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    h = _repair(h0, n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Coarse optimization at reduced resolution
    for ratio in [3, 5, 8]:
        if _el() > 55:
            break
        nc = max(48, n // ratio)
        f = n // nc
        hc = _downsample(h, f)
        for alpha in [3000, 50000, 500000]:
            if _el() > 75:
                break
            hc = _slsqp_opt(hc, alpha, 20, 400, 1e-13)
        h_up = _upsample(hc, f)
        c5u = _eval(h_up)
        if c5u < best_c5:
            best_c5 = c5u
            best_h = h_up.copy()

    # Phase 2: Nesterov accelerated gradient sweep
    nest_configs = [
        (10000, 30, 600, 0.004, 0.80),
        (100000, 40, 500, 0.003, 0.85),
        (1000000, 50, 400, 0.002, 0.90),
        (10000000, 60, 350, 0.001, 0.92),
        (100000000, 70, 300, 0.0005, 0.94),
    ]
    for alpha, ntop, steps, s0, mu in nest_configs:
        if _el() > budget_s * 0.45:
            break
        h_new = _nest_grad(best_h.copy(), alpha, ntop, steps, s0, mu)
        c5n = _eval(h_new)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new

    # Phase 3: SLSQP with increasing smoothness
    if _el() < budget_s - 200:
        for alpha, ntop, mx in [(200000, 40, 400), (2000000, 50, 350),
                                (20000000, 60, 300), (200000000, 80, 250)]:
            if _el() > budget_s - 150:
                break
            h_new = _slsqp_opt(best_h, alpha, ntop, mx, 1e-15)
            c5n = _eval(h_new)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h_new

    # Phase 4: Gradient-informed local search
    if _el() < budget_s - 80:
        for alpha, ntop, bs, nit in [(50000, 40, 2e-5, 50000),
                                      (500000, 50, 5e-6, 80000),
                                      (5000000, 60, 1e-6, 100000)]:
            remaining = budget_s - 80 - _el()
            if remaining < 5:
                break
            ntr = min(nit, int(remaining / 0.02))
            best_h, best_c5 = _grad_search(best_h, ntr, rng, alpha, ntop, bs)

    # Phase 5: Pure local search with decreasing steps
    if _el() < budget_s - 15:
        for bs in [5e-6, 1e-6, 2e-7, 5e-8]:
            remaining = budget_s - 15 - _el()
            if remaining < 2:
                break
            ntr = min(int(remaining / 0.012), 150000)
            best_h, best_c5 = _local_search(best_h, ntr, rng, bs, 2)

    # Phase 6: Ultra-fine 2-3 point polish with adaptive step
    if _el() < budget_s - 2:
        h2 = best_h.copy()
        nc2 = len(h2)
        c5c = _eval(h2)
        step = 3e-7
        npol = min(int((budget_s - 2 - _el()) / 0.002), 400000)
        no_improve = 0
        for _ in range(npol):
            if _el() > budget_s - 2:
                break
            npts = 2 if rng.rand() < 0.7 else 3
            idx = rng.randint(0, nc2, size=npts)
            d = step * rng.randn(npts)
            d -= d.mean()
            h2[np.arange(len(idx))] += 0  # no-op to avoid mutation
            ok = True
            for ii in range(npts):
                nv = h2[idx[ii]] + d[ii]
                if nv < 0 or nv > 1:
                    ok = False
                    break
                h2[idx[ii]] = nv
            if ok:
                h2 = _repair(h2, nc2)
                c5n = _eval(h2)
                if c5n < c5c:
                    c5c = c5n
                    no_improve = 0
                    if c5n < best_c5:
                        best_c5 = c5n
                        best_h = h2.copy()
                    step = min(step * 1.02, 5e-6)
                else:
                    step *= 0.95
                    no_improve += 1
                    if no_improve > 5000:
                        step = 2e-7
                        no_improve = 0
            else:
                # revert
                for ii in range(npts):
                    h2[idx[ii]] -= d[ii]

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
