  - So ∂corr/∂h[j] = (1-h[j+d]) * [j <= n-1-d] - h[j-d] * [j >= d]

- If d < 0 (let d' = -d):
  - Term where i = j: contributes (1-h[j-d']) if d' <= j <= n-1
  - Term where i-d' = j, i.e., i = j+d': contributes -h[j+d'] if 0 <= j+d' <= n-1, i.e., 0 <= j <= n-1-d'
  - So ∂corr/∂h[j] = (1-h[j-d']) * [j >= d'] - h[j+d'] * [j <= n-1-d']

                c5c = c5n
                if c5n < best_c:
                    best_c = c5n
                    best_h = h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return best_h, best_c

def _local_informed(h, n_iter, rng, base_step):
    n = len(h)
    c5c = _eval(h)
    k_w = int(np.argmax(np.correlate(h, 1.0-h, mode='full')))
    best_h = h.copy()
    best_c = c5c
    sh = (n-1) - k_w
    for _ in range(n_iter):
        moved = False
        if 0 < sh < n:
            i = rng.randint(n - sh)
            j = i + sh
            d = abs(base_step * rng.randn())
            ni, nj = h[i] - d, h[j] + d
            if 0.0 <= ni and 0.0 <= nj <= 1.0:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c:
                    c5c = c5n
                    k_w = int(np.argmax(np.correlate(h, 1.0-h, mode='full')))
                    sh = (n-1) - k_w
                    if c5n < best_c:
                        best_c = c5n
                        best_h = h.copy()
                else:
                    h[i], h[j] = h[i] + d, h[j] - d
                moved = True
        if not moved:
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 4)) % n
            d = base_step * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c:
                    c5c = c5n
                    if c5n < best_c:
                        best_c = c5n
                        best_h = h.copy()
                else:
                    h[i], h[j] = h[i] - d, h[j] + d
    return best_h, best_c

def _binary_local(h, n_iter, rng):
    """Binary flip local search for worst-lag improvement"""
    n = len(h)
    hb = (h >= 0.5).astype(float)
    s = hb.sum()
    target = n // 2
    if s != target:

        if s > target:

            for idx in order[:int(s - target)]:

            for idx in order[int(s - target):int(s)]:  # hmm this is wrong

            hb = _repair(h, n)
            hb = (hb >= 0.5).astype(float)

            # Just work with continuous
            return _local_informed(hb, n_iter, rng, 0.5)
    
    c5c = _eval(hb)
    best_h = hb.copy()
    best_c = c5c
    ones = np.where(hb == 1.0)[0].astype(int)
    zeros = np.where(hb == 0.0)[0].astype(int)
    
    for it in range(n_iter):
        # Weighted: 50% directed, 50% random
        if rng.rand() < 0.5:
            c_arr = np.correlate(hb, 1.0 - hb, mode='full')
            k_w = int(np.argmax(c_arr))
            d = k_w - (n - 1)
            # score[i] = contribution to worst lag if h[i] is 1
            # For flipping h[i] 1→0: Δc_worst = -(1-h[i+d]) + h[i-d] (for d>=0)
            # We want i where this is most negative (biggest reduction)
            # = h[i+d] - 1 + h[i-d] (for binary, with h[OOB]=0)
            # So we want to maximize h[i+d] + h[i-d] for h[i]=1
            scores_1 = np.zeros(ones.shape)
            for idx, i in enumerate(ones):
                v = 0.0
                if 0 <= i + d < n:
                    v += hb[i + d]
                if 0 <= i - d < n:
                    v += hb[i - d]
                scores_1[idx] = v
            # Flip the one with highest score (removing it most reduces worst lag)
            i_best = ones[np.argmax(scores_1)]
            j_rand = rng.choice(zeros)
        else:
            i_best = rng.choice(ones)
            j_rand = rng.choice(zeros)
        
        if i_best == j_rand:
            continue
        hb[i_best], hb[j_rand] = 0.0, 1.0
        new_ones = np.where(hb == 1.0)[0].astype(int)
        new_zeros = np.where(hb == 0.0)[0].astype(int)
        c5n = _eval(hb)
        if c5n < c5c - 1e-15:
            c5c = c5n
            ones, zeros = new_ones, new_zeros
            if c5n < best_c:
                best_c = c5n
                best_h = hb.copy()
        else:
            hb[i_best], hb[j_rand] = 1.0, 0.0
    
    return best_h, best_c
