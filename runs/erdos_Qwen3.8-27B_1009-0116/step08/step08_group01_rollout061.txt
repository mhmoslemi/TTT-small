import numpy as np

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5*(lo+hi)
        if float(np.clip(h-mid, 0, 1).sum()) > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h - 0.5*(lo+hi), 0, 1)

def _eval_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0-h, mode='full')) * 2.0/n)

def _c5_arr(h):
    n = len(h)
    return np.correlate(h, 1.0-h, mode='full') * (2.0/n)

def _sm_grad(h, alpha):
    n = len(h)
    dx = 2.0/n
    c = _c5_arr(h)
    m = c.max()
    s = np.clip(alpha*(c-m), -400, 400)
    s -= s.max()
    we = np.exp(s)
    ws = we.sum()
    if ws < 1e-300:
        return np.zeros(n)
    we /= ws
    g = np.zeros(n)
    for ki in range(2*n-1):
        wi = we[ki]
        if wi < 1e-14:
            continue
        sh = ki - (n-1)
        wk = wi * dx
        if sh >= 0:
            g[:n-sh] += wk*(1.0-h[sh:])
            g[sh:] -= wk*h[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk*(1.0-h[:n-sp])
            g[:n-sp] -= wk*h[sp:]
    return g

def _fista(h0, alpha, niter, target, step0):

    x = h0.copy()
    y = h0.copy()
    tm = 1.0

    bx = x.copy()
    for it in range(niter):
        g = _sm_grad(y, alpha)
        xn = _proj_sum(y - step*g, target)
        tmn = 0.5*(1.0+np.sqrt(1.0+4.0*tm*tm))
        y = xn + (tm-1.0)/tmn*(xn-x)
        x = xn
        tm = tmn
        if it % 200 == 0:

                bx = x.copy()
    cn = _eval_c5(x)
    if cn < bc:
        bc = cn
        bx = x.copy()
    return _proj_sum(bx, target), bc

def _pair_pass(h, target, offs, ts, rng):

    nimp = 0

            hi, hj = h[i], h[j]

                if c < bc:
                    bc = c
                    nimp += 1
                    hi, hj = ni, nj
                else:
                    h[i], h[j] = hi, hj
    return bc, nimp
