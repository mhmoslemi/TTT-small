                i1 = np.arange(sp,n); g[i1] += w[k]*(1.0-x[i1-sp])*dx
                i2 = np.arange(n-sp); g[i2] -= w[k]*x[i2+sp]*dx
        return g
    bnds = [(0.0,1.0)]*n
    cons = [{'type':'eq','fun':lambda x: x.sum()-n/2.0,'jac':lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter':maxiter,'ftol':1e-16})
        return _repair(res.x, n)
    except:
        return h

def _random_binary(n, rng):
    h = (rng.rand(n) < 0.5).astype(float)
    target = n//2
    ones = np.where(h==1.0)[0]; zeros = np.where(h==0.0)[0]
    diff = target - len(ones)
    if diff > 0: h[rng.choice(zeros,diff,replace=False)] = 1.0
    elif diff < 0: h[rng.choice(ones,-diff,replace=False)] = 0.0
    return h

def _perturb_bin(h, n, nswaps, rng):
    h = h.copy()
    ones = np.where(h>0.5)[0].tolist()
    zeros = np.where(h<=0.5)[0].tolist()
    for _ in range(nswaps):
        if len(ones)==0 or len(zeros)==0: break
        zi=zeros[rng.integers(len(zeros))]; oj=ones[rng.integers(len(ones))]
        h[zi]=1.0; h[oj]=0.0
        ones.remove(oj); zeros.remove(zi); ones.append(zi); zeros.append(oj)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    def _el(): return time.time() - t0

    n = len(initial_h_values)
    h0 = np.array(initial_h_values, dtype=float)
    h_bin = _binarize(h0, n)
    best_h = h_bin.copy()
    best_c5 = _eval_max(best_h)
    best_cont = _repair(h0.copy(), n)
    best_cont_c5 = _eval_max(best_cont)

    # Phase 1: Coarse-to-fine (40s)
    for ratio in [4, 2]:
        if _el() > 40: break
        nc = n // ratio
        hc = _binarize(_downsample(best_h, nc), nc)
        hc, _ = _binary_sa(hc, nc, 30000, rng, 0.02, 0.002)
        h_up = _binarize(_upsample(hc, ratio), n)
        if _eval_max(h_up) < best_c5:
            best_c5 = _eval_max(h_up); best_h = h_up.copy()

    # Phase 2: Main binary SA (300s)
    if _el() < budget_s - 500:
        best_h, best_c5 = _binary_sa(best_h, n, 500000, rng, 0.035, 0.003)

    # Phase 3: Greedy with 2-swaps (250s)
    if _el() < budget_s - 350:
        best_h, best_c5 = _binary_greedy(best_h, n, 300000, rng, 0.3)

    # Phase 4: Second SA round (150s)
    if _el() < budget_s - 250:
        best_h, best_c5 = _binary_sa(best_h, n, 250000, rng, 0.015, 0.001)

    # Phase 5: Multi-start (150s)
    if _el() < budget_s - 180:
        for attempt in range(3):
            if _el() > budget_s - 180: break
            if attempt == 0:
                start = _perturb_bin(best_h, n, 15, rng)
            elif attempt == 1:
                start = _random_binary(n, rng)
            else:
                start = _perturb_bin(h_bin, n, 30, rng)
            s_h, s_c = _binary_sa(start, n, 100000, rng, 0.04, 0.005)
            s_h, s_c = _binary_greedy(s_h, n, 100000, rng, 0.25)
            if s_c < best_c5:
                best_c5 = s_c; best_h = s_h.copy()

    # Phase 6: Continuous SLSQP refinement (80s)
    if _el() < budget_s - 80:
        h_ref = _repair(best_h + rng.randn(n)*0.005, n)
        h_ref = _slsqp_refine(h_ref, n, 150)
        c5r = _eval_max(h_ref)
        if c5r < best_c5:
            best_c5 = c5r; best_h = h_ref.copy()

    # Phase 7: Final greedy polish (remainder)
    if _el() < budget_s - 3:
        h_final = _binarize(best_h, n)
        cf = _eval_max(h_final)
        if cf < best_c5:
            best_c5 = cf; best_h = h_final.copy()
        # Time calibrate
        _tt = _el()
        for _ in range(40): np.correlate(best_h, 1.0-best_h, mode='full')
        eps = max((_el()-_tt)/40.0, 1e-7)
        ntr = int((budget_s - 3 - _el()) / eps * 0.85)
        best_h, best_c5 = _binary_greedy(best_h, n, max(ntr, 5000), rng, 0.2)

    best_c5 = _eval_max(best_h)
    return (best_h, best_c5, n)
