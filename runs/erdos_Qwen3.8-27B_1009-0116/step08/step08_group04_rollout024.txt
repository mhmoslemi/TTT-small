    return g

        h = _slsqp_solve(h, alpha, tk, mi, dl1)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h.copy()

    # Phase 2: Projected gradient Nesterov (100-280s)
    dl2 = min(deadline, t0 + 280)
    h_w = best_h.copy()
    for alpha0, st, ns in [(8000, 0.004, 2500), (50000, 0.002, 2500), (400000, 0.001, 2000)]:
        if time.time() > dl2:
            break
        h_w = _proj_grad(h_w, alpha0, 12, ns, st, dl2)
        c5v = _eval(h_w)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_w.copy()

    # Phase 3: Multi-resolution (280-400s)
    dl3 = min(deadline, t0 + 400)
    if time.time() < dl3 - 30:
        h_du = _down_up_optimize(best_h, n, dl3)
        c5v = _eval(h_du)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_du.copy()

    # Phase 4: Multi-start with diverse perturbations (400-560s)
    dl4 = min(deadline, t0 + 560)
    if time.time() < dl4 - 20:
        h_ms, c5_ms = _multistart(best_h, n, rng, dl4, 35)
        if c5_ms < best_c5:
            best_c5 = c5_ms
            best_h = h_ms.copy()

    # Phase 5: Subgradient on exact max (560-700s)
    dl5 = min(deadline, t0 + 700)
    h_sg = _subgrad_max(best_h.copy(), 5, 0.002, 5000, dl5)
    c5v = _eval(h_sg)
    if c5v < best_c5:
        best_c5 = c5v
        best_h = h_sg.copy()

    # Phase 6: Aggressive local search (700-950s)
    dl6 = min(deadline, t0 + 950)
    best_h, best_c5 = _local_search(best_h.copy(), dl6, rng)

    # Phase 7: Fine coordinate sweep (950-1060s)
    dl7 = min(deadline, t0 + 1060)
    best_h, best_c5 = _fine_coord(best_h.copy(), dl7, rng)

    # Phase 8: Final ultra-fine local search (1060-end)
    dl8 = deadline
    best_h, best_c5 = _local_search(best_h.copy(), dl8, rng, min_step=1e-11)

    # Final repair and evaluation
    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
