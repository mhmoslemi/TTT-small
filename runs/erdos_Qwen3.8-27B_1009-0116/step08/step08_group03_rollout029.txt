def _fft_size(n):
    m = 2 * n - 1
    N = 1
    while N < m:
        N <<= 1
    return N

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = 0.5 * n
    for _ in range(50):
        s = h.sum()
        if abs(s - t) < 1e-12:
            break
        h *= t / max(s, 1e-15)
        np.clip(h, 0.0, 1.0, out=h)
    d = t - h.sum()
    if abs(d) > 1e-9:
        idx = np.arange(n)
        if d > 0:
            order = idx[np.argsort(1.0 - h)]
        else:
            order = idx[np.argsort(h)]
        for i in order:
            if d > 0:
                a = min(d, 1.0 - h[i])
                h[i] += a
                d -= a
            else:
                a = min(-d, h[i])
                h[i] -= a
                d += a
            if abs(d) < 1e-12:
                break
    return h

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)
    x += (S - x.sum()) / x.size

import numpy as np

def _fft_size(n):
    m = 2 * n - 1
    N = 1
    while N < m:
        N <<= 1
    return N

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = 0.5 * n
    for _ in range(50):
        s = h.sum()
        if abs(s - t) < 1e-12:
            break
        h *= t / max(s, 1e-15)
        np.clip(h, 0.0, 1.0, out=h)
    d = t - h.sum()
    if abs(d) > 1e-9:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            if d > 0:
                a = min(d, 1.0 - h[i])
                h[i] += a; d -= a
            else:
                a = min(-d, h[i])
                h[i] -= a; d += a
            if abs(d) < 1e-12:
                break
    return h

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)
    x += (S - x.sum()) / x.size
    for _ in range(10):
        np.clip(x, 0.0, 1.0, out=x)
        d = S - x.sum()
        if abs(d) < 1e-11:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
    return x

def _c5(h):

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _ov(h):

    ac = np.fft.irfft(H * np.conj(H), N).real[:n]

    pref = np.concatenate([[0.0], np.cumsum(h)])

    ev = np.exp(np.clip(alpha * (P - mx), -500, 0)).sum() + np.exp(np.clip(alpha * (Nv - mx), -500, 0)).sum()
    return mx + np.log(max(ev, 1e-300)) / alpha

    P = np.zeros(n); Nv = np.zeros(n)

    if n > 1:

def _resample(h, m):
