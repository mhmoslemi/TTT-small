import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            a = min(abs(diff), (1.0 - h[idx]) if diff > 0 else h[idx])
            h[idx] += a if diff > 0 else -a
            diff -= a if diff > 0 else -a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_shift(h, p, dx):

    sh = (n - 1) - p
    g = np.zeros(n)
    if sh >= 0:
        g[:n - sh] += (1.0 - h[sh:])

    else:
        sp = -sh
        g[sp:] += (1.0 - h[:n - sp])
        g[:n - sp] -= h[sp:]
    return g * dx

def _smax(h, alpha, topk):
    c = _correl(h)
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        idx = np.arange(len(c))
        ct = c
    m = ct.max()
    s = np.clip(alpha * (ct - m), -400, 400)
    sm = s.max()
    return m + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha

def _grad_smax(h, alpha, topk):

    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
        ct = c[idx]
    else:
        idx = np.arange(len(c))

    s = np.clip(alpha * (ct - m), -400, 400)

    n = len(h)

    return g

def _nesterov(h, alpha, grow, topk, steps, step0, deadline):
    n = len(h)
    f = _smax(h, alpha, topk)
    v = np.zeros(n)
    beta = 0.85
    step = step0
    for it in range(steps):
        if time.time() > deadline:
            break
        if it > 0 and it % 150 == 0:
            alpha = min(alpha * grow, 5e7)
        g = _grad_smax(h, alpha, topk)
        gn = np.linalg.norm(g)
        if gn > 20:
            g *= 20.0 / gn
        v = beta * v + (1.0 - beta) * g

    return _repair(h, n)

def _fourier_anneal(h, niter, rng, deadline, T0, T1):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    t_grid = np.arange(n) / n * 2.0 * np.pi
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        nmod = rng.randint(1, 4)
        kmax = max(2, n // 3)
        ks = rng.randint(1, kmax, size=nmod)
        ph = rng.uniform(0, 2 * np.pi, size=nmod)
        am = rng.uniform(0.002, 0.015, size=nmod)
        d = np.zeros(n)

            h = hn
            c5c = c5n
            if c5n < bc - 1e-15:

                bh = h.copy()
    return bh, bc

def _slsqp_opt(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    def obj(x):
        return _smax(x, alpha, topk)
    def jac(x):
        return _grad_smax(x, alpha, topk)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(obj, h, jac=jac, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-16})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _resample(h, n_tgt):
    n_src = len(h)
    xs = np.linspace(0, 2, n_src, endpoint=False) + 1.0 / n_src
    xt = np.linspace(0, 2, n_tgt, endpoint=False) + 1.0 / n_tgt
    return _repair(np.clip(np.interp(xt, xs, h, period=2.0), 0, 1), n_tgt)

def _topk_focus(h, niter, deadline):

    c = _correl(h)
    bc = float(np.max(c))
    bh = h.copy()
    dx = 2.0 / n

def _pair_polish(h, niter, rng, deadline):

    c5c = _eval(h)
