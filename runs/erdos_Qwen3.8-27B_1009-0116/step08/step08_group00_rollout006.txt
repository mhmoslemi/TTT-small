        else:

            g[:n - M] -= dx * h[M:]

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    for _ in range(200):
        if s < 1e-15:
            return np.full(n, 0.5)
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_lag(h, k_idx, n, dx):
    """Gradient of correlation at index k_idx."""
    L = k_idx - (n - 1)
    M = abs(L)
    g = np.zeros(n)
    if M == 0:
        g[:] = dx * (1.0 - 2.0 * h)
    else:
        g[M:] += dx * (1.0 - h[:n - M])
        g[:n - M] -= dx * h[M:]
    return g

def _grad_smooth(h, alpha, m_lags):
    """Gradient of log-sum-exp of top m_lags."""
    n = len(h)
    dx = 2.0 / n
    c = _corr(h) * dx
    top = np.argsort(c)[-m_lags:]
    sm = c[top]
    sm -= sm.max()
    w = np.exp(alpha * sm)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k_idx in enumerate(top):
        if w[ii] < 1e-15:
            continue
        g += w[ii] * _grad_lag(h, k_idx, n, dx)
    return g

def _obj_smooth(h, alpha, m_lags):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h) * dx
    m = c.max()
    top = np.argsort(c)[-m_lags:]
    s = alpha * (c[top] - m)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _sum_fn(x, target):
    return x.sum() - target

def _sum_jac(x, target):
    return np.ones(len(x))

def _slsqp(h, alpha, m_lags, maxiter, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fn, 'jac': _sum_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(_obj_smooth, h, jac=_grad_smooth,
                       args=(alpha, m_lags), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h

def _nesterov(h, m_lags, steps, step0, mu, alpha):
    """Nesterov accelerated projected gradient."""
    n = len(h)
    v = np.zeros(n)
    step = step0
    cur = _eval(h)
    h_prev = h.copy()
    
    for _ in range(steps):
        y = np.clip(h + mu * (h - h_prev), 0, 1)
        y = _proj(y, n * 0.5)
        g = _grad_smooth(y, alpha, m_lags)
        v = mu * v + g
        h_prev = h.copy()
        
        taken = False
        for _ in range(10):
            h_new = _proj(h - step * v, n * 0.5)
            c_new = _eval(h_new)
            if c_new < cur - 1e-16:
                h = h_new
                cur = c_new
                step = min(step * 1.05, step0 * 5.0)
                taken = True
                break
            step *= 0.4
        if not taken:
            step = min(step * 1.3, step0 * 5.0)
    
    return _repair(h, n), cur

    n = len(h)
    c5 = _eval(h)
    best = h.copy()
    best_c5 = c5
    
    for it in range(n_iter):
        # Compute gradient
        g = _grad_smooth(h, alpha, m_lags)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:

        # Gradient step with small random perturbation
        step = 1e-5 / (g_norm + 1e-10)
        direction = -g + 0.1 * rng.randn(n)

        c5n = _eval(h_new)

            h = h_new
            c5 = c5n
            if c5 < best_c5:
                best_c5 = c5
                best = h.copy()
        else:
            # Try coordinate descent: adjust single coordinate with largest |g|
            j = int(np.argmax(np.abs(g)))
            delta = -np.sign(g[j]) * 5e-6 * (0.5 + rng.rand())
            h_new2 = h.copy()
            # Find partner to maintain sum
            k = (j + 1 + rng.randint(0, 10)) % n
            h_new2[j] += delta
            h_new2[k] -= delta
            if 0 <= h_new2[j] <= 1 and 0 <= h_new2[k] <= 1:
                c5n2 = _eval(h_new2)
                if c5n2 < c5:
                    h = h_new2
                    c5 = c5n2
                    if c5 < best_c5:
                        best_c5 = c5
                        best = h.copy()
    
    return best, best_c5

def _pairwise_polish(h, n_trials, rng):

    n = len(h)
    c5 = _eval(h)
    best = h.copy()
    best_c5 = c5
    
    for _ in range(n_trials):
        i = rng.randint(n)
