def _repair(h, n):

    target = n / 2.0

    # Bisection to find shift

            hi = mid
        else:

    if abs(diff) > 1e-12:

This is a min-max problem. A simple approach:

2. For each t, check feasibility: is there h such that c_k(h) ≤ t for all k, 0≤h≤1, sum(h)=n/2?
3. The feasibility check is a quadratic programming problem (c_k is quadratic in h)

But this QP is non-convex (product of two variables). So binary search doesn't directly work.

However, if we linearize: c_k(h) ≈ c_k(h₀) + ∇c_k(h₀)·(h-h₀), then the feasibility becomes a linear program!

This gives us a **successive linearization** algorithm:
1. Start with h₀
2. Compute c_k(h₀) for all k, find max
3. Linearize c_k around h₀
4. Add constraint: c_k_linearized ≤ t for the active k's
5. Solve the LP: min 0 s.t. c_k_lin(h) ≤ t, 0≤h≤1, sum(h)=n/2
6. Update h₀, go to 2

This is like a cutting-plane method! And LPs are very fast to solve.

def _cutting_planes(h, n, t_end, max_iter=200, tol=1e-8):
    """Cutting-plane method for min-max problem"""
    from scipy.optimize import linprog
    h_best = h.copy()
    best_t = _eval(h)
    
    for it in range(max_iter):
        if time.time() > t_end:
            break
        # Compute correlation and gradient
        dx = 2.0 / n
        c = np.correlate(h, 1.0-h, mode='full') * dx
        m = c.max()
        
        # Find active set (top lags)
        active_thresh = m - 0.001  # within 0.001 of max
        active = np.where(c >= active_thresh)[0]
        
        # Build LP: min t s.t. g_k^T h + d_k <= t, 0<=h<=1, sum(h)=n/2
        # Variables: [h(0), ..., h(n-1), t]
        # Constraints: g_k[0]*h[0] + ... + g_k[n-1]*h[n-1] - t <= d_k
        # where g_k = gradient of c_k at h, d_k = c_k(h) - g_k^T * h
        # Plus: sum(h) = n/2, 0 <= h[i] <= 1
        
        nvars = n + 1  # h(0..n-1), t
        # Objective: minimize t (last variable)
        c_obj = np.zeros(nvars)
        c_obj[-1] = 1.0
        
        # Inequality constraints: g_k^T h - t <= d_k
        # i.e., g_k[0]*h[0] + ... + g_k[n-1]*h[n-1] + (-1)*t <= d_k
        A_ub = []
        b_ub = []
        
        for k in active:

            d_k = c[k] - np.dot(gk, h)  # constant term after linearization
            row = np.zeros(nvars)
            row[:n] = gk
            row[-1] = -1.0
            A_ub.append(row)
            b_ub.append(d_k)
        
        A_ub = np.array(A_ub) if A_ub else np.zeros((1, nvars))
        b_ub = np.array(b_ub) if b_ub else np.array([0.0])
        
        # Equality: sum(h) = n/2
        A_eq = np.zeros((1, nvars))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n/2.0])
        
        bounds = [(0,1)]*n + [(0, 1.0)]  # h in [0,1], t in [0,1]
        
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, 
                      A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')

            t_new = res.x[-1]
            h_new = _repair(h_new, n)
            f_new = _eval(h_new)

                h_best = h_new.copy()
                h = h_new
        
        # Check convergence
        if m - active_thresh < tol:

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h + mid, 0, 1).sum() >= target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) * 0.5, 0, 1)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            a = min(abs(diff), max(0, (1-h[idx]) if diff > 0 else h[idx]))
            h[idx] += np.sign(diff) * a
            diff -= np.sign(diff) * a
            if abs(diff) < 1e-12:
                break
    return h

    return np.correlate(h, 1.0-h, mode='full') * (2.0/n)

def _grad_lag(h, n, k):
    """Gradient of c_k with respect to h, where c_k = corr[k] * dx"""
    dx = 2.0 / n
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        # c_k = dx * sum_{i=0}^{n-1-sh} h[i+sh] * (1-h[i])
        # d/dh[j]: 
        #   if j >= sh: h[j] appears as h[i+sh] with i = j-sh, contribution: dx*(1-h[j-sh])
        #   if j < n-sh: h[j] appears in (1-h[i]) with i=j, contribution: -dx*h[j+sh]
        g[sh:] += dx * (1.0 - h[:n-sh])
        g[:n-sh] -= dx * h[sh:]
    else:

        # c_k = dx * sum_{i=0}^{n-1-sp} h[i] * (1-h[i+sp])

    for k in top_idx:
        g += _grad_lag(h, n, k)
    g /= top_k
    return g

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h + mid, 0, 1).sum() >= target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) * 0.5, 0, 1)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            a = min(abs(diff), max(0.0, (1.0 - h[idx]) if diff > 0 else h[idx]))
            h[idx] += np.sign(diff) * a
            diff -= np.sign(diff) * a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)))

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_lag(h, n, k):
    dx = 2.0 / n
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        g[sh:] += dx * (1.0 - h[:n - sh])
        g[:n - sh] -= dx * h[sh:]
    else:
        sp = -sh
        g[:n - sp] += dx * (1.0 - h[sp:])
        g[sp:] -= dx * h[:n - sp]
    return g

def _topk_grad(h, n, k_top):
    c = _corr(h)
    idx = np.argsort(c)[-k_top:]
    g = np.zeros(n)
    for k in idx:
        g += _grad_lag(h, n, k)
    g /= float(k_top)
    return g

def _nesterov(h, n, t_end, k_top, steps, step0):
    best_h, best_f = h.copy(), _eval(h)
    v = np.zeros(n)
    mu = 0.85
    step = step0
    stall = 0
    for _ in range(steps):
        if time.time() > t_end:
            break
        g = _topk_grad(h, n, k_top)
        v = mu * v + (1.0 - mu) * g
        h_new = _repair(h - step * v, n)
        f_new = _eval(h_new)
        if f_new < best_f:
            best_f, best_h = f_new, h_new.copy()
            stall = 0
            step = min(step * 1.015, step0 * 6.0)
        else:
            stall += 1
            step *= 0.65
            if stall > 40:
                v[:] = 0.0
                step = step0 * 0.4
                stall = 0
        h = h_new
    return best_h, best_f

def _slsqp_run(h, alpha, n, maxiter, t_end):
    dx = 2.0 / n
    target = n / 2.0
    def _obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        s -= s.max()
        return m + np.log(np.sum(np.exp(s))) / alpha
    def _grad(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-14:
                continue
            g += w[k] * _grad_lag(x, n, k)
        return g
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_obj, h, jac=_grad, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _local_search(h, n, t_end, rng):
    best_h, best_f = h.copy(), _eval(h)
    step = 5e-4
    min_step = 5e-7
    stall = 0
    while time.time() < t_end:
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 8)) % n
        if i == j:
            continue
        d = step * rng.randn()
        ni, nj = best_h[i] + d, best_h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            best_h[i], best_h[j] = ni, nj
            f_new = _eval(best_h)
            if f_new < best_f:
                best_f = f_new
                stall = 0
                step = min(step * 1.01, 5e-3)
            else:
                best_h[i], best_h[j] = ni - d, nj + d
                stall += 1
                step = max(step * 0.85, min_step)
        else:
            stall += 1
            step = max(step * 0.9, min_step)
        if stall > 500 and step <= min_step * 1.5:
            break
    return best_h, best_f

def _block_search(h, n, t_end, rng):
    best_h, best_f = h.copy(), _eval(h)
    step = 2e-4
    while time.time() < t_end:
        i0 = rng.randint(n - 4)
        blen = rng.randint(2, 5)
        idx = np.arange(i0, min(i0 + blen, n))
        d = step * rng.randn(len(idx))
        d -= d.mean()
        new_vals = best_h[idx] + d
        if np.all(new_vals >= 0) and np.all(new_vals <= 1):
            best_h[idx] = new_vals
            f_new = _eval(best_h)
            if f_new < best_f:
                best_f = f_new
            else:
                best_h[idx] = new_vals - d
        step *= 0.98
    return best_h, best_f

    t0 = time.time()
    t_total_end = t0 + budget_s
    rng = np.random.RandomState(seed)
