def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if h.sum() < 1e-12:
        return np.full(n, 0.5)
    if abs(h.sum() - target) > 1e-10:
        lo, hi = -3.0, 3.0
        for _ in range(80):
            mid = (lo + hi) * 0.5
            if np.clip(h - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo + hi) * 0.5, 0, 1)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _grad_exact_topk(h, k_top):
    """Gradient of sum of top-k correlation values"""
    n = len(h)
    dx2 = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx2
    top_k = np.argsort(c)[-k_top:]

        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]
    return g

def _projected_grad_descent(h, k_top, n_iter, step0, t_end):
    """Projected gradient descent with adaptive step size"""
    n = len(h)
    f = _eval(h)
    v = np.zeros(n)

    step = step0
    best_h, best_f = h.copy(), f

            break
        g = _grad_exact_topk(h, k_top)
        h_prev = h.copy()
        f_prev = f
        v = mu * v + (1.0 - mu) * g
        h_new = _repair(h_prev - step * v, n)

        if f_new < f:

            step = min(step * 1.05, step0 * 5.0)
        else:
            step *= 0.5
            if step < 1e-12:
                v = np.zeros(n)
                step = step0
            h = h_prev
    return best_h, best_f

def _anneal_local(h, rng, t_end, n_tries_max=50000):
    """Simulated annealing style local search"""
    n = len(h)
    f_curr = _eval(h)
    best_h, best_f = h.copy(), f_curr
    t = 0.01  # temperature
    t_decay = 0.9999
    for _ in range(n_tries_max):
        if time.time() > t_end:
            break
        # Random 2-4 point perturbation
        n_pts = rng.choice([2, 3, 4], p=[0.4, 0.4, 0.2])

        # Perturb maintaining sum

        deltas -= deltas.mean()  # zero mean to preserve sum approximately

            if f_new < best_f:
                best_f, best_h = f_new, h.copy()
        t *= t_decay
    return best_h, best_f

def _multi_point_local(h, rng, t_end):

        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

        m = c.max()

            else:
                sp = -sh
                g[sp:] += wk * (1.0 - x[:n-sp])
                g[:n-sp] -= wk * x[sp:]
        return g
    
    def con_fun(x):
        return x.sum() - target
    def con_jac(x):
        return np.ones(n)

        c3 = _eval(h3)
        if c3 < best_c5:
            best_h, best_c5 = h3, c3

    # Phase 4: Gradient descent k=3
    t_ph = t0 + budget_s * 0.42
    if time.time() < t_ph:
        h4, f4 = _grad_descent(best_h.copy(), 3, 5000, 0.002, t_ph)
        if f4 < best_c5:
            best_h, best_c5 = h4, f4

    # Phase 5: SLSQP high alpha
    t_ph = t0 + budget_s * 0.52
    if time.time() < t_ph:
        h5 = _slsqp_run(best_h.copy(), 200000, 300, t_ph)
        c5 = _eval(h5)
        if c5 < best_c5:
            best_h, best_c5 = h5, c5

    # Phase 6: Gradient descent k=1 (exact max)
    t_ph = t0 + budget_s * 0.64
    if time.time() < t_ph:
        h6, f6 = _grad_descent(best_h.copy(), 1, 8000, 0.0008, t_ph)
        if f6 < best_c5:
            best_h, best_c5 = h6, f6

    # Phase 7: ILS perturb + optimize
    t_ph = t0 + budget_s * 0.82
    if time.time() < t_ph:
        h7, f7 = _ils_perturb_optimize(best_h, rng, t_ph, 15)
        if f7 < best_c5:
            best_h, best_c5 = h7, f7

    # Phase 8: Multi-scale local search (most important for final push)
    t_ph = t_global_end - 3
    if time.time() < t_ph:
        scales = [
            (8, 0.02),
            (5, 0.01),
            (3, 0.005),
            (2, 0.002),
            (2, 0.0005),
            (2, 0.0001),
        ]
        h8, f8 = _multi_scale_local(best_h, rng, t_ph, scales)
        if f8 < best_c5:
            best_h, best_c5 = h8, f8

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
