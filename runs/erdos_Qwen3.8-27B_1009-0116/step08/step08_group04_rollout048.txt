For 2-coordinate optimization: fix all h except h[i] and h[j], with the constraint h[i]+h[j] = s (constant).     h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)

                    a = min(diff, 1.0 - h[idx])

                    diff -= a

                    h[idx] -= a
                    diff += a

    return h

    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / len(h))
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]

        idx = np.arange(len(c))

    s = np.clip(alpha * (ct - m), -400, 400)

    return m + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha

def _smax_grad(h, alpha, topk):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    if topk < len(c):

        idx = np.arange(len(c))

    g = np.zeros(n)

        if w[ii] < 1e-10:

        shift = int(idx[ii]) - (n - 1)

            sp = -shift
            g[:n - sp] += w[ii] * (1.0 - h[sp:n])
            g[sp:] -= w[ii] * h[:n - sp]
    return g * dx

def _nesterov(h, alpha0, grow, topk, steps, step0, deadline):

    f = _smax_obj(h, alpha0, topk)
    v = np.zeros(n)
    beta, step, alpha = 0.83, step0, alpha0
    for it in range(steps):
        if time.time() > deadline:
            break
        if it > 0 and it % 400 == 0:

            beta = min(0.95, beta + 0.003)
        g = _smax_grad(h, alpha, topk)
        hp = h.copy()

        hn = _proj(h - step * v, n * 0.5)
        fn = _smax_obj(hn, alpha, topk)

            f = fn

def _slsqp(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    def obj(x):
        return _smax_obj(x, alpha, topk)
    def jac(x):
        return _smax_grad(x, alpha, topk)
    def con_fun(x):
        return x.sum() - n * 0.5
    def con_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        r = minimize(obj, h, jac=jac, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-18})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

    c5c = _eval(h)

                bh = h.copy()

def _block_anneal(h, niter, rng, deadline, T0, T1, blocks):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = blocks[rng.randint(len(blocks))]

        seg = h[i0:i0 + bs].copy()
        d = 0.006 * max(T, 1e-9) ** 0.5 * rng.randn(bs)
        d -= d.mean()

        hn = h.copy()
        hn[i0:i0 + bs] = seg

        c5n = _eval(hn)

            c5c = c5n
            if c5n < bc:
                bc = c5c
                bh = h.copy()
    return bh, bc
