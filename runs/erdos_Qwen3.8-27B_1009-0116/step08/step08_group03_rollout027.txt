def _resample(h, m):

def _sa_local_search(h0, rng, deadline, T0=5e-4, T1=1e-7):

        T = T0 * (T1 / T0) ** frac

        j = int(rng.randint(n))

            continue

        delta = (0.01 + 0.1 * T / T0) * rng.randn()
        delta = max(-0.5 * min(h[i], 1-h[i], h[j], 1-h[j]), 
                    min(0.5 * min(h[i], 1-h[i], h[j], 1-h[j]), delta))

        avail_min = min(h[i] if s > 0 else 1-h[i], h[j] if s < 0 else 1-h[j])

Actually, let me think about the gradient of corr[k] with respect to h[j]:

np.correlate(h, 1-h, 'full') has length 2n-1.
Index k in the result corresponds to: sum_i h[i] * (1-h)[i+k-(n-1)]
= sum_i h[i] * (1-h[i+k-n+1])

Let m = k - n + 1. Then:
corr[k] = sum_{i: 0<=i<n, 0<=i+m<n} h[i] * (1-h[i+m])

If m >= 0: sum_{i=0}^{n-1-m} h[i] * (1-h[i+m])
If m < 0: sum_{i=-m}^{n-1} h[i] * (1-h[i+m]) = sum_{i=0}^{n-1+m} h[i+m] * (1-h[i])

d corr[k] / d h[j]:
If m >= 0:
- Term where j = i: (1-h[j+m]) for j in [0, n-1-m]
- Term where j = i+m: -h[j-m] for j in [m, n-1]
So: grad[j] = (1-h[j+m]) (if j <= n-1-m) + (-h[j-m]) (if j >= m)

If m < 0, let p = -m > 0:
corr = sum_{i=0}^{n-1-p} h[i+p] * (1-h[i])
d/dh[j]:
- Term where j = i+p, i.e., i = j-p: (1-h[j-p]) for j in [p, n-1]
- Term where j = i: -h[j+p] for j in [0, n-1-p]
So: grad[j] = (1-h[j-p]) (if j >= p) + (-h[j+p]) (if j <= n-1-p)

This can be computed efficiently! For each active lag, compute its gradient and average.

But if there are many active lags, this could be expensive. Let's say we have at most 10 active lags. For each, the gradient computation is O(n). Total: O(10*n) per iteration. With n=560, that's O(5600) per iteration, which is very fast.

def _fft_size(n):
    m = 2*n - 1
    N = 1
    while N < m:

def _repair(h, n):
    t = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()

        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
        if abs(s - t) < 1e-12:
            break
    d = t - h.sum()
    if abs(d) > 1e-12:
        order = np.argsort(1.0-h) if d > 0 else np.argsort(h)
        for i in order:
            if d > 0:
                a = min(d, 1.0-h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a

def _proj(v, S):

def _lag_grad(h, m):
    """Gradient of corr at lag m (where m = k - (n-1)) with respect to h."""
    n = len(h)
    g = np.zeros(n)
    if m >= 0:
        # corr = sum_{i=0}^{n-1-m} h[i]*(1-h[i+m])

            g[p:]    += (1.0 - h[:n-p])       # d/dh[i+p] of h[i+p]*(1-h[i])
            g[:n-p]  += (-h[p:n])             # d/dh[i] of h[i+p]*(1-h[i])

def _smooth_value(h, alpha):

def _fft_size(n):

def _proj(v, S):

        P[idx] = pref[n - idx] - ac[idx]
        Nv[idx] = total - pref[idx] - ac[idx]

def _lag_gradient(h, m):
    n = len(h)
    g = np.zeros(n)
    if m >= 0 and m < n:
        g[:n-m] += (1.0 - h[m:n])
        g[m:] += (-h[:n-m])
    elif m < 0 and -m < n:
        p = -m
        g[p:] += (1.0 - h[:n-p])
        g[:n-p] += (-h[p:n])
    return g
