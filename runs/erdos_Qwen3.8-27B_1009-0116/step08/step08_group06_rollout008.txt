import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if h.sum() < 1e-12:
        return np.full(n, 0.5)
    if abs(h.sum() - target) > 1e-10:
        lo, hi = -3.0, 3.0
        for _ in range(80):
            mid = (lo + hi) * 0.5
            if np.clip(h - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo + hi) * 0.5, 0, 1)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _get_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_lag(h, k, n):
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        g[:n-sh] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:n-sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:n-sp])
        g[:n-sp] -= h[sp:]
    return g

def _topk_grad(h, k_top):
    n = len(h)
    c = _get_corr(h)
    idx = np.argpartition(c, -k_top)[-k_top:]
    g = np.zeros(n)
    for k in idx:
        g += _grad_lag(h, k, n)
    g /= k_top
    return g, c[idx]

def _lse_smooth(h, alpha):
    n = len(h)
    c = _get_corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

def _lse_grad(h, alpha):
    n = len(h)
    c = _get_corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-13:
            continue
        g += w[k] * _grad_lag(h, k, n)
    return g

def _slsqp_run(h, alpha, maxiter, ftol, target):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    def obj(x):
        return _lse_smooth(x, alpha)
    def jac(x):
        return _lse_grad(x, alpha)
    def cfun(x):
        return x.sum() - target
    def cjac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(obj, h, jac=jac, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _nesterov_topk(h, k_top, steps, step0, t_end, mu=0.8):
    n = len(h)
    f = _eval(h)
    v = np.zeros(n)
    step = step0
    for _s in range(steps):
        if time.time() > t_end:
            break
        g, _ = _topk_grad(h, k_top)
        h_prev = h.copy()
        v = mu * v + (1.0 - mu) * g
        h_new = _repair(h - step * v, n)
        f_new = _eval(h_new)
        if f_new < f:
            h, f = h_new, f_new
            step = min(step * 1.015, step0 * 6.0)
        else:
            step *= 0.45
            if step < 1e-11:
                v = np.zeros(n)
                step = step0 * 0.4
            h = _repair(h_prev - step * v, n)
            f = _eval(h)
    return _repair(h, n)

def _grad_polish(h, n_steps, t_end, rng):
    n = len(h)
    best_c = _eval(h)
    best_h = h.copy()
    step = 5e-4
    for _ in range(n_steps):
        if time.time() > t_end:
            break
        c = _get_corr(h)
        thresh = c.max() - 0.003
        nl = np.where(c >= thresh)[0]
        g = np.zeros(n)
        for k in nl:
            g += _grad_lag(h, k, n)
        g /= len(nl)
        accepted = False
        for _t in range(12):
            h_new = _repair(h - step * g, n)
            c_new = _eval(h_new)
            if c_new < best_c - 1e-15:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
                step = min(step * 1.08, 2e-3)
                accepted = True
                break
            step *= 0.5
        if not accepted and step < 1e-12:
            step = rng.uniform(1e-6, 5e-4)
    return best_h, best_c

def _two_pt_grad(h, n_tries, t_end, rng):
    n = len(h)
    best_c = _eval(h)
    best_h = h.copy()
    eps_list = [5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6]
    for _it in range(n_tries):
        if time.time() > t_end:
            break
        c = _get_corr(h)
        thresh = c.max() - 0.005
        nl = np.where(c >= thresh)[0]
        g = np.zeros(n)
        for k in nl:
            g += _grad_lag(h, k, n)
        g /= len(nl)
        lo_idx = np.argsort(g)[:30]
        hi_idx = np.argsort(g)[-30:]
        i = lo_idx[rng.randint(len(lo_idx))]
        j = hi_idx[rng.randint(len(hi_idx))]
        if i == j:
            continue
        for eps in eps_list:
            ni, nj = h[i] + eps, h[j] - eps
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c_new = _eval(h)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
                    break
                h[i], h[j] = ni - eps, nj + eps
    return best_h, best_c

def _random_restart_polish(h, n_rounds, t_end, rng):
    n = len(h)
    best = h.copy()
    best_c = _eval(best)
    for _r in range(n_rounds):
        if time.time() > t_end:

        d = rng.randn(n) * 0.02

        c_try = _eval(h_try)

            best = h_try.copy()
