import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_worst(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return float(c.max()), int(np.argmax(c))

def _proj_sum(h, target):
    lo, hi = -4.0, 4.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_lag(k, h, n):
    dx = 2.0 / n
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        i1 = np.arange(n - sh)
        g[i1] += (1.0 - h[i1 + sh])
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= h[i2 - sh]
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - h[i1 - sp])
        i2 = np.arange(n - sp)
        g[i2] -= h[i2 + sp]
    return g * dx

def _binary_sa(h0, n_iter, T0, T1, rng):
    n = len(h0)
    h = (h0 > 0.5).astype(float).copy()
    ones = np.where(h == 1.0)[0]
    nt = n // 2
    if len(ones) > nt:
        h[rng.choice(ones, len(ones) - nt, replace=False)] = 0.0
    elif len(ones) < nt:
        z = np.where(h == 0.0)[0]
        h[rng.choice(z, nt - len(ones), replace=False)] = 1.0
    cur = _eval(h)
    best_h, best_v = h.copy(), cur
    for it in range(n_iter):
        T = T0 * (T1 / T0) ** (it / max(n_iter - 1, 1))
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        nv = _eval(h)
        d = nv - cur
        if d < 0 or rng.rand() < np.exp(-d / max(T, 1e-15)):
            cur = nv
            if nv < best_v:
                best_v, best_h = nv, h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
    return best_h, best_v

def _nesterov_focus(h, n_steps, gamma, step0, top_k):
    n = len(h)
    dx = 2.0 / n
    t = 1.0
    h_old = h.copy()
    best_h, best_c = h.copy(), _eval(h)
    step = step0
    for it in range(n_steps):
        c = np.correlate(h, 1.0 - h, 'full') * dx
        if top_k > 0 and top_k < len(c):
            idx = np.argsort(c)[-top_k:]
        else:
            idx = np.arange(len(c))
        cs = c[idx]
        m = cs.max()
        s = np.clip(gamma * (cs - m), -400, 400)
        s -= s.max()
        w = np.exp(s); w /= w.sum()
        g = np.zeros(n)
        for ii in range(len(idx)):
            if w[ii] < 1e-15:
                continue
            g += w[ii] * _grad_lag(idx[ii], h, n)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = h + (t - 1.0) / t_new * (h - h_old)
        h_new = _proj_sum(y - step * g, n * 0.5)
        h_old = h
        h = h_new
        t = t_new
        c_new = _eval(h)
        if c_new > _eval(h_old) * 1.01:
            step *= 0.5
        elif it > 50 and it % 100 == 0:
            step *= 1.05
        if c_new < best_c:
            best_c, best_h = c_new, h_new.copy()
    return _repair(best_h, n), best_c

def _slsqp(h, alpha, maxiter, top_k):

    dx = 2.0 / n
    def obj_fn(x):
        c = np.correlate(x, 1.0 - x, 'full') * dx
        idx = np.argsort(c)[-top_k:] if 0 < top_k < len(c) else np.arange(len(c))
        cs = c[idx]; m = cs.max()
        s = np.clip(alpha * (cs - m), -400, 400); s -= s.max()
        return m + np.log(np.exp(s).sum()) / alpha
    def grad_fn(x):

            g += w[ii] * _grad_lag(idx[ii], x, n)
        return g
