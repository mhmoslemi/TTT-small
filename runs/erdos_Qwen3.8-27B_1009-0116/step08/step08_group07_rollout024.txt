                g[:n - sh] += wk * (1.0 - h[sh:])
                if sh > 0:
                    g[sh:] -= wk * h[:-sh]
            else:
                sp = -sh
                g[sp:] += wk * (1.0 - h[:-sp])
                g[:n - sp] -= wk * h[sp:]
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, target)
        fn = _eval(hn)
        if fn < f - 1e-16:
            h, f = hn, fn
            step = min(step * 1.01, step0 * 20.0)
            if f < best_f:
                best_f, best_h = f, h.copy()
        else:
            step *= 0.5
            h = _proj(h - step * v, target)
            f = _eval(h)
            if f < best_f:
                best_f, best_h = f, h.copy()
    return _repair(best_h, n), best_f

def _block_cd(h, deadline, n_blocks, blk_size):
    """Block coordinate descent: optimize blocks by local SLSQP."""
    n = len(h)
    c5c = _eval(h)
    bh = h.copy(); bc = c5c
    rng = np.random.RandomState(777)
    n_targets = min(2*n-1, 4*n)
    for _ in range(n_blocks):
        if time.time() > deadline:
            break
        # Pick a random block
        i0 = rng.randint(0, max(1, n - blk_size))
        idx = np.arange(i0, i0 + blk_size)
        h_fix = h.copy()
        try:
            def obj(x):
                hf = h_fix.copy()
                hf[idx] = x
                return _eval(hf)
            def jac(x):
                hf = h_fix.copy()
                hf[idx] = x
                c = _correl(hf)
                m = c.max()
                active = np.where(c >= m - 0.002)[0]
                g = _grad_shifts(hf, active)
                return g[idx]
            cons = []
            r = minimize(obj, h_fix[idx], jac=jac, method='SLSQP',
                         bounds=[(0,1)]*blk_size, options={'maxiter': 30, 'ftol': 1e-14})
            if r.success and obj(r.x) < c5c - 1e-15:
                h[idx] = np.clip(r.x, 0, 1)
                h = _repair(h, n)
                c5n = _eval(h)
                if c5n < bc:
                    bc = c5n; bh = h.copy()
        except Exception:
            pass
    return bh, bc

def _binary_push(h, deadline, n_iter):
    """Push h values toward 0 or 1 to encourage step-function structure."""
    n = len(h)
    c5c = _eval(h); bh = h.copy(); bc = c5c
    rng = np.random.RandomState(42)
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        target = 1.0 if h[i] > 0.45 else 0.0
        delta = target - h[i]
        if abs(delta) < 0.02:
            continue
        # Find compensating element
        if delta > 0:
            cands = np.where((h < 1.0 - abs(delta) - 1e-6) & (np.arange(n) != i))[0]
        else:
            cands = np.where((h > -delta + 1e-6) & (np.arange(n) != i))[0]
        if len(cands) == 0:
            continue
        j = cands[rng.randint(len(cands))]
        h[i] = target; h[j] -= delta
        c5n = _eval(h)
        if c5n < bc:
            bc = c5n; bh = h.copy()
        else:
            h[i] -= target - h[i]
            h[i] = target - delta
            h[j] += delta
    return _repair(bh, n), bc

def _polish(h, deadline, n_iter, scale):
    n = len(h)
    rng = np.random.RandomState(31)
    c5c = _eval(h); bh = h.copy(); bc = c5c
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 8)) % n
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < bc:
                bc = c5n; bh = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 12
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float).copy()
    n0 = len(h_init)
    best_h = _repair(h_init, n0)
    best_c5 = _eval(best_h)

    # Phase 1: Multi-scale coarse-to-fine (different from previous)
    coarse_n = max(48, n0 // 8)
    h_coarse = _downsample(best_h, coarse_n)
    dl1 = min(deadline - 500, t0 + 80)
    h_coarse, c5_coarse = _nesterov_v2(h_coarse, dl1, 800, 0.01, 0.82, adaptive=True)
    h_up = _upsample(h_coarse, n0)
    c5_up = _eval(h_up)
    if c5_up < best_c5:
        best_c5, best_h = c5_up, h_up.copy()

    # Phase 2: Nesterov adaptive at full scale (multiple restarts)
    dl2 = min(deadline - 350, t0 + 280)
    for step0, b0 in [(0.004, 0.85), (0.0015, 0.89)]:
        if time.time() > dl2:
            break
        h2, c5v = _nesterov_v2(best_h.copy(), dl2, 4000, step0, b0, adaptive=True)
        if c5v < best_c5:
            best_c5, best_h = c5v, h2.copy()

    # Phase 3: Soft-max Nesterov (smooth surrogate, different basin)
    dl3 = min(dl2 + 120, deadline - 250)
    for alpha, tk, s0 in [(40000, 8, 0.002), (150000, 5, 0.0008)]:
        if time.time() > dl3:
            break
        h3, c5v = _smax_nesterov(best_h.copy(), dl3, 2500, s0, alpha, tk)
        if c5v < best_c5:
            best_c5, best_h = c5v, h3.copy()

    # Phase 4: Binary push (encourage step-function structure)
    dl4 = min(dl3 + 100, deadline - 150)
    h4, c5v = _binary_push(best_h.copy(), dl4, 40000)
    if c5v < best_c5:
        best_c5, best_h = c5v, h4.copy()
    # Re-optimize after binary push
    if time.time() < dl4 + 80:
        h4b, c5v = _nesterov_v2(best_h.copy(), min(dl4 + 80, deadline - 80), 2000, 0.001, 0.9, True)
        if c5v < best_c5:
            best_c5, best_h = c5v, h4b.copy()

    # Phase 5: Block coordinate descent
    dl5 = min(deadline - 120, time.time() + 120)
    h5, c5v = _block_cd(best_h.copy(), dl5, n_blocks=300, blk_size=8)
    if c5v < best_c5:
        best_c5, best_h = c5v, h5.copy()

    # Phase 6: Final fine Nesterov
    dl6 = min(deadline - 60, time.time() + 120)
    h6, c5v = _nesterov_v2(best_h.copy(), dl6, 3000, 0.0005, 0.93, adaptive=True)
    if c5v < best_c5:
        best_c5, best_h = c5v, h6.copy()

    # Phase 7: Polish
    rem = max(0, deadline - time.time())
    n_pol = min(int(rem / 0.002), 80000)
    h7, c5v = _polish(best_h, deadline, n_pol, 2e-5)
    if c5v < best_c5:
        best_c5, best_h = c5v, h7.copy()

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
