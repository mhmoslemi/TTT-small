import numpy as np
import time

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _corr_vals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)
    x += (S - x.sum()) / x.size
    for _ in range(12):
        np.clip(x, 0, 1, out=x)
        d = S - x.sum()
        if abs(d) < 1e-13:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        if m.sum() == 0:
            break
        x[m] += d / m.sum()
    return x

def _repair(h, n):
    h = np.clip(h, 0, 1).copy()
    S = n * 0.5
    d = S - h.sum()
    if abs(d) < 1e-12:
        return h
    order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
    for i in order:
        if d > 0:
            a = min(d, 1.0 - h[i]); h[i] += a; d -= a
        else:
            a = min(-d, h[i]); h[i] -= a; d += a
        if abs(d) < 1e-13:
            break
    return h

def _fft_n(n):
    m = 2 * n - 1
    N = 1
    while N < m:
        N <<= 1
    return N

def _pn_arrays(h):
    n = len(h)
    Nf = _fft_n(n)
    H = np.fft.rfft(h, Nf)
    ac = np.fft.irfft(H * np.conj(H), Nf).real[:n]
    pref = np.empty(n + 1)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    total = pref[n]
    P = np.empty(n)
    P[0] = total - ac[0]
    Nv = np.zeros(n)
    if n > 1:
        idx = np.arange(1, n)
        P[idx] = pref[n - idx] - ac[idx]
        Nv[idx] = total - pref[idx] - ac[idx]
        mx = float(max(np.max(P), np.max(Nv)))
    else:
        mx = float(P[0])
    return P, Nv, mx

def _smooth_val(h, alpha):
    P, Nv, mx = _pn_arrays(h)
    n = len(h)
    dx = 2.0 / n
    e = np.exp(np.clip(alpha * (P[0] - mx), -400.0, 0.0))
    if n > 1:
        e += np.exp(np.clip(alpha * (P[1:] - mx), -400.0, 0.0)).sum()
        e += np.exp(np.clip(alpha * (Nv[1:] - mx), -400.0, 0.0)).sum()
    return dx * (mx + np.log(e) / alpha)

def _grad_weights(h, wP, wN):
    n = len(h)
    dx = 2.0 / n
    u = np.zeros(n)
    if n > 1:
        u[1:] = wP[1:] + wN[1:]
    cp = np.zeros(n)
    cn = np.zeros(n)
    if n > 1:
        cp[1:] = np.cumsum(wP[1:])
        cn[1:] = np.cumsum(wN[1:])
    linP = cp[n - 1::-1]
    linN = cn
    q = np.zeros(n)
    if n > 1 and np.any(u[1:]):
        Nf = _fft_n(n)
        H = np.fft.rfft(h, Nf)
        Urev = np.fft.rfft(u[::-1], Nf)
        Ur = np.fft.rfft(u, Nf)
        convR = np.fft.irfft(H * Urev, Nf)
        R = convR[n - 1:2 * n - 1].real
        convL = np.fft.irfft(H * Ur, Nf)
        L = convL[:n].real
        q = -(R + L)
    return dx * (wP[0] * (1.0 - 2.0 * h) + linP + linN + q)

def _smooth_grad(h, alpha):
    P, Nv, mx = _pn_arrays(h)
    n = len(h)
    wP = np.zeros(n)
    wN = np.zeros(n)
    e0 = np.exp(np.clip(alpha * (P[0] - mx), -400.0, 0.0))
    tot = e0
    if n > 1:
        ep = np.exp(np.clip(alpha * (P[1:] - mx), -400.0, 0.0))
        en = np.exp(np.clip(alpha * (Nv[1:] - mx), -400.0, 0.0))
        tot += ep.sum() + en.sum()
        if tot <= 0:
            tot = 1.0; e0 = 1.0
        wP[0] = e0 / tot
        wP[1:] = ep / tot
        wN[1:] = en / tot
    else:
        wP[0] = 1.0
    return _grad_weights(h, wP, wN)

def _fista(h0, alpha, max_time, L=10.0):
    if max_time <= 0.05:
        return h0.copy(), h0.copy(), _c5(h0)
    n = len(h0)
    S = n * 0.5
    t0 = time.time()
    h = _repair(h0, n)
    hp = h.copy()
    y = h.copy()
    t = 1.0
    fy = _smooth_val(h, alpha)
    fh = fy
    best_h = h.copy()
    best_c = _c5(h)
    stale = 0
    Lcur = L
    for it in range(500000):
        if time.time() - t0 > max_time:
            break
        if it:
            tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
            y = h + ((t - 1.0) / tn) * (h - hp)
            t = tn
            if y.min() < -0.01 or y.max() > 1.01 or abs(y.sum() - S) > 1e-7:
                y = _proj(y, S)
            fy = _smooth_val(y, alpha)
        else:
            y = h; fy = fh
        g = _smooth_grad(y, alpha)
        step = 1.0 / max(Lcur, 1e-12)
        accepted = False
        for _ in range(16):
            xn = _proj(y - step * g, S)
            fx = _smooth_val(xn, alpha)
            d = xn - y
            rhs = fy + float(np.dot(g, d)) + 0.5 * (1.0 / step) * float(np.dot(d, d))
            if fx <= rhs + 1e-10:
                accepted = True
                break
            step *= 0.5
        Lcur = 1.0 / max(step, 1e-12)
        if accepted:
            hp = h; h = xn; fh = fx; stale = 0
            Lcur = max(Lcur * 0.85, 1.0)
            if it % 8 == 0:
                c = _c5(h)
                if c < best_c:
                    best_c = c; best_h = h.copy()
        else:
            stale += 1; hp = h; y = h; t = 1.0; fy = fh
            Lcur = max(Lcur * 0.5, 1.0)
            if stale > 12:
                break
    c = _c5(h)
    if c < best_c:
        best_c = c; best_h = h.copy()
    return h, best_h, best_c

def _active_grad(h):
    P, Nv, mx = _pn_arrays(h)
    n = len(h)
    tol = max(1e-9, 6.0 * (2.0 / n))
    wP = np.zeros(n)
    wN = np.zeros(n)
    cnt = 0
    if P[0] >= mx - tol:
        wP[0] = 1.0; cnt += 1
    if n > 1:
        mP = P[1:] >= mx - tol
        if np.any(mP):
            wP[1:][mP] = 1.0; cnt += int(mP.sum())
        mN = Nv[1:] >= mx - tol
        if np.any(mN):
            wN[1:][mN] = 1.0; cnt += int(mN.sum())
    if cnt == 0:
        return np.zeros(n)
    inv = 1.0 / cnt
    wP *= inv; wN *= inv
    return _grad_weights(h, wP, wN)

def _local_search(h0, rng, deadline, start_step=3.0e-5):
    t0 = time.time()
    n = len(h0)
    S = n * 0.5
    h = h0.copy()
    c = _c5(h)
    best_c = c
    best_h = h.copy()
    P, Nv, _ = _pn_arrays(h)
    g = _active_grad(h)
    order = np.argsort(g)
    q = max(8, n // 6)
    low = order[:q]
    high = order[-q:]
    recompute = 1500
    it = 0
    while time.time() < deadline:
        if it >= recompute:
            cn = _c5(h)
            if cn < best_c:
                best_c = cn; best_h = h.copy()
            P, Nv, _ = _pn_arrays(h)
            g = _active_grad(h)
            order = np.argsort(g)
            low = order[:q]
            high = order[-q:]
            recompute = it + 1500
        frac = (time.time() - t0) / max(1e-9, deadline - t0)
        base = max(1e-12, start_step * (5e-6) ** (frac ** 1.5))
        r = rng.random()
        if r < 0.6:
            i = int(low[rng.randint(len(low))])
            j = int(high[rng.randint(len(high))])
        elif r < 0.8:
            i = int(rng.randint(0, n - 2))
            j = i + 1 + int(rng.randint(0, 3))
            if j >= n:
                j = n - 1
        else:
            i = int(rng.randint(n))
            j = int(rng.randint(n))
        if i == j:
            it += 1
            continue
        if g[i] <= g[j]:
            s = 1
        elif g[j] < g[i]:
            s = -1
        else:
            s = 1 if rng.random() < 0.5 else -1
        if s > 0:
            avail = min(1.0 - h[i], h[j])
        else:
            avail = min(h[i], 1.0 - h[j])
        if avail > 1e-14:
            mult = 0.3 + 2.0 * rng.random()
            if rng.random() < 0.02:
                mult *= 10.0
            d = s * min(avail, base * mult)
            h[i] += d; h[j] -= d
            cn = _c5(h)
            if cn < c - 1e-16:
                c = cn
                if c < best_c - 1e-16:
                    best_c = c; best_h = h.copy()
            else:
                h[i] -= d; h[j] += d
        it += 1
    c = _c5(h)
    if c < best_c:
        best_c = c; best_h = h.copy()
    return best_h, best_c

def _coord_descent(h0, deadline):
    n = len(h0)
    S = n * 0.5
    h = h0.copy()
    c = _c5(h)
    best_c = c
    best_h = h.copy()
    it = 0
    while time.time() < deadline:
        i = it % n
        old = h[i]
        for trial in (old - 0.01, old + 0.01, old - 0.005, old + 0.005, old - 0.001, old + 0.001):
            if trial < 0 or trial > 1:
                continue
            h[i] = trial
            cn = _c5(h)
            if cn < c - 1e-16:
                c = cn
                if c < best_c - 1e-16:
                    best_c = c; best_h = h.copy()
            else:
                h[i] = old
                break
        h[i] = old if h[i] != trial else h[i]
        h[i] = old
        it += 1
    return best_h, best_c

def _resample(h, m):
    n = len(h)
    if n == m:
        return _repair(h, n)
    if n % m == 0:
        out = h.reshape(m, n // m).mean(axis=1)
    else:
        x0 = (np.arange(n) + 0.5) * (2.0 / n)
        x1 = (np.arange(m) + 0.5) * (2.0 / m)
        out = np.interp(x1, x0, h)
    return _proj(out, m * 0.5)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3.0
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals():

    else:

        if c2 < best_c:
