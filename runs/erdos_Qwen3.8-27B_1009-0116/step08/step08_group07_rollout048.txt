```python
def _pmean_grad(h, a, topk):
    n = len(h)
    c = _correl(h)
    # Get top-k values
    idx = np.argpartition(c, -topk)[-topk:]
    ct = c[idx]
    M = (np.sum(ct**a) / topk) ** (1.0/a)
    # Weight for each top-k element
    w = (ct / M)**(a-1) / topk
    # Now compute gradient
    dx = 2.0/n
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-15: continue
        k = idx[ii]; sh = (n-1) - k; wk = w[ii] * dx
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - h[sh:])
            if sh > 0: g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n-sp] -= wk * h[sp:]
    return g, M
```

```python
def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _grad_fn(h, mode, **kw):

    g = np.zeros(n)

                if sh > 0: g[sh:] -= h[:-sh]

def _nesterov(h, mode, steps, step0, deadline, beta=0.85, **kw):

    target = n * 0.5

def _coord_descent(h, n_steps, rng, deadline, step_size=0.01):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    for _ in range(n_steps):
        if time.time() > deadline: break
        i, j = rng.randint(n), rng.randint(n)
        if i == j: continue
        d = step_size * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
                bh = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, c5c

def _anneal(h, niter, rng, deadline, blocks, T0, T1):
    n = len(h)
    c5c = _eval(h); bh = h.copy(); bc = c5c
    for it in range(niter):
        if time.time() > deadline: break
        T = T0 * (T1/T0)**(it/max(niter,1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n-bs))
        seg = h[i0:i0+bs].copy()
        d = 0.005 * max(T, 1e-9)**0.5 * rng.randn(bs)
        d -= d.mean(); seg += d; seg = np.clip(seg, 0, 1)

        if c5n < c5c or rng.random() < np.exp(-(c5n-c5c)/max(T, 1e-12)):
            h, c5c = hn, c5n
            if c5n < bc: bc, bh = c5n, h.copy()
    return bh, bc

        ct = c[idx]
        s = np.clip(alpha*(ct-m), -400, 400)

    def jac(x):

                     options={'maxiter':maxit, 'ftol':1e-16})

    except:
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _adj_grad(g, h, k, n, wk):
    sh = (n - 1) - k
    if sh >= 0:
        g[:n - sh] += wk * (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= wk * h[:-sh]
    else:
        sp = -sh
        g[sp:] += wk * (1.0 - h[:-sp])
        g[:n - sp] -= wk * h[sp:]

def _grad_fn(h, mode, **kw):
    n = len(h)
    c = _correl(h)
    m = c.max()
    dx = 2.0 / n
    g = np.zeros(n)
    if mode == 'exact':
        tol = kw.get('tol', 0.002)
        active = np.where(c >= m - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        for k in active:
            _adj_grad(g, h, k, n, dx / len(active))
    elif mode == 'smax':
        alpha = kw.get('alpha', 50000)
        topk = kw.get('topk', 10)
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
        s = np.clip(alpha * (ct - m), -400, 400)
        w = np.exp(s - s.max()); w /= w.sum()
        for ii in range(len(idx)):
            if w[ii] < 1e-13:
                continue
            _adj_grad(g, h, idx[ii], n, w[ii] * dx)
    elif mode == 'pmean':
        a = kw.get('a', 100.0)

            if w[ii] < 1e-15:
                continue
            _adj_grad(g, h, idx[ii], n, w[ii] * dx)
        m = M
    return g, m

def _nesterov(h, mode, steps, step0, deadline, beta=0.85, **kw):

    target = n * 0.5
    f = _eval(h)

def _slsqp_obj(x, alpha, topk):
    c = _correl(x)
    m = c.max()
    idx = np.argpartition(c, -topk)[-topk:]
    ct = c[idx]
    s = np.clip(alpha * (ct - m), -400, 400)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

    return _grad_fn(x, 'smax', alpha=alpha, topk=topk)[0]

    return x.sum() - n / 2.0

    return np.ones(n)

    if time.time() > deadline:
        return h

                     options={'maxiter': maxit, 'ftol': 1e-16})
        return _repair(r.x, n)
    except Exception:
        return h

def _coord_descent(h, n_steps, rng, deadline, step_size):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    for _ in range(n_steps):

                bh = h.copy()
```
