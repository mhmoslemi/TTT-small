import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12: break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target: lo = mid
        else: hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _smax_obj(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        idx = np.arange(len(c))
        ct = c
    s = np.clip(alpha * (ct - m), -500, 500)
    sm = s.max()
    return m + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha

def _smax_grad(h, alpha, topk):
    n = len(h)
    dx = 2.0 / n
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
        ct = c[idx]
    else:
        idx = np.arange(len(c))
        ct = c
    s = np.clip(alpha * (ct - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-15: continue
        k = int(idx[ii])
        lag = k - (n - 1)
        g[:] += w[ii]
        if lag >= 0:
            g[lag:] -= w[ii] * h[:n - lag]
            if lag > 0:
                g[:n - lag] -= w[ii] * h[lag:]
        else:
            mm = -lag
            g[:n - mm] -= w[ii] * h[mm:]
            if mm > 0:
                g[mm:] -= w[ii] * h[:n - mm]
    return g * dx

def _nesterov(h, alpha0, grow, topk0, topk1, steps, step0, deadline):
    n = len(h)
    f = _smax_obj(h, alpha0, topk0)
    v = np.zeros(n)
    step = step0
    alpha = alpha0
    for it in range(steps):
        if time.time() > deadline: break
        if it > 0 and it % 500 == 0:
            alpha = min(alpha * grow, 20000000.0)
            topk = max(5, int(topk0 * (topk1/topk0) ** (it/steps)))
        else:
            topk = max(5, int(topk0 * (topk1/topk0) ** (it/steps)))
        g = _smax_grad(h, alpha, topk)
        hp = h.copy()
        beta = 0.85
        v = beta * v + (1.0 - beta) * g
        for _ in range(10):
            hn = _proj(hp - step * v, n * 0.5)
            fn = _smax_obj(hn, alpha, topk)
            if fn < f: break
            step *= 0.5
            if step < 1e-12: break
        if step >= 1e-12:
            h = hn; f = fn
            step = min(step * 1.05, step0 * 5.0)
        else:
            break
    return _repair(h, n)

def _slsqp_block(h, alpha, topk, nblocks, deadline):
    n = len(h)
    bs = n // nblocks
    for sweep in range(20):
        if time.time() > deadline: break
        for b in range(nblocks):
            if time.time() > deadline: break
            i0, i1 = b * bs, (b + 1) * bs if b < nblocks - 1 else n
            bsz = i1 - i0
            rest_sum = h.sum() - h[i0:i1].sum()
            target_sum = n * 0.5 - rest_sum
            x0 = h[i0:i1].copy()
            def obj(x):
                hh = h.copy()
                hh[i0:i1] = x
                return _smax_obj(hh, alpha, topk)
            def jac(x):
                hh = h.copy()
                hh[i0:i1] = x
                return _smax_grad(hh, alpha, topk)[i0:i1]
            def con_fun(x):
                return x.sum() - target_sum
            def con_jac(x):
                return np.ones(bsz)
            try:

                           bounds=[(0.0, 1.0)] * bsz,
                           constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                           options={'maxiter': 200, 'ftol': 1e-14})

                    # Fix sum

                    if abs(diff) > 1e-10:
                        h[i0:i1] += diff / bsz
                        h[i0:i1] = np.clip(h[i0:i1], 0, 1)
            except Exception:
                pass
    return _repair(h, n)

def _pair_polish(h, niter, rng, deadline):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    step_scale = 1e-4
    for it in range(niter):
        if time.time() > deadline: break
        if it % 1000 == 0 and it > 0:
            step_scale *= 0.7
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, min(20, n-1))) % n
        if i == j: continue
        d = step_scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-17:
                c5c = c5n
                if c5n < bc:
                    bc = c5c; bh = h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return bh, bc

import numpy as np
from scipy.optimize import minimize
import time

_G = {}

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12: break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target: lo = mid
        else: hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _smax_obj(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        idx = np.arange(len(c))
        ct = c
    s = np.clip(alpha * (ct - m), -500, 500)
    sm = s.max()
    return m + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha

def _smax_grad_h(h, alpha, topk):
    n = len(h)
    dx = 2.0 / n
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
        ct = c[idx]
    else:
        idx = np.arange(len(c))
        ct = c
    s = np.clip(alpha * (ct - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-15: continue
        k = int(idx[ii])
        lag = k - (n - 1)
        g[:] += w[ii]
        if lag >= 0:
            g[lag:] -= w[ii] * h[:n - lag]
            if lag > 0:
                g[:n - lag] -= w[ii] * h[lag:]
        else:
            mm = -lag
            g[:n - mm] -= w[ii] * h[mm:]
            if mm > 0:
                g[mm:] -= w[ii] * h[:n - mm]
    return g * dx

def _pair_polish(h, niter, rng, deadline):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    ss = 1e-4
    for it in range(niter):

        if it % 500 == 0 and it > 0:
            ss *= 0.8
            if ss < 1e-7: break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, min(30, n-1))) % n
        if i == j: continue
