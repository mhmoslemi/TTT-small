        if time.time() - t0 > max_time: break
        if it:
            tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
            y = h + ((t - 1.0) / tn) * (h - hp)
            t = tn
            if y.min() < -1e-5 or y.max() > 1.0 + 1e-5 or abs(y.sum() - S) > 1e-8:
                y = _proj(y, S)
            fy = _smooth_value(y, alpha)
        else:
            y, fy = h, fh
        g = _smooth_grad(y, alpha)
        step = 1.0 / max(Lcur, 1e-12)
        accepted = False
        for _ in range(12):
            xn = _proj(y - step * g, S)
            fx = _smooth_value(xn, alpha)
            d = xn - y
            rhs = fy + float(np.dot(g, d)) + 0.5 * (1.0 / step) * float(np.dot(d, d))
            if fx <= rhs + 1e-10:
                accepted = True
                break
            step *= 0.5
        Lcur = 1.0 / max(step, 1e-12)
        if accepted:
            hp, h, fh = h, xn, fx
            stale = 0
            Lcur = max(Lcur * 0.85, 1.0)
            if it % 8 == 0:
                c = _c5_direct(h)
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
        else:
            stale += 1
            hp, y, t = h, h, 1.0
            fy, Lcur = fh, max(Lcur * 0.5, 1.0)
            if stale > 12: break
    c = _c5_direct(h)
    if c < best_c:
        best_c, best_h = c, h.copy()
    return h, best_h, best_c

def _targ_search(h, rng, deadline):
    n = len(h)
    h = h.copy()
    c = _c5_direct(h)
    best_c, best_h = c, h.copy()
    it = 0
    while time.time() < deadline:
        vals = _corr_all(h)
        mx = vals.max()
        tol = mx - 0.003
        active = np.where(vals >= tol)[0]
        shifted = active - (n - 1)
        improved = False
        for m in shifted[:4]:
            if m == 0:
                order = np.argsort(h - 0.5) ** 2
                hi_i = int(order[-1]); lo_i = int(order[0])
            else:
                ii = np.arange(max(0, -m), min(n, n - m))
                prod = h[ii] * (1.0 - h[ii + m])
                top = ii[np.argmax(prod)]
                if h[top] > 0.01:
                    hi_i = top
                else:
                    hi_i = top
                lo_i = int(rng.randint(n))
                for _ in range(50):
                    cand = int(rng.randint(n))
                    if m > 0 and 0 <= cand + m < n and h[cand + m] < h[hi_i + m]:
                        lo_i = cand
                        break
                    elif m < 0 and 0 <= cand + m < n and h[cand + m] < h[hi_i + m]:
                        lo_i = cand
                        break
            if hi_i == lo_i:
                continue
            if h[hi_i] < 0.02 or h[lo_i] > 0.98:
                continue
            delta = min(h[hi_i] * 0.3, (1.0 - h[lo_i]) * 0.3, 0.04)
            if delta < 1e-5:
                continue
            h[hi_i] -= delta
            h[lo_i] += delta
            cn = _c5_direct(h)
            if cn < c - 1e-14:
                c = cn
                improved = True
                if c < best_c:
                    best_c, best_h = c, h.copy()
            else:
                h[hi_i] += delta
                h[lo_i] -= delta
        if not improved and it % 200 == 0:
            for _ in range(10):
                a, b = int(rng.randint(n)), int(rng.randint(n))
                if a != b and h[a] > 0.05 and h[b] < 0.95:
                    d = min(h[a], 1.0 - h[b]) * 0.15 * (0.5 + rng.random())
                    if d > 1e-4:
                        h[a] -= d; h[b] += d
                        if _c5_direct(h) < c + 0.001:
                            c = _c5_direct(h)
                            if c < best_c:
                                best_c, best_h = c, h.copy()
                        else:
                            h[a] += d; h[b] -= d
        it += 1
    return best_h, best_c

def _local_search(h, rng, deadline, step0=3.0e-5):
    n = len(h)
    h = h.copy()
    c = _c5_direct(h)
    best_c, best_h = c, h.copy()
    P, Nv, _ = _pn_arrays(h)
    mx = max(float(P[0]), float(np.max(P)), float(np.max(Nv))) if n > 1 else float(P[0])
    tol = max(1e-9, 8.0 * 2.0 / n)
    wP = np.zeros(n); wN = np.zeros(n)
    if P[0] >= mx - tol: wP[0] = 1.0
    if n > 1:
        mP = P[1:] >= mx - tol
        mN = Nv[1:] >= mx - tol
        wP[1:][mP] = 1.0
        wN[1:][mN] = 1.0
    cnt = max(1, int(wP.sum() + wN.sum()))
    wP /= cnt; wN /= cnt
    g = _grad_from_weights(h, wP, wN)
    order = np.argsort(g)
    q = max(5, n // 7)
    low, high = order[:q], order[-q:]
    tl0 = time.time()
    recompute = it = 0
    while time.time() < deadline:
        if it >= recompute:
            c = _c5_direct(h)
            if c < best_c: best_c, best_h = c, h.copy()
            P, Nv, _ = _pn_arrays(h)
            mx2 = max(float(P[0]), float(np.max(P)), float(np.max(Nv))) if n > 1 else float(P[0])
            tol2 = max(1e-9, 8.0 * 2.0 / n)
            wP2 = np.zeros(n); wN2 = np.zeros(n)
            if P[0] >= mx2 - tol2: wP2[0] = 1.0
            if n > 1:
                mP2 = P[1:] >= mx2 - tol2
                mN2 = Nv[1:] >= mx2 - tol2
                wP2[1:][mP2] = 1.0
                wN2[1:][mN2] = 1.0
            cnt2 = max(1, int(wP2.sum() + wN2.sum()))
            wP2 /= cnt2; wN2 /= cnt2
            g = _grad_from_weights(h, wP2, wN2)
            order = np.argsort(g)
            low, high = order[:q], order[-q:]
            recompute = it + 1024
        frac = (time.time() - tl0) / max(1e-9, deadline - tl0)
        base = max(1e-10, step0 * (1e-6) ** frac)
        if rng.random() < 0.7:
            i, j = int(low[rng.randint(len(low))]), int(high[rng.randint(len(high))])
        else:
            i, j = int(rng.randint(n)), int(rng.randint(n))
        if i != j:
            s = 1 if g[i] <= g[j] else -1
            avail = min(1.0 - h[i], h[j]) if s > 0 else min(h[i], 1.0 - h[j])
            if avail > 1e-14:
                mult = 0.3 + 1.5 * rng.random()
                if rng.random() < 0.04: mult *= 10.0
                d = s * min(avail, base * mult)
                h[i] += d; h[j] -= d
                cn = _c5_direct(h)
                if cn < c - 1e-15:
                    c = cn
                    if c < best_c: best_c, best_h = c, h.copy()
                else:
                    h[i] -= d; h[j] += d
        it += 1
    return best_h, best_c

def _resample(h, m):
    n = len(h)
    if n == m: return _repair(h, n)
    if n % m == 0:
        out = h.reshape(m, n // m).mean(axis=1)
    else:
        x0 = (np.arange(n) + 0.5) * (2.0 / n)
        x1 = (np.arange(m) + 0.5) * (2.0 / m)
        out = np.interp(x1, x0, h)
    return _proj(out, m * 0.5)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(560, 0.5)
    n0 = len(h0)
    h0 = _repair(h0, n0)
    best, best_c = h0.copy(), _c5_direct(h0)

    # Phase 1: FISTA with multiple alphas
    end1 = t0 + min(120.0, 0.15 * budget_s)
    alphas = [50, 200, 800, 3000, 12000]
    per = (end1 - time.time()) / len(alphas)
    hcur = best.copy()
    for a in alphas:
        rem = end1 - time.time()
        if rem < 2: break
        hf, bh, bc = _fista(hcur, a, min(per, rem), 10.0)
        chf = _c5_direct(hf)
        if bc < best_c: best, best_c = bh.copy(), bc
        if chf < best_c: best, best_c = hf.copy(), chf
        hcur = hf if chf <= bc else bh

    # Phase 2: Targeted shift search
    end2 = t0 + 0.45 * budget_s
    if end2 > time.time():
        h0n = best if len(best) == n0 else _resample(best, n0)
        b, c = _targ_search(h0n, rng, end2)
        if c < best_c: best, best_c = b.copy(), c

    # Phase 3: Try coarser resolution then refine
    end3 = t0 + 0.65 * budget_s
    if budget_s > 400 and end3 > time.time():
        for m in (n0 // 2, n0 // 4):
            if time.time() >= end3 - 5: break
            hm = _resample(best, m)
            slot = min(25.0, end3 - time.time() - 3)
            if slot < 5: break
            hb, bcb, _ = _fista(hm, 500, slot, 10.0)
            chb = _c5_direct(hb)
            if chb < best_c or bcb < best_c:
                hu = _resample(hb if chb <= bcb else bcb, n0)
                cu = _c5_direct(hu)
                if cu < best_c: best, best_c = hu.copy(), cu

    # Phase 4: More FISTA at fine resolution with high alpha
    end4 = t0 + 0.85 * budget_s
    if end4 > time.time():
        hmain = best if len(best) == n0 else _resample(best, n0)
        for a in [20000, 80000, 300000]:
            rem = end4 - time.time()
            if rem < 3: break
            hf, bh, bc = _fista(hmain, a, rem * 0.7, 15.0)
            chf = _c5_direct(hf)
            if bc < best_c: best, best_c = bh.copy(), bc
            if chf < best_c: best, best_c = hf.copy(), chf
            hmain = hf if chf <= bc else bh

    # Phase 5: Final local search
    end5 = t0 + budget_s - 5.0
    if end5 > time.time():
        best, best_c = _local_search(best, rng, end5, 2.5e-5)

    # Final: try 1-h and reverse
    for hh in (1.0 - best, best[::-1]):
        hr = _proj(hh, len(hh) * 0.5)
        cr = _c5_direct(hr)
        if cr < best_c: best, best_c = hr.copy(), cr

    best = _repair(best, len(best))
    best_c = _c5_direct(best)
    return best, best_c, len(best)
