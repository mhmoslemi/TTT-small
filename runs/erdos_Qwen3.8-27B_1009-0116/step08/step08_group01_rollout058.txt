        if new.min() >= -1e-12 and new.max() <= 1.0 + 1e-12:

            cn = _eval(h)
            if cn < cur_c - 1e-15:
                cur_c = cn
                if cn < best_c:
                    best_c = cn
                    best_h = h.copy()
            else:
                h[i:i + bs] = old
    return best_h, best_c

def _two_pt(h, n_iter, ss, rng):
    n = len(h)
    cur_c = _eval(h)
    best_c = cur_c
    best_h = h.copy()
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 40)) % n
        d = ss * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            cn = _eval(h)
            if cn < cur_c:
                cur_c = cn
                if cn < best_c:
                    best_c = cn
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def _three_pt(h, n_iter, ss, rng):
    n = len(h)
    cur_c = _eval(h)
    best_c = cur_c
    best_h = h.copy()
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 15)) % n
        k = (i + 8 + rng.randint(0, 20)) % n
        d1, d2 = ss * rng.randn(), ss * rng.randn()
        ni, nj, nk = h[i] + d1, h[j] + d2, h[k] - d1 - d2
        if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= nk <= 1:
            s0, s1, s2 = h[i], h[j], h[k]
            h[i], h[j], h[k] = ni, nj, nk
            cn = _eval(h)
            if cn < cur_c:
                cur_c = cn
                if cn < best_c:
                    best_c = cn
                    best_h = h.copy()
            else:
                h[i], h[j], h[k] = s0, s1, s2
    return best_h, best_c

def _spectral_move(h, n_iter, rng, max_amp):
    n = len(h)
    cur_c = _eval(h)
    best_c = cur_c
    best_h = h.copy()
    for _ in range(n_iter):
        m = rng.randint(1, max(2, n // 30))
        phase = rng.rand() * 2.0 * np.pi
        amp = max_amp * rng.rand() * (1.0 if rng.rand() < 0.5 else -1.0)
        pert = amp * np.sin(2.0 * np.pi * m * np.arange(n) / n + phase)
        h_try = _repair(h + pert, n)
        cn = _eval(h_try)
        if cn < cur_c - 1e-15:
            cur_c = cn
            h = h_try.copy()
            if cn < best_c:
                best_c = cn
                best_h = h_try.copy()
    return best_h, best_c

def _slsqp(h, alpha, mxi, k):
    n = len(h)
    target = n * 0.5
    bnds = [(0.0, 1.0)] * n
    def _obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * 2.0 / n
        idx = np.argsort(c)[-k:]
        ct = c[idx]
        m = ct.max()
        s = np.clip(alpha * (ct - m), -400, 400)
        s -= s.max()
        return m + np.log(np.exp(s).sum()) / alpha
    def _con(x):
        return x.sum() - target
    def _conj(x):
        return np.ones(n)
    try:
        res = minimize(_obj, h, jac=lambda x: _sgrad(x, alpha, k),
                      method='SLSQP', bounds=bnds,
                      constraints=[{'type': 'eq', 'fun': _con, 'jac': _conj}],
                      options={'maxiter': mxi, 'ftol': 1e-15})
        return _repair(res.x, n)
    except:
        return h

def _downsample(h, f):
    n = len(h)
    m = n // f
    return h[:m * f].reshape(m, f).mean(axis=1).copy()

def _upsample(h, f):
    return np.repeat(h, f)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    def _el():
        return time.time() - t0

    n = len(initial_h_values)
    best_h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_c5 = _eval(best_h)

    # Phase 1: FISTA multi-scale
    f1_cfgs = [(400, 12, 0.012, 1500), (1500, 8, 0.008, 1200), (6000, 5, 0.005, 1000), (30000, 4, 0.003, 800), (200000, 3, 0.002, 600)]
    for alpha, k, s0, ni in f1_cfgs:
        if _el() > 300:
            break
        ht, c = _fista(best_h.copy(), alpha, k, ni, s0)
        c = _eval(ht)
        if c < best_c5:
            best_c5 = c
            best_h = ht.copy()

    # Phase 2: Multi-resolution
    if _el() < budget_s - 500:
        for ratio in [2, 4]:
            if _el() > budget_s - 450:
                break
            nc = n // ratio
            if nc < 40:
                continue
            hc = _repair(_downsample(best_h, ratio), nc)
            ht, c = _fista(hc, 3000, 6, 800, 0.03)
            h_up = _repair(_upsample(hc, ratio), n)
            # Blend: use coarse solution as guide
            h_upt = _repair(_upsample(h_up if c < best_c5 else hc, ratio), n)
            # Refine with FISTA on full
            h_ref, c_ref = _fista(h_upt, 5000, 5, 400, 0.005)
            c_ref = _eval(h_ref)
            if c_ref < best_c5:
                best_c5 = c_ref
                best_h = h_ref.copy()
            best_h, best_c5 = _two_pt(best_h, 5000, 1e-3, rng)

    # Phase 3: SLSQP
    if _el() < budget_s - 350:
        for alpha, k, mxi in [(8000, 6, 500), (40000, 4, 350), (200000, 3, 250)]:
            if _el() > budget_s - 300:
                break
            ht = _slsqp(best_h.copy(), alpha, mxi, k)
            c = _eval(ht)
            if c < best_c5:
                best_c5 = c
                best_h = ht.copy()

    # Phase 4: Spectral exploration
    if _el() < budget_s - 300:
        best_h, best_c5 = _spectral_move(best_h, 300, rng, 0.04)
        ht, c = _fista(best_h.copy(), 5000, 5, 400, 0.005)
        c = _eval(ht)
        if c < best_c5:
            best_c5 = c
            best_h = ht.copy()

    # Phase 5: Block local search at multiple scales
    if _el() < budget_s - 200:
        for bs, ss in [(3, 0.04), (5, 0.025), (8, 0.015), (12, 0.008)]:
            if _el() > budget_s - 150:
                break
            ni = min(int((budget_s - 150 - _el()) * 2000), 50000)
            best_h, best_c5 = _block_search(best_h, ni, bs, ss, rng)

    # Phase 6: Fine two-point (main refinement)
    if _el() < budget_s - 80:
        for ss in [2e-3, 5e-4, 1e-4, 3e-5, 8e-6]:
            if _el() > budget_s - 20:
                break
            ni = min(int((budget_s - 20 - _el()) * 2500), 100000)
            best_h, best_c5 = _two_pt(best_h, ni, ss, rng)

    # Phase 7: Three-point search
    if _el() < budget_s - 50:
        ni = min(int((budget_s - 50 - _el()) * 2000), 80000)
        best_h, best_c5 = _three_pt(best_h, ni, 1e-5, rng)

    # Phase 8: Final FISTA + ultra-fine
    if _el() < budget_s - 10:
        ht, c = _fista(best_h, 100000, 3, 300, 0.001)
        c = _eval(ht)
        if c < best_c5:
            best_c5 = c
            best_h = ht.copy()
        ni = min(int((budget_s - 3 - _el()) * 5000), 400000)
        best_h, best_c5 = _two_pt(best_h, ni, 1.5e-6, rng)

    # Final micro-polish
    if _el() < budget_s - 1:
        ni = min(int((budget_s - 1 - _el()) * 8000), 200000)
        best_h, best_c5 = _two_pt(best_h, ni, 5e-7, rng)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
