```python
def _repair(h, n):

    target = n / 2.0
    s = h.sum()
    if s < 1e-12:

        h = np.clip(h, 0.0, 1.0)
        s = h.sum()

                diff += a
```

```python
            contribs = h[idx] * (1.0 - h[idx + s])
            # Pick top few contributing pairs, try random ones
            top_pairs = np.argsort(contribs)[-5:]

            i = idx[p]
            j = i + s
            # Move mass from i to j (reduces h[i], increases h[j])
            # This reduces c[k_worst] by roughly dx * delta * (1 - h[j] + h[i])
            delta = rng.uniform(0.001, 0.01)
            if h[i] >= delta and h[j] + delta <= 1.0:
                h[i] -= delta
                h[j] += delta
                c_new = np.correlate(h, 1.0 - h, mode='full') * dx
                c5_new = c_new.max()
                if c5_new < c5:
                    c = c_new
                    c5 = c5_new
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h.copy()
                else:
                    h[i] += delta
                    h[j] -= delta
        elif s < 0:
            asp = -s
            idx = np.array(list(range(asp, n)))
            contribs = h[idx] * (1.0 - h[idx - asp])
            top_pairs = np.argsort(contribs)[-5:]
            p = top_pairs[rng.randint(len(top_pairs))]
            i = idx[p]
            j = i - asp
            delta = rng.uniform(0.001, 0.01)

                        best_c5 = c5
                        best_h = h.copy()
                else:
                    h[i] += delta
                    h[j] -= delta
    
    return best_h, best_c5
```

Let me try a different angle: **iterative reweighting / alternating minimization**.

Idea: 
- Fix a target value T for all lags
- For the current h, compute which lags exceed T
- Adjust h to reduce those specific lags
- Iterate

This is similar to the "hedge" algorithm in game theory:
- We have a "maximin" problem: min_h max_k c_k(h)
- By von Neumann's minimax theorem (if the problem were convex-concave), we could use:
  - Update h to minimize ∑_k p_k * c_k(h) where p_k are weights
  - Update p_k to put more weight on lags that are high

```python
def _hedge_optimize(h, n_iters, n):
    """Alternating: minimize weighted sum of c_k, update weights"""
    dx = 2.0 / n
    n_lags = 2 * n - 1
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    # Initialize weights proportional to c_k
    alpha = 500.0  # temperature
    w = np.exp(alpha * c / c.max())
    w /= w.sum()
    
    for it in range(n_iters):
        # Step 1: minimize ∑ w_k * c_k(h)
        # Gradient: ∑ w_k * ∂c_k/∂h[i]
        # ∂c_k/∂h[i] for shift s_k = k-(n-1):
        #   = dx * [(1-h[i+s_k]) if i+s_k valid] - dx * [h[i-s_k] if i-s_k valid]
        
        grad = np.zeros(n)
        for k in range(n_lags):
            if w[k] < 1e-10:
                continue
            s = k - n + 1
            wk = w[k] * dx
            if s >= 0:
                # h[i] appears as first factor: i in [0, n-1-s], contributes (1-h[i+s])
                i1 = np.arange(n - s)
                grad[i1] += wk * (1.0 - h[i1 + s])
                # h[i] appears in (1-h) factor: i in [s, n-1], contributes -h[i-s]
                if s > 0:
                    i2 = np.arange(s, n)
                    grad[i2] -= wk * h[i2 - s]
            else:
                sp = -s
                # h[i] as first factor: i in [sp, n-1], contributes (1-h[i-sp])
                i1 = np.arange(sp, n)
                grad[i1] += wk * (1.0 - h[i1 - sp])
                # h[i] in (1-h) factor: i in [0, n-1-sp], contributes -h[i+sp]
                i2 = np.arange(n - sp)
                grad[i2] -= wk * h[i2 + sp]
        
        # Projected gradient step
        # h_new = proj_{0≤h≤1, sum=h_target} (h - lr * grad)
        # But the constraint is sum(h) = n/2, so we need to project onto the simplex
        # bounded box
        
        lr = 0.001
        h_new = _proj(h - lr * grad, n / 2.0)
        
        c_new = np.correlate(h_new, 1.0 - h_new, mode='full') * dx
        if c_new.max() < c.max():
            h = h_new
            c = c_new
        
        # Step 2: update weights
        alpha *= 1.05  # decrease temperature (increase concentration)
        w = np.exp(alpha * c / c.max())
        w /= w.sum()
    
    return h, c.max()
```

```python
def _compute_grad_weighted(h, w, n):
    """Compute ∑_k w_k * ∂c_k/∂h, vectorized"""
    dx = 2.0 / n
    n_lags = 2 * n - 1
    n1 = 1.0 - h
    grad = np.zeros(n)
    
    # For each lag k with shift s = k-(n-1):
    # ∂c_k/∂h[i] = dx * [(1-h[i+s])*1[0≤i+s<n] - h[i-s]*1[0≤i-s<n]]
    
    # First term: ∑_k w_k * (1-h[i+s_k]) * 1[valid]
    # = ∑_k w_k * (1-h[i+k-n+1]) * 1[0≤i+k-n+1<n]
    # Let's compute this as: for each valid pair (i, j) where j = i+s_k for some k,
    # add w_k * (1-h[j]) to grad[i]
    # j = i + k - n + 1, so k = j - i + n - 1
    # Valid when 0≤j<n and 0≤k<2n-1, i.e., 0≤j-i+n-1<2n-1, i.e., -(n-1)≤j-i≤n-2
    # Since |j-i|≤n-1 always for i,j in [0,n), all pairs (i,j) are valid!
    # Wait: k = j-i+n-1. For k in [0, 2n-2]: 0≤j-i+n-1≤2n-2 → 1-n≤j-i≤n-1
    # Since j-i is in [1-n, n-1] for i,j in [0,n-1], all pairs are valid.
    
    # So first term: grad[i] += dx * ∑_j (1-h[j]) * w[j-i+n-1]
    # This is a convolution of (1-h) with a reversed w!
    
    # Let's define w_rev[k] = w[2n-2-k]
    # Then ∑_j (1-h[j]) * w[j-i+n-1] = [(1-h) * w_rev][i + n - 1] (convolution)
    
    # Actually, let me just use np.correlate or a direct loop for clarity.
    # For n=560, a loop of 1119 iterations with vectorized inner ops should be fast enough.
    
    for k in range(n_lags):
        wk = w[k]
        if wk < 1e-12:
            continue
        s = k - n + 1
        if s >= 0:
            # (1-h[i+s]) for i in [0, n-1-s]
            i1 = np.arange(n - s)
            grad[i1] += wk * n1[i1 + s]
            if s > 0:
                # -h[i-s] for i in [s, n-1]
                i2 = np.arange(s, n)
                grad[i2] -= wk * h[i2 - s]
        else:
            sp = -s
            i1 = np.arange(sp, n)
            grad[i1] += wk * n1[i1 - sp]
            i2 = np.arange(n - sp)
            grad[i2] -= wk * h[i2 + sp]
    
    return grad * dx
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            d = min(abs(diff), abs(1.0 - h[idx]) if diff > 0 else h[idx])
            if d < 1e-15:
                continue
            if diff > 0:
                h[idx] += d
                diff -= d
            else:
                h[idx] -= d
                diff += d
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_vector(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_weighted(h, w, n):
    dx = 2.0 / n
    n_lags = 2 * n - 1
    n1 = 1.0 - h
    grad = np.zeros(n)
    for k in range(n_lags):
        wk = w[k]
        if wk < 1e-12:
            continue
        s = k - n + 1
        if s >= 0:
            n_s = n - s
            grad[:n_s] += wk * n1[s:]
            if s > 0:
                grad[s:] -= wk * h[:n_s]
        else:
            sp = -s
            grad[sp:] += wk * n1[:n-sp]
            grad[:n-sp] -= wk * h[sp:]
    return grad * dx

def _slsqp_softmax(h, n, alpha, maxiter, top_k):
    dx = 2.0 / n
    n_lags = 2 * n - 1
    
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        if top_k > 0 and top_k < n_lags:
            idx = np.argsort(c)[-top_k:]
            cs = c[idx]
        else:
            cs = c
        m = cs.max()
        s = np.clip(alpha * (cs - m), -400, 400)
        s -= s.max()
        return m + np.log(np.exp(s).sum()) / alpha
    
    def jac(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        if top_k > 0 and top_k < n_lags:
            idx = np.argsort(c)[-top_k:]
            cs = c[idx]
        else:
            idx = np.arange(n_lags)
            cs = c
        m = cs.max()
        s = np.clip(alpha * (cs - m), -400, 400)
        s -= s.max()
        ew = np.exp(s)
        ew /= ew.sum()
        w = np.zeros(n_lags)
        w[idx] = ew
        return _grad_weighted(x, w, n)
    
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=jac, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return _repair(res.x, n)
    except:
        return h

def _hedge_descent(h, n, n_iter, t0, t_limit):
    dx = 2.0 / n
    n_lags = 2 * n - 1
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c5 = c.max()
    best_h = h.copy()
    best_c5 = c5
    
    # Initialize weights
    temp = 50.0
    w = np.exp(temp * c / max(c.max(), 1e-10))
    w /= w.sum()
    
    lr = 0.005
    for it in range(n_iter):
        if time.time() > t_limit:
            break
        
        g = _grad_weighted(h, w, n)
        # Projected step
        h_new = _proj(h - lr * g, n / 2.0)
        c_new = np.correlate(h_new, 1.0 - h_new, mode='full') * dx
        if c_new.max() < c5:
            h = h_new
            c = c_new
            c5 = c.max()
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            # Decrease learning rate slowly
            lr *= 0.9995
        else:
            # Try smaller step
            h_new2 = _proj(h - lr * 0.1 * g, n / 2.0)
            c_new2 = np.correlate(h_new2, 1.0 - h_new2, mode='full') * dx
            if c_new2.max() < c5:
                h = h_new2
                c = c_new2
                c5 = c.max()
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
            lr *= 0.9
    
        # Update weights (increase concentration on worst lags)
        temp *= 1.02
        w = np.exp(temp * c / max(c.max(), 1e-10))
        w /= w.sum()
    
    return _repair(best_h, n), best_c5

def _directed_search(h, n_iter, rng, n, step_init):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c5 = c.max()
    best_h = h.copy()
    best_c5 = c5
    step = step_init
    
    for it in range(n_iter):
        # Get worst lag
        k_worst = int(np.argmax(c))
        s = k_worst - n + 1
        
        if s >= 0:
            n_s = n - s
            if n_s <= 1:
                # Just do random move
                i = rng.randint(n)
                j = (i + 1 + rng.randint(0, 3)) % n
            else:
                # Find top contributors to this lag
                contribs = h[:n_s] * (1.0 - h[s:])
                # Pick a random one from top 20%
                thresh = np.percentile(contribs, 80)
                candidates = np.where(contribs > thresh)[0]
                if len(candidates) == 0:
                    i = rng.randint(n_s)
                else:
                    i = candidates[rng.randint(len(candidates))]
                j = i + s
        else:
            sp = -s
            if n - sp <= 1:
                i = rng.randint(n)
                j = (i + 1 + rng.randint(0, 3)) % n
            else:
                contribs = h[sp:] * (1.0 - h[:n-sp])
                thresh = np.percentile(contribs, 80)
                candidates = np.where(contribs > thresh)[0]
                if len(candidates) == 0:
                    i = sp + rng.randint(n - sp)
                else:
                    i = sp + candidates[rng.randint(len(candidates))]
                j = i - sp
        
        # Move mass from i to j (or j to i) to reduce the contribution
        d = step * rng.uniform(0.5, 1.5) * (1.0 if rng.rand() < 0.5 else -1.0)
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c_new = np.correlate(h, 1.0 - h, mode='full') * dx
            c5_new = c_new.max()
            if c5_new < c5:
                c = c_new
                c5 = c5_new
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                step *= 1.02
            else:
                h[i], h[j] = ni - d, nj + d
                step *= 0.95
    return _repair(best_h, n), best_c5

def _random_local(h, n_iter, rng, n, step):
    c = _eval(h)
    best_h = h.copy()
    best_c = c
    for _ in range(n_iter):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 5)) % n
        d = step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c_new = _eval(h)
            if c_new < c:
                c = c_new
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return _repair(best_h, n), best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    def _el():
        return time.time() - t0
    
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_c5 = _eval(h)
    best_h = h.copy()
    
    # Phase 1: SLSQP (up to 80s)
    slsqp_configs = [
        (5000, 0, 300),
        (20000, 5, 250),
        (100000, 3, 200),
        (500000, 2, 150),
    ]
    for alpha, tk, mit in slsqp_configs:
        if _el() > 80:
            break
        h_new = _slsqp_softmax(best_h, n, alpha, mit, tk)
        c5_new = _eval(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    
    # Phase 2: Hedge descent (up to 500s)
    if _el() < budget_s - 500:
        h2, c2 = _hedge_descent(best_h.copy(), n, 50000, t0, budget_s - 500)
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2.copy()
    
    # Phase 3: Directed local search (up to 900s)
    if _el() < budget_s - 200:
        # Calibrate
        _t = _el()
        for _ in range(30):
```

```python
def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            cap = 1.0 - h[idx] if diff > 0 else h[idx]
            a = min(abs(diff), cap)
            if a < 1e-15:
                continue
            if diff > 0:
                h[idx] += a; diff -= a
            else:
                h[idx] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_weighted(h, w, n):

    n1 = 1.0 - h
    grad = np.zeros(n)
    for k in range(n_lags):

        if wk < 1e-12:
            continue
        s = k - n + 1
        if s >= 0:
            ns = n - s
            grad[:ns] += wk * n1[s:]
            if s > 0:
                grad[s:] -= wk * h[:ns]
        else:
            sp = -s
            grad[sp:] += wk * n1[:n - sp]
            grad[:n - sp] -= wk * h[sp:]
    return grad * dx
```
