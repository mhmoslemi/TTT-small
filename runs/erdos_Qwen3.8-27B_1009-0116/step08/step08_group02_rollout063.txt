def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-10:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-10:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _binarize(h, n):
    target = n // 2
    idx = np.argpartition(-h, target)[:target]
    b = np.zeros(n)
    b[idx] = 1.0
    return b

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_lag(k, x, n):
    dx = 2.0 / n
    sh = (n - 1) - k
    g = np.zeros(n)
    if sh >= 0:
        i1 = np.arange(n - sh)
        g[i1] += (1.0 - x[i1 + sh])
        if sh > 0:
            i2 = np.arange(sh, n)
            g[i2] -= x[i2 - sh]
    else:
        sp = -sh
        i1 = np.arange(sp, n)
        g[i1] += (1.0 - x[i1 - sp])
        i2 = np.arange(n - sp)
        g[i2] -= x[i2 + sp]
    return g * dx

def _obj_soft(x, n, alpha, top_k):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    if top_k > 0 and top_k < len(c):
        idx = np.argsort(c)[-top_k:]
        cs = c[idx]
    else:
        cs = c
    m = cs.max()
    s = np.clip(alpha * (cs - m), -300, 300)
    s -= s.max()
    return m + np.log(np.exp(s).sum()) / alpha

def _grad_soft(x, n, alpha, top_k):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    if top_k > 0 and top_k < len(c):
        idx = np.argsort(c)[-top_k:]
        cs = c[idx]
    else:
        idx = np.arange(len(c))
        cs = c
    m = cs.max()
    s = np.clip(alpha * (cs - m), -300, 300)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-14:
            continue
        g += w[ii] * _grad_lag(idx[ii], x, n)
    return g

def _slsqp_soft(h, alpha, maxiter, top_k, n):
    bnds = [(0.0, 1.0)] * n
    def _f(x):
        return _obj_soft(x, n, alpha, top_k)
    def _j(x):
        return _grad_soft(x, n, alpha, top_k)
    def _cons_f(x):
        return x.sum() - n / 2.0
    def _cons_j(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': _cons_f, 'jac': _cons_j}]
    try:
        res = minimize(_f, h, jac=_j, method='SLSQP', bounds=bnds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return _repair(res.x, n)
    except Exception:
        return h

def _proj_grad(h, alpha, steps, step0, top_k, momentum, n):
    v = np.zeros(n)
    step = step0
    cur_c = _eval(h)
    for it in range(steps):
        c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        active = np.argsort(c)[-top_k:]
        c_a = c[active]
        m_a = c_a.max()
        s = np.clip(alpha * (c_a - m_a), -400, 400)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        g = np.zeros(n)
        for ii in range(len(active)):
            if w[ii] < 1e-14:
                continue
            g += w[ii] * _grad_lag(active[ii], h, n)
        v = momentum * v + (1.0 - momentum) * g
        taken = False
        for _t in range(12):
            h_new = _proj(h - step * v, n * 0.5)
            c_new = _eval(h_new)
            if c_new < cur_c - 1e-16:
                h = h_new
                cur_c = c_new
                step = min(step * 1.1, step0 * 12.0)
                taken = True
                break
            step *= 0.4
        if not taken:
            step = min(step * 1.4, step0 * 12.0)
        if it % 150 == 0:
            v *= 0.3
    return _repair(h, n)

        time_left = budget_s - 30 - _el()
        _t = _el()
        for _ in range(30):
            _ = _eval(best_h)
        eps = max((_el() - _t) / 30.0, 1e-8)
        ntries = min(int(time_left * (1.0/eps) * 0.7), 3000000)
        best_h, best_c5 = _local_swaptop(best_h, ntries, rng, n)

    # Phase 6: Ultra-fine 2-point polish
    if _el() < budget_s - 2:
        time_left = budget_s - 2 - _el()
        _t = _el()
        for _ in range(30):
            _ = _eval(best_h)
        eps = max((_el() - _t) / 30.0, 1e-8)
        ntries = min(int(time_left * (1.0/eps) * 0.8), 5000000)
        best_h, best_c5 = _local_2pt(best_h, ntries, rng, 3e-6, n)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
