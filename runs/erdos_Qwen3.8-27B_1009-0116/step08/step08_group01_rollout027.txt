import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _cvec(h):
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / len(h)

def _lse_grad(h, alpha, top_k):
    n = len(h)
    dx = 2.0 / n
    c = _cvec(h)
    idx = np.argsort(c)[-top_k:]
    ct = c[idx]
    m = ct.max()
    s = np.clip(alpha * (ct - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        L = k - (n - 1)
        wk = w[ii] * dx
        if L >= 0:
            if L == 0:
                g += wk * (1.0 - 2.0 * h)
            else:
                g[L:] += wk * (1.0 - h[:n - L])
                g[:n - L] -= wk * h[L:]
        else:
            M = -L
            g[M:] += wk * (1.0 - h[:n - M])
            g[:n - M] -= wk * h[M:]
    return g, w, idx

def _lse_fg(h, alpha, top_k, pen):

    target = n * 0.5
    c = _cvec(h)
    idx = np.argsort(c)[-top_k:]
    ct = c[idx]
    m = ct.max()
    s = np.clip(alpha * (ct - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    f = m + np.log(w.sum()) / alpha + pen * (h.sum() - target) ** 2
    g = np.zeros(len(h))
    for ii, k in enumerate(idx):

            continue

        if L >= 0:

            else:
                g[L:] += wk * (1.0 - h[:n - L])
                g[:n - L] -= wk * h[L:]
        else:
            M = -L
            g[M:] += wk * (1.0 - h[:n - M])
            g[:n - M] -= wk * h[M:]
    g += 2.0 * pen * (h.sum() - target)
    return f, g

def _lbfgs(h0, alpha, top_k, pen, maxiter):
    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(_lse_fg, h0, jac=True, method='L-BFGS-B',
                       bounds=bnds, args=(alpha, top_k, pen),
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10, 'maxfun': 20000})
        return _repair(res.x, n)
    except Exception:
        return h0

def _proj_target(h, target):

        else:

    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _mgd(h0, n_iter, alpha, top_k, step0, rng):

            else:

            tot = contrib.sum()

            probs = contrib / tot
