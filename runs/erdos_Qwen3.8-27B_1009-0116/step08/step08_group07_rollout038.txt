    dx = 2.0 / n
    for ii in range(len(idx)):
        if w[ii] < 1e-13:
            continue
        k = int(idx[ii]); sh = (n - 1) - k; wk = w[ii] * dx
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n-sp] -= wk * h[sp:]
    return g

def _opt_sub(h, topk, steps, step0, deadline, beta=0.85):
    n = len(h)
    f = _eval(h)
    v = np.zeros(n)
    step = step0
    bh, bf = h.copy(), f
    for _ in range(steps):
        if time.time() > deadline:
            break
        g = _subgrad(h, topk)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        v = beta * v + (1.0 - beta) * g
        hn = _repair(h - step * v / (gn + 1e-15), n)
        fn = _eval(hn)
        if fn < f - 1e-16:
            h, f = hn, fn
            step = min(step * 1.02, step0 * 12.0)
            if f < bf:
                bf, bh = f, h.copy()
        else:
            step *= 0.5
            if step < 1e-9:
                step = step0 * 0.1
    return _repair(bh, n), bf

def _opt_smooth(h, alpha, topk, steps, step0, deadline):
    n = len(h)
    f = _eval(h)
    step = step0
    bh, bf = h.copy(), f
    for _ in range(steps):
        if time.time() > deadline:
            break
        g = _sm_grad(h, alpha, topk)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        hn = _repair(h - step * g / gn, n)
        fn = _eval(hn)
        if fn < f - 1e-16:
            h, f = hn, fn
            step = min(step * 1.015, step0 * 6.0)
            if f < bf:
                bf, bh = f, h.copy()
        else:
            step *= 0.6
    return _repair(bh, n), bf

def _guided_micro(h, n_iter, rng, deadline):
    n = len(h)
    c5c = _eval(h)
    bh, bc = h.copy(), c5c
    g_cache = None
    for it in range(n_iter):
        if time.time() > deadline:
            break
        if it % 30 == 0 or g_cache is None:
            g_cache = _subgrad(h, 3)
        i = np.argmin(g_cache)
        j = np.argmax(g_cache)
        if h[i] <= 1e-10 or h[j] >= 1.0 - 1e-10:
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 5)) % n
        d = 2e-5 * rng.uniform(0.5, 2.0)
        ni, nj = h[i] - d, h[j] + d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
                if c5c < bc:
                    bc, bh = c5c, h.copy()
            else:
                h[i], h[j] = ni + d, nj - d
        if rng.random() < 0.4:
            i2 = rng.randint(n)
            j2 = (i2 + 1 + rng.randint(0, 4)) % n
            d2 = 8e-6 * rng.randn()
            ni2, nj2 = h[i2] + d2, h[j2] - d2
            if 0.0 <= ni2 <= 1.0 and 0.0 <= nj2 <= 1.0:
                h[i2], h[j2] = ni2, nj2
                c5n2 = _eval(h)
                if c5n2 < c5c:
                    c5c = c5n2
                    if c5c < bc:
                        bc, bh = c5c, h.copy()
                else:
                    h[i2], h[j2] = ni2 - d2, nj2 + d2
    return bh, bc

def _random_micro(h, n_iter, rng, deadline, scales):
    n = len(h)
    c5c = _eval(h)
    bh, bc = h.copy(), c5c
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 6)) % n
        d = scales[rng.randint(len(scales))] * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
                if c5c < bc:
                    bc, bh = c5c, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc

def _jitter_reopt(h, rng, deadline, topk, step0):
    n = len(h)
    c5c = _eval(h)
    for amp in [0.003, 0.0015, 0.0008]:
        if time.time() > deadline:
            break
        hn = _repair(h.copy() + amp * rng.randn(n), n)
        hn, cv = _opt_sub(hn, topk, 1500, step0, deadline, 0.85)
        if cv < c5c:
            c5c = cv
            h = hn.copy()
    return h, c5c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 6
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h, best_c5 = h.copy(), _eval(h)

    # Phase 1: Subgradient top-k (170s)
    dl1 = min(deadline, t0 + 170)
    for topk, s0, beta in [(5, 0.004, 0.85), (8, 0.003, 0.87), (12, 0.002, 0.88)]:
        if time.time() > dl1:
            break
        h2, cv = _opt_sub(best_h.copy(), topk, 5000, s0, dl1, beta)
        if cv < best_c5:
            best_c5, best_h = cv, h2.copy()

    # Phase 2: Smooth gradient (170s)
    dl2 = min(deadline, t0 + 340)
    for alpha, topk, s0 in [(80000, 5, 0.003), (300000, 3, 0.002), (1000000, 2, 0.001)]:
        if time.time() > dl2:
            break
        h2, cv = _opt_smooth(best_h.copy(), alpha, topk, 5000, s0, dl2)
        if cv < best_c5:
            best_c5, best_h = cv, h2.copy()

    # Phase 3: Jitter restarts (140s)
    dl3 = min(deadline, t0 + 480)
    h2, cv = _jitter_reopt(best_h.copy(), rng, dl3, 5, 0.002)
    if cv < best_c5:
        best_c5, best_h = cv, h2.copy()

    # Phase 4: Fine subgradient (130s)
    dl4 = min(deadline, t0 + 610)
    for topk, s0 in [(3, 0.0004), (2, 0.0002), (5, 0.0003)]:
        if time.time() > dl4:
            break
        h2, cv = _opt_sub(best_h.copy(), topk, 8000, s0, dl4, 0.92)
        if cv < best_c5:
            best_c5, best_h = cv, h2.copy()

    # Phase 5: Very high alpha smooth (100s)
    dl5 = min(deadline, t0 + 710)
    h2, cv = _opt_smooth(best_h.copy(), 2000000, 2, 8000, 0.0004, dl5)
    if cv < best_c5:
        best_c5, best_h = cv, h2.copy()

    # Phase 6: Guided micro-polish (150s)
    dl6 = min(deadline, t0 + 860)
    best_h, best_c5 = _guided_micro(best_h.copy(), 300000, rng, dl6)

    # Phase 7: Final random micro-polish (remaining)
    dl7 = deadline
    scales = np.array([3e-5, 1e-5, 5e-6, 2e-6, 1e-6])
    best_h, best_c5 = _random_micro(best_h.copy(), 500000, rng, dl7, scales)

    best_h = _repair(best_h, n0)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n0)
