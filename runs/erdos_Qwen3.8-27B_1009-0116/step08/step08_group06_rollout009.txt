from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if h.sum() < 1e-12:
        return np.full(n, 0.5)
    if abs(h.sum() - target) > 1e-10:
        lo, hi = -3.0, 3.0
        for _ in range(80):
            mid = (lo + hi) * 0.5
            if np.clip(h - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo + hi) * 0.5, 0, 1)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_full(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _grad_at_shift(h, sh, n):
    """Gradient of correlation at one shift."""
    g = np.zeros(n)
    if sh >= 0:
        g[:n-sh] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:n-sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:n-sp])
        g[:n-sp] -= h[sp:]
    return g

def _topk_objective_and_grad(h, k_top, n):
    c = _corr_full(h)
    idx = np.argsort(c)[-k_top:]
    g = np.zeros(n)
    for k in idx:
        sh = (n - 1) - k
        g += _grad_at_shift(h, sh, n)
    g /= k_top
    return c[idx].max(), g

def _slsqp_lse(h, alpha, maxiter, n, target):
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        return m + np.log(np.sum(np.exp(s - s.max()))) / alpha
    
    def grad(x):
        c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
        m = c.max()
        s = np.clip(alpha * (c - m), -500, 500)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-15:
                continue
            sh = (n - 1) - k
            wk = w[k] * (2.0 / n)
            if sh >= 0:
                g[:n-sh] += wk * (1.0 - x[sh:])
                if sh > 0:
                    g[sh:] -= wk * x[:n-sh]
            else:
                sp = -sh
                g[sp:] += wk * (1.0 - x[:n-sp])
                g[:n-sp] -= wk * x[sp:]
        return g
    
    bnds = [(0.0, 1.0)] * n
    res = minimize(obj, h, jac=grad, method='SLSQP',
                   bounds=bnds,
                   constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - target,
                                 'jac': lambda x: np.ones(n)}],
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return _repair(res.x, n)

def _subgrad_descent(h, n, n_iter, lr_init, t_end):
    h = h.copy()
    lr = lr_init
    best = _eval(h)
    best_h = h.copy()
    for it in range(n_iter):
        if time.time() > t_end:
            break
        c = _corr_full(h)
        m = c.max()
        active = np.where(c >= m - 1e-8)[0]
        g = np.zeros(n)

            g += (2.0 / (n * len(active))) * _grad_at_shift(h, sh, n)

def _sim_anneal(h, n, n_iter, t_start, t_end_anneal, t_deadline, rng):
    """Simulated annealing with pair perturbations."""
    h = h.copy()
    best = _eval(h)
    best_h = h.copy()
    t_anneal = t_start
    stall = 0
    max_stall = 200
    for it in range(n_iter):
        if time.time() > t_deadline:

            j = rng.randint(n)

        f_new = _eval(perturbed)
        delta = f_new - _eval(h)
        if delta < 0 or rng.rand() < np.exp(-max(delta, -0.1) / max(t_anneal, 1e-10)):

            if f_new < best:

                best_h = h.copy()
                stall = 0
            else:

        else:
            stall += 1
        if stall > max_stall and it > n_iter // 10:
            t_anneal *= 0.95
            stall = 0
        t_anneal = max(t_anneal * 0.999, 1e-6)
    return best_h, best

def _local_search(h, n, n_tries, rng, t_end, scale=1e-5):
    n = len(h)
    c5c = _eval(h)
    best_h, best_c = h.copy(), c5c

        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 4)) % n

            if c5n < c5c:

                if c5n < best_c:
                    best_c, best_h = c5n, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(np.float64)
    target = n / 2.0
    if h.sum() < 1e-12:
        return np.full(n, 0.5)
    if abs(h.sum() - target) > 1e-10:
        lo, hi = -3.0, 3.0
        for _ in range(80):
            mid = (lo + hi) * 0.5
            if np.clip(h - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo + hi) * 0.5, 0, 1)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr_full(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _grad_shift(h, sh, n):
    g = np.zeros(n)
    if sh >= 0:
        g[:n - sh] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:n - sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:n - sp])
        g[:n - sp] -= h[sp:]
    return g

class _ObjCtx:
    def __init__(self, alpha, n, target):
        self.alpha = alpha
        self.n = n
        self.target = target
    def obj(self, x):
        c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / self.n)
        m = c.max()
        s = np.clip(self.alpha * (c - m), -500, 500)
        return m + np.log(np.sum(np.exp(s - s.max()))) / self.alpha
    def grad(self, x):
        c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / self.n)
        m = c.max()
        s = np.clip(self.alpha * (c - m), -500, 500)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(self.n)
        for k in range(2 * self.n - 1):
            if w[k] < 1e-15:
                continue
            sh = (self.n - 1) - k
            wk = w[k] * (2.0 / self.n)
            g += wk * _grad_shift(x, sh, self.n)
        return g
    def con_fun(self, x):
        return x.sum() - self.target
    def con_jac(self, x):
        return np.ones(self.n)

def _slsqp_run(h, alpha, maxiter, n, target):
    ctx = _ObjCtx(alpha, n, target)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': ctx.con_fun, 'jac': ctx.con_jac}]
    try:
        res = minimize(ctx.obj, h, jac=ctx.grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _subgrad_run(h, n, n_iter, lr_init, k_top, t_end):
    h = h.copy()
    lr = lr_init
    best = _eval(h)
    best_h = h.copy()
    for _ in range(n_iter):
        if time.time() > t_end:
            break
        c = _corr_full(h)
        m = c.max()
        idx = np.argsort(c)[-k_top:]
        g = np.zeros(n)
        for k in idx:
            sh = (n - 1) - k
            g += _grad_shift(h, sh, n)
        g *= (2.0 / n) / k_top
        h_new = _repair(h - lr * g, n)
        f_new = _eval(h_new)
        if f_new < best:
            best = f_new
            best_h = h_new.copy()
        h = h_new
        if f_new < best + 1e-12:
            lr = min(lr * 1.005, lr_init * 5.0)
        else:
            lr *= 0.75
            if lr < 1e-15:
                lr = lr_init * 0.05
    return best_h, best

def _anneal_run(h, n, t_end, rng):

    best = _eval(h)
    best_h = h.copy()
    temp = 0.05
    decay = 0.9995
    stall = 0
    it = 0
    while time.time() < t_end:
        it += 1
        scale = min(temp, 0.3)
        npair = 1 if scale > 0.02 else (2 if scale > 0.005 else 3)
        trial = h.copy()
        for _ in range(npair):

            if i == j:

            d = scale * 0.5 * rng.randn()
            d = max(min(d, 0.49), -0.49)
            trial[i] += d
            trial[j] -= d
        trial = _repair(trial, n)
        f_new = _eval(trial)
        cur = _eval(h)
        delta = f_new - cur
        accept = delta < 0 or rng.rand() < np.exp(-max(delta, -0.5) / max(temp, 1e-12))
        if accept:
            h = trial
            if f_new < best:

                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        if stall > 500:
            temp *= 0.9
            stall = 0
        temp = max(temp * decay, 0.0001)
    return best_h, best

def _local_2pt(h, n, n_tries, rng, t_end, scale):
    c5c = _eval(h)
    best_h, best_c = h.copy(), c5c
    for _ in range(n_tries):

        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 5)) % n
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
                if c5n < best_c:
                    best_c, best_h = c5n, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def _local_3pt(h, n, n_tries, rng, t_end, scale):
    c5c = _eval(h)
    best_h, best_c = h.copy(), c5c
    for _ in range(n_tries):

            else:

def _multiscale_local(h, n, t_end, rng):
    scales = [5e-4, 1e-4, 3e-5, 1e-5, 3e-6, 1e-6]
