import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    for _ in range(100):
        s = h.sum()
        if s < 1e-15:
            return np.full(n, 0.5)
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        if abs(h.sum() - target) < 1e-13:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            a = min(abs(diff), (1.0 - h[idx]) if diff > 0 else h[idx])
            h[idx] += a if diff > 0 else -a
            diff -= a if diff > 0 else -a
            if abs(diff) < 1e-14:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_worst(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return float(c.max()), int(c.argmax())

def _proj(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_shift(h, s, n):
    dx = 2.0 / n
    g = np.zeros(n)
    if 0 < s < n:
        g[:n - s] += (1.0 - h[s:])
        g[s:] -= h[:n - s]
    elif -n < s < 0:
        sp = -s
        g[sp:] += (1.0 - h[:n - sp])
        g[:n - sp] -= h[sp:]
    return g * dx

def _combine_grad(h, n, top_k, alpha):
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    if top_k > 0 and top_k < len(c):
        idx = np.argsort(c)[-top_k:]
    else:
        idx = np.arange(len(c))
    ci = c[idx]
    m = ci.max()
    s = np.clip(alpha * (ci - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] > 1e-15:
            g += w[ii] * _grad_shift(h, (n - 1) - int(idx[ii]), n)
    return g, c, idx

def _slsqp_focus(h, alpha, maxiter, top_k, tlimit):
    n = len(h)
    dx = 2.0 / n
    t0 = time.time()

    def _obj(x):
        if time.time() - t0 > tlimit:
            return float(_eval(x))
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        if 0 < top_k < len(c):
            ci = c[np.argsort(c)[-top_k:]]
        else:
            ci = c
        m = ci.max()
        s = np.clip(alpha * (ci - m), -400, 400)
        s -= s.max()
        return float(m + np.log(np.exp(s).sum()) / alpha)

    def _grd(x):
        if time.time() - t0 > tlimit:
            return np.zeros(n)
        g, _, _ = _combine_grad(x, n, top_k, alpha)
        return g

    try:
        res = minimize(_obj, h, jac=_grd, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
                                     'jac': lambda x: np.ones(n)}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        if res.x is not None:
            rx = _repair(res.x, n)
            if _eval(rx) < _eval(h):
                return rx
    except:
        pass
    return h

def _pg_focus(h, alpha, n_steps, step0, top_k, mu, tlimit):
    n = len(h)
    t0 = time.time()
    v = np.zeros(n)
    step = step0

        v = mu * v + (1.0 - mu) * g
        taken = False

            cn = _eval(h_new)
            if cn < cur - 1e-17:
                h = h_new
                cur = cn
                step = min(step * 1.05, step0 * 50.0)
                taken = True
                break
            step *= 0.4
        if not taken:
            step = min(step * 1.3, step0 * 50.0)
        if it % 200 == 0:
            v *= 0.2
    return h, cur

            step *= 0.98

def _downsample(h, m):
    n = len(h)
    f = n // m
    return h[:m * f].reshape(m, f).mean(axis=1).copy()

def _upsample(h, n):
    m = len(h)
    f = n // m
    out = np.empty(n)
    for i in range(m):
        out[i * f:(i + 1) * f] = h[i]
    return out

def _polish(h, t0, tlimit, rng):
    n = len(h)
    cur = _eval(h)
    best_h = h.copy()
    best_c = cur
    step = 1e-5
    for _ in range(50000000):
        if time.time() - t0 > tlimit:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 3)) % n
        d = step * rng.randn()
        ni, nj = h[i] + d, h[j] - d

                    best_c = cn
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
                step *= 0.999
    return best_h, best_c
