    return g

def _grad_ps(x, p, dx2, n):
    c = np.correlate(x, 1.0 - x, mode='full') * dx2
    cs = np.maximum(c, 1e-30)
    S = np.sum(cs ** p)
    if S < 1e-25:
        return np.zeros(n)
    w = (cs ** (p - 1)) * dx2 * (S ** (1.0 / p - 1.0))
    return _grad_corr_w(x, w, n)

def _grad_topk(x, k_top, dx2, n):
    c = np.correlate(x, 1.0 - x, mode='full') * dx2
    idx = np.argpartition(c, -k_top)[-k_top:]
    w = np.zeros(2 * n - 1)
    w[idx] = dx2 / k_top
    return _grad_corr_w(x, w, n)

def _nesterov_ps(h, p_lo, p_hi, steps, step0, t_end):
    n = len(h)
    dx2 = 2.0 / n
    f = _eval(h)
    v = np.zeros(n)
    mu = 0.82
    step = step0
    for s in range(steps):
        if time.time() > t_end:
            break
        p = p_lo + (p_hi - p_lo) * s / max(steps, 1)
        g = _grad_ps(h, p, dx2, n)
        h_prev = h.copy()
        v = mu * v + (1.0 - mu) * g
        h_new = _repair(h - step * v, n)
        f_new = _eval(h_new)
        if f_new < f:
            h, f = h_new, f_new
            step = min(step * 1.03, step0 * 5.0)
        else:
            step *= 0.45
            if step < 1e-12:
                v = np.zeros(n)
                step = step0 * 0.3
            h = _repair(h_prev - step * v, n)
            f = _eval(h)
    return _repair(h, n)

def _slsqp_lse(h, alpha, maxiter, ftol, target):
    n = len(h)
    dx2 = 2.0 / n
    def obj_fn(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx2
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        return m + np.log(np.sum(np.exp(s - s.max()))) / alpha
    def grad_fn(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx2
        m = c.max()
        s = np.clip(alpha * (c - m), -400, 400)
        w = np.exp(s - s.max())
        w /= w.sum()
        return _grad_corr_w(x, w * dx2, n)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq',
             'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj_fn, h, jac=grad_fn, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _nesterov_topk(h, k, steps, step0, t_end):
    n = len(h)
    dx2 = 2.0 / n
    f = _eval(h)
    v = np.zeros(n)
    mu = 0.8
    step = step0
    for s in range(steps):
        if time.time() > t_end:
            break
        g = _grad_topk(h, k, dx2, n)
        h_prev = h.copy()
        v = mu * v + (1.0 - mu) * g
        h_new = _repair(h - step * v, n)
        f_new = _eval(h_new)
        if f_new < f:
            h, f = h_new, f_new
            step = min(step * 1.02, step0 * 8.0)
        else:
            step *= 0.4
            if step < 1e-11:
                v = np.zeros(n)
                step = step0 * 0.4
            h = _repair(h_prev - step * v, n)
            f = _eval(h)
    return _repair(h, n)

def _block_search(h, n_iters, rng, t_end):
    n = len(h)
    f = _eval(h)
    for _ in range(n_iters):
        if time.time() > t_end:
            break
        blen = rng.randint(2, 8)
        start = rng.randint(0, n - blen)
        idx = start + np.arange(blen)
        for _t in range(8):
            scale = 0.002 + 0.004 * rng.rand()
            d = scale * rng.randn(blen)
            d = d - d.mean()
            old = h[idx].copy()
            h[idx] = np.clip(old + d, 0, 1)
            h = _repair(h, n)
            f_new = _eval(h)
            if f_new < f:
                f = f_new
            else:
                h[idx] = old
                h = _repair(h, n)
    return _repair(h, n)

def _sa_search(h, n_moves, rng, t_end, T0, T1):
    n = len(h)
    f = _eval(h)
    best_h, best_f = h.copy(), f
    for m in range(n_moves):
        if time.time() > t_end:
            break
        frac = min(1.0, m / float(n_moves))
        T = T0 * (T1 / T0) ** frac
        np_ = rng.randint(2, 6)
        idx = rng.choice(n, np_, replace=False)
        scale = 0.0005 + 0.004 * (1.0 - frac)
        d = scale * rng.randn(np_)
        d = d - d.mean()
        old = h[idx].copy()
        h[idx] = np.clip(old + d, 0, 1)
        h = _repair(h, n)
        f_new = _eval(h)
        if f_new < f or rng.rand() < np.exp(-(f_new - f) / T):
            f = f_new
            if f < best_f:
                best_f = f
                best_h = h.copy()
        else:
            h[idx] = old
            h = _repair(h, n)
    return best_h, best_f

def _fine_2pt(h, n_moves, rng, t_end, scale):
    n = len(h)
    f = _eval(h)
    best_h, best_f = h.copy(), f
    for _ in range(n_moves):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = (i + rng.randint(1, 10)) % n
        d = scale * rng.randn()
        oi, oj = h[i], h[j]
        if 0 <= oi + d <= 1 and 0 <= oj - d <= 1:
            h[i], h[j] = oi + d, oj - d
            h = _repair(h, n)
            f_new = _eval(h)
            if f_new < f:
                f = f_new
                if f < best_f:
                    best_f = f
                    best_h = h.copy()
            else:
                h[i], h[j] = oi, oj
                h = _repair(h, n)
    return best_h, best_f

def _neighbor_rotate(h, n_moves, rng, t_end):
    n = len(h)
    f = _eval(h)
    best_h, best_f = h.copy(), f
    for _ in range(n_moves):
        if time.time() > t_end:
            break
        i = rng.randint(n - 2)
        a, b, c = h[i], h[i+1], h[i+2]
        d = 0.001 * rng.randn()
        h[i], h[i+1], h[i+2] = a + d, b - d, c
        if 0 <= h[i] <= 1 and 0 <= h[i+1] <= 1:
            h = _repair(h, n)
            f_new = _eval(h)
            if f_new < f:
                f = f_new
                if f < best_f:
                    best_f = f
                    best_h = h.copy()
            else:
                h[i], h[i+1], h[i+2] = a, b, c
                h = _repair(h, n)
    return best_h, best_f

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    t_end_all = t0 + budget_s
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    target = n / 2.0
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Power-sum gradient (broad basin, 12%)
    t1 = t0 + budget_s * 0.12
    hh = _nesterov_ps(best_h.copy(), 5, 40, 4000, 0.006, t1)
    cv = _eval(hh)
    if cv < best_c5:
        best_c5, best_h = cv, hh

    # Phase 2: SLSQP moderate (13%)
    t2 = t1 + budget_s * 0.13
    for alpha in [1500, 15000]:
        if time.time() > t2 - 8:
            break
        hh = _slsqp_lse(best_h.copy(), alpha, 800, 1e-15, target)
        cv = _eval(hh)
        if cv < best_c5:
            best_c5, best_h = cv, hh

    # Phase 3: Nesterov top-k adaptive (18%)
    t3 = t2 + budget_s * 0.18
    for k in [5, 15, 40]:
        tsub = t3 - (t3 - t2) * (40 - k) / 40.0
        tsub = min(tsub, t3 - 3)
        hh = _nesterov_topk(best_h.copy(), k, 4000, 0.003, tsub)
        cv = _eval(hh)
        if cv < best_c5:
            best_c5, best_h = cv, hh

    # Phase 4: SLSQP high alpha (10%)
    t4 = t3 + budget_s * 0.10
    for alpha in [100000, 1000000]:
        if time.time() > t4 - 10:
            break
        hh = _slsqp_lse(best_h.copy(), alpha, 500, 1e-16, target)
        cv = _eval(hh)
        if cv < best_c5:
            best_c5, best_h = cv, hh

    # Phase 5: Block search (15%)
    t5 = t4 + budget_s * 0.15
    if time.time() < t5 - 5:
        nm = min(int((t5 - time.time()) / 0.004), 4000)
        hh = _block_search(best_h.copy(), nm, rng, t5)
        cv = _eval(hh)
        if cv < best_c5:
            best_c5, best_h = cv, hh

    # Phase 6: Simulated annealing (18%)
    t6 = t5 + budget_s * 0.18
    if time.time() < t6 - 5:
        nm = min(int((t6 - time.time()) / 0.003), 35000)
        best_h, best_c5 = _sa_search(best_h, nm, rng, t6, 0.004, 1e-8)

    # Phase 7: Neighbor rotation (10%)
    t7 = t6 + budget_s * 0.10
    if time.time() < t7 - 3:
        nm = min(int((t7 - time.time()) / 0.002), 30000)
        best_h, best_c5 = _neighbor_rotate(best_h, nm, rng, t7)

    # Phase 8: Fine 2-point (7%)
    t8 = t7 + budget_s * 0.07
    if time.time() < t8 - 2:
        nm = min(int((t8 - time.time()) / 0.0015), 40000)
        best_h, best_c5 = _fine_2pt(best_h, nm, rng, t8, 3e-6)

    # Phase 9: Ultra-fine (5%)
    t9 = t8 + budget_s * 0.05
    if time.time() < t9 - 1:
        nm = min(int((t9 - time.time()) / 0.0015), 30000)
        best_h, best_c5 = _fine_2pt(best_h, nm, rng, t9, 1e-7)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
