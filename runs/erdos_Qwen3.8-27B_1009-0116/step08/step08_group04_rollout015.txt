def _grad_smax(h, alpha, topk):
    n = len(h)
    dx = 2.0 / n
    c = _correl(h)
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        idx = np.arange(len(c))
        ct = c
    s = np.clip(alpha * (ct - ct.max()), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-15:
            continue
        k_full = idx[ii]
        lag = k_full - (n - 1)
        if lag >= 0:
            l = lag
            g[:n-l] += w[ii] * dx * (1.0 - h[l:])
            g[l:] += w[ii] * dx * (-h[:n-l])
        else:
            l = -lag
            g[l:] += w[ii] * dx * (1.0 - h[:n-l])
            g[:n-l] += w[ii] * dx * (-h[l:])
    return g

            lr *= lr_decay
            alpha = min(alpha, 1e8)
            lr = max(lr, 1e-7)
        
        g = _grad_smax(h, alpha, topk)
        # Adam update

import numpy as np
from scipy.optimize import minimize
import time

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval(h):
    return float(np.max(_correl(h)))

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_smax(h, alpha, topk):
    n = len(h)
    dx = 2.0 / n
    c = _correl(h)
    m_c = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
    else:
        idx = np.arange(len(c))
    ct = c[idx]
    s = np.clip(alpha * (ct - m_c), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-15:
            continue
        lag = idx[ii] - (n - 1)
        l = abs(lag)
        if lag >= 0:
            g[:n - l] += w[ii] * dx * (1.0 - h[l:])
            g[l:] += w[ii] * dx * (-h[:n - l])
        else:
            g[l:] += w[ii] * dx * (1.0 - h[:n - l])
            g[:n - l] += w[ii] * dx * (-h[l:])
    return g

def _adam_opt(h, deadline, alpha0, a_grow, topk, lr0, lr_dec):
    n = len(h)
    target = n * 0.5
    m1 = np.zeros(n)
    m2 = np.zeros(n)
    t = 0
    best_h = h.copy()
    best_c = _eval(h)
    lr = lr0
    alpha = alpha0
    b1, b2 = 0.85, 0.97
    while time.time() < deadline:
        t += 1
        if t % 150 == 0:
            alpha = min(alpha * a_grow, 5e7)
            lr *= lr_dec
        g = _grad_smax(h, alpha, topk)
        m1 = b1 * m1 + (1 - b1) * g
        m2 = b2 * m2 + (1 - b2) * g * g
        m1h = m1 / (1 - b1 ** t)
        m2h = m2 / (1 - b2 ** t)
        h_new = _proj(h - lr * m1h / (np.sqrt(m2h) + 1e-8), target)
        c_new = _eval(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        if c_new <= _eval(h) + 1e-14:
            h = h_new
    return best_h, best_c

def _slsqp_opt(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    cons_v = n * 0.5
    def obj(x):
        c = _correl(x)
        m = c.max()
        ct = c if topk >= len(c) else c[np.argpartition(c, -topk)[-topk:]]
        s = np.clip(alpha * (ct - m), -400, 400)
        return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha
    def jac(x):
        return _grad_smax(x, alpha, topk)
    try:
        r = minimize(obj, h, jac=jac, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - cons_v,
                                   'jac': lambda x: np.ones(n)}],
                     options={'maxiter': maxit, 'ftol': 1e-16})
        return _repair(r.x, n)
    except Exception:
        return _repair(h, n)

def _targeted_anneal(h, niter, rng, deadline, T0, T1):
    n = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        c = _correl(h)
        worst = np.argmax(c)
        lag = worst - (n - 1)
        l = abs(lag)
        if lag >= 0:
            contrib = h[:n - l] * (1.0 - h[l:])
        else:
            contrib = h[l:] * (1.0 - h[:n - l])
        i_hi = np.argmax(contrib)
        if lag >= 0:
            pos_a = i_hi
            pos_b = i_hi + l
        else:
            pos_b = i_hi
            pos_a = i_hi + l
        d = 0.003 * max(T, 1e-10) ** 0.5 * rng.randn()
        ni, nj = h[pos_a] + d, h[pos_b] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[pos_a], h[pos_b] = ni, nj
            c5n = _eval(h)
            if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
                c5c = c5n
                if c5n < bc:
                    bc = c5n
                    bh = h.copy()
            else:
                h[pos_a], h[pos_b] = ni - d, nj + d
    return bh, bc

def _fine_polish(h, niter, rng, deadline):
    nc = len(h)
    c5c = _eval(h)
    bh = h.copy()
    bc = c5c
    step = 5e-6
    for _ in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(nc)
        j = (i + 1 + rng.randint(0, 12)) % nc
        d = step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < bc:
                    bc = c5n
                    bh = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
                step = max(step * 0.97, 1e-8)
    return bh, bc
