def exact_subgradient(h, alpha=None):
    """Compute gradient of smooth approximation (or exact subgradient if alpha is very large)."""
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full') * dx  # length 2n-1
    mx = np.max(corr)
    
    if alpha is not None:
        shifted = np.clip(alpha * (corr - mx), -400, 0)
        weights = np.exp(shifted)
        weights /= weights.sum()
    else:
        # Exact subgradient: uniform over active set
        tol = 1e-10
        active = corr >= mx - tol
        weights = np.zeros(2*n-1)
        weights[active] = 1.0 / active.sum()

    # Lag k corresponds to index n-1+k
    # weights[n-1+k] = w_k

    return grad

import numpy as np
import time

def _fft_size(n):
    N = 1
    while N < 4 * n:
        N <<= 1
    return N

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)
    x += (S - x.sum()) / x.size
    for _ in range(12):
        np.clip(x, 0.0, 1.0, out=x)
        d = S - x.sum()
        if abs(d) < 1e-12:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
    if abs(x.sum() - S) > 1e-9:
        lo, hi = float(np.min(v)) - 1.0, float(np.max(v)) + 1.0
        for _ in range(50):
            mid = 0.5 * (lo + hi)
            if np.clip(v - mid, 0.0, 1.0).sum() > S:
                lo = mid
            else:
                hi = mid
        x = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    return x

def _repair(h, n):
    t = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(50):
        s = h.sum()
        if abs(s - t) < 1e-13:
            break
        if s <= 0:
            return np.full(n, 0.5)
        h *= t / s
        np.clip(h, 0.0, 1.0, out=h)
    d = t - h.sum()
    if abs(d) > 1e-12:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _smooth_val(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    mx = float(np.max(corr))
    sh = np.clip(alpha * (corr - mx), -400.0, 0.0)
    return mx + float(np.log(np.exp(sh).sum())) / alpha

def _smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    mx = float(np.max(corr))
    sh = np.clip(alpha * (corr - mx), -400.0, 0.0)
    w = np.exp(sh)
    tot = w.sum()
    if tot < 1e-300:
        w = np.zeros(2 * n - 1)
        w[np.argmax(corr)] = 1.0
    else:
        w /= tot
    N = _fft_size(n)
    H = np.fft.rfft(h, N)
    Wf = np.fft.rfft(w, N)
    conv_wh = np.fft.irfft(Wf * H, N).real[:2 * n - 1]
    Part2 = conv_wh[n - 1:2 * n - 1]
    w2 = w[n - 1:2 * n - 1]
    W2r = np.fft.rfft(w2[::-1], N)
    conv_part3 = np.fft.irfft(W2r * H, N).real[:2 * n - 1]
    Part3 = conv_part3[n - 1:2 * n - 1]
    cum_w = np.zeros(2 * n)
    cum_w[1:] = np.cumsum(w)
    Part1 = cum_w[n:] - cum_w[:n]
    return dx * (Part1 - Part2 - Part3)

def _optimize_smooth(h0, alpha, budget, L0=8.0):
    n = len(h0)
    S = n * 0.5
    h = _repair(h0, n)
    t0 = time.time()
    mom = 0.9
    step = 1.0 / max(L0, 1e-12)
    best_h = h.copy()
    best_c = _c5(h)
    best_sv = _smooth_val(h, alpha)
    v = np.zeros(n)
    for it in range(500000):
        if time.time() - t0 > budget:
            break
        if it % 400 == 0:
            c = _c5(h)
            if c < best_c:
                best_c = c
                best_h = h.copy()
        g = _smooth_grad(h, alpha)
        v = mom * v + g
        step_eff = 1.0 / (np.linalg.norm(v) + 1e-15) * 0.8
        step_eff = max(1e-10, min(step_eff, 2.0 / L0))
        for _ in range(8):
            xn = _proj(h - step_eff * v, S)
            fx = _smooth_val(xn, alpha)

            desc = float(np.dot(g, xn - h))
            if fx <= fy + 1e-10 or desc > 0:

            step_eff *= 0.5

def _pn_precache(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    return corr, n

def _update_active(active, corr, n):
    mx = float(np.max(corr))
    tol = max(1e-8, 0.02 * (mx - np.min(corr) + 1e-15))
    active_lag = corr >= mx - tol
    active.fill(False)
    for k in range(2 * n - 1):
        if not active_lag[k]:
            continue
        lag = k - (n - 1)
        for i in range(n):
            ii = i - lag
            if 0 <= ii < n:
                active[i] = True
            ij = i + lag
            if 0 <= ij < n:
                active[i] = True
