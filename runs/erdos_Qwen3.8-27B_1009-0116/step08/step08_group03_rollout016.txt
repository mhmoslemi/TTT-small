import numpy as np
import time

def _fft_size(n):
    m = 2 * n - 1
    N = 1
    while N < m:
        N <<= 1
    return N

def _repair(h, n):
    t = n * 0.5
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if s <= 1e-15:
        return np.full(n, 0.5)
    for _ in range(100):
        h *= t / s
        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
        if abs(s - t) < 1e-14:
            break
    d = t - h.sum()
    if abs(d) > 1e-12:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            if d > 0:
                a = min(d, 1.0 - h[i])
                h[i] += a
                d -= a
            else:
                a = min(-d, h[i])
                h[i] -= a
                d += a
            if abs(d) < 1e-13:
                break
    return h

def _proj(v, S):
    x = np.array(v, dtype=float, copy=True)
    x += (S - x.sum()) / x.size
    for _ in range(20):
        np.clip(x, 0.0, 1.0, out=x)
        d = S - x.sum()
        if abs(d) < 1e-12:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
    return x

def _c5_direct(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _pn_arrays(h):
    n = len(h)
    N = _fft_size(n)
    H = np.fft.rfft(h, N)
    ac = np.fft.irfft(H * np.conj(H), N).real[:n]
    pref = np.empty(n + 1)
    pref[0] = 0.0
    np.cumsum(h, out=pref[1:])
    total = pref[n]
    P = np.empty(n)
    P[0] = total - ac[0]
    Nv = np.zeros(n)
    if n > 1:
        idx = np.arange(1, n)
        P[idx] = pref[n - idx] - ac[idx]
        Nv[idx] = total - pref[idx] - ac[idx]
        mx = float(max(np.max(P), np.max(Nv)))
    else:
        mx = float(P[0])
    return P, Nv, mx

def _smooth_value(h, alpha):

    n = len(h)
    dx = 2.0 / n
    e = np.exp(np.clip(alpha * (P[0] - mx), -400.0, 0.0))
    if n > 1:
        e += np.exp(np.clip(alpha * (P[1:] - mx), -400.0, 0.0)).sum()
        e += np.exp(np.clip(alpha * (Nv[1:] - mx), -400.0, 0.0)).sum()
    return dx * (mx + np.log(e) / alpha)

def _grad_from_weights(h, wP, wN):
    n = len(h)
    dx = 2.0 / n
    u = np.zeros(n)
    if n > 1:
        u[1:] = wP[1:] + wN[1:]
    cp = np.zeros(n)
    cn = np.zeros(n)
    if n > 1:
        cp[1:] = np.cumsum(wP[1:])
        cn[1:] = np.cumsum(wN[1:])
    linP = cp[n - 1::-1]
    linN = cn
    q = np.zeros(n)
    if n > 1 and np.any(u[1:]):
        Nf = _fft_size(n)
        H = np.fft.rfft(h, Nf)
        Urev = np.fft.rfft(u[::-1], Nf)
        Ur = np.fft.rfft(u, Nf)
        convR = np.fft.irfft(H * Urev, Nf)
        R = convR[n - 1:2 * n - 1].real
        convL = np.fft.irfft(H * Ur, Nf)
        L = convL[:n].real
        q = -(R + L)
    return dx * (wP[0] * (1.0 - 2.0 * h) + linP + linN + q)

def _smooth_grad(h, alpha):
    P, Nv, mx = _pn_arrays(h)
    n = len(h)
    wP = np.zeros(n)
    wN = np.zeros(n)
    e0 = np.exp(np.clip(alpha * (P[0] - mx), -400.0, 0.0))
    tot = e0
    if n > 1:
        ep = np.exp(np.clip(alpha * (P[1:] - mx), -400.0, 0.0))
        en = np.exp(np.clip(alpha * (Nv[1:] - mx), -400.0, 0.0))
        tot += ep.sum() + en.sum()
        if tot <= 0.0:
            tot = 1.0
            e0 = 1.0
        wP[0] = e0 / tot
        wP[1:] = ep / tot
        wN[1:] = en / tot
    else:
        wP[0] = 1.0
    return _grad_from_weights(h, wP, wN)

        wP[0] = 1.0
    return _grad_weights(h, wP, wN)

def _fista(h0, alpha, max_t, L0=10.0):
    if max_t <= 0:
        h = h0.copy()
        return h, h.copy(), _c5(h)
    n = len(h0)
    S = n * 0.5
    t0 = time.time()
    h = _repair(h0, n)
    hp = h.copy(); y = h.copy(); t = 1.0
    fy = _soft_val(h, alpha); fh = fy
    best_h = h.copy(); best_c = _c5(h)
    stale = 0; Lc = L0
    for it in range(500000):
        if time.time() - t0 > max_t:
            break
        if it:
            tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
            y = h + ((t - 1.0) / tn) * (h - hp)
            t = tn
            if y.min() < -1e-5 or y.max() > 1.0 + 1e-5 or abs(y.sum() - S) > 1e-8:
                y = _proj(y, S)
            fy = _soft_val(y, alpha)
        else:
            y = h; fy = fh
        g = _soft_grad(y, alpha)
        step = 1.0 / max(Lc, 1e-12)
        accepted = False
        for _ in range(14):
            xn = _proj(y - step * g, S)
            fx = _soft_val(xn, alpha)
            dv = xn - y
            rhs = fy + float(np.dot(g, dv)) + 0.5 * (1.0 / step) * float(np.dot(dv, dv))
            if fx <= rhs + 1e-11:
                accepted = True
                break
            step *= 0.5
        Lc = 1.0 / max(step, 1e-12)
        if accepted:
            hp = h; h = xn; fh = fx; stale = 0
            Lc = max(Lc * 0.85, 1.0)
            if it % 5 == 0:
                c = _c5(h)
                if c < best_c:
                    best_c = c; best_h = h.copy()
        else:
            stale += 1; hp = h; y = h; t = 1.0; fy = fh
            Lc = max(Lc * 0.5, 1.0)
            if stale > 12:
                break
    c = _c5(h)
    if c < best_c:
        best_c = c; best_h = h.copy()
    return h, best_h, best_c

def _sa_search(h0, rng, deadline):
    """Simulated annealing with targeted moves on active constraints"""
    n = len(h0)
    h = h0.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = float(corr.max())
    best_c = mx * (2.0 / n)
    best_h = h.copy()
    t_start = time.time()
    total_t = max(1.0, deadline - t_start)
    it = 0
    while time.time() < deadline:
        frac = min(1.0, (time.time() - t_start) / total_t)
        T = 2.0 * (0.005) ** frac
        k_act = int(np.argmax(corr)) - (n - 1)
        if rng.random() < 0.75:
            k = abs(k_act)
            if k > 0 and k < n:
                if k_act > 0:
                    valid = (h[:n-k] > 0.005) & (h[k:] < 0.995)
                else:
                    valid = (h[k:] > 0.005) & (h[:n-k] < 0.995)
                if valid.sum() > 2:
                    pos = np.nonzero(valid)[0]
                    i = int(pos[rng.randint(len(pos))])
                    if k_act > 0:
                        j = i + k
                    else:
                        j = i - k
                    if j < 0 or j >= n:
                        j = rng.randint(n)
                    if rng.random() < 0.5:
                        d = -rng.random() * 0.04
                    else:
                        d = rng.random() * 0.04
                    h[i] += d; h[j] -= d
                    c2 = np.correlate(h, 1.0 - h, mode='full')
                    mx2 = float(c2.max())
                    delta = mx2 - mx
                    if delta < -1e-14 or (delta > 0 and rng.random() < np.exp(-delta / max(T, 1e-15))):
                        mx = mx2; corr = c2
                        cn = mx * (2.0 / n)
                        if cn < best_c - 1e-14:
                            best_c = cn; best_h = h.copy()
                    else:
                        h[i] -= d; h[j] += d
                    it += 1
                    continue
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            it += 1
            continue
        d = (rng.random() - 0.5) * 0.06
        if h[i] + d < 0: d = -h[i]
        if h[i] + d > 1: d = 1.0 - h[i]
        if h[j] - d < 0: d = h[j]
        if h[j] - d > 1: d = h[j] - 1.0
        if abs(d) < 1e-10:
            it += 1
            continue
        h[i] += d; h[j] -= d
        c2 = np.correlate(h, 1.0 - h, mode='full')
        mx2 = float(c2.max())
        delta = mx2 - mx
        if delta < -1e-14 or (delta > 0 and rng.random() < np.exp(-delta / max(T, 1e-15))):
            mx = mx2; corr = c2
            cn = mx * (2.0 / n)
            if cn < best_c - 1e-14:
                best_c = cn; best_h = h.copy()
        else:
            h[i] -= d; h[j] += d
        it += 1
    return best_h, best_c

def _grad_local(h0, rng, deadline):
    """Gradient-guided local search with active constraint targeting"""
    n = len(h0)
    h = h0.copy()
    c = _c5(h)
    best_c = c; best_h = h.copy()
    P, Nv, _ = _pn(h)
    g = _active_grad(h, P, Nv)
    order = np.argsort(g)
    q = max(5, n // 8)
    low = order[:q]; high = order[-q:]
    recompute = 1024
    it = 0
    t0 = time.time()
    while time.time() < deadline:
        if it % recompute == 0:
            c = _c5(h)
            if c < best_c - 1e-14:
                best_c = c; best_h = h.copy()
            P, Nv, _ = _pn(h)
            g = _active_grad(h, P, Nv)
            order = np.argsort(g)
            low = order[:q]; high = order[-q:]
            recompute = it + 1024
        frac = min(1.0, (time.time() - t0) / max(1e-9, deadline - t0))
        base = max(1e-10, 3e-5 * (1e-6) ** frac)
        if rng.random() < 0.7:
            i = int(low[rng.randint(len(low))])
            j = int(high[rng.randint(len(high))])
        else:
            i = int(rng.randint(n))
            j = int(rng.randint(n))
        if i != j:
            s = 1 if g[i] <= g[j] else (-1 if g[j] < g[i] else (1 if rng.random() < 0.5 else -1))
            avail = min(1.0 - h[i], h[j]) if s > 0 else min(h[i], 1.0 - h[j])
            if avail > 1e-14:
                mult = 0.3 + 1.5 * rng.random()
                if rng.random() < 0.02:
                    mult *= 6.0
                d = s * min(avail, base * mult)
                h[i] += d; h[j] -= d
                cn = _c5(h)
                if cn < c - 1e-15:
                    c = cn
                    if c < best_c - 1e-15:
                        best_c = c; best_h = h.copy()
                else:
                    h[i] -= d; h[j] += d
        it += 1
    return best_h, best_c

def _active_grad(h, P, Nv):
    n = len(h)
    dx = 2.0 / n
    mx = float(P[0])
    if n > 1:
        mx = max(mx, float(np.max(P[1:])), float(np.max(Nv[1:])))
    tol = max(1e-9, 12.0 * dx)
    wP = np.zeros(n)
    wN = np.zeros(n)
    cnt = 0
    if P[0] >= mx - tol:
        wP[0] = 1.0; cnt += 1
    if n > 1:
        mP = P[1:] >= mx - tol
        if np.any(mP):
            pos = np.nonzero(mP)[0] + 1
            wP[pos] = 1.0; cnt += int(pos.size)
        mN = Nv[1:] >= mx - tol
        if np.any(mN):
            pos = np.nonzero(mN)[0] + 1
            wN[pos] = 1.0; cnt += int(pos.size)
    if cnt == 0:
        return np.zeros(n)
    inv = 1.0 / cnt
    wP *= inv; wN *= inv
    return _grad_weights(h, wP, wN)

def _resample(h, m):
    n = len(h)
    if n == m:
        return _repair(h, n)
    if n % m == 0:
        out = h.reshape(m, n // m).mean(axis=1)
    else:
        x0 = (np.arange(n) + 0.5) * (2.0 / n)
        x1 = (np.arange(m) + 0.5) * (2.0 / m)
        out = np.interp(x1, x0, h)
    return _proj(out, m * 0.5)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(560, 0.5)
    n0 = len(h0)
    h0 = _repair(h0, n0)
    best = h0.copy()
    best_c = _c5(best)
    for hh in (1.0 - h0.copy(), 0.5 + 0.5*(h0 - h0[::-1]), 0.5 + 0.5*(h0 + h0[::-1] - 1.0)):
        h2 = _proj(hh, n0 * 0.5)
        c = _c5(h2)
        if c < best_c:
            best = h2; best_c = c
    # Phase 1: FISTA with moderate alpha (first 120s)
    end1 = t0 + min(120.0, 0.12 * budget_s)
    if end1 > time.time():
        hcur = best.copy()
        tt = (end1 - time.time()) / 4.0
        for a in [30, 150, 800, 4000]:
            if time.time() >= end1 - 1:
                break
            hf, bh, bc = _fista(hcur, a, tt, 10.0)
            cf = _c5(hf)
            if bc < best_c:
                best = bh; best_c = bc
            if cf < best_c:
                best = hf; best_c = cf
            hcur = hf if cf <= bc else bh
    # Phase 2: Multi-resolution exploration (next 100s)
    end2 = t0 + min(220.0, 0.22 * budget_s)
    if end2 > time.time():
        for m in [280, 512, n0, 640]:
            if time.time() >= end2 - 3:
                break
            slot = min(25.0, end2 - time.time() - 2)
            if slot < 4:
                break
            hm = _resample(best, m)
            hcur = hm.copy()
            tt_per = slot / 3.0
            for a in [100, 1000, 10000]:
                if time.time() >= end2 - 2:
                    break
                hf, bh, bc = _fista(hcur, a, tt_per, 10.0)
                cf = _c5(hf)
                if bc < best_c:
                    hb = _resample(bh, n0) if len(bh) != n0 else bh
                    best = hb; best_c = _c5(hb)
                if cf < best_c:
                    hb = _resample(hf, n0) if len(hf) != n0 else hf
                    best = hb; best_c = _c5(hb)
                hcur = hf if cf <= bc else bh
    # Phase 3: High-alpha FISTA at n0 (next 300s)
    hmain = best if len(best) == n0 else _resample(best, n0)
    end3 = t0 + min(520.0, 0.52 * budget_s)
    if end3 > time.time():
        alphas = [200, 1000, 5000, 20000, 80000, 300000]
        tt_per = (end3 - time.time()) / len(alphas)
        hcur = hmain.copy()
        for a in alphas:
            if time.time() >= end3 - 2:
                break
            hf, bh, bc = _fista(hcur, a, tt_per, 10.0)
            cf = _c5(hf)
            if bc < best_c:
                best = bh; best_c = bc
            if cf < best_c:
                best = hf; best_c = cf
            hcur = hf if cf <= bc else bh
    # Phase 4: Simulated annealing
    end4 = t0 + min(780.0, 0.78 * budget_s)
    if end4 > time.time():
        best, best_c = _sa_search(best, rng, end4)
    # Phase 5: Gradient local search
    end5 = t0 + budget_s - 60.0
    if end5 > time.time():
        best, best_c = _grad_local(best, rng, end5)
    # Phase 6: Final SA polish
    end6 = t0 + budget_s - 2.0
    if end6 > time.time():
        best, best_c = _sa_search(best, rng, end6)
    # Final variants
    for hh in (1.0 - best, best[::-1], 0.5 + 0.5*(best - best[::-1]), 0.5 + 0.5*(best + best[::-1] - 1.0)):
        h2 = _proj(hh, len(hh) * 0.5)
        c = _c5(h2)
        if c < best_c:
            best = h2; best_c = c
    best = _repair(best, len(best))
    best_c = _c5(best)
    return best, best_c, len(best)
