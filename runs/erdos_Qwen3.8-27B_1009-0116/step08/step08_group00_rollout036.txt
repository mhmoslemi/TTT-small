def _sum_fn(x, half):
    return x.sum() - half

def _sum_jac(x):
    return np.ones(len(x))

def _slsqp(h, alpha, ml, mi, ft):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fn, 'jac': _sum_jac, 'args': (n * 0.5,)}]
    try:
        res = minimize(lambda x: _sv(x, alpha, ml)[0], h,
                       jac=lambda x: _sv(x, alpha, ml)[1],
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': mi, 'ftol': ft})
        return _repair(res.x, n)
    except Exception:
        return h

def _proj_grad(h, alpha, ml, steps, s0, mu=0.75):
    n = len(h)
    v = np.zeros(n)
    st = s0
    cc = _eval(h)
    for _ in range(steps):
        _, g = _sv(h, alpha, ml)
        v = mu * v + (1.0 - mu) * g
        taken = False
        for _ in range(12):

            cn = _eval(hn)

                taken = True
                break
            st *= 0.45
        if not taken:
            st = min(st * 1.4, s0 * 8)
    return _repair(h, n)

def _grad_pair_search(h, n_iter, rng, eta0):
    n = len(h)
    best_h = h.copy()
    best_c = _eval(best_h)
    for it in range(n_iter):
        eta = eta0 * (0.999 ** it)
        if eta < 1e-8:
            eta = 1e-8
        # Compute gradient (subgradient of max)
        c = np.correlate(best_h, 1-best_h, mode='full') * (2.0/n)
        kmax = np.argmax(c)
        # Gradient at the active lag
        L = kmax - (n-1)
        g = np.zeros(n)
        if L >= 0:
            g[L:] += (1.0 - best_h[:n-L])
            g[:n-L] -= best_h[L:]
        else:
            M = -L
            g[:n-M] += (1.0 - best_h[M:])
            g[M:] -= best_h[:n-M]
        g *= 2.0/n
        # Pick random pair
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, n-1)) % n
        delta = eta * (g[i] - g[j])
        for sign in ([1,-1] if delta > 0 else [-1,1]):
            d = sign * abs(delta)
            ni, nj = best_h[i]+d, best_h[j]-d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                best_h[i], best_h[j] = ni, nj
                cn = _eval(best_h)
                if cn < best_c:
                    best_c = cn
                else:
                    best_h[i], best_h[j] = ni-d, nj+d
                break
    return best_h, best_c

```python
import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= t / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - t) < 1e-12:
            break
    d = t - h.sum()
    if abs(d) > 1e-12:
        order = np.argsort(-h) if d > 0 else np.argsort(h)
        for idx in order:
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
            if abs(d) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _proj(h, t):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0, 1).sum() > t:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _og(x, a, ml):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    idx = np.argsort(c)[-ml:]
    s = np.clip(a * (c[idx] - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-18:
            continue
        L = idx[ii] - (n - 1)
        if L >= 0:
            g[L:] += w[ii] * (1 - x[:n - L])
            g[:n - L] -= w[ii] * x[L:]
        else:
            M = -L
            g[:n - M] += w[ii] * (1 - x[M:])
            g[M:] -= w[ii] * x[:n - M]
    return m + np.log(w.sum()) / a, g * dx

def _sf(x, hf):
    return x.sum() - hf

def _sj(x):
    return np.ones(len(x))

def _slsqp(h, a, ml, mi, ft):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sf, 'jac': _sj, 'args': (n * 0.5,)}]
    try:
        r = minimize(lambda x: _og(x, a, ml)[0], h,
                     jac=lambda x: _og(x, a, ml)[1],
                     method='SLSQP', bounds=bnds, constraints=cons,
                     options={'maxiter': mi, 'ftol': ft})
        return _repair(r.x, n)
    except Exception:
        return h

def _pgd(h, a, ml, steps, s0, mu=0.72):
    n = len(h)
    v = np.zeros(n)
    st = s0
    cc = _eval(h)
    for _ in range(steps):
        _, g = _og(h, a, ml)
        v = mu * v + (1.0 - mu) * g
        ok = False
        for _ in range(12):
            hn = _proj(h - st * v, n * 0.5)
            cn = _eval(hn)
            if cn < cc:
                h, cc, st, ok = hn, cn, min(st * 1.1, s0 * 8), True
                break
            st *= 0.45
        if not ok:
            st = min(st * 1.5, s0 * 8)
    return _repair(h, n)

def _gp(h, nit, rng, e0):
    n = len(h)
    dh = h.copy()
    dc = _eval(dh)
    for it in range(nit):
        e = e0 * max(0.2, 1.0 - it / nit)
        c = np.correlate(dh, 1 - dh, mode='full')
        top2 = np.argsort(c)[-2:]
        g = np.zeros(n)
        for km in top2:
            L = int(km) - (n - 1)
            if L >= 0:
                g[L:] += 1 - dh[:n - L]
                g[:n - L] -= dh[L:]
            else:
                M = -L
                g[:n - M] += 1 - dh[M:]
                g[M:] -= dh[:n - M]
        g *= 2.0 / n
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, n - 1)) % n
        d = e * (g[i] - g[j])
        if d > 0:
            seq = (1, -1)
        else:
            seq = (-1, 1)
        for sgn in seq:
            nd = sgn * abs(d)
            ni, nj = dh[i] + nd, dh[j] - nd
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                dh[i], dh[j] = ni, nj
                cn = _eval(dh)
                if cn >= dc:
                    dh[i], dh[j] = ni - nd, nj + nd
                else:
                    dc = cn
            break
    return dh, dc

def _rp(h, nit, rng, b0):

    dh = h.copy()
    dc = _eval(dh)
    b = b0
    for _ in range(nit):
        b = max(1e-7, b * 0.99995)
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, n - 1)) % n

            ni, nj = dh[i] + nd, dh[j] - nd

                cn = _eval(dh)
                if cn >= dc:
                    dh[i], dh[j] = ni - nd, nj + nd
                else:
