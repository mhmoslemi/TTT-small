```python
def _smooth_grad(h, weights):
    """
    h: array of length n
    weights: array of length 2n-1, the softmax weights for each correlation value
    Returns: gradient of sum_m weights[m]*c[m] w.r.t. h
    """
    n = len(h)
    grad = np.zeros(n)
    # For each m with significant weight:
    for m in range(2*n-1):
        if weights[m] < 1e-15:
            continue
        s = m - (n - 1)
        # T1[j] = (1-h[j+s]) if 0<=j+s<n
        # T2[j] = h[j-s] if 0<=j-s<n
        # grad[j] += weights[m] * (T1[j] - T2[j])
        # T1: j+s in [0,n-1] means j in [-s, n-1-s]
        j_lo = max(0, -s)
        j_hi = min(n-1, n-1-s)
        if j_lo <= j_hi:
            grad[j_lo:j_hi+1] += weights[m] * (1.0 - h[j_lo+s:j_hi+s+1])
        # T2: j-s in [0,n-1] means j in [s, s+n-1]
        j_lo2 = max(0, s)
        j_hi2 = min(n-1, s+n-1)
        if j_lo2 <= j_hi2:
            grad[j_lo2:j_hi2+1] -= weights[m] * h[j_lo2-s:j_hi2-s+1]
    return grad
```

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _proj(v, S):
    x = np.asarray(v, dtype=float)
    n = x.size
    lo = x.min() - 1.0
    hi = x.max() + 1.0
    for _ in range(50):
        mid = 0.5*(lo+hi)
        if np.clip(x-mid, 0, 1).sum() > S:
