import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_lag(h, k, dx):
    """Gradient of correlation at lag k (in samples) w.r.t. h."""
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        g[:n-k] += (1.0 - h[k:])
        if k > 0:
            g[k:] -= h[:-k]
    else:
        sp = -k
        g[sp:] += (1.0 - h[:-sp])
        g[:n-sp] -= h[sp:]
    return g * dx

def _smax_obj(h, alpha, topk):
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
        ct = c[idx]
    else:
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _grad_smax(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
        ct = c[idx]
    else:
        idx = np.arange(len(c))
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-14:
            continue
        g += w[ii] * _grad_lag(h, idx[ii] - (n-1), dx)
    return g

    f = _smax_obj(h, alpha, topk)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):

    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):

    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _grad_lag(h, k, dx):

    if k >= 0:
        g[:n - k] += (1.0 - h[k:])
        if k > 0:
            g[k:] -= h[:n - k]
    else:
        sp = -k
        g[:n - sp] += (1.0 - h[sp:])
        g[sp:] -= h[:n - sp]
    return g * dx

    else:

    s = np.clip(alpha * (ct - m), -400, 400)

def _hinge_obj(h, target, alpha):
    c = _correl(h)
    excess = np.maximum(c - target, 0)
    return np.sum(excess * excess) + alpha * (c.max() - target + 1e-10)

        h_up = _upscale(h_c, factor, n0)
        h_up = _repair(h_up, n0)
        h_up = _nesterov(h_up, 30000, 10, 600, 0.002, dl4, 0.80)
        c5v = _eval(h_up)
        if c5v < best_c5:
            best_c5 = c5v; best_h = h_up.copy()

    # Phase 5: Fourier perturbation restarts (560-720s)
    dl5 = min(deadline, t0 + 720)
    n = n0
    for trial in range(8):
        if time.time() > dl5:
            break
        h_p = best_h.copy()
        k1 = rng.randint(2, 15)
        k2 = rng.randint(15, 40)
        amp1 = rng.uniform(0.008, 0.04)
        amp2 = rng.uniform(0.003, 0.02)
        ph1 = rng.uniform(0, 2 * np.pi)
        ph2 = rng.uniform(0, 2 * np.pi)
        pert = amp1 * np.cos(2 * np.pi * k1 * np.arange(n) / n + ph1)
        pert += amp2 * np.sin(2 * np.pi * k2 * np.arange(n) / n + ph2)
        h_p += pert + rng.randn(n) * 0.003
        h_p = _repair(h_p, n)
        h_p = _nesterov(h_p, 5000, 20, 1000, 0.004, dl5, 0.84)
        h_p = _nesterov(h_p, 50000, 8, 800, 0.002, dl5, 0.80)
        c5v = _eval(h_p)
        if c5v < best_c5:
            best_c5 = c5v; best_h = h_p.copy()

    # Phase 6: Annealing (720-870s)
    dl6 = min(deadline, t0 + 870)
    blocks = [1, 2, 3, 5, 8, 13, 21, 34]
    h_a, c5_a = _anneal(best_h.copy(), 35000, rng, dl6, 0.004, 1e-10, blocks)
    if c5_a < best_c5:
        best_c5 = c5_a; best_h = h_a.copy()

    # Phase 7: Pair descent (870-980s)
    dl7 = min(deadline, t0 + 980)
    h_pd, c5_pd = _pair_descent(best_h.copy(), dl7, max_sweeps=25)
    if c5_pd < best_c5:
        best_c5 = c5_pd; best_h = h_pd.copy()

    # Phase 8: Final high-alpha Nesterov + micro-polish (980-end)
    dl8 = deadline
    h_f = _nesterov(best_h.copy(), 1500000, 5, 1200, 0.0005, dl8, 0.75)
    c5v = _eval(h_f)
    if c5v < best_c5:
        best_c5 = c5v; best_h = h_f.copy()

    # Micro single-coordinate polish
    rem = max(0, dl8 - time.time())
    nc = len(best_h)
    c5c = _eval(best_h)
    n_micro = min(int(rem / 0.0018), 40000)
    for _ in range(n_micro):
        if time.time() > dl8:
            break
        i = rng.randint(nc)
        j = (i + 1 + rng.randint(0, 4)) % nc
        for d in [3e-7, -3e-7, 1e-7, -1e-7]:
            ni, nj = best_h[i] + d, best_h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                best_h[i], best_h[j] = ni, nj
                c5n = _eval(best_h)
                if c5n < c5c:
                    c5c = c5n; best_c5 = c5c
                    break
                best_h[i], best_h[j] = ni - d, nj + d

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
