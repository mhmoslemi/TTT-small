import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_vec(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _grad_lse(h, alpha, top_k):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    idx = np.argsort(c)[-top_k:]
    c_top = c[idx]
    m = c_top.max()
    s = np.clip(alpha * (c_top - m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        sh = (n - 1) - k
        wk = w[ii] * dx
        if sh >= 0:
            cnt = n - sh
            g[:cnt] += wk * (1.0 - h[sh:sh + cnt])
            if sh > 0:
                g[sh:sh + cnt] -= wk * h[:cnt]
        else:
            sp = -sh
            cnt = n - sp
            g[sp:] += wk * (1.0 - h[:cnt])
            g[:cnt] -= wk * h[sp:]
    return g

def _proj_sum(h, target):
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _adam(h, n_iter, alpha, top_k, lr, b1, b2, eps):
    n = len(h)
    target = n * 0.5
    mv = np.zeros(n); vv = np.zeros(n)
    cur_c = _eval(h); best_h = h.copy(); best_c = cur_c
    for it in range(n_iter):
        g = _grad_lse(h, alpha, top_k)
        mv = b1 * mv + (1 - b1) * g
        vv = b2 * vv + (1 - b2) * g * g
        mhat = mv / (1 - b1 ** (it + 1))
        vhat = vv / (1 - b2 ** (it + 1))
        h_try = _proj_sum(h - lr * mhat / (np.sqrt(vhat) + eps), target)
        c_try = _eval(h_try)
        if c_try < cur_c:
            h = h_try; cur_c = c_try
            if c_try < best_c:
                best_c = c_try; best_h = h.copy()
        else:
            lr *= 0.92
    return _repair(best_h, n), best_c

def _fine_local(h, n_iter, step0, cool, rng):
    n = len(h)
    cur_c = _eval(h); best_c = cur_c; best_h = h.copy()
    for it in range(n_iter):
        frac = 1.0 - it / max(1, n_iter)
        step = step0 * (0.3 + 0.7 * frac ** cool)
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 15)) % n
        d = step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c_new = _eval(h)
            if c_new < cur_c:
                cur_c = c_new
                if c_new < best_c:
                    best_c = c_new; best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def _rounding(h, n_iter, rng):
    n = len(h)
    cur_c = _eval(h); best_c = cur_c; best_h = h.copy()
    order = rng.permutation(n)
    for i in range(min(n_iter, n * 4)):
        idx = order[i % n]
        target_val = 1.0 if h[idx] > 0.5 else 0.0
        if abs(h[idx] - target_val) < 1e-6:
            continue
        h_new = h.copy()
        h_new[idx] = target_val
        h_new = _repair(h_new, n)
        c_new = _eval(h_new)
        if c_new < cur_c:
            h = h_new; cur_c = c_new
            if c_new < best_c:
                best_c = c_new; best_h = h.copy()
    return best_h, best_c

def _downsample(h, f):
    n = len(h); m = n // f
    return h[:m * f].reshape(m, f).mean(axis=1).copy()

def _upsample(h, f):
    n = len(h); out = np.empty(n * f)
    for i in range(n):
        out[i * f:(i + 1) * f] = h[i]
    return out

def _slsqp(h, alpha, maxiter, top_k):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    target = n * 0.5
    def obj_f(x):
        c = _eval_vec(x)
        idx2 = np.argsort(c)[-top_k:]
        ct = c[idx2]; m2 = ct.max()
        s2 = np.clip(alpha * (ct - m2), -400, 400)
        s2 -= s2.max()
        return m2 + np.log(np.exp(s2).sum()) / alpha
    def grad_f(x):
        return _grad_lse(x, alpha, top_k)
    def sum_f(x):
        return x.sum() - target
    def sum_j(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': sum_f, 'jac': sum_j}]
    try:
        res = minimize(obj_f, h, jac=grad_f, method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return _repair(res.x, n)
    except Exception:
        return h
