            q = max(20, n // 8)
            if rng.random() < 0.5:

            else:
                j = order[rng.randint(q)]
                i = order[-(1 + rng.randint(q))]
        else:
            i = rng.randint(n)
            j = rng.randint(n)
        
        if i == j:
            it += 1
            continue

        # If we increase h[i] and decrease h[j]:
        #   change in overlap a is delta * (D[i,a] - D[j,a])

        diff = D[i] - D[j]  # shape (len(active),)
        
        # Max safe delta (for this direction)
        # Need: corr[active] + delta * diff <= c + margin
        # If diff[a] > 0: delta <= (c - corr[active[a]]) / diff[a]
        # If diff[a] <= 0: no constraint (delta can be anything ≥ 0)
        pos_mask = diff > 1e-15
        if pos_mask.any():
            max_d = (c + 1e-12 - corr[active[pos_mask]]) / diff[pos_mask]
            max_d = max_d.min()
        else:
            max_d = np.inf
        
        # Also constrain by box: h[i]+delta <= 1, h[j]-delta >= 0
        max_d = min(max_d, 1.0 - h[i], h[j])
        
        # Adaptive base step
        frac = (time.time() - tl0) / max(1e-9, deadline - tl0)
        base = 0.05 * (0.02) ** (frac * 0.5)

            it += 1

        h[j] -= target_d
        corr[active] += target_d * diff
        it += 1

    c = float(corr.max())
    if c < best_c:
        best_c = c
        best_h = h.copy()

def _proj(v, S):
    x = np.asarray(v, dtype=float)
    x = x + (S - x.sum()) / x.size
    for _ in range(30):
        np.clip(x, 0.0, 1.0, out=x)
        d = S - x.sum()
        if abs(d) < 1e-12:
            break
        m = (x < 1.0) if d > 0 else (x > 0.0)
        c = m.sum()
        if c == 0:
            break
        x[m] += d / c
    return x

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _corr_vals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _derivs(h, shifts):
    n = len(h)
    D = np.zeros((n, len(shifts)))
    for a in range(len(shifts)):
        s = shifts[a]
        if s >= 0:
            D[s:, a] += 1.0 - h[:n - s]
            D[:n - s, a] -= h[s:]
        else:
            ss = -s
            D[:n - ss, a] += 1.0 - h[ss:]
            D[ss:, a] -= h[:n - ss]
    return D

def _smooth_obj(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    mx = c.max()
    e = np.exp(np.clip(alpha * (c - mx), -400, 0))
    return dx * (mx + np.log(e.sum()) / alpha)

def _smooth_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    mx = c.max()
    e = np.exp(np.clip(alpha * (c - mx), -400, 0))
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        s = k - n + 1
        wk = w[k]
        if s >= 0:
            g[s:] += wk * (1.0 - h[:n - s])
            g[:n - s] -= wk * h[s:]
        else:
            ss = -s
            g[:n - ss] += wk * (1.0 - h[ss:])
            g[ss:] -= wk * h[:n - ss]
    return dx * g

def _fista(h0, alpha, max_time, L=8.0):
    n = len(h0)
    S = n * 0.5
    t0 = time.time()
    h = _repair(h0, n)
    hp = h.copy(); y = h.copy(); t = 1.0
    fy = _smooth_obj(y, alpha)
    best_h = h.copy(); best_f = fy
    Lcur = L
    for _ in range(300000):
        if time.time() - t0 > max_time:
            break
        if t > 1.0:
            tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
            y = h + ((t - 1.0) / tn) * (h - hp)
            t = tn
            if y.min() < -0.5 or y.max() > 1.5 or abs(y.sum() - S) > 0.1:
                y = _proj(y, S)
            fy = _smooth_obj(y, alpha)
        g = _smooth_grad(y, alpha)
        step = 1.0 / max(Lcur, 1e-10)
        ok = False
        for _ in range(10):
            xn = _proj(y - step * g, S)
            fx = _smooth_obj(xn, alpha)
            d = xn - y
            if fx <= fy + g @ d + 0.5 * (1.0 / step) * (d @ d) + 1e-10:
                ok = True; break
            step *= 0.5
        Lcur = 1.0 / max(step, 1e-10)
        if ok:
            hp = h; h = xn
            Lcur = max(Lcur * 0.9, 1.0)
            if fx < best_f:
                best_f = fx; best_h = h.copy()
        else:
            hp = h; y = h; t = 1.0
            Lcur = max(Lcur * 0.5, 1.0)
    return best_h

def _resample(h, m):
    n = len(h)
    if n == m:
        return _repair(h, n)
    x0 = (np.arange(n) + 0.5) * (2.0 / n)
    x1 = (np.arange(m) + 0.5) * (2.0 / m)
    return _proj(np.interp(x1, x0, h), m * 0.5)

def _local_search(h0, rng, deadline):
    n = len(h0)
    h = h0.copy()
    corr = _corr_vals(h)
    c = float(corr.max())
    best_c = c; best_h = h.copy()
    tol = max(0.01, 0.05 * c)
    active = np.where(corr > c - tol)[0]
    shifts = active - n + 1
    D = _derivs(h, shifts)
    tl0 = time.time()
    it = 0
    while time.time() < deadline - 0.3:
        if it % 800 == 0 and it > 0:
            corr = _corr_vals(h)
            c = float(corr.max())
            if c < best_c:
                best_c = c; best_h = h.copy()
            tol = max(0.01, 0.05 * c)
            active = np.where(corr > c - tol)[0]
            shifts = active - n + 1
            D = _derivs(h, shifts)
        g = D.mean(axis=1)
        if rng.random() < 0.80:
            order = np.argsort(g)
            q = max(16, n // 10)
            i = int(order[rng.randint(q)])
            j = int(order[n - 1 - rng.randint(q)])
        else:
            i = int(rng.randint(n)); j = int(rng.randint(n))
        if i == j:
            it += 1; continue
        diff = D[i] - D[j]
        pos = diff > 1e-14
        if pos.any():
            max_d = np.min((c + 1e-12 - corr[active[pos]]) / diff[pos])
        else:
            max_d = 1e9
        max_d = min(max_d, 1.0 - h[i], h[j])
        frac = (time.time() - tl0) / max(1e-9, deadline - tl0)
        base = 0.06 * (0.01) ** (frac * 0.6)
        target = min(max_d * 0.85, base * (0.4 + 1.2 * rng.random()))
        if rng.random() < 0.02:
            target = min(max_d * 0.9, 0.3 * (0.5 + rng.random()))
        if target < 1e-13:
            it += 1; continue
        h[i] += target; h[j] -= target
        corr[active] += target * diff
        it += 1
    corr = _corr_vals(h)
    c = float(corr.max())
    if c < best_c:
        best_c = c; best_h = h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(560, 0.5)
    n0 = len(h0)
    h0 = _repair(h0, n0)
    best = h0.copy()
    best_c = _c5(best)
    # Phase 1: quick FISTA (40s)
    e1 = t0 + min(40.0, 0.04 * budget_s)
    if time.time() < e1 - 2:
        hf = _fista(best, 100.0, e1 - time.time(), L=8.0)
        hc = _c5(hf)
        if hc < best_c:
            best_c = hc; best = hf
    # Phase 2: multi-res (40s)
    e2 = t0 + min(80.0, 0.08 * budget_s)
    if time.time() < e2 - 5 and budget_s > 200:
        for m in (n0 // 2, n0 // 4):
            if time.time() >= e2 - 4:
                break
            slot = min(25.0, e2 - time.time() - 2)
            if slot < 4:
                break
            hm = _resample(best, m)
            hf = _fista(hm, 400.0, slot, L=8.0)
            hu = _resample(hf, n0)
            hc = _c5(hu)
            if hc < best_c:
                best_c = hc; best = hu
    # Phase 3: FISTA increasing alpha (up to 250s from start)
    e3 = t0 + min(250.0, 0.25 * budget_s)
    if time.time() < e3 - 5:
        for a in (400, 1500, 6000, 25000, 100000):
            if time.time() >= e3 - 3:
                break
            slot = min(20.0, e3 - time.time() - 2)
            if slot < 3:
                break
            hf = _fista(best, a, slot, L=8.0)
            hc = _c5(hf)
            if hc < best_c:
                best_c = hc; best = hf
    # Phase 4: local search (remaining)
    e4 = t0 + budget_s - 1.0
    if time.time() < e4 - 1:
        best, best_c = _local_search(best, rng, e4)
    # Final variant checks
    cands = [best, _proj(1.0 - best, n0 * 0.5), _proj(best[::-1], n0 * 0.5)]
    for ch in cands:
        cc = _c5(ch)
        if cc < best_c:
            best_c = cc; best = ch
    best = _repair(best, n0)
    best_c = _c5(best)
    return best, best_c, n0
