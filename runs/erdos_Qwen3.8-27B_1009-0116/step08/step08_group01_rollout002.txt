def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)

        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(-h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return h

    c = np.correlate(h, 1.0-h, mode='full') * 2.0/n
    idx = np.argsort(c)[-top_k:]
    ct = c[idx]
    m = ct.max()
    s = np.clip(alpha*(ct-m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum())/alpha

    # The result has the linear correlation in positions 0..M+N-2

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0-h, mode='full')) * 2.0/n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(-h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _proj_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)*0.5, 0, 1)

def _lse_val(h, alpha, top_k):
    n = len(h)
    c = np.correlate(h, 1.0-h, mode='full') * 2.0/n
    idx = np.argsort(c)[-top_k:]
    ct = c[idx]
    m = ct.max()
    s = np.clip(alpha*(ct-m), -400, 400)
    s -= s.max()
    return m + np.log(np.exp(s).sum())/alpha

def _lse_grad(h, alpha, top_k):
    n = len(h)
    dx = 2.0/n
    c = np.correlate(h, 1.0-h, mode='full') * dx
    idx = np.argsort(c)[-top_k:]
    ct = c[idx]
    m = ct.max()
    s = np.clip(alpha*(ct-m), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for ii, k in enumerate(idx):
        if w[ii] < 1e-15:
            continue
        sh = (n-1) - k
        wk = w[ii] * dx
        if sh >= 0:
            g[:n-sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n-sp])
            g[:n-sp] -= wk * h[sp:]
    return g

def _mgd(h, n_iter, alpha, top_k, step0, mu=0.7):
    n = len(h)
    target = n * 0.5
    v = np.zeros(n)
    step = step0
    cur_c = _eval(h)
    stall = 0
    for it in range(n_iter):
        g = _lse_grad(h, alpha, top_k)
        v = mu * v + (1-mu) * g
        h_try = _proj_sum(h - step * v, target)
        c_try = _eval(h_try)
        if c_try < cur_c:
            if cur_c - c_try > 1e-15:
                stall = 0
            else:
                stall += 1
            h = h_try
            cur_c = c_try
            step = min(step * 1.03, step0 * 5.0)
        else:
            step *= 0.5
            step = max(step, step0 * 0.05)
        if stall > 200:
            break
    return h, cur_c

    target = n * 0.5
    bnds = [(0.0, 1.0)] * n

        return _lse_val(x, alpha, top_k)

        return _lse_grad(x, alpha, top_k)

    except:
