import cvxpy as cp

def _trust_region_step(h, k_star, r, target_c):
    n = len(h)
    # Compute gradient of c[k_star] at h
    g = np.zeros(n)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    L = k_star - (n - 1)
    if L >= 0:
        g[L:] += dx * (1.0 - h[:n-L])
        g[:n-L] -= dx * h[L:]
    else:
        M = -L
        g[:n-M] += dx * (1.0 - h[M:])
        g[M:] -= dx * h[:n-M]
    
    delta = cp.Variable(n)
    obj = cp.Minimize(cp.sum_squares(delta))  # Trust region
    cons = [
        g @ delta <= -0.01 * (c[k_star] * dx - target_c),  # Reduce the worst lag
        h + delta >= 0,
        h + delta <= 1,
        cp.sum(delta) == 0,
    ]
    prob = cp.Problem(obj, cons)
    try:
        prob.solve(solver=cp.CLARABEL)  # or ECOS, SCS
    except Exception:
        return h
    d = delta.value
    if d is None:
        return h
    return _repair(h + d, n)
