    lr = init_lr
    patience = 0
    for it in range(200000):
        if time.time() > t_end:
            break
        g, _ = _subgrad(h, n, top_count)
        g -= g.mean()
        improved = False
        for _trial in range(30):
            h_new = _repair(h - lr * g, n)
            c_new = _eval(h_new)
            if c_new < best_c:
                best_c, best_h = c_new, h_new.copy()
            if c_new < _eval(h) - 1e-14:
                h = h_new
                improved = True
                break
            lr *= 0.4
        if improved:
            patience = 0
            lr = min(lr * 1.01, init_lr * 5)
        else:
            patience += 1
            lr *= 0.95
            if patience > 200:
                break
    return best_h, best_c

def _local_search(h, n, t_end, rng, n_tries_max=50000):
    best_h, best_c = h.copy(), _eval(h)
    step = 5e-4
    c_curr = best_c
    for it in range(n_tries_max):
        if time.time() > t_end:
            break
        if it % 100 == 0:
            step *= 0.98
        mode = rng.randint(5)
        if mode == 0:
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 5)) % n
            d = step * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c_new = _eval(h)
                if c_new < c_curr:
                    c_curr = c_new
                    if c_new < best_c:
                        best_c, best_h = c_new, h.copy()
                else:
                    h[i], h[j] = h[i] - d, h[j] + d
        elif mode == 1:
            i = rng.randint(n)
            j = (i + 6 + rng.randint(0, 15)) % n
            d = step * 0.5 * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c_new = _eval(h)
                if c_new < c_curr:
                    c_curr = c_new
                    if c_new < best_c:
                        best_c, best_h = c_new, h.copy()
                else:
                    h[i], h[j] = h[i] - d, h[j] + d
        elif mode == 2:
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 3)) % n
            k2 = (i + 4 + rng.randint(0, 8)) % n
            d1, d2 = step * 0.3 * rng.randn(2)
            ni, nj, nk = h[i] + d1, h[j] + d2, h[k2] - d1 - d2
            if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= nk <= 1:
                h[i], h[j], h[k2] = ni, nj, nk
                c_new = _eval(h)
                if c_new < c_curr:
                    c_curr = c_new
                    if c_new < best_c:
                        best_c, best_h = c_new, h.copy()
                else:
                    h[i], h[j], h[k2] = h[i] - d1, h[j] - d2, h[k2] + d1 + d2
        elif mode == 3:
            # Gradient-informed: move against subgradient
            g, _ = _subgrad(h, n, 20)
            g -= g.mean()
            idx = np.argsort(np.abs(g))[-10:]
            i = rng.choice(idx)
            j = (i + 1 + rng.randint(0, 4)) % n
            direction = -np.sign(g[i])
            d = step * 0.3 * direction * abs(rng.randn())
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c_new = _eval(h)
                if c_new < c_curr:
                    c_curr = c_new
                    if c_new < best_c:
                        best_c, best_h = c_new, h.copy()
                else:
                    h[i], h[j] = h[i] - d, h[j] + d
        else:
            # Small 2-point fine tune
            i = rng.randint(n)
            j = (i + rng.randint(1, 10)) % n
            d = step * 0.1 * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c_new = _eval(h)
                if c_new < c_curr:
                    c_curr = c_new
                    if c_new < best_c:
                        best_c, best_h = c_new, h.copy()
                else:
                    h[i], h[j] = h[i] - d, h[j] + d
    return best_h, best_c

def _ultra_fine(h, n, t_end, rng):
    best_h, best_c = h.copy(), _eval(h)
    c_curr = best_c
    for it in range(200000):
        if time.time() > t_end:
            break
        step = 1e-6 * (1.0 + 0.5 * np.sin(it * 0.001))
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 7)) % n
        d = step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c_new = _eval(h)
            if c_new < c_curr:
                c_curr = c_new
                if c_new < best_c:
                    best_c, best_h = c_new, h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    t_global_end = t0 + budget_s
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    def _try(h_new):
        nonlocal best_h, best_c5
        c = _eval(h_new)
        if c < best_c5:
            best_c5, best_h = c, h_new.copy()
        return c

    # Phase 1: SLSQP moderate alpha (25% time)
    t_end1 = t0 + budget_s * 0.25
    for alpha in [2000, 20000, 200000]:
        if time.time() > t_end1 - 10:
            break
        c = _try(_slsqp_smooth(best_h.copy(), n, alpha, 800, t_end1))

    # Phase 2: Subgradient descent (25% time)
    t_end2 = t0 + budget_s * 0.50
    for top_k in [30, 80, 200]:
        if time.time() > t_end2 - 10:
            break
        lr0 = 0.005 if top_k <= 50 else 0.002
        best_h, best_c5 = _subgrad_descent(best_h.copy(), n, t_end2, top_k, lr0)
        if best_c5 > _eval(best_h):
            best_c5 = _eval(best_h)

    # Phase 3: SLSQP high alpha (15% time)
    t_end3 = t0 + budget_s * 0.65
    for alpha in [1000000, 10000000]:
        if time.time() > t_end3 - 10:
            break
        _try(_slsqp_smooth(best_h.copy(), n, alpha, 600, t_end3))

    # Phase 4: Multi-scale local search (25% time)
    t_end4 = t0 + budget_s * 0.90
    best_h, best_c5 = _local_search(best_h, n, t_end4, rng, 80000)

    # Phase 5: Ultra-fine polish (10% time)
    t_end5 = t0 + budget_s * 0.995
    best_h, best_c5 = _ultra_fine(best_h, n, t_end5, rng)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
