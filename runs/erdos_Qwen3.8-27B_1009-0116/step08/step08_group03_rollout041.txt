import numpy as np

def _fft_size(n):
    m = 2 * n - 1
    N = 1
    while N < m:

    return N

    t = n * 0.5

    s = h.sum()
    if s <= 1e-15:
        return np.full(n, 0.5)

        h *= t / s
        np.clip(h, 0.0, 1.0, out=h)
        s = h.sum()
        if abs(s - t) < 1e-13:
            break
    d = t - h.sum()
    if abs(d) > 1e-13:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in order:
            if d > 0:
                a = min(d, 1.0 - h[i])
                h[i] += a; d -= a
            else:
                a = min(-d, h[i])
                h[i] -= a; d += a

        if accel and it:
            tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
            y = h + ((t - 1.0) / tn) * (h - hp)
            t = tn
            if y.min() < -1e-4 or y.max() > 1.0 + 1e-4 or abs(y.sum() - S) > 1e-7:
                y = _proj(y, S)
            fy = _smooth_value(y, alpha)
        else:
            y = h; fy = fh
        g = _smooth_grad(y, alpha)
        step = 1.0 / max(Lcur, 1e-12)
        accepted = False
        for _ in range(16):
            xn = _proj(y - step * g, S)
            fx = _smooth_value(xn, alpha)
            dd = xn - y
            rhs = fy + float(np.dot(g, dd)) + 0.5 * (1.0 / step) * float(np.dot(dd, dd))
            if fx <= rhs + 1e-10:
                accepted = True; break
            step *= 0.5
        Lcur = 1.0 / max(step, 1e-12)
        if accepted:
            hp = h; h = xn; fh = fx; stale = 0
            Lcur = max(Lcur * 0.8, 1.0)
            if it % 5 == 0:
                c = _c5_direct(h)
                if c < best_c:
                    best_c = c; best_h = h.copy()
        else:
            stale += 1; hp = h; y = h; t = 1.0; fy = fh
            Lcur = max(Lcur * 0.5, 1.0)
            if stale > 12:
                break
    c = _c5_direct(h)
    if c < best_c:
        best_c = c; best_h = h.copy()
    return h, best_h, best_c

def _subgrad_max(h0, max_time, step_init=5e-4):
    n = len(h0)
    S = n * 0.5
    h = h0.copy()
    best_c = _c5_direct(h)
    best_h = h.copy()
    t0 = time.time()
    step = step_init
    stuck = 0
    for it in range(1000000):
        if time.time() - t0 > max_time:
            break
        P, Nv, mx = _pn_arrays(h)
        tol = max(1e-10, 5.0 * (2.0 / n))
        wP = np.zeros(n)
        wN = np.zeros(n)
        cnt = 0
        if P[0] >= mx - tol:
            wP[0] = 1.0; cnt += 1
        if n > 1:
            mP = P[1:] >= mx - tol
            if np.any(mP):
                wP[np.nonzero(mP)[0] + 1] = 1.0; cnt += mP.sum()
            mN = Nv[1:] >= mx - tol
            if np.any(mN):
                wN[np.nonzero(mN)[0] + 1] = 1.0; cnt += mN.sum()
        if cnt == 0:
            break
        wP /= cnt
        wN /= cnt
        g = _grad_weights(h, wP, wN)
        hn = _proj(h - step * g, S)
        cn = _c5_direct(hn)
        if cn < best_c - 1e-14:
            best_c = cn; best_h = hn.copy()
            h = hn
            stuck = 0
            step = min(step * 1.02, 0.05)
        else:
            stuck += 1
            step *= 0.95
            if stuck > 500:
                break
    return best_h, best_c

def _local_search(h0, rng, deadline):
    n = len(h0)
    h = h0.copy()
    c = _c5_direct(h)
    best_c = c
    best_h = h.copy()
    tl0 = time.time()
    P, Nv, _ = _pn_arrays(h)
    mx = max(P[0], np.max(P[1:]), np.max(Nv[1:]))
    g = np.zeros(n)
    tol = max(1e-10, 8.0 * 2.0 / n)
    if P[0] >= mx - tol: g += (1.0 - 2.0 * h) * (2.0 / n) / max(1, 1)
    if n > 1:
        wP = np.zeros(n); wN = np.zeros(n)
        cnt = 0
        if P[0] >= mx - tol: wP[0] = 1; cnt += 1
        mP = P[1:] >= mx - tol
        if np.any(mP):
            wP[np.nonzero(mP)[0]+1] = 1; cnt += mP.sum()
        mN = Nv[1:] >= mx - tol
        if np.any(mN):
            wN[np.nonzero(mN)[0]+1] = 1; cnt += mN.sum()
        if cnt > 0:
            wP /= cnt; wN /= cnt
            g = _grad_weights(h, wP, wN)
    order = np.argsort(g)
    q = max(8, n // 6)
    low = order[:q]
    high = order[-q:]
    recompute_at = 1024
    it = 0
    while time.time() < deadline:
        if it >= recompute_at:
            P, Nv, _ = _pn_arrays(h)
            mx = max(P[0], np.max(P[1:]), np.max(Nv[1:]))
            g = np.zeros(n)
            tol2 = max(1e-10, 8.0 * 2.0 / n)
            wP = np.zeros(n); wN = np.zeros(n)
            cnt = 0
            if P[0] >= mx - tol2: wP[0] = 1; cnt += 1
            if n > 1:
                mP = P[1:] >= mx - tol2
                if np.any(mP): wP[np.nonzero(mP)[0]+1] = 1; cnt += mP.sum()
                mN = Nv[1:] >= mx - tol2
                if np.any(mN): wN[np.nonzero(mN)[0]+1] = 1; cnt += mN.sum()
            if cnt > 0:
                wP /= cnt; wN /= cnt
                g = _grad_weights(h, wP, wN)
            else:
                g = rng.randn(n) * 1e-6
            order = np.argsort(g)
            low = order[:q]
            high = order[-q:]
            c = _c5_direct(h)
            if c < best_c:
                best_c = c; best_h = h.copy()
            recompute_at = it + 1024
        frac = min(1.0, (time.time() - tl0) / max(1e-9, deadline - tl0))
        base = max(1e-12, 3.0e-5 * (1e-4) ** frac)
        if rng.random() < 0.65:
            i = int(low[rng.randint(len(low))])
            j = int(high[rng.randint(len(high))])
        else:
            i = int(rng.randint(n))
            j = int(rng.randint(n))
        if i != j:
            s = 1 if g[i] <= g[j] else (-1 if g[j] < g[i] else (1 if rng.random() < 0.5 else -1))
            avail = min(1.0 - h[i], h[j]) if s > 0 else min(h[i], 1.0 - h[j])
            if avail > 1e-14:
                mult = 0.3 + 1.5 * rng.random()
                if rng.random() < 0.02:
                    mult *= 10.0
                d = s * min(avail, base * mult)
                h[i] += d; h[j] -= d
                cn = _c5_direct(h)
                if cn < c - 1e-15:
                    c = cn
                    if c < best_c:
                        best_c = c; best_h = h.copy()
                else:
                    h[i] -= d; h[j] += d
        it += 1
    return best_h, best_c

def _resample(h, m):
    n = len(h)
    if n == m:
        return _repair(h, n)
    if n % m == 0:
        out = h.reshape(m, n // m).mean(axis=1)
    else:
        x0 = (np.arange(n) + 0.5) * (2.0 / n)
        x1 = (np.arange(m) + 0.5) * (2.0 / m)
        out = np.interp(x1, x0, h)
    return _proj(out, m * 0.5)

def _run_alphas(h0, alphas, elapsed_budget, deadline, L=10.0):
    hcur = h0.copy()
    best_h = h0.copy()
    best_c = _c5_direct(best_h)
    per = elapsed_budget / max(1, len(alphas))
    for a in alphas:
        rem = deadline - time.time()
        if rem < 3.0 or per <= 0:
            break
        hf, bh, bc = _fista(hcur, a, min(per, rem), L, True)
        chf = _c5_direct(hf)
        chb = _c5_direct(bh)
        if bc < best_c:
            best_c = bc; best_h = bh
        if chf < best_c:
            best_c = chf; best_h = hf
        hcur = hf if chf <= chb else bh
    return best_h, best_c, hcur

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(560, 0.5)
    n0 = len(h0)
    h0 = _repair(h0, n0)
    best = h0.copy()
    best_c = _c5_direct(best)
    for hh in (1.0 - h0, h0[::-1], 0.5 + 0.5*(h0 - h0[::-1]), 0.5 + 0.5*(h0 + h0[::-1] - 1.0)):
        hh = _proj(hh, n0 * 0.5)
        c = _c5_direct(hh)
        if c < best_c:
            best = hh.copy(); best_c = c
    # Phase 1: quick FISTA
    end1 = t0 + min(65, 0.06 * budget_s)
    if end1 > time.time():
        b, c, _ = _run_alphas(best, [15, 60, 250, 1000, 4000], end1 - time.time(), end1)
        if c < best_c:
            best = b; best_c = c
    # Phase 2: multi-resolution
    end2 = t0 + min(max(0.0, budget_s - 350.0), (end1 - t0) + 200.0)
    if budget_s > 300 and end2 > time.time():
        for f in (2, 4):
            m = n0 // f
            if m < 32:
                continue
            if time.time() >= end2 - 5.0:
                break
            slot = min(40.0, end2 - time.time() - 2.0)
            if slot < 5:
                break
            hm = _resample(best, m)
            b, c, _ = _run_alphas(hm, [30, 200, 1500, 8000], slot, time.time() + slot)
            if c < best_c:
                best = b; best_c = c
            if len(best) != n0:
                hu = _resample(best, n0)
                cu = _c5_direct(hu)
                if cu < best_c:
                    best = hu; best_c = cu
    # Phase 3: main FISTA at full resolution
    hmain = best if len(best) == n0 else _resample(best, n0)
    end3 = t0 + budget_s - 250.0
    if end3 > time.time() + 10:
        b, c, _ = _run_alphas(
            hmain,
            [30, 150, 600, 2500, 10000, 50000, 250000, 1000000],
            end3 - time.time(), end3, 10.0
        )
        if c < best_c:
            best = b; best_c = c
        if len(best) != n0:
            endb = t0 + budget_s - 180.0
            if endb > time.time() + 5:
                b2, c2, _ = _run_alphas(best, [2000, 10000, 50000, 250000], endb - time.time(), endb)
                if c2 < best_c:
                    best = b2; best_c = c2
    # Phase 4: subgradient on max
    end4 = t0 + budget_s - 120.0
    if end4 > time.time() + 5:
        best, best_c = _subgrad_max(best, end4 - time.time(), 3e-4)
    # Phase 5: local search
    end5 = t0 + budget_s - 2.0
    if end5 > time.time():
        best, best_c = _local_search(best, rng, end5)
    # Final variants
    variants = [best, 1.0 - best, best[::-1],
                0.5 + 0.5*(best - best[::-1]), 0.5 + 0.5*(best + best[::-1] - 1.0)]
    for vv in variants:
        h2 = _proj(vv, len(vv) * 0.5)
        c = _c5_direct(h2)
        if c < best_c:
            best = h2; best_c = c
    best = _repair(best, len(best))
    best_c = _c5_direct(best)
    return best, best_c, len(best)
