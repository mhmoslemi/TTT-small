        if time.time() > deadline:

        active = np.where(c >= m - tol)[0]
        dx = 2.0 / n
        g = np.zeros(n)
        for k in active:
            sh = (n - 1) - k
            if sh >= 0:
                g[:n - sh] += (1.0 - h[sh:])
                if sh > 0:
                    g[sh:] -= h[:-sh]
            else:
                sp = -sh
                g[sp:] += (1.0 - h[:-sp])
                g[:n - sp] -= h[sp:]
        g *= dx / max(len(active), 1)
        v = beta * v + (1.0 - beta) * g
        hn = _proj_simplex(h - step * v, target)
        fn = _correl(hn).max()
        if fn < f - 1e-15:
            h, f = hn, fn
            step = min(step * 1.02, step0 * 25.0)
            if f < best_f:
                best_f, best_h = f, h.copy()
        else:
            step *= 0.5
            hn = _proj_simplex(h - step * v, target)
            f = _correl(hn).max()
            h = hn
            if f < best_f:
                best_f, best_h = f, h.copy()
    return _repair(best_h, n), best_f

def _anneal(h, niter, rng, deadline, blocks, T0, T1):
    n = len(h)
    c5c = _eval(h)
    bh, bc = h.copy(), c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0 + bs].copy()
        d = 0.006 * max(T, 1e-9) ** 0.5 * rng.randn(bs)
        d -= d.mean()
        seg = np.clip(seg + d, 0, 1)
        hn = h.copy()
        hn[i0:i0 + bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h, c5c = hn, c5n
            if c5n < bc:
                bc, bh = c5n, h.copy()
    return bh, bc

def _polish(h, niter, rng, deadline, scale):
    nc = len(h)
    c5c = _eval(h)
    bh, bc = h.copy(), c5c
    for _ in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(nc)
        j = (i + 1 + rng.randint(0, 10)) % nc
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < bc:
                    bc, bh = c5n, h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return bh, bc

def _slsqp(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    try:
        r = minimize(lambda x: _smax_obj(x, alpha, topk), h,
                     jac=lambda x: _grad_smax(x, alpha, topk)[0],
                     method='SLSQP', bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
                                    'jac': lambda x: np.ones(n)}],
                     options={'maxiter': maxit, 'ftol': 1e-15})
        return _repair(r.x, n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h, best_c5 = h.copy(), _eval(h)

    # Phase 1: SLSQP rapid refinement
    dl1 = min(deadline, t0 + 100)
    for alpha, tk, mi in [(6000, 25, 350), (25000, 12, 300)]:
        if time.time() > dl1:
            break
        h = _slsqp(h, alpha, tk, mi, dl1)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5, best_h = c5v, h.copy()

    # Phase 2: FISTA soft-max (accelerated, multi-scale)
    dl2 = min(deadline, t0 + 360)
    fista_cfgs = [(15000, 8, 3000), (60000, 5, 2500), (150000, 4, 2000)]
    for alpha, tk, ns in fista_cfgs:
        if time.time() > dl2:
            break
        h2, c5v = _fista(best_h.copy(), alpha, tk, ns, 0.003, dl2)
        if c5v < best_c5:
            best_c5, best_h = c5v, h2.copy()

    # Phase 3: Nesterov exact subgradient
    dl3 = min(deadline, t0 + 520)
    for tol, step0, mom, ns in [(0.004, 0.005, 0.84, 3500), (0.001, 0.0015, 0.90, 3000)]:
        if time.time() > dl3:
            break
        h2, c5v = _nesterov_exact(best_h.copy(), tol, ns, step0, dl3, beta=mom)
        if c5v < best_c5:
            best_c5, best_h = c5v, h2.copy()

    # Phase 4: Structured perturbations + FISTA re-optimization
    dl4 = min(deadline, t0 + 720)
    n = len(best_h)
    freqs = [3, 5, 8, 13, 21, 34]
    for freq in freqs:
        if time.time() > dl4:
            break
        d = 0.010 * np.sin(2.0 * np.pi * freq * np.arange(n) / n)
        d += 0.004 * rng.randn(n)
        d -= d.mean()
        hn = _repair(np.clip(best_h + d, 0, 1), n)
        h2, c5v = _fista(hn, 40000, 6, 1000, 0.002, dl4)
        if c5v < best_c5:
            best_c5, best_h = c5v, h2.copy()

    # Phase 5: Aggressive block annealing
    dl5 = min(deadline, t0 + 920)
    blocks = [2, 3, 5, 8, 13, 21]
    best_h, best_c5 = _anneal(best_h.copy(), 100000, rng, dl5, blocks, 0.008, 1e-11)

    # Phase 6: Post-anneal FISTA
    dl6 = min(deadline, t0 + budget_s - 40)
    h2, c5v = _fista(best_h.copy(), 100000, 4, 2000, 0.001, dl6)
    if c5v < best_c5:
        best_c5, best_h = c5v, h2.copy()

    # Phase 7: Fine Nesterov
    dl7 = min(deadline, t0 + budget_s - 20)
    h2, c5v = _nesterov_exact(best_h.copy(), 0.0004, 2000, 0.0004, dl7, beta=0.93)
    if c5v < best_c5:
        best_c5, best_h = c5v, h2.copy()

    # Phase 8: Coordinate polish
    rem = max(0, deadline - time.time())
    n_pol = min(int(rem / 0.0012), 80000)
    best_h, best_c5 = _polish(best_h, n_pol, rng, deadline, 8e-7)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
