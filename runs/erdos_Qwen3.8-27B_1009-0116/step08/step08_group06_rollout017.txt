def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if h.sum() < 1e-12:
        return np.full(n, 0.5)
    if abs(h.sum() - target) > 1e-10:
        lo, hi = -3.0, 3.0
        for _ in range(80):
            mid = (lo + hi) * 0.5
            if np.clip(h - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo + hi) * 0.5, 0, 1)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:

                    h[idx] -= a
                    diff += a

    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)

    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

        return _lse_obj(x, self.alpha, self.n)
    def con_fun(self, x):
        return x.sum() - self.target
    def con_jac(self, x):
        return np.ones(self.n)

def _slsqp(h, alpha, maxiter, ftol, target):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    ctx = _Ctx(alpha, n, target)
    cons = [{'type': 'eq', 'fun': ctx.con_fun, 'jac': ctx.con_jac}]
    try:

                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if h.sum() < 1e-12:
        return np.full(n, 0.5)
    if abs(h.sum() - target) > 1e-10:
        lo, hi = -3.0, 3.0
        for _ in range(80):
            mid = (lo + hi) * 0.5
            if np.clip(h - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo + hi) * 0.5, 0, 1)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _pair_dc(h, p, q, n, dc_buf):
    dc = dc_buf
    dc.fill(0.0)
    hi_p = min(p + n - 1, 2 * n - 2)
    dc[p:hi_p + 1] += (1.0 - 2.0 * h[:hi_p - p + 1])
    hi_q = min(q + n - 1, 2 * n - 2)
    dc[q:hi_q + 1] += (2.0 * h[:hi_q - q + 1] - 1.0)
    return dc

def _optimal_d(c, dc, d_lo, d_hi):
    for _ in range(30):
        if d_hi - d_lo < 1e-14:
            break
        d_mid = (d_lo + d_hi) * 0.5
        vals = c + dc * d_mid
        fmax = vals.max()
        active = vals >= fmax - 1e-10
        if dc[active].min() > 1e-12:
            d_hi = d_mid
        elif dc[active].max() < -1e-12:
            d_lo = d_mid
        else:
            break
    return (d_lo + d_hi) * 0.5

def _lse_tmp(x, alpha, n):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s - s.max()))) / alpha

class _Ctx:
    def __init__(self, alpha, n, target):
        self.alpha = alpha
        self.n = n
        self.target = target
    def obj(self, x):
        return _lse_tmp(x, self.alpha, self.n)
    def con_fun(self, x):
        return x.sum() - self.target
    def con_jac(self, x):
        return np.ones(self.n)

def _slsqp(h, alpha, maxiter, ftol, target):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    ctx = _Ctx(alpha, n, target)
    cons = [{'type': 'eq', 'fun': ctx.con_fun, 'jac': ctx.con_jac}]
    try:
        res = minimize(ctx.obj, h, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _swap_phase(h, c, n, dc_buf, n_swaps, rng, t_end, bias_extreme=True):
    c_max = c.max()
    stalled = 0
    for _ in range(n_swaps):
        if time.time() > t_end:
            break
        if bias_extreme:
            r = rng.random()
            if r < 0.5:
                p = int(np.argmax(np.where(rng.random(size=n) < 0.3, h, 0.0)))
            else:
                p = int(np.argmin(np.where(rng.random(size=n) < 0.3, 1.0 - h, 1.0)))
        else:
            p = rng.randint(n)
        q = (p + 1 + rng.randint(0, n - 2)) % n
        d_lo = max(-h[p], h[q] - 1.0)
        d_hi = min(1.0 - h[p], h[q])
        if d_hi - d_lo < 1e-14:
            continue
        _pair_dc(h, p, q, n, dc_buf)
        dc = dc_buf
        d_opt = _optimal_d(c, dc, d_lo, d_hi)
        if abs(d_opt) < 1e-14:
            stalled += 1
            continue
        h[p] += d_opt
        h[q] -= d_opt
        c += d_opt * dc
        c_max_new = c.max()
        if c_max_new > c_max + 1e-12:
            h[p] -= d_opt
            h[q] += d_opt
            c -= d_opt * dc
        else:
            c_max = c_max_new
            stalled = 0
        if ((c_max * 100000) % 1000) < 1:
            c = _corr(h, n)
            c_max = c.max()
    return h, c, c_max, stalled

def _local_search(h, rng, t_end, n):
    c5c = _eval(h)
    best_h, best_c = h.copy(), c5c
    while time.time() < t_end:
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 3)) % n
        k = (j + 1 + rng.randint(0, 3)) % n
        scale = 2e-6
        d1 = scale * rng.randn()
        d2 = scale * rng.randn()
        ni, nj, nk = h[i] + d1, h[j] + d2, h[k] - d1 - d2
        if 0 < ni < 1 and 0 < nj < 1 and 0 < nk < 1:
            h[i], h[j], h[k] = ni, nj, nk
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
                if c5n < best_c:
                    best_c, best_h = c5n, h.copy()
            else:
                h[i], h[j], h[k] = ni - d1, nj - d2, nk + d1 + d2
    return best_h, best_c
