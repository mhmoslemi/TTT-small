def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a

def _flip_search(h, n_iter, rng):
    """Binary flip-based search: swap a 0→1 at i and 1→0 at j"""
    n = len(h)
    b = _to_binary(h, n)
    best_b = b.copy()
    best_c = _eval(b)
    ones = np.where(b == 1.0)[0]
    zeros = np.where(b == 0.0)[0]
    c5 = best_c
    
    for _ in range(n_iter):
        # Random pair flip
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        b[i], b[j] = 1.0, 0.0
        c_new = _eval(b)
        if c_new < c5:
            c5 = c_new
            if c_new < best_c:
                best_c = c_new
                best_b = b.copy()
            ones = np.where(b == 1.0)[0]
            zeros = np.where(b == 0.0)[0]
        else:
            b[i], b[j] = 0.0, 1.0
    return best_b, best_c

def _sa_binary(h, n_iter, rng, T0, Tend):
    """Simulated annealing on binary representation"""
    n = len(h)
    b = _to_binary(h, n)
    best_b = b.copy()
    best_c = _eval(b)
    cur_c = best_c
    
    T = T0
    cooling = (Tend / T0) ** (1.0 / max(n_iter, 1))
    
    for it in range(n_iter):
        T = T0 * cooling ** it
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        if b[i] == b[j]:
            continue
        # Swap: b[i] and b[j] have opposite values
        b[i], b[j] = b[j], b[i]
        c_new = _eval(b)
        delta = c_new - cur_c
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            cur_c = c_new
            if c_new < best_c:
                best_c = c_new
                best_b = b.copy()
        else:
            b[i], b[j] = b[j], b[i]
    return best_b, best_c

def _downsample(h, factor):
    n = len(h)
    m = n // factor
    return h[:m*factor].reshape(m, factor).mean(axis=1).copy()

def _upsample(h, factor):

            g[sh:] -= h[:n-sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:n-sp])
        g[:n-sp] -= h[sp:]
    return g * dx

def _slsqp(h, alpha, maxiter, top_k):
    n = len(h)
    dx = 2.0 / n
    def obj(x):
        c = _fft_corr(x) * dx
        idx = np.argsort(c)[-top_k:] if 0 < top_k < len(c) else np.arange(len(c))
        cs = c[idx]; m = cs.max()
        s = np.clip(alpha*(cs-m), -400, 400); s -= s.max()
        return m + np.log(np.exp(s).sum())/alpha
    def grad(x):
        c = _fft_corr(x) * dx
        idx = np.argsort(c)[-top_k:] if 0 < top_k < len(c) else np.arange(len(c))
        cs = c[idx]; m = cs.max()
        s = np.clip(alpha*(cs-m), -400, 400); s -= s.max()
        w = np.exp(s); w /= w.sum()
        g = np.zeros(n)
        for ii in range(len(idx)):
            if w[ii] > 1e-15:
                g += w[ii] * _grad_lag(int(idx[ii]), x, n)
        return g
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return h

def _proj_grad(h, alpha, steps, step0, top_k, momentum):
    n = len(h)
    v = np.zeros(n)
    step = step0
    cur_c = _eval(h)
    for it in range(steps):
        c = _corr_vals(h)
        active = np.argsort(c)[-top_k:]
        c_a = c[active]; m_a = c_a.max()
        s = np.clip(alpha*(c_a-m_a), -400, 400); s -= s.max()
        w = np.exp(s); w /= w.sum()
        g = np.zeros(n)
        for ii in range(len(active)):
            if w[ii] > 1e-15:
                g += w[ii] * _grad_lag(int(active[ii]), h, n)
        v = momentum*v + (1.0-momentum)*g
        taken = False
        for _ in range(12):
            h_new = _proj(h - step*v, n*0.5)
            c_new = _eval(h_new)
            if c_new < cur_c - 1e-15:
                h = h_new; cur_c = c_new
                step = min(step*1.12, step0*10.0); taken = True; break
            step *= 0.35
        if not taken:
            step = min(step*1.5, step0*10.0)
        if it % 1000 == 0: v *= 0.2
    return h, cur_c

def _lag_focus(h, n_steps, k_target):
    """Minimize a single correlation lag via projected gradient"""
    n = len(h)
    cur_c = _eval(h)
    step = 0.003
    v = np.zeros(n)
    for it in range(n_steps):
        g = _grad_lag(k_target, h, n)
        v = 0.85*v + 0.15*g
        taken = False
        for _ in range(10):
            h_new = _proj(h - step*v, n*0.5)
            c_new = _eval(h_new)
            if c_new < cur_c - 1e-15:
                h = h_new; cur_c = c_new
                step = min(step*1.1, 0.03); taken = True; break
            step *= 0.4
        if not taken: step = min(step*1.4, 0.03)
    return h, cur_c

def _to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n); b[idx[:target]] = 1.0
    return b

def _flip_search(h, n_iter, rng):
    n = len(h)
    b = _to_binary(h, n)
    best_b = b.copy(); best_c = _eval(b); cur_c = best_c
    ones = np.where(b == 1.0)[0]; zeros = np.where(b == 0.0)[0]
    for _ in range(n_iter):
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        b[i], b[j] = 1.0, 0.0
        c_new = _eval(b)
        if c_new < cur_c:
            cur_c = c_new
            if c_new < best_c:
                best_c = c_new; best_b = b.copy()
            ones = np.where(b == 1.0)[0]; zeros = np.where(b == 0.0)[0]
        else:
            b[i], b[j] = 0.0, 1.0
    return best_b, best_c

def _sa_binary(h, n_iter, rng, T0, Tend):
    n = len(h)
    b = _to_binary(h, n)
    best_b = b.copy(); best_c = _eval(b); cur_c = best_c
    cool = (Tend/T0)**(1.0/max(n_iter, 1))
    T = T0
    for it in range(n_iter):
        T = T0 * cool**it
        i = rng.randint(n); j = rng.randint(n)
        if i == j or b[i] == b[j]: continue
        b[i], b[j] = b[j], b[i]
        c_new = _eval(b)
        d = c_new - cur_c
        if d < 0 or rng.rand() < np.exp(-d/max(T, 1e-12)):
            cur_c = c_new
            if c_new < best_c: best_c = c_new; best_b = b.copy()
        else:
            b[i], b[j] = b[j], b[i]
    return best_b, best_c

def _downsample(h, factor):
    n = len(h); m = n // factor
    return h[:m*factor].reshape(m, factor).mean(axis=1).copy()

def _upsample(h, factor):
    return np.repeat(h, factor)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    def _el(): return time.time() - t0

    n = len(initial_h_values)
    best_h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_c5 = _eval(best_h)

    # Phase 1: Coarse-to-fine SLSQP (25s)
    for ratio in [8, 4, 2]:
        if _el() > 25: break
        nc = max(24, n // ratio)
        hc = _repair(_downsample(best_h, n // nc), nc)
        for alpha in [5000, 50000]:
            if _el() > 25: break
            hc = _slsqp(hc, alpha, 80, 0)
        h_up = _repair(_upsample(hc, n // nc), n)
        if _eval(h_up) < best_c5: best_c5 = _eval(h_up); best_h = h_up

    # Phase 2: FFT projected gradient (120s)
    configs = [
        (2000, 4000, 0.005, 8, 0.7),
        (10000, 4000, 0.003, 6, 0.75),
        (50000, 3000, 0.002, 4, 0.8),
        (200000, 3000, 0.001, 3, 0.85),
        (1000000, 2500, 0.0005, 2, 0.9),
        (3000000, 2000, 0.0002, 2, 0.9),
        (10000000, 1500, 0.0001, 1, 0.92),
    ]
    for alpha, steps, s0, tk, mu in configs:
        if _el() > 150: break
        h_new, c_new = _proj_grad(best_h.copy(), alpha, steps, s0, tk, mu)
        if c_new < best_c5: best_c5 = c_new; best_h = h_new

    # Phase 3: Lag focus - minimize worst lag directly (80s)
    if _el() < budget_s - 500:
        for _ in range(8):
            if _el() > 175: break
            _, k_w = _eval_worst(best_h)
            h_new, c_new = _lag_focus(best_h.copy(), 5000, k_w)
            if c_new < best_c5: best_c5 = c_new; best_h = h_new

    # Phase 4: Binary flip search (250s)
    if _el() < budget_s - 280:
        _t = _el()
        for _ in range(50): _eval(best_h)
        eps = max((_el()-_t)/50.0, 1e-7)
        tl = budget_s - 280 - _el()
        nf = min(int(tl * 0.85 / eps), 80000000)
        b, cb = _flip_search(best_h, nf, rng)
        if cb < best_c5: best_c5 = cb; best_h = b.copy()
        # Continue from binary with gradient
        h_new, c_new = _proj_grad(best_h.copy(), 1000000, 2000, 0.0003, 2, 0.9)
        if c_new < best_c5: best_c5 = c_new; best_h = h_new

    # Phase 5: SLSQP polish (60s)
    if _el() < budget_s - 220:
        for alpha, tk, mit in [(50000, 4, 120), (200000, 2, 100), (1000000, 1, 80)]:
            if _el() > budget_s - 180: break
            h_new = _slsqp(best_h, alpha, mit, tk)
            c_new = _eval(h_new)
            if c_new < best_c5: best_c5 = c_new; best_h = h_new

    # Phase 6: More lag focus (100s)
    if _el() < budget_s - 120:
        for _ in range(12):
            if _el() > budget_s - 100: break
            _, k_w = _eval_worst(best_h)
            h_new, c_new = _lag_focus(best_h.copy(), 4000, k_w)
            if c_new < best_c5: best_c5 = c_new; best_h = h_new

    # Phase 7: Final binary flips + continuous (100s)
    if _el() < budget_s - 30:
        b = _to_binary(best_h, n)
        tl = budget_s - 30 - _el()
        _t = _el()
        for _ in range(100): _eval(b)
        eps = max((_el()-_t)/100.0, 1e-7)
        nf = min(int(tl * 0.6 / eps), 80000000)
        b, cb = _flip_search(best_h, nf, rng)
        if cb < best_c5: best_c5 = cb; best_h = b.copy()
        if _el() < budget_s - 5:
            h_new, c_new = _proj_grad(best_h.copy(), 5000000, 2000, 0.0001, 1, 0.92)
            if c_new < best_c5: best_c5 = c_new; best_h = h_new

    # Final ultra-fine polish
    if _el() < budget_s - 2:
        h_new, c_new = _proj_grad(best_h.copy(), 10000000, 1000, 0.00005, 1, 0.95)
        if c_new < best_c5: best_c5 = c_new; best_h = h_new

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
