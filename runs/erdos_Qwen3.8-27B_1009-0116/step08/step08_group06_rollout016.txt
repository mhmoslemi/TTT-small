                break
    return h


def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)


def _corr_vec(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)


def _topk_grad(h, k_top, temp):
    n = len(h)
    c = _corr_vec(h)
    idx = np.argsort(c)[-k_top:]
    c_sel = c[idx]
    w_raw = np.exp(temp * (c_sel - c_sel.max()))
    w_raw /= w_raw.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for i in range(k_top):
        k = idx[i]
        sh = n - 1 - k
        wk = w_raw[i] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g, float(c.max())


class _Ctx:
    def __init__(self, alpha, n, target):
        self.alpha = alpha
        self.n = n
        self.target = target
        self.dx = 2.0 / n

    def fun(self, x):
        c = np.correlate(x, 1.0 - x, mode='full') * self.dx
        m = c.max()
        s = np.clip(self.alpha * (c - m), -500, 500)
        return m + float(np.log(np.sum(np.exp(s - s.max())))) / self.alpha

    def jac(self, x):
        c = np.correlate(x, 1.0 - x, mode='full') * self.dx
        m = c.max()
        s = np.clip(self.alpha * (c - m), -500, 500)
        w = np.exp(s - s.max())
        w /= w.sum()
        g = np.zeros(self.n)
        for k in range(2 * self.n - 1):
            if w[k] < 1e-15:
                continue
            sh = self.n - 1 - k
            wk = w[k] * self.dx
            if sh >= 0:
                g[:self.n - sh] += wk * (1.0 - x[sh:])
                if sh > 0:
                    g[sh:] -= wk * x[:self.n - sh]
            else:
                sp = -sh
                g[sp:] += wk * (1.0 - x[:self.n - sp])
                g[:self.n - sp] -= wk * x[sp:]
        return g

    def cfun(self, x):
        return x.sum() - self.target

    def cjac(self, x):
        return np.ones(self.n)


def _slsqp_opt(h, alpha, maxiter, ftol, target, t_end):
    n = len(h)
    ctx = _Ctx(alpha, n, target)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': ctx.cfun, 'jac': ctx.cjac}]
    try:
        res = minimize(ctx.fun, h, jac=ctx.jac, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)


def _nesterov_adaptive(h, t_end, k_start, k_end, temp_start, temp_end, step0):
    n = len(h)
    f = _eval(h)
    v = np.zeros(n)
    step = step0
    t0 = time.time()
    it = 0
    no_improve = 0
    while time.time() < t_end:
        frac = min((time.time() - t0) / max(t_end - t0, 1e-6), 1.0)
        k_top = int(k_start + (k_end - k_start) * frac)
        k_top = max(k_top, 5)
        temp = temp_start * (temp_end / temp_start) ** frac
        g, cmax = _topk_grad(h, k_top, temp)
        mu = 0.85
        h_new = _repair(h + mu * (h - (h + step * (mu * v + (1 - mu) * g))) - step * (mu * v + (1 - mu) * g), n)
        # Simplified: just do standard momentum
        h_new = _repair(h - step * (mu * v + (1 - mu) * g), n)
        f_new = _eval(h_new)
        if f_new < f:
            v = mu * v + (1 - mu) * g
            h = h_new
            f = f_new
            no_improve = 0
            step = min(step * 1.03, step0 * 30)
        else:
            step *= 0.5
            no_improve += 1
            if no_improve > 200:
                # Random kick
                kick = 0.005 * np.random.randn(n)
                h = _repair(h + kick, n)
                f = _eval(h)
                v = np.zeros(n)
                step = step0
                no_improve = 0
        it += 1
    return h, f


def _swap_search(h, n_swaps, t_end, rng):
    n = len(h)
    best_h = h.copy()
    best_f = _eval(h)
    cur_f = best_f
    for _ in range(n_swaps):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        old_i, old_j = h[i], h[j]
        h[i], h[j] = old_j, old_i
        c = _corr_vec(h)
        f_new = float(c.max())
        if f_new < cur_f - 1e-14:
            cur_f = f_new
            if f_new < best_f:
                best_f = f_new
                best_h = h.copy()
        else:
            h[i], h[j] = old_i, old_j
    return best_h, best_f


def _two_point_search(h, n_iters, t_end, rng):
    n = len(h)
    best_h = h.copy()
    best_f = _eval(h)
    cur_f = best_f
    for _ in range(n_iters):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 4)) % n
        d = 5e-6 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            f_new = _eval(h)
            if f_new < cur_f - 1e-15:
                cur_f = f_new
                if f_new < best_f:
                    best_f = f_new
                    best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_f


def _three_point_search(h, n_iters, t_end, rng):
    n = len(h)
    best_h = h.copy()
    best_f = _eval(h)
    cur_f = best_f
    for _ in range(n_iters):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = (i + rng.randint(2, 8)) % n
        k = (i + rng.randint(9, 20)) % n
        d1 = 1.5e-5 * rng.randn()
        d2 = 1.5e-5 * rng.randn()
        ni, nj, nk = h[i] + d1, h[j] + d2, h[k] - d1 - d2
        if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= nk <= 1:
            oi, oj, ok = h[i], h[j], h[k]
            h[i], h[j], h[k] = ni, nj, nk
            f_new = _eval(h)
            if f_new < cur_f - 1e-15:
                cur_f = f_new
                if f_new < best_f:
                    best_f = f_new
                    best_h = h.copy()
            else:
                h[i], h[j], h[k] = oi, oj, ok
    return best_h, best_f


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    t_end_global = t0 + budget_s
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    target = n / 2.0
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    def _try(h_new):
        nonlocal best_h, best_c5
        c = _eval(h_new)
        if c < best_c5:
            best_c5 = c
            best_h = h_new.copy()
        return c

    # Phase 1: Nesterov adaptive (coarse to fine k)
    t1 = t0 + budget_s * 0.22
    h, _ = _nesterov_adaptive(h, t1, 10, 80, 0.5, 50.0, 0.003)
    _try(h)

    # Phase 2: SLSQP mid-alpha
    t2 = t0 + budget_s * 0.38
    for alpha in [2000, 20000]:
        if time.time() > t2 - 5:
            break
        h_s = _slsqp_opt(best_h.copy(), alpha, 800, 1e-15, target, t2)
        _try(h_s)

    # Phase 3: Nesterov high-k (fuller max approximation)
    t3 = t0 + budget_s * 0.55
    h, _ = _nesterov_adaptive(best_h.copy(), t3, 50, 300, 2.0, 200.0, 0.001)
    _try(h)

    # Phase 4: SLSQP high-alpha
    t4 = t0 + budget_s * 0.70
    for alpha in [200000, 2000000]:
        if time.time() > t4 - 10:
            break
        h_s = _slsqp_opt(best_h.copy(), alpha, 600, 1e-16, target, t4)
        _try(h_s)

    # Phase 5: Swap-based search (explores discrete structure)
    t5 = t0 + budget_s * 0.82
    h_sw, _ = _swap_search(best_h.copy(), int(2e9), t5, rng)
    _try(h_sw)

    # Phase 6: Nesterov very fine (all lags)
    t6 = t0 + budget_s * 0.90
    h, _ = _nesterov_adaptive(best_h.copy(), t6, 500, 1119, 50.0, 500.0, 0.0005)
    _try(h)

    # Phase 7: SLSQP ultra-high alpha
    if time.time() < t_end_global - 80:
        h_s = _slsqp_opt(best_h.copy(), 50000000, 500, 1e-17, target, t_end_global - 60)
        _try(h_s)

    # Phase 8: 3-point local search
    t8 = t_end_global - 20
    if time.time() < t8:
        h3, c3 = _three_point_search(best_h.copy(), 1000000, t8, rng)
        _try(h3)

    # Phase 9: 2-point ultra-fine
    if time.time() < t_end_global - 2:
        h2, c2 = _two_point_search(best_h.copy(), 2000000, t_end_global - 0.5, rng)
        _try(h2)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
