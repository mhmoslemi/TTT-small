def _repair(h, n):
    # Same as before but slightly optimized
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if h.sum() < 1e-12:

    if abs(h.sum() - target) > 1e-10:
        lo, hi = -3.0, 3.0
        for _ in range(60):
            mid = (lo + hi) * 0.5
            if np.clip(h - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo + hi) * 0.5, 0, 1)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)

                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

    # Compute gradient of f_{s_star}
    g = np.zeros(n)

            else:

        if m <= 0:
            return g
        g[:m] += dx * (1.0 - h[s:s + m])
        g[s:s + m] -= dx * h[:m]
    else:
        sp = -s
        m = n - sp
        if m <= 0:
            return g
        g[sp:sp + m] += dx * (1.0 - h[:m])
        g[:m] -= dx * h[sp:sp + m]
    return g

def _proj_grad(h, step0, n, t_end, max_iter=8000):
    f = _eval(h)
    best_h, best_f = h.copy(), f
    step = step0
    for _ in range(max_iter):
        if time.time() > t_end:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        k_star = int(np.argmax(c))
        s = k_star - (n - 1)
        g = _grad_shift(h, s, n)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        h_new = _repair(h - step * g / gn, n)
        f_new = _eval(h_new)
        if f_new < best_f:
            best_f, best_h = f_new, h_new.copy()
            step = min(step * 1.03, 0.05)
        else:
            step *= 0.6
            if step < 1e-13:
                h = _repair(h + 1e-4 * np.random.RandomState(int(time.time()*1e6)%100000).randn(n), n)
                step = step0
    return best_h, best_f

def _topk_cycling(h, k_top, step0, n, t_end, max_iter=5000):
    best_h, best_f = h.copy(), _eval(h)
    step = step0
    for _ in range(max_iter):
        if time.time() > t_end:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        top_k = np.argsort(c)[-k_top:]
        g = np.zeros(n)
        for k in top_k:
            s = k - (n - 1)
            g += _grad_shift(h, s, n)
        g /= k_top
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        h_new = _repair(h - step * g / gn, n)
        f_new = _eval(h_new)
        if f_new < best_f:
            best_f, best_h = f_new, h_new.copy()
            step = min(step * 1.02, 0.03)
        else:
            step *= 0.7
            if step < 1e-13:
                step = step0 * 0.5
    return best_h, best_f

def _multipoint_search(h, n, rng, t_end, npts, step0, max_iter=30000):
    c5c = _eval(h)
    best_h, best_c = h.copy(), c5c
    step = step0
    for _ in range(max_iter):
        if time.time() > t_end:
            break
        idx = rng.choice(n, size=npts, replace=False)
        deltas = rng.randn(npts)
        deltas -= deltas.mean()
        sd = np.std(deltas)
        if sd > 1e-15:
            deltas *= step / sd
        old = h[idx].copy()
        new = old + deltas
        if (new >= 0).all() and (new <= 1).all():
            h[idx] = new
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
                if c5n < best_c:
                    best_c, best_h = c5n, h.copy()
                step = min(step * 1.04, 1e-2)
            else:
                h[idx] = old
                step *= 0.92
        else:
            step *= 0.9
    return best_h, best_c

def _fft_perturb(h, n, rng, t_end, n_tries=3000, amp=0.015):
    best_h, best_c = h.copy(), _eval(h)
    for _ in range(n_tries):
        if time.time() > t_end:
            break
        H = np.fft.rfft(h.astype(float))
        pert = rng.randn(len(H)) * amp
        pert[0] = 0.0
        h_new = _repair(np.fft.irfft(H + pert, n=n), n)
        c5n = _eval(h_new)
        if c5n < best_c:
            best_c, best_h = c5n, h_new.copy()
    return best_h, best_c

def _two_point_ultra(h, n, rng, t_end, step0=3e-6):
    c5c = _eval(h)
    best_h, best_c = h.copy(), c5c
    step = step0
    while time.time() < t_end:
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 4)) % n
        d = step * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
                if c5n < best_c:
                    best_c, best_h = c5n, h.copy()
                step = min(step * 1.02, 2e-5)
            else:
                h[i], h[j] = ni - d, nj + d
                step *= 0.97
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    t_global_end = t0 + budget_s
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    target = n / 2.0
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    def _upd(hc):
        nonlocal best_h, best_c5
        hc = _repair(hc, n)
        c = _eval(hc)
        if c < best_c5:
            best_c5, best_h = c, hc.copy()

    # Phase 1: SLSQP moderate alpha
    if time.time() < t0 + budget_s * 0.12:
        _upd(_slsqp(best_h.copy(), 3000, 3000, 1e-16, target))
    if time.time() < t0 + budget_s * 0.22:
        _upd(_slsqp(best_h.copy(), 30000, 2500, 1e-16, target))

    # Phase 2: Projected gradient (exact subgradient)
    t_end2 = t0 + budget_s * 0.38
    if time.time() < t_end2:
        best_h, best_c5 = _proj_grad(best_h.copy(), 0.008, n, t_end2)
        _upd(best_h)

    # Phase 3: SLSQP high alpha
    if time.time() < t0 + budget_s * 0.50:
        _upd(_slsqp(best_h.copy(), 300000, 2000, 1e-16, target))
    if time.time() < t0 + budget_s * 0.60:
        _upd(_slsqp(best_h.copy(), 3000000, 1500, 1e-16, target))

    # Phase 4: Top-k cycling (k=10, then k=30)
    t_end4 = t0 + budget_s * 0.70
    if time.time() < t_end4:
        best_h, best_c5 = _topk_cycling(best_h.copy(), 10, 0.004, n, t_end4)
        _upd(best_h)
    if time.time() < t0 + budget_s * 0.76:
        best_h, best_c5 = _topk_cycling(best_h.copy(), 30, 0.002, n, t0 + budget_s * 0.76)
        _upd(best_h)

    # Phase 5: FFT perturbation
    if time.time() < t0 + budget_s * 0.82:
        best_h, best_c5 = _fft_perturb(best_h.copy(), n, rng, t0 + budget_s * 0.82, 4000, 0.012)
        _upd(best_h)

    # Phase 6: Multi-point local search (4 pts)
    t_end6 = t0 + budget_s * 0.91
    if time.time() < t_end6:
        best_h, best_c5 = _multipoint_search(best_h, n, rng, t_end6, 4, 5e-5, 40000)
        _upd(best_h)

    # Phase 7: Ultra-fine 2-point
    if time.time() < t_global_end - 2:
        best_h, best_c5 = _two_point_ultra(best_h, n, rng, t_global_end - 1, 2e-6)
        _upd(best_h)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
