def _delta_list(dmax):
    fr=[1e-4,1e-3,0.005,0.02,0.06,0.15,0.35,0.65,1.0]
    ds=[]
    for f in fr:
       d=f*dmax

    # also absolute small if dmax large
    if dmax > 1e-3:
       ds.extend([1e-5,1e-4,1e-3])

    out=[]
    for d in ds:
       if d <= 1e-12 or d > dmax*1.0001: continue
       if not out or d > out[-1]*1.12:
          out.append(d)
    if not out or out[-1] < dmax:
       out.append(dmax)
    return out

def _make_pools(g,h,pool_size):
    n=len(h)
    okin = h < 0.999
    okout = h > 0.001
    if okin.sum()==0 or okout.sum()==0: return np.empty(0,int), np.empty(0,int)
    low_idx=np.nonzero(okin)[0]
    high_idx=np.nonzero(okout)[0]
    low_idx=low_idx[np.argsort(g[low_idx])[:pool_size]]
    high_idx=high_idx[np.argsort(g[high_idx])[::-1][:pool_size]]

def _coord_lin(lin, h, r, dr, dx):

    n=len(h); mid=n-1
    lin[mid] += dx*dr*(1.0 - 2.0*h[r])

import numpy as np

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _eval(h):
    return float(np.max(_corr(h)))

def _proj(x, target):
    x = np.asarray(x, dtype=float)
    n = x.size
    target = float(target)
    y = None
    if x.min() >= -1e-12 and x.max() <= 1.0 + 1e-12:
        y = x + (target - float(x.sum())) / n
        if y.min() >= -1e-8 and y.max() <= 1.0 + 1e-8:
            y = np.clip(y, 0.0, 1.0)
        else:
            y = None
    if y is None:
        lo = float(x.min()) - 1.0
        hi = float(x.max()) + 1.0
        for _ in range(60):
            mid = 0.5 * (lo + hi)
            if np.clip(x - mid, 0.0, 1.0).sum() > target:
                lo = mid
            else:
                hi = mid
        y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(y.sum())
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - y)[::-1]
        else:
            order = np.argsort(y)[::-1]
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - y[idx])
                if a <= 0:
                    continue
                y[idx] += a
                diff -= a
            else:
                a = min(-diff, y[idx])
                if a <= 0:
                    continue
                y[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return y

def _default_h(n):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = 0.5 + 0.35 * np.cos(np.pi * x) + 0.15 * np.sin(2.0 * np.pi * x)
    return _proj(h, 0.5 * n)

def _down(h, m):
    n = len(h)
    if m >= n or m <= 0:
        return h.copy()
    x0 = np.linspace(0.0, 2.0, n, endpoint=False)
    x1 = np.linspace(0.0, 2.0, m, endpoint=False)
    return _proj(np.interp(x1, x0, h), 0.5 * m)

def _softmax_w(c, tau, uni=0.0, boost=0.0):
    m = c.size
    mid = (m - 1) // 2
    cm = float(c.max())
    if cm <= 1e-15 or tau <= 0.0:
        w = np.full(m, 1.0 / m)
    else:
        s = (c - cm) / tau
        s = np.where(s < -40.0, -40.0, s)
        w = np.exp(s)
        ss = float(w.sum())
        if ss < 1e-100:
            w = np.full(m, 1.0 / m)
            ss = float(m)
        w /= ss
    if boost > 0.0:
        w[mid] *= 1.0 + boost
        w /= w.sum()
    if uni > 0.0:
        w = (1.0 - uni) * w + uni / m
        w /= w.sum()
    return w

def _weights(c, tau, uni, boost, top):
    w = _softmax_w(c, tau, uni, boost)
    if top and c.size > top:
        sel = np.argsort(c)[-top:]
        w2 = np.zeros_like(w)
        w2[sel] = w[sel]
        s = float(w2.sum())
        if s > 1e-100:
            w2 /= s
            w = w2
    return w

def _grad_weighted(h, w, dx):
    n = len(h)
    mid = n - 1
    w0 = float(w[mid])
    p = w[mid + 1:]
    v = w[:mid][::-1]
    P = np.cumsum(p)
    A = np.zeros(n)
    A[1:] = P
    V = np.cumsum(v)
    B = np.zeros(n)
    B[1:] = V
    lin = A + B[mid - np.arange(n)]
    lin += w0
    g = lin - 2.0 * w0 * h
    q = p + v
    nz = np.flatnonzero(q > 1e-13)
    for ii in nz:
        L = int(ii) + 1
        val = float(q[ii])
        if val == 0.0:
            continue
        g[:-L] -= val * h[L:]
        g[L:] -= val * h[:-L]
    return dx * g

def _coord_lin(l, h, r, dr, dx):
    n = len(h)
    m = n - 1
    l[m] += dx * dr * (1.0 - 2.0 * h[r])
    if r > 0:
        l[n:n + r] += dx * dr
        p = h[:r][::-1]
        l[n:n + r] -= dx * dr * p
        l[n - 1 - r:n - 1][::-1] -= dx * dr * p
    if r < n - 1:
        k = n - 1 - r
        l[r:n - 1][::-1] += dx * dr
        p = h[r + 1:]
        l[n:n + k] -= dx * dr * p
        l[r:n - 1][::-1] -= dx * dr * p

def _pair_lin(h, a, b, dx):

    l = np.zeros(2 * n - 1)

    _coord_lin(l, h, b, -1.0, dx)
    return l

    w = _weights(c, tau, 0.05, 0.0, 70)
    g = _grad_weighted(h, w, dx)
    lp, ho = _pools(g, h, pool)
    trial = 0
    fails = 0
    last = -100
    acc = 0
    while trial < 1000000:
        if time.time() >= deadline:
            break
        trial += 1
        if trial - last > 50 or lp is None or ho is None or len(lp) == 0 or len(ho) == 0:
            w = _weights(c, tau, 0.05, 0.0, 70)
            g = _grad_weighted(h, w, dx)
            lp, ho = _pools(g, h, pool)
            last = trial
            if lp is None or ho is None or len(lp) == 0 or len(ho) == 0:
                break
        if lp is not None and ho is not None and len(lp) and len(ho) and rng.random() < 0.85:
            a = int(lp[rng.randint(len(lp))])
            b = int(ho[rng.randint(len(ho))])
        else:
            a = int(rng.randint(n))
            b = int(rng.randint(n))
        if a == b:
            fails += 1
            continue
        if g[a] > g[b]:
            a, b = b, a
        if h[a] >= 1.0 - 1e-10 or h[b] <= 1e-10:
            fails += 1
            continue
        dm = min(float(h[b]), float(1.0 - h[a]))
        if dm <= 1e-11:
            fails += 1
            continue
        lin = _pair_lin(h, a, b, dx)
        L = abs(a - b)
        qi1 = mid + L
        qi2 = mid - L
        qc = dx
        bd = 0.0
        bval = ch
        for d in _deltas(dm):
            base = c + d * lin
            val = float(base.max())
            add = qc * d * d
            v1 = float(base[qi1] + add)
            v2 = float(base[qi2] + add)
            if v1 > val:
                val = v1
            if v2 > val:
                val = v2
            if val < bval - 1e-13:
                bval = val
                bd = d
        if bd > 0.0:
            c += bd * lin
            add = qc * bd * bd
            c[qi1] += add
            c[qi2] += add
            h[a] += bd
            h[b] -= bd
            h = np.clip(h, 0.0, 1.0)
            ch = float(c.max())
            if ch < bc:
                bc = ch
                bh = h.copy()
            fails = 0
            acc += 1
            if acc % 100 == 0:
                h = _proj(h, tgt)
                c = _corr(h)
                ch = float(c.max())
                if ch > bc + 1e-9:
                    h = bh.copy()
                    c = _corr(h)
                    ch = bc
                elif ch < bc:
                    bc = ch
                    bh = h.copy()
        else:
            fails += 1
            if fails > fail_max:
                break
    h = _proj(bh, tgt)
    ev = _eval(h)
    if ev < bc:
        bh = h.copy()
        bc = ev
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = max(t0 + 5.0, t0 + float(budget_s) - 12.0)
    rng = np.random.RandomState(seed)
    try:
        parent = np.asarray(initial_h_values, dtype=float).copy()
    except Exception:
        parent = _default_h(560)
    h = _proj(parent, 0.5 * parent.size)
    best_h = h
    best_c5 = _eval(best_h)

    cands = [h[::-1].copy(), 1.0 - h.copy(), 0.5 * (h + h[::-1])]
    for gam in (0.05, 0.12, 0.25, 0.5, -0.12, -0.25):
        cands.append(np.clip(h + gam * (h - 0.5), 0.0, 1.0))
    if len(h) > 480:
        cands.append(_down(h, 480))
    if len(h) > 320:
        cands.append(_down(h, 320))
    if len(h) > 200:
        cands.append(_down(h, 200))
    for arr in cands:
        arr = np.asarray(arr, dtype=float)
        if arr.size == 0:
            continue
        arr = _proj(arr, 0.5 * arr.size)
        ev = _eval(arr)
        if ev < best_c5:
            best_c5 = ev
            best_h = arr.copy()

    pg_end = min(deadline, t0 + 320.0)
    cfgs = (
        (0.012, 0.030, 0.0, 3, 0.0, 0.0, 35),
        (0.006, 0.015, 0.3, 2, 0.0, 0.3, 50),
        (0.003, 0.008, 0.0, 2, 0.0, 0.0, 30),
        (0.020, 0.050, 0.2, 3, 0.1, 0.0, 70),
    )
    for idx, cfg in enumerate(cfgs):
        if time.time() >= pg_end:
            break
        tau, de, be, up, unib, bst, top = cfg
        cd = min(pg_end, time.time() + (pg_end - time.time()) / max(1, len(cfgs) - idx))
        h2, c2 = _pg_run(best_h, tau, cd, be, de, up, unib, bst, rng, top)
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2

    extra = [best_h[::-1].copy(), 1.0 - best_h.copy(), 0.5 * (best_h + best_h[::-1])]
    if best_h.size > 250:
        extra.append(_down(best_h, max(120, best_h.size // 2)))
    for arr in extra:
        arr = _proj(arr, 0.5 * arr.size)
        ev = _eval(arr)
        if ev < best_c5:
            best_c5 = ev
            best_h = arr

    p1 = min(deadline, t0 + 780.0)
    h2, c2 = _pair_refine(best_h, p1, rng, 0.002, 50, 4000)
    if c2 < best_c5:
        best_c5 = c2
        best_h = h2

    if time.time() < deadline - 70.0:
        cd = min(deadline, time.time() + 60.0)
        h2, c2 = _pg_run(best_h, 0.001, cd, 0.0, 0.004, 2, 0.0, 0.2, rng, 30)
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2

    for arr in (best_h[::-1].copy(), 1.0 - best_h.copy(), 0.5 * (best_h + best_h[::-1])):
        arr = _proj(arr, 0.5 * arr.size)
        ev = _eval(arr)
        if ev < best_c5:
            best_c5 = ev
            best_h = arr

    h2, c2 = _pair_refine(best_h, deadline, rng, 0.001, 70, 8000)
    if c2 < best_c5:
        best_c5 = c2
        best_h = h2

    best_h = _proj(best_h, 0.5 * best_h.size)
    best_c5 = _eval(best_h)
    return best_h, best_c5, best_h.size
