def _proj(h, target):

    # Actually simpler: find t such that sum(clip(h-t, 0, 1)) = target

def _grad_smax(h, alpha, topk):
    """Compute gradient of log-sum-exp approximation to max of correlation."""
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk < len(c):

        ct = c[idx]

    dx = 2.0 / n

    h2 = h.copy()

def _block_swap(dh, n, scale, rng, deadline, n_iter):
    """Try swapping mass between random positions."""
    c5c = _eval(dh)
    bh = dh.copy()
    bc = c5c
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = dh[i] + d, dh[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:

            c5n = _eval(dh)

                if c5n < bc:
                    bc, bh = c5n, dh.copy()
            else:
                dh[i], dh[j] = ni - d, nj + d
    return bh, bc

                     constraints=cons,

import numpy as np
from scipy.optimize import minimize

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(55):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    return _proj(np.clip(h, 0.0, 1.0), n * 0.5)

def _grad_smax(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
        ct = c[idx]
    else:
        idx = np.arange(len(c)); ct = c
    s = np.clip(alpha * (ct - m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-14:
            continue
        k = idx[ii]; sh = (n - 1) - k; wk = w[ii] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:n - sp])
            g[:n - sp] -= wk * h[sp:]
    return g, m

def _grad_exact(h, tol):
    n = len(h)
    c = _correl(h)
    m = c.max()
    active = np.where(c >= m - tol)[0]
    if len(active) < 1:
        active = np.array([c.argmax()])
    dx = 2.0 / n
    g = np.zeros(n)
    for k in active:

        else:
            sp = -sh
            g[sp:] += (1.0 - h[:n - sp])
            g[:n - sp] -= h[sp:]
    g *= dx / len(active)
    return g, m

def _nesterov(h0, n_iter, deadline, use_smax, alpha_start, alpha_end,
              topk, tol, step0, beta):
    n = len(h0)
    h = h0.copy()
    f = _eval(h)
    v = np.zeros(n)
    step = step0
    best_h, best_f = h.copy(), f
    target = n * 0.5
    for i in range(n_iter):
        if time.time() > deadline:
            break
        if use_smax:
            frac = i / max(n_iter - 1, 1)
            alpha = alpha_start * (alpha_end / alpha_start) ** frac
            g, _ = _grad_smax(h, alpha, topk)
        else:
            g, _ = _grad_exact(h, tol)
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, target)
        fn = _eval(hn)
        if fn < f - 1e-16:
            h, f = hn, fn
            step = min(step * 1.02, step0 * 25.0)
            if f < best_f:
                best_f, best_h = f, h.copy()
        else:
            step *= 0.35
            for _ in range(6):

                if fn < f:
                    h, f = hn, fn
                    break
                step *= 0.5
            if f < best_f:

def _fourier_perturb(h, rng, n_modes, amp):
    n = len(h)
    h2 = h.copy()
    for _ in range(n_modes):
        freq = rng.randint(1, n // 3)
        phase = rng.uniform(0, 2 * np.pi)
        d = amp * np.sin(2.0 * np.pi * freq * np.arange(n) / n + phase)
        d -= d.mean()
        h2 = np.clip(h2 + d, 0, 1)
    return _repair(h2, n)

def _slsqp(h0, alpha, topk, maxit, deadline):
    n = len(h0)
    if time.time() > deadline:
        return h0
    def obj(x):
        c = _correl(x)
        m = c.max()
        idx = np.argpartition(c, -topk)[-topk:] if topk < len(c) else np.arange(len(c))
        ct = c[idx]
        s = np.clip(alpha * (ct - m), -400, 400)
        return (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha
    def obj_g(x):
        g, _ = _grad_smax(x, alpha, topk)
        return g / alpha
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(obj, h0, jac=obj_g, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-13})
        return _repair(r.x, n)
    except Exception:
        return h0

def _swap_polish(h, n, scale, rng, deadline, n_iter):
    c5c = _eval(h)
    bh = h.copy(); bc = c5c
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        i = rng.randint(n); j = rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
                if c5n < bc:
                    bc, bh = c5n, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc
