    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _grad_smax(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
        ct = c[idx]

            continue

        if sh >= 0:

            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])

    m = c.max()

    else:
        ct = c

import numpy as np
import time

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _grad_smax(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
        ct = c[idx]
    else:
        idx = np.arange(len(c)); ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    w = np.exp(s - s.max()); w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-13:
            continue
        k = idx[ii]; sh = (n - 1) - k; wk = w[ii] * dx
        if sh >= 0:
            g[:n - sh] += wk * (1.0 - h[sh:])
            if sh > 0:
                g[sh:] -= wk * h[:-sh]
        else:
            sp = -sh
            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n - sp] -= wk * h[sp:]
    return g, m

def _nesterov_adapt(h, steps, step0, deadline, a0, a1, topk, beta):
    n = len(h)
    target = n * 0.5
    f = _eval(h)
    v = np.zeros(n)
    step = step0
    best_h, best_f = h.copy(), f
    stall = 0
    for it in range(steps):
        if time.time() > deadline:
            break
        frac = float(it) / max(steps, 1)
        alpha = a0 * (a1 / a0) ** frac
        g, _ = _grad_smax(h, alpha, topk)
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, target)
        fn = _eval(hn)
        if fn < f - 1e-15:
            h, f = hn, fn
            step = min(step * 1.02, step0 * 30.0)
            stall = 0
            if f < best_f:
                best_f, best_h = f, h.copy()
        else:
            step *= 0.45
            stall += 1
            h = _proj(h - step * v, target)
            f = _eval(h)
            if f < best_f:
                best_f, best_h = f, h.copy()
            if stall > 150:
                v = np.zeros(n)
                step = step0 * 0.5
                stall = 0
    return _repair(best_h, n), best_f

def _anneal(h, niter, rng, deadline, blocks, T0, T1):
    n = len(h)
    c5c = _eval(h); bh = h.copy(); bc = c5c
    for it in range(niter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0 + bs].copy()
        d = 0.006 * max(T, 1e-9) ** 0.5 * rng.randn(bs)
        d -= d.mean()
        seg += d; seg = np.clip(seg, 0, 1)
        hn = h.copy(); hn[i0:i0 + bs] = seg
        hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h, c5c = hn, c5n
            if c5n < bc:
                bc, bh = c5n, h.copy()
    return bh, bc

def _polish(h, niter, rng, deadline, scale):
    n = len(h)
    c5c = _eval(h); bh = h.copy(); bc = c5c
    for _ in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 10)) % n
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < bc:
                    bc, bh = c5n, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc

def _jitter_reopt(h, rng, alpha, topk, step0, deadline):
    n = len(h)
    c5c = _eval(h)
    for sc in [0.015, 0.005, 0.002]:
        if time.time() > deadline:
            break
        d = sc * rng.randn(n); d -= d.mean()
        hn = _repair(np.clip(h + d, 0, 1), n)
        hn2, c5n = _nesterov_adapt(hn, 400, step0, deadline, alpha, alpha * 4, topk, 0.9)
        if c5n < c5c:
            return hn2, c5n
    return h, c5c

def _block_init(n, rng):
    h = np.zeros(n)
    w = n // 4
    c1, c2 = n // 4, 3 * n // 4
    h[c1 - w // 2:c1 + w // 2] = 1.0
    h[c2 - w // 2:c2 + w // 2] = 1.0
    h += 0.02 * rng.randn(n)
    return _repair(np.clip(h, 0, 1), n)

def _sine_init(n, rng):
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + 0.25 * np.sin(2 * np.pi * x + rng.uniform(0, 6.28))
    h += 0.04 * rng.randn(n)
    return _repair(np.clip(h, 0, 1), n)

def _stair_init(n, rng):
    h = np.zeros(n)
    ns = rng.randint(3, 8)
    sw = n // ns
    for i in range(ns):
        h[i * sw:(i + 1) * sw] = rng.uniform(0.15, 0.85)
    h += 0.02 * rng.randn(n)
    return _repair(np.clip(h, 0, 1), n)

def _downsample(h, factor):
    n = len(h)
    nm = n // factor
    h2 = np.zeros(nm)
    for i in range(nm):
        h2[i] = np.mean(h[i * factor:(i + 1) * factor])
    return _repair(h2, nm)

def _upsample(h, factor):
    n = len(h)
    nm = n * factor
    h2 = np.repeat(h, factor).astype(float) + np.random.RandomState(0).randn(nm) * 0.005
    return _repair(np.clip(h2, 0, 1), nm)
