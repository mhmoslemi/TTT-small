        step = lr * m1h / (np.sqrt(m2h) + 1e-8)
        h_new = _proj(h - step, target)
        f_new = _eval(h_new)
        h, f = h_new, f_new
        lr *= decay
        if f < best_f:
            best_f, best_h = f, h.copy()
    return _repair(best_h, n), best_f

def _nesterov(h, steps, step0, deadline, tol, beta=0.9):
    n = len(h)
    target = n * 0.5
    f = _eval(h)
    v = np.zeros(n)
    step = step0
    best_h, best_f = h.copy(), f
    for _ in range(steps):
        if time.time() > deadline: break
        g, _ = _grad_exact(h, tol)
        v = beta * v + (1 - beta) * g
        h_new = _proj(h - step * v, target)
        f_new = _eval(h_new)
        if f_new < f - 1e-15:
            h, f = h_new, f_new
            step = min(step * 1.01, step0 * 20.0)
            if f < best_f: best_f, best_h = f, h.copy()
        else:
            step *= 0.5
            h = _proj(h - step * v, target)
            f = _eval(h)
            if f < best_f: best_f, best_h = f, h.copy()
    return _repair(best_h, n), best_f

def _worst_shift_descent(h, steps, lr, deadline):
    n = len(h)
    target = n * 0.5
    best_h, best_f = h.copy(), _eval(h)
    for _ in range(steps):
        if time.time() > deadline: break
        c = _correl(h)
        k_star = int(c.argmax())
        sh = (n - 1) - k_star
        dx = 2.0 / n
        g = np.zeros(n)
        if sh >= 0:
            g[:n - sh] += (1.0 - h[sh:]) * dx
            if sh > 0: g[sh:] -= h[:-sh] * dx
        else:
            sp = -sh
            g[sp:] += (1.0 - h[:-sp]) * dx
            g[:n - sp] -= h[sp:] * dx
        h_new = _proj(h - lr * g, target)
        f_new = _eval(h_new)
        if f_new <= best_f:
            h, best_f, best_h = h_new, f_new, h_new.copy()
    return _repair(best_h, n), best_f

def _jitter_reopt(h, rng, deadline, n2=None):
    n = len(h)
    c5c = _eval(h)
    d = 0.002 * rng.randn(n)
    d -= d.mean()
    hn = _repair(np.clip(h + d, 0, 1), n)
    hn2, c5n = _adam_opt(hn, 800, 0.003, deadline, 30000, 8)
    if c5n < c5c: return hn2, c5n
    d2 = 0.005 * rng.randn(n)
    d2 -= d2.mean()
    hn = _repair(np.clip(h + d2, 0, 1), n)
    hn2, c5n = _nesterov(hn, 2000, 0.001, deadline, 0.001)
    if c5n < c5c: return hn2, c5n
    return h, c5c

def _coord_polish(h, iters, rng, deadline):
    nc = len(h)
    c5c = _eval(h); bh = h.copy(); bc = c5c
    for _ in range(iters):
        if time.time() > deadline: break
        i = rng.randint(nc)
        j = (i + 1 + rng.randint(0, 7)) % nc
        d = 3e-7 * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < bc: bc, bh = c5n, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc

def _slsqp_smooth(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline: return h
    def obj(x):
        c = _correl(x)
        m = c.max()
        idx = np.argpartition(c, -topk)[-topk:] if topk < len(c) else np.arange(len(c))
        ct = c[idx]
        s = np.clip(alpha * (ct - m), -400, 400)
        return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha
    def grad(x):
        return _grad_sm(x, alpha, topk)[0]
    try:
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5, 'jac': lambda x: np.ones(n)}]
        r = minimize(obj, h, jac=grad, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-18})
        return _repair(r.x, n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h.copy(); best_c5 = _eval(best_h)

    def update(hh):
        nonlocal best_h, best_c5
        cv = _eval(hh)
        if cv < best_c5: best_c5, best_h = cv, hh.copy()

    # Phase 1: SLSQP warmup
    dl1 = min(deadline, t0 + 100)
    h = _slsqp_smooth(h, 5000, 20, 300, dl1)
    update(h)

    # Phase 2: Adam warmup (smooth)
    dl2 = min(deadline, t0 + 200)
    h2, c5v = _adam_opt(best_h.copy(), 999999, 0.008, dl2, 5000, 15, decay=0.9999)
    update(h2)
    h = h2 if c5v < best_c5 else best_h.copy()

    # Phase 3: Adam focus (sharper)
    dl3 = min(deadline, t0 + 380)
    h2, c5v = _adam_opt(best_h.copy(), 999999, 0.002, dl3, 50000, 6, b1=0.9, decay=0.99995)
    update(h2)
    h = h2 if c5v < best_c5 else best_h.copy()

    # Phase 4: Nesterov exact subgradient
    dl4 = min(deadline, t0 + 520)
    h2, c5v = _nesterov(best_h.copy(), 999999, 0.0015, dl4, 0.001, beta=0.88)
    update(h2)
    h = h2 if c5v < best_c5 else best_h.copy()

    # Phase 5: Worst-shift targeted descent
    dl5 = min(deadline, t0 + 600)
    h2, c5v = _worst_shift_descent(best_h.copy(), 999999, 0.001, dl5)
    update(h2)
    h = h2 if c5v < best_c5 else best_h.copy()

    # Phase 6: Adam very sharp
    dl6 = min(deadline, t0 + 720)
    h2, c5v = _adam_opt(best_h.copy(), 999999, 0.0005, dl6, 300000, 3, b1=0.92, decay=0.99998)
    update(h2)
    h = h2 if c5v < best_c5 else best_h.copy()

    # Phase 7: Jitter restarts
    dl7 = min(deadline, t0 + 880)
    for _ in range(4):
        if time.time() > dl7: break
        hj, cj = _jitter_reopt(best_h.copy(), rng, dl7)
        update(hj)

    # Phase 8: Fine Nesterov
    dl8 = min(deadline, t0 + budget_s - 60)
    h2, c5v = _nesterov(best_h.copy(), 999999, 0.0004, dl8, 0.0003, beta=0.93)
    update(h2)

    # Phase 9: Worst-shift squeeze
    dl9 = min(deadline, t0 + budget_s - 20)
    h2, c5v = _worst_shift_descent(best_h.copy(), 999999, 0.0003, dl9)
    update(h2)

    # Phase 10: Coordinate polish
    rem = max(0, deadline - time.time())
    n_pol = min(int(rem / 0.002), 40000)
    best_h, best_c5 = _coord_polish(best_h.copy(), n_pol, rng, deadline)

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
