def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if h.sum() < 1e-12:
        return np.full(n, 0.5)
    if abs(h.sum() - target) > 1e-10:
        lo, hi = -3.0, 3.0
        for _ in range(80):
            mid = (lo + hi) * 0.5
            if np.clip(h - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo + hi) * 0.5, 0, 1)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _sg_lag(h, k_idx, n):
    dx = 2.0 / n
    g = np.zeros(n)
    sh = (n - 1) - k_idx
    if sh >= 0:
        g[sh:] += dx * (1.0 - h[:n - sh])
        if sh > 0:
            g[sh:] -= dx * h[:n - sh]
    else:
        sp = -sh
        g[sp:] += dx * (1.0 - h[:n - sp])
        g[:n - sp] -= dx * h[sp:]
    return g

def _sg_topk(h, k_top, n):
    c = _corr(h, n)
    top_k = np.argsort(c)[-k_top:]
    g = np.zeros(n)
    for ki in top_k:
        g += _sg_lag(h, ki, n)
    g /= k_top
    return g

def _nesterov_k1(h, steps, step0, mu, t_end):
    n = len(h)
    f = _eval(h)
    v = np.zeros(n)
    step = step0
    stall = 0
    for _s in range(steps):
        if time.time() > t_end:
            break
        c = _corr(h, n)
        g = _sg_lag(h, int(np.argmax(c)), n)
        h_old = h.copy()
        v = mu * v + (1.0 - mu) * g
        h_new = _repair(h - step * v, n)
        f_new = _eval(h_new)
        if f_new < f - 1e-16:
            h, f = h_new, f_new
            step = min(step * 1.03, step0 * 30.0)
            stall = 0
        else:
            step *= 0.5
            stall += 1
            h = h_old
            if step < 1e-12 or stall > 200:
                v = np.zeros(n)
                step = step0 * 0.3
                stall = 0
                h = _repair(h - step * g, n)
                f = _eval(h)
    return _repair(h, n)

def _proj_grad_ls(h, k_top, steps, step0, t_end):
    n = len(h)
    f = _eval(h)
    step = step0
    for _s in range(steps):
        if time.time() > t_end:
            break
        g = _sg_topk(h, k_top, n)
        h_old = h.copy()
        improved = False
        for _trial in range(6):
            h_try = _repair(h_old - step * g, n)
            f_try = _eval(h_try)
            if f_try < f - 1e-15:
                h, f = h_try, f_try
                step = min(step * 1.08, step0 * 50.0)
                improved = True
                break
            step *= 0.45
        if not improved and step < 1e-13:
            break
    return _repair(h, n)

def _ps_slsqp(h, p, n, target, maxiter, t_end):
    n_use = n
    dx = 2.0 / n_use

    def _obj_ps(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        c = np.clip(c, 1e-10, None)
        return float(np.sum(c ** p) ** (1.0 / p))

    def _grad_ps(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        c = np.clip(c, 1e-10, None)
        S = np.sum(c ** p)
        pref = S ** (1.0 / p - 1.0)
        g = np.zeros(n_use)
        for k_idx in range(2 * n_use - 1):
            w = c[k_idx] ** (p - 1) * pref
            if w < 1e-16:
                continue
            sh = (n_use - 1) - k_idx
            if sh >= 0:
                g[sh:] += w * (1.0 - x[:n_use - sh])
                if sh > 0:
                    g[sh:] -= w * x[:n_use - sh]
            else:
                sp = -sh
                g[sp:] += w * (1.0 - x[:n_use - sp])
                g[:n_use - sp] -= w * x[sp:]
        return g

    def _con_fun(x):
        return x.sum() - target

    def _con_jac(x):
        return np.ones(n_use)

    bnds = [(0.0, 1.0)] * n_use
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(_obj_ps, h, jac=_grad_ps, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

    best_h, best_f = h.copy(), f
    for _ in range(n_trials):

        a, b = h[i], h[j]
        for t in [1e-4, -1e-4, 5e-5, -5e-5, 2e-4, -2e-4]:
            ni, nj = a + t, b - t
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h[i], h[j] = ni, nj
                fn = _eval(h)
                if fn < f:
                    f = fn
                    if fn < best_f:
                        best_f, best_h = fn, h.copy()
                else:
                    h[i], h[j] = a, b
                    break
                a, b = h[i], h[j]
    return best_h, best_f
