def _slp_step(h, n, n_iter_budget, t_end):
    """SLP: Sequential Linear Programming"""
    step_limit = 0.05  # max |delta| per coordinate
    for _ in range(n_iter_budget):
        if time.time() > t_end:
            break
        c = _corr(h)
        M = c.max()
        # Active set: all shifts within 10% of max
        active = np.where(c >= M * 0.95)[0]
        
        # Build LP
        # min t
        # s.t. c_k(h) + sum_i grad_k[i]*delta[i] <= t  for k in active
        #      -step_limit <= delta[i] - 0 <= step_limit  (implied by bounds on h)
        #      0 <= h[i] + delta[i] <= 1
        #      sum(delta) = 0
        
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0  # minimize t
        
        A_ub_rows = []
        b_ub_rows = []

            sh = (n - 1) - k
            grad = np.zeros(n)
            if sh >= 0:

                if sh > 0:

import numpy as np
from scipy.optimize import minimize, linprog
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    if h.sum() < 1e-12:
        return np.full(n, 0.5)
    if abs(h.sum() - target) > 1e-10:
        lo, hi = -3.0, 3.0
        for _ in range(80):
            mid = (lo + hi) * 0.5
            if np.clip(h - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo + hi) * 0.5, 0, 1)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _grad_one(h, k):
    """Gradient of c[k] w.r.t. h, without the dx factor"""
    n = len(h)
    g = np.zeros(n)
    sh = (n - 1) - k
    if sh >= 0:
        g[:n-sh] += (1.0 - h[sh:])
        if sh > 0:
            g[sh:] -= h[:n-sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:n-sp])
        g[:n-sp] -= h[sp:]
    return g

def _slp_phase(h, n, t_end, k_start, k_end, step_limit_start, step_limit_end):
    """SLP phase with increasing active set and decreasing step"""
    n_active_total = 2 * n - 1
    t_s = time.time()
    t_e = t_end
    total_iter = 0
    step_limit = step_limit_start
    
    while time.time() < t_e:
        c = _corr(h)
        M = c.max()
        # Active set size grows from k_start to k_end
        frac = min(1.0, (time.time() - t_s) / max(1e-9, t_e - t_s))
        k_active = int(k_start + frac * (k_end - k_start))
        k_active = min(k_active, n_active_total)
        
        # Get top k_active shifts
        top_k = np.argsort(c)[-k_active:]
        
        # Build LP: min t s.t. c_k(h) + grad_k' . delta <= t for k in top_k
        #            0 <= h[i]+delta[i] <= 1, sum(delta)=0, |delta[i]|<=step_limit
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        
        # Inequality constraints
        A_rows = []
        b_rows = []
        
        for k in top_k:
            g = _grad_one(h, k)
            row = np.concatenate([g, [-1.0]])
            A_rows.append(row)
            b_rows.append(-c[k])
        
        # Box constraints on h[i] + delta[i] in [0, 1]
        for i in range(n):
            row = np.zeros(n + 1); row[i] = 1.0
            A_rows.append(row); b_rows.append(1.0 - h[i])
            row = np.zeros(n + 1); row[i] = -1.0
            A_rows.append(row); b_rows.append(h[i])
        
        A_ub = np.array(A_rows)
        b_ub = np.array(b_rows)
        
        # Step limit: |delta[i]| <= step_limit
        # This is handled by the box constraints already (h[i] in [0,1])
        # But we also want to limit the step size. Add:
        # step_limit - delta[i] >= 0 and delta[i] + step_limit >= 0
        # Actually, the box constraints already limit delta to [-h[i], 1-h[i]]
        # For additional step limiting, we can scale: 
        # Let's just use the box constraints and reduce step_limit by
        # clipping delta after solving.
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([0.0])
        
        # Bounds: delta[i] in [-min(h[i], step_limit), min(1-h[i], step_limit)]
        bounds = []
        for i in range(n):
            lo = -min(h[i], step_limit)
            hi = min(1.0 - h[i], step_limit)
            bounds.append((lo, hi))
        bounds.append((0.0, None))  # t unbounded above, >= 0
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs', options={'presolve': True})
        except Exception:
            step_limit *= 0.5
            continue
        
        if res.success:
            delta = res.x[:n]
            # Clip to step_limit
            np.clip(delta, -step_limit, step_limit, out=delta)
            h_new = _repair(h + delta, n)
            f_new = _eval(h_new)
            f_cur = _eval(h)
            if f_new < f_cur:
                h = h_new
            else:
                # Try half step
                h_half = _repair(h + 0.5 * delta, n)
                if _eval(h_half) < f_cur:
                    h = h_half
                else:
                    step_limit *= 0.7
        
        step_limit = max(step_limit_end, step_limit * 0.995)
        total_iter += 1
    
    return _repair(h, n)

def _nesterov_active(h, n, k_active, n_steps, t_end, step0=0.005):
    """Nesterov-accelerated gradient on active set"""
    v = np.zeros(n)
    mu = 0.8
    step = step0
    f = _eval(h)
    for _ in range(n_steps):
        if time.time() > t_end:
            break
        c = _corr(h)
        top_k = np.argsort(c)[-k_active:]
        g = np.zeros(n)
        for k in top_k:
            g += _grad_one(h, k)
        g /= k_active
        v = mu * v + (1 - mu) * g
        h_new = _repair(h - step * v, n)
        f_new = _eval(h_new)
        if f_new < f:
            h, f = h_new, f_new
            step = min(step * 1.03, step0 * 5)
        else:
            step *= 0.5
            if step < 1e-13:
                v = np.zeros(n)
                step = step0 * 0.5
    return _repair(h, n)

def _fine_local(h, n, n_tries, rng, t_end, scale=2e-5):
    """Ultra-fine 2-point local search with adaptive step"""
    best_h, best_c = h.copy(), _eval(h)
    for _ in range(n_tries):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 5)) % n
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c_new = _eval(h)
            if c_new < best_c:
                best_c, best_h = c_new, h.copy()
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return best_h, best_c

def _s3_local(h, n, n_tries, rng, t_end, scale=3e-5):
    """3-point local search"""
    best_h, best_c = h.copy(), _eval(h)
    for _ in range(n_tries):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = (i + 2 + rng.randint(0, 4)) % n
        k = (i + 6 + rng.randint(0, 8)) % n
        d1, d2 = scale * rng.randn(), scale * rng.randn()
        ni, nj, nk = h[i]+d1, h[j]+d2, h[k]-d1-d2
        if 0 <= ni <= 1 and 0 <= nj <= 1 and 0 <= nk <= 1:
            h[i], h[j], h[k] = ni, nj, nk
            c_new = _eval(h)
            if c_new < best_c:
                best_c, best_h = c_new, h.copy()
            else:
                h[i], h[j], h[k] = ni-d1, nj-d2, nk+d1+d2
    return best_h, best_c

    for _ in range(ntries):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(0, 5)) % n
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            cn = _eval(h)
            if cn < best_c:
                best_c, best_h = cn, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def _targeted_local(h, n, ntries, rng, t_end, scale):
    """Local moves targeting the worst active shifts"""
    best_h, best_c = h.copy(), _eval(h)
    for _ in range(ntries):
        if time.time() > t_end:
            break
        c = _corr(h)
        M = c.max()
        near_max = np.where(c >= M - 0.03 * M)[0]
        k = near_max[rng.randint(len(near_max))]
        sh = (n - 1) - k
        if sh >= 0 and sh < n:
            i0 = rng.randint(n - sh)
            i, j = i0, i0 + sh
        elif sh < 0:
            sp = -sh
            i0 = rng.randint(n - sp)
            i, j = i0 + sp, i0
        else:
            i = rng.randint(n)
            j = (i + 1) % n
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            cn = _eval(h)
            if cn < best_c:
                best_c, best_h = cn, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    tg = t0 + budget_s
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: SLP coarse active set
    t1 = t0 + budget_s * 0.12
    if time.time() < t1:
        h = _slp_phase(h, n, t1, 5, 12, 0.1, 0.03)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5, best_h = c5v, h.copy()

    # Phase 2: Nesterov medium
    t2 = t0 + budget_s * 0.28
    if time.time() < t2:
        h = _nesterov_act(best_h.copy(), n, 15, 8000, t2, 0.004)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5, best_h = c5v, h.copy()

    # Phase 3: SLP medium active set
    t3 = t0 + budget_s * 0.48
    if time.time() < t3:
        h = _slp_phase(best_h.copy(), n, t3, 12, 30, 0.03, 0.006)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5, best_h = c5v, h.copy()

    # Phase 4: Nesterov larger active set
    t4 = t0 + budget_s * 0.64
    if time.time() < t4:
        h = _nesterov_act(best_h.copy(), n, 40, 6000, t4, 0.0015)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5, best_h = c5v, h.copy()

    # Phase 5: SLP fine active set
    t5 = t0 + budget_s * 0.80
    if time.time() < t5:
        h = _slp_phase(best_h.copy(), n, t5, 30, 60, 0.005, 0.001)
        c5v = _eval(h)
        if c5v < best_c5:
            best_c5, best_h = c5v, h.copy()

    # Phase 6: Targeted local search (worst shifts)
    t6 = t0 + budget_s * 0.87
    if time.time() < t6:
        best_h, best_c5 = _targeted_local(best_h, n, 12000, rng, t6, 4e-5)

    # Phase 7: 3-point local search
    t7 = t0 + budget_s * 0.94
    if time.time() < t7:
        best_h, best_c5 = _local3(best_h, n, 12000, rng, t7, 3e-5)

    # Phase 8: 2-point ultra-fine
    if time.time() < tg - 1:
        best_h, best_c5 = _local2(best_h, n, 20000, rng, tg - 0.5, 1e-5)

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
