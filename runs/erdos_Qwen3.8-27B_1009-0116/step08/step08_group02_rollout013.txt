import numpy as np

def _repair(h, n):
    h = np.clip(h, 0., 1.).copy()
    t = n / 2.
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    for _ in range(200):
        h *= t / s
        h = np.clip(h, 0, 1)
        s = h.sum()
        if abs(s - t) < 1e-12:
            break
    d = t - h.sum()
    if abs(d) > 1e-10:
        o = np.argsort(1 - h) if d > 0 else np.argsort(h)
        for i in o:
            if d > 0:
                a = min(d, 1 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1. - h, mode='full')) * 2.0 / n)

def _evalw(h):
    n = len(h)
    c = np.correlate(h, 1. - h, mode='full') * (2.0 / n)
    return float(c.max()), int(np.argmax(c)), c

def _proj(x, t):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        m = (lo + hi) * 0.5
        if np.clip(x - m, 0, 1).sum() > t:
            lo = m
        else:
            hi = m
    return np.clip(x - (lo + hi) * 0.5, 0, 1)

def _glag(k, h, n):
    dx = 2.0 / n
    sh = n - 1 - k
    g = np.zeros(n)
    if sh >= 0:
        g[sh:] += (1.0 - h[:n - sh]) * dx
        if sh > 0:
            g[:n - sh] -= h[sh:] * dx
    else:
        sp = -sh
        g[:n - sp] += (1.0 - h[sp:]) * dx
        g[sp:] -= h[:n - sp] * dx
    return g

def _nesterov(h, alpha, steps, s0, topk, deadline):

    y = h.copy()
    t_ = 1.0
    step = s0
    best_h = h.copy()

        idx = np.argsort(c)[-topk:]
        ca = c[idx]
        m = ca.max()
        s = np.clip(alpha * (ca - m), -300, 300)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        g = np.zeros(n)
        for ii in range(len(idx)):
            if w[ii] > 1e-12:
                g += w[ii] * _glag(idx[ii], y, n)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-18:

            best_c = c_new
            best_h = new_h.copy()
