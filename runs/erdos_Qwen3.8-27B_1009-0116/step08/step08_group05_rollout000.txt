def _fixsum(h,n):
    h=np.clip(np.asarray(h,dtype=float),0.0,1.0); t=n*0.5; d=t-h.sum()
    if d>1e-12:
        tot=float((1.0-h).sum())
        if tot>1e-15: h=h+(d/tot)*(1.0-h)
    elif d<-1e-12:
        tot=float(h.sum())
        if tot>1e-15: h=h*(t/tot)
    return np.clip(h,0.0,1.0)

def _proj(x,t):
    x=np.asarray(x,dtype=float); n=x.size
    if t<=0: return np.zeros(n)
    if t>=n: return np.ones(n)
    lo=float(x.min()); hi=float(x.max())
    if hi-lo<1e-13:
        out=x.copy()
    else:
        for _ in range(22):
            lam=(lo+hi)*0.5
            if float(np.clip(x-lam,0.0,1.0).sum())>t: lo=lam
            else: hi=lam
        out=np.clip(x-(lo+hi)*0.5,0.0,1.0)
    d=t-float(out.sum())
    if abs(d)>1e-10:
        m=(out>1e-9)&(out<1-1e-9); c=int(m.sum())
        if c: out=out+(d/float(c))*m
    return np.clip(out,0.0,1.0)

def _correl(h):
    n=h.size
    return np.correlate(h,1.0-h,mode='full')*(2.0/n)

def _eval(h):
    return float(_correl(h).max())

def _add_grad(h,off,n,w,g):
    if off>=0:
        g[off:]=g[off:]+w*(1.0-h[:n-off])
        g[:n-off]=g[:n-off]-w*h[off:]
    else:
        o=-off
        g[:n-o]=g[:n-o]+w*(1.0-h[o:])
        g[o:]=g[o:]-w*h[:n-o]

        bubs = (n * 0.5) * (cor[ix[0]] - cor[ix[1:]])
        bnds = [(-eps, eps)] * n
        try:
            res = linprog(gm[0], A_ub=gm[1:], b_ub=bubs, A_eq=np.ones((1, n)),
                          b_eq=np.zeros(1), bounds=bnds, method='highs')
        except Exception:
            break
        if not res.success:
            break
        d = res.x; mx = float(np.max(np.abs(d)))
        if mx < 1e-13:
            break
        d = d * (eps / mx); st = 1.0; moved = False
        for _ in range(6):
            hn = _fixsum(h + st * d, n); c2 = float(_correl(hn).max())
            if c2 < c - 1e-14:
                h = hn; c = c2; moved = True
                if c < bc:
                    bc = c; best = h.copy()
                break
            st = st * 0.5
        if not moved:
            h = _proj(h + 0.006 * rng.randn(n), n * 0.5)
            c2 = _eval(h)
            if c2 < bc:
                bc = c2; best = h.copy()
    return best, bc

def _polish(h0, n, dead, seed):
    h = _fixsum(h0, n).copy(); rng = np.random.RandomState(seed)
    c = _eval(h); best = h.copy(); bc = c
    scales = [2e-5, 8e-6, 4e-6, 1.5e-6]; ns = len(scales)
    t0 = time.time(); spanmax = max(1.0, dead - t0)
    for si, ss in enumerate(scales):
        dsub = t0 + spanmax * ((si + 1) / float(ns))
        while time.time() < min(dead, dsub) - 0.3:
            i = int(rng.randint(n)); j = (i + 1 + int(rng.randint(0, 6))) % n
            d = ss * (2.0 * rng.random() - 1.0)
            a = h[i] + d; b = h[j] - d
            if 0.0 <= a <= 1.0 and 0.0 <= b <= 1.0:
                h[i] = a; h[j] = b; c2 = _eval(h)
                if c2 < c - 1e-15:
                    c = c2
                    if c < bc:
                        bc = c; best = h.copy()
                else:
                    h[i] = a - d; h[j] = b + d
    return best, bc
