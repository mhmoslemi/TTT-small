    if n > 1:
        idx = np.arange(1, n)
        P[idx] = pref[n - idx] - ac[idx]
        Nv[idx] = total - pref[idx] - ac[idx]
        mx = float(max(np.max(P), np.max(Nv)))
    else:
        mx = float(P[0])
    return P, Nv, mx

def _grad_from_weights(h, wP, wN):
    n = len(h)
    dx = 2.0 / n
    u = np.zeros(n)
    if n > 1:
        u[1:] = wP[1:] + wN[1:]
    cp = np.zeros(n); cn = np.zeros(n)
    if n > 1:
        cp[1:] = np.cumsum(wP[1:])
        cn[1:] = np.cumsum(wN[1:])
    linP = cp[n - 1::-1]
    linN = cn
    q = np.zeros(n)
    if n > 1 and np.any(u[1:]):
        Nf = _fft_size(n)
        H = np.fft.rfft(h, Nf)
        Urev = np.fft.rfft(u[::-1], Nf)
        Ur = np.fft.rfft(u, Nf)
        q = -(np.fft.irfft(H * Urev, Nf)[n-1:2*n-1].real + np.fft.irfft(H * Ur, Nf)[:n].real)
    return dx * (wP[0] * (1.0 - 2.0 * h) + linP + linN + q)

def _smooth_value(h, alpha):
    P, Nv, mx = _pn_arrays(h)
    n = len(h)
    dx = 2.0 / n
    e = np.exp(np.clip(alpha * (P[0] - mx), -400.0, 0.0))
    if n > 1:
        e += np.exp(np.clip(alpha * (P[1:] - mx), -400.0, 0.0)).sum()
        e += np.exp(np.clip(alpha * (Nv[1:] - mx), -400.0, 0.0)).sum()
    return dx * (mx + np.log(e) / alpha)

def _smooth_grad(h, alpha):
    P, Nv, mx = _pn_arrays(h)
    n = len(h)
    wP = np.zeros(n); wN = np.zeros(n)
    e0 = np.exp(np.clip(alpha * (P[0] - mx), -400.0, 0.0))
    tot = e0
    if n > 1:
        ep = np.exp(np.clip(alpha * (P[1:] - mx), -400.0, 0.0))
        en = np.exp(np.clip(alpha * (Nv[1:] - mx), -400.0, 0.0))
        tot += ep.sum() + en.sum()
        if tot <= 0.0:
            tot, e0 = 1.0, 1.0
        wP[0] = e0 / tot; wP[1:] = ep / tot; wN[1:] = en / tot
    else:
        wP[0] = 1.0
    return _grad_from_weights(h, wP, wN)

def _fista(h0, alpha, max_time, L=15.0):
    if max_time <= 0.0:
        return h0.copy(), _c5_direct(h0)
    n = len(h0); S = n * 0.5
    t0 = time.time()
    h = _repair(h0, n); hp = h.copy(); y = h.copy(); t = 1.0
    fy = _smooth_value(h, alpha); fh = fy
    best_h = h.copy(); best_c = _c5_direct(h)
    stale = 0; Lcur = L
    for it in range(200000):
        if time.time() - t0 > max_time:
            break
        if it:
            tn = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
            y = h + ((t - 1.0) / tn) * (h - hp); t = tn
            if y.min() < -1e-5 or y.max() > 1.0 + 1e-5 or abs(y.sum() - S) > 1e-8:
                y = _proj(y, S)
            fy = _smooth_value(y, alpha)
        g = _smooth_grad(y, alpha)
        step = 1.0 / max(Lcur, 1e-12)
        accepted = False
        for _ in range(14):
            xn = _proj(y - step * g, S)
            fx = _smooth_value(xn, alpha)
            d = xn - y
            rhs = fy + float(np.dot(g, d)) + 0.5 * (1.0 / step) * float(np.dot(d, d))
            if fx <= rhs + 1e-11:
                accepted = True; break
            step *= 0.5
        Lcur = 1.0 / max(step, 1e-12)
        if accepted:
            hp = h; h = xn; fh = fx; stale = 0
            Lcur = max(Lcur * 0.82, 1.0)
            if it % 6 == 0:
                c = _c5_direct(h)
                if c < best_c:
                    best_c = c; best_h = h.copy()
        else:
            stale += 1; hp = h; y = h; t = 1.0; fy = fh
            Lcur = max(Lcur * 0.5, 1.0)
            if stale > 12:
                break
    c = _c5_direct(h)
    if c < best_c:
        best_c = c; best_h = h.copy()
    return best_h, best_c

def _resample(h, m):
    n = len(h)
    if n == m:
        return _repair(h, n)
    if n % m == 0:
        out = h.reshape(m, n // m).mean(axis=1)
    else:
        x0 = (np.arange(n) + 0.5) * (2.0 / n)
        x1 = (np.arange(m) + 0.5) * (2.0 / m)
        out = np.interp(x1, x0, h)
    return _proj(out, m * 0.5)

def _local_search_v2(h0, rng, deadline, init_step=5.0e-5):
    tl0 = time.time()
    n = len(h0); S = n * 0.5
    h = _repair(h0, n).copy()
    c = _c5_direct(h)
    best_c = c; best_h = h.copy()
    P, Nv, _ = _pn_arrays(h)
    mx = max(float(P[0]), float(np.max(P)), float(np.max(Nv)))
    tol = max(1e-9, 6.0 * (2.0 / n))
    g = np.zeros(n)
    recompute_every = 1024
    it = 0
    T = 0.02
    step = init_step
    no_improve = 0
    while time.time() < deadline - 0.5:
        if it >= recompute_every:
            c = _c5_direct(h)
            if c < best_c:
                best_c = c; best_h = h.copy()
            P, Nv, _ = _pn_arrays(h)
            mx = max(float(P[0]), float(np.max(P)), float(np.max(Nv)))
            wP = np.zeros(n); wN = np.zeros(n); cnt = 0
            if P[0] >= mx - tol:
                wP[0] = 1.0; cnt += 1
            mP = P[1:] >= mx - tol
            if np.any(mP):
                pos = np.nonzero(mP)[0] + 1; wP[pos] = 1.0; cnt += pos.size
            mN = Nv[1:] >= mx - tol
            if np.any(mN):
                pos = np.nonzero(mN)[0] + 1; wN[pos] = 1.0; cnt += pos.size
            if cnt > 0:
                wP /= cnt; wN /= cnt
                g = _grad_from_weights(h, wP, wN)
            recompute_every = it + 1024
            if no_improve > 5000:
                h = _proj(h + 0.01 * (rng.rand(n) - 0.5), S)
                c = _c5_direct(h)
                no_improve = 0
                step = init_step
            elif no_improve > 2000:
                T *= 0.95
            no_improve += 1
        frac = (time.time() - tl0) / max(1e-9, deadline - tl0)
        step = max(1e-8, init_step * (0.05) ** frac)
        if rng.random() < 0.75:
            gi = g[np.nonzero(g != 0.0)[0]] if np.any(g != 0.0) else g
            order = np.argsort(g)
            q = max(4, n // 10)
            i = int(order[rng.randint(q)])
            j = int(order[-q + rng.randint(q)])
        else:
            i = int(rng.randint(n)); j = int(rng.randint(n))
        if i == j:
            it += 1; continue
        if g[i] <= g[j]:
            s = 1
        elif g[j] < g[i]:
            s = -1
        else:
            s = 1 if rng.random() < 0.5 else -1
        if s > 0:
            avail = min(1.0 - h[i], h[j])
        else:
            avail = min(h[i], 1.0 - h[j])
        if avail > 1e-14:
            mult = 0.3 + 1.7 * rng.random()
            if rng.random() < 0.02:
                mult *= 6.0
            d = s * min(avail, step * mult)
            h[i] += d; h[j] -= d
            cn = _c5_direct(h)
            if cn < c:
                c = cn
                if c < best_c:
                    best_c = c; best_h = h.copy()
                no_improve = 0
            else:
                if rng.random() < np.exp(min(0.0, -(cn - c) / T)):
                    c = cn
                else:
                    h[i] -= d; h[j] += d
        it += 1
    c = _c5_direct(h)
    if c < best_c:
        best_c = c; best_h = h.copy()
    return best_h, best_c

def _grad_local(h0, rng, deadline, eta0=1e-4):
    n = len(h0); S = n * 0.5
    h = _repair(h0, n).copy()
    c = _c5_direct(h)
    best_c = c; best_h = h.copy()
    P, Nv, _ = _pn_arrays(h)
    mx = max(float(P[0]), float(np.max(P)), float(np.max(Nv)))
    tol = max(1e-9, 4.0 * (2.0 / n))
    wP = np.zeros(n); wN = np.zeros(n); cnt = 0
    if P[0] >= mx - tol:
        wP[0] = 1.0; cnt += 1
    mP = P[1:] >= mx - tol
    if np.any(mP):
        pos = np.nonzero(mP)[0] + 1; wP[pos] = 1.0; cnt += pos.size
    mN = Nv[1:] >= mx - tol
    if np.any(mN):
        pos = np.nonzero(mN)[0] + 1; wN[pos] = 1.0; cnt += pos.size
    g = np.zeros(n)
    if cnt > 0:
        wP /= cnt; wN /= cnt
        g = _grad_from_weights(h, wP, wN)
    eta = eta0
    it = 0
    while time.time() < deadline - 0.5:
        if it % 512 == 0:
            c = _c5_direct(h)
            if c < best_c:
                best_c = c; best_h = h.copy()
            P, Nv, _ = _pn_arrays(h)
            mx = max(float(P[0]), float(np.max(P)), float(np.max(Nv)))
            wP = np.zeros(n); wN = np.zeros(n); cnt = 0
            if P[0] >= mx - tol:
                wP[0] = 1.0; cnt += 1
            mP = P[1:] >= mx - tol
            if np.any(mP):
                pos = np.nonzero(mP)[0] + 1; wP[pos] = 1.0; cnt += pos.size
            mN = Nv[1:] >= mx - tol
            if np.any(mN):
                pos = np.nonzero(mN)[0] + 1; wN[pos] = 1.0; cnt += pos.size
            if cnt > 0:
                wP /= cnt; wN /= cnt
                g = _grad_from_weights(h, wP, wN)
            else:
                g = np.zeros(n)
        order = np.argsort(g)
        i = int(order[rng.randint(min(8, n))])
        j = int(order[max(0, n - 8) + rng.randint(min(8, n))])
        if i == j:
            it += 1; continue
        s = 1 if g[i] <= g[j] else -1
        if s > 0:
            avail = min(1.0 - h[i], h[j])
        else:
            avail = min(h[i], 1.0 - h[j])
        if avail > 1e-14:
            d = s * min(avail, eta)
            h[i] += d; h[j] -= d
            cn = _c5_direct(h)
            if cn < c:
                c = cn; eta = min(eta * 1.1, eta0)
            else:
                h[i] -= d; h[j] += d
                eta *= 0.9
        it += 1
    c = _c5_direct(h)
    if c < best_c:
        best_c = c; best_h = h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(560, 0.5)
    n0 = len(h0)
    h0 = _repair(h0, n0)
    best = h0.copy()
    best_c = _c5_direct(best)
    for hh in (_repair(1.0 - h0, n0), _proj(0.5 + 0.5*(h0 - h0[::-1]), n0*0.5), _proj(0.5 + 0.5*(h0 + h0[::-1] - 1.0), n0*0.5)):
        c = _c5_direct(hh)
        if c < best_c:
            best = hh; best_c = c
    end1 = t0 + min(80.0, 0.08 * budget_s)
    if end1 > time.time():
        hcur = best
        alphas = [15, 60, 200, 800, 3000, 12000]
        rem = end1 - time.time()
        per = rem / len(alphas)
        for a in alphas:
            if end1 - time.time() < 2.0:
                break
            b, c = _fista(hcur, a, per, 15.0)
            if c < best_c:
                best = b; best_c = c
            hcur = b
    end2 = t0 + min(0.45 * budget_s, t0 + (end1 - t0) + 200.0)
    if budget_s > 200 and end2 > time.time():
        for m in [n0 // 2, n0 * 2]:
            if end2 - time.time() < 10.0:
                break
            if 32 <= m <= 4096:
                slot = min(end2 - time.time() - 2.0, 80.0)
                hm = _resample(best, m)
                b, c = _fista(hm, 500, slot, 15.0)
                b2, c2 = _fista(b, 5000, min(slot, end2 - time.time() - 2.0), 15.0)
                if c2 < best_c:
                    best = b2; best_c = c2
                if len(best) != n0:
                    hu = _resample(best, n0)
                    cu = _c5_direct(hu)
                    if cu < best_c:
                        best = hu; best_c = cu
    end3 = t0 + budget_s - 120.0
    if end3 > time.time() + 5.0:
        hmain = best if len(best) == n0 else _resample(best, n0)
        b, c = _fista(hmain, 5000, end3 - time.time(), 15.0)
        if c < best_c:
            best = b; best_c = c
    if len(best) != n0:
        best = _resample(best, n0)
        best_c = _c5_direct(best)
    end_la = t0 + budget_s - 55.0
    if end_la > time.time() + 10.0:
        b, c = _local_search_v2(best, rng, end_la, 8.0e-5)
        if c < best_c:
            best = b; best_c = c
    end_lb = t0 + budget_s - 15.0
    if end_lb > time.time() + 5.0:
        b, c = _grad_local(best, rng, end_lb, 3.0e-5)
        if c < best_c:
            best = b; best_c = c
    end_lc = t0 + budget_s - 2.0
    if end_lc > time.time() + 2.0:
        b, c = _local_search_v2(best, rng, end_lc, 2.0e-6)
        if c < best_c:
            best = b; best_c = c
    best = _repair(best, len(best))
    best_c = _c5_direct(best)
    return best, best_c, len(best)
