    else:
        ct = c
    s = np.clip(alpha * (ct - m), -400, 400)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _grad_smax(h, alpha, topk):
    n = len(h); c = _correl(h); m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
    else:
        idx = np.arange(len(c))
    s = np.clip(alpha * (c[idx] - m), -400, 400)
    w = np.exp(s - s.max()); w /= w.sum()
    dx_v = 2.0 / n; g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] > 1e-15:
            g += w[ii] * _grad_lag(h, idx[ii], dx_v)
    return g

def _nesterov(h, alpha0, grow, topk, steps, step0, deadline):
    n = len(h); f = _smax_obj(h, alpha0, topk)
    v = np.zeros(n); beta = 0.84; step = step0; alpha = alpha0; stall = 0
    for it in range(steps):
        if time.time() > deadline: break
        if it > 0 and it % 150 == 0:
            alpha = min(alpha * grow, 8e7)
        g = _grad_smax(h, alpha, topk)
        hp = h.copy()
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, n * 0.5)
        fn = _smax_obj(hn, alpha, topk)
        if fn < f:
            h, f, step, stall = hn, fn, min(step * 1.015, step0 * 8.0), 0
        else:
            step *= 0.5
            hn = _proj(hp - step * v, n * 0.5); fn = _smax_obj(hn, alpha, topk)
            if fn < f: h, f, stall = hn, fn, 0
            else:
                step *= 0.4
                hn = _proj(hp - step * v, n * 0.5); fn = _smax_obj(hn, alpha, topk)
                if fn < f: h, f, stall = hn, fn, 0
                else: stall += 1
                if stall > 60: break
    return _repair(h, n)

def _jump_anneal(h, niter, rng, deadline, T0, T1):
    n = len(h); c5c = _eval(h); bh = h.copy(); bc = c5c
    for it in range(niter):
        if time.time() > deadline: break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        r = rng.random()
        if r < 0.35:
            bs = rng.randint(4, max(5, n // 12))
            i0 = rng.randint(0, n - bs)
            hn = h.copy(); hn[i0:i0+bs] = 1.0 - h[i0:i0+bs]
        elif r < 0.65:
            bs = rng.randint(3, max(4, n // 18))
            i0 = rng.randint(0, n - bs); i1 = rng.randint(0, n - bs)
            hn = h.copy(); tmp = hn[i0:i0+bs].copy()
            hn[i0:i0+bs] = hn[i1:i1+bs]; hn[i1:i1+bs] = tmp
        else:
            bs = rng.randint(8, max(9, n // 6))
            i0 = rng.randint(0, n - bs)
            hn = h.copy(); hn[i0:i0+bs] += 0.025 * rng.randn(bs)
        hn = _repair(hn, n); c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h, c5c = hn, c5n
            if c5n < bc: bc, bh = c5c, h.copy()
    return bh, bc

def _block_anneal(h, niter, rng, deadline, T0, T1, blocks):
    n = len(h); c5c = _eval(h); bh = h.copy(); bc = c5c
    for it in range(niter):
        if time.time() > deadline: break
        T = T0 * (T1 / T0) ** (it / max(niter, 1))
        bs = blocks[rng.randint(len(blocks))]
        i0 = rng.randint(0, max(1, n - bs))
        seg = h[i0:i0+bs].copy()
        amp = 0.002 + 0.006 * (T / T0)
        d = amp * max(T, 1e-9)**0.35 * rng.randn(bs); d -= d.mean()
        seg = np.clip(seg + d, 0, 1)
        hn = h.copy(); hn[i0:i0+bs] = seg; hn = _repair(hn, n)
        c5n = _eval(hn)
        if c5n < c5c or rng.random() < np.exp(-(c5n - c5c) / max(T, 1e-12)):
            h, c5c = hn, c5n
            if c5n < bc: bc, bh = c5c, h.copy()
    return bh, bc

def _pair_sweep(h, rng, deadline, max_sweeps=40):
    n = len(h); c5c = _eval(h); bh = h.copy(); bc = c5c
    offs = [1, 2, 3, 4, 5, 7, 11, 17, 29]
    for _ in range(max_sweeps):
        if time.time() > deadline: break
        for i in range(0, n, 3):
            if time.time() > deadline: break
            j = (i + offs[rng.randint(len(offs))]) % n
            d = 1e-4 * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h[i], h[j] = ni, nj; c5n = _eval(h)
                if c5n < c5c - 1e-18:
                    c5c = c5n
                    if c5n < bc: bc, bh = c5n, h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d
    return bh, bc

def _slsqp_fast(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline: return h
    def obj(x): return _smax_obj(x, alpha, topk)
    def jac(x): return _grad_smax(x, alpha, topk)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5, 'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(obj, h, jac=jac, method='SLSQP', bounds=[(0.0, 1.0)] * n,
                     constraints=cons, options={'maxiter': maxit, 'ftol': 1e-20})
        return _repair(r.x, n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 4
    n0 = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h, best_c5 = h.copy(), _eval(h)

    # Phase 1: SLSQP warm-up (0-100s)
    dl1 = min(deadline, t0 + 100)
    for alpha, tk, mi in [(4000, 35, 250), (18000, 18, 180)]:
        if time.time() > dl1: break
        h = _slsqp_fast(h, alpha, tk, mi, dl1)
        if _eval(h) < best_c5: best_c5, best_h = _eval(h), h.copy()

    # Phase 2: Nesterov multi-scale (100-380s)
    dl2 = min(deadline, t0 + 380)
    h1 = best_h.copy()
    for a0, gr, tk, st, ns in [(6000, 1.9, 30, 0.0035, 2500),
                                (40000, 1.5, 14, 0.0012, 2000),
                                (250000, 1.3, 7, 0.0005, 1500)]:
        if time.time() > dl2: break
        h1 = _nesterov(h1, a0, gr, tk, ns, st, dl2)
    if _eval(h1) < best_c5: best_c5, best_h = _eval(h1), h1.copy()

    # Phase 2b: Perturbed restart (380-480s)
    dl2b = min(deadline, t0 + 480)
    h2 = best_h.copy() + 0.04 * rng.randn(n0)
    h2 = _repair(h2, n0)
    h2 = _nesterov(h2, 8000, 1.8, 20, 1200, 0.003, dl2b)
    h2 = _nesterov(h2, 60000, 1.4, 8, 800, 0.0008, dl2b)
    if _eval(h2) < best_c5: best_c5, best_h = _eval(h2), h2.copy()

    # Phase 3: Jump annealing (480-600s)
    dl3 = min(deadline, t0 + 600)
    hj, cj = _jump_anneal(best_h.copy(), 25000, rng, dl3, 0.012, 1e-10)
    if cj < best_c5:
        best_c5, best_h = cj, hj.copy()
        if time.time() < dl3:
            best_h = _nesterov(best_h, 80000, 1.4, 8, 400, 0.001, dl3)
            best_c5 = _eval(best_h)

    # Phase 4: Block annealing (600-740s)
    dl4 = min(deadline, t0 + 740)
    blocks = [1, 2, 3, 5, 8, 13, 20, 35, 55, 80, 110]
    ha, ca = _block_anneal(best_h.copy(), 70000, rng, dl4, 0.0035, 1e-11, blocks)
    if ca < best_c5: best_c5, best_h = ca, ha.copy()

    # Phase 5: Nesterov high-alpha polish (740-840s)
    dl5 = min(deadline, t0 + 840)
    best_h = _nesterov(best_h, 500000, 1.3, 6, 1500, 0.0008, dl5)
    best_h = _nesterov(best_h, 4000000, 1.15, 3, 800, 0.0003, dl5)
    best_c5 = _eval(best_h)

    # Phase 6: Pair sweep (840-960s)
    dl6 = min(deadline, t0 + 960)
    best_h, best_c5 = _pair_sweep(best_h, rng, dl6, 50)

    # Phase 7: Final Nesterov + micro-polish (960-end)
    dl7 = deadline
    best_h = _nesterov(best_h, 8000000, 1.1, 2, 800, 0.0002, dl7)
    best_c5 = _eval(best_h)
    # Micro coordinate adjustments
    c5c = best_c5
    for _ in range(20000):
        if time.time() > dl7: break
        i = rng.randint(n0); j = (i + 1 + rng.randint(0, 15)) % n0
        d = 5e-6 * rng.randn()
        ni, nj = best_h[i] + d, best_h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            best_h[i], best_h[j] = ni, nj; c5n = _eval(best_h)
            if c5n < c5c - 1e-18: c5c, best_c5 = c5n, c5n
            else: best_h[i], best_h[j] = ni - d, nj + d
    best_h = _repair(best_h, n0)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n0)
