def _repair(h, n):
    # Ensure sum = n/2, 0 <= h <= 1
    target = n / 2.0
    h = np.clip(h, 0, 1)
    # Bisection on shift
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        hc = np.clip(h - mid, 0, 1)
        if hc.sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2, 0, 1)
    # Fine adjustment
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        # Distribute to elements with room
        room = np.where(diff > 0, 1.0 - h, h)
        idx = np.argsort(-room) if diff > 0 else np.argsort(h)
        # Actually distribute proportionally
        for idx_i in np.argsort(-room)[:int(abs(diff)*2)+10]:
            if diff > 0:
                a = min(diff, 1.0 - h[idx_i])
                h[idx_i] += a
                diff -= a
            else:
                a = min(-diff, h[idx_i])
                h[idx_i] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

    dx = 2.0 / n

    return float(c.max())

def _adam(h, n_steps, alpha, lr, top_k):
    n = len(h)
    target = n * 0.5
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    cur_c = _eval(h)
    for t in range(1, n_steps + 1):
        # Compute gradient of softmax
        c = _eval_all(h)
        idx = np.argsort(c)[-top_k:]
        ct = c[idx]
        cm = ct.max()
        s = np.clip(alpha * (ct - cm), -400, 400)
        s -= s.max()
        w = np.exp(s)
        w /= w.sum()
        
        # Gradient
        g = np.zeros(n)
        dx = 2.0 / n
        for ii, k in enumerate(idx):
            if w[ii] < 1e-15:
                continue
            sh = n - 1 - k
            wk = w[ii] * dx
            if sh >= 0:
                for i in range(n - sh):
                    g[i] += wk * (1.0 - h[i + sh])
                if sh > 0:
                    for i in range(sh, n):
                        g[i] -= wk * h[i - sh]
            else:
                sp = -sh
                for i in range(sp, n):
                    g[i] += wk * (1.0 - h[i - sp])
                for i in range(n - sp):
                    g[i] -= wk * h[i + sp]
        
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project
        h_try = _repair(h_new, n)
        c_try = _eval(h_try)
        if c_try < cur_c:
            h = h_try
            cur_c = c_try
    
    return h, cur_c

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) * 0.5, 0, 1)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _eval_all(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / n

def _grad_soft(h, alpha, top_k):
    n = len(h)
    c = _eval_all(h)
    idx = np.argsort(c)[-top_k:]
    ct = c[idx]
    cm = ct.max()
    s = np.clip(alpha * (ct - cm), -400, 400)
    s -= s.max()
    w = np.exp(s)
    w_sum = w.sum()
    if w_sum < 1e-30:
        return np.zeros(n)
    w /= w_sum
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-15:
            continue
        sh = n - 1 - idx[ii]
        wk = w[ii]
        if sh >= 0:
            if sh < n:
                g[sh:] += wk * (1.0 - h[:n - sh])
            if n - sh > 0:
                g[:n - sh] -= wk * h[sh:]
        else:
            sp = -sh
            g[:n - sp] += wk * (1.0 - h[sp:])
            g[sp:] -= wk * h[:n - sp]
    return g

def _adam_opt(h_init, n_steps, alpha, lr_init, top_k, n):
    h = h_init.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.8, 0.999, 1e-8
    cur_c = _eval(h)
    lr = lr_init
    for t in range(1, n_steps + 1):
        g = _grad_soft(h, alpha, top_k)
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * (g * g)
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_try = _repair(h_new, n)
        c_try = _eval(h_try)
        if c_try < cur_c:
            h = h_try
            cur_c = c_try
            lr = min(lr * 1.01, lr_init * 3.0)
        else:
            lr *= 0.7
    return h, cur_c

def _slsqp_opt(h, alpha, maxiter, top_k, n):
    bnds = [(0.0, 1.0)] * n
    target = n * 0.5
    def obj(x):
        c = _eval_all(np.clip(x, 0, 1))
        idx2 = np.argsort(c)[-top_k:]
        ct = c[idx2]
        m2 = ct.max()
        s2 = np.clip(alpha * (ct - m2), -400, 400)
        s2 -= s2.max()
        return m2 + np.log(np.exp(s2).sum()) / alpha
    def grad(x):
        return _grad_soft(np.clip(x, 0, 1), alpha, top_k)
    def sum_f(x):
        return x.sum() - target
    def sum_j(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': sum_f, 'jac': sum_j}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return _repair(res.x, n)
    except Exception:
        return h
