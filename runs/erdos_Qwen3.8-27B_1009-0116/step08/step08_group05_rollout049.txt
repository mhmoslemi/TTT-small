import numpy as np
from scipy.optimize import linprog
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _grad_at(h, k):
    n = len(h)
    sh = k - (n - 1)
    g = np.zeros(n)
    if sh >= 0:
        g[:n-sh] += (1.0 - h[sh:])
        g[sh:] -= h[:n-sh]
    else:
        sp = -sh
        g[sp:] += (1.0 - h[:n-sp])
        g[:n-sp] -= h[sp:]
    return g

def _lp_descent(h, max_steps, eps, topk, deadline):
    n = len(h)
    best_h, best_c5 = h.copy(), _eval(h)
    c_cur = best_c5
    stall = 0
    for step in range(max_steps):
        if time.time() > deadline:
            break
        c = _correl(h)
        cmax = c.max()
        thresh = cmax - 0.003 * max(cmax, 0.01)
        idx = np.where(c >= thresh)[0]
        if len(idx) > topk:
            idx = idx[np.argsort(-c[idx])[:topk]]
        idx = idx[np.argsort(-c[idx])]
        if len(idx) == 0:
            break
        n_active = min(len(idx), 16)
        grads = np.array([_grad_at(h, idx[i]) for i in range(n_active)])
        c_top = c[idx[:n_active]]
        # LP: min t s.t. c_top[i] + grads[i]·d <= t, |d|<=eps, sum(d)=0
        # vars: [d_0..d_{n-1}, t]
        A_ub = np.zeros((n_active, n + 1))
        for i in range(n_active):
            A_ub[i, :n] = grads[i]
            A_ub[i, n] = -1.0
        b_ub = -c_top
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        bnds = [(-eps, eps)] * n + [(cmax - 0.05, cmax + 0.05)]
        try:
            res = linprog(np.zeros(n+1), A_ub=A_ub, b_ub=b_ub,
                          A_eq=A_eq, b_eq=b_eq, bounds=bnds,
                          method='highs', options={'presolve': True})
        except Exception:
            break
        if not res.success or res.x is None:
            break
        d = res.x[:n]
        h_new = _repair(h + d, n)
        c_new = _eval(h_new)
        if c_new < c_cur - 1e-16:
            h, c_cur = h_new, c_new
            stall = 0
            if c_cur < best_c5:
                best_c5, best_h = c_cur, h.copy()
        else:
            stall += 1
            if stall % 40 == 0:

                d = rng2.randn(n) * 0.002
                d -= d.mean()
                h_try = _repair(h + d, n)

                if c_try < c_cur:
                    h, c_cur = h_try, c_try
                stall = 0
            if stall > 200:
                break
    return best_h, best_c5
