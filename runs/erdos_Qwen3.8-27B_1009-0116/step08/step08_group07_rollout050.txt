    return h

def _smax_grad(h, alpha, topk):

        idx = np.argpartition(c, -topk)[-topk:]; ct = c[idx]
    else:
        ct = c
    s = np.clip(alpha * (ct - m), -500, 500)
    return m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha

def _adam(h, alpha0, alpha1, steps, lr, deadline, topk=20):
    n = len(h)
    target = n * 0.5
    c5 = _eval(h)
    best_h, best_c5 = h.copy(), c5
    m1 = np.zeros(n); m2 = np.zeros(n)
    b1, b2 = 0.85, 0.999
    for t in range(steps):
        if time.time() > deadline:
            break
        frac = t / max(steps - 1, 1)
        alpha = alpha0 * (alpha1 / alpha0) ** frac
        g = _smax_grad(h, alpha, topk)
        m1 = b1 * m1 + (1 - b1) * g
        m2 = b2 * m2 + (1 - b2) * g * g
        m1h = m1 / (1 - b1 ** (t + 1))
        m2h = m2 / (1 - b2 ** (t + 1))
        hn = _proj(h - lr / (np.sqrt(m2h) + 1e-8) * m1h, target)
        c5n = _eval(hn)
        if c5n < best_c5:
            best_c5, best_h = c5n, hn.copy()
        h = hn; c5 = c5n
    return _repair(best_h, n), best_c5

def _nesterov(h, steps, step0, deadline, alpha=50000, topk=10, beta=0.88):
    n = len(h)
    target = n * 0.5
    f = _eval(h)
    v = np.zeros(n)
    step = step0
    best_h, best_f = h.copy(), f
    for _ in range(steps):
        if time.time() > deadline:
            break
        g = _smax_grad(h, alpha, topk)
        hp = h.copy()
        v = beta * v + (1.0 - beta) * g
        hn = _proj(h - step * v, target)
        fn = _eval(hn)
        if fn < f:
            h, f = hn, fn
            step = min(step * 1.01, step0 * 20.0)
            if f < best_f:
                best_f, best_h = f, h.copy()
        else:
            step *= 0.5
            h = _proj(hp - step * v, target)
            f = _eval(h)
            if f < best_f:
                best_f, best_h = f, h.copy()
    return _repair(best_h, n), best_f

def _slsqp(h, alpha, topk, maxit, deadline):
    n = len(h)
    if time.time() > deadline:
        return h
    try:
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
                 'jac': lambda x: np.ones(n)}]
        r = minimize(lambda x: _smax_obj(x, alpha, topk), h,
                     jac=lambda x: _smax_grad(x, alpha, topk),
                     method='SLSQP', bounds=[(0.0, 1.0)] * n,
                     constraints=cons,
                     options={'maxiter': maxit, 'ftol': 1e-18})
        if _eval(r.x) <= _eval(h) + 1e-15:
            return _repair(r.x, n)
    except Exception:
        pass
    return h

def _polish(h, niter, rng, deadline, scale=5e-6):
    nc = len(h); c5c = _eval(h); bh = h.copy(); bc = c5c
    for _ in range(niter):
        if time.time() > deadline:
            break
        i = rng.randint(nc)
        j = (i + 1 + rng.randint(0, 15)) % nc
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
            h[i], h[j] = ni, nj
            c5n = _eval(h)
            if c5n < c5c - 1e-19:
                c5c = c5n
                if c5n < bc:
                    bc, bh = c5n, h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return bh, bc

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _correl(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _smax_grad(h, alpha, topk):
    n = len(h)
    c = _correl(h)
    m = c.max()
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:].copy()
        ct = c[idx]
    else:
        idx = np.arange(len(c)); ct = c
    s = np.clip(alpha * (ct - m), -500, 500)
    w = np.exp(s - s.max()); w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for ii in range(len(idx)):
        if w[ii] < 1e-14:
            continue
        k = idx[ii]; sh = (n - 1) - k; wk = w[ii] * dx

            g[sp:] += wk * (1.0 - h[:-sp])
            g[:n - sp] -= wk * h[sp:]
    return g
