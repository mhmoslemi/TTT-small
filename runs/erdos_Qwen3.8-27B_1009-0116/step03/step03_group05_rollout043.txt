from scipy.optimize import linprog
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5_val(h, dx):
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) < 1e-12:
        return h
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                amt = min(diff, 1.0 - h[idx])
                h[idx] += amt
            else:
                amt = min(-diff, h[idx])
                h[idx] -= amt
            diff = target - h.sum()
            if abs(diff) < 1e-14:
                break
    return h

def _grad_lag(h, L):
    n = len(h)
    g = np.zeros(n)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[:n-L] += (1.0 - h[L:])
        g[L:] -= h[:n-L]
    else:
        M = -L
        g[M:] += (1.0 - h[:n-M])
        g[:n-M] -= h[M:]
    return g

def _lp_shift(h, top_idx, m):
    """Solve LP for best perturbation. Returns delta or None."""
    n = len(h)
    c = _corr(h)
    top_lags = [int(ki - (n-1)) for ki in top_idx[:m]]
    top_cv = [c[ki] for ki in top_idx[:m]]
    
    # Build LP: min t s.t. g_i·delta - t <= -c_i, sum(delta)=0, box
    A_ub = np.zeros((m, n+1))
    b_ub = np.zeros(m)
    for i in range(m):
        g = _grad_lag(h, top_lags[i])
        A_ub[i, :n] = g
        A_ub[i, n] = -1.0
        b_ub[i] = -top_cv[i]
    
    # Box constraints on delta
    A_box = np.zeros((2*n, n+1))
    b_box = np.zeros(2*n)
    A_box[:n, :n] = np.eye(n)
    b_box[:n] = 1.0 - h
    A_box[n:, :n] = -np.eye(n)
    b_box[n:] = h
    
    A_all = np.vstack([A_ub, A_box])
    b_all = np.concatenate([b_ub, b_box])
    
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    
    c_obj = np.zeros(n+1)
    c_obj[n] = 1.0
    
    res = linprog(c_obj, A_ub=A_all, b_ub=b_all, A_eq=A_eq, b_eq=np.array([0.0]),
                  bounds=[(None, None)] * (n+1), method='highs')
    if res.status != 0:
        return None
    return res.x[:n]

    """Block pattern: alternating 1s and 0s in n_clust groups."""
    h = np.zeros(n)
    block = n // n_clust
    for i in range(0, n, 2 * block):
        h[i:min(i+block, n)] = 1.0
    return _repair(h, n)

            c5_cur = cn
        else:

        T *= 0.9999
