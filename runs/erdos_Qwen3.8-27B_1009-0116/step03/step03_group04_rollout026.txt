def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -500.0, 500.0)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) > 1e-12:
        diff = target - s
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _subgrad_top(h, top_lags, n):

                g[j] -= h[i]
