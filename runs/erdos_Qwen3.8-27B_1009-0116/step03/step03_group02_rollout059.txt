```python
from scipy.optimize import linprog
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _grad_all(h, n):
    """Compute gradients for all lags. Returns (2n-1, n) array."""
    n_corr = 2 * n - 1
    grads = np.zeros((n_corr, n))
    for k in range(n_corr):
        lag = k - (n - 1)
        idx1 = np.arange(n) + lag
        m1 = (idx1 >= 0) & (idx1 < n)
        grads[k, m1] += (1.0 - h[idx1[m1]])
        idx2 = np.arange(n) - lag
        m2 = (idx2 >= 0) & (idx2 < n)
        grads[k, m2] -= h[idx2[m2]]
    return grads

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:

                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h
```

```python
def _grad_all(h, n):
    n_corr = 2*n - 1
    L = np.arange(n_corr) - (n-1)  # lags: -(n-1), ..., 0, ..., n-1
    all_idx = np.arange(n)
    grads = np.zeros((n_corr, n))
    for k in range(n_corr):
        Lk = L[k]
        idx_plus = all_idx + Lk  # l + L for each l
        mask_plus = (idx_plus >= 0) & (idx_plus < n)
        grads[k, mask_plus] = (1.0 - h[idx_plus[mask_plus]])
        idx_minus = all_idx - Lk  # l - L for each l
        mask_minus = (idx_minus >= 0) & (idx_minus < n)
        grads[k, mask_minus] -= h[idx_minus[mask_minus]]
    return grads
```

```python
import numpy as np
from scipy.optimize import linprog
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _grad_all(h, n):
    n_corr = 2 * n - 1
    L = np.arange(n_corr) - (n - 1)
    idx = np.arange(n)
    grads = np.zeros((n_corr, n))
    for k in range(n_corr):
        Lk = L[k]
        ip = idx + Lk
        mp = (ip >= 0) & (ip < n)
        grads[k, mp] = (1.0 - h[ip[mp]])
        im = idx - Lk
        mm = (im >= 0) & (im < n)
        grads[k, mm] -= h[im[mm]]
    return grads

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx_ in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx_])
                h[idx_] += a
                diff -= a
            else:
                a = min(-diff, h[idx_])
                h[idx_] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _slp_solve(h, n):
    corr = _corr(h)
    n_corr = 2 * n - 1
    grads = _grad_all(h, n)
    b = corr - grads @ h
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.zeros((n_corr + 2 * n, n + 1))
    b_ub = np.zeros(n_corr + 2 * n)
    A_ub[:n_corr, :n] = grads
    A_ub[:n_corr, n] = -1.0
    b_ub[:n_corr] = -b
    rows = np.arange(n)
    A_ub[n_corr:n_corr + n, rows] = 1.0
    b_ub[n_corr:n_corr + n] = 1.0
    A_ub[n_corr + n:, rows] = -1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = [(0.0, 1.0)] * n + [(None, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return _repair(res.x[:n], n)
    except Exception:
        pass
    return h

def _slp_loop(h, n, dx, max_iter, t_end, damp_init=1.0, damp_min=0.3):
    best_c5 = _c5(h, dx)
    best_h = h.copy()
    h_cur = h.copy()
    for it in range(max_iter):
        if time.time() > t_end:
            break
        damp = max(damp_min, damp_init * (1.0 - 0.02 * it))
        h_lp = _slp_solve(h_cur, n)
        h_new = (1.0 - damp) * h_cur + damp * h_lp
        h_new = _repair(h_new, n)
        chg = np.max(np.abs(h_new - h_cur))
        h_cur = h_new
        c5 = _c5(h_new, dx)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h_new.copy()
        if chg < 1e-11:
            break
    return best_h, best_c5

def _fvec(h, i, j, n):
    n_corr = 2 * n - 1
    L = np.arange(n_corr) - (n - 1)
    gd = np.zeros(n_corr)
    for k in range(n_corr):
        Lk = L[k]
        ip, im = i + Lk, i - Lk
        jp, jm = j + Lk, j - Lk
        if 0 <= ip < n:
            gd[k] += (1.0 - h[ip])
        if 0 <= im < n:
            gd[k] -= h[im]
        if 0 <= jp < n:
            gd[k] -= (1.0 - h[jp])
        if 0 <= jm < n:
            gd[k] += h[jm]
    return gd

def _sweep(h, n, t_end, n_coarse=20, n_bis=30):
    corr = _corr(h)
    improved = False
    order = np.random.permutation(n)
    for ai in range(n):
        if time.time() > t_end:
            break
        i = order[ai]
        for aj in range(ai + 1, n):
            if time.time() > t_end:
                break
            j = order[aj]
            hi, hj = h[i], h[j]
            d_min = max(-hi, hj - 1.0)
            d_max = min(1.0 - hi, hj)
            if d_max - d_min < 1e-13:
                continue
            gd = _fvec(h, i, j, n)
            ds = np.linspace(d_min, d_max, n_coarse)
            vals = np.max(corr[None, :] + ds[:, None] * gd[None, :], axis=1)
            bi = np.argmin(vals)
            if vals[bi] >= np.max(corr) - 1e-14:
                continue
            lo = max(d_min, ds[max(0, bi - 1)])
            hi_b = min(d_max, ds[min(n_coarse - 1, bi + 1)])
            for _ in range(n_bis):
                mid = 0.5 * (lo + hi_b)
                if np.max(corr + mid * gd) <= np.max(corr + hi_b * gd):
                    hi_b = mid
                else:
                    lo = mid
            d_best = 0.5 * (lo + hi_b)
            h[i] = hi + d_best
            h[j] = hj - d_best
            corr = corr + d_best * gd
            improved = True
    return h, corr, improved

def _fourier_init(n, seed=0):
    rng = np.random.RandomState(seed)
    x = np.linspace(0, 2.0, n, endpoint=False)
    h = np.zeros(n)
    h += 0.5 + 0.45 * np.cos(np.pi * x)
    h += 0.08 * np.cos(2.0 * np.pi * x) + 0.04 * np.cos(3.0 * np.pi * x)
    h += rng.normal(0, 0.03, n)
    return _repair(np.clip(h, 0, 1), n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0

    best_h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_c5 = _c5(best_h, dx)

    # Phase 1: SLP from multiple inits
    inits = [best_h.copy()]
    inits.append(_fourier_init(n, 1))
    inits.append(_fourier_init(n, 2))
    t_phase1 = t0 + budget_s * 0.25
    for h_init in inits:
        if time.time() > t_phase1:
            break
        h_slp, c5_slp = _slp_loop(h_init, n, dx, 200, t_phase1)
        if c5_slp < best_c5 - 1e-13:
            best_c5 = c5_slp
            best_h = h_slp.copy()

    # Phase 2: Coordinate sweeps + periodic SLP
    t_phase2 = t0 + budget_s * 0.70
    stall = 0
    while time.time() < t_phase2:
        h = best_h.copy()
        t_sweep_end = min(time.time() + 30, t_phase2)
        h, corr, imp = _sweep(h, n, t_sweep_end, 22, 32)
        c5 = float(np.max(corr) * dx)
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall >= 3 and time.time() < t_phase2 - 20:
            h_slp, c5_slp = _slp_loop(best_h, n, dx, 100, min(time.time() + 30, t_phase2 - 10))
            if c5_slp < best_c5 - 1e-13:
                best_c5 = c5_slp
                best_h = h_slp.copy()
                stall = 0
        if stall >= 6 and time.time() < t_phase2 - 20:
            h_pert = best_h + 0.03 * np.random.randn(n)
            h_pert = _repair(h_pert, n)
            h_slp, c5_slp = _slp_loop(h_pert, n, dx, 80, min(time.time() + 25, t_phase2 - 10))
            h_s, c_s, _ = _sweep(h_slp, n, min(time.time() + 15, t_phase2 - 5), 20, 28)
            if c_s < best_c5 - 1e-13:
                best_c5 = c_s
                best_h = h_s.copy()
            stall = 0

    # Phase 3: Final high-precision sweeps
    t_final = t0 + budget_s - 15
    for _ in range(5):
        if time.time() > t_final:
            break
        h = best_h.copy()
        h, corr, _ = _sweep(h, n, min(time.time() + 18, t_final), 30, 40)
        c5 = float(np.max(corr) * dx)
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            best_h = h.copy()

    # Phase 4: Final SLP polish
    if time.time() < t0 + budget_s - 3:
        h_slp, c5_slp = _slp_loop(best_h, n, dx, 300, t0 + budget_s - 3, damp_init=0.3, damp_min=0.1)
        if c5_slp < best_c5 - 1e-14:
            best_c5 = c5_slp
            best_h = h_slp.copy()

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
```
