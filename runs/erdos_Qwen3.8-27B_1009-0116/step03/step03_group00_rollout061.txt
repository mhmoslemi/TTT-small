The max of these linear functions is minimized at a point where two of them cross, or at a boundary (t such that h[i]+t or h[j]-t hits 0 or 1).

The breakpoints are:

- t = h[j] (h[j]-t = 0)
- t = -h[j] (h[j]-t = 1)
- All t where c_k(t) = c_m(t) for some k,m

```python
    #   if lag >= 0: (1-h[i+lag]) if i+lag < n, minus h[i-lag] if i>=lag
    #   if lag < 0: -(h[i+|lag|]) if i+|lag| < n, plus (1-h[i-|lag|]) if i>=|lag|

    # But that's 2n evaluations per pair, which is expensive.

    # For lag L (can be negative):
    # c_L = sum_{valid i} h[i] * (1-h[i+L])
    # dc_L/dh[p] = (1-h[p+L])*1[p+L valid] - h[p-L]*1[p-L valid]

        if 0 <= i+L < n: di += (1.0 - h[i+L])
        if 0 <= i-L < n: di -= h[i-L]

        if 0 <= j-L < n: dj -= h[j-L]
        d[k] = di - dj

    t_lo = max(-h[i], h[j] - 1.0)

    candidates = [t_lo, t_hi]

        h[j] += t
    
    # Also try the endpoint of the piecewise linear minimum
    # (which is where the slope changes sign)
    # Actually, the candidates above should capture this
    
    return best_t
```

```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, n):
    return float(np.max(_corr(h))) * (2.0 / n)

def _cvals(h, n):
    return _corr(h) * (2.0 / n)

def _repair(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return h

def _grad_c_k(x, n, lag):
    g = np.zeros(n)
    L = abs(lag)
    if L >= n:
        return g
    if lag >= 0:
        g[:n-L] += (1.0 - x[L:])
        g[L:] -= x[:n-L]
    else:
        g[L:] += (1.0 - x[:n-L])
        g[:n-L] -= x[L:]
    return g

def _lse_obj(x, n, alpha):
    c = _cvals(x, n)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _lse_grad(x, n, alpha):
    dx_val = 2.0 / n
    c = _cvals(x, n)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    nc = len(c)
    for k in range(nc):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        g += wk * _grad_c_k(x, n, lag)
    return g * dx_val

def _slsqp(h_init, n, alpha, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    return minimize(_lse_obj, h_init, jac=lambda x: _lse_grad(x, n, alpha),
                    method='SLSQP', bounds=bounds, constraints=cons,
                    options={'maxiter': maxiter, 'ftol': ftol})

def _pair_search(h, n, budget_end, rng, max_passes=200):
    """Greedy pair-wise mass transfer search."""
    dx_val = 2.0 / n
    c5_cur = _c5(h, n)
    deltas = np.array([0.0005, 0.001, 0.003, 0.01, -0.0005, -0.001, -0.003, -0.01])
    for p in range(max_passes):
        if time.time() > budget_end:
            break
        improved = False
        # Randomize the order each pass
        idx = np.random.permutation(n)
        for ii in range(n):
            if time.time() > budget_end:
                break
            i = idx[ii]
            hi = h[i]
            for jj in range(ii+1, n):
                if time.time() > budget_end:
                    break
                j = idx[jj]
                hj = h[j]
                for d in deltas:
                    ni, nj = hi + d, hj - d
                    if ni < -1e-12 or ni > 1+1e-12 or nj < -1e-12 or nj > 1+1e-12:
                        continue
                    h[i], h[j] = ni, nj
                    c = _c5(h, n)
                    if c < c5_cur - 1e-15:
                        c5_cur = c
                        hi, hj = ni, nj
                        improved = True
                    else:
                        h[i], h[j] = hi, hj
            h[i] = hi
        if not improved:
            break
    return h, c5_cur

def _interpolate(h_old, n_old, n_new):
    x_old = np.linspace(0, 2.0, n_old, endpoint=False) + 1.0/n_old
```

```python
def _pair_best_t(h, n, i, j):
    """Exact 1D minimization for pair (i,j)."""
    dx = 2.0 / n
    t_lo = max(-h[i], h[j] - 1.0)
    t_hi = min(1.0 - h[i], h[j])
    if t_lo >= t_hi - 1e-12:
        return 0.0
    
    c0 = _corr(h) * dx
    nc = len(c0)
    
    # d[k] = dc_k/dt
    d = np.empty(nc)
    for k in range(nc):
        lag = k - (n - 1)
        di = (1.0 - h[i+lag]) if (0 <= i+lag < n) else 0.0
        di -= h[i-lag] if (0 <= i-lag < n) else 0.0
        dj = (1.0 - h[j+lag]) if (0 <= j+lag < n) else 0.0
        dj -= h[j-lag] if (0 <= j-lag < n) else 0.0
        d[k] = (di - dj) * dx
    
    # f(t) = max_k (c0[k] + t * d[k])
    # This is convex piecewise linear
    # Find minimum by finding where slope crosses 0
    
    # Start at t=0, track the active (max) component
    t = 0.0
    for _iter in range(nc + 1):
        vals = c0 + t * d
        kmax = np.argmax(vals)
        slope = d[kmax]
        if abs(slope) < 1e-12:
            break
        if slope > 0:  # f increases with t, move left
            # Find t where another k catches up from below
            # c0[m] + t*d[m] = c0[kmax] + t*d[kmax]
            # t = (c0[kmax] - c0[m]) / (d[m] - d[kmax])
            best = t_lo
            for m in range(nc):
                if m == kmax: continue
                denom = d[m] - d[kmax]
                if denom > 1e-15:
                    tm = (c0[kmax] - c0[m]) / denom
                    if t_lo <= tm < t:
                        best = max(best, tm)
            t = best
        else:  # slope < 0, move right
            best = t_hi
            for m in range(nc):
                if m == kmax: continue
                denom = d[kmax] - d[m]
                if denom > 1e-15:
                    tm = (c0[m] - c0[kmax]) / denom
                    if t < tm <= t_hi:
                        best = min(best, tm)
            t = best
    
    t = max(t_lo, min(t_hi, t))
    # Verify
    h[i] += t; h[j] -= t
    ft = _c5(h, n)
    h[i] -= t; h[j] += t
    f0 = _c5(h, n)
    return t if ft < f0 - 1e-15 else 0.0
```

    if abs(diff) > 1e-14:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return h

def _lse_obj(x, n, alpha):
    c = _cvals(x, n)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _lse_grad(x, n, alpha):
    dx = 2.0 / n
    c = _cvals(x, n)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    g += w[n - 1] * (1.0 - 2.0 * x)
    for ml in range(1, n):
        wp = w[n - 1 + ml]
        wm = w[n - 1 - ml]
        if wp > 1e-14:
            g[:n - ml] += wp * (1.0 - x[ml:])
            g[ml:] -= wp * x[:n - ml]
        if wm > 1e-14:
            g[ml:] += wm * (1.0 - x[:n - ml])
            g[:n - ml] -= wm * x[ml:]
    return g * dx

def _slsqp(h_init, n, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [
        {'type': 'eq',
         'fun': lambda x: np.sum(x) - n / 2.0,
         'jac': lambda x: np.ones(n)}
    ]
    return minimize(_lse_obj, h_init,
                    jac=lambda x: _lse_grad(x, n, alpha),
                    method='SLSQP', bounds=bounds,
                    constraints=cons,
                    options={'maxiter': maxiter, 'ftol': 1e-16})

def _pair_best_t(h, n, i, j):
    dx = 2.0 / n
    t_lo = max(-h[i], h[j] - 1.0)
    t_hi = min(1.0 - h[i], h[j])
    if t_lo >= t_hi - 1e-12:
        return 0.0
    c0 = _corr(h) * dx
    nc = len(c0)
    d = np.empty(nc)
    for k in range(nc):
        lag = k - (n - 1)
        di = 0.0
        if 0 <= i + lag < n: di += 1.0 - h[i + lag]
        if 0 <= i - lag < n: di -= h[i - lag]
        dj = 0.0
        if 0 <= j + lag < n: dj += 1.0 - h[j + lag]
        if 0 <= j - lag < n: dj -= h[j - lag]
        d[k] = (di - dj) * dx
    t = 0.0
    for _ in range(nc + 1):
        vals = c0 + t * d
        kmax = np.argmax(vals)
        slope = d[kmax]
        if abs(slope) < 1e-12:
            break
        if slope > 0:
            best = t_lo
            for m in range(nc):
                if m == kmax: continue
                denom = d[m] - d[kmax]
                if denom > 1e-15:
                    tm = (c0[kmax] - c0[m]) / denom
                    if t_lo <= tm < t - 1e-15:
                        best = max(best, tm)
            t = best
        else:
            best = t_hi
            for m in range(nc):
                if m == kmax: continue
                denom = d[kmax] - d[m]
                if denom > 1e-15:
                    tm = (c0[m] - c0[kmax]) / denom
                    if t < tm <= t_hi + 1e-15:
                        best = min(best, tm)
            t = best
    t = max(t_lo, min(t_hi, t))
    h[i] += t
    h[j] -= t
    ft = _c5(h, n)
    h[i] -= t
    h[j] += t
    f0 = _c5(h, n)
    return t if ft < f0 - 1e-15 else 0.0
