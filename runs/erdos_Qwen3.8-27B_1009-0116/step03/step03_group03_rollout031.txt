import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return np.max(_corr(h)) * dx

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
            else:
                add = max(diff, h[idx] - 1.0)
            h[idx] += add
            diff -= add
            if abs(diff) < 1e-12:
                break
    return h

def _alpha_matrix(h, n):
    """Compute gradient matrix: alpha[m, i] = df_m/dh_i"""
    alpha = np.zeros((2*n-1, n))
    for m in range(2*n-1):
        k = m - (n-1)
        if k >= 0:
            alpha[m, :n-k] += (1.0 - h[k:])
            alpha[m, k:] -= h[:n-k]
        else:
            kp = -k
            alpha[m, kp:] += (1.0 - h[:n-kp])
            alpha[m, :n-kp] -= h[kp:]
    return alpha

def _pair_optimal_delta(c, alpha_diff, h_i, h_j):
    """Find optimal delta for pair update.
    c: current correlation values
    alpha_diff: slope for each lag = alpha[m,i] - alpha[m,j]
    h_i, h_j: current values
    Returns optimal delta (positive means h_i increases)
    """
    dmin = -h_j
    dmax = 1.0 - h_i
    if dmin >= dmax:
        return 0.0
    
    # Bisection on 1D convex piecewise-linear function
    lo, hi = dmin, dmax
    for _ in range(60):
        mid = (lo + hi) / 2.0
        vals = c + mid * alpha_diff
        max_val = vals.max()
        active_mask = vals >= max_val - 1e-12
        if active_mask.any():
            min_a = alpha_diff[active_mask].min()
            max_a = alpha_diff[active_mask].max()
        else:
            min_a = max_a = 0.0
        if min_a <= 1e-9 and max_a >= -1e-9:
            return mid
        elif max_a < 0:  # all active slopes negative, go right
            lo = mid
        else:  # all active slopes positive, go left
            hi = mid
    return (lo + hi) / 2.0

def _upsample(h_coarse, n_fine, target_fine):
    """Upsample coarse solution to fine grid using linear interpolation"""
    n_coarse = len(h_coarse)
    x_coarse = np.linspace(0, 2, n_coarse, endpoint=False) + (1.0/n_coarse)
    x_fine = np.linspace(0, 2, n_fine, endpoint=False) + (1.0/n_fine)
    h_fine = np.interp(x_fine, x_coarse, h_coarse)
    h_fine = np.clip(h_fine, 0, 1)
    h_fine *= target_fine / max(1e-12, h_fine.sum())
    h_fine = np.clip(h_fine, 0, 1)
    # Repair
    return _repair(h_fine, n_fine)

def _coord_descent(h, n, dx, max_iters=50000, verbose=False):
    """Coordinate descent with exact pair updates"""
    h = h.copy()
    target = n / 2.0
    c = _corr(h)
    c5_cur = c.max() * dx
    A = _alpha_matrix(h, n)
    rng = np.random.RandomState(42)

        i = rng.randint(n)
        j = (i + 1 + rng.randint(n-1)) % n
        
        alpha_diff = A[:, i] - A[:, j]
        delta = _pair_optimal_delta(c, alpha_diff, h[i], h[j])

        h[j] -= delta

        # Update
        c = _corr(h)
        new_c5 = c.max() * dx
        A = _alpha_matrix(h, n)
        c5_cur = new_c5
    
    return h, c5_cur

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5_val(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
            else:
                add = max(diff, h[idx] - 1.0)
            h[idx] += add
            diff -= add
            if abs(diff) < 1e-13:
                break
    return h

def _alpha_matrix(h, n):
    alpha = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        k = m - (n - 1)
        if k >= 0:
            alpha[m, :n-k] += (1.0 - h[k:])
            alpha[m, k:] -= h[:n-k]
        else:
            kp = -k
            alpha[m, kp:] += (1.0 - h[:n-kp])
            alpha[m, :n-kp] -= h[kp:]
    return alpha

def _pair_delta(c, a_diff, hi, hj):
    dmin = -hj
    dmax = 1.0 - hi
    if dmin >= dmax - 1e-14:
        return 0.0
    lo, hi_ = dmin, dmax
    for _ in range(45):
        mid = 0.5 * (lo + hi_)
        vals = c + mid * a_diff
        mv = vals.max()
        act = (vals >= mv - 1e-11).nonzero()[0]
        if len(act) == 0:
            act = np.array([int(np.argmax(vals))])
        mn = a_diff[act].min()
        mx = a_diff[act].max()
        if mn <= 1e-9 and mx >= -1e-9:
            return mid
        if mx < 0:
            lo = mid
        else:
            hi_ = mid
    return 0.5 * (lo + hi_)

def _smooth_obj(x, alpha_par):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha_par * (c - m), -500, 500)
    return float(m + np.log(np.sum(np.exp(s))) / alpha_par)

def _sum_res(x, target):
    return float(np.sum(x) - target)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0
    rng = np.random.RandomState(seed)

    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _c5_val(best_h, dx)

    def tleft():
        return budget_s - (time.time() - t0)

    # Phase 1: Random coordinate descent with exact pair updates
    c = _corr(h)
    A = _alpha_matrix(h, n)
    c5_cur = c.max() * dx
    stag = 0
    for it in range(800000):
        if tleft() < 350:
            break
        i = rng.randint(n)
        j = (i + rng.randint(1, n)) % n
        if j == i:
            j = (i + 1) % n
        a_diff = A[:, i] - A[:, j]
        delta = _pair_delta(c, a_diff, h[i], h[j])
        if abs(delta) < 1e-13:
            stag += 1
            if stag > 300:
                pi = rng.randint(n)
                pj = (pi + 1 + rng.randint(n-1)) % n
                hd = rng.uniform(0.005, 0.02) * (1 if rng.rand() < 0.5 else -1)
                if 0 <= h[pi]+hd <= 1 and 0 <= h[pj]-hd <= 1:
                    h[pi] += hd
                    h[pj] -= hd
                    h = _repair(h, n)
                    c = _corr(h)
                    A = _alpha_matrix(h, n)
                stag = 0
            continue
        h[i] += delta
        h[j] -= delta
        h = _repair(h, n)
        c = _corr(h)
        new_c5 = c.max() * dx
        if new_c5 < c5_cur - 1e-16:
            c5_cur = new_c5
            stag = 0
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        A = _alpha_matrix(h, n)

    # Phase 2: Systematic sweeps over all pairs
    if tleft() > 150:
        max_sweeps = 15
        for sweep in range(max_sweeps):
            if tleft() < 150:
                break
            c = _corr(h)
            A = _alpha_matrix(h, n)
            c5_cur = c.max() * dx
            any_imp = False
            for i in range(n):
                for j in range(i + 1, n):
                    a_diff = A[:, i] - A[:, j]
                    delta = _pair_delta(c, a_diff, h[i], h[j])
                    if abs(delta) < 1e-12:
                        continue
                    h[i] += delta
                    h[j] -= delta
                    h = _repair(h, n)
                    c = _corr(h)
                    nc5 = c.max() * dx
                    if nc5 < c5_cur - 1e-16:
                        c5_cur = nc5
                        any_imp = True
                        if nc5 < best_c5:
                            best_c5 = nc5
                            best_h = h.copy()
                    A = _alpha_matrix(h, n)
            if not any_imp:
                break

    # Phase 3: SLSQP polish on smoothed objective
    if tleft() > 80:
        bounds = [(0.0, 1.0)] * n
        for alpha_par in [20000, 100000]:
            if tleft() < 80:
                break
            try:
                res = minimize(_smooth_obj, best_h, args=(alpha_par,),
                              method='SLSQP', bounds=bounds,
                              constraints=[{'type': 'eq', 'fun': _sum_res, 'args': (target,)}],
                              options={'maxiter': 400, 'ftol': 1e-15, 'iprint': 0})
                hc = _repair(res.x, n)
                c5 = _c5_val(hc, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = hc
            except Exception:
                pass

    # Phase 4: Fine random exploration with varying step sizes
    if tleft() > 25:
        h = best_h.copy()
        c5_cur = _c5_val(h, dx)
        step_sets = [[0.0005, 0.001, 0.002, 0.005, -0.0005, -0.001, -0.002, -0.005]]
        for _ in range(300000):
            if tleft() < 25:
                break
            i = rng.randint(n)
            j = (i + 1 + rng.randint(n - 1)) % n
            steps = step_sets[0]
            for d in steps:
                ni, nj = h[i] + d, h[j] - d
                if 0 <= ni <= 1 and 0 <= nj <= 1:
                    h[i], h[j] = ni, nj
                    nc5 = _c5_val(h, dx)
                    if nc5 < c5_cur - 1e-16:
                        c5_cur = nc5
                        if nc5 < best_c5:
                            best_c5 = nc5
                            best_h = h.copy()
                        break
                    else:
                        h[i], h[j] = ni - d, nj + d
        final_c5 = _c5_val(h, dx)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = h.copy()

    # Phase 5: Final coordinate descent polish
    if tleft() > 15:
        h = best_h.copy()
        c = _corr(h)
        A = _alpha_matrix(h, n)
        c5_cur = c.max() * dx
        for it in range(200000):
            if tleft() < 15:
                break
            i = rng.randint(n)
            j = (i + 1 + rng.randint(n - 1)) % n
            a_diff = A[:, i] - A[:, j]
            delta = _pair_delta(c, a_diff, h[i], h[j])
            if abs(delta) < 1e-13:
                continue
            h[i] += delta
            h[j] -= delta
            h = _repair(h, n)
            c = _corr(h)
            nc5 = c.max() * dx
            if nc5 < c5_cur - 1e-16:
                c5_cur = nc5
                if nc5 < best_c5:
                    best_c5 = nc5
                    best_h = h.copy()
            A = _alpha_matrix(h, n)

    best_h = _repair(best_h, n)
    best_c5 = _c5_val(best_h, dx)
    return (best_h, best_c5, n)
