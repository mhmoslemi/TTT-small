def _obj_grad(h, n, alpha):
    h = np.asarray(h, dtype=float)
    dx_val = 2.0 / n
    c_raw = np.correlate(h, 1.0 - h, mode='full')
    c = c_raw * dx_val
    m = np.max(c)
    s = alpha * (c - m)
    s = np.clip(s, -400, 400)
    e = np.exp(s)
    w = e / e.sum()

    g = np.zeros(n)

        if w[k] < 1e-14:
            continue

        # d/dh[j]: for term h[i]*(1-h[i+ell])
        #   if i = j: contributes -h[i+ell] = -h[j+ell] (if j+ell valid)
        #   if i+ell = j i.e. i = j-ell: contributes (1-h[i]) = (1-h[j-ell]) (if j-ell valid)

        # h[j-ell] term: j-ell ranges over [0, n) => j in [ell, n+ell) => j in [max(0,ell), min(n, n+ell))

        # For valid i: 0≤j-ell<n and 0≤j<n

                    improved = True
                else:
                    h[i], h[j] = hi, hj
            h[i] = hi
        if not improved:
            break
    return h, c5_cur

def _micro_search(h, n, t0, timeout, rng):
    c5_cur = _c5(h, n)
    tiny = [0.0002, 0.0004, 0.0008, -0.0002, -0.0004, -0.0008]
    while time.time() - t0 < timeout:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = tiny[rng.randint(len(tiny))]
        hi, hj = h[i], h[j]
        ni, nj = hi + d, hj - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c = _c5(h, n)
            if c < c5_cur:
                c5_cur = c
            else:
                h[i], h[j] = hi, hjj = hj
                h[j] = hj
    return h, c5_cur

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    best_h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_c5 = _c5(best_h, n)

    def elapsed():
        return time.time() - t0

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq',
             'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]

    # Phase 1: SLSQP sweep (multiple alphas, multiple restarts)
    alphas = [1500, 5000, 15000, 40000, 100000, 250000]
    for alpha in alphas:
        if elapsed() > budget_s - 350:
            break
        for r in range(3):
            if elapsed() > budget_s - 350:
                break
            if r == 0:
                h_init = best_h.copy()
            else:
                h_init = _repair(best_h + 0.04 * (r + 1) * rng.randn(n), n)
            res = _slsqp_call(h_init, n, alpha, 700, _smooth_obj, bounds, cons)
            if res is not None:
                hc = _repair(res.x, n)
                c5 = _c5(hc, n)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = hc.copy()

    # Phase 2: Top-k SLSQP
    if elapsed() < budget_s - 300:
        for k_val in [3, 5, 8]:
            if elapsed() > budget_s - 280:
                break
            for r in range(3):
                if elapsed() > budget_s - 280:
                    break
                h_init = best_h.copy() if r == 0 else _repair(best_h + 0.03 * rng.randn(n), n)
                try:
                    res = minimize(lambda x, nn, a, kk: _topk_obj(x, nn, kk), h_init,
                                   args=(n, k_val),
                                   method='SLSQP', bounds=bounds, constraints=cons,
                                   options={'maxiter': 500, 'ftol': 1e-18})
                    hc = _repair(res.x, n)
                    c5 = _c5(hc, n)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = hc.copy()
                except Exception:
                    pass

    # Phase 3: SA + SLSQP interleaved (main optimization phase)
    sa_budget_per = 20  # seconds per SA segment
    n_segments = 12
    for seg in range(n_segments):
        if elapsed() > budget_s - 150:
            break
        # SA exploration
        sa_end = t0 + elapsed() + sa_budget_per
        if sa_end > t0 + budget_s - 120:
            sa_end = t0 + budget_s - 120
        h_sa, c5_sa, best_sa = _run_sa(best_h, n, best_c5, sa_budget_per,
                                       t0, t0 + budget_s - 120 - elapsed(),
                                       rng, T0=1.5e-4 * (1 + 0.5 * seg % 3))
        if best_sa < best_c5:
            best_c5 = best_sa
            best_h = h_sa.copy()
        # SLSQP exploitation
        if elapsed() < budget_s - 100:
            alpha = 30000 + 20000 * (seg % 4)
            res = _slsqp_call(best_h.copy(), n, alpha, 500, _smooth_obj, bounds, cons)
            if res is not None:
                hc = _repair(res.x, n)
                c5 = _c5(hc, n)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = hc.copy()

    # Phase 4: Intensive fine search
    if elapsed() < budget_s - 60:
        best_h, best_c5 = _fine_search(best_h.copy(), n, t0, t0 + budget_s - 40, rng)

    # Phase 5: SLSQP polish after fine search
    if elapsed() < budget_s - 30:
        for alpha in [50000, 150000]:
            if elapsed() > budget_s - 20:
                break
            res = _slsqp_call(best_h.copy(), n, alpha, 400, _smooth_obj, bounds, cons)
            if res is not None:
                hc = _repair(res.x, n)
                c5 = _c5(hc, n)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = hc.copy()

    # Phase 6: Try higher resolution n=100
    if elapsed() < budget_s - 50:
        n2 = 100
        x_old = np.linspace(0, 2.0, n, endpoint=False)
        x_new = np.linspace(0, 2.0, n2, endpoint=False)
        h2 = np.interp(x_new, x_old, best_h)
        h2 = _repair(h2, n2)
        bounds2 = [(0.0, 1.0)] * n2
        cons2 = [{'type': 'eq',
                  'fun': lambda x: np.sum(x) - n2 / 2.0,
                  'jac': lambda x: np.ones(n2)}]
        for alpha in [20000, 60000, 150000]:
            if elapsed() > budget_s - 25:
                break
            res = _slsqp_call(h2, n2, alpha, 400, _smooth_obj, bounds2, cons2)
            if res is not None:
                hc = _repair(res.x, n2)
                c5 = _c5(hc, n2)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = hc.copy()
                    n = n2
        if n == n2 and elapsed() < budget_s - 15:
            best_h, best_c5 = _fine_search(best_h.copy(), n, t0, t0 + budget_s - 10, rng)

    # Phase 7: Final micro search
    if elapsed() < budget_s - 5:
        best_h, best_c5 = _micro_search(best_h.copy(), n, t0, t0 + budget_s - 2, rng)

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, n)
    return (best_h, best_c5, n)
