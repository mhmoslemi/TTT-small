    if mask.sum() == 0:
        mask = np.ones(n, dtype=bool)
    per = diff / mask.sum()
    h[mask] += per
    h = np.clip(h, 0.0, 1.0)
    # Fix residual by adjusting individual entries
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        if residual > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for i in order:
            if residual > 0:
                add = min(residual, 1.0 - h[i])
                h[i] += add
                residual -= add
            else:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
            if abs(residual) < 1e-14:
                break
    return h

def _soft_max_obj(h, alpha, n, dx):
    corr = _corr(h)
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _topk_obj(h, k, n, dx):
    """Minimize the k-th largest correlation (for k=1, this is the max)"""
    corr = _corr(h)
    sorted_c = np.sort(corr)[::-1]
    # Smoothed top-k: average of top k with soft assignment
    threshold = sorted_c[k-1]
    weights = np.clip(corr - threshold, 0, 1) / k
    return np.sum(weights * corr) + 0.1 * sorted_c[k-1]

def _grad_cork(h, k, n):
    """Analytical gradient of correlation at lag k w.r.t. h"""
    # In numpy's full correlation, index n-1+k corresponds to lag k
    # c_{n-1+k} = sum_{i=0}^{n-1-k} h[i] * (1-h[i+k])  for k >= 0
    # c_{n-1+k} = sum_{i=|k|}^{n-1} h[i] * (1-h[i-|k|])  for k < 0
    grad = np.zeros(n)
    if k >= 0:
        for i in range(n - k):
            # d/dh[i] of h[i]*(1-h[i+k]) = (1-h[i+k])
            grad[i] += 1 - h[i+k]
            # d/dh[i+k] of h[i]*(1-h[i+k]) = -h[i]
            grad[i+k] -= h[i]
    else:
        k = -k
        for i in range(k, n):
            # c = sum h[i]*(1-h[i-k])
            # d/dh[i] = (1-h[i-k])
            grad[i] += 1 - h[i-k]
            # d/dh[i-k] = -h[i]
            grad[i-k] -= h[i]
    return grad

    # lag is the actual lag: -(n-1) to (n-1)

            grad[i+k] -= h[i]

            grad[i] -= h[i+k]

    for _ in range(80):

        s = np.clip(h - mid, 0, 1).sum()
        if s > target:

        else:

    """Iterative subgradient descent focusing on top-k correlations"""
    best = _c5(h, dx)
    h = h.copy()
    target = n / 2.0
    
    step_init = 0.05
    step_factor = 0.95
    step_min = 1e-8
    step = step_init
    no_improve = 0
    
    while time.time() < deadline:
        corr = _corr(h)
        # Top-k lags
        top_indices = np.argpartition(corr, -top_k)[-top_k:]
        top_indices = top_indices[np.argsort(corr[top_indices])[::-1]]
        
        # Combined gradient (weighted by correlation value)
        weights = corr[top_indices] ** 2
        weights /= np.sum(weights)
        combined_grad = np.zeros(n)
        for idx, w in zip(top_indices, weights):
            lag = idx - (n - 1)
            combined_grad += w * _grad_lag(h, lag, n)
        
        # Project gradient to sum=0 subspace
        combined_grad -= combined_grad.mean()
        
        # Line search
        improved = False
        for trial in range(30):
            s = step * (0.5 ** trial)
            h_new = _project(h - s * combined_grad, target)
            c5_new = _c5(h_new, dx)
            if c5_new < best - 1e-15:
                h = h_new
                best = c5_new
                improved = True
                break
        if not improved:
            no_improve += 1
            step *= step_factor
            if step < step_min or no_improve > 500:
                # Try random kick
                rng_kick = np.random.RandomState(int(time.time() * 1000) % 100000)
                i, j = rng_kick.choice(n, 2, replace=False)
                d = rng_kick.uniform(-0.03, 0.03)
                h[i] += d
                h[j] -= d
                h = _project(h, target)
                no_improve = 0
                step = step_init
        else:
            no_improve = 0
    
    return h, best
