import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return h

def _c5_eval(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _resample(h_old, n_old, n_new):
    xo = np.linspace(0.0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    hn = np.interp(xn, xo, h_old)
    return _repair(hn, n_new)

def _smooth_obj(x, alpha, dx):
    corr = np.correlate(x, 1.0 - x, mode='full')
    sc = alpha * corr / dx
    mx = np.max(sc)
    return (dx / alpha) * (mx + np.log(np.sum(np.exp(sc - mx))))

def _smooth_grad(x, alpha, dx):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    sc = alpha * corr / dx
    mx = np.max(sc)
    es = np.exp(sc - mx)
    w = es / np.sum(es)
    g = np.zeros(n)
    for p in range(2 * n - 1):
        wp = w[p]
        if wp < 1e-16:
            continue
        m = p - n + 1
        if m >= 0:
            g[m:n] += wp * (1.0 - x[:n - m])
            g[:n - m] -= wp * x[m:n]
        else:
            pm = -m
            g[:n - pm] += wp * (1.0 - x[pm:])
            g[pm:] -= wp * x[:n - pm]
    return g

def _eq_fun(x, target):
    return np.sum(x) - target

def _eq_jac(x, target):
    return np.ones(len(x))

def _slsqp(h, n, dx, alpha, maxiter):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac, 'args': (target,)}]
    best_h, best_c5 = h, _c5_eval(h, dx)
    try:
        res = minimize(_smooth_obj, h, method='SLSQP', jac=_smooth_grad,
                       args=(alpha, dx), bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        hc = _repair(res.x, n)
        c5 = _c5_eval(hc, dx)
        if c5 < best_c5:
            best_h, best_c5 = hc, c5
    except Exception:
        pass
    return best_h, best_c5

def _pair_local(h, n, dx, rng, deadline, n_tries=300000):
    best = _c5_eval(h, dx)
    h = h.copy()
    scales = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    per_scale = max(200, n_tries // len(scales))
    for sc in scales:
        improved = False
        for _ in range(per_scale):
            if time.time() > deadline:
                return h, best

            j = rng.randint(n)
            if i == j:
                continue
            d = sc * (2.0 * rng.random() - 1.0)
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            c5 = _c5_eval(h, dx)
            if c5 < best - 1e-15:
                best = c5
                improved = True
            else:
                h[i], h[j] = h[i] - d, h[j] + d
        if not improved and sc < 0.003:
            break
    return h, best

def _block_local(h, n, dx, rng, deadline, n_iters=30000):
