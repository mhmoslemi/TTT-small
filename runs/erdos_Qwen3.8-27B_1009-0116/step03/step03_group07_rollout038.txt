    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(5000):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        d = target - s
        if abs(d) < 1e-14:

        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)

                h[idx] += a
                d -= a
            else:

                h[idx] -= a
                d += a

import numpy as np
from scipy.optimize import minimize
import time

_gs = {}

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, n):
    return float(np.max(compute_corr(h)) * (2.0 / n))

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    tgt = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(5000):
        h *= tgt / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        d = tgt - s
        if abs(d) < 1e-14:
            break
        ord_ = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in ord_:
            if abs(d) < 1e-14:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx]); h[idx] += a; d -= a
            else:
                a = min(-d, h[idx]); h[idx] -= a; d += a
    return h

def obj_fn(h):
    n, al, tk = _gs['n'], _gs['al'], _gs['tk']
    fw, bp = _gs['fw'], _gs['bp']
    c = compute_corr(h)
    m = np.max(c)
    v = m + np.log(np.sum(np.exp(np.clip(al * (c - m), -800, 800)))) / al
    if tk > 1 and fw > 0:
        sc = np.sort(c)[::-1][:tk]
        v += fw * np.sum((sc - sc.mean())**2)
    if bp > 0:
        v += bp * np.sum(h * (1.0 - h)) / n
    v += 1e-3 * (h.sum() - n / 2.0)**2
    return v

def do_slsqp(h, n, al, tk, fw, bp, mi):
    _gs.update(n=n, al=al, tk=tk, fw=fw, bp=bp)
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(obj_fn, h, method='SLSQP', bounds=bnds,
                     options={'maxiter': mi, 'ftol': 1e-16})
        return repair(r.x, n)
    except Exception:
        return h

def do_lbfgs(h, n, al, tk, fw, bp, mi):
    _gs.update(n=n, al=al, tk=tk, fw=fw, bp=bp)
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(obj_fn, h, method='L-BFGS-B', bounds=bnds,
                     options={'maxiter': mi, 'ftol': 1e-18, 'gtol': 1e-14})
        return repair(r.x, n)
    except Exception:
        return h

def coord_sa(h, n, tend, rng, nr=500):
    h = h.copy()
    bc = compute_c5(h, n)
    bh, cc = h.copy(), bc
    tc, tf = 0.02, 5e-5
    dec = (tf / tc)**(1.0 / max(1, nr))
    stps = [0.05, 0.03, 0.02, 0.01, 0.005, 0.002, 0.001, 5e-4, 2e-4]
    for r in range(nr):
        if time.time() > tend:
            break
        for _ in range(4 * n):
            if time.time() > tend:
                break
            i = rng.randint(n); j = rng.randint(n)
            if i == j:
                continue
            hi, hj = h[i], h[j]
            for d in stps:
                hit = False
                for sg in (1, -1):
                    ni, nj = hi + sg * d, hj - sg * d
                    if 0 <= ni <= 1 and 0 <= nj <= 1:
                        h[i], h[j] = ni, nj
                        nc = compute_c5(h, n)
                        if nc < cc or rng.rand() < np.exp(-(nc - cc) / max(tc, 1e-10)):
                            cc = nc
                            if nc < bc:
                                bc, bh = nc, h.copy()
                            hit = True
                        else:
                            h[i], h[j] = hi, hj
                        break
                if hit:
                    break
        tc = max(tf, tc * dec)
        if cc > bc + 0.003:
            h, cc = bh.copy(), bc
    return bh, bc
