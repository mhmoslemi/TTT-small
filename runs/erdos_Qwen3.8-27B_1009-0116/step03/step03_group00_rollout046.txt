                        lambda x, a=alpha: _smooth_obj(x, n, a),
                        lambda x, a=alpha: _smooth_grad(x, n, a),
                        bounds, cons, 2000)
            c5 = _c5(hc, n)
            if c5 < best_c5 - 1e-16:
                best_c5 = c5
                best_h = hc.copy()

    # Phase 5: Systematic pair local search
    if elapsed() < budget_s - 15:
        h = best_h.copy()
        c5_cur = _c5(h, n)
        end_t = t0 + budget_s - 8
        steps = [0.001, 0.003, 0.008, 0.02, 0.05, -0.001, -0.003, -0.008, -0.02, -0.05]
        no_improve = 0
        while time.time() < end_t:
            improved = False
            for i in range(n):
                if time.time() > end_t:
                    break
                hi = h[i]
                for j in range(i + 1, n):
                    if time.time() > end_t:
                        break
                    hj = h[j]
                    for d in steps:
                        ni, nj = hi + d, hj - d
                        if ni < -1e-12 or ni > 1 + 1e-12 or nj < -1e-12 or nj > 1 + 1e-12:
                            continue
                        h[i], h[j] = ni, nj
                        c = _c5(h, n)
                        if c < c5_cur - 1e-15:
                            c5_cur = c
                            hi, hj = ni, nj
                            improved = True
                        else:
                            h[i], h[j] = hi, hj
                    if time.time() > end_t:
                        break
                h[i] = hi
            if not improved:
                no_improve += 1
                if no_improve >= 3:
                    break
            else:
                no_improve = 0
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h.copy()

    # Phase 6: Ultra-fine random coordinate moves
    if elapsed() < budget_s - 3:
        h = best_h.copy()
        c5_cur = _c5(h, n)
        end_t = t0 + budget_s - 1
        tiny = [0.0003, 0.0007, 0.0015, -0.0003, -0.0007, -0.0015]
        while time.time() < end_t:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = tiny[rng.randint(len(tiny))]
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c = _c5(h, n)
                if c < c5_cur - 1e-16:
                    c5_cur = c
                    if c < best_c5:
                        best_c5 = c
                        best_h = h.copy()
                else:
                    h[i], h[j] = h[i] - d, h[j] + d

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, n)
    return (best_h, best_c5, n)
