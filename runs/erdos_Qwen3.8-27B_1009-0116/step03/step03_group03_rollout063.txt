    """Find optimal h[j] given all other h values fixed."""
    # Compute corr values as functions of h[j]
    # corr[m] = intercept[m] + slope[m] * h[j]
    slopes = np.zeros(2*n-1)
    intercepts = np.zeros(2*n-1)
    
    # Current corr
    corr = np.correlate(h, 1-h, mode='full')
    hv = h[j]
    
    for m in range(2*n-1):
        # Determine which terms in corr[m] involve h[j]
        # corr[m] = sum_n h[n+m]* (1-h[n]) for valid n
        # h[j] appears when n+m = j (i.e., n = j-m) or n = j

        # But wait, we need n+m=j means n = j-m, and we need 0<=n<n

            # contribution to corr[m]: h[j] * (1-h[j-m])

        if 0 <= j+m < n:
            # (1-h[n]) = (1-h[j]), so contribution is h[j+m] * (1-h[j])
            # slope of this term w.r.t. h[j]: -h[j+m]

- Compute corr, find active set A = {m : corr[m] ≥ corr_max - ε}
- Average gradients over A to get g
- Solve: min g^T y s.t. 0≤y≤1, sum(y)=n/2

- Update: h ← h + λ*(y - h), λ = 2/(t+2) (standard FW step)

This is the **conditional gradient method** (Frank-Wolfe)! It's very clean and efficient.

    corr_max = corr.max()

    eps = 0.05 * corr_max  # or some fraction
    active = corr > (corr_max - eps)
    # Gradient of each active correlation
    K = 2*n - 1
    g = np.zeros(n)
    for m in active:
        for j in range(n):
            if 0 <= j - m < n:
                g[j] += (1.0 - h[j-m])
            if 0 <= j + m < n:
                g[j] += -h[j+m]

    order = np.argsort(g)
    y = np.zeros(n)
    y[order[:n//2]] = 1.0  # Most negative g get y=1
    
    # Update
    h_new = (1-lam)*h + lam*y
    return h_new, g

    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)

def _cd_sweep(h, n, n_grid, rng=None):

    K = 2 * n - 1
    lag_off = n - 1
    corr = _corr(h)
    cur_max = float(corr.max())
    improved = False
    
    # Random order if rng provided
    if rng is not None:
        order = list(range(n))
        rng.shuffle(order)
    else:
        order = list(range(n))
    
    for j in order:
        hv = h[j]
        slopes = np.zeros(K)
        for m in range(K):
            lag = m - lag_off
            s = 0.0
            if 0 <= j - lag < n:
                s += (1.0 - h[j - lag])
            if 0 <= j + lag < n:
                s += (-h[j + lag])
            slopes[m] = s
        intercepts = corr - hv * slopes
        
        # Coarse grid
        t_coarse = np.linspace(0.0, 1.0, n_grid)
        vals = intercepts[:, None] + slopes[:, None] * t_coarse[None, :]
        f_vals = vals.max(axis=0)
        best_c = f_vals.argmin()
        
        # Fine grid around best
        if n_grid >= 100:
            span = 4.0 / n_grid  # window width
            lo = max(0.0, t_coarse[best_c] - span)
            hi = min(1.0, t_coarse[best_c] + span)
            t_fine = np.linspace(lo, hi, 500)
            vals_f = intercepts[:, None] + slopes[:, None] * t_fine[None, :]
            f_f = vals_f.max(axis=0)
            best_f = f_f.argmin()
            new_j = t_fine[best_f]
            new_c = f_f[best_f]
        else:
            new_j = t_coarse[best_c]
            new_c = f_vals[best_c]
        
        if new_c < cur_max - 1e-14:
            h[j] = new_j
            corr = _corr(h)
            cur_max = float(corr.max())
            improved = True
    
    return h, cur_max, improved

def _cd_optimize(h, n, max_sweeps=200, n_grid=200, tol=1e-12, rng=None):
    """Run coordinate descent until convergence."""
    cur_max = float(_corr(h).max())

            cur_max = new_max
        elif not improved or abs(cur_max - new_max) < tol:

import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
            else:
                add = max(diff, h[idx] - 1.0)
            h[idx] += add
            diff -= add
            if abs(diff) < 1e-12:
                break
    return h

def _slopes_for_j(h, j, n):
    K = 2 * n - 1
    lag_off = n - 1
    slopes = np.zeros(K)
    for m in range(K):
        lag = m - lag_off
        s = 0.0
        if 0 <= j - lag < n:
            s += (1.0 - h[j - lag])
        if 0 <= j + lag < n:
            s += (-h[j + lag])
        slopes[m] = s
    return slopes

def _best_1d(intercepts, slopes, lo, hi, n_grid=1000):
    t_grid = np.linspace(lo, hi, n_grid)
    vals = intercepts[:, None] + slopes[:, None] * t_grid[None, :]
    f = vals.max(axis=0)
    idx = f.argmin()
    return t_grid[idx], f[idx]

def _cd_sweep(h, n, n_grid=500):
    K = 2 * n - 1
    corr = _corr(h)
    cur_max = float(corr.max())
    for j in range(n):
        hv = h[j]
        slopes = _slopes_for_j(h, j, n)
        intercepts = corr - hv * slopes
        new_j, new_c = _best_1d(intercepts, slopes, 0.0, 1.0, n_grid)
        if new_j != hv:
            h[j] = new_j
            corr = _corr(h)
            if float(corr.max()) < cur_max:
                cur_max = float(corr.max())
    return h, cur_max

def _cd_optimize(h, n, max_sweeps=100, n_grid=500, tol=1e-13):

    for _ in range(max_sweeps):
        h, new_max = _cd_sweep(h, n, n_grid)
        if new_max < cur_max - tol:
            cur_max = new_max
        else:
            if new_max > cur_max:
                pass  # Can happen due to grid; keep cur_max
            if abs(cur_max - new_max) < tol:
                break
    return h, cur_max

def _pair_cd(h, n, max_sweeps=50, n_grid=500):
    """Pair coordinate descent maintaining sum constraint."""
    K = 2 * n - 1
    corr = _corr(h)
    cur_max = float(corr.max())
    for _ in range(max_sweeps):
        improved = False
        for j in range(n):
            for k in range(j + 1, n):
                if h[j] <= 0 and h[k] >= 1:
                    continue
                sl_j = _slopes_for_j(h, j, n)
                sl_k = _slopes_for_j(h, k, n)
                diff_sl = sl_j - sl_k
                # Transfer δ from k to j: h[j]+δ, h[k]-δ
                # New corr[m] = corr[m] + δ * diff_sl[m]
                # δ range: [max(-h[k], 0-h[j])... wait
                # h[j]+δ ∈ [0,1] → δ ∈ [-h[j], 1-h[j]]
                # h[k]-δ ∈ [0,1] → δ ∈ [h[k]-1, h[k]]
                # Combined: δ ∈ [max(0-h[j], h[k]-1), min(1-h[j], h[k])]

                delta_hi = min(1.0 - h[j], h[k])
                if delta_lo >= delta_hi:

                if best_c < cur_max - 1e-14:

                    h[k] -= best_d
                    corr = _corr(h)
                    cur_max = float(corr.max())
                    improved = True
        if not improved:
            break

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
            else:
                add = max(diff, h[idx] - 1.0)
            h[idx] += add
            diff -= add
            if abs(diff) < 1e-12:
                break
    return h

def _slopes_for_j(h, j, n):
    K = 2 * n - 1
    lag_off = n - 1
    slopes = np.zeros(K)
    for m in range(K):
        lag = m - lag_off
        s = 0.0
        if 0 <= j - lag < n:
            s += (1.0 - h[j - lag])
        if 0 <= j + lag < n:
            s += (-h[j + lag])
        slopes[m] = s
    return slopes

def _best_1d(intercepts, slopes, lo, hi, n_grid):
    if lo >= hi:
        return lo, float(intercepts.max())
    t_grid = np.linspace(lo, hi, n_grid)
    vals = intercepts[:, None] + slopes[:, None] * t_grid[None, :]
    f = vals.max(axis=0)
    idx = int(f.argmin())
    return float(t_grid[idx]), float(f[idx])

def _cd_sweep_constrained(h, n, n_grid):
    K = 2 * n - 1
    corr = _corr(h)
    cur_max = float(corr.max())
    sl_n1 = _slopes_for_j(h, n - 1, n)
    for j in range(n - 1):
        hv = h[j]
        sl_j = _slopes_for_j(h, j, n)
        eff_sl = sl_j - sl_n1
        t_lo = max(0.0, hv + h[n - 1] - 1.0)
        t_hi = min(1.0, hv + h[n - 1])
        if t_lo >= t_hi:
            continue
        new_t, new_c = _best_1d(corr, eff_sl, t_lo, t_hi, n_grid)
        delta = new_t - hv
        if abs(delta) > 1e-14 and new_c < cur_max - 1e-14:
            h[j] = new_t
            h[n - 1] = h[n - 1] - delta
            h[n - 1] = np.clip(h[n - 1], 0.0, 1.0)
            corr = _corr(h)
            cur_max = float(corr.max())
            sl_n1 = _slopes_for_j(h, n - 1, n)
    return h, cur_max

def _cd_optimize(h, n, max_sweeps, n_grid, tol=1e-13):
    cur_max = float(_corr(h).max())
    for _ in range(max_sweeps):
        h, new_max = _cd_sweep_constrained(h, n, n_grid)
        if new_max < cur_max - tol:
            cur_max = new_max
        elif abs(cur_max - new_max) < tol:
            break
        else:
            break
    return h, cur_max

def _gen_pert(h, n, strategy, rng):
    if strategy == 0:
        return np.clip(h + 0.05 * rng.randn(n), 0, 1)
    elif strategy == 1:
        hp = h.copy()
        idx = rng.choice(n, size=min(5, n), replace=False)
        hp[idx] = rng.rand(len(idx))
        return hp
    elif strategy == 2:
        return np.where(h > 0.5, 1.0, 0.0)
    elif strategy == 3:
        hp = h.copy()
        idx = rng.choice(n, size=n // 4 + 1, replace=False)
        hp[idx] = 1.0 - hp[idx]
        return hp
    elif strategy == 4:
        hp = h.copy()
        idx = rng.choice(n, size=3, replace=False)
        hp[idx] = rng.rand(3)
        return hp
    elif strategy == 5:
        hp = h.copy()
        hp += 0.12 * rng.randn(n)
        return np.clip(hp, 0, 1)
    else:
        hp = h.copy()
        idx = rng.choice(n, size=n // 3 + 1, replace=False)
        hp[idx] = rng.rand(len(idx))
        return np.clip(hp, 0, 1)

def _smooth_obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sum_con(x, t):
    return np.sum(x) - t

    def tleft():

    # Phase 1: Coordinate descent with increasing precision (40% budget)
    phase1_end = t0 + budget_s * 0.40
