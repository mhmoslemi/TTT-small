                        if c5_cur < c5_best:
                            c5_best = c5_cur
                            h_best = h.copy()
                        improved = True
                        break
                    h[di] += d
                    h[dj] -= d
                if improved:
                    break
            if improved:
                break
        if not improved:
            if it > 0 and it % 5000 == 0:
                h = _project(h + 0.008 * rng.randn(n), target)
                c5_cur = _c5(h, dx)
            else:
                i = rng.randint(n)
                j = (i + 1 + rng.randint(n - 1)) % n
                d = 0.003 * rng.choice([-1, 1])
                ni, nj = h[i] + d, h[j] - d
                if 0 <= ni <= 1 and 0 <= nj <= 1:
                    h[i], h[j] = ni, nj
                    if _c5(h, dx) < c5_cur:
                        c5_cur = _c5(h, dx)
                        if c5_cur < c5_best:
                            c5_best = c5_cur
                            h_best = h.copy()

    # Phase 4: Ultra-fine random search
    if deadline - time.time() > 15:
        h = h_best.copy()
        c5_cur = _c5(h, dx)
        for _ in range(400000):
            if deadline - time.time() < 15:
                break
            i = rng.randint(n)
            j = (i + 1 + rng.randint(n - 1)) % n
            d = 0.0005 * rng.choice([-1, 1])
            if rng.rand() < 0.3:
                d = 0.0015 * rng.choice([-1, 1])
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5_new = _c5(h, dx)
                if c5_new < c5_cur - 1e-15:
                    c5_cur = c5_new
                    if c5_cur < c5_best:
                        c5_best = c5_cur
                        h_best = h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d

    # Phase 5: Final L-BFGS polish
    if deadline - time.time() > 30:
        hc = _lbfgs_step(h_best, target, 500000, 20, n, bounds)
        c5 = _c5(hc, dx)
        if c5 < c5_best:
            c5_best, h_best = c5, hc

    h_best = _project(h_best, target)
    c5_best = _c5(h_best, dx)
    return (h_best, c5_best, n)
