def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, n):
    dx = 2.0 / n
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return h

def _wsum_obj(x, n, K, alpha):
    # Weighted sum of top-K correlation values
    dx = 2.0 / n
    c = _corr(x) * dx
    # Use log-sum-exp for top K
    sorted_c = np.sort(c)[::-1]
    top = sorted_c[:K]
    m = top[0]
    s = alpha * (top - m)
    s = np.clip(s, -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _wsum_grad(x, n, K, alpha):
    dx = 2.0 / n
    c = _corr(x) * dx
    idx = np.argsort(c)[::-1][:K]
    m = c[idx[0]]
    s = alpha * (c[idx] - m)
    s = np.clip(s, -400, 400)
    e = np.exp(s)
    w = e / e.sum()
    # Gradient of the smooth max over top K
    g = np.zeros(n)
    nfull = len(c)  # 2n-1
    nmid = n - 1
    for ki in range(K):
        wk = w[ki] * dx
        k = idx[ki]
        lag = k - nmid
        i_min = max(0, -lag)
        i_max = min(n, n - lag)
        for i in range(i_min, i_max):
            jp = i + lag
            g[jp] += wk * (1.0 - x[i])
            g[i] -= wk * x[jp]
    return g

import numpy as np
from scipy.optimize import minimize
from functools import partial
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, n):
    dx = 2.0 / n
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = float(n) / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return h

def _obj_topk(x, n, K, alpha):
    dx = 2.0 / n
    c = _corr(x) * dx
    order = np.argsort(c)[::-1][:K]
    top = c[order]
    m = top[0]
    s = alpha * (top - m)
    s = np.clip(s, -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _obj_topk_grad(x, n, K, alpha):
    dx = 2.0 / n
    c = _corr(x) * dx
    order = np.argsort(c)[::-1][:K]
    m = c[order[0]]
    s = alpha * (c[order] - m)
    s = np.clip(s, -400, 400)
    e = np.exp(s)
    w = e / e.sum()
    nmid = n - 1
    g = np.zeros(n)
    for ki in range(K):
        wk = w[ki] * dx
        k = order[ki]
        lag = k - nmid
        i_min = max(0, -lag)
        i_max = min(n, n - lag)
        for i in range(i_min, i_max):
            jp = i + lag
            g[jp] += wk * (1.0 - x[i])
            g[i] -= wk * x[jp]
    return g

def _cons_fun(x, n):
    return np.sum(x) - float(n) / 2.0

def _cons_jac(x, n):
    return np.ones(n)

def _run_slsqp(h0, n, K, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    cf = partial(_cons_fun, n=n)
    cj = partial(_cons_jac, n=n)
    cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]
    res = minimize(
        _obj_topk, h0, method='SLSQP', bounds=bounds, constraints=cons,
        jac=_obj_topk_grad, args=(n, K, alpha),
        options={'maxiter': maxiter, 'ftol': 1e-18})
    return _repair(res.x, n)

def _grad_desc_topk(h, n, budget_end, rng):

    best_local = c5_cur
    h_best = h.copy()
    dx = 2.0 / n
    nmid = n - 1
    T = 0.5
    for it in range(50000):
        if time.time() > budget_end:
            break
        c = _corr(h) * dx
        order = np.argsort(c)[::-1]
        K_tail = max(10, len(c) // 6)
        top_idx = order[:K_tail]
        m = c[top_idx[0]]
        s = 8000.0 * (c[top_idx] - m)
        s = np.clip(s, -400, 400)
        e = np.exp(s)
        w = e / e.sum()
        g = np.zeros(n)
        for ki in range(K_tail):
            wk = w[ki] * dx
            k = top_idx[ki]
            lag = k - nmid
            i_min = max(0, -lag)
            i_max = min(n, n - lag)
            for i in range(i_min, i_max):
                jp = i + lag
                g[jp] += wk * (1.0 - h[i])
                g[i] -= wk * h[jp]
        step = T * 0.003 / (np.max(np.abs(g)) + 1e-12)
        h_new = _repair(h - step * g, n)
        c5_new = _c5(h_new, n)
        if c5_new < c5_cur or (T > 0.01 and rng.rand() < np.exp(-(c5_new - c5_cur) / (T * 0.001))):
            h = h_new
            c5_cur = c5_new
            if c5_cur < best_local:
                best_local = c5_cur
                h_best = h.copy()
        T *= 0.9995
        if T < 0.01:
            T = 0.01
    return h_best, best_local

def _fine_search(h, n, budget_end, rng):
    c5_cur = _c5(h, n)
    deltas = np.array([0.0004, 0.001, 0.003, 0.008, -0.0004, -0.001, -0.003, -0.008])
    for _ in range(800):
        if time.time() > budget_end:
            break
        improved = False
        npairs = min(3000, n * n)
        for _ in range(npairs):
            if time.time() > budget_end:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            hi, hj = h[i], h[j]
            for d in deltas:
                ni, nj = hi + d, hj - d
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
                h[i], h[j] = ni, nj
                c = _c5(h, n)
                if c < c5_cur - 1e-16:
                    c5_cur = c
                    hi, hj = ni, nj
                    improved = True
                else:
                    h[i], h[j] = hi, hj
            h[i] = hi
        if not improved:
            break
    return h, c5_cur

def _fourier_init(n, rng):
    h = np.full(n, 0.5)
    for mode in range(1, 6):
        amp = rng.uniform(0.05, 0.25) / mode
        phase = rng.uniform(0, 2 * np.pi)
        idx = np.arange(n)
        h += amp * np.cos(mode * np.pi * idx * 2.0 / n + phase)
    h = _repair(h, n)
    return h
