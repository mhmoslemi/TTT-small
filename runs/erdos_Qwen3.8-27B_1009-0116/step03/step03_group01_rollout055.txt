    target = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / len(h)))

def _sf(x, a):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(a * (c - m), -500, 500)
    s -= s.max()
    return m + np.log(np.sum(np.exp(s))) / a

def _sj(x, a):
    n = len(x)
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(a * (c - m), -500, 500)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(n):
        idx = k + n - 1
        if w[idx] < 1e-14:
            continue
        wl = w[idx]
        if k == 0:
            g += wl * (1.0 - 2.0 * x)
        else:
            g[:n-k] += wl * (1.0 - x[k:])
            g[k:] -= wl * x[:n-k]
    for m2 in range(n - 1):
        if w[m2] < 1e-14:
            continue
        L = n - 1 - m2
        g[L:] += w[m2] * (1.0 - x[:n-L])
        g[:n-L] -= w[m2] * x[L:]
    return g

def _con_fun(x):
    return np.sum(x) - _N * 0.5

def _con_jac(x):
    return np.ones(_N)

def _slsqp(h, n, alpha, maxiter):
    global _N
    _N = n
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]

                       options={'maxiter': maxiter, 'ftol': 1e-20})
        return _repair(res.x, n)
    except:
        return _repair(h, n)

def _sa(h, n, rng, deadline):

            j = (i + 1) % n
        d = rng.uniform(-step, step)
        ni = cur[i] + d
        nj = cur[j] - d
        if ni < 0.0 or ni > 1.0 or nj < 0.0 or nj > 1.0:

            cur_c5 = nc
        else:
            cur[i], cur[j] = oi, oj
    return best, best_c5

def _fine_scan(h, n, deadlines):

    cur = h.copy()
    cur_c5 = _eval(cur)
    best = cur.copy()
    best_c5 = cur_c5
    for dl in deadlines:
        improved = True
        while improved and time.time() < dl:
            improved = False
            for i in range(n):
                if time.time() > dl:
                    break
                hi = cur[i]
                for j in range(i + 1, n):
                    if time.time() > dl:
                        break
                    hj = cur[j]
                    # Try displacements
                    for d in (0.01, 0.005, 0.001, 0.0005, -0.01, -0.005, -0.001, -0.0005):

                                    best = cur.copy()
                            else:

    return best, best_c5
