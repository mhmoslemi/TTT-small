                    if elapsed() > budget_s - 25:

                    for _ in range(80):

                    best_c5 = ct
                    best_h = hf.copy()
                else:

    best_c5 = _c5(best_h, 2.0/best_n)

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(50):
        s = h.sum()
        if abs(s - target) < 1e-11:
            break
        h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-11:
        idx = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for i in idx:
            a = min(abs(diff), abs(1.0 - np.sign(diff) * h[i]) - 0.0 + abs(h[i] if diff < 0 else 1.0 - h[i]))
            if diff > 0:
                a = min(diff, 1.0 - h[i])
                h[i] += a
                diff -= a
            else:
                a = min(-diff, h[i])
                h[i] -= a
                diff += a
            if abs(diff) < 1e-11:
                break
    return h

def _grad_lag(h, m, n):
    k = m - (n - 1)
    g = np.zeros(n)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    elif k > 0:
        g[k:] = (1.0 - h[:n-k])
        g[:n-k] = -h[k:]
    else:
        L = -k
        g[:n-L] = -h[L:]
        g[L:] = (1.0 - h[:n-L])
    return g

def _softobj(x, alpha, lam, target):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.sum(np.exp(s)) / alpha + lam * (x.sum() - target) ** 2
