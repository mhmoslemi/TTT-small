def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()

    if s < 1e-14:

            idx = ms - k + n - 1
            grad[c1_lo:c1_hi+1] += pk * (1.0 - h[idx])

def adam_optimize(h, n, t_end, n_iter, alpha_start, alpha_end, lr_init, rng):
    """Adam optimizer with annealed alpha"""
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    m_opt = np.zeros(n)
    v_opt = np.zeros(n)
    c5 = compute_c5(h, n)
    
    for it in range(n_iter):
        if time.time() > t_end:
            break
        frac = it / max(n_iter, 1)
        alpha = alpha_start + (alpha_end - alpha_start) * frac**2
        lr = lr_init * (1.0 - 0.5 * frac)
        
        g = grad_softmax(h, n, alpha)
        
        m_opt = beta1 * m_opt + (1 - beta1) * g
        v_opt = beta2 * v_opt + (1 - beta2) * g * g
        m_hat = m_opt / (1 - beta1 ** (it + 1))
        v_hat = v_opt / (1 - beta2 ** (it + 1))
        
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = repair(h_new, n)
        
        c5_new = compute_c5(h_new, n)
        if c5_new < c5:
            c5 = c5_new
        h = h_new
    
    return h, c5
