    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _softobj(x, alpha, lam):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    s -= np.max(s)
    penalty = lam * (np.sum(x) - len(x) / 2.0) ** 2
    return m + np.log(np.sum(np.exp(s))) / alpha + penalty

def _softgrad(x, alpha, lam):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    s -= np.max(s)
    ew = np.exp(s)
    pw = ew / ew.sum()
    n = len(x)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        pk = pw[k]
        if pk < 1e-15:
            continue
        if L == 0:
            g += pk * (1.0 - 2.0 * x)
        elif L > 0:
            g[L:] += pk * (1.0 - x[:n - L])
            g[:n - L] -= pk * x[L:]
        else:
            M = -L
            g[:n - M] += pk * (1.0 - x[M:])
            g[M:] -= pk * x[:n - M]
    g *= 1.0 / alpha
    g += 2.0 * lam * (np.sum(x) - n / 2.0)
    return g

def _maxsubgrad(h):
    c = _corr(h)
    n = len(h)
    cm = np.max(c)
    spread = cm - np.min(c)
    if spread < 1e-12:
        spread = 1.0
    thresh = cm - 0.05 * spread
    active = np.where(c >= thresh)[0]
    g = np.zeros(n)
    for k in active:
        L = k - (n - 1)
        if L == 0:
            g += (1.0 - 2.0 * h)
        elif L > 0:
            g[L:] += (1.0 - h[:n - L])
            g[:n - L] -= h[L:]
        else:
            M = -L
            g[:n - M] += (1.0 - h[M:])
            g[M:] -= h[:n - M]
    g /= len(active)
    return g

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    def optimize_h(h_start, n_pts, dx_pts, bounds_pts, time_left_fn):
        bh = h_start.copy()
        bc5 = _c5(bh, dx_pts)
        target_pts = n_pts / 2.0

        for a in [500, 2000, 10000, 50000, 200000, 800000, 3000000]:
            if time_left_fn() < 30:
                break
            try:
                res = minimize(
                    lambda x, al=a: _softobj(x, al, 60.0), bh.copy(),
                    jac=lambda x, al=a: _softgrad(x, al, 60.0),
                    method='L-BFGS-B', bounds=bounds_pts,
                    options={'maxiter': 800, 'ftol': 1e-20, 'gtol': 1e-16})
                hc = _repair(res.x, n_pts)
                c5v = _c5(hc, dx_pts)
                if c5v < bc5:
                    bc5 = c5v
                    bh = hc
            except Exception:
                pass

        cons = [{'type': 'eq', 'fun': lambda x, tp=target_pts: np.sum(x) - tp,
                 'jac': lambda x, np2=n_pts: np.ones(np2)}]
        for a in [100000, 500000, 2000000]:
            if time_left_fn() < 25:
                break
            try:
                res = minimize(
                    lambda x, al=a: _softobj(x, al, 0.0), bh,
                    method='SLSQP', bounds=bounds_pts, constraints=cons,
                    options={'maxiter': 1200, 'ftol': 1e-19})
                hc = _repair(res.x, n_pts)
                c5v = _c5(hc, dx_pts)
                if c5v < bc5:
                    bc5 = c5v
                    bh = hc
            except Exception:
                pass

        h = bh.copy()
        c5c = _c5(h, dx_pts)
        step = 0.015
        for _it in range(800):
            if time_left_fn() < 20:
                break
            g = _maxsubgrad(h)
            g_c = g - g.mean()
            ng = np.linalg.norm(g_c)
            if ng < 1e-12:
                break
            g_c /= ng
            found = False
            for s_try in [step, step * 0.5, step * 0.25, step * 0.1]:
                h_new = _repair(h - s_try * g_c, n_pts)
                c_new = _c5(h_new, dx_pts)
                if c_new < c5c - 1e-16:
                    h = h_new
                    c5c = c_new
                    found = True
                    if c5c < bc5:
                        bc5 = c5c
                        bh = h.copy()
                    break
            if not found:
                step *= 0.5
                if step < 1e-7:
                    break
            else:
                step = min(step * 1.05, 0.04)

        h = bh.copy()
        c5c = _c5(h, dx_pts)
        dlist = np.array([0.0008, 0.002, 0.005, 0.012, 0.025,
                         -0.0008, -0.002, -0.005, -0.012, -0.025])
        for _sweep in range(150):
            if time_left_fn() < 15:
                break
            g = _softgrad(h, 500000, 0.0)
            order = np.argsort(np.abs(g))[::-1][:35]
            improved = False
            for ii in range(len(order)):
                i = order[ii]
                for jj in range(ii + 1, min(ii + 20, len(order))):
                    j = order[jj]
                    best_d = 0.0
                    best_c = c5c
                    for d in dlist:
                        ni, nj = h[i] + d, h[j] - d
                        if 0 <= ni <= 1 and 0 <= nj <= 1:
                            h[i], h[j] = ni, nj
                            c = _c5(h, dx_pts)
                            if c < best_c:
                                best_c = c
                                best_d = d
                            h[i], h[j] = ni - d, nj + d
                    if best_d != 0.0:
                        h[i] += best_d
                        h[j] -= best_d
                        h = _repair(h, n_pts)
                        c5c = _c5(h, dx_pts)
                        improved = True
                        if c5c < bc5:
                            bc5 = c5c
                            bh = h.copy()
                        break
            if not improved:
                break

        if time_left_fn() > 10:
            h = bh.copy()
            c5c = _c5(h, dx_pts)
            n_rand = min(30000, int((time_left_fn() - 10) * 3000))
            for _ in range(n_rand):
                if time_left_fn() < 10:
                    break
                i, j = rng.randint(n_pts), rng.randint(n_pts)
                if i == j:
                    continue
                d = rng.uniform(-0.005, 0.005)
                ni, nj = h[i] + d, h[j] - d
                if 0 <= ni <= 1 and 0 <= nj <= 1:
                    h[i], h[j] = ni, nj
                    c = _c5(h, dx_pts)
                    if c < c5c:
                        c5c = c
                        bc5 = c
                        bh = h.copy()
                    else:
                        h[i], h[j] = ni - d, nj + d
        return bh, bc5

    # Main: start from initial_h_values
    n = len(initial_h_values)
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    h0 = _repair(np.array(initial_h_values, dtype=float), n)
    best_h, best_c5 = optimize_h(h0, n, dx, bounds,
                                  lambda: budget_s - elapsed() - 150)

    # Try n=100
    if elapsed() < budget_s - 120:
        n2 = 100
        dx2 = 2.0 / n2
        bounds2 = [(0.0, 1.0)] * n2
        h2 = np.interp(np.linspace(0, 2, n2, endpoint=False),
                       np.linspace(0, 2, n, endpoint=False), best_h)
        h2 = _repair(h2, n2)
        h2r, c5_2 = optimize_h(h2, n2, dx2, bounds2,
                                lambda: budget_s - elapsed() - 60)
        if c5_2 < best_c5:
            best_h, best_c5, n, dx = h2r, c5_2, n2, dx2

    # Try n=140
    if elapsed() < budget_s - 80:
        n3 = 140
        dx3 = 2.0 / n3
        bounds3 = [(0.0, 1.0)] * n3
        h3 = np.interp(np.linspace(0, 2, n3, endpoint=False),
                       np.linspace(0, 2, n, endpoint=False), best_h)
        h3 = _repair(h3, n3)
        h3r, c5_3 = optimize_h(h3, n3, dx3, bounds3,
                                lambda: budget_s - elapsed() - 20)
        if c5_3 < best_c5:
            best_h, best_c5, n, dx = h3r, c5_3, n3, dx3

    # Final polish on best
    if elapsed() < budget_s - 3:
        h = best_h.copy()
        c5c = _c5(h, dx)
        tiny = np.array([0.0002, 0.0005, 0.001, -0.0002, -0.0005, -0.001])
        for _ in range(8000):
            if elapsed() > budget_s - 2:
                break
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = tiny[rng.randint(len(tiny))]
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c = _c5(h, dx)
                if c < c5c:
                    c5c = c
                    best_c5 = c
                    best_h = h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
