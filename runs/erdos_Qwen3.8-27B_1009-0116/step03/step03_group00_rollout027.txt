minimize max_k c[k] over h, subject to sum(h) = n/2, 0≤h≤1

Lagrangian: L(h, λ) = max_k c[k] + λ*(sum(h) - n/2)

For fixed λ, the subproblem is: minimize max_k c[k] + λ*sum(h) over 0≤h≤1
This separates by coordinate! For each coordinate i:
minimize max_k c[k] + λ*h[i] over h[i] ∈ [0,1]

And max_k c[k] = max_k (A_k + h[i]*B_k[lag]) where A_k and B_k depend on all other h[j].

So for fixed λ, the subproblem for coordinate i is:
min_{t∈[0,1]} max_k (A_k + t*B_k) + λ*t

This is still a 1D piecewise linear minimax! And it's separable per coordinate.

1. Initialize λ = 0
2. Repeat:
   a. For each coordinate i (in some order), solve the 1D subproblem to find optimal h[i]
   b. After one full pass, check sum(h) - n/2
   c. Update λ: if sum > n/2, increase λ; if sum < n/2, decrease λ
   d. Until convergence

                    h2 = hc.copy()
            except: pass
    
    # Phase 3: Increase to n=280
    if elapsed() < budget_s - 100:
        n3 = 280
        h3 = _interp(best_h, best_n, n3)
        h3 = _repair(h3, n3)
        for alpha in [20000, 60000]:
            if elapsed() > budget_s - 60: break
            hi = h3
            try:
                res = _slsqp(hi, n3, alpha, 300, t0+budget_s-60)
                hc = _repair(res.x, n3)
                c5 = _c5(hc, n3)
                if c5 < best_c5:
                    best_c5, best_h, best_n = c5, hc, n3
                    h3 = hc.copy()
            except: pass
    
    # Phase 4: Final local search
    if elapsed() < budget_s - 20:
        best_h, best_c5 = _local_search(best_h, best_n, t0+budget_s-20, rng)
    
    best_h = _repair(best_h, best_n)
    best_c5 = _c5(best_h, best_n)
    return (best_h, best_c5, best_n)

Actually, I realize there might be a much simpler vectorization using FFT. The correlation c[lag] = sum_i h[i]*(1-h[i+lag]) can be computed as:

Let's define the "full" correlation. For lag ∈ {-(n-1), ..., n-1}:
c[lag] = sum_{i: 0≤i<n, 0≤i+lag<n} h[i] * (1-h[i+lag])

This is like a cross-correlation of h with (1-h), but truncated to valid indices.

For the gradient:
g[j] = sum_lag w[lag] * [(1-h[j+lag])*I(0≤j+lag<n) - h[j-lag]*I(0≤j-lag<n)]

The first sum: sum_lag w[lag]*(1-h[j+lag])*I(0≤j+lag<n)
= sum_{k: 0≤k<n} w[k-j]*(1-h[k]) where k=j+lag, lag=k-j, and k-j ∈ [-(n-1), n-1] (always true for k,j∈[0,n))
= sum_k w[k-j]*(1-h[k]) for k in [max(0, j-(n-1)), min(n-1, j+(n-1))]

This is a convolution of (1-h) with w_reversed (or w depending on convention).

Similarly for the second term.

Using FFT:
First term = ifft(fft(1-h, 2n-1) * fft(w, 2n-1))[appropriate slice]
Second term = ifft(fft(h, 2n-1) * fft(w_reversed, 2n-1))[appropriate slice]

But this might be overkill for n=70-280. The loop version with numpy slices should be fast enough.

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, n):
    return float(np.max(_corr(h))) * (2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _obj(x, n, alpha):
    c = _corr(x) * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -350, 350)
    return m + np.log(np.exp(s).sum()) / alpha

def _grad(x, n, alpha):
    dxv = 2.0 / n
    c = _corr(x) * dxv
    m = c.max()
    s = np.clip(alpha * (c - m), -350, 350)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    L = 2 * n - 1
    for k in range(L):
        wk = w[k]
        if wk < 1e-12:
            continue
        lag = k - (n - 1)
        j1lo = max(0, -lag)
        j1hi = min(n, n - lag)
        if j1hi > j1lo:
            g[j1lo:j1hi] += wk * (1.0 - x[j1lo + lag:j1hi + lag])
        j2lo = max(0, lag)
        j2hi = min(n, n + lag)
        if j2hi > j2lo:
            g[j2lo:j2hi] -= wk * x[j2lo - lag:j2hi - lag]
    return g * dxv

def _slsqp(h0, n, alpha, maxiter, t_end):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    return minimize(lambda x: _obj(x, n, alpha), h0,
                    jac=lambda x: _grad(x, n, alpha),
                    method='SLSQP', bounds=bounds, constraints=cons,
                    options={'maxiter': maxiter, 'ftol': 1e-18})

def _interp(h, n_old, n_new):
    x_old = np.arange(n_old, dtype=float)
    x_new = np.linspace(0, n_old, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)

def _bangbang_search(h, n, t_end, rng):
    """Try setting pairs of coordinates to 0 or 1 (bang-bang)."""
    h = h.copy()
    c5_cur = _c5(h, n)
    for _ in range(200000):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        old_i, old_j = h[i], h[j]
        # Try 4 combos: (0,0),(0,1),(1,0),(1,1) adjusted for sum
        for v_i in [0.0, 1.0]:
            v_j = old_i + old_j - v_i
            if v_j < -1e-6 or v_j > 1 + 1e-6:
                continue
            v_j = max(0.0, min(1.0, v_j))
            h[i], h[j] = v_i, v_j
            c_new = _corr(h)
            c5_new = c_new.max() * (2.0 / n)
            if c5_new < c5_cur - 1e-14:
                c5_cur = c5_new
            else:
                h[i], h[j] = old_i, old_j
    return h, c5_cur

def _local_refine(h, n, t_end, rng):
    h = h.copy()
    c5_cur = _c5(h, n)
    steps = np.array([0.0005, 0.001, 0.003, 0.008, 0.02, 0.05])
    while time.time() < t_end:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = steps[rng.randint(len(steps))] * (1.0 if rng.rand() < 0.5 else -1.0)
        hi, hj = h[i], h[j]
        ni, nj = hi + d, hj - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5_new = _c5(h, n)
        if c5_new < c5_cur - 1e-15:
            c5_cur = c5_new
        else:
            h[i], h[j] = hi, hj
    return h, c5_cur

    rng = np.random.RandomState(seed)
    n = len(initial_h_values)

    best_c5 = _c5(best_h, n)
    best_n = n

    def elapsed():
        return time.time() - t0

    for alpha in [5000, 20000]:
        for r in range(4):
            if elapsed() > budget_s - 400:
                break
            hi = best_h.copy() if r == 0 else _repair(best_h + 0.02 * (1 + 0.4 * r) * rng.randn(n), n)
            try:
                res = _slsqp(hi, n, alpha, 400, t0 + budget_s - 400)
                hc = _repair(res.x, n)
                c5 = _c5(hc, n)
                if c5 < best_c5:
                    best_c5, best_h = c5, hc
            except Exception:
                pass

    # Phase 2: Main optimization at n=140
    n2 = 140
    h2 = _repair(_interp(best_h, best_n, n2), n2)
    c5_2 = _c5(h2, n2)
    if c5_2 < best_c5 + 0.01:
        best_c5_tmp = c5_2
    for alpha in [8000, 25000, 70000, 150000]:
        n_tries = 5 if alpha <= 25000 else 3
        for r in range(n_tries):
            if elapsed() > budget_s - 200:
                break
            hi = h2.copy() if r == 0 else _repair(h2 + 0.015 * rng.randn(n2), n2)
            try:
                res = _slsqp(hi, n2, alpha, 500, t0 + budget_s - 200)
                hc = _repair(res.x, n2)
                c5 = _c5(hc, n2)
                if c5 < best_c5:
                    best_c5, best_h, best_n = c5, hc, n2
                    h2 = hc.copy()
            except Exception:
                pass
        h2 = _repair(best_h.copy() if best_n == n2 else _interp(best_h, best_n, n2), n2)

    # Phase 3: Try n=280
    if elapsed() < budget_s - 120:
        n3 = 280
        h3 = _repair(_interp(best_h, best_n, n3), n3)
        for alpha in [30000, 100000]:
            if elapsed() > budget_s - 60:
                break
            for r in range(2):
                hi = h3.copy() if r == 0 else _repair(h3 + 0.01 * rng.randn(n3), n3)
                try:
                    res = _slsqp(hi, n3, alpha, 350, t0 + budget_s - 60)
                    hc = _repair(res.x, n3)
                    c5 = _c5(hc, n3)
                    if c5 < best_c5:
                        best_c5, best_h, best_n = c5, hc, n3
                        h3 = hc.copy()
                except Exception:
                    pass

    # Phase 4: Bang-bang search (exploit discreteness)
    if elapsed() < budget_s - 80:
        best_h, best_c5 = _bangbang_search(best_h, best_n, t0 + budget_s - 80, rng)

    # Phase 5: Fine local search
    if elapsed() < budget_s - 20:
        best_h, best_c5 = _local_refine(best_h, best_n, t0 + budget_s - 15, rng)

    # Phase 6: One more SLSQP polish on best
    if elapsed() < budget_s - 10:
        try:
            res = _slsqp(best_h, best_n, 200000, 200, t0 + budget_s - 5)
            hc = _repair(res.x, best_n)
            c5 = _c5(hc, best_n)
            if c5 < best_c5:
                best_c5, best_h = c5, hc
        except Exception:
            pass

    best_h = _repair(best_h, best_n)
    best_c5 = _c5(best_h, best_n)
    return (best_h, best_c5, best_n)
