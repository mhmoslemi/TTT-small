import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, n):
    return float(np.max(compute_corr(h)) * 2.0 / n)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    for _ in range(1000):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if abs(diff) < 1e-14:
                break
            room = (1.0 - h[i]) if diff > 0 else h[i]
            step = min(abs(diff), room) * (1.0 if diff > 0 else -1.0)
            h[i] += step
            diff -= step
    return h

def smooth_max_obj(h, alpha, topk, flat_w):
    corr = compute_corr(h)
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -100, 100)
    val = m + np.log(np.sum(np.exp(s))) / alpha
    if topk > 1:
        sc = np.sort(corr)[::-1][:topk]
        val += flat_w * np.sum((sc - sc.mean())**2)
    return val

def slsqp_optimize(h, n, alpha, topk, flat_w, maxiter, t_end):
    def obj(x):
        return smooth_max_obj(x, alpha, topk, flat_w)
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, jac=None,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return repair(res.x, n)
    except:
        return h

def full_sweep_cd(h, n, n_sweeps, t_end, rng):
    """Full coordinate descent: sweep all pairs, find best delta via sampling."""
    h = h.copy()
    best = compute_c5(h, n)
    for sweep in range(n_sweeps):
        if time.time() > t_end:
            break
        improved = False
        delta_steps = np.array([0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002])
        for i in range(n):
            for j in range(n):
                if i == j:
                    continue
                hi, hj = h[i], h[j]
                for d in delta_steps:
                    for sign in (1, -1):
                        nd = sign * d
                        ni, nj = hi + nd, hj - nd
                        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                            continue
                        h[i], h[j] = ni, nj
                        c5 = compute_c5(h, n)
                        if c5 < best - 1e-14:
                            best = c5
                            improved = True
                            # Try to go further in same direction
                            break
                        else:
                            h[i], h[j] = hi, hj
                    else:
                        continue
                    break
                if not improved:
                    h[i], h[j] = hi, hj
    return h, best

def subgrad_step(h, n, lr, t_end, n_steps):
    """Projected subgradient descent on the max."""

    target = n / 2.0
    best = compute_c5(h, n)

        if time.time() > t_end:
            break
        corr = compute_corr(h)
        m = corr.max()
        # Find active lags
        active = np.where(corr >= m - 1e-6)[0]
        # Compute subgradient (average of grads of active corr components)
        grad = np.zeros(n)
        for k in active:
            # d(corr_k)/d(h) - computed via the correlation structure
            # For np.correlate(h, 1-h, 'full'), index k corresponds to lag k-(n-1)
            lag = k - (n - 1)
            g = np.zeros(n)
            if lag >= 0:
                # corr = sum_{i=0}^{n-1-lag} h[i]*(1-h[i+lag])
                for i in range(n - lag):
                    g[i] += (1.0 - h[i+lag])
                    if i+lag < n:

def repair(h, n):

        if abs(diff) < 1e-14:

        for i in idx:
            if abs(diff) < 1e-14:
                break
            room = (1.0 - h[i]) if diff > 0 else h[i]

    corr = compute_corr(h)
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -80, 80)
    val = m + np.log(np.sum(np.exp(s))) / alpha
    if topk > 1:

        val += flat_w * np.sum((sc - sc.mean())**2)

            break
        for tk in [1, 3, 5]:
            if time.time() > t1:
                break
            for fw in [0.0, 0.005, 0.02]:
                if time.time() > t1:
                    break
                hn = slsqp(h, n, a, tk, fw, 300)
                c5n = compute_c5(hn, n)
                if c5n < best_c5 - 1e-14:
                    best_h, best_c5 = hn, c5n
                    h = hn
                elif c5n < compute_c5(h, n) - 1e-14:
                    h = hn
    if compute_c5(h, n) < best_c5:
        best_h, best_c5 = h.copy(), compute_c5(h, n)

    # Phase 2: Full CD (25% budget)
    t2 = t0 + budget_s * 0.50
    h_cd, c5_cd = full_cd(h, n, 80, t2)
    if c5_cd < best_c5 - 1e-14:
        best_h, best_c5 = h_cd, c5_cd
    h = best_h.copy()

    # Phase 3: Subgradient + focus (20% budget)
    t3 = t0 + budget_s * 0.68
    for _ in range(30):
        if time.time() > t3:
            break
        h_sg, c5_sg = subgrad_descent(h, n, t3, 15, 0.02)
        if c5_sg < best_c5 - 1e-14:
            best_h, best_c5 = h_sg, c5_sg
            h = h_sg
        h_fl = focus_lag_slsqp(h, n, t3)
        c5_fl = compute_c5(h_fl, n)
        if c5_fl < best_c5 - 1e-14:
            best_h, best_c5 = h_fl, c5_fl
            h = h_fl
        hn2 = slsqp(h, n, 50000, 3, 0.01, 200)
        c5_n2 = compute_c5(hn2, n)
        if c5_n2 < best_c5 - 1e-14:
            best_h, best_c5 = hn2, c5_n2
            h = hn2

    # Phase 4: L-BFGS polish (10% budget)
    t4 = t0 + budget_s * 0.80
    for a in [30000, 80000]:
        if time.time() > t4:
            break
        for pn in [500, 5000]:
            if time.time() > t4:
                break
            hn = lbfgs_pen(h, n, a, 3, 0.01, pn, 2000)
            c5n = compute_c5(hn, n)
            if c5n < best_c5 - 1e-14:
                best_h, best_c5 = hn, c5n
                h = hn

    # Phase 5: Final CD (10% budget)
    t5 = t0 + budget_s * 0.92
    h_cd2, c5_cd2 = full_cd(best_h, n, 60, t5)
    if c5_cd2 < best_c5 - 1e-14:
        best_h, best_c5 = h_cd2, c5_cd2

    # Phase 6: Try other n (5% budget)
    t6 = t0 + budget_s * 0.98
    for n_alt in [60, 65, 72, 75, 80, 90]:
        if time.time() > t6:
            break
        if n_alt == best_n:
            continue
        for stk in range(3):
            hs = make_diverse(n_alt, seed + n_alt * 7 + stk)
            for hs_i in hs[:3]:
                if time.time() > t6:
                    break
                ha = slsqp(hs_i, n_alt, 20000, 3, 0.01, 200)
                ha = full_cd(ha, n_alt, 10, t6)
                c5a = compute_c5(ha, n_alt)
                if c5a < best_c5 - 1e-14:
                    best_h, best_c5, best_n = ha, c5a, n_alt

    # Final SLSQP polish
    if best_h is not None and time.time() < t0 + budget_s * 0.995:
        hf = slsqp(best_h, best_n, 100000, 5, 0.02, 400)
        c5f = compute_c5(hf, best_n)
        if c5f < best_c5 - 1e-14:
            best_h, best_c5 = hf, c5f
        hc, c5c = full_cd(best_h, best_n, 30, t0 + budget_s * 0.998)
        if c5c < best_c5 - 1e-14:
            best_h, best_c5 = hc, c5c

    best_h = repair(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return (best_h, best_c5, best_n)
