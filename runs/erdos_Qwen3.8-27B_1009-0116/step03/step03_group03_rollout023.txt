def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            step = min(abs(diff), abs(1.0 - h[idx]) if diff > 0 else h[idx])
            if step < 1e-15:
                continue
            h[idx] += step if diff > 0 else -step
            diff -= step if diff > 0 else -step
            if abs(diff) < 1e-12:
                break
    return h

def _grad_lag(h, n, m):
    k = m - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n - k])
        g[:n - k] -= h[k:]
    else:
        k2 = -k
        g[:n - k2] += (1.0 - h[k2:])
        g[k2:] -= h[:n - k2]
    return g

    c = _corr(x)
    mx = np.max(c)
    s = np.clip(alpha * (c - mx), -50, 50)
    return mx + np.log(np.sum(np.exp(s))) / alpha
