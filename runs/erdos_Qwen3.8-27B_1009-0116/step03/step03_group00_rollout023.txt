This is a 1D convex PL minimization. I can solve it by:

2. Its subgradient at d is a_{k*(d)} where k*(d) = argmax_k (c[k] + d*a_k)

4. The minimum is where the subgradient contains 0

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return h

def _c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full')
    return float(np.max(c)) * dx

def _compute_S(h, n):
    """S[k, p] = d(c[k])/d(h[p]) where c[k] = corr_full[k]*dx"""
    dx = 2.0 / n
    v = 1.0 - h
    S = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        if k >= n-1:
            r = k - (n-1)
            for p in range(n):
                g = 0.0
                if p <= n-1-r:
                    g += v[p+r]
                if r <= p:
                    g -= h[p-r]
                S[k, p] = g * dx
        else:
            s = (n-1) - k
            for p in range(n):
                g = 0.0
                if p >= s:
                    g += v[p-s]
                if p <= n-1-s:
                    g -= h[p]
                S[k, p] = g * dx
    return S

def _optimal_d(c, S, i, j, h):
    """Find optimal d for pair swap (i,j) to minimize max(c + d*(S[:,i]-S[:,j]))"""
    d_min = max(-h[i], h[j]-1.0)
    d_max = min(1.0-h[i], h[j])
    if d_min > d_max:
        return 0.0
    a = S[:, i] - S[:, j]
    
    def f(d):
        return np.max(c + d * a)
    
    # Ternary search
    lo, hi = d_min, d_max
    for _ in range(30):
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        if f(m1) < f(m2):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2

def _pair_swap_local_search(h, n, budget_end, max_rounds=50):
    """Local search via optimal pair swaps"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, 'full')
    c = corr * dx
    c5_cur = np.max(c)
    S = _compute_S(h, n)
    
    for rnd in range(max_rounds):
        if time.time() > budget_end:
            break
        improved = False
        for i in range(n):
            if time.time() > budget_end:
                break
            for j in range(i+1, n):
                d = _optimal_d(c, S, i, j, h)
                if abs(d) < 1e-12:
                    continue
                h[i] += d
                h[j] -= d
                # Recompute
                corr = np.correlate(h, 1.0 - h, 'full')
                c = corr * dx
                new_c5 = np.max(c)
                if new_c5 < c5_cur - 1e-15:
                    c5_cur = new_c5
                    S = _compute_S(h, n)
                    improved = True
                else:
                    h[i] -= d
                    h[j] += d
        if not improved:
            break
    return h, c5_cur
