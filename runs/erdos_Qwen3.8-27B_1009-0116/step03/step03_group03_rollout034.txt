import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, dx):
    return np.max(compute_corr(h)) * dx

def grad_lag(h, n, m):
    k = m - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n - k])
        g[:n - k] -= h[k:]
    else:
        g[:n + k] += (1.0 - h[-k:])
        g[-k:] -= h[:n + k]
    return g

def smooth_fg(x, alpha, n):
    c = compute_corr(x)
    m = c.max()
    sh = np.clip(alpha * (c - m), -500, 500)
    es = np.exp(sh)
    es_sum = es.sum()
    f = m + np.log(es_sum) / alpha
    w = es / es_sum
    g = np.zeros(n)
    for mi in range(2 * n - 1):
        if w[mi] > 1e-12:
            g += w[mi] * grad_lag(x, n, mi)
    return f, g

def slsqp_obj(x, alpha, n):
    return smooth_fg(x, alpha, n)[0]

def slsqp_jac(x, alpha, n):
    return smooth_fg(x, alpha, n)[1]

def sum_con(x, target):
    return np.sum(x) - target

def sum_con_jac(x, target):
    return np.ones(len(x))

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
            else:
                add = max(diff, h[idx] - 1.0)
            h[idx] += add
            diff -= add
            if abs(diff) < 1e-10:
                break
    return h

def fw_point(g, n):
    target = n // 2
    idx = np.argsort(g)
    p = np.zeros(n)
    p[idx[:target]] = 1.0
    return p

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    con = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (target,)}]

    h0 = repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)

    def _tl():
        return budget_s - (time.time() - t0)

    # Phase 1: SLSQP with exact gradient, multi-alpha
    for alpha in [500, 2000, 8000, 30000, 100000]:
        if _tl() < 120:
            break
        try:
            res = minimize(
                slsqp_obj, best_h, args=(alpha, n), jac=slsqp_jac,
                method='SLSQP', bounds=bounds, constraints=con,
                options={'maxiter': 500, 'ftol': 1e-16, 'iprint': 0})
            hc = repair(res.x, n)
            c5 = compute_c5(hc, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, hc
        except Exception:
            pass

    # Phase 2: Frank-Wolfe + subgradient hybrid
    if _tl() > 100:
        h = best_h.copy()
        c_ideal = best_c5 / dx
        for it in range(80000):
            if _tl() < 100:
                break
            c = compute_corr(h)
            c_max = c.max()
            active = np.where(c > 0.99 * c_max)[0]
            g = np.zeros(n)
            for mi in active:
                g += grad_lag(h, n, mi)
            g /= len(active)
            gn = np.linalg.norm(g)
            if gn < 1e-14:
                break
            gap = c_max - c_ideal
            if gap < 1e-10:
                step = 0.02 / (1.0 + 0.001 * it)
            else:
                step = min(gap / (gn * gn), 0.3)
            # Mix of subgradient and FW
            p_fw = fw_point(g, n)
            h_new = h - step * g
            h_new = 0.5 * h_new + 0.5 * (h + 0.1 * (p_fw - h))
            h_new = repair(h_new, n)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_c5, best_h = c5_new, h_new.copy()
            c_ideal = min(c_ideal, c5_new / dx)
            h = h_new

    if _tl() > 60:

        T = 0.004

            if _tl() < 60:

            i = rng.randint(n)
            j = (i + 1 + rng.randint(n - 1)) % n
            d = rng.uniform(-0.1, 0.1)
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:

                c_new = compute_c5(h, dx)
                if c_new < c_cur or (c_new - c_cur) / T < np.log(rng.random()):
                    c_cur = c_new
                    if c_new < best_c5:
                        best_c5, best_h = c_new, h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d

    # Phase 4: Multi-restart SLSQP
    if _tl() > 80:
        for r in range(5):
            if _tl() < 80:
                break
            h_pert = best_h + 0.015 * rng.randn(n)
            h_pert = repair(h_pert, n)
            for alpha in [20000, 80000]:
                if _tl() < 50:
                    break
                try:
                    res = minimize(
                        slsqp_obj, h_pert, args=(alpha, n), jac=slsqp_jac,
                        method='SLSQP', bounds=bounds, constraints=con,
                        options={'maxiter': 400, 'ftol': 1e-16, 'iprint': 0})
                    hc = repair(res.x, n)
                    c5 = compute_c5(hc, dx)
                    if c5 < best_c5:
                        best_c5, best_h = c5, hc
                except Exception:
                    pass

    # Phase 5: Fine pair search
    if _tl() > 20:
        h = best_h.copy()
        c_cur = compute_c5(h, dx)
        for it in range(200000):
            if _tl() < 20:
                break
            i = rng.randint(n)
            j = (i + 1 + rng.randint(n - 1)) % n
            for d in (0.0003, 0.001, 0.003, -0.0003, -0.001, -0.003):
                ni, nj = h[i] + d, h[j] - d
                if 0 <= ni <= 1 and 0 <= nj <= 1:
                    h[i], h[j] = ni, nj
                    c = compute_c5(h, dx)
                    if c < c_cur:
                        c_cur = c
                        if c < best_c5:
                            best_c5, best_h = c, h.copy()
                        break
                    else:
                        h[i], h[j] = ni - d, nj + d

    # Phase 6: Final SLSQP polish
    if _tl() > 30:
        try:
            res = minimize(
                slsqp_obj, best_h, args=(200000, n), jac=slsqp_jac,
                method='SLSQP', bounds=bounds, constraints=con,
                options={'maxiter': 600, 'ftol': 1e-16, 'iprint': 0})
            hc = repair(res.x, n)
            c5 = compute_c5(hc, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, hc
        except Exception:
            pass

    best_h = repair(best_h, n)
    best_c5 = compute_c5(best_h, dx)
    return (best_h, best_c5, n)
