import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth_val_grad(h, alpha):
    corr = _corr(h)
    n = len(h)
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    w = np.exp(s)
    wsum = w.sum()
    w /= wsum
    val = m + np.log(wsum) / alpha
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        wk = w[m_idx]
        if wk < 1e-12:
            continue
        sh = m_idx - (n - 1)
        a = max(0, -sh)
        b = min(n - 1, n - 1 - sh)
        if a > b:
            continue
        j1lo = max(0, a + sh)
        j1hi = min(n - 1, b + sh)
        if j1lo <= j1hi:
            grad[j1lo:j1hi+1] += wk * (1.0 - h[j1lo - sh:j1hi - sh + 1])
        j2lo = max(0, a, -sh)
        j2hi = min(n - 1, b, n - 1 - sh)
        if j2lo <= j2hi:
            grad[j2lo:j2hi+1] -= wk * h[j2lo + sh:j2hi + sh + 1]
    return val, grad

def _obj(x, alpha, target, pen):
    v, g = _smooth_val_grad(x, alpha)
    v += pen * (np.sum(x) - target) ** 2
    g = g + 2.0 * pen * (np.sum(x) - target)
    return v, g

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return h

def _slsqp(h, n, dx, alpha, maxiter, deadline):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    best_h, best_c5 = h.copy(), _c5(h, dx)
    for pen in [10.0, 100.0, 1000.0]:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                lambda x: _obj(x, alpha, target, pen)[0],
                best_h, method='SLSQP', bounds=bounds, constraints=cons,
                jac=lambda x: _obj(x, alpha, target, pen)[1],
                options={'maxiter': maxiter, 'ftol': 1e-18}
            )
            hc = _repair(res.x, n)
            c5v = _c5(hc, dx)
            if c5v < best_c5:
                best_h, best_c5 = hc, c5v
        except Exception:
            pass
    return best_h, best_c5

def _topk_focus(h, n, dx, k, deadline):
    best_h, best_c5 = h.copy(), _c5(h, dx)
    for it in range(20):
        if time.time() >= deadline:
            break
        corr = _corr(best_h)
        top_idx = np.argsort(corr)[-k:]
        top_corr = corr[top_idx]
        t = np.max(top_corr) + 1e-8
        alpha = 5000.0
        def obj_k(x):
            ck = _corr(x)
            tk = ck[top_idx]
            m = np.max(tk)
            s = alpha * (tk - m)
            s = np.clip(s, -300, 300)
            return m + np.log(np.sum(np.exp(s))) / alpha
        bounds = [(0.0, 1.0)] * n

            c5v = _c5(hc, dx)
            if c5v < best_c5:
                best_h, best_c5 = hc, c5v
        except Exception:
            pass
    return best_h, best_c5

def _perturb(h, n, rng, kind):
    hp = h.copy()
    if kind == 0:
        block = rng.randint(3, 10)
        i = rng.randint(0, n - block)
        seg = hp[i:i+block].copy()
        hp[i:i+block] = np.roll(seg, rng.randint(1, block))
    elif kind == 1:
        f = rng.randint(1, 6)
        ph = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.03, 0.12)
        hp += amp * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
    elif kind == 2:
        f1, f2 = rng.randint(1, 5), rng.randint(1, 5)
        p1, p2 = rng.uniform(0, 6.28), rng.uniform(0, 6.28)
        amp = rng.uniform(0.03, 0.1)
        t = np.arange(n) / n
        hp += amp * np.sin(2*np.pi*f1*t + p1) * np.cos(2*np.pi*f2*t + p2)
    elif kind == 3:
        j = rng.randint(n // 5, 4 * n // 5)
        hp = np.concatenate([h[j:], h[:j]])
    elif kind == 4:
        m = rng.randint(n // 3, 2 * n // 3)
        hp[:m] = h[:m] + rng.uniform(-0.08, 0.08, m)
        hp[m:] = h[m:] + rng.uniform(-0.08, 0.08, n - m)
    else:
        hp = h + rng.normal(0, 0.05, n)
    return _repair(hp, n)

def _fine_search(h, n, dx, rng, deadline, n_tries=200000):
    best_c5 = _c5(h, dx)
    h = h.copy()
    scales = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    per_scale = max(1000, n_tries // len(scales))
    for sc in scales:
        improved = False
        count = 0
        for _ in range(per_scale):
            if time.time() >= deadline:
                return h, best_c5
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = sc * (2.0 * rng.random() - 1.0)
            ni, nj = h[i] + d, h[j] - d
            if ni < -1e-12 or ni > 1.0 + 1e-12 or nj < -1e-12 or nj > 1.0 + 1e-12:
                continue
            h[i], h[j] = ni, nj
            c5v = _c5(h, dx)
            count += 1
            if c5v < best_c5 - 1e-15:
                best_c5 = c5v
                improved = True
            else:
                h[i], h[j] = (h[i] - d) if i != j else h[i], (h[j] + d) if i != j else h[j]
                h[i], h[j] = ni - d + d, nj + d - d  # restore
                h[i], h[j] = ni - d, nj + d
        if not improved and sc < 0.002:
            break
    return h, best_c5

def _block_search(h, n, dx, rng, deadline, n_iters=20000):
    best_c5 = _c5(h, dx)
    h = h.copy()
    for _ in range(n_iters):
        if time.time() >= deadline:
            break
        bs = rng.choice([1, 2, 3])
        i = rng.randint(0, n - bs)
        j = rng.randint(0, n - bs)
        if abs(i - j) < bs:
            continue
        eps = rng.uniform(0.001, 0.01) * (1.0 if rng.random() < 0.5 else -1.0)
        ok = True
        if (h[i:i+bs] + eps).min() < -1e-12 or (h[i:i+bs] + eps).max() > 1.0 + 1e-12:
            ok = False
        if (h[j:j+bs] - eps).min() < -1e-12 or (h[j:j+bs] - eps).max() > 1.0 + 1e-12:
            ok = False
        if not ok:
            continue
        h[i:i+bs] += eps
        h[j:j+bs] -= eps
        c5v = _c5(h, dx)
        if c5v < best_c5 - 1e-15:
            best_c5 = c5v
        else:
            h[i:i+bs] -= eps
            h[j:j+bs] += eps
    return h, best_c5

def _anneal(h, n, dx, rng, deadline):
    best_c5 = _c5(h, dx)
    hc = h.copy()
    bc = best_c5
    T = 0.004
    Tmin = 0.00003
    cool = 0.9998
    while time.time() < deadline and T > Tmin:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = 0.01 * (2.0 * rng.random() - 1.0)
        ni, nj = hc[i] + d, hc[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        hc[i], hc[j] = ni, nj
        c5n = _c5(hc, dx)
        if c5n < bc or rng.random() < np.exp(-(c5n - bc) / T):
            bc = c5n
            if c5n < best_c5:
                best_c5 = c5n
        else:
            hc[i], hc[j] = ni - d, nj + d
        T *= cool
    return _repair(hc, n), best_c5
