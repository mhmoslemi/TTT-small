import numpy as np

import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, dx):
    return float(np.max(compute_corr(h)) * dx)

def smooth_max(c, alpha):
    m = float(np.max(c))
    if alpha < 1e-10:
        return m, np.full(len(c), 1.0/len(c))
    s = np.clip(alpha * (c - m), -500.0, 500.0)
    es = np.exp(s)
    weights = es / es.sum()
    val = m + np.log(es.sum()) / alpha
    return val, weights

def smooth_max_grad(c, h, n, alpha):
    """Returns gradient of smooth_max(correlate(h,1-h)) w.r.t. h"""
    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -500.0, 500.0)
    es = np.exp(s)
    w = es / es.sum()  # softmax weights over lags
    
    g = np.zeros(n)
    for k in range(len(c)):
        if w[k] < 1e-10:
            continue
        lag = k - (n - 1)
        for i in range(n):
            j1 = i + lag
            j2 = i - lag
            term = 0.0
            if 0 <= j1 < n:
                term += (1.0 - h[j1])
            if 0 <= j2 < n:
                term -= h[j2]
            g[i] += w[k] * term
    return g

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    # Proportional scaling
    if s > 1e-12:
        h = np.clip(h * target / s, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    # Distribute remaining
    if diff > 0:
        order = np.argsort(1.0 - h)
    else:
        order = np.argsort(h)
    for idx in order:
        if diff > 0:
            a = min(diff, 1.0 - h[idx])
            h[idx] += a
            diff -= a
        else:
            a = min(-diff, h[idx])
            h[idx] -= a
            diff += a
        if abs(diff) < 1e-12:
            break
    return h

                def sym_grad(x):
                    return sym_smooth_grad(x, alpha)
                res = minimize(sym_obj, x_init, jac=sym_grad, method='L-BFGS-B',
                               bounds=bounds_h,
                               options={'maxiter': 5000, 'ftol': 1e-20,
                                        'gtol': 1e-14, 'maxls': 50})
                hx = full_h_from_sym(res.x)
                hx = repair(hx, n)
                c5 = compute_c5(hx, dx)
                if c5 < best_c5:
                    best_h, best_c5 = hx.copy(), c5
                    x_init = res.x.copy()
            except Exception:
                pass

    # Phase 3: Active-set subgradient on top lags
    if elapsed() < budget_s - 120:
        h = best_h.copy()
        c5_cur = compute_c5(h, dx)
        for ph in range(12):
            if elapsed() > budget_s - 120:
                break
            step = 0.04 * (0.45 ** ph)
            for it in range(300):
                if elapsed() > budget_s - 120:
                    break
                c = compute_corr(h)
                top_k = 4
                active = np.argsort(c)[-top_k:]
                g = np.zeros(n)
                for k in active:
                    m_lag = k - (n - 1)
                    if m_lag >= 0:
                        nl = n - m_lag
                        if nl > 0:
                            g[m_lag:] += (1.0 - h[:nl])
                            g[:nl] -= h[m_lag:]
                    else:
                        m2 = -m_lag
                        nl = n - m2
                        if nl > 0:
                            g[:nl] += (1.0 - h[m2:])
                            g[m2:] -= h[:nl]
                g /= top_k
                h_new = repair(h - step * g, n)
                c5_new = compute_c5(h_new, dx)
                if c5_new < c5_cur - 1e-15:
                    h, c5_cur = h_new, c5_new
                    if c5_cur < best_c5:
                        best_h, best_c5 = h.copy(), c5_cur
                elif c5_new < c5_cur + 1e-12:
                    h = h_new
                else:
                    break

    # Phase 4: Pair local search (all pairs, multiple deltas)
    if elapsed() < budget_s - 50:
        h = best_h.copy()
        c5_cur = compute_c5(h, dx)
        deltas = [0.003, 0.001, 0.0004, -0.003, -0.001, -0.0004]
        for _pass in range(40):
            if elapsed() > budget_s - 50:
                break
            imp = False
            for i in range(n):
                if elapsed() > budget_s - 50:
                    break
                hi = h[i]
                for j in range(i + 1, n):
                    hj = h[j]
                    for d in deltas:
                        ni, nj = hi + d, hj - d
                        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                            continue
                        h[i], h[j] = ni, nj
                        cv = compute_c5(h, dx)
                        if cv < c5_cur - 1e-14:
                            c5_cur, hi, hj = cv, ni, nj
                            imp = True
                        else:
                            h[i], h[j] = hi, hj
                    h[i], h[j] = hi, hj
                h[i] = hi
            if c5_cur < best_c5:
                best_h, best_c5 = h.copy(), c5_cur
            if not imp:
                break

    # Phase 5: Ultra-fine random pairs
    if elapsed() < budget_s - 20:
        h = best_h.copy()
        c5_cur = compute_c5(h, dx)
        opts = [3e-4, 1e-4, 5e-5, -3e-4, -1e-4, -5e-5]
        while elapsed() < budget_s - 20:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = opts[rng.randint(len(opts))]
            ni, nj = h[i] + d, h[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h[i], h[j] = ni, nj
                cv = compute_c5(h, dx)
                if cv < c5_cur:
                    c5_cur = cv
                    if c5_cur < best_c5:
                        best_h, best_c5 = h.copy(), c5_cur
                else:
                    h[i], h[j] = h[i] - d, h[j] + d

    # Phase 6: Final SLSQP polish
    if elapsed() < budget_s - 5:
        try:
            def obj_final(x):
                return smooth_max(compute_corr(x), 200000)
            def grad_final(x):
                return smooth_grad_full(x, n, 200000)
            def eq_final(x):
                return np.sum(x) - n / 2.0
            res = minimize(obj_final, best_h, jac=grad_final, method='SLSQP',
                           bounds=bounds_n,
                           constraints=[{'type': 'eq', 'fun': eq_final}],
                           options={'maxiter': 2000, 'ftol': 1e-18})
            hc = repair(res.x, n)
            c5 = compute_c5(hc, dx)
            if c5 < best_c5:
                best_h, best_c5 = hc, c5
        except Exception:
            pass

    best_h = repair(best_h, n)
    best_c5 = compute_c5(best_h, dx)
    return (best_h, best_c5, n)
