The key insight: we want h to have a flat autocorrelation (like a Barker sequence or Zadoff-Chu sequence). This is a well-studied problem in signal processing!

Barker sequences have low autocorrelation sidelobes. For real-valued sequences with values in {0,1} (or {+1,-1}), the minimum achievable maximum sidelobe is known for small lengths.

Let me convert: if h[i] = (1 + s[i])/2 where s[i] ∈ {-1, +1} and sum s[i] = 0 (since sum h = n/2 means sum s = 0), then:
h[i]*h[j] = (1+s[i])(1+s[j])/4
sum_i h[i]*h[i+k] = sum_i (1+s[i])(1+s[i+k])/4 = (n + sum s + sum s[i+k] + sum s[i]*s[i+k])/4 = (n + 0 + 0 + A[k])/4 = (n + A[k])/4

where A[k] is the autocorrelation of s.

And corr[k] of h = n/2 - sum h[i]*h[i+k] = n/2 - (n+A[k])/4 = (2n - n - A[k])/4 = (n - A[k])/4

So minimizing max_k corr[k] is equivalent to maximizing min_k A[k] (the minimum autocorrelation of s).

Wait, that's not quite right because we're looking at linear (not circular) correlation, and the valid ranges are different.

But this gives me intuition: I want a sequence s with values in {-1,+1} (or near that), with as flat an autocorrelation as possible (all sidelobes as close to 0 as possible, or as equal as possible).

For the continuous case (h ∈ [0,1] not just {0,1}), the optimization is smoother and might achieve a lower max.

import numpy as np
from scipy.optimize import minimize
import time

_g_alpha = 5000
_g_topk = 5
_g_flatw = 0.01
_g_n = 70

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(5000):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        d = target - s
        if abs(d) < 1e-14:
            break
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-14:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
    return h

def smax(x):
    corr = get_corr(x)
    m = np.max(corr)
    s = _g_alpha * (corr - m)
    s = np.clip(s, -600, 600)
    val = m + np.log(np.sum(np.exp(s))) / _g_alpha
    if _g_topk > 1:
        sc = np.sort(corr)[::-1][:_g_topk]
        val += _g_flatw * np.sum((sc - sc.mean())**2)
    return val

def eq_constraint(x):
    return x.sum() - _g_n / 2.0

def nm_obj(x):
    if x.min() < -0.05 or x.max() > 1.05:
        return 1e6
    return smax(x) + 1000.0 * (x.sum() - _g_n / 2.0)**2

def slsqp(h, n, alpha, topk, flatw, maxiter):
    global _g_alpha, _g_topk, _g_flatw, _g_n
    _g_alpha, _g_topk, _g_flatw, _g_n = alpha, topk, flatw, n
    b = [(0.0, 1.0)] * n
    c = [{'type': 'eq', 'fun': eq_constraint}]
    try:
        r = minimize(smax, h, method='SLSQP', bounds=b, constraints=c,
                     options={'maxiter': maxiter, 'ftol': 1e-18})
        return repair(r.x, n)
    except Exception:
        return h

def nelder(h, n, maxiter):
    global _g_alpha, _g_topk, _g_flatw, _g_n
    _g_alpha, _g_topk, _g_flatw, _g_n = 50000, 5, 0.02, n
    sx = h + 0.005 * np.ones(n)
    try:
        r = minimize(nm_obj, h, method='Nelder-Mead',
                     options={'maxiter': maxiter, 'fatol': 1e-20, 'initial_simplex': sx})
        return repair(r.x, n)
    except Exception:
        return h

def coord(h, n, t_end, rng, passes):
    h = h.copy()
    best = compute_c5(h, n)
    for p in range(passes):
        if time.time() > t_end:
            break
        sc = max(0.015 * (0.93 ** p), 1e-7)
        for _ in range(4 * n):
            if time.time() > t_end:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                j = (j + 1) % n
            hi, hj = h[i], h[j]
            for sgn in (1, -1):
                for s in [sc, sc * 0.5, sc * 0.25, sc * 0.1]:
                    d = sgn * s
                    ni, nj = hi + d, hj - d
                    if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                        continue
                    h[i], h[j] = ni, nj
                    c5 = compute_c5(h, n)
                    if c5 < best - 1e-15:
                        best = c5
                        hi, hj = ni, nj
                        break
                    else:
                        h[i], h[j] = hi, hj
        if p % 50 == 49 and p > 0:
            h2 = slsqp(repair(h, n), n, 30000, 5, 0.02, 200)
            c5 = compute_c5(h2, n)
            if c5 < best:
                best = c5
                h = h2
    return h, best

def sa(h, n, t_end, rng):
    h = h.copy()
    best = compute_c5(h, n)
    bh = h.copy()
    T = 0.005
    step = 0.02
    for _ in range(500000):
        if time.time() > t_end:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            j = (j + 1) % n
        d = (rng.random() * 2 - 1) * step
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        nc = compute_c5(h, n)
        if nc < best:
            best = nc
            bh = h.copy()
        elif rng.random() >= np.exp(-(nc - best) / max(T, 1e-10)):
            h[i], h[j] = ni - d, nj + d
        T *= 0.99995
        step *= 0.99999
    return bh, best

def make_start(n, seed, kind):
    rng = np.random.RandomState(seed)
    if kind == 0:
        h = 0.5 + 0.3 * rng.randn(n)
    elif kind == 1:
        h = np.where(np.arange(n) < n // 2, 1.0, 0.0) + 0.05 * rng.randn(n)
    elif kind == 2:
        f = rng.randint(1, 6)
        ph = rng.uniform(0, 6.283)
        h = 0.5 + 0.3 * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
    elif kind == 3:
        s = rng.choice([-1.0, 1.0], n)
        for _ in range(200):
            if abs(s.sum()) <= 2:
                break
            idx = rng.randint(n)
            s[idx] *= -1
        h = (1.0 + s) / 2.0
    else:
        f = rng.randint(1, 4)
        h = 0.5 + 0.25 * np.cos(2 * np.pi * f * np.arange(n) / n + 0.3)
    return repair(h, n)
