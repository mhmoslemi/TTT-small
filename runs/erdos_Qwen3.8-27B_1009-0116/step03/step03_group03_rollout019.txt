def _repair(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if s < 1e-12:
        return np.full(len(h), 0.5)
    target_f = float(target)
    if s > 1e-12:
        h *= target_f / s
    h = np.clip(h, 0.0, 1.0)
    diff = target_f - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
            else:
                add = max(diff, h[idx] - 1.0)
            h[idx] += add
            diff -= add
            if abs(diff) < 1e-12:

    return h

        try:
            res = minimize(
                lambda x: _smooth_obj(x, 1000000, target),
                best_h, method='SLSQP', bounds=bounds,
                options={'maxiter': 1000, 'ftol': 1e-19, 'iprint': 0})
            hc = _repair(res.x, target)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, hc
        except Exception:
            pass

    # Phase 5: Fine perturbation
    if tleft() > 50:
        fp_budget = min(80, tleft() - 50)
        h_fp, c5_fp = _fine_perturb(best_h, dx, target, fp_budget, rng)
        if c5_fp < best_c5:
            best_c5, best_h = c5_fp, h_fp

    # Phase 6: Final SLSQP
    if tleft() > 25:
        try:
            res = minimize(
                lambda x: _smooth_obj(x, 2000000, target),
                best_h, method='SLSQP', bounds=bounds,
                options={'maxiter': 500, 'ftol': 1e-19, 'iprint': 0})
            hc = _repair(res.x, target)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, hc
        except Exception:
            pass

    # Phase 7: Ultra-fine
    if tleft() > 15:
        h_cur = best_h.copy()
        c5_cur = best_c5
        for _ in range(20000):
            if time.time() - t0 > budget_s - 10:
                break
            i = rng.randint(n)
            j = (i + 1 + rng.randint(n - 1)) % n
            d = (rng.rand() - 0.5) * 0.0008
            ni, nj = h_cur[i] + d, h_cur[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h_cur[i], h_cur[j] = ni, nj
                c = _c5(h_cur, dx)
                if c < c5_cur:
                    c5_cur = c
                    best_c5 = c
                    best_h = h_cur.copy()
                else:
                    h_cur[i], h_cur[j] = ni - d, nj + d

    best_h = _repair(best_h, target)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
