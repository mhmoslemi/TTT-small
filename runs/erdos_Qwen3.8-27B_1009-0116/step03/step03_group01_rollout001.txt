from scipy.optimize import minimize, linprog
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / len(h))

def _correlate_vals(h):
    return np.correlate(h, 1.0 - h, mode='full') * 2.0 / len(h)

def _grad_lag(x, k):
    """Gradient of correlation at lag k (0-indexed in full output)."""
    n = len(x)
    lag = k - (n - 1)
    g = np.zeros(n)
    lo = max(0, lag)
    hi = min(n, n + lag)
    for i in range(lo, hi):
        j = i - lag
        g[i] += (1.0 - x[j])
        g[j] -= x[i]
    return g * (2.0 / n)

def _smooth_obj(x, alpha):
    vals = _correlate_vals(x)
    m = vals.max()
    s = np.clip(alpha * (vals - m), -500, 500)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def _smooth_grad(x, alpha):
    n = len(x)
    vals = _correlate_vals(x)
    m = vals.max()
    s = np.clip(alpha * (vals - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    # Weighted sum of per-lag gradients
    g = np.zeros(n)
    for k in range(2*n-1):
        if w[k] > 1e-14:
            g += w[k] * _grad_lag(x, k)
    return g

def _slsqp(h, alpha, iters, ftol):
    n = len(h)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,

    res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha,),
