def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()

        h *= target / s
        h = np.clip(h, 0.0, 1.0)

                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a

import numpy as np
from scipy.optimize import minimize
import time

_state = {}

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12: break
    return h

def _eval(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _all_f(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_fk(h, k_idx):
    n = len(h)
    k = k_idx - (n - 1)
    g = np.zeros(n)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    elif k > 0:
        m = n - k
        g[:m] += 1.0 - h[k:]
        g[k:] -= h[:m]
    else:
        m = n + k
        kk = -k
        g[kk:] += 1.0 - h[:m]
        g[:m] -= h[kk:]
    return g

def _obj_func(x):
    return x[-1]

def _obj_grad(x):
    n = _state['n']
    g = np.zeros(n + 1)
    g[-1] = 1.0
    return g

def _ineq_func(x):
    n = _state['n']
    top = _state['top']
    hv = x[:n]
    tv = x[-1]
    c = np.correlate(hv, 1.0 - hv, mode='full')
    return tv - c[top]

def _ineq_jac(x):
    n = _state['n']
    top = _state['top']
    hv = x[:n]
    K = len(top)
    J = np.zeros((K, n + 1))
    for r in range(K):
        J[r, :n] = -_grad_fk(hv, top[r])
        J[r, -1] = 1.0
    return J

def _eq_func(x):
    n = _state['n']
    return np.sum(x[:n]) - n / 2.0

def _eq_jac(x):
    n = _state['n']
    J = np.zeros(n + 1)
    J[:n] = 1.0
    return J

def _solve_slsqp(h, n, top_idx, max_iter=500):
    _state['n'] = n
    _state['top'] = np.array(top_idx)
    x0 = np.concatenate([h, [float(np.max(np.correlate(h, 1.0-h, mode='full')))]])
    bnds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    cons = [
        {'type': 'ineq', 'fun': _ineq_func, 'jac': _ineq_jac},
        {'type': 'eq', 'fun': _eq_func, 'jac': _eq_jac}
    ]
    try:
        res = minimize(_obj_func, x0, jac=_obj_grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-16})
        return _repair(res.x[:n], n)
    except Exception:
        return h.copy()

def _proj_grad_descent(h, n, top_idx, eta_init, max_iters):
    h = h.copy()
    eta = eta_init
    c5_prev = _eval(h)
    for _ in range(max_iters):

        vals = c[top_idx]
        a = 50.0

def _pair_search(h, n, max_passes, time_limit_fn):
    h = h.copy()
    c5c = _eval(h)
    steps = np.array([0.01, 0.005, 0.002, 0.001, 0.0005, -0.01, -0.005, -0.002, -0.001, -0.0005])
    for _pass in range(max_passes):
        if time_limit_fn():
            break
        improved = False
        for i in range(n):
            if time_limit_fn():
                break
            hi = h[i]
            for j in range(i + 1, n):
                if time_limit_fn():
                    break
                hj = h[j]
                for d in steps:
                    ni = hi + d
                    nj = hj - d
                    if ni < -1e-12 or ni > 1 + 1e-12 or nj < -1e-12 or nj > 1 + 1e-12:
                        continue
                    h[i] = ni
                    h[j] = nj
                    c5n = _eval(h)
                    if c5n < c5c - 1e-16:
                        c5c = c5n
                        hi = ni
                        hj = nj
                        improved = True
                    else:
                        h[i] = hi
                        h[j] = hj
                h[j] = hj
            h[i] = hi
        if not improved:
            break
    return h, c5c
