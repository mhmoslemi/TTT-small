import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _subgrad_top(h, n, K):
    c = _corr(h)
    top = np.argsort(c)[-K:]
    g = np.zeros(n)
    for m in top:
        for i in range(n):
            j = i + m - (n - 1)
            if 0 <= j < n:
                g[i] += (1.0 - h[j])
                g[j] -= h[i]
    return g / len(top)

def _golden_pair(h, i, j, n, dx, niters=35):
    S = h[i] + h[j]
    lo = max(0.0, S - 1.0)
    hi_v = min(1.0, S)
    if hi_v - lo < 1e-14:
        return _c5(h, dx)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi_v
    c_pt = b - gr * (b - a)
    d_pt = a + gr * (b - a)
    h[i], h[j] = c_pt, S - c_pt
    fc = _c5(h, dx)
    h[i], h[j] = d_pt, S - d_pt
    fd = _c5(h, dx)
    for _ in range(niters):
        if fc < fd:
            b = d_pt
            d_pt, fd = c_pt, fc
            c_pt = b - gr * (b - a)
            h[i], h[j] = c_pt, S - c_pt
            fc = _c5(h, dx)
        else:
            a = c_pt
            c_pt, fc = d_pt, fd
            d_pt = a + gr * (b - a)
            h[i], h[j] = d_pt, S - d_pt
            fd = _c5(h, dx)
    best_t = (a + b) / 2.0
    h[i], h[j] = best_t, S - best_t
    return _c5(h, dx)

def _obj_smooth(x, alpha):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500.0, 500.0)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def _eq_con(x, target):
    return np.sum(x) - target
