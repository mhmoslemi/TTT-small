def compute_corr(h):

def compute_c5(h, dx):

    h[ones] = 1.0
    return h

    best_score = float('inf')
    best_h = h.copy()

    cur_h = h.copy()

        zeros = np.where(cur_h < 0.5)[0]

        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        # Swap

        new_score = max_score(cur_h, n)
        
        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
        elif new_score == cur_score:
