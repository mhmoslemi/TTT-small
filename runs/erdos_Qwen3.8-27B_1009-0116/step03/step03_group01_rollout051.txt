    """Gradient of overlap at a given lag (in 'full' correlate indexing)."""
    n = len(h)
    g = np.zeros(n)
    # lag here is the actual offset: for 'full' output index k, lag_offset = k - (n-1)
    # But let's work with lag_offset directly
    
    # For correlate(h, 1-h, 'full')[k] where k = lag_offset + n - 1:
    # If lag_offset >= 0: sum_{i=0}^{n-1-lag_offset} h[i+lag_offset] * (1-h[i])
    # If lag_offset < 0: sum_{i=-lag_offset}^{n-1} h[i+lag_offset] * (1-h[i])
    
    lo = lag_offset  # this is the actual offset
    
    if lo >= 0:

        # d/dh[j]: 
        #   j appears as i+lo when i = j-lo: contributes (1-h[j-lo]) for j-lo in [0, n-1-lo]
        #   j appears as i when i = j: contributes -h[j+lo] for j in [0, n-1-lo]
        for i in range(n - lo):
            g[i + lo] += (1 - h[i])
            g[i] -= h[i + lo]
    else:

        #   j appears as i when i = j: contributes -h[j-L] for j in [L, n-1]

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / len(h))

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

    c = _corr(x)
    m = c.max()
    s = np.clip(a * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / a

        if active[k]:
            lo = k - (n - 1)
            g += _grad_single_lag(h, lo)

def _opt_slqp(h, n, alpha, iters, ftol):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,

                       options={'maxiter': iters, 'ftol': ftol})

def _projected_subgrad(h, n, iters, lr_init=0.005, decay=0.999):
    """Nesterov accelerated projected gradient on true max objective."""
    h = h.copy()
    y = h.copy()
    t = 1.0

    best_h = h.copy()

        sg, m = _subgrad_max(y, tol=0.02)

            best_h = h_new.copy()
        h = h_new
        t = t_new

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / len(h))

        if w[k] < 1e-14: continue
        g[:n - M] += w[k] * (1.0 - x[M:])

    else:

    for k in range(2 * len(h) - 1):
        if active[k]:
            g += _grad_lag(h, k - (len(h) - 1))
    return g / n_act

        h = _repair(y - lr * sg, n)
        c5 = _eval(h)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()
        t = t_n; lr *= 0.99992
        if lr < 1e-7: break
    return best_h, best_c5

def _pair_sweep(h, n, n_trials):
    c5c = _eval(h)
    any_imp = False
    for i in range(n):
        for j in range(i + 1, n):
            d_lo = max(-h[i], h[j] - 1.0)
            d_hi = min(1.0 - h[i], h[j])
            if d_hi - d_lo < 1e-10: continue
            best_d, best_c5 = 0.0, c5c
            for tt in range(n_trials + 1):
                d = d_lo + (d_hi - d_lo) * tt / n_trials
                if abs(d) < 1e-12: continue
                h[i] += d; h[j] -= d
                c5n = _eval(h)
                if c5n < best_c5 - 1e-14:
                    best_c5, best_d = c5n, d
                h[i] -= d; h[j] += d
            if best_d != 0.0:
                h[i] += best_d; h[j] -= best_d
                c5c = best_c5; any_imp = True
    return h, c5c, any_imp

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)

    def _el():
        return time.time() - t0

    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h, best_c5 = h.copy(), _eval(h)

    # Phase 1: SLSQP warm-up with increasing precision
    for alpha in [500, 5000, 50000, 500000]:
        if _el() > budget_s - 350: break
        h_new = _slqp(best_h, n, alpha, 5000)
        c5 = _eval(h_new)
        if c5 < best_c5: best_c5, best_h = c5, h_new

    # Phase 2: Projected subgradient on true max (Nesterov)
    if _el() < budget_s - 280:
        h_new, c5_new = _proj_subgrad(best_h, n, 6000, 0.004)
        if c5_new < best_c5: best_c5, best_h = c5_new, h_new
    if _el() < budget_s - 220:
        h_new, c5_new = _proj_subgrad(best_h, n, 6000, 0.0015)
        if c5_new < best_c5: best_c5, best_h = c5_new, h_new

    # Phase 3: Multi-start SLSQP for escaping local optima
    for _ in range(10):
        if _el() > budget_s - 200: break
        sc = 0.005 + 0.035 * rng.rand()
        hp = _repair(best_h + sc * rng.randn(n), n)
        h_new = _slqp(hp, n, 300000, 2500)
        c5 = _eval(h_new)
        if c5 < best_c5: best_c5, best_h = c5, h_new

    # Phase 4: Exhaustive pairwise grid search (primary refinement)
    h = best_h.copy()
    for ntr in [30, 80, 150, 300]:
        while _el() < budget_s - 120:
            h, c5c, imp = _pair_sweep(h, n, ntr)
            if c5c < best_c5: best_c5, best_h = c5c, h.copy()
            if not imp: break

    # Phase 5: Simulated annealing with adaptive temperature
    if _el() < budget_s - 60:
        h = best_h.copy(); c5c = _eval(h)
        T = 0.004
        for _ in range(100000):
            if _el() > budget_s - 60: break
            i, j = rng.randint(n), rng.randint(n)
            if i == j: continue
            d = 0.006 * rng.randn()
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n - c5c < 0 or rng.rand() < np.exp(-(c5n - c5c) / T):
                    c5c = c5n
                    if c5n < best_c5: best_c5, best_h = c5n, h.copy()
                else:
                    h[i], h[j] = h[i] - d, h[j] + d
            T *= 0.99995
            if T < 1e-7: break

    # Phase 6: Final fine pairwise grid
    if _el() < budget_s - 30:
        h = best_h.copy()
        for _ in range(200):
            if _el() > budget_s - 30: break
            h, c5c, imp = _pair_sweep(h, n, 400)
            if c5c < best_c5: best_c5, best_h = c5c, h.copy()
            if not imp: break

    # Phase 7: Tiny random coordinate moves
    if _el() < budget_s - 10:
        h = best_h.copy(); c5c = _eval(h)
        for _ in range(50000):
            if _el() > budget_s - 10: break
            i, j = rng.randint(n), rng.randint(n)
            if i == j: continue
            d = rng.choice([1, -1]) * rng.choice([1e-4, 5e-5, 1e-5])
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-16:
                    c5c = c5n
                    if c5n < best_c5: best_c5, best_h = c5n, h.copy()
                else:
                    h[i], h[j] = h[i] - d, h[j] + d

    # Phase 8: Final SLSQP polish
    if _el() < budget_s - 5:
        h_new = _slqp(best_h, n, 5000000, 5000)
        c5 = _eval(h_new)
        if c5 < best_c5: best_c5, best_h = c5, h_new

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
