import numpy as np
from scipy.optimize import minimize
import time

def _compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _smooth_obj(x, alpha, target):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha + 100.0 * (np.sum(x) - target) ** 2

def _slsqp_polish(h, n, dx, alpha, maxiter=200):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(lambda x: _smooth_obj(x, alpha, target), h,
                       method='SLSQP', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        hc = _repair(res.x, n)
        return hc, _compute_c5(hc, dx)
    except Exception:
        return h, _compute_c5(h, dx)

def _binary_sa(h_init, n, dx, rng, deadline, T_start=None, T_end=None, restarts=1):
    target = n // 2
    best_h = np.zeros(n)
    best_c5 = 1e9
    
    for r in range(restarts):
        h = (h_init >= 0.5).astype(float) if r == 0 else _make_binary(n, rng, r % 5)
        # Fix count
        cnt = int(h.sum())
        if cnt > target:
            idx = np.where(h == 1)[0]
            h[idx[rng.randint(0, len(idx), cnt - target)]] = 0
        elif cnt < target:
            idx = np.where(h == 0)[0]
            h[idx[rng.randint(0, len(idx), target - cnt)]] = 1
        
        cur_c5 = _compute_c5(h, dx)
        if T_start is None:
            Ts = max(0.001, cur_c5 * 0.05)
        else:
            Ts = T_start
        Te = T_end if T_end is not None else max(1e-5, Ts * 0.01)
        T = Ts
        cool = (Te / Ts) ** (1.0 / max(100, int((deadline - time.time()) * 5000)))
        
        while time.time() < deadline:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i], h[j] = 0.0, 1.0
            new_c5 = _compute_c5(h, dx)
            delta = new_c5 - cur_c5
            if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
                cur_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
            else:
                h[i], h[j] = 1.0, 0.0
            T = max(Te, T * cool)
        if time.time() >= deadline:
            break
    
    return best_h, best_c5

def _make_binary(n, rng, style):
    target = n // 2
    h = np.zeros(n)
    if style == 0:
        idx = rng.permutation(n)
        h[idx[:target]] = 1.0
    elif style == 1:
        h = (np.arange(n) % 2).astype(float)
        if int(h.sum()) != target:
            h[-1] = 1.0
    elif style == 2:
        for i in range(n):
            if (i * target) % n < target * (i // target + 1) / (n // target + 1):
                h[i] = 1.0
        # Simpler: evenly spaced
        h = np.zeros(n)
        positions = (np.arange(target) * n // target).astype(int)
        h[positions] = 1.0
    elif style == 3:
        # Two blocks
        h[:target//2] = 1.0
        h[target + target//2:target + target//2 + target//2] = 1.0
        h[rng.choice(n, target - 2*(target//2), replace=False)] = 1.0
        # Resample to get exactly target
        cnt = int(h.sum())
        if cnt != target:
            h = np.zeros(n)
            idx = rng.permutation(n)
            h[idx[:target]] = 1.0
    else:
        h = (np.sin(2*np.pi*np.arange(n)/n * 3) > 0).astype(float)
        cnt = int(h.sum())
        if cnt > target:
            idx = np.where(h==1)[0]
            h[idx[rng.randint(0,len(idx),cnt-target)]] = 0
        elif cnt < target:
            idx = np.where(h==0)[0]
            h[idx[rng.randint(0,len(idx),target-cnt)]] = 1
    return h

def _local_refine(h, n, dx, rng, deadline):
    """Fine local search: try adjusting pairs."""
    best_c5 = _compute_c5(h, dx)
    h = h.copy()
    target = n / 2.0
    scales = [0.02, 0.01, 0.005, 0.002, 0.001]
    for scale in scales:
        improved = False
        for _ in range(20000):
            if time.time() >= deadline:
                return h, best_c5
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = scale * (2.0 * rng.random() - 1.0)
            ni, nj = h[i] + d, h[j] - d
            if ni < -1e-10 or ni > 1+1e-10 or nj < -1e-10 or nj > 1+1e-10:

import numpy as np
from scipy.optimize import minimize
import time

def _compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _smooth_obj(x, alpha, target):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha + 100.0 * (np.sum(x) - target) ** 2

def _slsqp_polish(h, n, dx, alpha, maxiter=200):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(lambda x: _smooth_obj(x, alpha, target), h,
                       method='SLSQP', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        hc = _repair(res.x, n)
        return hc, _compute_c5(hc, dx)
    except Exception:
        return h, _compute_c5(h, dx)

def _make_binary(n, rng, style):
    target = n // 2
    if style == 0:
        h = np.zeros(n)
        h[rng.permutation(n)[:target]] = 1.0
    elif style == 1:
        h = np.zeros(n)
        positions = (np.arange(target) * n // target).astype(int)

        if int(h.sum()) != target:
            h = np.zeros(n)
            h[rng.permutation(n)[:target]] = 1.0
    elif style == 2:
        h = (np.arange(n) % 2).astype(float)
        if int(h.sum()) != target:
            h = np.zeros(n)
            h[rng.permutation(n)[:target]] = 1.0
    elif style == 3:
        h = np.zeros(n)
        pos = rng.permutation(n)
        step = max(1, n // (target * 2))
        for i in range(target):
            h[(pos[i] * step) % n] = 1.0
        if int(h.sum()) != target:
            h = np.zeros(n)
            h[rng.permutation(n)[:target]] = 1.0
    else:
        h = (np.sin(2.0 * np.pi * rng.uniform(1, 5) * np.arange(n) / n +
              rng.uniform(0, 6.28)) > 0).astype(float)
        cnt = int(h.sum())
        if cnt > target:
            idx = np.where(h == 1)[0]
            h[idx[rng.randint(0, len(idx), cnt - target)]] = 0
        elif cnt < target:
            idx = np.where(h == 0)[0]
            h[idx[rng.randint(0, len(idx), target - cnt)]] = 1
    return h

def _binary_sa(h_init, n, dx, rng, deadline, T_start=None, T_end=None, n_restarts=1):
    target = n // 2
    best_h = np.zeros(n)
    best_c5 = 1e9
    for r in range(n_restarts):
        if r == 0 and h_init is not None:
            h = (h_init >= 0.5).astype(float)
            cnt = int(h.sum())
            if cnt > target:
                idx = np.where(h == 1)[0]
                h[idx[rng.randint(0, len(idx), cnt - target)]] = 0
            elif cnt < target:
                idx = np.where(h == 0)[0]
                h[idx[rng.randint(0, len(idx), target - cnt)]] = 1
        else:
            h = _make_binary(n, rng, r % 5)
        cur_c5 = _compute_c5(h, dx)
        Ts = T_start if T_start is not None else max(0.002, cur_c5 * 0.06)
        Te = T_end if T_end is not None else max(1e-5, Ts * 0.008)
        T = Ts
        est_steps = max(100, int((deadline - time.time()) * 6000))
        cool = max(1.0 - 1.0 / est_steps, (Te / Ts) ** (1.0 / est_steps))
        while time.time() < deadline:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i], h[j] = 0.0, 1.0
            new_c5 = _compute_c5(h, dx)
            delta = new_c5 - cur_c5
            if delta <= 0 or (T > Te and rng.random() < np.exp(-delta / T)):

                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
            else:
                h[i], h[j] = 1.0, 0.0
            T = max(Te, T * cool)
        if time.time() >= deadline:
            break
    return best_h, best_c5

def _local_refine(h, n, dx, rng, deadline):

    h = h.copy()
    for scale in [0.015, 0.008, 0.004, 0.002, 0.001, 0.0005]:

            else:

        if not improved and scale < 0.003:

    return _repair(h, n), best_c5
