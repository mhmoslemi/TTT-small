        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        d = target - s
        if abs(d) < 1e-14:
            break
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-14:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
    return h

def smax_obj(h, n, alpha, topk, fw):
    corr = compute_corr(h)
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    val = m + np.log(np.sum(np.exp(s))) / alpha
    if topk > 1:
        sc = np.sort(corr)[::-1][:topk]
        val += fw * (np.max(sc) - np.min(sc)) ** 2  # penalize spread in top-k
    return val

def slsqp_func(h, n, alpha, topk, fw, maxiter):
    def obj(x):
        return smax_obj(x, n, alpha, topk, fw)
    def con_eq(x):
        return x.sum() - n / 2.0

                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return repair(res.x, n)
    except:
        return h
