import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def _corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, n):
    dx = 2.0 / n
    return float(np.max(_corr(h, n))) * dx

def _c5_array(h, n):
    dx = 2.0 / n
    return _corr(h, n) * dx

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            for idx in np.argsort(1.0 - h):
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
                if abs(diff) < 1e-14: break
        else:
            for idx in np.argsort(h):
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
                if abs(diff) < 1e-14: break
    return h

def _project(h, n):
    """Project onto {0<=x<=1, sum=n/2}."""
    h = np.clip(h, 0.0, 1.0)
    diff = n/2.0 - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        for idx in np.argsort(1.0 - h):
            a = min(diff, 1.0 - h[idx])
            h[idx] += a; diff -= a
            if abs(diff) < 1e-12: break
    else:
        for idx in np.argsort(h):
            a = min(-diff, h[idx])
            h[idx] -= a; diff += a
            if abs(diff) < 1e-12: break
    return h

def _lse_obj_and_grad(h, n, alpha):
    c = _c5_array(h, n)
    m = np.max(c)
    s = np.clip(alpha*(c-m), -400, 400)
    e = np.exp(s)
    w = e / e.sum()
    obj = m + np.log(e.sum())/alpha
    dx = 2.0/n
    grad = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-14: continue
        lag = k-(n-1)
        i0 = max(0, -lag)
        i1 = min(n, n-lag)
        for i in range(i0, i1):
            jp = i+lag
            grad[jp] += w[k]*(1.0-h[i])
            grad[i] -= w[k]*h[jp]
    return obj, grad*dx

def _pd_step(h, n, alpha, lr, budget_end):
    """One batch of projected gradient steps."""
    for _ in range(200):
        if time.time() > budget_end: break
        obj, g = _lse_obj_and_grad(h, n, alpha)
        h_new = _project(h - lr*g, n)
        obj_new, _ = _lse_obj_and_grad(h_new, n, alpha)
        # Backtracking if needed
        if obj_new > obj + 1e-10:
            h_new = _project(h - lr*0.5*g, n)
        h = h_new
    return h

def _pair_sweep(h, n, budget_end, n_points_grid=7, refine=True):
    """Cyclic pair optimization with grid search + refinement."""
    best_c5 = _c5(h, n)
    improved_any = False
    for sweep in range(200):
        if time.time() > budget_end: break
        improved = False
        for i in range(n-1):
            if time.time() > budget_end: break
            hi = h[i]
            for j in range(i+1, n):
                if time.time() > budget_end: break
                hj = h[j]
                s = hi + hj
                lo = max(0.0, s-1.0)
                hi_bound = min(1.0, s)
                if hi_bound - lo < 1e-12: continue
                
                # Grid search
                if n_points_grid <= 3:
                    candidates = np.array([hi, lo, hi_bound])
                else:
                    candidates = np.linspace(lo, hi_bound, n_points_grid)
                    candidates = np.unique(np.concatenate([candidates, [hi]]))
                
                best_x = hi
                best_val = best_c5
                for x in candidates:
                    h[i] = x
                    h[j] = s - x
                    c = _c5(h, n)
                    if c < best_val:
                        best_val = c
                        best_x = x
                h[i] = best_x
                h[j] = s - best_x
                
                if best_val < best_c5 - 1e-15:
                    best_c5 = best_val
                    improved = True
                    # Refine with finer grid
                    if refine:
                        eps = (hi_bound - lo) / 50
                        candidates2 = np.linspace(max(lo, best_x-eps), min(hi_bound, best_x+eps), 5)
                        for x in candidates2:
                            h[i] = x
                            h[j] = s - x
                            c = _c5(h, n)
                            if c < best_val:
                                best_val = c
                                best_x = x
                        h[i] = best_x
                        h[j] = s - best_x
                        best_c5 = best_val
        if not improved:
            break
        improved_any = True
    return h, best_c5

def _fine_search(h, n, budget_end, rng):
    """Very fine coordinate search."""
    best_c5 = _c5(h, n)
    deltas = np.array([0.0005, 0.001, 0.003, 0.01, -0.0005, -0.001, -0.003, -0.01])
    for _pass in range(500):
        if time.time() > budget_end: break
        improved = False
        for i in range(n-1):
            if time.time() > budget_end: break
            for j in range(i+1, n):
                if time.time() > budget_end: break
                for d in deltas:
                    ni, nj = h[i]+d, h[j]-d
                    if ni < 0 or ni > 1 or nj < 0 or nj > 1: continue

                        improved = True
                    else:
                        h[i], h[j] = ni-d, nj+d
                h[i] = h[i]  # keep current
        if not improved: break
    return h, best_c5

            for idx in np.argsort(h):
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
                if abs(diff) < 1e-14: break
    return h

def _project(h, n):
    h = np.clip(h, 0.0, 1.0)
    diff = n / 2.0 - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        for idx in np.argsort(1.0 - h):
            a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            if abs(diff) < 1e-12: break
    else:
        for idx in np.argsort(h):
            a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-12: break
    return h

def _lse_grad(h, n, alpha):
    c = _c5_arr(h, n)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -350, 350)
    e = np.exp(s)
    w = e / e.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        i0 = max(0, -lag)
        i1 = min(n, n - lag)
        for i in range(i0, i1):
            jp = i + lag
            grad[jp] += wk * (1.0 - h[i])
            grad[i] -= wk * h[jp]
    return grad * dx

def _pd_phase(h, n, alpha, lr, budget_end):
    for _ in range(3000):
        if time.time() > budget_end:
            break
        g = _lse_grad(h, n, alpha)
        h_new = _project(h - lr * g, n)
        h = h_new
    return h

def _pair_sweep(h, n, budget_end, grid_n=9, refine_pts=9, max_sweeps=500):
    best_c5 = _c5(h, n)
    for sweep in range(max_sweeps):
        if time.time() > budget_end:
            break
        improved = False
        for i in range(n - 1):
            if time.time() > budget_end:
                break
            hi = h[i]
            for j in range(i + 1, n):
                if time.time() > budget_end:
                    break
                hj = h[j]
                s = hi + hj
                lo = max(0.0, s - 1.0)
                hb = min(1.0, s)
                if hb - lo < 1e-12:
                    continue
                cands = np.linspace(lo, hb, grid_n)
                if not np.isin(hi, cands, atol=1e-10):
                    cands = np.append(cands, hi)
                bx, bv = hi, best_c5
                for x in cands:
                    h[i] = x; h[j] = s - x
                    c = _c5(h, n)
                    if c < bv:
                        bv = c; bx = x
                h[i] = bx; h[j] = s - bx
                if bv < best_c5 - 1e-15:
                    best_c5 = bv; improved = True
                    eps = (hb - lo) / 80.0
                    lo2 = max(lo, bx - eps)
                    hi2 = min(hb, bx + eps)
                    cands2 = np.linspace(lo2, hi2, refine_pts)
                    for x in cands2:
                        h[i] = x; h[j] = s - x
                        c = _c5(h, n)
                        if c < bv:
                            bv = c; bx = x
                    h[i] = bx; h[j] = s - bx
                    best_c5 = bv
        if not improved:
            break
    return h, best_c5

def _micro_search(h, n, budget_end, rng):
    best_c5 = _c5(h, n)
    deltas = np.array([0.0005, 0.001, 0.002, 0.005, 0.01,
                       -0.0005, -0.001, -0.002, -0.005, -0.01])
    for _ in range(2000):
        if time.time() > budget_end:
            break
        imp = False
        for i in range(n - 1):
            if time.time() > budget_end:
                break
            for j in range(i + 1, n):
                if time.time() > budget_end:
                    break
                for d in deltas:
                    ni, nj = h[i] + d, h[j] - d
                    if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                        continue
                    h[i], h[j] = ni, nj
                    c = _c5(h, n)
                    if c < best_c5 - 1e-15:
                        best_c5 = c; imp = True
                    else:
                        h[i], h[j] = ni - d, nj + d
        if not imp:
            break
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    budget_end = t0 + budget_s

    n = len(initial_h_values)
    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _c5(best_h, n)

    # Phase 1: Projected gradient with LSE (20% budget)
    p1_end = t0 + budget_s * 0.20
    for alpha, lr in [(2000, 0.002), (5000, 0.001), (10000, 0.0005), (30000, 0.0002)]:
        if time.time() > p1_end:
            break
        h = _pd_phase(h, n, alpha, lr, p1_end)
        c5 = _c5(h, n)
        if c5 < best_c5:
            best_c5 = c5; best_h = h.copy()
        if time.time() < p1_end - 20:
            h = _repair(best_h + 0.03 * rng.randn(n), n)

    # Phase 2: Pair optimization main (50% budget)
    p2_end = t0 + budget_s * 0.70
    h = best_h.copy()
    h, c5v = _pair_sweep(h, n, min(p2_end, time.time() + budget_s * 0.30))
    if c5v < best_c5:
        best_c5 = c5v; best_h = h.copy()

    # Continue pair sweeps with shake-and-settle
    while time.time() < p2_end:
        h2 = _repair(best_h + 0.008 * rng.randn(n), n)
        h2, c5_2 = _pair_sweep(h2, n, min(p2_end, time.time() + 80))
        if c5_2 < best_c5:
            best_c5 = c5_2; best_h = h2.copy()
        # Also try without perturbation (refine)
        h3 = best_h.copy()
        h3, c5_3 = _pair_sweep(h3, n, min(p2_end, time.time() + 30))
        if c5_3 < best_c5:
            best_c5 = c5_3; best_h = h3.copy()

    # Phase 3: Micro search (15% budget)
    p3_end = t0 + budget_s * 0.87
    best_h, best_c5 = _micro_search(best_h, n, p3_end, rng)

    # Phase 4: Ultra-fine random pair moves (8% budget)
    p4_end = budget_end - 8
    h = best_h.copy()
    c5c = _c5(h, n)
    tiny = np.array([0.0002, 0.0004, 0.0008, 0.0015,
                     -0.0002, -0.0004, -0.0008, -0.0015])
    while time.time() < p4_end:
        i = rng.randint(n - 1)
        j = rng.randint(i + 1, n)
        d = tiny[rng.randint(len(tiny))]
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c = _c5(h, n)
            if c < c5c:
                c5c = c
                if c < best_c5:
                    best_c5 = c; best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d

    # Phase 5: One final pair polish
    if time.time() < budget_end - 15:
        h5 = best_h.copy()
        h5, c5f = _pair_sweep(h5, n, budget_end - 8)
        if c5f < best_c5:
            best_c5 = c5f; best_h = h5

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, n)
    return (best_h, best_c5, n)
