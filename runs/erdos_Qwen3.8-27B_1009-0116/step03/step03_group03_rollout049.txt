    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sum_con(x, target):
    return np.sum(x) - target

def _slsqp(h, n, target, bounds, alpha, maxiter, t0, deadline):
    if time.time() > deadline:
        return h
    try:
        res = minimize(
            _smooth_obj, h, args=(alpha,),
            method='SLSQP', bounds=bounds,
            constraints=[{'type': 'eq', 'fun': _sum_con, 'args': (target,)}],
            options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        return _repair(res.x, n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _c5(h, dx)

    # Phase 1: Exact coordinate descent (primary)
    dl1 = min(t0 + budget_s * 0.45, deadline)
    h = _coord_descent(h, n, max_sweeps=50000, t0=t0, deadline=dl1)
    h = _repair(h, n)
    c5 = _c5(h, dx)
    if c5 < best_c5:
        best_c5, best_h = c5, h.copy()

    # Phase 2: SLSQP polish with increasing alpha
    for alpha in [2000, 10000, 50000]:
        if time.time() > deadline - 60:
            break
        dl2 = min(time.time() + 120, deadline)
        h2 = _slsqp(best_h, n, target, bounds, alpha, 500, t0, dl2)
        c5 = _c5(h2, dx)
        if c5 < best_c5:
            best_c5, best_h = c5, h2

    # Phase 3: More coordinate descent from best
    if time.time() < deadline - 80:
        dl3 = min(time.time() + 200, deadline)
        h = _coord_descent(best_h, n, max_sweeps=50000, t0=t0, deadline=dl3)
        h = _repair(h, n)
        c5 = _c5(h, dx)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Phase 4: Multi-restart with perturbations
    for _r in range(8):
        if time.time() > deadline - 30:
            break
        h_p = best_h + 0.015 * rng.randn(n)
        h_p = _repair(h_p, n)
        dl4 = min(time.time() + 60, deadline)
        h_p = _coord_descent(h_p, n, max_sweeps=20000, t0=t0, deadline=dl4)
        h_p = _repair(h_p, n)
        c5 = _c5(h_p, dx)
        if c5 < best_c5:
            best_c5, best_h = c5, h_p.copy()
        # SLSQP from this point
        if time.time() < deadline - 30:
            h_s = _slsqp(h_p, n, target, bounds, 30000, 300, t0, deadline)
            c5 = _c5(h_s, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, h_s.copy()

    # Phase 5: Fine coordinate descent polish
    if time.time() < deadline - 10:
        dl5 = deadline
        h = _coord_descent(best_h, n, max_sweeps=100000, t0=t0, deadline=dl5)
        h = _repair(h, n)
        c5 = _c5(h, dx)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Phase 6: Tiny pair adjustments
    if time.time() < deadline - 2:
        h = best_h.copy()
        c_cur = _c5(h, dx)
        c_vec = _corr_vec(h)
        for _it in range(200000):
            if time.time() > deadline:
                break
            i = rng.randint(n)
            j = (i + 1 + rng.randint(n - 1)) % n
            for d in (0.0005, 0.002, -0.0005, -0.002):
                ni, nj = h[i] + d, h[j] - d
                if 0 <= ni <= 1 and 0 <= nj <= 1:
                    h[i], h[j] = ni, nj
                    h = _repair(h, n)
                    c_new = _c5(h, dx)
                    if c_new < c_cur:
                        c_cur = c_new
                        best_c5, best_h = c_cur, h.copy()
                        break
                    else:
                        h[i], h[j] = ni - d, nj + d
                        h = _repair(h, n)

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
