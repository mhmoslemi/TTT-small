def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)


def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(s))) / alpha


def _sym_to_full(v, half):
    h = np.empty(2 * half)
    h[:half] = v
    h[half:] = 1.0 - v
    return h


def _obj_sym(v, alpha):
    h = _sym_to_full(v, len(v))
    return _smooth_max(_corr(h), alpha)


def _obj_full(x, alpha):
    return _smooth_max(_corr(x), alpha)


def _eq_sum(x, target):
    return x.sum() - target


def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-12:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h
