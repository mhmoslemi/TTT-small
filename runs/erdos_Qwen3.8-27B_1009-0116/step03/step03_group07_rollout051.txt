def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(5000):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        d = target - s
        if abs(d) < 1e-14:
            break
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-14:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
    return h

def grad_lag(h, n, lag):
    grad = np.zeros(n)
    if lag == 0:
        return 1.0 - 2.0 * h
    elif lag > 0:
        i1 = np.arange(n - lag)
        i2 = i1 + lag
        grad[i1] += (1.0 - h[i2])
        grad[i2] += (-h[i1])

        i1 = np.arange(l, n)
        i2 = i1 - l
        grad[i1] += (1.0 - h[i2])
        grad[i2] += (-h[i1])
