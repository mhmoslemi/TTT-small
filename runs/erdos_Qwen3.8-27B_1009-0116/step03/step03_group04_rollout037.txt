                    diff -= a

                        best_c5, best_h = c5_cur, h.copy()

    # Phase 3: Coordinate descent on 35 vars (coarse then fine)
    if time.time() - t0 < budget_s - 50:
        x = best_h[:half].copy()
        c5_cur = _c5_of(_full_sym(x), dx)
        deltas_coarse = [0.01, 0.005, 0.002, 0.001, -0.01, -0.005, -0.002, -0.001]
        deltas_fine = [2e-4, 1e-4, 5e-5, -2e-4, -1e-4, -5e-5]
        for _pass in range(150):
            if time.time() - t0 > budget_s - 50:
                break
            dset = deltas_coarse if _pass < 40 else deltas_fine
            imp = False
            for j in range(half):
                if time.time() - t0 > budget_s - 50:
                    break
                bv, bc = x[j], c5_cur
                for d in dset:
                    v = x[j] + d
                    if 0.0 <= v <= 1.0:
                        x[j] = v
                        c = _c5_of(_full_sym(x), dx)
                        if c < bc:
                            bc, bv = c, v
                x[j] = bv
                if bc < c5_cur:
                    c5_cur = bc
                    imp = True
            if c5_cur < best_c5:
                best_c5, best_h = c5_cur, _full_sym(x).copy()
            if not imp:
                break

    # Phase 4: Pair moves on 35 vars
    if time.time() - t0 < budget_s - 15:
        x = best_h[:half].copy()
        c5_cur = _c5_of(_full_sym(x), dx)
        pair_deltas = [1e-4, 3e-4, 1e-3, -1e-4, -3e-4, -1e-3]
        for _pass in range(300):
            if time.time() - t0 > budget_s - 15:
                break
            imp = False
            for i in range(half):
                if time.time() - t0 > budget_s - 15:
                    break
                for j in range(i + 1, half):
                    hi, hj = x[i], x[j]
                    for d in pair_deltas:
                        ni, nj = hi + d, hj - d
                        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                            x[i], x[j] = ni, nj
                            c = _c5_of(_full_sym(x), dx)
                            if c < c5_cur - 1e-16:
                                c5_cur = c
                                imp = True
                            else:
                                x[i], x[j] = hi, hj
                    x[i], x[j] = hi, hj
            if c5_cur < best_c5:
                best_c5, best_h = c5_cur, _full_sym(x).copy()
            if not imp:
                break

    # Phase 5: Ultra-fine random pair search on 70-dim
    if time.time() - t0 < budget_s - 5:
        h = best_h.copy()
        c5_cur = _c5_of(h, dx)
        prefs = [5e-5, 1e-4, 2e-4, -5e-5, -1e-4, -2e-4]
        while time.time() - t0 < budget_s - 5:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = prefs[rng.randint(len(prefs))]
            ni, nj = h[i] + d, h[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h[i], h[j] = ni, nj
                c = _c5_of(h, dx)
                if c < c5_cur - 1e-16:
                    c5_cur = c
                    if c5_cur < best_c5:
                        best_c5, best_h = c5_cur, h.copy()
                        h = best_h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d

    # Phase 6: Final L-BFGS-B polish at very high alpha
    if time.time() - t0 < budget_s - 2:
        try:
            x0 = best_h[:half].copy()
            res = _lbfgs_sym(x0, bounds35, 2000000, 5000)
            x_new = np.clip(res.x, 0, 1)
            h_new = _full_sym(x_new)
            c_new = _c5_of(h_new, dx)
            if c_new < best_c5:
                best_c5, best_h = c_new, h_new.copy()
        except Exception:
            pass

    best_h = _repair(best_h, n)
    best_c5 = _c5_of(best_h, dx)
    return (best_h, best_c5, n)
