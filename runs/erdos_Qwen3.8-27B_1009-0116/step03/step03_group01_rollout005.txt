import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _cons_fun(x):
    return float(np.sum(x) - x.shape[0] * 0.5)

def _cons_jac(x):
    return np.ones(len(x))

def _obj_lse(h, alpha):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _obj_pnorm(h, p):
    c = _corr(h)
    m = c.max()
    if m <= 0:
        return 0.0
    ratio = np.clip(c / m, 0, 1)
    return m * float(np.sum(ratio ** p)) ** (1.0 / p)

def _obj_topk(h, k):
    c = _corr(h)
    idx = np.argpartition(c, -k)[-k:]
    return float(c[idx].mean())

def _do_slsqp(h, n, obj_func, args, iters, ftol):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(obj_func, h, args=args, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': iters, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _subgrad_opt(h, n, iters, step_init, decay, rng):
    h = h.copy().astype(float)
    best = h.copy()
    best_c5 = _eval(h)
    step = step_init
    for it in range(iters):
        c = _corr(h)
        top = np.argsort(c)[-3:]
        grad = np.zeros(n)
        for k in top:
            L = k - (n - 1)
            w = 1.0 / 3.0
            if L == 0:
                grad += w * (1.0 - 2.0 * h)
            elif L > 0:
                grad[:n - L] += w * (1.0 - h[L:])
                grad[L:] -= w * h[:n - L]
            else:
                L2 = -L
                grad[L2:] += w * (1.0 - h[:n - L2])
                grad[:n - L2] -= w * h[L2:]
        h_t = _repair(h - step * grad, n)
        c5t = _eval(h_t)
        if c5t < best_c5:
            best_c5 = c5t
            best = h_t.copy()
        if c5t <= _eval(h):
            h = h_t
        else:
            step *= 0.9
        step = max(step * decay, 0.005)
    return best, best_c5

def _random_pd(h, n, iters, scale, rng):
    h = h.copy().astype(float)
    c5c = _eval(h)
    no_impr = 0
    for _ in range(iters):
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue

            no_impr += 1
            if no_impr > 200:
                scale *= 0.7
                no_impr = 0
                if scale < 1e-7:
                    break
    return h, c5c

def _anneal(h, n, iters, T0, T1, rng):

    c5c = _eval(h)
    best = h.copy()
    best_c5 = c5c

    ratio = (T1 / T0) ** (1.0 / max(iters, 1))

        delta = c5n - c5c

            c5c = c5n
            if c5c < best_c5:
                best_c5 = c5c
                best = h.copy()
        else:
            h[i], h[j] = h[i] - d, h[j] + d
        T *= ratio
    return best, best_c5

def _resample(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    pos = x_new * n_old / 2.0
    i0 = np.floor(pos).astype(int) % n_old
    i1 = (i0 + 1) % n_old
    frac = pos - np.floor(pos)
    h_new = (1 - frac) * h_old[i0] + frac * h_old[i1]
    return _repair(h_new, n_new)

def _block_opt(h, n, rng, iters):
    """Optimize small blocks of coordinates jointly."""
    h = h.copy().astype(float)
    c5c = _eval(h)
    for _ in range(iters):
        bsz = rng.choice([3, 4, 5])
        idx = rng.choice(n, bsz, replace=False)
        vals = h[idx].copy()
        fixes = [v for v in vals if v < 1e-10 or v > 1 - 1e-10]
        # Only optimize if at least 2 are interior
        interior = [v for v in vals if 1e-10 < v < 1 - 1e-10]
        if len(interior) < 2:
            continue
        # Small local search on the block
        b_scale = 0.02
        for _b in range(30):
            bi = rng.randint(bsz)
            bj = rng.randint(bsz)
            if bi == bj:
                continue
            d = b_scale * rng.randn()
            nv_i = vals[bi] + d
            nv_j = vals[bj] - d
            if nv_i < 0 or nv_i > 1 or nv_j < 0 or nv_j > 1:
                continue
            vals[bi], vals[bj] = nv_i, nv_j
            h[:n] = h  # no-op
            h[idx] = vals
            c5n = _eval(h)
            if c5n < c5c:
                c5c = c5n
            else:
                h[idx] = vals
                vals[bi] -= d
                vals[bj] += d
                h[idx] = vals
        h[idx] = vals
    return h, c5c
