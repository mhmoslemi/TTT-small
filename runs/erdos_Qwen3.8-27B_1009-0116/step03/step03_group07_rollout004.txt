    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(5000):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        d = target - s
        if abs(d) < 1e-14:
            break
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-14:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
    return h

def slsqp_soft(h, n, alpha, maxiter):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        s = alpha * (corr - m)
        s = np.clip(s, -500, 500)
        return m + np.log(np.sum(np.exp(s))) / alpha
    def con_eq(x):
        return x.sum() - target
    cons = [{'type': 'eq', 'fun': con_eq}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return repair(res.x, n)
    except Exception:
        return repair(h, n)

def grad_cd(h, n, n_rounds, t_end, rng):
    """Gradient-informed coordinate descent with pair swaps."""
    h = h.copy()
    target = n / 2.0
    best = compute_c5(h, n)
    for r in range(n_rounds):
        if time.time() > t_end:
            break
        improved = False
        # Compute active lag and descent directions
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_val = np.max(corr)
        # Find all near-active lags (within 1e-6 of max)
        active = np.where(corr > max_val - 1e-6)[0]
        # Compute descent direction d[i] = -dC5/dh[i] for active lags
        d = np.zeros(n)
        for k_idx in active:
            lag = k_idx - (n - 1)
            for i in range(n):
                c = 0.0
                im = i - lag
                if 0 <= im < n:
                    c += (1.0 - h[im])
                ip = i + lag
                if 0 <= ip < n:
                    c -= h[ip]
                d[i] += -c
        d -= np.mean(d)  # project to sum-zero
        # Systematic sweep: try pairs in gradient order
        order = np.argsort(-d)  # indices with most negative gradient first (want to increase)
        order_neg = np.argsort(d)  # indices with most positive gradient first (want to decrease)
        n_try = min(30, n // 3)
        for a_idx in order[:n_try]:
            if time.time() > t_end:
                break
            for b_idx in order_neg[:n_try]:
                if a_idx == b_idx:
                    continue
                # Try moving mass from b_idx to a_idx
                max_step = min(1.0 - h[a_idx], h[b_idx])
                if max_step < 1e-10:
                    continue
                for frac in [1.0, 0.5, 0.25, 0.1]:
                    step = max_step * frac
                    h[a_idx] += step
                    h[b_idx] -= step
                    c5 = compute_c5(h, n)
                    if c5 < best - 1e-14:
                        best = c5
                        improved = True
                        break
                    h[a_idx] -= step
                    h[b_idx] += step
                if improved:
                    break
            if improved:
                break
        # Random pairs
        if not improved:
            for _ in range(200):
                if time.time() > t_end:
                    break
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                max_step = min(1.0 - h[i], h[j])
                if max_step < 1e-10:
                    continue
                for frac in [0.3, 0.15, 0.08, 0.04, 0.02, 0.01]:
                    step = max_step * frac
                    h[i] += step
                    h[j] -= step
                    c5 = compute_c5(h, n)
                    if c5 < best - 1e-14:
                        best = c5
                        improved = True
                        break
                    h[i] -= step
                    h[j] += step
                if not improved:
                    # Try reverse direction
                    max_step2 = min(1.0 - h[j], h[i])
                    if max_step2 > 1e-10:
                        for frac in [0.15, 0.08, 0.04, 0.02]:
                            step = max_step2 * frac
                            h[j] += step
                            h[i] -= step
                            c5 = compute_c5(h, n)
                            if c5 < best - 1e-14:
                                best = c5
                                improved = True
                                break
                            h[j] -= step
                            h[i] += step
        if not improved and r > 100:
            break
    return h, best

def reheat_cd(h, n, t_end, rng, n_rounds=200):
    """CD with periodic reheating to escape local minima."""
    best = compute_c5(h, n)
    best_h = h.copy()
    cur = h.copy()
    for phase in range(6):
        phase_end = min(t_end, t_end + (budget_ratio - phase * 0.01) * 100)
        # Reheat: add noise
        noise_scale = 0.05 * (1 - phase * 0.12)
        cur = cur + noise_scale * rng.randn(n)
        cur = repair(cur, n)
        c5_cur = compute_c5(cur, n)
        if c5_cur < best - 1e-14:
            best = c5_cur
            best_h = cur.copy()
        cur, c5_cd = grad_cd(cur, n, n_rounds, t_end, rng)
        if c5_cd < best - 1e-14:
            best = c5_cd
            best_h = cur.copy()
        else:
            cur = best_h.copy()
    return best_h, best

def make_start(n, seed_val, kind):
    rng = np.random.RandomState(seed_val)
    if kind == 0:
        h = 0.5 + 0.35 * rng.randn(n)
    elif kind == 1:
        h = np.where(np.arange(n) < n // 2, 1.0, 0.0)
        h = 0.7 * h + 0.3 * rng.uniform(0, 1, n)
    elif kind == 2:
        f1, f2 = rng.randint(2, 9), rng.randint(3, 15)
        p1, p2 = rng.uniform(0, 2 * np.pi), rng.uniform(0, 2 * np.pi)
        x = np.arange(n) / n
        h = 0.5 + 0.25 * np.sin(2 * np.pi * f1 * x + p1) + 0.15 * np.sin(2 * np.pi * f2 * x + p2)
    elif kind == 3:
        block = max(2, n // 6)
        h = np.zeros(n)
        for i in range(0, n, 2 * block):
            h[i:i + block] = 1.0
        h = 0.6 * h + 0.4 * rng.uniform(0, 1, n)
    elif kind == 4:
        x = np.arange(n) / n
        h = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x + 0.5)
    elif kind == 5:
        # Blocky: alternating high/low
        block = max(3, n // 10)
        h = np.zeros(n)
        for i in range(0, n, 2 * block):
            h[i:i + block] = 1.0
        h = h + rng.randn(n) * 0.1
    else:
        h = rng.uniform(0, 1, n)
    return repair(h, n)

budget_ratio = 1.0

def run(seed=42, budget_s=1100, **kwargs):
    global budget_ratio
    t0 = time.time()
    rng = np.random.RandomState(seed + 12345)
    best_h = None
    best_c5 = np.inf
    best_n = 70

    init_h = None
    init_n = 70
    try:
        init_h = np.array(initial_h_values, dtype=float).copy()
        init_n = len(init_h)
    except NameError:
        pass

    if init_h is not None:
        init_h = repair(init_h, init_n)
        best_h = init_h.copy()
        best_c5 = compute_c5(init_h, init_n)
        best_n = init_n
    else:
        best_h = repair(np.full(70, 0.5), 70)
        best_c5 = compute_c5(best_h, best_n)

    n_p = best_n
    h_work = best_h.copy()

    # Phase 1: SLSQP with a few alpha values
    elapsed_check = budget_s * 0.20
    for alpha in [3000, 10000, 30000, 80000]:
        if time.time() - t0 > elapsed_check:
            break
        h_new = slsqp_soft(h_work.copy(), n_p, alpha, 600)
        c5_new = compute_c5(h_new, n_p)
        if c5_new < best_c5 - 1e-14:
            best_h = h_new.copy()
            best_c5 = c5_new
            h_work = h_new

    # Phase 2: Intensive gradient-informed CD
    if time.time() - t0 < budget_s * 0.55:
        h_cd, c5_cd = grad_cd(h_work, n_p, 500, t0 + budget_s * 0.55, rng)
        if c5_cd < best_c5 - 1e-14:
            best_h = h_cd.copy()
            best_c5 = c5_cd
            h_work = h_cd

    # Phase 3: Reheat CD to escape local minima
    if time.time() - t0 < budget_s * 0.78:
        h_rh, c5_rh = reheat_cd(h_work, n_p, t0 + budget_s * 0.78, rng, 200)
        if c5_rh < best_c5 - 1e-14:
            best_h = h_rh.copy()
            best_c5 = c5_rh

    # Phase 4: Final SLSQP + CD polish on best
    if best_h is not None and time.time() - t0 < budget_s * 0.90:
        h_fin = slsqp_soft(best_h.copy(), best_n, 100000, 500)
        c5_fin = compute_c5(h_fin, best_n)
        if c5_fin < best_c5 - 1e-14:
            best_h = h_fin.copy()
            best_c5 = c5_fin
        h_cd2, c5_cd2 = grad_cd(best_h, best_n, 300, t0 + budget_s * 0.90, rng)
        if c5_cd2 < best_c5 - 1e-14:
            best_h = h_cd2.copy()
            best_c5 = c5_cd2

    # Phase 5: Try alternative n values
    if time.time() - t0 < budget_s * 0.96:
        for n_alt in [64, 80, 90]:
            if time.time() - t0 > budget_s * 0.96:
                break
            if n_alt == best_n:
                continue
            for kk in range(3):
                if time.time() - t0 > budget_s * 0.96:
                    break
                h_alt = make_start(n_alt, seed + n_alt * 7 + kk, kk % 6)
                h_alt = slsqp_soft(h_alt, n_alt, 15000, 400)
                c5_alt = compute_c5(h_alt, n_alt)
                if c5_alt < best_c5 - 1e-14:
                    best_h = h_alt.copy()
                    best_c5 = c5_alt
                    best_n = n_alt
                h_alt2, c5_alt2 = grad_cd(h_alt, n_alt, 100, t0 + budget_s * 0.96 - 5, rng)
                if c5_alt2 < best_c5 - 1e-14:
                    best_h = h_alt2.copy()
                    best_c5 = c5_alt2
                    best_n = n_alt

    # Phase 6: Final CD polish
    if best_h is not None and time.time() - t0 < budget_s * 0.995:
        h_p, c5_p = grad_cd(best_h, best_n, 200, t0 + budget_s * 0.995, rng)
        if c5_p < best_c5 - 1e-14:
            best_h = h_p.copy()
            best_c5 = c5_p

    if best_h is None:
        best_n = 70
        best_h = repair(np.full(best_n, 0.5), best_n)
        best_c5 = compute_c5(best_h, best_n)

    best_h = repair(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return (best_h, best_c5, best_n)
