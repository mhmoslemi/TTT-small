- Forward: j = i+k-(n-1), valid if 0 ≤ j ≤ n-1
- Backward: j' = i-k+(n-1), valid if 0 ≤ j' ≤ n-1

Note that for k = n-1 (zero lag), both forward and backward give j = i, j' = i.

For k ≠ n-1, forward and backward give different partners.

Let me just implement it straightforwardly for n=70 (which is small enough):

```python
def _smooth_grad(h, n, alpha):
    c = _corr(h)
    m = float(np.max(c))
    w = np.exp(alpha * (c - m))
    w /= w.sum()
    g = np.zeros(n)
    nm1 = n - 1
    for k in range(2*n - 1):
        if w[k] < 1e-10:
            continue
        wk = w[k]
        # Forward: j = i + k - nm1
        for i in range(n):
            j = i + k - nm1
            if 0 <= j < n:
                g[i] += wk * (1.0 - h[j])
            j2 = i - k + nm1
            if 0 <= j2 < n:
                g[i] -= wk * h[j2]
    return g
```

This is O(n * 2n) = O(n²) per iteration for the inner loops, which for n=70 is about 10K operations. Very fast.

    hi = float(v.max()) + 2.0

            lo = mid
        else:

    m = float(np.max(c))
    w = np.exp(np.clip(alpha * (c - m), -500, 500))
    s = float(w.sum())
    if s < 1e-300:
        return np.zeros(n)
    w /= s
    g = np.zeros(n)
    nm1 = n - 1
    i_idx = np.arange(n)

        if w[k] < 1e-12:
            continue
        jf = i_idx + k - nm1
        mf = (jf >= 0) & (jf < n)

        jb = i_idx - k + nm1
        mb = (jb >= 0) & (jb < n)
        g[mb] -= w[k] * h[jb[mb]]
    return g

def _eq_con(x, target):
    return float(np.sum(x)) - target

    res = minimize(_smooth_max_obj, x0, args=(alpha,), method='SLSQP',
                   bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': _eq_con, 'args': (target,)}],
                   options={'maxiter': maxiter, 'ftol': 1e-18})
    return res.x

    step = 0.05

            c5_new = _c5(h_new, dx)
            if c5_new < best - 1e-16:
                h = h_new
                best = c5_new
                step = min(step * 1.2, 0.5)
                break
            step *= 0.5
        if it % 200 == 199:
            step = max(step, 0.01)
    return h, best
