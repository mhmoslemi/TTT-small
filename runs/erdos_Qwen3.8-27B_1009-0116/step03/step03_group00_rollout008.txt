    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if abs(s - target) < 1e-13:
        return h
    # Water-filling projection
    if s < target:
        diff = target - s
        order = np.argsort(1.0 - h)
        for idx in order:
            add = min(diff, 1.0 - h[idx])
            h[idx] += add
            diff -= add
            if diff < 1e-14:
                break
    else:
        diff = s - target
        order = np.argsort(h)
        for idx in order:
            sub = min(diff, h[idx])
            h[idx] -= sub
            diff -= sub
            if diff < 1e-14:

    return h

    return np.correlate(h, 1.0 - h, mode='full') * dx

    return float(np.max(_c5_vec(h, n)))

def _smooth_grad(h, n, alpha):

    c = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(c)
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    e = np.exp(s)
    total = np.sum(e)
    if total < 1e-300:
        total = 1e-300
    w = e / total
    g = np.zeros(n)
    for k in range(len(c)):

            continue

        i_min = max(0, -L)

        g[i + L] += w[k] * (1.0 - h[i])
        g[i] -= w[k] * h[i + L]
    return g * dx

    h = h.copy()

        idx = np.argsort(c)[::-1][:k_top]
