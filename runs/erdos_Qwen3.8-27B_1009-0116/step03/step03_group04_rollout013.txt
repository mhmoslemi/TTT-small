    if s < 1e-12:

            j = i + m
            if 0 <= j < n:
                g[i] += (1.0 - h[j])
                g[j] -= h[i]
    return g / len(lags)

- In the sum, the term where p=i contributes h[i]*(1-h[i+m]) → changes by δ*(1-h[i+m]) if i+m valid
- The term where p+m=i (so p=i-m) contributes h[i-m]*(1-h[i]) → changes by -δ*h[i-m] if i-m valid

- The term where p=j contributes h[j]*(1-h[j+m]) → changes by -δ*(1-h[j+m]) if j+m valid
- The term where p+m=j (so p=j-m) contributes h[j-m]*(1-h[j]) → changes by +δ*h[j-m] if j-m valid

s[k] = [i+m valid ? (1-h[i+m]) : 0] - [i-m valid ? h[i-m] : 0] - [j+m valid ? (1-h[j+m]) : 0] + [j-m valid ? h[j-m] : 0]

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eq_target(x, target):
    return np.sum(x) - target

def _do_slsqp(x0, bounds, target, alpha, maxiter):
    def _obj(x):
        return _smooth_max(_corr(x), alpha)
    res = minimize(_obj, x0, method='SLSQP', bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': _eq_target, 'args': (target,)}],
                   options={'maxiter': maxiter, 'ftol': 1e-16})
    return res.x

def _pair_delta(h, c, i, j, n):
    n_corr = 2 * n - 1
    lags = np.arange(n_corr) - (n - 1)
    s = np.zeros(n_corr)
    im = i + lags
    imv = (im >= 0) & (im < n)
    s[imv] += (1.0 - h[im[imv]])
    im2 = i - lags
    im2v = (im2 >= 0) & (im2 < n)
    s[im2v] -= h[im2[im2v]]
    jm = j + lags
    jmv = (jm >= 0) & (jm < n)
    s[jmv] -= (1.0 - h[jm[jmv]])
    jm2 = j - lags
    jm2v = (jm2 >= 0) & (jm2 < n)
    s[jm2v] += h[jm2[jm2v]]
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if hi - lo < 1e-12:
        return 0.0, float(np.max(c))
    grid = np.linspace(lo, hi, 64)
    vals = np.max(c[:, None] + s[:, None] * grid, axis=0)
    bi = int(np.argmin(vals))
    r = 0.618033988749895
    a = grid[max(0, bi - 1)]
    b = grid[min(63, bi + 1)]
    for _ in range(40):
        m1 = b - r * (b - a)
        m2 = a + r * (b - a)
        if np.max(c + s * m1) < np.max(c + s * m2):
            b = m2
        else:
            a = m1
    d = 0.5 * (a + b)
    return float(d), float(np.max(c + s * d))

def _active_grad(h, n, K):
    c = _corr(h)
    top = np.argsort(c)[-K:]
    lags_k = top - (n - 1)
    g = np.zeros(n)
    for m in lags_k:
        for ii in range(n):
            jj = ii + m
            if 0 <= jj < n:
                g[ii] += (1.0 - h[jj])
                g[jj] -= h[ii]
    return g / K, float(c.max())
