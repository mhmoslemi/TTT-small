    return np.correlate(h, 1.0 - h, mode='full')

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) < 1e-9:
        return h
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-9:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
            else:
                add = max(diff, h[idx] - 1.0)
            h[idx] += add
            diff -= add
            if abs(diff) < 1e-9:
                break
    return h

def _set_coord(h, n, i, t):
    """Set h[i] = t while maintaining sum = n/2 and bounds [0,1]."""
    target = n / 2.0
    old = h[i]
    diff = old - t  # amount to redistribute to others
    h[i] = t
    # Subtract diff/n from each other element
    mask = np.ones(n, dtype=bool)
    mask[i] = False
    h[mask] -= diff / (n - 1)
    # Clip and fix
    h = np.clip(h, 0, 1)
    # Fix any remaining sum error
    error = target - h.sum()
    if abs(error) > 1e-12:
        # Distribute error
        if error > 0:
            order = np.argsort(1 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if error > 0:
                add = min(error, 1 - h[idx])
            else:
                add = max(error, h[idx] - 1)
            h[idx] += add
            error -= add
            if abs(error) < 1e-12:
                break
    return h

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-9:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                d = min(diff, 1.0 - h[idx]) if diff > 0 else max(diff, h[idx] - 1.0)
                h[idx] += d
                diff -= d
                if abs(diff) < 1e-9:
                    break
    return h

def _set_coord(h, n, i, t):
    target = n / 2.0
    old = h[i]
    if abs(t - old) < 1e-10:
        return h
    h = h.copy()
    h[i] = t
    diff = old - t
    others = np.ones(n, dtype=bool)
    others[i] = False
    h[others] -= diff / (n - 1)
    h = np.clip(h, 0.0, 1.0)
    err = target - h.sum()
    if abs(err) > 1e-10:
        order = np.argsort(1.0 - h) if err > 0 else np.argsort(h)
        for idx in order:
            d = min(err, 1.0 - h[idx]) if err > 0 else max(err, h[idx] - 1.0)
            h[idx] += d
            err -= d
            if abs(err) < 1e-10:
                break
    return h

def _coord_descent(h, n, dx, max_passes=40, n_grid=51):
    t_grid = np.linspace(0.0, 1.0, n_grid)
    best_c5 = _c5(h, dx)
    for p in range(max_passes):
        improved = False
        for i in range(n):
            old = h[i]
            best_t = old
            best_val = best_c5
            for t in t_grid:
                if abs(t - old) < 1e-8:
                    continue
                h2 = _set_coord(h, n, i, t)
                c = _c5(h2, dx)
                if c < best_val - 1e-13:
                    best_val = c
                    best_t = t
            if best_t != old:
                h = _set_coord(h, n, i, best_t)
                best_c5 = best_val
                improved = True
        if not improved:
            break
    return h, best_c5

def _sa_optimize(h, n, dx, t_start, t_end, n_iter, n_move, seed):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = _c5(h, dx)
    cur_c5 = best_c5
    T = t_start
    cool = (t_end / t_start) ** (1.0 / max(n_iter, 1))
    for it in range(n_iter):
        T *= cool
        idxs = rng.choice(n, size=n_move, replace=False)
        h_new = h.copy()
        for i in idxs:
            h_new[i] += rng.uniform(-0.08, 0.08)
        h_new = _repair(h_new, n)
        new_c5 = _c5(h_new, dx)
        delta = new_c5 - cur_c5
        if delta < -1e-14 or (T > 1e-12 and rng.rand() < np.exp(-delta / T)):
            h = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5 - 1e-14:
                best_c5 = new_c5
                best_h = h.copy()
    return best_h, best_c5

def _iterated_ls(h, n, dx, n_rounds, pert, max_cd, seed):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = _c5(h, dx)
    T = 0.015
    for r in range(n_rounds):
        h_p = best_h + rng.uniform(-pert, pert, size=n)
        h_p = _repair(h_p, n)
        h_l, c5_l = _coord_descent(h_p, n, dx, max_passes=max_cd, n_grid=41)
        if c5_l < best_c5 - 1e-14:
            best_c5, best_h = c5_l, h_l
            T = max(T * 0.9, 0.003)
        else:
            delta = c5_l - best_c5
            if T > 1e-10 and rng.rand() < np.exp(-delta / T):
                best_h, best_c5 = h_l, c5_l
            T *= 0.97
        pert = min(pert * 1.04, 0.12)
    return best_h, best_c5

def _grad_step(h, n, lr):
    c = _corr(h)
    m = c.max()
    active = np.where(c > 0.96 * m)[0]
    g = np.zeros(n)
    for mi in active:
        k = mi - (n - 1)
        if k >= 0:
            g[k:] += (1.0 - h[:n - k])
            g[:n - k] -= h[k:]
        else:
            g[:n + k] += (1.0 - h[-k:])
            g[-k:] -= h[:n + k]
    g /= len(active)
    return _repair(h - lr * g, n)

def _pair_polish(h, n, dx, n_trials, seed):
    rng = np.random.RandomState(seed)
    cur_c5 = _c5(h, dx)
    best_c5 = cur_c5
    steps = [0.0005, 0.001, 0.003, 0.008, -0.0005, -0.001, -0.003, -0.008]
    for _ in range(n_trials):
        i = rng.randint(n)
        j = (i + 1 + rng.randint(n - 1)) % n
        for d in steps:
            h[i] += d
            h[j] -= d
            if 0.0 <= h[i] <= 1.0 and 0.0 <= h[j] <= 1.0:
                h_r = _repair(h, n)
                c = _c5(h_r, dx)
                if c < cur_c5 - 1e-14:
                    cur_c5 = c
                    h = h_r
                    if c < best_c5:
                        best_c5 = c
                    break
            h[i] -= d
            h[j] += d
    return h, best_c5
