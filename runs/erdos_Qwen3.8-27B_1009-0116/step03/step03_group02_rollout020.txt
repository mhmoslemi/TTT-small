    return float(np.max(_corr(h)) * dx)

def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

    return _smooth_max(_corr(x), alpha)

    return float(np.sum(x)) - target

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break

    G = np.zeros((n, 2 * n - 1))

    idx2 = p + kk
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

    corr = _corr(h)
    G = _compute_G(h, n)
    cur_max = float(np.max(corr))
    order = np.random.permutation(n)
    n_improved = 0

        if time.time() > time_limit:

        gi = G[i]

            d_max = min(1.0 - hi, hj)

            hi_b = min(d_max, ds[min(n_coarse - 1, bi + 1)])

                if float(np.max(corr + mid * gd)) <= float(np.max(corr + hi_b * gd)):

            cur_max = float(np.max(corr))

    return h, corr, n_improved > 0

                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _g_vec(h, p, n):
    kk = np.arange(-(n - 1), n)
    g = np.zeros(2 * n - 1)
    idx1 = p - kk
    m1 = (idx1 >= 0) & (idx1 < n)
    g[m1] += (1.0 - h[idx1[m1]])
    idx2 = p + kk
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

def _compute_G(h, n):
    G = np.zeros((n, 2 * n - 1))
    for i in range(n):
        G[i] = _g_vec(h, i, n)
    return G

def _sweep_round(h, n, time_limit, n_coarse=31, n_bisect=42):
    corr = _corr(h)
    G = _compute_G(h, n)
    cur_max = float(np.max(corr))
    order = np.random.permutation(n)
    n_imp = 0
    for ai in range(n):
        i = order[ai]
        if time.time() > time_limit:
            break
        gi = G[i]
        for ai2 in range(ai + 1, n):
            j = order[ai2]
            hi, hj = h[i], h[j]
            d_min = max(-hi, hj - 1.0)
            d_max = min(1.0 - hi, hj)
            if d_max - d_min < 1e-14:
                continue
            gd = gi - G[j]
            ds = np.linspace(d_min, d_max, n_coarse)
            vals = np.max(corr[None, :] + ds[:, None] * gd[None, :], axis=1)
            bi = int(np.argmin(vals))
            if vals[bi] >= cur_max - 1e-14:
                continue
            lo = max(d_min, ds[max(0, bi - 1)])
            hi_b = min(d_max, ds[min(n_coarse - 1, bi + 1)])
            for _ in range(n_bisect):
                mid = 0.5 * (lo + hi_b)
                if float(np.max(corr + mid * gd)) <= float(np.max(corr + hi_b * gd)):
                    hi_b = mid
                else:
                    lo = mid
            d_best = 0.5 * (lo + hi_b)
            if abs(d_best) < 1e-14:
                continue
            h[i] = hi + d_best
            h[j] = hj - d_best
            corr = corr + d_best * gd
            nm = float(np.max(corr))
            if nm < cur_max:
                cur_max = nm
            n_imp += 1
    return h, corr, n_imp > 0

def _slsqp_step(h, n, target, bounds, alpha, maxiter):
    cons = [{'type': 'eq', 'fun': _eq_con, 'args': (target,)}]
    try:
        res = minimize(_obj_smax, h, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'maxls': 40})
        return _repair(res.x, n)
    except Exception:
        return h

def _optimize_from(h, n, dx, target, bounds, t0, budget_s, reserve):
    best_h = h.copy()
    best_c5 = _c5(best_h, dx)
    stall = 0
    rnd = 0
    while time.time() - t0 < budget_s - reserve:
        rnd += 1
        h_c = best_h.copy()
        chunk = min(min(60.0, (budget_s - reserve - (time.time() - t0)) * 0.7), 120.0)
        h_c, _, imp = _sweep_round(h_c, n, time.time() + chunk, 31, 42)
        if imp:
            c5 = _c5(h_c, dx)
            if c5 < best_c5 - 1e-15:
                best_c5 = c5
                best_h = h_c.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        if stall >= 3 and time.time() - t0 < budget_s - reserve - 60:
            alpha = 200 + 100 * (stall - 3)
            hc = _slsqp_step(best_h, n, target, bounds, alpha, 300)
            hc, _, _ = _sweep_round(hc, n, time.time() + 25, 25, 35)
            c5t = _c5(hc, dx)
            if c5t < best_c5 - 1e-15:
                best_c5 = c5t
                best_h = hc.copy()
                stall = 0
        if stall >= 6 and time.time() - t0 < budget_s - reserve - 80:
            scale = min(0.08, 0.02 + 0.01 * stall)
            hp = _repair(best_h + scale * np.random.randn(n), n)
            hc = _slsqp_step(hp, n, target, bounds, 500, 200)
            hc, _, _ = _sweep_round(hc, n, time.time() + 30, 31, 42)
            c5t = _c5(hc, dx)
            if c5t < best_c5 - 1e-15:
                best_c5 = c5t
                best_h = hc.copy()
            stall = 0
        if stall >= 10 and time.time() - t0 < budget_s - reserve - 100:
            stall = 0
            np.random.seed()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)

    # Quick SLSQP warm-up
    for alpha in [50, 200, 600]:
        if time.time() - t0 > budget_s * 0.05:
            break
        hc = _slsqp_step(h0, n, target, bounds, alpha, 300)
        c5c = _c5(hc, dx)
        if c5c < _c5(h0, dx) - 1e-15:
            h0 = hc.copy()

    # Main optimization loop with multi-start
    best_h, best_c5 = _optimize_from(h0, n, dx, target, bounds, t0, budget_s, 120)

    # Try a random restart if time allows
    if time.time() - t0 < budget_s - 200:
        h_rand = _repair(0.5 + 0.3 * np.random.randn(n), n)
        h_rand, c5_rand = _optimize_from(h_rand, n, dx, target, bounds, t0, budget_s, 60)
        if c5_rand < best_c5:
            best_c5 = c5_rand
            best_h = h_rand

    # Final high-precision polish
    if time.time() - t0 < budget_s - 20:
        for nc, nb in [(35, 50), (39, 55), (43, 60)]:
            if time.time() - t0 > budget_s - 12:
                break
            hc = best_h.copy()
            hc, _, _ = _sweep_round(hc, n, time.time() + 25, nc, nb)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hc.copy()

    # Final SLSQP polish
    if time.time() - t0 < budget_s - 4:
        try:
            hc = _slsqp_step(best_h, n, target, bounds, 3000, 500)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hc.copy()
        except Exception:
            pass

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
