    return float(np.max(corr) * dx)

    corr = np.correlate(h, 1.0 - h, mode='full')

def _local_search_binary(h, n, rng, max_rounds=100):

    h = h.copy()
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    best_mc = _max_corr(h, n)

        for i in range(len(ones)):

                    zeros = np.where(h < 0.5)[0]

                    h[oz] = 0.0

    T = T_init

        i = ones[rng.randint(len(ones))]

            h1, m1 = _sa(h0, n, rng, t1)
            h2, m2 = _ls(h1, n)

            # Optimize cycle 2
            t2 = time.time() + min(t_alloc * 0.30, rem() - 40)
            h3, m3 = _sa(h2, n, rng, t2, T0=0.7)
            h4, m4 = _ls(h3, n)

            # Multi-swap SA
            t3 = time.time() + min(t_alloc * 0.20, rem() - 30)
            h5, m5 = _sa_multi(h4, n, rng, t3)
            h6, m6 = _ls(h5, n)

            # Track best for this n
            cands = [(m2, h2), (m4, h4), (m6, h6)]
            cands.sort(key=lambda x: x[0])
            best_mc = cands[0][0]
            best_h = cands[0][1]
            c5_val = best_mc * 2.0 / n

            if c5_val < bc5:
                bc5 = c5_val
                bh = best_h.copy()
                bn = n

    # Continuous refinement of best binary solution
    if bh is not None and rem() > 30:
        t_end = time.time() + min(rem() - 10, 60)
        h_ref, c5_ref = _cont_refine(bh, bn, rng, t_end)
        if c5_ref < bc5:
            bc5 = c5_ref
            bh = h_ref

    # Final local search if time permits
    if rem() > 20:
        h_final, mc_final = _ls(bh, bn, rounds=20)
        c5_final = mc_final * 2.0 / bn
        if c5_final < bc5:
            bc5 = c5_final
            bh = h_final

    if bh is None:
        bh = np.zeros(70)
        bh[:35] = 1.0
        bn = 70
        bc5 = _mc(bh) * 2.0 / 70

    return (bh, bc5, bn)
