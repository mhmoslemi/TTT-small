```python
    w /= w.sum()

    lags = np.arange(-(n-1), n)  # shape (2n-1,)
    w_aligned = w  # w[k_full] where k_full = k - (n-1) + (n-1) = k + n - 1
```

```python
def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):

def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _grad_smooth_max(h, alpha):
    n = len(h)
    c = _corr(h)
    cmax = c.max()
    s = alpha * (c - cmax)
    s = np.clip(s, -500, 500)
    w = np.exp(s)
    w /= w.sum()
    lags = np.arange(-(n-1), n)
    grad = np.zeros(n)
    for p in range(n):
        # First term: sum over k where 0 <= p-k < n
        k_lo = max(p - n + 1, -(n-1))
        k_hi = min(p, n-1)
        if k_lo <= k_hi:
            ks = np.arange(k_lo, k_hi + 1)
            wks = w[ks + n - 1]
            hks = h[p - ks]
            grad[p] += wks.sum() - np.dot(wks, hks)

            wks2 = w[ks2 + n - 1]
            hks2 = h[p + ks2]
            grad[p] -= np.dot(wks2, hks2)
    return grad

def _project_sum(y, target):
    """Project y onto {x: sum(x)=target, 0<=x<=1}"""
    lo, hi = -5.0, 5.0

        if s > target:
            lo = mid
        else:

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _sweep_round(h, corr, time_limit, n_coarse=25, n_bisect=30):
    """Coordinate-pair sweeps with bisection."""
    n = len(h)
    improved = False
    order = np.random.permutation(n)
    for ai, i in enumerate(order):
        for j in order[ai+1:]:
            if time.time() > time_limit:

            hi, hj = h[i], h[j]

            gj = np.zeros(2*n-1)
            idx1 = i - kk

            gi[m1] += (1.0 - h[idx1[m1]])
            idx2 = i + kk
            m2 = (idx2 >= 0) & (idx2 < n)

            idx3 = j - kk
            m3 = (idx3 >= 0) & (idx3 < n)
            gj[m3] += (1.0 - h[idx3[m3]])
            idx4 = j + kk
            m4 = (idx4 >= 0) & (idx4 < n)
            gj[m4] -= h[idx4[m4]]
            gd = gi - gj
            
            ds = np.linspace(d_min, d_max, n_coarse)

            bi = np.argmin(vals)
            cur_max = np.max(corr)
            if vals[bi] >= cur_max - 1e-13:

            lo = max(d_min, ds[max(0, bi-1)])

            h[i] = hi + d_best
            h[j] = hj - d_best
            corr = corr + d_best * gd
            improved = True
    return h, corr, improved
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth_max(c, alpha):
    m = np.max(c)
    s = np.clip(alpha * (c - m), -300, 300)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _grad_sm(h, alpha):
    n = len(h)
    c = _corr(h)
    cm = c.max()
    s = np.clip(alpha * (c - cm), -300, 300)
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(n)
    for p in range(n):
        kl = max(p - n + 1, -(n - 1))
        kh = min(p, n - 1)
        if kl <= kh:
            ks = np.arange(kl, kh + 1)
            g[p] = np.dot(w[ks + n - 1], 1.0 - h[p - ks])
        kl2 = max(-p, -(n - 1))
        kh2 = min(n - 1 - p, n - 1)
        if kl2 <= kh2:
            ks2 = np.arange(kl2, kh2 + 1)
            g[p] -= np.dot(w[ks2 + n - 1], h[p + ks2])
    return g

def _proj(y, tgt):
    lo, hi = -5.0, 5.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(y - mid, 0, 1).sum() > tgt:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) * 0.5, 0, 1)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n * 0.5
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h *= tgt / s
    h = np.clip(h, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-13:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-13:
                break
    return h

def _V(h, n):
    kk = np.arange(-(n - 1), n)
    V = np.zeros((n, 2 * n - 1))
    for i in range(n):
        a = i - kk
        m = (a >= 0) & (a < n)
        V[i, m] += 1.0 - h[a[m]]
        b = i + kk
        m2 = (b >= 0) & (b < n)
        V[i, m2] -= h[b[m2]]
    return V

def _updV(V, i, j, d, n):
    ms = np.arange(n)
    a = ms - i
    ma = (a >= -(n-1)) & (a <= n-1)
    V[ma, a[ma]+n-1] -= d
    b = i - ms
    mb = (b >= -(n-1)) & (b <= n-1)
    V[mb, b[mb]+n-1] -= d
    c_ = ms - j
    mc = (c_ >= -(n-1)) & (c_ <= n-1)
    V[mc, c_[mc]+n-1] += d
    e = j - ms
    me = (e >= -(n-1)) & (e <= n-1)
    V[me, e[me]+n-1] += d

def _sweep(h, tl, nc=25, nb=35):
    n = len(h)
    corr = _corr(h)
    V = _V(h, n)
    imp = False
    order = np.random.permutation(n)
    for ai in range(n):
        i = order[ai]
        for aj in range(ai+1, n):
            j = order[aj]
            if time.time() > tl:
                return h, corr, imp
            dmin = max(-h[i], h[j]-1.0)
            dmax = min(1.0-h[i], h[j])
            if dmax - dmin < 1e-13:
                continue
            gd = V[i] - V[j]
            ds = np.linspace(dmin, dmax, nc)
            cm = corr.max()
            vals = (corr[None,:] + ds[:,None]*gd[None,:]).max(axis=1)
            bi = vals.argmin()
            if vals[bi] >= cm - 1e-14:
                continue
            lo = ds[max(0, bi-1)] if bi > 0 else dmin
            hi = ds[min(nc-1, bi+1)] if bi < nc-1 else dmax
            for _ in range(nb):
                mid = (lo+hi)*0.5
                if (corr+mid*gd).max() <= (corr+hi*gd).max():
                    hi = mid
                else:
                    lo = mid
            db = (lo+hi)*0.5
            h[i] += db
            h[j] -= db
            corr += db * gd
            _updV(V, i, j, db, n)
            imp = True
    return h, corr, imp

def _obj(x, alpha):
    return _smooth_max(_corr(x), alpha)

def _jac(x, alpha):
    return _grad_sm(x, alpha)

def _eqf(x, tgt):
    return x.sum() - tgt

def _slsqp(h, n, tgt, bnds, alpha, miter):
    try:
        cons = [{'type': 'eq', 'fun': _eqf, 'args': (tgt,)}]
        r = minimize(_obj, h, jac=_jac, args=(alpha,), method='SLSQP',
                     bounds=bnds, constraints=cons,
                     options={'maxiter': miter, 'ftol': 1e-16, 'maxls': 40})
        return _repair(r.x, n)
    except Exception:
        return h

def _pg(h, alpha, lr, nst, tgt):
    n = len(h)
    hp = h.copy()
    hc = h.copy()
    best = h.copy()
    best_o = _smooth_max(_corr(h), alpha)
    for _ in range(nst):
        y = _proj(hc + 0.85*(hc - hp), tgt)
        g = _grad_sm(y, alpha)
        t = _proj(y - lr*g, tgt)
        ot = _smooth_max(_corr(t), alpha)
        if ot >= _smooth_max(_corr(y), alpha) + 1e-10:
            lr *= 0.5
            t = _proj(y - lr*g, tgt)
        hp = hc
        hc = t
        o = _smooth_max(_corr(hc), alpha)
        if o < best_o - 1e-12:
            best_o = o
            best = hc.copy()
    return best

def run(seed=42, budget_s=1100, **kwargs):
```

```python
def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth_max(c, alpha):
    m = np.max(c)
    s = np.clip(alpha * (c - m), -300, 300)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _grad_sm(h, alpha):
    n = len(h)
    c = _corr(h)
    cm = c.max()
    s = np.clip(alpha * (c - cm), -300, 300)
    w = np.exp(s); w /= w.sum()
    g = np.zeros(n)
    for p in range(n):
        kl = max(p - n + 1, 1 - n)
        kh = min(p, n - 1)
        if kl <= kh:
            ks = np.arange(kl, kh + 1)
            g[p] = np.dot(w[ks + n - 1], 1.0 - h[p - ks])
        kl2 = max(-p, 1 - n)
        kh2 = min(n - 1 - p, n - 1)
        if kl2 <= kh2:

            g[p] -= np.dot(w[ks2 + n - 1], h[p + ks2])
    return g

        b = i + kk
        m2 = (b >= 0) & (b < n)
        V[i, m2] -= h[b[m2]]
    return V

    a = ms - i; ma = (a >= lb) & (a < n)

    V[mb, b[mb] + n - 1] -= d
    c2 = ms - j; mc = (c2 >= lb) & (c2 < n)
    V[mc, c2[mc] + n - 1] += d
    e = j - ms; me = (e >= lb) & (e < n)
    V[me, e[me] + n - 1] += d

def _jac_s(x, alpha):
```
