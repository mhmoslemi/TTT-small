def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    t_end = t0 + budget_s
    rng = np.random.RandomState(seed + 7777)
    best_h, best_c5, best_n = None, np.inf, 70

    try:
        init_h = np.array(initial_h_values, dtype=float).copy()
        best_n = len(init_h)
    except NameError:
        init_h = None
    if init_h is not None and init_h.size > 10:
        init_h = repair(init_h, best_n)
        best_h = init_h.copy()
        best_c5 = compute_c5(init_h, best_n)
    else:
        best_h = make_harmonic(70, seed, 0)
        best_c5 = compute_c5(best_h, 70)
        best_n = 70

    n_p = best_n
    h_work = best_h.copy()

    # Phase 1: Fourier global optimization (15%)
    t1 = t0 + budget_s * 0.15
    for nh in [4, 5, 6]:
        if time.time() >= t1:
            break
        h_f, c5_f = fourier_optimize(n_p, seed + nh*111, nh, t1)
        if c5_f < best_c5 - 1e-14:
            best_h = h_f.copy()
            best_c5 = c5_f
            h_work = h_f

    # Phase 2: SLSQP multi-parameter (20%)
    t2 = t0 + budget_s * 0.35
    for alpha in [3000, 10000, 40000, 100000, 200000]:
        if time.time() >= t2:
            break
        for topk in [1, 3, 5, 8]:
            if time.time() >= t2:
                break
            for fw in [0.0, 0.02, 0.1, 0.3]:
                if time.time() >= t2:
                    break
                h_new = do_slsqp(h_work.copy(), n_p, alpha, topk, fw, 350, t2)
                c5_new = compute_c5(h_new, n_p)
                if c5_new < best_c5 - 1e-14:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    h_work = h_new
                elif c5_new < compute_c5(h_work, n_p) - 1e-14:
                    h_work = h_new

    # Phase 3: L-BFGS-B (10%)
    t3 = t0 + budget_s * 0.45
    for alpha in [40000, 100000, 200000]:
        if time.time() >= t3:
            break
        for pen in [2000, 20000, 200000]:
            if time.time() >= t3:
                break
            h_new = do_lbfgs(h_work.copy(), n_p, alpha, 5, 0.1, pen, 3000, t3)
            c5_new = compute_c5(h_new, n_p)
            if c5_new < best_c5 - 1e-14:
                best_h = h_new.copy()
                best_c5 = c5_new
                h_work = h_new

    # Phase 4: Simulated annealing coordinate descent (35%)
    t4 = t0 + budget_s * 0.80
    h_sa, c5_sa = coord_sa(h_work, n_p, t4, rng)
    if c5_sa < best_c5 - 1e-14:
        best_h = h_sa.copy()
        best_c5 = c5_sa

    # Phase 5: High-alpha refinement + fine coord (12%)
    t5 = t0 + budget_s * 0.92
    for alpha in [200000, 400000]:
        if time.time() >= t5:
            break
        h_new = do_slsqp(best_h.copy(), best_n, alpha, 10, 0.2, 500, t5)
        c5_new = compute_c5(h_new, best_n)
        if c5_new < best_c5 - 1e-14:
            best_h = h_new.copy()
            best_c5 = c5_new
    h_f2, c5_f2 = coord_sa(best_h, best_n, t5, rng, T_sa=0.0003, cool=0.9998)
    if c5_f2 < best_c5 - 1e-14:
        best_h = h_f2.copy()
        best_c5 = c5_f2

    # Phase 6: Alternative n (7%)
    t6 = t0 + budget_s * 0.97
    for n_alt in [64, 72, 80, 76, 66]:
        if time.time() >= t6:
            break
        if n_alt == best_n:
            continue
        h_alt = make_harmonic(n_alt, seed + n_alt * 13, n_alt % 4)
        h_alt = do_slsqp(h_alt, n_alt, 50000, 5, 0.1, 500, t6)
        c5_alt = compute_c5(h_alt, n_alt)
        if c5_alt < best_c5 - 1e-12:
            h_alt, c5_alt = coord_sa(h_alt, n_alt, t6, rng)
            if c5_alt < best_c5 - 1e-14:
                best_h = h_alt.copy()
                best_c5 = c5_alt
                best_n = n_alt

    # Phase 7: Final ultra-fine (3%)
    t7 = t0 + budget_s * 0.995
    if best_h is not None and time.time() < t7:
        h_fin, c5_fin = coord_sa(best_h, best_n, t7, rng, T_sa=0.0001, cool=0.9999)
        if c5_fin < best_c5 - 1e-14:
            best_h = h_fin.copy()
            best_c5 = c5_fin

    if best_h is None:
        best_h = repair(np.full(70, 0.5), 70)
        best_n = 70
    best_h = repair(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return (best_h, best_c5, best_n)
