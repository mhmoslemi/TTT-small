def _local_fine(h, n, dx, rng, deadline, n_max_tries=100000):

    per_scale = max(1000, n_max_tries // len(scales))
    for scale in scales:

        for _ in range(per_scale):
            if time.time() >= deadline:
                return h, best_c5
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = scale * (2.0 * rng.random() - 1.0)
            ni, nj = h[i] + d, h[j] - d
            if ni < -1e-10 or ni > 1.0+1e-10 or nj < -1e-10 or nj > 1.0+1e-10:
                continue
            h[i], h[j] = ni, nj
            c5 = _c5(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                improved = True
            else:
                h[i], h[j] = h[i]-d, h[j]+d
        if not improved and scale < 0.001:
            break
    return h, best_c5

def _perturb(h, n, rng, kind):
    hp = h.copy()
    if kind == 0:
        # Block rotation
        b = rng.randint(3, 10)
        i = rng.randint(0, n-b)
        hp[i:i+b] = np.roll(hp[i:i+b], rng.randint(1,b))
    elif kind == 1:
        # Sine perturbation
        f = rng.randint(1, 6)
        ph = rng.uniform(0, 2*np.pi)
        hp += rng.uniform(0.03, 0.1) * np.sin(2*np.pi*f*np.arange(n)/n + ph)
    elif kind == 2:
        # Circular shift
        j = rng.randint(n//5, 4*n//5)
        hp = np.roll(hp, j)
    elif kind == 3:
        # Mirror
        hp = np.concatenate([hp[:n//2][::-1], hp[n//2:]])
    else:
        hp += rng.uniform(-0.05, 0.05, n)
    return _repair(hp, n)
