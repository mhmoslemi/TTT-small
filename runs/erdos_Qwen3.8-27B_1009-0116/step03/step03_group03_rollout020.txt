    # a_m = dc[m]/dh[i]
    # b_m = c[m] - a_m * h[i]

    b = c - a * t

    # at s=0: val = max(b)
    # at s=1: val = max(a + b)

                c5 = _c5(hc, dx)
                if c5 < best_c5:
                    best_c5, best_h = c5, hc
                # Polish with CD
                h2, c52 = _cd_optimize(hc.copy(), n, 100)
                if c52 < best_c5:
                    best_c5, best_h = c52, h2.copy()
            except Exception:
                pass

    # Phase 3: Diverse restarts + coordinate descent
    attempts = 0
    while tl() > 100 and attempts < 400:
        r = attempts % 8
        if r == 0:
            h_p = best_h + 0.003 * rng.randn(n)
        elif r == 1:
            h_p = best_h + 0.01 * rng.randn(n)
        elif r == 2:
            h_p = best_h + 0.04 * rng.randn(n)
        elif r == 3:
            h_p = _init_structured(n, rng)
        elif r == 4:
            h_p = best_h.copy()
            for _ in range(n // 8):
                i, j = rng.choice(n, 2, replace=False)
                h_p[[i, j]] = best_h[[j, i]]
        elif r == 5:
            h_p = 1.0 - best_h  # complement symmetry
        elif r == 6:
            sh = rng.randint(1, n // 3 + 1)
            h_p = np.roll(best_h, sh).copy()
        else:
            h_p = best_h + 0.008 * np.sin(2 * np.pi * rng.randint(1, 7) * np.arange(n) / n)
        h_p = _repair(h_p, n)
        h, c5 = _cd_optimize(h_p, n, 80)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()
        attempts += 1

    # Phase 4: SLSQP subgradient polish
    if tl() > 50:
        bounds = [(0.0, 1.0)] * n
        target = n / 2.0
        try:
            def obj_fn(x):
                c = _corr(x)
                return c.max()
            def jac_fn(x):
                c = _corr(x)
                mx = c.max()
                active = c >= mx - 1e-12
                g = np.zeros(n)
                for mi in np.where(active)[0]:
                    k = mi - (n - 1)
                    if k >= 0:
                        g[k:] += (1.0 - x[:n - k])
                        g[:n - k] -= x[k:]
                    else:
                        g[:n + k] += (1.0 - x[-k:])
                        g[-k:] -= x[:n + k]
                return g / max(1, len(active))
            res = minimize(obj_fn, best_h, jac=jac_fn, method='SLSQP', bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                                       'jac': lambda x: np.ones(n)}],
                          options={'maxiter': 500, 'ftol': 1e-20, 'iprint': 0})
            hc = _repair(res.x, n)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, hc
        except Exception:
            pass

    # Phase 5: Final deep coordinate descent
    if tl() > 20:
        h, c5 = _cd_optimize(best_h.copy(), n, 1000)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Phase 6: Fine random pair search
    if tl() > 10:
        h = best_h.copy()
        c5_cur = _c5(h, dx)
        for _ in range(300000):
            if tl() < 10:
                break
            i = rng.randint(n)
            j = (i + 1 + rng.randint(n - 1)) % n
            d = rng.choice([0.0003, 0.001, 0.003, 0.01, 0.03]) * (1 if rng.rand() < 0.5 else -1)
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                h_r = _repair(h, n)
                c5_new = _c5(h_r, dx)
                if c5_new < c5_cur - 1e-16:
                    c5_cur = c5_new
                    h = h_r
                    best_c5, best_h = c5_cur, h.copy()
                else:
                    h = h_r  # still use repaired version as base
        if c5_cur < best_c5:
            best_c5, best_h = c5_cur, _repair(h, n)

    # Phase 7: Final CD polish
    if tl() > 5:
        h, c5 = _cd_optimize(best_h.copy(), n, 200)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
