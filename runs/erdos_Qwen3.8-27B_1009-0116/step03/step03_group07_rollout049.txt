_state = {}

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(5000):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        d = target - s
        if abs(d) < 1e-14:
            break
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-14:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
    return h

    h = h.copy()
    best = compute_c5(h, n)
    for it in range(20000):

            break
        corr = compute_corr(h)
        m = np.max(corr)
        # Find top-k lags
        top_idx = np.argsort(corr)[::-1][:topk]
        c_top = corr[top_idx]
        # Weight: emphasize the max, gradually include others
        w = np.exp((c_top - c_top[0]) * _state['sub_alpha'])
        w /= w.sum()
        # Compute gradient of weighted sum of top correlations
        grad = np.zeros(n)

import numpy as np
from scipy.optimize import minimize
import time

_state = {}

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(5000):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        d = target - s
        if abs(d) < 1e-14:
            break
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-14:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
    return h

def obj_smooth(h):
    corr = compute_corr(h)
    m = np.max(corr)
    a = _state['alpha']
    s = a * (corr - m)
    s = np.clip(s, -500, 500)
    val = m + np.log(np.sum(np.exp(s))) / a
    k = _state['topk']
    if k > 1:
        sc = np.sort(corr)[::-1][:k]
        val += _state['flat_w'] * np.sum((sc - sc.mean())**2)
    return val

def eq_con(h):
    return h.sum() - _state['target']

def pen_obj(h):
    return obj_smooth(h) + _state['pen'] * (h.sum() - _state['target'])**2

def do_slsqp(h, n, alpha, topk, flat_w, maxiter):
    _state['alpha'] = alpha
    _state['topk'] = topk
    _state['flat_w'] = flat_w
    _state['target'] = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_con}]
    try:
        res = minimize(obj_smooth, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return repair(res.x, n)
    except Exception:
        return h

def do_lbfgs(h, n, alpha, topk, flat_w, pen, maxiter):
    _state['alpha'] = alpha
    _state['topk'] = topk
    _state['flat_w'] = flat_w
    _state['target'] = n / 2.0
    _state['pen'] = pen
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(pen_obj, h, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-12})
        return repair(res.x, n)
    except Exception:
        return h

def subgrad_descent(h, n, eta0, topk, sub_alpha, t_end, max_iter=50000):
    h = h.copy()
    best = compute_c5(h, n)
    eta = eta0
    n_improve = 0
    for it in range(max_iter):
        if time.time() > t_end:
            break
        corr = compute_corr(h)
        m = np.max(corr)
        top_idx = np.argsort(corr)[::-1][:topk]
        c_top = corr[top_idx]
        w = np.exp((c_top - c_top[0]) * sub_alpha)
        w /= w.sum()
        grad = np.zeros(n)
        j = np.arange(n)
        for wi, p in zip(w, top_idx):
            L = p - (n - 1)
            if L >= 0:
                m1 = (j >= L) & (j <= n - 1 - L)
                m2 = (j >= 0) & (j <= n - 1 - L)
                g = np.zeros(n)
                g[m1] += (1.0 - h[j[m1] - L])
                g[m2] -= h[j[m2] + L]
            else:
                aL = -L
                m1 = (j >= aL) & (j <= n - 1)
                g1 = np.zeros(n)
                g1[m1] += (1.0 - h[j[m1] - L])
                m2 = (j >= aL) & (j <= n - 1)
                g2 = np.zeros(n)
                g2[m2] -= h[j[m2] + L]
                g = g1 + g2
            grad += wi * g
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        h_new = h - (eta / gn) * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = repair(h_new, n)
        c5_new = compute_c5(h_new, n)
        if c5_new < best:
            h = h_new
            best = c5_new
            n_improve += 1
            eta *= 0.9998
        else:
            eta *= 0.99
            if eta < 1e-8:
                break
        if n_improve > 0 and it % 200 == 0:
            eta = max(eta, 0.003)
    return h, best

def coord_descent_focused(h, n, n_rounds, t_end, rng):
    h = h.copy()
    best = compute_c5(h, n)
    steps = [0.025, 0.012, 0.006, 0.003, 0.0015, 0.0008, 0.0004]
    for r in range(n_rounds):
        if time.time() > t_end:
            break
        corr = compute_corr(h)
        top_idx = np.argsort(corr)[::-1][:5]
        relevant = set()
        for p in top_idx:
            L = p - (n - 1)
            lo = max(0, -L)
            hi = min(n - 1, n - 1 - L)
            for i in range(lo, hi + 1):
                relevant.add(i)
                if i + L != i:
                    relevant.add(i + L)
        relevant = list(relevant) if relevant else list(range(n))
        improved = False
        n_tries = min(3 * n, 500)
        for _ in range(n_tries):
            if time.time() > t_end:
                break
            if rng.random() < 0.75:
                i = relevant[rng.randint(len(relevant))]
            else:
                i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            hi_v, hj_v = h[i], h[j]
            for d in steps:
                for sign in (1, -1):
                    delta = sign * d
                    ni, nj = hi_v + delta, hj_v - delta
                    if ni < -1e-10 or ni > 1.0 + 1e-10 or nj < -1e-10 or nj > 1.0 + 1e-10:
                        continue
                    h[i], h[j] = ni, nj
                    c5 = compute_c5(h, n)
                    if c5 < best - 1e-15:
                        best = c5
                        improved = True
                        break
                    h[i], h[j] = hi_v, hj_v
                if improved:
                    break
        if not improved and r > 100:
            break
    return h, best

def make_start(n, seed_val, kind):
    rng = np.random.RandomState(seed_val)
    if kind == 0:
        h = 0.5 + 0.3 * rng.randn(n)
    elif kind == 1:
        h = np.where(np.arange(n) < n // 2, 1.0, 0.0) + 0.1 * rng.randn(n)
    elif kind == 2:
        f = rng.randint(2, 8)
        ph = rng.uniform(0, 6.283)
        h = 0.5 + 0.35 * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
    elif kind == 3:
        block = max(2, n // 7)
        h = np.zeros(n)
        for i in range(0, n, 2 * block):
            h[i:i + block] = 1.0
        h = 0.5 * h + 0.2 * rng.randn(n)
    elif kind == 4:
        h = 0.5 + 0.3 * np.cos(np.pi * np.arange(n) / n) ** 3
    elif kind == 5:
        h = 0.5 + 0.25 * np.cos(2 * np.pi * 3 * np.arange(n) / n) + 0.15 * np.cos(2 * np.pi * 5 * np.arange(n) / n)
    else:
        h = rng.uniform(0, 1, n)
    return repair(h, n)
