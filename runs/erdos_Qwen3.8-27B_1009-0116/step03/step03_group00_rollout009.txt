import numpy as np
from scipy.optimize import minimize
import time

_N = 70
_ALPHA = 1000
_BOUNDS = [(0.0, 1.0)] * 70

def _c5_all(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5(h, n):
    return float(np.max(_c5_all(h, n)))

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

def _ls_obj(x):
    c = _c5_all(x, _N)
    m = np.max(c)
    s = _ALPHA * (c - m)
    s = np.clip(s, -400, 400)
    return m + np.log(np.sum(np.exp(s))) / _ALPHA

def _ls_grad(x):

    c = _c5_all(x, _N)
    m = np.max(c)
    s = _ALPHA * (c - m)
