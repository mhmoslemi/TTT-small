def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        # Project: scale then fix
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _corr_and_grad(h):

    n = len(h)

                    for sign in [1, -1]:
                        d_try = sign * d
                        ni, nj = h[i] + d_try, h[j] - d_try
                        if ni < -1e-12 or ni > 1 + 1e-12 or nj < -1e-12 or nj > 1 + 1e-12:
                            continue
                        h[i], h[j] = ni, nj
                        c5n = _eval(h)
                        if c5n < best_c5:
                            best_c5 = c5n
                            best_d = d_try
                        h[i], h[j] = ni - d_try, nj + d_try
                if best_d != 0.0:
                    h[i] += best_d
                    h[j] -= best_d
                    c5c = best_c5
                    improved = True
        if not improved:
            break
    return h, c5c

def _powell_polish(h, n, maxiter=500):
    def _obj_repaired(z):
        try:
            return _eval(_repair(z, n))
        except Exception:
            return 1.0
    try:
        res = fmin(_obj_repairment, h.copy(), disp=False, maxiter=maxiter, maxfun=100000,
                   args=(n,), full_output=True)
        return _repair(res[0], n), _eval(_repair(res[0], n))
    except Exception:
        return h, _eval(h)

def _obj_repairment(z, n):
    return _eval(_repair(z, n))

def _active_nlp(h0, n, epsilon=0.005, maxiter=2000):
    """Solve NLP considering only near-active constraints."""
    h = h0.copy()
    t0 = _eval(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    active = np.where(c > t0 - epsilon)[0]
    if len(active) < 3:
        top_k = np.argsort(c)[::-1][:max(3, len(active))]
        active = top_k
    idx = np.arange(n)
    lags_all = np.arange(2 * n - 1) - (n - 1)
    
    def _act_ineq(x):
        hh = x[:n]
        cc = np.correlate(hh, 1.0 - hh, mode='full') * (2.0 / n)
        return x[n] - cc[active]
    
    def _act_ineq_jac(x):
        hh = x[:n]
        J = np.zeros((len(active), n + 1))
        for ai, ki in enumerate(active):
            l = lags_all[ki]
            v1 = (idx - l) >= 0
            v1 &= (idx - l) < n
            v2 = (idx + l) >= 0
            v2 &= (idx + l) < n
            gk = np.zeros(n)
            if v1.any():
                gk[v1] += (1.0 - hh[idx[v1] - l])
            if v2.any():
                gk[v2] -= hh[idx[v2] + l]
            J[ai, :n] = -(2.0 / n) * gk
        J[:, n] = 1.0
        return J
    
    x0 = np.concatenate([h, [t0]])
    bnds = [(0.0, 1.0)] * n + [(-0.5, 1.5)]
    cons = [
        {'type': 'ineq', 'fun': _act_ineq, 'jac': _act_ineq_jac},
        {'type': 'eq', 'fun': _nlp_eq_fun, 'jac': _nlp_eq_jac, 'args': (n,)},
    ]
    try:
        res = minimize(_nlp_obj, x0, jac=_nlp_obj_jac, args=(n,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return _repair(res.x[:n], n)
    except Exception:
        pass
    return _repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)

    def _el():
        return time.time() - t0

    h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h.copy()
    best_c5 = _eval(best_h)

    # Phase 1: Full NLP solve from initial
    h1 = _solve_nlp(h, n, maxiter=3000)
    c5 = _eval(h1)
    if c5 < best_c5:
        best_c5, best_h = c5, h1.copy()

    # Phase 2: Active-set NLP (faster, focuses on bottleneck lags)
    if _el() < budget_s - 200:
        h2 = _active_nlp(best_h, n, epsilon=0.003, maxiter=3000)
        c5 = _eval(h2)
        if c5 < best_c5:
            best_c5, best_h = c5, h2.copy()

    # Phase 3: Basin hopping with NLP solves
    if _el() < budget_s - 150:
        sigmas = [0.01, 0.03, 0.06, 0.1, 0.15, 0.2, 0.02, 0.05, 0.12, 0.08,
                  0.04, 0.07, 0.14, 0.03, 0.09, 0.05, 0.11, 0.06, 0.04, 0.16]
        for trial in range(len(sigmas)):
            if _el() > budget_s - 150:
                break
            sig = sigmas[trial]
            hp = _repair(best_h + sig * rng.randn(n), n)
            method = _solve_nlp if (trial % 2 == 0) else _active_nlp
            h_new = method(hp, n, maxiter=2000)
            c5 = _eval(h_new)
            if c5 < best_c5:
                best_c5, best_h = c5, h_new.copy()

    # Phase 4: Coordinate descent (best-improvement)
    if _el() < budget_s - 60:
        h_cd, c5_cd = _coord_descent(best_h, n, max_sweeps=300)
        if c5_cd < best_c5:
            best_c5, best_h = c5_cd, h_cd

    # Phase 5: Re-run NLP from coordinate-descent result
    if _el() < budget_s - 40:
        h5 = _solve_nlp(best_h, n, maxiter=5000, ftol=1e-20)
        c5 = _eval(h5)
        if c5 < best_c5:
            best_c5, best_h = c5, h5.copy()
        h5b = _active_nlp(best_h, n, epsilon=0.001, maxiter=5000)
        c5 = _eval(h5b)
        if c5 < best_c5:
            best_c5, best_h = c5, h5b.copy()

    # Phase 6: Fine coordinate descent with small deltas
    if _el() < budget_s - 20:
        h = best_h.copy()
        c5c = _eval(h)
        fine_deltas = np.array([1e-4, 3e-4, 1e-3, 3e-3, 1e-2, -1e-4, -3e-4, -1e-3, -3e-3, -1e-2])
        for _sweep in range(200):
            if _el() > budget_s - 20:
                break
            improved = False
            for i in range(n):
                for j in range(i + 1, n):
                    best_d = 0.0
                    best_c5 = c5c
                    for d in fine_deltas:
                        ni, nj = h[i] + d, h[j] - d
                        if ni < -1e-12 or ni > 1 + 1e-12 or nj < -1e-12 or nj > 1 + 1e-12:
                            continue
                        h[i], h[j] = ni, nj
                        c5n = _eval(h)
                        if c5n < best_c5:
                            best_c5 = c5n
                            best_d = d
                        h[i], h[j] = ni - d, nj + d
                    if best_d != 0.0:
                        h[i] += best_d
                        h[j] -= best_d
                        c5c = best_c5
                        improved = True
                        if c5c < best_c5:
                            best_c5, best_h = c5c, h.copy()
            if not improved:
                break
        if c5c < best_c5:
            best_c5, best_h = c5c, h

    # Phase 7: Final NLP polish
    if _el() < budget_s - 5:
        h7 = _solve_nlp(best_h, n, maxiter=5000, ftol=1e-20)
        c5 = _eval(h7)
        if c5 < best_c5:
            best_c5, best_h = c5, h7.copy()

    best_h = _repair(best_h, n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
