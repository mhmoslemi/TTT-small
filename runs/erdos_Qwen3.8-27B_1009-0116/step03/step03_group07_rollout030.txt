                ha = linsq_refine(ha, n_alt, 20000, 3, 0.01, 1000, 1500)
            ca = compute_c5(ha, n_alt)
            if ca < best_c5 - 1e-14:
                best_h = ha.copy(); best_c5 = ca; best_n = n_alt

    # Phase 6: Final polish
    if time.time() < t0 + budget_s * 0.995:
        hp = finelinas(best_h.copy(), best_n, 200000, 5, 0.02, 50000, 800)
        cp = compute_c5(hp, best_n)
        if cp < best_c5 - 1e-14:
            best_h = hp.copy(); best_c5 = cp
        hc3, cc3 = coord_desc(best_h, best_n, t0 + budget_s * 0.998, rng)
        if cc3 < best_c5 - 1e-14:
            best_h = hc3.copy(); best_c5 = cc3

    if best_h is None:
        best_n = 70
        best_h = repair(np.full(best_n, 0.5), best_n)
        best_c5 = compute_c5(best_h, best_n)

    best_h = repair(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return (best_h, best_c5, best_n)
