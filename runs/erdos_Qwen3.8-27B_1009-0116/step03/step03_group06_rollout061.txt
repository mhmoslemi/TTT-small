def _compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _repair_continuous(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _to_binary(h, n, target_count):

        s = np.clip(s, -500, 500)
        return mm + np.log(np.sum(np.exp(s))) / alpha + 100.0 * (np.sum(x) - target) ** 2

                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _to_binary(h, n, target_count):
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:target_count]] = 1.0
    return h_bin

def _smooth_obj(x, alpha, target, pen):
    c = np.correlate(x, 1.0 - x, mode='full')
    mm = np.max(c)
    s = alpha * (c - mm)
    s = np.clip(s, -500, 500)
    return mm + np.log(np.sum(np.exp(s))) / alpha + pen * (np.sum(x) - target) ** 2

def _lbfgs_opt(h, n, dx, alpha, pen, maxiter):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    best_h, best_c5 = h, _compute_c5(h, dx)
    try:
        res = minimize(_smooth_obj, best_h, args=(alpha, target, pen),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-7})
        hc = _repair(res.x, n)
        c5 = _compute_c5(hc, dx)
        if c5 < best_c5:
            best_h, best_c5 = hc, c5
    except Exception:
        pass
    return best_h, best_c5

def _binary_sa(h, n, dx, rng, deadline):
    target = n // 2
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h <= 0.5)[0].tolist()
    best_h = h.copy()
    best_c5 = _compute_c5(h, dx)
    cur_c5 = best_c5
    T = 0.02 * dx
    T_min = 1e-7 * dx
    cool_rate = 0.99975
    stagnation = 0
    max_stag = 5000

    while time.time() < deadline:
        if T < T_min and stagnation > max_stag:
            T = 0.01 * dx
            stagnation = 0
        if stal_exceeded := stagnation > 20000:
            T = 0.03 * dx
            stagnation = 0
        move_type = rng.random()
        if move_type < 0.65:
            if not ones or not zeros:
                continue
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i], h[j] = 0.0, 1.0
            new_c5 = _compute_c5(h, dx)
            if new_c5 < cur_c5 or (cur_c5 - new_c5 < 0 and rng.random() < np.exp(-(new_c5 - cur_c5) / T)):
                cur_c5 = new_c5
                ones.remove(i)
                zeros.remove(j)
                ones.append(i)
                zeros.append(j)
                ones = np.where(h > 0.5)[0].tolist()
                zeros = np.where(h <= 0.5)[0].tolist()
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    stagnation = 0
                else:
                    stagnation += 1
            else:
                h[i], h[j] = 1.0, 0.0
                stagnation += 1
        elif move_type < 0.88:
            if len(ones) < 2 or len(zeros) < 2:
                continue
            i1, i2 = rng.choice(len(ones), 2, replace=False)
            j1, j2 = rng.choice(len(zeros), 2, replace=False)
            oi1, oi2 = ones[i1], ones[i2]
            oj1, oj2 = zeros[j1], zeros[j2]
            h[oi1], h[oi2] = 0, 0
            h[oj1], h[oj2] = 1, 1
            new_c5 = _compute_c5(h, dx)
            if new_c5 < cur_c5 or rng.random() < np.exp(-(new_c5 - cur_c5) / T):
                cur_c5 = new_c5
                ones = np.where(h > 0.5)[0].tolist()
                zeros = np.where(h <= 0.5)[0].tolist()
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    stagnation = 0
                else:
                    stagnation += 1
            else:
                h[oi1], h[oi2] = 1, 1
                h[oj1], h[oj2] = 0, 0
                stagnation += 1
        else:
            if len(ones) < 3 or len(zeros) < 3:
                continue
            sel = rng.choice(len(ones), 3, replace=False)
            sel2 = rng.choice(len(zeros), 3, replace=False)
            ois = np.array([ones[k] for k in sel])
            ojs = np.array([zeros[k] for k in sel2])
            h[ois] = 0
            h[ojs] = 1
            new_c5 = _compute_c5(h, dx)
            if new_c5 < cur_c5 or rng.random() < np.exp(-(new_c5 - cur_c5) / T):
                cur_c5 = new_c5
                ones = np.where(h > 0.5)[0].tolist()
                zeros = np.where(h <= 0.5)[0].tolist()
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                    stagnation = 0
                else:
                    stagnation += 1
            else:
                h[ois] = 1
                h[ojs] = 0
                stagnation += 1
        T = max(T_min, T * cool_rate)
    return best_h, best_c5

def _fine_local(h, n, dx, rng, deadline, n_tries):
    best_c5 = _compute_c5(h, dx)
    h = h.copy()
    scales = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    per_scale = max(200, n_tries // len(scales))
    for scale in scales:
        improved = False
        for _ in range(per_scale):
            if time.time() >= deadline:
                return h, best_c5
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = scale * (2.0 * rng.random() - 1.0)
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            c5 = _compute_c5(h, dx)
            if c5 < best_c5 - 1e-15:
                best_c5 = c5
                improved = True
            else:
                h[i], h[j] = h[i] - d, h[j] + d
        if not improved and scale < 0.002:
            break
    return h, best_c5

def _block_local(h, n, dx, rng, deadline, n_iters):
    best_c5 = _compute_c5(h, dx)
    h = h.copy()
    for bs in [2, 3]:
        for _ in range(n_iters // 2):
            if time.time() >= deadline:
                return h, best_c5
            i = rng.randint(0, n - bs + 1)
            j = rng.randint(0, n - bs + 1)
            if abs(i - j) < bs:
                continue
            eps = rng.uniform(0.002, 0.012) * (1.0 if rng.random() < 0.5 else -1.0)
            ok = True
            for di in range(bs):
                if h[i+di] + eps < -1e-10 or h[i+di] + eps > 1.0 + 1e-10:
                    ok = False
                if h[j+di] - eps < -1e-10 or h[j+di] - eps > 1.0 + 1e-10:
                    ok = False
            if not ok:
                continue
            h[i:i+bs] += eps
            h[j:j+bs] -= eps
            c5 = _compute_c5(h, dx)
            if c5 < best_c5 - 1e-15:
                best_c5 = c5
            else:
                h[i:i+bs] -= eps
                h[j:j+bs] += eps
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target_count = n // 2

    h0 = np.array(initial_h_values, dtype=float)
    h0 = np.clip(h0, 0, 1)
    h_bin = (h0 > 0.5).astype(float)
    cnt = int(h_bin.sum())
    if cnt != target_count:
        if cnt > target_count:
            ones_idx = np.where(h_bin == 1)[0]
            order = np.argsort(h0[ones_idx])
            for i in range(cnt - target_count):
                h_bin[ones_idx[order[i]]] = 0
        else:
            zeros_idx = np.where(h_bin == 0)[0]
            order = np.argsort(-h0[zeros_idx])
            for i in range(target_count - cnt):
                h_bin[zeros_idx[order[i]]] = 1

    best_h = h_bin.copy()
    best_c5 = _compute_c5(best_h, dx)

    def remaining():
        return budget_s - (time.time() - t_start)

    # Phase 1: Binary SA - multiple diverse restarts (45% budget)
    t_end1 = t_start + budget_s * 0.45
    n_restarts = 8
    for r in range(n_restarts):
        if time.time() >= t_end1 or remaining() < 350:
            break
        if r == 0:
            start_h = best_h.copy()
        elif r == 1:
            start_h = _to_binary(h0, n, target_count)
        elif r == 2:
            start_h = np.zeros(n)
            start_h[rng.choice(n, target_count, replace=False)] = 1.0
        elif r == 3:
            start_h = np.zeros(n)
            pos = np.arange(0, n, 2)
            start_h[pos[:target_count]] = 1.0
            for _ in range(8):
                oi = rng.choice(np.where(start_h == 1)[0])
                oj = rng.choice(np.where(start_h == 0)[0])
                start_h[oi], start_h[oj] = start_h[oj], start_h[oi]
        elif r == 4:
            start_h = np.zeros(n)
            period = 3
            for idx in range(n):
                if idx % (2 * period) < period:
                    start_h[idx] = 1.0
            start_h = _to_binary(start_h, n, target_count)
        elif r == 5:
            phase = rng.uniform(0, 2 * np.pi)
            start_h = np.zeros(n)
            t_arr = np.arange(n) / n
            for idx in range(n):
                start_h[idx] = 1.0 if np.sin(2*np.pi*3*t_arr[idx] + phase) > 0 else 0.0
            start_h = _to_binary(start_h, n, target_count)
        else:
            start_h = best_h.copy()
            for _ in range(15):
                oi = rng.choice(np.where(start_h == 1)[0])
                oj = rng.choice(np.where(start_h == 0)[0])
                start_h[oi], start_h[oj] = start_h[oj], start_h[oi]

        sa_time = min(55, (t_end1 - time.time()) / max(1, n_restarts - r))
        h_t, c5_t = _binary_sa(start_h, n, dx, rng, time.time() + sa_time)
        if c5_t < best_c5:
            best_h, best_c5 = h_t, c5_t

    # Phase 2: Continuous L-BFGS polish (20% budget)
    if remaining() > 250:
        t_end2 = time.time() + min(budget_s * 0.20, remaining() - 250)
        for alpha in [300, 1000, 5000, 20000, 80000]:
            if time.time() >= t_end2 or remaining() < 200:
                break
            for pen in [50.0, 200.0, 1000.0]:
                h_t, c5_t = _lbfgs_opt(best_h, n, dx, alpha, pen, maxiter=300)
                if c5_t < best_c5:
                    best_h, best_c5 = h_t, c5_t

    # Phase 3: Binary SA from continuous solution projection (15% budget)
    if remaining() > 200:
        t_end3 = time.time() + min(budget_s * 0.15, remaining() - 200)
        h_bin3 = _to_binary(best_h, n, target_count)
        h_t, c5_t = _binary_sa(h_bin3, n, dx, rng, t_end3)
        if c5_t < best_c5:
            best_h, best_c5 = h_t, c5_t

    # Phase 4: L-BFGS refinement (12% budget)
    if remaining() > 150:
        t_end4 = time.time() + min(budget_s * 0.12, remaining() - 130)
        for alpha in [10000, 50000, 200000]:
            if time.time() >= t_end4 or remaining() < 100:
                break
            h_t, c5_t = _lbfgs_opt(best_h, n, dx, alpha, 5000.0, maxiter=400)
            if c5_t < best_c5:
                best_h, best_c5 = h_t, c5_t

    # Phase 5: Block local search (7% budget)
    if remaining() > 100:
        t_end5 = time.time() + min(budget_s * 0.07, remaining() - 80)
        h_t, c5_t = _block_local(best_h, n, dx, rng, t_end5, n_iters=8000)
        if c5_t < best_c5:
            best_h, best_c5 = h_t, c5_t

    # Phase 6: Fine local search (6% budget)
    if remaining() > 60:
        t_end6 = time.time() + min(budget_s * 0.06, remaining() - 40)
        h_t, c5_t = _fine_local(best_h, n, dx, rng, t_end6, n_tries=60000)
        if c5_t < best_c5:
            best_h, best_c5 = h_t, c5_t

    # Phase 7: Final L-BFGS + fine polish
    if remaining() > 30:
        h_t, c5_t = _lbfgs_opt(best_h, n, dx, 100000, 10000.0, maxiter=200)
        if c5_t < best_c5:
            best_h, best_c5 = h_t, c5_t
    if remaining() > 10:
        t_end7 = time.time() + max(2, min(8, remaining() - 2))
        h_t, c5_t = _fine_local(best_h, n, dx, rng, t_end7, n_tries=10000)
        if c5_t < best_c5:
            best_h, best_c5 = h_t, c5_t

    best_h = _repair(best_h, n)
    best_c5 = _compute_c5(best_h, dx)
    return (best_h, best_c5, n)
