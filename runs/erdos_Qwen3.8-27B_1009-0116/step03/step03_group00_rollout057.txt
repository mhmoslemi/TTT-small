def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

    dx = 2.0 / n
    return float(np.max(_corr(h))) * dx

def _corn_vec(h, n):
    return _corr(h) * (2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return h

def _smooth_max(h, n, alpha):
    c = _corn_vec(h, n)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

    c = _corn_vec(h, n)
    m = np.max(c)

            g[jp] += wk * (1.0 - h[i])
            g[i] -= wk * h[jp]

            jp = i + lag
            g[jp] += (1.0 - h[i])
            g[i] -= h[jp]

    c5_cur = _c5(h, n)

        # Projected gradient step

        c5_new = _c5(h_new, n)

            c5_cur = c5_new
            lr = min(lr * 1.1, 0.05)
        else:
            lr *= 0.7
            if lr < 1e-8:

    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= tgt / s
    h = np.clip(h, 0.0, 1.0)

    if abs(diff) > 1e-14:

        for idx in order:

            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a

_TGT = 0.0
_ALPHA = 0.0
_KVAL = 0

def _init_con(n):
    global _N, _TGT
    _N = n
    _TGT = n / 2.0

def _con_fun(x):
    return float(np.sum(x) - _TGT)

def _con_jac(x):
    return np.ones(_N)

    global _KVAL, _N

def _corn(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (n / 2.0) / s
    h = np.clip(h, 0.0, 1.0)
    diff = (n / 2.0) - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-14:
