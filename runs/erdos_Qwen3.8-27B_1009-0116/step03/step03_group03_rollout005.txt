import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return np.max(_corr(h)) * dx

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
            else:
                add = max(diff, h[idx] - 1.0)
            h[idx] += add
            diff -= add
            if abs(diff) < 1e-12:
                break
    return h

def _upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    return h_new

    best_h, best_c = h.copy(), _c5(h, dx)
    for alpha_try in [40000, 100000]:
        if tleft() < 30:
            break
        try:
            def obj2(x, a=alpha_try):
                return _smooth_max(_corr(x), a) + 300.0 * (np.sum(x) - target) ** 2
            res = minimize(obj2, best_h, method='SLSQP', bounds=bounds,
                           options={'maxiter': max_iter, 'ftol': 1e-16, 'iprint': 0})
            hc = _repair(res.x, n)
            c5 = _c5(hc, dx)
            if c5 < best_c:
                best_h, best_c = hc, c5
        except Exception:
            pass
    return best_h, best_c

def _corr_grad(h, m_idx):
    n = len(h)
    k = m_idx - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n - k])
        g[:n - k] -= h[k:]
    else:
        j = -k
        g[:n - j] += (1.0 - h[j:])
        g[j:] -= h[:n - j]
    return g

def _local_search(h, dx, n, tleft, rng, max_iter=25000):
    best_h = h.copy()
    best_c = _c5(best_h, dx)
    no_improve = 0
    for it in range(max_iter):
        if tleft() < 8:
            break
        c_arr = _corr(best_h)
        top_k = min(3, len(c_arr))
        active = np.argsort(c_arr)[::-1][:top_k]
        best_imp = 1e-14
        best_mv = None
        for m in active:
            g = _corr_grad(best_h, m)
            src = np.where(g > 0.005)[0]
            dst = np.where(g < -0.005)[0]
            if len(src) == 0 or len(dst) == 0:
                continue
            si_list = src[rng.choice(len(src), size=min(15, len(src)), replace=False)]
            di_list = dst[rng.choice(len(dst), size=min(15, len(dst)), replace=False)]
            for si in si_list:
                for di in di_list:
                    dg = g[si] - g[di]
                    if dg <= 0:
                        continue
                    d_max = min(1.0 - best_h[di], best_h[si])
                    d_try = min(d_max, 0.04)
                    imp = dg * d_try
                    if imp > best_imp:
                        best_imp = imp
                        best_mv = (di, si, d_try)
        if best_mv is None:
            i = rng.randint(n)
            j = (i + 1 + rng.randint(n - 1)) % n
            d = 0.015 * (1.0 if best_h[i] < 0.5 else -1.0)
            ni, nj = best_h[i] + d, best_h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                best_h[i], best_h[j] = ni, nj
                best_h = _repair(best_h, n)
                nc = _c5(best_h, dx)
                if nc < best_c:
                    best_c = nc
                    no_improve = 0
                else:
                    best_h[i], best_h[j] = nj, ni
                    best_h = _repair(best_h, n)
                    no_improve += 1
            else:
                no_improve += 1
        else:
            di, si, d = best_mv
            best_h[di] += d
            best_h[si] -= d
            best_h = _repair(best_h, n)
            nc = _c5(best_h, dx)
            if nc < best_c:
                best_c = nc
                no_improve = 0
            else:
                best_h[di] -= d
                best_h[si] += d
                best_h = _repair(best_h, n)
                no_improve += 1
        if no_improve > 600:
            break
    return best_h, best_c

def _anneal(h, dx, n, tleft, rng, max_iter=40000):
    best_h = h.copy()
    best_c = _c5(best_h, dx)
    temp = 0.04
    no_imp = 0
    for it in range(max_iter):
        if tleft() < 8:
            break
        i = rng.randint(n)
        j = (i + 1 + rng.randint(n - 1)) % n
        d = (rng.rand() - 0.5) * 0.08
        ni, nj = best_h[i] + d, best_h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        best_h[i], best_h[j] = ni, nj
        best_h = _repair(best_h, n)
        nc = _c5(best_h, dx)
        if nc < best_c or rng.rand() < np.exp(-(nc - best_c) / temp):
            best_c = min(best_c, nc)
            no_imp = 0
        else:
            best_h[i], best_h[j] = nj, ni
            best_h = _repair(best_h, n)
            no_imp += 1
        temp = max(0.0005, temp * 0.99996)
        if no_imp > 400:
            no_imp = 0
            temp = min(0.04, temp * 1.5)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - (time.time() - t0)

    n0 = len(initial_h_values)
    dx0 = 2.0 / n0
    h_init = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
    best_h = h_init.copy()
    best_c = _c5(best_h, dx0)

    # Phase 1: SLSQP at n0
    if tleft() > 250:
        h, c = _slsqp_opt(h_init, dx0, n0, tleft, max_iter=500)
        if c < best_c:
            best_h, best_c = h, c

    # Phase 2: Local search at n0
    if tleft() > 200:
        h, c = _local_search(h_init, dx0, n0, tleft, rng, max_iter=20000)
        if c < best_c:
            best_h, best_c = h, c

    # Phase 3: Anneal at n0
    if tleft() > 150:
        h, c = _anneal(h_init, dx0, n0, tleft, rng, max_iter=25000)
        if c < best_c:
            best_h, best_c = h, c

    # Phase 4: Multi-resolution refinement
    for n_f in [n0 * 2, n0 * 3, n0 * 5]:
        if tleft() < 80:
            break
        dx_f = 2.0 / n_f
        h_f = _repair(_upsample(best_h, n_f), n_f)
        if tleft() > 70:
            h_f, c_f = _slsqp_opt(h_f, dx_f, n_f, tleft, max_iter=300)
            if c_f < best_c:
                best_h, best_c = h_f, c_f
        if tleft() > 40:
            h_f, c_f = _local_search(h_f, dx_f, n_f, tleft, rng, max_iter=15000)
            if c_f < best_c:
                best_h, best_c = h_f, c_f
        if tleft() > 25:
            h_f, c_f = _anneal(h_f, dx_f, n_f, tleft, rng, max_iter=15000)
            if c_f < best_c:
                best_h, best_c = h_f, c_f

    # Phase 5: Structured restarts
    if tleft() > 40:
        for n_s in [n0, n0 * 2]:
            if tleft() < 25:
                break
            dx_s = 2.0 / n_s
            p = max(3, n_s // 10)
            h_s = np.zeros(n_s)
            for i in range(n_s):
                phase = (i % p) / p
                h_s[i] = 1.0 - (phase if phase < 0.5 else (1.0 - phase) * 2.0)
                h_s[i] = max(0.0, min(1.0, h_s[i] * 1.5))
            h_s = _repair(h_s, n_s)
            h_s, c_s = _local_search(h_s, dx_s, n_s, tleft, rng, max_iter=8000)
            if c_s < best_c:
                best_h, best_c = h_s, c_s
            h_s2 = _repair(_upsample(best_h, n_s), n_s)
            h_s2, c_s2 = _local_search(h_s2, dx_s, n_s, tleft, rng, max_iter=8000)
            if c_s2 < best_c:
                best_h, best_c = h_s2, c_s2

    # Phase 6: Final SLSQP polish
    if tleft() > 30:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        h_b, c_b = _slsqp_opt(best_h, dx_b, n_b, tleft, max_iter=300)
        if c_b < best_c:
            best_h, best_c = h_b, c_b

    # Phase 7: Final local polish
    if tleft() > 15:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        h_b, c_b = _local_search(best_h, dx_b, n_b, tleft, rng, max_iter=10000)
        if c_b < best_c:
            best_h, best_c = h_b, c_b

    best_h = _repair(best_h, len(best_h))
    best_c = _c5(best_h, 2.0 / len(best_h))
    return (best_h, best_c, len(best_h))
