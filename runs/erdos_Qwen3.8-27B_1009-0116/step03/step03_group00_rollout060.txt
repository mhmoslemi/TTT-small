    if s < 1e-12:

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5v(h, n):
    return float(np.max(_corr(h))) * 2.0 / n

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= t / s
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-14:
        ord_ = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in ord_:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-14: break
    return h

def _obj(x, n, al):
    c = _corr(x) * 2.0 / n
    m = c.max()
    s = np.clip(al * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / al

def _jac(x, n, al):
    dx = 2.0 / n
    c = _corr(x) * dx
    m = c.max()
    s = np.clip(al * (c - m), -400, 400)
    e = np.exp(s); w = e / e.sum()
    g = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-12: continue
        lag = k - (n-1)
        if lag >= 0:
            u = n - lag
            g[:u] += w[k] * (1.0 - x[lag:lag+u])
            g[lag:lag+u] -= w[k] * x[:u]
        else:
            u = n + lag; l2 = -lag
            g[l2:l2+u] += w[k] * (1.0 - x[:u])
            g[:u] -= w[k] * x[l2:l2+u]
    return g * dx

def _slsqp(h0, n, al, mi):
    bnd = [(0.0,1.0)]*n
    con = [{'type':'eq','fun':lambda x: x.sum()-n/2.0,'jac':lambda x: np.ones(n)}]
    try:
        r = minimize(_obj, h0, jac=_jac, args=(n,al), method='SLSQP',
                     bounds=bnd, constraints=con, options={'maxiter':mi,'ftol':1e-16})
        return _repair(r.x, n), _c5v(_repair(r.x,n), n)
    except: return h0.copy(), _c5v(h0,n)

def _pair_grad(h, n, i, j):
    """Compute slope vector s_k = dc_k/dh[i] - dc_k/dh[j]"""
    dx = 2.0/n
    s = np.zeros(2*n-1)
    for k in range(2*n-1):
        lag = k - (n-1)
        def ga(t):
            if lag==0: return 1.0-2.0*h[t]
            if lag>0:
                v=0.0
                if t < n-lag: v += 1.0-h[t+lag]
                if t>=lag: v -= h[t-lag]
                return v
            l2=-lag; v=0.0
            if t>=l2: v += 1.0-h[t-l2]
            if t<n-l2: v -= h[t+l2]
            return v
        s[k] = (ga(i)-ga(j))*dx
    return s

def _pair_search(h, n, i, j):
    """Find best d for h[i]+=d, h[j]-=d minimizing max c_k. Returns best h, c5."""
    dx = 2.0/n
    c0 = _corr(h)*dx
    s = _pair_grad(h, n, i, j)
    dmin = max(-h[i], h[j]-1.0)
    dmax = min(1.0-h[i], h[j])
    if dmax - dmin < 1e-12: return h.copy(), _c5v(h,n)
    # Golden section search on [dmin, dmax]
    gr = (np.sqrt(5)-1)/2
    a, b = dmin, dmax
    c1 = b - gr*(b-a); c2 = a + gr*(b-a)
    h[i] += c1; h[j] -= c1
    f1 = _c5v(h, n)
    h[i] -= c1; h[j] += c1
    h[i] += c2; h[j] -= c2
    f2 = _c5v(h, n)
    h[i] -= c2; h[j] += c2
    for _ in range(30):
        if f1 < f2:
            b = c2; c2 = c1; f2 = f1
            c1 = b - gr*(b-a)
            h[i] += c1; h[j] -= c1; f1 = _c5v(h,n); h[i]-=c1; h[j]+=c1
        else:
            a = c1; c1 = c2; f1 = f2
            c2 = a + gr*(b-a)
            h[i] += c2; h[j] -= c2; f2 = _c5v(h,n); h[i]-=c2; h[j]+=c2
    d = (a+b)/2
    return d, min(f1,f2)

def _grad_search(h, n, budget_end, rng, topk=10):
    """Gradient-guided pair descent."""
    best_c5 = _c5v(h, n)
    dx = 2.0/n
    stagnant = 0

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5v(h, n):
    return float(np.max(_corr(h))) * 2.0 / n

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    t = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= t / s
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-14:
        ord_ = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in ord_:
            if d > 0:
                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a
            if abs(d) < 1e-14:
                break
    return h

def _obj_f(x, n, al):
    c = _corr(x) * 2.0 / n
    m = c.max()
    s = np.clip(al * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / al

def _jac_f(x, n, al):
    dx = 2.0 / n
    c = _corr(x) * dx
    m = c.max()
    s = np.clip(al * (c - m), -400, 400)
    e = np.exp(s); w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            u = n - lag
            g[:u] += w[k] * (1.0 - x[lag:lag + u])
            g[lag:lag + u] -= w[k] * x[:u]
        else:
            u = n + lag
            g[-lag:-lag + u] += w[k] * (1.0 - x[:u])
            g[:u] -= w[k] * x[-lag:-lag + u]
    return g * dx

def _slsqp(h0, n, al, mi):
    bnd = [(0.0, 1.0)] * n
    con = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0, 'jac': lambda x: np.ones(n)}]
    try:

                     bounds=bnd, constraints=con, options={'maxiter': mi, 'ftol': 1e-16})
        hc = _repair(r.x, n)
        return hc, _c5v(hc, n)
    except Exception:
        return h0.copy(), _c5v(h0, n)

def _subgrad(h, n, topk=1):
    c = _corr(h) * 2.0 / n
    idx = np.argsort(c)[::-1][:topk]
    g = np.zeros(n)
    for k in idx:
        lag = k - (n - 1)
        if lag >= 0:
            u = n - lag
            g[:u] += (1.0 - h[lag:lag + u])
            g[lag:lag + u] -= h[:u]
        else:
            u = n + lag
            g[-lag:-lag + u] += (1.0 - h[:u])
            g[:u] -= h[-lag:-lag + u]
    return g

def _pair_opt(h, n, i, j):
