def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _project(x, target):
    n = len(x)
    x = x.copy()
    lo = float(x.min()) - 2.0
    hi = float(x.max()) + 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def _grad_smooth(h, alpha):
    n = len(h)
    c = _corr(h)
    mx = float(c.max())
    shifted = np.clip(alpha * (c - mx), -500.0, 500.0)
    ew = np.exp(shifted)
    w = ew / float(ew.sum())
    nc = len(c)
    d = np.arange(nc) - (n - 1)
    g = np.zeros(n)
    for j in range(n):
        s = 0.0
        for m in range(nc):
            dm = d[m]
            ct = 1.0
            jp = j + dm
            if 0 <= jp < n:
                ct -= h[jp]
            jm = j - dm
            if 0 <= jm < n:
                ct -= h[jm]
            s += w[m] * ct
        g[j] = s
    return g

def _smooth_val(h, alpha):
    c = _corr(h)
    mx = float(c.max())
    shifted = np.clip(alpha * (c - mx), -500.0, 500.0)
    return mx + float(np.log(np.sum(np.exp(shifted)))) / alpha

                       constraints=[{'type': 'eq', 'fun': _eq}],

        return res.x

def _grad_smooth(h, alpha):
    n = len(h)
    c = _corr(h)
    mx = float(c.max())
    shifted = np.clip(alpha * (c - mx), -500.0, 500.0)
    ew = np.exp(shifted)
    w = ew / float(ew.sum())
    g = np.zeros(n)
    for m in range(2 * n - 1):
        d = m - (n - 1)
        wm = w[m]
        if wm < 1e-15:
            continue
        # First term: add wm to g[j] for j in [max(0,d), n-1-max(0,-d)]
        jlo = max(0, d)
        jhi = n - 1 - max(0, -d)
        if jlo <= jhi:
            g[jlo:jhi+1] += wm
        # Second term: -wm * h[j+d] for j where 0<=j+d<n
        # j+d in [0,n) means j in [-d, n-1-d], intersected with [0,n-1]
        # j in [max(0,-d), min(n-1, n-1-d)]
        jlo2 = max(0, -d)
        jhi2 = min(n - 1, n - 1 - d)
        if jlo2 <= jhi2:

        # Third term: -wm * h[j-d] for j where 0<=j-d<n

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _project(x, target):
    n = len(x)
    lo = float(x.min()) - 2.0
    hi = float(x.max()) + 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def _grad_smooth(h, alpha):
    n = len(h)
    c = _corr(h)
    mx = float(c.max())
    shifted = np.clip(alpha * (c - mx), -500.0, 500.0)
    ew = np.exp(shifted)
    w = ew / float(ew.sum())
    g = np.zeros(n)
    for m in range(2 * n - 1):
        d = m - (n - 1)
        wm = w[m]
        if wm < 1e-15:
            continue
        jlo = max(0, d)
        jhi = n - 1 - max(0, -d)
        if jlo <= jhi:
            g[jlo:jhi + 1] += wm
        jlo2 = max(0, -d)
        jhi2 = min(n - 1, n - 1 - d)
        if jlo2 <= jhi2:
            sl = max(0, d)
            en = sl + (jhi2 - jlo2 + 1)
            g[jlo2:jhi2 + 1] -= wm * h[sl:en]
        jlo3 = max(0, d)
        jhi3 = min(n - 1, n + d - 1)
        if jlo3 <= jhi3:
            st = max(0, -d)
            en = st + (jhi3 - jlo3 + 1)
            g[jlo3:jhi3 + 1] -= wm * h[st:en]
    return g

def _smooth_val(h, alpha):
    c = _corr(h)
    mx = float(c.max())
    shifted = np.clip(alpha * (c - mx), -500.0, 500.0)
    return mx + float(np.log(np.sum(np.exp(shifted)))) / alpha

def _slsqp_eq(x, target):
    return float(np.sum(x)) - target

def _slsqp_call(x, bounds, target, alpha, maxit):
    try:
        res = minimize(_smooth_val, x, method='SLSQP',
                       jac=_grad_smooth, args=(alpha,),
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _slsqp_eq, 'args': (target,)}],
                       options={'maxiter': maxit, 'ftol': 1e-15})
        return res.x
    except Exception:
        return x

def _nesterov_phase(h_start, n, target, alpha, T, budget_limit, rng):
    h = h_start.copy()
    h_prev = h.copy()
    c5_cur = _c5(h, 2.0 / n)
    for t in range(T):
        if time.time() > budget_limit:
            break
        if t == 0:
            y = h.copy()
        else:
            mu = min(0.9, (t + 1) / (t + 11.0))
            y = h + mu * (h - h_prev)
        y = _project(y, target)
        grad = _grad_smooth(y, alpha)
        step = 0.08 / (1.0 + 0.015 * t)
        h_new = _project(y - step * grad, target)
        c5_new = _c5(h_new, 2.0 / n)
        if c5_new > c5_cur + 1e-12:
            step *= 0.5
            h_new = _project(y - step * grad, target)
            c5_new = _c5(h_new, 2.0 / n)
        if c5_new > c5_cur + 1e-12:
            step *= 0.5
            h_new = _project(y - step * grad, target)
            c5_new = _c5(h_new, 2.0 / n)
        if c5_new <= c5_cur + 1e-14:
            h_prev = h
            h = h_new
            c5_cur = c5_new
        else:
            h_prev = h
        if t % 50 == 0 and rng.rand() < 0.3:
            h = _project(h + 0.005 * rng.randn(n), target)
            c5_cur = _c5(h, 2.0 / n)
    return h, c5_cur
