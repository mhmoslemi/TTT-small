    return float(c.max() * 2.0 / n)

    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()

    active = np.where(c >= m - 1e-12)[0]
    w = np.zeros(2*n - 1)
    w[active] = 1.0 / len(active)
    g = np.zeros(n)
    for k in active:
        lag = k - (n - 1)
        if lag >= 0:
            g[:n-lag] += (1.0 / len(active)) * (1.0 - h[lag:n])
            g[lag:n] += -(1.0 / len(active)) * h[:n-lag]
        else:
            L = -lag
            g[:n-L] += -(1.0 / len(active)) * h[L:n]
            g[L:n] += (1.0 / len(active)) * (1.0 - h[:n-L])
    return m * 2.0 / n, g * 2.0 / n

    h = np.clip(h, 0.0, 1.0).copy()

    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # Distribute the difference
    if diff > 0:
        # Need to increase: give to elements with most room
        room = 1.0 - h
        idx = np.argsort(-room)
        for i in idx:
            add = min(diff, room[i])
            h[i] += add
            diff -= add
            if diff < 1e-12:

            h[i] -= sub
            diff += sub

    return (m + np.log(np.sum(np.exp(s))) / _SLSQP_ALPHA) * 2.0 / n

def _slsqp_grad(x):
    n = _SC_N
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(_SLSQP_ALPHA * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            g[:n-lag] += w[k] * (1.0 - x[lag:n])
            g[lag:n] += -w[k] * x[:n-lag]
        else:
            L = -lag
            g[:n-L] += -w[k] * x[L:n]
            g[L:] += w[k] * (1.0 - x[:n-L])
    return g * 2.0 / n

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * 2.0 / n)

def _subgrad(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    active = np.where(np.abs(c - m) < 1e-10)[0]
    g = np.zeros(n)
    inv = 2.0 / (n * len(active))
    for k in active:
        lag = k - (n - 1)
        if lag >= 0:
            g[:n-lag] += inv * (1.0 - h[lag:n])
            g[lag:n] += -inv * h[:n-lag]
        else:
            L = -lag
            g[:n-L] += -inv * h[L:n]
            g[L:] += inv * (1.0 - h[:n-L])
    return g

def _psgd(h, n, rng, t_end):
    h = h.copy()
    c5c = _eval(h)
    step = 0.004
    stall = 0
    for _ in range(800000):
        if time.time() > t_end:
            break
        g = _subgrad(h)
        h_new = _repair(h - step * g, n)
        c5n = _eval(h_new)
        if c5n < c5c - 1e-15:
            h = h_new
            c5c = c5n
            step = min(step * 1.02, 0.03)
            stall = 0
        else:
            step *= 0.96
            stall += 1
            if stall > 150:
                h = _repair(h + 0.004 * rng.randn(n), n)
                c5c = _eval(h)
                step = 0.004
                stall = 0
    return h, c5c

def _slsqp(h, n, alpha, iters):
    global _SC_N, _SLSQP_ALPHA
    _SC_N = n
    _SLSQP_ALPHA = alpha
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sc_con_fun, 'jac': _sc_con_jac}]
    try:
        res = minimize(_slsqp_obj, h.copy(), jac=_slsqp_grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': iters, 'ftol': 1e-20})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _local_search(h, n, rng, t_end):
    h = h.copy()
    c5c = _eval(h)
    scale = 0.008
    stall = 0
    for _ in range(500000):
        if time.time() > t_end:
            break
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5c - 1e-15:
            c5c = c5n
            scale = min(scale * 1.03, 0.015)
            stall = 0
        else:
            h[i], h[j] = ni - d, nj + d
            scale *= 0.99
            stall += 1
            if stall > 300:
                scale = max(scale * 0.5, 1e-5)
                stall = 0
                if scale < 1e-7:
                    break
    return h, c5c

def _resample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old)
    h_new = np.interp(np.linspace(0, 2, n_new), x_old, h)
    return _repair(h_new, n_new)

def _pair_local(h, n, t_end):
    h = h.copy()
    c5c = _eval(h)
    steps = np.array([0.01, -0.01, 0.03, -0.03, 0.005, -0.005])
    for _pass in range(2000):
        if time.time() > t_end:
            break
        imp = False
        for i in range(n):
            if time.time() > t_end:
                break
            for j in range(i + 1, n):
                for d in steps:
                    ni, nj = h[i] + d, h[j] - d
                    if ni < -1e-12 or ni > 1 + 1e-12 or nj < -1e-12 or nj > 1 + 1e-12:
                        continue
                    h[i], h[j] = ni, nj
                    c5n = _eval(h)
                    if c5n < c5c - 1e-15:
                        c5c = c5n
                        imp = True
                    else:
                        h[i], h[j] = h[i] - d, h[j] + d
        if not imp:
            break
    return h, c5c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = 1.0
    best_n = 0

    def _update(h, c5):
        nonlocal best_h, best_c5, best_n
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = len(h)

    # Phase 1: Optimize initial (n=70)
    if 'initial_h_values' in dir() and initial_h_values is not None:
        n0 = len(initial_h_values)
        h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n0)
        _update(h0, _eval(h0))
        for alpha in [1000, 10000, 100000]:
            if time.time() - t0 > budget_s - 400:
                break
            h1 = _slsqp(best_h, best_n, alpha, 1500)
            _update(h1, _eval(h1))
        h1, c5_1 = _psgd(best_h, best_n, rng, t0 + 40)
        _update(h1, c5_1)
        h1, c5_1 = _local_search(best_h, best_n, rng, t0 + 80)
        _update(h1, c5_1)
        h1, c5_1 = _pair_local(best_h, best_n, t0 + 130)
        _update(h1, c5_1)

    # Phase 2: Try other n values
    n_list = [42, 50, 56, 63, 77, 84, 90, 100, 120, 140, 175, 200]
    n_list = [nn for nn in n_list if abs(nn - best_n) > 10]
    remaining = budget_s - 60 - (time.time() - t0)
    per_n = max(30, min(90, remaining / max(1, len(n_list))))

    for n in n_list:
        if time.time() - t0 > budget_s - 60:
            break
        t_n_end = time.time() + per_n
        # Multiple starts
        starts = []
        if best_h is not None and len(best_h) > 2:
            starts.append(_resample(best_h, n))
        starts.append(_repair(np.full(n, 0.5) + 0.15 * np.sin(2 * np.pi * np.arange(n) * 2.0 / n), n))
        starts.append(_repair(np.full(n, 0.5) + 0.2 * np.cos(2 * np.pi * np.arange(n) * 3.0 / n), n))
        starts.append(_repair(0.5 + 0.3 * rng.randn(n), n))

        for hs in starts:
            if time.time() > t_n_end - 10:
                break
            if n <= 150:
                h_opt = _slsqp(hs, n, 5000, 800)
            else:
                h_opt = hs
            _update(h_opt, _eval(h_opt))
            h_ps, c5_ps = _psgd(h_opt, n, rng, t_n_end - 15)
            _update(h_ps, c5_ps)
            h_ls, c5_ls = _local_search(h_ps, n, rng, t_n_end)
            _update(h_ls, c5_ls)

    # Phase 3: Final polish
    if time.time() - t0 < budget_s - 30 and best_h is not None:
        h_fin = _slsqp(best_h, best_n, 200000, 2000)
        _update(h_fin, _eval(h_fin))
        h_fin, c5_fin = _local_search(best_h, best_n, rng, t0 + budget_s - 10)
        _update(h_fin, c5_fin)
        h_fin, c5_fin = _pair_local(best_h, best_n, t0 + budget_s - 5)
        _update(h_fin, c5_fin)

    best_h = _repair(best_h, best_n)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, best_n)
