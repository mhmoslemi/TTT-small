import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-12:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / len(h))

def _smooth_max(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_jac(x, alpha):

            continue

        if L >= 0:

            i2 = np.arange(L, n)
            if len(i1) > 0:
                g[i1] += wl * (1.0 - x[i2])
            if len(i2) > 0:

                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _subgrad(h, n, steps, lr0, alpha):
    h = h.copy()
    c5 = _eval(h)
    lr = lr0
    avg_h = h.copy()
    for t in range(1, steps + 1):
        g = _smooth_jac(h, alpha)
        g -= g.mean()
        hp = h - lr * g
        hp = np.clip(hp, 0.0, 1.0)
        hp = _repair(hp, n)
        c5p = _eval(hp)
        if c5p < c5:
            h = hp
            c5 = c5p
            lr = min(lr * 1.3, 0.05)
        else:
            lr *= 0.7
            if lr < 1e-12:
                break
        avg_h += (1.0 / (t + 1)) * (h - avg_h)
    return _repair(avg_h, n)

    h = np.zeros(n)
    ones_needed = n // 2
    per = max(2, n // ones_needed * 2)
    half = per // 2
    for i in range(n):

            h[i] = 1.0
    h = _repair(h, n)
    return h

def _local_search(h, n, steps, base_scale, rng):
