import numpy as np
from scipy.optimize import minimize
import time

_st = {}

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(5000):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        d = target - s
        if abs(d) < 1e-14:
            break
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-14:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
    return h

def softmax_val(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def softmax_custom(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    alpha = _st['alpha']
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    val = m + np.log(np.sum(np.exp(s))) / alpha
    tk = _st['topk']
    if tk > 1:
        sc = np.sort(corr)[::-1][:tk]
        val += _st['fw'] * np.sum((sc - sc.mean())**2)
    return val

def grad_corr(h, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    alpha = _st['alpha']
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    e = np.exp(s)
    w = e / np.sum(e)
    shifts = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)
    for mi, sh in enumerate(shifts):
        wi = w[mi]
        if wi < 1e-16:
            continue
        if sh >= 0:
            i1 = np.arange(0, n - sh)
            grad[i1] += wi * (1.0 - h[i1 + sh])
            i2 = np.arange(sh, n)
            grad[i2] -= wi * h[i2 - sh]
        else:
            s2 = -sh
            i1 = np.arange(s2, n)
            grad[i1] += wi * (1.0 - h[i1 - s2])
            i2 = np.arange(0, n - s2)
            grad[i2] -= wi * h[i2 + s2]
    return grad

def slsqp_run(h, n, alpha, topk, fw, maxiter):
    _st['alpha'] = alpha
    _st['topk'] = topk
    _st['fw'] = fw
    _st['target'] = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def _obj(x):
        return softmax_custom(x)

    def _con(x):
        return x.sum() - _st['target']

    cons = [{'type': 'eq', 'fun': _con}]
    try:
        res = minimize(_obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return repair(res.x, n)
    except Exception:
        return repair(h, n)

def subgrad_step(h, n, alpha, lr, n_iter, t_end):

    h = h.copy()
    best_h = h.copy()

    target = n / 2.0
    v = np.zeros(n)

        g = grad_corr(h, n)
        lr_t = lr / (1.0 + 0.005 * it)
        v = mu * v - lr_t * g

        s = h_new.sum()

                h_new += d * room / room.sum()
            else:
                room = np.maximum(h_new, 1e-12)
                h_new += d * room / room.sum()
            h_new = np.clip(h_new, 0.0, 1.0)
        h_new = repair(h_new, n)
        h = h_new
        if it % 50 == 0:
            c5 = compute_c5(h, n)

                best_h = h.copy()
    c5 = compute_c5(h, n)

def sys_cd(h, n, max_rounds, t_end):

    best = compute_c5(h, n)
    for r in range(max_rounds):
        if time.time() > t_end:
            break
        improved = False
        for i in range(n):
            if time.time() > t_end:
                break
            hi = h[i]
            if hi < 0.005 or hi > 0.995:
                continue
            for j in range(i + 1, n):
                hj = h[j]
                if hj < 0.005 or hj > 0.995:
                    continue
                for amt in [0.04, 0.02, 0.01, 0.005]:
                    a1 = min(amt, 1.0 - hi, hj)
                    a2 = min(amt, hi, 1.0 - hj)
                    for d_i, d_j in [(a1, -a1), (-a1, a1), (a2, -a2), (-a2, a2)]:
                        if abs(d_i) < 1e-7:
                            continue
                        ni, nj = hi + d_i, hj + d_j
                        if ni < -1e-10 or ni > 1 + 1e-10 or nj < -1e-10 or nj > 1 + 1e-10:
                            continue
                        h[i], h[j] = ni, nj
                        c5 = compute_c5(h, n)
                        if c5 < best - 1e-14:
                            best = c5
                            improved = True
                            hi = h[i]
                            hj = h[j]
                            break
                        h[i], h[j] = hi, hj
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return h, best

def binary_snap(h, n, thresh):
    h2 = h.copy()
    h2[h2 < thresh] = 0.0
    h2[h2 > 1.0 - thresh] = 1.0
    return repair(h2, n)

        for i in range(0, n, 2 * bl):
            h[i:i + bl] = 1.0
        h = 0.5 * h + 0.2 * rng.randn(n)
    else:
        h = rng.uniform(0, 1, n)
    return repair(h, n)
