def _binary_sa(h, n, rng, deadline, T0=2.0, T_min=0.01, cool=0.9999):
    """Simulated annealing on binary vectors with exactly n/2 ones."""
    h = h.astype(int).copy()  # 0 or 1
    ones_idx = np.where(h == 1)[0]
    zeros_idx = np.where(h == 0)[0]
    
    corr = _binary_corr(h)
    best_corr = corr.max()
    best_h = h.copy()
    cur_corr = best_corr
    
    T = T0
    T_min = 0.01
    cool_rate = 0.99995
    
    while time.time() < deadline and T > T_min:
        # Random swap: pick random one and random zero
        i = ones_idx[rng.randint(len(ones_idx))]
        j = zeros_idx[rng.randint(len(zeros_idx))]
        
        h[i], h[j] = h[j], h[i]
        corr_new = _binary_corr(h)
        new_corr = corr_new.max()
        
        delta = new_corr - cur_corr
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            cur_corr = new_corr
            # Update tracked indices
            ones_idx[j] = i  # j is now 1
            zeros_idx[i] = j  # i is now 0
            if cur_corr < best_corr:
                best_corr = cur_corr
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]  # undo
        
        T *= cool_rate
    
    return best_h, best_corr * 2.0 / n

For a and v both of length N, k goes from -(N-1) to N-1.

So for our case, a = h, v = 1-h, both length 70.
corr[k] = ∑_{n: 0≤n≤69, 0≤n+k≤69} h_{n+k} * (1-h_n)

For k=0: ∑ h_n * (1-h_n) = 0 for binary
For k=1: ∑_{n=0}^{68} h_{n+1} * (1-h_n) = count of positions where h goes 0→1 at consecutive positions
For k=-1: ∑_{n=1}^{69} h_{n-1} * (1-h_n) = count of positions where h goes 1→0 at consecutive positions

In mode='full', the output has length 2N-1 = 139.
output[0] = corr[-(N-1)] = corr[-69]
output[N-1] = corr[0] = corr[0] = output[69]
output[2N-2] = corr[N-1] = corr[69] = output[138]

So output[69+j] = corr[j] for j = -69, ..., 69.
Or: output[i] = corr[i-69] for i = 0, ..., 138.

For the maximum, we take max over all 139 values. For binary, the center (i=69) is 0, so the max is at some other lag.

corr[k] = ∑ h_{n+k} * (1-h_n)

corr = np.correlate(h_int, 1 - h_int, mode='full')  # integer correlation

import numpy as np
from scipy.optimize import minimize
import time

_G_ALPHA = 500.0
_G_TARGET = 35.0
_G_PEN = 100.0

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _smooth_max(x, alpha):
    c = _corr(x)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _obj_fn(x):
    return _smooth_max(x, _G_ALPHA) + _G_PEN * (np.sum(x) - _G_TARGET) ** 2

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n / 2.0
    d = tgt - h.sum()
    if abs(d) < 1e-12:
        return h
    order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
    rem = d
    for i in order:
        if rem > 0:
            a = min(rem, 1.0 - h[i])
            h[i] += a
            rem -= a
        else:
            s = min(-rem, h[i])
            h[i] -= s
            rem += s
        if abs(rem) < 1e-12:
            break
    return h

def _lbfgs_run(h, n, dx, alpha, maxiter=300):
    global _G_ALPHA, _G_TARGET, _G_PEN
    _G_TARGET = n / 2.0
    _G_ALPHA = alpha
    best_h, best_c = h.copy(), _c5(h, dx)
    bounds = [(0.0, 1.0)] * n
    for pen in [50.0, 500.0, 5000.0]:
        _G_PEN = pen
        try:
            r = minimize(_obj_fn, best_h, method='L-BFGS-B', bounds=bounds,
                        options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-6})
            hc = _repair(r.x, n)
            c = _c5(hc, dx)
            if c < best_c:
                best_h, best_c = hc, c
        except Exception:
            pass
    return best_h, best_c

def _slsqp_run(h, n, dx, alpha, maxiter=400):
    global _G_ALPHA, _G_TARGET, _G_PEN
    _G_TARGET = n / 2.0
    _G_ALPHA = alpha
    _G_PEN = 1e6
    best_h, best_c = h.copy(), _c5(h, dx)
    bounds = [(0.0, 1.0)] * n
    try:
        r = minimize(_obj_fn, best_h, method='SLSQP', bounds=bounds,
                    options={'maxiter': maxiter, 'ftol': 1e-14})
        hc = _repair(r.x, n)
        c = _c5(hc, dx)
        if c < best_c:
            best_h, best_c = hc, c
    except Exception:
        pass
    return best_h, best_c

def _bmax(h_int):
    hf = h_int.astype(float)
    ci = np.correlate(h_int, 1 - h_int, mode='full')
    return int(ci.max())

def _fix_count(h, n, rng):
    h = h.copy()
    target = n // 2
    while h.sum() > target:
        i = rng.choice(np.where(h == 1)[0])
        h[i] = 0
    while h.sum() < target:
        i = rng.choice(np.where(h == 0)[0])
        h[i] = 1
    return h

def _binary_sa(h_init, n, rng, deadline):
    h = _fix_count((h_init > 0.5).astype(int), n, rng)
    best_h = h.copy()
    best_m = _bmax(h)

        cur_m = _bmax(h)

            zi = np.where(h == 0)[0]
            i = oi[rng.randint(len(oi))]
            j = zi[rng.randint(len(zi))]
            h[i], h[j] = 1, 0
            nm = _bmax(h)
            d = nm - cur_m
            if d <= 0 or rng.random() < np.exp(-float(d) / T):

def _fine_local(h, n, dx, rng, deadline, n_tries=40000):
    best_c5 = _c5(h, dx)
    h = h.copy()
    scales = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    per = max(200, n_tries // len(scales))
    for sc in scales:
        for _ in range(per):
            if time.time() >= deadline:
                return h, best_c5
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = sc * (2.0 * rng.random() - 1.0)
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            c = _c5(h, dx)
            if c < best_c5 - 1e-14:
                best_c5 = c
            else:
                h[i], h[j] = h[i] - d, h[j] + d
    return h, best_c5

def _struct_init(n, rng, kind):
    h = np.zeros(n)
    if kind == 0:
        idx = rng.permutation(n)[:n // 2]
        h[idx] = 1.0
    elif kind == 1:
        h[::2][:n // 2] = 1.0
    elif kind == 2:
        x = np.linspace(0, 2, n, endpoint=False)
        h = (np.sin(2 * np.pi * 3.0 * x / 2.0) > 0).astype(float)
    elif kind == 3:
        x = np.linspace(0, 2, n, endpoint=False)
        h = ((np.sin(2 * np.pi * 5.0 * x / 2.0) + 0.3 * np.sin(2 * np.pi * 2.0 * x / 2.0)) > 0).astype(float)
    elif kind == 4:
        bl = n // 10
        for b in range(10):
            s = b * bl
            e = min(s + bl, n)
            h[s:e] = 1.0 if b % 2 == 0 else 0.0
    elif kind == 5:
        h = rng.random(n)
        thr = np.sort(h)[n // 4]
        h = (h >= thr).astype(float)
    elif kind == 6:
        x = np.linspace(0, 2, n, endpoint=False)
        h = (np.cos(2 * np.pi * 7.0 * x / 2.0) * np.cos(2 * np.pi * 3.0 * x / 2.0) > -0.1).astype(float)
    else:
        h = rng.random(n)
        h = (h > 0.5).astype(float)
    return _repair(h * (n // 2) / max(h.sum(), 1e-6), n)
