import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_corr_max(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')))

def repair_binary(h, n, target):
    """Repair h to be binary with exactly target ones."""
    h = np.round(h)  # Round to 0 or 1
    h = np.clip(h, 0, 1)
    current_sum = h.sum()
    diff = target - current_sum
    if diff > 0:
        # Need to add ones: pick zeros to flip
        zeros = np.where(h == 0)[0]
        np.random.shuffle(zeros)
        for i in zeros[:int(diff)]:
            h[i] = 1
    elif diff < 0:
        # Need to remove ones: pick ones to flip
        ones = np.where(h == 1)[0]
        np.random.shuffle(ones)
        for i in ones[:int(-diff)]:
            h[i] = 0
    return h

def smooth_corr_max(h, alpha):
    """Smooth approximation of max correlation."""
    c = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(c)
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def sa_binary(h, n, dx, rng, deadline, t0=0.5, t_min=0.001, cooling=0.9998):
    """Simulated annealing on binary space with exactly n/2 ones."""
    target = n // 2
    h = repair_binary(h.copy(), n, target)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    cur_c5 = best_c5
    T = t0
    ones_pos = np.where(h == 1)[0]
    zeros_pos = np.where(h == 0)[0]
    
    while time.time() < deadline and T > t_min:
        # Pick a random 1 and a random 0, swap them
        oi = rng.randint(len(ones_pos))
        zi = rng.randint(len(zeros_pos))
        i = ones_pos[oi]
        j = zeros_pos[zi]
        
        h[i] = 0
        h[j] = 1
        new_c5 = compute_c5(h, dx)
        
        if new_c5 < cur_c5 or rng.random() < np.exp(-(new_c5 - cur_c5) / T):
            cur_c5 = new_c5
            # Update pos arrays
            ones_pos[oi] = j
            zeros_pos[zi] = i
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] = 1
            h[j] = 0
        
        T *= cooling
    
    return best_h, best_c5

from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def repair_binary(h, rng, target):
    h = np.round(h).astype(float)
    n = len(h)
    diff = target - int(h.sum())
    if diff > 0:
        zeros = np.where(h < 0.5)[0]
        perm = rng.permutation(len(zeros))
        for idx in perm[:diff]:
            h[zeros[idx]] = 1.0
    elif diff < 0:
        ones = np.where(h >= 0.5)[0]
        perm = rng.permutation(len(ones))
        for idx in perm[:(-diff)]:
            h[ones[idx]] = 0.0
    return h

def repair_continuous(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s < 1e-12:
        return np.full(len(h), target / len(h))
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx]); h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx]); h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def smooth_max(c, alpha):
    m = np.max(c)
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _cont_obj(x, alpha):
    return smooth_max(np.correlate(x, 1.0 - x, mode='full'), alpha)

def sa_binary(h, n, dx, rng, deadline, t0=0.5, t_min=0.0005, cooling=0.9999):
    target = n // 2
    h = repair_binary(h.copy(), rng, target)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    cur_c5 = best_c5
    T = t0
    no_improve = 0
    while time.time() < deadline and T > t_min:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = h[j], h[i]
        new_c5 = compute_c5(h, dx)
        if new_c5 < cur_c5 or rng.random() < np.exp(-(new_c5 - cur_c5) / T):
            cur_c5 = new_c5
            if cur_c5 <= best_c5:
                best_c5 = cur_c5; best_h = h.copy(); no_improve = 0
            else:
                no_improve += 1
        else:
            h[i], h[j] = h[j], h[i]
            no_improve += 1
        T *= cooling
        if no_improve > 3000:
            T = max(T, 0.08)
            no_improve = 0
    return best_h, best_c5

def steepest_descent(h, n, dx, max_iters=50, deadline=None):
    h = h.copy()
    best_c5 = compute_c5(h, dx)
    for _ in range(max_iters):
        if deadline and time.time() >= deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_delta = 1e-12
        best_i, best_j = -1, -1
        for i in ones:
            for j in zeros:
                h[i], h[j] = 0.0, 1.0
                c5 = compute_c5(h, dx)
                delta = best_c5 - c5
                if delta > best_delta:
                    best_delta = delta; best_i, best_j = int(i), int(j)

def continuous_refine(h, n, dx, deadline):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    for alpha in [50, 200, 1000, 5000]:
        if time.time() >= deadline:
            break
        try:
            res = minimize(_cont_obj, best_h, args=(alpha,), method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': 300, 'ftol': 1e-15})
            hc = repair_continuous(res.x, target)
            c5 = compute_c5(hc, dx)
            if c5 < best_c5:
                best_h, best_c5 = hc, c5
        except Exception:
            pass
    return best_h, best_c5
