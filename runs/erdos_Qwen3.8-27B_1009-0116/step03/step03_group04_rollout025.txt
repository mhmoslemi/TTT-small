import numpy as np
import time

def _repair(h, n):
    """Clip to [0,1] and adjust sum to n/2."""
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-14:
                    break
    return h

def _c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _smooth_c5(h, dx, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -500.0, 500.0)
    return (m + np.sum(np.exp(s)) / alpha) * dx, c

def _grad_smooth(c, h, n, alpha):
    """Compute gradient of smooth max of correlations."""
    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -500.0, 500.0)
    w = np.exp(s)
    w /= w.sum()
    # w has shape (2n-1,), corresponds to shifts s = -(n-1) to (n-1)
    grad = np.zeros(n)
    for idx in range(2*n-1):
        si = w[idx]
        if si < 1e-10:
            continue
        # shift = idx - (n-1)
        shift = idx - (n + 1)  # WRONG
        # Actually shift = idx - (n-1)
        shift = idx - (n - 1)
        # ∂corr_shift/∂h[p] for all p
        if shift >= 0:
            # valid p: 0 <= p <= n-1-shift (for first term)
            # valid p: shift <= p <= n-1 (for second term)
            for p in range(n - shift):
                grad[p] += si * (1.0 - h[p + shift])
            for p in range(shift, n):
                grad[p] -= si * h[p - shift]
        else:
            # shift < 0, let sh = -shift > 0
            sh = -shift
            # valid p: 0 <= p-sh < n AND 0 <= p < n => p >= sh
            # First term: (1-h[p+shift]) = (1-h[p-sh]) for p in [sh, n-1]
            # Second term: -h[p-shift] = -h[p+sh] for p in [0, n-1-sh]
            for p in range(sh, n):
                grad[p] += si * (1.0 - h[p - sh])
            for p in range(n - sh):
                grad[p] -= si * h[p + sh]

import numpy as np

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-14:
                    break
    return h

def _c5(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')))

def _grad_smooth(c, h, n, alpha):
    m = float(np.max(c))
    sv = np.clip(alpha * (c - m), -500.0, 500.0)
    w = np.exp(sv)
    wsum = w.sum()
    if wsum < 1e-300:
        w = np.zeros(len(c)); w[np.argmax(c)] = 1.0
    else:
        w /= wsum
    grad = np.zeros(n)
    nm1 = n - 1

        si = w[idx]

            L = n - sh

                grad[sh:sh+L] += si * (1.0 - h[:L])
                grad[:L] -= si * h[sh:sh+L]
    return grad

    topk = np.argsort(c)[-k:]
    w = np.zeros(len(c))

    # Same computation as _grad_smooth but with this w

            if L > 0:

        else:
            sh = -shift
            L = n - sh
            if L > 0:
                grad[sh:sh+L] += si * (1.0 - h[:L])
                grad[:L] -= si * h[sh:sh+L]
    return grad

def _obj_alpha(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -500.0, 500.0)
    return m + np.sum(np.exp(s)) / alpha

def _eq_constraint(x, target):
    return np.sum(x) - target

def _do_slsqp(x0, n, target, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    res = minimize(_obj_alpha, x0, args=(alpha,), method='SLSQP',
                   bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': _eq_constraint, 'args': (target,)}],
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return res.x

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-14:
                    break
    return h

def _c5val(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')))

def _grad_smooth(c, h, n, alpha):
    m = float(np.max(c))
    sv = np.clip(alpha * (c - m), -500.0, 500.0)
    w = np.exp(sv)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(c)); w[np.argmax(c)] = 1.0
    else:
        w /= ws
    grad = np.zeros(n)
    nm1 = n - 1
    for idx in range(2 * n - 1):
        si = w[idx]
        if si < 1e-10:
            continue
        sh = idx - nm1
        if sh >= 0:
            L = n - sh
            if L > 0:
                grad[:L] += si * (1.0 - h[sh:sh+L])
                grad[sh:sh+L] -= si * h[:L]
        else:
            sh2 = -sh
            L = n - sh2
            if L > 0:
                grad[sh2:sh2+L] += si * (1.0 - h[:L])
                grad[:L] -= si * h[sh2:sh2+L]
    return grad

def _topk_grad(c, h, n, k):
    topk = np.argsort(c)[-k:]
    w = np.zeros(len(c))
    w[topk] = 1.0 / k
    grad = np.zeros(n)
    nm1 = n - 1
    for idx in range(2 * n - 1):
        si = w[idx]
        if si < 1e-10:
            continue
        sh = idx - nm1
        if sh >= 0:
            L = n - sh
            if L > 0:
                grad[:L] += si * (1.0 - h[sh:sh+L])
                grad[sh:sh+L] -= si * h[:L]
        else:
            sh2 = -sh
            L = n - sh2
            if L > 0:
                grad[sh2:sh2+L] += si * (1.0 - h[:L])
                grad[:L] -= si * h[sh2:sh2+L]
    return grad

def _obj_slsqp(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(np.max(c))
    s = np.clip(alpha * (c - m), -500.0, 500.0)
    return m + np.sum(np.exp(s)) / alpha

def _eq_fn(x, target):
    return np.sum(x) - target

def _slsqp(x0, n, target, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    res = minimize(_obj_slsqp, x0, args=(alpha,), method='SLSQP',
                   bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': _eq_fn, 'args': (target,)}],
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return res.x

def _gradient_phase(h, n, dx, alphas, iters_per_alpha, step0, t0, budget_s, margin, rng):
    best_h = h.copy()
    best_c5 = _c5val(h) * dx
    h = h.copy()
    for alpha in alphas:
        if time.time() > t0 + budget_s - margin:
            break
        step = step0
        mom = np.zeros(n)
        beta = 0.85
        for it in range(iters_per_alpha):
            if time.time() > t0 + budget_s - margin:
                break
            c = np.correlate(h, 1.0 - h, mode='full')
            c5_cur = float(np.max(c)) * dx
            grad = _grad_smooth(c, h, n, alpha)
            mom = beta * mom - step * grad
            h_new = _repair(h + mom, n)
            c5_new = _c5val(h_new) * dx
            if c5_new < c5_cur - 1e-16:
                h = h_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                step = min(step * 1.005, 0.04)
            else:
                step *= 0.65
                mom *= 0.3
    return best_h, best_c5

def _topk_phase(h, n, dx, topks, iters, t0, budget_s, margin):
    best_h = h.copy()
    best_c5 = _c5val(h) * dx
    h = h.copy()
    for k in topks:
        if time.time() > t0 + budget_s - margin:
            break
        step = 0.008
        for it in range(iters):
            if time.time() > t0 + budget_s - margin:
                break
            c = np.correlate(h, 1.0 - h, mode='full')
            c5_cur = float(np.max(c)) * dx
            grad = _topk_grad(c, h, n, k)
            h_new = _repair(h - step * grad, n)
            c5_new = _c5val(h_new) * dx
            if c5_new < c5_cur - 1e-15:
                h = h_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
            else:
                step *= 0.7
    return best_h, best_c5

def _pair_search(h, n, dx, max_passes, t0, budget_s, margin):
    best_h = h.copy()
    best_c5 = _c5val(h) * dx
    h = h.copy()
    for p in range(max_passes):
        if time.time() > t0 + budget_s - margin:
            break
        c5_cur = _c5val(h) * dx
        scale = 1.0 / (1.0 + 0.3 * p)
        deltas = np.array([0.004, 0.0015, -0.004, -0.0015]) * scale
        imp = False
        for i in range(n):
            if time.time() > t0 + budget_s - margin:
                break
            hi = h[i]
            for j in range(i + 1, n):
                hj = h[j]
                for d in deltas:
                    ni, nj = hi + d, hj - d
                    if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                        continue
                    h[i], h[j] = ni, nj
                    c5_try = _c5val(h) * dx
                    if c5_try < c5_cur - 1e-14:
                        c5_cur, hi, hj = c5_try, ni, nj
                        imp = True
                    else:
                        h[i], h[j] = hi, hj
                h[i], h[j] = hi, hj
            h[i] = hi
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h.copy()
        if not imp:
            break
    return best_h, best_c5

def _random_pair(h, n, dx, t0, budget_s, margin, rng):
    best_h = h.copy()
    best_c5 = _c5val(h) * dx
    h = h.copy()
    while time.time() < t0 + budget_s - margin:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = rng.choice([1e-4, 2e-4, 5e-4, 1e-3, -1e-4, -2e-4, -5e-4, -1e-3])
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            h[i], h[j] = ni, nj
            c5_try = _c5val(h) * dx
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h.copy()
            else:
                h[i], h[j] = ni - d, nj + d
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
