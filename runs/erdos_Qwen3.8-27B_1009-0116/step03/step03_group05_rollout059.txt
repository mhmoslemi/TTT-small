def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _project(h, target):
    """Project h onto {x: 0<=x<=1, sum(x)=target}"""
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _grad_lag(h, L):
    """Gradient of c_L w.r.t. h"""
    n = len(h)
    g = np.zeros(n)
    if L == 0:
        g = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += 1.0 - h[:n-L]
        g[:n-L] -= h[L:]
    else:
        M = -L
        g[:n-M] += 1.0 - h[M:]
        g[M:] -= h[:n-M]
    return g

def _topk_grad(h, m):
    """Average gradient of top-m correlations"""
    c = _corr(h)
    n = len(h)
    top = np.argsort(c)[::-1][:m]
    g = np.zeros(n)
    for ki in top:
        L = ki - (n - 1)
        g += _grad_lag(h, L)
    g /= float(m)
    return g, c

    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)

    n = len(x)

            g[M:] -= pk * x[:n-M]

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _project(h, target):
    lo = float(h.min()) - 1.0
    hi = float(h.max()) + 1.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def _softobj(x, alpha, lam, target):
    c = _corr(x)
    m = float(c.max())
    s = np.clip(alpha * (c - m), -500.0, 500.0)
    p = lam * (float(np.sum(x)) - target) ** 2
    return m + np.log(np.sum(np.exp(s))) / alpha + p

def _softgrad(x, alpha, lam, target):
    c = _corr(x)
    m = float(c.max())
    s = np.clip(alpha * (c - m), -500.0, 500.0)
    ew = np.exp(s)
    pw = ew / float(ew.sum())
    n = len(x)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        pk = pw[k]
        if pk < 1e-14:
            continue
        if L == 0:
            g += pk * (1.0 - 2.0 * x)
        elif L > 0:
            g[L:] += pk * (1.0 - x[:n-L])
            g[:n-L] -= pk * x[L:]
        else:
            M = -L
            g[:n-M] += pk * (1.0 - x[M:])
            g[M:] -= pk * x[:n-M]
    g /= alpha
    g += 2.0 * lam * (np.sum(x) - target)
    return g

    alpha = 0.03
    for it in range(niter):

        top_idx = np.argsort(c)[::-1][:top_k]

                M = -L
                g[:n-M] += (1.0 - h[M:])

        h_new = _project(h - alpha * g, target)

def _coord_descent(h, n, dx, target, deadline):
    h = h.copy()
    c5_cur = _c5(h, dx)
    deltas = np.array([0.005, 0.01, 0.02, -0.005, -0.01, -0.02,
                       0.001, 0.002, -0.001, -0.002])
    improved_any = True
    while improved_any and time.time() < deadline:
        improved_any = False
        c = _corr(h)
        worst = int(np.argmax(c))
        L = worst - (n - 1)
        # Identify indices that affect the worst lag
        if L == 0:
            idxs = np.arange(n)
        elif L > 0:
            idxs = np.arange(n - L)
        else:
            idxs = np.arange(n + L)
        # Try to adjust pairs
        for i in idxs:
            if time.time() > deadline:
                break
            for j_offset in [1, 2, 3, n//4, n//2]:
                j = i + j_offset if i + j_offset < n else abs(i - j_offset)
                if i == j:
                    continue
                for d in deltas:
                    ni, nj = h[i] + d, h[j] - d
                    if 0 <= ni <= 1 and 0 <= nj <= 1:
                        h[i], h[j] = ni, nj
                        c_new = _c5(h, dx)
                        if c_new < c5_cur:
                            c5_cur = c_new
                            improved_any = True
                            break
                        h[i], h[j] = ni - d, nj + d
        h = _project(h, target)
        c5_cur = min(c5_cur, _c5(h, dx))
    return h, c5_cur

        bounds = [(0.0, 1.0)] * n

        if n == 70 and initial_h_values is not None:
            h0 = _project(np.array(initial_h_values, dtype=float), target)
        else:
            h0 = _cosine_init(n, rng)

        h = h0.copy()

        # Phase A: Subgradient
        dl_a = min(deadline - 60, time.time() + alloc * 0.4)
        h, bh, bc = _subgrad_opt(h, n, dx, target, rng, dl_a)
        if bc < _c5(h, dx):
            h = bh

        # Phase B: L-BFGS-B
        h = _lbfgs_opt(h, n, target, bounds, dl_a if time.time() < dl_a else deadline - 30)

        # Phase C: SLSQP
        dl_c = min(deadline - 15, time.time() + alloc * 0.2)
        if time.time() < dl_c:
            h = _slsqp_opt(h, n, target, bounds, dl_c)

        # Phase D: Another L-BFGS-B round
        if time.time() < deadline - 10:
            h = _lbfgs_opt(h, n, target, bounds, deadline - 10)

        # Phase E: Coordinate polish
        h, c5_coord = _coord_polish(h, n, dx, target, deadline - 2)
        c5_final = _c5(_project(h, target), dx)
        c5_final = min(c5_final, c5_coord)
        h = _project(h, target)
        c5_final = _c5(h, dx)

        # Phase F: Small perturbation + reoptimize
        if time.time() < deadline - 3:
            hp = h + 0.005 * rng.randn(n)
            hp = _project(hp, target)
            _lbfgs_opt(hp, n, target, bounds, deadline - 2)
            c5p = _c5(_project(hp, target), dx)
            if c5p < c5_final:
                h = _project(hp, target)
                c5_final = c5p

        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h.copy()
            best_n = n

    if best_h is None:
        n = 70
        best_h = _project(np.array(initial_h_values, dtype=float), n / 2.0)
        best_c5 = _c5(best_h, 2.0 / n)
        best_n = n

    best_h = _project(best_h, best_n / 2.0)
    best_c5 = _c5(best_h, 2.0 / best_n)
    return (best_h, best_c5, best_n)
