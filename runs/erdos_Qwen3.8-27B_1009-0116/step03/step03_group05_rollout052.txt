This is a linear programming problem in h for fixed c values! Wait, no, c depends on h in a quadratic way (h_i * (1-h_j) is bilinear).

But if we linearize: c[j] ≈ c0[j] + Σ_i (∂c[j]/∂h_i) * Δh_i

Then it becomes a LP. We can solve it iteratively (successive linearization).

This is essentially the **outer approximation** or **Benders decomposition** approach.

Let me implement this:
1. Compute c = _corr(h)
2. Find top-k shifts
3. Linearize c[j] around current h for j in top-k
4. Solve: min t s.t. c0[j] + g_j·Δh ≤ t, |Δh| ≤ ε, sum(Δh)=0, 0≤h+Δh≤1
5. Update h = h + Δh
6. Repeat

The subproblem in step 4 is a LP with 2n+1 variables (h_1,...,h_n, t) and k+1+n+2n constraints. For k=5 and n=70, that's ~146 variables and ~286 constraints. Very fast for CBC or GLPK.

```python
    top_k = np.argsort(c)[::-1][:k]
    
    h_var = cp.Variable(n)
    t = cp.Variable()

    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n/2.0]

    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.CBC)
```

```python
import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

_G = {}

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-10:
                break
    return h

def _grad_single(h, ki):
    """Gradient of correlation value ki w.r.t. h."""
    n = len(h)
    L = ki - (n - 1)
    g = np.zeros(n)
    if L == 0:
        g = 1.0 - 2.0 * h
    elif L > 0:
        g[:n-L] += 1.0 - h[L:]
        g[L:] -= h[:n-L]
    else:
        M = -L
        g[M:] += 1.0 - h[:-M] if M < n else 0.0
        g[:n-M] -= h[M:]
    return g
```

Wait, I need to double-check the gradient formulas.

For L > 0:
c = sum_{i=0}^{n-1-L} h[i] * (1 - h[i+L])

∂c/∂h[m]:
- If m < n-L: m appears as h[m] in the first factor → +（1-h[m+L]）
- If m >= L: m appears as h[m] in (1-h[m]) where the index shift means (1-h[m]) appears when i+L = m, i.e., i = m-L. So the term is h[m-L]*(1-h[m]), and ∂/∂h[m] = -h[m-L]

So:
- For m in [0, n-1-L]: grad += (1-h[m+L])
- For m in [L, n-1]: grad -= h[m-L]

In numpy:
g[:n-L] += (1 - h[L:])  # h[L:] has length n-L
g[L:] -= h[:n-L]       # h[:n-L] has length n-L

For L < 0, let M = -L:
c = sum_{i=M}^{n-1} h[i] * (1 - h[i-M])
  = sum_{j=0}^{n-1-M} h[j+M] * (1 - h[j])

∂c/∂h[m]:
- If m >= M: m appears as h[m] in first factor (j = m-M) → +(1-h[m-M])
- If m < n-M: m appears in (1-h[m]) → -h[m+M]

So:
- For m in [M, n-1]: grad += (1-h[m-M])
- For m in [0, n-1-M]: grad -= h[m+M]

In numpy:
g[M:] += (1 - h[:n-M])  # h[:n-M] has length n-M
g[:n-M] -= h[M:]        # h[M:] has length n-M

```python
import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

_G = {}

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-10:
                break
    return h

def _grad_single(h, ki):
    n = len(h)
    L = ki - (n - 1)
    g = np.zeros(n)
    if L == 0:
        return 1.0 - 2.0 * h
    elif L > 0:
        g[:n-L] += 1.0 - h[L:]
        g[L:] -= h[:n-L]
    else:
        M = -L
        g[M:] += 1.0 - h[:n-M]
        g[:n-M] -= h[M:]
    return g

def _soft_obj(x):
    a = _G['alpha']
    c = _corr(x)
    m = c.max()
    s = np.clip(a * (c - m), -500, 500)
    return float(m + np.log(np.sum(np.exp(s))) / a)

def _soft_grad(x):
    a = _G['alpha']
    c = _corr(x)
    m = c.max()
    s = np.clip(a * (c - m), -500, 500)
    ew = np.exp(s)
    pw = ew / np.sum(ew)
    n = len(x)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        pk = pw[k]
        if pk < 1e-14:
            continue
        if L == 0:
            g += pk * (1.0 - 2.0 * x)
        elif L > 0:
            g[:n-L] += pk * (1.0 - x[L:])
            g[L:] -= pk * x[:n-L]
        else:
            M = -L
            g[M:] += pk * (1.0 - x[:n-M])
            g[:n-M] -= pk * x[M:]
    g /= a
    return g

def _eq_con(x):
    return np.sum(x) - _G['target']

def _adam_opt(h, n, dx, alphas, max_iter_total, t_end):
    """ADAM optimization with increasing alpha."""
    h = h.copy()
    c5 = _c5(h, dx)
    for a in alphas:
        _G['alpha'] = a
        m = np.zeros(n)
        v = np.zeros(n)
        t = 0
        lr = 0.005
        for it in range(max_iter_total):
            if time.time() > t_end:
                return h, c5, True
            t += 1
            if t % 300 == 0:
                lr *= 0.7
            g = _soft_grad(h)
            m = 0.9 * m + 0.1 * g
            v = 0.999 * v + 0.001 * g * g
            mh = m / (1 - 0.9**t)
            vh = v / (1 - 0.999**t)
            h_new = h - lr * mh / (np.sqrt(vh) + 1e-8)
            h = _repair(h_new, n)
            c = _c5(h, dx)
            if c < c5:
                c5 = c
    return h, c5, False

def _lp_polish(h, n, dx, n_top, max_iter, t_end):
    """Iterative LP active-set optimization."""
    h = h.copy()
    c5 = _c5(h, dx)
    for it in range(max_iter):
        if time.time() > t_end:
            break
        c = _corr(h)
        top_idx = np.argsort(c)[::-1][:n_top]
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        cons = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2.0]
        for ki in top_idx:
            g = _grad_single(h, ki)
            cons.append(c[ki] + g @ (h_var - h) <= t_var)
        prob = cp.Problem(cp.Minimize(t_var), cons)
        solved = False
        for solver in [cp.GLPK, cp.CBC, cp.PDLP]:
            try:
                prob.solve(solver=solver)
                solved = True
                break
            except Exception:
                continue
        if not solved or h_var.value is None:
            break
        h_new = _repair(np.array(h_var.value), n)
        c_new = _c5(h_new, dx)
        if c_new < c5 - 1e-12:
            h = h_new
            c5 = c_new
        else:
            break
    return h, c5

def _slsqp_polish(h, n, dx, alpha, max_iter, t_end):
    """SLSQP with soft-max."""
    _G['alpha'] = alpha
    _G['target'] = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con}]
    try:
        res = minimize(_soft_obj, h, jac=_soft_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-16})
        hc = _repair(res.x, n)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _coord_descent(h, n, dx, max_sweeps, t_end):
    """Gradient-guided pair coordinate descent."""
    h = h.copy()
    c5 = _c5(h, dx)
    dlist = np.array([0.001, 0.003, 0.007, 0.015, 0.03,
                      -0.001, -0.003, -0.007, -0.015, -0.03])
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        c = _corr(h)
        top_k = np.argsort(c)[::-1][:5]
        # Compute gradient from top-k to find important coords
        g = np.zeros(n)
        for ki in top_k:
            g += _grad_single(h, ki)
        order = np.argsort(np.abs(g))[::-1][:30]
        improved = False
        for ii in range(len(order)):
            if time.time() > t_end:
                break
            i = order[ii]
            for jj in range(ii + 1, min(ii + 8, len(order))):
                j = order[jj]
                best_d = 0.0
                best_c = c5
                for d in dlist:
                    ni, nj = h[i] + d, h[j] - d
                    if 0 <= ni <= 1 and 0 <= nj <= 1:
                        h[i], h[j] = ni, nj
                        c = _c5(h, dx)
                        if c < best_c:
                            best_c = c
                            best_d = d
                        h[i], h[j] = ni - d, nj + d
                if best_d != 0.0:
                    h[i] += best_d
                    h[j] -= best_d
                    h = _repair(h, n)
                    c5 = _c5(h, dx)
                    improved = True
                    break
        if not improved:
            break
    return h, c5

def _init_two_block(n, seed):
    rng = np.random.RandomState(seed)
    x = np.arange(n, dtype=float)
    c1, c2 = n * 0.25, n * 0.75
```

```python
import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

_G = {}

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-10:
                break
    return h

def _grad_single(h, ki):
    n = len(h)
    L = ki - (n - 1)
    if L == 0:
        return 1.0 - 2.0 * h
    g = np.zeros(n)
    if L > 0:
        g[:n - L] += 1.0 - h[L:]
        g[L:] -= h[:n - L]
    else:
        M = -L
        g[M:] += 1.0 - h[:n - M]
        g[:n - M] -= h[M:]
    return g

def _soft_obj(x):
    a = _G['alpha']
    c = _corr(x)
    m = c.max()
    s = np.clip(a * (c - m), -500, 500)
    return float(m + np.log(np.sum(np.exp(s))) / a)

def _soft_grad(x):
    a = _G['alpha']
    c = _corr(x)
    m = c.max()
    s = np.clip(a * (c - m), -500, 500)
    ew = np.exp(s)
    pw = ew / np.sum(ew)
    n = len(x)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        pk = pw[k]
        if pk < 1e-14:
            continue
        if L == 0:
            g += pk * (1.0 - 2.0 * x)
        elif L > 0:
            g[:n - L] += pk * (1.0 - x[L:])
            g[L:] -= pk * x[:n - L]
        else:
            M = -L
            g[M:] += pk * (1.0 - x[:n - M])
            g[:n - M] -= pk * x[M:]
    g /= a
    return g

def _eq_con(x):
    return np.sum(x) - _G['target']

def _adam_opt(h, n, dx, t_end):
    h = h.copy()
    c5 = _c5(h, dx)
    alphas = [150, 500, 2000, 8000, 30000, 80000]
    for a in alphas:
        _G['alpha'] = a
        m = np.zeros(n)
        v = np.zeros(n)
        t = 0
        lr = 0.004
        base_lr = lr
        for it in range(4000):
            if time.time() > t_end:
                return h, c5
            t += 1
            if t % 400 == 0:
                lr = base_lr * (0.6 ** (t // 400))
            g = _soft_grad(h)
            m = 0.9 * m + 0.1 * g
            v = 0.999 * v + 0.001 * g * g
            mh = m / (1.0 - 0.9 ** t)
            vh = v / (1.0 - 0.999 ** t)
            h_new = h - lr * mh / (np.sqrt(vh) + 1e-8)
            h = _repair(h_new, n)
            c = _c5(h, dx)
            if c < c5:
                c5 = c
    return h, c5

def _lp_polish(h, n, dx, n_top, max_iter, t_end):
    h = h.copy()
    c5 = _c5(h, dx)
    for it in range(max_iter):
        if time.time() > t_end:
            break
        c = _corr(h)
        top_idx = np.argsort(c)[::-1][:n_top]
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        cons = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2.0,
                h_var - h <= 0.04, h - h_var <= 0.04]
        for ki in top_idx:
            g = _grad_single(h, ki)
            cons.append(c[ki] + g @ (h_var - h) <= t_var)
        solved = False
        for solver in [cp.PDLP, cp.GLPK, cp.CBC]:
            try:
                prob = cp.Problem(cp.Minimize(t_var), cons)
                prob.solve(solver=solver)
                if h_var.value is not None:
                    solved = True
                    break
            except Exception:
                continue
        if not solved:
            break
        h_new = _repair(np.array(h_var.value, dtype=float), n)
        c_new = _c5(h_new, dx)
        if c_new < c5 - 1e-12:
            h = h_new
            c5 = c_new
        else:
            if n_top > 3:
                n_top -= 1
            else:
                break
    return h, c5

def _slsqp_polish(h, n, dx, alpha, max_iter, t_end):
    _G['alpha'] = alpha
    _G['target'] = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con}]
    try:
        res = minimize(_soft_obj, h, jac=_soft_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-16})
        hc = _repair(res.x, n)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)

def _coord_descent(h, n, dx, max_sweeps, t_end):
    h = h.copy()
    c5 = _c5(h, dx)
    dlist = np.array([0.001, 0.003, 0.007, 0.015, 0.03,
                      -0.001, -0.003, -0.007, -0.015, -0.03])
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        c = _corr(h)
        top_k = np.argsort(c)[::-1][:5]
        g = np.zeros(n)
        for ki in top_k:
            g += _grad_single(h, ki)
        order = np.argsort(np.abs(g))[::-1][:30]
        improved = False
        for ii in range(len(order)):
            if time.time() > t_end:
                break
            i = order[ii]
            for jj in range(ii + 1, min(ii + 10, len(order))):
                j = order[jj]
                best_d = 0.0
                best_c = c5
                for d in dlist:
                    ni, nj = h[i] + d, h[j] - d
                    if 0 <= ni <= 1 and 0 <= nj <= 1:
                        h[i], h[j] = ni, nj
                        c = _c5(h, dx)
                        if c < best_c:
                            best_c = c
                            best_d = d
                        h[i], h[j] = ni - d, nj + d
                if best_d != 0.0:
                    h[i] += best_d
                    h[j] -= best_d
                    h = _repair(h, n)
                    c5 = _c5(h, dx)
                    improved = True
                    break
        if not improved:
            break
    return h, c5
```
