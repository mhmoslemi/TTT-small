def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(5000):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        d = target - s
        if abs(d) < 1e-14:
            break
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-14:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
    return h

def obj_smooth(h, n, alpha, topk, flat_w):
    corr = compute_corr(h)
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    val = m + np.log(np.sum(np.exp(s))) / alpha
    if topk > 1:
        sc = np.sort(corr)[::-1][:topk]
        val += flat_w * np.sum((sc - sc.mean())**2)
    val += 50.0 * (h.sum() - n / 2.0)**2
    return val

            if i == j:
                continue
            hi, hj = h[i], h[j]
            found = False
            for d in steps:
                for sign in (1, -1):
                    ni, nj = hi + sign * d, hj - sign * d
