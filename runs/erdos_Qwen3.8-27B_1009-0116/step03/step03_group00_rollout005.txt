def _eval_fourier(params, n):
    # params: [a1, b1, a2, b2, a3, b3, ...]
    i = np.arange(n)
    h = 0.5 * np.ones(n)
    for k in range(len(params)//2):
        a = params[2*k]
        b = params[2*k+1]
        h += a * np.cos(2*np.pi*(k+1)*i/n) + b * np.sin(2*np.pi*(k+1)*i/n)
    h = np.clip(h, 0, 1)
    # Repair integral
    s = h.sum()
    if s > 0:
        h *= (n/2.0) / s
        h = np.clip(h, 0, 1)
        # Final repair
        diff = n/2.0 - h.sum()
        if abs(diff) > 1e-12:
            idx = np.argsort(1-h) if diff > 0 else np.argsort(h)
            for j in idx:
                if diff > 0:
                    add = min(diff, 1-h[j])
                    h[j] += add; diff -= add
                else:
                    sub = min(-diff, h[j])
                    h[j] -= sub; diff += sub
                if abs(diff) < 1e-12: break
    return h

def _fourier_obj(params, n):

import time

def _c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return h

def _logsumexp_obj(x, n, alpha):
    c = _corr(x) * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _logsumexp_grad(x, n, alpha):
    dx_val = 2.0 / n
    c = _corr(x) * dx_val
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    e = np.exp(s)
    total = e.sum()
    weights = e / total
    g = np.zeros(n)
    kmax = len(c)
    for k in range(kmax):
        w = weights[k]
        if w < 1e-14:
            continue
        lag = k - (n - 1)
        i_min = max(0, -lag)
        i_max = min(n, n - lag)
        for i in range(i_min, i_max):
            jp = i + lag
            g[jp] += w * (1.0 - x[i])
            g[i] -= w * x[jp]
    return g * dx_val

def _fourier_to_h(params, n):
    i = np.arange(n, dtype=float)

        h += a * np.cos(2*np.pi*(k+1)*i/n)
        h += b * np.sin(2*np.pi*(k+1)*i/n)
    return _repair(h, n)

def _fourier_obj(params, n):
    h = _fourier_to_h(params, n)
    return _c5(h, n)

def _subgrad_step(h, n, lr, alpha):

    g = _logsumexp_grad(h, n, alpha)
    # Project gradient to maintain sum constraint
    g -= np.mean(g)
    h_new = h - lr * g
    return _repair(h_new, n)

import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, n):
    dx = 2.0 / n
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return h

def _logse_obj(x, n, alpha):
    c = _corr(x) * (2.0 / n)
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _logse_grad(x, n, alpha):
    dx_val = 2.0 / n
    c = _corr(x) * dx_val
    m = c.max()
    s = np.clip(alpha * (c - m), -400, 400)
    e = np.exp(s)
    total = e.sum()
    w = e / total
    g = np.zeros(n)
    for k in range(len(c)):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            for j in range(lag, n):
                g[j] += wk * (1.0 - x[j - lag])
            for j in range(0, n - lag):
                g[j] -= wk * x[j + lag]
        else:
            m2 = -lag
            for j in range(0, n - m2):
                g[j] += wk * (1.0 - x[j + m2])
            for j in range(m2, n):
                g[j] -= wk * x[j - m2]
    return g * dx_val

def _fourier_to_h(params, n):
    i = np.arange(n, dtype=float)
    h = np.full(n, 0.5)
    for k in range(len(params) // 2):
        h += params[2*k] * np.cos(2*np.pi*(k+1)*i/n)
        h += params[2*k+1] * np.sin(2*np.pi*(k+1)*i/n)
    return _repair(h, n)

def _fourier_obj(params, n):
    return _c5(_fourier_to_h(params, n), n)

def _slsqp_smooth(h0, n, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
             'jac': lambda x: np.ones(n)}]
    return minimize(_logse_obj, h0, jac=lambda x: _logse_grad(x, n, alpha),
                    args=(n, alpha), method='SLSQP',
                    bounds=bounds, constraints=cons,
                    options={'maxiter': maxiter, 'ftol': 1e-16})

def _pair_search(h, n, budget_end, window=30):
    c5_cur = _c5(h, n)
    deltas = np.array([0.001, 0.003, 0.008, 0.02, 0.05,
                       -0.001, -0.003, -0.008, -0.02, -0.05])
    for _pass in range(300):
        if time.time() > budget_end:
            break
        improved = False
        for i in range(n):
            if time.time() > budget_end:
                break
            hi = h[i]
            for j in range(i + 1, min(i + 1 + window, n)):
                if time.time() > budget_end:
                    break
                hj = h[j]
                for d in deltas:
                    ni, nj = hi + d, hj - d
                    if ni < -1e-12 or ni > 1 + 1e-12 or nj < -1e-12 or nj > 1 + 1e-12:
                        continue
                    h[i], h[j] = ni, nj
                    c = _c5(h, n)
                    if c < c5_cur - 1e-15:
                        c5_cur = c
                        hi, hj = ni, nj
                        improved = True
                    else:
                        h[i], h[j] = hi, hj
                h[i], h[j] = hi, hj
        if not improved:
            break
    return h, c5_cur

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    n = len(initial_h_values)
    best_h = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_c5 = _c5(best_h, n)

    # Phase 1: Fourier global search (different basins)
    if elapsed() < budget_s - 280:
        n_harm = 8
        bounds_f = [(-0.35, 0.35)] * (2 * n_harm)
        for trial in range(5):
            if elapsed() > budget_s - 280:
                break
            try:
                de_res = differential_evolution(
                    _fourier_obj, bounds_f, args=(n,),
                    seed=rng.randint(100000), maxiter=60, popsize=12,
                    tol=1e-6, mutation=(0.5, 1.0), recomb=0.8, polish=False)
                h_try = _fourier_to_h(de_res.x, n)
                c5_try = _c5(h_try, n)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try.copy()
            except Exception:
                pass

    # Phase 2: Multi-alpha SLSQP from best + random perturbations
    for alpha in [1500, 4000, 10000, 25000, 60000]:
        if elapsed() > budget_s - 200:
            break
        n_starts = 6 if alpha <= 4000 else 3
        for r in range(n_starts):
            if elapsed() > budget_s - 200:
                break
            if r == 0:
                h_init = best_h.copy()
            else:
                h_init = _repair(best_h + 0.015 * (1 + 0.4*r) * rng.randn(n), n)
            try:
                res = _slsqp_smooth(h_init, n, alpha, 600)
                hc = _repair(res.x, n)
                c5 = _c5(hc, n)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = hc.copy()
            except Exception:
                pass

    # Phase 3: Subgradient descent with adaptive step
    if elapsed() < budget_s - 160:
        h = best_h.copy()
        c5_cur = _c5(h, n)
        for a_s in [800, 2500, 8000]:
            lr = 0.05
            for step in range(3000):
                if elapsed() > budget_s - 160:
                    break
                g = _logse_grad(h, n, a_s)
                g -= np.mean(g)
                h_new = _repair(h - lr * g, n)
                c5_new = _c5(h_new, n)
                if c5_new <= c5_cur:
                    h = h_new
                    c5_cur = c5_new
                    if c5_cur < best_c5:
                        best_c5 = c5_cur
                        best_h = h.copy()
                    lr = min(lr * 1.005, 0.8)
                else:
                    lr *= 0.6
                if lr < 1e-7:
                    break

    # Phase 4: High-alpha SLSQP polish
    if elapsed() < budget_s - 100:
        for alpha in [50000, 100000]:
            if elapsed() > budget_s - 100:
                break
            for r in range(3):
                if elapsed() > budget_s - 100:
                    break
                h_init = best_h.copy() if r == 0 else _repair(best_h + 0.005*rng.randn(n), n)
                try:
                    res = _slsqp_smooth(h_init, n, alpha, 800)
                    hc = _repair(res.x, n)
                    c5 = _c5(hc, n)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = hc.copy()
                except Exception:
                    pass

    # Phase 5: Pair-move local search
    if elapsed() < budget_s - 60:
        budget_end = t0 + budget_s - 60
        h, c5 = _pair_search(best_h, n, budget_end)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 6: SA refinement
    if elapsed() < budget_s - 30:
        h = best_h.copy()
        c5_cur = _c5(h, n)
        T = 0.00015
        end_t = t0 + budget_s - 30
        while time.time() < end_t:
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 15)) % n
            d = (rng.rand() - 0.5) * 0.03
            hi, hj = h[i], h[j]
            ni, nj = hi + d, hj - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                T *= 0.999
                continue
            h[i], h[j] = ni, nj
            c5_new = _c5(h, n)
            if c5_new < c5_cur or (T > 1e-12 and rng.rand() < np.exp(-(c5_new - c5_cur) / T)):
                c5_cur = c5_new
                if c5_cur < best_c5:
                    best_c5 = c5_cur
                    best_h = h.copy()
            else:
                h[i], h[j] = hi, hj
            T *= 0.99995

    # Phase 7: Ultra-fine coordinate descent
    if elapsed() < budget_s - 5:
        h = best_h.copy()
        c5_cur = _c5(h, n)
        tiny = np.array([0.0001, 0.0003, 0.001, 0.003, -0.0001, -0.0003, -0.001, -0.003])
        end_t = t0 + budget_s - 5
        while time.time() < end_t:
            i = rng.randint(n)
            j = (i + 1 + rng.randint(0, 10)) % n
            d = tiny[rng.randint(len(tiny))]
            hi, hj = h[i], h[j]
            ni, nj = hi + d, hj - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c = _c5(h, n)
                if c < c5_cur:
                    c5_cur = c
                    if c < best_c5:
                        best_c5 = c
                        best_h = h.copy()
                else:
                    h[i], h[j] = hi, hj

    # Final verification
    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, n)
    return (best_h, best_c5, n)
