    active = [k - nm1 for k in range(len(corr)) if corr[k] >= thresh]
    if not active:
        active = [int(np.argmax(corr)) - nm1]
    if len(active) > 11:
        active = sorted(active, key=lambda k: -corr[nm1 + k])[:11]
    return sorted(active)

def active_obj(h, n, lags, alpha, var_w):
    corr = compute_corr(h)
    nm1 = n - 1
    vals = np.array([corr[nm1 + k] for k in lags])
    m = np.max(vals)
    if alpha < 1e12:
        s = alpha * (vals - m)
        s = np.clip(s, -400, 400)
        sm = m + np.log(np.sum(np.exp(s))) / alpha
    else:
        sm = m
    if var_w > 0:
        sm += var_w * np.sum((vals - vals.mean())**2)
    return sm

def do_slsqp_active(h, n, lags, alpha, var_w, maxiter):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def _obj(h_):
        return active_obj(h_, n, lags, alpha, var_w)

    def _eq(h_):
        return h_.sum() - target

    cons = [{'type': 'eq', 'fun': _eq}]
    try:
        res = minimize(_obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return repair(res.x, n)
    except Exception:
        return h

def do_lbfgs_active(h, n, lags, alpha, var_w, pen, maxiter):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def _pen_obj(h_):
        return active_obj(h_, n, lags, alpha, var_w) + pen * (h_.sum() - target)**2

    try:
        res = minimize(_pen_obj, h, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-12})
        return repair(res.x, n)
    except Exception:
        return h

def coord_descent(h, n, t_end, rng, max_rounds):
    h = h.copy()
    best = compute_c5(h, n)
    steps = [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for r in range(max_rounds):
        if time.time() > t_end:
            break
        improved = 0
        for _ in range(8 * n):
            if time.time() > t_end:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            hi, hj = h[i], h[j]
            bld = best
            bdi, bdj = 0.0, 0.0
            for d in steps:
                for sign in (1, -1):
                    delta = sign * d
                    ni, nj = hi + delta, hj - delta
                    if ni < -1e-10 or ni > 1.0 + 1e-10 or nj < -1e-10 or nj > 1.0 + 1e-10:
                        continue
                    h[i], h[j] = ni, nj
                    c5 = compute_c5(h, n)
                    if c5 < bld:
                        bld = c5
                        bdi, bdj = delta, -delta
                    h[i], h[j] = hi, hj
            if bld < best - 1e-14:
                h[i] += bdi
                h[j] += bdj
                best = bld
                improved += 1
        if improved == 0 and r > 20:
            break
    return h, best

def make_starts(n, seed_val):
    rng = np.random.RandomState(seed_val)
    starts = []
    starts.append(repair(0.5 + 0.25 * rng.randn(n), n))
    starts.append(repair(0.5 + 0.3 * np.sin(2 * np.pi * 2 * np.arange(n) / n + rng.uniform(0, 6)), n))
    blk = max(2, n // 5)
    h3 = np.zeros(n)
    for i in range(0, n, 2 * blk):
        h3[i:i + blk] = 1.0
    starts.append(repair(0.4 * h3 + 0.3 + 0.15 * rng.randn(n), n))
    starts.append(repair(0.5 + 0.2 * np.cos(np.pi * np.arange(n) / n)**2 + 0.1 * rng.randn(n), n))
    return starts

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed + 9999)
    best_h = None
    best_c5 = np.inf
    best_n = 70

    init_h = None
    try:
        init_h = np.array(initial_h_values, dtype=float).copy()
    except NameError:
        pass
    if init_h is not None:
        init_n = len(init_h)
        init_h = repair(init_h, init_n)
        best_h = init_h.copy()
        best_c5 = compute_c5(init_h, init_n)
        best_n = init_n
    else:
        best_h = repair(np.full(70, 0.5), 70)
        best_c5 = compute_c5(best_h, 70)
        best_n = 70

    # Phase 1: Active-set SLSQP + L-BFGS iterations on primary n
    t1 = t0 + budget_s * 0.35
    h_work = best_h.copy()
    for it in range(80):
        if time.time() > t1:
            break
        corr = compute_corr(h_work)
        active = find_active(corr, best_n, 0.25)
        improved = False
        for alpha in [3000, 15000, 60000]:
            if time.time() > t1:
                break
            for vw in [0.0, 0.005, 0.02]:
                if time.time() > t1:
                    break
                h_new = do_slsqp_active(h_work.copy(), best_n, active, alpha, vw, 350)
                c5_new = compute_c5(h_new, best_n)
                if c5_new < best_c5 - 1e-14:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    h_work = h_new
                    improved = True
                    break
                elif c5_new < compute_c5(h_work, best_n) - 1e-14:
                    h_work = h_new
            if improved:
                break
        if not improved and it > 10:
            # Try L-BFGS with penalty
            for alpha in [15000, 60000]:
                if time.time() > t1:
                    break
                for pen in [500, 5000]:
                    if time.time() > t1:
                        break
                    h_new = do_lbfgs_active(h_work.copy(), best_n, active, alpha, 0.01, pen, 2000)
                    c5_new = compute_c5(h_new, best_n)
                    if c5_new < best_c5 - 1e-14:
                        best_h = h_new.copy()
                        best_c5 = c5_new
                        h_work = h_new
                        improved = True
                        break
                if improved:
                    break

    # Phase 2: Coordinate descent
    t2 = t0 + budget_s * 0.60
    h_cd, c5_cd = coord_descent(h_work, best_n, t2, rng, 400)
    if c5_cd < best_c5 - 1e-14:
        best_h = h_cd.copy()
        best_c5 = c5_cd

    # Phase 3: Refined active-set with tighter fraction
    t3 = t0 + budget_s * 0.78
    h_work = best_h.copy()
    for it in range(50):
        if time.time() > t3:
            break
        corr = compute_corr(h_work)
        active = find_active(corr, best_n, 0.08)
        for alpha in [20000, 80000]:
            if time.time() > t3:
                break
            h_new = do_slsqp_active(h_work.copy(), best_n, active, alpha, 0.01, 500)
            c5_new = compute_c5(h_new, best_n)
            if c5_new < best_c5 - 1e-14:
                best_h = h_new.copy()
                best_c5 = c5_new
                h_work = h_new
                break

    # Phase 4: Try alternate n values
    t4 = t0 + budget_s * 0.93
    for n_alt in [50, 55, 60, 65, 75, 80, 90, 100]:
        if time.time() > t4:
            break
        if n_alt == best_n:
            continue
        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, n_alt)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = repair(h_interp, n_alt)
        h_alt = h_interp + 0.05 * rng.randn(n_alt)
        h_alt = repair(h_alt, n_alt)
        corr = compute_corr(h_alt)
        active = find_active(corr, n_alt, 0.25)
        for alpha in [10000, 40000]:
            if time.time() > t4:
                break
            h_alt = do_slsqp_active(h_alt, n_alt, active, alpha, 0.01, 300)
            c5_alt = compute_c5(h_alt, n_alt)
            if c5_alt < best_c5 - 1e-14:
                best_h = h_alt.copy()
                best_c5 = c5_alt
                best_n = n_alt

    # Phase 5: Final polish
    t5 = t0 + budget_s * 0.99
    if best_h is not None:
        corr = compute_corr(best_h)
        active = find_active(corr, best_n, 0.1)
        h_fin = do_slsqp_active(best_h.copy(), best_n, active, 100000, 0.02, 600)
        c5_fin = compute_c5(h_fin, best_n)
        if c5_fin < best_c5 - 1e-14:
            best_h = h_fin.copy()
            best_c5 = c5_fin
        h_p, c5_p = coord_descent(best_h, best_n, t5, rng, 150)
        if c5_p < best_c5 - 1e-14:
            best_h = h_p.copy()
            best_c5 = c5_p

    best_h = repair(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return (best_h, best_c5, best_n)
