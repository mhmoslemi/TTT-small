            i2 = j - lags; m2 = (i2 >= 0) & (i2 < n)
            for kk in active:
                if m1[kk]: g[j] += (1.0 - h[i1[kk]])
                if m2[kk]: g[j] -= h[i2[kk]]
        g /= len(active); g -= g.mean()
        improved = False
        for t in steps:
            for sgn in (-1, 1):
                h2 = _repair(np.clip(h - sgn * t * g, 0, 1), n)
                cv = _c5(h2, n)
                if cv < c5c - 1e-13:
                    h, c5c = h2, cv; improved = True; break
            if improved: break
        if improved: stall = 0
        else: stall += 1
    return h, c5c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h, best_c5 = h0.copy(), _c5(h0, n)

    def el(): return time.time() - t0

    # Phase 1: SLSQP log-sum-exp, increasing alpha, multiple restarts
    for alpha in [1500, 4000, 10000, 25000, 60000, 150000]:
        nr = 7 if alpha <= 10000 else 4
        for r in range(nr):
            if el() > budget_s - 250: break
            if r == 0:
                h_init = best_h.copy()
            else:
                sc = 0.025 * (1 + 0.25 * r)
                h_init = _repair(best_h + sc * rng.randn(n), n)
            try:
                hx = _repair(_slsqp(h_init, n, alpha, 600, 1e-18), n)
                cv = _c5(hx, n)
                if cv < best_c5: best_c5, best_h = cv, hx.copy()
            except Exception:
                pass

    # Phase 2: Top-k focused SLSQP
    for k in [2, 3, 5, 8]:
        if el() > budget_s - 180: break
        for r in range(3):
            if el() > budget_s - 180: break
            h_init = best_h.copy() if r == 0 else _repair(best_h + 0.015 * rng.randn(n), n)
            try:
                bnds = [(0.0, 1.0)] * n
                cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
                         'jac': lambda x: np.ones(n)}]
                res = minimize(_topk_obj, h_init, args=(n, k),
                               jac=lambda x: _topk_grad(x, n, k),
                               method='SLSQP', bounds=bnds, constraints=cons,
                               options={'maxiter': 500, 'ftol': 1e-17})
                hx = _repair(res.x, n); cv = _c5(hx, n)
                if cv < best_c5: best_c5, best_h = cv, hx.copy()
            except Exception:
                pass

    # Phase 3: Subgradient polishing
    if el() < budget_s - 120:
        hx, cv = _subgrad_polish(best_h, n, t0 + budget_s - min(80, budget_s - 120))
        if cv < best_c5: best_c5, best_h = cv, hx

    # Phase 4: Guided pair search (focus on active correlation indices)
    if el() < budget_s - 60:
        hx, cv = _pair_search(best_h, n, t0 + budget_s - 60, guided=True)
        if cv < best_c5: best_c5, best_h = cv, hx

    # Phase 5: Full pair search (all indices)
    if el() < budget_s - 30:
        hx, cv = _pair_search(best_h, n, t0 + budget_s - 30, guided=False)
        if cv < best_c5: best_c5, best_h = cv, hx

    # Phase 6: Ultra-fine random coordinate
    if el() < budget_s - 5:
        h = best_h.copy(); c5c = best_c5
        tiny = [0.0003, 0.0007, 0.0015, -0.0003, -0.0007, -0.0015]
        et = t0 + budget_s - 5
        while time.time() < et:
            i, j = int(rng.randint(n)), int(rng.randint(n))
            if i == j: continue
            d = tiny[int(rng.randint(6))]
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                cv = _c5(h, n)
                if cv < c5c - 1e-15:
                    c5c = cv
                    if cv < best_c5: best_c5, best_h = cv, h.copy()
                else:
                    h[i], h[j] = h[i] - d, h[j] + d

    # Phase 7: Try upsampled n
    if el() < budget_s - 80:
        n2 = 2 * n
        x_old = np.linspace(0, 2.0, n, endpoint=False)
        x_new = np.linspace(0, 2.0, n2, endpoint=False)
        h2 = np.interp(x_new, np.concatenate([x_old, [2.0]]),
                       np.concatenate([best_h, [best_h[0]]]))
        h2 = _repair(h2, n2)
        for alpha in [5000, 20000]:
            if el() > budget_s - 30: break
            try:
                hx = _repair(_slsqp(h2, n2, alpha, 400, 1e-17), n2)
                cv = _c5(hx, n2)
                if cv < best_c5:
                    best_c5, best_h, n = cv, hx, n2
            except Exception:
                pass
        if el() < budget_s - 15:
            hx, cv = _pair_search(best_h, n, t0 + budget_s - 15, guided=True)
            if cv < best_c5: best_c5, best_h = cv, hx

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, n)
    return (best_h, best_c5, n)
