import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
            else:
                add = max(diff, -h[idx])
            h[idx] += add
            diff -= add
            if abs(diff) < 1e-14:
                break
    return h

def _c5_vals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def _c5(h):
    return float(np.max(_c5_vals(h)))

def _smooth_obj(x, alpha):
    c = _c5_vals(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(x, alpha):
    n = len(x)
    dx_val = 2.0 / n
    c = _c5_vals(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -400, 400)
    e = np.exp(s)
    w = e / np.sum(e)
    g = np.zeros(n)
    n_corr = len(c)
    for k in range(n_corr):
        if w[k] < 1e-14:
            continue
        lag = k - (n - 1)
        a = max(0, -lag)
        b = min(n, n - lag)
        for i in range(a, b):
            g[i] += w[k] * (1.0 - x[i + lag])
            g[i + lag] += w[k] * (-x[i])
    return g * dx_val

def _slsqp(h_init, n, alpha, maxit):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    return minimize(
        lambda x: _smooth_obj(x, alpha), h_init,
        jac=lambda x: _smooth_grad(x, alpha),
        method='SLSQP', bounds=bounds, constraints=cons,
        options={'maxiter': maxit, 'ftol': 1e-16})

def _nesterov_phase(h_start, n, t_end, rng):
    h = h_start.copy()
    mom = np.zeros(n)
    lr = 0.05
    beta = 0.85
    alpha = 3000
    best_local = _c5(h)
    best_h_local = h.copy()
    stagnant = 0
    while time.time() < t_end:
        g = _smooth_grad(h, alpha)
        mom = beta * mom - lr * g
        h_new = _repair(h + mom, n)
        c_new = _c5(h_new)
        if c_new < best_local:
            h = h_new
            best_local = c_new
            best_h_local = h.copy()
            stagnant = 0
            lr = min(lr * 1.05, 0.3)
        else:
            stagnant += 1
            lr *= 0.7
            if stagnant > 30:
                # Perturb and restart momentum
                h = _repair(best_h_local + 0.03 * rng.randn(n), n)
                mom = np.zeros(n)
                lr = 0.05
                stagnant = 0
        if time.time() < t_end:

    return best_h_local, best_local

def _triple_search(h_start, n, t_end, rng):

    h = h_start.copy()
    c5_cur = _c5(h)
    while time.time() < t_end:

                if cn > c5_cur + 0.01:

    x_new = np.arange(n_new) / n_new
    h_ext = np.concatenate([h, h[:1]])
    return _repair(np.interp(x_new, x_old, h_ext), n_new)
