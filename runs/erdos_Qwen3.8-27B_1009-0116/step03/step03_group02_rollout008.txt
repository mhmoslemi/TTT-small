This is a nice idea but adds complexity. Let me keep it simpler: just use the random ordering but do more rounds.

Actually, I realize the key bottleneck in the previous code was:
1. The O(n²) pair loop with bisection is the main cost
2. The code was doing one round at a time and then checking if it improved
3. SLSQP was called periodically, which is expensive

My improvement: make the sweep_round do MULTIPLE passes internally (while improved), so it converges to a local optimum in one call. Then the outer loop just calls it repeatedly with increasing precision, and handles stall/restart.

Let me also make the bisection more precise in later phases. 50 bisection iterations gives precision of about 2^-50 * range, which is essentially exact (the range is at most 1, so precision is ~1e-15).

I think 45 bisection iterations is plenty. The key is doing many rounds.

import numpy as np
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                s2 = min(-diff, h[idx])
                h[idx] -= s2
                diff += s2
            if abs(diff) < 1e-12:
                break
    return h

def _pair_grad(h, i, j, L, n):
    g = np.zeros(2 * n - 1)
    m1 = (i + L >= 0) & (i + L < n)
    g[m1] += 1.0 - h[i + L[m1]]
    m2 = (i - L >= 0) & (i - L < n)
    g[m2] -= h[i - L[m2]]
    m3 = (j + L >= 0) & (j + L < n)
    g[m3] -= 1.0 - h[j + L[m3]]
    m4 = (j - L >= 0) & (j - L < n)
    g[m4] += h[j - L[m4]]
    return g

def _sweep_round(h, n, time_limit, n_coarse, n_bisect):
    corr = _corr(h)
    L = np.arange(-(n - 1), n)
    n_imp = 0
    improved = True
    while improved:
        if time.time() >= time_limit:
            break
        improved = False
        order = np.random.permutation(n)
        for ai in range(n - 1):
            i = order[ai]
            for aj in range(ai + 1, n):
                if time.time() >= time_limit:
                    return h, n_imp
                j = order[aj]
                hi, hj = h[i], h[j]
                d_lo = max(-hi, hj - 1.0)
                d_hi = min(1.0 - hi, hj)
                if d_hi - d_lo < 1e-14:
                    continue
                g = _pair_grad(h, i, j, L, n)
                ds = np.linspace(d_lo, d_hi, n_coarse)
                vals = np.max(corr[None, :] + ds[:, None] * g[None, :], axis=1)
                bi = int(np.argmin(vals))
                cur_max = float(np.max(corr))
                if vals[bi] >= cur_max - 1e-15:
                    continue
                lo = ds[max(0, bi - 1)]
                hi_b = ds[min(n_coarse - 1, bi + 1)]
                for _ in range(n_bisect):
                    mid = 0.5 * (lo + hi_b)
                    if np.max(corr + mid * g) <= np.max(corr + hi_b * g):
                        hi_b = mid
                    else:
                        lo = mid
                d_best = 0.5 * (lo + hi_b)
                h[i] = hi + d_best
                h[j] = hj - d_best
                corr = corr + d_best * g
                improved = True
                n_imp += 1
    h = _repair(h, n)
    return h, n_imp

def _gd_step(h, n, dx, alpha, lr, time_limit):
    idx = np.arange(n)
    for _ in range(150):
        if time.time() >= time_limit:
            break
        corr = _corr(h)
        c = alpha * corr * dx
        c -= np.max(c)
        eg = np.exp(c)
        p = eg / eg.sum()
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            pk = p[k]
            if pk < 1e-10:
                continue
            Lk = k - (n - 1)
            m1 = (idx + Lk >= 0) & (idx + Lk < n)
            grad[m1] += pk * (1.0 - h[idx[m1] + Lk])
            m2 = (idx - Lk >= 0) & (idx - Lk < n)
            grad[m2] -= pk * h[idx[m2] - Lk]
        h_new = _repair(np.clip(h - lr * grad, 0.0, 1.0), n)
        c5_new = _c5(h_new, dx)
        c5_cur = _c5(h, dx)
        if c5_new <= c5_cur + 1e-15:
            h = h_new
        else:
            lr *= 0.5
            if lr < 1e-13:
                break
    return h

def _focused_sweep(h, n, time_limit, n_coarse, n_bisect, top_k=3):
    corr = _corr(h)
    L = np.arange(-(n - 1), n)
    n_imp = 0
    top_lags = np.argsort(corr)[::-1][:top_k]
    hot = set()
    for tl in top_lags:
        Lk = tl - (n - 1)
        if Lk > 0:
            for ii in range(n - Lk):
                if h[ii] > 0.2 and h[ii + Lk] < 0.8:
                    hot.add(ii)
                    hot.add(ii + Lk)
        elif Lk < 0:
            for ii in range(-Lk, n):
                if h[ii] > 0.2 and h[ii + Lk] < 0.8:
                    hot.add(ii)
                    hot.add(ii + Lk)
        else:
            for ii in range(n):
                if 0.2 < h[ii] < 0.8:
                    hot.add(ii)
    hot_list = list(hot)
    other = [x for x in range(n) if x not in hot]
    np.random.shuffle(hot_list)
    np.random.shuffle(other)
    order = hot_list + other
    improved = True
    while improved and time.time() < time_limit:
        improved = False
        for ai in range(n - 1):
            i = order[ai]
            for aj in range(ai + 1, n):
                if time.time() >= time_limit:
                    return h, n_imp
                j = order[aj]
                hi, hj = h[i], h[j]
                d_lo = max(-hi, hj - 1.0)
                d_hi = min(1.0 - hi, hj)
                if d_hi - d_lo < 1e-14:
                    continue
                g = _pair_grad(h, i, j, L, n)
                ds = np.linspace(d_lo, d_hi, n_coarse)
                vals = np.max(corr[None, :] + ds[:, None] * g[None, :], axis=1)
                bi = int(np.argmin(vals))
                cur_max = float(np.max(corr))
                if vals[bi] >= cur_max - 1e-15:
                    continue
                lo = ds[max(0, bi - 1)]
                hi_b = ds[min(n_coarse - 1, bi + 1)]
                for _ in range(n_bisect):
                    mid = 0.5 * (lo + hi_b)
                    if np.max(corr + mid * g) <= np.max(corr + hi_b * g):
                        hi_b = mid
                    else:
                        lo = mid
                d_best = 0.5 * (lo + hi_b)
                h[i] = hi + d_best
                h[j] = hj - d_best
                corr = corr + d_best * g
                improved = True
                n_imp += 1
    h = _repair(h, n)
    return h, n_imp

def _elapsed(t0):
    return time.time() - t0

def _remaining(t0, budget_s):
    return budget_s - 3.0 - (time.time() - t0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()

    n = len(initial_h_values)

            c5g = _c5(h, dx)
