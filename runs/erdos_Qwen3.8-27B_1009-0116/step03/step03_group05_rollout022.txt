import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _sum_top_m(h, m, dx):
    c = _corr(h)
    top = np.sort(c)[::-1][:m]
    return float(np.sum(top)) * dx

def _grad_top_m(h, m):
    n = len(h)
    c = _corr(h)
    top_idx = np.argsort(c)[::-1][:m]
    g = np.zeros(n)
    for ki in top_idx:
        L = ki - (n - 1)
        if L == 0:
            g += (1.0 - 2.0 * h)
        elif L > 0:
            g[L:] += (1.0 - h[:n-L])
            g[:n-L] -= h[L:]
        else:
            M = -L
            g[:n-M] += (1.0 - h[M:])
            g[M:] -= h[:n-M]
    return g

def _softobj(x, alpha, lam):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    penalty = lam * (np.sum(x) - len(x) / 2.0) ** 2
    return m + np.log(np.sum(np.exp(s))) / alpha + penalty

def _softgrad(x, alpha, lam):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    ew = np.exp(s)
    pw = ew / ew.sum()
    n = len(x)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        pk = pw[k]
        if pk < 1e-14:
            continue
        if L == 0:
            g += pk * (1.0 - 2.0 * x)
        elif L > 0:
            g[L:] += pk * (1.0 - x[:n-L])
            g[:n-L] -= pk * x[L:]
        else:
            M = -L
            g[:n-M] += pk * (1.0 - x[M:])
            g[M:] -= x[:n-M]
    g *= 1.0 / alpha
    g += 2.0 * lam * (np.sum(x) - n / 2.0)
    return g

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = np.sum(h)
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

import numpy as np
from scipy.optimize import minimize
import time


def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _c5(h, dx):
    return float(np.max(_corr(h))) * dx


def _softobj(x, alpha, lam):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    penalty = lam * (np.sum(x) - len(x) / 2.0) ** 2
    return m + np.log(np.sum(np.exp(s))) / alpha + penalty


def _softgrad(x, alpha, lam):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    ew = np.exp(s)
    pw = ew / ew.sum()
    n = len(x)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        pk = pw[k]
        if pk < 1e-14:
            continue
        if L == 0:
            g += pk * (1.0 - 2.0 * x)
        elif L > 0:
            g[L:] += pk * (1.0 - x[:n - L])
            g[:n - L] -= pk * x[L:]
        else:
            M = -L
            g[:n - M] += pk * (1.0 - x[M:])
            g[M:] -= x[:n - M]
    g *= 1.0 / alpha
    g += 2.0 * lam * (np.sum(x) - n / 2.0)
    return g


def _topm_obj(x, m):
    c = _corr(x)
    top = np.sort(c)[::-1][:m]
    return float(np.sum(top))


def _topm_grad(x, m):
    n = len(x)
    c = _corr(x)
    top_idx = np.argsort(c)[::-1][:m]
    g = np.zeros(n)
    for ki in top_idx:
        L = ki - (n - 1)
        if L == 0:
            g += (1.0 - 2.0 * x)
        elif L > 0:
            g[L:] += (1.0 - x[:n - L])
            g[:n - L] -= x[L:]
        else:
            M = -L
            g[:n - M] += (1.0 - x[M:])
            g[M:] -= x[:n - M]

    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = np.sum(h)
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

    for m in [1, 3, 5, 8, 15]:

    if time.time() - t0 < budget_s - 25:

import numpy as np
from scipy.optimize import minimize
import time


def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _c5(h, dx):
    return float(np.max(_corr(h))) * dx


def _softobj(x, alpha, lam):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha + lam * (np.sum(x) - len(x) / 2.0) ** 2


def _softgrad(x, alpha, lam):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -500, 500)
    ew = np.exp(s)
    pw = ew / ew.sum()
    n = len(x)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        pk = pw[k]
        if pk < 1e-14:
            continue
        if L == 0:
            g += pk * (1.0 - 2.0 * x)
        elif L > 0:
            g[L:] += pk * (1.0 - x[:n - L])
            g[:n - L] -= pk * x[L:]
        else:
            M = -L
            g[:n - M] += pk * (1.0 - x[M:])
            g[M:] -= x[:n - M]
    g *= 1.0 / alpha
    g += 2.0 * lam * (np.sum(x) - n / 2.0)
    return g


def _topm_grad(x, m):
    n = len(x)
    c = _corr(x)
    top_idx = np.argsort(c)[::-1][:m]
    g = np.zeros(n)
    for ki in top_idx:
        L = ki - (n - 1)
        if L == 0:
            g += (1.0 - 2.0 * x)
        elif L > 0:
            g[L:] += (1.0 - x[:n - L])
            g[:n - L] -= x[L:]
        else:
            M = -L
            g[:n - M] += (1.0 - x[M:])
            g[M:] -= x[:n - M]
    return g


def _sum_eq(x, n):
    return np.sum(x) - n / 2.0


def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = np.sum(h)
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h


def _run_hlbfgs(h, n, bounds, t0, tlim):
    for alpha in [500, 2000, 8000, 40000, 200000, 800000]:
        if time.time() - t0 > tlim:
            break
        try:
            res = minimize(_softobj, h.copy(), args=(alpha, 40.0),
                           jac=_softgrad, method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 300, 'ftol': 1e-16})
            hc = _repair(res.x, n)
            if _c5(hc, 2.0 / n) < _c5(h, 2.0 / n):
                h = hc
        except Exception:
            pass
    return h


def _run_slseq(h, n, bounds, t0, tlim):
    cons = [{'type': 'eq', 'fun': _sum_eq, 'args': (n,)}]
    for alpha in [2000, 10000, 50000, 200000]:
        if time.time() - t0 > tlim:
            break
        try:
            res = minimize(_softobj, h.copy(), args=(alpha, 0.0),
                           method='SLSQP', bounds=bounds,
                           constraints=cons,
                           options={'maxiter': 500, 'ftol': 1e-16})
            hc = _repair(res.x, n)
            if _c5(hc, 2.0 / n) < _c5(h, 2.0 / n):
                h = hc
        except Exception:
            pass
    return h


def _run_topm_grad(h, n, t0, tlim):
    dx = 2.0 / n
    for m in [1, 2, 3, 5, 8, 12, 20]:
        if time.time() - t0 > tlim:
            break
        c5_l = _c5(h, dx)
        step = 0.006
        h_l = h.copy()
        for it in range(300):
            if time.time() - t0 > tlim:
                break
            g = _topm_grad(h_l, m)
            g -= g.mean()
            best_s, best_c, best_ht = 0.0, c5_l, None
            for s in [step, step * 0.5, step * 0.25, step * 0.1, step * 0.04]:
                ht = _repair(h_l - s * g, n)
                ct = _c5(ht, dx)
                if ct < best_c:
                    best_c, best_s, best_ht = ct, s, ht
            if best_s > 0 and best_ht is not None:
                h_l = best_ht
                c5_l = best_c
                if c5_l < _c5(h, dx):
                    h = h_l.copy()
                step = min(step * 1.4, 0.03)
            else:
                step *= 0.5
                if step < 1e-8:
                    break
    return h


def _run_pair_cd(h, n, t0, tlim, rng):
    dx = 2.0 / n
    h_l = h.copy()
    c5_l = _c5(h_l, dx)
    deltas = [0.004, 0.002, 0.001, 0.0005, 0.0002,
              -0.0002, -0.0005, -0.001, -0.002, -0.004]
    for sweep in range(200):

                    if 0 <= ni <= 1 and 0 <= nj <= 1:
                        h_work = h_l.copy()
                        h_work[i], h_work[j] = ni, nj
                        h_work = _repair(h_work, n)
                        c = _c5(h_work, dx)
                        if c < best_c:
                            best_c, best_d, best_ht = c, d, h_work

                    break

        if not improved:

    return h, c5_l, n
