    h = np.clip(h, 0, 1)

    for i in range(n // 2):

def _active_set_gradient(h, n, top_k=5):
    """Compute gradient of max over top-k shifts, using soft-max for the rest."""
    c = np.correlate(h, 1.0 - h, mode='full')
    # Find top-k shifts
    top_idx = np.argsort(c)[::-1][:top_k]
    # Gradient of sum of top-k correlations
    g = np.zeros(n)
    for ki in top_idx:
        L = ki - (n - 1)
        if L == 0:
            g += (1.0 - 2.0 * h)
        elif L > 0:
            g[L:] += (1.0 - h[:n-L])
            g[:n-L] -= h[L:]
        else:
            M = -L
            g[:n-M] += (1.0 - h[M:])
            g[M:] -= h[:n-M]
    # Project gradient to maintain sum constraint
    g -= g.mean()
    return g, c

    g = np.zeros(n)
    for ki in top_idx:
        L = ki - (n-1)
        if L == 0:
            g += (1.0 - 2.0*x)
        elif L > 0:
            g[L:] += (1.0 - x[:n-L])
            g[:n-L] -= x[L:]
        else:
            M = -L
            g[:n-M] += (1.0 - x[M:])
            g[M:] -= x[:n-M]
    g += 2.0*lam*(np.sum(x) - n/2.0)
    return g

```python
import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, n):
    return float(np.max(_corr(h))) * 2.0 / n

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _grad_shifts(h, n, idx_list):
    """Gradient of sum of correlations at given shift indices."""
    n_local = n
    g = np.zeros(n_local)
    m = np.arange(n_local)
    for k_idx in idx_list:
        k = k_idx - (n_local - 1)
        # First term: (1-h[m-k]) where 0 <= m-k < n
        mask1 = (m - k >= 0) & (m - k < n_local)
        g[mask1] += 1.0 - h[m[mask1] - k]
        # Second term: -h[m+k] where 0 <= m+k < n
        mask2 = (m + k >= 0) & (m + k < n_local)
        g[mask2] -= h[m[mask2] + k]
    return g

def _topk_obj(x, n, top_idx, lam):
    c = _corr(x)
    obj = np.sum(c[top_idx])
    penalty = lam * (np.sum(x) - n / 2.0) ** 2
    return obj + penalty

def _topk_grad(x, n, top_idx, lam):
    g = _grad_shifts(x, n, top_idx)
    g += 2.0 * lam * (np.sum(x) - n / 2.0)
    return g

def _project_sum(h, n):
    """Project h onto {sum h = n/2, 0 <= h <= 1} approximately."""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    # Distribute correction
    if abs(diff) > 1e-15:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for i in order:
            if diff > 0:
                a = min(diff, 1.0 - h[i])
                if a <= 0:
                    break
                h[i] += a
                diff -= a
            else:
                a = min(-diff, h[i])
                if a <= 0:
                    break
                h[i] -= a
                diff += a
            if abs(diff) < 1e-14:
                break
    return h

    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    def elapsed():
        return time.time() - t0
