                method='SLSQP', bounds=bounds, constraints=cons,
                options={'maxiter': 500, 'ftol': 1e-16})
            hc = _repair(res.x, n)
            c5 = _c5_val(hc, n)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hc.copy()
        except Exception:
            pass

    # Phase 3: Top-k SLSQP
    for k_val in [3, 5, 8, 12]:
        if elapsed() > budget_s - 150:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
                 'jac': lambda x: np.ones(n)}]
        try:
            res = minimize(
                lambda x: _topk_eval(x, n, k_val), best_h,
                jac=lambda x: _topk_grad(x, n, k_val),
                method='SLSQP', bounds=bounds, constraints=cons,
                options={'maxiter': 500, 'ftol': 1e-16})
            hc = _repair(res.x, n)
            c5 = _c5_val(hc, n)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hc.copy()
        except Exception:
            pass

    # Phase 4: Local search (structured)
    if elapsed() < budget_s - 120:
        best_h, best_c5 = _local_search(best_h, n, budget_s - 120, rng, focus_lags=10)

    # Phase 5: Annealing with restarts
    if elapsed() < budget_s - 60:
        for _restart in range(3):
            if elapsed() > budget_s - 60:
                break
            h_a = best_h.copy()
            if _restart > 0:
                h_a = _repair(h_a + 0.015 * rng.randn(n), n)
            h_a, c5_a = _anneal(h_a, n, budget_s - 60 + _restart * 5, rng)
            if c5_a < best_c5:
                best_c5 = c5_a
                best_h = h_a

    # Phase 6: Multi-resolution trial
    if elapsed() < budget_s - 80:
        for n_try in [100, 140]:
            if elapsed() > budget_s - 50:
                break
            h_mr, c5_mr, nr = _try_multires(best_h, n, n_try, budget_s - 50, rng)
            if c5_mr < best_c5:
                best_c5 = c5_mr
                best_h = h_mr
                n = nr

    # Phase 7: Final ultra-fine local search
    if elapsed() < budget_s - 10:
        h_fin = best_h.copy()
        c5_fin = _c5_val(h_fin, n)
        tiny = np.array([0.0005, 0.001, 0.003, -0.0005, -0.001, -0.003])
        c = np.correlate(h_fin, 1.0 - h_fin, mode='full')
        top_idx = np.argsort(c)[::-1][:6]
        end_t = t0 + budget_s - 5
        while time.time() < end_t:
            improved = False
            for tidx in top_idx:
                lag = tidx - (n - 1)
                imin = max(0, -lag)
                imax = min(n, n - lag)
                for i in range(imin, imax):
                    if time.time() > end_t:
                        break
                    jp = i + lag
                    hi, hj = h_fin[i], h_fin[jp]
                    for s in tiny:
                        ni, nj = hi + s, hj - s
                        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                            continue
                        h_fin[i], h_fin[jp] = ni, nj
                        c5_new = _c5_val(h_fin, n)
                        if c5_new < c5_fin - 1e-15:
                            c5_fin = c5_new
                            hi, hj = ni, nj
                            improved = True
                        else:
                            h_fin[i], h_fin[jp] = hi, hj
                    h_fin[i] = hi
                    h_fin[jp] = hj
            if not improved:
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                s = tiny[rng.randint(len(tiny))]
                hi, hj = h_fin[i], h_fin[j]
                ni, nj = hi + s, hj - s
                if 0 <= ni <= 1 and 0 <= nj <= 1:
                    h_fin[i], h_fin[j] = ni, nj
                    c5_new = _c5_val(h_fin, n)
                    if c5_new < c5_fin:
                        c5_fin = c5_new
                    else:
                        h_fin[i], h_fin[j] = hi, hj
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin.copy()

    best_h = _repair(best_h, n)
    best_c5 = _c5_val(best_h, n)
    return (best_h, best_c5, n)
