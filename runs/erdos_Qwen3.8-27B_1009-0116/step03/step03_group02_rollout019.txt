import time

def _corr(h):

def _c5(h, dx):

def _lsexp(c, alpha):

    s = c - m
    s = np.clip(s, -500, 500)
    e = np.exp(s)
    return m + np.log(np.sum(e)) / alpha

def _grad_lsexp(h, alpha, n):
    """Gradient of log-sum-exp(corr(h)/alpha) w.r.t. h"""
    corr = _corr(h)
    m = float(np.max(corr))

    grad = np.zeros(n)

def _g_vec(h, p, n):
    """d(corr)/d(h[p]) as a vector of length 2n-1"""
    kk = np.arange(-(n-1), n)
    g = np.zeros(2*n-1)
    idx1 = p - kk  # h[i+k] where i+k=p means i=p-k
    m1 = (idx1 >= 0) & (idx1 < n)
    g[m1] += (1.0 - h[idx1[m1]])
    idx2 = p + kk  # (1-h[i]) where i=p means h[i+k]=h[p+k]
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _pair_opt(h, i, j, corr, n, n_coarse=20, n_bisect=30):
    """Optimal transfer from j to i (or vice versa) preserving sum"""
    hi, hj = h[i], h[j]
    d_min = max(-hi, hj - 1.0)
    d_max = min(1.0 - hi, hj)
    if d_max - d_min < 1e-14:
        return h, corr, False
    gd = _g_vec(h, i, n) - _g_vec(h, j, n)
    ds = np.linspace(d_min, d_max, n_coarse)
    vals = np.max(corr[None, :] + ds[:, None] * gd[None, :], axis=1)
    bi = np.argmin(vals)
    if vals[bi] >= np.max(corr) - 1e-14:
        return h, corr, False
    lo = max(d_min, ds[max(0, bi-1)])
    hi_b = min(d_max, ds[min(n_coarse-1, bi+1)])
    for _ in range(n_bisect):
        mid = 0.5 * (lo + hi_b)
        if np.max(corr + mid * gd) <= np.max(corr + hi_b * gd):
            hi_b = mid
        else:
            lo = mid
    d_best = 0.5 * (lo + hi_b)
    h[i] = hi + d_best
    h[j] = hj - d_best
    corr = corr + d_best * gd
    return h, corr, True

def _grad_step(h, n, alpha, step_scale=0.1):
    """One projected gradient step on smooth max"""
    grad = _grad_lsexp(h, alpha, n)
    # Project gradient onto sum(h)=target subspace: subtract mean
    grad = grad - np.mean(grad)
    # Backtracking line search
    corr0 = _corr(h)
    max0 = np.max(corr0)
    step = step_scale
    for _ in range(20):

        h_new = _repair(h_new, n)
        if np.allclose(h_new, h, atol=1e-15):
            return h, False
        max_new = np.max(_corr(h_new))
        if max_new < max0:
            return h_new, True
        step *= 0.5
    return h, False

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _lsexp(c, alpha):
    m = float(np.max(c))
    s = np.clip(c - m, -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _g_vec(h, p, n):
    kk = np.arange(-(n - 1), n)
    g = np.zeros(2 * n - 1)
    idx1 = p - kk
    m1 = (idx1 >= 0) & (idx1 < n)
    g[m1] += (1.0 - h[idx1[m1]])
    idx2 = p + kk
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _pair_opt(h, i, j, corr, n, nc=22, nb=32):

    d_min = max(-hi, hj - 1.0)
    d_max = min(1.0 - hi, hj)
    if d_max - d_min < 1e-14:
        return h, corr, False
    gd = _g_vec(h, i, n) - _g_vec(h, j, n)
    ds = np.linspace(d_min, d_max, nc)

    bi = int(np.argmin(vals))
    if vals[bi] >= np.max(corr) - 1e-14:
        return h, corr, False
    lo = max(d_min, ds[max(0, bi - 1)])
    hi_b = min(d_max, ds[min(nc - 1, bi + 1)])
    for _ in range(nb):
        mid = 0.5 * (lo + hi_b)
        if np.max(corr + mid * gd) <= np.max(corr + hi_b * gd):
            hi_b = mid
        else:
            lo = mid
    d_best = 0.5 * (lo + hi_b)
    h[i] = hi + d_best
    h[j] = hj - d_best
    corr = corr + d_best * gd
    return h, corr, True

    improved = False

            i, j = int(order[ai]), int(order[aj])
            h, corr, imp = _pair_opt(h, i, j, corr, n, nc, nb)
            if imp:
                improved = True
    return h, corr, improved

def _grad_smooth(h, n, alpha):
    corr = _corr(h)
    m = float(np.max(corr))
    s = np.clip(corr - m, -500, 500)
    e = np.exp(s)
    w = e / np.sum(e)
    grad = np.zeros(n)
    for i in range(n):
        g = _g_vec(h, i, n)
        grad[i] = float(np.dot(w, g))
    return grad / alpha

def _grad_step(h, n, alpha, base_step=0.08):

    grad -= np.mean(grad)
    max0 = float(np.max(_corr(h)))

        max_new = float(np.max(_corr(h_new)))
        if max_new < max0 - 1e-15:
            return h_new, True
        step *= 0.5
    return h, False
