def _project(h, target):
    """Project h onto {0<=h<=1, sum(h)=target}"""
    h = np.clip(h, 0, 1)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        # Need to add diff total, distribute to elements with most room
        order = np.argsort(1.0 - h)  # most room first
        for idx in order:
            add = min(diff, 1.0 - h[idx])
            h[idx] += add
            diff -= add
            if diff < 1e-12:
                break
    else:
        # Need to remove -diff total, take from elements with most value
        order = np.argsort(h)  # least value first (least cost to reduce)
        for idx in order:
            sub = min(-diff, h[idx])
            h[idx] -= sub
            diff += sub
            if diff > -1e-12:
                break
    return h
