    constraints = []
    for L in range(-(n-1), n):
        valid_i = np.array([i for i in range(n) if 0 <= i + L < n])
        if len(valid_i) == 0:
            continue
        lin = np.zeros(n)
        const = 0.0
        for i in valid_i:
            j = i + L
            lin[i] += (1.0 - h0[j])
            lin[j] -= h0[i]
            const += h0[i] * h0[j]
        constraints.append((lin, const))
    return constraints

def _ccp_solve(h0, n, reg=0.01, solver=None):
    dx = 2.0 / n
    h_var = cp.Variable(n)
    t = cp.Variable()
    
    cons = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2.0]
    
    data = _build_lp_constraints(h0, n)
    for lin, const in data:
        cons.append(cp.multiply(dx, lin) @ h_var + dx * const <= t)
    
    prob = cp.Problem(cp.Minimize(t + reg * cp.sum_squares(h_var - h0)), cons)
    if solver:
        prob.solve(solver=solver)
    else:
        prob.solve()
    
    if h_var.value is not None:
        return np.array(h_var.value).flatten().reshape(n)
    return h0.copy()

            stagnant += 1
        if stagnant > 500:

def _sys_refine(h, n, t_end):

def _corr_vals(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

    return float(np.max(_corr_vals(h, n)))

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
