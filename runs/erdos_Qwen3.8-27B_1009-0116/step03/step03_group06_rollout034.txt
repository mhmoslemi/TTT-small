import numpy as np
from scipy.optimize import minimize
import time

def _compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _smooth_obj_grad(x, alpha):
    """Smooth max objective with analytical gradient."""
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    exp_s = np.exp(s)
    w = exp_s / np.sum(exp_s)
    obj = m + np.log(np.sum(exp_s)) / alpha
    
    # Gradient: for each j, sum over lags
    grad = np.zeros(n)
    # lag l corresponds to index l + (n-1) in corr
    for j in range(n):
        gj = 0.0
        for l in range(-(n-1), n):
            idx = l + (n - 1)
            wl = w[idx]
            if wl < 1e-14:
                continue
            i1 = j - l  # index in (1-h) for the h[j] term
            if 0 <= i1 < n:
                gj += wl * (1.0 - x[i1])
            i2 = j + l  # index in h for the (1-h[j]) term
            if 0 <= i2 < n:
                gj -= wl * x[i2]
        grad[j] = gj
    return obj, grad

def _slsqp_analytic(h, n, dx, alpha, maxiter=500):
    """SLSQP with analytical gradient."""
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return _smooth_obj_grad(x, alpha)[0]
    
    def grad(x):
        return _smooth_obj_grad(x, alpha)[1]
    
    def constr(x):
        return np.sum(x) - target
    
    def constr_jac(x):
        return np.ones(n)
    
    try:
        res = minimize(
            obj, h, jac=grad, method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constr, 'jac': constr_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-15}
        )
        hc = _repair(res.x, n)
        c5 = _compute_c5(hc, dx)
        return hc, c5
    except Exception:
        return h, _compute_c5(h, dx)

def _lbfgs_analytic(h, n, dx, alpha, maxiter=500, pen=100.0):
    """L-BFGS-B with analytical gradient and penalty for constraint."""
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        o, g = _smooth_obj_grad(x, alpha)
        return o + pen * (np.sum(x) - target)**2
    
    def grad(x):

        g = g + 2 * pen * (np.sum(x) - target)
        return g

            obj, h, jac=grad, method='L-BFGS-B',

        c5 = _compute_c5(hc, dx)

    except Exception:

def _grad_descent(h, n, dx, alpha, deadline, lr=None, n_iters=20000):
    """Steepest descent with analytical gradient and line search."""
    h = h.copy()
    best_c5 = _compute_c5(h, dx)
    if lr is None:
        lr = 0.01

        if gnorm < 1e-12:

    h = np.clip(h, 0.0, 1.0).copy()

    s = h.sum()
    if s < 1e-14:
        return np.full(n, target / n)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            order = np.argsort(-(1.0 - h))
        else:
            order = np.argsort(-h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add; diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub; diff += sub
            if abs(diff) < 1e-13: break
    return h

def _smooth_sg(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 500)
    es = np.exp(s)
    w = es / es.sum()
    obj = m + np.log(es.sum()) / alpha
    grad = np.zeros(n)
    for j in range(n):
        gj = 0.0
        for l in range(-(n-1), n):
            wl = w[l + n - 1]
            if wl < 1e-15: continue
            i1 = j - l
            if 0 <= i1 < n: gj += wl * (1.0 - h[i1])
            i2 = j + l
            if 0 <= i2 < n: gj -= wl * h[i2]
        grad[j] = gj
    return obj, grad

    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    def f(x): return _smooth_sg(x, alpha)[0]
    def g(x): return _smooth_sg(x, alpha)[1]
    def c(x): return np.sum(x) - target
    def cj(x): return np.ones(n)
    try:
        res = minimize(f, h, jac=g, method='SLSQP', bounds=bounds,
                       constraints=[{'type':'eq','fun':c,'jac':cj}],
                       options={'maxiter':maxiter,'ftol':1e-16})
        hc = _repair(res.x, n)
        return hc, _compute_c5(hc, dx)
    except:
        return h.copy(), _compute_c5(h, dx)

    bounds = [(0.0, 1.0)] * n

        return o + pen * (x.sum() - target)**2

def _repair(h, n):

def _pair(h, n, dx, rng, dl, ni=50000):

def _fine(h, n, dx, rng, dl, nt=50000):
