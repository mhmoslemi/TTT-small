import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-12:
                break
    return h

def _eval(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / len(h))

def _cor(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _cfun(x):
    return np.sum(x) - x.size / 2.0

def _cjac(x):
    return np.ones(x.size)

def _c_grad(h, ki, n):
    g = np.zeros(n)
    L = abs(ki - (n - 1))
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    else:
        for j in range(n - L):
            g[j] += 1.0 - h[j + L]
        for j in range(L, n):
            g[j] -= h[j - L]
    return g

def _sf(h, alpha):
    c = _cor(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sfd(h, alpha):
    n = len(h)
    c = _cor(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    w = np.exp(s - s.max())
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        g += w[k] * _c_grad(h, k, n)
    return g

def _slsqp(h, n, alpha, iters, ftol):
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cfun, 'jac': _cjac}]
    try:
        res = minimize(_sf, h, jac=_sfd, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': iters, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return _repair(h, n)

def _subgrad_refine(h, n, iters, eta0):
    h = h.copy()
    for it in range(iters):
        eta = eta0 * (1.0 / (1.0 + 0.01 * it))
        c = _cor(h)
        idx = np.argsort(c)[-4:]
        g = np.zeros(n)
        for ki in idx:
            g += _c_grad(h, ki, n)
        g /= len(idx)
        h_new = h - eta * g
        h_new = _repair(h_new, n)
        if _eval(h_new) <= _eval(h) + 1e-15:
            h = h_new
        else:
            eta = eta0
    return h

def _local_search(h, n, t_budget, rng):
    h = h.copy()
    c5c = _eval(h)
    dirs_fine = [5e-4, 1e-3, 2e-3, 5e-3, -5e-4, -1e-3, -2e-3, -5e-3]
    dirs_med = [1e-2, 2e-2, 5e-2, -1e-2, -2e-2, -5e-2]
    # Focus on boundary-adjacent indices
    crit_indices = np.where((h < 0.15) | (h > 0.85))[0]
    if len(crit_indices) < 10:
        crit_indices = np.arange(n)
    for ds in [dirs_med, dirs_fine]:
        for _pass in range(5000):
            if time.time() > t_budget:
                return h, c5c
            imp = False
            for i in crit_indices:
                if time.time() > t_budget:
                    return h, c5c
                hi = h[i]
                for j in range(n):
                    if j == i:
                        continue
                    if not (time.time() < t_budget):
                        return h, c5c
                    hj = h[j]
                    for d in ds:
                        ni, nj = hi + d, hj - d
                        if ni < -1e-12 or ni > 1 + 1e-12 or nj < -1e-12 or nj > 1 + 1e-12:
                            continue
                        h[i], h[j] = ni, nj
                        c5n = _eval(h)
                        if c5n < c5c - 1e-16:
                            c5c = c5n; imp = True
                        else:
                            h[i], h[j] = hi, hj
                    h[j] = hj
                h[i] = hi
            if not imp:
                break
    return h, c5c

def _sa_finetune(h, n, t_budget, rng):
    h = h.copy()
    c5c = _eval(h)
    T = 1e-4
    scale = 1e-3
    for it in range(200000):
        if time.time() > t_budget:
            break
        T *= 0.99995
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c5n = _eval(h)
        if c5n < c5c:
            c5c = c5n
            scale = min(scale * 1.05, 5e-3)
        else:
            delta = c5n - c5c
            if rng.rand() > np.exp(-delta / max(T, 1e-8)):
                h[i], h[j] = h[i] - d, h[j] + d
                c5c = c5n
                scale = min(scale * 1.02, 5e-3)
            else:
                h[i], h[j] = h[i] - d, h[j] + d
                scale *= 0.98
    return h, c5c

def _try_n(h_init, n_new, t_limit):
    from scipy.interpolate import interp1d
    x_old = np.linspace(0, 2, len(h_init), endpoint=False) + 1.0 / len(h_init)
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0 / n_new
    f = interp1d(x_old, h_init, kind='linear', fill_value=(h_init[0], h_init[-1]))
    h_new = np.clip(f(x_new), 0, 1)
    h_new = _repair(h_new, n_new)
    c5 = _eval(h_new)
    h_new = _slsqp(h_new, n_new, 50000, 1500, 1e-18)
    c5 = _eval(h_new)
    if time.time() < t_limit:
        h_new = _slsqp(h_new, n_new, 200000, 2000, 1e-19)
        c5 = _eval(h_new)
    return h_new, c5
