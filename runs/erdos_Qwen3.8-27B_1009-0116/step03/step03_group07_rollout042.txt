        if time.time() > t_end:
            return h
        h_new = repair(res.x, n)
        return h_new
    except Exception:
        return h

def anneal_coord(h, n, n_iter, t_end, rng, T_start=0.012, T_end=0.0003):
    h = h.copy()
    best_h = h.copy()
    best_c5, _ = compute_c5(h, n)
    steps = [0.05, 0.03, 0.02, 0.01, 0.006, 0.003, 0.001, 0.0005]
    for it in range(n_iter):
        if time.time() > t_end:
            break
        frac = it / max(n_iter, 1)
        T = T_start * (T_end / T_start) ** frac
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        hi, hj = h[i], h[j]
        accepted = False
        for d in steps:
            for sign in (1, -1):
                delta = sign * d
                ni = hi + delta
                nj = hj - delta
                if ni < -1e-10 or ni > 1.0 + 1e-10 or nj < -1e-10 or nj > 1.0 + 1e-10:
                    continue
                h[i], h[j] = ni, nj
                c5_new, _ = compute_c5(h, n)
                if c5_new < best_c5:
                    best_h = h.copy()
                    best_c5 = c5_new
                    hi, hj = ni, nj
                    accepted = True
                    break
                elif T > 1e-8 and rng.random() < np.exp(-(c5_new - best_c5) / T):
                    hi, hj = ni, nj
                    accepted = True
                    break
                h[i], h[j] = hi, hj
            if accepted:
                break
    return best_h, best_c5

def make_start(n, seq, kind):
    rng = np.random.RandomState(seq)
    if kind == 0:
        h = 0.5 + 0.4 * rng.randn(n)
    elif kind == 1:
        h = np.where(np.arange(n) < n // 2, 1.0, 0.0) + 0.12 * rng.randn(n)
    elif kind == 2:
        f = rng.randint(2, 10)
        ph = rng.uniform(0, 6.283)
        h = 0.5 + 0.4 * np.sin(2.0 * np.pi * f * np.arange(n) / n + ph)
    elif kind == 3:
        blk = max(2, n // rng.randint(4, 10))
        h = np.zeros(n)
        for i in range(0, n, 2 * blk):
            h[i:i + blk] = 1.0
        h = 0.35 * h + 0.65 * (0.5 + 0.12 * rng.randn(n))
    elif kind == 4:
        t = np.arange(n) / n
        h = 0.5 + 0.4 * np.cos(2.0 * np.pi * t) ** 2
    else:
        h = rng.uniform(0, 1, n)
    return repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    t_end = t0 + budget_s
    rng = np.random.RandomState(seed + 999)
    best_h = None
    best_c5 = np.inf
    best_n = 70

    init_h = None
    init_n = 70
    try:
        init_h = np.array(initial_h_values, dtype=float).copy()
        init_n = len(init_h)
    except NameError:
        pass

    if init_h is not None:
        init_h = repair(init_h, init_n)
        best_h = init_h.copy()
        best_c5, _ = compute_c5(init_h, init_n)
        best_n = init_n
    else:
        best_h = make_start(70, seed, 2)
        best_c5, _ = compute_c5(best_h, 70)
        best_n = 70

    h_work = best_h.copy()
    n_p = best_n

    # Phase 1: L-BFGS-B with escalating alpha and peak-spreading (35% budget)
    t1_end = t0 + budget_s * 0.35
    for alpha in [2000, 8000, 30000, 100000, 300000, 800000]:
        if time.time() > t1_end:
            break
        for topk in [1, 3, 5, 7]:
            if time.time() > t1_end:
                break
            for fw in [0.0, 0.003, 0.01, 0.03]:
                if time.time() > t1_end:
                    break
                h_new = lbfgs_run(h_work.copy(), n_p, alpha, topk, fw, 600, t1_end)
                c5_new, _ = compute_c5(h_new, n_p)
                if c5_new < best_c5 - 1e-12:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    h_work = h_new
                elif c5_new < compute_c5(h_work, n_p)[0] - 1e-12:
                    h_work = h_new

    # Phase 2: Annealed coordinate descent (25% budget)
    t2_end = t0 + budget_s * 0.60
    if h_work is not None:
        h_cd, c5_cd = anneal_coord(h_work, n_p, 30000, t2_end, rng, T_start=0.01, T_end=0.0002)
        if c5_cd < best_c5 - 1e-12:
            best_h = h_cd.copy()
            best_c5 = c5_cd
        h_work = best_h.copy()

    # Phase 3: High-alpha L-BFGS-B refinement (15% budget)
    t3_end = t0 + budget_s * 0.75
    if time.time() < t3_end:
        for alpha in [200000, 500000, 1000000]:
            if time.time() > t3_end:
                break
            for topk in [3, 5, 7, 9]:
                if time.time() > t3_end:
                    break
                h_new = lbfgs_run(best_h.copy(), best_n, alpha, topk, 0.01, 800, t3_end)
                c5_new, _ = compute_c5(h_new, best_n)
                if c5_new < best_c5 - 1e-12:
                    best_h = h_new.copy()
                    best_c5 = c5_new

    # Phase 4: Try alternative n values (12% budget)
    t4_end = t0 + budget_s * 0.87
    if time.time() < t4_end:
        for n_alt in [50, 55, 60, 65, 75, 80, 90]:
            if time.time() > t4_end:
                break
            if n_alt == best_n:
                continue
            for kk in range(2):
                if time.time() > t4_end:
                    break
                h_alt = make_start(n_alt, seed + n_alt * 31 + kk * 7, kk % 6)
                h_alt = lbfgs_run(h_alt, n_alt, 50000, 3, 0.01, 500, t4_end)
                h_alt = lbfgs_run(h_alt, n_alt, 300000, 5, 0.02, 400, t4_end)
                c5_alt, _ = compute_c5(h_alt, n_alt)
                if c5_alt < best_c5 - 1e-12:
                    best_h = h_alt.copy()
                    best_c5 = c5_alt
                    best_n = n_alt

    # Phase 5: Final anneal + L-BFGS polish (13% budget)
    t5_end = t_end - 1.0
    if best_h is not None and time.time() < t5_end:
        h_fin, c5_fin = anneal_coord(best_h, best_n, 15000, t5_end * 0.6, rng, T_start=0.004, T_end=0.0001)
        if c5_fin < best_c5 - 1e-12:
            best_h = h_fin.copy()
            best_c5 = c5_fin
        if time.time() < t5_end:
            h_pol = lbfgs_run(best_h.copy(), best_n, 500000, 5, 0.015, 600, t5_end)
            c5_pol, _ = compute_c5(h_pol, best_n)
            if c5_pol < best_c5 - 1e-12:
                best_h = h_pol.copy()
                best_c5 = c5_pol
        # Tiny final coordinate descent
        if time.time() < t5_end:
            h_t, c5_t = anneal_coord(best_h, best_n, 5000, t5_end, rng, T_start=0.001, T_end=0.00005)
            if c5_t < best_c5 - 1e-12:
                best_h = h_t.copy()
                best_c5 = c5_t

    if best_h is None:
        best_n = 70
        best_h = repair(np.full(best_n, 0.5), best_n)
        best_c5, _ = compute_c5(best_h, best_n)

    best_h = repair(best_h, best_n)
    best_c5, _ = compute_c5(best_h, best_n)
    return (best_h, best_c5, best_n)
