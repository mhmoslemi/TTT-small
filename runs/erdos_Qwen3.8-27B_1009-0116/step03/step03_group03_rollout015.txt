    return np.correlate(h, 1.0 - h, mode='full')

    return np.max(_corr(h)) * dx

def _compute_G(h, n):
    """Compute gradient matrix G where G[m,j] = d(c_m)/d(h_j)."""
    K = 2*n - 1
    G = np.zeros((K, n))
    for m in range(K):

            G[m, :n-kk] += 1.0 - h[kk:]
            # Second term: -h[j-kk] for j in [kk, n-1]
            G[m, kk:] -= h[:n-kk]

    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)

            else:
                add = max(remaining, h[idx] - 1.0)
            h[idx] += add
            remaining -= add
            if abs(remaining) < 1e-12:
                break
    return h

                G = _compute_G(h, n)
                improved = True
                if np.max(c) < best_c5:
                    best_c5 = np.max(c)
        
        if not improved:
            break
    
    return h, best_c5

def _solve_1d_fast(v, D, L, U):
    """Minimize max_m(v[m] + u*D[m]) over u in [L, U]. Returns (u*, f*)."""
    # Quick check at boundaries
    fL = np.max(v + L * D)
    fU = np.max(v + U * D)
    
    # Check subgradient at L
    active_L = (v + L * D) >= fL - 1e-15
    if np.min(D[active_L]) >= 0:
        return L, fL
    # Check subgradient at U  
    active_U = (v + U * D) >= fU - 1e-15
    if np.max(D[active_U]) <= 0:
        return U, fU
    
    # Bisection for interior minimum
    lo, hi = L, U
    for _ in range(25):
        mid = 0.5 * (lo + hi)
        f_mid_act = v + mid * D
        max_mid = np.max(f_mid_act)
        active = f_mid_act >= max_mid - 1e-15
        # If min active D > 0, function is increasing → go left
        # If max active D < 0, function is decreasing → go right
        if np.min(D[active]) > 1e-15:
            hi = mid
        elif np.max(D[active]) < -1e-15:
            lo = mid
        else:
            # 0 is in subgradient, we're at the minimum
            break
    u = 0.5 * (lo + hi)
    return u, np.max(v + u * D)

def _compute_G(h, n):
    K = 2 * n - 1
    G = np.zeros((K, n))

            G[m, :n - k] -= h[k:]
