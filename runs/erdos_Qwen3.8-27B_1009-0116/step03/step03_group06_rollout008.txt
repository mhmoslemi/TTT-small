    if remaining() > 400:
        tEnd = time.time() + min(120, remaining() - 350)
        n_c = max(14, n // 5)
        # Downsample best_h to coarse
        idx_c = np.linspace(0, n - 1, n_c).astype(int)
        hc = best_h[idx_c].copy()
        hc = _repair(hc, n_c)
        # Optimize coarse
        for alpha in [100, 500, 2000]:
            if time.time() >= tEnd:
                break
            hct, c5ct = _opt_slsqp(hc, n_c, 2.0 / n_c, alpha, maxiter=300)
            hc = hct
        # Upsample back
        hf = _coarse_to_fine(hc, n_c, n, dx)
        # Refine fine
        for alpha in [2000, 10000]:
            if time.time() >= tEnd:
                break
            hft, c5ft = _opt_slsqp(hf, n, dx, alpha, maxiter=400)
            if c5ft < best_c5:
                best_h, best_c5 = hft, c5ft

    # Phase 4: Multi-restart SLSQP (180s)
    tEnd = time.time() + min(180, remaining() - 300)
    for r in range(10):
        if time.time() >= tEnd or remaining() < 250:
            break
        if r == 0:
            start = best_h + rng.normal(0, 0.05, n)
        elif r == 1:
            start = _make_sinusoidal(n, rng, 3)
        elif r == 2:
            start = _make_sinusoidal(n, rng, 2)
        elif r == 3:
            # Block-permuted version
            blk = n // 7
            start = np.zeros(n)
            for b in range(7):
                seg = best_h[b*blk:(b+1)*blk] if b*blk < n else best_h[b*blk:]
                start[b*blk:b*blk+len(seg)] = seg
            # Apply random cyclic shift within one block
            k = rng.randint(0, blk)
            if k > 0 and k < n:
                start = np.concatenate([start[k:], start[:k]])
        else:
            start = best_h + rng.uniform(-0.08, 0.08, n)
        start = _repair(start, n)
        ht, c5t = _opt_slsqp(start, n, dx, 5000, maxiter=500)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 5: Targeted local search (150s)
    if remaining() > 200:
        tEnd = time.time() + min(150, remaining() - 180)
        ht, c5t = _targeted_local(best_h, n, dx, rng, tEnd, n_trials=40000)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 6: Block shift search (100s)
    if remaining() > 150:
        tEnd = time.time() + min(100, remaining() - 130)
        ht, c5t = _block_shift_search(best_h, n, dx, rng, tEnd, n_iters=3000)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 7: Annealing (100s)
    if remaining() > 130:
        tEnd = time.time() + min(100, remaining() - 110)
        ht, c5t = _anneal(best_h, n, dx, rng, tEnd)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 8: High-alpha SLSQP polish (100s)
    if remaining() > 100:
        tEnd = time.time() + min(100, remaining() - 80)
        for alpha in [30000, 100000, 300000]:
            if time.time() >= tEnd:
                break
            ht, c5t = _opt_slsqp(best_h, n, dx, alpha, maxiter=600)
            if c5t < best_c5:
                best_h, best_c5 = ht, c5t

    # Phase 9: Ultra-fine targeted search (60s)
    if remaining() > 60:
        tEnd = time.time() + min(60, remaining() - 40)
        ht, c5t = _targeted_local(best_h, n, dx, rng, tEnd, n_trials=30000)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 10: Final SLSQP + tiny local (40s)
    if remaining() > 40:
        tEnd = time.time() + min(30, remaining() - 15)
        ht, c5t = _opt_slsqp(best_h, n, dx, 200000, maxiter=400)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t
        if remaining() > 10:
            tEnd2 = time.time() + max(2, min(8, remaining() - 5))
            ht, c5t = _targeted_local(best_h, n, dx, rng, tEnd2, n_trials=10000)
            if c5t < best_c5:
                best_h, best_c5 = ht, c5t

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
