def pair_derivative(h, i, j, n):
    """Compute d(corr)/d(d) for h[i]+=d, h[j]-=d, for all lags."""
    derivs = np.zeros(2*n - 1)
    for k in range(2*n - 1):
        L = k - (n - 1)
        if L == 0:
            d_i = 1 - 2*h[i]
            d_j = -(1 - 2*h[j])
        elif L > 0:

            if j < n - L:

    n = len(x)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        pk = pw[k]
        if pk < 1e-14:
            continue
        if L == 0:
            g += pk * (1.0 - 2.0 * x)
        elif L > 0:
            g[L:] += pk * (1.0 - x[:n - L])
            g[:n - L] -= pk * x[L:]
        else:
            M = -L
            g[:n - M] += pk * (1.0 - x[M:])
            g[M:] -= x[:n - M] * pk
    g *= 1.0 / alpha
    g += 2.0 * lam * (np.sum(x) - n / 2.0)
    return g

def _pair_deriv(h, i, j, n):
    rd = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        di = 0.0
        dj = 0.0
        if L == 0:
            di = 1.0 - 2.0 * h[i]
            dj = 2.0 * h[j] - 1.0
        elif L > 0:
            if i < n - L:
                di += 1.0 - h[i + L]
            if i >= L:
                di -= h[i - L]
            if j < n - L:
                dj -= 1.0 - h[j + L]
            if j >= L:
                dj += h[j - L]
        else:
            M = -L
            if i >= M:
                di += 1.0 - h[i - M]
            if i < n - M:
                di -= h[i + M]
            if j >= M:
                dj -= 1.0 - h[j - M]
            if j < n - M:
                dj += h[j + M]
        rd[k] = di + dj
    return rd

def _best_pair_d(c_cur, h, i, j, n):
    dv = _pair_deriv(h, i, j, n)
    dlo = max(-h[i], -h[j])
    dhi = min(1.0 - h[i], h[j])
    if dhi - dlo < 1e-12:
        return 0.0
    f0 = float(np.max(c_cur))
    for _ in range(25):
        m1 = dlo + (dhi - dlo) / 3.0
        m2 = dhi - (dhi - dlo) / 3.0
        f1 = float(np.max(c_cur + m1 * dv))
        f2 = float(np.max(c_cur + m2 * dv))
        if f1 < f2:
            dhi = m2
        else:
            dlo = m1
    d = (dlo + dhi) / 2.0
    fd = float(np.max(c_cur + d * dv))
    if fd >= f0 - 1e-14:
        return 0.0
    return d

def _pair_sweep(h, c5_cur, n, dx, rng, max_sweeps, deadline, max_pairs_per_i):
    best_h = h.copy()
    best_c5 = c5_cur
    c_cur = _corr(h)
    for _sw in range(max_sweeps):
        if time.time() > deadline:
            break
        imp = False
        perm = rng.permutation(n)
        for a in range(n):
            if time.time() > deadline:
                break
            i = int(perm[a])
            for b in range(a + 1, min(a + max_pairs_per_i, n)):
                j = int(perm[b])
                d = _best_pair_d(c_cur, h, i, j, n)
                if abs(d) < 1e-12:
                    continue
                h[i] += d
                h[j] -= d
                if h[i] < 0 or h[i] > 1 or h[j] < 0 or h[j] > 1:
                    h[i] -= d
                    h[j] += d
                    continue
                c_new = _corr(h)
                c5_new = float(np.max(c_new)) * dx
                if c5_new < c5_cur - 1e-15:
                    c_cur = c_new
                    c5_cur = c5_new
                    imp = True
                    best_h = h.copy()
                    best_c5 = c5_new
                else:
                    h[i] -= d
                    h[j] += d
        if not imp:
            break
    return best_h, best_c5

def _sum_eq(x, target):
    return np.sum(x) - target

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    h0 = _repair(np.array(initial_h_values, dtype=float), n)
    best_h = h0.copy()
    best_c5 = _c5(best_h, dx)

    def _elapsed():
        return time.time() - t0

    # Phase 1: Analytical pair coordinate descent (wide search)
    dl1 = t0 + budget_s * 0.45
    best_h, best_c5 = _pair_sweep(best_h, best_c5, n, dx, rng,
                                  800, dl1, 8)

    # Phase 2: L-BFGS-B soft-max refinement
    if _elapsed() < budget_s - 250:
        for a in [500, 2000, 10000, 50000]:
            if _elapsed() > budget_s - 250:
                break
            try:
                res = minimize(partial(_softobj, alpha=a, lam=60.0),
                               best_h.copy(),
                               jac=partial(_softgrad, alpha=a, lam=60.0),
                               method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': 350, 'ftol': 1e-17})
                hc = _repair(res.x, n)
                c5 = _c5(hc, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = hc
            except Exception:
                pass

    # Phase 3: SLSQP with exact equality
    if _elapsed() < budget_s - 180:
        try:
            res = minimize(partial(_softobj, alpha=20000, lam=0.0),
                           best_h,
                           method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': partial(_sum_eq, target=target)}],
                           options={'maxiter': 500, 'ftol': 1e-17})
            hc = _repair(res.x, n)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hc
        except Exception:
            pass

    # Phase 4: Pair descent again (narrower, after smooth opt)
    dl4 = t0 + budget_s * 0.78
    best_h, best_c5 = _pair_sweep(best_h, best_c5, n, dx, rng,
                                  600, dl4, 5)

    # Phase 5: Nesterov subgradient on top-1 with perturbations
    if _elapsed() < budget_s - 60:
        h = best_h.copy()
        v = np.zeros(n)
        step = 0.005
        for _it in range(3000):
            if _elapsed() > budget_s - 60:
                break
            c = _corr(h)
            bad_k = int(np.argmax(c))
            L = bad_k - (n - 1)
            g = np.zeros(n)
            if L == 0:
                g[:] = 1.0 - 2.0 * h
            elif L > 0:
                g[L:] += (1.0 - h[:n - L])
                g[:n - L] -= h[L:]
            else:
                M = -L
                g[:n - M] += (1.0 - h[M:])
                g[M:] -= h[:n - M]
            g -= g.mean()
            step = max(1e-6, step * 0.999)
            h_new = _repair(h + v - step * g, n)
            v = h_new - h
            c5_new = _c5(h_new, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
            else:
                h = h_new  # keep momentum going, don't revert
        # Kick with sinusoidal perturbation
        for freq in range(2, 8):
            if _elapsed() > budget_s - 50:
                break
            for amp in [0.01, 0.02, 0.04]:
                if _elapsed() > budget_s - 50:
                    break
                pert = amp * np.sin(2.0 * np.pi * freq * np.arange(n) / n)
                pert -= pert.mean()
                hc = _repair(best_h + pert, n)
                c5 = _c5(hc, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = hc

    # Phase 6: Final micro-polish
    if _elapsed() < budget_s - 5:
        h = best_h.copy()
        c5_cur = best_c5
        for _ in range(80000):
            if _elapsed() > budget_s - 5:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = rng.uniform(-0.004, 0.004)
            ni, nj = h[i] + d, h[j] - d
            if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                continue
            h[i], h[j] = ni, nj
            c5_new = _c5(h, dx)
            if c5_new < c5_cur - 1e-15:
                c5_cur = c5_new
                best_c5 = c5_cur
                best_h = h.copy()
            else:
                h[i] -= d
                h[j] += d

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
