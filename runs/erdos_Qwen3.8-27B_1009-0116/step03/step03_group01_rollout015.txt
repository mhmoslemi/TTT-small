    lo = y.min() - target/len(y)

def corr_grad(h, offset):
    n = len(h)
    g = np.zeros(n)
    if offset >= 0:
        # c = sum_{i=0}^{n-1-offset} h[i+offset] * (1-h[i])
        # ∂c/∂h[j] from h[i+offset] term: (1-h[i]) where i = j-offset, valid if 0 ≤ j-offset and j-offset ≤ n-1-offset
        #   => offset ≤ j ≤ n-1
        lo1 = offset
        hi1 = n - 1
        g[lo1:hi1+1] += (1 - h[lo1-offset:hi1-offset+1])
        # ∂c/∂h[j] from (1-h[i]) term: -h[i+offset] where i = j, valid if 0 ≤ j ≤ n-1-offset
        lo2 = 0
        hi2 = n - 1 - offset
        g[lo2:hi2+1] -= h[lo2+offset:hi2+offset+1]
    else:
        m = -offset
        # c = sum_{i=0}^{n-1-m} h[i] * (1-h[i+m])
        # ∂c/∂h[j] from h[i] term: (1-h[i+m]) where i = j, valid if 0 ≤ j ≤ n-1-m
        lo1 = 0
        hi1 = n - 1 - m
        g[lo1:hi1+1] += (1 - h[lo1+m:hi1+m+1])
        # ∂c/∂h[j] from (1-h[i+m]) term: -h[i] where i+m = j, i = j-m, valid if m ≤ j ≤ n-1
        lo2 = m
        hi2 = n - 1
        g[lo2:hi2+1] -= h[lo2-m:hi2-m+1]
    return g

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / len(h))

def _sf(x, a):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = np.clip(a * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / a

def _corr_grad(h, offset):
    n = len(h)
    g = np.zeros(n)
    if offset >= 0:
        if offset < n:
            g[offset:] += (1.0 - h[:n - offset])
            g[:n - offset] -= h[offset:]
    else:
        m = -offset
        if m < n:
            g[:n - m] += (1.0 - h[m:])
            g[m:] -= h[:n - m]
    return g

    n = len(y)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(y - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid

def _substep(h, n, lr, c5_target):
    c = np.correlate(h, 1.0 - h, mode='full')
    k_star = np.argmax(c)
    offset = k_star - (n - 1)

    h_new = h - step * (2.0 / n) * g

            d = rng.choice([1.0, 1.5, 3.0]) * 5e-5 * rng.choice([-1.0, 1.0])
            ni, nj = h[i] + d, h[j] - d
            if 0 <= ni <= 1 and 0 <= nj <= 1:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-17:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5 = c5n
                        best_h = h.copy()
                else:
                    h[i], h[j] = h[i] - d, h[j] + d

    best_h = _repair(best_h, len(best_h))
    best_c5 = _eval(best_h)
    return (best_h, best_c5, len(best_h))
