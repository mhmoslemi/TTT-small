import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _lp_obj(x, p, lam, n):
    dx = 2.0 / n
    c = _corr(x) * dx
    lp = np.sum(c**p) ** (1.0/p)
    penalty = lam * (np.sum(x) - n/2.0)**2
    return lp + penalty

def _lp_grad(x, p, lam, n):
    dx = 2.0 / n
    c = _corr(x) * dx
    lp = np.sum(c**p) ** (1.0/p)
    if lp < 1e-30:
        return np.zeros(n) + 2.0*lam*(np.sum(x) - n/2.0)
    w = c**(p-1) * dx / lp  # weight for each correlation
    g = np.zeros(n)
    for k in range(2*n - 1):
        L = k - (n-1)
        wk = w[k]
        if wk < 1e-15:
            continue
        if L == 0:
            g += wk * (1.0 - 2.0*x)
        elif L > 0:
            g[L:] += wk * (1.0 - x[:n-L])
            g[:n-L] -= wk * x[L:]
        else:
            M = -L
            g[:n-M] += wk * (1.0 - x[M:])
            g[M:] -= wk * x[:n-M]
    g += 2.0 * lam * (np.sum(x) - n/2.0)
    return g

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-13:
                break
    return h

            g += (1.0 - 2.0 * h)
        elif L > 0:
            g[L:] += (1.0 - h[:n - L])
            g[:n - L] -= h[L:]
        else:
            M = -L
            g[:n - M] += (1.0 - h[M:])
            g[M:] -= h[:n - M]
    g -= g.mean()
    return g

def _pair_cd(h, n, dx, rng, budget_end):
    c5_cur = _c5(h, dx)
    steps = [0.01, 0.005, 0.002, 0.001]
    for _sweep in range(300):
        if time.time() > budget_end:
            break
        c = _corr(h)
        k_max = np.argmax(c)
        L = k_max - (n - 1)
        g = np.zeros(n)
        if L == 0:
            g = 1.0 - 2.0 * h
        elif L > 0:
            g[L:] += (1.0 - h[:n - L])
            g[:n - L] -= h[L:]
        else:
            M = -L
            g[:n - M] += (1.0 - h[M:])
            g[M:] -= h[:n - M]
        found = False
        i_dec = np.argmax(g)
        i_inc = np.argmin(g)
        if i_dec != i_inc:
            for s in steps:
                ni = h[i_dec] - s
                nj = h[i_inc] + s
                if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                    h[i_dec], h[i_inc] = ni, nj
                    c_new = _c5(h, dx)
                    if c_new < c5_cur - 1e-16:
                        c5_cur = c_new
                        found = True
                        break
                    h[i_dec], h[i_inc] = ni + s, nj - s
        if not found:
            found = False
            for _try in range(30):
                i, j = rng.randint(n), rng.randint(n)
                if i == j:
                    continue
                for s in [0.005, 0.002, 0.001]:
                    for sign in [1, -1]:
                        ni = h[i] + sign * s
                        nj = h[j] - sign * s
                        if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                            h[i], h[j] = ni, nj
                            c_new = _c5(h, dx)
                            if c_new < c5_cur - 1e-16:
                                c5_cur = c_new
                                found = True
                                break
                            h[i], h[j] = ni - sign * s, nj + sign * s
                    if found:
                        break
                if found:
                    break
            if not found:
                break
    return h, c5_cur

def _topk_equalize(h, n, dx, top_k, rng, budget_end):
    """Iteratively push down the top-k correlations by gradient steps."""
    c5_cur = _c5(h, dx)
    for step in [0.04, 0.02, 0.01, 0.005, 0.002, 0.001]:
        for _it in range(800):
            if time.time() > budget_end:
                return h, c5_cur
            h_new = _repair(h - step * _active_grad(h, top_k, n), n)
            c5_new = _c5(h_new, dx)
            if c5_new < c5_cur - 1e-16:
                h = h_new
                c5_cur = c5_new
            else:
                break
    return h, c5_cur

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    budget_end = t0 + budget_s

    h = _repair(np.array(initial_h_values, dtype=float), n)
    best_h = h.copy()
    best_c5 = _c5(best_h, dx)

    # Phase 1: Lp norm optimization with increasing p
    p_list = [4, 8, 16, 32, 64, 128, 256]
    for p in p_list:
        if time.time() > budget_end - 200:
            break
        for lam in [200.0, 2000.0]:
            if time.time() > budget_end - 180:
                break
            try:
                obj = _LpFun(p, lam, n)
                res = minimize(obj.f, h, jac=obj.g, method='L-BFGS-B',
                               bounds=bounds,
                               options={'maxiter': 600, 'ftol': 1e-19, 'gtol': 1e-14})
                h_new = _repair(res.x, n)
                c5_new = _c5(h_new, dx)
                if c5_new < _c5(h, dx):
                    h = h_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
            except Exception:
                pass

    # Phase 2: Top-k active gradient with decreasing step
    h = best_h.copy()
    for top_k in [3, 5, 7, 10]:
        if time.time() > budget_end - 150:
            break
        h, c5_cur = _topk_equalize(h, n, dx, top_k, rng, budget_end - 30)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h.copy()

    # Phase 3: SLSQP with exact equality constraint
    if time.time() < budget_end - 200:
        for p in [32, 128]:
            if time.time() > budget_end - 150:
                break
            try:
                obj = _LpFun(p, 0.0, n)
                con = _EqCon(n)
                res = minimize(obj.f, best_h, method='SLSQP',
                               bounds=bounds,
                               constraints=[{'type': 'eq', 'fun': con.f}],
                               options={'maxiter': 600, 'ftol': 1e-19})
                h_new = _repair(res.x, n)
                c5_new = _c5(h_new, dx)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new
            except Exception:
                pass

    # Phase 4: Pair coordinate descent
    if time.time() < budget_end - 80:
        h = best_h.copy()
        h, c5_cur = _pair_cd(h, n, dx, rng, budget_end - 50)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h.copy()

    # Phase 5: Multi-restart with small perturbations
    if time.time() < budget_end - 80:
        for _r in range(6):
            if time.time() > budget_end - 60:
                break
            hp = best_h + 0.005 * rng.randn(n)
            hp = _repair(hp, n)
            try:
                p_use = 256
                lam_use = 500.0
                obj = _LpFun(p_use, lam_use, n)
                res = minimize(obj.f, hp, jac=obj.g, method='L-BFGS-B',
                               bounds=bounds,
                               options={'maxiter': 300, 'ftol': 1e-19})
                h_new = _repair(res.x, n)
                c5_new = _c5(h_new, dx)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
            except Exception:
                pass

    # Phase 6: Final ultra-fine polish
    if time.time() < budget_end - 10:
        h = best_h.copy()
        c5_cur = _c5(h, dx)
        for _ in range(50000):
            if time.time() > budget_end - 5:
                break
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = rng.uniform(-0.002, 0.002)
            ni, nj = h[i] + d, h[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h[i], h[j] = ni, nj
                c = _c5(h, dx)
                if c < c5_cur:
                    c5_cur = c
                    best_c5 = c
                    best_h = h.copy()
                else:
                    h[i], h[j] = ni - d, nj + d
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h.copy()

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
