    return np.correlate(h, 1.0 - h, mode='full')

    return np.max(_corr(h)) * dx

    c = _corr(x)
    max_c = c.max()
    s = alpha * (c - max_c)
    s = np.clip(s, -50, 50)
    return max_c + np.log(np.sum(np.exp(s))) / alpha

def _soft_grad(x, alpha):
    c = _corr(x)
    max_c = c.max()
    s = alpha * (c - max_c)
    s = np.clip(s, -50, 50)
    exps = np.exp(s)
    weights = exps / exps.sum()  # softmax weights
    # Gradient of c_k w.r.t. x
    # c_k = sum_i x[i] * (1 - x[i+k_eff])  (with appropriate boundary)
    # dc_k/dx[i] = (1-x[i+k]) - x[i-k]  (where defined)
    n = len(x)
    g = np.zeros(n)
    for m in range(2*n-1):
        if weights[m] < 1e-10:
            continue
        k = m - (n-1)
        if k >= 0:
            # c[m] = sum_{i=0}^{n-1-k} x[i] * (1-x[i+k])
            g[k:] += (1.0 - x[:n-k]) * weights[m]
            g[:n-k] -= x[k:] * weights[m]
        else:
            # k < 0, let kk = -k
            kk = -k
            # c[m] = sum_{i=0}^{n-1-kk} x[i+kk] * (1-x[i])
            # = sum_{i=kk}^{n-1} x[i] * (1-x[i-kk])
            g[kk:] += (1.0 - x[:n-kk]) * weights[m]
            g[:n-kk] -= x[kk:] * weights[m]
    return g

    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
            else:
                add = max(diff, h[idx] - 1.0)
            h[idx] += add
            diff -= add
            if abs(diff) < 1e-10:
                break
    return h

    """Project onto {0<=h<=1, sum(h)=n/2}"""
    return _repair(h, n)

    return h, True

Wait, this is O(n) per evaluation (for np.correlate), and we do ~40 evaluations per pair. For n=70, np.correlate of two length-70 arrays produces a length-139 array. This is very fast, maybe 5μs. So 40 * 5μs = 200μs per pair step. 

If I do 70 coordinate pairs per sweep (or all pairs), that's 70 * 200μs = 14ms per sweep. I can do thousands of sweeps in the time budget.

But wait, I don't want to do all 70*69/2 = 2415 pairs. I'll do a sequential scan: for i=0 to n-1, for j=i+1 to n-1, do the pair step. That's 2415 steps per full sweep at 200μs each = ~0.5s per sweep. I can do about 2000 sweeps in 1000s. That should be more than enough to converge.

Actually, the golden section search takes 40 iterations, but many of the pairs won't improve (if the current solution is good). I can add a quick check: evaluate at δ=0 (current), δ=d_lo, δ=d_hi, and the midpoint. If none improve, skip the full search.

Let me also add a "restart" mechanism: if we haven't improved in X iterations, perturb the solution.

import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            add = min(diff, 1.0 - h[idx]) if diff > 0 else max(diff, h[idx] - 1.0)
            h[idx] += add
            diff -= add
            if abs(diff) < 1e-10:
                break
    return h

def _soft_obj(x, alpha):
    c = _corr(x)
    mc = c.max()
    s = np.clip(alpha * (c - mc), -40, 40)
    return mc + np.log(np.sum(np.exp(s))) / alpha

def _pair_step(h, n, i, j):

    d_lo = max(-hi, hj - 1.0)
    d_hi = min(1.0 - hi, hj)
    if d_hi - d_lo < 1e-14:
        return False
    dx = 2.0 / n

        h2 = h.copy()
        h2[i] = hi + d
        h2[j] = hj - d
        return float(np.max(_corr(h2)) * dx)

    f0 = f(0.0)

    if fm >= f0 - 1e-14 and f(d_lo) >= f0 - 1e-14 and f(d_hi) >= f0 - 1e-14:
        return False

    x1 = b - gr * (b - a)
    x2 = a + gr * (b - a)

            fx1 = f(x1)

            a, x1, fx1 = x1, x2, fx2
            x2 = a + gr * (b - a)
            fx2 = f(x2)
    best_d = (a + b) / 2.0

        best_d = d_lo
    elif f(d_hi) < f(best_d):
        best_d = d_hi
    h[i] = hi + best_d
    h[j] = hj - best_d
