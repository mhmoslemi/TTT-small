    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return np.max(_corr(h)) * dx

def _softmax_obj(h, beta):
    c = _corr(h)
    c_max = np.max(c)
    # Shift for numerical stability
    s = beta * (c - c_max)
    return c_max + np.log(np.sum(np.exp(s))) / beta

def _softmax_grad(h, beta):
    c = _corr(h)
    c_max = np.max(c)
    s = beta * (c - c_max)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    n = len(h)
    g = np.zeros(n)
    for j in range(2*n-1):

        if abs(w[j]) < 1e-15:

            g[k:] += w[j] * (1.0 - h[:n-k])
            g[:n-k] -= w[j] * h[k:]
        else:
            g[:n+k] += w[j] * (1.0 - h[-k:])
            g[-k:] -= w[j] * h[:n+k]
