    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _c5_val(best_h, dx)

    def tleft():
        return budget_s - (time.time() - t0)

    # Phase 1: SLSQP from multiple starts
    starts = [h0.copy()]
    r2 = 1.0 - h0[::-1].copy()
    starts.append(_repair(r2, n))
    sh = np.roll(h0, n // 5)
    starts.append(_repair(sh, n))
    for h_start in starts:
        if tleft() < 80:
            break
        for alpha in [10000, 50000]:
            if tleft() < 60:
                break
            try:
                res = minimize(
                    _smooth_obj, h_start, args=(alpha,),
                    method='SLSQP', bounds=bounds,
                    constraints=[{'type': 'eq', 'fun': _sum_con, 'args': (target,)}],
                    options={'maxiter': 400, 'ftol': 1e-14, 'iprint': 0})
                hc = _repair(res.x, n)
                c5 = _c5_val(hc, dx)
                if c5 < best_c5:
                    best_c5, best_h = c5, hc
            except Exception:
                pass

    # Phase 2: Subgradient with momentum
    if tleft() > 60:
        h = best_h.copy()
        c5_cur = _c5_val(h, dx)
        momentum = np.zeros(n)
        beta_m = 0.82
        for it in range(120000):
            if tleft() < 60:
                break
            c = _corr(h)
            max_c = c.max()
            if it < 5000:
                tf = 0.99
            elif it < 20000:
                tf = 0.97
            else:
                tf = 0.95
            active = np.where(c >= tf * max_c)[0]
            g = np.mean([_grad_lag(h, n, m) for m in active], axis=0)
            lr = 0.035 / (1.0 + 0.0002 * it)
            momentum = beta_m * momentum + (1.0 - beta_m) * g
            h_cand = _repair(h - lr * momentum, n)
            c5_new = _c5_val(h_cand, dx)
            if c5_new < c5_cur - 1e-15:
                h = h_cand
                c5_cur = c5_new
                if c5_cur < best_c5:
                    best_c5, best_h = c5_cur, h.copy()

    # Phase 3: Gradient-guided pair descent
    if tleft() > 35:
        h = best_h.copy()
        c5_cur = _c5_val(h, dx)
        stall = 0
        for rnd in range(30000):
            if tleft() < 35:
                break
            c_arr = _corr(h)
            max_c = c_arr.max()
            active = np.where(c_arr >= 0.995 * max_c)[0]
            g = np.mean([_grad_lag(h, n, m) for m in active], axis=0)
            order = np.argsort(g)
            inc = order[:20]
            dec = order[-20:][::-1]
            found = False
            for di in inc:
                if found or h[di] >= 1.0 - 1e-14:
                    continue
                for dj in dec:
                    if found or dj == di or h[dj] <= 1e-14:
                        continue
                    for d in (0.003, 0.001, 0.005, 0.0005):
                        ni_val = h[di] + d
                        nj_val = h[dj] - d
                        if 0.0 <= ni_val <= 1.0 and 0.0 <= nj_val <= 1.0:
                            h[di] = ni_val
                            h[dj] = nj_val
                            nc = _c5_val(h, dx)
                            if nc < c5_cur - 1e-14:
                                c5_cur = nc
                                found = True
                                break
                            h[di] = ni_val - d
                            h[dj] = nj_val + d
                        else:
                            break
            if found:
                stall = 0
                if c5_cur < best_c5:
                    best_c5, best_h = c5_cur, h.copy()
            else:
                stall += 1
                if stall > 300:
                    break

    # Phase 4: Low-temperature simulated annealing
    if tleft() > 25:
        h = best_h.copy()
        c5_cur = _c5_val(h, dx)
        T = 0.0004
        cool = 0.99998
        for it in range(400000):
            if tleft() < 25:
                break
            n_p = 1 + rng.randint(2)
            idxs = rng.choice(n, n_p, replace=False)
            hc = h.copy()
            for idx in idxs:
                hc[idx] += rng.randn() * 0.012
            hc = _repair(hc, n)
            c5_new = _c5_val(hc, dx)
            delta = c5_new - c5_cur
            if delta < 0 or (T > 1e-10 and rng.rand() < np.exp(-delta / T)):
                h = hc
                c5_cur = c5_new
                if c5_cur < best_c5:
                    best_c5, best_h = c5_cur, h.copy()
            T *= cool

    # Phase 5: Ultra-fine random pair polish
    if tleft() > 10:
        h = best_h.copy()
        c5_cur = _c5_val(h, dx)
        dvals = [0.0001, 0.0003, 0.0005, 0.001]
        for _ in range(300000):
            if tleft() < 10:
                break
            i = rng.randint(n)
            j = (i + 1 + rng.randint(n - 1)) % n
            d = rng.choice([-1.0, 1.0]) * dvals[rng.randint(4)]
            ni_val = h[i] + d
            nj_val = h[j] - d
            if 0.0 <= ni_val <= 1.0 and 0.0 <= nj_val <= 1.0:
                h[i] = ni_val
                h[j] = nj_val
                nc = _c5_val(h, dx)
                if nc < c5_cur - 1e-15:
                    c5_cur = nc
                    if c5_cur < best_c5:
                        best_c5, best_h = c5_cur, h.copy()
                else:
                    h[i] = ni_val - d
                    h[j] = nj_val + d

    # Phase 6: Final SLSQP polish
    if tleft() > 20:
        try:
            res = minimize(
                _smooth_obj, best_h, args=(200000,),
                method='SLSQP', bounds=bounds,
                constraints=[{'type': 'eq', 'fun': _sum_con, 'args': (target,)}],
                options={'maxiter': 300, 'ftol': 1e-16, 'iprint': 0})
            hc = _repair(res.x, n)
            c5 = _c5_val(hc, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, hc
        except Exception:
            pass

    best_h = _repair(best_h, n)
    best_c5 = _c5_val(best_h, dx)
    return (best_h, best_c5, n)
