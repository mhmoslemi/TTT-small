                    break
                for d in deltas:
                    ni, nj = h[i] + d, h[j] - d
                    if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                        continue
                    h[i], h[j] = ni, nj
                    c = _c5(h, n)
                    if c < c5c - 1e-15:
                        c5c = c
                        imp = True
                    else:
                        h[i], h[j] = ni - d, nj + d
        if not imp:
            break
    return h, c5c

def _anneal(h, n, t_end, rng, best_ref):
    c5c = _c5(h, n)
    T, Tmin, cool = 0.002, 1e-10, 0.9998
    steps = np.array([0.001, 0.003, 0.01, 0.03, 0.1, 0.25])
    while time.time() < t_end:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = steps[rng.randint(len(steps))] * (1 if rng.rand() < 0.5 else -1)
        hi, hj = h[i], h[j]
        ni, nj = hi + d, hj - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        cn = _c5(h, n)
        if cn < c5c or (T > Tmin * 10 and rng.rand() < np.exp(max(0.0, -(cn - c5c) / T))):
            c5c = cn
            if c5c < best_ref[0]:
                best_ref[0] = c5c
                best_ref[1][:] = h
        else:
            h[i], h[j] = hi, hj
        T = max(T * cool, Tmin)
    return best_ref[0], best_ref[1].copy()

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def el():
        return time.time() - t0

    n = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _c5(best_h, n)

    # Phase 1: SLSQP gradient optimization
    for alpha in [1500, 4000, 12000, 35000]:
        if el() > budget_s - 350:
            break
        for r in range(3):
            if el() > budget_s - 350:
                break
            h_init = best_h.copy() if r == 0 else _repair(best_h + (0.02 + 0.02 * r) * rng.randn(n), n)
            try:
                res = _slsqp(h_init, n, alpha, 400)
                hc = _repair(res.x, n)
                c5 = _c5(hc, n)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = hc.copy()
            except Exception:
                pass

    # Phase 2: Basin hopping
    if el() < budget_s - 300:
        ref = [best_c5, best_h.copy()]
        for h_hop in range(6):
            if el() > budget_s - 300:
                break
            scale = 0.04 + 0.03 * h_hop
            h_bp = _repair(best_h + scale * rng.randn(n), n)
            h_bp, c5_bp = _local_search(h_bp, n, np.array([0.01, 0.03, 0.08, -0.01, -0.03, -0.08]), 15, t0 + budget_s - 300)
            if c5_bp < best_c5:
                best_c5 = c5_bp
                best_h = h_bp.copy()
        best_h, best_c5 = _anneal(best_h, n, t0 + budget_s - 200, rng, [best_c5, best_h.copy()])

    # Phase 3: Aggressive multi-scale local search
    for sf in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02]:
        if el() > budget_s - 100:
            break
        dl = np.array([0.01, 0.02, 0.04, 0.08, 0.15, -0.01, -0.02, -0.04, -0.08, -0.15]) * sf
        h_c = best_h.copy()
        h_c, c5 = _local_search(h_c, n, dl, 40, t0 + budget_s - 100)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_c.copy()

    # Phase 4: Higher resolution n=100
    if el() < budget_s - 120:
        n2 = 100
        xo = np.linspace(0, 2.0, n, endpoint=False) + 1.0 / n
        xn = np.linspace(0, 2.0, n2, endpoint=False) + 1.0 / n2
        h2 = np.interp(xn, xo, best_h)
        h2 = _repair(h2, n2)
        for alpha in [5000, 15000]:
            if el() > budget_s - 60:
                break
            try:
                res = _slsqp(h2, n2, alpha, 300)
                hc = _repair(res.x, n2)
                c5 = _c5(hc, n2)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = hc
                    n = n2
            except Exception:
                pass
        if n == n2 and el() < budget_s - 40:
            dl = np.array([0.005, 0.01, 0.02, 0.05, -0.005, -0.01, -0.02, -0.05])
            best_h, best_c5 = _local_search(best_h, n, dl, 30, t0 + budget_s - 30)

    # Phase 5: Ultra-fine local search
    if el() < budget_s - 15:
        for sf in [0.01, 0.005, 0.002, 0.001]:
            if el() > budget_s - 10:
                break
            dl = np.array([0.01, 0.03, 0.1]) * sf
            best_h, best_c5 = _local_search(best_h, n, dl, 60, t0 + budget_s - 10)

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, n)
    return (best_h, best_c5, n)
