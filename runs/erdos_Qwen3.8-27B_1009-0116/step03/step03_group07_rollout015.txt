import numpy as np
from scipy.optimize import minimize
import time

_st = {}

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(8000):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
        d = target - s
        if abs(d) < 1e-13:
            break
        if d > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if abs(d) < 1e-13:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
    return h

def obj_sm(h):
    alpha = _st['alpha']
    topk = _st['topk']
    fw = _st['fw']
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.minimum(s, 0.0)
    val = m + np.log(np.sum(np.exp(s))) / alpha
    if topk > 1 and fw > 0.0:
        sc = np.sort(corr)[::-1][:topk]
        val += fw * np.var(sc)
    return val

def obj_pen(h):
    return obj_sm(h) + _st['pen'] * (h.sum() - _st['target']) ** 2

def eq_con(h):
    return h.sum() - _st['target']

def do_slsqp(h, n, alpha, topk, fw, maxiter):
    _st['alpha'] = alpha
    _st['topk'] = topk
    _st['fw'] = fw
    _st['target'] = n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_sm, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': eq_con}],
                       options={'maxiter': maxiter, 'ftol': 1e-19})
        return repair(res.x, n)
    except Exception:
        return h

def do_lbfgs(h, n, alpha, topk, fw, pen, maxiter):
    _st['alpha'] = alpha
    _st['topk'] = topk
    _st['fw'] = fw
    _st['target'] = n / 2.0
    _st['pen'] = pen
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_pen, h, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-15})
        return repair(res.x, n)
    except Exception:
        return h

def coord_desc(h, n, t_end, rng):
    h = h.copy()
    best = compute_c5(h, n)
    stall = 0
    while time.time() < t_end:
        improved = False
        n_try = 3000
        for _ in range(n_try):
            if time.time() > t_end:
                break
            if rng.rand() < 0.3:
                hi_idx = np.where(h > 0.5)[0]

                    j = rng.randint(n)
                else:
                    i = hi_idx[rng.randint(len(hi_idx))]
                    j = lo_idx[rng.randint(len(lo_idx))]

            else:
                i = rng.randint(n)
                j = rng.randint(n)
            if i == j:
                continue
            hi, hj = h[i], h[j]
            for d in (0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0004):
                found = False
                for sgn in (1, -1):
                    delta = sgn * d

                    if ni < -1e-10 or ni > 1.0 + 1e-10 or nj < -1e-10 or nj > 1.0 + 1e-10:

                    h[i], h[j] = ni, nj

    return h, best

    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur_c = best_c
    t_start = time.time()

                best_h = h.copy()
        elif rng.rand() < np.exp(-(c - cur_c) / max(T, 1e-15)):
            cur_c = c
        else:
            h[i], h[j] = hi, hj
    return best_h, best_c

def make_start(n, seed_val, kind):

    else:

    return repair(h, n)
