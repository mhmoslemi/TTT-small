def _max_subgrad(h, n, top_k=5):
    """Compute subgradient of max_k corr[k] at h, averaging over top-k correlations."""
    c = _corr(h)
    # Get top-k indices
    idx = np.argsort(c)[::-1][:top_k]
    # Average weight (equal weight for simplicity, or proportional)
    g = np.zeros(n)
    for ki in idx:
        L = ki - (n - 1)
        if L == 0:
            g += (1.0 - 2.0 * h)
        elif L > 0:
            g[L:] += (1.0 - h[:n-L])
            g[:n-L] -= h[L:]
        else:
            M = -L
            g[:n-M] += (1.0 - h[M:])
            g[M:] -= h[:n-M]
    g /= top_k
    return g

from scipy.optimize import minimize

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h))) * dx

def _repair(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                h[idx] -= a; diff += a
            if abs(diff) < 1e-13:
                break
    return h

def _binary_round(h, n):
    order = np.argsort(h)[::-1]
    b = np.zeros(n, dtype=float)
    b[order[:n // 2]] = 1.0
    return b

def _subgrad_top(h, n, topk):
    c = _corr(h)
    idx = np.argsort(c)[::-1][:topk]
    g = np.zeros(n)
    for ki in idx:
        L = ki - (n - 1)
        if L == 0:
            g += (1.0 - 2.0 * h)
        elif L > 0:
            g[L:] += (1.0 - h[:n - L])
            g[:n - L] -= h[L:]
        else:
            M = -L
            g[:n - M] += (1.0 - h[M:])
            g[M:] -= h[:n - M]
    g /= topk
    return g

def _softobj(x, alpha, lam):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -100, 100)
    penalty = lam * (np.sum(x) - len(x) / 2.0) ** 2
    return m + np.log(np.sum(np.exp(s))) / alpha + penalty

def _softgrad(x, alpha, lam):
    c = _corr(x)
    m = np.max(c)
    s = np.clip(alpha * (c - m), -100, 100)
    ew = np.exp(s)
    pw = ew / ew.sum()
    n = len(x)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        pk = pw[k]
        if pk < 1e-14:
            continue
        if L == 0:
            g += pk * (1.0 - 2.0 * x)
        elif L > 0:
            g[L:] += pk * (1.0 - x[:n - L])
            g[:n - L] -= pk * x[L:]
        else:
            M = -L
            g[:n - M] += pk * (1.0 - x[M:])
            g[M:] -= pk * x[:n - M]
    g *= 1.0 / alpha
    g += 2.0 * lam * (np.sum(x) - n / 2.0)
    return g

def _slsqp_obj(x):
    c = _corr(x)
    m = np.max(c)
    alpha = 80000
    s = np.clip(alpha * (c - m), -100, 100)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _slsqp_con(x):
    return np.sum(x) - len(x) / 2.0
