def softmax_gradient(h, n, alpha):
    corr = compute_corr(h)
    dx = 2.0 / n
    k_offset = n - 1
    s = alpha * (corr - corr.max())
    w = np.exp(s) / np.sum(np.exp(s))
    grad = np.zeros(n)
    for idx in range(len(corr)):
        wi = w[idx]
        if wi < 1e-8:
            continue
        k = idx - k_offset

            grad[k:] -= wi * h[:n - k]
        else:
            kk = -k
            grad[kk:] += wi * (1.0 - h[:n - kk])
            grad[:n - kk] -= wi * h[kk:]
    return dx * grad

def projected_soft_gd(h, n, alpha, lr_init, n_iter, t_end):

    best = compute_c5(h, n)
    best_h = h.copy()
    lr = lr_init
    stuck = 0
    for it in range(n_iter):
        if time.time() > t_end:

def coord_descent(h, n, t_end, rng):

    best = compute_c5(h, n)
    steps = [0.05, 0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005]
    rounds = 0
    while time.time() < t_end:
        rounds += 1
        improved = False
        for _ in range(8 * n):
            if time.time() > t_end:
                return h, best
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            hi, hj = h[i], h[j]
            h[i], h[j] = hi, hj

    state_al = alpha

        corr = compute_corr(hv)

        return m + np.log(np.sum(np.exp(s))) / state_al

    def cons_eq(hv):
