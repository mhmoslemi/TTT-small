import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth_max(c, alpha):
    m = np.max(c)
    s = np.clip(alpha * (c - m), -300.0, 300.0)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_max_weights(c, alpha):
    m = np.max(c)
    s = np.clip(alpha * (c - m), -300.0, 300.0)
    e = np.exp(s)
    return e / e.sum()

def _grad_corr_h(h, w):
    n = len(h)
    gh = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        lag = k - (n - 1)
        for i in range(n):
            j = i + lag
            if 0 <= j < n:
                gh[i] += wk * (1.0 - h[j])
                gh[j] -= wk * h[i]
    return gh

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    amt = min(diff, 1.0 - h[idx])
                    h[idx] += amt
                    diff -= amt
                else:
                    amt = min(-diff, h[idx])
                    h[idx] -= amt
                    diff += amt
                if abs(diff) < 1e-12:
                    break
    return h

    a = np.clip(a, 0.0, 1.0)

    w = _smooth_max_weights(_corr(h), alpha)
    gh = _grad_corr_h(h, w)
    nh = len(a)
    return gh[:nh] - gh[nh:]

                   jac=lambda x, al: _sym_obj_jac(x, al),
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18})
    return np.clip(res.x, 0.0, 1.0)

                   options={'maxiter': maxiter, 'ftol': 1e-18})
    return res.x
