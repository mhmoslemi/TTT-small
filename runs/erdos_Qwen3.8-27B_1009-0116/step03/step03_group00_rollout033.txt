    v = beta2 * v + (1-beta2) * grad**2

def _c5(h):
    n = len(h)
    c = np.correlate(h, 1.0-h, mode='full') * (2.0/n)
    return float(np.max(c))

def _project(h, n):
    t = n/2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s-t) < 1e-13:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo+hi)/2.0
        p = np.clip(h+mid, 0.0, 1.0)
        if p.sum() > t:
            hi = mid
        else:
            lo = mid
    h = np.clip(h+(lo+hi)/2.0, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-13:
        h = np.clip(h+d/n, 0.0, 1.0)
    return h

def _sm_grad(h, alpha):
    n = len(h)
    dx = 2.0/n
    cr = np.correlate(h, 1.0-h, mode='full')
    c = cr*dx
    m = np.max(c)
    s = np.clip(alpha*(c-m), -300, 300)
    e = np.exp(s)
    w = e/np.sum(e)
    g = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-14:
            continue
        lag = k-(n-1)
        if lag >= 0:
            if lag > 0:
                g[lag:] += w[k]*(1.0-h[:n-lag])
                g[:n-lag] += w[k]*(-h[lag:])
            else:
                g[:] += w[k]*(1.0-h)
        else:
            g[:n+lag] += w[k]*(1.0-h[-lag:])
            g[-lag:] += w[k]*(-h[:n+lag])
    return g*dx

def _adam(h0, n, alpha, steps, lr, tend, b1=0.85, b2=0.95):

    m1 = np.zeros(n)
    m2 = np.zeros(n)
    bh = h.copy()
    bc = _c5(h)
    for t in range(steps):
        if time.time() > tend:
            break
        g = _sm_grad(h, alpha)
        m1 = b1*m1+(1-b1)*g
        m2 = b2*m2+(1-b2)*g*g
        th = t+1
        hh = h-lr*m1/(1-b1**th)/(np.sqrt(m2/(1-b2**th))+1e-8)
        hh = _project(hh, n)
        c5 = _c5(hh)
        if c5 < bc:

            bh = hh.copy()
        h = hh

        if time.time() > tend:
            break
        g = _tk_grad(h, k)
        m1 = b1*m1+(1-b1)*g
        m2 = b2*m2+(1-b2)*g*g
        th = t+1
        hh = h-lr*m1/(1-b1**th)/(np.sqrt(m2/(1-b2**th))+1e-8)
        hh = _project(hh, n)
        c5 = _c5(hh)
        if c5 < bc:
            bc = c5
            bh = hh.copy()
        h = hh
    return bh, bc
