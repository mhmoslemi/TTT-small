                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _slsqp_analytical(h0, n, alpha, maxiter=500, ftol=1e-16):
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_obj_grad, h0, args=(alpha,), jac=True,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'maxls': 50})
        if res.success or res.x is not None:
            return _repair(res.x, n)
    except Exception:
        pass
    return h0

def _gradient_descent(h, n, alpha, lr, n_steps, momentum=0.9):
    """Projected gradient descent with momentum."""
    target = n / 2.0
    vel = np.zeros(n)
    for step in range(n_steps):
        _, g, _ = _smooth_max_grad(h, alpha)
        vel = momentum * vel - lr * g
        h_new = h + vel
        h_new = np.clip(h_new, 0.0, 1.0)
        # Fix sum constraint
        diff = target - h_new.sum()
        h_new += diff / n
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = _repair(h_new, n)
        if not np.allclose(h_new, h):
            h = h_new
    return h

def _pair_sweep(h, n, corr, time_limit, n_coarse=20, n_bisect=25):
    """Coordinate pair sweep."""
    improved = False
    order = np.random.permutation(n)
    for ai, i in enumerate(order):
        for j in order[ai+1:]:
            if time.time() > time_limit:
                return h, corr, improved
            hi, hj = h[i], h[j]
            d_min = max(-hi, hj-1.0)
            d_max = min(1.0-hi, hj)
            if d_max - d_min < 1e-12:
                continue
            gd = _g_vec(h, i, n) - _g_vec(h, j, n)
            ds = np.linspace(d_min, d_max, n_coarse)
            vals = np.max(corr[None,:] + ds[:,None]*gd[None,:], axis=1)
            bi = np.argmin(vals)
            cur_max = np.max(corr)
            if vals[bi] >= cur_max - 1e-13:
                continue
            # Bisection
            lo = max(d_min, ds[max(0,bi-1)])
            hi_b = min(d_max, ds[min(n_coarse-1, bi+1)])
            for _ in range(n_bisect):
                mid = 0.5*(lo+hi_b)
                if np.max(corr + mid*gd) <= np.max(corr + hi_b*gd):
                    hi_b = mid
                else:
                    lo = mid
            d_best = 0.5*(lo+hi_b)
            h[i] = hi + d_best
            h[j] = hj - d_best
            corr = corr + d_best*gd
            improved = True
    return h, corr, improved

def _g_vec(h, p, n):
    kk = np.arange(-(n-1), n)
    g = np.zeros(2*n-1)
    idx1 = p - kk
    m1 = (idx1 >= 0) & (idx1 < n)
    g[m1] += (1.0 - h[idx1[m1]])
    idx2 = p + kk
    m2 = (idx2 >= 0) & (idx2 < n)
    g[m2] -= h[idx2[m2]]
    return g

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

    return float(np.max(_corr(h)) * dx)

def _smooth_max_val(c, alpha):
    m = np.max(c)
    s = np.exp(alpha * (c - m))
    return m + np.log(s.sum()) / alpha

def _smooth_max_grad(h, alpha):
    n = len(h)
    c = _corr(h)
    m = np.max(c)
    e = np.exp(alpha * (c - m))
    w = e / e.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-16:
            continue
        ell = k - (n - 1)
        i1_lo = max(0, -ell)
        i1_hi = min(n, n - ell)
        if i1_hi > i1_lo:
            grad[i1_lo:i1_hi] += w[k] * (1.0 - h[i1_lo + ell:i1_hi + ell])
        i2_lo = max(0, ell)
        i2_hi = min(n, n + ell)
        if i2_hi > i2_lo:
            grad[i2_lo:i2_hi] -= w[k] * h[i2_lo - ell:i2_hi - ell]
    return _smooth_max_val(c, alpha), grad

    return np.sum(x) - len(x) / 2.0
