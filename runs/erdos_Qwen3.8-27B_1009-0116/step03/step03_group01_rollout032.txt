def _eval(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / len(h))

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                if a <= 0: continue
                h[idx] += a
                diff -= a
            else:
                a = min(-diff, h[idx])
                if a <= 0: continue
                h[idx] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    return h

import numpy as np
from scipy.optimize import minimize, minimize_nelder_mead

def _eval(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                if a <= 0: continue
                h[idx] += a; diff -= a
            else:
                a = min(-diff, h[idx])
                if a <= 0: continue
                h[idx] -= a; diff += a
            if abs(diff) < 1e-12: break
    return h

def _subgrad(h, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    k_max = int(np.argmax(c))
    lag = k_max - (n - 1)
    g = np.zeros(n)
    if lag >= 0:
        g[lag:] += 1.0 - h[:n - lag]
        g[0:n - lag] -= h[lag:]
    else:
        L = -lag
        g[0:n - L] += 1.0 - h[L:]
        g[L:] -= h[0:n - L]
    return c.max() * (2.0 / n), g, c.copy()

def _slsqp(h, n, t0, t_end, maxiter=3000, ftol=1e-16):
    def obj(x):
        return _subgrad(x, n)[0]

    bnds = [(0.0, 1.0)] * n
