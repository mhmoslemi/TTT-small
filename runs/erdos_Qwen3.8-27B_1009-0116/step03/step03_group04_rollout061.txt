import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-10:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _grad_lag(h, m, n):
    """Gradient of corr[m] w.r.t. h, where m is index in full correlation output."""
    g = np.zeros(n)
    k = m - (n - 1)  # actual lag
    if k >= 0:
        # corr[m] = sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])
        for i in range(n - k):
            g[i] -= h[i + k]  # d/dh[i] of h[i+k]*(1-h[i]) = -h[i+k]
            g[i+k] += (1.0 - h[i])  # d/dh[i+k] = (1-h[i])
    else:
        # corr[m] = sum_{i=0}^{n-1+k} h[i]*(1-h[i-k])
        kk = -k
        for i in range(n - kk):
            g[i] += (1.0 - h[i-kk])  # d/dh[i] of h[i]*(1-h[i-kk]) = (1-h[i-kk])
            g[i-kk] -= h[i]  # d/dh[i-kk] of h[i]*(1-h[i-kk]) = -h[i]
    return g

def _top_grad(h, n, K):
    """Gradient of sum of top-K correlations."""
    c = _corr(h)
    top_idx = np.argsort(c)[-K:]
    g = np.zeros(n)
    for m in top_idx:
        g += _grad_lag(h, m, n)
    return g / K, c, top_idx
