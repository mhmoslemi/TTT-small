def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a; diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a; diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / len(h))

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_weighted(h, w):
    m = len(h)
    g = np.zeros(m)
    for lc in range(2*m - 1):
        wk = w[lc]
        if wk < 1e-15:
            continue
        k = lc - (m - 1)
        if k == 0:
            g += wk * (1.0 - 2.0 * h)
        elif k > 0:
            g[:m-k] += wk * (1.0 - h[k:])
            g[k:] += wk * (-h[:m-k])
        else:
            kk = -k
            g[kk:] += wk * (1.0 - h[:m-kk])
            g[:m-kk] += wk * (-h[kk:])
    return g

def _project_feasible(h, m):
    target = m / 2.0
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo+hi), 0.0, 1.0)

    c = _corr(h)
    mx = c.max()
    s = alpha * (c - mx)
    s = np.clip(s, -500, 500)
    return mx + np.log(np.sum(np.exp(s - s.max()))) / alpha + s.max()/alpha

# log-sum-exp: log(sum(exp(alpha*(c_i - mx)))) / alpha + mx
# = log(sum(exp(alpha*c_i - alpha*mx))) / alpha + mx
# = (1/alpha) * log(sum(exp(alpha*(c_i - mx)))) + mx
# When alpha→∞, this → mx + log(K)/alpha → mx

def _soft_max_obj2(h, alpha):
    c = _corr(h)
    mx = c.max()
    s = np.clip(alpha * (c - mx), -500, 500)
    return mx + np.log(np.sum(np.exp(s))) / alpha

def _soft_max_grad(h, alpha):
    c = _corr(h)
    mx = c.max()
    s = np.clip(alpha * (c - mx), -500, 500)
    e = np.exp(s - s.max())
    w = e / e.sum()
    # Gradient of the softmax objective = (1/alpha) * alpha * weighted_avg_grad = weighted_avg_grad
    # Actually: d/dh [mx + log(sum exp(alpha*(c_i-mx)))/alpha]
    # If mx is fixed: = (1/alpha) * sum_i alpha * w_i * dc_i/dh = sum_i w_i * dc_i/dh
    return _grad_weighted(h, w)

def _subgrad_opt(h, m, budget_fn):
    """Projected subgradient with adaptive step size."""
    h_cur = h.copy()
    best_local = _eval(h_cur)
    best_local_h = h_cur.copy()
    
    for alpha in [50, 200, 1000, 5000, 20000]:
        for it in range(3000):
            if budget_fn():

            c = _corr(h_cur)
            mx = c.max()
            s = np.clip(alpha * (c - mx), -500, 500)
            e = np.exp(s - s.max())
            w = e / e.sum()
            
            g = _grad_weighted(h_cur, w)
            # Project gradient onto hyperplane
            g = g - g.mean()
            
            lr = 1.0 / (np.sqrt(it + 1) * np.sqrt(m) * 0.5)
            h_new = _project_feasible(h_cur - lr * g, m)
            
            cv = _eval(h_new)
            if cv < best_local:
                best_local = cv
                best_local_h = h_new.copy()
            h_cur = h_new
    
    return best_local_h, best_local

def _slsqp_opt(h, m, alpha, maxiter):
    bnds = [(0.0, 1.0)] * m
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - m/2.0,
             'jac': lambda x: np.ones(m)}]
    try:
        res = minimize(_soft_max_obj2, h, jac=_soft_max_grad,
                       args=(alpha,), method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return _repair(res.x, m)
    except:
        return _repair(h, m)

def _coord_descent(h, m, budget_fn):
    """Coordinate descent: move mass between pairs to reduce max overlap."""
    h = h.copy()
    cv = _eval(h)
    c = _corr(h)
    
    for _pass in range(200):
        if budget_fn():
            break
        imp = False
        # Focus on the most "strained" lags
        for i in range(m):
            for j in range(i+1, m):
                if budget_fn():
                    break
                # Try moving small amount from j to i
                for d in [0.01, -0.01, 0.005, -0.005]:
                    ni, nj = h[i]+d, h[j]-d
                    if ni < 0 or ni > 1 or nj < 0 or nj > 1:

                    h[i], h[j] = ni, nj
                    nc = _eval(h)
                    if nc < cv - 1e-14:
                        cv = nc; imp = True
                    else:
                        h[i], h[j] = ni-d, nj+d
        if not imp:
            break
    return h, cv

    h_new = np.interp(x_new, x_old, h_old, period=2.0)
    return _repair(h_new, n_new)

import numpy as np
from scipy.optimize import minimize
import time

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
            for idx in order:
                if diff > 0:
                    a = min(diff, 1.0 - h[idx])
                    h[idx] += a
                    diff -= a
                else:
                    a = min(-diff, h[idx])
                    h[idx] -= a
                    diff += a
                if abs(diff) < 1e-12:
                    break
    return h

def _eval(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / len(h))

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_weighted(h, w):
    m = len(h)
    g = np.zeros(m)
    for lc in range(2 * m - 1):
        wk = w[lc]
        if wk < 1e-15:
            continue
        k = lc - (m - 1)
        if k == 0:
            g += wk * (1.0 - 2.0 * h)
        elif k > 0:
            g[:m - k] += wk * (1.0 - h[k:])
            g[k:] -= wk * h[:m - k]
        else:
            kk = -k
            g[kk:] += wk * (1.0 - h[:m - kk])
            g[:m - kk] -= wk * h[kk:]
    return g

def _project_feasible(h, m):
    target = m / 2.0
    lo, hi = -5.0, 5.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _sf_obj(h, alpha):
    c = _corr(h)
    mx = c.max()
    s = np.clip(alpha * (c - mx), -500.0, 500.0)
    return mx + np.log(np.sum(np.exp(s))) / alpha

def _sf_grad(h, alpha):
    c = _corr(h)
    mx = c.max()
    s = np.clip(alpha * (c - mx), -500.0, 500.0)
    e = np.exp(s - s.max())
    w = e / e.sum()
    return _grad_weighted(h, w)

def _subgrad(h, m, deadline):
    h_cur = h.copy()
    best_l = _eval(h_cur)
    best_h = h_cur.copy()
    for alpha in [30, 100, 500, 2000, 8000, 30000]:
        for it in range(4000):
            if time.time() > deadline:
                return best_h, best_l
            c = _corr(h_cur)
            mx = c.max()
            s = np.clip(alpha * (c - mx), -500.0, 500.0)
            e = np.exp(s - s.max())
            w = e / e.sum()
            g = _grad_weighted(h_cur, w)
            g -= g.mean()
            lr = 1.0 / (np.sqrt(float(it + 1)) * np.sqrt(float(m)) * 0.4)
            h_new = _project_feasible(h_cur - lr * g, m)
            cv = _eval(h_new)
            if cv < best_l:
                best_l = cv
                best_h = h_new.copy()
            h_cur = h_new
    return best_h, best_l

def _slsqp(h, m, alpha, maxit, deadline):
    bnds = [(0.0, 1.0)] * m
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac, 'args': (m,)}]
    try:
        res = minimize(_sf_obj, h, jac=_sf_grad, args=(alpha,),
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxit, 'ftol': 1e-18})
        best = _repair(res.x, m)
    except Exception:
        best = _repair(h, m)
    return best

def _cons_fun(x, m):
    return x.sum() - m / 2.0

def _cons_jac(x, m):
    return np.ones(m)

def _coord_descent(h, m, deadline):
    h = h.copy()
    cv = _eval(h)
    steps = [0.02, -0.02, 0.01, -0.01, 0.005, -0.005, 0.002, -0.002]
    for _pass in range(500):
        if time.time() > deadline:
            break
        imp = False
        for i in range(m):
            hi = h[i]
            for j in range(i + 1, m):
                if time.time() > deadline:
                    break
                hj = h[j]
                for d in steps:
                    ni, nj = hi + d, hj - d
                    if ni < -1e-12 or ni > 1 + 1e-12 or nj < -1e-12 or nj > 1 + 1e-12:
                        continue
                    h[i], h[j] = ni, nj
                    nc = _eval(h)
                    if nc < cv - 1e-15:
                        cv = nc
                        hi, hj = ni, nj
                        imp = True
                    else:
                        h[i], h[j] = hi, hj
                h[j] = hj
            h[i] = hi
        if not imp:
            break
    return h, cv

def _interpolate(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old + 1)
    h_ext = np.concatenate([h_old, [h_old[0]]])
    x_new = np.linspace(0.0, 2.0, n_new + 1)
    h_mid = np.interp(x_new, x_old, h_ext)[:-1]
    return _repair(h_mid, n_new)

def _random_fine(h, m, rng, deadline):
    h = h.copy()
    cv = _eval(h)
    scale = 0.004
    no_imp = 0
    for _ in range(200000):
        if time.time() > deadline:
            break
        i = rng.randint(m)
        j = rng.randint(m)
        if i == j:
            continue
        d = scale * rng.randn()
        ni, nj = h[i] + d, h[j] - d
        if ni < 0.0 or ni > 1.0 or nj < 0.0 or nj > 1.0:
            continue
        h[i], h[j] = ni, nj
        nc = _eval(h)
        if nc < cv - 1e-15:
            cv = nc
            no_imp = 0
            scale = min(scale * 1.1, 0.015)
        else:
            h[i], h[j] = ni - d, nj + d
            no_imp += 1
            if no_imp > 800:
                scale *= 0.6
                no_imp = 0
                if scale < 1e-7:
                    break
    return h, cv
