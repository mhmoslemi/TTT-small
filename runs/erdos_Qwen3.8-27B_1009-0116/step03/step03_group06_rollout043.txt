        i = int(order_dec[rng.randint(8)])
        j = int(order_inc[rng.randint(8)])
        if i == j:
            i = int(rng.randint(n))
            j = int(rng.randint(n))
            if i == j:
                continue
        d = lr * rng.uniform(0.3, 1.0) * (1.0 if g[i] > g[j] else -1.0)
        # We want to decrease h where g is high (decrease helps)
        # and increase h where g is low
        # So: h[i] -= d, h[j] += d where g[i] > g[j]
        hi, hj = h[i] - d, h[j] + d
        if hi < 0 or hi > 1 or hj < 0 or hj > 1:
            d *= 0.3
            hi, hj = h[i] - d, h[j] + d
            if hi < 0 or hi > 1 or hj < 0 or hj > 1:
                continue
        h[i], h[j] = hi, hj
        c = _c5(h, dx)
        if c < best - 1e-14:
            best = c
            lr = min(0.05, lr * 1.02)
        else:
            h[i], h[j] = hi + d, hj - d
            lr = max(0.0005, lr * 0.97)
    return h, best

def _pair_local(h, n, dx, rng, deadline, ntry=200000):
    best = _c5(h, dx)
    h = h.copy()
    scales = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    ps = max(500, ntry // len(scales))
    for si, sc in enumerate(scales):
        imp = False
        for _ in range(ps):
            if time.time() >= deadline:
                return h, best
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = sc * (2.0 * rng.random() - 1.0)
            hi, hj = h[i] + d, h[j] - d
            if hi < -1e-10 or hi > 1.0 + 1e-10 or hj < -1e-10 or hj > 1.0 + 1e-10:
                continue
            h[i], h[j] = hi, hj
            c = _c5(h, dx)
            if c < best - 1e-14:
                best = c
                imp = True
            else:
                h[i], h[j] = hi - d, hj + d
        if not imp and si >= 3:
            break
    return h, best

def _slsqp_phase(h, n, dx, alphas, deadline):
    tgt = n / 2.0
    best_h, best_c5 = h.copy(), _c5(h, dx)
    for a in alphas:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                lambda x, aa=a: _smooth_max(_corr(x), aa) + 1000.0 * (np.sum(x) - tgt) ** 2,
                best_h, method='SLSQP', bounds=[(0.0, 1.0)] * n,
                options={'maxiter': 200, 'ftol': 1e-15}
            )
            hc = _repair(res.x, n)
            c = _c5(hc, dx)
            if c < best_c5:
                best_h, best_c5 = hc, c
        except Exception:
            pass
    return best_h, best_c5

def _perturb(h, n, rng, kind):
    hp = h.copy()
    if kind == 0:
        blk = rng.randint(5, 15)
        i = rng.randint(0, n - blk)
        hp[i:i+blk] = np.roll(hp[i:i+blk], rng.randint(1, blk))
    elif kind == 1:
        f = rng.randint(1, 6)
        ph = rng.uniform(0, 6.28)
        hp += rng.uniform(0.05, 0.15) * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
    elif kind == 2:
        j = rng.randint(1, n)
        hp = np.roll(hp, j)
    elif kind == 3:
        # Binary round and re-perturb
        hb = (hp > 0.5).astype(float)
        no = int(hb.sum())
        t = n // 2
        if no > t:
            oi = np.where(hb == 1)[0]
            hb[oi[rng.permutation(no)][:no - t]] = 0
        elif no < t:
            zi = np.where(hb == 0)[0]
            hb[zi[rng.permutation(n - no)][:t - no]] = 1
        return hb
    else:
        hp += rng.uniform(-0.08, 0.08, n)
    return _repair(hp, n)

def _multi_start(h, n, dx, rng, deadline, n_rest=10):
    bh, bc5 = h.copy(), _c5(h, dx)
    for r in range(n_rest):
        if time.time() >= deadline:
            break
        sp = min(60, deadline - time.time() - 10)
        if sp < 5:
            break
        hp = _perturb(bh, n, rng, r % 5)
        te1 = time.time() + sp * 0.4
        hp, c1 = _grad_local(hp, n, dx, rng, te1, max_it=15000)
        te2 = time.time() + sp * 0.8
        hp, c2 = _pair_local(hp, n, dx, rng, te2, ntry=20000)
        if c2 < bc5:
            bh, bc5 = hp, c2
    return bh, bc5
