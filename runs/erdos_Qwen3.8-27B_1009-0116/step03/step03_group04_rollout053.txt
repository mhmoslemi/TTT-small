import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smv(c, alpha):
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _sgrad(h, alpha, n):
    c = _corr(h)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        lag = k - (n - 1)
        lo = max(0, lag)
        hi = min(n - 1, n - 1 + lag)
        if lo <= hi:
            g[lo:hi + 1] += wk * (1.0 - h[lo - lag:hi - lag + 1])
        lo2 = max(0, -lag)
        hi2 = min(n - 1, n - 1 - lag)
        if lo2 <= hi2:
            g[lo2:hi2 + 1] += -wk * h[lo2 + lag:hi2 + lag + 1]
    return g

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    T = n / 2.0
    if h.sum() < 1e-12:
        return np.full(n, 0.5)
    if abs(h.sum() - T) > 1e-10:
        h *= T / h.sum()
        h = np.clip(h, 0.0, 1.0)
    d = T - h.sum()
    for _ in range(8):
        if abs(d) < 1e-14:
            break
        if d > 0:
            room = 1.0 - h
            order = np.argsort(-room)[:abs(d) * 2 + 2]
            for idx in order:
                if d <= 1e-14:
                    break
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
        else:
            order = np.argsort(h)[:abs(d) * 2 + 2]
            for idx in order:
                if -d <= 1e-14:
                    break
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
    return h

def _slsqp(x0, bounds, T, alpha, maxit, n):
    def obj(x):
        return _smv(_corr(x), alpha)
    def ograd(x):
        return _sgrad(x, alpha, n)
    def cfun(x):
        return x.sum() - T
    def cgrad(x):
        return np.ones(n)
    try:
        res = minimize(obj, x0, jac=ograd, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cfun, 'jac': cgrad}],
                       options={'maxiter': maxit, 'ftol': 1e-16})
        return res.x
    except Exception:
        return x0

    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    T = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def elapsed():
        return time.time() - t0

    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _c5(best_h, dx)

    # Symmetry: 1-h gives same C5
    h1 = _repair(1.0 - h0, n)
    c5_1 = _c5(h1, dx)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()

    # Phase 1: SLSQP with analytical gradient, increasing alpha
    for alpha in [300, 1000, 3000, 10000, 30000]:
        if elapsed() > budget_s - 350:
            break
        for att in range(2):

            sc = 0.0 if att == 0 else 0.004
            x0 = _repair(best_h + sc * rng.randn(n), n)
            mx = 1200 if alpha <= 3000 else 2000
            rx = _slsqp(x0, bounds, T, alpha, mx, n)
            hc = _repair(rx, n)
            c5 = _c5(hc, dx)
            if c5 < best_c5:

    # Phase 2: Nesterov GD with analytical smooth-max gradient
    if elapsed() < budget_s - 300:
        h = best_h.copy()
        alpha = 8000
        v = np.zeros(n)
        lr = 0.3
        mom = 0.85
        c5_cur = _c5(h, dx)
        stale = 0
        for it in range(8000):
            if elapsed() > budget_s - 300:
                break
            g = _sgrad(h, alpha, n)
            g -= g.mean()
            v = mom * v - lr * g
            h_new = _repair(h + v, n)
            c5_new = _c5(h_new, dx)
            if c5_new < c5_cur - 1e-16:
                h = h_new
                c5_cur = c5_new
                stale = 0
                lr = min(lr * 1.02, 3.0)
                if c5_cur < best_c5:
                    best_c5, best_h = c5_cur, h.copy()
            else:

                lr *= 0.7
                stale += 1

    if elapsed() < budget_s - 150:

        hc = _repair(rx, n)

        if c5 < best_c5:

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
