        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Phase 7: Ultra-fine nudge
    if _el() < budget_s - 5:
        h = best_h.copy()
        c5c = _eval(h)
        for _it in range(50000):
            if _el() > budget_s - 5:
                break
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            d = rng.choice([1, -1]) * 1e-5 * rng.choice([1.0, 2.0, 5.0, 10.0])
            ni, nj = h[i] + d, h[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h[i], h[j] = ni, nj
                c5n = _eval(h)
                if c5n < c5c - 1e-17:
                    c5c = c5n
                    if c5n < best_c5:
                        best_c5, best_h = c5n, h.copy()
                else:
                    h[i], h[j] = h[i] - d, h[j] + d

    best_h = _project(best_h, target)
    best_c5 = _eval(best_h)
    return (best_h, best_c5, n)
