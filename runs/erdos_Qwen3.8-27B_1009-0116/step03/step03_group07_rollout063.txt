import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-14:
        return np.full(n, 0.5)
    for _ in range(5000):
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target) < 1e-15:
            break
        d = target - s
        if abs(d) < 1e-15:
            break
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-15:
                break
            if d > 0:
                a = min(d, 1.0 - h[idx])
                h[idx] += a
                d -= a
            else:
                a = min(-d, h[idx])
                h[idx] -= a
                d += a
    return h

def grad_lse(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    p = np.exp(s - np.max(s))
    p /= np.sum(p)
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        w = p[idx]
        if w < 1e-14:
            continue
        lag = idx - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif lag > 0:
            i1 = np.arange(0, n - lag)
            grad[i1] += w * (1.0 - h[i1 + lag])
            i2 = np.arange(lag, n)
            grad[i2] -= w * h[i2 - lag]
        else:
            m2 = -lag
            i1 = np.arange(m2, n)
            grad[i1] += w * (1.0 - h[i1 - m2])
            i2 = np.arange(0, n - m2)
            grad[i2] -= w * h[i2 + m2]
    return grad

def lse_obj(h, alpha):
    corr = compute_corr(h)
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def lbfgs_opt(h, n, alpha, pen, maxit):
    target = n / 2.0
    def f(v):
        return lse_obj(v, alpha) + pen * (v.sum() - target) ** 2
    def g(v):
        return grad_lse(v, alpha) + 2 * pen * (v.sum() - target)
    b = [(0.0, 1.0)] * n
    try:
        r = minimize(f, h, jac=g, method='L-BFGS-B', bounds=b,
                     options={'maxiter': maxit, 'ftol': 1e-18, 'gtol': 1e-12})
        return repair(r.x, n)
    except Exception:
        return h

_s2 = {}

def obj_slsqp(h):
    corr = compute_corr(h)
    m = np.max(corr)
    a = _s2['alpha']
    t = _s2['k']
    s = a * (corr - m)
    s = np.clip(s, -500, 500)
    val = m + np.log(np.sum(np.exp(s))) / a
    if t > 1:
        sc = np.sort(corr)[::-1][:t]
        val += _s2['fw'] * np.sum((sc - sc[0] + (sc[0] - sc[-1])) ** 2) / t
    return val

def cons_eq(h):
    return h.sum() - _s2['target']

def slsqp_opt(h, n, alpha, topk, fw, maxit):
    _s2['alpha'] = alpha
    _s2['k'] = topk
    _s2['fw'] = fw
    _s2['target'] = n / 2.0
    b = [(0.0, 1.0)] * n
    c = [{'type': 'eq', 'fun': cons_eq}]
    try:
        r = minimize(obj_slsqp, h, method='SLSQP', bounds=b, constraints=c,
                     options={'maxiter': maxit, 'ftol': 1e-16})
        return repair(r.x, n)
    except Exception:
        return h

def block_cd(h, n, n_rounds, deadline, rng, bsz=3):
    h = h.copy()
    best = compute_c5(h, n)
    for r in range(n_rounds):
        if time.time() > deadline:
            break
        idxs = rng.choice(n, size=bsz, replace=False)
        h_block = h[idxs].copy()

        def sub_obj(x):
            saved = h[idxs].copy()
            h[idxs] = np.clip(x, 0, 1)
            val = compute_c5(h, n)
            h[idxs] = saved
            return val

        center = h_block.copy()
        radius = 0.15
        try:
            r2 = minimize(sub_obj, center, method='Nelder-Mead',
                          options={'maxiter': 200, 'xatol': 1e-8, 'fatol': 1e-10})
            h_trial = h.copy()
            h_trial[idxs] = np.clip(r2.x, 0, 1)
            h_trial = repair(h_trial, n)
            c5_trial = compute_c5(h_trial, n)
            if c5_trial < best - 1e-14:
                h = h_trial
                best = c5_trial
        except Exception:
            pass
    return h, best

def subgrad_polish(h, n, n_iter, deadline):
    h = h.copy()
    best = compute_c5(h, n)
    dx = 2.0 / n
    for it in range(n_iter):
        if time.time() > deadline:
            break
        corr = compute_corr(h)
        max_idx = np.argmax(corr)
        lag = max_idx - (n - 1)
        g = np.zeros(n)
        if lag == 0:
            g[:] = 1.0 - 2.0 * h
        elif lag > 0:
            i1 = np.arange(0, n - lag)
            g[i1] += (1.0 - h[i1 + lag])
            i2 = np.arange(lag, n)
            g[i2] -= h[i2 - lag]
        else:
            m2 = -lag
            i1 = np.arange(m2, n)
            g[i1] += (1.0 - h[i1 - m2])
            i2 = np.arange(0, n - m2)
            g[i2] -= h[i2 + m2]
        g *= dx
        lr = 0.05 / (1.0 + 0.05 * it)
        for _ in range(25):
            h_try = np.clip(h - lr * g, 0, 1)
            h_try = repair(h_try, n)
            c5_try = compute_c5(h_try, n)
            if c5_try < best - 1e-15:
                h = h_try
                best = c5_try
                break
            lr *= 0.6
    return h, best

def make_start(n, sv):
    rng = np.random.RandomState(sv)
    h = np.full(n, 0.5)
    for k in range(1, 6):
        amp = 0.12 / k
        ph = rng.uniform(0, 6.28)
        h += amp * np.sin(2 * np.pi * k * np.arange(n) / n + ph)
    return repair(h, n)
