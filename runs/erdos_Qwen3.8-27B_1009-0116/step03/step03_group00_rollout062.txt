            d = rng.choice([0.002, 0.005, 0.012, 0.03, 0.08, 0.2]) * (1 if rng.random() < 0.5 else -1)
        ni, nj = h[i] + d, h[j] - d
        hi, hj = h[i], h[j]
        if ni < -1e-12 or ni > 1 + 1e-12 or nj < -1e-12 or nj > 1 + 1e-12:
            continue
        h[i] = max(0.0, min(1.0, ni))
        h[j] = max(0.0, min(1.0, nj))
        c = _c5(h, n)
        if c < cur_c:
            cur_c = c
            if c < best_c:
                best_c = c
        elif T > 1e-13 and rng.random() < np.exp(-min((c - cur_c) / T, 400)):
            cur_c = c
        else:
            h[i], h[j] = hi, hj
        T = max(T * 0.99975, 1e-14)
    h = _repair(h, n)
    return h, _c5(h, n)

def _fine_polish(h, n, t_end, rng):
    cur_c = _c5(h, n)
    step = 0.0005
    while time.time() < t_end:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = step * (1 if rng.random() < 0.5 else -1)
        ni, nj = h[i] + d, h[j] - d
        if 0 <= ni <= 1 and 0 <= nj <= 1:
            hi, hj = h[i], h[j]
            h[i], h[j] = ni, nj
            c = _c5(h, n)
            if c < cur_c:
                cur_c = c
            else:
                h[i], h[j] = hi, hj
        if rng.random() < 0.01:
            step *= rng.choice([0.7, 1.4])
            step = max(1e-5, min(0.01, step))
    return h, cur_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    n = len(initial_h_values)
    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c = _c5(best_h, n)

    # Phase 1: SLSQP with soft-max (40% time)
    t_end1 = t0 + budget_s * 0.40
    alphas = [500, 2000, 8000, 30000, 100000]
    for alpha in alphas:
        n_restart = 4
        for r in range(n_restart):
            if time.time() > t_end1 - 5:
                break
            if r == 0:
                h_init = best_h.copy()
            else:
                scale = 0.015 + 0.01 * r
                h_init = _repair(best_h + scale * rng.randn(n), n)
            obj_fn = _make_soft_obj(n, alpha)
            grad_fn = _make_soft_grad(n, alpha)
            h_new = _slsqp_run(h_init, n, obj_fn, grad_fn, 800, t_end1 - 5)
            c = _c5(h_new, n)
            if c < best_c:
                best_c = c
                best_h = h_new.copy()

    # Phase 1b: Top-K focused SLSQP
    t_end1b = t0 + budget_s * 0.50
    for K in [2, 4, 6]:
        if time.time() > t_end1b - 5:
            break
        def obj_tk(x):
            return _topk_obj(x, n, K)
        def grad_tk(x):
            return _topk_grad(x, n, K)
        for r in range(3):
            if time.time() > t_end1b - 5:
                break
            h_init = best_h if r == 0 else _repair(best_h + 0.015 * rng.randn(n), n)
            h_new = _slsqp_run(h_init, n, obj_tk, grad_tk, 600, t_end1b - 5)
            c = _c5(h_new, n)
            if c < best_c:
                best_c = c
                best_h = h_new.copy()

    # Phase 2: Aggressive pair-grid local search (25% time)
    t_end2 = t0 + budget_s * 0.72
    best_h, best_c = _pair_grid(best_h, n, t_end2, nsamples=50)

    # Phase 3: Simulated annealing (15% time)
    t_end3 = t0 + budget_s * 0.87
    h_sa, c_sa = _sa_run(best_h, n, t_end3, rng)
    if c_sa < best_c:
        best_c = c_sa
        best_h = h_sa.copy()

    # Phase 4: Final pair-grid polish (7% time)
    t_end4 = t0 + budget_s * 0.96
    best_h, best_c = _pair_grid(best_h, n, t_end4, nsamples=80)

    # Phase 5: Ultra-fine random polish
    t_end5 = t0 + budget_s - 3
    best_h, best_c = _fine_polish(best_h, n, t_end5, rng)

    best_h = _repair(best_h, n)
    best_c = _c5(best_h, n)
    return (best_h, best_c, n)
