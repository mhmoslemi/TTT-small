def fft_corr(h, g):

    N = 2 * n

    G = np.fft.rfft(g[::-1], N)
    return np.fft.irfft(H * G, N)[:2*n-1]

    corr = fft_corr(h, 1.0 - h)

def project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        h += d / n
        h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-14:
        order = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for idx in order:
            if abs(d) < 1e-14:
                break

                a = min(d, 1.0 - h[idx]); h[idx] += a; d -= a

def subgrad_top(h, n, corr, top_m):

    grad = np.zeros(n)

            L = -lag

    steps = [0.03, 0.015, 0.007, 0.003, 0.001]

    target = n / 2.0

        vv = np.clip(v, 0, 1)
        corr = fft_corr(vv, 1.0 - vv)
        mx = np.max(corr)
        s = alpha * (corr - mx)
        s = np.clip(s, -400, 400)
        return mx + np.log(np.sum(np.exp(s))) / alpha
    def obj_grad(v):
        vv = np.clip(v, 0, 1)
        corr = fft_corr(vv, 1.0 - vv)
        mx = np.max(corr)
        top = np.where(corr > mx - 0.5 / alpha)[0]
        sg = subgrad_top(vv, n, corr, min(len(top), 10))
        return -sg
