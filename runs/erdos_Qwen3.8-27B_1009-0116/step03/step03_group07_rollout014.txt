        val += flat_w * np.sum((sc - sc.mean())**2)
    return val

        return soft_max_obj(hv, n, alpha, topk, flat_w)
    def eq_con(hv):
        return hv.sum() - target
    cons = [{'type': 'eq', 'fun': eq_con}]
    try:

                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})

            hi, hj = h[i], h[j]
            for d in steps:

                    ni = hi + delta
                    nj = hj - delta
                    if ni < 0 or ni > 1 or nj < 0 or nj > 1:

                    h[i], h[j] = ni, nj
                    c5 = compute_c5(h, n)

                        break
                    h[i], h[j] = hi, hj
                if improved:

            if i == j:
                continue
            hi, hj = h[i], h[j]
            if nested:
                found = False
                for d in base:
                    dd = d * scale
                    for sg in (1, -1):
                        delta = sg * dd
                        ni, nj = hi + delta, hj - delta
                        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                            continue
                        h[i], h[j] = ni, nj
                        c5 = compute_c5(h, n)
                        if c5 < best - 1e-16:
                            best = c5
                            improved = True
                            found = True
                            break
                        h[i], h[j] = hi, hj
                    if found:
                        break
            else:
                d = base[rng.randint(len(base))] * scale
                sg = 1.0 if rng.rand() < 0.5 else -1.0
                delta = sg * d
                ni, nj = hi + delta, hj - delta
                if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                    continue
                h[i], h[j] = ni, nj
                c5 = compute_c5(h, n)
                if c5 < best - 1e-16:
                    best = c5
                    improved = True
                else:
                    h[i], h[j] = hi, hj
        if not improved and r > 40:
            break
    return h, best

def fourier_start(n, M, sd):
    rng = np.random.RandomState(sd)
    t = np.arange(n)
    h = np.full(n, 0.5)
    for k in range(1, M + 1):
        h += rng.uniform(-0.4, 0.4) * np.cos(2.0 * np.pi * k * t / n)
        h += rng.uniform(-0.4, 0.4) * np.sin(2.0 * np.pi * k * t / n)
    return repair(h, n)

def interp_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return repair(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed + 777)
    best_h, best_c5, best_n = None, np.inf, 70

    try:
        ih = np.array(initial_h_values, dtype=float).copy()
        best_h = repair(ih, len(ih))
        best_n = len(best_h)
        best_c5 = compute_c5(best_h, best_n)
    except NameError:
        best_h = fourier_start(70, 12, seed)
        best_c5 = compute_c5(best_h, best_n)

    # Phase 1: SLSQP at original n
    nw = best_h.copy()
    for alpha in [5000, 20000, 100000, 300000]:
        if time.time() - t0 > budget_s * 0.15:
            break
        for tk in [1, 3, 5]:
            if time.time() - t0 > budget_s * 0.15:
                break
            hw = slsqp_opt(nw.copy(), best_n, alpha, tk, 0.01, 800)
            c5 = compute_c5(hw, best_n)
            if c5 < best_c5 - 1e-14:
                best_h, best_c5, nw = hw.copy(), c5, hw

    # Phase 2: L-BFGS-B at original n
    if time.time() - t0 < budget_s * 0.30:
        for alpha in [50000, 200000, 500000]:
            if time.time() - t0 > budget_s * 0.30:
                break
            for pen in [1000, 10000, 100000]:
                if time.time() - t0 > budget_s * 0.30:
                    break
                hw = lbfgs_opt(nw.copy(), best_n, alpha, 3, 0.01, pen, 5000)
                c5 = compute_c5(hw, best_n)
                if c5 < best_c5 - 1e-14:
                    best_h, best_c5, nw = hw.copy(), c5, hw

    # Phase 3: Coordinate descent at original n (nested, thorough)
    if time.time() - t0 < budget_s * 0.50:
        hc, c5c = coord_cd(nw, best_n, t0 + budget_s * 0.50, rng, 400, 8 * best_n, True)
        if c5c < best_c5 - 1e-14:
            best_h, best_c5 = hc.copy(), c5c
            nw = hc

    # Phase 4: Interpolate to larger n and optimize
    if time.time() - t0 < budget_s * 0.85:
        for n_new in [140, 200, 300, 500]:
            if time.time() - t0 > budget_s * 0.82:
                break
            h_interp = interp_h(best_h, n_new)
            # L-BFGS-B optimization
            h_interp = lbfgs_opt(h_interp, n_new, 30000, 3, 0.01, 5000, 3000)
            # Quick SLSQP
            h_interp = slsqp_opt(h_interp, n_new, 50000, 3, 0.01, 200)
            # Coordinate descent (non-nested for speed)
            per = max(200, 10000 // n_new)
            h_interp, c5_cd = coord_cd(h_interp, n_new, t0 + budget_s * 0.82, rng,
                                        200, per, False)
            c5_new = compute_c5(h_interp, n_new)
            if c5_new < best_c5 - 1e-6:
                best_h, best_c5, best_n = h_interp.copy(), c5_new, n_new

    # Phase 5: Try fresh Fourier starts at medium n
    if time.time() - t0 < budget_s * 0.92:
        for n_f in [100, 160, 250]:
            if time.time() - t0 > budget_s * 0.90:
                break
            if n_f == best_n:
                continue
            hf = fourier_start(n_f, min(15, n_f // 5), seed + n_f * 31)
            hf = lbfgs_opt(hf, n_f, 50000, 3, 0.01, 10000, 4000)
            hf = slsqp_opt(hf, n_f, 100000, 3, 0.01, 300)
            c5_f = compute_c5(hf, n_f)
            if c5_f < best_c5 - 1e-6:
                best_h, best_c5, best_n = hf.copy(), c5_f, n_f

    # Phase 6: Final polish on best solution
    if best_h is not None and time.time() - t0 < budget_s * 0.99:
        hw = slsqp_opt(best_h.copy(), best_n, 500000, 5, 0.02, 1000)
        c5 = compute_c5(hw, best_n)
        if c5 < best_c5 - 1e-14:
            best_h, best_c5 = hw.copy(), c5
        per = max(200, 8 * best_n) if best_n <= 100 else max(200, 5000 // best_n)
        nested = best_n <= 120
        hc, c5c = coord_cd(best_h, best_n, t0 + budget_s * 0.995, rng, 300, per, nested)
        if c5c < best_c5 - 1e-14:
            best_h, best_c5 = hc.copy(), c5c

    best_h = repair(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return (best_h, best_c5, best_n)
