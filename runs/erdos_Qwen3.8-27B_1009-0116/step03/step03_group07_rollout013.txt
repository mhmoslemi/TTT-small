import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    target = n / 2.0
    for _ in range(10000):
        s = h.sum()
        if abs(s - target) < 1e-14:
            return h
        h = np.clip(h * target / max(s, 1e-30), 0.0, 1.0)
        s = h.sum()
        d = target - s
        if abs(d) < 1e-14:
            return h
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if abs(d) < 1e-14:
                break
            a = min(abs(d), 1.0 - h[i]) if d > 0 else min(abs(d), h[i])
            if a > 0:
                h[i] += a if d > 0 else -a
    return h

_st = {}

def sm_func(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    a = _st['a']
    s = np.clip(a * (corr - m), -400, 400)
    return m + np.log(np.sum(np.exp(s))) / a + _st['pen'] * (h.sum() - _st['t'])**2

def eq_fn(h):
    return h.sum() - _st['t']

def slsqp(h, n, alpha, pen, maxiter):
    _st['a'] = alpha
    _st['t'] = n / 2.0
    _st['pen'] = pen
    b = [(0.0, 1.0)] * n
    c = [{'type': 'eq', 'fun': eq_fn}] if pen < 1 else []
    try:
        r = minimize(sm_func, h.copy(), method='SLSQP', bounds=b,
                     constraints=c, options={'maxiter': maxiter, 'ftol': 1e-16})
        return repair(r.x, n)
    except Exception:
        return h

def cd(h, n, rounds, t_end, rng, anneal=False):
    h = h.copy()
    best = compute_c5(h)
    steps = [0.05, 0.03, 0.02, 0.01, 0.006, 0.003, 0.0015, 0.0008, 0.0004, 0.0002]
    stall = 0
    temp = 0.001
    for r in range(rounds):
        if time.time() > t_end:
            break
        imp = False
        for _ in range(10 * n):
            if time.time() > t_end:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            hi, hj = h[i], h[j]
            for d in steps:
                for sgn in (1, -1):
                    di, dj = hi + sgn * d, hj - sgn * d
                    if di < -1e-10 or di > 1 + 1e-10 or dj < -1e-10 or dj > 1 + 1e-10:
                        continue
                    h[i], h[j] = di, dj
                    c5 = compute_c5(h)
                    if c5 < best - 1e-15:
                        best = c5
                        imp = True
                        break
                    elif anneal and c5 < best + temp and rng.rand() < 0.1:
                        best = c5
                        imp = True
                        break
                    h[i], h[j] = hi, hj
                if imp:
                    break
        if imp:
            stall = 0
        else:
            stall += 1
            if anneal:
                temp *= 0.9
            if stall > 300:
                break
    return h, best

def make_init(n, seed, kind):
    rng = np.random.RandomState(seed)
    if kind == 0:
        h = 0.5 + 0.35 * rng.randn(n)
    elif kind == 1:
        h = np.where(np.arange(n) < n // 2, 1.0, 0.0) + 0.08 * rng.randn(n)
    elif kind == 2:
        f = rng.randint(2, 6)
        p = rng.uniform(0, 6.28)
        h = 0.5 + 0.4 * np.sin(2 * np.pi * f * np.arange(n) / n + p)
    elif kind == 3:
        b = max(2, n // 6)
        h = np.zeros(n)
        for i in range(0, n, 2 * b):
            h[i:i + b] = 1.0
        h += 0.15 * rng.randn(n)
    else:
        h = rng.uniform(0, 1, n)
    return repair(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed + 999)
    best_h, best_c5, best_n = None, np.inf, 70

        best_h = make_init(70, seed, 1)

        best_c5 = compute_c5(best_h)

    hw = best_h.copy()

    # Phase 1: Escalating SLSQP (20% budget)
    t1 = t0 + budget_s * 0.20
    for alpha in [20000, 80000, 200000, 500000]:
        if time.time() > t1:

        h_new = slsqp(hw.copy(), best_n, alpha, 0.0, 800)
        c5 = compute_c5(h_new)
        if c5 < best_c5 - 1e-14:
            best_h, best_c5, hw = h_new.copy(), c5, h_new

    # Phase 2: Aggressive coordinate descent (25% budget)
    t2 = t0 + budget_s * 0.45
    hw, c5 = cd(best_h.copy(), best_n, 1000, t2, rng)

    # Phase 3: SLSQP on improved + annealed CD (20% budget)

    # Phase 4: Explore other n values (15% budget)

    return (best_h, best_c5, best_n)
