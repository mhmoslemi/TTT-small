```python
import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _obj_smax(x, alpha, dx):
    c = _corr(x)
    return _smooth_max(c, alpha)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _subgrad_step(h, n, dx, lr):
    """One subgradient step with projection."""
    corr = _corr(h)
    c5_val = np.max(corr)
    # Find active set (lags at or near max)
    active = np.where(corr >= c5_val - 1e-10)[0]
    n_active = len(active)
    # Compute subgradient (average over active lags)
    # grad of corr[k] w.r.t. h[m]: (1-h[m+k]) - h[m-k] (with bounds)
    grad = np.zeros(n)
    for k in active:
        lag = k - (n - 1)  # actual lag
        for m in range(n):
            mk = m + lag
            if 0 <= mk < n:
                grad[m] += (1.0 - h[mk])
            mk2 = m - lag
            if 0 <= mk2 < n:
                grad[m] -= h[mk2]
    grad /= n_active
    # Projected gradient step
    h_new = h - lr * grad
    h_new = _repair(h_new, n)
    return h_new, c5_val
```

```python
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _obj_smax(x, alpha):
    return _smooth_max(_corr(x), alpha)

def _eq_con(x, target):
    return float(np.sum(x)) - target

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _g_vec(h, p, n):
    g = np.zeros(2 * n - 1)
    q = np.arange(n)
    ks1 = (n - 1) - p + q
    m1 = (ks1 >= 0) & (ks1 < 2 * n - 1)
    g[ks1[m1]] += (1.0 - h[q[m1]])
    ks2 = (n - 1) + p - q
    m2 = (ks2 >= 0) & (ks2 < 2 * n - 1)
    g[ks2[m2]] -= h[q[m2]]
    return g

def _pair_sweep(h, n, corr, time_limit, n_coarse=30, n_refine=30):
    improved = False
    order = np.random.permutation(n)
    for ai in range(n - 1):
        i = order[ai]
        for aj in range(ai + 1, n):
            j = order[aj]
            if time.time() > time_limit:
                return h, corr, improved
            hi, hj = h[i], h[j]
            d_min = max(-hi, hj - 1.0)
            d_max = min(1.0 - hi, hj)
            if d_max - d_min < 1e-12:
                continue
            gd = _g_vec(h, i, n) - _g_vec(h, j, n)
            # Quadratic correction positions
            qd = np.zeros(2 * n - 1)
            delta = j - i
            k1 = (n - 1) - delta
            k2 = (n - 1) + delta
            if 0 <= k1 < 2 * n - 1:
                qd[k1] = 1.0
            if 0 <= k2 < 2 * n - 1 and k2 != k1:
                qd[k2] = 1.0
            # Coarse search
            ds = np.linspace(d_min, d_max, n_coarse)
            # Compute max values with quadratic term
            vals = np.empty(n_coarse)
            for di, d in enumerate(ds):
                new_c = corr + d * gd + d * d * qd
                vals[di] = np.max(new_c)
            bi = np.argmin(vals)
            cur_max = np.max(corr)
            if vals[bi] >= cur_max - 1e-14:
                continue
            # Refine around best
            lo = max(d_min, ds[max(0, bi - 1)])
            hi_r = min(d_max, ds[min(n_coarse - 1, bi + 1)])
            for _ in range(n_refine):
                mid = 0.5 * (lo + hi_r)
                # Evaluate at mid and hi_r
                f_mid = np.max(corr + mid * gd + mid * mid * qd)
                f_hi = np.max(corr + hi_r * gd + hi_r * hi_r * qd)
                # Also check lo
                if f_mid <= f_hi:
                    hi_r = mid
                else:
                    lo = mid
            d_best = 0.5 * (lo + hi_r)
            h[i] = hi + d_best
            h[j] = hj - d_best
            corr = corr + d_best * gd + d_best * d_best * qd
            improved = True
    return h, corr, improved
```

```python
def _golden_search(f, lo, hi, n_iter=40):
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(n_iter):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return 0.5 * (lo + hi)
```

But this requires many function evaluations, each O(n). For our purposes, the bisection approach should be fine since we're in a local neighborhood.

```python
vals_fine = np.array([np.max(corr + d * gd + d*d * qd) for d in ds_fine])
bi_fine = np.argmin(vals_fine)
d_best = ds_fine[bi_fine]
```

new_c = corr + d*gd + d²*qd

base = corr + d*gd (vector of length 2n-1)
base[k1] += d²  (if k1 valid)
base[k2] += d²  (if k2 valid)
max_val = max(base)

```python
import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smooth_max(c, alpha):
    m = float(np.max(c))
    s = alpha * (c - m)
    s = np.clip(s, -500, 500)
    return m + float(np.log(np.sum(np.exp(s)))) / alpha

def _obj_smax(x, alpha):
    return _smooth_max(_corr(x), alpha)

def _eq_con(x, target):
    return float(np.sum(x)) - target

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _g_vec(h, p, n):
    g = np.zeros(2 * n - 1)
    q = np.arange(n)
    ks1 = (n - 1) - p + q
    m1 = (ks1 >= 0) & (ks1 < 2 * n - 1)
    g[ks1[m1]] += (1.0 - h[q[m1]])
    ks2 = (n - 1) + p - q
    m2 = (ks2 >= 0) & (ks2 < 2 * n - 1)
    g[ks2[m2]] -= h[q[m2]]
    return g

def _eval_pair(corr, d, gd, qd):
    v = corr + d * gd
    nz = np.nonzero(qd)[0]
    for k in nz:
        v[k] += d * d * qd[k]
    return np.max(v)

def _pair_sweep(h, n, corr, time_limit, n_coarse=31, n_fine=25):
    improved = False
    order = np.random.permutation(n)
    for ai in range(n - 1):
        i = order[ai]
        for aj in range(ai + 1, n):
            j = order[aj]
            if time.time() > time_limit:
                return h, corr, improved
            hi, hj = h[i], h[j]
            d_min = max(-hi, hj - 1.0)
            d_max = min(1.0 - hi, hj)
            if d_max - d_min < 1e-12:
                continue
            gd = _g_vec(h, i, n) - _g_vec(h, j, n)
            qd = np.zeros(2 * n - 1)
            delta = j - i
            k1 = (n - 1) - delta
            k2 = (n - 1) + delta
            if 0 <= k1 < 2 * n - 1:
                qd[k1] = 1.0
            if 0 <= k2 < 2 * n - 1 and k2 != k1:
                qd[k2] = 1.0
            ds = np.linspace(d_min, d_max, n_coarse)
            vals = np.array([_eval_pair(corr, d, gd, qd) for d in ds])
            bi = np.argmin(vals)
            cur_max = np.max(corr)
            if vals[bi] >= cur_max - 1e-14:
                continue
            lo = max(d_min, ds[max(0, bi - 1)])
            hi_r = min(d_max, ds[min(n_coarse - 1, bi + 1)])
            ds_f = np.linspace(lo, hi_r, n_fine)
            vals_f = np.array([_eval_pair(corr, d, gd, qd) for d in ds_f])
            bi_f = np.argmin(vals_f)
            d_best = ds_f[bi_f]
            h[i] = hi + d_best
            h[j] = hj - d_best
            corr = corr + d_best * gd
            nz = np.nonzero(qd)[0]
            for k in nz:
                corr[k] += d_best * d_best * qd[k]
            improved = True
    return h, corr, improved

def _slsqp_step(h, n, target, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con, 'args': (target,)}]
    try:
        res = minimize(_obj_smax, h, args=(alpha,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'maxls': 40})
        return _repair(res.x, n)
    except Exception:
        return h

def _subgrad_step(h, n, lr):
    corr = _corr(h)
    cmax = float(np.max(corr))
    tol = cmax * 1e-6 + 1e-12
    active = np.where(corr >= cmax - tol)[0]
    grad = np.zeros(n)
    for k in active:
        lag = (n - 1) - k
        m = np.arange(n)
        mk = m + lag
        valid1 = (mk >= 0) & (mk < n)
        grad[valid1] += (1.0 - h[mk[valid1]])
        mk2 = m - lag
        valid2 = (mk2 >= 0) & (mk2 < n)
        grad[valid2] -= h[mk2[valid2]]
    grad /= len(active)
    h_new = h - lr * grad
    return _repair(h_new, n)

def _make_init(n, seed_val):
    rng = np.random.RandomState(seed_val)
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + 0.3 * np.cos(np.pi * x) + 0.1 * np.cos(2 * np.pi * x)
    h = _repair(h, n)
    h += 0.05 * rng.randn(n)
    h = _repair(h, n)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)
    n = len(initial_h_values)
    dx = 2.0 / n
    target = n / 2.0

    h0 = _repair(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _c5(best_h, dx)

    # Phase 1: SLSQP warm-up
    for alpha in [50, 200, 600, 1500]:
        if time.time() - t0 > budget_s * 0.10:
            break
        hc = _slsqp_step(best_h, n, target, alpha, 300)
        c5 = _c5(hc, dx)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = hc.copy()

    # Phase 2: Pair sweeps + subgradient + periodic SLSQP
    stall = 0
    iter_count = 0
    while time.time() - t0 < budget_s - 60:
        iter_count += 1
        h = best_h.copy()
        corr = _corr(h)
        time_left = budget_s - 60 - (time.time() - t0)
        chunk = min(35.0, time_left)
        h, corr, imp = _pair_sweep(h, n, corr, time.time() + chunk, 31, 25)
        if imp:
            c5 = float(np.max(corr) * dx)
            if c5 < best_c5 - 1e-15:
                best_c5 = c5
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        # Subgradient nudges every few iterations
        if iter_count % 3 == 0 and time.time() - t0 < budget_s - 80:
            lr = max(0.0005, 0.005 * (1.0 - (time.time() - t0) / budget_s))
            hs = _subgrad_step(best_h, n, lr)
            c5_s = _c5(hs, dx)
            if c5_s < best_c5 - 1e-14:
                best_c5 = c5_s
                best_h = hs.copy()
                stall = 0

        # Periodic SLSQP
        if iter_count % 5 == 0 and time.time() - t0 < budget_s - 80:
            alpha = 200 + 100 * (iter_count % 20)
            hc = _slsqp_step(best_h, n, target, alpha, 250)
            hc_corr = _corr(hc)
            hc, hc_corr, _ = _pair_sweep(hc, n, hc_corr, time.time() + 15, 25, 20)
            c5_t = float(np.max(hc_corr) * dx)
            if c5_t < best_c5 - 1e-15:
                best_c5 = c5_t
                best_h = hc.copy()
                stall = 0

        # Perturbation when stalled
        if stall >= 5 and time.time() - t0 < budget_s - 80:
            scale = min(0.04, 0.01 + 0.005 * stall)
            h_pert = best_h + scale * np.random.randn(n)
            h_pert = _repair(h_pert, n)
            hc = _slsqp_step(h_pert, n, target, 500, 200)
            hc_corr = _corr(hc)
            hc, hc_corr, _ = _pair_sweep(hc, n, hc_corr, time.time() + 20, 25, 20)
            c5_t = float(np.max(hc_corr) * dx)
            if c5_t < best_c5 - 1e-15:
                best_c5 = c5_t
                best_h = hc.copy()
            stall = 0

    # Phase 3: Final high-precision sweeps
    if time.time() - t0 < budget_s - 20:
        for _ in range(4):
            if time.time() - t0 > budget_s - 12:
                break
            h = best_h.copy()
            corr = _corr(h)
            h, corr, _ = _pair_sweep(h, n, corr, time.time() + 15, 41, 35)
            c5 = float(np.max(corr) * dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    # Phase 4: Final SLSQP polish
    if time.time() - t0 < budget_s - 5:
        try:
            hc = _slsqp_step(best_h, n, target, 3000, 500)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hc.copy()
        except Exception:
            pass

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
```
