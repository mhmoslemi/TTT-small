import time

def correlate_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_of(h, n):
    return float(np.max(correlate_full(h)) * 2.0 / n)

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(2000):
        s = h.sum()
        if s < 1e-15:
            h[:] = 0.5
            break
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
        if abs(h.sum() - target) < 1e-13:
            break
    d = target - h.sum()
    if abs(d) > 1e-13:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if abs(d) < 1e-13:

                a = min(d, 1.0 - h[i]); h[i] += a; d -= a
            else:
                a = min(-d, h[i]); h[i] -= a; d += a

def obj_and_grad(h, alpha, topk, flatw, pen):
    n = len(h)
    corr = correlate_full(h)
    m = np.max(corr)
    exp_args = np.clip(alpha * (corr - m), -300, 300)
    exp_s = np.exp(exp_args)
    sum_exp = np.sum(exp_s)
    w = exp_s / sum_exp
    val = m + np.log(sum_exp) / alpha
    if topk > 1:
        sc = np.sort(corr)[::-1][:topk]
        val += flatw * np.var(sc)
    val += pen * (h.sum() - n / 2.0) ** 2
    grad = np.zeros(n)
    for lag_idx in range(2 * n - 1):
        wi = w[lag_idx]
        if wi < 1e-12:
            continue
        lag = lag_idx - (n - 1)
        if lag >= 0:
            for j in range(n):
                g = 0.0
                if lag <= j <= n - 1:
                    g += (1.0 - h[j - lag])
                if 0 <= j <= n - 1 - lag:
                    g -= h[j + lag]
                grad[j] += wi * g
        else:
            L = -lag
            for j in range(n):
                g = 0.0
                if 0 <= j <= n - 1 - L:
                    g += (1.0 - h[j + L])
                if L <= j <= n - 1:
                    g -= h[j - L]
                grad[j] += wi * g
    if topk > 1:
        sc_sorted = np.sort(corr)[::-1][:topk]
        mean_sc = np.mean(sc_sorted)
        top_indices = np.argsort(-corr)[:topk]
        for ti in top_indices:
            gav = 2.0 * flatw * (corr[ti] - mean_sc) / topk
            lag = ti - (n - 1)
            for j in range(n):
                g = 0.0
                if lag >= 0:
                    if lag <= j:
                        g += (1.0 - h[j - lag])
                    if j <= n - 1 - lag:
                        g -= h[j + lag]
                else:
                    L = -lag
                    if j <= n - 1 - L:
                        g += (1.0 - h[j + L])
                    if j >= L:
                        g -= h[j - L]
                grad[j] += gav * g
    grad += 2.0 * pen * (h.sum() - n / 2.0)
    return val, grad

def do_slsqp(h, n, alpha, topk, flatw, maxiter):

def coord_descent(h, n, t_end, rng, n_rounds=500):
    h = h.copy()
    best = c5_of(h, n)
    steps = [0.05, 0.025, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    step_idx = 0
    stagnant = 0
    for r in range(n_rounds):
        if time.time() > t_end:
            break
        # Focus on active lag
        corr = correlate_full(h)
        active_lag = np.argmax(corr) - (n - 1)
        improved = False
        for _ in range(3 * n):
            if time.time() > t_end:
                break
            i = rng.randint(n)
            # Bias j towards i + active_lag
            if rng.rand() < 0.5:
                j = int(np.clip(i + active_lag, 0, n - 1))
            else:
                j = rng.randint(n)
            if i == j:
                continue
            hi, hj = h[i], h[j]
            for si in range(step_idx, min(step_idx + 3, len(steps))):
                d = steps[si]
                for sign in (1, -1):
                    ni, nj = hi + sign * d, hj - sign * d
                    if ni < -1e-10 or ni > 1.0 + 1e-10 or nj < -1e-10 or nj > 1.0 + 1e-10:
                        continue
                    h[i], h[j] = ni, nj
                    c5_new = c5_of(h, n)
                    if c5_new < best - 1e-14:
                        best = c5_new
                        improved = True
                        step_idx = max(0, si - 1)
                        break
                    h[i], h[j] = hi, hj
                if improved:
                    break
        if not improved:
            stagnant += 1
            if step_idx < len(steps) - 1:
                step_idx += 1
            if stagnant > 80:
                break
        else:
            stagnant = 0
    return h, best
