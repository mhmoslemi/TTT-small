    # Project onto {0≤h≤1, sum(h)=n/2}
    h = h.clip(0, 1)
    # Adjust to sum = n/2
    target = n/2
    diff = target - h.sum()
    if diff > 0:
        # Add to smallest elements
        order = np.argsort(h)
        for idx in order:
            add = min(diff, 1-h[idx])
            h[idx] += add
            diff -= add
            if diff <= 0: break

        for idx in order:
            sub = min(-diff, h[idx])
            h[idx] -= sub
            diff += sub
            if diff >= 0: break
    return h

Given h and we want to change h[i] += d, h[j] -= d (preserving the sum), what's the optimal d?

c[k] = sum_l h[l](1-h[l+lag_k])

∂c[k]/∂h[i] = (1-h[i+lag_k]) - h[i-lag_k] (for valid indices)
∂c[k]/∂h[j] = (1-h[j+lag_k]) - h[j-lag_k]

After the change:
c[k] + d * [(∂c[k]/∂h[i]) - (∂c[k]/∂h[j])] (first order)

But c[k] is quadratic in h, so:
c[k](h + d*e_i - d*e_j) = c[k](h) + d*(g_i[k] - g_j[k]) + d² * (second order terms)

The second order: c[k] involves terms h[l]*h[l+lag]. The cross terms involving h[i] and h[j]:
- h[i]*h[i+lag]: if i+lag = j, then we get h[i]*h[j], and the change is d*(-d) = -d²
- h[j]*h[j+lag]: if j+lag = i, then we get h[j]*h[i], and the change is (-d)*d = -d²
- etc.

So the second order is small (O(d²)) and for small d, the first order approximation is good.

The objective is max_k c[k]. The subgradient at the optimum is the indicator of the argmax. So the "gradient" direction is:
d * sum_{k in active} w_k * (g_i[k] - g_j[k])

To minimize, we want d * (sum w_k * g_i[k] - sum w_k * g_j[k]) < 0.

For the smart search:
1. Compute the "sensitivity" of h[i] and h[j]: s_i = weighted sum of (1-h[i+lag]-h[i-lag]) over active lags, similarly s_j.
2. The optimal d (ignoring constraints) is 0 if s_i ≈ s_j, or moves mass from higher sensitivity to lower.
3. The exact optimal d can be found by 1D bisection on the max of the linear functions.

- The function is max_k (c[k] + d * b_k) where b_k = g_i[k] - g_j[k]
- The minimum is where the two lines with the largest slopes on either side of 0 cross

    best_c = _c5(h, n)
    h = h.copy()
    deltas = np.array([0.001, 0.002, 0.005, 0.01, 0.02, 0.04,
                       -0.001, -0.002, -0.005, -0.01, -0.02, -0.04])
    for _pass in range(2000):
        if time.time() > t_end:
            break
        cv = _c5_vals(h, n)
        top_idx = np.argsort(cv)[::-1][:topk]
        improved = False
        n2 = n
        for i in range(n2):
            if time.time() > t_end:
                break
            for j in range(i + 1, n2):
                if time.time() > t_end:
                    break
                hi, hj = h[i], h[j]
                for d in deltas:
                    ni, nj = hi + d, hj - d
                    if ni < -1e-12 or ni > 1.0 + 1e-12 or nj < -1e-12 or nj > 1.0 + 1e-12:
                        continue
                    h[i], h[j] = ni, nj
                    c_new = _c5(h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        hi, hj = ni, nj
                        improved = True
                    else:
                        h[i], h[j] = hi, hj
        if not improved:
            break
    return h, best_c

def _fine_coord(h, n, t_end):
    best_c = _c5(h, n)
    h = h.copy()
    tiny = np.array([0.0002, 0.0005, 0.001, 0.002, -0.0002, -0.0005, -0.001, -0.002])
    rng = np.random.RandomState(12345)
    while time.time() < t_end:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = tiny[rng.randint(len(tiny))]
        hi, hj = h[i], h[j]
        ni, nj = hi + d, hj - d
        if 0.0 <= ni <= 1.0 and 0.0 <= hj + (hj - nj) <= 1.0:
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h[i], h[j] = ni, nj
                c = _c5(h, n)
                if c < best_c:
                    best_c = c
                else:
                    h[i], h[j] = hi, hj
    return h, best_c

def _anneal(h, n, t_end, rng):
    best_c = _c5(h, n)
    h = h.copy()
    c_cur = best_c
    T = 0.0002
    min_T = 1e-10
    cool = 0.9999
    while time.time() < t_end:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(-0.05, 0.05)
        hi, hj = h[i], h[j]
        ni, nj = hi + d, hj - d
        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
            continue
        h[i], h[j] = ni, nj
        c_new = _c5(h, n)
        if c_new < c_cur or (T > 1e-11 and rng.rand() < np.exp((c_cur - c_new) / T)):
            c_cur = c_new
            if c_cur < best_c:
                best_c = c_cur
        else:
            h[i], h[j] = hi, hj
        T = max(T * cool, min_T)
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    def remaining():
        return budget_s - elapsed()

    n = len(initial_h_values)
    h0 = _project(np.array(initial_h_values, dtype=float).copy(), n)
    best_h = h0.copy()
    best_c5 = _c5(best_h, n)

    # Phase 1: Nesterov projected gradient (fast, many iterations)
    alphas = [500, 1500, 4000, 10000, 25000]
    for alpha in alphas:
        if remaining() < 80:
            break
        if best_c5 < 0.38085 and alpha > 4000:
            break
        t_end = t0 + budget_s * (0.18 if alpha <= 1500 else 0.10)
        best_h, best_c5 = _nesterov_proj(best_h, n, alpha, 8000, t_end)

    # Phase 2: SLSQP with top-k objectives
    for kmax in [3, 5]:
        if remaining() < 100:
            break
        for r in range(6 if kmax == 3 else 4):
            if remaining() < 60:
                break
            if r == 0:
                h_init = best_h.copy()
            else:
                h_init = best_h + 0.02 * rng.randn(n)
                h_init = _project(h_init, n)
            hc, c5 = _slsqp_opt(h_init, n,
                                lambda x: _topk_obj_grad(x, n, kmax)[0],
                                lambda x: _topk_obj_grad(x, n, kmax)[1],
                                500)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hc

    # Phase 3: SLSQP with smooth max
    for alpha in [8000, 20000]:
        if remaining() < 80:
            break
        hc, c5 = _slsqp_opt(best_h, n,
                            lambda x: _smooth_obj_grad(x, n, alpha)[0],
                            lambda x: _smooth_obj_grad(x, n, alpha)[1],
                            800)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hc

    # Phase 4: Simulated annealing
    if remaining() > 120:
        t_end = t0 + min(budget_s - 60, elapsed() + 80)
        best_h, best_c5 = _anneal(best_h, n, t_end, rng)

    # Phase 5: Local search with pair exchanges
    if remaining() > 60:
        t_end = t0 + min(budget_s - 30, elapsed() + 120)
        best_h, best_c5 = _local_search(best_h, n, t_end, topk=5)

    # Phase 6: Fine coordinate search
    if remaining() > 20:
        t_end = t0 + budget_s - 10
        best_h, best_c5 = _fine_coord(best_h, n, t_end)

    # Phase 7: Try n=100
    if remaining() > 80:
        n2 = 100
        x_old = np.linspace(0, 2.0, n)
        x_new = np.linspace(0, 2.0, n2)
        h2 = np.interp(x_new, x_old, h0)
        h2 = _project(h2, n2)
        h2_best = h2.copy()
        h2_c5 = _c5(h2, n2)
        for alpha in [5000, 15000]:
            if remaining() < 40:
                break
            t_end = t0 + min(budget_s - 20, elapsed() + 30)
            h2_best, h2_c5 = _nesterov_proj(h2_best, n2, alpha, 5000, t_end)
        if remaining() > 40:
            hc2, c2 = _slsqp_opt(h2_best, n2,
                                  lambda x: _topk_obj_grad(x, n2, 3)[0],
                                  lambda x: _topk_obj_grad(x, n2, 3)[1], 400)
            if _c5(hc2, n2) < h2_c5:
                h2_best, h2_c5 = hc2, _c5(hc2, n2)
        if remaining() > 20:
            h2_best, h2_c5 = _local_search(h2_best, n2, t0 + budget_s - 10, topk=5)
        if h2_c5 < best_c5:
            best_h, best_c5, n = h2_best, h2_c5, n2

    best_h = _project(best_h, n)
    best_c5 = _c5(best_h, n)
    return (best_h, best_c5, n)
