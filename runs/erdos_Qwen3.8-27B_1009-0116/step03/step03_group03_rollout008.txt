            h[idx] += adj
            diff -= adj

    g = np.zeros(n)
    if k >= 0:

        # d/dh[i+k] for i in 0..n-1-k: -h[i]

        # Wait, let me re-derive.

        #   if j appears in h[i] (i.e., j=0..n-1-kk): contributes -h[j+kk]

import cvxpy as cp

def _lp_linearization(h_curr, n, t_init, num_shifts=139):
    """Solve LP relaxation with linearized quadratic terms."""
    h = cp.Variable(n)
    t = cp.Variable()
    target = n / 2.0

    for m in range(2 * n - 1):

        if k >= 0:

            # Linearize: ≈ sum h[i] - sum(h_curr[i]*h[i+k] + h[i]*h_curr[i+k] - h_curr[i]*h_curr[i+k])

At point h*, for the term h[i]*(1-h[i+k]):
f(h) = h[i] - h[i]*h[i+k]

= h*[i] - h*[i]*h*[i+k] + (1-h*[i+k])*(h[i]-h*[i]) - h*[i]*(h[i+k]-h*[i+k])

Wait, I need to be more careful. The term h[i]*h[i+k] is quadratic. The first-order Taylor expansion at h* is:
h[i]*h[i+k] ≈ h*[i]*h*[i+k] + (1)*h*[i+k]*(h[i]-h*[i]) + h*[i]*(1)*(h[i+k]-h*[i+k])
= h*[i]*h*[i+k] + h*[i+k]*h[i] - h*[i]*h*[i+k] + h*[i]*h[i+k] - h*[i]*h*[i+k]
= h*[i+k]*h[i] + h*[i]*h[i+k] - h*[i]*h*[i+k]

So the linearized c_k is:
sum_i [h[i] - (h*[i+k]*h[i] + h*[i]*h[i+k] - h*[i]*h*[i+k])]
= sum_i [(1-h*[i+k])*h[i] - h*[i]*h[i+k] + h*[i]*h*[i+k]]

The term h*[i]*h[i+k] is linear in h (h*[i] is a constant coefficient, h[i+k] is the variable). So yes, the whole expression IS linear in h! Great.

So the LP is:
min t
s.t. for each k: sum_i [(1-h*[i+k])*h[i] - h*[i]*h[i+k]] + const_k ≤ t
     sum h = 35, 0≤h≤1

where const_k = sum_i h*[i]*h*[i+k].

This is a valid LP! Let me implement it properly.

For k≥0 (i ranges 0 to n-1-k):
Linear part: sum_{i=0}^{n-1-k} [(1-h*[i+k])*h[i] - h*[i]*h[i+k]]
Const: sum_{i=0}^{n-1-k} h*[i]*h*[i+k]

For k<0, let kk=-k (i ranges 0 to n-1-kk):
c = sum_{i=0}^{n-1-kk} h[i+kk]*(1-h[i])
Linear: sum_{i=0}^{n-1-kk} [(1-h*[i])*h[i+kk] - h*[i+kk]*h[i]]
Const: sum_{i=0}^{n-1-kk} h*[i+kk]*h*[i]

This can be written as:
For k≥0: A_k @ h + b_k ≤ t where A_k has:
  A_k[i] += (1-h*[i+k]) for i in 0..n-1-k
  A_k[i+k] -= h*[i] for i in 0..n-1-k
  b_k = sum h*[i]*h*[i+k]

For k<0 (kk=-k):
  A_k[i+kk] += (1-h*[i]) for i in 0..n-1-kk
  A_k[i] -= h*[i+kk] for i in 0..n-1-kk
  b_k = sum h*[i+kk]*h*[i]

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n * 0.5
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    if abs(s - target) > 1e-12:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            adj = min(diff, 1.0 - h[idx]) if diff > 0 else max(diff, h[idx] - 1.0)
            h[idx] += adj
            diff -= adj
            if abs(diff) < 1e-14:
                break
    return h

def _grad_shift(h, n, k):
    g = np.zeros(n)
    if k >= 0:
        g[:n-k] += (1.0 - h[k:])
        g[k:] -= h[:n-k]
    else:
        kk = -k
        g[kk:] += (1.0 - h[:n-kk])
        g[:n-kk] -= h[kk:]
    return g
