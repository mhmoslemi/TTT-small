import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smax(c, alpha):
    m = c.max()
    s = np.clip(alpha * (c - m), -500.0, 0.0)
    return m + float(np.log(np.sum(np.exp(s))) / alpha)

def _corr_g(h, n, mi):
    k = mi - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += 1.0 - h[:n-k]
        g[:n-k] -= h[k:]
    else:
        g[:n+k] += 1.0 - h[-k:]
        g[-k:] -= h[:n+k]
    return g

def _repair(h, n, tgt):
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= tgt / s
    h = np.clip(h, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-12:
        o = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in o:
            a = min(d, 1.0 - h[i]) if d > 0 else max(d, h[i] - 1.0)
            h[i] += a
            d -= a
            if abs(d) < 1e-12:
                break
    return h

def _adam_phase(h, n, tgt, dx, iters, t_left_fn, lr0=0.1, alpha0=20000):
    m_a = np.zeros(n)
    v_a = np.zeros(n)
    t = 1

    best_h = h.copy()
    for i in range(iters):
        if t_left_fn() < 30:
            break
        c = _corr(h)
        mc = c.max()
        w = np.exp(np.clip(alpha0*(c-mc), -500.0, 0.0))
        w /= w.sum()
        g = np.zeros(n)
        act = np.where(w > 1e-5)[0]
        for mi in act:
            g += w[int(mi)] * _corr_g(h, n, int(mi))

        vh = v_a / (1 - 0.999**t)
        lr = lr0 / (1 + 0.0005*i)
        h2 = h - lr * mh / (np.sqrt(vh) + 1e-8)
        h2 = _repair(h2, n, tgt)
        c52 = _c5(h2, dx)
        if c52 < best:
            best = c52
            best_h = h2
            h = h2
        elif c52 < best + 1e-8:
            h = h2  # accept small worsenings for exploration
        t += 1
    return best_h, best

def _subgrad_phase(h, n, tgt, dx, iters, t_left_fn):

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smax(c, alpha):
    m = c.max()
    s = np.clip(alpha * (c - m), -500.0, 0.0)
    return m + float(np.log(np.sum(np.exp(s))) / alpha)

def _corr_g(h, n, mi):
    k = mi - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += 1.0 - h[:n-k]
        g[:n-k] -= h[k:]
    else:
        g[:n+k] += 1.0 - h[-k:]
        g[-k:] -= h[:n+k]
    return g

def _repair(h, n, tgt):
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h *= tgt / s
    h = np.clip(h, 0.0, 1.0)
    d = tgt - h.sum()
    if abs(d) > 1e-12:
        o = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in o:
            a = min(d, 1.0 - h[i]) if d > 0 else max(d, h[i] - 1.0)
            h[i] += a
            d -= a
            if abs(d) < 1e-12:
                break
    return h

def _adam_phase(h, n, tgt, dx, iters, tl_fn, lr0, alpha0):
    m_a = np.zeros(n)
    v_a = np.zeros(n)
    t = 1
    best = _c5(h, dx)
    best_h = h.copy()
    for i in range(iters):
        if tl_fn() < 25:
            break
        c = _corr(h)
        mc = c.max()
        w = np.exp(np.clip(alpha0 * (c - mc), -500.0, 0.0))
        w /= w.sum()
        g = np.zeros(n)
        act = np.where(w > 1e-5)[0]
        for mi in act:
            g += w[int(mi)] * _corr_g(h, n, int(mi))
        m_a = 0.9 * m_a + 0.1 * g
        v_a = 0.999 * v_a + 0.001 * g * g
        mh = m_a / (1.0 - 0.9 ** t)
        vh = v_a / (1.0 - 0.999 ** t)
        lr = lr0 / (1.0 + 0.0003 * i)
        h2 = h - lr * mh / (np.sqrt(vh) + 1e-8)
        h2 = _repair(h2, n, tgt)
        c52 = _c5(h2, dx)
        if c52 < best:
            best = c52
            best_h = h2
            h = h2
        elif c52 < best + 1e-7:
            h = h2
        t += 1
    return best_h, best

def _subgrad_phase(h, n, tgt, dx, iters, tl_fn):
    best = _c5(h, dx)
    best_h = h.copy()
    for i in range(iters):
        if tl_fn() < 15:
            break
        c = _corr(h)
        mc = c.max()
        act = np.where(c > 0.998 * mc)[0]
        g = np.zeros(n)
        for mi in act:
            g += _corr_g(h, n, int(mi))
        g /= len(act)
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        step = 0.03 / (1.0 + 0.002 * i)
        h2 = h - step * g
        h2 = _repair(h2, n, tgt)
        c52 = _c5(h2, dx)
        if c52 < best:
            best = c52
            best_h = h2
            h = h2
    return best_h, best

def _binary_sa(h, n, tgt, dx, iters, tl_fn, rng):
    hb = np.where(h >= 0.5, 1.0, 0.0)
    hb = _repair(hb, n, tgt)
    cb = _c5(hb, dx)
    best_b = hb.copy()
    best_cb = cb
    T = 0.05
    for i in range(iters):
        if tl_fn() < 12:
            break
        T *= 0.9998
        if rng.rand() < 0.6:
            pos = np.where(hb > 0.5)[0]
            neg = np.where(hb < 0.5)[0]
            if len(pos) == 0 or len(neg) == 0:
                break
            i1 = pos[rng.randint(len(pos))]
            i2 = neg[rng.randint(len(neg))]
            hb[i1], hb[i2] = hb[i2], hb[i1]
            cn = _c5(hb, dx)
            if cn <= cb or rng.rand() < np.exp(-(cn - cb) / max(T, 1e-10)):
                cb = cn
            else:
                hb[i1], hb[i2] = hb[i2], hb[i1]
        else:
            j = rng.randint(n - 1)
            if hb[j] != hb[j+1]:

                cn = _c5(hb, dx)

                    cb = cn

                    hb[j], hb[j+1] = hb[j+1], hb[j]
        if cb < best_cb:
            best_cb = cb
            best_b = hb.copy()
    return best_b, best_cb

def _fine_cd(h, n, tgt, dx, iters, tl_fn, rng):
    best = _c5(h, dx)
    best_h = h.copy()
    steps = [0.004, 0.002, 0.001, -0.001, -0.002, -0.004, 0.0005, -0.0005]
    for i in range(iters):
        if tl_fn() < 8:
            break
        idx = rng.randint(n)
        c = _corr(h)
        mc = c.max()
        act = np.where(c > 0.96 * mc)[0]
        g_avg = np.zeros(n)
        for mi in act:
            g_avg += _corr_g(h, n, int(mi))
        g_avg /= len(act)
        sd = -1 if g_avg[idx] > 0 else 1
        for d in [steps[k] for k in range(len(steps)) if steps[k]*sd > 0] + \
                [steps[k] for k in range(len(steps)) if steps[k]*sd < 0]:
            h2 = h.copy()
            h2[idx] += d
            h2 = _repair(h2, n, tgt)
            c52 = _c5(h2, dx)
            if c52 < best:
                best = c52
                best_h = h2
                h = h2
                break
    return best_h, best
