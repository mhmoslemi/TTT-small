            c5_r = _c5(h2, dx)

                best_c5, best_h = c5_r, h2.copy()

    # Phase 5: Fine random pair search
    if elapsed() < budget_s - 40:
        h = best_h.copy()
        c5_cur = _c5(h, dx)
        nsteps = 0
        while elapsed() < budget_s - 40 and nsteps < 2000000:
            nsteps += 1
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = rng.choice([1e-4, 2e-4, 4e-4, 8e-4, 1.5e-3, 3e-3, -1e-4, -2e-4, -4e-4, -8e-4, -1.5e-3, -3e-3])
            ni, nj = h[i] + d, h[j] - d
            if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                h[i], h[j] = ni, nj
                c = _c5(h, dx)
                if c < c5_cur - 1e-16:
                    c5_cur = c
                    if c5_cur < best_c5:
                        best_c5, best_h = c5_cur, h.copy()
                else:
                    h[i] = ni - d
                    h[j] = nj + d

    # Phase 6: Final L-BFGS-B polish
    if elapsed() < budget_s - 10:
        try:
            res = minimize(_obj_pen, best_h, args=(n, 200000, 100000.0),
                         method='L-BFGS-B', bounds=bounds,
                         jac=lambda x, n2, a2, m2: _grad_pen(x, n2, a2, m2),
                         options={'maxiter': 5000, 'ftol': 1e-20, 'gtol': 1e-16})
            hc = _repair(res.x, n)
            c5 = _c5(hc, dx)
            if c5 < best_c5:
                best_c5, best_h = c5, hc
        except Exception:
            pass

    # Phase 7: Ultra-fine pair scan on best
    if elapsed() < budget_s - 5:
        h = best_h.copy()
        max_passes2 = 100
        for p in range(max_passes2):
            if elapsed() > budget_s - 5:
                break
            imp = False
            for i in range(n):
                if elapsed() > budget_s - 5:
                    break
                for j in range(i + 1, n):
                    h, ok = _pair_scan(h, i, j, n, dx, nscan=40)
                    if ok:
                        imp = True
            c5_chk = _c5(h, dx)
            if c5_chk < best_c5:
                best_c5, best_h = c5_chk, h.copy()
            if not imp:
                break

    best_h = _repair(best_h, n)
    best_c5 = _c5(best_h, dx)
    return (best_h, best_c5, n)
