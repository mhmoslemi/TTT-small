```python
import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Initial grid-based configuration
    num_circles = 26
    grid_rows = 5
    grid_cols = 6
    grid_spacing = 0.25
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)

    # Initial grid
    for i in range(grid_rows):
        for j in range(grid_cols):
            idx = i * grid_cols + j
            x = j * grid_spacing
            y = i * grid_spacing
            centers[idx] = [x, y]
            radii[idx] = 0.05

    # Constraints for optimization
    def constraints(centers_flat):
        centers = centers_flat.reshape(-1, 2)
        radii = np.full(len(centers), 0.05)  # Default radius
        return {
            'type': 'ineq',
            'fun': lambda x: np.min(x[:, 0] - radii) + 1e-8,  # Left boundary
            'type': 'ineq',
            'fun': lambda x: np.min(1.0 - x[:, 0] - radii) + 1e-8,  # Right boundary
            'type': 'ineq',
            'fun': lambda x: np.min(x[:, 1] - radii) + 1e-8,  # Bottom boundary
            'type': 'ineq',
            'fun': lambda x: np.min(1.0 - x[:, 1] - radii) + 1e-8,  # Top boundary
            'type': 'ineq',
            'fun': lambda x: np.min(np.sqrt(np.sum((x[:, np.newaxis] - x[np.newaxis, :], axis=2))**2) - (radii + radii[np.newaxis, :]) + 1e-8)
        }

    # Define the objective function to maximize the sum of radii
    def objective(x):
        centers = x.reshape(-1, 2)
        return -np.sum(np.full(len(centers), 0.05))  # Minimize negative sum of radii

    # Prepare the initial guess
    initial_guess = centers.flatten()

    # Solve the optimization problem
    result = optimize.minimize(
        fun=objective,
        x0=initial_guess,
        method='SLSQP',
        jac=None,
        bounds=[(0, 1) for _ in range(2 * num_circles)],
        constraints=constraints,
        tol=1e-8
    )

    if not result.success:
        raise RuntimeError(f"Optimization failed: {result.message}")

    centers_opt = result.x.reshape(-1, 2)
    radii_opt = np.full(num_circles, 0.05)

    # Final check

    if not success:
        raise RuntimeError(msg)

    sum_radii = np.sum(radii_opt)
    return centers_opt, radii_opt, sum_radii
```

```python
def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
```
