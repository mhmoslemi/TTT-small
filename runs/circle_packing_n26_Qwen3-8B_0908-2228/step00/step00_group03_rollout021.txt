import scipy.optimize as optimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial hexagonal grid packing with 26 circles
    n = 26
    # Compute the number of rows and columns
    rows = int(np.ceil(np.sqrt(n)))
    cols = int(np.ceil(n / rows))
    
    # Compute the radius of a hexagonally packed circle
    # We will scale based on the unit square and try to fit as many as possible
    # The spacing between centers in a hexagonal arrangement is 2r
    # Let's find the initial radius based on the grid size
    # First, define the initial grid of centers
    # Using the hexagonal grid with vertical spacing sqrt(3)/2 * 2r = sqrt(3)r
    # We will adjust the grid to fit within [0,1]^2

    # Start with a hexagonal grid
    r_initial = 0.13
    centers = np.zeros((n, 2))
    count = 0
    for row in range(rows):
        for col in range(cols):
            if count >= n:
                break
            x = col * (2 * r_initial) + r_initial
            y = row * (np.sqrt(3) * r_initial) + r_initial
            if row % 2 == 1:
                x += r_initial
            if x < 1 - r_initial and y < 1 - r_initial:
                centers[count] = [x, y]
                count += 1

    # Now, optimize the radii
    # Define the optimization problem: maximize sum of radii with constraints
    # We define a function to compute objective and constraints
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    def constraint_overlap(i, j, radii):
        # Distance between centers i and j
        dx = centers[i, 0] - centers[j, 0]
        dy = centers[i, 1] - centers[j, 1]
        dist = np.sqrt(dx**2 + dy**2)
        return dist - (radii[i] + radii[j])  # >= 0 to ensure no overlap

    def constraint_boundary(i, radii):
        x, y = centers[i]
        return x - radii[i] >= -1e-12 and x + radii[i] <= 1 + 1e-12 and y - radii[i] >= -1e-12 and y + radii[i] <= 1 + 1e-12

    # Set up the optimization problem
    bounds = [(0.0, 0.5) for _ in range(n)]
    constraints = []

    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({'type': 'ineq', 'fun': lambda r, i=i, j=j: constraint_overlap(i, j, r)})
        constraints.append({'type': 'ineq', 'fun': lambda r, i=i: constraint_boundary(i, r)})

    # Run the optimization
    result = optimize.minimize(

        x0=np.full(n, r_initial),

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
