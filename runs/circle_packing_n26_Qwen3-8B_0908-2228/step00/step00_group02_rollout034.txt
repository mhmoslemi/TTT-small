import scipy.optimize as optimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    num_circles = 26

    cols = int(np.ceil(num_circles / rows))
    # Choose an initial approximate grid
    num_rows = 5
    num_cols = 5

    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)

        col = i % num_cols

            x += 0.5 / (num_cols - 1)
        centers[i] = np.array([x, y])
        radii[i] = 0.05  # Initial guess for radius

    # Define the objective function to maximize the sum of radii
    def objective(radii_flat):
        return -np.sum(radii_flat)  # Minimize negative sum (equivalent to maximizing sum)

    # Define constraints
    def constraints_func(x):
        centers_flat = x[:num_circles * 2].reshape(num_circles, 2)
        radii_flat = x[num_circles * 2:]
        constraints = []
        # Boundary constraints
        for i in range(num_circles):
            x_i, y_i = centers_flat[i]
            r_i = radii_flat[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12 or y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
