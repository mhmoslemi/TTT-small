from scipy.optimize import minimize
from scipy.spatial.distance import cdist
from scipy.optimize import Bounds

def run_packing():
    # Define initial configuration: Hexagonal close packing in a unit square
    n = 26
    rows = int(np.ceil(np.sqrt(n)))
    cols = int(np.ceil(n / rows))
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    for i in range(n):
        row = i // cols

        centers[i, 1] = y

    def objective(radii):
        return -np.sum(radii)

        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                bounds_violation += 10000  # Penalize hard boundary violations

        overlap_violation = 0

                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-12:
                    overlap_violation += 10000  # Penalize overlaps
        
        return bounds_violation + overlap_violation
    
    # Objective and constraints
    def total_objective(radii):
        return objective(radii) + constraints(centers, radii)
    
    # Bounds for radii
