# Define the run_packing function
def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial guess for circle centers (regular grid)
    centers = np.array([[x, y] for y in np.linspace(0.1, 0.9, 6) for x in np.linspace(0.1, 0.9, 6)])
    # Start with equal radii

    centers = centers[:26]
    radii = radii[:26]
    
    # Define the objective function to maximize the sum of radii
    def objective(x):
        # x is a flat array of [x1, y1, r1, x2, y2, r2, ..., x26, y26, r26]
        x_flat = np.array(x)
        centers_flat = x_flat[:26*2].reshape(26, 2)
        radii_flat = x_flat[26*2:].reshape(26)
        
        # Check if any radius is negative
        if np.any(radii_flat < 0):
            return -np.inf
        
        # Check if any circle is outside the unit square
        for i in range(26):
            x, y = centers_flat[i]
            r = radii_flat[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12 or 
                y - r < -1e-12 or y + r > 1 + 1e-12):
                return -np.inf
        
        # Check for overlap between circles
        for i in range(26):
            for j in range(i + 1, 26):
                dx = centers_flat[i, 0] - centers_flat[j, 0]
                dy = centers_flat[i, 1] - centers_flat[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii_flat[i] + radii_flat[j] - 1e-12:
                    return -np.inf
        
        # Return the negative sum of radii (we minimize this)
        return -np.sum(radii_flat)
    
    # Define the bounds for each variable
    bounds = []
    for i in range(26):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 1.0)])  # x, y, r for each circle
    
    # Optimize
    result = optimize.minimize(objective, x0=np.concatenate([centers.flatten(), radii]), 
                               bounds=bounds, method='L-BFGS-B', options={'maxiter': 500})
    
    # Extract the result
    optimized_centers = result.x[:26*2].reshape(26, 2)
    optimized_radii = result.x[26*2:].reshape(26)

        return np.zeros((26, 2)), np.zeros(26), 0.0

    if not valid:
        return np.zeros((26, 2)), np.zeros(26), 0.0
    
    return optimized_centers, optimized_radii, np.sum(optimized_radii)

# Define the validation function
def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
