import numpy as np
import scipy.optimize as opt
from typing import Tuple

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    # Initial hexagonal packing with 26 circles
    n = 26
    rows = int(np.sqrt(n)) + 1
    cols = int(np.ceil(n / rows))
    
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    for i in range(n):
        row = i // cols
        col = i % cols
        x = col + 0.5 * (1 - (row % 2))  # Offset alternate rows to make hexagonal
        y = row * np.sqrt(3) / 2
        centers[i] = [x, y]
        radii[i] = 0.05  # Initial small radius for optimization

    # Define the objective function: maximize the sum of radii
    def objective(radii_vec):
        return -np.sum(radii_vec)  # Negative for minimization

    # Define constraints
    # 1. Non-overlap constraints
    def non_overlap_cons(x):
        centers_flat = x[:n*2].reshape(n, 2)
        radii_flat = x[n*2:]
        for i in range(n):
            for j in range(i+1, n):
                dx = centers_flat[i, 0] - centers_flat[j, 0]
                dy = centers_flat[i, 1] - centers_flat[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii_flat[i] + radii_flat[j] - 1e-8:
                    return -1.0  # Violation
        return 0.0

    # 2. Boundary constraints
    def boundary_cons(x):
        centers_flat = x[:n*2].reshape(n, 2)
        radii_flat = x[n*2:]
        for i in range(n):
            if centers_flat[i, 0] - radii_flat[i] < 0 or centers_flat[i, 0] + radii_flat[i] > 1:
                return -1.0
            if centers_flat[i, 1] - radii_flat[i] < 0 or centers_flat[i, 1] + radii_flat[i] > 1:
                return -1.0
        return 0.0

    # 3. Non-negative radii
    def non_negative_radii(x):
        radii_flat = x[n*2:]
        if np.any(radii_flat < 0):
            return -1.0
        return 0.0

    # Combine constraints
    constraints = [
        {'type': 'ineq', 'fun': non_overlap_cons},
        {'type': 'ineq', 'fun': boundary_cons},
        {'type': 'ineq', 'fun': non_negative_radii}
    ]

    # Initial guess: Flatten centers and radii
    x0 = np.concatenate([centers.flatten(), radii])

    # Solve the optimization problem
    result = opt.minimize(
        objective,
        x0,
        method='SLSQP',
        constraints=constraints,
        bounds=[(0, 1) for _ in range(n*2)] + [(0, 1) for _ in range(n)]
    )

    # Extract results
    centers_opt = result.x[:n*2].reshape(n, 2)
    radii_opt = result.x[n*2:]
    sum_radii = np.sum(radii_opt)

    # Validate the packing
    is_valid, message = validate_packing(centers_opt, radii_opt)
    if not is_valid:
        raise ValueError(f"Invalid packing: {message}")

    return centers_opt, radii_opt, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
