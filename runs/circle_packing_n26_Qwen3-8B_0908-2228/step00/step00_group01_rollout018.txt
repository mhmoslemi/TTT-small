import numpy as np
import scipy.optimize as opt
import random

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial grid-based circle placement (basic uniform distribution)
    n = 26
    grid_size = 6  # approximate number of rows and columns to fit 26 circles
    spacing = 1.0 / (grid_size - 1)
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    idx = 0

    # Initial grid-based placement
    for i in range(grid_size):
        for j in range(grid_size):
            if idx >= n:
                break
            x = i * spacing
            y = j * spacing
            centers[idx] = [x, y]
            # Radius based on distance to boundaries
            r = min(x, 1 - x, y, 1 - y)
            radii[idx] = r
            idx += 1

    # Introduce some perturbations to move circles near boundaries
    for i in range(n):
        if np.random.random() < 0.3:
            # Move toward boundary
            x, y = centers[i]
            direction = np.random.choice(['left', 'right', 'top', 'bottom'])
            if direction == 'left':
                centers[i][0] = max(0.1, x - 0.05)
            elif direction == 'right':
                centers[i][0] = min(0.9, x + 0.05)
            elif direction == 'top':
                centers[i][1] = min(0.9, y + 0.05)
            elif direction == 'bottom':
                centers[i][1] = max(0.1, y - 0.05)
            # Adjust radius based on new position
            r = min(centers[i][0], 1 - centers[i][0], centers[i][1], 1 - centers[i][1])
            radii[i] = r

    # Optimization function to maximize sum of radii while ensuring non-overlap and boundary constraints
    def objective(radii):
        return -np.sum(radii)  # We minimize the negative of the sum

    # Constraint functions
    def constraint_boundaries(radii, centers):
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                return -1
        return 0

    def constraint_overlap(radii, centers):
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx**2 + dy**2)

        return 0

    # Optimization
    bounds = [(0, 1) for _ in range(n)]
    result = opt.minimize(

    sum_radii = np.sum(radii_opt)
