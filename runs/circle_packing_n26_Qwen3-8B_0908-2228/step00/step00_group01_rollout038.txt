import numpy as np
from scipy.optimize import minimize, Bounds

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def objective_function(x, num_circles=26):
    # x is a flat array of x, y, r for each circle
    centers = x[:num_circles * 2].reshape((num_circles, 2))
    radii = x[num_circles * 2:].reshape((num_circles,))
    
    # Start with an initial guess for sum_radii
    sum_radii = np.sum(radii)
    
    # Check if this configuration is valid
    is_valid, message = validate_packing(centers, radii)
    
    return -sum_radii if is_valid else float('inf')  # Minimize negative sum to maximize

def run_packing():

    horizontal_spacing = 2 * 1 / (np.sqrt(3))

    for i in range(5):
