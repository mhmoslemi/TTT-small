import numpy as np
import cvxpy as cp

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26

    # Step 1: Define variables
    centers = cp.Variable((n, 2))
    radii = cp.Variable(n)

    # Objective: Maximize the sum of radii
    objective = cp.Maximize(cp.sum(radii))

    # Constraints:
    constraints = []

    # Circles must fit within the unit square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        constraints += [
            x - r >= 0,
            x + r <= 1,
            y - r >= 0,
            y + r <= 1
        ]

    # Circles cannot overlap
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i][0] - centers[j][0]
            dy = centers[i][1] - centers[j][1]
            dist = cp.sqrt(dx**2 + dy**2)
            constraints += [
                dist >= radii[i] + radii[j]
            ]

    # Constraints for non-negative radii
    constraints += [radii >= 0]

    # Solve the optimization problem
    problem = cp.Problem(objective, constraints)
    result = problem.solve()

    # Extract the centers and radii
    centers_np = centers.value
    radii_np = radii.value

    # Convert to numpy arrays
    centers_arr = np.array(centers_np)
    radii_arr = np.array(radii_np)

    # Compute the total sum of radii
    sum_radii = np.sum(radii_arr)

    # Return the centers, radii, and sum of radii
    return (centers_arr, radii_arr, sum_radii)
