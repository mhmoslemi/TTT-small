def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    cols = 5
    centers = np.zeros((n, 2))

        return -np.sum(radii)  # We minimize the negative of the sum to maximize it

    def residuals(params):

        for i in range(n):
