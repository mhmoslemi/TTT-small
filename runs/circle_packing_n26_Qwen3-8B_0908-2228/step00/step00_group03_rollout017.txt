import scipy.optimize as opt

def run_packing():
    # Number of circles
    n_circles = 26

    # Define the bounds for circle centers (within [0,1]x[0,1]) and radii (>=0)
    bounds = [(0.0, 1.0) for _ in range(2 * n_circles)] + [(0.0, 1.0) for _ in range(n_circles)]

    # Initial guess based on hexagonal packing
    def initial_hexagonal_centers(n):
        # Create a hexagonal packing in a square
        # Limit rows to fit within the square
        max_rows = int(np.sqrt(n))
        centers = []
        for i in range(max_rows):
            for j in range(i + 1):
                x = j + 0.5 * (i % 2)
                y = i * np.sqrt(3) / 2
                if x < 1 and y < 1:
                    centers.append([x, y])
                if x + 0.5 < 1 and y < 1:
                    centers.append([x + 0.5, y])
        # If we have more than n, truncate
        return np.array(centers[:n])

    centers_initial = initial_hexagonal_centers(n_circles)
    # Normalize to fit in the square
    centers_initial = (centers_initial - centers_initial.min(axis=0)) / (centers_initial.max(axis=0) - centers_initial.min(axis=0))
    radii_initial = np.full(n_circles, 0.05)

    # Flatten centers and radii for optimization
    x0 = np.hstack((centers_initial.flatten(), radii_initial))

    # Objective function: maximize sum of radii
    def objective(x):
        return -x[-n_circles:]

    # Constraints: no overlaps and within square
    def constraints(x):
        centers = x[:-n_circles].reshape(n_circles, 2)
        radii = x[-n_circles:]
        # Constraints: each circle must be inside [0,1]^2
        for i in range(n_circles):
            x_i, y_i = centers[i]
            r_i = radii[i]
            if x_i - r_i < 0 or x_i + r_i > 1 or y_i - r_i < 0 or y_i + r_i > 1:
                return np.inf
        # Constraints: no overlaps between circles
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-6:
                    return np.inf
        return 0

    # Optimization using Sequential Quadratic Programming
    result = opt.minimize(
        objective,
        x0,
        bounds=bounds,
        constraints={"type": "ineq", "fun": constraints},
        method="SLSQP",
        options={"maxiter": 100, "ftol": 1e-8}

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
