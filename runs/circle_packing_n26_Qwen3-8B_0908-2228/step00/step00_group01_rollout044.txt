import numpy as np
import scipy.optimize as opt

def run_packing():
    # Initial guess: place circles in an approximate hexagonal packing
    # We start by forming a hexagonal grid and then fine-tune

    # Step 1: Define grid dimensions
    # Instead of perfect hexagonal, we'll consider a triangular lattice
    num_rows = 6
    num_cols = 5

    # Generate an approximate hexagonal grid
    centers = np.zeros((num_rows * num_cols, 2))
    for i in range(num_rows):
        for j in range(num_cols):
            x = j + 0.5 * (i % 2)
            y = i * np.sqrt(3) / 2
            centers[i * num_cols + j] = [x, y]

    # Step 2: Scale to fit within [0,1]^2
    # Current grid goes from 0 to num_cols + 0.5 in x and to num_rows * sqrt(3)/2 in y
    x_scale = 1.0 / (num_cols + 0.5)
    y_scale = 1.0 / (num_rows * np.sqrt(3) / 2)
    centers *= [x_scale, y_scale]

    # Step 3: Add more circles to reach 26
    add_circles = 26 - len(centers)
    added_centers = np.random.rand(add_circles, 2)
    added_centers *= 2  # Ensure they are within [0,1]^2

    centers = np.vstack((centers, added_centers))

    # Step 4: Initial radii (uniform)
    radii = np.ones(len(centers)) * 0.05

    # Step 5: Define objective function and constraints for optimization
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum for maximization

    def constraint_center(radii, centers):
        return np.sum(radii)  # Keep sum of radii as constraint

    def constraint_overlap(radii, centers):
        n = len(centers)
        dists = np.zeros(n * (n - 1) // 2)
        idx = 0
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dists[idx] = np.sqrt(dx**2 + dy**2) - (radii[i] + radii[j])
                idx += 1
        return np.min(dists)  # Ensure minimum distance is at least zero

    def constraint_boundaries(radii, centers):
        for i in range(len(centers)):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                return -np.inf
        return 0

    # Define bounds and constraints
    bounds = [(0, 0.1) for _ in range(len(centers))]  # Initial radius bounds
    constraints = [
        {'type': 'eq', 'fun': constraint_center, 'args': (centers,)},
        {'type': 'ineq', 'fun': constraint_overlap, 'args': (centers,)},
        {'type': 'ineq', 'fun': constraint_boundaries, 'args': (centers,)},
    ]

    # Run optimization
    result = opt.minimize(
        fun=objective,
        x0=radii,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP'
    )

    # Extract results
    centers_result = centers
    radii_result = result.x
    sum_radii = np.sum(radii_result)

    # Validate the packing
    valid, message = validate_packing(centers_result, radii_result)
    if not valid:
        raise ValueError(f"Invalid packing: {message}")

    return centers_result, radii_result, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
