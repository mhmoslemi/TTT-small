import scipy.optimize as optimize

def get_centers(radius, n):
    """Returns positions for n circles placed in two rows."""
    # Bottom row
    bottom_centers = np.array([[i * 2 * radius, radius] for i in range(n)])
    # Top row
    top_centers = np.array([[i * 2 * radius, 1 - radius] for i in range(n)])
    return np.vstack((bottom_centers, top_centers))

def compute_sum_radii(radii):
    """Returns the sum of the given radii."""
    return np.sum(radii)

    for i in range(n):

        if (x - r < -1e-12 or x + r > 1 + 1e-12

            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

            if dist < radii[i] + radii[j] - 1e-12:
