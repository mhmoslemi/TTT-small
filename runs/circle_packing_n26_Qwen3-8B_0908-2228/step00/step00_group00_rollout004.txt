import numpy as np
import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial estimate of the radius (will be optimized)
    max_possible_radius = 0.25  # A rough estimate based on the unit square area

    # Initial placement in a hexagonal pattern
    # We will adjust the radius and positions via optimization
    # We start with a hexagonal grid

    # Define the number of rows and columns
    num_rows = 5
    num_cols = 6
    num_circles = num_rows * num_cols

    # Offset for hexagonal packing
    offset_x = np.sqrt(3) / 2 * max_possible_radius
    offset_y = max_possible_radius

    # Initial centers
    centers = np.zeros((num_circles, 2))
    for i in range(num_rows):
        for j in range(num_cols):
            x = j * offset_x
            y = i * offset_y
            if i % 2 == 1:
                x += offset_x / 2
            centers[i * num_cols + j] = [x, y]

    # Ensure centers are within the unit square
    centers = np.clip(centers, 0, 1)

    # Initial guess for radii
    radii = np.full(num_circles, max_possible_radius)

    # Objective function: maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)  # Negative because we are minimizing

    # Constraints: non-overlapping and within the unit square
    def constraints(radii):
        # Each circle must stay within the unit square
        for i in range(num_circles):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
                return False
        # Circles must not overlap
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return False
        return True

    # Optimization setup
    bounds = [(0, 1) for _ in range(num_circles)]  # Radii must be non-negative

    # Optimize
    result = opt.minimize(
        objective,
        radii,
        method='SLSQP',
        bounds=bounds,
        constraints={
            'type': 'ineq',
            'fun': constraints,
            'jac': None
        }
    )

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
