import scipy.optimize as optimize

        x = col / 5.5 + 0.1  # Shift slightly inside square
        y = row / 3.5 + 0.1

    def objective(x):

        centers_opt = x[:2*n].reshape(n, 2)

    def constraints(x):

        radii_opt = x[2*n:]
        constraints = []

        for i in range(n):

            constraints.append((x_center - 1e-12 <= 1) & (x_center + 1e-12 >= 0))
