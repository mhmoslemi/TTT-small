def compute_total_radius(radii):
    return np.sum(radii)

def update_radii(centers, max_radius):
    n = centers.shape[0]
    radii = np.zeros(n)
    for i in range(n):
        x, y = centers[i]
        r = max_radius * np.min([x, 1 - x, y, 1 - y])
        radii[i] = r
    return radii
