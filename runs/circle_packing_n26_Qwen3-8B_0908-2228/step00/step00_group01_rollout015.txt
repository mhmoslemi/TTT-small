import numpy as np
import scipy.optimize as optimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Define the number of circles
    num_circles = 26

    # Initial estimate for radius (symmetric configuration)
    initial_radius = 1.0 / np.sqrt(2 * num_circles)  # Simple initial guess

    # Hexagonal lattice configuration
    rows = int(np.ceil(np.sqrt(num_circles)))
    cols = int(np.ceil(num_circles / rows))
    centers = np.zeros((num_circles, 2))

    for i in range(num_circles):
        row = i // cols
        col = i % cols
        x = col + 0.5 * (row % 2)
        y = row * np.sqrt(3) / 2
        centers[i] = [x, y]

    # Normalize centers to fit within [0,1]^2
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))

    # Initial guess for radii
    radii = np.full(num_circles, initial_radius)

    # Objective function to maximize sum of radii
    def objective(x):
        return -x  # Minimize negative sum of radii

    # Constraints for optimization
    def constraints(x):
        # x is a flattened array of radii
        radii = x.reshape(-1)
        # Check if all radii are non-negative
        if np.any(radii < 0):
            return np.inf
        # Check if circles are within the unit square
        for i in range(num_circles):
            x_center, y_center = centers[i]
            if (x_center - radii[i] < -1e-12 or x_center + radii[i] > 1 + 1e-12 or
                y_center - radii[i] < -1e-12 or y_center + radii[i] > 1 + 1e-12):
                return np.inf
        # Check for overlaps

            for j in range(i + 1, num_circles):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx ** 2 + dy ** 2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return np.inf
        return 0  # No constraints violated

    # Set up optimization problem
    x0 = radii.copy()
    bounds = [(0, 1.0) for _ in range(num_circles)]
    result = optimize.minimize(
        objective,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'ineq', 'fun': constraints},
        tol=1e-8,

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
