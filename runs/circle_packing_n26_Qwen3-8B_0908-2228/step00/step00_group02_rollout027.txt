from scipy.optimize import minimize
from scipy.optimize import Bounds

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    num_circles = 26
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)

    rows = 5
    cols = 6
    spacing = 0.2  # Initial spacing between centers

            y = i * spacing * np.sqrt(3) / 2

                x += spacing / 2

        return -np.sum(radii)  # We want to maximize sum of radii

        return np.array([centers[i, 0] - radii[i] >= 0 for i in range(num_circles)] +
                        [centers[i, 0] + radii[i] <= 1 for i in range(num_circles)] +
                        [centers[i, 1] - radii[i] >= 0 for i in range(num_circles)] +

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
