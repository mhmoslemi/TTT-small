from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
import logging

# Set up logging

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial hexagonal packing with spacing to allow for optimization
    num_circles = 26
    radius_initial = 0.1  # Initial guess for radius
    centers = []
    radii = []

    # Try to fit a hexagonal grid inside the unit square
    spacing = radius_initial * 2  # Initial spacing between centers
    rows = int(np.sqrt(num_circles))
    cols = int(np.ceil(num_circles / rows))

    # Adjust for the unit square bounds
    max_col_distance = 1 - spacing
    max_row_distance = 1 - spacing

    # Generate hexagonal grid
    for i in range(rows):
        for j in range(cols):
            x = j * spacing
            y = i * (np.sqrt(3)/2) * spacing
            if x + spacing > 1 or y + spacing > 1:
                continue  # Skip circles that exceed the unit square
            centers.append([x, y])
            radii.append(radius_initial)

    # Convert initial centers and radii to numpy arrays
    centers = np.array(centers)
    radii = np.array(radii)

    # Define the function to optimize: maximize the sum of radii
    def objective(radii_new, centers):
        return -np.sum(radii_new)  # Minimize the negative sum (maximize sum)

    # Define constraints:
    # 1. Circles stay within [0,1] x [0,1]
    # 2. Circles do not overlap
    def constraint_centers(radii_new, centers):
        return np.array([np.clip(c[0] - r, 0, 1) for c, r in zip(centers, radii_new)] +
                        [np.clip(c[1] - r, 0, 1) for c, r in zip(centers, radii_new)])

    def constraint_radii(radii_new):
        return np.array([r for r in radii_new])

    def constraint_overlap(radii_new, centers):
        distances = pdist(centers)
        min_distance = np.min(distances)
        if min_distance < 0:
            return min_distance + 1e-6
        return np.min(distances - (radii_new[np.newaxis, :] + radii_new[:, np.newaxis])) - 1e-6

    # Set bounds for radii (non-negative)
    bounds = [(0, 1) for _ in range(num_circles)]

    # Optimization
    result = minimize(
        fun=objective,
        x0=radii,
        args=(centers,),
        method='SLSQP',
        bounds=bounds,
        constraints=[

    total_sum_radii = np.sum(optimal_radii)
