def hexagonal_grid(n_circles):
    rows = int(np.ceil(np.sqrt(n_circles)))
    cols = int(np.ceil(n_circles / rows))
    centers = np.zeros((n_circles, 2))
    idx = 0
    for row in range(rows):
        for col in range(cols):
            if row % 2 == 0:
                x = col * (1 + 1/(rows)) * 0.5
            else:
                x = (col + 0.5) * (1 + 1/(rows)) * 0.5
            y = row * (1 + 1/(cols)) * 0.5
            centers[idx] = [x, y]
            idx += 1
    # Adjust the grid to fit in the unit square
    centers = centers / (np.max(centers, axis=0) + 1e-12)
    return centers

def circle_overlap_penalty(centers, radii):
    n = centers.shape[0]
    penalty = 0
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist_sq = dx*dx + dy*dy
            if dist_sq < (radii[i] + radii[j])**2 - 1e-12:
                penalty += (radii[i] + radii[j] - np.sqrt(dist_sq))**2
    return penalty

def boundary_penalty(centers, radii):
    n = centers.shape[0]
    penalty = 0
