    # num_circles: number of circles
    # square_size: size of the square (default 1)
    # padding: padding around the grid (default 0.1)
    
    # Determine number of rows and columns
    num_rows = int(np.sqrt(num_circles))

        num_rows += 1
    num_cols = num_rows + 1  # Hexagonal packing

                break
        if idx >= num_circles:
            break

    # Convert from [0, 2(1-padding)] to [0, 1]

    return centers
