import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26

    # Step 1: Hexagonal packing initialization
    radius_base = 0.14  # approximate for hexagonal arrangement in unit square
    # Compute number of rows and columns based on hexagonal packing
    rows = int(np.sqrt(n)) + 1
    cols = int(np.sqrt(n)) + 1

    # Adjust rows and cols to fit n circles
    while rows * cols > n:
        rows = int(np.sqrt(n)) + 1
        cols = int(np.sqrt(n)) + 1

    # Offset for hexagonal packing
    offset = radius_base * np.sqrt(3) / 2

    # Initialize centers
    centers = np.zeros((n, 2))
    idx = 0
    for r in range(rows):
        for c in range(cols):
            if idx >= n:
                break
            x = c * (2 * radius_base) + radius_base
            y = r * (radius_base * 2) + radius_base
            if r % 2 == 1:
                y += offset
            centers[idx] = [x, y]
            idx += 1

    # Step 2: Optimization setup
    # Variables: (x1, y1, r1, x2, y2, r2, ..., x26, y26, r26) => [0, 1] x [0, 1] x [0, 1] for each circle
    # Flatten centers and radii
    x = centers.flatten()
    r = np.full(n, radius_base)

    # Define objective function: maximize sum of radii
    def objective(params):
        new_x = params[:2*n].reshape(n, 2)
        new_r = params[2*n:]
        return -np.sum(new_r)  # Negative for minimization

    # Define constraints: non-overlap, boundary, and radius >= 0
    def constraint_overlap(params):
        new_x = params[:2*n].reshape(n, 2)
        new_r = params[2*n:]
        for i in range(n):
            for j in range(i + 1, n):
                dx = new_x[i, 0] - new_x[j, 0]
                dy = new_x[i, 1] - new_x[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < new_r[i] + new_r[j] - 1e-8:
                    return dist - new_r[i] - new_r[j] + 1e-8
        return 0.0

    def constraint_boundary(params):
        new_x = params[:2*n].reshape(n, 2)
        new_r = params[2*n:]
        for i in range(n):
            x, y = new_x[i]
            r = new_r[i]

                return x - r + 1e-8
        return 0.0

    def constraint_radius(params):
        new_r = params[2*n:]
        for r in new_r:
            if r < 0 - 1e-8:
                return r + 1e-8
        return 0.0

    bounds = Bounds(
        np.zeros(2*n + n),  # Lower bound: all values ≥ 0
        np.array([1.0] * 2*n + [1.0] * n)  # Upper bound: all values ≤ 1
    )

    initial_guess = np.concatenate([x, r])

    # Optimization options
    options = {
        'maxiter': 1000,
        'ftol': 1e-8,
        'disp': False
    }

    def objective_wrapper(params):
        return objective(params)

    constraints = [
        {'type': 'ineq', 'fun': constraint_overlap},
        {'type': 'ineq', 'fun': constraint_boundary},
        {'type': 'ineq', 'fun': constraint_radius}
    ]

    # Run optimization
    result = minimize(
        objective_wrapper,
        initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options=options
    )

    # Extract result

    sum_radii = np.sum(optimal_radii)

    if not is_valid:

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
