import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            # All at bounds, do water-filling
            if diff > 0:
                zeros = np.where(h == 0.0)[0]
                h[zeros[:int(diff)+1]] = 1.0
            else:
                ones = np.where(h == 1.0)[0]
                h[ones[:int(-diff)+1]] = 0.0
        h = np.clip(h, 0.0, 1.0)
    return h

def fix_binary_sum(h, target_sum):
    """Fix binary array to have exactly target_sum ones."""
    h = h.copy()
    ones = np.sum(h == 1.0)
    if ones > target_sum:
        # Need to turn some 1s to 0s
        excess = ones - target_sum
        one_indices = np.where(h == 1.0)[0]
        for i in one_indices[:int(excess)]:
            h[i] = 0.0
    elif ones < target_sum:
        deficit = target_sum - ones
        zero_indices = np.where(h == 0.0)[0]
        for i in zero_indices[:int(deficit)]:
            h[i] = 1.0
    return h

def smooth_max_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m))
    return m + np.log(np.sum(exp_vals)) / alpha

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m))
    w = exp_vals / np.sum(exp_vals)
    n2 = 2 * n - 1
    i_idx = np.arange(n)
    k_idx = np.arange(n2)
    u_mat = i_idx[:, None] - k_idx[None, :] + n - 1
    v_mat = i_idx[:, None] + k_idx[None, :] - n + 1
    valid_a = (u_mat >= 0) & (u_mat < n)
    valid_b = (v_mat >= 0) & (v_mat < n)
    u_clip = np.clip(u_mat, 0, n - 1)
    v_clip = np.clip(v_mat, 0, n - 1)
    grad = np.sum(valid_a * (1.0 - h[u_clip]) * w[None, :], axis=1) - \
           np.sum(valid_b * h[v_clip] * w[None, :] , axis=1)
    return grad

def smooth_max_obj_grad(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m))
    w = exp_vals / np.sum(exp_vals)
    obj = m + np.log(np.sum(exp_vals)) / alpha
    n = len(h)
    n2 = 2 * n - 1
    i_idx = np.arange(n)
    k_idx = np.arange(n2)
    u_mat = i_idx[:, None] - k_idx[None, :] + n - 1
    v_mat = i_idx[:, None] + k_idx[None, :] - n + 1
    valid_a = (u_mat >= 0) & (u_mat < n)
    valid_b = (v_mat >= 0) & (v_mat < n)
    u_clip = np.clip(u_mat, 0, n - 1)
    v_clip = np.clip(v_mat, 0, n - 1)
    grad = np.sum(valid_a * (1.0 - h[u_clip]) * w[None, :], axis=1) - \
           np.sum(valid_b * h[v_clip] * w[None, :], axis=1)
    return obj, grad

def binary_pair_swap(h, target_sum, max_rounds=100):
    """Exhaustive pair-swap local search on binary h."""
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h)
    for _ in range(max_rounds):
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_c5_local = best_c5
        best_pair = None
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                c5 = compute_c5(h)
                h[i] = 1.0
                h[j] = 0.0
                if c5 < best_c5_local - 1e-15:
                    best_c5_local = c5
                    best_pair = (i, j)
        if best_pair is None:
            break
        h[best_pair[0]] = 0.0
        h[best_pair[1]] = 1.0
        best_c5 = best_c5_local
    return h

def binary_sa(h, target_sum, rng, t_start, time_budget):
    """Simulated annealing on binary h."""
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    T_start = 0.01
    T_end = 0.001
    n_iter = 20000
    for step in range(n_iter):
        if time.time() - t_start > time_budget:
            break
        T = T_start * (T_end / T_start) ** (step / n_iter)
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c5

def fourier_to_h(coeffs, n):
    """Convert Fourier coefficients to h values on grid."""
    x = np.linspace(0, 2.0, n, endpoint=False)
    M = len(coeffs) // 2
    h = np.full(n, 0.5)
    for m in range(1, M + 1):
        a = coeffs[2 * (m - 1)]
        b = coeffs[2 * (m - 1) + 1]
        h += a * np.cos(np.pi * m * x) + b * np.sin(np.pi * m * x)
    return h

def h_to_fourier(h, M):
    """Extract Fourier coefficients from h."""
    n = len(h)
    x = np.linspace(0, 2.0, n, endpoint=False)
    coeffs = np.zeros(2 * M)
    for m in range(1, M + 1):
        coeffs[2 * (m - 1)] = 2.0 / n * np.sum(h * np.cos(np.pi * m * x))
        coeffs[2 * (m - 1) + 1] = 2.0 / n * np.sum(h * np.sin(np.pi * m * x))
    return coeffs

def fourier_optimize(h0, n, M, alpha, budget_s, t_start):
    """Optimize Fourier coefficients."""
    coeffs0 = h_to_fourier(h0, M)
    target_sum = n / 2.0
    
    def obj_func(coeffs):
        h = fourier_to_h(coeffs, n)
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
        return smooth_max_obj(h, alpha)
    
    def obj_grad_func(coeffs):
        h = fourier_to_h(coeffs, n)
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
        _, grad_h = smooth_max_obj_grad(h, alpha)
        # Chain rule: d(obj)/d(coeffs[m]) = sum_i grad_h[i] * d(h[i])/d(coeffs[m])
        x = np.linspace(0, 2.0, n, endpoint=False)
        grad_coeffs = np.zeros(2 * M)
        for m in range(1, M + 1):
            grad_coeffs[2 * (m - 1)] = np.sum(grad_h * np.cos(np.pi * m * x))
            grad_coeffs[2 * (m - 1) + 1] = np.sum(grad_h * np.sin(np.pi * m * x))
        return grad_coeffs
    
    try:
        res = minimize(obj_func, coeffs0, jac=obj_grad_func,
                      method='L-BFGS-B',
                      options={'maxiter': 500, 'ftol': 1e-16, 'gtol': 1e-10})
        h_new = fourier_to_h(res.x, n)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
    except Exception:
        h_new = h0.copy()
    return h_new

def lbsqp_smooth(h0, n, target_sum, alpha, maxiter, mu):
    """L-BFGS-B smooth optimization with penalty for sum constraint."""
    def penalty_obj(h):
        obj, grad = smooth_max_obj_grad(h, alpha)
        s = np.sum(h) - target_sum
        obj += mu * s * s
        grad += 2.0 * mu * s * np.ones(n)
        return obj, grad
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(penalty_obj, h0.copy(), jac=True,
                      method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
        h_opt = project_to_sum(res.x, target_sum)
    except Exception:
        h_opt = h0.copy()
    return h_opt

def make_structured_starts(n, target_sum, rng, M_fourier=20):

        h = np.zeros(n)

        starts.append(project_to_sum(fix_binary_sum(h, target_sum), target_sum))

        h = fourier_to_h(coeffs, n)
        starts.append(project_to_sum(h, target_sum))
    
    return starts
