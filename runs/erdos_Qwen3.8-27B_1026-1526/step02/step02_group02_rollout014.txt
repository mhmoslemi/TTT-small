import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(c.max())

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
            if not mask.any():
                break
            room = 1.0 - h[mask]
            max_per = room.max()
            step = min(diff / mask.sum(), max_per * 0.9999)
            h[mask] += step
        else:
            mask = h > 0.0
            if not mask.any():
                break
            room = h[mask]
            max_per = room.max()
            step = min(-diff / mask.sum(), max_per * 0.9999)
            h[mask] -= step
        np.clip(h, 0.0, 1.0, out=h)
    return h

def smooth_grad_obj(h, T):
    N = len(h)
    dx = 2.0 / N
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    c_max = c.max()
    logits = T * (c - c_max)
    logits = np.clip(logits, -300, 300)
    exp_l = np.exp(logits)
    total = exp_l.sum()
    w = exp_l / total
    obj = c_max + np.log(total) / T
    
    grad = np.zeros(N)
    for m in range(2 * N - 1):
        wm = w[m]
        if wm < 1e-10:
            continue
        offset = m - (N - 1)
        if offset >= 0:
            if N - offset > 0:
                grad[:N-offset] += wm * (1.0 - h[offset:N])
                grad[offset:] += -wm * h[:N-offset]
        else:
            la = -offset
            grad[la:] += wm * (1.0 - h[:N-la])
            if N - la > 0:
                grad[:N-la] += -wm * h[la:N]
    grad *= dx
    return obj, grad

def projected_gradient_nesterov(h0, n, T_start, T_end, n_iters, momentum=0.9):
    h = h0.copy()
    v = np.zeros_like(h)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    
    for it in range(n_iters):
        frac = it / max(1, n_iters - 1)
        T = T_start + (T_end - T_start) * frac
        lr = 0.05 / (1.0 + 0.1 * it)
        
        _, grad = smooth_grad_obj(h, T)
        
        # Nesterov momentum
        v = momentum * v + lr * grad
        h_new = h - v
        h_new = project_feasible(h_new, n)
        
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        
        h = h_new
    
    return best_h, best_c5

def slsqp_obj(x):
    return x[-1]

def slsqp_ineq(x):
    h = x[:-1]
    t = x[-1]
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return t - c * dx

def slsqp_eq(x):
    h = x[:-1]
    return np.sum(h) - len(h) / 2.0

def slsqp_eq_jac(x):
    n = len(x) - 1
    jac = np.zeros(len(x))
    jac[:n] = 1.0
    return jac

def make_slsqp_constraints():
    return [
        {'type': 'ineq', 'fun': slsqp_ineq},
        {'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac}
    ]

def slsqp_optimize(h_init, n, maxiter=3000, t_margin=0.05):
    c5_init = compute_c5(h_init)
    x0 = np.concatenate([h_init, [c5_init + t_margin]])
    bounds = [(0.0, 1.0)] * n + [(max(0.001, c5_init * 0.3), 2.0)]
    constraints = make_slsqp_constraints()
    result = minimize(slsqp_obj, x0, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
    if result.x is not None:
        h_out = project_feasible(result.x[:-1], n)
        return h_out, compute_c5(h_out)
    return h_init, c5_init

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(c.max())


def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
            if not mask.any():
                break
            room = 1.0 - h[mask]
            step = min(diff / mask.sum(), room.max() * 0.9999)
            if step <= 0:
                break
            h[mask] += step
        else:
            mask = h > 0.0
            if not mask.any():
                break
            room = h[mask]
            step = min(-diff / mask.sum(), room.max() * 0.9999)
            if step <= 0:
                break
            h[mask] -= step
        np.clip(h, 0.0, 1.0, out=h)
    return h


def smooth_grad_obj(h, T):
    N = len(h)
    dx = 2.0 / N
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    c_max = c.max()
    logits = T * (c - c_max)
    logits = np.clip(logits, -300, 300)
    exp_l = np.exp(logits)
    total = exp_l.sum()
    w = exp_l / total
    obj = c_max + np.log(total) / T

    grad = np.zeros(N)
    for m in range(2 * N - 1):
        wm = w[m]
        if wm < 1e-10:
            continue
        offset = m - (N - 1)
        if offset >= 0:
            if N - offset > 0:
                grad[:N - offset] += wm * (1.0 - h[offset:N])
                grad[offset:] += -wm * h[:N - offset]
        else:
            la = -offset
            grad[la:] += wm * (1.0 - h[:N - la])
            if N - la > 0:
                grad[:N - la] += -wm * h[la:N]
    grad *= dx
    return obj, grad


def pg_nesterov(h0, n, T_start, T_end, n_iters, momentum=0.92, lr_scale=0.04):
    h = h0.copy()
    v = np.zeros_like(h)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    for it in range(n_iters):
        frac = it / max(1, n_iters - 1)
        T = T_start + (T_end - T_start) * frac
        lr = lr_scale / (1.0 + 0.08 * it / n_iters)
        _, grad = smooth_grad_obj(h, T)
        v = momentum * v + lr * grad
        h_new = project_feasible(h - v, n)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5


def pg_adam(h0, n, T_start, T_end, n_iters, beta1=0.9, beta2=0.999, lr_base=0.015):
    h = h0.copy()
    m = np.zeros_like(h)
    v_hat = np.zeros_like(h)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    for it in range(n_iters):
        frac = it / max(1, n_iters - 1)
        T = T_start + (T_end - T_start) * frac
        lr = lr_base / (1.0 + 0.1 * it / n_iters)
        _, grad = smooth_grad_obj(h, T)
        m = beta1 * m + (1.0 - beta1) * grad
        v_hat = beta2 * v_hat + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat_c = v_hat / (1.0 - beta2 ** (it + 1))
        h_new = project_feasible(h - lr * m_hat / (np.sqrt(v_hat_c) + 1e-8), n)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5


def slsqp_obj(x):
    return x[-1]


def slsqp_ineq(x):
    h = x[:-1]
    t = x[-1]
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return t - c * dx


def slsqp_eq(x):
    h = x[:-1]
    return np.sum(h) - len(h) / 2.0


def slsqp_eq_jac(x):
    n = len(x) - 1
    jac = np.zeros(len(x))
    jac[:n] = 1.0
    return jac


def make_slsqp_constraints():
    return [
        {'type': 'ineq', 'fun': slsqp_ineq},
        {'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac}
    ]


def slsqp_optimize(h_init, n, maxiter=3000, t_margin=0.05):
    c5_init = compute_c5(h_init)
    x0 = np.concatenate([h_init, [c5_init + t_margin]])
    bounds = [(0.0, 1.0)] * n + [(max(0.001, c5_init * 0.3), 2.0)]
    constraints = make_slsqp_constraints()
    result = minimize(slsqp_obj, x0, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
    if result.x is not None:
        h_out = project_feasible(result.x[:-1], n)
        return h_out, compute_c5(h_out)
    return h_init, c5_init
