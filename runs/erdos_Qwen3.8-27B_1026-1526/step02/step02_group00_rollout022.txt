def compute_c5(h):

    dx = 2.0 / n

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):

        if abs(diff) < 1e-14:
            break

def projected_gradient_descent(h0, n, target_sum, alpha, lr, maxiter, tol=1e-12):
    """Custom projected gradient method with backtracking line search."""
    h = h0.copy()
    best_obj = corr_and_grad(h, alpha)[0]
    best_h = h.copy()
    
    for it in range(maxiter):

            # Still track best
            pass
        if obj < best_obj:

            best_h = h.copy()

        # The projection of -grad onto {v: sum(v) = 0} is -grad + mean(grad)
        grad_proj = -grad + np.mean(grad)

        for _ in range(30):

                break
            step *= 0.5

    return best_h, best_obj

def slsqp_local(h0, n, target_sum, alpha, maxiter):
    def f(h):
        return corr_and_grad(h, alpha)[0]
    def g(h):
        return corr_and_grad(h, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

1. Find the active set A = {k: corr[k] >= max(corr) - epsilon}
2. Compute average gradient over active set
3. Take a step in that direction
4. Project

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m_val))
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    term = valid_f * (1.0 - h[fc]) - valid_s * h[sc]
    grad = term @ w
    return obj, grad

def subgradient_step(h, step_size, epsilon=1e-4):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    active = corr >= m_val - epsilon * m_val
    w = active.astype(float) / np.sum(active)
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    term = valid_f * (1.0 - h[fc]) - valid_s * h[sc]
    grad = term @ w
    grad_proj = -grad + np.mean(grad)
    return h + step_size * grad_proj

def slsqp_optimize(h0, n, target_sum, alpha, maxiter):
    def f(h):
        return smooth_max_and_grad(h, alpha)[0]
    def g(h):
        return smooth_max_and_grad(h, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def parametric_sinusoidal(n, freq, amp, phase, target_sum):
    x = np.linspace(0, 2.0, n, endpoint=False)
    h = 0.5 + amp * np.sin(2.0 * np.pi * freq * x / 2.0 + phase)
    return project_to_sum(h, target_sum)

def block_pattern(n, period, on_len, target_sum):
    h = np.zeros(n)
    total = period + on_len
    for i in range(n):
        if (i % total) < on_len:
            h[i] = 1.0
    return project_to_sum(h, target_sum)

def interpolated_resample(h, n_old, n_new):
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h, period=2.0)

def make_diverse_starts(n, target_sum, rng, best_h=None, best_n=0):
    starts = []
    x = np.linspace(0, 2.0, n, endpoint=False)
    starts.append(np.full(n, 0.5))
    for freq in [1, 2, 3, 4, 5, 6, 7, 8, 10]:
        for amp in [0.35, 0.42, 0.47, 0.49]:
            for phase in [0.0, np.pi / 3.0, np.pi / 2.0, 2.0 * np.pi / 3.0]:
                s = 0.5 + amp * np.sin(2.0 * np.pi * freq * x / 2.0 + phase)
                starts.append(project_to_sum(s, target_sum))
    for period in [3, 4, 5, 6, 7, 8, 10, 12, 15]:
        for ratio in [0.4, 0.45, 0.5, 0.55, 0.6]:
            on_len = int(round(period * ratio / 2.0))
            if on_len > 0:
                starts.append(block_pattern(n, period, on_len, target_sum))
    for _ in range(6):
        f1 = rng.randint(1, 10)
        f2 = rng.randint(2, 15)
        a1 = rng.uniform(0.25, 0.45)
        a2 = rng.uniform(0.05, 0.25)
        p1 = rng.uniform(0, 2 * np.pi)
        p2 = rng.uniform(0, 2 * np.pi)
        s = 0.5 + a1 * np.sin(2 * np.pi * f1 * x / 2.0 + p1) + a2 * np.sin(2 * np.pi * f2 * x / 2.0 + p2)
        starts.append(project_to_sum(s, target_sum))
    starts.append(project_to_sum(rng.uniform(0, 1, n), target_sum))
    hb = (rng.rand(n) > 0.5).astype(float)
    attempts = 0
    while abs(hb.sum() - target_sum) > 1e-10 and attempts < 20000:
        if hb.sum() > target_sum:
            idx = rng.choice(np.where(hb > 0)[0])
            hb[idx] = 0.0
        else:
            idx = rng.choice(np.where(hb < 1)[0])
            hb[idx] = 1.0
        attempts += 1
    if abs(hb.sum() - target_sum) < 1e-10:
        starts.append(hb.copy())
    if best_h is not None and best_n > 0:
        if best_n == n:
            starts.append(best_h.copy())
        else:
            starts.append(project_to_sum(interpolated_resample(best_h, best_n, n), target_sum))
    return starts

def progressive_alpha_optimize(h0, n, target_sum, alphas, iters_per_alpha):
    h = h0.copy()
    for alpha in alphas:
        h = slsqp_optimize(h, n, target_sum, alpha, maxiter=iters_per_alpha)
    return h

def hybrid_smooth_subgradient(h0, n, target_sum, n_iter, alpha_start, alpha_end):
    h = h0.copy()
    ts = target_sum
    for i in range(n_iter):
        frac = i / max(1, n_iter - 1)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        if i % 10 == 0:
            h = slsqp_optimize(h, n, ts, alpha, maxiter=50)
        else:
            ss = 0.5 / (alpha + 1.0)
            h_new = subgradient_step(h, ss)
            h_new = project_to_sum(h_new, ts)
            if compute_c5(h_new) < compute_c5(h):
                h = h_new
    return h

def coordinate_descent_polish(h0, n, target_sum, max_rounds, rng):
    h = h0.copy()
    ts = target_sum
    cur_c5 = compute_c5(h)
    for rnd in range(max_rounds):
        improved = False
        order = rng.permutation(n)
        for i in order:
            j = (i + 1) % n
            old_hi = h[i]
            old_hj = h[j]
            for trial_hi in [old_hi + 0.1, old_hi - 0.1, old_hi + 0.05, old_hi - 0.05,
                            old_hi + 0.2, old_hi - 0.2, 0.0, 1.0, 0.5]:
                trial_hi = max(0.0, min(1.0, trial_hi))
                trial_hj = old_hi + old_hj - trial_hi
                if trial_hj < -0.01 or trial_hj > 1.01:
                    continue
                trial_hj = max(0.0, min(1.0, trial_hj))
                if abs(trial_hi + trial_hj - old_hi - old_hj) > 1e-8:
                    continue
                h_try = h.copy()
                h_try[i] = trial_hi
                h_try[j] = trial_hj
                if compute_c5(h_try) < cur_c5 - 1e-12:
                    h = h_try.copy()
                    cur_c5 = compute_c5(h)
                    improved = True
                    break
        if not improved:
            break
    return h, cur_c5

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()

    def time_left():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)
    best_h = None

    if 'initial_h_values' in globals() and initial_h_values is not None:
