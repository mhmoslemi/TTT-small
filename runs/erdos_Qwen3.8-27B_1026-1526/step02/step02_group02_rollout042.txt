```python
def compute_subgradient(h, n, top_frac=0.1):
    dx = 2.0 / n
    G = 1.0 - h
    c = np.correlate(h, G, mode='full') * dx
    T = c.max()
    # Active set: shifts where c is within tolerance of max
    tol = 0.01 * T  # or some fixed tolerance
    active = c >= T - tol
    n_active = active.sum()
    
    # For each i, compute sum over active shifts s of:
    # (1-h[i+s]) * I(i+s valid) - h[i-s] * I(i-s valid)
    # where s ranges over -(n-1) to n-1
    # In the full correlation array, index i_full = s + (n-1)
    
    grad = np.zeros(n)
    for i in range(n):
        s_vals = np.arange(-(n-1), n)
        # Filter to active shifts
        active_mask = active[np.arange(2*n-1)]  # already boolean
        # For shift s: contribution is (1-h[i+s])*I(0<=i+s<n) - h[i-s]*I(0<=i-s<n)
        # i+s is in [0,n) iff -i <= s < n-i
        # i-s is in [0,n) iff i-n < s <= i
        valid_plus = (s_vals >= -i) & (s_vals < n - i)
        valid_minus = (s_vals > i - n) & (s_vals <= i)
        
        idx_plus = i + s_vals  # index into h for h[i+s]
        idx_minus = i - s_vals  # index into h for h[i-s]
        
        contrib = np.zeros(2*n-1)
        contrib[valid_plus] = 1.0 - h[idx_plus[valid_plus]]
        contrib[valid_minus] -= h[idx_minus[valid_minus]]
        
        grad[i] = np.sum(contrib[active_mask]) / n_active * dx
    
    return grad, T, active
```

```python
import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            room = 1.0 - h[mask]
            pos_room = room[room > 1e-12]
            if len(pos_room) == 0:
                break
            step = min(diff / np.sum(mask), pos_room.min() * 0.5)
            h[mask] += step
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            room = h[mask]
            pos_room = room[room > 1e-12]
            if len(pos_room) == 0:
                break
            step = min(-diff / np.sum(mask), pos_room.min() * 0.5)
            h[mask] -= step
    return np.clip(h, 0.0, 1.0)

def c5_logsumexp(h, T):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    c_max = c.max()
    if c_max < -100:
        return float(c_max)
    shifted = T * (c - c_max)
    shifted = np.clip(shifted, -500, 500)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / T)

    c5_val = c5_logsumexp(x, T)
    sum_violation = np.sum(x) - n / 2.0
    return c5_val + penalty * sum_violation ** 2

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    T = c.max()

    return x[-1]

def slsqp_ineq(x):

    t = x[-1]
    n = len(h)

    c = np.correlate(h, 1.0 - h, mode="full")

def slsqp_optimize(h_init, n, maxiter=3000, t_margin=0.1):
```

```python
import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time


def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < (1.0 - 1e-15)
            if not np.any(mask):
                break
            room = 1.0 - h[mask]
            step = min(diff / np.sum(mask), room.min() * 0.999)
            h[mask] += step
        else:
            mask = h > 1e-15
            if not np.any(mask):
                break
            step = min(-diff / np.sum(mask), h[mask].min() * 0.999)
            h[mask] -= step
    return np.clip(h, 0.0, 1.0)


def c5_logsumexp(h, T):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    c_max = c.max()
    if c_max < -100:
        return float(c_max)
    shifted = T * (c - c_max)
    shifted = np.clip(shifted, -500, 500)
    return c_max + np.log(np.sum(np.exp(shifted))) / T


def lbf_obj(x, T, penalty):
    n = len(x)
    val = c5_logsumexp(x, T)
    sv = np.sum(x) - n / 2.0
    return val + penalty * sv * sv


def slsqp_obj(x):
    return x[-1]


def slsqp_ineq(x):
    h = x[:-1]
    t = x[-1]
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return t - c * dx


def slsqp_eq(x):
    return np.sum(x[:-1]) - len(x[:-1]) / 2.0


def slsqp_eq_jac(x):
    thetas = np.zeros(len(x))
    thetas[:-1] = 1.0
    return thetas


def slsqp_optimize(h_init, n, maxiter=3000, t_margin=0.1):
    c5_init = compute_c5_from_h(h_init)
    x0 = np.concatenate([h_init, [c5_init + t_margin]])
    bounds = [(0.0, 1.0)] * n + [(max(0.001, c5_init * 0.5), 2.0)]
    constraints = [
        {'type': 'ineq', 'fun': slsqp_ineq},
        {'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac}
    ]
    result = minimize(slsqp_obj, x0, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
    if result.x is not None:
        h_out = project_feasible(result.x[:-1], n)
        return h_out, compute_c5_from_h(h_out)

    if result.x is not None:
        h_out = project_feasible(result.x, n)
        return h_out, compute_c5_from_h(h_out)
    return h_init, compute_c5_from_h(h_init)


def fw_gradient(h, n, T):
    """Compute gradient of logsumexp objective."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    c_max = c.max()
    exp_c = np.clip(T * (c - c_max), -500, 500)
    weights = np.exp(exp_c)
    weights /= weights.sum()
    
    s_all = np.arange(-(n-1), n)
    grad = np.zeros(n)
    for i in range(n):
        vp = (i + s_all >= 0) & (i + s_all < n)
        vm = (i - s_all >= 0) & (i - s_all < n)
        grad[i] = dx * (np.sum(weights[vp] * (1.0 - h[i + s_all[vp]])) -
                         np.sum(weights[vm] * h[i - s_all[vm]]))
    return grad


def frank_wolfe(h_init, n, T, n_iters, time_limit):
    """Frank-Wolfe method with step size schedule."""
    t0 = time.time()
    h = project_feasible(h_init, n).copy()
    best_h = h.copy()
    best_c5 = compute_c5_from_h(best_h)
    
    for t in range(n_iters):
        if time.time() - t0 > time_limit:
            break
        grad = fw_gradient(h, n, T)
        # Find FW vertex
        sorted_idx = np.argsort(grad)
        target = n / 2.0
        v = np.zeros(n)
        n_full = int(target)
        v[sorted_idx[:n_full]] = 1.0
        frac = target - n_full
        if frac > 1e-10:
            v[sorted_idx[n_full]] = frac
        # Step size
        gamma = 2.0 / (t + 2)
        h_new = (1 - gamma) * h + gamma * v
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5_from_h(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        h = h_new
    return best_h, best_c5
```
