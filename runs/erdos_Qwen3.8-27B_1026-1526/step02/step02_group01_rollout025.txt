= h[j] * [(1-h[j-k]) * 1{k≤j} - h[j+k] * 1{j≤n-1-k}] + h[j+k] * 1{j≤n-1-k}

So for each lag k, c_k is linear in h[j] (when all other h's are fixed)! That means the max of linear functions is convex in h[j].

So for a fixed j and fixed other coordinates, the problem min_{h[j]} max_k c_k is a 1D convex optimization problem! We can solve it efficiently.

Specifically, for each lag k, c_k as a function of h[j] is:

The a_k and b_k can be precomputed for each j. Then we need to find t ∈ [0,1] that minimizes max_k (a_k * t + b_k).

1. Computing all (a_k, b_k) pairs
2. The function max_k(a_k*t + b_k) is a convex piecewise-linear function of t
3. Find the minimum over t ∈ [0,1]

The minimum of a convex piecewise-linear function occurs where the subgradient contains 0, or at a boundary. We can find it by:
- Evaluating at t=0 and t=1
- Finding the "crossover" points where the active lag changes
- Or just doing a ternary search / bisection on the subgradient

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / len(exp_vals)) / alpha
    
    g = np.zeros(n)
    for j in range(n):
        if j >= 0:
            ks = np.arange(0, j + 1)
            g[j] += np.dot(w[n - 1 + ks], 1.0 - h[j - ks])
        if j <= n - 1:
            ks2 = np.arange(0, n - j)
            g[j] -= np.dot(w[n - 1 + ks2], h[j + ks2])
        if n - 1 - j >= 1:
            ls = np.arange(1, n - j)
            g[j] += np.dot(w[n - 1 - ls], 1.0 - h[j + ls])
        if j >= 1:
            ls2 = np.arange(j, n)
            g[j] -= np.dot(w[n - 1 - ls2], h[j - ls2])
    return sm_val, g

def slsqp_optimize(h0, n, target_sum, alpha, maxiter):
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_and_grad(h, alpha)
        return v
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_and_grad(h, alpha)
        return g
    def con_fun(x):
        return np.sum(x) - target_sum
    def con_jac(x):
        return np.ones(n)
    
    constraints = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-13})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

    h_new = np.interp(x_new, x_old, h_old, period=2.0)

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / len(exp_vals)) / alpha
    g = np.zeros(n)
    for j in range(n):
        ks = np.arange(0, j + 1)
        g[j] += np.dot(w[n - 1 + ks], 1.0 - h[j - ks])
        ks2 = np.arange(0, n - j)
        g[j] -= np.dot(w[n - 1 + ks2], h[j + ks2])
        if n - 1 - j >= 1:
            ls = np.arange(1, n - j)
            g[j] += np.dot(w[n - 1 - ls], 1.0 - h[j + ls])
        if j >= 1:
            ls2 = np.arange(j, n)
            g[j] -= np.dot(w[n - 1 - ls2], h[j - ls2])
    return sm_val, g

def slsqp_optimize(h0, n, target_sum, alpha, maxiter):
    def obj_fn(x):
        h = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_and_grad(h, alpha)
        return v
    def grad_fn(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_and_grad(h, alpha)
        return g
    def con_fun(x):
        return np.sum(x) - target_sum
    def con_jac(x):
        return np.ones(n)
    constraints = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_fn, h0.copy(), jac=grad_fn, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def compute_ab(h, j, n):
    a = np.zeros(2 * n - 1)
    b = np.zeros(2 * n - 1)
    for idx in range(2 * n - 1):
        s = idx - (n - 1)
        if s >= 0:
            if s <= j and j <= n - 1:
                a[idx] += (1.0 - h[j - s])
            if j <= n - 1 - s:
                a[idx] -= h[j + s]
                b[idx] += h[j + s]
        else:
            m = -s
            if j <= n - 1 - m:
                a[idx] += (1.0 - h[j + m])
            if m <= j and j <= n - 1:
                a[idx] -= h[j - m]
                b[idx] += h[j - m]
    return a, b

def coord_1d_descent(h, n, target_sum, max_rounds, rng, t_start, time_limit):
    h = h.copy()
    c5_cur = compute_c5(h)
    dx = 2.0 / n
    grid = np.linspace(0.0, 1.0, 300)
    for rnd in range(max_rounds):
        if time.time() - t_start > time_limit:
            break
        improved = False
        idx_order = rng.permutation(n)
        for j in idx_order:
            a, b = compute_ab(h, j, n)
            vals = np.max(a[None, :] * grid[:, None] + b[None, :], axis=1)
            t_opt = grid[np.argmin(vals)]
            h[j] = t_opt
            h = project_to_sum(h, target_sum)
            c5_new = compute_c5(h)
            if c5_new < c5_cur - 1e-15:
                c5_cur = c5_new
                improved = True
        if not improved:
            break
    return h, c5_cur

def interpolate_h(h_old, n_old, n_new):
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return h_new
