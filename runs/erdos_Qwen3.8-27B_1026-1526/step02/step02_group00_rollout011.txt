```python
def coordinate_descent(h, n_steps=100, rng=None, time_limit=None, t_start=None):
    n = len(h)
    target_sum = n / 2.0
    h = h.copy()
    dx = 2.0 / n
    
    # Precompute lag indices
    n_corr = 2 * n - 1
    lags = np.arange(0, n_corr) - (n - 1)  # lags from -(n-1) to n-1
    
    for step in range(n_steps):
        # Time check
        if time_limit is not None:
            if time.time() - t_start > time_limit:
                break
        
        improved = False
        if rng is not None:
            order = rng.permutation(n)
        else:
            order = np.arange(n)
        
        for i in order:
            old_h_i = h[i]
            corr = np.correlate(h, 1.0 - h, mode="full")
            
            # Compute A[k] for this coordinate
            # A[k] for lag ℓ_k: (1-h[i+ℓ_k])*[i+ℓ_k in 0..n-1] - h[i-ℓ_k]*[i-ℓ_k in 0..n-1]
            A = np.zeros(n_corr)
            valid_p = (i + lags >= 0) & (i + lags < n)
            valid_m = (i - lags >= 0) & (i - lags < n)
            A[valid_p] += 1.0 - h[i + lags[valid_p]]
            A[valid_m] -= h[i - lags[valid_m]]
            
            B = corr - A * old_h_i
            
            # Minimize max(B + A*v) over v in [0,1]
            # Use ternary search
            def obj_1d(v):
                return np.max(B + A * v)
            
            lo, hi = 0.0, 1.0
            for _ in range(50):  # ~16 decimal digits of precision
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                if obj_1d(m1) < obj_1d(m2):
                    hi = m2
                else:
                    lo = m1
            v_new = (lo + hi) / 2.0
            
            h[i] = v_new
            improved = abs(v_new - old_h_i) > 1e-10
        
        # Project to sum constraint
        h = project_to_sum(h, target_sum)
        
        if not improved:
            break
    
    return h
```

```python
def lpsmooth_grad(h, p):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    n_corr = 2 * n - 1
    sp = np.sum(corr**p)
    corr_safe = np.maximum(corr, 1e-30)
    w = p * (corr_safe**(p-1)) * (sp**(1.0/p - 1.0)) * dx
    
    # offsets[i] for correlation index k: lag = k - (n-1)
    # For coordinate i:
    #   first term: h[i+lag] valid when 0 <= i+lag <= n-1
    #   second term: h[i-lag] valid when 0 <= i-lag <= n-1
    # lag = k - (n-1), k = 0..2n-2
    
    # Vectorized: create n x (2n-1) index matrices
    i_idx = np.arange(n)[:, None]  # n x 1
    k_idx = np.arange(n_corr)[None, :]  # 1 x (2n-1)
    lags = k_idx - (n - 1)  # 1 x (2n-1)
    
    valid1 = (i_idx + lags >= 0) & (i_idx + lags < n)  # n x (2n-1)
    valid2 = (i_idx - lags >= 0) & (i_idx - lags < n)  # n x (2n-1)
    
    # For valid1: h[i + lag] = h[i_idx[k] + lags[k]]
    idx1 = i_idx + lags  # n x (2n-1)
    idx1_clipped = np.clip(idx1, 0, n-1)
    
    # For valid2: h[i - lag] = h[i_idx[k] - lags[k]]
    idx2 = i_idx - lags
    idx2_clipped = np.clip(idx2, 0, n-1)
    
    # first[i,k] = w[k] * (1-h[i+lag]) * valid1[i,k]
    # second[i,k] = w[k] * h[i-lag] * valid2[i,k]
    
    term1 = w[None, :] * (1.0 - h[idx1_clipped]) * valid1
    term2 = w[None, :] * h[idx2_clipped] * valid2
    
    grad = np.sum(term1, axis=1) - np.sum(term2, axis=1)
    return grad
```

For coordinate i, given all other h fixed:
- corr[k] = B[k] + A[k] * v (linear in v)
- I want to minimize max_k (B[k] + A[k] * v) over v ∈ [0,1]

The optimal v is where the maximum of the lines is minimized. This can be found by:
- Evaluating max_k (B[k] + A[k]*v) at various v
- The function is convex, so ternary search works

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def lpsmooth(x, p):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    return float(np.sum(corr ** p) ** (1.0 / p) * dx)

def lpsmooth_grad(x, p):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    n_corr = 2 * n - 1
    sp = float(np.sum(corr ** p))
    if sp < 1e-30:
        return np.zeros(n)
    corr_safe = np.maximum(corr, 1e-25)
    w = p * (corr_safe ** (p - 1)) * (sp ** (1.0 / p - 1.0)) * dx
    i_idx = np.arange(n)[:, None]
    k_idx = np.arange(n_corr)[None, :]
    lags = k_idx - (n - 1)
    valid1 = (i_idx + lags >= 0) & (i_idx + lags < n)
    valid2 = (i_idx - lags >= 0) & (i_idx - lags < n)
    idx1 = np.clip(i_idx + lags, 0, n - 1)
    idx2 = np.clip(i_idx - lags, 0, n - 1)
    term1 = w[None, :] * (1.0 - x[idx1]) * valid1
    term2 = w[None, :] * x[idx2] * valid2
    return np.sum(term1, axis=1) - np.sum(term2, axis=1)

            corr = np.correlate(h, 1.0 - h, mode="full")
            n_corr = 2 * n - 1

            A = np.zeros(n_corr)
            valid_p = (i + lags >= 0) & (i + lags < n)
            valid_m = (i - lags >= 0) & (i - lags < n)
            A[valid_p] += 1.0 - h[i + lags[valid_p]]
            A[valid_m] -= h[i - lags[valid_m]]
            B = corr - A * old_val
            lo, hi = 0.0, 1.0

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    assert abs(best_h.sum() - best_n / 2.0) < 1e-6, f"Sum constraint violated: {best_h.sum()} vs {best_n/2.0}"
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
```
