State: binary array h of length n with exactly n/2 ones
Objective: minimize max(corr) where corr = np.correlate(h, 1-h, mode="full")
Move: pick a random 1 and a random 0, swap them (i.e., set the 1 to 0 and the 0 to 1)
This maintains the sum constraint automatically.

At each step:
- Compute new max(corr)
- If better, accept
- If worse, accept with probability exp(-Δ/T)

Temperature schedule: decrease over time.

def binary_sa(h0, n, target_sum, rng, max_steps=50000, T0=10.0, T1=0.1):
    """Simulated annealing on binary space."""
    h = (h0 > 0.5).astype(float)
    # Fix sum
    for _ in range(10000):
        if np.sum(h) > target_sum:
            ones = np.where(h > 0.5)[0]
            if len(ones) == 0:
                break
            h[rng.choice(ones)] = 0.0
        elif np.sum(h) < target_sum:
            zeros = np.where(h < 0.5)[0]
            if len(zeros) == 0:
                break
            h[rng.choice(zeros)] = 1.0

            break
    
    c5_cur = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_cur
    
    for step in range(max_steps):

        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        c5_new = compute_c5(h)
        delta = c5_new - c5_cur

            c5_cur = c5_new
            if c5_new < best_c5:

                best_h = h.copy()
        else:

            h[j] = 0.0
