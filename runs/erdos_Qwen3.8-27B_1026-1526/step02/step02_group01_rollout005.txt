    threshold = 1e-8
    active = np.where(w > threshold)[0]
    
    for k in active:
        wk = w[k]
        # corr[k] = sum_i h[i] * (1-h[i+k-n+1])
        i_lo = max(0, n - 1 - k)

        v, _ = eval_and_grad(h, alpha)
        return v
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = eval_and_grad(h, alpha)
        # Add constraint gradient (sum constraint)
        # Actually let's handle constraint separately
        return g
    def constraint_fun(x):
        return np.sum(x) - target_sum
    def constraint_jac(x):
        return np.ones(len(x))

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def eval_and_grad(h, alpha):
    n = len(h)
    n_corr = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    mval = np.max(corr)
    shifted = np.clip(alpha * (corr - mval), -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    val = mval + np.log(s / n_corr) / alpha
    g = np.zeros(n)
    active = np.where(w > 1e-10)[0]
    for k in active:
        wk = w[k]
        i_lo = max(0, n - 1 - k)
        i_hi = min(n - 1, 2 * n - 2 - k)
        idx = np.arange(i_lo, i_hi + 1)
        a = idx + k - n + 1
        mask_a = (a >= 0) & (a < n)
        if np.any(mask_a):
            g[idx[mask_a]] += wk * (1.0 - h[a[mask_a]])
        j2 = idx + k - n + 1
        mask_b = (j2 >= 0) & (j2 < n)
        if np.any(mask_b):
            g[j2[mask_b]] += wk * (-h[idx[mask_b]])
    return val, g

def slsqp_optimize(h0, target_sum, alpha, maxiter):
    n = len(h0)
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        v, _ = eval_and_grad(h, alpha)
        return v
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = eval_and_grad(h, alpha)
        return g
    def cfun(x):
        return np.sum(x) - target_sum
    def cjac(x):
        return np.ones(len(x))
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def lbfgs_optimize(h0, target_sum, alpha, maxiter):
    n = len(h0)
    penalty = 500.0
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        v, _ = eval_and_grad(h, alpha)
        v += penalty * (np.sum(h) - target_sum) ** 2
        return v
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = eval_and_grad(h, alpha)
        g += 2.0 * penalty * (np.sum(h) - target_sum)
        return g
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10,
                                'maxls': 30, 'maxfun': maxiter * 5})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def nesterov_gradient(h0, n, target_sum, alpha, n_iters, lr_init):
    h = h0.copy()
    y = h.copy()
    t = 1.0
    lr = lr_init
    c5_cur = compute_c5(h)
    h_best = h.copy()
    for it in range(n_iters):
        _, g = eval_and_grad(y, alpha)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        mo = (t - 1.0) / t_new
        h_new = np.clip(y - lr * g, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        y = h_new + mo * (h_new - h)
        y = np.clip(y, 0.0, 1.0)
        h = h_new
        t = t_new
        if (it + 1) % 100 == 0:
            c5_chk = compute_c5(h)
            if c5_chk < c5_cur - 1e-12:
                c5_cur = c5_chk
                h_best = h.copy()
                lr = min(lr * 1.02, 0.3)
            else:
                lr *= 0.9
    c5_final = compute_c5(h)
    if c5_final < c5_cur:
        c5_cur = c5_final
        h_best = h.copy()
    return h_best, c5_cur

def adaptive_coord_descent(h0, n, target_sum, max_rounds, rng, base_step=0.01):
    h = h0.copy()
    c5_cur = compute_c5(h)
    h_best = h.copy()
    for rnd in range(max_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            h_orig = h[i]
            best_c5 = c5_cur
            best_val = h_orig
            steps = [base_step * s for s in [2.0, 1.0, 0.5, 0.25, 0.1, -0.1, -0.25, -0.5, -1.0, -2.0]]
            for d in steps:
                h[i] = np.clip(h_orig + d, 0.0, 1.0)
                h = project_to_sum(h, target_sum)
                c5_new = compute_c5(h)
                if c5_new < best_c5 - 1e-15:
                    best_c5 = c5_new
                    best_val = h[i]
            if best_val != h_orig:
                h[i] = best_val
                h = project_to_sum(h, target_sum)
                c5_cur = best_c5
                improved = True
            else:
                h[i] = h_orig
        if not improved:
            base_step *= 0.5
            if base_step < 1e-5:
                break
    return h, c5_cur

def block_perturbation(h0, n, target_sum, rng, n_trials=200):
    h = h0.copy()
    c5_cur = compute_c5(h)
    h_best = h.copy()
    block_sizes = [3, 5, 8, 12, 20]
    for _ in range(n_trials):
        bs = block_sizes[rng.randint(len(block_sizes))]
        start = rng.randint(0, n - bs)
        h_trial = h.copy()
        block = h_trial[start:start+bs].copy()
        delta = rng.uniform(-0.1, 0.1)
        h_trial[start:start+bs] = np.clip(block + delta, 0.0, 1.0)
        h_trial = project_to_sum(h_trial, target_sum)
        c5_new = compute_c5(h_trial)
        if c5_new < c5_cur - 1e-14:
            h = h_trial
            c5_cur = c5_new
            h_best = h.copy()
    return h_best, c5_cur

def interpolate_h(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src, endpoint=False)
    x_dst = np.linspace(0, 2, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_src)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    def time_left():
        return budget_s - (time.time() - t_start)
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            ts = best_n / 2.0
            best_h = project_to_sum(best_h, ts)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass
    if best_h is None:
        best_n = 360
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)
    n_parent = best_n
    ts_parent = n_parent / 2.0
    if time_left() > 100:
        h_cur = best_h.copy()
        for alpha in [10.0, 50.0, 200.0, 1000.0, 5000.0, 20000.0, 100000.0]:
            if time_left() < 50:
                break
            maxiter = max(50, min(400, int(time_left() * 1.2)))
            h_opt = slsqp_optimize(h_cur, ts_parent, alpha, maxiter)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_parent
            h_cur = h_opt
    if time_left() > 150:
        h_lb = lbfgs_optimize(best_h, best_n / 2.0, 5000.0, 500)
        c5_lb = compute_c5(h_lb)
        if c5_lb < best_c5:
            best_c5 = c5_lb
            best_h = h_lb.copy()

    # Try n=720
    if time_left() > 250:
        n_hi = 720
        ts_hi = n_hi / 2.0
        h_interp = interpolate_h(best_h, best_n, n_hi)
        h_interp = project_to_sum(h_interp, ts_hi)
        c5_int = compute_c5(h_interp)
        h_cur = h_interp.copy()
        if c5_int < best_c5:
            best_c5 = c5_int
            best_h = h_interp.copy()
            best_n = n_hi
        for alpha in [20.0, 200.0, 2000.0, 10000.0, 50000.0]:
            if time_left() < 60:
                break
            maxiter = max(30, min(200, int(time_left() * 1.0)))
            h_opt = slsqp_optimize(h_cur, ts_hi, alpha, maxiter)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_hi
            h_cur = h_opt
        if time_left() > 60:
            h_lb = lbfgs_optimize(h_cur, ts_hi, 10000.0, 300)
            c5_lb = compute_c5(h_lb)
            if c5_lb < best_c5:
                best_c5 = c5_lb
                best_h = h_lb.copy()
                best_n = n_hi

    # Final polish on best
    if best_h is not None and time_left() > 30:
        n_f = best_n
        ts_f = n_f / 2.0
        for alpha in [50000.0, 200000.0]:
            if time_left() < 25:
                break
            h_pol = slsqp_optimize(best_h, ts_f, alpha, 600)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()
        if time_left() > 40:
            h_cd, c5_cd = adaptive_coord_descent(best_h, n_f, ts_f, 80, rng, base_step=0.005)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()
        if time_left() > 30:
            h_blk, c5_blk = block_perturbation(best_h, n_f, ts_f, rng, 300)
            if c5_blk < best_c5:
                best_c5 = c5_blk
                best_h = h_blk.copy()
        if time_left() > 25:
            h_pg, c5_pg = nesterov_gradient(best_h, n_f, ts_f, 20000.0,
                                           min(2000, int(time_left() * 2)), 0.01)
            if c5_pg < best_c5:
                best_c5 = c5_pg
                best_h = h_pg.copy()
        if time_left() > 20:
            h_cd2, c5_cd2 = adaptive_coord_descent(best_h, n_f, ts_f, 100, rng, base_step=0.002)
            if c5_cd2 < best_c5:
                best_c5 = c5_cd2
                best_h = h_cd2.copy()
        # Final SLSQP
        if time_left() > 15:
            h_fin = slsqp_optimize(best_h, ts_f, 200000.0, 800)
            c5_fin = compute_c5(h_fin)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin.copy()

    n_final = len(best_h)
    ts_final = n_final / 2.0
    best_h = project_to_sum(best_h, ts_final)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_final)
