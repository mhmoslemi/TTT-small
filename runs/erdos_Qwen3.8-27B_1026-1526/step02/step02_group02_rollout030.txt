    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):

            break
        if diff > 0:

            if not np.any(mask):

            h[mask] += diff / np.sum(mask)

            h[mask] -= (-diff) / np.sum(mask)

def lp_iteration(h, n, eps=0.01, max_lags=None):
    """One iteration of the iterated linearization method."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    c_max = c.max()
    
    # Find near-active lags
    active_mask = c >= c_max - eps
    active_indices = np.where(active_mask)[0]
    
    if max_lags is not None and len(active_indices) > max_lags:
        # Keep only the most active
        sorted_idx = active_indices[np.argsort(c[active_indices])[::-1][:max_lags]]
        active_indices = sorted_idx

        active_indices = np.array([np.argmax(c)])

    # Variables: h[0..n-1], t
    # Minimize t

    #   linearization of c[k] at h <= t
    #   0 <= h[i] <= 1
    #   sum(h) = n/2

    # Objective: minimize t (last variable)
    c_obj = np.zeros(n_vars)
    c_obj[-1] = 1.0

    # For each active lag, we need: linearized_c[k] <= t
    # i.e., linearized_c[k] - t <= 0

    # c_k(h) = sum_i h[i] * (1 - h[i+d_k]) * dx
    # d_k = k - n + 1 (the lag)

            if 0 <= i - d < n:

    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])

        return result.x[:n], result.x[-1]
    return h, c.max()

    best_alpha = 0.0

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.interpolate import interp1d
import time

def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def compute_all_lags(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            h[mask] -= (-diff) / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def solve_lp_iteration(h, n):
    """One iteration of successive linear programming."""
    dx = 2.0 / n
    n_lags = 2 * n - 1
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    
    # Build constraint matrix
    # For each lag k, the linearized constraint is:
    # c[k] + grad_k @ (h_new - h) <= t
    # i.e., grad_k @ h_new - t <= c[k] - grad_k @ h
    
    # grad_k[i] = dx * [(1-h[i+d]) if valid + (-h[i-d]) if valid]
    # where d = k - n + 1
    
    A = np.zeros((n_lags, n + 1))
    b = np.zeros(n_lags)
    
    for k in range(n_lags):
        d = k - n + 1
        i_min = max(0, -d)
        i_max = min(n - 1, n - 1 - d)
        
        if i_min > i_max:
            A[k, -1] = -1.0
            b[k] = 0.0
            continue
        
        valid = np.arange(i_min, i_max + 1)
        
        # Term 1: d/dh[i] of h[i]*(1-h[i+d]) = (1-h[i+d]) for i in valid
        A[k, valid] += (1.0 - h[valid + d]) * dx
        
        # Term 2: d/dh[j] of h[j-d]*(1-h[j]) = -h[j-d] for j = valid + d
        j_idx = valid + d
        j_valid = (j_idx >= 0) & (j_idx < n)
        if np.any(j_valid):
            A[k, j_idx[j_valid]] -= h[valid[j_valid]] * dx
        
        A[k, -1] = -1.0
        b[k] = c[k] - (A[k, :n] @ h)
    
    # Objective: minimize t
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Equality: sum(h) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = [(0, 1)] * n + [(0, None)]
    
    result = linprog(c_obj, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.success:
        return result.x[:n], result.x[-1]
    return h, c.max()
