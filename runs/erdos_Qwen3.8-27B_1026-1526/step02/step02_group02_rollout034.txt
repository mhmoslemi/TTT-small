def fourier_h(coeffs, n, M):
    """Reconstruct h from Fourier coefficients.
    coeffs = [a1, b1, a2, b2, ..., aM, bM]
    h(x) = 1/2 + sum_j (a_j cos(2πj x/2) + b_j sin(2πj x/2))

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:

        if diff > 0:

        else:

def c5_smoothed(h, T):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    c_max = c.max()
    shifted = T * (c - c_max)
    shifted = np.clip(shifted, -500, 500)
    lse = c_max + np.log(np.sum(np.exp(shifted))) / T
    return lse

This is not the same as c_k(h) = ∫ h(x)(1-h(x+k)) dx in general. So h and 1-h are not symmetric.

But c_k(h') = ∫ (1-h(x))h(x+k) dx = ∫ h(x+k) dx - ∫ h(x)h(x+k) dx
And c_k(h) = ∫ h(x)(1-h(x+k)) dx = ∫ h(x) dx - ∫ h(x)h(x+k) dx (over the overlap)

So c_k(h') + c_k(h) = ∫_overlap [h(x) + h(x+k)] dx - 2∫_overlap h(x)h(x+k) dx

Not obviously equal. So no simple symmetry.

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time


def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        val = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(val)
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    result = np.clip(h - lam, 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-13:
        mask = (result > 1e-14) & (result < 1.0 - 1e-14)
        if np.any(mask):
            result[mask] += diff / np.sum(mask)
            result = np.clip(result, 0.0, 1.0)
    return result


def c5_logsumexp(h, T):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    c_max = c.max()
    if c_max < -100:
        return float(c_max)
    shifted = T * (c - c_max)
    shifted = np.clip(shifted, -500, 500)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / T)


def lbfgsb_obj(x, T, penalty, n):
    val = c5_logsumexp(x, T)
    sum_v = np.sum(x) - n / 2.0
    return val + penalty * sum_v * sum_v


def lbfgsb_optimize(h_init, n, T, penalty, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    result = minimize(
        lbfgsb_obj, h_init, args=(T, penalty, n),
        method='L-BFGS-B', bounds=bounds,
        options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-15}
    )
    h_out = project_feasible(result.x, n)
    return h_out, compute_c5_from_h(h_out)


def slsqp_obj_full(x):
    return x[-1]


def slsqp_ineq_full(x):
    h = x[:-1]
    t = x[-1]
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return t - c * dx


def slsqp_eq_full(x):
    h = x[:-1]
    return np.sum(h) - len(h) / 2.0


def slsqp_eq_jac_full(x):
    n = len(x) - 1
    jac = np.zeros(len(x))
    jac[:n] = 1.0
    return jac


def slsqp_optimize(h_init, n, maxiter=3000, t_margin=0.1):
    c5_init = compute_c5_from_h(h_init)
    x0 = np.concatenate([h_init, [c5_init + t_margin]])
    bounds = [(0.0, 1.0)] * n + [(max(0.0001, c5_init * 0.3), 2.0)]
    constraints = [
        {'type': 'ineq', 'fun': slsqp_ineq_full},
        {'type': 'eq', 'fun': slsqp_eq_full, 'jac': slsqp_eq_jac_full}
    ]
    result = minimize(slsqp_obj_full, x0, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
    if result.x is not None:
        h_out = project_feasible(result.x[:-1], n)
        return h_out, compute_c5_from_h(h_out)
    return h_init, c5_init


def trust_constr_optimize(h_init, n, T, penalty, maxiter=2000):

def fourier_coeffs_from_h(h, M):
    n = len(h)
    x = np.arange(n) * 2.0 / n
    h_c = h - np.mean(h)
    coeffs = np.zeros(2 * M)
    for j in range(1, M + 1):
        coeffs[2*(j-1)] = 2.0 / n * np.sum(h_c * np.cos(np.pi * j * x))
        coeffs[2*(j-1)+1] = 2.0 / n * np.sum(h_c * np.sin(np.pi * j * x))
    return coeffs


def fourier_reconstruct(coeffs, n, M):
    x = np.arange(n) * 2.0 / n
    mean_val = 0.5
    h = np.full(n, mean_val)
    for j in range(1, M + 1):
        h += coeffs[2*(j-1)] * np.cos(np.pi * j * x)
        h += coeffs[2*(j-1)+1] * np.sin(np.pi * j * x)
    return h


def fourier_optimize(h_init, n, M, T, penalty, maxiter=1000):
    coeffs0 = fourier_coeffs_from_h(h_init, M)
    mean_val = np.mean(h_init)
    
    def fourier_obj(coeffs):
        h = fourier_reconstruct(coeffs, n, M)
        h = h + (mean_val - np.mean(h))
        h_clip = np.clip(h, 0.0, 1.0)
        viol = np.sum(np.maximum(0, -h)**2) + np.sum(np.maximum(0, h-1)**2)
        return c5_logsumexp(h_clip, T) + penalty * viol
    
    bounds = [(-0.4, 0.4)] * (2 * M)
    result = minimize(fourier_obj, coeffs0, method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
    h_out = fourier_reconstruct(result.x, n, M)
    h_out = h_out + (np.mean(h_init) - np.mean(h_out))
    h_out = project_feasible(h_out, n)
    return h_out, compute_c5_from_h(h_out)
