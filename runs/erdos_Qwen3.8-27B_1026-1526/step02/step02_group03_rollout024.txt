import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(values, tau):
    v = values / tau
    vmax = np.max(v)
    return tau * (vmax + np.log(np.sum(np.exp(v - vmax))))

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-14:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        if np.sum(proj) > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-14:
        idx = int(np.argmin(np.abs(result - 0.5)))
        result[idx] = np.clip(result[idx] + diff, 0.0, 1.0)
    return result

def slsqp_obj(x, n, tau):
    dx = 2.0 / n
    g = 1.0 - x
    corr = np.correlate(x, g, mode="full")
    return smooth_max(corr, tau) * dx

def slsqp_cons(x, n):
    return np.sum(x) - n / 2.0

def run_slsqp(h0, n, tau_sched, max_iter_each, ftol=1e-13):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_cons, 'args': (n,)}]
    current = project_feasible(h0.copy(), n)
    for tau in tau_sched:
        try:
            res = minimize(
                slsqp_obj, current,
                method='SLSQP',
                args=(n, tau),
                bounds=bounds,
                constraints=cons,
                options={'maxiter': max_iter_each, 'ftol': ftol}
            )
            current = project_feasible(res.x, n)
        except Exception:
            pass
    return current, evaluate_c5(current)

        h_new = current.copy()
        h_new[i] += delta
        h_new[j] -= delta
        h_new = project_feasible(h_new, n)

        if c5_new < current_c5 or rng.rand() < np.exp((current_c5 - c5_new) / t):

            current_c5 = c5_new

                best = current.copy()
                best_c5 = c5_new

    best = h.copy()
    best_c5 = evaluate_c5(best)
    step_sizes = [0.05, 0.03, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    for sweep in range(max_sweeps):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        improved = False
        for i in range(n):
            for d in step_sizes:
                for sign in [1, -1]:
                    h_new = best.copy()
                    h_new[i] += sign * d
                    h_new = project_feasible(h_new, n)
                    c5 = evaluate_c5(h_new)
                    if c5 < best_c5 - 1e-12:
                        best = h_new
                        best_c5 = c5
                        improved = True
                        d = 0.0
                        break
        if not improved:
            break
    return best, best_c5

    max_val = np.max(corr)
    threshold = max_val - 0.05 * max_val
    focus_mask = corr >= threshold
    
    def focused_obj(x, n_arg, tau_arg):
        dx = 2.0 / n_arg
        g = 1.0 - x
        c = np.correlate(x, g, mode="full")
        obj = smooth_max(c, tau_arg)
        if np.any(focus_mask):
            obj += 0.3 * smooth_max(c[focus_mask], tau_arg * 0.5)
        return obj * dx
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_cons, 'args': (n,)}]

        res = minimize(
            focused_obj, h0,
            method='SLSQP',
            args=(n, tau),
            bounds=bounds,
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-13}
        )
        return project_feasible(res.x, n)
    except Exception:
        return h0
