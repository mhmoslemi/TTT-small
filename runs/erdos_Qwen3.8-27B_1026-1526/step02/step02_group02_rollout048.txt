```python
            idx2 = range(js2 + k, je2 + k)  # indices into h for (j+k)

    """Take a projected gradient step"""
    h_new = h - step_size * grad
    return project_feasible(h_new, n)

    T = T_init

    best_c5 = compute_c5_from_h(h)
    momentum = np.zeros(n)

            best_c5 = c5_new
        
        if obj < best_c5 - 0.01:  # converged
            break
        
        h = h_new
        T = max(T_min, T * 0.98)
    
    return best_h, best_c5
```

```python
def coordinate_sweep(h, n):
    """One sweep of coordinate descent"""
    dx = 2.0 / n
    for j in range(n):
        # Compute B_k for all k (lag from -(n-1) to n-1)
        B = np.zeros(2*n-1)
        for k in range(-(n-1), n):
            b = 0.0
            if 0 <= j-k < n:
                b += (1.0 - h[j-k])
            if 0 <= j+k < n:
                b -= h[j+k]
            B[k + n - 1] = b
        
        # Current c
        c = np.correlate(h, 1.0-h, mode="full") * dx
        
        # A_k = c[k] - h[j] * B_k
        A = c - h[j] * B
        
        # Binary search on subgradient
        lo, hi = 0.0, 1.0
        for _ in range(40):
            mid = (lo + hi) / 2
            val = A + mid * B
            k_star = np.argmax(val)
            if B[k_star] > 0:
                hi = mid
            elif B[k_star] < 0:
                lo = mid
            else:
                break
        t_opt = (lo + hi) / 2
        h[j] = t_opt
    
    return h
```

```python
        for k in range(-(n-1), n):
            b = 0.0
            if 0 <= j-k < n:
                b += (1.0 - h[j-k])
            if 0 <= j+k < n:
                b -= h[j+k]
            B[k + n - 1] = b
        
        old_val = h[j]
        
        # A_k = c[k] - old_val * B_k
        A = c - old_val * B
        
        # Binary search
        lo, hi = 0.0, 1.0
        for _ in range(40):
            mid = (lo + hi) / 2
            val = A + mid * B
            k_star = np.argmax(val)
            bs = B[k_star]
            if bs > 1e-15:
                hi = mid
            elif bs < -1e-15:
                lo = mid
            else:
                break
        t_opt = (lo + hi) / 2
        
        # Update h and c
        delta = t_opt - old_val
        if abs(delta) > 1e-15:
            h[j] = t_opt
            c += delta * B
    
    return h
```
