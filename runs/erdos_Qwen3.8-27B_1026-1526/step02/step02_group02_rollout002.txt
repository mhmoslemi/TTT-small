        if time.time() - t0 > budget_s - 150:
            break
        frac = i / max(1, n_pert)
        scale = 0.06 * (1 - frac) ** 0.8 + 0.004
        c5_cur, c_vals = compute_c5(best_h)

        r = rng.random()
        if r < 0.25:
            h_pert = directed_perturb(best_h, c_vals, rng, scale, n)
        elif r < 0.45:
            h_pert = best_h + scale * rng.standard_normal(n)
            n_modes = rng.integers(1, 4)
            for _ in range(n_modes):
                freq = rng.integers(1, max(2, n // 4))
                phase = rng.random() * 2 * np.pi
                amp = scale * 0.5
                h_pert += amp * np.cos(2 * np.pi * freq * np.arange(n) / n + phase)
        elif r < 0.6:
            h_pert = best_h.copy()
            n_blocks = rng.integers(2, max(3, n // 15))
            for _ in range(n_blocks):
                bsize = rng.integers(4, max(5, n // 8))
                start = rng.integers(0, max(1, n - bsize))
                h_pert[start:start + bsize] += scale * (rng.random(bsize) - 0.5) * 2
        elif r < 0.75:
            n_push = max(4, int(n * 0.12))
            h_pert = best_h.copy()
            push_idx = rng.choice(n, min(n_push, n), replace=False)
            for idx_i in push_idx:
                if rng.random() < 0.5:
                    h_pert[idx_i] = min(1.0, h_pert[idx_i] + scale * 4.0)
                else:
                    h_pert[idx_i] = max(0.0, h_pert[idx_i] - scale * 4.0)
        else:
            h_pert = best_h + scale * 0.4 * rng.standard_normal(n)

        h_pert = project_feasible(h_pert, n)

        r2 = rng.random()
        if r2 < 0.45:
            h_new, c5_new = active_slsqp(h_pert, n, 2500, 12, 0.03, 0.006)
        elif r2 < 0.75:
            T_cur = 40 + 150 * frac
            h_new, c5_new = lbfgsb_opt(h_pert, T_cur, 80 + 400 * frac, 1500)
        else:
            h_new, c5_new = active_slsqp(h_pert, n, 3000, 10, 0.02, 0.004)

        if c5_new < best_c5 - 1e-13:
            best_h, best_c5 = h_new, c5_new

    for _ in range(25):
        if time.time() - t0 > budget_s - 100:
            break
        scale = 0.002 + 0.012 * (1 - _ / 25.0) ** 2
        h_pert = best_h + scale * rng.standard_normal(n)
        h_pert = project_feasible(h_pert, n)
        h_new, c5_new = active_slsqp(h_pert, n, 3500, 8, 0.008, 0.0015)
        if c5_new < best_c5 - 1e-14:
            best_h, best_c5 = h_new, c5_new

    for n_try in [360, 540]:
        if time.time() - t0 > budget_s - 180:
            break
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        f_interp = interp1d(x_old, best_h, kind='cubic', fill_value="extrapolate")
        h_interp = np.clip(f_interp(x_new), 0.0, 1.0)
        h_interp = project_feasible(h_interp, n_try)
        h_new, c5_new = active_slsqp(h_interp, n_try, 4000, 15, 0.05, 0.005)
        if c5_new < best_c5 - 1e-10:
            best_h = h_new
            best_c5 = c5_new
            n = n_try

    if time.time() - t0 < budget_s - 20:
        h_new, c5_new = active_slsqp(best_h, n, 10000, 20, 0.003, 0.0005)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    if time.time() - t0 < budget_s - 5:
        h_new, c5_new = lbfgsb_opt(best_h, 1.0, 20000, 3000)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    best_h = project_feasible(best_h, len(best_h))
    best_c5, _ = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
