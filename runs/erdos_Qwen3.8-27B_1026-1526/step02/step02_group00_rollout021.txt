    # j + m: need 0 ≤ j + m_k < n
    jp = j + m[None, :]
    valid2 = (jp >= 0) & (jp < n)
    
    jm_clipped = np.clip(jm, 0, n-1)
    jp_clipped = np.clip(jp, 0, n-1)
    
    grad = np.sum(w[None, :] * (valid1 * (1.0 - h[jm_clipped]) - valid2 * h[jp_clipped]), axis=1)
    
    return obj, grad

def slsqp_local(h0, n, target_sum, alpha, maxiter=600):
    def f(h):
        return smooth_max_and_grad(h, alpha)[0]
    def g(h):
        return smooth_max_and_grad(h, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def trust_constr_local(h0, n, target_sum, alpha, maxiter=400):
    def f(h):
        return smooth_max_and_grad(h, alpha)[0]
    def g(h):
        return smooth_max_and_grad(h, alpha)[1]
    from scipy.optimize import LinearConstraint
    A_eq = np.ones((1, n))
    cons = [LinearConstraint(A_eq, target_sum, target_sum),
            LinearConstraint(np.eye(n), 0, 1)]
    try:
        res = minimize(f, h0.copy(), method='trust-constr', jac=g,
                       constraints=cons, maxiter=maxiter,
                       options={'gtol': 1e-10, 'xtol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def active_set_optimize(h, n, target_sum, alpha, maxiter=200):
    """Optimize focusing on the active set (lags near the max)."""
    for _ in range(maxiter):
        corr = compute_corr(h)
        m_val = np.max(corr)
        # Find active lags (within tolerance of max)
        tol = 0.01 * m_val
        active = np.where(corr >= m_val - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        
        # Compute gradient of the max (approximately uniform over active set)
        w = np.zeros(2*n-1)
        w[active] = 1.0 / len(active)
        
        m = np.arange(2*n - 1) - (n - 1)
        j = np.arange(n)[:, None]
        jm = j - m[None, :]
        valid1 = (jm >= 0) & (jm < n)
        jp = j + m[None, :]
        valid2 = (jp >= 0) & (jp < n)
        jm_c = np.clip(jm, 0, n-1)
        jp_c = np.clip(jp, 0, n-1)
        
        grad = np.sum(w[None, :] * (valid1 * (1.0 - h[jm_c]) - valid2 * h[jp_c]), axis=1)
        
        # Project gradient to satisfy equality constraint
        grad_proj = grad - np.mean(grad)
        
        # Line search
        best_h = h.copy()
        best_c5 = compute_c5(h)
        step = 0.05
        for _ in range(20):
            h_trial = h + step * grad_proj
            h_trial = project_to_sum(h_trial, target_sum)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5 - 1e-12:
                best_h = h_trial.copy()
                best_c5 = c5_trial
                h = h_trial
                step *= 0.7
            else:
                step *= 0.5
                if step < 1e-8:
                    break
        
        if np.max(np.abs(best_h - h)) < 1e-12:
            break
    return best_h

def binary_flip_search_v2(h, n, target_sum, rng, max_rounds=100):

        best_move_type = None

                c5_test = compute_c5(h_test)

                    best_move_type = 'double'

            if best_move_type == 'single':
                i, j = best_move
                h[i] = 1.0
                h[j] = 0.0

def make_starts_v2(n, target_sum, rng, best_h, best_n):

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    shifted = np.clip(corr - m_val, -50.0/alpha, 0.0)
    exp_vals = np.exp(alpha * shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    
    m = np.arange(2*n - 1) - (n - 1)
    j = np.arange(n)[:, None]
    jm = j - m[None, :]
    valid1 = (jm >= 0) & (jm < n)
    jp = j + m[None, :]
    valid2 = (jp >= 0) & (jp < n)
    jm_c = np.clip(jm, 0, n-1)
    jp_c = np.clip(jp, 0, n-1)
    
    grad = np.sum(w[None, :] * (valid1 * (1.0 - h[jm_c]) - valid2 * h[jp_c]), axis=1)
    return obj, grad

def slsqp_local(h0, n, target_sum, alpha, maxiter=600):
    def f(h):
        return smooth_max_and_grad(h, alpha)[0]
    def g(h):
        return smooth_max_and_grad(h, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def lp_fixed_point(h0, n, target_sum, max_iters=100, time_limit=60):
    """LP-based fixed-point iteration."""
    try:
        import cvxpy as cp
    except ImportError:
        return h0
    
    t0 = time.time()
    h = h0.copy()
    
    for it in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        
        g = 1.0 - h
        h_var = cp.Variable(n)
        C_var = cp.Variable()
        
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target_sum]
        
        m_vals = np.arange(2*n - 1) - (n - 1)

            i_lo = max(0, m)
            i_hi = min(n, n + m)
            if i_lo >= i_hi:
                continue
            idx = np.arange(i_lo, i_hi)
            terms = h_var[idx] * g[idx - m]
            constraints.append(cp.sum(terms) * (2.0/n) <= C_var)

                h[j] = 1.0

                if c5_t < best_c5 - 1e-10:
                    best_c5 = c5_t
                    best_move = (i1, i2, j1, j2)

                    h[i1] = h[i2] = 0.0

                    h[s2] = 1.0

                break
        else:
