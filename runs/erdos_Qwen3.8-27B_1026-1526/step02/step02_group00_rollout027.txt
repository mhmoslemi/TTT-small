        w = exp_vals / np.sum(exp_vals)

        # Term 1: if 0 <= j+shift < n, then (1-h[j+shift])

        grad[lo2:hi2] += wi * (-h[lo2 - shift:hi2 - shift])

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        count = np.sum(mask)
        if count > 0:
            h[mask] += diff / count
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    exp_v = np.exp(alpha * (corr - m))
    s = np.sum(exp_v)
    w = exp_v / s
    obj = m + np.log(s) / alpha
    
    grad = np.zeros(n)
    for k in range(len(corr)):
        wi = w[k]
        if wi < 1e-12:
            continue
        shift = k - (n - 1)
        lo1 = max(0, -shift)
        hi1 = min(n, n - shift)
        lo2 = max(0, shift)
        hi2 = min(n, n + shift)
        if hi1 > lo1:
            grad[lo1:hi1] += wi * (1.0 - h[lo1 + shift:hi1 + shift])
        if hi2 > lo2:
            grad[lo2:hi2] -= wi * h[lo2 - shift:hi2 - shift]
    
    return obj, grad

    h = h0.copy()

        return smooth_max_obj_and_grad(x, alpha)[0]

             'jac': lambda x: np.ones(n)}]

    try:

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})

def coordinate_descent(h, n, target_sum, steps=3, delta=0.01):
    """Try adjusting pairs of coordinates."""
    h = h.copy()
    c5_cur = compute_c5(h)
    improved = True
    while improved and steps > 0:
        improved = False
        steps -= 1
        for i in range(n):

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        count = np.sum(mask)
        if count > 0:
            h[mask] += diff / count
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    exp_v = np.exp(alpha * (corr - m))
    s = np.sum(exp_v)
    w = exp_v / s
    obj = m + np.log(s) / alpha
    grad = np.zeros(n)
    thresh = 1e-10
    sig = np.where(w > thresh)[0]
    for k in sig:
        wi = w[k]
        shift = k - (n - 1)
        lo1 = max(0, -shift)
        hi1 = min(n, n - shift)
        lo2 = max(0, shift)
        hi2 = min(n, n + shift)
        if hi1 > lo1:
            grad[lo1:hi1] += wi * (1.0 - h[lo1 + shift:hi1 + shift])
        if hi2 > lo2:
            grad[lo2:hi2] -= wi * h[lo2 - shift:hi2 - shift]
    return obj, grad

def optimize_penalty(h0, n, target_sum, alpha, penalties, maxiter_per_level):
    h = h0.copy()
    for pen in penalties:
        def fun(x):
            obj, grad = smooth_max_obj_and_grad(x, alpha)
            s_dev = np.sum(x) - target_sum
            return obj + 0.5 * pen * s_dev * s_dev
        def jac(x):
            _, grad = smooth_max_obj_and_grad(x, alpha)
            s_dev = np.sum(x) - target_sum
            return grad + pen * s_dev * np.ones(n)
        try:
            res = minimize(fun, h, method='L-BFGS-B', jac=jac,
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': maxiter_per_level, 'ftol': 1e-14, 'gtol': 1e-7})
            h = res.x
        except Exception:
            pass
    return project_to_sum(h, target_sum)

def slsqp_optimize(h0, n, target_sum, alpha, maxiter):
    h = h0.copy()
    def fun(x):
        return smooth_max_obj_and_grad(x, alpha)[0]
    def jac(x):
        return smooth_max_obj_and_grad(x, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(fun, h, method='SLSQP', jac=jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h, target_sum)

def exact_max_polish(h, n, target_sum, max_rounds=200):
    h = h.copy()
    c5_cur = compute_c5(h)
    for r in range(max_rounds):
        corr = compute_corr(h)
        top_k = np.argsort(corr)[::-1][:min(5, len(corr))]
        best_improve = 0
        best_h = h
        for k_max in top_k:
            shift = k_max - (n - 1)
            grad = np.zeros(n)
            lo1 = max(0, -shift)
            hi1 = min(n, n - shift)
            lo2 = max(0, shift)
            hi2 = min(n, n + shift)
            if hi1 > lo1:

                    best_c5_new = c5_new
        if best_h is h:

        c5_cur = best_c5_new

    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    
    return (best_h, best_c5, best_n)
