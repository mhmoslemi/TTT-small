        sorted_h = np.sort(h)
        threshold = sorted_h[n // 2]
        h = (h >= threshold).astype(float)
    elif pattern_type == 4:
        period = max(4, n // 4)
        base = thue_morse_seq(period).astype(float)
        h = np.tile(base, n // period + 1)[:n]
    elif pattern_type == 5:
        period = max(4, n // 6)
        base = thue_morse_seq(period).astype(float)
        h = np.tile(base, n // period + 1)[:n]
    else:
        h = rng.uniform(0, 1, n)
        sorted_h = np.sort(h)
        threshold = sorted_h[n // 2]
        h = (h >= threshold).astype(float)
    h = project_to_sum(h, n / 2.0)
    return h

def try_round_binary(h, n, target_sum, rng, max_rounds=40):
    h_bin = h.copy()
    h_bin[h_bin > 0.5] = 1.0
    h_bin[h_bin <= 0.5] = 0.0
    h_bin = project_to_sum(h_bin, target_sum)
    c5_bin = compute_c5(h_bin)
    c5_cont = compute_c5(h)
    if c5_bin < c5_cont - 1e-6:
        h_bin, c5_bin = binary_local_search(h_bin, n, target_sum, max_rounds, rng)
        return h_bin, c5_bin
    return h, c5_cont

def coordinate_descent(h0, n, target_sum, max_rounds, rng):
    h = h0.copy()
    c5_cur = compute_c5(h)
    deltas = np.array([-0.02, -0.01, -0.005, -0.002, 0.002, 0.005, 0.01, 0.02])
    for rnd in range(max_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            best_val = c5_cur
            best_d = 0.0
            for d in deltas:
                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] + d, 0.0, 1.0)
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_val - 1e-14:
                    best_val = c5_new
                    best_d = d
            if best_d != 0.0:
                h[i] = np.clip(h[i] + best_d, 0.0, 1.0)
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True
        if not improved:
            break
    return h, c5_cur

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            ts = best_n / 2.0
            best_h = project_to_sum(best_h, ts)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 360
        best_h = make_binary_pattern(best_n, 0)
        best_c5 = compute_c5(best_h)

    def time_left():
        return budget_s - (time.time() - t_start)

    n_targets = []
    if best_n % 2 == 0 and best_n >= 8:
        n_targets.append(best_n)
    for nn in [480, 600, 720, 360]:
        if nn % 2 == 0 and nn not in n_targets:
            n_targets.append(nn)

    for n in n_targets:
        if time_left() < 120:
            break
        ts = n / 2.0
        if abs(ts - round(ts)) > 1e-9:
            continue

        starts = []
        if best_n == n and best_h is not None:
            starts.append(best_h.copy())
        else:
            x_old = np.linspace(0, 2, best_n, endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            h_interp = np.interp(x_new, x_old, best_h)
            starts.append(project_to_sum(h_interp, ts))

        for pt in range(6):
            starts.append(make_binary_pattern(n, pt, rng))

        x = np.linspace(0, 2, n, endpoint=False)
        for freq in [2, 3, 5]:
            for amp in [0.35, 0.5]:
                hs = 0.5 + amp * np.sin(2.0 * np.pi * freq * x / 2.0)
                starts.append(project_to_sum(hs, ts))

        for h_start in starts:
            if time_left() < 50:
                break
            h_start = project_to_sum(h_start, ts)
            c5_s = compute_c5(h_start)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_start.copy()
                best_n = n
            h_cur = h_start.copy()

            alphas = [5.0, 15.0, 50.0, 150.0, 400.0, 1000.0]
            for alpha in alphas:
                if time_left() < 40:
                    break
                maxiter = max(50, min(400, int(time_left() * 2.0)))
                h_opt = l_bfgs_optimize(h_cur, n, ts, alpha, maxiter, time_left() - 5)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    best_n = n
                h_cur = h_opt

            if time_left() > 50:
                n_pg = min(3000, int(time_left() * 2))
                h_pg, c5_pg = projected_gradient(h_cur, n, ts, 200.0, n_pg, 0.04, time_left() - 30, t_start)
                if c5_pg < best_c5:
                    best_c5 = c5_pg
                    best_h = h_pg.copy()
                    best_n = n
                h_cur = h_pg

            if time_left() > 30:
                n_bt = min(1500, int(time_left() * 3))
                h_bt, c5_bt = backtracking_gradient(h_cur, n, ts, 500.0, n_bt, 0.1, time_left() - 10, t_start)
                if c5_bt < best_c5:
                    best_c5 = c5_bt
                    best_h = h_bt.copy()
                    best_n = n
                h_cur = h_bt

            if time_left() > 20:
                h_r, c5_r = try_round_binary(h_cur, n, ts, rng, 30)
                if c5_r < best_c5:
                    best_c5 = c5_r
                    best_h = h_r.copy()
                    best_n = n

    if best_h is not None and time_left() > 30:
        n_f = best_n
        ts_f = n_f / 2.0
        for alpha in [800.0, 2000.0, 5000.0]:
            if time_left() < 20:
                break
            h_pol = l_bfgs_optimize(best_h, n_f, ts_f, alpha, 400, time_left() - 10)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

        if time_left() > 25:
            h_cd, c5_cd = coordinate_descent(best_h, n_f, ts_f, 60, rng)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()

        if time_left() > 20:
            h_pg, c5_pg = projected_gradient(best_h, n_f, ts_f, 2000.0, min(5000, int(time_left() * 3)), 0.01, time_left() - 10, t_start)
            if c5_pg < best_c5:
                best_c5 = c5_pg
                best_h = h_pg.copy()

        if time_left() > 15:
            h_r, c5_r = try_round_binary(best_h, n_f, ts_f, rng, 80)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()

    n_final = len(best_h)
    ts_final = n_final / 2.0
    best_h = project_to_sum(best_h, ts_final)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_final)
