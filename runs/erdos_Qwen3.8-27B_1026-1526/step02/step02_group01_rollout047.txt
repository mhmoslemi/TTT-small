def compute_corr_array(h):

        if abs(diff) < 1e-14:
            break

            h[mask] += diff / np.sum(mask)

def grad_single_lag(h, lag, n):
    """Vectorized gradient of correlation at given lag."""
    dx = 2.0 / n
    g = np.zeros(n)
    if lag >= 0:
        g[lag:] += (1.0 - h[:n-lag]) * dx
        g[:n-lag] -= h[lag:] * dx
    else:

        g[L:] += (1.0 - h[:n-L]) * dx
        g[:n-L] -= h[L:] * dx

def active_set_optimize(h0, n, target_sum, n_iters, top_m, init_lr, end_lr):
    """Active-set optimization focusing on top-m lags."""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(n_iters):
        corr = compute_corr_array(h)
        k_max = np.argmax(corr)
        m_val = corr[k_max]
        
        # Select top m lags
        top_idx = np.argsort(corr)[-top_m:]
        # Weight by proximity to max (softmax with high temperature)
        diffs = corr[top_idx] - m_val
        w = np.exp(diffs * 50.0)
        w /= w.sum()
        
        # Compute weighted gradient
        g = np.zeros(n)
        for idx, wi in zip(top_idx, w):
            lag = idx - (n - 1)
            g += wi * grad_single_lag(h, lag, n)
        
        # Adaptive learning rate
        lr = init_lr * (end_lr / init_lr) ** (it / max(1, n_iters))
        
        # Try the step
        h_new = h - lr * g
        h_new = project_to_sum(h_new, target_sum)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c5

def coordinate_descent_fast(h0, n, target_sum, max_rounds, rng):

    c5_cur = compute_c5(h)

        idx_order = rng.permutation(n)

def upsample(h_old, n_old, n_new):

    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def multi_lag_gradient(h, n, top_m):

    return float(np.max(corr) * dx)

def compute_corr_array(h):

    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_single_lag(h, lag, n):
    dx = 2.0 / n
    g = np.zeros(n)
    if lag >= 0:
        g[lag:] += (1.0 - h[:n-lag]) * dx
        g[:n-lag] -= h[lag:] * dx
    else:
        L = -lag
        g[L:] += (1.0 - h[:n-L]) * dx
        g[:n-L] -= h[L:] * dx
    return g

        diffs = corr[top_idx] - m_val
