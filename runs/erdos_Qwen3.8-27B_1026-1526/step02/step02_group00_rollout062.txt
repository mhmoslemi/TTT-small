import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_affine_box(h, target):
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_grad_from_weights(h, w):
    n = len(h)
    j = np.arange(n)[:, None]
    k = np.arange(2 * n - 1)[None, :]
    d = k - (n - 1)
    idx1 = j - d
    idx2 = j + d
    valid1 = (idx1 >= 0) & (idx1 < n)
    valid2 = (idx2 >= 0) & (idx2 < n)
    c1 = np.clip(idx1, 0, n - 1)
    c2 = np.clip(idx2, 0, n - 1)
    return (np.sum(valid1 * (1.0 - h[c1]) * w[None, :], axis=1)
            - np.sum(valid2 * h[c2] * w[None, :], axis=1))

def smooth_max_weights(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    obj = m_val + np.log(s) / alpha
    return obj, w

def get_exact_subgrad(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = float(np.max(corr))
    active = (corr >= max_val - 1e-12).astype(float)
    total = active.sum()
    if total == 0:
        w = np.zeros(2 * n - 1)
        w[np.argmax(corr)] = 1.0
    else:
        w = active / total
    grad = compute_grad_from_weights(h, w)
    return grad, max_val

    pc = 1e5

    def f_pen(x):
        obj, _ = smooth_max_weights(x, alpha)
        return obj + pc * (np.sum(x) - target) ** 2

    def g_pen(x):
        _, w = smooth_max_weights(x, alpha)
        grad = compute_grad_from_weights(x, w)
        return grad + 2.0 * pc * (np.sum(x) - target)

            idx_b = coord + dd
            va = (idx_a >= 0) & (idx_a < n)
            vb = (idx_b >= 0) & (idx_b < n)
            ca = np.clip(idx_a, 0, n - 1)
            cb = np.clip(idx_b, 0, n - 1)
            g = np.sum(va * (1.0 - h[ca]) * w) - np.sum(vb * h[cb] * w)
            if coord == i:
                grad_i = g
            else:
                grad_j_val = g
        return grad_i - grad_j_val

    a = lo + 1e-10
    b = hi - 1e-10
    fa = df_dt(a)
    if fa >= 0:
        t_opt = lo
    else:
        fb = df_dt(b)
        if fb <= 0:
            t_opt = hi
        else:
            for _ in range(40):
                mid = (a + b) / 2.0
                fm = df_dt(mid)
                if fm > 0:
                    b = mid
                else:
                    a = mid
            t_opt = (a + b) / 2.0
    h[i] = t_opt
    h[j] = s - t_opt

def pairwise_sweep(h, n, alpha, max_passes=5, step_size=2):
    target = n / 2.0
    best_c5 = compute_c5(h)
    for p in range(max_passes):
        improved = False
        for i in range(0, n, 1):
            for j in range(i + step_size, n, step_size):
                s = h[i] + h[j]
                lo = max(0.0, s - 1.0)
                hi = min(1.0, s)
                if hi - lo < 1e-12:
                    continue
                h_before_i = h[i]
                h_before_j = h[j]
                pairwise_bisection(h, i, j, alpha)
                if abs(h[i] - h_before_i) > 1e-8 or abs(h[j] - h_before_j) > 1e-8:
                    improved = True
        h = project_affine_box(h, target)
        c5_cur = compute_c5(h)
        if c5_cur < best_c5:
            best_c5 = c5_cur
        if not improved:
            break
    return h, best_c5

def binary_flip_search(h0, n, n_rounds=80, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    target = n / 2.0
    h = (h0 > 0.5).astype(float)
    attempts = 0
    while abs(h.sum() - target) > 1e-10 and attempts < 20000:
        if h.sum() > target:
            ones = np.where(h > 0.5)[0]
            if len(ones) == 0:
                break
            idx = rng.choice(ones)
            h[idx] = 0.0
        else:
            zeros = np.where(h < 0.5)[0]
            if len(zeros) == 0:
                break
            idx = rng.choice(zeros)
            h[idx] = 1.0
        attempts += 1
    if abs(h.sum() - target) > 1e-10:
        return h0.copy(), compute_c5(h0)

    best_c5 = compute_c5(h)
    for _ in range(n_rounds):
        improved = False
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_pair = None
        best_trial = best_c5
        n_trials = min(150, len(ones) * len(zeros))
        for _ in range(n_trials):
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h_trial = h.copy()
            h_trial[i] = 0.0
            h_trial[j] = 1.0
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_trial - 1e-12:
                best_trial = c5_trial
                best_pair = (i, j)
        if best_pair is None:
            break
        i, j = best_pair
        h[i] = 0.0
        h[j] = 1.0
        best_c5 = best_trial
    return h, best_c5

def upscale(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old, period=2.0)
    return project_affine_box(h_new, n_new / 2.0)

def make_diverse_starts(n, target, rng, best_h, best_n, count=8):
    starts = []
    x = np.linspace(0, 2, n, endpoint=False)
    for freq in [2, 3, 4, 5, 7, 11]:
        s = 0.5 + 0.35 * np.sin(2.0 * np.pi * freq * x / 2.0)
        starts.append(project_affine_box(s, target))
    for nblk in [4, 6, 8, 12]:
        blk = n // nblk
        h = np.zeros(n)
        for i in range(0, n, 2 * blk):
            h[i:min(i + blk, n)] = 1.0
        starts.append(project_affine_box(h, target))
    for _ in range(4):
        noise = rng.uniform(-0.15, 0.15, n)
        if best_h is not None and best_n > 0:
            if best_n == n:
                base = best_h + noise
            else:
                x_old = np.linspace(0, 2, best_n, endpoint=False)
                base = np.interp(x, x_old, best_h, period=2.0) + noise
        else:
            base = np.full(n, 0.5) + noise
        starts.append(project_affine_box(base, target))
    return starts[:count]

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()

    def time_left():
        return budget_s - (time.time() - t0)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
        best_n = len(best_h)
        best_h = project_affine_box(best_h, best_n / 2.0)
        best_c5 = compute_c5(best_h)

    if best_h is None:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    if best_n % 2 != 0:
        best_n = best_n // 2 * 2
        best_h = upscale(np.array(initial_h_values) if 'initial_h_values' in globals() and initial_h_values is not None else best_h, best_n)
        best_c5 = compute_c5(best_h)

    # Phase 1: L-BFGS-B alpha continuation at initial resolution
    n1 = best_n
    alphas1 = [5, 15, 50, 150, 500, 2000, 8000]
    for alpha in alphas1:
        if time_left() < 40:
            break
        h_new = lbfgs_smooth(best_h, n1, alpha, maxiter=1000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 2: Nesterov subgradient polish
    if time_left() > 150:
        h_new, c5_new = nesterov_polish(best_h, best_n, n_iter=8000)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Phase 3: Pairwise bisection refinement
    if time_left() > 120:
        h_new, c5_new = pairwise_sweep(best_h, best_n, alpha=2000, max_passes=8, step_size=3)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
        if time_left() > 60:
            h_new, c5_new = pairwise_sweep(best_h, best_n, alpha=8000, max_passes=4, step_size=1)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new

    # Phase 4: Upscale to higher resolutions
    for n_new in [360, 480, 600]:
        if time_left() < 200:
            break
        if n_new % 2 != 0:
            continue
        target_new = n_new / 2.0
        if abs(target_new - round(target_new)) > 1e-6:
            continue
        h_up = upscale(best_h, n_new)
        for alpha in [10, 50, 200, 1000, 5000]:
            if time_left() < 80:
                break
            h_new = lbfgs_smooth(h_up, n_new, alpha, maxiter=800)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                best_n = n_new
            h_up = h_new
        if time_left() > 100:
            h_new, c5_new = nesterov_polish(best_h, best_n, n_iter=5000)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
        if time_left() > 80:
            h_new, c5_new = pairwise_sweep(best_h, best_n, alpha=3000, max_passes=4, step_size=2)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
                best_n = n_new

    # Phase 5: Binary flip + resmooth
    if time_left() > 150:
        h_bin, c5_bin = binary_flip_search(best_h, best_n, n_rounds=100, rng=rng)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
        for alpha in [50, 500, 5000]:
            if time_left() < 40:
                break
            h_sm = lbfgs_smooth(h_bin, best_n, alpha, maxiter=600)
            c5_sm = compute_c5(h_sm)
            if c5_sm < best_c5:
                best_c5 = c5_sm
                best_h = h_sm.copy()

    # Phase 6: Diverse restarts
    if time_left() > 180:
        starts = make_diverse_starts(best_n, best_n / 2.0, rng, best_h, best_n, count=10)
        for h_start in starts:
            if time_left() < 30:
                break
            for alpha in [20, 200, 2000]:
                if time_left() < 20:
                    break
                h_new = lbfgs_smooth(h_start, best_n, alpha, maxiter=500)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h_start = h_new

    # Phase 7: Final high-precision polish
    if time_left() > 60:
        for alpha in [10000, 50000]:
            if time_left() < 30:
                break
            h_new = lbfgs_smooth(best_h, best_n, alpha, maxiter=2000)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    if time_left() > 40:
        h_new, c5_new = nesterov_polish(best_h, best_n, n_iter=5000)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    if time_left() > 30:
        h_new, c5_new = pairwise_sweep(best_h, best_n, alpha=20000, max_passes=3, step_size=1)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Final validation
    best_h = project_affine_box(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
