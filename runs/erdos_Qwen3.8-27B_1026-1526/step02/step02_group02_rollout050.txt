    for _ in range(3000):
        s = np.sum(h)
        if abs(s - target) < 1e-14:
            break
        if s < target:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += (target - s) / cnt
        else:
            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] -= (s - target) / cnt
    return np.clip(h, 0.0, 1.0)


def smooth_obj_grad(h, T, penalty=500.0):
    n = len(h)
    dx = 2.0 / n
    A = np.correlate(h, 1.0 - h, mode="full")
    c = A * dx
    c_max = c.max()
    shifted = np.clip(T * (c - c_max), -500, 500)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    p = exp_vals / total
    obj = c_max + np.log(total) / T
    sum_viol = np.sum(h) - n / 2.0
    obj += penalty * sum_viol ** 2
    grad = np.zeros(n)
    for j in range(len(p)):
        w = p[j]
        if w < 1e-15:
            continue
        k = j - (n - 1)
        if k >= 0:
            grad[k:] += w * (1.0 - h[:n - k])
            grad[:n - k] -= w * h[k:]
        else:
            s = -k
            grad[:n - s] += w * (1.0 - h[s:])
            grad[s:] -= w * h[:n - s]
    grad *= dx
    grad += 2.0 * penalty * sum_viol
    return obj, grad


def smooth_obj_only(h, T, penalty=500.0):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    c_max = c.max()
    shifted = np.clip(T * (c - c_max), -500, 500)
    obj = c_max + np.log(np.sum(np.exp(shifted))) / T
    sum_viol = np.sum(h) - n / 2.0
    obj += penalty * sum_viol ** 2
    return obj


def slsqp_active(h_init, n, max_constraints=40, maxiter=3000, t_margin=0.05):
    c = compute_all_c(h_init)
    c5_init = c.max()
    x0_t = c5_init + t_margin
    active_idx = np.argsort(c)[-max_constraints:]
    h_curr = h_init.copy()

    for iteration in range(10):
        m = len(active_idx)

        def slsqp_obj(x):
            return x[-1]

        def slsqp_ineq(x):
            h = x[:-1]
            t = x[-1]
            c_all = np.correlate(h, 1.0 - h, mode="full") * (2.0 / len(h))
            active_c = c_all[active_idx]
            return t - active_c

        def slsqp_eq(x):
            h = x[:-1]
            return np.sum(h) - len(h) / 2.0

        def slsqp_eq_jac(x):
            nn = len(x) - 1
            jac = np.zeros(len(x))
            jac[:nn] = 1.0
            return jac

        constraints = [
            {'type': 'ineq', 'fun': slsqp_ineq},
            {'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac}
        ]
        x0 = np.concatenate([h_curr, [x0_t]])
        bounds = [(0.0, 1.0)] * n + [(max(0.001, c5_init * 0.5), 2.0)]
        result = minimize(slsqp_obj, x0, method='SLSQP', bounds=bounds,
                          constraints=constraints,
                          options={'maxiter': maxiter, 'ftol': 1e-14})
        if result.x is not None:
            h_curr = project_feasible(result.x[:-1], n)
            t_val = result.x[-1]
        else:
            break

        c_all = compute_all_c(h_curr)
        new_c5 = c_all.max()
        if new_c5 <= t_val + 1e-8:
            break
        top_m = np.argsort(c_all)[-max_constraints:]
        if np.array_equal(top_m, active_idx):
            break
        active_idx = top_m
        x0_t = new_c5 + 0.01

    h_final = project_feasible(h_curr, n)
    return h_final, compute_c5_from_h(h_final)


def lbfgsb_optimize(h_init, T_list, penalty_list, maxiters):
    h = h_init.copy()
    c5 = compute_c5_from_h(h)
    bounds = [(0.0, 1.0)] * len(h)
    for i, (T, pen, maxit) in enumerate(zip(T_list, penalty_list, maxiters)):
        result = minimize(
            smooth_obj_only, h, args=(T, pen), jac=None,
            method='L-BFGS-B', bounds=bounds,
            options={'maxiter': maxit, 'ftol': 1e-15, 'gtol': 1e-10,
                     'maxls': 50, 'maxfun': maxit * 3}
        )
        h_new = project_feasible(result.x, len(h))
        c5_new = compute_c5_from_h(h_new)
        if c5_new < c5:
            h = h_new
            c5 = c5_new
    return h, c5


def lbfgsb_with_grad(h_init, T_list, penalty_list, maxiters):
    h = h_init.copy()
    c5 = compute_c5_from_h(h)
    bounds = [(0.0, 1.0)] * len(h)
    for T, pen, maxit in zip(T_list, penalty_list, maxiters):
        def f(x):
            return smooth_obj_grad(x, T, pen)[0]

        def g(x):
            return smooth_obj_grad(x, T, pen)[1]

        result = minimize(
            f, h, jac=g, method='L-BFGS-B', bounds=bounds,
            options={'maxiter': maxit, 'ftol': 1e-15, 'gtol': 1e-10,
                     'maxls': 40, 'maxfun': maxit * 4}
        )
        h_new = project_feasible(result.x, len(h))
        c5_new = compute_c5_from_h(h_new)
        if c5_new < c5:
            h = h_new
            c5 = c5_new
    return h, c5


def resample_h(h_old, n_old, n_new):
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    f_interp = interp1d(x_old, h_old, kind='linear', fill_value="extrapolate")
    h_new = np.clip(f_interp(x_new), 0, 1)
    return project_feasible(h_new, n_new)


def structured_init(n, rng):
    r = rng.random()
    if r < 0.2:
        t = np.linspace(0, 2, n)
        h = 0.5 + 0.3 * np.sin(2 * np.pi * t / 0.7 + rng.random() * 2 * np.pi)
    elif r < 0.4:
        t = np.linspace(0, 2, n)
        h = 0.5 + 0.25 * np.sin(2 * np.pi * t / 1.3 + rng.random() * 2 * np.pi)
        h += 0.15 * np.sin(2 * np.pi * t / 0.5 + rng.random() * 2 * np.pi)
    elif r < 0.6:
        h = np.zeros(n)
        n_blocks = rng.integers(2, 5)
        blocks = np.sort(rng.choice(n, n_blocks, replace=False))
        for i, b in enumerate(blocks):
            width = max(1, n // (2 * n_blocks + 1))
            if i % 2 == 0:
                h[max(0, b - width):b + width] = 0.8
            else:
                h[max(0, b - width):b + width] = 0.2
    else:
        h = np.full(n, 0.5) + 0.2 * rng.standard_normal(n)
    return project_feasible(h, n)


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = np.full(270, 0.5)

    n_parent = len(h_parent)
    h_parent = project_feasible(h_parent, n_parent)
    c5_parent = compute_c5_from_h(h_parent)

    best_h = h_parent.copy()
    best_c5 = c5_parent
    n_best = n_parent

    def elapsed():
        return time.time() - t0

    def remaining():
        return budget_s - elapsed()

    resolutions = [135, 270, 405, 540, 720]

    for n_target in resolutions:
        if remaining() < 80:
            break

        h_start = resample_h(h_parent, n_parent, n_target)
        c5_start = compute_c5_from_h(h_start)

        starts = [h_start]
        n_extra_starts = max(2, int(remaining() / 120))
        for _ in range(n_extra_starts):
            h_s = structured_init(n_target, rng)
            starts.append(h_s)
            h_p = h_start + 0.1 * rng.standard_normal(n_target)
            starts.append(project_feasible(h_p, n_target))
            h_p2 = h_start.copy()
            n_flip = max(3, n_target // 15)
            idx = rng.choice(n_target, n_flip, replace=False)
            h_p2[idx] = 1.0 - h_p2[idx]
            starts.append(project_feasible(h_p2, n_target))

        for h_s in starts:
            if remaining() < 40:
                break

            h_s = project_feasible(h_s, n_target)

            h_opt, c5_opt = lbfgsb_with_grad(
                h_s,
                T_list=[30.0, 100.0, 300.0, 800.0],
                penalty_list=[200.0, 500.0, 1000.0, 2000.0],
                maxiters=[200, 200, 150, 100]
            )

            if remaining() < 20:
                break

            h_opt, c5_opt = slsqp_active(
                h_opt, n_target,
                max_constraints=50,
                maxiter=2000,
                t_margin=0.03
            )

            if c5_opt < best_c5 - 1e-12:
                best_h = h_opt.copy()
                best_c5 = c5_opt
                n_best = n_target

        if remaining() < 60:
            break

        if compute_c5_from_h(h_start) < c5_parent - 1e-12:
            h_parent = resample_h(best_h, n_best, n_parent)
            c5_parent = compute_c5_from_h(h_parent)

    # Phase 2: focused refinement on best solution
    if remaining() > 50:
        n_refine = n_best
        h_refine = best_h.copy()

        for attempt in range(20):
            if remaining() < 30:
                break
            scale = 0.005 + 0.02 * (1 - attempt / 20) ** 2
            h_pert = h_refine + scale * rng.standard_normal(n_refine)
            h_pert = project_feasible(h_pert, n_refine)

            h_new, c5_new = lbfgsb_with_grad(
                h_pert,
                T_list=[200.0, 500.0],
                penalty_list=[1000.0, 3000.0],
                maxiters=[150, 100]
            )
            if remaining() < 20:
                break
            h_new, c5_new = slsqp_active(
                h_new, n_refine,
                max_constraints=60,
                maxiter=3000,
                t_margin=0.02
            )
            if c5_new < best_c5 - 1e-12:
                best_h = h_new
                best_c5 = c5_new
                h_refine = h_new.copy()

    # Phase 3: try different resolutions for final push
    if remaining() > 80:
        for n_try in [360, 480, 600, 810]:
            if remaining() < 60:
                break
            h_try = resample_h(best_h, n_best, n_try)
            h_try, c5_try = lbfgsb_with_grad(
                h_try,
                T_list=[100.0, 400.0, 1000.0],
                penalty_list=[500.0, 2000.0, 5000.0],
                maxiters=[150, 150, 100]
            )
            h_try, c5_try = slsqp_active(
                h_try, n_try,
                max_constraints=60,
                maxiter=3000,
                t_margin=0.02
            )
            if c5_try < best_c5 - 1e-12:
                best_h = h_try
                best_c5 = c5_try
                n_best = n_try

    # Final polishing
    if remaining() > 10:
        h_polish, c5_polish = slsqp_active(
            best_h, n_best,
            max_constraints=80,
            maxiter=5000,
            t_margin=0.01
        )
        if c5_polish < best_c5:
            best_h = h_polish
            best_c5 = c5_polish

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5_from_h(best_h)

    return best_h, best_c5, len(best_h)
