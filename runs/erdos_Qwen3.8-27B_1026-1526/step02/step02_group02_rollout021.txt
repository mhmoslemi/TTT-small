def compute_c5_from_h(h):

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            # distribute proportionally to available room
            room = 1.0 - h[mask]
            total_room = room.sum()
            if total_room < 1e-15:
                break
            h[mask] += diff * room / total_room
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            h[mask] += diff * h[mask] / h[mask].sum()
    return np.clip(h, 0.0, 1.0)

    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx

    free_vals = h_work[free_indices].copy()
    n_free = len(free_indices)

    def obj_free(x):
        x = np.clip(x, 0.0, 1.0)
        h_full = h_work.copy()
        h_full[free_indices] = x
        h_full = project_feasible(h_full, n)
        return c5_logsumexp(h_full, T)

    result = minimize(obj_free, free_vals, method='L-BFGS-B',
                      bounds=[(0.0, 1.0)] * n_free,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-13})
    if result.x is not None:
        h_new = h_work.copy()
        h_new[free_indices] = result.x
        h_new = project_feasible(h_new, n)
        return h_new, compute_c5_from_h(h_new)
    return h_work, compute_c5_from_h(project_feasible(h_work, n))


def discrete_local_search(h, n, n_iter, rng, c5_current):
    best_h = h.copy()
    best_c5 = c5_current
    for it in range(n_iter):
        c_vec = compute_c5_vector(best_h)
        active_shifts = np.where(c_vec > best_c5 - 0.005)[0]
        lag = active_shifts[len(active_shifts) // 2] - (n - 1)

        if rng.random() < 0.5:
            i1 = rng.integers(0, n)
            i2 = rng.integers(0, n)
            h_try = best_h.copy()
            h_try[i1], h_try[i2] = best_h[i2], best_h[i1]
        else:
            i1 = rng.integers(0, n)
            delta = (rng.random() - 0.5) * 0.1
            i2 = rng.integers(0, n)
            h_try = best_h.copy()
            h_try[i1] = np.clip(best_h[i1] + delta, 0, 1)
            h_try[i2] = np.clip(best_h[i2] - delta, 0, 1)

        h_try = project_feasible(h_try, n)
        c_try = compute_c5_from_h(h_try)
        if c_try < best_c5 - 1e-12:
            best_h = h_try
            best_c5 = c_try
    return best_h, best_c5


def multi_resolution_optimize(h_best, n_best, c5_best, n_target, T, penalty, maxiter, rng):
    x_old = np.linspace(0, 2, n_best, endpoint=False)
    x_new = np.linspace(0, 2, n_target, endpoint=False)
    f_interp = interp1d(x_old, h_best, kind='linear', fill_value="extrapolate")
    h_interp = np.clip(f_interp(x_new), 0, 1)
    h_interp = project_feasible(h_interp, n_target)

    h_new, c5_new = lbfgsb_optimize(h_interp, n_target, T=T, penalty=penalty, maxiter=maxiter)
    if c5_new < c5_best:
        return h_new, c5_new, n_target
    return h_best, c5_best, n_best


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        n_init = 135
        x = np.linspace(0, 2, n_init, endpoint=False)
        h0 = 0.5 + 0.3 * np.sin(2 * np.pi * 3 * x / 2)
        h0 = np.clip(h0, 0, 1)
        h0 = project_feasible(h0, n_init)

    n = len(h0)
    h0 = project_feasible(h0, n)

    best_h = h0.copy()
    best_c5 = compute_c5_from_h(best_h)
    n_best = n

    def elapsed():
        return time.time() - t0

    # Phase 1: L-BFGS-B temperature annealing
    temperatures = [80, 50, 35, 25, 18, 13, 10, 7, 5, 4, 3, 2.5, 2, 1.5, 1.2, 1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.15, 0.1]
    for T in temperatures:
        if elapsed() > budget_s - 150:
            break
        maxiter = 4000 if T > 10 else 5000
        h_new, c5_new = lbfgsb_optimize(best_h, n_best, T=T, penalty=200.0, maxiter=maxiter)
        if c5_new < best_c5 - 1e-13:
            best_h = h_new
            best_c5 = c5_new
            n_best = len(best_h)

    # Phase 2: Rounding and re-optimization
    if elapsed() < budget_s - 120:
        for threshold in [0.08, 0.05, 0.03, 0.02, 0.01]:
            if elapsed() > budget_s - 80:
                break
            h_round, c5_round = round_and_reoptimize(best_h, n_best, threshold, T=5.0, penalty=200.0, maxiter=3000)
            if c5_round < best_c5 - 1e-12:
                best_h = h_round
                best_c5 = c5_round
            h_round, c5_round = round_and_reoptimize(best_h, n_best, threshold, T=1.0, penalty=300.0, maxiter=3000)
            if c5_round < best_c5 - 1e-12:
                best_h = h_round
                best_c5 = c5_round

    # Phase 3: Discrete local search
    if elapsed() < budget_s - 100:
        n_disp = min(3000, int((budget_s - 100 - elapsed()) * 3))
        best_h, best_c5 = discrete_local_search(best_h, n_best, n_disp, rng, best_c5)

    # Phase 4: More L-BFGS-B refinement after discrete moves
    for T in [3.0, 1.5, 0.8, 0.4, 0.2, 0.1]:
        if elapsed() > budget_s - 80:
            break
        h_new, c5_new = lbfgsb_optimize(best_h, n_best, T=T, penalty=300.0, maxiter=4000)
        if c5_new < best_c5 - 1e-13:
            best_h = h_new
            best_c5 = c5_new

    # Phase 5: Multi-resolution
    n_candidates = [n_best - 30, n_best - 15, n_best, n_best + 15, n_best + 30, n_best + 60]
    for n_try in n_candidates:
        if elapsed() > budget_s - 60:
            break
        if n_try < 30:
            continue
        h_interp_res, c5_res, n_res = multi_resolution_optimize(
            best_h, n_best, best_c5, n_try, T=10.0, penalty=200.0, maxiter=3000, rng=rng)
        if c5_res < best_c5:
            best_h = h_interp_res
            best_c5 = c5_res
            n_best = n_res
            h_new, c5_new = lbfgsb_optimize(best_h, n_best, T=3.0, penalty=300.0, maxiter=4000)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    # Phase 6: Targeted active-set optimization
    if elapsed() < budget_s - 50:
        c_vec = compute_c5_vector(best_h)
        n_top = max(5, len(c_vec) // 10)
        top_indices = np.argsort(c_vec)[-n_top:]
        top_lags = np.sort(top_indices - (n_best - 1))

        # Perturb to break symmetry among active shifts
        for attempt in range(8):
            if elapsed() > budget_s - 30:
                break
            h_pert = best_h.copy()
            for lag in top_lags[:3]:
                if lag >= 0:
                    idx = np.random.default_rng(attempt + 42).integers(0, n_best - lag)
                    h_pert[idx] = np.clip(h_pert[idx] + 0.02 * (1 - 2 * attempt % 2), 0, 1)
                else:
                    idx = np.random.default_rng(attempt + 99).integers(-lag, n_best)
                    h_pert[idx] = np.clip(h_pert[idx] + 0.02 * (1 - 2 * attempt % 2), 0, 1)
            h_pert = project_feasible(h_pert, n_best)
            h_new, c5_new = lbfgsb_optimize(h_pert, n_best, T=5.0, penalty=200.0, maxiter=3000)
            if c5_new < best_c5 - 1e-13:
                best_h = h_new
                best_c5 = c5_new

    # Phase 7: Final intensive refinement
    if elapsed() < budget_s - 15:
        n_final = min(8000, int((budget_s - 15 - elapsed()) * 2))
        best_h, best_c5 = discrete_local_search(best_h, n_best, n_final, rng, best_c5)

    if elapsed() < budget_s - 5:
        h_new, c5_new = lbfgsb_optimize(best_h, n_best, T=0.1, penalty=500.0, maxiter=5000)
        if c5_new < best_c5 - 1e-14:
            best_h = h_new
            best_c5 = c5_new

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5_from_h(best_h)

    return best_h, best_c5, len(best_h)
