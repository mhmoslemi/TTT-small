def run_lp_refine(b, n, max_shifts=30):
    """LP refinement: find optimal fractional adjustment to binary pattern."""
    S = int(np.sum(b))
    target_sum = n / 2.0
    req = target_sum - S
    
    if abs(req) < 1e-10:
        return b.copy(), evaluate_c5(b)
    
    if abs(req) > 1.0:
        return None, float('inf')
    
    eps = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [
        eps >= -b,
        eps <= 1.0 - b,
        cp.sum(eps) == req
    ]
    
    shifts = list(range(-max_shifts, max_shifts + 1))
    
    for j in shifts:
        if j == 0:
            const = float(np.sum(b * (1 - b)))
            coeff = (1 - 2 * b).copy()  # d/d eps_i of b_i(1-b_i) + eps_i(1-b_i) - b_i*eps_i = (1-b_i) - b_i = 1-2b_i
            constraints.append(const + coeff @ eps <= t)
        elif j > 0:
            const = float(np.sum(b[:n-j] * (1 - b[j:])))
            coeff = np.zeros(n)
            coeff[:n-j] += (1.0 - b[j:])
            coeff[j:] += -b[:n-j]
            constraints.append(const + coeff @ eps <= t)
        else:
            j2 = -j
            const = float(np.sum(b[j2:] * (1 - b[:n-j2])))
            coeff = np.zeros(n)
            coeff[j2:] += (1.0 - b[:n-j2])
            coeff[:n-j2] += -b[j2:]
            constraints.append(const + coeff @ eps <= t)
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.ECOS, max_iter=10000)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=10000)
        except Exception:
            return None, float('inf')
    
    if eps.value is not None and t.value is not None and np.isfinite(t.value):
        h = project_feasible(b + eps.value, n)
        return h, float(t.value)
    return None, float('inf')

def coordinate_descent(h, n, max_sweeps=20):
    """Coordinate descent: optimize one element at a time."""
    best = h.copy()
    best_c5 = evaluate_c5(best)
    
    for sweep in range(max_sweeps):
        improved = False
        for i in range(n):
            # Best value for h[i] given all others fixed
            # O(j) is linear in h[i] for each j, so the max is piecewise linear
            # We can find the best h[i] by checking the breakpoints
            # Simpler: try a few candidate values
            h_save = best[i]
            for candidate in np.linspace(0, 1, 11):
                best[i] = candidate
                adjusted = project_feasible(best, n)
                c5 = evaluate_c5(adjusted)
                if c5 < best_c5 - 1e-12:
                    best = adjusted.copy()
                    best_c5 = c5
                    improved = True
            best[i] = h_save
        
        if not improved:
            break
    
    return best, best_c5

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_overlaps(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def p_norm_max(values, p):
    return np.sum(np.power(values, p)) ** (1.0 / p)

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        if np.sum(proj) > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-14:
        idx = int(np.argmin(np.abs(result - 0.5)))
        result[idx] = np.clip(result[idx] + diff, 0.0, 1.0)
    return result

        h_cd, c5_cd = coordinate_descent(best_h, n, max_sweeps=5, n_candidates=11)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd
    if best_h is None:
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c5 = evaluate_c5(best_h)
    final_c5 = evaluate_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, n
