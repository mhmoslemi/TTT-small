            # v1: 0 <= i + j_k < n
            v1 = (i + j_k >= 0) & (i + j_k < n)

    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)


def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):

            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
        else:

            if not np.any(mask):
                break

    return np.clip(h, 0.0, 1.0)


def solve_1d_minimax(a, b, n_iter=80):
    lo = 0.0
    hi = 1.0
    for _ in range(n_iter):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a * m1 + b)
        f2 = np.max(a * m2 + b)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    return 0.5 * (lo + hi)


def coordinate_descent(h_init, max_sweeps=300, time_limit=None, t_start=None):

    dx = 2.0 / n
    m_total = 2 * n - 1

    best_c5 = float(f.max())
    best_h = h.copy()
    
    for sweep in range(max_sweeps):
        if time_limit is not None and time.time() - t_start > time_limit:
            break
        
        for i in range(n):
            v1 = (i + j_k >= 0) & (i + j_k < n)
            v2 = (i - j_k >= 0) & (i - j_k < n)
            
            a = np.zeros(m_total)
            if np.any(v1):
                a[v1] += (1.0 - h[i + j_k[v1]]) * dx
            if np.any(v2):
                a[v2] -= h[i - j_k[v2]] * dx
            
            b = f - a * h[i]
            
            x_opt = solve_1d_minimax(a, b)
            
            delta = x_opt - h[i]
            if abs(delta) > 1e-15:
                f += a * delta
                h[i] = x_opt

        c5 = float(f.max())
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()

    dx = 2.0 / n
    m_total = 2 * n - 1
    
    def obj(x):
        return x[-1]
    
    def ineq(x):
        h = x[:-1]
        t = x[-1]
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        return t - c
    
    def eq(x):
        h = x[:-1]
        return np.sum(h) - n / 2.0
    
    def eq_jac(x):
        j = np.zeros(n + 1)
        j[:n] = 1.0
        return j

    bounds = [(0.0, 1.0)] * n + [(max(0.001, c5_init * 0.5), 2.0)]

        {'type': 'eq', 'fun': eq, 'jac': eq_jac}
    ]
    
    result = minimize(obj, x0, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
    if result.x is not None and result.success:
        h_out = project_feasible(result.x[:-1], n)
        return h_out, compute_c5_from_h(h_out)
