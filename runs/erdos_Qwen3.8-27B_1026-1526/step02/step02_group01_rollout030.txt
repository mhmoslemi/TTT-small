    c[-1] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target_sum])
    bounds = [(0, 1)] * n + [(0, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], float(res.x[n])
    return None, None

def sca_optimize(h0, n, max_iters, time_limit, top_k=None):
    h = h0.copy()
    target_sum = n / 2.0
    h = project_to_sum(h, target_sum)
    c5_cur = compute_c5(h)
    h_best = h.copy()
    c5_best = c5_cur
    t_start = time.time()
    for it in range(max_iters):
        if time.time() - t_start > time_limit:
            break
        A_ub, b_ub = build_lp_matrices(h, n, top_k)
        h_new, t_lp = solve_sca_lp(A_ub, b_ub, n, target_sum)
        if h_new is None:
            break
        found = False
        for beta in [1.0, 0.7, 0.5, 0.3, 0.15, 0.075, 0.0375]:
            h_try = (1.0 - beta) * h + beta * h_new
            h_try = np.clip(h_try, 0.0, 1.0)
            h_try = project_to_sum(h_try, target_sum)
            c5_try = compute_c5(h_try)
            if c5_try < c5_cur - 1e-15:
                h = h_try
                c5_cur = c5_try
                if c5_cur < c5_best:
                    c5_best = c5_cur
                    h_best = h.copy()
                found = True
                break
        if not found:
            break
    return h_best, c5_best

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / len(exp_vals)) / alpha
    g = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-15:
            continue
        d = k - (n - 1)
        if d >= 0:
            i1 = np.arange(0, n - d)
            g[i1] += wk * (1.0 - h[i1 + d])
            i2 = np.arange(d, n)
            if len(i2) > 0:
                g[i2] -= wk * h[i2 - d]
        else:
            dd = -d
            i1 = np.arange(dd, n)
            g[i1] += wk * (1.0 - h[i1 - dd])
            i2 = np.arange(0, n - dd)
            if len(i2) > 0:
                g[i2] -= wk * h[i2 + dd]
    return sm_val, g

def projected_gradient(h0, n, target_sum, alpha, n_iters, lr_init):
    h = h0.copy()
    c5_cur = compute_c5(h)
    h_best = h.copy()
    t = 1.0
    y = h.copy()
    lr = lr_init
    for it in range(n_iters):
        _, g = smooth_max_and_grad(y, alpha)
        h_new = np.clip(y - lr * g, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        mo = (t - 1.0) / t_new
        y = h_new + mo * (h_new - h)
        y = np.clip(y, 0.0, 1.0)
        h = h_new
        t = t_new
        c5_new = compute_c5(h)
        if c5_new < c5_cur:
            c5_cur = c5_new
            h_best = h.copy()
            lr = min(lr * 1.03, 0.3)
        else:
            lr *= 0.7
    return h_best, c5_cur

def coordinate_descent(h0, n, target_sum, max_rounds, rng):
    h = h0.copy()
    c5_cur = compute_c5(h)
    deltas = [-0.003, -0.001, -0.0005, 0.0005, 0.001, 0.003]
    for rnd in range(max_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            best_val = c5_cur
            best_delta = 0.0
            for d in deltas:
                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] + d, 0.0, 1.0)
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_val - 1e-15:
                    best_val = c5_new
                    best_delta = d
            if best_delta != 0.0:
                h[i] = np.clip(h[i] + best_delta, 0.0, 1.0)
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True
        if not improved:
            break
    return h, c5_cur

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    def time_left():
        return budget_s - (time.time() - t_start)
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            ts = best_n / 2.0
            best_h = project_to_sum(best_h, ts)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass
    if best_h is None:
        best_n = 360
        best_h = np.full(360, 0.5)
        best_c5 = compute_c5(best_h)
    # Phase 1: SCA with all lags
    if time_left() > 80:
        h_sca, c5_sca = sca_optimize(best_h, best_n, 500, max(20, min(time_left() * 0.35, 400)), top_k=None)
        if c5_sca < best_c5:
            best_c5 = c5_sca
            best_h = h_sca.copy()
    # Phase 2: SCA with top-K lags (faster per iteration, more iterations)
    if time_left() > 60:
        for top_k in [100, 50, 30]:
            if time_left() < 30:
                break
            h_sca_k, c5_sca_k = sca_optimize(best_h, best_n, 300, max(10, min(time_left() * 0.2, 200)), top_k=top_k)
            if c5_sca_k < best_c5:
                best_c5 = c5_sca_k
                best_h = h_sca_k.copy()
    # Phase 3: Projected gradient for diversity
    if time_left() > 60:
        n_cur = len(best_h)
        ts_cur = n_cur / 2.0
        for alpha in [50.0, 100.0, 200.0]:
            if time_left() < 30:
                break
            n_pg = min(2000, int(time_left() * 2))
            h_pg, c5_pg = projected_gradient(best_h, n_cur, ts_cur, alpha, n_pg, 0.03)
            if c5_pg < best_c5:
                best_c5 = c5_pg
                best_h = h_pg.copy()
    # Phase 4: Coordinate descent polishing
    if time_left() > 20:
        n_cur = len(best_h)
        ts_cur = n_cur / 2.0
        h_cd, c5_cd = coordinate_descent(best_h, n_cur, ts_cur, 80, rng)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
    # Phase 5: Try different n values
    for n_try in [240, 480, 400]:
        if time_left() < 120:
            break
        if n_try % 2 != 0:
            continue
        ts_try = n_try / 2.0
        if abs(ts_try - round(ts_try)) > 1e-9:
            continue
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_to_sum(h_interp, ts_try)
        c5_interp = compute_c5(h_interp)
        if c5_interp < best_c5:
            best_c5 = c5_interp
            best_h = h_interp.copy()
            best_n = n_try
        if time_left() > 60:
            h_sca_n, c5_sca_n = sca_optimize(h_interp, n_try, 300, max(15, time_left() * 0.25), top_k=None)
            if c5_sca_n < best_c5:
                best_c5 = c5_sca_n
                best_h = h_sca_n.copy()
                best_n = n_try
        if time_left() > 30:
            h_cd_n, c5_cd_n = coordinate_descent(best_h, n_try, ts_try, 50, rng)
            if c5_cd_n < best_c5:
                best_c5 = c5_cd_n
                best_h = h_cd_n.copy()
                best_n = n_try
    # Phase 6: Final SCA + coordinate descent
    if time_left() > 20:
        n_f = len(best_h)
        ts_f = n_f / 2.0
        h_sca_f, c5_sca_f = sca_optimize(best_h, n_f, 200, max(5, time_left() * 0.4), top_k=50)
        if c5_sca_f < best_c5:
            best_c5 = c5_sca_f
            best_h = h_sca_f.copy()
        if time_left() > 10:
            h_cd_f, c5_cd_f = coordinate_descent(best_h, n_f, ts_f, 100, rng)
            if c5_cd_f < best_c5:
                best_c5 = c5_cd_f
                best_h = h_cd_f.copy()
    # Final
    n_final = len(best_h)
    ts_final = n_final / 2.0
    best_h = project_to_sum(best_h, ts_final)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_final)
