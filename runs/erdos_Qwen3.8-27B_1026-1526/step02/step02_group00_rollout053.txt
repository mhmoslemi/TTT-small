    grad_x = grad_h[:n/2] - grad_h[n/2:]
    # Line search
    step = initial_step
    while obj_new > obj + c * step * grad_x @ dx:
        step /= 2
    x = clip(x - step * grad_x, 0, 1)

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_sum(h, target_sum):
    n = len(h)
    h = h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(h_proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2, 0.0, 1.0)

def soft_max_obj_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    g = corr * dx
    g_max = np.max(g)
    shifted = alpha * (g - g_max)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    obj = g_max + np.log(sum_exp) / alpha
    p = exp_vals / sum_exp
    
    # Gradient computation (vectorized O(n²))
    j_idx = np.arange(n)[:, None]
    k_idx = np.arange(2*n-1)[None, :]
    m1 = j_idx + k_idx - (n-1)
    m2 = j_idx - k_idx + (n-1)
    valid1 = (m1 >= 0) & (m1 < n)
    valid2 = (m2 >= 0) & (m2 < n)
    m1c = np.clip(m1, 0, n-1)
    m2c = np.clip(m2, 0, n-1)
    term1 = np.sum(valid1 * (1.0 - h[m1c]) * p[None, :], axis=1)
    term2 = np.sum(valid2 * h[m2c] * p[None, :], axis=1)
    grad = dx * (term1 - term2)
    return obj, grad

def anti_periodic_optimize(x0, alpha, n_max_iter, step_init):
    """Optimize x (length n/2) where h = [x, 1-x]"""
    h_half = x0.copy()
    n_half = len(h_half)
    n = 2 * n_half
    obj_prev = compute_c5(np.concatenate([h_half, 1.0 - h_half]))
    step = step_init
    for it in range(n_max_iter):
        h = np.concatenate([h_half, 1.0 - h_half])
        obj, grad_h = soft_max_obj_grad(h, alpha)
        grad_x = grad_h[:n_half] - grad_h[n_half:]
        
        # Try a few step sizes
        best_step = 0
        best_obj = obj_prev
        for trial_step in [step, step*0.5, step*0.1, step*2.0]:
            x_trial = np.clip(h_half - trial_step * grad_x, 0.0, 1.0)
            h_trial = np.concatenate([x_trial, 1.0 - x_trial])
            obj_trial = compute_c5(h_trial)
            if obj_trial < best_obj:
                best_obj = obj_trial
                best_step = trial_step
        
        if best_step > 0:
            h_half = np.clip(h_half - best_step * grad_x, 0.0, 1.0)
            step = best_step
            obj_prev = best_obj
        else:
            step *= 0.5
            if step < 1e-10:
                break
    h_half = np.clip(h_half, 0.0, 1.0)
    return h_half

def make_starts_anti(n_half, rng, best_h, best_n):
    """Generate starting points for the anti-periodic optimization."""
    starts = []
    # Uniform
    starts.append(np.full(n_half, 0.5))
    # Random

    # Blocks

        blk = n_half // nblks

    # From best solution

def general_optimize(h0, alpha, n_max_iter, step_init):
    """General optimization without anti-periodic constraint."""
    n = len(h0)
    target = n / 2.0
    h_curr = h0.copy()
    h_curr = project_sum(h_curr, target)
    obj_prev = compute_c5(h_curr)
    step = step_init
    for it in range(n_max_iter):
        obj, grad = soft_max_obj_grad(h_curr, alpha)
        # Gradient project out the sum constraint direction
        # The constraint is sum(h) = n/2, so we need grad to be orthogonal to ones
        grad_proj = grad - np.mean(grad)  # project onto sum=0 subspace
        # Try step sizes
        best_step = 0
        best_obj = obj_prev
        for trial_step in [step, step*0.5, step*0.1]:
            h_trial = np.clip(h_curr - trial_step * grad_proj, 0.0, 1.0)
            h_trial = project_sum(h_trial, target)
            obj_trial = compute_c5(h_trial)
            if obj_trial < best_obj:

            h_curr = np.clip(h_curr - best_step * grad_proj, 0.0, 1.0)
            h_curr = project_sum(h_curr, target)
            step = best_step
            obj_prev = best_obj
        else:
            step *= 0.5
            if step < 1e-10:
                break
    h_curr = project_sum(np.clip(h_curr, 0, 1), target)
    return h_curr
