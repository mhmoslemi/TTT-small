3. Set up the problem: minimize C subject to c[ki] ≤ C for i=1..m, and the constraints on h.
4. Solve using SLSQP or a similar method.
5. Check if other lags exceed C. If so, add them to the set and repeat.

This is essentially the "active set" method for minimax problems.

```python
def active_set_minimax(h0, n, target_sum, m_top=5, max_iter=100):
    """Minimize max(corr) using active set on top m lags."""
    h = h0.copy()
    corr = compute_corr(h)
    
    for it in range(max_iter):
        # Find top m lags
        top_idx = np.argsort(-corr)[:m_top]
        
        # Define objective: C (the max value)
        # Constraints: corr[ki] <= C for i in top_idx
        #             sum(h) = target_sum
        #             0 <= h <= 1
        
        # We'll use SLSQP with variables [h, C]
        # Objective: C
        # Constraints: 
        #   corr[ki] - C <= 0 for each ki in top_idx
        #   sum(h) - target_sum = 0
        
        def objective(x):
            h_sub = x[:n]
            C_val = x[n]
            return C_val
        
        constraints = []
        # corr[ki] <= C
        for ki in top_idx:
            def make_constraint(kk):
                def con(x):
                    h_sub = x[:n]
                    C_val = x[n]
                    c = compute_corr(h_sub)
                    return C_val - c[kk]  # >= 0
                return con
            constraints.append({'type': 'ineq', 'fun': make_constraint(int(ki))})
        
        # sum(h) = target_sum
        constraints.append({'type': 'eq', 'fun': lambda x: np.sum(x[:n]) - target_sum})
        
        bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
        
        x0 = np.concatenate([h, [np.max(corr) + 0.01]])
        
        try:
            res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                          constraints=constraints, options={'maxiter': 300, 'ftol': 1e-12})
            if res.success:
                h_new = res.x[:n]
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < compute_c5(h):
                    h = h_new
                    corr = compute_corr(h)
                else:
                    # Try increasing m_top
                    break
            else:
                break
        except:
            break
    
    return h, compute_c5(h)
```
