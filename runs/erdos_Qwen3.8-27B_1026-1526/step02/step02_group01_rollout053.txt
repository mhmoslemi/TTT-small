def fft_smooth_grad(h, alpha):
    """Compute smooth max value and its gradient using FFT-based convolution."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    m = np.max(corr)
    shifted = alpha * (corr - m)

    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    sm_val = m + np.log(s) / alpha
    w = exp_vals / s  # weights, length 2n-1
    
    # w[k] for k=0..2n-2 corresponds to lag k-(n-1)
    # w_pos[k] = w[n-1+k] for k=0..n-1 (lags 0, 1, ..., n-1)
    # w_neg[m] = w[n-1-m] for m=1..n-1 (lags -1, -2, ..., -(n-1))
    
    w_pos = w[n-1:]  # length n
    w_neg = np.zeros(n)
    w_neg[1:] = w[n-2::-1]  # w[n-2], w[n-3], ..., w[0], length n-1
    
    h_rev = h[::-1]
    
    conv_wp_h = fftconvolve(w_pos, h, mode='full')[:n]
    conv_wp_hr = fftconvolve(w_pos, h_rev, mode='full')
    conv_wn_h = fftconvolve(w_neg, h, mode='full')[:n]
    conv_wn_hr = fftconvolve(w_neg, h_rev, mode='full')
    
    cum_wp = np.cumsum(w_pos)
    cum_wn = np.cumsum(w_neg)
    
    P1 = cum_wp - conv_wp_h
    P2 = conv_wp_hr[n-1::-1][:n]
    P3 = cum_wn[::-1] - conv_wn_hr[n-1::-1][:n]
    P4 = conv_wn_h
    
    grad = dx * (P1 - P2 + P3 - P4)
    
    return sm_val, grad

    """FISTA (accelerated projected gradient) optimization."""
    h = h0.copy()
    y = h0.copy()
    c5_cur = compute_c5(h)
    h_best = h.copy()
    t = 1.0
    
    for it in range(n_iters):
        sm_val, g = fft_smooth_grad(y, alpha)
        h_new = np.clip(y - lr * g, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        mom = (t - 1.0) / t_new
        y = h_new + mom * (h_new - h)

        t = t_new
        c5_new = compute_c5(h)
        if c5_new < c5_cur:

import numpy as np
from scipy.signal import fftconvolve
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad_fft(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    sm_val = m + np.log(s) / alpha
    w = exp_vals / s
    
    # w has length 2n-1, indexed 0 to 2n-2
    # w[n-1+k] = weight for lag k (k >= 0)
    # w[n-1-m] = weight for lag -m (m > 0)
    
    w_pos = w[n-1:]  # length n, w_pos[k] = weight for lag k
    w_neg = np.zeros(n)  # w_neg[m] = weight for lag -m
    w_neg[1:] = w[n-2::-1]  # w_neg[1]=w[n-2], w_neg[2]=w[n-3], ..., w_neg[n-1]=w[0]
    
    h_rev = h[::-1]
    
    # Convolutions
    c1 = fftconvolve(w_pos, h, mode='full')[:n]        # P part 1: sum_k w_pos[k]*h[i-k]
    c2 = fftconvolve(w_pos, h_rev, mode='full')[:n]   # reversed
    c3 = fftconvolve(w_neg, h, mode='full')[:n]        # P part: sum_k w_neg[k]*h[i-k]
    c4 = fftconvolve(w_neg, h_rev, mode='full')[:n]   # reversed
    
    cum_wp = np.cumsum(w_pos)
    cum_wn = np.cumsum(w_neg)
    
    # P1[i] = sum_{k=0}^{i} w_pos[k]*(1-h[i-k]) = cum_wp[i] - c1[i]
    # P2[i] = sum_{k=0}^{n-1-i} w_pos[k]*h[i+k] = c2_rev[n-1-i] where c2_rev = c2 reversed
    #        Actually c2[m] = sum_k w_pos[k]*h_rev[m-k] = sum_k w_pos[k]*h[n-1-m+k]
    #        For m = n-1-i: c2[n-1-i] = sum_k w_pos[k]*h[n-1-(n-1-i)+k] = sum_k w_pos[k]*h[i+k]
    #        with valid k: 0<=k<n and 0<=i+k<n, so 0<=k<=n-1-i
    #        So P2[i] = c2[n-1-i]
    
    # P3[i] = sum_{m=1}^{n-1-i} w_neg[m]*(1-h[i+m]) = cum_wn[n-1-i] - (sum_{m=1}^{n-1-i} w_neg[m]*h[i+m])
    #        The second part: sum_{m=1}^{n-1-i} w_neg[m]*h[i+m] 
    #        = c4[n-1-i] (by same logic as P2, but w_neg[0]=0 so it's the same)
    #        P3[i] = cum_wn[n-1-i] - c4[n-1-i]
    
    # P4[i] = sum_{m=1}^{i} w_neg[m]*h[i-m] = c3[i] (since w_neg[0]=0)
    
    P1 = cum_wp - c1
    P2 = c2[n-1::-1][:n]  # P2[i] = c2[n-1-i]
    P3 = cum_wn[::-1] - c4[n-1::-1][:n]  # P3[i] = cum_wn[n-1-i] - c4[n-1-i]
    P4 = c3
    
    grad = dx * (P1 - P2 + P3 - P4)
    
    return sm_val, grad

def slsqp_polish(h0, target_sum, alpha, maxiter):

        return smooth_max_and_grad_fft(x, alpha)[0]

        return smooth_max_and_grad_fft(x, alpha)[1]

        {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,

    ]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_fn, h0.copy(), jac=grad_fn, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})

        return h0.copy()
