    n = len(h)
    h = h.copy()

        c_curr = h[j]
        
        # Compute B_k (coefficient of h[j] in c_k)
        B = np.zeros(2 * n - 1)
        k_start = max(0, nm1 - j)
        k_end = min(2 * n - 2, 2 * nm1 - j)
        m_idx = np.arange(k_start, k_end + 1)
        B[m_idx] += (1.0 - h[j - nm1 + m_idx])
        
        k_start2 = max(0, j)
        k_end2 = min(2 * n - 2, j + nm1)
        i_idx = np.arange(k_start2, k_end2 + 1)
        B[i_idx] += -h[j + nm1 - i_idx]
        
        # A_k = corr_k - B_k * c_curr
        A = corr - B * c_curr
        
        # Minimize f(t) = max_k (A_k + B_k * t) over t in [0,1]
        t_grid = np.linspace(0, 1, 20)
        vals = np.max(A[None, :] + B[None, :] * t_grid[:, None], axis=1)
        best_idx = np.argmin(vals)

            t_opt = 0.0
        elif best_idx == len(t_grid) - 1:
            t_opt = 1.0
        else:
            lo = t_grid[max(0, best_idx - 1)]
            hi = t_grid[min(len(t_grid)-1, best_idx + 1)]

                if np.max(A + B * mid) <= np.max(A + B * lo):
                    lo = mid
                else:
                    hi = mid
            t_opt = (lo + hi) / 2
        
        # Update

        if abs(delta) > 1e-15:

                if time.time() >= t_end2:
                    break
                h_st, corr_st, _ = pair_descent_sweep(h_st, corr_st, rng, n_cand=8)
            c5_st = compute_c5(h_st)
            if c5_st < best_c5:
                best_c5 = c5_st
                best_h = h_st.copy()
                best_n = n_tr

    # Phase 3: SLSQP polish (10% of budget)
    t_end3 = t_start + 0.90 * budget_s
    n_f = best_n
    ts_f = n_f / 2.0
    for alpha in [20, 100, 500]:
        if time.time() >= t_end3:
            break
        h_pol = slsqp_local(best_h, n_f, ts_f, alpha, maxiter=600)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
            best_n = n_f

    # Phase 4: Final pair descent polish (10% of budget)
    h_fin = best_h.copy()
    n_fin = best_n
    corr_fin = np.correlate(h_fin, 1.0 - h_fin, mode="full")
    while time.time() < t_start + budget_s - 5:
        h_fin, corr_fin, _ = pair_descent_sweep(h_fin, corr_fin, rng, n_cand=12)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()
            best_n = n_fin

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
