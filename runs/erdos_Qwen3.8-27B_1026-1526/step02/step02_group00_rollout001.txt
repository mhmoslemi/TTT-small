```python
def compute_all_lag_grads(h):
    n = len(h)
    # grad[k, i] = d corr[k] / d h[i]
    # corr[k] = sum_i h[i] * (1 - h[i + k - n + 1])
    # d corr[k] / d h[m] = (1 - h[m + k - n + 1]) - h[m - k + n - 1]
    # where h[j] = 0 for j outside [0, n-1]
    
    lags = 2*n - 1
    # For each lag k, the gradient w.r.t. h[m] is:
    # (1 - h[m + k - n + 1] if 0 <= m+k-n+1 < n) 
    #   - (h[m - k + n - 1] if 0 <= m-k+n-1 < n)
    
    # Vectorized:
    j = np.arange(n)[:, None]  # (n, 1)
    m_idx = np.arange(2*n-1)[None, :]  # (1, 2n-1)
    shift = m_idx - n + 1  # k - n + 1, shape (1, 2n-1)
    
    # First term: h[m] * (1 - h[m + shift])
    # d/d h[m'] = (1 - h[m' + shift]) if valid
    f_idx = j + shift  # (n, 2n-1)
    valid_f = (f_idx >= 0) & (f_idx < n)
    h_f = np.where(valid_f, h[np.clip(f_idx, 0, n-1)], 0.0)
    grad_first = (1.0 - h_f) * valid_f  # (n, 2n-1)
    
    # Second term: -h[m - shift] = -h[m + n - 1 - k]
    s_idx = j - shift  # (n, 2n-1)  
    valid_s = (s_idx >= 0) & (s_idx < n)
    h_s = np.where(valid_s, h[np.clip(s_idx, 0, n-1)], 0.0)
    grad_second = h_s * valid_s  # (n, 2n-1)
    
    return grad_first - grad_second  # (n, 2n-1)
```

```python
def compute_c5(h):

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)

        if abs(diff) < 1e-13:

        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m_val))
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j = np.arange(n)[:, None].astype(np.float64)
    m_idx = np.arange(2 * n - 1, dtype=np.float64)[None, :]
    shift = m_idx - n + 1
    f_idx = (j + shift).astype(int)
    valid_f = (f_idx >= 0) & (f_idx < n)
    fc = np.clip(f_idx, 0, n - 1)
    h_f = np.where(valid_f, h[fc], 0.0)
    g1 = (1.0 - h_f) * valid_f.astype(np.float64)
    s_idx = (j - shift).astype(int)
    valid_s = (s_idx >= 0) & (s_idx < n)
    sc = np.clip(s_idx, 0, n - 1)
    h_s = np.where(valid_s, h[sc], 0.0)
    g2 = h_s * valid_s.astype(np.float64)
    lag_grad = g1 - g2
    grad = lag_grad.T @ w
    return obj, grad

def nesterov_smooth_max(h0, n, target_sum, alpha, max_iter=10000, lr=None, momentum=0.9):
    h = h0.copy().astype(np.float64)
    y = h.copy()
    v = np.zeros(n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    if lr is None:
        lr = 0.3 / (alpha + 1.0)
    for it in range(max_iter):
        obj, grad_y = smooth_max_obj_grad(y, alpha)
        y = h + momentum * v
        obj2, grad_h = smooth_max_obj_grad(y, alpha)
        v = momentum * v - lr * grad_h
        h_new = h + v
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

def slsqp_local(h0, n, target_sum, alpha, maxiter=1000):
    def f_and_g(h):
        return smooth_max_obj_grad(h, alpha)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f_and_g, h0.copy(), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def coordinate_descent(h, target_sum, max_rounds=100):
    h = h.copy()
    n = len(h)
    dx = 2.0 / n
    for rnd in range(max_rounds):
        improved = False
        order = np.random.permutation(n)
        for i in order:
            old_val = h[i]
            for delta in [0.02, -0.02, 0.05, -0.05, 0.01, -0.01]:
                new_val = old_val + delta
                if new_val < 0 or new_val > 1:
                    continue
                h[i] = new_val
                h = project_to_sum(h, target_sum)
                c5_test = compute_c5(h)
                h[i] = old_val
                if c5_test < 0:
                    pass
            h[i] = old_val
        h = project_to_sum(h, target_sum)
    return h

def binary_tabu_search(h, n, target_sum, rng, max_rounds=200, tabu_size=50):
    h_bin = np.zeros(n)
    h_bin[h > 0.5] = 1.0
    attempts = 0
    while abs(h_bin.sum() - target_sum) > 1e-10 and attempts < 10000:
        if h_bin.sum() > target_sum:
            ones = np.where(h_bin > 0.5)[0]
            if len(ones) == 0:
                break
            h_bin[rng.choice(ones)] = 0.0
        else:
            zeros = np.where(h_bin < 0.5)[0]
            if len(zeros) == 0:
                break
            h_bin[rng.choice(zeros)] = 1.0
        attempts += 1
    if abs(h_bin.sum() - target_sum) > 1e-10:
        return h, compute_c5(h)
    c5_cur = compute_c5(h_bin)
    best_h = h_bin.copy()
    best_c5 = c5_cur
    tabu = [set() for _ in range(tabu_size)]
    tabu_idx = 0
    for rnd in range(max_rounds):
        ones = np.where(h_bin > 0.5)[0]
        zeros = np.where(h_bin < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_move = None
        best_c5_move = c5_cur
        n_trials = min(500, len(ones) * len(zeros))
        for _ in range(n_trials):
            i = rng.choice(ones)
            j = rng.choice(zeros)
            key = (i, j)
            if key in tabu[tabu_idx]:
                continue
            old_i = h_bin[i]
            old_j = h_bin[j]
            h_bin[i] = 0.0
            h_bin[j] = 1.0
            c5_test = compute_c5(h_bin)
            h_bin[i] = old_i
            h_bin[j] = old_j
            if c5_test < best_c5_move - 1e-12:
                best_c5_move = c5_test
                best_move = (i, j)
        if best_move is None:
            n_trials2 = min(100, len(ones) * len(zeros))
            for _ in range(n_trials2):
                i = rng.choice(ones)
                j = rng.choice(zeros)
                old_i = h_bin[i]
                old_j = h_bin[j]
                h_bin[i] = 0.0
                h_bin[j] = 1.0
                c5_test = compute_c5(h_bin)
                h_bin[i] = old_i
                h_bin[j] = old_j
                if c5_test < best_c5_move - 1e-12:
                    best_c5_move = c5_test
                    best_move = (i, j)
        if best_move is not None:
            i, j = best_move
            h_bin[i] = 0.0
            h_bin[j] = 1.0
            tabu[tabu_idx].add(best_move)
            c5_cur = best_c5_move
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h_bin.copy()
        tabu_idx = (tabu_idx + 1) % tabu_size
        if tabu_idx == 0:
            for t in tabu:
                t.clear()
    return best_h, best_c5

def upscale_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old, period=2.0)
    return h_new

def generate_starts(n, target_sum, rng, best_h=None, best_n=0):
    starts = []
    starts.append(np.full(n, 0.5))
    starts.append(project_to_sum(rng.uniform(0, 1, n), target_sum))
    x = np.linspace(0, 2, n, endpoint=False)
    for freq in [1, 2, 3, 4, 5, 6, 8, 12]:
        s = 0.5 + 0.35 * np.sin(2.0 * np.pi * freq * x / 2.0)
        starts.append(project_to_sum(s, target_sum))
    for nblks in [3, 4, 5, 6, 7, 8, 10, 12]:
        h = np.zeros(n)
        blk = n // nblks
        for i in range(0, n, 2 * blk):
            h[i:min(i + blk, n)] = 1.0
        starts.append(project_to_sum(h, target_sum))
    for _ in range(6):
        a1 = rng.uniform(0.15, 0.45)
        b1 = rng.uniform(-0.25, 0.25)
        f1 = rng.choice([1, 2, 3, 4, 5, 6, 7, 9])
        f2 = rng.choice([2, 3, 4, 5, 6, 7, 8, 10, 12])
        p1 = rng.uniform(0, 2 * np.pi)
        p2 = rng.uniform(0, 2 * np.pi)
        h = 0.5 + a1 * np.sin(2 * np.pi * f1 * x / 2 + p1) + b1 * np.sin(2 * np.pi * f2 * x / 2 + p2)
        starts.append(project_to_sum(h, target_sum))
    if best_h is not None and best_n > 0:
        if best_n == n:
            starts.append(best_h.copy())
        else:
            starts.append(upscale_h(best_h, n))
    hb = (rng.rand(n) > 0.5).astype(float)
    att = 0
    while abs(hb.sum() - target_sum) > 1e-10 and att < 10000:
        if hb.sum() > target_sum:
            idx = rng.choice(np.where(hb > 0.5)[0])
            hb[idx] = 0.0
        else:
            idx = rng.choice(np.where(hb < 0.5)[0])
            idx = rng.choice(np.where(hb < 0.5)[0]) if np.any(hb < 0.5) else 0
            hb[idx] = 1.0
        att += 1
    if abs(hb.sum() - target_sum) < 1e-10:
        starts.append(hb.copy())
    return starts

def time_remaining(t0, budget_s):
    return budget_s - (time.time() - t0)

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n // 2)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    # Phase 1: Nesterov accelerated smooth max at current resolution
    ts = best_n // 2
    for alpha in [10, 30, 100, 300, 1000, 3000]:
        if time_remaining(t0, budget_s) < 30:
            break
        h_new, c5_new = nesterov_smooth_max(best_h, best_n, ts, alpha,
                                             max_iter=8000, lr=0.5/(alpha+1), momentum=0.9)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h_slq = slsqp_local(best_h, best_n, ts, alpha, maxiter=500)
        c5_slq = compute_c5(h_slq)
        if c5_slq < best_c5:
            best_c5 = c5_slq
            best_h = h_slq.copy()

    # Phase 2: Multi-resolution search
    resolutions_to_try = []
    if best_n <= 240:
        resolutions_to_try = [320, 384, 480, 576]
    elif best_n <= 480:
        resolutions_to_try = [480, 576, 640, 720]
    else:
        resolutions_to_try = [best_n]

    resolutions_to_try = [best_n] + resolutions_to_try

    for n_tr in resolutions_to_try:
        if time_remaining(t0, budget_s) < 100:
            break
        if n_tr % 2 != 0:
            n_tr -= 1
        ts_tr = n_tr // 2
        if n_tr == best_n:
            continue
        if best_n > n_tr:
            h_start = project_to_sum(
                np.interp(np.linspace(0, 2, n_tr, endpoint=False),
                          np.linspace(0, 2, best_n, endpoint=False),
                          best_h, period=2.0), ts_tr)
        else:
            h_start = project_to_sum(upscale_h(best_h, n_tr), ts_tr)
        c5_s = compute_c5(h_start)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_start.copy()
            best_n = n_tr
        for alpha in [20, 100, 500]:
            if time_remaining(t0, budget_s) < 25:
                break
            h_new, c5_new = nesterov_smooth_max(h_start, n_tr, ts_tr, alpha,
                                                 max_iter=6000, lr=0.4/(alpha+1), momentum=0.9)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                best_n = n_tr
            h_slq = slsqp_local(h_new, n_tr, ts_tr, alpha, maxiter=600)
            c5_slq = compute_c5(h_slq)
            if c5_slq < best_c5:
                best_c5 = c5_slq
                best_h = h_slq.copy()
                best_n = n_tr

    # Phase 3: Multi-start diverse optimization at best resolution
    n_opt = best_n
    ts_opt = n_opt // 2
    starts = generate_starts(n_opt, ts_opt, rng, best_h, best_n)
    for h_start in starts[:8]:
        if time_remaining(t0, budget_s) < 40:
            break
        h_start = project_to_sum(h_start, ts_opt)
        c5_s = compute_c5(h_start)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_start.copy()
        for alpha in [30, 150, 600]:
            if time_remaining(t0, budget_s) < 20:
                break
            h_new, c5_new = nesterov_smooth_max(h_start, n_opt, ts_opt, alpha,
                                                 max_iter=5000, lr=0.3/(alpha+1), momentum=0.85)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_slq = slsqp_local(h_new, n_opt, ts_opt, alpha, maxiter=400)
            c5_slq = compute_c5(h_slq)
            if c5_slq < best_c5:
                best_c5 = c5_slq
                best_h = h_slq.copy()

    # Phase 4: Binary tabu search
    if time_remaining(t0, budget_s) > 80:
        h_bin, c5_bin = binary_tabu_search(best_h, best_n, ts_opt, rng,
                                            max_rounds=300, tabu_size=80)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
        h_smooth = slsqp_local(h_bin, best_n, ts_opt, 200, maxiter=1000)
        c5_smooth = compute_c5(h_smooth)
        if c5_smooth < best_c5:
            best_c5 = c5_smooth
            best_h = h_smooth.copy()

    # Phase 5: Shake and re-optimize
    for trial in range(20):
        if time_remaining(t0, budget_s) < 30:
            break
        noise = 0.1 * (1.0 - trial / 20.0) + 0.02
        h_shake = best_h + rng.uniform(-noise, noise, best_n)
        h_shake = project_to_sum(h_shake, ts_opt)
        for alpha in [50, 200, 800]:
            if time_remaining(t0, budget_s) < 15:
                break
            h_new, c5_new = nesterov_smooth_max(h_shake, best_n, ts_opt, alpha,
                                                 max_iter=4000, lr=0.3/(alpha+1), momentum=0.9)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_slq = slsqp_local(h_new, best_n, ts_opt, alpha, maxiter=400)
            c5_slq = compute_c5(h_slq)
            if c5_slq < best_c5:
                best_c5 = c5_slq
                best_h = h_slq.copy()

    # Phase 6: Final high-alpha polish
    if time_remaining(t0, budget_s) > 40:
        for alpha in [2000, 5000, 10000]:
            if time_remaining(t0, budget_s) < 20:
                break
            h_pol, c5_pol = nesterov_smooth_max(best_h, best_n, ts_opt, alpha,
                                                 max_iter=3000, lr=0.2/(alpha+1), momentum=0.95)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()
            h_slq = slsqp_local(best_h, best_n, ts_opt, alpha, maxiter=800)
            c5_slq = compute_c5(h_slq)
            if c5_slq < best_c5:
                best_c5 = c5_slq
                best_h = h_slq.copy()

    # Final validation
    best_h = project_to_sum(best_h, ts_opt)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
```
