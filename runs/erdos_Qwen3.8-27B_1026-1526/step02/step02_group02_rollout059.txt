    C = np.array([np.sum(h[:n-k] * (1 - h[k:])) for k in range(n)])
    
    for p in range(max_passes):
        improved = False
        for i in range(n):
            v = h[i]
            if v <= 1e-15 or v >= 1 - 1e-15:
                continue  # at boundary, can only move one way
            
            # Compute slopes
            slopes = np.zeros(n)
            upper_end = n - 1 - i
            lower_end = i
            if upper_end >= 0:
                slopes[:upper_end+1] += (1.0 - h[i:min(n, i+upper_end+1)])
            if lower_end >= 0:
                slopes[:lower_end+1] -= h[max(0, i-lower_end):i+1][::-1]
            
            # Try x = 0, x = 1, x = v (current)
            # Also try the "crossing point" of the two most important lines
            candidates = [0.0, 1.0, v]
            
            # Find the k that maximizes C at current v (this is the "most important" constraint)
            k_max = np.argmax(C)
            # Find the k with the most negative slope (would benefit most from increasing x)
            k_min_slope = np.argmin(slopes)
            # Find the k with the most positive slope (would be hurt most by increasing x)
            k_max_slope = np.argmax(slopes)
            
            # Add crossing points
            for k1 in [k_max, k_max_slope]:
                for k2 in [k_max, k_min_slope]:
                    if k1 != k2 and abs(slopes[k1] - slopes[k2]) > 1e-15:
                        u_cross = (C[k2] - C[k1]) / (slopes[k1] - slopes[k2])
                        x_cross = v + u_cross
                        if 0 <= x_cross <= 1:
                            candidates.append(x_cross)
            
            # Evaluate objective at each candidate
            best_x = v
            best_obj = np.max(C)
            for x in candidates:
                C_trial = C + (x - v) * slopes
                obj = np.max(C_trial)
                if obj < best_obj - 1e-15:
                    best_obj = obj
                    best_x = x

                delta = best_x - v
                h[i] = best_x
                C += delta * slopes
                improved = True
