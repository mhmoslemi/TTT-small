import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_topk(h, k_top):
    """Compute subgradient using top-k shifts of the correlation."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    # Get top-k indices
    idx = np.argsort(corr)[::-1][:k_top]
    weights = np.zeros(2*n - 1)
    for i in idx:
        weights[i] = 1.0
    # Compute gradient for each shift and average
    grad = np.zeros(n)
    for m_idx in idx:
        # shift = m_idx - (n-1)
        k = m_idx - (n - 1)
        if k == 0:
            # f_0 = sum h[j](1-h[j]) = sum h[j] - sum h[j]^2
            # grad = 1 - 2h
            grad += 1.0 - 2.0 * h
        elif k > 0:
            # f_k = sum_{j=0}^{n-1-k} h[j](1-h[j+k])
            # grad[j] = (1-h[j+k]) for j in [0, n-1-k]
            #         - h[j-k] for j in [k, n-1]
            upper = n - 1 - k
            grad[:upper+1] += (1.0 - h[k:upper+1+k])  # h[j+k] for j in [0, n-1-k]
            grad[k:] -= h[:upper+1]  # h[j-k] for j in [k, n-1]
        else:
            k_neg = -k
            # f_{-k'} = sum_{i=0}^{n-1-k'} h[i+k'](1-h[i])
            # grad[j] = (1-h[j-k']) for j in [k', n-1]
            #         - h[j+k'] for j in [0, n-1-k']
            upper = n - 1 - k_neg
            grad[k_neg:] += (1.0 - h[:upper+1])  # 1-h[j-k'] for j in [k', n-1]
            grad[:upper+1] -= h[k_neg:upper+1+k_neg]  # h[j+k'] for j in [0, n-1-k']
    grad /= len(idx)
    return grad, np.max(corr)

def topk_descent(h0, n, target_sum, k_top, lr, n_iters, momentum=0.9):
    """Momentum-based top-k subgradient descent."""
    h = h0.copy()
    v = np.zeros(n)
    c5_cur = compute_c5(h)
    h_best = h.copy()
    
    for it in range(n_iters):
        g, _ = subgradient_topk(h, k_top)
        v = momentum * v - lr * g
        h_new = h + v
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur:
            c5_cur = c5_new
            h = h_new
            h_best = h.copy()
        else:
            # revert, reduce lr
            lr *= 0.9
            v *= 0.5
    return h_best, c5_cur

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_for_shift(h, s):
    n = len(h)
    g = np.zeros(n)
    if s == 0:
        g[:] = 1.0 - 2.0 * h
    elif s > 0:
        upper = n - 1 - s
        g[0:upper+1] += (1.0 - h[s:upper+1+s])
        g[s:n] -= h[0:upper+1]
    else:
        t = -s
        upper = n - 1 - t
        g[t:n] += (1.0 - h[0:upper+1])
        g[0:upper+1] -= h[t:upper+1+t]
    return g

def topk_subgrad(h, k_top):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[::-1][:k_top]
    grad = np.zeros(n)
    for m_idx in top_idx:
        s = m_idx - (n - 1)
        grad += grad_for_shift(h, s)
    grad /= k_top
    return grad, np.max(corr)

def momentum_descent(h0, n, target_sum, k_top, lr_init, n_iters, mom=0.85):
    h = h0.copy()
    v = np.zeros(n)
    c5_cur = compute_c5(h)
    h_best = h.copy()
    lr = lr_init
    for it in range(n_iters):
        g, _ = topk_subgrad(h, k_top)
        v = mom * v - lr * g
        h_new = h + v
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur - 1e-16:
            c5_cur = c5_new
            h = h_new
            h_best = h.copy()
            lr = min(lr * 1.005, 0.005)
        else:
            lr *= 0.92
            v *= 0.5
            if lr < 1e-10:
                break
    return h_best, c5_cur

def slsqp_smooth(h0, n, target_sum, alpha, maxiter):
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode='full')
        m = np.max(corr)
        shifted = alpha * (corr - m)
        shifted = np.clip(shifted, -700.0, 700.0)
        exp_vals = np.exp(shifted)
        s = np.sum(exp_vals)
        return (m + np.log(s / len(exp_vals)) / alpha)
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode='full')
        m = np.max(corr)
        shifted = alpha * (corr - m)
        shifted = np.clip(shifted, -700.0, 700.0)
        exp_vals = np.exp(shifted)
        w = exp_vals / np.sum(exp_vals)
        g = np.zeros(n)
        for m_idx in range(len(corr)):
            wi = w[m_idx]
            if wi < 1e-15:
                continue
            s = m_idx - (n - 1)
            g += wi * grad_for_shift(h, s)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def coord_descent_focused(h0, n, target_sum, max_rounds, rng):
    h = h0.copy()
    c5_cur = compute_c5(h)
    deltas = [0.001, -0.001, 0.003, -0.003, 0.005, -0.005, 0.01, -0.01]
    for rnd in range(max_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            best_val = c5_cur
            best_d = 0.0
            for d in deltas:
                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] + d, 0.0, 1.0)
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_val - 1e-15:
                    best_val = c5_new
                    best_d = d
            if best_d != 0.0:
                h[i] = np.clip(h[i] + best_d, 0.0, 1.0)
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True
        if not improved:
            break
    return h, c5_cur

def shift_aware_cd(h0, n, target_sum, max_rounds, rng, k_focus=3):
    h = h0.copy()
    c5_cur = compute_c5(h)
    deltas = [0.0005, -0.0005, 0.001, -0.001, 0.002, -0.002, 0.005, -0.005]
    for rnd in range(max_rounds):
        corr = compute_corr(h)
        top_idx = np.argsort(corr)[::-1][:k_focus]
        active = set()
        for m_idx in top_idx:
            s = m_idx - (n - 1)
            if s >= 0:
                active.update(range(0, n - s))
            else:
                t = -s
                active.update(range(t, n))
        active_list = list(active)
        rng.shuffle(active_list)
        improved = False
        for i in active_list:
            best_val = c5_cur
            best_d = 0.0
            for d in deltas:
                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] + d, 0.0, 1.0)
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_val - 1e-15:
                    best_val = c5_new
                    best_d = d
            if best_d != 0.0:
                h[i] = np.clip(h[i] + best_d, 0.0, 1.0)
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True
        if not improved:
            break
    return h, c5_cur
