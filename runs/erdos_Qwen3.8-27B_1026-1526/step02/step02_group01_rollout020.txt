            g[j] -= np.dot(w[n - 1 - ls2], h[j - ls2])
    return sm_val, g

def slsqp_step(h0, n, target_sum, alpha, maxiter):
    def obj(x):
        ht = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_and_grad(ht, alpha)
        return v
    def grad(x):
        ht = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_and_grad(ht, alpha)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-13})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def coord_descent_ls(h0, n, target_sum, n_rounds, rng):
    h = h0.copy()
    c5_cur = compute_c5(h)
    for rnd in range(n_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            lo = max(0.0, h[i] - 0.5)
            hi = min(1.0, h[i] + 0.5)
            if hi - lo < 1e-6:
                continue
            def f1d(v):
                h_tmp = h.copy()
                h_tmp[i] = v
                h_tmp = project_to_sum(h_tmp, target_sum)
                return compute_c5(h_tmp)
            try:
                res = minimize_scalar(f1d, bounds=(lo, hi), method='bounded',
                                     options={'xatol': 1e-5})
                if res.fun < c5_cur - 1e-12:
                    h[i] = res.x
                    h = project_to_sum(h, target_sum)
                    c5_cur = res.fun
                    improved = True
            except Exception:
                pass
        if not improved:
            break
    return h, c5_cur

def coord_descent_fixed(h0, n, target_sum, n_rounds, rng, deltas):
    h = h0.copy()
    c5_cur = compute_c5(h)
    for rnd in range(n_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            best_val = c5_cur
            best_d = 0.0
            for d in deltas:
                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] + d, 0.0, 1.0)
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_val - 1e-13:
                    best_val = c5_new
                    best_d = d
            if best_d != 0.0:
                h[i] = np.clip(h[i] + best_d, 0.0, 1.0)
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True
        if not improved:
            break
    return h, c5_cur

def nesterov_gradient(h0, n, target_sum, alpha, n_iters, lr_init):
    h = h0.copy()
    y = h.copy()
    t_val = 1.0
    lr = lr_init
    h_best = h.copy()
    c5_cur = compute_c5(h)
    c5_best = c5_cur
    for it in range(n_iters):
        _, g = smooth_max_and_grad(y, alpha)
        h_new = np.clip(y - lr * g, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new > c5_cur * 1.01:
            lr *= 0.5
            continue
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        mo = (t_val - 1.0) / t_new
        y = h_new + mo * (h_new - h)
        y = np.clip(y, 0.0, 1.0)
        h = h_new
        t_val = t_new
        c5_cur = c5_new
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new.copy()
            lr = min(lr * 1.03, 0.3)
        else:
            lr *= 0.9
    return h_best, c5_best

def smooth_step(h, n, width):
    kernel = np.zeros(width)
    for i in range(width):
        kernel[i] = np.exp(-0.5 * ((i - width // 2) / max(1, width // 4)) ** 2)
    kernel /= np.sum(kernel)
    h_pad = np.concatenate([h[-width//2:], h, h[:width//2]])
    h_smooth = np.convolve(h_pad, kernel, mode='valid')
    return project_to_sum(h_smooth, n // 2.0)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    def time_left():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            ts = best_n / 2.0
            best_h = project_to_sum(best_h, ts)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 360
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    # Phase 1: Optimize at original resolution
    n1 = best_n
    ts1 = n1 / 2.0
    h_cur = best_h.copy()

    alphas_p1 = [5.0, 10.0, 20.0, 40.0, 80.0, 160.0, 320.0, 640.0, 1280.0, 2560.0]
    for alpha in alphas_p1:
        if time_left() < 60:
            break
        maxiter = max(50, min(500, int(time_left() * 0.8)))
        h_opt = slsqp_step(h_cur, n1, ts1, alpha, maxiter)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        h_cur = h_opt

    # Coordinate descent with line search
    if time_left() > 120:
        n_cd = min(20, max(5, int(time_left() // 10)))
        h_cd, c5_cd = coord_descent_ls(h_cur, n1, ts1, n_cd, rng)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            h_cur = h_cd

    # Fixed-step coordinate descent
    if time_left() > 60:
        deltas = [-0.02, -0.01, -0.005, -0.002, 0.002, 0.005, 0.01, 0.02]
        h_cd2, c5_cd2 = coord_descent_fixed(h_cur, n1, ts1, 30, rng, deltas)
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2.copy()
            h_cur = h_cd2

    # Nesterov gradient
    if time_left() > 80:
        n_pg = min(2000, int(time_left() * 2))
        h_pg, c5_pg = nesterov_gradient(h_cur, n1, ts1, 200.0, n_pg, 0.1)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()
            h_cur = h_pg

    # Phase 2: Try higher resolution
    n2 = n1 * 2
    if time_left() > 180 and n2 <= 1440:
        ts2 = n2 / 2.0
        x_old = np.linspace(0, 2, n1, endpoint=False)
        x_new = np.linspace(0, 2, n2, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h, period=2)
        h_interp = project_to_sum(h_interp, ts2)
        h_cur2 = h_interp

        alphas_p2 = [50.0, 100.0, 200.0, 400.0, 800.0, 1600.0]
        for alpha in alphas_p2:
            if time_left() < 60:
                break
            maxiter = max(30, min(300, int(time_left() * 0.6)))
            h_opt2 = slsqp_step(h_cur2, n2, ts2, alpha, maxiter)
            c5_opt2 = compute_c5(h_opt2)
            if c5_opt2 < best_c5:
                best_c5 = c5_opt2
                best_h = h_opt2.copy()
                best_n = n2
            h_cur2 = h_opt2

        if time_left() > 80:
            n_cd2 = min(10, max(3, int(time_left() // 12)))
            h_cd3, c5_cd3 = coord_descent_ls(h_cur2, n2, ts2, n_cd2, rng)
            if c5_cd3 < best_c5:
                best_c5 = c5_cd3
                best_h = h_cd3.copy()
                best_n = n2
                h_cur2 = h_cd3

        if time_left() > 60:
            h_cd4, c5_cd4 = coord_descent_fixed(h_cur2, n2, ts2, 20, rng,
                                                [-0.01, -0.005, -0.002, 0.002, 0.005, 0.01])
            if c5_cd4 < best_c5:
                best_c5 = c5_cd4
                best_h = h_cd4.copy()
                best_n = n2
                h_cur2 = h_cd4

        if time_left() > 40:
            n_pg2 = min(1500, int(time_left() * 1.5))
            h_pg2, c5_pg2 = nesterov_gradient(h_cur2, n2, ts2, 500.0, n_pg2, 0.05)
            if c5_pg2 < best_c5:
                best_c5 = c5_pg2
                best_h = h_pg2.copy()
                best_n = n2

    # Phase 3: Perturbation and re-optimization at best resolution
    if time_left() > 100:
        n_f = best_n
        ts_f = n_f / 2.0
        for pert_round in range(5):
            if time_left() < 30:
                break
            perturbation = 0.05 / (pert_round + 1)
            h_pert = best_h + rng.normal(0, perturbation, size=n_f)
            h_pert = project_to_sum(h_pert, ts_f)
            c5_pert = compute_c5(h_pert)

            if c5_pert < best_c5:
                best_c5 = c5_pert
                best_h = h_pert.copy()

            h_work = h_pert if c5_pert < best_c5 else best_h.copy()
            for alpha in [200.0, 500.0, 2000.0]:
                if time_left() < 25:
                    break
                h_opt = slsqp_step(h_work, n_f, ts_f, alpha, 200)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    h_work = h_opt
                else:
                    h_work = h_opt

            if time_left() > 30:
                h_cd, c5_cd = coord_descent_fixed(h_work, n_f, ts_f, 15, rng,
                                                  [-0.005, -0.002, -0.001, 0.001, 0.002, 0.005])
                if c5_cd < best_c5:
                    best_c5 = c5_cd
                    best_h = h_cd.copy()

    # Phase 4: Final polish
    if best_h is not None and time_left() > 20:
        n_p = len(best_h)
        ts_p = n_p / 2.0
        for alpha in [1000.0, 5000.0]:
            if time_left() < 15:
                break
            h_pol = slsqp_step(best_h, n_p, ts_p, alpha, 400)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol

        if time_left() > 15:
            h_cd, c5_cd = coord_descent_ls(best_h, n_p, ts_p, 5, rng)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd

    final_n = len(best_h)
    ts_final = final_n / 2.0
    best_h = project_to_sum(best_h, ts_final)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, final_n)
