```python
    n = len(h)
    c = np.correlate(h, 1-h, mode='full')
    c_max = np.max(c)
    # Get top-k indices
    top_k_indices = np.argsort(c)[-k_top:]
    # Smooth max over top-k
    top_k_c = c[top_k_indices]
    top_k_max = np.max(top_k_c)
    exp_vals = np.exp(alpha * (top_k_c - top_k_max))
    w_topk = exp_vals / np.sum(exp_vals)
    obj = top_k_max + np.log(np.sum(exp_vals)) / alpha
    
    # Gradient: only contributions from top-k lags
    w = np.zeros(2*n-1)
    w[top_k_indices] = w_topk
    
    grad = np.zeros(n)
    for j in range(n):
        # First sum
        first = np.dot(w[n-1:2*n-1-j], 1-h[j:n])
        # Second sum  
        second = np.dot(w[n-1:n-1+j+1][::-1], h[0:j+1])
        grad[j] = first - second
    
    return obj, grad
```

```python
        second = np.dot(w_pos[:j+1], h[j::-1])
        grad[j] = first - second
```

```python
    sm_val, sm_grad = smooth_max_obj_grad(h, alpha, k_top)
    sum_diff = np.sum(h) - target_sum
    obj = sm_val + penalty * sum_diff * sum_diff
    grad = sm_grad + 2.0 * penalty * sum_diff
```

```python
        first = np.dot(w_pos[:n-j], 1.0 - h[j:n])
        second = np.dot(w_pos[:j+1], h[j::-1])
        grad[j] = first - second
    return obj, grad

def obj_pen(x, alpha, penalty, target_sum):
    sm_val, _ = smooth_max_obj_grad(x, alpha)
    sd = np.sum(x) - target_sum
    return sm_val + penalty * sd * sd

def grad_pen(x, alpha, penalty, target_sum):
    _, sm_grad = smooth_max_obj_grad(x, alpha)
    sd = np.sum(x) - target_sum
    return sm_grad + 2.0 * penalty * sd

def upsample(h_old, n_new):

    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)
```
