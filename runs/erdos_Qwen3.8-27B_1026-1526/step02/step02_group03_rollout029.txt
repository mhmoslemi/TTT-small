    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def grad_overlap(h, k_idx):
    """Gradient of the overlap at a specific correlate index k_idx."""
    n = len(h)
    dx = 2.0 / n
    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    i_min = max(0, -lag)
    i_max = min(n-1, n-1-lag)
    if i_min > i_max:
        return grad
    # C = sum_{i=i_min}^{i_max} h[i] * (1 - h[i+lag])
    # dC/dh[i] = (1-h[i+lag]) for i in [i_min, i_max]
    # dC/dh[i+lag] = -h[i] for i in [i_min, i_max]
    idx1 = np.arange(i_min, i_max + 1)
    idx2 = idx1 + lag
    grad[idx1] += (1.0 - h[idx2])
    grad[idx2] += (-h[idx1])
    return grad * dx

def active_set_gradient(h, top_m=5, eps=0.01):
    """Compute weighted gradient of top-m active overlaps."""
    overlaps = compute_overlaps(h)
    cmax = np.max(overlaps)
    weights = np.maximum(overlaps - cmax + eps, 0) ** 3  # soft weighting
    if weights.sum() < 1e-15:
        # Just use the argmax
        return grad_overlap(h, np.argmax(overlaps)), cmax
    weights = weights / weights.sum()
    # Get top m indices
    top_idx = np.argsort(overlaps)[-top_m:]
    # Recompute weights only for top m
    w_top = np.maximum(overlaps[top_idx] - cmax + eps, 0) ** 3
    if w_top.sum() < 1e-15:
        w_top = np.ones(len(top_idx)) / len(top_idx)
    else:
        w_top = w_top / w_top.sum()

    for idx, w in zip(top_idx, w_top):
        grad += w * grad_overlap(h, idx)

            diff -= adj

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def grad_single_lag(h, lag_idx):
    n = len(h)
    dx = 2.0 / n
    lag = lag_idx - (n - 1)
    grad = np.zeros(n)
    i_min = max(0, -lag)
    i_max = min(n - 1, n - 1 - abs(lag))
    if i_min > i_max:
        return grad
    idx1 = np.arange(i_min, i_max + 1)
    if lag >= 0:
        idx2 = idx1 + lag
    else:
        idx2 = idx1 - lag
    grad[idx1] += (1.0 - h[idx2])
    grad[idx2] += (-h[idx1])
    return grad * dx

def top_lag_gradient(h, m=5):
    overlaps = compute_overlaps(h)
    cmax = np.max(overlaps)
    top_idx = np.argsort(overlaps)[-m:]
    weights = np.maximum(overlaps[top_idx] - cmax + 1e-4, 0) ** 2
    if weights.sum() < 1e-15:
        weights = np.ones(m) / m
    else:
        weights = weights / weights.sum()
    grad = np.zeros(len(h))
    for idx, w in zip(top_idx, weights):
        grad += w * grad_single_lag(h, idx)
    return grad, cmax, overlaps

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        if np.sum(proj) > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-13:
        sort_idx = np.argsort(np.abs(result - 0.5))
        for idx in sort_idx:
            if diff > 0:
                adj = min(diff, 1.0 - result[idx])
            else:
                adj = max(diff, -result[idx])
            result[idx] += adj
            diff -= adj
            if abs(diff) < 1e-13:
                break
    return result

def smooth_max_fn(values, tau):
    v = values / tau
    vmax = np.max(v)
    return tau * (vmax + np.log(np.sum(np.exp(v - vmax))))

def obj_smooth(x, n, tau):
    over = compute_overlaps(x)
    return smooth_max_fn(over, tau)

def con_sum(x, n):
    return np.sum(x) - n / 2.0

def slsqp_optimize(h0, n, tau, max_iter=200):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h0, method='SLSQP', args=(n, tau),
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h0
