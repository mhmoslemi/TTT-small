def compute_c_and_grad(x, T, penalty):
    """Compute log-sum-exp objective and its gradient."""
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full") * dx
    c_max = c.max()
    shifted = T * (c - c_max)
    shifted = np.clip(shifted, -300, 300)
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / exp_shifted.sum()
    
    lse_val = c_max + np.log(exp_shifted.sum()) / T
    
    # Gradient: for each shift i, the gradient of c[i] w.r.t. x
    # c[i] = sum_j x[j] * (1-x[j+p_i]) * dx where p_i = i - (n-1)
    # ∂c[i]/∂x[m] = dx * [(1-x[m+p_i])*I(valid) - x[m-p_i]*I(valid)]
    
    # Vectorized approach:
    # For shift p, the gradient with respect to x is:
    # g[m] = (1 - x[m+p]) - x[m-p] (with appropriate boundary handling)
    # This is like: g = (1 - shift(x, p)) - shift(x, -p)
    
    grad = np.zeros_like(x)
    p_values = np.arange(2*n-1) - (n-1)  # shifts from -(n-1) to (n-1)
    
    for i, p in enumerate(p_values):
        w = weights[i] / T
        if w < 1e-1:
            continue
        # ∂c[i]/∂x[m]:
        # Term 1: x[m] appears as first factor when j=m, need m+p in [0,n)
        #   contributes (1-x[m+p]) for m in [max(0,-p), min(n, n-p))
        # Term 2: x[m] appears as second factor (inside 1-x[...]) when m = j+p, j = m-p
        #   contributes -x[m-p] for m in [max(0,p), min(n, n+p))... 
        #   wait, j = m-p must be in [0,n) and also in the valid range for this shift
        
        # Let me think again. c[i] = sum_j x[j]*(1-x[j+p]) * dx
        # Valid j: j in [0,n) AND j+p in [0,n), so j in [max(0,-p), min(n, n-p))
        
        # ∂c[i]/∂x[m]:
        # From first factor x[j] where j=m: (1-x[m+p]) * dx, valid if m in [max(0,-p), min(n,n-p)) and m+p in [0,n)
        # From second factor (1-x[j+p]) where j+p=m, i.e., j=m-p: -x[m-p] * dx, valid if m-p in [max(0,-p), min(n,n-p)) and m in [0,n)
        
        lo_j = max(0, -p)
        hi_j = min(n, n - p)
        
        # Term 1: m in [lo_j, hi_j) and m+p in [0, n)
        m1_lo = max(lo_j, -p)  # m >= -p, but since lo_j >= 0 already, m1_lo = lo_j
        m1_hi = min(hi_j, n - p)  # m+p < n, so m < n-p
        # Actually m1_lo = lo_j = max(0, -p), m1_hi = min(n, n-p) = hi_j
        # So m1 range is [lo_j, hi_j), same as j range
        if m1_lo < m1_hi:
            grad[m1_lo:m1_hi] += w * dx * (1.0 - x[m1_lo + p:m1_hi + p])
        
        # Term 2: j = m-p in [lo_j, hi_j), so m in [lo_j+p, hi_j+p), and m in [0,n)
        m2_lo = max(lo_j + p, 0)
        m2_hi = min(hi_j + p, n)
        if m2_lo < m2_hi:
            grad[m2_lo:m2_hi] += w * dx * (-x[m2_lo - p:m2_hi - p])
    
    # Add penalty gradient
    sv = np.sum(x) - n / 2.0
    grad += 2.0 * penalty * sv
    
    return lse_val + penalty * sv * sv, grad

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time


def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    lo, hi = -5.0, 5.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(h_proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def lse_value_and_grad(x, T, penalty):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full") * dx
    c_max = c.max()
    shifted = T * (c - c_max)
    shifted = np.clip(shifted, -300.0, 300.0)
    exp_s = np.exp(shifted)
    exp_sum = exp_s.sum()
    weights = exp_s / exp_sum
    lse_val = c_max + np.log(exp_sum) / T
    sv = np.sum(x) - n / 2.0
    obj = lse_val + penalty * sv * sv
    grad = np.zeros_like(x)
    p_values = np.arange(2 * n - 1) - (n - 1)
    thresh = 1e-4
    for i in range(len(c)):
        w = weights[i] / T
        if w < thresh:
            continue
        p = p_values[i]
        lo_j = max(0, -p)
        hi_j = min(n, n - p)
        m1_lo = lo_j
        m1_hi = hi_j
        if m1_lo < m1_hi:
            grad[m1_lo:m1_hi] += w * dx * (1.0 - x[m1_lo + p:m1_hi + p])
        m2_lo = max(lo_j + p, 0)
        m2_hi = min(hi_j + p, n)
        if m2_lo < m2_hi:
            grad[m2_lo:m2_hi] += w * dx * (-x[m2_lo - p:m2_hi - p])
    grad += 2.0 * penalty * sv
    return obj, grad


def lbfgs_solve(h_init, T, penalty, maxiter=2000):
    n = len(h_init)
    bounds = [(0.0, 1.0)] * n

    def obj_func(x):
        val, _ = lse_value_and_grad(x, T, penalty)
        return val

    def grad_func(x):
        _, g = lse_value_and_grad(x, T, penalty)
        return g

    result = minimize(obj_func, h_init, jac=grad_func,
                      method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-13,
                               'maxls': 50})
    if result.x is not None:
        h_out = project_feasible(result.x, n)
        return h_out, compute_c5_from_h(h_out)
    return h_init, compute_c5_from_h(h_init)


def slsqp_t_obj(x):
    return x[-1]


def slsqp_ineq(x):
    h = x[:-1]
    t = x[-1]
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return t - c * dx


def slsqp_eq(x):
    h = x[:-1]
    return np.sum(h) - len(h) / 2.0


def slsqp_eq_jac(x):
    n = len(x) - 1
    jac = np.zeros(len(x))
    jac[:n] = 1.0
    return jac


def slsqp_solve(h_init, maxiter=5000):
    c5_init = compute_c5_from_h(h_init)
    n = len(h_init)
    x0 = np.concatenate([h_init, [c5_init * 1.05 + 0.03]])
    bounds = [(0.0, 1.0)] * n + [(0.01, 2.0)]
    constraints = [
        {'type': 'ineq', 'fun': slsqp_ineq},
        {'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac}
    ]
    result = minimize(slsqp_t_obj, x0, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
    if result.x is not None:
        h_out = project_feasible(result.x[:-1], n)
        return h_out, compute_c5_from_h(h_out)
    return h_init, c5_init


def coarse_to_fine(target_n, t0, budget_s, rng):
    current_h = np.full(30, 0.5)
    for n_target in [30, 50, 80, 130, target_n]:
        if n_target != 30:
            n_old = len(current_h)
            x_old = np.linspace(0, 2, n_old, endpoint=False)
            x_new = np.linspace(0, 2, n_target, endpoint=False)
            f_interp = interp1d(x_old, current_h, kind='linear', fill_value='extrapolate')
            current_h = np.clip(f_interp(x_new), 0, 1)
            current_h = project_feasible(current_h, n_target)
        elapsed = time.time() - t0
        remaining = budget_s - elapsed - 30
        if remaining < 50:
            break
        maxiter = max(300, min(4000, int(remaining * 1.5)))
        h_new, c5_new = lbfgs_solve(current_h, T=100.0, penalty=200.0, maxiter=maxiter)
        if compute_c5_from_h(h_new) < compute_c5_from_h(current_h):
            current_h = h_new
        if time.time() - t0 > budget_s - 40:
            break
        maxiter2 = max(200, min(3000, int(remaining * 0.8)))
        h_new2, c5_new2 = slsqp_solve(current_h, maxiter=maxiter2)
        if c5_new2 < compute_c5_from_h(current_h):
            current_h = h_new2
    return current_h


def directed_perturb(h, rng, scale, n_target_shifts=10):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    c_max = c.max()
    worst = np.where(c > c_max - 0.02 * c_max)[0]
    h_pert = h + scale * rng.standard_normal(n)
    for idx in worst:
        p = int(idx - (n - 1))
        lo_j = max(0, -p)
        hi_j = min(n, n - p)
        if p > 0:
            h_pert[lo_j:hi_j] -= scale * 0.3
        elif p < 0:
            h_pert[lo_j:hi_j] += scale * 0.3
    return project_feasible(h_pert, n)


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(270, 0.5)

    n_best = len(best_h)
    best_h = project_feasible(best_h, n_best)
    best_c5 = compute_c5_from_h(best_h)

    if time.time() - t0 < budget_s - 200:
        for T in [30, 80, 200, 500, 1000]:
            if time.time() - t0 > budget_s - 80:
                break
            h_new, c5_new = lbfgs_solve(best_h, T=T, penalty=float(T), maxiter=2000)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
                n_best = len(best_h)

    if time.time() - t0 < budget_s - 150:
        h_new, c5_new = slsqp_solve(best_h, maxiter=6000)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
            n_best = len(best_h)

    if time.time() - t0 < budget_s - 200:
        h_c2f = coarse_to_fine(n_best, t0, budget_s, rng)
        if len(h_c2f) == n_best:
            c5_c2f = compute_c5_from_h(h_c2f)
            if c5_c2f < best_c5:
                best_h = h_c2f
                best_c5 = c5_c2f

    n_pert = 70
    if time.time() - t0 < budget_s - 60:
        for attempt in range(n_pert):
            if time.time() - t0 > budget_s - 35:
                break
            frac = attempt / max(1, n_pert)
            scale = 0.07 * (1.0 - frac) ** 0.7 + 0.004
            r = rng.random()
            if r < 0.4:
                h_pert = directed_perturb(best_h, rng, scale)
            elif r < 0.7:
                h_pert = best_h + scale * rng.standard_normal(n_best)
                h_pert = project_feasible(h_pert, n_best)
            else:
                h_pert = best_h.copy()
                n_blocks = rng.integers(2, max(3, n_best // 15))
                for _ in range(n_blocks):
                    bsize = rng.integers(4, max(5, n_best // 8))
                    start = rng.integers(0, max(1, n_best - bsize))
                    h_pert[start:start + bsize] += scale * (rng.random(bsize) - 0.5) * 2.0
                h_pert = project_feasible(h_pert, n_best)

            if attempt % 3 == 0:
                h_new, c5_new = slsqp_solve(h_pert, maxiter=2500)
            elif attempt % 3 == 1:
                h_tmp, _ = lbfgs_solve(h_pert, T=150.0, penalty=300.0, maxiter=1000)
                h_new, c5_new = slsqp_solve(h_tmp, maxiter=2000)
            else:
                h_tmp, _ = lbfgs_solve(h_pert, T=50.0, penalty=100.0, maxiter=1500)
                h_new, c5_new = slsqp_solve(h_tmp, maxiter=2500)

            if c5_new < best_c5 - 1e-12:
                best_h = h_new
                best_c5 = c5_new
                n_best = len(best_h)

    for n_try in [200, 250, 300, 350, 400]:
        if time.time() - t0 > budget_s - 120:
            break
        x_old = np.linspace(0, 2, n_best, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        f_interp = interp1d(x_old, best_h, kind='linear', fill_value='extrapolate')
        h_interp = np.clip(f_interp(x_new), 0, 1)
        h_interp = project_feasible(h_interp, n_try)
        h_new, c5_new = lbfgs_solve(h_interp, T=200.0, penalty=400.0, maxiter=2000)
        if c5_new < compute_c5_from_h(h_interp):
            h_interp = h_new
        h_new2, c5_new2 = slsqp_solve(h_interp, maxiter=4000)
        if c5_new2 < best_c5 - 1e-10:
            best_h = h_new2
            best_c5 = c5_new2
            n_best = n_try

    if time.time() - t0 < budget_s - 20:
        for attempt in range(30):
            if time.time() - t0 > budget_s - 8:
                break
            scale = 0.001 + 0.004 * (1.0 - attempt / 30.0) ** 2
            h_pert = best_h + scale * rng.standard_normal(n_best)
            h_pert = project_feasible(h_pert, n_best)
            h_new, c5_new = slsqp_solve(h_pert, maxiter=2500)
            if c5_new < best_c5 - 1e-13:
                best_h = h_new
                best_c5 = c5_new
                n_best = len(best_h)

    if time.time() - t0 < budget_s - 5:
        maxiter_final = max(1000, int((budget_s - (time.time() - t0) - 3) * 2))
        best_h, best_c5 = slsqp_solve(best_h, maxiter=maxiter_final)

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5_from_h(best_h)
    return best_h, best_c5, len(best_h)
