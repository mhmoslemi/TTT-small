def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)

        if np.sum(proj) > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-12:

                if 0.0 <= new_val <= 1.0:

                    best = project_feasible(best, n)
                    c5 = evaluate_c5(best)
                    if c5 < best_c5 - 1e-16:
                        best_c5 = c5
                        improved = True
                    else:
                        best[i] = old
                        best = project_feasible(best, n)
    return best, best_c5

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -3.0, 3.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        if np.sum(proj) > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-12:
        idx = np.argsort(np.abs(result - 0.5))
        for i in idx:
            if diff > 0:
                add = min(diff, 1.0 - result[i])
                result[i] += add
                diff -= add
            else:
                sub = min(-diff, result[i])
                result[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return result

def smooth_max_obj(x, n, tau):
    dx = 2.0 / n
    g = 1.0 - x
    corr = np.correlate(x, g, mode="full")
    v = corr / tau
    vmax = np.max(v)
    smax = tau * (vmax + np.log(np.sum(np.exp(v - vmax))))
    return smax * dx

def sum_constraint(x, n):
    return np.sum(x) - n / 2.0

def sum_constraint_jac(x, n):
    return np.ones(n)

def slqp_refine(h0, n, tau_list, max_iter_each):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac, 'args': (n,)}]
    current = project_feasible(h0.copy(), n)
    for tau in tau_list:
        try:
            res = minimize(
                smooth_max_obj, current,
                method='SLSQP', args=(n, tau),
                bounds=bounds, constraints=cons,
                options={'maxiter': max_iter_each, 'ftol': 1e-16}
            )
            current = project_feasible(res.x, n)
        except Exception:
            pass
    return current, evaluate_c5(current)

def coord_descent(h, n, deadline):
    best = h.copy()
    best_c5 = evaluate_c5(best)
    steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    for step in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            if time.time() > deadline:
                break
            improved = False
            for i in range(n):
                for sign in (1, -1):
                    if time.time() > deadline:
                        break
                    h_new = best.copy()
                    h_new[i] += sign * step
                    h_new = project_feasible(h_new, n)
                    c5 = evaluate_c5(h_new)
                    if c5 < best_c5 - 1e-16:
                        best = h_new
                        best_c5 = c5
                        improved = True
                        break
    return best, best_c5

def pairwise_descent(h, n, deadline):
    best = h.copy()
    best_c5 = evaluate_c5(best)
    steps = [0.005, 0.002, 0.001, 0.0005]
    pairs = []
    for i in range(n):
        for j in range(i + 1, min(i + 15, n)):
            pairs.append((i, j))
    for lag in [1, 2, 3, 5, 8, 13, 20, 27, 30, 38]:
        for i in range(n - lag):
            pairs.append((i, i + lag))
    seen = set()
    unique_pairs = []
    for p in pairs:
        if p not in seen:
            seen.add(p)
            unique_pairs.append(p)
    pairs = unique_pairs
    improved = True
    while improved and time.time() < deadline:
        improved = False
        for (i, j) in pairs:
            for step in steps:
                for d in (1, -1):
                    if time.time() > deadline:
                        return best, best_c5
                    h_new = best.copy()
                    h_new[i] += d * step
                    h_new[j] -= d * step
                    h_new = project_feasible(h_new, n)
                    c5 = evaluate_c5(h_new)
                    if c5 < best_c5 - 1e-16:
                        best = h_new
                        best_c5 = c5
                        improved = True
                        break
                else:
                    continue
                break
    return best, best_c5

def active_lag_refine(h, n, deadline, max_rounds=50):
    best = h.copy()
    best_c5 = evaluate_c5(best)
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        dx = 2.0 / n
        g = 1.0 - best
        corr = np.correlate(best, g, mode="full")
        active = np.argmax(corr)
        def single_obj(x):
            gx = 1.0 - x
            c = np.correlate(x, gx, mode="full")
            return c[active] * dx
        try:
            res = minimize(
                single_obj, best,
                method='SLSQP',
                bounds=[(0.0, 1.0)] * n,
                constraints=[{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac, 'args': (n,)}],
                options={'maxiter': 200, 'ftol': 1e-14}
            )
            new_h = project_feasible(res.x, n)
            new_c5 = evaluate_c5(new_h)
            if new_c5 < best_c5 - 1e-16:
                best = new_h
                best_c5 = new_c5
            else:
                break
        except Exception:
            break
    return best, best_c5
