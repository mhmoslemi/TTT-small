import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def fft_correlation(h, w_arr, n):
    """Compute A[j] = sum_l w[l] * h[j+l] and B[j] = sum_l w[l] * h[j-l] for j=0..n-1
    where w[l] is w_arr[l + n - 1] for l = -(n-1)..(n-1)
    Returns (A, B) each of length n."""
    # h_ext: length 3n-2, with h at positions n-1..2n-2
    h_ext = np.zeros(3 * n - 2)
    h_ext[n-1:2*n-1] = h
    
    w_rev = w_arr[::-1]
    
    # Use FFT for convolution
    size = 2 ** int(np.ceil(np.log2(len(w_arr) + len(h_ext) - 1)))
    H_ext = np.fft.rfft(h_ext, size)
    W_arr = np.fft.rfft(w_arr, size)
    W_rev = np.fft.rfft(w_rev, size)
    
    conv_A = np.fft.irfft(W_rev * H_ext, size)[:3*n-2]
    conv_B = np.fft.irfft(W_arr * H_ext, size)[:3*n-2]
    
    # A[j] = conv_A[j + 2n - 2], B[j] = conv_B[j + 2n - 2]
    A = conv_A[2*n-2 : 3*n-2]
    B = conv_B[2*n-2 : 3*n-2]
    return A, B

def smooth_max_fft(h, alpha, n):
    """Compute smooth max surrogate and gradient using FFT."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    corr_scaled = corr * dx  # length 2n-1
    m = np.max(corr_scaled)
    
    shifted = alpha * (corr_scaled - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s  # softmax weights, length 2n-1
    sm_val = m + np.log(s / len(exp_vals)) / alpha
    
    # Gradient: grad[j] = dx * (1 - A[j] - B[j])
    A, B = fft_correlation(h, w, n)
    grad = dx * (1.0 - A - B)
    
    return sm_val, grad

def slsqp_optimize_fft(h0, n, target_sum, alpha, maxiter, time_limit):
    """SLSQP with FFT-based gradient."""
    t0 = time.time()
    
    def obj(x):
        if time.time() - t0 > time_limit:
            return np.inf
        h = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_fft(h, alpha, n)
        return v
    
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_fft(h, alpha, n)
        return g
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def projected_gradient_fft(h0, n, target_sum, alpha, n_iters, lr, time_limit):
    """FISTA with FFT-based gradient."""
    t0 = time.time()
    h = h0.copy()
    y = h.copy()
    t = 1.0
    c5_cur = compute_c5(h)
    h_best = h.copy()
    
    for it in range(n_iters):
        if time.time() - t0 > time_limit:
            break
        if it % 50 == 0 and alpha < 500:
            alpha *= 1.5
        _, g = smooth_max_fft(y, alpha, n)
        h_new = np.clip(y - lr * g, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        mo = (t - 1.0) / t_new
        y = h_new + mo * (h_new - h)
        y = np.clip(y, 0.0, 1.0)
        h = h_new
        t = t_new
        
        if it % 20 == 0:
            c5_new = compute_c5(h)
            if c5_new < c5_cur:
                c5_cur = c5_new
                h_best = h.copy()
                lr = min(lr * 1.1, 1.0)
            else:
                lr *= 0.7
    return h_best, c5_cur

def alternating_projection(h0, n, target_sum, alpha, n_iters, time_limit):
    """Alternating optimization: fix subset of variables, optimize others."""
    t0 = time.time()
    h = h0.copy()
    c5_cur = compute_c5(h)
    h_best = h.copy()
    dx = 2.0 / n
    
    for it in range(n_iters):
        if time.time() - t0 > time_limit:
            break
        
        # Compute current correlation
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        
        # Find the worst lag
        worst_k = np.argmax(corr)
        
        # Try to reduce the worst lag by adjusting h
        # For lag k, f(k) = sum_i h[i]*(1-h[i+k])
        # To reduce this, we want to decrease h[i] where h[i+k] is also high,
        # or increase h[i] where h[i+k] is low (but this increases other lags)
        
        # Simple approach: gradient step focused on worst lag
        lag = worst_k - (n - 1)
        # grad_i for the worst lag: (1-h[i+lag]) - h[i-lag]
        grad_worst = np.zeros(n)
        for i in range(n):
            ip = i + lag
            im = i - lag
            val = 0.0
            if 0 <= ip < n:
                val += (1.0 - h[ip])
            if 0 <= im < n:
                val -= h[im]
            grad_worst[i] = val
        
        # Step in negative gradient direction for worst lag
        lr = 0.01
        h_new = np.clip(h - lr * grad_worst, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        
        if c5_new < c5_cur:
            h = h_new
            c5_cur = c5_new
            h_best = h.copy()
    
    return h_best, c5_cur

def coordinate_descent_improved(h0, n, target_sum, max_rounds, rng, time_limit):
    """Improved coordinate descent with larger step sizes and better ordering."""
    t0 = time.time()
    h = h0.copy()
    c5_cur = compute_c5(h)
    h_best = h.copy()
    
    for rnd in range(max_rounds):
        if time.time() - t0 > time_limit:
            break
        improved = False
        # Sort by gradient magnitude (approximate)
        corr = np.correlate(h, 1.0 - h, mode='full')
        worst_lags = np.argsort(-np.abs(corr))[:n]
        idx_order = rng.permutation(n)
        
        for i in idx_order:
            if time.time() - t0 > time_limit:
                break
            # Try multiple step sizes
            best_val = c5_cur
            best_h_i = h[i]
            # Try values around current
            candidates = np.clip(h[i] + np.array([-0.05, -0.02, -0.01, -0.005, 0.005, 0.01, 0.02, 0.05]), 0, 1)
            candidates = np.unique(np.concatenate([[h[i]], candidates]))
            for hv in candidates:
                if abs(hv - h[i]) < 1e-10:

                h_new = h.copy()
                h_new[i] = hv
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_val - 1e-15:

                    best_h_i = hv
            if best_h_i != h[i]:
                h[i] = best_h_i
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True
    
    return h_best, c5_cur

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:

        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_grad_fft(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    corr_scaled = corr * dx
    m = np.max(corr_scaled)
    shifted = alpha * (corr_scaled - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / float(2*n - 1)) / alpha
    
    h_ext = np.zeros(3 * n - 2)
    h_ext[n-1:2*n-1] = h
    w_combined = w + w[::-1]
    
    fft_size = 1
    while fft_size < 5 * n - 2:
        fft_size *= 2
    
    H = np.fft.rfft(h_ext, fft_size)
    W = np.fft.rfft(w_combined, fft_size)
    combined = np.fft.irfft(W * H, fft_size)[2*n-2 : 3*n-2]
    
    grad = dx * (1.0 - combined)
    return sm_val, grad

def slsqp_run(h0, n, target_sum, alpha, maxiter, tlimit):
    t0 = time.time()
    def obj(x):
        if time.time() - t0 > tlimit:
            return np.inf
        h = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_grad_fft(h, alpha)
        return v
    def grad_fn(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_grad_fft(h, alpha)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad_fn, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def fista_run(h0, n, target_sum, alpha0, alpha_max, n_iters, lr, tlimit):

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0

        y = h_new + mo * (h_new - h)

        t = t_new

                h_best = h.copy()
                lr = min(lr * 1.05, 0.5)
            else:

    return h_best, c5_cur

def subgradient_averaging(h0, n, target_sum, n_iters, lr_init, tlimit):
    """Subgradient method on the non-smooth max with Polyak averaging."""
    t0 = time.time()
    h = h0.copy()
    h_avg = h.copy()
    c5_cur = compute_c5(h)
    h_best = h.copy()
    M = c5_cur  # running max of objective (for Polyak step size)
    
    for it in range(n_iters):
        if time.time() - t0 > tlimit:
            break
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        k_star = np.argmax(corr)
        lag = k_star - (n - 1)
        
        # Subgradient of f_{k*}(h) = sum_i h_i (1-h_{i+lag}) * dx
        sg = np.zeros(n)
        for i in range(n):
            ip = i + lag
            im = i - lag
            val = 0.0
            if 0 <= ip < n:
                val += (1.0 - h[ip])
            if 0 <= im < n:
                val -= h[im]
            sg[i] = val * dx
        
        lr = lr_init / (1.0 + 0.01 * it)
        h_new = np.clip(h - lr * sg, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        h = h_new
        h_avg = (it * h_avg + h) / (it + 1)
        
        if it % 20 == 0:
            c5_new = compute_c5(h)

            if c5_avg < c5_cur:

    h_best = h.copy()
    c5_cur = compute_c5(h)

        improved = False

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_grad_fft(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    corr_scaled = corr * dx
    m = np.max(corr_scaled)
    shifted = alpha * (corr_scaled - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / float(2 * n - 1)) / alpha
    h_ext = np.zeros(3 * n - 2)
    h_ext[n - 1:2 * n - 1] = h
    w_combined = w + w[::-1]
    fft_size = 1
    while fft_size < 5 * n - 2:
        fft_size *= 2
    H = np.fft.rfft(h_ext, fft_size)
    W = np.fft.rfft(w_combined, fft_size)
    combined = np.fft.irfft(W * H, fft_size)[2 * n - 2:3 * n - 2]
    grad = dx * (1.0 - combined)
    return sm_val, grad

def slsqp_run(h0, n, target_sum, alpha, maxiter, tlimit):
    t0 = time.time()
    def obj(x):
        if time.time() - t0 > tlimit:
            return np.inf
        h = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_grad_fft(h, alpha)
        return v
    def grad_fn(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_grad_fft(h, alpha)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad_fn, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def fista_run(h0, n, target_sum, alpha0, alpha_max, n_iters, lr, tlimit):
    t0 = time.time()
    h = h0.copy()
    y = h.copy()
    t = 1.0
    alpha = alpha0
    c5_cur = compute_c5(h)
    h_best = h.copy()
    for it in range(n_iters):
        if time.time() - t0 > tlimit:
            break
        if it % 80 == 0 and alpha < alpha_max:
            alpha *= 1.4
        _, g = smooth_max_grad_fft(y, alpha)
        h_new = np.clip(y - lr * g, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        mo = (t - 1.0) / t_new
        y = h_new + mo * (h_new - h)
        y = np.clip(y, 0.0, 1.0)
        h = h_new
        t = t_new
        if it % 40 == 0:
            c5_new = compute_c5(h)
            if c5_new < c5_cur:
                c5_cur = c5_new
                h_best = h.copy()
                lr = min(lr * 1.05, 0.8)
            else:
                lr *= 0.75
    return h_best, c5_cur

def subgradient_averaging(h0, n, target_sum, n_iters, lr_init, tlimit):
    t0 = time.time()
    h = h0.copy()
    h_avg = h.copy()
    c5_cur = compute_c5(h)
    h_best = h.copy()
    dx = 2.0 / n
    for it in range(n_iters):
        if time.time() - t0 > tlimit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        k_star = np.argmax(corr)
        lag = k_star - (n - 1)
        idx = np.arange(n)
        ip = idx + lag
        im = idx - lag
        sg = np.zeros(n)
        vp = (ip >= 0) & (ip < n)
        vm = (im >= 0) & (im < n)
        sg[vp] += (1.0 - h[ip[vp]])
        sg[vm] -= h[im[vm]]
        sg *= dx
        lr = lr_init / (1.0 + 0.005 * it)
        h_new = np.clip(h - lr * sg, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        h = h_new
        h_avg = (it * h_avg + h) / (it + 1)
        if it % 30 == 0:
            c5_new = compute_c5(h)
            if c5_new < c5_cur:
                c5_cur = c5_new
                h_best = h.copy()
            c5_avg = compute_c5(project_to_sum(h_avg, target_sum))
            if c5_avg < c5_cur:
                c5_cur = c5_avg
                h_best = project_to_sum(h_avg, target_sum).copy()
    return h_best, c5_cur

def coordinate_descent(h0, n, target_sum, max_rounds, rng, tlimit, fine):
    t0 = time.time()
    h = h0.copy()
    h_best = h.copy()
    c5_cur = compute_c5(h)
    if fine:
        deltas = np.array([-0.03, -0.015, -0.01, -0.005, -0.002, -0.001, 0.001, 0.002, 0.005, 0.01, 0.015, 0.03])
    else:
        deltas = np.array([-0.05, -0.03, -0.015, -0.01, -0.005, 0.005, 0.01, 0.015, 0.03, 0.05])
    for rnd in range(max_rounds):
        if time.time() - t0 > tlimit:
            break
        idx_order = rng.permutation(n)
        for i in idx_order:
            if time.time() - t0 > tlimit:
                break
            best_val = c5_cur
            best_d = 0.0
            for d in deltas:
                nv = h[i] + d
                if nv < 0.0 or nv > 1.0:
                    continue
                h_new = h.copy()
                h_new[i] = nv
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_val - 1e-15:
                    best_val = c5_new
                    best_d = d
            if best_d != 0.0:
                h[i] += best_d
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                h_best = h.copy()
    return h_best, c5_cur

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()

        return budget_s - (time.time() - t_start)
    rng = np.random.RandomState(seed)

    if best_h is None:

    n_targets = [best_n]
    if best_n <= 400:

        x = np.linspace(0, 2, n, endpoint=False)
