    start_time = time.time()
    best_h = None
    best_c5 = float('inf')

    try:
        parent_h = np.array(initial_h_values, dtype=float)
    except Exception:
        parent_h = None

    if parent_h is None:
        n = 77
        parent_h = np.full(n, 0.5)

    n = len(parent_h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_func, 'args': (n,)}]

    # Phase 1: SLSQP with smooth max, decreasing tau (25% budget)
    t1 = start_time + budget_s * 0.25
    current = project_feasible(parent_h.copy(), n)
    c5 = evaluate_c5(current)
    best_c5 = c5
    best_h = current.copy()

    tau_list = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    for tau in tau_list:
        if time.time() > t1:
            break
        try:
            res = minimize(obj_smooth, current, method='SLSQP',
                          args=(n, tau), bounds=bounds, constraints=cons,
                          options={'maxiter': 400, 'ftol': 1e-15})
            current = project_feasible(res.x, n)
            c5 = evaluate_c5(current)
            if c5 < best_c5:
                best_c5 = c5
                best_h = current.copy()
        except Exception:
            pass

    # Phase 2: Subgradient on true max (30% budget)
    t2 = start_time + budget_s * 0.55
    if time.time() < t2:
        h_sg, c5_sg = subgradient_optimize(best_h, n, n_steps=5000, lr_init=0.012, window_margin=0.004)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # Phase 3: SLSQP fine-tune with very small tau (15% budget)
    t3 = start_time + budget_s * 0.70
    if time.time() < t3:
        current = best_h.copy()
        for tau in [0.0005, 0.0002, 0.0001]:
            if time.time() > t3:
                break
            try:
                res = minimize(obj_smooth, current, method='SLSQP',
                              args=(n, tau), bounds=bounds, constraints=cons,
                              options={'maxiter': 500, 'ftol': 1e-16})
                current = project_feasible(res.x, n)
                c5 = evaluate_c5(current)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = current.copy()
            except Exception:
                pass

    # Phase 4: Second subgradient pass with smaller learning rate (10% budget)
    t4 = start_time + budget_s * 0.80
    if time.time() < t4:
        h_sg, c5_sg = subgradient_optimize(best_h, n, n_steps=3000, lr_init=0.005, window_margin=0.002)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # Phase 5: Coordinate descent (15% budget)
    t5 = start_time + budget_s * 0.95
    if time.time() < t5:
        h_coord, c5_coord = coordinate_optimize(best_h, n, max_time=t5)
        if c5_coord < best_c5:
            best_c5 = c5_coord
            best_h = h_coord.copy()

    # Phase 6: Final SLSQP polish
    if time.time() < start_time + budget_s * 0.99:
        try:
            res = minimize(obj_smooth, best_h, method='SLSQP',
                          args=(n, 0.0001), bounds=bounds, constraints=cons,
                          options={'maxiter': 500, 'ftol': 1e-16})
            final = project_feasible(res.x, n)
            c5 = evaluate_c5(final)
            if c5 < best_c5:
                best_c5 = c5
                best_h = final
        except Exception:
            pass
        # One more coordinate sweep
        if time.time() < start_time + budget_s * 0.995:
            h_coord, c5_coord = coordinate_optimize(best_h, n, max_time=start_time + budget_s * 0.995)
            if c5_coord < best_c5:
                best_c5 = c5_coord
                best_h = h_coord.copy()

    final_c5 = evaluate_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
