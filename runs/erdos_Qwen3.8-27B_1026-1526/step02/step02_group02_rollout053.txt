```python
def compute_smoothed_gradient(h, T):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    # c has length 2n-1, c[k] corresponds to shift k-(n-1)
    # Softmax weights
    Tc = T * c
    Tc = Tc - Tc.max()  # for numerical stability
    weights = np.exp(Tc)
    weights /= weights.sum()
    
    # Compute gradient as weighted sum of individual gradients
    grad = np.zeros(n)
    for idx in range(len(c)):
        if weights[idx] < 1e-15:
            continue
        k = idx - (n - 1)  # actual shift
        w = weights[idx]
        if k >= 0:
            grad[0:n-k] += w * (1 - h[k:n])
            grad[k:n] -= w * h[0:n-k]
        else:
            m = -k
            grad[m:n] += w * (1 - h[0:n-m])
            grad[0:n-m] -= w * h[m:n]
    return grad * dx
```

```python
import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_feasible(h):
    n = len(h)
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            if not np.any(mask):
                break
            room = (1.0 - h[mask])
            step = min(diff / np.sum(mask), room.min() * 0.9999)
            h[mask] += step
        else:
            mask = h > 1e-15
            if not np.any(mask):
                break
            room = h[mask]
            step = min(-diff / np.sum(mask), room.min() * 0.9999)
            h[mask] -= step
    return np.clip(h, 0.0, 1.0)

def c5_logsumexp(h, T):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    c_max = c.max()
    shifted = T * (c - c_max)
    shifted = np.clip(shifted, -500, 500)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / T)

def c5_logsumexp_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    Tc = T * c
    Tc_max = Tc.max()
    exp_Tc = np.exp(Tc - Tc_max)
    weights = exp_Tc / exp_Tc.sum()
    
    grad = np.zeros(n)
    for idx in range(len(c)):
        w = weights[idx]
        if w < 1e-14:
            continue
        k = idx - (n - 1)
        if k > 0:
            grad[k:n] += w * (1.0 - h[0:n-k])
            grad[0:n-k] -= w * h[k:n]
        elif k < 0:
            m = -k
            grad[0:n-m] += w * (1.0 - h[m:n])
            grad[m:n] -= w * h[0:n-m]
        else:
            grad += w * (1.0 - 2.0 * h)
    return grad * dx

def smooth_obj(h, T, penalty):
    n = len(h)
    c5_val = c5_logsumexp(h, T)
    sum_v = np.sum(h) - n / 2.0
    return c5_val + 0.5 * penalty * sum_v * sum_v

def smooth_obj_and_grad(h, T, penalty):
    n = len(h)
    c5_val = c5_logsumexp(h, T)
    sum_v = np.sum(h) - n / 2.0
    obj = c5_val + 0.5 * penalty * sum_v * sum_v
    grad = c5_logsumexp_grad(h, T) + penalty * sum_v * np.ones(n)
    return obj, grad

def lbfgsb_smooth(h_init, T, penalty, maxiter):
    n = len(h_init)
    bounds = [(0.0, 1.0)] * n
    result = minimize(
        smooth_obj_and_grad, h_init, args=(T, penalty),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12, 'maxcor': 50}
    )
    if result.x is not None:
        h_out = project_feasible(result.x)
        return h_out, compute_c5_from_h(h_out)
    return h_init, compute_c5_from_h(h_init)

def slsqp_obj(x):
    return x[-1]

def slsqp_ineq(x):
    h = x[:-1]
    t = x[-1]
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return t - c * dx

def slsqp_eq(x):
    h = x[:-1]
    return np.sum(h) - len(h) / 2.0

def slsqp_eq_jac(x):
    n = len(x) - 1
    jac = np.zeros(len(x))
    jac[:n] = 1.0
    return jac

def make_slsqp_constraints():
    return [
        {'type': 'ineq', 'fun': slsqp_ineq},
        {'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac}
    ]

def slsqp_optimize(h_init, maxiter=3000, t_margin=0.1):
    c5_init = compute_c5_from_h(h_init)
    n = len(h_init)
    x0 = np.concatenate([h_init, [c5_init + t_margin]])
    bounds = [(0.0, 1.0)] * n + [(max(0.001, c5_init * 0.3), 2.0)]
    constraints = make_slsqp_constraints()
    result = minimize(slsqp_obj, x0, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
    if result.x is not None:
        h_out = project_feasible(result.x[:-1])
        return h_out, compute_c5_from_h(h_out)
    return h_init, c5_init

def interpolate_to_n(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.arange(n_old)
    x_new = np.linspace(0, n_old - 1, n_new)
    f = interp1d(x_old, h_old, kind='linear', fill_value="extrapolate")
    h_new = f(x_new)
```
