            # -h[i+k] for i in 0..n-k-1, i.e., indices k..n-1 of h
            grad[:n-k] += pk * (-h[k:])

    # By the definition, for k >= n:
    # c[k] = sum_{i=0}^{2n-2-k} h[i] * (1-h[i + k - (n-1)])
    # Let j = k - (n-1), so j >= 1
    # c[k] = sum_{i=0}^{n-j} h[i] * (1-h[i+j])
    # This is the same as the "other direction" correlation.
    
    for k in range(n, 2*n-1):
        j = k - (n-1)  # j ranges from 1 to n-1
        pk = p[k] * dx
        # c[k] = sum_{i=0}^{n-1-j} h[i] * (1-h[i+j])
        # dc/dh[m]: 
        #   From h[i] where i=m: (1-h[m+j]) if m <= n-1-j
        #   From -h[i+j] where i+j=m, i=m-j: -h[m-j] if m-j >= 0 and m-j <= n-1-j, i.e., j <= m <= n-1
        if j == 0:
            pass
        else:
            # h[i] term: for i in 0..n-1-j, contributes (1-h[i+j])
            grad[:n-j] += pk * (1 - h[j:])
            # -h[i+j] term: for i in 0..n-1-j, i+j ranges from j to n-1
            grad[j:] += pk * (-h[:n-j])
    
    return obj, grad

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time


def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = diff / cnt
            h[mask] = np.minimum(1.0, h[mask] + step)
        else:
            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = -diff / cnt
            h[mask] = np.maximum(0.0, h[mask] - step)
    return np.clip(h, 0.0, 1.0)


def smooth_obj_and_grad(x, T, penalty):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    c_max = c.max()
    shifted = T * (c - c_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    p = exp_shifted / sum_exp
    obj = c_max + np.log(sum_exp) / T
    sum_dev = np.sum(x) - n / 2.0
    obj += penalty * sum_dev * sum_dev
    grad = np.zeros(2 * n - 1)
    for k in range(n):
        pk = p[k]
        if k == 0:
            grad[k:n] += pk * (1.0 - x)
            grad[0:n] += pk * (-x)
        else:
            grad[k:n] += pk * (1.0 - x[:n - k])
            grad[0:n - k] += pk * (-x[k:])
    for k in range(n, 2 * n - 1):
        j = k - (n - 1)
        pk = p[k]
        grad[0:n - j] += pk * (1.0 - x[j:n])
        grad[j:n] += pk * (-x[0:n - j])
    grad = grad[:n] * dx + 2.0 * penalty * sum_dev
    return obj, grad


def lbfgsb_optimize(h_init, T, penalty, maxiter=1000):
    n = len(h_init)

    def obj_func(x):
        return smooth_obj_and_grad(x, T, penalty)[0]

    def grad_func(x):
        return smooth_obj_and_grad(x, T, penalty)[1]

    bounds = [(0.0, 1.0)] * n
    result = minimize(obj_func, h_init, jac=grad_func, method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12, 'maxls': 50})
    if result.x is not None:
        h_out = project_feasible(result.x, n)
        return h_out, compute_c5_from_h(h_out)
    return h_init, compute_c5_from_h(h_init)


def generate_initial_candidates(n, rng):
    h0 = np.full(n, 0.5)
    candidates = [h0]
    for m in range(1, 8):
        for A in [0.1, 0.2, 0.3, 0.4]:
            x = np.arange(n)
            h_cand = 0.5 + A * np.sin(2.0 * np.pi * m * x / n)
            h_cand = np.clip(h_cand, 0, 1)
            h_cand = project_feasible(h_cand, n)
            candidates.append(h_cand.copy())
    for m in range(1, 5):
        x = np.arange(n)
        h_cand = 0.5 + 0.3 * np.cos(2.0 * np.pi * m * x / n)
        h_cand = np.clip(h_cand, 0, 1)
        h_cand = project_feasible(h_cand, n)
        candidates.append(h_cand.copy())
    tri = np.zeros(n)
    mid = n // 2
    for i in range(n):
        d = abs(i - mid)
        tri[i] = max(0, 1.0 - d / (n * 0.4))
    tri = 0.5 + 0.3 * (tri - tri.mean()) / (tri.max() + 1e-10)
    tri = np.clip(tri, 0, 1)
    tri = project_feasible(tri, n)
    candidates.append(tri)
    return candidates

            if c5_new < best_c5 - 1e-10:
                best_h = h_new
                best_c5 = c5_new
                h_interp = best_h.copy()
            else:
                h_interp = h_new

    if time.time() - t0 < budget_s - 60:

        for T in [3, 2, 1.5, 1, 0.7]:
            if time.time() - t0 > budget_s - 10:

            if c5_new < best_c5:

                best_c5 = c5_new

    n_polish = 50

    best_h = project_feasible(best_h, len(best_h))
