    Objective: sum_k softplus(alpha*(corr[k]*dx - T)) + mu*(sum(h)-target_sum)^2
    This is smooth and approximates: penalty for corr[k]*dx > T + sum constraint

    z_safe = np.clip(z, -500, 500)

    # corr[k] = sum_j h[j] * (1-h[j+k-(n-1)]) for valid j

    # d/corr[k]/dh[i] = (1-h[i+ell])*[0<=i+ell<n] - h[i-ell]*[0<=i-ell<n]

    # = sum_k w_corr[k] * (1-h[i+ell_k])*[valid] - sum_k w_corr[k] * h[i-ell_k]*[valid]
    
    # Use convolution-like computation
    # For each i: sum over k of w_corr[k] * (1-h[i+ell_k]) for valid
    # ell_k = k - (n-1), ranges from -(n-1) to (n-1)
    
    # Let's do it directly with broadcasting for n<=600
    i_arr = np.arange(n)[:, None]  # (n, 1)
    ell_arr = np.arange(2*n-1)[None, :] - (n-1)  # (1, 2n-1)
    
    idx_plus = i_arr + ell_arr  # (n, 2n-1)
    idx_minus = i_arr - ell_arr  # (n, 2n-1)
    
    valid_plus = (idx_plus >= 0) & (idx_plus < n)
    valid_minus = (idx_minus >= 0) & (idx_minus < n)
    
    # Clip for safe indexing
    ip = np.clip(idx_plus, 0, n-1)
    im = np.clip(idx_minus, 0, n-1)
    
    part1 = valid_plus * (1.0 - h[ip])  # (n, 2n-1)
    part2 = valid_minus * h[im]  # (n, 2n-1)
    
    grad_corr = np.sum(w_corr[None, :] * (part1 - part2), axis=1)
    
    grad = grad_corr + 2 * mu * (s - target_sum) * np.ones(n)
    
    return obj, grad
