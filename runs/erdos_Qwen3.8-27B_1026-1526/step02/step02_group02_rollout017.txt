import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time


def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            step = min(diff / np.sum(mask), (1.0 - h[mask]).min() * 0.9999)
            h[mask] += step
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            step = min(-diff / np.sum(mask), h[mask].min() * 0.9999)
            h[mask] -= step
    return np.clip(h, 0.0, 1.0)


def c5_smoothed(h, T):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    c_max = c.max()
    if c_max < -1e10:
        return 0.0
    shifted = T * (c - c_max)
    shifted = np.clip(shifted, -500, 500)
    return c_max + np.log(np.sum(np.exp(shifted))) / T


def slsqp_minmax(h_init, n, maxiter=10000, t_margin=0.05):
    c5_init = compute_c5_from_h(h_init)
    dx = 2.0 / n

    def obj(x):
        return x[-1]

    def ineq(x):
        h = x[:-1]
        t = x[-1]
        c = np.correlate(h, 1.0 - h, mode="full") * dx
        return t - c

    def eq(x):
        return np.sum(x[:-1]) - len(x) / 2.0

    def eq_jac(x):
        nv = len(x) - 1
        j = np.zeros(len(x))
        j[:nv] = 1.0
        return j

    x0 = np.concatenate([h_init, [c5_init + t_margin]])
    bounds = [(0.0, 1.0)] * n + [(max(0.001, c5_init * 0.3), 2.0)]
    constraints = [
        {'type': 'ineq', 'fun': ineq},
        {'type': 'eq', 'fun': eq, 'jac': eq_jac}
    ]

    result = minimize(obj, x0, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-15})

    if result.x is not None:
        h_out = project_feasible(result.x[:-1], n)
        return h_out, compute_c5_from_h(h_out)
    return h_init, c5_init


def lbfgsb_smooth(h_init, T, penalty, maxiter=5000):
    n = len(h_init)

    def obj(h):
        c5s = c5_smoothed(h, T)
        sv = np.sum(h) - n / 2.0
        return c5s + penalty * sv * sv

    result = minimize(obj, h_init, method='L-BFGS-B',
                      bounds=[(0.0, 1.0)] * n,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14})

    if result.x is not None:
        h_out = project_feasible(result.x, n)
        return h_out, compute_c5_from_h(h_out)
    return h_init, compute_c5_from_h(h_init)

    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float).copy()

    # Phase 3: SLSQP after smoothing

                amp = scale / np.sqrt(nf)

        elif r < 0.8:

            h_pert = 0.7 * best_h + 0.3 * h_pert

            if c5_new < best_c5:

        h_interp = np.clip(f_interp(x_new), 0, 1)
        h_interp = project_feasible(h_interp, n_try)

        if c5_new < best_c5 - 1e-12:

            if time.time() - t0 > budget_s - 15:

    best_c5 = compute_c5_from_h(best_h)
