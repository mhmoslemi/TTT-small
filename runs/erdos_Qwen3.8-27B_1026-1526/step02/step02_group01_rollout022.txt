```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        active = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(active):
            h[active] += diff / np.sum(active)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / (2*n - 1)) / alpha
    
    g = np.zeros(n)
    for j in range(n):
        i_lo = max(0, j - n + 1)
        i_hi = min(n - 1, j + n - 2)
        i = np.arange(i_lo, i_hi + 1)
        m1 = i - j + n - 1
        m2 = j - i + n - 1
        g[j] = np.dot(w[m1], 1.0 - h[i]) - np.dot(w[m2], h[i])
    return sm_val, g

def thue_morse(n):
    """Generate Thue-Morse sequence of length n, mapped to [0,1]."""
    tm = np.zeros(n)
    for i in range(1, n):
        tm[i] = tm[i >> 1] ^ (i & 1)
    return tm  # binary 0/1

def golden_ratio_binary(n):
    """Generate a balanced binary sequence based on golden ratio."""
    phi = (1 + np.sqrt(5)) / 2
    x = np.arange(n)
    return np.round(x * phi / n) - np.round((x - 1) * phi / n)

def slsqp_optimize(h0, n, target_sum, alpha, maxiter):
    def obj(x):
        v, _ = smooth_max_and_grad(np.clip(x, 0, 1), alpha)
        return v
    def grad(x):
        _, g = smooth_max_and_grad(np.clip(x, 0, 1), alpha)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def nesterov_gradient(h0, n, target_sum, alpha, n_iters, lr_init):
    h = h0.astype(float).copy()
    y = h.copy()
    t = 1.0
    lr = lr_init
    best_c5 = compute_c5(h)
    h_best = h.copy()
    for it in range(n_iters):
        _, g = smooth_max_and_grad(np.clip(y, 0, 1), alpha)
        h_new = np.clip(y - lr * g, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta = (t - 1.0) / t_new
        y = h_new + beta * (h_new - h)
        y = np.clip(y, 0.0, 1.0)
        h = h_new
        t = t_new
        # Adaptive step size
        c5_new = compute_c5(h)
        if c5_new < best_c5:
            best_c5 = c5_new
            h_best = h.copy()
            lr = min(lr * 1.02, 0.3)
        else:
            lr *= 0.9
    return h_best, best_c5

def coordinate_descent(h0, n, target_sum, max_rounds, rng, step_sizes=None):
    h = h0.copy()
    c5_cur = compute_c5(h)
    if step_sizes is None:
        step_sizes = [0.05, 0.02, 0.01, 0.005, -0.05, -0.02, -0.01, -0.005]

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        active = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(active):
            h[active] += diff / np.sum(active)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / (2 * n - 1)) / alpha
    g = np.zeros(n)
    for j in range(n):
        i_lo = max(0, j - n + 1)
        i_hi = min(n - 1, j + n - 2)
        i = np.arange(i_lo, i_hi + 1)
        m1 = i - j + n - 1
        m2 = j - i + n - 1
        g[j] = np.dot(w[m1], 1.0 - h[i]) - np.dot(w[m2], h[i])
    return sm_val, g


def thue_morse_gen(n):
    tm = np.zeros(n, dtype=np.float64)
    for i in range(1, n):
        tm[i] = tm[i >> 1] ^ (i & 1)
    return tm


def golden_ratio_binary(n):
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    x = np.arange(n, dtype=np.float64)
    return np.round(x * phi) - np.round((x - 0.5) * phi)


def slsqp_optimize(h0, n, target_sum, alpha, maxiter):
    def obj_func(x):
        v, _ = smooth_max_and_grad(np.clip(x, 0.0, 1.0), alpha)
        return v

    def grad_func(x):
        _, g = smooth_max_and_grad(np.clip(x, 0.0, 1.0), alpha)
        return g

    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_func, h0.copy(), jac=grad_func, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)


def nesterov_gradient(h0, n, target_sum, alpha, n_iters, lr_init):
    h = h0.astype(float).copy()
    y = h.copy()
    t_mom = 1.0
    lr = lr_init
    best_c5 = compute_c5(h)
    h_best = h.copy()
    for it in range(n_iters):
        _, g = smooth_max_and_grad(np.clip(y, 0.0, 1.0), alpha)
        h_new = np.clip(y - lr * g, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        beta = (t_mom - 1.0) / t_new
        y = h_new + beta * (h_new - h)
        y = np.clip(y, 0.0, 1.0)
        h = h_new
        t_mom = t_new
        c5_new = compute_c5(h)
        if c5_new < best_c5:
            best_c5 = c5_new
            h_best = h.copy()
            lr = min(lr * 1.01, 0.2)
        else:
            lr *= 0.92
    return h_best, best_c5


def coordinate_descent(h0, n, target_sum, max_rounds, rng):
    h = h0.copy()
    c5_cur = compute_c5(h)
    step_sizes = [0.04, 0.02, 0.01, 0.005, -0.04, -0.02, -0.01, -0.005,
                  0.08, -0.08, 0.002, -0.002]
    for rnd in range(max_rounds):
        improved = False
        idx_order = rng.permutation(n)

            best_val = c5_cur

                h_new = h.copy()

            break
    return h, c5_cur
