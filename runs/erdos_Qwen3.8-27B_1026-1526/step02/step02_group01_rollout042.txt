def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:

def fft_correlation(h):
    """Compute correlate(h, 1-h, mode='full') using FFT"""
    n = len(h)
    m = 1
    while m < 2*n - 1:
        m *= 2
    H = np.fft.rfft(h, m)
    G = np.fft.rfft(1.0 - h, m)
    corr = np.fft.irfft(H * G, m)[:2*n-1]
    return corr

    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / len(exp_vals)) / alpha

If h were constant at 1/2, then h(x)(1-h(x+k)) = 1/4 for all k, so C₅ = 1/4 * 2 = 1/2.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def fft_correlation(h):
    n = len(h)
    size = 1
    while size < 2 * n - 1:
        size *= 2
    H = np.fft.rfft(h, size)
    G = np.fft.rfft(1.0 - h, size)
    corr = np.fft.irfft(H * G, size)[:2*n-1]
    return corr

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = fft_correlation(h)
    m = float(np.max(corr))
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = float(np.sum(exp_vals))
    w = exp_vals / s
    sm_val = m + np.log(s / (2*n-1)) / alpha
    
    size = 1
    while size < 3 * n:
        size *= 2
    H = np.fft.rfft(h, size)
    Wf = np.fft.rfft(w, size)
    conv_wh = np.fft.irfft(Wf * H, size)
    w_rev = w[::-1]
    WR = np.fft.rfft(w_rev, size)
    conv_wrev_h = np.fft.irfft(WR * H, size)
    
    grad = np.empty(n)
    for idx in range(n):
        grad[idx] = 1.0 - conv_wh[idx + n - 1] - conv_wrev_h[idx + n - 3]
    return sm_val, grad

def gradient_descent_nesterov(h0, target_sum, alphas, lr_init, n_iters_per_alpha, time_limit):
    h = h0.copy()
    c5_cur = compute_c5(h)
    h_best = h.copy()
    t_start = time.time()
    
    for alpha in alphas:
        if time.time() - t_start > time_limit:
            break
        h = h.copy()
        y = h.copy()
        t = 1.0
        lr = lr_init
        alpha_iters = 0
        for it in range(n_iters_per_alpha):
            if time.time() - t_start > time_limit:
                break
            _, g = smooth_max_and_grad(y, alpha)
            h_new = np.clip(y - lr * g, 0.0, 1.0)
            h_new = project_to_sum(h_new, target_sum)
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
            mo = (t - 1.0) / t_new
            y = h_new + mo * (h_new - h)
            y = np.clip(y, 0.0, 1.0)
            h = h_new
            t = t_new
            alpha_iters += 1
            if it % 50 == 0:
                c5_new = compute_c5(h)
                if c5_new < c5_cur:
                    c5_cur = c5_new
                    h_best = h.copy()
                    lr *= 1.01
                elif it % 500 == 0:
                    lr *= 0.95
        c5_new = compute_c5(h)
        if c5_new < c5_cur:
            c5_cur = c5_new
            h_best = h.copy()
    
    return h_best, c5_cur

def slsqp_optimize(h0, target_sum, alpha, maxiter):
    n = len(h0)
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_and_grad(h, alpha)
        return v
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_and_grad(h, alpha)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def coordinate_descent(h0, target_sum, max_rounds, rng, step_sizes=None):
    h = h0.copy()
    c5_cur = compute_c5(h)
    if step_sizes is None:
        step_sizes = [0.001, 0.005, 0.01, 0.02, -0.001, -0.005, -0.01, -0.02]
    for rnd in range(max_rounds):
        improved = False
        idx_order = rng.permutation(len(h))
        for i in idx_order:
            best_val = c5_cur
            best_delta = 0.0
            for d in step_sizes:
                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] + d, 0.0, 1.0)
