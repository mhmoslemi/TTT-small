```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def binarize(h, target_ones):
    n = len(h)
    idx = np.argsort(h)[::-1][:target_ones]
    b = np.zeros(n)
    b[idx] = 1.0
    return b

def sa_binary(h0, seed, max_time_s, T0_scale=2.0):
    n = len(h0)
    target_ones = int(round(np.sum(h0)))
    b = binarize(h0, target_ones).copy()
    c5_cur = compute_c5(b)
    b_best = b.copy()
    c5_best = c5_cur
    
    rng = np.random.RandomState(seed)
    t_start = time.time()
    
    dx = 2.0 / n
    T0 = T0_scale * dx * 2  # initial temperature
    Tmin = 1e-10
    
    ones_list = list(np.where(b == 1.0)[0])
    zeros_list = list(np.where(b == 0.0)[0])
    
    # Track stagnation
    best_iter = 0
    total_iters = 0
    stagnation = 0
    
    while time.time() - t_start < max_time_s:
        progress = min(1.0, (time.time() - t_start) / max_time_s)
        T = T0 * (Tmin / T0) ** progress
        
        i_idx = rng.randint(len(ones_list))
        j_idx = rng.randint(len(zeros_list))
        i_pos = ones_list[i_idx]
        j_pos = zeros_list[j_idx]
        
        b[i_pos] = 0.0
        b[j_pos] = 1.0
        c5_new = compute_c5(b)
        total_iters += 1
        
        delta = c5_new - c5_cur
        if delta < 0 or (T > 1e-12 and rng.random() < np.exp(-delta / T)):
            c5_cur = c5_new
            if c5_new < c5_best - 1e-15:
                c5_best = c5_new
                b_best = b.copy()
                best_iter = total_iters
            ones_list[i_idx] = j_pos
            zeros_list[j_idx] = i_pos
        else:
            b[i_pos] = 1.0
            b[j_pos] = 0.0
        
        # Check stagnation
        if total_iters - best_iter > 50000:
            stagnation += 1
            if stagnation > 3:
                break
    
    return b_best, c5_best

def local_search_binary(b, max_time_s, rng):
    """Try all single swaps and accept improvements."""
    n = len(b)
    c5_cur = compute_c5(b)
    t_start = time.time()
    
    while time.time() - t_start < max_time_s:
        improved = False
        ones = np.where(b == 1.0)[0]
        zeros = np.where(b == 0.0)[0]
        
        for i_pos in ones:
            for j_pos in zeros:
                if abs(i_pos - j_pos) < 2:
                    continue
                b[i_pos] = 0.0
                b[j_pos] = 1.0
                c5_new = compute_c5(b)
                if c5_new < c5_cur - 1e-15:
                    c5_cur = c5_new
                    improved = True
                else:
                    b[i_pos] = 1.0
                    b[j_pos] = 0.0
        
        if not improved:
            break
    
    return b, c5_cur
```
