        return v

        _, g = smooth_max_and_grad(h, alpha)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,

    h = h0.copy()

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s) / alpha
    g = np.zeros(n)
    for l in range(-(n - 1), n):
        wl = w[l + n - 1]
        if wl < 1e-14:
            continue
        lo = max(0, l)
        hi = min(n, n + l)
        if lo < hi:
            g[lo:hi] += wl * (1.0 - h[lo - l:hi - l])
        lo2 = max(0, -l)
        hi2 = min(n, n - l)
        if lo2 < hi2:
            g[lo2:hi2] -= wl * h[lo2 + l:hi2 + l]
    return sm_val, g

def slsqp_optimize(h0, n, target_sum, alpha, maxiter):
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_and_grad(h, alpha)
        return v
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_and_grad(h, alpha)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

                if c5_pen < best_c5:
                    best_c5 = c5_pen
                    best_h = h_pen.copy()
                    best_n = n
                h_cur = h_pen

            if time_left() > 35:
                base_steps = [0.04, 0.02, 0.01, -0.04, -0.02, -0.01]
                max_rnd = max(10, min(30, int(time_left() // 3)))
                h_cd, c5_cd = coordinate_descent(h_cur, n, ts, max_rnd, base_steps)
                if c5_cd < best_c5:
                    best_c5 = c5_cd
                    best_h = h_cd.copy()
                    best_n = n

    if best_h is not None and time_left() > 25:
        n_f = best_n
        ts_f = n_f / 2.0

        for alpha in [1000.0, 2000.0, 4000.0, 8000.0]:
            if time_left() < 18:
                break
            h_pol = slsqp_optimize(best_h, n_f, ts_f, alpha, 900)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

        if time_left() > 25:
            C_pen = max(best_c5 - 0.002, 0.1)
            n_pen = min(4000, int(time_left() * 2))
            h_pen, c5_pen = penalty_descent(best_h, n_f, ts_f, C_pen, n_pen, 0.02)
            if c5_pen < best_c5:
                best_c5 = c5_pen
                best_h = h_pen.copy()

        if time_left() > 15:
            base_steps = [0.02, 0.01, 0.005, 0.002, -0.02, -0.01, -0.005, -0.002]
            h_cd, c5_cd = coordinate_descent(best_h, n_f, ts_f, 60, base_steps)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()

        if time_left() > 10:
            n_mom = min(3000, int(time_left() * 2))
            h_mom, c5_mom = momentum_gradient(best_h, n_f, ts_f, n_mom, 0.01, 200.0)
            if c5_mom < best_c5:
                best_c5 = c5_mom
                best_h = h_mom.copy()

    n_final = len(best_h)
    ts_final = n_final / 2.0
    best_h = project_to_sum(best_h, ts_final)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_final)
