def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx), c * dx

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def lag_gradient(h, ell):
    """Compute gradient of c[ell] (the correlation at lag ell) w.r.t. h."""
    n = len(h)
    grad = np.zeros(n)
    if ell >= 0:
        # h[m-ell] factor for m in [ell, n-1]
        if ell < n:
            grad[ell:] += (1.0 - h[:n-ell])
