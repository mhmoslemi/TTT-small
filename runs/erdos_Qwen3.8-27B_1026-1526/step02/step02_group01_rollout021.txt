Let me also add a **key new idea**: **alternating optimization between h and its "dual" representation**.

The overlap at shift k is: ∫ h(x)(1-h(x+k)) dx = ∫h - ∫h(x)h(x+k)dx = 1 - R(k)

where R(k) is the autocorrelation. So we want to maximize min_k R(k).

Now, R(k) = ∫ h(x)h(x+k) dx. If we think of h as a vector, R(k) = h^T P_k h where P_k is a "shift" matrix (symmetric part).

Actually, R(k) is not a quadratic form in h in the standard sense because it's h^T S_k h where S_k is the shift operator, and S_k is not symmetric in general (for linear, not circular, shifts).

But for circular shifts, R(k) = sum_i h[i] h[(i+k) mod n], which IS a quadratic form: h^T C_k h where C_k is the circulant matrix for shift k. And since C_k is symmetric (circulant matrices are symmetric iff the generating sequence is symmetric... actually C_k^T = C_{-k}, so C_k is symmetric iff k=0 or k=n/2).

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val_and_weights(corr, alpha):
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    sm_val = m + np.log(np.sum(exp_vals) / len(exp_vals)) / alpha
    return sm_val, w

def grad_wrt_h(h, w, dx):
    """Compute gradient of weighted sum of correlations w.r.t. h."""
    n = len(h)
    g = np.zeros(n)
    for j in range(n):
        # d(corr[k])/d(h[j]) = dx * [(1-h[j+s])*1_{0≤j+s<n} - h[j-s]*1_{0≤j-s<n}]
        # where s = k-(n-1), and w indexes k
        # Equivalently with s: weight is w[n-1+s]
        # First part: s from max(-j, -(n-1)) to min(n-1-j, n-1)
        s1_lo = max(-j, -(n-1))
        s1_hi = min(n-1-j, n-1)
        if s1_hi >= s1_lo:
            ss = np.arange(s1_lo, s1_hi + 1)
            g[j] += dx * np.dot(w[n-1 + ss], 1.0 - h[j + ss])
        # Second part: s from max(j-n+1, -(n-1)) to min(j, n-1)
        s2_lo = max(j - n + 1, -(n-1))
        s2_hi = min(j, n-1)
        if s2_hi >= s2_lo:
            ss = np.arange(s2_lo, s2_hi + 1)
            g[j] -= dx * np.dot(w[n-1 + ss], h[j - ss])
    return g

def adam_optimize(h0, target_sum, alpha_start, alpha_end, n_iters, lr):
    """Adam optimizer with smooth max objective."""
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    t = 0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    
    for it in range(n_iters):
        t += 1
        # Interpolate alpha
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max(1, n_iters - 1))
        
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        sm_val, w = smooth_max_val_and_weights(corr, alpha)
        g = grad_wrt_h(h, w, dx)
        
        # Adam update
        m_hat = beta1 * m_hat + (1 - beta1) * g
        v_hat = beta2 * v_hat + (1 - beta2) * g**2
        m_hat_unbias = m_hat / (1 - beta1**t)
        v_hat_unbias = v_hat / (1 - beta2**t)
        
        h_new = h - lr * m_hat_unbias / (np.sqrt(v_hat_unbias) + eps)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        h = h_new
    
    return best_h, best_c5

def coordinate_descent_line_search(h0, n, target_sum, max_rounds, rng):
    """Coordinate descent with line search."""
    h = h0.copy()
    c5_cur = compute_c5(h)
    for rnd in range(max_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            # Try to find the best value for h[i] via ternary search
            lo, hi = 0.0, 1.0
            # Evaluate at a few points
            best_val = c5_cur
            best_h_i = h[i]
            for frac in [0.0, 0.25, 0.5, 0.75, 1.0]:
                trial = lo + frac * (hi - lo)
                h_trial = h.copy()
                h_trial[i] = trial
                h_trial = project_to_sum(h_trial, target_sum)
                val = compute_c5(h_trial)
                if val < best_val:
                    best_val = val
                    best_h_i = trial
            if best_h_i != h[i]:
                h[i] = best_h_i
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True
        if not improved:
            break
    return h, c5_cur

def coord_descent_fast(h0, target_sum, max_rounds, rng):
    n = len(h0)
    h = h0.copy()
    c5_cur = compute_c5(h)
    for rnd in range(max_rounds):
        improved = False
        for i in rng.permutation(n):
            # Try changing h[i] and compensating with h[j]
            # Find a good j to compensate
            deltas = [0.01, 0.005, 0.001, -0.001, -0.005, -0.01]
            best_val = c5_cur
            best_move = None
            for d in deltas:
                new_val = h[i] + d
                if new_val < 0 or new_val > 1:
                    continue
                # Find j to compensate: need h[j] - d to be in [0,1]
                # Prefer j where h[j] is in (0,1)
                candidates = np.where((h > abs(d) + 1e-6) & (h < 1.0 - abs(d) - 1e-6) & (np.arange(n) != i))[0]
                if len(candidates) == 0:
                    continue
                # Try a few candidates
                try_j = candidates[rng.randint(len(candidates))]
                h_trial = h.copy()
                h_trial[i] = new_val
                h_trial[try_j] = h[try_j] - d
                if 0 <= h_trial[try_j] <= 1:
                    val = compute_c5(h_trial)
                    if val < best_val - 1e-15:
                        best_val = val
                        best_move = (i, d, try_j)
            if best_move is not None:
                i, d, j = best_move
                h[i] += d
                h[j] -= d
                c5_cur = best_val
                improved = True
        if not improved:
            break
    return h, c5_cur

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    if topk_idx is not None and len(topk_idx) < len(corr):
        c = corr[topk_idx]
        m_c = np.max(c)
        shifted = alpha * (c - m_c)
        shifted = np.clip(shifted, -700.0, 700.0)
        exp_vals = np.exp(shifted)
        s = np.sum(exp_vals)
        w_local = exp_vals / s
        sm_val = m_c + np.log(s / len(exp_vals)) / alpha
        w = np.zeros(len(corr))
        w[topk_idx] = w_local
    else:
        shifted = alpha * (corr - m)
        shifted = np.clip(shifted, -700.0, 700.0)
        exp_vals = np.exp(shifted)
        s = np.sum(exp_vals)
        w = exp_vals / s
        sm_val = m + np.log(s / len(exp_vals)) / alpha
    g = np.zeros(n)
    for j in range(n):
        s1_lo = max(-j, -(n - 1))
        s1_hi = min(n - 1 - j, n - 1)
        if s1_hi >= s1_lo:
            ss = np.arange(s1_lo, s1_hi + 1)
            g[j] += dx * np.dot(w[n - 1 + ss], 1.0 - h[j + ss])
        s2_lo = max(j - n + 1, -(n - 1))
        s2_hi = min(j, n - 1)
        if s2_hi >= s2_lo:
            ss = np.arange(s2_lo, s2_hi + 1)
            g[j] -= dx * np.dot(w[n - 1 + ss], h[j - ss])
    return sm_val, g

def adam_optimize(h0, target_sum, alpha_start, alpha_end, n_iters, lr, topk_refresh=200, topk_size=20):
    n = len(h0)
    h = h0.copy()
    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    t = 0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    topk_idx = None
    patience = 0
    for it in range(n_iters):
        t += 1
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max(1, n_iters - 1))
        if it % topk_refresh == topk_refresh - 1 and it > 0:
            corr = compute_c5_corr(h)
            topk_idx = np.argsort(corr)[-topk_size:]
        sm_val, g = smooth_max_and_grad(h, alpha, topk_idx)
        m_hat = beta1 * m_hat + (1.0 - beta1) * g
        v_hat = beta2 * v_hat + (1.0 - beta2) * g * g
        m_hat_unbias = m_hat / (1.0 - beta1 ** t)
        v_hat_unbias = v_hat / (1.0 - beta2 ** t)
        h_new = h - lr * m_hat_unbias / (np.sqrt(v_hat_unbias) + eps)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1
        h = h_new
        if patience > 5000:
            break
    return best_h, best_c5

def coord_descent_fast(h0, target_sum, max_rounds, rng):
    n = len(h0)
    h = h0.copy()
    c5_cur = compute_c5(h)
    deltas = [0.02, 0.01, 0.005, 0.002, 0.001, -0.001, -0.002, -0.005, -0.01, -0.02]
    for rnd in range(max_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            best_val = c5_cur
            best_d = 0.0
            best_j = -1
            for d in deltas:
                new_val = h[i] + d
                if new_val < -0.01 or new_val > 1.01:
                    continue
                candidates = np.where((np.arange(n) != i) & (h > 1e-6) & (h < 1.0 - 1e-6))[0]
                if len(candidates) == 0:
                    break
                j = candidates[rng.randint(len(candidates))]
                h_trial = h.copy()
                h_trial[i] = np.clip(new_val, 0.0, 1.0)
                h_trial[j] = h[j] - (h_trial[i] - h[i])
                if h_trial[j] < 0.0 or h_trial[j] > 1.0:
                    continue
                val = compute_c5(h_trial)
                if val < best_val - 1e-15:
                    best_val = val
                    best_d = d
                    best_j = j
            if best_j >= 0:
                h[i] = np.clip(h[i] + best_d, 0.0, 1.0)
                h[best_j] -= (h[i] - (h0[i] + best_d))
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True
        if not improved:
            break
    return h, c5_cur

def nesterov_gradient(h0, target_sum, alpha, n_iters, lr_init):
    n = len(h0)
    h = h0.copy()
    y = h.copy()
    t = 1.0
    lr = lr_init
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(n_iters):
        _, g = smooth_max_and_grad(y, alpha)
        h_new = np.clip(y - lr * g, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        mo = (t - 1.0) / t_new
        y = h_new + mo * (h_new - h)
        y = np.clip(y, 0.0, 1.0)
        h = h_new
        t = t_new
        c5_new = compute_c5(h)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h.copy()
            lr = min(lr * 1.02, 0.3)
        else:
            lr *= 0.9
            if lr < 1e-6:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            ts = best_n / 2.0
            best_h = project_to_sum(best_h, ts)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 360
        best_h = np.full(360, 0.5)
        best_c5 = compute_c5(best_h)

    def time_left():
        return budget_s - (time.time() - t_start)

    n0 = best_n
    ts0 = n0 / 2.0
    h_work = best_h.copy()

    if time_left() > 300:
        h_work, c5 = adam_optimize(h_work, ts0, 3.0, 50.0, 4000, 0.02)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_work.copy()

    if time_left() > 250:
        h_work, c5 = adam_optimize(best_h, ts0, 50.0, 300.0, 4000, 0.01)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_work.copy()

    if time_left() > 100:
        h_work, c5 = nesterov_gradient(best_h, ts0, 200.0, 3000, 0.03)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_work.copy()

    if time_left() > 80:
        h_work, c5 = coord_descent_fast(best_h, ts0, 40, rng)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_work.copy()
            best_n = n0

    n1 = 720
    ts1 = n1 / 2.0
    x_old = np.linspace(0, 2, best_n, endpoint=False)
    x_new = np.linspace(0, 2, n1, endpoint=False)
    h_up = np.interp(x_new, x_old, best_h)
    h_up = project_to_sum(h_up, ts1)
    c5_up = compute_c5(h_up)
    if c5_up < best_c5:
        best_c5 = c5_up
        best_h = h_up.copy()
        best_n = n1

    h_work = h_up.copy()
    if time_left() > 250:
        h_work, c5 = adam_optimize(h_work, ts1, 3.0, 50.0, 5000, 0.015)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_work.copy()
            best_n = n1

    if time_left() > 200:
        h_work, c5 = adam_optimize(best_h, ts1, 50.0, 400.0, 5000, 0.008)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_work.copy()
            best_n = n1

    if time_left() > 120:
        h_work, c5 = adam_optimize(best_h, ts1, 400.0, 1500.0, 4000, 0.004)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_work.copy()
            best_n = n1

    if time_left() > 80:
        h_work, c5 = nesterov_gradient(best_h, ts1, 500.0, 3000, 0.02)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_work.copy()
            best_n = n1

    if time_left() > 60:
        h_work, c5 = coord_descent_fast(best_h, ts1, 40, rng)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_work.copy()
            best_n = n1

    if time_left() > 250:
        n2 = 1080
        ts2 = n2 / 2.0
        x_new2 = np.linspace(0, 2, n2, endpoint=False)
        h_up2 = np.interp(x_new2, np.linspace(0, 2, best_n, endpoint=False), best_h)
        h_up2 = project_to_sum(h_up2, ts2)
        c5_up2 = compute_c5(h_up2)
        if c5_up2 < best_c5:
            best_c5 = c5_up2
            best_h = h_up2.copy()
            best_n = n2
        h_work = h_up2.copy()
        if time_left() > 150:
            h_work, c5 = adam_optimize(h_work, ts2, 5.0, 100.0, 4000, 0.01)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_work.copy()
                best_n = n2
        if time_left() > 100:
            h_work, c5 = adam_optimize(best_h, ts2, 100.0, 600.0, 3000, 0.005)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_work.copy()
                best_n = n2
        if time_left() > 50:
            h_work, c5 = coord_descent_fast(best_h, ts2, 25, rng)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_work.copy()
                best_n = n2

    if time_left() > 40:
        nf = best_n
        tsf = nf / 2.0
        h_work, c5 = adam_optimize(best_h, tsf, 500.0, 3000.0, 3000, 0.002)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_work.copy()
            best_n = nf

    if time_left() > 25:
        h_work, c5 = coord_descent_fast(best_h, best_n / 2.0, 30, rng)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_work.copy()
            best_n = len(h_work)

    nfinal = len(best_h)
    tsfinal = nfinal / 2.0
    best_h = project_to_sum(best_h, tsfinal)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, nfinal)
