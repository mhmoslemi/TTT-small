```python
def binary_local_search(h, n, target_sum):
    # h is binary, sum(h) = target_sum
    c = np.correlate(h, 1-h, mode='full')  # length 2n-1
    max_c = np.max(c)
    
    zeros = np.where(h == 0)[0]
    ones = np.where(h == 1)[0]
    
    # Precompute flip-up deltas for each zero position
    # flip_up[i] = change in c when h[i] goes 0->1
    flip_up = np.zeros((len(zeros), 2*n-1))
    for idx, i in enumerate(zeros):
        # For k >= 0: decrease by h[i+k] for k = 0..n-1-i
        ks = np.arange(0, n-i)
        flip_up[idx, n-1+ks] -= h[i+ks]
        # For k < 0 (m = -k): increase by (1-h[i-m]) for m = 1..i
        ms = np.arange(1, i+1)
        flip_up[idx, n-1-ms] += (1-h[i-ms])
    
    # Precompute flip-down deltas for each one position
    flip_down = np.zeros((len(ones), 2*n-1))
    for idx, j in enumerate(ones):
        # For k >= 0: decrease by (1-h[j-k]) for k = 0..j
        ks = np.arange(0, j+1)
        flip_down[idx, n-1+ks] -= (1-h[j-ks])
        # For k < 0 (m = -k): increase by h[j+m] for m = 1..n-1-j
        ms = np.arange(1, n-j)
        flip_down[idx, n-1-ms] += h[j+ms]
    
    # Now search for best pair
    best_max = max_c
    best_pair = None
    for idx_i, i in enumerate(zeros):
        base = c + flip_up[idx_i]
        for idx_j, j in enumerate(ones):
            new_c = base + flip_down[idx_j]
            m = np.max(new_c)
            if m < best_max:
                best_max = m
                best_pair = (idx_i, idx_j)
    
    return best_pair, best_max
```

```python
import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.count_nonzero(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, target_sum):
    """Convert h to binary with exactly target_sum ones."""
    h_bin = (h >= 0.5).astype(float)
    # Fix sum if needed
    current_sum = np.sum(h_bin)
    if current_sum > target_sum:
        # Need to flip some 1s to 0s
        excess = current_sum - target_sum
        ones_idx = np.where(h_bin == 1)[0]
        # Flip the ones with smallest h values (closest to 0.5)
        order = np.argsort(h[ones_idx])[:int(excess)]
        h_bin[ones_idx[order]] = 0
    elif current_sum < target_sum:
        deficit = target_sum - current_sum
        zeros_idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros_idx])[:int(deficit)]
        h_bin[zeros_idx[order]] = 1
    return h_bin

def grad_c_at_lag(h, k):
    """Gradient of c_k = sum h[i+k](1-h[i]) with respect to h."""
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        if k < n:
            g[k:] += (1.0 - h[:n-k])
            g[:n-k] -= h[k:]
    else:
        m = -k
        if m < n:
            g[m:] += (1.0 - h[:n-m])
            g[:n-m] -= h[m:]
    return g

def subgradient_optimize(h0, n, target_sum, n_iters, lr_init=0.1):
    """Subgradient method for minimizing max_k c_k."""
    h = h0.copy()
    lr = lr_init
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = np.argmax(corr) - (n - 1)
        g = grad_c_at_lag(h, k_star)
        h_new = h - lr * g
        h_new = project_to_sum(h_new, target_sum)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            lr = min(lr * 1.01, 0.5)
        else:
            lr *= 0.95
        h = h_new
    
    return best_h, best_c5

def binary_local_search(h_bin, n, target_sum, n_sweeps=100, rng=np.random.RandomState(42)):
    """Local search on binary solutions using pair flips."""
    h = h_bin.copy()
    c = np.correlate(h, 1.0 - h, mode='full').astype(float)
    max_c = float(np.max(c))
    
    zeros = np.where(h == 0)[0]
    ones = np.where(h == 1)[0]
    
    for sweep in range(n_sweeps):
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        
        if len(zeros) < 1 or len(ones) < 1:

        # Precompute flip-up deltas for zero positions
        n_zeros = len(zeros)
        n_ones = len(ones)
        
        # For each zero position i, compute the delta to c when h[i] goes 0->1
        # delta_up[i_idx, lag_index]
        delta_up = np.zeros((n_zeros, 2*n-1))
        for ii, i in enumerate(zeros):
            # Positive lags k = 0..n-1-i: c_k decreases by h[i+k]
            ks = np.arange(0, n-i)
            delta_up[ii, n-1+ks] -= h[i+ks]
            # Negative lags m = 1..i: c_{-m} increases by (1-h[i-m])
            if i > 0:
                ms = np.arange(1, i+1)
                delta_up[ii, n-1-ms] += (1.0 - h[i-ms])
        
        # For each one position j, compute the delta to c when h[j] goes 1->0
        delta_down = np.zeros((n_ones, 2*n-1))
        for jj, j in enumerate(ones):
            # Positive lags k = 0..j: c_k decreases by (1-h[j-k])
            ks = np.arange(0, j+1)
            delta_down[jj, n-1+ks] -= (1.0 - h[j-ks])
            # Negative lags m = 1..n-1-j: c_{-m} increases by h[j+m]
            if n-1-j > 0:
                ms = np.arange(1, n-j)
                delta_down[jj, n-1-ms] += h[j+ms]
        
        # Search for best pair
        best_max = max_c
        best_i = -1
        best_j = -1
        
        for ii in range(n_zeros):
            base = c + delta_up[ii]
            # For each j, compute max(c + delta_up[ii] + delta_down[jj])
            new_maxes = np.max(base[None, :] + delta_down, axis=1)
            j_best = np.argmin(new_maxes)
            if new_maxes[j_best] < best_max:
                best_max = new_maxes[j_best]
                best_i = ii
                best_j = j_best
        
        if best_i >= 0:
            i_flip = zeros[best_i]
            j_flip = ones[best_j]
            h[i_flip] = 1.0
            h[j_flip] = 0.0
            c = c + delta_up[best_i] + delta_down[best_j]
            max_c = best_max
        else:
            break
    
    return h, max_c * (2.0 / n)
```

```python
def multi_lag_subgradient(h, n, target_sum, n_iters, top_k=5, lr_init=0.05):

    lr = lr_init
    best_h = h.copy()
    best_c5 = compute_c5(h)

        corr = np.correlate(h, 1.0 - h, mode='full')
        # Get top_k lags
        top_indices = np.argsort(corr)[-top_k:]
        weights = np.exp((corr[top_indices] - corr[top_indices].max()) / 0.01)
        weights /= weights.sum()
```
