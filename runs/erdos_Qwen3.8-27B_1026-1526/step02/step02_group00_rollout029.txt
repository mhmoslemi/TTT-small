def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m_val))
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    n_corr = len(corr)
    grad = np.zeros(n)
    for idx in range(n_corr):
        if w[idx] < 1e-15:
            continue
        shift = idx - (n - 1)
        if shift == 0:
            grad += w[idx] * (1.0 - 2.0 * h)
        else:
            i_range = range(max(0, -shift), min(n, n - shift))
            hi = np.array([i + shift for i in i_range])
            li = np.array([i for i in i_range])
            grad[li] += w[idx] * h[hi]
            grad[hi] -= w[idx] * (1.0 - h[li])
    return obj, grad

def slsqp_obj(h, alpha):
    return smooth_max_and_grad(h, alpha)[0]

def slsqp_grad(h, alpha):
    return smooth_max_and_grad(h, alpha)[1]

def slsqp_sum_eq(x, target_sum):
    return np.sum(x) - target_sum

def slsqp_sum_jac(x, target_sum):
    return np.ones(len(x))

def slsqp_local(h0, n, target_sum, alpha, maxiter):
    cons = ([{'type': 'eq', 'fun': slsqp_sum_eq, 'jac': slsqp_sum_jac}],)
    bounds = [(0.0, 1.0)] * n
    def cons_with_target(x):
        return [{'type': 'eq', 'fun': lambda x, t=target_sum: np.sum(x) - t,
                 'jac': lambda x, t=target_sum: np.ones(len(x))}]
    try:
        res = minimize(lambda x: slsqp_obj(x, alpha), h0.copy(), method='SLSQP',
                       jac=lambda x: slsqp_grad(x, alpha),
                       bounds=bounds, constraints=cons_with_target,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def upscale_binary(h, n_old, n_new):
    x_old = np.arange(n_old, dtype=float)
    x_new = (np.arange(n_new, dtype=float) * n_old / n_new)
    h_interp = np.interp(x_new, x_old, h)
    target = n_new // 2
    h_bin = (h_interp >= 0.5).astype(float)
    diff = target - int(np.sum(h_bin))
    if diff > 0:
        below = np.where(h_bin < 0.5)[0]
        order = np.argsort(h_interp[below])[::-1]
        for i in range(min(diff, len(order))):
            h_bin[below[order[i]]] = 1.0
    elif diff < 0:
        above = np.where(h_bin > 0.5)[0]
        order = np.argsort(h_interp[above])
        for i in range(min(-diff, len(order))):
            h_bin[above[order[i]]] = 0.0
    return h_bin

    starts = []
    starts.append(np.full(n, 0.5))
    x = np.linspace(0, 2.0, n, endpoint=False)
    for freq in [1, 2, 3, 4, 5, 6, 7, 8, 10, 12]:
        s = 0.5 + 0.35 * np.sin(2.0 * np.pi * freq * x / 2.0)
        starts.append(project_to_sum(s, target_sum))
    for nblks in [2, 3, 4, 5, 6, 7, 8, 10, 12]:
        blk = max(1, n // (nblks * 2))
        h = np.zeros(n)
        for i in range(0, n, 2 * blk):
            h[i:min(i + blk, n)] = 1.0
        starts.append(project_to_sum(h, target_sum))
    for _ in range(6):
        a1 = rng.uniform(0.15, 0.45)
        a2 = rng.uniform(0.05, 0.35)
        f1 = rng.choice([1, 2, 3, 4, 5, 6, 7, 8, 10])
        f2 = rng.choice([2, 3, 4, 5, 6, 7, 8, 9, 10, 12, 14])
        p1 = rng.uniform(0, 2 * np.pi)
        p2 = rng.uniform(0, 2 * np.pi)
        h = 0.5 + a1 * np.sin(2.0 * np.pi * f1 * x / 2.0 + p1) + a2 * np.cos(2.0 * np.pi * f2 * x / 2.0 + p2)
        starts.append(project_to_sum(h, target_sum))
    if best_h is not None and best_n > 0:
        if best_n == n:
            starts.append(best_h.copy())
        else:
            x_old = np.linspace(0, 2, best_n, endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            starts.append(project_to_sum(np.interp(x_new, x_old, best_h, period=2.0), target_sum))
    hb = random_binary(int(target_sum), rng)
    starts.append(hb)
    return starts

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    def time_left():
        return budget_s - (time.time() - t_start)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n0 = best_n
    ts0 = n0 / 2.0

    if time_left() > 100:
        h_bin_init = (best_h > 0.5).astype(float)
        diff = ts0 - h_bin_init.sum()
        if diff > 0:
            idx = np.argsort(best_h)[::-1]
            for i in range(int(diff)):
                h_bin_init[idx[i]] = 1.0
        elif diff < 0:
            idx = np.argsort(best_h)
            for i in range(int(-diff)):
                h_bin_init[idx[i]] = 0.0
        h_ls, c5_ls, _ = efficient_local_search(h_bin_init, n0, time_left() - 10, rng, tabu_len=40)
        if c5_ls * (2.0 / n0) < best_c5:
            best_c5 = c5_ls * (2.0 / n0)
            best_h = h_ls.copy()
            best_n = n0

    if time_left() > 80:
        h_ml, c5_ml = multi_lag_local_search(best_h, best_n, time_left() - 10, rng, num_lags=3)
        c5_ml_real = compute_c5(h_ml)
        if c5_ml_real < best_c5:
            best_c5 = c5_ml_real
            best_h = h_ml.copy()

    resolutions = [360, 480, 600]
    for n_tr in resolutions:
        if time_left() < 120:
            break
        ts_tr = n_tr / 2.0
        if abs(ts_tr - round(ts_tr)) > 1e-6:
            continue

        h_up = upscale_binary(best_h, best_n, n_tr)
        h_up = project_to_sum(h_up, ts_tr)
        c5_up = compute_c5(h_up)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up.copy()
            best_n = n_tr

        h_bin = (h_up > 0.5).astype(float)
        diff = ts_tr - h_bin.sum()
        if diff > 0:
            idx = np.argsort(h_up)[::-1]
            for i in range(int(diff)):
                h_bin[idx[i]] = 1.0
        elif diff < 0:
            idx = np.argsort(h_up)
            for i in range(int(-diff)):
                h_bin[idx[i]] = 0.0

        if time_left() > 80:
            h_ls2, c5_ls2, _ = efficient_local_search(h_bin, n_tr, min(time_left() - 20, 300), rng, tabu_len=50)
            c5_ls2_real = compute_c5(h_ls2)
            if c5_ls2_real < best_c5:
                best_c5 = c5_ls2_real
                best_h = h_ls2.copy()
                best_n = n_tr

        if time_left() > 60:
            h_ml2, c5_ml2 = multi_lag_local_search(best_h, best_n, min(time_left() - 10, 200), rng, num_lags=3)
            c5_ml2_real = compute_c5(h_ml2)
            if c5_ml2_real < best_c5:
                best_c5 = c5_ml2_real
                best_h = h_ml2.copy()
                best_n = n_tr

    if time_left() > 60:
        n_f = best_n
        ts_f = n_f / 2.0
        for alpha in [50, 200, 1000]:
            if time_left() < 25:
                break
            h_pol = slsqp_local(best_h, n_f, ts_f, alpha, maxiter=500)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()
                best_n = n_f

    if time_left() > 40:
        n_f = best_n
        ts_f = n_f / 2.0
        h_bin_f = (best_h > 0.5).astype(float)
        diff = ts_f - h_bin_f.sum()
        if diff > 0:
            idx = np.argsort(best_h)[::-1]
            for i in range(int(diff)):
                h_bin_f[idx[i]] = 1.0
        elif diff < 0:
            idx = np.argsort(best_h)
            for i in range(int(-diff)):
                h_bin_f[idx[i]] = 0.0
        h_ls3, c5_ls3, _ = efficient_local_search(h_bin_f, n_f, time_left() - 5, rng, tabu_len=60)
        c5_ls3_real = compute_c5(h_ls3)
        if c5_ls3_real < best_c5:
            best_c5 = c5_ls3_real
            best_h = h_ls3.copy()
            best_n = n_f

    if time_left() > 30:
        n_f = best_n
        ts_f = n_f / 2.0
        starts = make_diverse_starts(n_f, ts_f, rng, best_h, best_n)
        for h_start in starts:
            if time_left() < 15:
                break
            h_b = (h_start > 0.5).astype(float)
            diff = ts_f - h_b.sum()
            if diff > 0:
                idx = np.argsort(h_start)[::-1]
                for i in range(int(diff)):
                    h_b[idx[i]] = 1.0
            elif diff < 0:
                idx = np.argsort(h_start)
                for i in range(int(-diff)):
                    h_b[idx[i]] = 0.0
            h_ls4, c5_ls4, _ = efficient_local_search(h_b, n_f, min(time_left() - 5, 60), rng, tabu_len=30)
            c5_ls4_real = compute_c5(h_ls4)
            if c5_ls4_real < best_c5:
                best_c5 = c5_ls4_real
                best_h = h_ls4.copy()
                best_n = n_f

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
