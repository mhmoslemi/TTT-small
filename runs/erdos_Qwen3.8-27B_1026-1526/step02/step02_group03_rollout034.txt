from scipy.interpolate import interp1d

    dx = 2.0 / n

    v = values / tau
    vmax = np.max(v)
    shifted = v - vmax
    return tau * (vmax + np.log(np.sum(np.exp(shifted))))

    target = n / 2.0
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        if np.sum(proj) > target:
            hi = mid
        else:
            lo = mid

        result[idx] = np.clip(result[idx] + diff, 0.0, 1.0)

    g = 1.0 - x
    corr = np.correlate(x, g, mode="full")
    obj = smooth_max(corr, tau) * dx
    cv = np.sum(x) - n / 2.0
    obj += penalty * cv * cv
    
    # Gradient
    v = corr / tau
    vmax = np.max(v)
    exp_v = np.exp(v - vmax)
    w = exp_v / np.sum(exp_v)

    m_idx = np.arange(n)[None, :].astype(float)

        return g

        h = rng.rand(n)
        starts.append(h)
    for _ in range(4):
        center = rng.uniform(0.1 * n, 0.9 * n)
        width = rng.uniform(0.08 * n, 0.3 * n)
        h = np.exp(-0.5 * ((np.arange(n, dtype=float) - center) / width) ** 2)
        h = 0.1 + 0.8 * h
        starts.append(np.clip(h, 0, 1))
    return starts[:num_starts]

    phase2_end = start_time + budget_s * 0.48
    phase3_end = start_time + budget_s * 0.63
    phase4_end = start_time + budget_s * 0.78
    phase5_end = start_time + budget_s * 0.94

    # Phase 1: L-BFGS-B on parent resolution
    if parent_h is not None and time.time() < phase1_end:
        tau_schedule = [0.4, 0.2, 0.1, 0.05, 0.025, 0.012, 0.006, 0.003]
        for tau in tau_schedule:
            if time.time() > phase1_end:
                break
            iters = 150 if tau > 0.05 else 250
            h_new = lbfgsb_optimize(best_h, n_cur, tau, iters, 5000.0)
            h_new = project_feasible(h_new, n_cur)
            c5 = evaluate_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()

    if time.time() < phase2_end:
        resolutions = [41, 45, 49, 53, 57, 61, 65, 69, 73, 81, 85, 89, 93, 97, 101, 105, 111, 117, 123, 129]
        for n_try in resolutions:
            if time.time() > phase2_end - 10:
                break
            if n_try == n_cur:
                continue
            x_old = np.linspace(0, 2, len(best_h), endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            f = interp1d(x_old, best_h, kind='cubic', fill_value='extrapolate', bounds_error=False)
            h_init = np.clip(f(x_new), 0, 1)
            h_init = project_feasible(h_init, n_try)
            tau_sched = [0.15, 0.07, 0.03, 0.01]
            for tau in tau_sched:
                h_init = lbfgsb_optimize(h_init, n_try, tau, 120, 5000.0)
                h_init = project_feasible(h_init, n_try)
            c5 = evaluate_c5(h_init)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_init.copy()
                n_cur = n_try

        # Try random starts at best resolution
        if time.time() < phase2_end:
            starts = generate_random_starts(n_cur, rng, 8)
            for h0 in starts:
                if time.time() > phase2_end - 5:
                    break
                h0 = project_feasible(h0, n_cur)
                tau_sched = [0.1, 0.04, 0.01]
                for tau in tau_sched:
                    h0 = lbfgsb_optimize(h0, n_cur, tau, 100, 5000.0)
                    h0 = project_feasible(h0, n_cur)
                c5 = evaluate_c5(h0)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h0.copy()

    if time.time() < phase3_end:
        n = len(best_h)
        tau_schedule = [0.003, 0.0015, 0.0008, 0.0004, 0.0002]

            if time.time() > phase3_end - 3:
                break

            h_new = project_feasible(h_new, n)

            if c5 < best_c5:

                best_h = h_new.copy()
