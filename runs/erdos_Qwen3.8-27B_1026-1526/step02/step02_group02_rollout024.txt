from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 0.999999
            if not np.any(mask):
                break
            step = diff / np.sum(mask)
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            mask = h > 0.000001
            if not np.any(mask):
                break
            step = (-diff) / np.sum(mask)
            h[mask] = np.maximum(h[mask] - step, 0.0)
    h = np.clip(h, 0.0, 1.0)
    # Exact correction
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        step = diff / n
        h = np.clip(h + step, 0.0, 1.0)
    return h

def slsqp_obj(x):
    return x[-1]

def slsqp_ineq(x):
    h = x[:-1]
    t = x[-1]
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return t - c * dx

def slsqp_eq(x):
    h = x[:-1]
    return np.sum(h) - len(h) / 2.0

def slsqp_eq_jac(x):
    n = len(x) - 1
    jac = np.zeros(len(x))
    jac[:n] = 1.0
    return jac

def slsqp_optimize(h_init, n, maxiter=5000, t_margin=0.1):
    c5_init = compute_c5_from_h(h_init)
    x0 = np.concatenate([h_init, [c5_init + t_margin]])
    bounds = [(0.0, 1.0)] * n + [(max(0.001, c5_init * 0.1), 2.0)]
    constraints = [
        {'type': 'ineq', 'fun': slsqp_ineq},
        {'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac}
    ]
    result = minimize(slsqp_obj, x0, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
    if result.x is not None:
        h_out = project_feasible(result.x[:-1], n)
        return h_out, compute_c5_from_h(h_out)
    return h_init, c5_init

def smooth_obj_lse(x, T, penalty):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full") * dx
    c_max = c.max()
    shifted = np.clip(T * (c - c_max), -500, 500)
    lse = c_max + np.log(np.sum(np.exp(shifted))) / T
    sum_viol = np.sum(x) - n / 2.0
    return lse + penalty * sum_viol ** 2

def lbfgsb_optimize(h_init, T, penalty, maxiter=5000):
    n = len(h_init)
    bounds = [(0.0, 1.0)] * n
    result = minimize(
        smooth_obj_lse, h_init, args=(T, penalty),
        method='L-BFGS-B', bounds=bounds,
        options={'maxiter': maxiter, 'gtol': 1e-14, 'ftol': 1e-18}

    if result.x is not None:
        h_out = project_feasible(result.x, n)
        return h_out, compute_c5_from_h(h_out)

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time


def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(3000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 0.9999999
            if not np.any(mask):
                break
            step = diff / np.sum(mask)
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            mask = h > 0.0000001
            if not np.any(mask):
                break
            step = (-diff) / np.sum(mask)
            h[mask] = np.maximum(h[mask] - step, 0.0)
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        step = diff / n
        h = np.clip(h + step, 0.0, 1.0)
    return h


def slsqp_obj(x):
    return x[-1]


def slsqp_ineq(x):
    h = x[:-1]
    t = x[-1]
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return t - c * dx


def slsqp_eq(x):
    h = x[:-1]
    return np.sum(h) - len(h) / 2.0


def slsqp_eq_jac(x):
    n = len(x) - 1
    jac = np.zeros(len(x))
    jac[:n] = 1.0
    return jac


def slsqp_optimize(h_init, n, maxiter=5000, t_margin=0.1):
    c5_init = compute_c5_from_h(h_init)
    x0 = np.concatenate([h_init, [c5_init + t_margin]])
    bounds = [(0.0, 1.0)] * n + [(max(0.001, c5_init * 0.1), 2.0)]
    constraints = [
        {'type': 'ineq', 'fun': slsqp_ineq},
        {'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac}
    ]
    result = minimize(slsqp_obj, x0, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
    if result.x is not None:
        h_out = project_feasible(result.x[:-1], n)
        return h_out, compute_c5_from_h(h_out)
    return h_init, c5_init


def smooth_obj_lse(x, T, penalty):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full") * dx
    c_max = c.max()
    shifted = np.clip(T * (c - c_max), -500, 500)
    lse = c_max + np.log(np.sum(np.exp(shifted))) / T
    sum_viol = np.sum(x) - n / 2.0
    return lse + penalty * sum_viol ** 2

    n = len(h_init)
    bounds = [(0.0, 1.0)] * n
    result = minimize(
        smooth_obj_lse, h_init, args=(T, penalty),
        method='L-BFGS-B', bounds=bounds,
        options={'maxiter': maxiter, 'gtol': 1e-14, 'ftol': 1e-18}
    )
    if result.x is not None:

        h_new, c5_new = slsqp_optimize(h_interp, n_try, maxiter=3000, t_margin=0.05)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
            best_n = n_try
        else:
            continue
        if remaining() > 80:
            for T in [80, 300]:
                if remaining() < 40:
                    break
                h_new, c5_new = lbfgsb_optimize(best_h, T, 50 + T * 0.5, maxiter=1500)
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new
                    best_n = len(best_h)
        if remaining() > 40 and best_n % 2 == 0:
            h_new, c5_new = ap_optimize(best_h[:best_n // 2], maxiter=2000)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
                best_n = len(best_h)

    n_perturb = 50
    for attempt in range(n_perturb):
        if remaining() < 35:
            break

        frac = attempt / max(1, n_perturb)
        if frac < 0.4:
            scale = 0.12 * (1.0 - frac / 0.4) ** 0.6 + 0.015
        elif frac < 0.7:
            scale = 0.04 * (1.0 - (frac - 0.4) / 0.3) ** 0.5 + 0.008
        else:
            scale = 0.012 * (1.0 - (frac - 0.7) / 0.3) ** 0.5 + 0.002

        ptype = int(rng.integers(0, 7))
        h_pert = make_perturbation(best_h, scale, rng, best_n, ptype)
        h_pert = project_feasible(h_pert, best_n)

        if attempt % 4 == 0 or frac < 0.3:
            h_new, c5_new = slsqp_optimize(h_pert, best_n, maxiter=2000, t_margin=0.03)
        elif attempt % 4 == 1:
            h_new, c5_new = lbfgsb_optimize(h_pert, 150, 150, maxiter=1500)
            h_new, c5_new = slsqp_optimize(h_new, best_n, maxiter=1000, t_margin=0.02)
        elif attempt % 4 == 2:
            h_new, c5_new = lbfgsb_optimize(h_pert, 500, 300, maxiter=2000)
            h_new, c5_new = slsqp_optimize(h_new, best_n, maxiter=1500, t_margin=0.02)
        else:
            if best_n % 2 == 0:
                a_pert = h_pert[:best_n // 2].copy()
                h_new, c5_new = ap_optimize(a_pert, maxiter=2000)
            else:
                h_new, c5_new = slsqp_optimize(h_pert, best_n, maxiter=1500, t_margin=0.03)

        if c5_new < best_c5 - 1e-12:
            best_h = h_new
            best_c5 = c5_new
            best_n = len(best_h)

    if remaining() > 100:
        n_fine = best_n + 90
        h_interp = interpolate_h(best_h, n_fine)
        h_new, c5_new = slsqp_optimize(h_interp, n_fine, maxiter=3000, t_margin=0.03)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
            best_n = n_fine
        if remaining() > 50 and best_n % 2 == 0:
            h_new, c5_new = ap_optimize(best_h[:best_n // 2], maxiter=2000)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
                best_n = len(best_h)

    if remaining() > 60:
        for T in [2000, 5000]:
            if remaining() < 30:
                break
            h_new, c5_new = lbfgsb_optimize(best_h, T, T * 0.4, maxiter=1500)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
                best_n = len(best_h)

    if remaining() > 20:
        best_h, best_c5 = slsqp_optimize(best_h, best_n, maxiter=8000, t_margin=0.005)
        best_n = len(best_h)

    if remaining() > 5:
        best_h = project_feasible(best_h, len(best_h))
        best_c5 = compute_c5_from_h(best_h)

    return best_h, best_c5, len(best_h)
