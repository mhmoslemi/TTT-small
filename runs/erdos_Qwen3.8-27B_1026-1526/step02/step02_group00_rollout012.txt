           grad[i] = (t1 + t2) * dx
       
       grad += 2.0 * lam * (np.sum(x) - target)
       return obj, grad

   def lbfgsb_run(h0, n, target, alpha, lam, maxiter=300):
       h0 = project_to_sum(h0, target)
       bounds = [(0.0, 1.0)] * n
       try:
           res = minimize(
               lambda x: smooth_obj_and_grad(x, alpha, lam, target)[0],
               h0,
               jac=lambda x: smooth_obj_and_grad(x, alpha, lam, target)[1],
               method='L-BFGS-B',
               bounds=bounds,
               options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10, 'maxls': 30}
           )
           return project_to_sum(res.x, target)
       except:
           return project_to_sum(h0, target)
