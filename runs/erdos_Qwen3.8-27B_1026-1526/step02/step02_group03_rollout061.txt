    if abs(s - target) < 1e-12:

        else:

    best = h.copy()
    best_c5 = evaluate_c5(best)
    x = np.linspace(0, 2, n, endpoint=False)

            if freq_idx*n >= n:  # Nyquist

**Innovation: Subgradient method with Nesterov acceleration**

The function f(h) = max_k g_k(h) is a max of differentiable functions. We can use the subgradient method:
- At each point h, the subgradient is the gradient of the active function (the one achieving the max)
- With Nesterov acceleration, this converges at rate O(1/T) for convex functions (our function isn't convex, but it might still help)

Let me implement this properly:

For g_k(h) where k is the lag index in np.correlate:
The correlation at index m (0-indexed, m from 0 to 2n-2) corresponds to offset = m - (n-1).

If offset >= 0: g_m(h) = Σ_{i=0}^{n-1-offset} h[i+offset] * (1-h[i]) * dx
Gradient: ∂g_m/∂h[j] = dx * [(1-h[j-offset])*1_{j>=offset} - h[j+offset]*1_{j+offset<n}]

Gradient: ∂g_m/∂h[j] = dx * [(1-h[j-d])*1_{j>=d} - h[j+d]*1_{j+d<n}]

def nesterov_subgradient(h0, n, iterations=5000, lr_init=0.05, lr_decay=0.999):
    best = h0.copy()
    best_c5 = evaluate_c5(best)
    y = h0.copy()  # Nesterov extrapolation point
    v = np.zeros(n)  # momentum
    lr = lr_init
    target = n / 2.0
    
    for t in range(iterations):
        # Compute overlaps at y
        dx = 2.0 / n
        g_y = 1.0 - y
        corr = np.correlate(y, g_y, mode="full") * dx
        m_max = np.argmax(corr)
        offset = m_max - (n - 1)
        d = abs(offset)
        
        # Compute gradient
        grad = np.zeros(n)
        if d == 0:
            # g = sum h[i]*(1-h[i]) * dx
            # grad[j] = dx * (1 - 2*h[j])
            grad = dx * (1.0 - 2.0 * y)
        else:
            for j in range(n):
                grad_j = 0.0
                if j >= d:
                    grad_j += (1.0 - y[j - d])
                if j + d < n:
                    grad_j -= y[j + d]
                grad[j] = grad_j * dx
        
        # Nesterov momentum
        v = 0.9 * v - lr * grad
        y_new = y + v
        y_new = project_feasible(y_new, n)
        
        # Update best
        c5 = evaluate_c5(y_new)
        if c5 < best_c5:
            best_c5 = c5
            best = y_new.copy()
        
        y = y_new
        lr = max(lr * lr_decay, 0.001)
    
    return best, best_c5

    best_c5 = evaluate_c5(best)
    h = h0.copy()
    y = h0.copy()
    lr = lr_init
    for t in range(iterations):
        grad, _ = compute_subgradient(y, n)
        beta_t = (1.0 + np.sqrt(1.0 + 4.0 * (t + 1) ** 2 * lr_decay ** 2)) / (2.0 * (t + 1) * lr_decay)
        beta_t = min(beta_t, 1.5)
        y_new = y - lr * grad
        y_new = project_feasible(y_new, n)
        if t > 0:
            y_new = y + beta_t * (y_new - h)
            y_new = project_feasible(y_new, n)
        h = y
        y = y_new
        c5 = evaluate_c5(y)
        if c5 < best_c5:
            best_c5 = c5
            best = y.copy()
        lr = max(lr * lr_decay, 0.0003)
    return best, best_c5

def interpolate_to_n(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    f = interp1d(x_old, h, kind='linear', fill_value='extrapolate')
    h_new = np.clip(f(x_new), 0, 1)
    return project_feasible(h_new, n_new)

def spatial_descent(h, n, sweeps=3):
    best = h.copy()
    best_c5 = evaluate_c5(best)
    step_sizes = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    for sweep in range(sweeps):
        improved = False
        for i in range(n):
            for s in step_sizes:
                for sign in [1, -1]:
                    h_new = best.copy()
                    h_new[i] = best[i] + sign * s
                    if h_new[i] < 0.0 or h_new[i] > 1.0:
                        continue
                    h_new = project_feasible(h_new, n)
                    c5 = evaluate_c5(h_new)
                    if c5 < best_c5:
                        best = h_new.copy()
                        best_c5 = c5
                        improved = True
        if not improved:
            break
    return best, best_c5

def fourier_refine(h, n, num_freq, iterations):
    best = h.copy()
    best_c5 = evaluate_c5(best)
    for it in range(iterations):
        changed = False
        for freq_idx in range(1, num_freq + 1):
            fft_best = np.fft.rfft(best - 0.5)
            orig = fft_best[freq_idx].copy()
            for scale in [0.7, 0.85, 0.95, 1.05, 1.15, 1.3]:
                fft_try = fft_best.copy()
                fft_try[freq_idx] = orig * scale
                h_try = 0.5 + np.fft.irfft(fft_try, n=n)
                h_try = project_feasible(h_try, n)
                c5 = evaluate_c5(h_try)
                if c5 < best_c5:
                    best = h_try.copy()
                    best_c5 = c5
                    changed = True
            magnitude = np.abs(orig)
            if magnitude > 0.005:
                phase = np.angle(orig)
                for dp in [-0.4, -0.2, -0.1, 0.1, 0.2, 0.4]:
                    fft_try = fft_best.copy()
                    fft_try[freq_idx] = magnitude * np.exp(1j * (phase + dp))
                    h_try = 0.5 + np.fft.irfft(fft_try, n=n)
                    h_try = project_feasible(h_try, n)
                    c5 = evaluate_c5(h_try)
                    if c5 < best_c5:
                        best = h_try.copy()
                        best_c5 = c5
                        changed = True
        if not changed:
            break
    return best, best_c5

def multi_subgradient(h0, n, iterations, lr_init, lr_decay, num_active=3):
    """Use weighted average of top-m subgradients for better direction"""
    best = h0.copy()
    best_c5 = evaluate_c5(best)
    h = h0.copy()
    v = np.zeros(n)
    lr = lr_init
    for t in range(iterations):
        dx = 2.0 / n
        g_h = 1.0 - h
        corr = np.correlate(h, g_h, mode="full") * dx
        top_idx = np.argsort(corr)[-num_active:]
        weights = np.exp(corr[top_idx] / 0.01)
        weights /= np.sum(weights)
        grad_total = np.zeros(n)
        for w, mi in zip(weights, top_idx):
            offset = mi - (n - 1)
            d = abs(offset)
            if d == 0:
                g = dx * (1.0 - 2.0 * h)
            else:
                g = np.zeros(n)
                g[d:] += (1.0 - h[:n - d])
                g[:n - d] -= h[d:]
                g *= dx
            grad_total += w * g
        v = 0.85 * v - lr * grad_total
        h_new = h + v
        h_new = project_feasible(h_new, n)
        c5 = evaluate_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best = h_new.copy()
        h = h_new
        lr = max(lr * lr_decay, 0.0002)
    return best, best_c5

def block_perturb_search(h, n, rng, max_rounds=60):
    best = h.copy()
    best_c5 = evaluate_c5(best)
    for r in range(max_rounds):
        block_len = rng.randint(5, min(15, n // 4))
        start = rng.randint(0, n - block_len)
        h_try = best.copy()
        h_try[start:start + block_len] += rng.randn(block_len) * 0.02
        h_try = project_feasible(h_try, n)
        h_opt = slsqp_optimize(h_try, n, tau=0.005, maxiter=40)
        c5 = evaluate_c5(h_opt)
        if c5 < best_c5:
            best = h_opt.copy()
            best_c5 = c5
    return best, best_c5

def active_set_sqp(h0, n, rounds=10, active_size=5):
    """Iteratively optimize with a smooth approximation of the active set max"""
    best = h0.copy()
    best_c5 = evaluate_c5(best)
    dx = 2.0 / n
    for r in range(rounds):
        g_best = 1.0 - best
        corr = np.correlate(best, g_best, mode="full") * dx
        top_idx = np.argsort(corr)[-active_size:]
        tau = max(0.001, 0.02 * (0.5 ** r))
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_constraint, 'args': (n,)}]
        def active_obj(x):
            gx = 1.0 - x
            cx = np.correlate(x, gx, mode="full") * dx
            active = cx[top_idx]
            amax = np.max(active)
            return amax + tau * np.log(np.sum(np.exp(np.clip((active - amax) / tau, -50, 50))))
        try:
            res = minimize(active_obj, best, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 150, 'ftol': 1e-13})
            candidate = project_feasible(res.x, n)
            c5 = evaluate_c5(candidate)
            if c5 < best_c5:
                best_c5 = c5
                best = candidate.copy()
        except Exception:
            pass
    return best, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    start_time = time.time()

    best_h = None
    best_c5 = float('inf')

    try:
        parent_h = np.array(initial_h_values, dtype=float)
    except Exception:
        parent_h = None

    if parent_h is not None:
        best_h = parent_h.copy()
        best_c5 = evaluate_c5(best_h)
    else:
        n0 = 77
        x0 = np.linspace(0, 2, n0, endpoint=False)
        best_h = 0.5 + 0.25 * np.sin(2.0 * np.pi * x0) + 0.1 * np.sin(4.0 * np.pi * x0 + 0.7)
        best_h = project_feasible(best_h, n0)
        best_c5 = evaluate_c5(best_h)

    n = len(best_h)

    # Phase 1: SLSQP with geometrically decreasing tau (40% budget)
    tau_schedule = [0.05, 0.025, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0003]
    for tau in tau_schedule:
        if time.time() - start_time > budget_s * 0.40:
            break
        h_opt = slsqp_optimize(best_h, n, tau=tau, maxiter=400, ftol=1e-13)
        c5 = evaluate_c5(h_opt)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt

    # Phase 2: Multi-subgradient with momentum (15% budget)
    if time.time() - start_time < budget_s * 0.55:
        n = len(best_h)
        h_ms, c5_ms = multi_subgradient(best_h, n, iterations=5000, lr_init=0.015, lr_decay=0.9997, num_active=3)
        if c5_ms < best_c5:
            best_c5 = c5_ms
            best_h = h_ms

    # Phase 3: Active set SQP (10% budget)
    if time.time() - start_time < budget_s * 0.65:
        n = len(best_h)
        h_as, c5_as = active_set_sqp(best_h, n, rounds=10, active_size=4)
        if c5_as < best_c5:
            best_c5 = c5_as
            best_h = h_as

    # Phase 4: Try nearby n values (10% budget)
    n = len(best_h)
    n_candidates = [n + 2, n - 2, n + 4, n + 6, n - 4]
    n_candidates = [x for x in n_candidates if x >= 31 and x % 2 == 1]
    for n_try in n_candidates:
        if time.time() - start_time > budget_s * 0.75:
            break
        h_interp = interpolate_to_n(best_h, n_try)
        for tau in [0.005, 0.002, 0.001]:
            if time.time() - start_time > budget_s * 0.75:
                break
            h_opt = slsqp_optimize(h_interp, n_try, tau=tau, maxiter=200)
            c5 = evaluate_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt
                n = n_try

    # Phase 5: Fourier refinement (8% budget)
    if time.time() - start_time < budget_s * 0.83:
        n = len(best_h)
        h_four, c5_four = fourier_refine(best_h, n, num_freq=10, iterations=4)
        if c5_four < best_c5:
            best_c5 = c5_four
            best_h = h_four

    # Phase 6: Spatial coordinate descent (8% budget)
    if time.time() - start_time < budget_s * 0.91:
        n = len(best_h)
        h_coord, c5_coord = spatial_descent(best_h, n, sweeps=4)
        if c5_coord < best_c5:
            best_c5 = c5_coord
            best_h = h_coord

    # Phase 7: Block perturbation + SLSQP (7% budget)
    if time.time() - start_time < budget_s * 0.98:
        n = len(best_h)
        h_block, c5_block = block_perturb_search(best_h, n, rng, max_rounds=40)
        if c5_block < best_c5:
            best_c5 = c5_block
            best_h = h_block

    # Final polish
    if time.time() - start_time < budget_s * 0.995:
        n = len(best_h)
        h_final = slsqp_optimize(best_h, n, tau=0.0002, maxiter=500, ftol=1e-14)
        c5 = evaluate_c5(h_final)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_final

    n = len(best_h)
    best_h = project_feasible(best_h, n)
    best_c5 = evaluate_c5(best_h)

    return best_h, best_c5, n
