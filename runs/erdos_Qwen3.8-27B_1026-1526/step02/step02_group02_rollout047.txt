Actually, coordinate descent is tricky here because changing h[i] affects many overlap values (all k values where h[i] participates).

Let me think about the gradient of the max. The subgradient of max_k f_k(h) at the active set A is:
∇_h [max_k f_k(h)] ∈ conv{∇f_k(h) : k ∈ A}

For SLSQP, it handles this internally. But I could implement a custom subgradient step:
1. Compute all f_k(h)
2. Find active set A (top few k values)
3. Compute ∇f_k for k ∈ A
4. Take a step in the negative weighted gradient direction
5. Project back to feasible set

The gradient of f_k(h) = ∑_m h[m](1-h[m+k])*dx with respect to h[i] is:
∂f_k/∂h[i] = dx * [(1-h[i+k]) - h[i-k]]  (where terms are 0 if index out of range)

Wait, more carefully:
f_k = ∑_m h[m] * (1 - h[m+k])

∂f_k/∂h[i] = (1-h[i+k]) - h[i] * (∂h[m+k]/∂h[i] when m+k=i) 
            = (1-h[i+k]) - h[i-k] (if m = i-k, then m+k = i)

So ∂f_k/∂h[i] = (1 - h[i+k]) * 1_{i+k valid} - h[i-k] * 1_{i-k valid}

(where we're thinking of the correlation structure)

This is actually computable in O(n) per k, and with a few active k values, the total gradient is O(|A|*n).

from scipy.interpolate import interp1d
import time

def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def get_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-16
            if not np.any(mask):
                break
            step = min(diff / np.sum(mask), (1.0 - h[mask]).min() * 0.9999)
            h[mask] += step
        else:
            mask = h > 1e-16
            if not np.any(mask):
                break
            step = min(-diff / np.sum(mask), h[mask].min() * 0.9999)
            h[mask] -= step
    return np.clip(h, 0.0, 1.0)

def slsqp_t_obj(x):
    return x[-1]

def slsqp_ineq_all(x):
    h = x[:-1]
    t = x[-1]
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return t - c * dx

def slsqp_eq_sum(x):
    return np.sum(x[:-1]) - len(x[:-1]) / 2.0

def slsqp_eq_jac(x):
    nv = len(x)
    jac = np.zeros(nv)
    jac[:-1] = 1.0
    return jac

def optimize_slsqp(h_init, maxiter=5000, t_margin=0.08):
    n = len(h_init)
    c5_init = compute_c5_from_h(h_init)
    x0 = np.concatenate([h_init, [c5_init + t_margin]])
    bounds = [(0.0, 1.0)] * n + [(c5_init * 0.3, 1.5)]
    constraints = [
        {'type': 'ineq', 'fun': slsqp_ineq_all},
        {'type': 'eq', 'fun': slsqp_eq_sum, 'jac': slsqp_eq_jac}
    ]
    res = minimize(slsqp_t_obj, x0, method='SLSQP', bounds=bounds,
                   constraints=constraints, options={'maxiter': maxiter, 'ftol': 1e-18})
    if res.x is not None and res.success or (res.x is not None and res.nit > 10):
        h_out = project_feasible(res.x[:-1], n)
        c5_out = compute_c5_from_h(h_out)
        return h_out, c5_out
    return h_init, c5_init

def logsumexp_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    cmax = c.max()
    s = T * (c - cmax)
    s = np.clip(s, -500, 500)
    return cmax + np.log(np.sum(np.exp(s))) / T

def lse_obj(x, T, pen):
    n = len(x)
    val = logsumexp_c5(x, T)
    sv = np.sum(x) - n / 2.0
    return val + pen * sv * sv

def optimize_lbfgs(h_init, T=50.0, pen=100.0, maxiter=2000):
    n = len(h_init)
    bounds = [(0.0, 1.0)] * n
    res = minimize(lse_obj, h_init, args=(T, pen), method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-15})
    if res.x is not None:
        h_out = project_feasible(res.x, n)
        return h_out, compute_c5_from_h(h_out)
    return h_init, compute_c5_from_h(h_init)

def gradient_subgradient_step(h, step_size, top_k=8):
    n = len(h)
    dx = 2.0 / n
    ov = np.correlate(h, 1.0 - h, mode="full") * dx
    c5_cur = ov.max()
    # Find active set (top k overlaps)
    active = np.argsort(ov)[-top_k:]
    # Compute gradient as weighted average of gradients at active k
    grad = np.zeros(n)
    weights = np.exp((ov[active] - c5_cur) / 0.01)  # softmax-like
    weights /= weights.sum()
    for i, k_idx in enumerate(active):
        lag = k_idx - (n - 1)  # actual lag
        # d/dh[j] of sum_m h[m]*(1-h[m+lag])
        # = (1-h[j+lag]) - h[j-lag] (with boundary handling)
        g = np.zeros(n)
        if lag >= 0:
            m_range = min(n - lag, n)
            g[:m_range] += (1.0 - h[lag:lag+m_range])  # (1-h[j+lag]) term
            g[lag:] -= h[:m_range]  # -h[j-lag] term
        else:
            m_range = min(n + lag, n)
            g[-lag:] += (1.0 - h[:m_range])  # (1-h[j+lag]) where j+lag = j-|lag|
            g[:m_range] -= h[-lag:]  # -h[j-lag] where j-lag = j+|lag|
        grad += weights[i] * g
    # Negate for descent, clip to [0,1]
    h_new = h - step_size * grad * dx / c5_cur
    h_new = project_feasible(h_new, n)
    return h_new, compute_c5_from_h(h_new)

def interpolate_to(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    f = interp1d(x_old, h_old, kind='cubic', fill_value='extrapolate')
    h_new = np.clip(f(x_new), 0, 1)
    return project_feasible(h_new, n_new)

        h_up = interpolate_to(best_h, n_target)
        c5_up = compute_c5_from_h(h_up)
        if c5_up > best_c5 + 0.005:
            continue
        rem = budget_s - 350 - elapsed
        iters = min(4000, max(800, int(rem * 1.2)))
        h_new, c5_new = optimize_slsqp_full(h_up, maxiter=iters, t_margin=0.06)
        if c5_new < best_c5:
            best_h, best_c5, n_best = h_new, c5_new, n_target
        elif c5_up < best_c5:
            best_h, best_c5, n_best = h_up, c5_up, n_target

    # Phase 4: Further SLSQP with topk for speed at higher n
    if time.time() - t0 < budget_s - 200 and n_best > 300:
        h_n, c5_n = optimize_slsqp_topk(best_h, topk=50, maxiter=3000, t_margin=0.04)
        if c5_n < best_c5:
            best_h, best_c5 = h_n, c5_n

    # Phase 5: L-BFGS-B sweep with increasing sharpness
    if time.time() - t0 < budget_s - 150:
        for T_val in [40.0, 100.0, 250.0]:
            if time.time() - t0 > budget_s - 120:
                break
            h_n, c5_n = optimize_lbfgs(best_h, T=T_val, pen=200.0, maxiter=1200)
            if c5_n < best_c5:
                best_h, best_c5 = h_n, c5_n

    # Phase 6: SLSQP + subgradient steps
    if time.time() - t0 < budget_s - 80:
        best_h, best_c5 = optimize_slsqp_full(best_h, maxiter=4000, t_margin=0.03)

    if time.time() - t0 < budget_s - 60:
        for _ in range(15):
            if time.time() - t0 > budget_s - 45:
                break
            h_s, c5_s = subgradient_step(best_h, step_size=0.3, n_active=10)
            if c5_s < best_c5:
                best_h, best_c5 = h_s, c5_s

    # Phase 7: Perturbation-based local search
    n_pert = 120
    if time.time() - t0 < budget_s - 30:
        for attempt in range(n_pert):
            if time.time() - t0 > budget_s - 12:
                break
            frac = attempt / max(1, n_pert)
            scale = 0.002 + 0.018 * (1.0 - frac) ** 1.8

            r = rng.random()
            if r < 0.30:
                h_p = best_h + scale * rng.standard_normal(n_best)
            elif r < 0.55:
                nf = rng.integers(1, max(2, n_best // 4))
                ph = rng.random() * 2 * np.pi
                h_p = best_h + scale * 2.0 * np.cos(2 * np.pi * nf * np.arange(n_best) / n_best + ph)
            elif r < 0.75:
                h_p = best_h.copy()
                bs = rng.integers(5, max(6, n_best // 10))
                st = rng.integers(0, max(1, n_best - bs))
                h_p[st:st + bs] += scale * 4.0 * rng.standard_normal(bs)
            else:
                h_p = best_h.copy()
                idx = rng.choice(n_best, size=max(5, n_best // 10), replace=False)
                h_p[idx] += scale * 3.0 * rng.standard_normal(len(idx))

            h_p = project_feasible(h_p, n_best)

            if time.time() - t0 > budget_s - 8:
                break

            if n_best <= 300:
                h_n, c5_n = optimize_slsqp_full(h_p, maxiter=1200, t_margin=0.02)
            else:
                h_n, c5_n = optimize_slsqp_topk(h_p, topk=40, maxiter=1000, t_margin=0.02)

            if c5_n < best_c5 - 1e-13:
                best_h, best_c5 = h_n, c5_n

    # Phase 8: Very fine final polishing
    if time.time() - t0 < budget_s - 8:
        for _ in range(30):
            if time.time() - t0 > budget_s - 4:
                break
            scale = 0.0005 + 0.002 * (1 - _ / 30.0) ** 2
            h_p = best_h + scale * rng.standard_normal(n_best)
            h_p = project_feasible(h_p, n_best)
            h_n, c5_n = optimize_slsqp_full(h_p, maxiter=1000, t_margin=0.01)
            if c5_n < best_c5 - 1e-14:
                best_h, best_c5 = h_n, c5_n

    # Final SLSQP polish
    if time.time() - t0 < budget_s - 2:
        best_h, best_c5 = optimize_slsqp_full(best_h, maxiter=3000, t_margin=0.01)

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5_from_h(best_h)

    return best_h, best_c5, len(best_h)
