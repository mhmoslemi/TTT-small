import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(h, alpha):
    corr = compute_corr(h)
    m = np.max(corr)
    if alpha < 1e6:
        exp_vals = np.exp(alpha * (corr - m))
        return m + np.log(np.sum(exp_vals)) / alpha
    return float(m)

def smooth_max_grad(h, alpha):
    corr = compute_corr(h)
    n = len(h)
    m = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m))
    w = exp_vals / np.sum(exp_vals)
    # Gradient: grad[i] = sum_m w[m] * d(corr[m])/dh[i]
    # d(corr[m])/dh[i] = (1-h[i+off]) * [valid] - h[i-off] * [valid, off!=0]
    # where off = m - n + 1
    offsets = np.arange(-(n-1), n)  # shape (2n-1,)
    grad = np.zeros(n)
    for i in range(n):
        # First part: (1-h[i+offset]) for valid offsets
        idx1 = i + offsets
        valid1 = (idx1 >= 0) & (idx1 < n)
        if np.any(valid1):
            grad[i] += np.dot(w[valid1], (1.0 - h[idx1[valid1]]))
        # Second part: -h[i-offset] for valid non-zero offsets
        idx2 = i - offsets
        valid2 = (idx2 >= 0) & (idx2 < n) & (offsets != 0)
        if np.any(valid2):
            grad[i] -= np.dot(w[valid2], h[idx2[valid2]])
    return grad

def slsqp_optimize(h0, n, target_sum, alpha, maxiter=500, ftol=1e-14):
    def obj(h):
        return smooth_max_obj(h, alpha)
    def grad(h):
        return smooth_max_grad(h, alpha)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), method='SLSQP', jac=grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)
