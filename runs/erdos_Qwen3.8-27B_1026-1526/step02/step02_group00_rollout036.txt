    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-16:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)

    corr = np.correlate(h, 1.0 - h, mode="full")

    shifted = np.clip(shifted, -50.0/alpha, 0.0)

        w[int(np.argmax(corr))] = 1.0
        obj = m_val
    else:
        w = exp_vals / w_sum
        obj = m_val + np.log(w_sum) / alpha

    n_c = len(corr)
    j_idx = np.arange(n)[:, None]  # shape (n, 1)
    m_idx = np.arange(n_c)[None, :]  # shape (1, n_c)
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

        return smooth_max_obj_and_grad(x, alpha)[0]

        return smooth_max_obj_and_grad(x, alpha)[1]

        return np.sum(x) - target_sum
    def con_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(func, h0.copy(), method='SLSQP', jac=jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_to_sum(res.x, target_sum)

        return project_to_sum(h0.copy(), target_sum)

def mirror_descent(h0, n, target_sum, alpha, n_iters=200, eta_init=0.01):
    h = h0.copy()
    for it in range(n_iters):
        obj, grad = smooth_max_obj_and_grad(h, alpha)
        eta = eta_init / (1.0 + 0.01 * it)

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = float(np.max(corr))
    shifted = corr - m_val
    shifted = np.clip(shifted, -50.0 / alpha, 0.0)
    exp_vals = np.exp(alpha * shifted)
    w_sum = float(np.sum(exp_vals))
    if w_sum < 1e-300:
        w = np.zeros(len(corr))
        w[int(np.argmax(corr))] = 1.0
        obj = m_val * dx
    else:
        w = exp_vals / w_sum
        obj = (m_val + np.log(w_sum) / alpha) * dx
    n_c = len(corr)
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(n_c)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = dx * np.sum(cf - cs, axis=1)
    return obj, grad

def slsqp_optimize(h0, n, target_sum, alpha, maxiter=1500):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (target_sum, n)}]
    try:

                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})

    except Exception:

def subgradient_averaged(h0, n, target_sum, alpha, n_iters=500, eta_init=0.05):

    h_ave = h.copy()

        h_new = project_to_sum(h_new, target_sum)

                continue
            for delta in deltas:
                if h[i] + delta > 1.0 or h[j] - delta < 0.0:
                    continue
                h[i] += delta
                h[j] -= delta
                c5_new = compute_c5(h)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    improved = True
                else:
                    h[i] -= delta
                    h[j] += delta
        if not improved:
            break
    return h, best_c5


def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old, period=2.0)
    return h_new


def make_diverse_starts(n, target_sum, rng, best_h, best_n, n_starts=8):
    starts = []
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    for s in range(n_starts):
        n_harm = rng.randint(2, 9)
        h = np.full(n, 0.5)
        for k in range(1, n_harm + 1):
            amp = rng.uniform(0.05, 0.35) / np.sqrt(k)
            phase = rng.uniform(0.0, 2.0 * np.pi)
            h += amp * np.sin(2.0 * np.pi * k * x / 2.0 + phase)
        starts.append(project_to_sum(h, target_sum))
    for nblk in [3, 5, 7, 9, 11]:
        h = np.zeros(n)
        blk = n // (2 * nblk)
        if blk > 0:
            for i in range(0, n, 2 * blk):
                h[i:min(i + blk, n)] = 1.0
            starts.append(project_to_sum(h, target_sum))
    if best_h is not None and best_n > 0:
        if best_n == n:
            starts.append(best_h.copy())
        else:
            starts.append(project_to_sum(interpolate_h(best_h, best_n, n), target_sum))
    return starts


def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - t_start)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP continuation at parent resolution
    n1 = best_n
    ts1 = n1 / 2.0
    for alpha in [20, 50, 100, 300, 1000, 3000, 10000]:
        if time_left() < 30:
            break
        h_new = slsqp_optimize(best_h, n1, ts1, alpha, maxiter=1200)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 2: Subgradient with averaging
    if time_left() > 60:
        for alpha in [50, 200, 500]:
            if time_left() < 30:
                break
            h_sg, c5_sg = subgradient_averaged(best_h, best_n, ts1, alpha,
                                               n_iters=300, eta_init=0.03)
            if c5_sg < best_c5:
                best_c5 = c5_sg
                best_h = h_sg.copy()
            h_next = slsqp_optimize(h_sg, best_n, ts1, alpha, maxiter=800)
            c5_next = compute_c5(h_next)
            if c5_next < best_c5:
                best_c5 = c5_next
                best_h = h_next.copy()

    # Phase 3: Pair sweep refinement
    if time_left() > 120:
        h_ps, c5_ps = pair_sweep(best_h, best_n, ts1, rng,
                                 n_rounds=50, n_pairs=4000)
        if c5_ps < best_c5:
            best_c5 = c5_ps
            best_h = h_ps.copy()
        h_pol = slsqp_optimize(best_h, best_n, ts1, 5000, maxiter=1500)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Phase 4: Multi-resolution exploration
    resolutions = [160, 180, 200, 240, 280, 300, 320, 360, 400, 480]
    for n_tr in resolutions:
        if time_left() < 80:
            break
        if n_tr % 2 != 0:
            continue
        ts_tr = n_tr / 2.0
        starts = make_diverse_starts(n_tr, ts_tr, rng, best_h, best_n, n_starts=4)
        for h_s in starts:
            if time_left() < 30:
                break
            h_s = project_to_sum(h_s, ts_tr)
            c5_s = compute_c5(h_s)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_s.copy()
                best_n = n_tr
            for alpha in [50, 200, 1000]:
                if time_left() < 20:
                    break
                h_opt = slsqp_optimize(h_s, n_tr, ts_tr, alpha, maxiter=1000)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    best_n = n_tr
                h_s = h_opt

    # Phase 5: Second pair sweep on best
    if time_left() > 80:
        ts_best = best_n / 2.0
        h_ps2, c5_ps2 = pair_sweep(best_h, best_n, ts_best, rng,
                                   n_rounds=40, n_pairs=5000)
        if c5_ps2 < best_c5:
            best_c5 = c5_ps2
            best_h = h_ps2.copy()

    # Phase 6: Final high-alpha polish
    if time_left() > 40:
        ts_f = best_n / 2.0
        for alpha in [5000, 20000, 50000]:
            if time_left() < 25:
                break
            h_pol = slsqp_optimize(best_h, best_n, ts_f, alpha, maxiter=2000)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()
        if time_left() > 30:
            h_final_ps, c5_final_ps = pair_sweep(best_h, best_n, ts_f, rng,
                                                 n_rounds=20, n_pairs=3000)
            if c5_final_ps < best_c5:
                best_c5 = c5_final_ps
                best_h = h_final_ps.copy()

    # Final validation
    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
