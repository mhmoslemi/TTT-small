    """Log-sum-exp approximation of max of correlations."""
    corrs = compute_corrs(h)
    c_max = corrs.max()
    if c_max < -100:
        return float(c_max)
    shifted = T * (corrs - c_max)
    shifted = np.clip(shifted, -400, 400)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / T)

    c5_val = lse_max(x, T)
    sum_v = np.sum(x) - len(x) / 2.0
    return c5_val + penalty * sum_v ** 2

def lse_optimize(h_init, T, penalty, maxiter=3000):
    n = len(h_init)
    bounds = [(0.0, 1.0)] * n
    result = minimize(lse_obj, h_init, args=(T, penalty),
                      method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14})

    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = diff / cnt
            h[mask] = np.minimum(1.0, h[mask] + step)
        else:
            mask = h > 1e-15
            cnt = np.sum(mask)
            if cnt == 0:
                break
            step = -diff / cnt
            h[mask] = np.maximum(0.0, h[mask] - step)
    return np.clip(h, 0.0, 1.0)

def c5_lse(h, T):
    corrs = compute_corrs(h)
    c_max = corrs.max()
    if c_max < -100:
        return float(c_max)
    shifted = T * (corrs - c_max)
    shifted = np.clip(shifted, -400, 400)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / T)

def lse_smooth_obj(x, T, penalty):
    val = c5_lse(x, T)
    sv = np.sum(x) - len(x) / 2.0
    return val + penalty * sv * sv

def lse_optimize(h_init, T, penalty, maxiter=5000):
    n = len(h_init)
    bounds = [(0.0, 1.0)] * n
    res = minimize(lse_smooth_obj, h_init, args=(T, penalty),
                   method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14})
    if res.x is not None:
        h_out = project_feasible(res.x, n)
        return h_out, compute_c5_from_h(h_out)
    return h_init, compute_c5_from_h(h_init)

def slsqp_full_obj(x):
    return x[-1]

def slsqp_full_ineq(x):
    h = x[:-1]
    t = x[-1]
    n = len(h)
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode="full") * dx
    return t - np.max(corrs)

def slsqp_full_eq(x):

    return np.sum(x[:-1]) - n / 2.0

def slsqp_full_eq_jac(x):

    jac = np.zeros(len(x))
    jac[:n] = 1.0
    return jac

    if res.x is not None:
        h_out = project_feasible(res.x[:-1], n)
        return h_out, compute_c5_from_h(h_out)

        if remaining() < 20:
            break
        scale = 0.002 + 0.01 * rng.random()
        h_pert = best_h + scale * rng.standard_normal(n_best)
        h_pert = project_feasible(h_pert, n_best)
        h_new, c5_new = slsqp_optimize(h_pert, maxiter=3000, t_margin=0.03)
        if c5_new < best_c5 - 1e-13:
            best_h = h_new
            best_c5 = c5_new

    # Phase 8: Final L-BFGS-B polish with very high T
    if remaining() > 30:
        h_new, c5_new = lse_optimize(best_h, T=500, penalty=1000, maxiter=5000)
        if c5_new < best_c5 - 1e-13:
            best_h = h_new
            best_c5 = c5_new

    # Phase 9: Final SLSQP polish
    if remaining() > 10:
        best_h, best_c5 = slsqp_optimize(best_h, maxiter=8000, t_margin=0.02)

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5_from_h(best_h)

    return best_h, best_c5, len(best_h)
