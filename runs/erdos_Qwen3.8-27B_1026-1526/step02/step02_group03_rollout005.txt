            perturb = project_feasible(perturb, n)
            tau_sched = [0.01, 0.005, 0.002, 0.001]
            h_opt, c5 = run_single_opt(perturb, n, tau_sched, 50, phase3_end)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()
        
        # Coordinate descent
        if time.time() < phase3_end:
            h_cd, c5_cd = coordinate_descent(best_h, n, sweeps=10, time_limit=phase3_end)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()
        
        # Pairwise descent
        if time.time() < phase3_end:
            h_pw, c5_pw = pairwise_descent(best_h, n, sweeps=5, time_limit=phase3_end, rng=rng)
            if c5_pw < best_c5:
                best_c5 = c5_pw
                best_h = h_pw.copy()
    
    # Phase 4: Final polishing (5% of time)
    phase4_end = start_time + budget_s * 0.995
    if best_h is not None and time.time() < phase4_end:
        n = len(best_h)
        h_final, c5_final = coordinate_descent(best_h, n, sweeps=5, time_limit=phase4_end)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()
    
    if best_h is None:
        if parent_h is not None:
            best_h = parent_h.copy()
            best_c5 = evaluate_c5(best_h)
        else:
            n = 77
            best_h = project_feasible(np.full(n, 0.5), n)
            best_c5 = evaluate_c5(best_h)
    
    final_c5 = evaluate_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
