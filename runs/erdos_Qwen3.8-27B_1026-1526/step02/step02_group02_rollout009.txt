def smooth_c5_and_grad(h, T):

    dx = 2.0 / n
    v = 1.0 - h
    c = np.correlate(h, v, mode="full") * dx

    exp_shifted = np.exp(shifted)
    S = exp_shifted.sum()
    p = exp_shifted / S
    
    val = c_max + np.log(S) / T
    
    grad = np.zeros(n)
    l = np.arange(2*n-1) - (n-1)  # l[k] = k - n + 1
    
    for i in range(n):
        # A[i] = sum_k p[k] * (1-h[i+l[k]]) for 0 <= i+l[k] < n
        mask1 = (i + l >= 0) & (i + l < n)
        if mask1.any():

            grad[i] += p[mask1] @ v[h_idx]
        
        # B[i] = sum_k p[k] * h[i-l[k]] for 0 <= i-l[k] < n

        if mask2.any():

            grad[i] -= p[mask2] @ h[h_idx]

    return val, grad

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            rooms = 1.0 - h[mask]
            if rooms.sum() < diff:
                h[mask] = 1.0
            else:
                h[mask] += diff / np.sum(mask)
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            h[mask] -= (-diff) / np.sum(mask)
    return np.clip(h, 0.0, 1.0)

def smooth_c5_val(x, T, penalty):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full") * dx
    c_max = c.max()
    shifted = np.clip(T * (c - c_max), -500, 500)
    S = np.exp(shifted).sum()
    val = c_max + np.log(S) / T
    sv = np.sum(x) - n / 2.0
    return val + penalty * sv * sv

def smooth_c5_grad(x, T, penalty):
    n = len(x)
    dx = 2.0 / n
    v = 1.0 - x
    c = np.correlate(x, v, mode="full") * dx
    c_max = c.max()
    shifted = np.clip(T * (c - c_max), -500, 500)
    exp_s = np.exp(shifted)
    S = exp_s.sum()
    p = exp_s / S
    val = c_max + np.log(S) / T
    
    # Gradient of log-sum-exp w.r.t. x
    l = np.arange(2*n-1) - (n-1)
    grad = np.zeros(n)
    for i in range(n):
        m1 = (i + l >= 0) & (i + l < n)
        if m1.any():
            grad[i] += p[m1] @ v[i + l[m1]]
        m2 = (i - l >= 0) & (i - l < n)
        if m2.any():
            grad[i] -= p[m2] @ x[i - l[m2]]
    grad *= dx
    
    # Add penalty gradient
    sv = np.sum(x) - n / 2.0
    grad += 2 * penalty * sv * np.ones(n)
    
    return val, grad

def lbfgsb_smooth(h_init, T, penalty, maxiter=2000):
    n = len(h_init)
    bounds = [(0.0, 1.0)] * n
    result = minimize(
        lambda x: smooth_c5_val(x, T, penalty),
        h_init,
        jac=lambda x: smooth_c5_grad(x, T, penalty)[1],
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12}
    )
    h_out = project_feasible(result.x, n)
    return h_out, compute_c5_from_h(h_out)

def slsqp_active(h_init, top_k, maxiter=3000, t_margin=0.1):
    n = len(h_init)
    c5_init = compute_c5_from_h(h_init)
    
    def obj(x):
        return x[-1]
    
    def ineq(x):
        h = x[:-1]
        t = x[-1]
        c = compute_corr(h)
        return t - c[top_k]
    
    def eq(x):
        return np.sum(x[:-1]) - n / 2.0
    
    def eq_jac(x):
        jac = np.zeros(x.shape)
        jac[:n] = 1.0
        return jac
    
    x0 = np.concatenate([h_init, [c5_init + t_margin]])
    bounds = [(0.0, 1.0)] * n + [(max(0.001, c5_init * 0.5), 2.0)]
    constraints = [
        {'type': 'ineq', 'fun': ineq},
        {'type': 'eq', 'fun': eq, 'jac': eq_jac}
    ]
    
    result = minimize(obj, x0, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
    
    if result.x is not None:
        h_out = project_feasible(result.x[:-1], n)
        return h_out, compute_c5_from_h(h_out)
    return h_init, c5_init

def resample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    f = interp1d(x_old, h, kind='linear', fill_value='extrapolate', bounds_error=False)
    h_new = np.clip(f(x_new), 0, 1)
    return project_feasible(h_new, n_new)

    if 'initial_h_values' in globals():

    best_h = h0.copy()
    best_c5 = compute_c5_from_h(best_h)
    
    def elapsed():
        return time.time() - t0
    
    # Phase 1: Quick SLSQP refinement (30s)

        top_k = np.argsort(c)[-30:]
        best_h, best_c5 = slsqp_active(best_h, top_k, maxiter=3000, t_margin=0.1)
    
    # Phase 2: Smooth optimization with increasing T (100s)

        h_new, c5_new = lbfgsb_smooth(best_h, T, pen, maxiter=1500)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new
    
    # Phase 3: Multi-start exploration (800s)
    n_perturb = 40
    for attempt in range(n_perturb):

            break
        
        frac = attempt / max(1, n_perturb)

        r = rng.random()
        if r < 0.3:

            scale = 0.08 * (1 - frac) ** 0.5 + 0.01
            h_pert = best_h.copy()
            n_modes = rng.integers(1, 5)
            for _ in range(n_modes):
                freq = rng.integers(1, max(2, n // 2))
                phase = rng.random() * 2 * np.pi
                h_pert += (scale / np.sqrt(n_modes)) * np.cos(2 * np.pi * freq * np.arange(n) / n + phase)
        elif r < 0.5:

            scale = 0.03 + 0.1 * (1 - frac)

        elif r < 0.7:

            h_pert = best_h.copy()
            n_blocks = rng.integers(2, max(3, n // 30))
            for _ in range(n_blocks):
                bsz = rng.integers(3, max(4, n // 8))
                start = rng.integers(0, max(1, n - bsz))
                h_pert[start:start+bsz] += rng.uniform(-0.15, 0.15)

            h_pert = best_h.copy()
            thresh = rng.uniform(0.3, 0.7)
            h_pert = np.where(h_pert > thresh, np.minimum(1.0, h_pert + 0.2), np.maximum(0.0, h_pert - 0.2))
        else:
            # Random restart (less connected to best)
            h_pert = best_h * 0.5 + 0.5 * np.clip(0.5 + 0.4 * rng.standard_normal(n), 0, 1)

            top_k_cur = np.argsort(c_cur)[-20:]
            h_new, c5_new = slsqp_active(h_pert, top_k_cur, maxiter=1500, t_margin=0.05)
        
        if c5_new < best_c5 - 1e-12:
            best_h, best_c5 = h_new, c5_new
    
    # Phase 4: Try different n (150s)
    for n_try in [540, 450, 360]:
        if elapsed() > budget_s - 150:

        h_interp = resample(best_h, n_try)

    def obj(x):
        return x[-1]

        t = x[-1]
        c = compute_corr(h)
        return t - c[top_k]
