    n = len(h)

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_at(h, k):
    """Compute subgradient of f_k(h) = sum_j h[j]*(1-h[j+k]) at point h."""
    n = len(h)
    g = np.zeros(n)
    ls = max(0, -k)
    le = min(n, n - k)
    if le > ls:
        g[ls:le] += (1.0 - h[ls+k:le+k])
    ls2 = max(0, k)
    le2 = min(n, n + k)
    if le2 > ls2:
        g[ls2:le2] -= h[ls2-k:le2-k]
    return g

def nesterov_subgrad(h0, n, target_sum, max_iters, lr_init, alpha_mom=0.9):
    """Nesterov accelerated subgradient descent."""

def coordinate_descent(h0, n, target_sum, max_rounds, rng, step_sizes=None):
    h = h0.copy()
    c5_cur = compute_c5(h)
    if step_sizes is None:
        step_sizes = [0.05, 0.02, 0.01, 0.005, -0.05, -0.02, -0.01, -0.005]
    
    for rnd in range(max_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            best_val = c5_cur
            best_delta = 0.0
            for d in step_sizes:
                new_val = h[i] + d
                if new_val < 0 or new_val > 1:
                    continue
                h[i] = new_val
                h = project_to_sum(h, target_sum)
                c5_new = compute_c5(h)
                if c5_new < best_val - 1e-15:
                    best_val = c5_new
                    best_delta = d
                h[i] = new_val - d  # restore
            if best_delta != 0.0:
                h[i] = np.clip(h[i] + best_delta, 0.0, 1.0)
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True
        if not improved:
            break
    return h, c5_cur

def smooth_max_grad(h, alpha):
    n = len(h)
    n_corr = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    val = m + np.log(s / n_corr) / alpha
    
    # Gradient: grad[j] = sum_m w[m] * d(corr[m])/d(h[j])
    # d(corr[m])/d(h[j]) = (1-h[j+m-n+1]) * 1_{0<=j+m-n+1<n} - h[j-m+n-1] * 1_{0<=j-m+n-1<n}
    g = np.zeros(n)
    for j in range(n):
        # First part: (1-h[j+m-n+1]) for m where 0 <= j+m-n+1 < n
        # j+m-n+1 >= 0 => m >= n-1-j
        # j+m-n+1 <= n-1 => m <= 2n-2-j
        m1_lo = max(0, n-1-j)
        m1_hi = min(n_corr-1, 2*n-2-j)
        if m1_hi >= m1_lo:
            idx1 = j + (np.arange(m1_lo, m1_hi+1) - n + 1)
            g[j] += np.dot(w[m1_lo:m1_hi+1], 1.0 - h[idx1])
        
        # Second part: -h[j-m+n-1] for m where 0 <= j-m+n-1 < n
        # j-m+n-1 >= 0 => m <= j+n-1
        # j-m+n-1 <= n-1 => m >= j
        m2_lo = max(0, j)
        m2_hi = min(n_corr-1, j+n-1)
        if m2_hi >= m2_lo:
            idx2 = j - (np.arange(m2_lo, m2_hi+1)) + n - 1
            g[j] -= np.dot(w[m2_lo:m2_hi+1], h[idx2])
    
    return val, g
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        n_free = np.sum(mask)
        if n_free > 0:
            h[mask] += diff / n_free
        else:
            # All at bounds, need to adjust
            if diff > 0:
                # Need to increase some 0s to positive
                idx = np.where(h == 0)[0][:int(abs(diff)) + 1]
                h[idx] = 1.0
            else:
                idx = np.where(h == 1)[0][:int(-diff) + 1]
                h[idx] = 0.0
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_grad(h, alpha):
    n = len(h)
    n_corr = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    val = m + np.log(s / n_corr) / alpha
    
    g = np.zeros(n)
    for j in range(n):
        m1_lo = max(0, n-1-j)
        m1_hi = min(n_corr-1, 2*n-2-j)
        if m1_hi >= m1_lo:
            idx1 = j + (np.arange(m1_lo, m1_hi+1) - n + 1)
            g[j] += np.dot(w[m1_lo:m1_hi+1], 1.0 - h[idx1])
        m2_lo = max(0, j)
        m2_hi = min(n_corr-1, j+n-1)
        if m2_hi >= m2_lo:
            idx2 = j - (np.arange(m2_lo, m2_hi+1)) + n - 1
            g[j] -= np.dot(w[m2_lo:m2_hi+1], h[idx2])
    
    return val, g

def proj_gradient(h0, n, target_sum, alpha, max_iters, lr_init, time_fn=None):
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    
    for it in range(max_iters):
        if time_fn is not None and time_fn() < 10:
            break
        
        val, g = smooth_max_grad(h, alpha)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-10:
            break
        
        # Step
        h_new = np.clip(h - lr * g, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        
        # Simple check: evaluate c5 periodically
        if (it + 1) % 100 == 0:
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
        
        h = h_new
        
        # Decay learning rate slowly
        if it % 500 == 499:
            lr *= 0.9
    
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    return best_h, best_c5

def nesterov_proj_grad(h0, n, target_sum, alpha, max_iters, lr_init, time_fn=None):
    h = h0.copy()
    y = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    t = 1.0

        y = np.clip(y, 0.0, 1.0)

    c5 = compute_c5(h)

def coordinate_descent_fast(h0, n, target_sum, max_rounds, rng, step_sizes):
    h = h0.copy()
    c5_cur = compute_c5(h)
    
    for rnd in range(max_rounds):
        improved = False
        # Try a random subset of coordinates
        n_try = min(n, max(30, n // 3))
        idx_order = rng.choice(n, size=n_try, replace=False)
        for i in idx_order:
            best_val = c5_cur
            best_delta = 0.0
            for d in step_sizes:
                new_val = h[i] + d
                if new_val < -0.01 or new_val > 1.01:
                    continue
                old = h[i]
                h[i] = np.clip(new_val, 0.0, 1.0)
                h = project_to_sum(h, target_sum)
                c5_new = compute_c5(h)
                h[i] = old
                if c5_new < best_val - 1e-12:
                    best_val = c5_new
                    best_delta = d
            if best_delta != 0.0:
                h[i] = np.clip(h[i] + best_delta, 0.0, 1.0)
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True
        if not improved:
            break
    return h, c5_cur

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
