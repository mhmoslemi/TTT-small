    best_h = project_feasible(h0.copy(), n)
    best_c5 = evaluate_c5(best_h)
    tau = tau_start
    alpha = 0.02
    iters_per_tau = 50
    
    while tau > tau_end and time.time() < time_limit:
        for _ in range(iters_per_tau):
            if time.time() > time_limit:
                break
            corr = np.correlate(best_h, 1.0 - best_h, mode="full")
            _, weights = smooth_max_and_weights(corr, tau)
            grad = compute_grad(best_h, n, weights)
            
            # Try a few step sizes
            improved = False
            for a in [alpha, alpha * 0.5, alpha * 2.0, alpha * 0.25]:
                new_h = project_feasible(best_h - a * grad, n)
                new_c5 = evaluate_c5(new_h)
                if new_c5 < best_c5:
                    best_h = new_h
                    best_c5 = new_c5
                    if a > alpha:
                        alpha = min(a * 1.1, 0.1)
                    improved = True
                    break
            if not improved:
                alpha *= 0.5
                if alpha < 1e-6:
                    # Random kick
                    best_h = project_feasible(best_h + rng.randn(n) * 0.01, n)
                    best_c5 = evaluate_c5(best_h)
                    alpha = 0.01
        
        tau *= 0.9
    
    return best_h, best_c5

def paired_coordinate_descent(h, n, max_sweeps=20, time_limit=None):
    """Paired coordinate descent: swap mass between two coordinates."""
    best = h.copy()
    best_c5 = evaluate_c5(best)
    
    for sweep in range(max_sweeps):
        if time_limit and time.time() > time_limit:
            break
        improved = False
        for i in range(n):
            for j in range(i+1, n):
                # Try moving mass from i to j and vice versa
                for delta in [0.01, 0.02, 0.03]:
                    # Move from i to j
                    trial = best.copy()
                    trial[i] -= delta
                    trial[j] += delta
                    if trial[i] >= 0 and trial[j] <= 1:
                        c5 = evaluate_c5(trial)
                        if c5 < best_c5:
                            best = trial
                            best_c5 = c5
                            improved = True
                            break
                    # Move from j to i
                    trial = best.copy()
                    trial[i] += delta
                    trial[j] -= delta
                    if trial[i] <= 1 and trial[j] >= 0:
                        c5 = evaluate_c5(trial)
                        if c5 < best_c5:
                            best = trial
                            best_c5 = c5
                            improved = True
                            break
        if not improved:
            break
    
    return best, best_c5

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        if np.sum(proj) > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-10:
        idx = int(np.argmin(np.abs(result - 0.5)))
        result[idx] = np.clip(result[idx] + diff, 0.0, 1.0)
    return result

def smooth_max_weights(corr, tau):
    v = corr / tau
    vmax = np.max(v)
    weights = np.exp(v - vmax)
    weights /= np.sum(weights)
    return weights

def compute_grad(h, n, weights):
    dx = 2.0 / n
    J = np.arange(n)[:, None]
    K = np.arange(-(n - 1), n)
    ip1 = J + K
    mask1 = ((ip1 >= 0) & (ip1 < n)).astype(float)
    ip2 = J - K
    mask2 = ((ip2 >= 0) & (ip2 < n)).astype(float)
    w = weights[None, :]
    term1 = np.sum(w * (1.0 - h)[ip1] * mask1, axis=1)
    term2 = -np.sum(w * h[ip2] * mask2, axis=1)
    return (term1 + term2) * dx
