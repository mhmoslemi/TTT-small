            if time.time() < t_phase3_end - 2:
                h_opt, c5_opt = slsqp_refine(h_sub, n_cur, [0.01, 0.003, 0.001], 80, t_phase3_end - 1)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()

    # Phase 3b: Try different n values briefly
    if time.time() < t_phase3_end + budget_s * 0.05:
        for n_try in [len(best_h) - 1, len(best_h) + 1, len(best_h) - 2, len(best_h) + 2]:
            if n_try < 20 or n_try > 100:
                continue
            if time.time() > t_phase3_end + budget_s * 0.05:
                break
            if parent is not None:
                xp = np.linspace(0, 2, len(parent), endpoint=False)
                xt = np.linspace(0, 2, n_try, endpoint=False)
                f = interp1d(xp, parent, kind='cubic', fill_value='extrapolate', bounds_error=False)
                h_try = np.clip(f(xt), 0, 1)
            else:
                h_try = np.full(n_try, 0.5)
            h_try = project_feasible(h_try, n_try / 2.0)
            h_opt, c5_opt = slsqp_refine(h_try, n_try, [0.02, 0.005, 0.001], 150, t_phase3_end + budget_s * 0.05)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
            h_cd_try, c5_cd_try = coord_descent_pairs(h_opt, n_try, sweeps=5, t_limit=t_phase3_end + budget_s * 0.05, pair_per_iter=2)
            if c5_cd_try < best_c5:
                best_c5 = c5_cd_try
                best_h = h_cd_try.copy()

    # Phase 4: Final intensive refinement
    if time.time() < t_final_end:
        n_cur = len(best_h)
        # Multiple rounds of coord descent + SLSQP
        for round_idx in range(6):
            if time.time() > t_final_end:
                break
            h_cd, c5_cd = coord_descent_pairs(best_h, n_cur, sweeps=15, t_limit=t_final_end, pair_per_iter=3)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()
            if time.time() > t_final_end - 5:
                break
            tau = 0.0005 * (0.5 ** round_idx)
            h_ref, c5_ref = slsqp_refine(best_h, n_cur, [tau], 300, t_final_end - 2)
            if c5_ref < best_c5:
                best_c5 = c5_ref
                best_h = h_ref.copy()

    # Final coordinate descent sweep
    if time.time() < t_start + budget_s * 0.995:
        n_cur = len(best_h)
        h_cd, c5_cd = coord_descent_pairs(best_h, n_cur, sweeps=20, t_limit=t_start + budget_s * 0.995, pair_per_iter=3)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Ensure feasibility
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final / 2.0)
    final_c5 = evaluate_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, n_final
