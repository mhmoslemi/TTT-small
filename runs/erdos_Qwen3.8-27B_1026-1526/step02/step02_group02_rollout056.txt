import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time


def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = int(mask.sum())
            if cnt == 0:
                break
            max_add = (1.0 - h[mask]).min()
            step = min(diff / cnt, max_add * 0.99)
            if step <= 0:
                break
            h[mask] += step
        else:
            mask = h > 0.0
            cnt = int(mask.sum())
            if cnt == 0:
                break
            max_sub = h[mask].min()
            step = min(-diff / cnt, max_sub * 0.99)
            if step <= 0:
                break
            h[mask] -= step
    return np.clip(h, 0.0, 1.0)


def interp_to_n(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return project_feasible(h.copy(), n_new)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    f = interp1d(x_old, h, kind='cubic', fill_value="extrapolate", assume_sorted=True)
    h_new = np.clip(f(x_new), 0, 1)
    return project_feasible(h_new, n_new)


def _s_obj(x):
    return x[-1]


def _s_ineq(x):
    n = len(x) - 1
    h = x[:-1]
    t = x[-1]
    c = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
    return t - c


def _s_eq(x):
    n = len(x) - 1
    return np.sum(x[:n]) - n / 2.0


def _s_eq_jac(x):
    n = len(x) - 1
    j = np.zeros(len(x))
    j[:n] = 1.0
    return j


def _s_constraints():
    return [
        {'type': 'ineq', 'fun': _s_ineq},
        {'type': 'eq', 'fun': _s_eq, 'jac': _s_eq_jac}
    ]


def slsqp_optimize(h_init, n, maxiter=5000, t_margin=0.05):
    c5_init = compute_c5_from_h(h_init)
    x0 = np.concatenate([h_init.copy(), [c5_init + t_margin]])
    bounds = [(0.0, 1.0)] * n + [(max(0.001, c5_init * 0.3), 2.0)]
    constraints = _s_constraints()
    result = minimize(_s_obj, x0, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-13})
    if result.x is not None:
        h_out = project_feasible(result.x[:-1], n)
        c5_out = compute_c5_from_h(h_out)
        if c5_out <= c5_init + 1e-10:
            return h_out, c5_out
    return h_init, c5_init


def slsqp_topk(h_init, n, maxiter=5000, t_margin=0.03, topk=15):
    c5_init = compute_c5_from_h(h_init)
    c = np.correlate(h_init, 1.0 - h_init, mode="full") * (2.0 / n)
    top_indices = np.argsort(c)[::-1][:topk]
    x0 = np.concatenate([h_init.copy(), [c5_init + t_margin]])
    bounds = [(0.0, 1.0)] * n + [(max(0.001, c5_init * 0.3), 2.0)]
    constraints = [
        {'type': 'eq', 'fun': _s_eq, 'jac': _s_eq_jac}
    ]
    for idx in top_indices:
        constraints.append({'type': 'ineq', 'fun': lambda x, i=idx: x[-1] - np.correlate(x[:-1], 1.0 - x[:-1], mode="full")[i] * (2.0 / (len(x) - 1))})
    result = minimize(_s_obj, x0, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-13})
    if result.x is not None:
        h_out = project_feasible(result.x[:-1], n)
        c5_out = compute_c5_from_h(h_out)
        if c5_out <= c5_init + 1e-10:
            return h_out, c5_out
    return h_init, c5_init


def smooth_subgrad_step(h, n, T, lr):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    c_max = c.max()
    c_shifted = np.clip(T * (c - c_max), -400, 400)
    weights = np.exp(c_shifted)
    wsum = weights.sum()
    if wsum < 1e-300:
        return h, c_max
    weights = weights / wsum

    grad = np.zeros(n)
    nh = n - 1
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-7:
            continue
        lag = m - nh
        if lag >= 0:
            if n - lag > 0:
                grad[:n - lag] += w * (1.0 - h[lag:])
                grad[lag:] -= w * h[:n - lag]
        else:
            ma = -lag
            if n - ma > 0:
                grad[ma:] += w * (1.0 - h[:n - ma])
                grad[:n - ma] -= w * h[ma:]

    grad *= dx
    h_new = h - lr * grad
    h_new = project_feasible(h_new, n)
    return h_new, c_max


def directed_perturb(h, n, scale, rng, n_active=6):
    h_pert = h.copy()
    c = np.correlate(h_pert, 1.0 - h_pert, mode="full")
    c_max = c.max()
    if c_max <= 1e-15:
        return h_pert
    thresh = c_max - 0.02 * c_max
    active = np.where(c >= thresh)[0]
    if len(active) > n_active:
        idx = np.argsort(c)[::-1][:n_active]
        active = idx

    for m in active:
        lag = m - (n - 1)
        if 0 <= lag < n:
            hi_other = h_pert[lag:] > 0.5
            if hi_other.any():
                h_pert[lag:][hi_other] = np.clip(h_pert[lag:][hi_other] + scale * rng.random(), 0, 1)
            lo_self = h_pert[:n - lag] < 0.3
            if lo_self.any():
                h_pert[:n - lag][lo_self] = np.clip(h_pert[:n - lag][lo_self] - scale * rng.random(), 0, 1)
        elif -n < lag < 0:
            ma = -lag
            if ma < n:
                hi_other = h_pert[:n - ma] > 0.5
                if hi_other.any():
                    h_pert[:n - ma][hi_other] = np.clip(h_pert[:n - ma][hi_other] + scale * rng.random(), 0, 1)
                lo_self = h_pert[ma:] < 0.3
                if lo_self.any():
                    h_pert[ma:][lo_self] = np.clip(h_pert[ma:][lo_self] - scale * rng.random(), 0, 1)

    return project_feasible(h_pert, n)


def fourier_perturb(h, n, scale, rng):
    h_pert = h.copy()
    n_modes = int(rng.integers(1, 4))
    for _ in range(n_modes):
        freq = int(rng.integers(1, max(2, n // 4)))
        phase = rng.random() * 2.0 * np.pi
        amp = scale / np.sqrt(n_modes)
        h_pert += amp * np.cos(2.0 * np.pi * freq * np.arange(n) / n + phase)
    return project_feasible(h_pert, n)


def binary_sharpen(h, n, alpha):
    h_sharp = alpha * h + (1.0 - alpha) * 0.5
    return project_feasible(np.clip(h_sharp, 0, 1), n)
