def coordinate_descent(h, n, budget, step_sizes=None):
    """Fine coordinate descent"""
    if step_sizes is None:
        step_sizes = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    
    best = h.copy()
    best_c5 = evaluate_c5(best)
    t_start = time.time()
    
    for step in step_sizes:
        if time.time() - t_start > budget * 0.8:
            break
        improved = True
        while improved and time.time() - t_start < budget * 0.8:
            improved = False
            for i in range(n):
                for d in [-step, step]:
                    h_new = best.copy()
                    # Maintain sum constraint by adjusting another element
                    j = (i + 1) % n
                    h_new[i] += d
                    h_new[j] -= d
                    if 0 <= h_new[i] <= 1 and 0 <= h_new[j] <= 1:
                        c5 = evaluate_c5(h_new)
                        if c5 < best_c5 - 1e-12:
                            best = h_new
                            best_c5 = c5
                            improved = True
    
    return best, best_c5

            starts.append(project_feasible(np.clip(f(x), 0, 1), n))

        h_tri = np.abs(np.mod(ar / n * freq, 2.0) - 1.0)

The overlap integral I(k) = ∫ h(x)(1-h(x+k)) dx can be rewritten as:
I(k) = ∫ h(x) dx - ∫ h(x)h(x+k) dx = 1 - ∫ h(x)h(x+k) dx (for k=0, it's 1 - ||h||²)

I(k) = ∫₀^{2-k} h(x) dx - ∫₀^{2-k} h(x)h(x+k) dx

The first term depends on k (it's the integral of h over [0, 2-k*dx]).
The second term is the autocorrelation of h at lag k.

Let me try yet another approach: **Use the subgradient of the max function.**

The subgradient of max_k I_k at a point h is in the convex hull of the gradients of the active lags (those achieving the max). So:

Direction = -avg(grad(I_k) for k in active_set)

This is basically a subgradient method. Let me implement this:

For each lag k, I_k(h) = sum_i h[i] * (1 - h[i+k]) (discrete approximation)
grad I_k / grad h[j] = (1-h[j+k]) for j < n-k (from the first term)
                           + (-h[j-k]) for j >= k (from the second term, since h[j] appears in h[i+k] when i = j-k)
                           - h[j] for j < n-k (from the -h[i]h[i+k] term when i=j)
                           - h[j-k] for j >= k (from the -h[i]h[i+k] term when i+k=j)

Wait, let me be more careful.

I_k = sum_{i=0}^{n-1-k} h[i] * (1 - h[i+k])
    = sum_{i=0}^{n-1-k} h[i] - sum_{i=0}^{n-1-k} h[i] * h[i+k]

dI_k / dh[j]:
- From the first sum: +1 if j <= n-1-k, else 0
- From the second sum: 
  - h[j] appears in h[i] when i=j (if j <= n-1-k): contributes -h[j+k]
  - h[j] appears in h[i+k] when i+k=j, i.e., i=j-k (if j-k >= 0 and j-k <= n-1-k, i.e., 0 <= j-k <= n-1-k): contributes -h[j-k]

So: dI_k/dh[j] = 1[j <= n-1-k] - 1[j <= n-1-k]*h[j+k] - 1[j >= k and j-k <= n-1-k]*h[j-k]

Simplifying the conditions:
- j <= n-1-k: first two terms
- j >= k and j <= n-1: third term (since j-k <= n-1-k iff j <= n-1)

So: dI_k/dh[j] = [j <= n-1-k] * (1 - h[j+k]) - [k <= j <= n-1] * h[j-k]

This is the gradient of I_k with respect to h[j].

For the subgradient method on max_k I_k:
1. Find all k* that achieve (or nearly achieve) the maximum
2. Compute the average gradient: g_j = (1/|S|) * sum_{k in S} dI_k/dh[j]
3. Take a step in the direction -g (projected onto feasible set)

            # The actual lag is k_idx - (n-1)
            k = k_idx - (n - 1)
            # dI_k/dh[j] = [j <= n-1-k] * (1 - h[j+k]) - [k <= j <= n-1] * h[j-k]

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return float(np.max(corr) * dx)

def get_lag_values(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    return np.correlate(h, g, mode="full") * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        if np.sum(proj) > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-12:
        ordering = np.argsort(np.abs(result - 0.5))
        remaining = diff
        for idx in ordering:
            if remaining > 0:
                add = min(remaining, 1.0 - result[idx])
                result[idx] += add
                remaining -= add
            elif remaining < 0:
                sub = min(-remaining, result[idx])
                result[idx] -= sub
                remaining += sub
            if abs(remaining) < 1e-14:
                break
    return result

def compute_subgradient(current, n, n_active=5):
    """Compute subgradient of max_k I_k at current point"""
    corr = get_lag_values(current)
    top_idx = np.argsort(corr)[-n_active:]
    grad = np.zeros(n)
    for k_idx in top_idx:
        k = int(k_idx) - (n - 1)
        if k >= 0:
            j_range = np.arange(n - k)
            if len(j_range) > 0:
                grad[j_range] += (1.0 - current[j_range + k])
            j_range2 = np.arange(k, n)
            if len(j_range2) > 0:
                grad[j_range2] -= current[j_range2 - k]
        else:
            m = -k
            j_range = np.arange(m, n)
            if len(j_range) > 0:
                grad[j_range] += (1.0 - current[j_range - m])
            j_range2 = np.arange(n - m)
            if len(j_range2) > 0:
                grad[j_range2] -= current[j_range2 + m]
    grad /= float(len(top_idx))
    return grad, corr[top_idx].max()

def subgradient_optimize(h0, n, budget, n_active=5):
    """Subgradient method with line search and averaging"""
    current = h0.copy()
    best = h0.copy()
    best_c5 = evaluate_c5(best)
    t_start = time.time()
    it = 0
    # Store iterates for averaging
    avg_window = 200
    iterates = []
    
    while time.time() - t_start < budget:
        # Check time more frequently
        if it % 50 == 0 and time.time() - t_start >= budget:
            break
        
        grad, active_val = compute_subgradient(current, n, n_active)
        
        # Line search: try several step sizes
        best_step = None
        best_val = best_c5
        for alpha in [0.05, 0.03, 0.02, 0.01, 0.005, 0.002]:
            new_h = current - alpha * grad
            new_h = project_feasible(new_h, n)
            val = evaluate_c5(new_h)
            if val < best_val:
                best_val = val
                best_step = new_h
        
        if best_step is not None:
            current = best_step
            if best_val < best_c5:
                best_c5 = best_val
                best = current.copy()
        else:
            # No improvement: try smaller steps or perturb
            for alpha in [0.001, 0.0005, 0.0002]:
                new_h = current - alpha * grad
                new_h = project_feasible(new_h, n)
                val = evaluate_c5(new_h)
                if val < best_c5:
                    best_c5 = val
                    best = new_h.copy()
                    current = new_h
                    break
            else:
                # Stuck: add small random perturbation
                current = project_feasible(current + np.random.randn(n) * 0.001, n)
        
        iterates.append(current.copy())
        if len(iterates) > avg_window:
            iterates.pop(0)
        
        it += 1
    
    # Try averaged solution
    if len(iterates) > 50:
        avg_h = np.mean(iterates[-avg_window:], axis=0)
        avg_h = project_feasible(avg_h, n)
        avg_c5 = evaluate_c5(avg_h)
        if avg_c5 < best_c5:
            best = avg_h
            best_c5 = avg_c5

import numpy as np
from scipy.optimize import minimize

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return float(np.max(corr) * dx)

def get_lag_values(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    return np.correlate(h, g, mode="full") * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        if np.sum(proj) > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-12:
        ordering = np.argsort(np.abs(result - 0.5))
        remaining = diff
        for idx in ordering:
            if remaining > 0:
                add = min(remaining, 1.0 - result[idx])
                result[idx] += add
                remaining -= add
            elif remaining < 0:
                sub = min(-remaining, result[idx])
                result[idx] -= sub
                remaining += sub
            if abs(remaining) < 1e-14:
                break
    return result

def compute_subgradient(current, n, n_active):
    corr = get_lag_values(current)
    top_idx = np.argsort(corr)[-n_active:]
    grad = np.zeros(n)
    for k_idx in top_idx:
        k = int(k_idx) - (n - 1)
        if k >= 0:
            j_range = np.arange(n - k)
            if len(j_range) > 0:
                grad[j_range] += (1.0 - current[j_range + k])
            j_range2 = np.arange(k, n)
            if len(j_range2) > 0:
                grad[j_range2] -= current[j_range2 - k]
        else:
            m = -k
            j_range = np.arange(m, n)
            if len(j_range) > 0:
                grad[j_range] += (1.0 - current[j_range - m])
            j_range2 = np.arange(n - m)
            if len(j_range2) > 0:
                grad[j_range2] -= current[j_range2 + m]
    grad /= float(len(top_idx))
    return grad, float(corr[top_idx].max())

def smooth_max_objective(h, n, tau):
    corr = get_lag_values(h)
    v = corr / tau
    vmax = np.max(v)
    return tau * (vmax + np.log(np.sum(np.exp(v - vmax))))

def slsqp_refine(h0, n, budget, tau=0.003):

        return smooth_max_objective(x, n, tau)

        return np.sum(x) - n / 2.0

    def sum_jac(x):
        return np.ones(n)
