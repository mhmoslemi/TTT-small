def evaluate_c5(h):
    n = len(h)

def smooth_max(values, tau):
    v = values / tau
    vmax = np.max(v)
    return tau * (vmax + np.log(np.sum(np.exp(v - vmax))))

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-14:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        if np.sum(proj) > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-14:
        # distribute the diff to elements that can accept it
        idx = np.argsort(np.abs(result - 0.5))
        for i in idx:
            if diff > 0 and result[i] < 1.0:
                amt = min(diff, 1.0 - result[i])
                result[i] += amt
                diff -= amt
            elif diff < 0 and result[i] > 0.0:
                amt = max(diff, -(result[i]))
                result[i] += amt
                diff -= amt
            if abs(diff) < 1e-14:
                break
    return result

def obj_smooth_penalized(x, n, tau, mu):
    dx = 2.0 / n
    g = 1.0 - x
    corr = np.correlate(x, g, mode="full")
    sm = smooth_max(corr, tau) * dx
    sm_grad_part = 0.0  # handled in gradient
    penalty = mu * (np.sum(x) - n/2.0) ** 2
    return sm + penalty

def obj_grad_penalized(x, n, tau, mu):
    dx = 2.0 / n
    g = 1.0 - x
    corr = np.correlate(x, g, mode="full")
    obj = smooth_max(corr, tau) * dx + mu * (np.sum(x) - n/2.0) ** 2
    
    # Gradient of smooth max part
    v = corr / tau
    vmax = np.max(v)
    w = np.exp(v - vmax)
    w /= np.sum(w)
    
    # d corr[k] / dx[i] for lag_k = k - (n-1)
    # = (1-x[i+lag_k]) * 1_{valid} - x[i-lag_k] * 1_{valid}
    grad = np.zeros(n)
    for k in range(2*n-1):
        wk = w[k]
        if wk < 1e-15:
            continue
        lag = k - (n-1)
        # Term 1: (1-x[i+lag]) for i+lag in [0,n-1]
        i1_start = max(0, -lag)
        i1_end = min(n, n - lag)
        if i1_end > i1_start:
            grad[i1_start:i1_end] += wk * (1.0 - x[i1_start+lag:i1_end+lag])
        # Term 2: -x[i-lag] for i-lag in [0,n-1]
        i2_start = max(0, lag)
        i2_end = min(n, n + lag)
        if i2_end > i2_start:
            grad[i2_start:i2_end] -= wk * x[i2_start-lag:i2_end-lag]
    grad *= dx
    
    # Gradient of penalty
    grad += 2.0 * mu * (np.sum(x) - n/2.0) * np.ones(n)
    
    return obj, grad

Let me also think about the subgradient method. The objective is f(h) = max_k c_k(h) * dx where c_k are quadratic.

The subgradient of f at h is any convex combination of the gradients of the active c_k. If we know which k are active (near the max), we can use:
- Pick a random active k
- Take a step in the direction -grad(c_k) (with appropriate scaling)
- Project to feasible set

This is the "random subgradient" method. With proper step-size decay, it converges to a local minimum.

For the step size, I'll use a diminishing sequence: alpha_t = c / sqrt(t) or c/t.

                diff -= amt

                diff -= amt

def subgradient_step(h, n, num_active=5, lr=0.01):
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    c5_vals = corr * dx
    max_val = np.max(c5_vals)
    
    # Find active lags (within some tolerance of max)
    tol = 0.005
    active = np.where(c5_vals > max_val - tol)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c5_vals)])
    if len(active) > num_active:
        # Pick random subset
        idx = np.random.choice(len(active), num_active, replace=False)
        active = active[idx]
    
    # Compute subgradient: average of gradients of active quadratics
    grad = np.zeros(n)
    for k in active:
        lag = k - (n-1)
        # d c[k] / dh[a] = (1-h[a-lag]) * 1_{a-lag in [0,n-1]} - h[a+lag] * 1_{a+lag in [0,n-1]}
        a1s = max(0, lag)

        a2s = max(0, -lag)
        a2e = min(n, n - lag)
        if a2e > a2s:

    # Step

    h_new = project_feasible(h_new, n)
    return h_new, evaluate_c5(h_new)

    lr_init = 0.008
    for it in range(max_iter):
        g = 1.0 - current
        corr = np.correlate(current, g, mode="full")
        vals = corr * dx
        max_val = np.max(vals)
        tol = max(0.003, 0.01 * max_val)
        active = np.where(vals > max_val - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(vals)])
        if len(active) > 6:
            idx = rand.choice(len(active), 6, replace=False)
            active = active[idx]
        grad = np.zeros(n)
        for k in active:
            lag = k - (n - 1)
            a1s = max(0, lag)
            a1e = min(n, n + lag)
            if a1e > a1s:
                grad[a1s:a1e] += (1.0 - current[a1s - lag:a1e - lag])
            a2s = max(0, -lag)
            a2e = min(n, n - lag)
            if a2e > a2s:
                grad[a2s:a2e] -= current[a2s + lag:a2e + lag]
        grad *= dx / len(active)
        lr = lr_init / (1.0 + 0.05 * it)
        h_new = current - lr * grad
        h_new = project_feasible(h_new, n)
        c5_new = evaluate_c5(h_new)
        if c5_new < current_c5:
            current = h_new
            current_c5 = c5_new
        else:
            lr = lr * 0.7
            h_new = current - lr * grad
            h_new = project_feasible(h_new, n)
            c5_new = evaluate_c5(h_new)
            if c5_new < current_c5:
                current = h_new
                current_c5 = c5_new
    return current, current_c5

def coordinate_descent(h, n, max_sweeps):
    best = h.copy()
    best_c5 = evaluate_c5(best)
    steps = [0.015, 0.008, 0.004, 0.002, 0.001, -0.001, -0.002, -0.004, -0.008, -0.015]
    for sweep in range(max_sweeps):
        improved = False
        for i in range(n):
            for d in steps:
                h_new = best.copy()
                h_new[i] = np.clip(h_new[i] + d, 0.0, 1.0)
                h_new = project_feasible(h_new, n)
                c5 = evaluate_c5(h_new)
                if c5 < best_c5 - 1e-12:
                    best = h_new
                    best_c5 = c5
                    improved = True
        if not improved:
            break
    return best, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')

    parent_h = None
    try:
        parent_h = np.array(initial_h_values, dtype=float)
    except Exception:
        pass

    n_list = []
    if parent_h is not None:
        n_list.append(len(parent_h))
    for nn in [75, 77, 79, 81, 83]:
        if nn not in n_list:
            n_list.append(nn)

    phase1_end = start_time + budget_s * 0.35
    phase2_end = start_time + budget_s * 0.60
    phase3_end = start_time + budget_s * 0.80
    phase4_end = start_time + budget_s * 0.93
    phase5_end = start_time + budget_s * 0.995

    for n in n_list:
        if time.time() > phase1_end:
            break
        bnds = [(0.0, 1.0)] * n
        starts = []
        if parent_h is not None:
            if len(parent_h) == n:
                starts.append(parent_h.copy())
                starts.append(project_feasible(parent_h + rng.randn(n) * 0.05, n))
            else:
                xp = np.linspace(0, 2, len(parent_h), endpoint=False)
                x_new = np.linspace(0, 2, n, endpoint=False)
                f = interp1d(xp, parent_h, kind='cubic', fill_value='extrapolate', bounds_error=False)
                starts.append(project_feasible(f(x_new), n))
                starts.append(project_feasible(f(x_new) + rng.randn(n) * 0.08, n))

        x = np.linspace(0, 2, n, endpoint=False)
        ar = np.arange(n, dtype=float)
        for freq in [1.0, 1.5, 2.0, 2.5, 3.0, 3.5, 4.0]:
            for phase in [0.0, np.pi / 3.0, 2.0 * np.pi / 3.0]:
                for amp in [0.3, 0.4, 0.45]:
                    h = 0.5 + amp * np.sin(2.0 * np.pi * freq * x / 2.0 + phase)
                    starts.append(project_feasible(h, n))
        for freq in [2, 3, 4, 5]:
            h_tri = np.abs(np.mod(ar / n * freq, 2.0) - 1.0)
            starts.append(project_feasible(h_tri, n))
        for _ in range(4):
            starts.append(project_feasible(rng.rand(n), n))
        starts.append(project_feasible(np.linspace(0.1, 0.9, n), n))
        starts.append(project_feasible(np.linspace(0.9, 0.1, n), n))
        for _ in range(3):
            center = rng.uniform(0.15 * n, 0.85 * n)
            width = rng.uniform(0.1 * n, 0.4 * n)
            h = 0.2 + 0.65 * np.exp(-0.5 * ((ar - center) / width) ** 2)
            starts.append(project_feasible(h, n))

        mu = 200.0
        tau_sched = [0.25, 0.1, 0.04, 0.015]
        for h0 in starts:
            if time.time() > phase1_end:
                break
            current = h0.copy()
            for tau in tau_sched:
                if time.time() > phase1_end:
                    break
                try:
                    res = minimize(
                        obj_smooth_only, current,
                        args=(n, tau, mu),
                        jac=grad_smooth_only,
                        method='L-BFGS-B',
                        bounds=bnds,
                        options={'maxiter': 300, 'ftol': 1e-14, 'gtol': 1e-12}
                    )
                    current = project_feasible(res.x, n)
                except Exception:
                    pass
            c5 = evaluate_c5(current)
            if c5 < best_c5:
                best_c5 = c5
                best_h = current.copy()

    if best_h is not None and time.time() < phase2_end:
        n = len(best_h)
        bnds = [(0.0, 1.0)] * n
        current = best_h.copy()
        for mu, tau_list in [(500.0, [0.01, 0.005, 0.002]), (2000.0, [0.002, 0.001, 0.0005])]:
            if time.time() > phase2_end:
                break
            for tau in tau_list:
                if time.time() > phase2_end:
                    break
                try:
                    res = minimize(
                        obj_smooth_only, current,
                        args=(n, tau, mu),
                        jac=grad_smooth_only,
                        method='L-BFGS-B',
                        bounds=bnds,
                        options={'maxiter': 500, 'ftol': 1e-15, 'gtol': 1e-14}
                    )
                    current = project_feasible(res.x, n)
                    c5 = evaluate_c5(current)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = current.copy()
                except Exception:
                    pass

    if best_h is not None and time.time() < phase3_end:
        n = len(best_h)
        elapsed_frac = (time.time() - start_time) / budget_s
        remaining = phase3_end - time.time()
        max_iter = int(remaining / 0.01)
        max_iter = max(100, min(max_iter, 20000))
        h_sub, c5_sub = subgradient_polish(best_h, n, max_iter, rng)
        if c5_sub < best_c5:
            best_c5 = c5_sub
            best_h = h_sub

    if best_h is not None and time.time() < phase4_end:
        n = len(best_h)
        bnds = [(0.0, 1.0)] * n
        current = best_h.copy()
        mu = 5000.0
        for tau in [0.0005, 0.0002, 0.0001]:
            if time.time() > phase4_end:
                break
            try:
                res = minimize(
                    obj_smooth_only, current,
                    args=(n, tau, mu),
                    jac=grad_smooth_only,
                    method='L-BFGS-B',
                    bounds=bnds,
                    options={'maxiter': 500, 'ftol': 1e-16, 'gtol': 1e-15}
                )
                current = project_feasible(res.x, n)
                c5 = evaluate_c5(current)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = current.copy()
            except Exception:
                pass
        h_sub2, c5_sub2 = subgradient_polish(best_h, n, 3000, rng)
        if c5_sub2 < best_c5:
            best_c5 = c5_sub2
            best_h = h_sub2

    if best_h is not None and time.time() < phase5_end:
        n = len(best_h)
        remaining_sweeps = int((phase5_end - time.time()) / 0.5)
        remaining_sweeps = max(2, min(remaining_sweeps, 15))
        h_cd, c5_cd = coordinate_descent(best_h, n, remaining_sweeps)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd

    if best_h is None:
        if parent_h is not None:
            best_h = parent_h.copy()
        else:
            n = 77
            best_h = project_feasible(np.full(n, 0.5), n)
        best_c5 = evaluate_c5(best_h)

    final_c5 = evaluate_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
