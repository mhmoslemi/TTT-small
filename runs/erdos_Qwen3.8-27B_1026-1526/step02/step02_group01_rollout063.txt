```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_c_at_lag(h, k_idx):
    n = len(h)
    dx = 2.0 / n
    shift = k_idx - (n - 1)
    grad = np.zeros(n)
    if shift >= 0:
        if shift < n:
            if shift > 0:
                grad[shift:] += (1.0 - h[:-shift])
                grad[:n - shift] -= h[shift:]
            else:
                grad[:] += (1.0 - h)
                grad[:] -= h
    else:
        s = -shift
        if s < n:
            grad[:n - s] += (1.0 - h[s:])
            grad[s:] -= h[:n - s]
    return grad * dx

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    n_corr = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / n_corr) / alpha
    grad = np.zeros(n)
    for k in range(n_corr):
        wk = w[k]
        if wk < 1e-12:
            continue
        shift = k - (n - 1)
        if shift >= 0:
            if shift < n:
                if shift > 0:
                    grad[shift:] += wk * (1.0 - h[:-shift])
                    grad[:n - shift] -= wk * h[shift:]
                else:
                    grad[:] += wk * (1.0 - h)
                    grad[:] -= wk * h
        else:
            s2 = -shift
            if s2 < n:
                grad[:n - s2] += wk * (1.0 - h[s2:])
                grad[s2:] -= wk * h[:n - s2]
    return sm_val, grad * dx

def slsqp_smooth_max(h0, n, target_sum, alpha, maxiter):
    def obj(x):
        v, _ = smooth_max_obj_grad(x, alpha)
        return v
    def grad(x):
        _, g = smooth_max_obj_grad(x, alpha)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def slsqp_single_lag(h0, n, target_sum, k_idx, maxiter):
    dx = 2.0 / n
    def obj(x):
        return float(np.correlate(x, 1.0 - x, mode='full')[k_idx] * dx)
    def grad(x):
        return grad_c_at_lag(x, k_idx)
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def cyclic_lag_optimize(h0, n, target_sum, max_rounds, time_limit):
    h = h0.copy()
    c5_cur = compute_c5(h)
    h_best = h.copy()
    t_start = time.time()
    for rnd in range(max_rounds):
        if time.time() - t_start > time_limit:
            break
        corr = compute_corr_values(h)
        cmax = np.max(corr)
        active = np.where(corr >= cmax - 1e-5)[0]
        improved = False
        for k_idx in active[:5]:
            if time.time() - t_start > time_limit:
                break
            h_new = slsqp_single_lag(h, n, target_sum, k_idx, maxiter=120)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-15:
                h = h_new
                c5_cur = c5_new
                h_best = h.copy()
                improved = True
        if not improved:
            break
    return h_best, c5_cur

def subgradient_optimize(h0, n, target_sum, n_iters, t_start_outer, time_limit):
    h = h0.copy()
    c5_cur = compute_c5(h)
    h_best = h.copy()
    sigma = 0.01
    for t in range(1, n_iters + 1):
        if time.time() - t_start_outer > time_limit:
            break
        corr = compute_corr_values(h)
        cmax = np.max(corr)
        eps = max(1e-4, 0.01 * cmax)
        active = np.where(corr >= cmax - eps)[0]
        g = np.zeros(n)
        for k in active:
            g += grad_c_at_lag(h, k)
        g /= len(active)
        eta = sigma / np.sqrt(t)
        h_new = h - eta * g
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur:
            c5_cur = c5_new
            h_best = h_new.copy()
            sigma = min(sigma * 1.02, 0.1)
        else:
            sigma *= 0.95
        h = h_new
    return h_best, c5_cur

def coordinate_descent_fine(h0, n, target_sum, max_rounds, rng, time_limit, t_start_outer):
    h = h0.copy()
    c5_cur = compute_c5(h)
    h_best = h.copy()
    step_sizes = [0.02, 0.01, 0.005, 0.002, 0.001, -0.001, -0.002, -0.005, -0.01, -0.02]
    for rnd in range(max_rounds):
        if time.time() - t_start_outer > time_limit:
            break
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            if time.time() - t_start_outer > time_limit:
                break
            best_val = c5_cur
            best_hi = h[i]
            for d in step_sizes:
                hi_try = h[i] + d
                if hi_try < 0 or hi_try > 1:
                    continue
                h_try = h.copy()
                h_try[i] = hi_try
                h_try = project_to_sum(h_try, target_sum)
                c5_try = compute_c5(h_try)
                if c5_try < best_val - 1e-15:
                    best_val = c5_try
                    best_hi = hi_try
            if best_hi != h[i]:
                h[i] = best_hi
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True
                if c5_cur < compute_c5(h_best) - 1e-15:
                    h_best = h.copy()
        if not improved:
            break
    return h_best, c5_cur

def multi_lag_average_slsqp(h0, n, target_sum, k_list, maxiter):

    n_k = len(k_list)

        total = 0.0

                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)
