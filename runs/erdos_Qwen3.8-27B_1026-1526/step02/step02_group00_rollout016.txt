This is related to the problem of finding binary sequences with good autocorrelation properties (like Barker sequences, M-sequences, etc.).

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    shifted = corr - m_val
    max_shifted = np.max(shifted)
    exp_vals = np.exp(alpha * (shifted - max_shifted))
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + max_shifted + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

_global_target_sum = 0.0

def _obj_args(h, alpha, target_sum, n):
    return smooth_max_and_grad(h, alpha)[0]

def _jac_args(h, alpha, target_sum, n):
    return smooth_max_and_grad(h, alpha)[1]

def _cons_fun(h):
    return np.sum(h) - _global_target_sum

def _cons_jac(h):
    return np.ones(len(h))

def slsqp_optimize(h0, n, target_sum, alpha, maxiter):
    global _global_target_sum
    _global_target_sum = target_sum
    cons = {'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_obj_args, h0.copy(), method='SLSQP',
                       jac=_jac_args, args=(alpha, target_sum, n),
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def binary_flip_optimize(h_bin, n, target_sum, rng, max_rounds=200, use_sa=False, T_init=0.005):
    h = h_bin.copy()
    c5_cur = compute_c5(h)
    T = T_init
    for rnd in range(max_rounds):
        improved = False
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_c5 = c5_cur
        best_pair = None
        n_trials = min(400, len(ones) * len(zeros))
        for _ in range(n_trials):
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h_test = h.copy()
            h_test[i] = 0.0
            h_test[j] = 1.0
            c5_test = compute_c5(h_test)
            delta = c5_test - c5_cur
            if use_sa and delta > 0 and rng.random() < np.exp(-delta / max(T, 1e-12)):

            h[j] = 1.0
            c5_cur = best_c5
            improved = True
            T *= 0.97

    return h, c5_cur

def subgradient_optimize(h0, n, target_sum, max_iter, step_init, momentum):
    h = h0.copy()
    v = np.zeros(n)
    step = step_init
    best_h = h.copy()
    best_val = compute_c5(h)
    for t in range(max_iter):
        corr = compute_corr(h)
        m_val = np.max(corr)
        k_star = np.argmax(corr)
        j_idx = np.arange(n)
        first_idx = j_idx - k_star + n - 1
        second_idx = j_idx + k_star - n + 1
        valid_f = (first_idx >= 0) & (first_idx < n)
        valid_s = (second_idx >= 0) & (second_idx < n)
        fc = np.clip(first_idx, 0, n - 1)
        sc = np.clip(second_idx, 0, n - 1)
        subgrad = np.zeros(n)
        subgrad[valid_f] -= (1.0 - h[fc[valid_f]])
        subgrad[valid_s] += h[sc[valid_s]]
        v = momentum * v - step * subgrad
        h_new = project_to_sum(h + v, target_sum)
        val_new = compute_c5(h_new)
        if val_new < best_val:
            best_val = val_new
            best_h = h_new.copy()
        h = h_new
        step *= 0.9995
    return best_h, best_val

    # Phase 6: Final binary polish with longer SA
    if check_budget(t_start, budget_s, 80):
        h_bin = make_binary(best_h, best_n, ts_best, rng)
        h_bin, c5_bin = binary_flip_optimize(h_bin, best_n, ts_best, rng, max_rounds=500, use_sa=True, T_init=0.004)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
        if check_budget(t_start, budget_s, 40):
            h_sm = slsqp_optimize(h_bin, best_n, ts_best, 1000, maxiter=2000)
            c5_sm = compute_c5(h_sm)
            if c5_sm < best_c5:
                best_c5 = c5_sm
                best_h = h_sm.copy()

    # Phase 7: Final high-alpha polish
    if check_budget(t_start, budget_s, 40):
        for alpha in [3000, 15000]:
            if not check_budget(t_start, budget_s, 30):
                break
            h_pol = slsqp_optimize(best_h, best_n, ts_best, alpha, maxiter=3000)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
```
