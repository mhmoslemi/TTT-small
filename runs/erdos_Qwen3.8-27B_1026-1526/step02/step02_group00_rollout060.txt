```python
def optimize_power_mean(h0, n, target_sum, p, mu, maxiter):
    def obj(bx):
        h = bx
        corr = np.correlate(h, 1.0-h, mode="full")
        corr_pos = np.maximum(corr, 1e-15)
        S = np.sum(corr_pos**p)
        f = S**(1.0/p) / p
        penalty = mu * (np.sum(h) - target_sum)**2
        return f + penalty
    
    def grad_func(bx):
        h = bx
        corr = np.correlate(h, 1.0-h, mode="full")
        corr_pos = np.maximum(corr, 1e-15)
        S = np.sum(corr_pos**p)
        # gradient of power mean
        alpha = S**(1.0/p - 1.0)
        w = alpha * corr_pos**(p-1)
        j_idx = np.arange(n)[:, None]
        m_idx = np.arange(2*n-1)[None, :]
        fi = j_idx - m_idx + n - 1
        si = j_idx + m_idx - n + 1
        vf = (fi >= 0) & (fi < n)
        vs = (si >= 0) & (si < n)
        fc = np.clip(fi, 0, n-1)
        sc = np.clip(si, 0, n-1)
        grad_pm = np.sum(w[None,:] * ((1.0-h[fc])*vf - h[sc]*vs), axis=1)
        # penalty gradient
        pg = 2.0 * mu * (np.sum(h) - target_sum) * np.ones(n)
        return grad_pm + pg
    
    res = minimize(obj, h0.copy(), jac=grad_func, method='L-BFGS-B',
                   bounds=[(0,1)]*n, options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8})
    return project_to_sum(res.x, target_sum)
```

```python
    best_h = project_to_sum(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
```
