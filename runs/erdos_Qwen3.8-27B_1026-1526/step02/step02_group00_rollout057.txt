So delta_k = (1-h[j+k]) - (1-h[i+k]) = h[i+k] - h[j+k]

This is for the "naive" correlation without boundary effects. With boundary effects (finite array), we need to be more careful, but for a periodic interpretation, it's exact.

So for each swap (i, j), the change in all correlations is:
delta_k = h[i+k] - h[j+k] for all k

This is just (shifted h by i) - (shifted h by j), which we can compute in O(n) time.

For the 0-1 local search with simulated annealing:
1. Pick a random swap (i, j) where h[i]=1, h[j]=0
2. Compute delta_k for all k
3. Compute new_corr_k = corr_k + delta_k
4. Compute new_max = max(new_corr_k)
5. Accept if new_max < old_max (always) or with probability exp(-(new_max - old_max)/T) (simulated annealing)

This is O(n) per step, which is very fast.

        if new_max < current_max or rng.uniform() < np.exp(-(new_max - current_max) / T):
            h[i] = 0.0
            h[j] = 1.0
            ones_mask[i] = False
            ones_mask[j] = True
            
            # Update shift_mat
            # Column (i-a)%n of row a becomes 0 (was 1)
            # Column (j-a)%n of row a becomes 1 (was 0)
            for a in range(n):
                ki = (i - a) % n
                kj = (j - a) % n
                shift_mat[a, ki] = 0.0
                shift_mat[a, kj] = 1.0
            
            c = c + delta
            current_max = new_max
    
    return h, current_max

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    n_corr = len(corr)
    m_val = np.max(corr)
    shifted = corr - m_val
    shifted = np.maximum(shifted, -60.0 / alpha)
    exp_vals = np.exp(alpha * shifted)
    s = np.sum(exp_vals)
    if s < 1e-300:
        w = np.full(n_corr, 1.0 / n_corr)
    else:
        w = exp_vals / s
    obj = m_val + np.log(max(s, 1e-300)) / alpha
    lag = np.arange(n_corr) - (n - 1)
    j_idx = np.arange(n)[:, None]
    second_idx = j_idx + lag[None, :]
    valid_s = (second_idx >= 0) & (second_idx < n)
    first_idx = j_idx - lag[None, :]
    valid_f = (first_idx >= 0) & (first_idx < n)
    sc = np.clip(second_idx, 0, n - 1)
    fc = np.clip(first_idx, 0, n - 1)
    cs = valid_s.astype(float) * (1.0 - h[sc]) * w[None, :]
    cf = valid_f.astype(float) * h[fc] * w[None, :]
    grad = np.sum(cs - cf, axis=1)
    return obj, grad

def lbfgs_optimize(h0, n, target_sum, alphas, maxiter_per_alpha, penalty=1e6):
    ts = target_sum
    h = h0.copy()
    for alpha in alphas:
        def obj_and_grad(x):
            om, og = smooth_max_obj_grad(x, alpha)
            pv = penalty * (np.sum(x) - ts) ** 2
            pg = 2.0 * penalty * (np.sum(x) - ts) * np.ones(n)
            return om + pv, og + pg
        try:
            res = minimize(obj_and_grad, h.copy(), method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-15, 'gtol': 1e-10})
            h = project_to_sum(res.x, ts)
        except Exception:
            pass
    return h

def 01_scout_local_search(h01, n, rng, n_iters, T_init, T_final):
    h = h01.copy().astype(float)
    rows = np.arange(n)[:, None]
    cols = np.arange(n)[None, :]
    shift_mat = h[(rows + cols) % n]
    h_float = h
    n_ones = int(np.sum(h_float > 0.5))
    idx_arr = np.arange(n)
    R = np.array([np.sum(h_float * h_float[(idx_arr + k) % n]) for k in range(n)])
    c = float(n_ones) - R
    current_max = float(np.max(c))
    ones_set = set(np.where(h > 0.5)[0].tolist())
    zeros_set = set(np.where(h < 0.5)[0].tolist())
    for it in range(n_iters):
        progress = it / max(n_iters - 1, 1)
        T = T_init * (T_final / T_init) ** progress
        T = max(T, 0.005)
        ones_arr = list(ones_set)
        zeros_arr = list(zeros_set)
        if not ones_arr or not zeros_arr:
            break
        i = rng.choice(ones_arr)
        j = rng.choice(zeros_arr)
        delta = shift_mat[i, :] - shift_mat[j, :]
        new_max = float(np.max(c + delta))
        if new_max < current_max or (new_max - current_max) < 0.0 and False or rng.uniform() < np.exp(-max(new_max - current_max, 0) / T):
            h[i] = 0.0
            h[j] = 1.0
            ones_set.discard(i)
            ones_set.add(j)
            zeros_set.discard(j)
            zeros_set.add(i)
            a_arr = np.arange(n)
            ki_arr = (i - a_arr) % n
            kj_arr = (j - a_arr) % n
            shift_mat[a_arr, ki_arr] = 0.0
            shift_mat[a_arr, kj_arr] = 1.0
            c = c + delta
            current_max = new_max
    return h, current_max

def best_improve_01(h01, n, rng, max_rounds=20):
    h = h01.copy().astype(float)
    rows = np.arange(n)[:, None]
    cols = np.arange(n)[None, :]
    shift_mat = h[(rows + cols) % n]
    h_float = h
    idx_arr = np.arange(n)
    R = np.array([np.sum(h_float * h_float[(idx_arr + k) % n]) for k in range(n)])
    n_ones = int(np.sum(h_float > 0.5))
    c = float(n_ones) - R
    current_max = float(np.max(c))
    for _ in range(max_rounds):
        ones_arr = np.where(h > 0.5)[0]
        zeros_arr = np.where(h < 0.5)[0]
        if len(ones_arr) == 0 or len(zeros_arr) == 0:
            break
        n_trials = min(500, len(ones_arr) * len(zeros_arr))
        best_delta_max = current_max
        best_i, best_j = None, None
        for _ in range(n_trials):
            i = rng.choice(ones_arr)
            j = rng.choice(zeros_arr)
            delta = shift_mat[i, :] - shift_mat[j, :]
            nm = float(np.max(c + delta))
            if nm < best_delta_max - 1e-10:
                best_delta_max = nm
                best_i, best_j = i, j
        if best_i is None:
            break
        i, j = best_i, best_j
        h[i] = 0.0
        h[j] = 1.0
        a_arr = np.arange(n)
        ki_arr = (i - a_arr) % n
        kj_arr = (j - a_arr) % n
        shift_mat[a_arr, ki_arr] = 0.0
        shift_mat[a_arr, kj_arr] = 1.0
        delta = shift_mat[j, :] - shift_mat[i, :]
        c = c + (h[i] * 0 + (1 - h[j]) * shift_mat[i, :] - h[i] * (1 - h[j]) * shift_mat[j, :])
        h_float = h
        R = np.array([np.sum(h_float * h_float[(idx_arr + k) % n]) for k in range(n)])
        c = float(n_ones) - R
        current_max = float(np.max(c))
    return h, current_max

def upsample_periodic(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    h_ext = np.concatenate([h_old, h_old[:n_new - n_old + 1] if n_new > n_old else h_old])
    x_ext = np.linspace(0, 2.0, n_old, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old, period=2.0)
    return h_new

def random_01(n, rng):
    h = np.zeros(n)
    h[:n // 2] = 1.0
    rng.shuffle(h)
    return h

def structured_01(n, rng):
    patterns = []
    for period in range(2, n + 1):
        if n % period == 0:
            ones_per_period = period // 2
            base = np.zeros(period)
            base[:ones_per_period] = 1.0
            if rng.random() < 0.5:
                rng.shuffle(base)
            h = np.tile(base, n // period).astype(float)
            patterns.append(h)
    if patterns:
        return patterns[rng.randint(len(patterns))]
    return random_01(n, rng)
