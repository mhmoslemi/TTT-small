At the optimum, let K* be the set of lags achieving the maximum. The KKT conditions suggest that at the optimum:
- For k in K*: the gradient of f_k with respect to h should be "balanced" (linearly dependent with the gradient of the equality constraint and the inequality constraints at the boundary).
- For k not in K*: f_k(h) < C₅.

The active set approach:
1. Compute all f_k values
2. Let K_active = {k : f_k > C₅ - ε} for some small ε
3. Optimize: min max_{k in K_active} f_k(h) subject to constraints
4. Update K_active
5. Repeat until convergence

For step 3, we can use SLSQP with constraints only for the active lags, which is a much smaller problem.

Let me think about another approach: **projected gradient with subgradient of the max**.

The function C₅(h) = max_k f_k(h) is the max of smooth functions, so it's convex if all f_k are convex. But as we discussed, f_k is not convex in general.

However, we can use a subgradient: any f_k' where f_k is active is a subgradient of the max. So we can do:
- Compute h, find active lags K*
- Compute subgradient: g = (1/|K*|) * sum_{k in K*} ∇f_k
- Update: h_new = project(h - α * g)

This is a subgradient method, which converges slowly but is simple. We could combine it with line search or momentum.

Actually, a better approach: **use the smooth max approximation with adaptive temperature**.

The log-sum-exp approximation: C₅_T(h) = (1/T) log(sum_k exp(T * f_k(h)))
approximates max_k f_k(h) as T → ∞.

The gradient is: ∇C₅_T = (sum_k f_k'(h) exp(T*f_k(h))) / (sum_k exp(T*f_k(h)))
= weighted average of gradients of active lags.

This is what the L-BFGS-B approach does. The issue is choosing T:
- Too small: poor approximation of the max
- Too large: the gradient is dominated by a single lag, losing information

**Adaptive T strategy**: Start with small T (smooth landscape) and gradually increase T (sharper maximum). This is like simulated annealing.

    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

        else:
            mask = h > 1e-15
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
    h = np.clip(h, 0.0, 1.0)
    return h


def project_onto_set(y, n):
    target = n / 2.0
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        h = np.clip(y - mid, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h


def c5_lse_shifted(x, T, n):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full") * dx
    c_max = c.max()
    if c_max < -1e-10:
        return float(c_max)
    shifted = T * (c - c_max)
    shifted = np.clip(shifted, -500, 500)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / T)


def lse_smooth_obj(x, T, penalty, n):
    c5_val = c5_lse_shifted(x, T, n)
    sum_viol = np.sum(x) - n / 2.0
    return c5_val + penalty * sum_viol ** 2


def slsqp_t_obj(x):
    return x[-1]


def slsqp_ineq_full(x):
    n = len(x) - 1
    h = x[:-1]
    t = x[-1]
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return t - c * dx


def slsqp_eq(x):
    n = len(x) - 1
    h = x[:-1]
    return np.sum(h) - n / 2.0


def slsqp_eq_jac(x):
    n = len(x) - 1
    jac = np.zeros(len(x))
    jac[:n] = 1.0
    return jac


def slsqp_optimize_full(h_init, n, maxiter=5000, t_margin=0.05):
    c5_init = compute_c5_from_h(h_init)
    t_init = c5_init + t_margin
    x0 = np.concatenate([h_init, [t_init]])
    bounds = [(0.0, 1.0)] * n + [(max(0.001, c5_init * 0.3), 2.0)]
    constraints = [
        {'type': 'ineq', 'fun': slsqp_ineq_full},
        {'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac}
    ]
    result = minimize(slsqp_t_obj, x0, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
    if result.x is not None:
        h_out = project_feasible(result.x[:-1], n)
        return h_out, compute_c5_from_h(h_out)
    return h_init, c5_init


_active_k_indices = None


def slsqp_ineq_active(x):
    n = len(x) - 1
    h = x[:-1]
    t = x[-1]
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return t - c[_active_k_indices] * dx


def slsqp_optimize_active(h_init, n, k_active, maxiter=3000, t_margin=0.03):
    global _active_k_indices
    _active_k_indices = k_active
    c5_init = compute_c5_from_h(h_init)
    t_init = c5_init + t_margin
    x0 = np.concatenate([h_init, [t_init]])
    bounds = [(0.0, 1.0)] * n + [(max(0.001, c5_init * 0.3), 2.0)]
    constraints = [
        {'type': 'ineq', 'fun': slsqp_ineq_active},
        {'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac}
    ]
    result = minimize(slsqp_t_obj, x0, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
    if result.x is not None:
        h_out = project_feasible(result.x[:-1], n)
        return h_out, compute_c5_from_h(h_out)
    return h_init, c5_init


def find_active_lags(h, n, threshold_frac=0.03):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    c_max = c.max()
    active = np.where(c > c_max - threshold_frac)[0]
    return active


def lse_optimize(h_init, n, T, penalty, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    result = minimize(lse_smooth_obj, h_init, args=(T, penalty, n),
                      method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-16})
    if result.x is not None:
        h_out = project_feasible(result.x, n)
        return h_out, compute_c5_from_h(h_out)
    return h_init, compute_c5_from_h(h_init)


def nesterov_accelerated(h_init, n, T, steps, lr_init, penalty):
    h_curr = h_init.copy()
    h_prev = h_init.copy()
    best_h = h_init.copy()
    best_c5 = compute_c5_from_h(h_init)
    y = h_init.copy()
    momentum = 0.0
    for t in range(1, steps + 1):
        mu = (1.0 + np.sqrt(1.0 + 4.0 * momentum ** 2)) / 2.0
        momentum = mu - 1.0
        y = h_curr + momentum / (1.0 + momentum) * (h_curr - h_prev)
        dx_v = 2.0 / n
        c = np.correlate(y, 1.0 - y, mode="full") * dx_v
        c_max = c.max()
        shifted = T * (c - c_max)
        shifted = np.clip(shifted, -500, 500)
        weights = np.exp(shifted)
        weights = weights / weights.sum()
        grad = np.zeros(n)
        for i in range(len(c)):
            k_shift = i - (n - 1)
            w = weights[i]
            if abs(w) < 1e-15:
                continue
            if k_shift == 0:
                contrib = dx_v * (1.0 - 2.0 * y)
            else:
                if k_shift > 0:
                    valid = np.arange(n - k_shift)
                    contrib = np.zeros(n)
                    contrib[valid] = dx_v * (1.0 - y[valid + k_shift])
                else:
                    ks = -k_shift
                    valid = np.arange(ks, n)
                    contrib = np.zeros(n)
                    contrib[valid] = dx_v * (1.0 - y[valid - ks])
            grad += w * contrib
        sum_viol = np.sum(y) - n / 2.0
        grad += penalty * 2.0 * sum_viol
        lr = lr_init / (1.0 + 0.01 * t)
        h_next = y - lr * grad
        h_next = project_onto_set(h_next, n)
        c5_curr = compute_c5_from_h(h_next)
        if c5_curr < best_c5:
            best_c5 = c5_curr
            best_h = h_next.copy()
        h_prev = h_curr
        h_curr = h_next
    return best_h, best_c5


def time_rem(t0, budget_s):
    return budget_s - (time.time() - t0)


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(270, 0.5)

    n = len(h0)
    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c5 = compute_c5_from_h(best_h)
    n_best = n

    rem = time_rem(t0, budget_s)

    if rem > 150:
        best_h, best_c5 = slsqp_optimize_full(best_h, n_best, maxiter=4000, t_margin=0.1)
        rem = time_rem(t0, budget_s)

    if rem > 80:
        for T_val in [10, 30, 80, 200, 500, 1000]:
            if rem < 30:
                break
            h_new, c5_new = lse_optimize(best_h, n_best, T=T_val, penalty=100.0, maxiter=1500)
            if c5_new < best_c5 - 1e-13:
                best_h = h_new
                best_c5 = c5_new
            rem = time_rem(t0, budget_s)

    if rem > 60:
        best_h, best_c5 = slsqp_optimize_full(best_h, n_best, maxiter=5000, t_margin=0.02)
        rem = time_rem(t0, budget_s)

    n_active_rounds = 5
    for rnd in range(n_active_rounds):
        if rem < 40:
            break
        k_act = find_active_lags(best_h, n_best, threshold_frac=0.02)
        if len(k_act) > 80:
            idx_order = np.argsort(np.correlate(best_h, 1.0 - best_h, mode="full"))[::-1]
            k_act = idx_order[:80]
        h_new, c5_new = slsqp_optimize_active(best_h, n_best, k_act, maxiter=4000, t_margin=0.01)
        if c5_new < best_c5 - 1e-13:
            best_h = h_new
            best_c5 = c5_new
        c_all = np.correlate(best_h, 1.0 - best_h, mode="full") * (2.0 / n_best)
        c_max = c_max_cur = c_all.max()
        non_active = np.setdiff1d(np.arange(len(c_all)), k_act)
        if len(non_active) > 0:
            max_non_active = c_all[non_active].max()
            if max_non_active > c_max - 1e-10:
                k_act_new = find_active_lags(best_h, n_best, threshold_frac=0.01)
                if len(k_act_new) > 80:
                    idx_order2 = np.argsort(c_all)[::-1]
                    k_act_new = idx_order2[:80]
                h_new, c5_new = slsqp_optimize_active(best_h, n_best, k_act_new, maxiter=3000, t_margin=0.01)
                if c5_new < best_c5 - 1e-13:
                    best_h = h_new
                    best_c5 = c5_new
        rem = time_rem(t0, budget_s)

    if rem > 60:
        h_new, c5_new = nesterov_accelerated(best_h, n_best, T=200, steps=200, lr_init=0.05, penalty=50.0)
        if c5_new < best_c5 - 1e-13:
            best_h = h_new
            best_c5 = c5_new
        rem = time_rem(t0, budget_s)

    n_perturb = 45
    for i in range(n_perturb):
        if rem < 25:
            break
        frac = i / max(1, n_perturb)
        scale = 0.12 * (1.0 - frac) ** 0.5 + 0.004
        r = rng.random()
        if r < 0.2:
            h_pert = best_h.copy()
            n_modes = rng.integers(1, 5)
            for _ in range(n_modes):
                freq = rng.integers(1, max(2, n_best // 3))
                phase = rng.random() * 2.0 * np.pi
                h_pert += (scale / np.sqrt(n_modes)) * np.cos(2.0 * np.pi * freq * np.arange(n_best) / n_best + phase)
        elif r < 0.35:
            h_pert = best_h + scale * rng.standard_normal(n_best)
        elif r < 0.5:
            h_pert = best_h.copy()
            n_blocks = rng.integers(2, 6)
            for _ in range(n_blocks):
                bsize = rng.integers(4, max(5, n_best // 10))
                start = rng.integers(0, max(1, n_best - bsize))
                h_pert[start:start + bsize] += scale * (rng.random() - 0.5) * 2.0
        elif r < 0.65:
            h_pert = best_h.copy()
            thr = 0.5 + 0.4 * (rng.random() - 0.5)
            h_pert = np.where(h_pert > thr, h_pert + scale, h_pert - scale)
        else:
            h_pert = best_h.copy()
            dx_v = 2.0 / n_best
            c_vals = np.correlate(best_h, 1.0 - best_h, mode="full") * dx_v
            c_max_v = c_vals.max()
            active_idx = np.where(c_vals > c_max_v - 0.04)[0]
            for k_idx in active_idx[:10]:
                k_shift = k_idx - (n_best - 1)
                if k_shift == 0:
                    h_pert -= scale * 0.3 * (1.0 - 2.0 * best_h)
                elif k_shift > 0:
                    valid = np.arange(n_best - k_shift)
                    grad_part = np.zeros(n_best)
                    grad_part[valid] = 1.0 - best_h[valid + k_shift]
                    h_pert -= scale * 0.3 * grad_part
                else:
                    ks = -k_shift
                    valid = np.arange(ks, n_best)
                    grad_part = np.zeros(n_best)
                    grad_part[valid] = 1.0 - best_h[valid - ks]
                    h_pert -= scale * 0.3 * grad_part
        h_pert = project_feasible(h_pert, n_best)

        if i % 3 == 0 and rem > 45:
            k_act2 = find_active_lags(h_pert, n_best, threshold_frac=0.03)
            if len(k_act2) > 60:
                c_tmp = np.correlate(h_pert, 1.0 - h_pert, mode="full")
                k_act2 = np.argsort(c_tmp)[::-1][:60]
            h_new, c5_new = slsqp_optimize_active(h_pert, n_best, k_act2, maxiter=3500, t_margin=0.02)
        else:
            h_new, c5_new = slsqp_optimize_active(h_pert, n_best, find_active_lags(h_pert, n_best, 0.03), maxiter=2000, t_margin=0.02)

        if c5_new < best_c5 - 1e-12:
            best_h = h_new
            best_c5 = c5_new
        rem = time_rem(t0, budget_s)

    for n_try in [405, 540]:
        if rem < 100:
            break
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        f_interp = interp1d(x_old, best_h, kind='linear', fill_value='extrapolate')
        h_interp = np.clip(f_interp(x_new), 0, 1)
        h_interp = project_feasible(h_interp, n_try)
        k_act3 = find_active_lags(h_interp, n_try, threshold_frac=0.03)
        if len(k_act3) > 80:
            c_tmp2 = np.correlate(h_interp, 1.0 - h_interp, mode="full")
            k_act3 = np.argsort(c_tmp2)[::-1][:80]
        h_new, c5_new = slsqp_optimize_active(h_interp, n_try, k_act3, maxiter=5000, t_margin=0.05)
        if c5_new < best_c5 - 1e-10:
            best_h = h_new
            best_c5 = c5_new
            n_best = n_try
        rem = time_rem(t0, budget_s)

    if rem > 40:
        best_h, best_c5 = slsqp_optimize_full(best_h, n_best, maxiter=8000, t_margin=0.005)
        rem = time_rem(t0, budget_s)

    if rem > 20:
        for T_fin in [50, 150, 400]:
            if rem < 10:
                break
            h_new, c5_new = lse_optimize(best_h, n_best, T=T_fin, penalty=200.0, maxiter=1000)
            if c5_new < best_c5 - 1e-13:
                best_h = h_new
                best_c5 = c5_new
        rem = time_rem(t0, budget_s)

    if rem > 15:
        k_final = find_active_lags(best_h, n_best, threshold_frac=0.01)
        if len(k_final) > 60:
            c_fin = np.correlate(best_h, 1.0 - best_h, mode="full")
            k_final = np.argsort(c_fin)[::-1][:60]
        best_h, best_c5 = slsqp_optimize_active(best_h, n_best, k_final, maxiter=5000, t_margin=0.005)

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5_from_h(best_h)
    return best_h, best_c5, len(best_h)
