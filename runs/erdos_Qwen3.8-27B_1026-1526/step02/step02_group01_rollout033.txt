    def constr_jac(x):
        return np.ones(len(x))
    constraints = [{'type': 'eq', 'fun': constr_fun, 'jac': constr_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad_fn, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def nesterov_pg(h0, n, target_sum, alpha, n_iters, lr_init):
    h = h0.copy()
    y = h0.copy()
    t = 1.0
    lr = lr_init
    c5_cur = compute_c5(h)
    h_best = h.copy()
    stall = 0
    for it in range(n_iters):
        _, g = smooth_max_value_and_grad(y, alpha)
        h_new = np.clip(y - lr * g, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        mo = (t - 1.0) / t_new
        y = h_new + mo * (h_new - h)
        y = np.clip(y, 0.0, 1.0)
        h = h_new
        t = t_new
        c5_new = compute_c5(h)
        if c5_new < c5_cur - 1e-15:
            c5_cur = c5_new
            h_best = h.copy()
            stall = 0
            lr = min(lr * 1.01, 0.4)
        else:
            stall += 1
            if stall % 30 == 0:
                lr *= 0.75
    return h_best, c5_cur

def exact_subgradient_step(h, n, target_sum, lr, top_k=3):
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_indices = np.argsort(corr)[::-1][:top_k]
    gap = corr[top_indices[0]] - corr[top_indices[-1]]
    if gap < 1e-10:
        weights = np.ones(top_k) / top_k
    else:
        raw_w = corr[top_indices] - corr[top_indices[0]] + 1.0
        weights = raw_w / np.sum(raw_w)
    grad = np.zeros(n)
    for ki, k in enumerate(top_indices):
        s = k - (n - 1)
        if s >= 0:
            i_start, i_end = 0, n - s
        else:
            i_start, i_end = -s, n
        g1 = np.zeros(n)
        g1[i_start:i_end] = 1.0 - h[i_start + s:i_end + s]
        g2 = np.zeros(n)
        j_start, j_end = i_start + s, i_end + s
        g2[j_start:j_end] = h[j_start - s:j_end - s]
        grad += weights[ki] * (g1 - g2)
    h_new = np.clip(h - lr * grad, 0.0, 1.0)
    h_new = project_to_sum(h_new, target_sum)
    return h_new

def subgradient_method(h0, n, target_sum, n_iters, lr_init):
    h = h0.copy()
    c5_cur = compute_c5(h)
    h_best = h.copy()
    lr = lr_init
    for it in range(n_iters):
        h = exact_subgradient_step(h, n, target_sum, lr, top_k=3)
        c5_new = compute_c5(h)
        if c5_new < c5_cur - 1e-14:
            c5_cur = c5_new
            h_best = h.copy()
            lr = min(lr * 1.005, 0.3)
        elif it % 50 == 0:
            lr *= 0.8
    return h_best, c5_cur

def coordinate_descent(h0, n, target_sum, max_rounds, rng, steps=None):
    h = h0.copy()
    c5_cur = compute_c5(h)
    if steps is None:
        steps = [0.01, 0.005, 0.002]
    for rnd in range(max_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            best_val = c5_cur
            best_delta = 0.0
            for d in steps:
                for sign in (-1.0, 1.0):
                    dd = sign * d
                    nh = h[i] + dd
                    if nh < -1e-10 or nh > 1.0 + 1e-10:
                        continue
                    h[i] = nh
                    h = project_to_sum(h, target_sum)
                    c5_new = compute_c5(h)
                    if c5_new < best_val - 1e-14:
                        best_val = c5_new
                        best_delta = dd
                    h[i] = nh - dd
            if best_delta != 0.0:
                h[i] += best_delta
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True
        if not improved:
            break
    return h, c5_cur

def make_starts(n, best_h, best_n, rng, target_sum):
    starts = []
    if best_n == n and best_h is not None:
        starts.append(best_h.copy())
    else:
        starts.append(project_to_sum(upsample(best_h, n), target_sum))
    starts.append(np.full(n, 0.5))
    hr = rng.uniform(0.0, 1.0, n)
    starts.append(project_to_sum(hr, target_sum))
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    for freq in (2, 3, 4, 5, 7):
        for phase in (0.0, np.pi / 4.0, np.pi / 2.0):
            hs = 0.5 + 0.3 * np.sin(2.0 * np.pi * freq * x + phase)
            starts.append(project_to_sum(hs, target_sum))
    period = max(3, n // 24)
    h_blk = np.zeros(n)
    for s in range(0, n, period * 2):
        h_blk[s:s + period] = 1.0
    starts.append(project_to_sum(h_blk, target_sum))
    h_blk2 = np.zeros(n)
    period2 = max(2, n // 40)
    for s in range(0, n, period2 * 3):
        h_blk2[s:s + period2] = 1.0
    starts.append(project_to_sum(h_blk2, target_sum))
    return starts

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    def time_left():
        return budget_s - (time.time() - t_start)
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            ts = best_n / 2.0
            best_h = project_to_sum(best_h, ts)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass
    if best_h is None:
        best_n = 180
        best_h = np.full(180, 0.5)
        best_c5 = compute_c5(best_h)
    n_targets = [360, 480, 600, 720]
    for n in n_targets:
        if time_left() < 100:
            break
        if n % 2 != 0:
            continue
        ts = n / 2.0
        if abs(ts - round(ts)) > 1e-9:
            continue
        starts = make_starts(n, best_h, best_n, rng, ts)
        np Starts_done = False
        for h_start in starts:
            if time_left() < 60:
                break
            h_start = project_to_sum(h_start, ts)
            c5_s = compute_c5(h_start)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_start.copy()
                best_n = n
            h_cur = h_start.copy()
            alphas_l = [30.0, 100.0, 300.0, 800.0]
            for alpha in alphas_l:
                if time_left() < 40:
                    break
                maxiter = max(30, min(300, int(time_left() * 1.2)))
                h_opt = lbfgsb_optimize(h_cur, n, ts, alpha, 50.0, maxiter)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    best_n = n
                h_cur = h_opt
            if time_left() > 50:
                n_pg = min(2000, int(time_left() * 2))
                h_pg, c5_pg = nesterov_pg(h_cur, n, ts, 100.0, n_pg, 0.03)
                if c5_pg < best_c5:
                    best_c5 = c5_pg
                    best_h = h_pg.copy()
                    best_n = n
                h_cur = h_pg
            if time_left() > 40:
                n_sub = min(1000, int(time_left() * 2))
                h_sub, c5_sub = subgradient_method(h_cur, n, ts, n_sub, 0.02)
                if c5_sub < best_c5:
                    best_c5 = c5_sub
                    best_h = h_sub.copy()
                    best_n = n
                h_cur = h_sub
            if time_left() > 30 and n <= 480:
                h_cd, c5_cd = coordinate_descent(h_cur, n, ts, 20, rng)
                if c5_cd < best_c5:
                    best_c5 = c5_cd
                    best_h = h_cd.copy()
                    best_n = n
    if best_h is None:
        best_h = np.full(360, 0.5)
        best_n = 360
        best_c5 = compute_c5(best_h)
    n_f = best_n
    ts_f = n_f / 2.0
    if time_left() > 80:
        for alpha in [500.0, 1500.0, 3000.0]:
            if time_left() < 30:
                break
            h_pol = lbfgsb_optimize(best_h, n_f, ts_f, alpha, 100.0, 300)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()
        if time_left() > 40:
            h_sqp = slsqp_optimize(best_h, n_f, ts_f, 2000.0, 200)
            c5_sqp = compute_c5(h_sqp)
            if c5_sqp < best_c5:
                best_c5 = c5_sqp
                best_h = h_sqp.copy()
    if time_left() > 60:
        n_pg = min(5000, int(time_left() * 3))
        h_pg, c5_pg = nesterov_pg(best_h, n_f, ts_f, 500.0, n_pg, 0.01)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()
    if time_left() > 40:
        n_sub = min(3000, int(time_left() * 2))
        h_sub, c5_sub = subgradient_method(best_h, n_f, ts_f, n_sub, 0.01)
        if c5_sub < best_c5:
            best_c5 = c5_sub
            best_h = h_sub.copy()
    if time_left() > 30 and n_f <= 480:
        h_cd, c5_cd = coordinate_descent(best_h, n_f, ts_f, 30, rng, steps=[0.005, 0.002, 0.001])
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
    if time_left() > 20 and n_f <= 360:
        h_cd2, c5_cd2 = coordinate_descent(best_h, n_f, ts_f, 20, rng, steps=[0.001, 0.0005])
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2.copy()
    n_final = len(best_h)
    ts_final = n_final / 2.0
    best_h = project_to_sum(best_h, ts_final)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_final)
