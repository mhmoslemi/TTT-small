import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            step = diff / np.sum(mask)
            step = min(step, (1.0 - h[mask]).min() * 0.999)
            if step <= 0:
                break
            h[mask] += step
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            step = -diff / np.sum(mask)
            step = min(step, h[mask].min() * 0.999)
            if step <= 0:
                break
            h[mask] -= step
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, T, penalty):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full") * dx
    c_max = c.max()
    shifted = np.clip(T * (c - c_max), -500, 500)
    val = c_max + np.log(np.sum(np.exp(shifted))) / T
    sv = np.sum(x) - n / 2.0
    return val + penalty * sv * sv

def smooth_grad(x, T, penalty):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full") * dx
    c_max = c.max()
    shifted = np.clip(T * (c - c_max), -500, 500)
    w = np.exp(shifted - shifted.max())
    w /= w.sum()
    
    first_sum = np.correlate(1.0 - x, w, mode='full')[n-1:2*n-1]
    second_sum = -np.convolve(x[::-1], w, mode='full')[:n]
    
    grad = T * dx * (first_sum + second_sum)
    sv = np.sum(x) - n / 2.0
    grad += 2.0 * penalty * sv
    return grad

def lbfgsb_opt(h0, T, penalty, maxiter=5000):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    res = minimize(smooth_obj, h0, args=(T, penalty),
                   jac=lambda x: smooth_grad(x, T, penalty),
                   method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14, 'maxls': 50})
    h_out = project_feasible(res.x, n)
    return h_out, compute_c5(h_out)

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(270, 0.5)
    
    n = len(h0)
    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    n_best = n
    
    # Phase 1: Optimize at current resolution with increasing T
    for T in [20, 50, 100, 200, 400, 800]:
        if time.time() - t0 > budget_s - 100:
            break
        penalty = min(5000.0, 50.0 * T)
        h_new, c5_new = lbfgsb_opt(best_h, T, penalty, maxiter=3000)
        if c5_new < best_c5 - 1e-12:
            best_h = h_new
            best_c5 = c5_new
    
    # Phase 2: Structured perturbations + re-optimization
    n_perturb = 30
    for attempt in range(n_perturb):
        if time.time() - t0 > budget_s - 80:
            break
        frac = attempt / max(1, n_perturb)
        scale = 0.08 * (1.0 - frac) ** 0.5 + 0.005
        
        # Fourier perturbation
        x = np.arange(n_best)
        n_modes = rng.integers(2, 6)
        pert = np.zeros(n_best)
        for _ in range(n_modes):
            freq = rng.integers(1, max(2, n_best // 2))
            phase = rng.random() * 2 * np.pi
            pert += (scale / np.sqrt(n_modes)) * np.cos(2 * np.pi * freq * x / n_best + phase)
        
        h_pert = project_feasible(best_h + pert, n_best)
        T_cur = 100.0 + 200.0 * frac
        penalty_cur = 200.0 + 1000.0 * frac
        h_new, c5_new = lbfgsb_opt(h_pert, T_cur, penalty_cur, maxiter=2000)
        if c5_new < best_c5 - 1e-12:
            best_h = h_new
            best_c5 = c5_new
    
    # Phase 3: Try different resolutions
    for n_try in [200, 300, 350, 400]:
        if time.time() - t0 > budget_s - 60:
            break
        x_old = np.linspace(0, 2, n_best, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        f_interp = interp1d(x_old, best_h, kind='linear', fill_value="extrapolate", bounds_error=False)
        h_interp = np.clip(f_interp(x_new), 0, 1)
        h_interp = project_feasible(h_interp, n_try)
        
        for T in [50, 200, 500]:
            if time.time() - t0 > budget_s - 30:
                break
            penalty = min(5000.0, 50.0 * T)
            h_new, c5_new = lbfgsb_opt(h_interp, T, penalty, maxiter=3000)
            h_interp = h_new
            if c5_new < best_c5 - 1e-12:
                best_h = h_new
                best_c5 = c5_new
                n_best = n_try
