- j appears in (1-h[n]) when n = j: contribution -h[j+s] if 0<=j+s<n

So: d(c[i])/d(h[j]) = (1-h[j-s]) * 1_{0<=j-s<n} - h[j+s] * 1_{0<=j+s<n}

Let me verify: h = [a, b, c, d], s=0:
c = a*(1-a) + b*(1-b) + c*(1-c) + d*(1-d)
dc/da = (1-a) - a = 1-2a

Using formula: (1-h[0-0]) * 1 - h[0+0] * 1 = (1-a) - a = 1-2a ✓

s=-1 (i = -1+3 = 2 for n=4):
c = h[0+s]*(1-h[0]) + h[1+s]*(1-h[1]) + ...
Wait, n ranges where 0<=n<n and 0<=n+s<n. For s=-1: 0<=n-1<4, so 1<=n<=4, i.e., n=1,2,3.

d(overlap(s))/d(h[j]) = dx * [(1-h[j-s]) * 1_{0<=j-s<n} - h[j+s] * 1_{0<=j+s<n}]

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def compute_all_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return c

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            if not np.any(mask):
                break
            room = 1.0 - h[mask]
            step = min(diff / np.sum(mask), room.min())
            h[mask] += step
        else:
            mask = h > 1e-15
            if not np.any(mask):
                break
            room = h[mask]
            step = min(-diff / np.sum(mask), room.min())
            h[mask] -= step
    return np.clip(h, 0.0, 1.0)

def overlap_grad(h, s):
    """Gradient of overlap at shift s (s = i - (n-1)) with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    # Term 1: (1-h[j-s]) for valid j-s
    lo = max(0, s)
    hi = min(n, n + s)
    # j ranges from lo to hi-1, j-s ranges from lo-s to hi-1-s
    grad[lo:hi] += (1.0 - h[lo-s:hi-s])
    
    # Term 2: -h[j+s] for valid j+s
    lo2 = max(0, -s)
    hi2 = min(n, n - s)
    grad[lo2:hi2] -= h[lo2+s:hi2+s]
    
    return grad

def slsqp_optimize(h_init, maxiter=5000, t_margin=0.1):
    n = len(h_init)
    c5_init = compute_c5_from_h(h_init)
    x0 = np.concatenate([h_init, [c5_init + t_margin]])
    bounds = [(0.0, 1.0)] * n + [(max(0.001, c5_init * 0.5), 2.0)]
    
    def obj(x):
        return x[-1]
    
    def ineq(x):
        h = x[:-1]
        t = x[-1]
        c = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
        return t - c
    
    def eq(x):
        return np.sum(x[:-1]) - n / 2.0
    
    def eq_jac(x):
        j = np.zeros(len(x))
        j[:n] = 1.0
        return j
    
    constraints = [
        {'type': 'ineq', 'fun': ineq},
        {'type': 'eq', 'fun': eq, 'jac': eq_jac}
    ]
    
    result = minimize(obj, x0, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
    
    if result.x is not None and result.success or (result.x is not None and compute_c5_from_h(project_feasible(result.x[:-1], n)) <= c5_init + 0.01):
        h_out = project_feasible(result.x[:-1], n)
        return h_out, compute_c5_from_h(h_out)
    return h_init, c5_init

def upscale(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    if n_new > n_old:
        f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    else:
        f = interp1d(x_old, h, kind='linear', fill_value='extrapolate')
    h_new = np.clip(f(x_new), 0, 1)
    return project_feasible(h_new, n_new)

def gradient_perturb(h, n, scale, top_k=5):
    """Generate a perturbation that reduces the top-k active overlaps."""
    overlaps = compute_all_overlaps(h)
    c_max = overlaps.max()
    # Find top k active lags
    active_idx = np.where(overlaps > c_max - 0.01)[0]
    if len(active_idx) == 0:
        active_idx = np.array([np.argmax(overlaps)])
    elif len(active_idx) > top_k * 3:
        # Take top top_k*3 by value
        top_positions = np.argsort(overlaps)[-top_k*3:]
        active_idx = top_positions
    
    # Compute weighted gradient (negative, to decrease overlaps)
    grad = np.zeros(n)
    for idx in active_idx:
        s = idx - (n - 1)
        w = overlaps[idx] / overlaps[active_idx].sum()
        grad += w * overlap_grad(h, s)
    
    # Normalize
    grad_norm = np.linalg.norm(grad)
    if grad_norm > 0:
        grad = grad / grad_norm
    
    return -grad * scale  # Negative gradient direction

def bimodal_push(h, strength):
    """Push h values toward 0 or 1, maintaining integral."""
    n = len(h)
    h_new = h.copy()
    # For values above 0.5, push toward 1; below 0.5, push toward 0
    above = h_new > 0.5
    below = h_new < 0.5
    
    h_new[above] = np.minimum(1.0, h_new[above] + strength * (1.0 - h_new[above]))
    h_new[below] = np.maximum(0.0, h_new[below] - strength * h_new[below])
    
    return project_feasible(h_new, n)

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    
    # Get initial
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(270, 0.5)
    
    n = len(h0)
    h = project_feasible(h0, n)
    best_h = h.copy()
    best_c5 = compute_c5_from_h(best_h)
    best_n = n
    
    def remaining():
        return budget_s - (time.time() - t0)
    
    # Phase 1: SLSQP at initial resolution
    if remaining() > 200:
        h, c5 = slsqp_optimize(h, maxiter=8000, t_margin=0.1)
        if c5 < best_c5:
            best_h, best_c5, best_n = h, c5, n
    
    # Phase 2: Upscale to 2x resolution
    if remaining() > 300:
        n2 = best_n * 2
        h_up = upscale(best_h, n2)
        h_up, c5_up = slsqp_optimize(h_up, maxiter=6000, t_margin=0.08)
        if c5_up < best_c5:
            best_h, best_c5, best_n = h_up, c5_up, n2
    
    # Phase 3: Gradient-informed perturbations + local SLSQP
    n_pert = 30
    if remaining() > 150:
        for attempt in range(n_pert):
            if remaining() < 60:
                break
            
            frac = attempt / max(1, n_pert)
            scale = 0.08 * (1.0 - frac) + 0.01
            
            # Choose perturbation type
            r = rng.random()
            if r < 0.4:
                # Gradient-informed perturbation
                h_pert = best_h + gradient_perturb(best_h, best_n, scale, top_k=5)
            elif r < 0.6:
                # Bimodal push
                h_pert = bimodal_push(best_h, strength=0.1 * (1-frac) + 0.02)
            elif r < 0.8:
                # Random Gaussian
                h_pert = best_h + scale * 0.5 * rng.standard_normal(best_n)
            else:
                # Localized perturbation
                h_pert = best_h.copy()
                bw = max(5, best_n // 20)
                center = rng.integers(0, best_n)
                idx = np.arange(max(0, center-bw), min(best_n, center+bw))
                h_pert[idx] += scale * rng.standard_normal(len(idx))
            
            h_pert = project_feasible(h_pert, best_n)
            
            # Quick SLSQP
            h_new, c5_new = slsqp_optimize(h_pert, maxiter=2000, t_margin=0.03)
            if c5_new < best_c5 - 1e-12:
                best_h, best_c5 = h_new, c5_new
    
    # Phase 4: Try another resolution
    if remaining() > 200:
        # Try 1.5x the current best (if not already 2x)
        n3 = int(best_n * 1.5)
        if n3 != best_n:
            h_up = upscale(best_h, n3)
            h_up, c5_up = slsqp_optimize(h_up, maxiter=5000, t_margin=0.05)
            if c5_up < best_c5:
                best_h, best_c5, best_n = h_up, c5_up, n3
    
    # Phase 5: Final polish with high-iteration SLSQP
    if remaining() > 50:
        h, c5 = slsqp_optimize(best_h, maxiter=15000, t_margin=0.02)
        if c5 < best_c5:
            best_h, best_c5 = h, c5
    
    # Phase 6: Tiny perturbations for final improvement
    if remaining() > 20:
        for _ in range(20):
            if remaining() < 8:
                break
            h_pert = best_h + 0.003 * rng.standard_normal(best_n)
            h_pert = project_feasible(h_pert, best_n)
            h_new, c5_new = slsqp_optimize(h_pert, maxiter=3000, t_margin=0.01)
            if c5_new < best_c5 - 1e-12:
                best_h, best_c5 = h_new, c5_new
    
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5_from_h(best_h)
    
    return best_h, best_c5, len(best_h)

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def compute_all_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            if not np.any(mask):
                break
            room = 1.0 - h[mask]
            if room.min() <= 0:
                break
            step = min(diff / np.sum(mask), room.min() * 0.9999)
            h[mask] += step
        else:
            mask = h > 1e-12
            if not np.any(mask):
                break
            room = h[mask]
            if room.min() <= 0:
                break
            step = min(-diff / np.sum(mask), room.min() * 0.9999)
            h[mask] -= step
    return np.clip(h, 0.0, 1.0)

def overlap_grad_nodx(h, s):
    """d(correlate at shift s)/d(h) without the dx factor."""
    n = len(h)
    grad = np.zeros(n)
    # (1-h[j-s]) for valid j-s in [0, n-1]
    lo1 = max(0, s)
    hi1 = min(n, n + s)
    if hi1 > lo1:
        grad[lo1:hi1] += (1.0 - h[lo1-s:hi1-s])
    # -h[j+s] for valid j+s in [0, n-1]
    lo2 = max(0, -s)
    hi2 = min(n, n - s)
    if hi2 > lo2:
        grad[lo2:hi2] -= h[lo2+s:hi2+s]
    return grad

def soft_max_and_grad(h, T, penalty):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    c_max = c.max()
    shifted = T * (c - c_max)
    shifted = np.clip(shifted, -500, 500)
    exp_s = np.exp(shifted)
    log_sum = np.log(exp_s.sum())
    obj = c_max + log_sum / T
    
    # Add constraint violation
    sum_v = np.sum(h) - n / 2.0
    obj += penalty * sum_v ** 2
    
    # Gradient
    weights = exp_s / exp_s.sum()
    grad = np.zeros(n)
    # Only compute for significant weights
    sig = weights > 1e-8
    if np.any(sig):
        for idx in np.where(sig)[0]:
            s = idx - (n - 1)
            grad += weights[idx] * overlap_grad_nodx(h, s)
        grad *= dx
    grad += 2 * penalty * sum_v
    
    return obj, grad

def lbfgsb_optimize(h_init, T, penalty, maxiter=3000):
    n = len(h_init)
    bounds = [(0.0, 1.0)] * n
    
    def obj_only(x):
        return soft_max_and_grad(x, T, penalty)[0]
    
    def grad_only(x):
        return soft_max_and_grad(x, T, penalty)[1]
    
    result = minimize(obj_only, h_init, jac=grad_only, method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-16})
    
    if result.x is not None:
        h_out = project_feasible(result.x, n)
        return h_out, compute_c5_from_h(h_out)
    return h_init, compute_c5_from_h(h_init)

        c5_out = compute_c5_from_h(h_out)
        if c5_out <= c5_init + 0.01:

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time


def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def compute_all_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx


def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            if not np.any(mask):
                break
            room = 1.0 - h[mask]
            if room.min() <= 0:
                break
            step = min(diff / np.sum(mask), room.min() * 0.9999)
            h[mask] += step
        else:
            mask = h > 1e-14
            if not np.any(mask):
                break
            room = h[mask]
            if room.min() <= 0:
                break
            step = min(-diff / np.sum(mask), room.min() * 0.9999)
            h[mask] -= step
    return np.clip(h, 0.0, 1.0)


def overlap_grad_nodx(h, s):
    n = len(h)
    grad = np.zeros(n)
    lo1 = max(0, s)
    hi1 = min(n, n + s)
    if hi1 > lo1:
        grad[lo1:hi1] += (1.0 - h[lo1 - s:hi1 - s])
    lo2 = max(0, -s)
    hi2 = min(n, n - s)
    if hi2 > lo2:
        grad[lo2:hi2] -= h[lo2 + s:hi2 + s]
    return grad


def soft_max_and_grad(h, T, penalty):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    c_max = c.max()
    shifted = T * (c - c_max)
    shifted = np.clip(shifted, -500, 500)
    exp_s = np.exp(shifted)
    log_sum = np.log(exp_s.sum())
    obj = c_max + log_sum / T
    sum_v = np.sum(h) - n / 2.0
    obj += penalty * sum_v ** 2
    weights = exp_s / exp_s.sum()
    grad = np.zeros(n)
    sig = weights > 1e-10
    if np.any(sig):
        idxs = np.where(sig)[0]
        for idx in idxs:
            s = idx - (n - 1)
            grad += weights[idx] * overlap_grad_nodx(h, s)
        grad *= dx
    grad += 2.0 * penalty * sum_v
    return obj, grad


def lbfgsb_optimize(h_init, T, penalty, maxiter=3000):
    n = len(h_init)
    bounds = [(0.0, 1.0)] * n

    def obj_only(x):
        return soft_max_and_grad(x, T, penalty)[0]

    def grad_only(x):
        return soft_max_and_grad(x, T, penalty)[1]

    result = minimize(obj_only, h_init, jac=grad_only, method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14})
    h_out = project_feasible(result.x, n)
    return h_out, compute_c5_from_h(h_out)


def slsqp_obj(x):
    return x[-1]


def slsqp_ineq(x, n):
    h = x[:-1]
    t = x[-1]
    c = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
    return t - c


def slsqp_eq(x, n):
    return np.sum(x[:-1]) - n / 2.0


def slsqp_eq_jac(x, n):
    j = np.zeros(n + 1)
    j[:n] = 1.0
    return j


def slsqp_optimize(h_init, maxiter=5000, t_margin=0.1):
    n = len(h_init)
    c5_init = compute_c5_from_h(h_init)
    x0 = np.concatenate([h_init, [c5_init + t_margin]])
    bounds = [(0.0, 1.0)] * n + [(max(0.001, c5_init * 0.3), 2.0)]

    def ineq(x):
        return slsqp_ineq(x, n)

    def eq(x):
        return slsqp_eq(x, n)

    def eq_jac(x):
        return slsqp_eq_jac(x, n)

    constraints = [
        {'type': 'ineq', 'fun': ineq},
        {'type': 'eq', 'fun': eq, 'jac': eq_jac}
    ]

    result = minimize(slsqp_obj, x0, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
    if result.x is not None:
        h_out = project_feasible(result.x[:-1], n)
        c5_out = compute_c5_from_h(h_out)
        if c5_out <= c5_init + 0.05:
            return h_out, c5_out
    return h_init, c5_init


def upscale(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = f(x_new)
    return project_feasible(np.clip(h_new, 0, 1), n_new)


def downscale(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = f(x_new)
    return project_feasible(np.clip(h_new, 0, 1), n_new)


def gradient_informed_perturb(h, n, scale, top_k=5):
    overlaps = compute_all_overlaps(h)
    top_idx = np.argsort(overlaps)[-top_k:]
    grad = np.zeros(n)
    for idx in top_idx:
        s = idx - (n - 1)
        grad += overlap_grad_nodx(h, s)
    grad_norm = np.linalg.norm(grad)
    if grad_norm > 1e-15:
        return h - scale * grad / grad_norm
    return h
