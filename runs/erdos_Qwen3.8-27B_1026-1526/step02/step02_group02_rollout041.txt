from scipy.interpolate import interp1d

def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_feasible(h, n):
    # Project onto {h in [0,1]^n : sum(h) = n/2}
    target = n / 2.0
    h = np.clip(np.array(h, dtype=float), 0.0, 1.0)
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
        else:
            mask = h > 1e-10
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
    return np.clip(h, 0.0, 1.0)

def smoothed_obj_and_grad(h, T, penalty):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = c.max()
    shifted = np.clip(T * (c - c_max), -500, 500)
    exp_shifted = np.exp(shifted)
    Z = exp_shifted.sum()
    log_sum_exp = c_max + np.log(Z) / T
    weights = exp_shifted / Z
    
    sum_viol = np.sum(h) - n / 2.0
    obj = log_sum_exp + 0.5 * penalty * sum_viol ** 2
    
    # Gradient
    lags = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)
    # Only include significant weights
    threshold = 1e-12 * Z  # relative to max weight
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-14:
            continue
        lag = lags[m]
        # Term 1: (1-h[j+lag])
        j1s = max(0, -lag)
        j1e = min(n, n - lag)
        if j1s < j1e:

        if j2s < j2e:
            grad[j2s:j2e] -= w * h[j2s - lag:j2e - lag]
    grad *= dx
    
    # Add penalty gradient
    grad += penalty * sum_viol
    
    return obj, grad

            return (np.inf, np.zeros(n))

        return val, grd
    
    result = minimize(fun, h_init, method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14, 'maxfun': 50000})
    h_out = project_feasible(result.x, n)
    return h_out, compute_c5_from_h(h_out)

from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time


def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-12
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def smoothed_obj_grad(h, T, penalty):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = c.max()
    shifted = np.clip(T * (c - c_max), -500, 500)
    exp_shifted = np.exp(shifted)
    Z = exp_shifted.sum()
    log_sum_exp = c_max + np.log(Z) / T
    weights = exp_shifted / Z

    sum_viol = np.sum(h) - n / 2.0
    obj = log_sum_exp + 0.5 * penalty * sum_viol * sum_viol

    lags = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-14:

        j1s = max(0, -lag)
        j1e = min(n, n - lag)
        if j1s < j1e:
            grad[j1s:j1e] += w * (1.0 - h[j1s + lag:j1e + lag])
        j2s = max(0, lag)
        j2e = min(n, n + lag)
        if j2s < j2e:
            grad[j2s:j2e] -= w * h[j2s - lag:j2e - lag]
    grad *= dx
    grad += penalty * sum_viol

    bounds = [(0.0, 1.0)] * n
    result = minimize(
        smoothed_obj_grad, h_init, args=(T, penalty),

        options={'maxiter': maxiter, 'ftol': ftol, 'gtol': gtol, 'maxfun': maxiter * 3}
    )
    h_out = project_feasible(result.x, n)
    return h_out, compute_c5_from_h(h_out)


def projected_grad_step(h, T, penalty, alpha):
    n = len(h)
    _, g = smoothed_obj_grad(h, T, penalty)
    y = h - alpha * g
    return project_feasible(y, n)


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(270, 0.5)
        idx = rng.choice(270, 135, replace=False)
        h0[idx] = 0.7
        h0[~np.isin(np.arange(270), idx)] = 0.3

    n = len(h0)
    best_h = project_feasible(h0, n)
    best_c5 = compute_c5_from_h(best_h)

    remaining = budget_s - (time.time() - t0)
    if remaining < 30:
        return best_h, best_c5, n

    T_schedule = [10, 20, 40, 70, 120, 200, 350, 600]
    for T in T_schedule:
        remaining = budget_s - (time.time() - t0)
        if remaining < 80:
            break
        pen = 30 + 0.8 * T
        h_new, c5_new = lbfgs_call(best_h, T, pen, maxiter=3000, ftol=1e-18, gtol=1e-15)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    remaining = budget_s - (time.time() - t0)
    if remaining < 60:
        best_h = project_feasible(best_h, n)
        best_c5 = compute_c5_from_h(best_h)
        return best_h, best_c5, n

    n_perturb = max(5, int(remaining * 0.35 / 8))
    n_perturb = min(n_perturb, 60)
    for idx_p in range(n_perturb):
        remaining = budget_s - (time.time() - t0)
        if remaining < 60:
            break
        frac = idx_p / max(1, n_perturb)
        scale = 0.12 * (1.0 - frac) ** 1.4 + 0.004
        r = rng.random()
        if r < 0.25:
            h_pert = best_h + scale * rng.standard_normal(n)
        elif r < 0.45:
            h_pert = best_h.copy()
            n_modes = int(rng.integers(1, 4))
            for _ in range(n_modes):
                freq = int(rng.integers(1, max(2, n // 3)))
                phase = rng.random() * 2 * np.pi
                amp = scale / np.sqrt(n_modes)
                h_pert += amp * np.cos(2 * np.pi * freq * np.arange(n) / n + phase)
        elif r < 0.6:
            h_pert = best_h.copy()
            n_blocks = int(rng.integers(2, max(3, n // 15)))
            for _ in range(n_blocks):
                bsize = int(rng.integers(4, max(5, n // 6)))
                start = int(rng.integers(0, max(1, n - bsize)))
                h_pert[start:start + bsize] += scale * (rng.random(bsize) - 0.5) * 3.0
        elif r < 0.8:
            h_pert = best_h.copy()
            n_push = max(5, int(n * 0.12))
            push_idx = rng.choice(n, n_push, replace=False)
            for idx in push_idx:
                if rng.random() < 0.5:
                    h_pert[idx] = min(1.0, h_pert[idx] + 0.4)
                else:
                    h_pert[idx] = max(0.0, h_pert[idx] - 0.4)
        else:
            h_pert = best_h.copy()
            swap_n = int(rng.integers(max(3, n // 30), max(4, n // 10)))
            above = np.where(h_pert > 0.55)[0]
            below = np.where(h_pert < 0.45)[0]
            if len(above) >= swap_n and len(below) >= swap_n:
                ia = rng.choice(above, swap_n, replace=False)
                ib = rng.choice(below, swap_n, replace=False)
                for k in range(swap_n):
                    old_a = h_pert[ia[k]]
                    old_b = h_pert[ib[k]]
                    h_pert[ia[k]] = old_b
                    h_pert[ib[k]] = old_a

        h_pert = project_feasible(h_pert, n)
        T_cur = 30.0 + 250.0 * frac
        pen_cur = 50.0 + 300.0 * frac
        h_new, c5_new = lbfgs_call(h_pert, T_cur, pen_cur, maxiter=2500, ftol=1e-17, gtol=1e-14)
        if c5_new < best_c5 - 1e-13:
            best_h = h_new
            best_c5 = c5_new

    remaining = budget_s - (time.time() - t0)
    if remaining > 200:
        n_coarse = max(50, n // 3)
        x_fine = np.linspace(0, 2, n, endpoint=False)
        x_coarse = np.linspace(0, 2, n_coarse, endpoint=False
            )
        h_coarse = np.array([best_h[int(i * n / n_coarse):int((i + 1) * n / n_coarse)].mean()
                             for i in range(n_coarse)])
        h_coarse = project_feasible(h_coarse, n_coarse)
        h_coarse, _ = lbfgs_call(h_coarse, 50, 100, maxiter=2000, ftol=1e-16, gtol=1e-13)
        h_up = np.interp(x_fine, x_coarse, h_coarse)
        h_up = project_feasible(h_up, n)
        h_new, c5_new = lbfgs_call(h_up, 50, 100, maxiter=2000, ftol=1e-16, gtol=1e-13)
        if c5_new < best_c5 - 1e-13:
            best_h = h_new
            best_c5 = c5_new

    remaining = budget_s - (time.time() - t0)
    if remaining > 150:
        n_high = min(n + 90, 540)
        if n_high != n:
            x_old = np.linspace(0, 2, n, endpoint=False)
            x_new_grid = np.linspace(0, 2, n_high, endpoint=False)
            f_interp = interp1d(x_old, best_h, kind='linear', fill_value='extrapolate')
            h_high = np.clip(f_interp(x_new_grid), 0, 1)
            h_high = project_feasible(h_high, n_high)
            h_high, c5_h = lbfgs_call(h_high, 50, 100, maxiter=3000, ftol=1e-17, gtol=1e-14)
            if c5_h < best_c5 - 1e-13:
                best_h = h_high
                best_c5 = c5_h
                n = n_high

    remaining = budget_s - (time.time() - t0)
    if remaining > 80:
        for T_fine in [400, 800, 1500]:
            remaining = budget_s - (time.time() - t0)
            if remaining < 30:
                break
            pen_fine = max(200, 5.0 * T_fine)
            h_new, c5_new = lbfgs_call(best_h, T_fine, pen_fine, maxiter=4000, ftol=1e-18, gtol=1e-15)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    remaining = budget_s - (time.time() - t0)
    if remaining > 30:
        n_final_polish = max(10, int(remaining // 4))
        for _ in range(n_final_polish):
            if time.time() - t0 > budget_s - 5:
                break
            scale = 0.002 + 0.01 * (1.0 - _ / max(1, n_final_polish)) ** 2
            h_pert = best_h + scale * rng.standard_normal(n)
            h_pert = project_feasible(h_pert, n)
            h_new, c5_new = lbfgs_call(h_pert, 200, 300, maxiter=1000, ftol=1e-17, gtol=1e-14)
            if c5_new < best_c5 - 1e-14:
                best_h = h_new
                best_c5 = c5_new

    remaining = budget_s - (time.time() - t0)
    if remaining > 15:
        h_new, c5_new = lbfgs_call(best_h, 1000, 1000, maxiter=5000, ftol=1e-18, gtol=1e-16)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5_from_h(best_h)
    return best_h, best_c5, len(best_h)
