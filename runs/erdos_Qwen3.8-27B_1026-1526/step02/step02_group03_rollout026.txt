    block_size = n // num_blocks
    for b in range(num_blocks):
        start = b * block_size
        end = start + block_size if b < num_blocks - 1 else n
        val = rng.uniform(0.1, 0.9)
        h[start:end] = val
        if b + 1 < num_blocks:
            h[start+1:end] = rng.uniform(0.1, 0.9)
    return project_feasible(h, n)

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    
    parent_h = None
    try:
        parent_h = np.array(initial_h_values, dtype=float)
    except Exception:
        pass
    
    def elapsed():
        return time.time() - start_time
    
    def remaining_frac():
        return max(0.0, 1.0 - elapsed() / budget_s)
    
    # Phase 1: Broad SLSQP search (30% budget)
    phase1_end = start_time + budget_s * 0.28
    
    n_candidates = []
    if parent_h is not None:
        n_candidates.append(len(parent_h))
    for nn in [47, 53, 63, 67, 71, 77, 81, 83, 89]:
        if nn not in n_candidates:
            n_candidates.append(nn)
    
    for n in n_candidates:
        if time.time() > phase1_end:
            break
        starts = []
        
        if parent_h is not None:
            if len(parent_h) == n:
                starts.append(parent_h.copy())
                starts.append(project_feasible(parent_h + rng.randn(n) * 0.05, n))
            else:
                xp = np.linspace(0, 2, len(parent_h), endpoint=False)
                f_interp = interp1d(xp, parent_h, kind='cubic', fill_value='extrapolate', bounds_error=False)
                x_new = np.linspace(0, 2, n, endpoint=False)
                starts.append(project_feasible(f_interp(x_new), n))
                starts.append(project_feasible(f_interp(x_new) + rng.randn(n) * 0.06, n))
        
        for _ in range(3):
            freq = rng.uniform(0.8, 4.0)
            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.3, 0.48)
            starts.append(generate_sin_start(n, freq, phase, amp, rng))
        
        for _ in range(2):
            starts.append(generate_multi_sin_start(n, rng))
        
        starts.append(generate_piecewise_start(n, rng))
        starts.append(project_feasible(rng.rand(n), n))
        
        for h0 in starts:
            if time.time() > phase1_end:
                break
            tau_sched = [0.15, 0.08, 0.04, 0.02, 0.01, 0.005]
            h_opt, c5 = run_slsqp(h0, n, tau_sched, 40)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()
    
    if best_h is None:
        if parent_h is not None:
            best_h = parent_h.copy()
            best_c5 = evaluate_c5(best_h)
        else:
            n = 77
            best_h = project_feasible(np.full(n, 0.5), n)
            best_c5 = evaluate_c5(best_h)
    
    # Phase 2: Exact coordinate descent (35% budget)
    phase2_end = start_time + budget_s * 0.63
    n_best = len(best_h)
    
    h_cd, c5_cd = exact_coord_descent(best_h, n_best, max_sweeps=30)
    if c5_cd < best_c5:
        best_c5 = c5_cd
        best_h = h_cd.copy()
    
    # Phase 3: Perturbation + escape + refine (25% budget)
    phase3_end = start_time + budget_s * 0.88
    n_best = len(best_h)
    
    attempt = 0
    while time.time() < phase3_end:
        attempt += 1
        frac = elapsed() / budget_s
        noise_scale = 0.003 + 0.03 * max(0.0, 1.0 - frac * 2.0)
        
        perturbed = best_h + rng.randn(n_best) * noise_scale
        perturbed = project_feasible(perturbed, n_best)
        
        tau_sched = [0.008, 0.004, 0.002]
        h_esc, c5_esc = run_slsqp(perturbed, n_best, tau_sched, 50)
        
        h_cd, c5_cd = exact_coord_descent(h_esc, n_best, max_sweeps=15)
        
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            attempt = 0
    
    # Phase 4: Fine SLSQP + final exact CD (7% budget)
    phase4_end = start_time + budget_s * 0.97
    n_best = len(best_h)
    
    while time.time() < phase4_end:
        tau_sched = [0.003, 0.001]
        h_ref, c5_ref = run_slsqp(best_h, n_best, tau_sched, 80)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref.copy()
        
        h_cd, c5_cd = exact_coord_descent(best_h, n_best, max_sweeps=20)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
        
        if time.time() > phase4_end - 5:
            break
    
    # Final
    if best_h is not None:
        n_best = len(best_h)
        h_cd, c5_cd = exact_coord_descent(best_h, n_best, max_sweeps=10)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
        final_c5 = evaluate_c5(best_h)
        if final_c5 < best_c5:
            best_c5 = final_c5
    
    return best_h, best_c5, len(best_h)
