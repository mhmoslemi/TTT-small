import time

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):

        if abs(diff) < 1e-14:

        h = np.clip(h, 0.0, 1.0)
    return h

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def get_gradient(h, w):
    """Compute gradient of sum_k w[k]*corr[k] w.r.t. h."""
    n = len(h)
    grad = np.zeros(n)
    nz = np.where(w > 1e-15)[0]
    for k in nz:
        d = k - (n - 1)
        i_start = max(0, -d)
        i_end = min(n, n - d)
        if i_end > i_start:
            idx_f = np.arange(i_start, i_end)
            idx_s = idx_f + d
            wf = w[k]
            grad[idx_f] += wf * (1.0 - h[idx_s])
            grad[idx_s] += wf * (-h[idx_f])
    return grad

def smooth_objective(h, alpha, topk=200):
    """Smoothed max objective and gradient."""
    n = len(h)
    corr = get_corr(h)
    nl = len(corr)
    if topk < nl:
        idx = np.argpartition(corr, -topk)[-topk:]
    else:
        idx = np.arange(nl)
    corr_sub = corr[idx]
    m_val = np.max(corr_sub)
    shifted = alpha * (corr_sub - m_val)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w_sub = exp_vals / s
    obj = (m_val + np.log(s) / alpha)  # this is the smooth max value
    # Construct full weight vector
    w = np.zeros(nl)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def smooth_obj_and_grad(h, alpha, topk=None):
    n = len(h)
    corr = get_corr(h)
    nl = len(corr)
    if topk is not None and topk < nl:
        idx = np.argpartition(corr, -topk)[-topk:]
    else:
        idx = np.arange(nl)
    corr_sub = corr[idx]
    m = np.max(corr_sub)
    x = alpha * (corr_sub - m)
    x = np.clip(x, -50.0, 50.0)
    e = np.exp(x)
    s = np.sum(e)
    w_sub = e / s
    obj = m + np.log(s) / alpha
    w = np.zeros(nl)
    w[idx] = w_sub
    grad = np.zeros(n)
    nz = np.where(w > 1e-14)[0]
    for ki in nz:
        d = ki - (n - 1)
        i_lo = max(0, -d)
        i_hi = min(n, n - d)
        if i_hi > i_lo:
            is_arr = np.arange(i_lo, i_hi)
            js_arr = is_arr + d
            wk = w[ki]
            grad[is_arr] += wk * (1.0 - h[js_arr])
            grad[js_arr] += wk * (-h[is_arr])
    return obj, grad

def adam_projected_step(h, grad, m_buf, v_buf, t, lr, b1, b2, eps, target_sum):
    m_buf[:] = b1 * m_buf + (1.0 - b1) * grad
    v_buf[:] = b2 * v_buf + (1.0 - b2) * (grad * grad)
    m_hat = m_buf / (1.0 - b1 ** t)
    v_hat = v_buf / (1.0 - b2 ** t)
    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    h_new = h + step
    h_new = project_to_sum(h_new, target_sum)
    return h_new, m_buf, v_buf

def slsqp_optimize(h0, target_sum, alpha, topk, maxiter):
    n = len(h0)
    def obj_fn(x):
        o, _ = smooth_obj_and_grad(x, alpha, topk)
        return o
    def grad_fn(x):
        _, g = smooth_obj_and_grad(x, alpha, topk)
        return g
    def con_fn(x):
        return np.sum(x) - target_sum
    def con_jac_fn(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': con_fn, 'jac': con_jac_fn}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_fn, h0.copy(), method='SLSQP', jac=grad_fn,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def random_swap_improve(h, target_sum, rng, n_rounds, time_fn, deadline):
    h = h.copy()
    c5_cur = compute_c5(h)
    for _ in range(n_rounds):
        if time_fn() > deadline:
            break
        improved = False
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        n_trials = min(300, max(50, len(ones) * len(zeros) // 2))
        for _ in range(n_trials):
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h_test = h.copy()
            h_test[i] = 1.0
            h_test[j] = 0.0
            c5_t = compute_c5(h_test)
            if c5_t < c5_cur - 1e-12:
                h = h_test
                c5_cur = c5_t
                improved = True
                break
        if not improved:
            break
    return h, c5_cur

def block_swap_improve(h, target_sum, rng, n_rounds, time_fn, deadline):
    h = h.copy()
    n = len(h)
    c5_cur = compute_c5(h)
    block_lengths = [1, 2, 3, 4, 5, 8, 10, 15, 20, 30]
    for _ in range(n_rounds):
        if time_fn() > deadline:
            break
        improved = False
        for bl in block_lengths:
            if bl >= n // 2:
                continue
            n_pos = n - bl
            if n_pos <= 1:
                continue
            for _ in range(30):
                pos_a = rng.randint(0, n_pos)
                pos_b = rng.randint(0, n_pos)
                if abs(pos_a - pos_b) < bl * 2:
                    continue
                block_a = h[pos_a:pos_a + bl].copy()
                block_b = h[pos_b:pos_b + bl].copy()
                if np.allclose(block_a, block_b):
                    continue
                h_test = h.copy()
                h_test[pos_a:pos_a + bl] = block_b
                h_test[pos_b:pos_b + bl] = block_a
                c5_t = compute_c5(h_test)
                if c5_t < c5_cur - 1e-12:
                    h = h_test
                    c5_cur = c5_t
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return h, c5_cur

def gradient_descent_run(h0, target_sum, n_iters, lr, alpha_start, alpha_end,
                         topk, rng, time_fn, deadline):
    h = h0.copy()
    n = len(h)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(n_iters):
        if time_fn() > deadline:
            break
        frac = float(it) / max(1, n_iters)
        alpha = alpha_start * (alpha_end / alpha_start) ** (frac * 0.7)
        obj, grad = smooth_obj_and_grad(h, alpha, topk)
        h, m_buf, v_buf = adam_projected_step(
            h, grad, m_buf, v_buf, it + 1, lr, 0.9, 0.999, 1e-8, target_sum)
        if it % 50 == 49:
            n_swaps = rng.randint(1, 5)
            for _ in range(n_swaps):
                ones = np.where(h > 0.5)[0]
                zeros = np.where(h < 0.5)[0]
                if len(ones) > 0 and len(zeros) > 0:
                    i = rng.choice(ones)
                    j = rng.choice(zeros)
                    h[i] = 1.0
                    h[j] = 0.0
                    h = project_to_sum(h, target_sum)
        c5_now = compute_c5(h)
        if c5_now < best_c5:
            best_c5 = c5_now
            best_h = h.copy()
    return best_h, best_c5

def interpolate_to_n(h, n_new, period):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0, period, n_old, endpoint=False)
    x_new = np.linspace(0, period, n_new, endpoint=False)
    return np.interp(x_new, x_old, h, period=period)

def make_diverse_starts(h_parent, n, target_sum, rng, n_starts):
    starts = []
    starts.append(h_parent.copy())
    for k in range(1, n_starts):
        noise_level = 0.1 + 0.3 * (k / n_starts)
        h_pert = h_parent + rng.uniform(-noise_level, noise_level, n)
        h_pert = project_to_sum(h_pert, target_sum)
        starts.append(h_pert)
    x = np.linspace(0, 2.0, n, endpoint=False)
    for freq in [1, 2, 3, 4, 5, 7, 11]:
        amp = rng.uniform(0.15, 0.4)
        phase = rng.uniform(0, 2 * np.pi)
        h_s = 0.5 + amp * np.sin(2.0 * np.pi * freq * x / 2.0 + phase)
        h_s = project_to_sum(h_s, target_sum)
        starts.append(h_s)
    nblks = rng.choice([4, 6, 8, 10, 12])
    h_blk = np.zeros(n)
    for i in range(0, n, 2 * (n // nblks)):
        end = min(i + n // nblks, n)
        h_blk[i:end] = 1.0
    h_blk = project_to_sum(h_blk, target_sum)
    starts.append(h_blk)
    return starts

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()

    def elapsed():
        return time.time() - t_start

    def time_left():
        return budget_s - elapsed()

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            if best_n % 2 == 0:
                target = best_n // 2
                best_h = project_to_sum(best_h, target)
                best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    ts = best_n // 2
    deadline = t_start + budget_s

    # Phase 1: Direct gradient optimization from parent (fast Adam iterations)
    if time_left() > 300:
        n_it = 5000
        h_r, c5_r = gradient_descent_run(
            best_h, ts, n_it, 0.02, 50, 500, 30, rng,
            lambda: elapsed(), t_start + budget_s - 60)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 2: SLSQP polish from gradient result
    if time_left() > 200:
        for alpha in [200, 500, 1000, 3000]:
            if time_left() < 30:
                break
            h_s = slsqp_optimize(best_h, ts, alpha, 40, 400)
            c5_s = compute_c5(h_s)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_s

    # Phase 3: Block swap local search
    if time_left() > 150:
        h_bin = (best_h > 0.5).astype(float)
        att = 0
        while abs(h_bin.sum() - ts) > 1e-10 and att < 20000:
            if h_bin.sum() > ts:
                idx = rng.choice(np.where(h_bin > 0)[0])
                h_bin[idx] = 0.0
            else:
                idx = rng.choice(np.where(h_bin < 1)[0])
                h_bin[idx] = 1.0
            att += 1
        if abs(h_bin.sum() - ts) < 1e-10:
            h_bswap, c5_bswap = block_swap_improve(
                h_bin, ts, rng, 200, lambda: elapsed(), t_start + budget_s - 40)
            if c5_bswap < best_c5:
                best_c5 = c5_bswap
                best_h = h_bswap.copy()
                h_rs, c5_rs = gradient_descent_run(
                    h_bswap, ts, 3000, 0.015, 100, 1000, 40, rng,
                    lambda: elapsed(), t_start + budget_s - 40)
                if c5_rs < best_c5:
                    best_c5 = c5_rs
                    best_h = h_rs

    # Phase 4: Random swap on binary + relax
    if time_left() > 150:
        h_bin2 = (best_h > 0.5).astype(float)
        att = 0
        while abs(h_bin2.sum() - ts) > 1e-10 and att < 20000:
            if h_bin2.sum() > ts:
                idx = rng.choice(np.where(h_bin2 > 0)[0])
                h_bin2[idx] = 0.0
            else:
                idx = rng.choice(np.where(h_bin2 < 1)[0])
                h_bin2[idx] = 1.0
            att += 1
        if abs(h_bin2.sum() - ts) < 1e-10:
            h_rswap, c5_rswap = random_swap_improve(
                h_bin2, ts, rng, 100, lambda: elapsed(), t_start + budget_s - 40)
            if c5_rswap < best_c5:
                best_c5 = c5_rswap
                best_h = h_rswap.copy()
                h_relax = slsqp_optimize(h_rswap, ts, 300, 50, 600)
                c5_relax = compute_c5(h_relax)
                if c5_relax < best_c5:
                    best_c5 = c5_relax
                    best_h = h_relax

    # Phase 5: Multi-start with diverse initializations
    if time_left() > 250:
        starts = make_diverse_starts(best_h, best_n, ts, rng, 12)
        for h_st in starts:
            if time_left() < 25:
                break
            h_r, c5_r = gradient_descent_run(
                h_st, ts, 2000, 0.015, 30, 500, 25, rng,
                lambda: elapsed(), t_start + budget_s - 40)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()
            if time_left() < 25:
                break
            h_pol = slsqp_optimize(h_r, ts, 500, 50, 300)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Phase 6: Multi-resolution search
    for n_alt in [120, 180, 360, 480]:
        if time_left() < 100:
            break
        ts_alt = n_alt // 2
        h_interp = interpolate_to_n(best_h, n_alt, 2.0)
        h_interp = project_to_sum(h_interp, ts_alt)
        h_r, c5_r = gradient_descent_run(
            h_interp, ts_alt, 3000, 0.015, 50, 800, 30, rng,
            lambda: elapsed(), t_start + budget_s - 50)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_alt
        if time_left() < 50:
            break
        h_pol = slsqp_optimize(h_r, ts_alt, 500, 500, 300)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
            best_n = n_alt

    # Phase 7: Final high-alpha polish on best solution
    if time_left() > 40:
        ts_final = best_n // 2
        for alpha in [1000, 5000, 15000]:
            if time_left() < 25:
                break
            h_pol = slsqp_optimize(best_h, ts_final, alpha, 60, 2000)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol

    # Phase 8: Final fine-tune with very small steps
    if time_left() > 30:
        ts_final = best_n // 2
        h_fine, c5_fine = gradient_descent_run(
            best_h, ts_final, 10000, 0.003, 2000, 5000, 20, rng,
            lambda: elapsed(), t_start + budget_s - 5)
        if c5_fine < best_c5:
            best_c5 = c5_fine
            best_h = h_fine

    # Final validation
    ts_fin = best_n // 2
    best_h = project_to_sum(best_h, ts_fin)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
