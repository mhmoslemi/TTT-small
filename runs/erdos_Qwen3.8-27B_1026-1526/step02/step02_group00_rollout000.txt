import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 0)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    # Gradient
    # d(corr[k])/d(h[i]) for each k,i
    # corr[k] = sum_j h[j] * (1-h[j+k-n+1])
    # d(corr[k])/d(h[i]) = (1-h[i+k-n+1]) - h[i-k+n-1] (where valid)
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2*n-1)[None, :]
    first_idx = j_idx - m_idx + n - 1  # For the (1-h[...]) term
    second_idx = j_idx + m_idx - n + 1  # For the h[...] term
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n-1)
    sc = np.clip(second_idx, 0, n-1)
    # grad[i] = sum_k w[k] * d(corr[k])/d(h[i])
    # = sum_k w[k] * ((1-h[first_idx[i,k]]) * valid_f[i,k] - h[second_idx[i,k]] * valid_s[i,k])
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def slsqp_optimize(h0, target_sum, alpha, maxiter=500):
    n = len(h0)
    def f_and_g(h):
        return smooth_max_and_grad(h, alpha)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        obj_val, grad_val = f_and_g(h0)
        res = minimize(lambda h: f_and_g(h)[0], h0.copy(), method='SLSQP',
                       jac=lambda h: f_and_g(h)[1],
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'catol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def binary_local_search(h_bin, n, target_sum, max_rounds=200):
    h = h_bin.copy()
    c5_cur = compute_c5(h)
    corr_cur = compute_corr(h)
    dx = 2.0 / n
    for rnd in range(max_rounds):
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_improve = 1e-10
        best_pair = None
        # Systematic search: try all pairs
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                c_new = compute_c5(h)
                h[i] = 1.0
                h[j] = 0.0
                if c_new < c5_cur - best_improve:
                    best_improve = c5_cur - c_new
                    best_pair = (i, j)

2. For each 0-position j, compute the "benefit" of adding it
3. Find the pair (i,j) that maximizes the improvement

This is still O(n²) per round but with better constants.

def binary_local_search(h_bin, n, max_rounds=500):
    h = h_bin.astype(float).copy()
    dx = 2.0 / n
    c5_cur = float(np.max(compute_corr(h)) * dx)
    for rnd in range(max_rounds):
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_c5 = c5_cur - 1e-12
        best_pair = None
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                c_new = float(np.max(compute_corr(h)) * dx)
                h[i] = 1.0
                h[j] = 0.0
                if c_new < best_c5:
                    best_c5 = c_new
                    best_pair = (i, j)
        if best_pair is None:
            break
        h[best_pair[0]] = 0.0
        h[best_pair[1]] = 1.0
        c5_cur = best_c5
    return h, c5_cur

- Move in the direction -∇a_{k*}(h)
- Project onto feasible set

This is simple but can be slow. A better variant is the "average subgradient" or using the smoothing approach.
