    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full") * dx
    c_max = c.max()
    if c_max < -500:

    val = sm
    if var_weight > 0:

    sv = np.sum(x) - n / 2.0
    val += penalty * sv * sv
    return val


def slsqp_obj(x):
    return x[-1]


def slsqp_ineq(x):
    h = x[:-1]
    t = x[-1]
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return t - c * dx


def slsqp_eq(x):
    h = x[:-1]
    return np.sum(h) - len(h) / 2.0


def slsqp_eq_jac(x):
    n = len(x) - 1
    jac = np.zeros(len(x))
    jac[:n] = 1.0
    return jac


def make_slsqp_constraints():
    return [
        {'type': 'ineq', 'fun': slsqp_ineq},
        {'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac}
    ]


def slsqp_optimize(h_init, n, maxiter=3000, t_margin=0.1):
    c5_init = compute_c5(h_init)
    x0 = np.concatenate([h_init, [c5_init + t_margin]])
    bounds = [(0.0, 1.0)] * n + [(max(0.001, c5_init * 0.3), 2.0)]
    constraints = make_slsqp_constraints()
    result = minimize(slsqp_obj, x0, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
    if result.x is not None:
        h_out = project_feasible(result.x[:-1], n)
        return h_out, compute_c5(h_out)
    return h_init, c5_init


def lbfgs_optimize(h_init, n, T, penalty, var_weight, maxiter=3000):

        method='L-BFGS-B', bounds=bounds,

            best_c5 = c5_new
            n_best = n_high

    # Phase 5: Structured perturbations with local refinement
    n_perturb = 40
    for attempt in range(n_perturb):
        if time_left() < 30:
            break

        frac = attempt / max(1, n_perturb)
        scale = 0.12 * (1.0 - frac) ** 0.7 + 0.005
        mode = int(rng.integers(0, 7))
        h_pert = perturb_structured(best_h, rng, scale, mode)
        h_pert = project_feasible(h_pert, n_best)

        if attempt < n_perturb // 3:
            h_new, c5_new = slsqp_optimize(h_pert, n_best, maxiter=3000, t_margin=0.06)
        elif attempt < 2 * n_perturb // 3:
            h_new, c5_new = lbfgs_optimize(h_pert, n_best, T=30.0,
                                           penalty=150.0, var_weight=0.0, maxiter=2000)
            if time_left() < 20:
                break
            h_new, c5_new = slsqp_optimize(h_new, n_best, maxiter=2000, t_margin=0.04)
        else:
            h_new, c5_new = slsqp_optimize(h_pert, n_best, maxiter=2000, t_margin=0.04)

        if c5_new < best_c5 - 1e-12:
            best_h = h_new
            best_c5 = c5_new

    # Phase 6: Bottleneck-lag guided perturbation
    if time_left() > 80:
        top_lags, _ = get_bottleneck_lags(best_h, n_top=15)
        for attempt in range(15):
            if time_left() < 15:
                break
            scale = 0.02 + 0.04 * (1 - attempt / 15.0)
            h_pert = best_h.copy()
            n_lags = len(top_lags)
            for li in range(n_lags):
                lag_idx = top_lags[li]
                lag = lag_idx - (n_best - 1)
                amp = scale / np.sqrt(n_lags) * (rng.random() - 0.5) * 2
                h_pert += amp * np.roll(1.0 - best_h, -lag)
            h_pert = project_feasible(h_pert, n_best)
            h_new, c5_new = slsqp_optimize(h_pert, n_best, maxiter=2500, t_margin=0.03)
            if c5_new < best_c5 - 1e-12:
                best_h = h_new
                best_c5 = c5_new

    # Phase 7: Fine polishing with SLSQP
    if time_left() > 40:
        for polish_round in range(3):
            if time_left() < 15:
                break
            maxiter = max(2000, int(time_left() * 60))
            best_h, best_c5 = slsqp_optimize(best_h, n_best, maxiter=maxiter, t_margin=0.02)

    # Phase 8: Tiny perturbations + SLSQP
    if time_left() > 30:
        for attempt in range(30):
            if time_left() < 8:
                break
            scale = 0.002 + 0.008 * (1 - attempt / 30.0) ** 2
            h_pert = best_h + scale * rng.standard_normal(n_best)
            h_pert = project_feasible(h_pert, n_best)
            h_new, c5_new = slsqp_optimize(h_pert, n_best, maxiter=2000, t_margin=0.01)
            if c5_new < best_c5 - 1e-14:
                best_h = h_new
                best_c5 = c5_new

    # Final: one more SLSQP pass
    if time_left() > 10:
        best_h, best_c5 = slsqp_optimize(best_h, n_best, maxiter=5000, t_margin=0.005)

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
