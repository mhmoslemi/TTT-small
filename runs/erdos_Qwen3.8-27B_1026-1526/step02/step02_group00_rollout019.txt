from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    shifted = corr - m_val
    exp_vals = np.exp(alpha * shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    
    i_idx = np.arange(n)[:, None]
    k_idx = np.arange(2 * n - 1)[None, :]
    r1 = i_idx - k_idx + n - 1
    valid1 = (r1 >= 0) & (r1 < n)
    r1c = np.clip(r1, 0, n - 1)
    r2 = i_idx + k_idx - n + 1
    valid2 = (r2 >= 0) & (r2 < n)
    r2c = np.clip(r2, 0, n - 1)
    
    term1 = valid1 * (1.0 - h[r1c])
    term2 = valid2 * h[r2c]
    grad = np.sum(w[None, :] * (term1 - term2), axis=1)
    
    return obj, grad

def fourier_eval(coeffs, n):
    """Evaluate h from Fourier coefficients [a0, a1, b1, a2, b2, ...]"""
    x = np.linspace(0, 2.0, n, endpoint=False)
    h = np.full(n, coeffs[0])
    n_modes = (len(coeffs) - 1) // 2
    for m in range(n_modes):
        a = coeffs[1 + 2*m]
        b = coeffs[2 + 2*m]
        h += a * np.cos(np.pi * m * x) + b * np.sin(np.pi * m * x)
    h = np.clip(h, 0.0, 1.0)
    h = project_to_sum(h, n / 2.0)
    return h

def fourier_obj(coeffs, n, alpha):
    h = fourier_eval(coeffs, n)
    return compute_c5(h)

def slsqp_local(h0, n, target_sum, alpha, maxiter):
    def f(h):
        return smooth_max_and_grad(h, alpha)[0]
    def g(h):
        return smooth_max_and_grad(h, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def nelder_mead_fourier(h_init, n, M, alpha, maxiter, budget_check):
    """Optimize Fourier coefficients using Nelder-Mead."""
    x = np.linspace(0, 2.0, n, endpoint=False)
    # Extract Fourier coefficients from h_init
    h = project_to_sum(h_init, n / 2.0)
    # Compute FFT to get initial coefficients
    fft_h = np.fft.rfft(h)
    # DC
    a0 = np.real(fft_h[0]) / n  # should be ~0.5
    coeffs = [a0]
    for m in range(1, M + 1):
        a = 2.0 * np.real(fft_h[m]) / n
        b = -2.0 * np.imag(fft_h[m]) / n
        coeffs.append(a)
        coeffs.append(b)
    coeffs = np.array(coeffs)
    
    def obj(c):
        if budget_check is not None and budget_check():
            return 1e10
        h_eval = fourier_eval(c, n)
        return compute_c5(h_eval)
    
    try:
        res = minimize(obj, coeffs, method='Nelder-Mead',
                       options={'maxiter': maxiter, 'xatol': 1e-8, 'fatol': 1e-10})
        h_result = fourier_eval(res.x, n)
        return h_result, res.x
    except Exception:
        return project_to_sum(h_init, n / 2.0), coeffs

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    shifted = corr - m_val
    exp_vals = np.exp(alpha * shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    i_idx = np.arange(n)[:, None]
    k_idx = np.arange(2 * n - 1)[None, :]
    r1 = i_idx - k_idx + n - 1
    valid1 = (r1 >= 0) & (r1 < n)
    r1c = np.clip(r1, 0, n - 1)
    r2 = i_idx + k_idx - n + 1
    valid2 = (r2 >= 0) & (r2 < n)
    r2c = np.clip(r2, 0, n - 1)
    term1 = valid1 * (1.0 - h[r1c])
    term2 = valid2 * h[r2c]
    grad = np.sum(w[None, :] * (term1 - term2), axis=1)
    return obj, grad

def slsqp_local(h0, n, target_sum, alpha, maxiter):
    def f(h):
        return smooth_max_and_grad(h, alpha)[0]
    def g(h):
        return smooth_max_and_grad(h, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def lbfgsb_local(h0, n, target_sum, alpha, maxiter):
    h0_clip = np.clip(h0, 1e-6, 1.0 - 1e-6)
    def f_and_g(x):
        h = np.concatenate([x, [target_sum - np.sum(x)]])
        h = np.clip(h, 1e-6, 1.0 - 1e-6)
        obj, grad_full = smooth_max_and_grad(h, alpha)
        grad = grad_full[:-1] - grad_full[-1]
        return obj, grad
    bounds = [(1e-6, 1.0 - 1e-6)] * (n - 1)
    try:
        res = minimize(f_and_g, h0_clip[:-1].copy(), method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-8})
        h_out = np.concatenate([res.x, [target_sum - np.sum(res.x)]])
        return project_to_sum(h_out, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def fourier_from_h(h, M):
    n = len(h)
    fft_h = np.fft.rfft(h)
    a0 = np.real(fft_h[0]) / n
    coeffs = [a0]
    for m in range(1, M + 1):
        a = 2.0 * np.real(fft_h[m]) / n
        b = -2.0 * np.imag(fft_h[m]) / n
        coeffs.append(a)
        coeffs.append(b)
    return np.array(coeffs)

def h_from_fourier(coeffs, n):
    x = np.linspace(0, 2.0, n, endpoint=False)
