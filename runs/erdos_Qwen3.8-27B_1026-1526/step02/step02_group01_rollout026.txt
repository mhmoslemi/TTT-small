def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):

    c5_cur = compute_c5(h)
    h_best = h.copy()
    
    for alpha in alphas:
        t = 1.0
        y = h.copy()
        lr = lr_init

                return h_best, c5_cur

            h_new = np.clip(y - lr * g, 0.0, 1.0)

    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_and_grad(h, alpha)
        return v
    
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_and_grad(h, alpha)
        return g
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

        if lag >= 0:
            if lag < n:
                g[lag:n] += wk * (1.0 - h[0:n-lag])
                g[0:n-lag] -= wk * h[lag:n]
        else:
            L = -lag
            if L < n:
                g[0:n-L] += wk * (1.0 - h[L:n])
                g[L:n] -= wk * h[0:n-L]
    return sm_val, g

def nesterov_optimize(h0, n, target_sum, alphas, iters_per_alpha, lr_init, time_limit):
    t_start = time.time()
    h = h0.copy()
    c5_cur = compute_c5(h)
    h_best = h.copy()
    for alpha in alphas:
        if time.time() - t_start > time_limit:
            break
        t = 1.0
        y = h.copy()
        lr = lr_init
        stagnant = 0
        for it in range(iters_per_alpha):
            if time.time() - t_start > time_limit:
                return h_best, c5_cur
            sm_val, g = smooth_max_and_grad(y, alpha)
            h_new = np.clip(y - lr * g, 0.0, 1.0)
            h_new = project_to_sum(h_new, target_sum)
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
            mo = (t - 1.0) / t_new
            y = h_new + mo * (h_new - h)
            y = np.clip(y, 0.0, 1.0)
            h = h_new
            t = t_new
            c5_new = compute_c5(h)
            if c5_new < c5_cur - 1e-15:
                c5_cur = c5_new
                h_best = h.copy()
                stagnant = 0
                lr = min(lr * 1.003, 3.0 / alpha)
            else:
                stagnant += 1
                if stagnant % 100 == 0:
                    lr *= 0.92
        h = h_best.copy()
    return h_best, c5_cur

def subgradient_top_lags(h0, n, target_sum, time_limit, rng, lr_init=0.003, top_k=5):
    t_start = time.time()
    h = h0.copy()
    c5_cur = compute_c5(h)
    h_best = h.copy()
    lr = lr_init
    stagnant = 0
    while time.time() - t_start < time_limit:
        corr = get_corr(h)
        top_indices = np.argsort(corr)[-top_k:]
        g = np.zeros(n)
        for k in top_indices:
            lag = k - (n - 1)
            if lag >= 0:
                g[lag:n] += (1.0 - h[0:n-lag])
                g[0:n-lag] -= h[lag:n]
            else:
                L = -lag
                g[0:n-L] += (1.0 - h[L:n])
                g[L:n] -= h[0:n-L]
        g /= float(top_k)
        h_new = h - lr * g
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur - 1e-15:
            h = h_new
            c5_cur = c5_new
            h_best = h.copy()
            stagnant = 0
            lr = min(lr * 1.01, 0.05)
        else:
            stagnant += 1
            lr *= 0.97
            if stagnant > 300:
                h = h_best + rng.uniform(-0.01, 0.01, n)
                h = np.clip(h, 0.0, 1.0)
                h = project_to_sum(h, target_sum)
                lr = lr_init
                stagnant = 0
    return h_best, c5_cur

def coordinate_descent_fine(h0, n, target_sum, time_limit, rng):
    t_start = time.time()
    h = h0.copy()
    c5_cur = compute_c5(h)
    step_sizes = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    for step_size in step_sizes:
        if time.time() - t_start > time_limit:
            break
        deltas = [-step_size, -step_size*0.5, step_size*0.5, step_size]
        for rnd in range(300):
            if time.time() - t_start > time_limit:
                break
            improved = False
            interior = np.where((h > 1e-6) & (h < 1.0 - 1e-6))[0]
            if len(interior) == 0:
                break
            idx_order = rng.permutation(interior)
            for i in idx_order:
                if time.time() - t_start > time_limit:
                    break
                for d in deltas:
                    h_new = h.copy()
                    h_new[i] = np.clip(h_new[i] + d, 0.0, 1.0)
                    h_new = project_to_sum(h_new, target_sum)
                    c5_new = compute_c5(h_new)
                    if c5_new < c5_cur - 1e-15:
                        h = h_new
                        c5_cur = c5_new
                        improved = True
                        break
            if not improved:
                break
    return h, c5_cur

def bang_bang_rounding(h0, n, target_sum, time_limit, rng):
    t_start = time.time()
    h = h0.copy()
    c5_cur = compute_c5(h)
    while time.time() - t_start < time_limit:
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            if time.time() - t_start > time_limit:
                break
            if h[i] > 0.05:
                h_new = h.copy()
                h_new[i] = 0.0
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < c5_cur - 1e-15:
                    h = h_new
                    c5_cur = c5_new
                    improved = True
                    continue
            if h[i] < 0.95:
                h_new = h.copy()
                h_new[i] = 1.0
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < c5_cur - 1e-15:
                    h = h_new
                    c5_cur = c5_new
                    improved = True
        if not improved:
            break
    return h, c5_cur

def slsqp_with_smmax(h0, n, target_sum, alpha, maxiter=200):
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_and_grad(h, alpha)
        return v
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_and_grad(h, alpha)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)
