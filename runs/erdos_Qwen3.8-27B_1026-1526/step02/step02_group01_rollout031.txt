def topL_smooth_max_grad(h, alpha, L):
    """Compute smooth max value and gradient using top-L lags."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
    m = np.max(corr)
    
    # Get top L indices
    top_idx = np.argsort(corr)[-L:]
    top_corr = corr[top_idx]
    
    # Softmax weights
    shifted = alpha * (top_corr - m)
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    
    # Smooth max value (approximation using top L)
    sm_val = m + np.log(s / L) / alpha  # approximation
    
    # Gradient: for each active lag k, ∂corr[k]/∂h[i] = (1-h[i+k]) - h[i-k]
    # Sum over k with weights w
    grad = np.zeros(n)
    for idx_i, k in enumerate(top_idx):
        wi = w[idx_i]
        # k is the index into corr (0 to 2n-2)
        # The lag value is k - (n-1)
        lag = k - (n - 1)
        if lag >= 0:
            # h[i+k] for i=0..n-1-k, h[i-k] for i=k..n-1
            grad[:n-lag] += wi * (1.0 - h[lag:n])
            grad[lag:] -= wi * h[:n-lag]
        else:
            lag_pos = -lag
            grad[lag_pos:] += wi * (1.0 - h[:n-lag_pos])
            grad[:n-lag_pos] -= wi * h[lag_pos:n]
    
    return sm_val, grad

    """Project onto {x: 0<=x<=1, sum(x)=target_sum} via bisection on offset."""
    n = len(h)
    if n == 0:
        return h

    hi = target_sum - np.sum(np.maximum(h, 0.0))

    # We want h_proj[i] = h[i] + c (same c for all i), clipped to [0,1].
    # But that's too restrictive. The optimal projection allows different adjustments.
    # The optimal projection is: h_proj[i] = clip(h[i] + c, 0, 1) for a single c.
    # This IS the correct Euclidean projection onto the intersect of the box and the hyperplane.
    
    # f(c) = sum(clip(h[i] + c, 0, 1)) is continuous and monotonically increasing
    # We need f(c) = target_sum
    
    # Lower bound: c such that all h[i]+c <= 0, i.e., c <= -max(h)
    # Upper bound: c such that all h[i]+c >= 1, i.e., c >= 1 - min(h)
    c_lo = -np.max(h)
    c_hi = 1.0 - np.min(h)

        if s < target_sum:
            c_lo = c_mid
        else:

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project_to_sum(h, target_sum):
    n = len(h)
    if n == 0:
        return h
    c_lo = -np.max(h)
    c_hi = 1.0 - np.min(h)
    for _ in range(80):
        c_mid = (c_lo + c_hi) / 2.0
        s = np.sum(np.clip(h + c_mid, 0.0, 1.0))
        if s < target_sum:
            c_lo = c_mid
        else:
            c_hi = c_mid
    c = (c_lo + c_hi) / 2.0
    return np.clip(h + c, 0.0, 1.0)


def topL_smooth_max_grad(h, alpha, L):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    L_eff = min(L, 2 * n - 1)
    top_idx = np.argsort(corr)[-L_eff:]
    top_corr = corr[top_idx]
    shifted = alpha * (top_corr - m)
    shifted = np.clip(shifted, -500.0, 500.0)

    s = np.sum(exp_vals)
    w = exp_vals / s
