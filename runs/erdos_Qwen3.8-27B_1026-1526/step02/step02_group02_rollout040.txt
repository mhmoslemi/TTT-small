    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            if not np.any(mask):

            step = min(diff / np.sum(mask), (1.0 - h[mask]).min() * 0.9999)
            h[mask] += step
        else:

            if not np.any(mask):

            step = min(-diff / np.sum(mask), h[mask].min() * 0.9999)
            h[mask] -= step
    return np.clip(h, 0.0, 1.0)

def c5_all_shifts(h):

    n = len(h)

def coordinate_descent_step(h, n, grid_size=64):

    grid = np.linspace(0, 1, grid_size)

        best_c5 = compute_c5_from_h(h)

        h[j] = best_val

def make_delta_c(h, n, j):
    """
    Precompute the vector d such that changing h[j] by delta changes c by delta * d.
    c = np.correlate(h, 1-h, mode="full")
    """
    # c[m] for m = 0, ..., 2n-2
    # c[m] = sum_i h[i] * (1-h[i+m-(n-1)])
    # Let shift = m - (n-1)
    # c[n-1+shift] = sum_i h[i] * (1-h[i+shift])
    
    # h[j] appears in c[n-1+shift] when:
    # - i = j: need 0 <= j+shift <= n-1
    # - i+shift = j: i = j-shift, need 0 <= j-shift <= n-1
    # Change in c[n-1+shift] = delta * [(1-h[j+shift])*I(0<=j+shift<=n-1) - h[j-shift]*I(0<=j-shift<=n-1)]
    # Special case shift=0: h[j]*(1-h[j]) changes by delta*(1-h[j-]*h[j]) where h[j*] is the old value
    # Actually for shift=0: the term is h[j](1-h[j]), old contribution = a(1-a), new = b(1-b)
    # Change = b(1-b) - a(1-a) = (b-a)(1-a-b) = delta*(1-a-b)
    # But for shift≠0: change = delta*(1-h[j+shift] - h[j-shift])
    
    d = np.zeros(2*n-1)
    for shift in range(-n+1, n):
        m = n - 1 + shift
        if m < 0 or m >= 2*n-1:
            continue
        if shift == 0:
            # Will be handled separately since it depends on both old and new values
            d[m] = 0  # placeholder
        else:
            contrib = 0.0
            if 0 <= j + shift <= n - 1:
                contrib += (1.0 - h[j + shift])
            if 0 <= j - shift <= n - 1:
                contrib -= h[j - shift]
            d[m] = contrib
    
    return d

For shift=0: c[n-1] includes the term h[j](1-h[j]). When h[j] changes from a to b:

- Change = b - b² - a + a² = (b-a) - (b²-a²) = (b-a)(1-a-b)

So for shift=0, the "derivative" is (1-a-b), which depends on both old and new values. This is the only special case.

For a grid search, I can handle it as:
- new_c[n-1] = old_c[n-1] + (x - old)*(1 - old - x) for the shift=0 contribution
- new_c[m] = old_c[m] + (x - old) * d[m] for m ≠ n-1

This is O(1) per m (since I precomputed d), so O(n) per candidate x, O(n * grid_size) per coordinate, O(n² * grid_size) per sweep.

For n=270, grid_size=81: 270² * 81 = 5.9M per sweep ≈ 0.02 seconds. I can do 5000 sweeps in 100 seconds!

def coord_descent_optimized(h, n, grid, max_sweeps, t0, time_limit):

            break
        
        any_change = False
        for j in range(n):
            old = h[j]
            best_c5 = c_max
            best_x = old

            # For shift ≠ 0: delta_c[shift] = (1-h[j+shift])*I(valid) - h[j-shift]*I(valid)
            # For shift = 0: special handling
            
            # Precompute the linear part
            # c_new[m] = c[m] + delta * linear_d[m] for m ≠ n-1
            # c_new[n-1] = c[n-1] + delta * (1 - old - x)  [quadratic in x]
            #          = c[n-1] + (x-old)*(1-old-x)
            
            linear_d = np.zeros(2*n-1)
            for shift in range(-n+1, n):
                if shift == 0:
                    continue
                m = n - 1 + shift
                if m < 0 or m >= 2*n-1:

                js = j + shift
                js_minus = j - shift
                if 0 <= js <= n-1:
                    contrib += (1.0 - h[js])
                if 0 <= js_minus <= n-1:
                    contrib -= h[js_minus]
                linear_d[m] = contrib
            
            # Now search over grid
            for x in grid:
                if abs(x - old) < 1e-12:
                    continue
                delta = x - old
                
                # Compute new c_max
                # For m ≠ n-1: new_c[m] = c[m] + delta * linear_d[m]
                # For m = n-1: new_c[n-1] = c[n-1] + delta * (1 - old - x)
                
                # New max = max(max_{m≠n-1}(c[m] + delta*linear_d[m]), c[n-1] + delta*(1-old-x))
                
                # For m ≠ n-1, this is: max(c[m] + delta*linear_d[m] for m ≠ n-1)
                # = max of an affine function of delta (piecewise linear in x)
                # Can compute as: c[m] + delta*linear_d[m] for all m ≠ n-1, take max
                # This is O(n) per candidate.
                
                c_temp = c + delta * linear_d
                c_temp[n-1] = c[n-1] + delta * (1.0 - old - x)
                new_max = c_temp.max() * dx
                
                if new_max < best_c5 - 1e-13:
                    best_c5 = new_max
                    best_x = x
            
            if best_x != old:
                # Update h and c
                h[j] = best_x
                delta = best_x - old
                c = c + delta * linear_d
                c[n-1] = c[n-1] + delta * (1.0 - old - best_x)
                c_max = c.max()
                any_change = True
        
        if not any_change:
            break
    
    return h, c_max

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            room = 1.0 - h[mask]
            step = min(diff / cnt, room.min() * 0.99999)
            h[mask] += step
        else:
            mask = h > 1e-15
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            room = h[mask]
            step = min(-diff / cnt, room.min() * 0.99999)
            h[mask] -= step
    return np.clip(h, 0.0, 1.0)


def compute_delta_linear(h, n, j):
    d = np.zeros(2 * n - 1)
    for shift in range(-n + 1, n):
        if shift == 0:
            continue
        m = n - 1 + shift
        if m < 0 or m >= 2 * n - 1:
            continue
        contrib = 0.0
        js = j + shift
        jm = j - shift
        if 0 <= js <= n - 1:
            contrib += (1.0 - h[js])
        if 0 <= jm <= n - 1:
            contrib -= h[jm]
        d[m] = contrib
    return d


def coord_descent(h, n, grid, max_sweeps, t0, time_limit, order_seed=None):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    c_max = c.max() * dx
    rng_local = np.random.default_rng(order_seed) if order_seed is not None else None

    for sweep in range(max_sweeps):
        if time.time() - t0 > time_limit:
            break
        if rng_local is not None and sweep % 10 == 0:
            order = rng_local.permutation(n)
        else:
            order = range(n)

        any_change = False
        for j in order:
            old = h[j]
            best_c5 = c_max
            best_x = old
            linear_d = compute_delta_linear(h, n, j)

            for x in grid:
                if abs(x - old) < 1e-12:
                    continue
                delta = x - old
                new_c = c + delta * linear_d
                new_c[n - 1] = c[n - 1] + delta * (1.0 - old - x)
                new_max = new_c.max() * dx
                if new_max < best_c5 - 1e-14:
                    best_c5 = new_max
                    best_x = x

            if best_x != old:
                delta = best_x - old
                c = c + delta * linear_d
                c[n - 1] = c[n - 1] + delta * (1.0 - old - best_x)
                h[j] = best_x
                c_max = c.max() * dx
                any_change = True

        if not any_change:
            break

    return h, c_max


def slsqp_obj(x):
    return x[-1]


def slsqp_ineq(x):
    h = x[:-1]
    t = x[-1]
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return t - c * dx


def slsqp_eq(x):
    return np.sum(x[:-1]) - len(x[:-1]) / 2.0


def slsqp_eq_jac(x):
    n = len(x) - 1
    jac = np.zeros(len(x))
    jac[:n] = 1.0
    return jac


def run_slsqp(h_init, t0, time_limit, maxiter=5000):
    n = len(h_init)
    c5_init = compute_c5(h_init)
    x0 = np.concatenate([h_init, [c5_init + 0.15]])
    bounds = [(0.0, 1.0)] * n + [(max(0.001, c5_init * 0.7), 2.0)]
    constraints = [
        {'type': 'ineq', 'fun': slsqp_ineq},
        {'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac}
    ]
    result = minimize(slsqp_obj, x0, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
    if result.x is not None:
        h_out = project_feasible(result.x[:-1], n)
        return h_out, compute_c5(h_out)
    return h_init, c5_init


def c5_logsumexp(h, T):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    c_max = c.max()
    shifted = T * (c - c_max)
    shifted = np.clip(shifted, -400, 400)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / T)


def smooth_obj(x, T, penalty):
    n = len(x)
    c5_val = c5_logsumexp(x, T)
    sum_violation = np.sum(x) - n / 2.0
    return c5_val + penalty * sum_violation ** 2


def run_lbfgs(h_init, T, penalty, t0, time_limit, maxiter=3000):
    n = len(h_init)
    bounds = [(0.0, 1.0)] * n
    result = minimize(
        smooth_obj, h_init, args=(T, penalty),
        method='L-BFGS-B', bounds=bounds,
        options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14}
    )
    if result.x is not None:
        h_out = project_feasible(result.x, n)
        return h_out, compute_c5(h_out)
    return h_init, compute_c5(h_init)
