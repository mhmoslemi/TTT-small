def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smax_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha

def smax_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    w /= np.sum(w)
    grad = np.zeros(n)
    for j in range(n):
        # sum over all shifts d = -(n-1) to n-1
        # dcorr_k/dh[j] = [j-d ∈ N(d)]*(1-h[j-d]) - [j ∈ N(d)]*h[j+d]
        # where N(d) = valid range of n for shift d
        for d in range(-(n-1), n):
            k = d + n - 1  # index into corr/w
            # Check if j-d is in N(d)
            nd_lo = max(0, -d)
            nd_hi = min(n, n - d) - 1
            if nd_lo <= j - d <= nd_hi:
                grad[j] += w[k] * (1.0 - h[j - d])
            # Check if j is in N(d)
            if nd_lo <= j <= nd_hi:
                grad[j] -= w[k] * h[j + d]
    return grad

def smax_obj_grad(h, alpha, penalty=0.0):
    n = len(h)
    target = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exp_shift = np.exp(alpha * (corr - m))
    se = np.sum(exp_shift)
    w = exp_shift / se
    obj = m + np.log(se) / alpha
    diff = np.sum(h) - target
    obj += 0.5 * penalty * diff * diff
    
    wr = w[n-1:]
    wl_full = np.zeros(n)
    wl_full[1:] = w[:n-1][::-1]
    
    rev_h = h[::-1]
    
    conv1 = np.convolve(wr, h, 'full')[:n]
    conv4 = np.convolve(wl_full, h, 'full')[:n]
    conv3 = np.convolve(wr, rev_h, 'full')[:n][::-1]
    conv_wl1_rev = np.convolve(wl_full[1:], rev_h, 'full')
    conv2 = np.zeros(n)
    conv2[:n-1] = conv_wl1_rev[:n-1][::-1]
    
    prefix_wr = np.concatenate([[0], np.cumsum(wr)])
    prefix_wl = np.concatenate([[0], np.cumsum(wl_full)])
    
    A = np.zeros(n)
    for j in range(n):
        A[j] = prefix_wr[j+1] + prefix_wl[n-j]
    
    grad = A - conv1 - conv2 - conv3 - conv4
    grad += penalty * diff
    
    return obj, grad

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smax_obj_grad(h, alpha, penalty):
    n = len(h)
    target = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exp_shift = np.exp(alpha * (corr - m))
    se = np.sum(exp_shift)
    w = exp_shift / se
    obj = m + np.log(se) / alpha
    diff = np.sum(h) - target
    obj += 0.5 * penalty * diff * diff
    
    wr = w[n-1:]
    wl_full = np.zeros(n)
    if n > 1:
        wl_full[1:] = w[:n-1][::-1]
    
    rev_h = h[::-1]
    
    conv1 = np.convolve(wr, h, 'full')[:n]
    conv4 = np.convolve(wl_full, h, 'full')[:n]
    conv3 = np.convolve(wr, rev_h, 'full')[:n][::-1]
    
    if n > 1:
        conv_wl1_rev = np.convolve(wl_full[1:], rev_h, 'full')
        conv2 = np.zeros(n)
        conv2[:n-1] = conv_wl1_rev[:n-1][::-1]
    else:
        conv2 = np.zeros(n)
    
    prefix_wr = np.concatenate([[0.0], np.cumsum(wr)])
    prefix_wl = np.concatenate([[0.0], np.cumsum(wl_full)])
    
    A = prefix_wr[1:n+1] + prefix_wl[1:n+1][::-1]

    # prefix_wl[1:n+1][::-1] gives prefix_wl[n], ..., prefix_wl[1]
    # So A[j] = prefix_wr[j+1] + prefix_wl[n-j] ✓
    
    grad = A - conv1 - conv2 - conv3 - conv4
    grad += penalty * diff
    
    return obj, grad

    def jac(x):
        return smax_obj_grad(x, alpha, penalty)[1]
    
    try:
        res = minimize(fun, h0.copy(), method='L-BFGS-B', jac=jac,
                      bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-10})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

    h = np.clip(x, 0, 1)
    obj, _ = smax_obj_grad(h, alpha, penalty)
    return obj

def de_wrap(x, bounds_flat, alpha, penalty):
    h = np.clip(x, 0, 1)

        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        delta = rng.uniform(-step_size, step_size)
        h_test = h.copy()
        h_test[i] = np.clip(h_test[i] + delta, 0.0, 1.0)
        h_test[j] = np.clip(h_test[j] - delta, 0.0, 1.0)
        if abs(h_test.sum() - target_sum) > 1e-8:
            continue
        c5_test = compute_c5(h_test)
        if c5_test < c5_cur - 1e-12:
            h = h_test
            c5_cur = c5_test
    return h, c5_cur


def binary_refine(h, n, target_sum, rng, max_rounds=100):
    h = h.copy()
    h_bin = (h > 0.5).astype(float)
    n_ones = int(h_bin.sum())
    n_target = n // 2
    if n_ones > n_target:
        ones_idx = np.where(h_bin == 1)[0]
        to_flip = rng.choice(ones_idx, n_ones - n_target, replace=False)
        h_bin[to_flip] = 0
    elif n_ones < n_target:
        zeros_idx = np.where(h_bin == 0)[0]
        to_flip = rng.choice(zeros_idx, n_target - n_ones, replace=False)
        h_bin[to_flip] = 1
    c5_cur = compute_c5(h_bin)
    for _ in range(max_rounds):
        improved = False
        ones = np.where(h_bin > 0.5)[0]
        zeros = np.where(h_bin < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        n_trials = min(300, len(ones) * len(zeros))
        best_c5 = c5_cur
        best_pair = None
        for _ in range(n_trials):
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h_test = h_bin.copy()
            h_test[i] = 0.0
            h_test[j] = 1.0
            c5_test = compute_c5(h_test)
            if c5_test < best_c5 - 1e-12:
                best_c5 = c5_test
                best_pair = (i, j)
        if best_pair is not None:
            h_bin[best_pair[0]] = 0.0
            h_bin[best_pair[1]] = 1.0
            c5_cur = best_c5
            improved = True
        if not improved:
            break
    return h_bin, c5_cur


def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float("inf")
    best_n = 0

    if "initial_h_values" in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 120
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    resolutions = [best_n, 60, 120, 180, 240, 320, 360]
    seen_res = set()
    ordered_res = []
    for r in resolutions:
        if r % 2 == 0 and r not in seen_res and r >= 20:
            seen_res.add(r)
            ordered_res.append(r)
    ordered_res.sort()

    for n_tr in ordered_res:
        if budget_s - (time.time() - t_start) < 50:
            break
        ts_tr = n_tr / 2.0
        starts = make_starts(n_tr, ts_tr, rng, best_h, best_n)

        for h_start in starts:
            if budget_s - (time.time() - t_start) < 25:
                break
            h_start = project_to_sum(h_start, ts_tr)
            c5_s = compute_c5(h_start)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_start.copy()
                best_n = n_tr

            alphas = [5, 15, 50, 200]
            penalties = [50, 500]
            for alpha in alphas:
                if budget_s - (time.time() - t_start) < 15:
                    break
                for penalty in penalties:
                    if budget_s - (time.time() - t_start) < 10:
                        break
                    h_opt = lbfgs_optimize(h_start, alpha, penalty, ts_tr, maxiter=400)
                    c5_opt = compute_c5(h_opt)
                    if c5_opt < best_c5:
                        best_c5 = c5_opt
                        best_h = h_opt.copy()
                        best_n = n_tr
                    h_start = h_opt

    if budget_s - (time.time() - t_start) > 60:
        n_f = best_n
        ts_f = n_f / 2.0
        n_restarts = min(15, int((budget_s - (time.time() - t_start)) / 40))
        for trial in range(n_restarts):
            if budget_s - (time.time() - t_start) < 25:
                break
            noise = 0.08 * (1.0 - trial / max(n_restarts, 1)) + 0.02
            h_shake = best_h + rng.uniform(-noise, noise, n_f)
            h_shake = project_to_sum(h_shake, ts_f)
            for alpha in [30, 200, 800]:
                if budget_s - (time.time() - t_start) < 15:
                    break
                h_opt = lbfgs_optimize(h_shake, alpha, 500, ts_f, maxiter=500)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    best_n = n_f
                h_shake = h_opt

    if budget_s - (time.time() - t_start) > 40:
        h_bin, c5_bin = binary_refine(best_h, best_n, best_n / 2.0, rng, max_rounds=60)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
        if budget_s - (time.time() - t_start) > 20:
            h_smooth = lbfgs_optimize(best_h, 200, 1000, best_n / 2.0, maxiter=800)
            c5_sm = compute_c5(h_smooth)
            if c5_sm < best_c5:
                best_c5 = c5_sm
                best_h = h_smooth.copy()

    if budget_s - (time.time() - t_start) > 30:
        n_ls = best_n
        ts_ls = n_ls / 2.0
        h_ls, c5_ls = coordinated_local_search(best_h, n_ls, ts_ls, rng,
                                                max_steps=min(3000, int((budget_s - (time.time() - t_start)) * 5)))
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()
        if budget_s - (time.time() - t_start) > 15:
            h_pol = lbfgs_optimize(best_h, 500, 2000, n_ls / 2.0, maxiter=1000)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    if budget_s - (time.time() - t_start) > 20:
        for alpha in [2000, 8000]:
            if budget_s - (time.time() - t_start) < 15:
                break
            h_pol = lbfgs_optimize(best_h, alpha, 5000, best_n / 2.0, maxiter=1500, gtol=1e-14)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
