Actually, let me reconsider the problem structure. The function we're minimizing is:
f(h) = max_k g_k(h) where g_k(h) = sum_x h[x](1-h[x+k-(n-1)])

Each g_k is a quadratic function of h (specifically, g_k = sum h[x] - sum h[x]h[x+k-(n-1)]). The constraint set is a polytope (box + hyperplane). This is a non-convex optimization problem.

One approach: **alternating minimization**. Fix a subset of "active" constraints (the shifts where the max is achieved), and solve the resulting problem. Then update the active set.

For a fixed set D of active shifts, we want to minimize:
max_{d in D} g_d(h)
subject to 0 ≤ h ≤ 1, sum h = n/2.

This is still non-convex (quadratic objective). But we can use a "water-filling" approach: introduce a variable t and minimize t subject to g_d(h) ≤ t for all d in D. This is a quadratically constrained quadratic program (QCQP), which is non-convex.

Alternatively, we can use the Dinkelbach-type approach or bisection on t.

For a fixed t, the feasibility problem is:
g_d(h) ≤ t for all d in D
0 ≤ h ≤ 1
sum h = n/2

This is still non-convex. But if we relax h[x]h[x'] ≤ min(h[x], h[x']) (which is always true), we get a linear relaxation:
sum_x h[x] - sum_x h[x]h[x+d] ≤ t
≤ sum_x h[x] - sum_x min(h[x], h[x+d]) ≤ t

The lower bound gives: sum h - sum min(h[x], h[x+d]) ≤ t
And sum min(h[x], h[x+d]) ≥ min(h[x]+h[x+d], ... ) hmm this isn't leading anywhere clean.

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def binary_local_search(h, max_sweeps=1000, time_limit=30):
    """Steepest descent local search on binary swaps."""
    n = len(h)
    h = h.copy()
    corr = compute_corr(h)
    max_c = np.max(corr)
    t_start = time.time()
    
    for sweep in range(max_sweeps):
        if time.time() - t_start > time_limit:
            break
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        best_improve = 0
        best_swap = None
        
        # For each (i,j) swap, compute the change in max correlation
        for i in ones:
            for j in zeros:
                # Compute delta corr for this swap
                # Affected k range
                k_min = max(0, n - 1 - max(i, j))
                k_max = min(2*n - 2, n - 1 + max(i, j))
                
                # For each affected k, compute new corr value
                new_max = max_c  # start with current max
                improved = False
                
                for k in range(k_min, k_max + 1):
                    q = i + k - (n - 1)
                    r = j + k - (n - 1)
                    delta = 0.0
                    if 0 <= q < n:
                        # h'[i] changed from 1 to 0
                        # The term involving h[i] in corr[k] was: 1 * (1 - h[q])
                        # Now it's: 0 * (1 - h'[q]) = 0
                        # But wait, if q == j, then h'[q] = 1, so (1-h'[q]) = 0
                        # The change in the h[i] contribution: was (1-h[q]), now 0
                        if q == j:
                            hq_new = 1.0
                        elif q == i:
                            hq_new = 0.0
                        else:
                            hq_new = h[q]
                        delta -= (1.0 - h[q])  # remove old contribution
                        # add new contribution (which is 0 since h'[i]=0)
                        # so net change from this term is -(1-h[q]) + 0 = -(1-h[q])
                        # but we need to be careful: if q==i, h[q] was 1, h'[q]=0
                        # if q==j, h[q] was 0, h'[q]=1
                        # Actually the formula is simpler:
                        # Old: h[i] * (1-h[q]) = 1 * (1-h[q])
                        # New: h'[i] * (1-h'[q]) = 0 * (1-h'[q]) = 0
                        # So change = -(1-h[q])
                        # But if q==i: old = 1*(1-1) = 0, new = 0*(1-0) = 0. change = 0.
                        #   And -(1-h[q]) = -(1-1) = 0. ✓
                        # If q==j: old = 1*(1-0) = 1, new = 0*(1-1) = 0. change = -1.
                        #   And -(1-h[q]) = -(1-0) = -1. ✓
                        # If q∉{i,j}: old = 1*(1-h[q]), new = 0. change = -(1-h[q]). ✓
                    
                    if 0 <= r < n:
                        # h'[j] changed from 0 to 1
                        # Old: h[j] * (1-h[s]) where s = j+k-(n-1) = r
                        #   = 0 * (1-h[r]) = 0
                        # New: h'[j] * (1-h'[r]) = 1 * (1-h'[r])
                        if r == i:
                            hr_new = 0.0
                            delta += (1.0 - hr_new)  # = 1
                        elif r == j:
                            hr_new = 1.0
                            delta += (1.0 - hr_new)  # = 0
                        else:
                            delta += (1.0 - h[r])
                    
                    new_val = corr[k] + delta
                    if new_val > new_max:
                        new_max = new_val
                
                improve = max_c - new_max
                if improve > best_improve + 1e-10:
                    best_improve = improve
                    best_swap = (i, j)
                    # Early exit if we found a big improvement
                    if best_improve > 0.5:  # Just a heuristic
                        break
            if time.time() - t_start > time_limit:
                break
        
        if best_swap is None:
            break
        
        i, j = best_swap
        h[i] = 0
        h[j] = 1
        corr = compute_corr(h)
        max_c = np.max(corr)
    
    return h, float(np.max(compute_corr(h)) * 2.0 / n)

def compute_c5(h):

    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def binary_steepest_descent(h, n, max_sweeps=500, time_limit=60):
    """Steepest descent: find the swap that most reduces max correlation."""
    h = h.copy()
    t_start = time.time()
    
    for sweep in range(max_sweeps):
        if time.time() - t_start > time_limit:
            break
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        current_corr = np.correlate(h, 1.0 - h, mode="full")
        current_max = float(np.max(current_corr))
        
        best_improve = 0
        best_swap = None
        
        # To save time, use first-improvement after a random shuffle
        # Or: evaluate all swaps but break early if improvement is large enough
        
        for i in ones:
            for j in zeros:
                h[i] = 0; h[j] = 1
                new_max = float(np.max(np.correlate(h, 1.0 - h, mode="full")))
                h[i] = 1; h[j] = 0
                
                if current_max - new_max > best_improve:
                    best_improve = current_max - new_max
                    best_swap = (i, j)
        
        if best_swap is None:
            break
        
        i, j = best_swap
        h[i] = 0; h[j] = 1
    
    return h

def simulated_annealing_binary(h, n, t_start_temp=0.5, t_end_temp=0.01, n_steps=50000, time_limit=60):
    """Simulated annealing on binary swaps."""
    h = h.copy()
    t0 = time.time()
    current_corr = np.correlate(h, 1.0 - h, mode="full")
    current_max = float(np.max(current_corr))
    best_h = h.copy()
    best_max = current_max
    
    for step in range(n_steps):
        if time.time() - t0 > time_limit:
            break
        
        progress = step / n_steps
        temp = t_start_temp * (t_end_temp / t_start_temp) ** progress
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = np.random.randint(len(ones))
        j = np.random.randint(len(zeros))
        i = ones[i]
        j = zeros[j]
        
        h[i] = 0; h[j] = 1
        new_max = float(np.max(np.correlate(h, 1.0 - h, mode="full")))
        
        delta = new_max - current_max
        if delta < 0 or np.random.rand() < np.exp(-delta / (temp * max(current_max, 1e-6))):
            current_max = new_max
            if current_max < best_max:
                best_max = current_max
                best_h = h.copy()
        else:
            h[i] = 1; h[j] = 0
    
    return best_h, best_max

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = float(np.max(corr))
    exp_vals = np.exp(alpha * (corr - m_val))
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    
    # Gradient
    # corr[k] = sum_x h[x] * (1-h[x+k-(n-1)])
    # d(corr[k])/d(h[y]) = (1-h[y+k-(n-1)]) * [y valid] + h[y-k+(n-1)] * [y-k+(n-1) valid] * (-1)
    # Wait: corr[k] = sum_x h[x](1-h[x+d]) where d = k-(n-1)
    # d/dh[y]: h[y](1-h[y+d]) contributes (1-h[y+d]) if y+d valid
    #          h[x](1-h[y]) contributes -h[x] if x = y-d, i.e., x = y-k+(n-1)
    
    d = np.arange(2*n-1) - (n-1)  # lag for each k
    
    # For each output y, the gradient is:
    # sum over valid k: w[k] * [(1-h[y+d_k]) * 1_{y+d_k in [0,n-1]} - h[y-d_k] * 1_{y-d_k in [0,n-1]}]
    
    # This is O(n * 2n) = O(n²) which is fine for n=240.
    # But let's vectorize.
    
    # grad[y] = sum_k w[k] * (1-h[y+d_k]) * valid1[y,k] - sum_k w[k] * h[y-d_k] * valid2[y,k]
    # where valid1[y,k] = 1 if 0 ≤ y+d_k < n, valid2[y,k] = 1 if 0 ≤ y-d_k < n
    
    # y+d_k in [0,n-1] means d_k in [-y, n-1-y]
    # y-d_k in [0,n-1] means d_k in [y-(n-1), y]
    
    # Let me compute this with matrix operations
    y_idx = np.arange(n)[:, None]  # (n, 1)
    d_row = d[None, :]  # (1, 2n-1)
    
    yd = y_idx + d_row  # (n, 2n-1): y + d_k
    ym = y_idx - d_row  # (n, 2n-1): y - d_k
    
    valid_yd = (yd >= 0) & (yd < n)  # (n, 2n-1)
    valid_ym = (ym >= 0) & (ym < n)  # (n, 2n-1)
    
    yd_clipped = np.clip(yd, 0, n-1)
    ym_clipped = np.clip(ym, 0, n-1)
    
    term1 = valid_yd * (1.0 - h[yd_clipped])  # (n, 2n-1)
    term2 = valid_ym * h[ym_clipped]  # (n, 2n-1)
    
    grad = term1 @ w - term2 @ w  # (n,)
    
    return obj, grad

def slsqp_optimize(h0, n, target_sum, alpha, maxiter=1000):
    def f(x):
        return smooth_max_and_grad(x, alpha)[0]
    def g(x):
        return smooth_max_and_grad(x, alpha)[1]
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)
