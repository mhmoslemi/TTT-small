import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            if not np.any(mask):
                break
            step = diff / np.sum(mask)
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            mask = h > 1e-15
            if not np.any(mask):
                break
            step = -diff / np.sum(mask)
            h[mask] = np.maximum(h[mask] - step, 0.0)
    return np.clip(h, 0.0, 1.0)

def compute_gradient(h, T):
    """Compute gradient of log-sum-exp of correlation values w.r.t. h."""
    n = len(h)
    dx = 2.0 / n
    ones_minus_h = 1.0 - h
    c = np.correlate(h, ones_minus_h, mode="full") * dx
    # c has length 2*n - 1, indexed from -(n-1) to (n-1)
    c_max = c.max()
    weights = np.exp((c - c_max) / T)
    weights /= weights.sum()
    
    # Gradient of sum_k weights[k] * c[k] w.r.t. h[m]
    # c[k] = sum_i h[i+k] * (1-h[i]) * dx (for the part where indices are valid)
    # d c[k] / d h[m] = (1-h[m-k])*dx [if m-k valid] + (-h[m+k])*dx [if m+k valid]
    
    grad = np.zeros(n)
    # For each lag k (offset from center of full correlation):
    # The full correlation is indexed 0 to 2*n-2, where lag = idx - (n-1)
    # c[idx] = sum_i h[i + lag] * (1-h[i]) where lag = idx - (n-1)
    # For lag >= 0: i ranges from 0 to n-1-lag
    #   d/dh[m] of h[m] in first factor: m = i+lag, i = m-lag, valid if 0 <= m-lag <= n-1-lag
    #   d/dh[m] of (1-h[m]) in second factor: i = m, valid if 0 <= m <= n-1-lag
    # For lag < 0: let lag' = -lag > 0
    #   c = sum_i h[i - lag'] * (1-h[i]), i ranges from lag' to n-1
    #   d/dh[m] of h[m] in first factor: m = i-lag', i = m+lag', valid if lag' <= m+lag' <= n-1, i.e., 0 <= m <= n-1-lag'
    #   d/dh[m] of (1-h[m]) in second factor: i = m, valid if lag' <= m <= n-1
    
    for idx in range(2*n - 1):
        w = weights[idx]
        if w < 1e-15:
            continue
        lag = idx - (n - 1)
        if lag >= 0:
            # h[i+lag] * (1-h[i]), i from 0 to n-1-lag
            # Contribution to grad[m] from first factor (h[m]): m = i+lag, i = m-lag
            #   m ranges from lag to n-1
            #   grad[m] += w * (1-h[m-lag]) * dx
            m_start, m_end = lag, n
            grad[m_start:m_end] += w * (1.0 - h[:n-lag]) * dx
            # Contribution to grad[m] from second factor (1-h[m]): m = i
            #   m ranges from 0 to n-1-lag
            #   grad[m] += w * (-h[m+lag]) * dx
            m_start2, m_end2 = 0, n - lag
            grad[m_start2:m_end2] += w * (-h[lag:]) * dx
        else:
            # lag < 0, let lp = -lag
            lp = -lag
            # h[i-lp] * (1-h[i]), i from lp to n-1
            # Contribution to grad[m] from first factor (h[m]): m = i-lp, i = m+lp
            #   m ranges from 0 to n-1-lp
            #   grad[m] += w * (1-h[m+lp]) * dx
            m_start, m_end = 0, n - lp
            grad[m_start:m_end] += w * (1.0 - h[lp:]) * dx
            # Contribution to grad[m] from second factor (1-h[m]): m = i
            #   m ranges from lp to n-1
            #   grad[m] += w * (-h[m-lp]) * dx
            m_start2, m_end2 = lp, n
            grad[m_start2:m_end2] += w * (-h[:n-lp]) * dx
    
    return grad, c, c_max

def subgradient_optimize(h_init, n, budget_s, t_start):
    """Optimize using subgradient method with log-sum-exp."""
    h = h_init.copy()
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c5 = compute_c5_from_h(best_h)
    
    T = 5.0  # Start with moderate temperature
    T_min = 0.1
    lr = 0.01
    n_iter = 0
    
    while time.time() - t_start < budget_s:
        dx = 2.0 / n
        grad, c, c_max = compute_gradient(h, T)
        
        # Line search
        for trial in range(20):
            h_new = h - lr * grad
            h_new = project_feasible(h_new, n)
            c5_new = compute_c5_from_h(h_new)
            if c5_new < best_c5 + 1e-10:
                break
            lr *= 0.7
        
        c5_new = compute_c5_from_h(h_new)
        if c5_new < best_c5 - 1e-15:
            best_h = h_new.copy()
            best_c5 = c5_new
            n_iter += 1
            # Decrease temperature slowly
            T = max(T_min, T * 0.995)
        else:
            lr *= 0.5
            T = max(T_min, T * 0.99)
        
        h = h_new
    
    return best_h, best_c5

Let me implement this coordinate descent as a final polishing step.

Actually, for the coordinate descent, the optimal x for each coordinate is:
- The point where the maximum of the linear functions is minimized
- This is a 1D minimax problem

For a set of linear functions f_k(x) = a_k + b_k * x, the minmax point is where the "envelope" has a local minimum. This occurs where two lines cross, and the slope changes from negative to positive.

The slope of f_k is b_k. The envelope slope is the max (of the active b_k values) minus... actually it's more subtle.

The envelope g(x) = max_k f_k(x) is convex. Its derivative is:
g'(x) = b_k* where k* is an active index with the largest b_k (for the subdifferential, it's the convex hull of active b_k values).

The minimum of g is where 0 ∈ subdifferential, i.e., where there exist active k1, k2 with b_k1 ≤ 0 ≤ b_k2.

So the optimal x is:
- If all active b_k > 0: x = 0 (or the boundary)
- If all active b_k < 0: x = 1 (or the boundary)
- Otherwise: find the crossing point x* where f_k1(x*) = f_k2(x*) with b_k1 < 0 < b_k2

Actually, the minmax of linear functions is found by:
x* = min { f_k(x) = f_l(x) for all pairs (k,l) } subject to x* being the argmax for both k and l.

This is getting complex. Let me just do a ternary search on x ∈ [0,1] for each coordinate. Since g(x) is convex, ternary search will converge quickly (20 iterations gives precision 1/3^20 ≈ 10^-10).

For each of n coordinates: 20 evaluations of g(x) = max of (2n-1) linear functions. Each evaluation is O(n). Total: n * 20 * n = 20 * n² ≈ 1.5M for n=270. Very fast.

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d

def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            step = diff / cnt
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            mask = h > 1e-12
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            step = -diff / cnt
            h[mask] = np.maximum(h[mask] - step, 0.0)
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, T, topk=None):
    n = len(h)
    dx = 2.0 / n
    g_arr = 1.0 - h
    c = np.correlate(h, g_arr, mode="full") * dx
    cmax = c.max()
    if topk is not None and topk < len(c):
        sel = np.argsort(c)[::-1][:topk]
    else:
        sel = np.arange(len(c))
    c_sub = c[sel]
    c_sub_max = c_sub.max()
    w = np.exp((c_sub - c_sub_max) / T)
    ws = w.sum()
    if ws < 1e-30:
        w = np.ones(len(sel)) / len(sel)
    else:
        w /= ws
    grad = np.zeros(n)
    for i in range(len(sel)):
        wi = w[i]
        if wi < 1e-14:
            continue
        lag = int(sel[i]) - (n - 1)
        if lag >= 0:
            grad[lag:] += wi * (1.0 - h[:n - lag]) * dx
            grad[:n - lag] += wi * (-h[lag:]) * dx
        else:
            lp = -lag
            grad[:n - lp] += wi * (1.0 - h[lp:]) * dx
            grad[lp:] += wi * (-h[:n - lp]) * dx
    return grad

def coord_descent_pass(h, n, max_rounds=3):
    dx = 2.0 / n
    for _ in range(max_rounds):
        changed = False
        for i in range(n):
            old = h[i]
            full_c = np.correlate(h, 1.0 - h, mode="full") * dx
            d_li = np.zeros(2 * n - 1)
            for li in range(2 * n - 1):
                j = i - li + (n - 1)
                if 0 <= j < n:
                    d_li[li] += (1.0 - h[j]) * dx
                j2 = i + li - (n - 1)
                if 0 <= j2 < n:
                    d_li[li] -= h[j2] * dx
            lo, hi = 0.0, 1.0
            for _ts in range(40):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1 = np.max(full_c + (m1 - old) * d_li)
                f2 = np.max(full_c + (m2 - old) * d_li)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
            best_x = (lo + hi) / 2.0
            new_c5 = np.max(full_c + (best_x - old) * d_li)
            cur_c5 = float(np.max(full_c))
            if new_c5 < cur_c5 - 1e-15:
                h[i] = best_x
                changed = True
        if not changed:
            break
    return h

def subgrad_optimize(h_init, n, t0, time_limit):
