    for rnd in range(num_rounds):
        if elapsed_fn() > time_limit:
            break
        corr = compute_corr_scaled(best)
        c_max = np.max(corr)
        sorted_idx = np.argsort(corr)[::-1]
        hot_lags = sorted_idx[:8]
        hot_vars = set()
        for m in hot_lags:
            k = m - (n - 1)
            if k >= 0:
                hot_vars.update(range(max(0, k), n))
                hot_vars.update(range(0, n - k))
            else:
                j = -k
                hot_vars.update(range(0, n - j))
                hot_vars.update(range(j, n))
        hot_list = list(hot_vars)
        if len(hot_list) == 0:
            hot_list = list(range(n))
        n_pert = np.random.randint(3, 9)
        idx = np.random.choice(len(hot_list), min(n_pert, len(hot_list)), replace=False)
        pert_vars = [hot_list[i] for i in idx]
        current = best.copy()
        for vi in pert_vars:
            current[vi] += np.random.randn() * 0.04
        current = project_feasible(current, n)
        c5_try = evaluate_c5(current)
        if c5_try < best_c5:
            best = current
            best_c5 = c5_try
    return best, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    elapsed_fn = lambda: time.time() - start_time

    parent_h = None
    try:
        parent_h = np.array(initial_h_values, dtype=float)
    except Exception:
        pass

    if parent_h is not None:
        n = len(parent_h)
        best_h = project_feasible(parent_h.copy(), n)
    else:
        n = 77
        x = np.linspace(0, 2, n, endpoint=False)
        best_h = project_feasible(0.5 + 0.3 * np.sin(2 * np.pi * 2.0 * x / 2.0), n)

    best_c5 = evaluate_c5(best_h)

    p1_end = start_time + budget_s * 0.12
    p2_end = start_time + budget_s * 0.45
    p3_end = start_time + budget_s * 0.70
    p4_end = start_time + budget_s * 0.90
    p5_end = start_time + budget_s * 0.98

    # Phase 1: Subgradient descent from parent and variants
    if time.time() < p1_end:
        best_h, best_c5 = subgradient_descent(best_h, n, elapsed_fn, p1_end * 0.6)

        if time.time() < p1_end:
            for trial in range(8):
                if time.time() > p1_end:
                    break
                x = np.linspace(0, 2, n, endpoint=False)
                freq = 1.5 + trial * 0.4
                phase = trial * 0.7
                h_try = project_feasible(0.5 + 0.35 * np.sin(2 * np.pi * freq * x / 2.0 + phase), n)
                h_try, c5_try = subgradient_descent(h_try, n, elapsed_fn, p1_end)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = h_try

    # Phase 2: Extended subgradient with restarts
    if time.time() < p2_end:
        for restart in range(12):
            t_remaining = p2_end - time.time()
            if t_remaining < 30:
                break
            frac = (time.time() - start_time) / budget_s
            noise = 0.005 + 0.02 * max(0, 1.0 - frac)
            start_pt = project_feasible(best_h + np.random.randn(n) * noise, n)
            h_try, c5_try = subgradient_descent(start_pt, n, elapsed_fn, min(time.time() + t_remaining / 14, p2_end))
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try

    # Phase 3: Targeted SA + general SA
    if time.time() < p3_end:
        frac3 = (time.time() - start_time) / budget_s
        t3 = p3_end - time.time()
        n_iters_sa = max(5000, int(t3 * 3000))
        best_h, best_c5 = simulated_annealing(best_h, n, n_iters_sa, 0.04, 0.002, elapsed_fn, p3_end * 0.5)
        if time.time() < p3_end:
            best_h, best_c5 = targeted_sa(best_h, n, elapsed_fn, p3_end, num_rounds=200)
        if time.time() < p3_end:
            best_h, best_c5 = subgradient_descent(best_h, n, elapsed_fn, p3_end)

    # Phase 4: Coordinate descent + fine SA
    if time.time() < p4_end:
        h_cd, c5_cd = coordinate_descent(best_h, n, sweeps=6, elapsed_fn=elapsed_fn, time_limit=p4_end * 0.6)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd
        if time.time() < p4_end:
            n_iters = max(3000, int((p4_end - time.time()) * 4000))
            best_h, best_c5 = simulated_annealing(best_h, n, n_iters, 0.02, 0.001, elapsed_fn, p4_end * 0.8)
        if time.time() < p4_end:
            h_cd2, c5_cd2 = coordinate_descent(best_h, n, sweeps=4, elapsed_fn=elapsed_fn, time_limit=p4_end)
            if c5_cd2 < best_c5:
                best_c5 = c5_cd2
                best_h = h_cd2

    # Phase 5: Final subgradient + coordinate polish
    if time.time() < p5_end:
        best_h, best_c5 = subgradient_descent(best_h, n, elapsed_fn, p5_end * 0.7)
        if time.time() < p5_end:
            h_cd3, c5_cd3 = coordinate_descent(best_h, n, sweeps=4, elapsed_fn=elapsed_fn, time_limit=p5_end)
            if c5_cd3 < best_c5:
                best_c5 = c5_cd3
                best_h = h_cd3

    # Final: try SLSQP for a few iterations as absolute final polish
    if time.time() < start_time + budget_s * 0.995:
        def obj_slsqp(x):
            return evaluate_c5(x)
        def con_slsqp(x):
            return np.sum(x) - n / 2.0
        try:
            res = minimize(obj_slsqp, best_h, method='SLSQP',
                          constraints=[{'type': 'eq', 'fun': con_slsqp}],
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': 200, 'ftol': 1e-14})
            h_sl = project_feasible(res.x, n)
            c5_sl = evaluate_c5(h_sl)
            if c5_sl < best_c5:
                best_c5 = c5_sl
                best_h = h_sl
        except Exception:
            pass

    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, n
