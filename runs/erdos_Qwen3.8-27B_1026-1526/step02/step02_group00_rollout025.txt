Let me now think about the overall algorithm:

1. Start from initial_h_values (n=240)
2. Do a few rounds of continuous optimization (L-BFGS-B with log-sum-exp)
3. Project to binary (threshold at 0.5, fix sum)
4. Do binary local search with efficient swap delta computation
5. Relax back to continuous
6. Repeat steps 2-5 a few times
7. Final polish

Let me also think about what "different" approach to take. The strategy mentions Bayesian optimization and FFT. Let me think about whether FFT-based methods can help.

**FFT-based approach:**
The correlation corr = np.correlate(h, 1-h, mode="full") can be computed via FFT in O(n log n) instead of O(n²). For n=240, this is 240*8 ≈ 1920 operations vs 240*240 = 57600. But the savings are modest for n=240.

More importantly, the gradient computation can also be done via FFT:
grad[i] = Σ_k w[k] * [(1-h[s_k(i)]) * 1[i is first for k] - h[f_k(i)] * 1[i is second for k]]

This is a convolution-like operation that can be done via FFT.

But for n=240, the direct computation is fast enough. The bottleneck is the number of optimization iterations, not the per-iteration cost.

Let me focus on making the optimization more effective:

1. **Adaptive α**: Start with small α (exploratory) and increase (exploitation).
2. **Active set tracking**: After convergence with a given α, identify the active shifts (those with corr close to max). For the next iteration, only use those shifts in the objective.
3. **Continuation method**: Solve at n/2, then use that as starting point for n.

Let me also consider a completely different approach: **Projected Gradient Method with Barzilai-Borwein step size**.

The BBO (Barzilai-Borwein) step size is:
α_k = ||s_{k-1}||² / (s_{k-1}^T y_{k-1}) or ||y_{k-1}||² / (s_{k-1}^T y_{k-1})

This often outperforms fixed step sizes for gradient methods.

But the issue is that the max function is not differentiable everywhere, so the gradient might not be well-defined. The log-sum-exp approximation helps with this.

Let me now write the full solution:

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

# Global parameters for optimization
_g_alpha = 100.0
_g_target_sum = 0.0
_g_penalty = 500.0

def _obj(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = _g_alpha * (corr - m_val)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    if s < 1e-300:
        s = 1e-300
    obj_val = m_val + np.log(s) / _g_alpha
    sum_viol = np.sum(h) - _g_target_sum
    obj_val += _g_penalty * sum_viol * sum_viol
    return obj_val

def _grad(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = _g_alpha * (corr - m_val)
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    cs_val = (sc := np.clip(second_idx, 0, n - 1))
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[cs_val] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    
    sum_viol = np.sum(h) - _g_target_sum
    grad += 2.0 * _g_penalty * sum_viol
    return grad

def swap_delta(h, n, a, b):
    """Compute delta in corr when h[a]: 1->0 and h[b]: 0->1."""
    delta = np.zeros(2 * n - 1)
    
    # a: 1->0, as first index
    ks = max(0, n - 1 - a)
    ke = min(2 * n - 2, 2 * n - 2 - a)
    if ks <= ke:
        s_vals = a + np.arange(ks, ke + 1) - n + 1
        delta[ks:ke + 1] -= (1.0 - h[s_vals])
    
    # a: 1->0, as second index  
    ks = max(0, a)
    ke = min(2 * n - 2, a + n - 1)
    if ks <= ke:
        f_vals = a - np.arange(ks, ke + 1) + n - 1
        valid = (f_vals >= 0) & (f_vals < n)
        if np.any(valid):
            k_valid = np.arange(ks, ke + 1)[valid]
            delta[k_valid] += h[f_vals[valid]]
    
    # b: 0->1, as first index
    ks = max(0, n - 1 - b)
    ke = min(2 * n - 2, 2 * n - 2 - b)
    if ks <= ke:
        s_vals = b + np.arange(ks, ke + 1) - n + 1
        delta[ks:ke + 1] += (1.0 - h[s_vals])
    
    # b: 0->1, as second index
    ks = max(0, b)
    ke = min(2 * n - 2, b + n - 1)
    if ks <= ke:
        f_vals = b - np.arange(ks, ke + 1) + n - 1
        valid = (f_vals >= 0) & (f_vals < n)
        if np.any(valid):
            k_valid = np.arange(ks, ke + 1)[valid]
            delta[k_valid] -= h[f_vals[valid]]
    
    return delta

def binary_local_search(h, n, target_sum, rng, max_rounds=50, trials_per_round=200):
    h = h.copy()

    max_corr = np.max(corr)
    
    for rnd in range(max_rounds):
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        best_delta_max = 1e-12
        best_a, best_b = -1, -1
        n_trials = min(trials_per_round, len(ones) * len(zeros))
        
        for _ in range(n_trials):
            a = ones[rng.randint(len(ones))]
            b = zeros[rng.randint(len(zeros))]
            d = swap_delta(h, n, a, b)
            new_corr = corr + d
            new_max = np.max(new_corr)
            if new_max - max_corr < best_delta_max:
                best_delta_max = new_max - max_corr
                best_a, best_b = a, b
                best_new_corr = new_corr.copy()
        
        if best_a < 0:
            break
        
        h[best_a] = 0.0
        h[best_b] = 1.0
        corr = best_new_corr
        max_corr = np.max(corr)
    
    return h, float(np.max(np.correlate(h, 1.0 - h, mode="full")) * 2.0 / n)

def make_binary_from_continuous(h, n, target_sum, rng):
    """Convert continuous h to binary with correct sum."""
    h_bin = (h > 0.5).astype(float)
    # If not exactly the right sum, adjust
    cur_sum = h_bin.sum()
    if cur_sum > target_sum:
        # Need to flip some 1s to 0s
        idx = np.where(h_bin == 1.0)[0]
        # Sort by h value (lowest first - these are the "least committed" 1s)
        order = np.argsort(h[idx])
        n_flip = int(cur_sum - target_sum)
        for i in range(min(n_flip, len(order))):
            h_bin[idx[order[i]]] = 0.0

        n_flip = int(target_sum - cur_sum)
        for i in range(min(n_flip, len(order))):
            h_bin[idx[order[i]]] = 1.0
    return h_bin
