        h = h_new
        t_k = t_new

        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
            n_same = 0
            lr = min(lr * 1.02, 0.5)

                lr *= 0.9

        h_opt, c5_opt = nesterov_grad(h_cur, n, ts, alpha, 5000, lr, tb)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        h_cur = h_opt
    if remaining() > 100:
        h_ls, c5_ls = targeted_local_search(h_cur, n, ts, 60, rng, min(80, remaining() - 20))
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()
        h_cur = h_ls
    if remaining() > 80:
        h_bs, c5_bs = binary_local_search(h_cur, n, ts, 30, rng, min(60, remaining() - 20))
        if c5_bs < best_c5:
            best_c5 = c5_bs
            best_h = h_bs.copy()
        h_cur = h_bs
    for n_new in [420, 480, 540, 600, 720]:
        if remaining() < 120:
            break
        ts_new = float(n_new) / 2.0
        if abs(ts_new - round(ts_new)) > 1e-9:
            continue
        h_interp = interpolate_h(best_h, n_new)
        h_interp = project_l2(h_interp, ts_new)
        c5_interp = compute_c5(h_interp)
        h_cur_new = h_interp.copy()
        for alpha in [30.0, 100.0, 400.0, 1200.0, 4000.0, 12000.0]:
            if remaining() < 60:
                break
            tb = min(40.0, remaining() * 0.1)
            lr = 0.3 / max(alpha, 1.0)
            h_opt, c5_opt = nesterov_grad(h_cur_new, n_new, ts_new, alpha, 4000, lr, tb)
            if c5_opt < c5_interp:
                c5_interp = c5_opt
                h_cur_new = h_opt
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_new
        if remaining() > 50:
            h_ls2, c5_ls2 = targeted_local_search(h_cur_new, n_new, ts_new, 40, rng,
                                                   min(40, remaining() - 20))
            if c5_ls2 < best_c5:
                best_c5 = c5_ls2
                best_h = h_ls2.copy()
                best_n = n_new
            h_cur_new = h_ls2
        if remaining() > 40:
            h_bs2, c5_bs2 = binary_local_search(h_cur_new, n_new, ts_new, 20, rng,
                                                 min(30, remaining() - 20))
            if c5_bs2 < best_c5:
                best_c5 = c5_bs2
                best_h = h_bs2.copy()
                best_n = n_new
    if best_h is not None and remaining() > 30:
        n_f = len(best_h)
        ts_f = float(n_f) / 2.0
        for alpha in [3000.0, 8000.0, 20000.0]:
            if remaining() < 15:
                break
            h_pol, c5_pol = nesterov_grad(best_h, n_f, ts_f, alpha, 2000,
                                           0.1 / alpha, min(20, remaining() - 5))
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()
        if remaining() > 20:
            h_ls3, c5_ls3 = targeted_local_search(best_h, n_f, ts_f, 40, rng,
                                                   min(15, remaining() - 5))
            if c5_ls3 < best_c5:
                best_c5 = c5_ls3
                best_h = h_ls3.copy()
        if remaining() > 15:
            h_bs3, c5_bs3 = binary_local_search(best_h, n_f, ts_f, 20, rng,
                                                 min(10, remaining() - 3))
            if c5_bs3 < best_c5:
                best_c5 = c5_bs3
                best_h = h_bs3.copy()
    n_final = len(best_h)
    ts_final = float(n_final) / 2.0
    best_h = project_l2(best_h, ts_final)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_final)
