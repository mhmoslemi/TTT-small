def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return float(np.max(corr) * dx)

def power_max(values, p):
    values = np.maximum(values, 0)
    return np.power(np.sum(np.power(values, p)), 1.0/p)

def smooth_max_soft(values, tau):
    v = values / tau
    vmax = np.max(v)
    return tau * (vmax + np.log(np.sum(np.exp(v - vmax))))

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    # Bisection to find the right shift
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        if np.sum(proj) > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-14:
        # Distribute the difference to elements closest to 0.5
        idx = np.argsort(np.abs(result - 0.5))
        for i in idx:
            adj = min(abs(diff), abs(1.0 - result[i]) if diff > 0 else abs(result[i]))
            result[i] += np.sign(diff) * min(adj, abs(diff))
            diff -= np.sign(diff) * min(adj, abs(diff))
            if abs(diff) < 1e-14:
                break
    return result

    h = np.full(n, 0.5)
    for k in range(1, K+1):
        a = params[2*(k-1)]
        b = params[2*(k-1)+1]
        h += a * np.cos(np.pi * k * x) + b * np.sin(np.pi * k * x)
    return project_feasible(h, n)

    dx = 2.0 / n

    dx = 2.0 / n
    g = 1.0 - h

def coordinate_descent_adaptive(h, n, max_sweeps=20, time_limit=None, start_time=None):
    best = h.copy()
    best_c5 = evaluate_c5(best)
    
    for sweep in range(max_sweeps):
        if time_limit is not None and time.time() - start_time > time_limit:
            break
        improved = False
        # Adaptive: try both directions with decreasing step sizes
        base_steps = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
        for i in range(n):
            if time_limit is not None and time.time() - start_time > time_limit:
                break
            for direction in [1, -1]:
                for step in base_steps:
                    delta = direction * step
                    new_val = best[i] + delta
                    if new_val < 0 or new_val > 1:
                        continue
                    h_new = best.copy()
                    h_new[i] = new_val
                    # Adjust one other coordinate to maintain sum
                    # Find coordinate closest to 0.5 (most flexible)
                    diff = best[i] - new_val
                    if abs(diff) < 1e-15:
                        continue
                    # We need to adjust another coordinate by -diff
                    # Find the best coordinate to adjust
                    candidates = [j for j in range(n) if j != i]
                    # Try the coordinate closest to 0.5
                    closest = min(candidates, key=lambda j: abs(best[j] - 0.5))
                    adj = -diff
                    h_new[closest] = best[closest] + adj
                    if h_new[closest] < 0 or h_new[closest] > 1:
                        # Try other candidates
                        found = False
                        for j in sorted(candidates, key=lambda j: abs(best[j] - 0.5)):
                            if j == i:

                            h_try = best.copy()
                            h_try[i] = new_val
                            h_try[j] = best[j] + adj
                            if 0 <= h_try[j] <= 1:
                                h_new = h_try
                                found = True
                                break
                        if not found:

                    c5 = evaluate_c5(h_new)
                    if c5 < best_c5 - 1e-12:
                        best = h_new
                        best_c5 = c5
                        improved = True
                        break  # Move to next coordinate
        if not improved:
            break
    return best, best_c5

def project_single_change(h, i, new_val, n):
    """Fast projection when only coordinate i changes."""
    target = n / 2.0
    old_sum = np.sum(h)
    new_sum = old_sum + (new_val - h[i])
    diff = target - new_sum
    
    # Adjust the coordinate closest to 0.5 (excluding i)
    h_new = h.copy()
    h_new[i] = np.clip(new_val, 0, 1)
    
    if abs(diff) < 1e-12:
        return h_new
    
    # Find best coordinate to adjust
    mask = np.ones(n, dtype=bool)
    mask[i] = False
    candidates = np.where(mask)[0]
    # Sort by distance to 0.5
    order = np.argsort(np.abs(h_new[candidates] - 0.5))
    
    for idx in order:
        j = candidates[idx]
        adj = diff
        # Can we adjust j by adj?
        new_j = h_new[j] + adj
        if 0 <= new_j <= 1:
            h_new[j] = new_j
            diff = 0
            break
        else:
            # Adjust as much as possible
            max_adj = 1.0 - h_new[j] if adj > 0 else h_new[j]
            h_new[j] += np.sign(adj) * max_adj
            diff -= np.sign(adj) * max_adj
    
    if abs(diff) > 1e-12:
        # Use full bisection as fallback
        h_new = project_feasible(h_new, n)
    
    return h_new

Now, the question is: will coordinate descent with exact max actually improve upon the SLSQP solution?

The answer is: it depends on whether SLSQP converged to a true local optimum of the exact max problem. The smooth max approximation means SLSQP is optimizing a slightly different objective. The coordinate descent on the exact max can potentially find a better solution.

However, the exact max is non-smooth (it's the max of many linear functions). At the optimum, only the "active" shifts (those achieving the max) matter. The coordinate descent will naturally focus on reducing the active shifts.

One potential issue: the coordinate descent might get stuck at a point where no single coordinate improvement helps, but a coordinated change of multiple coordinates would. This is the classic limitation of coordinate descent.

To mitigate this, I could:
1. Use "pair" coordinate descent (update two coordinates at once)
2. Add random perturbations between sweeps
3. Use a "shaking" approach (accept worse solutions temporarily)
