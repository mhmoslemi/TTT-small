import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(values, tau):
    v = values / tau
    vmax = np.max(v)
    return tau * (vmax + np.log(np.sum(np.exp(v - vmax))))

def objective_smooth(x, n, tau):
    dx = 2.0 / n
    g = 1.0 - x
    corr = np.correlate(x, g, mode="full")
    return smooth_max(corr, tau) * dx

def sum_constraint(x, n):
    return np.sum(x) - n / 2.0

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float, copy=True)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-15:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        if np.sum(proj) > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-14:
        idx = int(np.argmin(np.abs(result - 0.5)))
        result[idx] = np.clip(result[idx] + diff, 0.0, 1.0)
    return result

def transfer_ls(h, n, max_sweeps, deadline, rng, step_fracs):
    best = h.copy()
    best_c5 = evaluate_c5(best)
    for sweep in range(max_sweeps):
        if time.time() >= deadline:
            break
        improved = False
        for i in range(n):
            for j in range(n):
                if i == j:
                    continue
                max_d_fwd = min(best[i], 1.0 - best[j])
                if max_d_fwd > 1e-8:
                    for sf in step_fracs:
                        d = max_d_fwd * sf
                        hn = best.copy()
                        hn[i] -= d
                        hn[j] += d
                        c5 = evaluate_c5(hn)
                        if c5 < best_c5 - 1e-13:
                            best = hn
                            best_c5 = c5
                            improved = True
                            break
                max_d_bwd = min(best[j], 1.0 - best[i])
                if max_d_bwd > 1e-8:
                    for sf in step_fracs:
                        d = max_d_bwd * sf
                        hn = best.copy()
                        hn[i] += d
                        hn[j] -= d
                        c5 = evaluate_c5(hn)
                        if c5 < best_c5 - 1e-13:
                            best = hn
                            best_c5 = c5
                            improved = True
                            break
        if not improved:
            break
    return best, best_c5

def transfer_ls_random(h, n, max_iter, deadline, rng, step_fracs):
    best = h.copy()
    best_c5 = evaluate_c5(best)
    count = 0
    stall = 0
    while count < max_iter and time.time() < deadline:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            count += 1
            continue
        direction = rng.randint(2)
        if direction == 0:
            max_d = min(best[i], 1.0 - best[j])
        else:
            max_d = min(best[j], 1.0 - best[i])
        if max_d < 1e-8:
            count += 1
            continue
        sf = step_fracs[rng.randint(len(step_fracs))]
        d = max_d * sf

            hn[i] -= d
            hn[j] += d

            best_c5 = c5
            stall = 0
        else:
            stall += 1

        if stall > 500:
            break
    return best, best_c5

def slsqp_refine(h0, n, tau_list, max_iter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint, 'args': (n,)}]
    current = project_feasible(h0.copy(), n)
    for tau in tau_list:
        try:
            res = minimize(
                objective_smooth, current, method='SLSQP',
                args=(n, tau), bounds=bounds, constraints=cons,
                options={'maxiter': max_iter, 'ftol': 1e-14}
            )
            current = project_feasible(res.x, n)
        except Exception:
            pass
    return current, evaluate_c5(current)

def make_starts(n, parent, rng, count):

    x = np.linspace(0, 2, n, endpoint=False)

        h = np.sort(rng.rand(n))

    return starts[:count]
