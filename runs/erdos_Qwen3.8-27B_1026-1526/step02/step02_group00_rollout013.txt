import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m_val))
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    s_all = np.arange(2 * n - 1) - (n - 1)
    j_all = np.arange(n)
    grad = np.zeros(n)
    for j in range(n):
        g1 = s_all >= (-j)
        g1 &= (j + s_all) < n
        g2 = (j - s_all) >= 0
        g2 &= (j - s_all) < n
        grad[j] = np.sum(w[g1] * (1.0 - h[j + s_all[g1]])) - np.sum(w[g2] * h[j - s_all[g2]])
    return obj, grad

def smooth_max_and_grad_fast(h, alpha):

    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m_val))

    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    a_shifted = w
    grad = np.zeros(n)
    w1 = 1.0 - h
    for j in range(n):
        lo1 = max(0, j)
        hi1 = min(n, j + n)
        s_lo1 = lo1 - j
        s_hi1 = hi1 - j
        m_lo = s_lo1 + n - 1
        m_hi = s_hi1 + n - 1 - 1
        if m_hi >= m_lo:
            idx_f = np.arange(m_lo, m_hi + 1)
            idx_h = j - s_lo1 + np.arange(m_hi - m_lo + 1)
            if np.max(idx_h) < n and np.min(idx_h) >= 0:
                grad[j] += np.sum(a_shifted[idx_f] * w1[idx_h])
        lo2 = max(0, j)
        hi2 = min(n, j + n)
        s_lo2 = lo2 - j
        s_hi2 = hi2 - j
        m_lo2 = s_lo2 + n - 1
        m_hi2 = s_hi2 + n - 1 - 1
        if m_hi2 >= m_lo2:
            idx_f2 = np.arange(m_lo2, m_hi2 + 1)
            idx_h2 = j - (s_lo2) + np.arange(m_hi2 - m_lo2 + 1)
            if np.max(idx_h2) < n and np.min(idx_h2) >= 0:
                grad[j] -= np.sum(a_shifted[idx_f2] * h[idx_h2])
    return obj, grad

def slsqp_optimize(h0, target_sum, alpha, maxiter, ftol=1e-12):
    n = len(h0)
    def f(x):
        return smooth_max_and_grad(x, alpha)[0]
    def g(x):
        return smooth_max_and_grad(x, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def lbfgs_penalty(h0, target_sum, alpha, maxiter, lam=500.0):
    n = len(h0)
    def f(x):
        val, _ = smooth_max_and_grad(x, alpha)
        pen = lam * (np.sum(x) - target_sum) ** 2
        return val + pen
    def g(x):
        _, gr = smooth_max_and_grad(x, alpha)
        gr = gr + 2.0 * lam * (np.sum(x) - target_sum)
        return gr
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='L-BFGS-B', jac=g,
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

            if time_left() < 30:
                break
            h_opt = lbfgs_penalty(best_h, ts_main, alpha, maxiter=300, lam=1000.0)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Phase 4: Binary search from best continuous
    if time_left() > 100:
        h_bin2, c5_bin2 = binary_sa_search(best_h, ts_main, rng, time_limit=60, max_iter=1500)
        if c5_bin2 < best_c5:
            best_c5 = c5_bin2
            best_h = h_bin2.copy()
            h_opt = slsqp_optimize(best_h, ts_main, 2000, maxiter=800)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Phase 5: Upsample to higher resolution and optimize
    if time_left() > 200:
        n_up = n_main * 2
        ts_up = n_up / 2.0
        h_up = upsample(best_h, n_up)
        h_up = project_to_sum(h_up, ts_up)
        c5_up = compute_c5(h_up)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up.copy()
            best_n = n_up
            ts_main = ts_up
        
        for alpha in [20, 100, 500, 2000, 10000]:
            if time_left() < 30:
                break
            h_opt = slsqp_optimize(best_h, ts_main, alpha, maxiter=600)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                ts_main = len(best_h) / 2.0

        if time_left() > 80:
            h_bin3, c5_bin3 = binary_sa_search(best_h, ts_main, rng, time_limit=50, max_iter=1000)
            if c5_bin3 < best_c5:
                best_c5 = c5_bin3
                best_h = h_bin3.copy()
            h_opt = slsqp_optimize(best_h, ts_main, 5000, maxiter=1000)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Phase 6: Shake and re-optimize
    if time_left() > 60:
        for trial in range(8):
            if time_left() < 20:
                break
            noise = 0.08 + 0.04 * (1.0 - trial / 8.0)
            h_shake = best_h + rng.uniform(-noise, noise, len(best_h))
            h_shake = project_to_sum(h_shake, ts_main)
            for alpha in [100, 1000, 5000]:
                if time_left() < 15:
                    break
                h_opt = slsqp_optimize(h_shake, ts_main, alpha, maxiter=500)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                h_shake = h_opt

    # Phase 7: Final high-alpha polish
    if time_left() > 30:
        for alpha in [5000, 20000, 50000]:
            if time_left() < 15:
                break
            h_pol = slsqp_optimize(best_h, ts_main, alpha, maxiter=2000)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Final validation
    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
