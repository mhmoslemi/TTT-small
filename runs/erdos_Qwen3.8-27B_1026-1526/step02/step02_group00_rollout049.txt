def compute_c5(h):

    return float(np.max(corr) * dx)

def project_box_sum(h, target_sum):
    n = len(h)

        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")

def nesterov_momentum(h0, target_sum, alpha, lr0, max_iter, momentum=0.9):

    best_obj = smooth_max_obj_grad(h, alpha)[0]

        if it % 100 == 0:
            obj, _ = smooth_max_obj_grad(h, alpha)
            if obj < best_obj:

def slsqp_optimize(h0, target_sum, alpha, maxiter=500):
    n = len(h0)
    
    def fun(x):
        return smooth_max_obj_grad(x, alpha)[0]
    
    def jac(x):
        return smooth_max_obj_grad(x, alpha)[1]
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(fun, h0.copy(), method='SLSQP', jac=jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_opt = project_box_sum(res.x, target_sum)
        return h_opt
    except Exception:
        return project_box_sum(h0, target_sum)

            best_h = h_nm.copy()
    
    # SLSQP continuation at current resolution
    for alpha in [50, 200, 1000, 3000]:
        if time_left() < 20:
            break
        h_opt = slsqp_optimize(best_h, ts1, alpha, maxiter=500)
        c5_opt = compute_c5(h_opt)
