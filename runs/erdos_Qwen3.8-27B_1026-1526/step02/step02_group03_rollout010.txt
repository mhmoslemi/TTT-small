def project_onto_feasible(h, target_sum):
    """Project h onto {0<=x<=1, sum(x)=target_sum}"""
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-14:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5*(lo+hi), 0.0, 1.0)
    diff = target_sum - result.sum()
    if abs(diff) > 1e-12:
        # distribute among entries closest to 0.5
        dists = np.abs(result - 0.5)
        order = np.argsort(dists)
        remaining = abs(diff)
        sign = np.sign(diff)
        for idx in order:
            room = (1.0 - result[idx]) if sign > 0 else result[idx]
            adj = min(remaining, room)
            result[idx] += sign * adj
            remaining -= adj
            if remaining < 1e-14:
                break
    return result

        lr_cur = lr * (0.97 ** (it // 30))
        lr_cur = max(lr_cur, 0.001)
        cur_new = project_feasible(cur - lr_cur * grad, n / 2.0)
        c5_new = evaluate_c5(cur_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = cur_new.copy()
        elif c5_new > evaluate_c5(cur) + 1e-10:
            lr_cur *= 0.5
            cur_new = project_feasible(cur - lr_cur * grad, n / 2.0)
        cur = cur_new
    return best_h, best_c5

def generate_structured_starts(n, rng, num_starts=12):
    starts = []
    x = np.linspace(0, 2.0, n, endpoint=False)
    ar = np.arange(n, dtype=float)
    for freq in [1.0, 1.5, 2.0, 2.5, 3.0, 3.5, 4.0, 5.0]:
        for _ in range(2):
            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.25, 0.48)
            h = 0.5 + amp * np.sin(2.0 * np.pi * freq * x / 2.0 + phase)
            starts.append(np.clip(h, 0, 1))
    for _ in range(3):
        c = rng.uniform(0.2 * n, 0.8 * n)
        w = rng.uniform(0.1 * n, 0.4 * n)
        h = np.exp(-0.5 * ((ar - c) / w) ** 2)
        h = 0.1 * h + 0.4 * (1 - h)
        starts.append(np.clip(h, 0, 1))
    for _ in range(2):
        f1 = rng.uniform(1.0, 3.0)
        f2 = rng.uniform(3.0, 6.0)
        p1 = rng.uniform(0, 2 * np.pi)
        p2 = rng.uniform(0, 2 * np.pi)
        a1 = rng.uniform(0.15, 0.3)
        a2 = rng.uniform(0.1, 0.25)
        h = 0.5 + a1 * np.sin(2 * np.pi * f1 * x / 2.0 + p1) + a2 * np.sin(2 * np.pi * f2 * x / 2.0 + p2)
        starts.append(np.clip(h, 0, 1))
    for _ in range(2):
        h = rng.uniform(0.1, 0.9, n)
        kernel = np.exp(-np.arange(-3, 4) ** 2 / 4.0)
        kernel /= kernel.sum()
        h = np.convolve(h, kernel, mode='same')
        starts.append(np.clip(h, 0, 1))
    h_tri = np.abs(np.sin(np.pi * ar / n * 3))
    starts.append(np.clip(0.2 + 0.6 * h_tri, 0, 1))
    h_saw = (ar % 7) / 7.0
    starts.append(np.clip(0.15 + 0.7 * h_saw, 0, 1))
    return starts[:num_starts]

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    time_limit = [t0]
    best_h = None
    best_c5 = float('inf')

    parent_h = None
    try:
        parent_h = np.array(initial_h_values, dtype=float)
    except Exception:
        pass

    n_parent = len(parent_h) if parent_h is not None else 77

    # Phase 1: Multi-start optimization (35% time)
    t_phase1 = t0 + budget_s * 0.35
    n_vals = [n_parent]
    for nn in [77, 79, 81, 83, 85, 91, 93, 99, 101]:
        if nn not in n_vals and nn >= 41:
            n_vals.append(nn)
    n_vals = [nn for nn in n_vals if nn <= 101]
    n_vals.sort()

    if parent_h is not None:
        best_h = project_feasible(parent_h.copy(), n_parent / 2.0)
        best_c5 = evaluate_c5(best_h)

    for n in n_vals:
        if time.time() > t_phase1:
            break
        t_per_n = max(20.0, (t_phase1 - time.time()) / max(1, len(n_vals) - n_vals.index(n)))
        starts = []
        if parent_h is not None and n == n_parent:
            starts.append(parent_h.copy())
            starts.append(np.clip(parent_h + rng.randn(n) * 0.05, 0, 1))
            starts.append(np.clip(parent_h + rng.randn(n) * 0.12, 0, 1))
        elif parent_h is not None:
            xp = np.linspace(0, 2, len(parent_h), endpoint=False)
            xn = np.linspace(0, 2, n, endpoint=False)
            f_interp = np.interp(xn, xp, parent_h, period=2.0)
            starts.append(np.clip(f_interp, 0, 1))
            starts.append(np.clip(f_interp + rng.randn(n) * 0.08, 0, 1))
        if n == n_parent or parent_h is None:
            for h_s in generate_structured_starts(n, rng, num_starts=8):
                starts.append(h_s)
        starts.append(np.full(n, 0.5))
        starts.append(np.linspace(0.05, 0.95, n))
        starts.append(np.linspace(0.95, 0.05, n))

        n_starts = len(starts)
        t_per_start = max(5.0, t_per_n * 0.7 / max(1, n_starts))
        for h0 in starts:
            if time.time() > t_phase1:
                break
            h_opt, c5_opt = projected_gradient_opt(h0, n, t_per_start, time_limit)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Phase 2: Refine best with active-set gradient + coordinate descent (30% time)
    t_phase2 = t0 + budget_s * 0.65
    if best_h is not None and time.time() < t_phase2:
        n_best = len(best_h)
        for round_i in range(30):
            if time.time() > t_phase2:
                break
            h_a, c5_a = active_set_gradient(best_h, n_best, tau=0.003, lr=0.02,
                                            max_iter=300, time_limit_end=t_phase2)
            if c5_a < best_c5:
                best_c5 = c5_a
                best_h = h_a.copy()
            h_cd, c5_cd = coordinate_descent(best_h, n_best, max_rounds=8,
                                             time_limit_end=t_phase2)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()
            if time.time() > t_phase2:
                break
            noise = rng.randn(n_best) * 0.003
            h_pert = project_feasible(best_h + noise, n_best / 2.0)
            h_opt, c5_opt = projected_gradient_opt(h_pert, n_best, 15.0, time_limit)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Phase 3: Try multi-resolution (20% time)
    t_phase3 = t0 + budget_s * 0.85
    if best_h is not None and time.time() < t_phase3:
        n_best = len(best_h)
        for n_try in [n_best + 22, n_best + 24, n_best + 26]:
            if n_try > 120 or time.time() > t_phase3:
                break
            xp = np.linspace(0, 2, n_best, endpoint=False)
            xn = np.linspace(0, 2, n_try, endpoint=False)
            f_interp = np.interp(xn, xp, best_h, period=2.0)
            h_up = np.clip(f_interp, 0, 1)
            h_up = project_feasible(h_up, n_try / 2.0)
            h_opt, c5_opt = projected_gradient_opt(h_up, n_try, 30.0, time_limit)
            if c5_opt < best_c5:
                best_c5 = c5_opt
            if c5_opt < best_c5 or True:
                # Try to downsample back
                if len(h_opt) != n_best:
                    pass

    # Phase 4: Final fine-tuning (15% time)
    t_phase4 = t0 + budget_s * 0.97
    if best_h is not None and time.time() < t_phase4:
        n_best = len(best_h)
        for _ in range(50):
            if time.time() > t_phase4:
                break
            h_cd, c5_cd = coordinate_descent(best_h, n_best, max_rounds=3,
                                             time_limit_end=t_phase4)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()
            h_a, c5_a = active_set_gradient(best_h, n_best, tau=0.001, lr=0.005,
                                            max_iter=100, time_limit_end=t_phase4)
            if c5_a < best_c5:
                best_c5 = c5_a
                best_h = h_a.copy()

    # Final validation
    if best_h is None:
        if parent_h is not None:
            best_h = project_feasible(parent_h.copy(), n_parent / 2.0)
        else:
            n = 77
            best_h = project_feasible(np.full(n, 0.5), n / 2.0)
        best_c5 = evaluate_c5(best_h)

    final_c5 = evaluate_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
