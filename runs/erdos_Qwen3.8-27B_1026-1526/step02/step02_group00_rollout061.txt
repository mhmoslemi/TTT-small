from scipy.optimize import minimize
import time

def compute_corr_values(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def compute_c5(h):
    return float(np.max(compute_corr_values(h)))

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def weighted_gradient(h, weights):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    # For each lag k from -(n-1) to n-1
    for lag in range(n - 1, -1, -1):
        idx_pos = lag + n - 1  # index in corr array
        idx_neg = n - 1 - lag  # index in corr array for -lag
        w_pos = weights[idx_pos] * dx
        w_neg = weights[idx_neg] * dx
        if lag == 0:
            # c[0] = sum h[i](1-h[i])
            # grad = (1 - 2h)
            grad += (w_pos) * (1.0 - 2.0 * h)
        else:
            if w_pos != 0.0:
                # Positive lag: c = sum_{i=0}^{n-1-lag} h[i](1-h[i+lag])
                grad[:n - lag] += w_pos * (1.0 - h[lag:])
                grad[lag:] += w_pos * (-h[:n - lag])
            if w_neg != 0.0:
                # Negative lag (-lag): c = sum_{i=lag}^{n-1} h[i](1-h[i-lag])
                grad[lag:] += w_neg * (1.0 - h[:n - lag])
                grad[:n - lag] += w_neg * (-h[lag:])
    return grad

def smooth_max_obj(h, alpha):
    corr_vals = compute_corr_values(h)
    m = np.max(corr_vals)
    return m + np.log(np.sum(np.exp(alpha * (corr_vals - m)))) / alpha

def smooth_max_grad(h, alpha):
    corr_vals = compute_corr_values(h)
    m = np.max(corr_vals)
    exp_vals = np.exp(alpha * (corr_vals - m))
    weights = exp_vals / np.sum(exp_vals)
    return weighted_gradient(h, weights)

def projected_grad_step(h, grad, eta, target_sum):
    h_new = h - eta * grad
    return project_feasible(h_new, target_sum)

def optimize_smooth_max(h0, n, target_sum, alpha_schedule, steps_per_alpha, eta0):
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    
    for alpha in alpha_schedule:
        eta = eta0
        for step in range(steps_per_alpha):
            grad = smooth_max_grad(h, alpha)
            h_new = projected_grad_step(h, grad, eta, target_sum)
            val = compute_c5(h_new)
            if val < best_val:
                best_val = val
                best_h = h_new.copy()
            # Simple backtracking: if value increased, reduce step
            curr_val = compute_c5(h)
            if val > curr_val + 1e-10:
                eta *= 0.5
                # Retry with smaller step
                h_new = projected_grad_step(h, grad, eta, target_sum)
                val = compute_c5(h_new)
            h = h_new
            # Gradually increase step size
            eta = min(eta * 1.05, eta0)
    
    return best_h, best_val

    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:

def weighted_gradient(h, weights):
    """
    Compute gradient of sum_k weights[k] * c_k * dx
    where c = np.correlate(h, 1-h, mode='full')
    weights has length 2n-1
    """
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for lag in range(n - 1, -1, -1):
        w_pos = weights[lag + n - 1] * dx
        w_neg = weights[n - 1 - lag] * dx
        if lag == 0:
            if w_pos != 0.0:
                grad += w_pos * (1.0 - 2.0 * h)
        else:
            if w_pos != 0.0:
                grad[:n - lag] += w_pos * (1.0 - h[lag:n])
                grad[lag:n] += -w_pos * h[:n - lag]
            if w_neg != 0.0:
                grad[lag:n] += w_neg * (1.0 - h[:n - lag])
                grad[:n - lag] += -w_neg * h[lag:n]
    return grad

def smooth_max_weights(corr_vals, alpha):
    m = np.max(corr_vals)
    exp_vals = np.exp(alpha * (corr_vals - m))
    return exp_vals / np.sum(exp_vals)

def smooth_max_grad(h, alpha):
    corr_vals = compute_corr_values(h)
    weights = smooth_max_weights(corr_vals, alpha)
    return weighted_gradient(h, weights)

def optimize_with_smooth_max(h0, alpha_schedule, iters_per_alpha, target_sum, time_limit):
    """Optimize using smooth max with projected gradient descent."""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    t_start = time.time()
    
    momentum = np.zeros(n)
    beta = 0.9  # momentum parameter
    
    for alpha in alpha_schedule:
        if time.time() - t_start > time_limit:
            break
        eta = 1.0 / (4.0 * alpha * (2.0 / n))
        for it in range(iters_per_alpha):
            if time.time() - t_start > time_limit:
                break
            grad = smooth_max_grad(h, alpha)
            # Remove mean to preserve sum constraint
            grad -= np.mean(grad)
            # Momentum update
            momentum = beta * momentum + (1 - beta) * grad
            # Step
            h_new = h - eta * momentum
            h_new = project_feasible(h_new, target_sum)
            # Check improvement
            val = compute_c5(h_new)
            if val < best_val:
                best_val = val
                best_h = h_new.copy()
            # Adaptive step: if not improving, reduce eta
            curr_val = compute_c5(h)
            if val > curr_val * 1.001 + 1e-8:
                eta *= 0.5
                h_new = h - eta * momentum
                h_new = project_feasible(h_new, target_sum)
                val = compute_c5(h_new)
            else:
                eta = min(eta * 1.1, 1.0 / (2.0 * alpha * (2.0 / n)))
            h = h_new
            # Decay momentum over iterations
            if it % 500 == 0 and it > 0:
                beta = max(0.5, beta * 0.95)
    
    return best_h, best_val

        for it in range(iters_per_alpha):
            if time.time() - t_start > time_limit:
                break
            corr_vals = compute_corr_values(h)
            w = top_k_weights(corr_vals, k_top, alpha)
            grad = weighted_gradient(h, w)
            grad -= np.mean(grad)
            momentum = beta_mom * momentum + (1.0 - beta_mom) * grad
            h_new = h - eta * momentum
            h_new = project_feasible(h_new, target_sum)
            val_new = float(np.max(compute_corr_values(h_new)))
            curr_val = float(np.max(corr_vals))

            if val_new < best_val - 1e-14:
                best_val = val_new
                best_h = h_new.copy()
                stall = 0
            else:
                stall += 1

            if val_new > curr_val + 1e-10:
                eta *= 0.4
                h_new = h - eta * momentum
                h_new = project_feasible(h_new, target_sum)
                val_new = float(np.max(compute_corr_values(h_new)))
                if stall > 250:
                    break
            else:
                eta = min(eta * 1.03, eta_max)
            h = h_new
            if it % 300 == 0 and it > 0:
                k_top = max(3, int(k_top * 0.8))

    return best_h, best_val

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - t_start)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_feasible(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            best_h = None

    if best_h is None:
        best_n = 240
        best_h = project_feasible(np.full(best_n, 0.5) + 0.2 * rng.randn(best_n), best_n / 2.0)
        best_c5 = compute_c5(best_h)

    # Phase 1: Quick polish at native resolution
    if time_left() > 120:
        h_opt, c5_opt = optimize_smooth(
            best_h, [10, 50, 200], 3000, best_n / 2.0,
            min(time_left() - 60, 200)
        )
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 2: Main optimization with top-k method
    if time_left() > 200:
        h_opt, c5_opt = optimize_topk(
            best_h, [30, 100, 300, 1000], 4000, best_n / 2.0,
            min(time_left() - 100, 400), topk_frac=0.25
        )
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 3: Binary refinement
    if time_left() > 120:
        h_bin = binary_round(best_h, best_n / 2.0)
        h_bin, c5_bin = binary_local_search(h_bin, 300, min(time_left() - 60, 200), rng)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
        if time_left() > 80:
            h_pol, c5_pol = optimize_smooth(
                best_h, [100, 500, 2000], 2000, best_n / 2.0,
                min(time_left() - 40, 150)
            )
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Phase 4: Multi-resolution search
    resolutions = [240, 300, 360, 420, 480, 320, 280]
    for n_tr in resolutions:
        if time_left() < 100:
            break
        if n_tr % 2 != 0:
            continue
        h_start = interpolate_to_resolution(best_h, best_n, n_tr)
        h_opt, c5_opt = optimize_smooth(
            h_start, [20, 80, 300, 1000], 2500, n_tr / 2.0,
            min(time_left() - 60, 250)
        )
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_tr

    # Phase 5: Shake and re-optimize at best resolution
    if time_left() > 150:
        for trial in range(4):
            if time_left() < 50:
                break
            noise = 0.12 * (1.0 - trial / 4.0) + 0.03
            h_shake = best_h + noise * rng.randn(best_n)
            h_shake = project_feasible(h_shake, best_n / 2.0)
            h_opt, c5_opt = optimize_smooth(
                h_shake, [50, 200, 800], 2000, best_n / 2.0,
                min(time_left() - 30, 200)
            )
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Phase 6: Spectral starts
    if time_left() > 100:
        for trial in range(4):
            if time_left() < 40:
                break
            h_spec = make_spectral_start(best_n, rng.randint(4, 14), rng)
            h_opt, c5_opt = optimize_smooth(
                h_spec, [20, 100, 500], 2000, best_n / 2.0,
                min(time_left() - 20, 150)
            )
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Phase 7: Final high-alpha polish
    if time_left() > 40:
        h_opt, c5_opt = optimize_smooth(
            best_h, [2000, 8000, 20000], 3000, best_n / 2.0,
            max(0, time_left() - 10)
        )
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Final validation
    best_h = project_feasible(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
