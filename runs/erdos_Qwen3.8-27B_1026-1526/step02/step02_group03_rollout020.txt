    t5 = start_time + budget_s * 0.96
    current = best_h.copy()
    current_c5 = best_c5
    for sweep in range(300):
        if time.time() >= t5:
            break
        order = pairs[:]
        np.random.shuffle(order)
        current, current_c5, improved = do_swap_sweep(current, current_c5, t5, 15, order)
        if current_c5 < best_c5:
            best_h, best_c5 = current.copy(), current_c5
        if not improved and sweep > 2:
            break

    # Phase 6: Final coordinate pass
    t6 = start_time + budget_s * 0.99
    current = best_h.copy()
    current_c5 = best_c5
    current, current_c5 = multi_scale_coord(current, n, t6,
        [0.00002, 0.00001, 0.000005, 0.000001], max_passes=10)
    if current_c5 < best_c5:
        best_h, best_c5 = current.copy(), current_c5

    final_c5 = evaluate_c5(best_h)
    best_c5 = min(best_c5, final_c5)

    return best_h, best_c5, n
