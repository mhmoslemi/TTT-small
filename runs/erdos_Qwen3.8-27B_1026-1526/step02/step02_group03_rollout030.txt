def run_slsqp(h0, n, tau_sched, max_iter_each):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_con, 'args': (n,)}]
    current = project_feasible(h0.copy(), n)
    for tau in tau_sched:
        try:
            res = minimize(
                slsqp_obj, current,
                method='SLSQP',
                args=(n, tau),
                bounds=bounds,
                constraints=cons,
                options={'maxiter': max_iter_each, 'ftol': 1e-15}
            )
            current = project_feasible(res.x, n)
        except Exception:
            pass
    return current, evaluate_c5(current)

def subgradient_exact(h, n):
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    max_val = np.max(corr)
    tol = 1e-9
    active = np.where(corr >= max_val - tol)[0]
    grad = np.zeros(n)
    for k in active:
        lag = int(k) - (n - 1)
        j_s1 = max(0, lag)
        j_e1 = min(n, n + lag)
        if j_s1 < j_e1:
            grad[j_s1:j_e1] += (1.0 - h[j_s1 - lag:j_e1 - lag])
        j_s2 = max(0, -lag)
        j_e2 = min(n, n - lag)
        if j_s2 < j_e2:
            grad[j_s2:j_e2] -= h[j_s2 + lag:j_e2 + lag]
    return grad * dx

def nesterov_subgradient(h, n, time_limit):
    current = h.copy()
    y = h.copy()
    best = h.copy()
    best_c5 = evaluate_c5(current)
    momentum = np.zeros(n)
    t = 1.0
    alpha_base = 0.03
    no_improve = 0

    while time.time() < time_limit:
        t += 1.0
        beta = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        alpha = alpha_base / (np.sqrt(t) + 1.0)

        grad = subgradient_exact(y, n)
        y_new = current + momentum * (beta - 1.0) / beta - alpha * grad
        y_new = project_feasible(y_new, n)

        grad_y = subgradient_exact(y_new, n)
        step = y_new - current
        current_new = current - alpha * grad_y
        current_new = project_feasible(current_new, n)
        momentum = (current - current_new + step) / alpha / beta if alpha > 0 else step

        c5_new = evaluate_c5(current_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best = current_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 100 == 0:
                alpha_base = max(0.005, alpha_base * 0.8)

        current = current_new
        y = y_new

    return best, best_c5

def fourier_optimize(parent_h, n, K, max_iter=400):
    x = np.linspace(0, 2, n, endpoint=False)
    cos_terms = np.array([np.cos(np.pi * k * x) for k in range(1, K + 1)])
    sin_terms = np.array([np.sin(np.pi * k * x) for k in range(1, K + 1)])

    a_init = np.array([np.sum(parent_h * cos_terms[k]) / n * 2.0 for k in range(K)])
    b_init = np.array([np.sum(parent_h * sin_terms[k]) / n * 2.0 for k in range(K)])
    params0 = np.concatenate([a_init, b_init])

    def obj(params):
        a_k = params[:K]
        b_k = params[K:]
        h = np.full(n, 0.5)
        for k in range(K):
            h += a_k[k] * cos_terms[k]
            h += b_k[k] * sin_terms[k]
        h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, n)
        return evaluate_c5(h)

    result = minimize(obj, params0, method='Powell',
                      options={'maxiter': max_iter, 'xtol': 1e-13, 'ftol': 1e-13})
    a_k = result.x[:K]
    b_k = result.x[K:]
    h = np.full(n, 0.5)
    for k in range(K):
        h += a_k[k] * cos_terms[k]
        h += b_k[k] * sin_terms[k]
    h = project_feasible(h, n)
    return h, evaluate_c5(h)

def coordinate_optimize(h, n, time_limit):
    best = h.copy()
    best_c5 = evaluate_c5(best)
    steps = [0.015, -0.015, 0.008, -0.008, 0.004, -0.004, 0.002, -0.002, 0.001, -0.001]

    while time.time() < time_limit:
        improved = False
        for i in range(n):
            if time.time() > time_limit:
                break
            for d in steps:
                h_new = best.copy()
                h_new[i] = np.clip(h_new[i] + d, 0.0, 1.0)
                h_new = project_feasible(h_new, n)
                c5 = evaluate_c5(h_new)
                if c5 < best_c5 - 1e-13:
                    best = h_new
                    best_c5 = c5
                    improved = True
                    break
        if not improved:
            steps = [s * 0.6 for s in steps]
            if abs(steps[0]) < 1e-5:
                break
    return best, best_c5

def reweight_phase(h, n, time_limit):
    current = h.copy()
    best = h.copy()
    best_c5 = evaluate_c5(current)
    weights = None
    tau_rew = 0.01

    while time.time() < time_limit:
        dx = 2.0 / n
        g = 1.0 - current
        corr = np.correlate(current, g, mode="full")
        if weights is None:
            weights = np.ones(len(corr))
        else:
            weights = weights * 0.5 + 0.5 * np.exp(-corr / tau_rew)
            weights /= np.sum(weights)

        def obj_weighted(x):
            g2 = 1.0 - x
            c2 = np.correlate(x, g2, mode="full")
            float c2 = 0.0
            return np.sum(weights * c2) * dx

        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': slsqp_con, 'args': (n,)}]
        try:
            res = minimize(obj_weighted, current, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 50, 'ftol': 1e-14})
            current = project_feasible(res.x, n)
        except Exception:
            pass

        c5 = evaluate_c5(current)
        if c5 < best_c5:
            best_c5 = c5
            best = current.copy()
        tau_rew *= 0.95

    return best, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')

    try:
        parent_h = np.array(initial_h_values, dtype=float)
    except Exception:
        parent_h = None

    phase1_end = start_time + budget_s * 0.18
    if parent_h is not None:
        n = len(parent_h)
        h = project_feasible(parent_h.copy(), n)
        h, c5 = run_slsqp(h, n, [0.01, 0.005, 0.002, 0.001, 0.0005], 200)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    if best_h is None:
        n = 77
        x = np.linspace(0, 2, n, endpoint=False)
        for _ in range(8):
            if time.time() > phase1_end:
                break
            f1 = rng.uniform(1.5, 3.5)
            f2 = rng.uniform(2.5, 5.0)
            p1 = rng.uniform(0, 2 * np.pi)
            p2 = rng.uniform(0, 2 * np.pi)
            a1 = rng.uniform(0.25, 0.40)
            a2 = rng.uniform(0.05, 0.15)
            h0 = 0.5 + a1 * np.sin(2 * np.pi * f1 * x / 2.0 + p1) + a2 * np.sin(2 * np.pi * f2 * x / 2.0 + p2)
            h0 = project_feasible(h0, n)
            h, c5 = run_slsqp(h0, n, [0.05, 0.02, 0.01], 100)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        if best_h is None:
            best_h = project_feasible(np.full(77, 0.5), 77)
            best_c5 = evaluate_c5(best_h)

    n = len(best_h)

    phase2_end = start_time + budget_s * 0.40
    h_sub, c5_sub = nesterov_subgradient(best_h, n, phase2_end)
    if c5_sub < best_c5:
        best_c5 = c5_sub
        best_h = h_sub.copy()

    phase3_end = start_time + budget_s * 0.58
    for K in [12, 18, 25]:
        if time.time() > phase3_end:
            break
        h_four, c5_four = fourier_optimize(best_h, n, K, max_iter=300)
        if c5_four < best_c5:
            best_c5 = c5_four
            best_h = h_four.copy()
            n = len(best_h)

    phase4_end = start_time + budget_s * 0.75
    n = len(best_h)
    for attempt in range(40):
        if time.time() > phase4_end:
            break
        frac = attempt / 40.0
        noise = 0.002 + 0.008 * (1.0 - frac)
        h0 = best_h + rng.randn(n) * noise
        h0 = project_feasible(h0, n)
        h, c5 = run_slsqp(h0, n, [0.002, 0.001, 0.0005], 250)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    phase5_end = start_time + budget_s * 0.90
    n = len(best_h)
    h_coord, c5_coord = coordinate_optimize(best_h, n, phase5_end)
    if c5_coord < best_c5:
        best_c5 = c5_coord
        best_h = h_coord.copy()

    if time.time() < start_time + budget_s * 0.97:
        n = len(best_h)
        h, c5 = run_slsqp(best_h, n, [0.0003, 0.0001], 400)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    n = len(best_h)
    h_final, c5_final = coordinate_optimize(best_h, n, start_time + budget_s * 0.995)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h_final.copy()

    final_c5 = evaluate_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
