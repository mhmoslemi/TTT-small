    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)

    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0

        if np.sum(proj) > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)

    """Compute C5 and subgradient using only top_k active lags."""
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    val_max = np.max(corr)
    c5 = float(val_max * dx)
    # Get top_k indices
    sorted_idx = np.argsort(corr)[::-1][:top_k]
    # Average subgradient over top_k lags

        lo2 = max(0, m)
        hi2 = min(n, n + m)

            subgrad[lo2:hi2] -= h[lo2 - m:hi2 - m]

        sg_proj = sg - np.mean(sg)
        
        # Momentum update
        v = momentum * v + (1 - momentum) * sg_proj
        h_new = h - lr * v
        h_new = project_feasible(h_new, n)
        c5_new = evaluate_c5(h_new)
        
        if c5_new < best_c5 - 1e-12:

        else:
            lr = min(lr * 1.001, max_lr)
        
        step += 1
    
    return best_h, best_c5

    best = h.copy()
    best_c5 = evaluate_c5(best)

                    best_c5 = c5_new

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-13:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        if np.sum(proj) > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-13:
        idx = int(np.argmin(np.abs(result - 0.5)))
        result[idx] = np.clip(result[idx] + diff, 0.0, 1.0)
    return result

def compute_sg_eval(h, n, top_k=3):
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    val_max = np.max(corr)
    c5 = float(val_max * dx)
    sorted_idx = np.argsort(corr)[::-1][:top_k]
    subgrad = np.zeros(n)
    for k in sorted_idx:
        m = k - (n - 1)
        lo1 = max(0, -m)
        hi1 = min(n, n - m)
        if hi1 > lo1:
            subgrad[lo1:hi1] += (1.0 - h[lo1 + m:hi1 + m])
        lo2 = max(0, m)
        hi2 = min(n, n + m)
        if hi2 > lo2:
            subgrad[lo2:hi2] -= h[lo2 - m:hi2 - m]
    subgrad /= len(sorted_idx)
    return c5, subgrad

def smooth_max_obj(x, n, tau):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    v = corr / (tau * dx)
    vmax = np.max(v)
    return tau * dx * (vmax + np.log(np.sum(np.exp(v - vmax))))

def sum_constraint(x, n):
    return np.sum(x) - n / 2.0

def subgrad_momentum(h0, n, max_time, top_k=3, momentum=0.85, seed=0):
    rng = np.random.RandomState(seed)
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    h = h0.copy()
    v = np.zeros(n)
    lr = 1.0 / n
    min_lr = 1e-10
    max_lr = 10.0 / n
    no_improve = 0
    patience = 3000
    start = time.time()
    x_grid = np.linspace(0, 2, n, endpoint=False)
    while time.time() - start < max_time:
        c5, sg = compute_sg_eval(h, n, top_k)
        sg_proj = sg - np.mean(sg)
        v = momentum * v + (1.0 - momentum) * sg_proj
        h_new = h - lr * v
        h_new = project_feasible(h_new, n)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5 - 1e-13:
            best_h = h_new.copy()
            best_c5 = c5_new
            no_improve = 0
        else:
            no_improve += 1
        h = h_new
        if no_improve > patience:
            pert = np.zeros(n)
            for f in range(1, 8):
                pert += 0.04 * np.sin(2.0 * np.pi * f * x_grid + rng.uniform(0, 2.0 * np.pi))
            h = project_feasible(h + pert, n)
            best_h = project_feasible(np.maximum(best_h, 0.01), n)
            best_h = project_feasible(best_h + pert * 0.3, n)
            v = np.zeros(n)
            no_improve = 0
            lr = max(lr * 0.85, min_lr)
        elif no_improve > patience // 3:
            lr = max(lr * 0.995, min_lr)
        else:
            lr = min(lr * 1.002, max_lr)
    return best_h, best_c5

def coordinate_refine(h, n, t_end):
    best = h.copy()
    best_c5 = evaluate_c5(best)
    deltas = [-0.003, -0.001, 0.001, 0.003]
    while time.time() < t_end:
        improved = False
        for i in range(n):
            if time.time() > t_end:
                break
            for d in deltas:
                h_new = best.copy()
                h_new[i] = np.clip(h[i] + d, 0.0, 1.0)
                h_new = project_feasible(h_new, n)
                c5_new = evaluate_c5(h_new)
                if c5_new < best_c5 - 1e-14:
                    best = h_new
                    best_c5 = c5_new
                    improved = True
        if not improved:
            break
    return best, best_c5

def block_perturb_optimize(h, n, max_time, seed=0):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    current = h.copy()
    start = time.time()
    block_size = max(3, n // 10)
    while time.time() - start < max_time:
        idx = rng.randint(0, n - block_size)
        mask = np.zeros(n)
        mask[idx:idx + block_size] = 1.0
        pert = mask * rng.randn(n) * 0.08
        h_new = project_feasible(current + pert, n)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5 - 1e-13:
            best_h = h_new.copy()
            best_c5 = c5_new
            current = h_new.copy()
        elif c5_new < best_c5 + 0.01:
            if rng.rand() < 0.3:
                current = h_new.copy()
    return best_h, best_c5
