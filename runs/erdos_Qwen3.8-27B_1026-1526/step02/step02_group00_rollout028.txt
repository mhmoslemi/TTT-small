import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m_val))
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    # Gradient computation
    # corr[m] = sum_i h[i+m-(n-1)] * (1-h[i]) for the full correlation
    # Actually let me just use the same formula as before
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def slsqp_local(h0, n, target_sum, alpha, maxiter):
    def f(h):
        return smooth_max_and_grad(h, alpha)[0]
    def g(h):
        return smooth_max_and_grad(h, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

And for flipping h[j] from 0→1:
= +[0≤j-k<n] * (1-h[j-k]) - [0≤j+k<n] * h[j+k]

Wait, I need to be more careful. When h[j] goes from 0 to 1:
- For each idx where j = n+k (n = j-k): term goes from 0*(1-h[n]) to 1*(1-h[n]), change = +(1-h[j-k])
  valid if 0 ≤ j-k < n
- For each idx where j = n: term goes from h[n+k]*(1-0) to h[n+k]*(1-1)=0, change = -h[j+k]
  valid if 0 ≤ j+k < n

So total change for swap (i: 1→0, j: 0→1):
Δcorr[idx] = -[0≤i-k<n]*(1-h[i-k]) + [0≤i+k<n]*h[i+k] + [0≤j-k<n]*(1-h[j-k]) - [0≤j+k<n]*h[j+k]

For binary h, h[i-k] and h[j+k] are 0 or 1.

This means I can precompute the "effect vectors" for each position and then the effect of a swap is just a sum of two precomputed vectors. The new C5 is max(old_corr + effect) * dx.

        c5_cur = best_c5_round
        improved = True
    
    return h, c5_cur

def binary_anneal(h, n, target_sum, rng, t_start=0.5, t_end=0.01, n_iter=5000):
    h = h.copy()
    c5_cur = compute_c5(h)
    c5_best = c5_cur
    h_best = h.copy()
    
    for it in range(n_iter):
        T = t_start * (t_end / t_start) ** (it / n_iter)
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i], h[j] = 0.0, 1.0
        c5_new = compute_c5(h)
        if c5_new < c5_cur or rng.rand() < np.exp(-(c5_new - c5_cur) / T):
            c5_cur = c5_new
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
    
    return h_best, c5_best

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f.astype(float) * (1.0 - h[fc]) * w[None, :]
    cs = valid_s.astype(float) * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def slsqp_optimize(h0, n, target_sum, alpha, maxiter):
    def f(h):
        return smooth_max_and_grad(h, alpha)[0]
    def g(h):
        return smooth_max_and_grad(h, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def compute_neg_effect(h, n):
    """Compute effect on corr when each position i is flipped from 1 to 0."""
    effect = np.zeros((n, 2 * n - 1))
    k_offset = n - 1
    for i in range(n):
        for n_val in range(0, i + 1):
            if n_val < n:
                k_val = i - n_val
                if -(n - 1) <= k_val <= n - 1:
                    idx = k_val + k_offset
                    effect[i, idx] -= (1.0 - h[n_val])
        for k_val in range(max(-i, -(n - 1)), n - i):
            idx = k_val + k_offset
            if 0 <= idx < 2 * n - 1:
                effect[i, idx] += h[i + k_val]
    return effect

def binary_hill_climb(h, n, target_sum, max_rounds=200):
    h = h.copy()
    c5_cur = compute_c5(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    for _ in range(max_rounds):
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        neg_eff = compute_neg_effect(h, n)
        best_c5_round = c5_cur
        best_swap = None
        for i in ones:
            base = corr + neg_eff[i]
            diffs = base[None, :] - neg_eff[zeros, :]
            maxes = np.max(diffs, axis=1)
            best_j_local = np.argmin(maxes)
            if maxes[best_j_local] * dx < best_c5_round - 1e-14:
                best_c5_round = maxes[best_j_local] * dx
                best_swap = (i, zeros[best_j_local])
        if best_swap is None:
            break
        i, j = best_swap
        h[i] = 0.0
        h[j] = 1.0
        corr = compute_corr(h)
        c5_cur = best_c5_round
    return h, c5_cur

def binary_anneal(h, n, target_sum, rng, t_high=0.4, t_low=0.005, n_iter=5000):
    h = h.copy()
    c5_cur = compute_c5(h)
    c5_best = c5_cur
    h_best = h.copy()
    for it in range(n_iter):
        T = t_high * (t_low / t_high) ** (it / max(n_iter - 1, 1))
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        old_hi, old_hj = h[i], h[j]
        h[i], h[j] = 0.0, 1.0
        c5_new = compute_c5(h)
        delta = c5_new - c5_cur
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            c5_cur = c5_new
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h.copy()
        else:
            h[i], h[j] = old_hi, old_hj
    return h_best, c5_best

def round_to_binary(h, n, target_sum, rng):
    """Round h to binary with exactly target_sum ones."""
    n_target = int(round(target_sum))
    sorted_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:n_target]] = 1.0
    return h_bin

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n1 = best_n
    ts1 = n1 / 2.0

    # Phase 1: Continuous optimization with SLSQP
    for alpha in [5, 20, 100, 500, 2000, 10000]:
        if time.time() - t_start > budget_s - 300:
            break
        h_new = slsqp_optimize(best_h, n1, ts1, alpha, maxiter=1500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-10:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 2: Multiple binary roundings + hill climbing
    if time.time() - t_start < budget_s - 250:
        candidates = []
        # Deterministic rounding
        candidates.append(round_to_binary(best_h, n1, ts1, rng))
        # Stochastic roundings
        for _ in range(8):
            u = rng.rand(n1)
            h_stoch = np.zeros(n1)
            h_stoch[u < best_h] = 1.0
            defix = int(ts1) - int(h_stoch.sum())
            if defix > 0:
                zeros = np.where(h_stoch == 0)[0]
                rng.shuffle(zeros)
                h_stoch[zeros[:defix]] = 1.0
            elif defix < 0:
                ones = np.where(h_stoch == 1)[0]
                rng.shuffle(ones)
                h_stoch[ones[:defix]] = 0.0
            candidates.append(h_stoch)

        for h_cand in candidates:
            if time.time() - t_start > budget_s - 200:
                break
            c5_cand = compute_c5(h_cand)
            if c5_cand < best_c5:
                best_c5 = c5_cand
                best_h = h_cand.copy()
            # Hill climb from this candidate
            h_hc, c5_hc = binary_hill_climb(h_cand, n1, ts1, max_rounds=150)
            if c5_hc < best_c5:
                best_c5 = c5_hc
                best_h = h_hc.copy()

    # Phase 3: Simulated annealing from best binary solution
    if time.time() - t_start < budget_s - 200:
        h_bin = (best_h > 0.5).astype(float)
        fix_diff = int(ts1) - int(h_bin.sum())
        if fix_diff > 0:
            zeros = np.where(h_bin == 0)[0]
            order = np.argsort(-best_h[zeros])
            h_bin[zeros[order[:fix_diff]]] = 1.0
        elif fix_diff < 0:
            ones = np.where(h_bin == 1)[0]
            order = np.argsort(best_h[ones])
            h_bin[ones[order[:fix_diff]]] = 0.0

        # Multiple annealing runs with different seeds
        for run_idx in range(4):
            if time.time() - t_start > budget_s - 150:
                break
            rng_sa = np.random.RandomState(seed + 1000 + run_idx)
            t_high = 0.3 * (0.7 ** run_idx)
            n_iter = max(2000, 5000 - run_idx * 1000)
            h_sa, c5_sa = binary_anneal(h_bin, n1, ts1, rng_sa,
                                         t_high=t_high, t_low=0.003,
                                         n_iter=n_iter)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
                h_bin = h_sa.copy()
            # Hill climb after annealing
            h_hc, c5_hc = binary_hill_climb(h_sa, n1, ts1, max_rounds=100)
            if c5_hc < best_c5:
                best_c5 = c5_hc
                best_h = h_hc.copy()
                h_bin = h_hc.copy()

    # Phase 4: Final continuous optimization from best solution
    if time.time() - t_start < budget_s - 80:
        h_cont = best_h.copy()
        h_cont = h_cont + 0.02 * rng.randn(n1)
        h_cont = project_to_sum(h_cont, ts1)
        for alpha in [50, 500, 5000]:
            if time.time() - t_start > budget_s - 50:
                break
            h_opt = slsqp_optimize(h_cont, n1, ts1, alpha, maxiter=800)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
            h_cont = h_opt
        # Also try directly from the binary solution
        h_cont2 = best_h.astype(float)
        h_cont2 = project_to_sum(h_cont2 + 0.01 * rng.randn(n1), ts1)
        for alpha in [200, 2000]:
            if time.time() - t_start > budget_s - 30:
                break
            h_opt = slsqp_optimize(h_cont2, n1, ts1, alpha, maxiter=600)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
            h_cont2 = h_opt

    # Phase 5: Try higher resolution
    if time.time() - t_start < budget_s - 80:
        n_high = n1 * 2
        ts_high = n_high / 2.0
        x_old = np.linspace(0, 2.0, n1, endpoint=False)
        x_new = np.linspace(0, 2.0, n_high, endpoint=False)
        h_high = np.interp(x_new, x_old, best_h, period=2.0)
        h_high = project_to_sum(h_high, ts_high)
        c5_high = compute_c5(h_high)
        if c5_high < best_c5:
            best_c5 = c5_high
            best_h = h_high.copy()
            best_n = n_high
        # Optimize at higher resolution
        for alpha in [20, 200, 2000]:
            if time.time() - t_start > budget_s - 30:
                break
            h_opt = slsqp_optimize(best_h, best_n, best_n / 2.0, alpha, maxiter=600)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Phase 6: Final binary polish at best resolution
    if time.time() - t_start < budget_s - 20 and best_n == n1:
        h_bin_final = (best_h > 0.5).astype(float)
        fix_d = int(ts1) - int(h_bin_final.sum())
        if fix_d > 0:
            zeros = np.where(h_bin_final == 0)[0]
            order = np.argsort(-best_h[zeros])
            h_bin_final[zeros[order[:fix_d]]] = 1.0
        elif fix_d < 0:
            ones = np.where(h_bin_final == 1)[0]
            order = np.argsort(best_h[ones])
            h_bin_final[ones[order[:fix_d]]] = 0.0
        h_final, c5_final = binary_hill_climb(h_bin_final, n1, ts1, max_rounds=100)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Final validation
    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
