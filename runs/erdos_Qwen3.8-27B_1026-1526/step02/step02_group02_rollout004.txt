```python
def compute_gradient(h, T):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    # softmax
    c_max = c.max()
    shifted = T * (c - c_max)
    shifted = np.clip(shifted, -400, 400)
    p = np.exp(shifted)
    p /= p.sum()
    
    # Gradient
    grad = np.zeros(n)
    full_len = 2 * n - 1
    for k in range(full_len):
        if p[k] < 1e-15:
            continue
        m = k - (n - 1)
        # Term 1: (1 - h[j+m]) for valid j
        # j ranges [0, n-1], need 0 ≤ j+m < n, so j ranges [-m, n-1-m]
        j_min = max(0, -m)
        j_max = min(n-1, n-1-m)
        if j_min <= j_max:
            grad[j_min:j_max+1] += p[k] * (1.0 - h[j_min + m:j_max + 1 + m])
        # Term 2: -h[j-m] for valid j
        # j ranges [0, n-1], need 0 ≤ j-m < n, so j ranges [m, m+n-1]
        j_min2 = max(0, m)
        j_max2 = min(n-1, m + n - 1)
        if j_min2 <= j_max2:
            grad[j_min2:j_max2+1] -= p[k] * h[j_min2 - m:j_max2 + 1 - m]
    
    grad *= dx
    return grad, c.max()
```

```python
import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_box_sum(v, n):
    target = n / 2.0
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        if np.clip(v - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(50):
        d = target - x.sum()
        if abs(d) < 1e-14:
            break
        m = (x > 1e-15) & (x < 1.0 - 1e-15)
        if not np.any(m):
            break
        x[m] = np.clip(x[m] + d / m.sum(), 0.0, 1.0)
    return x

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    cm = c.max()
    s = T * (c - cm)
    s = np.clip(s, -450, 450)
    p = np.exp(s)
    ps = p.sum()
    if ps < 1e-300:
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        p /= ps
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-14:
            continue
        m = k - (n - 1)
        a = max(0, -m)
        b = min(n - 1, n - 1 - m)
        if a <= b:
            g[a:b + 1] += p[k] * (1.0 - h[a + m:b + 1 + m])
        a2 = max(0, m)
        b2 = min(n - 1, m + n - 1)
        if a2 <= b2:
            g[a2:b2 + 1] -= p[k] * h[a2 - m:b2 + 1 - m]
    return g * dx, cm

def pg_optimize(h0, n, T0, T1, lr0, lr1, niter, mom=0.0):
    h = h0.copy()
    vh = np.zeros(n)
    bh = h.copy()
    bc5 = compute_c5_from_h(h)
    for it in range(niter):
        f = it / max(1, niter)
        T = T0 * (T1 / T0) ** f
        lr = lr0 * (lr1 / lr0) ** f
        g, _ = smooth_grad(h, T)
        vh = mom * vh - lr * g
        hn = project_box_sum(h + vh, n)
        c5 = compute_c5_from_h(hn)
        if c5 < bc5:
            bc5 = c5
            bh = hn.copy()
        h = hn
    return bh, bc5

def slsqp_topk(h0, n, topk=40, maxiter=5000, tmargin=0.05):
    c5i = compute_c5_from_h(h0)
    c = np.correlate(h0, 1.0 - h0, mode="full")
    idx = np.argsort(c)[-topk:]
    dx = 2.0 / n
    x0 = np.concatenate([h0, [c5i + tmargin]])
    bnds = [(0.0, 1.0)] * n + [(c5i * 0.6, c5i + tmargin)]

    def obj(x):
        return x[-1]

    def ineq(x):
        c2 = np.correlate(x[:-1], 1.0 - x[:-1], mode="full")
        return x[-1] - c2[idx] * dx

    def eqc(x):
        return x[:-1].sum() - n / 2.0

    def eqj(x):
        j = np.zeros(n + 1)
        j[:n] = 1.0
        return j

    constraints = [{'type': 'ineq', 'fun': ineq}, {'type': 'eq', 'fun': eqc, 'jac': eqj}]
    r = minimize(obj, x0, method='SLSQP', bounds=bnds, constraints=constraints,
                 options={'maxiter': maxiter, 'ftol': 1e-15})
    if r.x is not None:
        ho = project_box_sum(r.x[:n], n)
        return ho, compute_c5_from_h(ho)
    return h0, c5i

def slsqp_active(h0, n, maxiter=5000, tmargin=0.03, tol=0.002):
    c5i = compute_c5_from_h(h0)
    c = np.correlate(h0, 1.0 - h0, mode="full") * (2.0 / n)
    active = np.where(np.abs(c - c.max()) < tol)[0]
    if len(active) == 0:
        active = np.argsort(c)[-30:]
    if len(active) < 5:
        active = np.argsort(c)[-40:]
    dx = 2.0 / n
    x0 = np.concatenate([h0, [c5i + tmargin]])
    bnds = [(0.0, 1.0)] * n + [(c5i * 0.5, c5i + tmargin)]

    def obj(x):
        return x[-1]

    def ineq(x):
        c2 = np.correlate(x[:-1], 1.0 - x[:-1], mode="full")
        return x[-1] - c2[active] * dx

    def eqc(x):
        return x[:-1].sum() - n / 2.0

    def eqj(x):
        j = np.zeros(n + 1)
        j[:n] = 1.0
        return j

    constraints = [{'type': 'ineq', 'fun': ineq}, {'type': 'eq', 'fun': eqc, 'jac': eqj}]
    r = minimize(obj, x0, method='SLSQP', bounds=bnds, constraints=constraints,
                 options={'maxiter': maxiter, 'ftol': 1e-15})
    if r.x is not None:
        ho = project_box_sum(r.x[:n], n)
        return ho, compute_c5_from_h(ho)
    return h0, c5i

def lbfgsb_smooth(h0, n, T, maxiter=3000):
    def obj(x):
        _, cm = smooth_grad(x, T)
        return cm

    bounds = [(0.0, 1.0)] * n
    r = minimize(obj, h0, method='L-BFGS-B', bounds=bounds,
                 options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14,
                          'maxfun': maxiter * 3})
    if r.x is not None:
        ho = project_box_sum(r.x, n)
        return ho, compute_c5_from_h(ho)
    return h0, compute_c5_from_h(h0)

def make_struct_init(n, mode, rng):
    if mode == 0:
        return np.full(n, 0.5)
    elif mode == 1:
        x = np.arange(n)
        h = 0.5 + 0.3 * np.sin(2 * np.pi * rng.integers(1, 5) * x / n + rng.random() * 2 * np.pi)
        return project_box_sum(h, n)
    elif mode == 2:
        h = np.zeros(n)
        blk = max(3, n // 20)
        val = 1.0
        for i in range(0, n, blk):
            h[i:min(i + blk, n)] = val
            val = 1.0 - val
        return project_box_sum(h, n)
    elif mode == 3:
        x = np.arange(n)
        h = 0.5 + sum(0.15 * np.sin(2 * np.pi * f * x / n + rng.random() * 2 * np.pi)
                      for f in rng.integers(1, 8, size=3))
        return project_box_sum(h, n)
    else:
        h = rng.random(n)
        return project_box_sum(h, n)

def run(seed=42, budget_s=1300, **kwargs):
```
