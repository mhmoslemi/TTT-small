def compute_smooth_grad(h, tau):
    n = len(h)
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    m = len(corr)
    v = corr / tau
    vmax = np.max(v)
    exp_v = np.exp(v - vmax)
    weights = exp_v / np.sum(exp_v)
    
    grad = np.zeros(n)
    for i in range(n):
        # First term
        m_lo = max(0, n - 1 - i)
        m_hi = min(2 * n - 2, 2 * n - 2 - i)
        if m_lo <= m_hi:
            partners = i + np.arange(m_lo, m_hi + 1) - (n - 1)
            grad[i] += np.dot(weights[m_lo:m_hi+1], g[partners])
        # Second term
        m_lo2 = max(0, i - (n - 1))
        m_hi2 = min(2 * n - 2, i + n - 1)
        if m_lo2 <= m_hi2:
            partners2 = i - np.arange(m_lo2, m_hi2 + 1) + (n - 1)
            grad[i] -= np.dot(weights[m_lo2:m_hi2+1], h[partners2])
    
    obj = tau * (vmax + np.log(np.sum(exp_v)))
    return grad, obj

        v_hat = beta2 * v_hat + (1 - beta2) * grad**2

        f1 = rng.uniform(1, 4)
        f2 = rng.uniform(4, 9)
        a1 = rng.uniform(0.15, 0.3)
        a2 = rng.uniform(0.05, 0.15)
        p1 = rng.uniform(0, 2 * np.pi)
        p2 = rng.uniform(0, 2 * np.pi)
        h = 0.5 + a1 * np.sin(2 * np.pi * f1 * x / 2 + p1) + a2 * np.sin(2 * np.pi * f2 * x / 2 + p2)
        starts.append(np.clip(h, 0, 1))
    for _ in range(2):
        starts.append(rng.rand(n))
    starts.append(np.linspace(0.01, 0.99, n))
    starts.append(np.linspace(0.99, 0.01, n))
    for _ in range(2):
        h_bin = (rng.rand(n) > 0.5).astype(float)
        starts.append(0.05 + 0.9 * h_bin)
    for _ in range(2):
        center = rng.uniform(0.1 * n, 0.9 * n)
        width = rng.uniform(0.1 * n, 0.4 * n)
        ar = np.arange(n, dtype=float)
        h = np.exp(-0.5 * ((ar - center) / width) ** 2)
        h = 0.1 + 0.8 * h
        starts.append(np.clip(h, 0, 1))
    return starts[:num_starts]

def ga_tournament_select(pop fitness_pop, fitnesses, rng, k=5):
    n = len(fitnesses)
    indices = rng.choice(n, size=k, replace=False)
    best_idx = indices[np.argmin(fitnesses[indices])]
    return pop[best_idx].copy()

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')

    parent_h = None
    try:
        parent_h = np.array(initial_h_values, dtype=float)
    except Exception:
        pass

    def elapsed():
        return time.time() - start_time

    def remaining():
        return budget_s - elapsed()

    phase1_end = start_time + budget_s * 0.28
    phase2_end = start_time + budget_s * 0.65
    phase3_end = start_time + budget_s * 0.88

    # Phase 1: Broad exploration with SLSQP across multiple n
    n_candidates = []
    if parent_h is not None:
        n_candidates.append(len(parent_h))
    for nn in [74, 76, 78, 80, 82, 84, 56, 60, 64, 68, 72, 77, 86, 90, 96]:
        if nn not in n_candidates:
            n_candidates.append(nn)

    for n in n_candidates:
        if time.time() > phase1_end:
            break
        starts = generate_starts(n, parent_h, rng, num_starts=5)
        for h0 in starts:
            if time.time() > phase1_end:
                break
            h_opt, c5 = slsqp_optimize(h0, n, [0.2, 0.08, 0.03, 0.01], 35)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()

    # Phase 2: Intensive refinement with Adam
    if best_h is not None:
        n = len(best_h)
        tau_levels = [0.08, 0.04, 0.02, 0.01, 0.005, 0.003, 0.001, 0.0005]
        for tau in tau_levels:
            if time.time() > phase2_end:
                break
            time_budget = phase2_end - start_time - elapsed()
            if time_budget < 5:
                break
            steps = int(min(30000, max(2000, time_budget * 800)))
            h_opt, c5 = adam_optimize(best_h, n, tau, steps, lr=0.03)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()
            # Also try Nesterov
            if time.time() < phase2_end:
                steps_n = int(min(10000, max(1000, time_budget * 300)))
                h_opt, c5 = nesterov_optimize(best_h, n, tau, steps_n)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt.copy()

    # Phase 3: Random restarts with Adam
    if best_h is not None and time.time() < phase3_end:
        n = len(best_h)
        while time.time() < phase3_end:
            frac = elapsed() / budget_s
            noise_scale = 0.004 + 0.04 * max(0.0, 1.0 - frac * 1.4)
            perturb = best_h + rng.randn(n) * noise_scale
            perturb = project_feasible(perturb, n)
            steps = 2000
            h_opt, c5 = adam_optimize(perturb, n, 0.01, steps, lr=0.015)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()
            if time.time() > phase3_end:
                break

    # Phase 4: Coordinate descent polishing
    if best_h is not None and time.time() < start_time + budget_s * 0.96:
        n = len(best_h)
        h_coord, c5_coord = coordinate_descent(best_h, n, max_sweeps=25)
        if c5_coord < best_c5:
            best_c5 = c5_coord
            best_h = h_coord.copy()

    # Phase 5: Final SLSQP with very small tau
    if best_h is not None and time.time() < start_time + budget_s * 0.995:
        n = len(best_h)
        bounds = [(0.0, 1.0)] * n
        while time.time() < start_time + budget_s * 0.995:
            for tau in [0.003, 0.001, 0.0005]:
                if time.time() > start_time + budget_s * 0.995:
                    break
                def obj_final(x):
                    return smooth_max_obj(x, n, tau)
                try:
                    res = minimize(
                        obj_final, best_h,
                        method='SLSQP',
                        bounds=bounds,
                        constraints=[{'type': 'eq', 'fun': constraint_eq, 'args': (n,)}],
                        options={'maxiter': 300, 'ftol': 1e-14}
                    )
                    current = project_feasible(res.x, n)
                    c5 = evaluate_c5(current)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = current.copy()
                except Exception:
                    pass

    # Phase 6: Final coordinate descent
    if best_h is not None:
        n = len(best_h)
        h_coord, c5_coord = coordinate_descent(best_h, n, max_sweeps=10)
        if c5_coord < best_c5:
            best_c5 = c5_coord
            best_h = h_coord.copy()

    if best_h is None:
        if parent_h is not None:
            best_h = parent_h.copy()
            best_c5 = evaluate_c5(best_h)
        else:
            n = 77
            best_h = project_feasible(np.full(n, 0.5), n)
            best_c5 = evaluate_c5(best_h)

    final_c5 = evaluate_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
