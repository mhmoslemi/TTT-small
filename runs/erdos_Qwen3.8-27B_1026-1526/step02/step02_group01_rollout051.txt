```python
    k = np.arange(-(n-1), n)  # lags
    # First term: for each j, sum over valid k of w_full[k+(n-1)] * (1-h[j-k])
    # Second term: for each j, sum over valid k of w_full[k+(n-1)] * h[j+k]
    
    g = np.zeros(n)
    for jm in range(n):
        # First term: (1-h[l]) where l = j-k, valid when 0<=l<n
        # l = j-k, so k = j-l, valid k: -(n-1)<=j-l<=n-1, 0<=l<n

        l_min = max(0, j - (n-1))
        l_max = min(j, n-1)
        if l_min <= l_max:
            l = np.arange(l_min, l_max+1)
            k_idx = j - l  # these are the k values
            g[jm] += np.dot(w_full[k_idx + (n-1)], 1-h[l])
        
        # Second term: h[l] where l = j+k, valid when 0<=l<n

        l_max2 = n-1
        if l_min2 <= l_max2:

            k_idx = l - j  # these are the k values
```

```python
def gradient_smooth_max(h, w_full):
    n = len(h)
    half = n - 1
    g = np.zeros(n)
    
    for ell in range(-(n-1), n):
        w = w_full[ell + half]
        if abs(w) < 1e-15:
            continue
        if ell >= 0:
            # First term: (1-h[j-ell]) for j in [ell, n-1]
            j_lo, j_hi = ell, n - 1
            if j_lo <= j_hi:
                g[j_lo:j_hi+1] += w * (1.0 - h[j_lo-ell:j_hi-ell+1])
            # Second term: -h[j+ell] for j in [0, n-1-ell]
            j_lo2, j_hi2 = 0, n - 1 - ell
            if j_lo2 <= j_hi2:
                g[j_lo2:j_hi2+1] -= w * h[j_lo2+ell:j_hi2+ell+1]
        else:
            p = -ell
            # First term: (1-h[j+p]) for j in [0, n-1-p]
            j_lo, j_hi = 0, n - 1 - p
            if j_lo <= j_hi:
                g[j_lo:j_hi+1] += w * (1.0 - h[j_lo+p:j_hi+p+1])
            # Second term: -h[j-p] for j in [p, n-1]
            j_lo2, j_hi2 = p, n - 1
            if j_lo2 <= j_hi2:
                g[j_lo2:j_hi2+1] -= w * h[j_lo2-p:j_hi2-p+1]
    return g
```

Combined with projected gradient with backtracking line search might be even better.

Let me also think about a different approach entirely: **subgradient method with projection**.

At each iteration:
1. Find k* = argmax corr
2. Compute the (sub)gradient of corr[k*] w.r.t. h
3. Take a projected step

This is simpler but converges slowly. The smooth max approach is essentially a smoothed version of this.

**Another idea**: Since the problem is min max, I can use a bisection approach:
- Binary search on the target value t
- For a given t, check if there exists h such that all correlations ≤ t/dx
- This is a feasibility problem (possibly convex if we reformulate)

Actually, the constraint corr_k ≤ t for all k is a quadratic constraint (corr_k is quadratic in h). So it's not convex in general.

Let me think about what makes this problem hard:
- The objective is the max of 2n-1 quadratic functions
- The constraints are 0≤h≤1 and sum(h)=n/2
- The landscape is non-convex

**Key insight for improvement**: The previous code's main weakness might be:
1. Not enough iterations of the gradient method
2. Not enough diversification (random restarts)
3. The SLSQP might not be effective with the smooth max (it's designed for smooth objectives, and the smooth max with large alpha is nearly non-smooth)

**My improved strategy**:
1. Start with initial_h_values (n=360)
2. For each resolution n in [360, 420, 480, 540, 600]:
   a. Interpolate current best to this resolution
   b. Run Adam with adaptive alpha (start low, increase)
   c. Run projected subgradient (pick k* random among top 5)
   d. Run coordinate descent with line search
   e. If time permits, try 2-3 random perturbations
3. Final polish on best solution

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_weights(corr, alpha):
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / len(exp_vals)) / alpha
    return sm_val, w

def gradient_smooth_max(h, w_full):
    n = len(h)
    half = n - 1
    g = np.zeros(n)
    for ell in range(-(n-1), n):
        w = w_full[ell + half]
        if abs(w) < 1e-15:
            continue
        if ell >= 0:
            j_lo, j_hi = ell, n - 1
            if j_lo <= j_hi:
                g[j_lo:j_hi+1] += w * (1.0 - h[j_lo-ell:j_hi-ell+1])
            j_lo2, j_hi2 = 0, n - 1 - ell
            if j_lo2 <= j_hi2:
                g[j_lo2:j_hi2+1] -= w * h[j_lo2+ell:j_hi2+ell+1]
        else:
            p = -ell
            j_lo, j_hi = 0, n - 1 - p
            if j_lo <= j_hi:
                g[j_lo:j_hi+1] += w * (1.0 - h[j_lo+p:j_hi+p+1])
            j_lo2, j_hi2 = p, n - 1
            if j_lo2 <= j_hi2:
                g[j_lo2:j_hi2+1] -= w * h[j_lo2-p:j_hi2-p+1]
    return g

def adam_optimize(h0, n, target_sum, alpha, lr, n_iters, time_fn):
    h = h0.copy()
    m_vec = np.zeros(n)
    v_vec = np.zeros(n)
    t = 0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    
    for it in range(n_iters):
        if time_fn() < 30:
            break
        t += 1
        corr = compute_corr(h)
        sm_val, w_full = smooth_max_and_weights(corr, alpha)
        g = gradient_smooth_max(h, w_full)
        
        m_vec = beta1 * m_vec + (1 - beta1) * g
        v_vec = beta2 * v_vec + (1 - beta2) * (g * g)
        m_hat = m_vec / (1.0 - beta1**t)
        v_hat = v_vec / (1.0 - beta2**t)
        
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_to_sum(np.clip(h_new, 0.0, 1.0), target_sum)
        h = h_new
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def projected_subgradient(h0, n, target_sum, n_iters, lr_init, time_fn):
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    t = 1.0
    y = h.copy()
    
    for it in range(n_iters):
        if time_fn() < 30:
            break
        corr = compute_corr(y)
        # Pick among top-5 correlations
        topk = np.argsort(corr)[-5:]
        k_star = topk[np.random.randint(5)]
        ell = k_star - (n - 1)
        
        w_single = np.zeros(2*n-1)
        w_single[k_star] = 1.0
        g = gradient_smooth_max(y, w_single)
        
        h_new = np.clip(y - lr * g, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        mu = (t - 1.0) / t_new
        y = h_new + mu * (h_new - h)
        y = np.clip(y, 0.0, 1.0)
        h = h_new
        t = t_new
        lr = lr_init / np.sqrt(t)
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def coordinate_descent(h0, n, target_sum, max_rounds, rng, time_fn):
    h = h0.copy()
    c5_cur = compute_c5(h)
    
    for rnd in range(max_rounds):
        if time_fn() < 30:
            break
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            # Try several step sizes
            best_val = c5_cur
            best_step = 0.0
            for step_frac in [-0.1, -0.05, -0.02, 0.02, 0.05, 0.1]:
                d = step_frac * (1.0 - 2.0 * h[i])  # step towards 0 or 1
                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] + d, 0.0, 1.0)
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_val - 1e-14:
                    best_val = c5_new
                    best_step = d
            if best_step != 0.0:
                h[i] = np.clip(h[i] + best_step, 0.0, 1.0)
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True
        if not improved:
            break
    return h, c5_cur

def slsqp_optimize(h0, n, target_sum, alpha, maxiter):
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_and_weights(compute_corr(h), alpha)
        return v
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, w = smooth_max_and_weights(compute_corr(h), alpha)
        return gradient_smooth_max(h, w)
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    def time_left():
        return budget_s - (time.time() - t_start)
    
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            ts = best_n / 2.0
            best_h = project_to_sum(best_h, ts)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass
    
    if best_h is None:
        best_n = 360
        best_h = np.full(360, 0.5)
        best_c5 = compute_c5(best_h)
    
    # Resolution targets
    n_targets = [360, 420, 480, 540]
    
    for n in n_targets:

            break
        if n % 2 != 0:

        ts = n / 2.0
        if abs(ts - round(ts)) > 1e-9:
            continue

        if best_n == n and best_h is not None:
            h_cur = best_h.copy()
        else:
            x_old = np.linspace(0, 2, best_n, endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            h_cur = np.interp(x_new, x_old, best_h)
            h_cur = project_to_sum(h_cur, ts)
        
        c5_cur = compute_c5(h_cur)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()
            best_n = n
        
        # Phase A: Adam with increasing alpha
        alphas = [20, 50, 100, 200, 400, 800]
        for alpha in alphas:
            if time_left() < 60:

            lr = 0.01

            if c5_opt < best_c5:

            h_cur = h_opt

                maxiter = max(50, min(300, int(time_left())))
                h_opt = slsqp_optimize(h_cur, n, ts, alpha, maxiter)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:

                    best_n = n
                h_cur = h_opt
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(600):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_weights(corr, alpha):
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / len(exp_vals)) / alpha
    return sm_val, w

def gradient_smooth_max(h, w_full):
    n = len(h)
    half = n - 1
    g = np.zeros(n)
    for ell in range(-(n - 1), n):
        w = w_full[ell + half]
        if abs(w) < 1e-16:
            continue
        if ell >= 0:
            j_lo = ell
            j_hi = n - 1
            if j_lo <= j_hi:
                g[j_lo:j_hi + 1] += w * (1.0 - h[j_lo - ell:j_hi - ell + 1])
            j_lo2 = 0
            j_hi2 = n - 1 - ell
            if j_lo2 <= j_hi2:
                g[j_lo2:j_hi2 + 1] -= w * h[j_lo2 + ell:j_hi2 + ell + 1]
        else:
            p = -ell
            j_lo = 0
            j_hi = n - 1 - p
            if j_lo <= j_hi:
                g[j_lo:j_hi + 1] += w * (1.0 - h[j_lo + p:j_hi + p + 1])
            j_lo2 = p
            j_hi2 = n - 1
            if j_lo2 <= j_hi2:
                g[j_lo2:j_hi2 + 1] -= w * h[j_lo2 - p:j_hi2 - p + 1]
    return g

def adam_optimize(h0, n, target_sum, alpha, lr, n_iters, time_fn):
    h = h0.copy()
    m_vec = np.zeros(n)
    v_vec = np.zeros(n)
    t = 0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    beta1 = 0.85
    beta2 = 0.999
    eps = 1e-8
    for it in range(n_iters):
        if time_fn() < 25:
            break
        t += 1
        corr = compute_corr(h)
        sm_val, w_full = smooth_max_and_weights(corr, alpha)
        g = gradient_smooth_max(h, w_full)
        m_vec = beta1 * m_vec + (1.0 - beta1) * g
        v_vec = beta2 * v_vec + (1.0 - beta2) * (g * g)
        m_hat = m_vec / (1.0 - beta1 ** t)
        v_hat = v_vec / (1.0 - beta2 ** t)
        step = m_hat / (np.sqrt(v_hat) + eps)
        step_norm = np.linalg.norm(step)
        if step_norm > 1.0:
            step *= 1.0 / step_norm
        h_new = h - lr * step
        h_new = project_to_sum(np.clip(h_new, 0.0, 1.0), target_sum)
        h = h_new
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def nesterov_subgradient(h0, n, target_sum, n_iters, lr_init, time_fn, rng):
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    t_val = 1.0
    y = h.copy()
    for it in range(n_iters):
        if time_fn() < 25:
            break
        corr = compute_corr(y)
        topk = np.argsort(corr)[-5:]
        k_star = topk[rng.randint(5)]
        w_single = np.zeros(2 * n - 1)
        w_single[k_star] = 1.0
        g = gradient_smooth_max(y, w_single)
        lr = lr_init / np.sqrt(t_val)
        h_new = np.clip(y - lr * g, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        mu = (t_val - 1.0) / t_new
        y = h_new + mu * (h_new - h)
        y = np.clip(y, 0.0, 1.0)
        h = h_new
        t_val = t_new
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def coordinate_descent(h0, n, target_sum, max_rounds, rng, time_fn):
    h = h0.copy()
    c5_cur = compute_c5(h)
    for rnd in range(max_rounds):
        if time_fn() < 25:
            break
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            best_val = c5_cur
            best_step = 0.0
            for sf in [-0.08, -0.04, -0.015, 0.015, 0.04, 0.08]:
                d = sf * (1.0 - 2.0 * h[i])
                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] + d, 0.0, 1.0)
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_val - 1e-15:
                    best_val = c5_new
                    best_step = d
            if best_step != 0.0:
                h[i] = np.clip(h[i] + best_step, 0.0, 1.0)
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True
        if not improved:
            break
    return h, c5_cur

def slsqp_optimize(h0, n, target_sum, alpha, maxiter):
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_and_weights(compute_corr(h), alpha)
        return v

    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, w = smooth_max_and_weights(compute_corr(h), alpha)
        return gradient_smooth_max(h, w)

    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)
```
