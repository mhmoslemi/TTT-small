def smooth_max_gradient_vec(h, alpha):

    sm_val, w = smooth_max_value_and_weight(corr, alpha)
    
    k_vals = np.arange(len(corr))

    idx_a = j_vals[:, None] + (n - 1) - k_vals[None, :]
    idx_b = j_vals[:, None] + k_vals[None, :] - (n - 1)
    
    # Create masked h arrays
    h_safe_a = np.where((idx_a >= 0) & (idx_a < n), h[np.clip(idx_a, 0, n-1)], 0.0)
    h_safe_b = np.where((idx_b >= 0) & (idx_b < n), h[np.clip(idx_b, 0, n-1)], 0.0)
    
    # A[j] = sum_k w[k] * h_safe_a[j,k]
    A = h_safe_a @ w  # shape (n,)
    B = h_safe_b @ w  # shape (n,)
    
    g = 1.0 - A - B
    return sm_val, g

```python
def project_box_sum(h, target_sum):
    """Project h onto {0<=x<=1, sum(x)=target_sum}"""
    n = len(h)
    # First clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    # Now adjust to get sum = target_sum
    # Use bisection on threshold: x_i = clip(h_i - t, 0, 1)
    # Find t such that sum(clip(h_i - t, 0, 1)) = target_sum
    
    # If sum is already correct, done
    if abs(np.sum(h) - target_sum) < 1e-14:
        return h
    
    # Bisection
    lo = -1.0  # h_i - lo >= 0 for all, so clip gives h_i
    hi = 2.0   # h_i - hi <= 0 for all, so clip gives 0
    
    for _ in range(200):
        mid = (lo + hi) / 2
        x = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(x)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
        if abs(s - target_sum) < 1e-15:
            break
    
    t = (lo + hi) / 2
    x = np.clip(h - t, 0.0, 1.0)
    # Final adjustment for floating point
    x += (target_sum - np.sum(x)) / n
    x = np.clip(x, 0.0, 1.0)
    return x
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_max_value_weight(corr, alpha):
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    if s < 1e-300:
        w = np.zeros(len(corr))
        w[int(np.argmax(corr))] = 1.0
        return m, w
    w = exp_vals / s
    sm_val = m + np.log(s) / alpha
    return sm_val, w

def smooth_max_grad_vec(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    sm_val, w = smooth_max_value_weight(corr, alpha)
    k_vals = np.arange(2 * n - 1)
    j_vals = np.arange(n)
    idx_a = j_vals[:, None] + (n - 1) - k_vals[None, :]
    idx_b = j_vals[:, None] + k_vals[None, :] - (n - 1)
    mask_a = (idx_a >= 0) & (idx_a < n)
    mask_b = (idx_b >= 0) & (idx_b < n)
    h_clip_a = np.clip(idx_a, 0, n - 1)
    h_clip_b = np.clip(idx_b, 0, n - 1)
    h_safe_a = np.where(mask_a, h[h_clip_a], 0.0)
    h_safe_b = np.where(mask_b, h[h_clip_b], 0.0)
    A = h_safe_a.dot(w)
    B = h_safe_b.dot(w)
    g = 1.0 - A - B
    return sm_val, g

def project_box_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:

def fista_with_backtrack(h0, n, target_sum, alpha, n_iters, time_limit):
    t_start = time.time()
    h = h0.copy()
    y = h0.copy()
    t_param = 1.0
    lr = 1.0 / (alpha + 2.0)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    for it in range(n_iters):
        if time.time() - t_start > time_limit:
            break
        sm_val, g = smooth_max_grad_vec(y, alpha)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:

def lbfgs_step(h0, n, target_sum, alpha, maxiter):
    penalty_coeff = 50.0

    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        corr = compute_corr(h)
        v, _ = smooth_max_value_weight(corr, alpha)
        p = penalty_coeff * (np.sum(h) - target_sum) ** 2
        return v + p

    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_grad_vec(h, alpha)
        g += 2.0 * penalty_coeff * (np.sum(h) - target_sum) * np.ones(n)
        return g

    try:
        res = minimize(obj, h0.copy(), jac=grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12, 'maxls': 30})
        return project_box_sum(res.x, target_sum)
    except Exception:
        return project_box_sum(h0, target_sum)

            best_val = c5_cur
            best_d = 0.0

                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] + d, 0.0, 1.0)

                h = project_box_sum(h, target_sum)
                c5_cur = best_val
                improved = True
        if not improved:
            break
