```python
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return float(np.max(corr) * dx)

def get_active_lags(h, tol=0.001):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full") * dx
    max_val = np.max(corr)
    return np.where(corr >= max_val - tol)[0]

def smooth_max(values, tau):
    v = values / tau
    vmax = np.max(v)
    return tau * (vmax + np.log(np.sum(np.exp(v - vmax))))

def objective_smooth(x, n, tau):
    dx = 2.0 / n
    g = 1.0 - x
    corr = np.correlate(x, g, mode="full")
    return smooth_max(corr, tau) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        if np.sum(proj) > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-14:
        idx = int(np.argmin(np.abs(result - 0.5)))
        result[idx] = np.clip(result[idx] + diff, 0.0, 1.0)
    return result

def pair_exchange_optimize(h, n, max_sweeps=50, base_delta=0.02):
    """Local search using pair exchanges that preserve the sum constraint."""
    best = h.copy()
    best_c5 = evaluate_c5(best)
    ar = np.arange(n)
    
    for sweep in range(max_sweeps):
        improved = False
        # Try pairs of positions
        for i in ar:
            for j in ar[i+1:]:
                # Try moving from i to j and from j to i
                for (src, dst) in [(i, j), (j, i)]:
                    if best[src] < 1e-8 or best[dst] > 1 - 1e-8:
                        continue

                    dist = abs(src - dst)
                    delta = base_delta / (1 + 0.1 * dist)
                    # Try a few delta values

                            best_c5 = c5
                            improved = True
                            delta = d  # remember good delta
                            break
        if not improved:
            break
    return best, best_c5
```

```python
            if 0 <= j - L < n:

            break  # No improvement possible
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_fn(values, tau):
    v = values / tau
    vmax = np.max(v)
    return tau * (vmax + np.log(np.sum(np.exp(v - vmax))))

def objective_smooth(x, n, tau):
    dx = 2.0 / n
    g = 1.0 - x
    corr = np.correlate(x, g, mode="full")
    return smooth_max_fn(corr, tau) * dx

def objective_grad(x, n, tau):
    dx = 2.0 / n
    g = 1.0 - x
    corr = np.correlate(x, g, mode="full")
    v = corr / tau
    vmax = np.max(v)
    exp_v = np.exp(v - vmax)
    weights = exp_v / np.sum(exp_v)
    # Gradient of smooth max: sum_k weights[k] * dc(k)/dx
    # dc(k)/dx[i] = (1-x[i-k]) * [0<=i-k<n] - x[i+k] * [0<=i+k<n]
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        k = k_idx - (n - 1)
        w = weights[k_idx]
        if w < 1e-10:
            continue
        for i in range(n):
            term = 0.0
            if 0 <= i - k < n:
                term += (1.0 - x[i - k])
            if 0 <= i + k < n:
                term -= x[i + k]
            grad[i] += w * term
    return smooth_max_fn(corr, tau) * dx, grad * dx
```

```python
                best_delta = d
        
        if best_delta < 1e-10 or (best_c5 - best_c5_trial) < min_improve:
            break
        
        best[p] -= best_delta
        best[q] += best_delta
        best_c5 = best_c5_trial
    
    return best, best_c5
```

Now, one potential issue: the gradient-guided exchange might get stuck in a local minimum where no single pair exchange improves the objective, but a sequence of exchanges would. This is the same issue as with coordinate descent.

To handle this, I can:
1. Allow "neutral" moves (that don't worsen the objective) to explore the landscape
2. Use simulated annealing (accept some worsening moves)
3. Try multiple random pair exchanges before concluding no improvement is possible

```python
            h_new[j] += d

            if current[j] < d or current[i] > 1 - d:
                continue

            h_new[j] -= d
            h_new[i] += d
        
        c5_new = evaluate_c5(h_new)

        if delta_c5 < 0 or np.random.random() < np.exp(-delta_c5 / temperature):
            current = h_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best = h_new.copy()
                best_c5 = c5_new
            # Anneal
            temperature *= 0.999
    
    return best, best_c5
```
