def modify_and_project(h, i, new_val, n):
    """Set h[i] = new_val and project to feasibility."""
    target = n / 2.0
    h = h.copy()
    h[i] = np.clip(new_val, 0.0, 1.0)
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-14:
        return h
    # Add diff to all other variables proportionally
    mask = np.ones(n, dtype=bool)
    mask[i] = False
    adjustable = h[mask]
    room_up = 1.0 - adjustable
    room_down = adjustable
    # Find how much we can add before hitting bounds
    if diff > 0:
        total_room = np.sum(room_up)
        if total_room < diff:
            # Can't add enough; set all to 1 where possible
            h[mask] = 1.0
            remaining = diff - total_room
            # Add remaining to variable i (might exceed 1)
            h[i] = min(1.0, h[i] + remaining)
            # If still not enough, we're infeasible (shouldn't happen)
        else:
            # Add uniformly
            h[mask] += diff / (n - 1)
    else:
        total_room = np.sum(room_down)
        if total_room < -diff:
            h[mask] = 0.0
            remaining = -diff - total_room
            h[i] = max(0.0, h[i] - remaining)
        else:
            h[mask] += diff / (n - 1)
    
    # Final clip and correction
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) > 1e-12:
        # Small correction
        idx = int(np.argmin(np.abs(h - 0.5)))
        h[idx] = np.clip(h[idx] + (target - s), 0.0, 1.0)
    return h

Actually wait - I just realized something. The evaluation uses `np.correlate(h, 1-h, mode="full") * dx`. The `mode="full"` gives 2n-1 values. The maximum over ALL these values is what we minimize. This includes lags from -(n-1) to n-1.

For lag k (positive), the correlation is sum_{i=0}^{n-1-k} h[i] * (1-h[i+k]).
For lag k (negative), it's sum_{i=-k}^{n-1} h[i+k] * (1-h[i]) = sum_{i=0}^{n-1+k} h[i] * (1-h[i-k]).

So the "effective" correlation at lag k only uses n-|k| terms (not all n terms). This means the correlation naturally decreases for large |k|, and the maximum is likely at small lags (k=0, 1, 2, ...).

At k=0: corr = sum h[i] * (1-h[i]) = sum h[i] - sum h[i]^2 = n/2 - sum h[i]^2.
To minimize this, we want to maximize sum h[i]^2, which means making h as "peaky" as possible (values close to 0 or 1).

But we also need to keep the other lags small. There's a trade-off.

This insight suggests that the optimal h should be close to a "square wave" (values near 0 or 1) with some structure to keep the autocorrelation flat.

import numpy as np

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    return np.correlate(h, g, mode="full") * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-13:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        if np.sum(proj) > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-13:
        idx = int(np.argmin(np.abs(result - 0.5)))
        result[idx] = np.clip(result[idx] + diff, 0.0, 1.0)
    return result

def modify_one(h, i, new_val, n):
    target = n / 2.0
    h2 = h.copy()
    h2[i] = np.clip(new_val, 0.0, 1.0)
    s = np.sum(h2)

        return h2

        total_room = np.sum(room)
        if total_room < diff * (1 - 1e-12):

def smooth_max(values, tau):
    v = values / tau
    vmax = np.max(v)
    return tau * (vmax + np.log(np.sum(np.exp(v - vmax))))

def slsqp_refine(h0, n, tau_list, max_iter, deadline):

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}]
    current = project_feasible(h0.copy(), n)
    best_c5 = evaluate_c5(current)
    for tau in tau_list:

            res = minimize(obj, current, method='SLSQP',
                          bounds=bounds, constraints=cons,
