```python
def evaluate_c5(h):

def project_feasible(h, n):

    # Find top_k critical lags
    idx = np.argsort(corr)[::-1][:top_k]

    for m in idx:

        # shift k = m - (n-1)

    h_new = project_feasible(h_new, n)
```

Wait, I need to reconsider. The subgradient step function I wrote has a nested loop which will be very slow for large n. Let me vectorize it.

For shift k (where k = m - (n-1)):
- First factor contribution: for j where 0<=j<n and 0<=j+k<n:
  - j ranges from max(0,-k) to min(n-1, n-1-k) (inclusive)
  - grad[j] += (1-h[j+k])
  
- Second factor contribution: for j where 0<=j-k<n and 0<=j<n:
  - j = i+k where 0<=i<n, so j ranges from k to k+n-1
  - Intersect with [0, n-1]: j from max(0,k) to min(n-1, k+n-1)
  - grad[j] -= h[j-k]

```python
def subgradient_step(h, top_k=1, lr=0.01):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full") * dx
    
    idx = np.argsort(corr)[::-1][:top_k]
    
    grad = np.zeros(n)
    for m in idx:
        k = m - (n - 1)
        # First factor: grad[j] += (1-h[j+k]) for valid j
        # j+k must be in [0, n-1] and j in [0, n-1]
        if k >= 0:
            j_idx = np.arange(0, n-k)
            grad[j_idx] += (1.0 - h[j_idx + k])
            # Second factor: grad[j] -= h[j-k] for valid j
            # j-k in [0, n-1] and j in [0, n-1] → j in [k, n-1]
            j2_idx = np.arange(k, n)
            grad[j2_idx] -= h[j2_idx - k]
        else:
            # k < 0, let kk = -k > 0
            kk = -k
            # j-k = j+kk in [0, n-1] → j in [0, n-1-kk] = [0, n-1+kk-1]... wait
            # j+k in [0, n-1] → j-kk in [0, n-1] → j in [kk, n-1+kk]... but j < n
            # So j in [kk, n-1] (since j < n and j >= kk)
            # Wait: j+k >= 0 → j-kk >= 0 → j >= kk
            # j+k <= n-1 → j-kk <= n-1 → j <= n-1+kk → always true since j < n <= n+kk
            # And j in [0, n-1]
            # So j in [max(0,kk), min(n-1, n-1+kk)] = [kk, n-1]
            j_idx = np.arange(kk, n)
            grad[j_idx] += (1.0 - h[j_idx - kk])
            # Second factor: j-k in [0, n-1] → j+kk in [0, n-1] → j in [-kk, n-1-kk]
            # j in [0, n-1] → j in [0, n-1-kk]
            j2_idx = np.arange(0, n-kk)
            grad[j2_idx] -= h[j2_idx + kk]
    
    h_new = h - lr * grad * dx
    h_new = project_feasible(h_new, n)
    return h_new, float(np.max(corr))
```

```python
import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        if np.sum(proj) > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-14:
        mask = (result > 0.005) & (result < 0.995)
        cnt = np.sum(mask)
        if cnt > 0:
            result[mask] += diff / cnt
            result = np.clip(result, 0.0, 1.0)
        else:
            idx = int(np.argmin(np.abs(result - 0.5)))
            result[idx] = np.clip(result[idx] + diff, 0.0, 1.0)
    return result

def subgradient_step(h, top_k=1, lr=0.05):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr_raw = np.correlate(h, g, mode="full")
    corr = corr_raw * dx
    idx = np.argsort(corr)[::-1][:top_k]
    grad = np.zeros(n)
    for m in idx:
        k = m - (n - 1)
        if k >= 0:
            ji = np.arange(0, n - k)
            grad[ji] += (1.0 - h[ji + k])
            j2 = np.arange(k, n)
            grad[j2] -= h[j2 - k]
        else:
            kk = -k
            ji = np.arange(kk, n)
            grad[ji] += (1.0 - h[ji - kk])
            j2 = np.arange(0, n - kk)
            grad[j2] -= h[j2 + kk]
    h_new = h - lr * grad * dx
    h_new = project_feasible(h_new, n)
    return h_new, float(np.max(corr))

def momentum_subgradient(h0, time_limit, top_k=3, lr0=0.05, noise_every=100):
    n = len(h0)
    best = h0.copy()
    best_c5 = evaluate_c5(best)
    velocity = np.zeros(n)
    beta = 0.8
    t0 = time.time()
    step = 0
    patience = 0
    rng_local = np.random.RandomState(int(time.time()) % 10000)
    while time.time() - t0 < time_limit:
        lr = lr0 * (0.997 ** step)
        lr = max(lr, lr0 * 0.05)
        h_new, c5_now = subgradient_step(best, top_k=top_k, lr=lr)
        velocity = beta * velocity + (1.0 - beta) * (h_new - best)
        h_cand = best + velocity
        h_cand = project_feasible(h_cand, n)
        c5_cand = evaluate_c5(h_cand)
        if c5_cand < best_c5:
            best = h_cand
            best_c5 = c5_cand
            patience = 0
        else:
            h_try = h_new
            c5_try = evaluate_c5(h_try)
            if c5_try < best_c5:
                best = h_try
                best_c5 = c5_try
                patience = 0
            else:
                patience += 1
        if patience > noise_every:
            noise_scale = 0.02 * (1.0 if best_c5 > 0.37 else 0.5)
            h_pert = best + rng_local.randn(n) * noise_scale
            h_pert = project_feasible(h_pert, n)
            c5_pert = evaluate_c5(h_pert)
            if c5_pert < best_c5:
                best = h_pert
                best_c5 = c5_pert
            patience = 0
        step += 1
    return best, best_c5

def sine_series_optimize(n, start_h, rng, time_limit):
    K = min(8, n // 6)
    hft = np.fft.rfft(start_h)
    D = 2 * K
    params = np.zeros(D)
    for k in range(K):
        kk = k + 1
        if kk < len(hft):
            params[2 * k] = 2.0 * np.abs(hft[kk]) / n
            params[2 * k + 1] = np.angle(hft[kk])
    x = np.linspace(0, 2, n, endpoint=False)

    def build_h(params_vec):
        h = np.full(n, 0.5)
        for k in range(K):
            kk = k + 1
            a = params_vec[2 * k]
            phi = params_vec[2 * k + 1]
            h += a * np.sin(kk * np.pi * x + phi)
        return np.clip(h, 0.0, 1.0)

    def obj(params_vec):
        hval = project_feasible(build_h(params_vec), n)
        return evaluate_c5(hval)

    best_c5 = obj(params)
    t0 = time.time()
    while time.time() - t0 < time_limit:
        try:
            res = minimize(obj, params, method='Nelder-Mead',
                          options={'maxiter': 150, 'xatol': 1e-7, 'fatol': 1e-9})
            if res.fun < best_c5:
                best_c5 = res.fun
                params = res.x.copy()
        except Exception:
            pass
        frac = (time.time() - t0) / time_limit
        noise = 0.008 * max(0.1, 1.0 - frac)
        params = params + rng.randn(D) * noise
        params[0::2] = np.clip(params[0::2], 0.0, 0.48)
    h_final = project_feasible(build_h(params), n)
    c5_final = evaluate_c5(h_final)
    return h_final, min(c5_final, best_c5)

def coordinate_refine(h, n, time_limit):
    best = h.copy()
    best_c5 = evaluate_c5(best)
    t0 = time.time()
    for sweep in range(30):
        if time.time() - t0 > time_limit:
            break
        improved = False
        for i in range(n):
            if time.time() - t0 > time_limit:
                break
            for d in [-0.015, -0.007, -0.003, 0.003, 0.007, 0.015]:
                h_new = best.copy()
                h_new[i] += d
                h_new = project_feasible(h_new, n)
                c5 = evaluate_c5(h_new)
                if c5 < best_c5 - 1e-12:
                    best = h_new
                    best_c5 = c5
                    improved = True
                    break
        if not improved:
            break
    return best, best_c5

def upsample_h(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate', bounds_error=False)
    h_new = f(x_new)
    return project_feasible(h_new, n_new)
```
