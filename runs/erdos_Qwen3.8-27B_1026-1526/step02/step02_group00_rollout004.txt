    if mu > 0:
        grad += 2.0 * mu * (np.sum(h) - n/2.0)
    
    return sm, grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_grad(h, alpha, mu=0.0):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    exp_c = np.exp(alpha * (corr - mc))
    w = exp_c / np.sum(exp_c)
    sm = mc + np.log(np.sum(exp_c)) / alpha
    if mu > 0:
        sm += mu * (np.sum(h) - n / 2.0) ** 2
    m_idx = np.arange(n)
    j_idx = np.arange(2 * n - 1)
    a = m_idx[:, None] - j_idx[None, :] + n - 1
    valid_a = (a >= 0) & (a < n)
    b = m_idx[:, None] + j_idx[None, :] - n + 1
    valid_b = (b >= 0) & (b < n)
    grad = np.zeros(n)
    if np.any(valid_a):
        a_clip = np.clip(a, 0, n - 1)
        grad += (valid_a * w[None, :] * (1.0 - h[a_clip])).sum(axis=1)
    if np.any(valid_b):
        b_clip = np.clip(b, 0, n - 1)
        grad -= (valid_b * w[None, :] * h[b_clip]).sum(axis=1)
    if mu > 0:
        grad += 2.0 * mu * (np.sum(h) - n / 2.0)
    return sm, grad

def lbfgs_optimize(h0, n, target_sum, alpha, mu, maxiter=500, time_limit=20):
    t0 = time.time()
    best_local = h0.copy()
    best_local_val = compute_c5(h0)
    def obj(h):
        val, _ = smooth_max_obj_grad(h, alpha, mu)
        return val
    def grad(h):
        _, g = smooth_max_obj_grad(h, alpha, mu)
        return g
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12})
        h_new = project_to_sum(res.x, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_local_val:
            best_local = h_new
            best_local_val = c5_new
    except Exception:
        pass
    return best_local, best_local_val

def subgradient_cycle(h, n, target_sum, n_iter=200, step_size=0.01):
    h = h.copy()
    for it in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        mc = np.max(corr)
        top_k = np.argsort(corr)[-5:]
        m_idx = np.arange(n)
        j_idx = np.arange(2 * n - 1)
        a = m_idx[:, None] - j_idx[None, :] + n - 1
        valid_a = (a >= 0) & (a < n)
        b = m_idx[:, None] + j_idx[None, :] - n + 1
        valid_b = (b >= 0) & (b < n)
        a_clip = np.clip(a, 0, n - 1)
        b_clip = np.clip(b, 0, n - 1)
        grad = np.zeros(n)
        for j in top_k:
            grad += valid_a[:, j] * (1.0 - h[a_clip[:, j]])
            grad -= valid_b[:, j] * h[b_clip[:, j]]
        grad /= len(top_k)
        h -= step_size * grad
        h = project_to_sum(h, target_sum)
        step_size *= 0.999
    return h

def make_ramp_start(n, rng, target_sum):
    x = np.linspace(0, 2.0, n, endpoint=False)
    n_ramps = rng.randint(1, 5)
    h = np.full(n, 0.5)
    for _ in range(n_ramps):
        x0 = rng.uniform(0, 2.0)
        width = rng.uniform(0.05, 0.3)
        amp = rng.uniform(0.2, 0.5) * rng.choice([-1, 1])
        h += amp * np.tanh((x - x0) / width)
    h = project_to_sum(h, target_sum)
    return h

def make_fourier_start(n, M, rng, target_sum):
    x = np.linspace(0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for m in range(1, M + 1):
        a = rng.uniform(-0.3, 0.3)
        b = rng.uniform(-0.3, 0.3)
        h += a * np.cos(np.pi * m * x) + b * np.sin(np.pi * m * x)
    h = project_to_sum(h, target_sum)
    return h

def make_binary_structured(n, rng, target_sum):

def coordination_pattern_search(h, n, target_sum, rng, max_rounds=100, n_trials=100):
    h = h.copy()
    c5_cur = compute_c5(h)
    for r in range(max_rounds):
        improved = False
        best_c5 = c5_cur
        best_move = None
        for _ in range(n_trials):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            h_test = h.copy()
            if h_test[i] < 0.5 and h_test[j] > 0.5:
                h_test[i] += 0.1
                h_test[j] -= 0.1
            elif h_test[i] > 0.5 and h_test[j] < 0.5:
                h_test[i] -= 0.1
                h_test[j] += 0.1
            else:
                continue
            h_test = np.clip(h_test, 0, 1)
            c5_test = compute_c5(h_test)
            if c5_test < best_c5 - 1e-12:
                best_c5 = c5_test
                best_move = h_test.copy()
        if best_move is None:
            break
        h = best_move
        c5_cur = best_c5
        improved = True
    return h, c5_cur

def interpolate_h(h_old, n_old, n_new, target_sum):
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, np.concatenate([h_old, [h_old[0]]]),
                      period=2.0) if hasattr(np.interp, '__call__') else np.interp(x_new, x_old, h_old)
    h_new = np.clip(h_new, 0, 1)
    return project_to_sum(h_new, target_sum)

            if time_left() < 15:
                break
            h_pol, c5_pol = lbfgs_optimize(best_h, n_f, ts_f, alpha, 0.1, maxiter=600, time_limit=12)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()
                best_n = n_f

    if time_left() > 20:
        n_down = max(120, best_n // 2)
        ts_down = n_down / 2.0
        if abs(ts_down - round(ts_down)) < 1e-6:
            h_down = interpolate_h(best_h, best_n, n_down, ts_down)
            for alpha in [100, 1000, 5000]:
                if time_left() < 10:
                    break
                h_d, c5_d = lbfgs_optimize(h_down, n_down, ts_down, alpha, 5.0, maxiter=400, time_limit=8)
                if c5_d < best_c5:
                    best_c5 = c5_d
                    best_h = h_d.copy()
                    best_n = n_down

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
