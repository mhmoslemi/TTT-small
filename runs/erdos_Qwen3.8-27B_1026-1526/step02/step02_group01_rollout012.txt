def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_value_and_weights(corr, alpha):
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / len(exp_vals)) / alpha
    return sm_val, w

def compute_gradient(h, w, dx):
    """Compute gradient of smooth max of corr*dx w.r.t. h.
    h: length n
    w: weights for each lag, length 2n-1, w[j] for lag k=j-(n-1)
    dx: step size
    """
    n = len(h)
    grad = np.zeros(n)
    offset = n - 1
    
    for i in range(n):

        k_lo = max(-i, -(n-1))

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -600.0, 600.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / len(exp_vals)) / alpha * dx

    offset = n - 1
    grad = np.zeros(n)
    for i in range(n):
        k_lo = max(-i, -(n - 1))
        k_hi = min(n - 1 - i, n - 1)
        if k_lo <= k_hi:
            ks = np.arange(k_lo, k_hi + 1)
            idx_h = i + ks
            idx_w = ks + offset
            grad[i] += np.dot(w[idx_w], 1.0 - h[idx_h])
        k_lo2 = max(i - (n - 1), -(n - 1))
        k_hi2 = min(i, n - 1)
        if k_lo2 <= k_hi2:
            ks2 = np.arange(k_lo2, k_hi2 + 1)
            idx_h2 = i - ks2
            idx_w2 = ks2 + offset
            grad[i] -= np.dot(w[idx_w2], h[idx_h2])
    grad *= dx
    return sm_val, grad

def nesterov_gradient(h0, n, target_sum, alpha, n_iters, lr_init, momentum=0.9):
    h = h0.copy()
    y = h.copy()
    c5_cur = compute_c5(h)
    h_best = h.copy()
    lr = lr_init
    v = np.zeros(n)
    stall = 0
    for it in range(n_iters):
        _, g = smooth_max_and_grad(y, alpha)
        v = momentum * v + (1.0 - momentum) * g
        h_new = np.clip(y - lr * v, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur - 1e-15:
            h = h_new.copy()
            y = h + momentum * (h - y)
            y = np.clip(y, 0.0, 1.0)
            c5_cur = c5_new
            h_best = h.copy()
            lr = min(lr * 1.03, 0.3)
            stall = 0
        else:
            lr *= 0.75
            y = h.copy()
            stall += 1
            if stall > 500:
                break
    return h_best, c5_cur

def backtracking_gradient(h0, n, target_sum, alpha, n_iters, lr_init):
    h = h0.copy()
    c5_cur = compute_c5(h)
    h_best = h.copy()
    lr = lr_init
    stall = 0
    for it in range(n_iters):
        _, g = smooth_max_and_grad(h, alpha)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        g = g / g_norm
        for _bt in range(30):
            h_new = np.clip(h - lr * g * g_norm, 0.0, 1.0)
            h_new = project_to_sum(h_new, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-15:
                break
            lr *= 0.5
        else:
            lr *= 0.5
            stall += 1
            if stall > 200:
                break
            continue
        h = h_new
        c5_cur = c5_new
        h_best = h.copy()
        lr = min(lr * 1.1, 0.5)
        stall = 0
    return h_best, c5_cur

def coordinate_descent(h0, n, target_sum, max_rounds, rng, step_sizes=None):
    h = h0.copy()
    c5_cur = compute_c5(h)
    if step_sizes is None:
        step_sizes = np.array([0.02, 0.01, 0.005, 0.002, 0.001, -0.001, -0.002, -0.005, -0.01, -0.02])
    for rnd in range(max_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            hi = h[i]
            best_val = c5_cur
            best_d = 0.0
            for d in step_sizes:
                hi_try = hi + d
                if hi_try < 0.0 or hi_try > 1.0:
                    continue
                h[i] = hi_try
                h = project_to_sum(h, target_sum)
                c5_try = compute_c5(h)
                if c5_try < best_val - 1e-15:
                    best_val = c5_try
                    best_d = d
            h[i] = hi + best_d
            h = project_to_sum(h, target_sum)
            c5_cur = best_val
            if best_d != 0.0:
                improved = True
        if not improved:
            break
    return h, c5_cur

def make_ramp(n):
    x = np.linspace(0, 2, n, endpoint=False)
    h = 1.0 - x / 2.0
    h[-1] = 1.0 - (2.0 - 2.0/n) / 2.0
    return h

def make_sinusoidal(n, freq, phase, amp):
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + amp * np.sin(2.0 * np.pi * freq * x / 2.0 + phase)
    return h

def make_tapered_ramp(n, p):
    x = np.linspace(0, 2, n, endpoint=False)
    t = 1.0 - x / 2.0
    h = t ** p
    s = np.sum(h)
    if s > 0:
        h *= (n / 2.0) / s
        h = np.clip(h, 0.0, 1.0)
    return h

def make_block_pattern(n, period, duty):
    h = np.zeros(n)
    for s in range(0, n, period):
        h[s:min(s + int(period * duty), n)] = 1.0
    return h

def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    remaining = lambda: budget_s - (time.time() - t_start)
    rng = np.random.RandomState(seed)
    best_h = None

    if 'initial_h_values' in globals() and initial_h_values is not None:

            best_n = len(best_h)

    if best_h is None:

    n_list = [360]
    if remaining() > 400:
        n_list.append(480)

    for n in n_list:
        if remaining() < 150:
            break
        if n % 2 != 0:
            continue
        ts = n / 2.0
        if abs(ts - round(ts)) > 1e-9:

        starts = []
        if best_n == n:

        else:
