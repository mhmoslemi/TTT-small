import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_value(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    return float(m + np.log(s / len(exp_vals)) / alpha), exp_vals / s

def smooth_max_grad_vec(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -700.0, 700.0)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    sm_val = float(m + np.log(np.sum(exp_vals) / len(exp_vals)) / alpha)
    n = len(h)
    one_minus_h = 1.0 - h
    h_rev = h[::-1].copy()
    corr1 = np.correlate(w, one_minus_h, mode='full')
    corr2 = np.correlate(w, h_rev, mode='full')
    grad = corr1[n-1:2*n-1][::-1] - corr2[n-1:2*n-1]
    return sm_val, grad

def slsqp_optimize(h0, n, target_sum, alpha, maxiter):
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_value(h, alpha)
        return v
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_grad_vec(h, alpha)
        return g
    def con_fun(x):
        return np.sum(x) - target_sum
    def con_jac(x):
        return np.ones(n)
    constraints = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def nesterov_descent(h0, n, target_sum, alpha, n_iters, lr_init):
    h = h0.copy().astype(float)
    y = h.copy()
    t = 1.0
    lr = lr_init
    c5_cur = compute_c5(h)
    h_best = h.copy()
    for it in range(n_iters):
        _, g = smooth_max_grad_vec(y, alpha)
        h_new = np.clip(y - lr * g, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        mo = (t - 1.0) / t_new
        t = t_new
        y = h_new + mo * (h_new - h)
        y = np.clip(y, 0.0, 1.0)
        h = h_new
        c5_new = compute_c5(h)
        if c5_new < c5_cur - 1e-15:
            c5_cur = c5_new
            h_best = h.copy()
            lr = min(lr * 1.03, 0.3)
        else:
            lr *= 0.75
            if lr < 1e-7:
                lr = lr_init * 0.1
    return h_best, c5_cur

def subgradient_nesterov(h0, n, target_sum, n_iters, lr_init):
    h = h0.copy().astype(float)
    y = h.copy()
    t = 1.0
    lr = lr_init
    c5_cur = compute_c5(h)
    h_best = h.copy()
    for it in range(n_iters):
        corr = np.correlate(y, 1.0 - y, mode='full')
        k_star = int(np.argmax(corr))
        n_local = len(y)
        g = np.zeros(n_local)
        h_y = y
        km = k_star - (n_local - 1)
        if km >= 0:
            idx = range(n_local - km)
            for i in idx:
                g[i] += h_y[i]
                g[i + km] -= h_y[i]
        else:
            idx = range(-km, n_local)
            for i in idx:
                g[i] += (1.0 - h_y[i + km])
                g[i + km] -= (1.0 - h_y[i])
        g = g / n_local
        h_new = np.clip(y - lr * g, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        mo = (t - 1.0) / t_new
        t = t_new
        y = h_new + mo * (h_new - h)
        y = np.clip(y, 0.0, 1.0)
        h = h_new
        c5_new = compute_c5(h)
        if c5_new < c5_cur - 1e-15:
            c5_cur = c5_new
            h_best = h.copy()
            lr = min(lr * 1.01, 0.2)
        else:
            lr *= 0.9
            if lr < 1e-8:
                lr = lr_init * 0.5
    return h_best, c5_cur

def coordinate_descent(h0, n, target_sum, max_rounds, rng):
    h = h0.copy().astype(float)
    c5_cur = compute_c5(h)
    for rnd in range(max_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            h_orig = h[i]
            for scale in [0.01, 0.003, -0.01, -0.003]:
                d = scale * (1.0 - 2.0 * h[i])
                h[i] = np.clip(h[i] + d, 0.0, 1.0)
                h = project_to_sum(h, target_sum)
                c5_new = compute_c5(h)
                if c5_new < c5_cur - 1e-15:
                    c5_cur = c5_new
                    improved = True
                    break
                h[i] = h_orig
        if not improved:
            break
    return h, c5_cur

def random_perturb(h0, n, target_sum, rng, strength):
    h = h0.copy().astype(float)
    n_pert = max(1, n // 10)
    idx = rng.choice(n, n_pert, replace=False)
    h[idx] += rng.uniform(-strength, strength, n_pert)
    h = np.clip(h, 0.0, 1.0)
    h = project_to_sum(h, target_sum)
    return h

def sentinel_initial_h_values(n):
    return np.full(n, 0.5)
