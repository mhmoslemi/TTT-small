import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / len(exp_vals)) / alpha
    g = np.zeros(n)
    active = np.where(w > 1e-12)[0]
    for idx in active:
        k = idx - (n - 1)
        wi = w[idx]
        if k >= 0:
            g[:n-k] += wi * (1.0 - h[k:])
            g[k:] -= wi * h[:n-k]
        else:
            ka = -k
            g[ka:] += wi * (1.0 - h[:n-ka])
            g[:n-ka] -= wi * h[ka:]
    return sm_val, g

def lag_gradient(h, k):
    """Gradient of a single correlation at lag k."""
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        g[:n-k] += (1.0 - h[k:])
        g[k:] -= h[:n-k]
    else:
        ka = -k
        g[ka:] += (1.0 - h[:n-ka])
        g[:n-ka] -= h[ka:]
    return g

def alternating_minimax(h0, n, target_sum, n_iters, t_start, budget_s):
    """Alternate: find worst lag, do projected gradient on it."""
    h = h0.copy()
    c5_cur = compute_c5(h)
    h_best = h.copy()
    lr = 0.1
    for it in range(n_iters):
        if time.time() - t_start > budget_s - 30:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = np.argmax(corr) - (n - 1)
        g = lag_gradient(h, k_star)
        h_new = np.clip(h - lr * g, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur:
            c5_cur = c5_new
            h_best = h_new.copy()
            h = h_new
            lr = min(lr * 1.1, 0.5)
        else:
            lr *= 0.7
    return h_best, c5_cur

def lbfgs_optimize(h0, n, target_sum, alpha, maxiter, t_start, budget_s, fallback_s=40):
    """L-BFGS-B with smooth max, projecting sum after each step."""
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_and_grad(h, alpha)
        return v
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_and_grad(h, alpha)
        return g
    
    # Handle sum constraint via penalty (simple approach)
    def obj_pen(x):
        h = np.clip(x, 0.0, 1.0)
        v, g = smooth_max_and_grad(h, alpha)
        penalty = 100.0 * (np.sum(h) - target_sum)
        gp = g + 100.0 * np.ones(n)
        return v + penalty, gp
    
    # Actually, let's just use SLSQP but with better settings
    # Or: optimize h[0:n-1] with h[n-1] = target_sum - sum(h[0:n-1])
    
    # Simplest: use the projection approach with Nesterov
    h = h0.copy()
    c5_cur = compute_c5(h)
    h_best = h.copy()
    y = h.copy()
    t_mom = 1.0
    lr = 0.05

            break
        h_trial = np.clip(y, 0.0, 1.0)

        mo = (t_mom - 1.0) / t_new

        y = np.clip(y, 0.0, 1.0)

def coordinate_descent(h0, n, target_sum, max_rounds, rng):
    h = h0.copy()
    c5_cur = compute_c5(h)
    for rnd in range(max_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            # Try several step sizes
            for d in [-0.02, -0.01, -0.003, 0.003, 0.01, 0.02]:
                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] + d, 0.0, 1.0)

                if c5_new < c5_cur - 1e-15:
                    h = h_new
                    c5_cur = c5_new
                    improved = True

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def lag_gradient(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        g[:n-k] += (1.0 - h[k:])
        g[k:] -= h[:n-k]
    else:
        ka = -k
        g[ka:] += (1.0 - h[:n-ka])
        g[:n-ka] -= h[ka:]
    return g

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / len(exp_vals)) / alpha
    g = np.zeros(n)
    active = np.where(w > 1e-12)[0]
    for idx in active:
        k = idx - (n - 1)
        wi = w[idx]
        if k >= 0:
            g[:n-k] += wi * (1.0 - h[k:])
            g[k:] -= wi * h[:n-k]
        else:
            ka = -k
            g[ka:] += wi * (1.0 - h[:n-ka])
            g[:n-ka] -= wi * h[ka:]
    return sm_val, g

def alternating_minimax(h0, n, target_sum, max_iters, t_start, budget_s, fallback=30):
    h = h0.copy()
    c5_cur = compute_c5(h)
    h_best = h.copy()
    lr = 0.08
    for it in range(max_iters):
        if time.time() - t_start > budget_s - fallback:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr)) - (n - 1)
        g = lag_gradient(h, k_star)
        h_new = np.clip(h - lr * g, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur - 1e-16:
            c5_cur = c5_new
            h_best = h_new.copy()
            h = h_new
            lr = min(lr * 1.03, 0.4)
        else:
            lr *= 0.75
    return h_best, c5_cur

def top_lag_descent(h0, n, target_sum, max_iters, t_start, budget_s, top_m=3, fallback=30):
    h = h0.copy()
    c5_cur = compute_c5(h)
    h_best = h.copy()
    lr = 0.05
    for it in range(max_iters):
        if time.time() - t_start > budget_s - fallback:

def nesterov_smooth(h0, n, target_sum, alpha, max_iters, t_start, budget_s, fallback=30):
    h = h0.copy()
    c5_cur = compute_c5(h)
    h_best = h.copy()
    y = h.copy()
    t_mom = 1.0
    lr = 0.04
    for it in range(max_iters):
        if time.time() - t_start > budget_s - fallback:

        h_trial = project_to_sum(h_trial, target_sum)
        _, g = smooth_max_and_grad(h_trial, alpha)

        c5_new = compute_c5(h)
        if c5_new < c5_cur - 1e-16:
            c5_cur = c5_new
            h_best = h.copy()
            lr = min(lr * 1.01, 0.25)
        else:
            lr *= 0.82
    return h_best, c5_cur

def pair_swap_descent(h0, n, target_sum, max_rounds, rng, t_start, budget_s, fallback=25):
    h = h0.copy()
    c5_cur = compute_c5(h)
    for rnd in range(max_rounds):
        if time.time() - t_start > budget_s - fallback:
            break
        improved = False
        n_trial = n * 4
        for _ in range(n_trial):
            i, j = rng.choice(n, size=2, replace=False)
            d = rng.uniform(-0.02, 0.02)
            hi = h[i] + d
            hj = h[j] - d
            if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                h[i], h[j] = hi, hj
                c5_new = compute_c5(h)
                if c5_new < c5_cur - 1e-15:
                    c5_cur = c5_new
                    improved = True
                else:
                    h[i], h[j] = hj, hi
        if not improved:
            break
    return h, c5_cur

def coord_fine_descent(h0, n, target_sum, max_rounds, rng, t_start, budget_s, fallback=25):
    h = h0.copy()
    c5_cur = compute_c5(h)
    step_sizes = [0.005, 0.002, 0.001, -0.001, -0.002, -0.005]
    for rnd in range(max_rounds):
        if time.time() - t_start > budget_s - fallback:
            break
        improved = False
        for i in rng.permutation(n):
            for d in step_sizes:
                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] + d, 0.0, 1.0)
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < c5_cur - 1e-15:
                    h = h_new
                    c5_cur = c5_new
                    improved = True
                    break
        if not improved:
            break
    return h, c5_cur

def slsqp_round(h0, n, target_sum, alpha, maxiter, t_start, budget_s, fallback=30):
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_and_grad(h, alpha)
        return v
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_and_grad(h, alpha)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    def time_left():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            ts = best_n / 2.0
            best_h = project_to_sum(best_h, ts)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 360
        best_h = np.full(360, 0.5)
        best_c5 = compute_c5(best_h)

    n_work = best_n
    ts = n_work / 2.0

    # Phase 1: Fast alternating minimax from parent
    if time_left() > 120:
        h1, c5_1 = alternating_minimax(best_h, n_work, ts, 8000, t_start, budget_s, fallback=60)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()

    # Phase 2: Top-k lag descent (focus on worst lags)
    if time_left() > 100:
        for top_m in [2, 3, 5]:
            if time_left() < 60:
                break
            h2, c5_2 = top_lag_descent(best_h, n_work, ts, 6000, t_start, budget_s, top_m=top_m, fallback=50)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h2.copy()

    # Phase 3: Nesterov with smooth max at multiple alphas
    if time_left() > 120:
        for alpha in [15.0, 40.0, 80.0, 150.0, 300.0]:
            if time_left() < 50:
                break
            n_it = int(min(3000, time_left() * 1.5))
            h3, c5_3 = nesterov_smooth(best_h, n_work, ts, alpha, n_it, t_start, budget_s, fallback=40)
            if c5_3 < best_c5:
                best_c5 = c5_3
                best_h = h3.copy()
            if time_left() < 50:
                break

    # Phase 4: SLSQP refinement
    if time_left() > 80:
        for alpha in [50.0, 150.0, 400.0]:
            if time_left() < 30:
                break
            h4 = slsqp_round(best_h, n_work, ts, alpha, 300, t_start, budget_s, fallback=25)
            c5_4 = compute_c5(h4)
            if c5_4 < best_c5:
                best_c5 = c5_4
                best_h = h4.copy()

    # Phase 5: Pair swap and fine coordinate descent
    if time_left() > 60:
        h5, c5_5 = pair_swap_descent(best_h, n_work, ts, 30, rng, t_start, budget_s, fallback=30)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()
    if time_left() > 40:
        h6, c5_6 = coord_fine_descent(best_h, n_work, ts, 20, rng, t_start, budget_s, fallback=25)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    # Phase 6: Try slightly higher resolution
    if time_left() > 150:
        n_hi = n_work + 60
        if n_hi % 2 == 0:
            ts_hi = n_hi / 2.0
            x_old = np.linspace(0, 2, n_work, endpoint=False)
            x_new = np.linspace(0, 2, n_hi, endpoint=False)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_to_sum(h_interp, ts_hi)
            c5_hi = compute_c5(h_interp)
            h_hi_best = h_interp.copy()
            if c5_hi < best_c5:
                best_c5 = c5_hi
                best_h = h_interp.copy()
                best_n = n_hi
            if time_left() > 80:
                h_hi, c5_hi2 = alternating_minimax(h_hi_best, n_hi, ts_hi, 5000, t_start, budget_s, fallback=40)
                if c5_hi2 < best_c5:
                    best_c5 = c5_hi2
                    best_h = h_hi.copy()
                    best_n = n_hi
                h_hi = h_hi_best.copy()
            if time_left() > 60:
                h_hi, c5_hi3 = top_lag_descent(h_hi_best, n_hi, ts_hi, 4000, t_start, budget_s, top_m=3, fallback=30)
                if c5_hi3 < best_c5:
                    best_c5 = c5_hi3
                    best_h = h_hi.copy()
                    best_n = n_hi
            if time_left() > 40:
                h_hi, c5_hi4 = nesterov_smooth(h_hi_best, n_hi, ts_hi, 50.0, 2000, t_start, budget_s, fallback=30)
                if c5_hi4 < best_c5:
                    best_c5 = c5_hi4
                    best_h = h_hi.copy()
                    best_n = n_hi

    # Phase 7: Diverse restarts at working resolution
    if time_left() > 100:
        n_work2 = best_n
        ts2 = n_work2 / 2.0
        x = np.linspace(0, 2, n_work2, endpoint=False)
        diverse_starts = [best_h.copy()]
        for freq in [2, 3, 5, 7]:
            hs = 0.5 + 0.3 * np.sin(2 * np.pi * freq * x / 2.0)
            diverse_starts.append(project_to_sum(hs, ts2))
        hr = rng.uniform(0.3, 0.7, n_work2)
        diverse_starts.append(project_to_sum(hr, ts2))
        for h_ds in diverse_starts:
            if time_left() < 40:
                break
            c5_ds = compute_c5(h_ds)
            if c5_ds < best_c5:
                best_c5 = c5_ds
                best_h = h_ds.copy()
                best_n = n_work2
            if time_left() < 30:
                break
            h_ds2, c5_ds2 = alternating_minimax(h_ds, n_work2, ts2, 4000, t_start, budget_s, fallback=25)
            if c5_ds2 < best_c5:
                best_c5 = c5_ds2
                best_h = h_ds2.copy()
                best_n = n_work2

    # Final polish
    if time_left() > 30:
        n_final = len(best_h)
        ts_final = n_final / 2.0
        h_pol, c5_pol = alternating_minimax(best_h, n_final, ts_final, 5000, t_start, budget_s, fallback=15)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
    if time_left() > 20:
        n_final = len(best_h)
        ts_final = n_final / 2.0
        h_pol2, c5_pol2 = pair_swap_descent(best_h, n_final, ts_final, 15, rng, t_start, budget_s, fallback=15)
        if c5_pol2 < best_c5:
            best_c5 = c5_pol2
            best_h = h_pol2.copy()
    if time_left() > 10:
        n_final = len(best_h)
        ts_final = n_final / 2.0
        h_pol3, c5_pol3 = top_lag_descent(best_h, n_final, ts_final, 3000, t_start, budget_s, top_m=2, fallback=10)
        if c5_pol3 < best_c5:
            best_c5 = c5_pol3
            best_h = h_pol3.copy()

    n_out = len(best_h)
    ts_out = n_out / 2.0
    best_h = project_to_sum(best_h, ts_out)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_out)
