    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            cnt = np.count_nonzero(mask)
            h[mask] += diff / cnt
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def get_corr_and_weights(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / (2 * n - 1)) / alpha
    return sm_val, w

def grad_from_weights(h, w):
    n = len(h)
    g = np.zeros(n)
    one_minus_h = 1.0 - h
    for j in range(n):
        m1_start = n - 1 - j
        m1_end = 2 * n - 1 - j  # exclusive
        g[j] += np.dot(w[m1_start:m1_end], one_minus_h)
        m2_start = j
        m2_end = j + n
        g[j] -= np.dot(w[m2_start:m2_end], h[::-1])
    return g

    target = n / 2.0
    big_lambda = 5000.0

        sm, w = get_corr_and_weights(h, alpha)
        pen = big_lambda * (np.sum(h) - target) ** 2
        return sm + pen

        sm, w = get_corr_and_weights(h, alpha)
        g = grad_from_weights(h, w)
        g += 2 * big_lambda * (np.sum(h) - target)
        return g

    try:
        res = minimize(obj, h0.copy(), jac=grad, method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-9})
        return project_to_sum(res.x, target)
    except Exception:
        return project_to_sum(h0, target)

    h = h0.copy()
    c5_cur = compute_c5(h)
    h_best = h.copy()
    momentum = np.zeros(n)
    beta = 0.9
    t = 1.0

        g = grad_from_weights(y, w)

        g += 2 * 100.0 * (np.sum(y) - target)

        y = np.clip(y, 0.0, 1.0)
        y = project_to_sum(y, target)
        
        h = h_new
        t = t_new

            c5_cur = c5_new
            h_best = h.copy()
            lr *= 1.01
        else:

            y = h.copy()

    h_best = h.copy()

        # Minimize: c^T x where x = [h_0, ..., h_{n-1}, t]
        # c = [0, ..., 0, 1]

from scipy.optimize import minimize, linprog
from scipy import sparse
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.count_nonzero(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def get_corr_weights(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / (2 * n - 1)) / alpha
    return sm_val, w

def grad_from_weights(h, w):
    n = len(h)
    g = np.zeros(n)
    one_minus_h = 1.0 - h
    h_rev = h[::-1]
    for j in range(n):
        g[j] += np.dot(w[n - 1 - j:2 * n - 1 - j], one_minus_h)
        g[j] -= np.dot(w[j:j + n], h_rev)
    return g

def lbfgs_smooth_max(h0, n, alpha, maxiter):
    target = n / 2.0
    big_lambda = 5000.0

    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        sm, w = get_corr_weights(h, alpha)
        pen = big_lambda * (np.sum(h) - target) ** 2
        return sm + pen

    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, w = get_corr_weights(h, alpha)
        g = grad_from_weights(h, w)
        g += 2.0 * big_lambda * (np.sum(h) - target)
        return g

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-9})
        return project_to_sum(np.clip(res.x, 0.0, 1.0), target)
    except Exception:
        return project_to_sum(h0, target)

        cols = []
        data = []

            rows.append(m)
            cols.append(n)
            data.append(-1.0)
        A_ub = sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n + 1))

                h_new = project_to_sum(h_new, target)
                c5_new = compute_c5(h_new)
                if c5_new < c5_cur:
                    c5_cur = c5_new
                    h_best = h_new.copy()
                h = h_new
        except Exception:
            pass
    return h_best, c5_cur

def projected_gradient(h0, n, alpha, n_iters, lr_init):
    target = n / 2.0
    h = h0.copy()
    c5_cur = compute_c5(h)
    h_best = h.copy()
    y = h.copy()
    lr = lr_init
    beta = 0.9
    momentum = np.zeros(n)
    t = 1.0
    for it in range(n_iters):
        _, w = get_corr_weights(y, alpha)
        g = grad_from_weights(y, w)
        g += 2.0 * 50.0 * (np.sum(y) - target)
        momentum = beta * momentum + (1.0 - beta) * g
        h_new = np.clip(y - lr * momentum, 0.0, 1.0)
        h_new = project_to_sum(h_new, target)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        mo = (t - 1.0) / t_new
        y = h_new + mo * (h_new - h)
        y = np.clip(y, 0.0, 1.0)
        y = project_to_sum(y, target)
        h = h_new
        t = t_new
        c5_new = compute_c5(h)
        if c5_new < c5_cur:
            c5_cur = c5_new
            h_best = h.copy()
            lr = min(lr * 1.02, 0.1)
        else:
            lr *= 0.85
            y = h.copy()
    return h_best, c5_cur

def coord_descent(h0, n, max_rounds, rng):
    target = n / 2.0
    h = h0.copy()
    c5_cur = compute_c5(h)
    deltas = [0.002, 0.008, 0.02, 0.05, -0.002, -0.008, -0.02, -0.05]
    for rnd in range(max_rounds):
        improved = False
        for i in rng.permutation(n):
            best = c5_cur
            best_d = 0.0
            for d in deltas:
                nv = h[i] + d
                if nv < -1e-9 or nv > 1.0 + 1e-9:
                    continue
                h[i] = nv
                h = project_to_sum(h, target)
                c5_new = compute_c5(h)
                if c5_new < best:
                    best = c5_new
                    best_d = d
            if best_d != 0.0:
                c5_cur = best
                improved = True
            else:
                h[i] = np.clip(h[i] - best_d, 0.0, 1.0)
                h = project_to_sum(h, target)
        if not improved:
            break
    return h, c5_cur

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()

    def _tl():
        return budget_s - (time.time() - t0)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 360
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n1 = best_n
    h1 = best_h.copy()

    if _tl() > 180:
        h1, c5_lp = alternating_lp(h1, n1, 150, 160)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h1.copy()
            best_n = n1

    if _tl() > 100:
        alphas_p1 = [10.0, 40.0, 150.0, 500.0, 2000.0]
        for alpha in alphas_p1:
            if _tl() < 30:
                break
            mi = max(30, min(200, int(_tl() * 0.4)))
            h_opt = lbfgs_smooth_max(h1, n1, alpha, mi)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n1
            h1 = h_opt

    if _tl() > 80:
        n_pg = min(1500, int(_tl() * 1.5))
        h_pg, c5_pg = projected_gradient(h1, n1, 300.0, n_pg, 0.015)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()
            best_n = n1
        h1 = h_pg

    n2 = 480
    if _tl() > 350:
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n2, endpoint=False)
        h2 = np.interp(x_new, x_old, best_h)
        h2 = project_to_sum(h2, n2 / 2.0)
        c5_2 = compute_c5(h2)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
            best_n = n2

        if _tl() > 180:
            h2, c5_lp2 = alternating_lp(h2, n2, 120, 170)
            if c5_lp2 < best_c5:
                best_c5 = c5_lp2
                best_h = h2.copy()
                best_n = n2

        alphas_p2 = [10.0, 40.0, 150.0, 500.0, 2000.0, 6000.0]
        for alpha in alphas_p2:
            if _tl() < 30:
                break
            mi = max(20, min(150, int(_tl() * 0.3)))
            h_opt = lbfgs_smooth_max(h2, n2, alpha, mi)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n2
            h2 = h_opt

        if _tl() > 60:
            n_pg2 = min(1000, int(_tl()))
            h_pg2, c5_pg2 = projected_gradient(h2, n2, 800.0, n_pg2, 0.008)
            if c5_pg2 < best_c5:
                best_c5 = c5_pg2
                best_h = h_pg2.copy()
                best_n = n2
            h2 = h_pg2

    if _tl() > 40:
        n_f = best_n
        h_f = best_h.copy()
        nr = min(40, int(_tl() / 4))
        h_cd, c5_cd = coord_descent(h_f, n_f, nr, rng)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    if _tl() > 20:
        n_f = best_n
        h_f = best_h.copy()
        h_pol = lbfgs_smooth_max(h_f, n_f, 5000.0, 100)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    n_final = len(best_h)
    best_h = project_to_sum(best_h, n_final / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_final)
