    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return float(np.max(corr) * dx)

def compute_all_f(h):
    """Compute f_s(h) for all shifts s"""
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return corr * dx

def compute_subgradient(h, active_set):
    """Compute subgradient of C5 = max f_s at h, using given active set"""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for s in active_set:
        if s >= 0:
            # f_s(h) = dx * sum_{i=0}^{n-1-s} h[i+s] * (1-h[i])
            # ∂f_s/∂h[j] = dx * [(1-h[j-s]) if j>=s] + dx * [-h[j+s] if j<=n-1-s]
            for j in range(n):
                if j >= s:
                    grad[j] += dx * (1 - h[j - s])
                if j <= n - 1 - s:
                    grad[j] += -dx * h[j + s]
        # For negative s, by symmetry f_{-s} = f_s, so we don't need to handle separately
    return grad / len(active_set)

def compute_subgradient_fast(h, active_set):
    """Vectorized subgradient computation"""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for s in active_set:
        if s == 0:
            grad[:n] += dx * (1 - h)  # (1-h[j-0]) for all j
            grad[:n] += -dx * h       # -h[j+0] for all j
            # Actually for s=0: ∂f_0/∂h[j] = dx*(1-h[j]) - dx*h[j] = dx*(1-2h[j])
        else:

            grad[s:] += dx * (1 - h[:n-s])
            # j <= n-1-s: contribution -h[j+s]
            grad[:n-s] += -dx * h[s:]
    return grad / len(active_set)

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)

        if np.sum(proj) > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)

        idx = int(np.argmin(np.abs(result - 0.5)))
        result[idx] = np.clip(result[idx] + diff, 0.0, 1.0)
    return result

    current = h.copy()
    velocity = np.zeros(n)
    lr = initial_lr
    best = current.copy()
    best_c5 = evaluate_c5(best)
    
    for it in range(max_iters):
        # Compute all f_s values
        f_vals = compute_all_f(current)
        c5 = np.max(f_vals)
        
        if c5 < best_c5:
            best_c5 = c5
            best = current.copy()

        shifts = np.arange(-(n-1), n)
        # Only consider non-negative shifts (by symmetry)
        active_mask = f_vals >= c5 - active_eps
        active_shifts = shifts[active_mask]
        # Remove negative shifts (redundant by symmetry)
        active_shifts = np.abs(active_shifts)
        active_shifts = np.unique(active_shifts)
        active_shifts = [int(s) for s in active_shifts if s < n]
        
        if len(active_shifts) == 0:
            active_shifts = [int(np.argmax(f_vals) - (n-1))]
            if active_shifts[0] < 0:
                active_shifts = [-active_shifts[0]]
        
        # Compute subgradient
        sg = compute_subgradient_fast(current, active_shifts)
        
        # Momentum update
        velocity = momentum * velocity - lr * sg
        candidate = current + velocity
        candidate = project_feasible(candidate, n)
        
        # Check if we improved
        new_c5 = evaluate_c5(candidate)
        if new_c5 < c5:
            current = candidate
        else:
            # Rejected - reduce learning rate
            lr *= 0.9
            velocity *= 0.5
        
        # Decay learning rate
        lr *= decay
    
    return project_feasible(best, n), best_c5

        if np.sum(proj) > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-14:
        idx = int(np.argmin(np.abs(result - 0.5)))
        result[idx] = np.clip(result[idx] + diff, 0.0, 1.0)
    return result

def compute_subgradient_fast(h, active_shifts, n):
    grad = np.zeros(n)
    dx = 2.0 / n
    for s in active_shifts:
        if s == 0:
            grad += dx * (1.0 - 2.0 * h)
        else:
            grad[s:] += dx * (1.0 - h[:n - s])
            grad[:n - s] += -dx * h[s:]
    return grad

def multiplicative_weights_opt(h, n, max_iters, tol=1e-10):
    current = h.copy()
    best = current.copy()
    best_c5 = evaluate_c5(best)
    lbfgs_count = 0

    for it in range(max_iters):
        f_vals = compute_all_f(current)
        c5 = float(np.max(f_vals))
        if c5 < best_c5:
            best_c5 = c5
            best = current.copy()
        shifts = np.arange(-(n - 1), n)
        active_mask = f_vals >= c5 - max(0.002, 0.005 * c5)
        active_shifts_raw = shifts[active_mask]
        active_shifts = sorted(set(int(abs(s)) for s in active_shifts_raw if abs(s) < n))
        if not active_shifts:
            active_shifts = [0]
        k = len(active_shifts)
        f_active = np.array([f_vals[s + (n - 1)] for s in active_shifts])
        weights = np.exp(f_active / 0.003)
        weights /= np.sum(weights)
        try:
            def obj(x):
                fv = compute_all_f(x)
                return float(np.sum(weights * fv[[s + (n - 1) for s in active_shifts]]))
            def grad(x):
                return compute_subgradient_fast(x, active_shifts, n) * 1.0
            bounds = [(0.0, 1.0)] * n
            cons = [{'type': 'eq', 'fun': constraint_func, 'args': (n,)}]
            res = minimize(obj, current, jac=grad, method='L-BFGS-B',
                           bounds=bounds, options={'maxiter': 30, 'ftol': 1e-14})
            current = project_feasible(res.x, n)
            lbfgs_count += 1
        except Exception:
            sg = compute_subgradient_fast(current, active_shifts, n)
            lr = 0.001
            for _ in range(30):
                candidate = project_feasible(current - lr * sg, n)
                new_c5 = evaluate_c5(candidate)
                if new_c5 < c5:
                    current = candidate
                    break
                lr *= 0.5
        if it % 200 == 0 and it > 0:
            c5_now = evaluate_c5(current)
            if c5_now > best_c5 + 0.01:
                current = best + np.random.randn(n) * 0.02
                current = project_feasible(current, n)
    return project_feasible(best, n), best_c5

def coordinate_descent(h, n, num_sweeps=3, step_sizes=None):
    if step_sizes is None:
        step_sizes = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    best = h.copy()
    best_c5 = evaluate_c5(best)
    for sweep in range(num_sweeps):
        improved = False
        for i in range(n):
            for step in step_sizes:
                for sign in [-1.0, 1.0]:
                    h_new = best.copy()
                    h_new[i] = np.clip(h_new[i] + sign * step, 0.0, 1.0)
                    h_new = project_feasible(h_new, n)
                    c5_new = evaluate_c5(h_new)
                    if c5_new < best_c5 - 1e-14:
                        best = h_new
                        best_c5 = c5_new
                        improved = True
        if not improved:
            break
    return best, best_c5

def fine_coordinate_descent(h, n, num_sweeps=5):
    best = h.copy()
    best_c5 = evaluate_c5(best)
    steps = [0.001, 0.0005, 0.0002, 0.0001]
    for sweep in range(num_sweeps):
        improved = False
        for i in range(n):
            for step in steps:
                for sign in [-1.0, 1.0]:
                    h_new = best.copy()
                    h_new[i] = np.clip(h_new[i] + sign * step, 0.0, 1.0)
                    h_new = project_feasible(h_new, n)
                    c5_new = evaluate_c5(h_new)
                    if c5_new < best_c5 - 1e-15:
                        best = h_new
                        best_c5 = c5_new
                        improved = True
        if not improved:
            break
    return best, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    parent_h = None
    try:
        parent_h = np.array(initial_h_values, dtype=float)
    except Exception:
        pass

    def elapsed():
        return time.time() - start_time

    phase1_end = start_time + budget_s * 0.28
    phase2_end = start_time + budget_s * 0.72

    n_list = []
    if parent_h is not None:
        n_list.append(len(parent_h))
    for nn in [51, 55, 59, 63, 67, 71, 75, 79, 83, 87, 91]:
        if nn not in n_list:
            n_list.append(nn)

    for n in n_list:
        if elapsed() > phase1_end:
            break
        x = np.linspace(0, 2, n, endpoint=False)
        if parent_h is not None:
            if len(parent_h) == n:
                h0 = parent_h.copy()
            else:
                xp = np.linspace(0, 2, len(parent_h), endpoint=False)
                f_interp = interp1d(xp, parent_h, kind='cubic', fill_value='extrapolate', bounds_error=False)
                h0 = np.clip(f_interp(x), 0, 1)
            h0 = project_feasible(h0, n)
            bounds = [(0.0, 1.0)] * n
            cons = [{'type': 'eq', 'fun': constraint_func, 'args': (n,)}]
            current = h0.copy()
            for tau in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]:
                if elapsed() > phase1_end:
                    break
                try:
                    res = minimize(objective_func, current, method='SLSQP',
                                   args=(n, tau), bounds=bounds, constraints=cons,
                                   options={'maxiter': 40, 'ftol': 1e-12})
                    current = project_feasible(res.x, n)
                except Exception:
                    pass
            c5 = evaluate_c5(current)
            if c5 < best_c5:
                best_c5 = c5
                best_h = current.copy()
        num_starts = 4 if n > 60 else 6
        for _ in range(num_starts):
            if elapsed() > phase1_end:
                break
            amp = rng.uniform(0.2, 0.48)
            freq1 = rng.uniform(0.5, 4.0)
            freq2 = rng.uniform(2.0, 6.0)
            ph1 = rng.uniform(0, 2 * np.pi)
            ph2 = rng.uniform(0, 2 * np.pi)
            h0 = 0.5 + amp * np.sin(2 * np.pi * freq1 * x / 2.0 + ph1)
            h0 += 0.15 * np.sin(2 * np.pi * freq2 * x / 2.0 + ph2)
            h0 = project_feasible(np.clip(h0, 0, 1), n)
            bounds = [(0.0, 1.0)] * n
            cons = [{'type': 'eq', 'fun': constraint_func, 'args': (n,)}]
            current = h0.copy()
            for tau in [0.03, 0.01, 0.003]:
                if elapsed() > phase1_end:
                    break
                try:
                    res = minimize(objective_func, current, method='SLSQP',
                                   args=(n, tau), bounds=bounds, constraints=cons,
                                   options={'maxiter': 30, 'ftol': 1e-12})
                    current = project_feasible(res.x, n)
                except Exception:
                    pass
            c5 = evaluate_c5(current)
            if c5 < best_c5:
                best_c5 = c5
                best_h = current.copy()

    if best_h is None:
        n = 77
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c5 = evaluate_c5(best_h)

    n_best = len(best_h)
    if elapsed() < phase2_end:
        current = best_h.copy()
        for attempt in range(30):
            if elapsed() > phase2_end * 0.95:
                break
            h_opt, c5 = multiplicative_weights_opt(current, n_best, max_iters=200)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()
                current = h_opt.copy()
            else:
                noise = 0.003 + 0.01 * (1.0 - attempt / 30.0)
                current = project_feasible(best_h + rng.randn(n_best) * noise, n_best)

    if elapsed() < start_time + budget_s * 0.88:
        h_cd, c5_cd = coordinate_descent(best_h, n_best, num_sweeps=5)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    if elapsed() < start_time + budget_s * 0.95:
        bounds = [(0.0, 1.0)] * n_best
        cons = [{'type': 'eq', 'fun': constraint_func, 'args': (n_best,)}]
        current = best_h.copy()
        for tau in [0.0005, 0.0002, 0.0001]:
            if elapsed() > start_time + budget_s * 0.95:
                break
            try:
                res = minimize(objective_func, current, method='SLSQP',
                               args=(n_best, tau), bounds=bounds, constraints=cons,
                               options={'maxiter': 100, 'ftol': 1e-14})
                current = project_feasible(res.x, n_best)
                c5 = evaluate_c5(current)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = current.copy()
            except Exception:
                pass

    if elapsed() < start_time + budget_s * 0.99:
        h_fine, c5_fine = fine_coordinate_descent(best_h, n_best, num_sweeps=4)
        if c5_fine < best_c5:
            best_c5 = c5_fine
            best_h = h_fine.copy()

    final_c5 = evaluate_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
