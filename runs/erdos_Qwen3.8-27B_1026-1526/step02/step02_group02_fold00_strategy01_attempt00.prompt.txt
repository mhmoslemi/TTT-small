$$C_5 = \max_k \int h(x)(1 - h(x+k)) dx$$

def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            room = 1.0 - h[mask]
            if room.min() <= 0:
                break
            step = min(diff / np.sum(mask), room.min() * 0.999)
            h[mask] += step
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            room = h[mask]
            if room.min() <= 0:
                break
            step = min(-diff / np.sum(mask), room.min() * 0.999)
            h[mask] -= step
    return np.clip(h, 0.0, 1.0)


def c5_logsumexp(h, T):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    c_max = c.max()
    if c_max < -1000:
        return float(c_max)
    shifted = T * (c - c_max)
    shifted = np.clip(shifted, -400, 400)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / T)


def smooth_obj(x, T, penalty):
    n = len(x)
    c5_val = c5_logsumexp(x, T)
    sum_violation = np.sum(x) - n / 2.0
    return c5_val + penalty * sum_violation ** 2


def slsqp_obj(x):
    return x[-1]


def slsqp_ineq(x):

def slsqp_eq(x):

def slsqp_eq_jac(x):

    return jac


def make_slsqp_constraints():
    return [

        {'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac}
    ]


def slsqp_optimize(h_init, n, maxiter=3000, t_margin=0.1):

    x0 = np.concatenate([h_init, [c5_init + t_margin]])
    bounds = [(0.0, 1.0)] * n + [(max(0.001, c5_init * 0.5), 2.0)]
    constraints = make_slsqp_constraints()
    result = minimize(slsqp_obj, x0, method='SLSQP', bounds=bounds,
                      constraints=constraints,

        return h_out, compute_c5_from_h(h_out)
    return h_init, c5_init


def lbfgsb_optimize(h_init, n, T, penalty, maxiter=3000):
