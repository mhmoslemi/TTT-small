        else:
            order = np.arange(n)

            old_val = h[i]
            j1_arr = i + np.arange(2*n-1) - n + 1  # for each k
            j2_arr = i - np.arange(2*n-1) + n - 1   # for each k

            valid_j2 = (j2_arr >= 0) & (j2_arr < n)

            A = np.zeros(2*n-1)

            mask_j2 = valid_j2 & (j2_arr != i)
            A[mask_j2] -= h[j2_clipped[mask_j2]]

            Q[j2_arr == i] = -1.0

            corr_cur = np.correlate(h, 1.0 - h, mode="full")

            # Evaluate on a grid

            f_vals = np.max(vals, axis=1)
            best_idx = np.argmin(f_vals)

            if best_idx == 0:
                t_new = 0.0
            elif best_idx == len(grid) - 1:
                t_new = 1.0
            else:

                t_hi = grid[min(len(grid)-1, best_idx+1)]

            if abs(t_new - old_val) > 1e-10:
                improved = True
        
        h = project_to_sum(h, ts)
        if not improved:
            break
    return h

def _golden_section_1d(f, a, b, tol=1e-8, maxiter=100):
    golden = (np.sqrt(5) - 1) / 2
    c = b - golden * (b - a)
    d = a + golden * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(maxiter):

            d = c
            fd = fc
            c = b - golden * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + golden * (b - a)
            fd = f(d)
    return (a + b) / 2

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h


def gs_min_1d(CONST, A, Q, lo, hi):
    def f(t):
        return np.max(CONST + A * t + Q * t * t)
    g = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - g * (hi - lo)
    d = lo + g * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(40):
        if hi - lo < 1e-12:
            break
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - g * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + g * (hi - lo)
            fd = f(d)
    return 0.5 * (lo + hi)


def cd_sweep(h, n, order):
    target = n / 2.0
    h = h.copy()
    for i in order:
        old = h[i]
        corr = np.correlate(h, 1.0 - h, mode="full")
        ks = np.arange(2 * n - 1)
        j1 = i + ks - n + 1
        j2 = i - ks + n - 1
        vj1 = (j1 >= 0) & (j1 < n)
        vj2 = (j2 >= 0) & (j2 < n)
        A = np.zeros(2 * n - 1)
        if np.any(vj1):
            A[vj1] += 1.0 - h[np.clip(j1, 0, n - 1)[vj1]]
        m2 = vj2 & (j2 != i)
        if np.any(m2):
            A[m2] -= h[np.clip(j2, 0, n - 1)[m2]]
        Q = np.zeros(2 * n - 1)
        Q[j2 == i] = -1.0
        CONST = corr - A * old - Q * old * old
        grid = np.linspace(0.0, 1.0, 33)
        vals = CONST[None, :] + A[None, :] * grid[:, None] + Q[None, :] * grid[:, None] ** 2
        fv = np.max(vals, axis=1)
        bi = int(np.argmin(fv))
        if bi == 0:
            h[i] = 0.0
        elif bi == 32:
            h[i] = 1.0
        else:
            h[i] = gs_min_1d(CONST, A, Q, grid[bi - 1], grid[bi + 1])
    return project_to_sum(h, target)


def cd_optimize(h0, n, max_sweeps, rng):
    h = h0.copy()
    best_c5 = compute_c5(h)
    stagnant = 0
    for s in range(max_sweeps):
        order = rng.permutation(n)
        h_new = cd_sweep(h, n, order)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            h = h_new
            stagnant = 0
        else:
            h = h_new
            stagnant += 1
            if stagnant > 20:
                break
    return h, best_c5
