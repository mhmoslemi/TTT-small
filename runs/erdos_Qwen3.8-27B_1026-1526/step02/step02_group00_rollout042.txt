   import cvxpy as cp
   import numpy as np
   import time
   from scipy.optimize import minimize
   from scipy.optimize import linprog

   def compute_c5(h):
       n = len(h)
       dx = 2.0 / n
       corr = np.correlate(h, 1.0 - h, mode="full")
       return float(np.max(corr) * dx)

   def project_to_sum(h, target_sum):
       h = np.clip(h.astype(float), 0.0, 1.0)
       for _ in range(500):
           diff = target_sum - np.sum(h)
           if abs(diff) < 1e-14:
               break
           mask = (h > 1e-12) & (h < 1.0 - 1e-12)
           if np.any(mask):
               h[mask] += diff / np.sum(mask)
           else:
               h += diff / len(h)
           h = np.clip(h, 0.0, 1.0)
       return h

   def solve_lp_given_q(q, n):
       """Given q (= 1-h), solve for h' minimizing max correlation."""
       t = cp.Variable()
       h_var = cp.Variable(n)
       
       constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n/2]
       
       for m in range(2*n - 1):
           k = m - (n - 1)  # actual lag
           if k >= 0:

               # = sum_{i=k}^{n-1} h'[i] * q[i-k]
               coeffs = np.zeros(n)
               coeffs[k:] = q[:n-k]
           else:  # k < 0, let d = -k

               # = sum_{i=0}^{n-1-d} h'[i] * q[i+d]
               coeffs = np.zeros(n)
               coeffs[:n-d] = q[d:]
           
           constraints.append(coeffs @ h_var <= t)
       
       prob = cp.Problem(cp.Minimize(t), constraints)
       prob.solve(solver=cp.ECOS, verbose=False)

    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def slsqp_refine(h0, n, target_sum, alpha, maxiter=500):
    def f(h):
        return smax_grad(h, alpha)[0]
    def g(h):
        return smax_grad(h, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def make_periodic_start(n, target_sum, period, ones_per_period):
    h = np.zeros(n)
    for i in range(n):
        if i % period < ones_per_period:
            h[i] = 1.0
    return project_to_sum(h, target_sum)

def make_random_binary(n, target_sum, rng):
    h = (rng.rand(n) > 0.5).astype(float)
    attempts = 0
    while abs(h.sum() - target_sum) > 1e-10 and attempts < 10000:
        if h.sum() > target_sum:
            idx = rng.choice(np.where(h > 0)[0])
            h[idx] = 0.0
        else:
            idx = rng.choice(np.where(h < 1)[0])
            h[idx] = 1.0
        attempts += 1
    return h

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()

    def time_left():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    # Phase 1: Alternating LP at current resolution
    if time_left() > 120:
        h_alt, c5_alt = alternating_lp(best_h, best_n, max_iters=50, time_limit=min(100, time_left() - 20))
        if c5_alt < best_c5:
            best_c5 = c5_alt
            best_h = h_alt.copy()

    # Phase 2: Binary local search from best continuous solution
    if time_left() > 100:
        h_bin = round_to_binary(best_h, best_n)
        if abs(h_bin.sum() - best_n / 2.0) < 1e-10:
            h_bs, c5_bs = binary_local_search(h_bin, best_n, max_iters=6000,
                                              time_limit=min(150, time_left() - 50), seed=seed)
            if c5_bs < best_c5:
                best_c5 = c5_bs
                best_h = h_bs.copy()

    # Phase 3: SLSQP polish
    if time_left() > 60:
        for alpha in [50, 200, 800]:
            if time_left() < 20:
                break
            h_pol = slsqp_refine(best_h, best_n, best_n / 2.0, alpha, maxiter=400)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Phase 4: Try different starting points at current resolution
    if time_left() > 150:
        starts = []
        # Periodic patterns
        for period in [4, 5, 6, 7, 8, 10, 12, 16]:
            for op in range(1, period // 2 + 1):
                if op * (best_n // period + 1) <= best_n // 2:
                    starts.append(make_periodic_start(best_n, best_n / 2.0, period, op))
        # Random binary starts
        for _ in range(6):
            starts.append(make_random_binary(best_n, best_n / 2.0, rng))
        # Interpolated best
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        noise_h = best_h + 0.1 * rng.randn(best_n)
        starts.append(project_to_sum(noise_h, best_n / 2.0))

        for h_s in starts:
            if time_left() < 40:
                break
            h_s = project_to_sum(h_s, best_n / 2.0)
            c5_s = compute_c5(h_s)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_s.copy()
            # Quick alternating LP from this start
            if time_left() > 60:
                h_res, c5_res = alternating_lp(h_s, best_n, max_iters=15,
                                               time_limit=min(30, time_left() - 30), damp=0.5)
                if c5_res < best_c5:
                    best_c5 = c5_res
                    best_h = h_res.copy()
            # Binary search from this start
            if time_left() > 50:
                h_b = round_to_binary(h_s, best_n)
                if abs(h_b.sum() - best_n / 2.0) < 1e-10:
                    h_rb, c5_rb = binary_local_search(h_b, best_n, max_iters=2000,
                                                      time_limit=min(25, time_left() - 30), seed=seed + 1)
                    if c5_rb < best_c5:
                        best_c5 = c5_rb
                        best_h = h_rb.copy()

    # Phase 5: Try higher resolution (n=480)
    if time_left() > 120:
        n2 = 480
        ts2 = n2 / 2.0
        # Upsample best solution
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n2, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h, period=2.0)
        h_up = project_to_sum(h_up, ts2)
        h_up_bin = round_to_binary(h_up, n2)
        c5_up = compute_c5(h_up_bin)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up_bin.copy()
            best_n = n2
        if time_left() > 60:
            h_hi, c5_hi = alternating_lp(h_up, n2, max_iters=25,
                                         time_limit=min(80, time_left() - 30), damp=0.5)
            if c5_hi < best_c5:
                best_c5 = c5_hi
                best_h = h_hi.copy()
                best_n = n2
        if time_left() > 50:
            h_b2 = round_to_binary(best_h, best_n)
            if abs(h_b2.sum() - best_n / 2.0) < 1e-10:
                h_r2, c5_r2 = binary_local_search(h_b2, best_n, max_iters=3000,
                                                  time_limit=min(40, time_left() - 20), seed=seed + 2)
                if c5_r2 < best_c5:
                    best_c5 = c5_r2
                    best_h = h_r2.copy()

    # Phase 6: Final SLSQP polish at best resolution
    if time_left() > 30:
        for alpha in [1000, 5000]:
            if time_left() < 15:
                break
            h_final = slsqp_refine(best_h, best_n, best_n / 2.0, alpha, maxiter=300)
            c5_final = compute_c5(h_final)
            if c5_final < best_c5:
                best_c5 = c5_final
                best_h = h_final.copy()

    # Final validation
    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
