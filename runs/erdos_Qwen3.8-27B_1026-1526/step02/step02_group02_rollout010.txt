import time

def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def compute_c_all(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return c * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            room = 1.0 - h[mask]
            step = min(diff / np.sum(mask), room.min() * 0.999)
            h[mask] += step
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            room = h[mask]
            step = min(-diff / np.sum(mask), room.min() * 0.999)
            h[mask] -= step
    return np.clip(h, 0.0, 1.0)

def slsqp_obj(x):
    return x[-1]

def slsqp_ineq(x):
    h = x[:-1]
    t = x[-1]
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return t - c * dx

def slsqp_eq(x):
    h = x[:-1]
    return np.sum(h) - len(h) / 2.0

def slsqp_eq_jac(x):
    n = len(x) - 1
    jac = np.zeros(len(x))
    jac[:n] = 1.0
    return jac

def slsqp_optimize(h_init, n, maxiter=3000, t_margin=0.1):

    x0 = np.concatenate([h_init, [c5_init + t_margin]])
    bounds = [(0.0, 1.0)] * n + [(max(0.001, c5_init * 0.5), 2.0)]
    constraints = [
        {'type': 'ineq', 'fun': slsqp_ineq},
        {'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac}
    ]
    result = minimize(slsqp_obj, x0, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
    if result.x is not None:
        h_out = project_feasible(result.x[:-1], n)
        return h_out, compute_c5_from_h(h_out)
    return h_init, c5_init

def compute_gradient_max(h, n, top_m=10):
    """Compute subgradient of max_k c[k]*dx w.r.t. h, averaged over top-m active constraints."""
    c_all_raw = np.correlate(h, 1.0 - h, mode="full")
    c_all = c_all_raw * (2.0 / n)
    c_max = c_all.max()
    
    # Find top-m active
    sorted_idx = np.argsort(c_all)[::-1]
    # Take those close to max (within 1%)
    threshold = c_max - 0.02 * max(c_max, 1e-10)
    active = sorted_idx[c_all[sorted_idx] >= threshold][:top_m]
    
    g = np.zeros(n)
    for idx in active:
        k = idx - (n - 1)
        if k >= 0:
            g[k:] += (1.0 - h[:n-k])
            g[:n-k] -= h[k:]
        else:
            kk = -k
            g[:n-kk] += (1.0 - h[kk:])
            g[kk:] -= h[:n-kk]
    
    g /= len(active)
    return g, c_max, len(active)

def proj_subgrad_optimize(h, n, max_iter=20000, decay=0.999, initial_alpha=0.005,
                          momentum=0.7, top_m=10, time_limit=60):

    t0 = time.time()
    best_h = h.copy()
    best_c5 = compute_c5_from_h(h)
    
    v = np.zeros(n)

def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            room = 1.0 - h[mask]
            step = min(diff / np.sum(mask), room.min() * 0.999)
            h[mask] += step
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            room = h[mask]
            step = min(-diff / np.sum(mask), room.min() * 0.999)
            h[mask] -= step
    return np.clip(h, 0.0, 1.0)

def slsqp_obj(x):
    return x[-1]

def slsqp_ineq(x):
    h = x[:-1]
    t = x[-1]
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return t - c * dx

def slsqp_eq(x):
    h = x[:-1]
    return np.sum(h) - len(h) / 2.0

def slsqp_eq_jac(x):
    n = len(x) - 1
    jac = np.zeros(len(x))
    jac[:n] = 1.0
    return jac

def slsqp_optimize(h_init, n, maxiter=3000, t_margin=0.1):
    c5_init = compute_c5_from_h(h_init)
    x0 = np.concatenate([h_init, [c5_init + t_margin]])
    bounds = [(0.0, 1.0)] * n + [(max(0.001, c5_init * 0.5), 2.0)]
    constraints = [

        {'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac}

        return h_out, compute_c5_from_h(h_out)

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time


def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            room = 1.0 - h[mask]
            step = min(diff / np.sum(mask), room.min() * 0.999)
            h[mask] += step
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            room = h[mask]
            step = min(-diff / np.sum(mask), room.min() * 0.999)
            h[mask] -= step
    return np.clip(h, 0.0, 1.0)


def slsqp_obj(x):
    return x[-1]


def slsqp_ineq(x):
    h = x[:-1]
    t = x[-1]
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return t - c * dx


def slsqp_eq(x):
    h = x[:-1]
    return np.sum(h) - len(h) / 2.0


def slsqp_eq_jac(x):
    n = len(x) - 1
    jac = np.zeros(len(x))
    jac[:n] = 1.0
    return jac


def slsqp_optimize(h_init, n, maxiter=3000, t_margin=0.1):
    c5_init = compute_c5_from_h(h_init)
    x0 = np.concatenate([h_init, [c5_init + t_margin]])
    bounds = [(0.0, 1.0)] * n + [(max(0.001, c5_init * 0.5), 2.0)]
    constraints = [
        {'type': 'ineq', 'fun': slsqp_ineq},
        {'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac}
    ]
    result = minimize(slsqp_obj, x0, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
    if result.x is not None:
        h_out = project_feasible(result.x[:-1], n)
        return h_out, compute_c5_from_h(h_out)
    return h_init, c5_init


def coord_descent(h_init, n, max_sweeps=2000, lam=800.0, time_limit=100, ls_iters=35):
    h = h_init.copy()
    dx = 2.0 / n
    target = n / 2.0
    c_all = np.correlate(h, 1.0 - h, mode="full")
    S = np.sum(h)
    dm = n - 1
    n_full = 2 * n - 1
    t0 = time.time()

    for sweep in range(max_sweeps):
        if time.time() - t0 > time_limit:
            break
        max_change = 0.0
        for i in range(n):
            old = h[i]
            A = np.zeros(n_full)
            k_min1 = -i
            k_max1 = n - 1 - i
            m1s = max(0, dm + k_min1)
            m1e = min(n_full - 1, dm + k_max1)
            if m1s <= m1e:
                kr = np.arange(m1s, m1e + 1) - dm
                A[m1s:m1e + 1] += (1.0 - h[i + kr])
            k_min2 = i - (n - 1)
            k_max2 = i
            m2s = max(0, dm + k_min2)
            m2e = min(n_full - 1, dm + k_max2)
            if m2s <= m2e:
                kr2 = np.arange(m2s, m2e + 1) - dm
                A[m2s:m2e + 1] -= h[i - kr2]

            B = c_all - A * old
            S_other = S - old

            lo, hi = 0.0, 1.0
            for _ in range(ls_iters):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1 = dx * np.max(A * m1 + B) + lam * (S_other + m1 - target) ** 2
                f2 = dx * np.max(A * m2 + B) + lam * (S_other + m2 - target) ** 2
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
            x_new = (lo + hi) * 0.5

            diff = x_new - old
            ad = abs(diff)
            if ad > 1e-12:
                if ad > max_change:
                    max_change = ad
                c_all += A * diff
                S += diff
                h[i] = x_new

        if max_change < 1e-11:
            break

    h = project_feasible(h, n)
    return h, compute_c5_from_h(h)


def sharpen_push(h, n, frac=0.15):
    h = h.copy()
    mask_mid = (h > 0.15) & (h < 0.85)
    if np.any(mask_mid):
        h[mask_mid] += frac * np.where(h[mask_mid] > 0.5, 1, -1) * (0.3 - 0.15 * np.abs(h[mask_mid] - 0.5))
    return project_feasible(h, n)
