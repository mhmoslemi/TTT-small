def evaluate_c5(h):

    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    """Project h onto {0<=x<=1, sum(x)=n/2}"""
    target = n / 2.0
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-10:
        return h
    # Binary search for offset
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        if np.sum(proj) > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    # Fix remaining error
    diff = target - np.sum(result)
    if abs(diff) > 1e-12:
        # Distribute to elements closest to 0.5
        sorted_idx = np.argsort(np.abs(result - 0.5))
        remaining = diff
        for idx in sorted_idx:
            if remaining > 0:
                add = min(remaining, 1.0 - result[idx])
                result[idx] += add
                remaining -= add
            elif remaining < 0:
                sub = min(-remaining, result[idx])
                result[idx] -= sub
                remaining += sub
            if abs(remaining) < 1e-14:
                break
    return result

def smooth_max(values, tau):
    v = values / tau
    vmax = np.max(v)
    return tau * (vmax + np.log(np.sum(np.exp(v - vmax))))

def subgradient_c5(h, n):
    """Compute exact subgradient of C5 w.r.t. h"""
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    k_star = int(np.argmax(corr))
    s = k_star - (n - 1)  # actual shift
    
    grad = np.zeros(n)
    # First term: (1-h[j-s]) if j-s valid
    if s >= 0:
        for j in range(s, n):
            grad[j] += (1.0 - h[j - s])
    else:
        ss = -s
        for j in range(n - ss):
            grad[j] += (1.0 - h[j + ss])
    
    # Second term: -h[j+s] if j+s valid
    if s >= 0:
        for j in range(n - s):
            grad[j] += -h[j + s]
    else:
        ss = -s
        for j in range(ss, n):
            grad[j] += -h[j - ss]
    
    return dx * grad, s
