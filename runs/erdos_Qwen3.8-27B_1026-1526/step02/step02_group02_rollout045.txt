   def smooth_grad(h, T, penalty):
       n = len(h)
       dx = 2.0 / n
       c = np.correlate(h, 1.0 - h, mode="full") * dx
       c_max = c.max()
       shifted = T * (c - c_max)
       shifted = np.clip(shifted, -500, 500)
       exp_shifted = np.exp(shifted)
       weights = exp_shifted / exp_shifted.sum()
       
       g = np.zeros(n)
       # For each lag m (index in full corr = m + n - 1):
       # c[m]*dx = dx * sum_i h[i+m]*(1-h[i])
       # d(c[m]*dx)/dh[j] = dx * [(1-h[j-m])*1(m<=j<=n-1-m) - h[j+m]*1(0<=j<=n-1-m)]  for m>0
       #                   = dx * (1-2h[j])  for m=0
       #                   = dx * [(1-h[j+m])*1(0<=j<=n-1-m) - h[j-m]*1(m<=j<=n-1)]  for m<0 (where m=-m')
       
       # Vectorize over all m at once:
       # For m > 0 (index = n-1+m in full corr, m = 1,...,n-1):
       #   g[j] += dx * weights[n-1+m] * [(1-h[j-m]) - h[j+m]]  (with appropriate masking)
       
       # Let me just loop over m (it's only n-1 iterations, each O(n) with numpy)
       for idx in range(2 * n - 1):
           w = weights[idx]
           if w < 1e-18:
               continue
           m = idx - (n - 1)
           if m == 0:
               g += w * (1.0 - 2.0 * h) * dx
           elif m > 0:
               mask1 = np.zeros(n, dtype=bool)
               if n - 2 * m > 0:
                   mask1[m:n-m] = True
                   g[m:n-m] += w * (1.0 - h[:n-2*m]) * dx
               mask2 = np.zeros(n, dtype=bool)
               mask2[:n-m] = True
               g[:n-m] -= w * h[m:] * dx
           else:
               mp = -m
               g[:n-mp] += w * (1.0 - h[mp:]) * dx
               g[mp:] -= w * h[:n-mp] * dx
       
       sv = np.sum(h) - n / 2.0
       g += 2.0 * penalty * sv
       return g
