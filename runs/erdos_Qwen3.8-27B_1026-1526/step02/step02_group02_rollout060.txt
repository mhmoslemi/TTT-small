I'm confusing the discrete and continuous. In the continuous case, the overlap at shift k is the integral, which for the step function is |k| for |k|≤1 and 2-|k| for 1≤|k|≤2. The max is 1 at |k|=1.

So C5 = 1 for the single step. That's awful.

What about a "gradual" step? h(x) = max(0, 1-|x-1|) (triangle centered at x=1)?
This is continuous and smooth. The integral is 1 (area of triangle with base 2 and height 1... area = 1·1 = 1? No, base is 2 (from 0 to 2) and height is 1, area = 1·2/2 = 1. ✓)

The overlap at shift k would be the integral of h(x)·(1-h(x+k)) dx. This is more complex to compute analytically, but intuitively, the gradual transition means the overlap is spread out over multiple k values, reducing the peak.

This suggests that the optimal h is NOT a sharp step but has some "smearing" of the transition.

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_and_grad(x, T, penalty):
    n = len(x)
    dx = 2.0 / n
    r = np.correlate(x, 1.0 - x, mode='full')  # length 2n-1, raw (no dx)
    
    # Smooth objective: (1/T) log(sum(exp(T * r * dx)))
    # + penalty * (sum(x) - n/2)^2
    c = r * dx
    c_max = c.max()
    exp_c = np.exp(T * (c - c_max))
    exp_sum = exp_c.sum()
    p = exp_c / exp_sum
    
    f = c_max + np.log(exp_sum) / T
    sum_viol = np.sum(x) - n / 2.0
    f += penalty * sum_viol ** 2
    
    # Gradient of smooth part:
    one_minus_h = 1.0 - x
    A = np.convolve(one_minus_h, p, mode='full')[n-1 : 2*n-1]
    B = np.correlate(p, x, mode='full')[n-1 : 2*n-1][::-1]
    g_smooth = dx * (A - B)
    
    # Gradient of penalty:
    g_penny = 2 * penalty * sum_viol * np.ones(n)
    
    return f, g_smooth + g_penny

def lbfgsb_run(h_init, T, penalty, maxiter):
    n = len(h_init)
    bounds = [(0.0, 1.0)] * n
    result = minimize(
        smooth_obj_and_grad, h_init, args=(T, penalty),
        method='L-BFGS-B', bounds=bounds,
        jac=True,
        options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-15, 'factr': 1e10}
    )
    h_out = project_feasible(result.x, n)
    return h_out, compute_c5_from_h(h_out)

def slsqp_active(h_init, n_active=30, maxiter=3000, t_margin=0.1):
    n = len(h_init)
    dx = 2.0 / n
    
    c5_init = compute_c5_from_h(h_init)
    t0_val = c5_init + t_margin
    
    def obj(x):
        return x[-1]
    
    def active_ineq(x):
        h = x[:-1]
        t = x[-1]
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        idx = np.argsort(c)[-n_active:]
        return t - c[idx]
    
    def eq_con(x):
        return np.sum(x[:-1]) - len(x[:-1]) / 2.0
    
    x0 = np.concatenate([h_init, [t0_val]])
    bounds = [(0.0, 1.0)] * n + [(max(0.001, c5_init * 0.3), 2.0)]
    constraints = [
        {'type': 'ineq', 'fun': active_ineq},
        {'type': 'eq', 'fun': eq_con}
    ]
    
    result = minimize(obj, x0, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
    
    if result.x is not None:
        h_out = project_feasible(result.x[:-1], n)
        return h_out, compute_c5_from_h(h_out)
    return h_init, c5_init

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.default_rng(seed)
    
    # Get initial solution
    if 'initial_h_values' in globals():
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(270, 0.5)
    
    n_best = len(best_h)
    best_c5 = compute_c5_from_h(best_h)
    
    # Phase 1: L-BFGS-B with increasing T
    T_schedule = [10, 25, 50, 100, 200, 400]
    for T in T_schedule:
        if time.time() - t_start > budget_s - 300:
            break
        h_new, c5_new = lbfgsb_run(best_h, T, 100.0, maxiter=3000)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
    
    # Phase 2: Active-set SLSQP
    if time.time() - t_start < budget_s - 200:
        h_new, c5_new = slsqp_active(best_h, n_active=40, maxiter=5000, t_margin=0.1)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
    
    # Phase 3: Multi-start perturbations
    n_perturb = 30
    if time.time() - t_start < budget_s - 150:
        for attempt in range(n_perturb):
            if time.time() - t_start > budget_s - 60:
                break
            
            frac = attempt / max(1, n_perturb)
            scale = 0.08 * (1.0 - frac) ** 0.5 + 0.01
            
            r = rng.random()
            if r < 0.3:
                h_pert = best_h + scale * rng.standard_normal(n_best)
            elif r < 0.5:
                # Sine perturbation
                freq = rng.integers(2, max(3, n_best // 5))
                phase = rng.random() * 2 * np.pi
                h_pert = best_h + scale * np.sin(2 * np.pi * freq * np.arange(n_best) / n_best + phase)
            elif r < 0.7:
                # Block perturbation
                n_blocks = rng.integers(3, max(4, n_best // 15))
                h_pert = best_h.copy()
                for _ in range(n_blocks):
                    bsize = rng.integers(5, max(6, n_best // 10))
                    start = rng.integers(0, max(1, n_best - bsize))
                    h_pert[start:start+bsize] += scale * 2 * (rng.random(bsize) - 0.5)
            else:
                # Push to 0-1
                h_pert = best_h.copy()
                n_push = int(n_best * 0.2)
                push_idx = rng.choice(n_best, n_push, replace=False)
                for idx in push_idx:
                    if h_pert[idx] > 0.5:
                        h_pert[idx] = min(1.0, h_pert[idx] + 0.3)
                    else:
                        h_pert[idx] = max(0.0, h_pert[idx] - 0.3)
            
            h_pert = project_feasible(h_pert, n_best)
            
            # Optimize the perturbed solution
            if attempt % 2 == 0:
                h_new, c5_new = lbfgsb_run(h_pert, 50 + 200 * frac, 100.0, maxiter=1500)
            else:
                h_new, c5_new = slsqp_active(h_pert, n_active=30, maxiter=2000, t_margin=0.05)
            
            if c5_new < best_c5 - 1e-10:
                best_h = h_new
                best_c5 = c5_new
    
    # Phase 4: Resolution changes
    for n_try in [240, 300, 360]:
        if time.time() - t_start > budget_s - 120:
            break
        x_old = np.linspace(0, 2, n_best, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        f_interp = interp1d(x_old, best_h, kind='linear', fill_value="extrapolate")
        h_interp = np.clip(f_interp(x_new), 0, 1)
        h_interp = project_feasible(h_interp, n_try)
        
        h_new, c5_new = lbfgsb_run(h_interp, 50, 100.0, maxiter=2000)
        if time.time() - t_start < budget_s - 60:
            h_new2, c5_new2 = slsqp_active(h_new, n_active=30, maxiter=2000, t_margin=0.05)
            if c5_new2 < c5_new:
                h_new, c5_new = h_new2, c5_new2
        
        if c5_new < best_c5 - 1e-10:
            best_h = h_new
            best_c5 = c5_new
            n_best = n_try
    
    # Phase 5: Fine polishing
    if time.time() - t_start < budget_s - 30:
        h_new, c5_new = slsqp_active(best_h, n_active=50, maxiter=8000, t_margin=0.02)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
    
    # Final micro-adjustments
    n_fine = 20
    for i in range(n_fine):
        if time.time() - t_start > budget_s - 10:
            break
        scale = 0.002 + 0.008 * (1 - i / n_fine)
        h_pert = best_h + scale * rng.standard_normal(n_best)
        h_pert = project_feasible(h_pert, n_best)
        h_new, c5_new = slsqp_active(h_pert, n_active=20, maxiter=1500, t_margin=0.01)
        if c5_new < best_c5 - 1e-12:
            best_h = h_new
            best_c5 = c5_new
    
    # Final projection and evaluation
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5_from_h(best_h)
    
    return best_h, best_c5, len(best_h)
