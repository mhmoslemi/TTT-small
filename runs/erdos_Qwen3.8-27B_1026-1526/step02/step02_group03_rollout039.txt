from scipy.optimize import minimize, minimize_scalar
from scipy.interpolate import interp1d

def evaluate_c5(h):

    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        if np.sum(proj) > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-14:
        idx = int(np.argmin(np.abs(result - 0.5)))
        result[idx] = np.clip(result[idx] + diff, 0.0, 1.0)
    return result

def smooth_max(values, tau):
    v = values / tau
    vmax = np.max(v)
    return tau * (vmax + np.log(np.sum(np.exp(v - vmax))))

def objective_func(x, n, tau):
    dx = 2.0 / n
    g = 1.0 - x
    corr = np.correlate(x, g, mode="full")
    return smooth_max(corr, tau) * dx

def constraint_func(x, n):
    return np.sum(x) - n / 2.0

def coord_line_search(i, h, n):
    """Find optimal h[i] using bounded scalar minimization."""
    def f(v):
        h_new = h.copy()
        h_new[i] = v
        h_new = project_feasible(h_new, n)
        return evaluate_c5(h_new)

    order = rng.permutation(n)

def pairwise_exchange(i, j, h, n):
    """Optimize h[i] and h[j] jointly, preserving sum exactly."""

    lo = max(0.0, s - 1.0)

    h_new = h.copy()
    h_new[i] = v_opt
    h_new[j] = s - v_opt

    return smooth_max(corr, tau) * dx

def slsqp_constraint(x, n):
    return np.sum(x) - n / 2.0

def coord_line_search(i, h, n):
    current_c5 = evaluate_c5(h)
    def f(v):
        h_new = h.copy()
        h_new[i] = v
        h_new = project_feasible(h_new, n)
        return evaluate_c5(h_new)
    try:
        res = minimize_scalar(f, bounds=(0.0, 1.0), method='bounded',
                            options={'xatol': 1e-8, 'maxiter': 80})
        v_opt = res.x
    except Exception:
        return h
    h_new = h.copy()
    h_new[i] = v_opt
    h_new = project_feasible(h_new, n)
    c5_new = evaluate_c5(h_new)
    if c5_new <= current_c5 + 1e-14:
        return h_new
    return h

def full_coord_sweep(h, n, rng):
    order = list(rng.permutation(n))
    for i in order:
        h = coord_line_search(i, h, n)
    return h

def pairwise_exchange(i, j, h):
    s = h[i] + h[j]
    lo = max(0.0, s - 1.0)
    hi = min(1.0, s)
    if hi - lo < 1e-12:
        return h
    current_c5 = evaluate_c5(h)
    def f(v):
        h_new = h.copy()
        h_new[i] = v
        h_new[j] = s - v
        return evaluate_c5(h_new)
    try:
        res = minimize_scalar(f, bounds=(lo, hi), method='bounded',
                            options={'xatol': 1e-9, 'maxiter': 80})
        v_opt = res.x
    except Exception:
        return h
    h_new = h.copy()
    h_new[i] = v_opt
    h_new[j] = s - v_opt
    c5_new = evaluate_c5(h_new)
    if c5_new <= current_c5 + 1e-14:
        return h_new
    return h

def triple_exchange(i, j, k, h):
    s = h[i] + h[j] + h[k]
    def f(params):
        v1, v2 = params
        v3 = s - v1 - v2
        if v3 < -1e-10 or v1 < -1e-10 or v2 < -1e-10:
            return 1.0
        if v1 > 1 + 1e-10 or v2 > 1 + 1e-10 or v3 > 1 + 1e-10:
            return 1.0
        h_new = h.copy()
        h_new[i] = v1
        h_new[j] = v2
        h_new[k] = v3
        return evaluate_c5(h_new)
    x0 = [h[i], h[j]]
    bounds = [(max(0, s-2), min(1, s)), (max(0, s-2), min(1, s))]
    try:
        res = minimize(f, x0, method='Nelder-Mead',
                      options={'xatol': 1e-8, 'fatol': 1e-10, 'maxiter': 200})
        v1, v2 = res.x
        v3 = s - v1 - v2
        if 0 - 1e-10 <= v1 <= 1 + 1e-10 and 0 - 1e-10 <= v2 <= 1 + 1e-10 and 0 - 1e-10 <= v3 <= 1 + 1e-10:
            h_new = h.copy()
            h_new[i] = v1
            h_new[j] = v2
            h_new[k] = v3
            c5_new = evaluate_c5(h_new)
            if c5_new <= evaluate_c5(h) + 1e-14:
                return h_new
    except Exception:
        pass
    return h

def interpolate_to_n(parent_h, new_n):
    np_old = len(parent_h)
    if np_old == new_n:
        return parent_h.copy()
    x_old = np.linspace(0, 2, np_old, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    try:
        f = interp1d(x_old, parent_h, kind='cubic', fill_value='extrapolate', bounds_error=False)
        h_new = f(x_new)
    except Exception:
        f = interp1d(x_old, parent_h, kind='linear', fill_value='extrapolate', bounds_error=False)
        h_new = f(x_new)
    return project_feasible(np.clip(h_new, 0, 1), new_n)

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    start_time = time.time()

    try:
        parent_h = np.array(initial_h_values, dtype=float)
    except Exception:
        parent_h = None

    # Phase 0: Multi-n exploration (8%)
    phase0_end = start_time + budget_s * 0.08
    candidates = []
    if parent_h is not None:
        np_parent = len(parent_h)
        n_trials = list(range(max(3, np_parent - 6), np_parent + 7))
    else:
        np_parent = 77
        n_trials = [73, 74, 75, 76, 77, 78, 79, 80, 81, 82, 83]

    for nt in n_trials:
        if time.time() > phase0_end:
            break
        if parent_h is not None:
            h_try = interpolate_to_n(parent_h, nt)
        else:
            h_try = project_feasible(np.full(nt, 0.5), nt)
        # Quick SLSQP
        bounds = [(0.0, 1.0)] * nt
        cons = [{'type': 'eq', 'fun': slsqp_constraint, 'args': (nt,)}]
        try:
            res = minimize(slsqp_objective, h_try, method='SLSQP',
                        args=(nt, 0.05), bounds=bounds, constraints=cons,
                        options={'maxiter': 80, 'ftol': 1e-10})
            h_try = project_feasible(res.x, nt)
        except Exception:
            pass
        c5_try = evaluate_c5(h_try)
        candidates.append((c5_try, h_try, nt))

    candidates.sort(key=lambda x: x[0])
    best_c5, best_h, n = candidates[0][0], candidates[0][1], candidates[0][2]

    # Phase 1: SLSQP refinement on best (10%)
    phase1_end = start_time + budget_s * 0.18
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_constraint, 'args': (n,)}]
    for tau in [0.02, 0.005, 0.001]:
        if time.time() > phase1_end:
            break
        try:
            res = minimize(slsqp_objective, best_h, method='SLSQP',
                        args=(n, tau), bounds=bounds, constraints=cons,
                        options={'maxiter': 200, 'ftol': 1e-13})
            candidate = project_feasible(res.x, n)
            c5 = evaluate_c5(candidate)
            if c5 < best_c5:
                best_c5 = c5
                best_h = candidate
        except Exception:
            pass

    # Phase 2: Coordinate descent + pairwise (42%)
    phase2_end = start_time + budget_s * 0.60
    stall = 0
    while time.time() < phase2_end:
        prev_c5 = best_c5
        best_h = full_coord_sweep(best_h, n, rng)
        c5 = evaluate_c5(best_h)
        if c5 < best_c5:
            best_c5 = c5

        for _ in range(400):
            if time.time() > phase2_end:
                break
            i, j = rng.choice(n, 2, replace=False)
            h_new = pairwise_exchange(int(i), int(j), best_h)
            if h_new is not best_h:
                c5_new = evaluate_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new
                else:
                    best_h = h_new

        if rng.random() < 0.3:
            for _ in range(30):
                if time.time() > phase2_end:
                    break
                i, j, k = rng.choice(n, 3, replace=False)
                h_new = triple_exchange(int(i), int(j), int(k), best_h)
                if h_new is not best_h:
                    c5_new = evaluate_c5(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new
                    else:
                        best_h = h_new

        if best_c5 < prev_c5 - 1e-10:
            stall = 0
        else:
            stall += 1
        if stall > 100:
            break

    # Phase 3: Simulated annealing (28%)
    phase3_end = start_time + budget_s * 0.88
    if time.time() < phase3_end:
        T = 0.0015
        T_min = 0.00003
        cooling = 0.9995
        stuck = 0
        max_sa = 20000

        for it in range(max_sa):
            if time.time() > phase3_end:
                break
            T = max(T_min, T * cooling)

            move_type = rng.random()
            if move_type < 0.30:
                bs = int(rng.choice([3, 4, 5, 7, 10]))
                start = rng.randint(0, n - bs + 1)
                h_new = best_h.copy()
                h_new[start:start+bs] += rng.randn(bs) * 0.03
                h_new = project_feasible(h_new, n)
            elif move_type < 0.55:
                i, j = rng.choice(n, 2, replace=False)
                h_new = pairwise_exchange(int(i), int(j), best_h)
                if h_new is best_h:
                    continue
            elif move_type < 0.75:
                n_flip = int(rng.choice([1, 2, 3, 4]))
                idx = rng.choice(n, n_flip, replace=False)
                h_new = best_h.copy()
                h_new[idx] = 1.0 - h_new[idx]
                h_new = project_feasible(h_new, n)
            elif move_type < 0.88:
                i, j, k = rng.choice(n, 3, replace=False)
                h_new = triple_exchange(int(i), int(j), int(k), best_h)
                if h_new is best_h:
                    continue
            else:
                h_new = project_feasible(best_h + rng.randn(n) * 0.01, n)

            c5_new = evaluate_c5(h_new)
            delta = c5_new - best_c5

            if delta < 0 or (delta > -50*T and rng.random() < np.exp(-delta / T)):
                best_h = h_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    stuck = 0
                    if time.time() < phase3_end and rng.random() < 0.4:
                        best_h = full_coord_sweep(best_h, n, rng)
                        c5 = evaluate_c5(best_h)
                        if c5 < best_c5:
                            best_c5 = c5
                else:
                    stuck += 1
            else:
                stuck += 1

            if stuck > 3000:
                stuck = 0
                T = min(T * 3, 0.003)

    # Phase 4: Final polishing (12%)
    phase4_end = start_time + budget_s * 0.995
    final_stall = 0
    while time.time() < phase4_end and final_stall < 20:
        prev_c5 = best_c5
        best_h = full_coord_sweep(best_h, n, rng)
        c5 = evaluate_c5(best_h)
        if c5 < best_c5:
            best_c5 = c5

        for _ in range(2000):
            if time.time() > phase4_end:
                break
            i, j = rng.choice(n, 2, replace=False)
            h_new = pairwise_exchange(int(i), int(j), best_h)
            if h_new is not best_h:
                c5_new = evaluate_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new
                else:
                    best_h = h_new

        if rng.random() < 0.5:
            for _ in range(100):
                if time.time() > phase4_end:
                    break
                i, j, k = rng.choice(n, 3, replace=False)
                h_new = triple_exchange(int(i), int(j), int(k), best_h)
                if h_new is not best_h:
                    c5_new = evaluate_c5(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new
                    else:
                        best_h = h_new

        if best_c5 < prev_c5 - 1e-12:
            final_stall = 0
        else:
            final_stall += 1

    # Final SLSQP
    if time.time() < start_time + budget_s * 0.998:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': slsqp_constraint, 'args': (n,)}]
        try:
            res = minimize(slsqp_objective, best_h, method='SLSQP',
                        args=(n, 0.0003), bounds=bounds, constraints=cons,
                        options={'maxiter': 300, 'ftol': 1e-15})
            candidate = project_feasible(res.x, n)
            c5 = evaluate_c5(candidate)
            if c5 < best_c5:
                best_c5 = c5
                best_h = candidate
        except Exception:
            pass

    best_h = full_coord_sweep(best_h, n, rng)
    final_c5 = evaluate_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = best_h.copy()

    return best_h, best_c5, n
