            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha):
    """Compute smooth max of corr(h) and its gradient w.r.t. h."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -500.0, 500.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / len(corr)) / alpha
    
    g = np.zeros(n)
    for k in range(n):
        # Positive lag k: index in corr = n-1+k
        wk = w[n - 1 + k]
        if wk < 1e-15:
            continue
        # dc_k/dh[i] for i in [k, n-1]: +(1-h[i-k])
        if k < n:
            idx = np.arange(k, n)
            g[idx] += wk * (1.0 - h[idx - k])
        # dc_k/dh[i] for i in [0, n-1-k]: -h[i+k]
        if n - k > 0:
            idx2 = np.arange(0, n - k)
            g[idx2] -= wk * h[idx2 + k]
        
        # Negative lag -k: index in corr = n-1-k
        wk_neg = w[n - 1 - k]
        if wk_neg < 1e-15:
            continue
        # dc_{-k}/dh[i] for i in [0, n-1-k]: +(1-h[i+k])
        idx3 = np.arange(0, n - k)
        g[idx3] += wk_neg * (1.0 - h[idx3 + k])
        # dc_{-k}/dh[i] for i in [k, n-1]: -h[i-k]
        idx4 = np.arange(k, n)
        g[idx4] -= wk_neg * h[idx4 - k]
    
    return sm_val, g

def nesterov_pg(h0, n, target_sum, alpha, n_iters, lr_init, t_start, time_budget):
    h = h0.copy()
    y = h.copy()
    t = 1.0
    lr = lr_init
    c5_cur = compute_c5(h)
    h_best = h.copy()
    c5_best = c5_cur
    
    for it in range(n_iters):
        if time.time() - t_start > time_budget:
            break
        _, gy = smooth_max_and_grad(y, alpha)
        h_new = y - lr * gy
        h_new = project_to_sum(h_new, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        mo = (t - 1.0) / t_new
        y = h_new + mo * (h_new - h)
        y = np.clip(y, 0.0, 1.0)
        h = h_new
        t = t_new
        
        if it % 100 == 0:
            c5_new = compute_c5(h)
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h.copy()
                lr = min(lr * 1.1, 1.0)
            else:
                lr *= 0.7
    
    c5_final = compute_c5(h)
    if c5_final < c5_best:
        c5_best = c5_final
        h_best = h.copy()
    return h_best, c5_best

def slsqp_optimize(h0, n, target_sum, alpha, maxiter, t_start, time_budget):
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_and_grad(h, alpha)
        return v
    
    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        _, g = smooth_max_and_grad(h, alpha)
        return g
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
        if time.time() - t_start > time_budget:
            return project_to_sum(h0, target_sum)
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

C(a, v)[k] = Σ_n a[n+k] · conj(v[n])

where the sum is over all n where both a[n+k] and v[n] are defined.

For mode='full', k ranges from -(len(v)-1) to len(a)-1, and the output is indexed 0 to len(a)+len(v)-2.

- If lag ≥ 0: n ranges from 0 to n-1-lag
- If lag < 0: n ranges from -lag to n-1

- For lag ≥ 0: output[n-1+lag] = Σ_{n=0}^{n-1-lag} a[n+lag] · v[n]
- For lag < 0 (let p = -lag > 0): output[n-1-p] = Σ_{n=p}^{n-1} a[n-p] · v[n]

from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -500.0, 500.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / len(corr)) / alpha
    
    g = np.zeros(n)
    # k = 0
    w0 = w[n-1]
    if w0 > 1e-15:
        g += w0 * (1.0 - 2.0 * h)
    # k > 0
    for k in range(1, n):
        wk_pos = w[n - 1 + k]
        wk_neg = w[n - 1 - k]
        # Positive lag k: out[n-1+k] = sum_{j=0}^{n-1-k} h[j+k]*(1-h[j])
        if wk_pos > 1e-15:
            idx_pos = np.arange(k, n)
            g[idx_pos] += wk_pos * (1.0 - h[idx_pos - k])
            idx_neg = np.arange(0, n - k)
            g[idx_neg] -= wk_pos * h[idx_neg + k]
        # Negative lag -k: out[n-1-k] = sum_{j=k}^{n-1} h[j-k]*(1-h[j])
        if wk_neg > 1e-15:
            idx_p2 = np.arange(0, n - k)
            g[idx_p2] += wk_neg * (1.0 - h[idx_p2 + k])
            idx_n2 = np.arange(k, n)
            g[idx_n2] -= wk_neg * h[idx_n2 - k]
    
    return sm_val, g

    y = h.copy()
    t = 1.0
    lr = lr_init
    c5_best = compute_c5(h)
    h_best = h.copy()
    lr_min = lr_init * 1e-4
    
    for it in range(n_iters):
        if time.time() - t_start > time_limit:
            break
        _, gy = smooth_max_and_grad(y, alpha)
        h_new = y - lr * gy
        h_new = project_to_sum(h_new, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        mo = (t - 1.0) / t_new
        y = np.clip(h_new + mo * (h_new - h), 0.0, 1.0)
        h = h_new
        t = t_new
        
        if it % 50 == 0 or it == n_iters - 1:
            c5_cur = compute_c5(h)
            if c5_cur < c5_best:
                c5_best = c5_cur
                h_best = h.copy()
                lr = min(lr * 1.05, 2.0)
            elif lr > lr_min:
                lr *= 0.75
    
    c5_final = compute_c5(h)
    if c5_final < c5_best:
        c5_best = c5_final
        h_best = h.copy()
    return h_best, c5_best

def slsqp_optimize(h0, n, target_sum, alpha, maxiter, t_start, time_limit):
    def obj_and_grad(x):
        h = np.clip(x, 0.0, 1.0)
        v, g = smooth_max_and_grad(h, alpha)
        return v, g
    
    def obj(x):
        v, _ = obj_and_grad(x)
        return v
    
    def grad(x):
        _, g = obj_and_grad(x)
        return g
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -500.0, 500.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / len(corr)) / alpha
    g = np.zeros(n)
    w0 = w[n - 1]
    if w0 > 1e-14:
        g += w0 * (1.0 - 2.0 * h)
    for k in range(1, n):
        wp = w[n - 1 + k]
        wn = w[n - 1 - k]
        if wp > 1e-14:
            ip = np.arange(k, n)
            g[ip] += wp * (1.0 - h[ip - k])
            iq = np.arange(0, n - k)
            g[iq] -= wp * h[iq + k]
        if wn > 1e-14:
            ir = np.arange(0, n - k)
            g[ir] += wn * (1.0 - h[ir + k])
            is_ = np.arange(k, n)
            g[is_] -= wn * h[is_ - k]
    return sm_val, g

def nesterov_pg(h0, n, target_sum, alpha, n_iters, lr_init, deadline):
    h = h0.copy()
    y = h.copy()
    t_param = 1.0
    lr = lr_init
    c5_best = compute_c5(h)
    h_best = h.copy()
    lr_floor = lr_init * 1e-5
    for it in range(n_iters):
        if time.time() > deadline:
            break
        _, gy = smooth_max_and_grad(y, alpha)
        h_new = y - lr * gy
        h_new = project_to_sum(h_new, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_param * t_param)) / 2.0
        mo = (t_param - 1.0) / t_new
        y = np.clip(h_new + mo * (h_new - h), 0.0, 1.0)
        h = h_new
        t_param = t_new
        if it % 50 == 0 or it == n_iters - 1:
            c5_cur = compute_c5(h)
            if c5_cur < c5_best:
                c5_best = c5_cur
                h_best = h.copy()
                lr = min(lr * 1.04, 1.0)
            elif lr > lr_floor:
                lr *= 0.72
    c5_final = compute_c5(h)
    if c5_final < c5_best:
        c5_best = c5_final
        h_best = h.copy()
    return h_best, c5_best

def slsqp_step(h0, n, target_sum, alpha, maxiter, deadline):
    def obj(x):
        hh = np.clip(x, 0.0, 1.0)
        v, _ = smooth_max_and_grad(hh, alpha)
        return v
    def grad(x):
        hh = np.clip(x, 0.0, 1.0)
        _, gg = smooth_max_and_grad(hh, alpha)
        return gg
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def coord_descend(h0, n, target_sum, max_rounds, rng, deadline):
    h = h0.copy()
    c5_best = compute_c5(h)
    h_best = h.copy()
    deltas = [0.01, -0.01, 0.005, -0.005, 0.02, -0.02, 0.002, -0.002]
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for i in rng.permutation(n):
            if time.time() > deadline:
                break
            h_save = h.copy()
            for d in deltas:
                h[i] = np.clip(h[i] + d, 0.0, 1.0)
                h = project_to_sum(h, target_sum)
                c5_new = compute_c5(h)
                if c5_new < c5_best - 1e-15:
                    c5_best = c5_new
                    h_best = h.copy()
                    improved = True
                    break
                h = h_save.copy()
        if not improved:
            break
    return h_best, c5_best

def interpolate_h(h_src, n_src, n_dst):
    x_old = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_new, x_old, h_src)
