    h = h0.copy()
    y = h.copy()
    t = 1.0
    lr = lr_init
    c5_cur = compute_c5(h)
    h_best = h.copy()
    c5_best = c5_cur
    t_start = time.time()
    stagnation = 0
    
    for it in range(max_iters):
        if time.time() - t_start > time_limit:
            break
        
        sm_val, g = smooth_max_and_grad(y, alpha)
        h_new = np.clip(y - lr * g, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        momentum = (t - 1.0) / t_new
        y = h_new + momentum * (h_new - h)
        y = np.clip(y, 0.0, 1.0)
        h = h_new
        t = t_new

        if c5_new < c5_best - 1e-12:
            c5_best = c5_new
            h_best = h.copy()
            lr = min(lr * 1.02, 0.5)
            stagnation = 0

def lsqpo_optimize(h0, n, target_sum, alpha, maxiter, time_limit):
    t_start = time.time()
    
    def obj_and_grad(x):
        h = np.clip(x, 0.0, 1.0)
        v, g = smooth_max_and_grad(h, alpha)
        return v, g
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]

    try:

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    nc = len(corr)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -700.0, 700.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    sm_val = m + np.log(s / nc) / alpha
    
    g = np.zeros(n)
    for k in range(-(n-1), n):
        wi = w[k + n - 1]
        if wi < 1e-18:
            continue
        js1 = max(0, k)
        je1 = min(n-1, k + n - 1)
        if js1 <= je1:
            g[js1:je1+1] += wi * (1.0 - h[js1-k:je1-k+1])
        js2 = max(0, -k)
        je2 = min(n-1, n-1-k)
        if je2 >= js2:
            g[js2:je2+1] -= wi * h[js2+k:je2+k+1]
    
    return sm_val, g

def compute_grad_matrix(h):
    """Compute the full gradient matrix for all lags."""
    n = len(h)
    nc = 2 * n - 1
    g = np.zeros((nc, n))
    for k in range(-(n-1), n):
        ki = k + n - 1
        js1 = max(0, k)
        je1 = min(n-1, k + n - 1)
        if js1 <= je1:
            g[ki, js1:je1+1] += (1.0 - h[js1-k:je1-k+1])
        js2 = max(0, -k)
        je2 = min(n-1, n-1-k)
        if js2 <= je2:
            g[ki, js2:je2+1] -= h[js2+k:je2+k+1]
    return g

def lp_optimize(h0, n, target_sum, max_iters, time_limit):
    """Successive linearization method."""
    t_start = time.time()
    h = h0.copy()
    c5_cur = compute_c5(h)
    
    for it in range(max_iters):
        if time.time() - t_start > time_limit:
            break
        
        corr = compute_corr(h)
        g_mat = compute_grad_matrix(h)
        
        # Set up LP: min t s.t. g_k^T h - t <= -corr_k + g_k^T h for all k
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        
        A_ub = np.zeros((2*n-1, n + 1))
        A_ub[:, :n] = g_mat
        A_ub[:, n] = -1.0
        b_ub = -corr + g_mat @ h
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target_sum])
        
        bounds = [(0.0, 1.0)] * n + [(-0.5, 2.0)]
        
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs',
                     options={'time_limit': max(1, time_limit - (time.time() - t_start))})
        
        if not res.success:
            break
        
        h_new = res.x[:n]
        # Line search
        best_h = h
        best_c5 = c5_cur
        for lr in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1]:
            h_try = h + lr * (h_new - h)
            h_try = project_to_sum(h_try, target_sum)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try
        h = best_h
        c5_cur = best_c5
    
    return h, c5_cur

def nesterov_gradient(h0, n, target_sum, alpha, max_iters, lr_init, time_limit):
    h = h0.copy()
    y = h.copy()
    t_mom = 1.0
    lr = lr_init
    c5_cur = compute_c5(h)
    h_best = h.copy()
    c5_best = c5_cur
    t_start = time.time()
    stagnation = 0
    
    for it in range(max_iters):
        if time.time() - t_start > time_limit:
            break
        
        sm_val, g = smooth_max_and_grad(y, alpha)
        h_new = np.clip(y - lr * g, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        momentum = (t_mom - 1.0) / t_new
        y = h_new + momentum * (h_new - h)
        y = np.clip(y, 0.0, 1.0)
        h = h_new
        t_mom = t_new
        
        c5_new = compute_c5(h)
        if c5_new < c5_best - 1e-13:
            c5_best = c5_new
            h_best = h.copy()
            lr = min(lr * 1.01, 0.5)
            stagnation = 0
        else:
            lr *= 0.95
            stagnation += 1
            if stagnation > 300:
                break
    
    return h_best, c5_best

        improved = False

            if best_delta != 0.0:
