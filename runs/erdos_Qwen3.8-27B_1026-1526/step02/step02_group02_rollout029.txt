def slp_step(h, n, m_active=30, method='highs'):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = c.max()
    
    # Get top m active lags
    top_idx = np.argsort(c)[-m_active:]
    
    # Build LP
    # Variables: h[0], ..., h[n-1], t
    # Minimize t
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    A_ub = []
    b_ub = []
    
    for k_idx in top_idx:
        k = k_idx - (n - 1)  # convert to lag
        grad = compute_lag_gradient(h, k, n)
        c_val = (c[k_idx] / dx)  # correlation value without dx
        row = np.zeros(n + 1)
        row[:n] = dx * grad
        row[-1] = -1.0
        rhs = dx * (grad @ h - c_val)
        A_ub.append(row)
        b_ub.append(rhs)
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = [(0, 1)] * n + [(0, None)]
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.success:
        return result.x[:n], c_obj
    return h, c_max

            stall = 0
        else:
            stall += 1

        if stall > 8:
            break

        threshold = c_max - 0.004
        active_idx = np.where(c_all >= threshold)[0]

        if len(active_idx) > m_active:
            top_order = np.argsort(c_all[active_idx])[::-1][:m_active]
            active_idx = active_idx[top_order]
        elif len(active_idx) < 5:
            active_idx = np.argsort(c_all)[::-1][:min(m_active, 2 * n - 1)]

        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0

        A_ub_list = []
        b_ub_list = []

        for k_idx in active_idx:
            k = k_idx - (n - 1)
            grad = compute_lag_gradient(h, k, n)
            c_val = c_all[k_idx] / dx

            row = np.zeros(n + 1)
            row[:n] = dx * grad
            row[-1] = -1.0
            rhs = dx * (grad @ h - c_val)

            A_ub_list.append(row)
            b_ub_list.append(rhs)

        A_ub = np.array(A_ub_list)
        b_ub = np.array(b_ub_list)

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])

        bounds = [(0, 1)] * n + [(0, None)]

        try:
            result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                             bounds=bounds, method='highs')
        except Exception:
            break

        if not result.success:
            stall += 3
            continue

        h_lp = result.x[:n]

        best_alpha = 0.0
        best_alpha_c5 = compute_c5_from_h(h)

        for alpha in [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.1, 0.05]:
            h_trial = (1.0 - alpha) * h + alpha * h_lp
            h_trial = project_feasible(h_trial, n)
            c5_trial = compute_c5_from_h(h_trial)
            if c5_trial < best_alpha_c5 - 1e-14:
                best_alpha = alpha
                best_alpha_c5 = c5_trial
                if c5_trial < 0.3807:
                    break

        if best_alpha == 0.0:
            stall += 3
            continue

        h = (1.0 - best_alpha) * h + best_alpha * h_lp
        h = project_feasible(h, n)

    return best_h, best_c5


def c5_logsumexp(h, T):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = c.max()
    shifted = T * (c - c_max)
    shifted = np.clip(shifted, -400, 400)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / T)


def smooth_obj(x, T, penalty):
    c5_val = c5_logsumexp(x, T)
    n = len(x)
    sum_violation = np.sum(x) - n / 2.0
    return c5_val + penalty * sum_violation ** 2


def lbfgsb_optimize(h_init, n, T, penalty, maxiter=500):
    bounds = [(0.0, 1.0)] * n
    try:
        result = minimize(
            smooth_obj, h_init, args=(T, penalty),
            method='L-BFGS-B', bounds=bounds,
            options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14}
        )
    except Exception:
        return h_init, compute_c5_from_h(h_init)
    if result.x is not None:
        h_out = project_feasible(result.x, n)
        return h_out, compute_c5_from_h(h_out)
    return h_init, compute_c5_from_h(h_init)


def slsqp_optimize(h_init, n, top_k=15, maxiter=3000):
    c5_init = compute_c5_from_h(h_init)
    c_all = compute_all_c(h_init)
    active_lags = np.argsort(c_all)[::-1][:top_k]

    def obj(x):
        return x[-1]

    def ineq_constr(x):
        h = x[:-1]
        t = x[-1]
        dx = 2.0 / n
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        return t - c[active_lags]

    def eq_constr(x):
        h = x[:-1]
        return np.sum(h) - n / 2.0

    x0 = np.concatenate([h_init, [c5_init + 0.05]])
    bounds = [(0.0, 1.0)] * n + [(0.01, 2.0)]
    constraints = [
        {'type': 'ineq', 'fun': ineq_constr},
        {'type': 'eq', 'fun': eq_constr}
    ]

    try:
        result = minimize(obj, x0, method='SLSQP', bounds=bounds,
                          constraints=constraints,
                          options={'maxiter': maxiter, 'ftol': 1e-15})
    except Exception:
        return h_init, c5_init

    if result.x is not None:
        h_out = project_feasible(result.x[:-1], n)
        return h_out, compute_c5_from_h(h_out)
    return h_init, c5_init


def active_lag_power_obj(x, active_idx, p, n):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    active_c = c[active_idx]
    return float(np.mean(active_c ** p))


def powered_lbfgsb(h_init, n, p=16, maxiter=500):
    c_all = compute_all_c(h_init)
    c_max = c_all.max()
    threshold = c_max - 0.005
    active_idx = np.where(c_all >= threshold)[0]
    if len(active_idx) > 30:
        active_idx = np.argsort(c_all[active_idx])[::-1][:30]

    bounds = [(0.0, 1.0)] * n
    try:
        result = minimize(
            active_lag_power_obj, h_init, args=(active_idx, p, n),
            method='L-BFGS-B', bounds=bounds,
            options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-13}
        )
    except Exception:
        return h_init, compute_c5_from_h(h_init)

    if result.x is not None:
        h_out = project_feasible(result.x, n)
        return h_out, compute_c5_from_h(h_out)
    return h_init, compute_c5_from_h(h_init)


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(135, 0.5)

    n = len(h0)
    h0 = project_feasible(h0, n)

    best_h = h0.copy()
    best_c5 = compute_c5_from_h(best_h)

    if time.time() - t0 < budget_s - 100:
        best_h, best_c5 = slp_optimize(best_h, n, m_active=30, max_iter=60,
                                       t0=t0, budget_s=budget_s)

    if time.time() - t0 < budget_s - 80:
        h_new, c5_new = slp_optimize(best_h, n, m_active=60, max_iter=50,
                                     t0=t0, budget_s=budget_s)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    if time.time() - t0 < budget_s - 60:
        h_new, c5_new = slp_optimize(best_h, n, m_active=100, max_iter=40,
                                     t0=t0, budget_s=budget_s)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    if time.time() - t0 < budget_s - 50:
        h_new, c5_new = powered_lbfgsb(best_h, n, p=16, maxiter=400)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    if time.time() - t0 < budget_s - 40:
        h_new, c5_new = powered_lbfgsb(best_h, n, p=32, maxiter=300)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    if time.time() - t0 < budget_s - 50:
        h_new, c5_new = slsqp_optimize(best_h, n, top_k=12, maxiter=4000)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    if time.time() - t0 < budget_s - 40:
        h_new, c5_new = lbfgsb_optimize(best_h, n, T=40.0, penalty=100.0, maxiter=500)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    n_perturb = 40
    if time.time() - t0 < budget_s - 50:
        for attempt in range(n_perturb):
            if time.time() - t0 > budget_s - 25:
                break

            frac = attempt / max(1, n_perturb)
            scale = 0.06 * (1.0 - frac) ** 0.8 + 0.002

            r = rng.random()
            if r < 0.3:
                h_pert = best_h + scale * rng.standard_normal(n)
            elif r < 0.5:
                n_modes = rng.integers(2, 5)
                h_pert = best_h.copy()
                for _ in range(n_modes):
                    freq = rng.integers(1, max(2, n // 4))
                    phase = rng.random() * 2.0 * np.pi
                    amp = scale / np.sqrt(n_modes)
                    h_pert += amp * np.cos(2.0 * np.pi * freq * np.arange(n) / n + phase)
            elif r < 0.7:
                n_push = max(5, int(n * 0.2))
                h_pert = best_h.copy()
                push_idx = rng.choice(n, min(n_push, n), replace=False)
                for idx in push_idx:
                    if rng.random() < 0.5:
                        h_pert[idx] = min(1.0, h_pert[idx] + scale * 4.0)
                    else:
                        h_pert[idx] = max(0.0, h_pert[idx] - scale * 4.0)
            else:
                n_blocks = rng.integers(3, max(4, n // 15))
                h_pert = best_h.copy()
                for _ in range(n_blocks):
                    bsize = rng.integers(5, max(6, n // 8))
                    start = rng.integers(0, max(1, n - bsize))
                    h_pert[start:start + bsize] += scale * 3.0 * (rng.random() - 0.5)

            h_pert = project_feasible(h_pert, n)

            h_new, c5_new = slp_optimize(h_pert, n, m_active=40, max_iter=30,
                                         t0=t0, budget_s=budget_s)
            if c5_new < best_c5 - 1e-12:
                best_h = h_new
                best_c5 = c5_new

    if time.time() - t0 < budget_s - 40:
        for n_try in [200, 300, 360]:
            if time.time() - t0 > budget_s - 20:
                break
            x_old = np.linspace(0, 2, n, endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            f_interp = interp1d(x_old, best_h, kind='linear', fill_value='extrapolate')
            h_interp = np.clip(f_interp(x_new), 0, 1)
            h_interp = project_feasible(h_interp, n_try)

            h_new, c5_new = slp_optimize(h_interp, n_try, m_active=40, max_iter=40,
                                         t0=t0, budget_s=budget_s)
            if c5_new < best_c5 - 1e-12:
                best_h = h_new
                best_c5 = c5_new
                n = n_try

    if time.time() - t0 < budget_s - 20:
        best_h, best_c5 = slp_optimize(best_h, n, m_active=50, max_iter=50,
                                       t0=t0, budget_s=budget_s)

    if time.time() - t0 < budget_s - 10:
        h_new, c5_new = slsqp_optimize(best_h, n, top_k=10, maxiter=5000)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    if time.time() - t0 < budget_s - 5:
        for attempt in range(20):
            if time.time() - t0 > budget_s - 2:
                break
            scale = 0.002 + 0.008 * (1.0 - attempt / 20.0) ** 2
            h_pert = best_h + scale * rng.standard_normal(n)
            h_pert = project_feasible(h_pert, n)
            h_new, c5_new = slp_optimize(h_pert, n, m_active=50, max_iter=20,
                                         t0=t0, budget_s=budget_s)
            if c5_new < best_c5 - 1e-13:
                best_h = h_new
                best_c5 = c5_new

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5_from_h(best_h)

    return best_h, best_c5, n
