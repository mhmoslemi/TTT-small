        if np.sum(proj) > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-14:
        order = np.argsort(np.abs(result - 0.5))
        for idx in order:
            if diff > 0:
                room = 1.0 - result[idx]
                adj = min(diff, room)
            else:
                room = result[idx]
                adj = max(diff, -room)
            result[idx] += adj
            diff -= adj
            if abs(diff) < 1e-14:
                break
    return result

def smooth_max(values, tau):
    v = values / tau
    vmax = np.max(v)
    return tau * (vmax + np.log(np.sum(np.exp(v - vmax))))

def obj_smooth(x, n, tau):
    vals = get_corr_scaled(x)
    return smooth_max(vals, tau)

def sum_constraint(x, n):
    return np.sum(x) - n / 2.0

def pair_cd(h, n, n_sweeps=10, rng=None):
    """Coordinate descent with pair moves."""
    if rng is None:
        rng = np.random.RandomState()
    best = h.copy()
    best_c5 = evaluate_c5(best)
    deltas = [0.002, 0.005, 0.01, 0.02, 0.03, 0.05, 0.08]
    for sweep in range(n_sweeps):
        improved = False
        pairs = []
        for _ in range(20):
            i = rng.randint(n)
            j = rng.randint(n)
            if i != j:
                pairs.append((i, j))
        for (i, j) in pairs:
            for d in deltas:
                for sign in [1, -1]:
                    h_new = best.copy()
                    h_new[i] += sign * d
                    h_new[j] -= sign * d
                    h_new = project_feasible(h_new, n)
                    c5 = evaluate_c5(h_new)
                    if c5 < best_c5 - 1e-12:
                        best = h_new
                        best_c5 = c5
                        improved = True
                        break
                if improved:
                    break
        if not improved:
            break
    return best, best_c5

def active_set_grad(h, n, eps_gap=0.005):
    """Compute gradient using active set (near-max terms)."""
    vals = get_corr_scaled(h)
    vmax = vals.max()
    active = np.where(vals > vmax - eps_gap)[0]
    # Compute gradient numerically for each active term, then average
    grad = np.zeros(n)
    eps = 1e-6
    for k in active:
        for j in range(n):
            h_p = h.copy()
            h_p[j] = min(1.0, h[j] + eps)
            h_m = h.copy()
            h_m[j] = max(0.0, h[j] - eps)
            vp = get_corr_scaled(h_p)[k]
            vm = get_corr_scaled(h_m)[k]
            denom = h_p[j] - h_m[j]
            if denom > 1e-10:
                grad[j] += (vp - vm) / denom
    grad /= len(active)
    return grad, vmax, active
