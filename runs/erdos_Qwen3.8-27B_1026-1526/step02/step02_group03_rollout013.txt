        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            best_step = 0.0
            best_val = best_c5
            for mult in [1.0, 0.5, 0.25, 2.0]:
                for sign in [1, -1]:
                    d = sign * step * mult
                    h_new = best.copy()
                    h_new[i] = float(np.clip(h_new[i] + d, 0.0, 1.0))
                    h_new = project_feasible(h_new, n)
                    c5 = evaluate_c5(h_new)
                    if c5 < best_val - 1e-13:
                        best_val = c5
                        best_step = d
            if best_step != 0.0:
                best[i] = float(np.clip(best[i] + best_step, 0.0, 1.0))
                best = project_feasible(best, n)
                best_c5 = evaluate_c5(best)
                improved = True
        if not improved:
            step *= 0.55
            stall_count += 1
            if step < 0.0015 or stall_count > 12:
                break
        else:
            stall_count = 0
    return best, best_c5

def block_optimize(h, n, deadline, rng):
    best = h.copy()
    best_c5 = evaluate_c5(best)
    for it in range(5000):
        if time.time() > deadline:
            break
        blen = int(rng.randint(2, min(25, n // 4)))
        bstart = int(rng.randint(0, n - blen))
        delta = float(rng.choice([-1.0, 1.0])) * float(rng.uniform(0.005, 0.1))
        h_new = best.copy()
        h_new[bstart:bstart + blen] += delta
        h_new = project_feasible(h_new, n)
        c5 = evaluate_c5(h_new)
        if c5 < best_c5 - 1e-13:
            best = h_new
            best_c5 = c5
    return best, best_c5

def swap_optimize(h, n, deadline, rng):
    best = h.copy()
    best_c5 = evaluate_c5(best)
    for it in range(8000):
        if time.time() > deadline:
            break
        idx = rng.randint(0, n, 2)
        i, j = int(idx[0]), int(idx[1])
        if i == j:
            continue
        delta = float(rng.uniform(0.01, 0.08))
        h_new = best.copy()
        if best[i] >= delta and best[j] <= 1.0 - delta:
            h_new[i] -= delta
            h_new[j] += delta
        elif best[j] >= delta and best[i] <= 1.0 - delta:
            h_new[j] -= delta
            h_new[i] += delta
        else:
            continue
        h_new = project_feasible(h_new, n)
        c5 = evaluate_c5(h_new)
        if c5 < best_c5 - 1e-13:
            best = h_new
            best_c5 = c5
    return best, best_c5

def lag_refine(h, n, deadline):
    rng_local = np.random.RandomState(int(time.time() * 1000) % 10000)
    best = h.copy()
    best_c5 = evaluate_c5(best)
    current = best.copy()
    for it in range(500):
        if time.time() > deadline:
            break
        top_k = int(rng_local.choice([3, 4, 5, 6, 8]))
        h_opt = slsqp_opt(current, n, top_lag_obj, (n, top_k, 0.003), maxiter=60)
        c5 = evaluate_c5(h_opt)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best = h_opt.copy()
            current = h_opt.copy()
        else:
            noise = 0.004 + 0.015 * max(0.0, 1.0 - it / 80.0)
            current = project_feasible(current + rng_local.randn(n) * noise, n)
    return best, best_c5

def multi_tau_slsqp(h0, n, deadline, taus, maxiter_each=50):
    current = h0.copy()
    for tau in taus:
        if time.time() > deadline:
            break
        current = slsqp_opt(current, n, smooth_max_obj, (n, tau), maxiter=maxiter_each)
    return current

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    best_h = None
    best_c5 = float('inf')

    try:
        best_h = np.array(initial_h_values, dtype=float)
        best_c5 = evaluate_c5(best_h)
    except Exception:
        pass

    if best_h is None:
        n = 77
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c5 = evaluate_c5(best_h)

    n = len(best_h)
    x = np.linspace(0.0, 2.0, n, endpoint=False)

    t1 = t0 + budget_s * 0.22
    t2 = t0 + budget_s * 0.50
    t3 = t0 + budget_s * 0.78
    t4 = t0 + budget_s * 0.97

    # Phase 1: Multi-start SLSQP with smooth max
    starts = [best_h.copy()]
    for sc in [0.03, 0.06, 0.10, 0.15]:
        starts.append(project_feasible(best_h + rng.randn(n) * sc, n))
    for freq in [1.0, 1.5, 2.0, 2.5, 3.0, 3.5]:
        for amp in [0.20, 0.30, 0.40]:
            ph = rng.uniform(0.0, 6.2832)
            hs = 0.5 + amp * np.sin(2.0 * np.pi * freq * x / 2.0 + ph)
            starts.append(project_feasible(hs, n))
    for _ in range(3):
        starts.append(project_feasible(rng.rand(n), n))
    hs_tri = np.abs(np.mod(np.arange(n, dtype=float) / n * 3.0, 2.0) - 1.0)
    starts.append(project_feasible(hs_tri, n))
    hs_lin = np.linspace(0.05, 0.95, n)
    starts.append(project_feasible(hs_lin, n))

    taus = [0.15, 0.08, 0.04, 0.02, 0.01, 0.005]
    for h0 in starts:
        if time.time() > t1:
            break
        h_cur = multi_tau_slsqp(h0, n, t1, taus, maxiter_each=45)
        c5 = evaluate_c5(h_cur)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_cur.copy()

    # Phase 2: Lag-focused iterative refinement
    best_h, best_c5 = lag_refine(best_h, n, t2)

    # Phase 3: Coordinate descent + block + swap
    h_cd, c5_cd = line_search_coord(best_h, n, t3, base_step=0.04)
    if c5_cd < best_c5:
        best_c5 = c5_cd
        best_h = h_cd.copy()

    h_bm, c5_bm = block_optimize(best_h, n, t3, rng)
    if c5_bm < best_c5:
        best_c5 = c5_bm
        best_h = h_bm.copy()

    h_sw, c5_sw = swap_optimize(best_h, n, t3, rng)
    if c5_sw < best_c5:
        best_c5 = c5_sw
        best_h = h_sw.copy()

    h_cd2, c5_cd2 = line_search_coord(best_h, n, t3, base_step=0.02)
    if c5_cd2 < best_c5:
        best_c5 = c5_cd2
        best_h = h_cd2.copy()

    # Phase 4: Final SLSQP polish + fine coordinate descent
    current = best_h.copy()
    fine_taus = [0.005, 0.003, 0.001, 0.0005]
    for it in range(200):
        if time.time() > t4:
            break
        h_opt = slsqp_opt(current, n, smooth_max_obj, (n, 0.002), maxiter=100, ftol=1e-14)
        c5 = evaluate_c5(h_opt)
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            best_h = h_opt.copy()
            current = h_opt.copy()
        else:
            current = project_feasible(current + rng.randn(n) * 0.002, n)

    if time.time() < t0 + budget_s * 0.99:
        h_cd3, c5_cd3 = line_search_coord(best_h, n, t0 + budget_s * 0.99, base_step=0.01)
        if c5_cd3 < best_c5:
            best_c5 = c5_cd3
            best_h = h_cd3.copy()

    final_c5 = evaluate_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, n
