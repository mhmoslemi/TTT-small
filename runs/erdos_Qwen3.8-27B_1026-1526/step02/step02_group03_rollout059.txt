def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)

        else:

        for j in range(shift, n):
            grad[j] -= h[j - shift] * dx

        # d/dh[j] = (1-h[j-s])*dx for j >= s (first factor)
        #          + (-h[j+s])*dx for j < n-s (second factor, h[j] = h[(j+s)-s])
        for j in range(s, n):
            grad[j] += (1.0 - h[j - s]) * dx
        for j in range(n - s):
            grad[j] -= h[j + s] * dx
    return grad

    corr = np.correlate(x, 1.0 - x, mode="full")
    return smooth_max(corr, tau) * dx

def cons_sum(x, n):
    return np.sum(x) - n / 2.0

def compute_subgradient_topk(h, n, k=3):
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    top_k = np.argsort(corr)[-k:]
    grad = np.zeros(n)
    for idx in top_k:
        shift = int(idx) - (n - 1)
        if shift >= 0:
            s = shift
            if s < n:
                grad[:n-s] += (1.0 - h[s:]) * dx
                grad[s:] -= h[:n-s] * dx
        else:
            s = -shift
            if s < n:
                grad[s:] += (1.0 - h[:n-s]) * dx
                grad[:n-s] -= h[s:] * dx
    return grad / k

def nesterov_subgrad(h0, n, n_iter, alpha0, decay, momentum, topk=3):
    h = h0.astype(float).copy()
    v = np.zeros(n)
    best_h = h0.copy()
    best_val = evaluate_c5(h0)
    for t in range(n_iter):
        alpha = alpha0 / (1.0 + t * decay)
        y = project_feasible(h + momentum * v, n)
        grad = compute_subgradient_topk(y, n, topk)
        v_new = momentum * v - alpha * grad
        h_new = project_feasible(h + v_new, n)
        val = evaluate_c5(h_new)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
        v = v_new
        h = h_new
    return best_h, best_val

def resample_h(h, n_old, n_new):
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate', bounds_error=False)
    h_new = np.clip(f(x_new), 0, 1)
    return project_feasible(h_new, n_new)

def coord_desc(h, n, time_limit, start_time, step_scales=None):
    best = h.copy()
    best_c5 = evaluate_c5(best)
    if step_scales is None:
        step_scales = [0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    sweep = 0
    while time.time() - start_time < time_limit:
        improved = False
        for i in range(n):
            if time.time() - start_time > time_limit:
                break
            for d in step_scales:
                for sign in [1, -1]:
                    h_new = best.copy()
                    h_new[i] = best[i] + sign * d
                    if h_new[i] < -0.01 or h_new[i] > 1.01:
                        continue
                    h_new = project_feasible(h_new, n)
                    c5 = evaluate_c5(h_new)
                    if c5 < best_c5:
                        best = h_new
                        best_c5 = c5
                        improved = True
        sweep += 1
        if not improved:
            break
        # Reduce steps as we converge
        if sweep > 8:
            step_scales = [s * 0.5 for s in step_scales]
    return best, best_c5
