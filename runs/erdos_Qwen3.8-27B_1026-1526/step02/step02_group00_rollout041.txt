```python
    corr = np.correlate(h, 1.0 - h, mode='full')
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def get_corr_weights(h, alpha):
    """Compute correlation and softmax weights"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -500.0, 500.0)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    return corr, w

def smooth_max_obj(h, alpha):
    corr, w = get_corr_weights(h, alpha)
    return np.max(corr) + np.log(np.sum(np.exp(np.clip(alpha*(corr - np.max(corr)), -500, 500)))) / alpha

def grad_smooth_max(h, alpha):
    """Gradient of smooth max of correlations"""
    n = len(h)
    corr, w = get_corr_weights(h, alpha)
    obj = np.max(corr) + np.log(np.sum(np.exp(np.clip(alpha*(corr - np.max(corr)), -500, 500)))) / alpha
    
    grad = np.zeros(n)
    active = np.where(w > 1e-10)[0]
    for k in active:
        wk = w[k]
        # (1-h[j-k]) valid when 0 <= j-k < n => j in [k, min(n, k+n))
        lo1 = k
        hi1 = min(n, k + n)
        if lo1 < hi1:
            grad[lo1:hi1] += wk * (1.0 - h[lo1-k:hi1-k])
        # -h[j+k] valid when 0 <= j+k < n => j in [0, min(n, n-k))
        hi2 = min(n, n - k)
        if hi2 > 0:
            grad[0:hi2] -= wk * h[k:hi2+k]
    
    return obj, grad

def projected_grad_descent(h0, target_sum, alpha, n_iter, time_limit, verbose=False):
    """Projected gradient descent with backtracking and Nesterov momentum"""
    n = len(h0)
    h = h0.copy()
    y = h0.copy()
    t = 1.0
    step = 1.0 / (alpha + 2.0 * n)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    
    for it in range(n_iter):
        obj, grad = grad_smooth_max(y, alpha)
        
        # Backtracking line search
        s = step
        h_trial = project_to_sum(y - s * grad, target_sum)
        obj_trial, _ = grad_smooth_max(h_trial, alpha)
        backtrack_count = 0
        while obj_trial > obj + 1e-10 * abs(obj) and s > 1e-10 and backtrack_count < 20:
            s *= 0.5
            h_trial = project_to_sum(y - s * grad, target_sum)
            obj_trial, _ = grad_smooth_max(h_trial, alpha)
            backtrack_count += 1
        
        h_new = h_trial
        
        # Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = h_new + ((t - 1.0) / t_new) * (h_new - h)
        y = project_to_sum(y, target_sum)
        
        # Evaluate true objective
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        
        # Update step size adaptively
        if backtrack_count == 0:
            step = min(s * 1.5, 1.0)
        else:
            step = s
        
        h = h_new
        t = t_new
        
        if time.time() > time_limit:
            break
    
    return best_h, best_c5

def binary_local_search(h_bin, n, target_sum, max_rounds, time_limit):
    """Local search on binary solutions using swap moves"""
    h = h_bin.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    current_max = np.max(corr)
    
    for rnd in range(max_rounds):
        if time.time() > time_limit:
            break
        
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        best_improvement = 0
        best_pair = None
        best_corr = None
        
        # Try a subset of swaps
        n_ones = len(ones)
        n_zeros = len(zeros)
        n_trials = min(n_ones * n_zeros, 5000)
        
        for trial in range(n_trials):
            i = ones[trial % n_ones]
            j = zeros[(trial * 7 + 3) % n_zeros]  # Use a simple hash for variety
            
            # Compute change in correlations after swap
            # Δcorr[k] = h[i-k] - h[j-k] + h[i+k] - h[j+k]
            delta = np.zeros(2*n-1)
            for k in range(2*n-1):
                d = 0.0
                # h[i-k] valid when 0 <= i-k < n
                if 0 <= i - k < n:
                    d += h[i-k]
                # -h[j-k] valid when 0 <= j-k < n
                if 0 <= j - k < n:
                    d -= h[j-k]
                # h[i+k] valid when 0 <= i+k < n
                if 0 <= i + k < n:
                    d += h[i+k]
                # -h[j+k] valid when 0 <= j+k < n
                if 0 <= j + k < n:
                    d -= h[j+k]
                delta[k] = d
            
            new_corr = corr + delta
            new_max = np.max(new_corr)
            improvement = current_max - new_max
            
            if improvement > best_improvement:
                best_improvement = improvement
                best_pair = (i, j)
                best_corr = new_corr
        
        if best_pair is None:
            break
        
        i, j = best_pair
        h[i] = 0.0
        h[j] = 1.0
        corr = best_corr
        current_max = np.max(corr)
    
    return h, current_max * (2.0 / n)
```
