    v = -values / tau

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return float(np.max(corr) * dx)

def autocorr_pos(h):
    n = len(h)
    full = np.correlate(h, h, mode="full")
    return full[n - 1:]

def softmin_w(R, tau):
    v = -R / tau
    v -= v.min()
    e = np.exp(v)
    return e / e.sum()

def grad_softmin(h, tau):
    n = len(h)
    R = autocorr_pos(h)
    w = softmin_w(R, tau)
    g = 2.0 * w[0] * h.copy()
    for s in range(1, n):
        if w[s] < 1e-12:
            continue
        ws = w[s]
        g[:n - s] += ws * h[s:]
        g[s:] += ws * h[:n - s]
    return g

def proj_feasible(h, tgt):
    h = np.clip(h, 0.0, 1.0).copy()
    s = h.sum()
    if abs(s - tgt) < 1e-14:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        p = np.clip(h + mid, 0.0, 1.0)
        if p.sum() > tgt:
            hi = mid
        else:
            lo = mid
    r = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    d = tgt - r.sum()
    if abs(d) > 1e-13:
        m = (r > 1e-8) & (r < 1.0 - 1e-8)
        if m.any():
            r[m] = np.clip(r[m] + d / float(m.sum()), 0.0, 1.0)
        else:
            i = int(np.argmin(np.abs(r - 0.5)))
            r[i] = np.clip(r[i] + d, 0.0, 1.0)
    return r

def sum_eq_constraint(x, tgt):
    return x.sum() - tgt

def slsqp_objective(x, n, tau):
    R = autocorr_pos(x)
    neg_rat = -R / tau
    mx = neg_rat.max()
    return tau * (mx + np.log(np.sum(np.exp(neg_rat - mx))))

def sg_ascent(h, n, tlim, rng):
    tgt = n / 2.0
    best = h.copy()
    bc5 = evaluate_c5(best)
    x = h.copy()
    alpha = 0.008
    alpha_max = 0.025
    tau0 = 0.08
    tau_min = 0.002
    stall = 0
    stall_max = 4000
    t0 = time.time()
    last_improve = t0

    while time.time() < tlim:
        elapsed = time.time() - t0
        total = tlim - t0
        frac = min(1.0, elapsed / max(total, 1e-9))
        tau = max(tau_min, tau0 * (tau_min / tau0) ** (frac ** 0.6))

        g = grad_softmin(x, tau)
        g -= g.mean()
        gn = np.linalg.norm(g)
        if gn < 1e-13:
            break

        step = alpha
        ok = False
        for _ in range(14):
            cand = proj_feasible(x + step * g / gn, tgt)
            c5 = evaluate_c5(cand)
            if c5 < bc5 - 1e-13:
                x = cand
                bc5 = c5
                best = cand.copy()
                stall = 0
                last_improve = time.time()
                ok = True
                break
            step *= 0.55

        if ok:
            alpha = min(alpha_max, alpha * 1.03)
        else:
            stall += 1
            if stall > stall_max or (time.time() - last_improve) > 30.0:
                ns = 0.008 * max(0.05, 1.0 - frac)
                x = proj_feasible(best + rng.randn(n) * ns, tgt)
                stall = 0
                last_improve = time.time()
            alpha = max(1e-6, alpha * 0.85)

    return best, bc5

def slsqp_polish(h, n, tlim, rng):
    tgt = n / 2.0
    best = h.copy()
    bc5 = evaluate_c5(best)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq_constraint, 'args': (tgt,)}]
    t0 = time.time()

    tau_list = [0.02, 0.01, 0.005, 0.003, 0.001]
    for tau in tau_list:
        if time.time() > tlim:
            break
        max_iters_this = 120
        restarts = 4
        for r in range(restarts):
            if time.time() > tlim:
                break
            noise = 0.015 if r == 0 else 0.008
            x0 = proj_feasible(best + rng.randn(n) * noise, tgt)
            try:
                res = minimize(
                    slsqp_objective, x0, args=(n, tau),
                    method='SLSQP', bounds=bounds,
                    constraints=cons,
                    options={'maxiter': max_iters_this, 'ftol': 1e-15}
                )
                x = proj_feasible(res.x, tgt)
                c5 = evaluate_c5(x)
                if c5 < bc5:
                    bc5 = c5
                    best = x.copy()
            except Exception:
                pass

    return best, bc5

def coord_refine(h, n, tlim):
    tgt = n / 2.0
    dx = 2.0 / n
    best = h.copy()
    R = autocorr_pos(best)
    bc5 = evaluate_c5(best)

    base_deltas = np.array([-0.04, -0.025, -0.015, -0.008, -0.004, -0.0015,
                            0.0015, 0.004, 0.008, 0.015, 0.025, 0.04])
    t0 = time.time()

    for sweep in range(30):
        if time.time() > tlim:
            break
        any_improved = False
        for j in range(n):
            if time.time() > tlim:
                return best, bc5
            coef = np.zeros(n)
            for s in range(1, n):
                c = 0.0
                if j + s < n and j <= n - 1 - s:
                    c += best[j + s]
                if j - s >= 0:
                    c += best[j - s]
                coef[s] = c

            for d in base_deltas:
                nv = best[j] + d
                if nv < -1e-12 or nv > 1.0 + 1e-12:
                    continue
                nv = max(0.0, min(1.0, nv))
                d = nv - best[j]
                if abs(d) < 1e-14:
                    continue
                new_R = R + d * coef
                new_R[0] = R[0] + 2.0 * best[j] * d + d * d
                new_c5 = (n / 2.0 - new_R.min()) * dx
                if new_c5 < bc5 - 1e-13:
                    best[j] = nv
                    R = new_R
                    bc5 = new_c5
                    any_improved = True
        if not any_improved:
            break

    return best, bc5

def fine_coord_refine(h, n, tlim):
    tgt = n / 2.0
    dx = 2.0 / n
    best = h.copy()
    R = autocorr_pos(best)
    bc5 = evaluate_c5(best)
    t0 = time.time()

    delta_scale = 0.003
    for sweep in range(50):

        deltas = np.array([-3, -2, -1, 1, 2, 3]) * delta_scale

            coef = np.zeros(n)

                if j + s < n and j <= n - 1 - s:

                if j - s >= 0:
                    c += best[j - s]
                coef[s] = c
            for d in deltas:
                nv = best[j] + d
                if nv < 0.0 or nv > 1.0:

                d_actual = nv - best[j]
                new_R = R + d_actual * coef
                new_R[0] = R[0] + 2.0 * best[j] * d_actual + d_actual * d_actual
                new_c5 = (n / 2.0 - new_R.min()) * dx
                if new_c5 < bc5 - 1e-14:
                    best[j] = nv
                    R = new_R
                    bc5 = new_c5
                    any_improved = True
        if not any_improved:
            delta_scale *= 0.6
            if delta_scale < 1e-5:
                break

    return best, bc5
