    diff = target - np.sum(result)
    if abs(diff) > 1e-14:
        idx = int(np.argmin(np.abs(result - 0.5)))
        result[idx] = np.clip(result[idx] + diff, 0.0, 1.0)
    return result

def obj_slsqp(x, n, tau):
    dx = 2.0 / n
    g = 1.0 - x
    corr = np.correlate(x, g, mode="full")
    v = corr / tau
    vmax = np.max(v)
    return float(tau * (vmax + np.log(np.sum(np.exp(v - vmax)))) * dx)

def obj_topk(x, n, k, tau):
    dx = 2.0 / n
    g = 1.0 - x
    corr = np.correlate(x, g, mode="full")
    sorted_corr = np.sort(corr)[::-1]
    top = sorted_corr[:k]
    v = top / tau
    vmax = np.max(v)
    smooth = tau * (vmax + np.log(np.sum(np.exp(v - vmax))))
    return float(smooth * dx)

def constraint_fn(x, n):
    return np.sum(x) - n / 2.0

def coordinate_descent(h, n, time_limit, max_sweeps=500, verbose=False):
    best = h.copy()
    best_c5 = evaluate_c5(best)
    sweep = 0
    while sweep < max_sweeps and time.time() < time_limit:
        improved = False
        offset = sweep % n
        if sweep < 20:
            steps = [0.03, 0.06, 0.12, 0.25]
        elif sweep < 50:
            steps = [0.01, 0.03, 0.06, 0.12]
        elif sweep < 100:
            steps = [0.005, 0.01, 0.03]
        else:
            steps = [0.002, 0.005, 0.01]
        for si in range(n):
            i = (si + offset) % n
            old_val = best[i]
            best_local_c5 = best_c5
            best_local_val = old_val
            for s in steps:
                for sign in (1, -1):
                    v = old_val + sign * s
                    if v < -1e-9 or v > 1.0 + 1e-9:
                        continue
                    trial = best.copy()
                    trial[i] = v
                    trial = project_feasible(trial, n)
                    c5 = evaluate_c5(trial)
                    if c5 < best_local_c5 - 1e-15:
                        best_local_c5 = c5
                        best_local_val = v
            if best_local_val != old_val:
                best[i] = best_local_val
                best = project_feasible(best, n)
                best_c5 = best_local_c5
                improved = True
        sweep += 1
        if not improved and sweep > 15:
            break
    return best, best_c5

def fine_coordinate_descent(h, n, time_limit, max_sweeps=1000):
    best = h.copy()
    best_c5 = evaluate_c5(best)
    sweep = 0
    while sweep < max_sweeps and time.time() < time_limit:
        improved = False
        offset = (sweep * 31) % n
        steps = [0.001, 0.003, 0.008, 0.02]
        if sweep > 50:
            steps = [0.0005, 0.001, 0.003]
        if sweep > 100:
            steps = [0.0003, 0.0005, 0.001]
        for si in range(n):
            i = (si + offset) % n
            old_val = best[i]
            best_local_c5 = best_c5
            best_local_val = old_val
            for s in steps:
                for sign in (1, -1):
                    v = old_val + sign * s
                    if v < -1e-9 or v > 1.0 + 1e-9:
                        continue
                    trial = best.copy()
                    trial[i] = v
                    trial = project_feasible(trial, n)
                    c5 = evaluate_c5(trial)
                    if c5 < best_local_c5 - 1e-15:
                        best_local_c5 = c5
                        best_local_val = v
            if best_local_val != old_val:
                best[i] = best_local_val
                best = project_feasible(best, n)
                best_c5 = best_local_c5
                improved = True
        sweep += 1
        if not improved and sweep > 20:
            break
    return best, best_c5

def multi_resolution_step(h, n, t_limit, rng):
    best = h.copy()
    best_c5 = evaluate_c5(best)
    x_low = np.linspace(0, 2, n, endpoint=False)
    for scale in [2, 3]:
        if time.time() > t_limit:
            break
        n_high = n * scale
        x_high = np.linspace(0, 2, n_high, endpoint=False)
        f = interp1d(x_low, best, kind='cubic', fill_value='extrapolate', bounds_error=False)
        h_high = np.clip(f(x_high), 0, 1)
        h_high = project_feasible(h_high, n_high)
        h_high, c5_high = coordinate_descent(h_high, n_high, min(time.time() + 8, t_limit), max_sweeps=8)
        f2 = interp1d(x_high, h_high, kind='cubic', fill_value='extrapolate', bounds_error=False)
        h_down = np.clip(f2(x_low), 0, 1)
        h_down = project_feasible(h_down, n)
        c5_down = evaluate_c5(h_down)
        if c5_down < best_c5:
            best_c5 = c5_down
            best = h_down.copy()
    return best, best_c5

def structured_perturb(h, n, rng, strategy):
    perturb = h.copy()
    if strategy == 0:
        scale = rng.uniform(0.002, 0.04)
        perturb += rng.randn(n) * scale
    elif strategy == 1:
        seg_len = rng.randint(5, min(15, n // 3))
        start = rng.randint(0, n - seg_len)
        sign = rng.choice([-1, 1])
        amp = rng.uniform(0.03, 0.15)
        perturb[start:start + seg_len] += sign * amp
    elif strategy == 2:
        freq = rng.uniform(1.0, 5.0)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.02, 0.1)
        x = np.linspace(0, 2, n, endpoint=False)
        perturb += amp * np.sin(2 * np.pi * freq * x / 2.0 + phase)
    elif strategy == 3:
        idx = rng.randint(0, n)
        perturb[idx] = rng.uniform(0, 1)
    elif strategy == 4:
        swap = rng.choice(n, size=2, replace=False)
        perturb[swap[0]], perturb[swap[1]] = h[swap[1]], h[swap[0]]
        perturb += rng.randn(n) * 0.005
    elif strategy == 5:
        perturb = h[::-1].copy()
        perturb += rng.randn(n) * 0.01
    else:
        mask = rng.rand(n) < 0.15
        perturb[mask] = rng.rand(np.sum(mask))
    return project_feasible(perturb, n)

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    try:
        parent = np.array(initial_h_values, dtype=float)
    except Exception:
        parent = None

    if parent is None:
        n = 77
        parent = project_feasible(np.full(n, 0.5), n)

    n = len(parent)
    best_h = parent.copy()
    best_c5 = evaluate_c5(best_h)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fn, 'args': (n,)}]

    t_phase1 = t0 + budget_s * 0.10
    best_h, best_c5 = coordinate_descent(best_h, n, t_phase1, max_sweeps=80)

    t_phase2 = t0 + budget_s * 0.22
    if time.time() < t_phase2:
        current = best_h.copy()
        for tau in [0.01, 0.005, 0.002, 0.001]:
            if time.time() > t_phase2:
                break
            try:
                res = minimize(obj_slsqp, current, method='SLSQP', args=(n, tau),
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 25, 'ftol': 1e-13})
                current = project_feasible(res.x, n)
                c5 = evaluate_c5(current)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = current.copy()
            except Exception:
                pass
        current = best_h.copy()
        try:
            res = minimize(obj_topk, current, method='SLSQP', args=(n, 5, 0.01),
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 25, 'ftol': 1e-13})
            current = project_feasible(res.x, n)
            c5 = evaluate_c5(current)
            if c5 < best_c5:
                best_c5 = c5
                best_h = current.copy()
        except Exception:
            pass

    t_phase3 = t0 + budget_s * 0.72
    if time.time() < t_phase3:
        no_improve_count = 0
        while time.time() < t_phase3:
            frac = (time.time() - t0) / budget_s
            strat = rng.randint(0, 6)
            pert = structured_perturb(best_h, n, rng, strat)
            h_opt, c5 = coordinate_descent(pert, n, min(time.time() + 4, t_phase3), max_sweeps=12)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()
                no_improve_count = 0
            else:
                no_improve_count += 1
            if rng.rand() < 0.25:
                try:
                    tau = 0.003 * (1.0 + 0.5 * (1.0 - frac))
                    res = minimize(obj_slsqp, h_opt, method='SLSQP', args=(n, tau),
                                   bounds=bounds, constraints=cons,
                                   options={'maxiter': 20, 'ftol': 1e-13})
                    h_opt = project_feasible(res.x, n)
                    c5 = evaluate_c5(h_opt)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_opt.copy()
                        no_improve_count = 0
                except Exception:
                    pass
            if no_improve_count > 60:
                no_improve_count = 0
                big_pert = best_h + rng.randn(n) * 0.08
                big_pert = project_feasible(big_pert, n)
                h_opt, c5 = coordinate_descent(big_pert, n, min(time.time() + 3, t_phase3), max_sweeps=10)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt.copy()

    t_phase4 = t0 + budget_s * 0.85
    if time.time() < t_phase4:
        best_h, best_c5 = multi_resolution_step(best_h, n, t_phase4, rng)

    t_phase5 = t0 + budget_s * 0.98
    if time.time() < t_phase5:
        best_h, best_c5 = fine_coordinate_descent(best_h, n, t_phase5, max_sweeps=500)

    if time.time() < t0 + budget_s * 0.995:
        current = best_h.copy()
        try:
            res = minimize(obj_slsqp, current, method='SLSQP', args=(n, 0.0005),
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 50, 'ftol': 1e-15})
            current = project_feasible(res.x, n)
            c5 = evaluate_c5(current)
            if c5 < best_c5:
                best_c5 = c5
                best_h = current.copy()
        except Exception:
            pass

    best_h, best_c5 = fine_coordinate_descent(best_h, n, t0 + budget_s * 0.998, max_sweeps=100)

    final_c5 = evaluate_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, n
