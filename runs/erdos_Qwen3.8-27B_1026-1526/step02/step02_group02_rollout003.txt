```python
    # 
    # Actually, np.correlate(h, 1-h, mode='full')[idx] where idx = n-1+k (k is the lag)
    # For k >= 0: sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])
    # For k < 0: let m = -k, sum_{i=0}^{n-1-m} h[i]*(1-h[i+m])

                    grad[:n-k] += wk * dx * (-h[k:])  # h[j+k] for j in [0,n-1-k] -> h[k..n-1]
            else:

                # c(-m) = sum_i h[i]*(1-h[i+m]), i in [0, n-1-m]
                # d/dh[j]: -h[j+m] for j in [0, n-1-m]; (1-h[j-m]) for j in [m, n-1]
                if m < n:

                    grad[m:] += wk * dx * (1.0 - h[:n-m])  # h[j-m] for j in [m,n-1]
    
    return smooth_val, grad

def smooth_obj_and_grad(x, T, penalty):
```

- k = 0: sum_{m=0}^{n-1} h[m*(1-h[m]) = sum h[m] - sum h[m]^2
- k = 1: sum_{m=0}^{n-2} h[m+1]*(1-h[m])
- k = -1: sum_{m=1}^{n-1} h[m-1]*(1-h[m])

So output[n-1] = k=0 term, output[n-1+1] = k=1 term, etc.
output index = n-1+k for lag k.

This confirms my earlier analysis. Good.

```python
def smooth_c5_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    c_max = c.max()
    shifted = np.clip(T * (c - c_max), -400, 400)
    exps = np.exp(shifted)
    Z = np.sum(exps)
    w = exps / Z  # weights, length 2n-1
    
    smooth_val = c_max + np.log(Z) / T
    
    # Gradient
    grad = np.zeros(n)
    # k=0 term (index n-1):
    w0 = w[n-1] * dx
    grad += w0 * (1.0 - 2.0 * h)
    
    # Positive lags k=1 to n-1 (indices n to 2n-2):
    for k in range(1, n):
        wk = w[n-1+k]
        if abs(wk) < 1e-18:
            continue
        wk_d = wk * dx
        # Contribution: wk_d * [(1-h[j-k]) for j in [k,n-1]  -  h[j+k] for j in [0,n-1-k]]
        grad[k:] += wk_d * (1.0 - h[:n-k])
        grad[:n-k] -= wk_d * h[k:]
    
    # Negative lags k=-1 to -(n-1) (indices 0 to n-2):
    for k in range(1, n):
        wk = w[n-1-k]
        if abs(wk) < 1e-18:
            continue
        wk_d = wk * dx
        # c(-k) = sum_i h[i]*(1-h[i+k]), i in [0, n-1-k]
        # d/dh[j]: (1-h[j+k]) for j in [0, n-1-k]  -  h[j-k] for j in [k, n-1]
        grad[:n-k] += wk_d * (1.0 - h[k:])
        grad[k:] -= wk_d * h[:n-k]
    
    return smooth_val, grad
```

Wait, I need to double-check the negative lag gradient.

c(-k) = sum_{i=0}^{n-1-k} h[i] * (1-h[i+k])
= sum_{i=0}^{n-1-k} h[i] - sum_{i=0}^{n-1-k} h[i]*h[i+k]

d/dh[j]:
- From sum h[i]: +1 if 0 ≤ j ≤ n-1-k
- From -sum h[i]*h[i+k]: 
  - -h[i+k] if j = i (i.e., 0 ≤ j ≤ n-1-k)
  - -h[i] if j = i+k (i.e., 0 ≤ j-k ≤ n-1-k, i.e., k ≤ j ≤ n-1)

So d c(-k)/dh[j] = dx * [(1 - h[j+k]) * 1_{j ≤ n-1-k} - h[j-k] * 1_{j ≥ k}]

Which matches what I had. Good.

```python
import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(c)), c


def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            if not np.any(mask):
                break
            step = min(diff / np.sum(mask), (1.0 - h[mask]).min() * 0.999)
            h[mask] += step
        else:
            mask = h > 1e-15
            if not np.any(mask):
                break
            step = min(-diff / np.sum(mask), h[mask].min() * 0.999)
            h[mask] -= step
    return np.clip(h, 0.0, 1.0)


def smooth_c5_and_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    c_max = c.max()
    shifted = np.clip(T * (c - c_max), -400, 400)
    exps = np.exp(shifted)
    Z = np.sum(exps)
    w = exps / Z
    smooth_val = c_max + np.log(Z) / T
    grad = np.zeros(n)
    w0 = w[n - 1] * dx
    grad += w0 * (1.0 - 2.0 * h)
    for k in range(1, n):
        wk = w[n - 1 + k]
        if abs(wk) < 1e-18:
            continue
        wk_d = wk * dx
        grad[k:] += wk_d * (1.0 - h[:n - k])
        grad[:n - k] -= wk_d * h[k:]
    for k in range(1, n):
        wk = w[n - 1 - k]
        if abs(wk) < 1e-18:
            continue
        wk_d = wk * dx
        grad[:n - k] += wk_d * (1.0 - h[k:])
        grad[k:] -= wk_d * h[:n - k]
    return smooth_val, grad


def smooth_obj(x, T, penalty):
    val, _ = smooth_c5_and_grad(x, T)
    sv = np.sum(x) - len(x) / 2.0
    val += penalty * sv * sv
    return val


def smooth_obj_grad(x, T, penalty):
    n = len(x)
    val, grad = smooth_c5_and_grad(x, T)
    sv = np.sum(x) - n / 2.0
    val += penalty * sv * sv
    grad += 2.0 * penalty * sv
    return val, grad


def slsqp_obj(x):
    return x[-1]


def slsqp_ineq(x):
    h = x[:-1]
    t = x[-1]
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return t - np.max(c)


def slsqp_eq(x):
    h = x[:-1]
    return np.sum(h) - len(h) / 2.0


def slsqp_eq_jac(x):
    n = len(x) - 1
    j = np.zeros(len(x))
    j[:n] = 1.0
    return j


def slsqp_optimize(h_init, maxiter=5000, t_margin=0.1):
    c5_init, _ = compute_c5(h_init)
    n = len(h_init)
    x0 = np.concatenate([h_init, [c5_init + t_margin]])
    bounds = [(0.0, 1.0)] * n + [(max(0.001, c5_init * 0.5), 2.0)]
    constraints = [
        {'type': 'ineq', 'fun': slsqp_ineq},
        {'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac}
    ]
    result = minimize(slsqp_obj, x0, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
    if result.x is not None:
        h_out = project_feasible(result.x[:-1], n)
        return h_out, compute_c5(h_out)[0]
    return h_init, c5_init


def lbfgsb_analytical(h_init, T, penalty, maxiter=5000):
    n = len(h_init)
    bounds = [(0.0, 1.0)] * n
    result = minimize(smooth_obj_grad, h_init, args=(T, penalty),
                      method='L-BFGS-B', bounds=bounds, jac=True,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
    if result.x is not None:
        h_out = project_feasible(result.x, n)
        return h_out, compute_c5(h_out)[0]
    return h_init, compute_c5(h_init)[0]


def make_structured_init(n, rng):
    r = rng.random()
    if r < 0.25:
        h = 0.5 + 0.3 * np.sin(np.pi * np.arange(n) / n + rng.random() * 2 * np.pi)
    elif r < 0.5:
        h = 0.5 + 0.25 * np.cos(2 * np.pi * np.arange(n) / n + rng.random() * 2 * np.pi)
    elif r < 0.7:
        cut1 = max(1, int(n * rng.uniform(0.2, 0.4)))
        cut2 = min(n - 1, int(n * rng.uniform(0.55, 0.75)))
        h = np.ones(n) * 0.3
        h[:cut1] = 0.7
        h[cut2:] = 0.7
    else:
        h = 0.5 + 0.2 * rng.standard_normal(n)
    return project_feasible(h, n)


def targeted_perturbation(h, c_vals, scale, rng):
    n = len(h)
    top_k = min(5, len(c_vals))
    idx = np.argsort(c_vals)[-top_k:]
    h_pert = h.copy()
    for idx_k in idx:
        k = idx_k - (n - 1)
        if k >= 0 and k < n:
            pert = np.zeros(n)
            for i in range(n - k):
                pert[i + k] += (1.0 - h[i]) * scale
                pert[i] -= h[i + k] * scale
        elif k < 0:
            m = -k
            if m < n:
                pert = np.zeros(n)
                for i in range(n - m):
                    pert[i] += (1.0 - h[i + m]) * scale
                    pert[i + m] -= h[i] * scale
        else:
            continue
        h_pert += 0.5 * pert
    return project_feasible(h_pert, n)


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(135, 0.5)

    n = len(h0)
    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)[0]
    n_best = n

    if time.time() - t0 < budget_s - 500:
        best_h, best_c5 = slsqp_optimize(best_h, maxiter=5000, t_margin=0.1)
        n_best = len(best_h)

    for T_val in [15, 40, 80, 150, 300, 600]:
        if time.time() - t0 > budget_s - 350:
            break
        h_new, c5_new = lbfgsb_analytical(best_h, T=T_val, penalty=300.0, maxiter=4000)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    if time.time() - t0 < budget_s - 250:
        best_h, best_c5 = slsqp_optimize(best_h, maxiter=6000, t_margin=0.05)

    elapsed = time.time() - t0
    remaining = budget_s - elapsed - 30
    n_perturb = int(min(80, remaining / 8))
    if n_perturb > 5 and time.time() - t0 < budget_s - 100:
        for attempt in range(n_perturb):
            if time.time() - t0 > budget_s - 30:
                break
            frac = attempt / max(1, n_perturb)
            scale = 0.08 * (1.0 - frac) ** 0.7 + 0.005
            c_vals = np.correlate(best_h, 1.0 - best_h, mode="full") * (2.0 / n_best)

            r = rng.random()
            if r < 0.3:
                h_pert = targeted_perturbation(best_h, c_vals, scale * 2, rng)
            elif r < 0.5:
                h_pert = best_h + scale * rng.standard_normal(n_best)
                h_pert = project_feasible(h_pert, n_best)
            elif r < 0.65:
                h_pert = best_h.copy()
                n_push = max(5, int(n_best * 0.2))
                push_idx = rng.choice(n_best, n_push, replace=False)
                for idx in push_idx:
                    if rng.random() < 0.5:
                        h_pert[idx] = min(1.0, h_pert[idx] + scale * 4)
                    else:
                        h_pert[idx] = max(0.0, h_pert[idx] - scale * 4)
                h_pert = project_feasible(h_pert, n_best)
            elif r < 0.8:
                n_modes = rng.integers(1, 3)
                h_pert = best_h.copy()
                for _ in range(n_modes):
                    freq = rng.integers(1, max(2, n_best // 4))
                    phase = rng.random() * 2 * np.pi
                    h_pert += (scale / np.sqrt(n_modes)) * np.cos(
                        2 * np.pi * freq * np.arange(n_best) / n_best + phase)
                h_pert = project_feasible(h_pert, n_best)
            else:
                h_pert = make_structured_init(n_best, rng)
                h_pert = project_feasible(h_pert, n_best)
                if np.sum(np.abs(h_pert - best_h)) > 5.0:
                    h_pert = 0.5 * best_h + 0.5 * h_pert
                    h_pert = project_feasible(h_pert, n_best)

            if attempt % 4 == 0:
                h_new, c5_new = lbfgsb_analytical(h_pert, T=50.0, penalty=300.0, maxiter=2000)
                if c5_new < best_c5 - 1e-12:
                    best_h, best_c5 = h_new, c5_new
                if time.time() - t0 < budget_s - 40:
                    h_new2, c5_new2 = slsqp_optimize(best_h, maxiter=3000, t_margin=0.03)
                    if c5_new2 < best_c5 - 1e-12:
                        best_h, best_c5 = h_new2, c5_new2
            else:
                h_new, c5_new = lbfgsb_analytical(h_pert, T=30.0, penalty=200.0, maxiter=1500)
                if c5_new < best_c5 - 1e-12:
                    best_h, best_c5 = h_new, c5_new

    for n_alt in [300, 360, 400, 480]:
        if time.time() - t0 > budget_s - 120:
            break
        x_old = np.linspace(0, 2, n_best, endpoint=False)
        x_new = np.linspace(0, 2, n_alt, endpoint=False)
        f_interp = interp1d(x_old, best_h, kind='linear', fill_value="extrapolate")
        h_interp = np.clip(f_interp(x_new), 0, 1)
        h_interp = project_feasible(h_interp, n_alt)
        h_new, c5_new = lbfgsb_analytical(h_interp, T=30.0, penalty=200.0, maxiter=3000)
        if c5_new < best_c5 - 1e-10:
            best_h, best_c5 = h_new, c5_new
            n_best = n_alt
        if time.time() - t0 < budget_s - 60:
            h_new2, c5_new2 = slsqp_optimize(best_h, maxiter=4000, t_margin=0.05)
            if c5_new2 < best_c5 - 1e-12:
                best_h, best_c5 = h_new2, c5_new2

    if time.time() - t0 < budget_s - 15:
        n_final = n_best
        for T_val in [200, 500]:
            if time.time() - t0 > budget_s - 8:

    if time.time() - t0 < budget_s - 5:
        best_h, best_c5 = slsqp_optimize(best_h, maxiter=8000, t_margin=0.02)

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)[0]
    return best_h, best_c5, len(best_h)
```
