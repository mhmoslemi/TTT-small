def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m_val))
    s_exp = np.sum(exp_vals)
    w = exp_vals / s_exp
    obj = m_val + np.log(s_exp) / alpha
    
    j_idx = np.arange(n)[:, None]
    k_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx + k_idx - (n - 1)
    second_idx = j_idx - k_idx + (n - 1)
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    grad = np.sum(valid_f * (1.0 - h[fc]) * w[None, :], axis=1) - \
           np.sum(valid_s * h[sc] * w[None, :], axis=1)
    return obj, grad

        return smooth_max_and_grad(h, alpha)[0]
    def g(h):
        return smooth_max_and_grad(h, alpha)[1]

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})

    except Exception:

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=np.zeros(2 * n - 1),
                          A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
            if res.success:
                h_a = project_feasible(res.x[:n], target_sum)
                c5_a = compute_c5(h_a)
                if c5_a < best_c5 - 1e-12:
                    best_c5 = c5_a
                    h = h_a
                    improved = True
        except Exception:
            pass
        
        A_ub = build_lp_matrices_f(h, n)
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=np.zeros(2 * n - 1),
                          A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
            if res.success:
                f_sol = project_feasible(res.x[:n], target_sum)
                h_b = project_feasible(1.0 - f_sol, target_sum)
                c5_b = compute_c5(h_b)
                if c5_b < best_c5 - 1e-12:
                    best_c5 = c5_b
                    h = h_b
                    improved = True
        except Exception:
            pass
        
        if not improved:
            break
    
    return h, best_c5

def binary_flip_search(h, n, target_sum, rng, max_rounds=100, n_trials=300):
    h = project_feasible((h > 0.5).astype(float).copy(), target_sum)
    c5_cur = compute_c5(h)
    for _ in range(max_rounds):
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_c5 = c5_cur
        best_pair = None
        n_t = min(n_trials, len(ones) * len(zeros))
        for _ in range(n_t):
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h_test = h.copy()
            h_test[i] = 1.0
            h_test[j] = 0.0
            c5_test = compute_c5(h_test)
            if c5_test < best_c5 - 1e-10:
                best_c5 = c5_test
                best_pair = (i, j)
        if best_pair is None:
            break
        i, j = best_pair
        h[i] = 1.0
        h[j] = 0.0
        c5_cur = best_c5
    return h, c5_cur

def multi_pair_flip(h, n, target_sum, rng, max_rounds=50):
    h = project_feasible((h > 0.5).astype(float).copy(), target_sum)
    c5_cur = compute_c5(h)
    for _ in range(max_rounds):
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        best_c5 = c5_cur
        best_pairs = None
        for _ in range(200):
            ii = rng.choice(ones, size=2, replace=False)
            jj = rng.choice(zeros, size=2, replace=False)
            h_test = h.copy()
            h_test[ii] = 1.0
            h_test[jj] = 0.0
            c5_test = compute_c5(h_test)
            if c5_test < best_c5 - 1e-10:
                best_c5 = c5_test
                best_pairs = (ii, jj)
        if best_pairs is None:
            break
        h[best_pairs[0]] = 1.0
        h[best_pairs[1]] = 0.0
        c5_cur = best_c5
    return h, c5_cur

def make_start_variants(h_best, n_new, target_sum, rng):
    starts = []
    if h_best is not None and len(h_best) > 0:
        x_old = np.linspace(0, 2, len(h_best), endpoint=False)
        x_new = np.linspace(0, 2, n_new, endpoint=False)
        h_interp = np.interp(x_new, x_old, h_best, period=2.0)
        h_interp = project_feasible(h_interp, target_sum)
        starts.append(h_interp)
        noise = 0.1
        h_noise = project_feasible(h_interp + rng.uniform(-noise, noise, n_new), target_sum)
        starts.append(h_noise)
    starts.append(np.full(n_new, 0.5))
    x = np.linspace(0, 2, n_new, endpoint=False)
    for freq in [1, 2, 3, 4, 5, 6, 8]:
        for amp in [0.3, 0.45]:
            s = 0.5 + amp * np.sin(2.0 * np.pi * freq * x / 2.0)
            starts.append(project_feasible(s, target_sum))
    for nblk in [2, 3, 4, 5, 6, 8, 10, 12]:
        blk = max(1, n_new // nblk)
        h_blk = np.zeros(n_new)
        for i in range(0, n_new, 2 * blk):
            h_blk[i:min(i + blk, n_new)] = 1.0
        starts.append(project_feasible(h_blk, target_sum))
    return starts

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    
    def time_left():
        return budget_s - (time.time() - t_start)
    
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_feasible(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass
    
    if best_h is None:
        best_n = 180
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)
    
    # Phase 1: Alternating LP on current resolution
    if time_left() > 60:
        h_lp, c5_lp = alternating_lp(best_h, best_n, best_n / 2.0, 
                                       max_rounds=50, budget_check=time_left)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp.copy()
    
    # Phase 2: SLSQP continuation on best solution
    n1 = best_n
    ts1 = n1 / 2.0
    for alpha in [5, 15, 40, 100, 300, 1000, 3000]:
        if time_left() < 30:
            break
        h_new = slsqp_optimize(best_h, n1, ts1, alpha, maxiter=1000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_n = n1
    
    # Phase 3: Binary and multi-pair flip search
    if time_left() > 60:
        n_b = best_n
        ts_b = n_b / 2.0
        h_bin = (best_h > 0.5).astype(float)
        attempts = 0
        while abs(h_bin.sum() - ts_b) > 1e-10 and attempts < 10000:
            if h_bin.sum() > ts_b:
                idx = rng.choice(np.where(h_bin > 0)[0])
                h_bin[idx] = 0.0
            else:
                idx = rng.choice(np.where(h_bin < 1)[0])
                h_bin[idx] = 1.0
            attempts += 1
        if abs(h_bin.sum() - ts_b) < 1e-10:
            h_bf, c5_bf = binary_flip_search(h_bin, n_b, ts_b, rng, max_rounds=80, n_trials=400)
            if c5_bf < best_c5:
                best_c5 = c5_bf
                best_h = h_bf.copy()
            h_mp, c5_mp = multi_pair_flip(h_bf, n_b, ts_b, rng, max_rounds=40)
            if c5_mp < best_c5:
                best_c5 = c5_mp
                best_h = h_mp.copy()
        # Refine the best binary solution with SLSQP
        if time_left() > 30:
            h_ref = slsqp_optimize(best_h, n_b, ts_b, 200, maxiter=1000)
            c5_ref = compute_c5(h_ref)
            if c5_ref < best_c5:
                best_c5 = c5_ref
                best_h = h_ref.copy()
    
    # Phase 4: Multi-resolution exploration
    resolutions = [40, 60, 80, 100, 120, 144, 160, 200, 240, 280, 320, 360]
    for n_tr in resolutions:
        if time_left() < 80:
            break
        ts_tr = n_tr / 2.0
        if abs(ts_tr - round(ts_tr)) > 1e-6:
            continue
        starts = make_start_variants(best_h, n_tr, ts_tr, rng)
        for h_start in starts:
            if time_left() < 30:
                break
            h_start = project_feasible(h_start, ts_tr)
            c5_s = compute_c5(h_start)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_start.copy()
                best_n = n_tr
            # LP on this resolution
            if time_left() > 40 and n_tr <= 200:
                h_lp2, c5_lp2 = alternating_lp(h_start, n_tr, ts_tr,
                                                max_rounds=20, budget_check=time_left)
                if c5_lp2 < best_c5:
                    best_c5 = c5_lp2
                    best_h = h_lp2.copy()
                    best_n = n_tr
                h_start = h_lp2
            # SLSQP refinement
            for alpha in [20, 100, 500]:
                if time_left() < 20:
                    break
                h_opt = slsqp_optimize(h_start, n_tr, ts_tr, alpha, maxiter=800)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    best_n = n_tr
                h_start = h_opt
    
    # Phase 5: Shake and re-optimize at best resolution
    n_s = best_n
    ts_s = n_s / 2.0
    for trial in range(20):
        if time_left() < 30:
            break
        noise = 0.12 * (1.0 - trial / 20.0) + 0.02
        h_shake = project_feasible(best_h + rng.uniform(-noise, noise, n_s), ts_s)
        if time_left() > 20 and n_s <= 200:
            h_lp3, c5_lp3 = alternating_lp(h_shake, n_s, ts_s,
                                            max_rounds=15, budget_check=time_left)
            h_shake = h_lp3
        for alpha in [50, 300]:
            if time_left() < 15:
                break
            h_opt = slsqp_optimize(h_shake, n_s, ts_s, alpha, maxiter=600)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_s
            h_shake = h_opt
    
    # Phase 6: Final high-alpha polish
    if time_left() > 30:
        n_f = best_n
        ts_f = n_f / 2.0
        for alpha in [2000, 8000, 30000]:
            if time_left() < 20:
                break
            h_pol = slsqp_optimize(best_h, n_f, ts_f, alpha, maxiter=2000)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()
    
    # Phase 7: Final binary flip on the best
    if time_left() > 40:
        n_f = best_n
        ts_f = n_f / 2.0
        h_bin2 = (best_h > 0.5).astype(float)
        attempts = 0
        while abs(h_bin2.sum() - ts_f) > 1e-10 and attempts < 10000:
            if h_bin2.sum() > ts_f:
                idx = rng.choice(np.where(h_bin2 > 0)[0])
                h_bin2[idx] = 0.0
            else:
                idx = rng.choice(np.where(h_bin2 < 1)[0])
                h_bin2[idx] = 1.0
            attempts += 1
        if abs(h_bin2.sum() - ts_f) < 1e-10:
            h_bf2, c5_bf2 = binary_flip_search(h_bin2, n_f, ts_f, rng, max_rounds=60, n_trials=500)
            if c5_bf2 < best_c5:
                best_c5 = c5_bf2
                best_h = h_bf2.copy()
            if time_left() > 20:
                h_r = slsqp_optimize(best_h, n_f, ts_f, 500, maxiter=800)
                c5_r = compute_c5(h_r)
                if c5_r < best_c5:
                    best_c5 = c5_r
                    best_h = h_r.copy()
    
    # Final validation
    best_h = project_feasible(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
