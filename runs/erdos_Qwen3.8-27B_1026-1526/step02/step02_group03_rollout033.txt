    phase1_end = start_time + budget_s * 0.35
    phase2_end = start_time + budget_s * 0.70
    phase3_end = start_time + budget_s * 0.88
    phase4_end = start_time + budget_s * 0.98

    n_list = [n_parent]
    for nn in [n_parent-1, n_parent+1, n_parent+2, n_parent-2, n_parent+4, n_parent-4,
               80, 84, 90, 96, 100, 104, 108, 112, 120, 128, 144, 160]:
        if nn >= 60 and nn not in n_list:
            n_list.append(nn)

    # Phase 1: Multi-start optimization
    for n in n_list:
        if time.time() > phase1_end:
            break
        starts = generate_starts(n, parent_h, rng)
        num_starts = min(len(starts), 12)
        for si in range(num_starts):
            if time.time() > phase1_end:
                break
            h0 = project_feasible(starts[si], n)
            if n <= 90:
                tau_sched = [0.12, 0.06, 0.03, 0.015, 0.008, 0.004]
                h_opt, c5 = solve_phase(h0, n, tau_sched, 80, "SLSQP", 0.0)
            else:
                tau_sched = [0.15, 0.08, 0.04, 0.02, 0.01]
                h_opt, c5 = solve_phase(h0, n, tau_sched, 60, "L-BFGS-B", 500.0)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()

    # Phase 2: Intensive local refinement
    if best_h is not None and time.time() < phase2_end:
        n = len(best_h)
        attempts = 0
        while time.time() < phase2_end and attempts < 600:
            attempts += 1
            frac = (time.time() - start_time) / budget_s
            noise_scale = 0.003 + 0.02 * max(0.0, 1.0 - frac * 2.0)
            perturb = project_feasible(best_h + rng.randn(n) * noise_scale, n)
            tau_sched = [0.008, 0.004, 0.002, 0.001]
            if n <= 90:
                h_opt, c5 = solve_phase(perturb, n, tau_sched, 80, "SLSQP", 0.0)
            else:
                h_opt, c5 = solve_phase(perturb, n, tau_sched, 60, "L-BFGS-B", 1000.0)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()

    # Phase 3: n-value exploration
    if best_h is not None and time.time() < phase3_end:
        n_best = len(best_h)
        xp = np.linspace(0, 2, n_best, endpoint=False)
        n_tries = [n_best-1, n_best+1, n_best+2, n_best-2,
                   n_best+3, n_best-3, n_best+4, n_best-4,
                   n_best+6, n_best+8, n_best-6]
        for n_try in n_tries:
            if time.time() > phase3_end or n_try < 60:
                continue
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            f_interp = interp1d(xp, best_h, kind='cubic', fill_value='extrapolate', bounds_error=False)
            h_init = np.clip(f_interp(x_new), 0, 1)
            h_init = project_feasible(h_init, n_try)
            if n_try <= 100:
                tau_sched = [0.04, 0.02, 0.01, 0.005, 0.002, 0.001]
                h_opt, c5 = solve_phase(h_init, n_try, tau_sched, 120, "SLSQP", 0.0)
            else:
                tau_sched = [0.05, 0.025, 0.01, 0.005]
                h_opt, c5 = solve_phase(h_init, n_try, tau_sched, 80, "L-BFGS-B", 2000.0)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()

    # Phase 4: Final polishing with coordinate descent + tiny SLSQP
    if best_h is not None and time.time() < phase4_end:
        n = len(best_h)
        h_coord, c5_coord = coordinate_descent(best_h, n, 200, rng, phase4_end - 5.0)
        if c5_coord < best_c5:
            best_c5 = c5_coord
            best_h = h_coord.copy()

        if time.time() < phase4_end - 2.0:
            for _ in range(20):
                if time.time() > phase4_end - 2.0:
                    break
                perturb = project_feasible(best_h + rng.randn(n) * 0.0005, n)
                tau_sched = [0.001, 0.0005]
                h_opt, c5 = solve_phase(perturb, n, tau_sched, 150, "SLSQP", 0.0)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt.copy()

    if best_h is not None and time.time() < start_time + budget_s * 0.99:
        n = len(best_h)
        h_coord, c5_coord = coordinate_descent(best_h, n, 100, rng, start_time + budget_s * 0.99)
        if c5_coord < best_c5:
            best_c5 = c5_coord
            best_h = h_coord.copy()

    if best_h is None:
        if parent_h is not None:
            best_h = parent_h.copy()
        else:
            best_h = project_feasible(np.full(77, 0.5), 77)
        best_c5 = evaluate_c5(best_h)

    final_c5 = evaluate_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
