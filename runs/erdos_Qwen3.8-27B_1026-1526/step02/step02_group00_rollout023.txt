def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

    n = len(h)
    return np.sum(h * (1.0 - np.roll(h, -k)))

def cyclic_overlaps(h):
    n = len(h)
    return np.array([cyclic_overlap(h, k) for k in range(n)])

    n = len(h)
    overlaps = cyclic_overlaps(h)
    k_star = np.argmax(overlaps)
    grad = (1.0 - np.roll(h, -k_star)) - np.roll(h, k_star)
    return float(np.max(overlaps)), grad, k_star

    n = len(h)
    overlaps = cyclic_overlaps(h)
    m_val = np.max(overlaps)
    exp_vals = np.exp(alpha * (overlaps - m_val))
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    grad = np.zeros(n)
    for k in range(n):
        if w[k] > 1e-10:
            grad += w[k] * ((1.0 - np.roll(h, -k)) - np.roll(h, k))
    return obj, grad

    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h


def cyclic_overlap_vec(h):
    n = len(h)
    hF = np.fft.fft(h)
    R = np.real(np.fft.ifft(hF * np.conj(hF)))
    return np.sum(h) - R


def smooth_max_cyclic(h, alpha):
    n = len(h)
    ov = cyclic_overlap_vec(h)
    m = np.max(ov)
    ev = np.exp(alpha * (ov - m))
    sw = np.sum(ev)
    w = ev / sw
    obj = m + np.log(sw) / alpha
    grad = np.zeros(n)
    for k in range(n):
        wk = w[k]
        if wk > 1e-10:
            grad += wk * ((1.0 - np.roll(h, -k)) - np.roll(h, k))
    return obj, grad


def _eq_fun(x):
    return np.sum(x) - x._target


def _eq_jac(x):
    return np.ones(len(x))


def slsqp_opt(h0, n, target_sum, alpha, maxiter=500):
    def obj_func(h):
        return smooth_max_cyclic(h, alpha)[0]

    def grad_func(h):
        return smooth_max_cyclic(h, alpha)[1]

    def cons_fun(x):
        return np.sum(x) - target_sum

    def cons_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_func, h0.copy(), method='SLSQP', jac=grad_func,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)


def subgrad_opt(h0, n, target_sum, max_iters=5000, alpha0=0.005, alpha_decay=0.9999, momentum=0.85):
    h = h0.copy()
    v = np.zeros(n)
    alpha = alpha0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    stall_count = 0
    stall_limit = 500

    for it in range(max_iters):
        ov = cyclic_overlap_vec(h)
        k_star = np.argmax(ov)
        grad = (1.0 - np.roll(h, -k_star)) - np.roll(h, k_star)

        v = momentum * v - alpha * grad
        h_new = h + v
        h_new = project_to_sum(h_new, target_sum)

        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > stall_limit:
            alpha = max(alpha * 0.5, 1e-6)
            v = np.zeros(n)
            stall_count = 0
            h = best_h.copy()

        h = h_new
        alpha *= alpha_decay

    return best_h, best_c5


def to_binary(h):
    n = len(h)
    target = n // 2
    h_bin = (h > 0.5).astype(float)
    while h_bin.sum() > target:
        ones = np.where(h_bin == 1)[0]
        idx = ones[np.argmin(h[ones])]
        h_bin[idx] = 0.0
    while h_bin.sum() < target:
        zeros = np.where(h_bin == 0)[0]
        idx = zeros[np.argmax(h[zeros])]
        h_bin[idx] = 1.0
    return h_bin


def local_binary_improve(h_bin, target_sum, rng, max_rounds=500, n_trials=300):
    n = len(h_bin)
    h = h_bin.copy()
    c5_cur = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_cur
    no_improve = 0

    for rnd in range(max_rounds):
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        best_c5_move = c5_cur
        best_move = None
        nt = min(n_trials, len(ones) * len(zeros))

        for _ in range(nt):
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h[i] = 0.0
            h[j] = 1.0
            c5_test = compute_c5(h)
            if c5_test < best_c5_move - 1e-13:
                best_c5_move = c5_test
                best_move = (int(i), int(j))
            h[i] = 1.0
            h[j] = 0.0

        if best_move is None:
            no_improve += 1
            if no_improve > 5:
                break
            continue

        i, j = best_move
        h[i] = 0.0
        h[j] = 1.0
        c5_cur = best_c5_move
        no_improve = 0

        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h.copy()

    return best_h, best_c5


def make_starts(n, target_sum, rng, best_h, best_n):
    starts = []
    starts.append(np.full(n, 0.5))
    starts.append(project_to_sum(rng.uniform(0, 1, n), target_sum))

    x = np.linspace(0, 2, n, endpoint=False)
    for freq in [1, 2, 3, 4, 5, 7]:
        s = 0.5 + 0.4 * np.sin(2.0 * np.pi * freq * x / 2.0)
        starts.append(project_to_sum(s, target_sum))

    for nblks in [2, 3, 4, 5, 6, 8]:
        blk = max(1, n // nblks)
        h_blk = np.zeros(n)
        for i in range(0, n, 2 * blk):
            h_blk[i:min(i + blk, n)] = 1.0
        starts.append(project_to_sum(h_blk, target_sum))

    for _ in range(4):
        a = rng.uniform(0.1, 0.45)
        f1 = rng.choice([1, 2, 3, 4])
        p1 = rng.uniform(0, 2 * np.pi)
        f2 = rng.choice([2, 3, 5, 7, 9])
        p2 = rng.uniform(0, 2 * np.pi)
        h_par = 0.5 + a * np.sin(2.0 * np.pi * f1 * x / 2.0 + p1) + 0.2 * np.sin(2.0 * np.pi * f2 * x / 2.0 + p2)
        starts.append(project_to_sum(h_par, target_sum))

    if best_h is not None and best_n > 0:
        if best_n == n:
            starts.append(best_h.copy())
        else:
            x_old = np.linspace(0, 2, best_n, endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            starts.append(project_to_sum(np.interp(x_new, x_old, best_h, period=2.0), target_sum))

    hb = (rng.rand(n) > 0.5).astype(float)
    attempts = 0
    while abs(hb.sum() - target_sum) > 1e-10 and attempts < 10000:
        if hb.sum() > target_sum:
            idx = rng.choice(np.where(hb > 0)[0])
            hb[idx] = 0.0
        else:
            idx = rng.choice(np.where(hb < 1)[0])
            hb[idx] = 1.0
        attempts += 1
    if abs(hb.sum() - target_sum) < 1e-10:
        starts.append(hb.copy())

    return starts


def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()

    def time_left():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    # Phase 1: Subgradient on initial solution
    n1 = best_n
    ts1 = n1 / 2.0
    if time_left() > 80:
        h_sg, c5_sg = subgrad_opt(best_h, n1, ts1, max_iters=8000, alpha0=0.004, alpha_decay=0.99997, momentum=0.9)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # Phase 2: SLSQP continuation on best solution
    if time_left() > 80:
        for alpha in [10, 30, 100, 300, 1000]:
            if time_left() < 30:
                break
            h_opt = slsqp_opt(best_h, best_n, best_n / 2.0, alpha, maxiter=1000)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Phase 3: Multi-resolution multi-start
    resolutions = [240, 120, 60, 180, 90, 300, 360, 400]
    for n_tr in resolutions:
        if time_left() < 80:
            break
        if n_tr % 2 != 0:
            continue
        ts_tr = n_tr / 2.0
        starts = make_starts(n_tr, ts_tr, rng, best_h, best_n)
        for h_start in starts:
            if time_left() < 30:
                break
            h_start = project_to_sum(h_start, ts_tr)
            c5_s = compute_c5(h_start)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_start.copy()
                best_n = n_tr

            for alpha in [50, 200, 1000]:
                if time_left() < 20:
                    break
                h_opt = slsqp_opt(h_start, n_tr, ts_tr, alpha, maxiter=600)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    best_n = n_tr
                h_start = h_opt

            if time_left() > 40:
                h_sg2, c5_sg2 = subgrad_opt(h_start, n_tr, ts_tr, max_iters=4000, alpha0=0.003, alpha_decay=0.99995, momentum=0.88)
                if c5_sg2 < best_c5:
                    best_c5 = c5_sg2
                    best_h = h_sg2.copy()
                    best_n = n_tr

    # Phase 4: Binary optimization
    if time_left() > 80:
        h_bin = to_binary(best_h)
        h_bin, c5_bin = local_binary_improve(h_bin, best_n / 2.0, rng, max_rounds=400, n_trials=400)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()

        if time_left() > 40:
            h_smooth = slsqp_opt(h_bin, best_n, best_n / 2.0, 300, maxiter=1500)
            c5_sm = compute_c5(h_smooth)
            if c5_sm < best_c5:
                best_c5 = c5_sm
                best_h = h_smooth.copy()

    # Phase 5: Shake and re-optimize
    n_best = best_n
    ts_best = n_best / 2.0
    for trial in range(25):
        if time_left() < 30:
            break
        noise = 0.12 * (1.0 - trial / 25.0) + 0.02
        h_shake = best_h + rng.uniform(-noise, noise, n_best)
        h_shake = project_to_sum(h_shake, ts_best)
        for alpha in [50, 300]:
            if time_left() < 20:
                break
            h_opt = slsqp_opt(h_shake, n_best, ts_best, alpha, maxiter=700)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
            h_shake = h_opt

    # Phase 6: Final polish
    if time_left() > 50:
        for alpha in [2000, 10000]:
            if time_left() < 25:
                break
            h_pol = slsqp_opt(best_h, best_n, best_n / 2.0, alpha, maxiter=2000)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

        if time_left() > 30:
            h_sg3, c5_sg3 = subgrad_opt(best_h, best_n, best_n / 2.0, max_iters=5000, alpha0=0.001, alpha_decay=0.99999, momentum=0.92)
            if c5_sg3 < best_c5:
                best_c5 = c5_sg3
                best_h = h_sg3.copy()

    # Phase 7: One more binary pass at the end
    if time_left() > 40:
        h_bin2 = to_binary(best_h)
        h_bin2, c5_bin2 = local_binary_improve(h_bin2, best_n / 2.0, rng, max_rounds=300, n_trials=500)
        if c5_bin2 < best_c5:
            best_c5 = c5_bin2
            best_h = h_bin2.copy()

    # Final validation
    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
