    return float(np.max(corr) * dx)

def get_overlap_values(h):

    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return corr * dx

def project_feasible(h):
    n = len(h)
    target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target_sum - np.sum(result)
    if abs(diff) > 1e-10:
        idx_order = np.argsort(np.abs(result - 0.5))
        for idx in idx_order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                add = min(1.0 - result[idx], diff)
                result[idx] += add
                diff -= add
            else:
                rem = min(result[idx], -diff)
                result[idx] -= rem
                diff += rem
    return result

def smooth_max(values, tau):
    v = values / tau
    vmax = np.max(v)
    return tau * (vmax + np.log(np.sum(np.exp(v - vmax))))

def compute_gradient_for_lag(h, lag_idx):
    n = len(h)
    s = lag_idx - (n - 1)
    grad = np.zeros(n)
    dx = 2.0 / n
    for j in range(n):
        jp = j + s
        if 0 <= jp < n:
            grad[j] += (1.0 - h[jp])
        jm = j - s
        if 0 <= jm < n:
            grad[j] -= h[jm]
    return grad * dx

def smooth_max_obj(x, tau):
    overlaps = get_overlap_values(x)
    return smooth_max(overlaps, tau)

    overlaps = get_overlap_values(x)
    sm = smooth_max(overlaps, tau)
    s = np.sum(x) - len(x) / 2.0
    return sm + penalty * s * s

def sum_constraint(x):

    n = len(h)
    overlaps = get_overlap_values(h)
    top_lags = np.argsort(overlaps)[-n_active:]
    grad = np.zeros(n)
    for lag in top_lags:
        grad += compute_gradient_for_lag(h, lag)
    grad /= n_active
    h_new = h - step_size * grad
    return project_feasible(h_new)

                            if p_c5 < best_c5:
                                best_c5 = p_c5
                                best_h = current.copy()
                        else:
                            break

            h_cd, cd_c5 = targeted_coordinate_descent(current, cd_steps_med, max_sweeps=1)
            if cd_c5 < current_c5:
                current = h_cd
                current_c5 = cd_c5
                if cd_c5 < best_c5:
                    best_c5 = cd_c5
                    best_h = current.copy()

    if best_h is not None and time.time() < phase3_end:
        fine_cd_steps = [0.01, 0.006, 0.004, 0.003, 0.002, 0.001]
        fine_pair_steps = [0.01, 0.006, 0.004, 0.002, 0.001]
        sg_steps_fine = [0.004, 0.002, 0.001, 0.0005, 0.0003]

        stagnant = 0
        while time.time() < phase3_end:
            h_sub = None
            for step in sg_steps_fine:
                h_try = active_subgradient_step(best_h, step, n_active=1)
                c5_try = evaluate_c5(h_try)
                if c5_try < best_c5:
                    h_sub = h_try
                    break
            if h_sub is not None:
                best_h = h_sub
                best_c5 = evaluate_c5(best_h)
                stagnant = 0
            else:
                h_cd, cd_c5 = targeted_coordinate_descent(best_h, fine_cd_steps, max_sweeps=2)
                if cd_c5 < best_c5:
                    best_h = h_cd
                    best_c5 = cd_c5
                    stagnant = 0
                else:
                    h_pair, pair_c5 = pair_coordinate_descent(best_h, fine_pair_steps, max_sweeps=2)
                    if pair_c5 < best_c5:
                        best_h = h_pair
                        best_c5 = pair_c5
                        stagnant = 0
                    else:
                        stagnant += 1
                        if stagnant > 5:
                            break

    phase4_end = start_time + budget_s * 0.98
    if best_h is not None and time.time() < phase4_end:
        ultra_steps = [0.004, 0.003, 0.002, 0.001, 0.0006, 0.0004, 0.0002]
        while time.time() < phase4_end:
            improved = False
            n = len(best_h)
            overlaps = get_overlap_values(best_h)
            worst_lag = int(np.argmax(overlaps))
            grad_worst = compute_gradient_for_lag(best_h, worst_lag)

            for i in range(n):
                if time.time() > phase4_end:
                    break
                if grad_worst[i] > 0:
                    dirs = [-1, 1]
                else:
                    dirs = [1, -1]
                for d in dirs:
                    found = False
                    for scale in ultra_steps:
                        h_new = best_h.copy()
                        h_new[i] = np.clip(h_new[i] + d * scale, 0.0, 1.0)
                        h_new = project_feasible(h_new)
                        c5_new = evaluate_c5(h_new)
                        if c5_new < best_c5:
                            best_h = h_new
                            best_c5 = c5_new
                            improved = True
                            found = True
                            break
                    if found:
                        break
            if not improved:
                h_pair, pair_c5 = pair_coordinate_descent(best_h, [0.003, 0.002, 0.001, 0.0005], max_sweeps=1)
                if pair_c5 < best_c5:
                    best_h = h_pair
                    best_c5 = pair_c5
                else:
                    h_sub = active_subgradient_step(best_h, 0.001, n_active=1)
                    s_c5 = evaluate_c5(h_sub)
                    if s_c5 < best_c5:
                        best_h = h_sub
                        best_c5 = s_c5
                    else:
                        break

    if best_h is None:
        best_h = project_feasible(np.full(n, 0.5))
        best_c5 = evaluate_c5(best_h)

    final_c5 = evaluate_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
