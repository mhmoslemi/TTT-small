    return best, best_c5

def coordinate_descent(h, n, max_sweeps=25, rng=None):
    best = h.copy()
    best_c5 = evaluate_c5(best)
    step_sizes = [0.08, 0.04, 0.02, 0.01, 0.005, 0.003, 0.001, 0.0005, 0.0002]
    for sweep in range(max_sweeps):
        improved = False
        si = min(sweep, len(step_sizes) - 1)
        step = step_sizes[si]
        for i in range(n):
            for d in [-step, step, -step * 0.5, step * 0.5]:
                h_new = best.copy()
                h_new[i] += d
                if h_new[i] < -1e-12 or h_new[i] > 1.0 + 1e-12:
                    continue
                h_new = project_feasible(h_new, n)
                c5_new = evaluate_c5(h_new)
                if c5_new < best_c5 - 1e-14:
                    best = h_new
                    best_c5 = c5_new
                    improved = True
                    break
        if not improved and sweep >= len(step_sizes):
            break
    return best, best_c5

def freq_perturb(h, n, rng, scale=0.2):
    H = np.fft.rfft(h)
    phases = rng.randn(len(H)) * scale
    H_pert = H * np.exp(1j * phases)
    h_pert = np.fft.irfft(H_pert, n=n)
    return project_feasible(h_pert, n)

def generate_structured(n, rng):
    x = np.linspace(0, 2, n, endpoint=False)
    ar = np.arange(n, dtype=float)
    f1 = rng.choice([1.0, 1.3, 1.5, 1.7, 2.0, 2.3, 2.5, 2.7, 3.0, 3.5])
    f2 = rng.choice([0.0, 0.7, 1.0, 1.5, 2.0, 2.5, 3.0])
    a1 = rng.uniform(0.25, 0.45)
    a2 = rng.uniform(0.03, 0.18)
    p1 = rng.uniform(0, 2 * np.pi)
    p2 = rng.uniform(0, 2 * np.pi)
    h = 0.5 + a1 * np.sin(2 * np.pi * f1 * x / 2.0 + p1) + a2 * np.sin(2 * np.pi * f2 * x / 2.0 + p2)
    if rng.rand() < 0.3:
        a3 = rng.uniform(0.02, 0.08)
        f3 = rng.choice([4.0, 5.0, 6.0, 7.0])
        p3 = rng.uniform(0, 2 * np.pi)
        h += a3 * np.sin(2 * np.pi * f3 * x / 2.0 + p3)
    h = np.clip(h, 0.0, 1.0)
    return project_feasible(h, n)

def multi_start_explore(n, rng, best_h, best_c5, time_limit):
    candidates = [(best_c5, best_h.copy())]
    while time.time() < time_limit:
        c = rng.randint(4)
        if c == 0:
            h_new = freq_perturb(best_h, n, rng, scale=rng.uniform(0.08, 0.35))
        elif c == 1:
            h_new = generate_structured(n, rng)
        elif c == 2:
            scale = rng.uniform(0.01, 0.08)
            h_new = project_feasible(best_h + rng.randn(n) * scale, n)
        else:
            h_new = project_feasible(rng.rand(n), n)
        h_opt, c5 = gradient_local_search(h_new, n, max_iter=80, rng=rng)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.copy()
        candidates.append((c5, h_opt.copy()))
        candidates.sort(key=lambda x: x[0])
        candidates = candidates[:15]
    return best_h, best_c5, candidates

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    start_time = time.time()

    parent_h = None
    try:
        parent_h = np.array(initial_h_values, dtype=float)
    except Exception:
        pass

    if parent_h is not None:
        n = len(parent_h)
        best_h = project_feasible(parent_h.copy(), n)
    else:
        n = 77
        x = np.linspace(0, 2, n, endpoint=False)
        best_h = project_feasible(0.5 + 0.3 * np.sin(2 * np.pi * 2.0 * x / 2.0), n)

    best_c5 = evaluate_c5(best_h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_constraint, 'args': (n,)}]

    phase1_end = start_time + budget_s * 0.18
    phase2_end = start_time + budget_s * 0.55
    phase3_end = start_time + budget_s * 0.85
    phase4_end = start_time + budget_s * 0.97

    current = best_h.copy()
    for tau in [0.08, 0.04, 0.02, 0.01, 0.005, 0.002]:
        if time.time() > phase1_end:
            break
        try:
            res = minimize(slsqp_objective, current, method='SLSQP',
                          args=(n, tau), bounds=bounds, constraints=cons,
                          options={'maxiter': 80, 'ftol': 1e-13})
            current = project_feasible(res.x, n)
            c5 = evaluate_c5(current)
            if c5 < best_c5:
                best_c5 = c5
                best_h = current.copy()
        except Exception:
            pass

    h_gs, c5_gs = gradient_local_search(best_h, n, max_iter=200, rng=rng)
    if c5_gs < best_c5:
        best_c5 = c5_gs
        best_h = h_gs.copy()

    best_h, best_c5, candidates = multi_start_explore(n, rng, best_h, best_c5, phase2_end)

    top_cands = [c[1] for c in candidates[:6]]
    top_cands.insert(0, best_h.copy())
    seen = [best_h.tolist()]

    for h_start in top_cands:
        if time.time() > phase3_end:
            break
        h_opt, c5 = gradient_local_search(h_start, n, max_iter=250, rng=rng)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.copy()
        if time.time() < phase3_end:
            h_cd, c5_cd = coordinate_descent(h_opt, n, max_sweeps=12, rng=rng)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()

    for attempt in range(200):
        if time.time() > phase3_end:
            break
        frac = (time.time() - start_time) / budget_s
        noise_scale = 0.001 + 0.008 * max(0.0, 1.0 - frac * 1.2)
        perturb = project_feasible(best_h + rng.randn(n) * noise_scale, n)
        h_opt, c5 = gradient_local_search(perturb, n, max_iter=100, rng=rng)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.copy()

    if time.time() < phase4_end:
        h_cd, c5_cd = coordinate_descent(best_h, n, max_sweeps=35, rng=rng)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    h_gs2, c5_gs2 = gradient_local_search(best_h, n, max_iter=200, rng=rng)
    if c5_gs2 < best_c5:
        best_c5 = c5_gs2
        best_h = h_gs2.copy()

    if time.time() < start_time + budget_s * 0.99:
        current = best_h.copy()
        for tau in [0.001, 0.0005]:
            if time.time() > start_time + budget_s * 0.99:
                break
            try:
                res = minimize(slsqp_objective, current, method='SLSQP',
                              args=(n, tau), bounds=bounds, constraints=cons,
                              options={'maxiter': 150, 'ftol': 1e-14})
                current = project_feasible(res.x, n)
                c5 = evaluate_c5(current)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = current.copy()
            except Exception:
                pass

    h_cd_final, c5_cd_final = coordinate_descent(best_h, n, max_sweeps=8, rng=rng)
    if c5_cd_final < best_c5:
        best_c5 = c5_cd_final
        best_h = h_cd_final.copy()

    final_c5 = evaluate_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, n
