    # Actually u = j+k-(n-1), and k ranges from n-1-j to n-1+j
    # So u ranges from 0 to 2j... but u must be < n
    # u = j+k-(n-1), k ∈ [n-1-j, n-1+j] ∩ [0, 2n-2]
    # u_low = j + max(0, n-1-j) - (n-1) = j + n-1-j - (n-1) = 0 (if n-1-j ≥ 0, i.e., j ≤ n-1)
    # u_high = j + min(2n-2, n-1+j) - (n-1) = j + n-1+j - (n-1) = 2j (if n-1+j ≤ 2n-2, i.e., j ≤ n-1)
    # So u ranges from 0 to min(2j, n-1)
    
    # Second term: -sum over k from j to j+n-1 of w[k] * h[j-k+(n-1)]
    # Let v = j-k+(n-1), k = j-v+(n-1), v ranges from n-1 to ... 
    # k ∈ [j, j+n-1] ∩ [0, 2n-2]
    # v = j-k+(n-1), so v ranges from j-(j+n-1)+(n-1) = 0 to j-j+(n-1) = n-1
    # Wait: k=j → v = j-j+(n-1) = n-1; k=j+n-1 → v = j-(j+n-1)+(n-1) = 0
    # So v ranges from 0 to n-1 (reversed)
    # k_low = max(0, j) = j (since j ≥ 0)
    # k_high = min(2n-2, j+n-1)
    # v_low = j - k_high + (n-1)
    # v_high = j - k_low + (n-1) = n-1
    
    # This is getting complicated. Let me just use the direct O(n²) loop.
    for j in range(n):
        # First term
        k_lo = max(0, n-1-j)
        k_hi = min(2*n-2, n-1+j)
        if k_hi >= k_lo:
            u = j + np.arange(k_lo, k_hi+1) - (n-1)
            mask = (u >= 0) & (u < n)
            if np.any(mask):
                grad[j] += np.dot(w[k_lo:k_hi+1][mask], 1.0 - h[u[mask]])
        # Second term
        k_lo2 = max(0, j)
        k_hi2 = min(2*n-2, j+n-1)
        if k_hi2 >= k_lo2:
            v = j - np.arange(k_lo2, k_hi2+1) + (n-1)
            mask = (v >= 0) & (v < n)
            if np.any(mask):
                grad[j] -= np.dot(w[k_lo2:k_hi2+1][mask], h[v[mask]])
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-11) & (h < 1.0 - 1e-11)
        if np.any(free):
            h[free] += diff / np.sum(free)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, target_sum):
    n = len(h)
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:int(round(target_sum))]] = 1.0
    return h_bin

def smooth_max_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    w /= np.sum(w)
    grad = np.zeros(n)
    for j in range(n):
        k_lo = max(0, n - 1 - j)
        k_hi = min(2 * n - 2, n - 1 + j)
        if k_hi >= k_lo:
            u = j + np.arange(k_lo, k_hi + 1) - (n - 1)
            mask = (u >= 0) & (u < n)
            if np.any(mask):
                grad[j] += np.dot(w[k_lo:k_hi + 1][mask], 1.0 - h[u[mask]])
        k_lo2 = max(0, j)
        k_hi2 = min(2 * n - 2, j + n - 1)
        if k_hi2 >= k_lo2:
            v = j - np.arange(k_lo2, k_hi2 + 1) + (n - 1)
            mask = (v >= 0) & (v < n)
            if np.any(mask):
                grad[j] -= np.dot(w[k_lo2:k_hi2 + 1][mask], h[v[mask]])
    return grad

def slsqp_opt(h0, n, target_sum, alpha, maxiter):
    def obj(x):
        return smooth_max_obj(x, alpha)
    def jac(x):
        return smooth_max_grad(x, alpha)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), method='SLSQP', jac=jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def binary_swap_search(h, n, max_rounds, rng, n_candidates=6000):
    h = h.copy()
    dx = 2.0 / n
    out = np.correlate(h, 1 - h, mode='full')
    c5_val = np.max(out)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for _ in range(max_rounds):
        best_delta = 1e10
        best_swap = None
        best_out = None
        n_pairs = len(ones) * len(zeros)
        if n_pairs <= n_candidates:
            candidates = [(int(ones[a]), int(zeros[b]))
                         for a in range(len(ones)) for b in range(len(zeros))]
        else:
            candidates = [(int(rng.choice(ones)), int(rng.choice(zeros)))
                         for _ in range(n_candidates)]
        for i, j in candidates:
            h[i] = 0
            h[j] = 1
            new_out = np.correlate(h, 1 - h, mode='full')
            new_c5 = np.max(new_out)
            delta = new_c5 - c5_val
            if delta < best_delta - 1e-12:
                best_delta = delta
                best_swap = (i, j)
                best_out = new_out.copy()
            h[i] = 1
            h[j] = 0
        if best_swap is None:
            break
        i, j = best_swap
        h[i] = 0
        h[j] = 1
        c5_val = c5_val + best_delta
        out = best_out
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
    return h, c5_val * dx

def sa_swap_search(h, n, max_rounds, rng, T0=2.0, T1=0.1):
    h = h.copy()
    dx = 2.0 / n
    out = np.correlate(h, 1 - h, mode='full')
    c5_val = np.max(out)
    best_c5 = c5_val
    best_h = h.copy()
    for rd in range(max_rounds):
        T = T0 * (T1 / T0) ** (rd / max_rounds)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(rng.choice(ones))
        j = int(rng.choice(zeros))
        h[i] = 0
        h[j] = 1
        new_out = np.correlate(h, 1 - h, mode='full')
        new_c5 = np.max(new_out)
        delta = new_c5 - c5_val
        if delta < 0 or (T > 0 and rng.rand() < np.exp(-delta / T)):
            c5_val = new_c5
            out = new_out
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h.copy()
        else:
            h[i] = 1
            h[j] = 0
    return best_h, best_c5 * dx

def coordinate_descent_continuous(h, n, target_sum, max_sweeps=50):
    h = h.copy()
    for sweep in range(max_sweeps):
        improved = False
        out = np.correlate(h, 1 - h, mode='full')
        for j in range(n):
            c = h[j]
            if c <= 1e-10 or c >= 1.0 - 1e-10:
                continue
            k_lo = max(0, n - 1 - j)
            k_hi = min(2 * n - 2, n - 1 + j)
            w_k = np.zeros(2 * n - 1)
            if k_hi >= k_lo:
                u = j + np.arange(k_lo, k_hi + 1) - (n - 1)
                mask = (u >= 0) & (u < n)
                w_k[k_lo:k_hi + 1][mask] = (1.0 - h[u[mask]])
            k_lo2 = max(0, j)
            k_hi2 = min(2 * n - 2, j + n - 1)
            if k_hi2 >= k_lo2:
                v = j - np.arange(k_lo2, k_hi2 + 1) + (n - 1)
                mask = (v >= 0) & (v < n)
                w_k[k_lo2:k_hi2 + 1][mask] -= h[v[mask]]
            a = out - c * w_k
            lo_v, hi_v = 0.0, 1.0
            for _ in range(50):
                mid = (lo_v + hi_v) / 2
                g_val = np.max(a + mid * w_k)
                active = np.abs(a + mid * w_k - g_val) < 1e-9
                if np.any(active):
                    min_w = np.min(w_k[active])
                    max_w = np.max(w_k[active])
                    if max_w < -1e-12:
                        lo_v = mid
                    elif min_w > 1e-12:
                        hi_v = mid
                    else:
                        break
            v_star = (lo_v + hi_v) / 2
            if abs(v_star - c) > 1e-8:
                h[j] = v_star
                improved = True
                out = np.correlate(h, 1 - h, mode='full')
        if not improved:
            break
    return project_to_sum(h, target_sum)

def make_binary_starts(n, target_sum, rng, best_h, best_n):
    starts = []
    if best_h is not None and best_n > 0:
        if best_n == n:
            starts.append(to_binary(best_h, target_sum))
        else:
            x_old = np.linspace(0, 2, best_n, endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            interp = np.interp(x_new, x_old, best_h, period=2.0)
            starts.append(to_binary(interp, target_sum))
    for trial in range(8):
        h_rand = rng.uniform(0, 1, n)
        starts.append(to_binary(h_rand, target_sum))
    for freq in [1, 2, 3, 4, 5]:
        x = np.linspace(0, 2, n, endpoint=False)
        h_pat = 0.5 + 0.4 * np.sin(2 * np.pi * freq * x / 2.0)
        starts.append(to_binary(h_pat, target_sum))
    return starts

def upsample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old, period=2.0)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()

    def time_left():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)
    best_h = None

    if 'initial_h_values' in dir() and initial_h_values is not None:
