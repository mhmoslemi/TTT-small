def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        if np.sum(proj) > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-14:
        mask = (result > 1e-10) & (result < 1.0 - 1e-10)
        if np.sum(mask) > 0:
            per = diff / np.sum(mask)
            result[mask] = np.clip(result[mask] + per, 0.0, 1.0)
        else:
            idx = int(np.argmin(np.abs(result - 0.5)))
            result[idx] = np.clip(result[idx] + diff, 0.0, 1.0)
    return result
