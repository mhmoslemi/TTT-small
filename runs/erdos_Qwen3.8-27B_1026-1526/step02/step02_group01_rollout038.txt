def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def subgrad_at_shift(h, s):
    """Compute subgradient of corr at shift s (where s = k - (n-1))"""
    n = len(h)
    grad = np.ones(n)
    # grad[j] = 1 - h[j-s] - h[j+s] (with zero padding)
    # h[j-s] is nonzero when 0 <= j-s < n, i.e., s <= j < n+s
    # h[j+s] is nonzero when 0 <= j+s < n, i.e., -s <= j < n-s
    
    if s >= 0:
        # h[j-s] is valid for j in [s, n-1]
        if s < n:
            grad[s:] -= h[:n-s]
        # h[j+s] is valid for j in [0, n-1-s]
        if s < n:
            grad[:n-s] -= h[s:]
    else:
        # s < 0, let t = -s
        t = -s
        # h[j-s] = h[j+t] is valid for j in [0, n-1-t]
        if t < n:
            grad[:n-t] -= h[t:]
        # h[j+s] = h[j-t] is valid for j in [t, n-1]
        if t < n:
            grad[t:] -= h[:n-t]
    
    return grad

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

    lr = lr_init
    stuck_count = 0
    
    for it in range(max_iters):
        corr = compute_corr(h)
        # Find top m shifts
        top_k = np.argsort(corr)[-top_m:]
        top_k = np.sort(top_k)
        
        # Weighted subgradient (weighted by how close to max)
        max_val = np.max(corr)
        weights = np.zeros(len(top_k))
        for i, k in enumerate(top_k):
            weights[i] = np.exp(-0.1 * (max_val - corr[k]))
        weights /= np.sum(weights)

            s = k - (n - 1)
            grad += weights[i] * subgrad_at_shift(h, s)
        
        # Momentum step
        v = momentum * v - lr * grad
        h_new = h + v
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur:
            c5_cur = c5_new
            h_best = h_new.copy()
            h = h_new
            stuck_count = 0
            lr = min(lr * 1.02, 0.5)
        else:
            h = h_new  # still use for exploration
            stuck_count += 1
            lr *= 0.95

                stuck_count = 0
                lr = max(lr * 0.5, 0.001)
        
        # Decay lr over time
        lr = max(lr * 0.9999, 0.001)
    
    return h_best, c5_cur
