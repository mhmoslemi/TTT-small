def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Project h onto {x : 0 <= x_i <= 1, sum(x) = target_sum}"""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        projected = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(projected)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    projected = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    return projected

def compute_gradient(h, beta):
    """Compute gradient of smooth max of correlation values w.r.t. h"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    values = corr * dx
    
    v = beta * values
    v_max = np.max(v)
    exp_v = np.exp(v - v_max)
    w = exp_v / np.sum(exp_v)
    
    grad = np.zeros(n)
    for j in range(n):
        W1 = 0.0
        A = 0.0
        B = 0.0
        # First sum: k from max(0,j) to min(2n-2, j+n-1)

        k_end = min(2 * n - 2, j + n - 1)

            A += wk * h[idx]

            B += w[k] * h[idx]

            beta *= 2

    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_gradient(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx
    v = beta * vals
    vm = np.max(v)
    ev = np.exp(v - vm)
    w = ev / np.sum(ev)
    h_rev = h[::-1]
    A = np.empty(n)
    B = np.empty(n)
    W1 = np.empty(n)
    for j in range(n):
        A[j] = np.dot(w[j:j + n], h_rev)
        B[j] = np.dot(w[n - 1 - j:2 * n - 1 - j], h)
        W1[j] = np.sum(w[j:j + n])
    return dx * (W1 - A - B)

def gradient_descent(h_init, n, max_time, t_start):
    target = n / 2.0
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    beta = 20.0
    lr = 0.05
    mu = 0.9
    vel = np.zeros(n)
    no_improve = 0
    while time.time() - t_start < max_time:
        g = compute_gradient(h, beta)
        vel = mu * vel - lr * g
        vel = np.clip(vel, -1.0, 1.0)
        h_new = project_to_feasible(h + vel, target)
        c5 = compute_c5(h_new)
        h = h_new
        if c5 < best_c5 - 1e-9:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
            lr = min(lr * 1.02, 0.5)
        else:
            no_improve += 1
            if no_improve > 80:
                h = best_h.copy()
                vel = np.zeros(n)
                beta = min(beta * 2.0, 3000.0)
                lr = max(lr * 0.6, 0.002)
                if beta > 3000.0:
                    beta = 15.0
                    lr = 0.04
                    mu = 0.85
                no_improve = 0
    return best_h, best_c5

def perturb_random(h, n, scale):
    h_new = h + np.random.randn(n) * scale
    return project_to_feasible(h_new, n / 2.0)

def perturb_structured(h, n):
    h_new = h.copy()
    freq = np.random.randint(1, 6)
    phase = np.random.uniform(0, 2.0 * np.pi)
    amp = 0.04 + 0.06 * np.random.rand()
    idx = np.arange(n)
    h_new += amp * np.sin(2.0 * np.pi * freq * idx / n + phase)
    return project_to_feasible(h_new, n / 2.0)

def perturb_binary(n):
    h = np.zeros(n)
    n_ones = int(n / 2)
    idx = np.random.choice(n, n_ones, replace=False)
    h[idx] = 1.0
    if n % 2 == 1:
        h[np.random.choice(n)] = 0.5
    return project_to_feasible(h, n / 2.0)

def local_search(h, n, max_time, t_start):
    target = n / 2.0
    best_c5 = compute_c5(h)
    h_curr = h.copy()
    step = 0.03
    while time.time() - t_start < max_time:
        n_changes = np.random.choice([1, 1, 2, 2, 3])
        n_changes = min(n_changes, n)
        idx = np.random.choice(n, n_changes, replace=False)
        h_try = h_curr.copy()
        for i in idx:
            h_try[i] += np.random.uniform(-step, step)
        h_try = project_to_feasible(h_try, target)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-10:
            best_c5 = c5_try
            h_curr = h_try
            step = min(step * 1.05, 0.15)
        else:
            step *= 0.99
            if step < 0.001:
                break
    return h_curr, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    n = 45
    target = n / 2.0
    
    try:
        h0 = np.array(initial_h_values, dtype=float)
        if len(h0) != n:
            h0 = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(h0)), h0)
        h0 = project_to_feasible(h0, target)
    except Exception:
        h0 = np.full(n, 0.5)
    
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    
    b1 = min(80.0, budget_s * 0.12)
    h_opt, c_opt = gradient_descent(h0, n, b1, time.time())
    if c_opt < best_c5:
        best_c5 = c_opt
        best_h = h_opt
    
    elapsed = time.time() - t0
    remaining = budget_s - elapsed - 40.0
    if remaining > 20:
        n_restarts = max(2, int(remaining / 18.0))
        n_restarts = min(n_restarts, 30)
        for i in range(n_restarts):
            elapsed = time.time() - t0
            if elapsed > budget_s - 40:
                break
            alloc = (budget_s - 40 - elapsed) / (n_restarts - i)
            alloc = max(alloc, 5.0)
            
            r = i % 5
            if r == 0:
                hp = perturb_random(best_h, n, 0.1)
            elif r == 1:
                hp = perturb_random(best_h, n, 0.05)
            elif r == 2:
                hp = perturb_structured(best_h, n)
            elif r == 3:
                hp = perturb_random(best_h, n, 0.2)
            else:
                hp = perturb_binary(n)
            
            h_opt, c_opt = gradient_descent(hp, n, alloc, time.time())
            if c_opt < best_c5:
                best_c5 = c_opt
                best_h = h_opt
    
    remaining = budget_s - (time.time() - t0) - 3.0
    if remaining > 5:
        h_fine, c_fine = local_search(best_h, n, remaining, time.time())
        if c_fine < best_c5:
            best_c5 = c_fine
            best_h = h_fine
    
    return (best_h, best_c5, n)
