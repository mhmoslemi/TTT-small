    target = n / 2.0
    h = h.copy()
    for _ in range(200):
        h = np.clip(h, 0, 1)
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        # Add diff/n to all elements, but only those that can absorb it
        h = h + diff / n
        # Clip again

    # Final adjustment: distribute remaining error
    diff = target - np.sum(h)

        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0, 1)

def project_onto_box_hyperplane(y, n):
    """Project y onto {x: sum(x) = n/2, 0 <= x_i <= 1}"""
    target = n / 2.0
    
    # Bisection for lambda
    lo = np.min(y) - 1 - target/n  # lower bound for lambda
    hi = np.max(y) + 1  # upper bound for lambda
    
    # Actually: f(lambda) = sum(clip(y - lambda, 0, 1))
    # f is decreasing in lambda
    # We want f(lambda) = target
    
    def f(lam):
        return np.sum(np.clip(y - lam, 0.0, 1.0))
    
    for _ in range(100):
        mid = (lo + hi) / 2
        if f(mid) > target:
            lo = mid
        else:
            hi = mid
    
    lam = (lo + hi) / 2
    x = np.clip(y - lam, 0.0, 1.0)
    # Small numerical fix
    diff = target - np.sum(x)
    if abs(diff) > 1e-12:
        # Distribute among non-boundary elements
        mask = (x > 1e-10) & (x < 1 - 1e-10)
        if np.sum(mask) > 0:
            x[mask] += diff / np.sum(mask)
            x = np.clip(x, 0, 1)
    return x

import numpy as np
from scipy.optimize import minimize
import time
import math

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_c5_val(h, s=200.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    scaled = s * corr * dx
    m = np.max(scaled)
    return float((m + np.log(np.sum(np.exp(scaled - m)))) / s)

def objective_pen(x, n, s=200.0, penalty=1e5):
    val = smooth_c5_val(x, s)
    viol = np.sum(x) - n / 2.0
    return val + penalty * viol * viol

def project_exact(h, n):
    target = float(n) / 2.0
    def f(lam):
        return float(np.sum(np.clip(h - lam, 0.0, 1.0)))
    lo, hi = -2.0, 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        if f(mid) > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(x)
    if abs(diff) > 1e-13:
        mask = (x > 1e-14) & (x < 1.0 - 1e-14)
        if np.any(mask):
            x[mask] += diff / np.sum(mask)
            x = np.clip(x, 0.0, 1.0)
    diff = target - np.sum(x)
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = np.argsort(-x)[:max(1, int(abs(diff)) + 1)]
        else:
            idx = np.argsort(x)[:max(1, int(abs(diff)) + 1)]
        x[idx] += diff
        x = np.clip(x, 0.0, 1.0)
    return x

def gen_random_start(n, rng, style=0):
    if style == 0:
        h = (rng.rand(n) < 0.5).astype(float)
    elif style == 1:
        h = rng.rand(n)
    elif style == 2:
        p = rng.randint(3, 12)
        pattern = (rng.rand(p) < 0.5).astype(float)
        reps = n // p + 1
        h = np.tile(pattern, reps)[:n]
    elif style == 3:
        freq = rng.randint(1, 6)
        phase = rng.rand() * 2.0 * np.pi
        x = np.linspace(0, 2, n)
        h = 0.5 + 0.3 * np.sin(2.0 * np.pi * freq * x / 2.0 + phase)
        h = np.clip(h, 0, 1)
    else:
        h = 0.5 + 0.4 * rng.randn(n)
        h = np.clip(h, 0, 1)
    return h

def optimize_local(h0, n, s=200.0, maxiter=500):
    bounds = [(0.0, 1.0)] * n
    penalty = 1e5
    res = minimize(
        objective_pen, h0.copy(),
        args=(n, s, penalty),
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10}
    )
    h_opt = project_exact(res.x, n)
    c5 = compute_c5(h_opt)
    return h_opt, c5

def subgradient_polish(h, n, max_iters=2000, learning_rate=0.01):
    h = h.copy()
    dx = 2.0 / n
    best = compute_c5(h)
    for it in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_star = int(np.argmax(corr))
        eps = 1e-7
        subgrad = np.zeros(n)
        for j in range(n):
            h[j] += eps
            c_plus = compute_c5(h)
            h[j] -= 2.0 * eps
            c_minus = compute_c5(h)
            h[j] += eps
            subgrad[j] = (c_plus - c_minus) / (2.0 * eps)
        step = learning_rate / (1.0 + 0.01 * it)
        h -= step * subgrad
        h = project_exact(h, n)
        c5 = compute_c5(h)
        if c5 < best:
            best = c5
            h_best = h.copy()
        else:
            h = h_best if 'h_best' in dir() else h
        if it % 100 == 0 and learning_rate < 1e-6:
            break
        if it > 100 and it % 200 == 0:
            learning_rate *= 0.5
    return h, best
