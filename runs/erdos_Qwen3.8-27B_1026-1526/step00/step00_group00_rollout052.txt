import numpy as np
from scipy.optimize import minimize
import time

_G_N = 45
_G_TARGET = 22.5

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def project_to_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _obj_func(x):
    return compute_c5(x, _G_N)

def _con_eq(x):
    return np.sum(x) - _G_TARGET

def _con_jac(x):
    return np.ones(_G_N)

def run(seed=42, budget_s=1300, **kwargs):
