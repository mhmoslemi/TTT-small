    dx = 2.0 / n_points
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return float(np.max(corr) * dx)

    # Numerically stable soft max
    x_max = np.max(x)
    if np.isinf(x_max):
        return x_max
    return x_max + np.log(np.sum(np.exp(beta * (x - x_max)))) / beta

import numpy as np
import scipy.optimize as optimize
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max_val(x, beta):
    x_max = np.max(x)
    return x_max + np.log(np.sum(np.exp(beta * (x - x_max)))) / beta

def c5_objective(h, n_points, beta):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * smooth_max_val(corr, beta)

def c5_gradient(h, n_points, beta):
    dx = 2.0 / n_points
    n = n_points
    corr = np.correlate(h, 1.0 - h, mode='full')
    x_max = np.max(corr)
    exp_diffs = beta * (corr - x_max)
    exp_vals = np.exp(exp_diffs)
    exp_sum = np.sum(exp_vals)
    weights = exp_vals / exp_sum
    
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-14:
            continue
        lag = m - n + 1
        i_start = max(0, -lag)
        i_end = min(n, n - lag)
        for i in range(i_start, i_end):
            j = i + lag
            grad[j] += w * (1.0 - h[i])
            grad[i] -= w * h[j]
    
    return dx * grad

def project_feasible(h, n_points):
    # Project to 0 <= h[i] <= 1 and sum(h) = n_points/2
    target_sum = n_points / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        # Distribute the difference
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h

import numpy as np
import scipy.optimize as optimize
import time

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max_val(x, beta):
    m = np.max(x)
    if m == -np.inf:
        return 0.0
    shifted = beta * (x - m)
    # Clip for numerical stability
    shifted = np.clip(shifted, -100, 100)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def c5_smooth(h, n_points, beta):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * smooth_max_val(corr, beta)

def c5_smooth_grad(h, n_points, beta):
    dx = 2.0 / n_points
    n = n_points
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -100, 100)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    weights = exp_vals / s
    
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        w = weights[m_idx]
        if w < 1e-14:
            continue
        lag = m_idx - n + 1
        i_start = max(0, -lag)
        i_end = min(n, n - lag)
        for i in range(i_start, i_end):
            j = i + lag
            grad[j] += w * (1.0 - h[i])
            grad[i] -= w * h[j]
    
    return dx * grad

def project_feasible(h, n_points):
    target = n_points / 2.0
    h = np.array(h, dtype=float, copy=True)
    h = np.clip(h, 0.0, 1.0)

        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h
