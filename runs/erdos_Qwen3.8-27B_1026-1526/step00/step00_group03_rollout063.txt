def repair_h(h, n):
    """Repair h to satisfy constraints: 0<=h<=1, sum(h)=n/2."""
    h = np.clip(h, 0.0, 1.0)
    target_sum = n / 2.0
    current_sum = h.sum()
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Scale to fix sum
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    else:
        h = np.full(n, 0.5)
    # Clip again after scaling
    h = np.clip(h, 0.0, 1.0)
    # Final adjustment
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        # Distribute the difference
        indices = np.argsort(h)
        if diff > 0:
            # Need to increase sum, add to smallest values
            for idx in indices:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add

            for idx in indices:

                h[idx] -= rem
                diff += rem
                if abs(diff) < 1e-10:
                    break

def evaluate_h(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_sum(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.argsort(h)
        for i in idx:
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
            if diff < 1e-12:
                break
    else:
        idx = np.argsort(h)[::-1]
        for i in idx:
            rem = min(-diff, h[i])
            h[i] -= rem
            diff += rem
            if -diff < 1e-12:
                break
    return h

        self.n = n

        m = corr.max()
        return float(m + np.log(np.sum(np.exp(self.alpha * (corr - m)))) / self.alpha)

    # Only use "active" correlations (within tol of max)
    active_mask = corrs > max_corr - tol
    active_indices = np.where(active_mask)[0]
    active_shifts = [(-(n-1) + idx) for idx in active_indices]
    
    # Compute gradients for active correlations
    num_active = len(active_shifts)
    # A constraint matrix: each row is [grad_k, -1] (for corr_k + (-1)*M <= 0 form)
    # Actually, let's formulate as:
    # minimize M
    # subject to corr_k(h') <= M for all active k
    #            0 <= h' <= 1
    #            sum(h') = n/2
    # Linearize: corr_k(h + d) ≈ corr_k(h) + grad_k · d
    #           corr_k(h) + grad_k · d <= M
    #           grad_k · d - M <= -corr_k(h)
    
    # Variables: d[0], ..., d[n-1], M (n+1 variables)
    # c = [0, ..., 0, 1] (minimize M)
    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.zeros((num_active, n + 1))
    b_ub = np.zeros(num_active)

        A_ub[i, :n] = g
        A_ub[i, n] = -1.0
        b_ub[i] = -corrs[active_indices[i]]

    b_eq[0] = 0.0  # sum(d) = 0 (since sum(h) = n/2 is already satisfied)

import numpy as np

# Top-level helper functions

def evaluate_h(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_sum(h, n):
    target = n / 2.0
    h = np.array(h, dtype=np.float64, copy=True)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.argsort(h)
        for i in idx:
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
            if diff < 1e-12:
                break
    else:
        idx = np.argsort(h)[::-1]
        for i in idx:
            rem = min(-diff, h[i])
            h[i] -= rem
            diff += rem
            if -diff < 1e-12:
                break
    return h

def compute_correlations_direct(h):
    n = len(h)
    num_shifts = 2 * n - 1
    corrs = np.zeros(num_shifts)

class SMO:
    """Smooth max objective for scipy."""
    def __init__(self, alpha, n):
        self.alpha = alpha
        self.n = n
    def __call__(self, x):
        dx = 2.0 / self.n
        corr = np.correlate(x, 1.0 - x, mode="full") * dx
        m = corr.max()
        return float(m + np.log(np.sum(np.exp(self.alpha * (corr - m)))) / self.alpha)

    """Sum constraint for scipy."""
    def __init__(self, n):
        self.n = n
