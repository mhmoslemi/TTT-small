import numpy as np
from scipy.optimize import minimize
import time


def _corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def _c5(h):
    return float(np.max(_corr_scaled(h)))


def _project(h, n):
    h = np.array(h, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    for _ in range(30):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        h += d / n
        h = np.clip(h, 0.0, 1.0)
    return h


def _try_coord(h, n, i, grid):
    target = n * 0.5
    best_v = _c5(h)
    best_h = h
    for t in grid:
        if abs(t - h[i]) < 1e-14:
            continue
        ht = h.copy()
        ht[i] = t
        rest_target = target - t
        rest_sum = ht.sum() - t
        diff = rest_target - rest_sum
        if abs(diff) > 1e-15:
            mask = np.ones(n, dtype=bool)
            mask[i] = False
            ht[mask] += diff / (n - 1)
            ht[mask] = np.clip(ht[mask], 0.0, 1.0)
            rest_sum2 = ht.sum() - t
            diff2 = rest_target - rest_sum2
            if abs(diff2) > 1e-15:
                ht[mask] += diff2 / (n - 1)
                ht[mask] = np.clip(ht[mask], 0.0, 1.0)
        v = _c5(ht)
        if v < best_v:
            best_v = v
            best_h = ht
    return best_v, best_h


def _coord_descent(h, n, grid, max_sweeps, deadline):
    h = np.array(h, dtype=np.float64).copy()
    cur = _c5(h)
    for _ in range(max_sweeps):
        if time.time() >= deadline:
            break
        imp = False
        for i in range(n):
            v, new_h = _try_coord(h, n, i, grid)
            if v < cur - 1e-15:
                cur = v
                h = new_h
                imp = True
        if not imp:
            break
    return h, cur

    h = np.clip(x, 0.0, 1.0)
    c = _corr_scaled(h)
    m = c.max()
    T = 100.0
    sm = m + np.log(np.sum(np.exp(T * (c - m)))) / T
    pen = 1e10 * (h.sum() - n * 0.5) ** 2
    return sm + pen
