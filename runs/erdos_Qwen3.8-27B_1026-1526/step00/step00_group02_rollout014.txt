            # Binary search for optimal h[i]
            # The objective as function of h[i] is max of linear functions (convex)
            # Try: 0, 1, and a few intermediate values
            candidates = [0.0, 1.0]

            if 0 < old_val < 1:

                if best_candidate_c5 < best_c5:
                    best_c5 = best_candidate_c5
                    best = h.copy()
        
        if not improved:

    # Use ternary search (or just dense grid + refinement)
    # For speed, use a grid of 1000 points then refine
    x_grid = np.linspace(0, 1, 1000)
    vals = np.max(base + x_grid[:, None] * coeff[None, :], axis=1)
    idx = np.argmin(vals)

    lo = x_grid[max(0, idx-1)]
    hi = x_grid[min(len(x_grid)-1, idx+1)]

        m2 = hi - (hi-lo)/3

minimize t
subject to: c_k(h_old) + ∇c_k(h_old)·(h_new - h_old) <= t for all k
             0 <= h_new[i] <= 1
             sum(h_new) = n/2

This is the linearized (first-order) approach, essentially a form of minorization-maximization or cutting plane.

def linearized_lp(h, dx, n, target_sum):
    """Solve the linearized LP at current point h."""
    c_full = np.correlate(h, 1 - h, mode="full") * dx
    
    t = cp.Variable()
    h_var = cp.Variable(n)
    cons = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target_sum]
    
    for k_idx in range(2 * n - 1):
        lag = k_idx - (n - 1)
        # Compute c_k and its gradient at h
        c_k = c_full[k_idx]
        grad_k = np.zeros(n)
        for i in range(n):
            g = 0.0
            j1 = i + lag
            if 0 <= j1 < n:
                g += (1.0 - h[j1])
            j2 = i - lag
            if 0 <= j2 < n:
                g += -h[j2]
            grad_k[i] = g
        grad_k *= dx
        # Linearization: c_k(h_new) ≈ c_k + grad_k · (h_new - h) = grad_k·h_new + (c_k - grad_k·h)
        offset = c_k - np.dot(grad_k, h)
        cons.append(cp.dot(grad_k, h_var) + offset <= t)
    
    prob = cp.Problem(cp.Minimize(t), cons)
    prob.solve(solver=cp.GLPK)  # or whatever is available
    
    if h_var.value is not None:
        return h_var.value, t.value
    return h, c_full.max()

def project_sum(h, target_sum):

def solve_lp(h, dx, target_sum):
    n = len(h)
    c_full = np.correlate(h, 1.0 - h, mode="full") * dx
    t = cp.Variable()
    h_var = cp.Variable(n)
    cons = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target_sum]
    for k_idx in range(2 * n - 1):
        lag = k_idx - (n - 1)
        c_k = c_full[k_idx]

                g += -h[j2]
            grad_k[i] = g * dx

    prob = cp.Problem(cp.Minimize(t), cons)

def solve_lp_scipy(h, dx, target_sum):
    n = len(h)
    c_full = np.correlate(h, 1.0 - h, mode="full") * dx
    
    # Variables: [h_0, ..., h_{n-1}, t]
    # Minimize: [0, ..., 0, 1] · x
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality constraints A_ub @ x <= b_ub
    A_ub_list = []
    b_ub_list = []
    
    for k_idx in range(2 * n - 1):
        lag = k_idx - (n - 1)
        c_k = c_full[k_idx]
        grad_k = np.zeros(n)
        for i in range(n):
            g = 0.0
            j1 = i + lag
            if 0 <= j1 < n:
                g += 1.0 - h[j1]
            j2 = i - lag
            if 0 <= j2 < n:
                g += -h[j2]
            grad_k[i] = g * dx
        offset = c_k - np.dot(grad_k, h)
        # grad_k · h_var - t <= -offset
        row = np.zeros(n + 1)
        row[:n] = grad_k
        row[n] = -1.0
        A_ub_list.append(row)
        b_ub_list.append(-offset)
    
    # h[i] >= 0: -h[i] <= 0
    for i in range(n):
        row = np.zeros(n + 1)
        row[i] = -1.0
        A_ub_list.append(row)
        b_ub_list.append(0.0)
    
    # h[i] <= 1: h[i] <= 1
    for i in range(n):
        row = np.zeros(n + 1)
        row[i] = 1.0
        A_ub_list.append(row)
        b_ub_list.append(1.0)
    
    A_ub = np.array(A_ub_list)
    b_ub = np.array(b_ub_list)
    
    # Equality: sum(h) = target_sum
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target_sum])
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=[(0, 1)] * n + [(None, None)], method='highs')
    
    if res.success:
        h_new = res.x[:n]
        return h_new, res.x[n]
    return h.copy(), np.max(c_full)

result has length 2n-1

For real arrays: result[m] = sum_j a[j] * b[j + m - n + 1]

So for our case: a = h, b = 1-h

Let lag = m - n + 1, so m = n - 1 + lag, lag ranges from -(n-1) to (n-1):

The derivative of c_{lag} with respect to h[i]:
c_{lag} = sum_j h[j] - h[j]*h[j+lag] (for valid j)

- If i is in the first slot (j=i): ∂(h[i]*(1-h[i+lag]))/∂h[i] = (1-h[i+lag])

import numpy as np
import time
from scipy.optimize import linprog

def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project_sum(h, target_sum):
    n = len(h)
    if target_sum <= 0.0:
        return np.zeros(n)
    if target_sum >= float(n):
        return np.ones(n)
    lo, hi = -1.0, 2.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) * 0.5
    return np.clip(h - t, 0.0, 1.0)

def solve_lp(h, dx, target_sum):
    n = len(h)
    c_full = np.correlate(h, 1.0 - h, mode="full") * dx
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    rows_ub = []
    vals_ub = []
    for k_idx in range(2 * n - 1):
        lag = k_idx - (n - 1)
        c_k = c_full[k_idx]
        gk = np.zeros(n)
        for i in range(n):
            g = 0.0
            j1 = i + lag
            if 0 <= j1 < n:
                g += 1.0 - h[j1]
            j2 = i - lag
            if 0 <= j2 < n:
                g += -h[j2]
            gk[i] = g * dx
        offset = c_k - np.dot(gk, h)
        row = np.zeros(n + 1)
        row[:n] = gk
        row[n] = -1.0
        rows_ub.append(row)
        vals_ub.append(-offset)
    neg_bound = np.zeros((n, n + 1))
    neg_bound[:n, :n] = -np.eye(n)
    rows_ub.append(neg_bound)
    vals_ub.extend([0.0] * n)
    pos_bound = np.zeros((n, n + 1))
    pos_bound[:n, :n] = np.eye(n)
    rows_ub.append(pos_bound)
    vals_ub.extend([1.0] * n)
    A_ub = np.array(rows_ub)
    b_ub = np.array(vals_ub)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target_sum])
    bounds = [(0.0, 1.0)] * n + [(None, None)]
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        hv = res.x[:n]
        hv = np.clip(hv, 0.0, 1.0)
        return hv, float(res.x[n])
    return h.copy(), float(np.max(c_full))

def optimize_from(h, dx, target_sum, n_restart, max_iters, t_end, rng):
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    for r in range(n_restart):
        if time.time() > t_end:
            break
        if r > 0:
            scale = 0.08 + 0.15 * rng.random()
            h = best_h + rng.uniform(-scale, scale, len(best_h))
            h = project_sum(h, target_sum)
        stuck = 0
        cur_c5 = compute_c5(h, dx)
        for it in range(max_iters):
            if time.time() > t_end:
                break
            h_new, _ = solve_lp(h, dx, target_sum)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new.copy()
            if c5_new < cur_c5 - 1e-10:
                h = h_new
                cur_c5 = c5_new
                stuck = 0
            else:
                stuck += 1
                if stuck >= 4:
                    break
    return best_h, best_c5
