    """Compute C5 = max_k correlate(h, 1-h, 'full') * dx"""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_constraint(h, n):
    """Project h onto {0≤h≤1, sum(h)=n/2}"""
    h = np.clip(h, 0, 1)
    target = n / 2.0
    # Iteratively adjust
    for _ in range(100):
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        # Add/subtract to elements that have room
        diff = target - s
        room_up = 1.0 - h  # room to go up
        room_down = h  # room to go down
        if diff > 0:
            # Need to increase sum
            total_room = room_up.sum()
            if total_room < 1e-12:
                break
            h = h + diff * room_up / total_room
        else:
            # Need to decrease sum
            total_room = room_down.sum()
            if total_room < 1e-12:
                break
            h = h + diff * room_down / total_room
        h = np.clip(h, 0, 1)
    return h

def smooth_max(x, temperature=0.01):
    """Smooth approximation of max using log-sum-exp"""
    x_shifted = x - np.max(x)
    return np.max(x) + temperature * np.log(np.sum(np.exp(x_shifted / temperature)))
    
def c5_smooth(h, temperature=0.01):
    """Smooth version of C5 for optimization"""
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return smooth_max(corr, temperature)

def optimize_step(h, n, temperature=0.01, max_iter=200):
    """One optimization step using L-BFGS-B"""
    def objective(x):
        h = project_to_constraint(x, n)
        return c5_smooth(h, temperature)
    
    def gradient(x):
        # Use finite differences for simplicity (n is small)
        eps = 1e-6
        grad = np.zeros(n)
        f0 = objective(x)
        for i in range(n):
            x_plus = x.copy()
            x_plus[i] += eps
            grad[i] = (objective(x_plus) - f0) / eps
        return grad
    
    result = minimize(objective, h, method='L-BFGS-B',

                for gi in range(len(generators)):
                    h_try = generators[gi](n)
                    h_try = project_feasible(h_try, n)
                    c5_try = evaluate_c5(h_try)
                    if c5_try < best_c5:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                h_start = project_feasible(interp_to(best_h, n), n)
            else:
                h_base = interp_to(best_h, n)
                sigma = 0.12 * (1.0 + 0.3 * ri)
                h_start = h_base + sigma * np.random.randn(n)
                h_start = project_feasible(h_start, n)

            time_for_opt = min(rem - 15, 150)
            if time_for_opt < 10:
                break

            use_smooth = (ri % 2 == 1)
            h_opt = optimize_once(h_start, n, time_for_opt, use_smooth)
            c5_opt = evaluate_c5(h_opt)

            if c5_opt < best_c5 - 1e-8:
                best_c5 = c5_opt
                best_h = h_opt.copy()

        # Local refinement at this n
        elapsed = time.time() - t0
        rem = budget_s - elapsed
        if rem > 40:
            h_try = best_h + 0.04 * np.random.randn(n)
            h_try = project_feasible(h_try, n)
            h_opt = optimize_once(h_try, n, min(rem - 10, 80), use_smooth=False)
            c5_opt = evaluate_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Final polish rounds
    elapsed = time.time() - t0
    rem = budget_s - elapsed
    nf = len(best_h)
    polish_rounds = min(5, max(1, int(rem / 40)))
    for _ in range(polish_rounds):
        if time.time() - t0 > budget_s - 10:
            break
        h_try = best_h + 0.02 * np.random.randn(nf)
        h_try = project_feasible(h_try, nf)
        h_opt = optimize_once(h_try, nf, 60, use_smooth=False)
        c5_opt = evaluate_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Snap near 0/1 values to help discrete structure
    best_h = np.where(best_h < 0.01, 0.0, best_h)
    best_h = np.where(best_h > 0.99, 1.0, best_h)
    best_h = project_feasible(best_h, len(best_h))
    c5_snapped = evaluate_c5(best_h)
    if c5_snapped > best_c5:
        # Revert snap if it made things worse
        best_h = project_feasible(best_h, len(best_h))
        best_h = np.where(best_h < 0.01, 0.0, best_h)
        best_h = np.where(best_h > 0.99, 1.0, best_h)
        best_h = project_feasible(best_h, len(best_h))
        best_c5 = evaluate_c5(best_h)

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = evaluate_c5(best_h)

    return best_h, best_c5, len(best_h)
