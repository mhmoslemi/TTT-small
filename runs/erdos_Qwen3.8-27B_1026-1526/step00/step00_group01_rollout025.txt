    return np.correlate(h, 1.0 - h, mode="full") * dx

    return m + t * np.log(np.sum(np.exp((v - m) / t)))

    sm = smooth_max(corr, temp)
    sum_err = np.sum(h) - n / 2.0
    return sm + penalty * sum_err**2
