from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    w = 1.0 - h
    corr = np.correlate(h, w, mode='full')
    return float(np.max(corr) * dx)

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    start_time = time.time()
    
    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)
    
    def elapsed():
        return time.time() - start_time
    
    def project_sum(x, target):
        """Project x onto {sum = target} while keeping in [0,1]."""
        x = np.clip(x, 0, 1)
        diff = target - np.sum(x)
        if abs(diff) < 1e-12:
            return x
        # Simple: distribute uniformly among coordinates that can absorb
        if diff > 0:
            mask = x < 1.0 - 1e-10
            if np.sum(mask) == 0:
                return x
            x[mask] += diff / np.sum(mask)
        else:
            mask = x > 1e-10
            if np.sum(mask) == 0:
                return x
            x[mask] += diff / np.sum(mask)
        return np.clip(x, 0, 1)
    
    def slsqp_solve(x0, maxiter=200, ftol=1e-12):
        def obj(x):
            return compute_c5(x, dx)
        
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        bounds = [(0.0, 1.0)] * n

                          constraints=constraints,
                          options={'maxiter': maxiter, 'ftol': ftol})
            x = np.clip(res.x, 0, 1)
            x = project_sum(x, target_sum)
            return x, compute_c5(x, dx)
