    """Project h onto the simplex {x: sum(x)=target, 0<=x<=1}"""
    h = h.copy()

    # Iterative projection
    for _ in range(100):

        if abs(diff) < 1e-12:

        # Add/subtract to elements that have room
        if diff > 0:
            room = 1.0 - h
            mask = room > 1e-10
            if not np.any(mask):

            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0, 1)
