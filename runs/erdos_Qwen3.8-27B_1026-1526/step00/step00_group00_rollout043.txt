import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)

def grad_corr_lag(h, lag_idx):
    n = len(h)
    L = lag_idx - (n - 1)
    grad = np.zeros(n)
    if L >= 0:
        idx = np.arange(n - L)
        grad[idx + L] += (1.0 - h[idx])
        grad[idx] -= h[idx + L]
    else:
        L2 = -L
        idx = np.arange(L2, n)
        grad[idx - L2] += (1.0 - h[idx])
        grad[idx] -= h[idx - L2]
    return grad

def project_sum(h, target_sum):
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def softmax_weights(corr, tau):
    tc = tau * corr
    tc_max = np.max(tc)
    e = np.exp(tc - tc_max)
    return e / np.sum(e)

def smooth_max_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    w = softmax_weights(corr, tau)
    grad = np.zeros(n)
    for k in range(len(corr)):
        if w[k] > 1e-12:
            grad += w[k] * grad_corr_lag(h, k)
    return dx * grad

def optimize_from(h, target_sum, n, time_limit, verbose=False):
    start = time.time()
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Continuation on tau
    tau_list = [2.0, 5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 500.0]
    
    for tau in tau_list:
        if time.time() - start > time_limit:

        vel = np.zeros(n)
        lr = 0.5 / np.sqrt(tau + 1)

                break

                    best_h = h_cur.copy()
                    if verbose:

            else:
                lr *= 0.95

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)

def grad_corr_lag(h, lag_idx):
    n = len(h)
    L = lag_idx - (n - 1)
    grad = np.zeros(n)
    if L >= 0:
        idx = np.arange(n - L)
        grad[idx + L] += (1.0 - h[idx])
        grad[idx] -= h[idx + L]
    else:
        L2 = -L
        idx = np.arange(L2, n)
        grad[idx - L2] += (1.0 - h[idx])
        grad[idx] -= h[idx - L2]
    return grad

def project_sum(h, target_sum):
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def smooth_max_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    tc = tau * corr
    tc_max = np.max(tc)
    e = np.exp(tc - tc_max)
    w = e / np.sum(e)
    grad = np.zeros(n)
    for k in range(len(corr)):
        if w[k] > 1e-12:
            grad += w[k] * grad_corr_lag(h, k)
    return dx * grad

def optimize_continuation(h, target_sum, n, time_limit):
    start = time.time()
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    taus = [2.0, 5.0, 10.0, 25.0, 50.0, 100.0, 200.0, 500.0, 1000.0]
    for tau in taus:
        if time.time() - start > time_limit:
            break
        h_cur = best_h.copy()
        vel = np.zeros(n)
        lr = 2.0 / (tau + 10.0)
        mu = 0.9
        no_improve = 0
        for step in range(20000):
            if time.time() - start > time_limit:
                break
            grad = smooth_max_grad(h_cur, tau)
            vel = mu * vel + (1.0 - mu) * grad
            h_new = project_sum(h_cur - lr * vel, target_sum)
            c5_new = compute_c5(h_new)
            c5_cur = compute_c5(h_cur)
            if c5_new < c5_cur - 1e-12:
                h_cur = h_new
                no_improve = 0
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_cur.copy()
            else:
                no_improve += 1
                lr *= 0.92
            lr *= 0.99998
            if lr < 1e-7:
                break
            if no_improve > 3000 and tau > 5:
                break
    return best_h, best_c5

def subgradient_polish(h, target_sum, n, time_limit):
    start = time.time()
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    vel = np.zeros(n)
    lr = 0.003
    for step in range(100000):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_val = np.max(corr)
        active = np.where(corr >= max_val - 0.03)[0]
        grad = np.zeros(n)
        for k in active:
            grad += grad_corr_lag(h, k)
        grad /= float(len(active))
        grad *= dx
        vel = 0.95 * vel + 0.05 * grad
        h_new = project_sum(h - lr * vel, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < compute_c5(h) - 1e-14:
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            lr *= 0.93
        if lr < 1e-9:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):

    np.random.seed(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:

    else:

    best_h, best_c5 = optimize_continuation(best_h, target_sum, n, t1)

    while time.time() - t0 < t2:
        remaining = t2 - (time.time() - t0)
        if remaining < 3.0:
            break
        scale = 0.005 + 0.04 * np.random.rand()
        h_pert = best_h + scale * np.random.randn(n)
        h_pert = project_sum(h_pert, target_sum)
        rt = min(remaining * 0.2, 15.0)
        h_opt, c5_opt = optimize_continuation(h_pert, target_sum, n, rt)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt

    if time.time() - t0 < budget_s * 0.88:
        t3 = budget_s * 0.96
        best_h, best_c5 = subgradient_polish(best_h, target_sum, n, budget_s - (time.time() - t0))

    best_h = project_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n)
