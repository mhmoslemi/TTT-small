class LogSumExpObjective:
    def __init__(self, beta, penalty_weight=0.0, target_sum=0.0):
        self._beta = beta
        self._penalty_weight = penalty_weight
        self._target_sum = target_sum

    def __call__(self, h):
        n = len(h)
        dx = 2.0 / n
        complement = 1.0 - h
        corr = np.correlate(h, complement, mode="full") * dx
        val = (1.0 / self._beta) * logsumexp(self._beta * corr)
        if self._penalty_weight > 0.0:
            val += self._penalty_weight * (np.sum(h) - self._target_sum) ** 2
        return val


class SumEqualityConstraint:
    def __init__(self, target_sum):
        self._target = target_sum

    def fun(self, x):
        return np.sum(x) - self._target

    def jac(self, x):
        return np.ones(len(x))


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    complement = 1.0 - h
    corr = np.correlate(h, complement, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h, target_sum):
    n = len(h)
    lo = float(-h.max()) - 1.0
    hi = float(1.0 - h.min()) + 1.0
    for _ in range(200):
        mid = (lo + hi) * 0.5
        h_proj = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(h_proj)
        if s < target_sum:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-14:
            break
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)


def optimize_local(h_start, n, target_sum, beta, maxiter, penalty_weight=0.0):
    bounds = [(0.0, 1.0)] * n
    obj = LogSumExpObjective(beta, penalty_weight, target_sum)
    
    # Use L-BFGS-B with penalty
    result = minimize(obj, h_start, method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxiter, 'gtol': 1e-10, 'ftol': 1e-12})
    
    if result.success or result.fun < obj(h_start):
        h_new = project_feasible(result.x, target_sum)
    else:
        h_new = h_start
    return h_new


def optimize_slsqp(h_start, n, target_sum, beta, maxiter):
    bounds = [(0.0, 1.0)] * n
    obj = LogSumExpObjective(beta)
    cons = [{'type': 'eq', 'fun': SumEqualityConstraint(target_sum).fun,
             'jac': SumEqualityConstraint(target_sum).jac}]
    
    result = minimize(obj, h_start, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
    
    if result.success:
        h_new = project_feasible(result.x, target_sum)
        return h_new
    return h_start


def generate_random_start(n, rng):
    h = rng.rand(n)
    return project_feasible(h, n * 0.5)


def generate_sinusoidal_start(n, rng, k_range=(1, 5)):
    k = rng.randint(k_range[0], k_range[1] + 1)
    phase = rng.uniform(0, 2 * np.pi)

    i = np.arange(n)
    h = 0.5 + amplitude * np.cos(2 * np.pi * k * i / n + phase)

class LogSumExpObj:
    def __init__(self, beta, penalty_weight, target_sum):
        self._beta = beta
        self._penalty_weight = penalty_weight
        self._target_sum = target_sum

    def __call__(self, h):
        n = len(h)
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        val = (1.0 / self._beta) * logsumexp(self._beta * corr)
        if self._penalty_weight > 0.0:
            val += self._penalty_weight * (np.sum(h) - self._target_sum) ** 2
        return val

class SumConstraint:
    def __init__(self, target):
        self._target = target

        return np.sum(x) - self._target
    def jac(self, x):
