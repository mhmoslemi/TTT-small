import numpy as np

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def compute_all_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_feasible(h, S):
    h = np.asarray(h, dtype=np.float64)
    h = np.clip(h, 0.0, 1.0)
    lo = -2.0
    hi = 2.0
    for _ in range(100):
        theta = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - theta, 0.0, 1.0))
        if s > S:
            lo = theta
        else:
            hi = theta
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def vsubgrad(h, k):
    n = len(h)
    sg = np.zeros(n)
    if k >= 0:
        if n - k > 0:
            sg[:n-k] += (1.0 - h[k:])
        if k > 0:
            sg[k:] -= h[:n-k]
        else:
            sg[:] -= h[:]
    else:
        km = -k
        if n - km > 0:
            sg[km:] += (1.0 - h[:n-km])
            sg[:n-km] -= h[km:]
    return sg
