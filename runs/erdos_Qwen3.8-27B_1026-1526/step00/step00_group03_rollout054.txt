import numpy as np
from scipy.optimize import minimize
import time


def repair(h, n):
    target = n / 2.0
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tr = room.sum()
            if tr < 1e-15:
                break
            h = h + room * (diff / tr)
        else:
            tr = h.sum()
            if tr < 1e-15:
                break
            h = h + h * (diff / tr)
    return np.clip(h, 0.0, 1.0)


def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def nm_objective(x, n):
    x = repair(x, n)
    return compute_c5(x, n)

    smooth_max = m + np.log(np.sum(exp_term)) / alpha
    sum_constraint = (x.sum() - n / 2.0) ** 2 * penalty
    return smooth_max * dx + sum_constraint
