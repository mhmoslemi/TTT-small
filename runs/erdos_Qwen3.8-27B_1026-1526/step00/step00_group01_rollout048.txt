            starts.append(np.clip(f2(x), 0, 1))

    for _ in range(4):
        starts.append(rng.rand(n))

    for freq in [1.3, 1.7, 2.1, 2.7, 3.3]:
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.3, 0.48)
        starts.append(np.clip(0.5 + amp * np.sin(2 * np.pi * freq * x + phase), 0, 1))

    starts.append(np.linspace(0.05, 0.95, n))
    starts.append(np.linspace(0.95, 0.05, n))

    center = rng.uniform(0.2 * n, 0.8 * n)
    width = rng.uniform(0.15 * n, 0.4 * n)
    bump = np.exp(-0.5 * ((ar - center) / width) ** 2)
    starts.append(np.clip(0.25 + 0.65 * bump, 0, 1))

    c1, c2 = 0.25 * n, 0.75 * n
    w = 0.12 * n
    b1 = np.exp(-0.5 * ((ar - c1) / w) ** 2)
    b2 = np.exp(-0.5 * ((ar - c2) / w) ** 2)
    starts.append(np.clip(0.15 + 0.75 * 0.5 * (b1 + b2), 0, 1))

    h_bin = (rng.rand(n) > 0.5).astype(float)
    starts.append(h_bin * 0.85 + 0.08)

    h_saw = np.mod(ar / n * 4.0, 2.0)
    h_saw = np.where(h_saw < 1.0, h_saw, 2.0 - h_saw)
    starts.append(np.clip(h_saw, 0, 1))

    return starts

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()

    best_h = None
    best_c5 = float('inf')

    parent_h = None
    try:
        parent_h = np.array(initial_h_values, dtype=float)
    except Exception:
        pass

    n_candidates = set()
    if parent_h is not None:
        n_candidates.add(len(parent_h))
    for nn in [41, 43, 45, 47, 49, 51, 53, 55, 57, 59, 61, 63, 65, 67, 69, 71, 73, 75]:
        n_candidates.add(nn)
    n_values = sorted(n_candidates)

    for n in n_values:
        if time.time() - start_time > budget_s * 0.78:
            break

        starts = generate_starts(n, parent_h, rng)
        elapsed = time.time() - start_time
        remaining = budget_s * 0.78 - elapsed
        if remaining < 15:
            continue

        for h0 in starts:
            if time.time() - start_time > budget_s * 0.82:
                break

            h0 = project_feasible(h0, n)
            bounds = [(0.0, 1.0)] * n
            cons = [{'type': 'eq', 'fun': constraint_func, 'args': (n,)}]
            current_h = h0.copy()

            taus = [0.08, 0.04, 0.02, 0.01]
            for tau in taus:
                if time.time() - start_time > budget_s * 0.82:
                    break
                try:
                    res = minimize(
                        obj_func, current_h,
                        method='SLSQP',
                        args=(n, tau),
                        bounds=bounds,
                        constraints=cons,
                        options={'maxiter': 50, 'ftol': 1e-12}
                    )
                    current_h = project_feasible(res.x, n)
                    c5 = evaluate_c5(current_h)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = current_h.copy()
                except Exception:
                    pass

            c5 = evaluate_c5(current_h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = current_h.copy()

    if best_h is not None and time.time() - start_time < budget_s * 0.90:
        n = len(best_h)
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': constraint_func, 'args': (n,)}]

        for tau in [0.005, 0.003, 0.001, 0.0005]:
            if time.time() - start_time > budget_s * 0.95:
                break
            for attempt in range(4):
                if time.time() - start_time > budget_s * 0.95:
                    break
                noise_scale = 0.005 + 0.01 * attempt
                perturb = best_h + rng.randn(n) * noise_scale
                perturb = project_feasible(perturb, n)
                try:
                    res = minimize(
                        obj_func, perturb,
                        method='SLSQP',
                        args=(n, tau),
                        bounds=bounds,
                        constraints=cons,
                        options={'maxiter': 120, 'ftol': 1e-14}
                    )
                    h_pol = project_feasible(res.x, n)
                    c5 = evaluate_c5(h_pol)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_pol
                except Exception:
                    pass

    if best_h is None:
        if parent_h is not None:
            best_h = parent_h.copy()
            best_c5 = evaluate_c5(best_h)
        else:
            n = 47
            best_h = project_feasible(np.full(n, 0.5), n)
            best_c5 = evaluate_c5(best_h)

    return best_h, best_c5, len(best_h)
