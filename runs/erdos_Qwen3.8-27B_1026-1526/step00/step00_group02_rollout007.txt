def project_onto_simplex(h, target):

        if total_room < 1e-12:

        if total < 1e-12:

        scale = min(-diff / total, 1.0)
        h = h * (1.0 - scale)
    h = np.clip(h, 0.0, 1.0)

        h += diff / n

from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    w = 1.0 - h
    corr = np.correlate(h, w, mode='full')
    return float(np.max(corr)) * dx

def compute_c5_components(h, dx):
    """Return all correlation values (before max)"""
    w = 1.0 - h
    corr = np.correlate(h, w, mode='full')
    return corr * dx

def smooth_max(c, T=10.0):
    """Smooth approximation of max"""
    c_max = np.max(c)
    c_shifted = c - c_max
    return c_max + (1.0/T) * np.log(np.sum(np.exp(T * c_shifted)))

def project(h, target):
    n = len(h)
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        total_room = room.sum()
        if total_room < 1e-14:
            return h
        frac = min(diff / total_room, 1.0)
        h += frac * room
    else:
        total = h.sum()
        if total < 1e-14:
            return h
        frac = min(-diff / total, 1.0)
        h -= frac * h
    h = np.clip(h, 0.0, 1.0)
    # Fine adjustment
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        uniform = h <= 1.0 - 1e-10
        if diff > 0 and np.any(h < 1.0):
            idx = np.where(h < 1.0)[0]
            per = diff / len(idx)
            for i in idx:
                h[i] = min(1.0, h[i] + per)
        elif diff < 0 and np.any(h > 0.0):
            idx = np.where(h > 0.0)[0]
            per = -diff / len(idx)
            for i in idx:
                h[i] = max(0.0, h[i] - per)
        h = np.clip(h, 0.0, 1.0)
    return h

def nelder_mead_optimize(h0, dx, target, max_iter=500, T_smooth=50.0):
    def objective(x):
        x = np.clip(x, 0.0, 1.0)
        # Soft constraint on sum
        s = np.sum(x)
        penalty = 100.0 * (s - target) ** 2
        c = compute_c5_components(x, dx)
        return smooth_max(c, T_smooth) + penalty
    
    result = minimize(objective, h0, method='Nelder-Mead',
                      options={'maxiter': max_iter, 'xatol': 1e-8, 'fatol': 1e-10,
                               'adaptive': True})
    h_out = project(result.x, target)
    return h_out, compute_c5(h_out, dx)

def coordinate_descent(h0, dx, target, n_steps=20000, t0=None, budget=None):
    n = len(h0)
    h = h0.copy()
    best_val = compute_c5(h, dx)

        if t0 is not None and budget is not None:

                break
        i = np.random.randint(n)
        # Try to adjust h[i]
        old = h[i]
        # Try moving toward 0 or 1
        for delta in [0.1, 0.01, 0.5, 0.05]:
            for direction in [1, -1]:
                h_new = h.copy()
                h_new[i] = old + direction * delta
                h_new = project(h_new, target)
                val = compute_c5(h_new, dx)
                if val < best_val:
                    h = h_new
                    best_val = val
                    break
            else:
                continue
            break
    return h, best_val

def simulated_annealing(h0, dx, target, n_steps=10000, T_init=0.05, T_min=0.001, t0=None, budget=None):
    n = len(h0)
    h = h0.copy()
    val = compute_c5(h, dx)
    T = T_init

        if delta < 0 or np.random.random() < np.exp(-delta / T):

def make_binary_start(n, ones_count, half_pos, half_val=0.5):

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    w = 1.0 - h
    corr = np.correlate(h, w, mode='full')
    return float(np.max(corr)) * dx

def compute_c5_components(h, dx):
    w = 1.0 - h
    corr = np.correlate(h, w, mode='full')
    return corr * dx

def smooth_max(c, T):
    c_shifted = c - np.max(c)
    exp_c = np.exp(T * c_shifted)
    return np.max(c) + (1.0 / T) * np.log(np.sum(exp_c))

def project(h, target):
    n = len(h)
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        total_room = room.sum()
        if total_room < 1e-14:
            return h
        frac = min(diff / total_room, 1.0)
        h = h + frac * room
    else:
        total = h.sum()
        if total < 1e-14:
            return h
        frac = min(-diff / total, 1.0)
        h = h * (1.0 - frac)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            candidates = np.where(h < 1.0 - 1e-12)[0]
        else:
            candidates = np.where(h > 1e-12)[0]
        if len(candidates) > 0:
            per = diff / len(candidates)
            for i in candidates:
                h[i] += per
            h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            idx = np.argmax(1.0 - h)
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = np.argmax(h)
            h[idx] = max(0.0, h[idx] + diff)
    return h

def nm_objective(x, dx, target, T):
    x = np.clip(x, 0.0, 1.0)
    s = np.sum(x)
    penalty = 500.0 * (s - target) ** 2
    c = compute_c5_components(x, dx)
    return smooth_max(c, T) + penalty

def nm_optimize(h0, dx, target, max_iter=400, T=30.0):
    h0_clipped = np.clip(h0, 1e-6, 1.0 - 1e-6)
    result = minimize(nm_objective, h0_clipped, args=(dx, target, T),
                      method='Nelder-Mead',
                      options={'maxiter': max_iter, 'xatol': 1e-8, 'fatol': 1e-12,
                               'adaptive': True, 'maxfev': max_iter * 3})
    h_out = project(result.x, target)
    return h_out, compute_c5(h_out, dx)

def bit_flip_search(h0, dx, target, n_rounds=5000, t0=None, budget=None):
    n = len(h0)
    h = h0.copy()
    best_val = compute_c5(h, dx)
    for rnd in range(n_rounds):
        if t0 is not None and budget is not None:
            if time.time() - t0 > budget * 0.85:
                break
        i = np.random.randint(n)
        old = h[i]
        if old < 0.01 or old > 0.99:
            for target_val in [1.0, 0.0, 0.5]:
                if abs(target_val - old) < 1e-6:
                    continue
                h_new = h.copy()
                h_new[i] = target_val
                h_new = project(h_new, target)
                val = compute_c5(h_new, dx)
                if val < best_val - 1e-10:
                    h = h_new
                    best_val = val
                    break
        else:
            for target_val in [1.0, 0.0, 0.5]:
                h_new = h.copy()
                h_new[i] = target_val
                h_new = project(h_new, target)
                val = compute_c5(h_new, dx)
                if val < best_val - 1e-10:
                    h = h_new
                    best_val = val
                    break
    return h, best_val

def coordinate_descent(h0, dx, target, n_steps=15000, t0=None, budget=None):
    n = len(h0)
    h = h0.copy()
    best_val = compute_c5(h, dx)
    np.random.seed(int(time.time() * 1000) % 100000)
    for step in range(n_steps):
        if t0 is not None and budget is not None:
            if time.time() - t0 > budget * 0.85:
                break
        i = np.random.randint(n)
        old = h[i]
        improved = False
        for delta in [0.05, 0.01, 0.2, 0.005, 0.5]:
            for direction in [1.0, -1.0]:
                h_new = h.copy()
                h_new[i] = old + direction * delta
                h_new = project(h_new, target)
                if np.array_equal(h_new, h):
                    continue
                val = compute_c5(h_new, dx)
                if val < best_val - 1e-10:
                    h = h_new
                    best_val = val
                    improved = True
                    break
            if improved:
                break
    return h, best_val
