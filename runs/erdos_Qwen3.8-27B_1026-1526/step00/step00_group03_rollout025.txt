    lo = -1.0
    hi = 2.0

        mid = (lo + hi) / 2
        projected = np.clip(h - mid, 0, 1)

            lo = mid
        else:
            hi = mid

import numpy as np
import time

def project_feasible(h, target_sum):
    lo = -2.0
    hi = 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) * 0.5
    h_proj = np.clip(h - tau, 0.0, 1.0)
    diff = target_sum - np.sum(h_proj)
    if abs(diff) > 1e-10:
        free = (h_proj > 1e-8) & (h_proj < 1.0 - 1e-8)
        if np.any(free):
            h_proj[free] += diff / np.sum(free)
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def coord_pass(h, target_sum, grid_size, t_deadline):
    n = len(h)
    best_obj = compute_c5(h)
    improved = False
    for i in range(n):
        if time.time() > t_deadline:
            break
        old_val = h[i]
        cands = np.linspace(0.0, 1.0, grid_size)
        bl_obj = best_obj
        bl_h = None
        for v in cands:
            if abs(v - old_val) < 1e-5:
                continue
            h2 = h.copy()
            h2[i] = v
            h2 = project_feasible(h2, target_sum)
            obj = compute_c5(h2)
            if obj < bl_obj - 1e-10:
                bl_obj = obj
                bl_h = h2
        if bl_h is not None:
            h = bl_h
            if bl_obj < best_obj:
                best_obj = bl_obj
                improved = True
    return h, best_obj, improved

def local_refine(h, target_sum, grid_size, t_deadline, radius):
    n = len(h)
    best_obj = compute_c5(h)
    for i in range(n):
        if time.time() > t_deadline:
            break
        old_val = h[i]
        lo = max(0.0, old_val - radius)

        cands = np.linspace(lo, hi, grid_size)
        bl_obj = best_obj
        bl_h = None
        for v in cands:
            if abs(v - old_val) < 1e-6:

            h2 = h.copy()
            h2[i] = v
            h2 = project_feasible(h2, target_sum)
            obj = compute_c5(h2)
            if obj < bl_obj - 1e-11:
                bl_obj = obj
                bl_h = h2
        if bl_h is not None:
            h = bl_h
            if bl_obj < best_obj:
                best_obj = bl_obj
    return h, best_obj
