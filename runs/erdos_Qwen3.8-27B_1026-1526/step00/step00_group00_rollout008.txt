    val = (cmax + np.log(sum_exp) / alpha) * dx
    grad = np.zeros(n)
    for p in range(2 * n - 1):
        if w[p] < 1e-10:
            continue
        offset = p - (n - 1)
        wp = w[p]
        if offset >= 0:
            j1 = np.arange(offset, n)
            grad[j1] += wp * (1.0 - h[j1 - offset])
            j2 = np.arange(0, n - offset)
            grad[j2] -= wp * h[j2 + offset]
        else:
            pa = -offset
            j1 = np.arange(0, n - pa)
            grad[j1] += wp * (1.0 - h[j1 + pa])
            j2 = np.arange(pa, n)
            grad[j2] -= wp * h[j2 - pa]
    return val, grad * dx

def constraint_eq(h, n):
    return np.sum(h) - n / 2.0

def constraint_eq_jac(h, n):
    return np.ones(n)

def local_search_swap(h, n, target_sum, n_iters, time_limit):
    t_start = time.time()
    best_c5 = compute_c5(h)
    h_curr = h.copy()
    no_improve = 0
    
    for it in range(n_iters):
        if time.time() - t_start > time_limit:
            break
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        # Determine max epsilon
        eps_max_fwd = min(h_curr[i], 1.0 - h_curr[j])
        eps_max_rev = min(h_curr[j], 1.0 - h_curr[i])
        if eps_max_fwd < 1e-8 and eps_max_rev < 1e-8:
            continue
        eps = max(eps_max_fwd, eps_max_rev) * (0.1 + 0.9 * np.random.rand())
        if eps < 1e-8:
            continue
        
        direction = np.random.rand()
        if direction < 0.5:
            if eps > eps_max_fwd:
                continue
            h_trial = h_curr.copy()
            h_trial[i] -= eps
            h_trial[j] += eps
        else:
            if eps > eps_max_rev:
                continue
            h_trial = h_curr.copy()
            h_trial[j] -= eps
            h_trial[i] += eps
        
        c5_trial = compute_c5(h_trial)
        if c5_trial < best_c5 - 1e-12:
            h_curr = h_trial
            best_c5 = c5_trial
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                # Perturb
                h_curr = h_curr + 0.02 * np.random.randn(n)
                h_curr = np.clip(h_curr, 0, 1)
                diff = target_sum - np.sum(h_curr)
                h_curr += diff / n
                h_curr = np.clip(h_curr, 0, 1)
                best_c5 = compute_c5(h_curr)
                no_improve = 0
    
    return h_curr, best_c5
