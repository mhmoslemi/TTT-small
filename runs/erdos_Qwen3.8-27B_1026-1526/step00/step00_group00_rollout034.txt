To minimize over t ∈ [0,1]:
Let a[m] = alpha[m] + lam, b[m] = gamma[m]
Minimize max_m (a[m]*t + b[m]) over t ∈ [0,1]

This is a convex piecewise-linear function. The minimum is found by:
- If min(a) ≥ 0: minimum at t=0
- If max(a) ≤ 0: minimum at t=1

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, n_points):
    dx = 2.0 / n_points
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return np.max(corr) * dx


def soft_max_c5(x, n_points, beta):
    dx = 2.0 / n_points
    one_minus_h = 1.0 - x
    corr = np.correlate(x, one_minus_h, mode='full') * dx
    max_val = np.max(corr)
    return max_val + np.log(np.sum(np.exp(corr - max_val))) / beta


class SmoothObjective:
    def __init__(self, n_pts, beta):
        self.n_pts = n_pts
        self.beta = beta
    def __call__(self, x):
        return soft_max_c5(x, self.n_pts, self.beta)


class DirectObjective:
    def __init__(self, n_pts):
        self.n_pts = n_pts
    def __call__(self, x):
        return compute_c5(x, self.n_pts)


class SumConstraintFun:
    def __init__(self, target_sum):
        self.target_sum = target_sum
    def fun(self, x):
        return np.sum(x) - self.target_sum
    def jac(self, x):
        return np.ones_like(x)


def project_to_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - h
            total_room = np.sum(room)
            if total_room > 1e-15:
                h = h + diff * room / total_room
        else:
            room = h
            total_room = np.sum(room)
            if total_room > 1e-15:
                h = h + diff * room / total_room
    return np.clip(h, 0.0, 1.0)


def solve_coord_problems(h, n, lam, max_sweeps=300, tol=1e-12):
    c_len = 2 * n - 1
    k_shift = np.arange(c_len) - (n - 1)
    one_minus_h = 1.0 - h
    c = np.correlate(h, one_minus_h, mode='full')
    j_arr = np.arange(n)
    for sweep in range(max_sweeps):
        max_change = 0.0
        order = np.random.permutation(n) if sweep > 0 else j_arr
        for j in order:
            alpha = np.zeros(c_len)
            v1 = (j - k_shift >= 0) & (j - k_shift < n)
            v2 = (j + k_shift >= 0) & (j + k_shift < n)
            if np.any(v1):
                idxs = j - k_shift[v1]
                alpha[v1] += 1.0 - h[idxs]
            if np.any(v2):
                idxs = j + k_shift[v2]
                alpha[v2] -= h[idxs]
            a = alpha + lam
            b = c - h[j] * alpha
            amin = np.min(a)
            amax = np.max(a)
            if amin >= -1e-14:
                t_opt = 0.0
            elif amax <= 1e-14:
                t_opt = 1.0
            else:
                t_lo = 0.0
                t_hi = 1.0
                for _ in range(50):
                    t_mid = 0.5 * (t_lo + t_hi)
                    vals = a * t_mid + b
                    max_val = np.max(vals)
                    active = vals >= max_val - 1e-13
                    if np.min(a[active]) > 1e-10:
                        t_hi = t_mid
                    elif np.max(a[active]) < -1e-10:
                        t_lo = t_mid
                    else:
                        break
                t_opt = 0.5 * (t_lo + t_hi)
            delta = t_opt - h[j]
            if abs(delta) > 1e-15:
                c += delta * alpha
                h[j] = t_opt
                max_change = max(max_change, abs(delta))
        if max_change < tol:
            break
    return h


def bisection_lambda(h_init, n, target_sum, max_iter=40, tol=1e-10):
    lam_lo = -100.0
    lam_hi = 100.0
    h_lo = solve_coord_problems(h_init.copy(), n, lam_lo, max_sweeps=100)
    if np.sum(h_lo) <= target_sum:
        return h_lo, lam_lo
    h_hi = solve_coord_problems(h_init.copy(), n, lam_hi, max_sweeps=100)
    if np.sum(h_hi) >= target_sum:
        return h_hi, lam_hi
    for _ in range(max_iter):
        lam_mid = 0.5 * (lam_lo + lam_hi)
        h_mid = solve_coord_problems(h_init.copy(), n, lam_mid, max_sweeps=300)
        sum_mid = np.sum(h_mid)
        if abs(sum_mid - target_sum) < tol:
            return h_mid, lam_mid
        if sum_mid > target_sum:
            lam_lo = lam_mid
            h_lo = h_mid
        else:
            lam_hi = lam_mid
            h_hi = h_mid
    if abs(np.sum(h_lo) - target_sum) < abs(np.sum(h_hi) - target_sum):
        return h_lo, lam_lo
    return h_hi, lam_hi


def interp_to_new_n(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False) + 1.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return h_new


def run(seed=42, budget_s=1300, **kwargs):

    start_time = time.time()

    n = len(initial_h_values)
    h = np.array(initial_h_values, dtype=float).copy()

    best_n = n

        return time.time() - start_time

        c5_cd = compute_c5(h_cd, n)
        if c5_cd < best_c5:
            best_h = h_cd.copy()
            best_c5 = c5_cd

    # Phase 2: SLSQP refinement
    if elapsed() < budget_s * 0.5:
        bounds = [(0.0, 1.0)] * n
        scf = SumConstraintFun(target_sum)
        constraint = {'type': 'eq', 'fun': scf.fun, 'jac': scf.jac}
        for beta in [20, 50, 100, 200, 500, 1000]:
            if elapsed() > budget_s * 0.5:
                break
            obj = SmoothObjective(n, beta)
            try:
                result = minimize(obj, best_h, method='SLSQP', bounds=bounds,
                                  constraints=[constraint],
                                  options={'maxiter': 500, 'ftol': 1e-14})
            except Exception:
                continue
            if result.x is not None:
                h_test = project_to_feasible(result.x, target_sum)
                c5_test = compute_c5(h_test, n)
                if c5_test < best_c5:
                    best_h = h_test.copy()
                    best_c5 = c5_cd if False else c5_test

    # Direct non-smooth SLSQP
    if elapsed() < budget_s * 0.65:

            if elapsed() > budget_s * 0.65:
                break
            obj = DirectObjective(n)

            if result.x is not None:

                else:

    # Phase 3: Perturbation + coordinate descent to escape local minima
    if elapsed() < budget_s * 0.75:

            if elapsed() > budget_s * 0.75:
                break
            scale = max(0.05, 0.3 * (1.0 - attempt / 30.0))

            break
        target_sum_new = n_new / 2.0
        h_new = interp_to_new_n(best_h, best_n, n_new)
        h_new = project_to_feasible(h_new, target_sum_new)

        h_cd_new, _ = bisection_lambda(h_new, n_new, target_sum_new, max_iter=30)
        h_cd_new = project_to_feasible(h_cd_new, target_sum_new)
        c5_new = compute_c5(h_cd_new, n_new)

        if c5_new < best_c5:

            # Try SLSQP from interpolated

            scf_new = SumConstraintFun(target_sum_new)
            constraint_new = {'type': 'eq', 'fun': scf_new.fun, 'jac': scf_new.jac}
            for beta in [50, 100]:

                obj = SmoothObjective(n_new, beta)

                except Exception:
                    continue
                if result.x is not None:
                    h_test = project_to_feasible(result.x, target_sum_new)
                    c5_test = compute_c5(h_test, n_new)
                    if c5_test < best_c5:
                        best_h = h_test.copy()
                        best_c5 = c5_test

                    best_h = h_test.copy()
                    best_c5 = c5_test
