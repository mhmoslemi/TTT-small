def compute_corr(h):
    n = len(h)
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return corr

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    return np.max(corr) * dx

def project_feasible(h, S):
    # Project h onto {x: 0<=x<=1, sum(x)=S}
    # Using bisection on the shift
    lo, hi = -h.max(), 1.0 - h.min()
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(h + mid, 0.0, 1.0)
        s = proj.sum()
        if s < S:
            lo = mid
        else:
            hi = mid
    mid = (lo + hi) / 2.0
    return np.clip(h + mid, 0.0, 1.0)

def smooth_max_gradient(corr_vals, dx, beta):
    # Returns (smooth_max_value, gradient_weights)
    scaled = beta * corr_vals * dx
    # For numerical stability
    max_scaled = np.max(scaled)
    exps = np.exp(scaled - max_scaled)
    logsumexp = max_scaled + np.log(np.sum(exps))
    weights = exps / np.sum(exps)
    smooth_val = logsumexp / beta
    return smooth_val, weights

def grad_smooth_max(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    smooth_val, weights = smooth_max_gradient(corr, dx, beta)
    
    # Compute gradient
    # For each m (0 to 2n-2), k = m - (n-1)
    # ∂corr[m]/∂h[j] = (1-h[j-k])*I(0≤j-k<n) - h[j+k]*I(0≤j+k<n)
    grad = np.zeros(n)

        if weights[m] < 1e-10:
            continue
        k = m - (n - 1)

            g = 0.0
            if 0 <= j - k < n:
                g += (1.0 - h[j - k])
            if 0 <= j + k < n:
                g -= h[j + k]
            grad[j] += weights[m] * g * dx
    
    return smooth_val, grad

def compute_gradient(h, weights, dx):
    n = len(h)
    # weights has length 2*n-1
    grad = np.zeros(n)
    for m in range(2*n-1):
        if weights[m] < 1e-12:
            continue
        k = m - (n-1)
        w = weights[m] * dx
        # First term: (1-h[j-k]) for j where 0<=j-k<n
        j_start = max(0, k)
        j_end = min(n, n+k)  # exclusive
        if j_start < j_end:
            idx = np.arange(j_start, j_end)
            grad[idx] += w * (1.0 - h[idx - k])
        # Second term: -h[j+k] for j where 0<=j+k<n
        j_start2 = max(0, -k)
        j_end2 = min(n, n-k)  # exclusive
        if j_start2 < j_end2:
            idx2 = np.arange(j_start2, j_end2)
            grad[idx2] -= w * h[idx2 + k]
    return grad

def compute_corr(h):

def softmax_weights(corr, dx, beta):
    v = beta * corr * dx
    v = v - v.max()
    e = np.exp(v)
    total = e.sum()
    if total < 1e-300:
        return np.ones(len(corr)) / len(corr)
    return e / total


def compute_grad(h, weights, dx):
    n = len(h)

        je2 = min(n, n - k)
        if js2 < je2:
            idx2 = np.arange(js2, je2)
            grad[idx2] -= w * h[idx2 + k]
    return grad
