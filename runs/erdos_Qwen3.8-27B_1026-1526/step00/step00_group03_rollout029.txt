```python
def directed_step(h, n, step_size):
    """Take one directed step to reduce the active constraint"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = np.argmax(corr)
    
    # Compute gradient
    grad = np.zeros(n)
    for i in range(n):
        val = 0.0
        nk = i - k_star  # from the h[n+k] term (n = i-k)
        if 0 <= nk < n:
            val += (1.0 - h[nk])
        pk = i + k_star  # from the (1-h[n]) term (n = i, so h[n+k] = h[i+k])
        if 0 <= pk < n:
            val -= h[pk]
        grad[i] = val
    
    # Best pairwise move: increase h[argmin(grad)], decrease h[argmax(grad)]
    i_add = np.argmin(grad)
    i_sub = np.argmax(grad)
    
    if i_add == i_sub:
        return h.copy()
    
    # Try moving mass from i_sub to i_add
    max_move = min(h[i_sub], 1.0 - h[i_add])
    d = min(step_size, max_move)
    
    if d <= 0:
        return h.copy()
    
    new_h = h.copy()
    new_h[i_add] += d
    new_h[i_sub] -= d
    
    return new_h
```

```python
def repair(h, n):
    h = np.asarray(h, dtype=np.float64).copy()
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-12
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_grad_for_k(h, n, k):
    """Compute gradient of corr[k] w.r.t. h"""
    grad = np.zeros(n)
    for i in range(n):
        val = 0.0
        nk = i - k
        if 0 <= nk < n:
            val += (1.0 - h[nk])
        pk = i + k
        if 0 <= pk < n:
            val -= h[pk]
        grad[i] = val
    return grad

def vectorized_grad_for_k(h, n, k):
    """Compute gradient of corr[k] w.r.t. h (vectorized)"""
    grad = np.zeros(n)
    if k >= 0:
        # nk = i - k is valid for i >= k
        valid_nk = np.arange(k, n)
        grad[valid_nk] += (1.0 - h[valid_nk - k])
        # pk = i + k is valid for i < n - k
        valid_pk = np.arange(0, n - k)
        grad[valid_pk] -= h[valid_pk + k]
    else:
        m = -k
        # nk = i - k = i + m is valid for i < n - m
        valid_nk = np.arange(0, n - m)
        grad[valid_nk] += (1.0 - h[valid_nk + m])
        # pk = i + k = i - m is valid for i >= m
        valid_pk = np.arange(m, n)
        grad[valid_pk] -= h[valid_pk - m]
    return grad
```
