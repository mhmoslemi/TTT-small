import numpy as np
import time

def compute_c5(h, dx):
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    return float(np.max(corr) * dx)

def compute_top_shifts(h, k=10):
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    n = len(h)
    top_idx = np.argsort(corr)[-k:]
    top_shifts = top_idx - (n - 1)
    return top_shifts, corr

def grad_at_shift(h, s):
    """Compute gradient of overlap(s) w.r.t. h."""
    n = len(h)
    grad = np.zeros(n)
    n_min = max(0, -s)
    n_max = min(n - 1, n - 1 - s)
    for j in range(n):
        val = 0.0
        # Check if j-s is in [n_min, n_max] (contribution from h[n+s] term where n=j-s)
        if n_min <= j - s <= n_max:
            val += 1.0
            val -= h[j - s]
        # Check if j is in [n_min, n_max] (contribution from (1-h[n]) term where n=j)
        if n_min <= j <= n_max:
            js = j + s
            if 0 <= js <= n - 1:
                val -= h[js]
        grad[j] = val
    return grad

def project_feasible(h, target_sum):
    """Project h onto {0<=x<=1, sum(x)=target_sum}."""
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
    h = np.clip(h, 0.0, 1.0)
    # Exact sum fix
    diff = target_sum - h.sum()
    if abs(diff) > 1e-15:
        idx = np.argsort(h)[::-1] if diff > 0 else np.argsort(h)
        for i in idx:
            if diff > 0:
                adj = min(diff, 1.0 - h[i])
            else:
                adj = max(diff, -h[i])
            if adj == 0:
                continue
            h[i] += adj
            diff -= adj
            if abs(diff) < 1e-15:
                break
    return h

def subgradient_optimize(h0, dx, n_iter=5000, top_k=5, alpha0=0.1, lr_decay=0.999):
    """Subgradient descent with top-k shifts."""
    h = h0.copy()
    n = len(h)
    target_sum = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    alpha = alpha0
    
    for t in range(n_iter):
        # Compute top shifts
        g = 1.0 - h
        corr = np.correlate(h, g, mode='full')
        top_idx = np.argsort(corr)[-top_k:]
        top_shifts = top_idx - (n - 1)
        
        # Weighted gradient
        top_corr = corr[top_idx]
        # Use softmax weights
        max_c = np.max(top_corr)
        weights = np.exp(top_corr - max_c)
        weights /= weights.sum()
        
        grad = np.zeros(n)
        for w, s in zip(weights, top_shifts):
            if w < 1e-10:
                continue
            grad += w * grad_at_shift(h, s)
        
        # Update
        h_new = h - alpha * grad
        h_new = project_feasible(h_new, target_sum)
        
        c5 = compute_c5(h_new, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        
        h = h_new
        alpha *= lr_decay
        if alpha < 1e-6:
            alpha = 1e-6
    
    return best_h, best_c5

    # I1[j] = 1 if n_min <= j-s <= n_max
    # i.e., n_min + s <= j <= n_max + s
    j1_min = n_min + s
    j1_max = n_max + s
    I1 = np.zeros(n)
    if j1_min <= j1_max:

        hi = min(n - 1, j1_max)
        I1[lo:hi+1] = 1.0

    I2 = np.zeros(n)
    if n_min <= n_max:
        I2[n_min:n_max+1] = 1.0

    if s >= 0:

        js = np.arange(0, n - s)  # j+s values
        # h[j+s] for j in [0, n-1-s]
        # We want: for j where I2[j]=1, subtract h[j+s]
        valid_j = np.arange(n_min, n_max + 1)
        # j+s for these j
        indices = valid_j + s
        indices = indices[(indices >= 0) & (indices < n)]
        valid_j = valid_j[(valid_j + s >= 0) & (valid_j + s < n)]
        grad[valid_j] -= h[valid_j + s]
    else:
        # j+s = j-|s| is valid for j in [|s|, n-1]
        valid_j = np.arange(n_min, n_max + 1)
        indices = valid_j + s
        valid_mask = (indices >= 0) & (indices < n)
        valid_j = valid_j[valid_mask]
        grad[valid_j] -= h[valid_j + s]
    
    # Subtract h[j-s] where I1[j]=1
    # j-s valid: 0 <= j-s <= n-1, i.e., s <= j <= n-1+s
    valid_j1 = np.where(I1 > 0)[0]
    js_minus = valid_j1 - s
    valid_mask = (js_minus >= 0) & (js_minus < n)
    valid_j1 = valid_j1[valid_mask]
    grad[valid_j1] -= h[valid_j1 - s]
    
    return grad
