from scipy.special import logsumexp

def evaluate(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n_points):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n_points / 2.0
    s = np.sum(h)
    if s < 1e-15:
        h[:] = 1.0 * target / n_points
        return h
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        idx = np.arange(n_points)
        np.random.shuffle(idx)
        for i in idx:
            if abs(diff) < 1e-12:
                break
            if diff > 0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            else:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
    return h

import numpy as np
import time
from scipy.optimize import minimize
from scipy.special import logsumexp

def evaluate(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def repair(h, n_points):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n_points / 2.0
    s = np.sum(h)
    if s < 1e-15:
        h[:] = target / n_points
        return h
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        for _ in range(n_points * 3):
            if abs(diff) < 1e-12:
                break
            if diff > 0:
                mask = h < 1.0
                idx = np.where(mask)[0]
                if len(idx) == 0:
                    break
                i = idx[0]
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            else:
                mask = h > 0.0
                idx = np.where(mask)[0]
                if len(idx) == 0:
                    break
                i = idx[0]
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
    return h

def coordinate_descent(h, n_points, max_sweeps=100, n_cands=101):
    n = n_points
    dx = 2.0 / n
    candidates = np.linspace(0.0, 1.0, n_cands)
    
    for sweep in range(max_sweeps):
        corr = np.correlate(h, 1.0 - h, mode='full')
        any_improved = False
        
        for j in range(n):
            old_val = h[j]
            
            # Compute slopes
            slopes = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                m = k - (n - 1)
                slope = 0.0
                jm = j - m
                if 0 <= jm < n:
                    slope += (1.0 - h[jm])
                jp = j + m
                if 0 <= jp < n:
                    slope += (-h[jp])
                slopes[k] = slope
            
            # Vectorized search over candidates
            new_corrs = corr[None, :] + (candidates[:, None] - old_val) * slopes[None, :]
            c5s = np.max(new_corrs, axis=1) * dx
            best_idx = int(np.argmin(c5s))
            best_val = candidates[best_idx]
            
            h[j] = best_val
            corr = corr + (best_val - old_val) * slopes
            if abs(best_val - old_val) > 1e-8:
                any_improved = True
        
        if not any_improved:
            break
    
    return h

def smooth_objective(h, n_points, beta):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    shifted = beta * corr - np.max(beta * corr)
    return float(np.log(np.sum(np.exp(shifted))) + np.max(beta * corr)) / beta
