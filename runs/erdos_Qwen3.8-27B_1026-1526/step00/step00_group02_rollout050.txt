import numpy as np
from scipy.optimize import minimize
import time

_target_val = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_obj(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    return m + (1.0 / tau) * np.log(np.sum(np.exp(tau * (corr - m))))

def sum_eq(x):
    return np.sum(x) - _target_val

def project_sum(h, target):
    h = h.astype(float).copy()
    diff = target - np.sum(h)
    if abs(diff) < 1e-15:
        return h
    n = len(h)
    h += diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(free):
            h[free] += diff / float(np.sum(free))
            h = np.clip(h, 0.0, 1.0)
    return h

def coord_descent_polish(h, target, max_sweeps=30):
    n = len(h)
    h = h.copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    for _ in range(max_sweeps):
        improved = False
        for i in range(n):
            h0 = h.copy(); h0[i] = 0.0
            c0 = np.correlate(h0, 1.0 - h0, mode='full')
            h1 = h.copy(); h1[i] = 1.0
            c1 = np.correlate(h1, 1.0 - h1, mode='full')
            a = c1 - c0
            b = c0
            xg = np.linspace(0.0, 1.0, 501)
            vals = np.max(b[:, None] + a[:, None] * xg[None, :], axis=0)
            idx = np.argmin(vals)
            x_new = xg[idx]
            val_new = vals[idx]
            # Check if improvement is significant
            cur_val = np.max(b + a * h[i])
            if val_new < cur_val - 1e-13 / dx:
                h[i] = x_new
                h = project_sum(h, target)
                c5_now = compute_c5(h)
                if c5_now < best_c5:
                    best_c5 = c5_now
                    improved = True
        if not improved:
            break
    return h, best_c5

def slsqp_optimize(h0, n, max_iter=5000):
    global _target_val
    target = n / 2.0
    _target_val = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq}]
    
    # Try soft objective first
    for tau in [200.0, 500.0]:
        try:
            res = minimize(soft_obj, h0, args=(tau,), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': max_iter, 'ftol': 1e-15})
            h0 = np.clip(res.x, 0.0, 1.0)
            h0 = project_sum(h0, target)
        except Exception:
            pass

        res = minimize(compute_c5, h0, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': max_iter, 'ftol': 1e-15})
        h0 = np.clip(res.x, 0.0, 1.0)
        h0 = project_sum(h0, target)
    except Exception:
        pass
    
    return h0
