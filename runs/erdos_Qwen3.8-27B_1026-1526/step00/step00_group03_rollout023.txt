    scaled = tau * x
    m = np.max(scaled)
    return (m + np.log(np.sum(np.exp(scaled - m)))) / tau

    h = h.reshape(-1)

    c5 = smooth_max(corr * dx, tau)
    penalty = lam * (np.sum(h) - n_points / 2.0) ** 2

def grad_func(h, n_points, tau, lam):
    # Numerical gradient (central difference)
    h = h.reshape(-1)
    eps = 1e-7
    grad = np.zeros_like(h)
    f0 = obj_func(h, n_points, tau, lam)
    for i in range(len(h)):
        h[i] += eps
        fp = obj_func(h, n_points, tau, lam)
        h[i] -= 2 * eps
        fm = obj_func(h, n_points, tau, lam)
        h[i] += eps
