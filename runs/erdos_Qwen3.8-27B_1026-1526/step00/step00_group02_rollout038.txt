from scipy.optimize import minimize, linprog
import time
import math

def project_to_simplex(h, target_sum):
    """Project h onto {0 <= h <= 1, sum(h) = target_sum}"""
    h = h.copy()
    h = np.clip(h, 0, 1)
    # If sum is too high, reduce
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    # Distribute the difference
    if diff > 0:
        # Need to increase sum by diff
        room = 1.0 - h
        # Add to entries with most room
        order = np.argsort(-room)
        for i in order:
            add = min(diff, room[i])
            h[i] += add
            diff -= add
            if diff <= 1e-10:
                break
    else:
        # Need to decrease sum by -diff
        order = np.argsort(h)
        for i in order:
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
            if diff >= -1e-10:
                break
    return np.clip(h, 0, 1)

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

from scipy.optimize import minimize
import time

def project_to_constraint(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        order = np.argsort(-room)
        for idx in order:
            add = min(diff, room[idx])
            h[idx] += add
            diff -= add
            if diff <= 1e-14:
                break
    else:
        order = np.argsort(h)
        for idx in order:
            sub = min(-diff, h[idx])
            h[idx] -= sub
            diff += sub
            if diff >= -1e-14:
                break

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def smooth_max_obj(corr, alpha):
    m = np.max(corr)
    if alpha * (m - m) > 500:  # overflow protection
        return m
    vals = alpha * (corr - m)
    return m + np.log(np.sum(np.exp(vals))) / alpha

    vals = alpha * (corr - m)

import numpy as np
from scipy.optimize import minimize
import time

def project_to_constraint(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        order = np.argsort(-room)
        for idx in order:
            add = min(diff, room[idx])
            h[idx] += add
            diff -= add
            if diff <= 1e-14:
                break
    else:
        order = np.argsort(h)
        for idx in order:
            sub = min(-diff, h[idx])
            h[idx] -= sub
            diff += sub
            if diff >= -1e-14:
                break
    return np.clip(h, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def smooth_max_grad(corr, h, alpha, dx, n):
    m = np.max(corr)
    vals = alpha * (corr - m)
    vals = np.clip(vals, -500, 500)
    weights = np.exp(vals)
    weights /= np.sum(weights)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        lag = abs(k - (n - 1))
        w = weights[k] * dx
        if w < 1e-12:
            continue
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        else:
            term1 = np.zeros(n)
            term1[lag:] = 1.0 - h[:n - lag]
            term2 = np.zeros(n)
            if n - lag > 0:
                term2[:n - lag] = -h[lag:]
            grad += w * (term1 + term2)
    return grad

    n = len(h)

    best_c5 = compute_c5(best_h)

            corr = np.correlate(h_curr, 1.0 - h_curr, mode="full") * dx
            grad = smooth_max_grad(corr, h_curr, alpha, dx, n)
            penalty = 50.0
            sum_viol = np.sum(h_curr) - target_sum
            grad += 2.0 * penalty * sum_viol
            gn = np.max(np.abs(grad))

            lr = min(0.1 / gn, 0.3)
            h_new = h_curr - lr * grad
            h_new = project_to_constraint(h_new, target_sum)
