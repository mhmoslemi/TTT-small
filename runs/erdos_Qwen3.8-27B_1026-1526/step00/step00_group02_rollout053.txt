```python
def coord_minimize(h, j, n, dx):
    """Minimize the objective with respect to h[j], keeping other coordinates fixed."""
    v = 1 - h
    # Compute A_k and B_k for each lag k (0 to 2n-2 in the full correlation array)
    # c[k] for k in [0, 2n-2] corresponds to lag = k - (n-1)
    
    A = np.zeros(2*n-1)
    B = np.zeros(2*n-1)
    
    for k_idx in range(2*n-1):
        # Compute c[k_idx] as A[k_idx] * h[j] + B[k_idx]
        # c[k_idx] = sum over valid n of h[n + (k_idx-(n-1))] * (1-h[n])
        lag = k_idx - (n-1)
        # n ranges from max(0, -lag) to min(n-1, n-1-lag)
        n_lo = max(0, -lag)
        n_hi = min(n-1, n-1-lag)
        b_val = 0.0
        a_val = 0.0
        for n_idx in range(n_lo, n_hi + 1):
            h_idx1 = n_idx + lag  # index into h (first factor)
            h_idx2 = n_idx  # index into 1-h (second factor)
            term = h[h_idx1] * v[h_idx2]
            if h_idx1 == j:
                a_val += v[h_idx2]
                # Don't add to b_val
            elif h_idx2 == j:
                a_val -= h[h_idx1]
                # Don't add to b_val
            else:
                b_val += term
        A[k_idx] = a_val
        B[k_idx] = b_val
    
    # Now minimize max_k (A[k]*t + B[k]) over t in [0,1]
    # This is a convex piecewise-linear function
    # Find all breakpoints in [0,1]
    candidates = [0.0, 1.0]
    for i in range(2*n-1):
        for k in range(i+1, 2*n-1):
            denom = A[i] - A[k]
            if abs(denom) > 1e-12:
                t = (B[k] - B[i]) / denom
                if 0.0 <= t <= 1.0:
                    candidates.append(t)
    
    best_val = np.inf
    best_t = 0.0
    for t in candidates:
        val = np.max(A * t + B)
        if val < best_val:
            best_val = val
            best_t = t
    
    h[j] = best_t
    return h
```

```python
            break
```

```python
            # Need to increase sum

            # Need to decrease sum
            capacities = h.copy()
        total_cap = np.sum(capacities)

            h += np.sign(diff) * capacities * (abs(diff) / total_cap)
        else:
            # Distribute proportionally
            h += diff * capacities / total_cap
        h = np.clip(h, 0, 1)
```

```python
def optimize_coordinate(h, j, n, dx, lam=100.0):
    n_pts = n
    v = 1.0 - h
    S = np.sum(h) - h[j]
    target = n_pts / 2.0
    n_corr = 2 * n_pts - 1
    k0 = n_pts - 1
    
    # Precompute A and B
    A = np.zeros(n_corr)
    B = np.zeros(n_corr)
    for ki in range(n_corr):
        lag = ki - (n_pts - 1)
        lo = max(0, -lag)
        hi = min(n_pts - 1, n_pts - 1 - lag)
        for ni in range(lo, hi + 1):
            i1 = ni + lag
            i2 = ni
            if i1 == j:
                if i2 == j:
                    A[ki] += 1.0
                    B[ki] += 0.0
                    # -t² term handled separately
                else:
                    A[ki] += v[i2]
            elif i2 == j:
                A[ki] += -h[i1]
            else:
                B[ki] += h[i1] * v[i2]
    
    def phi(t):
        corr = A * t + B
        corr[k0] -= t * t
        return np.max(corr) * dx + lam * (S + t - target) ** 2
    
    # Coarse grid
    ts = np.linspace(0, 1, 200)
    vals = np.array([phi(t) for t in ts])
    i_best = np.argmin(vals)
    
    # Fine grid around best
    if i_best == 0:
        fine = np.linspace(0, ts[1], 200)
    elif i_best == len(ts) - 1:
        fine = np.linspace(ts[-2], 1, 200)
    else:
        fine = np.linspace(ts[i_best-1], ts[i_best+1], 200)
    
    fine_vals = np.array([phi(t) for t in fine])
    i_fine = np.argmin(fine_vals)
    
    return fine[i_fine]
```

```python
def phi_vec(ts_grid, A, B, k0, dx, S, target, lam):
    # ts_grid: array of t values
    # Returns array of phi(t) values
    corr = np.outer(ts_grid, A) + B  # shape (len(ts_grid), n_corr)
    corr[:, k0] -= ts_grid**2
    max_corr = np.max(corr, axis=1)
    return max_corr * dx + lam * (S + ts_grid - target)**2
```
