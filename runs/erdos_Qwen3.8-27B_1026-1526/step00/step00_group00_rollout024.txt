Change when h_b: 0→1: Δ = 1*(1-h_{b-k}) - 0 = 1 - h_{b-k}

Term involving position b as the "second" index (h_i where i=b):
term_b_second = h_{b+k} (1 - h_b)
Change when h_b: 0→1: Δ = h_{b+k}(1-1) - h_{b+k}(1-0) = -h_{b+k}

Total change for h_b: 0→1: Δ_b(k) = (1 - h_{b-k}) + (-h_{b+k}) = 1 - h_{b-k} - h_{b+k}

Total change for swap (h_a: 1→0, h_b: 0→1):
Δc(k) = h_{a-k} - h_{a+k} - 1 + 1 - h_{b-k} - h_{b+k}
= h_{a-k} - h_{a+k} - h_{b-k} - h_{b+k}

All indices are taken mod n? No, they're just regular indices, and out-of-range terms are 0.

This is O(1) per lag to compute the change, so O(n) per swap evaluation. With n=45, that's very fast.

So the algorithm for binary local search:
1. Initialize h (binary, n/2 ones)
2. Compute c(k) for all k
3. Find k_max = argmax c(k)

5. Repeat until no improving swap is found

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_values(h, dx):
    """Compute all correlation values and return (c5_bound, correlation_array)"""
    n = len(h)
    # Use np.correlate for efficiency
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full")
    c5 = np.max(corr) * dx
    return c5, corr

def make_feasible(h, target_sum):
    """Project h to satisfy 0<=h<=1 and sum(h)=target_sum"""
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        # Water-filling to fix the sum
        for _ in range(100):
            if abs(np.sum(h) - target_sum) < 1e-12:
                break
            if np.sum(h) < target_sum:
                # Need to increase sum
                mask = h < 1.0
                if not np.any(mask):
                    break
                h[mask] += diff / np.sum(mask)
            else:
                mask = h > 0.0
                if not np.any(mask):
                    break
                h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h

def binary_local_search(h, dx, max_iters=1000, time_limit=None, start_time=None):
    """Local search on binary (or near-binary) sequence using swap moves"""
    n = len(h)
    target_sum = n / 2.0
    
    # Make sure h is binary
    # Sort by value, assign top n/2 to 1, rest to 0
    # But we want to preserve the structure, so let's just work with current h
    # Actually, for binary local search, h should already be binary or near-binary
    
    c5, corr = compute_c5_values(h, dx)
    best_c5 = c5
    best_h = h.copy()
    
    for iteration in range(max_iters):
        if time_limit and (time.time() - start_time) > time_limit:
            break
        
        current_max = np.max(corr)
        # Find positions with 1 and 0
        positions_1 = np.where(h >= 0.5)[0]
        positions_0 = np.where(h < 0.5)[0]
        
        if len(positions_1) == 0 or len(positions_0) == 0:
            break
        
        improved = False
        for a in positions_1:
            for b in positions_0:
                # Compute delta for all lags
                # delta_c[k] = h[a-k] + h[a+k] - h[b-k] - h[b+k]
                # We need to convert lag to index in corr array
                # corr has length 2n-1, index p corresponds to lag p-(n-1)
                delta = np.zeros(2*n-1)
                for k in range(-(n-1), n):
                    # h[a+k] and h[a-k] etc., with out-of-range = 0
                    val = 0.0
                    if 0 <= a-k < n:
                        val += h[a-k]
                    if 0 <= a+k < n:
                        val += h[a+k]
                    if 0 <= b-k < n:
                        val -= h[b-k]
                    if 0 <= b+k < n:
                        val -= h[b+k]
                    delta[k + n - 1] = val
                
                new_corr = corr + delta
                new_max = np.max(new_corr)
                if new_max < current_max - 1e-10:
                    # Accept the swap
                    h[a] = 0.0
                    h[b] = 1.0
                    corr = new_corr
                    current_max = new_max
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            break
        
        if current_max * dx < best_c5:
            best_c5 = current_max * dx
            best_h = h.copy()
    
    return best_h, best_c5
