    c5 = compute_c5(h, dx)

import numpy as np

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def penalized_objective(h, dx, n, penalty_weight):
    target = n / 2.0
    c5 = compute_c5(h, dx)
    penalty = penalty_weight * (np.sum(h) - target) ** 2
    return c5 + penalty


def coordinate_descent(h, dx, n, penalty_weight, max_sweeps, grid_size, time_limit, start_time, rng):
    h = h.astype(float).copy()
    best_obj = penalized_objective(h, dx, n, penalty_weight)

    for sweep in range(max_sweeps):
        if time.time() - start_time > time_limit:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            old_val = h[i]
            best_local_obj = best_obj
            best_local_val = old_val
            grid = np.linspace(0.0, 1.0, grid_size)
            for t in grid:
                if abs(t - old_val) < 1e-9:
                    continue
                h[i] = t
                obj = penalized_objective(h, dx, n, penalty_weight)
                if obj < best_local_obj:
                    best_local_obj = obj
                    best_local_val = t
            h[i] = best_local_val
            if best_local_obj < best_obj - 1e-12:
                best_obj = best_local_obj
                improved = True
        if not improved:
            break

    true_score = compute_c5(h, dx)
    return h, true_score

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        mask = h < 1.0
        if mask.sum() == 0:
            return h
        adj = diff / mask.sum()
        h[mask] = np.minimum(h[mask] + adj, 1.0)
    else:
        mask = h > 0.0
        if mask.sum() == 0:
            return h
        adj = (-diff) / mask.sum()
        h[mask] = np.maximum(h[mask] - adj, 0.0)
    return h
