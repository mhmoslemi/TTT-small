    order = np.argsort(g)

def c5_subgrad(h, dx, top_k=3):
    """Subgradient using top-k active lags."""
    c = np.correlate(h, 1.0 - h, mode="full")
    n = len(h)
    L = n - 1
    top = np.argsort(c)[::-1][:top_k]
    # Weight by proximity to max
    weights = np.exp((c[top] - c[top[0]]) / (10 * dx))
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for w, idx in zip(weights, top):
        m = int(idx) - L
        for i in range(n):
            ni = i - m
            if 0 <= ni < n:
                grad[i] += w * (1.0 - h[ni])
            nj = i + m
            if 0 <= nj < n:
                grad[i] -= w * h[nj]
    return grad * dx

def cg_vertex(g, target_sum):

    frac = target_sum - k_full
    h_v[order[:k_full]] = 1.0
    if 0 < frac < 1:
        h_v[order[k_full]] = frac
    return h_v
