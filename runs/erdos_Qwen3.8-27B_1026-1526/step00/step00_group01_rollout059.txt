```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def c5_gradient(h, n, eps):
    """Compute gradient of smoothed C5 objective."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    c = corr * dx
    c_max = np.max(c)
    
    # Weights for smooth max
    a = (c - c_max) / eps
    exp_a = np.exp(a - np.max(a))  # shifted for numerical stability
    w = exp_a / np.sum(exp_a)
    
    # Compute gradient of each c[k] w.r.t. h
    # c[k] = dx * f_{k-(n-1)}(h) where f is the correlation at that lag
    grad = np.zeros(n)
    
    for m in range(2 * n - 1):
        if w[m] < 1e-15:
            continue
        lag = m - (n - 1)
        L = abs(lag)
        if L == 0:
            # c = dx * sum_i h[i]*(1-h[i])
            # grad_j = dx * (1 - 2*h[j])
            grad += w[m] * dx * (1.0 - 2.0 * h)
        else:
            # Linear part gradient
            if lag > 0:
                # linear = sum_{j=L}^{n-1} h[j]
                lin_grad = np.zeros(n)
                lin_grad[L:] = 1.0
            else:
                # linear = sum_{j=0}^{n-1-L} h[j]
                lin_grad = np.zeros(n)
                lin_grad[:n-L] = 1.0
            
            # Bilinear part gradient: B = sum_{i=0}^{n-1-L} h[i+L]*h[i]
            # dB/dh[j] = h[j-L] * [L<=j<=n-1] + h[j+L] * [0<=j<=n-1-L]
            bil_grad = np.zeros(n)
            # First term: h[j-L] for j in [L, n-1]
            bil_grad[L:] -= h[:n-L]
            # Second term: h[j+L] for j in [0, n-1-L]
            bil_grad[:n-L] -= h[L:]
            
            grad += w[m] * dx * (lin_grad + bil_grad)
    
    return c_max + eps * np.log(np.sum(exp_a)), grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    """Project h onto {0 <= h <= 1, sum(h) = n/2}"""
    h = np.clip(h, 0, 1)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # Need to increase sum; increase where h < 1
            mask = h < 1.0
            if not np.any(mask):
                break
            step = min(diff, (1.0 - h[mask]).max()) * 0.5
            h[mask] = np.clip(h[mask] + step * (1.0 - h[mask]) / ((1.0 - h[mask]).sum() + 1e-30), 0, 1)
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            step = min(-diff, h[mask].max()) * 0.5
            h[mask] = np.clip(h[mask] + step * h[mask] / (h[mask].sum() + 1e-30), 0, 1)
    # Final exact fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        # Find the element with most room
        if diff > 0:
            idx = np.argmax(1.0 - h)
        else:
            idx = np.argmin(h)
        h[idx] += diff
        h = np.clip(h, 0, 1)
    return h

def smooth_max(y, eps):
    """Softplus-based smooth maximum."""
    y_max = np.max(y)
    return y_max + eps * np.log(np.sum(np.exp((y - y_max) / eps)))

def objective(x, n, eps, penalty_weight):
    h = x.copy()
    # Penalty for sum constraint
    target = n / 2.0
    sum_penalty = penalty_weight * (np.sum(h) - target) ** 2
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return smooth_max(corr, eps) + sum_penalty

def run_optimization(h0, n, eps_sequence, max_iter=200, time_limit=None, start_time=None):
    if start_time is None:
        start_time = time.time()
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    
    for eps in eps_sequence:
        if time_limit and time.time() - start_time > time_limit - 5:
            break
        res = minimize(
            objective, h0, args=(n, eps, 100.0),
            method='L-BFGS-B',
            bounds=[(0, 1)] * n,
            options={'maxiter': max_iter, 'ftol': 1e-12}
        )
        h_new = project_feasible(res.x, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h0 = h_new.copy()
    
    return best_h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            if not np.any(mask):
                break
            # Proportional redistribution
            room = (1.0 - h)[mask]
            h[mask] += diff * room / room.sum()
        else:
            mask = h > 1e-15
            if not np.any(mask):
                break
            room = h[mask]
            h[mask] += diff * room / room.sum()
    h = np.clip(h, 0.0, 1.0)
    # Exact fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        if diff > 0:
            idx = np.argmax(1.0 - h)
        else:
            idx = np.argmin(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def smooth_max(y, eps):
    y_max = np.max(y)
    return y_max + eps * np.log(np.sum(np.exp((y - y_max) / eps)))

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Get initial values
    if 'initial_h_values' in dir() or 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, dtype=float)
    else:
        n_init = 47
        h_init = np.full(n_init, 0.5)
    
    n_init = len(h_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    
    # Define objective for SLSQP
    def obj_and_grad(x, eps):
        n = len(x)
        dx = 2.0 / n
        corr = np.correlate(x, 1.0 - x, mode="full") * dx
        val = smooth_max(corr, eps)
        # Gradient
        c = corr
        c_max = np.max(c)
        exp_c = np.exp((c - c_max) / eps)
        w = exp_c / np.sum(exp_c)
        # Gradient of each c[m] w.r.t. x
        grad = np.zeros(n)
        n_corr = 2 * n - 1
        for m in range(n_corr):
            if w[m] < 1e-18:
                continue
            lag = m - (n - 1)
            L = abs(lag)
            if L == 0:
                grad += w[m] * dx * (1.0 - 2.0 * x)
            else:
                if lag > 0:
                    lin = np.zeros(n)
                    lin[L:] = 1.0
                else:
                    lin = np.zeros(n)
                    lin[:n-L] = 1.0
                bil = np.zeros(n)
                bil[L:] -= x[:n-L]
                bil[:n-L] -= x[L:]
                grad += w[m] * dx * (lin + bil)
        return val, grad
    
    # Trial n values
    n_values = [n_init]
    # Add some other n values to try
    for n_try in [46, 48, 50, 52, 54, 60, 80, 100]:
        if n_try != n_init:
            n_values.append(n_try)
    
    eps_sequence = [0.1, 0.05, 0.02, 0.01, 0.005, 0.001]
    
    for n in n_values:
        if time.time() - start_time > budget_s - 30:
            break
        
        # Create initial guess for this n
        if n == n_init:
            h0 = best_h.copy()  # Use best found so far as starting point for original n
            # Actually, for n == n_init, use the initial values
            h0 = project_feasible(h_init.copy(), n)
        else:
            # Interpolate initial_h_values to n points
            x_old = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n)
            h_interp = np.interp(x_new, x_old, h_init)
            h0 = project_feasible(h_interp, n)
        
        # Also try random starts
        n_restarts = 3
        starts = [h0]
        for _ in range(n_restarts):
            h_rand = np.random.uniform(0, 1, n)
            starts.append(project_feasible(h_rand, n))
        
        # Also try structured starts
        # Step function
        h_step = np.zeros(n)
        h_step[:n//2] = 1.0
        h_step = project_feasible(h_step, n)
        starts.append(h_step)
        
        # Alternating
        h_alt = np.where(np.arange(n) % 2 == 0, 1.0, 0.0)
        h_alt = project_feasible(h_alt, n)
        starts.append(h_alt)
        
        for h_start in starts:
            if time.time() - start_time > budget_s - 10:
                break
            
            current_h = h_start.copy()
            for eps in eps_sequence:
                if time.time() - start_time > budget_s - 10:
                    break
                try:
                    res = minimize(
                        lambda x, e=eps: obj_and_grad(x, e),
                        current_h,
                        method='SLSQP',
                        bounds=[(0, 1)] * n,
                        constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
                                      'jac': lambda x: np.ones(n)}],
                        options={'maxiter': 300, 'ftol': 1e-10}
                    )
                    h_new = project_feasible(res.x, n)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        # Also save back to current_h for this n
                    current_h = h_new
                except Exception:
                    break
        
        # After optimizing for this n, if n == n_init, update best
        # If n != n_init, just note the c5
    
    return best_h, best_c5, len(best_h)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    """Project h onto the feasible set {0<=h<=1, sum(h)=n/2}."""
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total_room = room.sum()
            if total_room < 1e-15:
                break
            h += diff * room / total_room
        else:
            total = h.sum()
            if total < 1e-15:
                break
            h += diff * h / total
        h = np.clip(h, 0.0, 1.0)
    # Final exact fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        if diff > 0:
            idx = np.argmax(1.0 - h)
        else:
            idx = np.argmin(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def c5_smoothed_and_grad(h, eps):
    """Compute smoothed C5 and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    c = corr * dx
    c_max = np.max(c)
    
    # Numerically stable log-sum-exp
    a = (c - c_max) / eps
    a_max = np.max(a)
    log_sum_exp = a_max + np.log(np.sum(np.exp(a - a_max)))
    val = c_max + eps * log_sum_exp
    
    # Weights
    w = np.exp(a - a_max)
    w /= w.sum()
    
    # Gradient computation
    grad = np.zeros(n)
    n_corr = 2 * n - 1
    
    for m in range(n_corr):
        if w[m] < 1e-18:
            continue
        lag = m - (n - 1)
        L = abs(lag)
        if L == 0:
            grad += w[m] * dx * (1.0 - 2.0 * h)
        elif lag > 0:
            # f_L = sum_{i=0}^{n-1-L} h[i+L]*(1-h[i])
            # df/dh[k] = (1-h[k-L])*1_{L<=k<=n-1} - h[k+L]*1_{0<=k<=n-1-L}
            term = np.zeros(n)
            term[L:] = 1.0 - h[:n-L]  # (1-h[k-L]) for k in [L, n-1]
            term[:n-L] -= h[L:]  # -h[k+L] for k in [0, n-1-L]
            grad += w[m] * dx * term
        else:
            # f_{-L} = sum_{j=0}^{n-1-L} h[j]*(1-h[j+L])
            # df/dh[k] = (1-h[k+L])*1_{0<=k<=n-1-L} - h[k-L]*1_{L<=k<=n-1}
            term = np.zeros(n)
            term[:n-L] = 1.0 - h[L:]  # (1-h[k+L]) for k in [0, n-1-L]
            term[L:] -= h[:n-L]  # -h[k-L] for k in [L, n-1]
            grad += w[m] * dx * term
    
    return val, grad
```
