    """Local search on binary vectors via 2-flips."""
    best = h_bin.copy()
    best_c = _c5(best)

def _gaussian_smooth(h, sigma=1.0):

    n = len(h)
    k = int(2 * sigma)
    if k <= 0:
        return h.copy()
    kernel = np.exp(-0.5 * (np.arange(-k, k + 1) / sigma) ** 2)

    out = np.convolve(h, kernel, mode='same')
    return out

        return np.ones(n)
    lo = -3.0
    hi = 3.0
    for _ in range(200):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > s:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _obj_smooth(x):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = corr.max()
    return (m + np.log(np.sum(np.exp(corr - m))) / _g_smooth_t) * dx

def _grad_smooth(x):
    return _full_smooth_grad(x)

def _con_fun(x):
    return x.sum() - _g_target_sum

def _con_jac(x):
    return np.ones(len(x))

def _sgd_optimize(h, s, n, iters, alpha, beta, time_limit):
    best = h.copy()
    best_c = _c5(best)
    mom = np.zeros(n)
    a = alpha
    t_start = time.time()
    for i in range(iters):
        if time.time() - t_start > time_limit:
            break
        g = _active_subgrad(h)
        mom = beta * mom + (1.0 - beta) * g
        h = _proj(h - a * mom, s)
        c = _c5(h)
        if c < best_c:
            best_c = c
            best = h.copy()
        a *= 0.9998
    return best, best_c

def _slsqp_optimize(h, s, maxiter=200):
    global _g_target_sum
    _g_target_sum = s
    n = len(h)
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_obj_smooth, h, method='SLSQP', jac=_grad_smooth,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None:
            x = _proj(res.x, s)
            if _c5(x) < _c5(h):
                return x, _c5(x)
    except Exception:
        pass
    return h, _c5(h)

def _binary_local_search(h_bin, time_limit=60):
    n = len(h_bin)
    best = h_bin.copy()
    best_c = _c5(best)
    t_start = time.time()
    while time.time() - t_start < time_limit:
        improved = False
        ones = np.where(best == 1.0)[0]
        zeros = np.where(best == 0.0)[0]
        # Try all 2-flips
        for i in zeros:
            for j in ones:
                trial = best.copy()
                trial[i] = 1.0
                trial[j] = 0.0
                c = _c5(trial)
                if c < best_c - 1e-15:
                    best = trial
                    best_c = c
                    improved = True
                    ones = np.where(best == 1.0)[0]
                    zeros = np.where(best == 0.0)[0]
                    break
            if improved:
                break
        if not improved:
            break
    return best, best_c

def _make_binary(h, n, target_int):
    """Create a binary vector from a continuous h by interpolation and rounding."""
    x_old = np.linspace(0.0, 2.0, len(h), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    h_new = _proj(h_new, float(target_int))
    idx = np.argsort(h_new)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target_int]] = 1.0
    return h_bin

def run(seed=42, budget_s=1300, **kwargs):
    global _g_target_sum, _g_smooth_t
    start_time = time.time()
    np.random.seed(seed)
    
    n0 = len(initial_h_values)
    h0 = np.array(initial_h_values, dtype=float)
    s0 = n0 / 2.0
    _g_target_sum = s0
    _g_smooth_t = 100.0
    
    best_h = h0.copy()
    best_c5 = _c5(h0)
    best_n = n0
    
    def time_left():
        return budget_s - (time.time() - start_time)
    
    # Phase 1: Subgradient optimization
    h = _proj(h0.copy(), s0)
    for restart in range(20):
        if time_left() < 600:
            break
        a0 = 0.5 / (1.0 + 0.5 * restart)
        h, c = _sgd_optimize(h, s0, n0, 5000, a0, 0.88, time_limit=60)
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()
            best_n = n0
    
    # Phase 2: SLSQP refinement with different smoothing levels
    for t_val in [50.0, 100.0, 200.0, 500.0]:
        if time_left() < 300:
            break
        _g_smooth_t = t_val
        h, c = _slsqp_optimize(best_h, best_n / 2.0, maxiter=300)
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()
    
    # Phase 3: Binary local search
    for n_try in [n0 - 1, n0 + 1]:
        if time_left() < 200:
            break
        if n_try < 10 or n_try % 2 != 0:
            continue
        target_int = n_try // 2
        h_bin = _make_binary(best_h, n_try, target_int)
        h_bin, c = _binary_local_search(h_bin, time_limit=80)
        if c < best_c5:
            best_c5 = c
            best_h = h_bin.copy()
            best_n = n_try
    
    # Phase 4: More subgradient with different parameters
    if time_left() > 120:
        for beta_val in [0.9, 0.95]:
            if time_left() < 60:
                break
            h = _proj(best_h.copy(), best_n / 2.0)
            h, c = _sgd_optimize(h, best_n / 2.0, best_n, 5000, 0.1, beta_val, time_limit=50)
            if c < best_c5:
                best_c5 = c
                best_h = h.copy()
    
    # Phase 5: Random restarts
    if time_left() > 100:
        for trial in range(10):
            if time_left() < 50:
                break
            h_rand = np.random.rand(n0)
            h_rand = _proj(h_rand, s0)
            h_rand, c = _sgd_optimize(h_rand, s0, n0, 3000, 0.3, 0.9, time_limit=30)
            if c < best_c5:
                best_c5 = c
                best_h = h_rand.copy()
                best_n = n0
    
    # Phase 6: Final SLSQP
    if time_left() > 30:
        _g_smooth_t = 500.0
        h, c = _slsqp_optimize(best_h, best_n / 2.0, maxiter=500)
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()
    
    # Verify and return
    best_h = _proj(best_h, best_n / 2.0)
    best_c5 = _c5(best_h)
    
    return best_h, best_c5, best_n

Hmm, I realize there are a few issues:
1. The `_binary_local_search` function does O(n^2) evaluations per "improvement step", and each evaluation is O(n^2) (for the correlate). For n=46, that's 529 * 4200 ≈ 2.2M operations per step. With maybe 23 improvements possible (each flip changes one 0 to 1), that's about 50M operations per full pass. This is too slow for the time limit.
