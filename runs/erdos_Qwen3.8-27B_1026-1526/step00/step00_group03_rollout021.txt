        t2 = time.time()
        time_for_try = min(remaining * 0.6, 400.0)
        h_try_opt, c5_try = optimize_n(h_try, n_try, dx_try, time_for_try + t2, start_time, seed + 1)
        if c5_try < best_c5_global:
            best_c5_global = c5_try
            best_h_global = h_try_opt.copy()
            best_n_global = n_try

        elapsed = time.time() - start_time
        remaining = budget_s - elapsed - 5.0

    if remaining > 60:
        n_try2 = 100
        dx_try2 = 2.0 / n_try2
        target_try2 = n_try2 * 0.5
        h_try2 = np.interp(
            np.linspace(0, 2, n_try2, endpoint=False),
            np.linspace(0, 2, best_n_global, endpoint=False),
            best_h_global
        )
        h_try2 = repair_h(h_try2, target_try2)

        t3 = time.time()
        time_for_try2 = min(remaining * 0.7, 350.0)
        h_try2_opt, c5_try2 = optimize_n(h_try2, n_try2, dx_try2, time_for_try2 + t3, start_time, seed + 2)
        if c5_try2 < best_c5_global:
            best_c5_global = c5_try2
            best_h_global = h_try2_opt.copy()
            best_n_global = n_try2

    if remaining > 60:
        n_try3 = 200
        dx_try3 = 2.0 / n_try3
        target_try3 = n_try3 * 0.5
        h_try3 = np.interp(
            np.linspace(0, 2, n_try3, endpoint=False),
            np.linspace(0, 2, best_n_global, endpoint=False),
            best_h_global
        )
        h_try3 = repair_h(h_try3, target_try3)

        t4 = time.time()
        time_for_try3 = max(30.0, remaining - 10.0)
        h_try3_opt, c5_try3 = optimize_n(h_try3, n_try3, dx_try3, time_for_try3 + t4, start_time, seed + 3)
        if c5_try3 < best_c5_global:
            best_c5_global = c5_try3
            best_h_global = h_try3_opt.copy()
            best_n_global = n_try3

    final_h = repair_h(best_h_global, best_n_global * 0.5)
    final_c5 = eval_c5(final_h, best_n_global)[0]

    return final_h, final_c5, best_n_global
