def optimize_coordinate(h, i, n):
    """Find optimal h[i] using efficient correlation update."""
    corr = np.correlate(h, 1-h, mode="full")
    old = h[i]
    
    # Compute slope vector
    slope = np.zeros(2*n-1)
    for p in range(2*n-1):
        k = p - (n-1)
        s = 0.0
        ip = i + k  # i+k
        if 0 <= ip < n:
            s += (1.0 - h[ip])
        ik = i - k  # i-k
        if 0 <= ik < n:
            s -= h[ik]

        new_corr[n-1] = corr[n-1] + t*(1-t) - old*(1-old)
        return np.max(new_corr)

    if best_idx == 0 or best_idx == len(grid)-1:
        h[i] = best_t
        return
    
    # Refine
    lo = grid[max(0, best_idx-1)]
    hi = grid[min(len(grid)-1, best_idx+1)]

            hi = mid2
        else:
            lo = mid1
        if hi - lo < 1e-12:
            break
    
    best_t = (lo + hi) / 2
    h[i] = best_t
