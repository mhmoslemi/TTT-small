def repair_h_vec(h, n_points):
    target = n_points / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        if abs(s - target) < 1e-14:
            break
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
    # Final fix
    s = np.sum(h)
    err = target - s
    if abs(err) > 1e-14:
        if err > 0:
            room = 1.0 - h
            tr = room.sum()
            if tr > 1e-15:
                h += err * room / tr
        else:
            tv = h.sum()
            if tv > 1e-15:
                h += err * h / tv
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_descent_paired(h, n_points, n_passes, n_grid, time_limit, start_time):
    """Coordinate descent with paired moves to maintain sum exactly."""
    h = h.copy()
    best_c5 = compute_c5_vec(h, n_points)
    
    for p in range(n_passes):
        if time.time() - start_time > time_limit:
            break
        for j in range(n_points):
            if time.time() - start_time > time_limit:
                return h, best_c5
            # Try paired moves with each k > j
            # For efficiency, try a subset
            # Find best single move for coordinate j
            # Compensate by uniform adjustment of others
            
            old_val = h[j]
            best_delta = 0.0
            best_obj = float('inf')
            
            # Move h[j] and compensate uniformly
            # new_h[j] = old_val + delta
            # new_h[k] = h[k] - delta/(n-1) for k != j
            # Constraint: 0 <= old_val + delta <= 1 and 0 <= h[k] - delta/(n-1) <= 1 for all k != j
            
            # delta range:
            d_lo = max(-old_val, -(1.0 - h[np.arange(n_points) != j].min()) * (n_points - 1))
            # Actually: for k != j, we need 0 <= h[k] - delta/(n-1) <= 1
            # So: delta/(n-1) <= h[k] and delta/(n-1) >= h[k] - 1
            # delta <= (n-1)*h[k] and delta >= (n-1)*(h[k]-1)
            # For all k: delta <= (n-1)*min(h[k]) and delta >= (n-1)*max(h[k]-1)
            # Also: -old_val <= delta <= 1 - old_val
            
            other_vals = h[np.arange(n_points) != j]
            d_lo = max(-old_val, (n_points - 1) * (other_vals.max() - 1.0))
            d_hi = min(1.0 - old_val, (n_points - 1) * other_vals.min())
            
            if d_lo >= d_hi:
                continue
            
            deltas = np.linspace(d_lo, d_hi, n_grid)
            for d in deltas:
                h[j] = old_val + d
                h[np.arange(n_points) != j] = other_vals - d / (n_points - 1)
                obj = compute_c5_vec(h, n_points)
                if obj < best_obj:
                    best_obj = obj
                    best_delta = d
            
            if best_delta != 0:
                h[j] = old_val + best_delta
                h[np.arange(n_points) != j] = other_vals - best_delta / (n_points - 1)
                if best_obj < best_c5:
                    best_c5 = best_obj
    
    return h, best_c5
