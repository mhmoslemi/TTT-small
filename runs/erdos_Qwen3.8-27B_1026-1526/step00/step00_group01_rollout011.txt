    return float(np.max(corr) * dx)

def fix_sum(h, target):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        order = np.argsort(-room)

            h[idx] += add
            rem -= add

            h[idx] -= sub
