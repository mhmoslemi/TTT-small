    if abs(diff) > 1e-10:

            mask = h < 1.0 - 1e-10
            if mask.sum() == 0:
                return h
            room = (1.0 - h[mask])
            total_room = room.sum()
            if total_room > 0:

            total_room = room.sum()
            if total_room > 0:

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] = np.minimum(1.0, h[mask] + diff / cnt)
        else:
            mask = h > 1e-15
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] = np.maximum(0.0, h[mask] + diff / cnt)
    return np.clip(h, 0.0, 1.0)

def smooth_c5(h, T=80.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = T * (corr - m)
    lse = m + np.log(np.sum(np.exp(shifted))) / T
    return float(lse * dx)

def smooth_grad(h, T=80.0, eps=1e-6):
    n = len(h)
    g = np.zeros(n)
    f0 = smooth_c5(h, T)

        hp = h.copy()
        hp[i] += eps

        fm = h.copy()
        fm[i] -= eps
        fm = project(fm, n)
        g[i] = (smooth_c5(hp, T) - smooth_c5(fm, T)) / (2.0 * eps)
    return g
