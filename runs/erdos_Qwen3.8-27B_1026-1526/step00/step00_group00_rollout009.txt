- The term where i+offset = j, i.e., i = j-offset: was h[j-offset]*(1-h[j]), now h[j-offset]*(1-t)
  Change: h[j-offset] * ((1-t) - (1-h[j])) = h[j-offset] * (h[j] - t)

So result[k] = result_old[k] + (t - h[j])*(1-h[j+offset]_valid) + h[j-offset]_valid * (h[j] - t)
= result_old[k] + (t - h[j]) * (1 - h[j+offset]) - (t - h[j]) * h[j-offset]
= result_old[k] + (t - h[j]) * (1 - h[j+offset] - h[j-offset])

where h[j+offset] and h[j-offset] are 0 if the index is out of bounds.

So result[k] = result_old[k] + (t - h_j) * S_k

where S_k = 1 - h[j+offset] - h[j-offset] (with boundary handling)

This means: result[k] is LINEAR in t!
result[k] = (result_old[k] - h_j * S_k) + t * S_k

So the objective max_k result[k] is the max of linear functions of t, which is convex in t.

To minimize it, I can use scipy.optimize.minimize_scalar with bounds [0, 1].

But I can also solve it more efficiently by noting that the optimum is at a point where two of the linear functions intersect (or at a boundary).

For practical purposes, minimize_scalar with method='bounded' should work fine. It does a golden section search and takes about 50 iterations to converge.

```python
import numpy as np
from scipy.optimize import minimize_scalar
import time


def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx


def normalize_h(h, target_sum):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < (1.0 - 1e-10)
    else:
        mask = h > 1e-10
    if not np.any(mask):
        return h
    n_adj = float(np.sum(mask))
    h[mask] += diff / n_adj
    h = np.clip(h, 0.0, 1.0)
    diff2 = target_sum - np.sum(h)
    if abs(diff2) > 1e-10:
        if diff2 > 0:
            mask2 = h < (1.0 - 1e-10)
        else:
            mask2 = h > 1e-10
        if np.any(mask2):
            candidates = np.where(mask2)[0]
            idx = candidates[np.argmin(np.abs(h[candidates] - 0.5))]
            h[idx] = np.clip(h[idx] + diff2, 0.0, 1.0)
    return h


def obj_1d_direct(t, h_template, j):
    h = h_template.copy()
    h[j] = t
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr))


def coord_descent_step(h, target_sum):
    n = len(h)
    h_cur = h.copy()
    for j in range(n):
        result = minimize_scalar(obj_1d_direct, args=(h_cur, j),
                                  bounds=(0.0, 1.0), method='bounded',
                                  options={'maxiter': 80, 'xatol': 1e-11})
        h_cur[j] = float(result.x)
    h_cur = normalize_h(h_cur, target_sum)
    return h_cur


def sa_perturb(h, n, target_sum, scale):
    i, j = np.random.choice(n, 2, replace=False)
    delta = np.random.uniform(-scale, scale)
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    h_new = normalize_h(h_new, target_sum)
    return h_new


def multi_perturb(h, n, target_sum, npert, scale):
    h_new = h.copy()
    indices = np.random.choice(n, npert, replace=False)
    deltas = np.random.uniform(-scale, scale, npert)
    for k in range(npert):
        h_new[indices[k]] += deltas[k]
    h_new = normalize_h(h_new, target_sum)
    return h_new


def binary_start(n, target_sum, rng):
    h = np.zeros(n)
    n_ones = int(np.round(target_sum))
    positions = rng.choice(n, n_ones, replace=False)
    h[positions] = 1.0
    diff = target_sum - np.sum(h)
    if diff > 0:
        zeros = np.where(h == 0)[0]
        if len(zeros) > 0:
            h[zeros[0]] = diff
    elif diff < 0:
        ones = np.where(h == 1)[0]
        if len(ones) > 0:
            h[ones[0]] = 1.0 + diff
    return normalize_h(h, target_sum)


def sinusoidal_start(n, target_sum, rng):
    freq = rng.uniform(0.5, 5.0)
    phase = rng.uniform(0, 2 * np.pi)
    h = 0.5 + 0.5 * np.sin(2.0 * np.pi * freq * np.arange(n) / n + phase)
    h = np.clip(h, 0, 1)
    return normalize_h(h, target_sum)


def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except (NameError, Exception):
        n = 45
        h0 = np.full(n, 0.5)

    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = normalize_h(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)

    T = 0.03
    no_improve = 0
    iteration = 0

    while time.time() - t0 < budget_s - 20:
        iteration += 1

        # Main: coordinate descent
        h_cd = coord_descent_step(best_h, target_sum)
        c5_cd = compute_c5(h_cd, dx)

        if c5_cd < best_c5 - 1e-12:
            best_h = h_cd
            best_c5 = c5_cd
            no_improve = 0
        else:
            no_improve += 1

            if no_improve <= 20:
                # Small SA perturbation
                h_try = sa_perturb(best_h, n, target_sum, 0.15)
                c5_try = compute_c5(h_try, dx)
                if c5_try < best_c5:
                    best_h = h_try
                    best_c5 = c5_try
                    no_improve = 0
                    T = 0.03
                else:
                    prob = np.exp(-(c5_try - best_c5) / T)
                    if np.random.random() < prob:
                        best_h = h_try
                    T = max(0.001, T * 0.99)
            elif no_improve <= 80:
                # Medium perturbation
                h_try = multi_perturb(best_h, n, target_sum,
                                      np.random.randint(2, 6), 0.25)
                c5_try = compute_c5(h_try, dx)
                if c5_try < best_c5:
                    best_h = h_try
                    best_c5 = c5_try
                    no_improve = 0
                    T = 0.04
                else:
                    prob = np.exp(-(c5_try - best_c5) / T)
                    if np.random.random() < prob:
                        best_h = h_try
                    T = max(0.001, T * 0.98)
            elif no_improve <= 200:
                # Large perturbation / partial restart
                keep_frac = 0.3
                n_keep = max(1, int(n * keep_frac))
                keep = np.random.choice(n, n_keep, replace=False)
                h_try = np.random.uniform(0, 1, n)
                h_try[keep] = best_h[keep]
                h_try = normalize_h(h_try, target_sum)
                c5_try = compute_c5(h_try, dx)
                if c5_try < best_c5:
                    best_h = h_try
                    best_c5 = c5_try
                    no_improve = 0
                    T = 0.05
                else:
                    h_try = best_h + np.random.normal(0, 0.15, n)
                    h_try = normalize_h(h_try, target_sum)
                    c5_try = compute_c5(h_try, dx)
                    if c5_try < best_c5:
                        best_h = h_try
                        best_c5 = c5_try
                        no_improve = 30
            else:
                # Full restarts with different structures
                structures = []
                structures.append(binary_start(n, target_sum, rng))
                structures.append(sinusoidal_start(n, target_sum, rng))
                structures.append(normalize_h(np.random.uniform(0, 1, n), target_sum))
                for h_struct in structures:
                    c5_s = compute_c5(h_struct, dx)
                    if c5_s < best_c5:
                        best_h = h_struct.copy()
                        best_c5 = c5_s
                        no_improve = 0
                        T = 0.05
                        break
                no_improve = max(0, no_improve - 100)

    # Final polish with multiple CD passes
    remaining = budget_s - (time.time() - t0)
    if remaining > 3:
        max_passes = min(30, int(remaining / 3))
        for _ in range(max_passes):
            h_f = coord_descent_step(best_h, target_sum)
            c5_f = compute_c5(h_f, dx)
            if c5_f < best_c5:
                best_h = h_f
                best_c5 = c5_f
            else:
                break

    best_h = normalize_h(best_h, target_sum)
    best_c5 = compute_c5(best_h, dx)

    return best_h, best_c5, n
```
