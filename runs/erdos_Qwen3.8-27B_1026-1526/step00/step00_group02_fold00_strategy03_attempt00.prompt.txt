import numpy as np
from scipy.optimize import minimize
import time

n = 45  # fixed number of samples
dx = 2.0 / n  # fixed discretization step

def objective(x):
    # Convert to numpy array if not
    if isinstance(x, (list, tuple)):
        x = np.array(x)
    
    # Calculate the shifted function 1-h
    w = 1 - x
    corr = np.correlate(x, w, mode='full')
    max_corr = np.max(corr) if len(corr) > 0 else 0
    return dx * max_corr

def normalize_vector(v, n):
    # Rescales to have sum 22.5 while preserving ratios
    current_sum = np.sum(v)
    if current_sum == 0:
        scale = 22.5 / (n * 0.5)  # 22.5 = n/2 * dx
        return v * scale
    scale = 22.5 / current_sum
    result = v * scale
    # After scaling, might violate bounds
    result = np.clip(result, 0, 1)
    corrected_sum = np.sum(result)
    # Minor readjustment since clipping changes the sum
    final_scale = 22.5 / corrected_sum
    return result * final_scale

def particle_swarm_optimization(n_particles=30, max_iterations=500, w=0.7298, c1=1.4944, c2=1.4944):
    # Convert initial_h_values to numpy array
    initial = np.array(initial_h_values)
    
    # Initialize particles
    particles = [initial.copy() for _ in range(n_particles)]
    velocities = np.random.uniform(-0.2, 0.2, (n_particles, n))
    
    # Initialize personal best and global best
    personal_best = [p.copy() for p in particles]
    global_best = initial.copy()
    best_score = objective(global_best)
    
    # PSO parameters
    w_personal = 2.0
    w_global = 2.0
    
    for it in range(max_iterations):
        # Calculate fitness for each particle
        scores = [objective(p) for p in particles]
        
        # Update personal best
        for i in range(n_particles):
            if scores[i] < objective(personal_best[i]):
                personal_best[i] = particles[i].copy()

        for i in range(n_particles):
            if scores[i] < best_score:
                best_score = scores[i]
                global_best = particles[i].copy()
        
        # Calculate average velocity and position
        avg_personal = np.array([personal_best[i] for i in range(n_particles)])
        avg_global = np.array([global_best.copy() for _ in range(n_particles)])
        
        # Update velocities and positions
        for i in range(n_particles):
            # Avoid boundaries
            if any(particles[i] < 0) or any(particles[i] > 1):
                continue  # Don't update out-of-bound particles

            velocity_mag = np.linalg.norm(velocities[i])
            v_inertia = w * velocities[i] + c1 * (personal_best[i] - particles[i]) / velocity_mag
            v_global = (c2 * (global_best - particles[i]) / velocity_mag) * w_personal
            
            # Add impact from particle positions
            if np.random.random() < 0.1:
                # Small movement based on wish vector
                wish_dir = avg_global[i] - particles[i]
                v_wish = 0.05 * wish_dir

            # Ensure velocities are within manageable range

            if speed > 0.05 * np.sqrt(n):

            particles[i] += velocities[i]

            # Re-normalize to meet sum constraint
