def compute_c5(h, n):

import numpy as np
import time

def compute_max_corr(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr)

def make_initial_h(n, seed, mode='random'):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    if mode == 'random':
        h[rng.choice(n, n//2, replace=False)] = 1.0
    elif mode == 'clumped':
        # Place ones in consecutive blocks
        block_size = max(1, n // 4)
        n_blocks = n // (2 * block_size)
        remaining = n - n_blocks * 2 * block_size
        pos = 0
        for b in range(n_blocks):
            h[pos:pos+block_size] = 1.0
            pos += 2 * block_size
        # Fill remaining
        if remaining > 0:
            positions = rng.choice(n, remaining, replace=False)
            for p in positions:
                if h[p] == 0:
                    h[p] = 1.0
                    break  # just one to maintain count
        # Ensure exactly n//2 ones
        n_ones = int(np.sum(h))
        if n_ones > n//2:
            ones = np.where(h == 1)[0]
            h[rng.choice(ones, n_ones - n//2, replace=False)] = 0.0
        elif n_ones < n//2:
            zeros = np.where(h == 0)[0]
            h[rng.choice(zeros, n//2 - n_ones, replace=False)] = 1.0
    elif mode == 'golden':
        # Golden ratio spacing
        phi = (1 + np.sqrt(5)) / 2
        positions = np.array([int(i * phi) % n for i in range(n//2)])
        positions = np.unique(positions)
        if len(positions) < n//2:
            remaining = rng.choice(n, n//2 - len(positions), replace=False)
            positions = np.unique(np.concatenate([positions, remaining]))
        h[positions[:n//2]] = 1.0
    
    return h

    rng = np.random.RandomState(seed)
    h = make_initial_h(n, seed, init_mode)
    
    current_c = compute_max_corr(h)
    best_h = h.copy()
    best_c = current_c

    T = 5.0
    min_T = 0.1
    
    while time.time() - start < budget_s and T > min_T:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h[i], h[j] = 0.0, 1.0
        new_c = compute_max_corr(h)
        
        if new_c < current_c or rng.rand() < np.exp(-(new_c - current_c) / T):
