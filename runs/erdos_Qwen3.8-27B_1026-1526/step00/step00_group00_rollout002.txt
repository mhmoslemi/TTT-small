    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr * dx)

def project_to_sum(h, target_sum):
    """Project h onto the set {x: sum(x) = target_sum, 0 <= x[i] <= 1}"""
    h = np.clip(h, 0, 1)

    diff = target_sum - current_sum
    if abs(diff) < 1e-10:
        return h
    # Distribute the difference

            return h
        h[mask] += new_diff / np.sum(mask)

Let me also consider: can I use CVXPY? The objective max_k corr_k is a max of quadratic functions. This isn't a standard convex problem (max of concave functions is not convex). So CVXPY won't directly help.

However, I could formulate it as:
minimize t
subject to: corr_k(h) <= t for all k
0 <= h[i] <= 1
sum(h) = n/2

Where corr_k(h) = sum_i h[i]*(1-h[i+k]) is quadratic (concave in h for fixed k, since it's -sum h[i]*h[i+k] + const).

The constraint corr_k(h) <= t becomes: sum_i h[i]*h[i+k] >= n/2 - t, which is a non-convex constraint (sum of products >= constant is not a convex constraint).

So this is a non-convex problem. CVXPY won't work directly.

Let me stick with SLSQP + simulated annealing + random restarts.

One more thought: the "smooth max" trick. I'll use:
smooth_max(x, α) = (1/α) * log(sum(exp(α*x_i)))

This is a smooth upper bound on max(x_i), and as α → ∞, it approaches max(x_i).

For the gradient: ∂ smooth_max / ∂ x_i = exp(α*x_i) / sum(exp(α*x_j))

This gives a soft "attention" on which k is contributing most to the max.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(vals, alpha):
    m = np.max(vals)
    exp_vals = np.exp(alpha * (vals - m))
    return m + np.log(np.sum(exp_vals) / len(exp_vals)) / alpha

def smooth_max_grad(vals, alpha):
    m = np.max(vals)
    exp_vals = np.exp(alpha * (vals - m))
    weights = exp_vals / np.sum(exp_vals)
    grad = alpha * (weights - weights * weights * 0)  # d/dx_i of log-sum-exp / alpha
    return weights, vals  # weights are the softmax; gradient of smooth_max w.r.t. each val

def optimize_h_slsqp(h0, n, target_sum, alpha, maxiter, ftol):
    def objective(h_flat):
        h_c = np.clip(h_flat, 0.0, 1.0)
        corr = np.correlate(h_c, 1.0 - h_c, mode="full")
        return smooth_max(corr, alpha)
    
    def gradient(h_flat):
        h_c = np.clip(h_flat, 0.0, 1.0)
        corr = np.correlate(h_c, 1.0 - h_c, mode="full")
        m = np.max(corr)
        exp_vals = np.exp(alpha * (corr - m))
        w = exp_vals / np.sum(exp_vals)
        # grad of smooth_max(corr, alpha) w.r.t. h = w^T * d(corr)/d(h)
        # corr_k = sum_i h[i]*(1-h[i+k]) for appropriate indices
        # d(corr_k)/d(h[j]) = (1 - h[j+k]) for i=j term, and -h[j-k] for the other term
        # This is complex; use numerical or just return objective only
        return None
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(objective, h0.copy(), method='SLSQP', bounds=bounds,
                       constraints=constraints, options={'maxiter': maxiter, 'ftol': ftol})
        return project_to_sum(res.x, target_sum)
    except:
        return project_to_sum(h0, target_sum)

def simulated_annealing(h0, n, target_sum, rng, max_iters, T0, T_end):
    h = h0.copy()
    c5_cur = compute_c5(h)
    c5_best = c5_cur
    h_best = h.copy()
    T = T0
    cool = (T_end / T0) ** (1.0 / max(1, max_iters))
    for it in range(max_iters):
        if rng.random() < 0.5:
            idx = rng.randint(n)
            delta = rng.uniform(-0.08, 0.08)
            h_new = h.copy()
            h_new[idx] = np.clip(h_new[idx] + delta, 0.0, 1.0)
            h_new = project_to_sum(h_new, target_sum)
        else:
            i, j = rng.choice(n, size=2, replace=False)
            delta = rng.uniform(-0.05, 0.05)
            h_new = h.copy()
            h_new[i] = np.clip(h_new[i] + delta, 0.0, 1.0)
            h_new[j] = np.clip(h_new[j] - delta, 0.0, 1.0)
            h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur or rng.random() < np.exp(-(c5_new - c5_cur) / max(T, 1e-10)):
            h = h_new
            c5_cur = c5_new
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h.copy()
        T = max(T * cool, T_end)
    return h_best, c5_best

def coordinate_descent(h0, n, target_sum, rng, max_rounds):
    h = h0.copy()
    c5_cur = compute_c5(h)
    for rnd in range(max_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            for delta in [-0.02, -0.01, -0.005, 0.005, 0.01, 0.02]:
                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] + delta, 0.0, 1.0)
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < c5_cur - 1e-12:
                    h = h_new
                    c5_cur = c5_new
                    improved = True
                    break
        if not improved:
            break
    return h, c5_cur

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    time_left = lambda: budget_s - (time.time() - t_start)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    # Initialize from provided parent
    initial_h = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            initial_h = np.array(initial_h_values, dtype=float)
            n_init = len(initial_h)
            ts = n_init / 2.0
            initial_h = project_to_sum(initial_h, ts)
            best_h = initial_h.copy()
            best_c5 = compute_c5(best_h)
            best_n = n_init
        except:
            pass
    
    if best_h is None:
        n = 45
        best_n = n
        best_h = np.full(n, 0.5)
        best_c5 = compute_c5(best_h)
    
    rng = np.random.RandomState(seed)
    
    n_candidates = [45, 50, 60, 72, 90, 100, 120, 144, 180, 200]
    
    for n in n_candidates:
        if time_left() < 40:
            break
        ts = n / 2.0
        if np.abs(ts - round(ts)) > 1e-6:
            continue  # skip non-integer target sums
        
        # Generate multiple starting points for this n
        starts = []
        
        # Start 0: from best if n matches
        if best_n == n and best_h is not None:
            starts.append(best_h.copy())
        
        # Start 1: uniform
        starts.append(np.full(n, 0.5))
        
        # Start 2: random
        hr = rng.uniform(0.0, 1.0, n)
        starts.append(project_to_sum(hr, ts))
        
        # Start 3: sinusoidal
        x = np.linspace(0, 2, n, endpoint=False)
        hs = 0.5 + 0.3 * np.sin(2 * np.pi * 2 * x / 2.0)
        starts.append(project_to_sum(hs, ts))
        
        # Start 4: two-tone blocks
        h4 = np.zeros(n)
        blk = n // 6
        h4[0:blk] = 1.0
        h4[2*blk:3*blk] = 1.0
        h4[4*blk:5*blk] = 1.0
        starts.append(project_to_sum(h4, ts))
        
        # Start 5: random binary-ish
        h5 = (rng.rand(n) > 0.5).astype(float)
        starts.append(project_to_sum(h5, ts))
        
        # Start 6: interpolate from best if different n
        if best_h is not None and best_n != n:
            from numpy import interp
            x_old = np.linspace(0, 2, best_n, endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            h6 = interp(x_new, x_old, best_h, period=2.0)
            starts.append(project_to_sum(h6, ts))
        
        for si, h_start in enumerate(starts):
            if time_left() < 30:
                break
            h_start = project_to_sum(h_start, ts)
            c5_s = compute_c5(h_start)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_start.copy()
                best_n = n
            
            # SLSQP with increasing alpha
            for alpha in [5.0, 15.0, 50.0, 150.0]:
                if time_left() < 25:
                    break
                maxiter = min(300, int(time_left() * 3))
                h_opt = optimize_h_slsqp(h_start, n, ts, alpha, maxiter, 1e-9)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    best_n = n
                h_start = h_opt
            
            # Simulated annealing refinement
            if time_left() > 60:
                sa_iters = min(8000, int(time_left() * 2))
                h_sa, c5_sa = simulated_annealing(h_start, n, ts, rng, sa_iters, 0.02, 0.001)
                if c5_sa < best_c5:
                    best_c5 = c5_sa
                    best_h = h_sa.copy()
                    best_n = n
                h_start = h_sa
            
            # Coordinate descent
            if time_left() > 40:
                h_cd, c5_cd = coordinate_descent(h_start, n, ts, rng, 50)
                if c5_cd < best_c5:
                    best_c5 = c5_cd
                    best_h = h_cd.copy()
                    best_n = n
    
    # Final polish on best solution
    if best_h is not None and time_left() > 20:
        n_f = best_n
        ts_f = n_f / 2.0
        for alpha in [100.0, 300.0, 1000.0]:
            if time_left() < 15:
                break
            h_pol = optimize_h_slsqp(best_h, n_f, ts_f, alpha, 500, 1e-10)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()
        
        if time_left() > 15:
            h_cd, c5_cd = coordinate_descent(best_h, n_f, ts_f, rng, 80)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()
    
    # Validate
    n_final = len(best_h)
    ts_final = n_final / 2.0
    best_h = project_to_sum(best_h, ts_final)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    
    return (best_h, best_c5, n_final)
