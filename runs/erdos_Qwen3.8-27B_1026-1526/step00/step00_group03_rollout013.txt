```python
def compute_gradient(h, p, dx, n):
    # p has length 2n-1, p[k] corresponds to shift s = k-(n-1)
    grad = np.zeros(n)
    for k in range(2*n-1):
        s = k - (n-1)
        if s >= 0:
            # term1: (1-h[i+s]) for i=0..n-1-s
            i_range = np.arange(n - s)
            grad[i_range] += p[k] * (1 - h[i_range + s])
            # term2: -h[i-s] for i=s..n-1
            i_range = np.arange(s, n)
            grad[i_range] -= p[k] * h[i_range - s]
        else:
            s_abs = -s
            # term1: (1-h[i+s]) for i=s_abs..n-1
            i_range = np.arange(s_abs, n)
            grad[i_range] += p[k] * (1 - h[i_range + s])
            # term2: -h[i-s] for i=0..n-1-s_abs
            i_range = np.arange(n - s_abs)
            grad[i_range] -= p[k] * h[i_range - s]
    
    grad *= dx
    return grad
```

```python
import numpy as np
import time
from scipy.optimize import minimize, Bounds
from scipy.special import logsumexp

def project_h(h, target_sum):
    h = np.clip(h, 0, 1)
    # Binary search for shift to make sum = target_sum
    s = np.sum(h)
    if abs(s - target_sum) < 1e-10:
        return h
    if s > target_sum:
        lo, hi = 0.0, 1.0
    else:
        lo, hi = -1.0, 0.0
    for _ in range(50):
        mid = (lo + hi) / 2
        h_shifted = np.clip(h + mid, 0, 1)
        s_shifted = np.sum(h_shifted)
        if s > target_sum:
            if s_shifted > target_sum:
                lo = mid
            else:
                hi = mid
        else:
            if s_shifted < target_sum:
                lo = mid
            else:
                hi = mid
    h = np.clip(h + (lo + hi) / 2, 0, 1)
    # Final adjustment: distribute remainder
    rem = target_sum - np.sum(h)
    if abs(rem) < 1e-9:
        return h
    # Add/subtract to elements not at boundary
    if rem > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    n_active = np.sum(mask)
    if n_active > 0:
        h[mask] += rem / n_active
        h = np.clip(h, 0, 1)
    return h

def smooth_obj(h, dx, temp, penalty):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    scaled = corr / temp
    max_s = np.max(scaled)
    exp_s = np.exp(scaled - max_s)
    obj = temp * (max_s + np.log(np.sum(exp_s))) * dx
    s_val = np.sum(h) - n / 2.0
    obj += penalty * s_val ** 2
    return obj

def smooth_grad(h, dx, temp, penalty):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    scaled = corr / temp
    max_s = np.max(scaled)
    exp_s = np.exp(scaled - max_s)
    sum_exp = np.sum(exp_s)
    p = exp_s / sum_exp
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        s = k - (n - 1)
        w = p[k]
        if s >= 0:
            idx1 = np.arange(n - s)
            grad[idx1] += w * (1.0 - h[idx1 + s])
            idx2 = np.arange(s, n)
            grad[idx2] -= w * h[idx2 - s]
        else:
            sa = -s
            idx1 = np.arange(sa, n)
            grad[idx1] += w * (1.0 - h[idx1 + s])
            idx2 = np.arange(n - sa)
            grad[idx2] -= w * h[idx2 - s]
    grad *= dx
    
    s_val = np.sum(h) - n / 2.0
    grad += 2 * penalty * s_val
    return grad

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

    best_c5 = compute_c5(h0, dx)

    # Add some perturbed starts

    # Alternating pattern with some smoothness

    h_struct = project_h(h_struct, target_sum)
    starts.append(h_struct)
    
    temp = 1.0
    penalty = 50.0
    
    for h_start in starts:
        if time.time() - t_start > budget_s - 60:
            break
        
        temp = 1.0
        penalty = 50.0
        current = h_start.copy()
        
        for outer in range(40):
            if time.time() - t_start > budget_s - 30:
                break
            
            def obj_func(x):
                return smooth_obj(x, dx, temp, penalty)
            
            def grad_func(x):
                return smooth_grad(x, dx, temp, penalty)
            
            bounds = Bounds(np.zeros(n_points), np.ones(n_points))
            
            try:
                result = minimize(
                    obj_func, current, jac=grad_func,

    # Final fix

    return best_h, best_c5, n_points
```
