    lo = max(0, k - n + 1)

def subgradient_step(h, n, step_size, rng):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    # Find the maximizing lag(s)
    max_val = np.max(corr)
    # Use top few for averaging
    threshold = max_val - 1e-6
    max_lags = np.where(corr >= threshold)[0]
    
    # Compute average subgradient
    sg = np.zeros(n)
    for k in max_lags:
        sg += subgradient_of_c5_at_k(k, h, n)
    sg /= len(max_lags)
    
    return h - step_size * sg
