        for idx in candidates:
            if diff > 0:
                h[idx] = min(1.0, h[idx] + per)
            else:
                h[idx] = max(0.0, h[idx] + per)
    # Final exact fix
    diff = sum_target - np.sum(h)

                    break
        else:
            for i in range(len(h)):
                if h[i] > 0.0:
                    h[i] = max(0.0, h[i] + diff)
                    break
    return h
