import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = beta * (x - m)
    # Clip to avoid overflow
    shifted = np.clip(shifted, -500, 500)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def obj_func(x, n, beta):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, beta) * (2.0 / n)

def sum_eq(x, n):
    return np.sum(x) - n / 2.0

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            # Need to increase sum: add to entries below 1
            mask = h < 1.0 - 1e-14
            if not np.any(mask):
                break
            num = min(int(np.ceil(diff)), int(np.sum(mask)))
            num = max(1, num)
            idx = np.where(mask)[0][:num]
            h[idx] = np.minimum(h[idx] + diff / num, 1.0)
        else:
            # Need to decrease sum: subtract from entries above 0
            mask = h > 1e-14
            if not np.any(mask):
                break
            num = min(int(np.ceil(-diff)), int(np.sum(mask)))
            num = max(1, num)
            idx = np.where(mask)[0][:num]
            h[idx] = np.maximum(h[idx] + diff / num, 0.0)
    # Exact fix
    diff = n / 2.0 - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = np.argmax(1.0 - h)
            h[idx] = min(h[idx] + diff, 1.0)
        else:
            idx = np.argmax(h)
            h[idx] = max(h[idx] + diff, 0.0)
    return h

def optimize_from(h0, n, max_iter, beta):

    bounds = [(0.0, 1.0)] * n
    constraints = [('eq', sum_eq, (n,))]

                      constraints=constraints,
                      args=(n, beta),
                      options={'maxiter': max_iter, 'ftol': 1e-10, 'iprint': 0})
        h_res = project_feasible(res.x, n)
        return h_res, compute_c5(h_res)
    except Exception:
        return None, None

    x_new = np.linspace(0.0, 2.0, n_new)
    return np.interp(x_new, x_old, h_old)
