The gradient with respect to h[i] is:

So the gradient is: Σ_m p_m * a_m

This is a weighted average of the coefficients, which is easy to compute!

def gradient_step(h, target, tau=100.0, lr=0.01):
    n = len(h)
    # Compute overlaps
    curr = np.correlate(h, 1.0 - h, mode='full')
    # Compute softmax weights
    weights = np.exp(tau * (curr - np.max(curr)))
    weights /= np.sum(weights)
    # Compute gradient
    grad = np.zeros(n)
    for i in range(n):
        a = np.zeros(2*n-1)
        for m in range(2*n-1):
            x = m - (n-1)
            c = 0.0
            ip = i + x
            if 0 <= ip < n:
                c += 1.0 - h[ip]
            im = i - x
            if 0 <= im < n:
                c -= h[im]
            a[m] = c
        grad[i] = np.sum(weights * a)
    # Projected gradient step
    h_new = h - lr * grad
    h_new = project_sum(h_new, target)
    return h_new
