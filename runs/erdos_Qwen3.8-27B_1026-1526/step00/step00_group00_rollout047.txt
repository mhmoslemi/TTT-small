def compute_gradient(h, k, n):
    """Compute gradient of r(k) w.r.t. h at point h.
    k is the index into np.correlate(h, 1-h, mode='full') result.
    np.correlate(h, 1-h, mode='full')[k] corresponds to lag k - (n-1)
    """
    lag = k - (n - 1)
    grad = np.ones(n)
    # Subtract h[j - lag] for valid j-lag
    # and h[j + lag] for valid j+lag
    if lag >= 0:
        # h[j - lag]: valid when j >= lag
        grad[lag:] -= h[:n-lag]
        # h[j + lag]: valid when j <= n-1-lag
        grad[:n-lag] -= h[lag:]

        # h[j + lag] = h[j - l]: valid when j >= l
        grad[l:] -= h[:n-l]
    return grad

        h_pol, c5_pol = slsqp_optimize(best_h, n_best, n_best / 2.0, max_iter=500)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, len(best_h)
