import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def grad_c_k(h, k, n, dx):
    shift = k - (n - 1)
    i = np.arange(n)
    grad = np.zeros(n)
    j1 = i + shift
    valid1 = (j1 >= 0) & (j1 < n)
    if np.any(valid1):
        grad[valid1] += (1.0 - h[j1[valid1]])
    j2 = i - shift
    valid2 = (j2 >= 0) & (j2 < n)
    if np.any(valid2):
        grad[valid2] -= h[j2[valid2]]
    return dx * grad


def project_to_constraint(h, target_sum):
    lo = -2.0
    hi = 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - proj.sum()
    if abs(diff) > 1e-12:
        mask = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        if np.any(mask):
            proj[mask] += diff / np.sum(mask)
            proj = np.clip(proj, 0.0, 1.0)
    return proj


def smooth_max_obj(x, dx, n, beta):
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    m = np.max(corr)
    shifted = beta * (corr - m)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def smooth_max_grad(x, dx, n, beta):
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    m = np.max(corr)
    shifted = beta * (corr - m)
    exp_shifted = np.exp(shifted)
    s = np.sum(exp_shifted)
    weights = exp_shifted / s
    g = np.zeros(n)
    for k in range(len(corr)):
        if weights[k] > 1e-10:
            g += weights[k] * grad_c_k(x, k, n, dx)
    return g
