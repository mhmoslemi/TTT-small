def subgradient_descent(h, n, dx, target_sum, time_limit=100.0, lr_init=0.01):
    h = h.copy()
    start = time.time()
    lr = lr_init
    best = compute_c5_value(h, dx)
    best_h = h.copy()
    
    while time.time() - start < time_limit:
        # Compute correlation vector
        corr = np.correlate(h, 1 - h, mode='full') * dx
        c_max = np.max(corr)
        active = np.where(corr >= c_max - 1e-10)[0]
        
        # Compute gradient of c[k] w.r.t. h for active k values
        # grad[i] = sum over active k of d c[k] / d h[i]
        grad = np.zeros(n)
        for k in active:
            for i in range(n):
                # d c[k] / d h[i]
                val = 0.0
                # h[i] appears as h[n+k] when n+k=i, i.e., n=i-k
                if 0 <= i - k < n:
                    val += (1 - h[i - k])
                # h[i] appears as h[n] when n=i
                if 0 <= i + k < n:
                    val -= h[i + k]
                grad[i] += val * dx
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            lr *= 0.5
            continue
        
        # Take step
        h_new = h - lr * grad / grad_norm
        
        # Project to constraint set
        h_new = project_to_constraint(h_new, target_sum)
        
        score = compute_c5_value(h_new, dx)
        if score < best:
            best = score
            best_h = h_new.copy()
            lr = min(lr * 1.05, 0.1)
        else:
            lr *= 0.7
            if lr < 1e-6:
                break
        
        h = h_new
    
    return best_h
