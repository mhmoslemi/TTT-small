def compute_grad_c(h, n):
    """Compute dc[k]/dh[i] for all k and i. Returns array of shape (2n-1, n)."""
    # For each shift s (ranging from -(n-1) to n-1), corresponding to k = s + (n-1):
    # dc[k]/dh[i] = (1-h[i-s]) * [0<=i-s<n] - h[i+s] * [0<=i+s<n]
    
    dc = np.zeros((2*n-1, n))
    for ki in range(2*n-1):
        s = ki - (n-1)
        # P: (1-h[i-s]) for i where 0 <= i-s < n
        i_vals = np.arange(n)
        idx_p = i_vals - s  # = i - s
        valid_p = (idx_p >= 0) & (idx_p < n)
        P = np.zeros(n)
        P[valid_p] = 1.0 - h[idx_p[valid_p]]
        
        # N: h[i+s] for i where 0 <= i+s < n
        idx_n = i_vals + s
        valid_n = (idx_n >= 0) & (idx_n < n)
        N = np.zeros(n)
        N[valid_n] = h[idx_n[valid_n]]
        
        dc[ki] = P - N
    
    return dc

def enforce_sum(h, target):
    """Project h to satisfy sum(h) = target while keeping h in [0,1]."""
    h = h.copy()
    diff = target - np.sum(h)
    for _ in range(50):
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not np.any(mask):
            break

        if fraction >= 1.0 or fraction <= -1.0:
            # Can't fix in one step
            if diff > 0:
                h[mask] = np.minimum(h[mask] + 1.0, 1.0)
            else:
                h[mask] = np.maximum(h[mask] - 1.0, 0.0)
        else:
            h[mask] += fraction
        h = np.clip(h, 0, 1)
        diff = target - np.sum(h)
    return h

    n = len(h)
    c = np.correlate(h, 1.0 - h, mode="full")
    scaled = alpha * dx * c
    max_s = np.max(scaled)
    
    penalty = penalty_weight * (np.sum(h) - target_sum)**2

        obj = max_s / alpha + penalty
        # Numerical gradient fallback
        return obj
    
    exp_s = np.exp(scaled - max_s)
    sum_exp = np.sum(exp_s)
    w = exp_s / sum_exp
    
    obj = (max_s + np.log(sum_exp)) / alpha + penalty
    return obj

        grad += 2.0 * penalty_weight * (np.sum(h) - target_sum)
        return grad

    sum_exp = np.sum(exp_s)
    w = exp_s / sum_exp

        idx_n = i_vals + s
        valid_n = (idx_n >= 0) & (idx_n < n)
        grad_corr[valid_n] -= wk * h[idx_n[valid_n]]

    n = len(h)

        h_test[i] = val
        return compute_c5(h_test, dx)

    grid = np.linspace(0, 1, 100)

    a = grid[max(0, best_idx-1)]

    c = b - gr * (b - a)

    fd = obj_1d(d)

            d = c

        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = obj_1d(d)

        idx_p = i_vals - s
        valid_p = (idx_p >= 0) & (idx_p < n)
        grad[valid_p] = dx * (1.0 - h[idx_p[valid_p]])
        idx_n = i_vals + s
        valid_n = (idx_n >= 0) & (idx_n < n)
        grad[valid_n] -= dx * h[idx_n[valid_n]]
        grad += 2.0 * pw * (np.sum(h) - target_sum)
        return obj, grad

    exp_s = np.exp(scaled - max_s)
    sum_exp = np.sum(exp_s)
    w = exp_s / sum_exp
    obj = (max_s + np.log(sum_exp)) / alpha + penalty

    grad_corr = np.zeros(n)
    i_vals = np.arange(n)
    for k in range(2 * n - 1):
        s = k - (n - 1)
        wk = w[k]
        if wk < 1e-14:
            continue
        idx_p = i_vals - s
        valid_p = (idx_p >= 0) & (idx_p < n)
        grad_corr[valid_p] += wk * (1.0 - h[idx_p[valid_p]])
        idx_n = i_vals + s
        valid_n = (idx_n >= 0) & (idx_n < n)
        grad_corr[valid_n] -= wk * h[idx_n[valid_n]]

    grad = dx * grad_corr + 2.0 * pw * (np.sum(h) - target_sum)
    return obj, grad

def coord_optimize(h, i, dx, target_sum):
    n = len(h)
    def obj_1d(val):
        ht = h.copy()
        ht[i] = val
        ht = enforce_sum(ht, target_sum)
        return compute_c5(ht, dx)
    grid = np.linspace(0.0, 1.0, 64)
    vals = np.array([obj_1d(g) for g in grid])
    bi = np.argmin(vals)
    a = grid[max(0, bi - 1)]
    b = grid[min(len(grid) - 1, bi + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = b - gr * (b - a)
    d1 = a + gr * (b - a)
    fc = obj_1d(c1)
    fd = obj_1d(d1)
    for _ in range(40):
        if fc < fd:
            b = d1
            d1 = c1
            fd = fc
            c1 = b - gr * (b - a)
            fc = obj_1d(c1)
        else:
            a = c1
            c1 = d1
            fc = fd
            d1 = a + gr * (b - a)
            fd = obj_1d(d1)
    return (a + b) / 2.0

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0

    h0 = np.clip(h0, 0.0, 1.0)
    h0 = enforce_sum(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)

    alphas = [5, 10, 20, 50, 100, 200, 400]
    perturbs = [0.0, 0.03, 0.07, 0.12, 0.2, 0.35, 0.5]
    pws = [500.0, 2000.0, 10000.0]

    trials = []
    for a in alphas:
        for p in perturbs:
            for pw in pws:
                trials.append((a, p, pw))
    rng.shuffle(trials)

    for idx, (alpha, pert, pw) in enumerate(trials):
        elapsed = time.time() - t0
        if elapsed > budget_s - 100:
            break

        if pert == 0.0:
            hs = best_h.copy()
        else:
            hs = best_h + rng.uniform(-pert, pert, n)
            hs = np.clip(hs, 0.0, 1.0)
            hs = enforce_sum(hs, target_sum)

        try:
            res = minimize(
                soft_max_obj_grad, hs,
                args=(alpha, dx, target_sum, pw),
                method='L-BFGS-B', jac=True,
                bounds=[(0.0, 1.0)] * n,
                options={'maxiter': 200, 'ftol': 1e-15, 'gtol': 1e-10, 'maxls': 40}
            )
            hn = np.clip(res.x, 0.0, 1.0)
            hn = enforce_sum(hn, target_sum)
            c5n = compute_c5(hn, dx)
            if c5n < best_c5 - 1e-12:
                best_c5 = c5n
                best_h = hn.copy()
        except Exception:
            pass

    elapsed = time.time() - t0
    max_cd = budget_s - 30 - elapsed
    if max_cd > 20:
        for sweep in range(50):
            if time.time() - t0 > budget_s - 30:
                break
            improved = False
            for i in range(n):
                if time.time() - t0 > budget_s - 30:
                    break
                old = best_h[i]
                new = coord_optimize(best_h, i, dx, target_sum)
                if abs(old - new) > 1e-10:
                    best_h[i] = new
                    best_h = enforce_sum(best_h, target_sum)
                    c5t = compute_c5(best_h, dx)
                    if c5t < best_c5:
                        best_c5 = c5t
                        improved = True
            if not improved:
                break

    for it in range(200):
        if time.time() - t0 > budget_s - 10:
            break
        step = max(0.001, 0.05 * (1.0 - it / 200.0))
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        delta = rng.uniform(-step, step)
        hn = best_h.copy()
        hn[i] = np.clip(hn[i] + delta, 0.0, 1.0)
        hn[j] = np.clip(hn[j] - delta, 0.0, 1.0)
        hn = enforce_sum(hn, target_sum)
        c5n = compute_c5(hn, dx)
        if c5n < best_c5 - 1e-12:
            best_c5 = c5n
            best_h = hn.copy()

    best_h = enforce_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h, dx)
    return best_h, best_c5, n
