import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(y, target_sum):
    """Project y onto {x: 0<=x_i<=1, sum(x)=target_sum}"""
    n = len(y)
    
    def do_project(tol):
        lo, hi = y.min() - 1, y.max() + 1
        for _ in range(100):
            mid = (lo + hi) / 2
            projected = np.clip(y - mid, 0, 1)
            s = projected.sum()
            if abs(s - target_sum) < 1e-12:
                break
            if s > target_sum:
                lo = mid
            else:
                hi = mid
        return np.clip(y - (lo + hi) / 2, 0, 1)
    
    return do_project(1e-12)

def compute_correlations(h):
    return np.correlate(h, 1 - h, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(compute_correlations(h)) * dx

def subgradient_at_h(h, k_lag):
    """Compute subgradient of c_k * dx wrt h, where k_lag is the lag index"""
    n = len(h)
    dx = 2.0 / n
    subgrad = np.zeros(n)
    for m in range(n):
        val = 0.0
        if 0 <= m - k_lag < n:
            val += (1 - h[m - k_lag])
        if 0 <= m + k_lag < n:
            val -= h[m + k_lag]
        subgrad[m] = val
    return subgrad * dx

import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(y, target):
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        p = np.clip(y - mid, 0.0, 1.0)
        s = p.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    return float(np.max(compute_corr(h)) * 2.0 / len(h))

def subgrad_lag(h, k):
    n = len(h)
    sg = np.zeros(n)
    if k == 0:
        sg[:] += (1.0 - h[:])
        sg[:] -= h[:]
    elif k > 0:
        sg[k:] += (1.0 - h[:n - k])
        sg[:n - k] -= h[k:]
    else:
        kk = -k
        sg[:n - kk] += (1.0 - h[kk:])
        sg[kk:] -= h[:n - kk]
    return sg

def subgrad_smoothed(h, corr, tau, topk=5):
    n = len(h)
    max_c = corr.max()
    exps = np.exp((corr - max_c) / tau)
    exs = exps.sum()
    if exs < 1e-300:
        return np.zeros(n)
    weights = exps / exs
    top_idx = np.argpartition(weights, -topk)[-topk:]
    sg = np.zeros(n)
    for ki in top_idx:
        w = weights[ki]
        if w > 1e-10:
            lag = int(ki) - (n - 1)
            sg += w * subgrad_lag(h, lag)
    return sg

def subgrad_top1(h, corr):
    n = len(h)
    ki = int(corr.argmax())
    lag = ki - (n - 1)
    return subgrad_lag(h, lag)

def local_subgrad(h, target, dx, n, n_iter, alpha, t0, budget_s, time_frac):
    for _ in range(n_iter):
        if time.time() - t0 > budget_s * time_frac:
            break
        corr = compute_corr(h)
        sg = subgrad_top1(h, corr)
        h = h - alpha * sg * dx
        h = project_feasible(h, target)
    return h

    n = len(h_init)
    target = n / 2.0
    dx = 2.0 / n

    h = project_feasible(h_init, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    def tleft():
        return budget_s - (time.time() - t0)

    # Phase 1: Subgradient descent with annealed smoothing
    h_curr = best_h.copy()
    tau = 0.3
    alpha = 0.15
    stall_count = 0
    stall_best = best_c5

    it = 0
    phase1_end = budget_s * 0.45
    while tleft() > budget_s - phase1_end:
        corr = compute_corr(h_curr)
        sg = subgrad_smoothed(h_curr, corr, tau)
        h_curr = h_curr - alpha * sg * dx
        h_curr = project_feasible(h_curr, target)

        c5 = compute_c5(h_curr)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_curr.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > 5000:
            h_curr = best_h + rng.normal(0, 0.05, n)
            h_curr = project_feasible(h_curr, target)
            stall_count = 0

        it += 1

            alpha = max(0.005, alpha * 0.995)

    # Phase 2: Nelder-Mead refinement

            x = project_feasible(x, target)

            if res.fun < best_c5:
