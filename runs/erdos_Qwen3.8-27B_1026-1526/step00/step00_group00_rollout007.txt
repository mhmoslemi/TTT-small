    t_start = time.time()

    def remaining():
        return budget_s - (time.time() - t_start)

    best_h = None
    best_c5 = np.inf
    best_n = 0

    def update_best(h, n):
        nonlocal best_h, best_c5, best_n
        c5 = compute_c5(h, n)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            best_n = n
        return c5

    # Phase 1: Optimize from initial_h_values
    h0 = np.array(initial_h_values).flatten().astype(float)
    n0 = len(h0)
    h0 = fix_sum(h0, n0)
    update_best(h0, n0)

    if remaining() > 180:
        h_opt = do_optimize_slsqp(h0, n0, [5.0, 20.0, 80.0, 300.0], max_iter_each=200)
        update_best(h_opt, n0)

    # Phase 2: Optimize at various n_points
    n_candidates = [46, 64, 80, 100, 128]
    for n in n_candidates:
        if remaining() < 150:
            break

        # Build multiple starting points
        starts = []
        h_interp = upsample_interp(h0, n)
        starts.append(h_interp)

        h_bin = random_binary(n, rng)
        starts.append(h_bin)

        h_tm = thue_morse_seq(n)
        # For even n, thue-morse has exactly n/2 ones
        if abs(np.sum(h_tm) - n / 2.0) > n / 2.0:
            h_tm[0] = 1.0 - h_tm[0]
        starts.append(fix_sum(h_tm, n))

        h_rnd = random_smooth(n, rng)
        starts.append(h_rnd)

        # Pick best start
        best_start = starts[0]
        best_start_c5 = compute_c5(best_start, n)
        for s in starts[1:]:
            c = compute_c5(s, n)
            if c < best_start_c5:
                best_start = s
                best_start_c5 = c
        update_best(best_start, n)

        # SLSQP optimization from best start
        if remaining() > 100:
            h_opt = do_optimize_slsqp(best_start, n, [10.0, 50.0, 200.0], max_iter_each=200)
            update_best(h_opt, n)

        # Coordinate descent refinement
        if remaining() > 60:
            h_cd, c5_cd = coordinate_descent(best_start, n, max_rounds=200)
            update_best(h_cd, n)

    # Phase 3: Further refinement on best solution
    if remaining() > 60:
        h_ref = do_optimize_slsqp(best_h, best_n, [50.0, 200.0, 500.0], max_iter_each=300)
        update_best(h_ref, best_n)

    # Phase 4: Coordinate descent on final best
    if remaining() > 30:
        h_cd, c5_cd = coordinate_descent(best_h, best_n, max_rounds=300)
        update_best(h_cd, best_n)

    # Final verification
    if best_h is not None:
        final_c5 = compute_c5(best_h, best_n)
        if final_c5 < best_c5:
            best_c5 = final_c5

    return (best_h, best_c5, best_n)
