    return np.max(corr) * dx

def c5_smooth(h, n, temperature):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    shifted = corr - m
    log_sum = np.log(np.maximum(np.sum(np.exp(temperature * shifted)) - 1.0, 1e-300))
    return m + log_sum / temperature
