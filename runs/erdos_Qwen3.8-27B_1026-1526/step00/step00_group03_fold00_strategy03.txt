def compute_overlap_integral(h):
    nt = 2*n_points - 1  # Result length
    overlap_array = np.zeros(nt)
    for lag in range(-n_points+1, n_points):
        if lag < 0:
            shifted_h = np.roll(h, lag)
        elif lag > 0:
            shifted_h = np.append(np.zeros(lag), h[:-lag])  # Simple rolling
        else:
            shifted_h = h.copy()
        for i in range(n_points):
            if i + lag < 0 or i + lag >= n_points:
                continue
            if h[i] > 0 and shifted_h[i + lag] > 0:
                if abs(i - (i+lag)) < 1:  # No overlap contribution
                    continue
                # But the correlation mode='full' naturally handles this
                overlap_array[lag] += h[i] * (1-shifted_h[i+lag]) * dx/2
    return np.max(overlap_array) * 2  # Scale to C₅ units
