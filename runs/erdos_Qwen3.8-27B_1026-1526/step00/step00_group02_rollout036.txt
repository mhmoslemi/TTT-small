    n = len(h)

    lo = h.min() - 1.0

        else:
            hi = mid

    """Compute C5 = max of full correlation of h and 1-h, times dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx, corr * dx

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')

                        decay=0.999, time_limit=None, start_time=None, seed=42):

    rng = np.random.RandomState(seed)

    best_c5 = compute_c5(h)
    alpha = alpha_init
    
    for step in range(n_steps):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break

        current_max = np.max(corr_vals)

        # Larger weight for lags closer to max
        weights = np.exp((corr_vals - current_max) / 0.01)
        weights /= np.sum(weights)

        # Find top lags
        top_k = np.argsort(corr_vals)[-10:][::-1]
        top_corr = corr_vals[top_k]
        top_weights = np.exp((top_corr - current_max) / 0.01)
        top_weights /= np.sum(top_weights)
        
        grad = np.zeros(n)
        for w, idx in zip(top_weights, top_k):
            k = int(idx - (n - 1))  # convert index to lag
            grad += w * gradient_of_correlation_fast(h, k)
        
        # Take step
        h_new = h - alpha * grad
        h_new = project_box_simplex(h_new, target_sum)

            # Don't decrease alpha when improving
            alpha = min(alpha * 1.005, 0.3)
        else:
            alpha *= decay
        
        h = h_new
    
    return best_h, best_c5

import numpy as np
import time

def project_box_simplex(h, target_sum):
    n = len(h)
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(h - lam, 0.0, 1.0)

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def grad_lag(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        if n - k > 0:
            g[k:] += 1.0 - h[:n-k]
            g[:n-k] -= h[k:]
    else:
        kp = -k
        if n - kp > 0:
            g[:n-kp] += 1.0 - h[kp:]
            g[kp:] -= h[:n-kp]
    return g

def optimize_subgrad(h, time_limit, start_time, seed, iters_per_check=1000):
    n = len(h)
    target = n * 0.5
    rng = np.random.RandomState(seed)
    
    best_h = h.copy()
    best_c5 = np.max(compute_corr(h))
    alpha = 0.1
    momentum = np.zeros(n)
    beta = 0.9
    
    it = 0
    while time.time() - start_time < time_limit:
        corr = compute_corr(h)
        current_max = np.max(corr)
        
        # Top lags with softmax weights
        top_m = min(10, len(corr))
        top_idx = np.argsort(corr)[-top_m:][::-1]
        top_vals = corr[top_idx]
        w = np.exp((top_vals - current_max) / 0.005)
        w /= w.sum()
        
        grad = np.zeros(n)
        for wi, idx in zip(w, top_idx):
            k = int(idx - (n - 1))
            grad += wi * grad_lag(h, k)
        
        momentum = beta * momentum - alpha * grad
        h_new = project_box_simplex(h + momentum, target)
        
        c5_new = np.max(compute_corr(h_new))
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            alpha = min(alpha * 1.01, 0.5)
        else:
            alpha *= 0.98
            if alpha < 0.005:
                # Add perturbation
                h_new = project_box_simplex(h + rng.randn(n) * 0.05, target)
                alpha = 0.05
            h = h_new if c5_new < current_max else h
        
        h = h_new
        it += 1
        
        if it % iters_per_check == 0 and alpha < 0.01:
            # Try a bigger perturbation
            h = project_box_simplex(h + rng.randn(n) * 0.1, target)
            alpha = 0.1
    
    return best_h, best_c5

def optimize_coordinate(h, time_limit, start_time, seed):
    """Pairwise coordinate descent"""
    n = len(h)
    target = n * 0.5
    rng = np.random.RandomState(seed)
    
    best_h = h.copy()
    best_c5 = np.max(compute_corr(h))
    
    order = list(range(n))
    
    while time.time() - start_time < time_limit:
        # Each sweep: for each pair (i, j), optimize
        # h[i] + h[j] = s (constant), minimize max correlation
        np.random.seed(seed + int(time.time() * 1000) % 100000)
        rng_pair = np.random.RandomState(int(time.time() * 1000) % 100000)
        
        # Random pairs
        for _ in range(200):
            if time.time() - start_time > time_limit:
                break
            i = rng_pair.randint(n)
            j = rng_pair.randint(n)
            if i == j:
                continue
            s = h[i] + h[j]
            # Try moving h[i] to new value, h[j] = s - new value
            # Objective: max correlation
            # Binary search on h[i] in [max(0, s-1), min(1, s)]
            lo = max(0.0, s - 1.0)
            hi = min(1.0, s)
            if lo >= hi:

def project_box_simplex(h, target_sum):
    n = len(h)
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(h - lam, 0.0, 1.0)

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def grad_lag(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        if n > k:
            g[k:n] += 1.0 - h[0:n-k]
            g[0:n-k] -= h[k:n]
    else:
        kp = -k
        if n > kp:
            g[0:n-kp] += 1.0 - h[kp:n]
            g[kp:n] -= h[0:n-kp]
    return g

def smooth_grad(h, tau):
    n = len(h)
    corr = compute_corr(h)
    max_c = corr.max()
    w = np.exp(tau * (corr - max_c))
    w_sum = w.sum()
    if w_sum < 1e-300:
        return np.zeros(n)
    w /= w_sum
    # Only use lags with significant weight
    active = w > 1e-6
    grad = np.zeros(n)
    for idx in np.where(active)[0]:
        k = int(idx - (n - 1))
        grad += w[idx] * grad_lag(h, k)
    return grad
