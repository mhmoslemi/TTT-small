    # Main optimization loop

        else:

    h_best = project_feasible(h_best, target_sum)
    final_c5 = evaluate_c5(h_best)
    return h_best, final_c5, n

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    proj = np.clip(h + lam, 0.0, 1.0)
    diff = target_sum - proj.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            candidates = np.where(proj < 1.0 - 1e-8)[0]
        else:
            candidates = np.where(proj > 1e-8)[0]
        if len(candidates) > 0:
            per = diff / len(candidates)
            proj[candidates] = np.clip(proj[candidates] + per, 0.0, 1.0)
            diff2 = target_sum - proj.sum()
            if abs(diff2) > 1e-12:
                idx = np.argmin(np.abs(proj - 0.5))
                proj[idx] = np.clip(proj[idx] + diff2, 0.0, 1.0)
    return proj
