def solve_1d_coord(h, j, n, mu):
    """Find optimal h[j] given all other coordinates fixed."""
    # Compute A_k and B_k for each lag offset d
    # C_k(h) = A_k + B_k * h[j] (where h[j] is the only variable)
    
    A_list = []
    B_list = []
    
    for d in range(-(n-1), n):
        i_lo = max(0, -d)
        i_hi = min(n-1, n-1-d)
        
        const = 0.0
        coeff = 0.0
        
        for i in range(i_lo, i_hi + 1):
            i_plus_d = i + d
            if i == j and 0 <= i_plus_d < n:
                # h[j] * (1 - h[j+d]) = h[j] - h[j]*h[j+d]
                # const: 0, coeff: 1 - h[j+d]
                coeff += 1.0 - h[i_plus_d]
            elif i_plus_d == j and 0 <= i < n:
                # h[i] * (1 - h[j]) = h[i] - h[i]*h[j]
                # const: h[i], coeff: -h[i]
                const += h[i]
                coeff -= h[i]
            else:
                # h[i] * (1 - h[i+d]), no h[j]
                const += h[i] * (1.0 - h[i_plus_d])
        
        A_list.append(const)
        B_list.append(coeff)
    
    A_arr = np.array(A_list)
    B_arr = np.array(B_list)
    
    # Now minimize: max(A_arr + B_arr * x) + mu * (S + x)^2
    # where S = sum of all other h[i] - n/2 = (sum(h) - h[j]) - n/2
    
    S = float(np.sum(h)) - h[j] - n / 2.0
    
    # The function is convex. Find its minimum on [0, 1].
    # Use the fact that the derivative is: B_{k*(x)} + 2*mu*(S+x)
    # where k*(x) is an active constraint at x.
    
    # Strategy: find all "critical points" where the identity of the active
    # constraint could change, plus where the derivative is zero.
    
    # For each pair of constraints i, m:
    #   A_i + B_i * x = A_m + B_m * x

    # At each critical point x_c, the active set changes.
