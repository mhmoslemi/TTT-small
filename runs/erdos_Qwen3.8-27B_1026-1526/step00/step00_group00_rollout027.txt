import cvxpy as cp

def solve_lp_step(coeff_matrix, n, target_sum):
    """
    Solve: max t
    s.t. coeff_matrix @ h >= t (for each row)
         0 <= h[i] <= 1
         sum(h) = target_sum
    
    coeff_matrix has shape (2n-1, n)
    """
    h = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == target_sum,
        coeff_matrix @ h >= t
    ]
    
    objective = cp.Maximize(t)
    prob = cp.Problem(objective, constraints)
    
    prob.solve(solver=cp.ECOS)  # or another solver
    
    if h.value is not None:
        return h.value, t.value
    return None, None
