def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    if s > 1e-12:
        h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if diff > 0:
        idx = np.argsort(h)
        for i in idx:
            if diff <= 1e-12:
                break
            add = min(1.0 - h[i], diff)
            h[i] += add
            diff -= add
    elif diff < 0:
        idx = np.argsort(h)
        for i in reversed(idx):
            if diff >= -1e-12:
                break
            sub = min(h[i], -diff)
            h[i] -= sub
            diff += sub
    return h

def build_constraints(n, dx):
    lags = np.arange(-(n-1), n)
    
    def cons_fun(v):
        t = v[-1]
        h = v[:-1]
        c = np.empty(2*n-1)
        for idx, lag in enumerate(lags):
            if lag == 0:
                r = np.sum(h * (1.0 - h))
            elif lag > 0:
                m = n - lag
                r = np.sum(h[lag:] * (1.0 - h[:m]))
            else:
                m = n + lag
                r = np.sum(h[:m] * (1.0 - h[-lag:]))
            c[idx] = t - dx * r
        return c
    
    def cons_jac(v):
        t = v[-1]
        h = v[:-1]
        jac = np.zeros((2*n-1, n+1))
        for idx, lag in enumerate(lags):
            jac[idx, -1] = 1.0
            for j in range(n):
                val = 0.0
                jp = j - lag
                jm = j + lag
                if 0 <= jp < n:
                    val += (1.0 - h[jp])
                if 0 <= jm < n:
                    val -= h[jm]
                jac[idx, j] = -dx * val
        return jac
    
    return cons_fun, cons_jac

    lags = np.arange(-(n-1), n)
    cons_fun, cons_jac = build_constraints(n, dx)
    eq_fun, eq_jac = build_eq(n)
    
    v0 = np.concatenate([h0, [compute_c5(h0, dx) + 1e-3]])
    
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    
    constraints = [
        {'type': 'ineq', 'fun': cons_fun, 'jac': cons_jac},
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
    ]
    
    result = minimize(
        lambda v: v[-1],
        v0,
        jac=lambda v: np.concatenate([np.zeros(n), [1.0]]),
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': maxiter, 'ftol': tol, 'disp': False}
    )
    
    if result.success or result.fun < compute_c5(h0, dx):
        h_opt = make_feasible(result.x[:n], n)
        return h_opt, compute_c5(h_opt, dx)
    return h0, compute_c5(h0, dx)

    """Refine h using coordinate descent."""

        improved = False

            # If we increase h[j] by d, decrease others proportionally

                else:

                    h = h_test2
                    current_c5 = c5_test
                    improved = True
                    break
        if not improved:
            break
    return h, current_c5
