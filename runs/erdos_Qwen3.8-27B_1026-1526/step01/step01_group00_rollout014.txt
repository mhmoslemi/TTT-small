def compute_gradient(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m))
    w = exp_vals / np.sum(exp_vals)
    L = m + np.log(np.sum(exp_vals)) / alpha
    
    # Lag k ranges from -(n-1) to n-1
    # w_lag[k] = w[k + (n-1)] for k = -(n-1), ..., n-1
    # For each j, grad[j] = sum_k w_lag[k] * [(1-h[j+k])*1[0≤j+k<n] - h[j-k]*1[0≤j-k<n]]
    
    # Vectorize: create a (n, 2n-1) matrix where entry [j, m] is the contribution
    # k = m - (n-1)
    # j+k = j + m - (n-1), valid if 0 ≤ j+m-n+1 < n
    # j-k = j - m + (n-1), valid if 0 ≤ j-m+n-1 < n
    
    js = np.arange(n)
    ms = np.arange(2*n-1)
    ks = ms - (n-1)  # lag values: -(n-1), ..., n-1
    
    # j + k matrix: shape (n, 2n-1)
    jk = js[:, None] + ks[None, :]  # (n, 2n-1)
    # j - k matrix
    jmk = js[:, None] - ks[None, :]  # (n, 2n-1)
    
    # Part 1: (1-h[j+k]) where 0 ≤ j+k < n
    mask1 = (jk >= 0) & (jk < n)  # (n, 2n-1)
    # Part 2: h[j-k] where 0 ≤ j-k < n
    mask2 = (jmk >= 0) & (jmk < n)  # (n, 2n-1)
    
    # For Part 1: need (1-h[j+k]) for valid entries
    # h values at indices jk where valid
    h_jk = np.where(mask1, h[np.clip(jk, 0, n-1)], 0)  # (n, 2n-1)
    part1 = np.where(mask1, 1.0 - h_jk, 0.0)
    
    # For Part 2: need h[j-k] for valid entries  
    h_jmk = np.where(mask2, h[np.clip(jmk, 0, n-1)], 0)
    part2 = h_jmk
    
    # grad[j] = sum_m w[m] * (part1[j,m] - part2[j,m])
    grad = (part1 - part2) @ w  # (n,)
    
    return L, grad, corr

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    # Clamp for numerical stability
    exp_vals = np.exp(alpha * (corr - m))
    s = np.sum(exp_vals)
    w = exp_vals / s
    L = m + np.log(s) / alpha
    
    # Vectorized gradient
    ns = np.arange(n)
    ms = np.arange(2 * n - 1)
    ks = ms - (n - 1)
    
    jk = ns[:, None] + ks[None, :]
    jmk = ns[:, None] - ks[None, :]
    
    mask1 = (jk >= 0) & (jk < n)
    mask2 = (jmk >= 0) & (jmk < n)
    
    # h at j+k positions (where valid)
    h_jk = np.where(mask1, h[np.clip(jk, 0, n-1)], 0.0)
    h_jmk = np.where(mask2, h[np.clip(jmk, 0, n-1)], 0.0)
    
    part1 = np.where(mask1, 1.0 - h_jk, 0.0)
    part2 = h_jmk
    
    grad = (part1 - part2) @ w
    
    return L, grad, corr

def objective_and_gradient(x, alpha, n):
    h = np.clip(x, 0.0, 1.0)
    val, grad, corr = smooth_max_val_and_grad(h, alpha)
    return val, grad

def run_slsqp(h0, n, alpha, maxiter=200, ftol=1e-10):
    target_sum = n / 2.0
    h0 = project_to_sum(h0, target_sum)
    
    def obj(x):
        return objective_and_gradient(x, alpha, n)[0]
    
    def grad(x):
        return objective_and_gradient(x, alpha, n)[1]
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.success or res.fun < float('inf'):
            return project_to_sum(res.x, target_sum)
    except:
        pass
    return project_to_sum(h0, target_sum)

def projected_gradient_descent(h0, n, alpha, max_iters=5000, lr_init=0.01):
    target_sum = n / 2.0
    h = project_to_sum(h0, target_sum).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    
    for it in range(max_iters):
        val, grad, corr = smooth_max_val_and_grad(h, alpha)
        
        # Gradient step
        h_new = h - lr * grad
        h_new = project_to_sum(h_new, target_sum)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        if c5_new < val + 1e-12:  # approximation check
            h = h_new
            lr = min(lr * 1.01, 0.1)
        else:
            lr *= 0.5
            if lr < 1e-8:
                break
    
    return best_h, best_c5

**Idea: Use the fact that corr_k = sum_i h[i](1-h[i+k]) is bilinear.**

For a fixed k, this is linear in h when the "other" index is fixed. Specifically:
corr_k = sum_{i: i+k < n} h[i] - sum_{i: i+k < n} h[i]*h[i+k]

The first term is linear in h. The second term is the "quadratic" part.

If I fix all h[i+k] values (i.e., fix the "target" positions), then corr_k is linear in h[i] for the "source" positions. This suggests a coordinate descent where at each step, I optimize over the h[i] values while keeping h[i+k] fixed.

But this doesn't directly help since the max is over all k.

**Idea: Use the dual problem.**

The problem min_h max_k corr_k can be written as:
min_h { t : corr_k(h) ≤ t for all k, 0 ≤ h ≤ 1, sum(h) = n/2 }

By LP duality (if we relax the quadratic terms), we get a lower bound. But the problem is non-convex due to the bilinear terms.

However, for a given h, the "active" lag k* is well-defined. The KKT conditions at the optimum would involve:
- ∇_h (sum_k λ_k * corr_k) = 0 for active λ_k > 0
- corr_k* = t (all active constraints are tight)
- sum(λ_k) = 1 (normalization)

This suggests an iterative algorithm:
1. Start with h, compute corr_k for all k
2. Set λ_k ∝ max(0, corr_k - t) (or λ_k = 1 for k = argmax)
3. Solve the subproblem: min_h sum_k λ_k * corr_k s.t. constraints
4. Update t
5. Iterate

This is like a "cutting plane" or "column generation" approach. The subproblem in step 3 is still non-convex (bilinear), but with fewer effective constraints, it might be easier.

Actually, for step 3, if λ is a probability distribution over k, then sum_k λ_k * corr_k is a weighted sum of bilinear terms, which is still non-convex. But it's a specific bilinear form that might be easier to optimize than the max.

Specifically: sum_k λ_k * corr_k = sum_k λ_k * sum_i h[i](1-h[i+k])
= sum_i h[i] * sum_k λ_k(1-h[i+k])

This is a quadratic in h (but not positive semi-definite in general).

import numpy as np
from scipy.optimize import minimize
import time

_n_curr = 0
_target_curr = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m))
    s = np.sum(exp_vals)
    w = exp_vals / s
    L = m + np.log(s) / alpha
    ns = np.arange(n)
    ms = np.arange(2 * n - 1)
    ks = ms - (n - 1)
    jk = ns[:, None] + ks[None, :]
    jmk = ns[:, None] - ks[None, :]
    mask1 = (jk >= 0) & (jk < n)
    mask2 = (jmk >= 0) & (jmk < n)
    h_jk = np.where(mask1, h[np.clip(jk, 0, n - 1)], 0.0)
    h_jmk = np.where(mask2, h[np.clip(jmk, 0, n - 1)], 0.0)
    part1 = np.where(mask1, 1.0 - h_jk, 0.0)
    part2 = h_jmk
    grad = (part1 - part2) @ w
    return L, grad, corr

def _obj_slsqp(x, alpha):
    h = np.clip(x, 0.0, 1.0)
    return smooth_max_val_and_grad(h, alpha)[0]

def _grad_slsqp(x, alpha):
    h = np.clip(x, 0.0, 1.0)
    return smooth_max_val_and_grad(h, alpha)[1]

def _con_fun(x):
    return np.sum(x) - _target_curr

def _con_jac(x):
    return np.ones(_n_curr)

def run_slsqp(h0, n, alpha, maxiter=200, ftol=1e-12):
    global _n_curr, _target_curr
    _n_curr = n
    _target_curr = n / 2.0
    h0 = project_to_sum(h0, _target_curr)
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(_obj_slsqp, h0, args=(alpha,), jac=lambda x: _grad_slsqp(x, alpha),
                       method='SLSQP', bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.fun < 1e6:
            return project_to_sum(res.x, _target_curr)
    except Exception:
        pass
    return project_to_sum(h0, _target_curr)

def interp_h(h_old, n_new):
    n_old = len(h_old)
    h_interp = np.zeros(n_new)
    for i in range(n_new):
        pos = 2.0 * i / n_new / (2.0 / n_old)
        idx = int(np.floor(pos)) % n_old
        frac = pos - np.floor(pos)
        h_interp[i] = (1.0 - frac) * h_old[idx] + frac * h_old[(idx + 1) % n_old]
    return h_interp

def pgd_optimize(h0, n, alpha, max_iters=5000, lr=0.005):
    ts = n / 2.0
    h = project_to_sum(h0, ts).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        val, grad, corr = smooth_max_val_and_grad(h, alpha)
        h_new = h - lr * grad
        h_new = project_to_sum(h_new, ts)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        if c5_new < val + 1e-6:
            h = h_new
            lr = min(lr * 1.005, 0.05)
        else:
            lr *= 0.7
            if lr < 1e-10:
                break
    return best_h, best_c5

def coord_descend(h0, n, target_sum, rng, max_rounds=200):

    ts_final = n_final / 2.0
    best_h = project_to_sum(best_h, ts_final)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_final)
