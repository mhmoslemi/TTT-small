from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)

        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-9) & (h < 1.0 - 1e-9)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:

        h = np.clip(h, 0.0, 1.0)

def get_time_left(t_start, budget_s):
    return budget_s - (time.time() - t_start)

def smooth_max_and_weights(corr, alpha):
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted -= np.max(shifted)  # numerical stability
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    sm = m + np.log(np.sum(exp_vals) + 1e-300) / alpha

    sm, w = smooth_max_and_weights(corr, alpha)

    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break

        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)

def smooth_max_value(corr, alpha):
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = shifted - np.max(shifted)
    exp_vals = np.exp(shifted)
    return m + np.log(np.sum(exp_vals) + 1e-300) / alpha

def compute_gradient(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = shifted - np.max(shifted)

        _, grad = compute_gradient(h, alpha)
        grad = grad - np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        grad = grad / gnorm
        vel = momentum * vel + (1.0 - momentum) * grad
        norm_vel = np.linalg.norm(vel)
        if norm_vel > 1e-12:
            vel = vel / norm_vel
        h_new = h + lr * vel
        h_new = project_to_sum(h_new, target_sum)
        obj_new = compute_c5(h_new)
        if obj_new < obj - 1e-15:
            h = h_new
            obj = obj_new
            patience = 0
            if obj < best_obj:
                best_obj = obj
                best_h = h.copy()
        else:
            patience += 1
            lr *= 0.85
            vel *= 0.5
            if patience > 50:
                break
    return best_h, best_obj

def sa_optimize(h, target_sum, t_start, budget_s, seed, max_time):
    n = len(h)
    rng = np.random.RandomState(seed)
    h = h.copy()
    obj = compute_c5(h)
    best_h = h.copy()
    best_obj = obj
    T = 0.008
    T_end = 0.0003
    remaining = min(get_time_left(t_start, budget_s), max_time)
    max_iters = int(remaining * 5000)
    if max_iters < 100:
        max_iters = 100
    cool = (T_end / T) ** (1.0 / max(1, max_iters))
    
    for it in range(max_iters):
        if get_time_left(t_start, budget_s) < 3:
            break
        move_type = rng.random()
        if move_type < 0.25:
            idx = rng.randint(n)
            delta = rng.normal(0, 0.06)
            h_new = h.copy()
            h_new[idx] = np.clip(h_new[idx] + delta, 0, 1)
            h_new = project_to_sum(h_new, target_sum)
        elif move_type < 0.5:
            blk = rng.randint(3, min(20, n // 3))
            start = rng.randint(0, n - blk)
            noise = rng.normal(0, 0.04, blk)
            h_new = h.copy()
            h_new[start:start + blk] = np.clip(h_new[start:start + blk] + noise, 0, 1)
            h_new = project_to_sum(h_new, target_sum)
        elif move_type < 0.75:
            _, grad = compute_gradient(h, 30.0)
            grad = grad - np.mean(grad)
            gn = np.linalg.norm(grad)
            if gn > 1e-10:
                grad = grad / gn
            idx = rng.randint(n)
            delta = rng.normal(0, 0.04)
            h_new = h.copy()
            h_new[idx] = np.clip(h_new[idx] + delta + 0.03 * grad[idx], 0, 1)
            h_new = project_to_sum(h_new, target_sum)
        else:
            i, j = rng.choice(n, size=2, replace=False)
            delta = rng.uniform(-0.1, 0.1)
            h_new = h.copy()
            h_new[i] = np.clip(h_new[i] + delta, 0, 1)
            h_new[j] = np.clip(h_new[j] - delta, 0, 1)
            h_new = project_to_sum(h_new, target_sum)
        
        obj_new = compute_c5(h_new)
        if obj_new < obj or rng.random() < np.exp(-(obj_new - obj) / T):
            h = h_new
            obj = obj_new
            if obj < best_obj:
                best_obj = obj
                best_h = h.copy()
        T = max(T * cool, T_end)
    
    return best_h, best_obj

def coord_descent_exact(h, target_sum, t_start, budget_s, max_rounds):
    n = len(h)
    rng = np.random.RandomState(999)
    h = h.copy()
    obj = compute_c5(h)
    
    for rnd in range(max_rounds):
        if get_time_left(t_start, budget_s) < 5:
            break
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            old = h[i]
            best_t = old
            best_o = obj
            for t in np.linspace(0, 1, 60):
                if abs(t - old) < 1e-4:
                    continue
                h_new = h.copy()
                h_new[i] = t
                h_new = project_to_sum(h_new, target_sum)
                o = compute_c5(h_new)
                if o < best_o:
                    best_o = o
                    best_t = t
            if best_t != old:
                h[i] = best_t
                h = project_to_sum(h, target_sum)
                obj = best_o
                improved = True
        if not improved:
            break
    return h, obj

def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return h_new
