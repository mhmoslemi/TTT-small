import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_obj_and_grad(h, beta, penalty=1000.0):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    c_full = np.correlate(h, g, mode="full")
    c = c_full * dx
    
    cm = c.max()
    if beta * cm > 700:
        # Numerical stability: just use the max
        obj = cm
        # Gradient will be approximate
        w = np.zeros(len(c))
        active = (c == cm)
        w[active] = 1.0 / np.sum(active)
    else:
        c_shifted = c - cm
        exp_vals = np.exp(beta * c_shifted)
        log_sum = np.log(np.sum(exp_vals))
        obj = cm + log_sum / beta
        w = exp_vals / np.sum(exp_vals)
    
    # Gradient of soft-max
    grad_sm = np.zeros(n)
    # Vectorized computation
    # For each i, Term A: sum over k in [i, min(i+n-1, 2n-2)] of w[k] * (1-h[i-(k-n+1)])
    # For each i, Term B: sum over k in [max(0,n-1-i), min(2n-2, 2n-2-i)] of w[k] * h[i+(k-n+1)]
    
    for i in range(n):
        # Term A
        k_lo = i
        k_hi = min(i + n - 1, 2*n - 2)
        if k_lo <= k_hi:
            idx = i - (np.arange(k_lo, k_hi+1) - (n-1))
            # idx = i - k + n - 1 = i + n - 1 - k
            # When k = k_lo = i: idx = n-1-i
            # When k = k_hi: idx = n-1-i-(k_hi-i) = n-1-k_hi
            grad_sm[i] += np.dot(w[k_lo:k_hi+1], g[idx])
        
        # Term B
        k_lo2 = max(0, n - 1 - i)
        k_hi2 = min(2*n - 2, 2*n - 2 - i)
        if k_lo2 <= k_hi2:
            idx2 = i + (np.arange(k_lo2, k_hi2+1) - (n-1))
            # idx2 = i + k - n + 1
            # When k = k_lo2 = n-1-i: idx2 = 0
            # When k = k_hi2: idx2 = i + k_hi2 - n + 1
            grad_sm[i] -= np.dot(w[k_lo2:k_hi2+1], h[idx2])
    
    grad_sm *= dx
    
    # Add penalty gradient
    penalty_grad = 2 * penalty * (np.sum(h) - n/2) * np.ones(n)
    
    total_obj = obj + penalty * (np.sum(h) - n/2)**2
    total_grad = grad_sm + penalty_grad
    
    return total_obj, total_grad

from scipy.interpolate import interp1d

def compute_c5_from_h(h):

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
        else:
            mask = h > 1e-14
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h


def soft_max_obj_and_grad(h, beta, penalty=500.0):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    c_full = np.correlate(h, g, mode="full")
    c = c_full * dx
    
    cm = np.max(c)
    c_shifted = c - cm
    max_exp = beta * np.max(c_shifted)
    if max_exp > 50:
        # Use log-sum-exp trick
        log_sum = np.max(c_shifted) + np.log(np.sum(np.exp(beta * (c_shifted - np.max(c_shifted)))))
        obj = cm + log_sum / beta
        exp_vals = np.exp(beta * (c_shifted - np.max(c_shifted)))
        w = exp_vals / np.sum(exp_vals)
    else:
        exp_vals = np.exp(beta * c_shifted)
        obj = cm + np.log(np.sum(exp_vals)) / beta
        w = exp_vals / np.sum(exp_vals)
    
    grad_sm = np.zeros(n)
    for i in range(n):
        k_lo = i
        k_hi = min(i + n - 1, 2 * n - 2)
        if k_lo <= k_hi:
            idx = i + n - 1 - np.arange(k_lo, k_hi + 1)
            grad_sm[i] += np.dot(w[k_lo:k_hi + 1], g[idx])
        k_lo2 = max(0, n - 1 - i)
        k_hi2 = min(2 * n - 2, 2 * n - 2 - i)
        if k_lo2 <= k_hi2:

            grad_sm[i] -= np.dot(w[k_lo2:k_hi2 + 1], h[idx2])
    
    grad_sm *= dx
    penalty_grad = 2.0 * penalty * (np.sum(h) - n / 2.0)
    total_obj = obj + penalty * (np.sum(h) - n / 2.0) ** 2
    total_grad = grad_sm + penalty_grad
    return total_obj, total_grad


def lbfgsb_optimize(h0, beta=50.0, penalty=500.0, maxiter=200):
    n = len(h0)
    result = minimize(soft_max_obj_and_grad, h0, jac=True, method='L-BFGS-B',
                      bounds=[(0, 1)] * n,
                      options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8,
                               'maxfun': 5000})
    if result.success or result.x is not None:
        return project_feasible(result.x, n), compute_c5_from_h(project_feasible(result.x, n))
    return h0, compute_c5_from_h(h0)


def slsqp_solve(h_init, n, maxiter=2000):
    c5_init = compute_c5_from_h(h_init)
    
    def obj_func(x):
        h = x[:-1]
        t = x[-1]
        return t
    
    def ineq_func(x):
        h = x[:-1]
        t = x[-1]
        dx = 2.0 / len(h)
        c = np.correlate(h, 1.0 - h, mode="full")
        return t - c * dx
    
    def eq_func(x):
        h = x[:-1]
        return np.sum(h) - len(h) / 2.0
    
    def eq_jac_func(x):
        n_loc = len(x) - 1
        jac = np.zeros(len(x))
        jac[:n_loc] = 1.0
        return jac
    
    x0 = np.concatenate([h_init, [c5_init + 0.02]])
    bounds = [(0.0, 1.0)] * n + [(max(0.001, c5_init * 0.8), 2.0)]
    constraints = [
        {'type': 'ineq', 'fun': ineq_func},
        {'type': 'eq', 'fun': eq_func, 'jac': eq_jac_func}
    ]
    result = minimize(obj_func, x0, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-14, 'maxls': 50})
    if result.x is not None:
        h_out = project_feasible(result.x[:-1], n)
        return h_out, compute_c5_from_h(h_out)
    return h_init, c5_init

        return x[-1]

        j = np.zeros(len(x))

        return j

        kh2 = min(2 * n - 2, 2 * n - 2 - i)
        if kl2 <= kh2:
            idx2 = i + np.arange(kl2, kh2 + 1) - n + 1
            grad[i] -= np.dot(w[kl2:kh2 + 1], h[idx2])
    grad *= dx
    pe = np.sum(h) - n / 2.0
    obj += penalty * pe * pe
    grad = grad + 2.0 * penalty * pe
    return obj, grad


def lbfgs_optimize(h0, beta, penalty, maxiter):
    n = len(h0)
    res = minimize(sm_obj_grad, h0, args=(beta, penalty), jac=True,
                   method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-9,
                            'maxfun': 4000, 'maxls': 50, 'm': 30})
    h_out = project_feasible(res.x, n)
    return h_out, compute_c5_from_h(h_out)


def slsqp_refine(h0, maxiter):
    n = len(h0)
    c5_0 = compute_c5_from_h(h0)

    def objf(x):
        return x[-1]

    def ineqf(x):
        h = x[:-1]
        dx = 2.0 / len(h)
        c = np.correlate(h, 1.0 - h, mode="full")
        return x[-1] - c * dx

    def eqf(x):
        return np.sum(x[:-1]) - len(x[:-1]) / 2.0

    def eqjac(x):
        nv = len(x) - 1
        j = np.zeros(len(x))
        j[:nv] = 1.0
        return j

    x0 = np.concatenate([h0, np.array([c5_0 + 0.02])])
    bnds = [(0.0, 1.0)] * n + [(max(0.001, c5_0 * 0.85), 2.0)]
    cons = [{'type': 'ineq', 'fun': ineqf},
            {'type': 'eq', 'fun': eqf, 'jac': eqjac}]
    res = minimize(objf, x0, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-14, 'maxls': 100})
    if res.x is not None:
        h_out = project_feasible(res.x[:-1], n)
        return h_out, compute_c5_from_h(h_out)
    return h0, c5_0


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(135, 0.5)

    n = len(h0)
    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c5 = compute_c5_from_h(best_h)

    def el():
        return time.time() - t0

    for beta in [5, 15, 40, 100, 250, 600, 1500]:
        if el() > budget_s - 120:
            break
        hn, cn = lbfgs_optimize(best_h, beta, 3000.0, 300)
        if cn < best_c5 - 1e-10:
            best_h, best_c5 = hn, cn

    if el() < budget_s - 180:
        hn, cn = slsqp_refine(best_h, 3000)
        if cn < best_c5 - 1e-10:
            best_h, best_c5 = hn, cn

    n_perturb = 50
    for att in range(n_perturb):
        if el() > budget_s - 100:
            break
        sc = 0.10 * (1.0 - att / float(n_perturb)) + 0.008
        pt = att % 5
        nn = len(best_h)
        if pt == 0:
            hp = best_h + sc * rng.standard_normal(nn)
        elif pt == 1:
            f = 1 + rng.integers(1, max(2, nn // 5))
            ph = rng.uniform(0, 2 * np.pi)
            hp = best_h + sc * np.sin(2 * np.pi * f * np.arange(nn) / nn + ph)
        elif pt == 2:
            hp = best_h.copy()
            bs = max(3, nn // 12)
            for _ in range(rng.integers(1, 4)):
                st = rng.integers(0, max(1, nn - bs))
                hp[st:st + bs] += sc * rng.choice([-1.0, 1.0])
        elif pt == 3:
            hp = best_h.copy()
            thr = 0.25 + 0.5 * rng.random()
            hp[hp > thr] += sc
            hp[hp < 1.0 - thr] -= sc
        else:
            hp = best_h.copy()
            mask = rng.random(nn) < 0.35
            hp[mask] += sc * rng.standard_normal(int(np.sum(mask)))

        hp = project_feasible(hp, nn)
        hn, cn = lbfgs_optimize(hp, 100, 3000.0, 150)
        if el() < budget_s - 100:
            hs, cs = slsqp_refine(hn, 800)
            if cs < cn:
                hn, cn = hs, cs
        if cn < best_c5 - 1e-10:
            best_h, best_c5 = hn, cn

    for nt in [270, 405, 675]:
        if el() > budget_s - 250:
            break
        xo = np.linspace(0, 2, len(best_h), endpoint=False)
        xn = np.linspace(0, 2, nt, endpoint=False)
        fi = interp1d(xo, best_h, kind='linear', fill_value="extrapolate", bounds_error=False)
        hi = np.clip(fi(xn), 0.0, 1.0)
        hi = project_feasible(hi, nt)
        hn, cn = lbfgs_optimize(hi, 40, 3000.0, 200)
        if el() < budget_s - 200:
            hn2, cn2 = lbfgs_optimize(hn, 200, 3000.0, 150)
            if cn2 < cn:
                hn, cn = hn2, cn2
        if el() < budget_s - 200:
            hs, cs = slsqp_refine(hn, 1200)
            if cs < cn:
                hn, cn = hs, cs
        if cn < best_c5 - 1e-10:
            best_h, best_c5 = hn, cn

    if el() < budget_s - 60:
        for _ in range(8):
            if el() > budget_s - 20:
                break
            hp = best_h + 0.008 * rng.standard_normal(len(best_h))
            hp = project_feasible(hp, len(best_h))
            hn, cn = lbfgs_optimize(hp, 2000, 5000.0, 80)
            if cn < best_c5 - 1e-12:
                best_h, best_c5 = hn, cn

    if el() < budget_s - 10:
        hn, cn = slsqp_refine(best_h, 2000)
        if cn < best_c5:
            best_h, best_c5 = hn, cn

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5_from_h(best_h)
    return best_h, best_c5, len(best_h)
