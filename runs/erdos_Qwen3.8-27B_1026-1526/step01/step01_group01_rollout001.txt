import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
            idx = np.where(mask)[0]
            if len(idx) == 0:
                break
            per = diff / len(idx)
            h[idx] = np.minimum(h[idx] + per, 1.0)
        else:
            mask = h > 1e-13
            idx = np.where(mask)[0]
            if len(idx) == 0:
                break
            per = diff / len(idx)
            h[idx] = np.maximum(h[idx] + per, 0.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(h)[::-1]
            for i in order:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
                if abs(diff) < 1e-14:
                    break
        else:
            order = np.argsort(h)
            for i in order:
                rem = min(-diff, h[i])
                h[i] -= rem
                diff += rem
                if abs(diff) < 1e-14:
                    break
    return h


def smooth_max_obj(x, n, beta):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500.0, 500.0)
    val = m + np.log(np.sum(np.exp(shifted))) / beta
    return val * (2.0 / n)


def slsqp_opt(h0, n, max_iter, beta, timeout):
    t_start = time.time()
    expired = [False]

    def obj(x):
        if time.time() - t_start > timeout:
            expired[0] = True
        h = np.clip(x, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode="full")
        m = np.max(corr)
        shifted = np.clip(beta * (corr - m), -500.0, 500.0)
        val = m + np.log(np.sum(np.exp(shifted))) / beta
        return val * (2.0 / n)

    def constraint_fun(x):
        return np.sum(x) - n / 2.0

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun}]
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-13, 'iprint': 0})
        if expired[0]:
            return None, None
        h_res = project_feasible(res.x, n)
        c5 = compute_c5(h_res)
        return h_res, c5
    except Exception:
        return None, None


def de_search(h_best, n, pop_size, generations, timeout, rng):
    t_start = time.time()
    pop = [h_best.copy()]
    for _ in range(pop_size - 1):
        sigma = 0.1 + 0.3 * rng.random()
        pert = h_best + rng.normal(0.0, sigma, n)
        pop.append(project_feasible(pert, n))

    scores = np.array([compute_c5(p) for p in pop])
    best_idx = int(np.argmin(scores))
    best_c5 = scores[best_idx]
    best_h = pop[best_idx].copy()

    for gen in range(generations):
        if time.time() - t_start > timeout:
            break
        for i in range(1, pop_size):
            if time.time() - t_start > timeout:
                break
            a, b, c = rng.choice(pop_size, 3, replace=False)
            F = 0.4 + 0.4 * rng.random()
            candidate = pop[a] + F * (pop[b] - pop[c])
            candidate = project_feasible(candidate, n)
            sc = compute_c5(candidate)
            if sc < scores[i]:
                pop[i] = candidate
                scores[i] = sc
                if sc < best_c5:
                    best_c5 = sc
                    best_h = candidate.copy()
            elif rng.random() < 0.1:
                pop[i] = candidate
                scores[i] = sc
    return best_h, best_c5


def multi_frequency_init(n, rng):
    x = np.arange(n, dtype=float) / n
    h = np.zeros(n)
    n_freqs = rng.randint(3, 8)
    for f in range(n_freqs):
        freq = rng.uniform(1.0, 8.0)
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.uniform(0.1, 0.3)
        h += amp * np.sin(2.0 * np.pi * freq * x + phase)
    h = 0.5 + 0.4 * h
    return project_feasible(h, n)


def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    return np.interp(x_new, x_old, h_old)


def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    best_h = None
    best_c5 = 1e9
    best_n = 190

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()

        best_h = h_init
        best_c5 = compute_c5(h_init)

    if best_h is None:
        best_h = np.ones(best_n) * 0.5
        best_c5 = compute_c5(best_h)

        return time.time() - t0

            break
        time_left = min(budget_s - elapsed() - 240, 180)
        if time_left < 20:
            break
        h_opt, c5_opt = slsqp_opt(best_h, best_n, 400, beta, time_left)
        if h_opt is not None and c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt

    # Phase 2: Differential Evolution for global exploration

        if time_left > 40:

            if c5_de < best_c5:

            # Refine DE result
            if elapsed() < budget_s - 180:
                h_opt, c5_opt = slsqp_opt(h_de, best_n, 300, 100.0, min(budget_s - elapsed() - 170, 120))
                if h_opt is not None and c5_opt < best_c5:
                    best_h, best_c5 = h_opt, c5_opt

    # Phase 3: Multi-start local search with diverse perturbations
    no_improve = 0
    max_no_improve = 50
    while elapsed() < budget_s - 200 and no_improve < max_no_improve:
        time_left = budget_s - elapsed() - 190
        if time_left < 30:
            break
        sigma = 0.06 + 0.12 * (0.92 ** min(no_improve, 30))
        if rng.random() < 0.15:
            sigma = 0.35 + 0.2 * rng.random()
        h_start = project_feasible(best_h + rng.normal(0.0, sigma, best_n), best_n)
        beta = 60.0 + 60.0 * rng.random()
        max_iter = int(min(300, max(100, time_left * 3)))
        h_opt, c5_opt = slsqp_opt(h_start, best_n, max_iter, beta, min(time_left, 60))
        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
            no_improve = 0
        else:
            no_improve += 1

    # Phase 4: Try different resolutions
    for n_try in [best_n + 10, best_n + 30, best_n + 60, best_n + 100]:
        if elapsed() > budget_s - 120:
            break
        time_left = budget_s - elapsed() - 110
        if time_left < 80:
            break
        h_interp = interpolate_h(best_h, best_n, n_try)
        h_interp = project_feasible(h_interp, n_try)
        h_opt, c5_opt = slsqp_opt(h_interp, n_try, 400, 80.0, min(time_left - 20, 150))
        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, n_try
            # Quick refinement at new resolution
            if elapsed() < budget_s - 100:
                h_opt2, c5_opt2 = slsqp_opt(best_h, best_n, 300, 200.0, min(budget_s - elapsed() - 90, 100))
                if h_opt2 is not None and c5_opt2 < best_c5:
                    best_h, best_c5 = h_opt2, c5_opt2

    # Phase 5: Coarser resolution attempt (easier landscape)
    if elapsed() < budget_s - 100:
        n_coarse = max(50, best_n - 40)
        h_coarse = interpolate_h(best_h, best_n, n_coarse)

        time_left = min(budget_s - elapsed() - 90, 120)
        h_opt, c5_opt = slsqp_opt(h_coarse, n_coarse, 500, 100.0, time_left)
        if h_opt is not None:
            # Upsample back
            h_up = interpolate_h(h_opt, n_coarse, best_n)
            h_up = project_feasible(h_up, best_n)
            c5_up = compute_c5(h_up)

    # Phase 6: Final high-beta refinement
    if elapsed() < budget_s - 30:
        time_left = min(budget_s - elapsed() - 20, 100)
        h_opt, c5_opt = slsqp_opt(best_h, best_n, 500, 300.0, time_left)
        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 7: Ultra-fine final polish

        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 8: Try a few structured inits if time permits
    if elapsed() < budget_s - 40:
        for _ in range(3):
            if elapsed() > budget_s - 35:

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, len(best_h))
