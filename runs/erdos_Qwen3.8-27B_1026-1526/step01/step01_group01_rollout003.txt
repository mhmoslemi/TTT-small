import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = beta * (x - m)
    shifted = np.clip(shifted, -500.0, 500.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def objective(x, n, beta, penalty=0.0):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    val = smooth_max(corr, beta) * dx
    if penalty > 0:
        val += penalty * (np.sum(x) - n/2.0)**2
    return val

def sum_constraint(x, n):
    return np.sum(x) - n/2.0

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
        else:
            mask = h > 1e-12
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h

def optimize_slsqp(h0, n, beta, max_iter=500, ftol=1e-12):

            i2_hi = j2_hi - offset
            grad[j2_lo:j2_hi] -= w * h[i2_lo:i2_hi]
    
    # Projected gradient step
    h_new = h - lr * grad
    h_new = project_feasible(h_new, n)
    return h_new

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
        else:
            mask = h > 1e-13
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    h = np.clip(h, 0.0, 1.0)
    return h


def obj_smooth(h, n, beta, penalty):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    val = smooth_max(corr, beta) * dx
    if penalty > 0:
        val += penalty * (np.sum(h) - n / 2.0) ** 2
    return val


def sum_eq(x, n):
    return np.sum(x) - n / 2.0


def optimize_slsqp(h0, n, beta, max_iter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, args=(n, beta, 0.0),
                       options={'maxiter': max_iter, 'ftol': ftol, 'iprint': 0})
        h_res = project_feasible(res.x, n)
        c5 = compute_c5(h_res)
        return h_res, c5
    except Exception:
        return None, None


def optimize_lbfgs(h0, n, beta, penalty, max_iter):
    try:
        res = minimize(obj_smooth, h0, method='L-BFGS-B',
                       args=(n, beta, penalty),
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-12})
        h_res = project_feasible(res.x, n)
        c5 = compute_c5(h_res)
        return h_res, c5
    except Exception:
        return None, None


def optimize_t_form(h0, n, max_iter):
    h0 = project_feasible(h0, n)
    n_vars = n + 1

    def obj_t(vars):
        h = vars[:n]
        t = vars[n]
        return t

    def h_bounds():
        return [(0.0, 1.0)] * n + [(0.0, 1.0)]

    def ineq_top(vars, k_top):
        h = vars[:n]
        t = vars[n]
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        top_idx = np.argsort(corr)[-k_top:]
        return t - corr[top_idx]

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x[:n]) - n / 2.0}]
    for k_top in [10, 30]:
        cons.append({'type': 'ineq', 'fun': lambda x, kt=k_top: ineq_top(x, kt)})

    init_t = compute_c5(h0) + 0.05
    x0 = np.concatenate([h0, [init_t]])
    bounds = h_bounds()
    try:
        res = minimize(obj_t, x0, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'iprint': 0})
        h_res = project_feasible(res.x[:n], n)
        c5 = compute_c5(h_res)

def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    return np.interp(x_new, x_old, h_old)

    h_p = h + rng.normal(0.0, sigma, n)
    return project_feasible(h_p, n)


def make_block_h(n, n_blocks, rng):
    h = np.zeros(n)
    block_size = n // n_blocks
    total_ones = n // 2
    fill_per_block = total_ones // n_blocks
    for b in range(n_blocks):
        start = b * block_size
        end = min(start + block_size, n)
        n_fill = min(fill_per_block, end - start)
        offsets = rng.choice(end - start, size=n_fill, replace=False)
        h[start + offsets] = 1.0
    return project_feasible(h, n)


def make_multisine(n, n_harm, rng):

        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.05, 0.3) / (k + 1)
        i = np.arange(n, dtype=float)
        h += amp * np.sin(2.0 * np.pi * freq * i / n + phase)
    return project_feasible(h, n)


def projected_subgradient(h, n, beta, lr, rng):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    shifted = np.clip(beta * (corr - np.max(corr)), -500.0, 500.0)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    v = 1.0 - h
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-10:
            continue
        offset = m - (n - 1)
        j_lo = max(0, -offset)
        j_hi = min(n, n - offset)
        if j_hi > j_lo:
            grad[j_lo:j_hi] += w * v[j_lo + offset:j_hi + offset]
        j2_lo = max(0, j_lo + offset)
        j2_hi = min(n, j_hi + offset)
        if j2_hi > j2_lo:
            i2_lo = j2_lo - offset
            i2_hi = j2_hi - offset
            grad[j2_lo:j2_hi] -= w * h[i2_lo:i2_hi]
    h_new = h - lr * grad * dx
    h_new = project_feasible(h_new, n)
    return h_new
