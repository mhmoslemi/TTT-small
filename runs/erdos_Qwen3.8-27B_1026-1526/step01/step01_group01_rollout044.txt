def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
            avail = (1.0 - h)[mask]
            total = avail.sum()
            if total < 1e-15:
                break
            h[mask] += avail * diff / total
        else:
            mask = h > 1e-13
            avail = h[mask]
            total = avail.sum()
            if total < 1e-15:
                break
            h[mask] -= avail * (-diff) / total
        h = np.clip(h, 0.0, 1.0)
    diff = n / 2.0 - np.sum(h)
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = np.where(h < 1.0)[0]
            if len(idx) > 0:
                h[idx[0]] = min(h[idx[0]] + diff, 1.0)
        else:
            idx = np.where(h > 0.0)[0]
            if len(idx) > 0:
                h[idx[0]] = max(h[idx[0]] + diff, 0.0)
    return h


def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def obj_smoothed(x, n, beta):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, beta) * (2.0 / n)


def sum_eq(x, n):
    return np.sum(x) - n / 2.0


def compute_subgrad_single(h, n, shift):
    grad = np.zeros(n)
    if shift >= 0:
        i_max = n - 1 - shift
        for i in range(i_max + 1):
            i2 = i + shift
            grad[i] += (1.0 - h[i2])
            grad[i2] += (-h[i])
    else:
        s = -shift
        i_min = s
        for i in range(i_min, n):
            i2 = i - s
            grad[i] += (1.0 - h[i2])
            grad[i2] += (-h[i])
    return grad


def compute_active_subgrad(h, n, rel_eps=0.02):
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(corr)
    active_mask = corr >= max_val - rel_eps * max_val
    active_lags = np.where(active_mask)[0]
    if len(active_lags) == 0:
        active_lags = np.array([np.argmax(corr)])
    grad = np.zeros(n)
    for lag_idx in active_lags:
        shift = lag_idx - n + 1
        grad += compute_subgrad_single(h, n, shift)
    grad /= len(active_lags)
    return grad, max_val


def fw_vertex(grad, n):
    order = np.argsort(grad)
    v = np.zeros(n)
    n_ones = n // 2
    v[order[:n_ones]] = 1.0
    return v


def fw_step_with_search(h, grad, n, max_t=0.5, n_tries=4):
    v = fw_vertex(grad, n)
    best_h = h.copy()
    best_f = compute_c5(h)
    for i in range(n_tries):
        t = max_t * (1.0 - i / n_tries)
        t = max(0.01, min(t, 1.0))
        h_trial = (1.0 - t) * h + t * v
        h_trial = project_feasible(h_trial, n)
        f_trial = compute_c5(h_trial)
        if f_trial < best_f - 1e-12:
            best_f = f_trial
            best_h = h_trial
    return best_h, best_f


def slsqp_optimize(h0, n, beta, max_iter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq, 'args': (n,)}]
    try:
        res = minimize(obj_smoothed, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, args=(n, beta),
                       options={'maxiter': max_iter, 'ftol': 1e-13, 'iprint': 0})
        h_res = project_feasible(res.x, n)
        return h_res, compute_c5(h_res)
    except Exception:
        return None, None


def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    return np.interp(x_new, x_old, h_old)


def coordinate_polish(h, n, max_passes=3):
    best_f = compute_c5(h)
    for _pass in range(max_passes):
        improved = False
        order = np.random.permutation(n)
        for i in order:
            for delta in [0.1, -0.1, 0.2, -0.2, 0.05, -0.05]:
                h_trial = h.copy()
                h_trial[i] = np.clip(h_trial[i] + delta, 0.0, 1.0)
                h_trial = project_feasible(h_trial, n)
                f_trial = compute_c5(h_trial)
                if f_trial < best_f - 1e-12:
                    h = h_trial
                    best_f = f_trial
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return project_feasible(h, n), best_f


def make_perturbed(h, n, sigma, rng):
    h_p = h + rng.normal(0.0, sigma, n)
    return project_feasible(h_p, n)


def make_binary_start(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h


def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    best_h = None
    best_c5 = 1e9
    best_n = 190

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
        h_init = project_feasible(h_init, n_init)
        best_h = h_init
        best_c5 = compute_c5(h_init)
        best_n = n_init

    if best_h is None:
        best_h = np.ones(190) * 0.5
        best_c5 = compute_c5(best_h)
        best_n = 190

    n = best_n

    phase1_end = t0 + 0.35 * budget_s
    phase2_end = t0 + 0.55 * budget_s
    phase3_end = t0 + 0.80 * budget_s

    # Phase 1: Subgradient Frank-Wolfe with momentum
    h = best_h.copy()
    velocity = np.zeros(n)
    mu = 0.7
    no_improve = 0

    while time.time() < phase1_end:
        grad, fval_scaled = compute_active_subgrad(h, n, rel_eps=0.015)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-10:
            break
        # Momentum update
        velocity = mu * velocity - grad / max(grad_norm, 1e-10)
        grad_eff = velocity
        h_new, f_new = fw_step_with_search(h, grad_eff, n, max_t=0.6, n_tries=5)
        if f_new < best_c5 - 1e-11:
            best_h = h_new.copy()
            best_c5 = f_new
            no_improve = 0
        else:
            h = h_new
            no_improve += 1
        h = h_new
        if no_improve > 200:
            # Restart with perturbation
            h = make_perturbed(best_h, n, 0.15, rng)
            velocity = np.zeros(n)
            no_improve = 0
        if no_improve > 600:
            break

    # Phase 2: SLSQP polish with increasing beta
    while time.time() < phase2_end:
        for beta in [40, 80, 150, 300, 600]:
            if time.time() > phase2_end - 5:
                break
            h_opt, c5_opt = slsqp_optimize(best_h, best_n, beta, 400)
            if h_opt is not None and c5_opt < best_c5:
                best_h, best_c5 = h_opt, c5_opt

    # Coordinate polish
    if time.time() < phase3_end - 30:
        h_cp, f_cp = coordinate_polish(best_h, best_n, max_passes=4)
        if f_cp < best_c5:
            best_h, best_c5 = h_cp, f_cp

    # Phase 3: Multi-start exploration
    while time.time() < phase3_end:
        sigma = 0.08 + 0.12 * rng.random()
        h_start = make_perturbed(best_h, best_n, sigma, rng)
        if rng.random() < 0.1:
            h_start = make_binary_start(best_n, rng)
        # Quick FW on the perturbed start
        for _ in range(100):
            grad, _ = compute_active_subgrad(h_start, best_n, rel_eps=0.02)
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-10:
                break
            h_start, _ = fw_step_with_search(h_start, grad / max(grad_norm, 1e-10),
                                             best_n, max_t=0.4, n_tries=3)
        c5_start = compute_c5(h_start)
        if c5_start < best_c5:
            h_opt, c5_opt = slsqp_optimize(h_start, best_n, 100, 200)
            if h_opt is not None and c5_opt < best_c5:
                best_h, best_c5 = h_opt, c5_opt
        elif c5_start < best_c5 + 0.005:
            # Close enough, try SLSQP polish
            h_opt, c5_opt = slsqp_optimize(h_start, best_n, 80, 150)
            if h_opt is not None and c5_opt < best_c5:
                best_h, best_c5 = h_opt, c5_opt

    # Phase 4: Try higher resolution
    for n_new in [238, 286, 334]:
        if time.time() > budget_s - 120:
            break
        h_interp = interpolate_h(best_h, best_n, n_new)
        h_interp = project_feasible(h_interp, n_new)
        # Quick FW at new resolution
        for _ in range(200):
            grad, _ = compute_active_subgrad(h_interp, n_new, rel_eps=0.02)
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-10:
                break
            h_interp, _ = fw_step_with_search(h_interp, grad / max(grad_norm, 1e-10),
                                              n_new, max_t=0.4, n_tries=3)
        c5_new_res = compute_c5(h_interp)
        if c5_new_res < best_c5:
            h_opt, c5_opt = slsqp_optimize(h_interp, n_new, 80, 300)
            if h_opt is not None and c5_opt < best_c5:
                best_h, best_c5, best_n = h_opt, c5_opt, n_new

    # Final polish
    remaining = budget_s - (time.time() - t0)
    if remaining > 60:
        h_opt, c5_opt = slsqp_optimize(best_h, best_n, 200, 500)
        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    if remaining > 100:
        h_opt, c5_opt = slsqp_optimize(best_h, best_n, 500, 400)
        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Final coordinate polish
    if time.time() < budget_s - 15:
        h_cp, f_cp = coordinate_polish(best_h, best_n, max_passes=2)
        if f_cp < best_c5:
            best_h, best_c5 = h_cp, f_cp

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, len(best_h))
