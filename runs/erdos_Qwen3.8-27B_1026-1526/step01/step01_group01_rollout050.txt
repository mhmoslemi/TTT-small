    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)

    return m + np.log(np.sum(np.exp(shifted))) / beta

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def obj_func(x, n, beta):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, beta) * (2.0 / n)


def sum_eq(x, n):
    return np.sum(x) - n / 2.0


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
        else:
            mask = h > 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    diff = n / 2.0 - np.sum(h)
    if abs(diff) > 1e-15:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(h[idx] + diff, 1.0)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(h[idx] + diff, 0.0)
    return h


def subgradient_at(h, tol=1e-6):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(corr)
    active = np.where(corr >= max_val - tol)[0]
    g = np.zeros(n)
    for lag_idx in active:
        k = lag_idx - (n - 1)
        lo1 = max(0, k)
        hi1 = min(n - 1, n - 1 + k)
        if lo1 <= hi1:
            idx = np.arange(lo1, hi1 + 1)
            g[idx] += (1.0 - h[idx - k])
        lo2 = max(0, -k)
        hi2 = min(n - 1, n - 1 - k)
        if lo2 <= hi2:
            idx2 = np.arange(lo2, hi2 + 1)
            g[idx2] += (-h[idx2 + k])
    g /= len(active)
    return g


def subgradient_single_lag(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    lag_idx = int(np.argmax(corr))
    k = lag_idx - (n - 1)
    g = np.zeros(n)
    lo1 = max(0, k)
    hi1 = min(n - 1, n - 1 + k)
    if lo1 <= hi1:
        idx = np.arange(lo1, hi1 + 1)
        g[idx] += (1.0 - h[idx - k])
    lo2 = max(0, -k)
    hi2 = min(n - 1, n - 1 - k)
    if lo2 <= hi2:
        idx2 = np.arange(lo2, hi2 + 1)
        g[idx2] += (-h[idx2 + k])
    return g


def subgradient_polish(h, n, max_iter=500, use_single=False):
    best_c5 = compute_c5(h)
    h = h.copy()
    stall = 0
    for it in range(max_iter):
        if use_single:
            g = subgradient_single_lag(h)
        else:
            g = subgradient_at(h, tol=1e-6)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        improved = False
        alpha = 0.5
        for _ in range(40):
            h_trial = project_feasible(h - alpha * g, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5 - 1e-13:
                h = h_trial
                best_c5 = c5_trial
                improved = True
                stall = 0
                break
            alpha *= 0.5
        if not improved:
            stall += 1
            if stall > 30:
                break
    return h, best_c5


def optimize_slsqp(h0, n, max_iter, beta):
    h0 = project_feasible(h0, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq, 'args': (n,)}]
    try:
        res = minimize(obj_func, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, args=(n, beta),
                       options={'maxiter': max_iter, 'ftol': 1e-14, 'iprint': 0})
        h_res = project_feasible(res.x, n)
        return h_res, compute_c5(h_res)
    except Exception:
        return None, None


def make_perturbed(h, n, sigma, rng):
    h_p = h + rng.normal(0.0, sigma, n)
    return project_feasible(h_p, n)


def make_antisymmetric_perturb(h, n, sigma, rng):
    noise = rng.normal(0.0, sigma, n)
    if n % 2 == 0:
        half = n // 2
        noise[half:] = -noise[:half]
    h_p = h + noise
    return project_feasible(h_p, n)


def make_structured_h(n, period, amplitude, phase, rng, base):
    i = np.arange(n, dtype=float)
    wave = amplitude * np.sin(2.0 * np.pi * i / period + phase)
    h = base + wave + rng.normal(0.0, 0.02, n)
    return project_feasible(h, n)


def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def pair_scan_refine(h, n, rng, max_pairs=100):
    best_c5 = compute_c5(h)
    h = h.copy()
    for _ in range(max_pairs):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        orig_i = h[i]
        orig_j = h[j]
        best_d = 0.0
        best_val = best_c5
        d_vals = np.linspace(-min(orig_i, 1.0 - orig_j), min(1.0 - orig_i, orig_j), 11)
        for d in d_vals:
            h[i] = orig_i + d
            h[j] = orig_j - d
            c5_trial = compute_c5(h)
            if c5_trial < best_val:
                best_val = c5_trial
                best_d = d
        h[i] = orig_i + best_d
        h[j] = orig_j - best_d
        if best_d != 0.0:
            best_c5 = best_val
    return h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    best_h = None
    best_c5 = 1e9
    best_n = 190

    init_hv = globals().get('initial_h_values', None)
    if init_hv is not None:
        h_init = np.array(init_hv, dtype=float).copy()

        h_init = project_feasible(h_init, n_init)
        best_h = h_init
        best_c5 = compute_c5(h_init)
        best_n = n_init

    if best_h is None:
        n_init = 190
        best_h = np.ones(n_init) * 0.5

        best_n = n_init

        return time.time() - t0

    # Phase 3: Pair scan refinement

    # Phase 4: Multiple perturbation restarts with SLSQP + polish
    no_improve = 0
    max_restarts = 80
    for restart in range(max_restarts):
        if elapsed() > budget_s - 120:
            break
        strategy = rng.random()
        if strategy < 0.35:
            sigma = 0.03 + 0.08 * (0.92 ** min(no_improve, 30))

            h_start = make_perturbed(best_h, best_n, sigma, rng)

            h_start = np.clip(best_h + rng.normal(0, 0.5, best_n), 0, 1)
            h_start = project_feasible(h_start, best_n)

        beta = 100.0 + 100.0 * rng.random()
        h_opt, c5_opt = optimize_slsqp(h_start, best_n, 400, beta)

            no_improve += 1

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, len(best_h))
