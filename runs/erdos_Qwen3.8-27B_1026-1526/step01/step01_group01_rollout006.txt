    shifted = np.clip(beta * (x - m), -500.0, 500.0)
    w = np.exp(shifted)
    return w / w.sum()

    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode="full")
    w = smooth_max_weights(corr, beta)
    
    lag = np.arange(len(corr)) - (n - 1)  # lag for each k

def project_feasible(h, n):

def exact_project(h, n):
    h = np.array(h, dtype=float).copy()
    lo = 0.0
    hi = 1.0
    target = n / 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)

def slsqp_optimize(h0, n, max_iter, beta):
    n = int(n)

    def obj(x):
        hx = np.clip(x, 0.0, 1.0)
        corr = np.correlate(hx, 1.0 - hx, mode="full")
        m = np.max(corr)
        shifted = np.clip(beta * (corr - m), -500.0, 500.0)
        return (m + np.log(np.sum(np.exp(shifted))) / beta) * (2.0 / n)

    def cons_fun(x):
        return np.sum(x) - n / 2.0

    hx = project_feasible(h0, n)
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, hx, method='SLSQP', bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': max_iter, 'ftol': 1e-12, 'iprint': 0})
        h_res = project_feasible(res.x, n)
        return h_res, compute_c5(h_res)
    except Exception:
        return None, None

    return np.interp(x_new, x_old, h_old)


def make_sinusoidal(n, freq, phase, amp, rng):
    i = np.arange(n, dtype=float)
    h = 0.5 + amp * np.sin(2.0 * np.pi * freq * i / n + phase)
    h = h + rng.normal(0.0, 0.02, n)
    return project_feasible(h, n)


def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    best_h = None
    best_c5 = 1e9
    best_n = 190

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        best_n = len(h_init)
        h_init = project_feasible(h_init, best_n)
        best_h = h_init
        best_c5 = compute_c5(h_init)

    if best_h is None:
        best_n = 190
        best_h = np.ones(best_n) * 0.5
        best_c5 = compute_c5(best_h)

    n = best_n

    if time.time() - t0 < budget_s - 100:
        h_opt, c5_opt = slsqp_optimize(best_h, n, 300, 80.0)
        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    if time.time() - t0 < budget_s - 300:
        h_opt, c5_opt = fista_optimize(best_h, n, 5000, 150.0)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    restarts = 0
    while time.time() - t0 < budget_s - 400:
        restarts += 1
        sigma = 0.15 * (0.92 ** min(restarts, 30))
        if rng.random() < 0.1:
            sigma = 0.5
        h_start = make_perturbation(best_h, n, sigma, rng)
        h_opt, c5_opt = fista_optimize(h_start, n, 3000, 150.0)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
        if rng.random() < 0.3:
            h_bin = make_binary_start(n, rng)
            h_bl, c5_bl = binary_local_search(h_bin, n, 500, rng)
            if c5_bl < best_c5:
                best_h, best_c5 = h_bl, c5_bl

    if time.time() - t0 < budget_s - 300:
        for freq in [3, 5, 7, 11, 13, 17]:
            if time.time() - t0 > budget_s - 280:
                break
            for amp in [0.2, 0.35, 0.5]:
                if time.time() - t0 > budget_s - 270:
                    break
                h_s = make_sinusoidal(n, freq, rng.uniform(0, 2 * np.pi), amp, rng)
                h_opt, c5_opt = fista_optimize(h_s, n, 2000, 120.0)
                if c5_opt < best_c5:
                    best_h, best_c5 = h_opt, c5_opt

    if time.time() - t0 < budget_s - 200:
        h_opt, c5_opt = fista_optimize(best_h, n, 5000, 300.0)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    if time.time() - t0 < budget_s - 150:
        h_opt, c5_opt = fista_optimize(best_h, n, 5000, 500.0)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    if time.time() - t0 < budget_s - 120:
        h_opt, c5_opt = slsqp_optimize(best_h, n, 400, 200.0)
        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    if time.time() - t0 < budget_s - 100:
        h_opt, c5_opt = fista_optimize(best_h, n, 3000, 800.0)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    for n_try in [285, 380]:
        if time.time() - t0 > budget_s - 80:
            break
        h_interp = interpolate_h(best_h, n, n_try)
        h_interp = project_feasible(h_interp, n_try)
        h_opt, c5_opt = fista_optimize(h_interp, n_try, 4000, 150.0)
        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, n_try
        if time.time() - t0 < budget_s - 60:
            h_opt2, c5_opt2 = slsqp_optimize(best_h, best_n, 200, 200.0)
            if h_opt2 is not None and c5_opt2 < best_c5:
                best_h, best_c5 = h_opt2, c5_opt2

    if time.time() - t0 < budget_s - 20:
        h_opt, c5_opt = fista_optimize(best_h, best_n, 2000, 1000.0)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    if time.time() - t0 < budget_s - 5:
        h_opt, c5_opt = slsqp_optimize(best_h, best_n, 300, 500.0)
        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
