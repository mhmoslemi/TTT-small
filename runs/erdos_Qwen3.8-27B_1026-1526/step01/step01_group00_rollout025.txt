import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.asarray(h, dtype=float)
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    return np.clip(h - lam, 0.0, 1.0)

def smooth_max_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx
    m = vals.max()
    exp_vals = np.exp(alpha * (vals - m))
    s = exp_vals.sum()
    w = exp_vals / s
    smooth_val = m + np.log(s) / alpha
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        L = k - (n - 1)
        wk = w[k]
        if wk < 1e-12:
            continue
        if L >= 0:
            if L < n:
                grad[L:n] += wk * (1.0 - h[0:n - L])
            if L > 0:
                grad[0:n - L] -= wk * h[L:n]
        else:
            M = -L
            grad[0:n - M] += wk * (1.0 - h[M:n])
            grad[M:n] -= wk * h[0:n - M]
    grad *= dx
    return smooth_val, grad

def exact_grad_at(h, k_star):
    n = len(h)
    dx = 2.0 / n
    L = k_star - (n - 1)
    grad = np.zeros(n)
    if L >= 0:
        if L < n:
            grad[L:n] += (1.0 - h[0:n - L])
        if L > 0:
            grad[0:n - L] -= h[L:n]
    else:
        M = -L
        grad[0:n - M] += (1.0 - h[M:n])
        grad[M:n] -= h[0:n - M]
    grad *= dx
    return grad

def gd_smooth(h, alpha, target_sum, max_iters, t_start, time_limit):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_val, _ = smooth_max_and_grad(h, alpha)
    step = 1.0 / (alpha * 0.02)
    stall = 0
    for it in range(max_iters):
        if time.time() - t_start > time_limit - 25:
            break
        val, grad = smooth_max_and_grad(h, alpha)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        step_size = step
        accepted = False
        for _ls in range(12):
            h_try = h - step_size * grad
            h_try = project_h(h_try, target_sum)
            new_val, _ = smooth_max_and_grad(h_try, alpha)
            if new_val < val - 1e-14:
                h = h_try
                accepted = True
                break
            step_size *= 0.5
        if not accepted:
            stall += 1
            if stall > 30:
                break
            step = step_size
        else:
            if stall > 0:
                stall = 0
            step = min(step * 1.02, 50.0 / (alpha + 1.0))
        c5_now = compute_c5(h)
        if c5_now < best_c5:
            best_c5 = c5_now
            best_h = h.copy()
    return best_h, best_c5

def gd_exact(h, target_sum, max_iters, t_start, time_limit):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if time.time() - t_start > time_limit - 15:
            break

        k_star = int(corr.argmax())
        grad = exact_grad_at(h, k_star)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break

                best_c5 = c5_try

                accepted = True

def interpolate_h(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_old = np.linspace(0, 2, n_src, endpoint=False)
    x_new = np.linspace(0, 2, n_dst, endpoint=False)
    h_new = np.interp(x_new, x_old, h_src, period=2.0)
    return h_new
