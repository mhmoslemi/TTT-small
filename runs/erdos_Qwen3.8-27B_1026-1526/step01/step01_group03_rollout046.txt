import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-14:
        idx = int(np.argmin(np.abs(result - 0.5)))
        result[idx] = np.clip(result[idx] + diff, 0.0, 1.0)
    return result

def smooth_max_fn(values, tau):
    v = values / tau
    vmax = np.max(v)
    shifted = v - vmax
    return tau * (vmax + np.log(np.sum(np.exp(shifted))))

def slsqp_obj(vec, n, tau):
    proj = np.clip(vec, 0.0, 1.0)
    g = 1.0 - proj
    corr = np.correlate(proj, g, mode="full")
    dx = 2.0 / n
    return float(smooth_max_fn(corr, tau) * dx)

def slsqp_con(vec, n):
    return np.sum(vec) - n / 2.0

def slsqp_con_jac(vec, n):
    return np.ones(n)

def de_objective(vec, n, tau):

    s = np.sum(proj)
    target = n / 2.0
    penalty = 5.0 * abs(s - target)
    g = 1.0 - proj
    corr = np.correlate(proj, g, mode="full")

    if len(parent_h) == n:
        return parent_h.copy()
    x_p = np.linspace(0, 2, len(parent_h), endpoint=False)
    x_new = np.linspace(0, 2, n, endpoint=False)
    f = interp1d(x_p, parent_h, kind='cubic', fill_value='extrapolate')
    return np.clip(f(x_new), 0.0, 1.0)
