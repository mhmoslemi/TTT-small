def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_box_sum(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        projected = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(projected)
        if s > target_sum:

        else:
            hi = mid
    projected = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target_sum - np.sum(projected)
    if abs(diff) > 1e-13:

def smooth_max(x, beta):
    m = np.max(x)
    if beta > 1e8:
        return m
    shifted = np.clip(beta * (x - m), -400.0, 400.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def obj_penalty(x, n, beta, penalty):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
    sm = smooth_max(corr, beta)
    sp = penalty * (np.sum(h) - 0.5 * n) ** 2
    return sm + sp


def obj_slsqp(x, n, beta):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
    return smooth_max(corr, beta)

def subgradient_refine(h, n, steps, step_init, tol_frac=0.02):
    h = project_box_sum(h.astype(float).copy(), 0.5 * n)
    step = step_init
    best_h = h.copy()

    for _ in range(steps):
        corr = np.correlate(h, 1.0 - h, mode="full")

        active_idx = np.where(active)[0]

            active_idx = np.array([np.argmax(corr)])
        grad = np.zeros(n)

            valid_plus = (m_plus >= 0) & (m_plus < n)

        n_init = len(h_init)
        h_init = project_box_sum(h_init, 0.5 * n_init)
        best_h = h_init
        best_c5 = compute_c5(h_init)
        best_n = n_init

    if best_h is None:
        best_h = np.ones(94) * 0.5
        best_c5 = compute_c5(best_h)
        best_n = 94

    if time.time() - t0 > budget_s - 50:
        return (best_h, best_c5, best_n)

    h_opt, c5_opt = optimize_lbfgs(best_h, best_n, max_time=min(150, budget_s * 0.12), beta=100, penalty=200)
    if c5_opt < best_c5:
        best_h, best_c5 = h_opt, c5_opt

    if time.time() - t0 < budget_s - 80:
        h_opt, c5_opt = optimize_slsqp(best_h, best_n, max_iter=400, beta=200)
        if c5_opt is not None and c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    if time.time() - t0 < budget_s - 60:
        h_opt, c5_opt = subgradient_refine(best_h, best_n, steps=400, step_init=0.08, tol_frac=0.03)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    if time.time() - t0 < budget_s - 50:
        h_opt, c5_opt = optimize_lbfgs(best_h, best_n, max_time=min(100, budget_s * 0.08), beta=400, penalty=500)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    no_improve = 0
    max_trials = 50
    trial = 0
    while time.time() - t0 < budget_s - 250 and no_improve < max_trials:
        trial += 1
        r = rng.random()
        if r < 0.25:
            sigma = 0.08 + 0.12 * (0.92 ** min(no_improve, 20))
            h_start = best_h + rng.normal(0, sigma, best_n)
            h_start = project_box_sum(h_start, 0.5 * best_n)
        elif r < 0.45:
            sigma = 0.35 + 0.1 * rng.random()
            h_start = best_h + rng.normal(0, sigma, best_n)
            h_start = project_box_sum(h_start, 0.5 * best_n)
        elif r < 0.60:
            n_peaks = int(rng.choice([2, 3, 4, 5, 6]))
            h_start = make_gaussian_structured(best_n, rng, n_peaks)
        elif r < 0.75:
            h_struct = make_gaussian_structured(best_n, rng, 3)
            alpha = rng.uniform(0.2, 0.8)
            h_start = alpha * best_h + (1.0 - alpha) * h_struct
            h_start = project_box_sum(h_start, 0.5 * best_n)
        elif r < 0.85:
            period = int(rng.choice([4, 5, 6, 7, 8, 10, 12]))
            amp = rng.uniform(0.15, 0.4)
            h_start = make_periodic(best_n, rng, period, amp)
        else:
            h_start = rng.uniform(0, 1, best_n)
            h_start = project_box_sum(h_start, 0.5 * best_n)

        remaining = budget_s - 250 - (time.time() - t0)
        if remaining < 25:
            break
        tper = min(50, remaining / 4.0)

        h_opt, c5_opt = optimize_lbfgs(h_start, best_n, max_time=tper, beta=150, penalty=300)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
            no_improve = 0
        else:
            no_improve += 1

        if no_improve % 8 == 0 and time.time() - t0 < budget_s - 100:
            h_opt, c5_opt = subgradient_refine(best_h, best_n, steps=200, step_init=0.06, tol_frac=0.02)
            if c5_opt < best_c5:
                best_h, best_c5 = h_opt, c5_opt

    for n_try in [380, 570, 760]:
        if time.time() - t0 > budget_s - 120:
            break
        h_interp = interpolate_h(best_h, best_n, n_try)
        h_interp = project_box_sum(h_interp, 0.5 * n_try)
        remaining = budget_s - 120 - (time.time() - t0)
        if remaining < 50:
            break
        h_opt, c5_opt = optimize_lbfgs(h_interp, n_try, max_time=int(remaining * 0.6), beta=120, penalty=300)
        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, n_try
        h_opt2, c5_opt2 = subgradient_refine(best_h, best_n, steps=150, step_init=0.05, tol_frac=0.02)
        if c5_opt2 < best_c5:
            best_h, best_c5 = h_opt2, c5_opt2

    remaining = budget_s - (time.time() - t0)
    if remaining > 50:
        h_opt, c5_opt = optimize_lbfgs(best_h, best_n, max_time=int(remaining * 0.45), beta=600, penalty=1000)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    remaining = budget_s - (time.time() - t0)
    if remaining > 30:
        h_opt, c5_opt = subgradient_refine(best_h, best_n, steps=300, step_init=0.04, tol_frac=0.01)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    remaining = budget_s - (time.time() - t0)
    if remaining > 20:
        h_opt, c5_opt = optimize_slsqp(best_h, best_n, max_iter=200, beta=800)
        if c5_opt is not None and c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    best_h = project_box_sum(best_h, 0.5 * len(best_h))
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, len(best_h))
