The gradient of (1/m) * ∑_i corr[k_i]^2 is (2/m) * ∑_i corr[k_i] * ∇corr[k_i].

This is a smooth objective (quadratic in the active correlations) and should converge faster than the max.

    best_c5 = compute_c5(h)
    dx = 2.0 / n

            best_c5 = c5
            best_h = h.copy()

        top_idx = np.argsort(corr)[-n_active:][::-1]
        
        # Compute gradient of sum of squared active correlations
        g = np.zeros(n)
        for k in top_idx:
            lag = k - (n - 1)
            # subgradient of corr[k]
            sg = np.zeros(n)
            if lag == 0:
                sg = 1.0 - 2.0 * h
            elif lag > 0:

                sg[:n-lag] -= h[lag:]
            else:
                m = -lag
                sg[:n-m] = (1.0 - h[m:])
                sg[m:] -= h[:n-m]
            g += corr[k] * sg
        
        # Normalize gradient
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        
        g /= gn
        
        # Line search
        step = 0.01
        improved = False
        for _ in range(20):

            h_new = project_onto_feasible(h_new, target_sum)
            c5_new = compute_c5(h_new)

                c5 = c5_new
                improved = True
                break

For the coordinate descent, the key insight is:
- When I change h[i] by δ, the change in corr[k] is δ * s_i[k] where s_i is the subgradient vector for coordinate i.
- I want to find the value of h[i] that minimizes max_k corr[k].
- This is a 1D minimization problem!

For a 1D minimization of max_k (corr[k] + δ * s_i[k]), I can use:
- Binary search on δ (but the objective is not monotone in δ)
- Golden section search
- Or just try a grid of values

Since corr[k] + δ * s_i[k] is linear in δ for each k, the max is a convex piecewise-linear function of δ. I can find its minimum by:
1. Finding all breakpoints (where two pieces cross): O((2n)^2) in the worst case, but usually much less
2. Evaluating at each breakpoint and at the boundaries

- Try values of h[i] in {0, 0.05, 0.1, ..., 0.95, 1.0} (21 values)
- For each, compute the change in corr (O(n)) and the new max (O(n))
- Total per coordinate: O(21 * n) = O(n)
- Total per round: O(n^2)

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_onto_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        nf = int(np.sum(free))
        if nf > 0:
            h[free] += diff / nf
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_subgrad_vec(h, i, n):
    ncorr = 2 * n - 1
    sg = np.zeros(ncorr)
    for k in range(ncorr):
        lag = k - (n - 1)
        val = 0.0
        j1 = i - lag
        if 0 <= j1 < n:
            val += (1.0 - h[j1])
        j2 = i + lag
        if 0 <= j2 < n:
            val -= h[j2]
        sg[k] = val
    return sg

def smooth_grad_descent(h0, n, target_sum, alpha, lr, t_start, time_limit):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    ncorr = 2 * n - 1
    momentum = 0.0

    while time.time() - t_start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = float(np.max(corr) * dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        m = float(np.max(corr))
        exp_s = np.exp(alpha * (corr - m))
        Z = float(np.sum(exp_s))
        w = exp_s / Z

        g = np.zeros(n)
        active_k = np.where(w > 1e-8)[0]
        for k in active_k:
            wk = w[k]
            lag = k - (n - 1)
            if lag == 0:
                g += wk * (1.0 - 2.0 * h)
            elif lag > 0:
                g[lag:] += wk * (1.0 - h[:n - lag])
                g[:n - lag] -= wk * h[lag:]
            else:
                mm = -lag
                g[:n - mm] += wk * (1.0 - h[mm:])
                g[mm:] -= wk * h[:n - mm]

        g *= dx
        g_new = g + momentum * g_prev if momentum > 0 else g
        g_prev = g.copy()
        momentum = 0.9

        step = lr
        h_new = h - step * g_new
        h_new = project_onto_feasible(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5:
            h = h_new
        else:
            lr *= 0.5
            if lr < 1e-10:
                lr = 1e-4
                momentum = 0.0
        lr = min(lr * 1.002, 0.1)

    return best_h, best_c5

def active_set_opt(h0, n, target_sum, top_k, t_start, time_limit):

def fast_coord_descent(h0, n, target_sum, t_start, time_limit, max_rounds=50000):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    ncorr = 2 * n - 1

    candidates = np.array([0.0, 0.05, 0.1, 0.15, 0.2, 0.25, 0.3, 0.35, 0.4,
                           0.45, 0.5, 0.55, 0.6, 0.65, 0.7, 0.75, 0.8, 0.85,
                           0.9, 0.95, 1.0])

    for rnd in range(max_rounds):
        if time.time() - t_start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = float(np.max(corr) * dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        improved = False
        start_i = (rnd * 7) % n
        for ii in range(n):
            i = (start_i + ii) % n
            subgrad = corr_subgrad_vec(h, i, n)
            best_val = h[i]
            best_c5_i = c5

            for cv in candidates:
                delta = cv - h[i]
                if abs(delta) < 1e-12:
                    continue
                new_max = np.max(corr + delta * subgrad)
                c5_candidate = new_max * dx
                if c5_candidate < best_c5_i - 1e-13:
                    best_c5_i = c5_candidate
                    best_val = cv

            if best_val != h[i]:
                h[i] = best_val
                h = project_onto_feasible(h, target_sum)
                corr = np.correlate(h, 1.0 - h, mode="full")
                c5 = float(np.max(corr) * dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                improved = True

        if not improved:
            break

    return best_h, best_c5
