    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m))
    sw = np.sum(exp_vals)
    w = exp_vals / sw
    grad_h = np.zeros(n)
    grad_h += w[n - 1] * (1.0 - 2.0 * h)
    for k in range(1, n):
        pw = w[n - 1 + k]
        nw = w[n - 1 - k]
        if pw > 1e-15 or nw > 1e-15:
            if pw > 1e-15:
                grad_h[k:] += pw * (1.0 - h[:n - k])
                grad_h[:n - k] -= pw * h[k:]
            if nw > 1e-15:
                grad_h[:n - k] += nw * (1.0 - h[k:])
                grad_h[k:] -= nw * h[:n - k]
    return grad_h

def projected_gradient_optimize(h0, n, target_sum, alpha_schedule, total_iters, rng, time_limit_fn):
    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_obj = compute_c5(h)
    momentum = 0.9
    lr = 0.01
    for it in range(total_iters):
        if time_limit_fn() < 5:
            break
        g = grad_smooth_max(h, alpha_schedule[it % len(alpha_schedule)])
        g = g - np.mean(g)
        v = momentum * v - lr * g
        h_new = h + v
        h_new = project_to_sum(h_new, target_sum)
        obj_new = compute_c5(h_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_h = h_new.copy()
        if obj_new < compute_c5(h) + 1e-10:
            h = h_new
            if it % 500 == 0:
                lr = max(lr * 0.95, 0.001)
        else:
            lr *= 0.7
            v *= 0.5
        if it % 200 == 199:
            mor = low()
            momentum = 0.9 if momentum < 0.95 else 0.8
        if it % 500 == 499:
            pert = rng.normal(0, 0.02, n)
            h_pert = project_to_sum(h + pert, target_sum)
            if compute_c5(h_pert) < compute_c5(h):
                h = h_pert
                v = np.zeros(n)
    return h, best_h, best_obj

def fast_gradient_descent(h0, n, target_sum, alpha, n_iters, lr_init, momentum, time_limit_fn, rng):
    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_val = compute_c5(h)
    lr = lr_init
    stalled = 0
    for it in range(n_iters):
        if time_limit_fn() < 3:
            break
        g = grad_smooth_max(h, alpha)
        g = g - np.mean(g)
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-12:
            g = g / g_norm
        v = momentum * v - lr * g
        h_new = project_to_sum(h + v, target_sum)
        val_new = compute_c5(h_new)
        if val_new < best_val - 1e-10:
            best_val = val_new
            best_h = h_new.copy()
            h = h_new
            stalled = 0
            lr = min(lr * 1.01, 0.05)
        else:
            h = h_new if val_new < compute_c5(h) else h
            stalled += 1
            if stalled % 10 == 0:
                lr *= 0.85
            if stalled > 50:
                pert = rng.normal(0, 0.03, n)
                h_tr = project_to_sum(h + pert, target_sum)
                if compute_c5(h_tr) < compute_c5(h):
                    h = h_tr
                    v = np.zeros(n)
                stalled = 0
                lr = lr_init * 0.5
    return h, best_h, best_val

def ga_optimize(h_init, n, target_sum, pop_size, n_gens, rng, time_limit_fn):
    pop = np.zeros((pop_size, n))
    pop[0] = h_init
    for i in range(1, pop_size):
        pop[i] = project_to_sum(rng.uniform(0, 1, n), target_sum)
    fitness = np.array([compute_c5(pop[i]) for i in range(pop_size)])
    best_idx = np.argmin(fitness)
    best_h = pop[best_idx].copy()
    best_val = fitness[best_idx]
    for gen in range(n_gens):
        if time_limit_fn() < 5:
            break
        new_pop = np.zeros((pop_size, n))
        new_pop[0] = pop[best_idx]
        for i in range(1, pop_size):
            if rng.random() < 0.9:
                t1 = rng.choice(pop_size, 3, replace=False)
                t2 = rng.choice(pop_size, 3, replace=False)
                p1 = pop[t1[fitness[t1].argmin()]]
                p2 = pop[t2[fitness[t2].argmin()]]
                mask = rng.random(n) < 0.5
                child = np.where(mask, p1, p2)
            else:
                child = pop[rng.randint(pop_size)].copy()
            child = child + rng.normal(0, 0.03, n)
            child = project_to_sum(child, target_sum)
            new_pop[i] = child
        pop = new_pop
        fitness = np.array([compute_c5(pop[i]) for i in range(pop_size)])
        bi = np.argmin(fitness)
        if fitness[bi] < best_val:
            best_val = fitness[bi]
            best_h = pop[bi].copy()
    return best_h, best_val

def coordinate_descent(h0, n, target_sum, n_rounds, rng, time_limit_fn):
    h = h0.copy()
    c5_cur = compute_c5(h)
    deltas = [0.001, 0.002, 0.005, 0.01, 0.02, 0.03, -0.001, -0.002, -0.005, -0.01, -0.02, -0.03]
    for rnd in range(n_rounds):
        if time_limit_fn() < 3:
            break
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            for d in deltas:
                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] + d, 0.0, 1.0)
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < c5_cur - 1e-12:
                    h = h_new
                    c5_cur = c5_new
                    improved = True
                    break
        if not improved:
            break
    return h, c5_cur

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    def time_left():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 180

    initial_h = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            initial_h = np.array(initial_h_values, dtype=float)
            n_init = len(initial_h)
            best_n = n_init
            target_sum = n_init / 2.0
            initial_h = project_to_sum(initial_h, target_sum)
            best_h = initial_h.copy()
            best_c5 = compute_c5(best_h)
        except:
            pass

    if best_h is None:
        best_n = 180
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n = best_n
    target_sum = n / 2.0

    if time_left() > 120:
        ga_h, ga_c5 = ga_optimize(best_h, n, target_sum, 40, 80, rng, time_left)
        if ga_c5 < best_c5:
            best_c5 = ga_c5
            best_h = ga_h.copy()

    alpha_schedules = [
        [5, 10, 20],
        [10, 20, 50],
        [20, 50, 100],
        [50, 100, 200],
        [100, 200, 500],
    ]

    starts = [best_h.copy()]
    for _ in range(4):
        h_r = best_h + rng.normal(0, 0.05, n)
        h_r = project_to_sum(h_r, target_sum)
        if compute_c5(h_r) < best_c5 + 0.02:
            starts.append(h_r)
    h_bin = (rng.rand(n) > 0.5).astype(float)
    h_bin = project_to_sum(h_bin, target_sum)
    starts.append(h_bin)
    x = np.linspace(0, 2, n, endpoint=False)
    h_sine = project_to_sum(0.5 + 0.3 * np.sin(4 * np.pi * x), target_sum)
    starts.append(h_sine)

    for si, h_start in enumerate(starts):
        if time_left() < 30:
            break
        h_start = project_to_sum(h_start, target_sum)
        c5_s = compute_c5(h_start)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_start.copy()

        for sched in alpha_schedules:
            if time_left() < 20:
                break
            n_iters = min(5000, max(500, int(time_left() * 2)))
            alpha = sched[-1]
            h_gd, h_best_gd, c5_gd = fast_gradient_descent(
                h_start, n, target_sum, alpha, n_iters, 0.02,
                0.9 if si == 0 else 0.8, time_left, rng
            )
            if c5_gd < best_c5:
                best_c5 = c5_gd
                best_h = h_best_gd.copy()
            h_start = h_gd

    if best_h is not None and time_left() > 20:
        for alpha in [200, 500, 1000, 2000]:
            if time_left() < 10:
                break
            h_gd, h_bgd, c5_gd = fast_gradient_descent(
                best_h, n, target_sum, alpha,
                min(3000, int(time_left() * 3)), 0.005, 0.9, time_left, rng
            )
            if c5_gd < best_c5:
                best_c5 = c5_gd
                best_h = h_bgd.copy()

    if best_h is not None and time_left() > 15:
        h_cd, c5_cd = coordinate_descent(best_h, n, target_sum, 100, rng, time_left)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    if best_h is not None and time_left() > 10:
        for alpha in [500, 1000, 2000]:
            if time_left() < 5:
                break
            h_gd, h_bgd, c5_gd = fast_gradient_descent(
                best_h, n, target_sum, alpha,
                min(2000, int(time_left() * 4)), 0.003, 0.9, time_left, rng
            )
            if c5_gd < best_c5:
                best_c5 = c5_gd
                best_h = h_bgd.copy()

    if best_h is not None and time_left() > 5:
        h_cd, c5_cd = coordinate_descent(best_h, n, target_sum, 50, rng, time_left)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    best_h = project_to_sum(best_h, target_sum)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n)
