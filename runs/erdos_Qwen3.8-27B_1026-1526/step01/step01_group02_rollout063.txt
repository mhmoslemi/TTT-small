    c = fftconvolve(h, 1.0 - h, mode='full')
    return float(np.max(c) * (2.0 / n))

    n = len(h)
    c = fftconvolve(h, 1.0 - h, mode='full')
    # Symmetric: c[k] = c[2n-2-k]
    # We only need k = 0, 1, ..., n-1 (lags 0 to n-1)
    # In the full output, lag 0 is at index n-1
    # Lag k is at index n-1+k

    return c[n-1:2*n-1] * (2.0 / n)

            break
        if diff > 0:
            mask = h < 1.0 - 1e-15

            mask = h > 1e-15
            if not np.any(mask):
                break

            step = min(-diff / np.sum(mask), room.min())
            h[mask] -= step

    n = len(h)

    exp_f = np.exp(tau * (f - max_f))
    soft_max = (1.0 / tau) * np.log(np.sum(exp_f)) + max_f
    return soft_max

    # k >= 1
    for k in range(1, n):
        if p[k] < 1e-18:
            continue
        coeff = p[k] * (2.0 / n)
        grad[k:] += coeff * (1.0 - h[:n-k])
        grad[:n-k] += coeff * (-h[k:])
    
    return obj, grad

def subgradient_polish(h, n, max_iter=2000, time_limit=30):
    """Polish using projected subgradient on the non-smooth max."""
    t_start = time.time()
    h_curr = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5_from_h(h_curr)
    c = 0.01  # step size parameter
    
    for it in range(max_iter):
        if time.time() - t_start > time_limit:
            break
        f = compute_correlations(h_curr)
        k_star = np.argmax(f)
        c5_curr = f[k_star]
        
        if c5_curr < best_c5 - 1e-12:
            best_c5 = c5_curr
            best_h = h_curr.copy()
        
        # Subgradient of f_{k_star}
        if k_star == 0:
            grad = (2.0 / n) * (1.0 - 2.0 * h_curr)
        else:
            k = k_star
            grad = np.zeros(n)
            grad[k:] += (2.0 / n) * (1.0 - h_curr[:n-k])
            grad[:n-k] += (2.0 / n) * (-h_curr[k:])
        
        alpha = c / (it + 1)
        h_new = h_curr - alpha * grad
        h_new = project_feasible(h_new, n)
        h_curr = h_new
    
    return best_h, best_c5
