def project_onto_feasible(x, n):
    """Project x onto {h : 0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    
    def sum_h(lam):
        return np.sum(np.clip(x + lam, 0.0, 1.0))
    
    # Bisection for lambda
    lo, hi = -max(x) - 1, 1 - min(x) + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        if sum_h(mid) < target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    return np.clip(x + lam, 0.0, 1.0)
