            best_c5 = c5_ref
            n_best = n_try

    elapsed = time.time() - t_start

    # Phase 9: Fine-tuning
    if elapsed < budget_s - 30:
        for scale in [0.04, 0.02, 0.01, 0.005]:
            for trial in range(20):
                if time.time() - t_start > budget_s - 10:
                    break
                h_pert = best_h + scale * rng.standard_normal(n_best)
                h_pert = project_feasible(h_pert, n_best)
                h_ref, c5_ref = slsqp_augmented(h_pert, maxiter=2000)
                if c5_ref < best_c5 - 1e-10:
                    best_h = h_ref
                    best_c5 = c5_ref
                    n_best = len(best_h)

    # Phase 10: Final coordinate descent + SLSQP
    elapsed = time.time() - t_start
    if elapsed < budget_s - 15:
        best_h, best_c5 = coordinate_descent(best_h, n_best, iterations=3)
        n_best = len(best_h)

    if time.time() - t_start < budget_s - 5:
        h_ref, c5_ref = slsqp_augmented(best_h, maxiter=5000)
        if c5_ref < best_c5 - 1e-12:
            best_h = h_ref
            best_c5 = c5_ref

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
