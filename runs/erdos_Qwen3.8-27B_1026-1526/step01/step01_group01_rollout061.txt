def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            if not np.any(mask):
                h += diff / n
                h = np.clip(h, 0, 1)
                break
            # Distribute to those not at 1
            add = diff / np.sum(mask)
            h[mask] = np.minimum(h[mask] + add, 1.0)
        else:
            mask = h > 1e-15
            if not np.any(mask):
                break
            sub = (-diff) / np.sum(mask)
            h[mask] = np.maximum(h[mask] - sub, 0.0)
    # Final correction
    diff = n / 2.0 - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(h[idx] + diff, 1.0)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(h[idx] + diff, 0.0)
    return h

def eval_smooth_max_grad(h, n, beta):
    """Evaluate smooth max of correlations and its gradient."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    n_full = 2 * n - 1
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500.0, 500.0)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    weights = exp_vals / sum_exp
    obj = (m + np.log(sum_exp) / beta) * dx
    
    # Gradient
    lags = np.arange(-(n-1), n)
    grad = np.zeros(n)

                   - np.sum(weights[mask1] * h[j - lags[mask1]]) 
                   - np.sum(weights[mask2] * h[j + lags[mask2]]))
    return obj, grad * dx

        pen = penalty * (np.sum(h) - n/2.0)**2

        _, g = eval_smooth_max_grad(h, n, beta)

        # Where h is at boundary, set gradient to 0 (or appropriate value)
        g[h <= 0.0] = 0.0
        g[h >= 1.0] = 0.0
        return g
    
    res = minimize(obj, h0, jac=grad, method='L-BFGS-B',
                   bounds=[(0,1)]*n, options={'maxiter': max_iter, 'ftol': ftol, 'gtol': 1e-8})
    h_res = project_feasible(res.x, n)
    return h_res, compute_c5(h_res)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-15
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    diff = n / 2.0 - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(h[idx] + diff, 1.0)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(h[idx] + diff, 0.0)
    return h

def sum_constraint(x, n):
    return np.sum(x) - n / 2.0

def sum_constraint_jac(x, n):
    return np.ones(n)

def eval_obj(x, n, beta):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500.0, 500.0)
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * (2.0 / n)

def active_gradient_refine(h, n, n_steps=200, tol_lag=0.01):
    dx = 2.0 / n
    h = project_feasible(h, n).copy()
    lags = np.arange(-(n-1), n)
    c5_current = compute_c5(h)
    
    for step in range(n_steps):
        corr = np.correlate(h, 1-h, mode="full")
        max_val = np.max(corr)
        active = np.where(corr >= max_val - tol_lag)[0]
        
        grad = np.zeros(n)
        for k in active:
            L = int(lags[k])
            j_start = max(0, L)
            j_end = min(n, L + n)
            if j_end > j_start:
                grad[j_start:j_end] += (1.0 - h[j_start-L:j_end-L])
            j_start2 = max(0, -L)
            j_end2 = min(n, n - L)
            if j_end2 > j_start2:
                grad[j_start2:j_end2] -= h[j_start2+L:j_end2+L]
        
        grad /= len(active)
        
        # Try decreasing step sizes
        improved = False
        for alpha in [0.05, 0.02, 0.01, 0.005, 0.001]:
            h_trial = project_feasible(h - alpha * grad, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < c5_current - 1e-12:
                h = h_trial
                c5_current = c5_trial
                improved = True
                break
        
        if not improved:

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.minimum(h[mask] + diff / cnt, 1.0)
        else:
            mask = h > 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] = np.maximum(h[mask] + diff / cnt, 0.0)
    diff = n / 2.0 - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(h[idx] + diff, 1.0)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(h[idx] + diff, 0.0)
    return h


def sum_constraint(x, n):
    return np.sum(x) - n / 2.0


def sum_constraint_jac(x, n):
    return np.ones(n)


def _obj(x, n, beta):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500.0, 500.0)
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * (2.0 / n)


def _grad(x, n, beta):
    h = np.clip(x, 0.0, 1.0)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500.0, 500.0)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    weights = exp_vals / sum_exp
    lags = np.arange(-(n - 1), n)
    grad = np.zeros(n)
    for j in range(n):
        m1 = (lags >= j - n + 1) & (lags <= j)
        m2 = (lags >= -j) & (lags <= n - 1 - j)
        grad[j] = (np.sum(weights[m1])
                   - np.sum(weights[m1] * h[j - lags[m1]])
                   - np.sum(weights[m2] * h[j + lags[m2]]))
    return grad * dx


def slsqp_optimize(h0, n, beta, max_iter=200):
    h0 = project_feasible(h0, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac, 'args': (n,)}]
    try:
        res = minimize(_obj, h0, jac=_grad, args=(n, beta),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-14, 'iprint': 0})
        h_res = project_feasible(res.x, n)
        return h_res, compute_c5(h_res)
    except Exception:
        return None, None


def active_gradient_refine(h, n, n_steps=300, tol_lag=0.01):
    h = project_feasible(h, n).copy()
    lags = np.arange(-(n - 1), n)
    c5_current = compute_c5(h)
    for step in range(n_steps):
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_val = np.max(corr)
        active = np.where(corr >= max_val - tol_lag)[0]
        grad = np.zeros(n)
        for k in active:
            L = int(lags[k])
            js1 = max(0, L)
            je1 = min(n, L + n)
            if je1 > js1:
                grad[js1:je1] += (1.0 - h[js1 - L:je1 - L])
            js2 = max(0, -L)
            je2 = min(n, n - L)
            if je2 > js2:
                grad[js2:je2] -= h[js2 + L:je2 + L]
        grad /= len(active)
        improved = False
        for alpha in [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]:
            h_trial = project_feasible(h - alpha * grad, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < c5_current - 1e-13:
                h = h_trial
                c5_current = c5_trial
                improved = True
                break
        if not improved:
            break
    return h, c5_current

def make_binary_relax(h, n, rng):

    n_ones = n // 2
    h_bin = np.zeros(n)
    h_bin[h_sorted_idx[-n_ones:]] = 1.0
    noise = rng.normal(0, 0.1, n)
    h_mix = 0.7 * h_bin + 0.3 * h
    h_mix = project_feasible(h_mix + noise, n)
    return h_mix


def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    best_h = None
    best_c5 = 1e9
    best_n = 190

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
        h_init = project_feasible(h_init, n_init)
        best_h = h_init
        best_c5 = compute_c5(h_init)
        best_n = n_init
    else:
        best_h = np.ones(190) * 0.5
        best_c5 = compute_c5(best_h)
        best_n = 190

    elapsed = time.time() - t0
    if elapsed < budget_s - 60:
        h_opt, c5_opt = slsqp_optimize(best_h, best_n, 100, 150)
        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    if time.time() - t0 < budget_s - 30:
        h_opt, c5_opt = active_gradient_refine(best_h, best_n, 400, 0.02)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    if time.time() - t0 < budget_s - 60:
        h_opt, c5_opt = slsqp_optimize(best_h, best_n, 500, 150)
        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    if time.time() - t0 < budget_s - 30:
        h_opt, c5_opt = active_gradient_refine(best_h, best_n, 500, 0.005)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    no_improve = 0
    while time.time() - t0 < budget_s - 350:
        if no_improve < 3:
            sigma = 0.15
        elif no_improve < 10:
            sigma = 0.10
        else:
            sigma = 0.05
        r = rng.random()
        if r < 0.2:
            h_start = make_clip_perturbed(best_h, best_n, int(2 + 4 * rng.random()), rng)
        elif r < 0.3:
            h_start = make_binary_relax(best_h, best_n, rng)
        else:
            h_start = make_perturbed(best_h, best_n, sigma, rng)
        h_opt, c5_opt = slsqp_optimize(h_start, best_n, 200, 120)
        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 35:
            break

    for n_try in [380, 570]:
        if time.time() - t0 > budget_s - 250:
            break
        h_interp = interpolate_h(best_h, best_n, n_try)
        h_opt, c5_opt = slsqp_optimize(h_interp, n_try, 100, 120)
        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, n_try
        if time.time() - t0 < budget_s - 150:
            h_opt2, c5_opt2 = active_gradient_refine(best_h, best_n, 300, 0.01)
            if c5_opt2 < best_c5:
                best_h, best_c5 = h_opt2, c5_opt2
        if time.time() - t0 < budget_s - 80:
            h_opt3, c5_opt3 = slsqp_optimize(best_h, best_n, 500, 100)
            if h_opt3 is not None and c5_opt3 < best_c5:
                best_h, best_c5 = h_opt3, c5_opt2 if c5_opt2 < c5_opt3 else c5_opt3
                best_c5 = min(best_c5, c5_opt3)

    no_improve2 = 0
    while time.time() - t0 < budget_s - 120:
        sigma = 0.03 + 0.05 * (0.7 ** min(no_improve2, 10))
        h_start = make_perturbed(best_h, best_n, sigma, rng)
        h_opt, c5_opt = slsqp_optimize(h_start, best_n, 300, 100)
        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
            no_improve2 = 0
        else:
            no_improve2 += 1
        if no_improve2 > 20:
            break

    if time.time() - t0 < budget_s - 30:
        h_opt, c5_opt = active_gradient_refine(best_h, best_n, 500, 0.002)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    if time.time() - t0 < budget_s - 20:
        h_opt, c5_opt = slsqp_optimize(best_h, best_n, 2000, 80)
        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    if time.time() - t0 < budget_s - 10:
        h_opt, c5_opt = active_gradient_refine(best_h, best_n, 300, 0.001)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, len(best_h))
