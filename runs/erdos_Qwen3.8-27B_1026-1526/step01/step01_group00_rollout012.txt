    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum} using iterative water-filling."""
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(corr, alpha):
    m = np.max(corr)
    return m + np.log(np.mean(np.exp(alpha * (corr - m)))) / alpha

def smooth_max_weights(corr, alpha):
    m = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m))
    return exp_vals / np.sum(exp_vals)

def grad_smooth_max(h, alpha):
    """Gradient of smooth_max(corr(h), alpha) w.r.t. h"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    w = smooth_max_weights(corr, alpha)
    
    grad = np.zeros(n)
    for k in range(2*n - 1):

        if wk < 1e-15:

            ss = -s
            if n - ss > 0:

    h = h0.copy()
    n = len(h)
    lr = lr_init
    best_h = h.copy()
    best_val = compute_c5(h)
    
    for step in range(n_steps):

        # Handle sum constraint: add Lagrange multiplier

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_c5(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    return m / alpha * np.log(np.sum(np.exp(alpha * (corr - m)))) + m * (1 - 1/np.sum(np.exp(alpha*(corr-m))))
    # Actually simpler:
    # smooth_max = m + log(mean(exp(alpha*(corr-m))))/alpha
    # = m + log(sum(exp(alpha*(corr-m))))/alpha - log(len(corr))/alpha
    
def smooth_max_and_weights(corr, alpha):
    m = np.max(corr)
    shifted = alpha * (corr - m)
    max_shifted = np.max(shifted)
    exp_shifted = np.exp(shifted - max_shifted)
    sum_exp = np.sum(exp_shifted)
    w = exp_shifted / sum_exp
    val = m + (max_shifted + np.log(sum_exp)) / alpha
    return val, w

def grad_smooth_max(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    w = smooth_max_weights(corr, alpha)
    
    grad = np.zeros(n)
    nm1 = n - 1
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - nm1

        else:

            if n - ss > 0:
                grad[ss:n] += wk * (1.0 - h[:n-ss])
                grad[:n-ss] -= wk * h[ss:]
    return grad

def smooth_max_weights(corr, alpha):
    m = np.max(corr)
    shifted = alpha * (corr - m)
    max_s = np.max(shifted)
    exp_s = np.exp(shifted - max_s)
    return exp_s / np.sum(exp_s)

def projected_grad_descent(h0, alpha, n_steps, target_sum, rng, lr_init=0.1):
    h = h0.copy()
    n = len(h)
    lr = lr_init
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for step in range(n_steps):
        grad = grad_smooth_max(h, alpha)
        
        # Augment with constraint gradient (Lagrange multiplier for sum)
        # Estimate: at free variables, grad should be constant (=-lambda)
        free = (h > 1e-8) & (h < 1.0 - 1e-8)
        if np.any(free):
            lam = np.median(grad[free])
        else:
            lam = 0.0
        grad_eff = grad - lam  # Remove the constant part (constraint direction)
        
        h_new = h - lr * grad_eff
        h_new = project_feasible(h_new, target_sum)
        
        # Evaluate
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        # Adaptive step size
        corr_old = np.correlate(h, 1.0 - h, mode="full")
        corr_new = np.correlate(h_new, 1.0 - h_new, mode="full")
        sm_old, _ = smooth_max_and_weights(corr_old, alpha)
        sm_new, _ = smooth_max_and_weights(corr_new, alpha)
        
        if sm_new < sm_old:
            h = h_new
            lr = min(lr * 1.02, 1.0)
        else:
            lr *= 0.6
        
        # Periodic perturbation
        if step % 500 == 499:
            idx = rng.randint(n)
            h_pert = h.copy()
            h_pert[idx] += rng.uniform(-0.05, 0.05)
            h_pert = project_feasible(h_pert, target_sum)
            if compute_c5(h_pert) < compute_c5(h):
                h = h_pert
    
    return best_h, best_c5

def coordinate_descent_polish(h0, target_sum, rng, n_rounds=200):
    h = h0.copy()
    n = len(h)
    c5_cur = compute_c5(h)
    
    for rnd in range(n_rounds):
        improved = False

                if c5_new < c5_cur - 1e-15:
                    h = h_new
                    c5_cur = c5_new
                    improved = True
                    break
        if not improved:
            break
    
    return h, c5_cur

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(3000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
        else:
            mask = h > 1e-13
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_weights(corr, alpha):
    m = np.max(corr)
    shifted = alpha * (corr - m)
    max_s = np.max(shifted)
    exp_s = np.exp(shifted - max_s)
    s = np.sum(exp_s)
    if s < 1e-300:
        w = np.zeros(len(corr))
        w[np.argmax(corr)] = 1.0
        return w
    return exp_s / s

def grad_smooth_max(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    w = smooth_max_weights(corr, alpha)
    grad = np.zeros(n)
    nm1 = n - 1
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-13:
            continue
        s = k - nm1
        if s >= 0:
            length = n - s
            if length > 0:
                grad[s:n] += wk * (1.0 - h[:length])
                grad[:length] -= wk * h[s:n]
        else:
            ss = -s
            length = n - ss
            if length > 0:
                grad[:length] += wk * (1.0 - h[ss:n])
                grad[ss:n] -= wk * h[:length]
    return grad

def projected_grad_descent(h0, alpha, n_steps, target_sum, rng, lr_init=0.1, time_fn=None, min_time=10):
    h = h0.copy()
    n = len(h)
    lr = lr_init
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = 0
    while step < n_steps:
        if time_fn is not None and time_fn() < min_time:
            break
        grad = grad_smooth_max(h, alpha)
        free = (h > 1e-7) & (h < 1.0 - 1e-7)
        if np.any(free):
            lam = np.median(grad[free])
        else:
            lam = 0.0
        grad_eff = grad - lam
        h_new = h - lr * grad_eff
        h_new = project_feasible(h_new, target_sum)
        corr_new = np.correlate(h_new, 1.0 - h_new, mode="full")
        c5_new = float(np.max(corr_new))
        corr_old_arr = np.correlate(h, 1.0 - h, mode="full")
        if c5_new < float(np.max(corr_old_arr)):
            h = h_new
            lr = min(lr * 1.03, lr_init * 3.0)
        else:
            lr *= 0.65
        c5_actual = compute_c5(h)
        if c5_actual < best_c5 - 1e-12:
            best_c5 = c5_actual
            best_h = h.copy()
        if step % 300 == 299:
            n_pert = max(1, n // 20)
            idxs = rng.choice(n, size=n_pert, replace=False)
            h_pert = h.copy()
            h_pert[idxs] += rng.uniform(-0.04, 0.04, n_pert)
            h_pert = project_feasible(h_pert, target_sum)
            if compute_c5(h_pert) < compute_c5(h):
                h = h_pert
        step += 1
    return best_h, best_c5

def coordinate_descent_polish(h0, target_sum, rng, n_rounds=200, time_fn=None, min_time=10):
    h = h0.copy()
    n = len(h)
    c5_cur = compute_c5(h)
    for rnd in range(n_rounds):
        if time_fn is not None and time_fn() < min_time:
            break
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            deltas = [-0.015, -0.008, -0.003, -0.001, 0.001, 0.003, 0.008, 0.015]
            for d in deltas:
                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] + d, 0.0, 1.0)
                h_new = project_feasible(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < c5_cur - 1e-14:
                    h = h_new
                    c5_cur = c5_new
                    improved = True
                    break
        if not improved:
            break
    return h, c5_cur

def multi_move_sa(h0, target_sum, rng, n_iters, T0, Tend, time_fn=None, min_time=15):
    h = h0.copy()
    n = len(h)
    c5_cur = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_cur
    T = T0
    cool_factor = (Tend / T0) ** (1.0 / max(1, n_iters))
    for it in range(n_iters):
        if time_fn is not None and time_fn() < min_time:
            break
        move_type = rng.random()
        if move_type < 0.4:
            i = rng.randint(n)
            delta = rng.uniform(-0.06, 0.06)
            h_new = h.copy()
            h_new[i] = np.clip(h_new[i] + delta, 0.0, 1.0)
            h_new = project_feasible(h_new, target_sum)
        elif move_type < 0.7:
            i, j = rng.choice(n, size=2, replace=False)
            delta = rng.uniform(-0.04, 0.04)
            h_new = h.copy()
            h_new[i] = np.clip(h_new[i] + delta, 0.0, 1.0)
            h_new[j] = np.clip(h_new[j] - delta, 0.0, 1.0)
            h_new = project_feasible(h_new, target_sum)
        else:
            n_block = rng.randint(3, 8)
            start = rng.randint(0, n - n_block)
            h_new = h.copy()
            h_new[start:start+n_block] += rng.uniform(-0.02, 0.02, n_block)
            h_new = project_feasible(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur:
            h = h_new
            c5_cur = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        elif rng.random() < np.exp(-(c5_new - c5_cur) / max(T, 1e-12)):
            h = h_new
            c5_cur = c5_new
        T = max(T * cool_factor, Tend)
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()

    def time_left():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float).copy()
            best_n = len(best_h)
            ts = best_n / 2.0
            best_h = project_feasible(best_h, ts)
            best_c5 = compute_c5(best_h)
        except Exception:
            best_h = None

    if best_h is None:
        best_n = 180
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n = best_n
    ts = n / 2.0

    h_cur = best_h.copy()

    for alpha in [8, 15, 30, 60, 120, 250, 500, 1000]:
        if time_left() < 80:
            break
        n_steps = min(15000, max(2000, int(time_left() * 20)))
        lr_init = max(0.05, 0.4 / np.sqrt(alpha))
        h_new, c5_new = projected_grad_descent(h_cur, alpha, n_steps, ts, rng, lr_init, time_left, 25)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_n = n
        h_cur = h_new.copy()

    n_restarts = 6
    for trial in range(n_restarts):
        if time_left() < 100:
            break
        h_pert = best_h.copy()
        n_pert = rng.randint(n // 6, n // 3)
        idxs = rng.choice(n, size=n_pert, replace=False)
        h_pert[idxs] += rng.uniform(-0.1, 0.1, n_pert)
        h_pert = project_feasible(h_pert, ts)
        h_opt, c5_opt = projected_grad_descent(h_pert, 150, 8000, ts, rng, 0.04, time_left, 30)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n
        if time_left() < 80:
            break
        h_sa, c5_sa = multi_move_sa(h_opt, ts, rng, 5000, 0.015, 0.0005, time_left, 25)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
            best_n = n

    for n_try in [144, 160, 180, 200, 240]:
        if time_left() < 70:
            break
        if n_try % 2 != 0:
            continue
        ts_try = n_try / 2.0
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_try = np.interp(x_new, x_old, best_h, period=2.0)
        h_try = project_feasible(h_try, ts_try)
        for a_try in [50, 200]:
            if time_left() < 45:
                break
            h_opt, c5_opt = projected_grad_descent(h_try, a_try, 6000, ts_try, rng, 0.06, time_left, 20)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_try
            h_try = h_opt.copy()

    if best_h is not None and time_left() > 30:
        n_f = best_n
        ts_f = n_f / 2.0
        h_pol, c5_pol = projected_grad_descent(best_h, 800, 4000, ts_f, rng, 0.015, time_left, 15)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
        if time_left() > 20:
            h_cd, c5_cd = coordinate_descent_polish(best_h, ts_f, rng, 150, time_left, 10)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()

    if best_h is not None and time_left() > 15:
        n_f = best_n
        ts_f = n_f / 2.0
        h_final = best_h.copy()
        close_to_zero = h_final < 0.002
        close_to_one = h_final > 0.998
        if np.sum(close_to_zero) + np.sum(close_to_one) > 0:
            h_final[close_to_zero] = 0.0
            h_final[close_to_one] = 1.0
            h_final = project_feasible(h_final, ts_f)
            c5_check = compute_c5(h_final)
            if c5_check < best_c5:
                best_c5 = c5_check
                best_h = h_final

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_feasible(best_h, best_n / 2.0)
    assert abs(np.sum(best_h) - best_n / 2.0) < 1e-10
    assert np.all(best_h >= 0.0) and np.all(best_h <= 1.0)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
