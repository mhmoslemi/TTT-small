            if c5_new < best_c5:

    no_improve = 0

                c5_new = compute_c5(h_new)
                if c5_new < best_c5:

    best_c5 = compute_c5(best_h)

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")


def project_to_constraint(y, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_cand = np.clip(y - mid, 0.0, 1.0)
        if np.sum(h_cand) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)


def obj_grad_smooth(h, n, beta):
    corr = get_corr(h)
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 500)
    exp_s = np.exp(shifted)
    w_sum = np.sum(exp_s)
    w = exp_s / w_sum
    val = (m + np.log(w_sum) / beta) * (2.0 / n)
    grad = np.zeros(n)
    lags = np.arange(-(n - 1), n)
    active = np.where(w > 1e-12)[0]
    for idx in active:
        L = lags[idx]
        wi = w[idx]
        if L >= 0:
            seg = n - L
            grad[L:] += wi * (1.0 - h[:seg])
            grad[:seg] += wi * (-h[L:])
        else:
            mval = -L
            seg = n - mval
            grad[:seg] += wi * (1.0 - h[mval:])
            grad[mval:] += wi * (-h[:seg])
    grad *= (2.0 / n)
    return val, grad


def obj_func_sm(h, n, beta):
    v, _ = obj_grad_smooth(h, n, beta)
    return v


def grad_func_sm(h, n, beta):
    _, g = obj_grad_smooth(h, n, beta)
    return g


def sum_constraint(h, n):
    return np.sum(h) - n / 2.0


def obj_grad_pen(h, n, beta, mu):
    corr = get_corr(h)
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 500)
    exp_s = np.exp(shifted)
    w_sum = np.sum(exp_s)
    w = exp_s / w_sum
    val = (m + np.log(w_sum) / beta) * (2.0 / n)
    grad = np.zeros(n)
    lags = np.arange(-(n - 1), n)
    active = np.where(w > 1e-12)[0]
    for idx in active:
        L = lags[idx]
        wi = w[idx]
        if L >= 0:
            seg = n - L
            grad[L:] += wi * (1.0 - h[:seg])
            grad[:seg] += wi * (-h[L:])
        else:
            mval = -L
            seg = n - mval
            grad[:seg] += wi * (1.0 - h[mval:])
            grad[mval:] += wi * (-h[:seg])
    grad *= (2.0 / n)
    s = np.sum(h) - n / 2.0
    val += mu * s * s
    grad += 2.0 * mu * s
    return val, grad


def obj_func_pen(h, n, beta, mu):
    v, _ = obj_grad_pen(h, n, beta, mu)
    return v


def grad_func_pen(h, n, beta, mu):
    _, g = obj_grad_pen(h, n, beta, mu)
    return g


def coordinate_descent(h, n, max_iter, time_limit, t0):
    best_c5 = compute_c5(h)
    delta = 0.02
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        for j in range(n):
            for sign in [1, -1]:
                h_try = h.copy()
                h_try[j] += sign * delta
                h_try = project_to_constraint(h_try, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-10:
                    h = h_try
                    best_c5 = c5_try
            delta = max(0.005, delta * 0.95)
    return h, best_c5


def projected_gradient_step(h, n, beta, alpha, t0, time_limit):
    _, grad = obj_grad_smooth(h, n, beta)
    s = np.sum(h) - n / 2.0
    grad += 50.0 * s
    h_new = project_to_constraint(h - alpha * grad, n)
    c5_new = compute_c5(h_new)
    return h_new, c5_new
