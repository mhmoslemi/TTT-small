                (0.03, 1500, 300),
                (0.015, 8000, 350),
                (0.008, 30000, 400),
                (0.004, 100000, 400),
            ]

            for tau, pw, maxiter in phases:
                if time.time() > time_limit_main:
                    break
                try:
                    res = minimize(
                        smooth_max_objective, h,
                        method='L-BFGS-B',
                        args=(n, tau, pw),
                        bounds=bounds,
                        options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-11, 'iprint': 0}
                    )
                    h = project_feasible(res.x, n)
                    c5 = evaluate_c5(h)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h.copy()
                except Exception:
                    pass

            if c0 < best_c5:
                best_c5 = c0
                best_h = h0.copy()

        print("n={} best={:.6f} t={:.0f}s".format(n, best_c5, time.time() - start_time))

    if best_h is None:
        if parent_h is not None:
            best_h = project_feasible(parent_h.copy(), len(parent_h))
            best_c5 = evaluate_c5(best_h)
        else:
            best_h = np.full(59, 0.5)
            best_c5 = evaluate_c5(best_h)

    if time.time() < start_time + budget_s * 0.92:
        n = len(best_h)
        h = best_h.copy()
        bounds = [(0.0, 1.0)] * n

        fine_phases = [
            (0.003, 50000, 400),
            (0.0015, 200000, 500),
            (0.0008, 500000, 500),
            (0.0004, 1000000, 500),
        ]

        for tau, pw, maxiter in fine_phases:
            if time.time() > start_time + budget_s * 0.94:
                break
            for attempt in range(3):
                if time.time() > start_time + budget_s * 0.94:
                    break
                if attempt == 0:
                    h_try = h.copy()
                else:
                    h_try = project_feasible(h + rng.randn(n) * 0.004 * attempt, n)
                try:
                    res = minimize(
                        smooth_max_objective, h_try,
                        method='L-BFGS-B',
                        args=(n, tau, pw),
                        bounds=bounds,
                        options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12, 'iprint': 0}
                    )
                    h_pol = project_feasible(res.x, n)
                    c5_pol = evaluate_c5(h_pol)
                    if c5_pol < best_c5:
                        best_c5 = c5_pol
                        best_h = h_pol.copy()
                        h = h_pol.copy()
                except Exception:
                    pass

        if time.time() < start_time + budget_s * 0.96:
            h_cp, c5_cp = coordinate_polish(best_h, n, max_sweeps=20)
            if c5_cp < best_c5:
                best_c5 = c5_cp
                best_h = h_cp

        if time.time() < start_time + budget_s * 0.97:
            h_ls, c5_ls = local_random_search(best_h, n, n_trials=400, rng=rng)
            if c5_ls < best_c5:
                best_c5 = c5_ls
                best_h = h_ls

            h_cp2, c5_cp2 = coordinate_polish(best_h, n, max_sweeps=15)
            if c5_cp2 < best_c5:
                best_c5 = c5_cp2
                best_h = h_cp2

    return best_h, best_c5, len(best_h)
