        if 0 <= j + offset < n:
            val += (1.0 - h[j + offset])
        # Second term: -h[j - offset] if j - offset in [0, n)
        if 0 <= j - offset < n:
            val -= h[j - offset]
        grad[j] = val
    return grad

Let me also think about what the optimal h looks like. For the minimum overlap problem, the optimal h should have a structure where it's high in some regions and low in others, arranged so that no shift k causes a large overlap between h=1 regions and h=0 regions.

For a single shift k, the overlap is ∫h(x)(1-h(x+k))dx. To minimize this for a particular k, we want h(x) and h(x+k) to be similar (both high or both low). But we need this for ALL k simultaneously, which is impossible to achieve perfectly (by a pigeonhole/averaging argument).

The average overlap over all k is:
(1/(2n-1)) * Σ_k ∫h(x)(1-h(x+k))dx = (1/(2n-1)) * ∫h(x)(1-Σ_k h(x+k))dx ≈ (1/(2n-1)) * ∫h(x)(1-h_bar)dx

where h_bar ≈ 0.5 (the average). Hmm, this isn't quite right because the shifts wrap differently.

Σ_k corr[k] = Σ_k Σ_i h[i] * (1-h[i+k]) (over valid indices)

from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def evaluate_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def smooth_max_val(values, tau):
    v = values / tau
    vmax = np.max(v)
    return tau * (vmax + np.log(np.sum(np.exp(v - vmax))))

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:

        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-14:
        idx = int(np.argmin(np.abs(result - 0.5)))
        result[idx] = np.clip(result[idx] + diff, 0.0, 1.0)

def grad_smooth_max(h, n, tau):

    corr = np.correlate(h, 1.0 - h, mode="full")
    v = corr / tau
    vmax = np.max(v)
    w = np.exp(v - vmax)
    w /= np.sum(w)
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        if w[k_idx] < 1e-12:
            continue
        m = k_idx - (n - 1)

        if m == 0:

        elif m > 0:

            grad[:n - m] -= wg * h[m:]
        else:
            p = -m
            grad[:n - p] += wg * (1.0 - h[p:])
            grad[p:] -= wg * h[:n - p]

def pso_run(n, rng, initial_h, n_particles, max_gen, time_limit):
    d = n
    particles = np.zeros((n_particles, d))
    velocities = np.zeros((n_particles, d))
    pbests = np.zeros((n_particles, d))
    pbest_fits = np.full(n_particles, np.inf)
    gbest = np.full(d, 0.5)
    gbest_fit = np.inf
    
    x = np.arange(n)
    for i in range(n_particles):
        if i == 0 and initial_h is not None:
            particles[i] = initial_h.copy()
        elif i == 1:
            particles[i] = np.clip(0.5 + 0.3*np.sin(2*np.pi*1.3*x/n + 0.5), 0, 1)
        elif i == 2:
            particles[i] = np.clip(0.5 + 0.25*np.sin(2*np.pi*2.1*x/n + 1.2) + 0.15*np.cos(2*np.pi*3.7*x/n), 0, 1)
        elif i == 3:
            particles[i] = np.clip(0.5 + 0.35*np.sin(2*np.pi*0.7*x/n), 0, 1)
        elif i == 4:
            particles[i] = np.clip(0.5 + 0.2*np.sin(2*np.pi*2.7*x/n + 0.7) + 0.2*np.sin(2*np.pi*1.9*x/n + 2.1), 0, 1)
        elif i == 5:
            particles[i] = rng.rand(n)
        else:
            particles[i] = np.clip(0.5 + 0.3*np.sin(2*np.pi*rng.uniform(0.5,4.0)*x/n + rng.uniform(0,6.28)), 0, 1)
        particles[i] = project_feasible(particles[i], n)
        pbests[i] = particles[i].copy()
        pbest_fits[i] = evaluate_c5(particles[i])
        if pbest_fits[i] < gbest_fit:
            gbest_fit = pbest_fits[i]
            gbest = particles[i].copy()
    
    start_t = time.time()
    stuck_count = 0
    
    for gen in range(max_gen):
        if time.time() - start_t > time_limit:
            break
        w_inert = 0.85 - 0.45 * gen / max_gen
        c1, c2 = 1.5, 1.5
        
        gen_best = gbest_fit
        for i in range(n_particles):
            r1 = rng.rand(d)
            r2 = rng.rand(d)
            velocities[i] = w_inert * velocities[i] + c1 * r1 * (pbests[i] - particles[i]) + c2 * r2 * (gbest - particles[i])
            velocities[i] = np.clip(velocities[i], -0.12, 0.12)
            particles[i] = np.clip(particles[i] + velocities[i], 0, 1)
            particles[i] = project_feasible(particles[i], n)

        if pbest_fits[i] < gbest_fit:
            gbest_fit = pbest_fits[i]
            gbest = pbests[i].copy()

def slsqp_refine(h0, n, tau_list, maxiter_list, time_limit, start_t):
    bounds = [(0.0, 1.0)] * n
    
    def constraint_func(h):
        return np.sum(h) - n / 2.0
    
    cons = [{'type': 'eq', 'fun': constraint_func}]
    current_h = h0.copy()
    current_c5 = evaluate_c5(current_h)
    
    for tau, maxiter in zip(tau_list, maxiter_list):
        if time.time() - start_t > time_limit:
            break
        def obj(h):
            dx = 2.0 / n
            corr = np.correlate(h, 1.0 - h, mode="full")
            return smooth_max_val(corr, tau) * dx
        try:
            res = minimize(obj, current_h, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-13})
            current_h = project_feasible(res.x, n)
            c5 = evaluate_c5(current_h)
            if c5 < current_c5:
                current_c5 = c5

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def smooth_max_val(values, tau):
    v = values / tau
    vmax = np.max(v)
    return tau * (vmax + np.log(np.sum(np.exp(v - vmax))))

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-14:
        idx = int(np.argmin(np.abs(result - 0.5)))
        result[idx] = np.clip(result[idx] + diff, 0.0, 1.0)

def grad_smooth_max(h, n, tau):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    v = corr / tau
    vmax = np.max(v)
    w = np.exp(v - vmax)
    w /= np.sum(w)
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        if w[k_idx] < 1e-12:
            continue
        m = k_idx - (n - 1)
        wg = w[k_idx]
        if m == 0:
            grad += wg * (1.0 - 2.0 * h)
        elif m > 0:
            grad[m:] += wg * (1.0 - h[:n - m])
            grad[:n - m] -= wg * h[m:]
        else:
            p = -m
            grad[:n - p] += wg * (1.0 - h[p:])
            grad[p:] -= wg * h[:n - p]
    return grad * dx

def evaluate_c5(h):

def smooth_max_val(values, tau):
    v = values / tau

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0

        mid = 0.5 * (lo + hi)

        if s > target:
            hi = mid
        else:

def grad_smooth_max(h, n, tau):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    v = corr / tau
    vmax = np.max(v)
    w = np.exp(v - vmax)
    w /= np.sum(w)
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        if w[k_idx] < 1e-12:
            continue
        m = k_idx - (n - 1)
        wg = w[k_idx]
        if m == 0:
            grad += wg * (1.0 - 2.0 * h)
        elif m > 0:
            grad[m:] += wg * (1.0 - h[:n - m])
            grad[:n - m] -= wg * h[m:]
        else:
            p = -m
            grad[:n - p] += wg * (1.0 - h[p:])
            grad[p:] -= wg * h[:n - p]

def slsqp_refine(h0, n, tau_list, maxiter_list, time_limit, start_t):
    bounds = [(0.0, 1.0)] * n

    def constraint_func(h):
        return np.sum(h) - n / 2.0

    cons = [{'type': 'eq', 'fun': constraint_func}]
    current_h = h0.copy()
    current_c5 = evaluate_c5(current_h)
    for tau, maxiter in zip(tau_list, maxiter_list):
        if time.time() - start_t > time_limit:
            break

        def obj(h):
            dx = 2.0 / n
            corr = np.correlate(h, 1.0 - h, mode="full")
            return smooth_max_val(corr, tau) * dx

        try:
            res = minimize(obj, current_h, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-13})
            current_h = project_feasible(res.x, n)
            c5 = evaluate_c5(current_h)
            if c5 < current_c5:
                current_c5 = c5
        except Exception:
            pass
    return current_h, current_c5
