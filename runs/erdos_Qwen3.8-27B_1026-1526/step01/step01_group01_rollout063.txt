```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
        else:
            mask = h > 1e-13
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    # Final exact fix
    diff = n / 2.0 - np.sum(h)
    if abs(diff) > 1e-15:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(h[idx] + diff, 1.0)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(h[idx] + diff, 0.0)
    return h

def solve_1d(D, s):
    """min_{t in [0,1]} max_j (D[j] + t * s[j])"""
    m = len(D)
    
    f0 = np.max(D)
    f1 = np.max(D + s)
    
    # Check endpoints
    tol = 1e-12
    active_0 = D >= f0 - tol
    if np.max(s[active_0]) >= -tol:
        return 0.0
    
    active_1 = (D + s) >= f1 - tol
    if np.min(s[active_1]) <= tol:
        return 1.0
    
    D_diff = D[:, None] - D[None, :]
    s_diff = s[:, None] - s[None, :]
    
    with np.errstate(divide='ignore', invalid='ignore'):
        t_mat = D_diff / s_diff
    
    valid = (t_mat > 1e-10) & (t_mat < 1.0 - 1e-10) & (np.abs(s_diff) > 1e-14)
    
    if not np.any(valid):
        return 0.0 if f0 <= f1 else 1.0
    
    candidates = t_mat[valid]
    candidates = np.unique(np.round(candidates, 10))
    
    vals = np.max(D[None, :] + candidates[:, None] * s[None, :], axis=1)
    best = int(np.argmin(vals))
    return candidates[best]

def coord_descent(h, n, lam, max_sweeps=100):
    """Run coordinate descent with Lagrangian for sum constraint."""
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode="full")
    s_vals = np.arange(2*n-1) - (n-1)
    
    for sweep in range(max_sweeps):
        max_change = 0.0
        for m in range(n):
            # Compute c_j
            c = np.zeros(2*n-1)
            for j in range(2*n-1):
                s = int(s_vals[j])
                c_j = 0.0
                if 0 <= m + s < n:
                    c_j += (1.0 - h[m + s])
                if 0 <= m - s < n:
                    c_j += (-h[m - s])
                c[j] = c_j
            
            D = corr - h[m] * c
            s_eff = c + lam
            
            t_new = solve_1d(D, s_eff)
            
            if abs(t_new - h[m]) > 1e-14:
                delta = t_new - h[m]
                h[m] = t_new
                corr += delta * c
                max_change = max(max_change, abs(delta))
        
        if max_change < 1e-12:
            break
    
    return h, corr
```

```python
def solve_1d_fast(D, s):
    """min_{t in [0,1]} max_j (D[j] + t * s[j])"""
    m = len(D)
    
    # Sort by slope
    idx = np.argsort(s)
    s_sorted = s[idx]
    D_sorted = D[idx]
    
    # Compute upper envelope
    # Lines: y = s_sorted[i] * t + D_sorted[i]
    # Intersect of lines i, i+1: t = (D[i] - D[i+1]) / (s[i+1] - s[i])
    
    # Standard algorithm:
    env = []  # list of (index, start_t)
    for i in range(m):
        if not env:
            env.append((i, -np.inf))
        else:
            # Check if line i is above the last line in the envelope at all t > start
            # Intersection of line i and last line in env:
            last_i = env[-1][0]
            last_start = env[-1][1]
            if s_sorted[i] == s_sorted[last_i]:
                # Same slope, keep the one with higher intercept
                if D_sorted[i] <= D_sorted[last_i]:
                    continue
                else:
                    # Replace last
                    env[-1] = (i, last_start)
                    # Need to recheck with previous
                    while len(env) >= 2:
                        prev_i = env[-2][0]
                        if s_sorted[i] == s_sorted[prev_i]:
                            if D_sorted[i] <= D_sorted[prev_i]:
                                env.pop()
                            continue
                        t_int = (D_sorted[prev_i] - D_sorted[i]) / (s_sorted[i] - s_sorted[prev_i])
                        if t_int <= env[-2][1]:
                            env.pop()
                        else:
                            env[-1] = (i, t_int)
                            break
                    if len(env) == 1:
                        env[0] = (i, -np.inf)
            else:
                t_int = (D_sorted[last_i] - D_sorted[i]) / (s_sorted[i] - s_sorted[last_i])
                if t_int <= last_start:
                    # Line i dominates the last line before it even starts
                    env.pop()
                    while env:
                        last_i = env[-1][0]
                        last_start = env[-1][1]
                        t_int = (D_sorted[last_i] - D_sorted[i]) / (s_sorted[i] - s_sorted[last_i])
                        if t_int > last_start:
                            break
                        env.pop()
                    if not env:
                        env.append((i, -np.inf))
                    else:
                        env.append((i, t_int))
                else:
                    env.append((i, t_int))
    
    # Now find the minimum on [0,1]
    # The envelope gives a convex PL function. Find where slope = 0.
    # Walk through the envelope segments.
    
    best_t = 0.0
    best_val = np.max(D_sorted + 0.0 * s_sorted)  # f(0)
    
    for seg_idx in range(len(env)):
        i = env[seg_idx][0]
        t_start = max(env[seg_idx][1], 0.0)
        if seg_idx < len(env) - 1:
            t_end = min(env[seg_idx+1][1], 1.0)
        else:
            t_end = 1.0
        
        if t_start > t_end:
            continue
        
        # On this segment, f(t) = s_sorted[i]*t + D_sorted[i]
        # It's linear, so minimum is at one of the endpoints
        val_start = s_sorted[i] * t_start + D_sorted[i]
        val_end = s_sorted[i] * t_end + D_sorted[i]
        
        if val_start < best_val:
            best_val = val_start
            best_t = t_start
        if val_end < best_val:
            best_val = val_end
            best_t = t_end
    
    # Also check t=0 and t=1
    val_0 = np.max(D)
    val_1 = np.max(D + s)
    if val_0 < best_val:
        best_val = val_0
        best_t = 0.0
    if val_1 < best_val:
        best_val = val_1
        best_t = 1.0
    
    return best_t
```

```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
        else:
            mask = h > 1e-13
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = n / 2.0 - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(h[idx] + diff, 1.0)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(h[idx] + diff, 0.0)
    return h

def solve_1d(D, s):
    """Solve min_{t in [0,1]} max_j (D[j] + t*s[j])"""
    m = len(D)
    f0 = np.max(D)
    f1 = np.max(D + s)
    
    # Check right derivative at 0
    active_0 = D >= f0 - 1e-14
    if np.max(s[active_0]) >= -1e-14:
        return 0.0
    
    # Check left derivative at 1
    active_1 = (D + s) >= f1 - 1e-14
    if np.min(s[active_1]) <= 1e-14:
        return 1.0
    
    # Find all intersection points in (0,1)
    D_diff = D[:, None] - D[None, :]
    s_diff = s[:, None] - s[None, :]
    
    with np.errstate(divide='ignore', invalid='ignore'):
        t_mat = D_diff / s_diff
    
    valid = (t_mat > 1e-12) & (t_mat < 1.0 - 1e-12) & (np.abs(s_diff) > 1e-14)
    
    if not np.any(valid):
        return 0.0 if f0 <= f1 else 1.0
    
    candidates = t_mat[valid]
    # Round to avoid near-duplicates
    candidates = np.unique(np.round(candidates, 12))
    
    if len(candidates) == 0:
        return 0.0 if f0 <= f1 else 1.0
    
    # Evaluate f at each candidate
    vals = np.max(D[None, :] + candidates[:, None] * s[None, :], axis=1)
    best = int(np.argmin(vals))
    return float(candidates[best])

def run_coord_descent(h, n, lam, max_sweeps, order, t_limit):
    """Run coordinate descent. Returns (h, corr, converged)."""
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode="full")
    s_vals = np.arange(2 * n - 1) - (n - 1)
    
    for sweep in range(max_sweeps):
        if time.time() > t_limit:
            break
        max_change = 0.0
        for m in order:
            ms = m + s_vals
            mms = m - s_vals
            alpha = (ms >= 0) & (ms < n)
            beta = (mms >= 0) & (mms < n)
            
            h_ms = np.zeros(2 * n - 1)
            if np.any(alpha):
                h_ms[alpha] = h[ms[alpha].astype(int)]
            h_mms = np.zeros(2 * n - 1)
            if np.any(beta):
                h_mms[beta] = h[mms[beta].astype(int)]
            
            c = (alpha * (1.0 - h_ms) - beta * h_mms).astype(float)
            
            D = corr - h[m] * c
            s_eff = c + lam
            t_new = solve_1d(D, s_eff)
            
            delta = t_new - h[m]
            if abs(delta) > 1e-14:
                h[m] = t_new
                corr += delta * c
```

```python
def solve_1d_approx(D, s, n_iter=50):
    """Approximate min_{t in [0,1]} max_j (D[j] + t*s[j]) using ternary search."""
    def f(t):
        return np.max(D + t * s)
    
    # Ternary search
    a, b = 0.0, 1.0
    for _ in range(n_iter):
        m1 = a + (b - a) / 3
        m2 = b - (b - a) / 3
        if f(m1) < f(m2):
            b = m2
        else:
            a = m1
    
    t_star = (a + b) / 2
    
    # Refine: check nearby breakpoints
    # The exact minimum is at or near t_star
    # Find the active set at t_star
    f_star = f(t_star)
    active = np.abs(D + t_star * s - f_star) < 1e-10
    if np.any(active):
        # The minimum might be at the intersection of two active lines
        # Check all pairs of active lines
        active_idx = np.where(active)[0]
        for i in range(len(active_idx)):
            for j in range(i+1, len(active_idx)):
                si = s[active_idx[i]]
                sj = s[active_idx[j]]
                if abs(si - sj) > 1e-14:
                    t = (D[active_idx[j]] - D[active_idx[i]]) / (si - sj)
                    if 0 <= t <= 1:
                        f_t = f(t)
                        if f_t < f_star:
                            t_star = t
                            f_star = f_t
    
    return t_star
```

```python
    def f_and_subgrad(t):
        vals = D + t * s
        f_val = np.max(vals)
        active = vals >= f_val - 1e-14
        return f_val, s[active]
    
    # Check endpoints
    f0, sg0 = f_and_subgrad(0.0)
    if np.min(sg0) >= -1e-14:
        return 0.0
    
    f1, sg1 = f_and_subgrad(1.0)
    if np.max(sg1) <= 1e-14:
        return 1.0

    a, b = 0.0, 1.0
    for _ in range(n_iter):
        mid = (a + b) / 2
        f_mid, sg_mid = f_and_subgrad(mid)
        if np.min(sg_mid) >= 0:
            # Subgradient is non-negative, minimum is at or to the left
            b = mid

            a = mid

                                best_val = f_t
                                best_t = t
```
