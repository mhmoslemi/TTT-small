import numpy as np

from scipy.interpolate import interp1d
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(values, tau):
    v = values / tau
    vmax = np.max(v)
    return tau * (vmax + np.log(np.sum(np.exp(v - vmax))))

def project_feasible(h, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-15:
        # Distribute the difference to the entries closest to 0.5
        idxs = np.argsort(np.abs(result - 0.5))[:10]
        per = diff / len(idxs)
        for idx in idxs:
            result[idx] = np.clip(result[idx] + per, 0.0, 1.0)
        # Final fix
        diff2 = target - np.sum(result)
        if abs(diff2) > 1e-14:
            idx = int(np.argmin(np.abs(result - 0.5)))
            result[idx] = np.clip(result[idx] + diff2, 0.0, 1.0)
    return result

def compute_subgradient(h, n, active_set):
    """Compute subgradient of max_k corr[k] at h, averaged over active set."""
    subgrad = np.zeros(n)
    for k in active_set:
        m = k - (n - 1)
        for i in range(n):
            val = 0.0
            if 0 <= i + m <= n - 1:
                val += (1.0 - h[i + m])
            if 0 <= i - m <= n - 1:
                val -= h[i - m]
            subgrad[i] += val
    subgrad /= len(active_set)
    return subgrad

def subgradient_descent(h_init, n, max_iter, start_time, budget_s, best_c5, best_h):
    """Projected subgradient descent with momentum."""
    h = h_init.copy()
    h_prev = h.copy()
    
    for t in range(max_iter):
        if time.time() > start_time + budget_s * 0.75:
            break
        
        dx = 2.0 / n
        g = 1.0 - h
        corr = np.correlate(h, g, mode="full")
        current_c5 = np.max(corr) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Find active set (within tolerance of max)
        max_val = np.max(corr)
        tol = 0.01 * max_val  # Include near-max lags
        active = np.where(corr >= max_val - tol)[0]
        
        # Compute subgradient
        subgrad = compute_subgradient(h, n, active)
        
        # Adaptive step size (decreasing)
        step_size = 0.1 / (1.0 + 0.01 * t)
        
        # Momentum (heavy ball)

        h_new = h - step_size * subgrad + momentum * (h - h_prev)
        
        # Project
        h_new = project_feasible(h_new, n)
        
        h_prev = h
        h = h_new
    
    return best_c5, best_h
