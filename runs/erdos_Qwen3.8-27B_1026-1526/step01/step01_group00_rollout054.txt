```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(corr, alpha):
    """Compute smooth max and its gradient with respect to corr values."""
    m = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m))
    w = exp_vals / np.sum(exp_vals)
    s = m + np.log(np.sum(exp_vals)) / alpha  # = m + np.log(len(corr))/alpha + log(mean(exp_vals))/alpha
    # Actually: smooth_max = (1/alpha) * log(sum(exp(alpha*corr)))
    # = (1/alpha) * log(exp(alpha*m) * sum(exp(alpha*(corr-m))))
    # = m + (1/alpha) * log(sum(exp(alpha*(corr-m))))
    # = m + (1/alpha) * log(sum(exp_vals))
    s = m + np.log(np.sum(exp_vals)) / alpha
    return s, w

def correlation_gradient(h, w):
    """
    Compute gradient of smooth_max(corr, alpha) with respect to h.
    
    w: weights from soft attention over correlation lags (same length as corr)
    corr = np.correlate(h, 1-h, "full"), length 2n-1
    corr[k] (0-indexed in full, corresponding to lag k-(n-1)) = sum_i h[i] * (1-h[i+lag])
    
    Actually, let me use the np.correlate convention:
    np.correlate(a, v, "full")[k] = sum_n a[n+k-(len(v)-1)] * v[n] for valid n
    Wait no. Let me check the numpy docs.
    
    numpy.correlate(a, v): c_k = sum_n a*_n+k * v_n
    In "full" mode, k ranges from -(len(v)-1) to len(a)-1
    The output is indexed 0 to len(a)+len(v)-2, where index k corresponds to lag k-(len(v)-1)
    
    So for a=h, v=1-h, both of length n:
    output[k] for k=0,...,2n-2 corresponds to lag = k-(n-1)
    
    output[k] = sum_n h[n + k - (n-1)] * (1-h)[n] for valid n
    Let lag = k - (n-1). Then:
    output[k] = sum_n h[n + lag] * (1-h[n]) for valid n (0 ≤ n < n, 0 ≤ n+lag < n)
    
    For a given j, d(output[k])/dh[j]:
    - h[j] appears in output[k] when n+lag = j, i.e., n = j-lag, contributing (1-h[j-lag]) if 0≤j-lag<n
    - (1-h[j]) appears (with -1 factor) when n = j, contributing -h[j+lag] if 0≤j+lag<n
    
    So dS/dh[j] = sum_k w[k] * ((1-h[j-lag_k]) * 1[0≤j-lag_k<n] - h[j+lag_k] * 1[0≤j+lag_k<n])
    where lag_k = k - (n-1)
    """
    n = len(h)
    lags = np.arange(len(w)) - (n - 1)  # lag for each k
    grad = np.zeros(n)
    for j in range(n):
        # First term: (1-h[j-lag]) for valid j-lag
        valid1 = (j - lags >= 0) & (j - lags < n)
        if np.any(valid1):
            indices = j - lags[valid1]
            grad[j] += np.sum(w[valid1] * (1 - h[indices]))
        # Second term: -h[j+lag] for valid j+lag
        valid2 = (j + lags >= 0) & (j + lags < n)
        if np.any(valid2):
            indices = j + lags[valid2]
            grad[j] -= np.sum(w[valid2] * h[indices])
    return grad

def projected_gradient_descent(h0, n, target_sum, alpha, max_iters, lr_init, time_limit, t_start):
    """Custom projected gradient descent with backtracking line search."""
    h = h0.copy()
    c5_best = compute_c5(h)
    h_best = h.copy()
    n_corr = 2 * n - 1
    
    for it in range(max_iters):
        if time.time() - t_start > time_limit:
            break
        
        # Compute current smooth max and weights
        corr = np.correlate(h, 1.0 - h, mode="full")
        s, w = smooth_max_and_grad(corr, alpha)
        
        # Compute gradient
        grad = correlation_gradient(h, w)
        
        # Project gradient: ensure we're moving in feasible direction
        # The constraint is sum(h) = target_sum, 0 ≤ h ≤ 1
        # Project gradient onto the tangent space of the equality constraint
        grad = grad - np.mean(grad)  # tangent to sum constraint
        
        # Backtracking line search
        lr = lr_init
        obj_cur = s
        for bt in range(30):
            h_new = h - lr * grad
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = project_to_sum(h_new, target_sum)
            
            corr_new = np.correlate(h_new, 1.0 - h_new, mode="full")
            obj_new, _ = smooth_max_and_grad(corr_new, alpha)
            
            if obj_new < obj_cur - 1e-12:
                break
            lr *= 0.5
        else:
            # No improvement, try smaller step
            break
        
        h = h_new
        c5_new = compute_c5(h)
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h.copy()
    
    return h_best, c5_best

def block_swap_search(h, n, target_sum, rng, max_rounds, time_limit, t_start):
    """Try swapping blocks of h to find better configurations."""
    h = h.copy()
    c5_cur = compute_c5(h)
    c5_best = c5_cur
    h_best = h.copy()
    
    for rnd in range(max_rounds):
        if time.time() - t_start > time_limit:
            break
        improved = False
        # Try different block sizes
        for block_size in [2, 3, 4, 5]:
            for start in range(0, n - block_size, block_size):
                if time.time() - t_start > time_limit:
                    break
                # Try swapping two blocks
                end = start + block_size
                if end >= n:
                    continue
                # Find another block to swap with
                for start2 in range(0, n - block_size, block_size):
                    if start2 == start:
                        continue
                    end2 = start2 + block_size
                    if end2 >= n:
                        continue
                    h_new = h.copy()
                    h_new[start:start+block_size] = h[start2:start2+block_size]
                    h_new[start2:start2+block_size] = h[start:start+block_size]
                    h_new = project_to_sum(h_new, target_sum)
                    c5_new = compute_c5(h_new)
                    if c5_new < c5_cur - 1e-10:
                        h = h_new
                        c5_cur = c5_new
                        improved = True
                        if c5_new < c5_best:
                            c5_best = c5_new
                            h_best = h.copy()
                        break
                if improved:
                    break
            if not improved:
                break
        if not improved:
            break
```

```
        grad[j] -= np.sum(w[valid2] * h[indices])
```

This is O(n * 2n) = O(n²) which is fine.

But I can vectorize this! Note that:
grad[j] = sum_i (1-h[i]) * w[i-j+(n-1)] * 1[0≤i<n, 0≤i-j+(n-1)<2n-1] - sum_i h[i] * w[j-i+(n-1)] * 1[0≤i<n, 0≤j-i+(n-1)<2n-1]

The first sum is: for each i, (1-h[i]) * w[i-j+(n-1)] where the index i-j+(n-1) must be in [0, 2n-2].
This is a convolution of (1-h) with w!
Similarly the second is a convolution of h with w (reversed).

Specifically:
First sum = [np.convolve(1-h, w, mode='full')][n-1 + j - i]... 

Hmm, this is getting complicated. Let me just use the O(n²) loop since for n=180 it's 65000 iterations which is trivial. For n=360 it's 260000, still fine. For n=720 it's 1M, still OK in a tight loop.

Actually, I can vectorize it as:
```python
def correlation_gradient(h, w):
    n = len(h)
    lags = np.arange(len(w)) - (n - 1)  # shape (2n-1,)
    
    # For each j, compute:
    # Term 1: sum_{k: 0<=j+lag_k<n} w[k] * (1-h[j+lag_k])
    # Term 2: sum_{k: 0<=j-lag_k<n} w[k] * h[j-lag_k]
    
    # Vectorized: create a matrix M where M[j,k] = 1 if 0<=j+lags[k]<n
    # This is a Toeplitz-like matrix
    
    # Actually, let's use the convolution approach:
    # Term 1[j] = sum_i (1-h[i]) * w[j-i+(n-1)] for valid indices
    # This is the cross-correlation of (1-h) with w, evaluated at positions...
    
    # Let's just do it with a loop over j, vectorized over k
    grad = np.zeros(n)
    for j in range(n):
        idx1 = j + lags  # = j + arange(2n-1) - (n-1) = arange(2n-1) + j - (n-1)
        valid1 = (idx1 >= 0) & (idx1 < n)
        if np.any(valid1):
            grad[j] += np.dot(w[valid1], 1.0 - h[idx1[valid1]])
        idx2 = j - lags  # = (n-1) - arange(2n-1) + j
        valid2 = (idx2 >= 0) & (idx2 < n)
        if np.any(valid2):
            grad[j] -= np.dot(w[valid2], h[idx2[valid2]])
    return grad
```

This is O(n²) but with numpy's fast dot product, it should be quite quick.

Alternatively, I can use FFT-based convolution:
Term 1[j] = sum_i (1-h[i]) * w[j-i+(n-1)]
Let me define g[i] = 1-h[i] and we want sum_i g[i] * w[j+(n-1)-i]
This is the cross-correlation of g and w at shift j+(n-1).
= np.correlate(g, w, mode='full')[j] (since correlate gives shifts from -(len(w)-1) to len(g)-1)

Wait, np.correlate(g, w, "full")[k] = sum_n g[n] * w[n+k-(len(w)-1)]
= sum_n g[n] * w[n+k-(2n-1-1)]... this is getting confusing with the same n for both array length and index.

Let me just stick with the O(n²) loop. It's simple and correct.

```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_weights(corr, alpha):
    m = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m))
    total = np.sum(exp_vals)
    w = exp_vals / total
    s = m + np.log(total) / alpha
    return s, w

def correlation_gradient(h, w):
    n = len(h)
    lags = np.arange(len(w)) - (n - 1)
    grad = np.zeros(n)
    for j in range(n):
        idx1 = j + lags
        valid1 = (idx1 >= 0) & (idx1 < n)
        if np.any(valid1):
            grad[j] += np.dot(w[valid1], 1.0 - h[idx1[valid1]])
        idx2 = j - lags
        valid2 = (idx2 >= 0) & (idx2 < n)
        if np.any(valid2):
            grad[j] -= np.dot(w[valid2], h[idx2[valid2]])
    return grad

def projected_grad_descent(h0, alpha, target_sum, max_iters, lr_init, t_start, time_limit):
    n = len(h0)
    h = h0.copy()
    c5_best = compute_c5(h)
    h_best = h.copy()
    lr = lr_init
    for it in range(max_iters):
        if time.time() - t_start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        obj_cur, w = smooth_max_and_weights(corr, alpha)
        grad = correlation_gradient(h, w)
        grad = grad - np.mean(grad)
        obj_new = float('inf')
        for bt in range(40):
            step = lr * (0.95 ** bt)
            h_new = np.clip(h - step * grad, 0.0, 1.0)
            h_new = project_to_sum(h_new, target_sum)
            corr_new = np.correlate(h_new, 1.0 - h_new, mode="full")
            obj_new, _ = smooth_max_and_weights(corr_new, alpha)
            if obj_new < obj_cur - 1e-14 * max(1.0, abs(obj_cur)):
                break
        else:
            break
        h = h_new
        c5_new = compute_c5(h)
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h.copy()
        lr = min(lr * 1.05, 0.5)
    return h_best, c5_best

def focused_refinement(h0, target_sum, rng, t_start, time_limit):
    n = len(h0)
    h = h0.copy()
    c5_best = compute_c5(h)
    h_best = h.copy()
    stall = 0
    for it in range(200000):
        if time.time() - t_start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_star = np.argmax(corr)
        lag = k_star - (n - 1)
        valid_i = []
        for i in range(n):
            if 0 <= i + lag < n:
                valid_i.append(i)
        valid_i = np.array(valid_i)
        if len(valid_i) == 0:
            break
        conflicts = h[valid_i] * (1.0 - h[valid_i + lag])
        top_k = min(8, len(valid_i))
        top_idx = np.argsort(conflicts)[-top_k:]
        moved = False
        for idx in top_idx:
            i = valid_i[idx]
            j = valid_i[idx] + lag
            for delta in [0.005, 0.01, 0.02, 0.03, 0.05]:
                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] - delta, 0.0, 1.0)
                h_new[j] = np.clip(h_new[j] + delta, 0.0, 1.0)
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < c5_best - 1e-12:
                    h = h_new
                    c5_best = c5_new
                    h_best = h.copy()
                    moved = True
                    break
            if moved:
                break
        if not moved and rng.random() < 0.7:
            i = valid_i[rng.randint(len(valid_i))]
            j = rng.randint(n)
            if i != j:
                delta = rng.choice([-0.01, -0.005, 0.005, 0.01, 0.02])
                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] + delta, 0.0, 1.0)
                h_new[j] = np.clip(h_new[j] - delta, 0.0, 1.0)
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < c5_best - 1e-12:
                    h = h_new
                    c5_best = c5_new
                    h_best = h.copy()
                    moved = True
        if moved:
            stall = 0
        else:
            stall += 1
            if stall > 500:
                break
    return h_best, c5_best

def fine_tune(h0, target_sum, rng, t_start, time_limit):
    n = len(h0)
    h = h0.copy()
    c5_best = compute_c5(h)
    h_best = h.copy()
    for it in range(500000):
        if time.time() - t_start > time_limit:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        delta = rng.uniform(-0.003, 0.003)
        h_new = h.copy()
        h_new[i] = np.clip(h_new[i] + delta, 0.0, 1.0)
        h_new[j] = np.clip(h_new[j] - delta, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5_best - 1e-13:
            h = h_new
            c5_best = c5_new
            h_best = h.copy()
    return h_best, c5_best

def multi_resolution(h_parent, n_target, t_start, time_limit, rng):
    resolutions = []
    n_cur = max(20, n_target // 6)
    if n_cur % 2 != 0:
        n_cur += 1
    while n_cur < n_target:
        n_next = min(int(n_cur * 1.6) + 2, n_target)
        if n_next % 2 != 0:
            n_next += 1
        if n_next <= n_cur:
            n_next = n_target
        resolutions.append(n_next)
        n_cur = n_next
    if n_target not in resolutions:
        resolutions.append(n_target)

    x_parent = np.linspace(0, 2, len(h_parent), endpoint=False)
    h_cur = np.interp(np.linspace(0, 2, resolutions[0], endpoint=False), x_parent, h_parent, period=2.0)
    n_cur = resolutions[0]
    h_cur = project_to_sum(h_cur, n_cur / 2.0)

    for n_new in resolutions:
        if time.time() - t_start > time_limit - 10:
            break
        ts = n_new / 2.0
        if n_new > n_cur:
            x_old = np.linspace(0, 2, n_cur, endpoint=False)
            x_new = np.linspace(0, 2, n_new, endpoint=False)
            h_cur = np.interp(x_new, x_old, h_cur, period=2.0)
            h_cur = project_to_sum(h_cur, ts)
        n_cur = n_new
        tl_remaining = time_limit - (time.time() - t_start)
        if tl_remaining < 15:
            break
        h_cur = projected_grad_descent(h_cur, 50.0, ts, 800, 0.04, t_start, time.time() + tl_remaining * 0.35)[0]
        tl_remaining = time_limit - (time.time() - t_start)
        if tl_remaining < 10:
            break
        h_cur = focused_refinement(h_cur, ts, rng, t_start, time.time() + tl_remaining * 0.4)[0]
    return h_cur

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()

    def tl():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in dir() and initial_h_values is not None:
        try:
            init = np.array(initial_h_values, dtype=float)
            n_init = len(init)
            ts_init = n_init / 2.0
            init = project_to_sum(init, ts_init)
            best_h = init.copy()
            best_c5 = compute_c5(best_h)
            best_n = n_init
        except Exception:
            pass

    if best_h is None:
        n = 180
        best_n = n
        best_h = np.full(n, 0.5)
        best_c5 = compute_c5(best_h)

    n = best_n
    ts = n / 2.0

    if tl() > 100:
        h_gd, c5_gd = projected_grad_descent(best_h, 30.0, ts, 3000, 0.05, t_start, t_start + tl() * 0.25)
        if c5_gd < best_c5:
            best_c5 = c5_gd
            best_h = h_gd.copy()

    if tl() > 80:
        for alpha in [80.0, 200.0]:
            if tl() < 40:
                break
            h_gd, c5_gd = projected_grad_descent(best_h, alpha, ts, 2000, 0.03, t_start, t_start + tl() * 0.2)
            if c5_gd < best_c5:
                best_c5 = c5_gd
                best_h = h_gd.copy()

    if tl() > 60:
        h_f, c5_f = focused_refinement(best_h, ts, rng, t_start, t_start + tl() * 0.25)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    if tl() > 120:
        h_mr = multi_resolution(best_h, 360, t_start, t_start + tl() * 0.3, rng)
        h_mr = project_to_sum(h_mr, 180.0)
        c5_mr = compute_c5(h_mr)
        if c5_mr < best_c5:
            best_c5 = c5_mr
            best_h = h_mr.copy()
            best_n = 360

    if tl() > 40:
        n = best_n
        ts = n / 2.0
        h_gd, c5_gd = projected_grad_descent(best_h, 500.0, ts, 1500, 0.02, t_start, t_start + tl() * 0.3)
        if c5_gd < best_c5:
            best_c5 = c5_gd
            best_h = h_gd.copy()

    if tl() > 30:
        n = best_n
        ts = n / 2.0
        h_f, c5_f = focused_refinement(best_h, ts, rng, t_start, t_start + tl() * 0.3)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    if tl() > 20:
        n = best_n
        ts = n / 2.0
        h_ft, c5_ft = fine_tune(best_h, ts, rng, t_start, t_start + tl() * 0.7)
        if c5_ft < best_c5:
            best_c5 = c5_ft
            best_h = h_ft.copy()

    n_final = len(best_h)
    ts_final = n_final / 2.0
    best_h = project_to_sum(best_h, ts_final)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n_final)
```
