from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(corr, alpha):
    m = np.max(corr)
    shifted = alpha * (corr - m)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    w = exp_vals / sum_exp
    sm = m + np.log(sum_exp) / alpha
    return sm, w

def compute_gradient(h, alpha):
    """Compute gradient of smooth max of correlations w.r.t. h"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    sm, w = smooth_max_and_grad(corr, alpha)
    
    # corr[m] = sum_i h[i] * (1-h)[i + n - 1 - m]
    # d(corr[m])/d(h[j]) = (1-h[j + n-1-m])_valid - h[j - n + 1 + m]_valid
    # Let d = n-1-m, so m = n-1-d
    # d(corr[n-1-d])/d(h[j]) = (1-h[j+d])_valid - h[j-d]_valid
    
    # grad[h[j]] = dx * sum_m w[m] * ((1-h[j+n-1-m])_valid - h[j-n+1+m]_valid)
    # Let d = n-1-m, so:
    # grad[h[j]] = dx * sum_d w[n-1-d] * ((1-h[j+d])_valid - h[j-d]_valid)
    
    grad = np.zeros(n)
    # d ranges from -(n-1) to n-1
    # w is indexed by m = 0 to 2n-2, corresponding to d = n-1-m
    # So w[m] corresponds to d = n-1-m
    
    for d in range(-(n-1), n):
        m = n - 1 - d
        wm = w[m]
        if wm < 1e-10:
            continue
        # (1-h[j+d]) for valid j: j in [0, n-1] and j+d in [0, n-1]
        # So j in [max(0,-d), min(n-1, n-1-d)]
        j_start = max(0, -d)
        j_end = min(n, n - d)  # exclusive
        if j_end > j_start:
            grad[j_start:j_end] += wm * (1.0 - h[j_start + d:j_end + d])
        
        # -h[j-d] for valid j: j in [0, n-1] and j-d in [0, n-1]
        # So j in [max(0,d), min(n-1, n-1+d)]
        j_start2 = max(0, d)
        j_end2 = min(n, n + d)  # exclusive
        if j_end2 > j_start2:
            grad[j_start2:j_end2] -= wm * h[j_start2 - d:j_end2 - d]
    
    grad *= dx
    return sm, grad

def projected_gradient_descent(h0, target_sum, alpha, n_iters, lr, rng):
    h = h0.copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    
    for it in range(n_iters):
        sm, grad = compute_gradient(h, alpha)
        
        # Adaptive learning rate
        h_new = h - lr * grad
        h_new = project_to_sum(h_new, target_sum)
        
        c5_new = compute_c5(h_new)
        if c5_new < c5:
            h = h_new
            c5 = c5_new
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:

            lr *= 0.5
            h_new = h - lr * grad
            h_new = project_to_sum(h_new, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < c5:
                h = h_new
                c5 = c5_new
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
        
        # Occasionally perturb
        if it % 100 == 50:
            h += rng.normal(0, 0.01, len(h))
            h = project_to_sum(h, target_sum)
    
    return best_h, best_c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = h.copy().astype(float)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_grad(corr, alpha, n):
    """Returns (value, gradient) of smooth max of corr w.r.t. h."""
    dx = 2.0 / n
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -700, 700)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    w = exp_vals / s
    val = (m + np.log(s) / alpha) * dx
    
    grad = np.zeros(n)
    for d in range(-(n-1), n):
        wm = w[n - 1 - d]
        if wm < 1e-10:
            continue
        js = max(0, -d)
        je = min(n, n - d)
        if je > js:
            grad[js:je] += wm * (1.0 - h[js+d:je+d])
        js2 = max(0, d)
        je2 = min(n, n + d)
        if je2 > js2:
            grad[js2:je2] -= wm * h[js2-d:je2-d]
    grad *= dx
    return val, grad

    n = len(h0)
    h = h0.copy()

    best_h = h.copy()
    best_c5 = c5
    v = np.zeros(n)

    lr = 0.1
    
    stalled = 0
    for it in range(n_iters):
        if time.time() - t_start > budget - reserve:
            break
        
        corr = np.correlate(h, 1.0 - h, mode="full")
        val, grad = smooth_max_grad(corr, alpha, n)

            c5 = c5_new
            stalled = 0

                best_h = h.copy()
            # Adaptively increase lr slightly when making progress

                v *= 0.5

                    best_h = h.copy()
    
    return best_h, best_c5

    return val, grad

def grad_descent(h0, target_sum, alpha, n_iters, t_start, budget, reserve, rng):
    n = len(h0)
    h = h0.copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    v = np.zeros(n)
    mu = 0.88
    lr = 0.08
    stalled = 0
    best_streak = 0
    for it in range(n_iters):
        if time.time() - t_start > budget - reserve:
            break
        val, grad = smooth_max_grad(h, alpha)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        v = mu * v - lr * grad
        h_new = project_to_sum(h + v, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5 - 1e-15:
            h = h_new
            c5 = c5_new
            stalled = 0
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                best_streak += 1
            lr = min(lr * 1.005, 0.4)
        else:
            stalled += 1
            best_streak = 0
            if stalled % 30 == 0:
                lr = max(lr * 0.85, 0.005)
                v *= 0.6
            if stalled > 200:
                h = best_h.copy()
                c5 = best_c5
                v = np.zeros(n)
                lr = 0.03
                stalled = 0
            h_new = project_to_sum(h - 0.02 * grad / max(gnorm, 1e-12), target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < c5:
                h = h_new
                c5 = c5_new
                v = -0.02 * grad / max(gnorm, 1e-12)
                stalled = 0
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
    return best_h, best_c5

def sa_optimize(h0, target_sum, n_iters, T0, T1, t_start, budget, reserve, rng):
    n = len(h0)
    h = h0.copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    T = T0
    cool = (T1 / T0) ** (1.0 / max(1, n_iters))
    for it in range(n_iters):
        if time.time() - t_start > budget - reserve:
            break
        r = rng.random()
        h_new = h.copy()
        if r < 0.35:
            k = rng.randint(1, max(2, n // 15))
            idx = rng.choice(n, size=k, replace=False)
            h_new[idx] += rng.normal(0, 0.025, k)
        elif r < 0.6:
            i, j = rng.choice(n, size=2, replace=False)
            delta = rng.uniform(-0.03, 0.03)
            h_new[i] += delta
            h_new[j] -= delta
        elif r < 0.8:
            i = rng.randint(n)
            h_new[i] += rng.normal(0, 0.04)
        else:
            start = rng.randint(0, n - 4)
            ln = rng.randint(3, min(7, n // 6))
            h_new[start:start + ln] += rng.normal(0, 0.015)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        diff = c5_new - c5
        if diff < 0 or (T > 1e-12 and rng.random() < np.exp(-diff / T)):
            h = h_new
            c5 = c5_new
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        T = max(T * cool, T1)
    return best_h, best_c5

def coord_refine(h0, target_sum, n_rounds, t_start, budget, reserve, rng):
    n = len(h0)
    h = h0.copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    steps = [0.0008, -0.0008, 0.0015, -0.0015, 0.0004, -0.0004, 0.003, -0.003]
    for rnd in range(n_rounds):
        if time.time() - t_start > budget - reserve:
            break
        improved = False
        for i in rng.permutation(n):
            base = h[i]
            for s in steps:
                h_new = h.copy()
                h_new[i] = np.clip(base + s, 0.0, 1.0)
                h_new = project_to_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < c5 - 1e-14:
                    h = h_new
                    c5 = c5_new
                    improved = True
                    break
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if not improved:
            break
    return best_h, best_c5

def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old, period=2.0)
    return h_new

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float)
        n = len(best_h)
        target_sum = n / 2.0
        best_h = project_to_sum(best_h, target_sum)
        best_c5 = compute_c5(best_h)
    else:
        n = 180
        target_sum = 90.0
        best_h = np.full(n, 0.5)
        best_c5 = compute_c5(best_h)

    h_work = best_h.copy()
    alphas = [25, 50, 100, 200, 400, 800, 1600, 3200]
    for i, alpha in enumerate(alphas):
        remaining = budget_s - (time.time() - t_start)
        reserve = max(150, int(remaining * 0.15))
        if time.time() - t_start > budget_s - reserve:
            break
        n_iter = min(8000, max(2000, int((budget_s - (time.time() - t_start) - reserve) / 0.0003)))
        h_work, c5 = grad_descent(h_work, target_sum, alpha, n_iter, t_start, budget_s, reserve, rng)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_work.copy()
        if i in [3, 6]:
            if time.time() - t_start < budget_s - 200:
                h_sa, c5_sa = sa_optimize(h_work, target_sum, 60000, 0.02, 0.0003, t_start, budget_s, 100, rng)
                if c5_sa < best_c5:
                    best_c5 = c5_sa
                    best_h = h_sa.copy()
                h_work = best_h.copy()

    if time.time() - t_start < budget_s - 100:
        T0 = 0.015
        h_work, c5 = sa_optimize(best_h, target_sum, 200000, T0, 0.0002, t_start, budget_s, 80, rng)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_work.copy()

    if time.time() - t_start < budget_s - 60:
        for alpha in [500, 1200, 3000]:
            if time.time() - t_start > budget_s - 45:
                break
            h_work, c5 = grad_descent(best_h, target_sum, alpha, 5000, t_start, budget_s, 35, rng)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_work.copy()

    if time.time() - t_start < budget_s - 15:
        h_work, c5 = coord_refine(best_h, target_sum, 300, t_start, budget_s, 5, rng)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_work.copy()

    if time.time() - t_start < budget_s - 10:
        for n_try in [n + 2, n + 4, n + 8]:
            if time.time() - t_start > budget_s - 8:
                break
            if n_try % 2 != 0:
                continue
            ts_try = n_try / 2.0
            h_try = interpolate_h(best_h, n_try)
            h_try = project_to_sum(h_try, ts_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                h_work = h_try
                for alpha in [100, 400, 1000]:
                    if time.time() - t_start > budget_s - 6:
                        break
                    h_work, c5_try = grad_descent(h_work, ts_try, alpha, 2000, t_start, budget_s, 4, rng)
                    if c5_try < best_c5:
                        best_c5 = c5_try
                        best_h = h_work.copy()
                        n = n_try

    best_h = project_to_sum(best_h, n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n)
