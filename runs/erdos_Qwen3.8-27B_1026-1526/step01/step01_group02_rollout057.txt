def project_feasible(h, n):

    h = np.array(h, dtype=float).copy()

            break

            mask = h < 1.0

                break
            step = min(diff / np.sum(mask), (1.0 - h[mask]).min())
            h[mask] += step
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            step = min(-diff / np.sum(mask), h[mask].min())
            h[mask] -= step

            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = np.argmax(h)
            h[idx] = max(0.0, h[idx] + diff)
    return h

        return t - c

        return np.sum(x[:-1]) - len(x[:-1]) / 2.0

        j = np.zeros(len(x))
