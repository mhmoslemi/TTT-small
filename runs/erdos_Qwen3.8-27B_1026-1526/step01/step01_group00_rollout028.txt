```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def get_active_lags(h, top_k=5):

    corr = np.correlate(h, 1.0 - h, mode="full")
    # The full correlation has length 2n-1
    # We want the lags that give the highest values
    idx = np.argsort(corr)[-top_k:]
    return idx

def targeted_mutation(h, target_sum, active_lags, rng, strength=0.03):
    """Mutation that specifically tries to reduce overlap at active lags"""
    n = len(h)
    h_new = h.copy()
    # For each active lag, try to reduce overlap
    for lag_idx in active_lags:
        # Convert lag index in full correlation to actual lag
        # full correlation index k corresponds to lag k - (n-1)
        lag = lag_idx - (n - 1)
        if lag == 0:
            continue  # can't easily reduce self-overlap
        # For positive lag: overlap is sum h[i]*(1-h[i+lag])
        # To reduce: increase h where h[i+lag] is already high (or decrease h[i] where h[i+lag] is low)
        # Simple approach: random perturbation biased towards reducing overlap

                i = j
                j = i - lag  # swap
            # Reduce overlap: if h[i] high and h[j] low, decrease h[i] or increase h[j]

def ga_tournament(pop, fits, rng, k=3):
    """Tournament selection"""
    idxs = rng.randint(0, len(pop), k)
    best_idx = idxs[np.argmin(fits[idxs])]  # minimize
    return pop[best_idx].copy()

def ga_uniform_crossover(p1, p2, rng):
    mask = rng.random(len(p1)) < 0.5
    c1 = np.where(mask, p1, p2)
    c2 = np.where(mask, p2, p1)
    return c1, c2

def ga_gaussian_mutation(h, rng, sigma=0.05, prob=0.1):
    h_new = h.copy()
    mut_mask = rng.random(len(h)) < prob
    h_new[mut_mask] += rng.normal(0, sigma, mut_mask.sum())
    return h_new

    step = 0.03

        # Adaptive step
        if no_improve > 50:
            step *= 0.8
            no_improve = 0
            if step < 0.001:

            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h.copy()
        else:
```

```python
    #          + δ * h[p-i] - δ*(1-h[p-i]) (from h[i] change, as 'v' component)
    #          - δ * h[p-j] + δ*(1-h[p-j]) (from h[j] change, as 'v' component)
    #
    # Wait, I need to be more careful.
    # corr[p] = sum_m h[m] * (1-h[p-m])
    # 
    # Effect of h[i] += δ:
    #   As 'a' component (m=i): adds δ * (1-h[p-i]) for valid p-i
    #   As 'v' component (p-m=i, so m=p-i): h[p-i] * (-δ) for valid p-i
    #   Total: δ * (1-h[p-i] - h[p-i]) = δ * (1 - 2*h[p-i]) for p-i in [0, n-1]
    #
    # Effect of h[j] -= δ:
    #   As 'a' component (m=j): adds (-δ) * (1-h[p-j]) for valid p-j
    #   As 'v' component (p-m=j, so m=p-j): h[p-j] * (+δ) for valid p-j
    #   Total: (-δ) * (1-h[p-j] - h[p-j]) = (-δ) * (1 - 2*h[p-j]) for p-j in [0, n-1]
    #
    # So Δcorr[p] = δ * (1 - 2*h[p-i]) - δ * (1 - 2*h[p-j]) for p in valid ranges
    #            = δ * [(1 - 2*h[p-i]) - (1 - 2*h[p-j])]
    #            = 2δ * (h[p-j] - h[p-i])
    #
    # Valid p for first term: p-i in [0, n-1], i.e., p in [i, i+n-1]
    # Valid p for second term: p-j in [0, n-1], i.e., p in [j, j+n-1]
    
    # d[p] = 2*(h[p-j] - h[p-i]) where both p-i and p-j are in [0, n-1]
    #        = 2*(h[p-j] - h[p-i]) + 2*h[p-i] where only p-i is valid (p-j not valid)
    #           wait no, if p-j not valid, the second term doesn't apply
    #        = -2*h[p-j] where only p-j is valid
    #        Hmm let me just compute it directly.
    
    d = np.zeros(n_corr)
    # From h[i] change:
    lo1 = max(i, 0)
    hi1 = min(i + n, n_corr)
    d[lo1:hi1] += (1.0 - 2.0 * h[lo1-i:hi1-i])
    # From h[j] change (with -δ):
    lo2 = max(j, 0)
    hi2 = min(j + n, n_corr)
    d[lo2:hi2] -= (1.0 - 2.0 * h[lo2-j:hi2-j])
    
    # Now we want to minimize max(corr + δ*d) over valid δ range
    # Valid range:
    lo_delta = max(-h[i], h[j] - 1.0)
    hi_delta = min(1.0 - h[i], h[j])
    
    if lo_delta >= hi_delta:
        return 0.0  # no feasible move
    
    # f(δ) = max_k (corr[k] + δ * d[k])
    # This is convex in δ. Find minimum over [lo_delta, hi_delta].
    
    # The minimum of a convex piecewise-linear function occurs at a breakpoint
    # or at a boundary. Breakpoints are where (corr[a] + δ*d[a]) = (corr[b] + δ*d[b])
    # i.e., δ = (corr[b] - corr[a]) / (d[a] - d[b])
    
    # Simple approach: use ternary search (since f is convex)
    # Or: evaluate at a few points and pick the best
    
    # For efficiency, let's use a bisection on the subgradient
    # f(δ) is convex, so its subgradient is monotonically increasing
    # f'(δ+) = max_{k: corr[k] + δ*d[k] = f(δ)} d[k]
    
    # Actually, simplest: just evaluate at a grid of points
    # Or use the fact that the optimal δ is at a breakpoint
    
    # Let me just do a fine search
    best_delta = 0.0
    best_val = np.max(corr)
    
    # Check boundaries
    for delta_test in [lo_delta, hi_delta]:
        val = np.max(corr + delta_test * d)
        if val < best_val:
            best_val = val
            best_delta = delta_test
    
    # Check some interior points
    # The optimal is where the two "tightest" constraints cross
    # For now, do a coarse-to-fine search
    n_grid = 50
    deltas = np.linspace(lo_delta, hi_delta, n_grid)
    vals = np.array([np.max(corr + δ * d) for δ in deltas])
    min_idx = np.argmin(vals)
    if vals[min_idx] < best_val:
        best_val = vals[min_idx]
        best_delta = deltas[min_idx]
    
    # Refine around the best
    if 0 < min_idx < n_grid - 1:
        lo = deltas[min_idx - 1]
        hi = deltas[min_idx + 1]
        for _ in range(20):
            mid1 = lo + (hi - lo) / 3
            mid2 = hi - (hi - lo) / 3
            if np.max(corr + mid1 * d) < np.max(corr + mid2 * d):
                hi = mid2
            else:
                lo = mid1
        best_delta = (lo + hi) / 2
        best_val = np.max(corr + best_delta * d)
    
    return best_delta
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def optimal_delta(corr, d, lo, hi):
    if lo >= hi:
        return 0.0
    best_d = 0.0
    best_v = float(np.max(corr))
    for dt in (lo, hi):
        v = float(np.max(corr + dt * d))
        if v < best_v:
            best_v = v
            best_d = dt
    n_grid = 30
    deltas = np.linspace(lo, hi, n_grid)
    for dt in deltas:
        v = float(np.max(corr + dt * d))
        if v < best_v:
            best_v = v
            best_d = dt
    if best_d > lo and best_d < hi:
        a, b = lo, hi
        for _ in range(30):
            m1 = a + (b - a) / 3.0
            m2 = b - (b - a) / 3.0
            if float(np.max(corr + m1 * d)) < float(np.max(corr + m2 * d)):
                b = m2
            else:
                a = m1
        best_d = 0.5 * (a + b)
        best_v = float(np.max(corr + best_d * d))
    return best_d

def pair_move_update(h, i, j, delta):
    n = len(h)
    n_corr = 2 * n - 1
    d = np.zeros(n_corr)
    lo1 = max(i, 0)
    hi1 = min(i + n, n_corr)
    d[lo1:hi1] += (1.0 - 2.0 * h[lo1 - i:hi1 - i])
    lo2 = max(j, 0)
    hi2 = min(j + n, n_corr)
    d[lo2:hi2] -= (1.0 - 2.0 * h[lo2 - j:hi2 - j])
    return d

def local_search_optimal_pairs(h0, n, target_sum, rng, time_limit, verbose=False):
    t0 = time.time()
    h = h0.copy()
    corr = compute_corr(h)
    dx = 2.0 / n
    c5_cur = float(np.max(corr)) * dx
    c5_best = c5_cur
    h_best = h.copy()
    patience = 0
    max_patience = 5000

    while time.time() - t0 < time_limit:
        if patience > max_patience:
            break

        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue

        d = pair_move_update(h, i, j, 1.0)
        lo = max(-h[i], h[j] - 1.0)
        hi = min(1.0 - h[i], h[j])
        if lo >= hi:
            continue

        best_delta = optimal_delta(corr, d, lo, hi)
        if abs(best_delta) < 1e-10:
            patience += 1
            continue

        h[i] += best_delta
        h[j] -= best_delta
        h[i] = np.clip(h[i], 0.0, 1.0)
        h[j] = np.clip(h[j], 0.0, 1.0)

        diff = target_sum - np.sum(h)
        if abs(diff) > 1e-10:

            if np.any(mask):
                h[mask] += diff / np.sum(mask)

            patience = 0
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h.copy()
        else:
            h[i] -= best_delta
            h[j] += best_delta
            h[i] = np.clip(h[i], 0.0, 1.0)
            h[j] = np.clip(h[j], 0.0, 1.0)
            diff = target_sum - np.sum(h)
            if abs(diff) > 1e-10:

            corr = compute_corr(h)

    return h_best, c5_best

def targeted_local_search(h0, n, target_sum, rng, time_limit):
    t0 = time.time()
    h = h0.copy()
    corr = compute_corr(h)
    dx = 2.0 / n
    c5_cur = float(np.max(corr)) * dx
    c5_best = c5_cur
    h_best = h.copy()
    n_corr = 2 * n - 1

    while time.time() - t0 < time_limit:
        top_p = int(np.argmax(corr))
        m_min = max(0, top_p - (n - 1))
        m_max = min(n - 1, top_p)
        if m_min > m_max:
            break
        m_range = np.arange(m_min, m_max + 1)
        terms = h[m_range] * (1.0 - h[top_p - m_range])
        top_terms = np.argsort(terms)[-5:]
        moved = False
        for ti in top_terms:
            m_star = int(m_range[ti])
            j_star = top_p - m_star
            if m_star == j_star:
                continue
            delta = min(0.03, 0.5 * h[m_star], 0.5 * (1.0 - h[j_star]))
            if delta < 1e-5:
                continue
            h[m_star] -= delta
            h[j_star] += delta
            h = np.clip(h, 0, 1)
            diff = target_sum - np.sum(h)
            if abs(diff) > 1e-10:
                mask = (h > 1e-10) & (h < 1.0 - 1e-10)
                if np.any(mask):
                    h[mask] += diff / np.sum(mask)
                    h = np.clip(h, 0, 1)
            new_corr = compute_corr(h)
            c5_new = float(np.max(new_corr)) * dx
            if c5_new < c5_cur - 1e-14:
                corr = new_corr
                c5_cur = c5_new
                moved = True
                if c5_new < c5_best:
                    c5_best = c5_new
                    h_best = h.copy()
            else:
                h[m_star] += delta
                h[j_star] -= delta
                h = np.clip(h, 0, 1)
                diff = target_sum - np.sum(h)
                if abs(diff) > 1e-10:
                    mask = (h > 1e-10) & (h < 1.0 - 1e-10)
                    if np.any(mask):
                        h[mask] += diff / np.sum(mask)
                        h = np.clip(h, 0, 1)
        if not moved:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            delta = rng.uniform(-0.02, 0.02)
            if h[i] + delta < 0 or h[i] + delta > 1 or h[j] - delta < 0 or h[j] - delta > 1:
                continue
            h[i] += delta
            h[j] -= delta
            h = np.clip(h, 0, 1)
            diff = target_sum - np.sum(h)
            if abs(diff) > 1e-10:

                    h[mask] += diff / np.sum(mask)

            new_corr = compute_corr(h)
            c5_new = float(np.max(new_corr)) * dx
            if c5_new < c5_cur - 1e-14:
                corr = new_corr
                c5_cur = c5_new
                if c5_new < c5_best:
                    c5_best = c5_new
                    h_best = h.copy()
            else:
                h[i] -= delta
                h[j] += delta
                h = np.clip(h, 0, 1)

                if abs(diff) > 1e-10:
                    mask = (h > 1e-10) & (h < 1.0 - 1e-10)
                    if np.any(mask):
                        h[mask] += diff / np.sum(mask)
                        h = np.clip(h, 0, 1)
                corr = compute_corr(h)
                c5_cur = float(np.max(corr)) * dx

    return h_best, c5_best

def ga_phase(h_init, n, target_sum, rng, time_limit, pop_size=60):
    t0 = time.time()
    x = np.linspace(0, 2, n, endpoint=False)
    population = [h_init.copy()]
    fitness = [compute_c5(h_init)]

    for freq in range(1, 10):
        h_s = 0.5 + 0.35 * np.sin(2 * np.pi * freq * x / 2.0 + rng.uniform(0, 2 * np.pi))
        h_s = project_to_sum(h_s, target_sum)
        population.append(h_s)
        fitness.append(compute_c5(h_s))

    for _ in range(max(0, pop_size - len(population))):
        h_r = rng.uniform(0, 1, n)
        h_r = project_to_sum(h_r, target_sum)
        population.append(h_r)
        fitness.append(compute_c5(h_r))

    best_h = population[0].copy()
    best_f = fitness[0]

    for gen in range(500):
        if time.time() - t0 > time_limit:
            break
        order = np.argsort(fitness)
        population = [population[i].copy() for i in order]
        fitness = [fitness[i] for i in order]

        elite_count = max(3, pop_size // 10)
        new_pop = [p.copy() for p in population[:elite_count]]
        new_fit = fitness[:elite_count]

        while len(new_pop) < pop_size:
            if time.time() - t0 > time_limit:
                break
            t1 = rng.randint(0, min(pop_size, 8))
            t2 = rng.randint(0, min(pop_size, 8))
            p1 = population[t1]
            p2 = population[t2]
            if rng.random() < 0.9:
                mask = rng.random(n) < 0.5
                c1 = np.where(mask, p1, p2).copy()
                c2 = np.where(mask, p2, p1).copy()
            else:
                c1 = p1.copy()
                c2 = p2.copy()
            sigma = 0.05 + 0.05 * (1.0 - gen / 500.0)
            c1 += rng.normal(0, sigma, n)
            c2 += rng.normal(0, sigma, n)
            c1 = project_to_sum(c1, target_sum)
            c2 = project_to_sum(c2, target_sum)
            f1 = compute_c5(c1)
            f2 = compute_c5(c2)
            new_pop.append(c1)
            new_fit.append(f1)
            if len(new_pop) < pop_size:
                new_pop.append(c2)
                new_fit.append(f2)

        population = new_pop[:pop_size]
        fitness = new_fit[:pop_size]
        min_i = int(np.argmin(fitness))
        if fitness[min_i] < best_f:
            best_f = fitness[min_i]
            best_h = population[min_i].copy()

        if gen % 15 == 14 and time.time() - t0 < time_limit * 0.8:
            h_ls, f_ls = targeted_local_search(population[0], n, target_sum, rng, 3.0)
            if f_ls < best_f:
                best_f = f_ls
                best_h = h_ls.copy()
            worst_i = int(np.argmax(fitness))
            population[worst_i] = h_ls
            fitness[worst_i] = f_ls

    return best_h, best_f

            remaining2 = t_phase2 - time.time()
            h_tg, f_tg = targeted_local_search(best_h, n, target_sum, rng, remaining2 * 0.5)
            if f_tg < best_c5:
                best_c5 = f_tg
                best_h = h_tg.copy()

    # Phase 3: Alternating refinement
    if time.time() < t_phase3:
        remaining = t_phase3 - time.time()
        n_rounds = 3
        for rnd in range(n_rounds):
            if time.time() > t_phase3 - 5:
                break
            sub_time = (t_phase3 - 5 - time.time()) / (n_rounds - rnd)
            h_ls, f_ls = local_search_optimal_pairs(best_h, n, target_sum, rng, sub_time * 0.7)
            if f_ls < best_c5:
                best_c5 = f_ls
                best_h = h_ls.copy()
            if time.time() < t_phase3 - 5:
                h_tg, f_tg = targeted_local_search(best_h, n, target_sum, rng, sub_time * 0.3)
                if f_tg < best_c5:
                    best_c5 = f_tg
                    best_h = h_tg.copy()

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_sum(best_h, target_sum)
    best_h = np.clip(best_h, 0.0, 1.0)
    final_sum = np.sum(best_h)
    if abs(final_sum - target_sum) > 1e-8:
        diff = target_sum - final_sum
        mask = (best_h > 1e-10) & (best_h < 1.0 - 1e-10)
        if np.any(mask):
            best_h[mask] += diff / np.sum(mask)
            best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, best_n)
```
