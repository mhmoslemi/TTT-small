    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argsort(corr)[-top_k:]
    top_vals = corr[idx]
    m = np.max(top_vals)
    exp_vals = np.exp(alpha * (top_vals - m))
    w = exp_vals / np.sum(exp_vals)
    obj = (m + np.log(np.sum(exp_vals)) / alpha) * dx
    grad = np.zeros(n)
    for l in range(top_k):
        lag = idx[l]
        d = lag - (n - 1)
        if d >= 0:
            grad[d:] += w[l] * (1.0 - h[:n - d])
            grad[:n - d] -= w[l] * h[d:]
        else:
            dn = -d
            grad[:n - dn] += w[l] * (1.0 - h[dn:])
            grad[dn:] -= w[l] * h[:n - dn]
    grad *= dx
    return obj, grad

def _opt_objective(x):
    o, _ = obj_grad(x, _opt_alpha, _opt_top_k)
    return o

def _opt_gradient(x):
    _, g = obj_grad(x, _opt_alpha, _opt_top_k)
    return g

def _constraint_fun(x):
    return np.sum(x) - _c5_target_sum

def _constraint_jac(x):
    return np.ones(len(x))

def slsqp_run(h0, alpha, top_k, maxiter, time_limit_s):
    global _opt_alpha, _opt_top_k, _c5_target_sum
    n = len(h0)
    _opt_alpha = alpha
    _opt_top_k = top_k
    _c5_target_sum = n / 2.0
    t0 = time.time()
    constraints = [{'type': 'eq', 'fun': _constraint_fun, 'jac': _constraint_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_opt_objective, h0.copy(), jac=_opt_gradient, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-13, 'disp': False})
        result = project_to_sum(res.x, n / 2.0)
    except Exception:
        result = project_to_sum(h0, n / 2.0)
    return result

def subgradient_refine(h0, n_iters, step_size_init, target_sum):
    h_cur = h0.copy()
    n = len(h_cur)
    c5_best = compute_c5(h_cur)
    h_best = h_cur.copy()
    step_size = step_size_init
    for it in range(n_iters):
        corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
        k_star = int(np.argmax(corr))
        d = k_star - (n - 1)
        grad = np.zeros(n)
        if d >= 0:
            grad[d:] += (1.0 - h_cur[:n - d])
            grad[:n - d] -= h_cur[d:]
        else:
            dn = -d
            grad[:n - dn] += (1.0 - h_cur[dn:])
            grad[dn:] -= h_cur[:n - dn]
        h_new = h_cur - step_size * grad
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new.copy()
        h_cur = h_new
        step_size *= 0.9995
    return h_best, c5_best

def stochastic_refine(h0, n_iters, target_sum, rng, t0_temp):
    h_cur = h0.copy()
    n = len(h_cur)
    c5_cur = compute_c5(h_cur)
    c5_best = c5_cur
    h_best = h_cur.copy()
    T = t0_temp
    cool = max(0.0001 / T, 0.99)
    for it in range(n_iters):
        r = rng.random()
        if r < 0.35:
            i = rng.randint(n)
            delta = rng.uniform(-0.04, 0.04)
            h_new = h_cur.copy()
            h_new[i] = np.clip(h_new[i] + delta, 0.0, 1.0)
            h_new = project_to_sum(h_new, target_sum)
        elif r < 0.7:
            i, j = rng.choice(n, size=2, replace=False)
            delta = rng.uniform(-0.025, 0.025)
            h_new = h_cur.copy()
            h_new[i] = np.clip(h_new[i] + delta, 0.0, 1.0)
            h_new[j] = np.clip(h_new[j] - delta, 0.0, 1.0)
            h_new = project_to_sum(h_new, target_sum)
        else:
            blk = rng.randint(3, 15)
            i = rng.randint(max(1, n - blk))
            delta = rng.uniform(-0.015, 0.015)
            h_new = h_cur.copy()
            h_new[i:i + blk] += delta
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur or rng.random() < np.exp(-(c5_new - c5_cur) / max(T, 1e-10)):
            h_cur = h_new
            c5_cur = c5_new
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h_new.copy()
        T = max(T * cool, 0.0001)
    return h_best, c5_best

def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old, period=2.0)
    target = n_new / 2.0
    h_new = project_to_sum(h_new, target)
    return h_new

def focused_perturb(h, rng, n_perturb):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[-5:]
    h_new = h.copy()
    for _ in range(n_perturb):
        lag = top_idx[rng.randint(len(top_idx))]
        d = lag - (n - 1)
        if d >= 0 and d < n:
            i = rng.randint(d, n)
            j = rng.randint(0, n - d)
            if i == j:
                continue
            delta = rng.uniform(-0.01, 0.01)
            h_new[i] = np.clip(h_new[i] + delta, 0.0, 1.0)
            h_new[j] = np.clip(h_new[j] - delta, 0.0, 1.0)
        elif d < 0:
            dn = -d
            i = rng.randint(0, n - dn)
            j = rng.randint(dn, n)
            if i == j:
                continue
            delta = rng.uniform(-0.01, 0.01)
            h_new[i] = np.clip(h_new[i] + delta, 0.0, 1.0)
            h_new[j] = np.clip(h_new[j] - delta, 0.0, 1.0)
    h_new = project_to_sum(h_new, n / 2.0)
    return h_new

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()

    def tl():
        return budget_s - (time.time() - t_start)

    best_h = None
    best_c5 = float('inf')
    best_n = 0
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in dir() and initial_h_values is not None:
        initial_h = np.array(initial_h_values, dtype=float)
        n_init = len(initial_h)
        ts = n_init / 2.0
        initial_h = project_to_sum(initial_h, ts)
        best_h = initial_h.copy()
        best_c5 = compute_c5(best_h)
        best_n = n_init
    else:
        best_n = 180
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP with various top_k and alpha at current resolution
    for top_k in [1, 2, 3, 5, 8, 12]:
        if tl() < 80:
            break
        for alpha in [30.0, 80.0, 200.0]:
            if tl() < 40:
                break
            h_opt = slsqp_run(best_h, alpha, top_k, 300, min(60, tl() - 5))
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Phase 2: Subgradient refinement
    if tl() > 120:
        n_iters = min(8000, int(tl() * 2))
        h_sg, c5_sg = subgradient_refine(best_h, n_iters, 0.008, best_n / 2.0)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # Phase 3: Another round of SLSQP after subgradient
    for top_k in [1, 3, 5]:
        if tl() < 50:
            break
        for alpha in [50.0, 150.0, 400.0]:
            if tl() < 30:
                break
            h_opt = slsqp_run(best_h, alpha, top_k, 400, min(50, tl() - 5))
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Phase 4: Stochastic local search
    if tl() > 100:
        n_sa = min(15000, int(tl() * 2))
        h_sa, c5_sa = stochastic_refine(best_h, n_sa, best_n / 2.0, rng, 0.003)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    # Phase 5: SLSQP + perturbation iterations
    for iteration in range(10):
        if tl() < 60:
            break
        h_pert = focused_perturb(best_h, rng, 5)
        c5_pert = compute_c5(h_pert)
        h_base_iter = h_pert if c5_pert < best_c5 else best_h.copy()
        if c5_pert < best_c5:
            best_c5 = c5_pert
            best_h = h_pert.copy()
        for top_k in [3, 5]:
            if tl() < 30:
                break
            h_opt = slsqp_run(h_base_iter, 100.0, top_k, 200, min(40, tl() - 5))
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                h_base_iter = h_opt.copy()

    # Phase 6: Try higher resolutions
    higher_ns = [240, 300, 360, 420, 480]
    for n_new in higher_ns:
        if tl() < 150:
            break
        if n_new <= best_n:
            continue
        ts_new = n_new / 2.0
        h_interp = interpolate_h(best_h, best_n, n_new)
        c5_interp = compute_c5(h_interp)
        if c5_interp < best_c5:
            best_c5 = c5_interp
            best_h = h_interp.copy()
            best_n = n_new
        else:
            pass
        for top_k in [3, 5, 8]:
            if tl() < 60:
                break
            for alpha in [50.0, 150.0]:
                if tl() < 30:
                    break
                h_opt = slsqp_run(h_interp, alpha, top_k, 200, min(50, tl() - 10))
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    best_n = n_new
                h_interp = h_opt
        if tl() > 80:
            n_sa2 = min(8000, int(tl() * 2))
            h_sa2, c5_sa2 = stochastic_refine(h_interp, n_sa2, ts_new, rng, 0.002)
            if c5_sa2 < best_c5:
                best_c5 = c5_sa2
                best_h = h_sa2.copy()
                best_n = n_new

    # Phase 7: Final intensive polish
    for _ in range(5):
        if tl() < 25:
            break
        h_opt = slsqp_run(best_h, 200.0, 3, 500, min(20, tl() - 3))
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    if tl() > 15:
        h_final_sg, c5_final_sg = subgradient_refine(best_h, 3000, 0.003, best_n / 2.0)
        if c5_final_sg < best_c5:
            best_c5 = c5_final_sg
            best_h = h_final_sg.copy()

    # Final validation
    n_final = len(best_h)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_sum(best_h, n_final / 2.0)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n_final)
