def _fftograd(h, n, tau):
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    v = corr / tau
    vmax = np.max(v)
    exp_v = np.exp(v - vmax)

    L = 1

        L *= 2
    h_pad = np.zeros(L)
    h_pad[:n] = h
    q_pad = np.zeros(L)
    q_pad[:2 * n - 1] = p
    qf_pad = np.zeros(L)
    qf_pad[:2 * n - 1] = p[::-1]
    H = np.fft.rfft(h_pad, L)
    B = np.fft.irfft(H * np.fft.rfft(q_pad, L), L)[:n]
    A = np.fft.irfft(H * np.fft.rfft(qf_pad, L), L)[:n]

def _obj_grad(h, n, tau, penalty):
