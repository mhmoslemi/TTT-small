def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_vectorized(h, n, m):
    dx = 2.0 / n
    grad = np.zeros(n)
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
    elif m > 0:
        grad[0:n-m] += 1.0
        grad[0:n-m] -= h[m:n]
        grad[m:n] -= h[0:n-m]
    else:
        m_abs = -m
        grad[m_abs:n] += 1.0
        grad[m_abs:n] -= h[0:n-m_abs]
        grad[0:n-m_abs] -= h[m_abs:n]
    return grad * dx

def interpolate_h(h_src, n_src, n_dst):

def slsqp_refine(h0, n, target_sum, maxiter=200, ftol=1e-10):
    def objective(h_flat):
        h_c = np.clip(h_flat, 0.0, 1.0)
        vals = compute_corr_vals(h_c)
        m = np.max(vals)
        alpha = 200.0
        exp_vals = np.exp(alpha * (vals - m))
        return m + np.log(np.sum(exp_vals) / len(exp_vals)) / alpha
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]

                       constraints=constraints, options={'maxiter': maxiter, 'ftol': ftol})

        return project_to_sum(h0, target_sum)

            break
        ts = scale_n / 2.0
        
        if n_current != scale_n:
            h_current = interpolate_h(h_current, n_current, scale_n)
            h_current = project_to_sum(h_current, ts)
            n_current = scale_n
        
        c5_cur = compute_c5(h_current)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_current.copy()
            best_n = scale_n
        
        # Subgradient with momentum
        h_sg, c5_sg = subgradient_optimize(
            h_current, scale_n, ts,
            max_iters=400, lr0=0.008, lr_decay=0.997,
            top_k=5, threshold=0.004, momentum=0.85, patience=150
        )
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
            best_n = scale_n
        h_current = h_sg
        
        # SLSQP refinement
        if time_left() > 40:
            h_sl = slsqp_refine(h_current, scale_n, ts, maxiter=150)
            c5_sl = compute_c5(h_sl)
            if c5_sl < best_c5:
                best_c5 = c5_sl
                best_h = h_sl.copy()
                best_n = scale_n
            h_current = h_sl
        
        # Fine coordinate descent
        if time_left() > 30:
            h_cd, c5_cd = fine_coordinate_descent(
                h_current, scale_n, ts, rng,
                max_rounds=15, step_sizes=[0.02, 0.01, 0.005, 0.002]
            )
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()
                best_n = scale_n
            h_current = h_cd
    
    # Phase 2: Intensive local search at best resolution
    n = best_n
    ts = n / 2.0
    h = best_h.copy()
    
    phase2_start = time.time()
    phase2_budget = 300
    
    while (time.time() - phase2_start) < phase2_budget and time_left() > 30:
        # Subgradient with varying parameters
        for top_k in [3, 5, 8]:
            if time_left() < 25:
                break
            h_sg, c5_sg = subgradient_optimize(
                h, n, ts,
                max_iters=200, lr0=0.005, lr_decay=0.998,
                top_k=top_k, threshold=0.003, momentum=0.8, patience=100
            )
            if c5_sg < best_c5:
                best_c5 = c5_sg
                best_h = h_sg.copy()
            h = h_sg
        
        # Pair search
        if time_left() > 20:
            h_ps, c5_ps = pair_search(h, n, ts, rng, n_iters=3000, step_range=(0.003, 0.02))
            if c5_ps < best_c5:
                best_c5 = c5_ps
                best_h = h_ps.copy()
            h = h_ps
        
        # Fine coordinate descent
        if time_left() > 20:
            h_cd, c5_cd = fine_coordinate_descent(
                h, n, ts, rng,
                max_rounds=10, step_sizes=[0.01, 0.005, 0.002, 0.001]
            )
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()
            h = h_cd
    
    # Phase 3: Try other resolutions
    if time_left() > 50:
        for n_try in [150, 180, 200, 240, 300, 360, 480]:
            if time_left() < 25:
                break
            if n_try == best_n:
                continue
            if n_try % 2 != 0:
                continue
            ts_try = n_try / 2.0
            if abs(ts_try - round(ts_try)) > 1e-6:
                continue
            
            h_try = interpolate_h(best_h, best_n, n_try)
            h_try = project_to_sum(h_try, ts_try)
            c5_try = compute_c5(h_try)
            
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = n_try
            
            if time_left() > 20:
                h_opt, c5_opt = subgradient_optimize(
                    h_try, n_try, ts_try,
                    max_iters=200, lr0=0.006, lr_decay=0.997,
                    top_k=5, threshold=0.004, momentum=0.85, patience=100
                )
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    best_n = n_try
                
                if time_left() > 15:
                    h_cd, c5_cd = fine_coordinate_descent(
                        h_opt, n_try, ts_try, rng,
                        max_rounds=8, step_sizes=[0.01, 0.005, 0.002]
                    )
                    if c5_cd < best_c5:
                        best_c5 = c5_cd
                        best_h = h_cd.copy()
                        best_n = n_try
    
    # Phase 4: Final polish
    if time_left() > 15:
        n_fin = best_n
        ts_fin = n_fin / 2.0
        
        h_pol = slsqp_refine(best_h, n_fin, ts_fin, maxiter=300)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
        
        if time_left() > 10:
            h_cd, c5_cd = fine_coordinate_descent(
                best_h, n_fin, ts_fin, rng,
                max_rounds=20, step_sizes=[0.005, 0.003, 0.002, 0.001, 0.0005]
            )
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()
        
        # One more round of very fine pair search
        if time_left() > 10:
            h_ps, c5_ps = pair_search(
                best_h, n_fin, ts_fin, rng,
                n_iters=5000, step_range=(0.001, 0.008)
            )
            if c5_ps < best_c5:
                best_c5 = c5_ps
                best_h = h_ps.copy()
    
    # Final validation
    n_final = len(best_h)
    ts_final = n_final / 2.0
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_sum(best_h, ts_final)
    best_c5 = compute_c5(best_h)
    
    return (best_h, best_c5, n_final)
