import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_exact(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_new = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(h_new)
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    h_new = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - np.sum(h_new)
    if abs(diff) > 1e-13:
        mask = (h_new > 1e-12) & (h_new < 1.0 - 1e-12)
        if np.any(mask):
            h_new[mask] += diff / np.count_nonzero(mask)
            h_new = np.clip(h_new, 0.0, 1.0)
    return h_new

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m))
    w = exp_vals / np.sum(exp_vals)
    f_val = m + np.log(np.sum(exp_vals)) / alpha
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        wk = w[k_idx]
        if wk < 1e-14:
            continue
        k = k_idx - (n - 1)
        j_lo = max(0, k)
        j_hi = min(n - 1, n - 1 + k)
        if j_lo <= j_hi:
            grad[j_lo:j_hi + 1] += wk * (1.0 - h[j_lo - k:j_hi - k + 1])
        j_lo2 = max(0, -k)
        j_hi2 = min(n - 1, n - 1 - k)
        if j_lo2 <= j_hi2:
            grad[j_lo2:j_hi2 + 1] -= wk * h[j_lo2 + k:j_hi2 + k + 1]
    return f_val, grad

def pgd_with_line_search(h0, target_sum, alpha, lr_init, max_iters, time_limit):
    h = h0.copy()
    c5_cur = compute_c5(h)
    best_c5 = c5_cur
    best_h = h.copy()
    t0 = time.time()
    step = lr_init
    stall = 0
    for it in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        f_val, grad = smooth_max_and_grad(h, alpha)
        h_trial = project_exact(h - step * grad, target_sum)
        c5_trial = compute_c5(h_trial)
        if c5_trial < c5_cur - 1e-14:
            h = h_trial
            c5_cur = c5_trial
            stall = 0
            if c5_cur < best_c5 - 1e-10:
                best_c5 = c5_cur
                best_h = h.copy()
            step = min(step * 1.1, 1.0)
        else:
            step *= 0.6
            stall += 1
            if stall > 30:
                break
            if step < 1e-12:
                break
    return best_h, best_c5

def fine_local_search(h0, target_sum, n, max_rounds, time_limit, seed=999):
    h = h0.copy()
    c5_cur = compute_c5(h)
    best_c5 = c5_cur
    best_h = h.copy()
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deltas = [0.005, -0.005, 0.002, -0.002, 0.01, -0.01, 0.001, -0.001]
    for rnd in range(max_rounds):
        if time.time() - t0 > time_limit:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() - t0 > time_limit:
                break
            for d in deltas:
                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] + d, 0.0, 1.0)
                h_new = project_exact(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < c5_cur - 1e-13:
                    h = h_new
                    c5_cur = c5_new
                    improved = True
                    if c5_cur < best_c5:
                        best_c5 = c5_cur
                        best_h = h.copy()
                    break
        if not improved:
            break
    return best_h, best_c5

def pair_local_search(h0, target_sum, n, max_rounds, time_limit, seed=777):
    h = h0.copy()

    best_c5 = c5_cur
    best_h = h.copy()
    t0 = time.time()
    rng = np.random.RandomState(seed)

            break

        for _ in range(n // 2):
            if time.time() - t0 > time_limit:
                break

                    h_new = h.copy()
                    h_new[i] = np.clip(h_new[i] + di, 0.0, 1.0)
                    h_new[j] = np.clip(h_new[j] + dj, 0.0, 1.0)
                    h_new = project_exact(h_new, target_sum)
                    c5_new = compute_c5(h_new)
                    if c5_new < c5_cur - 1e-13:

                        c5_cur = c5_new
                        improved = True
                        if c5_cur < best_c5:
                            best_c5 = c5_cur

                        break

            break
    return best_h, best_c5

def simulate_anneal_fine(h0, target_sum, n, max_iters, T0, T_min, seed=55):
    h = h0.copy()
    c5_cur = compute_c5(h)
    best_c5 = c5_cur
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    T = T0
    cool = (T_min / T0) ** (1.0 / max(1, max_iters))
    for it in range(max_iters):
        i = rng.randint(n)
        delta = rng.uniform(-0.03, 0.03)
        h_new = h.copy()
        h_new[i] = np.clip(h_new[i] + delta, 0.0, 1.0)
        h_new = project_exact(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur or rng.random() < np.exp(-max(0, c5_new - c5_cur) / max(T, 1e-15)):
            h = h_new
            c5_cur = c5_new
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h.copy()
        T = max(T * cool, T_min)
    return best_h, best_c5
