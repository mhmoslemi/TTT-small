import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.count_nonzero(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m))
    s = np.sum(exp_vals)
    w = exp_vals / s
    val = m + np.log(s) / alpha
    one_minus_h = 1.0 - h
    grad = np.empty(n)
    for j in range(n):
        grad[j] = np.dot(w[j:n+j][::-1], one_minus_h) - np.dot(w[n-1-j:2*n-1-j], h)
    return val, grad

    best_val = val
    momentum = np.zeros(n)

    for it in range(max_iter):

            break

        direction = grad_proj / grad_norm

            step *= 0.5
