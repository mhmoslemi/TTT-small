def compute_c5(h):

    dx = 2.0 / n

    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:

def smooth_max(vals, alpha):
    m = np.max(vals)
    if m < -1e30:
        return -1e30
    exp_vals = np.exp(alpha * (vals - m))
    return m + np.log(np.sum(exp_vals) / len(exp_vals)) / alpha

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def corr_array(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(vals, alpha):
    m = np.max(vals)
    exp_vals = np.exp(alpha * (vals - m))
    return m + np.log(np.sum(exp_vals) / len(exp_vals)) / alpha

def random_binary(n, rng, num_ones):
    h = np.zeros(n)
    ones = rng.choice(n, size=num_ones, replace=False)
    h[ones] = 1.0
    return h

def repair_binary(h, n, target_ones, rng):
    h = (h > 0.5).astype(float)
    cur_ones = int(np.sum(h))
    if cur_ones > target_ones:
        excess = cur_ones - target_ones
        ones_idx = np.where(h == 1.0)[0]
        remove = rng.choice(ones_idx, size=excess, replace=False)
        h[remove] = 0.0
    elif cur_ones < target_ones:
        deficit = target_ones - cur_ones
        zeros_idx = np.where(h == 0.0)[0]
        add = rng.choice(zeros_idx, size=deficit, replace=False)
        h[add] = 1.0
    return h

def greedy_local_search_binary(h0, n, target_ones, max_rounds, time_start, time_budget):
    h = h0.copy()
    c5_cur = compute_c5(h)
    corr_cur = corr_array(h)
    max_cur = np.max(corr_cur)
    
    for rnd in range(max_rounds):
        if (time.time() - time_start) > time_budget:
            break
        improved = False
        ones_idx = np.where(h == 1.0)[0]
        zeros_idx = np.where(h == 0.0)[0]
        best_improvement = 0
        best_i = -1
        best_j = -1
        
        for i in ones_idx:
            for j in zeros_idx:
                h_test = h.copy()
                h_test[i] = 0.0
                h_test[j] = 1.0
                corr_test = corr_array(h_test)
                max_test = np.max(corr_test)
                improvement = max_cur - max_test
                if improvement > best_improvement:
                    best_improvement = improvement
                    best_i = i
                    best_j = j
        
        if best_i >= 0 and best_improvement > 1e-12:
            h[best_i] = 0.0
            h[best_j] = 1.0
            corr_cur = corr_array(h)
            max_cur = np.max(corr_cur)
            improved = True
        
        if not improved:
            break
    
    c5_final = compute_c5(h)
    return h, c5_final

def binary_sa(h0, n, target_ones, rng, max_iters, T0, T_end, time_start, time_budget):
    h = h0.copy()
    c5_cur = compute_c5(h)
    max_cur = float(np.max(corr_array(h)))
    c5_best = c5_cur
    h_best = h.copy()
    T = T0
    cool = (T_end / T0) ** (1.0 / max(1, max_iters))
    
    for it in range(max_iters):
        if (time.time() - time_start) > time_budget:
            break
        ones_idx = np.where(h == 1.0)[0]
        zeros_idx = np.where(h == 0.0)[0]

        j = zeros_idx[rng.randint(len(zeros_idx))]

        h_new[i] = 0.0
        h_new[j] = 1.0

                c5_best = max_new * (2.0 / n)
                h_best = h.copy()
        
        T = max(T * cool, T_end)
    
    return h_best, c5_best

def tournament_select(fitnesses, rng, size):

def continuous_refine(h0, n, target_sum, time_start, time_budget):
    h = h0.astype(float).copy()
    h = project_to_sum(h, target_sum)
    c5_cur = compute_c5(h)
    
    if (time.time() - time_start) > time_budget - 30:
        return h, c5_cur
    
    alphas = [50.0, 200.0, 500.0]
    for alpha in alphas:
        if (time.time() - time_start) > time_budget - 10:
            break
        h_work = h.copy()
        
        def make_obj(a):
            def obj(x):
                xc = np.clip(x, 0.0, 1.0)
                corr = corr_array(xc)
                return smooth_max(corr, a)
            return obj
        
        obj_func = make_obj(alpha)
        try:
            res = minimize(obj_func, h_work, method='Nelder-Mead',
                          options={'maxiter': 3000, 'xatol': 1e-7, 'fatol': 1e-9, 'adaptive': True})
            h_cand = project_to_sum(res.x, target_sum)
            c5_cand = compute_c5(h_cand)
            if c5_cand < c5_cur:
                h = h_cand
                c5_cur = c5_cand
        except:
            pass
    
    return h, c5_cur

    if time_left_fn() > 15:
        n_final = len(best_h)
        ts_final = float(n_final // 2)
        h_final_bin = (best_h > 0.5).astype(float)
        h_final_bin = repair_binary(h_final_bin, n_final, n_final // 2, rng)
        h_pol, c5_pol = continuous_refine(h_final_bin, n_final, ts_final, t_start, int(time_left_fn() * 0.6))
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
    
    n_final = len(best_h)
    ts_final = float(n_final // 2)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_sum(best_h, ts_final)
    best_c5 = compute_c5(best_h)
    
    return (best_h, best_c5, n_final)
