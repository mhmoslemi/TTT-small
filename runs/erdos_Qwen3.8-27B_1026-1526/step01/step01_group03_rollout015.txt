import numpy as np

from scipy.interpolate import interp1d
import time


def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -3.0, 3.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    h2 = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(h2)
    if abs(diff) > 1e-14:
        order = np.argsort(np.abs(h2 - 0.5))
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                adj = min(diff, 1.0 - h2[idx])
                h2[idx] += adj
                diff -= adj
            else:
                adj = min(-diff, h2[idx])
                h2[idx] -= adj
                diff += adj
    return h2


def smooth_max_val(vals, tau):
    v = np.asarray(vals, dtype=float).copy()
    m = float(np.max(v))
    v -= m
    v /= tau
    ex = np.exp(v)
    s = float(np.sum(ex))
    return m + tau * np.log(s)


def find_active(h, n, top_frac=0.25):
    corr = np.correlate(h, 1.0 - h, mode="full")
    k = max(3, int(len(corr) * top_frac))
    thresh = np.partition(corr, len(corr) - k)[len(corr) - k]
    return np.where(corr >= thresh - 1e-10)[0]


def sm_objective(t, n, tau, active):
    dx = 2.0 / n
    corr = np.correlate(t, 1.0 - t, mode="full")
    return float(smooth_max_val(corr[active], tau) * dx)


def lbfgsb_optimize(h0, n, tau_list, t0, time_limit):
    h = project_feasible(h0.copy(), n)
    c5 = evaluate_c5(h)
    bounds = [(0.0, 1.0)] * n
    for tau in tau_list:
        if time.time() > time_limit:
            break
        active = find_active(h, n, 0.3)
        res = minimize(sm_objective, h.copy(), method="L-BFGS-B",
                       args=(n, tau, active), bounds=bounds,
                       options={"maxiter": 200, "ftol": 1e-15, "gtol": 1e-12})
        h_new = project_feasible(res.x, n)
        c5_new = evaluate_c5(h_new)
        if c5_new < c5:
            h = h_new
            c5 = c5_new
    return h, c5


def coord_1d_objective(t, h_save, idx, n, active):
    adj = (t - h_save[idx]) / (n - 1)
    h_new = h_save - adj
    h_new[idx] = t
    if h_new.min() < -1e-8 or h_new.max() > 1.0 + 1e-8:
        return 1e10
    corr = np.correlate(h_new, 1.0 - h_new, mode="full")
    return float(np.max(corr[active]))


def coord_descent(h, n, max_sweeps, t0, time_limit):
    h = h.copy()

            c5 = evaluate_c5(h)

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -3.0, 3.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(h + mid, 0.0, 1.0)) > target:
            hi = mid
        else:
            lo = mid
    r = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(r)
    if abs(diff) > 1e-14:
        order = np.argsort(np.abs(r - 0.5))
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - r[idx])
                r[idx] += a
                diff -= a
            else:
                a = min(-diff, r[idx])
                r[idx] -= a
                diff += a
    return r

import numpy as np
from scipy.optimize import minimize, minimize_scalar

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h, n):
    target = n / 2.0
    lo = -3.0
    hi = 3.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    r = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(r)
    if abs(diff) > 1e-14:
        order = np.argsort(np.abs(r - 0.5))
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                a = min(diff, 1.0 - r[idx])
                r[idx] += a
                diff -= a
            else:
                a = min(-diff, r[idx])
                r[idx] -= a
                diff += a
    return r


def smooth_max_val(vals, tau):
    v = np.asarray(vals, dtype=float).copy()
    m = float(v.max())
    v = (v - m) / tau
    return m + tau * float(np.log(np.sum(np.exp(v))))


def find_active(h, n, frac):
    corr = np.correlate(h, 1.0 - h, mode="full")
    k = max(3, int(len(corr) * frac))
    th = np.partition(corr, len(corr) - k)[len(corr) - k]
    return np.where(corr >= th - 1e-12)[0]


def sm_obj(t, n, tau, active):
    dx = 2.0 / n
    corr = np.correlate(t, 1.0 - t, mode="full")
    return smooth_max_val(corr[active], tau) * dx


def sm_obj_pen(t, n, tau, active, lam):
    dx = 2.0 / n
    corr = np.correlate(t, 1.0 - t, mode="full")
    sm = smooth_max_val(corr[active], tau) * dx
    pen = lam * (np.sum(t) - n / 2.0) ** 2
    return sm + pen


def lbfgsb_opt(h0, n, taus, t0, limit):
    h = project_feasible(np.array(h0, dtype=float), n)
    best = evaluate_c5(h)
    bd = [(0.0, 1.0)] * n
    for i, tau in enumerate(taus):
        if time.time() > limit:
            break
        act = find_active(h, n, 0.3)
        r = minimize(sm_obj, h.copy(), method="L-BFGS-B",
                     args=(n, tau, act), bounds=bd,
                     options={"maxiter": 500, "ftol": 1e-15, "gtol": 1e-12})
        hn = project_feasible(r.x, n)
        cn = evaluate_c5(hn)
        if cn < best:
            h = hn
            best = cn
    return h, best


def c1d_obj(t, hs, j, n, act):
    d = (t - hs[j]) / (n - 1)
    hn = hs - d
    hn[j] = t
    if hn.min() < -1e-8 or hn.max() > 1.0 + 1e-8:
        return 1e10
    c = np.correlate(hn, 1.0 - hn, mode="full")
    return float(c[act].max())


def coord_desc(h0, n, nsweeps, t0, limit):
    h = project_feasible(np.array(h0, dtype=float), n)
    bh = h.copy()
    bc = evaluate_c5(h)
    for s in range(nsweeps):
        if time.time() > limit:
            break
        act = find_active(h, n, 0.3)
        for j in range(n):
            if time.time() > limit:
                break
            sv = h.copy()
            lo = max(0.0, sv[j] - 0.25)
            hi = min(1.0, sv[j] + 0.25)
            if hi - lo < 1e-12:
                continue
            r = minimize_scalar(c1d_obj, bounds=(lo, hi), method="bounded",
                                args=(sv, j, n, act), options={"xatol": 1e-12})
