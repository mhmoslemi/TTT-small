    vmax = np.max(v)
    return tau * (vmax + np.log(np.sum(np.exp(v - vmax))))

def smooth_max_weights(corr, tau):
    v = corr / tau
    vmax = np.max(v)
    w = np.exp(v - vmax)
    return w / np.sum(w)

    g = 1.0 - h

def evaluate_c5(h):

    corr = np.correlate(h, g, mode="full")

def project_feasible(h, n):

    hi = 1.5

        if np.sum(proj) > target:
            hi = mid
        else:
            lo = mid

    diff = target - np.sum(result)
    if abs(diff) > 1e-10:
        order = np.argsort(np.abs(result - 0.5))
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - result[idx])
                result[idx] += add
                diff -= add
            else:
                sub = min(-diff, result[idx])
                result[idx] -= sub
                diff += sub

    vmax = np.max(v)
    return tau * (vmax + np.log(np.sum(np.exp(v - vmax))))

    w = np.exp(v - vmax)
    return w / np.sum(w)

            best_h = h_final.copy()

    if best_h is None:
        n = 59
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c5 = evaluate_c5(best_h)

    return best_h, best_c5, len(best_h)
