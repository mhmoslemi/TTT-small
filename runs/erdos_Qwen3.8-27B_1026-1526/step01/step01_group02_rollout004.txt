def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            # Distribute proportionally to room
            room = np.where(mask, 1.0 - h, 0)
            total_room = room.sum()
            if total_room < 1e-15:
                break
            h += diff * room / total_room
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            avail = np.where(mask, h, 0)
            total_avail = avail.sum()
            if total_avail < 1e-15:
                break
            h += diff * avail / total_avail
    return np.clip(h, 0.0, 1.0)

    """Compute gradient of max_k c[k]*dx using top-k active shifts."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")  # length 2n-1
    # Find top_k indices
    top_idx = np.argsort(c)[::-1][:top_k]

    top_vals = c[top_idx]
    weights = np.exp((top_vals - top_vals.max()) / dx)  # softmax

    grad = np.zeros(n)
    for m in range(len(c)):
        if weights[m] < 1e-14:
            continue
        shift = m - (n - 1)
        # (1-h[j-shift]) for valid j

        j_hi = min(n - 1, n - 1 + shift)
        if j_lo <= j_hi:
            idx = np.arange(j_lo, j_hi + 1)
            grad[idx] += weights[m] * (1.0 - h[idx - shift])
        # -h[j+shift] for valid j
        j_lo2 = max(0, -shift)
        j_hi2 = min(n - 1, n - 1 - shift)
        if j_lo2 <= j_hi2:
            idx2 = np.arange(j_lo2, j_hi2 + 1)
            grad[idx2] -= weights[m] * h[idx2 + shift]
    return grad * dx

    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            room = np.clip(room, 0, None)
            total = room.sum()
            if total < 1e-15:
                break
            h += diff * room / total
        else:
            avail = np.clip(h, 0, None)
            total = avail.sum()
            if total < 1e-15:
                break
            h += diff * avail / total
    return np.clip(h, 0.0, 1.0)

def compute_obj_grad(h, n, beta=100.0):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    max_c = c.max()
    log_w = beta * (c - max_c)
    log_w = log_w - log_w.max()
    w = np.exp(log_w)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
    else:
        w /= ws
    obj = (1.0 / beta) * (np.log(ws) + beta * max_c)  # smooth max value
    grad = np.zeros(n)
    for m in range(len(c)):
        if w[m] < 1e-13:
            continue
        s = m - (n - 1)
        lo1 = max(0, s)
        hi1 = min(n - 1, n - 1 + s)
        if lo1 <= hi1:

            grad[idx] += w[m] * (1.0 - h[idx - s])
        lo2 = max(0, -s)

        if lo2 <= hi2:
            idx2 = np.arange(lo2, hi2 + 1)
            grad[idx2] -= w[m] * h[idx2 + s]
    grad *= dx
    return obj, grad

        if obj_new < obj:

        else:
            alpha *= 0.5

            break
    return best_h, best_c5

def slsqp_polish(h, n, maxiter=2000):
    c5_init = compute_c5_from_h(h)
    x0 = np.concatenate([h, [c5_init + 0.02]])
    bounds = [(0.0, 1.0)] * n + [(max(0.001, c5_init * 0.8), 2.0)]
    
    def obj_slqp(x):
        return x[-1]
    
    def ineq_slqp(x):
        hvar = x[:-1]
        tvar = x[-1]
        nvar = len(hvar)
        dx = 2.0 / nvar
        cc = np.correlate(hvar, 1.0 - hvar, mode="full")
        return tvar - cc * dx
    # This is scalar, not vector. SLSQP needs vector inequality.
    # Actually for SLSQP, ineq should return array where each element >= 0
    # But we only have one constraint per k... that's 2n-1 constraints.
    
    # Better: just use the max formulation
    # t >= c[k]*dx for all k => 2n-1 inequality constraints
    # Too many for SLSQP. Let's use a single smooth constraint.
    
    # Alternative: use penalty method
    # Or just use the objective directly with a soft max
    
    # Actually, let's just use the previous approach: minimize t subject to t >= max(c)*dx
    # Which is: t - max(c)*dx >= 0 (single inequality, but non-smooth)
    # SLSQP can handle this numerically.
    
    constraints = [{'type': 'ineq', 'fun': ineq_slqp}]
    result = minimize(obj_slqp, x0, method='SLSQP', bounds=bounds,
                      constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
    if result.success or result.x is not None:
        h_out = project_feasible(result.x[:-1], n)
        return h_out
    return h

    n_old = len(h_old)

    h_new = np.clip(f(x_new), 0, 1)
    return project_feasible(h_new, n_new)

            break
    return best_h, best_c5


def slsqp_polish(h_init, n, maxiter=2000):
    c5_init = compute_c5_from_h(h_init)
    x0 = np.concatenate([h_init, [c5_init + 0.03]])
    bounds = [(0.0, 1.0)] * n + [(max(0.001, c5_init * 0.7), 2.0)]

    def obj_slqp(x):
        return x[-1]

    def ineq_slqp(x):
        hv = x[:-1]
        tv = x[-1]
        nv = len(hv)
        dxx = 2.0 / nv
        cc = np.correlate(hv, 1.0 - hv, mode="full")
        return tv - cc * dxx

    constraints = [{'type': 'ineq', 'fun': ineq_slqp}]
    result = minimize(obj_slqp, x0, method='SLSQP', bounds=bounds,
                      constraints=constraints, options={'maxiter': maxiter, 'ftol': 1e-14})
    if result.x is not None:
        return project_feasible(result.x[:-1], n)
    return h_init


def interp_h(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0 / n_new
    f = interp1d(x_old, h_old, kind='linear', fill_value='extrapolate', bounds_error=False)
    h_new = np.clip(f(x_new), 0, 1)
    return project_feasible(h_new, n_new)


def make_structure_h(n, kind, params, rng):
    x = np.linspace(0, 2, n, endpoint=False) + 1.0 / n
    if kind == 'sin':
        freq, amp = params
        h = 0.5 + amp * np.sin(2.0 * np.pi * freq * x / 2.0)
    elif kind == 'bump':
        center, sigma, amp = params
        h = 0.5 + amp * np.exp(-0.5 * ((x - center) / sigma) ** 2)
    elif kind == 'triangle':
        center, amp = params
        h = 0.5 + amp * np.maximum(0, 1 - np.abs(x - center) / 0.5)
    else:
        h = rng.random(n) + 0.5
    return project_feasible(h, n)


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    n_init = 135
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h0)
    else:
        h0 = np.full(n_init, 0.5)

    h0 = project_feasible(h0, n_init)
    best_h = h0.copy()
    best_c5 = compute_c5_from_h(best_h)
    best_n = n_init

    # Phase 1: Gradient descent at initial resolution
    remaining = budget_s - (time.time() - t0)
    if remaining > 20:
        tl = min(100, remaining * 0.15)
        h_gd, c5_gd = grad_descent_smooth(best_h, best_n, t0, time_limit=tl)
        if c5_gd < best_c5:
            best_h, best_c5, best_n = h_gd, c5_gd, best_n

    # Phase 2: Multi-resolution optimization
    resolutions = [54, 72, 90, 120, 135, 180, 240]
    for n_try in resolutions:
        if time.time() - t0 > budget_s - 200:
            break
        if n_try == best_n:
            continue
        h_try = interp_h(best_h, n_try)
        c5_try = compute_c5_from_h(h_try)
        remaining = budget_s - (time.time() - t0)
        tl = min(80, remaining * 0.12)
        if tl < 15:
            continue
        h_opt, c5_opt = grad_descent_smooth(h_try, n_try, t0, time_limit=tl)
        if c5_opt < best_c5 - 1e-10:
            best_h, best_c5, best_n = h_opt, c5_opt, n_try

    # Phase 3: Structured restarts at best resolution
    n_restarts = max(3, int((budget_s - (time.time() - t0)) / 35))
    n_restarts = min(n_restarts, 20)
    for ri in range(n_restarts):
        if time.time() - t0 > budget_s - 120:
            break
        kind = ['sin', 'bump', 'random', 'tri'][ri % 4]
        if kind == 'sin':
            params = (rng.integers(1, 5), 0.15 + 0.2 * rng.random())
        elif kind == 'bump':
            params = (rng.uniform(0.5, 1.5), 0.1 + 0.2 * rng.random(), 0.2 + 0.3 * rng.random())
        elif kind == 'tri':
            params = (rng.uniform(0.5, 1.5), 0.25 + 0.25 * rng.random())
        else:
            params = None
        h_struct = make_structure_h(best_n, kind, params, rng)
        # Blend with best
        blend = 0.3 + 0.4 * (ri / max(1, n_restarts))
        h_mix = project_feasible((1 - blend) * best_h + blend * h_struct, best_n)
        # Also add small random noise
        h_mix = project_feasible(h_mix + 0.03 * rng.standard_normal(best_n), best_n)
        tl = min(35, (budget_s - (time.time() - t0)) * 0.08)
        if tl < 8:
            continue
        h_opt, c5_opt = grad_descent_smooth(h_mix, best_n, t0, time_limit=tl, max_iters=800)
        if c5_opt < best_c5 - 1e-10:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 4: Random perturbation restarts
    n_rand = max(2, int((budget_s - (time.time() - t0)) / 30))
    n_rand = min(n_rand, 15)
    for ri in range(n_rand):
        if time.time() - t0 > budget_s - 80:
            break
        scale = 0.03 + 0.07 * rng.random()
        h_pert = project_feasible(best_h + scale * rng.standard_normal(best_n), best_n)
        tl = min(30, (budget_s - (time.time() - t0)) * 0.1)
        if tl < 8:
            continue
        h_opt, c5_opt = grad_descent_smooth(h_pert, best_n, t0, time_limit=tl, max_iters=600)
        if c5_opt < best_c5 - 1e-10:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 5: SLSQP polish at best resolution
    if time.time() - t0 < budget_s - 50:
        h_pol = slsqp_polish(best_h, best_n, maxiter=3000)
        c5_pol = compute_c5_from_h(h_pol)
        if c5_pol < best_c5:
            best_h, best_c5 = h_pol, c5_pol

    # Phase 6: SLSQP polish at nearby resolutions
    for n_try in [best_n // 2, best_n + best_n // 3, max(30, best_n - best_n // 4)]:
        if time.time() - t0 > budget_s - 40:
            break
        if n_try < 30 or n_try > 400 or n_try == best_n:
            continue
        h_try = interp_h(best_h, n_try)
        h_pol = slsqp_polish(h_try, n_try, maxiter=1500)
        c5_pol = compute_c5_from_h(h_pol)
        if c5_pol < best_c5:
            best_h, best_c5, best_n = h_pol, c5_pol, n_try

    # Phase 7: Final fine-tune gradient descent
    if time.time() - t0 < budget_s - 15:
        remaining = budget_s - (time.time() - t0) - 10
        if remaining > 10:
            h_gd, c5_gd = grad_descent_smooth(best_h, best_n, t0, time_limit=remaining * 0.6, max_iters=3000)
            if c5_gd < best_c5:
                best_h, best_c5 = h_gd, c5_gd

    # Final SLSQP if time allows
    if time.time() - t0 < budget_s - 10:
        h_pol = slsqp_polish(best_h, best_n, maxiter=1000)
        c5_pol = compute_c5_from_h(h_pol)
        if c5_pol < best_c5:
            best_h, best_c5 = h_pol, c5_pol

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5_from_h(best_h)
    return best_h, best_c5, len(best_h)
