from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def obj_func(args, n, beta):
    h = np.clip(args, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, beta) * (2.0 / n)

def sum_eq(args, n):
    return np.sum(args) - n / 2.0

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:

            if cnt == 0:
                break

        else:

            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt

    # Final fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            idx = int(np.argmin(h))
            h[idx] = min(h[idx] + diff, 1.0)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(h[idx] + diff, 0.0)
    return h

def slsqp_optimize(h0, n, max_iter, beta, ftol=1e-14):
    h0 = project_feasible(h0, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq, 'args': (n,)}]
    try:
        res = minimize(obj_func, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, args=(n, beta),
                       options={'maxiter': max_iter, 'ftol': ftol, 'iprint': 0})
        h_res = project_feasible(res.x, n)
        c5 = compute_c5(h_res)
        return h_res, c5
    except Exception:

def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False) + 1.0/n_old/2.0
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False) + 1.0/n_new/2.0
    h_interpolated = np.interp(x_new, x_old, h_old)

def downsample_h(h, n_old, n_new):
    """Downsample by averaging blocks"""
    if n_old % n_new != 0:
        # Resample to a common multiple then downsample
        ratio = n_old / n_new
        x_old = np.linspace(0.0, 2.0, n_old, endpoint=False) + 1.0/n_old/2.0
        x_new = np.linspace(0.0, 2.0, n_new, endpoint=False) + 1.0/n_new/2.0
        h_interp = np.interp(x_new, x_old, h)
        return project_feasible(h_interp, n_new)
    blocks = h.reshape(n_new, n_old // n_new)
    h_new = blocks.mean(axis=1)
    return project_feasible(h_new, n_new)

    h = project_feasible(h0.copy(), n)
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)

import numpy as np

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def obj_func(x, n, beta):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, beta) * (2.0 / n)

def sum_eq(x, n):
    return np.sum(x) - n / 2.0

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-12
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            idx = int(np.argmin(h))
            h[idx] = min(h[idx] + diff, 1.0)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(h[idx] + diff, 0.0)
    return h

def slsqp_optimize(h0, n, max_iter, beta, ftol=1e-14):
    h0 = project_feasible(h0, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq, 'args': (n,)}]
    try:
        res = minimize(obj_func, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, args=(n, beta),
                       options={'maxiter': max_iter, 'ftol': ftol, 'iprint': 0})
        h_res = project_feasible(res.x, n)
        c5 = compute_c5(h_res)
        return h_res, c5
    except Exception:
        return None, None

def interpolate_to_n(h, n_old, n_new):
    x_old = np.arange(n_old, dtype=float)
    x_new = np.linspace(0.0, n_old - 1.0, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def subgradient_method(h0, n, n_iters, alpha=None, momentum=0.9):
    h = project_feasible(h0.astype(float).copy(), n)
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    
    if alpha is None:
        alpha = 0.5 / np.sqrt(n)
    
    for it in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_star = int(np.argmax(corr))
        k = k_star - (n - 1)
        
        g = np.zeros(n)
        if k >= 0:
            g[k:] += (1.0 - h[:n-k])
            g[:n-k] -= h[k:]
        else:
            g[:n+k] += (1.0 - h[-k:])
            g[-k:] -= h[:n+k]
        
        a = alpha / np.sqrt(1.0 + 0.1 * it)
        h_y = h - momentum * v
        h_new = project_feasible(h_y - a * g, n)
        v = h_new - h
        h = h_new
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

            mask = h < 1.0 - 1e-12
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-12
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            idx = int(np.argmin(h))
            h[idx] = min(h[idx] + diff, 1.0)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(h[idx] + diff, 0.0)
    return h


def slsqp_full(h0, n, max_iter, beta, ftol=1e-14):
    h0 = project_feasible(h0, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq, 'args': (n,)}]
    try:
        res = minimize(obj_func_full, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, args=(n, beta),
                       options={'maxiter': max_iter, 'ftol': ftol, 'iprint': 0})
        h_res = project_feasible(res.x, n)
        c5 = compute_c5(h_res)
        return h_res, c5
    except Exception:
        return None, None


def slsqp_topk(h0, n, max_iter, beta, k_top, ftol=1e-14):
    h0 = project_feasible(h0, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq, 'args': (n,)}]
    try:
        res = minimize(obj_func_topk, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, args=(n, beta, k_top),
                       options={'maxiter': max_iter, 'ftol': ftol, 'iprint': 0})
        h_res = project_feasible(res.x, n)
        c5 = compute_c5(h_res)
        return h_res, c5
    except Exception:
        return None, None


def interpolate_to_n(h, n_old, n_new):
    x_old = np.arange(n_old, dtype=float)
    x_new = np.linspace(0.0, n_old - 1.0, n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)


def subgradient_method(h0, n, n_iters, alpha=None, momentum=0.9):
    h = project_feasible(h0.astype(float).copy(), n)
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    if alpha is None:
        alpha = 0.5 / np.sqrt(n)

    for it in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_star = int(np.argmax(corr))
        k = k_star - (n - 1)

        g = np.zeros(n)
        if k >= 0:
            g[k:] += (1.0 - h[:n - k])
            g[:n - k] -= h[k:]
        else:
            g[:n + k] += (1.0 - h[-k:])
            g[-k:] -= h[:n + k]

        a = alpha / np.sqrt(1.0 + 0.1 * it)
        h_y = h - momentum * v
        h_new = project_feasible(h_y - a * g, n)
        v = h_new - h
        h = h_new

        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    best_h = None
    best_c5 = 1e9
    best_n = 190

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
        h_init = project_feasible(h_init, n_init)
        best_h = h_init
        best_c5 = compute_c5(h_init)
        best_n = n_init

    if best_h is None:
        best_h = np.full(190, 0.5)
        best_c5 = compute_c5(best_h)
        best_n = 190

    # Phase 1: Initial SLSQP optimization
    if elapsed() < budget_s - 250:
        h_opt, c5_opt = slsqp_full(best_h, best_n, 300, 50.0)
        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 2: Coarse-to-fine at multiple scales
    if elapsed() < budget_s - 400:
        for n_coarse in [47, 71, 94]:
            if elapsed() > budget_s - 300:
                break
            h_c = interpolate_to_n(best_h, best_n, n_coarse)
            h_opt, c5_opt = slsqp_full(h_c, n_coarse, 200, 40.0)
            if h_opt is not None:
                h_up = interpolate_to_n(h_opt, n_coarse, best_n)
                h_opt2, c5_opt2 = slsqp_full(h_up, best_n, 250, 60.0)
                if h_opt2 is not None and c5_opt2 < best_c5:
                    best_h, best_c5 = h_opt2, c5_opt2

    # Phase 3: Main exploration loop with top-K and subgradient
    no_improve = 0
    max_no_improve = 30
    while elapsed() < budget_s - 280:
        strategy = rng.random()

        if strategy < 0.30:
            # Top-K optimization with small K
            k_top = int(rng.choice([3, 5, 7]))
            h_opt, c5_opt = slsqp_topk(best_h, best_n, 200, 100.0, k_top)
        elif strategy < 0.55:
            # Full SLSQP from perturbed start
            sigma = rng.uniform(0.08, 0.25)
            h_start = project_feasible(best_h + rng.normal(0, sigma, best_n), best_n)
            h_opt, c5_opt = slsqp_full(h_start, best_n, 200, 80.0)
        elif strategy < 0.75:
            # Subgradient exploration then SLSQP
            n_sub = int(rng.uniform(80, 200))
            h_sub, _ = subgradient_method(best_h, best_n, n_sub, momentum=rng.uniform(0.7, 0.95))
            h_opt, c5_opt = slsqp_full(h_sub, best_n, 200, 80.0)
        elif strategy < 0.90:
            # Larger perturbation restart
            sigma = rng.uniform(0.3, 0.5)
            h_start = project_feasible(best_h + rng.normal(0, sigma, best_n), best_n)
            h_opt, c5_opt = slsqp_full(h_start, best_n, 250, 60.0)
        else:
            # Coarse restart
            n_c = 47
            h_c = interpolate_to_n(best_h, best_n, n_c)
            h_c += rng.normal(0, 0.1, n_c)
            h_c = project_feasible(h_c, n_c)
            h_opt_c, _ = slsqp_full(h_c, n_c, 150, 40.0)
            if h_opt_c is not None:
                h_up = interpolate_to_n(h_opt_c, n_c, best_n)
                h_opt, c5_opt = slsqp_full(h_up, best_n, 250, 80.0)
            else:
                h_opt, c5_opt = None, None

        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= max_no_improve:
            break

    # Phase 4: Try higher resolution
    for n_try in [238, 286]:
        if elapsed() > budget_s - 150:
            break
        h_interp = interpolate_to_n(best_h, best_n, n_try)
        h_opt, c5_opt = slsqp_full(h_interp, n_try, 250, 50.0)
        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, n_try

    # Phase 5: Final polish with increasing beta
    for beta in [100, 300, 1000]:
        if elapsed() > budget_s - 25:
            break
        h_opt, c5_opt = slsqp_full(best_h, best_n, 400, beta, ftol=1e-15)
        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 6: Subgradient fine-tuning with small steps
    if elapsed() < budget_s - 10:
        h_sub, c5_sub = subgradient_method(best_h, best_n, 150, alpha=0.005, momentum=0.95)
        if c5_sub < best_c5:
            best_h, best_c5 = h_sub, c5_sub

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, len(best_h))
