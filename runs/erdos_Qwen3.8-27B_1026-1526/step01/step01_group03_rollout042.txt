import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return float(np.max(corr) * dx)

def get_correlations(h):
    n = len(h)
    g = 1.0 - h
    return np.correlate(h, g, mode="full")

def smooth_max_value(corr, tau):
    v = corr / tau
    vmax = np.max(v)
    return tau * (vmax + np.log(np.sum(np.exp(v - vmax))))

def smooth_max_weights(corr, tau):
    v = corr / tau
    vmax = np.max(v)
    w = np.exp(v - vmax)
    w /= np.sum(w)
    return w

def grad_smooth_corr(h, weights):
    """Gradient of smooth_max(corr(h, 1-h)) with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    for j in range(n):
        val = 0.0
        for m in range(2 * n - 1):
            k = m - (n - 1)
            wm = weights[m]
            if wm < 1e-12:
                continue
            if 0 <= j + k < n:
                val += wm * (1.0 - h[j + k])
            if 0 <= j - k < n:
                val -= wm * h[j - k]
        grad[j] = val
    return grad

def project_feasible(h, n):
    target = n / 2.0
    lo = -5.0
    hi = 5.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-14:
        idx = np.argsort(np.abs(result - 0.5))[:int(np.ceil(abs(diff)))]
        for i in idx:
            adj = np.clip(diff, -(result[i]), 1.0 - result[i])
            result[i] += adj
            diff -= adj
            if abs(diff) < 1e-14:
                break
    return result

def projected_gradient_step(h, grad, lr, n):
    h_new = h - lr * grad
    return project_feasible(h_new, n)

def backtracking_line_search(h, grad, n, tau, c0, lr_init, cmax=200):
    """Simple backtracking line search."""
    lr = lr_init
    for _ in range(30):
        h_new = projected_gradient_step(h, grad, lr, n)
        corr_new = get_correlations(h_new)
        val_new = smooth_max_value(corr_new, tau)
        if val_new < c0 - 1e-10 * lr * np.max(np.abs(grad)):
            return h_new, val_new, lr
        lr *= 0.5
    return h_new, val_new, lr

    h = project_feasible(h0.copy(), n)
    best_h = h.copy()
    best_val = float('inf')
    lr = lr_init
    
    for i, tau in enumerate(tau_schedule):
        if time_limit is not None and time.time() - start_time > time_limit:
            break
        
        corr = get_correlations(h)
        current_val = smooth_max_value(corr, tau)
        if current_val < best_val:
            best_val = current_val
            best_h = h.copy()
        
        for it in range(max_iters):
            if time_limit is not None and time.time() - start_time > time_limit:
                break
            
            corr = get_correlations(h)
            weights = smooth_max_weights(corr, tau)
            grad = grad_smooth_corr(h, weights)

                break
            
            h_new, val_new, lr = backtracking_line_search(h, grad, n, tau, smooth_max_value(corr, tau), lr)
            
            # Track best
            actual_c5 = evaluate_c5(h_new)
            if actual_c5 < best_val:
                best_val = actual_c5
                best_h = h_new.copy()
            
            h = h_new

        if i < len(tau_schedule) - 1:
            perturbation = rng.randn(n) * 0.02
            h = project_feasible(h + perturbation, n)
    
    return best_h, best_val

For the full mode, the valid range of `i + m - (n-1)` is [0, n-1], and `i` is [0, n-1].

`d/dh[j] sum_{i∈S} h[i](1-h[i+k])`
= sum_{i∈S} [δ_{ij}(1-h[i+k]) + h[i]·(-δ_{i+k,j})]

j-k∈S means 0≤j-k<n AND 0≤j-k+k<n = 0≤j<n. Since 0≤j<n is always true, this is just 0≤j-k<n.

def _get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def _smax(corr, tau):
    v = corr / tau
    m = np.max(v)
    return tau * (m + np.log(np.sum(np.exp(v - m))))

def _smax_w(corr, tau):
    v = corr / tau
    m = np.max(v)
    w = np.exp(v - m)
    s = np.sum(w)
    if s < 1e-300:
        return np.full(len(corr), 1.0 / len(corr))
    return w / s

def _grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = _get_corr(h)
    w = _smax_w(corr, tau)
    grad = np.zeros(n)
    for j in range(n):
        s = 0.0
        for m in range(2 * n - 1):
            k = m - (n - 1)
            wm = w[m]
            if wm < 1e-14:
