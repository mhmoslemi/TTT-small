            j2_lo = max(0, lag)
            j2_hi = min(n - 1, n - 1 + lag)
            if j2_lo <= j2_hi:
                idx2 = np.arange(j2_lo, j2_hi + 1)
                grad[idx2] -= w * h[idx2 - lag]
        grad *= (2.0 / n)
        return grad

    def _sum_fun(x):
        return np.sum(x) - n / 2.0

    cons = [{'type': 'eq', 'fun': _sum_fun}]
    try:
        res = minimize(_obj, h0, method='SLSQP', jac=_grad,
                       bounds=[(0.0, 1.0)] * n, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-14, 'iprint': 0})
        h_res = project_feasible(res.x, n)
        c5 = compute_c5(h_res)
        return h_res, c5
    except Exception:
        return None, None


def make_perturbed(h, sigma, rng):
    h_p = h + rng.normal(0.0, sigma, len(h))
    return project_feasible(h_p, len(h))


def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    return np.interp(x_new, x_old, h_old)


def make_periodic_variant(h, n, rng, strength=0.15):
    i = np.arange(n, dtype=float)
    for freq in [2.0, 3.0, 5.0, 7.0]:
        if rng.random() < 0.3:
            phase = rng.uniform(0, 2 * np.pi)
            h = h + strength * np.sin(2 * np.pi * freq * i / n + phase)
    return project_feasible(h, n)


def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    best_h = None
    best_c5 = 1e9
    best_n = 190

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
        h_init = project_feasible(h_init, n_init)
        best_h = h_init
        best_c5 = compute_c5(h_init)
        best_n = n_init

    if best_h is None:
        best_h = np.ones(190) * 0.5
        best_c5 = compute_c5(best_h)
        best_n = 190

    def elapsed():
        return time.time() - t0

    # Phase 1: L-BFGS-B with penalty, increasing beta
    for beta in [40, 100, 300, 800]:
        if elapsed() > budget_s - 400:
            break
        h_opt, c5_opt = optimize_lbfgs(best_h, best_n, beta, lam=500.0, max_iter=800, ftol=1e-14)
        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 2: SLSQP with analytical gradient and exact constraint
    for beta in [200, 500]:
        if elapsed() > budget_s - 350:
            break
        h_opt, c5_opt = optimize_slsq_gradient(best_h, best_n, beta, max_iter=300)
        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 3: Random restarts with L-BFGS-B (main exploration)
    no_improve = 0
    while elapsed() < budget_s - 300:
        sigma = 0.05 + 0.07 * (0.9 ** min(no_improve, 30))
        r = rng.random()
        if r < 0.08:
            sigma = 0.35
        elif r < 0.20:
            h_start = make_periodic_variant(best_h, best_n, rng, strength=0.1)
        else:
            h_start = make_perturbed(best_h, sigma, rng)
        beta = 150 if no_improve < 15 else 400
        h_opt, c5_opt = optimize_lbfgs(h_start, best_n, beta, lam=500.0, max_iter=600, ftol=1e-14)
        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 60:
            break

    # Phase 4: Multi-resolution refinement
    for n_new in [380, 570, 760]:
        if elapsed() > budget_s - 200:
            break
        h_interp = interpolate_h(best_h, best_n, n_new)
        h_interp = project_feasible(h_interp, n_new)
        for beta in [80, 300]:
            if elapsed() > budget_s - 120:
                break
            h_opt, c5_opt = optimize_lbfgs(h_interp, n_new, beta, lam=500.0, max_iter=500, ftol=1e-14)
            if h_opt is not None and c5_opt < best_c5:
                best_h, best_c5, best_n = h_opt, c5_opt, n_new

    # Phase 5: Final polish with very high beta (L-BFGS-B + SLSQP)
    for beta in [1000, 3000]:
        if elapsed() > budget_s - 40:
            break
        h_opt, c5_opt = optimize_lbfgs(best_h, best_n, beta, lam=1000.0, max_iter=1200, ftol=1e-15)
        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    if elapsed() < budget_s - 25:
        h_opt, c5_opt = optimize_slsq_gradient(best_h, best_n, 800, max_iter=200)
        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, len(best_h))
