            if time.time() - t_start > time_limit:
                break
        
        # Compute current autocorrelation
        auto = np.correlate(h, h, mode='full')
        
        # Determine active lags (those contributing to the max)
        costs = target_sum - auto  # n/2 - A_h(m)
        max_cost = np.max(costs)
        # Include lags within a fraction of the max
        active = np.where(costs >= max_cost * 0.5)[0]
        # Always include the top few
        top_k = np.argsort(costs)[-10:]
        active = np.unique(np.concatenate([active, top_k]))
        
        # Cyclic block updates
        for start in range(0, n):
            end = min(start + B, n)
            if end - start < 2:
                break
            actual_B = end - start
            h_slice = slice(start, end)
            h_block = h[h_slice].copy()
            S_block = np.sum(h_block)
            
            # Build LP
            A_rows = []
            b_rows = []
            
            for m_idx in active:
                m = m_idx - (n - 1)
                coeffs = np.zeros(actual_B)
                for bj in range(actual_B):
                    j = start + bj
                    c = 0.0
                    if 0 <= j + m < n:
                        c += h[j + m]
                    if 0 <= j - m < n:
                        c += h[j - m]
                    coeffs[bj] = c
                d_m = target_sum - auto[m_idx] + np.dot(coeffs, h_block)
                row = np.zeros(actual_B + 1)
                row[0] = -1.0
                row[1:] = -coeffs
                A_rows.append(row)
                b_rows.append(-d_m)
            
            A_ub = np.array(A_rows)
            b_ub = np.array(b_rows)
            c_obj = np.zeros(actual_B + 1)
            c_obj[0] = 1.0
            A_eq = np.zeros((1, actual_B + 1))
            A_eq[0, 1:] = 1.0
            b_eq = np.array([S_block])
            bounds = [(0, None)] + [(0, 1)] * actual_B
            
            try:
                res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                              bounds=bounds, method='highs')
                if res.success:
                    h[h_slice] = res.x[1:]
            except:
                pass
        
        # Evaluate
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h
