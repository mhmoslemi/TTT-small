import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-9) & (h < 1.0 - 1e-9)
        if np.any(mask):
            h[mask] += diff / mask.sum()
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_value(corr, alpha):
    m = np.max(corr)
    exps = np.exp(alpha * (corr - m))
    return m + np.log(exps.sum()) / alpha

def gradient_sm(corr, h, alpha):
    n = len(h)
    m = np.max(corr)
    exps = np.exp(alpha * (corr - m))
    w = exps / exps.sum()
    grad = np.zeros(n)
    for k in range(len(corr)):
        if w[k] < 1e-12:
            continue
        s = k - n + 1
        wk = w[k]
        # term 1: (1-h[t+s]) for valid t
        lo = max(0, -s)
        hi = min(n, n - s)
        if lo < hi:
            grad[lo:hi] += wk * (1.0 - h[lo+s:hi+s])
        # term 2: -h[t-s] for valid t  
        lo2 = max(0, s)
        hi2 = min(n, n + s)
        if lo2 < hi2:
            grad[lo2:hi2] -= wk * h[lo2-s:hi2-s]
    return grad

def projected_grad_descent(h0, n, target_sum, alpha, max_iters, lr_init=0.01):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode="full")
        grad = gradient_sm(corr, h, alpha)
        
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Backtracking line search
        lr = lr_init
        c5_cur = compute_c5(h)
        for ls in range(20):
            h_new = h - lr * grad
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = project_to_sum(h_new, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-12:
                h = h_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                break
            lr *= 0.5
        else:
            # No improvement, try smaller step
            lr = lr_init * (0.95 ** it)
            if lr < 1e-8:
                break
    
    return best_h, best_c5

def coordinate_descent(h0, n, target_sum, rng, max_rounds, step_sizes=None):
    if step_sizes is None:
        step_sizes = [0.05, 0.03, 0.02, 0.01, 0.005, 0.002, 0.001]
    h = h0.copy()
    c5_cur = compute_c5(h)
    
    for rnd in range(max_rounds):
        improved = False
        order = rng.permutation(n)
        for i in order:
            # Try paired move: increase h[i], decrease h[j] (or vice versa)
            best_val = c5_cur
            best_delta = None
            j = (i + 1) % n
            for step in step_sizes:
                for sign in [1, -1]:
                    h_new = h.copy()
                    h_new[i] = np.clip(h_new[i] + sign * step, 0.0, 1.0)
                    h_new[j] = np.clip(h_new[j] - sign * step, 0.0, 1.0)
                    h_new = project_to_sum(h_new, target_sum)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_val - 1e-12:
                        best_val = c5_new
                        best_delta = (step, sign)
            
            if best_delta is not None:
                step, sign = best_delta
                h[i] = np.clip(h[i] + sign * step, 0.0, 1.0)
                h[j] = np.clip(h[j] - sign * step, 0.0, 1.0)
                h = project_to_sum(h, target_sum)
                c5_cur = best_val
                improved = True
        
        if not improved:
            break
    
    return h, c5_cur

def simulated_annealing(h0, n, target_sum, rng, max_iters, T0=0.05, T_end=0.001):
    h = h0.copy()
    c5_cur = compute_c5(h)
    best_c5 = c5_cur
    best_h = h.copy()
    T = T0
    cool = (T_end / T0) ** (1.0 / max(1, max_iters))
    
    for it in range(max_iters):
        # Random block perturbation
        block_size = rng.choice([1, 2, 3, 5, 8], p=[0.3, 0.25, 0.2, 0.15, 0.1])
        start = rng.randint(0, n - block_size + 1)
        h_new = h.copy()
        h_new[start:start+block_size] += rng.uniform(-0.05, 0.05, block_size)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        
        c5_new = compute_c5(h_new)
        delta = c5_new - c5_cur
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            h = h_new
            c5_cur = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        T = max(T * cool, T_end)
    
    return best_h, best_c5

def subgradient_step(h, n, target_sum, alpha, lr):
    corr = np.correlate(h, 1.0 - h, mode="full")
    grad = gradient_sm(corr, h, alpha)
    h_new = h - lr * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_to_sum(h_new, target_sum)
    return h_new

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.any():
            h[mask] += diff / mask.sum()
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_and_weights(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    exps = np.exp(alpha * (corr - m))
    w = exps / exps.sum()
    return corr, w

def gradient_sm(corr, w, h):
    n = len(h)
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-10:
            continue
        s = k - n + 1
        lo = max(0, -s)
        hi = min(n, n - s)
        if lo < hi:
            grad[lo:hi] += wk * (1.0 - h[lo+s:hi+s])
        lo2 = max(0, s)
        hi2 = min(n, n + s)
        if lo2 < hi2:
            grad[lo2:hi2] -= wk * h[lo2-s:hi2-s]
    return grad

def grad_descent(h0, n, ts, alpha, max_iters, lr=0.02):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        corr, w = corr_and_weights(h, alpha)
        grad = gradient_sm(corr, w, h)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        step_lr = lr / max(1.0, gn * 5.0)
        for ls in range(30):
            h_new = h - step_lr * grad
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = project_to_sum(h_new, ts)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 or c5_new < compute_c5(h) - 1e-12:
                h = h_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                break
            step_lr *= 0.5
        # adapt lr
        if (it + 1) % 100 == 0:
            lr *= 0.9
    return best_h, best_c5

def coord_descent(h0, n, ts, rng, rounds, steps):
    h = h0.copy()
    c5_cur = compute_c5(h)
    for r in range(rounds):
        any_imp = False
        order = rng.permutation(n)
        for i in order:
            # paired move with j
            for j in [(i+1)%n, (i-1+n)%n, (i+n//3)%n, (i+n//2)%n]:
                for st in steps:
                    for sg in [1, -1]:
                        h2 = h.copy()
                        h2[i] = np.clip(h2[i] + sg*st, 0.0, 1.0)
                        h2[j] = np.clip(h2[j] - sg*st, 0.0, 1.0)
                        h2 = project_to_sum(h2, ts)
                        c52 = compute_c5(h2)

                            any_imp = True
                            break
                    if c52 < c5_cur - 1e-12:  # this check is redundant but safe
                        break
        if not any_imp:
            break
    return h, c5_cur

def sim_anneal(h0, n, ts, rng, iters, T0=0.04, T1=0.0005, block_sizes=(1,2,3,5,8)):
