    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            step = min(diff / np.sum(mask), (1.0 - h[mask]).min())
            h[mask] += step
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            step = min(-diff / np.sum(mask), h[mask].min())
            h[mask] -= step
    return np.clip(h, 0.0, 1.0)

def lse_smooth(c_vals, beta):
    c_max = np.max(c_vals)
    if beta > 50:
        shifted = beta * (c_vals - c_max)
        # Use log-sum-exp trick
        m = np.max(shifted)
        if m > 50:

        else:

    else:
        shifted = beta * (c_vals - c_max)
        log_sum = np.log(np.mean(np.exp(shifted)))
        return c_max + log_sum / beta

                n_best = n_try

    if time.time() - t0 < budget_s - 40:
        for beta_f in [500, 1000, 2000, 5000]:
            if time.time() - t0 > budget_s - 10:
                break
            h_new, c5_new = slsqp_smooth(best_h, len(best_h), beta_f, maxiter=2000)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    if time.time() - t0 < budget_s - 15:
        for _ in range(12):
            if time.time() - t0 > budget_s - 4:
                break
            h_pert = best_h + 0.005 * rng.standard_normal(len(best_h))
            h_pert = project_feasible(h_pert, len(best_h))
            h_new, c5_new = slsqp_smooth(h_pert, len(best_h), 2000, maxiter=600)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
