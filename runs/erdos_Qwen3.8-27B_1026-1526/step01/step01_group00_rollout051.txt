def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx), corr

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    if abs(np.sum(h) - target_sum) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h_out = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - np.sum(h_out)
    if abs(diff) > 1e-10:
        interior = (h_out > 1e-12) & (h_out < 1.0 - 1e-12)
        if np.sum(interior) > 0:
            h_out[interior] += diff / np.sum(interior)
            h_out = np.clip(h_out, 0.0, 1.0)
    return h_out

def smooth_max_val(corr, alpha):
    m = np.max(corr)
    shifted = alpha * (corr - m)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return m + (log_sum - np.log(len(corr))) / alpha

def get_weights(corr, alpha):
    m = np.max(corr)
    shifted = alpha * (corr - m)
    exp_vals = np.exp(shifted)
    return exp_vals / np.sum(exp_vals)

def gradient_smooth_max(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    sm = smooth_max_val(corr, alpha)
    weights = get_weights(corr, alpha)
    
    lags = np.arange(-(n-1), n)
    grad = np.zeros(n)
    for m in range(n):
        valid1 = (m - lags >= 0) & (m - lags < n)
        term1 = np.dot(weights[valid1], (1.0 - h[m - lags[valid1]]))
        valid2 = (m + lags >= 0) & (m + lags < n)
        term2 = np.dot(weights[valid2], h[m + lags[valid2]])
        grad[m] = term1 - term2
    return sm, grad

def projected_gradient_descent(h0, target_sum, alpha, n_iter, step_size, time_limit, t_start):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)[0]
    
    step = step_size
    for it in range(n_iter):
        if time.time() - t_start > time_limit:
            break
        sm, grad = gradient_smooth_max(h, alpha)
        
        # Projected gradient step
        h_trial = h - step * grad
        h_trial = project_to_sum(h_trial, target_sum)
        
        # Backtracking if needed
        sm_new, _ = gradient_smooth_max(h_trial, alpha)
        c5_trial = compute_c5(h_trial)[0]
        
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h_trial.copy()
        
        if sm_new <= sm + 1e-8:
            h = h_trial
        else:
            # Backtrack
            for _ in range(10):
                step *= 0.5
                h_trial = h - step * grad
                h_trial = project_to_sum(h_trial, target_sum)
                sm_new, _ = gradient_smooth_max(h_trial, alpha)
                if sm_new <= sm + 1e-8:

                    break
            else:

        # Occasionally increase step

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    if abs(np.sum(h) - target_sum) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h_out = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - np.sum(h_out)
    if abs(diff) > 1e-10:
        interior = (h_out > 1e-12) & (h_out < 1.0 - 1e-12)
        if np.sum(interior) > 0:
            h_out[interior] += diff / np.sum(interior)
            h_out = np.clip(h_out, 0.0, 1.0)
    return h_out

def smooth_max(corr, alpha):
    m = np.max(corr)
    shifted = alpha * (corr - m)
    exp_vals = np.exp(shifted)
    log_sum = np.log(np.sum(exp_vals))
    return m + (log_sum - np.log(len(corr))) / alpha

def get_weights(corr, alpha):
    m = np.max(corr)
    shifted = alpha * (corr - m)
    exp_vals = np.exp(shifted)
    return exp_vals / np.sum(exp_vals)

def gradient_smooth_max(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    sm = smooth_max(corr, alpha)
    weights = get_weights(corr, alpha)
    lags = np.arange(-(n-1), n)
    grad = np.zeros(n)
    for m in range(n):
        v1 = (m - lags >= 0) & (m - lags < n)
        term1 = np.dot(weights[v1], (1.0 - h[m - lags[v1]]))
        v2 = (m + lags >= 0) & (m + lags < n)
        term2 = np.dot(weights[v2], h[m + lags[v2]])
        grad[m] = term1 - term2
    return sm, grad

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    if abs(np.sum(h) - target_sum) < 1e-13:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h_out = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - np.sum(h_out)
    if abs(diff) > 1e-12:
        interior = (h_out > 1e-12) & (h_out < 1.0 - 1e-12)
        if np.sum(interior) > 0:
            h_out[interior] += diff / np.sum(interior)
            h_out = np.clip(h_out, 0.0, 1.0)
    return h_out

def smooth_max_val(corr, alpha):
    m = np.max(corr)
    shifted = alpha * (corr - m)
    exp_vals = np.exp(shifted)
    log_sum = np.log(np.sum(exp_vals))
    return m + (log_sum - np.log(len(corr))) / alpha

def get_weights(corr, alpha):
    m = np.max(corr)
    shifted = alpha * (corr - m)
    exp_vals = np.exp(shifted)
    return exp_vals / np.sum(exp_vals)

def gradient_smooth_max(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    sm = smooth_max_val(corr, alpha)
    weights = get_weights(corr, alpha)
    lags = np.arange(-(n - 1), n)
    grad = np.zeros(n)
    for m in range(n):
        v1 = (m - lags >= 0) & (m - lags < n)
        term1 = np.dot(weights[v1], (1.0 - h[m - lags[v1]]))
        v2 = (m + lags >= 0) & (m + lags < n)
        term2 = np.dot(weights[v2], h[m + lags[v2]])
        grad[m] = term1 - term2
    return sm, grad

def pgd_single_step(h, target_sum, alpha, step):
    sm, grad = gradient_smooth_max(h, alpha)
    lr = step
    for _ in range(20):
        h_try = h - lr * grad
        h_try = project_to_sum(h_try, target_sum)
        sm_try, _ = gradient_smooth_max(h_try, alpha)
        if sm_try <= sm + 1e-12:
            return h_try, sm_try, lr
        lr *= 0.5
    return h, sm, min(step, lr)

def block_perturb(h, rng, target_sum, block_size, scale):
    n = len(h)
    h_new = h.copy()
    start = rng.randint(0, n - block_size)
    h_new[start:start + block_size] += rng.uniform(-scale, scale, block_size)
    return project_to_sum(h_new, target_sum)

def pair_swap(h, rng, target_sum):
    n = len(h)
    h_new = h.copy()
    i, j = rng.choice(n, size=2, replace=False)
    h_new[i], h_new[j] = h_new[j], h_new[i]
    return h_new

def optimize_phase(h0, target_sum, alpha, max_iter, t_start, time_limit, rng):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_sm = smooth_max_val(np.correlate(h, 1.0 - h, mode="full"), alpha)
    sm = best_sm
    step = min(0.05, 1.0 / max(alpha, 10.0))
    no_improve = 0
    stall_limit = max(100, max_iter // 4)
    
    for it in range(max_iter):
        if time.time() - t_start > time_limit:
            break
        h, sm_new, step = pgd_single_step(h, target_sum, alpha, step)
        if sm_new < best_sm - 1e-10:
            best_sm = sm_new
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        sm = sm_new
        if no_improve > stall_limit:
            if rng.random() < 0.3:
                h = block_perturb(h, rng, target_sum, max(3, n // 30), 0.1)
            elif rng.random() < 0.5:
                h = pair_swap(h, rng, target_sum)
            else:
                h = h + rng.normal(0, 0.05, n)
                h = project_to_sum(h, target_sum)
            no_improve = 0
        if it % 200 == 100:
            step = min(step * 1.3, 0.1)
    return h, best_h
