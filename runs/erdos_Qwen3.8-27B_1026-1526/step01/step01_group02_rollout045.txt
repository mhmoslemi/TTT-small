import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time


def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            step = diff / np.sum(mask)
            step = min(step, (1.0 - h[mask]).min() - 1e-15)
            if step <= 0:
                break
            h[mask] += step
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            step = -diff / np.sum(mask)
            step = min(step, h[mask].min() - 1e-15)
            if step <= 0:
                break
            h[mask] -= step
    return np.clip(h, 0.0, 1.0)


def soft_max_val(c, beta):
    scaled = beta * c
    max_s = np.max(scaled)
    return max_s / beta + np.log(np.sum(np.exp(scaled - max_s))) / beta


def lbgfbs_obj(params, beta, n, pen_coeff):
    h = params
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    val = soft_max_val(c, beta)
    sum_pen = pen_coeff * (np.sum(h) - n / 2.0) ** 2
    return val + sum_pen


def lbgfbs_optimize(h_init, n, beta, maxiter=100, pen_coeff=1000.0):
    bounds = [(0.0, 1.0)] * n
    result = minimize(
        lbgfbs_obj, h_init.copy(),
        args=(beta, n, pen_coeff),
        method="L-BFGS-B", bounds=bounds,
        options={"maxiter": maxiter, "ftol": 1e-18, "gtol": 1e-14}
    )
    h_out = project_feasible(result.x, n)
    return h_out


def slsqp_obj(x, n):
    return x[-1]


def slsqp_ineq(x):
    h = x[:-1]
    t = x[-1]
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return t - c * dx


def slsqp_eq(x):
    h = x[:-1]
    return np.sum(h) - len(h) / 2.0


def slsqp_eq_jac(x):
    n = len(x) - 1
    jac = np.zeros(len(x))
    jac[:n] = 1.0
    return jac


def slsqp_minimize(h_init, n, maxiter=2000):
    c5_init = compute_c5_from_h(h_init)
    x0 = np.concatenate([h_init, [c5_init + 0.05]])
    bounds = [(0.0, 1.0)] * n + [(max(0.001, c5_init * 0.5), 2.0)]
    constraints = [
        {"type": "ineq", "fun": slsqp_ineq},
        {"type": "eq", "fun": slsqp_eq, "jac": slsqp_eq_jac}
    ]
    result = minimize(
        slsqp_obj, x0, args=(n,),
        method="SLSQP", bounds=bounds,
        constraints=constraints,
        options={"maxiter": maxiter, "ftol": 1e-15}
    )
    if result.x is not None:
        h_out = project_feasible(result.x[:-1], n)
        return h_out, compute_c5_from_h(h_out)
    return h_init, c5_init

def greedy_binary(n, rng):
    h = np.zeros(n)
    budget = n // 2
    for i in range(n):
