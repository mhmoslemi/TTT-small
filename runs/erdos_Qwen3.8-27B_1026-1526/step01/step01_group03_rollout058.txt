import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(values, tau):
    v = values / tau
    vmax = np.max(v)
    return tau * (vmax + np.log(np.sum(np.exp(v - vmax))))

def obj_smooth(h, n, tau):
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return smooth_max(corr, tau) * dx

def constr_sum(h, n):
    return np.sum(h) - n / 2.0

def project_feasible(h, n):
    target = n / 2.0
    lo = -5.0
    hi = 5.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-12:
        order = np.argsort(np.abs(result - 0.5))
        for idx in order:
            adj = result[idx] + diff
            if 0.0 <= adj <= 1.0:
                result[idx] = adj
                break
    return result

def do_slsqp(h, n, tau, maxiter=300):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constr_sum, 'args': (n,)}]
    try:
        res = minimize(
            obj_smooth, h,
            method='SLSQP',
            args=(n, tau),
            bounds=bounds,
            constraints=cons,
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()
