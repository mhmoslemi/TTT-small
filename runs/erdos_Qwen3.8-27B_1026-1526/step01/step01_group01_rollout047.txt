    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': max_iter, 'ftol': 1e-14, 'iprint': 0})
        h_opt = project_feasible(res.x, n)
        return h_opt, compute_c5(h_opt), True
    except Exception:
        return h0, compute_c5(h0), False


def nelder_mead_step(h0, n, max_iter=200):
    h0 = project_feasible(h0, n)

    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr) * (2.0 / n)

    try:
        res = minimize(obj, h0, method='Nelder-Mead',
                       options={'maxiter': max_iter, 'xatol': 1e-10, 'fatol': 1e-12, 'adaptive': True})
        h_opt = project_feasible(res.x, n)
        return h_opt, compute_c5(h_opt), True
    except Exception:
        return h0, compute_c5(h0), False


def subgradient_polish(h, n, n_steps=50, lr=0.001):
    h = project_feasible(h.copy(), n)
    lag_indices = precompute_lag_indices(n)
    for step in range(n_steps):
        c5_vals = get_lag_c5(h)
        max_val = np.max(c5_vals)
        active = np.where(c5_vals >= max_val - 0.0005)[0]
        if len(active) == 0:
            active = [int(np.argmax(c5_vals))]

        grad = np.zeros(n)
        for k in active:
            idx_a, idx_b = lag_indices[k]
            if len(idx_a) == 0:
                continue
            da = h[idx_a]
            db = h[idx_b]
            np.add.at(grad, idx_a, 1.0 - db)
            np.add.at(grad, idx_b, -da)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break

        lr_step = lr / (1.0 + step * 0.1)
        h_new = h - lr_step * grad
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        c5_old = compute_c5(h)
        if c5_new < c5_old:
            h = h_new
        else:
            lr *= 0.5
            if lr < 1e-8:
                break
    return h


def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
        h_init = project_feasible(h_init, n)
    else:
        n = 190
        h_init = np.ones(n) * 0.5

    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    best_n = n
    best_lag_indices = precompute_lag_indices(best_n)

    # Phase 1: Active-set optimization from initial
    if elapsed() < budget_s - 400:
        for it in range(5):
            if elapsed() > budget_s - 400:
                break
            active = find_active_lags(best_h, top_k=25, margin=0.003)
            h_opt, c5_opt, ok = active_set_slsqp(best_h, best_n, active, best_lag_indices, max_iter=400)
            if c5_opt < best_c5:
                best_h, best_c5 = h_opt, c5_opt
                best_lag_indices = precompute_lag_indices(best_n)

    # Phase 2: Subgradient polish
    if elapsed() < budget_s - 350:
        h_polished = subgradient_polish(best_h, best_n, n_steps=80, lr=0.002)
        c5_polished = compute_c5(h_polished)
        if c5_polished < best_c5:
            best_h, best_c5 = h_polished, c5_polished

    # Phase 3: More active-set iterations with updated active set
    if elapsed() < budget_s - 300:
        for it in range(5):
            if elapsed() > budget_s - 300:
                break
            active = find_active_lags(best_h, top_k=20, margin=0.001)
            h_opt, c5_opt, ok = active_set_slsqp(best_h, best_n, active, best_lag_indices, max_iter=300)
            if c5_opt < best_c5:
                best_h, best_c5 = h_opt, c5_opt
            h_polished = subgradient_polish(best_h, best_n, n_steps=40, lr=0.001)
            c5_polished = compute_c5(h_polished)
            if c5_polished < best_c5:
                best_h, best_c5 = h_polished, c5_polished

    # Phase 4: Smooth max with high beta
    if elapsed() < budget_s - 250:
        for beta in [200, 500, 1000, 2000]:
            if elapsed() > budget_s - 250:
                break
            h_opt, c5_opt, ok = smooth_max_slsqp(best_h, best_n, beta, max_iter=300)
            if c5_opt < best_c5:
                best_h, best_c5 = h_opt, c5_opt
                best_lag_indices = precompute_lag_indices(best_n)
            h_polished = subgradient_polish(best_h, best_n, n_steps=30, lr=0.001)
            c5_polished = compute_c5(h_polished)
            if c5_polished < best_c5:
                best_h, best_c5 = h_polished, c5_polished

    # Phase 5: Perturbation-based exploration
    no_improve = 0
    while elapsed() < budget_s - 200 and no_improve < 40:
        sigma = 0.03 + 0.07 * (0.92 ** min(no_improve, 20))
        h_pert = best_h + rng.normal(0, sigma, best_n)
        h_pert = project_feasible(h_pert, best_n)

        if rng.random() < 0.3:
            active = find_active_lags(h_pert, top_k=15, margin=0.002)
            h_opt, c5_opt, ok = active_set_slsqp(h_pert, best_n, active, best_lag_indices, max_iter=250)
        else:
            h_opt, c5_opt, ok = smooth_max_slsqp(h_pert, best_n, 300, max_iter=200)

        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
            no_improve = 0
            h_polished = subgradient_polish(best_h, best_n, n_steps=30, lr=0.001)
            c5_polished = compute_c5(h_polished)
            if c5_polished < best_c5:
                best_h, best_c5 = h_polished, c5_polished
        else:
            no_improve += 1

    # Phase 6: Nelder-Mead on a small subset of variables
    if elapsed() < budget_s - 150:
        h_opt, c5_opt, ok = nelder_mead_step(best_h, best_n, max_iter=300)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
        h_polished = subgradient_polish(best_h, best_n, n_steps=50, lr=0.001)
        c5_polished = compute_c5(h_polished)
        if c5_polished < best_c5:
            best_h, best_c5 = h_polished, c5_polished

    # Phase 7: Try higher resolution
    for n_new in [280, 380]:
        if elapsed() > budget_s - 100:
            break
        x_old = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, n_new)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, n_new)
        lag_idx_new = precompute_lag_indices(n_new)
        active = find_active_lags(h_interp, top_k=25, margin=0.003)
        h_opt, c5_opt, ok = active_set_slsqp(h_interp, n_new, active, lag_idx_new, max_iter=300)
        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, n_new
            best_lag_indices = lag_idx_new
            h_polished = subgradient_polish(best_h, best_n, n_steps=50, lr=0.001)
            c5_polished = compute_c5(h_polished)
            if c5_polished < best_c5:
                best_h, best_c5 = h_polished, c5_polished

    # Phase 8: Final polish - alternating active set and subgradient
    if elapsed() < budget_s - 30:
        for it in range(5):
            if elapsed() > budget_s - 30:
                break
            active = find_active_lags(best_h, top_k=30, margin=0.004)
            h_opt, c5_opt, ok = active_set_slsqp(best_h, best_n, active, best_lag_indices, max_iter=500)
            if c5_opt < best_c5:
                best_h, best_c5 = h_opt, c5_opt
            h_polished = subgradient_polish(best_h, best_n, n_steps=60, lr=0.001)
            c5_polished = compute_c5(h_polished)
            if c5_polished < best_c5:
                best_h, best_c5 = h_polished, c5_polished

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, len(best_h))
