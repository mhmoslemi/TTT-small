import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(vals, alpha):
    m = np.max(vals)
    if alpha <= 0:
        return m
    exp_vals = np.exp(alpha * (vals - m))
    return m + np.log(np.sum(exp_vals)) / alpha - np.log(len(vals)) / alpha

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    e = np.exp(alpha * (corr - m))
    s = np.sum(e)
    return m + np.log(s) / alpha

def smooth_grad(h, alpha):
    n = len(h)
    eps = 1e-6
    f0 = smooth_obj(h, alpha)
    grad = np.zeros(n)
    for i in range(n):
        hp = h.copy()
        hp[i] = min(1.0, max(0.0, h[i] + eps))
        hm = h.copy()
        hm[i] = min(1.0, max(0.0, h[i] - eps))
        grad[i] = (smooth_obj(hp, alpha) - smooth_obj(hm, alpha)) / (hp[i] - hm[i] + 1e-30)
    return grad

def slsqp_optimize(h0, target_sum, alpha, maxiter=500, ftol=1e-12):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return smooth_obj(x, alpha)
    
    def grad(x):
        return smooth_grad(x, alpha)
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.success or res.fun < smooth_obj(h0, alpha):
            return project_to_sum(res.x, target_sum)
    except:
        pass
    return project_to_sum(h0, target_sum)

def coord_descent_finew(h0, target_sum, n_rounds=20, rng=None):
    h = h0.copy()
    n = len(h)
    c5_cur = compute_c5(h)
    steps = [0.01, 0.005, 0.002, 0.001]
    for rnd in range(n_rounds):
        improved = False
        idx = rng.permutation(n) if rng is not None else np.arange(n)
        for i in idx:
            for s in steps:
                for sign in [1, -1]:
                    h_new = h.copy()
                    h_new[i] = np.clip(h_new[i] + sign * s, 0.0, 1.0)
                    h_new = project_to_sum(h_new, target_sum)
                    c5_new = compute_c5(h_new)
                    if c5_new < c5_cur - 1e-13:
                        h = h_new
                        c5_cur = c5_new
                        improved = True
                        break
                if improved:
                    break
        if not improved:
            break
    return h, c5_cur

def block_perturb_optimize(h0, target_sum, rng, alpha, n_blocks=8, block_size=4, maxiter=200):
    n = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    for _ in range(n_blocks):
        start = rng.randint(0, max(1, n - block_size))
        h_new = h0.copy()
        h_new[start:start+block_size] += rng.normal(0, 0.05, block_size)
        h_new = project_to_sum(h_new, target_sum)
        h_opt = slsqp_optimize(h_new, target_sum, alpha, maxiter=maxiter, ftol=1e-10)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
            h0 = h_opt
    return best_h, best_c5

def sinusoidal_perturb_optimize(h0, target_sum, rng, alpha, maxiter=200):
    n = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    for _ in range(12):
        freq = rng.randint(1, n // 4 + 1)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.02, 0.08)
        x = np.arange(n)
        pert = amp * np.sin(2 * np.pi * freq * x / n + phase)
        h_new = project_to_sum(h0 + pert, target_sum)
        h_opt = slsqp_optimize(h_new, target_sum, alpha, maxiter=maxiter, ftol=1e-10)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
            h0 = h_opt
    return best_h, best_c5

def upsample(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, np.concatenate([h_old, [h_old[0]]]), period=2.0)
    return h_new

def downsample(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, np.concatenate([h_old, [h_old[0]]]), period=2.0)
    return h_new

    def time_left():
        return budget_s - (time.time() - t_start)
    
    best_h = None

        maxiter = min(500, max(50, int(time_left() * 1.5)))
