def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-10:
        # Adjust the element closest to 0.5
        idx = np.argsort(np.abs(result - 0.5))[:int(abs(diff)]+1]
        for i in idx:
            adj = min(diff, 1.0 - result[i]) if diff > 0 else max(diff, -result[i])
            result[i] += adj
            diff -= adj
            if abs(diff) < 1e-12:
                break
    return result

def compute_correlations(h):
    n = len(h)
    g = 1.0 - h
    return np.correlate(h, g, mode="full")

def compute_subgradient(h, active_shifts):
    """Compute subgradient of max c_s at h, for given active shifts."""
    n = len(h)
    grad = np.zeros(n)
    for s in active_shifts:
        # s is the shift: c_s = sum_i h[i+s]*(1-h[i])
        # ∂c_s/∂h[j] = (1-h[j-s])*1[0≤j-s<n] - h[j+s]*1[0≤j+s<n]
        # j-s in [0,n-1] means j in [s, s+n-1] ∩ [0,n-1]
        # j+s in [0,n-1] means j in [-s, n-1-s] ∩ [0,n-1]
        
        for j in range(n):
            val = 0.0
            js = j - s
            if 0 <= js < n:
                val += (1.0 - h[js])
            js2 = j + s
            if 0 <= js2 < n:
                val -= h[js2]
            grad[j] += val
    # Average over active shifts
    m = len(active_shifts)
    if m > 0:
        grad /= m
    return grad

    grad = np.zeros(n)
    for s in active_shifts:
        g_s = np.zeros(n)
        if s >= 0:
            # (1-h[j-s]) for j in [s, n-1]
            g_s[s:] += (1.0 - h[:n-s])
            # -h[j+s] for j in [0, n-1-s]
            g_s[:n-s] -= h[s:]
        else:
            # (1-h[j-|s|]) for j in [0, n-1-|s|]

            # -h[j+|s|]) for j in [|s|, n-1]
            g_s[-s:] -= h[:n+s]
        grad += g_s
    m = len(active_shifts)
    if m > 0:
        grad /= m
    return grad

def smooth_max(values, tau):
    v = values / tau
    vmax = np.max(v)
    return tau * (vmax + np.log(np.sum(np.exp(v - vmax))))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, -10, 10)
    lo, hi = -5.0, 5.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-14:
        # Find elements to adjust
        order = np.argsort(np.abs(result - 0.5))
        for i in order:
            if diff > 0:
                adj = min(diff, 1.0 - result[i])
            else:
                adj = max(diff, -result[i])
            result[i] += adj
            diff -= adj
            if abs(diff) < 1e-14:
                break
    return result

    """Compute gradient of c_s w.r.t. h."""
    n = len(h)
    g = np.zeros(n)
    if s >= 0:
        g[s:] += (1.0 - h[:n-s])
        if n-s > 0:
            g[:n-s] -= h[s:]
    else:
        a = -s
        if n-a > 0:
            g[:n-a] += (1.0 - h[a:])
        if a > 0:
            g[a:] -= h[:n-a]
    return g

    n = len(h)
    grad = np.zeros(n)
    m = len(shifts)

    if m > 0:

                    best_h = h_fin
            except Exception:
                pass

    if best_h is None:
        n = 59
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c5 = evaluate_c5(best_h)

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = evaluate_c5(best_h)

    return best_h, best_c5, len(best_h)
