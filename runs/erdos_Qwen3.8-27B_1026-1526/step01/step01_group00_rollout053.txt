```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum}"""
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        # Water-filling
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
        else:
            break
    return h
```

```python
def compute_gradient(h, w, n):
    grad = np.zeros(n)
    for k in range(len(w)):
        if w[k] < 1e-14:
            continue
        lag = k - (n - 1)
        wk = w[k]
        # First part: (1-h[j-lag]) for valid j
        j_start = max(0, lag)
        j_end = min(n, n + lag)
        if j_end > j_start:
            h_idx = np.arange(j_start, j_end) - lag
            grad[j_start:j_end] += wk * (1.0 - h[h_idx])
        # Second part: -h[j+lag] for valid j
        j_start2 = max(0, -lag)
        j_end2 = min(n, n - lag)
        if j_end2 > j_start2:
            h_idx2 = np.arange(j_start2, j_end2) + lag
            grad[j_start2:j_end2] -= wk * h[h_idx2]
    return grad
```

This is O(n * num_active_k). For large alpha, num_active_k is small. For small alpha, it could be O(n) but that's still fine.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h

def corr_gradient(h, w, n):
    grad = np.zeros(n)
    for k in range(len(w)):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag <= 0:
            # First part: (1-h[j-lag]) for j in [0, n-1] with j-lag >= 0 => j >= -lag = |lag|
            js = max(0, -lag)
            je = n
            h_idx = np.arange(js, je) - lag
            grad[js:je] += wk * (1.0 - h[h_idx])
            # Second part: -h[j+lag] for j in [0, n-1] with j+lag < n => j < n-lag
            js2 = 0
            je2 = min(n, n + lag)  # n + lag could be less than n if lag < 0
            if je2 > js2:
                h_idx2 = np.arange(js2, je2) + lag
                grad[js2:je2] -= wk * h[h_idx2]
        else:
            # lag > 0
            # First part: (1-h[j-lag]) for j in [lag, n-1]
            js = lag
            je = n
            if je > js:
                h_idx = np.arange(js, je) - lag
                grad[js:je] += wk * (1.0 - h[h_idx])
            # Second part: -h[j+lag] for j in [0, n-1-lag]
            js2 = 0
            je2 = n - lag
            if je2 > js2:
                h_idx2 = np.arange(js2, je2) + lag
                grad[js2:je2] -= wk * h[h_idx2]
    return grad

def fista(h0, n, target_sum, alpha, n_iter, time_fn, time_limit):
    h = h0.copy()
    z = h0.copy()
    t = 1.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Initial step size
    step = 0.5 / alpha  # rough
    
    for it in range(n_iter):
        if time_fn() < time_limit:
            break
        
        corr = np.correlate(z, 1.0 - z, mode="full")
        m = np.max(corr)
        exp_v = np.exp(alpha * (corr - m))
        w = exp_v / np.sum(exp_v)
        
        grad = corr_gradient(z, w, n)
        
        # Backtracking line search
        f_z = m + np.log(np.sum(exp_v)) / alpha
        step_new = step
        h_candidate = z - step_new * grad
        h_candidate = project_box_sum(h_candidate, target_sum)
        
        for _ in range(20):
            corr_c = np.correlate(h_candidate, 1.0 - h_candidate, mode="full")
            f_c = np.max(corr_c) + np.log(np.sum(np.exp(alpha * (corr_c - np.max(corr_c))))) / alpha
            L_h = 2.0 * alpha * np.sum(exp_v) / (n + 1)  # rough Lipschitz
            if f_c <= f_z + np.dot(grad, h_candidate - z) + (1.0 / (2.0 * step_new)) * np.sum((h_candidate - z)**2):
                break
            step_new *= 0.5
            h_candidate = z - step_new * grad
            h_candidate = project_box_sum(h_candidate, target_sum)
        
        step = step_new * 1.5  # increase for next iter (with cap)
        step = min(step, 2.0 / alpha)
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        h_new = h_candidate  # This is the proximal step
        
        z = h_new + ((t - 1.0) / t_new) * (h_new - h)
        z = project_box_sum(z, target_sum)
        h = h_new
        t = t_new
        
        c5_it = compute_c5(h)
        if c5_it < best_c5:
            best_c5 = c5_it
            best_h = h.copy()
    
    return best_h, best_c5

def subgradient_step(h, n, target_sum, step_size):
    """One subgradient step of the exact max."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_max = np.argmax(corr)
    
    # Subgradient at k_max
    lag = k_max - (n - 1)
    grad = np.zeros(n)
    if lag <= 0:
        js = max(0, -lag)
        je = n
        h_idx = np.arange(js, je) - lag
        grad[js:je] += (1.0 - h[h_idx])
        je2 = min(n, n + lag)
        if je2 > 0:
            h_idx2 = np.arange(0, je2) + lag
            grad[0:je2] -= h[h_idx2]
    else:
        js = lag
        je = n
        if je > js:
            h_idx = np.arange(js, je) - lag
            grad[js:je] += (1.0 - h[h_idx])
        je2 = n - lag
        if je2 > 0:
            h_idx2 = np.arange(0, je2) + lag
            grad[0:je2] -= h[h_idx2]
    
    h_new = h - step_size * grad
    h_new = project_box_sum(h_new, target_sum)
    return h_new

def subgradient_descent(h0, n, target_sum, n_iter, time_fn, time_limit, step_init=0.01):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = step_init
    
    for it in range(n_iter):
        if time_fn() < time_limit:
            break
        
        h = subgradient_step(h, n, target_sum, step)
        c5_it = compute_c5(h)
        
        if c5_it < best_c5:
            best_c5 = c5_it
            best_h = h.copy()
            step = min(step * 1.1, 0.05)
        else:
            step *= 0.9
        
        # Also try with top-3 lags
        corr = np.correlate(h, 1.0 - h, mode="full")
        top_k = np.argsort(corr)[-3:]
        total_grad = np.zeros(n)
        for k in top_k:
            lag = k - (n - 1)
            if lag <= 0:
                js = max(0, -lag)
                je = n
                h_idx = np.arange(js, je) - lag
                total_grad[js:je] += (1.0 - h[h_idx])
                je2 = min(n, n + lag)
                if je2 > 0:
                    h_idx2 = np.arange(0, je2) + lag
                    total_grad[0:je2] -= h[h_idx2]
            else:
                js = lag
                je = n
                if je > js:
                    h_idx = np.arange(js, je) - lag
                    total_grad[js:je] += (1.0 - h[h_idx])
                je2 = n - lag
                if je2 > 0:
                    h_idx2 = np.arange(0, je2) + lag
                    total_grad[0:je2] -= h[h_idx2]
        
        h_trial = h - (step / 3.0) * total_grad
        h_trial = project_box_sum(h_trial, target_sum)
        c5_trial = compute_c5(h_trial)
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h_trial.copy()
            h = h_trial
    
    return best_h, best_c5

def coordinate_local_search(h0, n, target_sum, n_passes, time_fn, time_limit, rng):
    h = h0.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    
    for p in range(n_passes):
        if time_fn() < time_limit:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            c5_cur = compute_c5(h)
            # Try small adjustments
            for delta in [-0.01, -0.005, 0.005, 0.01]:
                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] + delta, 0.0, 1.0)
                h_new = project_box_sum(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-12:
                    h = h_new
                    best_c5 = c5_new
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5

def interpolate_to_resolution(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old, period=2.0)
    h_new = project_box_sum(h_new, n_new / 2.0)
    return h_new

def run(seed=42, budget_s=1300, **kwargs):

    def time_left():
        return budget_s - (time.time() - t_start)
    
    rng = np.random.RandomState(seed)
    
    best_h = None
    best_c5 = float('inf')
    best_n = 180
    
    # Initialize from parent
    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_c5 = compute_c5(best_h)
            best_h = project_box_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except:
            pass
    
    if best_h is None:

        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    for alpha in [10.0, 30.0, 100.0, 300.0]:

        if c5_fista < best_c5:
            best_c5 = c5_fista
            best_h = h_fista.copy()

            best_c5 = c5_cd
            best_h = h_cd.copy()
```

```python
def active_set_step(h, n, target_sum, step, top_m=3):
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_lags = np.argsort(corr)[-top_m:]
    total_grad = np.zeros(n)
    for k in top_lags:
        lag = k - (n - 1)
        if lag >= 0:
            js = lag
            je = n
            if je > js:
                h_idx = np.arange(js, je) - lag
                total_grad[js:je] += (1.0 - h[h_idx])
            je2 = n - lag
            if je2 > 0:
                h_idx2 = np.arange(0, je2) + lag
                total_grad[0:je2] -= h[h_idx2]
        else:
            js = -lag
            je = n
            if je > js:
                h_idx = np.arange(js, je) - lag
                total_grad[js:je] += (1.0 - h[h_idx])
            je2 = n + lag
            if je2 > 0:
                h_idx2 = np.arange(0, je2) + lag
                total_grad[0:je2] -= h[h_idx2]
    h_new = h - (step / top_m) * total_grad
    return project_box_sum(h_new, target_sum)
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(h, target_sum):
    h = h.astype(float).copy()
    np.clip(h, 0.0, 1.0, out=h)
    for _ in range(300):
        s = float(np.sum(h))
        diff = target_sum - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h

def grad_single_lag(h, lag, n):
    """Gradient of corr at given lag w.r.t. h."""
    g = np.zeros(n)
    if lag >= 0:
        # (1-h[j-lag]) for j in [lag, n-1]
        if lag < n:
            idx = np.arange(lag, n) - lag
            g[lag:n] += (1.0 - h[idx])
        # -h[j+lag] for j in [0, n-1-lag]
        cnt = n - lag
        if cnt > 0:
            idx2 = np.arange(0, cnt) + lag
            g[0:cnt] -= h[idx2]
    else:
        # lag < 0, let L = -lag
        L = -lag
        # (1-h[j+L]) for j in [0, n-1-L]
        cnt = n - L
        if cnt > 0:
            idx = np.arange(0, cnt) + L
            g[0:cnt] += (1.0 - h[idx])
        # -h[j-L] for j in [L, n-1]
        if L < n:
            idx2 = np.arange(L, n) - L
            g[L:n] -= h[idx2]
    return g

def fista_optimize(h0, n, target_sum, alpha, n_iter, elapsed_fn, time_limit):
    h = h0.copy()
    z = h0.copy()
    t = 1.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = 1.0 / (4.0 * alpha)
    
    for it in range(n_iter):
        if elapsed_fn() > 1300 - time_limit:
            break
        
        corr = np.correlate(z, 1.0 - z, mode="full")
        m = float(np.max(corr))
        e = np.exp(alpha * (corr - m))
        w = e / float(np.sum(e))
        
        # Gradient of smooth max
        grad = np.zeros(n)
        active = np.where(w > 1e-12)[0]
        for k in active:
            lag = k - (n - 1)
            grad += w[k] * grad_single_lag(z, lag, n)
        
        # Gradient step
        h_new = z - step * grad
        h_new = project_box_sum(h_new, target_sum)
        
        # Momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        z = h_new + ((t - 1.0) / t_new) * (h_new - h)
        z = project_box_sum(z, target_sum)
        h = h_new
