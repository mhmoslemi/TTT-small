    max_val = np.max(all_overlaps)

    grad = np.zeros(n)
    for m in active:

        g = np.zeros(n)

            if 0 <= jm < n:
                val -= h[jm]

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            cnt = float(np.sum(mask))
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 1e-14
            cnt = float(np.sum(mask))
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)[::-1]
            for idx in order:
                if diff <= 1e-14:
                    break
                step = min(1.0 - h[idx], diff)
                h[idx] += step
                diff -= step
        else:
            order = np.argsort(h)[::-1]
            for idx in order:
                if diff >= -1e-14:
                    break
                step = max(h[idx] + diff, 0.0) - h[idx]
                h[idx] += step
                diff -= step
    h = np.clip(h, 0.0, 1.0)
    return h


def min_1d_convex_pl(A, B):
    x_lo = 0.0
    x_hi = 1.0
    for _ in range(80):
        x_mid = 0.5 * (x_lo + x_hi)
        vals = A * x_mid + B
        max_val = np.max(vals)
        active = vals >= max_val - 1e-12
        if not np.any(active):
            active = vals >= max_val - 1e-8
        if np.all(A[active] > 1e-14):
            x_hi = x_mid
        elif np.all(A[active] < -1e-14):
            x_lo = x_mid
        else:
            return x_mid
    return 0.5 * (x_lo + x_hi)


def coord_descent(h_init, n, num_sweeps, order=None):
    h = h_init.copy()
    dx = 2.0 / n
    M = 2 * n - 1
    overlaps = np.correlate(h, 1.0 - h, mode="full") * dx
    L_range = np.arange(-(n - 1), n)
    for sweep in range(num_sweeps):
        seq = order if order is not None else range(n)
        for j in seq:
            jp_valid = (L_range >= -j) & (L_range < n - j)
            jm_valid = (L_range > j - n) & (L_range <= j)
            A = np.zeros(M)
            A[jp_valid] += (1.0 - h[j + L_range[jp_valid]])
            A[jm_valid] -= h[j - L_range[jm_valid]]
            A *= dx
            B = overlaps - A * h[j]
            x_new = min_1d_convex_pl(A, B)
            delta = x_new - h[j]
            if abs(delta) > 1e-15:
                h[j] = x_new
                overlaps += delta * A
        h = project_feasible(h, n)
        overlaps = np.correlate(h, 1.0 - h, mode="full") * dx
    return h


def smooth_max_obj(x, n, beta):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def slsqp_polish(h_init, n, beta, max_iter):
    h0 = project_feasible(h_init, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x, n=n: np.sum(x) - n / 2.0}]
    try:
        res = minimize(smooth_max_obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, args=(n, beta),
                       options={'maxiter': max_iter, 'ftol': 1e-14, 'iprint': 0})
        h_res = project_feasible(res.x, n)
        return h_res, compute_c5(h_res)
    except Exception:
        return None, None


def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def make_binary_random(n, rng):
    h = np.zeros(n)
    ones = n // 2
    idx = rng.choice(n, size=ones, replace=False)
    h[idx] = 1.0
    return h


def make_block_pattern(n, rng):
    h = np.zeros(n)
    for _ in range(n // 2):
        pos = rng.randint(0, n)
        length = rng.randint(2, max(3, n // 10))
        for k in range(length):
            h[(pos + k) % n] = 1.0
    count = np.sum(h == 1.0)
    target = n // 2
    if count > target:
        extra = np.where(h == 1.0)[0]
        remove = rng.choice(extra, size=count - target, replace=False)
        h[remove] = 0.0
    elif count < target:
        space = np.where(h == 0.0)[0]
        add = rng.choice(space, size=target - count, replace=False)
        h[add] = 1.0
    return h


def subgradient_polish(h_init, n, num_steps, lr_init):
    h = h_init.copy()
    dx = 2.0 / n
    M = 2 * n - 1
    L_range = np.arange(-(n - 1), n)
    for step in range(num_steps):
        overlaps = np.correlate(h, 1.0 - h, mode="full") * dx
        max_val = np.max(overlaps)
        tol = 0.02 * max_val
        active = np.where(overlaps > max_val - tol)[0]
        lr = lr_init / (1.0 + 0.01 * step)
        grad = np.zeros(n)
        for m in active:
            L = L_range[m]
            jp_valid = (L_range + 0 >= -(0))  # dummy, compute per-j below
            for j in range(n):
                val = 0.0
                jp = j + L
                jm = j - L
                if 0 <= jp < n:
                    val += (1.0 - h[jp])
                if 0 <= jm < n:
                    val -= h[jm]
                grad[j] += val * dx
        grad /= max(1, len(active))
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        h = h - lr * grad / grad_norm
        h = project_feasible(h, n)
    return h
