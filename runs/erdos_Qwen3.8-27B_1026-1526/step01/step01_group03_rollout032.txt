def smooth_max_obj(h, n, tau, lam):

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import CubicSpline
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_obj(h, n, tau, lam):
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    v = corr / tau
    vmax = np.max(v)
    sm = tau * (vmax + np.log(np.sum(np.exp(v - vmax))))
    penalty = lam * (np.sum(h) - n / 2.0) ** 2
    return sm * dx + penalty

def project_feasible(h, n):
    target = n / 2.0
    lo = -np.max(h) - 0.5
    hi = 1.0 - np.min(h) + 0.5
    for _ in range(120):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-13:
        if diff > 0:
            cands = np.where(result < 1.0 - 1e-12)[0]
            space = 1.0 - result[cands]
        else:
            cands = np.where(result > 1e-12)[0]
            space = result[cands]
        if len(cands) > 0:
            total = np.sum(space)
            if total > 1e-15:
                result[cands] += space * (diff / total)
                result = np.clip(result, 0.0, 1.0)
    return result

def interpolate_h(h_src, n_dst):
    x_src = np.linspace(0, 2, len(h_src), endpoint=False)
    x_dst = np.linspace(0, 2, n_dst, endpoint=False)
    cs = CubicSpline(x_src, h_src)
    return np.clip(cs(x_dst), 0.0, 1.0)

def coord_descent_polish(h, n, tau, lam, max_sweeps=15):
    dx = 2.0 / n
    h = h.copy()
    for sweep in range(max_sweeps):
        improved = False
        for i in range(n):
            best_obj = smooth_max_obj(h, n, tau, lam)
            best_val = h[i]
            for trial in np.linspace(0.0, 1.0, 21):
                h[i] = trial
                obj = smooth_max_obj(h, n, tau, lam)
                if obj < best_obj - 1e-15:
                    best_obj = obj
                    best_val = trial
            if abs(best_val - h[i]) > 1e-10:
                h[i] = best_val
                improved = True
            else:
                h[i] = best_val
        if not improved:
            break
    return h
