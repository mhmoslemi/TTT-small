```python
def compute_gradient(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
    w = np.exp(alpha * (corr - np.max(corr)))
    w /= np.sum(w)
    
    # Build the gradient
    # For each j, grad[j] = sum_k w[k] * [(1-h[j+lag[k]]) * valid1 - h[j-lag[k]] * valid2]
    # where lag[k] = k - (n-1)
    
    # Vectorize: create a matrix M where M[j, k] = (1-h[j+lag[k]]) if valid else 0
    # and N[j, k] = h[j-lag[k]] if valid else 0
    
    j = np.arange(n)
    k = np.arange(2*n-1)
    lags = k - (n-1)
    
    # j + lag: shape (n, 2n-1)
    jp = j[:, None] + lags[None, :]
    jm = j[:, None] - lags[None, :]
    
    valid_p = (jp >= 0) & (jp < n)
    valid_m = (jm >= 0) & (jm < n)
    
    # Clip for indexing
    jp_clipped = np.clip(jp, 0, n-1)
    jm_clipped = np.clip(jm, 0, n-1)
    
    term1 = np.where(valid_p, 1.0 - h[jp_clipped], 0.0)
    term2 = np.where(valid_m, h[jm_clipped], 0.0)
    
    grad = w[None, :] @ (term1 - term2)  # shape (n,)
    
    return grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def smooth_max_val(corr, alpha):
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha

def smooth_max_grad_h(h, alpha):
    """Compute gradient of smooth_max(corr(h), alpha) with respect to h."""
    n = len(h)
    corr = compute_corr(h)
    # Softmax weights
    m = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m))
    w = exp_vals / np.sum(exp_vals)
    
    # Compute gradient using the formula:
    # d(smooth_max)/d(h[j]) = sum_k w[k] * [(1-h[j+lag_k])*[valid] - h[j-lag_k]*[valid]]
    # where lag_k = (n-1-k)
    
    j = np.arange(n)
    k = np.arange(2*n-1)
    lags = (n-1) - k  # lag for each k
    
    jp = j[:, None] + lags[None, :]  # j + lag, shape (n, 2n-1)
    jm = j[:, None] - lags[None, :]  # j - lag, shape (n, 2n-1)
    
    valid_p = (jp >= 0) & (jp < n)
    valid_m = (jm >= 0) & (jm < n)
    
    jp_c = np.clip(jp, 0, n-1).astype(int)
    jm_c = np.clip(jm, 0, n-1).astype(int)
    
    h_jp = h[jp_c]  # shape (n, 2n-1)
    h_jm = h[jm_c]
    
    term1 = np.where(valid_p, 1.0 - h_jp, 0.0)
    term2 = np.where(valid_m, h_jm, 0.0)
    
    grad = (term1 - term2) @ w  # shape (n,)
    
    return grad

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def projected_gradient_descent(h0, n, target_sum, alpha, max_iters, lr_init, t_start, time_limit):
    h = h0.copy()
    c5_cur = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_cur
    lr = lr_init
    
    for it in range(max_iters):
        if time.time() - t_start > time_limit - 5:
            break
        
        grad = smooth_max_grad_h(h, alpha)
        # Project gradient to tangent space
        grad_tan = grad - np.mean(grad)
        
        # Backtracking line search
        obj_cur = smooth_max_val(compute_corr(h), alpha)
        best_h_ls = h.copy()
        best_obj_ls = obj_cur
        ls_lr = lr
        for bt in range(30):
            h_new = h - ls_lr * grad_tan
            h_new = project_to_sum(h_new, target_sum)
            obj_new = smooth_max_val(compute_corr(h_new), alpha)
            if obj_new < obj_cur - 1e-12:
                best_h_ls = h_new
                best_obj_ls = obj_new
                break
            ls_lr *= 0.5
        
        if best_obj_ls < obj_cur:
            h = best_h_ls
            lr = min(lr * 1.5, 0.1)  # increase step if successful
        else:
            lr = lr * 0.7  # decrease step
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5
```

```python
        grad_tan = grad - np.mean(grad)
        ls_lr = lr
        found = False
        for _ in range(25):
            h_new = h - ls_lr * grad_tan
            h_new = project_to_sum(h_new, target_sum)
            obj_new = smooth_obj(h_new, alpha)
            if obj_new < obj_cur - 1e-10:
                found = True
                break
            ls_lr *= 0.5
        if found:
            h = h_new
            obj_cur = obj_new
            lr = min(lr * 1.2, 2.0)
            stall = 0
        else:
            lr *= 0.5
            stall += 1
            if stall > 30 or lr < 1e-8:
                break
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def binary_local_search(h_init, n, t_start, time_budget):
    t_deadline = t_start + time_budget
    h_bin = (h_init > 0.5).astype(float)
    target = n // 2
    n_ones = int(np.sum(h_bin))
    rng = np.random.RandomState(77)
    if n_ones > target:
        ones = np.where(h_bin == 1)[0]
        to_rmv = rng.choice(ones, n_ones - target, replace=False)
        h_bin[to_rmv] = 0
    elif n_ones < target:
        zeros = np.where(h_bin == 0)[0]
        to_add = rng.choice(zeros, target - n_ones, replace=False)
        h_bin[to_add] = 1
    best_c5 = compute_c5(h_bin)
    best_h = h_bin.copy()
    for rnd in range(5000):
        if time.time() > t_deadline:
            break
        improved = False
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        order = rng.permutation(len(ones))
        for idx in order:
            i = ones[idx]
            j_candidates = zeros
            for jidx in range(len(j_candidates)):
                j = j_candidates[jidx]
                if i == j:
                    continue
                h_new = h_bin.copy()
                h_new[i] = 0.0
                h_new[j] = 1.0
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-12:
                    h_bin = h_new
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return best_h, best_c5

def quantize_refine(h, t_start, time_budget, alpha=100):
    t_deadline = t_start + time_budget
    n = len(h)
    target_sum = n / 2.0
    h_work = h.copy()
    best_h = h_work.copy()
    best_c5 = compute_c5(h_work)
    for _ in range(10):
        if time.time() > t_deadline:
            break
        h_q = h_work.copy()
        h_q[h_q > 0.85] = 1.0
        h_q[h_q < 0.15] = 0.0
        h_q = project_to_sum(h_q, target_sum)
        h_opt, c5_opt = pgd_optimize(h_q, alpha, t_start, min(time_budget - (time.time() - t_start), 60), lr_init=0.05)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            h_work = h_opt.copy()
        else:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 180
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    rng = np.random.RandomState(seed)
    n = best_n
    target_sum = n / 2.0
    elapsed_limit = budget_s - 5

    def remaining():
        return max(0, elapsed_limit - (time.time() - t_start))

    # Phase 1: PGD sweep with increasing alpha
    alphas = [15, 40, 80, 150, 250, 400]
    for a in alphas:
        if remaining() < 40:
            break
        alloc = min(remaining() * 0.25, 250)
        h_opt, c5_opt = pgd_optimize(best_h, a, t_start, alloc, lr_init=0.3)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 2: Restart with perturbations
    n_restarts = 6
    for r in range(n_restarts):
        if remaining() < 50:
            break
        sigma = 0.015 + 0.01 * r
        h_pert = best_h + rng.normal(0, sigma, n)
        h_pert = project_to_sum(h_pert, target_sum)
        alloc = min(remaining() * 0.2, 180)
        for a in [80, 200]:
            if remaining() < 30:
                break
            h_opt, c5_opt = pgd_optimize(h_pert, a, t_start, alloc * 0.5, lr_init=0.1)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                h_pert = h_opt.copy()

    # Phase 3: Quantize and refine
    if remaining() > 80:
        alloc = min(remaining() * 0.25, 200)
        h_q, c5_q = quantize_refine(best_h, t_start, alloc, alpha=150)
        if c5_q < best_c5:
            best_c5 = c5_q
            best_h = h_q.copy()

    # Phase 4: Binary local search
    if remaining() > 60:
        alloc = min(remaining() * 0.3, 300)
        h_b, c5_b = binary_local_search(best_h, n, t_start, alloc)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Phase 5: Final polish
    if remaining() > 30:
        for a in [200, 500, 1000]:
            if remaining() < 15:
                break
            h_opt, c5_opt = pgd_optimize(best_h, a, t_start, min(remaining() * 0.3, 80), lr_init=0.02)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Phase 6: Try higher resolution if time permits
    if remaining() > 100:
        n2 = (n // 10 + 1) * 10
        if n2 % 2 == 0 and n2 != n:
            x_old = np.linspace(0, 2, n, endpoint=False)
            x_new = np.linspace(0, 2, n2, endpoint=False)
            h2 = np.interp(x_new, x_old, best_h)
            h2 = project_to_sum(h2, n2 / 2.0)
            for a in [50, 150, 300]:
                if remaining() < 40:
                    break
                h_opt, c5_opt = pgd_optimize(h2, a, t_start, min(remaining() * 0.3, 150), lr_init=0.1)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    best_n = n2

    # Final validation
    n_final = len(best_h)
    best_h = project_to_sum(best_h, n_final / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_final)
```
