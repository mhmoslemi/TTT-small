def simulated_annealing(h_binary, n, n_iterations=100000, T0=0.01, T_end=0.0001, rng=None):
    if rng is None:
        rng = np.random.default_rng()
    h = h_binary.copy()
    c5 = compute_c5_from_h(h)
    best_c5 = c5
    best_h = h.copy()
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(n_iterations):
        T = T0 * (T_end / T0) ** (it / n_iterations)
        
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        
        if i == j:
            continue

        new_c5 = compute_c5_from_h(h)
        
        delta = new_c5 - c5
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            c5 = new_c5
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]

def compute_c5_from_h(h):

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            room = 1.0 - h[mask]
            step = min(diff / np.sum(mask), room.min())
            h[mask] += step
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            room = h[mask]
            step = min(-diff / np.sum(mask), room.min())

    c_max_idx = np.argmax(c)
    k_star = c_max_idx - (n - 1)

    else:

    else:

        g[-k_star:] -= h[:n + k_star]
    
    return g * dx, c.max() * dx

    best_h = h.copy()
    best_c5 = compute_c5_from_h(h)
    
    lr = lr_init
    for it in range(n_iter):

        h_new = h - lr * g

    return best_h, best_c5

def logsumexp_c5_grad(h, tau=0.01):

    w = w / np.sum(w)

    # c_k = dx * corr_k where corr_k = sum_n h[n+k] * (1-h[n]) (with appropriate bounds)
    # d(corr_k)/dh[i] = (1-h[i-k]) * [valid] - h[i+k] * [valid]
    # (using the numpy correlate convention where corr at index m corresponds to shift k = m-(n-1))

    for m in range(len(c)):
        if w[m] < 1e-15:

        if k >= 0:
            g[:n-k] -= w[m] * dx * h[k:]
        else:
            g[-k:] -= w[m] * dx * h[:n+k]
