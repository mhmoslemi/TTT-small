1. Find k* = argmax_k corr_k
2. Minimize corr_{k*}(h) subject to constraints
3. Repeat

from scipy.interpolate import interp1d

def evaluate_c5(h):

    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")

def smooth_max_obj(h, n, tau):
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    v = corr / tau
    vmax = np.max(v)
    return tau * (vmax + np.log(np.sum(np.exp(v - vmax)))) * dx

def smooth_max_grad(h, n, tau):
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    v = corr / tau
    vmax = np.max(v)
    weights = np.exp(v - vmax)
    weights /= np.sum(weights)
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        m = k - (n - 1)
        w = weights[k] * dx
        if abs(w) < 1e-14:
            continue
        i1_lo = max(0, m)
        i1_hi = min(n, n + m)
        if i1_hi > i1_lo:
            grad[i1_lo:i1_hi] += w * (1.0 - h[i1_lo - m:i1_hi - m])
        i2_lo = max(0, -m)
        i2_hi = min(n, n - m)
        if i2_hi > i2_lo:
            grad[i2_lo:i2_hi] -= w * h[i2_lo + m:i2_hi + m]
    return grad

def obj_with_grad(h, n, tau):
    return smooth_max_obj(h, n, tau), smooth_max_grad(h, n, tau)

def constraint_func(h, n):
    return np.sum(h) - n / 2.0

def project_feasible(h, n):
    target = n / 2.0
    lo = -np.max(h) - 1.0
    hi = 1.0 - np.min(h) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-12:
        order = np.argsort(np.abs(result - 0.5))
        for idx in order:
            if diff > 0:
                adjust = min(diff, 1.0 - result[idx])
            else:
                adjust = max(diff, -result[idx])
            result[idx] += adjust
            diff -= adjust
            if abs(diff) < 1e-12:
                break
    return result

    tau_schedule = [0.3, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.003, 0.0015, 0.001]

    h_current = parent_h.copy() if parent_h is not None else best_h.copy()
    h_current = project_feasible(h_current, n)

    for tau in tau_schedule:
        if elapsed(start_time) > budget_s * 0.40:
            break
        try:
            res = minimize(
                obj_with_grad, h_current,
                method='SLSQP',
                args=(n, tau),
                jac=True,
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 300, 'ftol': 1e-16}
            )
            h_current = project_feasible(res.x, n)
        except Exception:
            pass
        c5 = evaluate_c5(h_current)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_current.copy()

    x = np.linspace(0, 2, n, endpoint=False)

    diverse_starts = []
    if parent_h is not None:
        for scale in [0.03, 0.06, 0.1, 0.15, 0.25]:
            diverse_starts.append(np.clip(parent_h + rng.randn(n) * scale, 0, 1))
        for _ in range(6):
            h_p = parent_h.copy()
            n_pert = rng.randint(n // 5, n // 2)
            idx = rng.choice(n, size=n_pert, replace=False)
            h_p[idx] = np.clip(h_p[idx] + rng.randn(n_pert) * 0.35, 0, 1)
            diverse_starts.append(h_p)

    for _ in range(6):
        n_terms = rng.randint(2, 5)
        freqs = rng.uniform(0.5, 4.5, size=n_terms)
        phas = rng.uniform(0, 2 * np.pi, size=n_terms)
        amps = rng.uniform(0.08, 0.32, size=n_terms)
        h_ft = np.full(n, 0.5)
        for i in range(n_terms):
            h_ft += amps[i] * np.sin(2 * np.pi * freqs[i] * x + phas[i])
        diverse_starts.append(np.clip(h_ft, 0, 1))

    for _ in range(4):
        diverse_starts.append(rng.rand(n))

    coarse_to_fine = []
    for n_coarse in [20, 30]:
        if parent_h is not None:
            xc = np.linspace(0, 2, n_coarse, endpoint=False)
            f_interp = interp1d(x, parent_h, kind='cubic', fill_value='extrapolate')
            h_coarse = np.clip(f_interp(xc), 0, 1)
            h_coarse = project_feasible(h_coarse, n_coarse)
            bounds_c = [(0.0, 1.0)] * n_coarse
            cons_c = [{'type': 'eq', 'fun': constraint_func, 'args': (n_coarse,)}]
            for tau_c in [0.1, 0.03, 0.01]:
                if elapsed(start_time) > budget_s * 0.55:
                    break
                try:
                    res_c = minimize(
                        obj_with_grad, h_coarse,
                        method='SLSQP',
                        args=(n_coarse, tau_c),
                        jac=True,
                        bounds=bounds_c,
                        constraints=cons_c,
                        options={'maxiter': 200, 'ftol': 1e-14}
                    )
                    h_coarse = project_feasible(res_c.x, n_coarse)
                except Exception:
                    pass
            f_fine = interp1d(xc, h_coarse, kind='cubic', fill_value='extrapolate')
            h_fine = np.clip(f_fine(x), 0, 1)
            coarse_to_fine.append(h_fine)
        else:
            h_c = project_feasible(rng.rand(n_coarse), n_coarse)
            bounds_c = [(0.0, 1.0)] * n_coarse
            cons_c = [{'type': 'eq', 'fun': constraint_func, 'args': (n_coarse,)}]
            for tau_c in [0.05, 0.02]:
                if elapsed(start_time) > budget_s * 0.55:
                    break
                try:
                    res_c = minimize(
                        obj_with_grad, h_c,
                        method='SLSQP',
                        args=(n_coarse, tau_c),
                        jac=True,
                        bounds=bounds_c,
                        constraints=cons_c,
                        options={'maxiter': 150, 'ftol': 1e-14}
                    )
                    h_c = project_feasible(res_c.x, n_coarse)
                except Exception:
                    pass
            xc = np.linspace(0, 2, n_coarse, endpoint=False)
            f_fine = interp1d(xc, h_c, kind='cubic', fill_value='extrapolate')
            coarse_to_fine.append(np.clip(f_fine(x), 0, 1))

    diverse_starts.extend(coarse_to_fine)

    tau_fine = [0.05, 0.02, 0.01, 0.005]
    for h0 in diverse_starts:
        if elapsed(start_time) > budget_s * 0.80:
            break
        h0 = project_feasible(h0, n)
        for tau in tau_fine:
            if elapsed(start_time) > budget_s * 0.80:
                break
            try:
                res = minimize(
                    obj_with_grad, h0,
                    method='SLSQP',
                    args=(n, tau),
                    jac=True,
                    bounds=bounds,
                    constraints=cons,
                    options={'maxiter': 200, 'ftol': 1e-16}
                )
                h0 = project_feasible(res.x, n)
            except Exception:
                pass
            c5 = evaluate_c5(h0)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h0.copy()

    if best_h is not None and elapsed(start_time) < budget_s * 0.90:
        for round_num in range(10):
            if elapsed(start_time) > budget_s * 0.95:
                break
            for attempt in range(5):
                if elapsed(start_time) > budget_s * 0.95:
                    break
                scale = 0.008 * (1.0 - 0.08 * round_num)
                h_pert = best_h + rng.randn(n) * scale
                h_pert = project_feasible(h_pert, n)
                try:
                    res = minimize(
                        obj_with_grad, h_pert,
                        method='SLSQP',
                        args=(n, 0.001),
                        jac=True,
                        bounds=bounds,
                        constraints=cons,
                        options={'maxiter': 500, 'ftol': 1e-16}
                    )
                    h_pol = project_feasible(res.x, n)
                    c5 = evaluate_c5(h_pol)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_pol.copy()
                except Exception:
                    pass
            scale2 = 0.003 * (1.0 - 0.05 * round_num)
            h_pert2 = best_h + rng.randn(n) * scale2
            h_pert2 = project_feasible(h_pert2, n)
            try:
                res2 = minimize(
                    obj_with_grad, h_pert2,
                    method='SLSQP',
                    args=(n, 0.0005),
                    jac=True,
                    bounds=bounds,
                    constraints=cons,
                    options={'maxiter': 500, 'ftol': 1e-16}
                )
                h_pol2 = project_feasible(res2.x, n)
                c5_2 = evaluate_c5(h_pol2)
                if c5_2 < best_c5:
                    best_c5 = c5_2
                    best_h = h_pol2.copy()
            except Exception:
                pass

    best_h = project_feasible(best_h, n)
    best_c5 = evaluate_c5(best_h)

    return best_h, best_c5, n
