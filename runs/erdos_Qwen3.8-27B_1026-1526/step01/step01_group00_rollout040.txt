```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_gradient_single_lag(h, lag):
    """Compute d(corr_lag)/d(h) for a single lag."""
    n = len(h)
    v = 1.0 - h
    grad = np.zeros(n)
    if lag >= 0:
        span = n - lag
        if span > 0:
            grad[:span] += v[lag:]
            grad[lag:] += -h[:span]
    else:
        la = -lag
        span = n - la
        if span > 0:
            grad[la:] += v[:span]
            grad[:span] += -h[la:]
    return grad

def smooth_max_slsqp_obj(h, n, alpha, top_m=None):
    """Objective for SLSQP: smooth max of correlation values."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    if top_m is not None and top_m < len(corr):
        idx = np.argsort(corr)[-top_m:]
        corr_sel = corr[idx]
    else:
        corr_sel = corr
    m = np.max(corr_sel)
    exp_vals = np.exp(alpha * (corr_sel - m))
    return m + np.log(np.sum(exp_vals)) / alpha

def smooth_max_gradient(h, n, alpha, top_m=None):
    """Compute gradient of smooth max of correlation values."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    if top_m is not None and top_m < len(corr):
        idx = np.argsort(corr)[-top_m:]
        corr_sel = corr[idx]
    else:
        idx = np.arange(len(corr))
        corr_sel = corr
    m = np.max(corr_sel)
    exp_vals = np.exp(alpha * (corr_sel - m))
    w = exp_vals / np.sum(exp_vals)
    
    grad = np.zeros(n)
    for ki in range(len(corr_sel)):
        if w[ki] < 1e-10:
            continue
        lag = idx[ki] - (n - 1)
        grad += w[ki] * corr_gradient_single_lag(h, lag)
    return grad

def optimize_lbfgs(h0, n, target_sum, alpha, top_m, maxiter, time_limit):
    """Optimize using L-BFGS-B with smooth max objective."""
    t0 = time.time()
    
    def obj(h_flat):
        return smooth_max_slsqp_obj(h_flat, n, alpha, top_m)
    
    def grad(h_flat):
        return smooth_max_gradient(h_flat, n, alpha, top_m)
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='L-BFGS-B',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-12, 'maxls': 50})
        elapsed = time.time() - t0
        if elapsed > time_limit:
            return project_to_sum(h0, target_sum)
        return project_to_sum(res.x, target_sum)
    except:
        return project_to_sum(h0, target_sum)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(h, target_sum):
    """Project onto {x: 0<=x<=1, sum(x)=target}."""
    lo, hi = -5.0, 5.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_obj(h, alpha, top_m):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    if top_m and top_m < len(corr):
        idx = np.argpartition(corr, -top_m)[-top_m:]
        cs = corr[idx]
    else:
        cs = corr
    m = cs.max()
    e = np.exp(alpha * (cs - m))
    return m + np.log(e.sum()) / alpha

def smooth_max_grad(h, alpha, top_m):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    if top_m and top_m < len(corr):
        idx = np.argpartition(corr, -top_m)[-top_m:]
        cs = corr[idx]
    else:
        idx = np.arange(len(corr))
        cs = corr
    m = cs.max()
    e = np.exp(alpha * (cs - m))
    w = e / e.sum()
    
    g = np.zeros(n)
    v = 1.0 - h
    for i in range(len(cs)):
        wi = w[i]
        if wi < 1e-12:
            continue
        lag = idx[i] - n + 1
        if lag >= 0:
            s = n - lag
            if s > 0:
                g[:s] += wi * v[lag:]
                g[lag:] -= wi * h[:s]
        else:
            s = n + lag  # n - (-lag)
            if s > 0:
                g[-lag:] += wi * v[:s]
                g[:s] -= wi * h[-lag:]
    return g

def slsqp_run(h0, n, ts, alpha, top_m, maxiter, ftol):
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - ts, 'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(smooth_max_obj, h0.copy(), jac=lambda x: smooth_max_grad(x, alpha, top_m),
                     method='SLSQP', bounds=bnds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        if r.fun < smooth_max_obj(h0, alpha, top_m) + 1e-10:
            return project_box_sum(r.x, ts)
    except Exception:
        pass
    return project_box_sum(h0, ts)

def lbfgs_run(h0, n, ts, alpha, top_m, maxiter, penalty):
    def obj_pen(x):
        return smooth_max_obj(x, alpha, top_m) + penalty * (x.sum() - ts) ** 2
    
    def grad_pen(x):
        return smooth_max_grad(x, alpha, top_m) + 2 * penalty * (x.sum() - ts)
    
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(obj_pen, h0.copy(), jac=grad_pen, method='L-BFGS-B',
                     bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_box_sum(r.x, ts)
    except Exception:
```

```python
        ts_f = n_f / 2.0
        for alpha, topm in [(100000, 2), (200000, 2)]:
            if tl() < 8:
                break
            h_new = slsqp_opt(best_h, alpha, topm, 1000, 1e-14)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    n_final = best_n
    ts_final = n_final / 2.0
    best_h = project_box_sum(best_h, ts_final)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n_final)
```
