def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    lo = -5.0
    hi = 5.0
    for _ in range(200):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    # Fine adjustment
    diff = target - np.sum(result)
    if abs(diff) > 1e-10:
        # Distribute among the most "flexible" coordinates
        flex = np.abs(result - 0.5)
        idx = np.argsort(flex)[-int(abs(diff)*10)+1:]  # most flexible

                result[i] += add
                diff -= add
            else:
                sub = min(-diff, result[i])
                result[i] -= sub
                diff += sub

        grad[valid2] -= h[j[valid2] - shift]

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    lo = -5.0
    hi = 5.0
    for _ in range(200):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-12:
        idx = int(np.argmin(np.abs(result - 0.5)))
        result[idx] = float(np.clip(result[idx] + diff, 0.0, 1.0))
    return result

def obj_smooth_max(h, n, tau):
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    vals = corr / tau
    vmax = np.max(vals)
    smax = tau * (vmax + np.log(np.sum(np.exp(vals - vmax))))
    return smax * dx

def constraint_sum(h, n):
    return np.sum(h) - n / 2.0

def run_slqp(h, n, tau, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_sum, 'args': (n,)}]
    try:
        res = minimize(
            obj_smooth_max, h,
            method='SLSQP',
            args=(n, tau),
            bounds=bounds,
            constraints=cons,
            options={'maxiter': maxiter, 'ftol': ftol}
        )
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def compute_subgradient(h, n, active_frac=0.02):
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    max_val = np.max(corr)
    threshold = max_val * (1.0 - active_frac)
    active = np.where(corr >= threshold)[0]
    grad = np.zeros(n)
    for m in active:
        shift = m - (n - 1)
        j = np.arange(n)
        valid1 = (j + shift >= 0) & (j + shift < n)
        valid2 = (j - shift >= 0) & (j - shift < n)
        grad[valid1] += (1.0 - h[j[valid1] + shift])
        grad[valid2] -= h[j[valid2] - shift]
    grad /= float(len(active))
    return grad

def subgradient_method(h, n, n_iter, lr_init, rng):
    best = h.copy()
    best_val = evaluate_c5(h)
    current = h.copy()
    stall_count = 0
    for t in range(n_iter):
        lr = lr_init / (1.0 + 0.003 * t)
        grad = compute_subgradient(current, n)
        current = current - lr * grad
        current = project_feasible(current, n)
        val = evaluate_c5(current)
        if val < best_val:
            best_val = val
            best = current.copy()
            stall_count = 0
        else:
            stall_count += 1
        if stall_count > 2000:
            current = best + rng.randn(n) * 0.01
            current = project_feasible(current, n)
            stall_count = 0
    return best, best_val

def simulated_annealing(h, n, n_iter, rng, T_init=0.03):
    best = h.copy()
    best_val = evaluate_c5(h)
    current = h.copy()
    current_val = best_val
    for t in range(n_iter):
        frac = t / max(n_iter, 1)
        T = T_init * (1.0 - frac) ** 2
        i = int(rng.randint(n))
        delta = float(rng.uniform(-0.08, 0.08))
        trial = current.copy()
        trial[i] = float(np.clip(trial[i] + delta, 0.0, 1.0))
        trial = project_feasible(trial, n)
        trial_val = evaluate_c5(trial)
        if trial_val <= current_val:
            current = trial
            current_val = trial_val
            if trial_val < best_val:
                best_val = trial_val
                best = trial.copy()
        elif T > 1e-6 and rng.random() < np.exp(-(trial_val - current_val) / T):
            current = trial
            current_val = trial_val
    return best, best_val

def coordinate_polish(h, n, n_passes, rng):
    best = h.copy()
    best_val = evaluate_c5(h)
    idx_order = np.arange(n)
    for p in range(n_passes):
        np.random.seed(p + 100)
        np.random.shuffle(idx_order)
        for i in idx_order:
            current_val_i = best[i]
            candidates = np.linspace(max(0.0, current_val_i - 0.12), min(1.0, current_val_i + 0.12), 25)
            best_t = current_val_i
            best_obj = best_val
            for t in candidates:
                trial = best.copy()
                trial[i] = t
                trial = project_feasible(trial, n)
                obj = evaluate_c5(trial)
                if obj < best_obj - 1e-10:
                    best_obj = obj
                    best_t = t
            if best_t != current_val_i:
                best[i] = best_t
                best = project_feasible(best, n)
                best_val = evaluate_c5(best)
    return best, best_val
