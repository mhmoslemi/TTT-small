            if diff > 0:
                room = 1.0 - result[i]
                add = min(room, diff)
                result[i] += add
                diff -= add
            else:
                room = result[i]
                sub = min(room, -diff)
                result[i] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return result

def compute_lag_gradient(h, lag, n):
    """Compute gradient of corr[lag] w.r.t. h"""
    dx = 2.0 / n
    grad = np.zeros(n)
    if lag >= 0:
        for i in range(lag, n):
            grad[i] += (1.0 - h[i - lag])
        for i in range(0, n - lag):
            grad[i] -= h[i + lag]
    else:
        a = -lag
        for i in range(0, n - a):
            grad[i] += (1.0 - h[i + a])
        for i in range(a, n):
            grad[i] -= h[i - a]
    return grad * dx

def compute_lag_gradient(h, lag, n):

    if lag >= 0:

        grad[:n-lag] -= h[lag:]
    else:
        a = -lag
        grad[:n-a] += (1.0 - h[a:])
        grad[a:] -= h[:n-a]
    return grad * dx

This is much faster. Let me verify the logic:
- corr[k] = sum_j h[j+k] * (1-h[j]) for valid j
- lag = k - (n-1) in the np.correlate indexing

Wait, I need to be careful. In np.correlate(h, g, mode="full"):
- The output has length 2n-1
- Output index m corresponds to: sum_j h[j + (m - (n-1))] * g[j] where indices are valid
- So if m is the argmax, lag = m - (n-1)

For lag = 0: corr = sum_j h[j] * (1-h[j]) (autocorrelation-like)
For lag = 1: corr = sum_j h[j+1] * (1-h[j]) for j=0..n-2

∂(corr for lag=1)/∂h[i]:
- h[i] appears as h[j+1] when j = i-1, contributing (1-h[i-1]) if i-1 >= 0, i.e., i >= 1
- h[i] appears as h[j] in (1-h[j]) when j = i, contributing -h[i+1] if i+1 < n, i.e., i < n-1

So grad[i] = (1-h[i-1]) * [i>=1] - h[i+1] * [i<n-1]

For lag = 1:
- grad[lag:] = grad[1:] += (1-h[0:n-1]) → grad[i] += (1-h[i-1]) for i = 1..n-1 ✓
- grad[:n-1] -= h[1:] → grad[i] -= h[i+1] for i = 0..n-2 ✓

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(values, tau):
    v = values / tau
    vmax = np.max(v)
    return tau * (vmax + np.log(np.sum(np.exp(v - vmax))))

def project_feasible(h, n):
    target = n / 2.0
    lo = -3.0
    hi = 3.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-14:
        idx = np.argsort(np.abs(result - 0.5))
        for i in idx:
            if diff > 0:
                room = 1.0 - result[i]
                add = min(room, diff)
                result[i] += add
                diff -= add
            else:
                room = result[i]
                sub = min(room, -diff)
                result[i] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break
    return result

        grad[a:] -= h[:n - a]
    return grad * dx

    best_sub = c5_curr
    best_h_sub = h_curr.copy()

    stagnant = 0

    while time.time() < deadline:
        dx = 2.0 / n
        corr = np.correlate(h_curr, 1.0 - h_curr, mode="full")
        k_star = int(np.argmax(corr))
        lag = k_star - (n - 1)
        grad = compute_lag_gradient(h_curr, lag, n)
        h_new = project_feasible(h_curr - step * grad, n)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_sub - 1e-9:

            stagnant += 1
            if stagnant > 300:
                step *= 0.85
                stagnant = 0
                if step < 1e-5:
                    break
        h_curr = h_new
    return best_h_sub, best_sub

def multi_lag_subgradient_refine(h, n, max_time, rng, top_k=3):
    h_curr = h.copy()
    c5_curr = evaluate_c5(h_curr)
    best_sub = c5_curr
    best_h_sub = h_curr.copy()
    step = 0.03
    stagnant = 0

        dx = 2.0 / n
        corr = np.correlate(h_curr, 1.0 - h_curr, mode="full")
        max_val = np.max(corr)
        median_val = np.median(corr)
        top_idx = np.argsort(corr)[-top_k:]
        weights = np.zeros(2 * n - 1)
        for ti in top_idx:
