from scipy.interpolate import interp1d

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    lo = -3.0
    hi = 3.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-16:
        idx = int(np.argmin(np.abs(result - 0.5)))
        result[idx] = np.clip(result[idx] + diff, 0.0, 1.0)
    return result

def smooth_max(values, tau):
    v = values / tau
    vmax = np.max(v)
    return tau * (vmax + np.log(np.sum(np.exp(v - vmax))))

def smooth_obj(h, n, tau):
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return smooth_max(corr, tau) * dx

def sum_eq(x, n):
    return np.sum(x[:n]) - n / 2.0

def mm_obj(x, n):
    return x[-1]

def sum_eq_ext(x, n):
    return np.sum(x[:n]) - n / 2.0

def ineq_ext(x, n, active_k):
    h = x[:n]
    t = x[-1]
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return t - corr[active_k] * dx

def get_active_lags(h, n, num_active):
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    cs = corr * dx
    top = np.argsort(cs)[-num_active:]
    return np.sort(top.astype(int))

def resample_h(h, new_n):
    if len(h) == new_n:
        return h.copy()
    x_old = np.linspace(0, 2, len(h), endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    return np.clip(f(x_new), 0, 1)

def optimize_smooth(h0, n, taus, time_limit):
    h = project_feasible(h0, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq, 'args': (n,)}]
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    for tau in taus:
        if time.time() > time_limit:
            break
        try:
            res = minimize(smooth_obj, h, method='SLSQP', args=(n, tau),
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 400, 'ftol': 1e-15})
            if res.x is not None and np.all(np.isfinite(res.x)):
                h = project_feasible(res.x, n)
                c5 = evaluate_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
        except Exception:
            pass
    return best_h, best_c5

def optimize_active(h0, n, time_limit):
    h = project_feasible(h0, n)
    c5_init = evaluate_c5(h)
    x0 = np.concatenate([h, [c5_init + 0.01]])
    bounds = [(0.0, 1.0)] * n + [(-1.0, 2.0)]
    best_h = h.copy()
    best_c5 = c5_init
    active_count = 6
    for outer in range(40):
        if time.time() > time_limit:
            break
        active_k = get_active_lags(h, n, active_count)
        cons = [
            {'type': 'ineq', 'fun': ineq_ext, 'args': (n, active_k)},
            {'type': 'eq', 'fun': sum_eq_ext, 'args': (n,)}
        ]
        try:
            res = minimize(mm_obj, x0, method='SLSQP', args=(n,),
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 200, 'ftol': 1e-14})
            if res.x is not None and np.all(np.isfinite(res.x)):
                h_new = project_feasible(res.x[:n], n)
                c5_new = evaluate_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h = h_new.copy()
                x0 = np.concatenate([h, [best_c5 + 0.003]])
                if outer % 4 == 3 and active_count < 22:
                    active_count += 2
            else:
                break
        except Exception:
            break
    return best_h, best_c5

def coordinate_sweep(h, n, time_limit, num_sweeps=10):
    h = project_feasible(h, n.copy() if isinstance(n, np.ndarray) else h, n) if False else project_feasible(h, n)
    best_c5 = evaluate_c5(h)
    for sweep in range(num_sweeps):
        if time.time() > time_limit:
            break
        improved = False
        for j in range(n):
            if time.time() > time_limit:
                break
            for l in range(j + 1, n):
                c = h[j] + h[l]
                lo = max(0.0, c - 1.0)
                hi = min(1.0, c)
                if hi - lo < 1e-12:
                    continue
                a_lo = lo
                a_hi = hi
                for _ in range(60):
                    a1 = a_lo + (a_hi - a_lo) / 3.0
                    a2 = a_hi - (a_hi - a_lo) / 3.0
                    h[j] = a1
                    h[l] = c - a1
                    f1 = evaluate_c5(h)
                    h[j] = a2
                    h[l] = c - a2
                    f2 = evaluate_c5(h)
                    if f1 < f2:
                        a_hi = a2
                    else:
                        a_lo = a1
                a_opt = (a_lo + a_hi) / 2.0
                h[j] = a_opt
                h[l] = c - a_opt
                c5_now = evaluate_c5(h)
                if c5_now < best_c5:
                    best_c5 = c5_now
                    improved = True
        if not improved:
            break
    return h, best_c5

def generate_starts(n, parent_h, rng):
    starts = []
    x = np.linspace(0, 2, n, endpoint=False)
    if parent_h is not None:
        if len(parent_h) == n:
            starts.append(parent_h.copy())
            starts.append(np.clip(parent_h + rng.randn(n) * 0.03, 0, 1))
            starts.append(np.clip(parent_h + rng.randn(n) * 0.08, 0, 1))
            starts.append(np.clip(parent_h + rng.randn(n) * 0.15, 0, 1))
        else:
            hr = resample_h(parent_h, n)
            starts.append(hr)
            starts.append(np.clip(hr + rng.randn(n) * 0.05, 0, 1))
            starts.append(np.clip(hr + rng.randn(n) * 0.1, 0, 1))
    freqs = [1.3, 1.7, 2.1, 2.7, 3.3, 4.1, 5.3]
    for freq in freqs:
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.25, 0.45)
        starts.append(np.clip(0.5 + amp * np.sin(2 * np.pi * freq * x + phase), 0, 1))
    for _ in range(4):
        starts.append(rng.rand(n))
    ar = np.arange(n)
    for _ in range(2):
        center = rng.uniform(0.2 * n, 0.8 * n)
        width = rng.uniform(0.1 * n, 0.35 * n)
        bump = np.exp(-0.5 * ((ar - center) / width) ** 2)
        starts.append(np.clip(0.2 + 0.7 * bump, 0, 1))
    return starts

    t_phase3 = start_time + budget_s * 0.95

    coarse_taus = [0.1, 0.06, 0.04, 0.025, 0.015, 0.008]
    fine_taus = [0.005, 0.003, 0.0015, 0.0008, 0.0004]
    ultra_taus = [0.0002, 0.0001, 0.00005]

    # Phase 1: Explore with coarse smooth max on multiple n
    for n in n_list:
        if time.time() > t_phase1:
            break
        starts = generate_starts(n, parent_h, rng)
        time_per = max(3.0, (t_phase1 - time.time()) / (len(starts) * max(1, len(n_list) - n_list.index(n))))

            h_opt, c5_opt = optimize_smooth(h0, n, coarse_taus, t_phase1)

        h5, c5_5 = optimize_smooth(best_h, n, ultra_taus, t_phase3)
        if c5_5 < best_c5:
            best_c5 = c5_5

        h6, c5_6 = optimize_active(best_h, n, t_phase3)
