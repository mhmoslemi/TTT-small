    g = 1.0 - h

    vmax = np.max(v)
    return tau * (vmax + np.log(np.sum(np.exp(v - vmax))))

    if parent_h is not None:
        parent_h = project_feasible(parent_h, len(parent_h))
        parent_c5 = evaluate_c5(parent_h)
        best_c5 = parent_c5
        best_h = parent_h.copy()
    else:
        n = 59
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c5 = evaluate_c5(best_h)
    
    n_parent = len(best_h)
    
    grid_sizes = [n_parent]
    for nn in range(n_parent - 8, n_parent + 9):
        if nn > 10 and nn != n_parent:
            grid_sizes.append(nn)
    grid_sizes = sorted(set(grid_sizes))
    
    for n in grid_sizes:
        if time.time() - start_time > budget_s * 0.55:
            break
        
        if best_h is not None and len(best_h) == n:
            h0 = best_h.copy()
        elif best_h is not None:
            xp = np.linspace(0, 2, len(best_h), endpoint=False)
            x = np.linspace(0, 2, n, endpoint=False)
            f = interp1d(xp, best_h, kind='cubic', fill_value='extrapolate')
            h0 = np.clip(f(x), 0, 1)
            h0 = project_feasible(h0, n)
        else:
            h0 = project_feasible(np.full(n, 0.5), n)
        
        x = np.linspace(0, 2, n, endpoint=False)
        starts = [h0]
        
        if parent_h is not None and n != len(parent_h):
            xp = np.linspace(0, 2, len(parent_h), endpoint=False)
            fp = interp1d(xp, parent_h, kind='cubic', fill_value='extrapolate')
            starts.append(project_feasible(np.clip(fp(x), 0, 1), n))
        
        for freq in [1.5, 2.0, 2.5, 3.0, 3.5]:
            for phase in [0.0, 1.0, 2.0]:
                hs = 0.5 + 0.45 * np.sin(2.0 * np.pi * freq * x + phase)
                starts.append(project_feasible(np.clip(hs, 0, 1), n))
        
        for _ in range(3):
            starts.append(project_feasible(rng.rand(n), n))
        
        for h0 in starts:
            if time.time() - start_time > budget_s * 0.55:
                break
            
            current_h = h0.copy()
            
            for tau in [0.1, 0.05, 0.02, 0.01, 0.005]:
                if time.time() - start_time > budget_s * 0.55:
                    break
                current_h = slsqp_refine(current_h, n, tau, maxiter=100, ftol=1e-10)
                c5 = evaluate_c5(current_h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = current_h.copy()
            
            current_h, _ = coord_descent(current_h, n, max_iter=300, rng=rng)
            c5 = evaluate_c5(current_h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = current_h.copy()
            
            current_h, _ = swap_descent(current_h, n, max_iter=200, rng=rng)
            c5 = evaluate_c5(current_h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = current_h.copy()
    
    n = len(best_h)
    
    for tau in [0.003, 0.001, 0.0005, 0.0002, 0.0001]:
        if time.time() - start_time > budget_s * 0.90:
            break
        for attempt in range(6):
            if time.time() - start_time > budget_s * 0.90:
                break
            noise = rng.randn(n) * (0.005 + 0.005 * attempt)
            perturb = project_feasible(best_h + noise, n)
            current_h = slsqp_refine(perturb, n, tau, maxiter=300, ftol=1e-14)
            c5 = evaluate_c5(current_h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = current_h.copy()
            
            current_h, _ = coord_descent(best_h, n, max_iter=500, rng=rng)
            c5 = evaluate_c5(current_h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = current_h.copy()
    
    while time.time() - start_time < budget_s * 0.97:
        n = len(best_h)
        current_h, _ = coord_descent(best_h, n, max_iter=1000, rng=rng)
        c5 = evaluate_c5(current_h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = current_h.copy()
        else:
            break
        current_h, _ = swap_descent(best_h, n, max_iter=500, rng=rng)
        c5 = evaluate_c5(current_h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = current_h.copy()
    
    return best_h, best_c5, len(best_h)
