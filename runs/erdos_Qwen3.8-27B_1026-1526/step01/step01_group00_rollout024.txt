    lr0 = 0.1

    max_iters = 5000

        if time.time() - t_start > time_limit - 30:
            break
        
        # Interpolate alpha
        t_frac = it / max(1, n_iters)
        alpha = alpha_start * (alpha_end / alpha_start) ** t_frac

        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_circular_delta(h):
    """Compute δ_k = sum_i h[i](1-h[(i+k)%n]) for all k using FFT."""
    n = len(h)
    H = np.fft.rfft(h)
    circ_corr = np.fft.irfft(np.abs(H)**2)
    return np.sum(h) - circ_corr

def subgradient_optimize(h0, target_sum, t_start, time_limit, alpha_start=5.0, alpha_end=500.0, lr0=0.05, momentum=0.9):
    """Optimize using subgradient method with momentum and adaptive learning rate."""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n = len(h)
    v = np.zeros(n)  # momentum
    
    while time.time() < time_limit:
        # Adaptive alpha
        # Use time-based scheduling
        elapsed = time.time() - t_start
        remaining = time_limit - elapsed
        frac = 1.0 - remaining / (time_limit - t_start) if time_limit > t_start else 0
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
