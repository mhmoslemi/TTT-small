import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_c5_parts(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr, dx

def proj_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_grad(h, m):
    n = len(h)
    L = m - (n - 1)
    g = np.zeros(n)
    if L == 0:
        g = 1.0 - 2.0 * h
    elif L > 0:
        valid1 = np.arange(n) <= (n - 1 - L)
        cnt1 = np.sum(valid1)
        if cnt1 > 0:
            g[valid1] += (1.0 - h[L:L + cnt1])
        valid2 = np.arange(n) >= L
        cnt2 = np.sum(valid2)
        if cnt2 > 0:
            g[valid2] -= h[:cnt2]
    else:
        M = -L
        valid1 = np.arange(n) <= (n - 1 - M)
        cnt1 = np.sum(valid1)
        if cnt1 > 0:
            g[valid1] += (1.0 - h[M:M + cnt1])
        valid2 = np.arange(n) >= M
        cnt2 = np.sum(valid2)
        if cnt2 > 0:
            g[valid2] -= h[:cnt2]
    return g

def subgrad_descent(h0, n, ts, max_iters, lr_init=0.01, lr_final=0.0005, momentum=0.85, topk=5):
    h = h0.copy()
    v = np.zeros(n)
    c5_best = compute_c5(h)
    h_best = h.copy()
    best_iter = 0
    no_improve = 0
    
    for it in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode="full")
        sorted_idx = np.argsort(corr)[::-1]
        top_indices = sorted_idx[:topk]
        
        max_val = corr[sorted_idx[0]]
        weights = np.zeros(len(top_indices))
        for i2, idx in enumerate(top_indices):
            weights[i2] = np.exp(50.0 * (corr[idx] - max_val))
        weights /= np.sum(weights)
        
        g_total = np.zeros(n)
        for w, idx in zip(weights, top_indices):
            g_total += w * corr_grad(h, idx)
        
        lr = lr_init * (lr_final / lr_init) ** (float(it) / max(1, max_iters))
        v = momentum * v - lr * g_total
        h_new = proj_sum(h + v, ts)
        
        c5_new = compute_c5(h_new)
        if c5_new < c5_best - 1e-14:
            c5_best = c5_new
            h_best = h_new.copy()
            best_iter = it
            no_improve = 0
        else:
            no_improve += 1
        
        h = h_new
        if no_improve > 500:
            break
    
    return h_best, c5_best

def binary_push(h, ts, n_iter=200):
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    for it in range(n_iter):
        h_bin = (h_cur > 0.5).astype(float)
        h_bin = proj_sum(h_bin, ts)
        c5_bin = compute_c5(h_bin)
        if c5_bin < c5_cur:
            h_cur = h_bin
            c5_cur = c5_bin
        else:
            break
    return h_cur, c5_cur

def coordinate_refine(h0, n, ts, rng, max_rounds=100, deltas=None):
    if deltas is None:
        deltas = [0.3, -0.3, 0.1, -0.1, 0.05, -0.05, 0.02, -0.02, 0.01, -0.01, 0.5, -0.5]
    h = h0.copy()
    c5_cur = compute_c5(h)
    for rnd in range(max_rounds):
        improved = False
        idx_order = rng.permutation(n)
        for i in idx_order:
            best_d = 0.0
            best_c = c5_cur
            for d in deltas:
                h_try = h.copy()
                h_try[i] = np.clip(h_try[i] + d, 0.0, 1.0)
                h_try = proj_sum(h_try, ts)
                c_try = compute_c5(h_try)
                if c_try < best_c - 1e-14:
                    best_c = c_try
                    best_d = d
            if best_d != 0.0:
                h[i] = np.clip(h[i] + best_d, 0.0, 1.0)
                h = proj_sum(h, ts)
                c5_cur = compute_c5(h)
                improved = True
        if not improved:
            break
    return h, c5_cur

def slsqp_smooth(h0, n, ts, alpha, maxiter=300):
    def obj(x):
        hc = np.clip(x, 0.0, 1.0)
        corr = np.correlate(hc, 1.0 - hc, mode="full")
        m = np.max(corr)
        ev = np.exp(alpha * (corr - m))
        return m + np.log(np.sum(ev) / len(ev)) / alpha
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ts, 'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
        return proj_sum(res.x, ts)
    except:
        return h0.copy()
