    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)

                h_pert = best_h + scale * rng.standard_normal(n_best)
            elif r < 0.75:
                h_pert = best_h.copy()
                n_push = max(3, int(n_best * 0.15))
                push_idx = rng.choice(n_best, min(n_push, n_best), replace=False)
                for idx in push_idx:
                    if rng.random() < 0.5:
                        h_pert[idx] = min(1.0, h_pert[idx] + scale * 3.0)
                    else:
                        h_pert[idx] = max(0.0, h_pert[idx] - scale * 3.0)
            else:
                h_pert = best_h + scale * 0.5 * rng.standard_normal(n_best)
                h_pert = project_feasible(h_pert, n_best)
                T_cur = 50.0 + 100.0 * frac
                h_new, c5_new = lbfgsb_optimize(h_pert, n_best, T=T_cur, penalty=100.0 + 400.0 * frac, maxiter=1500)
                if c5_new < best_c5 - 1e-10:
                    best_h = h_new
                    best_c5 = c5_new
                continue

            h_pert = project_feasible(h_pert, n_best)
            if attempt % 3 == 0:
                h_new, c5_new = slsqp_optimize(h_pert, n_best, maxiter=2500, t_margin=0.05)
            else:
                h_new, c5_new = slsqp_optimize(h_pert, n_best, maxiter=1500, t_margin=0.05)

            if c5_new < best_c5 - 1e-10:
                best_h = h_new
                best_c5 = c5_new

    for n_try in [100, 150, 200, 270]:
        if time.time() - t0 > budget_s - 100:
            break
        n_alt = n_try
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_alt, endpoint=False)
        f_interp = interp1d(x_old, best_h, kind='linear', fill_value="extrapolate")
        h_interp = np.clip(f_interp(x_new), 0, 1)
        h_interp = project_feasible(h_interp, n_alt)

        h_new, c5_new = slsqp_optimize(h_interp, n_alt, maxiter=4000, t_margin=0.1)
        if c5_new < best_c5 - 1e-10:
            best_h = h_new
            best_c5 = c5_new
            n_best = n_alt

    if time.time() - t0 < budget_s - 20:
        for attempt in range(60):
            if time.time() - t0 > budget_s - 8:
                break
            scale = 0.003 + 0.015 * (1 - attempt / 60) ** 2
            h_pert = best_h + scale * rng.standard_normal(n_best)
            h_pert = project_feasible(h_pert, n_best)
            h_new, c5_new = slsqp_optimize(h_pert, n_best, maxiter=2000, t_margin=0.02)
            if c5_new < best_c5 - 1e-12:
                best_h = h_new
                best_c5 = c5_new

    if time.time() - t0 < budget_s - 3:
        best_h, best_c5 = slsqp_optimize(best_h, n_best, maxiter=8000, t_margin=0.01)

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5_from_h(best_h)

    return best_h, best_c5, len(best_h)
