```python
    """Compute O(k) values and optionally gradient of O(k)"""
    n = len(h)
    dx = 2.0 / n

    # c[n-1+k] for k=0,...,n-1 gives O(k)/dx
    O = c[n-1:n-1+n] * dx  # length n
```

Let me use a different approach: since the constraint set {h: 0 <= h <= 1, sum(h) = n/2} is a convex polytope, I can use projected gradient descent where the projection is onto this polytope.

The projection onto {h: 0 <= h <= 1, sum(h) = n/2} can be done efficiently:
1. Without the sum constraint, project onto [0,1]^n: clip to [0,1]
2. With the sum constraint, use a water-filling / bisection approach:
   Find tau such that sum(clip(h - tau, 0, 1)) = n/2

This is the standard projection onto the intersection of the simplex-like constraint and the box.

Let me code this up:

```python
def project_to_feasible(h, n):
    """Project h onto {0 <= h <= 1, sum(h) = n/2}"""
    target = n / 2.0
    # Bisection on tau
    # We want sum(clip(h - tau, 0, 1)) = target
    lo, hi = h.min() - 1, h.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2, 0, 1)
```

For the smooth max optimization, I'll use scipy's L-BFGS-B with analytic gradient. The objective is:

The gradient of the first term:

For each k, ∇O_k[j] = dx * [(1-h[j-k])*I(j>=k) - h[j+k]*I(j<n-k)]

```python
import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time


def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def project_feasible(h, n):
    """Project h onto {0 <= h <= 1, sum(h) = n/2} using bisection on tau"""
    target = n / 2.0
    h = np.array(h, dtype=float)
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def compute_O(h):
    """Compute O(k) = c[n-1+k]*dx for k=0,...,n-1"""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return c[n-1:n-1+n] * dx


def smooth_max_objective(h, beta, gamma):
    n = len(h)
    O = compute_O(h)
    # Log-sum-exp for numerical stability
    O_max = np.max(O)
    log_sum = O_max + np.log(np.sum(np.exp(beta * (O - O_max))))
    f = log_sum / beta
    # Penalty for constraint violation
    constraint_violation = np.sum(h) - n / 2.0
    f += gamma * constraint_violation ** 2
    return f


def smooth_max_gradient(h, beta, gamma):
    n = len(h)
    dx = 2.0 / n
    O = compute_O(h)
    # Softmax weights
    O_shifted = beta * (O - np.max(O))
    p = np.exp(O_shifted)
    p /= np.sum(p)
    
    # Gradient of smooth max = sum_k p_k * grad_O_k
    grad = np.zeros(n)
    for k in range(n):
        if k == 0:
            g_k = dx * (1.0 - 2.0 * h)
        else:
            g_k = np.zeros(n)
            g_k[k:] += dx * (1.0 - h[:n-k])
            g_k[:n-k] -= dx * h[k:]
        grad += p[k] * g_k
    
    # Gradient of penalty
    grad += 2.0 * gamma * (np.sum(h) - n / 2.0)
    return grad


def smooth_max_optimize(h_init, n, maxiter=500, start_beta=10.0, end_beta=200.0, gamma_start=1.0, gamma_end=100.0):
    h = h_init.copy()
    n_iter = maxiter
    for i in range(n_iter):
        frac = i / max(1, n_iter - 1)
        beta = start_beta * (end_beta / start_beta) ** frac
        gamma = gamma_start * (gamma_end / gamma_start) ** frac
        
        result = minimize(
            smooth_max_objective, h, args=(beta, gamma),
            jac=smooth_max_gradient, method='L-BFGS-B',
            bounds=[(0.0, 1.0)] * n,
            options={'maxiter': 20, 'ftol': 1e-14, 'gtol': 1e-8}
        )
        h = result.x
        h = np.clip(h, 0.0, 1.0)
    
    # Final refinement with hard max
    h = project_feasible(h, n)
    return h


def subgradient_step(h, step_size):
    """One subgradient step for the hard max"""
    n = len(h)
    O = compute_O(h)
    k_star = np.argmax(O)
    if k_star == 0:
        grad = (2.0 / n) * (1.0 - 2.0 * h)
    else:
        grad = np.zeros(n)
        grad[k_star:] += (2.0 / n) * (1.0 - h[:n-k_star])
        grad[:n-k_star] -= (2.0 / n) * h[k_star:]
    h_new = h - step_size * grad
    return project_feasible(h_new, n)


def subgradient_refine(h, n, maxiter=200, step_start=0.1, step_end=0.01):
    h = h.copy()
    best_c5 = compute_c5_from_h(h)
    best_h = h.copy()
    
    for i in range(maxiter):
        frac = i / max(1, maxiter - 1)
        step = step_start * (step_end / step_start) ** frac
        h_new = subgradient_step(h, step)
        c5_new = compute_c5_from_h(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    
    best_h = project_feasible(best_h, n)
    return best_h, compute_c5_from_h(best_h)


def upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    from scipy.interpolate import interp1d
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = np.clip(f(x_new), 0.0, 1.0)
    return project_feasible(h_new, n_new)

    best_h = h0.copy()

    best_n = len(best_h)
    
    # Phase 1: Upsample and optimize with smooth max
    for n_new in [270, 405]:
        if time.time() - t0 > budget_s - 300:

        h_up = upsample(best_h, n_new)
        h_opt = smooth_max_optimize(h_up, n_new, maxiter=300, start_beta=5.0, end_beta=100.0)
        h_opt = project_feasible(h_opt, n_new)
        c5_opt = compute_c5_from_h(h_opt)
        
        # Subgradient refinement
        h_ref, c5_ref = subgradient_refine(h_opt, n_new, maxiter=100)
        if c5_ref < c5_opt:
            h_opt = h_ref
            c5_opt = c5_ref
        
        if c5_opt < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5_opt
            best_n = n_new
    
    # Phase 2: Subgradient refinement on best solution
    if time.time() - t0 < budget_s - 60:
        h_ref, c5_ref = subgradient_refine(best_h, best_n, maxiter=300, step_start=0.05, step_end=0.002)
        if c5_ref < best_c5:
            best_h = h_ref
            best_c5 = c5_ref

    if time.time() - t0 < budget_s - 60:

            if c5_ref < best_c5:
```

```python
def smooth_max_gradient_fast(h, beta, gamma):
    n = len(h)
    dx = 2.0 / n
    O = compute_O(h)
    
    # Softmax weights
    O_max = np.max(O)
    log_p = beta * (O - O_max)
    log_p -= np.max(log_p)
    p = np.exp(log_p)
    p /= np.sum(p)
    
    # Compute gradient using convolution
    cumsum_p = np.cumsum(p)
    conv_ph = np.convolve(p, h, mode='full')[:n]  # first n elements
    h_rev = h[::-1]
    conv_ph_rev = np.convolve(p, h_rev, mode='full')
    
    # Second sum: conv_ph_rev[n-1-j] for j=0,...,n-1
    # We need indices n-1, n-2, ..., 0 of conv_ph_rev
    second_sum = conv_ph_rev[n-1::-1]  # index n-1, n-2, ..., 0
    
    grad = dx * (cumsum_p - conv_ph - second_sum)
    
    # Add penalty gradient
    grad += 2.0 * gamma * (np.sum(h) - n / 2.0)
    
    return grad
```

```python
import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time


def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float)
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def compute_O(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    # By symmetry, max is in c[n-1:]
    return c[n-1:n-1+n] * dx


def smooth_objective(h, beta, gamma):
    n = len(h)
    O = compute_O(h)
    O_max = np.max(O)
    f = O_max + np.log(np.sum(np.exp(beta * (O - O_max)))) / beta
    f += gamma * (np.sum(h) - n / 2.0) ** 2
    return f


def smooth_gradient(h, beta, gamma):
    n = len(h)
    dx = 2.0 / n
    O = compute_O(h)
    O_max = np.max(O)
    log_p = beta * (O - O_max)
    log_p -= np.max(log_p)
    p = np.exp(log_p)
    p /= np.sum(p)
    
    cumsum_p = np.cumsum(p)
    conv_ph = np.convolve(p, h, mode='full')[:n]
    h_rev = h[::-1]
    conv_ph_rev = np.convolve(p, h_rev, mode='full')[n-1::-1]
    
    grad = dx * (cumsum_p - conv_ph - conv_ph_rev)
    grad += 2.0 * gamma * (np.sum(h) - n / 2.0)
    return grad


def smooth_optimize(h_init, n, tau):
    """Run smooth max optimization with annealing. tau is time budget in seconds."""
    t_start = time.time()
    h = h_init.copy()
    
    n_levels = 50
    betas = np.geomspace(5.0, 300.0, n_levels)
    gammas = np.geomspace(1.0, 50.0, n_levels)
    
    for i in range(n_levels):
        if time.time() - t_start > tau:
            break
        beta = betas[i]
        gamma = gammas[i]
        result = minimize(
            smooth_objective, h, args=(beta, gamma),
            jac=smooth_gradient, method='L-BFGS-B',
            bounds=[(0.0, 1.0)] * n,
            options={'maxiter': 30, 'ftol': 1e-15, 'gtol': 1e-10}
        )
        h = result.x
        h = np.clip(h, 0.0, 1.0)
    
    return project_feasible(h, n)


def subgradient_refine(h, n, maxiter, step_start, step_end):
    h = h.copy()
    best_c5 = compute_c5_from_h(h)
    best_h = h.copy()
    
    for i in range(maxiter):
        frac = i / max(1, maxiter - 1)
        step = step_start * (step_end / step_start) ** frac
        O = compute_O(h)
        k_star = int(np.argmax(O))
        
        if k_star == 0:
            grad = (2.0 / n) * (1.0 - 2.0 * h)
        else:
            grad = np.zeros(n)
            grad[k_star:] += (2.0 / n) * (1.0 - h[:n-k_star])
            grad[:n-k_star] -= (2.0 / n) * h[k_star:]
        
        h_new = h - step * grad
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5_from_h(h_new)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    
    best_h = project_feasible(best_h, n)
    return best_h, compute_c5_from_h(best_h)


def upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = np.clip(f(x_new), 0.0, 1.0)
    return project_feasible(h_new, n_new)


def run(seed=42, budget_s=1300, **kwargs):

    rng = np.random.default_rng(seed)

            best_n = n1
    
    # Phase 2: Upsample to 405 and optimize
    if time.time() - t0 < budget_s - 200:

        remaining = budget_s - 200 - (time.time() - t0)
        h2 = upsample(best_h, n2)
        h2 = smooth_optimize(h2, n2, tau=remaining * 0.6)
        h2, c5_2 = subgradient_refine(h2, n2, maxiter=200, step_start=0.04, step_end=0.001)

            best_h = h2
            best_c5 = c5_2
            best_n = n2
    
    # Phase 3: Iterated refinement
    if time.time() - t0 < budget_s - 60:
        for _ in range(3):
            if time.time() - t0 > budget_s - 40:
                break
            remaining = budget_s - 40 - (time.time() - t0)

            h_ref, c5_ref = subgradient_refine(h_ref, best_n, maxiter=150, step_start=0.02, step_end=0.0005)

                best_c5 = c5_ref
    
    # Phase 4: Random restarts
    if time.time() - t0 < budget_s - 60:
        n_restarts = 15
        for ri in range(n_restarts):

            # Diverse initial conditions
            mode = ri % 3

    best_h = project_feasible(best_h, best_n)

    return best_h, best_c5, best_n
```

∇O(k)[j] = dx * [(1-h[j-k]) * I(j >= k) - h[j+k] * I(j < n-k)]

So:
∇f(h)[j] = dx * sum_k p_k * [(1-h[j-k]) * I(j >= k) - h[j+k] * I(j < n-k)]
= dx * [sum_{k <= j} p_k * (1-h[j-k]) - sum_{k < n-j} p_k * h[j+k]]

The first sum: sum_{k=0}^{j} p_k * (1-h[j-k]) = sum_{k=0}^{j} p_k - sum_{k=0}^{j} p_k * h[j-k]
= cumsum_p[j] - convolve(p, h, 'full')[j]

The second sum: sum_{k=0}^{n-1-j} p_k * h[j+k] = convolve(p, h[::-1], 'full')[n-1-j]

So:
∇f(h)[j] = dx * (cumsum_p[j] - convolve(p, h, 'full')[j] - convolve(p, h[::-1], 'full')[n-1-j])

This matches what I had before. Good.

Let me also add a check: for k=0, the formula should give:
∇O(0)[j] = dx * (1 - 2*h[j])

Using my formula with k=0:
Term 1: (1-h[j-0]) * I(j >= 0) = (1-h[j]) (always true for j >= 0)
Term 2: h[j+0] * I(j < n-0) = h[j] (always true for j < n)

So ∇O(0)[j] = dx * ((1-h[j]) - h[j]) = dx * (1-2*h[j]) ✓

But in my convolution formula, for k=0:
The first sum includes k=0: p_0 * (1-h[j])
The second sum includes k=0: p_0 * h[j]

So the contribution from k=0 to the gradient is:
dx * p_0 * [(1-h[j]) - h[j]] = dx * p_0 * (1-2*h[j]) ✓

Great, the formula handles k=0 correctly as part of the sum over all k.

Now let me finalize the code. I want to make sure it's efficient and robust.

One more optimization: instead of calling `compute_O` (which does a full correlate) in both the objective and gradient, I can compute it once and pass it around. But since scipy's minimize calls the objective and gradient separately, I'll just compute it in both. The overhead is small since correlate is O(n^2) and n is at most ~600.

Actually, let me reconsider the resolution. The previous solution used n=135 and got 0.380982. Going to n=270 doubles the resolution, and n=405 triples it. The key question is: does higher resolution actually help?

In the continuous problem, the optimal h is a measurable function. The discretization error is O(1/n) (for a reasonable h). So the C₅ value at resolution n is approximately C₅* + O(1/n), where C₅* is the true optimal value.

If the true optimal is around 0.3808 or so, then at n=135, the error is about O(1/135) ≈ 0.007, and at n=405, it's O(1/405) ≈ 0.002. So higher resolution should give a better approximation.

But there's a catch: with more variables, the optimization is harder (more local optima). So I need to make sure the optimization is good enough at the higher resolution.

I think the strategy of starting at low resolution and gradually increasing (with warm-starting) is the right approach.

Let me also add a "polishing" step at the very end: try small random perturbations of the best solution and keep the one that gives the lowest C₅. This is a form of local search.

Here's my final code:

```python
import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time


def compute_c5_from_h(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float)
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def compute_O(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return c[n-1:n-1+n] * dx


def smooth_obj(h, beta, gamma):
    n = len(h)
    O = compute_O(h)
    O_max = np.max(O)
    val = O_max + np.log(np.sum(np.exp(beta * (O - O_max)))) / beta
    val += gamma * (np.sum(h) - n / 2.0) ** 2
    return val


def smooth_grad(h, beta, gamma):
    n = len(h)
    dx = 2.0 / n
    O = compute_O(h)
    O_max = np.max(O)
    log_p = beta * (O - O_max)
    log_p -= np.max(log_p)
    p = np.exp(log_p)
    p /= np.sum(p)
    cumsum_p = np.cumsum(p)
    conv_ph = np.convolve(p, h, mode='full')[:n]
    conv_pr = np.convolve(p, h[::-1], mode='full')[n-1::-1]
    grad = dx * (cumsum_p - conv_ph - conv_pr)
    grad = grad + 2.0 * gamma * (np.sum(h) - n / 2.0)
    return grad


def smooth_optimize(h_init, n, tau):
    t_start = time.time()
    h = np.asarray(h_init, dtype=float).copy()
    n_levels = 60
    betas = np.geomspace(5.0, 400.0, n_levels)
    gammas = np.geomspace(0.5, 100.0, n_levels)
    for i in range(n_levels):
        if time.time() - t_start > tau:
            break
        beta = float(betas[i])
        gamma = float(gammas[i])
        res = minimize(smooth_obj, h, args=(beta, gamma),
                       jac=smooth_grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': 25, 'ftol': 1e-15, 'gtol': 1e-10})
        if res.x is not None:
            h = np.clip(res.x, 0.0, 1.0)
    return project_feasible(h, n)


def subgrad_refine(h, n, maxiter, step_start, step_end):
    h = np.asarray(h, dtype=float).copy()
    best_c5 = compute_c5_from_h(h)
    best_h = h.copy()
    for i in range(maxiter):
        frac = i / max(1, maxiter - 1)
        step = step_start * (step_end / step_start) ** frac
        O = compute_O(h)
        k_star = int(np.argmax(O))
        if k_star == 0:
            grad = (2.0 / n) * (1.0 - 2.0 * h)
        else:
            grad = np.zeros(n)
            grad[k_star:] += (2.0 / n) * (1.0 - h[:n - k_star])
            grad[:n - k_star] -= (2.0 / n) * h[k_star:]
        h_new = project_feasible(h - step * grad, n)
        c5_val = compute_c5_from_h(h_new)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5


def upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    f = interp1d(x_old, h, kind='linear', fill_value='extrapolate')
    h_new = np.clip(f(x_new), 0.0, 1.0)
    return project_feasible(h_new, n_new)


def local_search(h, n, rng, n_trials=50, scale=0.03):
    best_h = h.copy()
    best_c5 = compute_c5_from_h(h)
    for _ in range(n_trials):
        h_try = h + scale * rng.standard_normal(n)
        h_try = project_feasible(h_try, n)
        c5_try = compute_c5_from_h(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(135, 0.5)

    best_h = h0.copy()
    best_c5 = compute_c5_from_h(best_h)
    best_n = len(best_h)

    # Phase 1: Upsample to 270
    if time.time() - t0 < budget_s - 500:
        n1 = 270
        h1 = upsample(best_h, n1)
        tau1 = max(30, budget_s - 500 - (time.time() - t0))
        h1 = smooth_optimize(h1, n1, tau1)
        h1, c5_1 = subgrad_refine(h1, n1, 300, 0.05, 0.001)
        if c5_1 < best_c5:
            best_h, best_c5, best_n = h1, c5_1, n1

    # Phase 2: Upsample to 405
    if time.time() - t0 < budget_s - 250:
        n2 = 405
        h2 = upsample(best_h, n2)
        tau2 = max(30, budget_s - 250 - (time.time() - t0))
        h2 = smooth_optimize(h2, n2, tau2)
        h2, c5_2 = subgrad_refine(h2, n2, 300, 0.04, 0.001)
        if c5_2 < best_c5:
            best_h, best_c5, best_n = h2, c5_2, n2

    # Phase 3: Iterated refinement on best
    for _ in range(3):
        if time.time() - t0 > budget_s - 100:
            break
        remaining = budget_s - 100 - (time.time() - t0)
        h_ref = smooth_optimize(best_h, best_n, max(20, remaining * 0.4))
        h_ref, c5_ref = subgrad_refine(h_ref, best_n, 200, 0.02, 0.0005)
        if c5_ref < best_c5:
            best_h, best_c5 = h_ref, c5_ref

    # Phase 4: Local search
    if time.time() - t0 < budget_s - 50:
        best_h, best_c5 = local_search(best_h, best_n, rng, n_trials=100, scale=0.02)
        # Another round of subgradient after local search
        h_ref, c5_ref = subgrad_refine(best_h, best_n, 200, 0.015, 0.0003)
        if c5_ref < best_c5:
            best_h, best_c5 = h_ref, c5_ref

    # Phase 5: Random restarts
    if time.time() - t0 < budget_s - 80:
        n_restarts = 20
        for ri in range(n_restarts):
            if time.time() - t0 > budget_s - 30:
                break
            mode = ri % 4
            arange_n = np.arange(best_n)
            if mode == 0:
                h_rand = 0.5 + 0.35 * np.sin(2 * np.pi * (1 + ri * 0.7) * arange_n / best_n + rng.random() * 6.28)
            elif mode == 1:
                h_rand = 0.5 + 0.3 * rng.standard_normal(best_n)
            elif mode == 2:
                h_rand = 0.5 + 0.35 * np.cos(2 * np.pi * (2 + ri * 0.5) * arange_n / best_n + rng.random() * 3.14)
            else:
                h_rand = 0.5 + 0.25 * np.sin(2 * np.pi * (3 + ri) * arange_n / best_n) + 0.15 * np.cos(4 * np.pi * arange_n / best_n)
            h_rand = project_feasible(h_rand, best_n)

            remaining = budget_s - 30 - (time.time() - t0)
            tau_r = max(10, remaining * 0.25)
            h_opt = smooth_optimize(h_rand, best_n, tau_r)
            h_opt, c5_opt = subgrad_refine(h_opt, best_n, 100, 0.03, 0.0005)
            if c5_opt < best_c5:
                best_h, best_c5 = h_opt, c5_opt

    # Final local search
    if time.time() - t0 < budget_s - 20:
        best_h, best_c5 = local_search(best_h, best_n, rng, n_trials=50, scale=0.01)
        h_ref, c5_ref = subgrad_refine(best_h, best_n, 100, 0.01, 0.0002)
        if c5_ref < best_c5:
            best_h, best_c5 = h_ref, c5_ref

    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5_from_h(best_h)
    return best_h, best_c5, best_n
```
