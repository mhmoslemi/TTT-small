import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time
from scipy.ndimage import uniform_filter1d

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def sum_eq(x, n):
    return np.sum(x) - n / 2.0

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
        else:
            mask = h > 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    # Final correction
    diff = n / 2.0 - np.sum(h)
    if abs(diff) > 1e-15:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(h[idx] + diff, 1.0)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(h[idx] + diff, 0.0)

def obj_linf(x, n, beta):

    corr = np.correlate(h, 1.0 - h, mode="full")

def gradient_subgrad(h, n):

    # c_k = sum_i h[i]*(1-h[i+k])
    # dc_k/dh[i] = (1-h[i+k]) - h[i-k] (for valid indices)

    k_max = np.argmax(corr)

    lag = k_max - (n - 1)
    grad = np.zeros(n)

        term2 = 0.0

    best_h = h_cur.copy()

    pos = 0
    fill = True
    while pos < n:
        bsz = min(block_sizes[rng.randint(len(block_sizes))], n - pos)
        h[pos:pos+bsz] = 1.0 if fill else 0.0
        fill = not fill
        pos += bsz

def optimize_slsqp(h0, n, max_iter, beta):

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq, 'args': (n,)}]
    try:
        res = minimize(obj_linf, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, args=(n, beta),
                       options={'maxiter': max_iter, 'ftol': 1e-13, 'iprint': 0})
        h_res = project_feasible(res.x, n)
        return h_res, compute_c5(h_res)
    except Exception:
        return None, None

def optimize_lbfgs_barrier(h0, n, max_iter, beta, mu=0.1):
    # L-BFGS-B with a barrier for the sum constraint
    h0 = project_feasible(h0, n)
    
    def barrier_obj(x):
        h = np.clip(x, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode="full")
        sm = smooth_max(corr, beta) * (2.0 / n)
        # Barrier for sum constraint: mu * (sum(x) - n/2)^2
        penalty = mu * (np.sum(x) - n/2.0)**2
        # Barrier for bounds

            if val < 1e-10:

            elif val > 1.0 - 1e-10:
                bound_penalty -= mu * np.log(max(1.0 - val, 1e-30))
        return sm + penalty + bound_penalty

        h_res = project_feasible(res.x, n)

    except Exception:

- All other terms don't depend on h[j]

So ∂c_k/∂h[j] = (1-h[j+k]) * 1_{0<=j+k<n} - h[j-k] * 1_{0<=j-k<n}

This can be vectorized! For a given k (which is the lag, not the array index), the gradient is:
grad[j] = (1 - h[j+k]) * 1_{j+k<n} - h[j-k] * 1_{j-k>=0}

In numpy, if lag = k (positive) or k (negative):
- For lag > 0: grad[j] = (1-h[j+lag]) for j < n-lag, minus h[j-lag] for j >= lag
  Actually: grad = (1 - h[lag:]) - h[:-lag] (padded appropriately)

grad = np.zeros(n)
For lag k (can be positive or negative):
  If k >= 0:

    grad[k:] += (-h[:n-k])     # -h[j-k] for j=k..n-1

    grad[:n-k_neg] += (-h[k_neg:])     # -h[j-k] = -h[j+k_neg] for j=0..n-1-k_neg

This is O(n) per lag, and we only need it for the active lag(s).

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def sum_eq(x, n):
    return np.sum(x) - n / 2.0

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)

            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)

        else:
            idx = int(np.argmax(h))
            h[idx] = max(h[idx] + diff, 0.0)
    return h

    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, beta) * (2.0 / n)

        h_res = project_feasible(res.x, n)
        return h_res, compute_c5(h_res)
    except Exception:
        return None, None

def grad_c_k(h, lag):
    """Compute gradient of c_lag w.r.t. h."""
    n = len(h)
    grad = np.zeros(n)
    if lag >= 0:
        if lag < n:
            grad[:n-lag] += (1.0 - h[lag:])
            grad[lag:] += (-h[:n-lag])
    else:
        k = -lag
        if k < n:
            grad[k:] += (1.0 - h[:n-k])
            grad[:n-k] += (-h[k:])
    return grad

def subgradient_refine(h, n, max_iter=100, eta0=0.1):
    """Subgradient method focusing on the worst lag."""
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    dx = 2.0 / n
    eta = eta0
    for it in range(max_iter):
        corr = np.correlate(h_cur, 1.0 - h_cur, mode="full")
        # Find top few lags
        top_n = min(5, len(corr))
        top_indices = np.argsort(corr)[-top_n:]
        # Weight by how close to max
        weights = corr[top_indices] - corr[top_indices[0]] + 1.0
        weights = np.exp(weights * 2)  # Sharpen
        weights /= weights.sum()
        
        grad = np.zeros(n)
        for wi, ki in zip(weights, top_indices):
            lag = ki - (n - 1)
            grad += wi * grad_c_k(h_cur, lag)
        
        h_new = h_cur - eta * grad
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        h_cur = h_new
        eta *= 0.995
        if eta < 0.001:
            break
    return best_h, best_c5

    # Phase 2: Subgradient with momentum
    if time.time() - t0 < budget_s - 300:
        h_sub, c5_sub = subgradient_refine(best_h, best_n, max_iter=100, eta0=0.06, momentum=0.85)
        if c5_sub < best_c5:
            best_h, best_c5 = h_sub.copy(), c5_sub
        if time.time() - t0 < budget_s - 200:
            h_opt, c5_opt = optimize_slsqp(best_h, best_n, 300, 250)
            if h_opt is not None and c5_opt < best_c5:
                best_h, best_c5 = h_opt.copy(), c5_opt

    # Phase 3: Multi-start exploration
    no_improve = 0
    while time.time() - t0 < budget_s - 250 and no_improve < 55:
        if no_improve < 15:
            sigma = 0.08 + 0.03 * (0.9 ** min(no_improve, 20))
        else:
            sigma = 0.05 + 0.02 * (0.93 ** min(no_improve - 15, 30))
        h_start = project_feasible(best_h + rng.normal(0, sigma, best_n), best_n)
        h_opt, c5_opt = optimize_slsqp(h_start, best_n, 250, 100)
        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5 = h_opt.copy(), c5_opt
            no_improve = 0
            if time.time() - t0 < budget_s - 200:
                h_sub, c5_sub = subgradient_refine(best_h, best_n, max_iter=50, eta0=0.04, momentum=0.8)
                if c5_sub < best_c5:
                    best_h, best_c5 = h_sub.copy(), c5_sub
                    h_opt2, c5_opt2 = optimize_slsqp(best_h, best_n, 200, 200)
                    if h_opt2 is not None and c5_opt2 < best_c5:
                        best_h, best_c5 = h_opt2.copy(), c5_opt2
        else:
            no_improve += 1

        if no_improve % 10 == 5 and time.time() - t0 < budget_s - 180:
            h_bin = make_binary_random(best_n, rng)
            blend = 0.4 * best_h + 0.6 * h_bin
            h_blend = project_feasible(blend, best_n)
            h_opt, c5_opt = optimize_slsqp(h_blend, best_n, 300, 80)
            if h_opt is not None and c5_opt < best_c5:
                best_h, best_c5 = h_opt.copy(), c5_opt
                no_improve = 0

        if no_improve % 15 == 10 and time.time() - t0 < budget_s - 180:
            h_flat = project_feasible(best_h + rng.uniform(-0.3, 0.3, best_n), best_n)
            h_opt, c5_opt = optimize_slsqp(h_flat, best_n, 300, 60)
            if h_opt is not None and c5_opt < best_c5:
                best_h, best_c5 = h_opt.copy(), c5_opt
                no_improve = 0

    # Phase 4: Coordinate refinement at current n
    if time.time() - t0 < budget_s - 100:
        h_coord, c5_coord = coordinate_refine(best_h, best_n, sweeps=3, top_lags=3)
        if c5_coord < best_c5:
            best_h, best_c5 = h_coord.copy(), c5_coord
            h_opt, c5_opt = optimize_slsqp(best_h, best_n, 200, 300)
            if h_opt is not None and c5_opt < best_c5:
                best_h, best_c5 = h_opt.copy(), c5_opt

    # Phase 5: Upscale to higher n
    for n_try in [380, 570]:
        if time.time() - t0 > budget_s - 120:
            break
        h_interp = interpolate_h(best_h, best_n, n_try)
        h_interp = project_feasible(h_interp, n_try)
        h_opt, c5_opt = optimize_slsqp(h_interp, n_try, 350, 50)
        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt.copy(), c5_opt, n_try
            if time.time() - t0 < budget_s - 80:
                h_opt2, c5_opt2 = optimize_slsqp(best_h, best_n, 250, 150)
                if h_opt2 is not None and c5_opt2 < best_c5:
                    best_h, best_c5 = h_opt2.copy(), c5_opt2
                if time.time() - t0 < budget_s - 60:
                    h_opt3, c5_opt3 = subgradient_refine(best_h, best_n, max_iter=40, eta0=0.03, momentum=0.7)
                    if c5_opt3 < best_c5:
                        best_h, best_c5 = h_opt3.copy(), c5_opt3

    # Phase 6: Final high-beta SLSQP polish
    for beta_final in [300, 500]:
        if time.time() - t0 > budget_s - 35:
            break
        h_opt, c5_opt = optimize_slsqp(best_h, best_n, 300, beta_final)
        if h_opt is not None and c5_opt < best_c5:
            best_h, best_c5 = h_opt.copy(), c5_opt

    # Phase 7: Subgradient final polish
    if time.time() - t0 < budget_s - 20:
        h_sub, c5_sub = subgradient_refine(best_h, best_n, max_iter=50, eta0=0.02, momentum=0.7)
        if c5_sub < best_c5:
            best_h, best_c5 = h_sub.copy(), c5_sub

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, len(best_h))
