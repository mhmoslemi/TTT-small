def alternating_smooth(h, n, num_iters=50):
    dx = 2.0 / n
    h = h.copy()
    for it in range(num_iters):
        g = 1.0 - h
        corr = np.correlate(h, g, mode="full")
        k_star = np.argmax(corr)
        # k_star in [0, 2n-2], corresponds to lag k_star - (n-1)
        lag = k_star - (n - 1)
        if lag == 0:
            break
        # Smooth h at scale |lag|
        alpha = 0.01  # smoothing strength
        if lag > 0:
            # For i where i+lag < n: blend h[i] and h[i+lag]
            for i in range(n - lag):
                diff = h[i + lag] - h[i]
                h[i] += alpha * diff
                h[i + lag] -= alpha * diff
        else:
            for i in range(-lag):
                diff = h[i - lag] - h[i]
                h[i] += alpha * diff
                h[i - lag] -= alpha * diff
        h = np.clip(h, 0, 1)
        h = project_feasible(h, n)
    return h

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(values, tau):
    v = values / tau
    vmax = np.max(v)
    return tau * (vmax + np.log(np.sum(np.exp(v - vmax))))

def project_feasible(h, n):
    target = n / 2.0
    lo = -3.0
    hi = 3.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-14:
        order = np.argsort(np.abs(result - 0.5))
        for idx in order:
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                adj = min(diff, 1.0 - result[idx])
            else:
                adj = max(diff, -result[idx])
            result[idx] += adj
            diff -= adj
    return result

def equality_constraint(h, n):
    return np.sum(h) - n / 2.0

def obj_smooth(h, n, tau):
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return smooth_max(corr, tau) * dx

def obj_penalty(h, n, tau, mu):
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    base = smooth_max(corr, tau) * dx
    penalty = mu * (np.sum(h) - n / 2.0) ** 2
    return base + penalty

def compute_subgradient(h, n):
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    k_star = np.argmax(corr)
    lag = k_star - (n - 1)
    grad = np.zeros(n)
    if 0 <= lag < n:
        grad[lag:] += (1.0 - h[:-lag])
        grad[:-lag] += -h[lag:]
    elif -n < lag < 0:
        la = -lag
        grad[:-la] += (1.0 - h[la:])
        grad[la:] += -h[:-la]
    return grad * dx, int(lag), float(corr[k_star] * dx)

def subgradient_descent(h, n, max_iters, start_step, end_step):
    h = h.copy()
    best_local = evaluate_c5(h)
    best_h_local = h.copy()
    step = start_step
    for it in range(max_iters):
        ratio = it / max(1, max_iters - 1)
        step = start_step * (end_step / start_step) ** ratio
        grad, lag, val = compute_subgradient(h, n)
        h_new = h - step * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = project_feasible(h_new, n)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_local:
            best_local = c5_new
            best_h_local = h_new.copy()
        h = h_new
    return best_h_local, best_local

        h[lag_abs:] -= alpha * diff
        h = np.clip(h, 0, 1)
        h = project_feasible(h, n)
    return h

def slsqp_optimize(h, n, tau, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': equality_constraint, 'args': (n,)}]
    try:
        res = minimize(
            obj_smooth, h,
            method='SLSQP',
            args=(n, tau),
            bounds=bounds,
            constraints=cons,
            options={'maxiter': maxiter, 'ftol': 1e-13}
        )
        return project_feasible(res.x, n)
    except Exception:
        return h

def lbfgsb_optimize(h, n, tau, mu, maxiter):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            obj_penalty, h,
            method='L-BFGS-B',
            args=(n, tau, mu),
            bounds=bounds,
            options={'maxiter': maxiter, 'ftol': 1e-15}
        )
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def trustconstr_optimize(h, n, tau, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': equality_constraint, 'args': (n,)}]
    try:
        res = minimize(
            obj_smooth, h,
            method='trust-constr',
            args=(n, tau),
            bounds=bounds,
            constraints=cons,
            options={'maxiter': maxiter, 'gtol': 1e-8}
        )
        return project_feasible(np.clip(res.x, 0, 1), n)
    except Exception:
        return h

def generate_starts(n, best_h, rng):
    starts = []
    x = np.linspace(0, 2, n, endpoint=False)
    ar = np.arange(n, dtype=float)

    if best_h is not None:
        starts.append(best_h.copy())
        starts.append(np.clip(best_h + rng.randn(n) * 0.03, 0, 1))
        starts.append(np.clip(best_h + rng.randn(n) * 0.08, 0, 1))
        starts.append(np.clip(best_h + rng.randn(n) * 0.15, 0, 1))

    for freq in [1.1, 1.4, 1.7, 2.1, 2.5, 3.0, 3.7]:
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.25, 0.48)
        h_s = np.clip(0.5 + amp * np.sin(2 * np.pi * freq * x + phase), 0, 1)
        starts.append(h_s)

    starts.append(np.linspace(0.02, 0.98, n))
    starts.append(np.linspace(0.98, 0.02, n))
    starts.append(rng.rand(n))
    starts.append(rng.rand(n))

    for period in [2, 3, 4, 5]:
        h_saw = np.mod(ar / n * float(period), 2.0)
        h_saw = np.where(h_saw < 1.0, h_saw, 2.0 - h_saw)
        starts.append(np.clip(h_saw, 0, 1))

    for _ in range(3):
        c1 = rng.uniform(0.15, 0.45) * n
        c2 = rng.uniform(0.55, 0.85) * n
        w = rng.uniform(0.08, 0.2) * n
        b1 = np.exp(-0.5 * ((ar - c1) / w) ** 2)
        b2 = np.exp(-0.5 * ((ar - c2) / w) ** 2)
        starts.append(np.clip(0.1 + 0.8 * 0.5 * (b1 + b2), 0, 1))

    h_tri = np.abs(np.mod(ar / n * 2.0, 2.0) - 1.0)
    starts.append(np.clip(0.1 + 0.8 * h_tri, 0, 1))

    h_wave = 0.5 + 0.4 * np.cos(2 * np.pi * 2.0 * x) + 0.1 * np.sin(2 * np.pi * 5.0 * x)
    starts.append(np.clip(h_wave, 0, 1))

    return starts

def run(seed=42, budget_s=1300, **kwargs):
    rng = np.random.RandomState(seed)
    start_time = time.time()

    best_h = None
    best_c5 = float('inf')

    parent_h = None
    try:
        parent_h = np.array(initial_h_values, dtype=float)
    except Exception:
        pass

    def elapsed():
        return time.time() - start_time

    def time_rem():
        return budget_s - (time.time() - start_time)

    if parent_h is not None:
        best_h = parent_h.copy()
        best_c5 = evaluate_c5(best_h)
        n_base = len(best_h)
    else:
        n_base = 59
        best_h = project_feasible(np.full(n_base, 0.5), n_base)
        best_c5 = evaluate_c5(best_h)

    n = n_base

    # Phase 1: Alternating smoothing (fast structural improvement)
    if time_rem() > 50:
        h_smooth = alternating_smooth_pass(best_h, n, 200, 0.03)
        c5_s = evaluate_c5(h_smooth)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_smooth

        h_smooth2 = alternating_smooth_pass(best_h, n, 300, 0.01)
        c5_s2 = evaluate_c5(h_smooth2)
        if c5_s2 < best_c5:
            best_c5 = c5_s2
            best_h = h_smooth2

    # Phase 2: Subgradient descent
    if time_rem() > 100:
        h_sg, c5_sg = subgradient_descent(best_h, n, 500, 0.5, 0.01)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg

    # Phase 3: SLSQP with tau annealing from best
    if time_rem() > 100:
        h_curr = best_h.copy()
        tau_seq = [0.04, 0.02, 0.01, 0.005, 0.002, 0.001]
        for tau in tau_seq:
            if time_rem() < 40:
                break
            h_curr = slsqp_optimize(h_curr, n, tau, 100)
            c5 = evaluate_c5(h_curr)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_curr.copy()

    # Phase 4: Multiple diverse starts with multi-method optimization
    starts = generate_starts(n, best_h, rng)
    max_starts = min(len(starts), 14)

    for si in range(max_starts):
        if time_rem() < 40:
            break

        h0 = project_feasible(starts[si].copy(), n)

        # Method A: SLSQP
        if time_rem() > 50:
            h_a = h0.copy()
            for tau in [0.03, 0.01, 0.005]:
                if time_rem() < 20:
                    break
                h_a = slsqp_optimize(h_a, n, tau, 80)
            c5_a = evaluate_c5(h_a)
            if c5_a < best_c5:
                best_c5 = c5_a
                best_h = h_a.copy()

        # Method B: L-BFGS-B with penalty
        if time_rem() > 40:
            h_b = h0.copy()
            for mu in [100, 1000, 10000]:
                if time_rem() < 15:
                    break
                h_b = lbfgsb_optimize(h_b, n, 0.01, mu, 300)
            c5_b = evaluate_c5(h_b)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()

        # Method C: Trust-constr
        if time_rem() > 30:
            h_c = trustconstr_optimize(h0, n, 0.02, 100)
            c5_c = evaluate_c5(h_c)
            if c5_c < best_c5:
                best_c5 = c5_c
                best_h = h_c.copy()

    # Phase 5: Try different n values (interpolated from best)
    if time_rem() > 100:
        n_candidates = []
        for dn in [-6, -4, -2, 2, 4, 6, 8, 10]:
            nt = n_base + dn
            if 25 <= nt <= 120 and nt != n_base:
                n_candidates.append(nt)

        for nt in n_candidates:
            if time_rem() < 30:
                break

            x_new = np.linspace(0, 2, nt, endpoint=False)
            x_old = np.linspace(0, 2, len(best_h), endpoint=False)
            f_interp = interp1d(x_old, best_h, kind='cubic', fill_value='extrapolate', assume_sorted=True)
            h_try = np.clip(f_interp(x_new), 0, 1)
            h_try = project_feasible(h_try, nt)

            h_try = alternating_smooth_pass(h_try, nt, 50, 0.02)
            c5_t = evaluate_c5(h_try)

            for tau in [0.02, 0.01, 0.005]:
                if time_rem() < 15:
                    break
                h_try = slsqp_optimize(h_try, nt, tau, 60)
                c5_t = evaluate_c5(h_try)

            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_try.copy()
                n = nt

    # Phase 6: Final intensive fine-tuning
    if best_h is not None and time_rem() > 20:
        n_f = len(best_h)
        h_f = best_h.copy()

        # Very fine SLSQP
        for tau in [0.001, 0.0005, 0.0002, 0.0001]:
            if time_rem() < 15:
                break
            h_f = slsqp_optimize(h_f, n_f, tau, 200)
            c5_f = evaluate_c5(h_f)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f.copy()

        # Alternating smooth + optimize cycle
        for _ in range(3):
            if time_rem() < 20:
                break
            h_f = alternating_smooth_pass(best_h, n_f, 100, 0.005)
            h_f = slsqp_optimize(h_f, n_f, 0.001, 150)
            c5_f = evaluate_c5(h_f)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f.copy()

        # Perturbation restarts
        for i in range(10):
            if time_rem() < 15:
                break
            scale = 0.004 + 0.004 * i
            h_p = best_h + rng.randn(n_f) * scale
            h_p = project_feasible(h_p, n_f)
            h_p = slsqp_optimize(h_p, n_f, 0.001, 150)
            c5_p = evaluate_c5(h_p)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_p.copy()

        # Trust-constr final push
        if time_rem() > 15:
            h_tc = trustconstr_optimize(best_h, n_f, 0.001, 200)
            c5_tc = evaluate_c5(h_tc)
            if c5_tc < best_c5:
                best_c5 = c5_tc
                best_h = h_tc.copy()

    if best_h is None:
        if parent_h is not None:
            best_h = parent_h.copy()
            best_c5 = evaluate_c5(best_h)
        else:
            n = 59
            best_h = project_feasible(np.full(n, 0.5), n)
            best_c5 = evaluate_c5(best_h)

    return best_h, best_c5, len(best_h)
