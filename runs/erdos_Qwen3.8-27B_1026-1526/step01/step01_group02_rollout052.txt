    c = fftconvolve(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(np.array(h, dtype=float).copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
            np.clip(h, 0.0, 1.0, out=h)
        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
            np.clip(h, 0.0, 1.0, out=h)
    return h


def smooth_max_val_grad(h, beta, lam):
    n = len(h)
    c = fftconvolve(h, 1.0 - h, mode='full') * (2.0 / n)
    m = np.max(c)
    shifted = beta * (c - m)
    e = np.exp(np.clip(shifted, -500.0, 500.0))
    se = np.sum(e)
    logsumexp = m + np.log(se) / beta
    w = e / se
    grad = (2.0 / n) * np.correlate(w, 1.0 - 2.0 * h, mode='valid')
    sum_dev = np.sum(h) - n / 2.0
    penalty = lam * sum_dev * sum_dev
    grad = grad + 2.0 * lam * sum_dev
    return logsumexp + penalty, grad


def lbgfsb_optimize(h_init, n, beta_start, beta_end, n_steps, lam, maxiter, time_limit):
    t_start = time.time()
    h = project_feasible(h_init, n)
    best_h = h.copy()
    best_c5 = compute_c5_from_h(best_h)

    for step in range(n_steps):
        if time.time() - t_start > time_limit:
            break
        frac = step / max(1, n_steps - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        lam_eff = lam * (1.0 + 4.0 * frac)

        def obj(x):
            val, _ = smooth_max_val_grad(x, beta, lam_eff)
            return val

        def grad_fn(x):
            _, g = smooth_max_val_grad(x, beta, lam_eff)
            return g

        result = minimize(obj_fn, h, jac=grad_fn, method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-16, 'iprint': -1})
        if result.x is not None:
            h_new = project_feasible(np.clip(result.x, 0.0, 1.0), n)
            c5_new = compute_c5_from_h(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h = h_new

    return best_h, best_c5


def slsqp_refine(h_init, n, maxiter=2000):
    c5_init = compute_c5_from_h(h_init)

    def obj(x):
        return x[-1]

    def ineq(x):
        h = x[:-1]
        c = fftconvolve(h, 1.0 - h, mode='full') * (2.0 / n)
        return x[-1] - c

    def eq(x):
        return np.sum(x[:-1]) - n / 2.0

    x0 = np.concatenate([h_init, [c5_init + 0.1]])
    bounds = [(0.0, 1.0)] * n + [(max(0.01, c5_init * 0.5), 1.5)]
    constraints = [{'type': 'ineq', 'fun': ineq},
                   {'type': 'eq', 'fun': eq}]
    try:
        result = minimize(obj, x0, method='SLSQP', bounds=bounds,
                          constraints=constraints,
                          options={'maxiter': maxiter, 'ftol': 1e-15})
        if result.x is not None:
            h_out = project_feasible(result.x[:-1], n)
            return h_out, compute_c5_from_h(h_out)
    except Exception:
        pass
    return h_init, c5_init


def basin_hopping_step(h_center, n, rng, scale, beta, lam, maxiter):
    h_pert = h_center + scale * rng.standard_normal(n)
    h_pert = project_feasible(h_pert, n)

    def obj(x):
        val, _ = smooth_max_val_grad(x, beta, lam)
        return val

    def grad_fn(x):
        _, g = smooth_max_val_grad(x, beta, lam)
        return g

    result = minimize(obj, h_pert, jac=grad_fn, method='L-BFGS-B',
                      bounds=[(0.0, 1.0)] * n,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14, 'iprint': -1})
    if result.x is not None:
        h_new = project_feasible(np.clip(result.x, 0.0, 1.0), n)
        return h_new, compute_c5_from_h(h_new)
    return h_pert, compute_c5_from_h(h_pert)


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h0)
    else:
        n_init = 135
        h0 = np.full(n_init, 0.5)

    h0 = project_feasible(h0, n_init)
    best_h = h0.copy()
    best_c5 = compute_c5_from_h(best_h)
    n_best = n_init

    if time.time() - t0 < budget_s - 180:
        t_lim1 = min(100, (budget_s - 180) * 0.35)
        best_h, best_c5 = lbgfsb_optimize(best_h, n_best,
                                          beta_start=3.0, beta_end=800.0,
                                          n_steps=30, lam=30.0,
                                          maxiter=150, time_limit=t_lim1)
        n_best = len(best_h)

    if time.time() - t0 < budget_s - 90:
        n_attempts = 50
        for i in range(n_attempts):
            if time.time() - t0 > budget_s - 40:
                break
            frac = i / max(1, n_attempts)
            scale = 0.20 * (1.0 - frac) ** 1.5 + 0.015
            beta = 50.0 + 200.0 * frac
            h_new, c5_new = basin_hopping_step(best_h, n_best, rng, scale, beta, 50.0, 300)
            if c5_new < best_c5 - 1e-12:
                best_h = h_new
                best_c5 = c5_new

    for n_try in [200, 270, 360, 500]:
        if time.time() - t0 > budget_s - 150:
            break
        x_old = np.linspace(0.0, 2.0, n_best, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_try, endpoint=False)
        f_interp = interp1d(x_old, best_h, kind='linear', fill_value='extrapolate')
        h_interp = np.clip(f_interp(x_new), 0.0, 1.0)
        h_interp = project_feasible(h_interp, n_try)

        t_lim = min(60, (budget_s - time.time() - 150) * 0.5)
        if t_lim < 10:
            break
        h_alt, c5_alt = lbgfsb_optimize(h_interp, n_try,
                                        beta_start=5.0, beta_end=400.0,
                                        n_steps=15, lam=30.0,
                                        maxiter=100, time_limit=t_lim)
        if c5_alt < best_c5 - 1e-12:
            best_h = h_alt
            best_c5 = c5_alt
            n_best = n_try

    if time.time() - t0 < budget_s - 60:
        n_attempts2 = 20
        for i in range(n_attempts2):
            if time.time() - t0 > budget_s - 20:
                break
            scale = 0.05 + 0.05 * rng.random()
            h_new, c5_new = basin_hopping_step(best_h, n_best, rng, scale, 300.0, 80.0, 400)
            if c5_new < best_c5 - 1e-12:
                best_h = h_new
                best_c5 = c5_new

    if time.time() - t0 < budget_s - 30 and n_best <= 200:
        best_h, best_c5 = slsqp_refine(best_h, n_best, maxiter=3000)

    if time.time() - t0 < budget_s - 15:
        t_final = min(30, budget_s - time.time() - 10)
        best_h, best_c5 = lbgfsb_optimize(best_h, n_best,
                                          beta_start=200.0, beta_end=2000.0,
                                          n_steps=8, lam=100.0,
                                          maxiter=200, time_limit=t_final)

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5_from_h(best_h)
    return best_h, best_c5, len(best_h)
