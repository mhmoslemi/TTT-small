```python
def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float)
    if np.all(h >= -1e-12) and np.all(h <= 1 + 1e-12) and abs(np.sum(h) - target) < 1e-10:
        return np.clip(h, 0, 1)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_p = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(h_p)
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h_p = np.clip(h - lam, 0.0, 1.0)
    diff = target - np.sum(h_p)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h_p < 1.0 - 1e-10
        else:
            mask = h_p > 1e-10
        cnt = np.sum(mask)
        if cnt > 0:
            h_p[mask] += diff / cnt
            h_p = np.clip(h_p, 0.0, 1.0)
    return h_p

def get_correlation(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)

def smooth_max_value(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def smooth_max_weights(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500.0, 500.0)
    exp_s = np.exp(shifted)
    return exp_s / np.sum(exp_s)

def grad_smooth_max(h, n, beta):
    corr = get_correlation(h)
    w = smooth_max_weights(corr, beta)
    grad = np.zeros(n)
    # corr has 2n-1 entries, indexed 0 to 2n-2

            # Contribution from -h[j-m] for j = m, ..., n-1
            grad[m_val:] -= wi * h[:n-m_val]

    h = h0.copy()
    y = h0.copy()
    t_k = 1.0
    best_h = h0.copy()
    best_val = compute_c5(h0)
    
    for it in range(max_iter):

        g = grad_smooth_max(y, n, beta)

        h = h_new
        t_k = t_new
        
        val = compute_c5(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()
    
    return best_h, best_val
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float).copy()
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h_p = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(h_p)
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h_p = np.clip(h - lam, 0.0, 1.0)
    diff = target - np.sum(h_p)
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = h_p < 1.0 - 1e-12
        else:
            mask = h_p > 1e-12
        cnt = np.sum(mask)
        if cnt > 0:
            h_p[mask] += diff / cnt
            h_p = np.clip(h_p, 0.0, 1.0)
    return h_p


def get_corr_scaled(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)


def smooth_max_val(corr, beta):
    m = np.max(corr)
    s = np.clip(beta * (corr - m), -400.0, 400.0)
    return m + np.log(np.sum(np.exp(s))) / beta


def compute_gradient(h, n, beta):
    corr = get_corr_scaled(h)
    m = np.max(corr)
    s = np.clip(beta * (corr - m), -400.0, 400.0)
    exp_s = np.exp(s)
    w = exp_s / np.sum(exp_s)
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        k = idx - (n - 1)
        wi = w[idx]
        if wi < 1e-14:
            continue
        if k >= 0:
            grad[k:] += wi * (1.0 - h[:n - k])
            grad[:n - k] -= wi * h[k:]
        else:
            mv = -k
            grad[:n - mv] += wi * (1.0 - h[mv:])
            grad[mv:] -= wi * h[:n - mv]
    return grad * (2.0 / n)


def slsqp_optimize(h0, n, beta, max_iter=300, ftol=1e-14):
    def objective(x):
        h = np.clip(x, 0.0, 1.0)
        return smooth_max_val(get_corr_scaled(h), beta)

    def grad(x):
        h = np.clip(x, 0.0, 1.0)
        return compute_gradient(h, n, beta)

    def constraint_fun(x):
        return np.sum(x) - n / 2.0

    def constraint_grad(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_grad}]
    try:
        res = minimize(objective, h0, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': ftol, 'iprint': 0})
        h_res = project_feasible(res.x, n)
        c5 = compute_c5(h_res)
        return h_res, c5
    except Exception:
        return None, None


def fista_optimize(h0, n, max_iter=3000, beta=100.0):
    h = project_feasible(h0, n)
    y = h.copy()
    t_k = 1.0
    best_h = h.copy()
    best_val = compute_c5(h)
    lr = 5.0 / (n + 1.0)
    for it in range(max_iter):
        y_p = project_feasible(y, n)
        g = compute_gradient(y_p, n, beta)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        h_new = y_p - lr * g
        h_new = project_feasible(h_new, n)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = h_new + ((t_k - 1.0) / t_new) * (h_new - h)
        h = h_new
        t_k = t_new
        if it % 100 == 0:
            val = compute_c5(h)
            if val < best_val:
                best_val = val
                best_h = h.copy()
    val = compute_c5(h)
    if val < best_val:
        best_val = val
        best_h = h.copy()
    return best_h, best_val


def slsqp_subgradient(h0, n, max_iter=200):
    def objective(x):
        h = np.clip(x, 0.0, 1.0)
        corr = get_corr_scaled(h)
        return np.max(corr)

    def constraint_fun(x):
        return np.sum(x) - n / 2.0

    def constraint_grad(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_grad}]
    try:
        res = minimize(objective, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-14, 'iprint': 0})
        h_res = project_feasible(res.x, n)
        c5 = compute_c5(h_res)
        return h_res, c5
    except Exception:
        return None, None


def perturb_h(h, n, sigma, rng):
    h_p = h + rng.normal(0.0, sigma, n)
    return project_feasible(h_p, n)


def interpolate_to(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)


def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    best_h = None
    best_c5 = 1e9
    n = 190

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
        h_init = project_feasible(h_init, n)
        best_h = h_init
        best_c5 = compute_c5(h_init)
    else:
        best_h = np.ones(n) * 0.5
        best_c5 = compute_c5(best_h)

    if time.time() - t0 > budget_s - 50:
        return (best_h, best_c5, n)

    h_opt, c5_opt = slsqp_optimize(best_h, n, beta=50, max_iter=400)
    if c5_opt is not None and c5_opt < best_c5:
        best_h, best_c5 = h_opt, c5_opt

    h_opt, c5_opt = slsqp_optimize(best_h, n, beta=100, max_iter=400)
    if c5_opt is not None and c5_opt < best_c5:
        best_h, best_c5 = h_opt, c5_opt

    h_opt, c5_opt = slsqp_optimize(best_h, n, beta=200, max_iter=400)
    if c5_opt is not None and c5_opt < best_c5:
        best_h, best_c5 = h_opt, c5_opt

    if time.time() - t0 > budget_s - 150:
        best_h = project_feasible(best_h, n)
        return (best_h, best_c5, n)

    h_fista, c5_fista = fista_optimize(best_h, n, max_iter=5000, beta=80.0)
    if c5_fista < best_c5:
        best_h, best_c5 = h_fista, c5_fista
    h_opt, c5_opt = slsqp_optimize(best_h, n, beta=100, max_iter=300)
    if c5_opt is not None and c5_opt < best_c5:
        best_h, best_c5 = h_opt, c5_opt

    no_improve = 0
    restart_count = 0
    while time.time() - t0 < budget_s - 200 and no_improve < 30:
        restart_count += 1
        if restart_count % 5 == 0:
            sigma = 0.2
        else:
            sigma = 0.03 + 0.05 * rng.random()
        h_start = perturb_h(best_h, n, sigma, rng)
        beta_use = 80 + 40 * rng.random()
        h_opt, c5_opt = slsqp_optimize(h_start, n, beta=beta_use, max_iter=250)
        if c5_opt is not None and c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
            no_improve = 0
        else:
            no_improve += 1

    if time.time() - t0 > budget_s - 150:
        best_h = project_feasible(best_h, n)
        best_c5 = compute_c5(best_h)
        return (best_h, best_c5, n)

    h_fista2, c5_fista2 = fista_optimize(best_h, n, max_iter=5000, beta=150.0)
    if c5_fista2 < best_c5:
        best_h, best_c5 = h_fista2, c5_fista2
    h_opt, c5_opt = slsqp_optimize(best_h, n, beta=200, max_iter=300)
    if c5_opt is not None and c5_opt < best_c5:
        best_h, best_c5 = h_opt, c5_opt

    remaining = budget_s - (time.time() - t0)
    if remaining > 80:
        n_coarse = max(47, n // 4)
        h_coarse = interpolate_to(best_h, n, n_coarse)
        h_opt_c, c5_opt_c = slsqp_optimize(h_coarse, n_coarse, beta=100, max_iter=300)
        if c5_opt_c is not None:
            h_up = interpolate_to(h_opt_c, n_coarse, n)
            h_opt_r, c5_opt_r = slsqp_optimize(h_up, n, beta=150, max_iter=300)
            if c5_opt_r is not None and c5_opt_r < best_c5:
                best_h, best_c5 = h_opt_r, c5_opt_r

    remaining = budget_s - (time.time() - t0)
    if remaining > 40:
        h_opt, c5_opt = slsqp_optimize(best_h, n, beta=300, max_iter=500, ftol=1e-15)
        if c5_opt is not None and c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    remaining = budget_s - (time.time() - t0)
    if remaining > 25:
        h_opt, c5_opt = slsqp_subgradient(best_h, n, max_iter=200)
        if c5_opt is not None and c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n)
```
