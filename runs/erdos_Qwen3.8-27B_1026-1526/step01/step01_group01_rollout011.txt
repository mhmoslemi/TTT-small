    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
        else:
            mask = h > 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)

    if abs(diff) > 1e-15:

            idx = int(np.argmax(1.0 - h))
            h[idx] = min(h[idx] + diff, 1.0)

            idx = int(np.argmax(h))
            h[idx] = max(h[idx] + diff, 0.0)

    m = np.max(x)
    shifted = np.clip(beta * (x - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

    sm = smooth_max(corr, beta) * (2.0 / n)
    s = np.sum(h) - n / 2.0
    return sm + lam * s * s

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
        else:
            mask = h > 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    diff = n / 2.0 - np.sum(h)
    if abs(diff) > 1e-15:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(h[idx] + diff, 1.0)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(h[idx] + diff, 0.0)
    return h


def smooth_max_lse(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def smooth_max_pnorm(x, p):
    x_pos = np.maximum(x, 1e-12)
    if p <= 30:
        return float(np.mean(np.power(x_pos, p)) ** (1.0 / p))
    log_x = np.log(x_pos)
    p_log = p * log_x
    max_p = np.max(p_log)
    log_sum = max_p + np.log(np.sum(np.exp(p_log - max_p)))
    return float(np.exp((log_sum - np.log(len(x_pos))) / p))


def obj_lse_pen(x, n, beta, lam):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    sm = smooth_max_lse(corr, beta) * (2.0 / n)
    s = np.sum(h) - n / 2.0
    return sm + lam * s * s


def obj_pnorm_pen(x, n, p, lam):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode="full")
    sm = smooth_max_pnorm(corr, p) * (2.0 / n)
    s = np.sum(h) - n / 2.0
    return sm + lam * s * s


def optimize_lbfgsb(h0, n, use_p, p_val, beta_val, max_iter, lam):
    h0 = project_feasible(h0, n)
    bounds = [(0.0, 1.0)] * n
    maxfun = max(4000, n * 30)
    if use_p:
        res = minimize(obj_pnorm_pen, h0, method='L-BFGS-B',
                       bounds=bounds, args=(n, p_val, lam),
                       options={'maxiter': max_iter, 'ftol': 1e-16,
                                'gtol': 1e-9, 'maxfun': maxfun})
    else:
        res = minimize(obj_lse_pen, h0, method='L-BFGS-B',
                       bounds=bounds, args=(n, beta_val, lam),
                       options={'maxiter': max_iter, 'ftol': 1e-16,
                                'gtol': 1e-9, 'maxfun': maxfun})
    h_res = project_feasible(res.x, n)
    return h_res, compute_c5(h_res)


def optimize_slsqp(h0, n, beta, max_iter, lam):
    h0 = project_feasible(h0, n)
    bounds = [(0.0, 1.0)] * n

    def sum_fun(x):
        return np.sum(x) - n / 2.0

    cons = [{'type': 'eq', 'fun': sum_fun}]

    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode="full")
        return smooth_max_lse(corr, beta) * (2.0 / n)

    res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-14, 'iprint': 0})
    h_res = project_feasible(res.x, n)
    return h_res, compute_c5(h_res)

    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def make_perturbed(h, n, sigma, rng):
    return project_feasible(h + rng.normal(0.0, sigma, n), n)


def make_structured(n, kind, rng):

    else:
