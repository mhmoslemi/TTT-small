import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_B(h, j, n):
    nm1 = n - 1
    B = np.zeros(2 * n - 1)
    k_start = max(0, nm1 - j)
    k_end = min(2 * n - 2, 2 * nm1 - j)
    if k_start <= k_end:
        B[k_start:k_end + 1] += (1.0 - h[j - nm1 + np.arange(k_start, k_end + 1)])
    k_start2 = max(0, j)
    k_end2 = min(2 * n - 2, j + nm1)
    if k_start2 <= k_end2:
        B[k_start2:k_end2 + 1] += -h[j + nm1 - np.arange(k_start2, k_end2 + 1)]
    return B

def solve_1d(A, D, lo, hi):
    if hi - lo < 1e-14:
        return (lo + hi) * 0.5
    t_grid = np.linspace(lo, hi, 16)
    vals = np.max(A[None, :] + D[None, :] * t_grid[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        return lo
    if best_idx == len(t_grid) - 1:
        return hi
    lo_r = t_grid[best_idx - 1]
    hi_r = t_grid[best_idx + 1]
    for _ in range(40):
        mid = (lo_r + hi_r) * 0.5
        if np.max(A + D * mid) <= np.max(A + D * lo_r):
            lo_r = mid
        else:
            hi_r = mid
    return (lo_r + hi_r) * 0.5

def pair_descent_sweep(h, corr, rng, n_cand=12, max_rounds=3):
    n = len(h)
    improved_total = False
    for rnd in range(max_rounds):
        improved = False
        order = np.arange(n)
        rng.shuffle(order)
        for j in order:
            B_j = compute_B(h, j, n)
            best_val = float(np.max(corr))
            best_delta = 0.0
            best_D = None
            best_k = -1
            k_set = [(j + 1) % n, (j - 1 + n) % n, (j + n // 2) % n,
                     (j + n // 3) % n, (j + 2 * n // 3) % n,
                     (j + 2) % n, (j - 2 + n) % n, (j + n // 4) % n]
            for _ in range(max(0, n_cand - 8)):
                k_set.append(rng.randint(n))
            seen = set()
            for k in k_set:
                if k == j or k in seen:
                    continue
                seen.add(k)
                B_k = compute_B(h, k, n)
                D = B_j - B_k
                delta_lo = max(-h[j], h[k] - 1.0)
                delta_hi = min(1.0 - h[j], h[k])
                if delta_hi - delta_lo < 1e-13:
                    continue
                delta_opt = solve_1d(corr, D, delta_lo, delta_hi)
                val_opt = float(np.max(corr + D * delta_opt))
                if val_opt < best_val - 1e-13:
                    best_val = val_opt
                    best_delta = delta_opt
                    best_D = D
                    best_k = k
            if best_k >= 0 and abs(best_delta) > 1e-13:
                corr += best_delta * best_D
                h[j] += best_delta
                h[best_k] -= best_delta
                if h[j] < 0:
                    h[j] = 0.0

                    h[j] = 1.0
                if h[best_k] < 0:
                    h[best_k] = 0.0
                elif h[best_k] > 1:
                    h[best_k] = 1.0
                improved = True
        if not improved:
            break
        improved_total = True
    return h, corr, improved_total

        except Exception:
            pass
        h_cur = np.clip(h_cur, 0.0, 1.0)
        h_cur = project_to_sum(h_cur, ts)
        if time.time() > t_limit:
            break
    return h_cur

def generate_starts(n, rng, n_starts=6):
    starts = []
    h_uniform = np.full(n, 0.5)
    starts.append(h_uniform.copy())
    h_block = np.zeros(n)
    blk = max(1, n // 8)
    for i in range(0, n, 2 * blk):
        h_block[i:min(i + blk, n)] = 1.0
    starts.append(project_to_sum(h_block, n / 2.0))
    x = np.linspace(0, 2, n, endpoint=False)
    for freq in [1, 2, 3]:
        h_sine = 0.5 + 0.4 * np.sin(2 * np.pi * freq * x / 2.0 + rng.uniform(0, 2 * np.pi))
        starts.append(project_to_sum(h_sine, n / 2.0))
    h_rand = rng.uniform(0, 1, n)
    starts.append(project_to_sum(h_rand, n / 2.0))
    h_binary = (rng.uniform(0, 1, n) > 0.5).astype(float)
    starts.append(project_to_sum(h_binary, n / 2.0))
    return starts

def coarse_to_fine(h_best, n_best, n_target, rng, t_limit):
    if n_target == n_best:
        return h_best.copy()
    x_old = np.linspace(0, 2, n_best, endpoint=False)
    x_new = np.linspace(0, 2, n_target, endpoint=False)
    h_interp = np.interp(x_new, x_old, h_best, period=2.0)
    h_interp = project_to_sum(h_interp, n_target / 2.0)
    corr = np.correlate(h_interp, 1.0 - h_interp, mode="full")
    sweeps = 0
    while time.time() < t_limit and sweeps < 30:
        h_interp, corr, _ = pair_descent_sweep(h_interp, corr, rng, n_cand=10)
        sweeps += 1
    return h_interp

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float).copy()
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            best_h = None

    if best_h is None:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    def time_remaining():
        return max(0.0, budget_s - (time.time() - t_start))

    # Phase 1: L-BFGS-B optimization from multiple starts (55% budget)
    t_end1 = t_start + 0.55 * budget_s
    n_opt = best_n
    starts = []
    starts.append(best_h.copy())
    for i in range(5):
        h_pert = best_h + rng.normal(0, 0.08, n_opt)
        starts.append(project_to_sum(h_pert, n_opt / 2.0))
    for h_s in generate_starts(n_opt, rng, 6):
        if time.time() >= t_end1:
            break
        c5_s = compute_c5(h_s)
        if c5_s < best_c5 + 0.05:
            starts.append(h_s)

    for h_st in starts:
        if time.time() >= t_end1:
            break
        t_limit_st = min(t_end1 - 2, time.time() + (t_end1 - t_start) / max(1, len(starts) - len([s for s in starts if s is h_st])))
        h_opt = lbfgsb_optimize(h_st, n_opt, alpha_start=30, alpha_end=300,
                                penalty=500.0, maxiter=400, t_limit=t_end1 - 5)
        h_opt = project_to_sum(h_opt, n_opt / 2.0)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_opt

    # Phase 2: Pair descent intensive from best (20% budget)
    t_end2 = t_start + 0.75 * budget_s
    h_work = best_h.copy()
    n_w = best_n
    corr_w = np.correlate(h_work, 1.0 - h_work, mode="full")
    stall = 0
    while time.time() < t_end2:
        h_work, corr_w, imp = pair_descent_sweep(h_work, corr_w, rng, n_cand=14, max_rounds=2)
        c5_w = compute_c5(h_work)
        if c5_w < best_c5 - 1e-12:
            best_c5 = c5_w
            best_h = h_work.copy()
            stall = 0
        else:
            stall += 1
        if stall > 0 and stall % 50 == 0:
            noise_scale = min(0.1, 0.02 * stall / 50.0)
            h_work = project_to_sum(h_work + rng.normal(0, noise_scale, n_w), n_w / 2.0)
            corr_w = np.correlate(h_work, 1.0 - h_work, mode="full")
            if stall > 200:
                h_work = best_h.copy()
                corr_w = np.correlate(h_work, 1.0 - h_work, mode="full")
                stall = 0

    # Phase 3: Multi-resolution (15% budget)
    t_end3 = t_start + 0.90 * budget_s
    resolutions = sorted(set([best_n, best_n + 40, best_n + 80, best_n + 160,
                              max(120, best_n - 40)]))
    for n_tr in resolutions:
        if time.time() >= t_end3:
            break
        if abs(n_tr / 2.0 - round(n_tr / 2.0)) > 1e-6:
            continue
        h_tr = coarse_to_fine(best_h, best_n, n_tr, rng, t_limit=t_end3 - 5)
        c5_tr = compute_c5(h_tr)
        t_limit_tr = min(t_end3 - 3, time.time() + (t_end3 - time.time()) * 0.4)
        h_tr = lbfgsb_optimize(h_tr, n_tr, alpha_start=50, alpha_end=400,
                               penalty=500.0, maxiter=200, t_limit=t_limit_tr)
        h_tr = project_to_sum(h_tr, n_tr / 2.0)
        corr_tr = np.correlate(h_tr, 1.0 - h_tr, mode="full")
        while time.time() < t_end3 - 2:
            h_tr, corr_tr, _ = pair_descent_sweep(h_tr, corr_tr, rng, n_cand=10)
            c5_tr = compute_c5(h_tr)
            if c5_tr >= best_c5 and (corr_tr is None or not _):
                break
            if c5_tr < best_c5:
                best_c5 = c5_tr
                best_h = h_tr.copy()
                best_n = n_tr

    # Phase 4: Final L-BFGS-B polish + pair descent (10% budget)
    t_end4 = t_start + budget_s - 5
    h_fin = best_h.copy()
    n_fin = best_n
    h_fin = lbfgsb_optimize(h_fin, n_fin, alpha_start=100, alpha_end=500,
                            penalty=1000.0, maxiter=300, t_limit=t_end4 - 3)
    h_fin = project_to_sum(h_fin, n_fin / 2.0)
    c5_fin = compute_c5(h_fin)
    if c5_fin < best_c5:
        best_c5 = c5_fin
        best_h = h_fin.copy()
        best_n = n_fin

    corr_fin = np.correlate(best_h, 1.0 - best_h, mode="full")
    while time.time() < t_end4:
        best_h, corr_fin, _ = pair_descent_sweep(best_h, corr_fin, rng, n_cand=16, max_rounds=1)
        c5_fin = compute_c5(best_h)
        best_c5 = c5_fin

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
