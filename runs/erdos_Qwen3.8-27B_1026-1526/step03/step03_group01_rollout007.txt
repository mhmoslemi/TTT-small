    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    s = np.sum(exp_v)
    w = exp_v / s
    obj = m_val + (max_sh + np.log(s)) / alpha
    
    nj = np.arange(n)[:, None]
    nm = np.arange(2 * n - 1)[None, :]
    fi = nj - nm + n - 1
    si = nj + nm - n + 1
    vf = (fi >= 0) & (fi < n)
    vs = (si >= 0) & (si < n)
    fc = np.clip(fi, 0, n - 1)
    sc = np.clip(si, 0, n - 1)
    grad = np.sum(vf * (1.0 - h[fc]) * w[None, :] - vs * h[sc] * w[None, :], axis=1)
    return obj, grad

def slsqp_smooth(h0, n, ts, alpha, maxiter=2000, ftol=1e-14, gtol=1e-12):
    def f(x):
        return smooth_max_obj_grad(x, alpha)[0]
    def g(x):
        return smooth_max_obj_grad(x, alpha)[1]
    def cfun(x):
        return np.sum(x) - ts
    def cjac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(f, h0.copy(), method='SLSQP', jac=g, bounds=bnds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': ftol, 'gtol': gtol})
        result = project_to_sum(r.x, ts)
        return result
    except Exception:
        return project_to_sum(h0, ts)

def slsqp_tvar(h0, n, ts, alpha, maxiter=1000):
    """Minimize t subject to smooth_max constraint and box/sum constraints."""
    # Variables: [h_0, ..., h_{n-1}, t]
    n_total = n + 1
    ts_total = ts  # sum constraint on first n variables
    
    def f(x):
        h = x[:n]
        t = x[n]
        _, _ = smooth_max_obj_grad(h, alpha)  # just for compatibility
        return t
    
    def g(x):
        h = x[:n]
        # gradient w.r.t. h: from the smooth max constraint (as a barrier)
        _, grad_h = smooth_max_obj_grad(h, alpha)
        grad_t = np.zeros(1)
        grad_t[0] = 1.0
        return np.concatenate([grad_h * 0.01, grad_t])  # small weight on h grad
    
    # Actually this approach is tricky. Let me use the standard form:
    # min t s.t. smooth_max(h) - t <= 0, sum(h) = ts, 0<=h<=1, t>=0
    
    def f2(x):
        return x[n]
    
    def g2(x):
        hg = np.zeros(n)
        # Actually for SLSQP, the objective gradient should be just d(t)/dx = [0,...,0,1]
        # The constraint gradient is handled by SLSQP
        return np.concatenate([np.zeros(n), [1.0]])
    
    def con_fun(x):
        h = x[:n]
        t = x[n]
        obj_val, _ = smooth_max_obj_grad(h, alpha)
        return obj_val - t  # should be <= 0
    
    def con_jac(x):
        h = x[:n]
        _, grad_h = smooth_max_obj_grad(h, alpha)
        return np.concatenate([grad_h, [-1.0]])
    
    def sum_fun(x):
        return np.sum(x[:n]) - ts
    
    def sum_jac(x):
        return np.concatenate([np.ones(n), [0.0]])
    
    cons = [
        {'type': 'ineq', 'fun': lambda x: x[n] - smooth_max_obj_grad(x[:n], alpha)[0], 
         'jac': lambda x: np.concatenate([-smooth_max_obj_grad(x[:n], alpha)[1], [1.0]])},
        {'type': 'eq', 'fun': sum_fun, 'jac': sum_jac}
    ]
    bnds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    
    x0 = np.concatenate([h0, [compute_c5(h0) + 0.1]])
    
    try:
        r = minimize(f2, x0, method='SLSQP', jac=g2, bounds=bnds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        h_result = project_to_sum(r.x[:n], ts)
        return h_result, r.x[n]
    except Exception:

def block_slsqp(h0, n, ts, alpha, block_size, maxiter_per_block=500):
    """Optimize blocks of variables at a time."""
    h = h0.copy()
    n_blocks = n // block_size
    for b in range(n_blocks):
        start = b * block_size
        end = start + block_size
        fixed_sum = ts - np.sum(h[:start]) - np.sum(h[end:])
        
        def f_sub(x):
            h_full = h.copy()
            h_full[start:end] = x
            return smooth_max_obj_grad(h_full, alpha)[0]
        
        def g_sub(x):
            h_full = h.copy()
            h_full[start:end] = x
            _, grad_full = smooth_max_obj_grad(h_full, alpha)
            return grad_full[start:end]
        
        def sum_sub(x):
            return np.sum(x) - fixed_sum
        
        def sum_sub_jac(x):
            return np.ones(block_size)
        
        cons = [{'type': 'eq', 'fun': sum_sub, 'jac': sum_sub_jac}]
        bnds = [(0.0, 1.0)] * block_size
        
        try:
            r = minimize(f_sub, h[start:end].copy(), method='SLSQP', jac=g_sub,
                         bounds=bnds, constraints=cons,
                         options={'maxiter': maxiter_per_block, 'ftol': 1e-13})
            h[start:end] = r.x
        except Exception:
            pass
    
    return project_to_sum(h, ts)

def stochastic_block_cd(h, n, ts, rng, n_steps=30000, block_sizes=(1, 3, 5, 10)):
    """Stochastic coordinate descent with block moves."""
    h = h.copy()
    c5_cur = compute_c5(h)
    temp = 0.1
    for step in range(n_steps):
        bs = rng.choice(block_sizes)
        i = rng.randint(0, n - bs + 1)
        j = rng.randint(0, n - bs + 1)
        if i == j:
            continue
        delta = rng.uniform(0.05, 0.3)
        # Move: h[i:i+bs] -= delta*factor, h[j:j+bs] += delta*factor
        # Check feasibility
        max_delta = min(np.min(h[i:i+bs]), (1.0 - np.max(h[j:j+bs])))
        if max_delta < 1e-6:
            continue
        actual_delta = min(delta, max_delta * 0.8)
        h[i:i+bs] -= actual_delta
        h[j:j+bs] += actual_delta
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, ts)
        c5_new = compute_c5(h)
        if c5_new < c5_cur:
            c5_cur = c5_new
        else:
            accept_prob = np.exp(-(c5_new - c5_cur) / temp)
            if rng.random() < accept_prob:
                c5_cur = c5_new
            else:
                h[i:i+bs] += actual_delta
                h[j:j+bs] -= actual_delta
                h = np.clip(h, 0.0, 1.0)
                h = project_to_sum(h, ts)
        temp = max(1e-6, 0.1 * np.exp(-step / 3000.0))
    return h, c5_cur

def upsample_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old, period=2.0)
    return h_new

                h[i:i+bs] += actual_delta
                h[j:j+bs] -= actual_delta
                h = np.clip(h, 0.0, 1.0)
                h = project_to_sum(h, ts)
        temp = max(1e-7, 0.08 * np.exp(-step / 2500.0))
    return h, c5_cur

def upsample_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old, period=2.0)
    return h_new

def freq_perturb(h, rng, n_harm, amp):
    n = len(h)
    mean_h = np.mean(h)
    H = np.fft.rfft(h - mean_h)
    upper = min(n_harm, n // 2 - 1)
    for k in range(1, upper + 1):
        H[k] += amp * (rng.randn() + 1j * rng.randn())
    h_new = np.fft.irfft(H, n=n) + mean_h
    return h_new

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    def time_left():
        return budget_s - (time.time() - t_start)
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    try:
        if 'initial_h_values' in globals() and initial_h_values is not None:
            best_h = np.array(initial_h_values, dtype=float).copy()
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
    except Exception:
        pass
    if best_h is None:
        best_n = 480
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP continuation at parent resolution
    n1 = best_n
    ts1 = n1 / 2.0
    for alpha in [500, 2000, 8000, 30000]:
        if time_left() < 50:
            break
        h_new = slsqp_smooth(best_h, n1, ts1, alpha, maxiter=1200)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 2: Upsample to 600
    if time_left() > 80:
        n2 = 600
        ts2 = n2 / 2.0
        h_up = upsample_h(best_h, best_n, n2)
        h_up = project_to_sum(h_up, ts2)
        c5_up = compute_c5(h_up)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up.copy()
            best_n = n2
        for alpha in [500, 3000, 10000]:
            if time_left() < 35:
                break
            h_opt = slsqp_smooth(h_up, n2, ts2, alpha, maxiter=800)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n2
            h_up = h_opt

    # Phase 3: Upsample to 720
    if time_left() > 100:
        n3 = 720
        ts3 = n3 / 2.0
        h_up = upsample_h(best_h, best_n, n3)
        h_up = project_to_sum(h_up, ts3)
        c5_up = compute_c5(h_up)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up.copy()
            best_n = n3
        for alpha in [500, 2000, 8000]:
            if time_left() < 35:
                break
            h_opt = slsqp_smooth(h_up, n3, ts3, alpha, maxiter=700)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n3
            h_up = h_opt

    # Phase 4: Block SLSQP at best resolution
    n_b = best_n
    ts_b = n_b / 2.0
    for bs in [36, 72]:
        if time_left() < 50:
            break
        h_blk = block_slsqp(best_h, n_b, ts_b, 5000, bs, maxiter_per_block=150)
        c5_blk = compute_c5(h_blk)
        if c5_blk < best_c5:
            best_c5 = c5_blk
            best_h = h_blk.copy()

    # Phase 5: Stochastic block CD
    if time_left() > 100:
        h_st, c5_st = stochastic_block_cd(best_h, n_b, ts_b, rng, n_steps=35000, block_sizes=(1, 2, 3, 5, 8))
        if c5_st < best_c5:
            best_c5 = c5_st
            best_h = h_st.copy()
        if time_left() > 40:
            h_pol = slsqp_smooth(best_h, n_b, ts_b, 15000, maxiter=1000)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Phase 6: Frequency perturbation + re-optimization
    if time_left() > 80:
        for trial in range(15):
            if time_left() < 25:
                break
            amp = 0.04 * (1.0 - trial / 20.0) + 0.01
            nf = max(5, n_b // 40)
            h_sh = freq_perturb(best_h, rng, nf, amp)
            h_sh = project_to_sum(h_sh, ts_b)
            c5_sh = compute_c5(h_sh)
            if c5_sh < best_c5:
                best_c5 = c5_sh
                best_h = h_sh.copy()
            if time_left() > 20:
                h_ot = slsqp_smooth(h_sh, n_b, ts_b, 5000, maxiter=500)
                c5_ot = compute_c5(h_ot)
                if c5_ot < best_c5:
                    best_c5 = c5_ot
                    best_h = h_ot.copy()
                h_sh = h_ot

    # Phase 7: T-variable SLSQP formulation
    if time_left() > 60:
        for alpha in [5000, 30000]:
            if time_left() < 35:
                break
            h_tv = slsqp_tvar(best_h, n_b, ts_b, alpha, maxiter=800)
            c5_tv = compute_c5(h_tv)
            if c5_tv < best_c5:
                best_c5 = c5_tv
                best_h = h_tv.copy()

    # Phase 8: Ultra-high-alpha SLSQP polish
    if time_left() > 50:
        for alpha in [50000, 150000, 500000]:
            if time_left() < 25:
                break
            h_f = slsqp_smooth(best_h, n_b, ts_b, alpha, maxiter=1500)
            c5_f = compute_c5(h_f)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f.copy()

    # Phase 9: Fine stochastic refinement
    if time_left() > 50:
        h_s, c5_s = stochastic_block_cd(best_h, n_b, ts_b, rng, n_steps=15000, block_sizes=(1, 2, 3))
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()
        if time_left() > 25:
            h_fp = slsqp_smooth(best_h, n_b, ts_b, 200000, maxiter=1200)
            c5_fp = compute_c5(h_fp)
            if c5_fp < best_c5:
                best_c5 = c5_fp
                best_h = h_fp.copy()

    # Phase 10: Second t-variable pass + final polish
    if time_left() > 40:
        h_tv2 = slsqp_tvar(best_h, n_b, ts_b, 80000, maxiter=1000)
        c5_tv2 = compute_c5(h_tv2)
        if c5_tv2 < best_c5:
            best_c5 = c5_tv2
            best_h = h_tv2.copy()
        if time_left() > 20:
            h_fin = slsqp_smooth(best_h, n_b, ts_b, 300000, maxiter=1000)
            c5_fin = compute_c5(h_fin)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin.copy()

    # Final validation
    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    for _ in range(500):
        diff = best_n / 2.0 - np.sum(best_h)
        if abs(diff) < 1e-14:
            break
        mask = (best_h > 1e-10) & (best_h < 1.0 - 1e-10)
        if np.any(mask):
            best_h[mask] += diff / len(mask)
            best_h = np.clip(best_h, 0.0, 1.0)
        else:
            break
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
