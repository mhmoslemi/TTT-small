```python
import numpy as np
from scipy.optimize import linprog, minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_B_matrix(h, n):
    """Compute full B matrix: B[j, L] for all j, L."""
    nm1 = n - 1
    B = np.zeros((n, 2 * n - 1))
    j_range = np.arange(n)
    for L in range(2 * n - 1):
        idx1 = j_range + L - nm1
        valid1 = (idx1 >= 0) & (idx1 < n)
        B[valid1, L] += (1.0 - h[idx1[valid1]])
        idx2 = j_range - L + nm1
        valid2 = (idx2 >= 0) & (idx2 < n)
        B[valid2, L] -= h[idx2[valid2]]
    return B

def block_lp_update(h, s, m, B, corr, n):
    """Solve LP to optimally update block h[s:s+m] to minimize max correlation."""
    # Variables: x[0:m] = new h values, x[m] = t
    # rhs for inequality constraints
    block_idx = np.arange(s, s + m)
    B_block = B[block_idx, :]  # shape (m, 2n-1)
    h_block = h[block_idx]
    
    # rhs[L] = sum_i h_block[i] * B_block[i, L] - corr[L]
    rhs = B_block.T @ h_block - corr  # shape (2n-1,)
    
    # A_ub: shape (2n-1, m+1)
    A_ub = np.hstack([B_block.T, -np.ones((2 * n - 1, 1))])
    
    # Equality: sum x[0:m] = sum h_block
    A_eq = np.zeros((1, m + 1))
    A_eq[0, :m] = 1.0
    b_eq = np.sum(h_block)
    
    # Bounds
    bounds = [(0, 1)] * m + [(None, None)]
    
    c_obj = np.zeros(m + 1)
    c_obj[m] = 1.0
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=rhs, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        h[block_idx] = np.clip(res.x[:m], 0, 1)
        # Project to maintain exact sum
        h_block_new = h[block_idx]
        diff = b_eq - np.sum(h_block_new)
        if abs(diff) > 1e-10:
            h[block_idx] += diff / m
            h[block_idx] = np.clip(h[block_idx], 0, 1)
        return res.x[m]
    return None
```

```python
    L_row = np.arange(N)[None, :]  # (1, N)
```
