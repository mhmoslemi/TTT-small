    # = sum_m w[m] * (1-h[si]) for vs
    term1 = np.sum(vs * (1.0 - h[si]) * w[None, :], axis=1)
    # Second term: sum_m w[m] * h[j'-L(m)] for valid j'-L
    # = sum_m w[m] * h[fi] for vf
    term2 = np.sum(vf * h[fi] * w[None, :], axis=1)
    
    grad = term1 - term2
    return obj, grad

def lbfgs_opt(h0, n, target_sum, alpha, maxiter=5000):
    """Optimize using L-BFGS-B with projection."""
    def obj(x):
        return smooth_max_obj(x, alpha)
    def grad_fn(x):
        return smooth_max_grad(x, alpha)[1]
    
    # Use SLSQP with equality constraint for the sum
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum, 
             'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    
    try:
        result = minimize(obj, h0.copy(), method='SLSQP', jac=grad_fn,
                         bounds=bnds, constraints=cons,
                         options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-12})

    n = len(h)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break

            h[mask] += diff / n_mask
        else:
            break
        h = np.clip(h, 0.0, 1.0)

def smooth_max_and_grad(h, alpha):
    """Compute log-sum-exp smooth max of correlation and its gradient."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = (corr - m) * alpha
    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    w = exp_v / np.sum(exp_v)
    obj = m + (max_sh + np.log(np.sum(exp_v))) / alpha
    
    nj = np.arange(n)[:, None]
    nm = np.arange(2 * n - 1)[None, :]
    fi = nj - nm + n - 1
    si = nj + nm - n + 1
    vf = (fi >= 0) & (fi < n)
    vs = (si >= 0) & (si < n)
    fi_c = np.clip(fi, 0, n - 1)
    si_c = np.clip(si, 0, n - 1)
    
    grad = np.sum(vs * (1.0 - h[si_c]) * w[None, :], axis=1) - np.sum(vf * h[fi_c] * w[None, :], axis=1)

        # The contribution to corr[best_lag] from h[j] is:
        # h[j] * (1-h[j+L]) - h[j-L] where L = best_lag - (n-1)
        # To reduce corr[best_lag], we want to:
        # - Decrease h[j] where h[j]*(1-h[j+L]) is large
        # - Increase h[j] where h[j-L] is large (reduces the second term)

        # Find the j that contributes most to the correlation
        # Contribution of h[j] to corr[best_lag]:
        # If 0 ≤ j+L < n: term is h[j]*(1-h[j+L])
        # If 0 ≤ j-L < n: term is -h[j-L] (from the chain rule)
        # Net: (1-h[j+L]) - h[j-L] (with boundary handling)
        
        # For a simple greedy step: find j where decreasing h[j] most reduces corr[best_lag]
        # Decreasing h[j] by delta reduces corr[best_lag] by delta * (1-h[j+L]) (if valid)
        # And increases corr[best_lag] by delta * h[j-L] (if valid, because of the -h[j-L] term)
        # Net change: -delta * [(1-h[j+L]) - h[j-L]]
        # We want this to be negative, so we want (1-h[j+L]) - h[j-L] > 0
        
        # Compute the "gain" for each j
        si = np.arange(n) + L
        fi = np.arange(n) - L
        valid_s = (si >= 0) & (si < n)
        valid_f = (fi >= 0) & (fi < n)
        si_c = np.clip(si, 0, n-1)
        fi_c = np.clip(fi, 0, n-1)
        
        gain = np.zeros(n)
        gain[valid_s] += (1.0 - h[si_c[valid_s]])
        gain[valid_f] -= h[fi_c[valid_f]]
        
        # Find j with maximum gain (most beneficial to decrease h[j])
        # But we also need to maintain the sum constraint
        # If we decrease h[j] and increase h[k], the net effect on sum is 0
        
        # For simplicity, do a mass transfer: decrease h[j] by delta, increase h[k] by delta
        # Choose j with max gain, k with min gain (or 0)

        k_min = np.argmin(gain)

            c5_val = max_new

            break  # Stuck
