import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_B(h, j, n):
    nm1 = n - 1
    B = np.zeros(2 * n - 1)
    k_start = max(0, nm1 - j)
    k_end = min(2 * n - 2, 2 * nm1 - j)
    if k_start <= k_end:
        B[k_start:k_end + 1] += (1.0 - h[j - nm1 + np.arange(k_start, k_end + 1)])
    k_start2 = max(0, j)
    k_end2 = min(2 * n - 2, j + nm1)
    if k_start2 <= k_end2:
        B[k_start2:k_end2 + 1] += -h[j + nm1 - np.arange(k_start2, k_end2 + 1)]
    return B

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m_val))
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def adam_optimize(h0, target_sum, alphas, lrs, max_steps_each=1500):
    n = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    for alpha in alphas:
        for lr in lrs:
            h = h0.copy()
            m_vec = np.zeros(n)
            v_vec = np.zeros(n)
            beta1, beta2, eps = 0.85, 0.999, 1e-8
            best_local = compute_c5(h)
            best_local_h = h.copy()
            for t in range(1, max_steps_each + 1):
                obj, grad = smooth_max_and_grad(h, alpha)
                grad_proj = grad - np.mean(grad)
                m_vec = beta1 * m_vec + (1.0 - beta1) * grad_proj
                v_vec = beta2 * v_vec + (1.0 - beta2) * grad_proj ** 2
                m_hat = m_vec / (1.0 - beta1 ** t)
                v_hat = v_vec / (1.0 - beta2 ** t)
                h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
                h = project_to_sum(h, target_sum)
                c5_now = compute_c5(h)
                if c5_now < best_local:
                    best_local = c5_now
                    best_local_h = h.copy()
            if best_local < best_c5:
                best_c5 = best_local
                best_h = best_local_h.copy()
    return best_h, best_c5

def pair_descent_sweep(h, corr, rng, n_cand=10, bisection_iters=38):
    n = len(h)
    order = np.arange(n)
    rng.shuffle(order)
    improved = False
    for j in order:
        B_j = compute_B(h, j, n)
        best_val = float(np.max(corr))
        best_delta = 0.0
        best_D = None
        best_k = -1
        k_set = [(j + 1) % n, (j - 1 + n) % n, (j + n // 2) % n, (j + n // 3) % n, (j + 2 * n // 3) % n]
        for _ in range(max(0, n_cand - 5)):
            k_set.append(rng.randint(n))
        seen = {j}
        for k in k_set:
            if k < 0 or k >= n or k in seen:
                continue
            seen.add(k)
            if k == j:
                continue
            B_k = compute_B(h, k, n)
            D = B_j - B_k
            delta_lo = max(-h[j], h[k] - 1.0)
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-14:
                continue
            lo = delta_lo
            hi = delta_hi
            for _ in range(bisection_iters):
                mid = (lo + hi) * 0.5
                if np.max(corr + mid * D) <= np.max(corr + lo * D):
                    lo = mid
                else:
                    hi = mid
            delta_opt = (lo + hi) * 0.5
            val_opt = float(np.max(corr + delta_opt * D))
            if val_opt < best_val - 1e-14:
                best_val = val_opt
                best_delta = delta_opt
                best_D = D.copy()
                best_k = k
        if best_k >= 0 and abs(best_delta) > 1e-14:
            corr = corr + best_delta * best_D
            h[j] = np.clip(h[j] + best_delta, 0.0, 1.0)
            h[best_k] = np.clip(h[best_k] - best_delta, 0.0, 1.0)
            improved = True
    return h, corr, improved

def basin_hopping(h, rng, target_sum, t_end, n_desc_sweeps):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_h = h.copy()
    current_c5 = compute_c5(current_h)
    
    perturb_sizes = [0.03, 0.05, 0.08, 0.12, 0.18, 0.25]
    
    while time.time() < t_end:
        corr = np.correlate(current_h, 1.0 - current_h, mode="full")
        for _ in range(n_desc_sweeps):
            if time.time() >= t_end:
                break
            current_h, corr, _ = pair_descent_sweep(current_h, corr, rng, n_cand=12)
        current_c5 = compute_c5(current_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
        
        k = int(2 + rng.randint(len(perturb_sizes)))
        noise_amp = perturb_sizes[rng.randint(len(perturb_sizes))]
        n_perturb = max(4, n // k)
        idx = rng.choice(n, size=min(n_perturb, n), replace=False)
        current_h[idx] += rng.uniform(-noise_amp, noise_amp, size=len(idx))
        current_h = project_to_sum(current_h, target_sum)
    
    return best_h, best_c5

def slsqp_refine(h0, target_sum, alpha, maxiter):
    n = len(h0)
    def f(x):
        return smooth_max_and_grad(x, alpha)[0]
    def g(x):
        return smooth_max_and_grad(x, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)
