from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_C(h, n):
    """C[j, m] = coefficient of h[j] in corr[m] (derivative of corr[m] w.r.t. h[j])"""
    m_arr = np.arange(2*n-1)
    j_arr = np.arange(n)
    s1 = j_arr[:, None] + n - 1 - m_arr[None, :]
    valid1 = (s1 >= 0) & (s1 < n)
    s2 = j_arr[:, None] - n + 1 + m_arr[None, :]
    valid2 = (s2 >= 0) & (s2 < n)
    c1 = np.clip(s1, 0, n-1)
    c2 = np.clip(s2, 0, n-1)
    C = valid1.astype(float) * (1.0 - h[c1]) - valid2.astype(float) * h[c2]
    return C

def smooth_objective(h, n, alpha):
    dx = 2.0 / n
    corr = compute_corr(h)
    val = alpha * corr * dx
    val_shifted = val - np.max(val)
    logsumexp = np.max(val) + np.log(np.sum(np.exp(val_shifted)))
    return logsumexp / alpha

def smooth_grad(h, n, alpha):
    dx = 2.0 / n
    corr = compute_corr(h)
    val = alpha * corr * dx
    val_shifted = val - np.max(val)
    w = np.exp(val_shifted)
    w /= np.sum(w)
    C = compute_C(h, n)
    grad = dx * (C @ w)
    # Center the gradient (project out the sum constraint direction)
    grad -= np.mean(grad)
    return grad, np.max(corr) * dx

def solve_1d_max(A, D, lo, hi):
    """Minimize max(A + D*t) over t in [lo, hi]"""
    if hi - lo < 1e-14:
        return (lo + hi) * 0.5
    # Grid search for initial bracket
    t_grid = np.linspace(lo, hi, 16)
    vals = np.max(A[None, :] + D[None, :] * t_grid[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        return lo
    if best_idx == len(t_grid) - 1:
        return hi
    lo_r = t_grid[best_idx - 1]
    hi_r = t_grid[best_idx + 1]
    for _ in range(35):
        mid = (lo_r + hi_r) * 0.5
        if np.max(A + D * mid) <= np.max(A + D * lo_r):
            lo_r = mid
        else:
            hi_r = mid
    return (lo_r + hi_r) * 0.5

def pair_descent_sweep(h, C, corr, rng, n, n_cand=10):
    """One full sweep of pair coordinate descent"""
    order = np.random.permutation(n) if hasattr(rng, 'permutation') else rng.permutation(n)
    improved = False
    for j in order:
        if h[j] <= 1e-12 or h[j] >= 1.0 - 1e-12:
            continue
        best_val = np.max(corr)
        best_delta = 0.0
        best_D = None
        best_k = -1
        # Candidate k's
        k_set = [(j + 1) % n, (j + n - 1) % n, (j + n // 2) % n, (j + n // 3) % n, (j + 2 * n // 3) % n]
        for _ in range(max(0, n_cand - 5)):
            k_set.append(int(rng.randint(n)))
        seen = set()
        for k in k_set:
            if k == j or k in seen:

            seen.add(k)
            if h[k] <= 1e-12:
                continue
            D = C[j] - C[k]
            delta_lo = max(-h[j], -h[k])
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-13:
                continue
            delta_opt = solve_1d_max(corr, D, delta_lo, delta_hi)
            val_opt = np.max(corr + D * delta_opt)
            if val_opt < best_val - 1e-14:
                best_val = val_opt
                best_delta = delta_opt
                best_D = D
                best_k = k
        if best_k >= 0 and abs(best_delta) > 1e-13:

            h[j] += best_delta
            h[best_k] -= best_delta
            h[j] = max(0.0, min(1.0, h[j]))
            h[best_k] = max(0.0, min(1.0, h[best_k]))
            improved = True
    return h, corr, improved
