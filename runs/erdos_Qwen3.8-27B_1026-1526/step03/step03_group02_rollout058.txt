def binary_sa_improved(h_start, n, target_sum, t_end, rng):
    h = h_start.copy()
    c5_curr = compute_c5(h)
    best_c5 = c5_curr
    best_h = h.copy()
    T = 0.015
    min_T = 0.002
    no_improve = 0
    restart_count = 0
    
    while time.time() < t_end:
        # Choose move type
        move_type = rng.random()
        accepted = False
        revert_data = None
        
        if move_type < 0.6:
            # Single swap
            i = rng.randint(n)
            j = rng.randint(n)
            if h[i] == h[j]:
                continue
            h[i], h[j] = h[j], h[i]
            revert_data = (i, j)
        elif move_type < 0.85:
            # Block swap (length 2-4)
            blen = rng.randint(2, 5)
            i = rng.randint(0, n - blen)
            j = rng.randint(0, n - blen)
            if abs(i - j) < blen:
                continue
            block_i = h[i:i+blen].copy()
            block_j = h[j:j+blen].copy()
            if block_i.sum() != block_j.sum():
                continue
            h[i:i+blen] = block_j
            h[j:j+blen] = block_i
            revert_data = (i, j, blen, block_i, block_j)
        else:
            # Triple swap: find three positions with 2 ones, 1 zero (or vice versa)
            # and rotate them
            i = rng.randint(n)
            j = rng.randint(n)
            k = rng.randint(n)
            if len(set([i, j, k])) < 3:
                continue
            vals = [h[i], h[j], h[k]]
            if sum(vals) not in [1, 2]:
                continue
            # Rotate: h[i] -> h[j] -> h[k] -> h[i]
            h[i], h[j], h[k] = h[j], h[k], h[i]
            revert_data = (i, j, k, vals)
        
        c5_new = compute_c5(h)
        delta = c5_new - c5_curr
        if delta < 0 or (delta > 0 and np.log(rng.uniform()) < delta / T):
            accepted = True
            c5_curr = c5_new
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # Revert
            if revert_data is not None:
                rd = revert_data
                if len(rd) == 2:
                    i, j = rd
                    h[i], h[j] = h[j], h[i]
                elif len(rd) == 5:
                    i, j, blen, bi, bj = rd
                    h[i:i+blen] = bi
                    h[j:j+blen] = bj
                elif len(rd) == 4:
                    i, j, k, vals = rd
                    h[i], h[j], h[k] = vals[0], vals[1], vals[2]
        
        if no_improve > 50000:
            h = best_h.copy()
            c5_curr = best_c5
            T = min(0.02, T * 3.0)
            no_improve = 0
            restart_count += 1
        
        T = max(T * 0.999995, min_T)
    
    return best_h, best_c5

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def make_binary(h, n, target_sum, rng):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones == target_sum:
        return h_bin
    if n_ones > target_sum:
        excess = n_ones - target_sum
        ones_idx = np.where(h_bin == 1.0)[0]
        order = np.argsort(h[ones_idx])[:excess]
        h_bin[ones_idx[order]] = 0.0
    else:
        deficit = target_sum - n_ones
        zeros_idx = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h[zeros_idx])[:deficit]
        h_bin[zeros_idx[order]] = 1.0
    return h_bin

def upsample(h_src, n_src, n_dst, rng):

    h_dst = np.zeros(n_dst)
    for i in range(n_src):
        start = int(round(i * n_dst / n_src))
        end = int(round((i + 1) * n_dst / n_src))
        h_dst[start:end] = h_src[i]
    return h_dst

    n = len(h)
