def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break

            h[mask] += diff / np.sum(mask)
        else:

def compute_B_single(h, j, n):
    nm1 = n - 1
    B = np.zeros(2 * n - 1)

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m_val))
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

        return smooth_max_and_grad(h, alpha)[1]

    ks = max(0, nm1 - j)
    ke = min(2 * n - 2, nm1 + j)
    if ks <= ke:
        idx = np.arange(ks, ke + 1)
        B[idx] += (1.0 - h[j - nm1 + idx])
    ks2 = max(0, j - nm1)
    ke2 = min(2 * n - 2, j + nm1)
    if ks2 <= ke2:
        idx = np.arange(ks2, ke2 + 1)
        B[idx] -= h[j + nm1 - idx]
    return B

def solve_1d_optimal(corr, D, delta_lo, delta_hi, n_iter=45):
    if delta_hi - delta_lo < 1e-14:
        return (delta_lo + delta_hi) * 0.5
    for _ in range(n_iter):
        mid = (delta_lo + delta_hi) * 0.5
        val = corr + D * mid
        m_star = np.argmax(val)
        ds = D[m_star]
        if ds < -1e-12:
            delta_lo = mid
        elif ds > 1e-12:
            delta_hi = mid
        else:
            break
    return (delta_lo + delta_hi) * 0.5

def compute_s_all(h, w, n):
    nm1 = n - 1
    s = np.zeros(n)
    for k in range(n):
        ks = max(0, nm1 - k)
        ke = min(2 * n - 2, nm1 + k)
        if ks <= ke:
            idx = np.arange(ks, ke + 1)
            s[k] += np.dot(w[idx], 1.0 - h[k - nm1 + idx])
        ks2 = max(0, k - nm1)
        ke2 = min(2 * n - 2, k + nm1)
        if ks2 <= ke2:
            idx = np.arange(ks2, ke2 + 1)
            s[k] -= np.dot(w[idx], h[k + nm1 - idx])
    return s

def guided_sweep(h, corr, n, alpha, n_top=5):
    m_val = np.max(corr)
    exp_arg = alpha * (corr - m_val)
    exp_arg = np.clip(exp_arg, -700, 700)
    w = np.exp(exp_arg)
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w /= w_sum

    s = compute_s_all(h, w, n)
    order = np.argsort(s)[::-1]
    n_moves = 0
    current_max = np.max(corr)

    for j_pos in range(n):
        j_idx = int(order[j_pos])
        if j_idx < 0 or j_idx >= n:
            continue
        found = False
        for trial in range(n_top):
            k_idx = int(order[n - 1 - trial])
            if k_idx == j_idx:
                continue
            delta_lo = max(-h[j_idx], h[k_idx] - 1.0)
            delta_hi = min(1.0 - h[j_idx], h[k_idx])
            if delta_hi - delta_lo < 1e-13:
                continue
            B_j = compute_B_single(h, j_idx, n)
            B_k = compute_B_single(h, k_idx, n)
            D = B_j - B_k
            delta_opt = solve_1d_optimal(corr, D, delta_lo, delta_hi)
            new_val = np.max(corr + D * delta_opt)
            if new_val < current_max - 1e-12:
                corr += D * delta_opt
                h[j_idx] = min(1.0, max(0.0, h[j_idx] + delta_opt))
                h[k_idx] = min(1.0, max(0.0, h[k_idx] - delta_opt))
                current_max = new_val
                n_moves += 1
                found = True
                break
        if not found and n_top < n:
            k_idx = int(order[n // 2 + (j_pos % (n // 2))])
            if k_idx != j_idx:
                delta_lo = max(-h[j_idx], h[k_idx] - 1.0)
                delta_hi = min(1.0 - h[j_idx], h[k_idx])
                if delta_hi - delta_lo > 1e-13:
                    B_j = compute_B_single(h, j_idx, n)
                    B_k = compute_B_single(h, k_idx, n)
                    D = B_j - B_k
                    delta_opt = solve_1d_optimal(corr, D, delta_lo, delta_hi)
                    new_val = np.max(corr + D * delta_opt)
                    if new_val < current_max - 1e-12:
                        corr += D * delta_opt
                        h[j_idx] = min(1.0, max(0.0, h[j_idx] + delta_opt))
                        h[k_idx] = min(1.0, max(0.0, h[k_idx] - delta_opt))
                        current_max = new_val
                        n_moves += 1
    return h, corr, n_moves

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m_val))
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def slsqp_local(h0, n, target_sum, alpha, maxiter):
    def f(h):
        return smooth_max_and_grad(h, alpha)[0]
    def g(h):
        return smooth_max_and_grad(h, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float).copy()
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    t_end1 = t_start + 0.55 * budget_s
    h_work = best_h.copy()
    n_w = len(h_work)
    corr_w = np.correlate(h_work, 1.0 - h_work, mode="full")
    stall_count = 0
    sweep_count = 0

    while time.time() < t_end1:
        progress = min(1.0, sweep_count / 600.0)
        alpha = 30.0 + 470.0 * progress
        h_work, corr_w, n_moves = guided_sweep(h_work, corr_w, n_w, alpha, n_top=5)
        sweep_count += 1
        c5_w = float(np.max(corr_w) * 2.0 / n_w)
        if c5_w < best_c5 - 1e-10:
            best_c5 = c5_w
            best_h = h_work.copy()
            best_n = n_w
            stall_count = 0
        else:
            stall_count += 1
        if stall_count >= 150:
            noise_scale = 0.02 + 0.06 * min(1.0, stall_count / 400.0)
            h_work = project_to_sum(h_work + rng.uniform(-noise_scale, noise_scale, n_w), n_w / 2.0)
            corr_w = np.correlate(h_work, 1.0 - h_work, mode="full")
            h_work, corr_w, _ = guided_sweep(h_work, corr_w, n_w, 50.0, n_top=8)
            stall_count = 0
        if stall_count >= 500:
            h_work = project_to_sum(best_h + rng.uniform(-0.05, 0.05, n_w), n_w / 2.0)
            corr_w = np.correlate(h_work, 1.0 - h_work, mode="full")
            stall_count = 0

    t_end2 = t_start + 0.75 * budget_s
    resolutions = [160, 200, 240, 280, 320, 400, 480]
    for n_tr in resolutions:
        if time.time() >= t_end2:
            break
        ts_tr = n_tr / 2.0
        if abs(ts_tr - round(ts_tr)) > 1e-6:
            continue
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_tr, endpoint=False)
        starts = []
        starts.append(project_to_sum(np.interp(x_new, x_old, best_h, period=2.0), ts_tr))
        x_alt = np.linspace(0, 2, n_tr, endpoint=False)
        n_four = n_tr // 2
        h_four = 0.5 + 0.3 * np.cos(2 * np.pi * x_alt) + 0.15 * np.cos(4 * np.pi * x_alt + 0.5)
        starts.append(project_to_sum(h_four, ts_tr))
        starts.append(project_to_sum(rng.uniform(0, 1, n_tr), ts_tr))
        for h_st in starts:
            if time.time() >= t_end2:
                break
            h_st = project_to_sum(h_st, ts_tr)
            corr_st = np.correlate(h_st, 1.0 - h_st, mode="full")
            max_sweeps = min(30, max(3, int((t_end2 - time.time()) / 4.0)))
            for sw in range(max_sweeps):
                if time.time() >= t_end2:
                    break
                a = 80.0 + 20.0 * sw
                h_st, corr_st, _ = guided_sweep(h_st, corr_st, n_tr, a, n_top=5)
            c5_st = float(np.max(corr_st) * 2.0 / n_tr)
            if c5_st < best_c5:
                best_c5 = c5_st
                best_h = h_st.copy()
                best_n = n_tr

    t_end3 = t_start + 0.93 * budget_s
    n_f = best_n
    ts_f = n_f / 2.0
    for alpha in [100, 500, 2000]:
        if time.time() >= t_end3:
            break
        h_pol = slsqp_local(best_h, n_f, ts_f, alpha, maxiter=200)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
            best_n = n_f

    h_fin = best_h.copy()
    n_fin = best_n
    corr_fin = np.correlate(h_fin, 1.0 - h_fin, mode="full")
    while time.time() < t_start + budget_s - 3:
        h_fin, corr_fin, _ = guided_sweep(h_fin, corr_fin, n_fin, 300.0, n_top=8)
        c5_fin = float(np.max(corr_fin) * 2.0 / n_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()
            best_n = n_fin

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
