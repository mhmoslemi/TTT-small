def project_to_sum(h, target_sum):

            h[mask] += diff / np.sum(mask)
        else:

    return h

def project_to_polytope(y, target_sum):
    """Project y onto {h: 0<=h<=1, sum(h)=target_sum} using bisection on offset."""
    n = len(y)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def project_to_polytope(y, target_sum):
    n = len(y)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(y - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    offset = (lo + hi) / 2.0
    proj = np.clip(y - offset, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(proj)
        if abs(diff) < 1e-14:
            break
        mask = (proj > 1e-13) & (proj < 1.0 - 1e-13)
        if np.any(mask):
            proj[mask] += diff / np.sum(mask)
        else:
            proj += diff / n
        proj = np.clip(proj, 0.0, 1.0)
    return proj

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    grad = np.zeros(n)
    thresh = 1e-7
    active = np.where(w > thresh)[0]
    for m in active:
        wm = w[m]
        lag = m - (n - 1)
        if lag == 0:
            grad[:] += wm * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += wm * (1.0 - h[:n - lag])
            grad[:n - lag] -= wm * h[lag:]
        else:
            k = -lag
            grad[:n - k] += wm * (1.0 - h[k:])
            grad[k:] -= wm * h[:n - k]
    return obj, grad

def nesterov_pg(h0, target_sum, alpha, max_iters, lr_init, momentum):
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    lr = lr_init
    for it in range(max_iters):
        y = h + momentum * v
        obj, grad = smooth_max_and_grad(y, alpha)
        y_step = y - lr * grad
        h_new = project_to_polytope(y_step, target_sum)
        obj_new, grad_new = smooth_max_and_grad(h_new, alpha)
        if obj_new > obj + 0.5 * lr * (np.dot((h_new - y), grad)):
            lr = lr * 0.5
            h_new = project_to_polytope(y - lr * grad, target_sum)
        else:
            lr = min(lr * 1.05, lr_init * 2.0)
        v = h_new - h
        h = h_new
        if it % 100 == 0:
            c5_cur = compute_c5(h)
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h.copy()
    c5_final = compute_c5(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    return best_h, best_c5

def multi_lag_coord_desc(h, target_sum, max_iters, step_frac, top_k):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    stale = 0
    for it in range(max_iters):
        corr = compute_corr(h)
        n_corr = len(corr)
        top_indices = np.argsort(corr)[-top_k:]
        g = np.zeros(n)
        for m in top_indices:
            lag = m - (n - 1)
            if lag == 0:
                g[:] += (1.0 - 2.0 * h)
            elif lag > 0:
                g[lag:] += (1.0 - h[:n - lag])
                g[:n - lag] -= h[lag:]
            else:
                k = -lag
                g[:n - k] += (1.0 - h[k:])
                g[k:] -= h[:n - k]
        g /= float(top_k)
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        g_inc = np.where(can_inc, g, np.inf)
        g_dec = np.where(can_dec, g, -np.inf)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        if i_inc == i_dec:
            break
        if g[i_inc] >= g[i_dec] - 1e-12:
            break
        max_move = min(1.0 - h[i_inc], h[i_dec])
        move = step_frac * max_move
        if move < 1e-10:
            break
        h[i_inc] += move
        h[i_dec] -= move
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
        c5_new = compute_c5(h)
        if c5_new < c5 - 1e-13:
            c5 = c5_new
            stale = 0
        else:
            h[i_inc] -= move
            h[i_dec] += move
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, target_sum)
            stale += 1
            if stale > 200:
                break
    return h, c5

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h, period=2.0)

def slsqp_refine(h0, n, target_sum, alpha, maxiter):
    def obj_fn(x):
        return smooth_max_and_grad(x, alpha)[0]
    def grad_fn(x):
        return smooth_max_and_grad(x, alpha)[1]
    def sum_fn(x):
        return np.sum(x) - target_sum
    def sum_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': sum_fn, 'jac': sum_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_fn, h0.copy(), method='SLSQP', jac=grad_fn,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def random_fourier_init(n, rng, n_harm):
    h = np.full(n, 0.5)
    for _ in range(n_harm):
        freq = rng.uniform(1.0, 6.0)
        phase = rng.uniform(0, 2.0 * np.pi)
        amp = rng.uniform(0.05, 0.2)
        x = np.linspace(0, 2.0, n, endpoint=False)
        h += amp * np.cos(2.0 * np.pi * freq * x + phase)
    return project_to_polytope(h, n / 2.0)

    return time.time() - t0
