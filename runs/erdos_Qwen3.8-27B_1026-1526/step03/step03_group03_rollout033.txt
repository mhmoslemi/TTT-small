For each j, for each k≠j, compute D = B_j - B_k, and find the optimal delta.

The optimal delta minimizes max(corr + delta * D) subject to delta_lo <= delta <= delta_hi.

This is a 1D piecewise-linear minimax problem. The function f(delta) = max_m (corr[m] + D[m]*delta) is convex in delta (max of linear functions). We can find the minimum by:

2. Or using a ternary/bisection search

For efficiency, since n is small (≤500), computing max(corr + D*delta) is O(n) each time, and we need to do this for n² pairs. Total: O(n³) per sweep. For n=240, that's about 13.8M operations per sweep, which should take a few seconds.

Actually, let me think about a more efficient approach. Instead of trying all (j,k) pairs, I can:
1. For each j, compute B_j
2. For each k, compute D = B_j - B_k
3. Solve the 1D minimax

The 1D minimax: minimize max_m(a_m + b_m * t) over t in [lo, hi] where a_m = corr[m], b_m = D[m].

This is a linear programming problem in 1D. The optimal t is either at a breakpoint (where two terms in the max are equal) or at a boundary. The breakpoints are at t = (a_p - a_q)/(b_q - b_p) for pairs (p,q) where b_p ≠ b_q.

For an efficient solve:
- The function is convex, so we can use bisection on the derivative (subgradient)
- The subgradient at t is {sum of b_m for m in argmax}
- We want to find t where 0 is in the subgradient

Actually, the simplest approach for 1D convex minimization is:
- The function g(t) = max_m(a_m + b_m*t) is convex
- g'(t) (subgradient) = {b_m : a_m + b_m*t = g(t)} (any convex combination)
- We want to find t* where 0 is in the subgradient
- This means there exist m1, m2 such that a_{m1}+b_{m1}*t* = a_{m2}+b_{m2}*t* = g(t*) and b_{m1} < 0 < b_{m2} (or b_{m1} = 0 = b_{m2})

For the bisection approach:
- At a given t, compute g(t) and the set S(t) = {m : a_m + b_m*t ≈ g(t)}
- If min(b[S]) > 0: g is increasing, move left
- If max(b[S]) < 0: g is decreasing, move right
- Otherwise: t* found

This converges in O(n) per iteration, and we need O(log(precision/range)) iterations.

But actually, for our purposes, a simpler approach works well:
- Sample t at fine intervals, find the best, then refine with bisection.

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_B(h, j, n):
    nm1 = n - 1
    B = np.zeros(2 * n - 1)
    k_start = max(0, nm1 - j)
    k_end = min(2 * n - 2, 2 * nm1 - j)
    if k_start <= k_end:
        B[k_start:k_end + 1] += (1.0 - h[j - nm1 + np.arange(k_start, k_end + 1)])
    k_start2 = max(0, j)
    k_end2 = min(2 * n - 2, j + nm1)
    if k_start2 <= k_end2:
        B[k_start2:k_end2 + 1] += -h[j + nm1 - np.arange(k_start2, k_end2 + 1)]
    return B

def solve_1d_minimax(corr, D, lo, hi):
    """Solve min_t max_m(corr[m] + D[m]*t) for t in [lo, hi]."""
    if hi - lo < 1e-14:
        return (lo + hi) * 0.5
    
    # Bisection on subgradient
    # g(t) = max(corr + D*t), g'(t) subgradient = {D[m] : corr[m]+D[m]*t = g(t)}
    # We want 0 in subgradient
    
    for _ in range(60):
        mid = (lo + hi) * 0.5
        vals = corr + D * mid
        mval = np.max(vals)
        # Subgradient: values of D where corr + D*mid is close to max
        tol = 1e-12 * max(1.0, abs(mval))
        mask = (mval - vals) <= tol
        if np.any(mask):
            min_sg = np.min(D[mask])
            max_sg = np.max(D[mask])
            if min_sg >= 0:
                # g is non-decreasing at mid, go left or we're at optimum
                if abs(min_sg) < 1e-10:
                    return mid
                hi = mid
            elif max_sg <= 0:
                # g is non-increasing at mid, go right or we're at optimum
                if abs(max_sg) < 1e-10:
                    return mid
                lo = mid
            else:
                # 0 is in subgradient, mid is optimal
                return mid
        else:
            return mid  # shouldn't happen
    return (lo + hi) * 0.5

def full_pair_descent(h, corr, rng, n_cand=None, temperature=0.0):
    """Full (or partial) pair coordinate descent. Returns h, corr, improved."""
    n = len(h)
    order = np.arange(n)
    rng.shuffle(order)
    improved = False
    
    for j in order:
        B_j = compute_B(h, j, n)
        best_val = np.max(corr)
        best_delta = 0.0
        best_D = None
        best_k = -1
        best_new_val = best_val
        
        # Candidate k values
        if n_cand is None or n_cand >= n:
            candidates = [k for k in range(n) if k != j]
        else:
            # Mix of structured and random
            structured = [(j+1)%n, (j-1+n)%n, (j+n//2)%n, (j+n//3)%n, (j+2*n//3)%n]
            randoms = rng.choice(n, size=n_cand - len(structured), replace=False).tolist()
            candidates = list(set(structured + randoms) - {j})
        
        for k in candidates:
            if k == j:
                continue
            B_k = compute_B(h, k, n)
            D = B_j - B_k
            delta_lo = max(-h[j], h[k] - 1.0)
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-13:
                continue
            delta_opt = solve_1d_minimax(corr, D, delta_lo, delta_hi)
            val_opt = np.max(corr + D * delta_opt)
            
            # Simulated annealing: accept with probability
            if temperature > 0 and val_opt > best_val:
                if rng.expovariate(1.0) < (best_val - val_opt) / temperature:
                    pass  # accept anyway
                else:
                    continue
            
            if val_opt < best_new_val:
                best_new_val = val_opt
                best_delta = delta_opt
                best_D = D
                best_k = k
        
        if best_k >= 0 and abs(best_delta) > 1e-13:
            corr += best_delta * best_D
            h[j] = np.clip(h[j] + best_delta, 0.0, 1.0)
            h[best_k] = np.clip(h[best_k] - best_delta, 0.0, 1.0)
            improved = True
    
    return h, corr, improved

import time

    valid_s = (second_idx >= 0) & (second_idx < n)

    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)

    def f(x):
        return smooth_max_obj_grad(x, alpha)[0]
    def g(x):
        return smooth_max_obj_grad(x, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float).copy()
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    def time_left():
        return budget_s - (time.time() - t_start)

    # Phase 1: Intensive pair descent (55% budget)
    t_end1 = t_start + 0.55 * budget_s
    h_work = best_h.copy()
    n_w = len(h_work)
    corr_w = np.correlate(h_work, 1.0 - h_work, mode="full")
    stall = 0

    while time.time() < t_end1:
        h_work, corr_w, imp = pair_descent(h_work, corr_w, rng, n_cand=35)
        c5_w = compute_c5(h_work)
        if c5_w < best_c5 - 1e-10:
            best_c5 = c5_w
            best_h = h_work.copy()
            best_n = n_w
            stall = 0
        else:
            stall += 1

        if stall > 0 and stall % 100 == 0:
            if stall >= 600:
                n_kick = max(5, n_w // 10)
                idx = rng.choice(n_w, size=n_kick, replace=False)
                h_work[idx] = np.clip(h_work[idx] + rng.uniform(-0.4, 0.4, n_kick), 0, 1)
                h_work = project_to_sum(h_work, n_w / 2.0)
                corr_w = np.correlate(h_work, 1.0 - h_work, mode="full")
                stall = 0
            elif stall >= 300:
                idx = rng.choice(n_w, size=n_w // 8, replace=False)
                h_work[idx] = np.clip(1.0 - h_work[idx], 0, 1)
                h_work = project_to_sum(h_work, n_w / 2.0)
                corr_w = np.correlate(h_work, 1.0 - h_work, mode="full")
                stall = 0
            elif stall % 50 == 0:
                noise = 0.03 + 0.02 * (stall / 300.0)
                h_work = project_to_sum(h_work + rng.uniform(-noise, noise, n_w), n_w / 2.0)
                corr_w = np.correlate(h_work, 1.0 - h_work, mode="full")

    # Phase 2: Multi-resolution exploration (20% budget)
    t_end2 = t_start + 0.75 * budget_s
    resolutions = [120, 160, 200, 240, 280, 320, 400, 480]
    for n_tr in resolutions:
        if time.time() >= t_end2:
            break
        ts_tr = n_tr / 2.0
        if abs(ts_tr - round(ts_tr)) > 1e-6:
            continue

        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_tr, endpoint=False)
        cand_starts = []

        cand_starts.append(project_to_sum(np.interp(x_new, x_old, best_h, period=2.0), ts_tr))

        h_rand = project_to_sum(rng.uniform(0, 1, n_tr), ts_tr)
        cand_starts.append(h_rand)

        blk = n_tr // 4
        hb = np.zeros(n_tr)
        for i in range(0, n_tr, 2 * blk):
            hb[i:min(i + blk, n_tr)] = 1.0
        cand_starts.append(project_to_sum(hb, ts_tr))

        h_sin = np.clip(0.5 + 0.5 * np.sin(2 * np.pi * x_new), 0, 1)
        cand_starts.append(project_to_sum(h_sin, ts_tr))

        for h_st in cand_starts:
            if time.time() >= t_end2:
                break
            h_st = project_to_sum(h_st, ts_tr)
            corr_st = np.correlate(h_st, 1.0 - h_st, mode="full")
            n_sweeps = min(80, max(10, int((t_end2 - time.time()) / 1.5)))
            for _ in range(n_sweeps):
                if time.time() >= t_end2:
                    break
                h_st, corr_st, _ = pair_descent(h_st, corr_st, rng, n_cand=25)
            c5_st = compute_c5(h_st)
            if c5_st < best_c5:
                best_c5 = c5_st
                best_h = h_st.copy()
                best_n = n_tr

    # Phase 3: SLSQP polish (15% budget)
    t_end3 = t_start + 0.90 * budget_s
    n_f = best_n
    ts_f = n_f / 2.0
    alphas = [20, 50, 100, 300, 600, 1200]
    h_cur = best_h.copy()
    for alpha in alphas:
        if time.time() >= t_end3:
            break
        h_pol = slsqp_polish(h_cur, n_f, ts_f, alpha, maxiter=800)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
            best_n = n_f
        h_cur = h_pol.copy()

    # Phase 4: Final pair descent polish (10% budget)
    t_end4 = t_start + budget_s - 3
    h_fin = best_h.copy()
    n_fin = best_n
    corr_fin = np.correlate(h_fin, 1.0 - h_fin, mode="full")
    stall_fin = 0
    while time.time() < t_end4:
        h_fin, corr_fin, _ = pair_descent(h_fin, corr_fin, rng, n_cand=40)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5 - 1e-12:
            best_c5 = c5_fin
            best_h = h_fin.copy()
            best_n = n_fin
            stall_fin = 0
        else:
            stall_fin += 1
            if stall_fin > 200:
                n_kick = max(3, n_fin // 15)
                idx = rng.choice(n_fin, size=n_kick, replace=False)
                h_fin[idx] = np.clip(h_fin[idx] + rng.uniform(-0.15, 0.15, n_kick), 0, 1)
                h_fin = project_to_sum(h_fin, n_fin / 2.0)
                corr_fin = np.correlate(h_fin, 1.0 - h_fin, mode="full")
                stall_fin = 0

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
