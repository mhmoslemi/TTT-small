import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def lag_gradient(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[lag:] += (1.0 - h[:n - lag])
            g[:n - lag] -= h[lag:]
    else:
        k = -lag
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[:n - k] += (1.0 - h[k:])
            g[k:] -= h[:n - k]
    return g

def topk_gradient(h, top_k, temp):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    sorted_idx = np.argsort(corr)[::-1]
    top_idx = sorted_idx[:top_k]
    vals = corr[top_idx]
    shifted = vals - m_val
    w = np.exp(-temp * (-shifted))
    w = w / (np.sum(w) + 1e-30)
    grad = np.zeros(n)
    for i in range(top_k):
        if w[i] < 1e-10:
            continue
        lag = int(top_idx[i]) - (n - 1)
        grad += w[i] * lag_gradient(h, lag)
    return m_val, grad

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    grad = np.zeros(n)
    active = np.where(w > 1e-10)[0]
    for m in active:
        lag = int(m) - (n - 1)
        grad += w[m] * lag_gradient(h, lag)
    return obj, grad

    step = step_init
    stalled = 0

        if time.time() - t_start > time_limit:

        _, grad = topk_gradient(h, top_k, temp)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:

        momentum = mu * momentum + (1 - mu) * (-grad)
        h_new = h + step * momentum

    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_b)
