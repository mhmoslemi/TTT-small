        g_dec = np.where(can_dec, g, -np.inf)

        i_dec = int(np.argmax(g_dec))
        if i_inc == i_dec:
            break

            break

        move = step_frac * max_move
        if move < 1e-10:

        h[i_dec] -= move
        h = np.clip(h, 0.0, 1.0)

            h[i_dec] += move

- If j is in the range [k, n-1]: the term h[j]*(1-h[j-k]) contributes (1-h[j-k])
- For j in [k, n-1-k]: both terms apply, so (1-h[j-k]) - h[j+k]
- For j in [0, k-1]: only the first term, so -h[j+k]
- For j in [n-k, n-1]: only the second term, so (1-h[j-k])

This is exactly what corr_gradient does. So the gradient of the smooth max is:
g = sum_k w[k] * corr_gradient(h, k)

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")


def project_box_sum(v, target_sum):
    n = len(v)
    v = np.asarray(v, dtype=float).copy()
    lo = -2.0 - float(np.max(v))
    hi = 2.0 - float(np.min(v))
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(v + mid, 0.0, 1.0)
        s = np.sum(h)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(v + lam, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h


def corr_gradient(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[lag:] += (1.0 - h[:n - lag])
            g[:n - lag] -= h[lag:]
    else:
        k = -lag
        g[:n - k] += (1.0 - h[k:])
        g[k:] -= h[:n - k]
    return g


def smooth_max_fg(x, alpha, target_sum, penalty_w):
    n = len(x)
    corr = compute_corr(x)
    m_val = float(np.max(corr))
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    s = float(np.sum(exp_vals))
    obj = m_val + np.log(s) / alpha
    sum_diff = float(np.sum(x)) - target_sum
    obj += 0.5 * penalty_w * sum_diff * sum_diff
    w = exp_vals / s
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - x[fc]) * w[None, :]
    cs = valid_s * x[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    grad += penalty_w * sum_diff * np.ones(n)
    return obj, grad


def smooth_max_f(x, alpha, target_sum, penalty_w):
    return smooth_max_fg(x, alpha, target_sum, penalty_w)[0]


def smooth_max_g(x, alpha, target_sum, penalty_w):
    return smooth_max_fg(x, alpha, target_sum, penalty_w)[1]


def lbfgsb_run(h0, target_sum, alpha, penalty_w, maxiter):
    n = len(h0)
    try:
        res = minimize(smooth_max_f, h0.copy(), method='L-BFGS-B', jac=smooth_max_g,
                       bounds=[(0.0, 1.0)] * n,

                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14, 'maxls': 30})
        return project_box_sum(res.x, target_sum)
    except Exception:

def subgradient_opt(h0, target_sum, n_iters, eta0, decay, top_k, mom):
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for t in range(n_iters):
        corr = compute_corr(h)
        if top_k <= 1:
            m_idx = int(np.argmax(corr))
            g = corr_gradient(h, m_idx - (n - 1))
        else:
            top_idx = np.argsort(corr)[-top_k:]
            shifted = corr[top_idx] - float(np.max(corr[top_idx]))
            w = np.exp(10.0 * shifted)
            w /= float(np.sum(w))
            g = np.zeros(n)
            for idx, wi in zip(top_idx, w):
                g += wi * corr_gradient(h, int(idx) - (n - 1))
        eta = eta0 / (1.0 + decay * t)
        v = mom * v - eta * g
        h_new = project_box_sum(h + v, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5


def coordinate_descent(h, target_sum, max_iters, step_frac):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    stale = 0
    for it in range(max_iters):
        corr = compute_corr(h)
        m_star = int(np.argmax(corr))
        lag = m_star - (n - 1)
        g = corr_gradient(h, lag)
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        g_inc = np.where(can_inc, g, np.inf)
        g_dec = np.where(can_dec, g, -np.inf)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        if i_inc == i_dec:
            break
        if g[i_inc] >= g[i_dec] - 1e-12:
            break
        max_move = min(1.0 - h[i_inc], h[i_dec])
        move = step_frac * max_move
        if move < 1e-10:
            break
        h[i_inc] += move
        h[i_dec] -= move
        h = np.clip(h, 0.0, 1.0)
        h = project_box_sum(h, target_sum)
        c5_new = compute_c5(h)
        if c5_new >= c5 - 1e-13:
            h[i_inc] -= move
            h[i_dec] += move
            h = np.clip(h, 0.0, 1.0)
            h = project_box_sum(h, target_sum)
            stale += 1
            if stale > 50:
                break
        else:
            c5 = c5_new
            stale = 0
    return h, c5

def fourier_init(n, rng, n_freq):

        phase = rng.uniform(0.0, 2.0 * np.pi)

                best_n = n_tr
            h_interp = h_sg2

        # Coordinate descent
        if time_remaining() > 50:
            h_cd2, c5_cd2 = coordinate_descent(h_interp, ts_tr,
                                               max_iters=3000, step_frac=0.1)
            if c5_cd2 < best_c5:
                best_c5 = c5_cd2
                best_h = h_cd2.copy()
                best_n = n_tr
            h_interp = h_cd2

    n_best = best_n
    ts_best = n_best / 2.0

    # Phase 5: Perturbation trials
    n_trials = 12
    for trial in range(n_trials):
        if time_remaining() < 60:
            break
        noise_level = 0.02 + 0.05 * (1.0 - float(trial) / float(n_trials))
        h_shake = best_h + rng.uniform(-noise_level, noise_level, n_best)
        h_shake = project_box_sum(h_shake, ts_best)
        h_cd3, c5_cd3 = coordinate_descent(h_shake, ts_best,
                                           max_iters=2500, step_frac=0.1)
        if c5_cd3 < best_c5:
            best_c5 = c5_cd3
            best_h = h_cd3.copy()
        if time_remaining() < 40:
            break
        h_opt2 = lbfgsb_run(h_cd3, ts_best, 1500, 1e4, maxiter=200)
        c5_opt2 = compute_c5(h_opt2)
        if c5_opt2 < best_c5:
            best_c5 = c5_opt2
            best_h = h_opt2.copy()

    # Phase 6: Fourier-based diverse starts
    if time_remaining() > 80:
        for attempt in range(4):
            if time_remaining() < 40:
                break
            h_four = fourier_init(n_best, rng, n_freq=5 + attempt)
            h_cd4, c5_cd4 = coordinate_descent(h_four, ts_best,
                                               max_iters=3000, step_frac=0.1)
            h_opt3 = lbfgsb_run(h_cd4, ts_best, 500, 1e4, maxiter=200)
            c5_opt3 = compute_c5(h_opt3)
            if c5_opt3 < best_c5:
                best_c5 = c5_opt3
                best_h = h_opt3.copy()

    # Phase 7: Binary local search
    if time_remaining() > 80:
        h_bin = best_h.copy()
        n_ones = int(round(ts_best))
        sorted_idx = np.argsort(h_bin)
        h_bin[sorted_idx[:n_best - n_ones]] = 0.0
        h_bin[sorted_idx[n_best - n_ones:]] = 1.0
        h_bl, c5_bl = binary_local_search(h_bin, ts_best,
                                          max_iters=500, n_sample=15)
        if c5_bl < best_c5:
            best_c5 = c5_bl
            best_h = h_bl.copy()
        if time_remaining() > 40:
            h_cd5, c5_cd5 = coordinate_descent(best_h, ts_best,
                                               max_iters=5000, step_frac=0.1)
            if c5_cd5 < best_c5:
                best_c5 = c5_cd5
                best_h = h_cd5.copy()

    # Phase 8: Final polishing
    if time_remaining() > 60:
        h_pol = continuation_optimize(best_h, ts_best,
                                      alphas=[5000, 20000, 80000],
                                      penalty_w=1e4, maxiter_per=200)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    if time_remaining() > 30:
        h_cd6, c5_cd6 = coordinate_descent(best_h, ts_best,
                                           max_iters=15000, step_frac=0.05)
        if c5_cd6 < best_c5:
            best_c5 = c5_cd6
            best_h = h_cd6.copy()

    if time_remaining() > 20:
        h_pol2 = lbfgsb_run(best_h, ts_best, 100000, 1e4, maxiter=500)
        c5_pol2 = compute_c5(h_pol2)
        if c5_pol2 < best_c5:
            best_c5 = c5_pol2
            best_h = h_pol2.copy()

    best_h = project_box_sum(best_h, n_best / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_best)
