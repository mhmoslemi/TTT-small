import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_all_corrs(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_simplex_robust(h, target_sum):
    """Project h onto {x : 0<=x_i<=1, sum(x)=target_sum}"""
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        eps = 1e-14
        can_inc = h < 1.0 - eps
        can_dec = h > eps
        if diff > 0:
            mask = can_inc
            if np.any(mask):
                h[mask] += diff / int(np.sum(mask))
            else:
                break
        else:
            mask = can_dec
            if np.any(mask):
                h[mask] += diff / int(np.sum(mask))
            else:
                break
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_for_lag(h, lag):
    """Compute gradient of correlation at given lag w.r.t. h"""
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        return 1.0 - 2.0 * h
    elif lag > 0:
        g[:n-lag] += (1.0 - h[lag:])
        g[lag:] -= h[:n-lag]
    else:
        k = -lag
        g[k:] += (1.0 - h[:n-k])
        g[:n-k] -= h[k:]
    return g

                        eta0=0.1, decay=0.999, time_limit=None, t_start=None):

    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    eta = eta0
    patience = 200
    no_improve = 0
    
    for it in range(max_iters):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break
        
        corrs = compute_all_corrs(h)
        current_max = np.max(corrs)

        sorted_idx = np.argsort(corrs)[::-1]
        active = sorted_idx[:n_lags_active]
        
        # Compute subgradient as average of active lags
        g = np.zeros(n)
        for idx in active:
            lag = idx - (n - 1)
            g += grad_for_lag(h, lag)
        g /= len(active)

        h_new = project_simplex_robust(h_new, target_sum)

                no_improve = 0

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)

def smooth_max_objective_and_grad(h, alpha):
    n = len(h)
    corr = compute_all_corrs(h)
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = (m_val + np.log(np.sum(exp_vals)) / alpha) * (2.0/n)
    
    # Gradient
    grad = np.zeros(n)
    for idx in range(2*n-1):
        if w[idx] < 1e-15:
            continue
        lag = idx - (n-1)
        grad += w[idx] * grad_for_lag(h, lag)
    grad *= (2.0/n)
    return obj, grad

def slsqp_optimize(h, target_sum, alpha, maxiter=500, time_limit=None, t_start=None):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    constraint = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                  'jac': lambda x: np.ones(n)}
    
    def obj_grad(x):
        return smooth_max_objective_and_grad(x, alpha)
    
    try:
        res = minimize(obj_grad, h.copy(), method='SLSQP', jac=lambda x: smooth_max_objective_and_grad(x, alpha)[1],
                       bounds=bounds, constraints=constraint,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_simplex_robust(res.x, target_sum)
    except Exception:
        return h.copy()

def multiplicative_update(h, target_sum, max_iters=3000, eta0=0.05, time_limit=None, t_start=None):
    """Multiplicative (exponentiated gradient) update for the simplex"""
    n = len(h)
    h = h.copy()
    # Ensure h > 0 for multiplicative updates
    h = np.maximum(h, 1e-10)
    h = project_simplex_robust(h, target_sum)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    eta = eta0
    
    for it in range(max_iters):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break
        
        corrs = compute_all_corrs(h)
        current_max = np.max(corrs)
        
        # Top-k active lags
        n_active = max(5, min(20, n // 15))
        sorted_idx = np.argsort(corrs)[::-1]
        active = sorted_idx[:n_active]
        
        g = np.zeros(n)
        for idx in active:
            lag = idx - (n - 1)
            g += grad_for_lag(h, lag)
        g /= len(active)
        
        # Multiplicative update
        h_new = h * np.exp(-eta * g)
        h_new = project_simplex_robust(h_new, target_sum)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c5

        for inner_it in range(5):

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_all_corrs(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_simplex(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        eps = 1e-13
        mask = (h > eps) & (h < 1.0 - eps)
        if diff > 0:
            mask = h < 1.0 - eps
        else:
            mask = h > eps
        if np.any(mask):
            h[mask] += diff / int(np.sum(mask))
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        return 1.0 - 2.0 * h
    elif lag > 0:
        g[:n-lag] += (1.0 - h[lag:])
        g[lag:] -= h[:n-lag]
    else:
        k = -lag
        g[k:] += (1.0 - h[:n-k])
        g[:n-k] -= h[k:]
    return g

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    corr = compute_all_corrs(h)
    m = np.max(corr)
    s = alpha * (corr - m)
    s -= s.max()
    e = np.exp(s)
    w = e / e.sum()
    obj = (m + np.log(e.sum()) / alpha) * (2.0 / n)
    grad = np.zeros(n)
    for idx in range(2*n-1):
        if w[idx] > 1e-12:
            grad += w[idx] * grad_lag(h, idx - (n-1))
    grad *= (2.0 / n)
    return obj, grad

def slsqp(h, target_sum, alpha, maxiter=400, t_end=None):
    n = len(h)
    if t_end is not None and time.time() > t_end:
        return h
    try:
        res = minimize(
            lambda x: smooth_max_obj_grad(x, alpha)[0],
            h.copy(), method='SLSQP',
            jac=lambda x: smooth_max_obj_grad(x, alpha)[1],
            bounds=[(0.0,1.0)]*n,
            constraints=[{'type':'eq','fun':lambda x: np.sum(x)-target_sum,
                         'jac':lambda x: np.ones(n)}],
            options={'maxiter':maxiter,'ftol':1e-14})

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_all_corrs(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_simplex(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        eps = 1e-13
        if diff > 0:
            mask = h < 1.0 - eps
        else:
            mask = h > eps
        if np.any(mask):
            h[mask] += diff / int(np.sum(mask))
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        return 1.0 - 2.0 * h
    elif lag > 0:
        g[:n - lag] += (1.0 - h[lag:])
        g[lag:] -= h[:n - lag]
    else:
        k = -lag
        g[k:] += (1.0 - h[:n - k])
        g[:n - k] -= h[k:]
    return g

def verify_grad(h, j, tol=1e-6):
    n = len(h)
    corr = compute_all_corrs(h)
    num_grad = (corr[j] - compute_c5(np.concatenate([h[:j-(n-1)] if j>=n-1 else [], h[j-(n-1):]])) if False else 0)
    lag = j - (n - 1)
    return grad_lag(h, lag)

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    corr = compute_all_corrs(h)
    m = np.max(corr)
    s = alpha * (corr - m)
    s -= s.max()
    e = np.exp(s)
    w = e / e.sum()
    obj = (m + np.log(e.sum()) / alpha) * (2.0 / n)
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        if w[idx] > 1e-10:
            grad += w[idx] * grad_lag(h, idx - (n - 1))
    grad *= (2.0 / n)
    return obj, grad

def slsqp(h, target_sum, alpha, maxiter=400, t_end=None):
    n = len(h)
    if t_end is not None and time.time() > t_end:
        return h
    try:
        res = minimize(
            lambda x: smooth_max_obj_grad(x, alpha)[0],
            h.copy(), method='SLSQP',
            jac=lambda x: smooth_max_obj_grad(x, alpha)[1],
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                         'jac': lambda x: np.ones(n)}],
            options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_simplex(res.x, target_sum)
    except Exception:
        return h

def subgrad_descent(h, target_sum, n_active=15, maxit=3000, eta0=0.05, t_end=None):
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    eta = eta0
    pat = 0
    for it in range(maxit):
        if t_end is not None and time.time() > t_end:
            break
        corrs = compute_all_corrs(h)
        top = np.argsort(corrs)[::-1][:n_active]
        g = np.zeros(n)
        for idx in top:
            g += grad_lag(h, idx - (n - 1))
        g /= len(top)
        h = project_simplex(h - eta * g, target_sum)

def binary_snap(h, target_sum):

    num_ones = int(np.sum(h_bin))
