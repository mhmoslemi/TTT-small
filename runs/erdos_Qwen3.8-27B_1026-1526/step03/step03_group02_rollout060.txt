```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def make_binary(h, n, target_ones, rng):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones == target_ones:
        return h_bin
    if n_ones > target_ones:
        excess = n_ones - target_ones
        ones_idx = np.where(h_bin == 1.0)[0]
        order = np.argsort(h[ones_idx])[:excess]
        h_bin[ones_idx[order]] = 0.0
    else:
        deficit = target_ones - n_ones
        zeros_idx = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h[zeros_idx])[:deficit]
        h_bin[zeros_idx[order]] = 1.0
    return h_bin

            h[a:a+L], h[b:b+L] = h[b:b+L].copy(), h[a:a+L].copy()
        else:
            i1 = rng.randint(n)

            i2 = rng.randint(n)
            j2 = rng.randint(n)
            if i1 == j1 or i2 == j2:

        delta = c5_new - c5_curr

                no_improve = 0

                no_improve += 1
            else:

                no_improve += 1

            c5_curr = best_c5

            zeros = np.where(h == 0.0)[0]

            kz = rng.choice(zeros, size=min(n_kick, len(zeros)), replace=False)

    return np.repeat(h, 2)

def perturbed(h, n, rng, n_flips):
    h_new = h.copy()
    ones = np.where(h_new == 1.0)[0]
    zeros = np.where(h_new == 0.0)[0]

    h_new[sz] = 1.0
    return h_new

    best_c5 = compute_c5(best_h)
```

```python
        self.n = len(h)
```

```python
def greedy_maximin(n):
    """Place n/2 ones using maximin criterion."""
    h = np.zeros(n)
    target = n // 2
    positions = [0]
    h[0] = 1
    for _ in range(target - 1):
        # Find position that maximizes minimum distance to existing ones
        best_pos = 0
        best_min_dist = -1
        for pos in range(n):
            if h[pos] > 0:
                continue
            min_dist = min(abs(pos - p) for p in positions)
            if min_dist > best_min_dist:
                best_min_dist = min_dist
                best_pos = pos
        h[best_pos] = 1
        positions.append(best_pos)
    return h
```
