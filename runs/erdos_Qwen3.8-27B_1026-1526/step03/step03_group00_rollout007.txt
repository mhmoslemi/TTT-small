def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def multi_lag_gradient(h, corr, top_k=5):
    """Compute combined gradient from top K lags."""
    n = len(h)
    # Find top K lags
    sorted_lags = np.argsort(corr)[::-1][:top_k]
    max_val = corr[sorted_lags[0]]
    # Weight by how close to max each lag is
    weights = np.zeros(2*n-1)
    for i, lag_idx in enumerate(sorted_lags):
        weights[lag_idx] = max(0, (corr[lag_idx] - 0.9 * max_val) / max(1e-12, max_val))
    weights /= max(np.sum(weights), 1e-12)
    
    g = np.zeros(n)
    for lag_idx in sorted_lags:
        if weights[lag_idx] < 1e-10:
            continue
        lag = lag_idx - (n - 1)
        if lag >= 0:
            if lag == 0:
                g[:n] += weights[lag_idx] * (1.0 - 2.0 * h)
            else:
                g[lag:n] += weights[lag_idx] * (1.0 - h[:n-lag])
                g[:n-lag] -= weights[lag_idx] * h[lag:]
        else:
            k = -lag
            g[:n-k] += weights[lag_idx] * (1.0 - h[k:])
            g[k:n] -= weights[lag_idx] * h[:n-k]
    return g

    n = len(h0)
    h = h0.copy()
    c5 = compute_c5(h)
    
    for it in range(max_iters):
        corr = compute_corr(h)
        max_val = np.max(corr)
        # Get top 3 lags
        top_lags = np.argsort(corr)[::-1][:3]
        
        # Compute combined gradient
        g = np.zeros(n)
        for lag_idx in top_lags:
            lag = lag_idx - (n-1)
            if lag >= 0:

                    g[:n] += (1.0 - 2.0*h)
                else:
                    g[lag:n] += (1.0 - h[:n-lag])
                    g[:n-lag] -= h[lag:]
            else:
                k = -lag
                g[:n-k] += (1.0 - h[k:])
                g[k:n] -= h[:n-k]

        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        
        g_inc = np.where(can_inc, g, np.inf)
        g_dec = np.where(can_dec, g, -np.inf)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))

            break
        
        # Binary search for best step
        max_move = min(1.0 - h[i_inc], h[i_dec])

        for _ in range(8):
            mid = (lo + hi) / 2

            h[i_dec] -= mid
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, target_sum)

                lo = mid
                best_move = mid
            else:
                hi = mid

                grad += (1.0 - 2.0 * h)
            else:
                grad[lag:] += (1.0 - h[:n - lag])
                grad[:n - lag] -= h[lag:]
        else:
            k = -lag
            grad[:n - k] += (1.0 - h[k:])
            grad[k:] -= h[:n - k]
    return grad / float(top_k)

def momentum_subgradient(h0, target_sum, n_iters, alpha, lr, momentum, rng=None):
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    stuck = 0
    for it in range(n_iters):
        grad = smooth_max_grad(h, alpha)
        v = momentum * v + lr * grad
        h_new = h - v
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            stuck = 0
        else:
            stuck += 1
        h = h_new
        if stuck > 400 and it < n_iters - 100:
            lr *= 0.85
            v *= 0.5
            if rng is not None:
                h = h + rng.uniform(-0.005, 0.005, n)
                h = np.clip(h, 0.0, 1.0)
                h = project_to_sum(h, target_sum)
            stuck = 0
        if it % 1000 == 999:
            lr *= 0.85
    return best_h, best_c5

def targeted_descent(h0, target_sum, max_iters, top_k):
    n = len(h0)
    h = h0.copy()
    c5 = compute_c5(h)
    for it in range(max_iters):
        grad = multi_lag_grad(h, top_k)
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        g_inc = np.where(can_inc, grad, np.inf)
        g_dec = np.where(can_dec, grad, -np.inf)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        if i_inc == i_dec or grad[i_inc] >= grad[i_dec] - 1e-12:
            break
        max_move = min(1.0 - h[i_inc], h[i_dec])
        lo, hi = 0.0, max_move
        best_step = 0.0
        for _ in range(7):
            mid = (lo + hi) / 2.0
            h[i_inc] += mid
            h[i_dec] -= mid
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, target_sum)
            if compute_c5(h) < c5:
                lo = mid
                best_step = mid
            else:
                hi = mid
            h = np.clip(h, 0.0, 1.0)
        if best_step < 1e-9:
            break
        h[i_inc] += best_step
        h[i_dec] -= best_step
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
        c5 = compute_c5(h)
    return h, c5

def single_lag_cd(h0, target_sum, max_iters):
    n = len(h0)
    h = h0.copy()
    c5 = compute_c5(h)
    for it in range(max_iters):
        corr = compute_corr(h)
        m_star = int(np.argmax(corr))
        lag = m_star - (n - 1)
        g = np.zeros(n)
        if lag >= 0:
            if lag == 0:
                g[:] = 1.0 - 2.0 * h
            else:
                g[lag:] += (1.0 - h[:n - lag])
                g[:n - lag] -= h[lag:]
        else:
            k = -lag
            g[:n - k] += (1.0 - h[k:])
            g[k:] -= h[:n - k]
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        g_inc = np.where(can_inc, g, np.inf)
        g_dec = np.where(can_dec, g, -np.inf)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        if i_inc == i_dec or g[i_inc] >= g[i_dec] - 1e-12:
            break
        max_move = min(1.0 - h[i_inc], h[i_dec])
        lo, hi = 0.0, max_move
        best_step = 0.0
        for _ in range(6):
            mid = (lo + hi) / 2.0
            h[i_inc] += mid
            h[i_dec] -= mid
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, target_sum)
            if compute_c5(h) < c5:
                lo = mid
                best_step = mid
            else:
                hi = mid
            h = np.clip(h, 0.0, 1.0)
        if best_step < 1e-9:
            break
        h[i_inc] += best_step
        h[i_dec] -= best_step
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
        c5 = compute_c5(h)
    return h, c5

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h, period=2.0)

def _slsqp_obj(x, alpha, target_sum):
    corr = compute_corr(x)
    m = np.max(corr)
    s = alpha * (corr - m)
    s -= np.max(s)
    e = np.exp(s)
    return m + np.log(np.sum(e)) / alpha

def _slsqp_jac(x, alpha, target_sum):
    return smooth_max_grad(x, alpha)

def _slsqp_eq(x, alpha, target_sum):
    return np.sum(x) - target_sum

def _slsqp_eq_jac(x, alpha, target_sum):
    return np.ones(len(x))

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    def tl():
        return budget_s - (time.time() - t_start)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 540
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n_cur = best_n
    ts_cur = n_cur / 2.0

    for alpha, n_it, lr, mom in [(20, 3000, 0.04, 0.90), (60, 2500, 0.025, 0.92),
                                  (200, 2000, 0.015, 0.93), (600, 1500, 0.008, 0.94),
                                  (2000, 1000, 0.004, 0.95)]:
        if tl() < 100:
            break
        h_new, c5_new = momentum_subgradient(best_h, ts_cur, n_iters=n_it, alpha=alpha,
                                             lr=lr, momentum=mom, rng=rng)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    if tl() > 120:
        for top_k in [3, 5, 10]:
            if tl() < 50:
                break
            h_td, c5_td = targeted_descent(best_h, ts_cur, max_iters=4000, top_k=top_k)
            if c5_td < best_c5:
                best_c5 = c5_td
                best_h = h_td.copy()

    if tl() > 100:
        bounds_list = [(0.0, 1.0)] * n_cur
        for alpha in [300, 1000, 4000]:
            if tl() < 35:
                break
            try:
                args_t = (alpha, ts_cur)
                cons = [{'type': 'eq', 'fun': _slsqp_eq, 'jac': _slsqp_eq_jac, 'args': args_t}]
                res = minimize(_slsqp_obj, best_h.copy(), method='SLSQP',
                               jac=_slsqp_jac, args=args_t,
                               bounds=bounds_list, constraints=cons,
                               options={'maxiter': 700, 'ftol': 1e-16})
                h_pol = project_to_sum(res.x, ts_cur)
                c5_pol = compute_c5(h_pol)
                if c5_pol < best_c5:
                    best_c5 = c5_pol
                    best_h = h_pol.copy()
            except Exception:
                pass

    if tl() > 60:
        h_cd, c5_cd = single_lag_cd(best_h, ts_cur, max_iters=8000)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    for n_tr in [720, 900, 1080]:
        if tl() < 160:
            break
        ts_tr = n_tr / 2.0
        h_interp = interp_to_n(best_h, n_tr)
        h_interp = project_to_sum(h_interp, ts_tr)
        c5_interp = compute_c5(h_interp)

        for alpha, n_it, lr, mom in [(80, 1500, 0.012, 0.92), (400, 1000, 0.006, 0.94)]:
            if tl() < 70:
                break
            h_opt, c5_opt = momentum_subgradient(h_interp, ts_tr, n_iters=n_it,
                                                 alpha=alpha, lr=lr, momentum=mom, rng=rng)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_tr
            else:
                best_n = n_tr
                best_h = h_opt.copy()

        if tl() > 70:
            h_td2, c5_td2 = targeted_descent(best_h, n_tr / 2.0, max_iters=2500, top_k=5)
            if c5_td2 < best_c5:
                best_c5 = c5_td2
                best_h = h_td2.copy()
                best_n = n_tr

        if tl() > 50:
            h_cd2, c5_cd2 = single_lag_cd(best_h, n_tr / 2.0, max_iters=4000)
            if c5_cd2 < best_c5:
                best_c5 = c5_cd2
                best_h = h_cd2.copy()
                best_n = n_tr

    n_fin = best_n
    ts_fin = n_fin / 2.0

    if tl() > 40:
        for top_k in [2, 4, 7]:
            if tl() < 25:
                break
            h_td3, c5_td3 = targeted_descent(best_h, ts_fin, max_iters=3000, top_k=top_k)
            if c5_td3 < best_c5:
                best_c5 = c5_td3
                best_h = h_td3.copy()

    if tl() > 30:
        h_cd3, c5_cd3 = single_lag_cd(best_h, ts_fin, max_iters=10000)
        if c5_cd3 < best_c5:
            best_c5 = c5_cd3
            best_h = h_cd3.copy()

    if tl() > 25:
        h_mp, c5_mp = momentum_subgradient(best_h, ts_fin, n_iters=800, alpha=5000,
                                           lr=0.002, momentum=0.96, rng=rng)
        if c5_mp < best_c5:
            best_c5 = c5_mp
            best_h = h_mp.copy()

    if tl() > 15:
        h_cd4, c5_cd4 = single_lag_cd(best_h, ts_fin, max_iters=5000)
        if c5_cd4 < best_c5:
            best_c5 = c5_cd4
            best_h = h_cd4.copy()

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
