        max_move = min(1.0 - h[i_inc], h[i_dec])
        move = step_frac * max_move
        if move < 1e-12:
            break
        h[i_inc] += move
        h[i_dec] -= move
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
        c5_new = compute_c5(h)
        if c5_new >= c5 - 1e-14:
            h[i_inc] -= move
            h[i_dec] += move
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, target_sum)
            stale += 1
            if stale > 150:
                break
        else:
            c5 = c5_new
            stale = 0
    return h, c5

def coord_desc_adaptive(h, target_sum, max_iters, base_step, n_lags):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    step = base_step
    stale = 0
    for it in range(max_iters):
        corr = compute_corr(h)
        top_indices = np.argsort(corr)[-n_lags:]
        g = np.zeros(n)
        for idx in top_indices:
            lag = int(idx) - (n - 1)
            g += corr_gradient(h, lag)
        g /= float(n_lags)
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        g_inc = np.where(can_inc, g, 1e18)
        g_dec = np.where(can_dec, g, -1e18)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        if i_inc == i_dec:
            break
        if g[i_inc] >= g[i_dec] - 1e-13:
            break
        max_move = min(1.0 - h[i_inc], h[i_dec])
        move = step * max_move
        if move < 1e-12:
            step *= 0.5
            if step < 1e-6:
                break
            continue
        h[i_inc] += move
        h[i_dec] -= move
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
        c5_new = compute_c5(h)
        if c5_new >= c5 - 1e-14:
            h[i_inc] -= move
            h[i_dec] += move
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, target_sum)
            stale += 1
            step *= 0.8
            if stale > 200:
                break
        else:
            c5 = c5_new
            stale = 0
            step = min(step * 1.1, base_step)
    return h, c5

def fourier_candidate(n, n_modes, freq_scale, amp_scale, rng, blend, base_h):
    x = np.linspace(0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for m in range(n_modes):
        freq = rng.uniform(-freq_scale, freq_scale)
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = amp_scale / np.sqrt(float(m + 1))
        h += amp * np.sin(2.0 * np.pi * freq * x / 2.0 + phase)
    h = np.clip(h, 0.0, 1.0)
    h = project_to_sum(h, n / 2.0)
    if base_h is not None and blend > 0.0:
        h = (1.0 - blend) * h + blend * base_h
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, n / 2.0)
    return h

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h, period=2.0)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 540
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n_parent = best_n
    ts_parent = float(n_parent) / 2.0

    def elapsed():
        return time.time() - t_start

    def tl():
        return budget_s - elapsed()

    # Phase 1: SLSQP refinement at parent resolution
    for alpha in [20, 80, 300, 1200, 5000, 20000]:
        if tl() < 25:
            break
        h_new = slsqp_local(best_h, n_parent, ts_parent, alpha, maxiter=400)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 2: Adaptive coordinate descent at parent resolution
    if tl() > 50:
        h_cd, c5_cd = coord_desc_adaptive(best_h, ts_parent, max_iters=10000, base_step=0.15, n_lags=3)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 3: Multi-resolution refinement
    target_resolutions = []
    for mult in [1, 2, 3, 4]:
        nr = n_parent * mult
        if nr % 2 == 0 and nr <= 4000:
            target_resolutions.append(nr)
    target_resolutions = sorted(set(target_resolutions))

    current_h = best_h.copy()
    current_n = best_n

    for n_tr in target_resolutions:
        if tl() < 60:
            break
        ts_tr = float(n_tr) / 2.0

        if n_tr == current_n:
            h_interp = current_h.copy()
        else:
            h_interp = interp_to_n(current_h, n_tr)
            h_interp = project_to_sum(h_interp, ts_tr)

        c5_interp = compute_c5(h_interp)
        if c5_interp < best_c5:
            best_c5 = c5_interp
            best_h = h_interp.copy()
            best_n = n_tr

        # Coordinate descent at this resolution
        h_cd, c5_cd = coord_desc_adaptive(h_interp, ts_tr, max_iters=8000, base_step=0.12, n_lags=3)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            best_n = n_tr
            current_h = h_cd.copy()
            current_n = n_tr

        # SLSQP at lower resolutions
        if n_tr <= 1080 and tl() > 50:
            for alpha in [400, 2000, 8000]:
                if tl() < 30:
                    break
                h_opt = slsqp_local(current_h, n_tr, ts_tr, alpha, maxiter=300)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    best_n = n_tr
                    current_h = h_opt.copy()
                    current_n = n_tr

        # Additional coordinate descent after SLSQP
        if tl() > 30:
            h_cd2, c5_cd2 = coord_desc_adaptive(current_h, ts_tr, max_iters=5000, base_step=0.08, n_lags=5)
            if c5_cd2 < best_c5:
                best_c5 = c5_cd2
                best_h = h_cd2.copy()
                best_n = n_tr
                current_h = h_cd2.copy()
                current_n = n_tr

    # Phase 4: Diverse restarts
    n_best = best_n
    ts_best = float(n_best) / 2.0
    n_trials = 18

    for trial in range(n_trials):
        if tl() < 45:
            break

        r = float(trial) / float(n_trials)
        if trial % 4 == 0:
            noise = 0.02 + 0.06 * (1.0 - r)
            h_shake = best_h + rng.uniform(-noise, noise, n_best)
        elif trial % 4 == 1:
            h_shake = fourier_candidate(n_best, 10, 4.0, 0.3, rng, 0.5, best_h)
        elif trial % 4 == 2:
            h_shake = fourier_candidate(n_best, 18, 6.0, 0.25, rng, 0.3, best_h)
        else:
            h_shake = fourier_candidate(n_best, 6, 3.0, 0.4, rng, 0.7, best_h)

        h_shake = np.clip(h_shake, 0.0, 1.0)
        h_shake = project_to_sum(h_shake, ts_best)

        h_cd, c5_cd = coord_desc_adaptive(h_shake, ts_best, max_iters=5000, base_step=0.12, n_lags=3)

        if c5_cd < best_c5 + 0.0005 and tl() > 35:
            h_opt = slsqp_local(h_cd, n_best, ts_best, 800, maxiter=300)
            c5_opt = compute_c5(h_opt)
            if c5_opt < c5_cd:
                c5_cd = c5_opt
                h_cd = h_opt

        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 5: Final aggressive refinement
    if tl() > 40:
        h_fc, c5_fc = coord_desc_adaptive(best_h, ts_best, max_iters=20000, base_step=0.06, n_lags=5)
        if c5_fc < best_c5:
            best_c5 = c5_fc
            best_h = h_fc.copy()

    # Phase 6: Final SLSQP polishes with increasing alpha
    for alpha in [3000, 10000, 50000, 200000]:
        if tl() < 20:
            break
        h_pol = slsqp_local(best_h, n_best, ts_best, alpha, maxiter=1500)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Final cleanup
    best_h = project_to_sum(best_h, float(n_best) / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_best)
