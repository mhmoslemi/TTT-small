```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / float(np.sum(mask))
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(h, alpha):
    corr = compute_corr(h)
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    mk = np.max(shifted)
    exp_vals = np.exp(shifted - mk)
    return m_val + (mk + np.log(np.sum(exp_vals))) / alpha

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    mk = np.max(shifted)
    exp_vals = np.exp(shifted - mk)
    w = exp_vals / np.sum(exp_vals)
    
    # Only process lags with significant weight
    grad = np.zeros(n)
    significant = np.where(w > 1e-12)[0]
    for k_off in significant:
        wk = w[k_off]
        lag = k_off - (n - 1)
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += wk * (1.0 - h[:n-lag])
            grad[:n-lag] -= wk * h[lag:]
        else:
            k = -lag
            grad[:n-k] += wk * (1.0 - h[k:])
            grad[k:] -= wk * h[:n-k]
    return grad

def slsqp_optimize(h0, target_sum, alpha, maxiter):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{
        'type': 'eq',
        'fun': lambda x: np.sum(x) - target_sum,
        'jac': lambda x: np.ones(n)
    }]
    
    def obj_func(x):
        return smooth_max_obj(x, alpha)
    
    def grad_func(x):
        return smooth_max_grad(x, alpha)
    
    try:
        res = minimize(obj_func, h0.copy(), method='SLSQP', jac=grad_func,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'disp': 0})
        return project_to_sum(np.clip(res.x, 0, 1), target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def coord_desc_ls(h, target_sum, max_iters, time_limit):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    t0 = time.time()
    
    for it in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        
        corr = compute_corr(h)
        m_star = int(np.argmax(corr))
        lag = m_star - (n - 1)
        
        # Gradient of the max correlation (subgradient)
        g = np.zeros(n)
        if lag == 0:
            g[:] = 1.0 - 2.0 * h
        elif lag > 0:
            g[lag:] += (1.0 - h[:n-lag])
            g[:n-lag] -= h[lag:]
        else:
            k = -lag
            g[:n-k] += (1.0 - h[k:])
            g[k:] -= h[:n-k]
        
        # Find best pair: increase where g is most negative, decrease where g is most positive
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        
        g_inc = np.where(can_inc, g, np.inf)
        g_dec = np.where(can_dec, g, -np.inf)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        
        if i_inc == i_dec or g[i_inc] >= g[i_dec] - 1e-12:
            break
        
        # Line search: find optimal step
        # h_new[i_inc] = h[i_inc] + t, h_new[i_dec] = h[i_dec] - t
        t_max = min(1.0 - h[i_inc], h[i_dec])
        
        # Try a few candidate steps
        best_t = 0.0
        best_c5_ls = c5
        for frac in [0.1, 0.3, 0.5, 0.7, 0.9]:
            t = frac * t_max
            if t < 1e-12:
                continue
            h_test = h.copy()
            h_test[i_inc] += t
            h_test[i_dec] -= t
            h_test = project_to_sum(h_test, target_sum)
            c5_test = compute_c5(h_test)
            if c5_test < best_c5_ls:
                best_c5_ls = c5_test
                best_t = t
        
        if best_t < 1e-10:
            break
        
        h[i_inc] += best_t
        h[i_dec] -= best_t
        h = project_to_sum(h, target_sum)
        c5 = best_c5_ls
    
    return h, c5

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)

def fourier_perturb(h, rng, n_harmonics=5, amplitude=0.05):
    n = len(h)
    perturbation = np.zeros(n)
    for k in range(1, n_harmonics + 1):
        freq = rng.randint(1, n // 2) * k
        phase = rng.uniform(0, 2 * np.pi)
        amp = amplitude / k
        perturbation += amp * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
    return project_to_sum(np.clip(h + perturbation, 0, 1), n / 2.0)
```

```python
import numpy as np
from scipy.optimize import minima
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / float(np.sum(mask))
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(h, alpha):
    corr = compute_corr(h)
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    mk = np.max(shifted)
    ev = np.exp(shifted - mk)
    return m_val + (mk + np.log(np.sum(ev))) / alpha

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    mk = np.max(shifted)
    ev = np.exp(shifted - mk)
    w = ev / np.sum(ev)
    grad = np.zeros(n)
    sig = np.where(w > 1e-14)[0]
    for k_off in sig:
        wk = w[k_off]
        lag = k_off - (n - 1)
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += wk * (1.0 - h[:n - lag])
            grad[:n - lag] -= wk * h[lag:]
        else:
            k = -lag
            grad[:n - k] += wk * (1.0 - h[k:])
            grad[k:] -= wk * h[:n - k]
    return grad

def slsqp_optimize(h0, target_sum, alpha, maxiter):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]

    def obj_f(x):
        return smooth_max_obj(x, alpha)

    def grad_f(x):
        return smooth_max_grad(x, alpha)

    try:
        res = minimize(obj_f, h0.copy(), method='SLSQP', jac=grad_f,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_to_sum(np.clip(res.x, 0, 1), target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def multi_subgrad_coord(h, target_sum, max_iters, time_budget, top_k=3):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    t0 = time.time()
    stale = 0

    for it in range(max_iters):
        if time.time() - t0 > time_budget:
            break
        corr = compute_corr(h)
        top_idx = np.argsort(corr)[-top_k:]
        top_idx = np.sort(top_idx)

        g = np.zeros(n)
        for k_off in top_idx:
            lag = k_off - (n - 1)
            if lag == 0:
                g[:] += (1.0 - 2.0 * h)
            elif lag > 0:
                g[lag:] += (1.0 - h[:n - lag])
                g[:n - lag] -= h[lag:]
            else:
                kk = -lag
                g[:n - kk] += (1.0 - h[kk:])
                g[kk:] -= h[:n - kk]
        g /= float(top_k)

        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break

        g_inc = np.where(can_inc, g, np.inf)
        g_dec = np.where(can_dec, g, -np.inf)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))

        if i_inc == i_dec or g[i_inc] >= g[i_dec] - 1e-12:
            break

        t_max = min(1.0 - h[i_inc], h[i_dec])
        best_t = 0.0
        best_c5_ls = c5
        for frac in [0.05, 0.15, 0.3, 0.5, 0.7, 0.9]:
            t = frac * t_max
            if t < 1e-12:
                continue
            h_test = h.copy()
            h_test[i_inc] += t
            h_test[i_dec] -= t
            h_test = project_to_sum(h_test, target_sum)
            c5_test = compute_c5(h_test)
            if c5_test < best_c5_ls:
                best_c5_ls = c5_test
                best_t = t

        if best_t < 1e-10:
            stale += 1
            if stale > 30:
                break
            continue

        h[i_inc] += best_t
        h[i_dec] -= best_t
        h = project_to_sum(h, target_sum)
        c5 = best_c5_ls
        stale = 0

    return h, c5

def simulated_annealing(h, target_sum, rng, max_iters, time_budget):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    t0 = time.time()
    T_init = 0.002
    T_end = 0.00001

    for it in range(max_iters):
        if time.time() - t0 > time_budget:
            break
        T = T_init * (T_end / T_init) ** (float(it) / max(float(max_iters), 1.0))
        corr = compute_corr(h)
        m_star = int(np.argmax(corr))
        lag = m_star - (n - 1)

        g = np.zeros(n)
        if lag == 0:
            g[:] = 1.0 - 2.0 * h
        elif lag > 0:
            g[lag:] += (1.0 - h[:n - lag])
            g[:n - lag] -= h[lag:]
        else:
            kk = -lag
            g[:n - kk] += (1.0 - h[kk:])
            g[kk:] -= h[:n - kk]

        can_inc = np.where(h < 1.0 - 1e-10)[0]
        can_dec = np.where(h > 1e-10)[0]
        if len(can_inc) == 0 or len(can_dec) == 0:
            break

        n_try = min(20, len(can_inc))
        best_move_c5 = c5
        best_i_inc = -1
        best_i_dec = -1

        for _ in range(n_try):
            i_inc = int(can_inc[rng.randint(len(can_inc))])
            i_dec = int(can_dec[rng.randint(len(can_dec))])
            if i_inc == i_dec:
                continue
            t_move = min(1.0 - h[i_inc], h[i_dec]) * rng.uniform(0.1, 0.5)
            if t_move < 1e-12:
                continue
            h_test = h.copy()
            h_test[i_inc] += t_move
            h_test[i_dec] -= t_move
            h_test = project_to_sum(h_test, target_sum)
            c5_test = compute_c5(h_test)
            delta = c5_test - c5
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-8)):
                if c5_test < best_move_c5:
                    best_move_c5 = c5_test
                    best_i_inc = i_inc
                    best_i_dec = i_dec

        if best_i_inc >= 0:
            t_move = min(1.0 - h[best_i_inc], h[best_i_dec])
            for frac in [0.1, 0.3, 0.5, 0.8]:
                t = frac * t_move
                h_test = h.copy()
                h_test[best_i_inc] += t
                h_test[best_i_dec] -= t
                h_test = project_to_sum(h_test, target_sum)
                c5_test = compute_c5(h_test)
                if c5_test < c5:
                    h = h_test
                    c5 = c5_test
                    break

    return h, c5

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)

def fourier_perturb(h, rng, n_harm=5, amp=0.04):
    n = len(h)
    x = np.arange(n, dtype=float)
    pert = np.zeros(n)
    for k in range(1, n_harm + 1):
        freq = rng.randint(1, max(2, n // (2 * k)))
        phase = rng.uniform(0, 2.0 * np.pi)
        pert += (amp / k) * np.sin(2.0 * np.pi * freq * x / n + phase)
    return project_to_sum(np.clip(h + pert, 0, 1), n / 2.0)
```
