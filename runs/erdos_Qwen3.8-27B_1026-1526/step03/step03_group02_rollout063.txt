    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def make_binary(h, n, target_sum, rng):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.round(np.sum(h_bin)))
    if n_ones == target_sum:
        return h_bin
    if n_ones > target_sum:
        excess = n_ones - target_sum
        ones_idx = np.where(h_bin == 1.0)[0]
        order = np.argsort(h[ones_idx])[:excess]
        h_bin = h_bin.copy()
        h_bin[ones_idx[order]] = 0.0
    else:
        deficit = target_sum - n_ones
        zeros_idx = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h[zeros_idx])[:deficit]
        h_bin = h_bin.copy()
        h_bin[zeros_idx[order]] = 1.0
    return h_bin

def run_binary_sa(h_start, n, target_sum, t_end, rng, use_adj=True):
    h = h_start.copy()
    corr = np.correlate(h, 1.0 - h, mode="full")
    c5_curr = float(np.max(corr)) * (2.0 / n)
    best_c5 = c5_curr
    best_h = h.copy()
    
    T = 0.015
    min_T = 0.0015
    no_improve = 0
    stagn_limit = 25000
    
    while time.time() < t_end:
        r = rng.random()
        if use_adj and r < 0.3:
            # Near-neighbor swap
            i = rng.randint(n)
            while h[i] == 0.0:
                i = rng.randint(n)
            d = rng.randint(1, 8)
            if rng.random() < 0.5:
                j = (i + d) % n
            else:
                j = (i - d) % n
            if h[j] != 0.0:
                j = rng.randint(n)
                while h[j] != 0.0:
                    j = rng.randint(n)
        else:
            # Random swap
            i = rng.randint(n)
            while h[i] == 0.0:
                i = rng.randint(n)
            j = rng.randint(n)
            while h[j] != 0.0:
                j = rng.randint(n)
        
        if i == j:
            continue
        
        h[i], h[j] = h[j], h[i]
        new_corr = np.correlate(h, 1.0 - h, mode="full")
        new_max = float(np.max(new_corr))
        c5_new = new_max * (2.0 / n)
        
        delta = c5_new - c5_curr
        if delta < 0.0 or (delta / max(T, 1e-10)) < np.log(rng.uniform() + 1e-15):
            c5_curr = c5_new
            corr = new_corr
            if c5_new < best_c5 - 1e-13:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i], h[j] = h[j], h[i]
        
        if no_improve > stagn_limit:
            h = best_h.copy()
            c5_curr = best_c5
            T = min(max(T * 2.0, min_T), 0.03)
            no_improve = 0
        
        T = max(T * 0.99998, min_T)
    
    return best_h, best_c5

def smooth_max_f_g(h, n, alpha, topk=None):
    nw = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode="full")
    
    if topk is not None and topk < nw:
        idx = np.argpartition(corr, -topk)[-topk:]
        tk = corr[idx]
    else:
        idx = np.arange(nw)
        tk = corr
    
    mval = float(np.max(tk))
    sh = alpha * (tk - mval)
    sh = np.clip(sh, -500.0, 0.0)
    ev = np.exp(sh)
    w = ev / np.sum(ev)
    obj = mval + np.log(np.sum(ev)) / alpha
    
    fw = np.zeros(nw)
    fw[idx] = w
    
    j_idx = np.arange(n)[:, None]
    p_idx = np.arange(nw)[None, :]
    r1 = j_idx + p_idx - (n - 1)
    r2 = j_idx - p_idx + (n - 1)
    v1 = (r1 >= 0) & (r1 < n)
    v2 = (r2 >= 0) & (r2 < n)
    c1 = np.clip(r1, 0, n - 1)
    c2 = np.clip(r2, 0, n - 1)
    
    t1 = v1.astype(np.float64) * (1.0 - h[c1]) * fw[None, :]
    t2 = v2.astype(np.float64) * h[c2] * fw[None, :]
    grad = np.sum(t1, axis=1) - np.sum(t2, axis=1)
    
    return obj, grad

def slsqp_opt(h0, n, ts, alpha, topk, mi):
    global _g_n, _g_alpha, _g_topk, _g_target_sum
    _g_n = n
    _g_alpha = alpha
    _g_topk = topk
    _g_target_sum = float(ts)
    bd = [(0.0, 1.0)] * n
    cs = [{'type': 'eq', 'fun': _sum_f, 'jac': _sum_j}]
    try:
        r = minimize(_smax_f, h0.copy(), method='SLSQP', jac=_smax_g,
                     bounds=bd, constraints=cs,
                     options={'maxiter': mi, 'ftol': 1e-15})
        return project_sum(r.x, ts)
    except Exception:
        return project_sum(h0.copy(), ts)

def tc_opt(h0, n, ts, alpha, topk, mi):

    _g_n = n
    _g_alpha = alpha
    _g_topk = topk

        zi = np.where(hb == 0.0)[0]
        ordr = np.argsort(-h[zi])[:df]
        hb = hb.copy()
        hb[zi[ordr]] = 1.0
    return hb

def best_shift_bin(h, n):
    bc5 = compute_c5(h)
    bh = h.copy()
    for s in range(1, n):
        hs = np.roll(h, s)
        c5s = compute_c5(hs)
        if c5s < bc5:
            bc5 = c5s
            bh = hs.copy()
    return bh, bc5

def local_refine_bin(h, n, max_rounds=200):
    h = h.copy()
    c5 = compute_c5(h)
    for _ in range(max_rounds):
        improved = False
        for i in range(n):
            if h[i] != 1.0:
                continue
            for j in [(i - 1) % n, (i + 1) % n]:
                if h[j] != 0.0:
                    continue
                h[i], h[j] = h[j], h[i]
                c5n = compute_c5(h)
                if c5n < c5 - 1e-15:
                    c5 = c5n
                    improved = True
                else:
                    h[i], h[j] = h[j], h[i]
        if not improved:
            break
    return h, c5

def sa_bin(h0, n, ts, t_end, rng, adj_frac):
    h = h0.copy()
    dx = 2.0 / n
    c5c = compute_c5(h)
    bc5 = c5c
    bh = h.copy()
    T = 0.015
    mT = 0.0012
    ni = 0
    sl = 30000
    while time.time() < t_end:
        if rng.random() < adj_frac:
            i = rng.randint(n)
            while h[i] == 0.0:
                i = rng.randint(n)
            d = rng.randint(1, 8)
            j = (i + d) % n if rng.random() < 0.5 else (i - d) % n
            if h[j] == 1.0:
                j = rng.randint(n)
                while h[j] == 1.0:
                    j = rng.randint(n)
        else:
            i = rng.randint(n)
            while h[i] == 0.0:
                i = rng.randint(n)
            j = rng.randint(n)
            while h[j] == 1.0:
                j = rng.randint(n)
        if i == j:
            continue
        h[i], h[j] = h[j], h[i]
        nc5 = compute_c5(h)
        dn = nc5 - c5c
        if dn < 0.0 or dn / max(T, 1e-12) < np.log(rng.uniform() + 1e-15):
            c5c = nc5
            if nc5 < bc5 - 1e-13:
                bc5 = nc5
                bh = h.copy()
                ni = 0
            else:
                ni += 1
        else:
            h[i], h[j] = h[j], h[i]
        if ni > sl:
            h = bh.copy()
            c5c = bc5
            T = min(max(T * 1.8, mT), 0.03)
            ni = 0
        T = max(T * 0.99998, mT)
    return bh, bc5

def slsqp_opt(h0, n, ts, alpha, topk, mi):
    global _g_alpha, _g_topk, _g_ts
    _setup(n)
    _g_alpha = alpha
    _g_topk = topk
    _g_ts = float(ts)
    bd = [(0.0, 1.0)] * n
    cs = [{'type': 'eq', 'fun': _sfun, 'jac': _sjac}]
    try:
        r = minimize(_sf, h0.copy(), method='SLSQP', jac=_sg,
                     bounds=bd, constraints=cs,
                     options={'maxiter': mi, 'ftol': 1e-15})
        return project_sum(r.x, ts)
    except Exception:
        return project_sum(h0.copy(), ts)

def tc_opt(h0, n, ts, alpha, topk, mi):
    global _g_alpha, _g_topk
    _setup(n)
    _g_alpha = alpha
    _g_topk = topk
    A = np.ones((1, n))
    lc = LinearConstraint(A, ts, ts)
    bd = Bounds(np.zeros(n), np.ones(n))
    try:
        r = minimize(_sf, h0.copy(), method='trust-constr',
                     jac=_sg, bounds=bd, constraints=[lc],
                     options={'maxiter': mi, 'gtol': 1e-8, 'xrtol': 1e-10})
        return project_sum(r.x, ts)
    except Exception:
        return project_sum(h0.copy(), ts)

def interp_h(hs, ns, nd):
    xs = np.linspace(0.0, 2.0, ns, endpoint=False)
    xd = np.linspace(0.0, 2.0, nd, endpoint=False)
    return np.interp(xd, xs, hs, period=2.0)

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and globals()['initial_h_values'] is not None:
        try:
            best_h = np.array(globals()['initial_h_values'], dtype=np.float64)
            best_n = len(best_h)
            best_h = project_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    nm = best_n
    tsm = float(nm // 2)

    hb = make_bin(best_h, nm, int(tsm), rng)
    hb, c5sh = best_shift_bin(hb, nm)
    hb, c5lr = local_refine_bin(hb, nm)
    if c5lr < best_c5:
        best_c5 = c5lr
        best_h = hb.copy()

    sa1_end = t0 + 0.28 * budget_s
    hsa, c5sa = sa_bin(hb, nm, int(tsm), sa1_end, rng, 0.25)
    hsa, c5sa2 = best_shift_bin(hsa, nm)
    hsa, c5lr2 = local_refine_bin(hsa, nm)
    if c5lr2 < best_c5:
        best_c5 = c5lr2
        best_h = hsa.copy()

    sa2_end = t0 + 0.40 * budget_s
    if time.time() < sa2_end:
        rng2 = np.random.RandomState(seed + 7)
        hb2 = make_bin(best_h, nm, int(tsm), rng2)
        hsa2, c5sa3 = sa_bin(hb2, nm, int(tsm), sa2_end, rng2, 0.15)
        hsa2, c5sa4 = best_shift_bin(hsa2, nm)
        hsa2, c5lr3 = local_refine_bin(hsa2, nm)
        if c5lr3 < best_c5:
            best_c5 = c5lr3
            best_h = hsa2.copy()

    hw = best_h.copy()
    alphas = [3, 5, 10, 20, 50, 100, 200, 500, 1000, 2000, 5000, 10000, 20000, 50000, 100000]
    topks = [150, 150, 120, 100, 80, 70, 60, 50, 45, 35, 30, 25, 22, 20, 20]

    for ai in range(len(alphas)):
        tr = budget_s - (time.time() - t0)
        if tr < 25:
            break
        al = alphas[ai]
        tk = min(topks[ai], 2 * nm - 1)
        hw = slsqp_opt(hw, nm, tsm, al, tk, 600)
        c5x = compute_c5(hw)
        if c5x < best_c5 - 1e-11:
            best_c5 = c5x
            best_h = hw.copy()
        if al >= 1000:
            tr = budget_s - (time.time() - t0)
            if tr > 20:
                hw2 = tc_opt(hw, nm, tsm, al, tk, 200)
                c5x2 = compute_c5(hw2)
                if c5x2 < best_c5 - 1e-11:
                    best_c5 = c5x2
                    best_h = hw2.copy()
                if c5x2 < c5x:
                    hw = hw2.copy()

    h_snap = make_bin(best_h, nm, int(tsm), rng)
    h_snap, c5snap = best_shift_bin(h_snap, nm)
    h_snap, c5snap2 = local_refine_bin(h_snap, nm)
    if c5snap2 < best_c5:
        best_c5 = c5snap2
        best_h = h_snap.copy()

    for tri in range(8):
        tr = budget_s - (time.time() - t0)
        if tr < 20:
            break
        nl = 0.04 * (1.0 - tri / 8.0) + 0.01
        hs = best_h + rng.uniform(-nl, nl, nm)
        hs = project_sum(hs, tsm)
        for al, tk in [(2000, 40), (10000, 25), (30000, 20)]:
            if tr < 10:
                break
            ho = slsqp_opt(hs, nm, tsm, al, tk, 500)
            c5o = compute_c5(ho)
            if c5o < best_c5 - 1e-11:
                best_c5 = c5o
                best_h = ho.copy()
            hs = ho

    for nt in [336, 288]:
        tr = budget_s - (time.time() - t0)
        if tr < 60:
            break
        tst = float(nt // 2)
        hi = interp_h(best_h, best_n, nt)
        hi = project_sum(hi, tst)
        for al in [50, 500, 5000]:
            if tr < 25:
                break
            ht = slsqp_opt(hi, nt, tst, al, 30, 400)
            ct = compute_c5(ht)
            if ct < best_c5 - 1e-11:
                best_c5 = ct
                best_h = ht.copy()
                best_n = nt
            hi = ht

    best_h = project_sum(best_h, float(best_n) / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
