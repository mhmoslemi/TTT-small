    shifted -= np.max(shifted)

    obj = m_val + np.log(np.sum(exp_vals)) / alpha

        return smooth_max_obj_grad(x, alpha)[1]

def compute_c5(h):

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_gradient_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[lag:] += (1.0 - h[:n - lag])
            g[:n - lag] -= h[lag:]
    else:
        k = -lag
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[:n - k] += (1.0 - h[k:])
            g[k:] -= h[:n - k]
    return g

def smooth_max_obj_grad(h, alpha, top_frac):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    g = np.zeros(n)
    threshold = sorted(w)[-int((1-top_frac)*len(w)):]
    threshold = threshold[0] if len(threshold) > 0 else 0
    active = np.where(w > threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    for m_idx in active:
        lag = m_idx - (n - 1)
        g += w[m_idx] * corr_gradient_lag(h, lag)
    return obj, g

    def f(x):
        return smooth_max_obj_grad(x, alpha, top_frac)[0]
    def g(x):
        return smooth_max_obj_grad(x, alpha, top_frac)[1]

    bounds = [(0.0, 1.0)] * n
    try:

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_to_sum(res.x, target_sum)
    except Exception:
