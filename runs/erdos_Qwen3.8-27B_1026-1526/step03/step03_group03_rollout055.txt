    return B

def compute_gradient(h, w, n):
    nm1 = n - 1
    g = np.zeros(n)
    for j in range(n):
        m_lo = max(0, j + 1)
        m_hi = min(2 * nm1, j + nm1)
        if m_lo <= m_hi:
            m_range = np.arange(m_lo, m_hi + 1)
            q = j + nm1 - m_range
            g[j] += np.dot(w[m_range], 1.0 - h[q])
        m_lo2 = max(0, nm1 - j)
        m_hi2 = min(2 * nm1, 2 * nm1 - j)
        if m_lo2 <= m_hi2:
            m_range2 = np.arange(m_lo2, m_hi2 + 1)
            p = j + m_range2 - nm1
            g[j] += np.dot(w[m_range2], -h[p])
    return g

def solve_1d(A, D, lo, hi):
    if hi - lo < 1e-14:
        return (lo + hi) * 0.5
    t_grid = np.linspace(lo, hi, 16)
    vals = np.max(A[None, :] + D[None, :] * t_grid[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        return lo
    if best_idx == len(t_grid) - 1:
        return hi
    lo_r = t_grid[best_idx - 1]
    hi_r = t_grid[best_idx + 1]
    for _ in range(28):
        mid = (lo_r + hi_r) * 0.5
        if np.max(A + D * mid) <= np.max(A + D * lo_r):
            lo_r = mid
        else:
            hi_r = mid
    return (lo_r + hi_r) * 0.5

def grad_pair_sweep(h, corr, g, rng, n_cand=10):
    n = len(h)
    order = np.arange(n)
    rng.shuffle(order)
    improved = False
    for j in order:
        B_j = compute_B(h, j, n)
        best_val = float(np.max(corr))
        best_delta = 0.0
        best_k = -1
        best_D = None
        if g[j] > 0:
            neg_mask = g < 0
            sorted_cands = np.where(neg_mask)[0]
            if len(sorted_cands) > 0:
                sorted_cands = sorted_cands[np.argsort(-g[sorted_cands])][::-1]
            top = list(sorted_cands[:n_cand])
        else:
            pos_mask = g > 0
            sorted_cands = np.where(pos_mask)[0]
            if len(sorted_cands) > 0:
                sorted_cands = sorted_cands[np.argsort(g[sorted_cands])][::-1]
            top = list(sorted_cands[:n_cand])
        extra = [(j + 1) % n, (j - 1 + n) % n, (j + n // 2) % n]
        for _ in range(3):
            extra.append(rng.randint(n))
        seen = set()
        cand_list = []
        for c in top + extra:
            if c == j or c in seen:
                continue
            seen.add(c)
            cand_list.append(c)
        for k in cand_list:
            B_k = compute_B(h, k, n)
            D = B_j - B_k
            delta_lo = max(-h[j], h[k] - 1.0)
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-13:
                continue
            delta_opt = solve_1d(corr, D, delta_lo, delta_hi)
            val_opt = float(np.max(corr + D * delta_opt))
            if val_opt < best_val - 1e-13:
                best_val = val_opt
                best_delta = delta_opt
                best_D = D
                best_k = k
        if best_k >= 0 and abs(best_delta) > 1e-13:
            corr += best_delta * best_D
            h[j] = np.clip(h[j] + best_delta, 0.0, 1.0)
            h[best_k] = np.clip(h[best_k] - best_delta, 0.0, 1.0)
            improved = True
    return h, corr, improved

def gradient_step(h, corr, n, lr, alpha):
    m_val = np.max(corr)
    exp_v = np.exp(alpha * (corr - m_val))
    w = exp_v / np.sum(exp_v)
    g = compute_gradient(h, w, n)
    h_new = h - lr * g
    h_new = project_to_feasible(h_new, n / 2.0)
    return h_new

def slsqp_local(h0, n, target_sum, alpha, maxiter):
    def f_wrapper(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        m = np.max(corr)
        exp_v = np.exp(alpha * (corr - m))
        s = np.sum(exp_v)
        return m + np.log(s) / alpha

    def g_wrapper(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        m = np.max(corr)
        exp_v = np.exp(alpha * (corr - m))
        w = exp_v / np.sum(exp_v)
        return compute_gradient(h, w, n)

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f_wrapper, h0.copy(), method='SLSQP', jac=g_wrapper,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_feasible(res.x, target_sum)
    except Exception:
        return project_to_feasible(h0, target_sum)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals():
        try:
            best_h = np.array(initial_h_values, dtype=float).copy()
            best_n = len(best_h)
            best_h = project_to_feasible(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    def time_left():
        return budget_s - (time.time() - t_start)

    # Phase 1: Gradient-guided pair descent with periodic gradient steps (45%)
    t_end1 = t_start + 0.45 * budget_s
    h_work = best_h.copy()
    n_w = len(h_work)
    corr_w = compute_corr(h_work)
    stall = 0
    sweep_count = 0
    momentum = np.zeros(n_w)
    beta_mom = 0.9

    while time.time() < t_end1:
        alpha gm = 50
        m_val = np.max(corr_w)
        exp_v = np.exp(alpha gm * (corr_w - m_val)) if m_val > 0 else np.ones(len(corr_w))
        w = exp_v / np.sum(exp_v)
        g = compute_gradient(h_work, w, n_w)
        h_work, corr_w, imp = grad_pair_sweep(h_work, corr_w, g, rng, n_cand=12)
        sweep_count += 1
        c5_w = float(np.max(corr_w) * 2.0 / n_w)

        if c5_w < best_c5 - 1e-11:
            best_c5 = c5_w
            best_h = h_work.copy()
            best_n = n_w
            stall = 0
        else:
            stall += 1

        if sweep_count % 15 == 0 and time.time() < t_end1:
            lr = 0.002 * (0.95 ** (sweep_count // 15))
            h_work = gradient_step(h_work, corr_w, n_w, lr, 50)
            momentum = beta_mom * momentum + (1 - beta_mom) * (h_work - h_work)
            corr_w = compute_corr(h_work)

        if stall > 300:
            noise_scale = min(0.15, 0.02 * stall / 100)
            h_work = project_to_feasible(h_work + rng.uniform(-noise_scale, noise_scale, n_w), n_w / 2.0)
            corr_w = compute_corr(h_work)
            stall = 0

    # Phase 2: Multi-resolution exploration (30%)
    t_end2 = t_start + 0.75 * budget_s
    resolutions = [320, 400, 480, 560]
    for n_tr in resolutions:
        if time.time() >= t_end2:
            break
        ts_tr = n_tr / 2.0
        if abs(ts_tr - round(ts_tr)) > 1e-6:
            continue
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_tr, endpoint=False)
        h_interp = project_to_feasible(np.interp(x_new, x_old, best_h, period=2.0), ts_tr)
        h_rand = project_to_feasible(rng.uniform(0, 1, n_tr), ts_tr)

        for h_st in [h_interp, h_rand]:
            if time.time() >= t_end2 - 5:
                break
            corr_st = compute_corr(h_st)
            g_st = np.zeros(n_tr)
            m_tr = np.max(corr_st)
            ev_tr = np.exp(50 * (corr_st - m_tr)) if m_tr > 0 else np.ones(len(corr_st))
            w_tr = ev_tr / np.sum(ev_tr)
            g_st = compute_gradient(h_st, w_tr, n_tr)

            max_sweeps = max(5, int((t_end2 - 5 - time.time()) / 1.5))
            for s in range(max_sweeps):
                if time.time() >= t_end2 - 2:
                    break
                h_st, corr_st, _ = grad_pair_sweep(h_st, corr_st, g_st, rng, n_cand=8)
                c5_st = float(np.max(corr_st) * 2.0 / n_tr)
                if c5_st < best_c5:
                    best_c5 = c5_st
                    best_h = h_st.copy()
                    best_n = n_tr
                m_tr = np.max(corr_st)
                ev_tr = np.exp(50 * (corr_st - m_tr)) if m_tr > 0 else np.ones(len(corr_st))
                w_tr = ev_tr / np.sum(ev_tr)
                g_st = compute_gradient(h_st, w_tr, n_tr)

    # Phase 3: SLSQP polish (10%)
    t_end3 = t_start + 0.85 * budget_s
    n_f = best_n
    ts_f = n_f / 2.0
    for alpha in [30, 100, 300, 800]:
        if time.time() >= t_end3:
            break
        h_pol = slsqp_local(best_h, n_f, ts_f, alpha, maxiter=400)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
            best_n = n_f

    # Phase 4: Final intensive pair descent (15%)
    h_fin = best_h.copy()
    n_fin = best_n
    corr_fin = compute_corr(h_fin)
    fin_alpha = 80
    fin_stall = 0
    while time.time() < t_start + budget_s - 3:
        m_fin = np.max(corr_fin)
        ev_fin = np.exp(fin_alpha * (corr_fin - m_fin)) if m_fin > 0 else np.ones(len(corr_fin))
        w_fin = ev_fin / np.sum(ev_fin)
        g_fin = compute_gradient(h_fin, w_fin, n_fin)
        h_fin, corr_fin, _ = grad_pair_sweep(h_fin, corr_fin, g_fin, rng, n_cand=14)
        c5_fin = float(np.max(corr_fin) * 2.0 / n_fin)
        if c5_fin < best_c5 - 1e-12:
            best_c5 = c5_fin
            best_h = h_fin.copy()
            best_n = n_fin
            fin_stall = 0
        else:
            fin_stall += 1
        if fin_stall > 200:
            h_fin = project_to_feasible(h_fin + rng.uniform(-0.03, 0.03, n_fin), n_fin / 2.0)
            corr_fin = compute_corr(h_fin)
            fin_stall = 0
            fin_alpha = max(30, fin_alpha * 0.8)

    best_h = project_to_feasible(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
