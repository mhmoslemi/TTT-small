def smooth_max_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    w = exp_v / np.sum(exp_v)
    obj = m_val + (max_sh + np.log(np.sum(exp_v))) / alpha
    nj = np.arange(n)[:, None]
    nm = np.arange(2 * n - 1)[None, :]
    fi = nj - nm + n - 1
    si = nj + nm - n + 1
    vf = (fi >= 0) & (fi < n)
    vs = (si >= 0) & (si < n)
    fc = np.clip(fi, 0, n - 1)
    sc = np.clip(si, 0, n - 1)
    grad = np.sum(vf * (1.0 - h[fc]) * w[None, :] - vs * h[sc] * w[None, :], axis=1)
    return obj, grad

def slsqp_opt(h0, n, ts, alpha, maxiter=800):

        return smooth_max_obj_grad(x, alpha)[0]
    def g(x):
        return smooth_max_obj_grad(x, alpha)[1]

        temp = max(1e-6, alpha_sa * np.exp(-step / 4000.0))
    return h, c5_cur

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()

    def time_left():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        if 'initial_h_values' in globals() and initial_h_values is not None:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
    except Exception:
        pass

    if best_h is None:
        best_n = 480
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    target_ns = [512, 640]

    for n_tr in target_ns:
        if time_left() < 120:
            break
        if n_tr % 2 != 0:
            continue
        ts_tr = n_tr / 2.0
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_tr, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h, period=2.0)
        h_up = project_to_sum(h_up, ts_tr)

        # Phase A: MM-LP iterations
        h_mm, c5_mm = mm_sequence(h_up, n_tr, max_iter=300, time_limit=time_left() - 10, t_start=t_start)
        if c5_mm < best_c5:
            best_c5 = c5_mm
            best_h = h_mm.copy()
            best_n = n_tr

        # Phase B: SLSQP polish
        if time_left() > 60:
            for alpha in [200, 1000, 5000]:
                if time_left() < 25:
                    break
                h_s = slsqp_opt(best_h, best_n, best_n / 2.0, alpha, maxiter=800)
                c5_s = compute_c5(h_s)
                if c5_s < best_c5:
                    best_c5 = c5_s
                    best_h = h_s.copy()

    # Phase C: Multiple MM restarts with frequency perturbation
    n_b = best_n
    for trial in range(40):
        if time_left() < 40:
            break
        amp = 0.06 * (1.0 - trial / 50.0) + 0.01
        nf = max(3, n_b // 30)
        h_sh = freq_perturb(best_h, rng, nf, amp)
        h_mm2, c5_mm2 = mm_sequence(h_sh, n_b, max_iter=150, time_limit=time_left() - 10, t_start=t_start)
        if c5_mm2 < best_c5:
            best_c5 = c5_mm2
            best_h = h_mm2.copy()

    # Phase D: SLSQP polish at different alphas
    if time_left() > 80:
        for alpha in [500, 2000, 10000]:
            if time_left() < 30:
                break
            h_p = slsqp_opt(best_h, best_n, best_n / 2.0, alpha, maxiter=1500)
            c5_p = compute_c5(h_p)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_p.copy()

    # Phase E: Another round of MM with the improved solution
    if time_left() > 100:
        h_mm3, c5_mm3 = mm_sequence(best_h, best_n, max_iter=200, time_limit=time_left() - 10, t_start=t_start)
        if c5_mm3 < best_c5:
            best_c5 = c5_mm3
            best_h = h_mm3.copy()

    # Phase F: Stochastic CD for fine-tuning
    if time_left() > 120:
        h_cd, c5_cd = stochastic_cd(best_h, best_n, best_n / 2.0, rng,
                                    n_steps=60000, delta_range=(0.003, 0.12))
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
        # Polish after stochastic
        if time_left() > 50:
            h_pol = slsqp_opt(best_h, best_n, best_n / 2.0, 5000, maxiter=2000)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Phase G: Final MM squeeze
    if time_left() > 60:
        h_final_mm, c5_final_mm = mm_sequence(best_h, best_n, max_iter=100,
                                              time_limit=time_left() - 10, t_start=t_start)
        if c5_final_mm < best_c5:
            best_c5 = c5_final_mm
            best_h = h_final_mm.copy()

    # Phase H: Tiny stochastic + final SLSQP
    if time_left() > 50:
        h_t, c5_t = stochastic_cd(best_h, best_n, best_n / 2.0, rng,
                                  n_steps=30000, delta_range=(0.002, 0.06))
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()
        if time_left() > 30:
            h_f = slsqp_opt(best_h, best_n, best_n / 2.0, 20000, maxiter=3000)
            c5_f = compute_c5(h_f)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f.copy()

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    for _ in range(500):
        diff = best_n / 2.0 - np.sum(best_h)
        if abs(diff) < 1e-15:
            break
        mask = (best_h > 1e-12) & (best_h < 1.0 - 1e-12)
        if np.any(mask):
            best_h[mask] += diff / len(mask)
            best_h = np.clip(best_h, 0.0, 1.0)
        else:
            break
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
