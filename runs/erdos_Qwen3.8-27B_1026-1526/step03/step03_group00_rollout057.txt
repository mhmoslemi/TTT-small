def compute_c5(h):

def compute_corr(h):

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)

def corr_gradient(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[lag:] += (1.0 - h[:n - lag])
            g[:n - lag] -= h[lag:]
    else:
        k = -lag
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[:n - k] += (1.0 - h[k:])
            g[k:] -= h[:n - k]
    return g

    # Find lambda such that sum(v) = target_sum

    # If we set v=1 for the k smallest g values, sum = k

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    # Gradient: sum_k w[k] * grad_k(h)
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        if w[k_idx] < 1e-10:
            continue
        lag = k_idx - (n - 1)
        grad += w[k_idx] * corr_gradient(h, lag)
    return obj, grad

def slsqp_local(h0, n, target_sum, alpha, maxiter):

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_gradient(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[:n - lag] += (1.0 - h[lag:])
            g[lag:] -= h[:n - lag]
    else:
        k = -lag
        g[:n - k] += (1.0 - h[k:])
        g[k:] -= h[:n - k]
    return g

def smooth_max_val(h, alpha):
    corr = compute_corr(h)
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    return m_val + np.log(np.sum(exp_vals)) / alpha

_slsqp_alpha = 100.0
_slsqp_target = 0.0
_slsqp_n = 0

def _slsqp_f(x):
    return smooth_max_val(x, _slsqp_alpha)

def _slsqp_g(x):
    n = len(x)
    corr = compute_corr(x)
    m_val = np.max(corr)
    shifted = _slsqp_alpha * (corr - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        if w[k_idx] < 1e-12:
            continue
        lag = k_idx - (n - 1)
        if lag >= 0:
            if lag == 0:
                grad[:] += w[k_idx] * (1.0 - 2.0 * x)
            else:
                grad[:n - lag] += w[k_idx] * (1.0 - x[lag:])
                grad[lag:] -= w[k_idx] * x[:n - lag]
        else:
            k = -lag
            grad[:n - k] += w[k_idx] * (1.0 - x[k:])
            grad[k:] -= w[k_idx] * x[:n - k]
    return grad

def _slsqp_con_fun(x):
    return np.sum(x) - _slsqp_target

def _slsqp_con_jac(x):
    return np.ones(_slsqp_n)

def slsqp_local(h0, n, target_sum, alpha, maxiter):
    global _slsqp_alpha, _slsqp_target, _slsqp_n
    _slsqp_alpha = alpha
    _slsqp_target = target_sum
    _slsqp_n = n
    cons = [{'type': 'eq', 'fun': _slsqp_con_fun, 'jac': _slsqp_con_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_slsqp_f, h0.copy(), method='SLSQP', jac=_slsqp_g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def frank_wolfe(h0, target_sum, max_iters=5000):
    n = len(h0)
    h = h0.copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    stuck = 0
    for t in range(max_iters):
        corr = compute_corr(h)
        m_star = int(np.argmax(corr))
        lag = m_star - (n - 1)
        g = corr_gradient(h, lag)
        order = np.argsort(g)
        k_full = int(target_sum)
        frac = target_sum - k_full
        v = np.zeros(n)
        if k_full > 0:
            v[order[:k_full]] = 1.0
        if frac > 1e-10 and k_full < n:
            v[order[k_full]] = frac
        gamma = 2.0 / (t + 2)
        h_new = (1.0 - gamma) * h + gamma * v
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            stuck = 0
        else:
            stuck += 1
        h = h_new
        if stuck > 500:
            break
    return best_h, best_c5

def mirror_descent(h0, target_sum, max_iters=5000, eta_init=0.3):
    n = len(h0)
    h = np.maximum(h0, 1e-10).copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    stuck = 0
    for t in range(max_iters):
        corr = compute_corr(h)
        m_star = int(np.argmax(corr))
        lag = m_star - (n - 1)
        g = corr_gradient(h, lag)
        eta = eta_init / (1.0 + 0.005 * t)
        log_h = np.log(h) - eta * g
        log_h -= np.max(log_h)
        exp_vals = np.exp(log_h)
        h_new = exp_vals * (target_sum / np.sum(exp_vals))
        h_new = np.clip(h_new, 1e-10, 1.0)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            stuck = 0
        else:
            stuck += 1
            h_try = 0.5 * h + 0.5 * h_new
            h_try = project_to_sum(h_try, target_sum)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-12:
                best_c5 = c5_try
                best_h = h_try.copy()
                h = h_try
            else:
                h = h_new
        h = h_new
        if stuck > 800:
            break
    return best_h, best_c5

def coord_desc_ls(h0, target_sum, max_iters=20000, rng=None):
    n = len(h0)
    h = h0.copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    stuck = 0
    for it in range(max_iters):
        corr = compute_corr(h)
        m_star = int(np.argmax(corr))
        lag = m_star - (n - 1)
        g = corr_gradient(h, lag)
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        g_inc = np.where(can_inc, g, 1e10)
        g_dec = np.where(can_dec, g, -1e10)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        if i_inc == i_dec or g[i_inc] >= g[i_dec] - 1e-10:
            idx_inc = np.where(can_inc)[0]
            idx_dec = np.where(can_dec)[0]
            if len(idx_inc) == 0 or len(idx_dec) == 0:
                break
            i_inc = int(idx_inc[rng.randint(len(idx_inc))])
            i_dec = int(idx_dec[rng.randint(len(idx_dec))])
            if g[i_inc] >= g[i_dec] - 1e-10:
                stuck += 1
                if stuck > 100:
                    break
                continue
        max_move = min(1.0 - h[i_inc], h[i_dec])
        step = max_move
        improved = False
        for _ in range(15):
            h[i_inc] += step
            h[i_dec] -= step
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, target_sum)
            c5_new = compute_c5(h)
            if c5_new < c5 - 1e-13:
                c5 = c5_new
                improved = True
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                break
            h[i_inc] -= step
            h[i_dec] += step
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, target_sum)
            step *= 0.5
        if not improved:
            stuck += 1
            if stuck > 200:
                break
        else:
            stuck = 0
    return best_h, best_c5

def fourier_refine(h0, target_sum, modes_max=12):
    n = len(h0)
    x = np.linspace(0, 2.0, n, endpoint=False)
    h = h0.copy()
    c5 = compute_c5(h)
    basis = []
    for m in range(1, modes_max + 1):
        basis.append((m, 'cos', np.cos(m * np.pi * x)))
        basis.append((m, 'sin', np.sin(m * np.pi * x)))
    for step_size in [0.02, 0.01, 0.005, 0.002, 0.001]:
        for m, kind, vec in basis:
            for direction in [1, -1]:
                h_try = h + direction * step_size * vec
                h_try = np.clip(h_try, 0.0, 1.0)
                h_try = project_to_sum(h_try, target_sum)
                c5_try = compute_c5(h_try)
                if c5_try < c5 - 1e-12:
                    h = h_try
                    c5 = c5_try
    return h, c5

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h, period=2.0)

def nesterov_subgrad(h0, target_sum, max_iters=5000, gamma_init=0.5):
    n = len(h0)
    h = h0.copy()
    y = h0.copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    stuck = 0
    momentum = 0.9
    for t in range(max_iters):
        corr = compute_corr(y)
        m_star = int(np.argmax(corr))
        lag = m_star - (n - 1)
        g = corr_gradient(y, lag)
        gamma = gamma_init / (1.0 + 0.01 * t)
        y_new = y - gamma * g
        y_new = project_to_sum(y_new, target_sum)
        h_new = (1.0 - momentum) * h + momentum * y_new
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            stuck = 0
        else:
            stuck += 1
        c5 = c5_new
        h = h_new
        y = y_new
        if stuck > 500:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()

    def time_left():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 540
        best_h = np.full(best_n, 0.5)
        x = np.linspace(0, 2.0, best_n, endpoint=False)
        best_h = 0.5 + 0.3 * np.cos(np.pi * x) + 0.1 * np.cos(2 * np.pi * x)
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = project_to_sum(best_h, best_n / 2.0)
        best_c5 = compute_c5(best_h)

    n_parent = best_n
    ts_parent = n_parent / 2.0

    if time_left() > 120:
        h_fw, c5_fw = frank_wolfe(best_h, ts_parent, max_iters=4000)
        if c5_fw < best_c5:
            best_c5 = c5_fw
            best_h = h_fw.copy()

    if time_left() > 120:
        h_md, c5_md = mirror_descent(best_h, ts_parent, max_iters=4000, eta_init=0.2)
        if c5_md < best_c5:
            best_c5 = c5_md
            best_h = h_md.copy()

    if time_left() > 100:
        h_ns, c5_ns = nesterov_subgrad(best_h, ts_parent, max_iters=4000, gamma_init=0.3)
        if c5_ns < best_c5:
            best_c5 = c5_ns
            best_h = h_ns.copy()

    for alpha in [50, 200, 800, 3000]:
        if time_left() < 40:
            break
        h_opt = slsqp_local(best_h, n_parent, ts_parent, alpha, maxiter=800)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    if time_left() > 100:
        h_cd, c5_cd = coord_desc_ls(best_h, ts_parent, max_iters=20000, rng=rng)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    if time_left() > 80:
        h_four, c5_four = fourier_refine(best_h, ts_parent, modes_max=12)
        if c5_four < best_c5:
            best_c5 = c5_four
            best_h = h_four.copy()
        if time_left() > 50:
            h_cd2, c5_cd2 = coord_desc_ls(best_h, ts_parent, max_iters=15000, rng=rng)
            if c5_cd2 < best_c5:
                best_c5 = c5_cd2
                best_h = h_cd2.copy()

    for n_tr in [600, 720, 900]:
        if time_left() < 180:
            break
        ts_tr = n_tr / 2.0
        h_interp = interp_to_n(best_h, n_tr)
        h_interp = project_to_sum(h_interp, ts_tr)
        c5_interp = compute_c5(h_interp)
        if c5_interp < best_c5:
            best_c5 = c5_interp
            best_h = h_interp.copy()
            best_n = n_tr
        for alpha in [100, 500, 2000, 8000]:
            if time_left() < 50:
                break
            h_opt = slsqp_local(h_interp, n_tr, ts_tr, alpha, maxiter=600)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_tr
            h_interp = h_opt
        if time_left() > 60:
            h_cd, c5_cd = coord_desc_ls(h_interp, ts_tr, max_iters=12000, rng=rng)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()
                best_n = n_tr
        if time_left() > 50:
            h_four, c5_four = fourier_refine(h_interp, ts_tr, modes_max=10)
            if c5_four < best_c5:
                best_c5 = c5_four
                best_h = h_four.copy()
                best_n = n_tr

    n_best = best_n
    ts_best = n_best / 2.0
    x = np.linspace(0, 2.0, n_best, endpoint=False)

    for trial in range(18):
        if time_left() < 45:
            break
        trial_type = trial % 6
        if trial_type == 0:
            h_shake = best_h + rng.uniform(-0.04, 0.04, n_best)
        elif trial_type == 1:
            freq = rng.randint(1, 6)
            h_shake = best_h + 0.04 * np.sin(np.pi * freq * x + rng.uniform(0, 2 * np.pi))
        elif trial_type == 2:
            freq = rng.randint(2, 12)
            h_shake = best_h + 0.03 * np.cos(np.pi * freq * x + rng.uniform(0, 2 * np.pi))
        elif trial_type == 3:
            h_shake = best_h.copy()
            blk = n_best // rng.randint(3, 10)
            st = rng.randint(0, max(1, n_best - blk))
            h_shake[st:st + blk] += 0.08 * rng.random()
        elif trial_type == 4:
            pert = rng.randn(n_best) * 0.03
            pert = (pert + pert[::-1]) / 2.0
            h_shake = best_h + pert
        else:
            h_shake = best_h + 0.05 * rng.randn(n_best)
            h_shake = (h_shake * 0.5 + interp_to_n(h_shake, n_best) * 0.5)

        h_shake = np.clip(h_shake, 0.0, 1.0)
        h_shake = project_to_sum(h_shake, ts_best)

        h_cd, c5_cd = coord_desc_ls(h_shake, ts_best, max_iters=6000, rng=rng)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

        if time_left() < 35:
            break
        h_opt = slsqp_local(h_cd, n_best, ts_best, 500, maxiter=500)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    if time_left() > 60:
        h_cd, c5_cd = coord_desc_ls(best_h, ts_best, max_iters=25000, rng=rng)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    for alpha in [3000, 10000, 50000]:
        if time_left() < 25:
            break
        h_pol = slsqp_local(best_h, n_best, ts_best, alpha, maxiter=1200)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    if time_left() > 30:
        h_cd, c5_cd = coord_desc_ls(best_h, ts_best, max_iters=20000, rng=rng)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    if time_left() > 20:
        h_four, c5_four = fourier_refine(best_h, ts_best, modes_max=15)
        if c5_four < best_c5:
            best_c5 = c5_four
            best_h = h_four.copy()

    best_h = project_to_sum(best_h, n_best / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_best)
