import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_grad_single(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            g[:m] += (1.0 - h[lag:])
            g[lag:] -= h[:m]
    else:
        k = -lag
        m = n - k
        if m > 0:
            g[k:] += (1.0 - h[:m])
            g[:m] -= h[k:]
    return g

def topk_obj_grad(h, k):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[-k:]
    grad = np.zeros(n)
    for idx in top_idx:
        lag = int(idx) - (n - 1)
        grad += corr_grad_single(h, lag)
    return float(np.sum(corr[top_idx])), grad

def smoothmax_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted -= np.max(shifted)
    ev = np.exp(shifted)
    s = ev.sum()
    w = ev / s
    obj = m + np.log(s) / alpha
    grad = np.zeros(n)

        if w[ii] > 1e-7:
            lag = ii - (n - 1)
            grad += w[ii] * corr_grad_single(h, lag)
    return float(obj), grad

    bounds = [(0.0, 1.0)] * n
    
    def f_g(x):
        return smoothmax_obj_grad(x, alpha)
    
    def f(x):
        return f_g(x)[0]
    
    def g(x):
        return f_g(x)[1]
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,

        return h_res
    except Exception:
        return project_feasible(h0.copy(), target_sum)
