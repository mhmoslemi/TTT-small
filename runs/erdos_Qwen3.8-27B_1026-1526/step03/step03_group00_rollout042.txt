import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr_fft(h):
    n = len(h)
    N = 2 * n - 1
    A = np.fft.rfft(h, N)
    B = np.fft.rfft(1.0 - h, N)
    y = np.fft.irfft(A * np.conj(B), N)
    result = np.empty(2 * n - 1)
    result[:n-1] = y[n:]
    result[n-1:] = y[:n]
    return result

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_gradient(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[lag:] += (1.0 - h[:n - lag])
            g[:n - lag] -= h[lag:]
    else:
        k = -lag
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[:n - k] += (1.0 - h[k:])
            g[k:] -= h[:n - k]
    return g

def frank_wolfe(h0, target_sum, max_iter=2000, use_fft=True):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for t in range(max_iter):
        if use_fft:
            corr = compute_corr_fft(h)
        else:
            corr = np.correlate(h, 1.0 - h, mode="full")
        k_star = int(np.argmax(corr))
        lag = k_star - (n - 1)
        grad = corr_gradient(h, lag)
        # Linear minimization oracle
        sorted_idx = np.argsort(grad)
        v = np.zeros(n)
        v[sorted_idx[:target_sum]] = 1.0
        gamma = 2.0 / (t + 2)
        h = (1.0 - gamma) * h + gamma * v
        h = project_to_sum(h, target_sum)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def coordinate_descent_fft(h, target_sum, max_iters=20000, step_frac=0.15):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    stale = 0
    for it in range(max_iters):
        corr = compute_corr_fft(h)
        m_star = int(np.argmax(corr))
        lag = m_star - (n - 1)
        g = corr_gradient(h, lag)
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        g_inc = np.where(can_inc, g, np.inf)
        g_dec = np.where(can_dec, g, -np.inf)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        if i_inc == i_dec:
            break
        if g[i_inc] >= g[i_dec] - 1e-12:
            break
        # Line search: try several step sizes
        best_move = step_frac * min(1.0 - h[i_inc], h[i_dec])
        best_c5_try = c5
        for frac in [step_frac * 0.5, step_frac, step_frac * 1.5, step_frac * 2.0]:
            move = frac * min(1.0 - h[i_inc], h[i_dec])
            if move < 1e-12:
                continue
            h[i_inc] += move
            h[i_dec] -= move
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, target_sum)
            c5_try = compute_c5(h)
            if c5_try < best_c5_try:
                best_c5_try = c5_try
                best_move = move
                best_frac = frac
        # Apply best move
        h[i_inc] -= best_move  # undo
        h[i_dec] += best_move
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
        # Re-apply
        h[i_inc] += best_move
        h[i_dec] -= best_move
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
        c5_new = compute_c5(h)
        if c5_new >= c5 - 1e-14:
            h[i_inc] -= best_move
            h[i_dec] += best_move
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, target_sum)
            stale += 1
            if stale > 100:
                break
        else:

            stale = 0

One more thought: the `smooth_max_and_grad` function in the previous code is O(n * (2n-1)) = O(n²) because it constructs an n × (2n-1) matrix. This is the main bottleneck for SLSQP at high n. 

An alternative: compute the gradient of the smooth max more efficiently. The gradient is:
∇F_smooth = sum_k w_k * ∇g_k

where w_k = exp(α(g_k - max_g)) / sum_j exp(α(g_j - max_g))

We can compute this without constructing the full matrix:
- First compute all g_k (using FFT): O(n log n)
- Compute weights w_k: O(n)
- Then for each k with significant weight, add w_k * ∇g_k to the gradient: O(n * n_w) where n_w is the number of non-negligible weights

For large α, only a few weights are significant, so this is fast.

```python
def smooth_max_grad_efficient(h, alpha):
    n = len(h)
    corr = compute_corr_fft(h)
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    # Only use the top weights
    threshold = 1.0 / (2 * n - 1)  # uniform weight level
    significant = np.where(w > threshold)[0]
    grad = np.zeros(n)
    for kIdx in significant:
        lag = kIdx - (n - 1)
        grad += w[kIdx] * corr_gradient(h, lag)
    return obj, grad
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr_fft(h):
    n = len(h)
    N = 2 * n - 1
    A = np.fft.rfft(h, N)
    B = np.fft.rfft(1.0 - h, N)
    y = np.fft.irfft(A * np.conj(B), N)
    result = np.empty(2 * n - 1)
    result[:n - 1] = y[n:]
    result[n - 1:] = y[:n]
    return result

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_gradient(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[lag:] += (1.0 - h[:n - lag])
            g[:n - lag] -= h[lag:]
    else:
        k = -lag
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[:n - k] += (1.0 - h[k:])
            g[k:] -= h[:n - k]
    return g

def smooth_max_grad_efficient(h, alpha):
    n = len(h)
    corr = compute_corr_fft(h)
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    threshold = 1.0 / (2 * n - 1)
    significant = np.where(w > threshold)[0]
    grad = np.zeros(n)
    for kIdx in significant:
        lag = kIdx - (n - 1)
        grad += w[kIdx] * corr_gradient(h, lag)
    return obj, grad

def slsqp_local(h0, n, target_sum, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n

    def f(x):
        return smooth_max_grad_efficient(x, alpha)[0]

    def g(x):
        return smooth_max_grad_efficient(x, alpha)[1]

    def cfun(x):
        return np.sum(x) - target_sum

    def cjac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def frank_wolfe(h0, target_sum, max_iter=3000):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for t in range(max_iter):
        corr = compute_corr_fft(h)
        k_star = int(np.argmax(corr))
        lag = k_star - (n - 1)
        grad = corr_gradient(h, lag)
        sorted_idx = np.argsort(grad)
        v = np.zeros(n)
        v[sorted_idx[:target_sum]] = 1.0
        gamma = 2.0 / (t + 2)
        h = (1.0 - gamma) * h + gamma * v
        h = project_to_sum(h, target_sum)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def subgradient_momentum(h0, target_sum, max_iter=5000, momentum=0.85):
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for t in range(max_iter):
        corr = compose_corr_fft(h)
        k_star = int(np.argmax(corr))
        lag = k_star - (n - 1)
        grad = corr_gradient(h, lag)
        eta = 0.3 / (1.0 + 0.005 * t)
        v = momentum * v - eta * grad
        h = h + v
        h = project_to_sum(h, target_sum)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def coordinate_descent_fft(h, target_sum, max_iters=15000, step_frac=0.15):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    stale = 0
    for it in range(max_iters):
        corr = compose_corr_fft(h)
        m_star = int(np.argmax(corr))
        lag = m_star - (n - 1)
        g = corr_gradient(h, lag)
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        g_inc = np.where(can_inc, g, np.inf)
        g_dec = np.where(can_dec, g, -np.inf)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        if i_inc == i_dec:
            break
        if g[i_inc] >= g[i_dec] - 1e-12:
            break
        max_move = min(1.0 - h[i_inc], h[i_dec])
        best_c5_try = c5
        best_move = 0.0
        for frac in [0.3, 0.6, 1.0, 1.5]:
            move = step_frac * frac * max_move
            if move < 1e-12:
                continue
            h[i_inc] += move
            h[i_dec] -= move
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, target_sum)
            c5_try = compute_c5(h)
            if c5_try < best_c5_try:
                best_c5_try = c5_try
                best_move = move
                best_frac = frac
        if best_move > 0:
            h[i_inc] -= best_move
            h[i_dec] += best_move
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, target_sum)
            h[i_inc] += best_move
            h[i_dec] -= best_move
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, target_sum)
            c5_new = compute_c5(h)
            if c5_new < c5 - 1e-14:
                c5 = c5_new
                stale = 0
            else:
                h[i_inc] -= best_move
                h[i_dec] += best_move
                h = np.clip(h, 0.0, 1.0)
                h = project_to_sum(h, target_sum)
                stale += 1
                if stale > 200:
                    break
        else:
            stale += 1
            if stale > 200:
                break
    return h, c5

def multi_lag_cd(h, target_sum, max_iters=10000, step_frac=0.12, num_lags=3):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    stale = 0
    for it in range(max_iters):
        corr = compose_corr_fft(h)
        top_idx = np.argsort(corr)[-num_lags:]
        g = np.zeros(n)

            lag = idx - (n - 1)
            g += corr_gradient(h, lag)

        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):

        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        if i_inc == i_dec:
            break
        if g[i_inc] >= g[i_dec] - 1e-12:
            break

        move = step_frac * max_move
        h[i_inc] += move
        h[i_dec] -= move
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
        c5_new = compute_c5(h)
        if c5_new < c5 - 1e-14:

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)

def fourier_candidate(n, freq, amp, phase):
    x = np.linspace(0, 2.0, n, endpoint=False)
    h = 0.5 + amp * np.cos(np.pi * freq * x + phase)
    h = np.clip(h, 0.0, 1.0)
    h = project_to_sum(h, n // 2)
    return h

def compose_corr_fft(h):
    return compute_corr_fft(h)
