import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        n_active = np.sum(mask)
        if n_active > 0:
            h[mask] += diff / n_active
            h = np.clip(h, 0.0, 1.0)
        else:
            break
    return h

def slsqp_smooth(h0, n, ts, alpha, maxiter=500):
    def f(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = np.max(corr)
        shifted = alpha * (corr - m)
        shifted -= np.max(shifted)
        return m + (np.max(np.zeros(len(corr))) + np.log(np.sum(np.exp(shifted)))) / alpha
    
    def g(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = np.max(corr)
        shifted = alpha * (corr - m)
        shifted -= np.max(shifted)
        exp_v = np.exp(shifted)
        w = exp_v / np.sum(exp_v)
        # gradient computation
        grad = np.zeros(n)
        shifts = np.arange(-(n-1), n)
        for si, s in enumerate(shifts):
            ws = w[si]
            if ws < 1e-14:
                continue
            idx1 = np.arange(n) + s
            v1 = (idx1 >= 0) & (idx1 < n)
            grad[v1] += ws * (1.0 - x[idx1[v1]])
            idx2 = np.arange(n) - s
            v2 = (idx2 >= 0) & (idx2 < n)
            grad[v2] -= ws * x[idx2[v2]]
        return grad
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ts, 'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(f, h0.copy(), method='SLSQP', jac=g, bounds=bnds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-12})
        return project_to_sum(r.x, ts)
    except Exception:
        return project_to_sum(h0, ts)

        j = rng.randint(n)

        # Adaptive delta
        d = rng.uniform(0.01, 0.15)
        # Ensure feasibility after move
        hi_new = h[i] + d
        hj_new = h[j] - d
        if hi_new > 1.0:
            d = 1.0 - h[i]
            hi_new = 1.0
        if hj_new < 0.0:
            d = h[j]
            hj_new = 0.0
        if d < 1e-6:
            continue
        if hi_new > 1.0 or hj_new < 0.0:
            # Try reverse
            hi_new = h[i] - d
            hj_new = h[j] + d
            if hi_new < 0.0 or hj_new > 1.0:
                continue
            d_sign = -1
        else:
            d_sign = 1
        
        # Compute new correlation efficiently
        # Change at i: delta_i = d_sign * d
        # Change at j: delta_j = -d_sign * d
        # New h: h[i] -> h[i] + d_sign*d, h[j] -> h[j] - d_sign*d
        
        # Update correlation
        # For shift s (m = s+n-1 in full corr indexing):
        # delta_corr[m] = d_sign*d * [(1-h[i+s]) if valid - h[i-s] if valid]
        #                 - d_sign*d * [(1-h[j+s]) if valid - h[j-s] if valid]
        # = d_sign*d * [(1-h[i+s]) - h[i-s] - (1-h[j+s]) + h[j-s]] (with validity)
        
        # This is O(n) to compute
        new_corr = corr_cur.copy()
        di = d_sign * d
        # Effect of h[i] += di on corr:
        for s in range(-(n-1), n):
            m = s + n - 1
            delta = 0.0
            if 0 <= i + s < n:
                delta += (1.0 - h[i+s])
            if 0 <= i - s < n:
                delta -= h[i-s]
            if 0 <= j + s < n:
                delta -= (1.0 - h[j+s])
            if 0 <= j - s < n:
                delta += h[j-s]
            new_corr[m] += di * delta
        
        new_max = np.max(new_corr)
        c5_new = new_max * dx
        
        if c5_new < best_local:
            best_local = c5_new
            h[i] += di
            h[j] -= di
            c5_cur = c5_new
            corr_cur = new_corr
        else:

            if rng.random() < accept:
                h[i] += di
                h[j] -= di
                c5_cur = c5_new
                corr_cur = new_corr
            # else: reject, h unchanged
        
        temp = max(1e-6, 0.05 * np.exp(-step / 50000.0))
    
    return h, c5_cur, best_local

So ∂corr[m]/∂h[i] = (1-h[i+s])*[0≤i+s<n] - h[i-s]*[0≤i-s<n]

For a move δh[i] = di, δh[j] = -di:
δcorr[m] = di * [(1-h[i+s])*[0≤i+s<n] - h[i-s]*[0≤i-s<n]] - di * [(1-h[j+s])*[0≤j+s<n] - h[j-s]*[0≤j-s<n]]

Let me define for any index k and shift s:
D(k, s) = (1-h[k+s])*[0≤k+s<n] - h[k-s]*[0≤k-s<n]

Then δcorr[m] = di * (D(i, s) - D(j, s))

And D(k, s) for all s can be computed as:
D(k, :) = [(1-h[k+s]) for valid s] - [h[k-s] for valid s]

For fixed k, as s ranges from -(n-1) to (n-1):
- k+s ranges from k-(n-1) to k+n-1, valid when 0≤k+s<n, i.e., -k≤s≤n-1-k
- k-s ranges from k-(-(n-1)) to k-(n-1) = k+n-1 to k-n+1, valid when 0≤k-s<n, i.e., k-n+1≤s≤k

So D(k, s) is non-zero for s in [max(-(n-1), k-n+1), min(n-1, k)] ∪ ... wait, both conditions apply independently.

For the first term: valid when -k ≤ s ≤ n-1-k
For the second term: valid when k-n+1 ≤ s ≤ k

For k in [0, n-1]:
- First term valid: s ∈ [-k, n-1-k]
- Second term valid: s ∈ [k-n+1, k]

These overlap in s ∈ [max(-k, k-n+1), min(n-1-k, k)] = [max(-k, k-n+1), min(n-1-k, k)]

For k=0: first valid s ∈ [0, n-1], second valid s ∈ [-n+1, 0]. Overlap at s=0.
For k=n-1: first valid s ∈ [-(n-1), 0], second valid s ∈ [0, n-1]. Overlap at s=0.
For k=n/2: first valid s ∈ [-n/2, n/2-1], second valid s ∈ [-n/2+1, n/2]. Overlap in between.

```python
def compute_D(h, k, s_range):
    """Compute D(k, s) for all s in s_range."""
    n = len(h)
    D = np.zeros(len(s_range))
    ks = k + s_range
    ks_valid = (ks >= 0) & (ks < n)
    D[ks_valid] = 1.0 - h[ks[ks_valid]]
    km = k - s_range
    km_valid = (km >= 0) & (km < n)
    D[km_valid] -= h[km[km_valid]]
    return D
```

        shifted -= np.max(shifted)

            grad[v2] -= ws * x[im_vals[v2]]
