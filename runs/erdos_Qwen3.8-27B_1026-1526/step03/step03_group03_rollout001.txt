        else:
            h += diff / len(h)

    return h

def compute_subgrad(h, corr, alpha):
    """Compute softmax-weighted subgradient of max(corr)"""
    n = len(h)
    m_val = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m_val))
    w = exp_vals / np.sum(exp_vals)
    
    # Vectorized computation
    j_idx = np.arange(n)
    # First term: F[j] = sum_{i=j}^{n-1} w[i-j+n-1]*(1-h[i])
    # Second term: G[j] = sum_{i=0}^{j} w[n-1+j-i]*h[i]
    
    # Use broadcasting
    i = np.arange(n)[None, :]  # 1 x n
    j = np.arange(n)[:, None]  # n x 1
    
    mask_f = i >= j  # n x n
    idx_f = i - j + n - 1  # n x n, valid where mask_f
    F = np.sum(np.where(mask_f, (1-h)[None, :] * w[idx_f], 0), axis=1)
    
    mask_g = i <= j  # n x n
    idx_g = n - 1 + j - i  # n x n, valid where mask_g
    G = np.sum(np.where(mask_g, h[None, :] * w[idx_g], 0), axis=1)
    
    grad = F - G
    return grad

def subgradient_optimize(h0, n, target_sum, alpha, n_iters, rng):

    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 1e-9) & (h < 1.0 - 1e-9)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_B(h, j, n):
    nm1 = n - 1
    B = np.zeros(2 * n - 1)
    k_start = max(0, nm1 - j)
    k_end = min(2 * n - 2, 2 * nm1 - j)
    if k_start <= k_end:
        B[k_start:k_end+1] += (1.0 - h[j - nm1 + np.arange(k_start, k_end+1)])
    k_start2 = max(0, j)
    k_end2 = min(2 * n - 2, j + nm1)
    if k_start2 <= k_end2:
        B[k_start2:k_end2+1] += -h[j + nm1 - np.arange(k_start2, k_end2+1)]
    return B

def solve_1d(A, D, lo, hi):
    if hi - lo < 1e-14:
        return (lo + hi) * 0.5
    for _ in range(40):
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        if np.max(A + D * m1) <= np.max(A + D * m2):
            hi = m2
        else:
            lo = m1
    return (lo + hi) * 0.5

def pair_sweep(h, corr, rng, n_cand=14):
    n = len(h)
    order = np.arange(n)
    rng.shuffle(order)
    n_imp = 0
    for j in order:
        B_j = compute_B(h, j, n)
        best_val = np.max(corr)
        best_d = 0.0
        best_D = None
        best_k = -1
        k_set = [(j+1)%n, (j-1+n)%n, (j+n//2)%n, (j+n//3)%n,
                 (j+2*n//3)%n, (j+2)%n, (j-2+n)%n, (j+n//4)%n,
                 (j+3*n//4)%n, (j+4)%n, (j-4+n)%n, (j+n//5)%n]
        for _ in range(max(0, n_cand - 12)):
            k_set.append(rng.randint(n))
        seen = set()
        for k in k_set:
            if k == j or k in seen:
                continue
            seen.add(k)
            B_k = compute_B(h, k, n)
            D = B_j - B_k
            dlo = max(-h[j], h[k]-1.0)
            dhi = min(1.0-h[j], h[k])
            if dhi - dlo < 1e-13:
                continue
            d_opt = solve_1d(corr, D, dlo, dhi)
            v_opt = np.max(corr + D * d_opt)
            if v_opt < best_val - 1e-13:
                best_val = v_opt
                best_d = d_opt
                best_D = D
                best_k = k
        if best_k >= 0 and abs(best_d) > 1e-13:
            corr += best_d * best_D
            h[j] = np.clip(h[j] + best_d, 0, 1)
            h[best_k] = np.clip(h[best_k] - best_d, 0, 1)
            n_imp += 1
    return h, corr, n_imp

def snap_and_reopt(h, threshold=0.02):
    """Snap values near 0 or 1 to exact 0 or 1"""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    h[h < threshold] = 0.0
    h[h > 1.0 - threshold] = 1.0
    h = project_to_sum(h, target)
    return h

def gaussian_perturb(h, sigma=0.03, rng=None):
    """Apply a smooth random perturbation"""
    if rng is None:
        rng = np.random.RandomState(0)
    n = len(h)
    noise = rng.normal(0, sigma, n)
    # Smooth it
    kernel_size = 5
    kernel = np.exp(-0.5 * (np.arange(-kernel_size, kernel_size+1) / 1.5)**2)
    kernel /= kernel.sum()
    noise = np.convolve(noise, kernel, mode='same')
    h_new = h + noise
    h_new = project_to_sum(h_new, n / 2.0)
    return h_new
