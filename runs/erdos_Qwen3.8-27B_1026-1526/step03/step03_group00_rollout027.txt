Now, if I change h[j] from its current value to v, which C[k] values change?

C[k] = sum_i h[i+k] * (1-h[i])

h[j] appears in C[k] when:
- j = i+k, i.e., i = j-k. Then the term is h[j] * (1-h[j-k]) for valid j-k.
- j = i, i.e., the term is h[i+k] * (1-h[j]) = h[j+k] * (1-h[j]) for valid j+k.

= C[k]_old + (v - h[j]) * [(1-h[j-k]) * [0 ≤ j-k < n] - h[j+k] * [0 ≤ j+k < n]]

Let a_k = (1-h[j-k]) * [0 ≤ j-k < n] - h[j+k] * [0 ≤ j+k < n]

    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2, 0.0, 1.0)

def solve_1d_minimax(a, c):
    # Minimize max_k (a[k]*v + c[k]) over v in [0, 1]
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        vals = a * mid + c
        mval = vals.max()
        active = vals >= mval - 1e-12
        if not np.any(active):
            break
        max_slope = a[active].max()
        min_slope = a[active].min()
        if min_slope >= -1e-15:
            hi = mid
        elif max_slope <= 1e-15:
            lo = mid
        else:
            break
    return (lo + hi) / 2

    a = np.zeros(2*n - 1)
    valid1 = (j - k_vals >= 0) & (j - k_vals < n)
    if np.any(valid1):
        a[valid1] += (1.0 - h[j - k_vals[valid1]])
    valid2 = (j + k_vals >= 0) & (j + k_vals < n)
    if np.any(valid2):
        a[valid2] -= h[j + k_vals[valid2]]
    return a

    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for sweep in range(n_sweeps):
        if rng is not None:
            order = rng.permutation(n)
        else:
            order = range(n)
        
        for j in order:
            corr = np.correlate(h, 1.0 - h, mode="full")
            a = compute_a_vectorized(h, j, n)
            c = corr - h[j] * a
            v_star = solve_1d_minimax(a, c)
            h[j] = v_star
        
        h = project_exact(h, target_sum)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5
