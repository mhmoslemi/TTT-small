import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_B(h, j, n):
    nm1 = n - 1
    B = np.zeros(2 * n - 1)
    k_start = max(0, nm1 - j)
    k_end = min(2 * n - 2, 2 * nm1 - j)
    if k_start <= k_end:
        B[k_start:k_end + 1] += (1.0 - h[j - nm1 + np.arange(k_start, k_end + 1)])
    k_start2 = max(0, j)
    k_end2 = min(2 * n - 2, j + nm1)
    if k_start2 <= k_end2:
        B[k_start2:k_end2 + 1] += -h[j + nm1 - np.arange(k_start2, k_end2 + 1)]
    return B

def solve_1d(A, D, lo, hi):
    if hi - lo < 1e-14:
        return (lo + hi) * 0.5
    t_grid = np.linspace(lo, hi, 20)
    vals = np.max(A[None, :] + D[None, :] * t_grid[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        return lo
    if best_idx == len(t_grid) - 1:
        return hi
    lo_r = t_grid[best_idx - 1]
    hi_r = t_grid[best_idx + 1]
    for _ in range(45):
        mid = (lo_r + hi_r) * 0.5
        if np.max(A + D * mid) <= np.max(A + D * lo_r):
            lo_r = mid
        else:
            hi_r = mid
    return (lo_r + hi_r) * 0.5

def pair_descent_sweep(h, corr, rng, n_cand, target_lags=None):
    n = len(h)
    order = np.arange(n)
    rng.shuffle(order)
    improved = False
    for j in order:
        B_j = compute_B(h, j, n)
        best_val = np.max(corr)
        best_delta = 0.0
        best_D = None
        best_k = -1
        k_set = [(j + 1) % n, (j - 1 + n) % n, (j + n // 2) % n,
                 (j + n // 3) % n, (j + 2 * n // 3) % n, (j + n // 4) % n,
                 (j + 3 * n // 4) % n]
        if target_lags is not None:
            for tl in target_lags:
                k_set.append((j + tl) % n)
                k_set.append((j - tl) % n)
        for _ in range(max(0, n_cand - 7)):
            k_set.append(rng.randint(n))
        seen = set()
        for k in k_set:
            if k == j or k in seen:
                continue
            seen.add(k)
            B_k = compute_B(h, k, n)
            D = B_j - B_k
            delta_lo = max(-h[j], h[k] - 1.0)
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-13:
                continue
            delta_opt = solve_1d(corr, D, delta_lo, delta_hi)
            val_opt = np.max(corr + D * delta_opt)
            if val_opt < best_val - 1e-13:
                best_val = val_opt
                best_delta = delta_opt
                best_D = D
                best_k = k
        if best_k >= 0 and abs(best_delta) > 1e-13:
            corr += best_delta * best_D
            h[j] += best_delta
            h[best_k] -= best_delta
            if h[j] < 0:
                h[j] = 0.0
            elif h[j] > 1:
                h[j] = 1.0
            if h[best_k] < 0:
                h[best_k] = 0.0
            elif h[best_k] > 1:
                h[best_k] = 1.0
            improved = True
    return h, corr, improved

def get_target_lags(corr, n_top):
    idx = np.argsort(corr)[-n_top:]
    lags = []
    for i in idx:
        k = i - (len(corr) - 1) // 2
        if k != 0:
            lags.append(k)
    return lags

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    sh = corr - m_val
    max_sh = np.max(sh)
    if max_sh > 50:
        sh = sh - max_sh
        m_val_adj = m_val + max_sh
    else:
        m_val_adj = m_val
    exp_vals = np.exp(alpha * sh)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val_adj + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def slsqp_local(h0, n, target_sum, alpha, maxiter):
    def f(h):
        return smooth_max_and_grad(h, alpha)[0]
    def g(h):
        return smooth_max_and_grad(h, alpha)[1]
    def cfun(h):
        return np.sum(h) - target_sum
    def cjac(h):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def proximal_gradient(h0, n, target_sum, alpha, steps, lr):
    h = h0.copy()
    for _ in range(steps):
        _, grad = smooth_max_and_grad(h, alpha)
        h = h - lr * grad
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
    return h

def simulated_annealing(h, corr, rng, n, t_start, t_end, t0, tf):
    n_corr = len(corr)
    current_c5 = np.max(corr)
    improved = False
    n_sweep = max(1, int((t_end - time.time()) / 0.05))
    for sweep in range(n_sweep):
        if time.time() >= t_end:
            break
        progress = (time.time() - t_start) / max(1e-9, t_end - t_start)
        T = max(tf, t0 * (1 - progress) ** 2)
        order = np.arange(n)
        rng.shuffle(order)
        for j in order:
            if time.time() >= t_end:
                break
            k = rng.randint(n)
            if k == j:
                continue
            B_j = compute_B(h, j, n)
            B_k = compute_B(h, k, n)
            D = B_j - B_k
            delta_lo = max(-h[j], h[k] - 1.0)
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-13:
                continue
            delta = rng.uniform(delta_lo, delta_hi)
            new_corr = corr + delta * D
            new_max = np.max(new_corr)
            delta_obj = new_max - current_c5
            if delta_obj < 0 or rng.random() < np.exp(-delta_obj / max(T, 1e-10)):
                corr = new_corr
                h[j] += delta
                h[k] -= delta
                h[j] = np.clip(h[j], 0.0, 1.0)
                h[k] = np.clip(h[k], 0.0, 1.0)
                current_c5 = new_max
                improved = True
    return h, corr, improved

def upsample(h_old, n_new, target_sum):
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old, period=2.0)
    return project_to_sum(h_new, target_sum)
