    no_improve = 0

        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = np.argmax(corr)
        offset = max_idx - (n - 1)
        
        # Compute subgradient
        grad = np.zeros(n)
        for j in range(n):
            val = 0.0
            if 0 <= j - offset < n:
                val += (1.0 - h[j - offset])
            if 0 <= j + offset < n:
                val -= h[j + offset]
            grad[j] = val
        
        # Take step
        h_new = h - step_size * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_to_sum(h_new, target_sum)

        if c5_new < c5_curr - 1e-10:
            h = h_new
            c5_curr = c5_new
            no_improve = 0
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h.copy()
            step_size = min(step_size * 1.05, 0.5)

            no_improve = 0

def smooth_max_slsqp(h0, n, target_sum, alpha, maxiter=300):
    """SLSQP with smooth max - vectorized gradient"""
    n_corr = 2 * n - 1
    
    def f_and_g(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        m_val = np.max(corr)
        shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
        exp_vals = np.exp(shifted)
        w = exp_vals / exp_vals.sum()
        obj = m_val + np.log(exp_vals.sum()) / alpha
        
        # Vectorized gradient computation
        offsets = np.arange(n_corr) - (n - 1)  # shape (n_corr,)
        # For each j in 0..n-1, grad[j] = sum_k w[k] * [(1-h[j-off_k]) if valid - h[j+off_k] if valid]
        # This is hard to vectorize simply, so let's do it in a loop
        grad = np.zeros(n)
        for j in range(n):
            # Case 1: j - offset in [0, n)
            idx1 = j - offsets  # shape (n_corr,)
            valid1 = (idx1 >= 0) & (idx1 < n)
            contrib1 = np.sum(w[valid1] * (1.0 - h[idx1[valid1]]))
            # Case 2: j + offset in [0, n)
            idx2 = j + offsets  # shape (n_corr,)
            valid2 = (idx2 >= 0) & (idx2 < n)
            contrib2 = np.sum(w[valid2] * h[idx2[valid2]])
            grad[j] = contrib1 - contrib2
        return obj, grad
    
    def f(h):
        return f_and_g(h)[0]
    def g(h):
        return f_and_g(h)[1]
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

        h_new = project_to_sum(h_new, target_sum)
        
        c5_new = compute_c5(h_new)
        if c5_new < c5_curr - 1e-12:
            h = h_new
            c5_curr = c5_new
            no_improve = 0
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h.copy()
            step_size = min(step_size * 1.1, 1.0)
        else:
            no_improve += 1
            if no_improve > 20:
                step_size *= 0.8
                no_improve = 0
        
        # Increase alpha over time (makes smooth max → true max)
        alpha = min(alpha * 1.01, alpha_end)
        
        # Restart with noise if stuck
        if no_improve > 200:
            h = best_h + 0.03 * np.random.randn(n)
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, target_sum)
            c5_curr = compute_c5(h)
            no_improve = 0
            step_size = 0.1
    
    return best_h, best_c5

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(5000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def make_binary(h, n, target_sum, rng):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones == target_sum:
        return h_bin
    if n_ones > target_sum:
        excess = n_ones - target_sum
        ones_idx = np.where(h_bin == 1.0)[0]
        order = np.argsort(h[ones_idx])[:excess]
        h_bin[ones_idx[order]] = 0.0
    else:
        deficit = target_sum - n_ones
        zeros_idx = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h[zeros_idx])[:deficit]
        h_bin[zeros_idx[order]] = 1.0
    return h_bin

def interpolate_h(h_src, n_src, n_dst):
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_src, period=2.0)

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w = exp_vals / exp_vals.sum()
    obj = m_val + np.log(exp_vals.sum()) / alpha
    n_corr = 2 * n - 1
    j_idx = np.arange(n)[:, None]
    k_idx = np.arange(n_corr)[None, :]
    off = k_idx - (n - 1)
    idx1 = j_idx - off
    valid1 = (idx1 >= 0) & (idx1 < n)
    idx2 = j_idx + off
    valid2 = (idx2 >= 0) & (idx2 < n)
    clipped1 = np.clip(idx1, 0, n - 1)
    clipped2 = np.clip(idx2, 0, n - 1)
    part1 = np.where(valid1, 1.0 - h[clipped1], 0.0)
    part2 = np.where(valid2, h[clipped2], 0.0)
    grad = part1 @ w - part2 @ w
    return obj, grad

            else:

def subgradient_worst_lag(h0, n, target_sum, t_end):
