    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum, max_iter=500):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_gradient_single_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        g[lag:] += 1.0
        g[lag:] -= h[:n - lag]
        g[:n - lag] -= h[lag:]
    else:
        k = -lag
        g[:n - k] += 1.0
        g[:n - k] -= h[k:]
        g[k:] -= h[:n - k]
    return g

def multi_lag_gradient(h, tol_frac=0.01):
    """Average gradient over all lags within tol_frac of max"""
    corr = compute_corr(h)
    m_val = np.max(corr)
    threshold = m_val * (1.0 - tol_frac)
    active = np.where(corr >= threshold)[0]
    n = len(h)
    grad = np.zeros(n)
    for m_idx in active:
        lag = int(m_idx) - (n - 1)
        grad += corr_gradient_single_lag(h, lag)
    grad /= len(active)
    return grad, m_val

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-10:
            continue
        lag = m_idx - (n - 1)
        grad += w[m_idx] * corr_gradient_single_lag(h, lag)
    return obj, grad

def slsqp_optimize(h0, n, target_sum, alpha, maxiter):
    def f(x):
        return smooth_max_obj_grad(x, alpha)[0]
    def g(x):
        return smooth_max_obj_grad(x, alpha)[1]
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def momentum_coord_desc(h, target_sum, max_iters=10000, step_frac=0.12, momentum=0.3, tol_frac=0.02):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    vel = np.zeros(n)
    stale = 0
    
    for it in range(max_iters):
        grad, m_val = multi_lag_gradient(h, tol_frac)
        vel = momentum * vel + (1 - momentum) * grad
        
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        
        g_eff = vel
        g_inc = np.where(can_inc, g_eff, np.inf)
        g_dec = np.where(can_dec, g_eff, -np.inf)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        
        if i_inc == i_dec:
            break
        if g_eff[i_inc] >= g_eff[i_dec] - 1e-12:
            break
        
        max_move = min(1.0 - h[i_inc], h[i_dec])
        move = step_frac * max_move
        if move < 1e-12:
            break
        
        h[i_inc] += move
        h[i_dec] -= move
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
        
        c5_new = compute_c5(h)
        if c5_new >= c5 - 1e-14:
            h[i_inc] -= move
            h[i_dec] += move
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, target_sum)
            vel *= 0.5
            stale += 1
            if stale > 200:
                break
        else:
            c5 = c5_new
            stale = 0
    return h, c5

def projected_subgradient(h, target_sum, max_iters=5000, init_step=0.01, tol_frac=0.01):
    """Backtracking line search + projected subgradient"""
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    step = init_step
    stale = 0
    
    for it in range(max_iters):
        grad, m_val = multi_lag_gradient(h, tol_frac)
        # Normalize gradient
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-12:
            break
        grad = grad / g_norm
        
        # Backtracking line search
        accepted = False
        for ls in range(20):
            h_new = h - step * (g_norm / step) * (grad / g_norm) * step
            # Actually let's just do: h_new = h - step * grad_normalized * scale
            h_new = h - step * grad
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = project_to_sum(h_new, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < c5 - 1e-14:
                accepted = True
                break
            step *= 0.5
        if not accepted:
            stale += 1
            if stale > 50:
                break
            continue
        
        h = h_new
        c5 = c5_new
        step = min(step * 1.5, 0.1)
        stale = 0
    return h, c5

def interpolate(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)

    return np.interp(x_new, x_old, h, period=2.0)

def block_perturb(h, rng, n_blocks, amp):

    return h + noise

        max_move = min(1.0 - h[i_inc], h[i_dec])
        move = step_frac * max_move
        if move < 1e-12:
            break
        h[i_inc] += move
        h[i_dec] -= move
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
        c5_new = compute_c5(h)
        if c5_new >= c5 - 1e-14:
            h[i_inc] -= move
            h[i_dec] += move
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, target_sum)
            vel *= 0.5
            stale += 1
            if stale > 200:
                break
        else:
            c5 = c5_new
            stale = 0
    return h, c5

def subgradient_method(h, target_sum, max_iters=3000, init_step=0.005, tol_frac=0.01):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    step = init_step
    stale = 0
    for it in range(max_iters):
        grad, m_val = multi_lag_gradient(h, tol_frac)
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-12:
            break
        grad_unit = grad / g_norm
        accepted = False
        for ls in range(15):
            h_new = h - step * grad_unit
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = project_to_sum(h_new, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < c5 - 1e-14:
                accepted = True
                break
            step *= 0.5
        if not accepted:
            stale += 1
            if stale > 40:
                break
            step = init_step
            continue
        h = h_new
        c5 = c5_new
        step = min(step * 1.3, 0.05)
        stale = 0
    return h, c5

def interpolate(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h, period=2.0)

def block_perturb(h, rng, n_blocks, amp):
    n = len(h)
    block_size = n // n_blocks
    noise = np.zeros(n)
    for i in range(n_blocks):
        start = i * block_size
        end = min(start + block_size, n)
        noise[start:end] = rng.uniform(-amp, amp)
    return h + noise

def smooth_perturb(h, rng, freq_low=2, freq_high=8, amp=0.05):
    n = len(h)
    x = np.linspace(0, 2.0, n, endpoint=False)
    noise = np.zeros(n)
    n_harmonics = rng.randint(2, 5)
    for _ in range(n_harmonics):
        freq = rng.randint(freq_low, freq_high)
        phase = rng.uniform(0, 2 * np.pi)
        amp_i = rng.uniform(amp * 0.5, amp)
        noise += amp_i * np.sin(np.pi * freq * x + phase)
    return h + noise

def binary_snap(h, target_sum, threshold=0.5):
    h_bin = (h > threshold).astype(float)
    return project_to_sum(h_bin, target_sum)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()

    def time_left():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 540
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n_parent = best_n
    ts_parent = n_parent / 2.0

    # Phase 1: SLSQP at parent resolution with increasing alpha
    for alpha in [50, 200, 1000, 5000]:
        if time_left() < 25:
            break
        h_new = slsqp_optimize(best_h, n_parent, ts_parent, alpha, maxiter=1000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 2: Multi-lag coord descent at parent
    if time_left() > 60:
        h_cd, c5_cd = momentum_coord_desc(best_h, ts_parent, max_iters=20000, step_frac=0.1, momentum=0.3, tol_frac=0.015)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 3: Subgradient method at parent
    if time_left() > 40:
        h_sub, c5_sub = subgradient_method(best_h, ts_parent, max_iters=3000, init_step=0.004, tol_frac=0.01)
        if c5_sub < best_c5:
            best_c5 = c5_sub
            best_h = h_sub.copy()

    # Phase 4: Multi-resolution escalation
    target_resolutions = [720, 900, 1080, 1260]
    for n_tr in target_resolutions:
        if time_left() < 180:
            break
        if n_tr % 2 != 0:
            continue
        ts_tr = n_tr / 2.0
        h_interp = interpolate(best_h, n_tr)
        h_interp = project_to_sum(h_interp, ts_tr)
        c5_interp = compute_c5(h_interp)
        if c5_interp < best_c5:
            best_c5 = c5_interp
            best_h = h_interp.copy()
            best_n = n_tr

        # SLSQP at new resolution (fewer iterations for larger n)
        max_slsqp_iter = max(300, 1200 - n_tr // 2)
        for alpha in [100, 500, 2000]:
            if time_left() < 80:
                break
            h_opt = slsqp_optimize(h_interp, n_tr, ts_tr, alpha, maxiter=max_slsqp_iter)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_tr
            h_interp = h_opt

        # Coord descent
        if time_left() > 80:
            h_cd, c5_cd = momentum_coord_desc(h_interp, ts_tr, max_iters=15000, step_frac=0.08, momentum=0.25, tol_frac=0.01)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()
                best_n = n_tr

        # Subgradient
        if time_left() > 50:
            h_sub, c5_sub = subgradient_method(h_cd, ts_tr, max_iters=2000, init_step=0.003, tol_frac=0.008)
            if c5_sub < best_c5:
                best_c5 = c5_sub
                best_h = h_sub.copy()
                best_n = n_tr

    # Phase 5: Structured restarts at best resolution
    n_best = best_n
    ts_best = n_best / 2.0
    n_trials = 12
    for trial in range(n_trials):
        if time_left() < 70:
            break
        if trial % 3 == 0:
            n_blocks = rng.randint(3, 10)
            amp = 0.03 + 0.07 * (1.0 - trial / float(n_trials))
            h_shake = block_perturb(best_h, rng, n_blocks, amp)
        elif trial % 3 == 1:
            h_shake = smooth_perturb(best_h, rng, freq_low=2, freq_high=10, amp=0.04)
        else:
            h_shake = best_h + rng.uniform(-0.03, 0.03, n_best)
        h_shake = project_to_sum(h_shake, ts_best)

        h_cd, c5_cd = momentum_coord_desc(h_shake, ts_best, max_iters=6000, step_frac=0.1, momentum=0.3, tol_frac=0.02)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

        if time_left() < 40:
            break
        h_opt = slsqp_optimize(h_cd, n_best, ts_best, 5000, maxiter=500)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 6: Binary snap exploration
    if time_left() > 80:
        h_bin = binary_snap(best_h, ts_best, threshold=0.5)
        h_cd, c5_cd = momentum_coord_desc(h_bin, ts_best, max_iters=8000, step_frac=0.1, momentum=0.2, tol_frac=0.02)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
        if time_left() > 40:
            h_opt = slsqp_optimize(h_cd, n_best, ts_best, 3000, maxiter=600)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Phase 7: Final high-alpha SLSQP polishing
    for alpha in [10000, 50000]:
        if time_left() < 30:
            break
        h_pol = slsqp_optimize(best_h, n_best, ts_best, alpha, maxiter=2000)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Phase 8: Ultra-fine coordinate descent
    if time_left() > 40:
        h_final, c5_final = momentum_coord_desc(best_h, ts_best, max_iters=40000, step_frac=0.03, momentum=0.1, tol_frac=0.005)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Phase 9: Final subgradient pass
    if time_left() > 30:
        h_sub, c5_sub = subgradient_method(best_h, ts_best, max_iters=5000, init_step=0.002, tol_frac=0.005)
        if c5_sub < best_c5:
            best_c5 = c5_sub
            best_h = h_sub.copy()

    # Final projection
    best_h = project_to_sum(best_h, n_best / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return (best_h, best_c5, n_best)
