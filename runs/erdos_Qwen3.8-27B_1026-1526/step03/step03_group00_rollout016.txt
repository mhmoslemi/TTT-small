    top_idx = sorted_idx[:top_k]
    shifted = alpha * (corr[top_idx] - m_val)
    shifted -= shifted.max()
    exp_vals = np.exp(shifted)
    obj = m_val + (shifted.max() + np.log(exp_vals.sum())) / alpha
    return obj

def smooth_max_grad(h, alpha, top_k):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    sorted_idx = np.argsort(corr)[::-1]
    top_idx = sorted_idx[:top_k]
    shifted = alpha * (corr[top_idx] - m_val)
    shifted -= shifted.max()
    exp_vals = np.exp(shifted)
    w = exp_vals / exp_vals.sum()
    g = np.zeros(n)
    for idx, wi in zip(top_idx, w):
        lag = idx - (n - 1)
        g += wi * _lag_gradient(h, lag)
    return g

def slsqp_local(h0, n, target_sum, alpha, top_k, maxiter):
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    bounds = [(0.0, 1.0)] * n
    
    def f(x):
        return smooth_max_obj(x, alpha, top_k)
    def g(x):
        return smooth_max_grad(x, alpha, top_k)
    
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def annealed_coord(h, target_sum, rng, max_iters, init_temp, cool_rate,
                   min_temp, step_frac, top_k):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    best_c5 = c5
    best_h = h.copy()
    temp = init_temp
    stale = 0
    
    for it in range(max_iters):
        corr = compute_corr(h)
        if top_k == 1:
            m_star = int(np.argmax(corr))
            lag = m_star - (n - 1)
            g = _lag_gradient(h, lag)
        else:
            sorted_idx = np.argsort(corr)[::-1][:top_k]
            m_val = corr[sorted_idx[0]]
            shifted = 25.0 * (corr[sorted_idx] - m_val)
            shifted -= shifted.max()
            ev = np.exp(shifted)
            w = ev / ev.sum()
            g = np.zeros(n)
            for idx, wi in zip(sorted_idx, w):
                lag = idx - (n - 1)
                g += wi * _lag_gradient(h, lag)
        
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        
        g_inc = np.where(can_inc, g, np.inf)
        g_dec = np.where(can_dec, g, -np.inf)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        
        if i_inc == i_dec:
            break
        
        if g[i_inc] >= g[i_dec] - 1e-12:
            if temp > min_temp and rng.random() < 0.15:
                candidates_inc = np.where(can_inc)[0]
                candidates_dec = np.where(can_dec)[0]
                i_inc = int(rng.choice(candidates_inc))
                i_dec = int(rng.choice(candidates_dec))
                if g[i_inc] < g[i_dec]:
                    pass
                else:
                    stale += 1
                    if stale > 80:
                        break
                    temp *= cool_rate
                    continue
            else:
                stale += 1
                if stale > 80:
                    break
                temp *= cool_rate
                continue
        
        max_move = min(1.0 - h[i_inc], h[i_dec])
        move = step_frac * max_move
        if move < 1e-10:
            break
        
        h[i_inc] += move
        h[i_dec] -= move
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
        c5_new = compute_c5(h)
        
        delta = c5_new - c5
        if delta < 0 or (temp > min_temp and rng.random() < np.exp(-max(0, delta) / (temp + 1e-12))):
            c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
            stale = 0
        else:
            h[i_inc] -= move
            h[i_dec] += move
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, target_sum)
        
        temp *= cool_rate
    
    return best_h, best_c5

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h, period=2.0)

def fourier_perturb(h, rng, n_freqs, max_amp):
    n = len(h)
    x = np.linspace(0, 2.0, n, endpoint=False)
    pert = np.zeros(n)
    for _ in range(n_freqs):
        freq = rng.randint(1, max(2, n // 4))
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.01, max_amp)
        pert += amp * np.cos(2 * np.pi * freq * x / 2.0 + phase)
    return pert

def block_subgradient(h, target_sum, n_blocks, step_size, max_sweeps):
    n = len(h)
    h = h.copy()
    block_size = n // n_blocks
    c5 = compute_c5(h)
    
    for sweep in range(max_sweeps):
        improved = False
        for b in range(n_blocks):
            start = b * block_size
            end = start + block_size
            if end >= n:
                break
            idx = np.arange(start, end)
            h_block = h[idx].copy()
            h_rest_sum = np.sum(h) - np.sum(h_block)
            target_block_sum = target_sum - h_rest_sum
            
            corr = compute_corr(h)
            m_lag = int(np.argmax(corr)) - (n - 1)
            g_full = _lag_gradient(h, m_lag)
            g_block = g_full[idx]
            
            step = step_size * np.clip(-g_block, 0, None) + step_size * np.clip(g_block, 0, None)
            h_trial = h_block + 0.5 * (np.clip(-g_block, 0, None) - np.clip(g_block, 0, None)) * step_size
            h_trial = np.clip(h_trial, 0.0, 1.0)
            h_trial = project_to_sum(h_trial, target_block_sum)
            
            h_new = h.copy()
            h_new[idx] = h_trial
            c5_new = compute_c5(h_new)
            if c5_new < c5 - 1e-14:
                h = h_new
                c5 = c5_new
                improved = True
        if not improved:
            break
    
    return h, c5

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    
    def time_left():
        return budget_s - (time.time() - t_start)
    
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass
    
    if best_h is None:
        best_n = 540
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)
    
    n_cur = best_n
    ts_cur = n_cur / 2.0
    
    # Phase 1: SLSQP with various parameters on current resolution
    for alpha in [50, 200, 800, 3000, 10000]:
        if time_left() < 40:
            break
        top_k = min(8, 2 * n_cur)
        h_new = slsqp_local(best_h, n_cur, ts_cur, alpha, top_k, maxiter=800)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    
    # Phase 2: Annealed coordinate descent
    if time_left() > 120:
        h_ann, c5_ann = annealed_coord(best_h, ts_cur, rng, max_iters=8000,
                                       init_temp=0.05, cool_rate=0.9997,
                                       min_temp=0.0, step_frac=0.12, top_k=1)
        if c5_ann < best_c5:
            best_c5 = c5_ann
            best_h = h_ann.copy()
    
    if time_left() > 100:
        h_ann2, c5_ann2 = annealed_coord(best_h, ts_cur, rng, max_iters=6000,
                                         init_temp=0.03, cool_rate=0.9998,
                                         min_temp=0.0, step_frac=0.08, top_k=2)
        if c5_ann2 < best_c5:
            best_c5 = c5_ann2
            best_h = h_ann2.copy()
    
    # Phase 3: SLSQP with higher alpha for polishing
    for alpha in [5000, 20000, 50000]:
        if time_left() < 30:
            break
        top_k = min(5, 2 * n_cur)
        h_new = slsqp_local(best_h, n_cur, ts_cur, alpha, top_k, maxiter=1000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    
    # Phase 4: Upscale to higher resolution
    target_res = [720, 900, 1080]
    for n_tr in target_res:
        if time_left() < 150:
            break
        if n_tr <= best_n:
            continue
        ts_tr = n_tr / 2.0
        h_up = interp_to_n(best_h, n_tr)
        h_up = project_to_sum(h_up, ts_tr)
        c5_up = compute_c5(h_up)
        
        for alpha in [50, 300, 1500, 5000]:
            if time_left() < 40:
                break
            top_k = min(8, 2 * n_tr)
            h_opt = slsqp_local(h_up, n_tr, ts_tr, alpha, top_k, maxiter=600)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n_tr
                n_cur = n_tr
                ts_cur = ts_tr
            h_up = h_opt
        
        if time_left() > 60 and n_tr == best_n:
            h_ann, c5_ann = annealed_coord(best_h, ts_cur, rng, max_iters=5000,
                                           init_temp=0.03, cool_rate=0.9997,
                                           min_temp=0.0, step_frac=0.1, top_k=1)
            if c5_ann < best_c5:
                best_c5 = c5_ann
                best_h = h_ann.copy()
    
    # Phase 5: Diverse restarts
    n_restart = best_n
    ts_restart = n_restart / 2.0
    n_restarts = 8
    for r in range(n_restarts):
        if time_left() < 50:
            break
        if r < 3:
            pert = fourier_perturb(best_h, rng, n_freqs=4, max_amp=0.04)
            h_shake = best_h + pert
        elif r < 5:
            noise = rng.uniform(-0.03, 0.03, n_restart)
            h_shake = best_h + noise
        else:
            mask = rng.random(n_restart) < 0.3
            h_shake = best_h.copy()
            h_shake[mask] = rng.uniform(0, 1, mask.sum())
        
        h_shake = project_to_sum(h_shake, ts_restart)
        
        h_ann, c5_ann = annealed_coord(h_shake, ts_restart, rng, max_iters=3000,
                                       init_temp=0.04, cool_rate=0.9996,
                                       min_temp=0.0, step_frac=0.1, top_k=1)
        if c5_ann < best_c5:
            best_c5 = c5_ann
            best_h = h_ann.copy()
        
        if time_left() > 30:
            h_opt = slsqp_local(best_h, n_restart, ts_restart, 2000, 5, maxiter=500)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    
    # Phase 6: Final fine-tuning
    if time_left() > 60:
        h_fin, c5_fin = annealed_coord(best_h, ts_restart, rng, max_iters=10000,
                                       init_temp=0.01, cool_rate=0.9999,
                                       min_temp=0.0, step_frac=0.05, top_k=1)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()
    
    if time_left() > 40:
        for alpha in [20000, 80000]:
            if time_left() < 20:
                break
            h_pol = slsqp_local(best_h, best_n, ts_restart, alpha, 3, maxiter=1500)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()
    
    if time_left() > 30:
        h_fin2, c5_fin2 = annealed_coord(best_h, ts_restart, rng, max_iters=12000,
                                         init_temp=0.005, cool_rate=0.99995,
                                         min_temp=0.0, step_frac=0.04, top_k=1)
        if c5_fin2 < best_c5:
            best_c5 = c5_fin2
            best_h = h_fin2.copy()
    
    if time_left() > 20:
        h_fin3, c5_fin3 = annealed_coord(best_h, ts_restart, rng, max_iters=8000,
                                         init_temp=0.008, cool_rate=0.9999,
                                         min_temp=0.0, step_frac=0.06, top_k=2)
        if c5_fin3 < best_c5:
            best_c5 = c5_fin3
            best_h = h_fin3.copy()
    
    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
