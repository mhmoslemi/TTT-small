```python
def frank_wolfe_step(h, grad, n, target_sum):
    """One Frank-Wolfe step: find the 0-1 vertex in the feasible direction."""
    # min grad . v subject to 0 <= v <= 1, sum(v) = target_sum
    # Solution: v[i] = 1 for the target_sum smallest grad[i], 0 otherwise
    idx = np.argpartition(grad, int(target_sum))[:int(target_sum)]
    v = np.zeros(n)
    v[idx] = 1.0
    return v
```

```python
def smooth_max_grad(h, alpha, top_k=None):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    
    if top_k is not None and top_k < len(corr):
        idx = np.argpartition(corr, -top_k)[-top_k:]
    else:
        idx = np.arange(len(corr))
    
    corr_k = corr[idx]
    m = np.max(corr_k)
    shifted = alpha * (corr_k - m)
    shifted -= np.max(shifted)  # numerical stability
    exp_v = np.exp(shifted)
    w = exp_v / np.sum(exp_v)
    obj = m + (np.max(alpha * (corr_k - m)) + np.log(np.sum(exp_v))) / alpha
    
    # Compute gradient
    g = np.zeros(n)
    # Filter significant weights
    sig_mask = w > 1e-8
    sig_idx = idx[sig_mask]
    sig_w = w[sig_mask]
    
    for j, w_j in zip(sig_idx, sig_w):
        s = j - (n - 1)
        if s >= 0:
            # forward: (1-h[k-s]) for k in [s, n)
            g[s:] += w_j * (1.0 - h[:n-s])
            # backward: -h[k+s] for k in [0, n-s)
            g[:n-s] -= w_j * h[s:]
        else:
            t = -s
            # forward: (1-h[k+t]) for k in [0, n-t)
            g[:n-t] += w_j * (1.0 - h[t:])
            # backward: -h[k-t] for k in [t, n)
            g[t:] -= w_j * h[:n-t]
    
    return obj, g
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        n_mask = np.sum(mask)
        if n_mask > 0:
            h[mask] += diff / n_mask
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_grad(h, alpha, top_k=None):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    
    if top_k is not None and top_k < len(corr):
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(corr))
    
    corr_k = corr[top_idx]
    m = np.max(corr_k)
    shifted = alpha * (corr_k - m)
    max_shifted = np.max(shifted)
    exp_v = np.exp(shifted - max_shifted)
    w = exp_v / np.sum(exp_v)
    obj = m + (max_shifted + np.log(np.sum(exp_v))) / alpha
    
    g = np.zeros(n)
    sig_mask = w > 1e-8
    if not np.any(sig_mask):
        return obj, g
    
    sig_positions = top_idx[sig_mask]
    sig_w = w[sig_mask]
    
    for j, wj in zip(sig_positions, sig_w):
        s = j - (n - 1)
        if s >= 0:
            g[s:] += wj * (1.0 - h[:n-s])
            g[:n-s] -= wj * h[s:]
        else:
            t = -s
            g[:n-t] += wj * (1.0 - h[t:])
            g[t:] -= wj * h[:n-t]
    
    return obj, g

def lbfgs_step(h0, n, ts, alpha, maxiter=500, top_k=30, penalty=100.0):
    def obj_grad(x):
        obj, grad = smooth_max_obj_grad(x, alpha, top_k)
        sd = np.sum(x) - ts
        obj += penalty * sd * sd
        grad = grad + 2.0 * penalty * sd
        return obj, grad
    
    res = minimize(obj_grad, h0.copy(), method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14,
                            'maxfun': maxiter * 20})
    return project_to_sum(res.x, ts)

def fw_step(h, grad, n, ts):
    idx = np.argpartition(grad, int(ts))[:int(ts)]
    v = np.zeros(n)
    v[idx] = 1.0
    return v

def fw_optimize(h0, n, ts, alpha, n_iter=10000, top_k=20):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for t in range(n_iter):
        obj, grad = smooth_max_obj_grad(h, alpha, top_k)
        v = fw_step(h, grad, n, ts)
        gamma = 2.0 / (t + 2.0)
        h = (1.0 - gamma) * h + gamma * v
        # h should still have sum = ts since both h and v do
        if t % 500 == 0:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h

def stochastic_cd(h, n, ts, rng, n_steps=30000, delta_range=(0.01, 0.2)):
    h = h.copy()
    c5_cur = compute_c5(h)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        n_mask = np.sum(mask)
        if n_mask > 0:
            h[mask] += diff / n_mask
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_grad(h, alpha, top_k=None):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    if top_k is not None and top_k < len(corr):
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(corr))
    corr_k = corr[top_idx]
    m = np.max(corr_k)
    shifted = alpha * (corr_k - m)
    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    w = exp_v / np.sum(exp_v)
    obj = m + (max_sh + np.log(np.sum(exp_v))) / alpha
    g = np.zeros(n)
    sig = w > 1e-10
    if not np.any(sig):
        return obj, g
    pos = top_idx[sig]
    ws = w[sig]
    for j, wj in zip(pos, ws):
        s = j - (n - 1)
        if s >= 0:
            g[s:] += wj * (1.0 - h[:n - s])
            g[:n - s] -= wj * h[s:]
        else:
            t = -s
            g[:n - t] += wj * (1.0 - h[t:])
            g[t:] -= wj * h[:n - t]
    return obj, g

def lbfgs_optimize(h0, n, ts, alpha, maxiter=400, top_k=30, penalty=100.0):
    def obj_grad(x):
        obj, grad = smooth_max_obj_grad(x, alpha, top_k)
        sd = np.sum(x) - ts
        obj += penalty * sd * sd
        grad = grad + 2.0 * penalty * sd
        return obj, grad
    res = minimize(obj_grad, h0.copy(), method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14,
                            'maxfun': maxiter * 30})
    return project_to_sum(res.x, ts)

    x = np.linspace(0, 2, n, endpoint=False)
```
