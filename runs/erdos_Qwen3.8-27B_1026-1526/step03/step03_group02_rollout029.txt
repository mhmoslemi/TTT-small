def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

    try:

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(5000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def make_binary(h, n, target_sum, rng):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones == target_sum:
        return h_bin
    if n_ones > target_sum:
        excess = n_ones - target_sum
        ones_idx = np.where(h_bin == 1.0)[0]
        order = np.argsort(h[ones_idx])[:excess]
        h_bin[ones_idx[order]] = 0.0
    else:
        deficit = target_sum - n_ones
        zeros_idx = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h[zeros_idx])[:deficit]
        h_bin[zeros_idx[order]] = 1.0
    return h_bin

        j = zeros[rng.randint(len(zeros))]

        delta = c5_new - c5_curr

            zeros = np.where(h < 0.5)[0]

                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1

    return best_h, best_c5

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def slsqp_optimize(h0, n, target_sum, alpha, maxiter=800):
    def f(h):
        return smooth_max_and_grad(h, alpha)[0]
    def g(h):
        return smooth_max_and_grad(h, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def projected_gradient(h0, n, target_sum, alpha, lr, max_iter=2000):
    """Projected gradient descent on smooth max"""
    h = h0.copy()
    for _ in range(max_iter):
        obj, grad = smooth_max_and_grad(h, alpha)
        h_new = h - lr * grad
        h_new = project_to_sum(h_new, target_sum)
        obj_new = smooth_max_and_grad(h_new, alpha)[0]
        if obj_new < obj:
            h = h_new
        else:
            lr *= 0.9
        if lr < 1e-10:
            break
    return project_to_sum(h, target_sum)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(5000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        if np.any(mask):
            h[mask] += diff / float(np.sum(mask))
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)

def make_binary(h, n, target_sum, rng):

        return h_bin
    if n_ones > target_sum:
        excess = n_ones - target_sum
        ones_idx = np.where(h_bin == 1.0)[0]
        order = np.argsort(h[ones_idx])[:excess]
        h_bin[ones_idx[order]] = 0.0
    else:

        zeros_idx = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h[zeros_idx])[:deficit]

                h[i:i+blen] = np.concatenate((seg[:rot], seg[rot:]))
        if no_improve > 20000:
            h = best_h.copy()
            c5_curr = best_c5
            T = min(T * 2.5, 0.016)
            no_improve = 0
        T = max(T * 0.99999, 0.0012)
    return best_h, best_c5

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = np.clip(alpha * (corr - m_val), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def slsqp_optimize(h0, n, target_sum, alpha, maxiter=800):
    obj_cache = {}
    def f(h):
        key = tuple(np.round(h, 10))
        if key in obj_cache:
            return obj_cache[key]
        val = smooth_max_and_grad(h, alpha)[0]
        obj_cache.clear()
        obj_cache[key] = val
        return val
    def g(h):
        return smooth_max_and_grad(h, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def slsqp_no_cache(h0, n, target_sum, alpha, maxiter=800):
    def f(h):
        return smooth_max_and_grad(h, alpha)[0]
    def g(h):
        return smooth_max_and_grad(h, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def interpolated_h(h_src, n_src, n_dst):
    x_src = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    return np.interp(x_dst, x_src, h_src)

def local_improve(h, n, target_sum, max_rounds=50):
    h = h.copy()
    c5_curr = compute_c5(h)
    ones_set = set(np.where(h > 0.5)[0].tolist())
    zeros_set = set(np.where(h < 0.5)[0].tolist())
    for _ in range(max_rounds):
        improved = False
        ones_list = sorted(ones_set)
        zeros_list = sorted(zeros_set)
        rng_local = np.random.RandomState(0)
        trial_zeros = zeros_list if len(zeros_list) <= 30 else zeros_list[rng_local.choice(len(zeros_list), 30, replace=False)]
        for i in ones_list:
            for j in trial_zeros:
                h[i], h[j] = h[j], h[i]
                c5_new = compute_c5(h)
                if c5_new < c5_curr - 1e-14:
                    c5_curr = c5_new
                    ones_set.remove(i)
                    ones_set.add(j)
                    zeros_set.remove(j)
                    zeros_set.add(i)
                    improved = True
                else:
                    h[i], h[j] = h[j], h[i]
        if not improved:
            break
    return project_to_sum(h, target_sum), c5_curr

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - t_start)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n_main = best_n
    ts_main = n_main / 2.0

    # Phase 1: SA from parent (binary)
    h_bin = make_binary(best_h, n_main, int(ts_main), rng)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    sa_end1 = t_start + 0.22 * budget_s
    h_sa1, c5_sa1 = sa_swap(h_bin, n_main, int(ts_main), sa_end1, rng)
    if c5_sa1 < best_c5:
        best_c5 = c5_sa1
        best_h = h_sa1.copy()

    # Phase 2: SA with blocks from a different start
    h_bin2 = make_binary(best_h, n_main, int(ts_main), rng)
    sa_end2 = t_start + 0.35 * budget_s
    h_sa2, c5_sa2 = sa_blocks(h_bin2, n_main, int(ts_main), sa_end2, rng)
    if c5_sa2 < best_c5:
        best_c5 = c5_sa2
        best_h = h_sa2.copy()

    # Phase 3: SLSQP sequential polish
    for alpha in [8, 30, 100, 400, 1500, 6000]:
        if remaining() < 60:
            break
        h_pol = slsqp_no_cache(best_h, n_main, ts_main, alpha, maxiter=2000)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Phase 4: Random restart SA (different random seeds)
    for restart_seed in [12345, 67890, 11111]:
        if remaining() < 80:
            break
        h_rand = (rng.randint(0, 2, n_main) * 1.0)
        h_rand[binary_select(h_rand, int(ts_main), rng)] = 1.0
        h_rand[1 - binary_select(h_rand, int(ts_main), rng) == 0] = 0.0
        h_rb = make_binary(0.5 + 0.3 * (h_rand - 0.5), n_main, int(ts_main), rng)
        sa_e = t_start + budget_s - max(remaining() * 0.4, 100)
        h_sr, c5_sr = sa_swap(h_rb, n_main, int(ts_main), sa_e, rng, T_init=0.01)
        if c5_sr < best_c5:
            best_c5 = c5_sr
            best_h = h_sr.copy()
        h_rb2 = make_binary(best_h + rng.uniform(-0.05, 0.05, n_main), n_main, int(ts_main), rng)
        h_sr2, c5_sr2 = sa_blocks(h_rb2, n_main, int(ts_main), sa_e, rng)
        if c5_sr2 < best_c5:
            best_c5 = c5_sr2
            best_h = h_sr2.copy()

    # Phase 5: SLSQP again after SA
    for alpha in [3000, 15000, 60000]:
        if remaining() < 40:
            break
        h_pol2 = slsqp_no_cache(best_h, n_main, ts_main, alpha, maxiter=3000)
        c5_pol2 = compute_c5(h_pol2)
        if c5_pol2 < best_c5:
            best_c5 = c5_pol2
            best_h = h_pol2.copy()

    # Phase 6: Higher resolution n=480
    n_h = 480
    if remaining() > 150:
        ts_h = n_h / 2.0
        h_interp = interpolated_h(best_h, best_n, n_h)
        h_interp = project_to_sum(h_interp, ts_h)
        h_h_bin = make_binary(h_interp, n_h, int(ts_h), rng)
        sa_end_h = t_start + budget_s - 80
        h_h_sa, c5_h_sa = sa_swap(h_h_bin, n_h, int(ts_h), sa_end_h, rng,
                                  T_init=0.005, T_min=0.0008, cool=0.99998)
        h_h = h_h_sa.copy()
        c5_h = c5_h_sa
        if c5_h < best_c5:
            best_c5 = c5_h
            best_h = h_h.copy()
            best_n = n_h
        for alpha in [50, 500, 5000]:
            if remaining() < 30:
                break
            h_h = slsqp_no_cache(h_h, n_h, ts_h, alpha, maxiter=800)
            c5_h = compute_c5(h_h)
            if c5_h < best_c5:
                best_c5 = c5_h
                best_h = h_h.copy()
                best_n = n_h

    # Phase 7: Try n=320 if time allows
    if remaining() > 100 and best_n < 320:
        n_320 = 320
        ts_320 = n_320 / 2.0
        h_i320 = interpolated_h(best_h, best_n, n_320)
        h_i320 = project_to_sum(h_i320, ts_320)
        h_b320 = make_binary(h_i320, n_320, int(ts_320), rng)
        sa_e320 = t_start + budget_s - 60
        h_320, c5_320 = sa_swap(h_b320, n_320, int(ts_320), sa_e320, rng)
        for alpha in [100, 1000]:
            if remaining() < 20:
                break
            h_320 = slsqp_no_cache(h_320, n_320, ts_320, alpha, maxiter=600)
            c5_320 = compute_c5(h_320)
        if c5_320 < best_c5:
            best_c5 = c5_320
            best_h = h_320.copy()
            best_n = n_320

    # Final polish
    if remaining() > 10:
        alpha_final = 100000
        h_fin = slsqp_no_cache(best_h, best_n, best_n / 2.0, alpha_final, maxiter=2000)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)

def binary_select(h, target_ones, rng):
    n = len(h)
    idx = rng.permutation(n)
    return idx[:target_ones]
