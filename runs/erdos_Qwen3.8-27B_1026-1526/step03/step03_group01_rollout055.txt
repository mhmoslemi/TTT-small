```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    w = exp_v / np.sum(exp_v)
    obj = m_val + (max_sh + np.log(np.sum(exp_v))) / alpha
    
    nj = np.arange(n)[:, None]
    nm = np.arange(2 * n - 1)[None, :]
    fi = nj - nm + n - 1
    si = nj + nm - n + 1
    vf = (fi >= 0) & (fi < n)
    vs = (si >= 0) & (si < n)
    fc = np.clip(fi, 0, n - 1)
    sc = np.clip(si, 0, n - 1)
    grad = np.sum(vf * (1.0 - h[fc]) * w[None, :] - vs * h[sc] * w[None, :], axis=1)
    return obj, grad

def slsqp_opt(h0, n, ts, alpha, maxiter=800):
    def f(x):
        return smooth_max_obj_grad(x, alpha)[0]
    def g(x):
        return smooth_max_obj_grad(x, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ts, 'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(f, h0.copy(), method='SLSQP', jac=g, bounds=bnds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-13})
        return project_to_sum(r.x, ts)
    except Exception:
        return project_to_sum(h0, ts)

def nesterov_pgd(h0, n, ts, alpha, n_iter=3000, lr_init=0.01):
    h = h0.copy()
    h_prev = h.copy()
    lr = lr_init
    best_obj, _ = smooth_max_obj_grad(h, alpha)
    best_h = h.copy()
    stall = 0
    
    for t in range(n_iter):
        momentum = t / (t + 3.0)
        y = h + momentum * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        
        obj_y, grad_y = smooth_max_obj_grad(y, alpha)
        
        h_prev = h.copy()
        h_new = project_to_sum(y - lr * grad_y, ts)
        
        obj_new, _ = smooth_max_obj_grad(h_new, alpha)
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1
        
        h = h_new
        
        # Adaptive learning rate

            lr *= 0.5
        elif stall %  100 == 0 and stall > 0:
            lr *= 0.8
        
        if lr < 1e-10:
            break
        if stall > 500:
            break
    
    return best_h

def block_move(h, n, ts, rng, block_sizes=(1,2,3,4,5,8,10,16,20)):
    """Try moving a contiguous block of the function."""
    h_new = h.copy()

        return h
    # Choose random start position
    start = rng.randint(0, n - bs)
    # Choose shift amount

        return h
    # Extract block
    block = h_new[start:start+bs].copy()
    # Remove block (set to 0 or 1 depending on "background")
    # Actually, let's just shift the block
    new_start = start + shift

    h_new[start:start+bs] = 0.0  # This is too aggressive
    # Place at new position
    h_new[new_start:new_start+bs] = block
    
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_to_sum(h_new, ts)
    return h_new

def mass_transfer_cd(h, n, ts, rng, n_steps=50000, delta_range=(0.01, 0.3)):

            else:

                h[j] -= delta

                h = project_to_sum(h, ts)
        temp = max(1e-6, 0.3 * np.exp(-step / 8000.0))
```

Let me think about a completely different approach: **alternating direction method of multipliers (ADMM)** or **proximal gradient**.

For the problem:
minimize max_k f_k(h)
subject to: h ∈ C (convex set: 0 ≤ h ≤ 1, sum(h) = n/2)

We can write this as:
minimize t
subject to: f_k(h) ≤ t for all k
h ∈ C

The Lagrangian is:
L(h, t, λ) = t + sum_k λ_k * (f_k(h) - t)

where λ_k ≥ 0 and sum λ_k = 1 (normalization).

The dual function is:
g(λ) = min_{h ∈ C} [sum_k λ_k * f_k(h)]

Since sum_k λ_k = 1 and λ_k ≥ 0, this is minimizing a convex combination of the f_k's over the convex set C.

If each f_k were convex in h, we could solve the inner problem efficiently. But f_k(h) = sum_i h[i]*(1-h[i+k])*dx is not convex (it's a difference of products).

However, we can use the fact that f_k(h) = sum_i h[i]*dx - sum_i h[i]*h[i+k]*dx. The first term is linear, and the second is a sum of products (which is not convex but is "almost" convex in some sense).

This doesn't lead to a practical algorithm.

Let me try yet another approach: **subgradient method with cutting planes**.

At each iteration:
1. Evaluate all f_k(h)
2. Find k* = argmax_k f_k(h)
3. Take a step in the direction of -∇f_{k*}(h) (subgradient of the max)
4. Project to feasible set

This is simpler than the smooth max approach and might work well in practice.

```python
def subgradient_opt(h0, n, ts, n_iter=10000, lr_init=0.01):
    h = h0.copy()
    c5_cur = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_cur
    
    for t in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_star = np.argmax(corr)
        
        # Subgradient of f_{k*} at h
        # f_k(h) = sum_i h[i+k]*h[i] ... wait
        # f_k(h) = sum_n h[n+k] * (1-h[n])
        # d f_k / d h[i] = (1-h[i-k]) * 1_{valid} - h[i+k] * 1_{valid}
        
        n_pts = n
        s = k_star - (n_pts - 1)
        # d f_k / d h[i] = (1-h[i-s]) * 1_{0<=i-s<n} - h[i+s] * 1_{0<=i+s<n}
        
        i_vals = np.arange(n_pts)
        idx1 = i_vals - s
        idx2 = i_vals + s
        v1 = (idx1 >= 0) & (idx1 < n_pts)
        v2 = (idx2 >= 0) & (idx2 < n_pts)
        
        c1 = np.clip(idx1, 0, n_pts-1)
        c2 = np.clip(idx2, 0, n_pts-1)
        
        grad = (1.0 - h[c1]) * v1 - h[c2] * v2
        
        # Step
        lr = lr_init / (1.0 + t * 0.01)
        h_new = project_to_sum(h - lr * grad, ts)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        h = h_new
        
        if t % 10000 == 0:
            print(t, c5_new)
    
    return best_h, best_c5
```

```python
    # Get top m shifts
    top_idx = np.argsort(corr)[-m:]
    top_vals = corr[top_idx]
    max_val = top_vals[-1]
    
    # Weights: exponential of (val - max) / temperature
    temp = 0.01
    w = np.exp((top_vals - max_val) / temp)
    w /= np.sum(w)
    
    # Weighted gradient
    grad = np.zeros(n)
    n_pts = n
    for idx, weight in zip(top_idx, w):
        s = idx - (n_pts - 1)
        i_vals = np.arange(n_pts)
        idx1 = i_vals - s
        idx2 = i_vals + s
        v1 = (idx1 >= 0) & (idx1 < n_pts)
        v2 = (idx2 >= 0) & (idx2 < n_pts)
        c1 = np.clip(idx1, 0, n_pts-1)
        c2 = np.clip(idx2, 0, n_pts-1)
        grad += weight * ((1.0 - h[c1]) * v1 - h[c2] * v2)
    
    return grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    w = exp_v / np.sum(exp_v)
    obj = m_val + (max_sh + np.log(np.sum(exp_v))) / alpha
    nj = np.arange(n)[:, None]
    nm = np.arange(2 * n - 1)[None, :]
    fi = nj - nm + n - 1
    si = nj + nm - n + 1
    vf = (fi >= 0) & (fi < n)
    vs = (si >= 0) & (si < n)
    fc = np.clip(fi, 0, n - 1)
    sc = np.clip(si, 0, n - 1)
    grad = np.sum(vf * (1.0 - h[fc]) * w[None, :] - vs * h[sc] * w[None, :], axis=1)
    return obj, grad

def single_shift_grad(h, k_full):
    n = len(h)
    s = k_full - (n - 1)
    i_vals = np.arange(n)
    idx1 = i_vals - s
    idx2 = i_vals + s
    v1 = (idx1 >= 0) & (idx1 < n)
    v2 = (idx2 >= 0) & (idx2 < n)
    c1 = np.clip(idx1, 0, n - 1)
    c2 = np.clip(idx2, 0, n - 1)
    return (1.0 - h[c1]) * v1 - h[c2] * v2

def top_m_subgrad(h, m=8, temp=0.02):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[-m:]
    top_vals = corr[top_idx]
    max_val = top_vals[-1]
    w = np.exp((top_vals - max_val) / temp)
    w = w / np.sum(w)
    grad = np.zeros(n)
    for idx, weight in zip(top_idx, w):
        grad += weight * single_shift_grad(h, idx)
    return grad

def slsqp_opt(h0, n, ts, alpha, maxiter=800):
    def f_obj(x):
        return smooth_max_obj_grad(x, alpha)[0]
    def g_obj(x):
        return smooth_max_obj_grad(x, alpha)[1]
    def cons_fun(x):
        return np.sum(x) - ts
    def cons_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(f_obj, h0.copy(), method='SLSQP', jac=g_obj, bounds=bnds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-13})
        return project_to_sum(r.x, ts)
    except Exception:
        return project_to_sum(h0, ts)

def nesterov_pgd(h0, n, ts, alpha, n_iter=3000, lr_init=0.005):
    h = h0.copy()
    h_prev = h.copy()
    lr = lr_init
    best_obj, _ = smooth_max_obj_grad(h, alpha)
    best_h = h.copy()
    stall = 0
    for t in range(n_iter):
        mu = t / (t + 3.0)
        y = h + mu * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        obj_y, grad_y = smooth_max_obj_grad(y, alpha)
        h_prev = h.copy()
        h_new = project_to_sum(y - lr * grad_y, ts)
        obj_new, _ = smooth_max_obj_grad(h_new, alpha)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1
        h = h_new
        if obj_new > obj_y + 0.05:
            lr *= 0.5
        elif t > 0 and t % 500 == 0 and stall > 100:
            lr *= 0.7
        if lr < 1e-12:
            break
        if stall > 800:
            break
    return best_h

def top_m_pgd(h0, n, ts, n_iter=50000, lr_init=0.02, m=8, temp=0.02):
    h = h0.copy()
    c5_cur = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_cur
    best_stall = 0
    for t in range(n_iter):
        grad = top_m_subgrad(h, m=m, temp=temp)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-10:
            break
        lr = lr_init / (1.0 + 0.005 * t)
        h_new = project_to_sum(h - lr * grad, ts)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            best_stall = 0
        else:
            best_stall += 1
        h = h_new
        if best_stall > 3000:
            break
    return best_h, best_c5

def mass_transfer_sa(h, n, ts, rng, n_steps=50000, delta_range=(0.01, 0.25)):
    h = h.copy()
    c5_cur = compute_c5(h)
    temp = 0.2
    for step in range(n_steps):
        ones_idx = np.where(h > 0.03)[0]
        zeros_idx = np.where(h < 0.97)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        if i == j:
            continue
        delta = rng.uniform(delta_range[0], delta_range[1])
        if h[i] < delta:
            delta = max(h[i] * 0.5, 1e-5)
        if 1.0 - h[j] < delta:
            delta = max((1.0 - h[j]) * 0.5, 1e-5)
        if delta < 1e-5:
            continue
        h[i] -= delta
        h[j] += delta
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, ts)
        c5_new = compute_c5(h)
        if c5_new <= c5_cur:
            c5_cur = c5_new
        else:
            accept_prob = np.exp(-(c5_new - c5_cur) / temp)
            if rng.random() < accept_prob:
                c5_cur = c5_new
            else:
                h[i] += delta
                h[j] -= delta
                h = np.clip(h, 0.0, 1.0)
                h = project_to_sum(h, ts)
        temp = max(1e-7, 0.2 * np.exp(-step / 10000.0))
    return h, c5_cur

def freq_perturb(h, rng, n_harm, amp):
    n = len(h)
    mean_h = np.mean(h)
    H = np.fft.rfft(h - mean_h)
    upper = min(n_harm, n // 2 - 1)
    for k in range(1, upper + 1):
        H[k] += amp * (rng.randn() + 1j * rng.randn())
    h_new = np.fft.irfft(H, n=n) + mean_h
    return project_to_sum(h_new, n / 2.0 * 1.0)

def binary_push(h, ts, strength=0.1):
    n = len(h)
    h_new = h.copy()
    for i in range(n):
        if h_new[i] > 0.5:
            h_new[i] = min(1.0, h_new[i] + strength * (1.0 - h_new[i]))
        else:
            h_new[i] = max(0.0, h_new[i] - strength * h_new[i])
    return project_to_sum(h_new, ts)
```
