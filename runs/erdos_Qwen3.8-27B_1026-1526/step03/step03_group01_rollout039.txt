import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_all_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def overlap_grad(h, lag, n):
    """Gradient of overlap at given lag with respect to h."""
    # lag is the shift (can be negative)
    # grad[m] = (1-h[m-lag])*1[0<=m-lag<n] - h[m+lag]*1[0<=m+lag<n]
    m = np.arange(n)
    idx1 = m - lag
    idx2 = m + lag
    valid1 = (idx1 >= 0) & (idx1 < n)
    valid2 = (idx2 >= 0) & (idx2 < n)
    g = np.zeros(n)
    g[valid1] = (1.0 - h[idx1[valid1]])
    g[valid2] = g[valid2] - h[idx2[valid2]]
    return g

def smooth_max_value_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    w = exp_v / np.sum(exp_v)
    obj = (m_val + (max_sh + np.log(np.sum(exp_v))) / alpha) * dx
    
    # Gradient via matrix
    nj = np.arange(n)[:, None]
    nm = np.arange(2 * n - 1)[None, :]
    fi = nj - nm + n - 1
    si = nj + nm - n + 1
    vf = (fi >= 0) & (fi < n)
    vs = (si >= 0) & (si < n)
    fc = np.clip(fi, 0, n - 1)
    sc = np.clip(si, 0, n - 1)
    grad = np.sum(vf * (1.0 - h[fc]) * w[None, :] - vs * h[sc] * w[None, :], axis=1) * dx
    return obj, grad

def fista_opt(h0, n, ts, alpha, n_iter=3000):
    """FISTA on smooth max."""
    h = h0.copy()
    h_prev = h.copy()
    t_k = 1.0
    lr = 1.0  # initial step size
    obj_prev, _ = smooth_max_value_and_grad(h, alpha)
    best_obj = obj_prev
    best_h = h.copy()
    
    for i in range(n_iter):
        _, grad = smooth_max_value_and_grad(h, alpha)
        # Backtracking
        step = lr
        for _ in range(30):
            h_trial = project_to_sum(h - step * grad, ts)
            obj_trial, _ = smooth_max_value_and_grad(h_trial, alpha)
            if obj_trial <= obj_prev - 0.5 * step * np.dot(grad, h - h_trial):
                break
            step *= 0.5
        else:
            h_trial = project_to_sum(h - step * grad, ts)
            obj_trial, _ = smooth_max_value_and_grad(h_trial, alpha)
        
        if obj_trial < best_obj:
            best_obj = obj_trial
            best_h = h_trial.copy()
        
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = h_trial + ((t_k - 1.0) / t_next) * (h_trial - h)
        h = project_to_sum(y, ts)
        h_prev = h.copy()
        t_k = t_next
        lr = step
        obj_prev = smooth_max_value_and_grad(h, alpha)[0]
    
    # Final check
    obj_final, _ = smooth_max_value_and_grad(h, alpha)
    if obj_final < best_obj:
        best_h = h.copy()
    return best_h

def cutting_plane_opt(h0, n, ts, max_iter=200, tol=1e-6):
    """LP-based cutting plane method."""
    h = h0.copy()
    dx = 2.0 / n
    
    for it in range(max_iter):
        overlaps = compute_all_overlaps(h)
        t_cur = np.max(overlaps)
        
        # Find active lags (within tol of max)
        active = np.where(overlaps > t_cur - tol)[0]
        
        # Build LP
        # For each active lag, add a cutting plane
        c = np.zeros(n + 1)
        c[-1] = 1.0  # minimize t
        
        A_ub = []
        b_ub = []
        
        for k in active:
            lag = k - (n - 1)
            a = overlap_grad(h, lag, n)
            O_k = overlaps[k]
            rhs = np.dot(a, h) - O_k
            row = np.zeros(n + 1)
            row[:n] = a
            row[-1] = -1.0
            A_ub.append(row)
            b_ub.append(rhs)
        
        A_ub = np.array(A_ub)
        b_ub = np.array(b_ub)
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = [ts]
        
        bounds = [(0, 1)] * n + [(0, None)]
        
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
            if res.success:
                h_new = res.x[:n]
                t_new = res.x[-1]
                h_new = np.clip(h_new, 0, 1)
                h_new = project_to_sum(h_new, ts)
                c5_new = compute_c5(h_new)
                if c5_new < t_cur:
                    h = h_new
                else:
                    # LP solution doesn't improve; try a small step towards it
                    h = project_to_sum(0.5 * h + 0.5 * h_new, ts)
                    c5_new = compute_c5(h)
            else:
                break
        except Exception:
            break
        
        # Check convergence
        if abs(compute_c5(h) - t_cur) < 1e-10:
            break
    
    return h

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_all_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def overlap_grad_single(h, lag, n):
    m = np.arange(n)
    idx1 = m - lag
    idx2 = m + lag
    valid1 = (idx1 >= 0) & (idx1 < n)
    valid2 = (idx2 >= 0) & (idx2 < n)
    g = np.zeros(n)
    g[valid1] = (1.0 - h[idx1[valid1]])
    g[valid2] = g[valid2] - h[idx2[valid2]]
    return g

def smooth_max_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    s = np.sum(exp_v)
    w = exp_v / s
    obj = (m_val + (max_sh + np.log(s)) / alpha) * dx
    return obj, w, corr

def smooth_max_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    obj, w, corr = smooth_max_obj(h, alpha)
    nj = np.arange(n)[:, None]
    nm = np.arange(2 * n - 1)[None, :]
    fi = nj - nm + n - 1
    si = nj + nm - n + 1
    vf = (fi >= 0) & (fi < n)
    vs = (si >= 0) & (si < n)
    fc = np.clip(fi, 0, n - 1)
    sc = np.clip(si, 0, n - 1)
    grad = np.sum(vf * (1.0 - h[fc]) * w[None, :] - vs * h[sc] * w[None, :], axis=1) * dx
    return obj, grad

def fista_opt(h0, n, ts, alpha, n_iter=2000):
    h = h0.copy()
    h_y = h.copy()
    t_k = 1.0
    lr = 1.0 / (4.0 * alpha / n)
    if lr > 1.0:
        lr = 1.0
    obj_prev, _ = smooth_max_grad(h, alpha)
    best_obj = obj_prev
    best_h = h.copy()
    for i in range(n_iter):
        _, grad = smooth_max_grad(h_y, alpha)
        step = lr
        for _ in range(30):
            h_trial = project_to_sum(h_y - step * grad, ts)
            obj_trial, _ = smooth_max_grad(h_trial, alpha)
            if obj_trial <= obj_prev - 0.4 * step * np.dot(grad, h_y - h_trial):
                break
            step *= 0.5
        if obj_trial < best_obj - 1e-15:
            best_obj = obj_trial
            best_h = h_trial.copy()
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        h = h_trial
        h_y = h + ((t_k - 1.0) / t_next) * (h - h_y + h)
        h_y = h + ((t_k - 1.0) / t_next) * (h_trial - h_y)
        h_y = project_to_sum(h_y, ts)
        t_k = t_next
        lr = step
        obj_prev = smooth_max_grad(h, alpha)[0]
    obj_final, _ = smooth_max_grad(h, alpha)
    if obj_final < best_obj:
        best_h = h.copy()
    return best_h

def slsqp_step(h0, n, ts, alpha, maxiter=500):
    def f_obj(x):
        return smooth_max_grad(x, alpha)[0]
    def g_obj(x):
        return smooth_max_grad(x, alpha)[1]
    def c_fun(x):
        return np.sum(x) - ts
    def c_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(f_obj, h0.copy(), method='SLSQP', jac=g_obj,
                     bounds=bnds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-14})
        return project_to_sum(r.x, ts)
    except Exception:
        return project_to_sum(h0, ts)

def cutting_plane_opt(h0, n, ts, max_iter=100, n_active=10):
    h = h0.copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    for it in range(max_iter):
        overlaps = compute_all_overlaps(h)
        t_cur = np.max(overlaps)
        if t_cur >= best_c5 + 1e-12:
            break
        sorted_idx = np.argsort(overlaps)[::-1]
        active = sorted_idx[:n_active]
        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0
        A_ub_list = []
        b_ub_list = []

            row[-1] = -1.0

        b_ub = np.array(b_ub_list)
        A_eq = np.zeros((1, n + 1))

        b_eq = np.array([ts])

        try:

            pass

                h[j] -= delta
