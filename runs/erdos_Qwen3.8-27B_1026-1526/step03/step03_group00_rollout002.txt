import numpy as np
from scipy.optimize import minimize
import time

def fft_corr_overlap(h):
    n = len(h)
    N = 2 * n - 1
    return np.fft.ifft(np.fft.fft(h, N) * np.fft.fft(1.0 - h[::-1], N)).real

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = fft_corr_overlap(h)
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_step(h, alpha, lam=0.9):
    n = len(h)
    target_sum = n / 2.0
    corr = fft_corr_overlap(h)
    m_star = int(np.argmax(corr))
    k = m_star - (n - 1)
    # Gradient of f_k with respect to h
    g = np.zeros(n)
    if k >= 0:
        if k == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[k:] += (1.0 - h[:n-k])
            g[:n-k] -= h[k:]
    else:
        kk = -k
        g[:n-kk] += (1.0 - h[kk:])
        g[kk:] -= h[:n-kk]
    # Projected gradient step
    h_new = h - alpha * g
    h_new = project_feasible(h_new, target_sum)
    return h_new

def momentum_subgrad(h, target_sum, steps, alpha0, decay, momentum):
    n = len(h)
    h = h.copy()
    v = np.zeros(n)
    c5 = compute_c5(h)
    for t in range(steps):
        alpha = alpha0 * (decay ** t)
        corr = fft_corr_overlap(h)
        m_star = int(np.argmax(corr))
        k = m_star - (n - 1)
        g = np.zeros(n)
        if k >= 0:
            if k == 0:
                g[:] = 1.0 - 2.0 * h
            else:
                g[k:] += (1.0 - h[:n-k])
                g[:n-k] -= h[k:]
        else:
            kk = -k
            g[:n-kk] += (1.0 - h[kk:])
            g[kk:] -= h[:n-kk]
        v = momentum * v - alpha * g
        h_new = h + v
        h_new = project_feasible(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5:
            h = h_new
            c5 = c5_new
        else:
            # Revert
            v = momentum * v  # keep some momentum
            h = project_feasible(h, target_sum)
    return h, c5

import numpy as np
from scipy.optimize import minimize
import time

def fft_corr(h):
    n = len(h)
    N = 2 * n - 1
    return np.fft.ifft(np.fft.fft(h, N) * np.fft.fft(1.0 - h[::-1], N)).real

def c5_val(h):
    corr = fft_corr(h)
    return float(2.0 * np.max(corr) / len(h))

def proj(h, ts):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        d = ts - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h > 1e-13) & (h < 1.0 - 1e-13)
        if m.any():
            h[m] += d / m.sum()
        else:
            h += d / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_lag(h, k):
    n = len(h)
    g = np.zeros(n)
    if k > 0:
        g[k:] += 1.0 - h[:n - k]
        g[:n - k] -= h[k:]
    elif k < 0:
        kk = -k
        g[:n - kk] += 1.0 - h[kk:]
        g[kk:] -= h[:n - kk]
    else:
        g[:] = 1.0 - 2.0 * h
    return g

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = fft_corr(h)
    mval = np.max(corr)
    s = alpha * (corr - mval)
    s -= np.max(s)
    ev = np.exp(s)
    w = ev / ev.sum()
    obj = mval + np.log(ev.sum()) / alpha
    g = np.zeros(n)
    thresh = 1.0 / (n * 10)
    active = np.where(w > thresh)[0]
    for m in active:
        k = int(m) - (n - 1)
        g += w[m] * grad_lag(h, k)
    return obj, g

def slsqp_opt(h0, n, ts, alpha, maxiter):
    cons_sum = lambda x: np.sum(x) - ts
    jac_sum = lambda x: np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_sum, 'jac': jac_sum}]
    def fun(x):
        return smooth_max_grad(x, alpha)[0]
    def jac(x):
        return smooth_max_grad(x, alpha)[1]
    try:
        res = minimize(fun, h0.copy(), method='SLSQP', jac=jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return proj(res.x, ts)
    except Exception:
        return proj(h0.copy(), ts)

    n = len(h)
    h = h.copy()
    c5 = c5_val(h)
    stale = 0
    for it in range(max_iters):
        corr = fft_corr(h) if use_fft else np.correlate(h, 1.0 - h, mode="full")
        mstar = int(np.argmax(corr))
        k = mstar - (n - 1)
        g = grad_lag(h, k)
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not can_inc.any() or not can_dec.any():
            break
        g_inc = np.where(can_inc, g, np.inf)
        g_dec = np.where(can_dec, g, -np.inf)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        if i_inc == i_dec:
            break
        if g[i_inc] >= g[i_dec] - 1e-12:
            break
        max_move = min(1.0 - h[i_inc], h[i_dec])

            stale = 0
    return h, c5
