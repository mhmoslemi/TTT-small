    w = exp_v / np.sum(exp_v)
    obj = m_val + (max_shifted + np.log(np.sum(exp_v))) / alpha
    nj = np.arange(n)[:, None]
    nm = np.arange(2 * n - 1)[None, :]
    lags = nm - (n - 1)
    fi = nj - lags
    si = nj + lags
    vf = (fi >= 0) & (fi < n)
    vs = (si >= 0) & (si < n)
    fc = np.clip(fi, 0, n - 1)
    sc = np.clip(si, 0, n - 1)
    grad = np.sum(vf * (1.0 - h[fc]) * w[None, :] - vs * h[sc] * w[None, :], axis=1) * dx
    return obj, grad

def topk_obj_grad(h, K):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    idx = np.argsort(corr)[::-1][:K]
    w = np.zeros(2 * n - 1)
    w[idx] = 1.0 / K
    obj = np.sum(corr[idx])
    nj = np.arange(n)[:, None]
    nm = np.arange(2 * n - 1)[None, :]
    lags = nm - (n - 1)
    fi = nj - lags
    si = nj + lags
    vf = (fi >= 0) & (fi < n)
    vs = (si >= 0) & (si < n)
    fc = np.clip(fi, 0, n - 1)
    sc = np.clip(si, 0, n - 1)
    grad = np.sum(vf * (1.0 - h[fc]) * w[None, :] - vs * h[sc] * w[None, :], axis=1) * dx
    return obj, grad

def slsqp_smooth(h0, n, ts, alpha, maxiter=1000):
    def f(x):
        return smooth_max_obj_grad(x, alpha)[0]
    def g(x):
        return smooth_max_obj_grad(x, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ts, 'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(f, h0.copy(), method='SLSQP', jac=g, bounds=bnds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14})
        return project_to_sum(r.x, ts)
    except Exception:
        return project_to_sum(h0.copy(), ts)

def slsqp_topk(h0, n, ts, K, maxiter=800):
    def f(x):
        return topk_obj_grad(x, K)[0]
    def g(x):
        return topk_obj_grad(x, K)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ts, 'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(f, h0.copy(), method='SLSQP', jac=g, bounds=bnds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14})
        return project_to_sum(r.x, ts)
    except Exception:
        return project_to_sum(h0.copy(), ts)

def proj_subgrad(h0, n, ts, n_iter=5000, lr_init=0.01):
    h = h0.copy()
    lr = lr_init
    best_obj = compute_c5(h)
    best_h = h.copy()
    stall = 0
    for i in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        dx = 2.0 / n
        c_vals = corr * dx
        m_val = np.max(c_vals)
        active = np.where(np.abs(c_vals - m_val) < 1e-12)[0]
        if len(active) > 8:
            top_idx = np.argsort(c_vals)[::-1][:8]
            active = top_idx
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / len(active)
        nj = np.arange(n)[:, None]
        nm = np.arange(2 * n - 1)[None, :]
        lags = nm - (n - 1)
        fi = nj - lags
        si = nj + lags
        vf = (fi >= 0) & (fi < n)
        vs = (si >= 0) & (si < n)
        fc = np.clip(fi, 0, n - 1)
        sc = np.clip(si, 0, n - 1)
        grad = np.sum(vf * (1.0 - h[fc]) * w[None, :] - vs * h[sc] * w[None, :], axis=1) * dx
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        h_new = project_to_sum(h - lr * grad / gnorm, ts)
        obj_new = compute_c5(h_new)
        if obj_new < best_obj - 1e-15:
            h = h_new
            best_obj = obj_new
            best_h = h.copy()
            stall = 0
            lr = min(lr * 1.03, 0.3)
        else:
            lr *= 0.5
            stall += 1
            if stall > 150 or lr < 1e-14:
                break
    return best_h, best_obj

def stochastic_cd(h, n, ts, rng, n_steps=100000, delta_min=0.005, delta_max=0.3, temperature=0.01):
    h = h.copy()
    c5_cur = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_cur
    for step in range(n_steps):
        if len(h) == 0:
            break
        mask_high = h > 0.02
        mask_low = h < 0.98
        if not np.any(mask_high) or not np.any(mask_low):
            break
        idx_h = np.where(mask_high)[0]
        idx_l = np.where(mask_low)[0]
        i = rng.choice(idx_h)
        j = rng.choice(idx_l)
        if i == j:
            continue
        delta = rng.uniform(delta_min, delta_max)
        if h[i] < delta:
            delta = h[i] * 0.4
        if 1.0 - h[j] < delta:
            delta = (1.0 - h[j]) * 0.4
        if delta < 1e-5:
            continue
        h[i] -= delta
        h[j] += delta
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, ts)
        c5_new = compute_c5(h)
        if c5_new < c5_cur - 1e-15:
            c5_cur = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            temp = temperature * np.exp(-step / max(1, n_steps // 3))
            accept_prob = np.exp(-(c5_new - c5_cur) / max(temp, 1e-10))
            if rng.random() < accept_prob:
                c5_cur = c5_new
            else:
                h[i] += delta
                h[j] -= delta
                h = np.clip(h, 0.0, 1.0)
                h = project_to_sum(h, ts)
    return best_h, best_c5

def upsample(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old, period=2.0)
    return project_to_sum(h_new, n_new / 2.0)

def freq_perturb(h, rng, n_harm, amp):
    n = len(h)
    mean_h = np.mean(h)
    H = np.fft.rfft(h - mean_h)
    upper = min(n_harm, n // 2 - 1)
    for k in range(1, upper + 1):
        H[k] += amp * (rng.randn() + 1j * rng.randn())
    h_new = np.fft.irfft(H, n=n) + mean_h
    return project_to_sum(h_new, n / 2.0)

def make_diverse_starts(n, ts, rng, best_h, best_n, num_starts=8):
    starts = []
    x = np.linspace(0, 2, n, endpoint=False)
    starts.append(np.full(n, 0.5))
    starts.append(project_to_sum(rng.uniform(0, 1, n), ts))
    for f in [2, 3, 5, 7, 11]:
        phase = rng.uniform(0, 2 * np.pi)
        s = 0.5 + 0.3 * np.sin(2.0 * np.pi * f * x / 2.0 + phase)
        starts.append(project_to_sum(s, ts))
    for nb in [4, 6, 8, 12]:
        blk = n // nb
        h_b = np.zeros(n)
        for i in range(0, n, 2 * blk):
            h_b[i:min(i + blk, n)] = 1.0
        starts.append(project_to_sum(h_b, ts))
    if best_h is not None and best_n > 0:
        if best_n == n:
            starts.append(best_h.copy())
        else:
            starts.append(upsample(best_h, n))
    for _ in range(3):
        h_mix = np.zeros(n)
        nf = rng.randint(3, 8)
        for _ in range(nf):
            f = rng.randint(1, 20)
            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.05, 0.2)
            h_mix += amp * np.sin(2.0 * np.pi * f * x / 2.0 + phase)
        h_mix += 0.5
        starts.append(project_to_sum(h_mix, ts))
    return starts[:num_starts]

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()

    def time_left():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        if 'initial_h_values' in globals() and initial_h_values is not None:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
    except Exception:
        pass

    if best_h is None:
        best_n = 480
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    # Phase 1: Upsample to n=960 and quick optimize
    n_target = 960
    if time_left() > 60:
        h_up = upsample(best_h, n_target)
        c5_up = compute_c5(h_up)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up.copy()
            best_n = n_target
        else:
            best_h = h_up.copy()
            best_n = n_target

    # Phase 2: Continuation SLSQP at n=960
    n_cur = best_n
    ts_cur = n_cur / 2.0
    alphas = [20, 80, 300, 1000, 4000, 15000]
    for alpha in alphas:
        if time_left() < 40:
            break
        h_opt = slsqp_smooth(best_h, n_cur, ts_cur, alpha, maxiter=1200)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 3: Top-K objective optimization
    if time_left() > 80:
        for K in [3, 5, 8]:
            if time_left() < 30:
                break
            h_topk = slsqp_topk(best_h, n_cur, ts_cur, K, maxiter=800)
            c5_topk = compute_c5(h_topk)
            if c5_topk < best_c5:
                best_c5 = c5_topk
                best_h = h_topk.copy()
            # Polish the topk result with smooth max
            if time_left() > 30:
                h_pol = slsqp_smooth(h_topk, n_cur, ts_cur, 5000, maxiter=600)
                c5_pol = compute_c5(h_pol)
                if c5_pol < best_c5:
                    best_c5 = c5_pol
                    best_h = h_pol.copy()

    # Phase 4: Projected subgradient with exact max
    if time_left() > 60:
        h_sg, c5_sg = proj_subgrad(best_h, n_cur, ts_cur, n_iter=4000, lr_init=0.02)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
        # Polish
        if time_left() > 30:
            h_pol2 = slsqp_smooth(best_h, n_cur, ts_cur, 10000, maxiter=1000)
            c5_pol2 = compute_c5(h_pol2)
            if c5_pol2 < best_c5:
                best_c5 = c5_pol2
                best_h = h_pol2.copy()

    # Phase 5: Stochastic CD with large steps
    if time_left() > 120:
        h_st, c5_st = stochastic_cd(best_h, n_cur, ts_cur, rng,
                                    n_steps=80000, delta_min=0.01, delta_max=0.35, temperature=0.008)
        if c5_st < best_c5:
            best_c5 = c5_st
            best_h = h_st.copy()
        if time_left() > 30:
            h_pol3 = slsqp_smooth(best_h, n_cur, ts_cur, 8000, maxiter=800)
            c5_pol3 = compute_c5(h_pol3)
            if c5_pol3 < best_c5:
                best_c5 = c5_pol3
                best_h = h_pol3.copy()

    # Phase 6: Frequency perturbation + re-optimize
    if time_left() > 80:
        for trial in range(20):
            if time_left() < 35:
                break
            amp = 0.06 * (1.0 - trial / 25.0) + 0.01
            nf = max(5, n_cur // 30)
            h_sh = freq_perturb(best_h, rng, nf, amp)
            c5_sh = compute_c5(h_sh)
            if c5_sh < best_c5:
                best_c5 = c5_sh
                best_h = h_sh.copy()
            alpha_sh = [200, 2000]
            for a in alpha_sh:
                if time_left() < 20:
                    break
                h_ot = slsqp_smooth(h_sh, n_cur, ts_cur, a, maxiter=500)
                c5_ot = compute_c5(h_ot)
                if c5_ot < best_c5:
                    best_c5 = c5_ot
                    best_h = h_ot.copy()
                h_sh = h_ot

    # Phase 7: Diverse restarts
    if time_left() > 80:
        starts = make_diverse_starts(n_cur, ts_cur, rng, best_h, best_n, num_starts=10)
        for h_s in starts:
            if time_left() < 25:
                break
            c5_s = compute_c5(h_s)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_s.copy()
            for a in [100, 2000]:
                if time_left() < 15:
                    break
                h_o = slsqp_smooth(h_s, n_cur, ts_cur, a, maxiter=500)
                c5_o = compute_c5(h_o)
                if c5_o < best_c5:
                    best_c5 = c5_o
                    best_h = h_o.copy()
                h_s = h_o

    # Phase 8: Stochastic CD small steps, low temperature
    if time_left() > 80:
        h_st2, c5_st2 = stochastic_cd(best_h, n_cur, ts_cur, rng,
                                      n_steps=60000, delta_min=0.003, delta_max=0.12, temperature=0.003)
        if c5_st2 < best_c5:
            best_c5 = c5_st2
            best_h = h_st2.copy()
        if time_left() > 30:
            h_pol4 = slsqp_smooth(best_h, n_cur, ts_cur, 20000, maxiter=1500)
            c5_pol4 = compute_c5(h_pol4)
            if c5_pol4 < best_c5:
                best_c5 = c5_pol4
                best_h = h_pol4.copy()

    # Phase 9: Final SLSQP polish with very high alpha
    if time_left() > 40:
        for alpha_f in [15000, 50000]:
            if time_left() < 25:
                break
            h_f = slsqp_smooth(best_h, n_cur, ts_cur, alpha_f, maxiter=2000)
            c5_f = compute_c5(h_f)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f.copy()

    # Phase 10: Tiny stochastic refinement
    if time_left() > 30:
        h_t, c5_t = stochastic_cd(best_h, n_cur, ts_cur, rng,
                                  n_steps=30000, delta_min=0.002, delta_max=0.06, temperature=0.001)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()

    # Phase 11: Try downsample to n=480 if it gives better result
    if time_left() > 30:
        h_ds = upsample(best_h, 480)
        c5_ds = compute_c5(h_ds)
        h_ds_opt = slsqp_smooth(h_ds, 480, 240.0, 10000, maxiter=1000)
        c5_ds_opt = compute_c5(h_ds_opt)
        if c5_ds_opt < best_c5:
            best_c5 = c5_ds_opt
            best_h = h_ds_opt.copy()
            best_n = 480

    # Final validation
    best_n = len(best_h)
    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    for _ in range(300):
        diff = best_n / 2.0 - np.sum(best_h)
        if abs(diff) < 1e-15:
            break
        mask = (best_h > 1e-10) & (best_h < 1.0 - 1e-10)
        if np.any(mask):
            best_h[mask] += diff / len(mask)
            best_h = np.clip(best_h, 0.0, 1.0)
        else:
            break
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
