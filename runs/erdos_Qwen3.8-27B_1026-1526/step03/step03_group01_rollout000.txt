import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-16:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / n
    shifted = alpha * corr
    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    obj = (max_sh + np.log(np.sum(exp_v))) / alpha * dx
    return obj

def smooth_max_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    shifted = alpha * corr
    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    w = exp_v / np.sum(exp_v)
    nj = np.arange(n)[:, None]
    nm = np.arange(2 * n - 1)[None, :]
    fi = nj - nm + n - 1
    si = nj + nm - n + 1
    vf = (fi >= 0) & (fi < n)
    vs = (si >= 0) & (si < n)
    fc = np.clip(fi, 0, n - 1)
    sc = np.clip(si, 0, n - 1)
    grad = np.sum(vf * (1.0 - h[fc]) * w[None, :] - vs * h[sc] * w[None, :], axis=1) * dx
    return grad

def obj_and_grad(h, alpha):
    return smooth_max_obj(h, alpha), smooth_max_grad(h, alpha)

def lbfgs_opt(h0, n, ts, alpha, maxiter=5000, ftol=1e-18):

        obj = smooth_max_obj(x, alpha) + penalty
        grad = smooth_max_grad(x, alpha) + 2e6 * (np.sum(x) - ts)
        return obj, grad

                     bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-14})
        return project_to_sum(np.clip(r.x, 0.0, 1.0), ts)

        return project_to_sum(r.x, ts)
    except Exception:
        return project_to_sum(h0, ts)

def nesterov_pg(h0, n, ts, alpha, n_iter=5000, lr_init=0.01):
    h = h0.copy()
    h_prev = h0.copy()
    t = 1.0
    lr = lr_init
    y = h.copy()
    obj_y, _ = obj_and_grad(y, alpha)
    best_obj = obj_y
    best_y = y.copy()
    for i in range(n_iter):
        obj_y, grad_y = obj_and_grad(y, alpha)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = y + (t / t_new - 1.0) * (y - h_prev)
        y_new = y_new - lr * grad_y
        y_new = project_to_sum(y_new, ts)
        h_new = y_new.copy()
        obj_new, _ = obj_and_grad(h_new, alpha)
        if obj_new < best_obj:
            best_obj = obj_new
            best_y = h_new.copy()
        # Simple step size adaptation
        diff = np.linalg.norm(h_new - h_prev)
        if diff < 1e-10:
            lr *= 0.5
        else:
            lr = min(lr * 1.01, 0.1)
        h_prev = h_prev  # keep for momentum
        h = h_new
        h_prev = h
        t = t_new
        y = h
    return best_y

def targeted_perturb(h, rng, n_active=20, amp=0.02):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    # Find active lags
    active = np.where(corr > m_val - 0.01 * m_val)[0]
    # Perturb h at positions that affect active lags
    h_new = h.copy()
    for _ in range(20):
        if len(active) == 0:
            break
        m_idx = rng.choice(active)
        d = m_idx - (n - 1)  # the lag
        # Perturb a random position
        j = rng.randint(n)
        h_new[j] += amp * (rng.rand() - 0.5) * 2
    h_new = project_to_sum(np.clip(h_new, 0, 1), n / 2.0)
    return h_new

def multi_alpha_opt(h0, n, ts, alphas, maxiter_per_alpha=500):
    h = h0.copy()
    for alpha in alphas:
        h = slsqp_opt(h, n, ts, alpha, maxiter=maxiter_per_alpha)
    return h

def near_binary(h, eps=0.05):
    # Push h towards binary (0 or 1) with a small epsilon
    n = len(h)
    ts = n / 2.0
    h_bin = np.where(h > 0.5, 1.0 - eps, eps)
    h_bin = project_to_sum(h_bin, ts)
    return h_bin
