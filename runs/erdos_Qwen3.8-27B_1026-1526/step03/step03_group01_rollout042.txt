def compute_g_array(h):
    """Compute g(s)*dx for all shifts s, returned as array of length 2*n-1."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h.astype(np.float64), 1.0 - h.astype(np.float64), mode="full")
    return corr * dx

def compute_c5(h):
    return float(np.max(compute_g_array(h)))

def make_binary(h, rng, n):
    """Convert a continuous h to binary with exactly n/2 ones."""
    h_bin = (h >= 0.5).astype(np.float64)
    n_ones = int(np.sum(h_bin))
    target = n // 2
    if n_ones > target:
        ones = np.where(h_bin == 1)[0]
        # Flip the ones with lowest h values first
        order = np.argsort(h[ones])[:n_ones - target]
        h_bin[ones[order]] = 0
    elif n_ones < target:
        zeros = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros])[:target - n_ones]
        h_bin[zeros[order]] = 1
    return h_bin

import numpy as np

def compute_g(h, n=None):
    """Compute the g array (correlation values, not yet multiplied by dx)."""
    if n is None:
        n = len(h)
    corr = np.correlate(h.astype(np.float64), 1.0 - h.astype(np.float64), mode="full")
    return corr

def compute_c5_from_g(g, n):
    dx = 2.0 / n
    return float(np.max(g)) * dx

def make_binary_from_h(h, rng, n):
    h_bin = (h >= 0.5).astype(np.float64)
    n_ones = int(np.sum(h_bin))
    target = n // 2
    if n_ones > target:
        ones = np.where(h_bin == 1)[0]
        # Flip the ones with lowest h values
        order = np.argsort(h[ones])[:n_ones - target]
        h_bin[ones[order]] = 0
    elif n_ones < target:
        zeros = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros])[:target - n_ones]
        h_bin[zeros[order]] = 1
    return h_bin

def binary_swap_delta(h, i, j, n):
    """Compute delta in g (before dx) for swapping h[i]:1->0 and h[j]:0->1."""
    delta = np.zeros(2*n-1, dtype=np.float64)
    h_rev = h[::-1]
    
    # Effect 1: delta[n-1-i : 2n-1-i] += h - 1
    delta[n-1-i : 2*n-1-i] += (h - 1.0)
    
    # Effect 2: delta[i : i+n] += h[::-1]
    delta[i : i+n] += h_rev
    
    # Effect 3: delta[n-1-j : 2n-1-j] += 1 - h
    delta[n-1-j : 2*n-1-j] += (1.0 - h)
    
    # Effect 4: delta[j : j+n] += -h[::-1]
    delta[j : j+n] += (-h_rev)
    
    return delta

def sa_binary(h, n, g, rng, n_moves, T_start, T_end, beta):
    """Simulated annealing on binary h, operating on g array directly."""
    best_g = g.copy()
    best_c5 = float(np.max(g)) * (2.0/n)
    cur_c5 = best_c5
    cur_max = float(np.max(g))
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    
    for step in range(n_moves):
        T = T_start * (T_end/T_start) ** (step / max(n_moves-1, 1))
        
        # Pick random i from ones, j from zeros
        ii = rng.randint(len(ones))
        jj = rng.randint(len(zeros))
        i = ones[ii]
        j = zeros[jj]
        if i == j:
            continue
        
        delta = binary_swap_delta(h, i, j, n)
        new_g = g + delta
        new_max = float(np.max(new_g))
        new_c5 = new_max * (2.0/n)
        
        if new_c5 < cur_c5 or rng.random() < np.exp(-(new_c5 - cur_c5) / max(T, 1e-10)):
            # Accept
            h[i] = 0
            h[j] = 1
            ii = np.where(ones == i)[0]
            if len(ii) > 0:
                ones[ii[0]] = zeros[jj]
                jj = np.where(zeros == j)[0]
                if len(jj) > 0:
                    zeros[jj[0]] = i
            else:
                # Should not happen
                ones = np.append(ones, j)
                zeros = np.delete(zeros, jj)
            
            g = new_g
            cur_c5 = new_c5
            cur_max = new_max
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_g = g.copy()
                best_h = h.copy()
        else:
            # Reject: do nothing (h and g unchanged)
            pass
    
    return best_h, best_g, best_c5

def compute_g_array(h):
    n = len(h)
    corr = np.correlate(h.astype(np.float64), 1.0 - h.astype(np.float64), mode="full")
    return corr

def compute_c5_from_g(g, n):
    return float(np.max(g)) * (2.0 / n)

def make_binary(h, rng, n):
    h_bin = np.zeros(n, dtype=np.float64)
    h_bin[h >= 0.5] = 1.0
    n_ones = int(np.sum(h_bin))
    target = n // 2
    if n_ones > target:
        ones = np.where(h_bin == 1)[0]
        order = np.argsort(-h[ones])  # flip the ones with lowest h values... wait
        # We want to flip the ones that are "least 1-like" (lowest h value)
        order = np.argsort(h[ones])[:n_ones - target]
        h_bin[ones[order]] = 0
    elif n_ones < target:
        zeros = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros])[:target - n_ones]
        h_bin[zeros[order]] = 1
    return h_bin

def swap_delta(h, i, j, n):
    """Delta in g for swapping h[i]:1->0, h[j]:0->1."""
    delta = np.zeros(2*n-1, dtype=np.float64)
    h1 = h - 1.0
    h1n = 1.0 - h
    hr = h[::-1]
    hr_neg = -hr
    delta[n-1-i : 2*n-1-i] += h1
    delta[i : i+n] += hr
    delta[n-1-j : 2*n-1-j] += h1n
    delta[j : j+n] += hr_neg
    return delta

def sa_loop(h, n, g, rng, n_moves, T_start, T_end, best_h, best_c5):
    """Run SA for n_moves steps, return updated best."""
    dx = 2.0 / n
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    n_half = n // 2
    cur_c5 = float(np.max(g)) * dx
    g_tmp = np.empty(2*n-1, dtype=np.float64)
    
    for step in range(n_moves):
        T = T_start * (T_end / T_start) ** (step / max(n_moves-1, 1))
        
        ii = rng.randint(n_half)
        jj = rng.randint(n_half)
        i = ones[ii]
        j = zeros[jj]
        if i == j:
            continue
        
        delta = swap_delta(h, i, j, n)
        np.add(g, delta, out=g_tmp)
        new_c5 = float(np.max(g_tmp)) * dx
        
        if new_c5 < cur_c5:
            accept = True
        else:
            delta_c5 = new_c5 - cur_c5
            if delta_c5 > 5.0 * T:  # too bad, definitely reject
                accept = False
            else:
                accept = (rng.random() < np.exp(-delta_c5 / T))
        
        if accept:
            h[i] = 0.0
            h[j] = 1.0
            ones[ii] = j
            zeros[jj] = i
            g, g_tmp = g_tmp, g
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
    
    return h, g, best_h, best_c5
