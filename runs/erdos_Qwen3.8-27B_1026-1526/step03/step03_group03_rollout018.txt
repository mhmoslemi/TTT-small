    k_start = max(0, nm1 - j)
    k_end = min(2 * n - 2, 2 * nm1 - j)
    if k_start <= k_end:
        B[k_start:k_end + 1] += (1.0 - h[j - nm1 + np.arange(k_start, k_end + 1)])
    k_start2 = max(0, j)
    k_end2 = min(2 * n - 2, j + nm1)
    if k_start2 <= k_end2:
        B[k_start2:k_end2 + 1] += -h[j + nm1 - np.arange(k_start2, k_end2 + 1)]
    return B

def solve_1d(A, D, lo, hi):
    if hi - lo < 1e-14:
        return (lo + hi) * 0.5
    t_grid = np.linspace(lo, hi, 24)
    vals = np.max(A[None, :] + D[None, :] * t_grid[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        return lo
    if best_idx == len(t_grid) - 1:
        return hi
    lo_r = t_grid[best_idx - 1]
    hi_r = t_grid[best_idx + 1]
    for _ in range(45):
        mid = (lo_r + hi_r) * 0.5
        if np.max(A + D * mid) <= np.max(A + D * lo_r):
            lo_r = mid
        else:
            hi_r = mid
    return (lo_r + hi_r) * 0.5

def pair_descent_sweep(h, corr, rng, n_cand=12, use_gradient_hints=False):
    n = len(h)
    order = np.arange(n)
    rng.shuffle(order)
    improved = False
    top_lags = np.argsort(corr)[-6:] if use_gradient_hints else np.array([], dtype=int)
    for j in order:
        B_j = compute_B(h, j, n)
        best_val = np.max(corr)
        best_delta = 0.0
        best_D = None
        best_k = -1
        k_set = set()
        k_set.add((j + 1) % n)
        k_set.add((j - 1 + n) % n)
        k_set.add((j + n // 2) % n)
        k_set.add((j + n // 3) % n)
        k_set.add((j + 2 * n // 3) % n)
        k_set.add((j + n // 4) % n)
        k_set.add((j + 3 * n // 4) % n)
        k_set.add((j + n // 5) % n)
        k_set.add((j + 2 * n // 5) % n)
        if use_gradient_hints:
            for tk in top_lags:
                offset = int(tk) - (n - 1)
                k_set.add((j + offset + 1) % n)
                k_set.add((j - offset - 1) % n)
                k_set.add((j + offset) % n)
        extra_needed = max(0, n_cand - len(k_set))
        for _ in range(extra_needed):
            k_set.add(rng.randint(n))
        for k in k_set:
            if k == j:
                continue
            B_k = compute_B(h, k, n)
            D = B_j - B_k
            delta_lo = max(-h[j], h[k] - 1.0)
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-14:
                continue
            delta_opt = solve_1d(corr, D, delta_lo, delta_hi)
            val_opt = np.max(corr + D * delta_opt)
            if val_opt < best_val - 1e-15:
                best_val = val_opt
                best_delta = delta_opt
                best_D = D
                best_k = k
        if best_k >= 0 and abs(best_delta) > 1e-15:
            corr += best_delta * best_D
            h[j] = np.clip(h[j] + best_delta, 0.0, 1.0)
            h[best_k] = np.clip(h[best_k] - best_delta, 0.0, 1.0)
            improved = True
    return h, corr, improved

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    if total < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w = exp_vals / total
    obj = m_val + np.log(total + 1e-300) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

def slsqp_optimize(h0, n, target_sum, alpha, maxiter):
    def f(h):
        return smooth_max_and_grad(h, alpha)[0]
    def g(h):
        return smooth_max_and_grad(h, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_to_sum(res.x, target_sum)
    except Exception:
        pass
    return project_to_sum(h0, target_sum)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float).copy()
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 240
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n = best_n
    target_sum = n / 2.0

    # Phase 1: Pair descent (30% of budget)
    t_end1 = t_start + 0.30 * budget_s
    h_work = best_h.copy()
    corr_w = np.correlate(h_work, 1.0 - h_work, mode="full")
    stall = 0
    noise = 0.05

    while time.time() < t_end1:
        h_work, corr_w, _ = pair_descent_sweep(h_work, corr_w, rng, n_cand=14, use_gradient_hints=True)
        c5_w = compute_c5(h_work)
        if c5_w < best_c5 - 1e-12:
            best_c5 = c5_w
            best_h = h_work.copy()
            best_n = n
            stall = 0
            noise = 0.05
        else:
            stall += 1
        if stall > 0 and stall % 40 == 0:
            if stall > 250:
                noise = 0.12
            elif stall > 100:
                noise = 0.07
            else:
                noise = 0.03
            h_work = project_to_sum(h_work + rng.uniform(-noise, noise, n), target_sum)
            corr_w = np.correlate(h_work, 1.0 - h_work, mode="full")
            stall = 0

    # Phase 2: SLSQP with progressive smoothing (45% of budget)
    t_end2 = t_start + 0.75 * budget_s
    alphas = [5, 15, 40, 100, 300, 800, 2000, 5000]
    h_sls = best_h.copy()

    for alpha in alphas:
        if time.time() >= t_end2 - 20:
            break
        remaining = t_end2 - 20 - time.time()
        est_iters_per_sec = 3.0
        maxiter = max(100, int(remaining * est_iters_per_sec * 0.6))
        maxiter = min(maxiter, 5000)
        h_sls = slsqp_optimize(h_sls, n, target_sum, alpha, maxiter)
        c5_s = compute_c5(h_sls)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_sls.copy()
            best_n = n
        # Quick pair descent cleanup after each SLSQP
        h_tmp = h_sls.copy()
        corr_t = np.correlate(h_tmp, 1.0 - h_tmp, mode="full")
        n_sweeps = min(8, max(2, int((t_end2 - time.time()) / 3.0)))
        for _ in range(n_sweeps):
            if time.time() >= t_end2 - 15:
                break
            h_tmp, corr_t, _ = pair_descent_sweep(h_tmp, corr_t, rng, n_cand=12, use_gradient_hints=True)
        c5_t = compute_c5(h_tmp)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_tmp.copy()
            best_n = n
            h_sls = h_tmp.copy()

    # Phase 3: Final alternating SLSQP + pair descent (15% of budget)
    t_end3 = t_start + 0.90 * budget_s
    while time.time() < t_end3:
        h_curr = best_h.copy()
        h_curr = slsqp_optimize(h_curr, n, target_sum, 3000, maxiter=400)
        c5_s = compute_c5(h_curr)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_curr.copy()
        h_curr = best_h.copy()
        corr_c = np.correlate(h_curr, 1.0 - h_curr, mode="full")
        for _ in range(4):
            if time.time() >= t_end3:
                break
            h_curr, corr_c, _ = pair_descent_sweep(h_curr, corr_c, rng, n_cand=16, use_gradient_hints=True)
        c5_c = compute_c5(h_curr)
        if c5_c < best_c5:
            best_c5 = c5_c
            best_h = h_curr.copy()
            best_n = n

    # Phase 4: Intensive final pair descent (10% of budget)
    h_fin = best_h.copy()
    corr_fin = np.correlate(h_fin, 1.0 - h_fin, mode="full")
    t_final_end = t_start + budget_s - 2
    while time.time() < t_final_end:
        h_fin, corr_fin, _ = pair_descent_sweep(h_fin, corr_fin, rng, n_cand=16, use_gradient_hints=True)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5 - 1e-14:
            best_c5 = c5_fin
            best_h = h_fin.copy()
            best_n = n

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
