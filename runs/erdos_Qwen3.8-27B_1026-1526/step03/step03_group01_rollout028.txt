import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = alpha * (corr - m)
    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    w = exp_v / np.sum(exp_v)
    obj = m + (max_sh + np.log(np.sum(exp_v))) / alpha
    nk = 2 * n - 1
    js = np.arange(n)[:, None]
    ks = np.arange(nk)[None, :]
    fi = js + ks - (n - 1)
    si = js - ks + (n - 1)
    vf = (fi >= 0) & (fi < n)
    vs = (si >= 0) & (si < n)
    fc = np.clip(fi, 0, n - 1)
    sc = np.clip(si, 0, n - 1)
    dc = vf.astype(np.float64) * (1.0 - h[fc]) - vs.astype(np.float64) * h[sc]
    grad = dc @ w
    return obj, grad

def nesterov_pg(h0, target_sum, alpha, n_iter, lr_init, beta, time_budget):
    n = len(h0)
    h = h0.copy()
    obj, _ = lse_and_grad(h, alpha)
    best_obj = obj
    best_h = h.copy()
    v = np.zeros(n)
    lr = lr_init
    stall = 0
    t0 = time.time()
    for it in range(n_iter):
        if time.time() - t0 > time_budget:
            break
        y = h + beta * v
        y = project_to_sum(y, target_sum)
        obj_y, grad_y = lse_and_grad(y, alpha)
        new_v = beta * v + grad_y
        h_new = y - lr * new_v
        h_new = project_to_sum(h_new, target_sum)
        obj_new = lse_and_grad(h_new, alpha)[0]
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_h = h_new.copy()
            stall = 0
            lr = min(lr * 1.015, 0.08)
        else:
            stall += 1
            lr *= 0.75
            if stall > 800 or lr < 1e-9:
                break
        v = h_new - h
        h = h_new
    return best_h, best_obj

def freq_perturb(h, rng, n_harm, amp):
    n = len(h)
    mean_h = np.mean(h)
    H = np.fft.rfft(h - mean_h)
    upper = min(n_harm, n // 2 - 1)
    for k in range(1, upper + 1):
        H[k] += amp * (rng.randn() + 1j * rng.randn())
    h_new = np.fft.irfft(H, n=n) + mean_h
    return project_to_sum(h_new, n // 2 * 1.0)

def snap_and_reoptimize(h, target_sum, alpha, n_iter, threshold=0.05):
    n = len(h)
    h_free = h.copy()
    fixed_mask = (h_free < threshold) | (h_free > 1.0 - threshold)
    h_free[fixed_mask] = np.where(h_free[fixed_mask] < 0.5, 0.0, 1.0)
    h_free = project_to_sum(h_free, target_sum)
    obj, grad = lse_and_grad(h_free, alpha)
    v = np.zeros(n)
    lr = 0.003
    best_obj = obj
    best_h = h_free.copy()
    stall = 0
    for it in range(n_iter):
        y = h_free + 0.8 * v
        y = np.clip(y, 0.0, 1.0)
        y[fixed_mask] = h_free[fixed_mask]
        y = project_to_sum(y, target_sum)
        y[fixed_mask] = h_free[fixed_mask]
        obj_y, grad_y = lse_and_grad(y, alpha)
        new_v = 0.8 * v + grad_y
        h_new = y - lr * new_v
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new[fixed_mask] = h_free[fixed_mask]
        h_new = project_to_sum(h_new, target_sum)
        h_new[fixed_mask] = h_free[fixed_mask]
        obj_new = lse_and_grad(h_new, alpha)[0]
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_h = h_new.copy()
            stall = 0
            lr = min(lr * 1.01, 0.05)
        else:
            stall += 1
            lr *= 0.8
            if stall > 400:
                break
        v = h_new - h_free
        h_free = h_new
    return best_h

def continuation_opt(h_init, target_sum, alphas, n_iter_per_alpha, time_budget, lr_init=0.003):
    h = h_init.copy()
    c5_cur = compute_c5(h)
    for alpha in alphas:
        remaining = time_budget - (time.time() - t_global_start)
        if remaining < 5:
            break
        iter_budget = max(500, int(n_iter_per_alpha * min(1.0, remaining / 60.0)))
        h, _ = nesterov_pg(h, target_sum, alpha, iter_budget, lr_init, 0.8, remaining - 5)
        c5_new = compute_c5(h)
        c5_cur = min(c5_cur, c5_new)
    return h, c5_cur

t_global_start = time.time()

def run(seed=42, budget_s=1300, **kwargs):
    global t_global_start
    t_global_start = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        if 'initial_h_values' in globals() and initial_h_values is not None:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
    except Exception:
        pass

    if best_h is None:
        best_n = 480
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    n = best_n
    ts = n / 2.0

    # Phase 1: Continuation with Nesterov projected gradient
    alphas1 = [5, 15, 50, 150, 500, 2000, 8000]
    for alpha in alphas1:
        if time.time() - t_global_start > budget_s - 30:
            break
        h_opt, _ = nesterov_pg(best_h, ts, alpha, 4000, 0.003, 0.8,
                               budget_s - (time.time() - t_global_start) - 25)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 2: Snap and re-optimize (active set)
    if time.time() - t_global_start < budget_s - 60:
        for thresh in [0.03, 0.05, 0.08, 0.12]:
            if time.time() - t_global_start > budget_s - 40:
                break
            h_snap = snap_and_reoptimize(best_h, ts, 5000, 3000, threshold=thresh)
            c5_snap = compute_c5(h_snap)
            if c5_snap < best_c5:
                best_c5 = c5_snap
                best_h = h_snap.copy()
                # Continue snapping with same threshold
                n = best_n
                ts = n / 2.0

    # Phase 3: Multiple frequency perturbations + re-optimization
    for trial in range(30):
        if time.time() - t_global_start > budget_s - 50:
            break
        amp = 0.06 * (1.0 - trial / 40.0) + 0.01
        nf = max(3, best_n // 30)
        h_sh = freq_perturb(best_h, rng, nf, amp)
        c5_sh = compute_c5(h_sh)
        if c5_sh < best_c5:
            best_c5 = c5_sh
            best_h = h_sh.copy()
        for alpha in [200, 2000]:

                                  min(30, budget_s - (time.time() - t_global_start) - 25))
            c5_ot = compute_c5(h_ot)
            if c5_ot < best_c5:
                best_c5 = c5_ot
                best_h = h_ot.copy()
            h_sh = h_ot

    # Phase 4: Try different starting structures at same resolution
    x = np.linspace(0, 2, n, endpoint=False)
    starts = []
    for f in [1, 2, 3, 4, 5, 6, 7, 8, 10, 12]:
        for phase_shift in [0.0, 0.5, 1.0]:

        if time.time() - t_global_start > budget_s - 60:
            break
        c5_s = compute_c5(h_s)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

        if c5_s - best_c5 < 0.05:

                if time.time() - t_global_start > budget_s - 30:

        c5_up = compute_c5(h_up)

            best_c5 = c5_up

        else:

    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
