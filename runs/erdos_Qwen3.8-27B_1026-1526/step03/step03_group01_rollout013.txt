def smooth_max_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    s = np.exp(alpha * (corr - m))
    obj = m + np.log(np.sum(s)) / alpha
    w = s / np.sum(s)
    return obj, w

def grad_smooth(h, w):
    n = len(h)
    nj = np.arange(n)[:, None]
    nm = np.arange(2 * n - 1)[None, :]
    fi = nj - nm + n - 1
    si = nj + nm - n + 1
    vf = (fi >= 0) & (fi < n)
    vs = (si >= 0) & (si < n)
    fc = np.clip(fi, 0, n - 1)
    sc = np.clip(si, 0, n - 1)
    grad = np.sum(vf * (1.0 - h[fc]) * w[None, :] - vs * h[sc] * w[None, :], axis=1)
    return grad

```python
import numpy as np
from scipy.optimize import minimize
import time

# All helper functions at top level

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def _con_fun(x):
    return np.sum(x) - len(x) / 2.0

def _con_jac(x):
    return np.ones(len(x))

def _obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    s = np.exp(alpha * (corr - m))
    return m + np.log(np.sum(s)) / alpha

def _grad(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    s = np.exp(alpha * (corr - m))
    w = s / np.sum(s)
    n = len(x)
    nj = np.arange(n)[:, None]
    nm_arr = np.arange(2 * n - 1)[None, :]
    fi = nj - nm_arr + n - 1
    si = nj + nm_arr - n + 1
    vf = (fi >= 0) & (fi < n)
    vs = (si >= 0) & (si < n)
    fc = np.clip(fi, 0, n - 1)
    sc = np.clip(si, 0, n - 1)
    return np.sum(vf * (1.0 - x[fc]) * w[None, :] - vs * x[sc] * w[None, :], axis=1)

def slsqp_opt(h0, alpha, maxiter):
    n = len(h0)
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        r = minimize(_obj, h0.copy(), args=(alpha,), method='SLSQP',
                     jac=_grad, bounds=bnds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-13})
        return project_to_sum(r.x, n / 2.0)
    except Exception:
        return project_to_sum(h0, n / 2.0)

def momentum_grad(h, alpha, n_iter, lr_init, beta):
    """Projected gradient descent with momentum."""
    h = h.copy()
    n = len(h)
    ts = n / 2.0
    lr = lr_init
    v = np.zeros(n)
    best_h = h.copy()
    best_obj = _obj(h, alpha)
    stall = 0
    for i in range(n_iter):
        g = np.correlate(h, 1.0 - h, mode="full")
        m = np.max(g)
        s = np.exp(alpha * (g - m))
        w = s / np.sum(s)
        grad = _grad(h, alpha)
        v = beta * v + (1.0 - beta) * grad
        h_new = project_to_sum(h - lr * v, ts)
        obj_new = _obj(h_new, alpha)
        if obj_new < best_obj - 1e-15:
            best_h = h_new.copy()
            best_obj = obj_new
            h = h_new
            stall = 0
            lr = min(lr * 1.01, 0.3)
        else:
            lr *= 0.7
            stall += 1
            if stall > 200 or lr < 1e-14:
                break
    return best_h

def stochastic_search(h, rng, n_steps, delta_lo, delta_hi):
    """Stochastic mass transfer search."""
    n = len(h)
    h = h.copy()
    ts = n / 2.0
    c5_cur = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_cur
    temp = 0.15
    for step in range(n_steps):
        # Pick two indices to transfer mass
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        delta = rng.uniform(delta_lo, delta_hi)
        if h[i] < delta:
            delta = h[i] * 0.8
        if 1.0 - h[j] < delta:
            delta = (1.0 - h[j]) * 0.8
        if delta < 1e-5:
            continue
        h[i] -= delta
        h[j] += delta
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, ts)
        c5_new = compute_c5(h)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
        if c5_new < c5_cur or (c5_new - c5_cur > -100 and rng.random() < np.exp(-(c5_new - c5_cur) / temp)):
            c5_cur = c5_new
        else:
            h[i] += delta
            h[j] -= delta
            h = np.clip(h, 0.0, 1.0)
            h = project_to_sum(h, ts)
        temp = max(1e-5, 0.15 * np.exp(-step / max(1, n_steps // 3)))
    return best_h, best_c5

def fourier_perturb(h, rng, n_harm, amp):
    n = len(h)
    mean_h = np.mean(h)
    H = np.fft.rfft(h - mean_h)
    upper = min(n_harm, n // 2 - 1)
    for k in range(1, upper + 1):
        H[k] += amp * (rng.randn() + 1j * rng.randn())
    h_new = np.fft.irfft(H, n=n) + mean_h
    return project_to_sum(h_new, n / 2.0)

def upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    return project_to_sum(h_new, n_new / 2.0)

def make_structured_starts(n, rng, best_h, best_n):
    """Generate diverse starting points."""
    starts = []
    x = np.linspace(0, 2, n, endpoint=False)
    # Constant
    starts.append(np.full(n, 0.5))
    # Random
    starts.append(project_to_sum(rng.uniform(0, 1, n), n / 2.0))
    # Sine waves
    for f in [2, 3, 5, 7, 11]:
        s = 0.5 + 0.35 * np.sin(2.0 * np.pi * f * x / 2.0 + rng.uniform(0, 2 * np.pi))
        starts.append(project_to_sum(s, n / 2.0))
    # Block patterns
    for nb in [4, 6, 8, 12]:
        blk = n // nb
        h_b = np.zeros(n)
        for i in range(0, n, 2 * blk):
            h_b[i:min(i + blk, n)] = 1.0
        starts.append(project_to_sum(h_b, n / 2.0))
    # Upsampled best
    if best_h is not None:
        if best_n == n:
            starts.append(best_h.copy())
        else:
            starts.append(upsample(best_h, n))
    # Random Fourier mix
    for _ in range(3):
        h_mix = np.zeros(n)
        nf = rng.randint(3, 8)
        freqs = rng.choice(np.arange(1, 20), size=nf, replace=False)
        for f in freqs:

        h_mix += 0.5
        starts.append(project_to_sum(h_mix, n / 2.0))
    return starts
