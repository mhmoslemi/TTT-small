```python
def compute_swap_delta(h, a, b):
    n = len(h)
    delta = np.zeros(2*n-1)
    # a goes from 1 to 0
    # First term: index j = a+k-n+1, valid when j in [0,n-1], i.e., k in [n-1-a, 2n-2-a]
    k_lo = max(0, n-1-a)
    k_hi = min(2*n-2, 2*n-2-a)
    if k_lo <= k_hi:
        j = np.arange(k_lo, k_hi+1) + (n-1-a)  # j = a+k-n+1
        delta[k_lo:k_hi+1] += h[j] - 1
    # Second term: index j = a-k+n-1, valid when j in [0,n-1], i.e., k in [a, a+n-1]
    k_lo2 = max(0, a)
    k_hi2 = min(2*n-2, a+n-1)
    if k_lo2 <= k_hi2:
        j = a + n - 1 - np.arange(k_lo2, k_hi2+1)  # j = a-k+n-1
        delta[k_lo2:k_hi2+1] += h[j]
    # b goes from 0 to 1
    # Third term: index j = b+k-n+1, valid when j in [0,n-1]
    k_lo3 = max(0, n-1-b)
    k_hi3 = min(2*n-2, 2*n-2-b)
    if k_lo3 <= k_hi3:
        j = np.arange(k_lo3, k_hi3+1) + (n-1-b)
        delta[k_lo3:k_hi3+1] += 1 - h[j]
    # Fourth term: index j = b-k+n-1
    k_lo4 = max(0, b)
    k_hi4 = min(2*n-2, b+n-1)
    if k_lo4 <= k_hi4:
        j = b + n - 1 - np.arange(k_lo4, k_hi4+1)
        delta[k_lo4:k_hi4+1] -= h[j]
    return delta
```

def compute_swap_delta(h, a, b, n):
    delta = np.zeros(2*n-1)
    nm1 = n - 1
    
    # a: 1 -> 0
    # First factor: i=a, j=a+k-nm1, valid when 0<=j<=n-1, i.e., nm1-a <= k <= 2*nm1-a
    k1_lo = max(0, nm1 - a)
    k1_hi = min(2*nm1, 2*nm1 - a)
    if k1_lo <= k1_hi:
        js = np.arange(k1_lo, k1_hi+1) + nm1 - a
        delta[k1_lo:k1_hi+1] += h[js] - 1.0
