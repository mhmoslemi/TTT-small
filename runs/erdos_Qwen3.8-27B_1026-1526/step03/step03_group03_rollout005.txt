    k_end = min(2 * n - 2, 2 * nm1 - j)

        idx = np.arange(k_start, k_end + 1)
        B[k_start:k_end + 1] += (1.0 - h[j - nm1 + idx])
    k_start2 = max(0, j)
    k_end2 = min(2 * n - 2, j + nm1)
    if k_start2 <= k_end2:
        idx2 = np.arange(k_start2, k_end2 + 1)
        B[k_start2:k_end2 + 1] += -h[j + nm1 - idx2]
    return B

def solve_1d(A, D, lo, hi):
    if hi - lo < 1e-14:
        return (lo + hi) * 0.5
    n_grid = 20
    t_grid = np.linspace(lo, hi, n_grid)
    vals = np.max(A[None, :] + D[None, :] * t_grid[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        return lo
    if best_idx == n_grid - 1:
        return hi
    lo_r = t_grid[best_idx - 1]
    hi_r = t_grid[best_idx + 1]
    for _ in range(42):
        mid = (lo_r + hi_r) * 0.5
        if np.max(A + D * mid) <= np.max(A + D * lo_r):
            hi_r = mid
        else:
            lo_r = mid
    return (lo_r + hi_r) * 0.5

def pair_descent_sweep(h, corr, rng, n_cand=12, temp=0.0):
    n = len(h)
    order = np.arange(n)
    rng.shuffle(order)
    improved = False
    for j in order:
        B_j = compute_B(h, j, n)
        cur_max = np.max(corr)
        best_val = cur_max
        best_delta = 0.0
        best_D = None
        best_k = -1
        cands = set()
        for off in range(1, 6):
            cands.add((j + off) % n)
            cands.add((j - off + n) % n)
        cands.add((j + n // 2) % n)
        cands.add((j + n // 3) % n)
        cands.add((j + 2 * n // 3) % n)
        cands.add((j + n // 4) % n)
        cands.add((j + n // 6) % n)
        for _ in range(max(0, n_cand - len(cands))):
            cands.add(rng.randint(n))
        cands.discard(j)
        cand_list = list(cands)
        rng.shuffle(cand_list)
        cand_list = cand_list[:n_cand]
        for k in cand_list:
            B_k = compute_B(h, k, n)
            D = B_j - B_k
            delta_lo = max(-h[j], h[k] - 1.0)
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-13:
                continue
            delta_opt = solve_1d(corr, D, delta_lo, delta_hi)
            val_opt = np.max(corr + D * delta_opt)
            if temp > 1e-10:
                accept = val_opt < best_val - 1e-13 or (val_opt - best_val) < temp * rng.normal()
            else:
                accept = val_opt < best_val - 1e-13
            if accept and val_opt < best_val - 1e-13:
                best_val = val_opt
                best_delta = delta_opt
                best_D = D
                best_k = k
        if best_k >= 0 and abs(best_delta) > 1e-13:
            corr += best_delta * best_D
            h[j] += best_delta
            h[best_k] -= best_delta
            h[j] = np.clip(h[j], 0.0, 1.0)
            h[best_k] = np.clip(h[best_k], 0.0, 1.0)
            improved = True
    return h, corr, improved

def fourier_perturb(h, rng, n_modes=4, amp=0.08):
    n = len(h)
    x = np.arange(n)
    ps = (h > 1e-12) & (h < 1.0 - 1e-12)
    ps_count = np.sum(ps)
    if ps_count < 2:
        return h
    for _ in range(n_modes):
        m = rng.randint(1, n // 4 + 1)
        phase = rng.uniform(0, 2 * np.pi)
        noise = amp * np.sin(2 * np.pi * m * x / n + phase)
        h[ps] += noise[ps]
    h = np.clip(h, 0.0, 1.0)
    return project_to_sum(h, n / 2.0)

def smooth_max_obj(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    obj = m_val + np.log(s) / alpha
    return obj, corr, exp_vals / s

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m_val))
    w = exp_vals / np.sum(exp_vals)
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    return obj, grad

def slsqp_local(h0, n, target_sum, alpha, maxiter):
    def f_and_g(h):
        return smooth_max_grad(h, alpha)
    def f(h):
        return f_and_g(h)[0]
    def g(h):
        return f_and_g(h)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def interp_to_res(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, np.concatenate([h_old, h_old[:1]]), period=2.0)
    return project_to_sum(h_new, n_new / 2.0)

def block_start(n, rng, n_blocks):
    h = np.zeros(n)
    seg = n // (2 * n_blocks)
    for i in range(n_blocks):
        start = i * 2 * seg + rng.randint(0, max(1, seg))
        h[start:min(start + seg, n)] = 1.0
    return project_to_sum(h, n / 2.0)

def sinusoidal_start(n, rng):
    x = np.linspace(0, 2, n, endpoint=False)
    a0 = 0.5
    m1 = rng.choice([1, 2, 3])
    a1 = rng.uniform(0.1, 0.3)
    p1 = rng.uniform(0, 2 * np.pi)
    h = a0 + a1 * np.sin(m1 * np.pi * x + p1)
    h = np.clip(h, 0.0, 1.0)
    return project_to_sum(h, n / 2.0)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float).copy()
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 240
        best_h = project_to_sum(sinusoidal_start(best_n, rng), best_n / 2.0)
        best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t_start

    def remaining():
        return budget_s - elapsed()

    # Phase 1: Multi-resolution cascade (25% budget)
    t_end1 = t_start + 0.25 * budget_s
    res_cascade = []
    current_n = best_n
    for n_tr in [80, 120, 160, 200, 240, 280, 320, 360, 400, 440, 480]:
        if time.time() >= t_end1:
            break
        ts_tr = n_tr / 2.0
        if abs(ts_tr - round(ts_tr)) > 1e-6:
            continue
        if current_n > 0:
            h_tr = interp_to_res(best_h, n_tr)
        else:
            h_tr = project_to_sum(rng.uniform(0, 1, n_tr), ts_tr)
        corr_tr = np.correlate(h_tr, 1.0 - h_tr, mode="full")
        max_sw = min(40, max(3, int(remaining() * 0.15 / 1.5)))
        for _ in range(max_sw):
            if time.time() >= t_end1 - 5:
                break
            h_tr, corr_tr, _ = pair_descent_sweep(h_tr, corr_tr, rng, n_cand=10)
        c5_tr = compute_c5(h_tr)
        if c5_tr < best_c5:
            best_c5 = c5_tr
            best_h = h_tr.copy()
            best_n = n_tr
        res_cascade.append((n_tr, h_tr.copy()))

    # Phase 2: Intensive pair descent on best resolution (45% budget)
    t_end2 = t_start + 0.70 * budget_s
    n_work = best_n
    h_work = best_h.copy()
    corr_work = np.correlate(h_work, 1.0 - h_work, mode="full")
    stall = 0
    sweeps_done = 0
    while time.time() < t_end2:
        temp = 0.02 * max(0, 1.0 - sweeps_done / 300.0)
        h_work, corr_work, imp = pair_descent_sweep(h_work, corr_work, rng, n_cand=12, temp=temp)
        sweeps_done += 1
        c5_w = np.max(corr_work) * (2.0 / n_work)
        if c5_w < best_c5 - 1e-10:
            best_c5 = c5_w
            best_h = h_work.copy()
            best_n = n_work
            stall = 0
        else:
            stall += 1
        if stall == 50:
            h_work = fourier_perturb(h_work, rng, n_modes=3, amp=0.05)
            corr_work = np.correlate(h_work, 1.0 - h_work, mode="full")
            stall = 0
        elif stall == 150:
            h_work = fourier_perturb(h_work, rng, n_modes=5, amp=0.10)
            corr_work = np.correlate(h_work, 1.0 - h_work, mode="full")
            stall = 0
        elif stall == 300:
            h_work = project_to_sum(h_work + rng.uniform(-0.15, 0.15, n_work), n_work / 2.0)
            corr_work = np.correlate(h_work, 1.0 - h_work, mode="full")
            stall = 0
        elif stall >= 500:
            h_work = block_start(n_work, rng, 3 + rng.randint(0, 3))
            corr_work = np.correlate(h_work, 1.0 - h_work, mode="full")
            stall = 0

    # Phase 3: Try alternate resolutions with best solution as seed (15% budget)
    t_end3 = t_start + 0.85 * budget_s
    alt_res = [int(best_n * 0.75), int(best_n * 0.9), int(best_n * 1.15), int(best_n * 1.3)]
    for n_alt in alt_res:
        if time.time() >= t_end3:
            break
        n_alt = max(80, min(560, n_alt))
        ts_alt = n_alt / 2.0
        if abs(ts_alt - round(ts_alt)) > 1e-6:
            n_alt = int(round(ts_alt)) * 2
            ts_alt = n_alt / 2.0
        h_alt = interp_to_res(best_h, n_alt)
        corr_alt = np.correlate(h_alt, 1.0 - h_alt, mode="full")
        max_sw = min(50, max(5, int(remaining() * 0.12 / 1.5)))
        for _ in range(max_sw):
            if time.time() >= t_end3:
                break
            h_alt, corr_alt, _ = pair_descent_sweep(h_alt, corr_alt, rng, n_cand=10)
        c5_alt = compute_c5(h_alt)
        if c5_alt < best_c5:
            best_c5 = c5_alt
            best_h = h_alt.copy()
            best_n = n_alt

    # Phase 4: SLSQP polish with soft-max (10% budget)
    t_end4 = t_start + 0.95 * budget_s
    n_f = best_n
    ts_f = n_f / 2.0
    for alpha in [50, 200, 800]:
        if time.time() >= t_end4:
            break
        h_pol = slsqp_local(best_h, n_f, ts_f, alpha, maxiter=800)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5 - 1e-12:
            best_c5 = c5_pol
            best_h = h_pol.copy()
            best_n = n_f

    # Phase 5: Final pair descent polish (5% budget)
    h_fin = best_h.copy()
    n_fin = best_n
    corr_fin = np.correlate(h_fin, 1.0 - h_fin, mode="full")
    while time.time() < t_start + budget_s - 5:
        h_fin, corr_fin, _ = pair_descent_sweep(h_fin, corr_fin, rng, n_cand=14)
        c5_fin = np.max(corr_fin) * (2.0 / n_fin)
        if c5_fin < best_c5 - 1e-12:
            best_c5 = c5_fin
            best_h = h_fin.copy()
            best_n = n_fin

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
