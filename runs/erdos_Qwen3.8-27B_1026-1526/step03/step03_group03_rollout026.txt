Wait, the bisection on the subgradient is not quite right. f(δ) = max_l (A[l] + D[l]*δ). The subgradient at δ is {D[l] : l ∈ argmax}. At the optimum, 0 should be in the convex hull of the subgradient, which means there should be both positive and negative D values at the argmax (or a single D=0).

A better approach: f is convex piecewise linear. Its minimum is at a point where the slope changes from negative to positive. The slope at δ is D[argmax(A + D*δ)] if unique, or any value in [min D[l], max D[l]] for l in argmax.

For bisection:
- If f'(δ) > 0 (i.e., D[argmax] > 0 when unique), we need to go left
- If f'(δ) < 0, go right

But when there are ties, the subgradient is an interval. If 0 is in that interval, we're at the optimum.

A simpler approach that works well in practice: just use the original grid + bisection approach, but with more iterations.

        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_B(h, j, n):
    nm1 = n - 1
    B = np.zeros(2 * n - 1)
    k_start = max(0, nm1 - j)
    k_end = min(2 * n - 2, 2 * nm1 - j)
    if k_start <= k_end:
        B[k_start:k_end + 1] = (1.0 - h[j - nm1 + np.arange(k_start, k_end + 1)])
    k_start2 = max(0, j)
    k_end2 = min(2 * n - 2, j + nm1)
    if k_start2 <= k_end2:
        B[k_start2:k_end2 + 1] += -h[j + nm1 - np.arange(k_start2, k_end2 + 1)]
    return B

def solve_1d(A, D, lo, hi):
    if hi - lo < 1e-14:
        return (lo + hi) * 0.5
    t_grid = np.linspace(lo, hi, 12)
    vals = np.max(A[None, :] + D[None, :] * t_grid[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        return lo
    if best_idx == len(t_grid) - 1:
        return hi
    lo_r = t_grid[best_idx - 1]
    hi_r = t_grid[best_idx + 1]
    for _ in range(35):
        mid = (lo_r + hi_r) * 0.5
        if np.max(A + D * mid) <= np.max(A + D * lo_r):
            lo_r = mid
        else:
            hi_r = mid
    return (lo_r + hi_r) * 0.5

def pair_sweep(h, corr, rng, n_cand, temperature=0.0):
    n = len(h)
    order = np.arange(n)
    rng.shuffle(order)
    n_improved = 0
    n_accepted_worse = 0
    
    for j in order:
        B_j = compute_B(h, j, n)
        
        # Generate candidates
        k_candidates = [(j + 1) % n, (j - 1 + n) % n, (j + n//2) % n,
                       (j + n//3) % n, (j + 2*n//3) % n,
                       (j + 2) % n, (j + 3) % n, (j + n//4) % n,
                       (j + 3*n//4) % n]
        for _ in range(max(0, n_cand - len(k_candidates))):
            k_candidates.append(rng.randint(n))
        
        seen = set()
        best_delta = 0.0
        best_D = None
        best_k = -1
        best_val = np.max(corr)
        
        for k in k_candidates:
            k = int(k)
            if k == j or k in seen:
                continue
            seen.add(k)
            B_k = compute_B(h, k, n)
            D = B_j - B_k
            delta_lo = max(-h[j], -1.0 + h[k])
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-13:
                continue
            delta_opt = solve_1d(corr, D, delta_lo, delta_hi)
            val_opt = np.max(corr + D * delta_opt)
            if val_opt < best_val - 1e-14:
                best_val = val_opt
                best_delta = delta_opt
                best_D = D
                best_k = k
        
        if best_k >= 0 and abs(best_delta) > 1e-13:
            new_max = np.max(corr + best_delta * best_D)
            old_max = np.max(corr)
            accept = (new_max <= old_max + 1e-14)
            if not accept and temperature > 0:
                delta_obj = (new_max - old_max) / (2.0 / n)  # Convert to C5 units
                if delta_obj < 0.01:  # Only consider small worsening
                    accept = (rng.random() < np.exp(-delta_obj / temperature))
            if accept:
                corr = corr + best_delta * best_D
                h[j] = np.clip(h[j] + best_delta, 0.0, 1.0)
                h[best_k] = np.clip(h[best_k] - best_delta, 0.0, 1.0)
                if new_max < old_max - 1e-12:
                    n_improved += 1
                else:
                    n_accepted_worse += 1
    
    return h, corr, n_improved > 0

def fourier_move(h, rng, alpha_max=0.05):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for m in range(1, 8):
        freq = 2.0 * np.pi * m / n
        for _ in range(3):
            phase = rng.uniform(0, 2 * np.pi)
            direction = np.cos(freq * np.arange(n) + phase)
            direction -= np.mean(direction)
            
            pos_mask = direction > 1e-10
            neg_mask = direction < -1e-10
            A_max = alpha_max
            if np.any(pos_mask):
                A_max = min(A_max, np.min((1.0 - h[pos_mask]) / direction[pos_mask]) * 0.9)
            if np.any(neg_mask):
                A_max = min(A_max, np.min(h[neg_mask] / (-direction[neg_mask])) * 0.9)
            
            if A_max < 1e-10:
                continue
            
            for A_frac in [0.3, 0.6, 1.0]:
                A = A_frac * A_max
                h_trial = h + A * direction
                h_trial = project_to_sum(h_trial, target)
                c5_trial = compute_c5(h_trial)
                if c5_trial < best_c5:
                    best_c5 = c5_trial
                    best_h = h_trial.copy()
    
    return best_h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_B(h, j, n):
    nm1 = n - 1
    B = np.zeros(2 * n - 1)
    k_start = max(0, nm1 - j)
    k_end = min(2 * n - 2, 2 * nm1 - j)
    if k_start <= k_end:
        B[k_start:k_end + 1] = (1.0 - h[j - nm1 + np.arange(k_start, k_end + 1)])
    k_start2 = max(0, j)
    k_end2 = min(2 * n - 2, j + nm1)
    if k_start2 <= k_end2:
        B[k_start2:k_end2 + 1] += -h[j + nm1 - np.arange(k_start2, k_end2 + 1)]
    return B

def solve_1d(A, D, lo, hi):
    if hi - lo < 1e-14:
        return (lo + hi) * 0.5
    t_grid = np.linspace(lo, hi, 12)
    vals = np.max(A[None, :] + D[None, :] * t_grid[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        return lo
    if best_idx == len(t_grid) - 1:
        return hi
    lo_r = t_grid[best_idx - 1]
    hi_r = t_grid[best_idx + 1]
    for _ in range(40):
        mid = (lo_r + hi_r) * 0.5
        if np.max(A + D * mid) <= np.max(A + D * lo_r):
            lo_r = mid
        else:
            hi_r = mid
    return (lo_r + hi_r) * 0.5

def gradient_at_max(h, corr):
    n = len(h)
    m_star = int(np.argmax(corr))
    lag = m_star - (n - 1)
    grad = np.zeros(n)
    jp = np.arange(n) + lag
    jm = np.arange(n) - lag
    valid_p = (jp >= 0) & (jp < n)
    valid_m = (jm >= 0) & (jm < n)
    grad[valid_p] += (1.0 - h[jp[valid_p]])
    grad[valid_m] -= h[jm[valid_m]]
    return grad

def smooth_grad(h, corr, alpha):
    n = len(h)
    cmax = np.max(corr)
    w = np.exp(alpha * (corr - cmax))
    w_sum = np.sum(w)
    if w_sum < 1e-30:
        return gradient_at_max(h, corr)
    w = w / w_sum
    grad = np.zeros(n)
    L_values = np.arange(-(n - 1), n)
    for i, L in enumerate(L_values):
        wi = w[n - 1 + i]
        if wi < 1e-12:
            continue
        jp = np.arange(n) + L
        jm = np.arange(n) - L
        valid_p = (jp >= 0) & (jp < n)
        valid_m = (jm >= 0) & (jm < n)
        grad[valid_p] += wi * (1.0 - h[jp[valid_p]])
        grad[valid_m] -= wi * h[jm[valid_m]]
    return grad

def pair_sweep(h, corr, rng, n_cand, temperature=0.0, use_grad_order=False):
    n = len(h)
    if use_grad_order:
        grad = gradient_at_max(h, corr)
        order = np.argsort(-np.abs(grad))
        idx = np.arange(n)
        for i in range(n - 1, 0, -1):
            j = rng.randint(i + 1)
            order[i], order[j] = order[j], order[i]
    else:
        order = np.arange(n)
        rng.shuffle(order)
    n_improved = 0
    for j in order:
        B_j = compute_B(h, j, n)
        k_candidates = [(j + 1) % n, (j - 1 + n) % n, (j + n // 2) % n,
                       (j + n // 3) % n, (j + 2 * n // 3) % n,
                       (j + 2) % n, (j + 3) % n, (j + n // 4) % n,
                       (j + 3 * n // 4) % n, (j + 4) % n]
        for _ in range(max(0, n_cand - len(k_candidates))):
            k_candidates.append(rng.randint(n))
        seen = set()
        best_delta = 0.0
        best_D = None
        best_k = -1
        best_val = np.max(corr)
        for k in k_candidates:
            k = int(k)
            if k == j or k in seen:
                continue
            seen.add(k)
            B_k = compute_B(h, k, n)
            D = B_j - B_k
            delta_lo = max(-h[j], -1.0 + h[k])
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-13:
                continue

                best_val = val_opt
                best_delta = delta_opt
                best_D = D
                best_k = k

            new_max = best_val
            old_max = np.max(corr)
            accept = (new_max <= old_max + 1e-14)

                if new_max < old_max - 1e-12:
                    n_improved += 1
    return h, corr, n_improved > 0

def fourier_perturb(h, rng, alpha_max=0.04):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for m in range(1, 10):
        freq = 2.0 * np.pi * m / n
        for _ in range(4):
            phase = rng.uniform(0, 2 * np.pi)
            direction = np.cos(freq * np.arange(n) + phase)
            direction -= np.mean(direction)
            dn = np.sqrt(np.sum(direction ** 2))
            if dn < 1e-10:
                continue
            direction /= dn
            pos_mask = direction > 1e-10
            neg_mask = direction < -1e-10
            A_cap = alpha_max
            if np.any(pos_mask):
                A_cap = min(A_cap, 0.9 * np.min((1.0 - h[pos_mask]) / direction[pos_mask]))
            if np.any(neg_mask):
                A_cap = min(A_cap, 0.9 * np.min(h[neg_mask] / (-direction[neg_mask])))
            if A_cap < 1e-10:

                A = A_frac * A_cap
                h_trial = h + A * direction

                c5_trial = compute_c5(h_trial)
                if c5_trial < best_c5:
                    best_c5 = c5_trial

    grad = smooth_grad(h, corr, smooth_alpha)
    h_new = h - alpha * grad
    h_new = project_to_sum(h_new, n / 2.0)
