    # But only use where valid
    delta += valid1 * h1
    
    # Term 2: a=j, need 0<=j+m<n
    idx2 = j + m_vals
    valid2 = (idx2 >= 0) & (idx2 < n)
    h2 = np.where(idx2 == i, 0.0, np.where(idx2 == j, 1.0, h[np.clip(idx2, 0, n-1)]))
    delta -= valid2 * h2
    
    # Term 3: a=i-m, need 0<=i-m<n
    idx3 = i - m_vals
    valid3 = (idx3 >= 0) & (idx3 < n)
    # delta contribution: -(1-h[idx3]) but with modifications
    # if idx3==i (m=0): old=0, new=0, delta=0
    # if idx3==j: old=1, new=0, delta=-1
    # else: delta = -(1-h[idx3])
    h3 = np.where(idx3 == i, 0.0, np.where(idx3 == j, 1.0, h[np.clip(idx3, 0, n-1)]))
    # delta3 = -(1-h[idx3]) for general case, but 0 if idx3==i, -1 if idx3==j
    delta3 = -(1.0 - h[np.clip(idx3, 0, n-1)])
    delta3 = np.where(idx3 == i, 0.0, delta3)
    delta += valid3 * delta3
    
    # Term 4: a=j-m, need 0<=j-m<n
    idx4 = j - m_vals
    valid4 = (idx4 >= 0) & (idx4 < n)
    # delta contribution: 1-h'[idx4]
    # if idx4==j (m=0): 1-1=0
    # if idx4==i: 1-0=1
    # else: 1-h[idx4]
    h4_new = np.where(idx4 == i, 0.0, np.where(idx4 == j, 1.0, h[np.clip(idx4, 0, n-1)]))
    delta4 = 1.0 - h4_new
    delta4 = np.where(idx4 == j, 0.0, delta4)
    delta += valid4 * delta4
    
    return delta
