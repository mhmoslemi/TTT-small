import time

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):

    for _ in range(500):

        else:

        h = np.clip(h, 0.0, 1.0)

def active_gradient(h, corr, n_active=10, rel_tol=1e-3):
    """Compute gradient as average over active lags."""
    n = len(h)
    corr_max = np.max(corr)
    threshold = corr_max - rel_tol * corr_max
    active_idx = np.where(corr >= threshold)[0]
    if len(active_idx) == 0:
        active_idx = np.array([np.argmax(corr)])
    # Limit to n_active closest to max
    if len(active_idx) > n_active:
        sorted_active = active_idx[np.argsort(corr[active_idx])[::-1][:n_active]]
    else:
        sorted_active = active_idx
    g = np.zeros(n)
    for idx in sorted_active:
        lag = int(idx - (n - 1))
        g += corr_gradient(h, lag)
    g /= len(sorted_active)
    return g

def corr_gradient(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        g[lag:] += (1.0 - h[:n - lag])
        g[:n - lag] -= h[lag:]
    else:
        k = -lag
        g[:n - k] += (1.0 - h[k:])
        g[k:] -= h[:n - k]
    return g

def smooth_max_obj_and_grad(h, alpha, penalty):
    """Smooth max objective with sum constraint penalty."""
    corr = compute_corr(h)
    corr_max = np.max(corr)
    shifted = alpha * (corr - corr_max)
    shifted -= np.max(shifted)  # numerical stability
    exp_vals = np.exp(shifted)
    log_sum_exp = np.log(np.sum(exp_vals))
    obj = (corr_max + log_sum_exp / alpha) * (2.0 / len(h))
    # Sum constraint penalty
    s = np.sum(h)
    target = len(h) / 2.0
    obj += penalty * (s - target) ** 2
    # Gradient
    w = exp_vals / np.sum(exp_vals)
    n = len(h)
    g = np.zeros(n)
    for idx in range(len(corr)):
        if w[idx] < 1e-10:
            continue
        lag = idx - (n - 1)
        g += w[idx] * corr_gradient(h, lag)
    g *= (2.0 / n)
    g += 2.0 * penalty * (s - target) * np.ones(n)
    return obj, g
