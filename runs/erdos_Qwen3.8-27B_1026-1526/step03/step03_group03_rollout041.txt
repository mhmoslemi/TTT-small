def frank_wolfe_step(h, corr, B, t_alpha=2.0):
    n = len(h)
    m = len(corr)
    # Compute weights (soft max)
    alpha = 50.0
    m_val = np.max(corr)

    W = exp_vals / np.sum(exp_vals)
    
    # Gradient: grad[i] = (B^T @ W)[i]
