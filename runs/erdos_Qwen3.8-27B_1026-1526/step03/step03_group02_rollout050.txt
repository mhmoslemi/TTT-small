def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_lag_values(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

        if abs(diff) < 1e-13:
            break

        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)

    # wait, n=j-k must satisfy 0 <= j-k <= n-1-k, i.e., k <= j and j-k <= n-1-k, i.e., j <= n-1 (always true)
    # and j-k >= 0, i.e., k <= j

    # So: k in [0, j] (and k < n)

    # Before: (1-h[j])*h[j+k] = 1*h[j+k] = h[j+k] (since h[j]=0)
    # After: (1-h[j])*h[j+k] = 0*h[j+k] = 0 (since h[j]=1 now)
    # Delta: -h[j+k] for k in [0, n-1-j]
    
    # Change due to h[i]: 1→0
    # h[i] appears as h[n+k] when n=i-k (if k<=i<n)

    # After: (1-h[i-k])*h[i] = (1-h[i-k])*0 = 0
    # Delta: -(1-h[i-k]) for k in [0, i]

    # Before: (1-h[i])*h[i+k] = 0*h[i+k] = 0 (since h[i]=1)
    # After: (1-h[i])*h[i+k] = 1*h[i+k] = h[i+k] (since h[i]=0 now)
    # Delta: +h[i+k] for k in [0, n-1-i]
    
    # Total delta for lag k:
    # delta_k = [k<=j] * (1-h[j-k]) - [k<=n-1-j] * h[j+k] - [k<=i] * (1-h[i-k]) + [k<=n-1-i] * h[i+k]

    # After the swap: h[i]=0, h[j]=1.

    k_max = max(i, n-1-i, j, n-1-j)

        if k <= n-1-j and j+k < n:

        # From i: 1→0
        if k <= i:
            d -= (1.0 - h[i-k])
        if k <= n-1-i and i+k < n:
            d += h[i+k]
        deltas[k] = d
    
    return deltas, k_max
