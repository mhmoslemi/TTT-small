       """Project v onto {0 <= x <= 1, sum(x) = target_sum}."""
       # Find lambda such that sum(clip(v_i + lambda, 0, 1)) = target_sum
       lo, hi = -max(v) - 1, -min(v) + 1  # wide bounds
       for _ in range(60):
           mid = (lo + hi) / 2
           proj = np.clip(v + mid, 0, 1)
           s = np.sum(proj)
           if s > target_sum:
               lo = mid
           else:
               hi = mid
