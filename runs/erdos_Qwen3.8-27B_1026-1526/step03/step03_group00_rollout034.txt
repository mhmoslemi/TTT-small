def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def round_to_binary(h, target_ones):
    n = len(h)
    h_bin = (h >= 0.5).astype(float)
    n_ones = int(h_bin.sum())
    if n_ones > target_ones:
        ones_idx = np.where(h_bin > 0.5)[0]
        # Remove the ones with lowest h values (closest to 0.5 from below)
        scores = h[ones_idx]
        remove_idx = ones_idx[np.argsort(scores)[:n_ones - target_ones]]
        h_bin[remove_idx] = 0
    elif n_ones < target_ones:
        zeros_idx = np.where(h_bin < 0.5)[0]
        # Add the zeros with highest h values (closest to 0.5 from above)
        scores = h[zeros_idx]
        add_idx = zeros_idx[np.argsort(scores)[::-1][:target_ones - n_ones]]
        h_bin[add_idx] = 1
    return h_bin

def binary_sa(h_init, rng, max_iters, t_start_temp, t_stop_temp, time_limit):
    n = len(h_init)
    target_ones = n // 2
    h = round_to_binary(h_init, target_ones) if not np.all((h_init == 0) | (h_init == 1)) else h_init.copy()
    
    ones_list = list(np.where(h > 0.5)[0])
    zeros_list = list(np.where(h < 0.5)[0])
    
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    stale_count = 0
    
    t_deadline = time.time() + time_limit
    
    for it in range(max_iters):
        if time.time() > t_deadline:
            break
        
        frac = min(1.0, it / max(1, max_iters))
        T = t_start_temp * (t_stop_temp / t_start_temp) ** frac
        
        # Random swap
        i = ones_list[rng.randint(len(ones_list))]
        j = zeros_list[rng.randint(len(zeros_list))]
        
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h)
        
        delta = new_c5 - current_c5
        accept = delta <= 0
        if not accept and T > 1e-15:
            if delta < T * 5:  # avoid overflow in exp

                best_h = h.copy()
                stale_count = 0
            else:
                stale_count += 1
        else:
            h[i] = 1.0
            h[j] = 0.0
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def round_to_binary(h, target_ones):
    h_bin = (h >= 0.5).astype(float)
    n_ones = int(h_bin.sum())
    if n_ones > target_ones:
        ones_idx = np.where(h_bin > 0.5)[0]
        scores = h[ones_idx]
        remove_idx = ones_idx[np.argsort(scores)[:n_ones - target_ones]]
        h_bin[remove_idx] = 0.0
    elif n_ones < target_ones:
        zeros_idx = np.where(h_bin < 0.5)[0]
        scores = h[zeros_idx]
        add_idx = zeros_idx[np.argsort(scores)[::-1][:target_ones - n_ones]]
        h_bin[add_idx] = 1.0
    return h_bin

def make_block_pattern(n, block_len, rng):
    target = n // 2
    h = np.zeros(n)
    period = 2 * block_len
    shift = rng.randint(0, period)

        if pos < block_len:
            h[i] = 1.0
    n_ones = int(h.sum())
    if n_ones > target:
        mask = h > 0.5
        h[mask] = (h[mask] - 0.3)
        h_bin = round_to_binary(h, target)
        return h_bin
    elif n_ones < target:
        mask = h < 0.5
        h[mask] = (h[mask] + 0.3)
        h_bin = round_to_binary(h, target)
        return h_bin
    return h

def make_random_binary(n, rng):
    target = n // 2
    perm = rng.permutation(n)
    h = np.zeros(n)
    h[perm[:target]] = 1.0
    return h

def make_interleaved_pattern(n, rng, gap):
    target = n // 2
    h = np.zeros(n)
    positions = []
    i = 0
    while len(positions) < target and i < n:
        positions.append(i)
        i += gap
    if len(positions) < target:

        available = [j for j in range(n) if j not in set(positions)]
        rng.shuffle(available)
        positions.extend(available[:remaining])
    h[np.array(positions[:target])] = 1.0
    return h

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha

    grad = np.zeros(n)
    for k in range(-n + 1, n):
        if w[n - 1 + k] < 1e-12:
            continue
        wk = w[n - 1 + k]
        if k >= 0:
            valid = slice(k, n)
            valid2 = slice(0, n - k)
            grad[k:] += wk * (1.0 - h[:n - k])
            grad[:n - k] -= wk * h[k:]
        else:
            kk = -k
            grad[:n - kk] += wk * (1.0 - h[kk:])
            grad[kk:] -= wk * h[:n - kk]
    return obj, grad

def slsqp_local(h0, n, target_sum, alpha, maxiter):
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n

    def f(x):
        return smooth_max_and_grad(x, alpha)[0]

    def g(x):
        return smooth_max_and_grad(x, alpha)[1]

    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

            h_shake_bin = round_to_binary(h_shake, len(best_h) // 2)
            h_res, c5_res = binary_sa(h_shake_bin, rng, min(30, time_left() * 0.3),
                                       t_init=0.003, t_final=0.0002, p_reverse=0.1)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = h_res.copy()

    if time_left() > 30:
        h_final = best_h.copy()
        ts_final = len(h_final) / 2.0
        h_final = project_to_sum(h_final, ts_final)
        h_final = np.clip(h_final, 0.0, 1.0)
        best_c5 = compute_c5(h_final)
        best_h = h_final
        best_n = len(h_final)

    return (best_h, best_c5, best_n)
