    active = np.where(overlaps >= threshold)[0]
    if len(active) < n_top:
        active = np.argsort(overlaps)[-n_top:]
    return active

def slsqp_solve(h0, n, ts, alpha, maxiter=800, active_idx=None):
    def f(x):
        if active_idx is not None:
            return smooth_max_obj_grad_active(x, alpha, active_idx)[0]
        return smooth_max_obj_grad_full(x, alpha)[0]

    def g(x):
        if active_idx is not None:
            return smooth_max_obj_grad_active(x, alpha, active_idx)[1]
        return smooth_max_obj_grad_full(x, alpha)[1]

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ts, 'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(f, h0.copy(), method='SLSQP', jac=g, bounds=bnds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14})
        return project_to_sum(r.x, ts)
    except Exception:
        return project_to_sum(h0, ts)

def proj_grad_nesterov(h0, n, ts, alpha, n_iter=8000, lr_init=0.005, momentum=0.9):
    h = h0.copy()
    y = h0.copy()
    v = np.zeros(n)
    lr = lr_init
    beta = momentum
    obj_prev, _ = smooth_max_obj_grad_full(h, alpha)
    best_h = h.copy()
    best_obj = obj_prev
    stall = 0

    for i in range(n_iter):
        _, grad = smooth_max_obj_grad_full(h, alpha)
        v = beta * v - lr * grad
        y = h + v
        h_new = project_to_sum(y, ts)
        obj_new, _ = smooth_max_obj_grad_full(h_new, alpha)

        if obj_new < obj_prev - 1e-16:
            h = h_new
            obj_prev = obj_new
            if obj_new < best_obj:
                best_obj = obj_new
                best_h = h.copy()
            stall = 0
            lr = min(lr * 1.03, 0.05)
        else:
            v = beta * v
            lr *= 0.5
            stall += 1
            if stall > 300 or lr < 1e-12:
                break

    return best_h

def proj_grad_active(h0, n, ts, alpha, n_iter=5000, lr_init=0.01):
    h = h0.copy()
    lr = lr_init
    obj_prev, _ = smooth_max_obj_grad_full(h, alpha)
    best_h = h.copy()
    best_obj=obj_prev
    stall = 0
    active = find_active_set(h, n_top=6)

    for i in range(n_iter):
        if i % 200 == 0:
            active = find_active_set(h, n_top=6)
        _, grad = smooth_max_obj_grad_active(h, alpha, active)
        h_new = project_to_sum(h - lr * grad, ts)
        obj_new, _ = smooth_max_obj_grad_full(h_new, alpha)
        if obj_new < obj_prev - 1e-16:
            h = h_new
            obj_prev = obj_new
            if obj_new < best_obj:
                best_obj = obj_new
                best_h = h.copy()
            stall = 0
            lr = min(lr * 1.02, 0.05)
        else:
            lr *= 0.5
            stall += 1
            if stall > 200 or lr < 1e-12:
                break

    return best_h

def stochastic_cd(h, n, ts, rng, n_steps=40000, delta_range=(0.01, 0.25), temp_init=0.15):
    h = h.copy()
    c5_cur = compute_c5(h)
    temp = temp_init
    for step in range(n_steps):
        if step % 100 == 0:
            temp = max(1e-6, temp_init * np.exp(-step / (n_steps * 0.3)))
        ones_idx = np.where(h > 0.02)[0]
        zeros_idx = np.where(h < 0.98)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        if i == j:
            continue
        delta = rng.uniform(delta_range[0], delta_range[1])
        if h[i] < delta:
            delta = h[i] * 0.4
        if 1.0 - h[j] < delta:
            delta = (1.0 - h[j]) * 0.4
        if delta < 1e-5:
            continue
        h[i] -= delta
        h[j] += delta
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, ts)
        c5_new = compute_c5(h)
        if c5_new < c5_cur:
            c5_cur = c5_new
        else:
            accept_prob = np.exp(-(c5_new - c5_cur) / temp)
            if rng.random() < accept_prob:
                c5_cur = c5_new
            else:
                h[i] += delta
                h[j] -= delta
                h = np.clip(h, 0.0, 1.0)
                h = project_to_sum(h, ts)
    return h, c5_cur

def block_cd(h, n, ts, rng, n_steps=10000, block_size=8):
    h = h.copy()
    c5_cur = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_cur
    for step in range(n_steps):
        start = rng.randint(0, n - block_size)
        mask = np.zeros(n, dtype=bool)
        mask[start:start + block_size] = True
        perturbation = rng.normal(0, 0.05, block_size)
        h_trial = h.copy()
        h_trial[start:start + block_size] += perturbation
        h_trial = project_to_sum(h_trial, ts)
        c5_trial = compute_c5(h_trial)
        if c5_trial < best_c5:
            h = h_trial
            best_h = h.copy()
            best_c5 = c5_trial
    return best_h, best_c5

def upsample(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old, period=2.0)
    return h_new

def multi_alpha_optimize(h, n, ts, time_fn, min_time=15):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alphas = [30, 100, 500, 2000, 8000, 30000]
    for alpha in alphas:
        if time_fn() < min_time:
            break
        active = find_active_set(best_h, n_top=6)
        h_new = slsqp_solve(best_h, n, ts, alpha, maxiter=600, active_idx=active)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    return best_h, best_c5

def nesterov_phase(h, n, ts, time_fn, min_time=20):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for alpha in [3000, 10000, 50000]:
        if time_fn() < min_time:
            break
        h_new = proj_grad_nesterov(best_h, n, ts, alpha, n_iter=5000, lr_init=0.003, momentum=0.85)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    return best_h, best_c5

def active_refine(h, n, ts, time_fn, min_time=15):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for alpha in [5000, 20000, 100000]:
        if time_fn() < min_time:
            break
        h_new = proj_grad_active(best_h, n, ts, alpha, n_iter=3000, lr_init=0.005)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()

    def time_left():
        return budget_s - (time.time() - t_start)

    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        if 'initial_h_values' in globals() and initial_h_values is not None:
            best_h = np.array(initial_h_values, dtype=float)
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
    except Exception:
        pass

    if best_h is None:
        best_n = 480
        best_h = np.full(best_n, 0.5)
        best_c5 = compute_c5(best_h)

    # Phase 1: Optimize at original resolution with multi-alpha
    n0 = best_n
    ts0 = n0 / 2.0
    if time_left() > 60:
        best_h, best_c5 = multi_alpha_optimize(best_h, n0, ts0, time_left, min_time=20)

    # Phase 2: Nesterov projected gradient at original resolution
    if time_left() > 50:
        best_h, best_c5 = nesterov_phase(best_h, n0, ts0, time_left, min_time=20)

    # Phase 3: Try higher resolutions
    target_ns = [512, 544, 576, 640, 720]
    for n_tr in target_ns:
        if time_left() < 100:
            break
        ts_tr = n_tr / 2.0
        h_up = upsample(best_h, n_tr)
        h_up = project_to_sum(h_up, ts_tr)
        c5_up = compute_c5(h_up)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up.copy()
            best_n = n_tr
        else:
            best_h = h_up.copy()
            best_n = n_tr

        # Optimize at this resolution
        if time_left() > 80:
            best_h, best_c5 = multi_alpha_optimize(best_h, best_n, best_n / 2.0, time_left, min_time=30)

        if time_left() > 60:
            best_h, best_c5 = nesterov_phase(best_h, best_n, best_n / 2.0, time_left, min_time=25)

        if time_left() > 50:
            best_h, best_c5 = active_refine(best_h, best_n, best_n / 2.0, time_left, min_time=20)

        if time_left() < 80:
            break

    # Phase 4: Go back to the best resolution and do intensive optimization
    # Determine the best resolution
    # Re-evaluate at current best
    n_best = best_n
    ts_best = n_best / 2.0

    # Phase 5: Stochastic CD
    if time_left() > 80:
        h_cd, c5_cd = stochastic_cd(best_h, n_best, ts_best, rng, n_steps=60000,
                                     delta_range=(0.01, 0.3), temp_init=0.1)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 6: Block coordinate descent
    if time_left() > 60:
        h_bl, c5_bl = block_cd(best_h, n_best, ts_best, rng, n_steps=8000, block_size=6)
        if c5_bl < best_c5:
            best_c5 = c5_bl
            best_h = h_bl.copy()

    # Phase 7: Final intensive optimization at best resolution
    if time_left() > 50:
        best_h, best_c5 = multi_alpha_optimize(best_h, n_best, ts_best, time_left, min_time=20)

    if time_left() > 40:
        best_h, best_c5 = nesterov_phase(best_h, n_best, ts_best, time_left, min_time=15)

    if time_left() > 30:
        best_h, best_c5 = active_refine(best_h, n_best, ts_best, time_left, min_time=15)

    # Phase 8: Tiny stochastic refinement
    if time_left() > 40:
        h_t, c5_t = stochastic_cd(best_h, n_best, ts_best, rng, n_steps=30000,
                                   delta_range=(0.003, 0.08), temp_init=0.05)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()

    # Phase 9: Final SLSQP polish with very high alpha
    if time_left() > 25:
        for alpha in [50000, 200000]:
            if time_left() < 15:
                break
            active = find_active_set(best_h, n_top=5)
            h_f = slsqp_solve(best_h, n_best, ts_best, alpha, maxiter=1000, active_idx=active)
            c5_f = compute_c5(h_f)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f.copy()

    # Phase 10: One more resolution check - try to find if a slightly different resolution helps
    if time_left() > 60:
        for n_try in [best_n - 32, best_n + 32, best_n + 64]:
            if n_try < 200 or n_try > 1000:
                continue
            if time_left() < 40:
                break
            ts_try = n_try / 2.0
            h_try = upsample(best_h, n_try)
            h_try = project_to_sum(h_try, ts_try)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = n_try
                ts_best = n_try / 2.0
                if time_left() > 30:
                    best_h, best_c5 = multi_alpha_optimize(best_h, best_n, ts_best, time_left, min_time=15)

    # Final validation
    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    for _ in range(300):
        diff = best_n / 2.0 - np.sum(best_h)
        if abs(diff) < 1e-14:
            break
        mask = (best_h > 1e-12) & (best_h < 1.0 - 1e-12)
        if np.any(mask):
            best_h[mask] += diff / len(mask)
            best_h = np.clip(best_h, 0.0, 1.0)
        else:
            break

    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
