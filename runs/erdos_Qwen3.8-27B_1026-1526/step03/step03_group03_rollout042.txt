import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_B_correct(h, j, n):
    """Compute the correct gradient B_j where B_j[m] = d(corr[m])/d(h[j])"""
    nm1 = n - 1
    B = np.zeros(2 * n - 1)
    lag = np.arange(2 * n - 1) - nm1
    first_idx = j - lag
    second_idx = j + lag
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    B[valid_f] += (1.0 - h[first_idx[valid_f]])
    B[valid_s] -= h[second_idx[valid_s]]
    return B

def solve_1d(A, D, lo, hi):
    """Find t in [lo, hi] that minimizes max(A + D*t)"""
    if hi - lo < 1e-14:
        return (lo + hi) * 0.5
    t_grid = np.linspace(lo, hi, 16)
    vals = np.max(A + D[None, :] * t_grid[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        return lo
    if best_idx == len(t_grid) - 1:
        return hi
    lo_r = t_grid[best_idx - 1]
    hi_r = t_grid[best_idx + 1]
    for _ in range(40):
        mid = (lo_r + hi_r) * 0.5
        if np.max(A + D * mid) <= np.max(A + D * lo_r):
            lo_r = mid
        else:
            hi_r = mid

                best_delta = delta_opt
                best_D = D
                best_k = k
        if best_k >= 0 and abs(best_delta) > 1e-12:
            corr = corr + best_delta * best_D
            h[j] += best_delta
            h[best_k] -= best_delta
            h[j] = np.clip(h[j], 0.0, 1.0)
            h[best_k] = np.clip(h[best_k], 0.0, 1.0)
            improved = True
    return h, corr, improved

def smooth_max_obj_grad(h, alpha):
    """Compute smooth max of corr and its gradient."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m_val))
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    
    # Gradient: d(corr)/d(h[j]) weighted by w
    # corr[m] = sum_n h[n+lag](1-h[n]) where lag = m-(n-1)
    # d(corr[m])/d(h[j]) = (1-h[j-lag])*I[valid] - h[j+lag]*I[valid]
    # grad[j] = sum_m w[m] * d(corr[m])/d(h[j])
    #         = sum_m w[m] * [(1-h[j-lag])*I[0<=j-lag<n] - h[j+lag]*I[0<=j+lag<n]]
    #         = sum_{m: j-lag(m) in [0,n)} w[m]*(1-h[j-lag(m)]) - sum_{m: j+lag(m) in [0,n)} w[m]*h[j+lag(m)]
    
    # Let me compute this efficiently.
    # For the first term: sum over m where 0 <= j-(m-n+1) < n, i.e., j-n+1 <= m-n+1 <= j, i.e., j <= m <= j+n-1
    #   w[m] * (1 - h[j-(m-n+1)]) = w[m] * (1 - h[2j-m+n-1-j])... 
    #   Let t = j-(m-n+1) = j-m+n-1. As m ranges from max(0,j) to min(2n-2, j+n-1), t ranges from j to j-(j) = 0... 
    #   Actually t = j - (m - n + 1) = j - m + n - 1.
    #   When m = j: t = n-1. When m = j+n-1: t = 0.
    #   So t ranges from n-1 down to 0 (clamped).
    #   First term for j: sum_{m=max(0,j)}^{min(2n-2, j+n-1)} w[m] * (1-h[j-m+n-1])
    
    # Let me just do it directly with vectorization.
    nm1 = n - 1
    j_idx = np.arange(n)
    lag = np.arange(2 * n - 1) - nm1  # shape (2n-1,)
    
    # For each j, the first term sums w[m]*(1-h[j-lag[m]]) over valid m
    # j-lag[m] in [0,n) means lag[m] in [j-n+1, j], i.e., m-nm1 in [j-n+1, j], i.e., m in [j, j+n-1]
    # clamped to [0, 2n-2]
    
    # Let's use a different approach: for each m, contribute w[m]*(1-h[j-lag[m]]) to grad[j] if valid,
    # and -w[m]*h[j+lag[m]] to grad[j] if valid.
    
    # First term: for m = 0..2n-2, lag[m] = m-nm1
    #   Contributes to grad[j] for j = lag[m] + t where t = 0..n-1 (i.e., j-lag[m] = t)
    #   Contribution: w[m] * (1-h[t])
    #   So: for m, t in range: grad[lag[m]+t] += w[m]*(1-h[t])
    
    # Second term: for m = 0..2n-2, lag[m] = m-nm1
    #   Contributes to grad[j] for j = lag[m] + t where t = 0..n-1 (i.e., j+lag[m] = ... wait
    #   j+lag[m] in [0,n) means j = s - lag[m] where s = 0..n-1 (j+lag[m] = s)
    #   Contribution: -w[m] * h[s]
    #   So: for m, s in range: grad[s-lag[m]] += -w[m]*h[s] if 0 <= s-lag[m] < n
    
    # Let me use a matrix approach but it might be too slow for large n.
    # Alternative: use FFT or direct computation.
    
    # For n=240, 2n-1=479. The matrix would be 240x479 which is fine.
    # But let's do it more efficiently.
    
    # First term: grad[j] += sum_{m: j-n+1<=m<=j+n-1, 0<=m<2n-1} w[m]*(1-h[j-(m-n+1)])
    # Let t = j-(m-n+1), so m = j-t+n-1. As t goes 0 to n-1, m goes j+n-1 down to j.
    # First term: grad[j] += sum_{t=0}^{n-1} w[j-t+n-1]*(1-h[t]) (with clamping)
    
    # This is a cross-correlation! 
    # First term = correlate((1-h), w) at specific positions... 
    # Actually: sum_t (1-h[t]) * w[j-t+n-1] = (cross-correlation of (1-h) and w)[j+n-1 - (n-1)] 
    # Hmm, let me just use np.correlate.
    
    # sum_t (1-h[t]) * w[j+n-1-t] for t=0..n-1 (with w clamped to [0, 2n-2])
    # This is like a convolution.
    
    # Let me define: a = 1-h (length n), b = w (length 2n-1)
    # first_term[j] = sum_{t=0}^{n-1} a[t] * b[j+n-1-t] (where b is zero outside [0, 2n-2])
    # = (a cross-correlated with b)[j+n-1]... 
    
    # Actually, np.correlate(b, a, mode='full')[j+n-1] = sum_t b[j+n-1+t] * a[t]... no.
    # np.correlate(v, a)[k] = sum_n v[n+k] * a[n]
    # So np.correlate(w, 1-h)[k] = sum_n w[n+k] * (1-h[n])
    # For k = j+n-1-n+1 = j... hmm.
    
    # Let me just do it numerically with a loop or vectorized approach.
    # For n=240, a loop of 240 iterations each with O(n) work = O(n²) = 57600, which is fine.
    
    grad = np.zeros(n)
    for j in range(n):
        # First term: sum over m where 0 <= j-(m-n+1) < n
        # j-(m-n+1) >= 0 → m <= j+n-1
        # j-(m-n+1) < n → m > j-1 → m >= j
        # So m from max(0, j) to min(2n-2, j+n-1)
        m_lo = j  # since j >= 0
        m_hi = min(2 * n - 2, j + n - 1)
        if m_lo <= m_hi:
            t = j - (np.arange(m_lo, m_hi + 1) - nm1)  # = j - m + n - 1, should be in [0, n-1]
            grad[j] += np.sum(w[m_lo:m_hi+1] * (1.0 - h[t]))
        # Second term: sum over m where 0 <= j+(m-n+1) < n
        # j+(m-n+1) >= 0 → m >= n-1-j
        # j+(m-n+1) < n → m < 2n-1-j → m <= 2n-2-j
        # So m from max(0, n-1-j) to min(2n-2, 2n-2-j)
        m_lo2 = max(0, nm1 - j)
        m_hi2 = min(2 * n - 2, 2 * n - 2 - j)
        if m_lo2 <= m_hi2:
            s = j + (np.arange(m_lo2, m_hi2 + 1) - nm1)  # = j + m - n + 1, should be in [0, n-1]
            grad[j] -= np.sum(w[m_lo2:m_hi2+1] * h[s])
    
    return obj, grad

def slsqp_local(h0, n, target_sum, alpha, maxiter=500):
    h0 = h0.copy()
    
    def objective_and_grad(x):
        return smooth_max_obj_grad(x, alpha)
    
    def obj(x):
        return objective_and_grad(x)[0]
    
    def grad(x):
        return objective_and_grad(x)[1]
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0.copy(), method='SLSQP', jac=grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)
