def compute_c5(h):

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break

            h[mask] += diff / np.sum(mask)
        else:

def smooth_max_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    w = exp_v / np.sum(exp_v)
    obj = m_val + np.log(np.sum(exp_v)) / alpha
    
    # Gradient computation
    # Only compute for significant weights
    thresh = 1e-10
    active = np.where(w > thresh)[0]
    grad = np.zeros(n)
    shifts = active - (n - 1)

        s = shifts[idx]

def projected_gradient_method(h0, n, ts, alpha, n_iter, lr_init, momentum):
    h = h0.copy()
    v = np.zeros(n)  # momentum
    lr = lr_init
    obj_prev, _ = smooth_max_and_grad(h, alpha)
    
    for i in range(n_iter):

        v = momentum * v + (1 - momentum) * grad

        return smooth_max_and_grad(x, alpha)[0]
    def g(x):
        return smooth_max_and_grad(x, alpha)[1]
