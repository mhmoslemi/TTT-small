import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_B(h, j, n):
    nm1 = n - 1
    B = np.zeros(2 * n - 1)
    k_start = max(0, nm1 - j)
    k_end = min(2 * n - 2, 2 * nm1 - j)
    if k_start <= k_end:
        B[k_start:k_end + 1] += (1.0 - h[j - nm1 + np.arange(k_start, k_end + 1)])
    k_start2 = max(0, j)
    k_end2 = min(2 * n - 2, j + nm1)
    if k_start2 <= k_end2:
        B[k_start2:k_end2 + 1] += -h[j + nm1 - np.arange(k_start2, k_end2 + 1)]
    return B

def solve_1d(A, D, lo, hi):
    if hi - lo < 1e-14:
        return (lo + hi) * 0.5
    t_grid = np.linspace(lo, hi, 12)
    vals = np.max(A[None, :] + D[None, :] * t_grid[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        return lo
    if best_idx == len(t_grid) - 1:
        return hi
    lo_r = t_grid[best_idx - 1]
    hi_r = t_grid[best_idx + 1]
    for _ in range(32):
        mid = (lo_r + hi_r) * 0.5
        if np.max(A + D * mid) <= np.max(A + D * lo_r):
            lo_r = mid
        else:
            hi_r = mid
    return (lo_r + hi_r) * 0.5

def pair_descent_sweep(h, corr, rng, n_cand=10):
    n = len(h)
    order = np.arange(n)
    rng.shuffle(order)
    improved = False
    for j in order:
        B_j = compute_B(h, j, n)
        best_val = np.max(corr)
        best_delta = 0.0
        best_D = None
        best_k = -1
        k_set = [(j + 1) % n, (j - 1 + n) % n, (j + n // 2) % n, (j + n // 3) % n, (j + 2 * n // 3) % n]
        for _ in range(max(0, n_cand - 5)):
            k_set.append(rng.randint(n))
        seen = set()
        for k in k_set:
            if k == j or k in seen:
                continue
            seen.add(k)
            B_k = compute_B(h, k, n)
            D = B_j - B_k
            delta_lo = max(-h[j], h[k] - 1.0)
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-13:
                continue
            delta_opt = solve_1d(corr, D, delta_lo, delta_hi)
            val_opt = np.max(corr + D * delta_opt)
            if val_opt < best_val - 1e-13:
                best_val = val_opt
                best_delta = delta_opt
                best_D = D
                best_k = k
        if best_k >= 0 and abs(best_delta) > 1e-13:
            corr += best_delta * best_D
            h[j] += best_delta
            h[best_k] -= best_delta
            if h[j] < 0:
                h[j] = 0.0
            elif h[j] > 1:
                h[j] = 1.0
            if h[best_k] < 0:
                h[best_k] = 0.0
            elif h[best_k] > 1:
                h[best_k] = 1.0
            improved = True
    return h, corr, improved

def smooth_max_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m_val))
    w = exp_vals / np.sum(exp_vals)
    obj = (m_val + np.log(np.sum(exp_vals)) / alpha) * dx
    # Gradient: for each j, grad[j] = dx * sum_m w[m] * dcorr[m]/dh[j]
    # dcorr[m]/dh[j] = (1-h[j+m-(n-1)]) if valid - h[j-m+(n-1)] if valid
    # This is equivalent to: convolve(w, 1-h) and convolve(w, h) with appropriate indexing
    # Let's use the direct approach
    grad = np.zeros(n)
    for j in range(n):
        # First term: sum over m where j+m-n+1 is in [0, n-1]
        # m ranges from n-1-j to 2n-2-j
        m_lo = max(0, n - 1 - j)
        m_hi = min(2*n-2, 2*n-2-j)
        if m_lo <= m_hi:
            idx = j + np.arange(m_lo, m_hi+1) - (n-1)
            grad[j] += np.sum(w[m_lo:m_hi+1] * (1.0 - h[idx]))
        # Second term: sum over m where j-m+n-1 is in [0, n-1]
        m_lo2 = max(0, j)
        m_hi2 = min(2*n-2, j+n-1)
        if m_lo2 <= m_hi2:
            idx2 = j - np.arange(m_lo2, m_hi2+1) + (n-1)
            grad[j] -= np.sum(w[m_lo2:m_hi2+1] * h[idx2])
    grad *= dx
    return obj, grad

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)

        diff = target_sum - np.sum(h)

def compute_B(h, j, n):
    nm1 = n - 1
    B = np.zeros(2 * n - 1)
    k_start = max(0, nm1 - j)
    k_end = min(2 * n - 2, 2 * nm1 - j)

    return B

    if hi - lo < 1e-14:
        return (lo + hi) * 0.5

        return lo
    if best_idx == len(t_grid) - 1:
        return hi

    hi_r = t_grid[best_idx + 1]

    n = len(h)
    order = np.arange(n)
    rng.shuffle(order)
    improved = False

        best_val = np.max(corr)
        best_delta = 0.0
        best_D = None

                continue
            seen.add(k)
            B_k = compute_B(h, k, n)
            D = B_j - B_k

def solve_1d(A, D, lo, hi):
    if hi - lo < 1e-14:
        return (lo + hi) * 0.5
    t_grid = np.linspace(lo, hi, 12)
    vals = np.max(A[None, :] + D[None, :] * t_grid[:, None], axis=1)
    best_idx = int(np.argmin(vals))
    if best_idx == 0:
        return lo
    if best_idx == len(t_grid) - 1:
        return hi
    lo_r = t_grid[best_idx - 1]
    hi_r = t_grid[best_idx + 1]
    for _ in range(32):
        mid = (lo_r + hi_r) * 0.5
        if np.max(A + D * mid) <= np.max(A + D * lo_r):
            lo_r = mid
        else:
            hi_r = mid
    return (lo_r + hi_r) * 0.5

def pair_descent_sweep(h, corr, rng, n_cand=10):
    n = len(h)
    order = np.arange(n)
    rng.shuffle(order)
    improved = False
    for j in order:
        B_j = compute_B(h, j, n)
        best_val = np.max(corr)
        best_delta = 0.0
        best_D = None
        best_k = -1
        k_set = [(j + 1) % n, (j - 1 + n) % n, (j + n // 2) % n, (j + n // 3) % n, (j + 2 * n // 3) % n]
        for _ in range(max(0, n_cand - 5)):
            k_set.append(rng.randint(n))
        seen = set()
        for k in k_set:
            if k == j or k in seen:
                continue
            seen.add(k)
            B_k = compute_B(h, k, n)
            D = B_j - B_k
            delta_lo = max(-h[j], h[k] - 1.0)
            delta_hi = min(1.0 - h[j], h[k])
            if delta_hi - delta_lo < 1e-13:
                continue
            delta_opt = solve_1d(corr, D, delta_lo, delta_hi)
            val_opt = np.max(corr + D * delta_opt)
            if val_opt < best_val - 1e-13:
                best_val = val_opt
                best_delta = delta_opt
                best_D = D
                best_k = k
        if best_k >= 0 and abs(best_delta) > 1e-13:
            corr += best_delta * best_D
            h[j] += best_delta
            h[best_k] -= best_delta
            h[j] = np.clip(h[j], 0.0, 1.0)
            h[best_k] = np.clip(h[best_k], 0.0, 1.0)
            improved = True
    return h, corr, improved

def smooth_max_obj_grad(h, alpha, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted = np.clip(shifted, -500, 0)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    log_sum = np.log(np.sum(exp_vals))
    obj = (m_val + log_sum / alpha) * dx
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx + m_idx - (n - 1)
    second_idx = j_idx - m_idx + (n - 1)
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    term1 = valid_f * (1.0 - h[fc])
    term2 = valid_s * h[sc]
    grad = dx * np.sum(w[None, :] * (term1 - term2), axis=1)
    if lam > 0:
        s = np.sum(h) - n / 2.0
        obj += lam * s * s
        grad += 2.0 * lam * s
    return obj, grad

def lbfgs_smooth(h0, n, alpha, lam, maxiter):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            lambda x: smooth_max_obj_grad(x, alpha, lam)[0],
            h0.copy(),
            method='L-BFGS-B',
            jac=lambda x: smooth_max_obj_grad(x, alpha, lam)[1],
            bounds=bounds,
            options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10, 'maxfun': 5000}
        )
        return np.clip(res.x, 0.0, 1.0)
    except Exception:
        return h0.copy()

def smooth_max_rounds(h, n, target_sum, alpha_list, lam, time_limit):
    t0 = time.time()
    h_cur = h.copy()
    best_local = h_cur.copy()
    best_c5_local = compute_c5(h_cur)
    for alpha in alpha_list:
        if time.time() - t0 > time_limit:
            break
        mi = 200 if alpha < 100 else 400
        h_new = lbfgs_smooth(h_cur, n, alpha, lam, mi)
        h_new = project_to_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5_local:
            best_c5_local = c5_new
            best_local = h_new.copy()
        if c5_new <= compute_c5(h_cur) + 1e-9:
            h_cur = h_new
    return best_local, best_c5_local

def fourier_start(rng, n, scale):
    freqs = np.arange(1, n // 2 + 1)
    a = rng.randn(len(freqs)) * scale
    b = rng.randn(len(freqs)) * scale
    x = np.arange(n) * 2.0 / n
    h = np.zeros(n)
    for k_idx in range(len(freqs)):
        h += a[k_idx] * np.cos(np.pi * freqs[k_idx] * x) + b[k_idx] * np.sin(np.pi * freqs[k_idx] * x)
    h = 0.5 + h
    h = np.clip(h, 0.0, 1.0)
    return project_to_sum(h, n / 2.0)

def periodic_start(rng, n, period):
    x = np.arange(n)
    phase = rng.uniform(0, period)
    h = np.where(((x + phase) % period) < period * 0.5, 1.0, 0.0)
    return project_to_sum(h, n / 2.0)

def sigmoid_start(rng, n, scale):
    A = rng.randn(n) * scale
    h = 1.0 / (1.0 + np.exp(-A))
    return project_to_sum(h, n / 2.0)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        try:
            best_h = np.array(initial_h_values, dtype=float).copy()
            best_n = len(best_h)
            best_h = project_to_sum(best_h, best_n / 2.0)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    if best_h is None:
        best_n = 240
        best_h = project_to_sum(rng.uniform(0, 1, best_n), best_n / 2.0)
        best_c5 = compute_c5(best_h)

    t_end1 = t_start + 0.40 * budget_s
    n1 = best_n
    ts1 = n1 / 2.0

    h_res = smooth_max_rounds(best_h, n1, ts1, [30, 80, 200, 500, 1000], 500.0, t_end1 - time.time())
    if h_res[1] < best_c5:
        best_c5 = h_res[1]
        best_h = h_res[0].copy()

    if time.time() < t_end1:
        h4 = fourier_start(rng, n1, 0.08)
        h_res = smooth_max_rounds(h4, n1, ts1, [50, 150, 400], 500.0, t_end1 - time.time())
        if h_res[1] < best_c5:
            best_c5 = h_res[1]
            best_h = h_res[0].copy()

    if time.time() < t_end1:
        hs = sigmoid_start(rng, n1, 1.2)
        h_res = smooth_max_rounds(hs, n1, ts1, [50, 200, 500], 500.0, t_end1 - time.time())
        if h_res[1] < best_c5:
            best_c5 = h_res[1]
            best_h = h_res[0].copy()

    t_end2 = t_start + 0.62 * budget_s
    for n_tr in [240, 248, 256, 264]:
        if time.time() >= t_end2:
            break
        ts_tr = n_tr / 2.0
        if abs(ts_tr - round(ts_tr)) > 1e-6:
            continue
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_tr, endpoint=False)
        h_interp = project_to_sum(np.interp(x_new, x_old, best_h, period=2.0), ts_tr)
        h_res = smooth_max_rounds(h_interp, n_tr, ts_tr, [100, 300, 800], 2000.0, min(50, t_end2 - time.time()))
        if h_res[1] < best_c5:
            best_c5 = h_res[1]
            best_h = h_res[0].copy()
            best_n = n_tr
        if time.time() < t_end2:
            hp = periodic_start(rng, n_tr, n_tr // 6)
            h_res = smooth_max_rounds(hp, n_tr, ts_tr, [100, 300], 2000.0, min(30, t_end2 - time.time()))
            if h_res[1] < best_c5:
                best_c5 = h_res[1]
                best_h = h_res[0].copy()
                best_n = n_tr

    t_end3 = t_start + 0.82 * budget_s
    starts = [best_h.copy()]
    if best_n >= 240:
        starts.append(fourier_start(rng, best_n, 0.06))
        starts.append(periodic_start(rng, best_n, best_n // 8))

    for h_st in starts:
        if time.time() >= t_end3:
            break
        n_st = len(h_st)
        ts_st = n_st / 2.0
        h_st = project_to_sum(h_st, ts_st)
        corr_st = np.correlate(h_st, 1.0 - h_st, mode="full")
        stall = 0
        while time.time() < t_end3:
            h_st, corr_st, imp = pair_descent_sweep(h_st, corr_st, rng, n_cand=10)
            if not imp:
                stall += 1
                if stall >= 5:
                    h_st = project_to_sum(h_st + rng.uniform(-0.04, 0.04, n_st), ts_st)
                    corr_st = np.correlate(h_st, 1.0 - h_st, mode="full")
                    stall = 0
            else:
                stall = 0
        c5_st = compute_c5(h_st)
        if c5_st < best_c5:
            best_c5 = c5_st
            best_h = h_st.copy()
            best_n = n_st

    h_fin = best_h.copy()
    n_fin = best_n
    ts_fin = n_fin / 2.0
    corr_fin = np.correlate(h_fin, 1.0 - h_fin, mode="full")
    stall_fin = 0
    while time.time() < t_start + budget_s - 5:
        h_fin, corr_fin, imp = pair_descent_sweep(h_fin, corr_fin, rng, n_cand=12)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5 - 1e-10:
            best_c5 = c5_fin
            best_h = h_fin.copy()
            best_n = n_fin
            stall_fin = 0
        else:
            stall_fin += 1
            if stall_fin >= 30:
                h_fin = project_to_sum(h_fin + rng.uniform(-0.02, 0.02, n_fin), ts_fin)
                corr_fin = np.correlate(h_fin, 1.0 - h_fin, mode="full")
                stall_fin = 0

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
