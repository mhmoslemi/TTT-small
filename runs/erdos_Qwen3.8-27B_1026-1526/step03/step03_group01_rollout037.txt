import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx


def project_box_hyperplane(h, ts):
    h = h.astype(float).copy()
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        x = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(x)
        if s > ts:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    diff = ts - np.sum(result)
    if abs(diff) > 1e-14:
        mask = (result > 1e-12) & (result < 1.0 - 1e-12)
        if np.any(mask):
            result[mask] += diff / np.sum(mask)
            result = np.clip(result, 0.0, 1.0)
    return result


def smooth_c5_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    max_sh = np.max(shifted)
    exp_v = np.exp(shifted - max_sh)
    w = exp_v / np.sum(exp_v)
    obj = (m_val + (max_sh + np.log(np.sum(exp_v))) / alpha) * (2.0 / n)
    nj = np.arange(n)[:, None]
    nm = np.arange(2 * n - 1)[None, :]
    fi = nj - nm + n - 1
    si = nj + nm - n + 1
    vf = (fi >= 0) & (fi < n)
    vs = (si >= 0) & (si < n)
    fc = np.clip(fi, 0, n - 1)
    sc = np.clip(si, 0, n - 1)
    grad = np.sum(vf * (1.0 - h[fc]) * w[None, :] - vs * h[sc] * w[None, :], axis=1)
    grad *= (2.0 / n)
    return obj, grad


def projected_gradient(h, ts, alpha, n_steps, lr_init=0.05):
    h = h.copy()
    lr = lr_init
    obj, _ = smooth_c5_obj_grad(h, alpha)
    stall = 0
    for step in range(n_steps):
        _, grad = smooth_c5_obj_grad(h, alpha)
        grad_tan = grad - np.mean(grad)
        if np.max(np.abs(grad_tan)) < 1e-12:
            break
        h_new = project_box_hyperplane(h - lr * grad, ts)
        obj_new, _ = smooth_c5_obj_grad(h_new, alpha)
        if obj_new < obj - 1e-15:
            h = h_new
            obj = obj_new
            lr = min(lr * 1.08, 0.5)
            stall = 0
        else:
            lr *= 0.5
            stall += 1
            if stall > 80 or lr < 1e-14:
                break
    return h, obj


def slsqp_opt(h0, ts, alpha, maxiter=600):
    n = len(h0)

    def f(x):
        return smooth_c5_obj_grad(x, alpha)[0]

    def g(x):
        return smooth_c5_obj_grad(x, alpha)[1]

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ts, 'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:
        r = minimize(f, h0.copy(), jac=g, method='SLSQP', bounds=bnds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-11})
        return project_box_hyperplane(r.x, ts)
    except Exception:
        return project_box_hyperplane(h0, ts)


def perturb_global(h, rng, amp):
    n = len(h)
    noise = rng.randn(n) * amp
    h_new = h + noise
    return project_box_hyperplane(h_new, n / 2.0)


def perturb_local_block(h, rng, block_size_max=30):
    n = len(h)
    block_size = rng.randint(3, block_size_max + 1)
    start = rng.randint(0, max(1, n - block_size))
    idx = np.arange(start, min(start + block_size, n))
    t = np.linspace(0, 1, len(idx))
    bump = rng.uniform(-0.15, 0.15) * np.sin(np.pi * t)
    h_new = h.copy()
    h_new[idx] += bump
    return project_box_hyperplane(h_new, n / 2.0)


def perturb_swap(h, rng, n_swaps=5):
    h_new = h.copy()
    n = len(h)
    for _ in range(n_swaps):
        i = rng.randint(0, n)
        j = rng.randint(0, n)
        if i != j:
            h_new[i], h_new[j] = h_new[j], h_new[i]
    return project_box_hyperplane(h_new, n / 2.0)

    h_new = np.interp(x_new, x_old, h_old, period=2.0)
