   **Idea: Use a "minimax" solver directly**.
   
   The problem min_h max_k C_k(h) can be reformulated as:
   min_h t
   subject to: C_k(h) ≤ t for all k
              0 ≤ h_i ≤ 1 for all i
              sum h_i = n/2
   
   This is a non-convex problem (due to the bilinear terms in C_k), but we can use a **branch-and-bound** or **successive convex approximation** approach.
   
   Successive convex approximation: linearize C_k around the current point h*:
   C_k(h) ≈ C_k(h*) + ∇C_k(h*)'(h - h*)
   
   Then solve:
   min t
   subject to: C_k(h*) + ∇C_k(h*)'(h - h*) ≤ t for all k
              0 ≤ h_i ≤ 1
              sum h_i = n/2
   
   This is a linear program! We can solve it with any LP solver.
   
   Then update h* to the solution and repeat. This is the **cutting plane** or **outer approximation** method.
   
   The advantage: each subproblem is a linear program, which is fast and has a unique solution.
   The disadvantage: it may converge slowly and get stuck at non-optimal points.
   
   But combined with pair descent, it might be very effective!

   ```python
   from scipy.optimize import linprog
   
   def solve_linearized(h_cur, n, max_iter=100):
       dx = 2.0 / n
       target_sum = n / 2.0
       h = h_cur.copy()
       
       for it in range(max_iter):
           C = np.correlate(h, 1-h, mode="full")
           B_all = compute_all_B(h, n)  # shape (n, 2n-1)
           
           # Minimize t
           # s.t. C_k + sum_j B_all[j,k] * (h_j - h_cur_j) ≤ t
           #      => sum_j B_all[j,k] * h_j - t ≤ -C_k + sum_j B_all[j,k] * h_cur_j
           #      => sum_j B_all[j,k] * h_j - t ≤ B_all[:,k] @ h_cur - C_k
           
           # Variables: [h_0, ..., h_{n-1}, t]
           # c = [0, ..., 0, 1] (minimize t)
           
           c = np.zeros(n + 1)
           c[n] = 1  # minimize t
           
           # Inequality constraints: A_ub @ x ≤ b_ub
           # For each k: sum_j B_all[j,k] * h_j - t ≤ B_all[:,k] @ h_cur - C_k
           # A_ub row k: [B_all[:,k], -1]
           
           A_ub = np.hstack([B_all.T, -np.ones(2*n-1).reshape(-1,1)])
           b_ub = B_all @ h_cur - C
           
           # Equality: sum h_j = target_sum
           A_eq = np.zeros((1, n+1))
           A_eq[0, :n] = 1.0
           b_eq = np.array([target_sum])
           
           bounds = [(0,1)]*n + [(-np.inf, np.inf)]
           
           res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                        bounds=bounds, method='highs')
           
           if not res.success:
               break
           
           h_new = res.x[:n]
           h_new = np.clip(h_new, 0, 1)
           
           # Check if converged
           if np.max(np.abs(h_new - h)) < 1e-10:
               h = h_new
               break
           
           h = h_new
       
       return h
   ```

   ```python
   import numpy as np
   from scipy.optimize import minimize, linprog
   import time

   def compute_c5(h):
       n = len(h)
       dx = 2.0 / n
       corr = np.correlate(h, 1.0 - h, mode="full")
       return float(np.max(corr) * dx)

   def project_to_sum(h, target_sum):
       h = np.clip(h.astype(float), 0.0, 1.0)
       for _ in range(200):
           diff = target_sum - np.sum(h)
           if abs(diff) < 1e-14:
               break
           mask = (h > 1e-12) & (h < 1.0 - 1e-12)
           if np.any(mask):
               h[mask] += diff / np.sum(mask)
           else:
               h += diff / len(h)
           h = np.clip(h, 0.0, 1.0)
       return h

   def compute_B_single(h, j, n):
       nm1 = n - 1
       B = np.zeros(2 * n - 1)
       k_start = max(0, nm1 - j)
       k_end = min(2 * n - 2, 2 * nm1 - j)
       if k_start <= k_end:
           B[k_start:k_end+1] += (1.0 - h[j - nm1 + np.arange(k_start, k_end+1)])
       k_start2 = max(0, j)
       k_end2 = min(2 * n - 2, j + nm1)
       if k_start2 <= k_end2:
           B[k_start2:k_end2+1] += -h[j + nm1 - np.arange(k_start2, k_end2+1)]
       return B

   def compute_all_B(h, n):
       B_all = np.zeros((n, 2*n-1))
       for j in range(n):
           B_all[j] = compute_B_single(h, j, n)
       return B_all

   def solve_1d(A, D, lo, hi):
       if hi - lo < 1e-14:
           return (lo + hi) * 0.5
       # Quick grid search
       t_grid = np.linspace(lo, hi, 16)
       vals = np.max(A[None,:] + D[None,:] * t_grid[:,None], axis=1)
       best_idx = int(np.argmin(vals))
       if best_idx == 0:
           lo_r, hi_r = t_grid[0], t_grid[1]
       elif best_idx == len(t_grid)-1:
           lo_r, hi_r = t_grid[-2], t_grid[-1]
       else:
           lo_r, hi_r = t_grid[best_idx-1], t_grid[best_idx+1]
       # Bisection
       for _ in range(40):
           mid = (lo_r + hi_r) * 0.5
           if np.max(A + D * mid) <= np.max(A + D * lo_r):
               lo_r = mid
           else:
               hi_r = mid
       return (lo_r + hi_r) * 0.5

   def pair_descent_all(h, C, B_all, rng, n_cand=None):
       """Pair descent sweep. If n_cand is None, try all pairs."""
       n = len(h)
       order = np.arange(n)
       rng.shuffle(order)
       improved = False
       
       for j in order:
           best_val = np.max(C)
           best_delta = 0.0
           best_k = -1
           
           if n_cand is None:
               k_set = [k for k in range(n) if k != j]
           else:
               k_set = [(j+1)%n, (j-1+n)%n, (j+n//2)%n, (j+n//3)%n, (j+2*n//3)%n]
               for _ in range(max(0, n_cand - 5)):
                   k_set.append(rng.randint(n))
           
           seen = set()
           for k in k_set:
               if k == j or k in seen:
                   continue
               seen.add(k)
               D = B_all[j] - B_all[k]
               delta_lo = max(-h[j], h[k] - 1.0)
               delta_hi = min(1.0 - h[j], h[k])
               if delta_hi - delta_lo < 1e-13:
                   continue
               delta_opt = solve_1d(C, D, delta_lo, delta_hi)
               val_opt = np.max(C + D * delta_opt)
               if val_opt < best_val - 1e-14:
                   best_val = val_opt
                   best_delta = delta_opt
                   best_k = k
           
           if best_k >= 0 and abs(best_delta) > 1e-13:
               C += best_delta * (B_all[j] - B_all[best_k])
               h[j] = np.clip(h[j] + best_delta, 0, 1)
               h[best_k] = np.clip(h[best_k] - best_delta, 0, 1)
               improved = True
       
       return h, C, improved

   def lp_relinearize(h, n, max_iters=50):
       """Successive LP relinearization."""
       target_sum = n / 2.0
       dx = 2.0 / n
       
       for it in range(max_iters):
           C = np.correlate(h, 1-h, mode="full")
           B_all = compute_all_B(h, n)
           
           c_obj = np.zeros(n + 1)
           c_obj[n] = 1.0
           
           A_ub = np.hstack([B_all.T, -np.ones(2*n-1).reshape(-1, 1)])
           b_ub = B_all @ h - C
           
           A_eq = np.zeros((1, n+1))
           A_eq[0, :n] = 1.0
           b_eq = np.array([target_sum])
           
           bounds = [(0,1)]*n + [(-1e3, 1e3)]
           
           try:
               res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                           bounds=bounds, method='highs', options={'presolve': True})
               if not res.success:
                   break
               h_new = np.clip(res.x[:n], 0, 1)
               h_new = project_to_sum(h_new, target_sum)
               if np.max(np.abs(h_new - h)) < 1e-10:
                   h = h_new
                   break
               h = h_new
           except Exception:
               break
       
       return h

   def smooth_max_and_grad(h, alpha):
       n = len(h)
       corr = np.correlate(h, 1.0 - h, mode="full")
       m_val = np.max(corr)
       shifted = corr - m_val
       # Numerical stability
       max_shift = np.max(shifted)
       if max_shift > 700:
           shifted = shifted - max_shift
       exp_vals = np.exp(alpha * shifted)
       exp_sum = np.sum(exp_vals)
       if exp_sum == 0 or not np.isfinite(exp_sum):
           obj = m_val
           w = np.zeros(len(corr))
           w[np.argmax(corr)] = 1.0
       else:
           obj = m_val + np.log(exp_sum) / alpha
           w = exp_vals / exp_sum
       
       # Gradient
       B_all = compute_all_B(h, n)
       grad = w @ B_all  # shape (n,)
       
       return obj, grad

   def slsqp_local(h0, n, target_sum, alpha, maxiter=1000):
       def f(h):
           return smooth_max_and_grad(h, alpha)[0]
       def g(h):
           return smooth_max_and_grad(h, alpha)[1]
       
       cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                'jac': lambda x: np.ones(n)}]
       bounds = [(0.0, 1.0)] * n
       
       try:
           res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                         bounds=bounds, constraints=cons,
                         options={'maxiter': maxiter, 'ftol': 1e-16})
           if res.success:
               return project_to_sum(res.x, target_sum)
           return project_to_sum(h0, target_sum)
       except Exception:
           return project_to_sum(h0, target_sum)
