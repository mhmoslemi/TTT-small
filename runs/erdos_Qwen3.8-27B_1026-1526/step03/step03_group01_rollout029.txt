def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_alls(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)

        else:

        h = np.clip(h, 0.0, 1.0)
    return h

def slogsumexp_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    s = alpha * (corr - m)
    s -= np.max(s)
    e = np.exp(s)
    w = e / e.sum()
    obj = m + (np.log(e.sum())) / alpha
    # gradient
    nj = np.arange(n)[:, None]
    nm = np.arange(2*n-1)[None, :]
    p = nm - (n-1)
    fi = nj + p
    si = nj - p
    vf = (fi >= 0) & (fi < n)
    vs = (si >= 0) & (si < n)
    fc = np.clip(fi, 0, n-1)
    sc = np.clip(si, 0, n-1)
    grad = np.sum(vf * (1.0 - h[fc]) * w[None, :] - vs * h[sc] * w[None, :], axis=1)
    return obj, grad

def slsqp_step(h0, n, ts, alpha, maxiter=1000):
    def f(x):
        return slogsumexp_grad(x, alpha)[0]
    def g(x):
        return slogsumexp_grad(x, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ts, 'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n

                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14})

def exact_subgrad_step(h, n, ts, n_active=5, lr=0.5):
    """Projected gradient step using exact subgradient of max."""
    lags = compute_alls(h)
    m_val = np.max(lags)
    # Find active lags (within small epsilon of max)
    thresh = m_val - 1e-6
    active = np.where(lags >= thresh)[0]
    if len(active) == 0:
        active = np.argmax(lags)
        active = [active]
    
    # Compute gradient for each active lag
    # For lag index k in full correlation, shift p = k - (n-1)
    # d/dh[j] g_k = dx * [(1-h[j+p])*I(j+p in [0,n-1]) - h[j-p]*I(j-p in [0,n-1])]
    dx = 2.0 / n
    n_lags = 2*n - 1
    p_vals = np.arange(n_lags) - (n-1)
    
    grad = np.zeros(n)
    for k in active:
        p = p_vals[k]
        if p == 0:
            grad += dx * (1.0 - 2.0 * h)
        elif p > 0:
            # (1-h[j+p]) for j in [0, n-1-p]
            grad[:n-p] += dx * (1.0 - h[p:])
            # -h[j-p] for j in [p, n-1]

        else:

            grad[:n-s] += dx * (1.0 - h[s:])
            grad[s:] += dx * (-h[:n-s])
    
    # Average over active lags
    grad /= len(active)
    
    # Projected gradient step
    h_new = project_to_sum(h - lr * grad, ts)
    return h_new, m_val

    nm = np.arange(2*n-1)[None, :]
    p = nm - (n-1)
    fi = nj + p
    si = nj - p
    vf = (fi >= 0) & (fi < n)
    vs = (si >= 0) & (si < n)
    fc = np.clip(fi, 0, n-1)
    sc = np.clip(si, 0, n-1)
    grad = np.sum(vf * (1.0 - h[fc]) * w[None, :] - vs * h[sc] * w[None, :], axis=1)
    return obj, grad

def slsqp_opt(h0, n, ts, alpha, maxiter=3000):
    def f(x):
        return smooth_max_and_grad(x, alpha)[0]
    def g(x):
        return smooth_max_and_grad(x, alpha)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ts, 'jac': lambda x: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g, bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-15})
        return project_to_sum(res.x, ts)
    except Exception:
        return project_to_sum(h0, ts)

def subgrad_descent(h, n, ts, n_iter=5000, lr_init=2.0):
    h = h.copy()
    dx = 2.0 / n
    n_lags = 2*n - 1
    p_vals = np.arange(n_lags) - (n-1)
    lr = lr_init
    c5_cur = compute_c5(h)
    stall = 0
    for it in range(n_iter):
        lags = np.correlate(h, 1.0-h, mode="full") * dx
        m_val = np.max(lags)
        thresh = m_val - 1e-8 * max(1.0, abs(m_val))
        active = np.where(lags >= thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(lags)])
        
        grad = np.zeros(n)
        for k in active:
            p = p_vals[k]
            if p == 0:
                grad += dx * (1.0 - 2.0 * h)
            elif p > 0:
                grad[:n-p] += dx * (1.0 - h[p:])
                grad[p:] -= dx * h[:n-p]
            else:
                s = -p
                grad[s:] += dx * (1.0 - h[:n-s])
                grad[:n-s] -= dx * h[s:]
        grad /= len(active)
        
        h_new = project_to_sum(h - lr * grad, ts)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur - 1e-15:
            h = h_new
            c5_cur = c5_new
            stall = 0
            lr = min(lr * 1.05, 10.0)
        else:
            stall += 1
            lr *= 0.5
            if stall > 200 or lr < 1e-10:
                break
    return h, c5_cur
