def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def make_binary(h, n, target_sum, rng):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones == target_sum:
        return h_bin
    if n_ones > target_sum:
        excess = n_ones - target_sum
        ones_idx = np.where(h_bin == 1.0)[0]
        order = np.argsort(h[ones_idx])[:excess]
        h_bin[ones_idx[order]] = 0.0
    else:
        deficit = target_sum - n_ones
        zeros_idx = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h[zeros_idx])[:deficit]
        h_bin[zeros_idx[order]] = 1.0
    return h_bin

def binary_sa_advanced(h_start, n, target_sum, t_end, rng, verbose=False):
    h = h_start.copy().astype(float)
    corr = compute_corr(h)
    c5_curr = float(np.max(corr))
    best_c5 = c5_curr
    best_h = h.copy()
    
    T = 0.02
    min_T = 0.002
    max_T = 0.03
    no_improve = 0
    accept_count = 0
    total_count = 0
    STABLE = 5000
    
    while time.time() < t_end:
        # Adaptive temperature
        if total_count > 2000:
            rate = accept_count / total_count
            if rate < 0.1:
                T = min(T * 1.5, max_T)
            elif rate > 0.6:
                T = max(T * 0.7, min_T)
            accept_count = 0
            total_count = 0
        
        total_count += 1
        move_type = rng.randint(3)
        
        if move_type == 0:  # Single swap
            i = rng.randint(n)
            j = rng.randint(n)
            if h[i] == h[j]:
                continue
            h[i], h[j] = h[j], h[i]
            new_corr = compute_corr(h)
            c5_new = float(np.max(new_corr))
            delta = c5_new - c5_curr
            if delta < 0 or np.log(rng.uniform()) < delta / T:
                c5_curr = c5_new
                corr = new_corr
                accept_count += 1
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h[i], h[j] = h[j], h[i]
        
        elif move_type == 1:  # Double swap
            i1 = rng.randint(n)
            j1 = rng.randint(n)
            i2 = rng.randint(n)
            j2 = rng.randint(n)
            if len(set([i1, j1, i2, j2])) < 4:
                continue
            if h[i1] == h[j1] or h[i2] == h[j2]:
                continue
            h[i1], h[j1] = h[j1], h[i1]
            h[i2], h[j2] = h[j2], h[i2]
            new_corr = compute_corr(h)
            c5_new = float(np.max(new_corr))
            delta = c5_new - c5_curr
            if delta < 0 or np.log(rng.uniform()) < delta / T:
                c5_curr = c5_new
                corr = new_corr
                accept_count += 1
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h[i1], h[j1] = h[j1], h[i1]
                h[i2], h[j2] = h[j2], h[i2]
        
        else:  # Block move
            L = rng.randint(2, 6)
            start = rng.randint(n - L)
            if h[start:start+L].sum() == 0 or h[start:start+L].sum() == L:
                continue
            target_start = rng.randint(n - L)
            if abs(target_start - start) < L:
                continue
            block = h[start:start+L].copy()
            h[start:start+L] = h[target_start:target_start+L]
            h[target_start:target_start+L] = block
            new_corr = compute_corr(h)
            c5_new = float(np.max(new_corr))
            delta = c5_new - c5_curr
            if delta < 0 or np.log(rng.uniform()) < delta / T:
                c5_curr = c5_new
                corr = new_corr
                accept_count += 1
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h[start:start+L] = block
                h[target_start:target_start+L] = h[start:start+L]
        
        if no_improve > 30000:
            h = best_h.copy()
            c5_curr = best_c5
            corr = compute_corr(h)
            T = min(T * 2.0, max_T)
            no_improve = 0
        
        T = max(T * 0.99999, min_T)
    
    return best_h, best_c5 * (2.0 / n)

    w = exp_vals / np.sum(exp_vals)
    n = len(h)
    n_corr = len(corr)
    j_idx = np.arange(n)[:, None]
    k_idx = np.arange(n_corr)[None, :]
    first_idx = j_idx - k_idx + (n - 1)
    second_idx = j_idx + k_idx - (n - 1)
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return grad

def slsqp_optimize(h0, n, target_sum, alpha, maxiter=1000):
    def f(h):
        return smooth_max_loss(h, alpha)[0]
    def g(h):
        _, corr = smooth_max_loss(h, alpha)
        return smooth_max_grad(h, corr, alpha)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, project_to_sum(h0, target_sum), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0, target_sum)

def binary_sa(h_start, n, target_sum, t_end, rng):
    h = h_start.copy().astype(float)
    corr = np.correlate(h, 1.0 - h, mode="full")
    c5_curr = float(np.max(corr))
    best_c5 = c5_curr
    best_h = h.copy()
    T = 0.015
    min_T = 0.001
    no_improve = 0
    
    while time.time() < t_end:
        # Adaptive T based on recent acceptance
        i = rng.randint(n)
        j = rng.randint(n)
        if h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        new_corr = np.correlate(h, 1.0 - h, mode="full")
        c5_new = float(np.max(new_corr))
        delta = c5_new - c5_curr
        if delta < 0 or np.log(rng.uniform()) < delta / T:
            c5_curr = c5_new
            corr = new_corr
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i], h[j] = h[j], h[i]
        
        if no_improve > 20000:
            h = best_h.copy()
            c5_curr = best_c5
            corr = np.correlate(h, 1.0 - h, mode="full")
            T = min(T * 2.0, 0.025)
            no_improve = 0
        T = max(T * 0.999995, min_T)
    
    return best_h, best_c5 * (2.0 / n)

def block_sa(h_start, n, target_sum, t_end, rng):
    h = h_start.copy().astype(float)
    corr = np.correlate(h, 1.0 - h, mode="full")
    c5_curr = float(np.max(corr))
    best_c5 = c5_curr
    best_h = h.copy()
    T = 0.02
    min_T = 0.003
    no_improve = 0
    
    while time.time() < t_end:
        r = rng.random()
        if r < 0.5:
            # Single swap
            i = rng.randint(n)
            j = rng.randint(n)
            if h[i] == h[j]:
                continue
            h[i], h[j] = h[j], h[i]
        elif r < 0.8:
            # Double swap
            i1, j1 = rng.randint(n), rng.randint(n)
            i2, j2 = rng.randint(n), rng.randint(n)
            if len(set([i1,j1,i2,j2])) < 4:
                continue
            if h[i1] == h[j1] or h[i2] == h[j2]:
                continue
            h[i1], h[j1] = h[j1], h[i1]
            h[i2], h[j2] = h[j2], h[i2]
        else:
            # Block move
            L = rng.randint(2, 6)
            s1 = rng.randint(n - L)
            s2 = rng.randint(n - L)
            if abs(s1 - s2) < L:
                continue
            blk1 = h[s1:s1+L].copy()
            blk2 = h[s2:s2+L].copy()
            if np.array_equal(blk1, blk2):
                continue
            h[s1:s1+L] = blk2
            h[s2:s2+L] = blk1
        
        new_corr = np.correlate(h, 1.0 - h, mode="full")
        c5_new = float(np.max(new_corr))
        delta = c5_new - c5_curr
        if delta < 0 or np.log(rng.uniform()) < delta / T:
            c5_curr = c5_new
            corr = new_corr
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # Revert
            if r < 0.5:
                h[i], h[j] = h[j], h[i]
            elif r < 0.8:
                h[i1], h[j1] = h[j1], h[i1]
                h[i2], h[j2]
