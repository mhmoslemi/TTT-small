def project_to_sum(h, target_sum):
    h = h.astype(float).copy()

    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break

    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)
    ew = np.exp(shifted)

    dx = 2.0 / n

    return float(np.max(corr) * dx)

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:

        h = np.clip(h, 0.0, 1.0)

def topk_calc(h, K, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)
    ew = np.exp(shifted)
    wsum = np.sum(ew)
    w = ew / wsum
    obj = m_val + np.log(wsum) / alpha
    kk = min(K, len(corr))
    top_idx = np.argsort(w)[-kk:][::-1]
    tw = w[top_idx] / np.sum(w[top_idx])
    lags = top_idx - (n - 1)
    grad = np.zeros(n)
    for idx in range(kk):
        lag = lags[idx]
        wt = tw[idx]
        if lag >= 0:
            if lag == 0:
                grad[:] += wt * (1.0 - 2.0 * h)
            else:
                grad[:n-lag] += wt * (1.0 - h[lag:])
                grad[lag:] -= wt * h[:n-lag]

            if kk2 > 0:
                grad[kk2:] += wt * (1.0 - h[:n-kk2])
                grad[:n-kk2] -= wt * h[kk2:]
    return obj, grad
