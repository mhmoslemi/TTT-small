```python
def proximal_gradient_subgradient(h, n, target_sum, n_iters, step_size):
    """FISTA-like method for minimizing max_k corr(h,1-h,k)*dx"""
    dx = 2.0/n
    h_curr = h.copy()
    h_prev = h.copy()
    t = 1.0
    
    for it in range(n_iters):
        # Compute all correlations
        corr = np.correlate(h_curr, 1.0-h_curr, mode="full")
        active_lag = np.argmax(corr)
        L = active_lag - (n-1)
        
        # Compute subgradient at active lag
        g = np.zeros(n)
        if L >= 0:
            if L == 0:
                g[:] = 1.0 - 2.0*h_curr
            else:
                g[L:] += (1.0 - h_curr[:n-L])
                g[:n-L] -= h_curr[L:]
        else:
            k = -L
            g[:n-k] += (1.0 - h_curr[k:])
            g[k:] -= h_curr[:n-k]
        
        # Cos.step
        h_proposed = h_curr - step_size * dx * g
        h_proj = project_to_sum(h_proposed, target_sum)
        
        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0*t*t)) / 2.0
        h_curr = h_proj + ((t-1.0)/t_new) * (h_proj - h_prev)
        h_curr = project_to_sum(h_curr, target_sum)
        h_prev = h_proj
        t = t_new
    
    return h_curr
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-13) & (h < 1.0 - 1e-13)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha, top_k=1):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    # Use soft-max over top_k lags
    sorted_idx = np.argsort(corr)[::-1]
    top_indices = sorted_idx[:top_k]
    top_corr = corr[top_indices]
    shifted = alpha * (top_corr - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    
    # Gradient: for each top lag, compute gradient and weight
    j_idx = np.arange(n)[:, None]
    grad = np.zeros(n)
    for idx, wi in zip(top_indices, w):
        lag = idx - (n - 1)
        if lag >= 0:
            if lag == 0:
                g = 1.0 - 2.0 * h
            else:
                g = np.zeros(n)
                g[lag:] += (1.0 - h[:n-lag])
                g[:n-lag] -= h[lag:]
        else:
            k = -lag
            g = np.zeros(n)
            g[:n-k] += (1.0 - h[k:])
            g[k:] -= h[:n-k]
        grad += wi * g
    return obj, grad

def slsqp_local(h0, n, target_sum, alpha, maxiter, top_k=1):
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    def f(x):
        return smooth_max_and_grad(x, alpha, top_k)[0]
    def g(x):
        return smooth_max_and_grad(x, alpha, top_k)[1]
    
    try:
        res = minimize(f, h0.copy(), method='SLSQP', jac=g,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def subgradient_descent(h, target_sum, n_iters, step_size_init, top_k=3):
    """Subgradient method with averaging for min max corr"""
    n = len(h)
    dx = 2.0/n
    h_curr = h.copy()
    best_h = h.copy()
    best_val = compute_c5(h_curr)
    h_avg = h_curr.copy()
    
    for it in range(n_iters):
        corr = compute_corr(h_curr)
        sorted_idx = np.argsort(corr)[::-1]
        top_indices = sorted_idx[:top_k]
        top_corr = corr[top_indices]
        
        # Soft weights
        temps = 1.0 / (1.0 + it * 0.01)
        weights = np.exp(temps * (top_corr - top_corr[0]))
        weights /= weights.sum()
        
        # Subgradient
        g = np.zeros(n)
        for idx, wi in zip(top_indices, weights):
            lag = idx - (n-1)
            if lag >= 0:
                if lag == 0:
                    g += wi * (1.0 - 2.0*h_curr)
                else:
                    g[lag:] += wi * (1.0 - h_curr[:n-lag])
                    g[:n-lag] -= wi * h_curr[lag:]
            else:
                k = -lag
                g[:n-k] += wi * (1.0 - h_curr[k:])
                g[k:] -= wi * h_curr[:n-k]
        
        # Step
        step = step_size_init / (1.0 + 0.05 * it)
        h_proposed = h_curr - step * dx * g
        h_curr = project_to_sum(h_proposed, target_sum)
        
        # Running average
        h_avg = ((it+1) * h_curr + it * h_avg) / (it+1)
        h_avg = project_to_sum(h_avg, target_sum)
        
        val = compute_c5(h_curr)
        if val < best_val:
            best_val = val
            best_h = h_curr.copy()
        
        val_avg = compute_c5(h_avg)
        if val_avg < best_val:
            best_val = val_avg
            best_h = h_avg.copy()
    
    return best_h, best_val

def pairwise_swap_descent(h, target_sum, max_iters=10000, top_k=2):
    """More efficient than coordinate descent: try pairs of swaps"""
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    stale = 0
    
    for it in range(max_iters):
        corr = compute_corr(h)
        sorted_idx = np.argsort(corr)[::-1]
        # Use gradient from top lag
        m_star = sorted_idx[0]
        lag = m_star - (n-1)
        
        g = np.zeros(n)
        if lag >= 0:
            if lag == 0:
                g[:] = 1.0 - 2.0*h
            else:
                g[lag:] += (1.0 - h[:n-lag])
                g[:n-lag] -= h[lag:]
        else:
            k = -lag
            g[:n-k] += (1.0 - h[k:])
            g[k:] -= h[:n-k]
        
        can_inc = h < 1.0 - 1e-10
        can_dec = h > 1e-10
        if not np.any(can_inc) or not np.any(can_dec):
            break
        
        g_inc = np.where(can_inc, g, np.inf)
        g_dec = np.where(can_dec, g, -np.inf)
        i_inc = int(np.argmin(g_inc))
        i_dec = int(np.argmax(g_dec))
        
        if i_inc == i_dec or g[i_inc] >= g[i_dec] - 1e-12:
            break
        
        max_move = min(1.0 - h[i_inc], h[i_dec])
        move = 0.1 * max_move
        if move < 1e-10:
            break
        
        h[i_inc] += move
        h[i_dec] -= move
        h = np.clip(h, 0.0, 1.0)
        h = project_to_sum(h, target_sum)
        
        c5_new = compute_c5(h)
        if c5_new >= c5 - 1e-14:
            h[i_inc] -= move
            h[i_dec] += move
            h = project_to_sum(h, target_sum)
            stale += 1
            if stale > 100:
                break
        else:
            c5 = c5_new
            stale = 0
    
    return h, c5

def binarize_pressure(h, target_sum, strength=0.1):
    """Push values towards 0 or 1"""
    h = h.copy()
    mask_up = h > 0.5
    mask_down = h < 0.5
    h[mask_up] = h[mask_up] + strength * (1.0 - h[mask_up])
    h[mask_down] = h[mask_down] - strength * h[mask_down]
    return project_to_sum(h, target_sum)

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h, period=2.0)
```

```python
        for a in [80, 300, 1200, 4000]:
            if tl() < 35:
                break
            hi = slsqp_optimize(hi, ntr, tst, a, 500, 2)
            c5n = compute_c5(hi)
            if c5n < best_c5:
                best_c5 = c5n
                best_h = hi.copy()
                best_n = ntr
        if tl() > 35:
            hcd2, c5cd2 = coord_descent(hi, tst, 6000, 0.1)
            if c5cd2 < best_c5:
                best_c5 = c5cd2
                best_h = hcd2.copy()
                best_n = ntr

    # Phase 5: Binarization perturbation
    nb = best_n
    tsb = nb / 2.0
    for eps in [0.008, 0.004, 0.002, 0.001]:
        if tl() < 35:
            break
        hb = binarize_push(best_h, tsb, eps)
        ho = slsqp_optimize(hb, nb, tsb, 2000, 500, 2)
        c5o = compute_c5(ho)
        if c5o < best_c5:
            best_c5 = c5o
            best_h = ho.copy()
        hcd3, c5cd3 = coord_descent(ho, tsb, 4000, 0.08)
        if c5cd3 < best_c5:
            best_c5 = c5cd3
            best_h = hcd3.copy()

    # Phase 6: Random restarts
    for trial in range(10):
        if tl() < 45:
            break
        ns = 0.015 + 0.025 * (trial / 10.0)
        hs = best_h + rng.uniform(-ns, ns, nb)
        hs = project_to_sum(hs, tsb)
        ho = slsqp_optimize(hs, nb, tsb, 800, 400, 2)
        c5o = compute_c5(ho)
        if c5o < best_c5:
            best_c5 = c5o
            best_h = ho.copy()
        if tl() > 30:
            hcd4, c5cd4 = coord_descent(ho, tsb, 4000, 0.1)
            if c5cd4 < best_c5:
                best_c5 = c5cd4
                best_h = hcd4.copy()

    # Phase 7: Final high-alpha polish
    for a in [8000, 20000, 50000]:
        if tl() < 25:
            break
        hp = slsqp_optimize(best_h, nb, tsb, a, 1200, 3)
        c5p = compute_c5(hp)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()

    # Phase 8: Final long coordinate descent
    if tl() > 30:
        hfin, c5fin = coord_descent(best_h, tsb, 20000, 0.06)
        if c5fin < best_c5:
            best_c5 = c5fin
            best_h = hfin.copy()

    # Phase 9: One more SLSQP if time
    if tl() > 20:
        hp2 = slsqp_optimize(best_h, nb, tsb, 100000, 1500, 3)
        c5p2 = compute_c5(hp2)
        if c5p2 < best_c5:
            best_c5 = c5p2
            best_h = hp2.copy()

    best_h = project_to_sum(best_h, best_n / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, best_n)
```
