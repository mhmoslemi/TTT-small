import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_sum(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / np.sum(mask)
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m_val = np.max(corr)
    shifted = alpha * (corr - m_val)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    obj = m_val + np.log(np.sum(exp_vals)) / alpha
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    first_idx = j_idx - m_idx + n - 1
    second_idx = j_idx + m_idx - n + 1
    valid_f = (first_idx >= 0) & (first_idx < n)
    valid_s = (second_idx >= 0) & (second_idx < n)
    fc = np.clip(first_idx, 0, n - 1)
    sc = np.clip(second_idx, 0, n - 1)
    cf = valid_f * (1.0 - h[fc]) * w[None, :]
    cs = valid_s * h[sc] * w[None, :]
    grad = np.sum(cf - cs, axis=1)
    return obj, grad

class SmoothMaxProb:
    def __init__(self, alpha, target_sum, mu=0.0):
        self.alpha = alpha
        self.target_sum = target_sum
        self.mu = mu
    def f(self, x):
        obj, _ = smooth_max_and_grad(x, self.alpha)
        if self.mu > 0:
            d = np.sum(x) - self.target_sum
            obj += self.mu * d * d
        return obj
    def g(self, x):
        _, grad = smooth_max_and_grad(x, self.alpha)
        if self.mu > 0:
            d = np.sum(x) - self.target_sum
            grad = grad + 2.0 * self.mu * d
        return grad

def optimize_lbfgs(h0, alpha, target_sum, mu=1000.0, maxiter=500):
    prob = SmoothMaxProb(alpha, target_sum, mu)
    bounds = [(0.0, 1.0)] * len(h0)
    try:
        res = minimize(prob.f, h0.copy(), jac=prob.g, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-12})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)

def optimize_slsqp(h0, alpha, target_sum, maxiter=500):
    n = len(h0)
    prob = SmoothMaxProb(alpha, target_sum, mu=0.0)
    bounds = [(0.0, 1.0)] * n
    
    class ConsFn:
        def __init__(self, ts, n):
            self.ts = ts
            self.n = n
        def fun(self, x):
            return np.sum(x) - self.ts
        def jac(self, x):
            return np.ones(self.n)
    
    cfn = ConsFn(target_sum, n)
    cons = [{'type': 'eq', 'fun': cfn.fun, 'jac': cfn.jac}]
    try:

                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_to_sum(res.x, target_sum)
    except Exception:
        return project_to_sum(h0.copy(), target_sum)
