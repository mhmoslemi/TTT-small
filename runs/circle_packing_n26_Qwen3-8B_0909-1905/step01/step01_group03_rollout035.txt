import scipy.optimize as optimization
from typing import Tuple, List

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:

    n_circles = 26

            row = i // cols
            col = i % cols
            x = col + 0.5 * (row % 2)
            y = row * np.sqrt(3) / 2
            centers[i] = [x, y]
        # Scale to fit within the unit square
        centers[:, 0] /= (cols + 1)
        centers[:, 1] /= (rows + 1)
        return centers

    initial_radii = np.array([0.03] * n_circles)

    def objective(radii):

    def constraint_bounds(radii):

        return np.all((centers[:, 0] - radii) >= 0) and np.all((1 - centers[:, 0] - radii) >= 0) and \
