import numpy as np
import scipy.optimize as opt
import math

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Helper function to compute the maximum possible radius given positions
    def max_radius(centers, i):
        if i == 0:
            # First circle: place it in the center
            return 0.4  # Approximate value based on square packing
        else:
            # Calculate distance to all previous circles and to boundaries
            min_dist = float('inf')
            for j in range(i):
                dist = np.linalg.norm(centers[i] - centers[j])
                min_dist = min(min_dist, dist)
            # Distance to nearest boundary
            dist_to_boundary = min(centers[i][0], 1 - centers[i][0], centers[i][1], 1 - centers[i][1])
            return min(min_dist, dist_to_boundary) / 2

    # Generate an initial hexagonal packing
    def hexagonal_packing(n, x_size, y_size, spacing):
        centers = []
        for i in range(n):
            row = i // 3
            col = i % 3
            x = col * spacing
            y = row * (spacing * math.sqrt(3) / 2)
            if row % 2 == 1:
                x += spacing / 2
            centers.append([x, y])
        centers = np.array(centers)
        # Scale to fit within the square
        scale = min(x_size / centers[:, 0].max(), y_size / centers[:, 1].max())
        centers *= scale
        return centers

    # Objective function to maximize the sum of radii
    def objective(x, centers):
        # x is a flattened array of radii
        radii = x.reshape(-1, 1)
        sum_radii = np.sum(radii)
        return -sum_radii  # Minimize negative sum for maximization

    # Constraint function: circles cannot overlap or go outside the square
    def constraints(x, centers):
        radii = x.reshape(-1, 1)
        constraints = []
        # Boundary constraints
        for i in range(len(centers)):
            x_center, y_center = centers[i]
            r = radii[i]
            # Ensure circle stays within the square
            constraints.append((x_center - r >= 0) * 1e6)
            constraints.append((x_center + r <= 1) * 1e6)
            constraints.append((y_center - r >= 0) * 1e6)
            constraints.append((y_center + r <= 1) * 1e6)
        # Pairwise distance constraints
        for i in range(len(centers)):
            for j in range(i + 1, len(centers)):
                dist = np.linalg.norm(centers[i] - centers[j])
                constraints.append((dist - (radii[i] + radii[j]) >= -1e-6) * 1e6)
        return np.array(constraints)

    # Initial hexagonal packing
    initial_centers = hexagonal_packing(26, 1, 1, 0.1)
    print("Initial centers shape:", initial_centers.shape)

    # Initial guess for radii (based on max possible radius for each circle)
    initial_radii = np.array([max_radius(initial_centers, i) for i in range(26)])
    print("Initial radii:", initial_radii)

    # Flatten the radii for optimization
    x0 = initial_radii.flatten()

    # Define bounds for radii (non-negative)
    bounds = [(0, None) for _ in range(26)]

    # Perform optimization
    result = opt.minimize(
        fun=objective,
        x0=x0,
        bounds=bounds,
        method='SLSQP',
        constraints={
            'type': 'ineq',
            'fun': lambda x: constraints(x, initial_centers),
            'jac': None
        }
    )

    # Extract the optimized radii and centers
    optimized_radii = result.x.reshape(-1, 1)
    optimized_centers = initial_centers

    # Validate the packing

    if not valid:
        raise ValueError(f"Packing is invalid: {message}")

    # Return the results
    return optimized_centers, optimized_radii.flatten(), np.sum(optimized_radii)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
