import scipy.optimize as optimize

def run_packing():

    def generate_initial_guess(n_circles):

        cols = (n_circles + rows - 1) // rows
        centers = np.zeros((n_circles, 2))
        for i in range(n_circles):
            row = i // cols
            col = i % cols
            x = col + 0.5 * (row % 2)
            y = row * (np.sqrt(3)/2) + 0.5
            x = x / (cols - 1)
            y = y / (rows - 1)
            centers[i] = [x, y]
        return centers

    # Objective function to maximize sum of radii

        return -np.sum(radii)  # Minimize negative sum

    def constraint_square(radii, centers):
        return np.array([np.clip(centers[i, 0] - radii[i], -1e-12, 1 + 1e-12) for i in range(len(radii))]) \
               + np.array([np.clip(centers[i, 1] - radii[i], -1e-12, 1 + 1e-12) for i in range(len(radii))])

    def constraint_overlap(radii, centers):
        n = len(radii)
        dists = np.zeros((n, n))
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dists[i, j] = np.sqrt(dx*dx + dy*dy)
                dists[j, i] = dists[i, j]
        return np.array([dists[i, j] - (radii[i] + radii[j]) for i in range(n) for j in range(n) if i < j])

    centers = generate_initial_guess(n)

    constraints = [

    result = optimize.minimize(
