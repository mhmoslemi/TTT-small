def validate_packing(centers, radii):

        return False, "NaN values present"

        r = radii[i]

def create_individual(n_circles):
    # Generate a random initial guess: positions and radii
    centers = np.random.uniform(0.1, 0.9, (n_circles, 2))
    radii = np.random.uniform(0.01, 0.05, n_circles)
    return np.concatenate((centers.reshape(n_circles * 2), radii))

def evaluate_individual(individual, n_circles):
    centers = individual[:n_circles * 2].reshape(n_circles, 2)
    radii = individual[n_circles * 2:]
    
    sum_radii = -np.sum(radii)
    
    # Boundary penalty
    boundary_penalty = 0
    for i in range(n_circles):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12:
            boundary_penalty += 1e6 * (x - r + 1e-12)
        elif x + r > 1 + 1e-12:
            boundary_penalty += 1e6 * (x + r - 1 - 1e-12)
        if y - r < -1e-12:
            boundary_penalty += 1e6 * (y - r + 1e-12)
        elif y + r > 1 + 1e-12:
            boundary_penalty += 1e6 * (y + r - 1 - 1e-12)
    
    # Overlap penalty
    overlap_penalty = 0
    for i in range(n_circles):

def run_packing():

    # Define the fitness function
