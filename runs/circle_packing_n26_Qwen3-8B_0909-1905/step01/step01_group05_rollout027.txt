```python
import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds

def run_packing():
    n = 26

    # Initial placement: hexagonal packing
    # 6 rows with 4, 4, 4, 4, 4, 6 circles
    # First define the positions of the centers in the grid
    rows = [4, 4, 4, 4, 4, 6]
    y_coords = np.linspace(0.5, 0.5, len(rows))
    x_start = 0.25
    x_step = 0.5 / (np.max(rows) - 1)
    centers = []

    for i, row in enumerate(rows):
        y = y_coords[i]
        row_centers = []
        for j in range(row):
            x = x_start + j * x_step
            row_centers.append([x, y])
        centers.extend(row_centers)

    # Convert to numpy array
    centers = np.array(centers)

    # Initial radii: all equal and scaled to fit in the square
    initial_radii = np.ones(n) * 0.2

    # Define the objective function (minimize negative sum of radii)
    def objective(x):
        r = x[:n]
        return -np.sum(r)

    # Define the constraints
    # Constraint 1: Circle centers are within the square
    def constraint1(x):
        r = x[:n]
        c = x[n:]
        for i in range(n):
            if c[i] < 0 or c[i] > 1:
                return np.inf
        return 0

    # Constraint 2: No circles overlap
    def constraint2(x):
        r = x[:n]
        c = x[n:]
        for i in range(n):
            for j in range(i + 1, n):
                dx = c[i*2] - c[j*2]
                dy = c[i*2 + 1] - c[j*2 + 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < r[i] + r[j] - 1e-8:
                    return np.inf
        return 0

    # Combine constraints
    constraints = [
        {'type': 'ineq', 'fun': constraint1},
        {'type': 'ineq', 'fun': constraint2}
    ]

    # Set bounds for the centers and radii
    bounds = Bounds(
        lb=np.zeros(2 * n + n),
        ub=np.ones(2 * n + n)
    )

    # Use the initial guess
    initial_guess = np.concatenate([centers.flatten(), initial_radii])

    # Solve the optimization
    result = optimize.minimize(
        fun=objective,
        x0=initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        tol=1e-10,
        options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-8}
    )

    # Extract the centers and radii
    centers_opt = result.x[n:2*n].reshape(n, 2)
    radii_opt = result.x[2*n:3*n]

    # Validate the packing
    valid, message = validate_packing(centers_opt, radii_opt)

    if not valid:
        raise RuntimeError(message)

    sum_radii = np.sum(radii_opt)

    return centers_opt, radii_opt, sum_radii
```

```python
def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
```
