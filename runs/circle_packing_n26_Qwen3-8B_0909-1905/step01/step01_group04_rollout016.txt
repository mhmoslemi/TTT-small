import numpy as np
import scipy.optimize as opt

def run_packing():
    n = 26
    # Initial guess: Hexagonal packing in the center
    # Let's place circles in a hexagonal grid in the center of the square
    # Initial spacing: 0.2 (this can be adjusted for better results)
    spacing = 0.2
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Calculate initial positions and radii
    # We start with hexagonal grid of 5 rows
    rows = 5
    cols = 5
    for i in range(rows):
        for j in range(cols):
            x = j * spacing
            y = i * spacing + (j % 2) * spacing / 2
            # Adjust to center the grid
            x = x - (cols - 1) * spacing / 2 + 0.5
            y = y - (rows - 1) * spacing / 2 + 0.5
            centers[i * cols + j] = np.array([x, y])
            radii[i * cols + j] = spacing / 2

    # Optimization settings
    bounds = []
    for _ in range(n):
        bounds.append((0, 1))
    for _ in range(n):
        bounds.append((0, 1))

    # Objective function: maximize sum of radii
    def objective(radii_vector):
        return -np.sum(radii_vector)

    # Constraint: no overlapping circles
    def constraint_overlap(x):
        centers_vector = x[:n*2].reshape(n, 2)
        radii_vector = x[n*2:].reshape(n)
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers_vector[i, 0] - centers_vector[j, 0]
                dy = centers_vector[i, 1] - centers_vector[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii_vector[i] + radii_vector[j] - 1e-8:
                    return -1
        return 0

    # Constraint: each circle must be within the unit square
    def constraint_bounds(x):
        centers_vector = x[:n*2].reshape(n, 2)
        radii_vector = x[n*2:].reshape(n)
        for i in range(n):
            x_i, y_i = centers_vector[i]
            r_i = radii_vector[i]
            if x_i - r_i < 0 or x_i + r_i > 1 or y_i - r_i < 0 or y_i + r_i > 1:
                return -1
        return 0

    # Combine all constraints into a single function
    def constraint(x):
        c1 = constraint_overlap(x)
        c2 = constraint_bounds(x)
        return c1 + c2

    x0 = np.zeros(n * 2 + n)

    result = opt.minimize(

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
