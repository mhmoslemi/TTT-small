import numpy as np
import scipy.optimize as optimize
from concurrent.futures import ThreadPoolExecutor

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def objective_function(params, n_circles):
    centers = params[:n_circles*2].reshape(n_circles, 2)
    radii = params[n_circles*2:]
    
    sum_radii = -np.sum(radii)
    
    boundary_penalty = 0
    for i in range(n_circles):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12:
            boundary_penalty += 1e5 * (x - r + 1e-12)
        if x + r > 1 + 1e-12:
            boundary_penalty += 1e5 * (x + r - 1 - 1e-12)
        if y - r < -1e-12:
            boundary_penalty += 1e5 * (y - r + 1e-12)
        if y + r > 1 + 1e-12:
            boundary_penalty += 1e5 * (y + r - 1 - 1e-12)
    
    overlap_penalty = 0
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                overlap_penalty += 1e5 * (radii[i] + radii[j] - dist - 1e-12)
    
    return sum_radii + boundary_penalty + overlap_penalty

def generate_initial_guess_hexagonal(n_circles):
    centers = np.zeros((n_circles, 2))
    radii = np.zeros(n_circles)
    col = 0
    row = 0
    for i in range(n_circles):
        col = i % 5
        row = i // 5
        x = col / 4.0 + 0.1
        y = row / 4.0 + 0.1
        centers[i, 0] = x
        centers[i, 1] = y
        radii[i] = 0.05
    return centers, radii

def generate_initial_guess_random(n_circles):
    centers = np.random.uniform(0.1, 0.9, (n_circles, 2))
    radii = np.random.uniform(0.03, 0.05, n_circles)
    return centers, radii

def run_packing():
    n_circles = 26
    n_params = n_circles * 2 + n_circles
    
    initial_guesses = [
        generate_initial_guess_hexagonal(n_circles),
        generate_initial_guess_random(n_circles)
    ]
    
    results = []
    
    def optimize_packing(initial_guess):
        centers = initial_guess[0]
        radii = initial_guess[1]
        initial_params = np.concatenate((centers.reshape(n_circles * 2), radii))
        bounds = []
        for i in range(n_circles):
            bounds.append((0.0, 1.0))  # x
            bounds.append((0.0, 1.0))  # y
            bounds.append((0.0, 1.0))  # radius
        result = optimize.minimize(
            fun=lambda x: objective_function(x, n_circles),
            x0=initial_params,
            bounds=bounds,
            method='Nelder-Mead',
            options={'maxiter': 200, 'tol': 1e-8}
        )
        centers_final = result.x[:n_circles * 2].reshape(n_circles, 2)

        sum_radii = np.sum(radii_final)
        return (centers_final, radii_final, sum_radii)
    
    with ThreadPoolExecutor(max_workers=2) as executor:
        results = list(executor.map(optimize_packing, initial_guesses))
    
    best_result = max(results, key=lambda x: x[2])
    return best_result[0], best_result[1], best_result[2]
