import scipy.optimize as opt
from typing import Tuple

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    np.random.seed(42)
    
    # Initial guess: place circles in a hexagonal grid within the unit square
    # We'll use a hybrid packing: most circles in a grid, some in corners/edges
    n_circles = 26
    centers = np.zeros((n_circles, 2))
    radii = np.zeros(n_circles)

    # Hexagonal grid
    rows = 5
    cols = 5
    spacing = 0.25

    for i in range(rows):
        for j in range(cols):
            center_idx = i * cols + j
            x = j * spacing
            y = i * spacing * np.sqrt(3) / 2
            centers[center_idx] = [x, y]
            radii[center_idx] = 0.05
    
    # Add a few circles to the corners and edges
    # Bottom-left corner
    centers[25] = [0.05, 0.05]
    radii[25] = 0.05
    # Bottom-right corner
    centers[24] = [0.95, 0.05]
    radii[24] = 0.05
    # Top-left corner
    centers[23] = [0.05, 0.95]
    radii[23] = 0.05
    # Top-right corner
    centers[22] = [0.95, 0.95]
    radii[22] = 0.05
    
    # Now, optimize the configuration
    def objective(radii_flat):
        return -np.sum(radii_flat)

    def constraints(radii_flat):
        radii = radii_flat.reshape(-1)
        centers = np.zeros((n_circles, 2))
        for i in range(n_circles):
            x = centers[i, 0]
            y = centers[i, 1]
            r = radii[i]
            # Enforce boundary constraints
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                return np.inf  # Penalize boundary violations
            # Enforce non-overlap between circles
            for j in range(i + 1, n_circles):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-12:
                    return np.inf  # Penalize overlap
        return 0

    # Optimization parameters
    bounds = [(0.0, 0.1) for _ in range(n_circles)]
    res = opt.minimize(
        objective,
        radii,
        bounds=bounds,
        method='SLSQP',
        constraints={'type': 'ineq', 'fun': constraints},
        tol=1e-6,
        options={'maxiter': 1000}

    centers = np.array(res.x).reshape(-1, 2)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
