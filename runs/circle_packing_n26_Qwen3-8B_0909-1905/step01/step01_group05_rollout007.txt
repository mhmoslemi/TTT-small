        dx = c1[0] - c2[0]

        dist_sq = dx * dx + dy * dy
        return dist_sq < (r1 + r2) ** 2 - 1e-12

        return (c[0] - r >= -1e-12 and c[0] + r <= 1 + 1e-12 and

        return -np.sum(radii)
