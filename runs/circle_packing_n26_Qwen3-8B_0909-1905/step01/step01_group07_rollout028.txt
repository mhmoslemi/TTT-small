import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
import time

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial guess: two rows of circles in hexagonal pattern
    # First row at y = 0.2, spaced horizontally
    radius_guess = 0.05
    num_circles = 26
    centers_guess = np.zeros((num_circles, 2))
    for i in range(num_circles):
        row_index = i // 13
        col_index = i % 13
        x = col_index * (1 + radius_guess * 2) / 13
        y = row_index * (1 + radius_guess * 2) / 2 if row_index == 0 else row_index * (1 + radius_guess * 2) / 2
        x_center = x
        y_center = y
        centers_guess[i, 0] = x_center
        centers_guess[i, 1] = y_center

    # Objective function: maximize sum of radii
    def objective(radii):
        return -np.sum(radii)

    # Constraints
    def constraint_center_bounds(radii):
        return np.concatenate([radii >= 0, centers_guess[:, 0] - radii <= 1, centers_guess[:, 1] - radii <= 1,
                              centers_guess[:, 0] + radii >= 0, centers_guess[:, 1] + radii >= 0])

    def constraint_overlap(radii):
        n = len(radii)
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers_guess[i, 0] - centers_guess[j, 0]
                dy = centers_guess[i, 1] - centers_guess[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return -np.inf
        return np.sum(radii)

    bounds = Bounds(np.zeros(num_circles), np.inf)
    constraints = [
        {'type': 'ineq', 'fun': constraint_center_bounds},
        {'type': 'ineq', 'fun': constraint_overlap}
    ]

    # Initial guess for radii
    radii_guess = np.full(num_circles, radius_guess)

    # Optimization settings
    result = minimize(
        objective,
        radii_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        tol=1e-6,
        options={'maxiter': 1000}
    )

    centers = centers_guess
    radii = result.x
    sum_radii = -result.fun

    # Validate the solution
    is_valid, msg = validate_packing(centers, radii)
    if not is_valid:
        raise ValueError(f"Validation failed: {msg}")

    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
