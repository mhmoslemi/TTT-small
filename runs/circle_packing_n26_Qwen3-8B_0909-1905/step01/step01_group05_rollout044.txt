import numpy as np
import scipy.optimize as opt
from typing import Tuple

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    # Define a grid of 26 points (5x5 grid with 1 point removed)
    # We'll use a 5x5 grid to get 25 points and then add one more to reach 26
    num_circles = 26
    grid = np.array([
        [i/4, j/4] for i in range(5) for j in range(5)
    ])
    centers = grid[:25]
    centers = np.vstack([centers, [0.5, 0.5]])  # Add one at the center

    # Initial radii: assume each circle has a radius equal to the minimum of the distances to the boundary
    # For now, let's use a uniform initial estimate
    radii = np.full(num_circles, 0.1)  # Initial radius is small

    # Define the optimization variables
    x = np.concatenate([centers.flatten(), radii])

    # Define the objective function to maximize the sum of radii
    def objective(x):
        centers = x[:2*num_circles].reshape(num_circles, 2)
        radii = x[2*num_circles:]
        return -np.sum(radii)  # We minimize the negative sum to maximize the sum

    # Define constraints
    def inequality_constraints(x):
        centers = x[:2*num_circles].reshape(num_circles, 2)
        radii = x[2*num_circles:]
        constraints = []
        for i in range(num_circles):
            x_i, y_i = centers[i]
            r_i = radii[i]
            # Distance to each boundary
            dist_left = x_i - r_i
            dist_right = 1 - x_i - r_i
            dist_bottom = y_i - r_i
            dist_top = 1 - y_i - r_i
            # Ensure center stays inside the square
            constraints.append(dist_left >= -1e-12)
            constraints.append(dist_right >= -1e-12)
            constraints.append(dist_bottom >= -1e-12)
            constraints.append(dist_top >= -1e-12)
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                constraints.append(dist - radii[i] - radii[j] >= -1e-12)
        return constraints

    # Bounds: centers in [0,1]^2, radii > 0
    bounds = []
    for _ in range(num_circles):
        bounds.extend([(0.0, 1.0), (0.0, 1.0)])
    for _ in range(num_circles):
        bounds.extend([(0.0, 1.0)])

    # Solve the optimization
    result = opt.minimize(
        fun=objective,
        x0=x,
        bounds=bounds,
        constraints={'type': 'ineq', 'fun': inequality_constraints},
        method='SLSQP',
        options={'ftol': 1e-8, 'maxiter': 1000}
    )

    # Extract the results
    centers = result.x[:2*num_circles].reshape(num_circles, 2)
    radii = result.x[2*num_circles:]
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
