import scipy.optimize as optimize

def run_packing():

    n = 26

    centers = np.zeros((n, 2))
    radii = np.full(n, radius_initial)

    # Hexagonal packing setup
    rows = int(np.ceil(np.sqrt(n)))
    cols_per_row = np.ceil(n / rows).astype(int)
    row_height = 2 * radius_initial * np.sqrt(3)  # Height of each row
    col_width = 2 * radius_initial  # Width of each column

    col_idx = 0
    row_idx = 0
    for i in range(n):
        x = col_idx * col_width
        y = row_idx * row_height
        if i % 2 == 1:
            x += col_width / 2
        centers[i, 0] = x
        centers[i, 1] = y
        col_idx += 1
        if col_idx == cols_per_row:
            col_idx = 0
            row_idx += 1

    # Define objective function
    def objective(radii_vec):
        return -np.sum(radii_vec)  # Negative for maximization

    # Define constraints
    def constraint_boundary(x):
        return x

    def constraint_overlap(x):
        centers = x[:n, :2]
        radii = x[n:]
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return -1
        return 0

    # Bounds: centers in [0,1], radii >= 0
    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0))  # x

    # Initial guess
    initial_guess = np.zeros(2 * n + n)
    for i in range(n):

        initial_guess[2 * i + 1] = centers[i, 1]

    # Optimization
    result = optimize.minimize(

        initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=[

            {'type': 'ineq', 'fun': constraint_overlap}
        ]

    centers_opt = result.x[:2 * n].reshape(n, 2)
