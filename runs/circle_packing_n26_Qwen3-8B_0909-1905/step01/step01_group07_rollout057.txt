import numpy as np
import cvxpy as cp
from typing import Tuple

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    # Initial hexagonal grid-based configuration
    num_circles = 26
    # We roughly divide the space into 5 rows, each with 5 or 6 circles
    # Initial grid shape is roughly 5x5 with some overlap at the top
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)

    # Start building the hexagonal grid
    for i in range(num_circles):
        row = i // 5
        col = i % 5
        # Hexagonal grid spacing
        x = col * (1.0 / 3.5)
        y = row * (np.sqrt(3) / 2.5)
        # Adjust for containment
        if row % 2 == 1:
            x += 0.25
        centers[i, 0] = x
        centers[i, 1] = y
        radii[i] = 0.15

    # Now optimize this configuration
    # Objective: maximize sum of radii
    # Constraints: circles are within the square, radii non-negative, no overlap

    # Optimization variables
    x = cp.Variable((num_circles, 2))
    r = cp.Variable(num_circles, nonneg=True)

    # Objective function
    obj = cp.Maximize(cp.sum(r))

    # Constraints
    constraints = []
    for i in range(num_circles):
        constraints.append(cp.pnorm(x[i] - np.array([0.5, 0.5]), 2) <= 0.5 + r[i])
        constraints.append(cp.pnorm(x[i] - np.array([0.0, 0.5]), 2) <= 0.5 + r[i])
        constraints.append(cp.pnorm(x[i] - np.array([0.5, 0.0]), 2) <= 0.5 + r[i])
        constraints.append(cp.pnorm(x[i] - np.array([0.0, 0.0]), 2) <= 0.5 + r[i])
        constraints.append(cp.pnorm(x[i] - np.array([1.0, 0.5]), 2) <= 0.5 + r[i])
        constraints.append(cp.pnorm(x[i] - np.array([0.5, 1.0]), 2) <= 0.5 + r[i])
        constraints.append(cp.pnorm(x[i] - np.array([1.0, 1.0]), 2) <= 0.5 + r[i])
        constraints.append(cp.pnorm(x[i] - np.array([1.0, 0.0]), 2) <= 0.5 + r[i])

        # Fallback to initial configuration if optimization fails

    # Extract the optimized centers and radii
    optimized_centers = np.reshape(x.value, (num_circles, 2))

    if not is_valid:

    return optimized_centers, optimized_radii, np.sum(optimized_radii)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
