import numpy as np
import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial guess: a hexagonal packing in a square
    n = 26
    max_radius = 0.1  # Initial estimate for max radius

    # Compute hexagonal grid layout
    rows = int(np.sqrt(n))
    cols = n // rows + (1 if n % rows != 0 else 0)

    # Create grid of positions
    positions = np.zeros((n, 2))
    for i in range(n):
        row = i // cols
        col = i % cols
        x = col + 0.5 * (row % 2)
        y = row * np.sqrt(3) / 2
        positions[i] = [x, y]

    # Scale to unit square
    positions -= positions.min()
    positions /= positions.max()

    # Initial guess for radii
    radii = np.ones(n) * max_radius

    # Objective function: maximize total radius
    def objective(radii_vector):
        return -np.sum(radii_vector)  # Minimize negative of sum

    # Constraints
    def constraint_overlap(x, y, r1, r2):
        return np.sqrt((x[0] - x[1])**2 + (y[0] - y[1])**2) - (r1 + r2)

    # Pack constraints
    constraints = []

    for i in range(n):
        for j in range(i + 1, n):
            def cons(x, i=i, j=j):
                return constraint_overlap(x[i * 2:(i + 1) * 2], x[j * 2:(j + 1) * 2], radii[i], radii[j])
            constraints.append({'type': 'ineq', 'fun': cons})
    
    def constraint_boundary(x, i=i):
        return np.array([x[i * 2] - radii[i], 1.0 - x[i * 2], x[i * 2 + 1] - radii[i], 1.0 - x[i * 2 + 1]])
    
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: constraint_boundary(x, i)[0]})
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: constraint_boundary(x, i)[1]})
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: constraint_boundary(x, i)[2]})
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: constraint_boundary(x, i)[3]})

    # Bounds for positions and radii
    bounds = []
    for i in range(n):
        bounds.append((0, 1))  # x
        bounds.append((0, 1))  # y
        bounds.append((0, 1))  # r

    # Optimization
    result = opt.minimize(
        fun=objective,
        x0=np.concatenate((positions.flatten(), radii)),
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 1000, 'ftol': 1e-9, 'disp': False}
    )

    # Extract results
    centers = result.x[:n * 2].reshape(n, 2)
    radii = result.x[n * 2:]
    sum_radii = np.sum(radii)

    # Validate result
    valid, message = validate_packing(centers, radii)
    if not valid:
        raise RuntimeError(f"Validation failed: {message}")

    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
