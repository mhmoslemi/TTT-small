from scipy.optimize import Bounds

def objective_function(x, n):

    radii = x[2*n:]
    return -np.sum(radii)  # Maximize sum of radii

def build_constraints(n):
    constraints = []
    for i in range(n):
        # Center of circle i must lie within the unit square
        constraints.append({
            'type': 'ineq',
            'fun': lambda x, i=i: x[2*i] - 1e-12  # x-coordinate >= 0

def refine_edges(centers, radii):

    for i in range(centers.shape[0]):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0:

            centers[i, 0] = 1 - r
        if y - r < 0:
            centers[i, 1] = r
        elif y + r > 1:
            centers[i, 1] = 1 - r
    return centers

def validate_packing(centers, radii):
