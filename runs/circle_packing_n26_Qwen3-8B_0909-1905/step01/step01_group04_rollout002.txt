    radius_initial = 0.08  # Initial radius for all circles

    hex_grid_spacing = 2 * radius_initial * math.sqrt(3)  # Horizontal spacing

        centers[i] = [x, y]

        radii = x_flat[:, 2]

        obj = np.sum(radii)
        return -obj  # Minimize the negative to maximize

    # 2. No circles can overlap
    # 3. Radii must be non-negative
