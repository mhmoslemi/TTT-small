import scipy.optimize as opt

def run_packing():

    max_rad = max_r

            x = (i + j) * (np.sqrt(3)/2) * max_rad

            idx = i * (i + 1) // 2 + j
            if idx < n:
                centers[idx] = np.array([x, y])

    # Adjust the hexagonal grid to fit between 0 and 1
    # Approximate center positions
    centers = (centers - centers.min(axis=0)) / (centers.max(axis=0) - centers.min(axis=0))
    centers = np.clip(centers, 0, 1)

    # Define objective function (negative of sum of radii to minimize)
    def objective_function(radii):
        return -np.sum(radii)

    # Define constraints
    def constraint_boundary(radii, centers):
        constraints = []
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-6 or x + r > 1 + 1e-6:
                constraints.append(-1)

                constraints.append(-1)
        return np.array(constraints)

                if dist < radii[i] + radii[j] - 1e-6:
                    constraints.append(-1)
        return np.array(constraints)

    bounds = [(0, 1) for _ in range(n)]
