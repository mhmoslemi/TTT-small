import scipy.optimize as optimize

    col_count = 6

                y = i * (np.sqrt(3) * radius_initial)
            centers[row_idx] = [x, y]

    def objective(x):
        centers_flat = x[:2 * num_circles].reshape(num_circles, 2)
        radii_flat = x[2 * num_circles:]
        return -np.sum(radii_flat)

    # Define constraints

        centers_flat = x[:2 * num_circles].reshape(num_circles, 2)
