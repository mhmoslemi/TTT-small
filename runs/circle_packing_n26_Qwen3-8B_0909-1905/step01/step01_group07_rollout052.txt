import scipy.optimize as opt
from functools import partial

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # First layer (outermost) circles
    for i in range(16):  # 16 outermost circles
        angle = 2 * np.pi * i / 16
        r = 0.18
        x = 0.5 + r * np.cos(angle)
        y = 0.5 + r * np.sin(angle)
        centers[i] = [x, y]
        radii[i] = r

    # Second layer (inner circles)

        angle = 2 * np.pi * i / 10
        r = 0.15
        x = 0.5 + r * np.cos(angle)
        y = 0.5 + r * np.sin(angle)
        centers[i] = [x, y]
        radii[i] = r

    # Now perform an optimization to refine the configuration

    # Objective function to maximize total radius
    def objective(radii):
        return -np.sum(radii)  # negative for minimization

    # Constraint: all circles within [0,1]^2
    def boundary_constraints(radii, centers):
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                return False
        return True

    # Constraint: circles do not overlap
    def non_overlap_constraints(radii, centers):
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j]:
                    return False
        return True

    # Optimization bounds
    bounds = [(0.0, 0.5) for _ in range(n)]

    # Constraints in the form required by scipy.optimize
    def constraints(radii):
        if not boundary_constraints(radii, centers):
            return np.array([1.0])  # Bad constraint
        if not non_overlap_constraints(radii, centers):
            return np.array([1.0])  # Bad constraint
        return np.array([0.0])

    # Run the optimization
    result = opt.minimize(
        fun=objective,
        x0=radii,
        bounds=bounds,
        constraints=partial(constraints, centers=centers),
        method='SLSQP',
        options={'maxiter': 500, 'ftol': 1e-9, 'eps': 1e-8}

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
