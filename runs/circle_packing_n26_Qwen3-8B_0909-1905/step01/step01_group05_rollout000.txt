import numpy as np
from scipy.optimize import minimize
import matplotlib.pyplot as plt

def run_packing():
    # Number of circles
    n = 26

    # Initial guess: hexagonal grid with circles packed in a staggered formation
    # We will place 6 rows with 5, 5, 5, 5, 5, and 6 circles in each row
    # Initial grid spacing for hexagonal packing
    spacing = 0.3

    # Calculate positions in a hexagonal grid
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Place 6 rows of circles
    for row in range(6):
        for col in range(5 if row % 2 == 0 else 6):
            x = col * spacing
            y = row * spacing * np.sqrt(3) / 2
            centers[row * 5 + col] = [x, y]
            radii[row * 5 + col] = 0.1  # Initial radius

    # Define the optimization function
    def objective(radii):
        # Objective: maximize the sum of radii
        return -np.sum(radii)  # Negative for minimization

    # Define the constraints
    def constraints_func(radii):
        # Constraints:
        # 1. All circles must be within [0,1]^2
        # 2. All circles must be non-overlapping
        # 3. All radii must be non-negative

        # Convert radii to numpy array
        radii_array = np.array(radii)

        # Apply constraints
        # 1. Check boundary constraints
        for i in range(n):
            x, y = centers[i]
            r = radii_array[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                return np.inf

        # 2. Check non-overlapping constraints
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < radii_array[i] + radii_array[j] - 1e-12:
                    return np.inf

        # 3. Check non-negative radii
        for r in radii_array:
            if r < 0:
                return np.inf

        return -np.sum(radii_array)  # Negative for minimization

    # Optimization
    result = minimize(
        objective,
        radii,
        method='SLSQP',
        bounds=[(0, 1.0) for _ in range(n)],
        constraints=[{'type': 'ineq', 'fun': constraints_func}],
        options={'maxiter': 1000, 'ftol': 1e-12, 'disp': False}
    )

    # Extract the result
    optimized_radii = result.x
    optimized_centers = centers.copy()

    # Compute the sum of radii
    sum_radii = np.sum(optimized_radii)

    # Display the packing (optional)
    plt.figure(figsize=(6, 6))
    for i in range(n):

    plt.xlim(0, 1)
    plt.ylim(0, 1)
    plt.axis('equal')
    plt.show()

    return optimized_centers, optimized_radii, sum_radii
