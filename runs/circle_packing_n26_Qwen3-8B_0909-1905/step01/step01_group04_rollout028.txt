def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    num_circles = 26
    max_radius = 1 / np.sqrt(2)  # Max radius without overlapping with neighbors

    # Start with a hexagonal lattice
    centers = []
    radii = []

    # Define number of rows and columns for hexagonal packing
    rows = 5
    cols = 5
    if num_circles > rows * cols:
        rows += 1
        cols += 1

    # Determine the grid of hexagonal points
    for i in range(rows):
        for j in range(cols):
            x = j + 0.5 * i
            y = i
            # Normalize to unit square
            x = x / (cols - 1) if cols > 1 else 0.5
            y = y / (rows - 1) if rows > 1 else 0.5
            centers.append([x, y])
            radii.append(max_radius * 0.9)  # Start with a slightly smaller radius

    # Convert to numpy arrays
    centers = np.array(centers)
    radii = np.array(radii)

    # Define the objective function to maximize sum of radii
    def objective(x):
        return -np.sum(x)  # Minimize negative sum to maximize sum

    # Define the constraints
    def constraint_overlap(x, centers, radii):

                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < x[i] + x[j] - 1e-8:
                    return -1  # Violation of non-overlap
        return 0

    def constraint_boundary(x, centers, radii):
        for i in range(len(x)):
            x_pos = centers[i, 0]
            y_pos = centers[i, 1]
            r = x[i]
            if x_pos - r < 0 or x_pos + r > 1 or y_pos - r < 0 or y_pos + r > 1:
                return -1  # Violation of boundary constraints
        return 0

    # Define the bounds for radii (non-negative)
    bounds = [(0, 1) for _ in range(num_circles)]

    # Optimization
    result = optimize.minimize(
        fun=objective,
        x0=radii,
        method='SLSQP',
        bounds=bounds,
        constraints=[

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
