import scipy.optimize as optimize

def run_packing():

    n = 26
    
    # Initial guess for centers (hexagonal packing)
    max_rows = int(np.sqrt(n))
    centers = np.zeros((n, 2))
    idx = 0
    for row in range(max_rows):
        for col in range(n // max_rows):
            if row % 2 == 0:
                x = col * (1 + 1 / (n // max_rows))
            else:
                x = col * (1 + 1 / (n // max_rows)) + 0.5 / (n // max_rows)
            y = row * np.sqrt(3) / 2
            centers[idx] = [x, y]
            idx += 1
    
    # Initial guess for radii (uniform)
    radii = np.ones(n) * 0.15

    # Objective function: maximize sum of radii
    def objective(radii):
        return -np.sum(radii)

    # Constraint functions
    def constraint_boundary(radii):
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                return -1
        return 0

    def constraint_overlap(radii):
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-12:
                    return -1
        return 0

    cons = [
        {'type': 'ineq', 'fun': constraint_boundary},
        {'type': 'ineq', 'fun': constraint_overlap}
    ]

    bounds = [(0, 1) for _ in range(n)]

    # Optimization
    result = optimize.minimize(
        fun=objective,
        x0=radii,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        tol=1e-8,
        options={'maxiter': 1000, 'ftol': 1e-9}
    )

    radii_result = result.x

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
