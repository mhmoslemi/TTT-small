```python
    # and adjust radii to maximize sum while satisfying constraints

    n = 26
    # Number of rows
    rows = 6
    # Number of columns
    cols = 5
    # Initial spacing between centers
    spacing = 0.7

    # Initial positions
    centers = np.zeros((n, 2))
    idx = 0
    for row in range(rows):

            x = col * spacing
            y = row * spacing + (col % 2) * spacing / 2  # Offset for hexagonal packing
            centers[idx] = [x, y]
            idx += 1

    # Initial radii
    radii = np.ones(n) * 0.1

    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize

    def constraint_center_bounds(x):
        return np.array([x[0] - x[2] >= -1e-12, x[0] + x[2] <= 1 + 1e-12,

        for i in range(n):

                if dists[i, j] < radii[i] + radii[j] - 1e-12:
                    return np.array([0.0])
        return np.array([1.0])

    x0 = np.concatenate([centers.flatten(), radii])

        bounds.extend([(0, 1), (0, 1)])
```

```python
def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
```
