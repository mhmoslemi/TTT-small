import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run_packing():
    # Initial guess: 26 evenly spaced circles in a grid
    num_circles = 26
    radius_guess = 0.1 * np.ones(num_circles)
    
    # Grid layout: 5 rows and 5 columns (with one extra circle)
    grid = np.zeros((num_circles, 2))
    idx = 0
    for i in range(5):
        for j in range(5):
            grid[idx, 0] = j / 4.0
            grid[idx, 1] = i / 4.0
            idx += 1
    # Add an extra circle in the center
    grid[25, 0] = 0.5
    grid[25, 1] = 0.5
    
    centers = grid
    radii = radius_guess
    
    # Objective function: maximize sum of radii (minimize negative sum)
    def objective(x):
        return -np.sum(x)
    
    # Optimization bounds: radii must be non-negative and circles fit in [0,1]^2
    bounds = Bounds(0, np.inf)
    
    # Constraints: circles stay within the unit square and no overlap
    def constraint_square(x):
        # x is a flat array of all centers and radii
        # First 2*num_circles entries are centers, then the last num_circles are radii
        centers_flat = x[:2*num_circles].reshape(num_circles, 2)
        radii_flat = x[2*num_circles:]
        
        # Check that all circles are within [0,1]^2
        for i in range(num_circles):
            x_center, y_center = centers_flat[i]
            r = radii_flat[i]
            if not (0 <= x_center - r and x_center + r <= 1 and 0 <= y_center - r and y_center + r <= 1):
                return np.inf  # Penalize invalid positions
        
        # Check that no circles overlap
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers_flat[i, 0] - centers_flat[j, 0]
                dy = centers_flat[i, 1] - centers_flat[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii_flat[i] + radii_flat[j] - 1e-12:
                    return np.inf  # Penalize overlap
        
        return 0  # No constraints violated
    
    # Optimization: minimize negative sum of radii
    x0 = np.concatenate([centers.flatten(), radii])
    result = minimize(
        objective,

        bounds=bounds,
        constraints={'type': 'ineq', 'fun': constraint_square},

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
