import numpy as np
import cvxpy as cp

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Number of circles
    n = 26

    # Generate an initial hexagonal grid of 26 points in the unit square
    def generate_hexagonal_grid(n):
        # Find the approximate dimensions for a hexagonal packing
        cols = int(np.sqrt(2 * n))
        rows = int(np.ceil(n / cols))
        
        # Intial positions
        centers = np.zeros((n, 2))
        idx = 0
        for row in range(rows):
            for col in range(cols):
                x = col * (1 + 1/np.sqrt(3)) / (cols - 0.5)
                y = row * (np.sqrt(3) / 2) / (rows - 0.5)
                if row % 2 == 1:
                    x += 0.5 / (cols - 0.5)
                centers[idx] = [x, y]
                idx += 1

        # Scale the centers to fit within [0, 1] x [0, 1]
        centers /= max(np.max(centers), 1.0)
        return centers

    centers = generate_hexagonal_grid(n)
    radii = np.full(n, 0.05)

    # Optimize the radius values with constraints
    x = cp.Variable((n, 2))
    r = cp.Variable(n)

    # Constraints
    constraints = []

    # Bound the centers within [0,1]x[0,1]
    constraints += [0 <= x[i, 0] <= 1 for i in range(n)]
    constraints += [0 <= x[i, 1] <= 1 for i in range(n)]

    # All radii non-negative
    constraints += [r[i] >= 0 for i in range(n)]

    # Ensure circles do not overlap
    for i in range(n):
        for j in range(i + 1, n):
            dist = cp.sqrt((x[i, 0] - x[j, 0])**2 + (x[i, 1] - x[j, 1])**2)
            constraints += [dist >= r[i] + r[j]]

    # Ensure circles are inside the square
    for i in range(n):
        constraints += [x[i, 0] - r[i] >= -1e-8]
        constraints += [x[i, 0] + r[i] <= 1 + 1e-8]
        constraints += [x[i, 1] - r[i] >= -1e-8]
        constraints += [x[i, 1] + r[i] <= 1 + 1e-8]

    # Objective: Maximize the sum of radii
    obj = cp.Maximize(cp.sum(r))

    # Solve the optimization problem
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.SCS, verbose=False)

    # Extract the results
    optimal_centers = x.value
    optimal_radii = r.value
    sum_radii = np.sum(optimal_radii)

    # Validate the packing
    valid, msg = validate_packing(optimal_centers, optimal_radii)
    if not valid:
        raise ValueError(f"Packing validation failed: {msg}")

    return optimal_centers, optimal_radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
