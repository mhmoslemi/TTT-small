import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing():
    # Initial estimate for number of circles
    n = 26
    # Initial guess for circle centers (hexagonal packing)
    # We'll choose a hexagonal grid near the center
    # Create a grid with 5 rows and 6 columns, with some adjustment for square boundary
    
    # Define center of square
    center_x, center_y = 0.5, 0.5
    
    # Generate hexagonal grid
    # Define spacing and offsets
    spacing = 0.2
    offset_x = spacing * np.sqrt(3) / 2
    offset_y = spacing / 2
    
    # Create grid
    centers = []
    for i in range(5):
        for j in range(6):
            x = center_x + offset_x * (j - 2.5)
            y = center_y + offset_y * (i - 2)
            centers.append([x, y])
    
    # Add a few extra circles near the corners to fill gaps
    extra_centers = [
        [0.1, 0.1],
        [0.9, 0.1],
        [0.1, 0.9],
        [0.9, 0.9],
        [0.4, 0.95],
        [0.6, 0.95]
    ]
    centers += extra_centers
    
    # Convert to numpy arrays
    centers = np.array(centers)
    
    # Initial guess for radii (all circles have the same radius for simplicity)
    radii = np.ones(n) * 0.05
    
    # Objective function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum
    
    # Constraints
    def constraint_overlap(radii, centers):
        dist_matrix = cdist(centers, centers)
        for i in range(len(centers)):
            for j in range(i + 1, len(centers)):
                if dist_matrix[i, j] < radii[i] + radii[j] - 1e-8:
                    return False
        return True
    
    def constraint_boundary(radii, centers):
        for i in range(len(centers)):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-8 or x + r > 1.0 + 1e-8 or y - r < -1e-8 or y + r > 1.0 + 1e-8:
                return False
        return True
    
    # Combine constraints into a single function
    def constraint(x):
        radii = x
        if not constraint_boundary(radii, centers):
            return np.inf
        if not constraint_overlap(radii, centers):
            return np.inf
        return 0
    
    # Bounds for radii (must be non-negative)
    bounds = [(0, 0.5) for _ in range(n)]
    
    # Optimization
    result = minimize(
        objective,
        radii,
        bounds=bounds,
        constraints={"type": "ineq", "fun": constraint},
        method="SLSQP",
        options={"ftol": 1e-9, "maxiter": 1000}
    )
    
    # Extract results
    best_radii = result.x

    best_sum = np.sum(best_radii)

    if not is_valid:
        raise ValueError(f"Packing is invalid due to: {message}")
    
    return best_centers, best_radii, best_sum

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
