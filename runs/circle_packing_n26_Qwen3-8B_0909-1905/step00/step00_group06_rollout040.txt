import scipy.optimize as optimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    n_circles = 26

    # Use a grid of hexagonal circles, evenly spaced
    def initial_hexagonal_grid(n, side_length):
        rows = int(np.ceil(np.sqrt(n)))
        cols = int(np.ceil(n / rows))

                x = j * (2 * side_length) + (side_length * 0.5)

                    x -= side_length
                centers.append([x, y])
        return np.array(centers)

    side_length = 0.15

    initial_radii = np.ones(n_circles) * 0.05

        radii = x[2 * n_circles:]

        centers = x[:2 * n_circles].reshape(n_circles, 2)
        radii = x[2 * n_circles:]
        return np.array([(x - r < -1e-12 or x + r > 1 + 1e-12) for x, y, r in zip(centers[:, 0], centers[:, 1], radii)])

    def no_overlap(x):
        centers = x[:2 * n_circles].reshape(n_circles, 2)

        max_dist = np.max(radii[:, np.newaxis] + radii[np.newaxis, :])
        return min_dist - max_dist + 1e-12

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
