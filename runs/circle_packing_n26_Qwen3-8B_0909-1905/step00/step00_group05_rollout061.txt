import scipy.optimize as opt
import scipy.spatial.distance as dist
from typing import Tuple

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    # Initialize 26 circles with equal radii and positions based on hexagonal packing
    num_circles = 26
    initial_radius = 0.05  # Start with a low radius to avoid overlap
    radii = np.full(num_circles, initial_radius)
    
    # Generate hexagonal packing grid
    rows = int(np.ceil(np.sqrt(num_circles)))
    cols = int(np.ceil(num_circles / rows))
    centers = np.zeros((num_circles, 2))
    for i in range(num_circles):
        row = i // cols
        col = i % cols

        centers[i] = [x, y]

    def objective_function(radii):

        return -np.sum(radii)  # Minimize negative sum for optimization

    def constraints_function(radii, centers):

        n = len(radii)
        dist_matrix = dist.cdist(centers, centers)

                    return np.inf  # Penalize overlap
        return 0

        return [(0, 1.0) for _ in radii]

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
