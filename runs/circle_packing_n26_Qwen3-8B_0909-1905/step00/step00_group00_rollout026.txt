import scipy.optimize as opt
from typing import Tuple

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:

    # Each row will be offset by half the diameter (diameter = 2r) of the previous row
    # Start with a radius of 0.05 (arbitrary small value)
    r_init = 0.05
    num_circles = 26

    # How many rows to fit in the square?
    # Assuming a hexagonal packing, the number of rows in a square is approximately sqrt(2)
    num_rows = int(np.sqrt(num_circles))
    num_cols = num_rows
    # We'll allow one more row if the number of circles exceeds num_rows^2
    if num_rows * num_rows < num_circles:
        num_rows += 1
    num_cols = num_rows if num_rows * num_rows >= num_circles else num_rows + 1

    # Compute spacing between centers
    spacing = 2 * r_init
    # Set up the grid
    centers = np.zeros((num_circles, 2))
    index = 0
    for i in range(num_rows):
        for j in range(num_cols):
            if index >= num_circles:
                break
            x = j * spacing
            y = i * spacing * np.sqrt(3) / 2

                continue
            centers[index] = [x, y]
            index += 1
    # Add remaining circles around the grid
    while index < num_circles:
        x = np.random.rand()
        y = np.random.rand()
        if x + r_init > 1 or y + r_init > 1:
            continue
        centers[index] = [x, y]
        index += 1

    # Let's now optimize the centers and radii
    # Objective function to maximize the sum of radii
    def objective(x):
        # x contains flattened centers and radii

        sum_radii = np.sum(radii_flat)
        return -sum_radii  # Minimize the negative sum

    # Constraints: circle centers must be within [0,1]^2
    def constraint_centers(x):
        centers_flat = x[:2 * num_circles].reshape(num_circles, 2)
        return np.array([x[0] for x in centers_flat]) - 1e-12  # Ensure centers are > 0

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
