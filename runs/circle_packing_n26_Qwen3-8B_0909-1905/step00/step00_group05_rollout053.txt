import scipy.optimize as opt
from scipy.spatial.distance import cdist

def run_packing():

    num_circles = 26
    # Determine rows and columns (rough estimate for hexagonal packing)
    rows = int(np.ceil(np.sqrt(num_circles)))
    cols = int(np.ceil(num_circles / rows))

    # Initial spacing between circles
    spacing = 1.0 / (cols + 1)  # columns (x-direction)
    spacing_y = 1.0 / (rows + 1)  # rows (y-direction)

    # Initial centers based on hexagonal packing
    centers = np.zeros((num_circles, 2))
    for i in range(num_circles):

        return -np.sum(x)  # Minimize negative of sum of radii

        return np.array([0.0, 0.0, 1.0, 1.0])  # Ensure all centers lie within [0,1]^2

        return np.array([0.0, 0.0])  # Ensure all radii are non-negative

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12

            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
