from scipy.optimize import minimize
from scipy.optimize import Bounds

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    n_circles = 26

    rows = int(np.sqrt(n_circles))
    cols = n_circles // rows + (1 if n_circles % rows else 0)
    centers = np.zeros((n_circles, 2))
    r = np.full(n_circles, radius_initial)

        row = i // cols
        col = i % cols
        x = 0.5 + col * (1 / (cols - 1)) - 0.5

        centers[i, 0] = x
        centers[i, 1] = y

    x = np.reshape(centers, (n_circles * 2, 1))
    r_opt = r.copy()

        return -np.sum(r_opt)  # Minimize negative of sum

    def constraint_circle_to_bound(x_flat):
