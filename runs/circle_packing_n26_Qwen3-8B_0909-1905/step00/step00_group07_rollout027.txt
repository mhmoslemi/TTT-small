import scipy.optimize as opt
import math

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial estimate for centers using hexagonal packing
    n = 26
    centers = np.zeros((n, 2))
    radius_initial = 0.05
    radii = np.full(n, radius_initial)

    # Generate a hexagonal grid of circles in the unit square
    # Create a hexagonal grid with rows and columns
    rows = int(math.ceil(math.sqrt(n)))
    cols = int(math.ceil(n / rows))

    # Calculate the minimum radius that allows hexagonal packing
    spacing = 0.75 * (1 - 2 * radius_initial)
    for i in range(n):
        col = i % cols
        row = i // cols
        x = col * spacing + radius_initial
        y = row * (spacing * math.sqrt(3) / 2) + radius_initial
        if y > 1 - radius_initial:
            y = 1 - radius_initial
        centers[i] = [x, y]
        radii[i] = radius_initial

    # Optimization function to minimize negative sum of radii (i.e., maximize sum of radii)
    def objective(x):
        centers_flat = x[:2*n].reshape(n, 2)
        radii_flat = x[2*n:]

        # Penalize for circles going out of bounds
        out_bound_penalty = 0
        for i in range(n):
            x_i, y_i = centers_flat[i]
            r_i = radii_flat[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12:
                out_bound_penalty += 1e6 * (x_i - r_i + 1e-12) if x_i - r_i < -1e-12 else 1e6 * (x_i + r_i - 1 - 1e-12)
            if y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
                out_bound_penalty += 1e6 * (y_i - r_i + 1e-12) if y_i - r_i < -1e-12 else 1e6 * (y_i + r_i - 1 - 1e-12)

        # Penalize for circle overlap
        overlap_penalty = 0
        for i in range(n):

                dx = centers_flat[i][0] - centers_flat[j][0]
                dy = centers_flat[i][1] - centers_flat[j][1]
                dist = math.hypot(dx, dy)
                if dist < radii_flat[i] + radii_flat[j] - 1e-12:

        return -sum_radii + total_penalty

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
