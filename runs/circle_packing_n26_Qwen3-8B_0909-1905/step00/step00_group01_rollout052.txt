import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Define initial grid layout with hexagonal spacing
    num_circles = 26
    # Grid size parameters
    num_rows = 5
    num_cols = 5
    spacing = 0.2  # Initial horizontal spacing
    offset = spacing / 2  # Hexagonal offset
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)

    # Create base grid
    for i in range(num_rows):
        for j in range(num_cols):
            idx = i * num_cols + j
            if i % 2 == 0:
                x = j * spacing
                y = i * spacing
            else:
                x = j * spacing + offset
                y = i * spacing + offset
            centers[idx] = [x, y]

    # Initial guess for radii: all circles fit in a square
    radii_initial = np.full(num_circles, 0.05)
    radii = radii_initial.copy()

    # Define the objective function: maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize the negative to maximize

    # Define constraints for boundary and overlap
    def constraint_boundary(radii):
        return np.sum((centers[:, 0] - radii) >= 0) + np.sum((1 - centers[:, 0] - radii) >= 0) + \
               np.sum((centers[:, 1] - radii) >= 0) + np.sum((1 - centers[:, 1] - radii) >= 0)

    def constraint_overlap(radii):
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < radii[i] + radii[j] - 1e-8:
                    return -1
        return 0

    # Bounds for radii: non-negative
    bounds = [(0, 1) for _ in range(num_circles)]

    # Set up optimization problem
    constraints = [
        {'type': 'ineq', 'fun': lambda r: constraint_boundary(r)},
        {'type': 'ineq', 'fun': lambda r: constraint_overlap(r)}
    ]

    # Run the optimization
    result = optimize.minimize(
        objective,
        radii,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        tol=1e-10,
        options={"maxiter": 100}
    )

    # Extract the final result
    centers_final = centers.copy()
    radii_final = result.x
    sum_radii = np.sum(radii_final)

    # Validate the packing
    valid, message = validate_packing(centers_final, radii_final)
    if not valid:
        raise ValueError(f"Packing validation failed: {message}")

    return centers_final, radii_final, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
