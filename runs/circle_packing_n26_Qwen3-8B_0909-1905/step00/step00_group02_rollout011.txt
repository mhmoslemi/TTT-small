import numpy as np
import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Define the number of circles
    n_circles = 26

    # Initial guess: place centers in a grid and define an initial radius
    # Use a hexagonal grid with some spacing
    centers = np.zeros((n_circles, 2))
    radii = np.zeros(n_circles)
    r_initial = 0.05  # Initial radius

    # Place circles in a hexagonal grid
    rows = int(np.sqrt(n_circles))
    cols = rows
    spacing = 0.15
    for i in range(rows):
        for j in range(cols):
            idx = i * cols + j
            if idx >= n_circles:
                break
            x = j * spacing
            y = i * spacing * np.sqrt(3) / 2
            if i % 2 == 1:
                x += spacing / 2
            centers[idx] = [x, y]
            radii[idx] = r_initial

    # Define the objective function to maximize (we minimize the negative)
    def objective(x):
        return -np.sum(x[26:])  # Radii are from index 26 onwards

    # Define the constraints
    def constraints(x):
        # Extract centers and radii
        centers = x[:26*2].reshape((26, 2))
        radii = x[26*2:]

        # Constraint 1: All centers must be within the unit square
        # Constraint 2: All circles must not overlap
        # Constraint 3: All radii must be non-negative

        # Check constraints to evaluate feasibility
        for i in range(26):
            x_i, y_i = centers[i]
            r_i = radii[i]
            if x_i - r_i < 0 or x_i + r_i > 1 or y_i - r_i < 0 or y_i + r_i > 1:
                return [1e6]  # Penalize by a large value if constraint violated
            for j in range(i + 1, 26):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j]:
                    return [1e6]  # Penalize by a large value if constraint violated
        return [0]  # Feasible

    # Define bounds: each coordinate for centers is between 0 and 1, radii are >= 0
    bounds = []
    for i in range(26):
        bounds.extend([(0.0, 1.0), (0.0, 1.0)])  # x and y for each center
    bounds.extend([(0.0, 1.0) for _ in range(26)])  # radii for each circle

    # Run optimization
    result = opt.minimize(
        objective,
        x0=np.concatenate((centers.flatten(), radii)),
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'ineq', 'fun': constraints}
    )

    centers_opt = result.x[:26*2].reshape((26, 2))
    radii_opt = result.x[26*2:]
    total_radius = np.sum(radii_opt)

    # Validate the result (optional but recommended)
    valid, msg = validate_packing(centers_opt, radii_opt)
    if not valid:
        raise ValueError(msg)

    return centers_opt, radii_opt, total_radius

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
