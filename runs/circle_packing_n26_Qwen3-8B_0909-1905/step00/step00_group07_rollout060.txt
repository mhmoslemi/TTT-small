import numpy as np
import scipy.optimize as optimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial grid-based packing for 26 circles
    # Grid dimensions
    rows = 5
    cols = 6
    spacing = 1.0 / (rows + 1)
    centers = np.zeros((26, 2))
    radii = np.zeros(26)
    for i in range(rows):
        for j in range(cols):
            x = j * spacing + spacing / 2
            y = i * spacing + spacing / 2
            idx = i * cols + j
            centers[idx] = [x, y]
            # Initial radius is the minimum distance to a boundary or neighbor
            dist_to_boundary = min(x, 1 - x, y, 1 - y)
            if i > 0 and j > 0:
                dist_to_neighbor = np.sqrt((x - centers[i * cols + j - 1][0])**2 + (y - centers[i * cols + j - 1][1])**2)
                dist_to_boundary = min(dist_to_boundary, dist_to_neighbor)
            radii[idx] = dist_to_boundary * 0.9  # Small buffer

    def objective(radii_vec):
        return -np.sum(radii_vec)  # Minimize negative sum (maximize sum)

    def constraint_boundary(radii_vec):
        for i in range(26):
            x, y = centers[i]
            r = radii_vec[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                return -1
        return 0

    def constraint_overlap(radii_vec):
        for i in range(26):
            for j in range(i + 1, 26):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx ** 2 + dy ** 2)
                if dist < radii_vec[i] + radii_vec[j] - 1e-12:
                    return -1
        return 0

    # Optimization bounds
    bounds = [(0, 1) for _ in range(26)]

    # Constraints
    cons = [
        {'type': 'ineq', 'fun': constraint_boundary},
        {'type': 'ineq', 'fun': constraint_overlap}
    ]

    # Initial guess
    initial_guess = radii.copy()

    # Solve optimization problem
    result = optimize.minimize(
        objective,
        initial_guess,
        bounds=bounds,
        constraints=cons,
        method='SLSQP',
        options={'eps': 1e-6, 'maxiter': 200}
    )

    if not result.success:
        raise RuntimeError("Optimization failed")

    final_radii = result.x
    final_centers = centers.copy()

    sum_radii = np.sum(final_radii)
    return final_centers, final_radii, sum_radii
