import numpy as np
import scipy.optimize as optimize
import math

def run_packing():
    # Initial guess: hexagonal packing with rows
    n = 26
    rows = int(math.ceil(math.sqrt(n)))
    cols = int(math.ceil(n / rows))
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Define spacing based on initial guess
    spacing = 1.0
    for i in range(n):
        row = i // cols
        col = i % cols
        x = col * spacing
        y = row * spacing * math.sqrt(3) / 2
        centers[i] = np.array([x, y])

    # Initial radii: same for all
    radius_initial = 0.1
    radii[:] = radius_initial

    def objective(radii):
        return -np.sum(radii)

    def constraint_boundary(radii):
        return np.sum((centers[i] - radii[i]) >= 0 for i in range(n)) + np.sum((1 + radii[i] - centers[i]) >= 0 for i in range(n))

    def constraint_overlap(radii):
        overlaps = 0
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-12:
                    overlaps += 1
        return overlaps

    # Set up optimization using scipy.optimize.minimize
    # We optimize -sum(radii) to maximize sum(radii)
    result = optimize.minimize(
        objective,
        radii,
        method='SLSQP',
        constraints=[
            {'type': 'ineq', 'fun': lambda r: constraint_boundary(r)},
            {'type': 'ineq', 'fun': lambda r: -constraint_overlap(r)}
        ],
        bounds=[(0, 1) for _ in range(n)]
    )

    centers_final = centers
    radii_final = result.x
    sum_radii = np.sum(radii_final)

    # Validate the packing
    valid, msg = validate_packing(centers_final, radii_final)
    if not valid:
        raise ValueError(msg)

    return centers_final, radii_final, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
