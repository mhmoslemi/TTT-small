import numpy as np
import scipy.optimize as optimize

def run_packing():
    n = 26

    # Initial guess based on a nested hexagonal pattern
    # First, place 1 central circle
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Start with a 1x1 grid with a central circle
    centers[0] = [0.5, 0.5]
    radii[0] = 0.1

    # Create a hexagonal pattern around the center
    angle_step = 2 * np.pi / 6
    for i in range(1, n):
        angle = i * angle_step
        radius = 0.1 + (i - 1) * 0.05
        centers[i] = [
            0.5 + radius * np.cos(angle),
            0.5 + radius * np.sin(angle)
        ]
        radii[i] = 0.05

    # Define the objective function to maximize the sum of radii
    def objective(x):
        return -np.sum(x)

    # Define bounds for the radii
    bounds = [(0, 1) for _ in range(n)]

    # Define constraints
    def constraint_center(x, i):
        return (centers[i][0] - x[i] < -1e-12 or centers[i][0] + x[i] > 1 + 1e-12 or
                centers[i][1] - x[i] < -1e-12 or centers[i][1] + x[i] > 1 + 1e-12)

    def constraint_overlap(x, i, j):
        dx = centers[i][0] - centers[j][0]
        dy = centers[i][1] - centers[j][1]
        dist = np.sqrt(dx*dx + dy*dy)
        return dist - (x[i] + x[j]) + 1e-12

    # Build the constraints
    constraints = []
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: -constraint_center(x, i)})
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i, j=j: -constraint_overlap(x, i, j)})

    # Run the optimization
    result = optimize.minimize(
        objective,
        radii,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        tol=1e-10
    )

    # Extract the optimal radii and centers
    optimal_radii = result.x
    optimal_centers = np.zeros((n, 2))
    for i in range(n):
        optimal_centers[i] = centers[i]

    # Validate the packing

    if not is_valid:
        raise ValueError(f"Invalid packing: {message}")

    sum_radii = np.sum(optimal_radii)
    return optimal_centers, optimal_radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
