import numpy as np
import scipy.optimize as opt
import math

def run_packing():

    def hexagonal_grid(num_circles):
        # Hexagonal grid with circle spacing equal to 2*r_initial
        r_initial = 0.04
        positions = []
        for i in range(num_circles):
            row = i // 5
            col = i % 5
            x = col * 2 * r_initial
            y = row * (np.sqrt(3) * r_initial)
            positions.append((x, y))
        return np.array(positions)

    # Define initial radii for the hexagonal grid
    def initial_radii(num_circles):
        return np.full(num_circles, 0.04)

    # Objective function: maximize sum of radii
    def objective(params, centers, radii, n):
        # params: flattened array of (x, y, r) for each circle
        # Reconstruct centers and radii
        new_centers = params[:n*2].reshape(n, 2)
        new_radii = params[n*2:].reshape(n)
        return -np.sum(new_radii)  # Negative for minimization

    # Constraint function: circle within the unit square
    def square_constraint(params, centers, radii, n):
        new_centers = params[:n*2].reshape(n, 2)
        new_radii = params[n*2:].reshape(n)
        for i in range(n):
            x, y = new_centers[i]
            r = new_radii[i]
            if (x - r < 0 or x + r > 1 or
                y - r < 0 or y + r > 1):
                return 1.0  # Violation
        return 0.0

    # Constraint function: circles do not overlap
    def no_overlap_constraint(params, centers, radii, n):
        new_centers = params[:n*2].reshape(n, 2)
        new_radii = params[n*2:].reshape(n)
        for i in range(n):
            for j in range(i + 1, n):
                dx = new_centers[i, 0] - new_centers[j, 0]
                dy = new_centers[i, 1] - new_centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < new_radii[i] + new_radii[j] - 1e-6:
                    return 1.0  # Violation
        return 0.0

    # Initial packing (hexagonal grid with 25 circles)
    n = 26
    # Start with a hexagonal grid and add one extra circle in a corner
    initial_centers = hexagonal_grid(25)
    initial_centers = np.vstack((initial_centers, [0.1, 0.1]))  # Add corner circle

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
