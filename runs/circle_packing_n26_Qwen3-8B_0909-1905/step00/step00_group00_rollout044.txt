import numpy as np
import scipy.optimize as optimize
import matplotlib.pyplot as plt

def run_packing():
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Generate an initial guess using hexagonal packing in the unit square
    radius_grid = np.sqrt(2 / (n + 1))  # Initial estimate for radius
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col * (1 + radius_grid) / (n ** 0.5)
        y = row * (1 + radius_grid) / (n ** 0.5)
        centers[i] = [x, y]
        radii[i] = radius_grid

    # Optimization variables
    x_bounds = np.zeros((n, 2, 2))
    for i in range(n):
        x_bounds[i, 0, 0] = 0.0
        x_bounds[i, 0, 1] = 1.0
        x_bounds[i, 1, 0] = 0.0
        x_bounds[i, 1, 1] = 1.0

    r_bounds = np.zeros((n, 2))
    for i in range(n):
        r_bounds[i, 0] = 0.0
        r_bounds[i, 1] = 1.0

    # Optimization objective
    def objective(vars):
        centers_flat = vars[:n * 2].reshape(n, 2)
        radii_flat = vars[n * 2:].reshape(n)
        return -np.sum(radii_flat)  # Minimize negative sum to maximize

    # Constraints
    def constraint_center_bounds(vars):
        centers_flat = vars[:n * 2].reshape(n, 2)
        return np.concatenate([
            centers_flat[:, 0] - 1e-12,
            1.0 - centers_flat[:, 0] - 1e-12,
            centers_flat[:, 1] - 1e-12,
            1.0 - centers_flat[:, 1] - 1e-12
        ])

    def constraint_radius_bounds(vars):
        radii_flat = vars[n * 2:].reshape(n)
        return np.concatenate([
            radii_flat - 1e-12,
            1.0 - radii_flat - 1e-12
        ])

    def constraint_circle_distances(vars):
        centers_flat = vars[:n * 2].reshape(n, 2)
        radii_flat = vars[n * 2:].reshape(n)
        constraints = []
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers_flat[i, 0] - centers_flat[j, 0]
                dy = centers_flat[i, 1] - centers_flat[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                constraints.append(dist - (radii_flat[i] + radii_flat[j]) + 1e-12)
        return constraints

    # Initial guess
    initial_guess = np.concatenate([centers.flatten(), radii])

    # Setup optimization
    cons = [
        {'type': 'ineq', 'fun': constraint_center_bounds},
        {'type': 'ineq', 'fun': constraint_radius_bounds},
        {'type': 'ineq', 'fun': constraint_circle_distances}
    ]

    result = optimize.minimize(objective, initial_guess, method='SLSQP', constraints=cons, bounds=[x_bounds.flatten(), r_bounds.flatten()], tol=1e-8)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
