import numpy as np
import scipy.optimize as optimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial grid-like packing
    num_circles = 26
    grid_size = int(np.ceil(np.sqrt(num_circles)))

    radii = np.zeros(num_circles)

    spacing = 1.0 / (grid_size + 1)

        y = row * spacing + spacing

        radii[i] = spacing / 2

        # x contains all parameters (centers and radii)

        x_centers = x[:2 * num_circles].reshape(num_circles, 2)

        # Objective: maximize sum of radii
        return -np.sum(x_radii)  # Negative for minimization
        
    # Constraints
    def constraints(x):
        x_centers = x[:2 * num_circles].reshape(num_circles, 2)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
