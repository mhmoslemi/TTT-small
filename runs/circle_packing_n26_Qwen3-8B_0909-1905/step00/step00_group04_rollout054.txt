import numpy as np
import scipy.optimize as opt

def get_hexagonal_grid(nx, ny, spacing):
    """Generate a hexagonal grid of circles with given number of rows and columns."""
    centers = []
    for i in range(ny):
        for j in range(nx):
            x = j * spacing * np.sqrt(3)/2
            y = i * spacing * 3/2
            centers.append([x, y])
    return np.array(centers)

def get_grid_grid(nx, ny, spacing):
    """Generate a square grid of circles."""
    centers = []
    for i in range(ny):
        for j in range(nx):
            x = j * spacing
            y = i * spacing
            centers.append([x, y])
    return np.array(centers)

def validate_packing(centers, radii):

    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]

            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def objective_function(variables, centers, radii, n_circles):
    """Objective function to maximize the sum of radii."""
    return -np.sum(radii)  # Minimize the negative sum

def constraints(variables, centers, radii, n_circles):

    # Flatten the variables into centers and radii
    x = variables[:2 * n_circles]
    r = variables[2 * n_circles:]
    centers_new = x.reshape((n_circles, 2))

    # If all constraints are satisfied, return 0
    return 0
