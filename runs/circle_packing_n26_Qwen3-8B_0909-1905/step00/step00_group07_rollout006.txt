import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
from scipy.optimize import NonlinearConstraint

def run_packing():
    # Initial grid-based packing configuration
    n = 26
    num_rows = 5
    num_cols = 5
    spacing = 0.4
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    idx = 0
    for i in range(num_rows):
        for j in range(num_cols):
            if i % 2 == 0:
                x = j * spacing
            else:
                x = (j + 0.5) * spacing
            y = i * spacing
            centers[idx] = [x, y]
            radii[idx] = 0.05
            idx += 1

    # Objective function: maximize sum of radii
    def objective(radii_flat):
        return -np.sum(radii_flat)  # Minimize negative sum to maximize

    # Constraints: non-overlap and boundary
    def non_overlap_constraint(x):
        centers = x[:n*2].reshape(n, 2)
        radii = x[n*2:]
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-8:
                    return dist - (radii[i] + radii[j] - 1e-8)
        return 0.0

    def boundary_constraint(x):
        centers = x[:n*2].reshape(n, 2)
        radii = x[n*2:]
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-8 or x + r > 1 + 1e-8 or y - r < -1e-8 or y + r > 1 + 1e-8:
                return 1.0
        return 0.0

    # Initial guess: flatten centers and radii
    x0 = np.concatenate([centers.flatten(), radii])

    # Bounds for centers (within unit square)
    center_bounds = Bounds([0, 0] * n, [1, 1] * n)
    # Bounds for radii (non-negative)
    radius_bounds = Bounds([0] * n, [1] * n)

    # Non-linear constraint for non-overlap
    non_overlap = NonlinearConstraint(non_overlap_constraint, 0, 0)
    # Non-linear constraint for boundary conditions
    boundary = NonlinearConstraint(boundary_constraint, 0, 0)

    # Optimization
    result = minimize(

        x0,
        bounds=[center_bounds, radius_bounds],

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
