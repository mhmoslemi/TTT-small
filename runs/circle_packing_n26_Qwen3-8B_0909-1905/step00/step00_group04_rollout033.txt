def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial grid packing with hexagonal arrangement
    num_circles = 26
    rows = int(np.sqrt(num_circles)) + 1  # Approximate number of rows
    cols = num_circles // rows + 1

    # Hexagonal packing: x = i * 1.5 * r, y = j * sqrt(3) * r
    # Start with an arbitrary radius for grid-optimized spacing
    initial_radius = 0.05
    centers = np.zeros((num_circles, 2))
    idx = 0
    for j in range(rows):
        for i in range(cols):
            if idx >= num_circles:
                break
            x = i * 1.5 * initial_radius
            y = j * np.sqrt(3) * initial_radius
            # Adjust to stay within the unit square

                x = 1 - initial_radius
            if y + initial_radius > 1:
                y = 1 - initial_radius
            centers[idx] = [x, y]
            idx += 1

    # Initial guess for radii: equal small radii
    initial_radii = np.full(num_circles, initial_radius)

    # Constraints: non-overlap, boundary, and non-negative
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    def constraint_non_overlap(i, j, centers, radii):
        def cons(radii):
            dist = np.linalg.norm(centers[i] - centers[j])
            return dist - (radii[i] + radii[j])
        return cons

    def constraint_boundary(i, centers, radii):

            return (x - radii[i]) - (-1e-12), (x + radii[i]) - (1 + 1e-12), \
                   (y - radii[i]) - (-1e-12), (y + radii[i]) - (1 + 1e-12)
        return cons

    # Create optimization problem

    constraints = []

            constraints.append({
                'type': 'ineq',

            })

            # Overlap constraints

            constraints.append({

            })

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
