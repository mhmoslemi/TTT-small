```python
def run_packing():
    # Initial grid-based arrangement of 26 circles
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # First, arrange in a hexagonal grid as a starting point
    # Pair of rows for 26 circles: rows 1 through 5, with two rows having 6 circles and the rest having 5
    def hexagonal_grid(num_rows, num_cols_per_row):
        positions = []
        for row in range(num_rows):
            for col in range(num_cols_per_row):
                x = col + 0.5 * (row % 2)
                y = row * (np.sqrt(3) / 2)
                positions.append((x, y))
        return np.array(positions)

    # Define the initial grid layout
    rows = 5
    cols = 5  # Will be adjusted based on the number of circles
    positions = hexagonal_grid(rows, cols)
    centers[:len(positions)] = positions
    radii[:len(positions)] = 0.05  # Initial small radius

    # Define objective function: maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)  # Negative for minimization

    # Define constraints: no overlaps, all in the unit square
    def constraint_overlap(radii, centers):
        constraints = []
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                lower = radii[i] + radii[j] - 1e-8
                constraints.append(dist - lower)
        return np.array(constraints)

    def constraint_boundary(radii, centers):
        constraints = []
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            # Check if circle is in the unit square
            constraints.append(x - r)
            constraints.append(x + r - 1)
            constraints.append(y - r)
            constraints.append(y + r - 1)
        return np.array(constraints)

    # Convert centers to a numpy array for optimization
    centers = np.array(centers)

    # Define optimization problem
    bounds = [(0, 0.5) for _ in range(n)]  # Upper bound for radii

    # Initial guess: initial radii
    initial_radii = np.ones(n) * 0.05

    # Optimization
    result = minimize(
        fun=objective,
        x0=initial_radii,
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': constraint_overlap, 'args': (centers,)},
            {'type': 'ineq', 'fun': constraint_boundary, 'args': (centers,)}
```

```python
def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
```
