import numpy as np
import scipy.optimize as optimize
import math

def validate_packing(centers, radii):

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def get_initial_config():
    # Place circles in a hexagonal-like pattern
    rows = 5
    cols_per_row = [6, 6, 6, 6, 6]  # 5 rows with 6 circles each
    centers = []
    for r in range(rows):
        for c in range(cols_per_row[r]):
            x = c / (cols_per_row[r] - 1) * 1.0
            y = r * (math.sqrt(3) / 2) / (rows - 1) * 1.0
            centers.append([x, y])
    return np.array(centers)

def run_packing():

    centers = get_initial_config()
    n = len(centers)
    radii = np.array([1.0 / n] * n)  # Start with equal radii

    # Objective function to maximize the sum of radii
    def objective(radii_vec):
        return -np.sum(radii_vec)

    # Constraints: non-overlapping circles
    def non_overlap_constraints(radii_vec, centers):
        constraints = []
        for i in range(n):
            for j in range(i + 1, n):

                x2, y2 = centers[j]

                    r2 = radii_vec[j]

    # Bounds: radii must be non-negative

    # Optimization
    result = optimize.minimize(

        raise RuntimeError(f"Packing invalid: {msg}")

    # Return the result
    return (optimized_centers, optimized_radii, np.sum(optimized_radii))
