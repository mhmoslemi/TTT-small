import numpy as np
import scipy.optimize as optimize
import math

def run_packing():
    """
    Pack 26 circles in a unit square to maximize the sum of radii.
    
    Returns:
        tuple[np.ndarray, np.ndarray, float]: (centers, radii, sum_radii)
    """
    # Initial hexagonal packing with 26 circles
    num_circles = 26
    rows = int(math.ceil(math.sqrt(num_circles)))
    cols_per_row = int(math.ceil(num_circles / rows))
    
    # Hexagonal grid spacing (in unit square)
    dx = 1.0 / cols_per_row
    dy = np.sqrt(3.0) / 2.0 / rows
    
    centers = np.zeros((num_circles, 2))
    idx = 0
    for i in range(rows):
        for j in range(cols_per_row):
            if i % 2 == 1:
                x = j * dx + dx / 2.0
            else:
                x = j * dx
            y = i * dy
            centers[idx] = [x, y]
            idx += 1

    # Try to maximize total radius
    def objective(radii):
        return -np.sum(radii)  # Maximize sum of radii

    def constraints(centers, radii):
        # Non-overlapping constraint
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dist = np.linalg.norm(centers[i] - centers[j])
                if dist < radii[i] + radii[j] - 1e-8:
                    return False
        # Boundary constraint
        for i in range(num_circles):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-8 or x + r > 1 + 1e-8 or y - r < -1e-8 or y + r > 1 + 1e-8:
                return False
        return True

    # Initial radii: start with equal radii (based on packing)
    radii = np.full(num_circles, 1.0 / (2 * (cols_per_row + rows)))

    # Optimization problem to maximize radii
    def optimize_func(radii):

            return -np.sum(radii)

        def constraint_func(radii):
            return constraints(centers, radii)

        # Set bounds for radii
        bounds = [(0.0, 1.0) for _ in range(num_circles)]

        # Use SLSQP for constrained optimization
        result = optimize.minimize(
            objective_func,
            radii,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'ineq', 'fun': constraint_func}
        )
        return result.x

    # Run optimization
    radii = optimize_func(radii)

    return centers, radii, sum_radii


def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
