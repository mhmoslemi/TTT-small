import numpy as np
import scipy.optimize as opt

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def objective_function(x, n_circles=26):
    # x is a flat array of [x1, y1, r1, x2, y2, r2, ..., xn, yn, rn]
    centers = x.reshape((n_circles, 2))
    radii = x[n_circles*2:].reshape((n_circles,))
    # Penalize for being out of bounds
    out_of_bounds = np.sum(np.abs(centers - np.array([[0, 0], [1, 1]])) < 1e-6)
    # Penalize for overlap
    overlap_penalty = 0
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-6:
                overlap_penalty += (radii[i] + radii[j] - dist) ** 2
    # Penalize for negative radii
    neg_radii_penalty = np.sum(np.maximum(-radii, 0) ** 2)
    # Objective: minimize the penalty
    return out_of_bounds + overlap_penalty + neg_radii_penalty

def run_packing():

    # Optimization using L-BFGS-B
    result = opt.minimize(

        x0,
        method='L-BFGS-B',

        tol=1e-8,
        maxiter=1000

    # Extract the optimized centers and radii
