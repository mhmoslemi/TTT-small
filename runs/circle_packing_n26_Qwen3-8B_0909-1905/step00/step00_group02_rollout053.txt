import numpy as np
import scipy.optimize as optimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial placement: Hexagonal packing with some offset to spread out circles
    num_circles = 26
    rows = int(np.ceil(np.sqrt(num_circles)))
    cols = int(np.ceil(num_circles / rows))
    
    # Initial grid spacing
    grid_spacing = 1.0 / (cols + 1)
    grid_spacing_y = 1.0 / (rows + 1)
    
    # Generate initial centers
    centers = np.zeros((num_circles, 2))
    idx = 0
    for i in range(rows):
        for j in range(cols):
            if i % 2 == 0:
                x = j * grid_spacing + grid_spacing / 2
            else:
                x = j * grid_spacing + grid_spacing / 2 + grid_spacing / 2
            y = i * grid_spacing_y + grid_spacing_y / 2
            centers[idx] = [x, y]
            idx += 1
    
    # Initial radii (all equal for start)
    radii = np.full(num_circles, 0.05)
    
    # Define energy function to minimize (negative of sum of radii)
    def energy_function(params):
        # params: flattened array of centers and radii
        flat_centers = params[:num_circles * 2].reshape(num_circles, 2)
        flat_radii = params[num_circles * 2:]
        radii = flat_radii.reshape(num_circles)
        
        # Penalty for circles outside the square
        boundary_penalty = 0.0
        for i in range(num_circles):
            x, y = flat_centers[i]
            r = radii[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                boundary_penalty += 1e4 * max(0, (x - r - 0), (1 - x - r), (y - r - 0), (1 - y - r))
        
        # Penalty for circles overlapping
        overlap_penalty = 0.0
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = flat_centers[i, 0] - flat_centers[j, 0]
                dy = flat_centers[i, 1] - flat_centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < radii[i] + radii[j] - 1e-12:
                    overlap_penalty += 1e4 * (radii[i] + radii[j] - dist)
        
        # Penalty for negative radii
        negative_radius_penalty = 0.0
        for r in radii:
            if r < 0:
                negative_radius_penalty += 1e6 * (-r)
        
        # Total energy (negative of sum of radii)
        total_energy = -np.sum(radii) + boundary_penalty + overlap_penalty + negative_radius_penalty
        return total_energy
    
    # Flatten the centers and radii for optimization
    initial_params = np.concatenate([centers.flatten(), radii])
    
    # Constraint: All radii must be non-negative
    bounds = []
    for _ in range(num_circles * 2):
        bounds.append((0, 1))  # Radii constrained to [0, 1]
    
    # Optimization
    result = optimize.minimize(
        energy_function,
        initial_params,
        method='L-BFGS-B',
        bounds=bounds,
        jac=None
    )
    
    # Extract the optimal centers and radii
    optimal_params = result.x
    centers = optimal_params[:num_circles * 2].reshape(num_circles, 2)
    radii = optimal_params[num_circles * 2:]
    
    # Validate the packing
    is_valid, message = validate_packing(centers, radii)
    if not is_valid:
        raise ValueError(message)
    
    # Return the result
    return centers, radii, np.sum(radii)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
