import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds

def run_packing():
    # Number of circles
    n = 26

    # Define radii as a starting guess (each circle has the same initial radius)
    initial_guess = np.full(n, 0.05)
    # Define a grid of centers based on hexagonal packing
    centers = np.zeros((n, 2))
    radius = 0.05

    # Hexagonal grid with rows and columns
    columns = int(np.ceil(np.sqrt(n)))
    rows = int(np.ceil(n / columns))

    for i in range(rows):
        for j in range(columns):
            idx = i * columns + j
            if idx >= n:
                break
            x = j + 0.5 * (i % 2)
            y = i * np.sqrt(3) / 2
            centers[idx] = [x, y]

    # Scale centers to fit within the unit square
    centers = centers / np.max(centers)

    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum for maximization

    def constraint_boundary(radii):
        # Check if each circle fits within the boundary
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                return -1
        return 0

    def constraint_overlap(radii):
        # Check if any two circles overlap
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return -1
        return 0

    # Define optimization bounds (radii can't be negative)
    bounds = Bounds(np.zeros(n), np.inf)

    # Optimization options
    options = {
        'maxiter': 1000,
        'ftol': 1e-10,
        'gtol': 1e-10
    }

    # Multi-start optimization for global convergence
    best_sum = 0
    best_radii = np.zeros(n)
    best_centers = np.zeros((n, 2))

    for _ in range(5):
        # Random perturbation of centers
        centers += np.random.uniform(-0.1, 0.1, centers.shape)
        centers = np.clip(centers, 0, 1)

        # Optimize
        result = opt.minimize(
            objective,
            initial_guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[
                {'type': 'ineq', 'fun': constraint_boundary},
                {'type': 'ineq', 'fun': constraint_overlap}
            ],
            options=options
        )

        if result.success:
            if -result.fun > best_sum:
                best_sum = -result.fun
                best_radii = result.x
                best_centers = centers

    # Validate the final packing
    valid, msg = validate_packing(best_centers, best_radii)
    if not valid:
        raise ValueError(f"Packing failed to validate: {msg}")

    return best_centers, best_radii, best_sum

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
