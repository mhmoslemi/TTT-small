import numpy as np
import cvxpy as cp

def run_packing():
    # Initial guess: grid-like placement
    num_circles = 26
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)

    # Generate an initial grid of points in a square
    # We use a hexagonal-like grid to ensure better spacing
    # Parameters for grid size and spacing
    grid_rows = 5
    grid_cols = 6
    spacing = 0.25  # spacing between centers of neighboring circles

    for i in range(grid_rows):
        for j in range(grid_cols):
            index = i * grid_cols + j
            if index >= num_circles:
                break
            x = j * spacing
            y = i * spacing * np.sqrt(3) / 2
            # Adjust to stay within [0,1]×[0,1]
            if x > 1 - 1e-6:
                x = 1 - 1e-6
            if y > 1 - 1e-6:
                y = 1 - 1e-6
            centers[index] = [x, y]
            radii[index] = 0.01  # initial guess for radius

    # Define variables
    r = cp.Variable(num_circles, nonneg=True)
    c = cp.Variable((num_circles, 2), lb=np.array([0.0]*num_circles*2), ub=np.array([1.0]*num_circles*2))

    # Objective: maximize sum of radii
    objective = cp.Maximize(cp.sum(r))

    # Constraints
    constraints = []
    # All circles within the unit square
    for i in range(num_circles):
        constraints.append(c[i, 0] <= 1.0 - 1e-6)
        constraints.append(c[i, 0] >= 0.0 + 1e-6)
        constraints.append(c[i, 1] <= 1.0 - 1e-6)
        constraints.append(c[i, 1] >= 0.0 + 1e-6)

    # Circle non-overlap constraints
    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            dist = cp.sqrt((c[i, 0] - c[j, 0])**2 + (c[i, 1] - c[j, 1])**2)
            constraints.append(dist >= r[i] + r[j] - 1e-6)

    # Set up the problem
    problem = cp.Problem(objective, constraints)

    # Solve the problem
    problem.solve(solver=cp.SCS, verbose=False)

    # Extract the solution
    centers = c.value
    radii = r.value
    sum_radii = np.sum(radii)

    # Validate the solution
    is_valid, message = validate_packing(centers, radii)
    if not is_valid:
        raise ValueError(f"Packing is invalid: {message}")

    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
