import scipy.optimize as optimize

def run_packing():

    n_circles = 26
    rows = 7
    cols = 4  # max number of circles per row
    radius_initial = 0.05  # initial guess for radii

    max_row_height = 2 * radius_initial * np.sqrt(3)  # height of hexagonal row
    row_height = max_row_height / (rows - 1)  # spacing between rows
    col_width = 2 * radius_initial  # spacing between columns

            centers[index] = [x, y]

    # Initial guess for radii
    radii = np.full(n_circles, radius_initial)

    def objective(x):

    # Constraint: non-overlapping circles
    def constraint_nonoverlap(x, centers, radii):

        n = len(x)
        radii = x

                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-6:

    def constraint_bounds(x, centers):

        return 0
