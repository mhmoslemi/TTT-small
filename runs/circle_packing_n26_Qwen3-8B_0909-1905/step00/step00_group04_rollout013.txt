import numpy as np
import scipy.optimize as opt

def run_packing():
    # Define the number of circles
    n = 26
    
    # Initial guess: Hexagonal packing layout
    # Start with a grid of rows and columns, calculated to fit in the unit square
    # We'll start with a 5-row, 6-column grid and fit in the square
    rows = 5
    cols = 6
    radius_initial = 0.1  # Initial estimate for radius
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Create an initial hexagonal grid placement inside the square
    # Hexagonal lattice: vertical spacing is sqrt(3)/2 times horizontal spacing
    h = 1.0 / (cols - 1) * 0.9
    v = np.sqrt(3)/2 * h

    for i in range(rows):
        for j in range(cols):
            x = j * h
            y = i * v
            # Offset even rows by half the horizontal spacing
            if i % 2 == 1:
                x += h / 2
            # Adjust to stay within square [0,1]x[0,1]
            x = np.clip(x, 0, 1)
            y = np.clip(y, 0, 1)
            idx = i * cols + j
            centers[idx] = [x, y]
            radii[idx] = radius_initial
    
    # Objective function: Maximize the sum of radii
    def objective(x_vec):
        x = x_vec[:n*2].reshape(n, 2)
        r = x_vec[n*2:]
        return -np.sum(r)  # Negative for minimization

    # Constraints
    def constraint_boundary(x_vec):
        x = x_vec[:n*2].reshape(n, 2)
        r = x_vec[n*2:]
        for i in range(n):
            x_i, y_i = x[i]
            r_i = r[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12 or y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
                return -1
        return 0

    def constraint_overlap(x_vec):

        r = x_vec[n*2:]

                dist = np.sqrt(dx**2 + dy**2)
                if dist < r[i] + r[j] - 1e-12:
                    return -1
        return 0

    # Define constraints
    cons = [

        {'type': 'ineq', 'fun': constraint_overlap}
    ]

    x0 = np.concatenate([centers.reshape(-1), radii])

    # Run optimization
    result = opt.minimize(

        x0=x0,
        method='SLSQP',

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
