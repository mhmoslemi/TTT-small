import scipy.optimize as optimize

def run_packing():

    # Adjust column and row spacing for better fit

    row_spacing = np.sqrt(3) / 2 * col_spacing  # Vertical distance between rows

    # Generate centers in a hexagonal grid
    centers = []
    radii = []
    
    col_count = 6  # Columns for even distribution
    row_count = 5  # Rows for even distribution

    for col in range(col_count):
        for row in range(row_count):
            x = col * col_spacing
            y = row * row_spacing
            # Adjust for boundary constraints
            x = np.clip(x, 0, 1)
            y = np.clip(y, 0, 1)
            centers.append([x, y])
            radii.append(0.02)  # Initial small radius

    centers = np.array(centers)
    radii = np.array(radii)

    # Objective function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative of sum to maximize

    # Constraint function to ensure circles are inside the square
    def constraint_inside(radii):

            r = radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:

        return 0

    def constraint_nonoverlap(radii):
