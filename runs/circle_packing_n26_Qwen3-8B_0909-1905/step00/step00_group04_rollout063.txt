import numpy as np
import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    n = 26

    # This is a simplified initial grid placement (hexagonal)
    # We determine how many rows and columns are needed

    # Hexagonal packing in a square
    # Each circle can be considered as a grid point in a hexagonal lattice
    # Number of rows will be approx sqrt(n) and columns approx sqrt(n)
    rows = int(np.ceil(np.sqrt(n)))
    cols = int(np.ceil(n / rows))

    # Generate hexagonal lattice coordinates
    centers = np.zeros((n, 2))
    idx = 0
    for r in range(rows):
        for c in range(cols):
            x = c + 0.5 * (r % 2)
            y = r * np.sqrt(3) / 2
            if idx < n:
                centers[idx] = [x, y]
                idx += 1

    # Scale the grid to fit in the unit square
    max_x = np.max(centers[:, 0])
    max_y = np.max(centers[:, 1])
    scale = 1.0 / max(max_x, max_y)
    centers *= scale

    # Initial guess for radii: equal radius (unit circle area)
    # Total area of n circles: n * π * r^2
    # Unit square area is 1, so maximum possible sum of areas is 1
    # So we solve for r such that n * π * r^2 = 1 => r = (1 / (n * π))^(1/2)
    initial_radius = np.sqrt(1.0 / (n * np.pi))
    radii = np.full(n, initial_radius)

    def objective(radii_flat):
        return -np.sum(radii_flat)

    # Create a function to calculate the constraint violations
    def constraints(radii_flat):

        # Constraint 1: circles must be within the unit square
        # Constraint 2: circles must not overlap
