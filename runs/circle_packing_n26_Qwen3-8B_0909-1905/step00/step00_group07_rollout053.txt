import scipy.optimize as optimize

        radii[i] = 0.05

    def objective(x):

        centers_flat = x[:2*n].reshape(n, 2)
        radii_flat = x[2*n:]
        return -np.sum(radii_flat)  # Minimize negative sum => Maximize sum

    # Bounds for centers (x, y) in [0,1]
    center_bounds = [(0, 1) for _ in range(2*n)]
    # Bounds for radii (must be non-negative)
    radius_bounds = [(0, 1) for _ in range(n)]

    # Constraint: each circle must be entirely inside the unit square
    def square_boundary_constraint(x):

        radii_flat = x[2*n:]

    constraints = [

    result = optimize.minimize(
