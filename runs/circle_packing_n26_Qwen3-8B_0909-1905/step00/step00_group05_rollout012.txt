import numpy as np
import scipy.optimize as opt
from scipy.spatial.distance import cdist

def run_packing():
    # Initial guess: evenly spaced grid with equal radii
    n = 26
    grid = np.linspace(0, 1, 7)  # 7 points to get 6 intervals (can be adjusted for finer spacing)
    x_coords = grid
    y_coords = grid
    grid_centers = np.array([(x, y) for x in x_coords for y in y_coords])
    grid_centers = grid_centers[:n]
    initial_radii = np.full(n, 0.05)

    # Objective function: maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    # Constraint: boundary (circle must be within [0,1]^2)
    def boundary_constraint(radii):
        for i in range(n):
            x, y = grid_centers[i]
            r = radii[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                return np.inf
        return 0

    # Constraint: non-overlapping between circles
    def non_overlap_constraint(radii):
        dist_matrix = cdist(grid_centers, grid_centers)
        for i in range(n):
            for j in range(i + 1, n):
                if dist_matrix[i, j] < radii[i] + radii[j] - 1e-8:
                    return np.inf
        return 0

    # Combine constraints
    constraints = [
        {'type': 'ineq', 'fun': boundary_constraint},
        {'type': 'ineq', 'fun': non_overlap_constraint}
    ]

    # Optimization bounds
    bounds = [(0, 0.5) for _ in range(n)]  # Reasonable upper bound for radii

    # Run optimization
    result = opt.minimize(
        fun=objective,
        x0=initial_radii,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        tol=1e-8,
        options={'maxiter': 500}
    )

    # Extract final centers and radii
    final_centers = grid_centers
    final_radii = result.x

    # Validate the packing
    is_valid, message = validate_packing(final_centers, final_radii)
    if not is_valid:
        raise ValueError(f"Validation failed: {message}")

    # Return centers, radii, and sum of radii
    return final_centers, final_radii, np.sum(final_radii)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
