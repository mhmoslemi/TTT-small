import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import NonlinearConstraint

def run_packing():
    n = 26  # Number of circles
    # Initial hexagonal grid configuration
    rows = int(np.ceil(np.sqrt(n)))
    cols = int(np.ceil(n / rows))
    
    # Generate initial positions in a hexagonal grid
    centers = np.zeros((n, 2))
    r = 1.0 / (np.sqrt(3) * (cols + 1))  # Initial radius
    radius_guess = np.full(n, r)

    # Hexagonal grid positions
    for i in range(n):
        row = i // cols
        col = i % cols
        x = col + 0.5 * (row % 2)
        y = row * np.sqrt(3) / 2
        centers[i, 0] = x
        centers[i, 1] = y

    # Scale positions to fit in the unit square
    max_x = np.max(centers[:, 0])
    max_y = np.max(centers[:, 1])
    centers /= np.max([max_x, max_y])

    # Define the cost function to maximize sum of radii (minimize negative sum)
    def objective(x):
        return -np.sum(x)

    # Define constraints
    def constraint_center(x, i):
        # Constraint: circle i must be within the unit square
        x_i, y_i = centers[i]
        r_i = x[i]
        if (x_i - r_i < 0 or x_i + r_i > 1 or y_i - r_i < 0 or y_i + r_i > 1):
            return 1e-6  # Penalize out-of-bounds circles
        return 0

    def constraint_overlap(x, i, j):
        # Constraint: circles i and j must not overlap
        x_i, y_i = centers[i]
        x_j, y_j = centers[j]
        r_i = x[i]
        r_j = x[j]
        dist = np.sqrt((x_i - x_j)**2 + (y_i - y_j)**2)
        if dist < r_i + r_j - 1e-6:
            return 1e-6  # Penalize overlapping circles
        return 0

    # Set up the bounds: radii must be non-negative
    bounds = Bounds(np.zeros(n), np.inf)

    # Set up nonlinear constraints
    nlp_constraints = []
    for i in range(n):
        nlp_constraints.append(NonlinearConstraint(lambda x: constraint_center(x, i), 0, 1e6))
    for i in range(n):
        for j in range(i+1, n):
            nlp_constraints.append(NonlinearConstraint(lambda x: constraint_overlap(x, i, j), 0, 1e6))

    # Run the optimization
    result = minimize(
        objective,
        radius_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=nlp_constraints,
        options={'maxiter': 500, 'ftol': 1e-9, 'disp': False}
    )

    # Extract results
    radii = result.x
    centers = centers
    sum_radii = np.sum(radii)

    # Validate the packing
    is_valid, msg = validate_packing(centers, radii)
    if not is_valid:
        raise Exception(f"Validation failed: {msg}")

    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
