import numpy as np
import scipy.optimize as opt

def run_packing():
    # Initial guess using hexagonal packing
    num_circles = 26
    num_rows = 4  # Approximately number of rows that fit in the square
    num_cols = 6  # Number of columns per row
    spacing = 1.0 / (num_cols + 1)  # Distance between centers horizontally
    row_spacing = np.sqrt(3) / 2 * spacing  # Distance between rows vertically

    centers = np.zeros((num_circles, 2))
    idx = 0
    for row in range(num_rows):
        for col in range(num_cols):
            x = col * spacing + spacing / 2
            y = row * row_spacing + spacing / 2
            centers[idx] = [x, y]
            idx += 1

    # Initial radius guess: all same
    initial_radii = np.full(num_circles, 0.1)  # Initial guess for radii

    # Objective function to maximize (sum of radii)
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum

    # Constraints: no overlapping between any two circles
    def constraint_overlap(radii):
        n = len(radii)
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx ** 2 + dy ** 2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return -1.0
        return 0.0

    # Constraint: all circles must stay within the unit square
    def constraint_boundary(radii):
        for i in range(len(radii)):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                return -1.0
        return 0.0

    # Bounds: radii must be non-negative
    bounds = [(0.0, 1.0) for _ in range(num_circles)]

        {'type': 'ineq', 'fun': constraint_boundary}

    # Optimization
    result = opt.minimize(

        initial_radii,
        bounds=bounds,
        constraints=constraints,

        options={'disp': False, 'maxiter': 100}
