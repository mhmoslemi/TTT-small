import numpy as np
import cvxpy as cp

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Define the number of circles
    n = 26

    # Initial grid setup (columns and rows of circles)
    num_cols = 6
    num_rows = 5
    grid_centers = np.array([[i / (num_cols - 1), j / (num_rows - 1)] for j in range(num_rows) for i in range(num_cols)])

    # Optimize centers and radii to maximize the sum of radii
    # Define variables
    centers = cp.Variable((n, 2))
    radii = cp.Variable(n)

    # Objective: Maximize the sum of radii
    objective = cp.Maximize(cp.sum(radii))

    # Constraints
    constraints = []

    # Boundary constraints: each circle must lie in [0, 1]^2
    for i in range(n):
        constraints.append(centers[i, 0] - radii[i] >= 0)
        constraints.append(centers[i, 0] + radii[i] <= 1)
        constraints.append(centers[i, 1] - radii[i] >= 0)
        constraints.append(centers[i, 1] + radii[i] <= 1)

    # Non-overlap constraint between any pair of circles
    for i in range(n):
        for j in range(i + 1, n):
            dist = cp.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)
            constraints.append(dist >= radii[i] + radii[j])

    # Initial values to start the optimization
    initial_centers = grid_centers
    initial_radii = np.ones(n) * 0.1  # Start with small radii

    # Add initial values as guesses
    centers_initial = cp.Parameter((n, 2), value=initial_centers)
    radii_initial = cp.Parameter(n, value=initial_radii)

    # Reformulate the problem using initial values
    centers = cp.Variable((n, 2), initial=centers_initial)
    radii = cp.Variable(n, initial=radii_initial)

    # Rebuild constraints with new variables
    constraints = []
    for i in range(n):
        constraints.append(centers[i, 0] - radii[i] >= 0)
        constraints.append(centers[i, 0] + radii[i] <= 1)
        constraints.append(centers[i, 1] - radii[i] >= 0)
        constraints.append(centers[i, 1] + radii[i] <= 1)

    for i in range(n):
        for j in range(i + 1, n):
            dist = cp.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)
            constraints.append(dist >= radii[i] + radii[j])

    # Solve the optimization problem
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.SCS, max_iter=1000)

    # Extract the final centers and radii
    centers_final = centers.value
    radii_final = radii.value

    # Compute the total sum of radii
    total_sum_radii = np.sum(radii_final)
