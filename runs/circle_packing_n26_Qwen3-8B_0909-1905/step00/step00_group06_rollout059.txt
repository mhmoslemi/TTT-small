import numpy as np
import scipy.optimize as opt
import matplotlib.pyplot as plt

def run_packing():
    # Initial guess: hexagonal packing with 5 rows
    n = 26
    rows = 5
    cols = 6
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Hexagonal packing in unit square
    for i in range(rows):
        for j in range(cols):
            # Offset for even rows
            offset = 0.5 * (1 - (i % 2))
            x = j * (1 / (cols - 1)) + offset
            y = i * (np.sqrt(3) / (rows - 1)) + 0.5 * (1 - (i % 2))
            idx = i * cols + j
            if idx < n:
                centers[idx] = [x, y]
                radii[idx] = 0.05  # Initial radius

    # Objective function to maximize sum of radii
    def objective(x):
        # x is a concatenated array of centers and radii
        centers_flat = x[:n * 2].reshape(n, 2)
        radii_flat = x[n * 2:].reshape(n)
        return -np.sum(radii_flat)  # Negative for minimization

    # Constraints
    def constraint_centers(x):
        centers_flat = x[:n * 2].reshape(n, 2)
        return np.array([c - r for c, r in zip(centers_flat, x[n * 2:])])

    def constraint_bounding_box(x):
        centers_flat = x[:n * 2].reshape(n, 2)
        radii_flat = x[n * 2:].reshape(n)
        return np.array([c[0] - r for c, r in zip(centers_flat, radii_flat)] + [1 - c[0] - r for c, r in zip(centers_flat, radii_flat)] +
                        [c[1] - r for c, r in zip(centers_flat, radii_flat)] + [1 - c[1] - r for c, r in zip(centers_flat, radii_flat)])

    def constraint_circle_overlap(x):
        centers_flat = x[:n * 2].reshape(n, 2)
        radii_flat = x[n * 2:].reshape(n)
        overlaps = []
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers_flat[i, 0] - centers_flat[j, 0]
                dy = centers_flat[i, 1] - centers_flat[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                overlaps.append(dist - (radii_flat[i] + radii_flat[j]))
        return np.array(overlaps)

    # Initial guess
    initial_guess = np.concatenate([centers.flatten(), radii])

    # Optimization
    bounds = []
    for i in range(n):
        bounds += [(0, 1) for _ in range(2)]  # for each center

        {'type': 'ineq', 'fun': constraint_centers},
        {'type': 'ineq', 'fun': constraint_bounding_box},

        method='SLSQP',
        bounds=bounds,
        constraints=constraints

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
