from scipy.optimize import differential_evolution
import math

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    # Define initial configuration with refined placement
    def generate_initial_centers(n):
        cols = int(math.sqrt(n)) + 1
        rows = (n + cols - 1) // cols
        spacing = 1.0 / (rows + 1)
        centers = np.zeros((n, 2))
        idx = 0
        for r in range(rows):
            for c in range(cols):
                x = c * spacing + spacing / 2
                y = r * spacing + spacing / 2

                    x += spacing / 2

                    centers[idx] = [x, y]
                    idx += 1

        return centers

    centers = generate_initial_centers(n)
    radii = np.full(n, 0.05)

    def objective(radii):
        return -np.sum(radii)

    # Define constraints to ensure no overlaps and within square
    def constraint_overlap(i, j, centers, radii):

        return dist - radii[i] - radii[j] + 1e-12

    def constraint_boundary(i, centers, radii):
        x, y = centers[i]
        r = radii[i]
        return (x - r + 1e-12) * (x + r - 1 - 1e-12) * (y - r + 1e-12) * (y + r - 1 - 1e-12)

    bounds = [(0, 1) for _ in range(n * 2)]

    constraints = []

    result = differential_evolution(

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
