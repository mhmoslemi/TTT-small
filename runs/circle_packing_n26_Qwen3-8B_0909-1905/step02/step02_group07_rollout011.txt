from scipy.optimize import minimize
import math
from multiprocessing import Pool, cpu_count

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

        return -np.sum(radii)

        return dist - radii[i] - radii[j] + 1e-12

    def constraint_boundary(i, centers, radii):
        x, y = centers[i]
        r = radii[i]
        return (x - r + 1e-12) * (x + r - 1 - 1e-12) * (y - r + 1e-12) * (y + r - 1 - 1e-12)

    def optimize_one_config(centers):

            constraints.append({'type': 'ineq', 'fun': lambda r, i=i: constraint_boundary(i, centers, r)})

                constraints.append({'type': 'ineq', 'fun': lambda r, i=i, j=j: constraint_overlap(i, j, centers, r)})

        result = minimize(

            radii,
            args=(centers,),
            method='SLSQP',
            constraints=constraints,
            options={'ftol': 1e-9, 'maxiter': 500}
        )
