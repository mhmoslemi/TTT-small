import numpy as np
from scipy.optimize import minimize
import math

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    # Generate hexagonal initial layout within the unit square
    cols = int(math.sqrt(n))
    rows = (n + cols - 1) // cols
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Hexagonal lattice
    for i in range(n):
        # Use hexagonal grid inspired by sphere packing in 2D
        # Generate coordinates based on hexagonal lattice
        row = i // cols
        col = i % cols
        x = col + 0.5 * (row % 2)
        y = row * np.sqrt(3)/2
        # Scale to unit square
        x = x / (cols + 1)
        y = y / (rows + 1)
        centers[i] = [x, y]
        radii[i] = 0.01  # Initial very small radius

    # Define objective function to maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)

    # Define constraint functions
    def constraint_boundary(i, centers, radii):
        x, y = centers[i]
        r = radii[i]
        return (x - r >= -1e-12) * (x + r <= 1 + 1e-12) * (y - r >= -1e-12) * (y + r <= 1 + 1e-12)

    def constraint_overlap(i, j, centers, radii):
        dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
        return dist - radii[i] - radii[j] + 1e-12

    # Create constraints
    constraints = []

    # Boundary constraints
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': lambda r, i=i: constraint_boundary(i, centers, r)})

    # Overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({'type': 'ineq', 'fun': lambda r, i=i, j=j: constraint_overlap(i, j, centers, r)})

    # Perform optimization with COBYLA method, which handles bounds well
    initial_radii = np.full(n, 0.01)
    result = minimize(objective, initial_radii, method='COBYLA', constraints=constraints)

    # Extract result
    final_radii = result.x
    final_sum_radii = np.sum(final_radii)

    # Validate the packing
    valid, msg = validate_packing(centers, final_radii)
    if not valid:
        raise RuntimeError(f"Invalid packing: {msg}")

    return centers, final_radii, final_sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
