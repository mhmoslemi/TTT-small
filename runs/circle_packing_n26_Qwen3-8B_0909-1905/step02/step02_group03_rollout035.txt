from scipy.optimize import minimize
import math
import random

    cols = int(math.sqrt(n))

    radii = np.zeros(n)

    def objective(radii):

    # Constraints to ensure non-overlap and within square

def validate_packing(centers, radii):

            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"
