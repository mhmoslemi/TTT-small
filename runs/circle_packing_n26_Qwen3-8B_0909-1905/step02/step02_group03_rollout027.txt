import numpy as np
from scipy.optimize import minimize
import math

# Function to create a hexagonal grid of circles
def create_hexagonal_grid(n, rows, cols, spacing):
    centers = np.zeros((n, 2))
    idx = 0
    for r in range(rows):
        for c in range(cols):
            x = c * spacing + spacing / 2
            y = r * spacing + spacing / 2
            if r % 2 == 1:
                x += spacing / 2
            if idx < n:
                centers[idx] = [x, y]
                idx += 1
    return centers

# Function to calculate the distance between two points
def distance(p1, p2):
    return np.sqrt(np.sum((p1 - p2) ** 2))

# Function to calculate the constraint for no overlap
def overlap_constraint(i, j, centers, radii):
    return distance(centers[i], centers[j]) - radii[i] - radii[j] + 1e-12

# Function to calculate the boundary constraint
def boundary_constraint(i, centers, radii):
    x, y = centers[i]
    r = radii[i]
    return (x - r + 1e-12) * (x + r - 1 - 1e-12) * (y - r + 1e-12) * (y + r - 1 - 1e-12)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = distance(centers[i], centers[j])
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26

    # Create a denser initial hexagonal grid
    # Use more rows and columns to better fill the square
    cols = int(math.sqrt(n))
    rows = (n + cols - 1) // cols

    centers = create_hexagonal_grid(n, rows, cols, spacing)

    # Initial guess for radii
    initial_radius = np.full(n, spacing / 2)

    # Objective function to maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)

    # For constraints, we need to pass the centers and radii
    # Let's define an optimizer with more control

    # Define constraints
    constraints = []
    for i in range(n):
        # Boundary constraint
        constraints.append({'type': 'ineq', 'fun': lambda r, i=i, centers=centers: boundary_constraint(i, centers, r)})

    for i in range(n):
        for j in range(i + 1, n):
            # No overlap constraint
            constraints.append({'type': 'ineq', 'fun': lambda r, i=i, j=j, centers=centers: overlap_constraint(i, j, centers, r)})

    # Use COBYLA as the optimization method (better for constrained problems)
    result = minimize(
