from scipy.optimize import minimize
import math

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    # Initial configuration: hexagonal grid
    cols = int(math.sqrt(n))
    rows = (n + cols - 1) // cols
    spacing = 1.0 / (rows + 1)
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    idx = 0
    for r in range(rows):
        for c in range(cols):
            x = c * spacing + spacing / 2
            y = r * spacing + spacing / 2
            if r % 2 == 1:
                x += spacing / 2
            if idx < n:
                centers[idx] = [x, y]
                radii[idx] = spacing / 2
                idx += 1

    # Define objective: maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)

    # Define constraints
    def constraint_boundary(i, centers, radii):
        x, y = centers[i]
        r = radii[i]
        return (x - r + 1e-12) * (x + r - 1 - 1e-12) * (y - r + 1e-12) * (y + r - 1 - 1e-12)

    def constraint_overlap(i, j, centers, radii):
        x1, y1 = centers[i]
        x2, y2 = centers[j]
        r1 = radii[i]
        r2 = radii[j]
        dist = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)
        return dist - r1 - r2 + 1e-12

    constraints = []

        constraints.append({'type': 'ineq', 'fun': lambda r, i=i: constraint_boundary(i, centers, r)})

            constraints.append({'type': 'ineq', 'fun': lambda r, i=i, j=j: constraint_overlap(i, j, centers, r)})

    # Perform optimization
    result = minimize(
        objective,
        radii,
        method='SLSQP',
        constraints=constraints,
        options={'ftol': 1e-9, 'maxiter': 500}
    )

    final_radii = result.x

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
