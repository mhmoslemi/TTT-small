from scipy.optimize import minimize
import math
import random

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Initial hexagonal grid layout with spacing
    cols = int(math.sqrt(n))
    rows = (n + cols - 1) // cols
    spacing = 1.0 / (rows + 1)
    idx = 0
    for r in range(rows):
        for c in range(cols):
            x = c * spacing + spacing / 2
            y = r * spacing + spacing / 2
            if r % 2 == 1:
                x += spacing / 2
            if idx < n:
                centers[idx] = [x, y]
                radii[idx] = spacing / 2
                idx += 1

    # Objective function to maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)

    # Constraints for boundaries and non-overlapping
    def constraint_boundary(i, centers, radii):
        x, y = centers[i]
        r = radii[i]
        return (x - r + 1e-12) * (x + r - 1 - 1e-12) * (y - r + 1e-12) * (y + r - 1 - 1e-12)

    def constraint_overlap(i, j, centers, radii):
        dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
        return dist - radii[i] - radii[j] + 1e-12

    # Create constraints
    constraints = []

        constraints.append({'type': 'ineq', 'fun': lambda r, i=i: constraint_boundary(i, centers, r)})

            constraints.append({'type': 'ineq', 'fun': lambda r, i=i, j=j: constraint_overlap(i, j, centers, r)})

    # Perform optimization
    initial_radius = np.full(n, spacing / 2)
    result = minimize(objective, initial_radius, method='SLSQP', constraints=constraints)

    # Extract result

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12

            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
