from scipy.optimize import minimize
import math

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    n = 26

    # Initial layout: hexagonal packing
    # This is a grid of circles where each row is offset
    # We'll start by calculating the positions for a hex grid
    rows = int(math.ceil(math.sqrt(n)))
    cols = int(math.ceil(n / rows))
    spacing = 1.0 / (rows + 1)
    spacing_y = 1.0 / (rows + 1)
    spacing_x = 1.0 / (cols + 1)

    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    idx = 0

    for r in range(rows):
        for c in range(cols):
            x = c * spacing_x + spacing_x / 2
            y = r * spacing_y + spacing_y / 2
            if r % 2 == 1:
                x += spacing_x / 2
            if idx < n:
                centers[idx] = [x, y]
                radii[idx] = spacing_x / 2
                idx += 1

    def objective(radii):
        return -np.sum(radii)

    # Define constraints for boundaries and non-overlapping
    def constraint_boundary(i, centers, radii):
        x, y = centers[i]
        r = radii[i]
        return (x - r + 1e-12) * (x + r - 1 - 1e-12) * (y - r + 1e-12) * (y + r - 1 - 1e-12)

    def constraint_overlap(i, j, centers, radii):
        dx = centers[i, 0] - centers[j, 0]
        dy = centers[i, 1] - centers[j, 1]
        dist = np.sqrt(dx * dx + dy * dy)
        return dist - radii[i] - radii[j] + 1e-12

    constraints = []

        constraints.append({'type': 'ineq', 'fun': lambda r, i=i: constraint_boundary(i, centers, r)})

            constraints.append({'type': 'ineq', 'fun': lambda r, i=i, j=j: constraint_overlap(i, j, centers, r)})

    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': lambda r, i=i: r[i]})

    # Perform optimization
    initial_radius = np.full(n, spacing_x / 2)
    result = minimize(objective, initial_radius, method='SLSQP', constraints=constraints, options={'ftol': 1e-10, 'maxiter': 1000})

    final_radii = result.x

    valid, msg = validate_packing(centers, final_radii)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
