from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    rows = 5  # number of rows in hexagonal layout
    cols = 6  # number of columns in hexagonal layout
    # Adjust based on the number of circles and unit square
    cols = (n + rows - 1) // rows
    rows = (n + cols - 1) // cols

    # Initial placement: hexagonal lattice with optimizations
    spacing = 1.0 / (rows + 1)
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    idx = 0
    for r in range(rows):
        for c in range(cols):
            # Shift x for even rows to simulate hexagonal packing
            x = c * spacing + spacing / 2
            y = r * spacing + spacing / 2
            if r % 2 == 1:
                x += spacing / 2
            if idx < n:
                centers[idx] = [x, y]
                radii[idx] = spacing / 2
                idx += 1

    # Objective function: maximize the sum of radii (minimize negative sum)
    def objective(radii):
        return -np.sum(radii)

    # Boundary constraint: ensure circles stay within [0,1]^2
    def boundary_constraint(radii):

            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                return -1.0
        return 0.0

    # Overlap constraint: ensure circles do not overlap
    def overlap_constraint(radii):

            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return -1.0
        return 0.0

    # Regularization term to discourage redundant spacing
    def regularization(radii):
        return 0.1 * np.sum(radii ** 2)

    # Optimization
    # Define bounds for all radii (0 to 1)
    bounds = [(0, 1.0) for _ in range(n)]
    
    # Start with the initial guess
    result = minimize(
        lambda r: objective(r) + regularization(r),
        radii,
        method='SLSQP',
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': boundary_constraint},
            {'type': 'ineq', 'fun': overlap_constraint}

    )

    final_radii = result.x

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12

            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
