    c = get_corr(h, n)

def nesterov_smooth(h, n, alpha, max_iter, deadline, step_decay=0.999):

    y = h.copy()
    t_k = 1.0
    step = 0.1
    beta = 1.0 / (2.0 * n)  # Lipschitz estimate (rough)

            break

        c_max = c.max()
        shifted = alpha * (c - c_max)
        # Softmax-weighted gradient
        w = np.exp(shifted - shifted.max())
        w /= w.sum()
        
        # Gradient of LSE objective

                g[m:] -= y[:n-m]
            else:
                g[-m:] += (1.0 - y[:n+m])
                g[:n+m] -= y[-m:]
            grad += w[k] * g
        
        # Step
        lr = step / (1.0 + 0.01 * it)
        x_new = project_feasible(y - lr * grad, n)
        
        # Nesterov update

        y = project_feasible(y, n)
        x = x_new
        t_k = t_new
    
    return x

    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def project_feasible(h, n):

            tot = room.sum()
            if tot < 1e-15:

            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)

def smooth_lse(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

            return x.sum() - n / 2.0

            return np.ones(n)

            scale = 0.008 * np.exp(-0.06 * (r - 5)) + 0.001
            hp = project_feasible(h_best + rng.normal(0, scale, n_cur), n_cur)
        h_o, c_o = exact_cd(hp, n_cur, 250, deadline, rng)
        if c_o < best_c - 1e-13:
            h_best, best_c = h_o, c_o
            if time.time() < deadline - 70:
                h_p, c_p = slsqp_opt(h_best, n_cur, [100000, 500000, 2000000], 400, deadline)
                if c_p < best_c - 1e-13:
                    h_best, best_c = h_p, c_p
                h_p2, c_p2 = exact_cd(h_best, n_cur, 200, deadline, rng)
                if c_p2 < best_c - 1e-13:
                    h_best, best_c = h_p2, c_p2

    # Phase 7: Final intensive polish
    n_fin = len(h_best)
    h_f, c_f = exact_cd(h_best, n_fin, 800, deadline, rng)
    if c_f < best_c - 1e-14:
        h_best, best_c = h_f, c_f
    if time.time() < deadline - 15:
        h_f2, c_f2 = slsqp_opt(h_best, n_fin, [500000, 2000000, 8000000], 500, deadline)
        if c_f2 < best_c - 1e-14:
            h_best, best_c = h_f2, c_f2
        h_f3, c_f3 = exact_cd(h_best, n_fin, 400, deadline, rng)
        if c_f3 < best_c - 1e-14:
            h_best, best_c = h_f3, c_f3

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
