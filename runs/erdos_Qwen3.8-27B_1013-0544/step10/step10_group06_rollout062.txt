        neg, pos = a_nd < -1e-13, a_nd > 1e-13
        if np.any(neg): xl = max(xl, float(np.max((T - c_nd[neg]) / a_nd[neg])))
        if np.any(pos): xh = min(xh, float(np.min((T - c_nd[pos]) / a_nd[pos])))
    xl, xh = max(xl, 0.0), min(xh, 1.0)
    if hd:
        Cq = T - cd_ + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            if xl < r1: xh = min(xh, r1)
            elif xh > r2: xl = max(xl, r2)
    xl, xh = max(xl, 0.0), min(xh, 1.0)
    if xl > xh + 1e-10: return h0
    return float(np.clip((xl + xh) / 2.0, 0.0, 1.0))

def cd(h, n, mp, dl, rng=None, nb=40):
    h = h.copy()
    for _ in range(mp):
        if time.time() > dl: break
        c = np.correlate(h, 1.0 - h, mode='full')
        imp = False
        order = rng.permutation(n) if rng is not None else np.arange(n)
        for p in order:
            if time.time() > dl: break
            if h[p] < 1e-14 or h[p] > 1.0 - 1e-14: continue
            xn = solve_coord(c, h, p, n, nb)
            if abs(xn - h[p]) > 1e-13:
                h[p] = xn; imp = True
        if not imp: break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def lbf(h, n, alpha, lam, mx, dl):
    if time.time() > dl - 10: return h, compute_c5(h, n)
    try:
        r = minimize(lse_grad, h, args=(n, alpha, lam), method='L-BFGS-B',
                     jac=True, bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': mx, 'ftol': 1e-18, 'gtol': 1e-14})
        nh = project_feasible(r.x, n)
        return nh, compute_c5(nh, n)
    except Exception:
        return h, compute_c5(h, n)
