import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def get_subgradient(h, n):
    """Compute subgradient of max_k c_k w.r.t. h."""
    c_raw = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    c = c_raw * dx
    m = c.max()
    # Active set: indices where c is within epsilon of max
    active = c >= m - 1e-6
    # For each active k, compute gradient of c_k w.r.t. h
    # c_k = dx * sum_i h[i] * (1-h[i+k]) for valid i
    # dc_k/dh[p] = dx * (1-h[p+k]) - dx * h[p-k] (for valid indices)
    # More precisely, using the correlation structure:
    # c_k = dx * sum over valid i of h[i] * (1 - h[i+k])
    # If we change h[p]: 
    #   - It appears as h[i] when i=p: contributes dx*(1-h[p+k])
    #   - It appears as h[i+k] when i+k=p, i=p-k: contributes -dx*h[p-k]
    # So dc_k/dh[p] = dx*((1-h[p+k] if valid) - h[p-k] if valid)
    
    g = np.zeros(n)
    for k_idx in np.where(active)[0]:
        k = k_idx - (n - 1)  # shift
        # h[p+k] valid when 0 <= p+k < n
        # h[p-k] valid when 0 <= p-k < n
        p_range = np.arange(n)
        valid_f = (p_range + k >= 0) & (p_range + k < n)
        valid_b = (p_range - k >= 0) & (p_range - k < n)
        grad_k = np.zeros(n)
        grad_k[valid_f] += 1.0 - h[p_range[valid_f] + k]
        grad_k[valid_b] -= h[p_range[valid_b] - k]
        g += grad_k
    g *= dx / active.sum()
    return g, m

def subgradient_descent(h, n, steps, deadline):
    """Subgradient descent with Polyak step size."""
    h = h.copy()
    h_avg = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    C_hat = best_c  # upper bound on optimal value
    for t in range(1, steps + 1):
        if time.time() > deadline:
            break
        g, m = get_subgradient(h, n)
        # Polyak step: alpha = (C_hat - current_obj) / ||grad||^2
        # But our objective is max_k c_k, and we're at value m
        # Use alpha = (C_hat - m + epsilon) / (||g||^2 + epsilon)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-12:
            break
        alpha = (C_hat - m + 1e-4) / (g_norm ** 2 + 1e-12)
        alpha = min(alpha, 1.0 / t)  # also use standard step for safety
        alpha = min(alpha, 0.5)  # cap step size
        
        h_new = h - alpha * g
        h_new = project_feasible(h_new, n)
        
        # Update average
        h_avg = (t * h_avg + h_new) / (t + 1)
        
        c_avg = compute_c5(h_avg, n)
        c_new = compute_c5(h_new, n)
        
        if c_avg < C_hat:
            C_hat = c_avg
        if c_avg < best_c:
            best_c = c_avg
            best_h = h_avg.copy()
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        
        h = h_new
    return best_h, best_c

def project_feasible(h, n):

    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break

def solve_coord(c, h, p, n):
    h0 = h[p]
    m_min = max(0, n - 1 - p)
    m_max = min(2 * n - 2, 2 * n - 2 - p)
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms].copy()
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    unaff = 0.0
    if m_min > 0:
        unaff = float(np.max(c[:m_min]))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))

    neg_mask = a < -1e-14
    pos_mask = a > 1e-14
    zero_mask = np.abs(a) <= 1e-14
    L_lo, L_hi = unaff, cur_max
    for _ in range(40):
        if L_hi - L_lo < 1e-13:
            break

        if x_lo > x_hi + 1e-11:

                r2 = (a_d + sq) / 2.0
                if x_lo > h0 + r1 and x_hi < h0 + r2:
                    feasible = False
        if feasible:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0

def exact_cd(h, n, max_passes, deadline):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            xn = solve_coord(c, h, p, n)
            if abs(xn - h0) > 1e-12:
                delta = xn - h0
                m_min = max(0, n - 1 - p)
                m_max = min(2 * n - 2, 2 * n - 2 - p)
                ms = np.arange(m_min, m_max + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                c[ms] += delta * a
                diag_i = np.where(ms == (n - 1))[0]
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def subgrad_cd(h, n, deadline, top_k=30):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h_avg = h.copy()
    t = 0
    C_hat = best_c
    while time.time() < deadline:
        t += 1
        c_raw = np.correlate(h, 1.0 - h, mode='full')
        dx = 2.0 / n
        c = c_raw * dx
        m = c.max()
        # Top k active
        idx = np.argsort(c)[::-1][:top_k]
        idx = idx[c[idx] >= m - 1e-5]
        if len(idx) == 0:
            idx = np.array([np.argmax(c)])
        
        g = np.zeros(n)
        p_arr = np.arange(n)
        for ki in idx:
            k = ki - (n - 1)
            vf = (p_arr + k >= 0) & (p_arr + k < n)
            vb = (p_arr - k >= 0) & (p_arr - k < n)
            if np.any(vf):
                g[vf] += 1.0 - h[p_arr[vf] + k]
            if np.any(vb):
                g[vb] -= h[p_arr[vb] - k]
        g *= dx / len(idx)
        
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-10:
            break
        # Step size: balance between Polyak and diminishing
        alpha = (C_hat - m + 0.01) / (g_norm * g_norm + 1e-10)
        alpha = min(alpha, 2.0 / (t + 10))
        alpha = max(alpha, 0.001)
        alpha = min(alpha, 1.0)
        
        h_new = h - alpha * g
        h_new = project_feasible(h_new, n)
        
        # Update average
        h_avg = (t * h_avg + h_new) / (t + 1)
        
        c_new = compute_c5(h_new, n)
        c_avg = compute_c5(h_avg, n)
        
        if c_avg < C_hat:
            C_hat = c_avg
        c_best = min(c_new, c_avg)
        if c_best < best_c:
            best_c = c_best
            if c_new <= c_avg:
                best_h = h_new.copy()
            else:
                best_h = h_avg.copy()
        
        h = h_new
        
        # Every 50 steps, do a short CD to polish
        if t % 50 == 0:
            h_cd, c_cd = exact_cd(h, n, 3, time.time() + 5)
            if c_cd < best_c:
                best_c = c_cd
                best_h = h_cd.copy()
            h = h_cd
    
    # Final CD polish
    h_final, c_final = exact_cd(best_h, n, 50, deadline)
    if c_final < best_c:
        best_c = c_final
        best_h = h_final.copy()
    
    return best_h, best_c

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def _cons_fun(x):
    return x.sum() - len(x) / 2.0

def _cons_jac(x):
    return np.ones(len(x))

def slsqp_opt(h0, n, alphas, maxiter, deadline):
    best_h, best_c = h0.copy(), compute_c5(h0, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-15})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def solve_coord(c, h, p, n):
    h0 = h[p]
    m_min = max(0, n - 1 - p)
    m_max = min(2 * n - 2, 2 * n - 2 - p)
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms].copy()
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    unaff = 0.0
    if m_min > 0:
        unaff = float(np.max(c[:m_min]))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    cur_max = float(np.max(c))
    neg_mask = a < -1e-14
    pos_mask = a > 1e-14
    zero_mask = np.abs(a) <= 1e-14
    L_lo, L_hi = unaff, cur_max
    for _ in range(40):
        if L_hi - L_lo < 1e-13:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        feasible = True
        if np.any(zero_mask) and np.any(c_base[zero_mask] > L + 1e-12):
            feasible = False
        if feasible and np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if feasible and np.any(pos_mask):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        if x_lo > x_hi + 1e-11:
            feasible = False
        if feasible and has_diag:
            cd = c_base[diag_idx[0]]
            ad = a[diag_idx[0]]
            disc = ad * ad + 4.0 * (cd - L)
            if disc >= 0:
                sq = np.sqrt(disc)
                r1 = (ad - sq) / 2.0
                r2 = (ad + sq) / 2.0
                if x_lo > h0 + r1 and x_hi < h0 + r2:
                    feasible = False
        if feasible:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        cd = c_base[diag_idx[0]]
        ad = a[diag_idx[0]]
        disc = ad * ad + 4.0 * (cd - L)
        if disc >= 0:
            sq = np.sqrt(disc)
            r1 = (ad - sq) / 2.0
            r2 = (ad + sq) / 2.0
            if x_lo <= h0 + r1:
                x_hi = min(x_hi, h0 + r1)
            elif x_hi >= h0 + r2:
                x_lo = max(x_lo, h0 + r2)
            else:
                if abs(x_lo - (h0 + r1)) <= abs(x_hi - (h0 + r2)):
                    x_hi = min(x_hi, h0 + r1)
                else:
                    x_lo = max(x_lo, h0 + r2)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def exact_cd(h, n, max_passes, deadline):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            xn = solve_coord(c, h, p, n)
            if abs(xn - h0) > 1e-12:
                delta = xn - h0
                m_min = max(0, n - 1 - p)
                m_max = min(2 * n - 2, 2 * n - 2 - p)
                ms = np.arange(m_min, m_max + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                c[ms] += delta * a
                di = np.where(ms == (n - 1))[0]
                if len(di) > 0:
                    c[ms[di[0]]] -= delta * delta
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def subgrad_explore(h, n, deadline, top_k=40):

    best_h = h.copy()
    best_c = compute_c5(h, n)
    h_avg = h.copy()
    t = 0
    C_hat = best_c
    while time.time() < deadline:
        t += 1
        c_raw = np.correlate(h, 1.0 - h, mode='full')
        dx = 2.0 / n
        c = c_raw * dx
        m = c.max()
        idx = np.argsort(c)[::-1][:top_k]
        idx = idx[c[idx] >= m - 1e-5]
        if len(idx) == 0:
            idx = np.array([np.argmax(c)])
        g = np.zeros(n)
        p_arr = np.arange(n)
        for ki in idx:
            k = ki - (n - 1)
            vf = (p_arr + k >= 0) & (p_arr + k < n)
            vb = (p_arr - k >= 0) & (p_arr - k < n)
            if np.any(vf):
                g[vf] += 1.0 - h[p_arr[vf] + k]
            if np.any(vb):
                g[vb] -= h[p_arr[vb] - k]
        g *= dx / len(idx)
        gn = np.linalg.norm(g)
        if gn < 1e-10:
            break
        alpha = (C_hat - m + 0.005) / (gn * gn + 1e-10)
        alpha = min(alpha, 1.5 / (t + 5))
        alpha = max(alpha, 0.002)
        alpha = min(alpha, 0.8)
        h_new = project_feasible(h - alpha * g, n)
        h_avg = (t * h_avg + h_new) / (t + 1)
        c_new = compute_c5(h_new, n)
        c_avg = compute_c5(h_avg, n)
        if c_avg < C_hat:
            C_hat = c_avg
        if min(c_new, c_avg) < best_c:
            best_c = min(c_new, c_avg)
            best_h = h_new.copy() if c_new <= c_avg else h_avg.copy()
        h = h_new
        if t % 40 == 0:
            h_cd, c_cd = exact_cd(h, n, 3, time.time() + 3)
            if c_cd < best_c:

                best_h = h_cd.copy()

    return h_f, c_f

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def _cons_fun(x):
    return x.sum() - len(x) / 2.0

def _cons_jac(x):
    return np.ones(len(x))

def slsqp_opt(h0, n, alphas, maxiter, deadline):
    best_h, best_c = h0.copy(), compute_c5(h0, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-15})
            nh = project_feasible(res.x, n)
            nc = compute_c5(nh, n)
            if nc < best_c:
                best_h, best_c = nh, nc
        except Exception:
            pass
    return best_h, best_c

def interpolate_h(h_src, n_src, n_tgt):
    x_s = np.linspace(0, 2, n_src, endpoint=False)
    x_t = np.linspace(0, 2, n_tgt, endpoint=False)
    return project_feasible(np.interp(x_t, x_s, h_src), n_tgt)

    return project_feasible(0.5 + 0.4 * h, n)
