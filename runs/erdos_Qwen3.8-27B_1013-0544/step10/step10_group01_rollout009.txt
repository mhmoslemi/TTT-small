def smooth_h_fg(h,n,beta):
    dx=2.0/n
    c=np.correlate(h,1.0-h,mode='full')
    C=c*dx; mx=C.max(); z=beta*(C-mx); ew=np.exp(z); s=ew.sum(); w=ew/(s if s>0 else 1)
    val = mx + (np.log(s) if s>0 else 0.0)/beta
    a=1.0-h
    wrev=w[::-1]
    conv1=np.convolve(a,wrev,mode='full')
    conv2=np.convolve(h,w,mode='full')
    T1=conv1[(n-1):(n-1+n)]
    T2=conv2[(n-1):(n-1+n)]
    grad=dx*(T1-T2)

import numpy as np
import time
from scipy.optimize import minimize


def compute_c5(h, n):
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c)) * dx


def enforce_exact(h, n):
    target = 0.5 * float(n)
    for _ in range(6):
        diff = target - float(h.sum())
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h


def project_feasible(y, n):
    h = np.asarray(y, dtype=float)
    if h.size != n:
        h = np.resize(h, int(n))
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * float(n)
    for _ in range(30):
        diff = target - float(h.sum())
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = float(room.sum())
            if tot <= 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = float(h.sum())
            if tot <= 1e-15:
                break
            h += (diff / tot) * h
        h = np.clip(h, 0.0, 1.0)
    return enforce_exact(h, int(n))


def fix_sum_keep(h, n, p):
    h = np.asarray(h, dtype=float)
    target = 0.5 * float(n)
    for _ in range(8):
        diff = target - float(h.sum())
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            room[p] = 0.0
            tot = float(room.sum())
            if tot <= 1e-15:
                return project_feasible(h, int(n))
            h += (diff / tot) * room
        else:
            mass = h.copy()
            mass[p] = 0.0
            tot = float(mass.sum())
            if tot <= 1e-15:
                return project_feasible(h, int(n))
            h += (diff / tot) * mass
        h = np.clip(h, 0.0, 1.0)
    return enforce_exact(h, int(n))


def expit_vec(z):
    z = np.asarray(z, dtype=float)
    z = np.clip(z, -50.0, 50.0)
    return 1.0 / (1.0 + np.exp(-z))


def logistic_h(x, n):
    x = np.asarray(x, dtype=float)
    target = 0.5 * float(n)
    theta = float(np.mean(x))
    lo = float(np.min(x)) - 50.0
    hi = float(np.max(x)) + 50.0
    h = None
    for _ in range(12):
        if theta < lo:
            theta = lo + 1e-6
        elif theta > hi:
            theta = hi - 1e-6
        h = expit_vec(x - theta)
        err = float(h.sum() - target)
        v = h * (1.0 - h)
        s = float(v.sum())
        if abs(err) < 1e-12:
            break
        if s < 1e-10:
            for __ in range(40):
                mid = 0.5 * (lo + hi)
                sm = float(expit_vec(x - mid).sum())
                if sm > target:
                    lo = mid
                else:
                    hi = mid
            theta = 0.5 * (lo + hi)
            h = expit_vec(x - theta)
            break
        theta += err / s
    if h is None:
        h = expit_vec(x - theta)
    v = h * (1.0 - h)
    return h, v


def h_to_x(h, n):
    eps = 1e-10
    hc = np.clip(h, eps, 1.0 - eps)
    x = np.log(hc / (1.0 - hc))
    return x - float(np.mean(x))


def smooth_h_fg(h, n, beta):
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode='full')
    cm = float(np.max(c))
    z = beta * dx * (c - cm)
    ew = np.exp(z)
    s = float(ew.sum())
    if s > 0.0:
        w = ew / s
        val = cm * dx + float(np.log(s)) / beta
    else:
        w = np.full(2 * n - 1, 1.0 / (2.0 * n - 1.0))
        val = cm * dx
    wr = w[::-1]
    c1 = np.convolve(1.0 - h, wr, mode='full')
    c2 = np.convolve(h, w, mode='full')
    start = int(n - 1)
    grad = dx * (c1[start:start + n] - c2[start:start + n])
    return float(val), grad


def smooth_x_fg(x, n, beta):
    h, v = logistic_h(x, int(n))
    val, gh = smooth_h_fg(h, int(n), beta)
    s = float(v.sum())
    if s > 1e-10:
        t = float(np.dot(gh, v))
        gx = v * (gh - t / s)
    else:
        gx = gh * min(1.0, s / 1e-6)
    if not np.all(np.isfinite(gx)):
        gx = np.zeros(int(n))
    if not np.isfinite(val):
        val = 1e100
    return float(val), gx

def lfgsb_optimize(h, n, alphas, maxiter_each, deadline):
    n = int(n)
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(-50.0, 50.0)] * n
    for beta in alphas:

        x0 = h_to_x(best_h, n)
        try:

                           bounds=bounds, jac=True,
                           options={'maxiter': int(maxiter_each), 'disp': False})
            if hasattr(res, 'x') and res.x is not None:
                hx, _ = logistic_h(res.x, n)
                hs = project_feasible(hx, n)
                cs = compute_c5(hs, n)
                if cs < best_c - 1e-15:

                    best_c = cs
        except Exception:
            pass
    return best_h, float(best_c)


def sum_eq(x, n):
    return float(np.sum(x) - 0.5 * float(n))


def sum_eq_jac(x, n):
    return np.ones(int(n))


def slsqp_optimize(h, n, alphas, maxiter_each, deadline):
    n = int(n)
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (n,)}]

            break
        try:

                           bounds=bounds, constraints=cons,
                           options={'maxiter': int(maxiter_each), 'ftol': 1e-15})
            if hasattr(res, 'x') and res.x is not None:

                cs = compute_c5(hs, n)

                    best_c = cs
        except Exception:
            pass
    return best_h, float(best_c)


def make_cd_data(n):
    n = int(n)
    nm1 = n - 1
    data = []
    for p in range(n):
        m_lo = p
        m_hi = nm1 + max(nm1 - p, p)
        m_hi = min(2 * n - 2, m_hi)
        L = m_hi - m_lo + 1
        d = np.arange(m_lo, m_hi + 1) - nm1
        jf = p + d
        jb = p - d
        vf = (jf >= 0) & (jf < n)
        vb = (jb >= 0) & (jb < n)
        f_pos = np.nonzero(vf)[0]
        f_nei = jf[vf].astype(np.int64)
        b_pos = np.nonzero(vb)[0]
        b_nei = jb[vb].astype(np.int64)
        diag = nm1 - m_lo if (m_lo <= nm1 and nm1 <= m_hi) else -1
        data.append((m_lo, m_hi, int(diag), f_pos, f_nei, b_pos, b_nei))
    return data


def _cd_eval(c_aff, a, diag, unaff, h0, d):
    if abs(d) < 1e-16:
        mx = float(np.max(c_aff))
    else:
        vals = c_aff + d * a
        if diag >= 0:
            vals[diag] = c_aff[diag] + d * (1.0 - 2.0 * h0) - d * d
        mx = float(np.max(vals))
    if unaff > mx:
        mx = unaff
    return mx


def coord_descent(h, n, max_passes, deadline, grid_pts=16, gold_iters=22,
                  max_updates=25, threshold=1e-8, seed=1):
    n = int(n)
    if grid_pts < 4:
        grid_pts = 4
    if gold_iters < 5:
        gold_iters = 5
    if time.time() > deadline - 2.0:
        h = project_feasible(h, n)
        return h, float(compute_c5(h, n))

    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    cd_data = make_cd_data(n)
    rng = np.random.RandomState(int(seed))
    total_lags = 2 * n - 1
    gr = 0.5 * (np.sqrt(5.0) - 1.0)

    for _ in range(int(max_passes)):
        if time.time() > deadline:
            break
        h = best_h.copy()
        c = np.correlate(h, 1.0 - h, mode='full')
        updates = 0
        order = rng.permutation(n)

        for p in order:
            if time.time() > deadline:
                break
            m_lo, m_hi, diag, f_pos, f_nei, b_pos, b_nei = cd_data[p]
            L = m_hi - m_lo + 1
            if L <= 0:
                continue
            h0 = float(h[p])
            dlo = -h0
            dhi = 1.0 - h0
            if dhi - dlo < 1e-14:
                continue

            c_aff = c[m_lo:m_hi + 1]
            a = np.zeros(L)
            if f_nei.size:
                a[f_pos] += 1.0 - h[f_nei]
            if b_nei.size:
                a[b_pos] -= h[b_nei]

            unaff = 0.0
            if m_lo > 0:
                unaff = float(np.max(c[:m_lo]))
            rem = m_hi + 1
            if rem < total_lags:
                u2 = float(np.max(c[rem:]))
                if u2 > unaff:
                    unaff = u2

            f0 = _cd_eval(c_aff, a, diag, unaff, h0, 0.0)
            d_grid = np.linspace(dlo, dhi, int(grid_pts))
            vals = c_aff[None, :] + d_grid[:, None] * a[None, :]
            if diag >= 0:
                c0 = c_aff[diag]
                vals[:, diag] = c0 + d_grid * (1.0 - 2.0 * h0) - d_grid * d_grid
            rowmax = vals.max(axis=1)
            if unaff > 0.0:
                rowmax = np.maximum(rowmax, unaff)

            idx = int(np.argmin(rowmax))
            db = float(d_grid[idx])
            fb = float(rowmax[idx])
            if f0 < fb:
                db = 0.0
                fb = f0

            G = len(d_grid)
            if db == 0.0:
                i = int(np.argmin(np.abs(d_grid)))
            else:
                i = idx
            i0 = max(0, min(i, G - 2))
            lo_i = max(0, i0 - 1)
            hi_i = min(G - 1, i0 + 1)
            glo = float(d_grid[lo_i])
            ghi = float(d_grid[hi_i])
            if db < glo:
                glo = db
            if db > ghi:
                ghi = db

            if ghi > glo + 1e-14:
                x1 = ghi - gr * (ghi - glo)
                x2 = glo + gr * (ghi - glo)
                f1 = _cd_eval(c_aff, a, diag, unaff, h0, x1)
                f2 = _cd_eval(c_aff, a, diag, unaff, h0, x2)
                for __ in range(int(gold_iters)):
                    if ghi - glo < 1e-14:
                        break
                    if f1 < f2:
                        ghi = x2
                        x2 = x1
                        f2 = f1
                        x1 = ghi - gr * (ghi - glo)
                        f1 = _cd_eval(c_aff, a, diag, unaff, h0, x1)
                    else:
                        glo = x1
                        x1 = x2
                        f1 = f2
                        x2 = glo + gr * (ghi - glo)
                        f2 = _cd_eval(c_aff, a, diag, unaff, h0, x2)
                dm = 0.5 * (glo + ghi)
                fm = _cd_eval(c_aff, a, diag, unaff, h0, dm)
                if fm < fb:
                    db = dm
                    fb = fm

            if fb < f0 - 1e-12:
                nv = h0 + db
                if nv < threshold:
                    nv = 0.0
                elif nv > 1.0 - threshold:
                    nv = 1.0
                if nv != h0:
                    h[p] = nv
                    h = fix_sum_keep(h, n, p)
                    updates += 1
                    if updates % 10 == 0:

                        if ct < best_c - 1e-15:

                    if updates >= int(max_updates):

            best_h = h.copy()

    return best_h, float(compute_c5(best_h, n))


def resample_h(h, src_n, dst_n):
    src_n = int(src_n)
    dst_n = int(dst_n)
    if src_n == dst_n:
        return h.copy()
    x0 = (np.arange(src_n) + 0.5) * (2.0 / float(src_n))
    x1 = (np.arange(dst_n) + 0.5) * (2.0 / float(dst_n))
    return np.interp(x1, x0, h)


def polish_candidates(h, n):
    n = int(n)
    best = project_feasible(h, n)
    bc = compute_c5(best, n)
    cands = []
    cands.append(best[::-1].copy())
    cands.append(0.5 * (best + best[::-1]))
    cands.append(0.5 * (best + 1.0 - best[::-1]))
    kern = np.ones(3.0) / 3.0
    cands.append(np.convolve(best, kern, 'same'))
    for cand in cands:
        try:
            cand2 = project_feasible(cand, n)
            cv = compute_c5(cand2, n)
            if cv < bc - 1e-15:
                best = cand2
                bc = cv
        except Exception:
            pass
    for thr in [1e-6, 1e-5, 1e-4, 5e-4, 1e-3, 3e-3, 1e-2, 3e-2]:
        cr = best.copy()
        cr[cr < thr] = 0.0
        cr[cr > 1.0 - thr] = 1.0
        cr = project_feasible(cr, n)
        cv = compute_c5(cr, n)
        if cv < bc - 1e-15:
            best = cr
            bc = cv
    return best, float(bc)

            if cand_dead - time.time() > 10.0:
                h_s, cs = coord_descent(h_s, nt, 100000, cand_dead,
                                        grid_pts=12, gold_iters=18,
                                        max_updates=20, threshold=1e-8,
                                        seed=int(seed) + 101 + nt)
            h_s = project_feasible(h_s, nt)
            cs = float(compute_c5(h_s, nt))
            if cs < best_c - 1e-15:
                best_h, best_n, best_c = h_s, int(len(h_s)), cs
                best_h, best_c = polish_candidates(best_h, best_n)

    left = deadline - time.time()
    if left > 18.0:
        final_dead = min(deadline - 3.0, time.time() + min(70.0, 0.55 * left))
        h, c = coord_descent(best_h, best_n, 100000, final_dead,
                             grid_pts=20, gold_iters=26, max_updates=18,
                             threshold=1e-9, seed=int(seed) + 991)
        if c < best_c - 1e-15:
            best_h, best_n, best_c = h, int(len(h)), float(c)

    if best_n <= 220 and deadline - time.time() > 45.0:
        h, c = slsqp_optimize(best_h, best_n, get_alphas(best_n)[-3:], 45, deadline - 3.0)
        if c < best_c - 1e-15:
            best_h, best_n, best_c = h, int(len(h)), float(c)

    if deadline - time.time() > 4.0:
        best_h, best_c = polish_candidates(best_h, best_n)

    best_h = project_feasible(best_h, best_n)
    best_c = float(compute_c5(best_h, best_n))
    return best_h, best_c, int(len(best_h))
