import numpy as np
from scipy.optimize import minimize, minimize_scalar

import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def solve_1d(c, h, p, n):

    vb = (jb >= 0) & (jb < n)

def update_corr(c, h, p, old_val, new_val, n):
    delta = new_val - old_val
    if abs(delta) < 1e-14:
        return c
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))

        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):

    diag_i = np.where(ms == (n - 1))[0]
    c[ms] += delta * a
    if len(diag_i) > 0:

def exact_cd(h, n, max_passes, deadline, rng=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for it in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        order = list(range(n))
        if rng is not None and it % 2 == 1:
            rng.shuffle(order)
        for p in order:
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            xn = solve_1d(c, h, p, n)
            if abs(xn - h0) > 1e-12:
                update_corr(c, h, p, h0, xn, n)
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def find_free_vars(h, tol=0.02):
    mask = (h > tol) & (h < 1.0 - tol)
    return np.where(mask)[0]

def nelder_mead_free(h, n, free_idx, maxiter, deadline):
    """Optimize only free variables using Nelder-Mead."""
    if len(free_idx) < 2:
        return h, compute_c5(h, n)
    
    h_work = h.copy()
    
    def obj(x):
        h_work[free_idx] = x
        return compute_c5(h_work, n)
    
    x0 = h_work[free_idx].copy()
    try:
        res = minimize(obj, x0, method='Nelder-Mead',
                       options={'maxiter': maxiter, 'xatol': 1e-12, 'fatol': 1e-14})
        if res.fun < obj(x0):
            h_work[free_idx] = res.x
            h_work = project_feasible(h_work, n)
            return h_work, compute_c5(h_work, n)
    except Exception:
        pass
    return h, compute_c5(h, n)

def topk_objective(h, n, k=20):
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    top_k = np.sort(c)[-k:]
    # Smooth max of top-k using log-sum-exp
    alpha = 100.0
    mx = top_k.max()
    return mx + np.log(np.sum(np.exp(-alpha * (mx - top_k)))) / alpha

def slsqp_topk(h, n, k, maxiter, deadline):

    bounds = [(0.0, 1.0)] * n
    
    def cf(x):
        return x.sum() - n / 2.0
    
    def cj(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]
    
    try:
        res = minimize(topk_objective, best_h, args=(n, k), method='SLSQP',

        new_c = compute_c5(new_h, n)
        if new_c < best_c - 1e-14:

def smooth_lse(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.sum(np.exp(shifted - mx))
    return (m + lse / alpha) * (2.0 / n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def solve_1d(c, h, p, n):
    h0 = h[p]
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms].copy()
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    if has_diag:
        c_diag = c_base[diag_idx[0]]
        a[diag_idx] = 0.0
    unaff = 0.0
    if m_min > 0:
        unaff = float(np.max(c[:m_min]))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    cur_max = float(np.max(c))
    neg_mask = a < -1e-15
    pos_mask = a > 1e-15
    zero_mask = np.abs(a) <= 1e-15
    L_lo, L_hi = unaff, cur_max
    for _ in range(60):
        if L_hi - L_lo < 1e-13:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        feasible = True
        if np.any(zero_mask) and np.any(c_base[zero_mask] > L + 1e-14):
            feasible = False
        if feasible and np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if feasible and np.any(pos_mask):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        if x_lo > x_hi + 1e-13:
            feasible = False
        if feasible and has_diag:
            Cq = L - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                if not (max(x_lo, 0.0) < r1 and min(x_hi, 1.0) > r2):
                    feasible = False
        if feasible:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Cq = L - c_diag + h0 - h0 * h0

    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def update_corr(c, h, p, old_val, new_val, n):
    delta = new_val - old_val
    if abs(delta) < 1e-14:
        return c
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    diag_i = np.where(ms == (n - 1))[0]
    c[ms] += delta * a
    if len(diag_i) > 0:
        c[ms[diag_i[0]]] -= delta * delta
    return c

def exact_cd(h, n, max_passes, deadline, rng=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for it in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        order = list(range(n))
        if rng is not None and it % 2 == 1:
            rng.shuffle(order)
        for p in order:
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            xn = solve_1d(c, h, p, n)
            if abs(xn - h0) > 1e-12:
                update_corr(c, h, p, h0, xn, n)
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def find_free(h, tol):
    return np.where((h > tol) & (h < 1.0 - tol))[0]

def nm_free(h, n, free, maxiter, deadline):
    if len(free) < 2:
        return h, compute_c5(h, n)
    h_work = h.copy()
    base_val = compute_c5(h_work, n)

    def obj(x):
        h_work[free] = np.clip(x, 0.0, 1.0)
        return compute_c5(h_work, n)

    x0 = h_work[free].copy()
    try:
        res = minimize(obj, x0, method='Nelder-Mead',
                       options={'maxiter': maxiter, 'xatol': 1e-10, 'fatol': 1e-14,
                                'adaptive': True})
        if res.fun < base_val - 1e-14:
            h_work[free] = np.clip(res.x, 0.0, 1.0)
            h_work = project_feasible(h_work, n)
            return h_work, compute_c5(h_work, n)
    except Exception:
        pass
    return h, base_val

def smooth_lse(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_lse(h, n, alpha, maxiter):
    best_h, best_c = h.copy(), compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    def cf(x):
        return x.sum() - n / 2.0
    def cj(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]
    try:
        res = minimize(smooth_lse, best_h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20})
        new_h = project_feasible(res.x, n)
        new_c = compute_c5(new_h, n)
        if new_c < best_c - 1e-14:
            best_h, best_c = new_h, new_c
    except Exception:
        pass
    return best_h, best_c

def interp_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)
