def smooth_value_and_grad(h,n,alpha,mu,L,orient):
    dx=2.0/n
    om=1.0-h
    c=np.correlate(h,om,'full')
    m=c.max()
    z=alpha*(c-m)
    zmax=z.max()
    center=z-zmax
    expv=np.exp(center)
    se=expv.sum()
    if se == 0:
       w=np.zeros_like(c); amax=np.argmax(c); w[amax]=1.0
    else:
       w=expv/se
    lse=zmax+np.log(se)
    val=dx*(m+lse/alpha)
    w_eff = w if orient==1 else w[::-1]
    Fw=np.fft.rfft(w_eff,L)
    ga=np.fft.irfft(Fw*np.fft.rfft(om,L),L)[:3*n-2]
    gb=np.fft.irfft(Fw*np.fft.rfft(h[::-1],L),L)[:3*n-2]

def project_box_sum(x, S):
    x = np.asarray(x, dtype=float)
    n = x.size
    S = float(min(max(S, 0.0), float(n)))
    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(45):
        mid = 0.5*(lo+hi)
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > S:
            lo = mid
        else:
            hi = mid
    t = 0.5*(lo+hi)
    y = np.clip(x - t, 0.0, 1.0)
    for _ in range(10):
        diff = S - float(y.sum())
        if abs(diff) < 1e-12:
            break
        mask = (y > 1e-14) & (y < 1.0 - 1e-14)
        if mask.any():
            y[mask] += diff / float(mask.sum())
        elif diff > 0:
            room = 1.0 - y
            tot = float(room.sum())
            if tot > 1e-15:
                y += (diff / tot) * room
        else:
            tot = float(y.sum())
            if tot > 1e-15:
                y += (diff / tot) * y
        y = np.clip(y, 0.0, 1.0)
    return y

import numpy as np

def next_fft_len(m):
    L = 1
    while L < m:
        L *= 2
    return L


def fft_convolve(a, b):
    L = next_fft_len(len(a) + len(b) - 1)
    out = np.fft.irfft(np.fft.rfft(a, L) * np.fft.rfft(b, L), L)
    return out[:len(a) + len(b) - 1]


def project_box_sum(x, S):
    x = np.asarray(x, dtype=float)
    n = x.size
    S = float(min(max(S, 0.0), float(n)))
    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(45):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(10):
        diff = S - float(y.sum())
        if abs(diff) < 1e-12:

def softmax_weights(c, alpha):
    if alpha <= 0.0:

            if np.isfinite(fp) and fp <= f_h + 1e-3 * s * gd + 1e-13:
                accepted = True
                best_p = p
                best_fp = fp
                best_exact = exact_p
                break
            s *= 0.5

        if not accepted:
            if best_p is not None and np.isfinite(best_fp) and best_fp <= f_h + 1e-9:
                accepted = True
            else:
                no += 1
                h_prev = h.copy()
                t = 1
                step = max(0.5 * s, step_base * 1e-5)
                if no > patience:
                    alpha = max(base_alpha, 0.5 * alpha)
                    step = step_base
                    no = 0
                    if rng is not None:
                        sc = 0.02 * max(0.1, 10.0 / (alpha + 10.0))
                        h = project_box_sum(h + rng.normal(0.0, sc, n), S)
                    c = c_full(h)
                    w = active_weights(c)
                    w_eff = w if orient == 1 else w[::-1]
                    Fw = np.fft.rfft(w_eff, L)
                    f_h = dx * float(np.dot(w, c))
                    exact_h = float(np.max(c) * dx)
                    continue
                continue

        if best_p is None:
            best_p = project_box_sum(y, S)
            best_fp, best_exact = fixed_w_value_only(best_p, n, w, dx)

        if best_fp < f_h - 1e-10:
            step = min(step * 1.05, 5.0 * step_base)
        else:
            step = max(step * 0.95, step_base * 1e-4)

        h_prev = h
        h = best_p
        t += 1
        f_h = best_fp
        exact_h = best_exact

        if np.isfinite(exact_h) and exact_h < best_c - 1e-13:
            best_c = exact_h
            best_h = h.copy()
            no = 0
        else:
            no += 1
            if no > patience:
                alpha = max(base_alpha, 0.5 * alpha)
                step = step_base
                h_prev = h.copy()
                t = 1
                no = 0
                if rng is not None:
                    sc = 0.02 * max(0.1, 10.0 / (alpha + 10.0))
                    h = project_box_sum(h + rng.normal(0.0, sc, n), S)
                    c = c_full(h)
                    w = active_weights(c)
                    w_eff = w if orient == 1 else w[::-1]
                    Fw = np.fft.rfft(w_eff, L)
                    f_h = dx * float(np.dot(w, c))
                    exact_h = float(np.max(c) * dx)

    h = project_box_sum(h, S)
    c_end = exact_c5(h, n)
    if c_end < best_c:
        best_h, best_c = h, c_end
    return best_h, best_c


def lbfgs_polish(h, n, L, stop_time, orient):
    h = project_box_sum(h, 0.5 * n)
    best_h, best_c = h, exact_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for a in (2.0, 10.0, 50.0, 200.0):
        if time.time() > stop_time - 2.0:
            break
        try:
            res = minimize(
                lbfgs_fn, h, args=(n, a, 100.0, L, orient),
                jac=lbfgs_jac, method='L-BFGS-B', bounds=bounds,
                options={'maxiter': 50, 'maxfun': 200, 'maxls': 8,
                         'ftol': 1e-14, 'gtol': 1e-8}
            )
            x = project_box_sum(res.x, 0.5 * n)
            c = exact_c5(x, n)
            if c < best_c - 1e-12:
                best_h, best_c = x, c
        except Exception:
            pass
    return best_h, best_c


def calibrate_orientation():
    n = 16
    h = np.linspace(0.2, 0.8, n)
    h = project_box_sum(h, 0.5 * n)
    c1 = c_full(h)
    c2 = fft_convolve((1.0 - h)[::-1], h)
    scale = max(1.0, float(np.max(np.abs(c1))))
    if np.max(np.abs(c1 - c2)) <= 1e-8 * scale:
        return 1
    if c2.size == c1.size and np.max(np.abs(c1 - c2[::-1])) <= 1e-8 * scale:
        return -1
    return 1


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget_s = float(budget_s)
    usable = max(1.0, budget_s - 3.0)
    t_end = t0 + budget_s - 2.0
    rng = np.random.RandomState(seed)

    try:
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.full(179, 0.5)

    n0 = len(h0)
    h0 = project_box_sum(h0, 0.5 * n0)
    best_h = h0.copy()
    best_c = exact_c5(h0, n0)
    best_n = n0

    if budget_s <= 12.0:
        return best_h, best_c, best_n

    orient = calibrate_orientation()

    if usable < 60.0:
        Lb = next_fft_len(3 * best_n - 2)
        hf, cf = fista_alt_adaptive(best_h, best_n, Lb, min(t_end - 1.0, t0 + usable * 0.95),
                                    orient, rng, alpha0=1.0, max_iters=20000)
        if cf < best_c:
            best_h, best_c, best_n = hf, cf, len(hf)
        best_h = project_box_sum(best_h, 0.5 * best_n)
        return best_h, exact_c5(best_h, best_n), best_n

    L0 = next_fft_len(3 * n0 - 2)
    stop1 = min(t_end - 60.0, t0 + 0.32 * usable)
    if time.time() < stop1:
        h1, c1 = fista_alt_adaptive(h0, n0, L0, stop1, orient, rng, alpha0=0.2)
        if c1 < best_c:
            best_h, best_c, best_n = h1, c1, n0

    variants = []
    base_h = best_h.copy()
    base_n = best_n
    Sbase = 0.5 * base_n

    hs = project_box_sum(0.5 * (base_h + base_h[::-1]), Sbase)
    variants.append(hs)
    for sc in (0.02, 0.06):
        hp = project_box_sum(base_h + rng.normal(0.0, sc, base_n), Sbase)
        variants.append(hp)
    ii = np.arange(base_n)
    x1 = 0.5 + 0.28 * np.sin(6.0 * np.pi * ii / base_n) + 0.14 * np.cos(10.0 * np.pi * ii / base_n)
    x1 = project_box_sum(0.72 * base_h + 0.28 * x1, Sbase)
    variants.append(x1)
    x2 = 0.5 + 0.22 * np.sin(4.0 * np.pi * ii / base_n) - 0.18 * np.cos(14.0 * np.pi * ii / base_n)
    x2 = project_box_sum(0.78 * base_h + 0.22 * x2, Sbase)
    variants.append(x2)

    stop2 = min(t_end - 30.0, t0 + 0.50 * usable)
    for k, hv in enumerate(variants):
        if time.time() > stop2:
            break
        rem = max(1, len(variants) - k)
        gap = (stop2 - time.time()) / float(rem)
        if gap < 4.0:
            break
        hv_c = exact_c5(hv, base_n)
        if hv_c < best_c:
            best_h, best_c, best_n = hv.copy(), hv_c, base_n
        stopv = min(stop2, time.time() + gap * 0.92)
        if time.time() < stopv:
            hv_opt, hv_c2 = fista_alt_adaptive(hv, base_n, L0, stopv, orient, rng, alpha0=0.5)
            if hv_c2 < best_c:
                best_h, best_c, best_n = hv_opt, hv_c2, base_n

    ns = set()
    for f in (0.75, 0.55, 0.40, 0.30, 0.20, 0.12, 0.08):
        ns.add(int(round(best_n * f)))
    for m in (80, 100, 120, 150, 200, 250, 350, 500, 650, 800, 900, 1000):
        ns.add(m)
    ns = [m for m in ns if 32 <= m <= 1000 and m != best_n]
    ns.sort(reverse=True)
    ns = ns[:8]

    stop3 = min(t_end - 25.0, t0 + 0.82 * usable)
    for i, nn in enumerate(ns):
        if time.time() > stop3:
            break
        rem = max(1, len(ns) - i)
        gap = (stop3 - time.time()) / float(rem)
        if gap < 6.0:
            break
        hv = interp_h(best_h, best_n, nn)
        hv_c = exact_c5(hv, nn)
        if hv_c < best_c:
            best_h, best_c, best_n = hv.copy(), hv_c, nn
        stopv = min(stop3, time.time() + gap * 0.92)
        if time.time() < stopv:
            Lc = next_fft_len(3 * nn - 2)
            hv_opt, hv_c2 = fista_alt_adaptive(hv, nn, Lc, stopv, orient, rng, alpha0=0.8)
            if hv_c2 < best_c:
                best_h, best_c, best_n = hv_opt, hv_c2, nn

    Lb = next_fft_len(3 * best_n - 2)
    stop_final = min(t_end - 2.0, t0 + usable - 1.0)
    if time.time() < stop_final:
        hf, cf = fista_alt_adaptive(best_h, best_n, Lb, stop_final, orient, rng, alpha0=4.0)
        if cf < best_c:
            best_h, best_c, best_n = hf, cf, best_n

    if time.time() < stop_final - 20.0:
        stop_lb = min(stop_final, time.time() + 35.0)
        hb, cb = lbfgs_polish(best_h, best_n, Lb, stop_lb, orient)
        if cb < best_c:
            best_h, best_c = hb, cb

    best_h = project_box_sum(best_h, 0.5 * best_n)
    cp = exact_c5(best_h, best_n)
    if cp < best_c:
        best_c = cp
    return best_h, best_c, best_n
