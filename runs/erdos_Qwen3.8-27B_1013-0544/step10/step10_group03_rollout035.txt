    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def solve_coord(c, h, p, n):
    h0 = h[p]
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))

    diag_idx = np.where(ms == (n - 1))[0]

    cur_max = float(np.max(c))
    neg_mask = a < -1e-15
    pos_mask = a > 1e-15
    zero_mask = np.abs(a) <= 1e-15
    L_lo, L_hi = unaff, cur_max

        if L_hi - L_lo < 1e-14:

        L = (L_lo + L_hi) / 2.0

            feasible = False
        if feasible and np.any(neg_mask):

        if feasible and np.any(pos_mask):

                r2 = (1.0 + np.sqrt(disc)) / 2.0
                xc_lo, xc_hi = max(x_lo, 0.0), min(x_hi, 1.0)
                if not (xc_lo < r1 and xc_hi > r2):
                    feasible = False
        if feasible:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):

    if np.any(pos_mask):

    x_lo = max(x_lo, 0.0)

            else:

    if x_lo > x_hi:
        return h0, a, ms, diag_idx
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0)), a, ms, diag_idx

def exact_cd(h, n, max_passes=200, deadline=None):

    c = np.correlate(h, 1.0 - h, mode='full')
    for _pass in range(max_passes):

            break
        improved = False
        for p in range(n):

            xn, a, ms, diag_idx = solve_coord(c, h, p, n)
            if abs(xn - h0) > 1e-12:

def smooth_obj_and_grad(x, n, alpha):

    dx = 2.0 / n
    ac = alpha * c
    ac_s = ac - ac.max()

            s2 = np.sum(w[js] * x[p - ls])
            grad[p] = dx * (s1 - s2)
    return obj, grad
