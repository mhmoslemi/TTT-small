import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def smooth_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = c.max()
    shifted = alpha * (c - c_max)
    mx = shifted.max()
    exp_vals = np.exp(shifted - mx)
    w = exp_vals / exp_vals.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for p in range(n):
        m_min = n - 1 - p
        m_max = 2 * n - 2 - p
        ms = np.arange(m_min, m_max + 1)
        jf = p + ms - (n - 1)
        jb = p - ms + (n - 1)
        vf = (jf >= 0) & (jf < n)
        vb = (jb >= 0) & (jb < n)
        contrib = np.zeros(len(ms))
        if np.any(vf):
            contrib[vf] += (1.0 - h[jf[vf]])
        if np.any(vb):
            contrib[vb] -= h[jb[vb]]
        diag_mask = (ms == (n - 1))
        contrib[diag_mask] = (1.0 - 2.0 * h[p])
        grad[p] = dx * np.dot(w[ms], contrib)
    return grad, c_max

def smooth_obj_val(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = c.max()
    shifted = alpha * (c - c_max)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (c_max + lse / alpha) * (2.0 / n)

def fista_smooth(h, n, alpha, max_iter, deadline):
    h_curr = h.copy()
    h_prev = h.copy()
    k = 1
    lr = 1.0 / (alpha * 2.0 * n / 2.0)  # initial step size estimate
    # Actually, the Lipschitz constant of the smooth objective is related to alpha and the correlation structure.
    # Let's use backtracking.
    obj_val = smooth_obj_val(h_curr, n, alpha)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * k * k)) / 2.0
        h_y = h_curr + (k - 1.0) / (k + 2.0) * (h_curr - h_prev)
        h_y = np.clip(h_y, 0.0, 1.0)
        grad, _ = smooth_grad(h_y, n, alpha)
        # Backtracking line search
        lr_trial = lr
        for _ in range(50):
            h_try = h_y - lr_trial * grad
            h_try = project_feasible(h_try, n)
            obj_try = smooth_obj_val(h_try, n, alpha)
            if obj_try <= obj_val + np.dot(grad, h_try - h_y) + 1.0 / (2.0 * lr_trial) * np.sum((h_try - h_y) ** 2):
                break
            lr_trial *= 0.5
        lr = lr_trial
        h_prev = h_curr.copy()
        h_curr = h_y - lr * grad
        h_curr = project_feasible(h_curr, n)
        obj_curr = smooth_obj_val(h_curr, n, alpha)
        k = t_k
        if obj_curr < obj_val - 1e-15 and it % 10 == 0:
            obj_val = obj_curr
        else:
            # If we didn't decrease, update obj_val anyway for the line search
            obj_val = max(obj_val, obj_curr)
    return h_curr

def solve_1d_bisect(c, h, p, n):
    h0 = h[p]
    m_min = n - 1 - p

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def solve_1d_bisect(c, h, p, n):
    h0 = h[p]
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms]
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    if has_diag:
        a[diag_idx] = 0.0
    unaff = 0.0
    if m_min > 0:
        unaff = float(np.max(c[:m_min]))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    cur_max = float(np.max(c))
    neg_mask = a < -1e-15
    pos_mask = a > 1e-15
    zero_mask = np.abs(a) <= 1e-15
    L_lo, L_hi = unaff, cur_max
    for _ in range(55):
        if L_hi - L_lo < 1e-14:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        feasible = True
        if np.any(zero_mask) and np.any(c_base[zero_mask] > L):
            feasible = False
        if feasible and np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if feasible and np.any(pos_mask):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        if x_lo > x_hi + 1e-13:
            feasible = False
        if feasible and has_diag:
            Cq = L - c_base[diag_idx[0]] + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                xc_lo, xc_hi = max(x_lo, 0.0), min(x_hi, 1.0)
                if not (xc_lo < r1 and xc_hi > r2):
                    feasible = False
        if feasible:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Cq = L - c_base[diag_idx[0]] + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            xc_lo, xc_hi = max(x_lo, 0.0), min(x_hi, 1.0)
            if (xc_lo < r1 and xc_hi > r2) or (x_lo >= r1 and x_hi <= r2):
                if h0 - r1 <= r2 - h0:
                    x_hi = min(x_hi, r1)
                else:
                    x_lo = max(x_lo, r2)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def exact_cd(h, n, max_passes, deadline, random_order=False, rng=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        if random_order and rng is not None:
            order = rng.permutation(n)
        else:
            order = np.arange(n)
        for p in order:
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            xn = solve_1d_bisect(c, h, p, n)
            if abs(xn - h0) > 1e-13:
                delta = xn - h0
                ms = np.arange(n - 1 - p, 2 * n - 1 - p)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                diag_i = np.where(ms == (n - 1))[0]
                c[ms] += delta * a
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def grad_smooth(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = c.max()
    shifted = alpha * (c - c_max)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for p in range(n):
        m_min = n - 1 - p
        m_max = 2 * n - 2 - p
        ms = np.arange(m_min, m_max + 1)
        jf = p + ms - (n - 1)
        jb = p - ms + (n - 1)
        vf = (jf >= 0) & (jf < n)
        vb = (jb >= 0) & (jb < n)
        contrib = np.zeros(len(ms))
        if np.any(vf):
            contrib[vf] += (1.0 - h[jf[vf]])
        if np.any(vb):
            contrib[vb] -= h[jb[vb]]
        contrib[ms == (n - 1)] = (1.0 - 2.0 * h[p])
        grad[p] = dx * np.dot(w[ms], contrib)
    return grad

def smooth_gd(h, n, alpha, n_steps, lr, deadline):
    h = h.copy()
    for _ in range(n_steps):
        if time.time() > deadline:
            break
        g = grad_smooth(h, n, alpha)
        h = h - lr * g
        h = project_feasible(h, n)
    return h

    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:

        n = len(h_best)
    except Exception:

        n = 500

    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)

    # Phase 1: Strong CD convergence
    h1, c1 = exact_cd(h_best, n, 400, deadline)
    if c1 < best_c:
        h_best, best_c = h1, c1

    # Phase 2: Smooth GD with varying alpha + CD polish
    if time.time() < deadline - 400:
        for alpha in [100, 300, 1000, 3000]:
            if time.time() > deadline - 300:
                break
            for lr in [0.005, 0.002]:
                if time.time() > deadline - 280:
                    break
                hg = smooth_gd(h_best, n, alpha, 200, lr, deadline)
                ho, co = exact_cd(hg, n, 250, deadline)
                if co < best_c:
                    h_best, best_c = ho, co

    # Phase 3: Structured perturbations
    if time.time() < deadline - 300:
        x = np.linspace(0, 2, n)
        # Fourier mode perturbations
        for freq in [2, 4, 6, 8, 12, 16]:
            if time.time() > deadline - 200:

            for amp in [0.05, 0.1, 0.15]:
                if time.time() > deadline - 190:

                phase = rng.uniform(0, 2.0 * np.pi)
                hp = h_best + amp * np.sin(np.pi * freq * x + phase)
                hp = project_feasible(hp, n)
                ho, co = exact_cd(hp, n, 250, deadline)
                if co < best_c:
                    h_best, best_c = ho, co

        # Block structure perturbations
        for nb in [4, 6, 8, 12]:
            if time.time() > deadline - 100:

            block_size = n // nb
            hp = h_best.copy()
            for bi in range(nb):
                start = bi * block_size
                end = start + block_size if bi < nb - 1 else n
                shift = rng.uniform(-0.15, 0.15)
                hp[start:end] += shift
            hp = project_feasible(hp, n)
            ho, co = exact_cd(hp, n, 250, deadline)

                h_best, best_c = ho, co

    # Phase 4: Random-order CD escape
    if time.time() < deadline - 150:

        if c2 < best_c:
            h_best, best_c = h2, c2

    # Phase 5: Multi-n exploration
    if time.time() < deadline - 200:
        n_list = [n - 50, n + 50, n - 100, n + 100, 400, 600, 800]
        n_list = [x for x in n_list if 200 <= x <= 1200 and x != n]
        for n_try in n_list:
            if time.time() > deadline - 100:
                break
            x_old = np.linspace(0, 2, n)
            x_new = np.linspace(0, 2, n_try)
            h_int = project_feasible(np.interp(x_new, x_old, h_best), n_try)
            ho, co = exact_cd(h_int, n_try, 300, deadline)
            if co < best_c:

    if time.time() < deadline - 100:
        for scale in [0.02, 0.04, 0.06, 0.03]:
            if time.time() > deadline - 50:
                break
            c_cur = np.correlate(h_best, 1.0 - h_best, mode='full')
            cmax = c_cur.max()

            n_act = max(1, int(active.sum()))
            grad = np.zeros(n)
            for p in range(n):
                m_min = n - 1 - p
                m_max = 2 * n - 2 - p
                ms = np.arange(m_min, m_max + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                contrib = np.zeros(len(ms))
                if np.any(vf):
                    contrib[vf] += (1.0 - h_best[jf[vf]])
                if np.any(vb):
                    contrib[vb] -= h_best[jb[vb]]
                contrib[ms == (n - 1)] = (1.0 - 2.0 * h_best[p])
                grad[p] = np.sum(contrib[active[ms]]) / n_act
            hp = project_feasible(h_best - scale * grad, n)
            ho, co = exact_cd(hp, n, 250, deadline)
            if co < best_c:
                h_best, best_c = ho, co

    # Phase 7: Final intensive CD
    if time.time() < deadline - 3:
        n = len(h_best)
        hf, cf = exact_cd(h_best, n, 500, deadline)
        if cf < best_c:
            h_best, best_c = hf, cf

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
