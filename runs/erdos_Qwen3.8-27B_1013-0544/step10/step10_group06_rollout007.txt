import numpy as np
import time
from scipy.optimize import minimize
from scipy.fft import rfft, irfft, next_fast_len

def c5_raw(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')))

def project_h(y, target, iters=30, polish=False):
    y = np.asarray(y, dtype=float)
    low, high = y.min() - 1.0, y.max()
    for _ in range(iters):
        mid = 0.5 * (low + high)
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            low = mid
        else:
            high = mid
    h = np.clip(y - 0.5 * (low + high), 0.0, 1.0)
    if polish:
        for _ in range(60):
            diff = target - h.sum()
            if abs(diff) < 1e-13:
                break
            if diff > 0:
                room = 1.0 - h
                tot = room.sum()
                if tot < 1e-15:
                    break
                h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
            else:
                tot = h.sum()
                if tot < 1e-15:
                    break
                h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def calibrate_orientation():
    a = np.array([0.11, 0.73, 0.42, 0.88, 0.25, 0.56, 0.07, 0.64])
    n = len(a)
    off = n - 1
    v = 1.0 - a
    c = np.correlate(a, v, mode='full')
    m = np.empty(2 * n - 1)
    for d in range(-n + 1, n):
        if d >= 0:
            m[off + d] = np.dot(a[:n - d], v[d:])
        else:
            m[off + d] = np.dot(a[-d:], v[:-d])
    return 0 if np.linalg.norm(m - c) <= np.linalg.norm(m - c[::-1]) else 1

def fft_slices(h, w, n):
    off = n - 1
    L = next_fast_len(3 * n - 2)
    H = rfft(h, L)
    out1 = irfft(H * rfft(w, L), L)[off:off + n]
    out2 = irfft(H * rfft(w[::-1], L), L)[off:off + n]
    return out1, out2

def apply_weights(w, h, n):
    off = n - 1
    pref = np.empty(2 * n)
    pref[0] = 0.0
    np.cumsum(w, out=pref[1:])
    i = np.arange(n)
    t1a = pref[off + n - i] - pref[off - i]
    t2, t1b = fft_slices(h, w, n)
    return t1a - t1b - t2

def lse_val_grad(h, n, alpha, orient):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = float(np.max(c))
    z = alpha * (c - m)
    zmax = float(np.max(z))
    ez = np.exp(z - zmax)
    s = float(np.sum(ez))
    val = m + (zmax + np.log(s)) / alpha
    w = ez / s
    if orient:
        w = w[::-1]
    return val, apply_weights(w, h, n)

def slsqp_obj(x, n, alpha, orient):
    v, _ = lse_val_grad(x, n, alpha, orient)
    return v * (2.0 / n)

def slsqp_jac(x, n, alpha, orient):
    _, g = lse_val_grad(x, n, alpha, orient)
    return g * (2.0 / n)

def sum_fun(x, n):
    return x.sum() - 0.5 * n

def sum_jac(x, n):
    return np.ones(n)

def slsqp_smooth(h, n, alphas, maxiter, deadline, orient):
    best = h.copy()
    best_r = c5_raw(best)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac, 'args': (n,)}]
    for alpha in alphas:
        if time.time() > deadline - 60:
            break
        try:
            res = minimize(
                slsqp_obj, best, args=(n, alpha, orient),
                jac=slsqp_jac, method='SLSQP', bounds=bounds, constraints=cons,
                options={'maxiter': maxiter, 'ftol': 1e-11, 'maxls': 8, 'disp': False}
            )
            x = project_h(res.x, 0.5 * n, 50, True)
            r = c5_raw(x)
            if r < best_r:
                best, best_r = x, r
        except Exception:
            pass
    return best, best_r

def try_accept(h, n, best_h, best_r):
    h = project_h(h, 0.5 * n, 50, True)
    r = c5_raw(h)
    if r < best_r:
        return h, r
    return best_h, best_r

def adam_pg(h, n, alpha, iters, lr, deadline, orient):

    best = x.copy()
    best_r = c5_raw(x)

    v = np.zeros(n)
    t = 0
    b1, b2 = 0.9, 0.999

        t += 1
        m = b1 * m + (1.0 - b1) * g
        v = b2 * v + (1.0 - b2) * g * g
        y = x - lr * (m / (1.0 - b1 ** t)) / (np.sqrt(v / (1.0 - b2 ** t)) + 1e-8)
        x = project_h(y, 0.5 * n, 25, False)
        if it % 35 == 0 or it == iters - 1:
            r = c5_raw(x)
            if r < best_r:
                best, best_r = x.copy(), r
    return best, best_r

def perturb_h(h, amp, rng):
    n = len(h)
    x = 2.0 * np.arange(n) / n
    p = np.zeros(n)
    for j in range(1, 6):
        p = p + rng.normal() * np.cos(j * np.pi * x) + rng.normal() * np.sin(j * np.pi * x)
    p -= p.mean()
    mx = float(np.max(np.abs(p)))
    if mx > 0.0:
        p = p / mx
    return h + amp * p

def fw_candidate(h, n, orient, alpha):
    _, g = lse_val_grad(h, n, orient, alpha)
    o = np.argsort(g)
    s = np.zeros(n)
    s[o[:n // 2]] = 1.0
    return s

def path_best(a, b, n, grid=13):
    best = a.copy()
    br = c5_raw(a)
    for i in range(1, grid + 1):
        lam = float(i) / grid
        x = (1.0 - lam) * a + lam * b
        r = c5_raw(x)
        if r < br:
            br = r
            best = x.copy()
    return best

def coord_descent(h, n, passes, deadline, rng=None, random=True, nbisect=24):
