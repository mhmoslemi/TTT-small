            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def smooth_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = c.max()
    exp_c = np.exp(alpha * (c - c_max))
    w = exp_c / exp_c.sum()
    obj = c_max + np.log(exp_c.sum()) / alpha
    grad = np.zeros(n)
    for s in range(-(n - 1), n):
        ws = w[s + n - 1]
        if ws < 1e-14:
            continue
        if s >= 0:
            m = n - s
            if m > 0:
                grad[:m] += ws * (1.0 - h[s:]) * dx
                grad[s:] -= ws * h[:m] * dx
        else:
            ss = -s
            m = n - ss
            if m > 0:
                grad[ss:] += ws * (1.0 - h[:m]) * dx
                grad[:m] -= ws * h[ss:] * dx
    return obj, grad

def momentum_smooth(h, n, deadline, alphas, iters, lr):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        mom = np.zeros(n)
        h_cur = best_h.copy()
        for it in range(iters):
            if time.time() > deadline:
                break
            _, grad = smooth_grad(h_cur, n, alpha)
            mom = 0.85 * mom + 0.15 * grad
            h_new = project_feasible(h_cur - lr * mom, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
            h_cur = h_new
    return best_h, best_c

def _cd_obj(c_aff, a, d, diag_idx, has_diag, h0, unaff_max):
    vals = c_aff + d * a
    if has_diag:
        vals = vals.copy()
        vals[diag_idx] = c_aff[diag_idx] + d * (1.0 - h0 - d)
    r = float(np.max(vals))
    return max(r, unaff_max)

def coord_descent(h, n, max_passes, deadline, gs_iters=50):
    h = h.copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-13 or h0 >= 1.0 - 1e-13:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(vf):
                a[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                a[vb] -= h[jb[vb]]
            diag_pos = np.where(ms == (n - 1))[0]
            has_diag = len(diag_pos) > 0
            diag_idx = int(diag_pos[0]) if has_diag else -1
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            d_lo = -h0
            d_hi = 1.0 - h0
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            f1 = _cd_obj(c_aff, a, d1, diag_idx, has_diag, h0, unaff_max)
            f2 = _cd_obj(c_aff, a, d2, diag_idx, has_diag, h0, unaff_max)
            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-15:
                    break
                if f1 < f2:
                    d_hi, d2, f2 = d2, d1, f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    f1 = _cd_obj(c_aff, a, d1, diag_idx, has_diag, h0, unaff_max)
                else:
                    d_lo, d1, f1 = d1, d2, f2
                    d2 = d_lo + gr * (d_hi - d_lo)
                    f2 = _cd_obj(c_aff, a, d2, diag_idx, has_diag, h0, unaff_max)
            d_opt = (d_lo + d_hi) / 2.0
            f_opt = _cd_obj(c_aff, a, d_opt, diag_idx, has_diag, h0, unaff_max)
            f_curr = _cd_obj(c_aff, a, 0.0, diag_idx, has_diag, h0, unaff_max)
            if f_opt < f_curr - 1e-14:
                h[p] = h0 + d_opt
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def slsqp_opt(h, n, alphas, maxiter=200, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n

    def objective(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        shifted = 100.0 * (c - m)
        mx = shifted.max()
        lse = mx + np.log(np.sum(np.exp(shifted - mx)))
        return (m + lse / 100.0)

    def objective_grad(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        shifted = 100.0 * (c - m)
        exp_c = np.exp(shifted - shifted.max())
        w = exp_c / exp_c.sum()
        g = np.zeros(n)
        for s in range(-(n - 1), n):
            ws = w[s + n - 1]
            if ws < 1e-14:
                continue
            if s >= 0:
                mm = n - s
                if mm > 0:
                    g[:mm] += ws * (1.0 - x[s:]) * dx * 100.0
                    g[s:] -= ws * x[:mm] * dx * 100.0
            else:
                ss = -s
                mm = n - ss
                if mm > 0:
                    g[ss:] += ws * (1.0 - x[:mm]) * dx * 100.0
                    g[:mm] -= ws * x[ss:] * dx * 100.0
        return g

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _ceq, 'jac': _ceq_jac, 'args': (n,)}]
    for alpha_val in alphas:
        if deadline is not None and time.time() > deadline:
            break
        try:
            res = minimize(objective, best_h, jac=objective_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def _ceq(x, n):
    return x.sum() - n / 2.0

def _ceq_jac(x, n):
    return np.ones(n)

def structured_perturb(h, n, rng, mode):
    x = np.linspace(0, 2, n)
    if mode == 0:
        return project_feasible(h + 0.02 * rng.normal(0, 1, n), n)
    elif mode == 1:
        freq = 3 + rng.randint(0, 4)
        phase = rng.uniform(0, 2 * np.pi)
        return project_feasible(h + 0.03 * np.sin(2 * np.pi * freq * x / 2 + phase), n)
    elif mode == 2:
        block = max(1, n // 9)
        h2 = h.copy()
        start = rng.randint(0, max(1, n - block))
        h2[start:start + block] = 1.0 - h2[start:start + block]
        return project_feasible(h2, n)
    else:
        freq = 1 + rng.randint(0, 3)
        h2 = h * (1.0 + 0.03 * np.cos(2 * np.pi * freq * x / 2))
        return project_feasible(h2, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 700
        h_init = np.full(n, 0.5)
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: Momentum smooth gradient on parent
    rem = deadline - time.time()
    if rem > 150:
        h1, c1 = momentum_smooth(best_h, n, deadline - 150,
                                  [50, 100, 200, 500], max(50, int(rem // 6)), lr=1.5)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Phase 2: Coordinate descent polish
    rem = deadline - time.time()
    if rem > 120:
        h2, c2 = coord_descent(best_h, n, 60, deadline - 120, gs_iters=55)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Phase 3: SLSQP refinement
    rem = deadline - time.time()
    if rem > 100 and len(best_h) <= 800:
        h3, c3 = slsqp_opt(best_h, len(best_h), [10000, 50000], 200, deadline - 100)
        if c3 < best_c:
            best_h, best_c = h3, c3

    # Phase 4: Multi-resolution exploration
    x_best = np.linspace(0, 2, len(best_h))
    n_list = [500, 600, 700, 800, 900, 1000]
    for n_try in n_list:
        rem = deadline - time.time()
        if rem < 100:
            break
        if n_try == len(best_h):
            continue
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_best, best_h), n_try)
        h_o, c_o = momentum_smooth(h_int, n_try, deadline - 100, [100, 300], 80, lr=1.2)
        h_o, c_o = coord_descent(h_o, n_try, 40, deadline - 100, gs_iters=45)
        if c_o < best_c:
            best_h, best_c = h_o, c_o
            x_best = np.linspace(0, 2, len(best_h))

    # Phase 5: Structured restarts
    rem = deadline - time.time()
    if rem > 120:
        n_cur = len(best_h)
        n_restarts = max(3, int(rem // 50))
        for r in range(n_restarts):
            if time.time() > deadline - 20:
                break
            hp = structured_perturb(best_h, n_cur, rng, r % 4)
            h_o, c_o = momentum_smooth(hp, n_cur, deadline - 20, [100, 300], 60, lr=1.2)
            h_o, c_o = coord_descent(h_o, n_cur, 30, deadline - 20, gs_iters=40)
            if c_o < best_c:
                best_h, best_c = h_o, c_o

    # Phase 6: Final polish
    rem = deadline - time.time()
    if rem > 20:
        h_f, c_f = coord_descent(best_h, len(best_h), 50, deadline - 3, gs_iters=60)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
        if rem > 40 and len(best_h) <= 800:
            h_f, c_f = slsqp_opt(best_h, len(best_h), [50000, 100000], 300, deadline - 3)
            if c_f < best_c:
                best_h, best_c = h_f, c_f
            h_f, c_f = coord_descent(best_h, len(best_h), 30, deadline - 3, gs_iters=55)
            if c_f < best_c:
                best_h, best_c = h_f, c_f

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
