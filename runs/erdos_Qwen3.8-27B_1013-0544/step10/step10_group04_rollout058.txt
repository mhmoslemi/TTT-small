        pm = a > 1e-15
        zm = np.abs(a) <= 1e-15
        feas = True
        if np.any(zm) and np.any(c_base[zm] > L):
            feas = False
        if feas and np.any(nm):
            x_lo = max(x_lo, float(np.max((L - c_base[nm]) / a[nm])))
        if feas and np.any(pm):
            x_hi = min(x_hi, float(np.min((L - c_base[pm]) / a[pm])))
        if x_lo > x_hi + 1e-14:
            feas = False
        if feas and has_diag:
            Cq = L - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                lo2, hi2 = max(x_lo, 0.0), min(x_hi, 1.0)
                if not (lo2 < r1 and hi2 > r2):
                    feas = False
        L_hi = L if feas else L_lo
        L_lo = L if not feas else L_lo
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    nm = a < -1e-15
    pm = a > 1e-15
    if np.any(nm):
        x_lo = max(x_lo, float(np.max((L - c_base[nm]) / a[nm])))
    if np.any(pm):
        x_hi = min(x_hi, float(np.min((L - c_base[pm]) / a[pm])))
    x_lo, x_hi = max(x_lo, 0.0), min(x_hi, 1.0)
    if has_diag:
        Cq = L - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            if x_lo > r1 and x_lo < r2:
                x_lo = r2
            elif x_hi > r1 and x_hi < r2:
                x_hi = r1
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def exact_cd(h, n, max_passes, deadline, random_order=False, rng=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        order = rng.permutation(n) if (random_order and rng is not None) else np.arange(n)
        for p in order:
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            xn = solve_1d(c, h, p, n)
            if abs(xn - h0) > 1e-13:
                delta = xn - h0
                ms = np.arange(n - 1 - p, 2 * n - 1 - p)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                di = np.where(ms == n - 1)[0]
                c[ms] += delta * a
                if len(di) > 0:
                    c[ms[di[0]]] -= delta * delta
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_box_sum(h, n, n / 2.0)
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    lse = mx + np.log(np.sum(np.exp(sh - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_opt(h, n, alphas, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        def obj(x):
            return smooth_obj(x, n, alpha) + 1e4 * (x.sum() - n / 2.0) ** 2
        try:
            res = minimize(obj, best_h, method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14})
            new_h = project_box_sum(res.x, n, n / 2.0)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def binary_local_search(h, n, deadline, rng, max_iters=200000):
    target = n // 2
    h_bin = (h >= 0.5).astype(float)
    ones = np.where(h_bin == 1)[0]
    zeros = np.where(h_bin == 0)[0]
    if len(ones) != target:
        idx = np.argsort(h)[::1]
        h_bin = np.zeros(n)
        h_bin[idx[:target]] = 1.0
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
    c_cur = compute_c5(h_bin, n)
    t_start = time.time()
    temp = 0.01
    for it in range(max_iters):
        if time.time() > deadline:
            break
        temp = max(0.001, temp * 0.9997)
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        h_bin[i], h_bin[j] = 1.0, 0.0
        c_new = compute_c5(h_bin, n)
        delta = c_new - c_cur
        if delta < 0 or (delta > 0 and rng.rand() < np.exp(-delta / temp)):
            c_cur = c_new
            ones = (ones + 1) % (len(ones) + 1)
            ones = ones[ones < len(ones)]
            ones = np.where(h_bin == 1)[0]
            zeros = np.where(h_bin == 0)[0]
        else:
            h_bin[i], h_bin[j] = 0.0, 1.0
    return h_bin, c_cur

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = np.full(500, 0.5)
        n = 500
    h_best = project_box_sum(h_best, n, n / 2.0)
    best_c = compute_c5(h_best, n)

    # Phase 1: Quick CD polish
    h1, c1 = exact_cd(h_best, n, 80, min(deadline, t0 + 60))
    if c1 < best_c:
        h_best, best_c = h1, c1

    # Phase 2: L-BFGS-B with multiple sharpness levels
    if time.time() < deadline - 200:
        h2, c2 = slsqp_opt(h_best, n, [50000, 200000, 500000, 1000000], 400, min(deadline, t0 + 200))
        if c2 < best_c:
            h_best, best_c = h2, c2
        h2b, c2b = exact_cd(h_best, n, 100, min(deadline, t0 + 300))
        if c2b < best_c:
            h_best, best_c = h2b, c2b

    # Phase 3: Binary rounding + annealing + re-continuous
    if time.time() < deadline - 200:
        hb, cb = binary_local_search(h_best, n, min(deadline, t0 + 500), rng, max_iters=150000)
        # Re-continuous: treat binary as starting point, add small noise, optimize
        h_relax = np.clip(hb + rng.normal(0, 0.05, n), 0.0, 1.0)
        h_relax = project_box_sum(h_relax, n, n / 2.0)
        h3, c3 = exact_cd(h_relax, n, 100, min(deadline, t0 + 650))
        if c3 < best_c:
            h_best, best_c = h3, c3
        # Also try the binary solution directly
        if cb < best_c:
            h_best, best_c = hb, cb

    # Phase 4: Interpolate to larger n and optimize
    n_list = [900, 998, 800]
    for n_try in n_list:
        if time.time() > deadline - 180:
            break
        x_old = np.linspace(0, 2, len(h_best), endpoint=False) + 1.0 / len(h_best)
        x_new = np.linspace(0, 2, n_try, endpoint=False) + 1.0 / n_try
        h_int = np.interp(x_new, x_old, h_best, period=2.0)
        h_int = project_box_sum(h_int, n_try, n_try / 2.0)
        hi, ci = slsqp_opt(h_int, n_try, [100000, 500000], 200, min(deadline, time.time() + 120))
        hi, ci = exact_cd(hi, n_try, 80, min(deadline, time.time() + 60))
        if ci < best_c - 1e-7:
            h_best, best_c = hi, ci

    # Phase 5: Diverse restarts
    n = len(h_best)
    for s in [0.01, 0.02, 0.03, 0.05]:
        if time.time() > deadline - 60:
            break
        hp = project_box_sum(h_best + rng.normal(0, s, n), n, n / 2.0)
        ho, co = exact_cd(hp, n, 100, min(deadline, time.time() + 40))
        if co < best_c:
            h_best, best_c = ho, co

    # Phase 6: Final CD polish
    if time.time() < deadline - 3:
        hf, cf = exact_cd(h_best, len(h_best), 200, deadline)
        if cf < best_c:
            h_best, best_c = hf, cf

    h_best = project_box_sum(h_best, len(h_best), len(h_best) / 2.0)
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
