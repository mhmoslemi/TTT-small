import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    """Project h onto {0<=x<=1, sum(x)=n/2} using bisection on Lagrange multiplier."""
    h = h.copy()
    target = n / 2.0
    # Bisection on lambda
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - (lo+hi)/2.0, 0.0, 1.0)
    # Exact correction for floating point
    diff = target - x.sum()
    if abs(diff) > 1e-15:
        # Distribute evenly among non-bound variables
        free = (x > 1e-14) & (x < 1.0 - 1e-14)
        if free.sum() > 0:
            x[free] += diff / free.sum()
            x = np.clip(x, 0.0, 1.0)
    return x

def lse_smooth_max(h, n, alpha):
    """Compute smooth approximation of max(c[k]) using log-sum-exp."""
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = np.max(shifted)
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n), c

def lse_gradient(h, n, alpha):
    """Compute gradient of the LSE smooth max objective."""
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = np.max(shifted)
    weights = np.exp(shifted - mx)
    weights /= weights.sum()
    
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-15:
            continue
        kf = k - (n - 1)
        # First term: for j where i=j-kf is valid, contribution is w*(1-h[i])
        j_lo = max(0, kf)
        j_hi = min(n - 1, n - 1 + kf)
        if j_lo <= j_hi:
            i_indices = np.arange(j_lo, j_hi + 1) - kf
            grad[j_lo:j_hi+1] += w * (1.0 - h[i_indices])
        # Second term: for j where i=j+kf is valid, contribution is -w*h[i]
        if kf >= 0:
            j_lo2 = 0
            j_hi2 = n - 1 - kf
        else:
            j_lo2 = -kf
            j_hi2 = n - 1
        if j_lo2 <= j_hi2:
            i_indices2 = np.arange(j_lo2, j_hi2 + 1) + kf
            grad[j_lo2:j_hi2+1] -= w * h[i_indices2]
    grad *= dx
    return lse_smooth_max(h, n, alpha)[0], grad

def fista(h, n, alpha, n_steps, deadline, lr=None):
    """FISTA (accelerated proximal gradient) on the LSE objective."""
    h = project_feasible(h.copy(), n)
    y = h.copy()
    t_k = 1.0
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    if lr is None:
        # Estimate Lipschitz constant
        # L ≈ max over k of sum of |partial c[k]/partial h[i]|^2 ... rough bound
        # For now use a conservative step
        lr = 1.0 / (alpha * 2.0 * (2.0/n) * n)  # very rough
        lr = min(lr, 1.0)
    
    for s in range(n_steps):
        if time.time() > deadline:
            break
        f, grad = lse_gradient(y, n, alpha)
        # Gradient step
        z = y - lr * grad
        # Proximal step (projection)
        h_new = project_feasible(z, n)
        # Update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k**2)) / 2.0
        y = h_new + (t_k - 1.0) / t_new * (h_new - h)
        h = h_new
        t_k = t_new
        
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    return best_h, best_c

def lse_obj_and_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = np.max(shifted)
    w = np.exp(shifted - mx)
    w /= w.sum()
    val = (m + np.log(np.sum(w)) / alpha) * (2.0 / n)
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        kf = k - (n - 1)
        i1_lo = max(0, kf)

        best_c, best_h = c_s2, h_s2.copy()
    # Phase 5: FISTA high alpha
    if time.time() < deadline - 30:
        for alpha in [20000, 50000]:
            if time.time() > deadline - 10:
                break
            h_new, c_new = fista_run(best_h, n, alpha, 40, deadline)
            if c_new < best_c:
                best_c, best_h = c_new, h_new.copy()
    # Phase 6: Multi-start diversification
    if time.time() < deadline - 100:
        for trial in range(3):
            if time.time() > deadline - 30:
                break
            h_try = structured_init(n, rng)
            h_try, c_try = coord_descent(h_try, n, 20, deadline, rng, 35)
            if c_try < best_c:
                best_c, best_h = c_try, h_try.copy()
    # Phase 7: Try different n
    if time.time() < deadline - 60:
        for n_try in [n + 100, n - 100, 500, 900]:
            if time.time() > deadline - 15 or n_try < 200:
                break
            x_orig = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
            h_opt, c_opt = coord_descent(h_int, n_try, 25, deadline, rng, 40)
            if c_opt < best_c:
                best_c, best_h = c_opt, h_opt.copy()
    # Phase 8: Final polish
    remaining = deadline - time.time()
    if remaining > 15:
        max_p = max(15, int(remaining * 0.8))
        h_pf, c_pf = coord_descent(best_h, len(best_h), max_p, deadline, rng, 50)
        if c_pf < best_c:
            best_c, best_h = c_pf, h_pf.copy()
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
