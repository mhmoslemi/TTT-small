    # Condition: max(0, t-j) <= t < min(n, j+t) [since n-m_k = n-j+t]
    # Also 0 <= k <= 2n-2, i.e., 0 <= j-t+n-1 <= 2n-2, i.e., 1-n <= t-j <= n-1, i.e., t in [j-1+1-n... 
    
    # This is way too complicated analytically. Let me just use a direct but efficient computation.
    
    # Alternative: use the fact that the gradient of the sum of weighted correlations can be computed
    # as a convolution-like operation.
    
    # Let me define u = w (the weights, length 2n-1)
    # Then grad[j] = dx * [sum_k w_k * (1-h[j-m_k]) * 1[valid] - sum_k w_k * h[j+m_k] * 1[valid]]
    
    # The first sum is a cross-correlation of w with (1-h), and the second is a cross-correlation of w with h.
    # Specifically:
    # sum_k w_k * h[j-m_k] = (w * h)[j] where * denotes appropriate correlation
    # Let's think of it as: for fixed j, sum over k of w[k] * h[j - (k-(n-1))]
    # = sum over k of w[k] * h[j - k + n - 1]
    # Let t = j - k + n - 1, so k = j - t + n - 1
    # = sum over t of w[j-t+n-1] * h[t] (where valid indices)
    # This is a correlation!
    
    # So: sum_k w_k * h[j-m_k] = correlate(h, w, 'full')[j + (n-1)] 
    # Wait, let me verify. np.correlate(a, b, 'full')[i] = sum_j a[j] * b[j + i - (len(b)-1)]
    # Hmm, actually the definition varies. Let me just compute it directly.
    
    # For n up to ~1000, a direct O(n^2) gradient computation is fine (10^6 operations).
    
    # Actually for the purpose of L-BFGS-B, I can just compute the gradient numerically
    # or use a simpler approach: just use the objective without gradient and let L-BFGS-B
    # compute numerical gradients. For n=700, this might be slow but should work.
    
    # Better: let me compute the gradient analytically.
    # grad[j] = dx * sum_k w[k] * d(c[k])/d(h[j])
    # where c[k] = sum_i h[i+m_k]* (1-h[i]), m_k = k-(n-1)
    
    # d(c[k])/d(h[j]) = (1-h[j-m_k])*I(j-m_k in [0,n) and j in [0,n)) 
    #                   + (-h[j+m_k])*I(j in [0,n) and j+m_k in [0,n))
    
    # Note: j is always in [0,n) since we're computing grad[j] for j in [0,n).
    
    # First part: sum_k w[k] * (1-h[j-m_k]) * I(0 <= j-m_k < n)
    # = sum_k w[k] * (1-h[j-m_k]) * I(max(0, -m_k) <= j-m_k < min(n, n-m_k))
    # Hmm, the valid range for i is max(0,-m_k) <= i < min(n, n-m_k), and i = j-m_k.
    # So: I(max(0,-m_k) <= j-m_k < min(n, n-m_k))
    # = I(j-m_k >= 0 and j-m_k < n and j-m_k >= -m_k and j-m_k < n-m_k)
    # = I(j >= m_k and j < n+m_k and j >= 0 and j < n)
    # Since j is already in [0,n):
    # = I(j >= m_k and j < n+m_k)
    
    # So first part = sum_k w[k] * (1-h[j-m_k]) * I(m_k <= j < n+m_k)
    # = sum_{k: m_k <= j < n+m_k} w[k] * (1-h[j-m_k])
    
    # Second part: sum_k w[k] * (-h[j+m_k]) * I(0 <= j+m_k < n)
    # = -sum_{k: -m_k <= j < n-m_k} w[k] * h[j+m_k]
    
    # Let me compute this with numpy:
    grad = np.zeros(n)
    m = np.arange(2*n-1) - (n-1)
    
    # First part: for each j, sum over k where m_k <= j < n+m_k
    # Equivalent: for each k, add w[k]*(1-h[j-m_k]) to grad[j] for j in [m_k, n+m_k-1] ∩ [0,n-1]
    for k in range(2*n-1):
        mk = m[k]
        j_lo = max(0, mk)
        j_hi = min(n, n + mk + 1)  # j < n+mk, so j <= n+mk-1
        if j_lo >= j_hi:
            continue
        t = j_lo - mk  # t = j - m_k, ranges from j_lo-mk to j_hi-1-mk
        t_hi = j_hi - mk
        valid_t = (t >= 0) & (t < n)
        # Hmm this is getting complicated. Let me just do it directly.
        # Actually for n=700, the total work is sum over k of O(n) = O(n^2) = 490000, which is fine.
    
    # Let me use a simpler direct approach:
    # For each j, compute the gradient contribution directly.
    # This is O(n^2) total which is fine for n=700.
    
    # Actually, let me just use the correlation-based approach:
    # grad[j] = dx * [correlate_w_1mh[j] - correlate_w_h[j]]
    # where correlate_w_1mh is the appropriate cross-correlation
    
    # Let me define:
    # A[j] = sum_k w[k] * (1-h[j-m_k]) * I(valid)
    # B[j] = sum_k w[k] * h[j+m_k] * I(valid)
    
    # A is like: for each k, w[k]*(1-h[j-m_k]) for valid j. If we define v = 1-h (length n),
    # then A[j] = sum_k w[k] * v[j-m_k] where j-m_k in [0,n).
    # This is the cross-correlation of v and w: A = correlate(v, w_reversed, 'full') ... 
    # Actually, let me think of it as a convolution.
    
    # A[j] = sum_k w[k] * v[j-m_k] = sum_k w[k] * v[j-(k-(n-1))]
    # = sum_k w[k] * v[j-k+n-1]
    # Let l = k-(n-1) = m_k. Then A[j] = sum_l w[l+(n-1)] * v[j-l]
    # This is (w_shifted * v)[j] where w_shifted[l] = w[l+n-1] (for l in [-(n-1), n-1])
    # which is a convolution (not correlation).
    
    # In numpy: np.convolve(v, w_shifted) gives (v * w_shifted)[j] = sum_i v[i] * w_shifted[j-i]
    # which is sum_i v[i] * w_shifted[j-i]. We want sum_l v[j-l] * w_shifted[l] = sum_l' v[j-l'] * w_shifted[l'] where l' = -l... 
    
    # OK this is getting too complicated. Let me just do the O(n^2) computation directly.
    # For n=700, that's 490000 operations per gradient evaluation, which is fast.
    
    # Actually, let me use a vectorized approach:
    # Create a matrix M where M[j,k] = 1 if the term (1-h[j-m_k]) is valid, else 0
    # Then A = (1-h) @ M.T * w ... no this doesn't work simply.
    
    # Let me just do it the straightforward way:
    # A[j] = sum_{k: 0<=j-m_k<n} w[k] * (1-h[j-m_k])
    # B[j] = sum_{k: 0<=j+m_k<n} w[k] * h[j+m_k]
    
    # Vectorized:
    # For A: create indices. For each k, the valid j range is [max(0,m_k), min(n, n+m_k))
    #   and t = j - m_k ranges from max(0,m_k)-m_k to min(n,n+m_k)-1-m_k = max(-m_k, 0) to min(n-1, n-1-m_k+m_k) ... 
    
    # I'll just loop over k since 2n-1 ~ 1400 iterations is fast:
    one_minus_h = 1.0 - h
    for k in range(2*n-1):
        mk = k - (n-1)
        # A contribution: j in [max(0, mk), min(n, n+mk)), t = j-mk in [0, n)
        j_lo = max(0, mk)
        j_hi = min(n, n + mk)
        if j_lo < j_hi:
            t_lo = j_lo - mk
            t_hi = j_hi - mk
            wk = weights[k]
            grad[j_lo:j_hi] += wk * one_minus_h[t_lo:t_hi]
        # B contribution: j in [max(0, -mk), min(n, n-mk)), t = j+mk in [0, n)
        j_lo2 = max(0, -mk)
        j_hi2 = min(n, n - mk)
        if j_lo2 < j_hi2:
            t_lo2 = j_lo2 + mk
            t_hi2 = j_hi2 + mk
            wk = weights[k]
            grad[j_lo2:j_hi2] -= wk * h[t_lo2:t_hi2]
    
    grad *= dx
    return f, grad

def lbgfb_opt(h, n, alphas, maxiter, deadline):
    """L-BFGS-B optimization with smooth max."""
    best_h, best_c = h.copy(), compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(obj_and_grad, best_h, args=(n, alpha, deadline),
                          method='L-BFGS-B', jac=True, bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-12,
                                   'maxls': 50})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c - 1e-12:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def exact_cd(h, n, max_passes, deadline):
    """Exact 1D coordinate descent."""
    h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            # Compute the quadratic in x for coordinate p
            # c[k] for k where p is involved
            m_min = n - 1 - p
            m_max = 2 * n - 2 - p
            ms = np.arange(m_min, m_max + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(vf):
                a[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                a[vb] -= h[jb[vb]]
            c = np.correlate(h, 1.0 - h, mode='full')
            c_base = c[ms].copy()
            diag_idx = np.where(ms == (n - 1))[0]
            has_diag = len(diag_idx) > 0
            if has_diag:
                c_diag = c_base[diag_idx[0]]
                a[diag_idx] = 0.0
            
            unaff = 0.0
            if m_min > 0:
                unaff = max(unaff, float(np.max(c[:m_min])))
            if m_max < 2 * n - 2:
                unaff = max(unaff, float(np.max(c[m_max + 1:])))
            cur_max = float(np.max(c))
            
            neg_mask = a < -1e-15
            pos_mask = a > 1e-15
            zero_mask = np.abs(a) <= 1e-15
            
            L_lo, L_hi = unaff, cur_max
            for _ in range(45):
                L = (L_lo + L_hi) / 2.0
                x_lo, x_hi = 0.0, 1.0
                ok = True
                if np.any(zero_mask) and np.any(c_base[zero_mask] > L + 1e-15):
                    ok = False
                if ok and np.any(neg_mask):
                    x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
                if ok and np.any(pos_mask):
                    x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
                if x_lo > x_hi + 1e-14:
                    ok = False
                if ok and has_diag:
                    Cq = L - c_diag + h0 - h0 * h0
                    disc = 1.0 - 4.0 * Cq
                    if disc >= 0:
                        r1 = (1.0 - np.sqrt(disc)) / 2.0
                        r2 = (1.0 + np.sqrt(disc)) / 2.0
                        if not (max(x_lo, 0.0) < r1 and min(x_hi, 1.0) > r2):
                            ok = False
                if ok:
                    L_hi = L
                else:
                    L_lo = L
            L = L_hi
            x_lo, x_hi = 0.0, 1.0
            if np.any(neg_mask):
                x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
            if np.any(pos_mask):
                x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
            x_lo = max(x_lo, 0.0)
            x_hi = min(x_hi, 1.0)
            if has_diag:
                Cq = L - c_diag + h0 - h0 * h0
                disc = 1.0 - 4.0 * Cq
                if disc >= 0:
                    r1 = (1.0 - np.sqrt(disc)) / 2.0
                    r2 = (1.0 + np.sqrt(disc)) / 2.0
                    if max(x_lo, 0.0) < r1 and min(x_hi, 1.0) > r2:
                        if h0 - r1 <= r2 - h0:
                            x_hi = min(x_hi, r1)
                        else:
                            x_lo = max(x_lo, r2)
                    elif x_lo >= r1 and x_hi <= r2:
                        if h0 - r1 <= r2 - h0:
                            x_hi = min(x_hi, r1)
                        else:
                            x_lo = max(x_lo, r2)
            if x_lo > x_hi:
                xn = h0
            else:
                xn = float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))
            if abs(xn - h0) > 1e-12:
                h[p] = xn
                improved = True
        h = project_feasible(h, n)
        if not improved:
            break
    return h, compute_c5(h, n)
