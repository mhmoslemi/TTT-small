def compute_c5(h, n):

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:

            tot = h.sum()

def cd_1d_fast(c, h, p, n):

    h0 = h[p]
    if h0 < 1e-12 or h0 > 1.0 - 1e-12:

    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_opt(h, n, alphas, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:

            break

                return z.sum() - n / 2.0

def linearized_cd(h, n, deadline, max_iters=50):
    h = h.copy()
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        c_max = c.max()
        # Active set: shifts where c[k] > c_max - tolerance
        tol = max(1e-4, c_max * 0.01)
        S = np.where(c > c_max - tol)[0]
        if len(S) == 0:
            S = np.arange(len(c))
        # Compute gradient of c[k] for k in S
        # c[k] = sum_j h[j] * (1 - h[j + k - n + 1])
        # dc[k]/dh[i] = (1 - h[i + k - n + 1]) * I(valid) - h[i - k + n - 1] * I(valid)
        grad = np.zeros((len(S), n))
        for idx, k in enumerate(S):
            offset = k - (n - 1)  # the shift
            # Term 1: (1 - h[i + offset]) for valid i + offset
            i_vals = np.arange(n)
            j1 = i_vals + offset
            mask1 = (j1 >= 0) & (j1 < n)
            g1 = np.zeros(n)
            g1[mask1] = 1.0 - h[j1[mask1]]
            # Term 2: -h[i - offset] for valid i - offset
            j2 = i_vals - offset
            mask2 = (j2 >= 0) & (j2 < n)
            g2 = np.zeros(n)
            g2[mask2] = h[j2[mask2]]
            grad[idx] = g1 - g2
        # Linearize: c[k] ≈ c[k]_current + grad[k] @ (h - h_current)
        # = (c[k]_current - grad[k] @ h_current) + grad[k] @ h
        const = c[S] - grad @ h
        # LP: minimize t
        # subject to: grad[k] @ h + const[k] <= t for k in S
        #              sum(h) = n/2
        #              0 <= h <= 1
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [cp.sum(h_var) == n / 2.0,
                      h_var >= 0, h_var <= 1]
        for idx in range(len(S)):
            constraints.append(grad[idx] @ h_var + const[idx] <= t_var)
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iter=10000)
        except Exception:
            try:
                prob.solve(solver=cp.CBC)
            except Exception:
                break
        if h_var.value is None:
            break
        h_new = np.clip(h_var.value, 0, 1)
        h_new = project_feasible(h_new, n)
        # Line search
        c_new = compute_c5(h_new, n)
        if c_new <= c_max + 1e-10:

                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                c[ms] += delta * a
                diag_i = np.where(ms == (n - 1))[0]
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True
        h = project_feasible(h, n)
        if not improved:
            break
    return h

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_round(h, n, alphas, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            def _cf(z):
                return z.sum() - n / 2.0
            def _cj(z):
                return np.ones(n)
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': _cf, 'jac': _cj}],
                           options={'maxiter': maxiter, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c - 1e-15:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def linearized_update(h, n, tol_frac, deadline):
    """Solve LP with linearized active constraints to get a new h."""
    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(c.max())
    tol = max(1e-6, c_max * tol_frac)
    S = np.where(c > c_max - tol)[0]
    if len(S) == 0:
        S = np.argmax(c)
        S = np.array([S])
    if len(S) > 300:
        S = np.argsort(c)[-300:]
    S = np.sort(S)
    grad = np.zeros((len(S), n))
    for idx, k in enumerate(S):
        offset = int(k) - (n - 1)
        i_vals = np.arange(n)
        j1 = i_vals + offset
        mask1 = (j1 >= 0) & (j1 < n)
        g1 = np.zeros(n)
        g1[mask1] = 1.0 - h[j1[mask1]]
        j2 = i_vals - offset
        mask2 = (j2 >= 0) & (j2 < n)
        g2 = np.zeros(n)
        g2[mask2] = h[j2[mask2]]
        grad[idx] = g1 - g2
    const = c[S].astype(float) - grad @ h
    try:
        import cvxpy as cp
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1]
        for idx in range(len(S)):
            constraints.append(grad[idx] @ h_var + const[idx] <= t_var)
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        prob.solve(solver=cp.ECOS, max_iter=20000)
        if h_var.value is None:
            return None
        h_new = np.clip(h_var.value, 0.0, 1.0)
        h_new = project_feasible(h_new, n)
        return h_new
    except Exception:
        return None

def run_lp_cd(h, n, max_iters, deadline, tol_fracs):
    h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        improved = False
        for tf in tol_fracs:
            if time.time() > deadline:
                break
            h_lp = linearized_update(h, n, tf, deadline)
            if h_lp is None:
                continue
            c_new = compute_c5(h_lp, n)
            if c_new < best_c - 1e-12:
                h = h_lp
                best_c = c_new
                improved = True
            else:
                for alpha in [0.5, 0.25, 0.1, 0.05]:
                    h_mix = (1.0 - alpha) * h + alpha * h_lp
                    h_mix = project_feasible(h_mix, n)
                    c_mix = compute_c5(h_mix, n)
                    if c_mix < best_c - 1e-12:
                        h = h_mix
                        best_c = c_mix
                        improved = True
                        break
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = np.full(400, 0.5)
        n = 400
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)
    if time.time() < deadline - 150:
        h1, c1 = slsqp_round(h_best, n, [2000, 8000, 30000], 200, deadline)
        if c1 < best_c:
            h_best, best_c = h1, c1
    h2 = run_cd(h_best, n, 300, deadline)
    c2 = compute_c5(h2, n)
    if c2 < best_c:
        h_best, best_c = h2, c2
    n = len(h_best)
    if time.time() < deadline - 200:
        h3, c3 = run_lp_cd(h_best, n, 40, deadline, [0.02, 0.05, 0.1])
        if c3 < best_c:
            h_best, best_c = h3, c3
            h4 = run_cd(h_best, n, 100, deadline)
            c4 = compute_c5(h4, n)
            if c4 < best_c:
                h_best, best_c = h4, c4
    n_list = [500, 550, 600, 650, 750, 800, 850, 900, 1000, 1200]
    for n_try in n_list:
        if time.time() > deadline - 120:
            break
        if n_try == len(h_best):
            continue
        x_old = np.linspace(0, 2, len(h_best))
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_old, h_best), n_try)
        h_o = run_cd(h_int, n_try, 150, deadline)
        c_o = compute_c5(h_o, n_try)
        if c_o < best_c - 1e-8:
            h_best, best_c, n = h_o, c_o, n_try
            h_o2, c_o2 = run_lp_cd(h_o, n_try, 20, deadline, [0.03, 0.08])
            if c_o2 < best_c:
                h_best, best_c = h_o2, c_o2
    n = len(h_best)
    if time.time() < deadline - 150:
        h5, c5 = slsqp_round(h_best, n, [30000, 100000, 300000, 800000], 300, deadline)
        if c5 < best_c:
            h_best, best_c = h5, c5
    if time.time() < deadline - 100:
        h6, c6 = run_lp_cd(h_best, n, 50, deadline, [0.01, 0.03, 0.08, 0.15])
        if c6 < best_c:
            h_best, best_c = h6, c6
    n = len(h_best)
    for r in range(12):
        if time.time() > deadline - 40:
            break
        bs = rng.randint(5, max(6, n // 15))
        st = rng.randint(0, max(1, n - bs))
        hp = h_best.copy()
        hp[st:st + bs] = np.clip(hp[st:st + bs] + rng.uniform(-0.15, 0.15), 0, 1)
        hp = project_feasible(hp, n)
        h_o = run_cd(hp, n, 150, deadline)
        c_o = compute_c5(h_o, n)
        if c_o < best_c:
            h_best, best_c = h_o, c_o
        hp2 = project_feasible(h_best + rng.normal(0, 0.01 + 0.01 * r, n), n)
        h_o2 = run_cd(hp2, n, 80, deadline)
        c_o2 = compute_c5(h_o2, n)
        if c_o2 < best_c:
            h_best, best_c = h_o2, c_o2
    if time.time() < deadline - 10:
        h_f = run_cd(h_best, len(h_best), 200, deadline)
        c_f = compute_c5(h_f, len(h_best))
        if c_f < best_c:
            h_best, best_c = h_f, c_f
        if time.time() < deadline - 5:
            h_lp2, c_lp2 = run_lp_cd(h_best, len(h_best), 20, deadline, [0.02, 0.06])
            if c_lp2 < best_c:
                h_best, best_c = h_lp2, c_lp2
    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
