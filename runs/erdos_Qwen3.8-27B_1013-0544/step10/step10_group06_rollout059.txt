    """Coordinate descent with h[i] + h[n-1-i] = 1 enforced."""
    h = h.copy()
    # First, symmetrize
    m = n // 2
    for i in range(m):
        h[n-1-i] = 1.0 - h[i]
    # Now optimize only h[0..m-1]
    for _pass in range(max_passes):
        if time.time() > deadline:

        c = np.correlate(h, 1.0 - h, mode='full')

def lse_obj_and_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    w = np.exp(s - mx)

            if kf == 0:
                grad[i] += w[k] * (1.0 - 2.0 * x[i]) * dx
    return lse_val, grad
