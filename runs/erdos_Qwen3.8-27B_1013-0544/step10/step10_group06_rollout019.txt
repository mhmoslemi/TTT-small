import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def coord_descent(h, n, max_passes, deadline, rng=None, random_order=True, nbisect=50):
    h = h.copy()
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        if random_order and rng is not None:
            order = rng.permutation(n)
        else:
            order = np.arange(n)
        for p in order:
            if time.time() > deadline:
                break
            x_new = cd_solve_coord(c, h, p, n, nbisect)
            if abs(x_new - h[p]) > 1e-13:
                h[p] = x_new
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def lse_smoothedgrad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    mc = c.max()
    exp_c = np.exp(alpha * (c - mc))
    S = exp_c.sum()
    w = exp_c / S
    dx = 2.0 / n
    obj = dx * (mc + np.log(S) / alpha)
    grad = np.zeros(n)
    for s in range(-(n - 1), n):
        m = s + n - 1
        w_m = w[m]
        if w_m < 1e-12:
            continue
        if s >= 0:
            if s < n:
                grad[s:] += w_m * (1.0 - x[:n - s])
                grad[:n - s] -= w_m * x[s:]
        else:
            d = -s
            if d < n:
                grad[:n - d] += w_m * (1.0 - x[d:])
                grad[d:] -= w_m * x[:n - d]
    return obj, dx * grad

def lse_step(h, n, alpha, lr, deadline):
    for _ in range(200):
        if time.time() > deadline:
            break
        obj, grad = lse_smoothedgrad(h, n, alpha)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        obj_new = compute_c5(h_new, n)
        if obj_new < compute_c5(h, n) + 1e-15:
            h = h_new
        else:
            lr *= 0.5
            if lr < 1e-10:
                break
    return h

def symmetrize(h, n):
    h_sym = 0.5 * (h + 1.0 - h[::-1].copy())
    return project_feasible(h_sym, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 700
        x = np.linspace(0, 2, n)
        h_init = 0.5 + 0.15 * np.sin(np.pi * x) + 0.05 * np.sin(3 * np.pi * x)
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    # Phase 1: Quick CD
    h1, c1 = coord_descent(best_h, n, 30, deadline, rng, True, 50)
    if c1 < best_c:
        best_h, best_c = h1, c1
    # Phase 2: Gradient-based diversification
    for alpha in [2000, 8000, 20000]:
        if time.time() > deadline - 60:
            break
        h_g = lse_step(best_h.copy(), n, alpha, 0.02, deadline)
        h_g, c_g = coord_descent(h_g, n, 40, deadline, rng, True, 50)
        if c_g < best_c:
            best_h, best_c = h_g, c_g
    # Phase 3: Deep CD
    h3, c3 = coord_descent(best_h, n, 100, deadline, rng, False, 50)
    if c3 < best_c:
        best_h, best_c = h3, c3
    # Phase 4: Symmetrize + CD
    h_s = symmetrize(best_h, n)
    c_s = compute_c5(h_s, n)
    if c_s < best_c:
        best_h, best_c = h_s, c_s
        h_s2, c_s2 = coord_descent(h_s, n, 60, deadline, rng, False, 50)
        if c_s2 < best_c:
            best_h, best_c = h_s2, c_s2
    # Phase 5: Try other n values
    for n_try in [500, 600, 800, 900]:
        if time.time() > deadline - 120:
            break
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_old, best_h), n_try)
        h_t, c_t = coord_descent(h_int, n_try, 40, deadline, rng, True, 50)
        for alpha in [5000, 20000]:
            if time.time() > deadline - 60:
                break
            h_t = lse_step(h_t, n_try, alpha, 0.02, deadline)
            h_t, c_t = coord_descent(h_t, n_try, 30, deadline, rng, True, 50)
        h_s2 = symmetrize(h_t, n_try)
        c_s2 = compute_c5(h_s2, n_try)
        if c_s2 < c_t:
            h_t, c_t = h_s2, c_s2
            h_t, c_t = coord_descent(h_t, n_try, 30, deadline, rng, False, 50)
        if c_t < best_c:
            best_h, best_c = h_t, c_t
    # Phase 6: Final escalation
    if time.time() < deadline - 30:
        n_cur = len(best_h)
        remaining = deadline - time.time() - 5
        max_p = max(30, int(remaining / max(0.1, remaining / 60)))
        h_f, c_f = coord_descent(best_h, n_cur, max_p, deadline, rng, False, 55)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
