def feasible_1d(A,B,C,T):

          hi = min(hi, float(np.min(u)))
       neg = blin < -1e-14
       if np.any(neg):

       ac=A[conv]; bc=B[conv]; cc=C[conv]
       disc = bc*bc - 4*ac*(cc - T)
       if np.any(disc < -1e-12): return False
       r = np.sqrt(np.maximum(disc,0.0))
       r1 = (-bc - r)/(2*ac); r2=(-bc + r)/(2*ac)
       # sort (for a positive, r1<=r2)

          disc=b*b -4*a*(cc-T)
          if disc < -1e-12:
             continue

def project_feasible(h, n):

def coord_coeff(c, h, x0, p, n, anti, d):
    L = 2 * n - 1
    A = np.zeros(L, dtype=float)
    B = np.zeros(L, dtype=float)

    j1 = p + d
    v1 = (j1 >= 0) & (j1 < n)

        mqq = v2 & (j2 == q)
        if np.any(mqq):
            A[mqq] += -1.0
            B[mqq] += 1.0
        lin2 = v2 & ~((j2 == p) | (j2 == q))
        if np.any(lin2):

        mask3 = v3 & (i1 != p) & (i1 != q)

        i2 = q - d
        v4 = (i2 >= 0) & (i2 < n)

        lin1 = v1 & (j1 != p)

    C = c - A * x0 * x0 - B * x0

def coord_descent(h_in, n, anti, passes, deadline, rng, recompute, nbisect, random_order=True):

        m = n // 2
        x = np.asarray(h_in[:m], dtype=float).copy()
        x = np.clip(x, 0.0, 1.0)

        x = np.clip(np.asarray(h_in, dtype=float).copy(), 0.0, 1.0)
        h = x

    c = np.correlate(h, 1.0 - h, mode="full")

        if time.time() > deadline:
            break

                cnt += 1
                if recompute <= 1 or (cnt % recompute == 0):
                    c = np.correlate(h, 1.0 - h, mode="full")

        if not improved:
            break

        if not anti:
            h = project_feasible(x, n)
            x = h
            c = np.correlate(h, 1.0 - h, mode="full")

        cur_c = compute_c5(h, n)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()

    if anti:
        c = np.correlate(h, 1.0 - h, mode="full")
        cur_c = float(np.max(c) * (2.0 / n))
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()

    return best_h, best_c


def lse_anti_value(x, n, alpha):
    h = anti_full(np.asarray(x, dtype=float), n)
    c = np.correlate(h, 1.0 - h, mode="full")
    m0 = float(np.max(c))
    sh = alpha * (c - m0)
    sw = float(np.sum(np.exp(sh)))
    if sw <= 0.0:
        sw = 1.0
    return (2.0 / float(n)) * (m0 + math.log(sw) / float(alpha))


def lse_anti_grad(x, n, alpha):
    h = anti_full(np.asarray(x, dtype=float), n)
    c = np.correlate(h, 1.0 - h, mode="full")
    m0 = float(np.max(c))
    sh = alpha * (c - m0)
    w = np.exp(sh)
    sw = float(w.sum())
    if sw <= 0.0:
        return np.zeros(n // 2)
    w /= sw

    gh = np.zeros(n, dtype=float)
    nz = np.where(w > 1e-13)[0]
    for ell in nz:
        ww = float(w[ell])
        dd = int(ell - (n - 1))
        if dd >= 0:
            end = n - dd
            if end > 0:
                gh[:end] += ww * (1.0 - h[dd:])
                gh[dd:] -= ww * h[:end]
        else:
            a = -dd
            end = n - a
            if end > 0:
                gh[a:] += ww * (1.0 - h[:end])
                gh[:end] -= ww * h[a:]

    m = n // 2
    gx = gh[:m] - gh[m:][::-1]
    return gx * (2.0 / float(n))


def lbfgs_polish(x, n, alpha, maxiter):
    m = n // 2
    bounds = [(0.0, 1.0)] * m
    try:
        res = minimize(
            lse_anti_value,
            np.asarray(x, dtype=float).copy(),
            args=(n, float(alpha)),
            method="L-BFGS-B",
            jac=lse_anti_grad,
            bounds=bounds,
            options={"maxiter": int(maxiter), "ftol": 1e-15, "disp": False, "maxls": 30},
        )
        out = res.x if hasattr(res, "x") else x
        return np.clip(np.asarray(out, dtype=float), 0.0, 1.0)
    except Exception:
        return np.clip(np.asarray(x, dtype=float), 0.0, 1.0)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget_s = float(budget_s)
    deadline = t0 + max(10.0, budget_s) - 4.0
    rng = np.random.RandomState(int(seed))

    try:
        h0 = np.asarray(initial_h_values, dtype=float).copy()
        n0 = int(len(h0))
    except Exception:
        h0 = None
        n0 = 0

    if h0 is None or n0 < 50:
        n0 = 700
        h0 = default_h(n0, rng)

    if n0 % 2 != 0:
        n0 += 1
        h0 = interpolate_h(h0, n0 - 1, n0)

    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    best_n = n0

    if deadline - time.time() > 120.0:
        href, cref = coord_descent(h0, best_n, False, 5, deadline - 180.0, rng, 3, 28, True)
        if cref < best_c:
            best_h, best_c = href, cref

    if deadline - time.time() > 260.0:
        xa = anti_project(best_h, best_n)
        ha, ca = coord_descent(xa, best_n, True, 35, deadline - 160.0, rng, 1, 30, True)
        if ca < best_c:
            best_h, best_c, best_n = ha, ca, len(ha)

    if deadline - time.time() > 260.0:
        xa = anti_project(best_h, best_n)
        for alpha in [500.0, 2000.0]:
            if deadline - time.time() < 110.0:
                break
            xa = lbfgs_polish(xa, best_n, alpha, 45)
            if deadline - time.time() < 70.0:
                continue
            hx, cx = coord_descent(xa, best_n, True, 6, deadline - 50.0, rng, 1, 30, False)
            if cx < best_c:
                best_h, best_c = hx, cx
            xa = anti_project(best_h, best_n)

    if deadline - time.time() > 140.0:
        hf = project_feasible(best_h.copy(), best_n)
        hf, cf = coord_descent(hf, best_n, False, 10, deadline - 80.0, rng, 3, 28, True)
        if cf < best_c:
            best_h, best_c = hf, cf

    base_n = best_n
    opts = []
    for v in [base_n, base_n - 140, 560, 420, 640, base_n + 140, 840, 500, 360]:
        if 120 <= v < 1000 and v % 2 == 0 and v not in opts:
            opts.append(v)
    opts = opts[:5]

    for ntry in opts:
        if deadline - time.time() < 110.0:
            break
        hi = interpolate_h(best_h, best_n, ntry)
        xi = anti_project(hi, ntry)
        passes = 20 if ntry <= 650 else 10
        hx, cx = coord_descent(xi, ntry, True, passes, deadline - 70.0, rng, 1, 30, True)
        if cx < best_c:
            best_h, best_c, best_n = hx, cx, ntry
            if deadline - time.time() > 70.0:
                hf = project_feasible(best_h.copy(), best_n)
                hf, cf = coord_descent(hf, best_n, False, 4, deadline - 35.0, rng, 3, 28, False)
                if cf < best_c:
                    best_h, best_c = hf, cf

    if deadline - time.time() > 240.0:
        for sigma in [0.06, 0.03]:
            if deadline - time.time() < 130.0:
                break
            xa = anti_project(best_h, best_n)
            xb = np.clip(xa + sigma * rng.normal(size=best_n // 2), 0.0, 1.0)
            hx, cx = coord_descent(xb, best_n, True, 18, deadline - 80.0, rng, 1, 30, True)
            if cx < best_c:
                best_h, best_c = hx, cx

    rem = deadline - time.time()
    if rem > 60.0:
        xa = anti_project(best_h, best_n)
        passes = int(min(60, max(5, rem / 2.0)))
        hx, cx = coord_descent(xa, best_n, True, passes, deadline - 5.0, rng, 1, 30, False)
        if cx < best_c:
            best_h, best_c = hx, cx

    rem = deadline - time.time()
    if rem > 30.0:
        hf = project_feasible(best_h.copy(), best_n)
        passes = int(min(25, max(3, rem / 2.0)))
        hf, cf = coord_descent(hf, best_n, False, passes, deadline - 3.0, rng, 3, 28, False)
        if cf < best_c:
            best_h, best_c = hf, cf

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, int(best_n)
