        x = 0.5 * (x[:half] + x[half:])
        x = np.concatenate([x, x])
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_opt(h, n, alphas, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq',
                 'fun': lambda x, n_=n: x.sum() - n_ / 2.0,
                 'jac': lambda x, n_=n: np.ones(n_)}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def fine_tune(h, n, deadline, rng):
    """Try very small perturbations around current best."""
    h_best, c_best = h.copy(), compute_c5(h, n)
    step_sizes = [0.001, 0.0005, 0.0002, 0.0001]
    for ss in step_sizes:
        if time.time() > deadline:
            break
        for _ in range(50):
            if time.time() > deadline:
                break
            p = rng.randint(n)
            h_try = h_best.copy()
            h_try[p] = np.clip(h_best[p] + rng.uniform(-ss, ss), 0, 1)
            h_try = project_feasible(h_try, n)
            c_try = compute_c5(h_try, n)
            if c_try < c_best - 1e-15:
                h_best, c_best = h_try, c_try
    return h_best, c_best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        n = 700
        h_best = np.full(n, 0.5)
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)

    # Phase 1: Thorough CD from initial
    h1, c1 = exact_cd(h_best, n, 500, deadline)
    if c1 < best_c:
        h_best, best_c = h1, c1

    # Phase 2: Periodic constraint CD
    if time.time() < deadline - 300:
        hp = enforce_periodic(h_best.copy(), n)
        hp = project_feasible(hp, n)
        hp, cp = exact_cd(hp, n, 300, deadline, periodic=True)
        if cp < best_c:
            h_best, best_c = hp, cp

    # Phase 3: SLSQP refinement with multiple alphas
    if time.time() < deadline - 200:
        h3, c3 = slsqp_opt(h_best, n, [3000, 10000, 30000, 100000, 300000], 400, deadline - 100)
        if c3 < best_c:
            h_best, best_c = h3, c3
        h_best, best_c = exact_cd(h_best, n, 300, deadline)

    # Phase 4: Structured restarts
    restarts = []
    # Complement
    restarts.append(1.0 - h_best)
    # Periodic average
    hp2 = enforce_periodic(h_best.copy(), n)
    hp2 = project_feasible(hp2, n)
    restarts.append(hp2)
    # Small shifts
    for s in [1, 2, -1, -2]:
        hs = np.roll(h_best, s)
        hs = project_feasible(hs, n)
        restarts.append(hs)
    # Noise at various scales
    for sig in [0.01, 0.03, 0.05]:
        hn = project_feasible(h_best + rng.normal(0, sig, n), n)
        restarts.append(hn)

    for hr in restarts:
        if time.time() > deadline - 80:
            break
        hr = project_feasible(hr, n)
        ho, co = exact_cd(hr, n, 200, deadline - 40)
        if co < best_c:
            h_best, best_c = ho, co

    # Phase 5: Fine-tune around best
    if time.time() < deadline - 30:
        h_best, best_c = fine_tune(h_best, n, deadline - 10, rng)

    # Phase 6: Final CD polish
    if time.time() < deadline - 3:
        hf, cf = exact_cd(h_best, n, 500, deadline)
        if cf < best_c:
            h_best, best_c = hf, cf

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
