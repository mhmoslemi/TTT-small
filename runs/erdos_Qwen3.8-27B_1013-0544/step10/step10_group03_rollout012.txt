       one_minus = 1.0 - h
       block = int(block) # constant
       count = 0
       for p in perm:
          if time.time() >= deadline: break
          y = h[p]
          x, a = solve_1d(c, h, p, n, one_minus, iters)
          delta = x - y
          if abs(delta) > 1e-13:

def project_feasible(h, n):
    h = np.clip(np.array(h, float, copy=True), 0.0, 1.0)
    t = 0.5 * n
    for _ in range(80):
        d = t - h.sum()
        if abs(d) < 1e-13: return h
        if d > 0:
            r = 1.0 - h; s = r.sum()
            if s <= 1e-15: break
            h = np.clip(h + d / s * r, 0.0, 1.0)
        else:
            s = h.sum()
            if s <= 1e-15: break

        else:

import time
import numpy as np

def project_feasible(h, n):
    h = np.clip(np.array(h, dtype=float, copy=True), 0.0, 1.0)
    t = 0.5 * n
    for _ in range(80):
        d = t - h.sum()
        if abs(d) < 1e-13:
            return h
        if d > 0.0:
            r = 1.0 - h
            s = r.sum()
            if s <= 1e-15:
                break
            h = np.clip(h + (d / s) * r, 0.0, 1.0)
        else:
            s = h.sum()
            if s <= 1e-15:
                break
            h = np.clip(h + (d / s) * h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-9:
        if d > 0.0:
            for i in range(n):
                if d <= 0.0:
                    break
                a = min(d, 1.0 - h[i])
                h[i] += a
                d -= a
        else:
            d = -d
            for i in range(n):
                if d <= 0.0:
                    break
                a = min(d, h[i])
                h[i] -= a
                d -= a
    return h

def c5_of(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def addc(lst, x):
    if x < 0.0:
        x = 0.0
    elif x > 1.0:
        x = 1.0
    if not np.isfinite(x):
        return
    for v in lst:
        if abs(v - x) < 1e-12:
            return
    lst.append(float(x))

def build_coeff(h, om, p, n):
    a = om.copy()
    t0 = 2 * p - n + 1
    if t0 < 0:
        t0 = 0
    t1 = 2 * p
    if t1 > n - 1:
        t1 = n - 1
    if t1 >= t0:
        lo = 2 * p - t1
        hi = 2 * p - t0
        a[t0:t1 + 1] -= h[lo:hi + 1][::-1]
    a[p] = 0.0
    return a

def solve_1d(c, h, p, n, om, iters, drift):
    st = n - 1 - p
    cs = c[st:st + n]
    a = build_coeff(h, om, p, n)
    y = float(h[p])
    eps = 1e-14

    neg = a < -eps
    pos = a > eps
    zero = ~neg & ~pos
    zero[p] = False
    hn = bool(np.any(neg))
    hp = bool(np.any(pos))
    hz = bool(np.any(zero))

    if hn:
        an = a[neg]
        cn = cs[neg]
        ns = 1.0 / an
        nc = y - cn * ns
    if hp:
        an = a[pos]
        cn = cs[pos]
        ps = 1.0 / an
        pc = y - cn * ps
    if hz:
        zm = float(cs[zero].max())

    un = 0.0
    if st > 0:
        un = max(un, float(c[:st].max()))
    en = st + n
    if en < c.size:
        un = max(un, float(c[en:].max()))

    Ll = un
    Lh = float(c.max())
    if Lh < Ll:
        Lh = Ll

    for _ in range(iters):
        L = 0.5 * (Ll + Lh)
        ok = True
        if hz and zm > L + 1e-12:
            ok = False
        xl = 0.0
        xh = 1.0
        if ok and hn:
            v = float(np.max(ns * L + nc))
            if v > xl:
                xl = v
        if ok and hp:
            v = float(np.min(ps * L + pc))
            if v < xh:
                xh = v
        if xl > xh + 1e-12:
            ok = False
        if ok:
            D = L - cs[p] + y * (1.0 - y)
            if D < -1e-12:
                ok = False
            elif D < 0.25:
                if D < 0.0:
                    D = 0.0
                r = float(np.sqrt(max(0.0, 1.0 - 4.0 * D)))
                r1 = 0.5 * (1.0 - r)
                r2 = 0.5 * (1.0 + r)
                if xl > r1 + 1e-12 and xh < r2 - 1e-12:
                    ok = False
        if ok:
            Lh = L
        else:
            Ll = L

    L = Lh
    xl = 0.0
    xh = 1.0
    if hn:
        xl = max(xl, float(np.max(ns * L + nc)))
    if hp:
        xh = min(xh, float(np.min(ps * L + pc)))
    if xl > xh + 1e-10:
        return y, a

    xl = max(0.0, xl)
    xh = min(1.0, xh)
    ints = []
    D = L - cs[p] + y * (1.0 - y)
    if D < -1e-12:
        return y, a
    if D < 0.25:
        if D < 0.0:
            D = 0.0
        r = float(np.sqrt(max(0.0, 1.0 - 4.0 * D)))
        r1 = 0.5 * (1.0 - r)
        r2 = 0.5 * (1.0 + r)
        rA = min(xh, r1)
        if rA >= xl - 1e-12:
            ints.append((xl, rA))
        lB = max(xl, r2)
        if xh >= lB - 1e-12:
            ints.append((lB, xh))
    else:
        ints.append((xl, xh))

    cands = []
    for l, r in ints:
        addc(cands, l)
        addc(cands, r)
        addc(cands, 0.5 * (l + r))
        addc(cands, min(max(y, l), r))

    bx = y
    bk = None
    for x in cands:
        d = x - y
        seg = cs + d * a
        seg[p] += d * (1.0 - 2.0 * y) - d * d
        mx = float(seg.max())
        if mx < un:
            mx = un
        if mx <= L + 1e-7:
            s1 = abs(d)
            if drift > 1e-10 and d < 0.0:
                s1 += 0.5 * (-d)
            elif drift < -1e-10 and d > 0.0:
                s1 += 0.5 * d
            key = (mx, s1)
            if bk is None or key < bk:
                bk = key
                bx = x
    return bx, a

def cd_polish(h, n, deadline, iters=45, block=32, seed=1, order="random", max_perturb=3):
    h = project_feasible(h, n)
    om = 1.0 - h
    cc = np.correlate(h, om, mode="full")
    bh = h.copy()
    bc = c5_of(h, n)
    rs = np.random.RandomState(seed)
    target = 0.5 * n
    sh = float(h.sum())
    noimp = 0
    pert = 0

    while time.time() < deadline:
        if order == "random":
            perm = rs.permutation(n)
        else:
            perm = np.arange(n)
        om = 1.0 - h
        changed = 0
        for p in perm:
            if time.time() >= deadline:
                break
            y = h[p]
            x, a = solve_1d(cc, h, p, n, om, iters, target - sh)
            d = x - y
            if abs(d) > 1e-13:
                st = n - 1 - p
                cc[st:st + n] += d * a
                cc[n - 1] += d * (1.0 - 2.0 * y) - d * d
                h[p] = x
                om[p] = 1.0 - x
                sh += d
                changed += 1
                if changed >= block and time.time() + 0.25 < deadline:
                    h = project_feasible(h, n)
                    om = 1.0 - h
                    cc = np.correlate(h, om, mode="full")
                    sh = float(h.sum())
                    changed = 0

        h = project_feasible(h, n)
        om = 1.0 - h
        cc = np.correlate(h, om, mode="full")
        sh = float(h.sum())
        cur = c5_of(h, n)

        if cur < bc - 1e-12:
            bc = cur
            bh = h.copy()
            noimp = 0
        else:
            noimp += 1
            if noimp >= 4:
                if pert < max_perturb and time.time() < deadline - 15:
                    sig = 0.01 * (pert + 1)
                    h = bh + rs.normal(0.0, sig, n)
                    h = project_feasible(h, n)
                    om = 1.0 - h
                    cc = np.correlate(h, om, mode="full")
                    sh = float(h.sum())
                    pert += 1
                    noimp = 0
                else:
                    break
    return bh, bc

def resample_h(h, no, nn):
    xo = (np.arange(no) + 0.5) * (2.0 / no)
    xn = (np.arange(nn) + 0.5) * (2.0 / nn)
    return project_feasible(np.interp(xn, xo, h), nn)

def fourier_start(n, seed, modes, scale):
    rs = np.random.RandomState(seed)
    x = (np.arange(n) + 0.5) * (2.0 / n)
    v = np.full(n, 0.5)
    for f in range(1, modes + 1):
        amp = scale / (f ** 0.8)
        if amp < 1e-4:
            break
        v += amp * np.cos(np.pi * f * x + rs.uniform(0.0, 2.0 * np.pi))
        v += 0.7 * amp * np.sin(np.pi * f * x + rs.uniform(0.0, 2.0 * np.pi))
    v += 0.03 * rs.rand(n)
    return project_feasible(v, n)

def make_starts(h, n, rng, seed):
    L = []
    tb = np.zeros(n)
    tb[np.argsort(-h)[:max(1, n // 2)]] = 1.0
    tb = project_feasible(tb, n)
    L.append((tb, n))
    L.append((project_feasible(0.75 * tb + 0.25 * rng.rand(n), n), n))
    L.append((project_feasible(h + 0.12 * rng.normal(0.0, 1.0, n), n), n))
    L.append((fourier_start(n, seed + 7, 10, 0.45), n))
    for off in (-100, 100, 200):
        nn = n + off
        if nn % 2:
            nn += 1
        if 320 <= nn <= 980 and nn != n:
            L.append((resample_h(h, n, nn), nn))
    return L

    t0 = time.time()

    n_best = len(best_h)
    best_h = project_feasible(best_h, n_best)
    best_c = c5_of(best_h, n_best)
    return best_h, best_c, n_best
