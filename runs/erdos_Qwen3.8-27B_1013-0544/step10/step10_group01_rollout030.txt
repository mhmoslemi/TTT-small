import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def exact_1d_min(c, a, d_lo, d_hi):
    """Minimize max_m(c[m] + a[m]*d) for d in [d_lo, d_hi]."""
    if d_hi - d_lo < 1e-15:
        return d_lo
    lo, hi = d_lo, d_hi
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        vals = c + a * mid
        mx = vals.max()
        # Find active set (within tolerance)
        tol = 1e-10 * (1.0 + abs(mx))
        active = vals >= mx - tol
        if active.sum() == 0:
            active = np.zeros_like(c, dtype=bool)
            active[np.argmax(vals)] = True
        a_act = a[active]
        if a_act.min() <= 0 and a_act.max() >= 0:
            return mid
        elif a_act.min() > 0:
            hi = mid
        else:
            lo = mid
    return 0.5 * (lo + hi)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def exact_1d_min(c, a, d_lo, d_hi):
    if d_hi - d_lo < 1e-15:
        return d_lo
    lo, hi = d_lo, d_hi
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        vals = c + a * mid
        mx = vals.max()
        tol = 1e-10 * (1.0 + abs(mx))
        active = vals >= mx - tol
        if not active.any():
            active = np.zeros(len(c), dtype=bool)
            active[np.argmax(vals)] = True
        a_act = a[active]
        if a_act.min() <= 0 and a_act.max() >= 0:
            return mid
        elif a_act.min() > 0:
            hi = mid
        else:
            lo = mid
    return 0.5 * (lo + hi)

def compute_coeffs(p, h, n):
    a = np.zeros(2 * n - 1)
    s1_lo = max(-(n - 1), p - n + 1)
    s1_hi = min(n - 1, p)
    if s1_lo <= s1_hi:
        ks = np.arange(s1_lo, s1_hi + 1)
        a[ks + (n - 1)] += (1.0 - h[p - ks])
    s2_lo = max(-(n - 1), -p)
    s2_hi = min(n - 1, n - 1 - p)
    if s2_lo <= s2_hi:
        ks = np.arange(s2_lo, s2_hi + 1)
        a[ks + (n - 1)] -= h[p + ks]
    return a

def coord_descent(h, n, max_passes, deadline, random_order=False, rng=None):
    h = h.copy()
    best_c = compute_c5(h, n)
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        order = list(range(n))
        if random_order and rng is not None:
            rng.shuffle(order)
        for p in order:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-10 or h0 >= 1.0 - 1e-10:
                continue
            a = compute_coeffs(p, h, n)
            d_opt = exact_1d_min(c, a, -h0, 1.0 - h0)
            if abs(d_opt) > 1e-13:
                h[p] = h0 + d_opt
                improved = True
        if improved:
            mask_lo = h < 1e-8
            mask_hi = h > 1.0 - 1e-8
            if mask_lo.any():
                h[mask_lo] = 0.0
            if mask_hi.any():
                h[mask_hi] = 1.0
            h = project_feasible(h, n)
            c_new = compute_c5(h, n)
            if c_new < best_c:
                best_c = c_new
        else:
            break
    final_c = compute_c5(h, n)
    return h, min(best_c, final_c)

def slsqp_refine(h, n, maxiter=300):
    def smooth_obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        alpha = 10000.0
        shifted = alpha * (c - m)
        mx = shifted.max()
        lse = mx + np.log(np.sum(np.exp(shifted - mx)))
        return (m + lse / alpha) * (2.0 / n)

    def sum_eq(x):
        return x.sum() - n / 2.0

    def sum_eq_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac}]
    try:
        res = minimize(smooth_obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        if res.x is not None:
            new_h = project_feasible(res.x, n)
            return new_h, compute_c5(new_h, n)
    except Exception:
        pass
    return h, compute_c5(h, n)
