            tot = room.sum()

def exact_cd(h, n, max_passes, deadline, order=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    if order is None:
        orders = list(range(n))
    else:
        orders = [order]
    
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in orders:
            if time.time() > deadline:
                return h, compute_c5(h, n)
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            xn = solve_1d_bisect(c, h, p, n)
            if abs(xn - h0) > 1e-12:
                delta = xn - h0
                ms = np.arange(n - 1 - p, 2 * n - 1 - p)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                diag_i = np.where(ms == (n - 1))[0]
                c[ms] += delta * a
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-12:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

        #      0 <= h[i] <= 1
        #      sum h[i] = n/2
        #
        # where a[k,i] = (1-h[j_i(k)]) - h[i_k(j)] ...

        #
        # c[k] = sum_{i valid} h[i] * (1 - h[i + k - (n-1)])
        # Linearized at h0:
        # c[k] ≈ sum_{i valid} [h0[i]*h0[j] + (1-h0[j])*h[i] - h0[i]*h[j]]

        # where const[k] = sum_{i valid} h0[i]*h0[i+k-(n-1)]
        # and coeff[k,i] = (1-h0[i+k-(n-1)]) [from second term, when i is the first index]
        #                  - h0[i-(k-(n-1))] [from third term, when i is the second index]
        #                  (with appropriate bounds checking)
        
        A_rows = []
        b_vals = []
        
        # For each k in 0..2n-2:
        for k in range(2 * n - 1):
            # Valid i range: max(0, n-1-k) <= i <= min(n-1, n-1+k)
            i_min = max(0, n - 1 - k)
            i_max = min(n - 1, n - 1 + k)
            
            if i_min > i_max:
                continue
                
            j_offset = k - (n - 1)  # j = i + j_offset
            
            # const = sum h0[i] * h0[i + j_offset]
            const = 0.0
            coeff = np.zeros(n)
            
            for i in range(i_min, i_max + 1):
                j = i + j_offset
                const += h[i] * h[j]
                coeff[i] += (1.0 - h[j])  # from (1-h[j])*h[i]
                coeff[j] -= h[i]           # from -h[i]*h[j]
            
            # Constraint: const + sum coeff[i]*h[i] <= t
            # => sum coeff[i]*h[i] - t <= -const
            # In standard form for linprog: A @ [h, t] <= b
            row = np.zeros(n + 1)
            row[:n] = coeff
            row[n] = -1.0  # coefficient of t
            A_rows.append(row)
            b_vals.append(-const)
        
        A = np.array(A_rows)
        b = np.array(b_vals)
        
        # Objective: minimize t => c_vec = [0, ..., 0, 1]
        c_vec = np.zeros(n + 1)
        c_vec[n] = 1.0

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])

        bounds = [(0, 1)] * n + [(0, None)]

            res = linprog(c_vec, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
            if res.success:
                new_h = res.x[:n]
                new_h = project_feasible(new_h, n)
                new_c = compute_c5(new_h, n)
                old_c = compute_c5(h, n)
                if new_c < old_c:
                    h = new_h
            else:
                break
        except Exception:
            break
    
    return h, compute_c5(h, n)

            h[add] = 1
    h = h.astype(float)
    cur = compute_c5(h, n)
    best = cur
    best_h = h.copy()
    temp = 0.008
    for it in range(max_iters):
        if time.time() > deadline:
            break
        p = rng.randint(n)
        if h[p] == 1:
            cands = np.where(h == 0)[0]
        else:
            cands = np.where(h == 1)[0]
        if len(cands) == 0:
            continue
        q = cands[rng.randint(len(cands))]
        h[p], h[q] = 1 - h[p], 1 - h[q]
        new_c = compute_c5(h, n)
        if new_c < best:
            best = new_c
            best_h = h.copy()
            cur = new_c
        elif rng.rand() > np.exp(-(new_c - cur) / (temp + 1e-12)):
            h[p], h[q] = 1 - h[p], 1 - h[q]
        else:
            cur = new_c
        temp *= 0.9997
    return best_h, best

def thermal_push(h, n, beta, deadline):
    h_new = 1.0 / (1.0 + np.exp(-beta * (h - 0.5)))
    return project_feasible(h_new, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = np.full(700, 0.5)
        n = 700
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)

    # Phase 1: CD polish with random ordering
    if time.time() < deadline - 350:
        h, c = exact_cd(h_best, n, 100, deadline - 350, rng)
        if c < best_c:
            h_best, best_c = h, c

    # Phase 2: SCA LP guided optimization
    if time.time() < deadline - 250:
        for _ in range(8):
            if time.time() > deadline - 250:
                break
            h_sca = sca_lp(h_best, n)
            h_cd, c_cd = exact_cd(h_sca, n, 25, deadline - 250, rng)
            if c_cd < best_c:
                h_best, best_c = h_cd, c_cd

    # Phase 3: Binary SA from random init
    if time.time() < deadline - 200:
        h_rand = np.zeros(n)
        h_rand[:n // 2] = 1.0
        rng.shuffle(h_rand)
        h_sa, c_sa = binary_sa(h_rand, n, 60000, deadline - 250, rng)
        h_cd, c_cd = exact_cd(project_feasible(h_sa, n), n, 60, deadline - 200, rng)
        if c_cd < best_c:
            h_best, best_c = h_cd, c_cd

    # Phase 4: Binary SA from parent threshold
    if time.time() < deadline - 150:
        h_thr = (h_best >= 0.5).astype(float)
        h_thr = project_feasible(h_thr, n)
        h_sa, c_sa = binary_sa(h_thr, n, 40000, deadline - 200, rng)
        h_cd, c_cd = exact_cd(project_feasible(h_sa, n), n, 60, deadline - 150, rng)
        if c_cd < best_c:
            h_best, best_c = h_cd, c_cd

    # Phase 5: Thermal push + CD
    if time.time() < deadline - 120:
        for beta in [15, 25, 40, 60]:
            if time.time() > deadline - 120:
                break
            h_tp = thermal_push(h_best, n, beta, deadline)
            h_cd, c_cd = exact_cd(h_tp, n, 40, deadline - 120, rng)
            if c_cd < best_c:
                h_best, best_c = h_cd, c_cd

    # Phase 6: Multi-start perturbation
    if time.time() < deadline - 80:
        for r in range(10):
            if time.time() > deadline - 80:
                break
            s = 0.003 + 0.012 * r
            hp = project_feasible(h_best + rng.normal(0, s, n), n)
            h_cd, c_cd = exact_cd(hp, n, 35, deadline - 80, rng)
            if c_cd < best_c:
                h_best, best_c = h_cd, c_cd

    # Phase 7: Multi-n exploration
    for n_try in [600, 800, 900, 550]:
        if time.time() > deadline - 60:
            break
        if n_try == n:
            continue
        x_old = np.linspace(0, 2, n)
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_old, h_best), n_try)
        h_cd, c_cd = exact_cd(h_int, n_try, 50, deadline - 60, rng)
        if c_cd < best_c:
            h_best, best_c = h_cd, c_cd
            n = n_try

    # Phase 8: Final intensive CD
    n = len(h_best)
    h_f, c_f = exact_cd(h_best, n, 300, deadline)
    if c_f < best_c:
        h_best, best_c = h_f, c_f

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
