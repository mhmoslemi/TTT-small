import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def _sum_eq(x, n):
    return x.sum() - n / 2.0

def _sum_eq_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alphas, maxiter_each=300):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac, 'args': (n,)}]
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-15})
            if res.x is not None:
                new_h = project_feasible(res.x, n)
                new_c = compute_c5(new_h, n)
                if new_c < best_c:
                    best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

    vals = c_aff + d * a
    if has_diag:
        vals[diag_idx] = c_aff[diag_idx] + d * (1.0 - h0 - d)
    result = float(np.max(vals))
    if unaff_max > result:
        result = unaff_max
    return result

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-13 or h0 >= 1.0 - 1e-13:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)

            jb = p - ms + (n - 1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(valid_f):
                a[valid_f] += (1.0 - h[jf[valid_f]])
            if np.any(valid_b):
                a[valid_b] -= h[jb[valid_b]]
            diag_pos = np.where(ms == (n - 1))[0]
            has_diag = len(diag_pos) > 0
            diag_idx = int(diag_pos[0]) if has_diag else -1
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:

            d_lo = -h0

            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-15:
                    break
                if f1 < f2:
                    d_hi = d2
                    d2 = d1
                    f2 = f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
                else:
                    d_lo = d1
                    d1 = d2
                    f1 = f2
                    d2 = d_lo + gr * (d_hi - d_lo)
                    f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            d_opt = (d_lo + d_hi) / 2.0
            f_opt = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d_opt)

            if f_opt < f_curr - 1e-14:

                improved = True
        if improved:
            h[np.where(h < threshold)[0]] = 0.0
            h[np.where(h > 1.0 - threshold)[0]] = 1.0
            h = project_feasible(h, n)
            c_new = compute_c5(h, n)
            if c_new < best_c:
                best_c = c_new
        else:
            break
    final_c = compute_c5(h, n)
    return h, min(best_c, final_c)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def _sum_eq(x, n):
    return x.sum() - n / 2.0

def _sum_eq_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alphas, maxiter_each=300):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac, 'args': (n,)}]
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-15})
            if res.x is not None:
                new_h = project_feasible(res.x, n)
                new_c = compute_c5(new_h, n)
                if new_c < best_c:
                    best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

    vals = c_aff + d * a
    if has_diag:

    result = float(np.max(vals))
    if unaff_max > result:
        result = unaff_max
    return result

    h2, c2 = coord_descent(best_h, n, 120, pd, gs_iters=55, threshold=1e-5)
    if c2 < best_c:
        best_h, best_c = h2, c2

    # Phase 3: Multi-resolution cycles
    x_orig = np.linspace(0, 2, len(best_h))
    n_cur = len(best_h)
    pd = t0 + 480
    for n_small in [400, 300, 250]:
        if time.time() > pd - 80:
            break
        x_s = np.linspace(0, 2, n_small)
        h_s = project_feasible(np.interp(x_s, x_orig, best_h), n_small)
        h_s, c_s = slsqp_optimize(h_s, n_small, [2000, 8000], 150)
        h_s, c_s = coord_descent(h_s, n_small, 80, pd, gs_iters=55)
        h_back = project_feasible(np.interp(x_orig, x_s, h_s), n_cur)
        h_back, c_back = coord_descent(h_back, n_cur, 50, pd, gs_iters=50)
        if c_back < best_c:
            best_h, best_c = h_back, c_back
        if c_s < best_c:
            h_up = project_feasible(np.interp(x_orig, x_s, h_s), n_cur)
            h_up, c_up = coord_descent(h_up, n_cur, 40, pd, gs_iters=50)
            if c_up < best_c:
                best_h, best_c = h_up, c_up

    # Phase 4: Diversified restarts
    pd = t0 + 730
    n_cur = len(best_h)
    x = np.linspace(0, 2, n_cur)
    n_restarts = max(3, int((pd - time.time()) / 80))
    for r in range(n_restarts):
        if time.time() > pd - 30:
            break
        if r % 4 == 0:
            hp = project_feasible(best_h + 0.015 * rng.randn(n_cur), n_cur)
        elif r % 4 == 1:
            f = 3.0 + 0.7 * r
            hp = project_feasible(best_h + 0.03 * np.sin(2 * np.pi * f * (x - 1.0)), n_cur)
        elif r % 4 == 2:
            mask = (rng.rand(n_cur) < 0.08).astype(float)
            hp = project_feasible(best_h + 0.4 * mask - 0.2, n_cur)
        else:
            hp = best_h.copy()
            lo, hi = int(n_cur * 0.25), int(n_cur * 0.75)
            hp[lo:hi] += 0.025 * rng.randn(hi - lo)
            hp = project_feasible(hp, n_cur)
        h_o, c_o = coord_descent(hp, n_cur, 45, pd, gs_iters=45)
        if c_o < best_c:
            best_h, best_c = h_o, c_o

    # Phase 5: Final aggressive CD with adaptive threshold
    pd = deadline
    n_cur = len(best_h)
    h_f, c_f = coord_descent(best_h, n_cur, 400, pd, gs_iters=65, threshold=5e-6)
    if c_f < best_c:
        best_h, best_c = h_f, c_f

    # Phase 6: Final SLSQP polish
    if time.time() < deadline - 5:
        n_cur = len(best_h)
        h_f, c_f = slsqp_optimize(best_h, n_cur, [200000, 800000], 300)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
        best_h = project_feasible(best_h, n_cur)
        best_c = compute_c5(best_h, n_cur)

    return best_h, best_c, len(best_h)
