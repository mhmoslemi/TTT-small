        # And the smooth max = cm + (1/alpha) * log-sum-exp(alpha*(c-cm))
        # = cm + (1/alpha) * (mx + log(sum(exp(shifted-mx))))
        # But we want the smooth max of c, which is approx = cm + (1/alpha)*log(sum(exp(alpha*(c-cm))))
        # = (1/alpha) * log(sum(exp(alpha*c))) ... no that's not right either
        # 
        # Let me use: smooth_max(c, alpha) = (1/alpha) * log(sum(exp(alpha * c)))
        # This is a standard smooth approximation to max(c)
        # grad smooth_max/grad c = exp(alpha*c) / sum(exp(alpha*c))
        
        # Redo: use soft = (1/alpha) * log(sum(exp(alpha * c)))
        # To avoid overflow: let max_c = c.max()
        # soft = (1/alpha) * (alpha*max_c + log(sum(exp(alpha*(c-max_c)))))
        #       = max_c + (1/alpha) * log(sum(exp(alpha*(c-max_c))))
        
        max_c = c.max()
        exp_terms = np.exp(alpha * (c - max_c))
        log_sum = np.log(exp_terms.sum())
        soft = max_c + log_sum / alpha
        f = soft * (2.0 / n)
        
        if f < best_f:
            best_f = f
            best_h = h.copy()
        
        # Gradient: grad_soft = (2/n) * exp_terms / exp_terms.sum()
        w = exp_terms / exp_terms.sum()
        
        # grad_f/grad_h[p] = (2/n) * sum_m w[m] * dc[m]/dh[p]
        # dc[m]/dh[p] = (1-h[p+m-n+1]) - h[p-m+n+1] (with validity checks)
        # For m = n-1: dc[n-1]/dh[p] = (1-h[p]) - h[p] = 1-2h[p] but wait
        # c[m] = sum_i h[i]*(1-h[i+m-n+1])
        # dc[m]/dh[p] = (1-h[p+m-n+1]) [if p+m-n+1 in [0,n)] 
        #              + (-1) * sum_{i: i+m-n+1=p} h[i]
        # The second sum: i+m-n+1=p => i=p-m+n-1
        # So: dc[m]/dh[p] = (1-h[p+m-n+1]) [if valid] - h[p-m+n-1] [if valid]
        # For m=n-1: both terms involve h[p], giving (1-h[p]) - h[p] = 1-2h[p]
        # But wait, that double counts! When m=n-1, p+m-n+1=p and p-m+n-1=p
        # So dc[n-1]/dh[p] = (1-h[p]) + (-h[p]) = 1-2h[p]. That's correct.
        
        grad = np.zeros(n)
        for p in range(n):
            m_lo = n - 1 - p
            m_hi = 2 * n - 2 - p
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            contrib = np.zeros(len(ms))
            if np.any(vf):
                contrib[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                contrib[vb] -= h[jb[vb]]
            diag_mask = (ms == (n - 1))
            if np.any(diag_mask):
                contrib[diag_mask] = 1.0 - 2.0 * h[p]
            grad[p] = (2.0 / n) * np.dot(w[m_lo:m_hi+1], contrib)
        
        # Adam update
        beta1, beta2, eps = 0.8, 0.999, 1e-8
        m_v = beta1 * m_v + (1 - beta1) * grad
        v_v = beta2 * v_v + (1 - beta2) * grad**2
        m_hat = m_v / (1 - beta1**t)
        v_hat = v_v / (1 - beta2**t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, n)
        
        # Check if we made progress
        f_new = compute_c5(h_new, n)
        if f_new < compute_c5(h, n) + 0.001:
            h = h_new
        else:
            lr *= 0.5
    
    return h, best_h, best_f

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def solve_1d_bisect(c, h, p, n):
    h0 = h[p]
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms]
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    if has_diag:
        c_diag = c_base[diag_idx[0]]
        a[diag_idx] = 0.0
    unaff = 0.0
    if m_min > 0:
        unaff = float(np.max(c[:m_min]))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    cur_max = float(np.max(c))
    neg_mask = a < -1e-14
    pos_mask = a > 1e-14
    zero_mask = np.abs(a) <= 1e-14
    L_lo, L_hi = unaff, cur_max
    for _ in range(40):
        if L_hi - L_lo < 1e-11:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        feasible = True
        if np.any(zero_mask) and np.any(c_base[zero_mask] > L):
            feasible = False
        if feasible and np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if feasible and np.any(pos_mask):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        if x_lo > x_hi + 1e-12:
            feasible = False
        if feasible and has_diag:
            Cq = L - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                xc_lo, xc_hi = max(x_lo, 0.0), min(x_hi, 1.0)
                if not (xc_lo < r1 and xc_hi > r2):
                    feasible = False
        if feasible:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Cq = L - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            if h0 - r1 <= r2 - h0:
                x_hi = min(x_hi, r1)
            else:
                x_lo = max(x_lo, r2)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def exact_cd(h, n, max_passes, deadline):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            xn = solve_1d_bisect(c, h, p, n)
            if abs(xn - h0) > 1e-10:
                delta = xn - h0
                ms = np.arange(n - 1 - p, 2 * n - 1 - p)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                diag_i = np.where(ms == (n - 1))[0]
                c[ms] += delta * a
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def random_order_cd(h, n, max_passes, deadline, rng):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        order = rng.permutation(n)
        for p in order:
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            xn = solve_1d_bisect(c, h, p, n)
            if abs(xn - h0) > 1e-10:
                delta = xn - h0
                ms = np.arange(n - 1 - p, 2 * n - 1 - p)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                diag_i = np.where(ms == (n - 1))[0]
                c[ms] += delta * a
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def block_flip_perturb(h, n, rng):
    h = h.copy()
    block_len = rng.randint(n // 6, n // 2)
    start = rng.randint(0, n - block_len)
    h[start:start + block_len] = 1.0 - h[start:start + block_len]
    return project_feasible(h, n)

def sine_perturb(h, n, rng):
    h = h.copy()
    freq = rng.randint(1, 5)
    phase = rng.uniform(0, 2 * np.pi)
    amp = rng.uniform(0.05, 0.2)
    x = np.linspace(0, 2 * np.pi * freq, n) + phase
    h = h + amp * np.sin(x)
    return project_feasible(h, n)

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    c = np.correlate(h, 1.0 - h, mode='full')
    cm = c.max()
    def obj(x):
        cc = np.correlate(x, 1.0 - x, mode='full')
        ccmax = cc.max()
        shifted = alpha * (cc - ccmax)
        mx2 = shifted.max()
        lse = mx2 + np.log(np.sum(np.exp(shifted - mx2)))
        return (ccmax + lse / alpha) * (2.0 / n)
    def obj_grad(x):
        cc = np.correlate(x, 1.0 - x, mode='full')
        ccmax = cc.max()
        exp_t = np.exp(alpha * (cc - ccmax))
        w = exp_t / exp_t.sum()
        grad = np.zeros(n)
        for p in range(n):
            m_lo = n - 1 - p
            m_hi = 2 * n - 2 - p
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            contrib = np.zeros(len(ms))
            if np.any(vf):
                contrib[vf] += (1.0 - x[jf[vf]])
            if np.any(vb):
                contrib[vb] -= x[jb[vb]]
            diag_mask = (ms == (n - 1))
            if np.any(diag_mask):
                contrib[diag_mask] = 1.0 - 2.0 * x[p]
            grad[p] = (2.0 / n) * np.dot(w[m_lo:m_hi + 1], contrib)
        return grad
    def cons_fun(x):
        return x.sum() - n / 2.0
    def cons_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

    # Phase 3: Sine perturbations + CD

        if time.time() > deadline - 250:

    # Phase 4: Multi-n interpolation + optimize

    # Phase 5: SLSQP on smooth approximation (different alphas)
