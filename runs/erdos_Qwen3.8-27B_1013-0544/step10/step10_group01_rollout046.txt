def _penalty_obj_and_grad(x, n, alpha, mu):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = np.clip(alpha * (c - m), -500, 500)

        wk = w[k]
        if abs(wk) < 1e-14:

    grad *= (2.0 / n)

    # Final CD if time remains
    if time.time() < deadline - 1:
        best_h, best_c = coord_descent(best_h, len(best_h), 30, deadline, gs_iters=70, threshold=2e-5)

    return best_h, best_c, len(best_h)
