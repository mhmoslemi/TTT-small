             + contribution from p=j: (b-t)*(1-h'[j+s]) - b*(1-h[j+s])
             + contribution from p=i-s (if valid): h'[i-s]*(1-(a+t)) - h[i-s]*(1-a)
             + contribution from p=j-s (if valid): h'[j-s]*(1-(b-t)) - h[j-s]*(1-b)

For terms where i+s ∉ {i,j} and j+s ∉ {i,j} and i-s ∉ {i,j} and j-s ∉ {i,j}:
= (a+t)*(1-h[i+s]) - a*(1-h[i+s]) + (b-t)*(1-h[j+s]) - b*(1-h[j+s])
+ h[i-s]*(1-a-t) - h[i-s]*(1-a) + h[j-s]*(1-b+t) - h[j-s]*(1-b)
= t*(1-h[i+s]) - t*(1-h[j+s]) - t*h[i-s] + t*h[j-s]
= t * ((1-h[i+s]) - (1-h[j+s]) - h[i-s] + h[j-s])

So for most shifts, the change is LINEAR in t! This makes the optimization over t a 1D problem: minimize max_m (c[m] + t * coef[m]) over t in an interval, where coef[m] is known.

This is a "min of max of lines" problem, which can be solved by finding the intersection points.

This is much better than the single-coordinate approach because:
1. It maintains feasibility (no projection needed)
2. The optimization over t is 1D and exact
3. It's a different neighborhood than single-coordinate CD

            if f_opt * dx < best_c - 1e-14:
                h[i] = a + t_opt
                h[j] = b - t_opt
                c = c.copy()
                # Update c analytically
                c += t_opt * coef
                c[n-1] += -2.0 * t_opt * t_opt
                s_diff = j - i
                if s_diff != 0:
                    m1 = n - 1 + s_diff
                    m2 = n - 1 - s_diff
                    if 0 <= m1 < 2*n-1:
                        c[m1] += t_opt * t_opt
                    if 0 <= m2 < 2*n-1:
                        c[m2] += t_opt * t_opt
                best_c = float(c.max()) * dx
                improved = True
        
        if not improved:
            break
    
    return h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def pair_cd(h, n, deadline, max_passes=500, n_pairs=300, seed=0, sweep=False):
    h = h.copy()
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    c = np.correlate(h, 1.0 - h, mode='full')
    best_c = float(c.max()) * dx
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    shifts = np.arange(2 * n - 1) - (n - 1)
    
    for p in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        # Decide which pairs to try
        if sweep and p >= 5:
            # Systematic sweep: all pairs (or a large subset)
            pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
            rng.shuffle(pairs)
        else:
            pairs = [(rng.randint(0, n), rng.randint(0, n)) for _ in range(n_pairs)]
        
        for i, j in pairs:
            if time.time() > deadline:
                break
            if i == j:
                continue
            a, b = h[i], h[j]
            if a < 1e-12 or a > 1 - 1e-12 or b < 1e-12 or b > 1 - 1e-12:
                continue
            t_lo = max(-a, b - 1.0)
            t_hi = min(1.0 - a, b)
            if t_hi - t_lo < 1e-13:
                continue
            # Compute linear coefficient for each correlation index
            coef = np.zeros(2 * n - 1)
            vi_f = (shifts >= -i) & (shifts <= n - 1 - i)
            if np.any(vi_f):
                coef[vi_f] += (1.0 - h[i + shifts[vi_f]])
            vj_f = (shifts >= -j) & (shifts <= n - 1 - j)
            if np.any(vj_f):
                coef[vj_f] -= (1.0 - h[j + shifts[vj_f]])
            vi_b = (shifts >= i - n + 1) & (shifts <= i)
            if np.any(vi_b):
                coef[vi_b] -= h[i - shifts[vi_b]]
            vj_b = (shifts >= j - n + 1) & (shifts <= j)
            if np.any(vj_b):
                coef[vj_b] += h[j - shifts[vj_b]]
            sd = j - i
            # Golden section on t
            def eval_f(t):
                vals = c + t * coef
                vals[n - 1] -= 2.0 * t * t
                if sd != 0:
                    m1 = n - 1 + sd
                    m2 = n - 1 - sd
                    if 0 <= m1 < 2 * n - 1:
                        vals[m1] += t * t
                    if 0 <= m2 < 2 * n - 1:
                        vals[m2] += t * t
                return float(np.max(vals))
            
            tl, th = t_lo + 1e-14, t_hi - 1e-14
            tm1 = th - gr * (th - tl)
            tm2 = tl + gr * (th - tl)
            fm1 = eval_f(tm1)
            fm2 = eval_f(tm2)
            for _ in range(35):
                if th - tl < 1e-13:
                    break
                if fm1 < fm2:
                    th, tm2, fm2 = tm2, tm1, fm1
                    tm1 = th - gr * (th - tl)
                    fm1 = eval_f(tm1)
                else:
                    tl, tm1, fm1 = tm1, tm2, fm2
                    tm2 = tl + gr * (th - tl)
                    fm2 = eval_f(tm2)
            t_opt = (tl + th) / 2.0
            f_opt = eval_f(t_opt)
            if f_opt * dx < best_c - 1e-14:
                h[i] = a + t_opt
                h[j] = b - t_opt
                c = c + t_opt * coef
                c[n - 1] -= 2.0 * t_opt * t_opt
                if sd != 0:
                    m1 = n - 1 + sd
                    m2 = n - 1 - sd
                    if 0 <= m1 < 2 * n - 1:
                        c[m1] += t_opt * t_opt
                    if 0 <= m2 < 2 * n - 1:
                        c[m2] += t_opt * t_opt
                best_c = float(c.max()) * dx
                improved = True
        if not improved:
            break
    return h, best_c

def single_cd(h, n, deadline, max_passes=30, gs_iters=45):
    h = h.copy()
    dx = 2.0 / n
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-13 or h0 >= 1.0 - 1e-13:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            a = np.zeros(len(ms))
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            if np.any(vf):
                a[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                a[vb] -= h[jb[vb]]
            dp = np.where(ms == (n - 1))[0]
            has_diag = len(dp) > 0
            di = int(dp[0]) if has_diag else -1
            c_aff = c[ms]
            unaff = 0.0
            if m_lo > 0:
                unaff = max(unaff, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff = max(unaff, float(c[m_hi + 1:].max()))
            d_lo, d_hi = -h0, 1.0 - h0
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            def cf(d):
                vals = c_aff + d * a
                if has_diag:
                    vals[di] = c_aff[di] + d * (1.0 - h0 - d)
                return max(float(np.max(vals)), unaff)
            f1, f2 = cf(d1), cf(d2)
            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-14:
                    break
                if f1 < f2:
                    d_hi, d2, f2 = d2, d1, f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    f1 = cf(d1)
                else:
                    d_lo, d1, f1 = d1, d2, f2
                    d2 = d_lo + gr * (d_hi - d_lo)
                    f2 = cf(d2)
            d_opt = (d_lo + d_hi) / 2.0
            if cf(d_opt) < cf(0.0) - 1e-14:
                h[p] = h0 + d_opt
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def swap_search_01(h, n, deadline, max_iters=500, n_tries=2000, seed=0):
    rng = np.random.RandomState(seed)
    h = h.copy()
    dx = 2.0 / n
    # Round to 0-1
    idx_sorted = np.argsort(-h)
    h[idx_sorted[:n // 2]] = 1.0
    h[idx_sorted[n // 2:]] = 0.0
    c = np.correlate(h, 1.0 - h, mode='full')
    best_c = float(c.max()) * dx
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    shifts = np.arange(2 * n - 1) - (n - 1)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        best_gain = 1e-10
        best_i, best_j = None, None
        for _ in range(n_tries):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            sd = np.arange(1)  # just for shape
            # Delta in c for swap i(1->0), j(0->1) at each shift
            # delta[m] = h[i+s] + h[i-s] - h[j+s] - h[j-s] (0-1 values, 0 if out of range)
            vi_f = (shifts >= -i) & (shifts <= n - 1 - i)
            vj_f = (shifts >= -j) & (shifts <= n - 1 - j)
            vi_b = (shifts >= i - n + 1) & (shifts <= i)
            vj_b = (shifts >= j - n + 1) & (shifts <= j)
            delta = np.zeros(2 * n - 1)
            if np.any(vi_f):
                delta[vi_f] += h[i + shifts[vi_f]]
            if np.any(vi_b):
                delta[vi_b] += h[i - shifts[vi_b]]
            if np.any(vj_f):
                delta[vj_f] -= h[j + shifts[vj_f]]
            if np.any(vj_b):
                delta[vj_b] -= h[j - shifts[vj_b]]
            new_max = float(np.max(c + delta)) * dx
            if new_max < best_c - best_gain:
                best_gain = best_c - new_max
                best_i, best_j = i, j
        if best_i is not None:
            h[best_i], h[best_j] = 0.0, 1.0
            vi_f = (shifts >= -best_i) & (shifts <= n - 1 - best_i)
            vj_f = (shifts >= -best_j) & (shifts <= n - 1 - best_j)
            vi_b = (shifts >= best_i - n + 1) & (shifts <= best_i)
            vj_b = (shifts >= best_j - n + 1) & (shifts <= best_j)
            delta = np.zeros(2 * n - 1)
            if np.any(vi_f):
                delta[vi_f] += h[best_i + shifts[vi_f]]
            if np.any(vi_b):
                delta[vi_b] += h[best_i - shifts[vi_b]]
            if np.any(vj_f):
                delta[vj_f] -= h[best_j + shifts[vj_f]]
            if np.any(vj_b):
                delta[vj_b] -= h[best_j - shifts[vj_b]]
            c = c + delta
            best_c = float(c.max()) * dx
            ones.remove(best_i)
            zeros.remove(best_j)
            ones.append(best_j)
            zeros.append(best_i)
    return h, best_c

            best_h, best_c = h3, c3
    
    # Phase 4: Pair CD again with sweep

        if c4 < best_c:
            best_h, best_c = h4, c4

            if c5 < best_c:
                best_h, best_c = h5, c5

    best_h = project_feasible(best_h, len(best_h))

    return best_h, best_c, len(best_h)
