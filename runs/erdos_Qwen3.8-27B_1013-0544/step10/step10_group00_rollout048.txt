def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def grad_c_at_lag(h, s):
    n = len(h)
    if s == 0:
        return 1.0 - 2.0 * h
    grad = np.zeros(n)
    if s > 0:
        grad[s:] += (1.0 - h[:n - s])
        grad[:n - s] -= h[s:]
    else:
        s2 = -s
        grad[:n - s2] += (1.0 - h[s2:])
        grad[s2:] -= h[:n - s2]

def project_feasible(h, n):

    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)

def _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d):
    vals = c_aff + d * a
    if has_diag:
        vals[diag_idx] = c_aff[diag_idx] + d * (1.0 - h0 - d)
    result = float(np.max(vals))
    if unaff_max > result:
        result = unaff_max
    return result

def top_k_push(h, n, k=5, step=0.02, deadline=None):
    c = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(c)[-k:]
    grad = np.zeros(n)

        s = mk - (n - 1)
        grad += grad_c_at_lag(h, s)
    h_new = project_feasible(h - step * grad, n)

def nesterov_phase(h, n, deadline, max_iters=3000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    y = h.copy()
    beta_t = 1.0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        beta_t = (1.0 + np.sqrt(1.0 + 4.0 * beta_t * beta_t)) / 2.0
        momentum = (beta_t - 1.0) / (beta_t + 1.0)
        y = h + momentum * (h - y)
        yc = np.clip(y, 0, 1)
        c = np.correlate(yc, 1.0 - yc, mode='full')
        k_star = int(np.argmax(c))
        s = k_star - (n - 1)
        grad = grad_c_at_lag(yc, s)
        # Also add top-3 for robustness
        top3 = np.argsort(c)[-3:]
        for mk in top3:
            s2 = mk - (n - 1)
            if s2 != s:
                grad += 0.5 * grad_c_at_lag(yc, s2)
        t = it + 1
        eta = 0.05 / (1.0 + 0.3 * np.sqrt(t))
        h = project_feasible(y - eta * grad, n)
        c_val = compute_c5(h, n)
        if c_val < best_c:
            best_h = h.copy()
            best_c = c_val
    return best_h, best_c

    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * dx

def slsqp_opt(h, n, alphas, maxiter=400):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        def fun(x):
            return smooth_obj(x, n, alpha)
        def cons_fun(x):
            return x.sum() - n / 2.0
        try:
            res = minimize(fun, best_h, method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': cons_fun,
                                         'jac': lambda x: np.ones(n)}],
                           options={'maxiter': maxiter, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 700
        h_init = np.full(n, 0.5)
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    # Phase 1: Intensive CD
    d1 = deadline - 350
    h1, c1 = coord_descent(best_h, n, 150, d1, gs_iters=60)
    if c1 < best_c:
        best_h, best_c = h1, c1
    # Phase 2: Nesterov subgradient
    if time.time() < deadline - 250:
        h2, c2 = nesterov_phase(best_h, n, deadline - 250, max_iters=2000)
        if c2 < best_c:
            best_h, best_c = h2, c2
        h2, c2 = coord_descent(h2, n, 80, deadline - 250, gs_iters=55)
        if c2 < best_c:
            best_h, best_c = h2, c2
    # Phase 3: Top-k push + CD
    if time.time() < deadline - 200:
        for _ in range(50):
            if time.time() > deadline - 200:
                break
            h3, c3 = top_k_push(best_h, n, k=5, step=0.015)
            h3, c3 = coord_descent(h3, n, 40, deadline - 200, gs_iters=50)
            if c3 < best_c:
                best_h, best_c = h3, c3
            else:
                break
    # Phase 4: Try different n
    x_orig = np.linspace(0, 2, len(best_h))
    for n_try in [500, 600, 800, 900, 1000]:
        if time.time() > deadline - 120:
            break
        if n_try == len(best_h):
            continue
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h4, c4 = coord_descent(h_int, n_try, 80, deadline - 120, gs_iters=55)
        if c4 < best_c:
            h4, c4 = slsqp_opt(h4, n_try, [5000, 20000], 300)
            h4, c4 = coord_descent(h4, n_try, 50, deadline - 120, gs_iters=55)
        if c4 < best_c:
            best_h, best_c = h4, c4
            x_orig = np.linspace(0, 2, n_try)
    # Phase 5: Multi-restart
    if time.time() < deadline - 60:
        n_cur = len(best_h)
        x = np.linspace(0, 2, n_cur)
        for r in range(12):
            if time.time() > deadline - 60:
                break
            if r % 3 == 0:
                hp = project_feasible(best_h + 0.03 * rng.normal(0, 1, n_cur), n_cur)
            elif r % 3 == 1:
                hp = project_feasible(best_h + 0.02 * np.sin(2 * np.pi * (2 + r) * x), n_cur)
            else:
                block = max(10, n_cur // 10)
                hp = best_h.copy()
                start = int(rng.randint(0, n_cur - block))
                hp[start:start + block] = 1.0 - hp[start:start + block]
                hp = project_feasible(hp, n_cur)
            h5, c5 = coord_descent(hp, n_cur, 50, deadline - 60, gs_iters=50)
            if c5 < best_c:
                h5, c5 = nesterov_phase(h5, n_cur, deadline - 60, max_iters=500)
                h5, c5 = coord_descent(h5, n_cur, 40, deadline - 60, gs_iters=55)
            if c5 < best_c:
                best_h, best_c = h5, c5
    # Phase 6: Final polish
    if time.time() < deadline - 3:
        h6, c6 = slsqp_opt(best_h, len(best_h), [10000, 50000, 200000], 500)
        if c6 < best_c:
            best_h, best_c = h6, c6
        h6, c6 = coord_descent(best_h, len(best_h), 60, deadline - 3, gs_iters=65)
        if c6 < best_c:
            best_h, best_c = h6, c6
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
