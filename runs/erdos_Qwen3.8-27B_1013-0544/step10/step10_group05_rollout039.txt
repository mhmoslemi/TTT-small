```python
def smooth_obj_lse(x, n, alpha):
    """Log-sum-exp smooth approximation of max."""
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    if m < -1e10:
        return 0.0
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

    if c_pos.max() < 1e-20:
        return 0.0
    return (np.sum(c_pos ** p) / len(c_pos)) ** (1.0 / p) * (2.0 / n)
```

```python
def grad_c5_smooth(h, n, alpha):
    """Gradient of the log-sum-exp smooth C5 objective."""
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    
    grad = np.zeros(n)
    for p in range(n):
        # Forward: h[p] as first factor
        j_start = max(0, p - (n-1))
        j_end = min(n-1, n-1+p)
        if j_end >= j_start:
            k_indices = (n-1-p) + np.arange(j_start, j_end+1)
            grad[p] += w[k_indices] @ (1.0 - h[j_start:j_end+1])
        # Backward: h[p] as second factor (in 1-h[...])
        j_start_b = max(0, p - (n-1))
        j_end_b = min(n-1, p + (n-1))
        if j_end_b >= j_start_b:
            k_indices_b = (n-1+p) - np.arange(j_start_b, j_end_b+1)
            grad[p] += w[k_indices_b] @ (-h[j_start_b:j_end_b+1])
    
    grad *= (2.0 / n)
    return grad
```

```python
    grad[j] += (-h[i])  # d/dh[j] of h[i]*(1-h[j])
```

This is correct! When i=j (diagonal), both updates go to the same index: grad[i] += (1-h[i]) + (-h[i]) = 1-2h[i] ✓

```python
def run_cd(h, n, max_passes, deadline, rng=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        order = rng.permutation(n) if (rng is not None) else range(n)
        for p in order:
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            xn = solve_coord(c, h, p, n)
            if abs(xn - h0) > 1e-14:
                delta = xn - h0
                m_min = max(0, n - 1 - p)
                m_max = min(2 * n - 2, 2 * n - 2 - p)
                ms = np.arange(m_min, m_max + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                c[ms] += delta * a
                diag_i = np.where(ms == (n - 1))[0]
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def subgradient_step(h, n, eta, top_k):
    c = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(c)[::-1][:top_k]
    grad = np.zeros(n)
    for k in idx:
        i_start = max(0, n - 1 - k)
        i_end = min(n - 1, n - 1 + k)
        for i in range(i_start, i_end + 1):
            j = i + k - (n - 1)
            grad[i] += (1.0 - h[j])
            grad[j] -= h[i]
    norm_g = np.linalg.norm(grad)
    if norm_g < 1e-15:
        return h.copy()
    h_new = h - eta * grad / (norm_g * len(idx))
    return project_feasible(h_new, n)

def smooth_obj_lse(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def smooth_grad_lse(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    grad = np.zeros(n)
    for p in range(n):
        j_s = max(0, p - (n - 1))
        j_e = min(n - 1, n - 1 + p)
        k_f = (n - 1 - p) + np.arange(j_s, j_e + 1)
        grad[p] += w[k_f] @ (1.0 - h[j_s:j_e + 1])
        k_b = (n - 1 + p) - np.arange(j_s, j_e + 1)
        grad[p] += w[k_b] @ (-h[j_s:j_e + 1])
    grad *= (2.0 / n)
    return grad

def lbfgs_opt(h0, n, alpha, maxiter, deadline, penalty=1e5):
    h = h0.copy()
    def obj(x):
        return smooth_obj_lse(x, n, alpha) + penalty * (x.sum() - n / 2.0) ** 2
    def grad(x):
        return smooth_grad_lse(x, n, alpha) + 2 * penalty * (x.sum() - n / 2.0) * np.ones(n)
    bounds = [(0.0, 1.0)] * n
    best_h, best_c = h.copy(), compute_c5(h, n)
    try:
        res = minimize(obj, best_h, jac=grad, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
        new_h = project_feasible(res.x, n)
        new_c = compute_c5(new_h, n)
        if new_c < best_c:
            best_h, best_c = new_h, new_c
    except Exception:
        pass
    if time.time() > deadline:
        return best_h, best_c
    best_h, best_c = run_cd(best_h, n, 5, deadline)
    return best_h, best_c

def sg_cd_cycle(h, n, sg_steps, cd_passes, deadline, rng, top_k=10, eta0=0.05):
    h = h.copy()
    c = compute_c5(h, n)
    for i in range(sg_steps):
        if time.time() > deadline:
            break
        eta = eta0 * (1.0 + 0.1 * i)
        h_new = subgradient_step(h, n, eta, top_k)
        h_new, c_new = run_cd(h_new, n, cd_passes, deadline)
        if c_new < c:
            h, c = h_new, c_new
        else:
            h = subgradient_step(h, n, eta * 0.5, top_k)
            h, c = run_cd(h, n, cd_passes, deadline)
    return h, c

def interpolate_h(h_src, n_src, n_tgt):

    x_tgt = np.linspace(0, 2, n_tgt, endpoint=False)
    return project_feasible(np.interp(x_tgt, x_src, h_src), n_tgt)

def fourier_init(n, K, rng):
    x = np.linspace(0, 2, n, endpoint=False)

    return project_feasible(h, n)
```
