        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def obj_grad(h, n, alpha, pen):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    shifted = alpha * (corr - m)
    mx = shifted.max()
    w_un = np.exp(shifted - mx)
    w_sum = w_un.sum()
    fn = dx * (m + (mx + np.log(w_sum)) / alpha)
    w = w_un / w_sum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            i_hi = n - s
            if i_hi > 0:
                grad[:i_hi] += wk * (1.0 - h[s:s + i_hi])
            if s > 0:
                grad[s:] -= wk * h[:i_hi]
        else:
            s2 = -s
            i_c = n - s2
            if i_c > 0:
                grad[s2:] += wk * (1.0 - h[:i_c])
                grad[:i_c] -= wk * h[s2:s2 + i_c]
    grad *= dx
    if pen > 0:
        sv = h.sum() - n / 2.0
        grad += 2.0 * pen * sv
        fn += pen * sv * sv
    return fn, grad

def lse_run(h, n, alpha, pen, maxiter, deadline):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_grad, h, args=(n, alpha, pen), method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-8,
                                               'ftol': 1e-12, 'maxfun': maxiter * 3})
        return project_feasible(res.x, n)
    except Exception:
        return h

def solve_coord_1d(corr, h, p, n, nbisect=40):
    h0 = h[p]
    if h0 < 1e-14 or h0 > 1.0 - 1e-14:
        return h0
    lo_a = max(0, n - 1 - p)
    hi_a = min(2 * n - 2, n - 1 + p)
    lo_b = max(0, p)
    hi_b = min(2 * n - 2, p + n - 1)
    m_lo = min(lo_a, lo_b)
    m_hi = max(hi_a, hi_b)
    ms = np.arange(m_lo, m_hi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = corr[ms].copy()
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    c_diag = 0.0
    if has_diag:
        c_diag = c_base[diag_idx[0]]
        a[diag_idx] = 0.0
    non_diag = np.ones(len(ms), dtype=bool)
    if has_diag:
        non_diag[diag_idx] = False
    a_nd = a[non_diag]
    c_nd = c_base[non_diag]
    unaff_max = 0.0
    if m_lo > 0:
        unaff_max = max(unaff_max, float(np.max(corr[:m_lo])))
    if m_hi < 2 * n - 2:
        unaff_max = max(unaff_max, float(np.max(corr[m_hi + 1:])))
    cur_max = float(np.max(corr))
    L_lo = unaff_max
    L_hi = cur_max
    for _ in range(nbisect):
        if L_hi - L_lo < 1e-14:
            break
        T = (L_lo + L_hi) / 2.0
        feasible = True
        if unaff_max > T + 1e-15:
            feasible = False
        elif feasible and len(a_nd) > 0:
            x_lo_c = 0.0
            x_hi_c = 1.0
            neg = a_nd < -1e-13
            pos = a_nd > 1e-13
            if np.any(neg):
                x_lo_c = max(x_lo_c, float(np.max((T - c_nd[neg]) / a_nd[neg])))
            if np.any(pos):
                x_hi_c = min(x_hi_c, float(np.min((T - c_nd[pos]) / a_nd[pos])))
            if x_lo_c > x_hi_c + 1e-14:
                feasible = False
        if feasible and has_diag:
            Cq = T - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                if len(a_nd) > 0:
                    neg = a_nd < -1e-13
                    pos = a_nd > 1e-13
                    xl = 0.0
                    xh = 1.0
                    if np.any(neg):
                        xl = max(xl, float(np.max((T - c_nd[neg]) / a_nd[neg])))
                    if np.any(pos):
                        xh = min(xh, float(np.min((T - c_nd[pos]) / a_nd[pos])))
                    xl = max(xl, 0.0)
                    xh = min(xh, 1.0)
                    if xl > xh + 1e-14:
                        feasible = False
                    elif xl >= r1 and xh <= r2:
                        feasible = False
                else:
                    if 0.0 >= r1 and 1.0 <= r2:
                        feasible = False
        if feasible:
            L_hi = T
        else:
            L_lo = T
    T = (L_lo + L_hi) / 2.0
    x_lo = 0.0
    x_hi = 1.0
    if len(a_nd) > 0:
        neg = a_nd < -1e-13
        pos = a_nd > 1e-13
        if np.any(neg):
            x_lo = max(x_lo, float(np.max((T - c_nd[neg]) / a_nd[neg])))
        if np.any(pos):
            x_hi = min(x_hi, float(np.min((T - c_nd[pos]) / a_nd[pos])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Cq = T - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            xl2 = max(x_lo, 0.0)
            xh2 = min(x_hi, 1.0)
            if xl2 < r1:
                x_hi = min(x_hi, r1)
            elif xh2 > r2:
                x_lo = max(x_lo, r2)
            else:
                mid = (r1 + r2) / 2.0
                if h0 - r1 <= r2 - h0:
                    x_hi = min(x_hi, r1 + 1e-9)
                else:
                    x_lo = max(x_lo, r2 - 1e-9)
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if x_lo > x_hi + 1e-10:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def coord_descent(h, n, max_passes, deadline, rng, nbisect=40):
    h = h.copy()
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        order = rng.permutation(n)
        for p in order:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            x_new = solve_coord_1d(corr, h, p, n, nbisect)
            if abs(x_new - h0) > 1e-12:
                h[p] = x_new
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def make_perturbation(base, n, rng, mode):
    x = np.linspace(0, 2, n)
    if mode == 0:
        m = rng.randint(1, 5)
        amp = rng.uniform(0.02, 0.08)
        phase = rng.uniform(0, 2 * np.pi)
        h = base + amp * np.sin(np.pi * m * x + phase)
    elif mode == 1:
        h = base + rng.uniform(-0.05, 0.05, n)
    elif mode == 2:
        h = base + 0.05 * np.sin(2 * np.pi * rng.randint(1, 8) * x)
    else:
        h = 0.7 * base + 0.3 * (1.0 - base[::-1])
    return project_feasible(np.clip(h, 0, 1), n)

def optimize_one(h, n, deadline, rng):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    alphas = [20, 100, 500, 2000, 8000, 20000]
    maxiters = [100, 150, 200, 300, 300, 400]
    pens = [500, 500, 1000, 1000, 2000, 2000]
    for i, (al, mi, pe) in enumerate(zip(alphas, maxiters, pens)):
        if time.time() > deadline - 10:
            break
        h_new = lse_run(best_h, n, al, pe, mi, deadline)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_h, best_c = h_new, c_new
    h_cd, c_cd = coord_descent(best_h, n, 60, deadline, rng, 40)
    if c_cd < best_c:
        best_h, best_c = h_cd, c_cd
    if time.time() < deadline - 15:
        h_cd2, c_cd2 = coord_descent(best_h, n, 40, deadline, rng, 45)
        if c_cd2 < best_c:
            best_h, best_c = h_cd2, c_cd2
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
    except Exception:
        n_parent = 700
        x = np.linspace(0, 2, n_parent)
        h_parent = 0.5 + 0.15 * np.sin(np.pi * x) + 0.05 * np.sin(3 * np.pi * x)
        h_parent = np.clip(h_parent, 0, 1)

    grid_sizes = [350, 500, 700]
    best_h = None
    best_c = 1e9
    best_n = 0

    for n in grid_sizes:
        if time.time() > deadline - 30:
            break
        if n == n_parent:
            h_start = project_feasible(h_parent, n)
        else:
            x_old = np.linspace(0, 2, n_parent)
            x_new = np.linspace(0, 2, n)
            h_start = project_feasible(np.interp(x_new, x_old, h_parent), n)
        h_opt, c_opt = optimize_one(h_start, n, deadline, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()
            best_n = n

    if time.time() > deadline - 20:
        result = project_feasible(best_h, best_n)
        return result, compute_c5(result, best_n), best_n

    n_start = max(100, min(best_n, 500))
    x_best = np.linspace(0, 2, best_n)
    n_start_x = np.linspace(0, 2, n_start)
    h_base = np.interp(n_start_x, x_best, best_h)
    h_base = project_feasible(h_base, n_start)

    for trial in range(6):
        if time.time() > deadline - 40:
            break
        h_pert = make_perturbation(h_base, n_start, rng, trial % 4)
        h_opt, c_opt = optimize_one(h_pert, n_start, deadline, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()
            best_n = n_start

    if time.time() < deadline - 30:
        h_final = project_feasible(best_h, best_n)
        h_cd_f, c_cd_f = coord_descent(h_final, best_n, 50, deadline, rng, 45)
        if c_cd_f < best_c:
            best_h, best_c = h_cd_f, c_cd_f
        if time.time() < deadline - 10:
            for al in [30000, 50000]:
                if time.time() > deadline - 5:
                    break
                h_new = lse_run(best_h, best_n, al, 5000, 200, deadline)
                c_new = compute_c5(h_new, best_n)
                if c_new < best_c:
                    best_h, best_c = h_new, c_new
                h_cd3, c_cd3 = coord_descent(best_h, best_n, 50, deadline, rng, 45)
                if c_cd3 < best_c:
                    best_h, best_c = h_cd3, c_cd3

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
