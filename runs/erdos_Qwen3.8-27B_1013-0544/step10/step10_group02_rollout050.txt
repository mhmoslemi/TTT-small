def solve_1d(c, h, p, n):
    h0 = h[p]
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms].copy()
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    if has_diag:
        c_diag = c_base[diag_idx[0]]
        a[diag_idx] = 0.0
    unaff = 0.0
    if m_min > 0:
        unaff = float(np.max(c[:m_min]))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    cur_max = float(np.max(c))
    neg_mask = a < -1e-15
    pos_mask = a > 1e-15
    zero_mask = np.abs(a) <= 1e-15
    L_lo, L_hi = unaff, cur_max
    for _ in range(50):
        if L_hi - L_lo < 1e-12:
            break

    delta = new_val - old_val
    if abs(delta) < 1e-14:
        return c
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)

        a[vb] -= h[jb[vb]]

    c[ms] += delta * a

def lse_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    grad = np.zeros(n)
    i = np.arange(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-11:
            continue
        jf = i + k - (n - 1)
        vf = (jf >= 0) & (jf < n)
        jb = i - k + (n - 1)
        vb = (jb >= 0) & (jb < n)
        if np.any(vf):
            grad[vf] += wk * (1.0 - h[jf[vf]])
        if np.any(vb):
            grad[vb] -= wk * h[jb[vb]]
    f_val = (m + np.log(w.sum()) / alpha) * (2.0 / n)
    return grad, f_val

def slsqp_opt(h, n, alphas, maxiter, deadline):

    for alpha in alphas:
        if time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        def cf(x):
            return x.sum() - n / 2.0
        def cj(x):
            return np.ones(n)
        cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]
        def obj(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            m = c.max()
            sh = alpha * (c - m)
            mx = sh.max()
            lse = mx + np.log(np.sum(np.exp(sh - mx)))
            return (m + lse / alpha) * (2.0 / n)
        try:
            res = minimize(obj, best_h, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c - 1e-12:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

    for r in range(40):
        if time.time() > deadline - 200:
            break
        frac = 0.08 + 0.15 * (1.0 - r / 40.0)
        n_pert = max(5, int(n * frac))
        idx = rng.choice(n, n_pert, replace=False)
        scale = 0.005 + 0.03 * (1.0 - r / 40.0)
        hp = h_best.copy()
        hp[idx] = np.clip(hp[idx] + rng.normal(0, scale, n_pert), 0, 1)
        hp = project_feasible(hp, n)
        h_o, c_o = exact_cd(hp, n, 60, deadline, rng)
        if c_o < best_c - 1e-10:
            h_best, best_c = h_o, c_o

    # Phase 4: SLSQP
    if time.time() < deadline - 120:
        h4, c4 = slsqp_opt(h_best, n, [20000, 80000, 300000], 250, deadline)
        if c4 < best_c - 1e-12:
            h_best, best_c = h4, c4
        h_o, c_o = exact_cd(h_best, n, 100, deadline, rng)
        if c_o < best_c:
            h_best, best_c = h_o, c_o

    # Phase 5: Multi-resolution
    n_list = [650, 750, 800, 550, 600, 900, 450, 1000]
    for n_try in n_list:
        if time.time() > deadline - 40:
            break
        h_int = interp_h(h_best, len(h_best), n_try)
        h_o, c_o = exact_cd(h_int, n_try, 100, deadline, rng)
        if c_o < best_c - 1e-9:
            h_best, best_c = h_o, c_o

    # Phase 6: Final polish
    h_f, c_f = exact_cd(h_best, len(h_best), 300, deadline, rng)
    if c_f < best_c:
        h_best, best_c = h_f, c_f

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
