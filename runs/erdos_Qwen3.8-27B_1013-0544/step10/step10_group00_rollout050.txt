            d_opt = (d_lo + d_hi) / 2.0
            f_opt = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d_opt)
            f_curr = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, 0.0)
            if f_opt < f_curr - 1e-15:
                h[p] = h0 + d_opt
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def block_flip_perturb(h, n, rng):
    h = h.copy()
    block_sizes = [n // 10, n // 7, n // 5, n // 4]
    bs = block_sizes[rng.randint(len(block_sizes))]
    start = rng.randint(0, n - bs)
    h[start:start + bs] = 1.0 - h[start:start + bs]
    return project_feasible(h, n)

def fourier_perturb(h, n, freq, amplitude, phase):
    x = np.arange(n) * 2.0 / n
    pert = amplitude * np.sin(np.pi * freq * x + phase)
    return project_feasible(h + pert, n)

def slsqp_subspace(h, n, m_indices, deadline):
    h0 = h.copy()
    vals = h0[m_indices].copy()
    lo = np.zeros(len(m_indices))
    hi = np.ones(len(m_indices))
    offset = h0.sum() - vals.sum()
    
    def sub_obj(v):
        temp = h0.copy()
        temp[m_indices] = v
        temp = project_feasible(temp, n)
        return compute_c5(temp, n)
    
    try:
        res = minimize(sub_obj, vals, method='Nelder-Mead',
                       options={'maxiter': 200, 'xatol': 1e-10, 'fatol': 1e-12,
                                'maxfev': 300})
        h_new = h0.copy()
        h_new[m_indices] = res.x
        h_new = project_feasible(h_new, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return h0, compute_c5(h0, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n = 500

    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: Thorough coordinate descent
    if time.time() < deadline - 350:
        h1, c1 = coord_descent(best_h, n, 120, deadline - 350, gs_iters=60)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Phase 2: Block flip perturbations
    if time.time() < deadline - 250:
        for r in range(12):
            if time.time() > deadline - 250:
                break
            hp = block_flip_perturb(best_h, n, rng)
            h_o, c_o = coord_descent(hp, n, 50, deadline - 250, gs_iters=50)
            if c_o < best_c:
                best_h, best_c = h_o, c_o

    # Phase 3: Low-frequency Fourier perturbations
    if time.time() < deadline - 180:
        for freq in range(1, 7):
            if time.time() > deadline - 180:
                break
            for amp in [0.03, 0.07, 0.12]:
                if time.time() > deadline - 180:
                    break
                phase = rng.uniform(0, 2 * np.pi)
                hp = fourier_perturb(best_h, n, freq, amp, phase)
                h_o, c_o = coord_descent(hp, n, 45, deadline - 180, gs_iters=50)
                if c_o < best_c:
                    best_h, best_c = h_o, c_o

    # Phase 4: Gaussian perturbations
    if time.time() < deadline - 120:
        for r in range(8):
            if time.time() > deadline - 120:
                break
            scale = 0.015 + 0.025 * r / 8.0
            hp = project_feasible(best_h + rng.normal(0, scale, n), n)
            h_o, c_o = coord_descent(hp, n, 40, deadline - 120, gs_iters=50)
            if c_o < best_c:
                best_h, best_c = h_o, c_o

    # Phase 5: Try different n values
    if time.time() < deadline - 80:
        x_orig = np.linspace(0, 2, len(best_h))
        for n_try in [400, 550, 700, 850, 1000]:
            if time.time() > deadline - 80:
                break
            if n_try == len(best_h):
                continue
            x_new = np.linspace(0, 2, n_try)
            h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
            h_o, c_o = coord_descent(h_int, n_try, 60, deadline - 80, gs_iters=55)
            if c_o < best_c:
                best_h, best_c = h_o, c_o
                x_orig = np.linspace(0, 2, n_try)

    # Phase 6: Final CD polish with high accuracy
    if time.time() < deadline - 10:
        n_cur = len(best_h)
        h_f, c_f = coord_descent(best_h, n_cur, 40, deadline - 10, gs_iters=70)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
        # One more round
        if time.time() < deadline - 5:
            h_f2, c_f2 = coord_descent(best_h, n_cur, 20, deadline - 5, gs_iters=70)
            if c_f2 < best_c:
                best_h, best_c = h_f2, c_f2

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
