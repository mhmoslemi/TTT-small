def project_feasible(h, n):

def smooth_obj_grad(x, n, alpha):

    m = c.max()
    shifted = alpha * (c - m)

        _, g = smooth_obj_grad(x, n, alpha)
        return g

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    shifted -= shifted.max()
    S = np.exp(shifted).sum()
    return (m + np.log(S) / alpha) * (2.0 / n)

def smooth_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    shifted -= shifted.max()
    exp_s = np.exp(shifted)
    S = exp_s.sum()
    w = exp_s / S
    dx = 2.0 / n
    A = np.convolve(w, x, 'full')[:n]
    B = np.convolve(w, x[::-1], 'full')[n-1::-1][:n]
    return dx * (1.0 - A - B)

def cons_fun(x, n):
    return x.sum() - n / 2.0

def cons_jac(x, n):
    return np.ones(len(x))

def slsqp_opt(h0, n, alpha, maxiter, deadline):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_obj, h0, jac=smooth_grad, args=(n, alpha),
                       method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac, 'args': (n,)}],
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h0

def solve_coord(c, h, p, n):
    h0 = h[p]
    m_min = max(0, n - 1 - p)
    m_max = min(2 * n - 2, 2 * n - 2 - p)
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms]
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    unaff = 0.0
    if m_min > 0:
        unaff = float(np.max(c[:m_min]))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    cur_max = float(np.max(c))
    neg_mask = a < -1e-14
    pos_mask = a > 1e-14
    zero_mask = np.abs(a) <= 1e-14
    L_lo, L_hi = unaff, cur_max
    for _ in range(50):
        if L_hi - L_lo < 1e-15:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        feasible = True
        if np.any(zero_mask) and np.any(c_base[zero_mask] > L + 1e-14):
            feasible = False
        if feasible and np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if feasible and np.any(pos_mask):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        if x_lo > x_hi + 1e-12:
            feasible = False
        if feasible and has_diag:
            c_diag = c_base[diag_idx[0]]
            a_d = a[diag_idx[0]]
            disc = a_d * a_d + 4.0 * (c_diag - L)
            if disc >= 0:
                sq = np.sqrt(disc)
                r1 = (a_d - sq) / 2.0
                r2 = (a_d + sq) / 2.0
                if x_lo > h0 + r1 and x_hi < h0 + r2:
                    feasible = False
        if feasible:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo, x_hi = max(x_lo, 0.0), min(x_hi, 1.0)
    if has_diag:
        c_diag = c_base[diag_idx[0]]
        a_d = a[diag_idx[0]]
        disc = a_d * a_d + 4.0 * (c_diag - L)
        if disc >= 0:
            sq = np.sqrt(disc)
            r1 = (a_d - sq) / 2.0
            r2 = (a_d + sq) / 2.0
            if x_lo <= h0 + r1:
                x_hi = min(x_hi, h0 + r1)
            elif x_hi >= h0 + r2:
                x_lo = max(x_lo, h0 + r2)
            else:
                if abs(x_lo - (h0 + r1)) <= abs(x_hi - (h0 + r2)):
                    x_hi = min(x_hi, h0 + r1)
                else:
                    x_lo = max(x_lo, h0 + r2)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def cd_pass(c, h, n, order):
    improved = False
    for p in order:
        h0 = h[p]
        if h0 < 1e-14 or h0 > 1.0 - 1e-14:
            continue
        xn = solve_coord(c, h, p, n)
        if abs(xn - h0) > 1e-14:
            delta = xn - h0
            m_min = max(0, n - 1 - p)
            m_max = min(2 * n - 2, 2 * n - 2 - p)
            ms = np.arange(m_min, m_max + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(vf):
                a[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                a[vb] -= h[jb[vb]]
            c[ms] += delta * a
            diag_i = np.where(ms == (n - 1))[0]
            if len(diag_i) > 0:
                c[ms[diag_i[0]]] -= delta * delta
            h[p] = xn
            improved = True
    return improved

def exact_cd(h, n, max_passes, deadline, rng=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    seq = np.arange(n)
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        if rng is not None and rng.rand() < 0.3:
            order = rng.permutation(n)
        else:
            order = seq
        if not cd_pass(c, h, n, order):
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def interpolate_h(h_src, n_src, n_tgt):
    x_src = np.linspace(0, 2, n_src, endpoint=False)
    x_tgt = np.linspace(0, 2, n_tgt, endpoint=False)
    return project_feasible(np.interp(x_tgt, x_src, h_src), n_tgt)

def fourier_init(n, K, rng):
    x = np.linspace(0, 2, n, endpoint=False)
    h = np.zeros(n)
    for k in range(1, K + 1):
        amp = rng.normal(0, 0.8 / np.sqrt(k))
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(2 * np.pi * k * x + phase)
        amp2 = rng.normal(0, 0.8 / np.sqrt(k))
        phase2 = rng.uniform(0, 2 * np.pi)
        h += amp2 * np.sin(2 * np.pi * k * x + phase2)
    h = 0.5 + 0.25 * h
    return project_feasible(h, n)

def block_init(n, rng, nblocks):
    h = np.zeros(n)
    bs = n // nblocks
    for i in range(nblocks):
        val = rng.choice([0.0, 0.2, 0.5, 0.8, 1.0])
        h[i*bs:(i+1)*bs] = val
    if nblocks * bs < n:
        h[nblocks*bs:] = rng.uniform(0, 1)
    return project_feasible(h, n)

def refine(h, n, deadline, rng):

    for alpha in [3000, 10000, 50000, 200000, 1000000]:
