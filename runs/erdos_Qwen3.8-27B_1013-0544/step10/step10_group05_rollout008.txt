```python
import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def c_vec(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def cons_fun(x):
    return x.sum() - x.size / 2.0

def cons_jac(x):
    return np.ones(x.size)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    lse = mx + np.log(np.sum(np.exp(s - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_opt(h0, n, alphas, maxiter, deadline):
    best_h, best_c = h0.copy(), compute_c5(h0, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=[(0.0, 1.0)] * n, constraints=[{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}],
                           options={'maxiter': maxiter, 'ftol': 1e-15})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
        # re-project & small exact CD
        best_h, best_c = quick_cd(best_h, n, 15, deadline)
    return best_h, best_c

def wsd_pass(h, n, steps, deadline, lam=30.0, B=28):
    h = project_feasible(h.copy(), n)
    c = c_vec(h, n)
    best_h, best_c = h.copy(), c.max() * 2.0 / n
    step = 1.0 / n
    for s in range(steps):
        if time.time() > deadline: break
        cx = c.max()
        idx = np.argpartition(-c, B)[:B]
        idx = idx[c[idx] > cx - 0.6]   # near-max lags
        w = np.exp(lam * (c[idx] - cx)); w /= w.sum()
        g = np.zeros(n)
        for i, L in enumerate(idx):
            Lf = np.abs(int(L - (n - 1)))
            if Lf == 0:
                g += w[i] * (2.0 * h - 1.0)
            else:
                g[Lf:] += w[i] * (1.0 - h[:n - Lf])
                g[:n - Lf] -= w[i] * (1.0 - h[Lf:])
        g -= g.mean()
        h = project_feasible(h - step * g, n)
        c = c_vec(h, n)
        if c.max() < best_c:
            best_c, best_h = float(c.max()) * 2.0 / n, h.copy()
            step = max(step * 1.02, 0.5 / n)
        else:
            step *= 0.75
    return best_h, best_c

def smooth_jac(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    e = np.exp(s - mx)
    p = e / e.sum()
    # add mass to argmax
    p[np.where(np.isclose(c, m, atol=1e-12))] += 1.0
    top = p > 1e-10
    idx = np.where(top)[0]
    g = np.zeros(n)
    for Li in idx:
        w = p[Li]
        Lf = int(abs(Li - (n - 1)))
        if Lf == 0:
            g += w * (2.0 * x - 1.0)
        else:
            g[Lf:] += w * (1.0 - x[:n - Lf])
            g[:n - Lf] -= w * (1.0 - x[Lf:])
    return (2.0 / n) * g
```

def pair_cd(h, n, topB, topD, passes, deadline):
    h = project_feasible(h.copy(), n)
    c = c_vec(h, n)
    best_c = float(np.max(c)) * 2.0 / n
    best_h = h.copy()
    kk = np.linspace(0.0, 1.0, 9)
    for _ in range(passes):
        if time.time() > deadline:
            break
        c = c_vec(h, n)
        Ls = np.argpartition(-c, topB)[:topB]
        for Li in Ls:
            Lf = int(abs(Li - (n - 1)))
            if Lf == 0 or Lf >= max(2, n - 2):
                continue
            Lm = 2 * (n - 1) - Li
            c2 = c.copy(); c2[Li] = -1e9; c2[Lm] = -1e9
            rest = float(np.max(c2))
            cert = c.sum()
            a_band = h[:n - Lf]; b_band = h[Lf:n]

def gd_smooth(h, n, alpha, iters, deadline):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n); best_h = h.copy()
    step = 0.5 / n
    for _ in range(iters):
        if time.time() > deadline: break
        g = smooth_jac(h, n, alpha)
        gn = np.linalg.norm(g)
        if gn < 1e-15: break
        h = h - (step / gn) * g      # stefanich-ato: g/|g| × step
        h = project_feasible(h, n)
        cc = compute_c5(h, n)
        if cc < best_c:
            best_c, best_h = cc, h.copy()

def c_vec(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, n):
    return float(np.max(c_vec(h, n)) * (2.0 / n))

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = 0.5 * n
    for _ in range(40):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0.0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        m = (h > 1e-10) & (h < 1.0 - 1e-10)
        if m.any():
            h = h.copy()
            h[m] += diff / float(m.sum())
    return h

def smooth_obj(x, n, alpha):
    c = c_vec(x, n)
    m = float(np.max(c))
    s = alpha * (c - m)
    mx = float(np.max(s))
    e = np.exp(s - mx)
    return (m + (mx + np.log(e.sum())) / alpha) * (2.0 / n)

def smooth_jac(x, n, alpha):
    c = c_vec(x, n)
    m = float(np.max(c))
    s = alpha * (c - m)
    mx = float(np.max(s))
    p = np.exp(s - mx)
    p /= p.sum()
    p[np.isclose(c, m, rtol=1e-12, atol=1e-15)] += 1.0
    g = np.zeros(n)
    for j in np.where(p > 1e-9)[0]:
        w = p[j]
        Lf = int(abs(j - (n - 1)))
        if Lf == 0:
            g += w * (2.0 * x - 1.0)
        else:
            g[Lf:] += w * (1.0 - x[:n - Lf])
            g[:n - Lf] -= w * (1.0 - x[Lf:])
    return (2.0 / n) * g

def c_vec(h, n):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h, n):
    return float(np.max(c_vec(h, n)) * (2.0 / n))


def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = 0.5 * n
    for _ in range(60):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0.0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        m = (h > 1e-10) & (h < 1.0 - 1e-10)
        if m.any():
            h[m] += diff / float(m.sum())
    return h


def sym_point(h, n):
    return project_feasible(0.5 * (h + h[::-1].copy()), n)


def smooth_obj(x, n, alpha):
    c = c_vec(x, n)
    m = float(np.max(c))
    s = alpha * (c - m)
    mx = float(np.max(s))
    e = np.exp(s - mx)
    return (m + (mx + np.log(e.sum())) / alpha) * (2.0 / n)


def smooth_jac(x, n, alpha):
    c = c_vec(x, n)
    m = float(np.max(c))
    s = alpha * (c - m)
    mx = float(np.max(s))
    p = np.exp(s - mx)
    p /= p.sum()
    p[np.isclose(c, m, rtol=1e-12, atol=1e-15)] += 1.0
    g = np.zeros(n)
    for j in np.where(p > 1e-9)[0]:
        w = p[j]
        Lf = int(abs(j - (n - 1)))
        if Lf == 0:
            g += w * (2.0 * x - 1.0)
        else:
            g[Lf:] += w * (1.0 - x[:n - Lf])
            g[:n - Lf] -= w * (1.0 - x[Lf:])
    return (2.0 / n) * g


def gd_smooth(h, n, alpha, iters, deadline):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    step = 2.0e-3
    for _ in range(iters):
        if time.time() > deadline:
            break
        g = smooth_jac(h, n, alpha)
        gn = float(np.linalg.norm(g))
        if gn < 1e-16:
            break
        h = project_feasible(h - (step / gn) * g, n)
        cc = compute_c5(h, n)
        if cc < best_c:
            best_c, best_h = cc, h.copy()
            step = min(step * 1.12, 9.0e-3)
        else:
            step = max(step * 0.62, 3.0e-4)
    return best_h, best_c


def t_update(c, h, t, d, n):
    if t > 0:
        c[n:n + t] += d * (1.0 - h[t - 1::-1])
    if n - 1 - t > 0:
        c[t:n - 1] += d * (1.0 - h[t + 1:n])
    c[n - 1] += d * (1.0 - 2.0 * h[t]) - d * d


def solve_coord(c, h, p, n):
    h0 = h[p]
    m0 = max(0, n - 1 - p)
    m1 = min(2 * n - 2, 2 * n - 2 - p)
    ms = np.arange(m0, m1 + 1)
    a = np.zeros(len(ms))
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    if vf.any():
        a[vf] += 1.0 - h[jf[vf]]
    if vb.any():
        a[vb] -= h[jb[vb]]
    c_base = c[ms]
    zero = np.abs(a) <= 1e-14
    neg = a < -1e-14
    pos = a > 1e-14
    zmax = float(np.max(c_base[zero])) if zero.any() else 0.0
    diag = (m0 <= n - 1) and (n - 1 <= m1)
    if diag:
        idq = n - 1 - m0
        ad, cd0 = float(a[idq]), float(c_base[idq])
    else:
        ad, cd0 = 0.0, 0.0
    unaff = 0.0
    if m0 > 0:
        unaff = max(unaff, float(np.max(c[:m0])))
    if m1 < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m1 + 1:])))
    L_lo, L_hi = max(unaff, zmax), float(np.max(c))
    for _ in range(34):
        if L_hi - L_lo < 1e-13:
            break
        L = 0.5 * (L_lo + L_hi)
        x_lo, x_hi = 0.0, 1.0
        ok = True
        if neg.any():
            x_lo = max(x_lo, float(np.max((L - c_base[neg]) / a[neg])))
        if pos.any():
            x_hi = min(x_hi, float(np.min((L - c_base[pos]) / a[pos])))
        if x_lo > x_hi + 1e-13:
            ok = False
        if ok and diag:
            disc = ad * ad + 4.0 * (cd0 - L)
            if disc >= 0.0:
                sq = float(np.sqrt(disc))
                r1 = h0 + 0.5 * (ad - sq)
                r2 = h0 + 0.5 * (ad + sq)
                if x_lo > r1 and x_hi < r2:
                    ok = False
        if ok:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if neg.any():
        x_lo = max(x_lo, float(np.max((L - c_base[neg]) / a[neg])))
    if pos.any():
        x_hi = min(x_hi, float(np.min((L - c_base[pos]) / a[pos])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if diag:
        disc = max(0.0, ad * ad + 4.0 * (cd0 - L))
        sq = float(np.sqrt(disc))
        r1 = h0 + 0.5 * (ad - sq)
        r2 = h0 + 0.5 * (ad + sq)
        if x_lo <= r1:
            x_hi = min(x_hi, r1)
        elif x_hi >= r2:
            x_lo = max(x_lo, r2)
        elif abs(x_lo - r1) <= abs(x_hi - r2):
            x_hi = min(x_hi, r1)
        else:
            x_lo = max(x_lo, r2)
    if x_lo > x_hi + 1e-12:
        return h0
    return float(np.clip(0.5 * (x_lo + x_hi), 0.0, 1.0))


def cd_core(h, n, passes, deadline, rng, shuffled):
    h = project_feasible(h.copy(), n)
    c = c_vec(h, n)
    best_c = float(np.max(c)) * (2.0 / n)
    best_h = h.copy()
    for _ in range(passes):
        if time.time() > deadline:
            break
        cur = float(np.max(c))
        moved = False
        order = (rng.permutation(n) if shuffled else np.arange(n))
        for p in order:
            h0 = h[p]
            if h0 <= 1e-13 or h0 >= 1.0 - 1e-13:
                continue
            xn = solve_coord(c, h, p, n)
            d = xn - h0
            if abs(d) <= 1e-14:
                continue
            t_update(c, h, p, d, n)
            h[p] = xn
            cur_m = float(np.max(c))
            if cur_m < cur:
                cur = cur_m
            moved = True
        h = project_feasible(h, n)
        if abs(h.sum() - 0.5 * n) > 1e-9:
            c = c_vec(h, n)
            cur = float(np.max(c))
        cc = cur * (2.0 / n)
        if cc < best_c:
            best_c, best_h = cc, h.copy()
        if not moved:
            break
    return best_h, best_c


def pair_cd(h, n, topB, topD, passes, deadline):
    h = project_feasible(h.copy(), n)
    c = c_vec(h, n)
    best_c = float(np.max(c)) * (2.0 / n)
    best_h = h.copy()
    kk = np.linspace(0.0, 1.0, 9)
    nb = min(topB, 2 * n - 2)
    for _ in range(passes):
        if time.time() > deadline:
            break
        c = c_vec(h, n)
        cur = float(np.max(c))
        Ls = np.argpartition(-c, nb)[:nb]
        for Li in Ls:
            Lf = int(abs(Li - (n - 1)))
            if Lf == 0 or n - Lf < 6:
                continue
            Lm = 2 * (n - 1) - Li
            old = c.copy()
            c[Li] = -1e9
            c[Lm] = -1e9
            rest = float(np.max(c))
            c = old
            val = h[:n - Lf] * (1.0 - h[Lf:])
            for m in np.argsort(-val)[:topD]:
                m = int(m)
                u0, v0 = float(h[m]), float(h[m + Lf])
                if u0 <= 1e-12 or v0 <= 1e-12:
                    continue
                cL0, cN0 = float(c[Li]), float(c[Lm])
                bu, bv, bs = -1.0, -1.0, 1e18
                for u in kk:
                    ho = 1.0 - u
                    for v in kk:
                        CL = cL0 - v0 + v0 * u + v - v * u
                        CN = cN0 - u0 + u0 * v + u - u * v
                        o = rest if rest > CL else (CL if CL > CN else CN)
                        if o < bs:
                            bs, bu, bv = o, u, v
                if bu < 0.0:
                    continue
                d1, d2 = bu - u0, bv - v0
                if abs(d1) < 1e-14 and abs(d2) < 1e-14:
                    continue
                pre = c.copy()
                if d1 != 0.0:
                    t_update(c, h, m, d1, n)
                    h[m] = bu
                if d2 != 0.0:
                    t_update(c, h, m + Lf, d2, n)
                    h[m + Lf] = bv
                nm = float(np.max(c))
                if nm < cur - 1e-12:
                    h = project_feasible(h, n)
                    c = c_vec(h, n)
                    nm = float(np.max(c))
                    if nm < cur:
                        cur = nm
                        if nm * (2.0 / n) < best_c:
                            best_c = nm * (2.0 / n)
                            best_h = h.copy()
                else:
                    c = pre
                    h[m] = u0
                    h[m + Lf] = v0
    return best_h, best_c


def wsd_pass(h, n, steps, deadline, B):
    h = project_feasible(h.copy(), n)
    c = c_vec(h, n)
    best_c = float(np.max(c)) * (2.0 / n)
    best_h = h.copy()
    step = 1.5e-3
    nb = min(B, 2 * n - 2)
    for _ in range(steps):
        if time.time() > deadline:
            break
        cm = float(np.max(c))
        idx = np.argpartition(-c, nb)[:nb]
        w = np.exp(30.0 * (c[idx] - cm))
        w /= w.sum()
        g = np.zeros(n)
        for i in range(len(idx)):
            Lf = int(abs(idx[i] - (n - 1)))
            if Lf == 0:
                g += w[i] * (2.0 * h - 1.0)
            else:
                g[Lf:] += w[i] * (1.0 - h[:n - Lf])
                g[:n - Lf] -= w[i] * (1.0 - h[Lf:])
        g -= g.mean()
        h = project_feasible(h - step * g, n)
        c = c_vec(h, n)
        cc = float(np.max(c)) * (2.0 / n)
        if cc < best_c:
            best_c, best_h = cc, h.copy()
            step = min(step * 1.08, 6.0e-3)
        else:
            step = max(step * 0.7, 2.0e-4)
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + max(120.0, float(budget_s)) - 20.0
    rng = np.random.RandomState(seed)
    try:
        h = np.asarray(initial_h_values, dtype=float).copy()
        n = int(h.size)
    except Exception:
        h = np.full(600, 0.5)
        n = 600
    h = sym_point(h, n)
    best_h, best_c = h.copy(), compute_c5(h, n)

    hh, cc = gd_smooth(best_h, n, 4.0e3, 60, deadline)
    if cc < best_c:
        best_h, best_c = hh, cc
    hh, cc = cd_core(best_h, n, 35, deadline, rng, False)
    if cc < best_c:
        best_h, best_c = hh, cc
    hh, cc = gd_smooth(best_h, n, 2.5e4, 140, deadline)
    if cc < best_c:
        best_h, best_c = hh, cc
    hh, cc = cd_core(best_h, n, 50, deadline, rng, False)
    if cc < best_c:
        best_h, best_c = hh, cc
    hh, cc = wsd_pass(best_h, n, 50, deadline, 25)
    if cc < best_c:
        best_h, best_c = hh, cc
    hh, cc = cd_core(best_h, n, 55, deadline, rng, False)
    if cc < best_c:
        best_h, best_c = hh, cc
    hh, cc = pair_cd(best_h, n, 14, 5, 5, deadline)
    if cc < best_c:
        best_h, best_c = hh, cc
    hh, cc = cd_core(best_h, n, 60, deadline, rng, True)
    if cc < best_c:
        best_h, best_c = hh, cc

    for temp in (0.1, 0.2):
        if time.time() > deadline - 150:
            break
        nz = sym_point(project_feasible(best_h + rng.normal(0.0, temp, best_h.size), best_h.size), best_h.size)
        hh, cc = gd_smooth(nz, nz.size, 1.2e4, 90, deadline)
        hh, cc = cd_core(hh, nz.size, 30, deadline, rng, True)
        hh, cc = wsd_pass(hh, nz.size, 25, deadline, 25)
        hh, cc = cd_core(hh, nz.size, 25, deadline, rng, False)
        if cc < best_c:
            best_h, best_c = hh, cc

    if time.time() < deadline - 110:
        n2 = 500
        xs = np.linspace(0.0, 2.0, best_h.size, endpoint=False)
        xt = np.linspace(0.0, 2.0, n2, endpoint=False)
        h2 = sym_point(project_feasible(np.interp(xt, xs, best_h), n2), n2)
        hh, cc = gd_smooth(h2, n2, 2.0e4, 60, deadline)
        hh, cc = cd_core(hh, n2, 25, deadline, rng, False)
        hh, cc = wsd_pass(hh, n2, 15, deadline, 25)
        if cc < best_c - 1e-7:
            best_h, best_c = hh, cc

    best_n = int(best_h.size)
    hh, cc = cd_core(best_h, best_n, 60, deadline, rng, False)
    if cc < best_c:
        best_h, best_c = hh, cc
    hh, cc = pair_cd(best_h, best_n, 14, 5, 3, deadline)
    if cc < best_c:
        best_h, best_c = hh, cc
    hs = sym_point(best_h, best_n)
    cs = compute_c5(hs, best_n)
    if cs <= best_c + 1e-12:
        best_h, best_c = hs, min(cs, best_c)
    if time.time() < deadline - 5:
        hh, cc = cd_core(best_h, best_n, 30, deadline, rng, True)
        if cc < best_c:
            best_h, best_c = hh, cc

    best_h = project_feasible(best_h, int(best_h.size))
    best_c = compute_c5(best_h, int(best_h.size))
    return best_h, float(best_c), int(best_h.size)
