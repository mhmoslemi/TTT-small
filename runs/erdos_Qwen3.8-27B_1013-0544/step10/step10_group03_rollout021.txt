def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def solve_1d(c, h, p, n):
    h0 = h[p]
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms]
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    if has_diag:
        a[diag_idx] = 0.0
    unaff = 0.0
    if m_min > 0:
        unaff = float(np.max(c[:m_min]))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    cur_max = float(np.max(c))
    L_lo, L_hi = unaff, cur_max
    c_diag = c_base[diag_idx[0]] if has_diag else 0.0
    for _ in range(55):
        if L_hi - L_lo < 1e-16:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        feasible = True
        neg_mask = a < -1e-15
        pos_mask = a > 1e-15
        zero_mask = np.abs(a) <= 1e-15
        if np.any(zero_mask) and np.any(c_base[zero_mask] > L):
            feasible = False
        if feasible and np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if feasible and np.any(pos_mask):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        if x_lo > x_hi + 1e-14:
            feasible = False
        if feasible and has_diag:
            Cq = L - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                xc_lo, xc_hi = max(x_lo, 0.0), min(x_hi, 1.0)
                if not (xc_lo < r1 and xc_hi > r2):
                    feasible = False
        if feasible:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    neg_mask = a < -1e-15
    pos_mask = a > 1e-15
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Cq = L - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            if h0 - r1 <= r2 - h0:
                x_hi = min(x_hi, r1)
            else:
                x_lo = max(x_lo, r2)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def update_c(c, h, p, old_val, new_val, n):
    delta = new_val - old_val
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    diag_i = np.where(ms == (n - 1))[0]
    c[ms] += delta * a
    if len(diag_i) > 0:
        c[ms[diag_i[0]]] -= delta * delta

def exact_cd(h, n, deadline, rng):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    best_max = float(np.max(c))
    tol = 1e-13
    stagnation = 0
    for it in range(100000):
        if time.time() > deadline:
            break
        improved = False
        order = rng.permutation(n)
        for p in order:
            h0 = h[p]
            if h0 < tol or h0 > 1.0 - tol:
                continue
            xn = solve_1d(c, h, p, n)
            if abs(xn - h0) > tol:
                h[p] = xn
                update_c(c, h, p, h0, xn, n)
                nm = float(np.max(c))
                if nm < best_max:
                    best_max = nm
                improved = True
        if not improved:
            stagnation += 1
            if stagnation >= 3:
                break
        else:
            stagnation = 0
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-13:
            c = np.correlate(h, 1.0 - h, mode='full')
            best_max = float(np.max(c))
    return h, best_max * (2.0 / n)

def binary_push(h, n, strength):
    h = h.copy()
    mask = h < 0.5
    h[mask] = np.maximum(0.0, h[mask] - strength)
    h[~mask] = np.minimum(1.0, h[~mask] + strength)
    return project_feasible(h, n)

def local_perturb(h, n, rng, scale):
    idx = rng.choice(n, size=max(1, n // 10), replace=False)
    h = h.copy()
    h[idx] += rng.normal(0, scale, len(idx))
    return project_feasible(h, n)

    strengths = [0.1, 0.2, 0.35, 0.5, 0.7]

        if co < best_c - 1e-10:

    # Phase 3: Local perturbation + CD (explore nearby basins)
    p3_end = t0 + budget_s * 0.92
    scales = [0.01, 0.02, 0.03, 0.05, 0.08, 0.12]
    for sc in scales:
        if time.time() > p3_end - 25:
            break
        for attempt in range(2):

    return h_best, best_c, n_best
