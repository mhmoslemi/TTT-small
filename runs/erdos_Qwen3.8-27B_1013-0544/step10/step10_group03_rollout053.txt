        Cq = L - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            if h0 - r1 <= r2 - h0:
                x_hi = min(x_hi, r1)
            else:
                x_lo = max(x_lo, r2)
    if x_lo > x_hi:
        return h0
    if mode == 'close':
        dl = abs(h0 - x_lo)
        dh = abs(h0 - x_hi)
        return float(np.clip(x_lo if dl <= dh else x_hi, 0.0, 1.0))
    if mode == 'low':
        return float(np.clip(x_lo, 0.0, 1.0))
    if mode == 'high':
        return float(np.clip(x_hi, 0.0, 1.0))
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def exact_cd(h, n, max_passes, deadline, order=None, mode='mid'):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        if order is None:
            seq = range(n)
        elif order == 'rev':
            seq = range(n - 1, -1, -1)
        else:
            seq = order
        for p in seq:
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            xn = solve_1d_bisect(c, h, p, n, mode=mode)
            if abs(xn - h0) > 1e-12:
                delta = xn - h0
                m_min = n - 1 - p
                m_max = 2 * n - 2 - p
                ms = np.arange(m_min, m_max + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                diag_i = np.where(ms == (n - 1))[0]
                c[ms] += delta * a
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-12:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_opt(h, n, alphas, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
                 'jac': lambda x: np.ones(len(x))}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def value_sorted_order(h, n):
    return list(np.argsort(h))

def reverse_value_order(h, n):
    return list(np.argsort(h)[::-1])

def targeted_perturb(h, n, rng, strength):
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = c.max()
    active_lags = c > cmax - 0.05 * cmax
    active_indices = np.zeros(n, dtype=bool)
    for lag in np.where(active_lags)[0]:
        k = lag - (n - 1)
        i_start = max(0, -k)
        i_end = min(n, n - k)
        if i_start < i_end:
            active_indices[i_start:i_end] = True
    n_active = active_indices.sum()
    if n_active == 0:
        active_indices[:] = True
        n_active = n
    idx = np.where(active_indices)[0]
    n_pick = max(1, int(n_active * 0.3))
    chosen = rng.choice(idx, size=min(n_pick, len(idx)), replace=False)
    hp = h.copy()
    hp[chosen] += rng.normal(0, strength, len(chosen))
    return project_feasible(hp, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = np.full(700, 0.5)
        n = 700
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)

    # Phase 1: Quick CD forward
    if time.time() < deadline - 500:
        h1, c1 = exact_cd(h_best, n, 100, deadline - 500, mode='mid')
        if c1 < best_c:
            h_best, best_c = h1, c1

    # Phase 2: CD reverse order
    if time.time() < deadline - 450:
        h2, c2 = exact_cd(h_best, n, 200, deadline - 450, order='rev', mode='mid')
        if c2 < best_c:
            h_best, best_c = h2, c2

    # Phase 3: CD value-sorted (low to high)
    if time.time() < deadline - 400:
        vo = value_sorted_order(h_best, n)
        h3, c3 = exact_cd(h_best, n, 200, deadline - 400, order=vo, mode='close')
        if c3 < best_c:
            h_best, best_c = h3, c3

    # Phase 4: CD value-sorted (high to low)
    if time.time() < deadline - 350:
        rvo = reverse_value_order(h_best, n)
        h4, c4 = exact_cd(h_best, n, 200, deadline - 350, order=rvo, mode='close')
        if c4 < best_c:
            h_best, best_c = h4, c4

    # Phase 5: CD with 'low' mode
    if time.time() < deadline - 300:
        h5, c5 = exact_cd(h_best, n, 200, deadline - 300, mode='low')
        if c5 < best_c:
            h_best, best_c = h5, c5

    # Phase 6: SLSQP with diverse alphas
    if time.time() < deadline - 250:
        h6, c6 = slsqp_opt(h_best, n, [5000, 20000, 100000, 500000], 400, deadline - 250)
        if c6 < best_c:
            h_best, best_c = h6, c6

    # Phase 7: Alternating CD with different modes
    for cycle in range(8):
        if time.time() > deadline - 200:
            break
        modes = ['close', 'mid', 'low', 'high']
        m = modes[cycle % 4]
        o = 'rev' if cycle % 2 == 0 else None
        hc, cc = exact_cd(h_best, n, 150, deadline - 200, order=o, mode=m)
        if cc < best_c:
            h_best, best_c = hc, cc

    # Phase 8: Targeted perturbation + CD
    for r in range(12):
        if time.time() > deadline - 150:
            break
        s = 0.001 + 0.004 * r
        if r % 2 == 0:
            hp = targeted_perturb(h_best, n, rng, s)
        else:
            hp = project_feasible(h_best + rng.normal(0, s, n), n)
        h_o, c_o = exact_cd(hp, n, 200, deadline - 150, mode='close')
        if c_o < best_c:
            h_best, best_c = h_o, c_o
        if time.time() > deadline - 200:
            break
        h_s, c_s = slsqp_opt(h_o, n, [30000, 200000], 150, deadline - 200)
        if c_s < best_c:
            h_best, best_c = h_s, c_s

    # Phase 9: Mixed CD cycles (alternate forward/reverse with different modes)
    if time.time() < deadline - 100:
        for i in range(6):
            if time.time() > deadline - 100:
                break
            o = 'rev' if i % 2 else None
            m = 'close' if i % 3 == 0 else 'mid'
            hc, cc = exact_cd(h_best, n, 100, deadline - 100, order=o, mode=m)
            if cc < best_c:
                h_best, best_c = hc, cc

    # Phase 10: Final intensive CD
    if time.time() < deadline - 3:
        n = len(h_best)
        h_f, c_f = exact_cd(h_best, n, 300, deadline, mode='close')
        if c_f < best_c:
            h_best, best_c = h_f, c_f
        h_f2, c_f2 = exact_cd(h_best, n, 200, deadline, order='rev', mode='mid')
        if c_f2 < best_c:
            h_best, best_c = h_f2, c_f2

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
