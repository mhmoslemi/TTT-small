    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            xn = solve_coord(c, h, p, n)
            if abs(xn - h0) > 1e-13:
                delta = xn - h0
                m_min = max(0, n - 1 - p)
                m_max = min(2 * n - 2, 2 * n - 2 - p)
                ms = np.arange(m_min, m_max + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                c[ms] += delta * a
                diag_i = np.where(ms == (n - 1))[0]
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def slsqp_opt(h0, n, params, maxiter, deadline):
    best_h, best_c = h0.copy(), compute_c5(h0, n)
    for alpha, beta in params:
        if time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
        try:
            res = minimize(lambda x: smooth_obj(x, n, alpha, beta), best_h,
                         method='SLSQP', bounds=bounds, constraints=cons,
                         options={'maxiter': maxiter, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def binary_restart(h, n, rng, deadline):
    h_bin = (h > 0.5).astype(float)
    target = n // 2
    num_ones = int(h_bin.sum())
    if num_ones > target:
        ones = np.where(h_bin == 1)[0]
        to_flip = rng.choice(ones, num_ones - target, replace=False)
        h_bin[to_flip] = 0
    elif num_ones < target:
        zeros = np.where(h_bin == 0)[0]
        to_flip = rng.choice(zeros, target - num_ones, replace=False)
        h_bin[to_flip] = 1
    for _ in range(30):
        p = rng.randint(n)
        q = rng.randint(n)
        if p == q:
            continue
        if h_bin[p] == h_bin[q]:
            continue
        h_bin[p], h_bin[q] = h_bin[q], h_bin[p]
    return project_feasible(h_bin + rng.normal(0, 0.01, n), n)

def interpolate_h(h_src, n_src, n_tgt):
    x_src = np.linspace(0, 2, n_src, endpoint=False)
    x_tgt = np.linspace(0, 2, n_tgt, endpoint=False)
    return project_feasible(np.interp(x_tgt, x_src, h_src), n_tgt)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = np.full(600, 0.5)
        n = 600
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)

    # Phase 1: SLSQP with step penalty (key new element)
    params1 = [(3000, 0.0), (10000, 0.001), (30000, 0.005), (100000, 0.01), (300000, 0.05)]
    h1, c1 = slsqp_opt(h_best, n, params1, 300, deadline)
    if c1 < best_c:
        h_best, best_c = h1, c1

    # Phase 2: Nesterov projected gradient with step penalty
    for alpha, beta in [(50000, 0.01), (200000, 0.0), (500000, 0.05)]:
        if time.time() > deadline - 60:
            break
        h2, c2 = nesterov_projected(h_best, n, alpha, beta, 5000, deadline)
        if c2 < best_c:
            h_best, best_c = h2, c2

    # Phase 3: Exact CD for precise convergence
    h3, c3 = exact_cd(h_best, n, 300, deadline)
    if c3 < best_c:
        h_best, best_c = h3, c3

    # Phase 4: Binary restarts + refinement
    for r in range(8):
        if time.time() > deadline - 120:
            break
        h_pb = binary_restart(h_best, n, rng, deadline)
        h4, c4 = slsqp_opt(h_pb, n, [(50000, 0.01), (200000, 0.0)], 200, deadline)
        h4, c4 = exact_cd(h4, n, 200, deadline)
        if c4 < best_c:
            h_best, best_c = h4, c4

    # Phase 5: Nesterov with different LR schedules
    for lr_mult in [0.5, 2.0, 5.0]:
        if time.time() > deadline - 90:
            break
        alpha_g = 100000
        lr = 2.0 / (alpha_g * (2 * n - 1)) * lr_mult
        h5, c5 = nesterov_projected(h_best, n, alpha_g, 0.0, 3000, deadline, lr=lr)
        if c5 < best_c:
            h_best, best_c = h5, c5

    # Phase 6: Different resolution
    for n_try in [n + 20, n - 20, 800"]:
        if time.time() > deadline - 100:
            break
        if n_try <= 100 or n_try == n or n_try > 1000:
            continue
        h_int = interpolate_h(h_best, n, n_try)
        h6, c6 = slsqp_opt(h_int, n_try, [(100000, 0.01)], 150, deadline)
        h6, c6 = exact_cd(h6, n_try, 200, deadline)
        if c6 < best_c - 1e-7:
            h_best, best_c = h6, c6
            n = n_try

    # Phase 7: Final interleaved refinement
    n = len(h_best)
    for _ in range(3):
        if time.time() > deadline - 20:
            break
        h7, c7 = slsqp_opt(h_best, n, [(500000, 0.01), (2000000, 0.0)], 150, deadline)
        h7, c7 = exact_cd(h7, n, 200, deadline)
        if c7 < best_c:
            h_best, best_c = h7, c7

    # Phase 8: Final CD polish
    if time.time() < deadline - 2:
        hf, cf = exact_cd(h_best, len(h_best), 200, deadline)
        if cf < best_c:
            h_best, best_c = hf, cf

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
