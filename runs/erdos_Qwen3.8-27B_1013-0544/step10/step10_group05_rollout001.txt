def solve_coord(c,h,p,n):
    h0=h[p]
    n1=n-1
    mmin=max(0,n1-p); mmax=min(2*n1,2*n1-p)
    ms=np.arange(mmin,mmax+1)
    jf=p+ms-n1; jb=p-ms+n1
    vf=(jf>=0)&(jf<n); vb=(jb>=0)&(jb<n)
    a=np.zeros(len(ms))
    if np.any(vf): a[vf]+=1.0-h[jf[vf]]
    if np.any(vb): a[vb]-=h[jb[vb]]
    cb=c[ms]
    di=np.where(ms==n1)[0]
    has=di.size>0
    unaff=0.0
    if mmin>0: unaff=float(np.max(c[:mmin]))
    if mmax<2*n1: unaff=max(unaff,float(np.max(c[mmax+1:])))
    cur=float(np.max(c))
    neg=a<-1e-14; pos=a>1e-14; zero=np.abs(a)<=1e-14
    Llo, Lhi=unaff, cur
    for _ in range(35):
        if Lhi-Llo<1e-13: break
        L=(Llo+Lhi)*0.5
        xlo, xhi=0.0,1.0
        ok=True
        if np.any(zero) and np.any(cb[zero]>L+1e-13): ok=False
        if ok and np.any(neg): xlo=max(xlo,float(np.max((L-cb[neg])/a[neg])))
        if ok and np.any(pos): xhi=min(xhi,float(np.min((L-cb[pos])/a[pos])))
        if xlo>xhi+1e-12: ok=False
        if ok and has:
            cd=cb[di[0]]; ad=a[di[0]]
            disc=ad*ad+4.0*(cd-L)
            if disc>=0.0:
                sq=np.sqrt(disc); r1=h0+(ad-sq)/2.0; r2=h0+(ad+sq)/2.0
                if xlo>r1+1e-13 and xhi<r2-1e-13: ok=False
        if ok: Lhi=L
        else: Llo=L
    L=Lhi
    xlo,xhi=0.0,1.0
    if np.any(neg): xlo=max(xlo,float(np.max((L-cb[neg])/a[neg])))
    if np.any(pos): xhi=min(xhi,float(np.min((L-cb[pos])/a[pos])))
    xlo=max(xlo,0.0); xhi=min(xhi,1.0)
    if xlo>xhi:
        return h0, ms, a
    if has:
        cd=cb[di[0]]; ad=a[di[0]]
        disc=ad*ad+4.0*(cd-L)
        if disc>=0.0:
            sq=np.sqrt(disc); r1=h0+(ad-sq)/2.0; r2=h0+(ad+sq)/2.0
            bestx=None; bestd=1e100

                    if abs(cand-h0)<bestd: bestd=abs(cand-h0); bestx=cand
            if xhi>=r2:

                if lo<=hi+1e-13:
                    cand=min(max(h0,lo),hi)
                    if abs(cand-h0)<bestd: bestd=abs(cand-h0); bestx=cand
            if bestx is None: return h0,ms,a
            return float(np.clip(bestx,0.0,1.0)), ms, a
    # no diag
    cand=min(max(h0,xlo),xhi)
    return float(np.clip(cand,0.0,1.0)), ms, a

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))

def project(h, n):
    h = np.clip(np.asarray(h, dtype=float).copy(), 0.0, 1.0)
    t = 0.5 * n
    for _ in range(40):
        d = t - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0.0:
            r = 1.0 - h
            s = r.sum()
            if s < 1e-15:
                break
            h = np.clip(h + d * r / s, 0.0, 1.0)
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = np.clip(h + d * h / s, 0.0, 1.0)
    for _ in range(12):
        d = t - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0.0:
            idx = np.where(h < 1.0 - 1e-14)[0]
        else:
            idx = np.where(h > 1e-14)[0]
        if idx.size == 0:
            break
        h[idx] += d / idx.size
        h[idx] = np.clip(h[idx], 0.0, 1.0)
    return h

def update_best(h, n, best_h, best_c, best_n):
    h = project(h, n)
    c = compute_c5(h)
    if c < best_c - 1e-12 or (abs(c - best_c) <= 1e-12 and (best_n == 0 or n < best_n)):
        return h.copy(), c, len(h)
    return best_h, best_c, best_n

def interpolate_h(h0, n0, n1):
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
    x1 = np.linspace(0.0, 2.0, n1, endpoint=False)
    return project(np.interp(x1, x0, h0), n1)

def symmetrize(h):
    return project(0.5 * (h + h[::-1]), len(h))

def mode_init(n, seed, modes, sc):
    rs = np.random.RandomState(seed)
    i = np.arange(n)
    f = np.zeros(n)
    for m in modes:
        f += rs.normal(0.0, 0.9) * np.cos(2.0 * np.pi * m * i / n + rs.uniform(0.0, 2.0 * np.pi))
    f = np.tanh(sc * f)
    return project(0.5 + 0.5 * f, n)

def block_init(n, rng, B):
    idx = (np.arange(n) * B / float(n)).astype(int)
    idx = np.clip(idx, 0, B - 1)
    b = 0.5 + 0.4 * np.cos(2.0 * np.pi * np.arange(B) / max(1, B)) + rng.normal(0.0, 0.08, B)
    return project(b[idx], n)

def subgrad_grad(h, top, w):
    n = len(h)
    g = np.zeros(n)
    nm = n - 1
    for ii in range(len(top)):
        lag = int(top[ii]) - nm
        ww = float(w[ii])
        if ww <= 1e-18:
            continue
        if lag >= 0:
            if lag == 0:
                g += ww * (1.0 - 2.0 * h)
            else:
                g[:n - lag] += ww * (1.0 - h[lag:])
                g[lag:] -= ww * h[:-lag]
        else:
            l = -lag
            g[l:] += ww * (1.0 - h[:-l])
            g[:-l] -= ww * h[l:]
    return g

def subgrad_opt(h0, deadline, rng):
    n = len(h0)
    h = project(h0, n)
    best_h = h.copy()
    best_c = compute_c5(h)
    K = max(16, min(80, n // 4))
    no = 0
    it = 0
    while time.time() < deadline and it < 6000:
        cv = np.correlate(h, 1.0 - h, mode="full")
        mx = float(cv.max())
        cur = mx * (2.0 / n)
        if cur < best_c:
            best_c = cur
            best_h = h.copy()
            no = 0
        if mx <= 1e-14:
            break
        alphas = (5.0, 10.0, 20.0, 40.0, 80.0)
        alpha = mx / alphas[it % len(alphas)]
        if K < cv.size:
            top = np.argpartition(cv, -K)[-K:]
        else:
            top = np.arange(cv.size)
        sh = (cv[top] - mx) / alpha
        w = np.exp(sh)
        s = w.sum()
        if s < 1e-300:
            w = np.full_like(w, 1.0 / len(w))
        else:
            w /= s
        g = subgrad_grad(h, top, w)
        gn = float(np.max(np.abs(g)))
        if gn < 1e-14:
            h = project(h + rng.normal(0.0, 0.05, n), n)
            no += 1
            it += 1
            continue

        cand_c = cur

    return best_h, best_c

def solve_coord(c, h, p, n):

    nm = n - 1
    m0 = max(0, nm - p)
    m1 = min(2 * nm, 2 * nm - p)
    ms = np.arange(m0, m1 + 1)

    jb = p - ms + nm
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += 1.0 - h[jf[vf]]
    if np.any(vb):

    cb = c[ms]

    if m1 < 2 * nm:

    neg = a < -1e-14

    for _ in range(34):

            break

        xlo, xhi = 0.0, 1.0
        ok = True
        if np.any(zero) and np.any(cb[zero] > L + 1e-13):
            ok = False
        if ok and np.any(neg):

        else:
            Llo = L
    L = Lhi
    xlo, xhi = 0.0, 1.0
    if np.any(neg):
        xlo = max(xlo, float(np.max((L - cb[neg]) / a[neg])))
    if np.any(pos):
        xhi = min(xhi, float(np.min((L - cb[pos]) / a[pos])))
    xlo = max(0.0, xlo)
    xhi = min(1.0, xhi)
    if xlo > xhi:
        return h0, ms, a
    if has:
        cd = cb[di[0]]
        ad = a[di[0]]
        disc = ad * ad + 4.0 * (cd - L)
        if disc >= 0.0:
            sq = np.sqrt(disc)
            r1 = h0 + (ad - sq) / 2.0

                lo = max(0.0, xlo)
                hi = min(1.0, xhi, r1)
                if lo <= hi + 1e-13:

                    d0 = abs(cand - h0)
                    if d0 < bd:

            if xhi >= r2:

                hi = min(1.0, xhi)

            return float(np.clip(bx, 0.0, 1.0)), ms, a
    cand = min(max(h0, xlo), xhi)
    return float(np.clip(cand, 0.0, 1.0)), ms, a

def coordinate_polish(h0, max_passes, deadline, rng=None):

    h = project(h0, n)
    c = np.correlate(h, 1.0 - h, mode="full")

    best_h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:

        ord = np.arange(n) if rng is None else rng.permutation(n)

                c[ms] += delta * a
                di = np.where(ms == n - 1)[0]

        if not improved:

    cur = compute_c5(h)
    if cur < best_c:

    return best_h, best_c

        ns = int(v)

            ns += 1
        if 80 <= ns <= 900:
            ns_set.add(ns)
    for ns in sorted(ns_set):
        if time.time() > stop - 40:
            break
        ld = min(stop - 40, time.time() + 65)
        if ld - time.time() < 20:
            continue
        starts = [
            interpolate_h(best_h, best_n, ns),
            mode_init(ns, seed + ns, (1, 2, 3), 1.0),
            block_init(ns, rng, 12)
        ]
        for h0 in starts:
            if time.time() > ld:
                break
            a1 = coordinate_polish(h0, 6, ld, rng)
            a2 = subgrad_opt(a1[0], ld, rng)
            a3 = coordinate_polish(a2[0], 6, ld, rng)
            best_h, best_c, best_n = update_best(a3[0], ns, best_h, best_c, best_n)

    if time.time() < dl - 60:
        for r in range(12):
            if time.time() > dl - 60:
                break
            temp = 0.03 + 0.012 * r
            hp = project(best_h + rng.normal(0.0, temp, best_n), best_n)
            p1 = coordinate_polish(hp, 8, min(dl - 40, time.time() + 50), rng)
            p2 = subgrad_opt(p1[0], min(dl - 40, time.time() + 50), rng)
            best_h, best_c, best_n = update_best(p2[0], best_n, best_h, best_c, best_n)

    if time.time() < dl - 3:
        ph, pc = coordinate_polish(best_h, 150, max(dl - 2, time.time() + 1), rng)
        best_h, best_c, best_n = update_best(ph, len(ph), best_h, best_c, best_n)
        if time.time() < dl - 2:
            sh, sc = subgrad_opt(best_h, min(dl - 1, time.time() + 15), rng)
            best_h, best_c, best_n = update_best(sh, len(sh), best_h, best_c, best_n)

    bh = project(best_h, len(best_h))
    cf = compute_c5(bh)
    if cf <= best_c + 1e-12:
        return bh, cf, len(bh)
    return best_h, best_c, len(best_h)
