    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()

            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)

def update_corr_fast(c, h, p, old_val, new_val, n):
    delta = new_val - old_val
    if abs(delta) < 1e-14:
        return
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))

        a[vf] += (1.0 - h[jf[vf]])

def smooth_lse_val(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def lbfgs_smooth(h, n, alphas, maxiter, deadline, rng):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        # Use L-BFGS-B with penalty for constraint
        penalty = 10000.0
        def obj(x):
            h_clip = np.clip(x, 0, 1)
            return smooth_lse_val(h_clip, n, alpha) + penalty * (h_clip.sum() - n/2.0)**2
        res = minimize(obj, best_h, method='L-BFGS-B', 
                       bounds=[(0,1)]*n, options={'maxiter': maxiter, 'ftol': 1e-15})
        new_h = project_feasible(res.x, n)
        new_c = compute_c5(new_h, n)
        if new_c < best_c - 1e-12:
            best_h, best_c = new_h, new_c
    return best_h, best_c

def swap_sa(h, n, steps, deadline, rng, t_start=0.05, t_end=0.001):
    """Simulated annealing with swap moves"""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur_c = best_c
    for i in range(steps):
        if time.time() > deadline:
            break
        t = t_start * (t_end / t_start) ** (i / steps)
        # Random swap: pick two positions, exchange values
        i1, i2 = rng.randint(0, n, 2)
        if i1 == i2:
            continue
        # Try swap
        h[i1], h[i2] = h[i2], h[i1]
        new_h = project_feasible(h, n)
        new_c = compute_c5(new_h, n)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / t):
            h = new_h
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            h[i1], h[i2] = h[i2], h[i1]  # revert
    return best_h, best_c

def block_perturb(h, n, rng, block_size):
    """Add a block perturbation"""
    h_new = h.copy()
    a = rng.randint(0, n - block_size)
    b = a + block_size

    a2 = rng.randint(0, n - block_size)

    b2 = a2 + block_size
    amt = rng.uniform(0.05, 0.2)
    h_new[a:b] += amt
    h_new[a2:b2] -= amt
    return project_feasible(h_new, n)
