    h1, c1 = run_lbfgs(h_best, n, [500, 2000, 8000, 30000], 500, deadline)
    if c1 < best_c:
        h_best, best_c = h1, c1

    # Phase 2: Nesterov subgradient
    if time.time() < deadline - 200:
        h2 = nesterov_subgrad(h_best, n, 2000, 200, deadline)
        c2 = compute_c5(h2, n)
        if c2 < best_c:
            h_best, best_c = h2, c2

    # Phase 3: Exact CD polish
    h3, c3 = cd_1d(h_best, n, 500, deadline)
    if c3 < best_c:
        h_best, best_c = h3, c3

    # Phase 4: Multi-n exploration
    n_list = [500, 600, 800, 900, 1000, 1100, 1200, 1400, 1600]
    for n_try in n_list:
        if time.time() > deadline - 150: break
        if n_try == n: continue
        h_int = resample(h_best, len(h_best), n_try)
        h_o, c_o = run_lbfgs(h_int, n_try, [1000, 5000, 20000], 300, deadline)
        h_o, c_o2 = cd_1d(h_o, n_try, 300, deadline)
        if c_o2 < c_o: h_o, c_o = h_o, c_o2
        if c_o < best_c - 1e-7:
            h_best, best_c, n = h_o, c_o, n_try

    # Phase 5: Targeted perturbation + re-optimize
    for r in range(15):
        if time.time() > deadline - 60: break
        s = 0.003 + 0.003 * r
        hp = targeted_perturb(h_best, n, rng, s)
        h_o, c_o = cd_1d(hp, n, 200, deadline)
        if c_o < best_c:
            h_best, best_c = h_o, c_o

    # Phase 6: Random shake
    for r in range(8):
        if time.time() > deadline - 40: break
        s = 0.002 + 0.004 * r
        hp = project_feasible(h_best + rng.normal(0, s, n), n)
        h_o, c_o = cd_1d(hp, n, 300, deadline)
        if c_o < best_c:
            h_best, best_c = h_o, c_o

    # Phase 7: Final L-BFGS polish
    n = len(h_best)
    if time.time() < deadline - 30:
        h_f, c_f = run_lbfgs(h_best, n, [50000, 100000], 200, deadline)
        if c_f < best_c: h_best, best_c = h_f, c_f

    # Phase 8: Final CD
    if time.time() < deadline - 3:
        h_f, c_f = cd_1d(h_best, len(h_best), 200, deadline)
        if c_f < best_c: h_best, best_c = h_f, c_f

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
