    target=0.5*n

       diff=target-h.sum()
       if abs(diff)<1e-13: break
       if diff>0:
          room=1.0-h; tot=room.sum()
          if tot<=1e-15: break
          h=np.clip(h + (diff/tot)*room,0.0,1.0)

          tot=h.sum()

def project_feasible(h,n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = 0.5 * n
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot <= 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot <= 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 0.0:
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) > 0:
            h[idx] += diff / len(idx)
            h = np.clip(h, 0.0, 1.0)
    return h

def make_ou(n, alpha, rng):
    u = np.empty(n, dtype=float)
    u[0] = rng.normal()
    if n == 1: return u
    coeff = math.sqrt(max(0.0, 1.0 - alpha*alpha))
    for i in range(1,n):
        u[i] = alpha*u[i-1] + coeff*rng.normal()
    return u

import numpy as np
import math
import time


def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = 0.5 * n
    for _ in range(120):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot <= 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot <= 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 0.0:
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) > 0:
            h[idx] += diff / len(idx)
            h = np.clip(h, 0.0, 1.0)
    return h


def compute_c5(h, n):
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(c.max() * (2.0 / n))


def resample_h(h, old_n, new_n):
    if old_n == new_n:
        return np.array(h, dtype=float, copy=True)
    x_old = np.linspace(0.0, 2.0, old_n)
    x_new = np.linspace(0.0, 2.0, new_n)
    return np.interp(x_new, x_old, h)


def threshold_binary(n, scores):
    scores = np.asarray(scores, dtype=float)
    k = n // 2
    out = np.zeros(n, dtype=float)
    if k <= 0:
        return out
    idx = np.argpartition(scores, k - 1)[:k]
    out[idx] = 1.0
    return out


def make_ou(n, alpha, rng):
    u = np.empty(n, dtype=float)
    u[0] = rng.normal()
    if n == 1:
        return u
    coeff = math.sqrt(max(0.0, 1.0 - alpha * alpha))
    for i in range(1, n):
        u[i] = alpha * u[i - 1] + coeff * rng.normal()
    return u


def make_sine(n, f, rng):
    x = np.arange(n, dtype=float)
    f2 = min(0.85, 2.0 * f + rng.uniform(-0.04, 0.04))
    p1 = rng.uniform(0.0, 2.0 * math.pi)
    p2 = rng.uniform(0.0, 2.0 * math.pi)
    s = np.sin(2.0 * math.pi * f * x + p1)
    s += 0.6 * np.sin(2.0 * math.pi * f2 * x + p2)
    s += 0.35 * rng.normal(size=n)
    return s


def add_candidate(candidates, h, n, do_project):
    if do_project:
        h = project_feasible(h, int(n))
    else:
        h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    n = int(h.size)
    c = compute_c5(h, n)
    candidates.append((c, h, n))


def _cand_key(item):
    return item[0]


def _cd_fval(c_aff, a, buf, diag, unaff, d):
    np.multiply(a, d, out=buf)
    buf += c_aff
    buf[diag] -= d * d
    r = float(buf.max())
    if unaff > r:
        r = unaff
    return r


def coord_descent_inc(h, n, max_passes, deadline, gs_iters=30, rng=None):
    h = project_feasible(h, int(n))
    n = int(h.size)
    c = np.correlate(h, 1.0 - h, mode="full").astype(float)
    full_len = 2 * n - 1
    a_full = np.empty(full_len, dtype=float)
    buf_full = np.empty(full_len, dtype=float)
    order = np.arange(n)
    gr = (math.sqrt(5.0) - 1.0) / 2.0

    for _ in range(int(max_passes)):
        if time.time() >= deadline:
            break
        if rng is not None:
            rng.shuffle(order)

        improved = 0.0
        for p in order:
            h0 = float(h[p])
            lo = -h0
            hi = 1.0 - h0
            if hi - lo <= 1e-15:
                continue

            side = p if p >= n - 1 - p else n - 1 - p
            s_lo = -side
            s_hi = side
            m_lo = n - 1 + s_lo
            L = 2 * side + 1
            m_hi = m_lo + L - 1

            a = a_full[:L]
            buf = buf_full[:L]
            a.fill(0.0)

            st1 = max(s_lo, -(n - 1 - p))
            en1 = min(s_hi, p)
            if st1 <= en1:
                ln = en1 - st1 + 1
                off = st1 - s_lo
                imin = p - en1
                imax = p - st1
                view = h[imin:imax + 1][::-1]
                a[off:off + ln] += 1.0 - view

            st2 = max(s_lo, -p)
            en2 = min(s_hi, n - 1 - p)
            if st2 <= en2:
                ln = en2 - st2 + 1
                off = st2 - s_lo
                imin = p + st2
                imax = p + en2
                view = h[imin:imax + 1]
                a[off:off + ln] -= view

            diag = -s_lo
            c_aff = c[m_lo:m_hi + 1]
            unaff = 0.0
            if m_lo > 0:
                unaff = max(unaff, float(c[:m_lo].max()))
            if m_hi < full_len - 1:
                unaff = max(unaff, float(c[m_hi + 1:].max()))

            f0 = _cd_fval(c_aff, a, buf, diag, unaff, 0.0)
            d1 = lo + gr * (hi - lo)
            d2 = hi - gr * (hi - lo)
            f1 = _cd_fval(c_aff, a, buf, diag, unaff, d1)
            f2 = _cd_fval(c_aff, a, buf, diag, unaff, d2)

            for __ in range(int(gs_iters)):
                if hi - lo < 1e-14:
                    break
                if f1 < f2:
                    hi = d2
                    d2 = d1
                    f2 = f1
                    d1 = hi - gr * (hi - lo)
                    f1 = _cd_fval(c_aff, a, buf, diag, unaff, d1)
                else:
                    lo = d1
                    d1 = d2
                    f1 = f2
                    d2 = lo + gr * (hi - lo)
                    f2 = _cd_fval(c_aff, a, buf, diag, unaff, d2)

            dm = 0.5 * (lo + hi)
            fm = _cd_fval(c_aff, a, buf, diag, unaff, dm)
            fl = _cd_fval(c_aff, a, buf, diag, unaff, lo)
            if fl < fm:
                fm = fl
                dm = lo
            fh = _cd_fval(c_aff, a, buf, diag, unaff, hi)
            if fh < fm:
                fm = fh
                dm = hi
            if f0 < fm:
                fm = f0
                dm = 0.0

            if fm < f0 - 1e-14:
                np.multiply(a, dm, out=buf)
                buf += c_aff
                buf[diag] -= dm * dm
                c[m_lo:m_hi + 1] = buf
                h[p] = float(np.clip(h0 + dm, 0.0, 1.0))
                improved += f0 - fm

        if improved < 1e-13:
            break
        if time.time() >= deadline:
            break
        c = np.correlate(h, 1.0 - h, mode="full").astype(float)

    h = project_feasible(h, n)
    c = np.correlate(h, 1.0 - h, mode="full")
    return h, float(c.max() * (2.0 / n))


def apply_flip_binary(c, h, p, flip, n):
    h0 = float(h[p])
    side = p if p >= n - 1 - p else n - 1 - p
    s_lo = -side
    m_lo = n - 1 + s_lo

    st1 = max(s_lo, -(n - 1 - p))
    en1 = min(side, p)
    if st1 <= en1:
        ln = en1 - st1 + 1
        off = st1 - s_lo
        imin = p - en1
        imax = p - st1
        view = h[imin:imax + 1][::-1]
        if flip > 0:
            c[m_lo + off:m_lo + off + ln] += 1.0 - view
        else:
            c[m_lo + off:m_lo + off + ln] -= 1.0 - view

    st2 = max(s_lo, -p)
    en2 = min(side, n - 1 - p)
    if st2 <= en2:
        ln = en2 - st2 + 1
        off = st2 - s_lo
        imin = p + st2
        imax = p + en2
        view = h[imin:imax + 1]
        if flip > 0:
            c[m_lo + off:m_lo + off + ln] -= view
        else:
            c[m_lo + off:m_lo + off + ln] += view

    c[n - 1] -= 1.0
    h[p] = h0 + flip


def binary_hill_climb(h, n, deadline, rng, max_trials=5000, temperature=0.002):
    n = int(n)
    h = (np.asarray(h, dtype=float) > 0.5).astype(float)
    target = n // 2
    cursum = int(h.sum())
    if cursum > target:
        idx = np.flatnonzero(h > 0.5)
        rng.shuffle(idx)
        h[idx[:cursum - target]] = 0.0
    elif cursum < target:
        idx = np.flatnonzero(h < 0.5)
        rng.shuffle(idx)
        h[idx[:target - cursum]] = 1.0

    c = np.correlate(h, 1.0 - h, mode="full").astype(float)
    cur = float(c.max())
    best = cur
    best_h = h.copy()
    ones = np.flatnonzero(h > 0.5)
    zeros = np.flatnonzero(h < 0.5)
    T = float(temperature)
    stall = 0

    for _ in range(int(max_trials)):
        if time.time() > deadline:
            break
        if ones.size == 0 or zeros.size == 0:
            break

        io = rng.randint(ones.size)
        jo = rng.randint(zeros.size)
        i = int(ones[io])
        j = int(zeros[jo])
        if i == j:
            continue

        apply_flip_binary(c, h, i, -1, n)
        apply_flip_binary(c, h, j, 1, n)
        new = float(c.max())
        delta_c = (new - cur) * (2.0 / n)
        accept = new < cur
        if (not accept) and T > 1e-5 and delta_c < 0.05:
            accept = rng.random() < math.exp(-delta_c / T)

        if accept:
            cur = new
            ones[io] = j
            zeros[jo] = i
            if cur < best:
                best = cur
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            apply_flip_binary(c, h, i, 1, n)
            apply_flip_binary(c, h, j, -1, n)

        T *= 0.9995
        if stall > 2000:
            scores = best_h + 0.25 * rng.normal(size=n)
            h = threshold_binary(n, scores)
            c = np.correlate(h, 1.0 - h, mode="full").astype(float)
            cur = float(c.max())
            if cur < best:
                best = cur
                best_h = h.copy()
            ones = np.flatnonzero(h > 0.5)
            zeros = np.flatnonzero(h < 0.5)
            stall = 0

    return best_h, best * (2.0 / n)


def active_relax(h, n, deadline, rng, max_rounds=2):
    h = project_feasible(h, int(n))
    n = int(h.size)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    for _ in range(int(max_rounds)):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode="full")
        top_count = 16
        if c.size <= top_count:
            top = np.arange(c.size)
        else:
            top = np.argpartition(c, -top_count)[-top_count:]

        g = np.zeros(n, dtype=float)
        for mm in top:
            s = int(mm - (n - 1))
            if s >= 0:
                L = n - s
                if L > 0:
                    g[s:n] += 1.0 - h[0:L]
                    g[0:L] -= h[s:n]
            else:
                r = -s
                L = n - r
                if L > 0:
                    g[0:L] += 1.0 - h[r:n]
                    g[r:n] -= h[0:L]

        g -= g.mean()
        gmax = float(np.max(np.abs(g)))
        if gmax <= 1e-12:
            break
        g /= gmax

        accepted = False
        for tau in (0.08, 0.04, 0.02):
            if time.time() > deadline:
                break
            hn = project_feasible(h - tau * g, n)
            sub_deadline = min(deadline, time.time() + 18.0)
            hn2, c2 = coord_descent_inc(hn, n, 10, sub_deadline, gs_iters=20, rng=rng)
            if c2 < best_c:
                h = hn2
                best_c = c2
                best_h = h.copy()
                accepted = True
                break
        if not accepted:
            break

    return best_h, best_c

            h_init = resample_h(h0, n, n_new)
            n = n_new
        else:
            h_init = h0

    best_h = project_feasible(h_init, n)
    best_n = int(best_h.size)
    best_c = compute_c5(best_h, best_n)

    phase1_end = min(deadline - 40.0, t0 + 0.24 * budget)
    if phase1_end > t0 + 10.0:
        h1, c1 = coord_descent_inc(best_h, best_n, 1000, phase1_end, gs_iters=32, rng=rng)
        if c1 < best_c:
            best_h, best_c, best_n = h1, c1, int(h1.size)

    phase2_end = min(deadline - 140.0, t0 + 0.36 * budget)
    if phase2_end > time.time() + 25.0 and budget >= 400.0:
        h2, c2 = active_relax(best_h, best_n, phase2_end, rng, max_rounds=2)
        if c2 < best_c:
            best_h, best_c, best_n = h2, c2, int(h2.size)

    n_set = set([best_n, 384, 512, 640, 700, 800])
    n_list = sorted([mm for mm in n_set if 256 <= mm <= 1000 and mm % 2 == 0])
    if best_n % 2 == 0 and 256 <= best_n <= 1000 and best_n not in n_list:
        n_list.append(best_n)
    if not n_list:
        n_list = [640]
    n_list = n_list[:6]

    x_best = np.linspace(0.0, 2.0, best_n)
    candidates = []

    for nt in n_list:
        if nt == best_n:
            add_candidate(candidates, np.array(best_h, dtype=float, copy=True), nt, False)
            hb = threshold_binary(nt, best_h + 0.02 * rng.normal(size=nt))
            add_candidate(candidates, hb, nt, False)
            add_candidate(candidates, 1.0 - hb, nt, False)
            add_candidate(candidates, hb[::-1], nt, False)
        else:
            h_int = np.interp(np.linspace(0.0, 2.0, nt), x_best, best_h)
            add_candidate(candidates, h_int, nt, True)
            hb = threshold_binary(nt, h_int)
            add_candidate(candidates, hb, nt, False)
            add_candidate(candidates, 1.0 - hb, nt, False)
            add_candidate(candidates, hb[::-1], nt, False)

        for alpha in (0.2, 0.4, 0.6, 0.8, 0.95):
            for rep in range(2):
                hb = threshold_binary(nt, make_ou(nt, alpha, rng))
                add_candidate(candidates, hb, nt, False)
                if rep == 0:
                    add_candidate(candidates, 1.0 - hb, nt, False)

        for f in (0.1, 0.22, 0.36):
            hb = threshold_binary(nt, make_sine(nt, f, rng))
            add_candidate(candidates, hb, nt, False)

    if candidates:
        candidates.sort(key=_cand_key)
        pool = []
        for item in candidates[:10]:
            if item[0] < best_c + 0.035:
                pool.append(item)
        for nt in n_list:
            for item in candidates:
                if item[2] == nt:
                    pool.append(item)
                    break
        if len(pool) < 4:
            for item in candidates[:4]:
                pool.append(item)

        pool_deadline = deadline - 95.0
        bin_budget_used = 0
        for item in pool:
            if time.time() > pool_deadline:
                break
            cf, hf, nf = item
            nf = int(nf)
            if nf != nf:
                continue
            remaining = pool_deadline - time.time()
            if remaining < 25.0:
                break
            if best_n == nf and np.array_equal(hf, best_h):
                continue
            if cf > best_c + 0.025:
                limit = min(pool_deadline, time.time() + max(20.0, 0.45 * remaining))
            else:
                limit = min(pool_deadline, time.time() + max(55.0, 0.65 * remaining))

            if (
                nf % 2 == 0
                and bin_budget_used < 2
                and cf < best_c + 0.02
                and np.max(np.abs(2.0 * hf - 1.0)) < 1e-8
                and limit - time.time() > 35.0
            ):
                bin_deadline = min(limit - 25.0, time.time() + 25.0)
                hb, cb = binary_hill_climb(hf, nf, bin_deadline, rng, max_trials=6000, temperature=0.0018)
                bin_budget_used += 1
                if cb < best_c:
                    best_h, best_c, best_n = hb, cb, nf
                if cb < cf:
                    hf, cf = hb, cb

            hf = project_feasible(hf, nf)
            hnew, cnew = coord_descent_inc(hf, nf, 700, limit, gs_iters=30, rng=rng)
            if cnew < best_c:
                best_h, best_c, best_n = hnew, cnew, int(hnew.size)

    if time.time() < deadline - 75.0:
        final_cands = []
        add_candidate(final_cands, 1.0 - best_h, best_n, True)
        add_candidate(final_cands, best_h[::-1], best_n, False)
        hs = 0.5 * (best_h + best_h[::-1])
        add_candidate(final_cands, hs, best_n, True)
        add_candidate(final_cands, threshold_binary(best_n, best_h + 0.07 * rng.normal(size=best_n)), best_n, False)
        add_candidate(final_cands, threshold_binary(best_n, best_h[::-1] + 0.07 * rng.normal(size=best_n)), best_n, False)
        if final_cands:
            final_cands.sort(key=_cand_key)
            fin_deadline = deadline - 35.0
            ran = 0
            for item in final_cands:
                if ran >= 3 or time.time() > fin_deadline:
                    break
                cf, hf, nf = item
                if cf >= best_c + 0.018:
                    continue
                ran += 1
                lim = min(fin_deadline, time.time() + max(18.0, 0.7 * (fin_deadline - time.time())))
                hnew, cnew = coord_descent_inc(hf, nf, 300, lim, gs_iters=32, rng=rng)
                if cnew < best_c:
                    best_h, best_c, best_n = hnew, cnew, int(hnew.size)

    if time.time() < deadline - 2.0:
        h_final, c_final = coord_descent_inc(best_h, best_n, 1000, deadline - 1.0, gs_iters=36, rng=rng)
        if c_final < best_c:
            best_h, best_c, best_n = h_final, c_final, int(h_final.size)

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, float(best_c), int(best_n)
