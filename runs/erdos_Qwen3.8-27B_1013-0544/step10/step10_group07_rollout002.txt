        else:

            if tot < 1e-15:

def coordinate_update(c, h, p, n, grid_count=25, xatol=1e-10, local=True):
    M = 2*n - 1
    h0 = float(h[p])
    if n <= 1: return c,h0,False
    lo = p if p < n - 1 - p else n - 1 - p
    alt1 = 2*n - 2 - p
    alt2 = n - 1 + p
    hi = alt1 if alt1 > alt2 else alt2
    if lo < 0: lo=0
    if hi > M-1: hi=M-1
    if lo > hi: return c,h0,False
    ms = np.arange(lo, hi+1, dtype=int)
    L = ms.size
    a = np.zeros(L, dtype=float)
    jf = p + ms - (n - 1)
    vf = (jf >= 0) & (jf < n) & (jf != p)
    if np.any(vf):
        a[vf] += 1.0 - h[jf[vf]]
    jb = p + (n-1) - ms
    vb = (jb >= 0) & (jb < n) & (jb != p)
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    base = c[ms].copy()
    base -= h0 * a
    diag_pos = (n - 1) - lo
    if 0 <= diag_pos < L and lo <= n-1 <= hi:
        base[diag_pos] -= h0 * (1.0 - h0)
    else:
        diag_pos = -1
    max_out = -np.inf
    if lo > 0:
        max_out = float(np.max(c[:lo]))

import numpy as np
from scipy.optimize import minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    if n <= 0:
        return 0.0
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c)) * (2.0 / n)

def raw_max(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")))

def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(60):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    diff = target - h.sum()
    if diff > 1e-13:
        order = np.argsort(1.0 - h)[::-1]
    elif diff < -1e-13:
        order = np.argsort(h)
    else:
        return h
    rem = diff
    for idx in order:
        if rem > 0:
            add = min(1.0 - h[idx], rem)
        else:
            add = -min(h[idx], -rem)
        h[idx] += add
        rem -= add
        if abs(rem) < 1e-13:
            break
    return h

def _coord_scalar_f(x, base, a, diag_pos, max_out):
    val = base + a * x
    if diag_pos >= 0:
        val[diag_pos] += x - x * x
    m = float(val.max())
    if max_out > m:
        return max_out
    return m

    if hi > M - 1:

    if lo > hi:

    if np.any(vf):

    vb = (jb >= 0) & (jb < n) & (jb != p)
    if np.any(vb):

    if diag_pos >= 0:

    g = np.unique(np.append(g, h0))

    if new_max <= current + 1e-7:
        c[ms] = new_vals
        h[p] = best_x
        return c, best_x, True
    return c, h0, False

def coordinate_cd(start_h, n, max_passes, deadline, grid_count=25, seed=1,
                  stagn_limit=10, local=True):
    h = project_feasible(start_h, n)
    c = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / n

    best_h = h.copy()
    best_c = float(np.max(c)) * dx
    cur_h = h.copy()

    rng = np.random.RandomState(seed)
    stagn = 0
    pass_no = 0
    block = max(8, n // 25)

    for _ in range(max_passes):
        if time.time() > deadline:
            break

        c = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        if pass_no == 0:
            order = np.arange(n)
        elif pass_no == 1:
            order = np.arange(n - 1, -1, -1)
        else:
            order = rng.permutation(n)
        pass_no += 1

        streak = 0
        for pos in order:
            if time.time() > deadline - 0.5:
                break
            c, _, _ = coordinate_update(
                c, cur_h, int(pos), n, grid_count, 1e-10, local
            )
            streak += 1
            if streak >= block:
                cur_h = project_feasible(cur_h, n)
                c = np.correlate(cur_h, 1.0 - cur_h, mode="full")
                streak = 0

        if time.time() > deadline:
            break

        cur_h = project_feasible(cur_h, n)
        c = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        new_c = float(np.max(c)) * dx

        if new_c < best_c - 1e-14:
            best_h = cur_h.copy()
            best_c = new_c
            stagn = 0
        else:
            stagn += 1
            if stagn >= stagn_limit:
                break

    return best_h, best_c

def _pair_eval(x, h, p, q, S):
    y = S - x
    if y < -1e-12 or y > 1.0 + 1e-12:
        return 1e100
    hh = h.copy()
    hh[p] = x
    hh[q] = y
    return raw_max(hh)

def pair_cd(start_h, n, max_pairs, deadline, seed=99, grid_count=9):
    h = project_feasible(start_h, n)
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_raw = raw_max(h)
    dx = 2.0 / n
    rng = np.random.RandomState(seed)

    it = 0
    while it < max_pairs and time.time() < deadline:
        it += 1
        if it <= n:
            p = (it - 1) % n
            q = (p + 1) % n
        else:
            p = int(rng.randint(0, n))
            q = int(rng.randint(0, n))
            if p == q:
                q = (q + 1) % n

        S = float(h[p] + h[q])
        lo = 0.0 if S > 1.0 else S - 1.0
        hi = 1.0 if S < 1.0 else S
        if hi - lo < 1e-12:
            continue

        x0 = float(h[p])
        g = np.linspace(lo, hi, max(7, int(grid_count)))
        g = np.unique(np.append(g, x0))
        m = len(g)

        vals = np.empty(m)
        best_x = x0
        best_val = cur_raw
        imin = 0

        for i in range(m):
            x = float(g[i])
            if abs(x - x0) < 1e-13:
                val = cur_raw
            else:
                val = _pair_eval(x, h, p, q, S)
            vals[i] = val
            if val < best_val:
                best_val = val
                best_x = x
                imin = i

        if best_val <= cur_raw + 1e-8 and m > 1:
            l = float(g[imin - 1]) if imin > 0 else float(lo)
            r = float(g[imin + 1]) if imin + 1 < m else float(hi)
            if r - l > 1e-11:
                try:
                    res = minimize_scalar(
                        _pair_eval,
                        bounds=(l, r),
                        args=(h, p, q, S),
                        method="bounded",
                        options={"xatol": 1e-9},
                    )
                    if res.success and np.isfinite(res.fun) and res.fun < best_val - 1e-10:
                        best_x = float(res.x)
                        best_val = float(res.fun)
                except Exception:
                    pass

        best_x = float(np.clip(best_x, lo, hi))
        if best_val < cur_raw - 1e-10:
            h[p] = best_x
            h[q] = S - best_x
            cur_raw = best_val
            if cur_raw * dx < best_c:
                best_h = h.copy()
                best_c = cur_raw * dx

    h = project_feasible(h, n)
    c5 = compute_c5(h)
    if c5 < best_c:
        best_h = h.copy()
        best_c = c5
    return best_h, best_c

def resample_h(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return project_feasible(h, n_new)
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    return project_feasible(np.interp(x_new, x_old, h), n_new)

def make_smooth_start(n, seed):
    rng = np.random.RandomState(seed)
    x = np.linspace(0.0, 2.0, n)
    f1 = int(rng.randint(1, 12))
    f2 = int(rng.randint(1, 18))
    a1 = float(rng.uniform(0.15, 0.35))
    a2 = float(rng.uniform(0.05, 0.20))
    p1 = float(rng.uniform(0.0, 2.0 * np.pi))
    p2 = float(rng.uniform(0.0, 2.0 * np.pi))
    h = 0.5
    h = h + a1 * np.sin(2.0 * np.pi * f1 * x / 2.0 + p1)
    h = h + a2 * np.sin(2.0 * np.pi * f2 * x / 2.0 + p2)
    return project_feasible(np.clip(h, 0.0, 1.0), n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget = float(budget_s)
    reserve = min(30.0, max(5.0, budget * 0.05))
    deadline = t0 + max(10.0, budget - reserve)

    try:
        h0 = np.asarray(initial_h_values, dtype=float).copy()
        n0 = int(len(h0))
        if n0 <= 1:
            raise ValueError("bad length")
    except Exception:
        n0 = 600
        h0 = make_smooth_start(n0, seed)

    if n0 > 1000:
        h0 = resample_h(h0, 700)
        n0 = 700

    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_c = compute_c5(best_h)

    remain = deadline - t0
    main_alloc = min(remain, 650.0, remain * 0.70)
    if main_alloc >= 5.0:
        h1, c1 = coordinate_cd(
            best_h, n0, 100000, min(deadline, time.time() + main_alloc),
            grid_count=23, seed=seed + 1, stagn_limit=12, local=True
        )
        if c1 < best_c:
            best_h, best_c = h1, c1

    rem = deadline - time.time()
    if rem > 140.0:
        alloc = min(rem, 350.0, rem * 0.50)
        h2, c2 = coordinate_cd(
            best_h, len(best_h), 100000, min(deadline, time.time() + alloc),
            grid_count=31, seed=seed + 77, stagn_limit=10, local=True
        )
        if c2 < best_c:
            best_h, best_c = h2, c2

    rem = deadline - time.time()
    if rem > 80.0:
        nb = len(best_h)
        max_pairs = int(min(600, max(200, 2 * nb)))
        alloc = min(rem, 120.0, rem * 0.30)
        h3, c3 = pair_cd(
            best_h, nb, max_pairs, min(deadline, time.time() + alloc),
            seed=seed + 321, grid_count=9
        )
        if c3 < best_c:
            best_h, best_c = h3, c3

    seen = set()
    n_opts = []
    cur_n = len(best_h)
    candidates = [600, 500, 400, 300, 250, 200, 150, 800, 900, 700,
                  cur_n - 100, cur_n + 100]
    for m in candidates:
        if 100 <= m < 1000 and m != cur_n and m not in seen:
            seen.add(m)
            n_opts.append(m)

    for m in n_opts:
        rem = deadline - time.time()
        if rem < 70.0:
            break
        alloc = min(rem, 180.0, rem * 0.28)
        h_start = resample_h(best_h, m)
        h_try, c_try = coordinate_cd(
            h_start, m, 100000, min(deadline, time.time() + alloc),
            grid_count=23, seed=seed + m, stagn_limit=8, local=True
        )
        if c_try < best_c:
            best_h, best_c = h_try, c_try

        if m in (400, 300) and time.time() + 50.0 < deadline:
            h_rand = make_smooth_start(m, seed + 31 * m)
            h_try2, c_try2 = coordinate_cd(
                h_rand, m, 100000, min(deadline, time.time() + 50.0),
                grid_count=23, seed=seed + 37 * m, stagn_limit=6, local=True
            )
            if c_try2 < best_c:
                best_h, best_c = h_try2, c_try2

    nb = len(best_h)
    rem = deadline - time.time()
    if rem > 90.0:
        for nz in (0.02, 0.01, 0.05):
            if time.time() > deadline - 40.0:
                break
            rsh = np.random.RandomState(seed + 1000 + int(nz * 10000))
            h_sh = project_feasible(best_h + rsh.normal(0.0, nz, nb), nb)
            alloc = min(60.0, (deadline - time.time()) * 0.5)
            if alloc < 25.0:
                break
            hs, cs = coordinate_cd(
                h_sh, nb, 100000, min(deadline, time.time() + alloc),
                grid_count=23, seed=seed + 900 + int(nz * 100),
                stagn_limit=6, local=True
            )
            if cs < best_c:
                best_h, best_c = hs, cs

    nb = len(best_h)
    if time.time() < deadline - 25.0:
        hf, cf = coordinate_cd(
            best_h, nb, 100000, deadline - 5.0,
            grid_count=33, seed=seed + 1234, stagn_limit=8, local=True
        )
        if cf < best_c:
            best_h, best_c = hf, cf

    best_h = project_feasible(best_h, nb)
    best_c = compute_c5(best_h)
    return best_h, float(best_c), int(nb)
