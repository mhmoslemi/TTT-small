        pos = a_nd > 1e-13
        if np.any(neg):
            x_lo = max(x_lo, float(np.max((T - c_nd[neg]) / a_nd[neg])))
        if np.any(pos):
            x_hi = min(x_hi, float(np.min((T - c_nd[pos]) / a_nd[pos])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Cq = T - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            if x_lo < r1:
                x_hi = min(x_hi, r1)
            elif x_hi > r2:
                x_lo = max(x_lo, r2)
            else:
                if h0 - r1 <= r2 - h0:
                    x_hi = min(x_hi, r1 + 1e-9)
                else:
                    x_lo = max(x_lo, r2 - 1e-9)
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if x_lo > x_hi + 1e-10:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def coord_descent(h, n, max_passes, deadline, rng=None, nbisect=50):
    h = h.copy()
    prev_c = compute_c5(h, n)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = 0
        order = rng.permutation(n) if rng is not None else np.arange(n)
        for p in order:
            if time.time() > deadline:
                break
            x_new = solve_coord(c, h, int(p), n, nbisect)
            if abs(x_new - h[p]) > 1e-14:
                h[p] = x_new
                improved += 1
        if improved == 0:
            break
        h = project_feasible(h, n)
        cur_c = compute_c5(h, n)
        if cur_c < prev_c - 1e-12:
            prev_c = cur_c
    return h, compute_c5(h, n)

def lse_obj(x, n, alpha, lam):
    c = np.correlate(x, 1.0 - x, mode='full')
    cm = c.max()
    shifted = alpha * (c - cm)
    sm = shifted.max()
    lse = cm + (sm + np.log(np.sum(np.exp(shifted - sm)))) / alpha
    s = x.sum() - n / 2.0
    return (lse + lam * s * s) * (2.0 / n)

def lse_grad(x, n, alpha, lam):
    c = np.correlate(x, 1.0 - x, mode='full')
    cm = c.max()
    shifted = alpha * (c - cm)
    sm = shifted.max()
    w = np.exp(shifted - sm)
    w_sum = w.sum()
    w_norm = w / w_sum
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w_norm[k]
        if wk < 1e-16:
            continue
        d = k - (n - 1)
        if d > 0:
            grad[d:] += wk * (1.0 - x[:n - d])
            grad[:n - d] -= wk * x[d:]
        elif d < 0:
            d2 = -d
            grad[:n - d2] += wk * (1.0 - x[d2:])
            grad[d2:] -= wk * x[:n - d2]
        else:
            grad[:] += wk * (1.0 - 2.0 * x)
    s = x.sum() - n / 2.0
    grad += 2.0 * lam * s
    return grad * dx

def lbfgs_step(h, n, alpha, lam, maxiter, deadline):
    bounds = [(0.0, 1.0)] * n
    t_start = time.time()
    max_time = max(10, deadline - t_start - 5)
    try:
        res = minimize(lse_obj, h, jac=lse_grad, args=(n, alpha, lam),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': 1e-14, 'maxfun': 5000})
    except Exception:
        return h
    h_new = project_feasible(res.x, n)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 700
        x = np.linspace(0, 2, n)
        h = np.clip(0.5 + 0.15 * np.sin(np.pi * x) + 0.05 * np.sin(3 * np.pi * x), 0, 1)
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    # Phase 1: Quick CD baseline
    h, best_c = coord_descent(h, n, 15, deadline, rng, 45)
    # Phase 2: L-BFGS-B continuation
    for alpha in [300, 1500, 8000, 40000, 200000]:
        if time.time() > deadline - 80:
            break
        h = lbfgs_step(h, n, alpha, 1e5, 800, deadline)
        c_new = compute_c5(h, n)
        if c_new < best_c:
            best_c = c_new
    # Phase 3: CD polish
    h, c3 = coord_descent(h, n, 60, deadline, rng, 50)
    if c3 < best_c:
        best_c = c3
    # Phase 4: Perturbation restarts
    best_h = h.copy()
    for trial in range(8):
        if time.time() > deadline - 120:
            break
        sigma = 0.005 + 0.005 * trial
        h_p = project_feasible(best_h + sigma * rng.randn(n), n)
        h_p, c_p = coord_descent(h_p, n, 15, deadline, rng, 45)
        if time.time() < deadline - 60:
            h_p = lbfgs_step(h_p, n, 40000, 1e5, 500, deadline)
            c_p = compute_c5(h_p, n)
            h_p, c_p = coord_descent(h_p, n, 20, deadline, rng, 50)
        if c_p < best_c:
            best_c = c_p
            best_h = h_p.copy()
    h = best_h
    # Phase 5: Try different n
    for n_try in [500, 1000, 350]:
        if time.time() > deadline - 200:
            break
        x_orig = np.linspace(0, 2, len(h))
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, h), n_try)
        h_t, c_t = coord_descent(h_int, n_try, 15, deadline, rng, 45)
        if time.time() < deadline - 80:
            for a2 in [5000, 50000]:
                h_t = lbfgs_step(h_t, n_try, a2, 1e5, 500, deadline)
                c_t = compute_c5(h_t, n_try)
            h_t, c_t = coord_descent(h_t, n_try, 25, deadline, rng, 50)
        if c_t < best_c:
            best_c = c_t
            h = h_t
    # Phase 6: Final high-precision CD
    n_final = len(h)
    remaining = deadline - time.time() - 5
    max_p = max(10, int(remaining / 0.8))
    h, final_c = coord_descent(h, n_final, max_p, deadline, None, 55)
    if final_c < best_c:
        best_c = final_c
    h = project_feasible(h, n_final)
    best_c = compute_c5(h, n_final)
    return h, best_c, n_final
