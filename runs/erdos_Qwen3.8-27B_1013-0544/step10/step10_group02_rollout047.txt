    L_lo, L_hi = unaff, cur_max
    for _ in range(60):
        if L_hi - L_lo < 1e-13:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        feasible = True
        if np.any(zero_mask) and np.any(c_base[zero_mask] > L + 1e-14):
            feasible = False
        if feasible and np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if feasible and np.any(pos_mask):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        if x_lo > x_hi + 1e-13:
            feasible = False
        if feasible and has_diag:
            Cq = L - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                xc_lo, xc_hi = max(x_lo, 0.0), min(x_hi, 1.0)
                if not (xc_lo < r1 and xc_hi > r2):
                    feasible = False
        if feasible:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Cq = L - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            xc_lo, xc_hi = max(x_lo, 0.0), min(x_hi, 1.0)
            if xc_lo < r1 and xc_hi > r2:
                if h0 - r1 <= r2 - h0:
                    x_hi = min(x_hi, r1)
                else:
                    x_lo = max(x_lo, r2)
            elif x_lo >= r1 and x_hi <= r2:
                if h0 - r1 <= r2 - h0:
                    x_hi = min(x_hi, r1)
                else:
                    x_lo = max(x_lo, r2)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def update_corr(c, h, p, old_val, new_val, n):
    delta = new_val - old_val
    if abs(delta) < 1e-14:
        return c
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    diag_i = np.where(ms == (n - 1))[0]
    c[ms] += delta * a
    if len(diag_i) > 0:
        c[ms[diag_i[0]]] -= delta * delta
    return c

def exact_cd(h, n, max_passes, deadline, rng=None, order_mode='sequential'):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for it in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        order = list(range(n))
        if order_mode == 'random' and rng is not None:
            rng.shuffle(order)
        elif order_mode == 'adaptive' and rng is not None:
            key = h * (1.0 - h)
            if it % 2 == 1:
                key = key[::-1].copy()
            order = list(np.argsort(-key))
        for p in order:
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            xn = solve_1d(c, h, p, n)
            if abs(xn - h0) > 1e-12:
                update_corr(c, h, p, h0, xn, n)
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def smooth_lse(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_opt(h, n, alphas, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        def cf(x):
            return x.sum() - n / 2.0
        def cj(x):
            return np.ones(n)
        cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]
        try:
            res = minimize(smooth_lse, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c - 1e-12:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def interp_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def structured_perturb(h, n, rng, mode):
    h_new = h.copy()
    if mode == 'shift':
        s = rng.randint(1, max(2, n // 15))
        h_new = np.roll(h_new, s)
        h_new = project_feasible(h_new, n)
    elif mode == 'region_flip':
        start = rng.randint(0, n)
        length = rng.randint(3, max(4, n // 8))
        end = min(start + length, n)
        h_new[start:end] = 1.0 - h_new[start:end]
        h_new = project_feasible(h_new, n)
    elif mode == 'local_noise':
        start = rng.randint(0, max(1, n - n // 8))
        length = rng.randint(max(2, n // 20), max(3, n // 5))
        end = min(start + length, n)
        h_new[start:end] += rng.normal(0, 0.08, end - start)
        h_new = project_feasible(h_new, n)
    else:
        h_new = project_feasible(h + rng.normal(0, 0.03, n), n)
    return h_new

def refine_from_start(h_start, n, deadline, rng):
    best_h, best_c = h_start.copy(), compute_c5(h_start, n)
    h1, c1 = exact_cd(best_h, n, 120, deadline, rng, 'sequential')
    if c1 < best_c:
        best_h, best_c = h1, c1
    if time.time() < deadline - 30:
        h2, c2 = slsqp_opt(best_h, n, [50000, 200000], 150, deadline)
        if c2 < best_c - 1e-11:
            best_h, best_c = h2, c2
    if time.time() < deadline - 15:
        h3, c3 = exact_cd(best_h, n, 80, deadline, rng, 'random')
        if c3 < best_c:
            best_h, best_c = h3, c3
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = np.full(500, 0.5)
        n = 500
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)

    # Phase 1: CD to local optimum
    h1, c1 = exact_cd(h_best, n, 200, deadline, rng, 'sequential')
    if c1 < best_c:
        h_best, best_c = h1, c1

    # Phase 2: SLSQP with geometric alpha sweep
    alphas = [5000, 20000, 80000, 300000, 1000000, 3000000]
    h2, c2 = slsqp_opt(h_best, n, alphas, 250, deadline)
    if c2 < best_c - 1e-12:
        h_best, best_c = h2, c2

    # Phase 3: CD after SLSQP with adaptive ordering
    h3, c3 = exact_cd(h_best, n, 250, deadline, rng, 'adaptive')
    if c3 < best_c - 1e-12:
        h_best, best_c = h3, c3

    # Phase 4: Multi-resolution exploration
    n_base = len(h_best)
    n_list = []
    for d in [-3, 3, -7, 7, -13, 13, -31, 31, -64, 64]:
        nn = n_base + d
        if 150 <= nn <= 1500:
            n_list.append(nn)
    for nn in [200, 300, 350, 400, 450, 500, 550, 600, 650, 750, 800, 850, 900, 1000, 1100, 1200]:
        if abs(nn - n_base) > 5:
            n_list.append(nn)
    n_list = sorted(set(n_list), key=lambda x: abs(x - n_base))

    for n_try in n_list:
        if time.time() > deadline - 250:
            break
        h_int = interp_h(h_best, len(h_best), n_try)
        h_o, c_o = refine_from_start(h_int, n_try, deadline - 100, rng)
        if c_o < best_c - 1e-9:
            h_best, best_c = h_o, c_o

    # Phase 5: Structured perturbations + refinement
    n_cur = len(h_best)
    for r in range(50):
        if time.time() > deadline - 120:
            break
        modes = ['shift', 'region_flip', 'local_noise', 'random']
        mode = modes[r % len(modes)]
        hp = structured_perturb(h_best, n_cur, rng, mode)
        h_o, c_o = refine_from_start(hp, n_cur, deadline - 80, rng)
        if c_o < best_c - 1e-9:
            h_best, best_c = h_o, c_o
            # On success, try SLSQP at higher alpha
            if time.time() < deadline - 120:
                h_o2, c_o2 = slsqp_opt(h_best, n_cur, [500000, 2000000], 150, deadline)
                if c_o2 < best_c - 1e-10:
                    h_best, best_c = h_o2, c_o2
                h_o3, c_o3 = exact_cd(h_best, n_cur, 100, deadline, rng, 'adaptive')
                if c_o3 < best_c - 1e-10:
                    h_best, best_c = h_o3, c_o3

    # Phase 6: Annealed restarts
    n_cur = len(h_best)
    for r in range(20):
        if time.time() > deadline - 80:
            break
        scale = 0.06 * (1.0 - r / 20.0) + 0.01
        hp = project_feasible(h_best + rng.normal(0, scale, n_cur), n_cur)
        h_o, c_o = refine_from_start(hp, n_cur, deadline - 60, rng)
        if c_o < best_c - 1e-9:
            h_best, best_c = h_o, c_o

    # Phase 7: Final polish
    n_cur = len(h_best)
    for om in ['sequential', 'adaptive', 'random']:
        if time.time() > deadline - 5:
            break
        h_f, c_f = exact_cd(h_best, n_cur, 200, deadline, rng, om)
        if c_f < best_c - 1e-12:
            h_best, best_c = h_f, c_f

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
