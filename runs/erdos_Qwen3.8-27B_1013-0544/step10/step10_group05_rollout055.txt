def subgradient_step(h, c, n, lr):
    cmax = c.max()
    epsilon = 1e-6 * cmax + 1e-12
    active = c >= cmax - epsilon
    n_active = active.sum()
    if n_active == 0:
        return h
    sg = np.zeros(n)
    for p in range(n):
        m_min = max(0, n - 1 - p)
        m_max = min(2 * n - 2, 2 * n - 2 - p)
        ms = np.arange(m_min, m_max + 1)
        act = active[ms]
        if not act.any():
            continue
        jf = p + ms[act] - (n - 1)
        jb = p - ms[act] + (n - 1)
        sg_p = 0.0
        vf = (jf >= 0) & (jf < n)
        if vf.any():
            sg_p += np.sum(1.0 - h[jf[vf]])
        vb = (jb >= 0) & (jb < n)
        if vb.any():
            sg_p -= np.sum(h[jb[vb]])
        if (ms[act] == (n - 1)).any():
            sg_p -= h[p]
        sg[p] = sg_p / n_active
    return project_feasible(h - lr * sg, n)

                jac[p] = jac_p * dx
            return jac
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
                 'jac': lambda x: np.ones(n)}]
        try:
            res = minimize(obj, best_h, jac=obj_jac, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = np.full(600, 0.5)
        n = 600
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)

    # Phase 1: CD from initial (forward + random)
    h1, c1 = run_cd(h_best, n, 100, deadline, rng, use_greedy=False)
    if c1 < best_c:
        h_best, best_c = h1, c1
    h1, c1 = run_cd(h_best, n, 100, deadline, rng, use_greedy=True)
    if c1 < best_c:
        h_best, best_c = h1, c1

    # Phase 2: SLSQP refinement
    h2, c2 = slsqp_opt(h_best, n, [10000, 50000, 200000], 300, deadline)
    if c2 < best_c:
        h_best, best_c = h2, c1
    h2, c2 = run_cd(h_best, n, 200, deadline, rng, use_greedy=False)
    if c2 < best_c:
        h_best, best_c = h2, c2

    # Phase 3: Subgradient-guided exploration
    for it in range(30):
        if time.time() > deadline - 300:
            break
        c_curr = np.correlate(h_best, 1.0 - h_best, mode='full')
        lr = 0.02 * (1.0 - it / 40.0) + 0.005
        h_sg = subgradient_step(h_best, c_curr, n, lr)
        noise = rng.normal(0, 0.005, n)
        h_pert = project_feasible(h_sg + noise, n)
        h_o, c_o = run_cd(h_pert, n, 80, deadline, rng, use_greedy=True)
        if c_o < best_c:
            h_best, best_c = h_o, c_o

    # Phase 4: Try different n values
    n_candidates = [500, 550, 600, 650, 750, 800]
    for n_try in n_candidates:
        if time.time() > deadline - 200:
            break
        if n_try == n or n_try > 1000:
            continue
        h_int = interpolate_h(h_best, n, n_try)
        h_o, c_o = run_cd(h_int, n_try, 100, deadline, rng, use_greedy=False)
        h_o, c_o = slsqp_opt(h_o, n_try, [50000, 200000], 200, deadline)
        h_o, c_o = run_cd(h_o, n_try, 150, deadline, rng, use_greedy=True)
        if c_o < best_c - 1e-8:
            h_best, best_c = h_o, c_o
            n = n_try

    # Phase 5: Binary and Fourier restarts at best n
    n = len(h_best)
    inits = [binary_init(n, rng), fourier_init(n, 12, rng), fourier_init(n, 20, rng)]
    for h_init in inits:
        if time.time() > deadline - 150:
            break
        h_o, c_o = run_cd(h_init, n, 150, deadline, rng, use_greedy=False)
        h_o, c_o = slsqp_opt(h_o, n, [100000], 200, deadline)
        h_o, c_o = run_cd(h_o, n, 100, deadline, rng, use_greedy=True)
        if c_o < best_c:
            h_best, best_c = h_o, c_o

    # Phase 6: Final polish - alternating CD orders
    for _ in range(5):
        if time.time() > deadline - 10:
            break
        h_o, c_o = run_cd(h_best, n, 50, deadline, rng, use_greedy=False)
        h_o, c_o = run_cd(h_o, n, 50, deadline, rng, use_greedy=True)
        if c_o < best_c:
            h_best, best_c = h_o, c_o

    # Final
    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
