```python
def binary_sa(h_bin, n, rng, deadline, max_iters=500000):
    """Simulated annealing on binary configuration."""
    h = h_bin.copy()
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    best_c = c.max() * dx
    best_h = h.copy()
    
    T = 0.01  # initial temperature
    T_min = 0.001
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        # Pick random positions to swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        # Swap
        h[i], h[j] = h[j], h[i]
        new_c = np.correlate(h, 1.0 - h, mode='full').max() * dx
        
        if new_c < best_c or rng.random() < np.exp(-(new_c - best_c) / T):
            # Accept
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
            # Update ones and zeros
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        else:
            # Reject
            h[i], h[j] = h[j], h[i]
        
        # Cool down
        T = max(T_min, T * 0.9999)
    
    return best_h, best_c
```

```python
def compute_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full')
    logits = alpha * c
    logits -= logits.max()
    w = np.exp(logits)
    w /= w.sum()
    
    # grad[j] = dx * sum_m w[m] * ((1-h[j-s_m]) - h[j+s_m])
    # where s_m = m - (n-1)
    s = np.arange(2*n-1) - (n-1)  # shape (2n-1,)
    
    # For each j, we need:
    # sum_m w[m] * h[j - s_m] (valid where 0 <= j-s_m < n)
    # sum_m w[m] * h[j + s_m] (valid where 0 <= j+s_m < n)
    
    # Create index arrays
    # j ranges over [0, n), s ranges over [-(n-1), n-1]
    # j - s ranges over [0-(n-1), (n-1)-(n-1)] = [-(n-1), n-1]... no
    # j - s: j in [0,n), s in [-(n-1), n-1] → j-s in [-(n-1), 2n-2]... 
    # Valid when 0 <= j-s < n
    
    # Vectorized: create a (n, 2n-1) matrix
    J, S = np.meshgrid(np.arange(n), s, indexing='ij')
    idx1 = J - S  # shape (n, 2n-1)
    idx2 = J + S
    valid1 = (idx1 >= 0) & (idx1 < n)
    valid2 = (idx2 >= 0) & (idx2 < n)
    
    # Clip for safe indexing
    idx1_clipped = np.clip(idx1, 0, n-1)
    idx2_clipped = np.clip(idx2, 0, n-1)
    
    h1 = h[idx1_clipped] * valid1  # (n, 2n-1)
    h2 = h[idx2_clipped] * valid2
    
    term1 = h1 @ w  # (n,) = sum_m h[j-s_m] * w[m] for valid m
    term2 = h2 @ w
    
    grad = dx * (1.0 - term1 - term2)
    return grad
```

Let me be very careful about the CD implementation to make it as efficient as possible.

For the CD, the key optimization is that when we change h[p] by delta d, the new c[m] = c[m] + d*a[m] - d² (where the -d² comes from the diagonal term at m=n-1). So the new max is max(unaffected_max, max_m (c[m] + d*a[m] - d² * I[m==n-1])).

This is a piecewise quadratic function of d, and we can find its minimum using golden section search (which the previous code already does). The key is to make each evaluation O(1) by precomputing the relevant quantities.

Let me also add a "block CD" that updates 2 coordinates at a time. For 2 coordinates p and q, changing h[p] by d1 and h[q] by d2:
c'[m] = c[m] + d1*a_p[m] + d2*a_q[m] + (diagonal corrections)

This is a 2D problem, solvable with a small grid search or nested golden section.

Actually, for the 2D case, the objective is:
f(d1, d2) = max_m (c[m] + d1*a_p[m] + d2*a_q[m] - d1²*I[m==n-1] - d2²*I[m==n-1])

Wait, the diagonal correction: when we change h[p] by d1, the term at m=n-1 (s=0) gets an additional -d1*h[p+d1]... no, it gets more complex.

Actually, for the diagonal term (s=0):
c[n-1] = sum_i h[i](1-h[i]) = sum_i h[i] - sum_i h[i]²

If h[p] changes by d1: c[n-1] changes by d1 - (h[p]²+d1²+2d1*h[p] - h[p]²) = d1 - d1² - 2*d1*h[p]
= d1(1-2*h[p]) - d1²

So for the 2D case at the diagonal: -d1² - d2² (plus linear terms)
For off-diagonal terms: linear in d1, d2 (no quadratic)

So f(d1, d2) = max(unaffected, max_{m≠n-1} (c[m] + d1*a_p[m] + d2*a_q[m]), c[n-1] + d1*(1-2h[p]) + d2*(1-2h[q]) - d1² - d2²)

This is a max of affine functions and a concave quadratic. The minimum is at a point where 2-3 constraints are active. We can solve this with a small convex optimization.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def _sum_eq(x, n):
    return x.sum() - n / 2.0

def _sum_eq_jac(x, n):
    return np.ones(n)

def compute_lse_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    logits = alpha * c
    logits -= logits.max()
    w = np.exp(logits)
    w /= w.sum()
    s = np.arange(2 * n - 1) - (n - 1)
    J, S = np.meshgrid(np.arange(n), s, indexing='ij')
    idx1 = J - S
    idx2 = J + S
    v1 = (idx1 >= 0) & (idx1 < n)
    v2 = (idx2 >= 0) & (idx2 < n)
    i1 = np.clip(idx1, 0, n - 1)
    i2 = np.clip(idx2, 0, n - 1)
    term1 = (h[i1] * v1).dot(w)
    term2 = (h[i2] * v2).dot(w)
    return dx * (1.0 - term1 - term2)

def nesterov_lse(h, n, rng, deadline, max_iters=30000):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    x = h.copy()
    y = h.copy()
    v = np.zeros(n)
    step = 0.01
    for alpha in [200, 500, 1000, 2000, 5000]:
        it = 0
        for it in range(max_iters):
            if time.time() > deadline:
                break
            v = 0.9 * v + compute_lse_grad(y, n, alpha)
            y = x - step * v
            y = project_feasible(y, n)
            c_new = compute_c5(y, n)
            if c_new < best_c:
                best_c = c_new
                best_h = y.copy()
            if it % 200 == 199 and it > 0:
                step *= 0.7
            x = y
        step = 0.01
    return best_h, best_c

def _cd_fval(c_aff, a, diag_mask, unaff_max, d):
    vals = c_aff + d * a
    vals[diag_mask] = c_aff[diag_mask] + d * a[diag_mask] - d * d
    result = float(np.max(vals))
    if unaff_max > result:
        result = unaff_max
    return result

def coord_descent(h, n, deadline, max_passes=500, gs_iters=60):
    h = h.copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    diag = n - 1
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-14 or h0 >= 1.0 - 1e-14:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(vf):
                a[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                a[vb] -= h[jb[vb]]
            diag_mask = (ms == diag)
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            d_lo = -h0
            d_hi = 1.0 - h0
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            f1 = _cd_fval(c_aff, a, diag_mask, unaff_max, d1)
            f2 = _cd_fval(c_aff, a, diag_mask, unaff_max, d2)
            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-16:
                    break
                if f1 < f2:
                    d_hi, d2, f2 = d2, d1, f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    f1 = _cd_fval(c_aff, a, diag_mask, unaff_max, d1)
                else:
                    d_lo, d1, f1 = d1, d2, f2
                    d2 = d_lo + gr * (d_hi - d_lo)
                    f2 = _cd_fval(c_aff, a, diag_mask, unaff_max, d2)
            d_opt = (d_lo + d_hi) / 2.0
            f_opt = _cd_fval(c_aff, a, diag_mask, unaff_max, d_opt)
            f_curr = _cd_fval(c_aff, a, diag_mask, unaff_max, 0.0)
            if f_opt < f_curr - 1e-15:
                h[p] = h0 + d_opt
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def slsqp_optimize(h, n, alphas, maxiter_each=200, deadline=None):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac, 'args': (n,)}]
    for alpha in alphas:
        if deadline and time.time() > deadline:
            break
        try:
            def _obj(x, n=n, alpha=alpha):
                dx = 2.0 / n
                c = np.correlate(x, 1.0 - x, mode='full')
                sh = alpha * c
                mx = sh.max()
                return (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha * dx
            res = minimize(_obj, best_h, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter_each, 'ftol': 1e-14})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def subgradient_phase(h, n, deadline, max_iters=3000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    mom = np.zeros(n)
    beta = 0.5
    step = 0.02
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        m_star = int(np.argmax(c))
        s = m_star - (n - 1)
        grad = np.zeros(n)
        for j in range(n):
            g = 0.0
            if 0 <= j + s < n:
                g += (1.0 - h[j + s])
            if 0 <= j - s < n:
                g -= h[j - s]
            if s == 0:
                g = (1.0 - 2.0 * h[j])
            grad[j] = g
        mom = beta * mom + (1.0 - beta) * grad
        step = 0.02 / (1.0 + 0.005 * it)
        h = project_feasible(h - step * mom, n)
        c_new = compute_c5(h, n)
        if c_new < best_c:
            best_h, best_c = h.copy(), c_new
    return best_h, best_c

def two_point_cd(h, n, deadline, max_passes=20):
    h = h.copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    diag = n - 1
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in range(0, n - 1, 2):
            if time.time() > deadline:
                break
            h0a, h0b = h[p], h[p + 1]
            if h0a <= 1e-14 or h0a >= 1 - 1e-14 or h0b <= 1e-14 or h0b >= 1 - 1e-14:
                continue
            m_lo = max(0, n - 1 - max(p, p+1))
            m_hi = min(2*n-2, n - 1 + min(p, p+1))
            ms = np.arange(m_lo, m_hi + 1)
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2*n-2:
                unaff_max = max(unaff_max, float(c[m_hi+1:].max()))
            aff_a = np.zeros(len(ms))
            aff_b = np.zeros(len(ms))
            for j in [p, p+1]:
                jf = j + ms - (n-1)
                jb = j - ms + (n-1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                arr = aff_a if j == p else aff_b
                if np.any(vf): arr[vf] += (1.0 - h[jf[vf]])
                if np.any(vb): arr[vb] -= h[jb[vb]]
            c_aff = c[ms]
            diag_mask = (ms == diag)
            da_lo, da_hi = -h0a, 1.0 - h0a
            db_lo, db_hi = -h0b, 1.0 - h0b
            def eval2(da, db):
                vals = c_aff + da * aff_a + db * aff_b
                vals[diag_mask] = c_aff[diag_mask] + da*aff_a[diag_mask] + db*aff_b[diag_mask] - da*da - db*db
                return max(float(np.max(vals)), unaff_max)
            best_f = eval2(0.0, 0.0)
            cands = [(0.3, 0.3), (0.3, -0.3), (-0.3, 0.3), (-0.3, -0.3),
                     (0.5, 0.5), (0.5, -0.5), (-0.5, 0.5), (-0.5, -0.5)]
            best_d = (0.0, 0.0)
            for sa in [0.1, 0.3, 0.5, 0.8]:
                for sb in [0.1, 0.3, 0.5, 0.8]:
                    for sgn_a in [-1, 1]:
                        for sgn_b in [-1, 1]:
                            da = sgn_a * sa * (1.0 - h0a)
                            db = sgn_b * sb * (1.0 - h0b)
                            da = np.clip(da, -h0a, 1.0 - h0a)
                            db = np.clip(db, -h0b, 1.0 - h0b)
                            f = eval2(da, db)
                            if f < best_f:
                                best_f = f
                                best_d = (da, db)
            if best_f < eval2(0.0, 0.0) - 1e-14:
                h[p] = h0a + best_d[0]
                h[p+1] = h0b + best_d[1]
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)
```
