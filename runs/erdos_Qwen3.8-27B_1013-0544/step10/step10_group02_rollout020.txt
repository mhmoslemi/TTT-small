            grad += pk * (1.0 - 2.0 * h)
        else:

                grad[shift:] -= pk * bwd
            else:
                s = -shift

                grad[:n-s] += pk * fwd
                # Backward: i from s to n-1
                bwd = h[:n-s]
                grad[s:] -= pk * bwd

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def solve_1d_fast(c, h, p, n):
    h0 = h[p]
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf): a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb): a[vb] -= h[jb[vb]]
    c_base = c[ms].copy()
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    if has_diag:
        c_diag = c_base[diag_idx[0]]
        a[diag_idx] = 0.0
    unaff = 0.0
    if m_min > 0: unaff = float(np.max(c[:m_min]))
    if m_max < 2 * n - 2: unaff = max(unaff, float(np.max(c[m_max + 1:])))
    L_lo, L_hi = unaff, float(np.max(c_base))
    for _ in range(30):
        if L_hi - L_lo < 1e-14: break
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        feasible = True
        zmq = np.abs(a) <= 1e-15
        if np.any(zmq) and np.any(c_base[zmq] > L + 1e-14): feasible = False
        nmk = a < -1e-15
        pmk = a > 1e-15
        if feasible and np.any(nmk):
            x_lo = max(x_lo, float(np.max((L - c_base[nmk]) / a[nmk])))
        if feasible and np.any(pmk):
            x_hi = min(x_hi, float(np.min((L - c_base[pmk]) / a[pmk])))
        if x_lo > x_hi + 1e-13: feasible = False
        if feasible and has_diag:
            Cq = L - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:

                r2 = (1.0 + np.sqrt(disc)) / 2.0
                xc_lo, xc_hi = max(x_lo, 0.0), min(x_hi, 1.0)
                if not (xc_lo < r1 and xc_hi > r2): feasible = False
        if feasible: L_hi = L
        else: L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    nmk = a < -1e-15
    pmk = a > 1e-15
    if np.any(nmk): x_lo = max(x_lo, float(np.max((L - c_base[nmk]) / a[nmk])))
    if np.any(pmk): x_hi = min(x_hi, float(np.min((L - c_base[pmk]) / a[pmk])))
    x_lo, x_hi = max(x_lo, 0.0), min(x_hi, 1.0)
    if has_diag:

                elif x_lo < r1 and x_hi > r2:
                    if h0 - r1 <= r2 - h0: x_hi = min(x_hi, r1)
                    else: x_lo = max(x_lo, r2)

    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def exact_cd(h, n, max_passes, deadline, rng, order_mode='seq'):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for it in range(max_passes):
        if time.time() > deadline: break
        improved = False
        if order_mode == 'seq':
            order = list(range(n))
        else:
            order = rng.permutation(n)
        for p in order:
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13: continue
            xn = solve_1d_fast(c, h, p, n)
            if abs(xn - h0) < 1e-12: continue
            delta = xn - h0
            m_min = n - 1 - p
            m_max = 2 * n - 2 - p
            ms = np.arange(m_min, m_max + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(vf): a[vf] += (1.0 - h[jf[vf]])
            if np.any(vb): a[vb] -= h[jb[vb]]
            c[ms] += delta * a
            diag_i = np.where(ms == (n - 1))[0]
            if len(diag_i) > 0: c[ms[diag_i[0]]] -= delta * delta
            h[p] = xn
            improved = True
        if not improved: break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def smooth_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    shifted = alpha * c - alpha * c.max()
    w = np.exp(shifted)
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-16: continue
        wk = w[k]
        p_lo = max(0, k)
        p_hi = min(n - 1, k + n - 1)
        if p_lo <= p_hi:
            grad[p_lo:p_hi+1] += wk * (1.0 - h[p_lo-k:p_hi-k+1])
        p_lo2 = max(0, -k)
        p_hi2 = min(n - 1, n - 1 - k)
        if p_lo2 <= p_hi2:
            grad[p_lo2:p_hi2+1] -= wk * h[p_lo2+k:p_hi2+k+1]
    return grad

def nesterov_opt(h, n, alpha, max_iter, deadline, lr_init=None):
    h = project_feasible(h.copy(), n)
    h_prev = h.copy()
    if lr_init is None:
        lr = 2.0 / (alpha * n) * 50
    else:
        lr = lr_iter
    beta = 1.0
    best_h, best_c = h.copy(), compute_c5(h, n)
    for it in range(max_iter):
        if time.time() > deadline: break
        y = h + (beta - 1.0) / (beta + 1.0) * (h - h_prev)
        y = project_feasible(y, n)
        g = smooth_grad(y, n, alpha)
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-8:
            step = lr / np.sqrt(it + 1)
            h_new = h - step * g / g_norm * np.min(1.0, g_norm)
        else:
            h_new = h.copy()
        h_new = project_feasible(h_new, n)
        c_now = compute_c5(h_new, n)
        if c_now < best_c:
            best_h, best_c = h_new.copy(), c_now
        h_prev = h
        h = h_new
        beta = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
    return best_h, best_c
