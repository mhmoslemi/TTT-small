        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        feasible = True
        if np.any(zero_mask) and np.any(c_base[zero_mask] > L + 1e-14):
            feasible = False
        if feasible and np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if feasible and np.any(pos_mask):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        if x_lo > x_hi + 1e-12:
            feasible = False
        if feasible and has_diag:
            c_diag = c_base[diag_idx[0]]
            a_d = a[diag_idx[0]]
            disc = a_d * a_d + 4.0 * (c_diag - L)
            if disc >= 0:
                sq = np.sqrt(disc)
                r1 = (a_d - sq) / 2.0
                r2 = (a_d + sq) / 2.0
                if x_lo > h0 + r1 and x_hi < h0 + r2:
                    feasible = False
        if feasible:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        c_diag = c_base[diag_idx[0]]
        a_d = a[diag_idx[0]]
        disc = a_d * a_d + 4.0 * (c_diag - L)
        if disc >= 0:
            sq = np.sqrt(disc)
            r1 = (a_d - sq) / 2.0
            r2 = (a_d + sq) / 2.0
            if x_lo <= h0 + r1:
                x_hi = min(x_hi, h0 + r1)
            elif x_hi >= h0 + r2:
                x_lo = max(x_lo, h0 + r2)
            else:
                if abs(x_lo - (h0 + r1)) <= abs(x_hi - (h0 + r2)):
                    x_hi = min(x_hi, h0 + r1)
                else:
                    x_lo = max(x_lo, h0 + r2)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def cd_pass(h, c, n, order):
    improved = False
    for p in order:
        h0 = h[p]
        if h0 < 1e-13 or h0 > 1.0 - 1e-13:
            continue
        xn = solve_coord(c, h, p, n)
        if abs(xn - h0) > 1e-13:
            delta = xn - h0
            m_min = max(0, n - 1 - p)
            m_max = min(2 * n - 2, 2 * n - 2 - p)
            ms = np.arange(m_min, m_max + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(vf):
                a[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                a[vb] -= h[jb[vb]]
            c[ms] += delta * a
            diag_i = np.where(ms == (n - 1))[0]
            if len(diag_i) > 0:
                c[ms[diag_i[0]]] -= delta * delta
            h[p] = xn
            improved = True
    return improved

def run_cd(h, n, max_passes, deadline, rng, mode='fwd'):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    best_c = compute_c5(h, n)
    stall = 0
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        if mode == 'fwd':
            order = range(n)
        elif mode == 'bwd':
            order = range(n - 1, -1, -1)
        else:
            order = rng.permutation(n)
        improved = cd_pass(h, c, n, order)
        if improved:
            h = project_feasible(h, n)
            if abs(h.sum() - n / 2.0) > 1e-10:
                c = np.correlate(h, 1.0 - h, mode='full')
            cur_c = compute_c5(h, n)
            if cur_c < best_c:
                best_c = cur_c
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        if stall > 30:
            break
    return h, compute_c5(h, n)

def slsqp_opt(h0, n, alphas, maxiter, deadline):
    best_h, best_c = h0.copy(), compute_c5(h0, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def interp_to(h_src, n_src, n_tgt):
    x_src = np.linspace(0, 2, n_src, endpoint=False)
    x_tgt = np.linspace(0, 2, n_tgt, endpoint=False)
    return project_feasible(np.interp(x_tgt, x_src, h_src), n_tgt)

def fourier_init(n, K, rng):
    x = np.linspace(0, 2, n, endpoint=False)
    h = np.zeros(n)
    for k in range(1, K + 1):
        amp = rng.normal(0, 1.0 / np.sqrt(k))
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(2 * np.pi * k * x / 2.0 + phase)
    h = 0.5 + 0.4 * h
    return project_feasible(h, n)

def variance_reducing_cd(h, n, max_passes, deadline, rng, beta=0.3):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c_mean = c.mean()
        c_spread = c.max() - c.min()
        if c_spread < 1e-10:
            break
        order = rng.permutation(n)
        improved = False
        for p in order:
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            xn = solve_coord(c, h, p, n)
            if abs(xn - h0) > 1e-13:
                delta = xn - h0
                m_min = max(0, n - 1 - p)
                m_max = min(2 * n - 2, 2 * n - 2 - p)
                ms = np.arange(m_min, m_max + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                c[ms] += delta * a
                diag_i = np.where(ms == (n - 1))[0]
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n_best = len(h_best)
    except Exception:
        h_best = np.full(700, 0.5)
        n_best = 700
    h_best = project_feasible(h_best, n_best)
    best_c = compute_c5(h_best, n_best)

    # Phase 1: Quick SLSQP warmup from parent
    h1, c1 = slsqp_opt(h_best, n_best, [50000, 500000], 80, deadline)
    if c1 < best_c:
        h_best, best_c = h1, c1

    # Phase 2: Coarse-to-fine multi-scale
    coarse_ns = [70, 140, 280]
    for nc in coarse_ns:
        if time.time() > deadline - 400:
            break
        h_c = interp_to(h_best, n_best, nc)
        h_c, c_c = run_cd(h_c, nc, 300, deadline, rng, mode='rand')
        h_c, c_c = variance_reducing_cd(h_c, nc, 100, deadline, rng)
        # Interpolate back up
        h_up = interp_to(h_c, nc, n_best)
        h_up, c_up = run_cd(h_up, n_best, 80, deadline, rng, mode='rand')
        if c_up < best_c:
            h_best, best_c = h_up, c

        # Also try at the coarse level directly
        if c_c * (n_best / nc) < best_c * 1.01:  # promising
            h_up2 = interp_to(h_c, nc, n_best)
            h_up2, c_up2 = slsqp_opt(h_up2, n_best, [100000], 60, deadline)
            h_up2, c_up2 = run_cd(h_up2, n_best, 100, deadline, rng, mode='rand')
            if c_up2 < best_c:
                h_best, best_c = h_up2, c_up2

    # Phase 3: Intensive multi-ordering CD at full resolution
    for mode in ['fwd', 'bwd', 'rand', 'rand', 'rand']:
        if time.time() > deadline - 200:
            break
        h3, c3 = run_cd(h_best, n_best, 150, deadline, rng, mode=mode)
        if c3 < best_c:
            h_best, best_c = h3, c3

    # Phase 4: Variance-reducing CD
    if time.time() < deadline - 150:
        h4, c4 = variance_reducing_cd(h_best, n_best, 200, deadline, rng)
        if c4 < best_c:
            h_best, best_c = h4, c4

    # Phase 5: SLSQP polish with high alpha
    if time.time() < deadline - 100:
        h5, c5 = slsqp_opt(h_best, n_best, [1000000, 5000000], 100, deadline)
        if c5 < best_c:
            h_best, best_c = h5, c5

    # Phase 6: Perturbation restarts
    for r in range(10):
        if time.time() > deadline - 60:
            break
        temp = 0.005 + 0.01 * r
        hp = project_feasible(h_best + rng.normal(0, temp, n_best), n_best)
        h6, c6 = run_cd(hp, n_best, 100, deadline, rng, mode='rand')
        if c6 < best_c:
            h_best, best_c = h6, c6

    # Phase 7: Final polish
    if time.time() < deadline - 5:
        h7, c7 = run_cd(h_best, n_best, 100, deadline, rng, mode='fwd')
        if c7 < best_c:
            h_best, best_c = h7, c7
        h7, c7 = run_cd(h_best, n_best, 50, deadline, rng, mode='bwd')
        if c7 < best_c:
            h_best, best_c = h7, c7

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
