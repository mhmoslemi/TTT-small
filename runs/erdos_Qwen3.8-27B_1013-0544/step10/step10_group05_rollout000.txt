import time
import numpy as np
from scipy.optimize import minimize
from scipy.signal import fftconvolve


def compute_c5(h, n):
    dx = 2.0 / float(n)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def project_box_sum(z, target):
    z = np.asarray(z, dtype=float)
    n = z.size
    lo = -float(np.min(z))
    hi = 1.0 - float(np.max(z))
    for _ in range(55):
        mid = 0.5 * (lo + hi)
        if float(np.clip(z + mid, 0.0, 1.0).sum()) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z + 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(5):
        diff = target - float(h.sum())
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.any():
            h[mask] += diff / float(mask.sum())
        else:
            h += diff / float(n)
        h = np.clip(h, 0.0, 1.0)
    return h


def corr_fft(h, n):
    if n < 192:
        return np.correlate(h, 1.0 - h, mode='full')
    return fftconvolve(h, (1.0 - h)[::-1], mode='full')[::-1]


def grad_from_w(h, n, w):
    N = n - 1
    A = fftconvolve(1.0 - h, w[::-1], mode='full')[N:N + n]
    B = fftconvolve(h, w, mode='full')[N:N + n]
    return A - B


def lbfgs_fun(x, n, alpha, pen):
    c = corr_fft(x, n)
    m = float(c.max())
    s = alpha * (c - m)
    e = np.exp(s)
    z = float(e.sum())
    val = m + np.log(z) / alpha
    d = float(x.sum() - 0.5 * n)
    val += pen * d * d
    return float(val)


def lbfgs_jac(x, n, alpha, pen):
    c = corr_fft(x, n)
    m = float(c.max())
    s = alpha * (c - m)
    e = np.exp(s)
    w = e / float(e.sum())
    g = grad_from_w(x, n, w) / alpha
    d = float(x.sum() - 0.5 * n)
    if pen:
        g = g + (2.0 * pen * d) * np.ones_like(x)
    return g


def lbfgs_opt(h, n, alphas, maxiter, deadline, pen=1000.0):
    h = project_box_sum(h, 0.5 * n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    if n <= 0 or maxiter <= 0:
        return best_h, best_c
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(lbfgs_fun, h, args=(n, float(alpha), pen),
                           jac=lbfgs_jac, method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': int(maxiter), 'ftol': 1e-14,
                                    'gtol': 1e-10, 'maxcor': 10, 'maxls': 20,
                                    'disp': False})
            hp = project_box_sum(res.x, 0.5 * n)
            c = compute_c5(hp, n)
            if c < best_c - 1e-12:
                best_h = hp
                best_c = c
                h = hp
        except Exception:
            pass
    return best_h, best_c

    if diag >= 0:
        c0d = float(base[diag])
        ad = float(a[diag])
        disc = ad * ad + 4.0 * (c0d - hi)
        if disc >= 0.0:
            sq = np.sqrt(disc)
            r1 = 0.5 * (ad - sq)
            r2 = 0.5 * (ad + sq)
            f1 = h0 + r1
            f2 = h0 + r2
            if f1 < x < f2:
                choose = None
                if xlo - 1e-12 <= f1 <= xhi + 1e-12:
                    choose = f1
                if xlo - 1e-12 <= f2 <= xhi + 1e-12:
                    if choose is None or abs(f2 - h0) < abs(choose - h0):
                        choose = f2
                if choose is not None:
                    x = float(np.clip(choose, xlo, xhi))
                else:
                    x = float(np.clip(xlo if h0 <= f1 else xhi, xlo, xhi))
    x = float(np.clip(x, 0.0, 1.0))
    if abs(x - h0) <= 1e-14:
        return h0, None
    delta = x - h0
    tmp = base + delta * a
    if diag >= 0:
        tmp[diag] -= delta * delta
    if float(tmp.max()) >= cur - 1e-11:
        return h0, None
    return x, a


def exact_cd(h, n, max_passes, deadline, reverse=False):
    h = project_box_sum(h, 0.5 * n)
    meta = coord_meta(n)
    N = n - 1
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        changed = False
        rng = range(n - 1, -1, -1) if reverse else range(n)
        for p in rng:
            if time.time() > deadline:
                break
            h0 = float(h[p])
            x, a = solve_coord(c, h, p, meta)
            if a is not None and abs(x - h0) > 1e-14:
                low, high = meta[p][0], meta[p][1]
                delta = x - h0
                c[low:high + 1] += delta * a
                if meta[p][8] >= 0:
                    c[N] -= delta * delta
                h[p] = x
                changed = True
        if changed:
            h = project_box_sum(h, 0.5 * n)
        else:
            break
    h = project_box_sum(h, 0.5 * n)
    return h, compute_c5(h, n)


def fourier_init(n, rng, K=10):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for k in range(1, K + 1):
        amp = 0.45 / np.sqrt(float(k)) * rng.normal(0.0, 1.0)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        h += amp * np.cos(0.5 * np.pi * k * x + ph)
    return project_box_sum(np.clip(h, 0.0, 1.0), 0.5 * n)


def interp_h(hs, ns, nt):
    xs = np.linspace(0.0, 2.0, ns, endpoint=False)
    xt = np.linspace(0.0, 2.0, nt, endpoint=False)
    return project_box_sum(np.interp(xt, xs, hs), 0.5 * nt)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 8.0
    rng = np.random.RandomState(seed)
    try:
        best_h = np.array(initial_h_values, dtype=float).copy()
        best_n = len(best_h)
    except Exception:
        best_n = 700
        best_h = fourier_init(best_n, rng, 12)
    if best_n > 1100:
        best_h = interp_h(best_h, best_n, 1000)
        best_n = 1000
    best_h = project_box_sum(best_h, 0.5 * best_n)
    best_c = compute_c5(best_h, best_n)

    hr = project_box_sum(0.5 * (best_h + best_h[::-1]), 0.5 * best_n)
    cr = compute_c5(hr, best_n)
    if cr < best_c - 1e-12:
        best_h, best_c = hr, cr
    for sh in (1, 2, 3):
        if time.time() > deadline - 800:
            break
        hsh = np.roll(best_h, sh)
        csh = compute_c5(hsh, best_n)
        if csh < best_c - 1e-12:
            best_h, best_c = hsh, csh

    stop1 = min(deadline - 550.0, time.time() + 90.0)
    h1, c1 = lbfgs_opt(best_h, best_n, (100.0, 500.0, 2000.0), 55, stop1, 1000.0)
    if c1 < best_c - 1e-12:
        best_h, best_c = h1, c1

    stop2 = min(deadline - 300.0, time.time() + 330.0)
    mpass = 35 if best_n <= 750 else 22
    h2, c2 = exact_cd(best_h, best_n, mpass, stop2, reverse=False)
    if c2 < best_c - 1e-12:
        best_h, best_c = h2, c1 < best_c and c2 or c2

    stop3 = min(deadline - 230.0, time.time() + 55.0)
    h3, c3 = lbfgs_opt(best_h, best_n, (5000.0, 20000.0), 40, stop3, 1200.0)
    if c3 < best_c - 1e-12:
        best_h, best_c = h3, c3

    stop4 = min(deadline - 160.0, time.time() + 110.0)
    h4, c4 = exact_cd(best_h, best_n, 14, stop4, reverse=True)
    if c4 < best_c - 1e-12:
        best_h, best_c = h4, c4

    seq = []
    for d in (-1, 1, -2, 2, -5, 5, -10, 10, -20, 20):
        nt = best_n + d
        if 200 <= nt <= 1200 and nt != best_n:
            seq.append(nt)
    for nt in (300, 400, 500, 600, 700, 800, 900, 1000):
        if 200 <= nt <= 1200 and nt != best_n:
            seq.append(nt)
    seen = set()
    cands = []
    for nt in seq:
        if nt not in seen:
            seen.add(nt)
            cands.append(nt)

    for nt in cands:
        if time.time() > deadline - 80.0:
            break
        sub = min(deadline - 70.0, time.time() + 75.0)
        h_try = interp_h(best_h, best_n, nt)
        h_try, c_try = lbfgs_opt(h_try, nt, (300.0, 3000.0), 35, sub - 10.0, 900.0)
        if c_try < best_c - 1e-12:
            best_h, best_c, best_n = h_try, c_try, nt
        if time.time() > sub - 12.0:
            continue
        h_try, c_try = exact_cd(h_try, nt, 10, min(sub - 3.0, deadline - 50.0), reverse=False)
        if c_try < best_c - 1e-12:
            best_h, best_c, best_n = h_try, c_try, nt

    for r in range(4):
        if time.time() > deadline - 60.0:
            break
        sub = min(deadline - 50.0, time.time() + 80.0)
        h_try = fourier_init(best_n, rng, 7 + 2 * r)
        h_try, c_try = lbfgs_opt(h_try, best_n, (300.0, 3000.0), 35, sub - 15.0, 900.0)
        if time.time() > sub - 10.0:
            continue
        h_try, c_try = exact_cd(h_try, best_n, 8, sub - 3.0, reverse=False)
        if c_try < best_c - 1e-12:
            best_h, best_c = h_try, c_try

    h_final, c_final = exact_cd(best_h, best_n, 28, max(t0 + 1.0, deadline - 2.0), reverse=False)
    if c_final < best_c - 1e-12:
        best_h, best_c = h_final, c_final

    best_h = project_box_sum(best_h, 0.5 * len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
