def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def solve_coord(c, h, p, n):
    # Exact 1D coordinate descent
    h0 = h[p]
    m_min = max(0, n - 1 - p)
    m_max = min(2 * n - 2, 2 * n - 2 - p)
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms].copy()
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    unaff = 0.0
    if m_min > 0:
        unaff = float(np.max(c[:m_min]))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    cur_max = float(np.max(c))
    neg_mask = a < -1e-14
    pos_mask = a > 1e-14
    zero_mask = np.abs(a) <= 1e-14
    L_lo, L_hi = unaff, cur_max
    for _ in range(50):
        if L_hi - L_lo < 1e-14:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        feasible = True
        if np.any(zero_mask) and np.any(c_base[zero_mask] > L + 1e-14):
            feasible = False
        if feasible and np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if feasible and np.any(pos_mask):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        if x_lo > x_hi + 1e-12:
            feasible = False
        if feasible and has_diag:
            c_diag = c_base[diag_idx[0]]
            a_d = a[diag_idx[0]]
            disc = a_d * a_d + 4.0 * (c_diag - L)
            if disc >= 0:
                sq = np.sqrt(disc)
                r1 = (a_d - sq) / 2.0
                r2 = (a_d + sq) / 2.0
                if x_lo > h0 + r1 and x_hi < h0 + r2:
                    feasible = False
        if feasible:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:

            r2 = (a_d + sq) / 2.0
            if x_lo <= h0 + r1:
                x_hi = min(x_hi, h0 + r1)
            elif x_hi >= h0 + r2:
                x_lo = max(x_lo, h0 + r2)
            else:
                if abs(x_lo - (h0 + r1)) <= abs(x_hi - (h0 + r2)):
                    x_hi = min(x_hi, h0 + r1)
                else:
                    x_lo = max(x_lo, h0 + r2)
    if x_lo > x_hi:

    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))
