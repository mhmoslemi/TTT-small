def project_feasible(h,n):

def compute_c5(h, n):

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def project_feasible(h, n):

            tot = s
            if tot < 1e-15:
                break
            h = h + (d / tot) * h
    return np.clip(h, 0.0, 1.0)

                sum_h = float(h.sum())
                c = np.correlate(h, 1.0 - h, mode='full')
                cmax = float(np.max(c))

        if not improved:
            break

        h = project_feasible(h, n)
        c = np.correlate(h, 1.0 - h, mode='full')
        cmax = float(np.max(c))

    return h, cmax


def smooth_obj(x):
    n = G_N
    alpha = G_ALPHA
    c = np.correlate(x, 1.0 - x, mode='full')
    y = alpha * c
    m = float(np.max(y))
    e = np.exp(y - m)
    lse = (m + float(np.log(np.sum(e)))) / alpha
    return lse * G_DX


def smooth_jac(x):
    n = G_N
    alpha = G_ALPHA
    dx = G_DX
    c = np.correlate(x, 1.0 - x, mode='full')
    y = alpha * c
    m = float(np.max(y))
    e = np.exp(y - m)
    w = e / float(np.sum(e))

    base = 1.0 - x[::-1]
    grad = np.empty(n)

    for p in range(n):
        ww = w[p:p + n]
        g = float(np.dot(ww, base))

        l = n - 1 - 2 * p
        if l < 0:
            l = 0
        u = 2 * n - 2 - 2 * p
        if u >= n:
            u = n - 1
        if l <= u:
            v = 2 * p + l - (n - 1)
            g -= float(np.dot(ww[l:u + 1], x[v:v + (u - l + 1)]))

        grad[p] = dx * g

    return grad


def optimize_smooth(h, n, alphas, deadline, maxit):
    global G_N, G_ALPHA, G_DX
    G_N = n
    G_DX = 2.0 / n

    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    work = best_h.copy()

    bounds = [(0.0, 1.0)] * n
    A_eq = np.ones((1, n), dtype=float)
    b_eq = np.array([0.5 * n], dtype=float)

    for al in alphas:
        if time.time() > deadline - 20:
            break
        G_ALPHA = float(al)
        x0 = project_feasible(work, n)
        try:
            res = minimize(smooth_obj, x0, jac=smooth_jac, method='L-BFGS-B',
                           bounds=bounds, A_eq=A_eq, b_eq=b_eq,
                           options={'maxiter': int(maxit), 'ftol': 1e-12, 'gtol': 1e-9})
            if hasattr(res, 'x') and np.all(np.isfinite(res.x)):
                cand = project_feasible(res.x, n)
                cc = compute_c5(cand, n)
                if cc < best_c:
                    best_c = cc
                    best_h = cand.copy()
                    work = cand.copy()
                elif cc < best_c + 0.002:
                    work = cand.copy()
                else:
                    work = best_h.copy()
        except Exception:
            work = best_h.copy()

    return best_h, best_c


def perturb_h(h, n, scale, rng, fraction=0.0):
    hp = h + rng.normal(0.0, scale, n)
    if fraction > 0.0:
        m = rng.rand(n) < fraction
       _vals = rng.uniform(0.0, 1.0, n)
        hp[m] = _vals[m]
    return project_feasible(hp, n)


def refine_candidate(h, n, deadline, cd1, alphas, maxit, cd2, rng=None):
    h, c = exact_cd(h, n, cd1, deadline)

    if rng is not None and time.time() < deadline - 40:
        hr, cr = exact_cd(h, n, 5, deadline, random_order=True, rng=rng)
        if cr < c:
            h, c = hr, cr

    if time.time() < deadline - 35:
        ho, co = optimize_smooth(h, n, alphas, deadline, maxit)
        if co < c:
            h, c = ho, co

    h, c = exact_cd(h, n, cd2, deadline)

    if rng is not None and time.time() < deadline - 10:
        hr, cr = exact_cd(h, n, 4, deadline, random_order=True, rng=rng)
        if cr < c:
            h, c = hr, cr

    return h, c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget = float(budget_s)
    reserve = min(20.0, 0.1 * max(1.0, budget))
    deadline = t0 + max(1.0, budget - reserve)
    rng = np.random.RandomState(seed)

    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n_best = len(h_best)
    except Exception:
        n_best = 700
        x = np.linspace(0.0, 2.0, n_best, endpoint=False)
        h_best = 0.5 + 0.1 * np.cos(math.pi * x)
        h_best = project_feasible(h_best, n_best)

    h_best = project_feasible(h_best, n_best)
    best_c = compute_c5(h_best, n_best)

    h0, c0 = refine_candidate(h_best, n_best, deadline, 10,
                              [20.0, 100.0, 500.0, 2000.0], 30, 25, rng)
    if c0 < best_c:
        h_best, best_c = h0, c0

    n_list = [350, 500, 600, 650, 750, 800, 450, 900]
    for n_try in n_list:
        if time.time() > deadline - 70:
            break
        if int(n_try) == int(n_best):
            continue
        x_old = np.linspace(0.0, 2.0, n_best, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_try, endpoint=False)
        h_try = np.interp(x_new, x_old, h_best)
        h_try = project_feasible(h_try, n_try)
        h_try = perturb_h(h_try, n_try, 0.004, rng, 0.0)
        ht, ct = refine_candidate(h_try, n_try, deadline, 6,
                                  [20.0, 100.0, 500.0], 25, 15, rng)
        if ct < best_c:
            h_best, best_c, n_best = ht, ct, n_try

    for scale, frac in [(0.003, 0.0), (0.008, 0.02), (0.02, 0.05), (0.05, 0.10)]:
        if time.time() > deadline - 45:
            break
        n_cur = len(h_best)
        hp = perturb_h(h_best, n_cur, scale, rng, frac)
        ht, ct = refine_candidate(hp, n_cur, deadline, 4,
                                  [50.0, 300.0, 1500.0], 20, 10, rng)
        if ct < best_c:
            h_best, best_c = ht, ct

    if time.time() < deadline - 2:
        ht, ct = exact_cd(h_best, len(h_best), 100000, deadline)
        if ct < best_c:
            h_best, best_c = ht, ct

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, int(len(h_best))
