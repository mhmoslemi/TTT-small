```python
def sa_01(h, n, iterations, deadline, rng):
    """Simulated annealing for 0-1 Erdős overlap problem."""
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    M = float(c.max())
    best_M = M
    best_h = h.copy()
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    T_start = M * 0.1  # Initial temperature
    T_end = M * 0.001   # Final temperature
    
    for it in range(iterations):
        if time.time() > deadline:
            break
            
        # Annealing schedule
        T = T_start * (T_end / T_start) ** (it / iterations)
        
        # Pick random pair to swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        # Swap
        h[i], h[j] = 0, 1
        c_new = np.correlate(h, 1.0 - h, mode='full')
        new_M = float(c_new.max())
        
        if new_M <= best_M or rng.random() < np.exp((best_M - new_M) / max(T, 1e-10)):
            # Accept
            c = c_new
            M = new_M
            if new_M < best_M:
                best_M = new_M
                best_h = h.copy()
            # Update lists
            ones[ones == i] = j
            zeros[zeros == j] = i
        else:
            # Reject
            h[i], h[j] = 1, 0
    
    return best_h, best_M * 2.0 / n
```

import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14: break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14: break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def round_to_01(h, n, rng):
    target = n // 2
    h01 = (h >= 0.5).astype(float)
    # Adjust
    while h01.sum() > target:
        ones = np.where(h01 == 1)[0]

        flip = ones[np.argmin(h[ones])]

        h01[flip] = 1
    return h01

    zeros_list = np.where(h == 0)[0].copy()

    no_improve = 0
    improve_interval = 5000

        progress = min(1.0, n_iter / 500000.0)
        T = t_start * (t_end / t_start) ** progress

        j_idx = rng.randint(zeros_list.shape[0])
        i = ones_list[i_idx]
        j = zeros_list[j_idx]
        
        h[i], h[j] = 0, 1
        c_new = np.correlate(h, 1 - h, mode='full')
        new_M = int(c_new.max())
        
        delta = new_M - M
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            c = c_new
            M = new_M
            # Update lists
            ones_list[i_idx] = j
            zeros_list[j_idx] = i
            if new_M < best_M:
                best_M = new_M
                best_h = h.copy()
                no_improve = 0
            else:

            # Restart with higher temperature
            T = t_start
            no_improve = 0
            improve_interval = max(2000, improve_interval // 2)

def coord_descent_frac(h, n, deadline, max_passes=50):
    """Coordinate descent for fractional solution."""
    h = h.copy()
    best_c = compute_c5(h, n)
    target = n / 2.0
    
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1 - h, mode='full')
        improved = False
        # Try adjusting each coordinate
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-10 or h0 >= 1 - 1e-10:
                continue
            # Binary search for optimal h[p] in [0, 1]
            # The objective as a function of h[p] is piecewise linear
            # (max of linear functions)
            # We can find the optimum by checking the "breakpoints"
            
            # For each lag k, the contribution of h[p] to c(k) is:
            # c(k) has a term involving h[p] that is linear in h[p]
            # The max of these is a convex piecewise linear function of h[p]
            # We want to minimize it, so the optimum is at a breakpoint or at a boundary
            
            # Simpler: ternary search (since it's convex in h[p])
            lo, hi = 0.0, 1.0
            for _ in range(25):
                m1 = lo + (hi - lo) / 3
                m2 = hi - (hi - lo) / 3
                h[p] = m1
                c1 = np.max(np.correlate(h, 1 - h, mode='full'))
                h[p] = m2
                c2 = np.max(np.correlate(h, 1 - h, mode='full'))
                if c1 < c2:
                    hi = m2
                else:
                    lo = m1
            h[p] = (lo + hi) / 2
            c_val = compute_c5(h, n)
            if c_val < best_c - 1e-12:
                improved = True
            else:
                h[p] = h0
        
        if improved:
            h = project_feasible(h, n)
            best_c = compute_c5(h, n)
        else:
            break
    
    return h, best_c
