    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0

        if abs(diff) < 1e-14:
            break
        if diff > 0:

            tot = room.sum()

            if tot < 1e-15:

def slsqp_target(h, n, target, maxiter=500):

    return new_h, compute_c5(new_h, n)

def equalize_step(h, n, top_k=3, step_size=0.01):
    """Move h to equalize the top correlations"""
    c = np.correlate(h, 1.0 - h, mode='full')
    top_lags = np.argsort(c)[-top_k:]
    # Combined gradient (negative) for reducing top lags
    grad = np.zeros(n)
    for k_idx in top_lags:
        s = k_idx - (n - 1)
        for j in range(n):
            g = 0.0
            if 0 <= j + s < n:
                g += (1.0 - h[j + s])
            if 0 <= j - s < n:
                g -= h[j - s]
            if s == 0:
                g = 1.0 - 2.0 * h[j]
            grad[j] += g
    # Normalize
    gn = np.linalg.norm(grad)
    if gn > 1e-15:
        h = h - step_size * grad / gn
    h = project_feasible(h, n)
    return h

def anneal_quantized(h, n, deadline, n_levels=20, n_iters=5000):

    h = h.copy()

    T_start = 0.05

    curr_c = best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def _sum_eq(x, n):
    return x.sum() - n / 2.0

def _sum_eq_jac(x, n):
    return np.ones(n)

def lse_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    if m <= 0:
        return 0.0
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * dx

def target_obj(x, n, T_raw):
    c = np.correlate(x, 1.0 - x, mode='full')
    excess = np.maximum(0.0, c - T_raw)
    return float(np.sum(excess * excess))

def _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d):
    vals = c_aff + d * a
    if has_diag:
        vals[diag_idx] = c_aff[diag_idx] + d * (1.0 - h0 - d)
    result = float(np.max(vals))
    if unaff_max > result:
        result = unaff_max
    return result

                break
            h0 = h[p]
            if h0 <= 1e-13 or h0 >= 1.0 - 1e-13:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(vf):
                a[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                a[vb] -= h[jb[vb]]
            dp = np.where(ms == (n - 1))[0]
            has_diag = len(dp) > 0
            diag_idx = int(dp[0]) if has_diag else -1
            c_aff = c[ms]
            unaff = 0.0
            if m_lo > 0:
                unaff = max(unaff, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff = max(unaff, float(c[m_hi+1:].max()))
            d_lo, d_hi = -h0, 1.0 - h0
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff, h0, d1)
            f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff, h0, d2)
            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-15:
                    break
                if f1 < f2:
                    d_hi, d2, f2 = d2, d1, f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff, h0, d1)
                else:
                    d_lo, d1, f1 = d1, d2, f2
                    d2 = d_lo + gr * (d_hi - d_lo)
                    f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff, h0, d2)
            d_opt = (d_lo + d_hi) / 2.0
            if _cd_fval(c_aff, a, diag_idx, has_diag, unaff, h0, d_opt) < \
               _cd_fval(c_aff, a, diag_idx, has_diag, unaff, h0, 0.0) - 1e-14:
                h[p] = h0 + d_opt
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def slsqp_lse(h, n, alphas, maxiter_each=400):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac, 'args': (n,)}]
    for alpha in alphas:
        if time.time() > best_h is None:
            break
        try:
            res = minimize(lse_obj, best_h, args=(n, alpha), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter_each, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def slsqp_target(h, n, target_c5, maxiter=400):
    T_raw = target_c5 * n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac, 'args': (n,)}]
    try:
        res = minimize(target_obj, h, args=(n, T_raw), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-20})
        new_h = project_feasible(res.x, n)
        return new_h, compute_c5(new_h, n)
    except Exception:
        return h.copy(), compute_c5(h, n)

def equalize_step(h, n, top_k=4, step_size=0.004):
    c = np.correlate(h, 1.0 - h, mode='full')
    top_lags = np.argsort(c)[-top_k:]
    grad = np.zeros(n)
    for k_idx in top_lags:
        s = k_idx - (n - 1)
        if s == 0:
            grad += (1.0 - 2.0 * h)
        else:
            if s > 0:
                for j in range(s, n):
                    grad[j] += (1.0 - h[j - s])
                for j in range(0, n - s):
                    grad[j] -= h[j + s]
            else:
                s2 = -s
                for j in range(0, n - s2):
                    grad[j] += (1.0 - h[j + s2])
                for j in range(s2, n):
                    grad[j] -= h[j - s2]
    gn = np.linalg.norm(grad)
    if gn > 1e-15:
        h = h - step_size * grad / gn
    return project_feasible(h, n)

def fourier_init(n, freq, amp, phase=0.0):
    x = np.arange(n)
    h = 0.5 + amp * np.sin(2.0 * np.pi * freq * x / n + phase)
    return project_feasible(h, n)

def sturmian_init(n):
    phi = (np.sqrt(5.0) - 1.0) / 2.0
    x = np.arange(n)
    h = ((x * phi) % 1.0).copy()
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(500, 0.5)
        n = 500
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    # Phase 1: CD polish
    if time.time() < deadline - 380:
        h1, c1 = coord_descent(best_h, n, 35, deadline - 380, gs_iters=55)
        if c1 < best_c:
            best_h, best_c = h1, c1
    # Phase 2: SLSQP LSE with increasing alpha
    if time.time() < deadline - 300:
        h2, c2 = slsqp_lse(best_h, n, [1000, 5000, 20000, 100000, 500000, 2000000], 400)
        if c2 < best_c:
            best_h, best_c = h2, c2
        h2, c2 = coord_descent(best_h, n, 30, deadline - 300, gs_iters=55)
        if c2 < best_c:
            best_h, best_c = h2, c2
    # Phase 3: Target-level SLSQP
    if time.time() < deadline - 220:
        for attempt in range(3):
            target_c5 = best_c - 0.00003 * (attempt + 1)
            h3, c3 = slsqp_target(best_h, n, target_c5, 400)
            if c3 < best_c:
                best_h, best_c = h3, c3
            else:
                break
        h3, c3 = coord_descent(best_h, n, 30, deadline - 220, gs_iters=55)
        if c3 < best_c:
            best_h, best_c = h3, c3
    # Phase 4: Equalization + CD
    if time.time() < deadline - 170:
        for _ in range(50):
            if time.time() > deadline - 190:
                break
            best_h = equalize_step(best_h, n, top_k=4, step_size=0.003)
        h4, c4 = coord_descent(best_h, n, 35, deadline - 170, gs_iters=55)
        if c4 < best_c:
            best_h, best_c = h4, c4
        if time.time() < deadline - 140:
            h4, c4 = slsqp_lse(best_h, n, [100000, 500000], 400)
            if c4 < best_c:
                best_h, best_c = h4, c4
    # Phase 5: Try nearby n values
    x_orig = np.linspace(0, 2, len(best_h))
    for n_try in [n + 20, n - 20, n + 40]:
        if time.time() > deadline - 80:
            break
        if n_try == n or n_try < 200:
            continue
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_o, c_o = coord_descent(h_int, n_try, 40, deadline - 80, gs_iters=55)
        if time.time() < deadline - 50:
            h_o, c_o = slsqp_lse(h_o, n_try, [50000, 200000], 350)
            h_o, c_o = coord_descent(h_o, n_try, 25, deadline - 50, gs_iters=50)
        if c_o < best_c:
            best_h, best_c = h_o, c_o
            x_orig = np.linspace(0, 2, n_try)
    # Phase 6: Multi-restart with diverse strategies
    if time.time() < deadline - 50:
        n_cur = len(best_h)
        remaining = deadline - time.time() - 50
        n_restarts = max(4, int(remaining / 45))
        for r in range(n_restarts):
            if time.time() > deadline - 50:
                break
            if r % 5 == 0:
                hp = project_feasible(best_h + rng.normal(0, 0.02, n_cur), n_cur)
            elif r % 5 == 1:
                hp = fourier_init(n_cur, max(2, n_cur // 14), 0.25, phase=r * 0.7)
                hp = project_feasible(0.7 * best_h + 0.3 * hp, n_cur)
            elif r % 5 == 2:
                hp = sturmian_init(n_cur)
                hp = project_feasible(0.6 * best_h + 0.4 * hp, n_cur)
            elif r % 5 == 3:
                block = n_cur // 6
                hp = best_h.copy()
                start = (r * block) % max(1, n_cur - block)
                hp[start:start+block] = 1.0 - hp[start:start+block]
                hp = project_feasible(hp, n_cur)
            else:
                x = np.linspace(0, 2, n_cur)
                hp = project_feasible(best_h + 0.03 * np.sin(2*np.pi*(3+r)*x), n_cur)
            h_o, c_o = coord_descent(hp, n_cur, 30, deadline - 50, gs_iters=50)
            if c_o < best_c:
                if time.time() < deadline - 30:
                    h_o, c_o = slsqp_lse(h_o, n_cur, [50000, 200000], 300)
                    h_o, c_o = coord_descent(h_o, n_cur, 25, deadline - 30, gs_iters=50)
            if c_o < best_c:
                best_h, best_c = h_o, c_o
    # Phase 7: Final polish
    if time.time() < deadline - 10:
        n_cur = len(best_h)
        h_f, c_f = slsqp_lse(best_h, n_cur, [200000, 1000000], 500)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
        h_f, c_f = coord_descent(best_h, n_cur, 50, deadline - 10, gs_iters=60)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
