    """Project h onto {x : 0<=x<=1, sum(x)=target}"""
    lo, hi = h.min() - 1, h.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid

def lag_balancing(h, n, max_iter, deadline):
    """Cutting-plane style method for min max_k c_k"""
    h = h.copy()
    dx = 2.0 / n
    active = set()  # active lags
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1-h, mode='full') * dx
        max_c = c.max()
        # Find lags within tolerance of max
        tol = 0.001
        active = set(np.where(c >= max_c - tol)[0].tolist())
        
        # Build linearized constraints
        # For each active lag m, constraint: c[m] + dc[m]/dh · (h_new - h) ≤ L
        # dc[m]/dh[p] = dx * [(1-h[p+k]) - h[p-k]] (with bounds)
        m_active = list(active)
        A_rows = []
        b_vals = []
        for m in m_active:
            k = m - (n-1)
            grad = np.zeros(n)
            if k >= 0:
                grad[:n-k] += (1.0 - h[k:])
                grad[k:] -= h[:n-k]
            else:
                d = -k
                grad[d:] += (1.0 - h[:n-d])
                grad[:n-d] -= h[d:]
            grad *= dx
            A_rows.append(grad)
            b_vals.append(c[m] - grad @ h)  # c[m] + grad·(h_new - h) ≤ L → grad·h_new ≤ L - c[m] + grad·h
        # sum(h_new) = n/2
        # 0 ≤ h_new ≤ 1
        # min L = max of (c[m] - grad[m]·h + grad[m]·h_new) for m in active
        # = min max_m (b[m] + A[m]·h_new)
        
        # This is: min L s.t. A[m]·h_new - L ≤ b[m] for all m
        #                      sum(h_new) = n/2
        #                      0 ≤ h_new ≤ 1
        
        # Use cvxpy to solve
        import cvxpy as cp
        h_var = cp.Variable(n)
        L_var = cp.Variable()
        constraints = [cp.sum(h_var) == n/2.0, h_var >= 0, h_var <= 1]
        for i, m in enumerate(m_active):
            constraints.append(A_rows[i] @ h_var - L_var <= b_vals[i])
        obj = cp.Minimize(L_var)
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.GLOP)  # or CLP, CBC
        except:
            try:
                prob.solve(solver=cp.CBC)
            except:
                break
        
        if prob.status not in ['optimal', 'optimal_inaccurate']:
            break
        
        h_lp = h_var.value
        # Damped step
        alpha = 0.5  # step size
        h_new = h + alpha * (h_lp - h)
        h_new = project_feasible(h_new, n)

    return h, compute_c5(h, n)

    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def exact_cd(h, n, max_passes, deadline):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            xn = cd_1d(c, h, p, n)
            if abs(xn - h0) > 1e-12:
                delta = xn - h0
                m_min = n - 1 - p
                m_max = 2 * n - 2 - p
                ms = np.arange(m_min, m_max + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                a = np.zeros(len(ms))
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                diag_i = np.where(ms == (n - 1))[0]
                c[ms] += delta * a
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True
        h = project_feasible(h, n)
        if not improved:
            break
    return h, compute_c5(h, n)

def cutting_plane(h, n, max_iters, deadline):
    dx = 2.0 / n
    h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1 - h, mode='full') * dx
        max_c = c.max()
        tol = max(0.003, 0.01 * max_c)
        active = np.where(c >= max_c - tol)[0]
        if len(active) == 0:
            break
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_ub_list = []
        b_ub_list = []
        for m in active:
            k = m - (n - 1)
            grad = np.zeros(n)
            if k >= 0:
                grad[:n - k] += (1.0 - h[k:])
                grad[k:] -= h[:n - k]
            else:
                d = -k
                grad[d:] += (1.0 - h[:n - d])
                grad[:n - d] -= h[d:]
            grad *= dx
            A_ub_list.append(np.append(grad, -1.0))
            b_ub_list.append(c[m] - grad @ h)
        if not A_ub_list:
            break
        A_ub = np.array(A_ub_list)
        b_ub = np.array(b_ub_list)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        bounds = [(0, 1)] * n + [(0, 1)]
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
        except Exception:
            break
        if res.status != 0:
            break
        h_lp = res.x[:n]
        improved = False
        for alpha in [1.0, 0.7, 0.4, 0.2, 0.1]:
            h_trial = project_feasible(h + alpha * (h_lp - h), n)
            c_trial = compute_c5(h_trial, n)
            if c_trial < best_c - 1e-10:
                h = h_trial
                best_c = c_trial
                improved = True
                break
        if not improved:
            break
    return h, best_c

def binary_project(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1][:target]
    b = np.zeros(n)
    b[idx] = 1.0
    return b

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 10

    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = np.full(400, 0.5)
        n = 400

    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)

    # Phase 1: Cutting-plane LP (new approach)
    if time.time() < deadline - 200:
        h1, c1 = cutting_plane(h_best, n, 40, deadline - 180)
        if c1 < best_c:
            h_best, best_c = h1, c1

    # Phase 2: Exact 1D coordinate descent
    if time.time() < deadline - 120:
        h2, c2 = exact_cd(h_best, n, 300, deadline - 100)
        if c2 < best_c:
            h_best, best_c = h2, c2

    # Phase 3: Binary projection + re-optimize
    if time.time() < deadline - 80:
        b = binary_project(h_best, n)
        h3, c3 = cutting_plane(b, n, 25, deadline - 60)
        h3, c3b = exact_cd(h3, n, 150, deadline - 50)
        if c3b < best_c:
            h_best, best_c = h3, c3b

    # Phase 4: Multi-n exploration
    n_list = [350, 500, 600, 700, 800, 900, 1000]
    for n_try in n_list:
        if time.time() > deadline - 60:
            break
        if n_try == len(h_best):
            continue
        x_old = np.linspace(0, 2, len(h_best))
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_old, h_best), n_try)
        h_o, c_o = cutting_plane(h_int, n_try, 20, deadline - 40)
        h_o, c_o2 = exact_cd(h_o, n_try, 100, deadline - 30)
        if c_o2 < best_c - 1e-8:
            h_best, best_c = h_o, c_o2

    # Phase 5: Perturbation restarts
    for r in range(8):
        if time.time() > deadline - 40:
            break
        s = 0.003 + 0.004 * r
        hp = project_feasible(h_best + rng.normal(0, s, len(h_best)), len(h_best))
        h_o, c_o = cutting_plane(hp, len(h_best), 20, deadline - 20)
        h_o, c_o2 = exact_cd(h_o, len(h_best), 100, deadline - 15)
        if c_o2 < best_c:
            h_best, best_c = h_o, c_o2

    # Phase 6: Final CD polish
    n = len(h_best)
    if time.time() < deadline - 3:
        h_f, c_f = exact_cd(h_best, n, 300, deadline)
        if c_f < best_c:
            h_best, best_c = h_f, c_f

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
