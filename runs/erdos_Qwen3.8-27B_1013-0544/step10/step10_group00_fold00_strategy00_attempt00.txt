```python
        if n_curr is None:  # Automatic higher resolution
            n_curr = n_parent * 2**(i+1)
            
        if time.time() > deadline: break
        
        # Global restart initialization
        h_g = np.random.rand(n_parent) * 2 - 1  # Stochastic base
        h_g = project_feasible(h_g[:n_curr], n_curr)
        
        if i == 0:  # Coarse resolution CD
            perform_coord_descent(h_g, n_curr, deadline, max_iter=300)
        elif i == 1:  # Medium resolution SGD
            perform_sgd(h_g, n_curr, deadline, max_iters=1000, restarts=5)
        
        # Accept updates if better
        if compute_c5(h_g, n_curr) < compute_c5(h, n_parent):
            h = h_g.copy()
            n_parent = n_curr
    
    # Phase 2: Final stochastic refinement
    perform_global_sgd(h, n_parent, deadline, include_parent=True)
    h = project_feasible(h, n_parent)
    
    # Phase 3: Adaptable final polish
    if time.time() < deadline - 10:
        tune_slsqp(h, n_parent, deadline, alpha_range=[10, 100, 1000])
        h = perform_coord_descent(h, n_parent, deadline - 5, max_iter=50)
    
    return h, compute_c5(h, n_parent), len(h)
    
def project_feasible(h, n, tol=1e-14):
    # Clip to [0,1] and rebalance
    h_clipped = np.clip(h, 0, 1)
    total = np.sum(h_clipped)
    diff = 1.0 - (n / 2) / ((total if total!=0 else 1) * dx)  # Fixed dx=2/n
    
    # Simplex projection
    h_proj = (1 - diff) * h_clipped
    return h_proj
```

```python
def compute_stochastic_gradient(c_val):
    # Sample-augmented gradient estimation
    grad = np.random.uniform(0, 1, len(c_val))
    grad = (c_val - np.median(c_val) + 0.1) / np.std(c_val)
    return grad / (1 + np.exp(-5*grad))
```
