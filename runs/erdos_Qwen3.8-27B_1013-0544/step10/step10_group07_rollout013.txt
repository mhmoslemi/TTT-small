def maxgrad_active(h,n,alpha=20.0,K=16):
   c=np.correlate(h,1-h,'full')
   top=np.argsort(c)[-K:] if c.size>K else np.arange(c.size)
   vals=c[top]
   z=alpha*(vals-vals.max())
   w=np.exp(z); w/=w.sum()

import numpy as np
import time

GOLD = (np.sqrt(5.0) - 1.0) / 2.0


def compute_c5(h, n):
    dx = 2.0 / float(n)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)


def project_feasible(y, target=None):
    y = np.asarray(y, dtype=float).copy()
    n = len(y)
    if target is None:
        target = 0.5 * float(n)
    if target <= 0.0:
        return np.zeros(n, dtype=float)
    if target >= float(n):
        return np.ones(n, dtype=float)

    if np.all(y >= -1e-14) and np.all(y <= 1.0 + 1e-14):
        s = float(y.sum())
        if abs(s - target) <= 1e-11:
            np.clip(y, 0.0, 1.0, out=y)
            return y

    lo = min(float(y.min()), 0.0) - 1.0
    hi = max(float(y.max()), 1.0) + 1.0
    for _ in range(55):
        tau = 0.5 * (lo + hi)
        s = float(np.clip(y - tau, 0.0, 1.0).sum())
        if s > target:
            lo = tau
        else:
            hi = tau

    tau = 0.5 * (lo + hi)
    out = np.clip(y - tau, 0.0, 1.0)
    diff = target - float(out.sum())
    if abs(diff) > 1e-13:
        room = np.where((out > 1e-10) & (out < 1.0 - 1e-10))[0]
        if room.size:
            out[room] += diff / float(room.size)
            np.clip(out, 0.0, 1.0, out=out)
    return out


def maxgrad_active(h, n, alpha=20.0, K=16):
    c = np.correlate(h, 1.0 - h, mode="full")
    if c.size > K:
        top = np.argsort(c)[-K:]
    else:
        top = np.arange(c.size)
    vals = c[top]
    z = alpha * (vals - vals.max())
    w = np.exp(z)
    w /= w.sum()
    g = np.zeros(n, dtype=float)
    lm1 = n - 1
    for idx, wt in zip(top, w):
        if wt < 1e-12:
            continue
        k = int(idx) - lm1
        if k >= 0:
            if k == 0:
                g += wt * (1.0 - 2.0 * h)
            else:
                g[k:n] += wt * (1.0 - h[:n - k])
                g[:n - k] -= wt * h[k:n]
        else:
            q = -k
            g[:n - q] += wt * (1.0 - h[q:n])
            g[q:n] -= wt * h[:n - q]
    return g


def quad_max(Q0, Q1, Q2, t):
    return float(np.max(Q0 + Q1 * t + Q2 * t * t))


def optimize_pair(h, c, n, p, q, grid_count=20, golden_iters=35):
    if p == q:
        return None
    if p > q:
        p, q = q, p
    d = q - p
    if d <= 0 or d >= n:
        return None

    a = float(h[p])
    b = float(h[q])
    s = a + b
    if s <= 1e-9 or s >= 2.0 - 1e-9:
        return None

    lo = max(0.0, s - 1.0)
    hi = min(1.0, s)
    if hi - lo <= 1e-9:
        return None

    m = 2 * n - 1
    diag = n - 1
    x1 = diag - d
    x2 = diag + d

    hmask = h.copy()
    hmask[p] = 0.0
    hmask[q] = 0.0
    rev = 1.0 - hmask[::-1]

    Lp = np.zeros(m, dtype=float)
    Lq = np.zeros(m, dtype=float)
    Lp[p:p + n] += rev
    Lp[n - 1 - p:n + n - 1] -= hmask
    Lq[q:q + n] += rev
    Lq[n - 1 - q:n + n - 1] -= hmask

    ab = a * b
    Q0 = c + a * (Lq - Lp)
    Q0[diag] -= 2.0 * ab
    Q0[x1] += ab
    Q0[x2] += ab

    Qs = c + b * (Lp - Lq)
    Qs[diag] -= 2.0 * ab
    Qs[x1] += ab
    Qs[x2] += ab

    hm = 0.5 * s
    Qm = c + (hm - a) * Lp + (hm - b) * Lq
    Qm[diag] += a * a + b * b - 0.5 * s * s
    mc = ab - 0.25 * s * s
    Qm[x1] += mc
    Qm[x2] += mc

    s2 = s * s
    Q1 = (4.0 * Qm - Qs - 2.0 * Q0) / s
    Q2 = (2.0 * (Qs - 2.0 * Qm + Q0)) / s2

    grid = np.linspace(lo, hi, grid_count)
    g2 = grid * grid
    G = Q0[None, :] + Q1[None, :] * grid[:, None] + Q2[None, :] * g2[:, None]
    vals = G.max(axis=1)
    i0 = int(np.argmin(vals))

    left = lo if i0 == 0 else float(grid[i0 - 1])
    right = hi if i0 == grid_count - 1 else float(grid[i0 + 1])

    if right - left > 1e-8:
        a1 = left + (1.0 - GOLD) * (right - left)
        a2 = left + GOLD * (right - left)
        f1 = quad_max(Q0, Q1, Q2, a1)
        f2 = quad_max(Q0, Q1, Q2, a2)
        for _ in range(golden_iters):
            if f1 < f2:
                right = a2
                a2 = a1
                f2 = f1
                a1 = left + (1.0 - GOLD) * (right - left)
                f1 = quad_max(Q0, Q1, Q2, a1)
            else:
                left = a1
                a1 = a2
                f1 = f2
                a2 = left + GOLD * (right - left)
                f2 = quad_max(Q0, Q1, Q2, a2)
        tg = 0.5 * (left + right)
    else:
        tg = float(grid[i0])

    if quad_max(Q0, Q1, Q2, tg) > float(vals[i0]) + 1e-12:
        tg = float(grid[i0])

    if tg < lo:
        tg = lo
    elif tg > hi:
        tg = hi

    tq = s - tg
    c_new = Q0 + Q1 * tg + Q2 * tg * tg
    return tg, tq, c_new


def generate_candidates(h, c, n, lags, rng,
                        max_candidates=180, top_lags=6, pairs_per_lag=25):
    cands = []
    seen = set()
    top_idx = np.argsort(c)[::-1][:top_lags]

    for idx in top_idx:
        k = int(lags[idx])
        if k == 0:
            continue
        d = abs(k)
        if d <= 0 or d >= n:
            continue
        if k > 0:
            scores = h[d:] * (1.0 - h[:-d])
        else:
            scores = h[:-d] * (1.0 - h[d:])
        sz = scores.size
        if sz == 0:
            continue
        maxs = float(scores.max())
        if maxs <= 1e-8:
            continue

        cnt = min(pairs_per_lag, sz)
        if sz > cnt:
            part = np.argpartition(scores, -cnt)[-cnt:]
        else:
            part = np.arange(sz)

        extra = min(5, max(0, sz - cnt))
        if extra:
            high = np.nonzero(scores > 0.25 * maxs)[0]
            if high.size > extra:
                chosen = np.concatenate((part, high[rng.randint(0, high.size, extra)]))
            else:
                chosen = part
        else:
            chosen = part

        for r in chosen:
            rr = int(r)
            if rr < 0 or rr + d >= n:
                continue
            sv = float(h[rr] + h[rr + d])
            if sv <= 1e-7 or sv >= 2.0 - 1e-7:
                continue
            key = rr * n + (rr + d)
            if key not in seen:
                seen.add(key)
                cands.append((rr, rr + d))

    for sign in (1, 0):
        d = 1
        if sign:
            scores = h[1:] * (1.0 - h[:-1])
        else:
            scores = h[:-1] * (1.0 - h[1:])
        sz = scores.size
        if sz == 0:
            continue
        maxs = float(scores.max())
        if maxs <= 1e-8:
            continue
        cnt = min(15, sz)
        if sz > cnt:
            part = np.argpartition(scores, -cnt)[-cnt:]
        else:
            part = np.arange(sz)
        for r in part:
            rr = int(r)
            q = rr + d
            if rr < 0 or q >= n:
                continue
            sv = float(h[rr] + h[q])
            if sv <= 1e-7 or sv >= 2.0 - 1e-7:
                continue
            key = rr * n + q
            if key not in seen:
                seen.add(key)
                cands.append((rr, q))

    offs = []
    for idx in top_idx:
        d = abs(int(lags[idx]))
        if 0 < d < n and d not in offs:
            offs.append(d)
    if 1 not in offs:
        offs.append(1)
    if 2 not in offs and 2 < n:
        offs.append(2)

    for _ in range(15):
        if not offs:
            break
        d = offs[rng.randint(0, len(offs))]
        rr = rng.randint(0, n - d)
        q = rr + d
        sv = float(h[rr] + h[q])
        if sv <= 1e-7 or sv >= 2.0 - 1e-7:
            continue
        key = rr * n + q
        if key not in seen:
            seen.add(key)
            cands.append((rr, q))

    if len(cands) > max_candidates:
        cands = cands[:max_candidates]
    return cands


def targeted_search(h, n, deadline, seed):
    h = project_feasible(h, 0.5 * float(n))
    c = np.correlate(h, 1.0 - h, mode="full")
    lags = np.arange(2 * n - 1) - (n - 1)
    dx = 2.0 / float(n)

    cur_max = float(c.max())
    best_max = cur_max
    best_h = h.copy()

    rng = np.random.RandomState(seed)
    noim = 0
    shake = 0
    it = 0

    while time.time() < deadline:
        it += 1
        if cur_max * dx < best_max * dx:
            best_max = cur_max
            best_h = h.copy()

        if noim > 10:
            if shake & 1:
                sig = max(0.004, 0.035 * (0.75 ** shake))
                y = project_feasible(h + rng.normal(0.0, sig, n), 0.5 * float(n))
            else:
                grad = maxgrad_active(h, n)
                step = 0.015 + 0.003 * min(shake, 6)
                y = project_feasible(h - step * grad, 0.5 * float(n))
            c = np.correlate(y, 1.0 - y, mode="full")
            h = y
            cur_max = float(c.max())
            noim = 0
            shake += 1
            continue

        cands = generate_candidates(h, c, n, lags, rng)
        improved = False
        accepts = 0

        for p, q in cands:
            if time.time() > deadline:
                break
            res = optimize_pair(h, c, n, p, q)
            if res is None:
                continue
            tp, tq, cnew = res
            nmax = float(cnew.max())
            if nmax < cur_max - 1e-10:
                h[p] = tp
                h[q] = tq
                c = cnew
                cur_max = nmax
                improved = True
                accepts += 1
                if cur_max * dx < best_max * dx:
                    best_max = cur_max
                    best_h = h.copy()
                if accepts % 30 == 0:
                    c = np.correlate(h, 1.0 - h, mode="full")
                    cur_max = float(c.max())
                    if cur_max * dx < best_max * dx:
                        best_max = cur_max
                        best_h = h.copy()

        if improved:
            noim = 0
            c = np.correlate(h, 1.0 - h, mode="full")
            cur_max = float(c.max())
            if cur_max * dx < best_max * dx:
                best_max = cur_max
                best_h = h.copy()
        else:
            noim += 1

    h = project_feasible(h, 0.5 * float(n))
    c = np.correlate(h, 1.0 - h, mode="full")
    cf = float(c.max())
    if cf * dx < best_max * dx:
        return h, cf * dx
    return best_h, best_max * dx


def default_h(n):

    n_best = n0
    best_c = compute_c5(best_h, n_best)

    phase1_end = min(deadline - 20.0, t0 + max(90.0, 0.45 * float(budget_s)))
    if phase1_end > time.time():
        h1, c1 = targeted_search(h0, n_best, phase1_end, seed + 1)
        if c1 < best_c - 1e-12:
            best_h = h1
            best_c = c1
            n_best = len(h1)

    n_list = [200, 250, 300, 350, 400, 500, 600, 800, 1000]
    for idx, n_try in enumerate(n_list):
        now = time.time()
        rem = deadline - now
        if rem < 50.0:
            break
        if n_try == n_best:
            continue

        left_count = len(n_list) - idx
        alloc = min(80.0, max(30.0, (rem - 50.0) / float(max(1, left_count))))
        if alloc < 20.0:
            continue
        phase_dead = now + alloc

        x_new = np.linspace(0.0, 2.0, n_try)
        if n_best % n_try == 0 and len(best_h) == n_best:
            br = n_best // n_try
            h_int = best_h.reshape(n_try, br).mean(axis=1)
        else:
            x_old = np.linspace(0.0, 2.0, n_best)
            h_int = np.interp(x_new, x_old, best_h)

        if idx < 3:
            cyc = 3.0 + float(idx)
            ph = 0.1 * float(idx)
            pat = 0.5 + 0.40 * np.cos(cyc * np.pi * x_new + ph) + 0.15 * np.sin((cyc + 2.0) * np.pi * x_new)
            h_int = 0.75 * h_int + 0.25 * pat

        h_int = project_feasible(h_int, 0.5 * float(n_try))
        if phase_dead > time.time():
            h_n, c_n = targeted_search(h_int, n_try, phase_dead, seed + 1000 + idx)
            if c_n < best_c - 1e-12:
                best_h = h_n
                best_c = c_n
                n_best = len(h_n)

    now = time.time()
    if deadline - now > 40.0:
        h_f, c_f = targeted_search(best_h, n_best, deadline - 15.0, seed + 98765)
        if c_f < best_c - 1e-12:
            best_h = h_f
            best_c = c_f
            n_best = len(h_f)

    best_h = project_feasible(best_h, 0.5 * float(len(best_h)))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
