    target=n/2.0

       diff=target-h.sum()
       if abs(diff)<1e-13: return h
       if diff>0:

          if tot<=1e-15: return h
          h=np.clip(h + (diff/tot)*room,0.0,1.0)

import time
import numpy as np

def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            return h
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot <= 1e-15:
                return h
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot <= 1e-15:
                return h
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h)[::-1] if diff > 0 else np.argsort(h)[::-1]
        for idx in order:
            i = int(idx)
            h[i] = min(1.0, max(0.0, h[i] + diff))
            diff = target - h.sum()
            if abs(diff) < 1e-13:
                break
    return h

def build_V(h, n):
    off = n - 1
    L = 2 * n - 1
    V = np.zeros((n, L), dtype=float)
    V[:, off] = 1.0 - 2.0 * h
    for t in range(1, n):
        m = n - t
        V[t:, off + t] -= h[:m]
        V[:m, off + t] += 1.0 - h[t:]
        V[t:, off - t] += 1.0 - h[:m]
        V[:m, off - t] -= h[t:]
    return V

import time
import numpy as np


def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            return h
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot <= 1e-15:
                return h
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot <= 1e-15:
                return h
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)[::-1]
        else:
            order = np.argsort(h)[::-1]
        for idx in order:
            i = int(idx)
            h[i] = min(1.0, max(0.0, h[i] + diff))
            diff = target - h.sum()
            if abs(diff) < 1e-13:
                break
    return h


def raw_c5(h, n):
    return np.correlate(h, 1.0 - h, mode="full")


def objective_c5(h, n):
    return float(np.max(raw_c5(h, n)) * (2.0 / float(n)))


def build_V(h, n):
    off = n - 1
    L = 2 * n - 1
    V = np.zeros((n, L), dtype=float)
    V[:, off] = 1.0 - 2.0 * h
    for t in range(1, n):
        m = n - t
        V[t:, off + t] -= h[:m]
        V[:m, off + t] += 1.0 - h[t:]
        V[t:, off - t] += 1.0 - h[:m]
        V[:m, off - t] -= h[t:]
    return V


def update_V(V, p, delta, q, off):
    if delta == 0.0:
        return
    V[q, off + p - q] -= delta
    V[q, off + q - p] -= delta


def softmax_weights(c, k_focus=96):
    L = len(c)
    if L == 1:
        return np.array([1.0], dtype=float)
    order = np.argsort(c)[::-1]
    kk = min(k_focus, L - 1)
    cmax = float(c[order[0]])
    cfocus = float(c[order[kk]])
    gap = cmax - cfocus
    if gap < 1e-8:
        gap = max(cmax - float(c[order[-1]]), 1e-6)
    temp = max(0.01, gap / 6.0)
    z = (c - cmax) / temp
    w = np.exp(z)
    s = w.sum()
    if not np.isfinite(s) or s <= 0.0:
        w = np.zeros(L, dtype=float)
        w[order[0]] = 1.0
    else:
        w /= s
    return w


def move_value(work, tmp, c, D, t, idx0, q0, idx1, q1, idx2, q2):
    np.copyto(work, c)
    if t != 0.0:
        np.multiply(D, t, out=tmp)
        np.add(work, tmp, out=work)
    tt = t * t
    if q0 != 0.0:
        work[idx0] += q0 * tt
    if q1 != 0.0:
        work[idx1] += q1 * tt
    if q2 != 0.0:
        work[idx2] += q2 * tt
    return float(work.max())


def screen_pair(c, D, off, a, b, tmax, f0, work, tmp):
    if tmax <= 1e-12:
        return None
    idx0 = off
    q0 = -2.0
    d = b - a
    if d != 0:

        q1 = 1.0
        idx2 = off - d
        q2 = 1.0
    else:
        idx1 = 0
        q1 = 0.0
        idx2 = 0
        q2 = 0.0
    best_f = f0
    best_t = 0.0
    best_m = 0
    count = 16
    for m in range(1, count + 1):
        t = tmax * (float(m) / float(count))
        f = move_value(work, tmp, c, D, t, idx0, q0, idx1, q1, idx2, q2)
        if f < best_f:
            best_f = f
            best_t = t
            best_m = m
    if best_m == 0:
        return None
    return best_t, best_f, best_m

        q1 = 0.0

            if f < best_f:

            if f < best_f:
                best_f = f
                best_t = key
    width = max(tmax / float(count), 1e-10)
    for _ in range(4):
        left = max(0.0, best_t - width)
        right = min(tmax, best_t + width)
        if right - left <= 1e-12:
            break
        step = (right - left) / 32.0
        for m in range(33):
            t = left + step * float(m)
            f = move_value(work, tmp, c, D, t, idx0, q0, idx1, q1, idx2, q2)
            if f < best_f:
                best_f = f
                best_t = t
        width *= 0.5
    if best_f < f0 - 1e-9:
        return best_t, best_f
    return None


def pair_cd_optimize(h, n, deadline, max_substeps=50000, rng=None):
    h = project_feasible(h, n).copy()
    c = raw_c5(h, n)
    V = build_V(h, n)
    off = n - 1
    L = 2 * n - 1
    q = np.arange(n)
    cmax = float(c.max())
    if rng is None:
        rng = np.random.RandomState(123)
    if n <= 300:
        K = 28
    elif n <= 600:
        K = 24
    elif n <= 900:
        K = 18
    else:
        K = 14
    work = np.empty(L, dtype=float)
    tmp = np.empty(L, dtype=float)
    D = np.empty(L, dtype=float)
    for _ in range(max_substeps):
        if time.time() >= deadline:
            break
        w = softmax_weights(c, 96)
        G = V @ w
        if not np.all(np.isfinite(G)):
            break
        md = h > 1e-10
        mr = h < 1.0 - 1e-10
        K = int(min(K, md.sum(), mr.sum()))
        if K < 2:
            break
        desc = np.argsort(G)[::-1]
        asc = np.argsort(G)
        donors = []
        for idx in desc:
            if md[idx]:
                donors.append(int(idx))
                if len(donors) >= K:
                    break
        recips = []
        for idx in asc:
            if mr[idx]:
                recips.append(int(idx))
                if len(recips) >= K:
                    break
        top_a = -1
        top_b = -1
        top_t = 0.0
        top_f = 1e100
        for a in donors:
            ha = h[a]
            for b in recips:
                if a == b:
                    continue
                tmax = min(ha, 1.0 - h[b])
                if tmax <= 1e-10:
                    continue
                np.subtract(V[b], V[a], out=D)
                res = screen_pair(c, D, off, a, b, tmax, cmax, work, tmp)
                if res is not None and res[1] < top_f:
                    top_t, top_f, _ = res
                    top_a = a
                    top_b = b
        for _rr in range(K):
            if time.time() >= deadline:
                break
            a = int(rng.randint(n))
            if not md[a]:
                continue
            b = int(rng.randint(n))
            if not mr[b] or a == b:
                continue
            tmax = min(h[a], 1.0 - h[b])
            if tmax <= 1e-10:
                continue
            np.subtract(V[b], V[a], out=D)
            res = screen_pair(c, D, off, a, b, tmax, cmax, work, tmp)
            if res is not None and res[1] < top_f:
                top_t, top_f, _ = res
                top_a = a
                top_b = b
        if top_a < 0:
            break
        a = top_a
        b = top_b
        tmax = min(h[a], 1.0 - h[b])
        np.subtract(V[b], V[a], out=D)
        ref = refine_pair(c, D, off, a, b, tmax, cmax, top_t, top_f, work, tmp)
        if ref is None:
            break
        t, f = ref
        if not (f < cmax - 1e-7):
            break
        t = min(t, h[a], 1.0 - h[b])
        if t <= 1e-12:
            break
        old_cmax = cmax
        prev_h = h.copy()
        h[a] -= t
        h[b] += t
        if h[a] < 0.0 or h[b] > 1.0:
            h = project_feasible(h, n)
            c = raw_c5(h, n)
            V = build_V(h, n)
            new_cmax = float(c.max())
        else:
            c = c + t * D
            tt = t * t
            c[off] -= 2.0 * tt
            if a != b:
                c[off + b - a] += tt
                c[off + a - b] += tt
            update_V(V, a, -t, q, off)
            update_V(V, b, t, q, off)
            new_cmax = float(c.max())
        if new_cmax >= old_cmax - 1e-9:
            h = prev_h
            c = raw_c5(h, n)
            V = build_V(h, n)
            cmax = float(c.max())
            break
        cmax = new_cmax
    h = project_feasible(h, n)
    return h, objective_c5(h, n)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget = float(budget_s)
    deadline = t0 + budget - 5.0
    if deadline <= t0 + 1.0:
        deadline = t0 + max(10.0, budget)
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = None
    if h_init is None or len(h_init) == 0:
        n = 500
        x = np.linspace(0.0, 2.0, n, endpoint=False)
        h_init = 0.5 + 0.5 * np.cos(np.pi * x)
    else:
        n = len(h_init)
    best_h = project_feasible(h_init, n)
    best_n = n
    best_c = objective_c5(best_h, best_n)
    reserve = max(20.0, 0.20 * budget)
    main_end = min(deadline, max(time.time() + 2.0, deadline - reserve))
    h_o, c_o = pair_cd_optimize(best_h, best_n, main_end, 50000, rng)
    if c_o < best_c:
        best_h, best_c = h_o, c_o
    coarse_end = min(deadline, max(time.time() + 1.0, deadline - max(20.0, 0.15 * budget)))
    for n_try in [500, 350, 250]:
        if n_try == best_n or time.time() >= coarse_end:
            continue
        x0 = np.linspace(0.0, 2.0, best_n, endpoint=False)
        x1 = np.linspace(0.0, 2.0, n_try, endpoint=False)
        h_i = project_feasible(np.interp(x1, x0, best_h), n_try)
        sub_end = min(coarse_end, time.time() + 40.0)
        if sub_end <= time.time():
            break
        h_o, c_o = pair_cd_optimize(h_i, n_try, sub_end, 20000, rng)
        if c_o < best_c:
            best_h, best_c, best_n = h_o, c_o, n_try
    var_end = min(deadline, max(time.time() + 1.0, deadline - max(10.0, 0.05 * budget)))
    count = 0
    while time.time() < var_end:
        count += 1
        rb = best_n
        x = np.linspace(0.0, 2.0, rb, endpoint=False)
        kind = (count - 1) % 7
        if kind == 0:
            hp = best_h + rng.normal(0.0, 0.04, rb)
        elif kind == 1:
            hp = best_h + rng.normal(0.0, 0.08, rb)
        elif kind == 2:
            hp = best_h + 0.08 * np.sin(2.0 * np.pi * x + rng.uniform(0.0, 2.0 * np.pi))
        elif kind == 3:
            hp = best_h + 0.08 * np.sin(3.0 * np.pi * x + rng.uniform(0.0, 2.0 * np.pi))
        elif kind == 4:
            hp = np.roll(best_h, max(1, rb // 7))
        elif kind == 5:
            hp = best_h.copy()
            block = max(1, rb // 10)
            st = int(rng.randint(0, max(1, rb - block)))
            hp[st:st + block] = 1.0 - hp[st:st + block]
        else:
            hp = 1.0 - best_h
        hp = project_feasible(hp, rb)
        sub_end = min(var_end, time.time() + 30.0)
        if sub_end <= time.time():
            break
        h_o, c_o = pair_cd_optimize(hp, rb, sub_end, 10000, rng)
        if c_o < best_c:
            best_h, best_c = h_o, c_o
    if time.time() < deadline:
        h_o, c_o = pair_cd_optimize(best_h, best_n, deadline, 100000, rng)
        if c_o < best_c:
            best_h, best_c = h_o, c_o
    best_h = project_feasible(best_h, best_n)
    best_c = objective_c5(best_h, best_n)
    if not np.isfinite(best_c):
        best_h = np.full(best_n, 0.5, dtype=float)
        best_c = 0.5
    return best_h, float(best_c), best_n
