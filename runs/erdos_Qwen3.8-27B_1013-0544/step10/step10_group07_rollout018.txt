    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)

            break

            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_opt(h, n, alphas, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': lambda z: z.sum() - n/2.0,
                                         'jac': lambda z: np.ones(n)}],
                           options={'maxiter': maxiter, 'ftol': 1e-20})
            new_h = project_feasible(res.x, n)

            if new_c < best_c - 1e-12:

        except Exception:
            pass

**Idea: Use the structure of the problem more effectively.**

The correlation c_m = sum_i h[i] * (1-h[i+m]) can be written as:

The range of i depends on m. For m ≥ 0: i ranges from 0 to n-1-m.
For m < 0: i ranges from -m to n-1.

**Another idea: Use the fact that the optimal solution likely has many values at 0 or 1 (binary-like).**

If h is binary (0 or 1), then h*(1-h) = 0, so the diagonal c_0 = 0. The problem becomes:
minimize max_{m≠0} sum_i h[i] * (1-h[i+m]) * dx

With the constraint sum(h) = n/2.

This is a combinatorial problem. The optimal binary solution would be a subset of {0,...,n-1} of size n/2 that minimizes the maximum "discrepancy" at each shift.

This is related to the "maximum cross-correlation" problem or "balanced incomplete block designs."

For a binary h, c_m = |A ∩ (A+m)| complement... actually:
c_m = sum_i h[i]*(1-h[i+m]) = |{i : h[i]=1, h[i+m]=0}|

This is the number of 1s in A that, when shifted by m, land outside A.

If A has n/2 elements, then:
|{i : h[i]=1, h[i+m]=0}| + |{i : h[i]=1, h[i+m]=1}| = |A ∩ [0, n-1-m]| (for m > 0)
and |{i : h[i]=0, h[i+m]=1}| + |{i : h[i]=1, h[i+m]=1}| = |(A+m) ∩ [0, n-1]| (for m > 0)

So c_m = |A ∩ [0, n-1-m]| - |A ∩ (A-m) ∩ [0, n-1]| (roughly)
