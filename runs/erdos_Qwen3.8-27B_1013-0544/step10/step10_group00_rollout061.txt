```python
def compute_gradient(x, n, w):
    # w has length 2n-1, w[k] is the weight for shift s = k-(n-1)
    dx = 2.0 / n
    # Build the gradient using matrix-vector product
    # grad[p] = dx * sum_k w[k] * ((1-x[p-s_k]) - x[p+s_k])
    #         = dx * (sum_k w[k]) - dx * sum_k w[k]*x[p-s_k] - dx * sum_k w[k]*x[p+s_k]
    
    # For the two sums, use advanced indexing
    p_idx = np.arange(n)
    s_vals = np.arange(-(n-1), n)  # shape (2n-1,)
    
    # Forward: F[p] = sum_s w[s+n-1] * x[p+s] for valid p+s
    fp = p_idx[:, None] + s_vals[None, :]  # shape (n, 2n-1)
    valid_f = (fp >= 0) & (fp < n)
    x_f = np.where(valid_f, x[np.clip(fp, 0, n-1)], 0.0)
    F = x_f @ w  # shape (n,)
    
    # Backward: B[p] = sum_s w[s+n-1] * x[p-s] for valid p-s
    bp = p_idx[:, None] - s_vals[None, :]
    valid_b = (bp >= 0) & (bp < n)
    x_b = np.where(valid_b, x[np.clip(bp, 0, n-1)], 0.0)
    B = x_b @ w
    
    grad = dx * (1.0 - F - B)
    return grad
```

24. **Another idea: use the structure of the correlation matrix**

The correlation c_k = sum_i h[i]*(1-h[i+k]) can be written as:
c_k = sum_i h[i] - (h ⋆ h)[k]  (for the "full" part)

where (h ⋆ h)[k] is the autocorrelation at lag k.

Ignoring boundary effects, we want to maximize the minimum autocorrelation. This is related to the problem of designing sequences with minimum autocorrelation (or "good autocorrelation properties" in signal processing).

Known results:
- For binary sequences, the minimum autocorrelation is related to Barker sequences
- For real-valued sequences, the optimal design is related to "flat spectrum" (all autocorrelation values equal)

If we could make all autocorrelation values equal, we'd have a "flat" autocorrelation, which would minimize the maximum overlap.

The autocorrelation spectrum is |H(ω)|² where H is the DFT of h. A flat autocorrelation means |H(ω)| is constant, i.e., h is a "CAZAC" (Constant Amplitude Zero Autocorrelation) sequence... but we want constant non-zero autocorrelation, not zero.

For real-valued h with 0 ≤ h[i] ≤ 1, the "flattest" autocorrelation would be the one where all lags have the same value. The sum of all autocorrelation values is sum_i h[i]² (by Parseval's theorem applied to the autocorrelation). So the "flat" value would be (sum h[i]²) / (2n-1).

To minimize the maximum overlap, we want the minimum autocorrelation to be as large as possible, which means making the autocorrelation as flat as possible.

```python
    valid_b = (bp >= 0) & (bp < n)
    valid_f = (fp >= 0) & (fp < n)
    x_b = np.where(valid_b, x[np.clip(bp, 0, n-1)], 0.0)
    x_f = np.where(valid_f, x[np.clip(fp, 0, n-1)], 0.0)
    B = x_b @ w
    F = x_f @ w
    grad = dx * (1.0 - F - B)
    return obj, grad

def lbfgsb_optimize(h, n, alphas, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(smooth_obj_and_grad, best_h, args=(n, alpha),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 800, 'ftol': 1e-20, 'gtol': 1e-12})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def _cd_fval(c_aff, a, diag_mask, unaff_max, d):
    vals = c_aff + d * a
    if np.any(diag_mask):
        vals = vals.copy()
        vals[diag_mask] -= d * d
    result = float(np.max(vals))
    if unaff_max > result:
        result = unaff_max
    return result

def coord_descent(h, n, max_passes, deadline, gs_iters=40):
    h = h.copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-13 or h0 >= 1.0 - 1e-13:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            s_k = ms - (n - 1)
            fp_idx = p - s_k
            bp_idx = p + s_k
            valid_f = (fp_idx >= 0) & (fp_idx < n)
            valid_b = (bp_idx >= 0) & (bp_idx < n)
            a = np.zeros(len(ms))
            if np.any(valid_f):
                a[valid_f] += (1.0 - h[fp_idx[valid_f]])
            if np.any(valid_b):
                a[valid_b] -= h[bp_idx[valid_b]]
            diag_mask = (s_k == 0)
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            d_lo, d_hi = -h0, 1.0 - h0
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            f1 = _cd_fval(c_aff, a, diag_mask, unaff_max, d1)
            f2 = _cd_fval(c_aff, a, diag_mask, unaff_max, d2)
            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-14:
                    break
                if f1 < f2:
                    d_hi, d2, f2 = d2, d1, f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    f1 = _cd_fval(c_aff, a, diag_mask, unaff_max, d1)
                else:
                    d_lo, d1, f1 = d1, d2, f2
                    d2 = d_lo + gr * (d_hi - d_lo)
                    f2 = _cd_fval(c_aff, a, diag_mask, unaff_max, d2)
            d_opt = (d_lo + d_hi) / 2.0
            f_opt = _cd_fval(c_aff, a, diag_mask, unaff_max, d_opt)
            f_curr = _cd_fval(c_aff, a, diag_mask, unaff_max, 0.0)
            if f_opt < f_curr - 1e-14:
                h[p] = h0 + d_opt
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def block_flip_perturb(h, n, rng, n_blocks=3):
    hp = h.copy()
    for _ in range(n_blocks):
        block_len = max(5, n // 20)
        start = rng.randint(0, n - block_len)
        hp[start:start+block_len] = 1.0 - hp[start:start+block_len]
    return project_feasible(hp, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(700, 0.5)
        n = 700
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    # Phase 1: L-BFGS-B with increasing alpha
    if time.time() < deadline - 200:
        h1, c1 = lbfgsb_optimize(best_h, n, [500, 2000, 10000, 50000], deadline - 180)
        if c1 < best_c:
            best_h, best_c = h1, c1
        h1, c1 = coord_descent(best_h, n, 60, deadline - 180, gs_iters=40)
        if c1 < best_c:
            best_h, best_c = h1, c1
    # Phase 2: Multi-restart with diverse perturbations
    x_cur = np.linspace(0, 2, len(best_h))
    for r in range(30):
        if time.time() > deadline - 120:
            break
        n_cur = len(best_h)
        scale = 0.005 + 0.03 * (r / 30.0)
        mode = r % 5
        if mode == 0:
            hp = project_feasible(best_h + rng.normal(0, scale, n_cur), n_cur)
        elif mode == 1:
            hp = block_flip_perturb(best_h, n_cur, rng, n_blocks=2)
        elif mode == 2:
            x = np.linspace(0, 2, n_cur)
            freq = 2 + (r % 7)
            hp = project_feasible(best_h + scale * 2 * np.sin(np.pi * freq * x + r), n_cur)
        elif mode == 3:
            bsize = max(10, n_cur // 10)
            hp = best_h.copy()
            i = rng.randint(0, n_cur - 2*bsize)
            j = rng.randint(0, n_cur - 2*bsize)
            hp[i:i+bsize], hp[j:j+bsize] = hp[j:j+bsize].copy(), hp[i:i+bsize].copy()
            hp = project_feasible(hp, n_cur)
        else:
            hp = best_h.copy()
            for _ in range(n_cur // 50):
                i, j = rng.randint(0, n_cur, 2)
                tmp = hp[i]
                hp[i] = hp[j]
                hp[j] = tmp
            hp = project_feasible(hp, n_cur)
        h_o, c_o = lbfgsb_optimize(hp, n_cur, [2000, 10000, 50000], deadline - 120)
        if c_o < best_c or c_o < best_c + 0.0005:
            h_o, c_o = coord_descent(h_o, n_cur, 50, deadline - 120, gs_iters=40)
        if c_o < best_c:
            best_h, best_c = h_o, c_o
            x_cur = np.linspace(0, 2, n_cur)
    # Phase 3: Try different n values
    for n_try in [500, 700, 900, 1000, 1200, 1500]:
        if time.time() > deadline - 80:
            break
        if n_try == len(best_h):
            continue
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_cur, best_h), n_try)
        h_o, c_o = lbfgsb_optimize(h_int, n_try, [2000, 10000, 50000], deadline - 80)
        h_o, c_o = coord_descent(h_o, n_try, 50, deadline - 80, gs_iters=40)
        if c_o < best_c:
            best_h, best_c = h_o, c_o
            x_cur = x_new
    # Phase 4: Final polish
    if time.time() < deadline - 10:
        h_f, c_f = coord_descent(best_h, len(best_h), 80, deadline - 10, gs_iters=50)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
        h_f, c_f = lbfgsb_optimize(best_h, len(best_h), [50000, 200000], deadline - 10)
        h_f, c_f = coord_descent(h_f, len(best_h), 30, deadline - 10, gs_iters=50)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
```
