def solve_1d(L, c_base, a, h0, c_diag, has_diag):
    zero_mask = np.abs(a) <= 1e-15
    neg_mask = a < -1e-15
    pos_mask = a > 1e-15
    x_lo, x_hi = 0.0, 1.0
    if np.any(zero_mask):
        if np.any(c_base[zero_mask] > L + 1e-15):
            return h0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    if x_lo > x_hi:
        return h0
    if has_diag:
        Cq = L - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:

            r2 = (1.0 + np.sqrt(disc)) / 2.0

            else:
                x_lo = max(x_lo, r2)

        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

        else:
            x_old = np.linspace(0, 2, len(h_best))
            x_new = np.linspace(0, 2, n_try)
            h_start = project_feasible(np.interp(x_new, x_old, h_best), n_try)
        t_phase = min(deadline - 250, time.time() + 80)
        hb, cb = binary_sa(h_start, n_try, t_phase, rng, T0=3.0, T_min=0.08, cool=0.9998)
        hc, cc = exact_cd(hb, n_try, 150, min(deadline - 200, time.time() + 30))
        if cc < best_c:
            h_best, best_c, n = hc, cc, n_try

    # Phase 2: More binary SA rounds on best n
    n = len(h_best)
    for r in range(5):
        if time.time() > deadline - 200:
            break
        t_phase = min(deadline - 180, time.time() + 60)
        T0 = 2.5 - 0.3 * r
        hb, cb = binary_sa(h_best, n, t_phase, rng, T0=T0, T_min=0.05, cool=0.9999)
        hc, cc = exact_cd(hb, n, 200, min(deadline - 180, time.time() + 40))
        if cc < best_c:
            h_best, best_c = hc, cc

    # Phase 3: Perturbation + binary SA + CD
    n = len(h_best)
    for r in range(8):
        if time.time() > deadline - 120:
            break
        s = 0.03 + 0.02 * r
        hp = project_feasible(h_best + rng.normal(0, s, n), n)
        t1 = min(deadline - 100, time.time() + 30)
        hb, cb = binary_sa(hp, n, t1, rng, T0=2.0, T_min=0.05, cool=0.9999)
        t2 = min(deadline - 100, time.time() + 20)
        hc, cc = exact_cd(hb, n, 150, t2)
        if cc < best_c:
            h_best, best_c = hc, cc

    # Phase 4: Subgradient + CD cycles
    if time.time() < deadline - 60:
        for _ in range(4):
            if time.time() > deadline - 60:
                break
            hs = subgrad_cd(h_best, n, min(deadline - 50, time.time() + 25), n_subgrad=5, scale=0.003)
            if hs[1] < best_c:
                h_best, best_c = hs

    # Phase 5: Multi-n final sweep
    if time.time() < deadline - 50:
        for n_try in [n - 50, n + 50, n + 100]:
            if n_try < 200 or time.time() > deadline - 40:
                continue
            x_old = np.linspace(0, 2, len(h_best))
            x_new = np.linspace(0, 2, n_try)
            h_int = project_feasible(np.interp(x_new, x_old, h_best), n_try)
            t1 = min(deadline - 30, time.time() + 15)
            hb, cb = binary_sa(h_int, n_try, t1, rng, T0=1.5, T_min=0.05, cool=0.9999)
            t2 = min(deadline - 30, time.time() + 10)
            hc, cc = exact_cd(hb, n_try, 100, t2)
            if cc < best_c - 1e-9:
                h_best, best_c, n = hc, cc, n_try

    # Phase 6: Final intensive CD
    if time.time() < deadline - 5:
        n = len(h_best)
        hf, cf = exact_cd(h_best, n, 300, deadline)
        if cf < best_c:
            h_best, best_c = hf, cf

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
