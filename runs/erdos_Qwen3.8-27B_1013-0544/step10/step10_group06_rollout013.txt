def coord_feasible(T, M, b_pos, a_pos, b_neg, a_neg, cdiag, y):
    xlo = 0.0; xhi = 1.0
    if T < M - 1e-10:
        return False
    if b_pos.size:
        v = float(np.min((T - b_pos) / a_pos))
        if v < xhi: xhi = v
    if b_neg.size:
        v = float(np.max((T - b_neg) / a_neg))
        if v > xlo: xlo = v
    if xlo > xhi + 1e-10: return False
    D = T - cdiag + y - y*y
    if D < -1e-10: return False
    if D < 0.25 - 1e-12:
        disc = 1.0 - 4.0*D
        if disc < 0.0: disc = 0.0
        s = float(np.sqrt(disc))

        if f1 < f2 and xlo > f1 + 1e-10 and xhi < f2 + 1e-10:

def coord_step(c, h, p, n, nbisect, lags, not_diag):
    y = float(h[p])
    if y <= 1e-15 or y >= 1.0 - 1e-15: return False
    diag = n - 1
    a = np.zeros(len(lags), dtype=float)
    jf = p + lags
    m1 = (jf >= 0) & (jf < n)
    if np.any(m1): a[m1] += (1.0 - h[jf[m1]])
    jb = p - lags
    m2 = (jb >= 0) & (jb < n)
    if np.any(m2): a[m2] -= h[jb[m2]]
    a[diag] = 0.0
    eps = 1e-12
    nd = (np.abs(a) > eps) & not_diag
    unmask = (~nd) & not_diag
    M = float(np.max(c[unmask])) if np.any(unmask) else 0.0
    U = float(np.max(c))
    if M >= U - 1e-10: return False
    c_nd = c[nd]
    a_nd = a[nd]
    b = c_nd - a_nd * y
    pos = a_nd > eps
    neg = a_nd < -eps
    b_pos = b[pos]; a_pos = a_nd[pos]
    b_neg = b[neg]; a_neg = a_nd[neg]
    cdiag = float(c[diag])
    lo = max(0.0, M)
    hi = U
    if not coord_feasible(hi, M, b_pos, a_pos, b_neg, a_neg, cdiag, y):
        hi = U + 1e-8
    for _ in range(nbisect):
        mid = 0.5*(lo+hi)
        if coord_feasible(mid, M, b_pos, a_pos, b_neg, a_neg, cdiag, y):
            hi = mid
        else:
            lo = mid

    x = coord_choose_x(T, M, b_pos, a_pos, b_neg, a_neg, cdiag, y)

    if nd.any():

import numpy as np

def project_feasible(z, n):
    z = np.asarray(z, dtype=float).copy()
    target = 0.5 * n
    lo = -float(np.max(z)) - 1.0
    hi = 1.0 - float(np.min(z))
    for _ in range(60):
        t = 0.5 * (lo + hi)
        if np.clip(z + t, 0.0, 1.0).sum() < target:
            lo = t
        else:
            hi = t
    h = np.clip(z + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-13:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] += diff / float(mask.sum())
            h = np.clip(h, 0.0, 1.0)
    return h

def calc_c(h, n):
    return np.correlate(1.0 - h, h, mode='full')

    if d < now + 5.0:

    if d > deadline - 2.0:

def coord_step(c, h, p, n, nbisect, lags, not_diag):

    if y <= 1e-15 or y >= 1.0 - 1e-15:
        return False
    diag = n - 1
    a = np.zeros(len(lags), dtype=float)
    jf = p + lags

    if np.any(m2):

    a[diag] = 0.0

    M = float(np.max(c[unmask])) if np.any(unmask) else 0.0

    if M >= U - 1e-10:
        return False

    a_nd = a[nd]

    a_pos = a_nd[pos]
    b_neg = b[neg]
    a_neg = a_nd[neg]
    cdiag = float(c[diag])
    lo = max(0.0, M)

    if not coord_feasible(hi, M, b_pos, a_pos, b_neg, a_neg, cdiag, y):

        newmax = v
    if newmax >= U - 1e-11:
        return False
    if nd.any():
        c[nd] += a_nd * delta
    c[diag] += x - x * x - y + y * y

def coord_cd(h, n, max_passes, deadline, rng, nbisect=30, sym=0, random_order=True):
    h = np.asarray(h, dtype=float)
    if h.size != n:
        h = interp_to(h, n)
    if sym == 2 and n % 2:
        n -= 1
    if sym == 2:
        h = apply_symmetry(h, n, sym)
    else:

    c = calc_c(h, n)
    best = c5_from_c(c, n)
    best_h = h.copy()
    diag = n - 1
    L = 2 * n - 1
    lags = np.arange(L, dtype=int) - diag
    not_diag = np.ones(L, dtype=bool)
    not_diag[diag] = False
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        if random_order:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

                break

                improved = True

            break

        c2 = calc_c(h2, n)
        cc = c5_from_c(c2, n)
        if cc < best - 1e-14:
            best = cc
            best_h = h2.copy()
        h = h2
        c = c2
    h2 = apply_symmetry(h, n, sym)
    cc = c5_from_c(calc_c(h2, n), n)
    if cc < best:
        best = cc
        best_h = h2.copy()
    return best_h, best

def lse_obj_and_c5(h, alpha):
    n = len(h)
    c = calc_c(h, n)
    m = float(np.max(c))
    dx = 2.0 / n
    z = alpha * (c - m)
    z = np.clip(z, -745.0, 0.0)
    s = float(np.exp(z).sum())
    if s <= 0.0:
        s = 1.0
    obj = (m + float(np.log(s)) / alpha) * dx
    return obj, m * dx

def lse_obj_grad(h, alpha):
    n = len(h)
    c = calc_c(h, n)
    m = float(np.max(c))
    dx = 2.0 / n
    z = alpha * (c - m)
    z = np.clip(z, -745.0, 0.0)
    ez = np.exp(z)
    s = float(ez.sum())
    if s <= 0.0:
        w = np.full(2 * n - 1, 1.0 / (2 * n - 1))
        s = 1.0
    else:
        w = ez / s
        w[w < 1e-15] = 0.0
    obj = (m + float(np.log(s)) / alpha) * dx
    g = np.zeros(n)
    n0 = n - 1
    for idx in range(2 * n - 1):
        wt = float(w[idx])
        if wt <= 1e-15:
            continue
        e = idx - n0
        if e >= 0:
            L = n - e
            if L > 0:
                g[:L] += wt * (1.0 - h[e:n])
                g[e:] -= wt * h[:L]
        else:
            f = -e
            L = n - f
            if L > 0:
                g[f:n] += wt * (1.0 - h[:L])
                g[:L] -= wt * h[f:n]
    return obj, g * dx

def lse_refine(h0, n, deadline, alphas, base_step=0.05):
    h = project_feasible(h0, n)
    c = calc_c(h, n)
    best = c5_from_c(c, n)
    best_h = h.copy()
    delta = base_step
    for alpha in alphas:
        if time.time() > deadline:
            break
        noimp = 0
        for _ in range(500):
            if time.time() > deadline:
                break
            obj, grad = lse_obj_grad(h, alpha)
            g = grad - np.mean(grad)
            gn = float(np.linalg.norm(g))
            if gn < 1e-13:
                break
            dirn = g / gn
            accepted = False
            d = delta
            for __ in range(30):
                if time.time() > deadline:
                    break
                z = project_feasible(h - d * dirn, n)
                zobj, zc = lse_obj_and_c5(z, alpha)
                if zc < best + 1e-12 and zobj <= obj + 1e-11:
                    h = z
                    accepted = True
                    if zc < best:
                        best = zc
                        best_h = z.copy()
                    break
                d *= 0.5
                if d < 1e-6:
                    break
            if accepted:
                noimp = 0
                if delta < 0.2:
                    delta *= 1.25
            else:
                noimp += 1
                delta *= 0.55
                if delta < 1e-5 or noimp > 10:
                    break
    return best_h, best

def make_initial(n, seed):
    rng = np.random.RandomState(seed)
    x = np.linspace(0.0, 2.0, n)
    h = 0.5 + 0.14 * np.sin(np.pi * x) + 0.07 * np.sin(3.0 * np.pi * x)
    noise = rng.randn(n)
    noise -= noise.mean()
    h += 0.04 * noise
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + max(20.0, float(budget_s) - 5.0)
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        h0 = make_initial(700, seed)
    n0 = int(len(h0))
    if n0 < 50:
        h0 = make_initial(700, seed)
        n0 = 700
    if n0 > 1000:
        h0 = interp_to(h0, 1000)
        n0 = 1000
    best_h = project_feasible(h0, n0)
    best_c = c5_from_c(calc_c(best_h, n0), n0)
    best_n = n0
    if best_n != 700:
        h7 = interp_to(best_h, 700)
        c7 = c5_from_c(calc_c(h7, 700), 700)
        if c7 < best_c:
            best_h = h7
            best_c = c7
            best_n = 700

    end1 = deadline_at(t0, deadline, budget_s, 0.30, 0.45, time.time())
    h_try, c_try = coord_cd(best_h, best_n, 120, end1, rng, 34, 0, True)

    end2 = deadline_at(t0, deadline, budget_s, 0.52, 0.38, time.time())
    h_try, c_try = lse_refine(best_h, best_n, end2, [100, 1000, 10000], 0.06)
    if c_try < best_c:
        best_h, best_c, best_n = h_try, c_try, len(h_try)

    end3 = deadline_at(t0, deadline, budget_s, 0.68, 0.28, time.time())
    h_try, c_try = coord_cd(best_h, best_n, 200, end3, rng, 34, 0, True)
    if c_try < best_c:
        best_h, best_c, best_n = h_try, c_try, len(h_try)

    end_anti = deadline_at(t0, deadline, budget_s, 0.78, 0.20, time.time())
    nA = best_n if best_n % 2 == 0 else best_n - 1
    if nA >= 100 and time.time() < end_anti - 10:
        hA = apply_symmetry(best_h, nA, 2)
        h_try, c_try = coord_cd(hA, nA, 100, end_anti, rng, 34, 2, True)
        if c_try < best_c:
            best_h, best_c, best_n = h_try, c_try, len(h_try)

    end_multi = deadline_at(t0, deadline, budget_s, 0.90, 0.10, time.time())
    cands = [560, 700, 840, 1000, 500, 900]
    seen = {best_n}
    for nt in cands:
        if time.time() > end_multi - 40:
            break
        if nt in seen or nt < 200 or nt > 1000:
            continue
        seen.add(nt)
        rem = end_multi - time.time()
        if rem < 70:
            break
        sub_end = time.time() + min(150.0, rem * 0.55)
        h_try = interp_to(best_h, nt)
        h_try, c_try = coord_cd(h_try, nt, 80, sub_end, rng, 32, 0, True)
        if c_try < best_c:
            best_h, best_c, best_n = h_try, c_try, len(h_try)
        if time.time() < sub_end - 15:
            sub_end2 = min(end_multi, time.time() + 60.0)
            h_try2, c_try2 = lse_refine(h_try, nt, sub_end2, [1000], 0.04)
            if c_try2 < best_c:
                best_h, best_c, best_n = h_try2, c_try2, len(h_try2)
            if time.time() < end_multi - 15:
                h_try3, c_try3 = coord_cd(best_h, best_n, 50, end_multi, rng, 30, 0, True)
                if c_try3 < best_c:
                    best_h, best_c, best_n = h_try3, c_try3, len(h_try3)

    end_pert = deadline - 3
    eps_list = [0.07, 0.045, 0.028, 0.014, 0.007]
    n_best = len(best_h)
    for eps in eps_list:
        for _ in range(2):
            if time.time() > end_pert - 25:
                break
            noise = rng.randn(n_best)
            noise -= noise.mean()
            z = project_feasible(best_h + eps * noise, n_best)
            rem = end_pert - time.time()
            if rem < 30:
                break
            sub_end = time.time() + min(80.0, rem * 0.4)
            h_try, c_try = coord_cd(z, n_best, 30, sub_end, rng, 28, 0, True)
            if c_try < best_c:
                best_h, best_c, best_n = h_try, c_try, len(h_try)
                n_best = len(best_h)

    best_h = project_feasible(best_h, len(best_h))
    best_c = c5_eval(best_h, len(best_h))
    return best_h, best_c, len(best_h)
