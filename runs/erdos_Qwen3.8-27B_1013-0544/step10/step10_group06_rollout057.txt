def project_feasible(h, n):

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def enforce_symmetry(h, n):
    h_sym = 0.5 * (h + 1.0 - h[::-1])
    return project_feasible(h_sym, n)

    sh = alpha * (c - m)
    mx = sh.max()
    lse = mx + np.log(np.sum(np.exp(sh - mx)))
    return (m + lse / alpha) * (2.0 / n)

def lse_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-18:
            continue
        kf = k - (n - 1)
        wf, wb = max(0, kf), min(n, n + kf)
        for i in range(wf, wb):
            jf = i + kf
            if 0 <= jf < n:
                grad[i] += w[k] * (1.0 - x[jf]) * dx
            jb = i - kf
            if 0 <= jb < n:
                grad[i] -= w[k] * x[jb] * dx
        if kf == 0:
