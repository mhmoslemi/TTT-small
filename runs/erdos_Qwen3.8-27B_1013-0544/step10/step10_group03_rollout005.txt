def compute_c5(h, n):
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * (2.0 / float(n)))


def finalize_sum(h, n):
    target = 0.5 * n
    for _ in range(12):
        err = target - float(h.sum())
        if abs(err) < 1e-13:
            break
        if err > 0.0:
            idx = int(np.argmax(1.0 - h))
            add = min(err, 1.0 - float(h[idx]))
            if add <= 0.0:
                break
            h[idx] += add
        else:
            idx = int(np.argmax(h))
            sub = min(-err, float(h[idx]))
            if sub <= 0.0:
                break
            h[idx] -= sub
    return h


def project_feasible(h, n):

def coord_pre(n):

    m_start = n - 1 - R

    return R, m_start, off_q, off_b


def diag_roots(cq):
