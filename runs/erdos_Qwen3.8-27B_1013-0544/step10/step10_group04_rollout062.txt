    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def cd_pass(h, n, c, order):
    """One pass of coordinate descent. Returns updated h and c."""
    improved = False
    for p in order:
        h0 = h[p]
        if h0 < 1e-13 or h0 > 1.0 - 1e-13:
            continue
        xn = solve_1d(c, h, p, n)
        if abs(xn - h0) > 1e-13:
            delta = xn - h0
            m_min = n - 1 - p
            m_max = 2 * n - 2 - p
            ms = np.arange(m_min, m_max + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            delta_c = np.zeros(len(ms))
            if np.any(vf):
                delta_c[vf] -= h[jf[vf]] * delta
            if np.any(vb):
                delta_c[vb] += (1.0 - h[jb[vb]]) * delta
            # Diagonal correction
            diag_pos = p  # position of n-1 in ms array
            if 0 <= diag_pos < len(ms):
                delta_c[diag_pos] -= delta * delta
            c[ms[0]:ms[-1]+1] += delta_c
            h[p] = xn
            improved = True
    return h, c, improved

def exact_cd(h, n, max_passes, deadline, random_order=False, rng=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for it in range(max_passes):
        if time.time() > deadline:
            break
        if random_order and rng is not None:
            order = rng.permutation(n)
        else:
            order = np.arange(n)
        h, c, improved = cd_pass(h, n, c, order)
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def smooth_max(c, alpha):
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return m + lse / alpha

def smooth_obj_and_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    val = smooth_max(c, alpha) * (2.0 / n)
    # Gradient: d/dx_p of c_k for each k
    # c_k = sum_i x[i] * (1-x[i+k-(n-1)])
    grad = np.zeros(n)
    for p in range(n):
        m_min = n - 1 - p
        m_max = 2 * n - 2 - p
        ms = np.arange(m_min, m_max + 1)
        jf = p + ms - (n - 1)
        jb = p - ms + (n - 1)
        vf = (jf >= 0) & (jf < n)
        vb = (jb >= 0) & (jb < n)
        contrib = np.zeros(len(ms))
        if np.any(vf):
            contrib[vf] += (1.0 - x[jf[vf]])
        if np.any(vb):
            contrib[vb] -= x[jb[vb]]
        diag_pos = (n-1) - m_min
        if 0 <= diag_pos < len(ms):
            contrib[diag_pos] = (1.0 - 2.0 * x[p])
        # Weight by softmax of c
        c_shifted = alpha * (c - c.max())
        weights = np.exp(c_shifted - c_shifted.max())
        weights = weights / weights.sum()
        grad[p] = np.sum(contrib * weights[ms]) * (2.0 / n)
    return val, grad

def slsqp_opt(h, n, alphas, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        
        def fun(x):
            return smooth_max(np.correlate(x, 1.0-x, mode='full'), alpha) * (2.0/n)
        
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
                 'jac': lambda x: np.ones(n)}]
        try:
            res = minimize(fun, best_h, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def threshold_and_project(h, n, threshold=0.5):
    """Push values toward 0 or 1, then project."""
    h2 = np.where(h > threshold, 1.0, 0.0)
    return project_feasible(h2, n)

def block_flip(h, n, block_size, rng):
    """Flip a random block: h[i] -> 1-h[i] for i in a random block."""
    h2 = h.copy()
    start = rng.randint(0, max(1, n - block_size))
    h2[start:start+block_size] = 1.0 - h2[start:start+block_size]
    return project_feasible(h2, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def solve_1d(c, h, p, n):
    h0 = h[p]
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms]
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    if has_diag:
        c_diag = c_base[diag_idx[0]]
        a[diag_idx] = 0.0
    unaff = 0.0
    if m_min > 0:
        unaff = max(unaff, float(np.max(c[:m_min])))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    cur_max = float(np.max(c))
    neg_mask = a < -1e-15
    pos_mask = a > 1e-15
    zero_mask = np.abs(a) <= 1e-15
    L_lo, L_hi = unaff, cur_max

        x_lo, x_hi = 0.0, 1.0
        ok = True
        if np.any(zero_mask) and np.any(c_base[zero_mask] > L):
            ok = False
        if ok and np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if ok and np.any(pos_mask):

        if x_lo > x_hi + 1e-13:
            ok = False

            Cq = L - c_diag + h0 - h0 * h0

                disc = 1.0 - 4.0 * Cq
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                if x_lo > r1 and x_hi < r2:
                    ok = False
        L_hi = L if ok else L_lo
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo, x_hi = max(x_lo, 0.0), min(x_hi, 1.0)

def cd_pass(h, c, n, order):
    improved = False
    for p in order:
        h0 = h[p]
        if h0 < 1e-13 or h0 > 1.0 - 1e-13:
            continue
        xn = solve_1d(c, h, p, n)
        if abs(xn - h0) > 1e-13:
            d = xn - h0
            ms = np.arange(n - 1 - p, 2 * n - 1 - p)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            dc = np.zeros(len(ms))
            if np.any(vf):
                dc[vf] -= h[jf[vf]] * d
            if np.any(vb):
                dc[vb] += (1.0 - h[jb[vb]]) * d
            dp = p
            if 0 <= dp < len(ms):
                dc[dp] -= d * d
            c[ms[0]:ms[-1] + 1] += dc
            h[p] = xn
            improved = True
    return improved

def exact_cd(h, n, max_passes, deadline, random_order=False, rng=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    recompute = False
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        order = rng.permutation(n) if (random_order and rng is not None) else np.arange(n)
        imp = cd_pass(h, c, n, order)
        if not imp:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
            recompute = True
    return h, compute_c5(h, n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return (m + mx + np.log(np.sum(np.exp(sh - mx)))) / alpha * (2.0 / n)

def slsqp_opt(h, n, alpha, maxiter, deadline):
    bounds = [(0.0, 1.0)] * n
    def fun(x):
        return smooth_obj(x, n, alpha)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(fun, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h

def gen_restarts(h, n, rng):
    restarts = []
    for s in [0.02, 0.04, 0.07, 0.1, 0.15, 0.2]:
        restarts.append(project_feasible(h + rng.normal(0, s, n), n))

    for bs in [5, 15, 30, 50]:

        st = rng.randint(0, max(1, n - bs))

        restarts.append(project_feasible(h2, n))
    nb = max(4, n // 70)

    for f in [3, 7, 13, 23]:

    return restarts
