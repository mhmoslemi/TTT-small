    if verify:
        new_aff = aff + a * delta
        new_aff[diag] -= delta * delta
        nm = max(unaff, float(np.max(new_aff)))
        if nm > cur_max + 1e-8:
            return False

    c[start:end + 1] += a * delta
    c[np1] -= delta * delta
    h[p] = x
    return True


def exact_cd(h, n, max_passes, deadline, bisect_iters=28, verify=True):
    h = project_feasible(h.copy(), n)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    best_c = float(np.max(c)) * dx
    best_h = h.copy()

    for _ in range(max_passes):
        if time.time() > deadline:
            break
        cur = float(np.max(c)) * dx
        if cur < best_c:
            best_c = cur
            best_h = h.copy()

        changed = False
        for p in range(n):
            if time.time() > deadline:
                break
            if coord_update_inplace(c, h, p, n, bisect_iters, verify):
                changed = True

        if not changed:
            break

        h = project_feasible(h, n)
        c = np.correlate(h, 1.0 - h, mode="full")

    h = project_feasible(h, n)
    c = np.correlate(h, 1.0 - h, mode="full")
    cur = float(np.max(c)) * (2.0 / n)
    if cur < best_c:
        best_c = cur
        best_h = h.copy()
    return best_h, best_c


def smooth_value(x):
    n = _SL_N
    alpha = _SL_ALPHA
    c = np.correlate(x, 1.0 - x, mode="full")
    m = float(np.max(c))
    sh = alpha * (c - m)
    mx = float(np.max(sh))
    return float((m + np.log(np.sum(np.exp(sh - mx))) / alpha) * (2.0 / n))


def smooth_jac(x):
    n = _SL_N
    alpha = _SL_ALPHA
    c = np.correlate(x, 1.0 - x, mode="full")
    m = float(np.max(c))
    sh = alpha * (c - m)
    mx = float(np.max(sh))
    lam = np.exp(sh - mx)
    lam /= lam.sum()

    g = np.zeros(n, dtype=float)
    np1 = n - 1
    for t, w in enumerate(lam):
        if w <= 1e-11:
            continue
        k = t - np1
        if k >= 0:
            mmm = k
            L2 = n - mmm
            if L2 > 0:
                g[mmm:n] += w * (1.0 - x[0:L2])
                g[0:L2] -= w * x[mmm:n]
        else:
            mmm = -k
            L2 = n - mmm
            if L2 > 0:
                g[0:L2] += w * (1.0 - x[mmm:n])
                g[mmm:n] -= w * x[0:L2]
    return (g * (2.0 / n)) / alpha


def slsqp_constraint(x):
    return float(x.sum() - 0.5 * _SL_N)


def slsqp_constraint_jac(x):
    return np.ones(_SL_N, dtype=float)


def slsqp_opt(h, n, alphas, maxiter, deadline):
    global _SL_N, _SL_ALPHA
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": slsqp_constraint, "jac": slsqp_constraint_jac}]
    _SL_N = int(n)

    for alpha in alphas:
        if time.time() > deadline:
            break
        _SL_ALPHA = float(alpha)
        try:
            res = minimize(
                smooth_value,
                best_h,
                method="SLSQP",
                jac=smooth_jac,
                bounds=bounds,
                constraints=cons,
                options={"maxiter": int(maxiter), "ftol": 1e-10},
            )
            cand = project_feasible(res.x, n)
            cc = compute_c5(cand, n)
            if cc < best_c:
                best_c = cc
                best_h = cand
        except Exception:
            pass
    return best_h, best_c


def resample_h(h, n):
    m = len(h)
    if m == n:
        return project_feasible(h.copy(), n)
    xo = np.linspace(0.0, 2.0, m, endpoint=False)
    xn = np.linspace(0.0, 2.0, n, endpoint=False)
    return project_feasible(np.interp(xn, xo, h), n)


def random_h(n, rng, mode=0):
    if mode == 0:
        h = np.zeros(n, dtype=float)
        k = int(n // 2)
        if k > 0:
            h[rng.choice(n, k, replace=False)] = 1.0
        return project_feasible(h, n)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    f = np.zeros(n, dtype=float)
    for m in range(1, 6):
        f += rng.uniform(0.3, 1.3) * np.sin(np.pi * m * x + rng.uniform(0.0, 2.0 * np.pi))
    q = float(np.sort(f)[n // 2])
    h = 1.0 / (1.0 + np.exp(-10.0 * (f - q)))
    return project_feasible(h, n)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget_s = float(budget_s)
    deadline = t0 + max(1.0, budget_s - 5.0)
    rng = np.random.RandomState(seed)

    try:
        h_best = np.asarray(initial_h_values, dtype=float).copy()
        n_best = int(len(h_best))
    except Exception:
        h_best = random_h(700, rng, 0)
        n_best = 700

    h_best = project_feasible(h_best, n_best)
    c_best = compute_c5(h_best, n_best)

    # Strong polish of the parent construction.
    lim = min(deadline, t0 + max(30.0, 0.22 * budget_s))
    h1, c1 = exact_cd(h_best, n_best, 90, lim, 30, True)
    if c1 < c_best:
        h_best, n_best, c_best = h1, int(len(h1)), float(c1)

    # Short smooth-max polish.
    if n_best <= 800 and time.time() < deadline - 60.0:
        lim = min(deadline - 30.0, t0 + max(60.0, 0.40 * budget_s))
        h2, c2 = slsqp_opt(h_best, n_best, [300, 1200, 4000], 12, lim)
        if c2 < c_best:
            h_best, n_best, c_best = h2, int(len(h2)), float(c2)

    # Second exact coordinate pass.
    lim = min(deadline - 120.0, t0 + max(90.0, 0.62 * budget_s))
    h3, c3 = exact_cd(h_best, n_best, 160, lim, 30, True)
    if c3 < c_best:
        h_best, n_best, c_best = h3, int(len(h3)), float(c3)

    # Sweep nearby resolutions.
    seen = set()
    for d in (-240, -160, -100, -50, -20, 20, 50, 100, 160, 240):
        nt = n_best + d
        if nt % 2 == 0 and 80 <= nt <= 1000 and nt not in seen:
            seen.add(nt)
            if time.time() > deadline - 120.0:
                break
            lim = min(deadline - 100.0, time.time() + 80.0)
            hs = resample_h(h_best, nt)
            hs, cs = exact_cd(hs, nt, 80, lim, 28, False)
            if cs < c_best:
                h_best, n_best, c_best = hs, nt, float(cs)
            elif nt <= 500 and cs < c_best + 0.003 and time.time() < deadline - 100.0:
                lim2 = min(deadline - 80.0, time.time() + 50.0)
                hb2, cb2 = slsqp_opt(h_best, n_best, [800, 3000], 12, lim2)
                if cb2 < c_best:
                    h_best, n_best, c_best = hb2, int(len(hb2)), float(cb2)

    # Small random / smooth restarts to escape local basins.
    small_ns = (80, 120, 160, 200, 280, 360, 480)
    for ns in small_ns:
        if time.time() > deadline - 160.0:
            break
        nstarts = 3 if ns <= 160 else 2
        for mm in range(nstarts):
            if time.time() > deadline - 130.0:
                break
            lim = min(deadline - 110.0, time.time() + 35.0)
            hr = random_h(ns, rng, mm % 2)
            hr, cr = exact_cd(hr, ns, 90, lim, 24, False)
            if cr < c_best:
                h_best, n_best, c_best = hr, ns, float(cr)
            if cr < c_best + 0.002 and n_best != ns and time.time() < deadline - 90.0:
                lim2 = min(deadline - 70.0, time.time() + 45.0)
                h7 = resample_h(h_best if c_best < cr else hr, n_best)
                h7, c7 = exact_cd(h7, n_best, 50, lim2, 26, False)
                if c7 < c_best:
                    h_best, n_best, c_best = h7, n_best, float(c7)

    # Controlled shaking of the best state.
    for rr in range(8):
        if time.time() > deadline - 70.0:
            break
        eps = 0.002 + 0.0035 * rr
        valid = n_best if n_best <= best_n if "best_n" in dir() else n_best
        lim = min(deadline - 45.0, time.time() + 45.0)
        hp = project_feasible(h_best + rng.normal(0.0, eps, int(len(h_best))), int(len(h_best)))
        hp, cp = exact_cd(hp, int(len(h_best)), 60, lim, 28, False)
        if cp < c_best:
            h_best, n_best, c_best = hp, int(len(hp)), float(cp)

    # One more large random restart if enough time remains.
    if time.time() < deadline - 120.0 and n_best >= 300:
        lim = min(deadline - 80.0, time.time() + 70.0)
        hr = random_h(n_best, rng, 0)
        hr, cr = exact_cd(hr, n_best, 70, lim, 26, False)
        if cr < c_best:
            h_best, n_best, c_best = hr, n_best, float(cr)

    # Final polish.
    h4, c4 = exact_cd(h_best, n_best, 200, deadline - 15.0, 32, True)
    if c4 < c_best:
        h_best, n_best, c_best = h4, int(len(h4)), float(c4)

    if n_best <= 650 and time.time() < deadline - 25.0:
        h5, c5 = slsqp_opt(h_best, n_best, [1500, 6000], 10, deadline - 5.0)
        if c5 < c_best:
            h_best, n_best, c_best = h5, int(len(h5)), float(c5)

    h_best = project_feasible(h_best, n_best)
    c_best = compute_c5(h_best, n_best)
    return h_best, float(c_best), int(n_best)
