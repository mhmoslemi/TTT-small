    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms]
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    unaff = 0.0
    if m_min > 0:
        unaff = float(np.max(c[:m_min]))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    cur_max = float(np.max(c))
    neg_mask = a < -1e-14
    pos_mask = a > 1e-14
    zero_mask = np.abs(a) <= 1e-14
    L_lo, L_hi = unaff, cur_max
    for _ in range(40):
        if L_hi - L_lo < 1e-13:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        feasible = True
        if np.any(zero_mask) and np.any(c_base[zero_mask] > L + 1e-13):
            feasible = False
        if feasible and np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if feasible and np.any(pos_mask):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        if x_lo > x_hi + 1e-11:
            feasible = False
        if feasible and has_diag:
            cd_ = c_base[diag_idx[0]]
            ad = a[diag_idx[0]]
            disc = ad * ad + 4.0 * (cd_ - L)
            if disc >= 0:
                sq = np.sqrt(disc)
                r1 = (ad - sq) / 2.0
                r2 = (ad + sq) / 2.0
                if x_lo > h0 + r1 + 1e-11 and x_hi < h0 + r2 - 1e-11:
                    feasible = False
        if feasible:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        cd_ = c_base[diag_idx[0]]
        ad = a[diag_idx[0]]
        disc = ad * ad + 4.0 * (cd_ - L)
        if disc >= 0:

            r1 = (ad - sq) / 2.0
            r2 = (ad + sq) / 2.0
            if x_lo <= h0 + r1:

            elif x_hi >= h0 + r2:
                x_lo = max(x_lo, h0 + r2)
            else:

                    x_hi = min(x_hi, h0 + r1)
                else:
                    x_lo = max(x_lo, h0 + r2)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def cd(h, n, max_passes, deadline, rng, random_order=False):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        if random_order:
            order = rng.permutation(n)
        else:
            order = range(n)
        for p in order:
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            xn = solve_coord(c, h, p, n)
            if abs(xn - h0) > 1e-13:
                delta = xn - h0
                m_min = max(0, n - 1 - p)
                m_max = min(2 * n - 2, 2 * n - 2 - p)
                ms = np.arange(m_min, m_max + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                c[ms] += delta * a
                di = np.where(ms == (n - 1))[0]
                if len(di) > 0:
                    c[ms[di[0]]] -= delta * delta
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def slsqp_smooth(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def _cons_fun(x):
    return x.sum() - len(x) / 2.0

def _cons_jac(x):
    return np.ones(len(x))

def slsqp_opt(h0, n, alphas, maxiter, deadline):
    best_h, best_c = h0.copy(), compute_c5(h0, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(slsqp_smooth, best_h, args=(n, alpha), method='SLSQP',
                           bounds=[(0, 1)] * n,
                           constraints=[{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}],
                           options={'maxiter': maxiter, 'ftol': 1e-15})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

    xt = np.linspace(0, 2, n_tgt, endpoint=False)
    return project_feasible(np.interp(xt, xs, h_src), n_tgt)

def subgrad_step(h, n, n_active=30, step=0.02):
    """Take a step in the negative subgradient direction of the max."""
    c = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(c)[-n_active:]
    grad = np.zeros(n)
    for m in idx:
        lag = m - (n - 1)
        if lag >= 0:
            # c[m] = sum_{i=0}^{n-1-lag} h[i+lag]*(1-h[i])
            # ∂c[m]/∂h[p] = (1-h[p-lag]) if p>=lag, minus h[p+lag] if p+lag<n

            for p in range(0, n - lag):

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def solve_coord(c, h, p, n):
    h0 = h[p]
    m_min = max(0, n - 1 - p)
    m_max = min(2 * n - 2, 2 * n - 2 - p)
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms]
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    unaff = 0.0
    if m_min > 0:
        unaff = float(np.max(c[:m_min]))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    cur_max = float(np.max(c))
    neg_mask = a < -1e-14
    pos_mask = a > 1e-14
    zero_mask = np.abs(a) <= 1e-14
    L_lo, L_hi = unaff, cur_max
    for _ in range(40):
        if L_hi - L_lo < 1e-13:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        feasible = True
        if np.any(zero_mask) and np.any(c_base[zero_mask] > L + 1e-13):
            feasible = False
        if feasible and np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if feasible and np.any(pos_mask):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        if x_lo > x_hi + 1e-11:
            feasible = False
        if feasible and has_diag:
            cd_ = c_base[diag_idx[0]]
            ad = a[diag_idx[0]]
            disc = ad * ad + 4.0 * (cd_ - L)
            if disc >= 0:
                sq = np.sqrt(disc)
                r1 = (ad - sq) / 2.0
                r2 = (ad + sq) / 2.0
                if x_lo > h0 + r1 + 1e-11 and x_hi < h0 + r2 - 1e-11:
                    feasible = False
        if feasible:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        cd_ = c_base[diag_idx[0]]
        ad = a[diag_idx[0]]
        disc = ad * ad + 4.0 * (cd_ - L)
        if disc >= 0:
            sq = np.sqrt(disc)
            r1 = (ad - sq) / 2.0
            r2 = (ad + sq) / 2.0
            if x_lo <= h0 + r1:
                x_hi = min(x_hi, h0 + r1)
            elif x_hi >= h0 + r2:
                x_lo = max(x_lo, h0 + r2)
            else:
                if abs(x_lo - (h0 + r1)) <= abs(x_hi - (h0 + r2)):
                    x_hi = min(x_hi, h0 + r1)
                else:
                    x_lo = max(x_lo, h0 + r2)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def cd(h, n, max_passes, deadline, rng, random_order=False):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        order = rng.permutation(n) if random_order else range(n)
        for p in order:
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            xn = solve_coord(c, h, p, n)
            if abs(xn - h0) > 1e-13:
                delta = xn - h0
                m_min = max(0, n - 1 - p)
                m_max = min(2 * n - 2, 2 * n - 2 - p)
                ms = np.arange(m_min, m_max + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                c[ms] += delta * a
                di = np.where(ms == (n - 1))[0]
                if len(di) > 0:
                    c[ms[di[0]]] -= delta * delta
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def subgrad_perturb(h, n, n_active, step):
    c = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(c)[-n_active:]
    grad = np.zeros(n)
    for m in idx:
        lag = m - (n - 1)
        if lag >= 0:
            grad[lag:] += (1.0 - h[:n - lag])
            grad[:n - lag] -= h[lag:]
        else:
            nl = -lag
            grad[:n - nl] += (1.0 - h[nl:])
            grad[nl:] -= h[:n - nl]
    gmax = np.abs(grad).max()
    if gmax > 1e-10:
        grad /= gmax
    return project_feasible(h - step * grad, n)

def _cons_fun(x):
    return x.sum() - len(x) / 2.0

def _cons_jac(x):
    return np.ones(len(x))

def slsqp_smooth(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_opt(h0, n, alpha, maxiter, deadline):
    best_h, best_c = h0.copy(), compute_c5(h0, n)
    if time.time() > deadline:
        return best_h, best_c
    try:
        res = minimize(slsqp_smooth, best_h, args=(n, alpha), method='SLSQP',
                       bounds=[(0, 1)] * n,
                       constraints=[{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        new_h = project_feasible(res.x, n)
        new_c = compute_c5(new_h, n)
        if new_c < best_c:
            best_h, best_c = new_h, new_c
    except Exception:
        pass
    return best_h, best_c

    xt = np.linspace(0, 2, n_tgt, endpoint=False)
    return project_feasible(np.interp(xt, xs, h_src), n_tgt)
