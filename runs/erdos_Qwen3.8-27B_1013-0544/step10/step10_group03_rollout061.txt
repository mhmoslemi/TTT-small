def project_feasible(h, n):
    target = n / 2.0

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def ps_obj(x, n, p):
    dx = 2.0 / n

    c = np.maximum(c, 1e-30)
    return (np.sum(c**p))**(1.0/p) + (x.sum() - n/2.0)**2 * 10.0

def ps_grad(x, n, p):

    ps = np.sum(c**p)

def lbfgs_opt(h, n, p_list, maxiter, deadline):

    for p in p_list:

        try:
            res = minimize(ps_obj, best_h, args=(n, p), jac=ps_grad,
                         method='L-BFGS-B', bounds=[(0,1)]*n,
                         options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:

        except Exception:
            pass
    return best_h, best_c

    jb = p - ms + (n - 1)
    a = np.zeros(len(ms))
    vf = (jf >= 0) & (jf < n)

    a[vb] -= h[jb[vb]]

    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline: break
        improved = False
        for p in range(n):
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13: continue

            if abs(xn - h0) > 1e-12:
                delta = xn - h0
                m_min = n - 1 - p
                m_max = 2 * n - 2 - p
                ms = np.arange(m_min, m_max + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf): a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb): a[vb] -= h[jb[vb]]
                c[ms] += delta * a
                diag_i = np.where(ms == (n - 1))[0]
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True
        if not improved: break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-12:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

        if c_r < best_c:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-15: break
        if diff > 0:
            room = 1.0 - h; tot = room.sum()
            if tot < 1e-15: break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def ps_obj(x, n, p):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    c = np.maximum(c, 1e-30)
    return (np.sum(c**p))**(1.0/p) + 50.0*(x.sum() - n/2.0)**2

def ps_grad(x, n, p):
    dx = 2.0 / n
    r = np.correlate(x, 1.0 - x, mode='full')
    c = np.maximum(r * dx, 1e-30)
    ps = np.sum(c**p)
    base = ps**(1.0/p - 1.0)
    w = base * (c**(p-1)) * dx
    grad = np.zeros(n)
    for m in range(2*n - 1):
        shift = m - (n - 1); wm = w[m]
        if abs(wm) < 1e-28: continue
        if shift >= 0:
            j_lo, j_hi = 0, n - shift
        else:
            j_lo, j_hi = -shift, n
        if j_lo < j_hi:
            ks = np.arange(j_lo, j_hi) + shift
            grad[j_lo:j_hi] += wm * (1.0 - x[ks])
        if shift > 0:
            j2_lo, j2_hi = shift, n
        else:
            j2_lo, j2_hi = 0, n + shift
        if j2_lo < j2_hi:
            ks2 = np.arange(j2_lo, j2_hi) - shift
            grad[j2_lo:j2_hi] -= wm * x[ks2]
    grad += 100.0 * (x.sum() - n/2.0)
    return grad

def lbfgs_opt(h, n, p_list, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for p in p_list:
        if time.time() > deadline: break
        try:
            res = minimize(ps_obj, best_h, args=(n, p), jac=ps_grad,
                         method='L-BFGS-B', bounds=[(0.0,1.0)]*n,
                         options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def solve_1d_bisect(c, h, p, n):
    h0 = h[p]
    m_min = n - 1 - p; m_max = 2*n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1); jb = p - ms + (n - 1)
    a = np.zeros(len(ms))
    vf = (jf >= 0) & (jf < n); vb = (jb >= 0) & (jb < n)
    a[vf] += (1.0 - h[jf[vf]]); a[vb] -= h[jb[vb]]
    c_base = c[ms].copy()
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    if has_diag: a[diag_idx] = 0.0
    unaff = 0.0
    if m_min > 0: unaff = float(np.max(c[:m_min]))
    if m_max < 2*n - 2: unaff = max(unaff, float(np.max(c[m_max+1:])))
    cur_max = float(np.max(c))
    neg_m = a < -1e-15; pos_m = a > 1e-15; zero_m = np.abs(a) <= 1e-15
    L_lo, L_hi = unaff, cur_max
    for _ in range(55):
        if L_hi - L_lo < 1e-14: break
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0; feasible = True
        if np.any(zero_m) and np.any(c_base[zero_m] > L + 1e-14): feasible = False
        if feasible and np.any(neg_m):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_m]) / a[neg_m])))
        if feasible and np.any(pos_m):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_m]) / a[pos_m])))
        if x_lo > x_hi + 1e-12: feasible = False
        if feasible and has_diag:
            Cq = L - c_base[diag_idx[0]] + h0 - h0*h0
            disc = 1.0 - 4.0*Cq
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc))/2.0; r2 = (1.0 + np.sqrt(disc))/2.0
                if not (max(x_lo,0.0) < r1 and min(x_hi,1.0) > r2): feasible = False
        L_hi = L if feasible else L_hi
        L_lo = L_lo if feasible else L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_m): x_lo = max(x_lo, float(np.max((L - c_base[neg_m]) / a[neg_m])))
    if np.any(pos_m): x_hi = min(x_hi, float(np.min((L - c_base[pos_m]) / a[pos_m])))
    x_lo = max(x_lo, 0.0); x_hi = min(x_hi, 1.0)
    if has_diag:
        Cq = L - c_base[diag_idx[0]] + h0 - h0*h0
        disc = 1.0 - 4.0*Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc))/2.0; r2 = (1.0 + np.sqrt(disc))/2.0
            if h0 - r1 <= r2 - h0: x_hi = min(x_hi, r1)
            else: x_lo = max(x_lo, r2)
    if x_lo > x_hi: return h0
    return float(np.clip((x_lo + x_hi)/2.0, 0.0, 1.0))

def exact_cd(h, n, max_passes, deadline, rng=None, randomized=False):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    order = list(range(n)) if rng is None else list(range(n))
    for _ in range(max_passes):
        if time.time() > deadline: break
        if randomized and rng is not None: rng.shuffle(order)
        improved = False
        for p in order:
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13: continue
            xn = solve_1d_bisect(c, h, p, n)
            if abs(xn - h0) > 1e-13:
                delta = xn - h0
                m_min = n-1-p; m_max = 2*n-2-p
                ms = np.arange(m_min, m_max+1)
                jf = p + ms - (n-1); jb = p - ms + (n-1)
                a = np.zeros(len(ms))
                vf = (jf>=0)&(jf<n); vb = (jb>=0)&(jb<n)
                if np.any(vf): a[vf] += (1.0-h[jf[vf]])
                if np.any(vb): a[vb] -= h[jb[vb]]
                c[ms] += delta * a
                diag_i = np.where(ms == (n-1))[0]
                if len(diag_i) > 0: c[ms[diag_i[0]]] -= delta*delta
                h[p] = xn; improved = True
        if not improved: break
        h = project_feasible(h, n)
        if abs(h.sum() - n/2.0) > 1e-12:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def fourier_perturb(h, n, amp, rng):
    h_new = h.copy(); x = np.arange(n)
    for _ in range(3):
        f = rng.randint(1, max(2, n//4))
        ph = rng.uniform(0, 2*np.pi)
        h_new += amp * np.sin(2*np.pi*f*x/n + ph)
    return project_feasible(h_new, n)
