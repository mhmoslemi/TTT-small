    lse = mx + np.log(np.sum(np.exp(s - mx)))
    return (m + lse / alpha) * (2.0 / n)

        h[i], h[j] = new_i, new_j
        c_new = compute_c5(h, n)
        if c_new < c_cur - 1e-14:
            c_cur = c_new
        else:
            h[i], h[j] = old_i, old_j
    return h, c_cur

def fista_solve(h, n, alpha, steps, deadline):
    x = h.copy()
    z = h.copy()
    t = 1.0
    c = np.correlate(x, 1.0 - x, mode='full')
    L_bound = 4.0 * n / (alpha * 2.0)  # rough Lipschitz bound
    for k in range(steps):
        if time.time() > deadline:
            break
        w = np.exp(alpha * (c - c.max()))
        w /= w.sum()
        grad = np.zeros(n)
        for p in range(n):
            m_min = n - 1 - p
            m_max = 2 * n - 2 - p
            ms = np.arange(m_min, m_max + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            contrib = np.zeros(len(ms))
            if np.any(vf):
                contrib[vf] += (1.0 - x[jf[vf]])
            if np.any(vb):
                contrib[vb] -= x[jb[vb]]
            di = ms == (n - 1)
            if np.any(di):
                contrib[di] = 1.0 - 2.0 * x[p]
            grad[p] = np.dot(w[ms], contrib)
        grad *= (2.0 / n)
        L_step = 2.0 / (np.max(np.abs(grad)) + 1e-10)
        L_step = min(L_step, 0.5 / alpha)
        z = np.clip(z - L_step * grad, 0.0, 1.0)
        z = project_feasible(z, n)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        x = z + ((t - 1.0) / t_new) * (z - x)
        x = np.clip(x, 0.0, 1.0)
        t = t_new
        c = np.correlate(z, 1.0 - z, mode='full')
    return project_feasible(z, n), compute_c5(z, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        n = 500
        h_best = np.full(n, 0.5)
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)

    # Phase 1: FISTA with smooth objective (fast, good for initialization)
    if time.time() < deadline - 400:
        for alpha in [5000, 20000, 80000]:
            if time.time() > deadline - 300:
                break
            hf, cf = fista_solve(h_best, n, alpha, 300, deadline - 200)
            if cf < best_c:
                h_best, best_c = hf, cf

    # Phase 2: SLSQP with moderate smoothing
    if time.time() < deadline - 300:
        h1, c1 = slsqp_phase(h_best, n, [30000, 100000], 200, deadline - 250)
        if c1 < best_c:
            h_best, best_c = h1, c1

    # Phase 3: Heavy CD (fixed order)
    h2, c2 = do_cd(h_best, n, 800, deadline, rand_order=False, nb=25)
    if c2 < best_c:
        h_best, best_c = h2, c2

    # Phase 4: CD with random order (multiple seeds)
    for s in [1, 2, 3]:
        if time.time() > deadline - 100:
            break
        rng2 = np.random.RandomState(seed + s * 1000)
        h3, c3 = do_cd(h_best, n, 400, deadline - 80, rand_order=True, rng=rng2, nb=25)
        if c3 < best_c:
            h_best, best_c = h3, c3

    # Phase 5: Multi-n exploration
    n_list = [400, 550, 600, 450, 650]
    for n_try in n_list:
        if time.time() > deadline - 120:
            break
        if n_try == len(h_best):
            continue
        x_old = np.linspace(0, 2, len(h_best))
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_old, h_best), n_try)
        ho, co = do_cd(h_int, n_try, 400, deadline - 80, nb=25)
        if co < best_c - 1e-7:
            h_best, best_c, n = ho, co, n_try

    # Phase 6: Pair shift local search
    if time.time() < deadline - 30:
        n = len(h_best)
        h_best, best_c = pair_shift(h_best, n, 5000, deadline - 10, rng)

    # Phase 7: Final CD polish with higher precision
    if time.time() < deadline - 5:
        n = len(h_best)
        hf, cf = do_cd(h_best, n, 300, deadline - 2, nb=40)
        if cf < best_c:
            h_best, best_c = hf, cf

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
