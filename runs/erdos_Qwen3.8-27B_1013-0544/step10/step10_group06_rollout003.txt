def project_box_sum(y, S):
    y = np.asarray(y, dtype=float)
    y = np.clip(y, 0.0, 1.0)
    s = y.sum()
    if abs(s - S) < 1e-14:
        return y
    lo, hi = -1.0, 1.0
    for _ in range(80):
        tau = 0.5*(lo+hi)
        t = np.clip(y - tau, 0.0, 1.0)
        st = t.sum()
        if st > S:
            lo = tau
        else:
            hi = tau
    tau = 0.5*(lo+hi)
    h = np.clip(y - tau, 0.0, 1.0)
    # final exact
    diff = S - h.sum()
    for _ in range(5):
        if abs(diff) < 1e-13: break
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
            room = 1.0 - h[idx]
            if room <= 1e-15: break
            add = min(diff, room)
        else:
            idx = int(np.argmax(h))
            room = h[idx]
            if room <= 1e-15: break
            add = max(diff, -room)
        h[idx] += add
        diff = S - h.sum()
    return h

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def project_box_sum(y, S):
    y = np.asarray(y, dtype=float)
    y = np.clip(y, 0.0, 1.0)
    if abs(float(y.sum()) - S) < 1e-14:
        return y
    lo, hi = -1.0, 1.0
    for _ in range(80):
        tau = 0.5 * (lo + hi)
        t = np.clip(y - tau, 0.0, 1.0)
        st = float(t.sum())
        if st > S:
            lo = tau
        else:
            hi = tau
    h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    diff = S - float(h.sum())
    for _ in range(8):
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            idx = int(np.argmax(1.0 - h))
            room = 1.0 - h[idx]
            if room <= 1e-15:
                break
            add = min(diff, room)
        else:
            idx = int(np.argmax(h))
            room = h[idx]
            if room <= 1e-15:
                break
            add = max(diff, -room)
        h[idx] += add
        diff = S - float(h.sum())
    return h


def get_aff(p, n):
    center = n - 1

    hi1 = center - p
    lo2 = p - center
    hi2 = p

    s2 = max(0, lo2 - dmin)
    e2 = min(L, hi2 - dmin + 1)
    return dmin, dmax, L, d, s1, e1, s2, e2

            xh = min(xh, float(np.min(h0 + (T - cp) / ap)))
        if an.size:
            xl = max(xl, float(np.max(h0 + (T - cn) / an)))
        if xl > xh + 1e-13:
            feasible = False
        else:
            if xl < 0.0:
                xl = 0.0
            if xh > 1.0:
                xh = 1.0
            if has_diag:
                Cq = T - cdiag + h0 - h0 * h0
                if Cq < -1e-12:
                    feasible = False
                elif Cq > 0.25 + 1e-12:
                    pass
                else:
                    disc = max(0.0, 1.0 - 4.0 * Cq)
                    sq = np.sqrt(disc)
                    r1 = 0.5 * (1.0 - sq)
                    r2 = 0.5 * (1.0 + sq)
                    left = (max(xl, 0.0) <= min(xh, r1) + 1e-13)
                    right = (max(xl, r2) <= min(xh, 1.0) + 1e-13)
                    if not (left or right):
                        feasible = False
        if feasible:
            hiT = T
        else:
            loT = T

    T = hiT
    xl = 0.0
    xh = 1.0
    if ap.size:
        xh = min(xh, float(np.min(h0 + (T - cp) / ap)))
    if an.size:
        xl = max(xl, float(np.max(h0 + (T - cn) / an)))
    xl = max(xl, 0.0)
    xh = min(xh, 1.0)

    intervals = []
    if not has_diag:
        if xl <= xh + 1e-13:
            intervals.append((xl, xh))
    else:
        Cq = T - cdiag + h0 - h0 * h0
        if Cq > 0.25 + 1e-12:
            if xl <= xh + 1e-13:
                intervals.append((xl, xh))
        elif Cq >= -1e-12:
            disc = max(0.0, 1.0 - 4.0 * Cq)
            sq = np.sqrt(disc)
            r1 = 0.5 * (1.0 - sq)
            r2 = 0.5 * (1.0 + sq)
            l1 = max(xl, 0.0)
            rr1 = min(xh, r1)
            if l1 <= rr1 + 1e-13:
                intervals.append((l1, rr1))
            l2 = max(xl, r2)
            rr2 = min(xh, 1.0)
            if l2 <= rr2 + 1e-13:
                intervals.append((l2, rr2))

    if not intervals:
        return h0

    best_iv = None
    best_dist = 1e100
    for l, r in intervals:
        if l > r:
            l, r = r, l
        if h0 < l:
            dist = l - h0
        elif h0 > r:
            dist = h0 - r
        else:
            dist = 0.0
        if dist < best_dist:
            best_dist = dist
            best_iv = (l, r)

    l, r = best_iv
    if l > r:
        l, r = r, l
    x = float(np.clip(h0, l, r))
    if cap is not None and cap >= 0.0:
        delta = x - h0
        if delta > cap:
            x = h0 + cap
        elif delta < -cap:
            x = h0 - cap
    if x < 0.0:
        x = 0.0
    elif x > 1.0:
        x = 1.0
    return x


def apply_c_delta_aff(c, h, p, old, new, aff, n):
    if aff is None:
        return
    dmin, dmax, L, d, s1, e1, s2, e2 = aff
    delta = new - old
    if abs(delta) < 1e-15:
        return
    center = n - 1
    change = np.zeros(L)
    if s1 < e1:
        change[s1:e1] += (1.0 - h[p + d[s1:e1]]) * delta
    if s2 < e2:
        change[s2:e2] -= h[p - d[s2:e2]] * delta
    if dmin <= 0 <= dmax:
        change[-dmin] -= delta * delta
    c[center + d] += change


def cd_opt(h_in, n, max_passes, deadline, cap, nbisect, proj_every, random_order, rng):
    target = 0.5 * n
    h = project_box_sum(np.clip(h_in, 0.0, 1.0), target)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    no_improve = 0

    for _pass in range(max_passes):
        if time.time() > deadline - 2.0:
            break
        pass_improved = False
        if random_order:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

        for step, p in enumerate(order):
            if time.time() > deadline - 2.0:
                break
            p = int(p)
            if proj_every > 0 and step > 0 and step % proj_every == 0:
                h = project_box_sum(h, target)
                c = np.correlate(h, 1.0 - h, mode='full')
            aff = get_aff(p, n)
            x = solve_coord_aff(c, h, p, aff, n, nbisect, cap)
            delta = x - h[p]
            if abs(delta) > 1e-12:
                apply_c_delta_aff(c, h, p, h[p], x, aff, n)
                h[p] = x
                pass_improved = True

        h = project_box_sum(h, target)
        c_new = compute_c5(h, n)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if c_new > best_c + 1e-5:
                h = best_h.copy()
        c = np.correlate(h, 1.0 - h, mode='full')
        if not pass_improved or no_improve >= 5:
            break

    return best_h, best_c


def lse_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(np.max(c))
    shifted = alpha * (c - m)
    mx = float(np.max(shifted))
    s = float(np.sum(np.exp(shifted - mx)))
    return (m + (mx + np.log(s)) / alpha) * (2.0 / n)


def lse_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(np.max(c))
    shifted = alpha * (c - m)
    mx = float(np.max(shifted))
    ev = np.exp(shifted - mx)
    s = float(ev.sum())
    w = ev / s
    val = (m + (mx + np.log(s)) / alpha) * (2.0 / n)

    grad = np.zeros(n)
    center = n - 1
    nz = np.where(w > 1e-15)[0]
    for k in nz:
        ws = float(w[k])
        d = int(k - center)
        if d == 0:
            grad += ws * (1.0 - 2.0 * x)
        elif d > 0:
            grad[:n - d] += ws * (1.0 - x[d:])
            grad[d:] -= ws * x[:n - d]
        else:
            m = -d
            grad[m:] += ws * (1.0 - x[:n - m])
            grad[:n - m] -= ws * x[m:]
    grad *= 2.0 / n
    return val, grad


def sum_func(x, n):
    return float(np.sum(x) - 0.5 * n)


def sum_jac(x, n):
    return np.ones(n)


def slsqp_opt(h, n, alphas, maxiter_each, deadline):
    target = 0.5 * n
    h = project_box_sum(np.clip(h, 0.0, 1.0), target)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0) for _ in range(n)]
    cons = [{'type': 'eq', 'fun': sum_func, 'jac': sum_jac, 'args': (n,)}]

    for alpha in alphas:
        if time.time() > deadline - 10.0:
            break
        try:
            res = minimize(
                lse_obj, best_h, args=(n, float(alpha)), method='SLSQP',
                bounds=bounds, constraints=cons,
                options={'maxiter': int(maxiter_each), 'ftol': 1e-12, 'disp': False}
            )
            x = project_box_sum(np.clip(res.x, 0.0, 1.0), target)
            cc = compute_c5(x, n)
            if cc < best_c:
                best_c = cc
                best_h = x.copy()
        except Exception:
            pass
    return best_h, best_c


def smooth_candidate(h, n, radius=4):
    r = max(1, int(radius))
    t = np.arange(-r, r + 1)
    kern = np.exp(-0.5 * (t / float(r)) ** 2)
    kern /= kern.sum()
    y = np.convolve(h, kern, mode='same')
    return project_box_sum(np.clip(y, 0.0, 1.0), 0.5 * n)


def perturb_candidate(h, n, rng, amt):
    y = h.copy()
    y += amt * rng.standard_normal(n)
    x = np.linspace(0.0, 2.0, n)
    for _ in range(2):
        f = rng.uniform(0.5, 4.0)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        amp = amt * rng.uniform(0.3, 1.0)
        y += amp * np.sin(2.0 * np.pi * f * x + ph)
    return project_box_sum(np.clip(y, 0.0, 1.0), 0.5 * n)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget = float(budget_s)
    deadline = t0 + budget - 2.0
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 700
        x = np.linspace(0.0, 2.0, n0)
        h0 = 0.5 + 0.25 * np.sin(np.pi * x) + 0.1 * np.sin(3.0 * np.pi * x)

    h = project_box_sum(np.clip(h0, 0.0, 1.0), 0.5 * n0)
    best_n = n0
    best_h = h.copy()
    best_c = compute_c5(h, best_n)

    phase_dead = min(deadline - 350.0, t0 + 220.0)
    if time.time() < phase_dead:
        hh, cc = cd_opt(h, best_n, 30, phase_dead, 0.04, 18, 40, True, rng)
        if cc < best_c:
            best_c, best_h, h = cc, hh, hh

    if time.time() < deadline - 250.0:
        hd = min(deadline - 250.0, time.time() + 120.0)
        hh, cc = slsqp_opt(h, best_n, [300.0, 2000.0], 25, hd)
        if cc < best_c:
            best_c, best_h, h = cc, hh, hh

    if time.time() < deadline - 150.0:
        hd = min(deadline - 150.0, time.time() + 300.0)
        hh, cc = cd_opt(h, best_n, 120, hd, 0.008, 24, 50, True, rng)
        if cc < best_c:
            best_c, best_h, h = cc, hh, hh

    if time.time() < deadline - 120.0:
        hd = min(deadline - 120.0, time.time() + 100.0)
        hh, cc = slsqp_opt(h, best_n, [8000.0], 25, hd)
        if cc < best_c:
            best_c, best_h, h = cc, hh, hh

    if time.time() < deadline - 90.0:
        cand_dead = min(deadline - 30.0, time.time() + 180.0)
        target = 0.5 * best_n
        cands = []
        ysym = 0.5 * (best_h + 1.0 - best_h[::-1])
        cands.append(project_box_sum(np.clip(ysym, 0.0, 1.0), target))
        cands.append(smooth_candidate(best_h, best_n, 4))
        cands.append(perturb_candidate(best_h, best_n, rng, 0.012))
        cands.append(perturb_candidate(best_h, best_n, rng, 0.025))

        for hcan in cands:
            if time.time() > cand_dead - 40.0:
                break
            ch, cc = cd_opt(hcan, best_n, 8, cand_dead, 0.02, 18, 30, True, rng)
            cand_h, cand_c = ch, cc
            if time.time() < cand_dead - 40.0 and cand_c < best_c + 0.01:
                sh, sc = slsqp_opt(ch, best_n, [2000.0], 15, cand_dead)
                th, tc = cd_opt(sh, best_n, 8, cand_dead, 0.008, 22, 30, False, rng)
                if tc < cand_c:
                    cand_h, cand_c = th, tc
                elif sc < cand_c:
                    cand_h, cand_c = sh, sc
            if cand_c < best_c:
                best_c, best_h, h = cand_c, cand_h, cand_h

    if time.time() < deadline - 5.0:
        hh, cc = cd_opt(h, best_n, 100000, deadline - 5.0, 0.002, 25, 60, True, rng)
        if cc < best_c:
            best_c, best_h = cc, hh

    best_h = project_box_sum(np.clip(best_h, 0.0, 1.0), 0.5 * best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, float(best_c), int(best_n)
