import numpy as np

from scipy.optimize import minimize_scalar

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project_feasible(a, n):
    a = np.asarray(a, dtype=float)
    target = 0.5 * n
    lo, hi = -2.0, 2.0
    for _ in range(70):
        t = 0.5 * (lo + hi)
        if np.clip(a + t, 0.0, 1.0).sum() < target:
            lo = t
        else:
            hi = t
    z = np.clip(a + 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(12):
        diff = target - z.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - z
            tot = room.sum()
            if tot <= 1e-16:
                break
            z = np.clip(z + diff * room / tot, 0.0, 1.0)
        else:
            tot = z.sum()
            if tot <= 1e-16:
                break
            z = np.clip(z + diff * z / tot, 0.0, 1.0)
    return z

def var_Lpos(h, r):
    n = len(h)
    L = np.zeros(n)
    L[0] = 1.0 - 2.0 * h[r]
    if r + 1 < n:
        L[1:n - r] += 1.0 - h[r + 1:n]
    if r > 0:
        L[1:r + 1] -= h[r - 1::-1]
    return L

def var_Lneg(h, r):
    n = len(h)
    L = np.zeros(n)
    if r > 0:
        L[1:r + 1] += 1.0 - h[r - 1::-1]
    if r + 1 < n:
        L[1:n - r] -= h[r + 1:n]
    return L

def pair_eval_scalar(t, base_pos, base_neg, Ap, An, k, dx):
    pos = base_pos + t * Ap
    neg = base_neg + t * An
    pos[0] -= 2.0 * t * t
    if k > 0:
        pos[k] += t * t
        neg[k] += t * t
    return dx * float(max(pos.max(), neg[1:].max()))

def pair_eval_grid(tvec, base_pos, base_neg, Ap, An, k, dx):
    t2 = tvec * tvec
    pos = base_pos[None, :] + tvec[:, None] * Ap[None, :]
    pos[:, 0] -= 2.0 * t2
    if k > 0:
        pos[:, k] += t2
    neg = base_neg[1:][None, :] + tvec[:, None] * An[1:][None, :]
    if k > 0:
        neg[:, k - 1] += t2
    return dx * np.maximum(pos.max(axis=1), neg.max(axis=1))

def pair_search(h, p, q, n, base_pos, base_neg, cur_c, grid_pts):
    if n < 2 or p == q:
        return h, cur_c
    oldp = float(h[p])
    oldq = float(h[q])
    lo = max(-oldp, oldq - 1.0)
    hi = min(1.0 - oldp, oldq)
    if hi - lo <= 1e-13:
        return h, cur_c
    dx = 2.0 / n
    Ap = var_Lpos(h, p) - var_Lpos(h, q)
    An = var_Lneg(h, p) - var_Lneg(h, q)
    k = abs(p - q)
    grid_pts = int(max(64, min(grid_pts, 512)))
    tvec = np.linspace(lo, hi, grid_pts)
    vals = pair_eval_grid(tvec, base_pos, base_neg, Ap, An, k, dx)
    idx = int(np.argmin(vals))
    best_t = float(tvec[idx])
    best_c = float(vals[idx])
    if best_c < cur_c - 1e-12:
        if idx == 0:
            a, b = lo, float(tvec[1])
        elif idx == grid_pts - 1:
            a, b = float(tvec[-2]), hi
        else:
            a, b = float(tvec[idx - 1]), float(tvec[idx + 1])
        if b > a:
            res = minimize_scalar(pair_eval_scalar, bounds=(a, b),
                                  args=(base_pos, base_neg, Ap, An, k, dx),
                                  method="bounded",
                                  options={"xatol": 1e-10, "maxiter": 120})
            rf = res.fun
            if rf is not None and rf < best_c:
                best_t = float(res.x)

    if best_c < cur_c - 5e-12:

        h2[p] = oldp + best_t
        h2[q] = oldq - best_t
        h2 = project_feasible(h2, n)
        c2 = compute_c5(h2, n)
        if c2 < cur_c - 5e-12:
            return h2, c2
    return h, cur_c

def pos_neg_corr(h):
    n = len(h)
    omh = 1.0 - h
    pos = np.empty(n)
    neg = np.empty(n)
    pos[0] = np.dot(h, omh)
    neg[0] = pos[0]
    for k in range(1, n):
        a = h[:-k]
        b = h[k:]
        pos[k] = np.dot(a, omh[k:])
        neg[k] = np.dot(b, omh[:-k])
    return pos, neg

def c_from_pos_neg(pos, neg, n):

        if not chosen:

            r1 = 0.5 * (1.0 - np.sqrt(disc))

def update_corr(c, h, p, old, new, n):
    d = new - old
    if abs(d) < 1e-14:
        return
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(ms.size)
    if np.any(vf):
        a[vf] = 1.0 - h[jf[vf]]
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c[ms] += d * a
    di = np.where(ms == (n - 1))[0]
    if di.size:
        c[ms[di[0]]] -= d * d

def exact_cd(h, n, max_passes, deadline, rng):
    h = h.copy()
    for it in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode="full")
        order = np.arange(n)
        if rng is not None and it % 2 == 1:
            rng.shuffle(order)
        improved = False
        for p in order:
            if time.time() > deadline:
                h = project_feasible(h, n)
                return h, compute_c5(h, n)
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            xn = solve_1d(c, h, p, n)
            if abs(xn - h0) > 1e-11:
                update_corr(c, h, p, h0, xn, n)
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    h = project_feasible(h, n)
    return h, compute_c5(h, n)
