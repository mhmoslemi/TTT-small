```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def solve_1d(c, h, p, n):
    h0 = h[p]
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf): a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb): a[vb] -= h[jb[vb]]
    c_base = c[ms]
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    if has_diag:
        c_diag = c_base[diag_idx[0]]
        a[diag_idx] = 0.0
    unaff = 0.0
    if m_min > 0: unaff = float(np.max(c[:m_min]))
    if m_max < 2 * n - 2: unaff = max(unaff, float(np.max(c[m_max + 1:])))
    cur_max = float(np.max(c))
    neg_mask = a < -1e-15
    pos_mask = a > 1e-15
    zero_mask = np.abs(a) <= 1e-15
    L_lo, L_hi = unaff, cur_max
    for _ in range(40):
        if L_hi - L_lo < 1e-15: break
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        feasible = True
        if np.any(zero_mask) and np.any(c_base[zero_mask] > L):
            feasible = False
        if feasible and np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if feasible and np.any(pos_mask):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        if x_lo > x_hi + 1e-14: feasible = False
        if feasible and has_diag:
            Cq = L - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                xc_lo, xc_hi = max(x_lo, 0.0), min(x_hi, 1.0)
                if not (xc_lo < r1 and xc_hi > r2): feasible = False
        if feasible: L_hi = L
        else: L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):

    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Cq = L - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:

                else: x_lo = max(x_lo, r2)
    if x_lo > x_hi: return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def update_correlation(c, h, p, delta, n):
    h0 = h[p]
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf): a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb): a[vb] -= h[jb[vb]]
    c[ms] += delta * a
    diag_idx = np.where(ms == (n - 1))[0]
    if len(diag_idx) > 0:
        c[ms[diag_idx[0]]] -= delta * delta
    h[p] = h0 + delta

def exact_cd(h, n, max_passes, deadline, order_type='fixed', rng=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline: break
        improved = False
        if order_type == 'fixed':
            order = np.arange(n)
        elif order_type == 'reverse':
            order = np.arange(n - 1, -1, -1)
        else:
            order = rng.permutation(n)
        for p in order:
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13: continue
            xn = solve_1d(c, h, p, n)
            if abs(xn - h0) > 1e-12:
                update_correlation(c, h, p, xn - h0, n)
                improved = True
        if not improved: break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def greedy_cd(h, n, max_passes, deadline):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    cur_max = float(np.max(c))
    for _ in range(max_passes):
        if time.time() > deadline: break
        # Compute optimal values and improvements for all coordinates
        best_p, best_new_p, best_improve = -1, 0, 0
        for p in range(n):
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13: continue
            xn = solve_1d(c, h, p, n)
            if abs(xn - h0) < 1e-12: continue
            # Estimate improvement: compute new max correlation
            delta = xn - h0
            # Save state to try
            old_h = h[p]
            m_min = n - 1 - p
            m_max = 2 * n - 2 - p
            ms = np.arange(m_min, m_max + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(vf): a[vf] += (1.0 - h[jf[vf]])
            if np.any(vb): a[vb] -= h[jb[vb]]
            old_c = c[ms].copy()
            c[ms] += delta * a
            diag_idx = np.where(ms == (n - 1))[0]
            if len(diag_idx) > 0: c[ms[diag_idx[0]]] -= delta * delta
            h[p] = xn
            new_max = float(np.max(c))
            c[ms] = old_c
            h[p] = old_h
            improve = cur_max - new_max
            if improve > best_improve:
                best_improve = improve
                best_p = p
                best_new_p = xn
        if best_p < 0: break
        # Apply the best update
        delta = best_new_p - h[best_p]
        update_correlation(c, h, best_p, delta, n)
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(c))
    return h, compute_c5(h, n)
```

```python
def smooth_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    max_c = c.max()
    shifted = alpha * (c - max_c)
    weights = np.exp(shifted - shifted.max())
    weights /= weights.sum()
    grad = np.zeros(n)
    for p in range(n):
        m_min = n - 1 - p
        m_max = 2 * n - 2 - p
        ms = np.arange(m_min, m_max + 1)
        jf = p + ms - (n - 1)
        jb = p - ms + (n - 1)
        vf = (jf >= 0) & (jf < n)
        vb = (jb >= 0) & (jb < n)
        contrib = np.zeros(len(ms))
        if np.any(vf): contrib[vf] += (1.0 - h[jf[vf]])
        if np.any(vb): contrib[vb] -= h[jb[vb]]
        diag = ms == (n - 1)
        contrib[diag] = 1.0 - 2.0 * h[p]
        grad[p] = np.dot(weights[ms], contrib)
    return (2.0 / n) * grad
```

```python
def smooth_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    mc = c.max()
    sh = alpha * (c - mc)
    w = np.exp(sh - sh.max())
    w /= w.sum()
    g = np.zeros(n)
    for p in range(n):
        mn = n - 1 - p
        mx = 2 * n - 2 - p
        ms = np.arange(mn, mx + 1)
        jf = p + ms - (n - 1)
        jb = p - ms + (n - 1)
        vf = (jf >= 0) & (jf < n)
        vb = (jb >= 0) & (jb < n)
        ct = np.zeros(len(ms))
        if np.any(vf):
            ct[vf] += 1.0 - h[jf[vf]]
        if np.any(vb):
            ct[vb] -= h[jb[vb]]
        dg = ms == n - 1
        ct[dg] = 1.0 - 2.0 * h[p]
        g[p] = np.dot(w[ms], ct)
    return (2.0 / n) * g
```

```python
        c_g = np.correlate(h_best, 1.0 - h_best, mode='full')

        for _ in range(800):
            if time.time() > deadline - 10:
                break
            best_p, best_xn, best_imp = -1, 0.0, 0.0
            sample = rng.choice(n, size=min(40, n), replace=False)
            for p in sample:
                h0 = h_g[p]
                if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                    continue
                xn = solve_1d(c_g, h_g, p, n)
                if abs(xn - h0) < 1e-12:
                    continue
                d = xn - h0
                mn = n - 1 - p
                mx = 2 * n - 2 - p
                ms = np.arange(mn, mx + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += 1.0 - h_g[jf[vf]]
                if np.any(vb):
                    a[vb] -= h_g[jb[vb]]
                oc = c_g[ms].copy()
                c_g[ms] += d * a
                di = np.where(ms == n - 1)[0]
                if len(di) > 0:
                    c_g[ms[di[0]]] -= d * d
                nm2 = float(np.max(c_g)) * (2.0 / n)
                c_g[ms] = oc
                if best_c - nm2 > best_imp:
                    best_imp = best_c - nm2
                    best_p, best_xn = p, xn
            if best_p < 0:
                break
            d = best_xn - h_g[best_p]
            upd_c(c_g, h_g, best_p, d, n)
            h_g = project_feasible(h_g, n)
            if abs(h_g.sum() - n / 2.0) > 1e-10:
                c_g = np.correlate(h_g, 1.0 - h_g, mode='full')
        gc = compute_c5(h_g, n)
        if gc < best_c:
            h_best, best_c = h_g, gc

    for nt in [500, 400, 600, 550, 450, 650, 350]:
        if time.time() > deadline - 25:
            break
        if nt == n:
            continue
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, nt, endpoint=False)
        h_i = project_feasible(np.interp(x_new, x_old, h_best, period=2), nt)
        hh, cc = exact_cd(h_i, nt, 40, deadline, rng, 'fixed')
        if cc < best_c:
            h_best, best_c, n = hh, cc, nt

    if time.time() < deadline - 3:
        hh, cc = exact_cd(h_best, n, 100, deadline, rng, 'fixed')
        if cc < best_c:
            h_best, best_c = hh, cc
        hh, cc = exact_cd(h_best, n, 50, deadline, rng, 'rev')
        if cc < best_c:
            h_best, best_c = hh, cc

    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)
    return h_best, best_c, n
```
