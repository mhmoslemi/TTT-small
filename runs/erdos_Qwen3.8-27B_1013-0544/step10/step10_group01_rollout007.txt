def project_feasible(x,n):
    x=np.clip(np.asarray(x,dtype=float),0,1)
    target=0.5*n
    diff=target-x.sum()
    if abs(diff)<1e-13: return x
    # First try uniform over interior, repeat up to 10
    for _ in range(10):
        if diff>0:
            mask=x<1-1e-14
        else:
            mask=x>1e-14
        if not np.any(mask): break
        x[mask]+=diff/np.count_nonzero(mask)
        x=np.clip(x,0,1)
        diff=target-x.sum()
        if abs(diff)<1e-13: break
    if abs(diff)>=1e-12:
       # L2 bisection
       lo=-1.0; hi=1.0
       for _ in range(50):
          mid=(lo+hi)/2
          y=np.clip(x+mid,0,1)
          if y.sum() < target: lo=mid
          else: hi=mid

          idx=interior; x[idx[0]] += diff

def project_feasible(x,n):
    x = np.clip(np.asarray(x, dtype=float), 0.0, 1.0)
    target = 0.5 * n
    diff = target - x.sum()
    if abs(diff) < 1e-13:
        return x
    # quick uniform
    for _ in range(5):
        if diff > 0:
            mask = x < 1.0 - 1e-14
        else:
            mask = x > 1e-14
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            break
        x[mask] += diff / cnt
        x = np.clip(x, 0.0, 1.0)
        diff = target - x.sum()
        if abs(diff) < 1e-13:
            break
    if abs(diff) >= 1e-12:
        lo, hi = -1.0, 1.0
        for _ in range(45):
            mid = 0.5*(lo+hi)
            y = np.clip(x + mid, 0.0, 1.0)
            if y.sum() < target:
                lo = mid
            else:
                hi = mid
        x = np.clip(x + 0.5*(lo+hi), 0.0, 1.0)
        diff = target - x.sum()

        if np.any(interior):

            x += diff / x.size

import numpy as np
import time

GR = 0.6180339887498949

def corr_fwd(h,n):
    return np.correlate(h, 1.0 - h, mode='full')[::-1]

def c5(h,n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def project_feasible(x,n):
    x = np.clip(np.asarray(x, dtype=float), 0.0, 1.0)
    target = 0.5 * n
    diff = target - x.sum()
    if abs(diff) < 1e-13:
        return x
    for _ in range(4):
        if diff > 0:
            mask = x < 1.0 - 1e-14
        else:
            mask = x > 1e-14
        cnt = int(np.count_nonzero(mask))
        if cnt == 0:
            break
        x[mask] += diff / cnt
        x = np.clip(x, 0.0, 1.0)
        diff = target - x.sum()
        if abs(diff) < 1e-13:
            break
    if abs(diff) >= 1e-12:
        lo, hi = -1.0, 1.0
        for _ in range(40):
            mid = 0.5 * (lo + hi)
            if np.clip(x + mid, 0.0, 1.0).sum() < target:
                lo = mid
            else:
                hi = mid
        x = np.clip(x + 0.5*(lo+hi), 0.0, 1.0)
        diff = target - x.sum()
    if diff != 0.0:
        interior = (x > 1e-14) & (x < 1.0 - 1e-14)
        if np.any(interior):
            x[interior] += diff / np.count_nonzero(interior)
        else:
            x += diff / x.size
    return np.clip(x, 0.0, 1.0)

def eval_coord(c_aff, B, diag, c_diag, h0, unaff, d):
    v = c_aff + d * B
    v[diag] = c_diag + (1.0 - 2.0 * h0) * d - d * d
    mx = float(v.max())
    return mx if mx > unaff else unaff

def min_step_coord(c_aff, B, diag, c_diag, h0, unaff, fcur, gs=12):
    lo = -h0
    hi = 1.0 - h0
    span = hi - lo
    if span < 1e-14:
        return 0.0
    best_d = 0.0
    best_f = fcur
    pts = (0.0, lo, lo + 0.25*span, lo + 0.5*span, lo + 0.75*span, hi)
    for aa in pts:
        fv = fcur if abs(aa) < 1e-15 else eval_coord(c_aff, B, diag, c_diag, h0, unaff, aa)
        if fv < best_f:
            best_f = fv; best_d = aa
    l = lo; r = hi
    d1 = r - GR*(r - l)
    d2 = l + GR*(r - l)
    f1 = fcur if abs(d1) < 1e-15 else eval_coord(c_aff, B, diag, c_diag, h0, unaff, d1)
    f2 = fcur if abs(d2) < 1e-15 else eval_coord(c_aff, B, diag, c_diag, h0, unaff, d2)
    for _ in range(gs):
        if f1 < f2:
            r = d2; d2 = d1; f2 = f1
            d1 = r - GR*(r - l)
            f1 = fcur if abs(d1) < 1e-15 else eval_coord(c_aff, B, diag, c_diag, h0, unaff, d1)
        else:
            l = d1; d1 = d2; f1 = f2
            d2 = l + GR*(r - l)
            f2 = fcur if abs(d2) < 1e-15 else eval_coord(c_aff, B, diag, c_diag, h0, unaff, d2)
    dg = 0.5*(l + r)
    fg = fcur if abs(dg) < 1e-15 else eval_coord(c_aff, B, diag, c_diag, h0, unaff, dg)
    if fg < best_f:
        best_f = fg; best_d = dg
    if best_f < fcur - 1e-12:
        return float(best_d)
    return 0.0

        if imp:
            if abs(h.sum() - 0.5 * n) > 1e-8:
                h = project_feasible(h, n)
            c = corr_fwd(h, n)
            cv = float(c.max() * (2.0 / n))
            if cv < bc:
                bh = h.copy()
                bc = cv
            stale = 0
        else:
            stale += 1
            if stale >= 2:
                break
    h = project_feasible(h, n)
    cv = c5(h, n)
    if cv < bc:
        bh = h.copy()
        bc = cv
    return bh, bc

def grad_from_indices(idx, h, n):
    g = np.zeros(n, dtype=float)
    ctr = n - 1
    for l in idx:
        l = int(l)
        s = l - ctr
        if s >= 0:
            m = n - s
            if m > 0:
                g[:m] += 1.0 - h[s:s + m]
                g[s:] += -h[:m]
        else:
            t = -s
            m = n - t
            if m > 0:
                g[t:] += 1.0 - h[:m]
                g[:m] += -h[t:]
    return g

def dir_line_search(h, d, n, f0, gs=8):
    eps = 1e-14
    pos = d > eps
    neg = d < -eps
    lo, hi = -1e6, 1e6
    if np.any(pos):
        dp = d[pos]
        hp = h[pos]
        lo = max(lo, float(np.min(-hp / dp)), 0.0)
        hi = min(hi, float(np.min((1.0 - hp) / dp)))
    if np.any(neg):
        dn = -d[neg]
        hn = h[neg]
        lo = max(lo, float(np.min((1.0 - hn) / dn)))
        hi = min(hi, float(np.min(hn / dn)))
    lo = max(lo, -5.0)
    hi = min(hi, 5.0)
    lo = min(lo, 0.0)
    hi = max(hi, 0.0)
    if hi - lo < 1e-12:
        return 0.0, f0
    ba = 0.0
    bf = f0
    sp = hi - lo
    for aa in (0.0, lo, lo + 0.25 * sp, lo + 0.5 * sp, lo + 0.75 * sp, hi):
        if abs(aa) < 1e-15:
            continue
        f = c5(h + aa * d, n)
        if f < bf:
            bf = f
            ba = aa
    l, r = lo, hi
    d1 = r - GR * (r - l)
    d2 = l + GR * (r - l)
    f1 = f0 if abs(d1) < 1e-15 else c5(h + d1 * d, n)
    f2 = f0 if abs(d2) < 1e-15 else c5(h + d2 * d, n)
    for _ in range(gs):
        if f1 < f2:
            r = d2
            d2 = d1
            f2 = f1
            d1 = r - GR * (r - l)
            f1 = f0 if abs(d1) < 1e-15 else c5(h + d1 * d, n)
        else:
            l = d1
            d1 = d2
            f1 = f2
            d2 = l + GR * (r - l)
            f2 = f0 if abs(d2) < 1e-15 else c5(h + d2 * d, n)
    dg = 0.5 * (l + r)
    fg = f0 if abs(dg) < 1e-15 else c5(h + dg * d, n)
    if fg < bf:
        bf = fg
        ba = dg
    return float(ba), float(bf)

def active_dir_search(h0, n, stop, rng, gs=8):
    h = project_feasible(h0, n)
    bc = c5(h, n)
    fails = 0
    while time.time() < stop:
        c = corr_fwd(h, n)
        L = 2 * n - 1
        T = min(L, max(8, L // 5))
        top = np.argsort(c)[::-1][:T]
        kmax = int(min(T, max(4, T // 3)))
        kmin = int(max(3, T // 10))
        if kmin > kmax:
            kmin = kmax
        K = int(rng.randint(kmin, kmax + 1))
        sel = top[rng.choice(T, K, replace=False)]
        g = grad_from_indices(sel, h, n)
        g -= g.mean()
        nm = float(np.linalg.norm(g))
        if nm < 1e-12:
            break
        d = -g / nm
        aa, f = dir_line_search(h, d, n, bc, gs)
        done = False
        if aa != 0.0:
            for sc in (1.0, 0.5, 0.25, 0.1):
                hh = project_feasible(h + aa * sc * d, n)
                ff = c5(hh, n)
                if ff < bc:
                    h = hh
                    bc = ff
                    fails = 0
                    done = True
                    break
        if not done:
            fails += 1
            if fails >= 8:
                break
    h = project_feasible(h, n)
    bc = c5(h, n)
    return h, bc

def resample(h, n):
    if len(h) == n:
        return h.copy()
    return np.interp(np.linspace(0.0, 2.0, n), np.linspace(0.0, 2.0, len(h)), h)

def perturb_h(base, n, rng, kind):
    x = np.linspace(0.0, 2.0, n)
    if kind <= 0:
        h = base + rng.normal(0.0, 0.04, n)
    elif kind == 1:
        h = base + 0.03 * np.sin(np.pi * rng.uniform(0.5, 3.0) * x + rng.uniform(0.0, 6.283185307179586))
    elif kind == 2:
        h = base.copy()
        for _ in range(5):
            a = int(rng.randint(0, max(1, n - 1)))
            b = int(rng.randint(a + 1, n))
            h[a:min(b, n)] += rng.uniform(-0.15, 0.15)
    elif kind == 3:
        h = (base > rng.uniform(0.3, 0.7)).astype(float) + rng.normal(0.0, 0.005, n)
    elif kind == 4:
        h = 1.0 - base[::-1]
    else:
        theta = 0.5 + rng.uniform(-0.08, 0.08)
        h = ((np.arange(n) * theta) % 1.0) < 0.5
        h = h.astype(float) + rng.normal(0.0, 0.01, n)
    return project_feasible(h, n)

def optimize_candidate(h, n, stop, rng, quick=False):
    h = project_feasible(h, n)
    bh = h.copy()
    bc = c5(h, n)
    rem = stop - time.time()
    if rem > 8.0:
        gs = 12 if (quick or n > 500) else 16
        h, c = cd_polish(h, n, time.time() + rem * 0.55, rng, gs=gs)
        if c < bc:
            bh, bc = h, c
        rem = stop - time.time()
        if rem > 8.0:
            gs = 6 if (quick or n > 500) else 9
            h, c = active_dir_search(h, n, time.time() + rem * 0.35, rng, gs=gs)
            if c < bc:
                bh, bc = h, c
    rem = stop - time.time()
    if rem > 3.0:
        gs = 14 if (quick or n > 500) else 18
        h, c = cd_polish(h, n, stop - 1, rng, gs=gs)
        if c < bc:
            bh, bc = h, c
        rem = stop - time.time()
        if rem > 4.0:
            h, c = active_dir_search(h, n, stop - 1, rng, gs=5)
            if c < bc:
                bh, bc = h, c
    bh = project_feasible(bh, n)
    bc = c5(bh, n)
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = float(budget_s)
    deadline = t0 + max(10.0, total - 5.0)
    avail = max(20.0, total - 15.0)
    reserve = max(10.0, 0.10 * avail)
    if reserve > 0.35 * (deadline - t0):
        reserve = 0.35 * (deadline - t0)
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 300
        h0 = np.full(n0, 0.5)

    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_c = c5(h0, n0)

    stop = min(deadline, t0 + 0.34 * avail)
    h, c = optimize_candidate(h0, n0, stop, rng, quick=False)
    if c < best_c:
        best_h, best_c = h, c

    low = []
    for tt in (100, 200, 350, 550):
        if tt < n0:
            low.append(tt)
    if len(low) > 2 or n0 < 550:
        low = low[:3]
    hh = n0 // 2
    if 20 <= hh < n0 and hh not in low:
        low.append(hh)
    low = sorted(set(low))[:3]

    for nn in low:
        if time.time() > deadline - reserve:
            break
        stop = min(deadline - reserve, time.time() + 0.07 * avail)
        hs = project_feasible(resample(best_h, nn), nn)
        hs = perturb_h(hs, nn, rng, 0 if rng.rand() < 0.7 else 3)
        h, c = optimize_candidate(hs, nn, stop, rng, quick=True)
        if c < best_c:
            best_h, best_c = h, c

    while time.time() < deadline - reserve:
        if len(best_h) > 400 and rng.rand() < 0.25:
            nn = int(min(300, max(100, len(best_h) // 2)))
            base = resample(best_h, nn)
        else:
            nn = len(best_h)
            base = best_h.copy()
        kind = int(rng.randint(0, 6))
        hs = perturb_h(base, nn, rng, kind)
        stop = min(deadline - reserve, time.time() + max(22.0, 0.035 * avail))
        h, c = optimize_candidate(hs, nn, stop, rng, quick=True)
        if c < best_c:
            best_h, best_c = h, c

    if n0 > len(best_h) and time.time() < deadline - reserve:
        stop = min(deadline - reserve, time.time() + 0.05 * avail)
        h, c = optimize_candidate(resample(best_h, n0), n0, stop, rng, quick=True)
        if c < best_c:
            best_h, best_c = h, c

    h, c = optimize_candidate(best_h, len(best_h), deadline, rng, quick=False)
    if c < best_c:
        best_h, best_c = h, c

    best_h = project_feasible(best_h, len(best_h))
    best_c = c5(best_h, len(best_h))
    return best_h, float(best_c), int(len(best_h))
