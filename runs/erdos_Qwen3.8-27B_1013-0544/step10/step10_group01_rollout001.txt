def project_feasible(x,n):
    h=np.asarray(x,float)
    target=n/2.0
    if abs(h.sum()-target)<1e-12: h=np.clip(h,0,1); enforce; return
    lo,hi = -2.0,2.0
    for _ in range(80):
       mid=(lo+hi)/2
       y=np.clip(h+mid,0,1)
       if y.sum() < target: lo=mid else hi=mid
    y=np.clip(h+(lo+hi)/2,0,1)
    # exact sum adjustment
    residual = target - y.sum()
    if abs(residual)>1e-12:
       # adjust interior values
       mask = (y>1e-12)&(y<1-1e-12)
       cnt=mask.sum()
       if cnt:
          y[mask]+=residual/cnt
          y=np.clip(y,0,1)

       for _ in range(10):

          if abs(resid)<1e-13: break

          else: room=y; room[y<=1e-12]=0; tot=room.sum()

def coord_descent(h,n,max_passes,deadline,samples=21,gs_iters=24,rng=None,lam=0.0):
    h=project_feasible(np.array(h,dtype=float,copy=True),n)
    best_h=h.copy(); best_c=compute_c5(h,n)
    c=np.correlate(h,1.0-h,'full')
    for it in range(max_passes):
        if time.time()>deadline: break
        if rng is not None:
            order = rng.permutation(n)
        else:
            order = np.arange(n)
        updated=False
        for pos in range(n):
            if time.time()>deadline: break
            p=int(order[pos])
            h0=h[p]
            if h0 <= 1e-10 or h0 >= 1.0-1e-10: continue
            lo = p if p < n-1-p else n-1-p
            hi = p+n-1 if p+n-1 > 2*n-2-p else 2*n-2-p
            L=hi-lo+1
            a=np.zeros(L)
            r1 = p if p>lo else lo
            r2 = p+n-1 if p+n-1<hi else hi
            if r1 <= r2:
               s = p+n-1-r1; e = p+n-1-r2
               if s == e:
                  a[r1-lo] += 1.0 - h[s]
               else:
                  a[r1-lo:r2+1-lo] += 1.0 - h[s:e:-1]
            r3 = n-1-p if n-1-p>lo else lo
            r4 = 2*n-2-p if 2*n-2-p<hi else hi
            if r3 <= r4:
               s = p+r3-(n-1); e = p+r4-(n-1)
               a[r3-lo:r4+1-lo] -= h[s:e+1]
            c_aff = c[lo:hi+1]
            has_diag = lo <= n-1 and n-1 <= hi
            diag = n-1-lo if has_diag else 0
            c_diag = c[diag] if has_diag else 0.0
            d_opt,f_opt,f_curr = _line_min(c_aff,a,h0,has_diag,diag,c_diag,lam,samples,gs_iters)
            if f_opt < f_curr - 1e-13:
               h[p]=h0+d_opt

               tmp += c_aff

            if lam>0.5: lam=0.5

def coord_descent(h, n, max_passes, deadline, samples=19, gs_iters=20, rng=None, coeff_mode=0):

def _smooth_fg(h, n, alpha, gmode):
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode="full")
    m = float(c.max())
    e = np.exp(alpha * (c - m))
    s = float(e.sum())
    if s <= 1e-300:
        p = np.zeros_like(e)
        p[np.argmax(c)] = 1.0
    else:
        p = e / s
    F = (m + math.log(s) / alpha) * dx
    if gmode == 0:
        t1 = np.convolve(1.0 - h, p, mode="full")[n - 1:2 * n - 1]
        t2 = np.convolve(h, p[::-1], mode="full")[n - 1:2 * n - 1]
    else:
        t1 = np.convolve(1.0 - h, p[::-1], mode="full")[n - 1:2 * n - 1]
        t2 = np.convolve(h, p, mode="full")[n - 1:2 * n - 1]
    g = (t1 - t2) * dx
    return F, g

def _proj_fun_grad(z, n, alpha, gmode):
    h = project_feasible(z, n)
    F, g = _smooth_fg(h, n, alpha, gmode)
    active = (h > 1e-9) & (h < 1.0 - 1e-9)
    if not np.any(active):
        gz = np.zeros(n, dtype=float)
    else:
        ga = g[active]
        gz = np.zeros(n, dtype=float)
        gz[active] = ga - ga.mean()
    return F, gz


def lbfgs_project(h, n, alpha, maxfun, maxiter, gmode):
    try:
        z = np.array(h, dtype=float, copy=True)
        bounds = [(-1.0, 2.0)] * n
        opts = {
            "maxfun": int(maxfun),
            "maxiter": int(maxiter),
            "ftol": 1e-12,
            "gtol": 1e-8,
            "disp": False,
        }
        res = minimize(
            _proj_fun_grad,
            z,
            args=(n, alpha, gmode),
            method="L-BFGS-B",
            jac=True,
            bounds=bounds,
            options=opts,
        )
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return None

    stop = min(deadline - 350.0, now + limit)
    if time.time() < stop:
        h1, c1 = coord_descent(
            best_h, n_best, 1000, stop, samples=19, gs_iters=20, rng=rng, coeff_mode=coeff_mode
        )
        if c1 < best_c:
            best_h, best_c = h1, c1
            n_best = len(best_h)

    if use_grad:
        for alpha in (30.0, 160.0, 900.0):
            if time.time() > deadline - 280.0:
                break
            if n_best <= 300:
                mf, mi = 220, 90
            elif n_best <= 550:
                mf, mi = 120, 60
            else:
                mf, mi = 70, 35
            hs = lbfgs_project(best_h, n_best, alpha, mf, mi, grad_mode)
            if hs is None:
                continue
            cs = compute_c5(hs, n_best)
            if cs < best_c + 1e-12:
                h2, c2 = coord_descent(
                    hs, n_best, 60, min(deadline - 240.0, time.time() + 35.0),
                    samples=15, gs_iters=15, rng=rng, coeff_mode=coeff_mode
                )
                if c2 < best_c:
                    best_h, best_c = h2, c2
                    n_best = len(best_h)

    if time.time() < deadline - 200.0:
        for frac in (2, 3):
            nt = int(max(140, n_best // frac))
            if nt >= n_best or time.time() > deadline - 150.0:
                continue
            hs = project_feasible(interp_h(best_h, nt), nt)
            rem = min(170.0 if nt <= 280 else 110.0, max(40.0, deadline - 150.0 - time.time()))
            hs, cs = coord_descent(
                hs, nt, 1000, time.time() + rem,
                samples=21, gs_iters=23, rng=rng, coeff_mode=coeff_mode
            )
            if cs < best_c:
                best_h, best_c = hs, cs
                n_best = nt

    if time.time() < deadline - 80.0:
        n_high = max(n0, n_best)
        if n_high != n_best:
            hu = project_feasible(interp_h(best_h, n_high), n_high)
        else:
            hu = best_h.copy()
        rem = min(200.0 if n_high > 600 else 260.0, max(30.0, deadline - 40.0 - time.time()))
        hu, cu = coord_descent(
            hu, n_high, 1000, time.time() + rem,
            samples=23, gs_iters=27, rng=rng, coeff_mode=coeff_mode
        )
        if cu < best_c:
            best_h, best_c = hu, cu
            n_best = len(best_h)

    if use_grad and time.time() < deadline - 25.0:
        n_cur = len(best_h)
        if n_cur <= 400:
            mf, mi = 90, 40
        else:
            mf, mi = 45, 25
        hs = lbfgs_project(best_h, n_cur, 250.0, mf, mi, grad_mode)
        if hs is not None:
            cs = compute_c5(hs, n_cur)
            if cs < best_c:
                best_h, best_c = hs, cs
                n_best = len(best_h)

    if time.time() < deadline - 10.0:
        n_cur = len(best_h)
        hf, cf = coord_descent(
            best_h, n_cur, 50, deadline - 4.0,
            samples=25, gs_iters=29, rng=rng, coeff_mode=coeff_mode
        )
        if cf < best_c:
            best_h, best_c = hf, cf
            n_best = len(best_h)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
