import time
import numpy as np

# ---- basic ----
def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * dx)

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    for _ in range(600):
        d = target - h.sum()
        if abs(d) <= 1e-13:
            break
        if d > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h = np.clip(h + d / s * room, 0.0, 1.0)
        else:
            s = h.sum()
            if s < 1e-14: break
            h = np.clip(h + d / s * h, 0.0, 1.0)
    # exact-ish final adjust
    d = target - h.sum()
    if abs(d) > 0 and abs(d) < 0.5:
        eps = d / n
        if np.all(h + eps >= 0.0) and np.all(h + eps <= 1.0):
            h = h + eps
    return h

import time
import warnings
import numpy as np
from scipy.optimize import minimize

warnings.simplefilter('ignore')


def compute_c5(h, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * (2.0 / n))


def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = 0.5 * n
    for _ in range(800):
        d = target - h.sum()
        if abs(d) <= 1e-13:
            break
        if d > 0:
            room = 1.0 - h
            s = room.sum()
            if s <= 1e-15:
                break
            h = np.clip(h + d / s * room, 0.0, 1.0)
        else:
            s = h.sum()
            if s <= 1e-15:
                break
            h = np.clip(h + d / s * h, 0.0, 1.0)
    return h


def make_sum_exact(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    t = 0.5 * n
    order = np.argsort(np.abs(h - 0.5))
    for i in order[:12]:
        h[i] += t - h.sum()
        if h[i] < 0.0:
            h[i] = 0.0
        elif h[i] > 1.0:
            h[i] = 1.0
    return h


def solve_coord(c, h, p, n, nbisect):
    h0 = h[p]
    if h0 < 1e-14 or h0 > 1.0 - 1e-14:
        return h0
    lo = max(0, n - 1 - p)
    hi = min(2 * n - 2, n - 1 + p)
    ms = np.arange(lo, hi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(ms.size)
    if np.any(vf):
        a[vf] += 1.0 - h[jf[vf]]
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    cb = c[ms].copy()
    dp = np.where(ms == (n - 1))[0]
    has = dp.size > 0
    c_diag = (cb[int(dp[0])] + (1.0 - cb[int(dp[0])])) * 0.5 if has else 0.0
    nd = np.ones(ms.size, dtype=bool)
    a_d = 0.0
    if has:
        a_d = a[dp]
        nd[dp] = False
        a[dp] = 0.0
    an, cn = a[nd], cb[nd]
    unaff = 0.0
    if lo > 0:
        unaff = max(unaff, float(c[:lo].max()))
    if hi < 2 * n - 2:
        unaff = max(unaff, float(c[hi + 1:].max()))
    Llo, Lhi = unaff, float(c.max())
    for _ in range(nbisect):
        if Lhi - Llo < 1e-12:
            break
        T = 0.5 * (Llo + Lhi)
        neg = an < -1e-12
        pos = an > 1e-12
        xl, xh = 0.0, 1.0
        if neg.any():
            xl = max(xl, float(np.max((T - cn[neg]) / an[neg])))
        if pos.any():
            xh = min(xh, float(np.min((T - cn[pos]) / an[pos])))
        feas = False
        if xl <= xh + 1e-12:
            if not has:
                feas = True
            else:
                Cq = T - c_diag + h0 - h0 * h0
                if Cq < 0.0:
                    pass
                elif Cq >= 0.25:
                    feas = True
                else:
                    r1 = 0.5 * (1.0 - np.sqrt(1.0 - 4.0 * Cq))
                    r2 = 0.5 * (1.0 + np.sqrt(1.0 - 4.0 * Cq))
                    if xl >= r1 and xh <= r2:
                        pass
                    else:
                        feas = True
        if feas:
            Lhi = T
        else:
            Llo = T
    T = 0.5 * (Llo + Lhi)
    xl, xh = 0.0, 1.0
    if an.size:
        neg = an < -1e-12
        pos = an > 1e-12
        if neg.any():
            xl = max(xl, float(np.max((T - cn[neg]) / an[neg])))
        if pos.any():
            xh = min(xh, float(np.min((T - cn[pos]) / an[pos])))
    if has and not np.isinf(xl / (xh + 1e-30)):
        Cq = T - c_diag + h0 - h0 * h0
        if Cq >= 0.0 and Cq < 0.25:
            r1 = 0.5 * (1.0 - np.sqrt(1.0 - 4.0 * Cq))
            r2 = 0.5 * (1.0 + np.sqrt(1.0 - 4.0 * Cq))
            if xl < r1:
                xh = min(xh, r1)
            elif xh > r2:
                xl = max(xl, r2)
            else:
                if h0 - r1 <= r2 - h0:
                    xh = min(xh, r1 + 1e-9)
                else:
                    xl = max(xl, r2 - 1e-9)
    xl = max(0.0, xl)
    xh = min(1.0, xh)
    if xl > xh + 1e-10:
        return h0
    return float(np.clip(0.5 * (xl + xh), 0.0, 1.0))


def coord_descent(h, n, max_passes, deadline, rng, random_order, nbisect):
    h = project_feasible(h, n)
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = corr_full(h)
        order = rng.permutation(n) if (random_order and rng is not None) else np.arange(n)
        moves = 0
        for p in order:
            h0 = h[p]
            if h0 <= 1e-13 or h0 >= 1.0 - 1e-13:
                continue
            xn = solve_coord(c, h, p, n, nbisect)
            if abs(xn - h0) > 1e-12:
                h[p] = xn
                moves += 1
        if moves == 0:
            h = project_feasible(h, n)
            break
        if time.time() > deadline:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)


def sum_con(x, n):
    return x.sum() - 0.5 * n


def sum_con_jac(x, n):
    return np.ones(n)


def lse_f(x, n, alpha):
    c = corr_full(x)
    m = c.max()
    e = np.exp(alpha * (c - m))
    return (m + np.log(e.sum()) / alpha) * (2.0 / n)


def lse_g(x, n, alpha):
    c = corr_full(x)
    m = c.max()
    e = np.exp(alpha * (c - m))
    w = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        d = k - (n - 1)
        if d >= 0:
            g[d:] += wk * (1.0 - x[:n - d])
            g[:n - d] -= wk * x[d:]
        else:
            e2 = -d
            g[:n - e2] += wk * (1.0 - x[e2:])
            g[e2:] -= wk * x[:n - e2]
    g *= 2.0 / n
    return g


def slsqp_lse(h, n, alpha, maxiter, deadline):
    cur = project_feasible(h, n)
    bc = compute_c5(cur, n)
    if time.time() > deadline - 6:
        return cur, bc
    con = {'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(lse_f, cur, args=(n, alpha), jac=lse_g, method='SLSQP',
                       bounds=bnds, constraints=con,
                       options={'maxiter': int(maxiter), 'ftol': 1e-13})
        cand = project_feasible(res.x, n)
        cc = compute_c5(cand, n)
        if cc < bc:
            return cand, cc
    except Exception:
        pass
    return cur, bc


def grad_ck(x, n, k):
    g = np.zeros(n)
    d = k - (n - 1)
    if d >= 0:
        g[d:] += 1.0 - x[:n - d]
        g[:n - d] -= x[d:]
    else:
        e = -d
        g[:n - e] += 1.0 - x[e:]
        g[e:] -= x[:n - e]
    return g


def push_fun(x, n, idx, T, beta):
    c = corr_full(x)
    v = (c[idx] - T) * (2.0 / n)
    return float(np.logaddexp(0.0, beta * v).sum())


def push_jac(x, n, idx, T, beta):
    c = corr_full(x)
    t = beta * (c[idx] - T) * (2.0 / n)
    w = 1.0 / (1.0 + np.exp(-t))
    g = np.zeros(n)
    for j in range(idx.size):
        if w[j] > 1e-12:
            g += w[j] * grad_ck(x, n, int(idx[j]))
    g *= beta * (2.0 / n)
    return g


def active_push(h, n, deadline):
    cur = project_feasible(h, n)
    bc = compute_c5(cur, n)
    bnds = [(0.0, 1.0)] * n
    con = {'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}
    for _ in range(9):
        if time.time() > deadline - 45:
            break
        c = corr_full(cur)
        M = c.max()
        idx = np.sort(np.argsort(c)[-12:])
        for T in (M - 0.9, M - 0.5, M - 0.25):
            try:
                res = minimize(push_fun, cur, args=(n, idx, T, 30.0), jac=push_jac,
                               method='SLSQP', bounds=bnds, constraints=con,
                               options={'maxiter': 200, 'ftol': 1e-13})
                cand = project_feasible(res.x, n)
                cc = compute_c5(cand, n)
                if cc < bc - 1e-12:
                    cur, bc = cand, cc
                    break
            except Exception:
                break
    return cur, bc


def refine_top(h, n, topk, niter, deadline):
    cur = project_feasible(h, n)
    bc = compute_c5(cur, n)
    for _ in range(niter):
        if time.time() > deadline - 25:
            break
        c = corr_full(cur)
        idx = np.argsort(c)[-topk:]
        g = np.zeros(n)
        for k in idx:
            g += grad_ck(cur, n, int(k))
        g /= topk
        ng = float(np.linalg.norm(g))
        if ng < 1e-9:
            break
        bt = bc
        best = None
        for f in (0.05, 0.15, 0.4, 1.0):
            eta = f * 0.35 / ng
            cand = project_feasible(cur - eta * g, n)
            cc = compute_c5(cand, n)
            if cc < bt:
                bt, best = cc, cand
        if best is None:
            break
        cur, bc = best, bt
    return cur, bc


def dual_of(h, n):
    return project_feasible(1.0 - h[::-1], n)


def combo_h(n, p, rng):
    base = np.zeros(p)
    base[:p // 2] = 1.0
    if rng is not None and p >= 4:
        base = base[rng.permutation(p)]
    v = np.tile(base, n // p)
    v = np.clip(v, 0.0, 1.0)
    if v.size < n:
        v = np.concatenate([v, np.zeros(n - v.size)])
    return project_feasible(v[:n], n)


def sine_h(n, m, phi, amp):
    x = np.linspace(0.0, 2.0, n)
    v = 0.5 + amp * np.sin(m * np.pi * x + phi)
    return project_feasible(np.clip(v, 0.0, 1.0), n)


def jitter(h, n, rng, s):
    w = rng.standard_normal(n)
    w = np.convolve(w, np.array([1.0, 2.0, 1.0]) / 4.0, mode='same')
    w /= max(1e-9, float(np.abs(w).max()))
    return project_feasible(np.clip(h + s * w, 0.0, 1.0), n)


def resample_step(h, n_new):
    n0 = h.size
    j = np.minimum(n0 - 1, (np.arange(n_new) * n0 / n_new).astype(int))
    return h[j]


def polish_start(h, n, end_t, rng, alpha=900, mit=140, cd_passes=7):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    h2, c2 = slsqp_lse(best_h, n, alpha, mit, end_t)
    if c2 < best_c:
        best_h, best_c = h2, c2
    h2, c2 = coord_descent(best_h, n, cd_passes, end_t, rng, True, 28)
    if c2 < best_c:
        best_h, best_c = h2, c2
    return best_h, best_c

        if cc < best_c:

            hh, cc = active_push(best_h, ntr, deadline)
            if cc < best_c:
                best_h, best_c = hh, cc
            hh, cc = coord_descent(best_h, ntr, 10, deadline, rng, True, 30)
            if cc < best_c:
                best_h, best_c = hh, cc

    # Phase 5: final polish
    nf = best_h.size
    hh, cc = slsqp_lse(best_h, nf, 30000, 220, deadline)
    if cc < best_c:
        best_h, best_c = hh, cc
    nlp = int(max(8.0, min(60.0, (deadline - time.time() - 8.0) / 1.0)))
    hh, cc = coord_descent(best_h, nf, nlp, deadline, rng, False, 42)
    if cc < best_c:
        best_h, best_c = hh, cc
    hh, cc = refine_top(best_h, nf, 12, 70, deadline)
    if cc < best_c:
        best_h, best_c = hh, cc

    best_h = make_sum_exact(project_feasible(best_h, nf), nf)
    best_c = compute_c5(best_h, nf)
    if time.time() > deadline:
        pass
    return best_h, best_c, nf
