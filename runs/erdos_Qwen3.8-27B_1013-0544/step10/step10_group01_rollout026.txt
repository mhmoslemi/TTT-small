def subgradient_step(h, n, step_size, deadline):
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = c.max()
    active = c > cmax - step_size * 0.1
    g = np.zeros(n)
    for k in np.where(active)[0]:
        shift = int(k) - (n - 1)
        # gradient of c[k] w.r.t. h
        # c[k] = sum_i h[i]*(1-h[i+shift])
        # dc[k]/dh[p] = (1-h[p+shift]) - h[p-shift] (with boundary handling)
        for i in range(n):
            ip = i + shift
            im = i - shift
            grad = 0.0
            if 0 <= ip < n:
                grad += (1.0 - h[ip])
            if 0 <= im < n:
                grad -= h[im]
            g[i] += grad
    g /= active.sum()
    h_new = project_feasible(h - step_size * g, n)
    return h_new

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)
