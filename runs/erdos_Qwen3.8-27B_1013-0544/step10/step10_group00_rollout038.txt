**Key improvement ideas:**

1. **Faster CD with early termination**: If a coordinate is at 0 or 1, skip it. If the best possible improvement from a coordinate is less than the current gap to the max, skip it.

2. **Multi-start CD with different orderings**: Instead of always going left to right, try random orderings. This can escape local optima.

3. **Block moves**: Instead of changing one coordinate, change a small block (2-4 coordinates) simultaneously. Use a 2D/3D/4D search.

4. **Nelder-Mead on the whole vector (reduced)**: Use Nelder-Mead on a subset of the most "active" coordinates (those with h values between 0.1 and 0.9).

5. **Different n**: The solution quality might depend on n. Try n values that are factors of common numbers for better symmetry.

```python
import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d):
    vals = c_aff + d * a
    if has_diag:
        vals[diag_idx] = c_aff[diag_idx] + d * (1.0 - h0 - d)
    result = float(np.max(vals))
    if unaff_max > result:
        result = unaff_max
    return result

def coord_descent(h, n, max_passes, deadline, gs_iters=50, order=None):
    h = h.copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        if order is None:
            indices = range(n)
        else:
            indices = order
        for p in indices:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-13 or h0 >= 1.0 - 1e-13:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(valid_f):
                a[valid_f] += (1.0 - h[jf[valid_f]])
            if np.any(valid_b):
                a[valid_b] -= h[jb[valid_b]]
            diag_pos = np.where(ms == (n - 1))[0]
            has_diag = len(diag_pos) > 0
            diag_idx = int(diag_pos[0]) if has_diag else -1
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            d_lo = -h0
            d_hi = 1.0 - h0
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
            f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-15:
                    break
                if f1 < f2:
                    d_hi = d2
                    d2 = d1
                    f2 = f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
                else:
                    d_lo = d1
                    d1 = d2
                    f1 = f2
                    d2 = d_lo + gr * (d_hi - d_lo)
                    f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            d_opt = (d_lo + d_hi) / 2.0
            f_opt = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d_opt)
            f_curr = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, 0.0)
            if f_opt < f_curr - 1e-14:
                h[p] = h0 + d_opt
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
```
```python
    return h, compute_c5(h, n)
```

Wait, this is inefficient because coord_descent recomputes the correlation each pass. Let me restructure.

Actually, let me think about this differently. The key bottleneck for improvement is:
1. The CD converges to a local optimum
2. We need to escape that local optimum

The most effective way to escape is through **structured perturbations** that are large enough to jump out of the local basin but small enough to land in a nearby region.

Let me think about what kind of perturbation would be effective:
- **Phase shift**: Shift the entire h by a small amount (circular shift). This preserves all properties except boundaries.
- **Frequency modulation**: Multiply the "transition regions" by a different amount.
- **Anti-symmetric perturbation**: Add a perturbation that is anti-symmetric about the center. This preserves the sum constraint.

The anti-symmetric perturbation is interesting! If we add δ(x) to h(x) and -δ(x) to h(2-x), the sum is preserved. This is a natural symmetry-breaking move.

Let me implement this:

```python
def antisymmetric_perturb(h, n, rng, scale):
    # Add anti-symmetric perturbation: h'[x] = h[x] + scale*noise[x], h'[2-x] = h[2-x] - scale*noise[x]
    h = h.copy()
    noise = rng.normal(0, 1, n)
    # Anti-symmetric about center
    half = n // 2
    for i in range(half):
        j = n - 1 - i
        delta = scale * noise[i]
        h[i] += delta
        h[j] -= delta
    return project_feasible(h, n)
```

This is a good move because:
1. It preserves the sum exactly (before projection)
2. It's a structured perturbation that explores a specific direction
3. The scale can be annealed over time

Now let me also think about a **circular shift** move:
```python
def circular_shift(h, n, shift):
    return np.roll(h, shift)
```
This preserves the sum and all correlation values (since correlation is shift-invariant for periodic functions). But our function is NOT periodic (it's defined on [0,2] with the correlation computed as a linear correlation, not circular). So a circular shift would change the correlation values at the boundaries.

Actually wait, let me re-examine. The correlation `np.correlate(h, 1-h, mode='full')` computes:
c[m] = sum_i h[i] * (1-h[i + m - (n-1)])
for valid i (where both indices are in [0, n-1]).

This is NOT the same as circular correlation. So a circular shift would change the values.

def block_perturb(h, n, rng, scale):

    # Pick a random block and add noise

    start = rng.randint(0, n - block_len)

def swap_perturb(h, n, rng, num_swaps=10):
    h = h.copy()
    for _ in range(num_swaps):
        i, j = rng.choice(n, 2, replace=False)
        h[i], h[j] = h[j], h[i]
    return h  # sum preserved

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * dx

def _sum_eq(x, n):
    return x.sum() - n / 2.0

def _sum_eq_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alphas, maxiter_each=200):
    from scipy.optimize import minimize
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac, 'args': (n,)}]
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-14})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c
