def lp_tangent_step(h, n, c, K=120, t_hi=None):
    nvar=n+1
    T=float(c.max())
    L=len(c)
    idx = np.argpartition(c, -K)[-K:] (if K<L else all)
    # maybe include random? we can mix: if K>0, top K
    rows=[]
    rhs=[]
    for r in idx:
       q = r-(n-1)
       g=np.zeros(n)
       if q>=0:
          N=n-q
          g[:N] += 1.0 - h[q:q+N]
          if q>0: g[q:] += -h[:N]
          else: g[:] += -h[:]  # makes 1-2h
       else:
          a=-q
          g[a:] += 1.0 - h[:n-a]
          g[:n-a] += -h[a:]
       const = c[r] - g @ h
       rows.append(np.concatenate((g, [-1.0])))
       rhs.append(g @ h - c[r])
    A=np.array(rows)
    b=np.array(rhs)
    cobj=np.zeros(nvar); cobj[-1]=1
    bounds=[(0,1)]*n + [(0, t_hi if t_hi is not None else T)]
    Aeq=np.ones((1,nvar)); Aeq[0,-1]=0; beq=target
    res=linprog(cobj, A_ub=A,b_ub=b,A_eq=Aeq,b_eq=target,bounds=bounds,method='highs')
    if not res.success: return None
    h_res=res.x[:n]

    h = np.clip(h,0.0,1.0)

    if abs(s-target) <= 1e-10:

    lo,hi = -1.0,1.0

    for _ in range(50):

       if s < target: lo=mid
       else: hi=mid
    tau=(lo+hi)/2
    h=np.clip(h+tau,0,1)
    diff=target-h.sum()
    # fix diff with interior points
    idx=np.where((h>1e-12)&(h<1.0-1e-12))[0]
    if idx.size==0:

       if abs(diff) <= 1e-10: pass
       else:
          # can't? just scale?

def lp_spare(h,n,c,K,trust=0.08):
    L=2*n-1
    T=float(c.max())
    if K<=0: return None
    if K>=L: idx=np.arange(L)
    else: idx=np.argpartition(c,L-K)[L-K:]
    K=len(idx)
    A=np.zeros((K,n+1),float)
    b=np.empty(K,float)
    for i,r in enumerate(idx):
       q=int(r-(n-1))
       if q>=0:
          N=n-q
          A[i,:N] += 1.0 - h[q:q+N]
          if q>0: A[i,q:] += -h[:N]
          else: A[i,:] += -h

          A[i,:n-a] += -h[a:]

def _cd_fval(c_aff,a,diag_idx,has_diag,unaff_max,h0,d):
    vals = c_aff + d*a
    if has_diag:
       vals[diag_idx] = c_aff[diag_idx] + d*(1.0 - h0 - d)

def coord_descent(h,n,max_passes,deadline,gs_iters=18,max_step=0.05,rng=None):
    h=project_feasible(h,n)
    target=0.5*float(n)
    c=get_c(h,n)
    best_c5=float(c.max())*2.0/float(n)
    L=2*n-1
    gr=(np.sqrt(5.0)-1.0)/2.0
    for _ in range(max_passes):
       if time.time() > deadline: break
       h_pass=h.copy()
       c_work=c.copy()
       cmax_work=float(c_work.max())
       pass_sum=h_pass.sum()
       upd=0
       improved=False
       order=np.arange(n)
       if rng is not None: rng.shuffle(order)
       for p in order:
          if time.time() > deadline: break
          h0=float(h_pass[p])
          if h0 <= 1e-12 or h0 >= 1.0-1e-12: continue
          radius = max(p, n-1-p)
          m_lo = (n-1)-radius
          m_hi = (n-1)+radius
          ms = np.arange(m_lo, m_hi+1)
          a = np.zeros(ms.size, dtype=float)
          q = ms - (n-1)
          idx_f = p + q
          vf = (idx_f >= 0) & (idx_f < n)
          if np.any(vf): a[vf] += 1.0 - h_pass[idx_f[vf]]
          idx_b = p - q
          vb = (idx_b >= 0) & (idx_b < n)
          if np.any(vb): a[vb] -= h_pass[idx_b[vb]]
          c_aff = c_work[ms]
          diag = np.nonzero(ms == (n-1))[0]
          has_diag = diag.size>0
          diag_idx = int(diag[0]) if has_diag else -1
          unaff=0.0
          if m_lo>0: unaff=max(unaff,float(np.max(c_work[:m_lo])))
          if m_hi<L-1: unaff=max(unaff,float(np.max(c_work[m_hi+1:])))
          hi=min(1.0-h0, max_step)
          lo=max(-h0, -max_step)
          if pass_sum > target + 1.0: hi=min(hi,1e-12)
          if pass_sum < target - 1.0: lo=max(lo,-1e-12)
          if hi - lo < 1e-14: continue
          d1=hi-gr*(hi-lo); d2=lo+gr*(hi-lo)
          f1=_cd_fval(c_aff,a,diag_idx,has_diag,unaff,h0,d1)
          f2=_cd_fval(c_aff,a,diag_idx,has_diag,unaff,h0,d2)
          for _it in range(gs_iters):
             if hi-lo < 1e-14: break
             if f1 < f2:
                hi=d2; d2=d1; f2=f1
                d1=hi-gr*(hi-lo)
                f1=_cd_fval(c_aff,a,diag_idx,has_diag,unaff,h0,d1)
             else:
                lo=d1; d1=d2; f1=f2
                d2=lo+gr*(hi-lo)
                f2=_cd_fval(c_aff,a,diag_idx,has_diag,unaff,h0,d2)
          d_opt=(lo+hi)*0.5
          f_opt=_cd_fval(c_aff,a,diag_idx,has_diag,unaff,h0,d_opt)
          f_cur=_cd_fval(c_aff,a,diag_idx,has_diag,unaff,h0,0.0)
          if f_opt < f_cur - 1e-13:
             h_pass[p]=h0+d_opt
             new_vals = c_aff + d_opt*a
             if has_diag: new_vals[diag_idx] = c_aff[diag_idx] + d_opt*(1.0 - h0 - d_opt)
             c_work[ms] = new_vals
             improved=True
             pass_sum += d_opt
             upd += 1
             if upd >= 40 or abs(pass_sum-target)>3.0:
                h_pass = project_feasible(h_pass,n)
                c_work = get_c(h_pass,n)
                pass_sum = h_pass.sum()
                upd = 0
       if improved:
          h_pass = project_feasible(h_pass,n)
          new_c5 = float(np.max(get_c(h_pass,n))) * 2.0 / float(n)
          if new_c5 < best_c5 - 1e-14:
             h=h_pass; best_c5=new_c5; c=get_c(h,n)
          else:
             break
       else:
          break
    final_c5 = float(np.max(get_c(h,n))) * 2.0 / float(n)
    if final_c5 < best_c5: best_c5=final_c5
    return h,best_c5

import numpy as np
import time
from scipy.optimize import linprog


def project_feasible(h, n):
    n = int(n)
    target = 0.5 * float(n)
    h = np.asarray(h, dtype=float)
    if not np.all(np.isfinite(h)) or h.size != n:
        return np.full(n, 0.5)
    h = np.clip(h, 0.0, 1.0)
    if abs(h.sum() - target) <= 1e-10:
        return h

    lo, hi = -1.0, 1.0
    for _ in range(45):
        mid = 0.5 * (lo + hi)
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid

    h = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - h.sum()

    if abs(diff) > 1e-11:
        idx = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if idx.size:
            if diff > 0.0:
                if diff <= 1.0 - h[idx[0]]:
                    h[idx[0]] += diff
                else:
                    room = 1.0 - h[idx]
                    cap = room.sum()
                    if cap > 0:
                        h[idx] += diff * (room / cap)
            else:
                if -diff <= h[idx[0]]:
                    h[idx[0]] += diff
                else:
                    floor = h[idx]
                    cap = floor.sum()
                    if cap > 0:
                        h[idx] += diff * (floor / cap)
            h = np.clip(h, 0.0, 1.0)
        else:
            h = np.clip(h + diff / float(n), 0.0, 1.0)

        diff = target - h.sum()
        if abs(diff) > 1e-12:
            idx2 = np.where((h > 1e-14) & (h < 1.0 - 1e-14))[0]
            if idx2.size:
                h[idx2[0]] += diff
                h = np.clip(h, 0.0, 1.0)
    return h


def get_c(h, n):
    return np.correlate(h, 1.0 - h, mode="full")


def compute_c5(h, n):
    return float(np.max(get_c(h, n))) * (2.0 / float(n))


def transfer(h, m):
    m = int(m)
    h = np.asarray(h, dtype=float)
    if m == len(h):
        return project_feasible(h, m)
    x1 = np.linspace(0.0, 2.0, len(h))
    x2 = np.linspace(0.0, 2.0, m)
    return project_feasible(np.interp(x2, x1, h), m)


def lp_spare(h, n, c, K, trust=0.08):
    L = 2 * n - 1
    T = float(c.max())
    K = int(K)
    if K <= 0:
        return None
    if K >= L:
        idx = np.arange(L)
    else:
        idx = np.argpartition(c, L - K)[L - K:]
    K = len(idx)

    A = np.zeros((K, n + 1), dtype=float)
    b = np.empty(K, dtype=float)

    for i, r in enumerate(idx):
        q = int(r - (n - 1))
        if q >= 0:
            N = n - q
            A[i, :N] += 1.0 - h[q:q + N]
            if q > 0:
                A[i, q:] += -h[:N]
            else:
                A[i, :] += -h
        else:
            a = -q
            A[i, a:] += 1.0 - h[:n - a]
            A[i, :n - a] += -h[a:]
        A[i, n] = -1.0
        dot = float(np.dot(A[i, :n], h))
        b[i] = dot - float(c[r])

    obj = np.zeros(n + 1)
    obj[n] = 1.0
    Aeq = np.ones((1, n + 1))
    Aeq[0, n] = 0.0

    if trust is not None and trust > 0:
        lo = np.maximum(0.0, h - float(trust))
        hi = np.minimum(1.0, h + float(trust))
        bounds = [(float(lo[i]), float(hi[i])) for i in range(n)]
    else:
        bounds = [(0.0, 1.0) for _ in range(n)]
    bounds.append((0.0, T + 1e-9))

    try:
        res = linprog(
            obj,
            A_ub=A,
            b_ub=b,
            A_eq=Aeq,
            b_eq=np.array([0.5 * float(n)]),
            bounds=bounds,
            method="highs"
        )
    except Exception:
        return None

    if not res.success or res.x is None:
        return None
    return res.x[:n]


def try_direction(h, d, n, cur_c5, deadline):
    if float(np.max(np.abs(d))) < 1e-13:
        return None, cur_c5

    target = 0.5 * float(n)
    best = cur_c5
    best_h = None
    alphas = (1.0, 0.8, 0.6, 0.45, 0.3, 0.18, 0.1)

    for a in alphas:
        if time.time() > deadline:
            break
        h2 = h + a * d
        if (np.any(h2 < -1e-7) or np.any(h2 > 1.0 + 1e-7)
                or abs(h2.sum() - target) > 1e-5):
            h2 = project_feasible(h2, n)
        c2 = compute_c5(h2, n)
        if c2 < best - 1e-15:
            best = c2
            best_h = h2
    return best_h, best


def optimize_lp(h, n, deadline, max_steps, K, trust=0.08):
    h = project_feasible(h, n)
    c = get_c(h, n)
    cur_c5 = float(c.max()) * (2.0 / float(n))
    best_h = h.copy()
    best_c5 = cur_c5

    L = 2 * n - 1
    Kcur = min(max(1, int(K)), L)
    trust_cur = float(trust)
    no = 0

    for _ in range(int(max_steps)):
        if time.time() > deadline - (25.0 if Kcur > 450 else 5.0):
            break

        h_lp = lp_spare(h, n, c, Kcur, trust_cur)
        if h_lp is None:
            no += 1
            if no >= 3:
                break
            Kcur = min(Kcur * 2, L)
            continue

        h_try, c_try = try_direction(h, h_lp - h, n, cur_c5, deadline)
        if h_try is not None and c_try < cur_c5 - 1e-14:
            h = h_try
            cur_c5 = c_try
            c = get_c(h, n)
            best_h = h.copy()
            best_c5 = cur_c5
            no = 0
            if Kcur < L:
                Kcur = min(Kcur + 24, L)
            trust_cur = max(0.02, trust_cur * 0.9)
        else:
            no += 1
            if no == 1:
                Kcur = min(Kcur + 48, L)
            elif no == 2:
                trust_cur = min(0.35, trust_cur * 1.6)
            elif no >= 4:
                break

    final_c5 = compute_c5(h, n)
    if final_c5 < best_c5:
        return h, final_c5
    return best_h, best_c5


def _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d):
    vals = c_aff + d * a
    if has_diag:
        vals[diag_idx] = c_aff[diag_idx] + d * (1.0 - h0 - d)
    result = float(np.max(vals))
    if unaff_max > result:
        result = unaff_max
    return result


def coord_descent(h, n, max_passes, deadline, gs_iters=18, max_step=0.05, rng=None):
    h = project_feasible(h, n)
    target = 0.5 * float(n)
    c = get_c(h, n)
    best_c5 = float(c.max()) * (2.0 / float(n))
    L = 2 * n - 1
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    for _ in range(int(max_passes)):
        if time.time() > deadline:
            break

        h_pass = h.copy()
        c_work = c.copy()
        pass_sum = float(h_pass.sum())
        upd = 0
        improved = False

        order = np.arange(n)
        if rng is not None:
            rng.shuffle(order)

        for p in order:
            if time.time() > deadline:
                break

            h0 = float(h_pass[p])
            if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
                continue

            radius = max(p, n - 1 - p)
            m_lo = (n - 1) - radius
            m_hi = (n - 1) + radius
            ms = np.arange(m_lo, m_hi + 1)

            a = np.zeros(ms.size, dtype=float)
            q = ms - (n - 1)

            idx_f = p + q
            vf = (idx_f >= 0) & (idx_f < n)
            if np.any(vf):
                a[vf] += 1.0 - h_pass[idx_f[vf]]

            idx_b = p - q
            vb = (idx_b >= 0) & (idx_b < n)
            if np.any(vb):
                a[vb] -= h_pass[idx_b[vb]]

            c_aff = c_work[ms]
            diag = np.nonzero(ms == (n - 1))[0]
            has_diag = diag.size > 0
            diag_idx = int(diag[0]) if has_diag else -1

            unaff = 0.0
            if m_lo > 0:
                unaff = max(unaff, float(np.max(c_work[:m_lo])))
            if m_hi < L - 1:
                unaff = max(unaff, float(np.max(c_work[m_hi + 1:])))

            hi = min(1.0 - h0, float(max_step))
            lo = max(-h0, -float(max_step))
            if pass_sum > target + 1.0:
                hi = min(hi, 1e-12)
            if pass_sum < target - 1.0:
                lo = max(lo, -1e-12)

            if hi - lo < 1e-14:
                continue

            d1 = hi - gr * (hi - lo)
            d2 = lo + gr * (hi - lo)
            f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff, h0, d1)
            f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff, h0, d2)

            for _ in range(int(gs_iters)):
                if hi - lo < 1e-14:
                    break
                if f1 < f2:
                    hi = d2
                    d2 = d1
                    f2 = f1
                    d1 = hi - gr * (hi - lo)
                    f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff, h0, d1)
                else:
                    lo = d1
                    d1 = d2
                    f1 = f2
                    d2 = lo + gr * (hi - lo)
                    f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff, h0, d2)

            d_opt = 0.5 * (lo + hi)
            f_opt = _cd_fval(c_aff, a, diag_idx, has_diag, unaff, h0, d_opt)
            f_cur = _cd_fval(c_aff, a, diag_idx, has_diag, unaff, h0, 0.0)

            if f_opt < f_cur - 1e-13:
                h_pass[p] = h0 + d_opt
                new_vals = c_aff + d_opt * a
                if has_diag:
                    new_vals[diag_idx] = c_aff[diag_idx] + d_opt * (1.0 - h0 - d_opt)
                c_work[ms] = new_vals
                improved = True
                pass_sum += d_opt
                upd += 1

                if upd >= 40 or abs(pass_sum - target) > 3.0:
                    h_pass = project_feasible(h_pass, n)
                    c_work = get_c(h_pass, n)
                    pass_sum = float(h_pass.sum())
                    upd = 0

        if improved:
            h_pass = project_feasible(h_pass, n)
            new_c5 = compute_c5(h_pass, n)
            if new_c5 < best_c5 - 1e-14:
                h = h_pass
                best_c5 = new_c5
                c = get_c(h, n)
            else:
                break
        else:
            break

    final_c5 = compute_c5(h, n)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return project_feasible(h, n), best_c5


def make_perturb(h, n, kind, rng):
    h = project_feasible(h, n)
    if kind == 0:
        return project_feasible(h + 0.05 * rng.normal(size=n), n)

    x = np.linspace(0.0, 2.0, n)
    if kind == 1:
        f = 1.0 + 3.0 * rng.rand()
        ph = 2.0 * np.pi * rng.rand()
        return project_feasible(h + 0.07 * np.sin(np.pi * f * x + ph), n)

    if kind == 2:
        y = h.copy()
        m = rng.rand(n) < 0.25
        if np.any(m):
            y[m] += 0.3 * (rng.random(int(np.count_nonzero(m))) - 0.5)
        return project_feasible(y, n)

    y = 0.5 * np.ones(n)
    for freq in (1, 2, 3):
        amp = 0.04 + 0.08 * rng.rand()
        ph = 2.0 * np.pi * rng.rand()
        y += amp * np.sin(np.pi * freq * x + ph)
    return project_feasible(0.45 * h + 0.55 * y, n)
