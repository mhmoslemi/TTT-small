def overlap_vector(h, n):
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, mode="full") * dx


def project_simplex(v, target=None):
    v = np.asarray(v, dtype=float)
    n = v.size
    if target is None:
        target = 0.5 * n
    if target <= 0.0:
        return np.zeros(n)
    if target >= n:
        return np.ones(n)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(64):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(v - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - float(h.sum())
    for _ in range(12):
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.any():
            if diff > 0:
                idx = int(np.argmax(np.where(free, 1.0 - h, -1.0)))
                h[idx] = min(1.0, h[idx] + diff)
            else:
                idx = int(np.argmax(np.where(free, h, -1.0)))
                h[idx] = max(0.0, h[idx] + diff)
        else:
            if diff > 0:
                idx = int(np.argmax(1.0 - h))
                h[idx] = min(1.0, h[idx] + diff)
            else:
                idx = int(np.argmax(h))
                h[idx] = max(0.0, h[idx] + diff)
        diff = target - float(h.sum())
    return h

def pair_cd(h, n, deadline, strides, patience=4):
    h = np.array(h, dtype=float, copy=True)
    n = int(n)
    dx = 2.0 / float(n)
    m = 2 * n - 1
    n1 = n - 1
    v = np.arange(n, dtype=np.int64)
    A = np.zeros(m)
    B = np.zeros(m)
    C = np.zeros(m)
    g = overlap_vector(h, n)
    cur_max = float(np.max(g))
    pass_i = 0
    no_improve = 0
    while time.time() < deadline:
        stride = int(strides[pass_i % len(strides)]) % n
        if stride == 0:
            stride = 1
        improved = False
        acc = 0
        for p in range(n):
            if p and (p % 120 == 0) and time.time() > deadline:
                break
            q = (p + stride) % n
            if q == p:
                continue
            res = try_pair_update(p, q, h, g, cur_max, n, dx, n1, v, A, B, C)
            if res is not None:
                x, y, nm, idx, nv = res
                h[p] = x
                h[q] = y
                g[idx] = nv
                cur_max = nm
                improved = True
                acc += 1
                if acc > 0 and (acc % 250 == 0):
                    g = overlap_vector(h, n)
                    cur_max = float(np.max(g))
        pass_i += 1
        if not improved:
            no_improve += 1
            if no_improve >= patience:
                break
        else:
            no_improve = 0
        g = overlap_vector(h, n)
        cur_max = float(np.max(g))
    return h, g, cur_max
