import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def _sum_eq(x, n):
    return x.sum() - n / 2.0

def _sum_eq_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alphas, maxiter_each=300):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac, 'args': (n,)}]
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-15})
            if res.x is not None:
                new_h = project_feasible(res.x, n)
                new_c = compute_c5(new_h, n)
                if new_c < best_c:
                    best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d):
    vals = c_aff + d * a
    if has_diag:
        vals[diag_idx] = c_aff[diag_idx] + d * (1.0 - h0 - d)
    result = float(np.max(vals))
    if unaff_max > result:
        result = unaff_max
    return result

def coord_descent(h, n, max_passes, deadline, gs_iters=50):
    h = h.copy()
    best_c = compute_c5(h, n)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    improved_any = False
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(valid_f):
                a[valid_f] += (1.0 - h[jf[valid_f]])
            if np.any(valid_b):
                a[valid_b] -= h[jb[valid_b]]
            diag_pos = np.where(ms == (n - 1))[0]
            has_diag = len(diag_pos) > 0
            diag_idx = int(diag_pos[0]) if has_diag else -1
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            d_lo = -h0
            d_hi = 1.0 - h0
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
            f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-15:
                    break
                if f1 < f2:
                    d_hi = d2; d2 = d1; f2 = f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
                else:
                    d_lo = d1; d1 = d2; f1 = f2
                    d2 = d_lo + gr * (d_hi - d_lo)
                    f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            d_opt = (d_lo + d_hi) / 2.0
            f_opt = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d_opt)
            f_curr = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, 0.0)
            if f_opt < f_curr - 1e-14:
                h[p] = h0 + d_opt
                improved = True
        if improved:
            h[np.where(h < 1e-10)[0]] = 0.0
            h[np.where(h > 1.0 - 1e-10)[0]] = 1.0
            h = project_feasible(h, n)
            c_new = compute_c5(h, n)
            if c_new < best_c:
                best_c = c_new
            improved_any = True
        else:
            if not improved_any:
                break
            improved_any = False
    final_c = compute_c5(h, n)
    return h, min(best_c, final_c)

def block_cd(h, n, block_size, deadline, rng):
    """Block coordinate descent: optimize small blocks of coordinates simultaneously."""
    h = h.copy()
    best_c = compute_c5(h, n)
    imp = 0
    for iteration in range(500):
        if time.time() > deadline:
            break
        # Pick a random block
        start = rng.randint(0, n - block_size)
        block = list(range(start, start + block_size))
        x0 = h[block].copy()
        # Define sub-objective
        def sub_obj(xb):
            h_test = h.copy()
            h_test[block] = xb
            h_test = np.clip(h_test, 0, 1)
            return compute_c5(h_test, n)
        res = minimize(sub_obj, x0, method='Nelder-Mead',
                       options={'maxiter': 200, 'xatol': 1e-10, 'fatol': 1e-12})
        if res.fun < best_c - 1e-12:
            h[block] = np.clip(res.x, 0, 1)
            h = project_feasible(h, n)
            best_c = compute_c5(h, n)
            imp += 1
    return h, best_c, imp

def project_feasible(h, n):

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def _sum_eq(x, n):
