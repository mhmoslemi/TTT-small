def solve_block_2(c, h, p, n):
    """Solve 2D subproblem for positions p and p+1."""
    h0, h1 = h[p], h[p+1]
    
    # Compute affected range
    m_min = n - 1 - p
    m_max = 2 * n - 2 - (p + 1)
    # Actually for position p: m ranges from n-1-p to 2n-2-p
    # For position p+1: m ranges from n-1-(p+1) to 2n-2-(p+1)
    # Combined: m ranges from n-1-(p+1) to 2n-2-p
    
    m_lo = n - 1 - (p + 1)
    m_hi = 2 * n - 2 - p
    ms = np.arange(max(0, m_lo), min(2*n-1, m_hi) + 1)
    
    # Coefficients for position p
    jf_p = p + ms - (n - 1)
    jb_p = p - ms + (n - 1)
    vf_p = (jf_p >= 0) & (jf_p < n)
    vb_p = (jb_p >= 0) & (jb_p < n)
    ap = np.zeros(len(ms))
    if np.any(vf_p):
        ap[vf_p] += (1.0 - h[jf_p[vf_p]])
    if np.any(vb_p):
        ap[vb_p] -= h[jb_p[vb_p]]
    
    # Coefficients for position p+1
    jf_q = (p + 1) + ms - (n - 1)
    jb_q = (p + 1) - ms + (n - 1)
    vf_q = (jf_q >= 0) & (jf_q < n)
    vb_q = (jb_q >= 0) & (jb_q < n)
    aq = np.zeros(len(ms))
    if np.any(vf_q):
        aq[vf_q] += (1.0 - h[jf_q[vf_q]])
    if np.any(vb_q):
        aq[vb_q] -= h[jb_q[vb_q]]
    
    # Diagonal indices
    diag_p = (ms == (n - 1)) & (p >= 0)  # position p is diagonal at m=n-1
    diag_q = (ms == (n - 1))
    
    # Current correlation values
    c_base = c[ms].copy()
    
    # Unaffected max
    affected_mask = np.zeros(2*n-1, dtype=bool)
    affected_mask[ms] = True
    unaff = float(np.max(c[~affected_mask])) if np.any(~affected_mask) else 0.0
    
    # Objective: smooth max
    from scipy.optimize import minimize as sci_min

        x, y = xy[0] - h0, xy[1] - h1  # deltas
        new_c = c_base + x * ap + y * aq
        # Diagonal corrections
        if np.any(diag_p):

        if np.any(diag_q):
            new_c[diag_q] -= y * y
        # Note: if both diag_p and diag_q are True at the same index,
        # we've subtracted both, which is correct
        m = new_c.max()

        mx = shifted.max()

        return (m + lse / alpha)

    res = sci_min(obj, x0, method='L-BFGS-B', bounds=bounds, options={'maxiter': 100, 'ftol': 1e-15})

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def solve_1d(c, h, p, n):
    h0 = h[p]
    m_min = max(0, n - 1 - p)
    m_max = min(2 * n - 2, 2 * n - 2 - p)
    if m_min > m_max:
        return h0
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms].copy()
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    if has_diag:
        c_diag = c_base[diag_idx[0]]
        a[diag_idx] = 0.0
    unaff = 0.0
    if m_min > 0:
        unaff = float(np.max(c[:m_min]))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    cur_max = float(np.max(c))
    neg_mask = a < -1e-15
    pos_mask = a > 1e-15
    zero_mask = np.abs(a) <= 1e-15
    L_lo, L_hi = unaff, cur_max
    for _ in range(60):
        if L_hi - L_lo < 1e-13:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        feasible = True
        if np.any(zero_mask) and np.any(c_base[zero_mask] > L + 1e-14):
            feasible = False
        if feasible and np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if feasible and np.any(pos_mask):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        if x_lo > x_hi + 1e-13:
            feasible = False
        if feasible and has_diag:
            Cq = L - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                xc_lo, xc_hi = max(x_lo, 0.0), min(x_hi, 1.0)
                if not (xc_lo < r1 and xc_hi > r2):
                    feasible = False
        if feasible:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Cq = L - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            xc_lo, xc_hi = max(x_lo, 0.0), min(x_hi, 1.0)
            if xc_lo < r1 and xc_hi > r2:

                    x_hi = min(x_hi, r1)
                else:
                    x_lo = max(x_lo, r2)
            elif x_lo >= r1 and x_hi <= r2:
                if h0 - r1 <= r2 - h0:
                    x_hi = min(x_hi, r1)
                else:
                    x_lo = max(x_lo, r2)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def update_corr(c, h, p, old_val, new_val, n):
    delta = new_val - old_val
    if abs(delta) < 1e-14:
        return c
    m_min = max(0, n - 1 - p)
    m_max = min(2 * n - 2, 2 * n - 2 - p)
    if m_min > m_max:
        return c
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    diag_i = np.where(ms == (n - 1))[0]
    c[ms] += delta * a
    if len(diag_i) > 0:
        c[ms[diag_i[0]]] -= delta * delta
    return c

def exact_cd(h, n, max_passes, deadline, rng=None, recompute_every=20):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for it in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        order = list(range(n))
        if rng is not None and it % 3 == 1:
            rng.shuffle(order)
        for p in order:
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            xn = solve_1d(c, h, p, n)
            if abs(xn - h0) > 1e-12:
                update_corr(c, h, p, h0, xn, n)
                h[p] = xn
                improved = True
        if not improved:
            break
        if (it + 1) % recompute_every == 0:
            h = project_feasible(h, n)
            c = np.correlate(h, 1.0 - h, mode='full')
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def smooth_lse(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_opt(h, n, alphas, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        def cf(x):
            return x.sum() - n / 2.0
        def cj(x):
            return np.ones(n)
        cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]
        try:
            res = minimize(smooth_lse, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-20})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c - 1e-12:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def interp_h(h_old, n_old, n_new):

    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)

    # Phase 1: Strong initial CD
    h1, c1 = exact_cd(h_best, n, 300, deadline, rng, recompute_every=15)
    if c1 < best_c:
        h_best, best_c = h1, c1

    # Phase 2: Block CD (2-coordinate pairs) for several passes
    c_vec = np.correlate(h_best, 1.0 - h_best, mode='full')
    for bp in range(10):
        if time.time() > deadline - 400:
            break
        h_best, c_vec = block_cd_pass(h_best, n, c_vec, deadline, rng)
        h_best = project_feasible(h_best, n)
        c_now = compute_c5(h_best, n)
        if c_now < best_c:
            best_c = c_now
        c_vec = np.correlate(h_best, 1.0 - h_best, mode='full')

    # Phase 3: CD after block CD
    h3, c3 = exact_cd(h_best, n, 200, deadline, rng, recompute_every=15)
    if c3 < best_c:
        h_best, best_c = h3, c3

    # Phase 4: SLSQP with diverse alphas
    h4, c4 = slsqp_opt(h_best, n, [3000, 8000, 20000, 50000, 120000, 300000], 500, deadline)
    if c4 < best_c:
        h_best, best_c = h4, c4

    # Phase 5: CD after SLSQP
    h5, c5 = exact_cd(h_best, n, 200, deadline, rng, recompute_every=15)
    if c5 < best_c:
        h_best, best_c = h5, c4

    # Phase 6: Large perturbations + CD
    n = len(h_best)
    for scale in [0.15, 0.12, 0.09, 0.07, 0.05, 0.04]:
        for attempt in range(3):
            if time.time() > deadline - 150:
                break
            hp = project_feasible(h_best + rng.normal(0, scale, n), n)
            hp, c = exact_cd(hp, n, 60, deadline, rng, recompute_every=10)
            if c < best_c:
                h_best, best_c = hp, c
            # Also try a shifted version
            shift = rng.randint(1, max(2, n // 10))
            hs = project_feasible(np.roll(h_best, shift), n)
            hs, cs = exact_cd(hs, n, 40, deadline, rng, recompute_every=10)
            if cs < best_c:
                h_best, best_c = hs, cs

    # Phase 7: Multi-resolution exploration
    n_list = [500, 600, 800, 400, 700, 550]
    for n_try in n_list:
        if time.time() > deadline - 200:
            break
        h_int = interp_h(h_best, len(h_best), n_try)
        h_o, c_o = exact_cd(h_int, n_try, 150, deadline, rng, recompute_every=15)
        if c_o < best_c - 1e-10:
            h_best, best_c = h_o, c_o
            c_o_vec = np.correlate(h_best, 1.0 - h_best, mode='full')
        if time.time() < deadline - 120:
            h_o2, c_o2 = slsqp_opt(h_o, n_try, [10000, 50000, 150000], 300, deadline)
            if c_o2 < best_c - 1e-10:
                h_best, best_c = h_o2, c_o2
            h_o3, c_o3 = exact_cd(h_o2, n_try, 100, deadline, rng, recompute_every=15)
            if c_o3 < best_c - 1e-10:
                h_best, best_c = h_o3, c_o3

    # Phase 8: Final block CD + CD polish
    n_final = len(h_best)
    if time.time() < deadline - 80:
        c_vec = np.correlate(h_best, 1.0 - h_best, mode='full')
        for bp in range(5):
            if time.time() > deadline - 60:
                break
            h_best, c_vec = block_cd_pass(h_best, n_final, c_vec, deadline, rng)
            h_best = project_feasible(h_best, n_final)
            c_now = compute_c5(h_best, n_final)
            if c_now < best_c:
                best_c = c_now
            c_vec = np.correlate(h_best, 1.0 - h_best, mode='full')

    # Phase 9: Final CD
    h_f, c_f = exact_cd(h_best, n_final, 400, deadline, rng, recompute_every=10)
    if c_f < best_c:
        h_best, best_c = h_f, c_f

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
