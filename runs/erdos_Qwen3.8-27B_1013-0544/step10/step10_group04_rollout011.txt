import time
import numpy as np


def project_box_sum(y, n):
    target = 0.5 * float(n)
    y = np.asarray(y, dtype=float)
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(35):
        tau = 0.5 * (lo + hi)
        val = np.clip(y - tau, 0.0, 1.0).sum()
        if val > target:
            lo = tau
        else:
            hi = tau
    tau = 0.5 * (lo + hi)
    x = np.clip(y - tau, 0.0, 1.0)
    diff = target - x.sum()
    for _ in range(25):
        if abs(diff) < 1e-14:
            break
        if diff > 0.0:
            room = 1.0 - x
            tot = room.sum()
            if tot < 1e-15:
                break
            x = np.clip(x + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = x.sum()
            if tot < 1e-15:
                break
            x = np.clip(x + (diff / tot) * x, 0.0, 1.0)
    return x


def compute_cfull(h):
    b = 1.0 - h
    return np.convolve(h, b[::-1], mode='full')


def exact_c5_from_cfull(c, n):
    return float(np.max(c) * (2.0 / float(n)))


def exact_c5(h, n):
    return exact_c5_from_cfull(compute_cfull(h), n)


def lag_arrays(c, n):
    cp = np.empty(n)
    cn = np.empty(n)
    cp[0] = c[n - 1]
    cn[0] = c[n - 1]
    if n > 1:
        cp[1:] = c[n - 2::-1]
        cn[1:] = c[n:]
    return cp, cn


def smooth_value(c, alpha):
    mx = float(np.max(c))
    z = alpha * (c - mx)
    ez = np.exp(z)
    return mx + float(np.sum(ez)) / alpha


def conv_right(a, w):
    return np.convolve(a, w[::-1], mode='full')[len(a) - 1:2 * len(a) - 1]


def conv_left(a, w):
    return np.convolve(a, w, mode='full')[:len(a)]


def smooth_grad(c, h, n, alpha):
    mx = float(np.max(c))
    ez = np.exp(alpha * (c - mx))
    w = ez / ez.sum()
    wpos = np.zeros(n)
    wneg = np.zeros(n)
    if n > 1:
        wpos[1:] = w[n - 2::-1]
        wneg[1:] = w[n:]
    w0 = w[n - 1]
    u = 1.0 - h
    grad = w0 * (1.0 - 2.0 * h)
    grad += conv_right(u, wpos) - conv_right(h, wneg)
    grad += conv_left(u, wneg) - conv_left(h, wpos)
    return grad


def projected_gradient_optimize(h, n, alphas, steps_per_alpha, deadline):
    h = project_box_sum(h, n)
    c = exact_c5(h, n)
    best_h = h.copy()
    best_c = c
    for alpha in alphas:
        if time.time() > deadline:
            break
        h_cur = best_h.copy()
        patience = 0
        for _ in range(steps_per_alpha):
            if time.time() > deadline:
                break
            cfull = compute_cfull(h_cur)
            f = smooth_value(cfull, alpha)
            grad = smooth_grad(cfull, h_cur, n, alpha)
            gnorm = float(np.linalg.norm(grad))
            if gnorm < 1e-12:
                break
            lr = 0.1 / max(1.0, gnorm)
            found = False
            y = None
            ye = c
            for __ in range(12):
                y = project_box_sum(h_cur - lr * grad, n)
                yc = compute_cfull(y)
                yf = smooth_value(yc, alpha)
                ye = float(np.max(yc) * (2.0 / float(n)))
                if yf < f - 1e-12 or ye < best_c - 1e-12:
                    found = True
                    break
                lr *= 0.5
            if not found:
                break
            h_cur = y
            if ye < best_c - 1e-14:
                best_c = ye
                best_h = y.copy()
                patience = 0
            else:
                patience += 1
                if patience > 10:
                    break
    return best_h, best_c


def coord_base(Cpos, Cneg, p, n):
    rr = n - 1 - p
    rl = p
    R = rr if rr > rl else rl
    if R + 1 < n:
        bp = float(np.max(Cpos[R + 1:]))
        bn = float(np.max(Cneg[R + 1:]))
        return max(bp, bn), R
    return 0.0, R


def build_aff(Cpos, Cneg, h, p, n, R, cbuf, abuf):
    if R > 0:
        rr = n - 1 - p
        rl = p
        ap = abuf[:R]
        an = abuf[R:2 * R]
        ap.fill(0.0)
        an.fill(0.0)
        if rr > 0:
            hp = h[p + 1:p + 1 + rr]
            ap[:rr] += 1.0 - hp
            an[:rr] -= hp
        if rl > 0:
            hm = h[p - rl:p][::-1]
            ap[:rl] -= hm
            an[:rl] += 1.0 - hm
        cbuf[:R] = Cpos[1:1 + R]
        cbuf[R:2 * R] = Cneg[1:1 + R]
    return cbuf[:2 * R], abuf[:2 * R]


def apply_update(Cpos, Cneg, h, p, n, z, x, coeff, R):
    delta = x - z
    if abs(delta) <= 1e-15:
        return
    if R > 0:
        Cpos[1:1 + R] += coeff[:R] * delta
        Cneg[1:1 + R] += coeff[R:2 * R] * delta
    Cpos[0] += x - x * x - (z - z * z)
    Cneg[0] = Cpos[0]
    h[p] = x


def fast_coord(Cpos, Cneg, h, p, n, grid, cbuf, abuf):
    z = h[p]
    base, R = coord_base(Cpos, Cneg, p, n)
    c0 = Cpos[0]
    c_aff, a_aff = build_aff(Cpos, Cneg, h, p, n, R, cbuf, abuf)
    if c_aff.size == 0:
        return z, R
    affm = float(np.max(c_aff))
    cm = max(base, affm, c0)
    if base >= cm - 1e-13:
        return z, R
    vals = c_aff[:, None] + a_aff[:, None] * (grid[None, :] - z)
    rowmax = np.max(vals, axis=0)
    rowmax = np.maximum(rowmax, base)
    diag = c0 - z + z * z + grid - grid * grid
    rowmax = np.maximum(rowmax, diag)
    best_val = float(np.min(rowmax))
    if best_val >= cm - 1e-13:
        return z, R
    idx = int(np.argmin(rowmax))
    x = float(grid[idx])
    lo = max(0.0, x - 0.04)
    hi = min(1.0, x + 0.04)
    if hi - lo > 1e-12:
        local = np.linspace(lo, hi, 9)
        if lo > 0.0 and hi < 1.0:
            local[4] = x
        vloc = c_aff[:, None] + a_aff[:, None] * (local[None, :] - z)
        rloc = np.max(vloc, axis=0)
        rloc = np.maximum(rloc, base)
        dloc = c0 - z + z * z + local - local * local
        rloc = np.maximum(rloc, dloc)
        j = int(np.argmin(rloc))
        if rloc[j] < best_val - 1e-14:
            x = float(local[j])
            best_val = float(rloc[j])
    dx = x - z
    if abs(dx) <= 1e-13:
        return z, R
    newaff = c_aff + a_aff * dx
    newmax = max(base, float(np.max(newaff)), c0 - z + z * z + x - x * x)
    if newmax >= cm - 1e-13:
        return z, R
    return x, R


def coord_feasible(L, a_neg, off_neg, a_pos, off_pos, const_max, c0, z):
    if L < const_max - 1e-13:
        return False, 0.0, 1.0, -1.0, 1.0
    lo = 0.0
    hi = 1.0
    if a_neg.size:
        v = float(np.max((L + off_neg) / a_neg))
        if v > lo:
            lo = v
    if a_pos.size:
        v = float(np.min((L + off_pos) / a_pos))
        if v < hi:
            hi = v
    if lo < 0.0:
        lo = 0.0
    if hi > 1.0:
        hi = 1.0
    if lo > hi + 1e-13:
        return False, lo, hi, -1.0, 1.0
    cq = L - c0 + z - z * z
    if cq < -1e-12:
        return False, lo, hi, -1.0, 1.0
    if cq >= 0.25 - 1e-14:
        return True, lo, hi, -1.0, 1.0
    disc = 1.0 - 4.0 * cq
    if disc < 0.0:
        return False, lo, hi, -1.0, 1.0
    s = float(np.sqrt(disc))
    r1 = 0.5 * (1.0 - s)
    r2 = 0.5 * (1.0 + s)
    if not (lo <= r1 + 1e-13 or hi >= r2 - 1e-13):
        return False, lo, hi, r1, r2
    return True, lo, hi, r1, r2


def choose_coord_x(lo, hi, r1, r2):
    lo = max(0.0, float(lo))
    hi = min(1.0, float(hi))
    if lo > hi:
        return float(np.clip(0.5 * (lo + hi), 0.0, 1.0))
    if r1 < 0.0:
        return float(0.5 * (lo + hi))
    best_len = -1.0
    best_x = lo
    end = min(hi, r1)
    if end >= lo - 1e-12:
        length = end - lo
        if length > 1e-12:
            best_x = 0.5 * (lo + end)
            best_len = length
        else:
            best_x = max(lo, min(hi, r1))
            best_len = 0.0
    start = max(lo, r2)
    if hi >= start - 1e-12:
        length = hi - start
        if length > best_len + 1e-12:
            if length > 1e-12:
                best_x = 0.5 * (start + hi)
            else:
                best_x = max(lo, min(1.0, start))
            best_len = length
    return float(np.clip(best_x, 0.0, 1.0))


def exact_coord(Cpos, Cneg, h, p, n, cbuf, abuf):
    z = h[p]
    base, R = coord_base(Cpos, Cneg, p, n)
    c0 = Cpos[0]
    c_aff, a_aff = build_aff(Cpos, Cneg, h, p, n, R, cbuf, abuf)
    if c_aff.size == 0:
        return z, R
    cm = max(base, float(np.max(c_aff)), c0)
    if base >= cm - 1e-13:
        return z, R
    const_max = base
    zero = np.abs(a_aff) <= 1e-14
    if np.any(zero):
        const_max = max(const_max, float(np.max(c_aff[zero])))
    if const_max >= cm - 1e-13:
        return z, R
    negmask = a_aff < -1e-14
    posmask = a_aff > 1e-14
    a_neg = a_aff[negmask]
    off_neg = -c_aff[negmask] + a_neg * z
    a_pos = a_aff[posmask]
    off_pos = -c_aff[posmask] + a_pos * z
    Llo = float(const_max)
    Lhi = float(cm)
    if Lhi < Llo:
        Lhi = Llo
    for _ in range(35):
        L = 0.5 * (Llo + Lhi)
        ok, _, _, _, _ = coord_feasible(L, a_neg, off_neg, a_pos, off_pos,
                                        const_max, c0, z)
        if ok:
            Lhi = L
        else:
            Llo = L
        if Lhi - Llo < 1e-12:
            break
    if Lhi >= cm - 1e-12:
        return z, R
    ok, lo, hi, r1, r2 = coord_feasible(Lhi, a_neg, off_neg, a_pos, off_pos,
                                        const_max, c0, z)
    if not ok:
        return z, R
    x = choose_coord_x(lo, hi, r1, r2)
    dx = x - z
    if abs(dx) <= 1e-13:
        return z, R
    newaff = c_aff + a_aff * dx
    newmax = max(base, float(np.max(newaff)), c0 - z + z * z + x - x * x)
    if newmax >= cm - 1e-13:
        return z, R
    return x, R


def cd_fast_pass(h, n, grid, passes, deadline, rng, random_order=False):
    h = project_box_sum(h, n)
    cfull = compute_cfull(h)
    Cpos, Cneg = lag_arrays(cfull, n)
    best_c = exact_c5_from_cfull(cfull, n)
    best_h = h.copy()
    sz = max(1, 2 * max(0, n - 1))
    cbuf = np.empty(sz)
    abuf = np.empty(sz)
    for _ in range(passes):
        if time.time() > deadline:
            break
        if random_order:
            order = rng.permutation(n)
        else:
            order = np.arange(n)
        changed = False
        for p in order:
            if time.time() > deadline:
                break
            z = h[p]
            x, R = fast_coord(Cpos, Cneg, h, p, n, grid, cbuf, abuf)
            if abs(x - z) > 1e-13:
                apply_update(Cpos, Cneg, h, p, n, z, x, abuf, R)
                changed = True
        if not changed:
            break
        h = project_box_sum(h, n)
        new_c = exact_c5(h, n)
        if new_c < best_c - 1e-13:
            best_h = h.copy()
            best_c = new_c
            cfull = compute_cfull(h)
            Cpos, Cneg = lag_arrays(cfull, n)
        else:
            h = best_h.copy()
            break
    return best_h, best_c


def cd_exact_pass(h, n, passes, deadline):
    h = project_box_sum(h, n)
    cfull = compute_cfull(h)
    Cpos, Cneg = lag_arrays(cfull, n)
    best_c = exact_c5_from_cfull(cfull, n)
    best_h = h.copy()
    sz = max(1, 2 * max(0, n - 1))
    cbuf = np.empty(sz)
    abuf = np.empty(sz)
    for _ in range(passes):
        if time.time() > deadline:
            break
        changed = False
        for p in np.arange(n):
            if time.time() > deadline:
                break
            z = h[p]
            x, R = exact_coord(Cpos, Cneg, h, p, n, cbuf, abuf)
            if abs(x - z) > 1e-13:

        if not changed:

        h = project_box_sum(h, n)

        if new_c < best_c - 1e-13:
            best_h = h.copy()

def resample_h(h, n_src, n_dst):
    if n_dst == n_src:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, n_src, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    y = np.interp(x_new, x_old, h)
    return project_box_sum(y, n_dst)


def polish_light(h, n, grid, deadline, rng):
    h = project_box_sum(h, n)
    c = exact_c5(h, n)
    if time.time() < deadline - 12:
        h1, c1 = cd_fast_pass(h, n, grid, 3, deadline - 12, rng, False)
        if c1 < c:
            h, c = h1, c1
    if time.time() < deadline - 6:
        h2, c2 = cd_exact_pass(h, n, 1, deadline - 4)
        if c2 < c:
            h, c = h2, c2
    if time.time() < deadline - 10:
        h3, c3 = cd_fast_pass(h, n, grid, 2, deadline, rng, True)
        if c3 < c:
            h, c = h3, c3
    return h, c


def polish_main(h, n, grid, pgd_deadline, fast_deadline, exact_deadline, rng):
    h = project_box_sum(h, n)
    c = exact_c5(h, n)
    if time.time() < pgd_deadline:
        h1, c1 = projected_gradient_optimize(h, n, [60, 200, 600], 18, pgd_deadline)
        if c1 < c:
            h, c = h1, c1
    if time.time() < fast_deadline - 30:
        h2, c2 = cd_fast_pass(h, n, grid, 14, fast_deadline - 30, rng, False)
        if c2 < c:
            h, c = h2, c2
    if time.time() < fast_deadline:
        h3, c3 = cd_fast_pass(h, n, grid, 8, fast_deadline, rng, True)
        if c3 < c:
            h, c = h3, c3
    if time.time() < exact_deadline:
        h4, c4 = cd_exact_pass(h, n, 5, exact_deadline)
        if c4 < c:
            h, c = h4, c4
    return h, c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 5.0
    rng = np.random.RandomState(seed)

    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n_best = len(h_best)
    except Exception:
        n_best = 700
        h_best = np.full(n_best, 0.5)

    h_best = project_box_sum(h_best, n_best)
    c_best = exact_c5(h_best, n_best)

    grid = np.linspace(0.0, 1.0, 31)

    if budget_s > 700:
        pgd_dl = t0 + min(150.0, max(45.0, float(budget_s) - 950.0))
        fast_dl = t0 + min(540.0, max(pgd_dl + 80.0, float(budget_s) - 560.0))
        exact_dl = t0 + min(660.0, max(fast_dl + 80.0, float(budget_s) - 440.0))
        noise_dl = t0 + min(880.0, max(exact_dl + 80.0, float(budget_s) - 220.0))
    else:
        pgd_dl = t0 + max(20.0, 0.15 * float(budget_s))
        fast_dl = t0 + max(pgd_dl + 20.0, 0.55 * float(budget_s))
        exact_dl = t0 + max(fast_dl + 20.0, 0.75 * float(budget_s))
        noise_dl = deadline

    h_main, c_main = polish_main(h_best, n_best, grid, pgd_dl, fast_dl, exact_dl, rng)
    if c_main < c_best:
        h_best, c_best = h_main, c_main

    if noise_dl > time.time():
        for s in [0.006, 0.012, 0.02, 0.035]:
            rem = noise_dl - time.time()
            if rem < 160.0:
                break
            att_dl = time.time() + max(45.0, min(rem - 5.0, 170.0))
            h_try = project_box_sum(h_best + rng.normal(0.0, s, n_best), n_best)
            h_try, c_try = polish_light(h_try, n_best, grid, att_dl, rng)
            if c_try < c_best - 1e-11:
                h_best, c_best = h_try, c_try

    cand = []
    for v in [n_best, n_best - 50, n_best + 50, n_best - 100,
              n_best + 100, 500, 600, 650, 700, 750, 800, 850, 900, 450, 550]:
        iv = int(v)
        if iv not in cand and 60 < iv < 1000:
            cand.append(iv)

    n_end = deadline - 8.0
    for n_try in cand:
        rem = n_end - time.time()
        if rem < 130.0:
            break
        if n_try == n_best and c_best <= 0.380860:
            continue
        att_dl = time.time() + max(45.0, min(rem - 6.0, 150.0))
        h_try = resample_h(h_best, n_best, n_try)
        h_try, c_try = polish_light(h_try, n_try, grid, att_dl, rng)
        if c_try < c_best - 1e-11:
            h_best, c_best, n_best = h_try, c_try, n_try

    if time.time() < n_end - 90.0 and n_best % 2 == 0:
        m = n_best // 2
        base = resample_h(h_best, n_best, m)
        full = np.empty(n_best)
        full[:m] = base
        full[m:] = 1.0 - base
        full = project_box_sum(full, n_best)
        att_dl = n_end - 10.0
        h_try, c_try = polish_light(full, n_best, grid, att_dl, rng)
        if c_try < c_best - 1e-11:
            h_best, c_best = h_try, c_try

    h_best = project_box_sum(h_best, n_best)
    c_best = exact_c5(h_best, n_best)
    return h_best, c_best, n_best
