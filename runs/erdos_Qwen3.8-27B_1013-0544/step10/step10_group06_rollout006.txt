def coord_descent(h,n,max_passes,deadline,rng,orient,random_order=True,nbisect=25):
    h=h.copy()

       if time.time() > deadline: break
       c=np.correlate(h,1.0-h,mode='full')
       cur_max=float(c.max())
       if random_order and rng is not None: order=rng.permutation(n)
       else: order=np.arange(n)
       improved=False
       for p in order:
          if time.time()>deadline: break
          h0=h[p]
          x_new,m_nd,a=solve_coord(c,h,p,n,orient,nbisect)
          if abs(x_new-h0)>1e-13:

    m=float(c.max())

    mx=float(z.max())

def coord_descent(h, n, max_passes, deadline, rng, orient, random_order=True, nbisect=25):
    h = project_feasible(h, n).copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        if random_order and rng is not None:
            order = rng.permutation(n)
        else:
            order = np.arange(n)
        improved = False
        for p in order:
            if time.time() > deadline:
                break
            h0 = float(h[p])
            xnew, m_nd, a_nd = solve_coord(c, h, p, n, orient, nbisect)
            if abs(xnew - h0) > 1e-13:
                delta = xnew - h0
                h[p] = xnew
                if m_nd.size:
                    c[m_nd] += a_nd * delta
                c[n - 1] += xnew * (1.0 - xnew) - h0 * (1.0 - h0)
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    z = alpha * (c - m)
    mx = float(z.max())
    e = np.exp(z - mx)
    val = m + (mx + float(np.log(np.sum(e)))) / alpha
    res = float(x.sum() - 0.5 * n)
    return val + pen * res * res

def lse_gradient(x, n, alpha, orient, pen):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    z = alpha * (c - m)
    mx = float(z.max())
    e = np.exp(z - mx)
    w = e / float(np.sum(e))
    grad = np.zeros(n)
    for m_i in range(len(w)):
        wt = float(w[m_i])
        if wt <= 1e-14:
            continue
        lag = m_i - (n - 1)
        if orient == 0:
            if lag >= 0:
                i1 = n - lag
                if i1 <= 0:
                    continue
                grad[:i1] += wt * (1.0 - x[lag:])
                grad[lag:] -= wt * x[:i1]
            else:
                d = -lag
                i1 = n - d
                grad[d:] += wt * (1.0 - x[:i1])
                grad[:i1] -= wt * x[d:]
        else:
            if lag >= 0:
                i1 = n - lag
                if i1 <= 0:
                    continue
                grad[lag:] += wt * (1.0 - x[:i1])
                grad[:i1] -= wt * x[lag:]
            else:
                d = -lag
                i1 = n - d
                grad[:i1] += wt * (1.0 - x[d:])
                grad[d:] -= wt * x[:i1]
    res = float(x.sum() - 0.5 * n)
    if pen != 0.0:
        grad += (2.0 * pen * res)
    return grad

def lse_lbfgs(h, n, alphas, iters, orient, deadline):
    best = project_feasible(h, n).copy()
    best_c = compute_c5(best, n)
    for alpha in alphas:
        if time.time() > deadline - 20:
            break
        try:
            bounds = [(0.0, 1.0)] * n
            res = minimize(lse_value, best, args=(n, alpha, orient, 8.0),
                           jac=lse_gradient, method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': iters, 'ftol': 1e-12, 'disp': False})
            cand = project_feasible(np.asarray(res.x, dtype=float), n)
            cc = compute_c5(cand, n)
            if cc < best_c:
                best = cand
                best_c = cc
        except Exception:
            pass
    return best, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget = max(60.0, float(budget_s))
    deadline = t0 + budget - 4.0
    rng = np.random.RandomState(seed)
    TARGET = 0.3808

    try:
        h = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h = None

    if h is None or h.size == 0:
        n = 700
        x = sample_grid(n)
        h = 0.5 + 0.25 * np.sin(np.pi * x) + 0.12 * np.sin(3.0 * np.pi * x)
        h = h + 0.05 * rng.normal(size=n)

    n = int(h.size)
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    best_n = n
    orient = detect_orientation(best_h, best_n)

    core_time = min(230.0, max(30.0, 0.23 * (budget - 20.0)))
    end = min(deadline, time.time() + core_time)
    if end > time.time() + 3:
        hh, cc = coord_descent(best_h, best_n, 100000, end, rng, orient, True, 25)
        if cc < best_c:
            best_h, best_c = hh, cc

    if time.time() < deadline - 100:
        trans = (0.5 * (best_h + 1.0 - best_h[::-1]), 1.0 - best_h, best_h[::-1])
        for tc in trans:
            if time.time() > deadline - 60:
                break
            tc = project_feasible(tc, best_n)
            tc_c = compute_c5(tc, best_n)
            if tc_c < best_c:
                best_h, best_c = tc, tc_c
                end2 = min(deadline, time.time() + 30)
                hh2, cc2 = coord_descent(best_h, best_n, 100000, end2, rng, orient, True, 24)
                if cc2 < best_c:
                    best_h, best_c = hh2, cc2

    if best_c > TARGET and time.time() < deadline - 130:
        alphas = (700, 7000, 70000)
        iters = (22, 18, 14)
        for a, it in zip(alphas, iters):
            if time.time() > deadline - 30:
                break
            hh, cc = lse_lbfgs(best_h, best_n, [a], it, orient, deadline)
            if cc < best_c:
                best_h, best_c = hh, cc
        if time.time() < deadline - 20:
            hh, cc = coord_descent(best_h, best_n, 100000, min(deadline, time.time() + 35),
                                   rng, orient, False, 26)
            if cc < best_c:
                best_h, best_c = hh, cc

    if time.time() < deadline - 80:
        for amp in (0.02, 0.06):
            if time.time() > deadline - 40:
                break
            hp = project_feasible(best_h + amp * rng.normal(size=best_n), best_n)
            hh, cc = coord_descent(hp, best_n, 100000, min(deadline, time.time() + 25),
                                   rng, orient, True, 24)
            if cc < best_c:
                best_h, best_c = hh, cc

    if time.time() < deadline - 160:
        old_x = sample_grid(best_n)
        for n_try in (900, 1000):
            if time.time() > deadline - 110:
                break
            if n_try == best_n:
                continue
            new_x = sample_grid(n_try)
            ht = project_feasible(np.interp(new_x, old_x, best_h), n_try)
            ot = detect_orientation(ht, n_try)
            hh, cc = coord_descent(ht, n_try, 100000, min(deadline, time.time() + 70),
                                   rng, ot, True, 25)
            if cc < best_c:
                best_h, best_c, best_n, orient = hh, cc, n_try, ot
                if time.time() < deadline - 35:
                    hh2, cc2 = coord_descent(best_h, best_n, 100000, min(deadline, time.time() + 25),
                                             rng, orient, False, 25)
                    if cc2 < best_c:
                        best_h, best_c = hh2, cc2

    if time.time() < deadline - 8:
        hh, cc = coord_descent(best_h, best_n, 100000, deadline - 2, rng, orient, False, 28)
        if cc < best_c:
            best_h, best_c = hh, cc

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
