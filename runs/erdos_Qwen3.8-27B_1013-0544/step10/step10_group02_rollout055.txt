def update_corr(c, h, p, old_val, new_val, n):

    if abs(delta) < 1e-14:
        return
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]

    if len(diag_i) > 0:

                pass  # natural order

            else:

def lse_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_opt(h, n, alphas, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
                 'jac': lambda x: np.ones(n)}]
        try:
            res = minimize(lse_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c - 1e-12:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

                continue
            xn = solve_1d(c, h, p, n)
            if abs(xn - h0) > 1e-13:
                update_corr(c, h, p, h0, xn, n)
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = get_corr(h, n)
    return h, compute_c5(h, n)

def lse_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def _cf(x):
    return x.sum() - _current_n / 2.0

def _cj(x):
    return np.ones(_current_n)

def slsqp_opt(h, n, alphas, maxiter, deadline):
    global _current_n
    _current_n = n
    best_h, best_c = h.copy(), compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj}]
    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(lse_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c - 1e-12:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def interp_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def antisymmetrize(h, n):
    half = n // 2
    hs = h.copy()
    for i in range(half):
        avg = (h[i] + (1.0 - h[i + half])) / 2.0
        hs[i] = avg
        hs[i + half] = 1.0 - avg
    return project_feasible(hs, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = np.full(500, 0.5)
        n = 500
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)

    # Phase 1: Heavy CD with varied ordering
    h1, c1 = exact_cd(h_best, n, 500, deadline, rng)
    if c1 < best_c:
        h_best, best_c = h1, c1

    # Phase 2: SLSQP refinement
    h2, c2 = slsqp_opt(h_best, n, [2000, 8000, 30000, 100000, 300000], 400, deadline)
    if c2 < best_c - 1e-12:
        h_best, best_c = h2, c2

    # Phase 3: CD after SLSQP
    h3, c3 = exact_cd(h_best, n, 400, deadline, rng)
    if c3 < best_c - 1e-12:
        h_best, best_c = h3, c3

    # Phase 4: Anti-symmetric exploration
    if time.time() < deadline - 200:
        n_cur = len(h_best)
        h_as = antisymmetrize(h_best, n_cur)
        h_as2, c_as = exact_cd(h_as, n_cur, 200, deadline, rng)
        if c_as < best_c - 1e-9:
            h_best, best_c = h_as2, c_as

    # Phase 5: Multi-resolution
    n_list = [400, 450, 500, 550, 600, 650, 750, 800, 850, 900, 1000, 350, 480, 560]
    for n_try in n_list:
        if time.time() > deadline - 250:
            break
        if n_try == len(h_best):
            continue
        h_int = interp_h(h_best, len(h_best), n_try)
        h_o, c_o = exact_cd(h_int, n_try, 150, deadline, rng)
        if c_o < best_c - 1e-9:
            h_best, best_c = h_o, c_o
        if time.time() < deadline - 120:
            h_o2, c_o2 = slsqp_opt(h_o, n_try, [5000, 30000, 100000], 250, deadline)
            if c_o2 < best_c - 1e-9:
                h_best, best_c = h_o2, c_o2
            h_o3, c_o3 = exact_cd(h_o2, n_try, 100, deadline, rng)
            if c_o3 < best_c - 1e-9:
                h_best, best_c = h_o3, c_o3

    # Phase 6: Diverse restarts
    n_cur = len(h_best)
    for r in range(35):
        if time.time() > deadline - 80:
            break
        if r < 12:
            scale = 0.025 * (1.0 - r / 12.0) + 0.004
            hp = project_feasible(h_best + rng.normal(0, scale, n_cur), n_cur)
        elif r < 20:
            shift = rng.randint(1, max(2, n_cur // 5))
            hp = np.roll(h_best, shift)
            hp = project_feasible(hp + rng.normal(0, 0.008, n_cur), n_cur)
        elif r < 28:
            freq = rng.randint(1, 7)
            phase = rng.uniform(0, 2 * np.pi)
            pert = 0.015 * np.cos(2 * np.pi * freq * np.arange(n_cur) / n_cur + phase)
            hp = project_feasible(h_best + pert, n_cur)
        else:
            h_as = antisymmetrize(h_best, n_cur)
            hp = project_feasible(h_as + rng.normal(0, 0.01, n_cur), n_cur)
        h_o, c_o = exact_cd(hp, n_cur, 80, deadline, rng)
        if c_o < best_c - 1e-9:
            h_best, best_c = h_o, c_o

    # Phase 7: Final polish
    n_final = len(h_best)
    h_f, c_f = exact_cd(h_best, n_final, 600, deadline, rng)
    if c_f < best_c - 1e-12:
        h_best, best_c = h_f, c_f

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
