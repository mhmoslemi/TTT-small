def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)

        if abs(diff) < 1e-14:
            break

            tot = room.sum()
            if tot < 1e-15:

            if tot < 1e-15:
