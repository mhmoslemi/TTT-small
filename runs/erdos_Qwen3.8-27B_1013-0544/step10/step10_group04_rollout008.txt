def smooth_obj(x,n,alpha,pen):
    c=compute_c(x)
    m=float(np.max(c))
    sh=alpha*(c-m)
    mx=float(np.max(sh))
    lse=mx + math.log(float(np.sum(np.exp(sh - mx))))
    return (m + lse/alpha)*(2.0/n) + pen*(x.sum()-0.5*n)**2

import numpy as np
import math
import time


def corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def met(c, n):
    return float(np.max(c)) * (2.0 / n)


def project(h, n):
    h = np.asarray(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    t = 0.5 * n
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(h + mid, 0.0, 1.0)) > t:
            lo = mid
        else:
            hi = mid
    y = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(100):
        r = t - y.sum()
        if abs(r) < 1e-13:
            break
        if r > 0:
            m = y < 1.0 - 1e-14
        else:
            m = y > 1e-14
        if not m.any():
            break
        y[m] += r / float(m.sum())
        np.clip(y, 0.0, 1.0, out=y)
    return y


def apply_move(h, c, n, p, x):
    x = float(x)
    if x < 0.0:
        x = 0.0
    elif x > 1.0:
        x = 1.0
    h0 = float(h[p])
    delta = x - h0
    if abs(delta) < 1e-16:
        return 0.0
    dlo = min(-p, p - (n - 1))
    dhi = max(p, n - 1 - p)
    ds = np.arange(dlo, dhi + 1)
    a = np.zeros(ds.size)
    i1 = p - ds
    v1 = (i1 >= 0) & (i1 < n)
    if v1.any():
        a[v1] += 1.0 - h[i1[v1]]
    i2 = p + ds
    v2 = (i2 >= 0) & (i2 < n)
    if v2.any():
        a[v2] -= h[i2[v2]]
    mlo = n - 1 + dlo
    c[mlo:mlo + ds.size] += a * delta
    c[n - 1] -= delta * delta
    h[p] = x
    return delta


def coord_feasible(L, cont, cd, q0, invp, offp, invn, offn):
    if L < cont - 1e-12:
        return False
    lo = 0.0
    hi = 1.0
    if invn.size:
        v = float(np.max(L * invn - offn))
        if v > lo:
            lo = v
    if invp.size:
        v = float(np.min(L * invp - offp))
        if v < hi:
            hi = v
    if lo < 0.0:
        lo = 0.0
    if hi > 1.0:
        hi = 1.0
    if lo > hi + 1e-11:
        return False
    Cq = L - cd + q0
    if Cq < -1e-11:
        return False
    if Cq < 0.0:
        Cq = 0.0
    if Cq < 0.25 - 1e-14:
        disc = 1.0 - 4.0 * Cq
        if disc < 0.0:
            return False
        sq = math.sqrt(disc)
        r1 = 0.5 * (1.0 - sq)
        r2 = 0.5 * (1.0 + sq)
        if lo <= r1 + 1e-11 and lo <= hi + 1e-11:
            if min(lo, hi) <= max(0.0, r1) + 1e-11:
                if (lo <= r1 + 1e-11 and hi >= 0.0 - 1e-11):
                    return True
        if lo >= r2 - 1e-11 and hi >= r2 - 1e-11:
            if max(lo, r2) <= min(hi, 1.0) + 1e-11:
                return True
        return False
    return True


def cd_pass(h, c, n, order, step):
    t = 0.5 * n
    sw = float(np.sum(h))
    cm = float(np.max(c))
    eps = 1e-12
    emp = np.empty(0.0)
    for p in order:
        h0 = float(h[p])
        dlo = min(-p, p - (n - 1))
        dhi = max(p, n - 1 - p)
        ds = np.arange(dlo, dhi + 1)
        a = np.zeros(ds.size)
        i1 = p - ds
        v1 = (i1 >= 0) & (i1 < n)
        if v1.any():
            a[v1] += 1.0 - h[i1[v1]]
        i2 = p + ds
        v2 = (i2 >= 0) & (i2 < n)
        if v2.any():
            a[v2] -= h[i2[v2]]

        mlo = n - 1 + dlo
        muh = n - 1 + dhi
        cseg = c[mlo:muh + 1]

        u = -1.0
        if mlo > 0:
            u = float(np.max(c[:mlo]))
        if muh + 1 < c.size:
            vv = float(np.max(c[muh + 1:]))
            if vv > u:
                u = vv
        if u >= cm - 1e-13:
            continue

        diag = n - 1 - mlo
        pos = a > eps
        neg = a < -eps
        pos[diag] = False
        neg[diag] = False
        zero = ~(pos | neg)
        zero[diag] = False

        cont = u
        if zero.any():
            vv = float(np.max(cseg[zero]))
            if vv > cont:
                cont = vv
        if cont < 0.0:
            cont = 0.0

        cd = float(cseg[diag])
        q0 = h0 * (1.0 - h0)
        dm = cd - q0
        lb = cont if cont > dm else dm
        if lb >= cm - 1e-12:
            continue

        if pos.any():
            ap = a[pos]
            bp = cseg[pos] - h0 * ap
            invp = 1.0 / ap
            offp = bp * invp
        else:
            invp = emp
            offp = emp

        if neg.any():
            an = a[neg]
            bn = cseg[neg] - h0 * an
            invn = 1.0 / an
            offn = bn * invn
        else:
            invn = emp
            offn = emp

        llo = lb
        lhi = cm
        for _ in range(32):
            lm = 0.5 * (llo + lhi)
            if coord_feasible(lm, cont, cd, q0, invp, offp, invn, offn):
                lhi = lm
            else:
                llo = lm
        L = lhi

        lo = 0.0
        hi = 1.0
        if invn.size:
            vv = float(np.max(L * invn - offn))
            if vv > lo:
                lo = vv
        if invp.size:
            vv = float(np.min(L * invp - offp))
            if vv < hi:
                hi = vv
        if lo < 0.0:
            lo = 0.0
        if hi > 1.0:
            hi = 1.0
        if hi < lo - 1e-12:
            continue

        if lo < h0 - step:
            lo = h0 - step
        if hi > h0 + step:
            hi = h0 + step
        if hi < lo - 1e-12:
            continue

        Cq = L - cd + q0
        if Cq < -1e-11:
            continue
        if Cq < 0.0:
            Cq = 0.0

        tp = h0
        if abs(sw - t) > 1e-12:
            bb = 0.25 * (t - sw) / float(n)
            if bb > 0.02:
                bb = 0.02
            elif bb < -0.02:
                bb = -0.02
            tp = h0 + bb
            if tp < 0.0:
                tp = 0.0
            elif tp > 1.0:
                tp = 1.0

        if Cq >= 0.25 - 1e-14:
            ivals = ((0.0, 1.0),)
        else:
            disc = 1.0 - 4.0 * Cq
            if disc < 0.0:
                disc = 0.0
            sq = math.sqrt(disc)
            r1 = 0.5 * (1.0 - sq)
            r2 = 0.5 * (1.0 + sq)
            ivals = ((0.0, r1), (r2, 1.0))

        bx = None
        bd = 1e100
        for A, B in ivals:
            ll = lo if lo > A else A
            hh = hi if hi < B else B
            if ll <= hh + 1e-12:
                if tp < ll:
                    xx = ll
                elif tp > hh:
                    xx = hh
                else:
                    xx = tp
                dd = abs(xx - tp)
                if dd < bd:
                    bd = dd
                    bx = xx
        if bx is None:
            continue

        delta = bx - h0
        if abs(delta) < 1e-13:
            continue

        c[mlo:muh + 1] += a * delta
        c[n - 1] -= delta * delta
        h[p] = bx

        xmax = float(np.max(cseg))
        if u > xmax:
            xmax = u
        if xmax >= cm - 1e-12:
            apply_move(h, c, n, p, h0)
            continue
        cm = xmax
        sw += delta

    return h, c, cm


def refine(h0, n, deadline, best_c, rng, max_blocks, random_every=2):
    h = project(h0, n)
    c = corr(h)
    cur = met(c, n)
    if cur < best_c:
        best_c = cur
    for b in range(max_blocks):
        if time.time() > deadline - 10:
            break
        hw = h.copy()
        cw = c.copy()
        step = 0.5
        if b > 1:
            step = 0.3
        if b > 6:
            step = 0.15
        if b > 20:
            step = 0.07
        if b % random_every == 0:
            order = np.arange(n)
        else:
            order = rng.permutation(n)
        hw, cw, _ = cd_pass(hw, cw, n, order, step)
        hp = project(hw, n)
        cp = corr(hp)
        val = met(cp, n)
        if val < cur - 1e-14:
            cur = val
            h = hp
            c = cp
            if val < best_c:
                best_c = val
        else:
            break
    return h, c, cur, best_c


def swap_search(h, n, deadline, rng):
    h = h.copy()
    c = corr(h)
    cur = met(c, n)
    best_c = cur
    best_h = h.copy()
    maxa = 50000
    for k in range(maxa):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        oi = float(h[i])
        oj = float(h[j])
        if abs(oi - oj) < 1e-13:
            continue
        apply_move(h, c, n, i, oj)
        apply_move(h, c, n, j, oi)
        val = met(c, n)
        temp = 0.004 * math.exp(-k / 20000.0)
        accept = val < cur - 1e-14
        if not accept and temp > 1e-10 and val - cur < 0.01:
            if rng.random() < math.exp(-(val - cur) / temp):
                accept = True
        if accept:
            cur = val
            if val < best_c:
                best_c = val
                best_h = h.copy()
        else:
            apply_move(h, c, n, j, oj)
            apply_move(h, c, n, i, oi)
        if k % 2000 == 0:
            c = corr(h)
            cur = met(c, n)
            if cur < best_c:
                best_c = cur
                best_h = h.copy()
            if cur - best_c > 0.01:
                h = best_h.copy()
                c = corr(h)
                cur = best_c
    return best_h, best_c


def resample_h(h, no, nn):
    xo = np.linspace(0.0, 2.0, no)
    xn = np.linspace(0.0, 2.0, nn)
    return np.interp(xn, xo, h)
