def project(y, target):

    h=np.clip(y,0.0,1.0)
    for _ in range(25):
        diff=target-h.sum()
        if abs(diff)<1e-12: break
        if diff>0:
            free=h<1.0-1e-12
        else:
            free=h>1e-12
        cnt=free.sum()
        if cnt==0: break
        h[free]+=diff/cnt
        h=np.clip(h,0.0,1.0)
    if abs(h.sum()-target)>1e-8:
        s=np.sort(y); pref=np.empty(len(y)+1); pref[0]=0.0; pref[1:]=np.cumsum(s)
        lo=s[0]-1.0; hi=s[-1]+1.0

            i0=int(np.searchsorted(s,m,'right'))

def compute_c5(h, n):

def project(y, target):
    y = np.asarray(y, dtype=float)
    h = np.clip(y, 0.0, 1.0)
    for _ in range(25):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = h < 1.0 - 1e-12
        else:
            free = h > 1e-12
        cnt = int(free.sum())
        if cnt == 0:
            break
        h[free] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    if abs(h.sum() - target) > 1e-8:
        n = len(y)
        s = np.sort(y)
        pref = np.empty(n + 1)
        pref[0] = 0.0
        pref[1:] = np.cumsum(s)
        lo = s[0] - 1.0
        hi = s[-1] + 1.0
        for _ in range(55):
            mid = 0.5 * (lo + hi)
            i0 = int(np.searchsorted(s, mid, 'right'))
            i1 = int(np.searchsorted(s, mid + 1.0, 'left'))
            val = (pref[i1] - pref[i0]) - (i1 - i0) * mid + (n - i1)
            if val > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        fr = np.flatnonzero((h > 1e-12) & (h < 1.0 - 1e-12))
        if fr.size:
            h[fr[0]] += diff

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def project(y, target):
    y = np.asarray(y, dtype=float)
    h = np.clip(y, 0.0, 1.0)
    for _ in range(25):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = h < 1.0 - 1e-12
        else:
            free = h > 1e-12
        cnt = int(free.sum())
        if cnt == 0:
            break
        h[free] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    if abs(h.sum() - target) > 1e-8:
        n = len(y)
        s = np.sort(y)
        pref = np.empty(n + 1)
        pref[0] = 0.0
        pref[1:] = np.cumsum(s)
        lo = s[0] - 1.0
        hi = s[-1] + 1.0
        for _ in range(55):
            mid = 0.5 * (lo + hi)
            i0 = int(np.searchsorted(s, mid, 'right'))
            i1 = int(np.searchsorted(s, mid + 1.0, 'left'))
            val = (pref[i1] - pref[i0]) - (i1 - i0) * mid + (n - i1)
            if val > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        fr = np.flatnonzero((h > 1e-12) & (h < 1.0 - 1e-12))
        if fr.size:
            h[fr[0]] += diff
    return h

def smooth(x, alpha, need_grad=True):
    n = len(x)
    dx = 2.0 / n
    b = 1.0 - x
    c = np.correlate(x, b, mode='full')
    m = float(np.max(c))
    tmp = np.clip(alpha * (c - m), -700.0, 0.0)
    w = np.exp(tmp)
    ws = float(w.sum())
    if ws <= 0.0:
        ws = 1.0
    else:
        w /= ws
    val = dx * (m + np.log(ws) / alpha)
    if not need_grad:
        return val, None, dx * m
    off = n - 1
    t1 = np.convolve(w, b, 'full')
    t2 = np.convolve(w[::-1], x, 'full')
    grad = dx * (t1[off:off + n] - t2[off:off + n])
    return val, grad, dx * m

def pen_obj(x, n, alpha, mu):
    v = smooth(x, alpha, False)[0]
    d = x.sum() - 0.5 * n
    return float(v + 0.5 * mu * d * d)

def pen_grad(x, n, alpha, mu):
    _, g, _ = smooth(x, alpha, True)
    d = x.sum() - 0.5 * n
    return g + mu * d * np.ones(len(x))

def slq_obj(x, n, alpha):
    return smooth(x, alpha, False)[0]

def slq_grad(x, n, alpha):
    return smooth(x, alpha, True)[1]

def sum_eq(x, n):
    return x.sum() - 0.5 * n

def sum_jac(x, n):
    return np.ones(len(x))

def lbfgs_phase(h, n, deadline, alphas=(10.0, 50.0, 200.0, 1000.0)):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    x = project(h, 0.5 * n)
    bounds = [(0.0, 1.0)] * n
    for mu in (10.0, 100.0):
        for a in alphas:

                break
            try:
                res = minimize(pen_obj, x, args=(n, a, mu), method='L-BFGS-B',
                               jac=pen_grad, bounds=bounds,
                               options={'maxiter': 50, 'maxfun': 140,
                                        'ftol': 1e-12, 'gtol': 1e-6})
                xp = project(res.x, 0.5 * n)

                    best_h = xp
                    x = xp
                else:
                    x = xp if c < best_c + 0.002 else best_h
            except Exception:
                x = best_h
    return best_h, best_c

def slq_phase(h, n, deadline, alphas=(500.0, 2000.0), maxiter_each=20):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_jac, 'args': (n,)}]
    for a in alphas:
        if time.time() > deadline:
            break
        try:

                           jac=slq_grad, bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-12})
            xp = project(res.x, 0.5 * n)

                h = xp

        except Exception:
            h = best_h
    return best_h, best_c

def adam_smooth(h, n, alphas, lr, deadline, rng):
    target = 0.5 * n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h = h.copy()
    mom = np.zeros(n)
    var = np.zeros(n)
    b1, b2 = 0.9, 0.999
    per = int(min(5000, max(200, 25000.0 / max(1, len(alphas)))))
    stall = 0
    for ai, a in enumerate(alphas):
        if time.time() > deadline:
            break
        mom.fill(0.0)
        var.fill(0.0)
        it = 0
        lr_use = lr / (1.0 + 0.4 * ai)
        for _ in range(per):
            if time.time() > deadline:
                break
            it += 1
            val, g, cc = smooth(h, a)
            if cc < best_c:
                best_c = cc
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
            mom = b1 * mom + (1.0 - b1) * g
            var = b2 * var + (1.0 - b2) * (g * g)
            mhat = mom / (1.0 - b1 ** it)
            vhat = var / (1.0 - b2 ** it)
            step = lr_use * mhat / (np.sqrt(vhat) + 1e-8)
            hp = project(h - step, target)
            c1 = compute_c5(hp, n)
            if c1 > cc + 1e-9:
                hp = project(h - 0.25 * step, target)
                c1 = compute_c5(hp, n)
            if c1 <= cc + 1e-9:
                h = hp
                if c1 < best_c:
                    best_c = c1
                    best_h = h.copy()
                    stall = 0
            if stall > 200:
                h = project(best_h + 0.015 * rng.standard_normal(n), target)
                mom.fill(0.0)
                var.fill(0.0)
                stall = 0
        if compute_c5(h, n) > best_c + 0.005:
            h = best_h.copy()
            mom.fill(0.0)
            var.fill(0.0)
    return best_h, best_c

def binarize(h, n):
    k = int(n // 2)
    if k <= 0:
        return h.copy()
    idx = np.argpartition(-h, k - 1)[:k]
    b = np.zeros(n)
    b[idx] = 1.0
    return b
