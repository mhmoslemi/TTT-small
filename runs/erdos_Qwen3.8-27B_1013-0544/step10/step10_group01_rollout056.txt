import numpy as np
import time

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def project_feasible(x, n):
    """Exact Euclidean projection onto {h: 0<=h<=1, sum(h)=n/2}."""
    target = n / 2.0
    # Binary search for threshold
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def compute_smooth_obj_grad(h, n, alpha):
    """Compute log-sum-exp smooth max and gradient."""
    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = c.max()
    exp_c = np.exp(alpha * (c - c_max))
    sum_exp = exp_c.sum()
    obj = (c_max + np.log(sum_exp) / alpha) * (2.0 / n)
    w = exp_c / sum_exp
    
    # Compute gradient (O(n*(2n-1)) with numpy vectorization)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-10:
            continue
        # First term: (1-h[p+m-n+1]) for valid p
        p_lo = max(0, n - 1 - m)
        p_hi = min(n - 1, 2 * n - 2 - m)
        if p_lo <= p_hi:
            ps = np.arange(p_lo, p_hi + 1)
            grad[ps] += wm * (1.0 - h[ps + m - n + 1])
        # Second term: -h[p-m+n-1] for valid p
        q_lo = max(0, m - n + 1)
        q_hi = min(n - 1, m - 1)
        if q_lo <= q_hi:
            qs = np.arange(q_lo, q_hi + 1)
            grad[qs] -= wm * h[qs - m + n - 1]
    grad *= (2.0 / n)
    return obj, grad

def projected_gradient_descent(h, n, alpha_start, alpha_end, n_iters, lr, momentum, deadline):
    """PGD with Nesterov momentum and adaptive alpha."""
    h = h.copy()
    v = np.zeros_like(h)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    beta = momentum
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        frac = it / max(n_iters - 1, 1)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        obj, grad = compute_smooth_obj_grad(h, n, alpha)
        v = beta * v + (1 - beta) * (-grad)
        h = project_feasible(h + lr * v, n)
        c = compute_c5(h, n)
        if c < best_c:
            best_h = h.copy()
            best_c = c
    return best_h, best_c

def coord_descent(h, n, max_passes, deadline, gs_iters=50, threshold=1e-6):
    """Coordinate descent with golden section search."""
    h = h.copy()
    best_c = compute_c5(h, n)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    dx = 2.0 / n
    
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
                continue
            # Determine which correlation lags are affected
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(valid_f):
                a[valid_f] += (1.0 - h[jf[valid_f]])
            if np.any(valid_b):
                a[valid_b] -= h[jb[valid_b]]
            diag_pos = np.where(ms == (n - 1))[0]
            has_diag = len(diag_pos) > 0
            diag_idx = int(diag_pos[0]) if has_diag else -1
            c_aff = c[ms].copy()
            # Unaffected max
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            
            # Golden section search for d in [-h0, 1-h0]
            d_lo, d_hi = -h0, 1.0 - h0
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            
            def fval(d):
                vals = c_aff + d * a
                if has_diag:
                    vals[diag_idx] = c_aff[diag_idx] + d * (1.0 - h0 - d)
                return float(np.max(vals))
            
            f1, f2 = fval(d1), fval(d2)
            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-14:
                    break
                if f1 < f2:
                    d_hi, d2, f2 = d2, d1, f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    f1 = fval(d1)
                else:
                    d_lo, d1, f1 = d1, d2, f2
                    d2 = d_lo + gr * (d_hi - d_lo)
                    f2 = fval(d2)
            
            d_opt = (d_lo + d_hi) / 2.0
            if fval(d_opt) < fval(0.0) - 1e-13:
                h[p] = h0 + d_opt
                improved = True
        
        if improved:

            h = project_feasible(h, n)
            c_new = compute_c5(h, n)
            if c_new < best_c:
                best_c = c_new
        else:
            break
    
    return h, best_c

import numpy as np
import time

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def project_feasible(x, n):
    target = n / 2.0
    lo, hi = x.min() - 0.5, x.max() + 0.5
    for _ in range(70):
        mid = (lo + hi) / 2.0
        if np.clip(x - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d):
    vals = c_aff + d * a
    if has_diag:
        vals[diag_idx] = c_aff[diag_idx] + d * (1.0 - h0 - d)
    result = float(np.max(vals))
    if unaff_max > result:
        result = unaff_max
    return result

def coord_descent(h, n, max_passes, deadline, gs_iters=55, threshold=1e-6):
    h = h.copy()
    best_c = compute_c5(h, n)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(valid_f):
                a[valid_f] += (1.0 - h[jf[valid_f]])
            if np.any(valid_b):
                a[valid_b] -= h[jb[valid_b]]
            diag_pos = np.where(ms == (n - 1))[0]
            has_diag = len(diag_pos) > 0
            diag_idx = int(diag_pos[0]) if has_diag else -1
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            d_lo, d_hi = -h0, 1.0 - h0
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
            f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-14:
                    break
                if f1 < f2:
                    d_hi, d2, f2 = d2, d1, f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
                else:
                    d_lo, d1, f1 = d1, d2, f2
                    d2 = d_lo + gr * (d_hi - d_lo)
                    f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            d_opt = (d_lo + d_hi) / 2.0
            if _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d_opt) < \
               _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, 0.0) - 1e-13:
                h[p] = h0 + d_opt
                improved = True
        if improved:
            h[h < threshold] = 0.0
            h[h > 1.0 - threshold] = 1.0
            h = project_feasible(h, n)
            c_new = compute_c5(h, n)
            if c_new < best_c:
                best_c = c_new
        else:
            break
    return h, best_c

def smooth_obj_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = c.max()
    exp_c = np.exp(alpha * (c - c_max))
    sum_exp = exp_c.sum()
    obj = (c_max + np.log(sum_exp) / alpha) * (2.0 / n)
    w = exp_c / sum_exp
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-11:
            continue
        p_lo = max(0, n - 1 - m)
        p_hi = min(n - 1, 2 * n - 2 - m)
        if p_lo <= p_hi:
            ps = np.arange(p_lo, p_hi + 1)
            grad[ps] += wm * (1.0 - h[ps + m - n + 1])
        q_lo = max(0, m - n + 1)
        q_hi = min(n - 1, m - 1)
        if q_lo <= q_hi:
            qs = np.arange(q_lo, q_hi + 1)
            grad[qs] -= wm * h[qs - m + n - 1]
    grad *= (2.0 / n)
    return obj, grad

def pgd_optimize(h, n, alpha_start, alpha_end, n_iters, lr, momentum, deadline):
    h = h.copy()
    v = np.zeros_like(h)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(n_iters):
        if time.time() > deadline:
            break
        frac = it / max(n_iters - 1, 1)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        obj, grad = smooth_obj_grad(h, n, alpha)
        v = momentum * v - (1 - momentum) * grad
        h = project_feasible(h + lr * v, n)
        c = compute_c5(h, n)
        if c < best_c:
            best_h = h.copy()
            best_c = c
    return best_h, best_c

def slsqp_step(h, n, alpha, maxiter=200):
    def _obj(x):
        return smooth_obj_grad(x, n, alpha)[0]
    def _grad(x):
        return smooth_obj_grad(x, n, alpha)[1]
    def _cons_eq(x):
        return x.sum() - n / 2.0
    from scipy.optimize import minimize
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_obj, h, jac=_grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h
