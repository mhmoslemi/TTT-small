    x_hi = min(x_hi, 1.0)
    if has_diag:
        Cq = L - c_base[diag_idx[0]] + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            if max(x_lo, 0.0) < r1 and min(x_hi, 1.0) > r2:
                x_hi = r1 if (h0 - r1) <= (r2 - h0) else x_hi
                x_lo = x_lo if (h0 - r1) <= (r2 - h0) else r2
            elif x_lo >= r1 and x_hi <= r2:
                mid = (r1 + r2) / 2.0
                if h0 < mid:
                    x_lo = r2
                else:
                    x_hi = r1
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def exact_cd(h, n, max_passes, deadline):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            xn = cd_1d(c, h, p, n)
            if abs(xn - h0) > 1e-12:
                delta = xn - h0
                m_min = n - 1 - p
                m_max = 2 * n - 2 - p
                ms = np.arange(m_min, m_max + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                diag_i = np.where(ms == (n - 1))[0]
                c[ms] += delta * a
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True
        h = project_feasible(h, n)
        if not improved:
            break
    return h, compute_c5(h, n)

def cd_pair(h, n, p, deadline):
    p2 = p + 1
    s = n / 2.0 - h[p] - h[p2]
    a_lo = max(0.0, s - 1.0)
    a_hi = min(1.0, s)
    if a_hi - a_lo < 1e-12:
        return
    h_orig = h.copy()
    h_orig[p] = 0.0
    h_orig[p2] = 0.0
    c_base = np.correlate(h_orig, 1.0 - h_orig, mode='full')
    k_all = np.arange(2 * n - 1) - (n - 1)
    pk = p - k_all
    pk2 = p + k_all
    pkv = (pk >= 0) & (pk < n)
    pk2v = (pk2 >= 0) & (pk2 < n)
    hp = np.zeros(2 * n - 1)
    hp2 = np.zeros(2 * n - 1)
    if np.any(pkv):
        hp[pkv] = h_orig[pk[pkv]]
    if np.any(pk2v):
        hp2[pk2v] = h_orig[pk2[pk2v]]
    q_p = p + 1 - k_all
    q_p2 = p + 1 + k_all
    qp1v = (q_p >= 0) & (q_p < n)
    qp2v = (q_p2 >= 0) & (q_p2 < n)
    hq1 = np.zeros(2 * n - 1)
    hq2 = np.zeros(2 * n - 1)
    if np.any(qp1v):
        hq1[qp1v] = h_orig[q_p[qp1v]]
    if np.any(qp2v):
        hq2[qp2v] = h_orig[q_p2[qp2v]]
    A = 1.0 - hp2 - hp
    B = 1.0 - hq2 - hq1
    A0 = hp2
    B0 = hq2
    cross_lin = np.zeros(2 * n - 1)
    cross_quad = np.zeros(2 * n - 1)
    idx1 = n
    idxm1 = n - 2
    cross_quad[idx1] = 1.0
    cross_lin[idx1] = -(s + 1.0)
    cross_quad[idxm1] = 1.0
    cross_lin[idxm1] = (1.0 - s)
    Q0 = c_base + A0 + B0 + s * B
    Q1 = A - B + cross_lin
    Q2 = cross_quad
    grid = np.linspace(a_lo, a_hi, 100)
    vals = Q0[None, :] + Q1[None, :] * grid[:, None] + Q2[None, :] * grid[:, None] ** 2
    fvals = vals.max(axis=1)
    i0 = int(fvals.argmin())
    lo = grid[max(0, i0 - 1)]
    hi = grid[min(len(grid) - 1, i0 + 1)]
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    a1 = hi - (hi - lo) / phi
    a2 = lo + (hi - lo) / phi
    f1 = np.max(Q0 + Q1 * a1 + Q2 * a1 ** 2)
    f2 = np.max(Q0 + Q1 * a2 + Q2 * a2 ** 2)
    for _ in range(80):
        if hi - lo < 1e-14:
            break
        if f1 < f2:
            hi = a2; a2 = a1; f2 = f1
            a1 = hi - (hi - lo) / phi
            f1 = np.max(Q0 + Q1 * a1 + Q2 * a1 ** 2)
        else:
            lo = a1; a1 = a2; f1 = f2
            a2 = lo + (hi - lo) / phi
            f2 = np.max(Q0 + Q1 * a2 + Q2 * a2 ** 2)
    a_star = float(np.clip((lo + hi) / 2.0, a_lo, a_hi))
    h[p] = a_star
    h[p2] = s - a_star

def exact_cd_pair(h, n, max_passes, deadline, stride=1):
    h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(0, n - 1, stride):
            h_old = (h[p], h[p + 1])
            cd_pair(h, n, p, deadline)
            if abs(h[p] - h_old[0]) > 1e-12 or abs(h[p + 1] - h_old[1]) > 1e-12:
                improved = True
        h = project_feasible(h, n)
        if not improved:
            break
    return h, compute_c5(h, n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_opt(h, n, alphas, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': lambda z: z.sum() - n/2.0,
                                         'jac': lambda z: np.ones(n)}],
                           options={'maxiter': maxiter, 'ftol': 1e-20})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c - 1e-14:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def resample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return project_feasible(np.interp(x_new, x_old, h_old), n_new)

def multi_start_cd(h_best, n, best_c, deadline, rng, n_restarts, perturb_scale):
    for r in range(n_restarts):
        if time.time() > deadline:
            break
        s = perturb_scale * (1.0 + 0.3 * r)
        hp = project_feasible(h_best + rng.normal(0, s, n), n)
        h_o, c_o = exact_cd(hp, n, 500, deadline)
        if c_o < best_c:
            h_o, c_o = exact_cd_pair(h_o, n, 100, deadline, stride=1)
            if c_o < best_c:
                h_best, best_c = h_o, c_o
    return h_best, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 8
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = np.full(500, 0.5)
        n = 500
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)

    # Phase 1: SLSQP warmup with annealing
    if time.time() < deadline - 300:
        h1, c1 = slsqp_opt(h_best, n, [3000, 10000, 50000, 200000], 300, deadline)
        if c1 < best_c:
            h_best, best_c = h1, c1

    # Phase 2: Exact 1D CD
    if time.time() < deadline - 200:
        h2, c2 = exact_cd(h_best, n, 1000, deadline)
        if c2 < best_c:
            h_best, best_c = h2, c2

    # Phase 3: Pair CD with stride 1 (all overlapping pairs)
    if time.time() < deadline - 150:
        h3, c3 = exact_cd_pair(h_best, n, 300, deadline, stride=1)
        if c3 < best_c:
            h_best, best_c = h3, c2

    # Phase 4: Multi-n exploration
    n_list = [1000, 1400, 900, 600]
    for n_try in n_list:
        if time.time() > deadline - 150:
            break
        if abs(n_try - n) < 50:
            continue
        remaining = deadline - time.time()
        if remaining < 120:
            break
        h_int = resample(h_best, len(h_best), n_try)
        # Quick CD on new grid
        h_o, c_o = exact_cd(h_int, n_try, min(400, int(remaining / 2)), deadline)
        if c_o < best_c - 1e-8:
            h_o, c_o2 = exact_cd_pair(h_o, n_try, 150, deadline, stride=1)
            if c_o2 < c_o:
                h_o, c_o = h_o, c_o2
            if c_o < best_c:
                h_best, best_c, n = h_o, c_o, n_try

    # Phase 5: Multi-start with diverse perturbations
    if time.time() < deadline - 100:
        h_best, best_c = multi_start_cd(h_best, n, best_c, deadline, rng, 15, 0.008)

    # Phase 6: Refine with pair CD stride 1
    if time.time() < deadline - 60:
        h6, c6 = exact_cd_pair(h_best, n, 200, deadline, stride=1)
        if c6 < best_c:
            h_best, best_c = h6, c6

    # Phase 7: Final SLSQP polish
    if time.time() < deadline - 30:
        h7, c7 = slsqp_opt(h_best, n, [100000, 300000, 500000], 200, deadline)
        if c7 < best_c:
            h_best, best_c = h7, c7

    # Phase 8: Final CD
    if time.time() < deadline - 5:
        h8, c8 = exact_cd(h_best, n, 500, deadline)
        if c8 < best_c:
            h_best, best_c = h8, c8

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
