            # Pick from [x_lo, min(x_hi, r1)] or [max(x_lo, r2), x_hi]
            left_lo, left_hi = max(x_lo, 0.0), min(x_hi, r1)
            right_lo, right_hi = max(x_lo, r2), min(x_hi, 1.0)
            cands = []
            if left_hi >= left_lo - 1e-13:
                cands.append((left_lo + left_hi) / 2.0)
            if right_hi >= right_lo - 1e-13:
                cands.append((right_lo + right_hi) / 2.0)
            if cands:
                # Pick closest to h0
                return float(np.clip(min(cands, key=lambda x: abs(x - h0)), 0.0, 1.0))
            return h0
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def exact_cd(h, n, max_passes, deadline, rng=None):

    c = np.correlate(h, 1.0 - h, mode='full')
    for it in range(max_passes):

            break
        improved = False
        order = np.arange(n) if rng is None else rng.permutation(n)

                jb = p - ms + (n - 1)

                    a[vb] -= h[jb[vb]]

                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= delta * delta

            if abs(h.sum() - n / 2.0) > 1e-12:

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_opt(h, n, alphas, maxiter, deadline):

            break

        def _cons_fun(x):
            return x.sum() - n / 2.0
        def _cons_jac(x):
            return np.ones(n)

                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-16})

            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def make_binary_init(n, rng):

    idx = np.argsort(-weight)

    h[idx[:k]] = 1.0

def project_feasible(h, n):
