        x_lo, x_hi = 0.0, 1.0
        feasible = True
        if np.any(zero_mask) and np.any(c_base[zero_mask] > L + 1e-13):
            feasible = False
        if feasible and np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if feasible and np.any(pos_mask):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        if x_lo > x_hi + 1e-12:
            feasible = False
        if feasible and has_diag:
            Cq = L - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                xc_lo, xc_hi = max(x_lo, 0.0), min(x_hi, 1.0)
                if not (xc_lo < r1 and xc_hi > r2):
                    feasible = False
        if feasible:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Cq = L - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            xc_lo, xc_hi = max(x_lo, 0.0), min(x_hi, 1.0)
            if xc_lo < r1 and xc_hi > r2:
                if h0 - r1 <= r2 - h0:
                    x_hi = min(x_hi, r1)
                else:
                    x_lo = max(x_lo, r2)
            elif x_lo >= r1 and x_hi <= r2:
                if h0 - r1 <= r2 - h0:
                    x_hi = min(x_hi, r1)
                else:
                    x_lo = max(x_lo, r2)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def update_corr(c, h, p, old_val, new_val, n):
    delta = new_val - old_val
    if abs(delta) < 1e-14:
        return c
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    diag_i = np.where(ms == (n - 1))[0]
    c[ms] += delta * a
    if len(diag_i) > 0:
        c[ms[diag_i[0]]] -= delta * delta
    return c

def exact_cd(h, n, max_passes, deadline, rng=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for it in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        order = list(range(n))
        if rng is not None and it % 3 == 1:
            rng.shuffle(order)
        elif rng is not None and it % 3 == 2:
            order = order[::-1]
        for p in order:
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            xn = solve_1d(c, h, p, n)
            if abs(xn - h0) > 1e-11:
                update_corr(c, h, p, h0, xn, n)
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-9:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def smooth_lse(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_continuation(h, n, alphas, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        def cf(x):
            return x.sum() - n / 2.0
        def cj(x):
            return np.ones(n)
        cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]
        try:
            res = minimize(smooth_lse, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-17})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c - 1e-12:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def fourier_explore(h, n, n_iters, deadline, rng, best_c):
    h = h.copy()
    cur_c = best_c
    for it in range(n_iters):
        if time.time() > deadline:
            break
        scale = 0.02 * (1.0 - it / float(n_iters)) + 0.003
        H = np.fft.rfft(h)
        freqs = np.fft.rfftfreq(n)
        mask = freqs > 0
        noise = (rng.randn(len(H)[mask]) + 1j * rng.randn(len(H)[mask])) * scale
        H_new = H.copy()
        H_new[mask] += noise
        h_new = np.real(np.fft.irfft(H_new, n))
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < cur_c - 1e-10:
            h = h_new
            cur_c = c_new
    return h, cur_c

def interp_h(h_old, n_old, n_new):
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def greedy_order(n, c, h, rng):
    """Order coordinates by potential improvement (approximate)."""
    scores = np.zeros(n)
    for p in range(n):
        scores[p] = h[p] * (1.0 - h[p])
    order = np.argsort(-scores)
    if rng is not None and rng.rand() < 0.3:
        rng.shuffle(order)
    return list(order)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = np.full(500, 0.5)
        n = 500
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)

    # Phase 1: Heavy exact CD
    h1, c1 = exact_cd(h_best, n, 400, deadline, rng)
    if c1 < best_c:
        h_best, best_c = h1, c1

    # Phase 2: SLSQP continuation (smooth -> sharp)
    h2, c2 = slsqp_continuation(h_best, n, [100, 500, 2000, 10000, 50000, 200000], 300, deadline)
    if c2 < best_c - 1e-12:
        h_best, best_c = h2, c2

    # Phase 3: CD after SLSQP
    h3, c3 = exact_cd(h_best, n, 300, deadline, rng)
    if c3 < best_c - 1e-12:
        h_best, best_c = h3, c3

    # Phase 4: Fourier-domain exploration
    h4, c4 = fourier_explore(h_best, n, 200, deadline, rng, best_c)
    if c4 < best_c - 1e-10:
        h_best, best_c = h4, c3 if c3 < best_c else c4
        h_best, best_c = h4, c4
    # CD after Fourier
    h4b, c4b = exact_cd(h_best, n, 200, deadline, rng)
    if c4b < best_c - 1e-10:
        h_best, best_c = h4b, c4b

    # Phase 5: Multi-resolution exploration
    n_list = [600, 650, 700, 750, 800, 550, 450, 900, 1000]
    for n_try in n_list:
        if time.time() > deadline - 200:
            break
        h_int = interp_h(h_best, len(h_best), n_try)
        h_o, c_o = exact_cd(h_int, n_try, 250, deadline, rng)
        if c_o < best_c - 1e-9:
            h_best, best_c = h_o, c_o
        if time.time() < deadline - 100:
            h_o2, c_o2 = slsqp_continuation(h_o, n_try, [5000, 50000, 200000], 200, deadline)
            if c_o2 < best_c - 1e-9:
                h_best, best_c = h_o2, c_o2
            h_o3, c_o3 = exact_cd(h_o2, n_try, 150, deadline, rng)
            if c_o3 < best_c - 1e-9:
                h_best, best_c = h_o3, c_o3
        # Fourier at this resolution
        if time.time() < deadline - 80:
            h_o4, c_o4 = fourier_explore(h_best, len(h_best), 80, deadline, rng, best_c)
            if c_o4 < best_c - 1e-9:
                h_best, best_c = h_o4, c_o4
            h_o5, c_o5 = exact_cd(h_best, len(h_best), 100, deadline, rng)
            if c_o5 < best_c - 1e-9:
                h_best, best_c = h_o5, c_o5

    # Phase 6: Annealed restarts with CD
    n = len(h_best)
    for r in range(30):
        if time.time() > deadline - 60:
            break
        frac = float(r) / 30.0
        scale = 0.06 * (1.0 - frac) + 0.005
        hp = project_feasible(h_best + rng.normal(0, scale, n), n)
        h_o, c_o = exact_cd(hp, n, 100, deadline, rng)
        if c_o < best_c - 1e-10:
            h_best, best_c = h_o, c_o

    # Phase 7: Final long CD polish
    n = len(h_best)
    h_f, c_f = exact_cd(h_best, n, 500, deadline, rng)
    if c_f < best_c - 1e-13:
        h_best, best_c = h_f, c_f

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
