import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def solve_coord_fast(c, h, p, n):
    """Solve for optimal h[p] given current correlation c."""
    h0 = h[p]
    m_min = max(0, n - 1 - p)
    m_max = min(2 * n - 2, 2 * n - 2 - p)
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    
    c_base = c[ms].copy()
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    unaff = 0.0
    if m_min > 0:
        unaff = float(np.max(c[:m_min]))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    cur_max = float(np.max(c))
    
    # Bisection for optimal L
    L_lo, L_hi = unaff, cur_max
    for _ in range(40):
        if L_hi - L_lo < 1e-13:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        feasible = True
        # Linear constraints from forward/backward terms
        neg_mask = a < -1e-14
        pos_mask = a > 1e-14
        if np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if np.any(pos_mask):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        if x_lo > x_hi + 1e-12:
            feasible = False
        # Quadratic constraint from diagonal
        if feasible and has_diag:
            di = diag_idx[0]
            disc = a[di]**2 + 4.0 * (c_base[di] - L)
            if disc >= 0:
                sq = np.sqrt(disc)
                r1 = (a[di] - sq) / 2.0
                r2 = (a[di] + sq) / 2.0
                if x_lo > h0 + r1 and x_hi < h0 + r2:
                    feasible = False
        if feasible:
            L_hi = L
        else:
            L_lo = L
    
    L = L_hi
    # Find feasible x range
    x_lo, x_hi = 0.0, 1.0
    neg_mask = a < -1e-14
    pos_mask = a > 1e-14
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        di = diag_idx[0]
        disc = a[di]**2 + 4.0 * (c_base[di] - L)
        if disc >= 0:
            sq = np.sqrt(disc)
            r1 = (a[di] - sq) / 2.0
            r2 = (a[di] + sq) / 2.0
            if x_lo <= h0 + r1:
                x_hi = min(x_hi, h0 + r1)
            elif x_hi >= h0 + r2:
                x_lo = max(x_lo, h0 + r2)
            else:
                if abs(x_lo - (h0 + r1)) <= abs(x_hi - (h0 + r2)):
                    x_hi = min(x_hi, h0 + r1)
                else:
                    x_lo = max(x_lo, h0 + r2)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def cd_pass(h, n, order=None, deadline=None):
    """One full pass of coordinate descent."""
    c = np.correlate(h, 1.0 - h, mode='full')
    nm1 = n - 1
    if order is None:
        order = range(n)
    for p in order:
        if deadline is not None and time.time() > deadline:
            break
        h0 = h[p]
        if h0 < 1e-12 or h0 > 1.0 - 1e-12:
            continue
        xn = solve_coord_fast(c, h, p, n)
        if abs(xn - h0) > 1e-12:
            delta = xn - h0
            m_min = max(0, nm1 - p)
            m_max = min(2 * n - 2, 2 * n - 2 - p)
            ms = np.arange(m_min, m_max + 1)
            jf = p + ms - nm1
            jb = p - ms + nm1
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(vf):
                a[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                a[vb] -= h[jb[vb]]
            c[ms] += delta * a
            di = np.where(ms == nm1)[0]
            if len(di) > 0:
                c[ms[di[0]]] -= delta * delta
            h[p] = xn
    return h, c

def cd_optimize(h, n, max_passes, deadline, rng=None):
    """Run coordinate descent to convergence."""
    h = h.copy()
    for it in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        if rng is not None and it > 0 and it % 3 == 0:
            order = rng.permutation(n)
        else:
            order = range(n)
        h, c = cd_pass(h, n, order, deadline)
        h = project_feasible(h, n)
        # Check convergence
        c_new = np.correlate(h, 1.0 - h, mode='full')
        if np.max(np.abs(c_new - c)) < 1e-14:
            break
        c = c_new
    return h, compute_c5(h, n)

def targeted_perturbation(h, n, rng, n_top=10):
    """Perturb h to target the binding constraints."""
    c = np.correlate(h, 1.0 - h, mode='full')
    # Find top lags
    top_k = np.argsort(c)[-n_top:][::-1]  # largest first
    # For each top lag, compute gradient of c[k] w.r.t. h
    # c[k] = sum_i h[i] * (1-h[i+k-n+1]) (with boundary handling)
    # dc[k]/dh[p] = (1-h[p+k-n+1]) - h[p-k+n-1] (with appropriate bounds)

        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    cur_max = float(np.max(c))
    L_lo, L_hi = unaff, cur_max
    for _ in range(45):
        if L_hi - L_lo < 1e-13:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        feasible = True
        zm = np.abs(a) <= 1e-14
        if np.any(zm) and np.any(c_base[zm] > L + 1e-14):
            feasible = False
        nm = a < -1e-14
        pm = a > 1e-14
        if feasible and np.any(nm):
            x_lo = max(x_lo, float(np.max((L - c_base[nm]) / a[nm])))
        if feasible and np.any(pm):
            x_hi = min(x_hi, float(np.min((L - c_base[pm]) / a[pm])))
        if x_lo > x_hi + 1e-12:
            feasible = False
        if feasible and has_diag:
            di = diag_idx[0]
            disc = a[di] * a[di] + 4.0 * (c_base[di] - L)
            if disc >= 0:
                sq = np.sqrt(disc)
                r1 = (a[di] - sq) / 2.0
                r2 = (a[di] + sq) / 2.0
                if x_lo > h0 + r1 and x_hi < h0 + r2:
                    feasible = False
        if feasible:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    nm = a < -1e-14
    pm = a > 1e-14
    if np.any(nm):
        x_lo = max(x_lo, float(np.max((L - c_base[nm]) / a[nm])))
    if np.any(pm):
        x_hi = min(x_hi, float(np.min((L - c_base[pm]) / a[pm])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        di = diag_idx[0]
        disc = a[di] * a[di] + 4.0 * (c_base[di] - L)
        if disc >= 0:
            sq = np.sqrt(disc)
            r1 = (a[di] - sq) / 2.0
            r2 = (a[di] + sq) / 2.0
            if x_lo <= h0 + r1:
                x_hi = min(x_hi, h0 + r1)
            elif x_hi >= h0 + r2:
                x_lo = max(x_lo, h0 + r2)
            else:
                if abs(x_lo - (h0 + r1)) <= abs(x_hi - (h0 + r2)):
                    x_hi = min(x_hi, h0 + r1)
                else:
                    x_lo = max(x_lo, h0 + r2)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def cd_single(h, n, c, p):
    """Update coordinate p, return (new_h_p, delta)."""
    h0 = h[p]
    if h0 < 1e-12 or h0 > 1.0 - 1e-12:
        return h0, 0.0
    xn = solve_coord(c, h, p, n)
    return xn, xn - h0

def apply_update(c, h, p, delta, n):
    nm1 = n - 1
    m_min = max(0, nm1 - p)
    m_max = min(2 * n - 2, 2 * n - 2 - p)
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - nm1
    jb = p - ms + nm1
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c[ms] += delta * a
    di = np.where(ms == nm1)[0]
    if len(di) > 0:
        c[ms[di[0]]] -= delta * delta
    h[p] += delta

def cd_optimize(h, n, max_passes, deadline, rng=None, randomize=True):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for it in range(max_passes):
        if time.time() > deadline:
            break
        if randomize and rng is not None and it > 0:
            order = rng.permutation(n)
        else:
            order = np.arange(n)
        changed = False
        for p in order:
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            xn, delta = cd_single(h, n, c, p)
            if abs(delta) > 1e-12:
                apply_update(c, h, p, delta, n)
                changed = True
        if not changed:
            break
        h = project_feasible(h, n)
        c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def perturb_block_flip(h, n, rng):
    L = rng.randint(8, max(10, n // 5))
    start = rng.randint(0, n - L)
    h2 = h.copy()
    h2[start:start + L] = 1.0 - h2[start:start + L]
    return project_feasible(h2, n)

def perturb_gaussian(h, n, rng, sigma):
    return project_feasible(h + rng.normal(0, sigma, n), n)

def perturb_sinusoid(h, n, rng):
    freq = rng.randint(1, 15)
    amp = rng.uniform(0.08, 0.35)
    phase = rng.uniform(0, 2 * np.pi)
    x = np.linspace(0, 1, n)
    return project_feasible(h + amp * np.sin(2 * np.pi * freq * x + phase), n)

def perturb_sparse(h, n, rng):
    k = rng.randint(3, 15)
    idx = rng.choice(n, k, replace=False)
    h2 = h.copy()
    h2[idx] += rng.uniform(-0.5, 0.5, k)
    return project_feasible(h2, n)

def perturb_shift(h, n, rng):
    s = rng.randint(1, 6)
    return project_feasible(np.roll(h, s) + rng.normal(0, 0.02, n), n)

def interpolate_to_n(h_src, n_src, n_tgt):
    x_src = np.linspace(0, 2, n_src, endpoint=False)
    x_tgt = np.linspace(0, 2, n_tgt, endpoint=False)
    return project_feasible(np.interp(x_tgt, x_src, h_src), n_tgt)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = np.full(700, 0.5)
        n = 700

    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)

    # Phase 1: Initial CD convergence
    h_best, best_c = cd_optimize(h_best, n, 150, deadline, rng)

    # Phase 2: Diverse perturbation exploration
    pert_funcs = [perturb_block_flip, None, perturb_sinusoid, perturb_sparse, perturb_shift]
    n_rounds = 50
    for r in range(n_rounds):
        if time.time() > deadline - 80:
            break
        ptype = r % 5
        if ptype == 1:
            sigma = 0.04 + 0.04 * (r // 5)
            h_pert = perturb_gaussian(h_best, n, rng, sigma)
        else:
            pf = pert_funcs[ptype]
            h_pert = pf(h_best, n, rng)
        h_new, c_new = cd_optimize(h_pert, n, 60, deadline, rng)
        if c_new < best_c:
            h_best, best_c = h_new, c_new
            h_best, best_c = cd_optimize(h_best, n, 40, deadline, rng)

    # Phase 3: Try different n values
    n_candidates = sorted(set([n - 2, n + 2, n + 4, n - 4, n + 6, n + 8]))
    for n_try in n_candidates:
        if time.time() > deadline - 60:
            break
        if n_try < 200 or n_try > 1500 or n_try == n or n_try % 2 != 0:
            continue
        h_int = interpolate_to_n(h_best, n, n_try)
        h_new, c_new = cd_optimize(h_int, n_try, 80, deadline, rng)
        if c_new < best_c:
            h_best, best_c = h_new, c_new
            n = n_try

    # Phase 4: Second round of perturbations (finer)
    for r in range(30):
        if time.time() > deadline - 40:
            break
        ptype = r % 5
        if ptype == 1:
            h_pert = perturb_gaussian(h_best, n, rng, 0.02 + 0.02 * (r // 5))
        elif ptype == 0:
            h_pert = perturb_block_flip(h_best, n, rng)
        elif ptype == 2:
            h_pert = perturb_sinusoid(h_best, n, rng)
        elif ptype == 3:
            h_pert = perturb_sparse(h_best, n, rng)
        else:
            h_pert = perturb_shift(h_best, n, rng)
        h_new, c_new = cd_optimize(h_pert, n, 50, deadline, rng)
        if c_new < best_c:
            h_best, best_c = h_new, c_new
            h_best, best_c = cd_optimize(h_best, n, 30, deadline, rng)

    # Phase 5: Final long CD polish
    h_best, best_c = cd_optimize(h_best, n, 300, deadline, rng)

    # Phase 6: Quick final perturbations
    for r in range(8):
        if time.time() > deadline - 3:
            break
        h_pert = perturb_gaussian(h_best, n, rng, 0.015 + 0.01 * r)
        h_new, c_new = cd_optimize(h_pert, n, 30, deadline, rng)
        if c_new < best_c:
            h_best, best_c = h_new, c_new

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
