def project_feasible(x,n):
    x=np.asarray(x,dtype=float).copy()
    if x.size != n:
       x=np.resize(x,n)
    if not np.all(np.isfinite(x)): x=np.full(n,0.5)
    target=0.5*n
    s=x.sum()
    if abs(s-target) < 1e-15 and np.all((x>=-1e-15)&(x<=1+1e-15)):
       return np.clip(x,0,1)
    lo=-float(np.max(x))-1.0
    hi=1.0-float(np.min(x))+1.0
    for _ in range(70):
       mid=0.5*(lo+hi)
       y=np.clip(x+mid,0.0,1.0)
       sm=float(y.sum())
       if sm < target: lo=mid
       else: hi=mid
    y=np.clip(x+0.5*(lo+hi),0.0,1.0)
    # small correction to interior
    resid=target-float(y.sum())
    if abs(resid)>0 and np.any((y>1e-12)&(y<1-1e-12)):
       mask=(y>1e-12)&(y<1-1e-12)
       y[mask]+=resid/float(mask.sum())
       y=np.clip(y,0,1.0)
    return y

import numpy as np
import time

def project_feasible(x, n):
    x = np.asarray(x, dtype=float)
    if x.size != n:
        x = np.resize(x, n).copy()
    else:
        x = x.copy()
    if not np.all(np.isfinite(x)):
        return np.full(n, 0.5)
    target = 0.5 * n
    if abs(float(x.sum()) - target) < 1e-15 and np.all((x >= -1e-15) & (x <= 1.0 + 1e-15)):
        return np.clip(x, 0.0, 1.0)
    lo = -float(np.max(x)) - 1.0
    hi = 1.0 - float(np.min(x)) + 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        y = np.clip(x + mid, 0.0, 1.0)
        if float(y.sum()) < target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    resid = target - float(y.sum())
    if abs(resid) > 0.0:
        mask = (y > 1e-12) & (y < 1.0 - 1e-12)
        if np.any(mask):
            y[mask] += resid / float(mask.sum())
            y = np.clip(y, 0.0, 1.0)
    return y

def compute_c5(h, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * (2.0 / float(n)))

def solve_coord(c_sub, a, h0, high, diag_idx):
    pos = a > 1e-14
    neg = a < -1e-14
    zero = ~(pos | neg)
    zero[diag_idx] = False
    pos_any = bool(np.any(pos))
    neg_any = bool(np.any(neg))
    zero_any = bool(np.any(zero))
    if pos_any:
        c_pos = c_sub[pos]
        a_pos = a[pos]
    if neg_any:
        c_neg = c_sub[neg]
        a_neg = a[neg]
    c_zero_max = float(np.max(c_sub[zero])) if zero_any else -1e300
    c_diag = float(c_sub[diag_idx])
    high = max(float(high), float(np.max(c_sub)))
    low = 0.0
    h0m1 = h0 * (1.0 - h0)
    for _ in range(42):
        if high - low < 1e-11:
            break
        mid = 0.5 * (low + high)
        tlo = -h0
        thi = 1.0 - h0
        if pos_any:
            v = float(np.min((mid - c_pos) / a_pos))
            if v < thi:
                thi = v
        if neg_any:
            v = float(np.max((mid - c_neg) / a_neg))
            if v > tlo:
                tlo = v
        ok = True
        if zero_any and c_zero_max > mid + 1e-11:
            ok = False
        if ok and (thi < -h0 - 1e-12 or tlo > 1.0 - h0 + 1e-12 or tlo > thi + 1e-12):
            ok = False
        if ok:
            xlo = tlo + h0
            xhi = thi + h0
            if xlo < 0.0:
                xlo = 0.0
            if xhi > 1.0:
                xhi = 1.0
            if xlo > xhi + 1e-12:
                ok = False
            else:
                D = mid - c_diag + h0m1
                if D < -1e-11:
                    ok = False
                elif D < 0.25:
                    disc = 1.0 - 4.0 * D
                    if disc < 0.0:
                        disc = 0.0
                    rt = np.sqrt(disc)
                    r1 = 0.5 * (1.0 - rt)
                    r2 = 0.5 * (1.0 + rt)
                    if not (xhi <= r1 + 1e-11 or xlo >= r2 - 1e-11):
                        ok = False
        if ok:
            high = mid
        else:
            low = mid
    L = high
    tlo = -h0
    thi = 1.0 - h0
    if pos_any:
        v = float(np.min((L - c_pos) / a_pos))
        if v < thi:
            thi = v
    if neg_any:
        v = float(np.max((L - c_neg) / a_neg))
        if v > tlo:
            tlo = v
    xlo = tlo + h0
    xhi = thi + h0
    if xlo < 0.0:
        xlo = 0.0
    if xhi > 1.0:
        xhi = 1.0
    if xlo > xhi:
        xlo, xhi = min(xlo, xhi), max(xlo, xhi)
    D = L - c_diag + h0m1
    if D < 0.25:
        disc = 1.0 - 4.0 * D
        if disc < 0.0:
            disc = 0.0
        rt = np.sqrt(disc)
        r1 = 0.5 * (1.0 - rt)
        r2 = 0.5 * (1.0 + rt)
        bx = float(np.clip(h0, 0.0, 1.0))
        bd = 1e100
        al = max(xlo, 0.0)
        bl = min(xhi, r1)
        if al <= bl + 1e-12:
            cand = al if h0 < al else (bl if h0 > bl else h0)
            d1 = abs(cand - h0)
            if d1 < bd:

            d1 = abs(cand - h0)
            if d1 < bd:
                bx = cand
    else:
        bx = float(np.clip(h0, xlo, xhi))
    return float(np.clip(bx, 0.0, 1.0))

def coordinate_update(h, c, p, n, max_bound):
    h0 = h[p]
    m_l = p if p < n - 1 - p else n - 1 - p
    m_r = p + n - 1 if p + n - 1 > 2 * n - 2 - p else 2 * n - 2 - p
    d = np.arange(m_l, m_r + 1) - (n - 1)
    a = np.zeros(m_r - m_l + 1, dtype=float)
    jb = p - d
    mh = (jb >= 0) & (jb < n)
    if np.any(mh):
        a[mh] += 1.0 - h[jb[mh]]
    jf = p + d
    ml = (jf >= 0) & (jf < n)
    if np.any(ml):
        a[ml] -= h[jf[ml]]
    diag_idx = (n - 1) - m_l
    a[diag_idx] = 0.0
    xn = solve_coord(c[m_l:m_r + 1], a, h0, max_bound, diag_idx)
    if abs(xn - h0) > 1e-13:
        delta = xn - h0
        c[m_l:m_r + 1] += delta * a
        c[m_l + diag_idx] += xn - xn * xn - h0 + h0 * h0
        h[p] = xn
        lm = float(np.max(c[m_l:m_r + 1]))
        if lm > max_bound:
            max_bound = lm
    return max_bound

def cd_refine(h, n, deadline, init_chunk=None):
    h = project_feasible(h, n)
    c = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / float(n)
    best_cmax = float(np.max(c))
    best_c = best_cmax * dx
    best_h = h.copy()
    cur_h = h.copy()
    cur_c = c.copy()
    cur_max = best_cmax
    if init_chunk is None:
        init_chunk = max(8, min(n, n // 8))
    chunk = int(init_chunk)
    no_progress = 0
    while time.time() < deadline:
        progressed = False
        for start in range(0, n, chunk):
            if time.time() > deadline:
                break
            hs = cur_h.copy()
            cs = cur_c.copy()
            ms = cur_max
            end = min(n, start + chunk)
            mb = cur_max
            last_p = start - 1
            for p in range(start, end):
                if time.time() > deadline:
                    break
                last_p = p
                mb = coordinate_update(cur_h, cur_c, p, n, mb)
                if (p - start) % 16 == 15:
                    mb = float(np.max(cur_c))
            if last_p >= start:
                ph = project_feasible(cur_h, n)
                pc = np.correlate(ph, 1.0 - ph, mode='full')
                pcmax = float(np.max(pc))
                if pcmax < best_cmax - 1e-12:
                    best_h = ph.copy()
                    best_cmax = pcmax
                    best_c = pcmax * dx
                    cur_h = ph
                    cur_c = pc
                    cur_max = pcmax
                    progressed = True
                else:
                    cur_h = hs
                    cur_c = cs
                    cur_max = ms
        if time.time() > deadline:
            break
        if progressed:
            no_progress = 0
            med = max(8, n // 10)
            if chunk > med:
                chunk = med
        else:
            no_progress += 1
            if chunk < n:
                chunk = min(n, chunk * 2)
            elif no_progress >= 3:
                break
    return best_h, float(best_c)

def interpolate_h(old, n):
    n0 = len(old)
    if n0 == n:
        return old.copy()
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
    x1 = np.linspace(0.0, 2.0, n, endpoint=False)
    return project_feasible(np.interp(x1, x0, old), n)

def scale_h(h, s):
    return project_feasible(np.clip(0.5 + s * (h - 0.5), 0.0, 1.0), len(h))

def contrast_h(h, lam):
    return project_feasible(0.5 + 0.5 * np.tanh(lam * (h - 0.5)), len(h))

def binary_push_h(h, st):
    x = h.copy()
    x[x < 0.5] -= st
    x[x >= 0.5] += st
    return project_feasible(np.clip(x, 0.0, 1.0), len(x))

def perturb_h(h, sg, rng):
    return project_feasible(h + rng.normal(0.0, sg, len(h)), len(h))

def make_fallback(n):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = 0.5 + 0.35 * np.sin(np.pi * x) + 0.18 * np.sin(2.0 * np.pi * x + 0.5)
    return project_feasible(np.clip(h, 0.0, 1.0), n)

def make_ns(n):
    out = []
    for f in (0.5, 0.6666667, 0.75, 0.8, 1.0, 1.1, 1.25, 1.5):
        m = int(round(n * f))
        if 50 <= m <= 1000 and m not in out:
            out.append(m)
    for m in (80, 120, 160, 200, 250, 300, 400, 500, 600, 700, 800, 900, 999):
        if m not in out:
            out.append(m)
    return out

def optimize_candidate(h, n, end_deadline, best_h, best_c, best_n, over_thresh, init_chunk):
    h = project_feasible(h, n)
    c = compute_c5(h, n)
    if c < best_c:
        best_h, best_c, best_n = h, c, n
    if c <= best_c + over_thresh and time.time() < end_deadline:
        h2, c2 = cd_refine(h, n, end_deadline, init_chunk)
        if c2 < best_c:
            best_h, best_c, best_n = h2, c2, n
    return best_h, best_c, best_n
