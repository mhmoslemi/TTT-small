    diag_i = np.where(ms==(n-1))[0]
    c[ms] += delta*a
    if len(diag_i)>0: c[ms[diag_i[0]]] -= delta*delta
    return c

def exact_cd(h, n, max_passes, deadline, rng=None):
    h = h.copy()
    c = np.correlate(h, 1.0-h, mode='full')
    for it in range(max_passes):
        if time.time() > deadline: break
        improved = False
        order = list(range(n))
        if rng is not None and it % 3 == 2:
            rng.shuffle(order)
        for p in order:
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0-1e-13: continue
            xn = solve_1d(c, h, p, n)
            if abs(xn-h0) > 1e-12:
                update_corr(c, h, p, h0, xn, n)
                h[p] = xn
                improved = True
        if not improved: break
        h = project_feasible(h, n)
        if abs(h.sum()-n/2.0) > 1e-10:
            c = np.correlate(h, 1.0-h, mode='full')
    return h, compute_c5(h, n)

def subgrad_step(h, n, eta):
    """Take a subgradient step on the max correlation."""
    c = np.correlate(h, 1.0-h, mode='full')
    m_star = int(np.argmax(c))
    # Subgradient w.r.t. h[p]
    sg = np.zeros(n)
    offset = m_star - (n-1)
    # Forward: h[p] contributes (1-h[p+offset]) if valid
    for p in range(n):
        j = p + offset
        if 0 <= j < n:
            sg[p] += (1.0 - h[j])
        i = p - offset
        if 0 <= i < n:
            sg[p] -= h[i]
    h_new = project_feasible(h - eta * sg, n)
    return h_new

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0-x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted-mx)))
    return (m + lse/alpha) * (2.0/n)

def slsqp_opt(h, n, alphas, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline: break
        bounds = [(0.0, 1.0)]*n
        def cf(x): return x.sum() - n/2.0
        def cj(x): return np.ones(n)
        cons = [{'type':'eq', 'fun':cf, 'jac':cj}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter':maxiter, 'ftol':1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c - 1e-14:
                best_h, best_c = new_h, new_c
        except: pass
    return best_h, best_c

def interp_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return project_feasible(np.interp(x_new, x_old, h_old), n_new)
