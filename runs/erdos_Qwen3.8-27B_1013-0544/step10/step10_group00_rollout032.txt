    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def fast_cd(h, n, max_passes, deadline, gs_iters=45, recompute_every=5):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full').astype(float).copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    pass_count = 0
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = 0
        for p in range(n):
            h0 = h[p]
            if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            a = np.zeros(len(ms))
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            valid_f = (jf >= 0) & (jf < n) & (ms != (n - 1))
            valid_b = (jb >= 0) & (jb < n) & (ms != (n - 1))
            if np.any(valid_f):
                a[valid_f] += (1.0 - h[jf[valid_f]])
            if np.any(valid_b):
                a[valid_b] -= h[jb[valid_b]]
            diag_pos = np.where(ms == (n - 1))[0]
            has_diag = len(diag_pos) > 0
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            c_aff = c[ms]
            d_lo, d_hi = -h0, 1.0 - h0
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            f1 = max(float(np.max(c_aff + d1 * a)), unaff_max)
            if has_diag:
                di = int(diag_pos[0])
                f1 = max(f1, float(c_aff[di] + d1 * (1.0 - h0) - d1 * d1))
            f2 = max(float(np.max(c_aff + d2 * a)), unaff_max)
            if has_diag:
                f2 = max(f2, float(c_aff[di] + d2 * (1.0 - h0) - d2 * d2))
            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-11:
                    break
                if f1 < f2:
                    d_hi, d2, f2 = d2, d1, f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    tmp = max(float(np.max(c_aff + d1 * a)), unaff_max)
                    if has_diag:
                        tmp = max(tmp, float(c_aff[di] + d1 * (1.0 - h0) - d1 * d1))
                    f1 = tmp
                else:
                    d_lo, d1, f1 = d1, d2, f2
                    d2 = d_lo + gr * (d_hi - d_lo)
                    tmp = max(float(np.max(c_aff + d2 * a)), unaff_max)
                    if has_diag:
                        tmp = max(tmp, float(c_aff[di] + d2 * (1.0 - h0) - d2 * d2))
                    f2 = tmp
            d_opt = 0.5 * (d_lo + d_hi)
            f_opt = max(float(np.max(c_aff + d_opt * a)), unaff_max)
            if has_diag:
                f_opt = max(f_opt, float(c_aff[di] + d_opt * (1.0 - h0) - d_opt * d_opt))
            f_curr = max(float(np.max(c_aff)), unaff_max)
            if has_diag:
                f_curr = max(f_curr, float(c_aff[di]))
            if f_opt < f_curr - 1e-12:
                h[p] = h0 + d_opt
                c[ms] = c_aff + d_opt * a
                if has_diag:
                    c[ms[di]] = c_aff[di] + d_opt * (1.0 - h0) - d_opt * d_opt
                improved += 1
        pass_count += 1
        if improved == 0:
            break
        if pass_count % recompute_every == 0:
            h = project_feasible(h, n)
            c = np.correlate(h, 1.0 - h, mode='full').astype(float)
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * dx

def _sum_eq(x, n):
    return x.sum() - n / 2.0

def _sum_eq_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alphas, maxiter_each=200):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac, 'args': (n,)}]
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-15})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def subgradient_phase(h, n, deadline, max_iters=3000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    mom = np.zeros(n)
    beta = 0.6
    step = 0.1
    for it in range(max_iters):
        if time.time() > deadline:
            break
        step = max(0.002, 0.1 / (1.0 + 0.02 * it))
        c = np.correlate(h, 1.0 - h, mode='full')
        m_star = int(np.argmax(c))
        s = m_star - (n - 1)
        grad = np.zeros(n)
        for j in range(n):
            g = 0.0
            if 0 <= j + s < n:
                g += (1.0 - h[j + s])
            if 0 <= j - s < n:
                g -= h[j - s]
            if s == 0:
                g = (1.0 - 2.0 * h[j])
            grad[j] = g
        mom = beta * mom + (1.0 - beta) * grad
        h = project_feasible(h - step * mom, n)
        c_new = compute_c5(h, n)
        if c_new < best_c:
            best_h, best_c = h.copy(), c_new
    return best_h, best_c

def binary_threshold_attempt(h, n, deadline):
    """Try to improve by pushing toward binary values."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for t in np.linspace(0.3, 0.7, 9):
        if time.time() > deadline:
            break
        hb = np.where(h > t, 1.0, 0.0)
        hb = project_feasible(hb, n)
        hc, cc = fast_cd(hb, n, 20, deadline, gs_iters=30)
        if cc < best_c:
            best_h, best_c = hc, cc
    return best_h, best_c

def multi_scale(h_best, n_best, deadline):
    """Try different resolutions."""
    x_orig = np.linspace(0, 2, n_best)
    for n_try in [400, 500, 600, 800, 1000, 1200]:
        if time.time() > deadline - 60:
            break
        if abs(n_try - n_best) < 50:
            continue
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, h_best), n_try)
        h_o, c_o = fast_cd(h_int, n_try, 40, deadline - 60, gs_iters=45)
        if c_o < compute_c5(h_best, n_best):
            h_o, c_o = slsqp_optimize(h_o, n_try, [5000, 20000], 200)
            h_o, c_o = fast_cd(h_o, n_try, 30, deadline - 60, gs_iters=50)
            if c_o < compute_c5(h_best, n_best):
                h_best, n_best = h_o, n_try
                x_orig = np.linspace(0, 2, n_best)
    return h_best, n_best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 500
        h_init = np.full(n, 0.5)

    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: Fast CD
    if time.time() < deadline - 300:
        h1, c1 = fast_cd(best_h, n, 150, deadline - 300, gs_iters=45)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Phase 2: SLSQP polish
    if time.time() < deadline - 250:
        h2, c2 = slsqp_optimize(best_h, n, [2000, 10000, 50000], 300)
        if c2 < best_c:
            best_h, best_c = h2, c2
        h2, c2 = fast_cd(h2, n, 50, deadline - 250, gs_iters=45)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Phase 3: Subgradient escape
    if time.time() < deadline - 200:
        h3, c3 = subgradient_phase(best_h, n, deadline - 200, max_iters=2000)
        if c3 < best_c:
            best_h, best_c = h3, c3
        h3, c3 = fast_cd(h3, n, 60, deadline - 200, gs_iters=45)
        if c3 < best_c:
            best_h, best_c = h3, c3

    # Phase 4: Binary threshold
    if time.time() < deadline - 150:
        h4, c4 = binary_threshold_attempt(best_h, n, deadline - 150)
        if c4 < best_c:
            best_h, best_c = h4, c4
        h4, c4 = fast_cd(h4, n, 40, deadline - 150, gs_iters=45)
        if c4 < best_c:
            best_h, best_c = h4, c4

    # Phase 5: Multi-scale
    if time.time() < deadline - 120:
        best_h, n = multi_scale(best_h, n, deadline - 120)
        best_c = compute_c5(best_h, n)

    # Phase 6: Multi-restart with structured perturbations
    if time.time() < deadline - 80:
        n_cur = len(best_h)
        x = np.linspace(0, 2, n_cur)
        remaining = deadline - time.time() - 80
        n_restarts = max(6, int(remaining / 30))
        for r in range(min(n_restarts, 20)):
            if time.time() > deadline - 80:
                break
            scale = 0.005 + 0.03 * (r / max(1, n_restarts - 1))
            if r % 5 == 0:
                hp = project_feasible(best_h + rng.normal(0, scale, n_cur), n_cur)
            elif r % 5 == 1:
                freq = 3 + r
                hp = project_feasible(best_h + scale * 2 * np.sin(2 * np.pi * freq * x / 2.0), n_cur)
            elif r % 5 == 2:
                block = n_cur // 5
                hp = best_h.copy()
                start = (r * block * 3) % max(1, n_cur - block)
                hp[start:start + block] = 1.0 - hp[start:start + block]
                hp = project_feasible(hp, n_cur)
            elif r % 5 == 3:
                hp = project_feasible(np.roll(best_h, r * 13, axis=0), n_cur)
            else:
                hp = project_feasible(np.clip(best_h + scale * rng.randn(n_cur), 0, 1), n_cur)
            h_o, c_o = fast_cd(hp, n_cur, 30, deadline - 80, gs_iters=40)
            if c_o < best_c:
                best_h, best_c = h_o, c_o

    # Phase 7: Final polish
    if time.time() < deadline - 15:
        n_cur = len(best_h)
        h_f, c_f = slsqp_optimize(best_h, n_cur, [10000, 50000, 200000], 400)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
        h_f, c_f = fast_cd(best_h, n_cur, 80, deadline - 15, gs_iters=55)
        if c_f < best_c:
            best_h, best_c = h_f, c_f

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
