```python
def subgradient_step(h, n, step_size, top_k=10):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    # Find top_k lags
    idx = np.argsort(c)[-top_k:]
    active_weights = np.zeros(len(c))
    active_weights[idx] = 1.0 / len(idx)
    
    grad = np.zeros(n)
    for k in idx:
        lag = k - (n - 1)
        for p in range(n):
            if 0 <= p - lag < n:
                grad[p] += active_weights[k] * (1.0 - h[p - lag])
            if 0 <= p + lag < n:
                grad[p] -= active_weights[k] * h[p + lag]
    
    h_new = h - step_size * grad
    h_new = project_feasible(h_new, n)
    return h_new, m
```

```python
import cvxpy as cp

def qp_step(h, n, top_k=20, max_iter=100, tol=1e-8):
    """Second-order cone / QP step using linearization."""
    h = h.copy()
    for iter_i in range(max_iter):
        c_arr = np.correlate(h, 1.0 - h, mode='full')
        m = c_arr.max()
        c5 = m * 2.0 / n
        
        # Top k lags
        idx = np.argsort(c_arr)[-top_k:]
        
        # Compute gradients for these lags
        grads = np.zeros((len(idx), n))
        for ii, k in enumerate(idx):
            lag = k - (n - 1)
            if lag >= 0:
                if n - lag > 0:
                    grads[ii, lag:n] += (1.0 - h[0:n-lag])
                if n - 1 - lag >= 0:
                    grads[ii, 0:n-lag] -= h[lag:n]
            else:
                lag2 = -lag
                if n - lag2 > 0:
                    grads[ii, 0:n-lag2] += (1.0 - h[lag2:n])
                grads[ii, lag2:n] -= h[0:n-lag2]
        
        # Solve QP:
        # min t
        # s.t. c[k_i] + grad_i . (h' - h) <= t/dx  for each i
        #      sum(h') = n/2
        #      0 <= h' <= 1
        dx = 2.0 / n
        z = cp.Variable(n)
        t = cp.Variable()
        constraints = [0 <= z, z <= 1, cp.sum(z) == n / 2.0]
        residual = c_arr[idx] - np.dot(grads, h)  # c[k_i] - grad_i . h
        # c[k_i] + grad_i . (z - h) = c[k_i] + grad_i . z - grad_i . h = residual[i] + grad_i . z
        for i in range(len(idx)):
            constraints.append(residual[i] + grads[i] @ z <= t / dx)
        obj = cp.Minimize(t)
        prob = cp.Problem(obj, constraints)
        prob.solve(solver=cp.ECOS)  # or SCS, or other
        if z.value is None:
            break
        new_h = np.clip(z.value, 0, 1)
        # Project to fix sum constraint
        new_h = project_feasible(new_h, n)
        new_c5 = compute_c5(new_h, n)
        if new_c5 < c5 - tol:
            h = new_h
        else:
            break
    return h, compute_c5(h, n)
```

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def lag_gradient(h, n, k):
    """Compute gradient of c[k] w.r.t. h."""
    lag = k - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        if n - lag > 0:
            grad[lag:n] += (1.0 - h[0:n-lag])
        if n - 1 - lag >= 0:
            grad[0:n-lag] -= h[lag:n]
    else:
        lag2 = -lag
        if n - lag2 > 0:
            grad[0:n-lag2] += (1.0 - h[lag2:n])
        grad[lag2:n] -= h[0:n-lag2]
    return grad

def cd_1d(c, h, p, n):
    h0 = h[p]
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms].copy()
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    if has_diag:
        c_diag = c_base[diag_idx[0]]
        a[diag_idx] = 0.0
    unaff = 0.0
    if m_min > 0:
        unaff = max(unaff, float(np.max(c[:m_min])))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    cur_max = float(np.max(c))
    neg_mask = a < -1e-15
    pos_mask = a > 1e-15
    zero_mask = np.abs(a) <= 1e-15
    L_lo, L_hi = unaff, cur_max
    for _ in range(45):
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        ok = True
        if np.any(zero_mask) and np.any(c_base[zero_mask] > L + 1e-15):
            ok = False
        if ok and np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if ok and np.any(pos_mask):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        if x_lo > x_hi + 1e-14:
            ok = False
        if ok and has_diag:
            Cq = L - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                if not (max(x_lo, 0.0) < r1 and min(x_hi, 1.0) > r2):
                    ok = False
        if ok:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Cq = L - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            if max(x_lo, 0.0) < r1 and min(x_hi, 1.0) > r2:
                x_hi = min(x_hi, r1) if h0 - r1 <= r2 - h0 else max(x_lo, r2)
                x_lo = min(x_lo, r2) if h0 - r1 <= r2 - h0 else max(x_lo, r2)
            elif x_lo >= r1 and x_hi <= r2:
                x_hi = min(x_hi, r1) if h0 - r1 <= r2 - h0 else max(x_lo, r2)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def exact_cd(h, n, max_passes, deadline):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            xn = cd_1d(c, h, p, n)
            if abs(xn - h0) > 1e-12:
                delta = xn - h0
                m_min = n - 1 - p
                m_max = 2 * n - 2 - p
                ms = np.arange(m_min, m_max + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                diag_i = np.where(ms == (n - 1))[0]
                c[ms] += delta * a
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True
        h = project_feasible(h, n)
        if not improved:
            break
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def slp_optimize(h_init, n, deadline, top_k=30, max_iter=200):
    """Successive Linear Programming (cutting plane) method."""
    h = h_init.copy()
    dx = 2.0 / n
    cuts = []  # (residual, grad) pairs
    best_h, best_c5 = h.copy(), compute_c5(h, n)
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        c_arr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = c_arr.max() * dx
        
        if c5_now < best_c5:
            best_h, best_c5 = h.copy(), c5_now
        
        # Add cuts for top k lags
        idx = np.argsort(c_arr)[-top_k:]
        for k in idx:
            lag = k - (n - 1)
            grad = lag_gradient(h, n, k)
            residual = c_arr[k] - np.dot(grad, h)
            cuts.append((residual, grad))
        
        # Keep only recent cuts
        if len(cuts) > 60:
            cuts = cuts[-40:]
        
        # Solve LP: min t s.t. residual_i + grad_i · z ≤ t/dx, 0≤z≤1, sum(z)=n/2
        # Variables: [z (n), t] -> total n+1
        m_cuts = len(cuts)
        # A_ub: m_cuts x (n+1), b_ub: m_cuts
        A_ub = np.zeros((m_cuts, n + 1))
        b_ub = np.zeros(m_cuts)
        for i, (res, grad) in enumerate(cuts):
            A_ub[i, :n] = grad
            A_ub[i, n] = -1.0 / dx
            b_ub[i] = -res
        
        # A_eq: 1 x (n+1), b_eq: 1
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        
        # Objective: minimize t
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        
        bounds = [(0, 1)] * n + [(0, None)]
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
        except Exception:
            break
        
        if res.status != 0:
            try:
                res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                             bounds=bounds, method='revised simplex')
            except Exception:
                break
        
        if res.status != 0 or res.x is None:
            break
        
        z_new = np.clip(res.x[:n], 0, 1)
        t_new = res.x[n]
        z_new = project_feasible(z_new, n)
        c5_new = compute_c5(z_new, n)
        
        # Check gap
        gap = c5_new - t_new
        if gap < 1e-7:
            h = z_new
            best_h, best_c5 = h.copy(), c5_new
            break
        
        # Move to new point (damped if needed)
        h = z_new
    
    return best_h, best_c5
```

```python
def all_lag_gradients(h, n):
    """Compute gradients of all c[k] w.r.t. h. Returns (n,) x (2n-1,) array."""
    grads = np.zeros((2*n-1, n))

    return grads
```
