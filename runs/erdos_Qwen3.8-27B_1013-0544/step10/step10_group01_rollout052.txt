def project_polytope(z, n):
    """Project z onto {h: 0≤h≤1, sum(h)=n/2}"""
    target = n / 2.0
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h = np.clip(z - mid, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(z - (lo+hi)/2.0, 0.0, 1.0)
    # Fix any remaining error
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        # Distribute to non-bound components
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() > 0:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def smooth_val_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = c.max()
    exp_c = np.exp(alpha * (c - c_max))
    w = exp_c / exp_c.sum()
    f = (c_max + np.log(exp_c.sum()) / alpha) * (2.0 / n)
    grad = np.zeros(n)
    omh = 1.0 - h
    for idx in range(2 * n - 1):
        s = idx - (n - 1)
        wi = w[idx]
        if wi < 1e-18:
            continue
        if s >= 0:
            grad[:n-s] += wi * omh[s:]
            grad[s:] -= wi * h[:n-s]
        else:
            grad[-s:] += wi * omh[:n+s]
            grad[:n+s] -= wi * h[-s:]
    grad *= (2.0 / n)
    return f, grad

def proj_gradient(h, n, alpha, max_iters, deadline, lr=None):
    h = h.copy()
    if lr is None:
        lr = 1.0 / (2.0 * alpha)
    f_curr, grad = smooth_val_grad(h, n, alpha)
    gnorm2 = float(np.dot(grad, grad))
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if gnorm2 < 1e-20:
            break
        step = lr
        h_new = h
        f_new = f_curr
        for _ in range(40):
            h_trial = project_feasible(h - step * grad, n)
            f_trial, _ = smooth_val_grad(h_trial, n, alpha)
            if f_trial <= f_curr - 1e-5 * step * gnorm2:
                h_new = h_trial
                f_new = f_trial
                break
            step *= 0.5
        h = h_new
        lr = step * 1.4
        f_curr, grad = smooth_val_grad(h, n, alpha)
        gnorm2 = float(np.dot(grad, grad))
    return h

def _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d):
    vals = c_aff + d * a
    if has_diag:
        vals[diag_idx] = c_aff[diag_idx] + d * (1.0 - h0 - d)
    result = float(np.max(vals))
    if unaff_max > result:
        result = unaff_max
    return result

def coord_descent(h, n, max_passes, deadline, gs_iters=55, threshold=1e-7):
    h = h.copy()
    best_c = compute_c5(h, n)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-13 or h0 >= 1.0 - 1e-13:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(valid_f):
                a[valid_f] += (1.0 - h[jf[valid_f]])
            if np.any(valid_b):
                a[valid_b] -= h[jb[valid_b]]
            diag_pos = np.where(ms == (n - 1))[0]
            has_diag = len(diag_pos) > 0
            diag_idx = int(diag_pos[0]) if has_diag else -1
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            d_lo = -h0
            d_hi = 1.0 - h0
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
            f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-15:
                    break
                if f1 < f2:
                    d_hi = d2; d2 = d1; f2 = f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
                else:
                    d_lo = d1; d1 = d2; f1 = f2
                    d2 = d_lo + gr * (d_hi - d_lo)
                    f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            d_opt = (d_lo + d_hi) / 2.0
            f_opt = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d_opt)
            f_curr = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, 0.0)
            if f_opt < f_curr - 1e-15:
                h[p] = h0 + d_opt
                improved = True
        if improved:
            h[np.where(h < threshold)[0]] = 0.0
            h[np.where(h > 1.0 - threshold)[0]] = 1.0
            h = project_feasible(h, n)
            c_new = compute_c5(h, n)
            if c_new < best_c:
                best_c = c_new
        else:
            break
    final_c = compute_c5(h, n)
    return h, min(best_c, final_c)

    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:

    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
