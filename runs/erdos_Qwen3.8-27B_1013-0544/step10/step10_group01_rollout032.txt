            d_lo = -h0
            d_hi = 1.0 - h0
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            f1 = cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
            f2 = cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-16:
                    break
                if f1 < f2:
                    d_hi = d2; d2 = d1; f2 = f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    f1 = cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
                else:
                    d_lo = d1; d1 = d2; f1 = f2
                    d2 = d_lo + gr * (d_hi - d_lo)
                    f2 = cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            d_opt = (d_lo + d_hi) / 2.0
            f_opt = cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d_opt)
            f_curr = cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, 0.0)
            if f_opt < f_curr - 1e-15:
                h[p] = h0 + d_opt
                improved = True
        if improved:
            h[h < threshold] = 0.0
            h[h > 1.0 - threshold] = 1.0
            h = project_feasible(h, n)
            c_new = compute_c5(h, n)
            if c_new < best_c:
                best_c = c_new
        else:
            break
    return h, min(best_c, compute_c5(h, n))

def binary_sa(h, n, max_iters, deadline, rng, T_init=0.005, T_final=0.0005):
    """Simulated annealing on binary solutions via swap moves."""
    h_bin = (h > 0.5).astype(float)
    ones_count = int(h_bin.sum())
    target = n // 2
    if ones_count > target:
        idx1 = np.where(h_bin == 1)[0]
        order = np.argsort(h[idx1])[:ones_count - target]
        h_bin[idx1[order]] = 0
    elif ones_count < target:
        idx0 = np.where(h_bin == 0)[0]
        order = np.argsort(-h[idx0])[:target - ones_count]
        h_bin[idx0[order]] = 1
    
    cur = h_bin.copy()
    cur_c = compute_c5(cur, n)
    best = cur.copy()
    best_c = cur_c
    t_start = time.time()
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        progress = it / max(max_iters - 1, 1)
        T = T_init * (T_final / T_init) ** progress
        
        ones_idx = np.where(cur == 1)[0]
        zeros_idx = np.where(cur == 0)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i = ones_idx[rng.randint(len(ones_idx))]
        j = zeros_idx[rng.randint(len(zeros_idx))]
        cur[i] = 0
        cur[j] = 1
        new_c = compute_c5(cur, n)
        delta = new_c - cur_c
        if delta < 0 or (T > 1e-7 and rng.random() < np.exp(-delta / T)):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best = cur.copy()
        else:
            cur[i] = 1
            cur[j] = 0
    
    return best, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n = 200

    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)

    # Phase 1: Quick SLSQP polish
    if time.time() < deadline - 400:
        h1, c1 = slsqp_step(best_h, n, [2000, 10000], 250)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Phase 2: Coordinate descent
    if time.time() < deadline - 300:
        h2, c2 = coord_descent(best_h, n, 50, deadline - 300, gs_iters=55)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Phase 3: Binary SA (key new component)
    if time.time() < deadline - 200:
        n_sa_iter = max(5000, int((deadline - time.time() - 200) * 2000))
        hb, cb = binary_sa(best_h, n, n_sa_iter, deadline - 200, rng)
        if cb < best_c - 1e-7:
            # Polish the binary solution with CD
            h_pol, c_pol = coord_descent(hb, n, 40, deadline - 150, gs_iters=55)
            if c_pol < best_c:
                best_h, best_c = h_pol, c_pol
            elif cb < best_c:
                best_h, best_c = hb, cb

    # Phase 4: Multi-resolution search
    for n_try in [500, 350]:
        if time.time() > deadline - 150:
            break
        x_orig = np.linspace(0, 2, len(best_h))
        x_s = np.linspace(0, 2, n_try)
        h_s = project_feasible(np.interp(x_s, x_orig, best_h), n_try)
        h_s, c_s = slsqp_step(h_s, n_try, [5000, 20000], 200)
        h_s, c_s = coord_descent(h_s, n_try, 40, deadline - 120, gs_iters=50)
        if time.time() < deadline - 100:
            h_sb, c_sb = binary_sa(h_s, n_try, 3000, deadline - 100, rng)
            if c_sb < c_s:
                h_s, c_s = h_sb, c_sb
        h_back = project_feasible(np.interp(x_orig, x_s, h_s), len(best_h))
        h_back, c_back = coord_descent(h_back, len(best_h), 30, deadline - 80, gs_iters=55)
        if c_back < best_c:
            best_h, best_c = h_back, c_back

    # Phase 5: Multi-restart with diverse perturbations
    if time.time() < deadline - 200:
        n_cur = len(best_h)
        x = np.linspace(0, 2, n_cur)
        remaining = deadline - time.time() - 100
        n_restarts = max(4, int(remaining / 60))
        for r in range(n_restarts):
            if time.time() > deadline - 80:
                break
            if r % 4 == 0:
                hp = project_feasible(best_h + rng.normal(0, 0.015, n_cur), n_cur)
            elif r % 4 == 1:
                shift = rng.randint(1, max(2, n_cur // 40))
                hp = project_feasible(np.roll(best_h, shift), n_cur)
            elif r % 4 == 2:
                w = n_cur // 5
                s = rng.randint(0, n_cur - w)
                hp = best_h.copy()
                hp[s:s+w] = 1.0 - hp[s:s+w]
                hp = project_feasible(hp, n_cur)
            else:
                freq = 1.0 + 2.0 * r
                hp = project_feasible(best_h + 0.03 * np.sin(np.pi * freq * x), n_cur)
            
            h_o, c_o = slsqp_step(hp, n_cur, [3000, 15000], 150)
            h_o, c_o = coord_descent(h_o, n_cur, 25, deadline - 80, gs_iters=50)
            if time.time() < deadline - 60:
                h_ob, c_ob = binary_sa(h_o, n_cur, 2000, deadline - 60, rng)
                if c_ob < c_o:
                    h_o, c_o = c_ob and h_ob or h_o, c_ob if c_ob < c_o else (h_o, c_o)
            if c_o < best_c:
                best_h, best_c = h_o, c_o

    # Phase 6: Final polish
    if time.time() < deadline - 10:
        h_f, c_f = coord_descent(best_h, len(best_h), 30, deadline - 5, gs_iters=60)
        if c_f < best_c:
            best_h, best_c = h_f, c_f

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
