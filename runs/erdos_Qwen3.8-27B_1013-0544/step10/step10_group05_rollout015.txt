                x_lo = max(x_lo, h0 + r2)
            else:
                if abs(x_lo - (h0 + r1)) <= abs(x_hi - (h0 + r2)):
                    x_hi = min(x_hi, h0 + r1)
                else:
                    x_lo = max(x_lo, h0 + r2)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def cd_pass(c, h, n, order, deadline):
    improved = False
    for p in order:
        if time.time() > deadline:
            break
        h0 = h[p]
        if h0 < 1e-14 or h0 > 1.0 - 1e-14:
            continue
        xn = solve_coord(c, h, p, n)
        if abs(xn - h0) > 1e-14:
            delta = xn - h0
            m_min = max(0, n - 1 - p)
            m_max = min(2 * n - 2, 2 * n - 2 - p)
            ms = np.arange(m_min, m_max + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(vf):
                a[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                a[vb] -= h[jb[vb]]
            c[ms] += delta * a
            diag_i = np.where(ms == (n - 1))[0]
            if len(diag_i) > 0:
                c[ms[diag_i[0]]] -= delta * delta
            h[p] = xn
            improved = True
    return h, c, improved

def run_cd(h, n, max_passes, deadline, rng=None, random_order=False):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for it in range(max_passes):
        if time.time() > deadline:
            break
        if random_order and rng is not None:
            order = rng.permutation(n)
        else:
            order = np.arange(n)
        h, c, improved = cd_pass(c, h, n, order, deadline)
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-12:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def slsqp_opt(h0, n, alphas, maxiter, deadline):
    best_h, best_c = h0.copy(), compute_c5(h0, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def fourier_init(n, K, rng):
    x = np.linspace(0, 2, n, endpoint=False)
    h = np.zeros(n)
    for k in range(1, K + 1):
        amp = rng.normal(0, 1.0 / np.sqrt(k))
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(2 * np.pi * k * x / 2.0 + phase)
    h = 0.5 + 0.4 * h
    return project_feasible(h, n)

def interpolate_h(h_src, n_src, n_tgt):
    x_src = np.linspace(0, 2, n_src, endpoint=False)
    x_tgt = np.linspace(0, 2, n_tgt, endpoint=False)
    return project_feasible(np.interp(x_tgt, x_src, h_src), n_tgt)

def multi_res_refine(h_best, n, deadline, rng):
    """Optimize at coarser resolution, then refine back."""
    best_h, best_c = h_best.copy(), compute_c5(h_best, n)
    for n_coarse in [n // 2, n // 3]:
        if time.time() > deadline - 100:
            break
        if n_coarse < 100:
            continue
        h_c = interpolate_h(h_best, n, n_coarse)
        h_c, c_c = slsqp_opt(h_c, n_coarse, [5000, 50000, 500000], 200, deadline)
        h_c, c_c = run_cd(h_c, n_coarse, 300, deadline)
        h_fine = interpolate_h(h_c, n_coarse, n)
        h_fine, c_fine = slsqp_opt(h_fine, n, [10000, 100000, 1000000], 150, deadline)
        h_fine, c_fine = run_cd(h_fine, n, 200, deadline)
        if c_fine < best_c:
            best_h, best_c = h_fine, c_fine
    return best_h, best_c

def diverse_restart(h_best, n, deadline, rng, n_restarts):
    best_h, best_c = h_best.copy(), compute_c5(h_best, n)
    for r in range(n_restarts):
        if time.time() > deadline - 60:
            break
        temp = 0.002 + 0.008 * r
        hp = project_feasible(h_best + rng.normal(0, temp, n), n)
        h_o, c_o = slsqp_opt(hp, n, [5000, 50000, 500000, 5000000], 100, deadline)
        h_o, c_o = run_cd(h_o, n, 100, deadline, rng, True)
        if c_o < best_c:
            best_h, best_c = h_o, c_o
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = np.full(600, 0.5)
        n = 600
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)

    # Phase 1: Long CD to convergence
    h1, c1 = run_cd(h_best, n, 500, deadline)
    if c1 < best_c:
        h_best, best_c = h1, c1

    # Phase 2: SLSQP with wide alpha sweep
    h2, c2 = slsqp_opt(h_best, n, [50, 200, 1000, 5000, 20000, 100000, 500000, 2000000], 300, deadline)
    if c2 < best_c:
        h_best, best_c = h2, c2

    # Phase 3: Another long CD
    h3, c3 = run_cd(h_best, n, 500, deadline)
    if c3 < best_c:
        h_best, best_c = h3, c3

    # Phase 4: SLSQP at high alpha (near-exact max)
    h4, c4 = slsqp_opt(h_best, n, [500000, 2000000, 10000000], 400, deadline)
    if c4 < best_c:
        h_best, best_c = h4, c4

    # Phase 5: CD again
    h5, c5 = run_cd(h_best, n, 300, deadline)
    if c5 < best_c:
        h_best, best_c = h5, c5

    # Phase 6: Multi-resolution refinement
    if time.time() < deadline - 200:
        h6, c6 = multi_res_refine(h_best, n, deadline, rng)
        if c6 < best_c:
            h_best, best_c = h6, c6

    # Phase 7: Diverse restarts with small perturbations
    if time.time() < deadline - 120:
        h7, c7 = diverse_restart(h_best, n, deadline, rng, 12)
        if c7 < best_c:
            h_best, best_c = h7, c7

    # Phase 8: Final alternating SLSQP+CD
    for _ in range(3):
        if time.time() > deadline - 30:
            break
        h_a, c_a = slsqp_opt(h_best, n, [1000000, 5000000], 200, deadline)
        if c_a < best_c:
            h_best, best_c = h_a, c_a
        h_b, c_b = run_cd(h_best, n, 200, deadline)
        if c_b < best_c:
            h_best, best_c = h_b, c_b

    # Final polish
    if time.time() < deadline - 2:
        h_f, c_f = run_cd(h_best, n, 100, deadline)
        if c_f < best_c:
            h_best, best_c = h_f, c_f

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
