```
def coordinate_descent(h, n, stop_time, patience=20, iters=20, tol=1e-13):
    h = project_feasible(h)
    n = int(h.size)
    c = np.correlate(h, 1.0 - h, 'full')
    dx = 2.0 / n
    best_h = h.copy()
    best_c = float(np.max(c)) * dx
    plan = make_plan(n)
    om_rev = 1.0 - h[::-1]
    stall = 0
    while time.time() < stop_time:
        for p in range(n):
            if time.time() > stop_time:
                break
            h0 = h[p]
            x, coef = coord_update(h, om_rev, c, plan[p], p, n, tol, iters)
            delta = x - h0

                c[m_lo:m_lo+L] += delta * coef
                c[n-1] += x * (1.0 - x) - h0 * (1.0 - h0)
                h[p] = x

            best_c = cv
            best_h = h.copy()
            stall = 0

            stall += 1
            if stall >= patience:
```

```
def coordinate_descent(h,n,stop,patience=20,iters=20,tol=1e-13):
    h=project_feasible(h); n=int(h.size)
    c=np.correlate(h,1-h,'full'); dx=2/n
    best=h.copy(); bestc=float(c.max())*dx
    plan=make_plan(n); om=1-h[::-1]; stall=0
    while time.time()<stop:
       for p in range(n):
          if time.time()>stop: break
          h0=h[p]
          x,coef=coord_update(h,om,c,plan[p],p,n,tol,iters)
          d=x-h0
          if abs(d)>1e-13:
             m_lo,a0,b0,LL,diag=plan[p]
             c[m_lo:m_lo+LL]+=d*coef
             c[n-1]+=x*(1-x)-h0*(1-h0)
             h[p]=x; om[n-1-p]=1-x
       h=project_feasible(h); c=np.correlate(h,1-h,'full'); cv=float(c.max())*dx
       if cv<bestc-1e-12: bestc=cv; best=h.copy(); stall=0
       else:
          stall+=1
          if stall>=patience: break
       om=1-h[::-1]
    return project_feasible(best), bestc
```

```
def coordinate_descent(h, n, stop_time, patience=20, iters=20, tol=1e-13):
    h = project_feasible(h)
    n = int(h.size)
    c = np.correlate(h, 1.0 - h, 'full')
    dx = 2.0 / n
    best_h = h.copy()
    best_c = float(np.max(c)) * dx
    plan = make_plan(n)
    om = 1.0 - h[::-1]
    stall = 0
    while time.time() < stop_time:
        for p in range(n):
            if time.time() > stop_time:
                break
            h0 = h[p]
            x, coef = coord_update(h, om, c, plan[p], p, n, tol, iters)
            d = x - h0
            if abs(d) > 1e-13:
                m_lo, a0, b0, Ln, diag = plan[p]
                c[m_lo:m_lo + Ln] += d * coef
                c[n - 1] += x * (1.0 - x) - h0 * (1.0 - h0)
                h[p] = x
                om[n - 1 - p] = 1.0 - x
        h = project_feasible(h)
        c = np.correlate(h, 1.0 - h, 'full')
        cv = float(np.max(c)) * dx
        if cv < best_c - 1e-12:
            best_c = cv
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall >= patience:
                break
        om = 1.0 - h[::-1]
    return project_feasible(best_h), best_c
```

```python
import numpy as np
import time


def c5_value(h):
    n = h.size
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * 2.0 / n)


def project_feasible(h, target=None):
    h = np.array(h, dtype=float, copy=True)
    n = h.size
    target = 0.5 * n if target is None else float(target)
    np.clip(h, 0.0, 1.0, out=h)
    D = target - float(h.sum())
    if abs(D) < 1e-14:
        return h

    if D > 0.0:
        lo, hi = 0.0, 1.0
    else:
        lo, hi = -1.0, 0.0

    for _ in range(45):
        mid = 0.5 * (lo + hi)
        hp = np.clip(h + mid, 0.0, 1.0)
        s = float(hp.sum())
        if (D > 0.0 and s > target) or (D < 0.0 and s < target):
            hi = mid
        else:
            lo = mid

    h = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)

    for _ in range(3):
        D = target - float(h.sum())
        if abs(D) < 1e-15:
            break
        m = (h < 1.0) if D > 0.0 else (h > 0.0)
        if np.any(m):
            h[m] += D / float(m.sum())
            np.clip(h, 0.0, 1.0, out=h)

    D = target - float(h.sum())
    if abs(D) > 1e-15:
        m = (h < 1.0) if D > 0.0 else (h > 0.0)
        if np.any(m):
            h[m] += D / float(m.sum())
            np.clip(h, 0.0, 1.0, out=h)
    return h


def make_plan(n):
    plan = []
    for p in range(n):
        A0 = n - 1 - p
        B0 = p
        m_lo = A0 if A0 < B0 else B0
        a0 = A0 - m_lo
        b0 = B0 - m_lo
        Ln = 2 * n - 1 - 2 * m_lo
        diag = n - 1 - m_lo
        plan.append((m_lo, a0, b0, Ln, diag))
    return plan


def coord_update(h, om_rev, c, entry, p, n, tol, iters):
    m_lo, a0, b0, Ln, diag = entry
    h0 = float(h[p])
    cs = c[m_lo:m_lo + Ln]
    cmax = float(np.max(c))

    if float(np.max(cs)) + 1e-12 < cmax:
        return h0, None

    f0 = h0 * (1.0 - h0)
    coef = np.zeros(Ln)
    coef[a0:a0 + n] -= h
    coef[b0:b0 + n] += om_rev
    coef[diag] = np.nan

    pos = coef > tol
    neg = coef < -tol
    zero = (coef >= -tol) & (coef <= tol)

    cd = float(cs[diag])
    zm = float(np.max(cs[zero])) if np.any(zero) else 0.0
    pa = bool(np.any(pos))
    na = bool(np.any(neg))

    if pa:
        ap = coef[pos]
        cp = cs[pos]
        ip = 1.0 / ap
        bp = h0 - cp * ip
    if na:
        an = coef[neg]
        cn = cs[neg]
        ivn = 1.0 / an
        bn = h0 - cn * ivn

    Lhi = cmax
    Llo = zm
    if Lhi <= Llo:
        Lhi = Llo + 1e-12

    for _ in range(iters):
        Lv = 0.5 * (Llo + Lhi)
        ok = (zm <= Lv)
        xlo = 0.0
        xhi = 1.0

        if ok and pa:
            xhi = min(xhi, float(np.min(bp + Lv * ip)))
        if ok and na:
            xlo = max(xlo, float(np.max(bn + Lv * ivn)))

        if ok:
            if xlo < 0.0:
                xlo = 0.0
            if xhi > 1.0:
                xhi = 1.0
            if xlo > xhi + 1e-13:
                ok = False
            T = Lv - cd + f0
            if ok and T < -1e-12:
                ok = False
            elif ok and T < 0.25:
                disc = 1.0 - 4.0 * T
                if disc < 0.0:
                    ok = False
                else:
                    r1 = 0.5 * (1.0 - np.sqrt(disc))
                    r2 = 0.5 * (1.0 + np.sqrt(disc))
                    xl = xlo
                    xh = xhi
                    left = (xl <= r1 + 1e-12) and (min(xh, r1) >= xl - 1e-12)
                    right = (xh >= r2 - 1e-12) and (max(xl, r2) <= xh + 1e-12)
                    if not (left or right):
                        ok = False

        if ok:
            Lhi = Lv
        else:
            Llo = Lv

    Lv = Lhi + 1e-11 * max(1.0, abs(Lhi))
    if zm > Lv + 1e-10:
        coef[diag] = 0.0
        return h0, coef

    xlo = 0.0
    xhi = 1.0
    if pa:
        xhi = min(xhi, float(np.min(bp + Lv * ip)))
    if na:
        xlo = max(xlo, float(np.max(bn + Lv * ivn)))
    if xlo < 0.0:
        xlo = 0.0
    if xhi > 1.0:
        xhi = 1.0

    T = Lv - cd + f0
    intervals = []
    if T >= -1e-10:
        if T >= 0.25:
            intervals.append((xlo, xhi))
        else:
            if T < 0.0:
                T = 0.0
            disc = 1.0 - 4.0 * T
            if disc >= 0.0:
                r1 = 0.5 * (1.0 - np.sqrt(disc))
                r2 = 0.5 * (1.0 + np.sqrt(disc))
                if xlo <= r1 + 1e-12:
                    bb = min(xhi, r1)
                    if bb >= xlo - 1e-12:
                        intervals.append((xlo, bb))
                if xhi >= r2 - 1e-12:
                    aa = max(xlo, r2)
                    if xhi >= aa - 1e-12:
                        intervals.append((aa, xhi))

    bx = h0
    bd = 2.0
    found = False
    for aa, bb in intervals:
        if aa < 0.0:
            aa = 0.0
        if bb > 1.0:
            bb = 1.0
        if bb >= aa - 1e-12:
            xc = h0
            if xc < aa:
                xc = aa
            if xc > bb:
                xc = bb
            d = abs(xc - h0)
            if not found or d < bd:
                bx = xc
                bd = d
                found = True

    coef[diag] = 0.0
    if not found:
        return h0, coef
    return float(np.clip(bx, 0.0, 1.0)), coef


def coordinate_descent(h, n, stop_time, patience=20, iters=20, tol=1e-12):
    h = project_feasible(h)
    n = int(h.size)
    c = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / n

    best_h = h.copy()
    best_c = float(np.max(c)) * dx
    plan = make_plan(n)
    om = 1.0 - h[::-1]
    stall = 0

    while time.time() < stop_time:
        for p in range(n):
            if time.time() > stop_time:
                break
            h0 = h[p]
            x, coef = coord_update(h, om, c, plan[p], p, n, tol, iters)
            if coef is None:
                continue
            d = x - h0
            if abs(d) > 1e-13:
                m_lo, a0, b0, Ln, diag = plan[p]
                c[m_lo:m_lo + Ln] += d * coef
                c[n - 1] += x * (1.0 - x) - h0 * (1.0 - h0)
                h[p] = x
                om[n - 1 - p] = 1.0 - x

        h = project_feasible(h)
        c = np.correlate(h, 1.0 - h, mode="full")
        cv = float(np.max(c)) * dx

        if cv < best_c - 1e-12:
            best_c = cv
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall >= patience:
                break

        om = 1.0 - h[::-1]

    return project_feasible(best_h), best_c


def random_h(n, rng, kind):
    x = np.linspace(0.0, 2.0, n)
    if kind == 0:
        h = np.full(n, 0.5)
        for _ in range(4):
            f = int(rng.randint(1, max(2, n // 10)))
            amp = float(rng.uniform(0.05, 0.35))
            ph = float(rng.uniform(0.0, 2.0 * np.pi))
            h += amp * np.sin(np.pi * f * x + ph)
    elif kind == 1:
        h = (rng.rand(n) < 0.5).astype(float)
    else:
        h = 0.5 + 0.25 * np.sin(np.pi * x) + 0.15 * np.sin(2.0 * np.pi * x + 0.7)
    return project_feasible(h)


def perturb_h(h, rng, sigma):
    n = h.size
    r = int(rng.randint(0, 4))
    if r == 0:
        h2 = h.copy()
        for _ in range(3):
            l = int(rng.randint(0, n))
            w = int(rng.randint(1, max(2, n // 8)))
            rr = min(n, l + w)
            h2[l:rr] = 1.0 - h2[l:rr]
    elif r == 1:
        h2 = h + float(sigma) * np.sign(h - 0.5)
    else:
        x = np.linspace(0.0, 2.0, n)
        h2 = h + float(sigma) * rng.normal(size=n)
        for _ in range(2):
            f = int(rng.randint(1, 6))
            amp = float(rng.uniform(0.0, sigma))
            ph = float(rng.uniform(0.0, 2.0 * np.pi))
            h2 += amp * np.sin(np.pi * f * x + ph)
    return project_feasible(h2)
```
