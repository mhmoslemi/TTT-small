```python
import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)))

def project_feasible(x, n):
    t = 0.5 * float(n)
    x = np.clip(np.asarray(x, dtype=float), 0.0, 1.0).copy()
    lo, hi = -1.0, 2.0
    for _ in range(64):
        m = 0.5 * (lo + hi)
        s = np.clip(x - m, 0.0, 1.0).sum()
        if s > t:
            lo = m
        else:
            hi = m
    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(25):
        d = t - y.sum()
        if abs(d) < 1e-13:
            break
        f = (y > 1e-12) & (y < 1.0 - 1e-12)
        if f.any():
            y[f] += d / f.sum()
        else:
            y += d / max(1.0, float(n))
        np.clip(y, 0.0, 1.0, out=y)
    return y

def binary_topk(h, n):
    t = 0.5 * float(n)
    o = np.argsort(h)[::-1]
    b = np.zeros(n)
    k = int(t)
    if k > 0:
        b[o[:k]] = 1.0
    if n > k and t - k > 1e-12:
        b[o[k]] = min(1.0, max(0.0, t - k))
    return b

def resample_h(h, n1, n2):
    if n1 == n2:
        return project_feasible(h.copy(), n2)
    x2 = np.linspace(0.0, 2.0, n2, endpoint=False)
    x1 = np.linspace(0.0, 2.0, n1, endpoint=False)
    return project_feasible(np.interp(x2, x1, h), n2)

def fourier_init(n, K, rng):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    z = np.zeros(n)
    for k in range(1, K + 1):
        z += rng.normal(0.0, 1.0 / np.sqrt(k)) * np.cos(np.pi * k * x + rng.uniform(0.0, 2.0 * np.pi))
    h = 0.5 + 0.5 * np.tanh(0.9 * z)
    return project_feasible(h, n)

def _smooth_core(x, n, alpha, lam):
    h = np.asarray(x, dtype=float)
    dx = 2.0 / n
    with np.errstate(over='ignore', under='ignore'):
        raw = np.correlate(h, 1.0 - h, mode='full') * dx
        m = float(raw.max())
        if alpha <= 1e-8:
            w = np.full(raw.shape, 1.0 / raw.size)
            s = float(raw.mean())
        else:
            e = np.exp(alpha * (raw - m))
            se = e.sum()
            s = m + float(np.log(se)) / alpha
            w = e / se
        g = np.correlate(1.0 - h, w, mode='full') - np.convolve(h, w, mode='full')
        g = g[n - 1:n - 1 + n] * dx
    diff = float(h.sum() - 0.5 * n)
    e2 = diff / max(1.0, float(n))
    f = s + lam * e2 * e2
    if lam:
        g = g + (2.0 * lam * e2 / max(1.0, float(n))) * np.ones(n)
    if not np.isfinite(f):
        f, g = 1e100, np.zeros_like(h)
    return float(f), g

def _fun_tuple(x, n, alpha, lam):
    return _smooth_core(x, n, alpha, lam)

def _fun_f(x, n, alpha, lam):
    return _smooth_core(x, n, alpha, lam)[0]

def _fun_g(x, n, alpha, lam):
    return _smooth_core(x, n, alpha, lam)[1]

def lbfgs_repair(h0, n, alphas, lams, maxiter, deadline):
    h = project_feasible(h0.copy(), n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    opts = {'maxiter': maxiter, 'ftol': 1e-10, 'gtol': 1e-6, 'maxls': 20}
    for alpha in alphas:
        for lam in lams:
            if time.time() > deadline:
                break
            res = None
            try:
                res = minimize(_fun_tuple, h, args=(n, alpha, lam), method='L-BFGS-B',
                               bounds=bounds, options=opts)
            except Exception:
                try:
                    res = minimize(_fun_f, h, args=(n, alpha, lam), jac=_fun_g,
                                   method='L-BFGS-B', bounds=bounds, options=opts)
                except Exception:
                    res = None
            if res is None or not hasattr(res, 'x'):
                continue
            hp = project_feasible(res.x, n)
            c = compute_c5(hp, n)
            if c < best_c:
                h, best_c = hp, c
            else:
                h = project_feasible(h, n)
    return h, best_c

def build_coord_info(n):
    L = n - 1
    infos = []
    for p in range(n):
        mn = max(0, L - p)
        mx = min(2 * L, 2 * L - p)
        ms = np.arange(mn, mx + 1, dtype=np.intp)
        f = p + ms - L
        b = p - ms + L
        f = np.where((f >= 0) & (f < n), f, -1)
        b = np.where((b >= 0) & (b < n), b, -1)
        infos.append((mn, mx, ms, f, b, L - mn))
    return infos

def solve_coord(c, h, p, info):

    h0 = float(h[p])
    a = np.zeros(ms.size, dtype=float)
    vf = fidx >= 0
    if vf.any():
        a[vf] += 1.0 - h[fidx[vf]]
    vb = bidx >= 0
    if vb.any():
        a[vb] -= h[bidx[vb]]
    cb = c[ms]
    U = 0.0
    if mn > 0:

    if mx < len(c) - 1:

    cur = float(c.max())
    tol = 1e-14
    neg = a < -tol
    pos = a > tol
    zero = ~(neg | pos)
    cneg, aneg = cb[neg], a[neg]
    cpos, apos = cb[pos], a[pos]
    zmax = float(np.max(cb[zero] - a[zero] * h0)) if zero.any() else -np.inf
    Cd = float(cb[di] - h0 * (1.0 - h0))
    lo, hi = U, cur
    for _ in range(28):
        Lm = 0.5 * (lo + hi)
        xl, xh = 0.0, 1.0
        if neg.any():
            xl = max(xl, float(np.max((Lm - cneg) / aneg + h0)))
        if pos.any():
            xh = min(xh, float(np.min((Lm - cpos) / apos + h0)))
        if xl < 0.0:
            xl = 0.0
        if xh > 1.0:
            xh = 1.0
        ok = True
        if zmax > Lm + 1e-12:
            ok = False
        elif xh < xl - 1e-12:
            ok = False
        else:
            D = 1.0 + 4.0 * Cd - 4.0 * Lm
            if D > 0.0:
                sq = np.sqrt(D)
                r1 = 0.5 * (1.0 - sq)
                r2 = 0.5 * (1.0 + sq)
                if xl > r1 + 1e-12 and xh < r2 - 1e-12:
                    ok = False
        if ok:
            hi = Lm
        else:
            lo = Lm
    Ls = hi
    xl, xh = 0.0, 1.0
    if neg.any():
        xl = max(xl, float(np.max((Ls - cneg) / aneg + h0)))
    if pos.any():
        xh = min(xh, float(np.min((Ls - cpos) / apos + h0)))
    if xl < 0.0:
        xl = 0.0
    if xh > 1.0:
        xh = 1.0
    if xh < xl - 1e-12:
        return h0, a
    D = 1.0 + 4.0 * Cd - 4.0 * Ls
    seg = [(xl, xh)] if D <= 0.0 else []
    if D > 0.0:
        sq = np.sqrt(D)
        r1 = 0.5 * (1.0 - sq)
        r2 = 0.5 * (1.0 + sq)
        if xl <= r1 + 1e-12:
            seg.append((xl, min(xh, r1)))
        if xh >= r2 - 1e-12:
            seg.append((max(xl, r2), xh))
    cands = [h0, 0.0, 1.0, xl, xh]
    for s0, e0 in seg:
        if e0 - s0 > 1e-14:
            cands += [s0, e0, 0.5 * (s0 + e0)]
        elif e0 >= s0 - 1e-14:
            cands.append(0.5 * (s0 + e0))
    if D > 0.0:
        if -0.1 < r1 < 1.1:
            cands.append(r1)
        if -0.1 < r2 < 1.1:
            cands.append(r2)
    bestx, bestv = h0, cur + 1.0
    for xx in cands:
        xx = float(np.clip(xx, 0.0, 1.0))
        vals = cb + a * (xx - h0)
        vals[di] = -xx * xx + xx + Cd
        val = U
        if vals.size:
            val = max(val, float(np.max(vals)))
        if val < bestv:
            bestv, bestx = val, xx
    if bestv > cur + 1e-9:
        return h0, a
    return bestx, a

def exact_cd(h, n, passes, deadline, infos):
    if infos is None:
        return project_feasible(h, n), compute_c5(h, n)
    h = project_feasible(h.copy(), n)
    c = np.correlate(h, 1.0 - h, mode='full')
    pk = max(24, n // 12)
    for _ in range(passes):
        if time.time() > deadline:
            break
        improved = False
        changed = 0
        curmax = float(c.max())
        for p in range(n):
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            info = infos[p]
            xn, a = solve_coord(c, h, p, info)
            dxx = xn - h0
            if abs(dxx) < 1e-13:
                continue
            ms, di = info[2], info[5]
            c[ms] += dxx * a
            c[ms[di]] -= dxx * dxx
            newmax = float(np.max(c))
            if newmax > curmax + 1e-11:
                c[ms] -= dxx * a
                c[ms[di]] += dxx * dxx
            else:
                h[p] = xn
                curmax = newmax
                improved = True
                changed += 1
                if changed % pk == 0:
                    hp = project_feasible(h, n)
                    if abs(hp.sum() - 0.5 * n) > 1e-12:
                        h = hp
                        c = np.correlate(h, 1.0 - h, mode='full')
                        curmax = float(c.max())
                    else:
                        h = hp
        if not improved:
            break
        h = project_feasible(h, n)
        c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    reserve = min(10.0, 0.1 * float(budget_s))
    end = t0 + float(budget_s) - reserve
    if end <= t0:
        end = t0 + max(1.0, 0.8 * float(budget_s))
    rng = np.random.RandomState(seed)
    try:
        h_best = np.asarray(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = np.full(700, 0.5)
        n = 700
    if n < 2:
        h_best = np.full(700, 0.5)
        n = 700
    if n > 1100:
        ns = 700
        h_best = np.interp(np.linspace(0, 2, ns, endpoint=False),
                           np.linspace(0, 2, n, endpoint=False), h_best)
        n = ns
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)
    infos = build_coord_info(n) if n <= 1000 else None

    d1 = min(t0 + min(180.0, 0.18 * float(budget_s)), end)
    if time.time() < d1:
        h2, c2 = lbfgs_repair(h_best, n, [50.0, 500.0, 5000.0, 50000.0], [50.0], 35, d1)
        if c2 < best_c:
            h_best, best_c = h2, c2

    d2 = min(t0 + min(430.0, 0.38 * float(budget_s)), end)
    for nb in (35, 50, 70):
        if time.time() > d2 - 20:
            break
        hc = resample_h(h_best, n, nb)
        hco, _ = lbfgs_repair(hc, nb, [500.0, 5000.0, 20000.0], [50.0], 45,
                              min(time.time() + 45.0, d2))
        hf = resample_h(hco, nb, n)
        cf = compute_c5(hf, n)
        if cf < best_c:
            h_best, best_c = hf, cf

    d3 = min(t0 + min(760.0, 0.66 * float(budget_s)), end)
    starts = []
    for nb in (35, 50):
        starts.append(resample_h(resample_h(h_best, n, nb), nb, n))
    starts.append(binary_topk(h_best, n) + 0.02 * rng.normal(size=n))
    starts.append(fourier_init(n, 12, rng))
    starts.append(fourier_init(n, 24, rng))
    for s in starts:
        if time.time() > d3 - 20:
            break
        s = project_feasible(s, n)
        h2, c2 = lbfgs_repair(s, n, [2000.0, 20000.0], [50.0], 35,
                              min(time.time() + 45.0, d3))
        if c2 < best_c:
            h_best, best_c = h2, c2

    if infos is not None and time.time() < end - 220:
        h2, c2 = exact_cd(h_best, n, 8, min(end - 220, time.time() + 120), infos)
        if c2 < best_c:
            h_best, best_c = h2, c2

    d4 = max(time.time(), end - 90.0)
    if time.time() < d4:
        h2, c2 = lbfgs_repair(h_best, n, [50000.0, 200000.0], [500.0], 55, d4)
        if c2 < best_c:
            h_best, best_c = h2, c2

    if infos is not None and time.time() < end - 8:
        h2, c2 = exact_cd(h_best, n, 12, end - 5, infos)
        if c2 < best_c:
            h_best, best_c = h2, c2

    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)

    hb = project_feasible(binary_topk(h_best, n), n)
    cb = compute_c5(hb, n)
    if cb < best_c:
        h_best, best_c = hb, cb

    return h_best, float(best_c), int(n)
```
