def cd_choose_x(L, a,b,doth,h0, tol=1e-12):
    lo=0.0; hi=1.0
    m = a < -tol
    if np.any(m):
       r=(L-b[m])/a[m]
       lo=max(lo,float(np.max(r)))
    m = a > tol
    if np.any(m):
       r=(L-b[m])/a[m]
       hi=min(hi,float(np.min(r)))
    if lo<0: lo=0
    if hi>1: hi=1
    if lo > hi + 1e-10:
       return float(np.clip(h0,0,1))
    lq=L-doth
    if lq < 0:
       return float(np.clip(h0,lo,hi))
    if lq >= 0.25:
       return float(np.clip(h0,lo,hi))
    disc=1-4*lq
    if disc <0: disc=0
    sq=np.sqrt(disc); r1=(1-sq)*0.5; r2=(1+sq)*0.5
    bestx=None; bestd=1e30

          d=abs(x-h0)
       if d < bestd:
          bestd=d; bestx=x
    if bestx is None: return float(np.clip(h0,lo,hi))
    return float(np.clip(bestx,0,1))

def smooth_grad(x,n,tau,rho):
    c=np.correlate(x,1.0-x,mode='full')
    mx=float(c.max())
    z=(c-mx)/tau
    ew=np.exp(z); w=ew/float(ew.sum())
    g=np.zeros(n)
    L=2*n-1
    for idx in range(L):
       wi=w[idx]
       if wi <= 1e-15: continue
       k=idx-(n-1)
       ps = -k if k<0 else 0
       pe = n-1-k if n-1-k < n-1 else n-1
       if ps <= pe:
           g[ps:pe+1] += wi*(1.0-x[ps+k:pe+k+1])
       ps2 = k if k>0 else 0
       pe2 = n-1+k if n-1+k < n-1 else n-1
       if ps2 <= pe2:
           g[ps2:pe2+1] -= wi*x[ps2-k:pe2-k+1]
    s=x.sum()-0.5*n
    if s: g += 2.0*rho*s
    return g

import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(v, n):
    v = np.asarray(v, dtype=float).copy()
    np.clip(v, 0.0, 1.0, out=v)
    target = 0.5 * n
    if abs(v.sum() - target) > 1e-13:
        lo, hi = -1.0, 1.0
        for _ in range(55):
            mid = 0.5 * (lo + hi)
            if np.clip(v + mid, 0.0, 1.0).sum() < target:
                lo = mid
            else:
                hi = mid
        v = np.clip(v + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - v.sum()
    if abs(diff) > 1e-12:
        if diff > 0.0:
            idx = np.where(v < 1.0)[0]
        else:
            idx = np.where(v > 0.0)[0]
        rem = diff
        for i in idx:
            if rem > 0.0:
                inc = min(1.0 - v[i], rem)
                v[i] += inc
                rem -= inc
            else:
                dec = max(-v[i], rem)
                v[i] += dec
                rem -= dec
            if abs(rem) < 1e-14:
                break
    return v

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))

def smooth_obj(x, n, tau, rho):
    c = np.correlate(x, 1.0 - x, mode="full")
    mx = float(c.max())
    z = (c - mx) / tau
    ew = np.exp(z)
    val = mx + tau * float(np.log(ew.sum()))
    s = x.sum() - 0.5 * n
    return val + rho * s * s

def smooth_grad(x, n, tau, rho):
    c = np.correlate(x, 1.0 - x, mode="full")
    mx = float(c.max())
    z = (c - mx) / tau
    ew = np.exp(z)
    w = ew / float(ew.sum())
    g = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = float(w[idx])
        if wi <= 1e-15:
            continue
        k = idx - (n - 1)
        ps = -k if k < 0 else 0
        pe = n - 1 - k if n - 1 - k < n - 1 else n - 1
        if ps <= pe:
            g[ps:pe + 1] += wi * (1.0 - x[ps + k:pe + k + 1])
        ps2 = k if k > 0 else 0
        pe2 = n - 1 + k if n - 1 + k < n - 1 else n - 1
        if ps2 <= pe2:
            g[ps2:pe2 + 1] -= wi * x[ps2 - k:pe2 - k + 1]
    s = x.sum() - 0.5 * n
    if s != 0.0:
        g += 2.0 * rho * s
    return g

def smooth_phase(h, n, end_time, bounds):
    best_h = h.copy()
    best_c = compute_c5(best_h)
    settings = [
        (30.0, 1000.0, 50),
        (15.0, 1000.0, 50),
        (7.0, 3000.0, 45),
        (3.5, 5000.0, 40),
        (1.5, 8000.0, 35),
        (0.8, 8000.0, 25),
    ]
    stalls = 0
    for tau, rho, mit in settings:
        if time.time() > end_time - 8:
            break
        try:
            res = minimize(
                smooth_obj, best_h, args=(n, tau, rho),
                jac=smooth_grad, method="L-BFGS-B", bounds=bounds,
                options={"maxiter": mit, "ftol": 1e-12, "gtol": 1e-7, "maxls": 20}
            )
            hh = project_feasible(res.x, n)
            cc = compute_c5(hh)
            if cc < best_c - 1e-11:
                best_h = hh
                best_c = cc
                stalls = 0
            else:
                stalls += 1
                if stalls >= 4:
                    break
        except Exception:
            break
    return best_h, best_c

def cd_feasible(L, a, b, doth):
    tol = 1e-13
    lo, hi = 0.0, 1.0
    m = a < -tol
    if np.any(m):
        lo = max(lo, float(np.max((L - b[m]) / a[m])))
    m = a > tol
    if np.any(m):
        hi = min(hi, float(np.min((L - b[m]) / a[m])))
    m = np.abs(a) <= tol
    if np.any(m) and float(np.max(b[m])) > L + 1e-11:
        return False
    lo = max(0.0, lo)
    hi = min(1.0, hi)
    if lo > hi + 1e-11:
        return False
    lq = L - doth
    if lq < -1e-10:
        return False
    if lq < 0.25 - 1e-13:
        disc = 1.0 - 4.0 * lq
        if disc < 0.0:

            else:

        sq = np.sqrt(disc)
        r1 = 0.5 * (1.0 - sq)
        r2 = 0.5 * (1.0 + sq)
        if hi < r1 - 1e-11 or lo > r2 + 1e-11:
            return False
    return True

def cd_choose_x(L, a, b, doth, h0):
    tol = 1e-13
    lo, hi = 0.0, 1.0
    m = a < -tol
    if np.any(m):
        lo = max(lo, float(np.max((L - b[m]) / a[m])))
    m = a > tol
    if np.any(m):

    lo = max(0.0, lo)
    hi = min(1.0, hi)

    if lq < 0.0:
        return float(np.clip(h0, lo, hi))
    if lq >= 0.25 - 1e-13:
        return float(np.clip(h0, lo, hi))
    disc = max(0.0, 1.0 - 4.0 * lq)
    sq = np.sqrt(disc)
    r1 = 0.5 * (1.0 - sq)
    r2 = 0.5 * (1.0 + sq)
    bestx = None
    bestd = 1e300
    for Lr, Rr in ((max(lo, 0.0), min(hi, r1)), (max(lo, r2), min(hi, 1.0))):

def cd_min_x(c, h, p, n, segmax, bits=22):
    start = n - 1 - p
    cseg = c[start:start + n]
    h0 = h[p]
    if segmax <= 0.0:
        return None
    a = 1.0 - h
    i0 = max(0, 2 * p - (n - 1))
    i1 = min(n - 1, 2 * p)
    if i0 <= i1:
        j0 = 2 * p - i1
        j1 = 2 * p - i0
        a[i0:i1 + 1] -= h[j0:j1 + 1][::-1]
    b = cseg - a * h0
    a_lin = a.copy()
    b_lin = b.copy()
    a_lin[p] = 0.0
    b_lin[p] = -1e300
    doth = cseg[p] - h0 * (1.0 - h0)
    low, high = 0.0, float(segmax)

        return None

        if time.time() < final_end - 10:
            h, best_c = cd_loop(h, n, final_end, rng, pass_time=20, max_passes=40)
            n = len(h)
    except Exception:
        pass

    h = project_feasible(h, len(h))
    best_c = compute_c5(h)
    return h, best_c, len(h)
