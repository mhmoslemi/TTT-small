def update_corr(c, h, p, old_val, new_val, n):

    m_min = n - 1 - p

    vb = (jb >= 0) & (jb < n)

    if np.any(vb):
        a[vb] -= h[jb[vb]]

    c[ms] += delta * a
    if len(diag_i) > 0:

    return c

            break

        order = list(range(n))
        if rng is not None and it % 2 == 1:
            rng.shuffle(order)
        for p in order:

            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue

                h[p] = xn

            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def compute_c5(h, n):

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)

        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def update_corr(c, h, p, old_val, new_val, n):
    delta = new_val - old_val
    if abs(delta) < 1e-14:
        return c
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    diag_i = np.where(ms == (n - 1))[0]
    c[ms] += delta * a
    if len(diag_i) > 0:
        c[ms[diag_i[0]]] -= delta * delta

def exact_cd(h, n, max_passes, deadline, rng=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for it in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        order = list(range(n))
        if rng is not None and it % 2 == 1:
            rng.shuffle(order)
        for p in order:
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            xn = solve_1d(c, h, p, n)
            if abs(xn - h0) > 1e-12:
                update_corr(c, h, p, h0, xn, n)
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def smooth_obj_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    M = c.max()
    sht = alpha * (c - M)
    mx = sht.max()
    e = np.exp(sht - mx)
    w = e / e.sum()
    lse = mx + np.log(np.sum(e))
    val = (M + lse / alpha) * (2.0 / n)
    grad = np.zeros(n)
    omh = 1.0 - h
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        s = k - (n - 1)
        plo = max(0, -s)
        phi = min(n - 1, n - 1 - s)
        if plo <= phi:
            grad[plo:phi + 1] += wk * omh[plo + s:phi + s + 1]
        plo = max(0, s)
        phi = min(n - 1, n - 1 + s)
        if plo <= phi:
            grad[plo:phi + 1] -= wk * h[plo - s:phi - s + 1]
    return val, grad * (2.0 / n)

def smooth_slsqp(h_init, n, alphas, maxiter, deadline):
    best_h, best_c = h_init.copy(), compute_c5(h_init, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline:
            break
        def obj(x):
            v, _ = smooth_obj_grad(x, n, alpha)
            return v
        def grd(x):
            _, g = smooth_obj_grad(x, n, alpha)
            return g
        def cf(x):
            return x.sum() - n / 2.0
        def cj(x):
            return np.ones(n)
        cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]
        try:
            res = minimize(obj, best_h, jac=grd, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c - 1e-12:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def interp_h(h_old, n_old, n_new):

    h_new = np.interp(x_new, x_old, h_old)
