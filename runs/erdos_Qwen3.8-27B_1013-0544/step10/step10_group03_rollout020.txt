import time

def solve_coord(c, h, p, n):
    h0 = h[p]
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms]
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    if has_diag:
        a[diag_idx] = 0.0
    unaff = 0.0
    if m_min > 0:
        unaff = float(np.max(c[:m_min]))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    cur_max = float(np.max(c))
    neg_mask = a < -1e-15
    pos_mask = a > 1e-15
    zero_mask = np.abs(a) <= 1e-15
    L_lo, L_hi = unaff, cur_max
    for _ in range(55):
        if L_hi - L_lo < 1e-14:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        feasible = True
        if np.any(zero_mask) and np.any(c_base[zero_mask] > L):
            feasible = False
        if feasible and np.any(neg_mask):

        if feasible and np.any(pos_mask):

            feasible = False

                r2 = (1.0 + np.sqrt(disc)) / 2.0

        else:

    x_lo, x_hi = 0.0, 1.0

        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def exact_cd(h, n, max_passes, deadline):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            xn = solve_coord(c, h, p, n)
            if abs(xn - h0) > 1e-12:
                delta = xn - h0
                ms = np.arange(n - 1 - p, 2 * n - 1 - p)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                diag_i = np.where(ms == (n - 1))[0]
                c[ms] += delta * a
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-12:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def binary_local_search(h, n, deadline, max_flips=20000):
    """Local search on binary rounded version."""
    target = int(n // 2)
    hb = (h > 0.5).astype(float)
    if hb.sum() > target:
        order = np.argsort(hb)
        for i in range(len(order)):
            if hb.sum() == target:
                break
            if hb[order[i]] > 0.5:
                hb[order[i]] = 0.0
    elif hb.sum() < target:
        order = np.argsort(-hb)
        for i in range(len(order)):
            if hb.sum() == target:
                break
            if hb[order[i]] < 0.5:
                hb[order[i]] = 1.0
    c_cur = compute_c5(hb, n)
    for _ in range(max_flips):
        if time.time() > deadline:
            break
        p = np.random.randint(n)
        if hb[p] == 0.0:
            q = np.random.randint(n)
            if hb[q] == 1.0:
                hb[p], hb[q] = 1.0, 0.0
        else:
            q = np.random.randint(n)
            if hb[q] == 0.0:
                hb[p], hb[q] = 0.0, 1.0
        c_new = compute_c5(hb, n)
        if c_new > c_cur + 1e-15:
            hb[p], hb[q] = hb[q], hb[p]
    return hb, c_cur

def perturb_block(h, n, rng, block_len=20):
    """Perturb a random contiguous block."""
    hp = h.copy()
    start = rng.randint(0, max(1, n - block_len))
    seg = hp[start:start + block_len].copy()
    hp[start:start + block_len] = 1.0 - seg
    return project_feasible(hp, n)

def perturb_sinusoidal(h, n, rng):
    """Add a low-frequency sinusoidal perturbation."""
    freq = rng.uniform(1, 8)
    phase = rng.uniform(0, 2 * np.pi)
    amp = rng.uniform(0.02, 0.08)
    x = np.linspace(0, 2, n)
    hp = h + amp * np.sin(2 * np.pi * freq * x / 2.0 + phase)
    return project_feasible(hp, n)

def perturb_binary_flip(h, n, rng, frac=0.1):
    """Flip a random fraction of coordinates."""
    hp = h.copy()
    n_flip = max(1, int(frac * n))
    idx = rng.choice(n, n_flip, replace=False)
    hp[idx] = 1.0 - hp[idx]
    return project_feasible(hp, n)

    n_coarse = min(200, n_final)

        h = project_feasible(np.interp(x_new, x_coarse, h), ns)
        h, c = exact_cd(h, ns, 150, deadline)
        x_coarse = x_new
    return h, c
