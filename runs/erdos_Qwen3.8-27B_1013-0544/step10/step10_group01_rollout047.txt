                c = np.correlate(best_h, 1.0 - best_h, mode='full')
                top_k = np.argsort(c)[-5:]
                hp = best_h.copy()
                for lag in top_k:
                    off = int(lag) - (n_cur - 1)
                    lo = max(0, -off)
                    hi = min(n_cur, n_cur - off)
                    if hi > lo:
                        idx = np.random.permutation(np.arange(lo, hi))[:3]
                        hp[idx] = 1.0 - hp[idx]
                hp = project_feasible(hp, n_cur)
            elif r % 6 == 2:
                freq = 1.5 + 2.0 * rng.rand()
                hp = project_feasible(best_h + 0.02 * np.sin(np.pi * freq * x), n_cur)
            elif r % 6 == 3:
                blk = n_cur // 8
                st = rng.randint(0, n_cur - blk)
                hp = best_h.copy()
                hp[st:st+blk] = 1.0 - hp[st:st+blk]
                hp = project_feasible(hp, n_cur)
            elif r % 6 == 4:
                mask = (rng.rand(n_cur) < 0.04).astype(float)
                hp = project_feasible(best_h + 0.15 * (mask - 0.5), n_cur)
            else:
                hp = project_feasible(best_h * (1.0 + 0.015 * rng.normal(0, 1, n_cur)), n_cur)
            h_o, c_o = coord_descent(hp, n_cur, 60, deadline - 40, gs_iters=60, threshold=1e-5)
            if c_o < best_c:
                best_h, best_c = h_o, c_o

    # Phase 4: Try finer resolution
    if time.time() < deadline - 80:
        n_cur = len(best_h)
        for n_try in [n_cur + 25, n_cur - 25]:
            if time.time() > deadline - 50:
                break
            if n_try < 100:
                continue
            x_f = np.linspace(0, 2, n_try)
            h_f = project_feasible(np.interp(x_f, np.linspace(0, 2, n_cur), best_h), n_try)
            h_f, c_f = coord_descent(h_f, n_try, 50, deadline - 50, gs_iters=60, threshold=1e-5)
            if c_f < best_c:
                best_h, best_c = h_f, c_f
                n_cur = n_try

    # Phase 5: Final CD polish with tight threshold
    if time.time() < deadline - 15:
        n_cur = len(best_h)
        h_f, c_f = coord_descent(best_h, n_cur, 100, deadline - 10, gs_iters=70, threshold=1e-6)
        if c_f < best_c:
            best_h, best_c = h_f, c_f

    # Phase 6: Final SLSQP
    if time.time() < deadline - 8:
        n_cur = len(best_h)
        h_f, c_f = slsqp_optimize(best_h, n_cur, [500000, 2000000], 400)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
        best_h = project_feasible(best_h, n_cur)
        best_c = compute_c5(best_h, n_cur)

    return best_h, best_c, len(best_h)
