import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * dx)

def compute_all_c(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx, dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def sum_eq(x, n):
    return x.sum() - n / 2.0

def sum_eq_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alphas, maxiter_each=300):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (n,)}]
    for alpha in alphas:

                if new_c < best_c:

            pass
    return best_h, best_c

    c, dx = compute_all_c(h, n)
    # c has 2n-1 elements
    k_max = np.argmax(c)
    # Gradient of c[k_max] w.r.t. h[p]
    # c[k] = sum_i h[i] * (1 - h[i+k-(n-1)])
    # dc[k]/dh[p] = (1 - h[p+k-(n-1)]) [if valid] + h[i] * (-1) [where i+k-(n-1)=p]

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

    return x.sum() - n / 2.0

def coord_descent(h, n, max_passes, deadline, gs_iters=55, threshold=0.0):

    best_c = compute_c5(h, n)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for pass_num in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False

            if h0 <= 1e-13 or h0 >= 1.0 - 1e-13:

            c_aff = c[ms]

            d2 = d_lo + gr * (d_hi - d_lo)
            f1 = cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
            f2 = cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)

                    d2 = d1
                    f2 = f1

    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
