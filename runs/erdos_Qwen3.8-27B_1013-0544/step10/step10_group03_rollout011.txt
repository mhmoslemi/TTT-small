def get_c(h):
    gh = 1.0 - h
    if len(h) <= 128:

        return np.real(sig.correlate(h, gh, mode='full', method='fft'))
    except Exception:
        return np.correlate(h, gh, mode='full')


def calibrate_lag_sign():
    n = 140

    b = np.zeros(n)

    a[p] = 1.0
    b[q] = 1.0

    k = int(np.argmax(c))
    m_true = q - p
    if abs(k - ((n - 1) + m_true)) <= 1.0:
        return 1
    return -1


def project_box_sum(u, target):
    u = np.asarray(u, dtype=float).copy()
    if u.size == 0:
        return u
    target = float(target)
    a = float(np.min(u)) - 1.0

        s = float(np.clip(u - lam, 0.0, 1.0).sum())
        if s > target:
            a = lam
        else:
            b = lam
    h = np.clip(u - 0.5 * (a + b), 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-14:
        act = (h > 1e-14) & (h < 1.0 - 1e-14)

def full_from_half(u, n):

    h[m:] = u[::-1]

def smooth_obj_full(h, n, alpha, scale):

    m = float(np.max(c))
    y = alpha * (c - m)
    e = np.exp(y)

    if not np.isfinite(s) or s <= 0.0:

        e[int(np.argmax(c))] = 1.0
        s = 1.0
    return scale * (m + np.log(s) / alpha) * (2.0 / float(n))


def smooth_obj_half(u, n, alpha, scale):

            h_proj = project_box_sum(h_proj, target)
        c_proj = get_c(h_proj)
        new_val = float(np.max(c_proj) * (2.0 / float(n)))
        if new_val < c_best - 1e-12:
            h_best = h_proj
            c_best = new_val
            h = h_proj
            c = c_proj
        else:
            break
    h_best = project_box_sum(h_best, target)
    c_best = compute_c5(h_best, int(n))
    return h_best, c_best


def refine_half(u, n, stop, rng):
    n = int(n)
    target = n / 4.0
    u = project_box_sum(u, target)
    h = full_from_half(u, n)
    c_best = compute_c5(h, n)
    u_best = u.copy()
    h_best = h.copy()

    if stop - time.time() > 40.0:
        pg_stop = time.time() + min(220.0, 0.45 * (stop - time.time()))
        u1, c1, h1 = pgd_half(u_best, n, pg_stop, rng)
        if c1 < c_best - 1e-12:
            u_best, c_best, h_best = u1, c1, h1

    rem = stop - time.time()
    if rem > 30.0:
        m = n // 2
        for k in (10, 16, 24):
            if time.time() > stop - 12.0:
                break
            if k >= m:
                continue
            xm = (np.arange(m) + 0.5) / float(m)
            xk = (np.arange(k) + 0.5) / float(k)
            p0 = np.clip(np.interp(xk, xm, u_best), 0.0, 1.0)
            bounds = [(0.0, 1.0) for _ in range(k)]
            maxit = 70 if k <= 12 else 45
            try:
                res = opt.minimize(
                    param_obj, p0, method='Powell', bounds=bounds,
                    args=(n, k, 100.0, 1.0),
                    options={'maxiter': maxit, 'ftol': 1e-11, 'xtol': 1e-9}
                )
                u2 = param_build_half(np.clip(res.x, 0.0, 1.0), n, k)
                h2 = full_from_half(u2, n)
                c2 = compute_c5(h2, n)
                if c2 < c_best - 1e-12:
                    u_best, c_best, h_best = u2, c2, h2
            except Exception:
                pass
            if rng is not None and time.time() < stop - 18.0:
                pr = rng.uniform(0.0, 1.0, k)
                try:
                    res = opt.minimize(
                        param_obj, pr, method='Powell', bounds=bounds,
                        args=(n, k, 100.0, 1.0),
                        options={'maxiter': 35, 'ftol': 1e-11, 'xtol': 1e-9}
                    )
                    u3 = param_build_half(np.clip(res.x, 0.0, 1.0), n, k)
                    h3 = full_from_half(u3, n)
                    c3 = compute_c5(h3, n)
                    if c3 < c_best - 1e-12:
                        u_best, c_best, h_best = u3, c3, h3
                except Exception:
                    pass

    if time.time() < stop - 12.0:
        h4, c4 = approx_cd(h_best, n, 5, stop - 2.0, rng, True)
        if c4 < c_best - 1e-12:
            c_best = c4
            h_best = h4
            u_best = to_half(h4, n)

    h_best = project_box_sum(h_best, n / 2.0)
    c_best = compute_c5(h_best, n)
    u_best = to_half(h_best, n)
    return u_best, c_best, h_best


def run(seed=42, budget_s=1100, **kwargs):
    global LAG_SIGN
    t0 = time.time()
    deadline = t0 + float(budget_s) - 4.0
    LAG_SIGN = calibrate_lag_sign()
    rng = np.random.RandomState(seed)

    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n_best = int(len(h_best))
    except Exception:
        h_best = np.full(700, 0.5)
        n_best = 700

    if n_best <= 0:
        h_best = np.full(700, 0.5)
        n_best = 700
    if n_best % 2:
        h_best = np.concatenate([h_best, h_best[-1:]])
        n_best += 1
    h_best = project_box_sum(h_best, n_best / 2.0)
    best_c = compute_c5(h_best, n_best)
    best_n = n_best

    cands = []
    raw = [n_best, n_best - 80, n_best - 40, n_best + 40, n_best + 80, 600, 750, 850, 950]
    for v in raw:
        vv = int(v)
        if vv % 2:
            vv -= 1
        if 300 <= vv <= 1000 and vv not in cands:
            cands.append(vv)
    if best_n not in cands:
        cands.insert(0, best_n)
    cands = cands[:5]

    for idx, cand in enumerate(cands):
        rem = deadline - time.time()
        if rem < 100.0:
            break
        if idx == 0:
            alloc = min(380.0, rem - 160.0)
        elif idx == 1:
            alloc = min(260.0, rem - 120.0)
        elif idx == 2:
            alloc = min(180.0, rem - 80.0)
        else:
            alloc = min(120.0, rem - 40.0)
        if alloc < 60.0:
            break
        stop = time.time() + alloc
        if stop > deadline - 30.0:
            stop = deadline - 30.0

        if cand == len(h_best):
            h_start = h_best.copy()
        else:
            x_old = (np.arange(len(h_best)) + 0.5) * (2.0 / len(h_best))
            x_new = (np.arange(cand) + 0.5) * (2.0 / cand)
            h_start = np.interp(x_new, x_old, h_best)
        h_start = project_box_sum(h_start, cand / 2.0)

        u0 = to_half(h_start, cand)
        u1, c1, h1 = refine_half(u0, cand, stop, rng)
        if c1 < best_c - 1e-12:
            best_c = c1
            best_n = cand
            h_best = h1

        rem = deadline - time.time()
        if rem > 70.0:
            stop2 = time.time() + min(70.0, rem - 50.0)
            if stop2 > deadline - 10.0:
                stop2 = deadline - 10.0
            base = h1 if c1 < compute_c5(h_start, cand) else h_start
            h2, c2 = approx_cd(base, cand, 5, stop2, rng, False)
            if c2 < best_c - 1e-12:
                best_c = c2
                best_n = cand
                h_best = h2

    rem = deadline - time.time()
    if rem > 70.0:
        h3, c3 = approx_cd(h_best, len(h_best), 10, deadline - 3.0, rng, False)
        if c3 < best_c - 1e-12:
            best_c = c3
            best_n = len(h3)
            h_best = h3

    h_best = project_box_sum(h_best, len(h_best) / 2.0)
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
