import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def lse_obj_and_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    exps = np.exp(shifted - mx)
    s = exps.sum()
    dx = 2.0 / n
    val = (m + mx / alpha + np.log(s) / alpha) * dx
    w = exps / s
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        wm = w[m_idx]
        if wm < 1e-14:
            continue
        s_val = m_idx - (n - 1)
        if s_val >= 0:
            ah = np.zeros(n)
            ah[s_val:] = 1.0 - h[:n - s_val]
            bh = np.zeros(n)
            bh[:n - s_val] = h[s_val:]
        else:
            ah = np.zeros(n)
            ah[:n + s_val] = 1.0 - h[-s_val:]
            bh = np.zeros(n)
            bh[-s_val:] = h[:n + s_val]
        grad += wm * (ah - bh)
    grad *= dx
    return val, grad

def solve_1d(c, h, p, n):

    if np.any(vb):

def update_corr(c, h, p, old_val, new_val, n):
    delta = new_val - old_val
    if abs(delta) < 1e-14:
        return c
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    diag_i = np.where(ms == (n - 1))[0]
    c[ms] += delta * a
    if len(diag_i) > 0:
        c[ms[diag_i[0]]] -= delta * delta
    return c

def exact_cd(h, n, max_passes, deadline, rng=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for it in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        order = list(range(n))
        if rng is not None and it % 2 == 1:
            rng.shuffle(order)
        for p in order:
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            xn = solve_1d(c, h, p, n)
            if abs(xn - h0) > 1e-12:
                update_corr(c, h, p, h0, xn, n)
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

        def cj(x):
            return np.ones(n)
        cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]
        try:
            res = minimize(lse_obj_and_grad, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons, jac=True,
                           options={'maxiter': maxiter, 'ftol': 1e-16})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c - 1e-12:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def interp_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def block_perturb(h, n, rng, block_frac=0.15):
    h_new = h.copy()
    blk = max(2, int(n * block_frac))
    start = rng.randint(0, n - blk)
    h_new[start:start+blk] = rng.uniform(0, 1, blk)
    return project_feasible(h_new, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = np.full(500, 0.5)
        n = 500
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)

    # Phase 1: CD + SLSQP on parent
    h1, c1 = exact_cd(h_best, n, 400, deadline, rng)
    if c1 < best_c:
        h_best, best_c = h1, c1
    h2, c2 = slsqp_opt(h_best, n, [3000, 10000, 40000, 150000, 500000], 300, deadline)
    if c2 < best_c - 1e-12:
        h_best, best_c = h2, c2
    h3, c3 = exact_cd(h_best, n, 300, deadline, rng)
    if c3 < best_c - 1e-12:
        h_best, best_c = h3, c3

    # Phase 2: Multi-resolution sweep
    n_list = [600, 650, 750, 800, 850, 900, 950, 1050, 1100, 550, 500, 450, 400]
    n_curr = len(h_best)
    for n_try in n_list:
        if time.time() > deadline - 150:
            break
        elapsed = time.time() - t0
        remaining = budget_s - elapsed - 150
        if remaining < 30:
            break
        h_int = interp_h(h_best, n_curr, n_try)
        budget_per_n = min(60, remaining / max(1, len(n_list)))
        dl = min(deadline, time.time() + budget_per_n)
        h_o, c_o = exact_cd(h_int, n_try, 150, dl, rng)
        if c_o < best_c - 1e-9:
            h_best, best_c, n_curr = h_o, c_o, n_try
        if time.time() < deadline - 100:
            h_o2, c_o2 = slsqp_opt(h_o, n_try, [5000, 30000, 100000], 150, min(deadline, time.time()+20))
            if c_o2 < best_c - 1e-9:
                h_best, best_c, n_curr = h_o2, c_o2, n_try
            h_o3, c_o3 = exact_cd(h_best, n_curr, 100, min(deadline, time.time()+15), rng)
            if c_o3 < best_c - 1e-9:
                h_best, best_c, n_curr = h_o3, c_o3, n_try

    # Phase 3: Block perturbations + CD
    n = len(h_best)
    for r in range(30):
        if time.time() > deadline - 60:
            break
        hp = block_perturb(h_best, n, rng, block_frac=0.1 + 0.05 * rng.rand())
        h_o, c_o = exact_cd(hp, n, 100, min(deadline, time.time() + 25), rng)
        if c_o < best_c - 1e-9:
            h_best, best_c = h_o, c_o
        hgp = project_feasible(h_best + rng.normal(0, 0.03, n), n)
        h_o2, c_o2 = exact_cd(hgp, n, 80, min(deadline, time.time() + 20), rng)
        if c_o2 < best_c - 1e-9:
            h_best, best_c = h_o2, c_o2

    # Phase 4: Final SLSQP + CD polish
    n = len(h_best)
    h_f, c_f = slsqp_opt(h_best, n, [2000, 8000, 30000, 100000, 400000], 200, deadline)
    if c_f < best_c - 1e-12:
        h_best, best_c, n = h_f, c_f, n
    h_final, c_final = exact_cd(h_best, n, 500, deadline, rng)
    if c_final < best_c - 1e-12:
        h_best, best_c = h_final, c_final

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
