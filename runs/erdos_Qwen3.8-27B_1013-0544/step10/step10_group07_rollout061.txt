import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_box_sum(y, target):
    """Project y onto {0 <= x <= 1, sum(x) = target}"""
    n = len(y)
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    x = np.clip(y - lam, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-14:
        free = (x > 1e-13) & (x < 1.0 - 1e-13)
        if np.any(free):
            x[free] += diff / free.sum()
            x = np.clip(x, 0.0, 1.0)
    return x

def smooth_grad(y, n, alpha):
    """Gradient of smooth approximation (1/alpha)*log(sum(exp(alpha*corr*dx)))"""
    dx = 2.0 / n
    corr = np.correlate(y, 1.0 - y, mode='full')
    shifted = alpha * corr * dx
    shifted -= shifted.max()
    p = np.exp(shifted)
    p /= p.sum()

    term1 = np.correlate(1.0 - y, p, mode='full')

def projected_gradient(h, n, alpha, max_iter, deadline):
    """Projected gradient for smooth approximation"""
    dx = 2.0 / n
    h = h.copy()
    y = h.copy()
    t_k = 1.0
    step = 1.0 / (4.0 * dx * n)  # conservative initial step
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        corr = np.correlate(y, 1.0 - y, mode='full')
        shifted = alpha * corr * dx
        shifted -= shifted.max()
        p = np.exp(shifted)
        p /= p.sum()
        
        term1 = np.correlate(1.0 - y, p, mode='full')
        term2 = np.convolve(p, y, mode='full')
        grad = dx * (term1[n-1:2*n-1] - term2[n-1:2*n-1])
        
        # Backtracking line search
        for _ in range(30):
            new_h = project_box_sum(y - step * grad, n / 2.0)
            new_corr = np.correlate(new_h, 1.0 - new_h, mode='full')
            new_shifted = alpha * new_corr * dx
            new_shifted -= new_shifted.max()
            new_val = (new_shifted.max() + np.log(np.exp(new_shifted).sum())) / alpha * dx
            old_corr = np.correlate(y, 1.0 - y, mode='full')
            old_shifted = alpha * old_corr * dx
            old_shifted -= old_shifted.max()
            old_val = (old_shifted.max() + np.log(np.exp(old_shifted).sum())) / alpha * dx
            
            if new_val <= old_val + 1e-12:
                break
            step *= 0.5
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y_new = new_h + ((t_k - 1.0) / t_new) * (new_h - h)
        h = new_h
        y = np.clip(y_new, -2.0, 3.0)
        t_k = t_new
    
    return h
