import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def build_jacobian(h, n):
    J = np.zeros((2 * n - 1, n))
    for m in range(2 * n - 1):
        lag = m - (n - 1)
        if lag >= 0:
            idx1 = np.arange(0, n - lag)
            J[m, idx1] += (1.0 - h[idx1 + lag])
            if lag > 0:
                idx2 = np.arange(lag, n)
                J[m, idx2] -= h[idx2 - lag]
        else:
            idx1 = np.arange(-lag, n)
            J[m, idx1] += (1.0 - h[idx1 + lag])
            idx2 = np.arange(0, n + lag)
            J[m, idx2] -= h[idx2 - lag]
    return J

def solve_lp_step(h, n, dmax):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    J = build_jacobian(h, n)
    delta = cp.Variable(n)
    C = cp.Variable()
    lb = -np.minimum(h, dmax)
    ub = np.minimum(1.0 - h, dmax)
    cons = [C - dx * (J @ delta) >= dx * corr,
            cp.sum(delta) == 0,
            delta >= lb, delta <= ub]
    prob = cp.Problem(cp.Minimize(C), cons)
    try:
        prob.solve(solver=cp.GLOP)
    except Exception:
        try:
            prob.solve(solver=cp.GLPK)
        except Exception:
            return None, None
    if delta.value is None:
        return None, None
    return delta.value, float(C.value)

        c_best = c_cur

        else:

        if C_lp is not None and C_lp >= best_c - 1e-9:
            break
    return h, best_c

def cd_1d(c, h, p, n):
    """Exact 1D coordinate descent for variable p."""
    h0 = h[p]
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms].copy()
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    if has_diag:
        a[diag_idx] = 0.0
    unaff = 0.0
    if m_min > 0:
        unaff = max(unaff, float(np.max(c[:m_min])))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    cur_max = float(np.max(c))
    L_lo, L_hi = unaff, cur_max
    for _ in range(40):
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        ok = True
        zero_mask = np.abs(a) <= 1e-15
        neg_mask = a < -1e-15
        pos_mask = a > 1e-15
        if np.any(zero_mask) and np.any(c_base[zero_mask] > L + 1e-15):
            ok = False
        if ok and np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if ok and np.any(pos_mask):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        if x_lo > x_hi + 1e-14:
            ok = False
        if ok and has_diag:
            c_diag = c_base[diag_idx[0]]
            Cq = L - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                if not (max(x_lo, 0.0) < r1 and min(x_hi, 1.0) > r2):
                    ok = False
        if ok:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    neg_mask = a < -1e-15
    pos_mask = a > 1e-15
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def exact_cd(h, n, max_passes, deadline):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):

            if h0 < 1e-12 or h0 > 1.0 - 1e-12:

                m_min = n - 1 - p

                ms = np.arange(m_min, m_max + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))

                h[p] = xn
                improved = True

            alpha *= 0.5

            if C_lp is not None and C_lp >= best_c - 1e-9:
                break
            dmax *= 0.5
        else:
            dmax = min(0.12, dmax * 1.3)
        if dmax < 1e-6:
            break
    return h, best_c

def cd_1d(c, h, p, n):
    h0 = h[p]
    mlo = n - 1 - p
    mhi = 2 * n - 2 - p
    ms = np.arange(mlo, mhi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += 1.0 - h[jf[vf]]
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms].copy()
    di = np.where(ms == n - 1)[0]
    has_d = len(di) > 0
    if has_d:
        a[di] = 0.0
    unaff = 0.0
    if mlo > 0:
        unaff = max(unaff, float(c[:mlo].max()))
    if mhi < 2 * n - 2:
        unaff = max(unaff, float(c[mhi + 1:].max()))
    Llo, Lhi = unaff, float(c.max())
    for _ in range(40):
        L = (Llo + Lhi) * 0.5
        xlo, xhi = 0.0, 1.0
        ok = True
        zm = np.abs(a) <= 1e-15
        nm = a < -1e-15
        pm = a > 1e-15
        if np.any(zm) and c_base[zm].max() > L + 1e-15:
            ok = False
        if ok and np.any(nm):
            xlo = max(xlo, ((L - c_base[nm]) / a[nm]).max())
        if ok and np.any(pm):
            xhi = min(xhi, ((L - c_base[pm]) / a[pm]).min())
        if xlo > xhi + 1e-14:
            ok = False
        if ok and has_d:
            cq = L - c_base[di[0]] + h0 * (1.0 - h0)
            disc = 1.0 - 4.0 * cq
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) * 0.5
                r2 = (1.0 + np.sqrt(disc)) * 0.5
                if not (max(xlo, 0.0) < r1 and min(xhi, 1.0) > r2):
                    ok = False
        if ok:
            Lhi = L
        else:
            Llo = L
    L = Lhi
    xlo, xhi = 0.0, 1.0
    nm = a < -1e-15
    pm = a > 1e-15
    if np.any(nm):
        xlo = max(xlo, ((L - c_base[nm]) / a[nm]).max())
    if np.any(pm):
        xhi = min(xhi, ((L - c_base[pm]) / a[pm]).min())
    xlo, xhi = max(xlo, 0.0), min(xhi, 1.0)
    if xlo > xhi:
        return h0
    return float(np.clip((xlo + xhi) * 0.5, 0.0, 1.0))

def exact_cd(h, n, max_passes, deadline, rng=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    order_all = list(range(n))
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        if rng is not None:
            order = order_all[:]
            rng.shuffle(order)
        else:
            order = order_all
        imp = False
        for p in order:
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            xn = cd_1d(c, h, p, n)
            if abs(xn - h0) > 1e-12:
                d = xn - h0
                mlo, mhi = n - 1 - p, 2 * n - 2 - p
                ms = np.arange(mlo, mhi + 1)
                jf, jb = p + ms - (n - 1), p - ms + (n - 1)
                vf, vb = (jf >= 0) & (jf < n), (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += 1.0 - h[jf[vf]]
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                c[ms] += d * a
                di = np.where(ms == n - 1)[0]
                if len(di) > 0:
                    c[ms[di[0]]] -= d * d
                h[p] = xn
                imp = True
        h = project_feasible(h, n)
        if not imp:
            break
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 10
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = np.full(500, 0.5)
        n = 500
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)

    # Phase 1: SLP with LP
    h1, c1 = slp_phase(h_best, n, 60, deadline)
    if c1 < best_c:
        h_best, best_c = h1, c1

    # Phase 2: CD
    h2, c2 = exact_cd(h_best, n, 150, deadline)
    if c2 < best_c:
        h_best, best_c = h2, c2

    # Phase 3: SLP
    h3, c3 = slp_phase(h_best, n, 50, deadline)
    if c3 < best_c:
        h_best, best_c = h3, c3

    # Phase 4: CD (randomized)
    h4, c4 = exact_cd(h_best, n, 200, deadline, rng)
    if c4 < best_c:
        h_best, best_c = h4, c4

    # Phase 5: SLP + CD polish
    h5, c5v = slp_phase(h_best, n, 40, deadline)
    if c5v < best_c:
        h_best, best_c = h5, c5v
    h6, c6 = exact_cd(h_best, n, 300, deadline)
    if c6 < best_c:
        h_best, best_c = h6, c6

    # Phase 6: Try nearby n values
    for nt in [650, 750, 800]:
        if time.time() > deadline - 150:
            break
        if nt == n:
            continue
        xo = np.linspace(0, 2, len(h_best))
        xw = np.linspace(0, 2, nt)
        hi = project_feasible(np.interp(xw, xo, h_best), nt)
        ho, co = slp_phase(hi, nt, 40, deadline)
        ho, co = exact_cd(ho, nt, 200, deadline)
        if co < best_c - 1e-8:
            h_best, best_c, n = ho, co, nt

    # Phase 7: Final polish
    if time.time() < deadline - 30:
        hf, cf = slp_phase(h_best, n, 30, deadline)
        if cf < best_c:
            h_best, best_c = hf, cf
        hf2, cf2 = exact_cd(h_best, n, 400, deadline)
        if cf2 < best_c:
            h_best, best_c = hf2, cf2

    # Phase 8: One more SLP
    if time.time() < deadline - 15:
        hf3, cf3 = slp_phase(h_best, n, 20, deadline)
        if cf3 < best_c:
            h_best, best_c = hf3, cf3

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
