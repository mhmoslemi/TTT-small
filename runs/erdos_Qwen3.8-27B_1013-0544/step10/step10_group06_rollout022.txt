    dx = 2.0/n
    c = np.correlate(h, 1.0-h, mode='full')
    # Soft constraint: Σ max(0, c_k - T/dx)²
    violations = np.maximum(0, c - T/dx)
    penalty = violations @ violations
    # Equality constraint
    eq_viol = (h.sum() - n/2.0) ** 2
    return penalty + mu * eq_viol

    dx = 2.0/n
    c = np.correlate(h, 1.0-h, mode='full')
    viol = np.maximum(0.0, c - T/dx)
    w = 2.0 * viol  # weights for each k
    active = viol > 0
    if not np.any(active):
        obj = mu * (h.sum() - n/2.0)**2
        grad = 2.0 * mu * (h.sum() - n/2.0)
        return obj, grad * np.ones(n)
    
    grad = np.zeros(n)
    for k in np.where(active)[0]:
        wk = w[k]
        s = k - (n-1)
        if s == 0:
            grad += wk * (1.0 - 2.0*h)
        elif s > 0:

            grad[:n-m] += -wk * h[m:]

    grad += eq_term
    obj = (viol @ viol) + mu * (h.sum() - n/2.0)**2

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def build_full(half, n):
    h = np.empty(n)
    h[:n//2] = half
    h[n//2:] = 1.0 - half[::-1]
    return h

def c5_from_half(half, n):
    h = build_full(half, n)
    return compute_c5(h, n)

def penalty_obj_and_grad_half(half, n, T):
    h = build_full(half, n)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    viol = np.maximum(0.0, c - T / dx)
    obj = float(viol @ viol)
    if obj < 1e-20:
        return obj, np.zeros_like(half)
    w = 2.0 * viol
    # Compute full gradient
    nfull = 2 * n - 1
    grad_full = np.zeros(n)
    active = viol > 0
    for k in np.where(active)[0]:
        wk = w[k]
        s = k - (n - 1)
        if s == 0:
            grad_full += wk * (1.0 - 2.0 * h)
        elif s > 0:
            grad_full[:n - s] += wk * (1.0 - h[s:])
            grad_full[s:] -= wk * h[:n - s]
        else:
            m = -s
            grad_full[m:] += wk * (1.0 - h[:n - m])
            grad_full[:n - m] -= wk * h[m:]
    # Project gradient to half
    nh = n // 2
    grad_half = grad_full[:nh] - grad_full[nh:][::-1]
    return obj, grad_half

def lse_obj_grad_half(half, n, alpha):
    h = build_full(half, n)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    weights = np.exp(shifted - mx)
    weights /= weights.sum()
    lse = m + (mx + np.log(weights.sum())) / alpha
    obj = lse * dx
    # Gradient
    grad_full = np.zeros(n)
    for k in range(2 * n - 1):
        wk = weights[k]
        if wk < 1e-15:
            continue
        s = k - (n - 1)
        if s == 0:
            grad_full += wk * (1.0 - 2.0 * h)
        elif s > 0:

            grad_full[s:] -= wk * h[:n - s]
        else:
            m2 = -s
            grad_full[m2:] += wk * (1.0 - h[:n - m2])
            grad_full[:n - m2] -= wk * h[m2:]

def cd_full(h, n, max_passes, deadline, rng, nbisect=35):
    """Standard coordinate descent on full h."""
    h = h.copy()
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        order = rng.permutation(n)
        for p in order:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            x_new = _solve_coord(c, h, p, n, nbisect)
            if abs(x_new - h0) > 1e-12:
                h[p] = x_new
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def _solve_coord(c, h, p, n, nbisect=35):
    """Solve the 1D subproblem for coordinate p."""
    h0 = h[p]
    lo_a = max(0, n - 1 - p)
    hi_a = min(2 * n - 2, n - 1 + p)
    lo_b = max(0, p)
    hi_b = min(2 * n - 2, p + n - 1)
    m_lo = min(lo_a, lo_b)
    m_hi = max(hi_a, hi_b)
    ms = np.arange(m_lo, m_hi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms].copy()
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    c_diag = 0.0
    if has_diag:
        c_diag = c_base[diag_idx[0]]
        a[diag_idx] = 0.0
    non_diag = np.ones(len(ms), dtype=bool)
    if has_diag:
        non_diag[diag_idx] = False
    a_nd = a[non_diag]
    c_nd = c_base[non_diag]
    unaff_max = 0.0
    if m_lo > 0:
        unaff_max = max(unaff_max, float(np.max(c[:m_lo])))
    if m_hi < 2 * n - 2:
        unaff_max = max(unaff_max, float(np.max(c[m_hi + 1:])))
    cur_max = float(np.max(c))
    L_lo = unaff_max
    L_hi = cur_max
    for _ in range(nbisect):
        if L_hi - L_lo < 1e-13:
            break
        T = (L_lo + L_hi) / 2.0
        feasible = True
        if unaff_max > T + 1e-15:

        elif feasible and len(a_nd) > 0:

            x_hi_c = 1.0

            if np.any(pos):

            if x_lo_c > x_hi_c + 1e-14:

                    xl = max(xl, 0.0)

    x_lo = max(x_lo, 0.0)

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)

def build_full(half, n):

    h = np.empty(n)
    h[:nh] = half
    h[nh:] = 1.0 - half[::-1]

        else:
            m2 = -s

            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def antisym(h, n):
    nh = n // 2
    h2 = h.copy()
    h2[:nh] = 0.5 * (h[:nh] + 1.0 - h[nh:][::-1])
    h2[nh:] = 1.0 - h2[:nh][::-1]
    return project_feasible(h2, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 700
        x = np.linspace(0, 2, n)
        h = np.clip(0.5 + 0.3 * np.sin(np.pi * x), 0, 1)
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    nh = n // 2
    bounds = [(0.0, 1.0)] * nh
    # Phase 1: Quick CD
    h1, c1 = cd_full(h, n, 8, deadline, rng, 30)
    if c1 < best_c:
        best_h, best_c = h1, c1
    # Phase 2: Antisymmetrize
    h2 = antisym(best_h, n)
    c2 = compute_c5(h2, n)
    if c2 < best_c:
        best_h, best_c = h2, c2
    half_best = best_h[:nh].copy()
    # Phase 3: L-BFGS-B with LSE (anti-symmetric)
    for alpha in [2000, 8000, 30000, 80000]:
        if time.time() > deadline - 120:
            break
        try:
            res = minimize(lse_obj_half, half_best, args=(n, alpha),
                           jac=lse_jac_half, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': 600, 'ftol': 1e-16, 'gtol': 1e-9})
            half_opt = np.clip(res.x, 0, 1)
            h_opt = build_full(half_opt, n)
            c_opt = compute_c5(h_opt, n)
            if c_opt < best_c:
                best_h, best_c = h_opt, c_opt
                half_best = half_opt
        except Exception:
            pass
    # Phase 4: CD polish
    if time.time() < deadline - 40:
        h3 = antisym(best_h, n)
        rem = deadline - time.time() - 5
        mp = max(5, int(rem / 2.5))
        h3, c3 = cd_full(h3, n, mp, deadline, rng, 38)
        if c3 < best_c:
            best_h, best_c = h3, c3
    # Phase 5: Bisection on T with penalty
    T_hi = best_c
    T_lo = 0.32
    half_bis = best_h[:nh].copy()
    for _ in range(30):
        if time.time() > deadline - 60:
            break
        T = (T_lo + T_hi) / 2.0
        try:
            res = minimize(penalty_obj_half, half_bis, args=(n, T),
                           jac=penalty_jac_half, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': 800, 'ftol': 1e-16, 'gtol': 1e-9})
            half_opt = np.clip(res.x, 0, 1)
            h_opt = build_full(half_opt, n)
            c_opt = compute_c5(h_opt, n)
            if c_opt <= T * (1 + 1e-5) + 1e-9:
                T_hi = T
                half_bis = half_opt
            else:
                T_lo = T
            if c_opt < best_c:
                best_h, best_c = h_opt, c_opt
                half_bis = half_opt
        except Exception:
            break
    # Phase 6: Random restarts (anti-symmetric)
    if time.time() < deadline - 80:
        for _restart in range(5):
            if time.time() > deadline - 30:
                break
            half_r = 0.5 + rng.randn(nh) * 0.15
            half_r = np.clip(half_r, 0, 1)
            half_r = 0.5 * (half_r + 1.0 - half_r[::-1])
            half_r = np.clip(half_r, 0, 1)
            try:
                res = minimize(lse_obj_half, half_r, args=(n, 20000),
                               jac=lse_jac_half, method='L-BFGS-B',
                               bounds=bounds,
                               options={'maxiter': 400, 'ftol': 1e-14, 'gtol': 1e-7})
                half_opt = np.clip(res.x, 0, 1)
                h_opt = build_full(half_opt, n)
                h_opt, c_opt = cd_full(h_opt, n, 10, deadline, rng, 30)
                if c_opt < best_c:
                    best_h, best_c = h_opt, c_opt
            except Exception:
                pass
    # Phase 7: Final CD polish
    if time.time() < deadline - 3:
        rem = deadline - time.time() - 2
        mp = max(3, int(rem / 2.0))
        h_f, c_f = cd_full(best_h, n, mp, deadline, rng, 40)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
