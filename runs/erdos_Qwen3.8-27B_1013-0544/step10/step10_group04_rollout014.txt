_G_MU = 1.0e5


def project_box_sum(y, S):
    y = np.asarray(y, dtype=float)
    h = np.clip(y, 0.0, 1.0)
    s = float(h.sum())
    if abs(s - S) < 1e-10:
        return h
    lo = -float(h.max()) - 1.0
    hi = 1.0 - float(h.min()) + 1.0
    for _ in range(70):
        lam = 0.5 * (lo + hi)
        hh = np.clip(y + lam, 0.0, 1.0)
        ss = float(hh.sum())
        if ss < S:
            lo = lam
        else:
            hi = lam
    lam = 0.5 * (lo + hi)
    h = np.clip(y + lam, 0.0, 1.0)
    diff = S - float(h.sum())
    if abs(diff) > 1e-14:
        idx = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if idx.size:
            h[idx[0]] = float(np.clip(h[idx[0]] + diff, 0.0, 1.0))
        else:
            h = np.clip(h + diff / float(h.size), 0.0, 1.0)
    return h


def raw_value(h, n):
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(c.max()) * (2.0 / float(n))


def smooth_value(h, n, alpha):
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode="full")
    v = c * dx
    mx = float(v.max())
    z = alpha * (v - mx)
    z -= float(z.max())
    ew = np.exp(z)
    s = float(ew.sum())
    F = mx + (float(z.max()) + float(np.log(s))) / float(alpha)
    return F, mx


def eval_smooth_grad(h, n, alpha, pre):
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode="full")
    v = c * dx
    mx = float(v.max())
    z = alpha * (v - mx)
    z -= float(z.max())
    ew = np.exp(z)
    s = float(ew.sum())
    w = ew / s
    F = mx + (float(z.max()) + float(np.log(s))) / float(alpha)
    one = 1.0 - h
    g = np.empty(n, dtype=float)
    kAs = pre["kA"]
    kBs = pre["kB"]
    for p in range(n):
        g[p] = float(np.dot(w[kAs[p]], one) - np.dot(w[kBs[p]], h))
    g *= dx
    return F, g, mx


def precompute(n):
    m = 2 * n - 1
    kAl = []
    kB_l = []
    U_l = []
    posA_l = []
    posB_l = []
    comp_l = []
    dp_l = []
    base = np.arange(n, dtype=np.int64)
    for p in range(n):
        kA = np.int64(n - 1 + p - base)
        kB = np.int64(n - 1 - p + base)
        U = np.unique(np.concatenate((kA, kB)))
        mask = np.zeros(m, dtype=bool)
        mask[U] = True
        comp = np.flatnonzero(~mask)
        posA = np.searchsorted(U, kA)
        posB = np.searchsorted(U, kB)
        dp = int(np.searchsorted(U, n - 1))
        if dp >= U.size or int(U[dp]) != n - 1:
            dp = -1
        kAl.append(kA)
        kB_l.append(kB)
        U_l.append(U)
        posA_l.append(posA)
        posB_l.append(posB)
        comp_l.append(comp)
        dp_l.append(dp)
    return {
        "kA": kAl,
        "kB": kB_l,
        "U": U_l,
        "posA": posA_l,
        "posB": posB_l,
        "comp": comp_l,
        "dp": dp_l,
    }


def linear_bounds(L, h0, c, a, pos, neg, zero):
    if zero.size and np.any(c[zero] > L + 1e-11):
        return None
    lo = 0.0
    hi = 1.0
    if pos.size:
        u = h0 + float(np.min((L - c[pos]) / a[pos]))
        if u < hi:
            hi = u
    if neg.size:
        l = h0 + float(np.max((L - c[neg]) / a[neg]))
        if l > lo:
            lo = l
    if lo < 0.0:
        lo = 0.0
    if hi > 1.0:
        hi = 1.0
    if lo > hi + 1e-11:
        return None
    return lo, hi


def diag_check(L, h0, dval, lo, hi):
    Cq = L - dval + h0 - h0 * h0
    disc = 1.0 - 4.0 * Cq
    if disc < 0.0:
        return True
    r1 = 0.5 * (1.0 - float(np.sqrt(disc)))
    r2 = 0.5 * (1.0 + float(np.sqrt(disc)))
    return (min(hi, r1) >= lo) or (max(lo, r2) <= hi)


def point_value(x, c_aff, a, dp, h0, unaff):
    d = x - h0
    m = c_aff + a * d
    if dp >= 0:
        m[dp] = c_aff[dp] + (1.0 - 2.0 * h0) * d - d * d
    val = float(m.max())
    if unaff > val:
        val = unaff
    return val


def coord_line(c, h, p, n, pre):
    U = pre["U"][p]
    h0 = h[p]
    a = np.zeros(U.size, dtype=float)
    a[pre["posA"][p]] += 1.0 - h
    a[pre["posB"][p]] -= h
    c_aff = c[U]
    dp = pre["dp"][p]
    comp = pre["comp"][p]
    unaff = float(np.max(c[comp])) if comp.size else -1.0e30
    cur = max(unaff, float(c_aff.max()))

    a_lin = a.copy()
    c_lin = c_aff.copy()
    if dp >= 0:
        a_lin[dp] = 0.0
        c_lin[dp] = -1.0e30

    pos = np.nonzero(a_lin > 1e-15)[0]
    neg = np.nonzero(a_lin < -1e-15)[0]
    zero = np.nonzero((a_lin >= -1e-15) & (a_lin <= 1e-15))[0]

    Llo = 0.0 if unaff < 0.0 else unaff
    Lhi = cur
    if Lhi <= Llo + 1e-12:
        return h0, None, dp

    dval = c_aff[dp] if dp >= 0 else 0.0
    for _ in range(34):
        L = 0.5 * (Llo + Lhi)
        b = linear_bounds(L, h0, c_lin, a_lin, pos, neg, zero)
        if b is None:
            Llo = L
        else:
            lo, hi = b
            if dp >= 0 and not diag_check(L, h0, dval, lo, hi):
                Llo = L
            else:
                Lhi = L

    L = Lhi
    b = linear_bounds(L, h0, c_lin, a_lin, pos, neg, zero)
    if b is None:
        return h0, None, dp
    lo, hi = b

    intervals = []
    if dp >= 0:
        Cq = L - dval + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc < 0.0:
            intervals.append((lo, hi))
        else:
            r1 = 0.5 * (1.0 - float(np.sqrt(disc)))
            r2 = 0.5 * (1.0 + float(np.sqrt(disc)))
            intervals.append((lo, min(hi, r1)))
            intervals.append((max(lo, r2), hi))
    else:
        intervals.append((lo, hi))

    cands = [h0]
    for al, ah in intervals:
        if al < 0.0:
            al = 0.0
        if ah > 1.0:
            ah = 1.0
        if al <= ah:
            cands.append(al)
            cands.append(ah)
            cands.append(0.5 * (al + ah))

    bestx = h0
    bestm = cur
    for x in cands:
        if x < 0.0:
            x = 0.0
        elif x > 1.0:
            x = 1.0
        m = point_value(x, c_aff, a, dp, h0, unaff)
        if m < bestm:
            bestm = m
            bestx = x

    if bestm < cur - 1e-10:
        return bestx, a, dp
    return h0, None, dp


def cd_refine(h, n, pre, passes, deadline, rng, random=False, skip_bounds=True):
    S = 0.5 * float(n)
    h = project_box_sum(h, S)
    c = np.correlate(h, 1.0 - h, mode="full")
    best_h = h.copy()
    best_c = float(c.max()) * (2.0 / float(n))
    passes = int(max(0, passes))

    for _ in range(passes):
        if time.time() > deadline:
            break
        if random:
            order = rng.permutation(n)
        else:
            order = np.arange(n)
        improved = False
        for p in order:
            if time.time() > deadline:
                break
            hp = h[p]
            if skip_bounds and (hp <= 1e-13 or hp >= 1.0 - 1e-13):
                continue
            res = coord_line(c, h, p, n, pre)
            xn = res[0]
            if abs(xn - hp) > 1e-12:
                delta = xn - hp
                U = pre["U"][p]
                dp = pre["dp"][p]
                c[U] += delta * res[1]
                if dp >= 0:
                    c[U[dp]] -= delta * delta
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_box_sum(h, S)
        c = np.correlate(h, 1.0 - h, mode="full")
        val = float(c.max()) * (2.0 / float(n))
        if val < best_c:
            best_c = val
            best_h = h.copy()
        elif val > best_c + 1e-7:
            h = best_h.copy()
            c = np.correlate(h, 1.0 - h, mode="full")

    best_h = project_box_sum(best_h, S)
    return best_h, raw_value(best_h, n)


def projected_gradient(h, n, pre, alpha, max_iter, deadline):
    S = 0.5 * float(n)
    h = project_box_sum(h, S)
    best_h = h.copy()
    best_raw = raw_value(h, n)
    step = 30.0
    max_iter = int(max(0, max_iter))
    rejects = 0

    for _ in range(max_iter):
        if time.time() > deadline:
            break
        F, g, raw = eval_smooth_grad(h, n, alpha, pre)
        if float(np.linalg.norm(g)) < 1e-12:
            break
        h_new = project_box_sum(h - step * g, S)
        F_new, raw_new = smooth_value(h_new, n, alpha)
        if raw_new < best_raw - 1e-11 or F_new < F - 1e-10:
            h = h_new
            if raw_new < best_raw:
                best_raw = raw_new
                best_h = h.copy()
            step = min(step * 1.05, 90.0)
            rejects = 0
        else:
            rejects += 1
            step *= 0.5
            if step < 1e-5 or rejects > 8:
                break

    best_h = project_box_sum(best_h, S)
    return best_h, raw_value(best_h, n)


def _lb_obj(x):
    F, _ = smooth_value(x, _G_N, _G_ALPHA)
    s = float(x.sum()) - 0.5 * float(_G_N)
    return F + 0.5 * _G_MU * s * s


def _lb_grad(x):
    _, g, _ = eval_smooth_grad(x, _G_N, _G_ALPHA, _G_PRE)
    s = float(x.sum()) - 0.5 * float(_G_N)
    return g + _G_MU * s * np.ones_like(x)


def lbfgs_opt(h, n, pre, alpha, mu, maxiter):
    global _G_N, _G_PRE, _G_ALPHA, _G_MU
    h = np.asarray(h, dtype=float).copy()
    _G_N = int(n)
    _G_PRE = pre
    _G_ALPHA = float(alpha)
    _G_MU = float(mu)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_lb_obj, h, jac=_lb_grad, method="L-BFGS-B",
                       bounds=bounds,
                       options={"maxiter": int(maxiter), "ftol": 1e-12, "maxls": 20})
        h2 = project_box_sum(res.x, 0.5 * float(n))
        return h2, raw_value(h2, n)
    except Exception:
        return h, raw_value(h, n)


def parent_to_n(parent, n):
    if parent is None or len(parent) == 0:
        y = np.full(n, 0.5, dtype=float)
    else:
        xp = np.linspace(0.0, 2.0, len(parent))
        xn = np.linspace(0.0, 2.0, n)
        y = np.interp(xn, xp, parent)
    return project_box_sum(y, 0.5 * float(n))


def binary_threshold(y, n):
    S = 0.5 * float(n)
    b = (y > 0.5).astype(float)
    s = float(b.sum())
    if s < S:
        zero = np.where(b == 0)[0]
        k = int(S - s + 0.5)
        if k > 0 and zero.size:
            idx = zero[np.argsort(-y[zero])[:k]]
            b[idx] = 1.0
    elif s > S:
        one = np.where(b == 1)[0]
        k = int(s - S + 0.5)
        if k > 0 and one.size:
            idx = one[np.argsort(y[one])[:k]]
            b[idx] = 0.0
    return project_box_sum(b, S)


def block_start(n, rng):
    nb = max(1, n // 12)
    v = rng.random(nb)
    h = np.repeat(v, n // nb + 1)
    h = h[:n]
    return project_box_sum(h, 0.5 * float(n))


def make_candidates(yref, n, best_h, rng):
    S = 0.5 * float(n)
    x = np.linspace(0.0, 2.0, n)
    cands = [yref]

    if best_h is not None:
        if len(best_h) == n:
            h = np.clip(best_h, 0.0, 1.0)
        else:
            xp = np.linspace(0.0, 2.0, len(best_h))
            h = np.interp(x, xp, best_h)
        cands.append(project_box_sum(h, S))

    cands.append(binary_threshold(yref, n))

    for s in (0.05, 0.18, 0.35):
        cands.append(project_box_sum(yref + rng.normal(0.0, s, n), S))

    for f in (1, 2, 3):
        phase = float(rng.random()) * 6.283185307179586
        cands.append(project_box_sum(0.5 + 0.5 * np.sin(2.0 * np.pi * f * x + phase), S))

    cands.append(block_start(n, rng))
    return cands[:10]


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 8.0
    rng = np.random.RandomState(seed)

    try:
        parent = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        parent = None

    if parent is not None and len(parent) > 0:
        best_n = len(parent)
        best_h = parent_to_n(parent, best_n)
    else:
        best_n = 512
        best_h = np.full(best_n, 0.5, dtype=float)
        best_h = project_box_sum(best_h, 0.5 * float(best_n))
    best_c = raw_value(best_h, best_n)

    ns = []
    if parent is not None and len(parent) > 0:
        ns.append(len(parent))
    for mm in (512, 384, 320):
        if mm not in ns:
            ns.append(mm)

    for n in ns:
        if time.time() > deadline - 180:
            break
        pre = precompute(n)
        yref = parent_to_n(parent, n) if parent is not None else parent_to_n(None, n)
        cands = make_candidates(yref, n, best_h, rng)

        for cand in cands:
            if time.time() > deadline - 120:
                break
            hc, cc = cd_refine(cand, n, pre, 2, deadline - 120, rng)
            if cc + 0.003 < best_c:
                hc2, cc2 = cd_refine(hc, n, pre, 10, deadline - 90, rng)
                if cc2 < best_c * 0.999:
                    hc3, cc3 = projected_gradient(hc2, n, pre, 2000, 50, deadline - 80)
                    hc4, cc4 = cd_refine(hc3, n, pre, 8, deadline - 70, rng, random=True)
                    if cc4 < best_c:
                        best_h, best_c, best_n = hc4, cc4, n

        del pre

    pre = precompute(best_n)

    if time.time() < deadline - 120:
        h1, c1 = cd_refine(best_h, best_n, pre, 12, deadline - 120, rng,
                           skip_bounds=False)
        if c1 < best_c:
            best_h, best_c = h1, c1

    if time.time() < deadline - 100:
        h2, c2 = projected_gradient(best_h, best_n, pre, 1500, 60, deadline - 100)
        if c2 < best_c:
            best_h, best_c = h2, c2

    if time.time() < deadline - 90:
        h3, c3 = cd_refine(best_h, best_n, pre, 18, deadline - 90, rng)
        h3b, c3b = cd_refine(h3, best_n, pre, 12, deadline - 80, rng, random=True)
        if c3b < best_c:
            best_h, best_c = h3b, c3b

    if time.time() < deadline - 70:
        h4, c4 = lbfgs_opt(best_h, best_n, pre, 2000, 1.0e5, 60)
        if c4 < best_c:
            best_h, best_c = h4, c4

    if time.time() < deadline - 60:
        h5, c5 = cd_refine(best_h, best_n, pre, 25, deadline - 60, rng)
        if c5 < best_c:
            best_h, best_c = h5, c5

    if time.time() < deadline - 30:
        h6, c6 = lbfgs_opt(best_h, best_n, pre, 8000, 1.0e6, 40)
        h6, c6 = cd_refine(h6, best_n, pre, 10, deadline - 30, rng, random=True)
        if c6 < best_c:
            best_h, best_c = h6, c6

    best_h = project_box_sum(best_h, 0.5 * float(best_n))
    best_c = raw_value(best_h, best_n)
    return best_h, float(best_c), int(best_n)
