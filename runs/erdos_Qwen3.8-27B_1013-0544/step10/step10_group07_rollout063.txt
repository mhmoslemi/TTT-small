import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def lse_value_and_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    out = shifted.max()
    exp_s = np.exp(shifted - out)
    w = exp_s / exp_s.sum()
    val = (m + (out + np.log(exp_s.sum())) / alpha) * (2.0 / n)
    # Gradient
    one_minus_h = 1.0 - h
    A = np.correlate(one_minus_h, w, mode='full')[2*n-2 : 3*n-2][::-1]
    B = np.convolve(h, w, mode='full')[n-1 : 2*n-1]
    grad = (2.0 / n) * (A - B)
    return val, grad

def pgd_optimize(h, n, alphas, iters_per_alpha, deadline):
    h = h.copy()
    best_h = h.copy()
    best_val = lse_value_and_grad(h, n, alphas[-1])[0]
    best_c = compute_c5(h, n)
    
    for alpha in alphas:
        if time.time() > deadline:
            break
        f, g = lse_value_and_grad(h, n, alpha)
        lr = 1.0
        for it in range(iters_per_alpha):
            if time.time() > deadline:
                break
            # Backtracking line search
            for _ in range(50):
                h_new = project_feasible(h - lr * g, n)
                f_new, _ = lse_value_and_grad(h_new, n, alpha)
                if f_new <= f - 1e-4 * lr * np.sum(g * (h_new - h)):
                    break
                lr *= 0.5
            else:
                break
            h = h_new
            f = f_new
            g = lse_value_and_grad(h, n, alpha)[1]
            # Update best
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
    
    return best_h, best_c

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def lse_val_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    e = np.exp(sh - mx)
    w = e / e.sum()
    val = (m + (mx + np.log(e.sum())) / alpha) * (2.0 / n)
    omh = 1.0 - h
    A = np.correlate(omh, w, mode='full')[2*n-2 : 3*n-2][::-1]
    B = np.convolve(h, w, mode='full')[n-1 : 2*n-1]
    grad = (2.0 / n) * (A - B)
    return val, grad

def pgd_momentum(h, n, alpha, max_iter, lr, momentum, deadline):
    h = h.copy()
    h_prev = h.copy()
    v = np.zeros(n)
    f, g = lse_val_grad(h, n, alpha)
    best_h = h.copy()
    best_f = f
    for it in range(max_iter):
        if time.time() > deadline:
            break
        v = momentum * v + g
        h_new = project_feasible(h - lr * v, n)
        f_new, g = lse_val_grad(h_new, n, alpha)
        if f_new < f:
            h, f, h_prev = h_new, f_new, h
            if f < best_f:
                best_f = f
                best_h = h.copy()
        else:
            # Undo step, reduce effective step
            h = h
            # Just use h_prev as is
            h_prev = h
            g, _ = lse_val_grad(h, n, alpha) if False else (g, None)
        h_prev = h if f_new < f else h_prev
    return best_h, best_f

    # Gradient: grad_j = sum_k lam[k] * dc_k/dh_j
