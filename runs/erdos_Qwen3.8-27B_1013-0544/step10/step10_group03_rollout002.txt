def subgrad_step(h,n, eps_frac=0.05, step=0.02):
 c=correlate_full(h); cmax=c.max()
 if cmax<=0: return h
 mask=c >= cmax - eps_frac*cmax
 grad=np.zeros(n)
 for m in np.nonzero(mask)[0]:
   t=m-(n-1)
   if t==0:
      grad += 1-2*h
   elif t>0:
      # i=0..n-t-1
      grad[:n-t] += 1 - h[t:n]
      grad[t:] += -h[:n-t]
   else:
      d=-t
      # i=d..n-1? c[ -d]=∑_{i=d}^{n-1} h_i(1-h_{i-d})
      # grad[i] += 1 - h_{i-d}; grad[i-d] += -h_i
      grad[d:] += 1 - h[:n-d]
      grad[:n-d] += -h[d:]
 gmax=np.max(np.abs(grad))
 if gmax==0: return h
 h2=h - (step/gmax)*grad
 return project_feasible(h2,n)

def make_subspace_basis(n,d,seed):
    rng=np.random.RandomState(seed)
    x=np.linspace(0.0,2.0,n,endpoint=False)
    U=np.empty((n,d))
    for j in range(d):
       f=rng.uniform(0.75,6.0)
       ph=rng.uniform(0.0,2.0*np.pi)
       U[:,j]=np.cos(np.pi*f*x + ph)
    U -= U.mean(axis=0)
    Q,R=np.linalg.qr(U)
    Q=Q[:,:d]
    for j in range(d):
       if R[j,j] < 0: Q[:,j]*=-1.0
       norm=np.linalg.norm(Q[:,j])
       if norm>1e-12: Q[:,j]/=norm
    return Q

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / float(n)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def correlate_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = 0.5 * n
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    diff = target - h.sum()
    for _ in range(8):
        if abs(diff) < 1e-13:
            break
        m = (h > 1e-13) & (h < 1.0 - 1e-13)
        if not m.any():
            break
        h[m] += diff / int(m.sum())
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
    if abs(diff) > 1e-9:
        lo, hi = -10.0, 10.0
        for _ in range(4):
            s0 = np.clip(h + lo, 0.0, 1.0).sum()
            s1 = np.clip(h + hi, 0.0, 1.0).sum()
            if s0 <= target <= s1:
                break
            lo *= 2.0
            hi *= 2.0
        for _ in range(45):
            mid = 0.5 * (lo + hi)
            if np.clip(h + mid, 0.0, 1.0).sum() > target:
                hi = mid
            else:
                lo = mid
        h = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
        diff = target - h.sum()
        m = (h > 1e-13) & (h < 1.0 - 1e-13)
        if m.any():
            h[m] += diff / int(m.sum())
        h = np.clip(h, 0.0, 1.0)
    return h

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / float(n)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def correlate_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = 0.5 * n
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    diff = target - h.sum()
    for _ in range(8):
        if abs(diff) < 1e-13:
            break
        m = (h > 1e-13) & (h < 1.0 - 1e-13)
        if not m.any():
            break
        h[m] += diff / int(m.sum())
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
    if abs(diff) > 1e-9:
        lo, hi = -10.0, 10.0
        for _ in range(4):
            s0 = np.clip(h + lo, 0.0, 1.0).sum()
            s1 = np.clip(h + hi, 0.0, 1.0).sum()
            if s0 <= target <= s1:
                break
            lo *= 2.0
            hi *= 2.0
        for _ in range(45):
            mid = 0.5 * (lo + hi)
            if np.clip(h + mid, 0.0, 1.0).sum() > target:
                hi = mid
            else:
                lo = mid
        h = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
        diff = target - h.sum()
        m = (h > 1e-13) & (h < 1.0 - 1e-13)
        if m.any():
            h[m] += diff / int(m.sum())
        h = np.clip(h, 0.0, 1.0)
    return h

def default_h(n):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    return project_feasible(1.0 - np.abs(x - 1.0), n)

def interp_h(h, n_old, n_new):
    if n_old == n_new:
        return h.copy()
    x0 = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x1 = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return project_feasible(np.interp(x1, x0, h), n_new)

def build_flip_indices(n):
    out = []
    for p in range(n):
        klo = max(0, 2 * p - n + 1)
        khi = min(n - 1, 2 * p)
        if klo <= khi:
            idx = 2 * p - np.arange(klo, khi + 1, dtype=np.int64)
        else:
            idx = np.empty(0, dtype=np.int64)
        out.append((klo, khi, idx))
    return out

def feasible_data(L, Apos, Bpos, Aneg, Bneg, maxzero, C0):
    tol = 1e-12
    if maxzero > L + tol:
        return False
    lo, hi = 0.0, 1.0
    if Apos.size:
        hi = min(hi, float(np.min((L - Bpos) / Apos)))
    if Aneg.size:
        lo = max(lo, float(np.max((L - Bneg) / Aneg)))
    lo = max(0.0, lo)
    hi = min(1.0, hi)
    if lo > hi + 1e-11:
        return False
    if L < C0 - 1e-11:
        return False
    if L >= C0 + 0.25 - 1e-12:
        return True
    delta = L - C0
    if delta < 0.0:
        delta = 0.0
    if delta > 0.25:
        delta = 0.25
    disc = 1.0 - 4.0 * delta
    if disc < 0.0:
        disc = 0.0
    r1 = 0.5 * (1.0 - np.sqrt(disc))
    r2 = 0.5 * (1.0 + np.sqrt(disc))
    return (lo <= r1 + 1e-11) or (hi >= r2 - 1e-11)

def choose_x_data(L, Apos, Bpos, Aneg, Bneg, C0, v):
    tol = 1e-11
    lo, hi = 0.0, 1.0
    if Apos.size:
        hi = min(hi, float(np.min((L - Bpos) / Apos)))
    if Aneg.size:
        lo = max(lo, float(np.max((L - Bneg) / Aneg)))
    lo = max(0.0, lo)
    hi = min(1.0, hi)
    if lo > hi:
        return float(np.clip(v, 0.0, 1.0))
    intervals = []
    if L >= C0 + 0.25 - tol:
        intervals.append((lo, hi))
    else:
        if L < C0 - tol:
            return float(np.clip(v, 0.0, 1.0))
        delta = L - C0
        if delta < 0.0:
            delta = 0.0
        if delta > 0.25:
            delta = 0.25
        disc = 1.0 - 4.0 * delta
        if disc < 0.0:
            disc = 0.0
        r1 = 0.5 * (1.0 - np.sqrt(disc))
        r2 = 0.5 * (1.0 + np.sqrt(disc))
        if lo <= r1 + tol:
            b = hi if hi < r1 else r1
            if lo <= b + tol:
                intervals.append((lo, b))
        if hi >= r2 - tol:
            a = lo if lo > r2 else r2
            if a <= hi + tol:
                intervals.append((a, hi))
    if not intervals:
        return float(np.clip(0.5 * (lo + hi), 0.0, 1.0))
    best_mid = 0.5 * (lo + hi)
    best_d = 1e100
    for a, b in intervals:
        mid = 0.5 * (a + b)
        d = abs(mid - v)
        if d < best_d:
            best_d = d
            best_mid = mid
    return float(np.clip(best_mid, 0.0, 1.0))

def exact_cd(h, n, max_passes, deadline):
    h = project_feasible(h, n)
    c = correlate_full(h)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    idxs = build_flip_indices(n)
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        full_max = float(np.max(c))
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            v = h[p]
            m0 = n - 1 - p
            csl = c[m0:m0 + n]
            unaff = 0.0
            if m0 > 0:
                unaff = max(unaff, float(np.max(c[:m0])))
            end = m0 + n
            if end < c.size:
                unaff = max(unaff, float(np.max(c[end:])))
            A = 1.0 - h
            klo, khi, idx = idxs[p]
            if klo <= khi:
                A[klo:khi + 1] -= h[idx]
            A[p] = 0.0
            B = csl - A * v
            B[p] = -1e100
            C0 = csl[p] - v * (1.0 - v)
            pos = A > 1e-14
            neg = A < -1e-14
            zero = ~(pos | neg)
            Apos = A[pos]
            Bpos = B[pos]
            Aneg = A[neg]
            Bneg = B[neg]
            maxzero = float(np.max(B[zero])) if zero.any() else -1e100
            low = max(0.0, float(C0), float(unaff))
            high = full_max if full_max > low else low
            for __ in range(40):
                mid = 0.5 * (low + high)
                if feasible_data(mid, Apos, Bpos, Aneg, Bneg, maxzero, C0):
                    high = mid
                else:
                    low = mid
            x = choose_x_data(high, Apos, Bpos, Aneg, Bneg, C0, v)
            if abs(x - v) > 1e-12:
                new_csl = csl + A * (x - v)
                new_csl[p] = C0 + x * (1.0 - x)
                nm = float(np.max(new_csl))
                if unaff > nm:
                    nm = unaff
                if nm < full_max - 1e-12:
                    c[m0:m0 + n] = new_csl
                    h[p] = x
                    full_max = nm
                    improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        c = correlate_full(h)
        cv = compute_c5(h, n)
        if cv < best_c:
            best_c = cv
            best_h = h.copy()
        if cv > best_c + 1e-10:
            h = best_h.copy()
            c = correlate_full(h)
    return best_h, best_c

def binary_from_h(h, n):
    hb = (np.asarray(h, dtype=float) > 0.5).astype(float)
    target = n // 2
    diff = int(round(hb.sum() - target))
    if diff > 0:
        idx = np.flatnonzero(hb)
        idx = idx[np.argsort(np.asarray(h)[idx])]
        for i in range(diff):
            hb[idx[i]] = 0.0
    elif diff < 0:
        idx = np.flatnonzero(1.0 - hb)
        idx = idx[np.argsort(-np.asarray(h)[idx])]
        for i in range(-diff):
            hb[idx[i]] = 1.0
    return hb

def flip_update(c, h, p, n, new_val, idxs):
    v = float(h[p])
    new_val = float(new_val)
    delta = new_val - v
    if abs(delta) < 1e-14:
        return
    m0 = n - 1 - p
    A = 1.0 - h
    klo, khi, idx = idxs[p]
    if klo <= khi:
        A[klo:khi + 1] -= h[idx]
    A[p] = 0.0
    c[m0:m0 + n] += delta * A
    h[p] = new_val

def binary_local(h, n, max_iters, deadline, seed):
    hb = binary_from_h(h, n)
    c = correlate_full(hb)
    best_max = int(np.rint(float(np.max(c))))
    best_active = int(np.count_nonzero(c >= best_max - 0.5))
    best_h = hb.copy()
    best_c = 2.0 * best_max / float(n)
    idxs = build_flip_indices(n)
    ones_list = np.flatnonzero(hb).tolist()
    zeros_list = np.flatnonzero(1.0 - hb).tolist()
    rng = np.random.RandomState(seed)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if not ones_list or not zeros_list:
            break
        p = ones_list[int(rng.randint(len(ones_list)))]
        q = zeros_list[int(rng.randint(len(zeros_list)))]
        if p == q:
            continue
        flip_update(c, hb, p, n, 0.0, idxs)
        flip_update(c, hb, q, n, 1.0, idxs)
        nm = int(np.rint(float(np.max(c))))
        na = int(np.count_nonzero(c >= nm - 0.5))
        accept = False
        if nm < best_max:
            accept = True
        elif nm == best_max:
            if na < best_active:
                accept = True
            elif na == best_active and rng.random() < 0.04:
                accept = True
        if accept:
            ones_list = np.flatnonzero(hb).tolist()
            zeros_list = np.flatnonzero(1.0 - hb).tolist()
            if nm < best_max:
                best_max = nm
                best_active = na
                best_c = 2.0 * nm / float(n)
                best_h = hb.copy()
            elif nm == best_max and na < best_active:
                best_active = na
                best_h = hb.copy()
        else:
            flip_update(c, hb, q, n, 0.0, idxs)
            flip_update(c, hb, p, n, 1.0, idxs)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def cosine_perturb(h, n, trials, scales, rng, deadline, passes):
    best = h.copy()
    best_c = compute_c5(best, n)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    d = min(14, max(4, n // 4))
    for t in range(trials):
        if time.time() > deadline:
            break
        U = np.empty((n, d))
        for j in range(d):
            f = rng.uniform(0.75, 8.0)
            ph = rng.uniform(0.0, 2.0 * np.pi)
            U[:, j] = np.cos(np.pi * f * x + ph)
        U -= U.mean(axis=0)
        for j in range(d):
            nn = np.linalg.norm(U[:, j])
            if nn > 1e-12:
                U[:, j] /= nn
        scale = float(scales[t % len(scales)])
        z = rng.normal(0.0, 1.0, d) * scale
        h0 = project_feasible(h + U * z, n)
        h1, c1 = exact_cd(h0, n, passes, deadline)
        if c1 < best_c:
            best = h1
            best_c = c1
    return best, best_c
