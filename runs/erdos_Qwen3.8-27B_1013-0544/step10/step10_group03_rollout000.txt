def project_feasible(h, n=None):

def coordinate_update(h, c, p, n, lam, sum_h, target):

    Dp = p
    Dn = n - 1 - p
    D = Dp if Dp > Dn else Dn

            U = v
    D0 = c[n - 1] - h0 * (1.0 - h0)
    if D0 < 0.0 and D0 > -1e-10:
        D0 = 0.0
    cur_max = float(cseg.max())
    if U > cur_max:
        cur_max = U
    B = A + lam
    B[D] = 0.0
    eps = 1e-14
    pos = B > eps
    pos[D] = False
    neg = B < -eps
    neg[D] = False
    c_pos = cseg[pos]
    B_pos = B[pos]
    c_neg = cseg[neg]
    B_neg = B[neg]
    zero_non = ~(pos | neg)
    zero_non[D] = False
    if zero_non.any():
        zero_cmax = float(cseg[zero_non].max())
    else:
        zero_cmax = -1e100
    non_all = np.ones(M, dtype=bool)
    non_all[D] = False
    c_non = cseg[non_all]
    A_non = A[non_all]
    dmin = h0 - 1.0
    dmax = 1.0 - h0
    lb = -1e100
    if c_pos.size:
        lb = max(lb, float(np.max(c_pos + B_pos * dmin)))
    if c_neg.size:
        lb = max(lb, float(np.max(c_neg + B_neg * dmax)))
    if zero_non.any():
        lb = max(lb, float(cseg[zero_non].max()))
    if lam > eps:
        lb = max(lb, U + lam * dmin)
    elif lam < -eps:
        lb = max(lb, U + lam * dmax)
    else:
        lb = max(lb, U)
    lb = max(lb, min(D0 + lam * dmin, D0 + lam * dmax))
    if lb < -1e4:
        lb = -1e4
    hi = cur_max
    if hi < lb:
        hi = lb
    ok = phi_state(hi, h0, lam, D0, U, c_pos, B_pos, c_neg, B_neg, zero_cmax)[0]
    for _ in range(20):
        if ok:
            break
        hi += max(1.0, abs(hi) * 0.25)
        ok = phi_state(hi, h0, lam, D0, U, c_pos, B_pos, c_neg, B_neg, zero_cmax)[0]
    if not ok:
        return 0.0, cur_max, sum_h
    lo = min(lb, hi)
    oklo = phi_state(lo, h0, lam, D0, U, c_pos, B_pos, c_neg, B_neg, zero_cmax)[0]
    if not oklo:
        for _ in range(36):
            mid = 0.5 * (lo + hi)
            if phi_state(mid, h0, lam, D0, U, c_pos, B_pos, c_neg, B_neg, zero_cmax)[0]:
                hi = mid
            else:
                lo = mid
    L = hi + 1e-11
    ok, xl, xh, has, rr1, rr2 = phi_state(L, h0, lam, D0, U, c_pos, B_pos, c_neg, B_neg, zero_cmax)
    if not ok:
        return 0.0, cur_max, sum_h
    step = (target - sum_h) / float(n)
    if step > 0.1:
        step = 0.1
    elif step < -0.1:
        step = -0.1
    x_Des = h0 + step
    x = choose_candidate(xl, xh, has, rr1, rr2, c_non, A_non, D0, U, h0, x_Des)
    x = float(np.clip(x, 0.0, 1.0))
    delta = x - h0
    if abs(delta) < 1e-13:
        return 0.0, cur_max, sum_h
    c[r0:r1 + 1] += A * delta
    c[n - 1] = D0 + x * (1.0 - x)
    h[p] = x
    sum_h += delta
    new_seg = float(c[r0:r1 + 1].max())
    if U > new_seg:
        new_seg = U
    return delta, new_seg, sum_h


def cd_fixed_lam(h, c, n, lam, passes, deadline, rng, target):
    if c is None:
        c = np.correlate(h, 1.0 - h, mode="full")
    sum_h = float(h.sum())
    for _ in range(int(max(1, passes))):
        if time.time() >= deadline:
            break
        order = rng.permutation(n)
        maxd = 0.0
        for p in order:
            if time.time() >= deadline:
                break
            delta, _new_max, sum_h = coordinate_update(h, c, p, n, lam, sum_h, target)
            if abs(delta) > maxd:
                maxd = abs(delta)
        c = np.correlate(h, 1.0 - h, mode="full")
        if maxd < 1e-11:
            break
    sum_h = float(h.sum())
    return h, c, sum_h


def make_starts(base, n, rng):
    out = []
    base = project_feasible(base, n)
    s1 = np.zeros(n)
    cnt = n // 2
    if cnt:
        idx = rng.choice(n, cnt, replace=False)
        s1[idx] = 1.0
    out.append(project_feasible(s1, n))
    out.append(project_feasible(base + 0.08 * rng.normal(size=n), n))
    x = np.linspace(0.0, 2.0, n)
    s3 = 0.5 + 0.28 * np.sin(np.pi * x) + 0.17 * np.sin(2.0 * np.pi * x)
    out.append(project_feasible(s3, n))
    s4 = (base > 0.5).astype(float)
    out.append(project_feasible(s4, n))
    return out


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 5.0
    if deadline < t0 + 5.0:
        deadline = t0 + 10.0
    rng = np.random.RandomState(int(seed) % (2**31 - 1))
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        n = 700
        h_best = np.full(n, 0.5)
    if n <= 2:
        n = 700
        h_best = np.full(n, 0.5)
    target = 0.5 * float(n)
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best)
    base_h = h_best.copy()
    reserve = max(120.0, 0.34 * (deadline - t0))
    coarse = []
    lam_cand = (-4.0, -1.5, -0.4, 0.0, 0.4, 1.5, 4.0)
    for lam in lam_cand:
        if time.time() > deadline - reserve:
            break
        ht = base_h.copy()
        ct = np.correlate(ht, 1.0 - ht, mode="full")
        ht, ct, st = cd_fixed_lam(ht, ct, n, lam, 2, deadline - reserve, rng, target)
        hp = project_feasible(ht, n)
        cp = compute_c5(hp)
        meanwork = st / float(n)
        coarse.append((cp, lam, meanwork))
        if cp < best_c:
            best_c = cp
            h_best = hp
    lam_best = -0.5
    if coarse:
        best_score = 1e100
        for cp, lam, mw in coarse:
            score = cp + 0.12 * abs(mw - 0.5)
            if score < best_score:
                best_score = score
                lam_best = lam
    h_work = h_best.copy()
    c_work = np.correlate(h_work, 1.0 - h_work, mode="full")
    sum_work = float(h_work.sum())
    lam_use = float(lam_best)
    no_improve = 0
    while time.time() < deadline - 120.0:
        h_work, c_work, sum_work = cd_fixed_lam(h_work, c_work, n, lam_use, 3, deadline - 120.0, rng, target)
        hp = project_feasible(h_work, n)
        cp = compute_c5(hp)
        if cp < best_c - 1e-12:
            best_c = cp
            h_best = hp
            no_improve = 0
        else:
            no_improve += 1
        err = sum_work - target
        if abs(err) > 0.12 * n:
            h_work = hp.copy()
            c_work = np.correlate(h_work, 1.0 - h_work, mode="full")
            sum_work = float(h_work.sum())
            if abs(lam_use) > 1e-3:
                lam_use *= 0.5
            else:
                lam_use = -0.2
        else:
            lam_use += 12.0 * err / float(n)
            if lam_use > 8.0:
                lam_use = 8.0
            if lam_use < -8.0:
                lam_use = -8.0
        if no_improve > 8:
            break
    if time.time() < deadline - 80.0 and abs(lam_use) > 1e-4:
        for mult in (0.5, 1.5, -0.5):
            if time.time() >= deadline - 60.0:
                break
            ht = h_best.copy()
            ct = np.correlate(ht, 1.0 - ht, mode="full")
            ht, ct, st = cd_fixed_lam(ht, ct, n, lam_use * mult, 3, deadline - 60.0, rng, target)
            hp = project_feasible(ht, n)
            cp = compute_c5(hp)
            if cp < best_c:
                best_c = cp
                h_best = hp
    if time.time() < deadline - 120.0:
        starts = make_starts(h_best, n, rng)
        for st in starts[:3]:
            if time.time() >= deadline - 60.0:
                break
            for lam_try in (lam_use, 0.0):
                if time.time() >= deadline - 45.0:
                    break
                ht = st.copy()
                ct = np.correlate(ht, 1.0 - ht, mode="full")
                ht, ct, _st = cd_fixed_lam(ht, ct, n, lam_try, 3, deadline - 45.0, rng, target)
                hp = project_feasible(ht, n)
                cp = compute_c5(hp)
                if cp < best_c:
                    best_c = cp
                    h_best = hp
    if time.time() < deadline - 8.0:
        for lam_try in (0.0, lam_use):
            if time.time() >= deadline - 1.0:
                break
            ht = h_best.copy()
            ct = np.correlate(ht, 1.0 - ht, mode="full")
            ht, ct, _st = cd_fixed_lam(ht, ct, n, lam_try, 2, deadline - 0.2, rng, target)
            hp = project_feasible(ht, n)
            cp = compute_c5(hp)
            if cp < best_c:
                best_c = cp
                h_best = hp
    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best)
    return h_best, best_c, len(h_best)
