    best_c = compute_c5(h, n)

        return x.sum() - n / 2.0
    
    def _sum_eq_jac(x, n):
        return np.ones(n)
    
    def smooth_obj(x, n, alpha):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        lse = mx + np.log(np.sum(np.exp(shifted - mx)))
        return (m + lse / alpha) * (2.0 / n)
    
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac, 'args': (n,)}]
    for alpha in alphas:

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)

        if abs(diff) < 1e-13:

            freq = 1.0 + 3.0 * rng.rand()
            harm = rng.randint(1, 5)
            hp = project_feasible(best_h + 0.015 * np.sin(harm * np.pi * freq * x), n)
        elif r < 45:
            bl = rng.randint(5, 40)
            st = rng.randint(0, n - bl)
            pert = np.zeros(n)
            pert[st:st+bl] = 0.25 * rng.choice([-1, 1], size=bl)
            hp = project_feasible(best_h + pert, n)
        elif r < 60:
            nf = rng.randint(2, 12)
            fl = rng.choice(n, size=nf, replace=False)
            pert = np.zeros(n)
            pert[fl] = 0.5 * rng.choice([-1, 1], size=nf)
            hp = project_feasible(best_h + pert, n)
        elif r < 75:
            # Swap-pair perturbation
            ones = np.where(best_h >= 0.9)[0]
            zeros = np.where(best_h <= 0.1)[0]
            hp = best_h.copy()
            if len(ones) > 0 and len(zeros) > 0:
                ns = rng.randint(1, 6)
                for _ in range(ns):
                    p = ones[rng.randint(len(ones))]
                    q = zeros[rng.randint(len(zeros))]
                    if p != q:
                        hp[p], hp[q] = hp[q], hp[p]
            hp = project_feasible(hp, n)
        else:
            s = 1.0 + 0.04 * rng.randn()
            hp = project_feasible(s * best_h, n)
        h_o, c_o = coord_descent(hp, n, 25, deadline, gs_iters=50, threshold=0.03)
        if c_o < best_c:
            h_o, c_o = subspace_slsqp(h_o, n, thresh=0.05, alphas=[15000], maxiter=200)
            h_o, c_o = coord_descent(h_o, n, 20, deadline, gs_iters=55, threshold=0.01)
            h_o, c_o = swap_search(h_o, n, rng, max_passes=8, deadline=deadline, n_tries=600)
            if c_o < best_c:
                best_h, best_c = h_o, c_o

    # Phase 5: Final polish
    if time.time() < deadline - 40:
        hf, cf = coord_descent(best_h, n, 30, deadline, gs_iters=65, threshold=0.005)
        if cf < best_c:
            best_h, best_c = hf, cf
    if time.time() < deadline - 15:
        hf, cf = subspace_slsqp(best_h, n, thresh=0.03, alphas=[100000], maxiter=400)
        if cf < best_c:
            best_h, best_c = hf, cf
    if time.time() < deadline - 8:
        hf, cf = slsqp_smooth(best_h, n, [200000, 800000], maxiter=300)
        if cf < best_c:
            best_h, best_c = hf, cf
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
