                    r2 = (1.0 + np.sqrt(disc)) * 0.5
                    if x_hi > r2 and x_lo < r1:
                        ok = False
                    elif x_hi > r2:
                        x_lo = max(x_lo, r2)
                    elif x_lo < r1:
                        x_hi = min(x_hi, r1)
            if ok:
                L_hi = L
            else:
                L_lo = L
        L = L_hi
        x_lo = 0.0
        x_hi = 1.0
        if np.any(neg):
            x_lo = max(x_lo, float(np.max((L - c[neg]) / a[neg])))
        if np.any(pos):
            x_hi = min(x_hi, float(np.min((L - c[pos]) / a[pos])))
        x_lo = max(x_lo, 0.0)
        x_hi = min(x_hi, 1.0)
        Cq = L - c[diag] + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) * 0.5
            r2 = (1.0 + np.sqrt(disc)) * 0.5
            if x_hi > r2 and x_lo < r1:
                if abs(h0 - r1) <= abs(h0 - r2):
                    x_hi = min(x_hi, r1)
                else:
                    x_lo = max(x_lo, r2)
            elif x_hi > r2:
                x_lo = max(x_lo, r2)
            elif x_lo < r1:
                x_hi = min(x_hi, r1)
        if x_lo > x_hi + 1e-14:
            continue
        x_new = float(np.clip((x_lo + x_hi) * 0.5, 0.0, 1.0))
        if abs(x_new - h0) > 1e-13:
            delta = x_new - h0
            c += delta * a
            c[diag] -= delta * delta
            h[p] = x_new
    h = project_feasible(h, n)
    return h

def pair_cd_pass(h, n):
    dx = 2.0 / n
    for p in range(0, n - 1, 2):
        s = n / 2.0 - h[p] - h[p + 1]
        a_lo = max(0.0, s - 1.0)
        a_hi = min(1.0, s)
        if a_hi - a_lo < 1e-12:
            continue
        h_save = h.copy()
        h_save[p] = 0.0
        h_save[p + 1] = 0.0
        c_base = np.correlate(h_save, 1.0 - h_save, mode='full')
        k_all = np.arange(2 * n - 1) - (n - 1)
        pk = p - k_all
        pk2 = p + 1 + k_all
        pkv = (pk >= 0) & (pk < n)
        pk2v = (pk2 >= 0) & (pk2 < n)
        hp = np.zeros(2 * n - 1)
        hp2 = np.zeros(2 * n - 1)
        if np.any(pkv):
            hp[pkv] = h_save[pk[pkv]]
        if np.any(pk2v):
            hp2[pk2v] = h_save[pk2[pk2v]]
        q1 = p + 1 - k_all
        q2 = p + 1 + k_all
        q1v = (q1 >= 0) & (q1 < n)
        q2v = (q2 >= 0) & (q2 < n)
        hq1 = np.zeros(2 * n - 1)
        hq2 = np.zeros(2 * n - 1)
        if np.any(q1v):
            hq1[q1v] = h_save[q1[q1v]]
        if np.any(q2v):
            hq2[q2v] = h_save[q2[q2v]]
        A = 1.0 - hp2 - hp
        B = 1.0 - hq2 - hq1
        Q0 = c_base + hp2 + hq2 + s * B
        Q1 = A - B
        Q2 = np.zeros(2 * n - 1)
        Q2[n] = 1.0
        Q2[n - 2] = 1.0
        Q1[n] -= (s + 1.0)
        Q1[n - 2] += (1.0 - s)
        phi = (1.0 + np.sqrt(5.0)) / 2.0
        a1 = a_hi - (a_hi - a_lo) / phi
        a2 = a_lo + (a_hi - a_lo) / phi
        f1 = np.max(Q0 + Q1 * a1 + Q2 * a1 ** 2)
        f2 = np.max(Q0 + Q1 * a2 + Q2 * a2 ** 2)
        for _ in range(45):
            if a_hi - a_lo < 1e-13:
                break
            if f1 < f2:
                a_hi, a2 = a2, a1
                f2 = f1
                a1 = a_hi - (a_hi - a_lo) / phi
                f1 = np.max(Q0 + Q1 * a1 + Q2 * a1 ** 2)
            else:
                a_lo, a1 = a1, a2
                f1 = f2
                a2 = a_lo + (a_hi - a_lo) / phi
                f2 = np.max(Q0 + Q1 * a2 + Q2 * a2 ** 2)
        a_star = float(np.clip((a_lo + a_hi) * 0.5, a_lo, a_hi))
        h[p] = a_star
        h[p + 1] = s - a_star
    h = project_feasible(h, n)
    return h

def smooth_max_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    lse = mx + np.log(np.sum(np.exp(sh - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_polish(h, n, alpha, maxiter, deadline):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_max_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq',
                                     'fun': lambda z: z.sum() - n / 2.0,
                                     'jac': lambda z: np.ones(n)}],
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n_best = len(h_best)
    except Exception:
        h_best = np.full(500, 0.5)
        n_best = 500
    h_best = project_feasible(h_best, n_best)
    best_c = compute_c5(h_best, n_best)
    # Phase 1: SLSQP warmup on initial
    if time.time() < deadline - 200:
        h1 = slsqp_polish(h_best, n_best, 10000, 200, deadline)
        c1 = compute_c5(h1, n_best)
        if c1 < best_c:
            h_best, best_c = h1, c1
    # Phase 2: CD on initial n
    for _ in range(200):
        if time.time() > deadline - 150:
            break
        h_best = cd_pass(h_best, n_best)
    best_c = compute_c5(h_best, n_best)
    # Phase 3: Pair CD
    for _ in range(100):
        if time.time() > deadline - 120:
            break
        h_best = pair_cd_pass(h_best, n_best)
        h_best = cd_pass(h_best, n_best)
    best_c = compute_c5(h_best, n_best)
    # Phase 4: Multi-n exploration
    n_list = list(range(500, 1400, 25))
    n_list.sort(key=lambda k: abs(k - n_best))
    for n_try in n_list:
        if time.time() > deadline - 80:
            break
        if n_try == n_best:
            continue
        x_old = np.linspace(0, 2, n_best)
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_old, h_best), n_try)
        for _ in range(80):
            if time.time() > deadline - 40:
                break
            h_int = cd_pass(h_int, n_try)
        h_int = pair_cd_pass(h_int, n_try)
        h_int = cd_pass(h_int, n_try)
        c_try = compute_c5(h_int, n_try)
        if c_try < best_c - 1e-9:
            h_best, best_c, n_best = h_int, c_try, n_try
    # Phase 5: Multi-start perturbation
    scales = [0.002, 0.004, 0.008, 0.015, 0.03, 0.003, 0.006, 0.012, 0.02, 0.005]
    for i, s in enumerate(scales):
        if time.time() > deadline - 50:
            break
        hp = project_feasible(h_best + rng.normal(0, s, n_best), n_best)
        for _ in range(60):
            if time.time() > deadline - 30:
                break
            hp = cd_pass(hp, n_best)
        hp = pair_cd_pass(hp, n_best)
        c_try = compute_c5(hp, n_best)
        if c_try < best_c:
            h_best, best_c = hp, c_try
    # Phase 6: SLSQP polish on best
    if time.time() < deadline - 60:
        h5 = slsqp_polish(h_best, n_best, 50000, 200, deadline)
        c5 = compute_c5(h5, n_best)
        if c5 < best_c:
            h_best, best_c = h5, c5
    # Phase 7: Final CD squeeze
    for _ in range(150):
        if time.time() > deadline - 2:
            break
        h_best = cd_pass(h_best, n_best)
    h_best = project_feasible(h_best, n_best)
    best_c = compute_c5(h_best, n_best)
    return h_best, best_c, n_best
