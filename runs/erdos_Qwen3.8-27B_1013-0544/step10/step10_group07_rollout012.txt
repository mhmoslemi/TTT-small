import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def precompute_grad_matrices(n):
    js = np.arange(n)
    ks = np.arange(2 * n - 1)
    idx1 = (js[:, None] + ks[None, :] - (n - 1)).clip(0, n - 1).astype(np.int32)
    idx2 = (js[:, None] - ks[None, :] + (n - 1)).clip(0, n - 1).astype(np.int32)
    valid1 = (js[:, None] + ks[None, :] - (n - 1))
    valid1 = (valid1 >= 0) & (valid1 < n)
    valid2 = (js[:, None] - ks[None, :] + (n - 1))
    valid2 = (valid2 >= 0) & (valid2 < n)
    return idx1, idx2, valid1, valid2

def obj_grad(h, n, alpha, idx1, idx2, valid1, valid2):
    c = np.correlate(h, 1.0 - h, mode='full')
    cm = c.max()
    shifted = alpha * (c - cm)
    exp_s = np.exp(shifted)
    exp_sum = exp_s.sum()
    p = exp_s / exp_sum
    obj = (cm + np.log(exp_sum) / alpha) * (2.0 / n)
    
    h1 = 1.0 - h[idx1]
    h1[~valid1] = 0.0
    h2 = h[idx2]
    h2[~valid2] = 0.0
    grad = (2.0 / n) * (h1 @ p - h2 @ p)
    return obj, grad

def nesterov_gd(h, n, alpha, max_iters, deadline, initial_step=None):
    idx1, idx2, valid1, valid2 = precompute_grad_matrices(n)
    h = h.copy()
    h_prev = h.copy()
    if initial_step is None:
        step = 2.0 / alpha
    else:
        step = initial_step
    mu = 0.9  # momentum
    best_obj = obj_grad(h, n, alpha, idx1, idx2, valid1, valid2)[0]
    best_h = h.copy()
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        # Nesterov extrapolation
        h_ex = h + mu * (h - h_prev)
        h_ex = np.clip(h_ex, 0.0, 1.0)
        # Ensure sum constraint approximately
        obj_e, grad_e = obj_grad(h_ex, n, alpha, idx1, idx2, valid1, valid2)
        
        # Backtracking line search from h
        h_try = h - step * grad_e
        h_try = project_feasible(h_try, n)
        obj_t, _ = obj_grad(h_try, n, alpha, idx1, idx2, valid1, valid2)
        
        backtracks = 0
        while obj_t > obj_e - 1e-12 * step * np.abs(grad_e @ (h - h_ex)) and backtracks < 20:
            step *= 0.5
            h_try = h - step * grad_e
            h_try = project_feasible(h_try, n)
            obj_t, _ = obj_grad(h_try, n, alpha, idx1, idx2, valid1, valid2)
            backtracks += 1
        
        h_prev = h
        h = h_try
        step *= 1.05  # slightly increase step
        
        if obj_t < best_obj:
            best_obj = obj_t
            best_h = h.copy()
    
    return best_h

def cd_polish(h, n, max_passes, deadline):
    """Simple coordinate descent polishing."""
    h = h.copy()
    best_c = compute_c5(h, n)
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-10 or h0 > 1.0 - 1e-10:
                continue
            # Try small perturbations
            best_val = compute_c5(h, n)
            for delta in [0.01, -0.01, 0.005, -0.005, 0.02, -0.02]:
                h[p] = h0 + delta
                if h[p] < 0 or h[p] > 1:
                    h[p] = h0
                    continue
                h[p] = np.clip(h[p], 0, 1)
                # Adjust sum
                h_t = project_feasible(h, n)
                c_t = compute_c5(h_t, n)
                if c_t < best_val - 1e-12:
                    best_val = c_t
                    h = h_t
                    improved = True
                    break
            h[p] = h0 if abs(compute_c5(h, n) - best_val) > 1e-12 else h[p]
        if not improved:
            break
    return h, compute_c5(h, n)
