import numpy as np
import time
import math
from scipy.optimize import minimize

def grid(n):
    n = int(n)
    return (np.arange(n) + 0.5) * (2.0 / float(n))

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / float(n)))

def project_feasible(h, n):
    h = np.clip(np.array(h, dtype=float, copy=True), 0.0, 1.0)
    target = 0.5 * float(n)
    if abs(h.sum() - target) < 1e-13:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(h + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(120):
        diff = target - float(h.sum())
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if idx.size == 0:
            break
        step = diff / float(idx.size)
        h[idx] = np.clip(h[idx] + step, 0.0, 1.0)
    return h

def fallback_h(n):
    x = grid(n)
    h = 0.5 + 0.25 * np.sin(3.0 * np.pi * x) + 0.12 * np.sin(7.0 * np.pi * x + 0.7) + 0.08 * np.sin(13.0 * np.pi * x + 1.3)
    return project_feasible(h, n)

def smooth_loss(x, n, alpha, lam):
    x = np.clip(np.asarray(x, dtype=float), 0.0, 1.0)
    c = np.correlate(x, 1.0 - x, mode="full")
    cm = float(c.max())
    z = alpha * (c - cm)
    lse = cm + float(np.log(np.sum(np.exp(z)))) / alpha
    s = float(x.sum()) - 0.5 * float(n)
    return lse + lam * s * s

def smooth_grad(x, n, alpha, lam):
    x = np.clip(np.asarray(x, dtype=float), 0.0, 1.0)
    c = np.correlate(x, 1.0 - x, mode="full")
    cm = float(c.max())
    z = alpha * (c - cm)
    e = np.exp(z)
    w = e / float(e.sum())
    d = np.zeros(n, dtype=float)
    nn = int(n)
    N = nn - 1
    xx = x
    for q in range(2 * nn - 1):
        wq = float(w[q])
        if wq <= 1e-16:
            continue
        l = q - N
        if l >= 0:
            m = nn - l
            if m <= 0:
                continue
            d[l:l + m] += wq * (1.0 - xx[:m])
            d[:m] -= wq * xx[l:l + m]
        else:
            r = -l
            m = nn - r
            if m <= 0:
                continue
            d[:m] += wq * (1.0 - xx[r:r + m])
            d[r:r + m] -= wq * xx[:m]
    s = float(xx.sum()) - 0.5 * float(nn)
    d += (2.0 * lam * s) * np.ones(nn)
    return d

def smooth_optimize(h, n, alphas, deadline, maxiter=60, lam=1.0):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur = h.copy()
    bounds = [(0.0, 1.0)] * int(n)
    scale = max(1.0, float(n) / 700.0)
    for base in alphas:
        if time.time() > deadline:
            break
        alpha = max(1.0, float(base) * scale)
        try:
            res = minimize(smooth_loss, cur, args=(int(n), alpha, lam), jac=smooth_grad,
                           method="L-BFGS-B", bounds=bounds,
                           options={"maxiter": maxiter, "ftol": 1e-12, "gtol": 1e-8, "maxls": 30})
            nh = project_feasible(res.x, n)
            nc = compute_c5(nh, n)
            if nc < best_c - 1e-14:
                best_h, best_c = nh.copy(), nc
            if nc < best_c + 0.02:
                cur = nh.copy()
            else:
                cur = best_h.copy()
            if abs(float(res.x.sum()) - 0.5 * float(n)) > 1.0:
                lam = min(1000.0, max(lam * 5.0, 10.0))
        except Exception:
            pass
    return best_h, best_c

def cd1d_update(c, h, n, p):
    h0 = float(h[p])
    N = int(n) - 1
    if p <= N - p:
        qmin, qmax = p, 2 * N - p
    else:
        qmin, qmax = N - p, N + p
    L = qmax - qmin + 1
    A = np.zeros(L, dtype=float)
    s1 = p - qmin
    A[s1:s1 + n] += 1.0
    A[s1:s1 + n] -= h[::-1]
    s2 = (N - p) - qmin
    A[s2:s2 + n] -= h
    idxd = N - qmin
    c_aff = c[qmin:qmax + 1]
    unaff = 0.0
    if qmin > 0:
        unaff = max(unaff, float(np.max(c[:qmin])))
    if qmax < 2 * N:
        unaff = max(unaff, float(np.max(c[qmax + 1:])))
    cur = max(unaff, float(np.max(c_aff)))
    if cur <= unaff + 1e-13:
        return False
    c_diag = float(c_aff[idxd])
    A_diag = float(A[idxd])
    neg = A < -1e-14
    pos = A > 1e-14
    neg[idxd] = False
    pos[idxd] = False
    neg_n = int(np.count_nonzero(neg))
    pos_n = int(np.count_nonzero(pos))
    if neg_n:
        c_neg = c_aff[neg].copy()
        A_neg = A[neg].copy()
    else:
        c_neg = np.empty(0)
        A_neg = np.empty(0)
    if pos_n:
        c_pos = c_aff[pos].copy()
        A_pos = A[pos].copy()
    else:
        c_pos = np.empty(0)
        A_pos = np.empty(0)
    zero = (np.abs(A) <= 1e-14)
    zero[idxd] = False
    zero_max = float(np.max(c_aff[zero])) if np.any(zero) else -1e300
    lz, hz = -h0, 1.0 - h0
    loL, hiL = unaff, cur
    for _ in range(34):
        Lm = 0.5 * (loL + hiL)
        ok = True
        if zero_max > Lm + 1e-14:
            ok = False
        zlo, zhi = lz, hz
        if ok and neg_n:
            t = float(np.max((Lm - c_neg) / A_neg))
            if t > zlo:
                zlo = t
        if ok and pos_n:
            t = float(np.min((Lm - c_pos) / A_pos))
            if t < zhi:
                zhi = t
        if ok and zlo > zhi + 1e-14:
            ok = False
        if ok:
            disc = A_diag * A_diag + 4.0 * (c_diag - Lm)
            if disc >= 0.0:
                sq = math.sqrt(disc)
                r1 = 0.5 * (A_diag - sq)
                r2 = 0.5 * (A_diag + sq)
                if zlo > r1 + 1e-14 and zhi < r2 - 1e-14:
                    ok = False
        if ok:
            hiL = Lm
        else:
            loL = Lm
    Lbest = hiL
    zlo, zhi = lz, hz
    if neg_n:
        t = float(np.max((Lbest - c_neg) / A_neg))
        if t > zlo:
            zlo = t
    if pos_n:
        t = float(np.min((Lbest - c_pos) / A_pos))
        if t < zhi:
            zhi = t
    disc = A_diag * A_diag + 4.0 * (c_diag - Lbest)
    cands = [0.0, lz, hz, zlo, zhi]
    if disc >= 0.0:
        sq = math.sqrt(disc)
        r1 = 0.5 * (A_diag - sq)
        r2 = 0.5 * (A_diag + sq)
        cands.extend([r1, r2])
        if r1 > lz:
            cands.append(0.5 * (lz + min(r1, hz)))
        if r2 < hz:
            cands.append(0.5 * (max(r2, lz) + hz))
    bestz, bestv = 0.0, cur
    for zz in cands:
        if zz < lz or zz > hz:
            continue
        vals = c_aff + A * zz
        vals[idxd] -= zz * zz
        v = max(float(np.max(vals)), unaff)
        if v < bestv - 1e-14 or (abs(v - bestv) <= 1e-14 and abs(zz) < abs(bestz)):
            bestv, bestz = v, float(zz)
    if abs(bestz) < 1e-13 or bestv > cur - 1e-14:
        return False
    xn = h0 + bestz
    if xn <= 0.0 or xn >= 1.0:
        return False
    h[p] = float(xn)
    c[qmin:qmax + 1] += bestz * A
    c[N] -= bestz * bestz
    return True

def exact_cd(h, n, max_passes, deadline, rng):
    h = project_feasible(h, n)
    c = np.correlate(h, 1.0 - h, mode="full")
    best_h = h.copy()
    best_c = compute_c5(h, n)
    stagn = 0
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        order = np.arange(n)
        if rng is not None:
            rng.shuffle(order)
        moved = False
        for p in order:
            if cd1d_update(c, h, int(n), int(p)):
                moved = True
        h = project_feasible(h, n)
        c = np.correlate(h, 1.0 - h, mode="full")
        val = compute_c5(h, n)
        if val < best_c - 1e-14:
            best_h, best_c, stagn = h.copy(), val, 0
        else:
            stagn += 1
            if (not moved and stagn >= 3) or stagn >= 12:
                break
    return best_h, best_c

def interpolate_h(h, n_from, n_to):
    n_from, n_to = int(n_from), int(n_to)
    if n_from == n_to:
        return project_feasible(h, n_from)
    return project_feasible(np.interp(grid(n_to), grid(n_from), h), n_to)

def perturb_h(h, n, rng, style, strength):
    out = h.copy()
    nn = int(n)
    if style == 0:
        out += rng.normal(0.0, strength, nn)
    elif style == 1:
        nb = 2 + rng.randint(0, 3)
        for _ in range(nb):
            st = rng.randint(0, nn)
            w = max(5, int(nn * (0.02 + 0.03 * rng.rand())))
            en = min(nn, st + w)
            out[st:en] = 1.0 - out[st:en]
    elif style == 2:
        x = grid(nn)
        low = np.zeros(nn, dtype=float)
        for _ in range(5):
            f = 1 + rng.randint(0, 12)
            ph = rng.uniform(0.0, 2.0 * np.pi)
            low += rng.normal(0.0, strength) * np.sin(np.pi * f * x + ph)
        out += low
    else:
        out = 0.5 * (out + out[::-1])
    return project_feasible(out, nn)

def run(seed=42, budget_s=1100, **kwargs):

    hbest = project_feasible(hbest, int(nbest))
    cbest = compute_c5(hbest, int(nbest))
    return hbest, cbest, int(nbest)
