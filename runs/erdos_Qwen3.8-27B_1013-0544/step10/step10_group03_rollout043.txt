Actually, one more important consideration: the parent's `exact_cd` function has a subtle issue. After each full pass, it calls `project_feasible` to maintain the sum constraint. But this projection can change the values in ways that undo the CD improvements. The parent handles this by recomputing c if the sum is off, but the projection itself can be suboptimal.

A better approach: instead of projecting after each pass, project only when the sum is significantly off, and use a more careful projection that minimizes the change to h.

The parent's projection is:
- If sum > target: scale down proportional to h (h * target/sum)
- If sum < target: distribute the deficit proportional to room (1-h)

def project_l2(h, n):
    """Project h onto {0<=x<=1, sum(x)=n/2} in L2 sense."""
    target = n / 2.0
    h = h.copy()
    for _ in range(100):
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        # Binary search for lambda
        lo, hi = -max(h.max(), 1.0), max(1.0 - h.min(), 1.0)
        for _ in range(60):
            mid = (lo + hi) / 2
            h_proj = np.clip(h + mid, 0.0, 1.0)
            s_proj = h_proj.sum()
            if s_proj > target:
                hi = mid
            else:
                lo = mid
        h = np.clip(h + (lo + hi) / 2, 0.0, 1.0)
    return h

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_l2(h, n):
    target = n / 2.0
    h = h.copy()
    for _ in range(200):
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        lo, hi = -np.max(np.abs(h)) - 1.0, np.max(np.abs(1.0 - h)) + 1.0
        for _ in range(60):
            mid = (lo + hi) / 2.0
            s_mid = np.clip(h + mid, 0.0, 1.0).sum()
            if s_mid > target:
                hi = mid
            else:
                lo = mid
        h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    return h

def solve_1d_bisect(c, h, p, n):
    h0 = h[p]
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms]
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    if has_diag:
        a[diag_idx] = 0.0
    unaff = 0.0
    if m_min > 0:
        unaff = float(np.max(c[:m_min]))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    cur_max = float(np.max(c))
    neg_mask = a < -1e-15
    pos_mask = a > 1e-15
    zero_mask = np.abs(a) <= 1e-15
    L_lo, L_hi = unaff, cur_max
    for _ in range(80):
        if L_hi - L_lo < 1e-14:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        feasible = True
        if np.any(zero_mask) and np.any(c_base[zero_mask] > L):
            feasible = False
        if feasible and np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if feasible and np.any(pos_mask):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        if x_lo > x_hi + 1e-13:
            feasible = False
        if feasible and has_diag:
            c_diag = c_base[diag_idx[0]]
            Cq = L - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                xc_lo, xc_hi = max(x_lo, 0.0), min(x_hi, 1.0)
                if not (xc_lo < r1 and xc_hi > r2):
                    feasible = False
        if feasible:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        c_diag = c_base[diag_idx[0]]
        Cq = L - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            if h0 - r1 <= r2 - h0:
                x_hi = min(x_hi, r1)
            else:
                x_lo = max(x_lo, r2)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def exact_cd(h, n, max_passes, deadline):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    c5 = compute_c5(h, n)
    for it in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            xn = solve_1d_bisect(c, h, p, n)
            if abs(xn - h0) > 1e-12:
                delta = xn - h0
                ms = np.arange(n - 1 - p, 2 * n - 1 - p)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                diag_i = np.where(ms == (n - 1))[0]
                c[ms] += delta * a
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_l2(h, n)
        new_c5 = compute_c5(h, n)
        c5 = new_c5
        if abs(h.sum() - n / 2.0) > 1e-12:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, c5

def thue_morse_seed(n, rng):
    h = np.zeros(n)
    for i in range(n):
        x = i
        cnt = 0
        while x:
            cnt ^= 1
            x &= x - 1
        h[i] = float(cnt)
    ones = (h == 1.0).sum()
    target = n // 2
    if ones > target:
        idx = np.where(h == 1.0)[0]
        flip = rng.choice(idx, ones - target, replace=False)
        h[flip] = 0.0
    elif ones < target:
        idx = np.where(h == 0.0)[0]
        flip = rng.choice(idx, target - ones, replace=False)
        h[flip] = 1.0
    return h

def golden_ratio_seed(n, rng):
    phi_inv = 1.0 / ((1.0 + np.sqrt(5.0)) / 2.0)
    x = (np.arange(n) * phi_inv) % 1.0
    order = np.argsort(x)
    h = np.zeros(n)
    h[order[:n // 2]] = 1.0
    return h

def random_binary_seed(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h

def soft_from_binary(h_bin, blend, rng):
    h = 0.5 + (h_bin - 0.5) * (1.0 - blend)
    h += rng.normal(0, 0.02, len(h))
    return project_l2(np.clip(h, 0.0, 1.0), len(h))

def binary_local_search(h_bin, n, swaps, deadline):
    h = h_bin.copy()
    c5 = compute_c5(h, n)
    for _ in range(swaps):
        if time.time() > deadline:

        i = np.random.randint(n)
        j = np.random.randint(n)
        if h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        c_new = compute_c5(h, n)
        if c_new < c5:
            c5 = c_new
        else:
            h[i], h[j] = h[j], h[i]
    return h, c5
