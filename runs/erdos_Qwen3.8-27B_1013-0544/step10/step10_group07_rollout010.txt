import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def smooth_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    exps = np.exp(shifted - mx)
    w = exps / exps.sum()  # softmax weights
    # grad[p] = (2/n) * sum_m w[m] * [(1-h[p-s])*[valid] - h[p+s]*[valid]]
    # where s = m - (n-1)
    grad = np.zeros(n)
    shifts = np.arange(2*n-1) - (n-1)
    for m in range(2*n-1):
        s = shifts[m]
        wm = w[m]
        if wm < 1e-15:
            continue
        # term1: (1-h[p-s]) for valid p-s
        q = np.arange(n) - s
        vq = (q >= 0) & (q < n)
        grad[vq] += wm * (1.0 - x[q[vq]])
        # term2: -h[p+s] for valid p+s
        r = np.arange(n) + s
        vr = (r >= 0) & (r < n)
        grad[vr] -= wm * x[r[vr]]
    grad *= (2.0 / n)
    return grad

def slsqp_polish(h, n, alpha, maxiter, deadline):
    def obj(z):
        return smooth_obj(z, n, alpha)
    def grad(z):
        return smooth_grad(z, n, alpha)
    def cons_fun(z):
        return z.sum() - n/2.0
    def cons_jac(z):
        return np.ones(n)
    
    best_h = h.copy()
    best_c = compute_c5(h, n)
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP',
                      bounds=[(0,1)]*n,
                      constraints=[{'type':'eq','fun':cons_fun,'jac':cons_jac}],
                      options={'maxiter':maxiter,'ftol':1e-20,'iprint':0})
        new_h = project_feasible(res.x, n)
        new_c = compute_c5(new_h, n)
        if new_c < best_c:
            best_h, best_c = new_h, new_c
    except Exception:
        pass
    return best_h, best_c

    neg_mask = a < -1e-15

        disc = 1.0 - 4.0*Cq

    return float(np.clip((x_lo + x_hi)/2.0, 0.0, 1.0))

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def smooth_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    exps = np.exp(shifted - mx)
    w = exps / exps.sum()
    grad = np.zeros(n)
    shifts = np.arange(2 * n - 1) - (n - 1)
    idxs = np.arange(n)
    for m_i in range(2 * n - 1):
        wm = w[m_i]
        if wm < 1e-18:
            continue
        s = shifts[m_i]
        q = idxs - s
        vq = (q >= 0) & (q < n)
        if np.any(vq):
            grad[vq] += wm * (1.0 - x[q[vq]])
        r = idxs + s
        vr = (r >= 0) & (r < n)
        if np.any(vr):
            grad[vr] -= wm * x[r[vr]]
    grad *= (2.0 / n)
    return grad

def slsqp_opt(h, n, alpha, maxiter, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), jac=lambda z: smooth_grad(z, n, alpha),
                       method='SLSQP', bounds=[(0, 1)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda z: z.sum() - n / 2.0,
                                     'jac': lambda z: np.ones(n)}],
                       options={'maxiter': maxiter, 'ftol': 1e-22, 'iprint': 0})
        new_h = project_feasible(res.x, n)
        new_c = compute_c5(new_h, n)
        if new_c < best_c:
            best_h, best_c = new_h, new_c
    except Exception:
        pass
    return best_h, best_c

def cd_1d(c, h, p, n):
    h0 = h[p]
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms].copy()
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    if has_diag:
        c_diag = c_base[diag_idx[0]]
        a[diag_idx] = 0.0
    unaff = 0.0
    if m_min > 0:
        unaff = max(unaff, float(np.max(c[:m_min])))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    cur_max = float(np.max(c))
    neg_mask = a < -1e-15
    pos_mask = a > 1e-15
    zero_mask = np.abs(a) <= 1e-15
    L_lo, L_hi = unaff, cur_max
    for _ in range(50):
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        ok = True
        if np.any(zero_mask) and np.any(c_base[zero_mask] > L + 1e-15):
            ok = False
        if ok and np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if ok and np.any(pos_mask):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        if x_lo > x_hi + 1e-14:
            ok = False
        if ok and has_diag:
            Cq = L - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                if not (max(x_lo, 0.0) < r1 and min(x_hi, 1.0) > r2):
                    ok = False
        if ok:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Cq = L - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            if max(x_lo, 0.0) < r1 and min(x_hi, 1.0) > r2:
                x_hi = min(x_hi, r1) if h0 - r1 <= r2 - h0 else max(x_lo, r2)
                x_lo = min(x_lo, r2) if h0 - r1 <= r2 - h0 else max(x_lo, r2)
            elif x_lo >= r1 and x_hi <= r2:
                x_hi = min(x_hi, r1) if h0 - r1 <= r2 - h0 else max(x_lo, r2)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def exact_cd(h, n, max_passes, deadline, rng=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        order = np.arange(n) if rng is None else rng.permutation(n)
        for p in order:
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            xn = cd_1d(c, h, p, n)
            if abs(xn - h0) > 1e-12:
                delta = xn - h0
                m_min = n - 1 - p
                m_max = 2 * n - 2 - p
                ms = np.arange(m_min, m_max + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                diag_i = np.where(ms == (n - 1))[0]
                c[ms] += delta * a
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True
        h = project_feasible(h, n)
        if not improved:
            break
    return h, compute_c5(h, n)
