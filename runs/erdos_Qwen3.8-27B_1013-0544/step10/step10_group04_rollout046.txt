import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def solve_1d_bisect(c, h, p, n):
    h0 = h[p]
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms]
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    if has_diag:
        c_diag = c_base[diag_idx[0]]
        a[diag_idx] = 0.0
    unaff = 0.0
    if m_min > 0:
        unaff = float(np.max(c[:m_min]))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    cur_max = float(np.max(c))
    neg_mask = a < -1e-15
    pos_mask = a > 1e-15
    zero_mask = np.abs(a) <= 1e-15
    L_lo, L_hi = unaff, cur_max
    for _ in range(50):
        if L_hi - L_lo < 1e-15:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        feasible = True
        if np.any(zero_mask) and np.any(c_base[zero_mask] > L):
            feasible = False
        if feasible and np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if feasible and np.any(pos_mask):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        if x_lo > x_hi + 1e-14:
            feasible = False
        if feasible and has_diag:
            Cq = L - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                xc_lo, xc_hi = max(x_lo, 0.0), min(x_hi, 1.0)
                if not (xc_lo < r1 and xc_hi > r2):
                    feasible = False
        if feasible:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Cq = L - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0

            if (xc_lo < r1 and xc_hi > r2) or (x_lo >= r1 and x_hi <= r2):

        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def exact_cd(h, n, max_passes, deadline, order_func=None, rng=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        if order_func is not None and rng is not None:
            order = order_func(n, rng)
        else:
            order = np.arange(n)
        for p in order:
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            xn = solve_1d_bisect(c, h, p, n)
            if abs(xn - h0) > 1e-13:
                delta = xn - h0
                ms = np.arange(n - 1 - p, 2 * n - 1 - p)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                diag_i = np.where(ms == (n - 1))[0]
                c[ms] += delta * a
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def random_order(n, rng):
    return rng.permutation(n)

def reverse_order(n, rng):
    return np.arange(n - 1, -1, -1)

def serpentine_order(n, rng):
    order = np.arange(n)
    for i in range(0, n, 2):
        end = min(i + 2, n)
        if (i // 2) % 2 == 1:
            order[i:end] = order[i:end][::-1]
    return order

def multi_start_variant(h, n, variant_id, rng):
    x = h.copy()
    if variant_id == 0:
        return x[::-1].copy()
    elif variant_id == 1:
        k = rng.randint(1, n)
        return np.roll(x, k).copy()
    elif variant_id == 2:
        freq = rng.randint(2, 8)
        x = x + 0.02 * np.sin(2 * np.pi * freq * np.linspace(0, 2, n))
        return x
    elif variant_id == 3:
        x = x + 0.05 * rng.normal(0, 1, n)
        return x
    elif variant_id == 4:
        blk = n // 4
        b1, b2 = sorted(rng.choice(range(4), 2, replace=False))
        x[b1*blk:(b1+1)*blk], x[b2*blk:(b2+1)*blk] = x[b2*blk:(b2+1)*blk], x[b1*blk:(b1+1)*blk]
        return x
    elif variant_id == 5:
        freq = rng.randint(3, 15)
        x = x + 0.03 * np.cos(2 * np.pi * freq * np.linspace(0, 2, n) + rng.uniform(0, 2*np.pi))
        return x
    elif variant_id == 6:
        x = x + 0.01 * rng.normal(0, 1, n)
        return x
    elif variant_id == 7:
        mid = n // 3
        x[mid:2*mid] = 1.0 - x[mid:2*mid]
        return x
    else:
        x = 0.7 * x + 0.3 * rng.uniform(0, 1, n)
        return x

def coordinate_perturb_search(h, n, deadline, rng):
    h = h.copy()
    best_c = compute_c5(h, n)
    deltas = np.array([0.001, 0.003, 0.005, 0.01, 0.02, 0.05, -0.001, -0.003, -0.005, -0.01, -0.02, -0.05])

        return h, best_c
