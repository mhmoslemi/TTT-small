        obj, grad = soft_max_obj_grad(y, n, alpha)
        # Adaptive step size
        step = 1.0 / (alpha * 4.0)
        # Backtracking
        z = project_feasible(y - step * grad, n)
        z_obj = None
        for bs in range(10):
            z_obj, _ = soft_max_obj_grad(z, n, alpha)
            if z_obj <= obj + 0.5 * step * np.dot(grad, h - y):
                break
            step *= 0.5
            z = project_feasible(y - step * grad, n)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = z + (t_k - 1.0) / t_new * (z - h)
        h = z.copy()
        t_k = t_new
        if obj < best_obj:
            best_obj = obj
            best_h = h.copy()
    return best_h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-14:
        return h.copy()
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s_try = np.clip(h - mid, 0.0, 1.0).sum()
        if s_try > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    h_proj = np.clip(h - tau, 0.0, 1.0)
    diff = target - h_proj.sum()
    if abs(diff) > 1e-11:
        if diff > 0:
            mask = h_proj < 1.0 - 1e-10
        else:
            mask = h_proj > 1e-10
        if mask.any():
            h_proj[mask] += diff / mask.sum()
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def apply_symmetry(h, n):
    return project_feasible(0.5 * (h + 1.0 - h[::-1]), n)

def soft_max_obj_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = float(c.max())
    shifted = alpha * (c - m)
    mx = float(shifted.max())
    exp_vals = np.exp(shifted - mx)
    lse = mx + np.log(float(exp_vals.sum()))
    obj = (m + lse / alpha) * (2.0 / n)
    weights = exp_vals / float(exp_vals.sum())
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-14:
            continue
        kf = k - (n - 1)
        if kf >= 0:
            grad[kf:] += w * (1.0 - h[:n - kf]) * dx
            grad[:n - kf] -= w * h[kf:] * dx
        else:
            jk = -kf
            grad[:n - jk] += w * (1.0 - h[jk:]) * dx
            grad[jk:] -= w * h[:n - jk] * dx
    return obj, grad

def fista_opt(h, n, alphas, max_iter_each, deadline):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    for alpha in alphas:
        if time.time() > deadline - 10:
            break
        h_cur = best_h.copy()
        y = h_cur.copy()
        t_k = 1.0
        step = 2.0 / (n * alpha)
        for it in range(max_iter_each):
            if time.time() > deadline - 5:
                break
            obj, grad = soft_max_obj_grad(y, n, alpha)
            z = project_feasible(y - step * grad, n)
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
            y = z + (t_k - 1.0) / t_new * (z - h_cur)
            h_cur = z.copy()
            t_k = t_new
            c_cur = compute_c5(h_cur, n)
            if c_cur < best_c:
                best_c = c_cur
                best_h = h_cur.copy()
    return best_h, best_c

def solve_coord_fast(c, h, p, n, nbisect=35):
    h0 = h[p]
    if h0 < 1e-14 or h0 > 1.0 - 1e-14:
        return h0
    lo_a = max(0, n - 1 - p)
    hi_a = min(2 * n - 2, n - 1 + p)
    m_lo = min(lo_a, max(0, p))
    m_hi = max(hi_a, min(2 * n - 2, p + n - 1))
    ms = np.arange(m_lo, m_hi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms].copy()
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    c_diag = 0.0
    if has_diag:
        c_diag = c_base[diag_idx[0]]
        a[diag_idx] = 0.0
    non_diag = np.ones(len(ms), dtype=bool)
    if has_diag:
        non_diag[diag_idx] = False
    a_nd = a[non_diag]
    c_nd = c_base[non_diag]
    unaff_max = 0.0
    if m_lo > 0:
        unaff_max = max(unaff_max, float(np.max(c[:m_lo])))
    if m_hi < 2 * n - 2:
        unaff_max = max(unaff_max, float(np.max(c[m_hi + 1:])))
    cur_max = float(np.max(c))
    L_lo = unaff_max
    L_hi = cur_max
    for _ in range(nbisect):
        if L_hi - L_lo < 1e-14:
            break
        T = (L_lo + L_hi) / 2.0
        feasible = True
        if unaff_max > T + 1e-15:
            feasible = False
        elif feasible and len(a_nd) > 0:
            x_lo_c = 0.0
            x_hi_c = 1.0
            neg = a_nd < -1e-13
            pos = a_nd > 1e-13
            if np.any(neg):
                x_lo_c = max(x_lo_c, float(np.max((T - c_nd[neg]) / a_nd[neg])))
            if np.any(pos):
                x_hi_c = min(x_hi_c, float(np.min((T - c_nd[pos]) / a_nd[pos])))
            if x_lo_c > x_hi_c + 1e-14:
                feasible = False
        if feasible and has_diag:
            Cq = T - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                if len(a_nd) > 0:
                    neg = a_nd < -1e-13
                    pos = a_nd > 1e-13
                    xl = 0.0
                    xh = 1.0
                    if np.any(neg):
                        xl = max(xl, float(np.max((T - c_nd[neg]) / a_nd[neg])))
                    if np.any(pos):
                        xh = min(xh, float(np.min((T - c_nd[pos]) / a_nd[pos])))
                    xl = max(xl, 0.0)
                    xh = min(xh, 1.0)
                    if xl > xh + 1e-14:
                        feasible = False
                    elif xl >= r1 and xh <= r2:
                        feasible = False
                else:
                    if 0.0 >= r1 and 1.0 <= r2:
                        feasible = False
                    elif 0.0 < r1 and 1.0 > r2:
                        feasible = False
        if feasible:
            L_hi = T
        else:
            L_lo = T
    T = (L_lo + L_hi) / 2.0
    x_lo = 0.0
    x_hi = 1.0
    if len(a_nd) > 0:
        neg = a_nd < -1e-13
        pos = a_nd > 1e-13
        if np.any(neg):
            x_lo = max(x_lo, float(np.max((T - c_nd[neg]) / a_nd[neg])))
        if np.any(pos):
            x_hi = min(x_hi, float(np.min((T - c_nd[pos]) / a_nd[pos])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Cq = T - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            if x_lo < r1:
                x_hi = min(x_hi, r1)
            elif x_hi > r2:
                x_lo = max(x_lo, r2)
            else:
                if h0 - r1 <= r2 - h0:
                    x_hi = min(x_hi, r1 + 1e-9)
                else:
                    x_lo = max(x_lo, r2 - 1e-9)
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if x_lo > x_hi + 1e-10:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def coord_descent(h, n, max_passes, deadline, rng=None, random_order=True, nbisect=35):
    h = h.copy()
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        if random_order and rng is not None:
            order = rng.permutation(n)
        else:
            order = np.arange(n)
        for p in order:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            x_new = solve_coord_fast(c, h, p, n, nbisect)
            if abs(x_new - h0) > 1e-13:
                h[p] = x_new
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def perturb(h, n, sigma, rng):
    noise = rng.randn(n) * sigma
    return project_feasible(h + noise, n)

def resolve_at_n(h_src, n_src, n_target, deadline, rng):
    x_src = np.linspace(0, 2, n_src)
    x_tgt = np.linspace(0, 2, n_target)
    h_new = project_feasible(np.interp(x_tgt, x_src, h_src), n_target)
    h_new = apply_symmetry(h_new, n_target)
    h_new, c_new = fista_opt(h_new, n_target, [500, 2000], 200, deadline)
    h_new, c_new = coord_descent(h_new, n_target, 50, deadline, rng, True, 35)
    return h_new, c_new
