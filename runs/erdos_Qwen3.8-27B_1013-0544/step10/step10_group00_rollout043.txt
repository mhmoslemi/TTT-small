```python
def adam_optimize(h, n, deadline, lr=0.01, alpha=1000):
    # Adam state
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    t = 0
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    while time.time() < deadline:
        t += 1
        # Compute correlation
        c = np.correlate(h, 1.0 - h, mode='full')
        # Compute smooth objective gradient
        m_max = c.max()
        shifted = alpha * (c - m_max)
        weights = np.exp(shifted - shifted.max())
        weights /= weights.sum()
        
        # Gradient: grad[j] = sum_m weights[m] * dc[m]/dh[j]
        grad = np.zeros(n)
        for j in range(n):
            g = 0.0
            for s in range(-n+1, n):
                m_idx = n - 1 + s
                w = weights[m_idx]
                if w < 1e-10:
                    continue
                # dc[m_idx]/dh[j] contribution
                js = j - s
                js2 = j + s
                if 0 <= js < n:
                    g += w * (1.0 - h[js])
                if 0 <= js2 < n and s != 0:
                    g -= w * h[js2]
                elif s == 0:
                    # diagonal: h[j]*(1-h[j]) derivative is 1-2h[j]
                    # already counted in the first term when s=0
                    pass
            grad[j] = g
        
        # Adam update
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_h, best_c = h_new.copy(), c_new
        h = h_new
    
    return best_h, best_c
```
