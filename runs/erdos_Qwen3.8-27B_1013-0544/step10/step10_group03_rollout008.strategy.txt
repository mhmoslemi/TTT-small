    ```python
    def de(evaluate, n_vars, lower_bounds, upper_bounds, popsize=20, mutation_factor=0.5, crossover_rate=0.7, max_generations=20, seed=None):
        # We assume the evaluate function will handle bounds by passing through the corrected chromosome?
        # We decide that evaluate can handle the chromosome without bounds checking.

        rng = np.random.RandomState(seed)
        # Initialize population
        # We require the chromosome length of n_vars.
        # Each chromosome is a list of n_vars real numbers.
        pop = rng.uniform(lower_bounds[0], upper_bounds[0], size=(popsize, n_vars))
        pop = pop.tolist()
        best_fitness = float('inf')
        best_solution = None
        # DE/breeding/crossover type
        for generation in range(max_generations):
            if time.time() > deadline:  # if we have a deadline, passed in evaluate or in the main function?
                break
            # For each individual in the population
            for i in range(popsize):
                if time.time() > deadline: break
                candidate = None
                while candidate is None:
                    # Mutation
                    # Select three distinct random indices
                    idx = rng.choice(popsize, 3, replace=False)
                    a = pop[idx[0]]
                    b = pop[idx[1]]
                    c = pop[idx[2]]
                    # Mutation: DE/rand/1
                    mutant = [a[j] + mutation_factor * (b[j] - c[j]) for j in range(n_vars)]
                    # Optionally, clip to bounds (not inside evaluate)
                    mutant = [max(lower_bounds[j], min(upper_bounds[j], x)) for j, x in enumerate(mutant)]

                # Crossover
                cr = rng.random() < crossover_rate
                child = []
                for j in range(n_vars):
                    if cr or rng.random() < 0.5:
                        child.append(mutant[j])
                    else:
                        child.append(pop[i][j])
                child_fitness = evaluate(child)
                # Forced elitism: keep the best between child and parent
                if child_fitness < evaluate(pop[i]):
                    pop[i] = child
                    if child_fitness < best_fitness:
                        best_fitness = child_fitness
                        best_solution = child

        return best_solution, best_fitness
    ```
