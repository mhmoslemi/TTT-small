def alloc_exact(total,k,rng):
   if total<=k:
      x=np.ones(k,dtype=int); # total==k usually, if total<k impossible but return min?
      return x
   x=np.ones(k,dtype=int)
   rem=total-k
   # Dirichlet weights
   w=rng.random(k)+1e-3
   x_norm=np.floor((rem*w/w.sum())).astype(int)
   x += x_norm
   # ensure min 1
   # adjust diff
   diff=total-x.sum()
   idxs=np.arange(k)
   while diff>0:
      for _ in range(diff): x[rng.randint(k)]+=1
   while diff<0:
      # pick x>1
      cand=np.where(x>1)[0]
      if cand.size==0: break
      x[rng.randint(cand.size)] -=1; diff+=1

```python
import numpy as np
import time
from scipy.optimize import minimize

UPDATE_CACHE = {}
SL_N = 0
SL_ALPHA = 1.0


def compute_c5(h, n):
    h = np.asarray(h, dtype=float)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)


def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(600):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = np.clip(h + (d / s) * room, 0.0, 1.0)
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = np.clip(h + (d / s) * h, 0.0, 1.0)
    return h


def smooth_lse(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode="full")
    m = float(c.max())
    z = alpha * (c - m)
    zm = z.max()
    return (m + (zm + np.log(np.sum(np.exp(z - zm)))) / alpha) * (2.0 / n)


def smooth_lse_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode="full")
    cm = float(c.max())
    z = alpha * (c - cm)
    zm = z.max()
    w = np.exp(z - zm)
    s = float(w.sum())
    if s <= 0.0:
        return np.zeros(n)
    w /= s
    g = np.zeros(n)
    nm1 = n - 1
    for r in range(2 * n - 1):
        wr = w[r]
        if wr < 1e-18:
            continue
        m = r - nm1
        if m >= 0:
            ln = n - m
            if ln > 0:
                g[:ln] += wr * (1.0 - h[m:m + ln])
                g[m:] -= wr * h[:ln]
        else:
            d = -m
            ln = n - d
            if ln > 0:
                g[d:] += wr * (1.0 - h[:ln])
                g[:ln] -= wr * h[d:]
    return (2.0 / n) * g


def slq_obj(x):
    return smooth_lse(x, SL_N, SL_ALPHA)


def slq_grad(x):
    return smooth_lse_grad(x, SL_N, SL_ALPHA)


def slq_cons_fun(x):
    return float(x.sum()) - 0.5 * SL_N


def slq_cons_jac(x):
    return np.ones_like(x, dtype=float)


def slsqp_opt(h, n, alphas, maxiter, deadline):
    global SL_N, SL_ALPHA
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        best_h = project_feasible(best_h, n)
        SL_N = n
        SL_ALPHA = float(alpha)
        try:
            res = minimize(slq_obj, best_h, jac=slq_grad, method="SLSQP",
                           bounds=[(0.0, 1.0)] * n,
                           constraints=[{"type": "eq", "fun": slq_cons_fun,
                                         "jac": slq_cons_jac}],
                           options={"maxiter": maxiter, "ftol": 1e-15})
            cand = project_feasible(res.x, n)
            cc = compute_c5(cand, n)
            if cc < best_c - 1e-12:
                best_h, best_c = cand, cc
        except Exception:
            pass
    return best_h, best_c


def build_update_data(n):
    data = []
    full = 2 * n - 1
    for p in range(n):
        ms = np.arange(n - 1 - p, 2 * n - 2 - p + 1, dtype=np.int32)
        add = p + ms - (n - 1)
        sub = p - ms + (n - 1)
        av = (add >= 0) & (add < n)
        sv = (sub >= 0) & (sub < n)
        add_s = np.where(av, add, -1).astype(np.int32)
        sub_s = np.where(sv, sub, -1).astype(np.int32)
        cadd = ms[av].astype(np.int32)
        hadd = add[av].astype(np.int32)
        csub = ms[sv].astype(np.int32)
        hsub = sub[sv].astype(np.int32)
        dloc = int(np.where(ms == n - 1)[0][0])
        local = np.full(full, -1, dtype=np.int16)
        local[ms] = np.arange(n, dtype=np.int16)
        data.append((add_s, sub_s, cadd, hadd, csub, hsub, n - 1, dloc, local))
    return data


def get_data(n):
    d = UPDATE_CACHE.get(n)
    if d is None:
        if len(UPDATE_CACHE) >= 3:
            UPDATE_CACHE.clear()
        d = build_update_data(n)
        UPDATE_CACHE[n] = d
    return d


def apply_update(c, h, p, y, dp):
    old = float(h[p])
    delta = y - old
    if abs(delta) < 1e-15:
        return
    add, sub, cadd, hadd, csub, hsub, dc, dloc, local = dp
    if hadd.size:
        c[cadd] += delta * (1.0 - h[hadd])
    if hsub.size:
        c[csub] -= delta * h[hsub]
    c[dc] -= delta * delta
    h[p] = y


def active_cost(cvals, b, lin_mask, didx, dval, old, y):
    if dval is None and didx < 0:
        vals = cvals + (y - old) * b
        return float(np.max(vals))
    vals = cvals.copy()
    if np.any(lin_mask):
        vals[lin_mask] += (y - old) * b[lin_mask]
    if didx >= 0:
        D = dval - old + old * old
        vals[didx] = D + y - y * y
    return float(np.max(vals))


def solve_one(cvals, b, lin_mask, didx, dval, old, high_max):
    if high_max <= 1e-14:
        return old
    neg = lin_mask & (b < -1e-12)
    pos = lin_mask & (b > 1e-12)
    zero = lin_mask & (np.abs(b) <= 1e-12)
    lo, hi = 0.0, float(high_max)
    for _ in range(48):
        L = 0.5 * (lo + hi)
        ok = True
        if ok and np.any(zero) and float(np.max(cvals[zero])) > L + 1e-10:
            ok = False
        xlo, xhi = 0.0, 1.0
        if ok and np.any(neg):
            v = old + (L - cvals[neg]) / b[neg]
            xlo = max(xlo, float(np.max(v)))
        if ok and np.any(pos):
            v = old + (L - cvals[pos]) / b[pos]
            xhi = min(xhi, float(np.min(v)))
        if ok:
            if xlo > 1.0 + 1e-10 or xhi < -1e-10:
                ok = False
            else:
                xlo = max(0.0, min(1.0, xlo))
                xhi = max(0.0, min(1.0, xhi))
                if xlo > xhi + 1e-10:
                    ok = False
        if ok and didx >= 0:
            D = dval - old + old * old
            T = L - D
            if T < -1e-10:
                ok = False
            elif T <= 0.25:
                rad = np.sqrt(max(0.0, 1.0 - 4.0 * T))
                r1 = 0.5 * (1.0 - rad)
                r2 = 0.5 * (1.0 + rad)
                if not ((xlo <= r1 + 1e-10) or (xhi >= r2 - 1e-10)):
                    ok = False
        if ok:
            hi = L
        else:
            lo = L
    L = hi
    xlo, xhi = 0.0, 1.0
    if np.any(neg):
        v = old + (L - cvals[neg]) / b[neg]
        xlo = max(xlo, float(np.max(v)))
    if np.any(pos):
        v = old + (L - cvals[pos]) / b[pos]
        xhi = min(xhi, float(np.min(v)))
    if xlo > 1.0 + 1e-10 or xhi < -1e-10:
        return old
    xlo = max(0.0, min(1.0, xlo))
    xhi = max(0.0, min(1.0, xhi))
    if xlo > xhi:
        return old
    intervals = [(xlo, xhi)]
    if didx >= 0:
        D = dval - old + old * old
        T = L - D
        intervals = []
        if T > 0.25:
            intervals = [(xlo, xhi)]
        else:
            rad = np.sqrt(max(0.0, 1.0 - 4.0 * T))
            r1 = 0.5 * (1.0 - rad)
            r2 = 0.5 * (1.0 + rad)
            if xlo <= r1 + 1e-10:
                a, b2 = max(xlo, 0.0), min(xhi, r1)
                if a <= b2:
                    intervals.append((a, b2))
            if xhi >= r2 - 1e-10:
                a, b2 = max(xlo, r2), min(xhi, 1.0)
                if a <= b2:
                    intervals.append((a, b2))
    best_y = old
    best_val = float(active_cost(cvals, b, lin_mask, didx, dval, old, old))
    for a, b2 in intervals:
        if a < b2:
            y = 0.5 * (a + b2)
            val = active_cost(cvals, b, lin_mask, didx, dval, old, y)
            if val < best_val:
                best_val, best_y = val, y
    return float(np.clip(best_y, 0.0, 1.0))


def top_active(c, cur_max, tol, max_active):
    if c.size <= max_active:
        return np.arange(c.size, dtype=np.int32)
    top = np.argpartition(-c, max_active)[:max_active]
    thr = np.where(c >= cur_max - tol)[0]
    if thr.size <= max(2 * max_active, 64):
        return np.sort(np.unique(np.concatenate((top, thr))))
    return np.sort(top)


def fast_cd(h, n, passes, deadline, rng, active_tol=3.0, max_active=90):
    h = project_feasible(h, n).copy()
    c = np.correlate(h, 1.0 - h, mode="full").astype(float)
    data = get_data(n)
    dx = 2.0 / n
    cur_max = float(c.max())
    best_c = cur_max * dx
    best_h = h.copy()
    active = top_active(c, cur_max, active_tol, max_active)
    no_imp = 0
    for it in range(passes):
        if time.time() > deadline:
            break
        order = list(range(n))
        if it % 2 == 1:
            rng.shuffle(order)
        improved = False
        for p in order:
            if time.time() > deadline:
                break
            dp = data[p]
            loc = dp[8][active]
            mask = loc >= 0
            if not np.any(mask):
                continue
            lin = loc[mask]
            act = active[mask]
            old = float(h[p])
            old_max = cur_max
            k = lin.size
            b = np.zeros(k)
            av = dp[0][lin]
            mm = av >= 0
            if np.any(mm):
                b[mm] += 1.0 - h[av[mm]]
            sv = dp[1][lin]
            mm = sv >= 0
            if np.any(mm):
                b[mm] -= h[sv[mm]]
            lin_mask = np.ones(k, dtype=bool)
            didx = -1
            dval = None
            if dp[7] >= 0:
                hit = np.where(lin == dp[7])[0]
                if hit.size:
                    didx = int(hit[0])
                    lin_mask[didx] = False
                    dval = c[n - 1]
            y = solve_one(c[act], b, lin_mask, didx, dval, old, old_max)
            if abs(y - old) > 1e-9:
                apply_update(c, h, p, y, dp)
                nm = float(np.max(c))
                if nm < old_max - 1e-12:
                    cur_max = nm
                    improved = True
                    cv = nm * dx
                    if cv < best_c - 1e-12:
                        best_c, best_h = cv, h.copy()
                else:
                    apply_update(c, h, p, old, dp)
        if (it + 1) % 10 == 0:
            h = project_feasible(h, n)
            c = np.correlate(h, 1.0 - h, mode="full").astype(float)
            cur_max = float(c.max())
            cv = cur_max * dx
            if cv < best_c - 1e-12:
                best_c, best_h = cv, h.copy()
                no_imp = 0
            elif not improved:
                no_imp += 1
        if not improved and (it + 1) % 40 == 0:
            active_tol = min(12.0, active_tol * 1.25)
        active = top_active(c, cur_max, active_tol, max_active)
    best_h = project_feasible(best_h, n)
    return best_h, compute_c5(best_h, n)


def alloc_exact(total, k, rng):
    if k <= 0:
        return np.empty(0, dtype=int)
    x = np.ones(k, dtype=int)
    rem = total - k
    if rem <= 0:
        x = np.zeros(k, dtype=int)
        if total > 0:
            idx = np.sort(rng.choice(k, size=min(k, total), replace=False))
            x[idx] = 1
        return x
    w = rng.random(k) + 1e-3
    x += np.floor(rem * w / w.sum()).astype(int)
    diff = int(total - x.sum())
    guard = 0
    while diff != 0 and guard < 50000:
        if diff > 0:
            x[rng.randint(k)] += 1
            diff -= 1
        else:
            cand = np.where(x > 1)[0]
            if cand.size == 0:
                break
            x[rng.randint(cand.size)] -= 1
            diff += 1
        guard += 1
    return x


def random_run_h(n, r, start, rng):
    n = int(n)
    h = np.zeros(n, dtype=float)
    len1 = alloc_exact(n // 2, r, rng)
    len0 = alloc_exact(n // 2, r, rng)
    pos = 0
    one = (start == 1)
    i1 = i0 = 0
    for _ in range(2 * r):
        if one:
            l = int(len1[i1])
            i1 += 1
        else:
            l = int(len0[i0])
            i0 += 1
        if l > 0 and pos < n:
            h[pos:pos + l] = 1.0 if one else 0.0
            pos += l
        if pos >= n:
            break
        one = not one
    return h


def ar_h(n, rho, gamma, gain, rng):
    n = int(n)
    s = 0.0
    a = np.empty(n, dtype=float)
    noise = rng.standard_normal(n)
    for i in range(n):
        s = rho * s + noise[i]
        a[i] = np.tanh(gamma * s)
    a -= a.mean()
    mx = np.max(np.abs(a))
    if mx > 1e-12:
        a = a / mx * gain
    a = np.clip(a, -1.0, 1.0)
    return project_feasible(0.5 + 0.5 * a, n)


def interp_h(h_old, n_old, n_new):
    if n_old == n_new:
        return project_feasible(h_old, n_new)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return project_feasible(np.interp(x_new, x_old, h_old), n_new)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 4.0
    rng = np.random.RandomState(seed)

    try:
        best_h = np.array(initial_h_values, dtype=float).copy()
        best_n = len(best_h)
    except Exception:
        best_n = 400
        best_h = np.full(best_n, 0.5)

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)

    if time.time() < deadline - 300:
        h1, c1 = fast_cd(best_h, best_n, 80, deadline, rng, 3.0, 90)
        if c1 < best_c:
            best_h, best_c = h1, c1
    if time.time() < deadline - 220:
        h2, c2 = slsqp_opt(best_h, best_n, (700, 3000, 15000, 80000, 300000), 220, deadline)
        if c2 < best_c:
            best_h, best_c = h2, c2
    if time.time() < deadline - 120:
        h3, c3 = fast_cd(best_h, best_n, 80, deadline, rng, 2.5, 110)
        if c3 < best_c:
            best_h, best_c = h3, c3

    ns = [best_n]
    if 300 <= best_n <= 1200:
        ns += [500, 700, 900]
    ns = sorted(set(x for x in ns if x >= 200 and x <= 1200 and x % 2 == 0))

    for nt in ns:
        if time.time() > deadline - 180:
            break
        for r in (3, 4, 5, 6, 8, 10, 14):
            if time.time() > deadline - 120:
                break
            for _ in range(2):
                if time.time() > deadline - 120:
                    break
                hr = random_run_h(nt, r, rng.randint(2), rng)
                cr = compute_c5(hr, nt)
                if cr < max(0.52, best_c + 0.055):
                    ht, ct = fast_cd(hr, nt, 35, deadline, rng, 3.0, 80)
                    if ct < best_c:
                        best_h, best_c, best_n = ht, ct, nt
                    if time.time() < deadline - 90 and ct < best_c + 0.002:
                        ht2, ct2 = slsqp_opt(ht, nt, (2000, 12000, 60000), 140, deadline)
                        if ct2 < best_c:
                            best_h, best_c, best_n = ht2, ct2, nt
                        ht3, ct3 = fast_cd(ht2, nt, 25, deadline, rng, 2.5, 90)
                        if ct3 < best_c:
                            best_h, best_c, best_n = ht3, ct3, nt

        if time.time() > deadline - 120:
            break
        for rho in (0.94, 0.97, 0.992):
            for gam in (1.5, 4.0):
                if time.time() > deadline - 120:
                    break
                hr = ar_h(nt, rho, gam, 0.92, rng)
                cr = compute_c5(hr, nt)
                if cr < max(0.50, best_c + 0.045):
                    ht, ct = fast_cd(hr, nt, 30, deadline, rng, 3.0, 80)
                    if ct < best_c:
                        best_h, best_c, best_n = ht, ct, nt
                    if time.time() < deadline - 80 and ct < best_c + 0.002:
                        ht2, ct2 = slsqp_opt(ht, nt, (3000, 20000, 100000), 120, deadline)
                        if ct2 < best_c:
                            best_h, best_c, best_n = ht2, ct2, nt

    if time.time() < deadline - 140:
        for nn in (500, 650, 800):
            if nn == best_n or time.time() > deadline - 100:
                continue
            hi = interp_h(best_h, best_n, nn)
            ht, ct = fast_cd(hi, nn, 60, deadline, rng, 2.5, 95)
            if ct < best_c:
                best_h, best_c, best_n = ht, ct, nn
            if time.time() < deadline - 70:
                ht2, ct2 = slsqp_opt(ht, nn, (3000, 25000, 120000), 150, deadline)
                if ct2 < best_c:
                    best_h, best_c, best_n = ht2, ct2, nn
                ht3, ct3 = fast_cd(ht2, nn, 30, deadline, rng, 2.0, 110)
                if ct3 < best_c:
                    best_h, best_c, best_n = ht3, ct3, nn

    if time.time() < deadline - 90:
        h4, c4 = slsqp_opt(best_h, best_n, (2000, 20000, 120000, 600000), 250, deadline)
        if c4 < best_c:
            best_h, best_c = h4, c4
    if time.time() < deadline - 25:
        h5, c5 = fast_cd(best_h, best_n, 120, deadline, rng, 2.0, 120)
        if c5 < best_c:
            best_h, best_c = h5, c5

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
```
