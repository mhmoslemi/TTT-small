            lr *= 0.5
        if not accepted:
            lr = max(lr * 0.8, 1e-10)
            if lr < 1e-9:
                break
    return best_h, best_c

def smooth_max_obj(x, n, p):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    return np.sum(c ** p) / n

def lbfgsb_optimize(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for p in [8, 16, 32, 64]:
        if time.time() > deadline - 10:
            break
        res = minimize(smooth_max_obj, best_h, args=(n, p), method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': 300, 'ftol': 1e-18, 'gtol': 1e-12})
        if res.x is not None:
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_h, best_c = h_new, c
    return best_h, best_c

def cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d):
    vals = c_aff + d * a
    if has_diag:
        vals[diag_idx] = c_aff[diag_idx] + d * (1.0 - h0 - d)
    result = float(np.max(vals))
    if unaff_max > result:
        result = unaff_max
    return result

def coord_descent(h, n, max_passes, deadline, gs_iters=50, threshold=1e-5):
    h = h.copy()
    best_c = compute_c5(h, n)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-13 or h0 >= 1.0 - 1e-13:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(valid_f):
                a[valid_f] += (1.0 - h[jf[valid_f]])
            if np.any(valid_b):
                a[valid_b] -= h[jb[valid_b]]
            diag_pos = np.where(ms == (n - 1))[0]
            has_diag = len(diag_pos) > 0
            diag_idx = int(diag_pos[0]) if has_diag else -1
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            d_lo = -h0
            d_hi = 1.0 - h0
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            f1 = cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
            f2 = cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-15:
                    break
                if f1 < f2:
                    d_hi, d2, f2 = d2, d1, f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    f1 = cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
                else:
                    d_lo, d1, f1 = d1, d2, f2
                    d2 = d_lo + gr * (d_hi - d_lo)
                    f2 = cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            d_opt = (d_lo + d_hi) / 2.0
            f_opt = cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d_opt)
            f_curr = cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, 0.0)
            if f_opt < f_curr - 1e-14:
                h[p] = h0 + d_opt
                improved = True
        if improved:
            h[np.where(h < threshold)[0]] = 0.0
            h[np.where(h > 1.0 - threshold)[0]] = 1.0
            h = project_feasible(h, n)
            c_new = compute_c5(h, n)
            if c_new < best_c:
                best_c = c_new
        else:
            break
    return h, min(best_c, compute_c5(h, n))

def binary_local_search(h, n, max_rounds, deadline, rng):
    h_b = (h >= 0.5).astype(float)
    target = n // 2
    n_ones = int(h_b.sum())
    if n_ones > target:
        ones_idx = np.where(h_b == 1)[0]
        order = np.argsort(h[ones_idx])
        for i in range(n_ones - target):
            h_b[ones_idx[order[i]]] = 0.0
    elif n_ones < target:
        zeros_idx = np.where(h_b == 0)[0]
        order = np.argsort(-h[zeros_idx])
        for i in range(target - n_ones):
            h_b[zeros_idx[order[i]]] = 1.0
    best_c = compute_c5(h_b, n)
    best_hb = h_b.copy()
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        c = np.correlate(h_b, 1.0 - h_b, mode='full')
        max_idx = int(c.argmax())
        k_star = max_idx - (n - 1)
        ones_idx = np.where(h_b == 1)[0]
        zeros_idx = np.where(h_b == 0)[0]
        improved = False
        n_try = min(len(ones_idx), 300)
        idx_sample = rng.choice(len(ones_idx), n_try, replace=False)
        for ii in idx_sample:
            i = int(ones_idx[ii])
            ik = i - k_star
            ipk = i + k_star
            delta_target = (1.0 - (h_b[ik] if 0 <= ik < n else 0)) + \
                           (h_b[ipk] if 0 <= ipk < n else 0)
            for jj in range(min(len(zeros_idx), 40)):
                j = int(zeros_idx[jj])
                jk = j - k_star
                jpk = j + k_star
                delta = 0.0
                if 0 <= ik < n:
                    delta -= (1.0 - h_b[ik])
                if 0 <= jk < n and jk != i:
                    delta += (1.0 - h_b[jk])
                if 0 <= ipk < n and ipk != i and ipk != j:
                    delta += h_b[ipk]
                if 0 <= jpk < n and jpk != i and jpk != j:
                    delta -= h_b[jpk]
                if delta < -1e-10 and c[max_idx] + delta < c.max() - 0.5:
                    h_b[i] = 0.0
                    h_b[j] = 1.0
                    nc = compute_c5(h_b, n)
                    if nc < best_c:
                        best_c = nc
                        best_hb = h_b.copy()
                        improved = True
                        break
                    h_b[i] = 1.0
                    h_b[j] = 0.0
        if not improved:
            break
    return best_hb, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n = 200
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    # Phase 1: Quick CD
    h1, c1 = coord_descent(best_h, n, 30, deadline, gs_iters=50)
    if c1 < best_c:
        best_h, best_c = h1, c1
    # Phase 2: Hinge projected gradient
    T0 = best_c
    h2, c2 = proj_grad_hinge(best_h, n, lambda t: T0 * (1.0 - 0.3 * t), 400, deadline)
    if c2 < best_c:
        best_h, best_c = h2, c2
    # Phase 3: L-BFGS-B with power mean
    h3, c3 = lbfgsb_optimize(best_h, n, deadline)
    if c3 < best_c:
        best_h, best_c = h3, c3
    # Phase 4: Binary local search
    if time.time() < deadline - 120:
        hb, cb = binary_local_search(best_h, n, 30, deadline, rng)
        if cb < best_c * 1.001:
            h_interp = project_feasible(np.interp(
                np.linspace(0, 2, n), np.linspace(0, 2, n), hb), n)
            h4, c4 = coord_descent(h_interp, n, 40, deadline, gs_iters=55)
            if c4 < best_c:
                best_h, best_c = h4, c4
    # Phase 5: Hinge again with lower target
    if time.time() < deadline - 100:
        T0 = best_c
        h5, c5 = proj_grad_hinge(best_h, n, lambda t: T0 * (1.0 - 0.5 * t), 300, deadline)
        if c5 < best_c:
            best_h, best_c = h5, c5
    # Phase 6: Multi-restart
    if time.time() < deadline - 200:
        n_cur = len(best_h)
        x = np.linspace(0, 2, n_cur)
        n_restarts = max(3, int((deadline - time.time() - 100) / 80))
        for r in range(n_restarts):
            if time.time() > deadline - 80:
                break
            if r % 3 == 0:
                hp = project_feasible(best_h + rng.normal(0, 0.02, n_cur), n_cur)
            elif r % 3 == 1:
                freq = 1.0 + 2.0 * r
                hp = project_feasible(best_h + 0.03 * np.sin(np.pi * freq * x), n_cur)
            else:
                mask = (rng.rand(n_cur) < 0.08).astype(float)
                hp = project_feasible(best_h + 0.4 * mask - 0.2, n_cur)
            ho, co = proj_grad_hinge(hp, n_cur, lambda t: best_c * (1.0 - 0.4 * t), 150, deadline)
            ho, co = coord_descent(ho, n_cur, 30, deadline, gs_iters=50)
            if co < best_c:
                best_h, best_c = ho, co
    # Phase 7: Final CD polish
    if time.time() < deadline - 10:
        h_f, c_f = coord_descent(best_h, len(best_h), 50, deadline, gs_iters=65)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
