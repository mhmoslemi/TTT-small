def lse_obj_fast(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    dx = 2.0 / n
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    val = (m + lse / alpha) * dx
    return val

def lse_grad_fast(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    dx = 2.0 / n
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    first = np.convolve(1.0 - x, w, mode='full')[:n]
    second = np.correlate(x, w, mode='full')[:n]
    return dx * (first - second)

def coord_descent(h, n, max_passes, deadline, rng, nbisect=45):

        improved_any = improved

def slsqp_phase(h, n, alphas, maxiter, deadline):

            if new_c < best_c:

        for p in order:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            x_new = solve_coord(c, h, p, n, nbisect)
            if abs(x_new - h0) > 1e-14:
                h[p] = x_new
                improved = True
        h = project_feasible(h, n)
        if not improved:
            break
    return h, compute_c5(h, n)

def slsqp_phase(h, n, alphas, maxiter, deadline):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    for alpha in alphas:
        if time.time() > deadline - 30:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac, 'args': (n,)}]
        try:
            res = minimize(lse_obj_grad, best_h, args=(n, alpha), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-15, 'maxls': 30})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def window_perturb(h, n, rng, n_iter, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    win_sizes = [10, 20, 30, 50, 80]
    for it in range(n_iter):
        if time.time() > deadline:
            break
        wl = win_sizes[rng.randint(len(win_sizes))]
        if wl >= n:
            wl = n // 3
        start = rng.randint(0, n - wl)
        h_try = h.copy()
        scale = rng.uniform(0.03, 0.15)
        h_try[start:start + wl] += rng.randn(wl) * scale
        h_try = project_feasible(h_try, n)
        c_try = compute_c5(h_try, n)
        if c_try < best_c:
            h = h_try
            best_c = c_try
    return h, best_c

def symmetrize(h, n):
    h_rev = h[::-1]
    h_sym = 0.5 * (h + 1.0 - h_rev)
    return project_feasible(h_sym, n)

def multiscale_opt(h, n, n_coarse, deadline):
    x_fine = np.linspace(0, 2, n)
    x_coarse = np.linspace(0, 2, n_coarse)
    h_c = project_feasible(np.interp(x_coarse, x_fine, h, clip=True), n_coarse)
    h_c, c_c = slsqp_phase(h_c, n_coarse, [500, 5000, 20000], 100, deadline)
    h_c, c_c = coord_descent(h_c, n_coarse, 15, deadline, np.random.RandomState(0), 40)
    h_up = project_feasible(np.interp(x_fine, x_coarse, h_c, clip=True), n)
    return h_up, c_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 700
        x = np.linspace(0, 2, n)
        h_init = np.clip(0.5 + 0.25 * np.sin(np.pi * x) + 0.1 * np.sin(3 * np.pi * x), 0, 1)
    best_h = project_feasible(h_init, n)
    best_c = compute_c5(best_h, n)
    # Phase 1: Initial CD
    h1, c1 = coord_descent(best_h, n, 8, deadline, rng, 40)
    if c1 < best_c:
        best_h, best_c = h1, c1
    # Phase 2: SLSQP ramp
    h2, c2 = slsqp_phase(best_h, n, [200, 1000, 5000, 20000], 200, deadline)
    if c2 < best_c:
        best_h, best_c = h2, c2
    # Phase 3: CD refinement
    h3, c3 = coord_descent(best_h, n, 25, deadline, rng, 50)
    if c3 < best_c:
        best_h, best_c = h3, c3
    # Phase 4: Window perturb + refine
    h4, c4 = window_perturb(best_h, n, rng, 300, deadline)
    if c4 < best_c:
        best_h, best_c = h4, c4
    h4b, c4b = coord_descent(best_h, n, 20, deadline, rng, 50)
    if c4b < best_c:
        best_h, best_c = h4b, c4b
    # Phase 5: High-alpha SLSQP
    if time.time() < deadline - 80:
        h5, c5 = slsqp_phase(best_h, n, [50000, 100000, 300000], 200, deadline)
        if c5 < best_c:
            best_h, best_c = h5, c5
    # Phase 6: Multi-scale
    if time.time() < deadline - 80:
        h6, c6 = multiscale_opt(best_h, n, n // 2, deadline)
        if c6 < best_c:
            best_h = h6
            best_c = compute_c5(best_h, n)
            h6b, c6b = coord_descent(best_h, n, 20, deadline, rng, 45)
            if c6b < best_c:
                best_h, best_c = h6b, c6b
            h6c, c6c = slsqp_phase(best_h, n, [5000, 50000], 150, deadline)
            if c6c < best_c:
                best_h, best_c = h6c, c6c
    # Phase 7: Symmetrize + reoptimize
    if time.time() < deadline - 60:
        h_sym = symmetrize(best_h, n)
        c_sym = compute_c5(h_sym, n)
        if c_sym < best_c:
            h_sym, c_sym = coord_descent(h_sym, n, 15, deadline, rng, 45)
            if c_sym < best_c:
                best_h, best_c = h_sym, c_sym
    # Phase 8: More perturbations with smaller scale
    if time.time() < deadline - 40:
        h8, c8 = window_perturb(best_h, n, rng, 200, deadline)
        if c8 < best_c:
            best_h, best_c = h8, c4
            h8b, c8b = coord_descent(best_h, n, 15, deadline, rng, 50)
            if c8b < best_c:
                best_h, best_c = h8b, c8b
            if time.time() < deadline - 30:
                h8c, c8c = slsqp_phase(best_h, n, [100000], 100, deadline)
                if c8c < best_c:
                    best_h, best_c = h8c, c8c
    # Phase 9: Final polish
    if time.time() < deadline - 10:
        max_p = max(5, int((deadline - time.time() - 3) / 1.5))
        h9, c9 = coord_descent(best_h, n, max_p, deadline, rng, 55)
        if c9 < best_c:
            best_h, best_c = h9, c9
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
