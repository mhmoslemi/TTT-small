import numpy as np
import math
import time
from scipy.optimize import minimize

_SLSQP_N = 64

def compute_c5(h, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * (2.0 / n))

def project_feasible(h, n):
    target = 0.5 * n
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0.0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        idx = np.flatnonzero(h < 1.0 - 1e-12) if diff > 0 else np.flatnonzero(h > 1e-12)
        if idx.size:
            h[idx[0]] = min(1.0, max(0.0, h[idx[0]] + diff))
    return h

def default_h(n):
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = 0.5 + 0.45 * np.cos(math.pi * x) + 0.15 * np.cos(2.0 * math.pi * x)
    return project_feasible(h, n)

def make_symmetric(h):
    n = len(h)
    return project_feasible(0.5 * (h + h[::-1]), n)

def resample_h(h, new_n):
    old_n = len(h)
    xo = (np.arange(old_n) + 0.5) * (2.0 / old_n)
    xn = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    return project_feasible(np.interp(xn, xo, h), new_n)

def cd_step(c, h, p, n):
    center = n - 1
    x0 = float(h[p])
    mn = min(center - p, p)
    mx = max(2 * n - 2 - p, center + p)
    ms = np.arange(mn, mx + 1)
    q = ms - center
    j = p + q
    i = p - q
    f = (j >= 0) & (j < n)
    s = (i >= 0) & (i < n)
    d = f & s
    L = len(ms)
    co = np.zeros(L)
    b = c[ms].copy()
    fm = f & ~d
    if np.any(fm):
        a = 1.0 - h[j[fm]]
        co[fm] = a
        b[fm] -= a * x0
    sm = s & ~d
    if np.any(sm):
        a = -h[i[sm]]
        co[sm] = a
        b[sm] -= a * x0
    di = np.nonzero(d)[0]
    has_d = di.size > 0
    if has_d:
        bd = float(c[ms[di[0]]] - x0 + x0 * x0)
        co[di[0]] = 0.0
    else:
        bd = 0.0
    nd = ~d
    pos = nd & (co > 1e-14)
    neg = nd & (co < -1e-14)
    zmask = nd & ~pos & ~neg
    pb = b[pos] if np.any(pos) else np.empty(0)
    pc = co[pos] if np.any(pos) else np.empty(0)
    nb = b[neg] if np.any(neg) else np.empty(0)

    unaff = 0.0
    if mn > 0:

    bnd = b[nd]
    L0 = unaff
    L1 = unaff
    if bnd.size:

        if ok and xl > xh + tol:
            ok = False

                ok = False

                    T = 0.0
                rr1 = 0.5 * (1.0 - math.sqrt(max(0.0, 1.0 - 4.0 * T)))
                rr2 = 0.5 * (1.0 + math.sqrt(max(0.0, 1.0 - 4.0 * T)))
                if not (xl <= rr1 + tol or xh >= rr2 - tol):
                    ok = False
        if ok:
            L_hi = mid
        else:
            L_lo = mid
    Lm = L_hi
    ok = zmax <= Lm + tol
    xl = 0.0
    xh = 1.0
    if ok and pb.size:
        xh = min(xh, float(np.min((Lm - pb) / pc)))
    if ok and nb.size:

    r2 = 0.5

            ok = False

            r2 = 0.5 * (1.0 + math.sqrt(max(0.0, 1.0 - 4.0 * T)))

                ok = False

        return False
    comps = []
    if has_d and T < 0.25 - 1e-13:
        lo = max(xl, 0.0)
        hi = min(xh, r1)
        if lo <= hi:
            comps.append((lo, hi))
        lo = max(xl, r2)
        hi = min(xh, 1.0)

        hi = min(xh, 1.0)
        if lo <= hi:
            comps.append((lo, hi))

        cand = float(np.clip(x0, lo, hi))

            bx = cand

        return False
    h[p] = bx
    idx = ms[nd]
    if idx.size:
        c[idx] = b[nd] + co[nd] * bx
    if has_d:
        c[center] = bd + bx - bx * bx
    return True

def coordinate_descent(h, n, max_passes, deadline):
    h = project_feasible(h, n)
    c = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    best_c = float(np.max(c) * dx)
    best_h = h.copy()
    target = 0.5 * n
    no_imp = 0
    for it in range(max_passes):
        if time.time() > deadline:
            break
        changed = 0
        order = range(n - 1, -1, -1) if (it & 1) else range(n)
        for p in order:
            if cd_step(c, h, p, n):
                changed += 1
        if abs(h.sum() - target) > 1e-10:
            hn = project_feasible(h, n)
            if np.max(np.abs(hn - h)) > 1e-15:
                h = hn
                c = np.correlate(h, 1.0 - h, mode='full')
                changed += 1
        cur = float(np.max(c) * dx)
        if cur < best_c:
            best_c = cur
            best_h = h.copy()
            no_imp = 0
        else:
            no_imp += 1
        if changed == 0 and no_imp >= 2:
            break
        if no_imp >= 28:
            break
    best_h = project_feasible(best_h, n)
    return best_h, compute_c5(best_h, n)

def smooth_objective(z, n, alpha):
    z = np.clip(z, 0.0, 1.0)
    c = np.correlate(z, 1.0 - z, mode='full')
    m = float(c.max())
    e = np.exp(alpha * (c - m))
    val = (m + math.log(float(e.sum())) / alpha) * (2.0 / n)
    return float(100.0 * val)

def smooth_gradient(z, n, alpha):
    z = np.clip(z, 0.0, 1.0)
    c = np.correlate(z, 1.0 - z, mode='full')
    m = float(c.max())
    e = np.exp(alpha * (c - m))
    s = float(e.sum())
    if s <= 0.0 or not math.isfinite(s):
        p = np.full(2 * n - 1, 1.0 / (2 * n - 1))
    else:
        p = e / s
    ones = 1.0 - z
    zrev = z[::-1]
    center = n - 1
    g = np.empty(n)
    for k in range(n):
        g[k] = np.dot(p[center - k:2 * n - 1 - k], ones) - np.dot(p[k:n + k], zrev)
    return g * (200.0 / n)

def sum_constraint(z):
    return float(np.sum(z) - 0.5 * _SLSQP_N)

def sum_constraint_jac(z):
    return np.ones(_SLSQP_N, dtype=float)

def slsqp_polish(h, n, alpha, maxiter, deadline):
    global _SLSQP_N
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h_res = best_h.copy()
    if time.time() > deadline:
        return best_h, best_c, h_res
    _SLSQP_N = int(n)
    maxiter = int(min(maxiter, max(8, maxiter * (700.0 / max(n, 1)))))
    try:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac}]
        res = minimize(smooth_objective, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, jac=smooth_gradient, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-10})
        x = np.asarray(res.x, dtype=float)
        if np.all(np.isfinite(x)):
            h_res = project_feasible(x, n)
            c_res = compute_c5(h_res, n)
            if c_res < best_c:
                best_h, best_c = h_res, c_res
    except Exception:
        pass
    return best_h, best_c, h_res

def cd_try(hc, bh, bc, passes, deadline):
    if time.time() > deadline:
        return bh, bc
    h, c = coordinate_descent(hc, len(hc), passes, deadline)
    if c < bc:
        return h, c
    return bh, bc

def slp_try(hc, bh, bc, alpha, iters, deadline):

    h, c, hr = slsqp_polish(hc, len(hc), alpha, iters, deadline)
    if c < bc:

    return bh, bc, hr
