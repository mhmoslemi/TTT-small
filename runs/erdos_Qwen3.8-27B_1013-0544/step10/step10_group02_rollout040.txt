    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)

        disc = 1.0 - 4.0 * Cq
        if disc >= 0:

            r2 = (1.0 + np.sqrt(disc)) / 2.0

            if xc_lo < r1 and xc_hi > r2:

                    x_hi = min(x_hi, r1)
                else:
                    x_lo = max(x_lo, r2)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def update_corr(c, h, p, old_val, new_val, n):
    delta = new_val - old_val
    if abs(delta) < 1e-14:
        return c
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    diag_i = np.where(ms == (n - 1))[0]
    c[ms] += delta * a
    if len(diag_i) > 0:
        c[ms[diag_i[0]]] -= delta * delta
    return c

def exact_cd(h, n, max_passes, deadline, rng=None):

    for it in range(max_passes):

            break
        improved = False
        order = list(range(n))
        if rng is not None and it % 3 == 1:
            rng.shuffle(order)

                h[p] = xn
                improved = True
        if not improved:

def project_feasible(h, n):

def smooth_obj_grad(x, n, alpha, penalty):
    dx = 2.0 / n
    h = np.clip(x, 0.0, 1.0)
    c = np.correlate(h, 1.0 - h, mode='full')
    shifted = -alpha * c
    mx = shifted.max()
    ev = np.exp(shifted - mx)
    w = ev / ev.sum()
    obj = (-mx + np.log(ev.sum())) / alpha * dx + penalty * (h.sum() - n / 2.0) ** 2
    grad = np.zeros(n)
    m_all = np.arange(2 * n - 1)
    d_all = (n - 1) - m_all
    for p in range(n):
        m_lo = max(0, n - 1 - p)
        m_hi = min(2 * n - 2, 2 * n - 2 - p)
        if m_lo > m_hi:
            continue
        ms = m_all[m_lo:m_hi + 1]
        ds = d_all[m_lo:m_hi + 1]
        jf = p + ds
        jb = p - ds
        contrib = np.zeros(len(ms))
        vf = (jf >= 0) & (jf < n)
        vb = (jb >= 0) & (jb < n)
        if np.any(vf):
            contrib[vf] += (1.0 - h[jf[vf]])
        if np.any(vb):
            contrib[vb] -= h[jb[vb]]
        grad[p] = np.dot(w[m_lo:m_hi + 1], contrib)
    grad *= dx
    grad += 2.0 * penalty * (h.sum() - n / 2.0)
    return obj, grad

def lbfgs_opt(h, n, alphas, maxiter, deadline, penalty=100.0):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline - 30:
            break
        try:
            res = minimize(smooth_obj_grad, best_h, args=(n, alpha, penalty),
                          method='L-BFGS-B', jac=True,
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c - 1e-12:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def zero_mean_perturb(h, n, rng, scale):
    noise = rng.normal(0, scale, n)
    noise -= noise.mean()
    return project_feasible(h + noise, n)

def interp_h(h_old, n_old, n_new):

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = np.full(700, 0.5)
        n = 700
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)

    # Phase 1: Strong CD with mixed orderings
    h1, c1 = exact_cd(h_best, n, 200, deadline, rng)
    if c1 < best_c:
        h_best, best_c = h1, c1

    # Phase 2: L-BFGS with analytical gradient
    h2, c2 = lbfgs_opt(h_best, n, [3000, 10000, 40000], 300, deadline, penalty=50.0)
    if c2 < best_c - 1e-12:
        h_best, best_c = h2, c2

    # Phase 3: CD after L-BFGS
    h3, c3 = exact_cd(h_best, n, 150, deadline, rng)
    if c3 < best_c - 1e-12:
        h_best, best_c = h3, c3

    # Phase 4: Multi-resolution
    n_list = [600, 550, 650, 750, 500, 800]
    for n_try in n_list:
        if time.time() > deadline - 200:
            break
        h_int = interp_h(h_best, len(h_best), n_try)
        h_o, c_o = exact_cd(h_int, n_try, 150, deadline, rng)
        if c_o < best_c - 1e-9:
            h_best, best_c = h_o, c_o
        if time.time() < deadline - 80:
            h_o2, c_o2 = lbfgs_opt(h_o, n_try, [10000, 30000], 200, deadline, penalty=50.0)
            if c_o2 < best_c - 1e-9:
                h_best, best_c = h_o2, c_o2
            h_o3, c_o3 = exact_cd(h_o2, n_try, 100, deadline, rng)
            if c_o3 < best_c - 1e-9:
                h_best, best_c = h_o3, c_o3

    # Phase 5: Annealed restarts (zero-mean perturbations preserve sum)
    n = len(h_best)
    for r in range(30):
        if time.time() > deadline - 60:
            break
        scale = 0.05 * (1.0 - r / 30.0) + 0.005
        hp = zero_mean_perturb(h_best, n, rng, scale)
        h_o, c_o = exact_cd(hp, n, 80, deadline, rng)
        if c_o < best_c - 1e-9:
            h_best, best_c = h_o, c_o
        if time.time() > deadline - 40:
            break
        if c_o < best_c:
            h_o2, c_o2 = lbfgs_opt(h_o, n, [20000], 100, deadline, penalty=50.0)
            if c_o2 < best_c - 1e-9:
                h_best, best_c = h_o2, c_o2

    # Phase 6: Final polish
    n = len(h_best)
    h_f, c_f = exact_cd(h_best, n, 300, deadline, rng)
    if c_f < best_c - 1e-12:
        h_best, best_c = h_f, c_f
    if time.time() < deadline - 20:
        h_g, c_g = lbfgs_opt(h_best, n, [50000], 200, deadline, penalty=200.0)
        if c_g < best_c - 1e-12:
            h_best, best_c = h_g, c_g

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
