def min_convex_plinear(c, a, lo, hi):
    """Find argmin_{d in [lo,hi]} max_i (c[i] + d*a[i])"""
    m = len(c)
    
    # Sort by slope
    order = np.argsort(a)
    a_s = a[order]
    c_s = c[order]
    
    # Remove duplicates (same slope, keep max c)
    # Build hull
    hull_a = []
    hull_c = []
    i = 0
    while i < m:
        ai = a_s[i]
        ci = c_s[i]
        # Find max c for this slope
        while i + 1 < m and a_s[i+1] == ai:
            if c_s[i+1] > ci:
                ci = c_s[i+1]
            i += 1
        i += 1
        # Add to hull with convexity check
        while len(hull_a) >= 2:
            # Check if new line makes the last hull line obsolete
            a_prev = hull_a[-2]
            c_prev = hull_c[-2]
            a_last = hull_a[-1]
            c_last = hull_c[-1]
            # Intersection of prev and last: d_pl = (c_prev - c_last) / (a_last - a_prev)
            # Intersection of last and new: d_ln = (c_last - ci) / (ai - a_last)
            # If d_pl >= d_ln, then last line is never the max -> remove it
            denom1 = a_last - a_prev
            denom2 = ai - a_last
            if denom1 == 0:
                hull_a.pop()
                hull_c.pop()
                continue
            if denom2 == 0:
                # Same slope as last, shouldn't happen after dedup
                break
            d_pl = (c_prev - c_last) / denom1
            d_ln = (c_last - ci) / denom2
            if d_pl >= d_ln:
                hull_a.pop()
                hull_c.pop()
            else:
                break
        hull_a.append(ai)
        hull_c.append(ci)
    
    # Now hull_a, hull_c define the upper envelope
    # Find min over [lo, hi]
    best_d = lo
    best_f = max(c + lo * a)
    
    # Check hi
    f_hi = max(c + hi * a)
    if f_hi < best_f:
        best_f = f_hi
        best_d = hi
    
    # Check interior breakpoints
    for j in range(len(hull_a) - 1):
        d_cross = (hull_c[j] - hull_c[j+1]) / (hull_a[j+1] - hull_a[j])
        if lo <= d_cross <= hi:
            f_cross = hull_c[j] + d_cross * hull_a[j]
            if f_cross < best_f:
                best_f = f_cross
                best_d = d_cross

def project_feasible(h, n):
    target = n / 2.0
