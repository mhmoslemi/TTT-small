def compute_c5(h, n):

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:

def solve_1d(c, h, p, n):

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return m + lse / alpha

def slsqp_opt(h, n, alphas, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            bounds = [(0.0, 1.0)] * n
            cons = [{'type': 'eq',
                     'fun': lambda x: x.sum() - n / 2.0,
                     'jac': lambda x: np.ones(n)}]
            res = minimize(smooth_obj, best_h, args=(n, alpha),
                           method='SLSQP', bounds=bounds,
                           constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def perturb_block(h, n, rng):
    hp = h.copy()
    size = rng.randint(4, max(5, n // 8))
    start = rng.randint(0, n - size)
    hp[start:start + size] = 1.0 - hp[start:start + size]
    return project_feasible(np.clip(hp, 0, 1), n)

def perturb_swap(h, n, rng):
    hp = h.copy()
    size = rng.randint(4, max(5, n // 8))
    s1 = rng.randint(0, n - size)
    s2 = rng.randint(0, n - size)
    if abs(s1 - s2) < size:
        return hp
    tmp = hp[s1:s1 + size].copy()
    hp[s1:s1 + size] = hp[s2:s2 + size]
    hp[s2:s2 + size] = tmp
    return project_feasible(hp, n)

def perturb_noise(h, n, rng, scale):
    hp = h + rng.normal(0, scale, n)
    return project_feasible(np.clip(hp, 0, 1), n)

def binary_push(h, n, strength):
    hb = h.copy()
    hb[hb < 0.5] = np.maximum(0.0, hb[hb < 0.5] - strength)
    hb[hb >= 0.5] = np.minimum(1.0, hb[hb >= 0.5] + strength)
    return project_feasible(hb, n)

        h_o, c_o = exact_cd(hp, n, 250, deadline - 150, recompute_every=10)
        if c_o < best_c:
            h_best, best_c = h_o, c_o

    # Phase 5: Binary push refinement
    for strength in [0.02, 0.06, 0.12, 0.2]:
        if time.time() > deadline - 100:
            break
        hb = binary_push(h_best, n, strength)
        h_o, c_o = exact_cd(hb, n, 200, deadline - 100, recompute_every=10)
        if c_o < best_c:
            h_best, best_c = h_o, c_o

    # Phase 6: Final intensive CD
    if time.time() < deadline - 3:
        h_f, c_f = exact_cd(h_best, len(h_best), 400, deadline, recompute_every=10)
        if c_f < best_c:
            h_best, best_c = h_f, c_f

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
