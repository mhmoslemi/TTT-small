import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(8000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-16:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-16:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def update_corr(c, h, p, old_val, new_val, n):
    delta = new_val - old_val
    if abs(delta) < 1e-15:
        return c
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    diag_i = np.where(ms == (n - 1))[0]
    c[ms] += delta * a
    if len(diag_i) > 0:
        c[ms[diag_i[0]]] -= delta * delta
    return c

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    s = shifted - shifted.max()
    lse = shifted.max() + np.log(np.sum(np.exp(s)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_opt(h, n, alphas, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n
        def cf(x):
            return x.sum() - n / 2.0
        def cj(x):
            return np.ones(n)
        cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c - 1e-12:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)
