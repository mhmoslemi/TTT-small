            a = np.zeros(len(ms))
            if np.any(vf):
                a[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                a[vb] -= h[jb[vb]]
            c[ms] += delta * a
            diag_i = np.where(ms == (n - 1))[0]
            if len(diag_i) > 0:
                c[ms[diag_i[0]]] -= delta * delta
            h[p] = xn
            improved = True
    return h, c, improved

def cd_refine(h, n, max_passes, deadline, rng=None, max_bisect=35):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    prev_c = compute_c5(h, n)
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        order = rng.permutation(n) if (rng is not None) else np.arange(n)
        h, c, improved = cd_pass(h, n, c, order, deadline, max_bisect)
        if not improved:
            break
        h = project_feasible(h, n)
        c = np.correlate(h, 1.0 - h, mode='full')
        cur_c = compute_c5(h, n)
        if prev_c - cur_c < 1e-12:
            break
        prev_c = cur_c
    return h, compute_c5(h, n)

def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old, period=2.0)
    return project_feasible(h_new, n_new)

def add_penalty_constraint(h, n, alpha, mu, maxiter, deadline):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(lse_obj, h, args=(n, alpha, mu), method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14,
                                'maxls': 20, 'maxfun': maxiter * n + 100})
        return project_feasible(res.x, n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        n = 700
        h_best = np.full(n, 0.5)

    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)

    # Phase 1: L-BFGS-B with increasing sharpness
    for alpha in [500, 2000, 8000, 30000, 100000]:
        if time.time() > deadline - 200:
            break
        mu = 50.0
        h_try = add_penalty_constraint(h_best, n, alpha, mu, 200, deadline)
        c_try = compute_c5(h_try, n)
        if c_try < best_c:
            h_best, best_c = h_try, c_try

    # Phase 2: Coordinate descent (main workhorse)
    h_best, best_c = cd_refine(h_best, n, 80, deadline, max_bisect=40)

    # Phase 3: Random-order CD to escape local optima
    if time.time() < deadline - 150:
        h_best, best_c = cd_refine(h_best, n, 40, deadline, rng=rng, max_bisect=30)

    # Phase 4: Multi-start with diverse perturbations
    if time.time() < deadline - 100:
        n = len(h_best)
        for trial in range(12):
            if time.time() > deadline - 60:
                break
            # Different perturbation strategies
            if trial % 3 == 0:
                # Small noise
                hp = h_best + rng.normal(0, 0.02, n)
            elif trial % 3 == 1:
                # Permutation of values
                perm = rng.permutation(n)
                hp = h_best[perm]
            else:
                # Block swap
                hp = h_best.copy()
                a, b = sorted(rng.choice(n, 2, replace=False))
                length = min(int(n * 0.1), b - a)
                if length > 5:
                    hp[a:a+length], hp[b-length+1:b+1] = hp[b-length+1:b+1].copy(), hp[a:a+length].copy()
            hp = project_feasible(hp, n)
            # Quick L-BFGS-B + CD
            hp = add_penalty_constraint(hp, n, 10000, 50.0, 100, deadline)
            hp, c_try = cd_refine(hp, n, 15, deadline, max_bisect=30)
            if c_try < best_c:
                h_best, best_c = hp, c_try

    # Phase 5: Multi-n exploration
    if time.time() < deadline - 120:
        n_best = len(h_best)
        for n_try in [550, 750, 900, 650]:
            if time.time() > deadline - 60:
                break
            h_int = interpolate_h(h_best, n_best, n_try)
            h_int = add_penalty_constraint(h_int, n_try, 5000, 50.0, 150, deadline)
            h_int, c_try = cd_refine(h_int, n_try, 30, deadline, max_bisect=35)
            if c_try < best_c - 1e-7:
                h_best, best_c, n_best = h_int, c_try, n_try

    # Phase 6: Final intensive CD
    n = len(h_best)
    if time.time() < deadline - 10:
        h_best, best_c = cd_refine(h_best, n, 60, deadline, max_bisect=50)

    # Phase 7: Final L-BFGS-B polish
    if time.time() < deadline - 5:
        h_try = add_penalty_constraint(h_best, n, 200000, 100.0, 100, deadline)
        c_try = compute_c5(h_try, n)
        if c_try < best_c:
            h_best, best_c = h_try, c_try
        h_best, best_c = cd_refine(h_best, n, 20, deadline, max_bisect=50)

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
