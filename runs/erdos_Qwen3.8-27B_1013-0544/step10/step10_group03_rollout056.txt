∂c[m]/∂h[p]:
- If p is in the "first" position (i=p): contribution = (1-h[p+m-(n-1)]) [if p+m-(n-1) is valid and ≠ p]
- If p is in the "second" position (i+m-(n-1)=p, i.e., i=p-m+(n-1)): contribution = -h[p-m+(n-1)] [if i is valid and ≠ p]
- If m=n-1 (so both positions are p): contribution = (1-h[p]) - h[p] = 1-2h[p]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def solve_1d(c, h, p, n):
    h0 = h[p]
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms]
    diag_off = p
    c_diag = c_base[diag_off]
    a_b = a.copy()
    a_b[diag_off] = 0.0
    unaff = 0.0
    if m_min > 0:
        unaff = float(c[:m_min].max())
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(c[m_max + 1:].max()))
    cur_max = float(c.max())
    neg_m = a_b < -1e-15
    pos_m = a_b > 1e-15
    zero_m = ~neg_m & ~pos_m
    L_lo = max(unaff, -1e-10)
    L_hi = cur_max
    for _ in range(45):
        L = (L_lo + L_hi) * 0.5
        x_lo = 0.0
        x_hi = 1.0
        ok = True
        if np.any(zero_m) and np.any(c_base[zero_m] > L):
            ok = False
        if ok and np.any(neg_m):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_m]) / a_b[neg_m])))
        if ok and np.any(pos_m):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_m]) / a_b[pos_m])))
        if x_lo > x_hi + 1e-12:
            ok = False
        if ok:
            Cq = L - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc < 0:
                ok = False
            else:
                r1 = (1.0 - np.sqrt(disc)) * 0.5
                r2 = (1.0 + np.sqrt(disc)) * 0.5
                if max(x_lo, 0.0) >= r1 or min(x_hi, 1.0) <= r2:
                    ok = False
        if ok:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo = 0.0
    x_hi = 1.0
    if np.any(neg_m):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_m]) / a_b[neg_m])))
    if np.any(pos_m):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_m]) / a_b[pos_m])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    Cq = L - c_diag + h0 - h0 * h0
    disc = 1.0 - 4.0 * Cq
    if disc >= 0:
        r1 = (1.0 - np.sqrt(disc)) * 0.5
        r2 = (1.0 + np.sqrt(disc)) * 0.5
        if h0 - r1 <= r2 - h0:
            x_hi = min(x_hi, r1)
        else:
            x_lo = max(x_lo, r2)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) * 0.5, 0.0, 1.0))

def exact_cd(h, n, max_passes, deadline):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        max_change = 0.0
        for p in range(n):
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            xn = solve_1d(c, h, p, n)
            delta = xn - h0
            if abs(delta) > 1e-12:
                m_min = n - 1 - p
                m_max = 2 * n - 2 - p
                ms = np.arange(m_min, m_max + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                c[ms] += delta * a
                c[n - 1] -= delta * delta
                h[p] = xn
                max_change = max(max_change, abs(delta))
        if max_change < 1e-13:
            break
        if abs(h.sum() - n / 2.0) > 1e-10:
            h = project_feasible(h, n)
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def slsqp_opt(h, n, alphas, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline:
            break
        def cons_fun(x):
            return x.sum() - n / 2.0
        def cons_jac(x):
            return np.ones(n)
        cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-15})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def smooth_perturb(h, n, rng, scale):
    noise = rng.normal(0, 1, n)
    fft_n = np.fft.rfft(noise)
    cutoff = max(2, n // 8)
    fft_n[cutoff:] = 0
    noise = np.fft.irfft(fft_n, n=n)
    noise /= (np.std(noise) + 1e-15)
    return project_feasible(h + scale * noise, n)
