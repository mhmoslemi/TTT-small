import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, n, alpha):
    dx = 2.0 / n
    N = 2 * n
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(1.0 - h, N)
    corr = np.fft.irfft(H * np.conj(G), N)
    full = np.empty(2 * n - 1)
    full[n - 1:] = corr[:n]
    full[:n - 1] = corr[n + 1:2 * n]
    c = full * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    exp_s = np.exp(shifted - mx)
    w = exp_s / exp_s.sum()
    f_val = m + (mx + np.log(exp_s.sum())) / alpha
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-13:
            continue
        s = k - (n - 1)
        if s >= 0:
            grad[s:] += wk * (1.0 - h[:n - s])
            grad[:n - s] -= wk * h[s:]
        else:
            ss = -s
            grad[:n - ss] += wk * (1.0 - h[ss:])
            grad[ss:] -= wk * h[:n - ss]
    grad *= dx
    return f_val, grad

def smooth_grad_phase(h, n, deadline, alpha_start=5, alpha_end=500, n_alphas=15, iters_per=60):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alphas = np.geomspace(alpha_start, alpha_end, n_alphas)
    for alpha in alphas:
        if time.time() > deadline - 3:
            break
        h_cur = best_h.copy()
        lr = 0.015
        mom = np.zeros(n)
        for it in range(iters_per):
            if time.time() > deadline - 3:
                break
            _, grad = smooth_max_and_grad(h_cur, n, alpha)
            mom = 0.8 * mom + 0.2 * grad
            h_new = project_feasible(h_cur - lr * mom, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_h = h_new.copy()
                best_c = c_new
            h_cur = h_new
            lr *= 0.99
    return best_h, best_c

def _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d):
    vals = c_aff + d * a
    if has_diag:
        vals[diag_idx] = c_aff[diag_idx] + d * (1.0 - h0 - d)
    result = float(np.max(vals))
    if unaff_max > result:
        result = unaff_max
    return result

def coord_descent(h, n, max_passes, deadline, gs_iters=50):
    h = h.copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-13 or h0 >= 1.0 - 1e-13:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(valid_f):
                a[valid_f] += (1.0 - h[jf[valid_f]])
            if np.any(valid_b):
                a[valid_b] -= h[jb[valid_b]]
            diag_pos = np.where(ms == (n - 1))[0]
            has_diag = len(diag_pos) > 0
            diag_idx = int(diag_pos[0]) if has_diag else -1
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            d_lo = -h0
            d_hi = 1.0 - h0
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
            f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-15:
                    break
                if f1 < f2:
                    d_hi = d2
                    d2 = d1
                    f2 = f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
                else:
                    d_lo = d1
                    d1 = d2
                    f1 = f2
                    d2 = d_lo + gr * (d_hi - d_lo)
                    f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            d_opt = (d_lo + d_hi) / 2.0
            f_opt = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d_opt)
            f_curr = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, 0.0)
            if f_opt < f_curr - 1e-14:
                h[p] = h0 + d_opt
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def penalty_optimize(h, n, deadline, max_iters=200):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    pen = 10.0
    lr = 0.01
    for it in range(max_iters):
        if time.time() > deadline:
            break
        _, grad_sm = smooth_max_and_grad(h, n, 200.0)
        pen_grad = 2.0 * pen * (h.sum() - n / 2.0) / n
        grad = grad_sm + pen_grad
        h_new = np.clip(h - lr * grad, 0.0, 1.0)
        c_new = compute_c5(h_new, n)
        if c_new < best_c and abs(h_new.sum() - n / 2.0) < n * 0.01:
            best_h = h_new.copy()
            best_c = c_new
        h = h_new
        lr *= 0.995
        pen *= 1.05
    return best_h, best_c
