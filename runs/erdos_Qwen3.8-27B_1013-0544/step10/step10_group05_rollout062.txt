import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def solve_coord(c, h, p, n):
    h0 = h[p]
    m_min = max(0, n - 1 - p)
    m_max = min(2 * n - 2, 2 * n - 2 - p)
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms]
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    unaff = 0.0
    if m_min > 0:
        unaff = float(np.max(c[:m_min]))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    cur_max = float(np.max(c))
    neg_mask = a < -1e-14
    pos_mask = a > 1e-14
    zero_mask = np.abs(a) <= 1e-14
    L_lo, L_hi = unaff, cur_max
    for _ in range(45):
        if L_hi - L_lo < 1e-15:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        feasible = True
        if np.any(zero_mask) and c_base[zero_mask].max() > L + 1e-14:
            feasible = False
        if feasible and np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if feasible and np.any(pos_mask):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        if x_lo > x_hi + 1e-12:
            feasible = False
        if feasible and has_diag:
            c_diag = float(c_base[diag_idx[0]])
            a_d = a[diag_idx[0]]
            disc = a_d * a_d + 4.0 * (c_diag - L)
            if disc >= 0:
                sq = np.sqrt(disc)
                r1 = (a_d - sq) / 2.0
                r2 = (a_d + sq) / 2.0
                if x_lo > h0 + r1 and x_hi < h0 + r2:
                    feasible = False
        if feasible:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        c_diag = float(c_base[diag_idx[0]])
        a_d = a[diag_idx[0]]
        disc = a_d * a_d + 4.0 * (c_diag - L)
        if disc >= 0:
            sq = np.sqrt(disc)
            r1 = (a_d - sq) / 2.0
            r2 = (a_d + sq) / 2.0
            if x_lo <= h0 + r1:
                x_hi = min(x_hi, h0 + r1)
            elif x_hi >= h0 + r2:
                x_lo = max(x_lo, h0 + r2)
            else:
                if abs(x_lo - (h0 + r1)) <= abs(x_hi - (h0 + r2)):
                    x_hi = min(x_hi, h0 + r1)
                else:
                    x_lo = max(x_lo, h0 + r2)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def cd_pass(h, c, n, order):
    improved = False
    for p in order:
        h0 = h[p]
        if h0 < 1e-13 or h0 > 1.0 - 1e-13:
            continue
        xn = solve_coord(c, h, p, n)
        if abs(xn - h0) > 1e-13:
            delta = xn - h0
            m_min = max(0, n - 1 - p)
            m_max = min(2 * n - 2, 2 * n - 2 - p)
            ms = np.arange(m_min, m_max + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(vf):
                a[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                a[vb] -= h[jb[vb]]
            c[ms] += delta * a
            diag_i = np.where(ms == (n - 1))[0]
            if len(diag_i) > 0:
                c[ms[diag_i[0]]] -= delta * delta
            h[p] = xn
            improved = True
    return improved

def cd_optimize(h, n, max_passes, deadline, rng=None, order='seq'):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        if order == 'rand' and rng is not None:
            ord_ = rng.permutation(n)
        else:
            ord_ = np.arange(n)
        if not cd_pass(h, c, n, ord_):
            break
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def binarize(h, n, rng):
    h_bin = (h >= 0.5).astype(float)
    cur_sum = int(h_bin.sum())
    target = n // 2
    if cur_sum > target:
        excess = cur_sum - target
        ones = np.where(h_bin == 1.0)[0]
        idx = np.argsort(h[ones])[:excess]
        h_bin[ones[idx]] = 0.0
    elif cur_sum < target:
        deficit = target - cur_sum
        zeros = np.where(h_bin == 0.0)[0]
        idx = np.argsort(-h[zeros])[:deficit]
        h_bin[zeros[idx]] = 1.0
    return h_bin

def random_binary(n, rng):
    h = np.zeros(n)
    positions = rng.choice(n, size=n // 2, replace=False)
    h[positions] = 1.0
    return h

def interpolate_h(h_src, n_src, n_tgt):
    x_src = np.linspace(0, 2, n_src, endpoint=False)
    x_tgt = np.linspace(0, 2, n_tgt, endpoint=False)
    return project_feasible(np.interp(x_tgt, x_src, h_src), n_tgt)

def swap_search(h, n, deadline, rng, max_iters=20000):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    cur_max = c.max()
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        delta_max = min(h[j], 1.0 - h[i])
        if delta_max < 1e-6:
            continue
        delta = 0.7 * delta_max
        # Compute effect on c
        m_min_i = max(0, n - 1 - i)
        m_max_i = min(2 * n - 2, 2 * n - 2 - i)
        m_min_j = max(0, n - 1 - j)
        m_max_j = min(2 * n - 2, 2 * n - 2 - j)
        whole_min = min(m_min_i, m_min_j)
        whole_max = max(m_max_i, m_max_j)
        ms = np.arange(whole_min, whole_max + 1)
        jf_i = i + ms - (n - 1)
        jb_i = i - ms + (n - 1)
        jf_j = j + ms - (n - 1)
        jb_j = j - ms + (n - 1)
        ai = np.zeros(len(ms))
        vf = (jf_i >= 0) & (jf_i < n)
        vb = (jb_i >= 0) & (jb_i < n)
        if np.any(vf):
            ai[vf] += (1.0 - h[jf_i[vf]])
        if np.any(vb):
            ai[vb] -= h[jb_i[vb]]
        aj = np.zeros(len(ms))
        vf = (jf_j >= 0) & (jf_j < n)
        vb = (jb_j >= 0) & (jb_j < n)
        if np.any(vf):
            aj[vf] += (1.0 - h[jf_j[vf]])
        if np.any(vb):
            aj[vb] -= h[jb_j[vb]]
        d = ai - aj
        # Check if this swap reduces the max
        new_c_seg = c[ms] + delta * d
        # Second order: -delta^2 where both i and j are involved
        k_ij = j - i + n - 1
        k_ji = i - j + n - 1
        if whole_min <= k_ij <= whole_max:
            idx = k_ij - whole_min
            new_c_seg[idx] -= delta * delta
        if whole_min <= k_ji <= whole_max:
            idx = k_ji - whole_min
            new_c_seg[idx] -= delta * delta
        # Check unchanged part
        unch_max = c[ms].max()
        # Full new max
        new_full_max = max(new_c_seg.max(), unch_max)
        if new_full_max < cur_max - 1e-10:
            c[ms] = new_c_seg
            cur_max = new_full_max
            h[i] += delta
            h[j] -= delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, n)
            if abs(h.sum() - n / 2.0) > 1e-10:
                c = np.correlate(h, 1.0 - h, mode='full')
                cur_max = c.max()
    return h, compute_c5(h, n)

        h_b2 = binarize(h_best, len(h_best), rng)
        h_f2, c_f2 = cd_optimize(h_b2, len(h_best), 200, deadline, order='seq')
        if c_f2 < best_c:
            h_best, best_c = h_f2, c_f2

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
