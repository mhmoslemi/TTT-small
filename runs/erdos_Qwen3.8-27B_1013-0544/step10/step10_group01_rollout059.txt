def project_onto_simplex_box(y, target):
    """Project y onto {h: sum(h) = target, 0<=h_i<=1}."""
    # Bisection on tau
    lo, hi = -1.0, 2.0  # tau range
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(y - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2, 0, 1)

def nesterov_subgrad(h, n, alpha, max_iters, deadline):
    """Nesterov accelerated projected subgradient for log-sum-exp objective."""
    x = h.copy()
    y = h.copy()
    t_k = 1.0
    best_h = h.copy()
    best_val = compute_c5(h, n)
    L = 4.0 * alpha * n / (2.0)  # Lipschitz constant estimate (rough)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        # Gradient at y
        c = compute_corr(y, n)
        m = c.max()
        shifted = alpha * (c - m)
        max_s = shifted.max()
        weights = np.exp(shifted - max_s)
        weights /= weights.sum()
        
        # Compute gradient
        grad = np.zeros(n)
        significant = weights > 1e-8
        for idx in np.where(significant)[0]:
            s = idx - (n - 1)
            w = weights[idx]
            if s == 0:
                grad += w * (1.0 - 2.0 * y)
            elif s > 0:
                grad[s:] += w * (1.0 - y[:n-s])
                grad[:n-s] -= w * y[s:]
            else:
                p = -s
                grad[:n-p] += w * (1.0 - y[p:])
                grad[p:] -= w * y[:n-p]
        
        # Gradient step
        step = 2.0 / (t_k + 1)  # Polyak-like step
        y = project_onto_simplex_box(y - step * grad / 2.0, n / 2.0)
        
        # Update t
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        x_new = project_onto_simplex_box(y + (t_k - 1.0) / t_new * (y - x), n / 2.0)
        
        x = x_new
        t_k = t_new
        
        c_x = compute_c5(x, n)
        if c_x < best_val:
            best_val = c_x
            best_h = x.copy()
    
    return best_h, best_val

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * dx)

def compute_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n, iters=200):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(iters):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d):
    vals = c_aff + d * a
    if has_diag:
        vals[diag_idx] = c_aff[diag_idx] + d * (1.0 - h0 - d)
    result = float(np.max(vals))
    if unaff_max > result:
        result = unaff_max
    return result

def coord_descent(h, n, max_passes, deadline, gs_iters=55, threshold=1e-8):
    h = h.copy()
    best_c = compute_c5(h, n)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for p_pass in range(max_passes):
        if time.time() > deadline:
            break
        c = compute_corr(h, n)
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= threshold or h0 >= 1.0 - threshold:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(valid_f):
                a[valid_f] += (1.0 - h[jf[valid_f]])
            if np.any(valid_b):
                a[valid_b] -= h[jb[valid_b]]
            diag_pos = np.where(ms == (n - 1))[0]
            has_diag = len(diag_pos) > 0
            diag_idx = int(diag_pos[0]) if has_diag else -1
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            d_lo = -h0
            d_hi = 1.0 - h0
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
            f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-15:
                    break
                if f1 < f2:
                    d_hi = d2
                    d2 = d1
                    f2 = f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
                else:
                    d_lo = d1
                    d1 = d2
                    f1 = f2
                    d2 = d_lo + gr * (d_hi - d_lo)
                    f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            d_opt = (d_lo + d_hi) / 2.0
            f_opt = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d_opt)
            f_curr = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, 0.0)
            if f_opt < f_curr - 1e-14:
                h[p] = h0 + d_opt
                improved = True
        if improved:
            h[np.where(h < threshold)[0]] = 0.0
            h[np.where(h > 1.0 - threshold)[0]] = 1.0
            h = project_feasible(h, n)
            c_new = compute_c5(h, n)
            if c_new < best_c:
                best_c = c_new
        else:
            break
    final_c = compute_c5(h, n)
    return h, min(best_c, final_c)

def project_onto_box_sum(y, target):
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:

def nesterov_smooth(h, n, alpha_list, max_iters, deadline):
    x = h.copy()
    best_h = h.copy()
    best_val = compute_c5(h, n)
    t_k = 1.0
    for alpha in alpha_list:
        step_base = 1.0 / (alpha * 4.0)
        for it in range(max_iters):
            if time.time() > deadline:
                break
            c = compute_corr(x, n)
            m = c.max()
            shifted = alpha * (c - m)
            max_s = shifted.max()
            w = np.exp(shifted - max_s)
            w /= w.sum()
            grad = np.zeros(n)
            sig = w > 1e-8
            for idx in np.where(sig)[0]:
                s = idx - (n - 1)
                wi = w[idx]
                if s == 0:
                    grad += wi * (1.0 - 2.0 * x)
                elif s > 0:
                    grad[s:] += wi * (1.0 - x[:n - s])
                    grad[:n - s] -= wi * x[s:]
                else:
                    p = -s
                    grad[:n - p] += wi * (1.0 - x[p:])
                    grad[p:] -= wi * x[:n - p]
            step = step_base / (t_k + 1.0)
            y = project_onto_box_sum(x - step * grad, n / 2.0)
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0

            t_k = t_new
            c_x = compute_c5(x, n)
            if c_x < best_val:
                best_val = c_x
                best_h = x.copy()
        t_k = 1.0
    return best_h, best_val
