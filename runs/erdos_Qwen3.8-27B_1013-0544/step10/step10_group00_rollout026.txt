def slsqp_opt(h, n, alphas, maxiter, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x, n=n: x.sum() - n / 2.0}]
    for alpha in alphas:
        if time.time() > deadline:
            break

                c = np.correlate(x, 1.0 - x, mode='full')

                s = alpha * (c - m)
                sm = s.max()
                lse = sm + np.log(np.sum(np.exp(s - sm)))
