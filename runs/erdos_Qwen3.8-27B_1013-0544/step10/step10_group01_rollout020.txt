import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:

            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)

def build_grad_matrices(n):
    ks = np.arange(n)
    ms = np.arange(2 * n - 1)
    idx1 = ks[:, None] + ms[None, :] - (n - 1)
    valid1 = (idx1 >= 0) & (idx1 < n)
    M1_idx = np.where(valid1, idx1, 0)
    M1_mask = valid1
    idx2 = ks[:, None] - ms[None, :] + (n - 1)
    valid2 = (idx2 >= 0) & (idx2 < n)
    M2_idx = np.where(valid2, idx2, 0)
    M2_mask = valid2
    return M1_idx, M1_mask, M2_idx, M2_mask

def softmax_grad(h, w, M1_idx, M1_mask, M2_idx, M2_mask):
    v = 1.0 - h
    M1 = np.where(M1_mask, v[M1_idx], 0.0)
    M2 = np.where(M2_mask, h[M2_idx], 0.0)

            h = h_new

def _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d):
    vals = c_aff + d * a
    if has_diag:
        vals[diag_idx] = c_aff[diag_idx] + d * (1.0 - h0) + d * d
    result = float(np.max(vals))
    if unaff_max > result:
        result = unaff_max
    return result

def coord_descent(h, n, max_passes, deadline, gs_iters=50, threshold=1e-6):
    h = h.copy()
    best_c = compute_c5(h, n)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(max_passes):
        if time.time() > deadline:

        improved = False

            if time.time() > deadline:

            h0 = h[p]
            if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(valid_f):

            if np.any(valid_b):
                a[valid_b] -= h[jb[valid_b]]
            diag_pos = np.where(ms == (n - 1))[0]

            d2 = d_lo + gr * (d_hi - d_lo)
            f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
            f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            for _ in range(gs_iters):

                if f1 < f2:

            h = project_feasible(h, n)
