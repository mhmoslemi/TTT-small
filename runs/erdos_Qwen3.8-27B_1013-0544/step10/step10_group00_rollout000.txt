```python
import numpy as np
import time
from numpy import sqrt


def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    for _ in range(10):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = h + (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = h + (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0.0:
            i = int(np.argmax(h))
            h[i] = min(1.0, h[i] + diff)
        else:
            i = int(np.argmin(h))
            h[i] = max(0.0, h[i] + diff)
    return h


def pair_fval_surr(c_aff, B, unaff, d, tmp):
    np.multiply(B, d, out=tmp)
    tmp += c_aff
    r = float(tmp.max())
    if unaff > r:
        r = unaff
    return r


def pair_fval_exact(c_aff, B, unaff, d, diag_pos, c1, c2, tmp):
    np.multiply(B, d, out=tmp)
    tmp += c_aff
    if d != 0.0:
        dd = d * d
        tmp[diag_pos] -= 2.0 * dd
        if c1 >= 0:
            tmp[c1] += dd
        if c2 >= 0:
            tmp[c2] += dd
    r = float(tmp.max())
    if unaff > r:
        r = unaff
    return r


def pair_golden_surr(c_aff, B, unaff, lo, hi, tmp, iters):
    if hi - lo < 1e-14:
        d = lo
        return d, pair_fval_surr(c_aff, B, unaff, d, tmp)
    gr = (sqrt(5.0) - 1.0) / 2.0
    d1 = hi - gr * (hi - lo)
    d2 = lo + gr * (hi - lo)
    f1 = pair_fval_surr(c_aff, B, unaff, d1, tmp)
    f2 = pair_fval_surr(c_aff, B, unaff, d2, tmp)
    for _ in range(iters):
        if hi - lo < 1e-13:
            break
        if f1 < f2:
            hi = d2
            d2 = d1
            f2 = f1
            d1 = hi - gr * (hi - lo)
            f1 = pair_fval_surr(c_aff, B, unaff, d1, tmp)
        else:
            lo = d1
            d1 = d2
            f1 = f2
            d2 = lo + gr * (hi - lo)
            f2 = pair_fval_surr(c_aff, B, unaff, d2, tmp)
    d = 0.5 * (lo + hi)
    return d, pair_fval_surr(c_aff, B, unaff, d, tmp)


def apply_pair(h, c, p, q, n, work, buf, golden_iters, s_base):
    if p == q:
        return None
    hp = float(h[p])
    hq = float(h[q])
    lo = -hq
    hi = 1.0 - hp
    if hi - lo < 1e-14:
        return None

    Rp = p if p > n - 1 - p else n - 1 - p
    Rq = q if q > n - 1 - q else n - 1 - q
    R = Rp if Rp > Rq else Rq
    klo = n - 1 - R
    khi = n - 1 + R
    L = 2 * R + 1

    s = s_base[klo:khi + 1]
    B = buf[:L]
    B.fill(0.0)

    fw = p + s
    m = (fw >= 0) & (fw < n)
    B[m] += 1.0 - h[fw[m]]
    bk = p - s
    m = (bk >= 0) & (bk < n)
    B[m] -= h[bk[m]]

    fw = q + s
    m = (fw >= 0) & (fw < n)
    B[m] -= 1.0 - h[fw[m]]
    bk = q - s
    m = (bk >= 0) & (bk < n)
    B[m] += h[bk[m]]

    c_aff = c[klo:khi + 1]
    unaff = 0.0
    if klo > 0:
        unaff = float(c[:klo].max())
    if khi < 2 * n - 2:
        u = float(c[khi + 1:].max())
        if u > unaff:
            unaff = u

    tmp = work[:L]
    diag = R
    g = q - p
    c1 = -1
    c2 = -1
    if g != 0:
        pos1 = R - g
        if 0 <= pos1 < L:
            c1 = pos1
        pos2 = R + g
        if 0 <= pos2 < L and pos2 != c1:
            c2 = pos2

    d_surr, _ = pair_golden_surr(c_aff, B, unaff, lo, hi, tmp, golden_iters)
    f0 = pair_fval_exact(c_aff, B, unaff, 0.0, diag, c1, c2, tmp)
    best_d = 0.0
    best_f = f0

    cand = [
        0.0, lo, hi, d_surr,
        0.5 * (lo + hi),
        lo + 0.25 * (hi - lo),
        lo + 0.75 * (hi - lo),
        0.5 * (hq - hp), -0.5 * (hq - hp)
    ]
    span = hi - lo
    for z in (-0.25, -0.1, 0.1, 0.25):
        cand.append(d_surr + z * span)
    if c1 >= 0:
        cand.append(-0.5 * float(B[c1]))
    if c2 >= 0:
        cand.append(-0.5 * float(B[c2]))
    cand.append(0.25 * float(B[diag]))

    for d in cand:
        if d < lo:
            d = lo
        elif d > hi:
            d = hi
        f = pair_fval_exact(c_aff, B, unaff, d, diag, c1, c2, tmp)
        if f + 1e-12 < best_f:
            best_f = f
            best_d = d

    imp = best_f + 1e-12 < f0
    if imp:
        h[p] = hp + best_d
        h[q] = hq - best_d
        if h[p] < 0.0:
            h[p] = 0.0
        elif h[p] > 1.0:
            h[p] = 1.0
        if h[q] < 0.0:
            h[q] = 0.0
        elif h[q] > 1.0:
            h[q] = 1.0

        new_aff = c_aff + B * best_d
        if best_d != 0.0:
            dd = best_d * best_d
            new_aff[diag] -= 2.0 * dd
            if c1 >= 0:
                new_aff[c1] += dd
            if c2 >= 0:
                new_aff[c2] += dd
            c[klo:khi + 1] = new_aff
    return (imp, best_f)


def cd_adjacent(h, n, deadline, passes, golden_iters, s_base, work, buf):
    c = np.correlate(h, 1.0 - h, mode="full")
    cmax = float(c.max())
    for it in range(passes):
        if time.time() > deadline:
            break
        improved = False
        start = it % 2
        for p in range(start, n - 1, 2):
            if time.time() > deadline:
                break
            res = apply_pair(h, c, p, p + 1, n, work, buf, golden_iters, s_base)
            if res is None:
                continue
            imp, cmax = res
            if imp:
                improved = True
        if not improved:
            break
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode="full")
        cmax = float(c.max())
    h = project_feasible(h, n)
    c = np.correlate(h, 1.0 - h, mode="full")
    cmax = float(c.max())
    return h, cmax * (2.0 / n)


def cd_steepest(h, n, deadline, iterations, golden_iters, s_base, work, buf, topK=8):
    c = np.correlate(h, 1.0 - h, mode="full")
    cmax = float(c.max())
    grad = np.zeros(n)
    noimp = 0
    mlen = len(c)
    for _ in range(iterations):
        if time.time() > deadline:
            break
        if mlen > topK:
            top = np.argpartition(c, -topK)[-topK:]
        else:
            top = np.arange(mlen)
        grad.fill(0.0)
        for k in top:
            kk = int(k)
            s = n - 1 - kk
            if s > 0:
                e = n - s
                grad[:e] += 1.0 - h[s:]
                grad[s:] -= h[:e]
            elif s < 0:
                t = -s
                e = n - t
                grad[t:] += 1.0 - h[:e]
                grad[:e] -= h[t:]
            else:
                grad += 1.0 - 2.0 * h
        grad -= grad.mean()
        p = int(np.argmin(grad))
        q = int(np.argmax(grad))
        if p == q:
            noimp += 1
            if noimp > 120:
                break
            continue
        res = apply_pair(h, c, p, q, n, work, buf, golden_iters, s_base)
        if res is None:
            noimp += 1
        else:
            imp, cmax = res
            if imp:
                noimp = 0
            else:
                noimp += 1
        if noimp > 120:
            break
    h = project_feasible(h, n)
    c = np.correlate(h, 1.0 - h, mode="full")
    cmax = float(c.max())
    return h, cmax * (2.0 / n)


def cd_ring(h, n, dist, deadline, passes, golden_iters, s_base, work, buf, count=140):
    c = np.correlate(h, 1.0 - h, mode="full")
    cmax = float(c.max())
    step = max(1, n // max(1, count))
    for it in range(passes):
        if time.time() > deadline:
            break
        improved = False
        shift = (it * step) % n
        for p in range(0, n, step):
            if time.time() > deadline:
                break
            pp = (p + shift) % n
            qq = (pp + dist) % n
            if pp == qq:
                continue
            res = apply_pair(h, c, pp, qq, n, work, buf, golden_iters, s_base)
            if res is None:
                continue
            imp, cmax = res
            if imp:
                improved = True
        if not improved:
            break
    h = project_feasible(h, n)
    c = np.correlate(h, 1.0 - h, mode="full")
    cmax = float(c.max())
    return h, cmax * (2.0 / n)


def cd_random(h, n, deadline, iterations, golden_iters, s_base, work, buf, rng):
    c = np.correlate(h, 1.0 - h, mode="full")
    cmax = float(c.max())
    noimp = 0
    for _ in range(iterations):
        if time.time() > deadline:
            break
        p = int(rng.randint(n))
        q = int(rng.randint(n))
        if p == q:
            continue
        res = apply_pair(h, c, p, q, n, work, buf, golden_iters, s_base)
        if res is None:
            noimp += 1
        else:
            imp, cmax = res
            if imp:
                noimp = 0
            else:
                noimp += 1
        if noimp > 250:
            break
    h = project_feasible(h, n)
    c = np.correlate(h, 1.0 - h, mode="full")
    cmax = float(c.max())
    return h, cmax * (2.0 / n)


def template_blocks(h, blocks):
    n = len(h)
    out = h.copy()
    for bb in range(blocks):
        s = int(bb * n / blocks)
        e = int((bb + 1) * n / blocks)
        if e > s:
            out[s:e] = h[s:e].mean()
    return out


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 5.0
    rng = np.random.RandomState(seed)

    try:
        h0 = np.asarray(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 700
        x = np.linspace(0.0, 2.0, n)
        h0 = 0.5 + 0.22 * np.sin(np.pi * 5.0 * x) + 0.08 * rng.randn(n)

    best_h = project_feasible(h0, n)
    best_c = compute_c5(best_h, n)

    m = 2 * n - 1
    s_base = n - 1 - np.arange(0, m)
    work = np.empty(m)
    buf = np.empty(m)

    rem = deadline - time.time()
    lim = min(deadline - 420.0, time.time() + max(60.0, 0.35 * rem))
    if time.time() < lim:
        hc, cc = cd_adjacent(best_h.copy(), n, lim, 1000, 14, s_base, work, buf)
        if cc < best_c:
            best_h, best_c = hc.copy(), cc

    rem = deadline - time.time()
    lim = min(deadline - 340.0, time.time() + max(40.0, 0.30 * rem))
    if time.time() < lim:
        hc, cc = cd_steepest(best_h.copy(), n, lim, 50000, 8, s_base, work, buf, 8)
        if cc < best_c:
            best_h, best_c = hc.copy(), cc

    rem = deadline - time.time()
    lim = min(deadline - 260.0, time.time() + max(30.0, 0.25 * rem))
    if time.time() < lim:
        hc, cc = cd_adjacent(best_h.copy(), n, lim, 1000, 16, s_base, work, buf)
        if cc < best_c:
            best_h, best_c = hc.copy(), cc

    limitD = deadline - 120.0
    remainingD = max(0.0, limitD - time.time())
    if remainingD > 30.0:
        dists = (2, 3, 5, 8, 13, 21, 34, 55)
        per = max(8.0, min(45.0, remainingD / len(dists)))
        for dist in dists:
            lim = min(limitD, time.time() + per)
            if lim <= time.time() + 1.0:
                break
            hc, cc = cd_ring(best_h.copy(), n, dist, lim, 4, 8, s_base, work, buf, 130)
            if cc < best_c:
                best_h, best_c = hc.copy(), cc

    limitR = deadline - 85.0
    if time.time() < limitR:
        remR = max(0.0, limitR - time.time())
        iters = max(300, int(remR * 7.0))
        hc, cc = cd_random(best_h.copy(), n, limitR, iters, 7, s_base, work, buf, rng)
        if cc < best_c:
            best_h, best_c = hc.copy(), cc

    limitT = deadline - 65.0
    if time.time() < limitT:
        cands = [np.sort(best_h)[::-1], template_blocks(best_h, 40)]
        x = np.linspace(0.0, 2.0, n)
        for fr in (3, 5, 7):
            cands.append(0.5 + 0.45 * np.sin(np.pi * float(fr) * x))
        for cand in cands:
            if time.time() >= limitT - 12.0:
                break
            tl = min(limitT - 12.0, time.time() + 35.0)
            cur = project_feasible(cand, n)
            hc, cc = cd_adjacent(cur, n, tl, 8, 10, s_base, work, buf)
            if cc <= best_c + 0.006 and time.time() < limitT - 4.0:
                tl2 = min(limitT - 4.0, time.time() + 25.0)
                hc, cc = cd_steepest(hc, n, tl2, 350, 8, s_base, work, buf, 8)
            if cc < best_c:
                best_h, best_c = hc.copy(), cc

    limitF = deadline - 3.0
    if time.time() < limitF:
        remF = max(0.0, limitF - time.time())
        lim = min(limitF, time.time() + max(10.0, 0.45 * remF))
        hc, cc = cd_steepest(best_h.copy(), n, lim, 50000, 8, s_base, work, buf, 8)
        if cc < best_c:
            best_h, best_c = hc.copy(), cc
        if time.time() < limitF:
            hc, cc = cd_adjacent(best_h.copy(), n, limitF, 1000, 18, s_base, work, buf)
            if cc < best_c:
                best_h, best_c = hc.copy(), cc

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
```
