import time

            tot = h.sum()
            if tot < 1e-15: break

def obj_pmean(x, n, p, penalty):
    """Power mean objective with penalty for sum constraint"""
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    # Power mean of c
    c_pos = np.maximum(c, 0)
    pm = (np.mean(c_pos ** p)) ** (1.0 / p)
    # Penalty for sum constraint
    s = x.sum() - n / 2.0
    return (pm + penalty * s * s) * dx

def _sum_eq(x, n):
    return x.sum() - n / 2.0

def _sum_eq_jac(x, n):
    return np.ones(n)

def slsqp_power_mean(h, n, ps, maxiter_each=200, penalty=100.0):

    best_c = compute_c5(h, n)

                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-18})

            new_c = compute_c5(new_h, n)

def vectorized_subgradient(h, n):

    m_star = int(np.argmax(c))
    s = m_star - (n - 1)
    # For lag s, gradient wrt h[j] is:
    # (1-h[j+s]) if j+s is valid, plus -h[j-s] if j-s is valid
    grad = np.zeros(n)
    if s >= 0:
        # j+s valid: j from 0 to n-1-s
        grad[:n-s] += (1.0 - h[s:])
        # j-s valid: j from s to n-1, j-s from 0 to n-1-s
        grad[s:] -= h[:n-s]
    else:
        s2 = -s
        grad[s2:] += (1.0 - h[:n-s2])
        grad[:n-s2] -= h[s2:]
    # At s=0, the contribution is 1-2*h[j] (already handled by the above)
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(80):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def _sum_eq(x, n):
    return x.sum() - n / 2.0

def _sum_eq_jac(x, n):
    return np.ones(n)

def obj_pmean(x, n, p, penalty):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    c_pos = np.maximum(c, 0.0)
    pm = (np.mean(c_pos ** p)) ** (1.0 / p)
    s = x.sum() - n / 2.0
    return (pm + penalty * s * s) * dx

def slsqp_optimize(h, n, ps, maxiter_each=200, penalty=50.0):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac, 'args': (n,)}]
    for p in ps:
        try:
            res = minimize(obj_pmean, best_h, args=(n, p, penalty),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def fast_cd(h, n, max_passes, deadline, gs_iters=60):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        cmax = float(c.max())
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(vf):
                a[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                a[vb] -= h[jb[vb]]
            dp = np.where(ms == (n - 1))[0]
            has_d = len(dp) > 0
            di = int(dp[0]) if has_d else -1
            unaff = 0.0
            if m_lo > 0:
                unaff = max(unaff, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff = max(unaff, float(c[m_hi + 1:].max()))
            c_aff = c[ms]

            def fval(d):
                vals = c_aff + d * a
                if has_d:
                    vals[di] = c_aff[di] + d * (1.0 - 2.0 * h0 - d)
                return max(float(np.max(vals)), unaff)

            dlo, dhi = -h0, 1.0 - h0
            d1 = dhi - gr * (dhi - dlo)
            d2 = dlo + gr * (dhi - dlo)
            f1, f2 = fval(d1), fval(d2)
            for _ in range(gs_iters):
                if dhi - dlo < 1e-14:
                    break
                if f1 < f2:
                    dhi, d2, f2 = d2, d1, f1
                    d1 = dhi - gr * (dhi - dlo)
                    f1 = fval(d1)
                else:
                    dlo, d1, f1 = d1, d2, f2
                    d2 = dlo + gr * (dhi - dlo)
                    f2 = fval(d2)
            dopt = (dlo + dhi) / 2.0
            if fval(dopt) < cmax - 1e-13:
                h[p] = h0 + dopt
                c[ms] = c_aff + dopt * a
                if has_d:
                    c[n - 1] = c_aff[di] + dopt * (1.0 - 2.0 * h0 - dopt)
                cmax = max(float(c.max()), unaff)
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def vec_subgrad(h, n, topk=3):
    c = np.correlate(h, 1.0 - h, mode='full')
    k = min(topk, len(c))
    top_idx = np.argsort(c)[-k:]
    grad = np.zeros(n)
    for mi in top_idx:
        s = int(mi) - (n - 1)
        if s == 0:
            grad += (1.0 - 2.0 * h)
        elif s > 0:
            grad[:n - s] += (1.0 - h[s:])
            grad[s:] -= h[:n - s]
        else:
            s2 = -s
            grad[s2:] += (1.0 - h[:n - s2])
            grad[:n - s2] -= h[s2:]
    return grad / k

def subgrad_explore(h, n, deadline, max_iters=5000, topk=3):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    mom = np.zeros(n)
    beta = 0.85
    base_step = 0.04
    for it in range(max_iters):
        if time.time() > deadline:
            break
        step = base_step / (1.0 + 0.003 * it)
        g = vec_subgrad(h, n, topk)
        mom = beta * mom + (1.0 - beta) * g
        h_new = project_feasible(h - step * mom, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_h, best_c = h_new.copy(), c_new
        h = h_new
    return best_h, best_c
