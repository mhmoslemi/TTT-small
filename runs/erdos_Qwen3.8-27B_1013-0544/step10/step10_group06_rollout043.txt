import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def apply_symmetry(h, n):
    h = h.copy()
    for i in range(n // 2):
        j = n - 1 - i
        h[i] = 0.5 * (h[i] + 1.0 - h[j])
        h[j] = 1.0 - h[i]
    if n % 2 == 1:
        h[n // 2] = 0.5
    return h

def solve_coord_sym(c, h, p, n, nbisect=40):
    """Solve for optimal h[p] given symmetry h[p]+h[n-1-p]=1"""
    a = h[p]
    if a < 1e-14 or a > 1.0 - 1e-14:
        return a
    np_ = n - 1 - p
    # d[m] = h[n-1-p-k] - h[p-k] - h[p+k] + h[n-1-p+k] where k = m-(n-1)
    # m ranges from 0 to 2n-2, k from -(n-1) to n-1
    ms = np.arange(2 * n - 1)
    ks = ms - (n - 1)
    # For each m, d[m] = h[np_-k] - h[p-k] - h[p+k] + h[np_+k]
    # where h[idx] = 0 if idx out of [0, n-1]
    idx1 = np_ - ks  # n-1-p-k
    idx2 = p - ks    # p-k
    idx3 = p + ks    # p+k
    idx4 = np_ + ks  # n-1-p+k
    
    d = np.zeros(2 * n - 1)
    for idx_arr, sign in [(idx1, 1), (idx2, -1), (idx3, -1), (idx4, 1)]:
        valid = (idx_arr >= 0) & (idx_arr < n)
        d[valid] += sign * h[idx_arr[valid]]
    
    # c[m](x) = c[m] + (x-a)*d[m]
    # We want to minimize max_m c[m](x) over x in [0,1]
    # This is minimizing a convex function (max of linear functions)
    
    # Binary search on T: find minimum T such that there exists x with all c[m](x) <= T
    # For a given T, we need: c[m] + (x-a)*d[m] <= T for all m
    # If d[m] > 0: x <= (T - c[m]) / d[m] + a
    # If d[m] < 0: x >= (T - c[m]) / d[m] + a
    # If d[m] = 0: need c[m] <= T
    
    cur_max = float(np.max(c))
    # Find the range of possible T values
    # At x=0: max_m (c[m] - a*d[m])
    # At x=1: max_m (c[m] + (1-a)*d[m])
    val_0 = float(np.max(c - a * d))
    val_1 = float(np.max(c + (1 - a) * d))
    
    L_lo = min(val_0, val_1, cur_max)  # lower bound on optimal T
    L_hi = max(val_0, val_1)  # upper bound (we can always achieve max(val_0, val_1))
    
    # Actually, the optimal T is between min(cur_max, min(val_0, val_1)) and max(val_0, val_1)
    # But cur_max is the current value, which might not be achievable (we're changing h[p])
    # The achievable range is [min over x of max_m c[m](x), max(val_0, val_1)]
    # Let's just use [0, max(val_0, val_1)] as bounds for bisection
    
    L_lo = 0.0
    L_hi = max(val_0, val_1)
    
    for _ in range(nbisect):
        if L_hi - L_lo < 1e-14:
            break
        T = (L_lo + L_hi) / 2.0
        # Check if feasible: exists x in [0,1] such that c[m] + (x-a)*d[m] <= T for all m
        x_lo = 0.0
        x_hi = 1.0
        pos = d > 1e-14
        neg = d < -1e-14
        if np.any(pos):
            x_hi = min(x_hi, float(np.min((T - c[pos]) / d[pos] + a)))
        if np.any(neg):
            x_lo = max(x_lo, float(np.max((T - c[neg]) / d[neg] + a)))
        zero_d = (d >= -1e-14) & (d <= 1e-14)
        if np.any(zero_d) and np.max(c[zero_d]) > T + 1e-14:
            x_lo = 1.0  # infeasible
        if x_lo > x_hi + 1e-14:
            L_lo = T
        else:
            L_hi = T
    
    T = (L_lo + L_hi) / 2.0
    x_lo = 0.0
    x_hi = 1.0
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        x_hi = min(x_hi, float(np.min((T - c[pos]) / d[pos] + a)))
    if np.any(neg):
        x_lo = max(x_lo, float(np.max((T - c[neg]) / d[neg] + a)))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if x_lo > x_hi + 1e-10:
        return a
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def coord_descent_sym(h, n, max_passes, deadline, rng, nbisect=40):
    h = apply_symmetry(h, n)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        order = rng.permutation(n // 2)
        for p_idx in order:
            if time.time() > deadline:
                break
            p = p_idx  # p in [0, n//2)
            x_new = solve_coord_sym(c, h, p, n, nbisect)
            if abs(x_new - h[p]) > 1e-13:
                h[p] = x_new
                h[n - 1 - p] = 1.0 - x_new
                improved = True
        if not improved:
            break
    return h, compute_c5(h, n)

def coord_descent_asym(h, n, max_passes, deadline, rng, nbisect=35):
    h = h.copy()
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        order = rng.permutation(n)
        for p in order:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            # Standard asymmetric coordinate update
            # c[m](x) = c[m] + (x-h0) * d[m]
            # d[m] = (1-h[p+m-(n-1)]) - h[p-m+(n-1)]
            ms = np.arange(2 * n - 1)
            kf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            d = np.zeros(2 * n - 1)
            vf = (kf >= 0) & (kf < n)
            vb = (jb >= 0) & (jb < n)
            d[vf] += (1.0 - h[kf[vf]])
            d[vb] -= h[jb[vb]]
            
            # Bisection on T
            val_0 = float(np.max(c - h0 * d))
            val_1 = float(np.max(c + (1 - h0) * d))
            L_lo = 0.0
            L_hi = max(val_0, val_1)
            for _ in range(nbisect):
                if L_hi - L_lo < 1e-14:
                    break
                T = (L_lo + L_hi) / 2.0
                x_lo_c = 0.0
                x_hi_c = 1.0
                pos = d > 1e-14
                neg = d < -1e-14
                if np.any(pos):
                    x_hi_c = min(x_hi_c, float(np.min((T - c[pos]) / d[pos] + h0)))
                if np.any(neg):
                    x_lo_c = max(x_lo_c, float(np.max((T - c[neg]) / d[neg] + h0)))
                zero_d = (d >= -1e-14) & (d <= 1e-14)
                if np.any(zero_d) and np.max(c[zero_d]) > T + 1e-14:
                    x_lo_c = 1.0
                if x_lo_c > x_hi_c + 1e-14:
                    L_lo = T
                else:
                    L_hi = T
            T = (L_lo + L_hi) / 2.0
            x_lo_c = 0.0
            x_hi_c = 1.0
            pos = d > 1e-14
            neg = d < -1e-14
            if np.any(pos):
                x_hi_c = min(x_hi_c, float(np.min((T - c[pos]) / d[pos] + h0)))
            if np.any(neg):
                x_lo_c = max(x_lo_c, float(np.max((T - c[neg]) / d[neg] + h0)))
            x_lo_c = max(x_lo_c, 0.0)
            x_hi_c = min(x_hi_c, 1.0)
            if x_lo_c > x_hi_c + 1e-10:
                x_new = h0
            else:
                x_new = float(np.clip((x_lo_c + x_hi_c) / 2.0, 0.0, 1.0))
            if abs(x_new - h0) > 1e-13:
                h[p] = x_new
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def apply_symmetry(h, n):
    h = h.copy()
    for i in range(n // 2):
        j = n - 1 - i
        h[i] = 0.5 * (h[i] + 1.0 - h[j])
        h[j] = 1.0 - h[i]
    if n % 2 == 1:
        h[n // 2] = 0.5
    return h

def solve_coord_1d(c, d, a, nbisect=40):
    """Find x in [0,1] minimizing max_m (c[m] + (x-a)*d[m])"""
    if a < 1e-15 or a > 1.0 - 1e-15:
        return a
    val_0 = float(np.max(c - a * d))
    val_1 = float(np.max(c + (1.0 - a) * d))
    L_lo = 0.0
    L_hi = max(val_0, val_1)
    for _ in range(nbisect):
        if L_hi - L_lo < 1e-15:
            break
        T = (L_lo + L_hi) / 2.0
        x_lo = 0.0
        x_hi = 1.0
        pos = d > 1e-14
        neg = d < -1e-14
        if np.any(pos):
            x_hi = min(x_hi, float(np.min((T - c[pos]) / d[pos] + a)))
        if np.any(neg):
            x_lo = max(x_lo, float(np.max((T - c[neg]) / d[neg] + a)))
        zd = (d >= -1e-14) & (d <= 1e-14)
        if np.any(zd) and float(np.max(c[zd])) > T + 1e-14:
            x_lo = 1.0
        if x_lo > x_hi + 1e-14:
            L_lo = T
        else:
            L_hi = T
    T = (L_lo + L_hi) / 2.0
    x_lo = 0.0
    x_hi = 1.0
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        x_hi = min(x_hi, float(np.min((T - c[pos]) / d[pos] + a)))
    if np.any(neg):
        x_lo = max(x_lo, float(np.max((T - c[neg]) / d[neg] + a)))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if x_lo > x_hi + 1e-10:
        return a
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def coord_descent(h, n, max_passes, deadline, rng, symmetric=False, nbisect=40):
    h = h.copy()
    if symmetric:
        h = apply_symmetry(h, n)
    npairs = n // 2 if symmetric else n
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        order = rng.permutation(npairs)
        for idx in order:
            if time.time() > deadline:
                break
            if symmetric:
                p = int(idx)
                q = n - 1 - p
                a = h[p]
                if a < 1e-15 or a > 1.0 - 1e-15:
                    continue
                ks = np.arange(2 * n - 1) - (n - 1)
                idx1 = q - ks
                idx2 = p - ks
                idx3 = p + ks
                idx4 = q + ks
                d = np.zeros(2 * n - 1)
                for ia, s in [(idx1, 1), (idx2, -1), (idx3, -1), (idx4, 1)]:
                    v = (ia >= 0) & (ia < n)
                    d[v] += s * h[ia[v]]
                x_new = solve_coord_1d(c, d, a, nbisect)
                if abs(x_new - a) > 1e-13:
                    h[p] = x_new
                    h[q] = 1.0 - x_new
                    improved = True
            else:
                p = int(idx)
                a = h[p]
                if a < 1e-15 or a > 1.0 - 1e-15:
                    continue
                kf = p + np.arange(2 * n - 1) - (n - 1)
                jb = p - np.arange(2 * n - 1) + (n - 1)
                d = np.zeros(2 * n - 1)
                vf = (kf >= 0) & (kf < n)

                x_new = solve_coord_1d(c, d, a, nbisect)

                    h[p] = x_new
                    improved = True
        if not improved:
            break
        if not symmetric:

    return h, compute_c5(h, n)

Let me implement the gradient correctly:

The log-sum-exp objective: F(h) = (m + (1/α) * log(∑_k exp(α * c[k]))) * dx

where m = max_k c[k].

∂F/∂h[p] = (1/α) * ∑_k w_k * ∂c[k]/∂h[p] * dx

where w_k = exp(α * (c[k] - m)) / ∑_j exp(α * (c[j] - m))

And ∂c[k]/∂h[p] is the "influence" of h[p] on c[k], which is exactly the d[m] we computed in coordinate descent (evaluated at the current h).

So the gradient is:
∇h[p] F = (1/α) * dx * ∑_k w_k * d_k[p]

where d_k[p] = (1-h[p+k']) - h[p-k''] (for the asymmetric case, with appropriate index adjustments).

This is a weighted sum of the "influence vectors", weighted by the softmax weights. It's O(n * (2n-1)) to compute, which is O(n^2). For n=700, that's 490K operations, very fast.

```python
def lse_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    shifted -= shifted.max()
    w = np.exp(shifted)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    ms = np.arange(2 * n - 1)
    # For each lag k, the influence on h[p] is:
    # (1-h[p-k]) if p-k in [0,n-1] minus h[p+k] if p+k in [0,n-1]
    # where k = m - (n-1)
    # We can compute this as a correlation-like operation
    # ∂c/∂h[p] = ∑_k w[k] * [(1-h[p-k_k]) - h[p+k_k]] * dx
    # where k_k = k - (n-1)
    # This is: dx * [correlate(w_shifted, ones-like) ... ]
    # Actually, let's think of it as:
    # term1[p] = ∑_k w[k] * (1-h[p-k_k]) = ∑_k w[k] - ∑_k w[k]*h[p-k_k]
    #          = sum(w) - (w * h)_correlation[p]  ... not quite
    # 
    # Let me just do it with numpy broadcasting for the lags that have significant weight
    significant = w > 1e-10
    for k_idx in np.where(significant)[0]:
        wv = w[k_idx]
        kf = int(k_idx - (n - 1))
        if kf >= 0:
            # (1-h[p-kf]) for p in [kf, n-1]
            grad[kf:] += wv * dx * (1.0 - h[:n-kf])
            # -h[p+kf] for p in [0, n-1-kf]
            grad[:n-kf] -= wv * dx * h[kf:]
        else:
            kf_abs = -kf
            # (1-h[p+kf_abs]) for p in [0, n-1-kf_abs]
            grad[:n-kf_abs] += wv * dx * (1.0 - h[kf_abs:])
            # -h[p-kf_abs] for p in [kf_abs, n-1]
            grad[kf_abs:] -= wv * dx * h[:n-kf_abs]
    return grad
```
