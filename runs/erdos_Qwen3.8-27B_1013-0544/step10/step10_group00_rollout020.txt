    vals = c_aff + d * a
    if has_diag:
        vals[diag_idx] = c_aff[diag_idx] + d * (1.0 - h0 - d)
    result = float(np.max(vals))
    if unaff_max > result:
        result = unaff_max
    return result

def coord_descent(h, n, max_passes, deadline, gs_iters=50, order='natural'):
    h = h.copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    best_c = compute_c5(h, n)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(c.max())
        improved = False
        if order == 'natural':
            p_order = range(n)
        else:
            p_order = np.random.permutation(n)
        for p in p_order:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-13 or h0 >= 1.0 - 1e-13:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(vf):
                a[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                a[vb] -= h[jb[vb]]
            dp = np.where(ms == (n - 1))[0]
            has_diag = len(dp) > 0
            diag_idx = int(dp[0]) if has_diag else -1
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            d_lo, d_hi = -h0, 1.0 - h0
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
            f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-14:
                    break
                if f1 < f2:
                    d_hi, d2, f2 = d2, d1, f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
                else:
                    d_lo, d1, f1 = d1, d2, f2
                    d2 = d_lo + gr * (d_hi - d_lo)
                    f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            d_opt = (d_lo + d_hi) / 2.0
            f_opt = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d_opt)
            if f_opt < cur_max - 1e-14:
                h[p] = h0 + d_opt
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        nc = compute_c5(h, n)
        if nc < best_c:
            best_c = nc
    return h, best_c

The Hessian of c[s] with respect to x is a 2n×2n matrix (well, it's bilinear so the "Hessian" is really the matrix of the bilinear form). For a fixed s, the contribution to the Hessian is sparse (only involves pairs (i, i+s)).

H_f = (1/alpha) * [diag(exp(alpha*(c-m))) - exp(2*alpha*(c-m)) / sum(exp(alpha*(c-m)))] ⊙ H_c

where ⊙ is some appropriate product. The eigenvalues are bounded by:
λ_max(H_f) ≤ alpha * max_s |λ_max(H_c[s])| 

For c[0] = Σ x[i]² * (-1) + Σ x[i], the Hessian is -2I (eigenvalues -2).
For c[s] (s≠0), the Hessian involves off-diagonal entries, with eigenvalues bounded by about 2.

import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            r = 1.0 - h
            t = r.sum()
            if t < 1e-15:
                break
            h = np.clip(h + (d / t) * r, 0.0, 1.0)
        else:
            t = h.sum()
            if t < 1e-15:
                break
            h = np.clip(h + (d / t) * h, 0.0, 1.0)
    return h

def smooth_obj_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    mx = c.max()
    val = (mx + np.log(np.sum(np.exp(alpha * (c - mx)))) / alpha) * dx
    w = alpha * np.exp(alpha * (c - mx))
    w /= w.sum()
    grad = np.zeros(n)
    active = np.where(w > 1e-12)[0]
    for s_idx in active:
        s = int(s_idx) - (n - 1)
        if s == 0:
            grad += w[s_idx] * (1.0 - 2.0 * x)
        else:
            jf = np.arange(n) + s
            jb = np.arange(n) - s
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            if np.any(vf):
                grad[vf] += w[s_idx] * (1.0 - x[jf[vf]])
            if np.any(vb):
                grad[vb] -= w[s_idx] * x[jb[vb]]
    return val, grad * dx

    best_x = h.copy()

    for alpha in alphas:
        if time.time() > deadline - 10:
            break
        iters = 3000
        xk = x.copy()
        yk = x.copy()
        tk = 1.0
        L = 8.0 * alpha / n
        _, fk = smooth_obj_grad(xk, n, alpha)
        for it in range(iters):
            if time.time() > deadline - 10:
                break
            x_new = project_feasible(yk - fk / L, n)
            _, fn = smooth_obj_grad(x_new, n, alpha)
            if it % 200 == 0:
                c5 = compute_c5(x_new, n)
                if c5 < best_c:
                    best_c = c5
                    best_x = x_new.copy()
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * tk * tk))
            yk = x_new + ((tk - 1.0) / t_new) * (x_new - xk)
            yk = project_feasible(yk, n)
            xk = x_new
            fk = fn
            tk = min(t_new, 400.0)
        x = best_x
    return best_x, best_c

def slsqp_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    mx = c.max()
    return (mx + np.log(np.sum(np.exp(alpha * (c - mx)))) / alpha) * dx

def _slsqp_eq(x, n):
    return x.sum() - n / 2.0

def _slsqp_eq_jac(x, n):
    return np.ones(n)

def slsqp_refine(h, n, deadline, alphas=None):
    if alphas is None:
        alphas = [500, 200, 100]
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_eq, 'jac': _slsqp_eq_jac, 'args': (n,)}]
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        try:
            res = minimize(slsqp_obj, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 400, 'ftol': 1e-18})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def mass_redistribute(h, n, k, rng):
    hp = h.copy()
    idx = rng.choice(n, size=k, replace=False)
    vals = hp[idx].copy()
    total = vals.sum()
    for _ in range(50):
        new_vals = np.abs(rng.normal(0.5 * n / k, 0.3 * n / k, k))
        new_vals = np.clip(new_vals, 0, 1)
        s = new_vals.sum()
        if s < 1e-10:
            break
        new_vals = new_vals * total / s
        new_vals = np.clip(new_vals, 0, 1)
        vals = new_vals
        break
    hp[idx] = vals
    return project_feasible(hp, n)
