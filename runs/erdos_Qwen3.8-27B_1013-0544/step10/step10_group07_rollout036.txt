def compute_c5(h, n):

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)

def cd_1d(c, h, p, n):

def block_cd_step(h, n, idxs, deadline):
    """Optimize h[idxs] jointly, fixing all other points."""
    b = len(idxs)
    target_sum = sum(n/2.0 - h[j] for j in range(n) if j not in set(idxs))
    # Actually, target sum for the block is: n/2 - sum(h[j] for j not in idxs)
    fixed_sum = sum(h[j] for j in range(n) if j not in set(idxs))
    target = n/2.0 - fixed_sum
    
    def obj(x):
        h_trial = h.copy()
        h_trial[idxs] = x
        c = np.correlate(h_trial, 1.0 - h_trial, mode='full')
        return c.max() * (2.0/n)

        h_trial = h.copy()
        h_trial[idxs] = x
        c = np.correlate(h_trial, 1.0 - h_trial, mode='full')
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        lse = mx + np.log(np.sum(np.exp(shifted - mx)))
        return (m + lse / alpha) * (2.0/n)
    
    bounds = [(0, 1)] * b
    x0 = h[idxs].copy()
    
    # Simple constraint: sum(x) = target
    try:
        res = minimize(smooth_obj, x0, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - target,

                       options={'maxiter': 200, 'ftol': 1e-15})

            c_after = compute_c5(h_trial, n)
            if c_after < c_before:
                h[idxs] = x_new
                return True
    except Exception:
        pass
    return False

def subgradient_step(h, n, step_size, deadline):
    """One subgradient step."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    k_star = np.argmax(c)
    # Subgradient w.r.t. h[p]:
    # c[k_star] = sum_i h[i] * (1-h[i+k_star-(n-1)])
    # d c[k_star] / d h[p] = (1-h[p+k_star-(n-1)]) - h[p-(k_star-(n-1))]
    # with appropriate boundary handling
    shift = k_star - (n - 1)
    sg = np.zeros(n)
    # Forward: h[p] appears as h[i] with i=p, contributing (1-h[p+shift])
    for p in range(n):
        coeff = 0.0
        jf = p + shift
        if 0 <= jf < n:
            coeff += (1.0 - h[jf])
        jb = p - shift
        if 0 <= jb < n:
            coeff -= h[jb]
        sg[p] = coeff
    
    # Normalize
    sg_norm = np.linalg.norm(sg)
    if sg_norm < 1e-15:
        return h, False
    
    h_new = h - step_size * sg / sg_norm
    h_new = project_feasible(h_new, n)
    return h_new, True

import time

def project_feasible(h, n):

        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:

        else:

def cd_1d(c, h, p, n):
    h0 = h[p]
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms].copy()
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    if has_diag:
        c_diag = c_base[diag_idx[0]]
        a[diag_idx] = 0.0
    unaff = 0.0
    if m_min > 0:
        unaff = max(unaff, float(np.max(c[:m_min])))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    cur_max = float(np.max(c))
    neg_mask = a < -1e-15
    pos_mask = a > 1e-15
    zero_mask = np.abs(a) <= 1e-15
    L_lo, L_hi = unaff, cur_max
    for _ in range(45):
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        ok = True
        if np.any(zero_mask) and np.any(c_base[zero_mask] > L + 1e-15):
            ok = False
        if ok and np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if ok and np.any(pos_mask):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        if x_lo > x_hi + 1e-14:
            ok = False

            Cq = L - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                if not (max(x_lo, 0.0) < r1 and min(x_hi, 1.0) > r2):
                    ok = False
        if ok:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:

                x_lo = min(x_lo, r2) if h0 - r1 <= r2 - h0 else max(x_lo, r2)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

    for _ in range(max_passes):

                m_min = n - 1 - p
                m_max = 2 * n - 2 - p

def smooth_gradient_step(h, n, alpha, lr, deadline):
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    # Softmax weights
    shifted = alpha * (c - m)
    weights = np.exp(shifted - shifted.max())
    weights = weights / weights.sum()
    # Gradient of smooth max w.r.t. c
    # c[k] = sum_i h[i] * (1-h[i+k-(n-1)])
    # d c[k] / d h[p] = (1-h[p+k-(n-1)]) - h[p-k+(n-1)] (with boundary checks)
    grad = np.zeros(n)
    for k in range(len(c)):
        w = weights[k]
        if w < 1e-10:
            continue
        shift = k - (n - 1)
        for p in range(n):
            coeff = 0.0
            jf = p + shift
            if 0 <= jf < n:
                coeff += (1.0 - h[jf])
            jb = p - shift
            if 0 <= jb < n:
                coeff -= h[jb]
            grad[p] += w * coeff
    h_new = h - lr * grad
    h_new = project_feasible(h_new, n)
    return h_new

- For p where p+shift is valid: add w * (1-h[p+shift])
- For p where p-shift is valid: subtract w * h[p-shift]

grad = np.zeros(n)
for k in range(len(c)):

    if shift >= 0:
        p_range = range(n - shift)
        grad[:n-shift] += w * (1.0 - h[shift:])
        grad[shift:] -= w * h[:n-shift]
    else:
        s = -shift
        p_range = range(n - s)
        grad[s:] += w * (1.0 - h[:n-s])
        grad[:n-s] -= w * h[s:]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def cd_1d(c, h, p, n):
    h0 = h[p]
    m_min = n - 1 - p
    m_max = 2 * n - 2 - p
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms].copy()
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    if has_diag:
        c_diag = c_base[diag_idx[0]]
        a[diag_idx] = 0.0
    unaff = 0.0
    if m_min > 0:
        unaff = max(unaff, float(np.max(c[:m_min])))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    cur_max = float(np.max(c))
    neg_mask = a < -1e-15
    pos_mask = a > 1e-15
    zero_mask = np.abs(a) <= 1e-15
    L_lo, L_hi = unaff, cur_max
    for _ in range(45):
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        ok = True
        if np.any(zero_mask) and np.any(c_base[zero_mask] > L + 1e-15):
            ok = False
        if ok and np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if ok and np.any(pos_mask):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        if x_lo > x_hi + 1e-14:
            ok = False
        if ok and has_diag:
            Cq = L - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                if not (max(x_lo, 0.0) < r1 and min(x_hi, 1.0) > r2):
                    ok = False
        if ok:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        Cq = L - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            if max(x_lo, 0.0) < r1 and min(x_hi, 1.0) > r2:
                x_hi = min(x_hi, r1) if h0 - r1 <= r2 - h0 else max(x_lo, r2)
                x_lo = min(x_lo, r2) if h0 - r1 <= r2 - h0 else max(x_lo, r2)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def exact_cd(h, n, max_passes, deadline):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            xn = cd_1d(c, h, p, n)
            if abs(xn - h0) > 1e-12:
                delta = xn - h0
                m_min = n - 1 - p
                m_max = 2 * n - 2 - p
                ms = np.arange(m_min, m_max + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                diag_i = np.where(ms == (n - 1))[0]
                c[ms] += delta * a
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True
        h = project_feasible(h, n)
        if not improved:
            break
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def block_smooth_obj(x_free, n, h_base, idxs, target, alpha, b_free):
    x_full = np.empty(len(idxs))
    x_full[:b_free] = x_free
    x_full[b_free] = target - x_free.sum()
    if x_full[b_free] < -0.05 or x_full[b_free] > 1.05:
        return 1e10 + (x_full[b_free] - 0.5) ** 2 * 100.0
    h_trial = h_base.copy()
    h_trial[idxs] = np.clip(x_full, 0.0, 1.0)
    c = np.correlate(h_trial, 1.0 - h_trial, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def block_cd_step(h, n, idxs, alpha, deadline):
    b = len(idxs)
    b_free = b - 1
    idx_arr = np.array(idxs)
    fixed_mask = np.ones(n, dtype=bool)
    fixed_mask[idx_arr] = False
    target = n / 2.0 - h[fixed_mask].sum()
    x0 = h[idx_arr].copy()
    x0_free = x0[:b_free].copy()
    bounds = [(0.0, 1.0)] * b_free
    try:
        res = minimize(block_smooth_obj, x0_free,
                       args=(n, h, idxs, target, alpha, b_free),
                       method='SLSQP', bounds=bounds,
                       options={'maxiter': 60, 'ftol': 1e-14})
        x_full = np.empty(b)
        x_full[:b_free] = np.clip(res.x, 0, 1)
        x_full[b_free] = target - x_full[:b_free].sum()
        if 0 <= x_full[b_free] <= 1:
            c_before = compute_c5(h, n)
            h_trial = h.copy()
            h_trial[idx_arr] = x_full
            c_after = compute_c5(h_trial, n)
            if c_after < c_before:
                h[idx_arr] = x_full
                return True
    except Exception:
        pass
    return False

def full_smooth_obj(x, n, alpha, penalty):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    base = (m + lse / alpha) * (2.0 / n)
    cv = x.sum() - n / 2.0
    return base + penalty * cv * cv

def slsqp_penalized(h, n, alphas, penalties, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline:
            break
        for penalty in penalties:
            if time.time() > deadline:
                break
            try:
                res = minimize(full_smooth_obj, best_h, args=(n, alpha, penalty),
                               method='SLSQP', bounds=bounds,
                               options={'maxiter': maxiter, 'ftol': 1e-18})
                new_h = project_feasible(res.x, n)
                new_c = compute_c5(new_h, n)
                if new_c < best_c - 1e-12:
                    best_h, best_c = new_h, new_c
            except Exception:
                pass
    return best_h, best_c

def do_block_phase(h, n, n_steps, rng, deadline, alpha_list):
    for it in range(n_steps):
        if time.time() > deadline:
            break
        bs = int(rng.choice([3, 4, 5, 6, 7]))
        if bs >= n:
            bs = max(3, n // 5)
        start = int(rng.randint(0, n - bs + 1))
        idxs = list(range(start, start + bs))
        alpha = float(rng.choice(alpha_list))
        block_cd_step(h, n, idxs, alpha, deadline)
        if it % 25 == 0:
            h, _ = exact_cd(h, n, 15, deadline)
    return h
