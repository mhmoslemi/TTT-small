```python
def compute_gradient(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    # softmax
    c_shifted = c - c.max()
    w = np.exp(alpha * c_shifted)
    w /= w.sum()
    # For each p, g[p] = sum_m w[m] * a[p,m]
    # where a[p,m] = (1-h[p+m-(n-1)]) if valid - h[p-m+(n-1)] if valid
    g = np.zeros(n)
    for p in range(n):
        m_min = n - 1 - p
        m_max = 2 * n - 2 - p
        ms = np.arange(m_min, m_max + 1)
        jf = p + ms - (n - 1)
        jb = p - ms + (n - 1)
        a = np.zeros(len(ms))
        vf = (jf >= 0) & (jf < n)
        a[vf] += (1.0 - h[jf[vf]])
        vb = (jb >= 0) & (jb < n)
        a[vb] -= h[jb[vb]]
        g[p] = np.dot(w[ms], a)
    return g, w, c
```

    x = h.copy()
    y = h.copy()  # momentum variable
    v = np.zeros(n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

            break
        # compute gradient at y
        g, w, c = compute_gradient(y, n, alphas[t])
        # Nesterov update
        beta = (1 + np.sqrt(1 + 4*t*t)) / (2*t + 2)  # standard Nesterov momentum
        v = beta * v - lr * g
        x_new = x + v
        x_new = project_feasible(x_new, n)
        y_new = x_new + (beta_new - beta) * (x_new - x)  # Nesterov extrapolation
        y_new = np.clip(y_new, 0, 1)
        x, y = x_new, y_new
        c_new = compute_c5(x, n)
        if c_new < best_c:
            best_h = x.copy()
            best_c = c_new
    return best_h, best_c

            ok = False
        if ok and np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if ok and np.any(pos_mask):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        if ok and x_lo > x_hi + 1e-13:
            ok = False
        if ok and has_diag:
            Cq = L - c_diag + h0 * (1.0 - h0)
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                if not (x_lo <= r1 or x_hi >= r2):
                    ok = False
        if ok:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag and x_lo <= x_hi:
        Cq = L - c_diag + h0 * (1.0 - h0)
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            s1lo, s1hi = x_lo, min(x_hi, r1)
            s2lo, s2hi = max(x_lo, r2), x_hi
            c1 = (s1lo + s1hi) / 2.0 if s1lo <= s1hi else -1
            c2 = (s2lo + s2hi) / 2.0 if s2lo <= s2hi else -1
            if c1 >= 0 and c2 >= 0:
                x_lo, x_hi = (s1lo, s1hi) if abs(h0 - c1) <= abs(h0 - c2) else (s2lo, s2hi)
            elif c1 >= 0:
                x_lo, x_hi = s1lo, s1hi
            elif c2 >= 0:
                x_lo, x_hi = s2lo, s2hi
            else:
                return h0
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def exact_cd(h, n, max_passes, deadline):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            xn = cd_1d(c, h, p, n)
            if abs(xn - h0) > 1e-12:
                delta = xn - h0
                m_min = n - 1 - p
                m_max = 2 * n - 2 - p
                ms = np.arange(m_min, m_max + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                a = np.zeros(len(ms))
                if np.any((jf >= 0) & (jf < n)):
                    a[(jf >= 0) & (jf < n)] += (1.0 - h[jf[(jf >= 0) & (jf < n)]])
                if np.any((jb >= 0) & (jb < n)):
                    a[(jb >= 0) & (jb < n)] -= h[jb[(jb >= 0) & (jb < n)]]
                diag_i = np.where(ms == (n - 1))[0]
                c[ms] += delta * a
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True
        h = project_feasible(h, n)
        if not improved:
            break
    return h, compute_c5(h, n)

def grad_smooth_max(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    c_s = c - c.max()
    w = np.exp(alpha * c_s)
    w /= w.sum()
    g = np.zeros(n)
    for p in range(n):
        m_min = n - 1 - p
        m_max = 2 * n - 2 - p
        ms = np.arange(m_min, m_max + 1)
        jf = p + ms - (n - 1)
        jb = p - ms + (n - 1)
        a = np.zeros(len(ms))
        vf = (jf >= 0) & (jf < n)
        vb = (jb >= 0) & (jb < n)
        if np.any(vf):
            a[vf] += (1.0 - h[jf[vf]])
        if np.any(vb):
            a[vb] -= h[jb[vb]]
        g[p] = np.dot(w[ms], a)
    return g, c.max() * (2.0 / n)

def momentum_opt(h, n, deadline, max_iter=5000):
    x = h.copy()
    v = np.zeros(n)
    best_h, best_c = x.copy(), compute_c5(x, n)
    beta = 0.9
    lr = 0.01
    alpha = 100.0
    for t in range(max_iter):
        if time.time() > deadline:
            break
        g, _ = grad_smooth_max(x, n, alpha)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-12:
            break
        g = g / g_norm
        v = beta * v - lr * g
        x_new = project_feasible(x + v, n)
        c_new = compute_c5(x_new, n)
        if c_new < best_c:
            best_h, best_c = x_new.copy(), c_new
        x = x_new
        if t % 200 == 0:
            alpha *= 1.5
            lr *= 0.95
    return best_h, best_c

def block3_cd(h, n, max_passes, deadline):
    h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(0, n - 2, 3):
            s = h[p] + h[p+1] + h[p+2]
            h_save = (h[p], h[p+1], h[p+2])
            h_base = h.copy()
            h_base[p] = 0.0
            h_base[p+1] = 0.0
            h_base[p+2] = 0.0
            c0 = np.correlate(h_base, 1.0 - h_base, mode='full')
            # Build sensitivity for each of the 3 points
            sens = [None, None, None]
            for i in range(3):
                pp = p + i
                m_min = n - 1 - pp
                m_max = 2 * n - 2 - pp
                ms = np.arange(m_min, m_max + 1)
                jf = pp + ms - (n - 1)
                jb = pp - ms + (n - 1)
                a = np.zeros(len(ms))
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                if np.any(vf):
                    a[vf] += (1.0 - h_base[jf[vf]])
                if np.any(vb):
                    a[vb] -= h_base[jb[vb]]
                diag_i = np.where(ms == (n - 1))[0]
                if len(diag_i) > 0:
                    a[diag_i] = 0.0
                sens[i] = (ms, a)
            # Parameterize: h[p]=a, h[p+1]=b, h[p+2]=s-a-b
            # Grid search
            lo_a, hi_a = max(0, s-2), min(1, s)
            if hi_a - lo_a < 1e-12:
                continue
            best_val = compute_c5(h, n)
            best_ab = (h_save[0], h_save[1])
            ngrid = 30
            for ga in np.linspace(lo_a, hi_a, ngrid):
                lo_b = max(0, s - ga - 1)
                hi_b = min(1, s - ga)
                if hi_b - lo_b < 1e-12:
                    continue
                for gb in np.linspace(lo_b, hi_b, ngrid):
                    h[p], h[p+1], h[p+2] = ga, gb, s - ga - b
                    # Quick eval: only affected lags
                    affected = set()
                    for i in range(3):
                        affected.update(range(sens[i][0][0], sens[i][0][-1]+1))
                    c_new = c0.copy()
                    for i in range(3):
                        ms_i, a_i = sens[i]
                        c_new[ms_i] += (h[p+i] - 0.0) * a_i
                    c_new[n-1] += ga*(1-ga) + gb*(1-gb) + (s-ga-gb)*(1-(s-ga-gb))
                    val = np.max(c_new) * (2.0/n)
                    if val < best_val:
                        best_val = val
                        best_ab = (ga, gb)
            h[p], h[p+1], h[p+2] = best_ab[0], best_ab[1], s - best_ab[0] - best_ab[1]
            if abs(h[p] - h_save[0]) > 1e-10 or abs(h[p+1] - h_save[1]) > 1e-10 or abs(h[p+2] - h_save[2]) > 1e-10:
                improved = True
        h = project_feasible(h, n)
        if not improved:
            break
    return h, compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = np.full(500, 0.5)
        n = 500
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)
    # Phase 1: 1D CD (fixed version)
    h1, c1 = exact_cd(h_best, n, 2000, deadline)
    if c1 < best_c:
        h_best, best_c = h1, c1
    # Phase 2: 3D block CD
    if time.time() < deadline - 200:
        h2, c2 = block3_cd(h_best, n, 50, deadline)
        if c2 < best_c:
            h_best, best_c = h2, c2
    # Phase 3: More 1D CD after block
    if time.time() < deadline - 200:
        h3, c3 = exact_cd(h_best, n, 1000, deadline)
        if c3 < best_c:
            h_best, best_c = h3, c3
    # Phase 4: Momentum gradient
    if time.time() < deadline - 300:
        h4, c4 = momentum_opt(h_best, n, deadline - 300, 3000)
        if c4 < best_c:
            h_best, best_c = h4, c4
            h5, c5 = exact_cd(h_best, n, 500, deadline)
            if c5 < best_c:
                h_best, best_c = h5, c5
    # Phase 5: Multi-n exploration
    n_list = [400, 500, 600, 800, 900, 1000]
    for n_try in n_list:
        if time.time() > deadline - 150:
            break
        if n_try == len(h_best):
            continue
        x_old = np.linspace(0, 2, len(h_best))
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_old, h_best), n_try)
        h_o, c_o = exact_cd(h_int, n_try, 800, deadline)
        if time.time() < deadline - 100:
            h_o2, c_o2 = block3_cd(h_o, n_try, 30, deadline)
            if c_o2 < c_o:
                h_o, c_o = h_o2, c_o2
            h_o3, c_o3 = exact_cd(h_o, n_try, 500, deadline)
            if c_o3 < c_o:
                h_o, c_o = h_o3, c_o3
        if c_o < best_c - 1e-8:
            h_best, best_c, n = h_o, c_o, n_try
    # Phase 6: Shake and settle
    for r in range(15):
        if time.time() > deadline - 30:
            break
        s = 0.003 + 0.003 * r
        hp = project_feasible(h_best + rng.normal(0, s, len(h_best)), len(h_best))
        h_o, c_o = exact_cd(hp, len(h_best), 400, deadline)
        if c_o < best_c:
            h_best, best_c = h_o, c_o
    # Phase 7: Final polish
    if time.time() < deadline - 5:
        h_f, c_f = exact_cd(h_best, len(h_best), 1000, deadline)
        if c_f < best_c:
            h_best, best_c = h_f, c_f
    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
