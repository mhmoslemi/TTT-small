```python
def lse_gradient(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    shifted = alpha * (c - c.max())
    w = np.exp(shifted - shifted.max())
    w /= w.sum()
    dx = 2.0 / n
    # For each p, compute sum over shifts
    # shift s corresponds to w index s + (n-1)
    # contribution to grad[p] from shift s:
    #   w[s] * (1-h[p+s]) if 0<=p+s<n
    #   -w[s] * h[p-s] if 0<=p-s<n
    # Build a matrix or use broadcasting
    # For n=700, a 700x1399 matrix is fine
    shifts = np.arange(-(n-1), n)  # all possible shifts
    # For position p and shift s: forward index is p+s, backward index is p-s
    P = np.arange(n)[:, None]  # (n, 1)
    S = shifts[None, :]  # (1, 2n-1)
    fwd = P + S  # (n, 2n-1)
    bwd = P - S  # (n, 2n-1)
    fwd_valid = (fwd >= 0) & (fwd < n)
    bwd_valid = (bwd >= 0) & (bwd < n)
    # (1-h[fwd]) for valid entries
    fwd_val = np.where(fwd_valid, 1.0 - h[np.clip(fwd, 0, n-1)], 0.0)
    bwd_val = np.where(bwd_valid, h[np.clip(bwd, 0, n-1)], 0.0)
    grad = dx * (w[None, :] * fwd_val - w[None, :] * bwd_val).sum(axis=1)
    return grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def solve_coord(c, h, p, n, nbisect=30):
    h0 = h[p]
    if h0 < 1e-14 or h0 > 1.0 - 1e-14:
        return h0
    lo = max(0, n - 1 - p)
    hi = min(2 * n - 2, n - 2 + p)
    lo2 = max(0, p)
    hi2 = min(2 * n - 2, p + n - 1)
    m_lo = min(lo, lo2)
    m_hi = max(hi, hi2)
    ms = np.arange(m_lo, m_hi + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms].copy()
    di = np.where(ms == (n - 1))[0]
    has_diag = len(di) > 0
    c_diag = c_base[di[0]] if has_diag else 0.0
    if has_diag:
        a[di] = 0.0
    nd = np.ones(len(ms), dtype=bool)
    if has_diag:
        nd[di] = False
    a_nd, c_nd = a[nd], c_base[nd]
    unaff = 0.0
    if m_lo > 0:
        unaff = max(unaff, float(np.max(c[:m_lo])))
    if m_hi < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_hi + 1:])))
    L_lo, L_hi = unaff, float(np.max(c))
    for _ in range(nbisect):
        if L_hi - L_lo < 1e-14:
            break
        T = (L_lo + L_hi) / 2.0
        ok = unaff <= T + 1e-15
        if ok and len(a_nd) > 0:
            xl, xh = 0.0, 1.0
            neg, pos = a_nd < -1e-13, a_nd > 1e-13
            if np.any(neg):
                xl = max(xl, float(np.max((T - c_nd[neg]) / a_nd[neg])))
            if np.any(pos):
                xh = min(xh, float(np.min((T - c_nd[pos]) / a_nd[pos])))
            xl, xh = max(xl, 0.0), min(xh, 1.0)
            if xl > xh + 1e-14:
                ok = False
        if ok and has_diag:
            Cq = T - c_diag + h0 - h0 * h0
            disc = 1.0 - 4.0 * Cq
            if disc >= 0:
                r1 = (1.0 - np.sqrt(disc)) / 2.0
                r2 = (1.0 + np.sqrt(disc)) / 2.0
                xl2, xh2 = 0.0, 1.0
                if len(a_nd) > 0:
                    neg, pos = a_nd < -1e-13, a_nd > 1e-13
                    if np.any(neg):
                        xl2 = max(xl2, float(np.max((T - c_nd[neg]) / a_nd[neg])))
                    if np.any(pos):
                        xh2 = min(xh2, float(np.min((T - c_nd[pos]) / a_nd[pos])))
                    xl2, xh2 = max(xl2, 0.0), min(xh2, 1.0)
                if xl2 > r1 and xh2 < r2:
                    ok = False
        L_hi = T if ok else L_lo
        L_lo = L_lo if ok else T
    T = (L_lo + L_hi) / 2.0
    xl, xh = 0.0, 1.0
    if len(a_nd) > 0:
        neg, pos = a_nd < -1e-13, a_nd > 1e-13
        if np.any(neg):
            xl = max(xl, float(np.max((T - c_nd[neg]) / a_nd[neg])))
        if np.any(pos):
            xh = min(xh, float(np.min((T - c_nd[pos]) / a_nd[pos])))
    xl, xh = max(xl, 0.0), min(xh, 1.0)
    if has_diag:
        Cq = T - c_diag + h0 - h0 * h0
        disc = 1.0 - 4.0 * Cq
        if disc >= 0:
            r1 = (1.0 - np.sqrt(disc)) / 2.0
            r2 = (1.0 + np.sqrt(disc)) / 2.0
            if xl < r1:
                xh = min(xh, r1)
            elif xh > r2:
                xl = max(xl, r2)
            else:
                xl = max(xl, r2 - 1e-9) if h0 - r1 > r2 - h0 else min(xh, r1 + 1e-9)
    xl, xh = max(xl, 0.0), min(xh, 1.0)
    return float(np.clip((xl + xh) / 2.0, 0.0, 1.0)) if xl <= xh + 1e-10 else h0

def cd_refine(h, n, max_passes, deadline, rng, nbisect=30):
    h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        order = rng.permutation(n)
        for p in order:
            if time.time() > deadline:
                break
            old = h[p]
            if old < 1e-14 or old > 1.0 - 1e-14:
                continue
            x_new = solve_coord(c, h, p, n, nbisect)
            if abs(x_new - old) > 1e-13:
                h[p] = x_new
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def lse_obj_grad(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    sh = alpha * (c - c.max())
    w = np.exp(sh - sh.max())
    w /= w.sum()
    obj = (c.max() + np.log(w.sum() * np.exp(sh.max() - sh.max())) / alpha) * (2.0 / n)
    obj = (c.max() + np.log(np.sum(np.exp(sh))) - sh.max()) / alpha * (2.0 / n) + c.max() * (2.0 / n)
    # Simpler: obj = (c.max() + (sh.max() + log(sum(exp(sh - sh.max()))))/alpha) * dx
    dx = 2.0 / n
    obj = (c.max() + (sh.max() + np.log(np.sum(np.exp(sh - sh.max())))) / alpha) * dx
    P = np.arange(n)[:, None]
    S = np.arange(-(n - 1), n)[None, :]
    fwd = P + S
    bwd = P - S
    fv = (fwd >= 0) & (fwd < n)
    bv = (bwd >= 0) & (bwd < n)
    fwd_val = np.where(fv, 1.0 - x[np.clip(fwd, 0, n - 1)], 0.0)
    bwd_val = np.where(bv, x[np.clip(bwd, 0, n - 1)], 0.0)
    grad = dx * (w[None, :] * fwd_val - w[None, :] * bwd_val).sum(axis=1)
    return obj, grad

def sum_con(x, n):
    return x.sum() - n / 2.0

def lse_optimize(h, n, alphas, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline - 20:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': lambda x, n=n: np.ones(n), 'args': (n,)}]
        try:
            res = minimize(lse_obj_grad, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def make_perturbation(h, n, rng, typ):
    if typ % 5 == 0:
        sigma = rng.uniform(0.02, 0.08)
        return project_feasible(h + sigma * rng.randn(n), n)
    elif typ % 5 == 1:
        a = rng.randint(0, n - 20)
        b = a + rng.randint(10, n - a)
        h2 = h.copy()
        h2[a:b] = 1.0 - h2[a:b]
        return project_feasible(h2, n)
    elif typ % 5 == 2:
        shift = rng.randint(5, n // 4)
        h2 = np.roll(h, shift)
        blend = rng.uniform(0.3, 0.7)
        h2 = blend * h + (1 - blend) * h2
        return project_feasible(h2, n)
    elif typ % 5 == 3:
        spec = np.fft.rfft(h)
        k = len(spec)
        mask = np.zeros(k)
        mask[rng.randint(1, k - 1, size=min(3, k - 1))] = 1.0
        spec2 = spec * (1 + 0.05 * mask * rng.randn(k))
        h2 = np.fft.irfft(spec2, n=n)
        return project_feasible(h2, n)
    else:
        blocks = rng.randint(3, 8)
        h2 = h.copy()
        for _ in range(blocks):
            a = rng.randint(0, n - 10)
            b = a + rng.randint(5, n - a)
            h2[a:b] += rng.uniform(-0.15, 0.15)
        return project_feasible(h2, n)
```
