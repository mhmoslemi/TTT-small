def flow_direction(s,n,k,rng):
    pos=np.where(s>1e-8)[0]; neg=np.where(s<-1e-8)[0]
    kmin=max(1,min(k,len(pos),len(neg)))
    if kmin<1: return None

    ni=rng.choice(len(neg), kmin, replace=False)

import numpy as np
import time

COORD_CACHE = {}

def project_box_sum(x, lb, ub, target):
    x = np.asarray(x, dtype=float)
    lo = lb - float(np.max(x)) - 1.0
    hi = ub - float(np.min(x)) + 1.0
    target = float(target)
    for _ in range(45):
        lam = 0.5 * (lo + hi)
        if float(np.sum(np.clip(x + lam, lb, ub))) < target:
            lo = lam
        else:
            hi = lam
    return np.clip(x + 0.5 * (lo + hi), lb, ub)

def project_h(x, n):
    return project_box_sum(x, 0.0, 1.0, 0.5 * n)

def project_s(x, n):
    return project_box_sum(x, -1.0, 1.0, 0.0)

def c5_h(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def prepare_s(s, n):
    k = np.arange(n)
    pref = np.empty(n + 1, dtype=float)
    pref[0] = 0.0
    pref[1:] = np.cumsum(s)
    P = pref[n - k]
    A = np.correlate(s, s, mode='full')[n - 1:n - 1 + n]
    Ns = n - k
    val = (Ns + 2.0 * np.abs(P) - A) / (2.0 * n)
    return P, A, Ns, val

def lse_s(s, n, alpha):
    P, A, Ns, val = prepare_s(s, n)
    mx = float(val.max())
    sh = alpha * (val - mx)
    msh = float(sh.max())
    e = np.exp(sh - msh)
    se = float(e.sum())
    w = e / se
    sign = np.sign(P)
    sign[np.abs(P) < 1e-13] = 0.0
    gP = np.cumsum(w * sign)[::-1]
    gA = np.convolve(w, s, mode='full')[:n] + np.convolve(w[::-1], s, mode='full')[n - 1:n - 1 + n]
    return mx + (msh + np.log(se)) / alpha, (2.0 * gP - gA) / (2.0 * n), P, A, Ns

def eval_dir(P, Q, A0, R, D2, Ns, t, n):
    return float(np.max((Ns + 2.0 * np.abs(P + t * Q) - (A0 + t * R + t * t * D2)) / (2.0 * n)))

def feasible_tmax(s, d):
    t = 2.0 - 1e-9
    pos = d > 1e-15
    neg = d < -1e-15
    if np.any(pos):
        t = min(t, float(np.min((1.0 - s[pos]) / d[pos])))
    if np.any(neg):
        t = min(t, float(np.min((s[neg] + 1.0) / (-d[neg]))))
    return max(0.0, t)

def line_dir(s, n, d, Ccur, P, A, Ns, tmax, maxevals):

def flow_direction(s, n, k, rng):
    pos = np.where(s > 1e-8)[0]
    neg = np.where(s < -1e-8)[0]
    kk = min(k, len(pos), len(neg))

    d[neg[ni]] = 1.0
    d = d - d.mean()
    md = float(np.max(np.abs(d)))
    if md < 1e-14:
        return None
    return d / md

                if xhi < r1 - 1e-12 and xlo > r2 + 1e-12:
                    ok = False
        if ok:
            Lhi = L
        else:
            Llo = L

    L = Lhi
    xlo, xhi = 0.0, 1.0
    ok = L + 1e-12 >= czmax
    if ok and hneg:
        v = float(np.max((L - cneg + aneg * h0) / aneg))
        if v > xlo:
            xlo = v
        if xlo > 1.0 + 1e-8:
            ok = False
    if ok and hpos:
        v = float(np.min((L - cpos + apos * h0) / apos))
        if v < xhi:
            xhi = v
        if xhi < -1e-8:
            ok = False
    if ok:
        xlo = max(0.0, xlo)
        xhi = min(1.0, xhi)
        if xlo > xhi + 1e-8:
            ok = False
    if not ok:
        return h0, ms, a, dp
    if not has:
        if h0 < xlo:
            x = xlo
        elif h0 > xhi:
            x = xhi
        else:
            x = h0
        return x, ms, a, dp

    disc = ad * ad + 4.0 * (cd - L)
    if disc < 0.0:
        return float(np.clip(h0, xlo, xhi)), ms, a, dp
    r1 = h0 + 0.5 * (ad - np.sqrt(disc))
    r2 = h0 + 0.5 * (ad + np.sqrt(disc))
    best_x = None
    best_e = 1e100
    e1 = min(xhi, r1)
    if xlo <= e1 + 1e-12:
        x1 = float(np.clip(h0, xlo, e1))
        e = abs(x1 - h0)
        if e < best_e:
            best_x, best_e = x1, e
    s2 = max(xlo, r2)
    if s2 <= xhi + 1e-12:
        x2 = float(np.clip(h0, s2, xhi))
        e = abs(x2 - h0)
        if e < best_e:
            best_x, best_e = x2, e
    if best_x is None:
        best_x = h0
    return best_x, ms, a, dp

def exact_cd(h, n, max_passes, deadline):
    h = project_h(h, n)
    c = np.correlate(h, 1.0 - h, mode='full')
    infos = get_coord_infos(n)
    for pass_no in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        order = infos if pass_no % 2 == 0 else infos[::-1]
        for info in order:
            p = info[0]
            x, ms, a, dp = solve_coord(c, h, info)
            delta = x - h[p]
            if abs(delta) > 1e-12:
                h[p] = x
                c[ms] += delta * a
                if dp >= 0:
                    c[ms[dp]] -= delta * delta
                improved = True
        if not improved:
            break
        h = project_h(h, n)
        c = np.correlate(h, 1.0 - h, mode='full')
    return h, c5_h(h, n)

def interpolate_h(h_src, n_src, n_tgt):
    xs = np.linspace(0.0, 2.0, n_src, endpoint=False)
    xt = np.linspace(0.0, 2.0, n_tgt, endpoint=False)
    return project_h(np.interp(xt, xs, h_src), n_tgt)

def default_h(n, rng):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = (0.5 + 0.2 * np.cos(2.0 * np.pi * x + rng.uniform(0.0, 2.0 * np.pi)) +
         0.13 * np.cos(4.0 * np.pi * x + rng.uniform(0.0, 2.0 * np.pi)) +
         0.07 * np.sin(6.0 * np.pi * x))
    return project_h(h, n)

def optimize_set(h, n, deadline, rng):
    h = project_h(h, n)
    best_h = h.copy()
    best_C = c5_h(h, n)
    cur_h = h.copy()
    cur_C = best_C
    cur_s = project_s(2.0 * cur_h - 1.0, n)
    phase = 0
    while time.time() < deadline and phase < 120:
        phase += 1
        for alpha in (20, 120, 600):
            if time.time() > deadline:
                break
            s2, _ = grad_line_search(cur_s, n, alpha, deadline, cur_C, 4, 28)
            h2 = project_h(0.5 * (1.0 + s2), n)
            h2, C2 = exact_cd(h2, n, 1, deadline)
            if C2 < cur_C - 1e-13:
                cur_h, cur_C = h2, C2
                cur_s = project_s(2.0 * cur_h - 1.0, n)
                if cur_C < best_C:
                    best_h, best_C = cur_h.copy(), cur_C
        if time.time() > deadline:
            break
        for k in (2, 8):
            if time.time() > deadline:
                break
            s2, _ = flow_line_search(cur_s, n, deadline, cur_C, 3, k, rng)
            h2 = project_h(0.5 * (1.0 + s2), n)
            h2, C2 = exact_cd(h2, n, 1, deadline)
            if C2 < cur_C - 1e-13:
                cur_h, cur_C = h2, C2
                cur_s = project_s(2.0 * cur_h - 1.0, n)
                if cur_C < best_C:
                    best_h, best_C = cur_h.copy(), cur_C
        if time.time() < deadline:
            h2, C2 = exact_cd(cur_h, n, 2, deadline)
            if C2 < cur_C - 1e-13:
                cur_h, cur_C = h2, C2
                cur_s = project_s(2.0 * cur_h - 1.0, n)
                if cur_C < best_C:
                    best_h, best_C = cur_h.copy(), cur_C
    best_h = project_h(best_h, n)
    return best_h, c5_h(best_h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget = max(60.0, float(budget_s))
    deadline = t0 + budget - 8.0
    rng = np.random.RandomState(seed)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 700
        h = default_h(n, rng)

    h, C = exact_cd(h, n, 2, deadline)
    d1 = min(deadline, t0 + 0.45 * budget)
    h, C = optimize_set(h, n, d1, rng)
    n = len(h)

    alt_dead = min(deadline, t0 + 0.75 * budget)
    cands = [350, 500, 600, 800, max(160, int(0.7 * n)), max(160, int(1.3 * n))]
    seen = set()
    for ntry in cands:
        if time.time() > alt_dead - 35:
            break
        if ntry < 120 or ntry > 1200 or ntry == n or ntry in seen:
            continue
        seen.add(ntry)
        sub_dead = min(alt_dead, time.time() + 45.0)
        if sub_dead <= time.time() + 10.0:
            continue
        h_try = interpolate_h(h, n, ntry)
        h_try, Ctry = optimize_set(h_try, ntry, sub_dead, rng)
        if Ctry < C - 1e-12:
            h, C = h_try, Ctry
            n = ntry

    h, C = optimize_set(h, n, deadline, rng)
    h = project_h(h, n)
    C = c5_h(h, n)
    return h, C, n
