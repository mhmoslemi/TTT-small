def compute_c5(h,n):

def project_feasible(h,n):
    target=n*0.5
    h=np.asarray(h,dtype=float).copy()
    h=np.clip(h,0.0,1.0)
    d=target-h.sum()
    if d>1e-12:
        cap=1.0-h
        lo=0.0; hi=1.0
        for _ in range(40):
            t=(lo+hi)*0.5
            if float(np.sum(np.minimum(cap,t))) < d:
                lo=t
            else:
                hi=t
        t=(lo+hi)*0.5
        h += np.minimum(cap,t)
    elif d<-1e-12:
        cap=h
        need=-d
        lo=0.0; hi=1.0
        for _ in range(40):
            t=(lo+hi)*0.5
            if float(np.sum(np.minimum(cap,t))) < need:
                lo=t
            else:
                hi=t
        t=(lo+hi)*0.5
        h -= np.minimum(cap,t)
    r=target-h.sum()
    if abs(r)>1e-13:

            if s>1e-300: h += (r/s)*cap

            s=h.sum()

    return np.clip(h,0.0,1.0)

import numpy as np
import time


def compute_c5(h, n):
    dx = 2.0 / float(n)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)


def project_feasible(h, n):
    h = np.clip(np.array(h, dtype=float), 0.0, 1.0)
    target = n * 0.5
    d = target - float(h.sum())
    if d > 1e-14:
        room = 1.0 - h
        s = float(room.sum())
        if s > 1e-300:
            h = h + (d / s) * room
    elif d < -1e-14:
        s = float(h.sum())
        if s > 1e-300:
            h = h * (target / s)
    d = target - float(h.sum())
    if abs(d) > 1e-13:
        h = h + d / float(n)
        h = np.clip(h, 0.0, 1.0)
    return h


def cd_step(c, h, p, n, hsum, target, sum_tol):
    h0 = float(h[p])
    cm = n - 1
    alt = cm - p
    smin = p if p < alt else alt
    e1 = p + cm
    e2 = 2 * cm - p
    smax = e1 if e1 > e2 else e2

    c0 = c[smin:smax + 1].copy()
    a = np.zeros(smax - smin + 1, dtype=float)

    st = p if p > smin else smin
    en = p + cm if p + cm < smax else smax
    if st <= en:
        q0 = st - p
        q1 = en - p
        hi = cm - q0
        lo = cm - q1
        a[st - smin:en - smin + 1] += 1.0 - h[lo:hi + 1][::-1]

    st = cm - p if cm - p > smin else smin
    en = 2 * cm - p if 2 * cm - p < smax else smax
    if st <= en:
        i0 = st - cm + p
        i1 = en - cm + p
        a[st - smin:en - smin + 1] -= h[i0:i1 + 1]

    co = cm - smin
    c_center = float(c0[co])
    a[co] = 0.0

    eps = 1e-14
    pos = a > eps
    neg = a < -eps
    zero = ~pos & ~neg
    zero[co] = False

    pc = c0[pos]
    pa = a[pos]
    nc = c0[neg]
    na = a[neg]
    zc = c0[zero]
    zmax = float(np.max(zc)) if zc.size else -1e300

    unaff = -1e300
    if smin > 0:
        m = float(np.max(c[:smin]))
        if m > unaff:
            unaff = m
    if smax < 2 * n - 2:
        m = float(np.max(c[smax + 1:]))
        if m > unaff:
            unaff = m
    if unaff < -1e299:
        unaff = 0.0

    c0max = float(np.max(c0))
    oldmax = c0max if c0max > unaff else unaff

    err = float(hsum) - float(target)
    tol = sum_tol
    if tol < abs(err):
        tol = abs(err) + 1e-12
    slo = h0 - tol - err
    shi = h0 + tol - err

    loL = unaff
    hiL = oldmax
    if hiL - loL > 1e-14:
        for _ in range(24):
            L = (loL + hiL) * 0.5
            xlo = 0.0
            xhi = 1.0
            feasible = True

            if zc.size and zmax > L + 1e-13:
                feasible = False

            if feasible and pc.size:
                v = float(np.min((L - pc) / pa + h0))
                if v < xhi:
                    xhi = v

            if feasible and nc.size:
                v = float(np.max((L - nc) / na + h0))
                if v > xlo:
                    xlo = v

            if xlo < slo:
                xlo = slo
            if xhi > shi:
                xhi = shi

            xl = xlo if xlo > 0.0 else 0.0
            xh = xhi if xhi < 1.0 else 1.0
            if xl > xh + 1e-12:
                feasible = False

            if feasible:
                Cq = L - c_center + h0 * (1.0 - h0)
                if Cq < -1e-13:
                    feasible = False
                elif Cq < 0.25 - 1e-13:
                    disc = 1.0 - 4.0 * Cq
                    if disc < 0.0:
                        disc = 0.0
                    r1 = (1.0 - np.sqrt(disc)) * 0.5
                    r2 = 1.0 - r1
                    if xl > r1 + 1e-12 and xh < r2 - 1e-12:
                        feasible = False

            if feasible:
                hiL = L
            else:
                loL = L

    L = hiL
    xlo = 0.0
    xhi = 1.0

    if pc.size:
        v = float(np.min((L - pc) / pa + h0))
        if v < xhi:
            xhi = v
    if nc.size:
        v = float(np.max((L - nc) / na + h0))
        if v > xlo:
            xlo = v

    if xlo < slo:
        xlo = slo
    if xhi > shi:
        xhi = shi

    xl = max(0.0, xlo)
    xh = min(1.0, xhi)
    if xl > xh + 1e-12:
        return oldmax, 0.0, False

    Cq = L - c_center + h0 * (1.0 - h0)
    if Cq >= 0.25 - 1e-13:
        if h0 < xl:
            x = xl
        elif h0 > xh:
            x = xh
        else:
            x = h0
    else:
        if Cq < -1e-13:
            return oldmax, 0.0, False
        disc = 1.0 - 4.0 * Cq
        if disc < 0.0:
            disc = 0.0
        r1 = (1.0 - np.sqrt(disc)) * 0.5
        r2 = 1.0 - r1

        best = h0
        bestd = 1e100

        A = xl
        B = xh if xh < r1 else r1
        if A <= B + 1e-12:
            cand = A if h0 < A else (B if h0 > B else h0)
            dval = abs(cand - h0)
            if dval < bestd:
                bestd = dval
                best = cand

        A = xl if xl > r2 else r2
        B = xh
        if A <= B + 1e-12:
            cand = A if h0 < A else (B if h0 > B else h0)
            dval = abs(cand - h0)
            if dval < bestd:
                bestd = dval
                best = cand

        x = best

    x = float(np.clip(x, 0.0, 1.0))
    delta = x - h0
    if abs(delta) < 1e-13:
        return oldmax, 0.0, False

    cnew = c0 + a * delta
    cnew[co] = c_center + x * (1.0 - x) - h0 * (1.0 - h0)
    nneg = cnew < 0.0
    if np.any(nneg):
        cnew[nneg] = 0.0

    newmax = float(np.max(cnew))
    if unaff > newmax:
        newmax = unaff

    if newmax > oldmax + 1e-8:
        return oldmax, 0.0, False

    c[smin:smax + 1] = cnew
    h[p] = x
    return newmax, delta, True


def cd_polish(h, n, deadline, random_order=False, rng=None, sym=False, sum_tol=None):
    if time.time() >= deadline:
        h = project_feasible(h, n)
        return h, compute_c5(h, n)

    h = project_feasible(h, n)
    target = n * 0.5
    if sum_tol is None:
        sum_tol = max(1.0, 0.004 * n)

    c = np.correlate(h, 1.0 - h, mode="full")
    cur = float(c.max())
    best_h = h.copy()
    best_raw = cur
    best_c = cur * 2.0 / float(n)
    hsum = target

    while time.time() < deadline:
        if random_order and rng is not None:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

        changed_pass = False
        for idx in range(n):
            if time.time() > deadline:
                break
            p = int(order[idx])
            newmax, delta, changed = cd_step(c, h, p, n, hsum, target, sum_tol)
            if changed:
                hsum += delta
                cur = newmax
                changed_pass = True

        if sym:
            h = 0.5 * (h + h[::-1])
        h = project_feasible(h, n)
        c = np.correlate(h, 1.0 - h, mode="full")
        cur = float(c.max())
        hsum = target
        c5 = cur * 2.0 / float(n)

        if c5 < best_c - 1e-12:
            best_c = c5
            best_raw = cur
            best_h = h.copy()

        if not changed_pass:
            break

        if c5 > best_c + 1e-4:
            h = best_h.copy()
            c = np.correlate(h, 1.0 - h, mode="full")
            cur = best_raw
            hsum = target

    best_h = project_feasible(best_h, n)
    return best_h, compute_c5(best_h, n)


def interp_h(h_old, n_new):
    n_old = len(h_old)
    if n_old <= 0:
        return project_feasible(np.full(n_new, 0.5), n_new)
    if n_old == n_new:
        return project_feasible(h_old.copy(), n_new)
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    return project_feasible(np.interp(x_new, x_old, h_old), n_new)


def smooth_h(h, w):
    n = len(h)
    if w <= 1:
        return project_feasible(h.copy(), n)
    ker = np.ones(w) / float(w)
    out = np.convolve(h, ker, mode="same")
    return project_feasible(out, n)


def make_start(n, rng):
    t = np.arange(n, dtype=float)
    k = max(3, n // 25)
    h = 0.5
    h += 0.25 * np.sin(2.0 * np.pi * t / float(k))
    h += 0.06 * np.sin(2.0 * np.pi * t / 7.0 + rng.uniform(0.0, 6.0))
    h += 0.04 * rng.normal(0.0, 1.0, n)
    return project_feasible(h, n)
