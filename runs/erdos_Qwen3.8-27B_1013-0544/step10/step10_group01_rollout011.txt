def smooth_grad(x,n,alpha):
    c=np.correlate(x,1-x,'full')
    m=c.max()
    shifted=alpha*(c-m)
    shifted-=shifted.max()
    p=np.exp(shifted)
    s=p.sum()
    if s<=0: p[np.argmax(c)]=1
    else: p/=s
    g=np.zeros(n)
    center=n-1
    for l in range(1,n):
       wp=p[center-l]
       if wp>1e-18:
          g[:n-l] += wp*(1-x[l:])
          g[l:] -= wp*x[:-l]
       wn=p[center+l]
       if wn>1e-18:
          g[:n-l] -= wn*x[l:]
          g[l:] += wn*(1-x[:-l])
    w0=p[center]
    if w0>1e-18:
       g += w0*(1-2*x)
    return (2.0/n)*g

def project_feasible(h,n):
    h=np.clip(np.asarray(h,dtype=float).copy(),0.0,1.0)
    target=n/2.0
    for _ in range(200):
        diff=target-h.sum()
        if abs(diff)<1e-13: break
        if diff>0:
            room=1.0-h
            tot=room.sum()
            if tot<1e-14: break
            h=np.clip(h + (diff/tot)*room,0.0,1.0)
        else:
            tot=h.sum()
            if tot<1e-14: break
            h=np.clip(h + (diff/tot)*h,0.0,1.0)
    # final exact-ish correction
    diff=target-h.sum()
    if abs(diff)>1e-15:
       # distribute to elements with room
       if diff>0:
          room=1.0-h
          idx=np.where(room>1e-12)[0]
          if len(idx): h[idx]=np.clip(h[idx]+diff/len(idx),0,1)
       else:
          idx=np.where(h>1e-12)[0]
          if len(idx): h[idx]=np.clip(h[idx]+diff/len(idx),0,1)
    return h
```

import numpy as np
import time
from scipy.optimize import minimize, linprog

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(120):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    diff = target - h.sum()
    if diff > 1e-14:
        idx = np.where(h < 1.0 - 1e-12)[0]
        if idx.size:
            h[idx] = np.clip(h[idx] + diff / idx.size, 0.0, 1.0)
    elif diff < -1e-14:
        idx = np.where(h > 1e-12)[0]
        if idx.size:
            h[idx] = np.clip(h[idx] + diff / idx.size, 0.0, 1.0)
    return h

def symmetrize_feasible(h, n):
    return project_feasible(0.5 * (h + h[::-1]), n)

def downsample(h, nin, nout):
    if nin == nout:
        return project_feasible(np.asarray(h, dtype=float), nout)
    x0 = np.linspace(0.0, 2.0, nin)
    x1 = np.linspace(0.0, 2.0, nout)
    return project_feasible(np.interp(x1, x0, np.asarray(h, dtype=float)), nout)

def make_cos(n):
    x = np.linspace(0.0, 2.0, n)
    return project_feasible(0.5 + 0.5 * np.cos(np.pi * x), n)

def make_ramp(n):
    x = np.linspace(0.0, 2.0, n)
    return project_feasible(np.clip(1.0 - np.abs(x - 1.0), 0.0, 1.0), n)

def random_start(n, seed, scale):
    rng = np.random.RandomState(seed)
    return project_feasible(np.full(n, 0.5) + scale * rng.normal(size=n), n)

def make_binary_sym(n, seed):
    rng = np.random.RandomState(seed)
    m = n // 2
    target = n // 4
    x = (rng.rand(m) > 0.5).astype(float)
    cur = int(x.sum())
    if cur > target:
        idx = rng.choice(np.flatnonzero(x), size=cur - target, replace=False)
        x[idx] = 0.0
    elif cur < target:
        idx = rng.choice(np.flatnonzero(1.0 - x), size=target - cur, replace=False)
        x[idx] = 1.0
    if n % 2 == 0:
        h = np.empty(n, dtype=float)
        h[:m] = x
        h[m:] = x[::-1]
    else:
        h = np.empty(n, dtype=float)
        h[:m] = x
        h[m] = 0.5
        h[m + 1:] = x[::-1]
    return project_feasible(h, n)

def _f_lin(c_aff, a, unaff, d):
    val = float(np.max(c_aff + d * a))
    return val if val > unaff else unaff

def _f_quad(c_aff, a, b, unaff, d):
    val = float(np.max(c_aff + d * a + d * d * b))
    return val if val > unaff else unaff

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    sh = alpha * (c - m)
    mx = float(sh.max())
    return dx * (m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)

def smooth_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    sh = alpha * (c - m)
    mx = float(sh.max())
    p = np.exp(sh - mx)
    s = p.sum()
    if s <= 0.0:
        p = np.zeros_like(p)
        p[int(np.argmax(c))] = 1.0
    else:
        p /= s
    g = np.zeros(n, dtype=float)
    center = n - 1
    active = np.flatnonzero(p > 1e-12)
    for idx in active:
        w = float(p[idx])
        if idx < center:
            l = center - idx
            g[:n - l] += w * (1.0 - x[l:])
            g[l:] -= w * x[:-l]
        elif idx > center:
            l = idx - center
            g[:n - l] -= w * x[l:]
            g[l:] += w * (1.0 - x[:-l])
        else:
            g += w * (1.0 - 2.0 * x)
    return dx * g

def _sum_fun(x, n):
    return x.sum() - n / 2.0

def _sum_jac(x, n):
    return np.ones(n, dtype=float)

def slsqp_smooth(h, n, alphas, maxiter, deadline):
    best_h = project_feasible(np.asarray(h, dtype=float), n)
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (n,)}]
    for alpha in alphas:
        if time.time() > deadline - 25:
            break
        try:
            res = minimize(smooth_obj, best_h, args=(n, float(alpha)), jac=smooth_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': int(maxiter), 'ftol': 1e-13})
            if res.x is not None:
                hh = project_feasible(res.x, n)
                cc = compute_c5(hh, n)
                if cc < best_c:
                    best_h, best_c = hh, cc
        except Exception:
            pass
    return best_h, best_c

def lp_step(h, n, active_count):
    h = np.asarray(h, dtype=float)
    c = np.correlate(h, 1.0 - h, mode='full')
    base_c = float(c.max()) * (2.0 / n)
    m = min(int(active_count), len(c))
    active = np.argsort(c)[-m:][::-1]
    A = np.zeros((m, n + 1), dtype=float)
    b = np.zeros(m, dtype=float)
    center = n - 1
    for r, idx in enumerate(active):
        row = A[r]
        if idx < center:
            l = center - idx
            row[:n - l] += 1.0 - h[l:]
            row[l:] -= h[:-l]
        elif idx > center:
            l = idx - center
            row[:n - l] -= h[l:]
            row[l:] += 1.0 - h[:-l]
        else:
            row[:n] = 1.0 - 2.0 * h
        row[n] = -1.0
        b[r] = -float(c[idx])
    A_eq = np.ones((1, n + 1), dtype=float)
    A_eq[0, n] = 0.0
    lo = np.where(h < 1e-12, 0.0, -h)
    hi = np.where(h > 1.0 - 1e-12, 0.0, 1.0 - h)
    bnds = list(zip(lo, hi)) + [(-1e9, 1e9)]
    obj = np.zeros(n + 1, dtype=float)
    obj[n] = 1.0
    try:
        try:
            res = linprog(obj, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=[0.0],
                          bounds=bnds, method='highs')
        except Exception:
            res = linprog(obj, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=[0.0],
                          bounds=bnds, method='simplex')
        if not res.success or res.x is None:
            return h, base_c
        d = res.x[:n]
        if not np.isfinite(d).all() or np.allclose(d, 0.0):
            return h, base_c
    except Exception:
        return h, base_c
    raw_h = h.copy()
    raw_c = base_c
    for lam in np.linspace(0.0, 1.0, 32):
        hh = h + lam * d
        if hh.min() < -1e-9 or hh.max() > 1.0 + 1e-9:
            continue
        hh = np.clip(hh, 0.0, 1.0)
        cc = compute_c5(hh, n)
        if cc < raw_c:
            raw_c = cc
            raw_h = hh.copy()
    out = project_feasible(raw_h, n)
    cc = compute_c5(out, n)
    if cc < base_c:
        return out, cc
    return h, base_c

def coord_sym_cd(h, n, max_passes, deadline, gs_iters=35):

            i1 = p - k

            a[mk] += 1.0 - h[i1[mk]]
            i2 = p + k
            mk = (i2 >= 0) & (i2 < n)
            a[mk] -= h[i2[mk]]
            i1 = q - k
            mk = (i1 >= 0) & (i1 < n)
            a[mk] -= 1.0 - h[i1[mk]]
            i2 = q + k
            mk = (i2 >= 0) & (i2 < n)
            a[mk] += h[i2[mk]]
            b = np.zeros(ms.size, dtype=float)
            D = q - p
            ip = center - D
            if mlo <= ip <= mhi:

            iq = center + D
            if mlo <= iq <= mhi:
                b[iq - mlo] += 1.0
            if mlo <= center <= mhi:

            c_aff = c[ms]

                    d1 = hi - gr * (hi - lo)
                    f1 = _f_lin(c_aff, a, unaff, d1)
                else:

                    d2 = lo + gr * (hi - lo)
                    f2 = _f_lin(c_aff, a, unaff, d2)
            dlin = 0.5 * (lo + hi)
            span = hi - lo
            cand = [0.0, dlin, lo, hi,
                    dlin - 0.5 * span, dlin + 0.5 * span,
                    lo + 0.25 * span, lo + 0.75 * span]
            for t in range(8):
                cand.append(lo + (hi - lo) * (t / 8.0))
            best_val = np.inf
            best_d = 0.0
            for dd in cand:
                if dd < lo - 1e-14 or dd > hi + 1e-14:
                    continue
                val = _f_quad(c_aff, a, b, unaff, dd)
                if val < best_val:
                    best_val = val
                    best_d = dd
            if best_val < max_c - 1e-11:
                h[p] = A0 + best_d
                h[q] = B0 - best_d
                c[ms] += best_d * a + best_d * best_d * b
                max_c = best_val
                improved = True
                if dx * max_c < best_c - 1e-15:
                    best_c = dx * max_c
                    best_h = h.copy()
        if not improved:
            break
        h = 0.5 * (h + h[::-1])
        h = project_feasible(h, n)
        c = np.correlate(h, 1.0 - h, mode='full')
        max_c = float(c.max())
        if dx * max_c < best_c:
            best_c = dx * max_c
            best_h = h.copy()
        else:
            h = best_h.copy()
            c = np.correlate(h, 1.0 - h, mode='full')
            max_c = float(c.max())
    return best_h, best_c

def optimize_candidate(h, n, deadline, do_slsqp=True, do_lp=True, do_cd=True,
                       slsqp_alphas=None, slsqp_maxiter=70,
                       lp_iters=8, cd_passes=18, active_count=90):

        h2, c2 = slsqp_smooth(best_h, n, slsqp_alphas, slsqp_maxiter, deadline)
        if c2 < best_c:

        if cs < best_c:
            best_h, best_c = hs, cs
    if do_cd and (n % 2 == 0) and time.time() < deadline - 25:
        h2, c2 = coord_sym_cd(best_h, n, cd_passes, deadline,
                              gs_iters=35 if n < 520 else 30)
        if c2 < best_c:
            best_h, best_c = h2, c2
        hs = symmetrize_feasible(best_h, n)
        cs = compute_c5(hs, n)
        if cs < best_c:
            best_h, best_c = hs, cs
    if do_lp and lp_iters > 0 and time.time() < deadline - 25:
        for _ in range(min(6, int(lp_iters))):
            if time.time() > deadline - 18:
                break
            h2, c2 = lp_step(best_h, n, active_count)
            if c2 < best_c - 1e-12:
                best_h, best_c = h2, c2
            else:
                break
        hs = symmetrize_feasible(best_h, n)
        cs = compute_c5(hs, n)
        if cs < best_c:
            best_h, best_c = hs, cs
    return best_h, compute_c5(best_h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + max(60.0, float(budget_s)) - 10.0
    try:
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = None
    if h0 is None or h0.size == 0:
        h0 = make_cos(320)
        n0 = 320
    else:
        n0 = int(len(h0))
    best_h = project_feasible(h0, n0)
    best_n = n0
    best_c = compute_c5(best_h, best_n)
    hs = symmetrize_feasible(best_h, best_n)
    cs = compute_c5(hs, best_n)
    if cs < best_c:
        best_h, best_c = hs, cs

    if time.time() < deadline - 120:
        if best_n > 550:
            h2, c2 = optimize_candidate(best_h, best_n, deadline - 120,
                                        do_slsqp=True,
                                        slsqp_alphas=[20000],
                                        slsqp_maxiter=35,
                                        do_lp=True, lp_iters=5,
                                        do_cd=True, cd_passes=8,
                                        active_count=55)
        else:
            h2, c2 = optimize_candidate(best_h, best_n, deadline - 120,
                                        do_slsqp=True,
                                        slsqp_alphas=[3000, 12000],
                                        slsqp_maxiter=80,
                                        do_lp=True, lp_iters=8,
                                        do_cd=True, cd_passes=20,
                                        active_count=100)
        if c2 < best_c:
            best_h, best_c, best_n = h2, c2, len(h2)

    if time.time() < deadline - 270:
        if best_n >= 500:
            n_small = 320
        else:
            n_small = max(128, best_n // 2)
        if n_small % 2:
            n_small += 1
        base = downsample(best_h, best_n, n_small)
        candidates = [
            base,
            symmetrize_feasible(base, n_small),
            make_binary_sym(n_small, seed + 11),
            random_start(n_small, seed + 13, 0.055),
            make_cos(n_small)
        ]
        max_starts = 4 if n_small <= 340 else 3
        for st in candidates[:max_starts]:
            if time.time() > deadline - 65:
                break
            h2, c2 = optimize_candidate(st, n_small, deadline - 55,
                                        do_slsqp=True,
                                        slsqp_alphas=[2000, 8000, 30000],
                                        slsqp_maxiter=70,
                                        do_lp=True, lp_iters=10,
                                        do_cd=True, cd_passes=24,
                                        active_count=110)
            if c2 < best_c:
                best_h, best_c, best_n = h2, c2, int(len(h2))

    if time.time() < deadline - 230 and best_n > 320:
        n_med = 480 if best_n >= 600 else 400
        if n_med % 2:
            n_med += 1
        base = downsample(best_h, best_n, n_med)
        candidates = [base, symmetrize_feasible(base, n_med), make_binary_sym(n_med, seed + 23)]
        max_starts = 2 if n_med >= 480 else 3
        for st in candidates[:max_starts]:
            if time.time() > deadline - 70:
                break
            h2, c2 = optimize_candidate(st, n_med, deadline - 60,
                                        do_slsqp=True,
                                        slsqp_alphas=[4000, 16000],
                                        slsqp_maxiter=70,
                                        do_lp=True, lp_iters=8,
                                        do_cd=True, cd_passes=16,
                                        active_count=90)
            if c2 < best_c:
                best_h, best_c, best_n = h2, c2, int(len(h2))

    if time.time() < deadline - 35:
        h2, c2 = optimize_candidate(best_h, best_n, deadline,
                                    do_slsqp=False,
                                    do_lp=True,
                                    lp_iters=12 if best_n < 600 else 8,
                                    do_cd=True,
                                    cd_passes=22 if best_n < 600 else 12,
                                    active_count=110 if best_n < 600 else 60)
        if c2 < best_c:
            best_h, best_c, best_n = h2, c2, int(len(h2))

    if time.time() < deadline - 15:
        hs = symmetrize_feasible(best_h, best_n)
        cs = compute_c5(hs, best_n)
        if cs < best_c:
            best_h, best_c = hs, cs

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, int(best_n)
