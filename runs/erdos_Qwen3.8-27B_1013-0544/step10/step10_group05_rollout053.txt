        if m >= 0:
            # p ranges from m to n-1
            g[m:] += (1.0 - h[:n-m])
        else:
            # p ranges from 0 to n-1+m
            g[:n+m] += (1.0 - h[-m:])
        # Term 2: -h[p+m] for p+m in [0,n-1]
        if m >= 0:
            # p ranges from 0 to n-1-m
            g[:n-m] -= h[m:]
        else:
            # p ranges from -m to n-1
            g[-m:] -= h[:n+m]
    g /= len(active_lags)

def project_feasible(h, n):
    target = n / 2.0

        diff = target - h.sum()
        if abs(diff) < 1e-15:

        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:

def solve_coord(c, h, p, n):

    m_min = max(0, n - 1 - p)
    m_max = min(2 * n - 2, 2 * n - 2 - p)

    jb = p - ms + (n - 1)

    vb = (jb >= 0) & (jb < n)

    if np.any(vb):

    g = np.zeros(n)
    for m in active_ms:
        if m >= 0:
            if m < n:
                g[m:] += (1.0 - h[:n - m])
                g[:n - m] -= h[m:]
        else:
            mm = -m
            if mm < n:
                g[:n - mm] += (1.0 - h[mm:])
                g[mm:] -= h[:n - mm]
    g /= len(active_ms)
    return g

            h = h_new
            step = min(step * 1.1, 0.1)
        else:

            if step < 1e-6:
                break
    return best_h, best_c

For a fixed p, min_h ∑_k p_k c_k is a non-convex problem (because c_k is bilinear in h). But it's a quadratic minimization with constraints.

This is a quadratic form in h. For fixed p, this is a QP (quadratic program) with linear constraints.
