    # Compute subgradient
    grad = np.zeros(n)
    for k in np.where(active)[0]:
        shift = k - (n-1)
        # g_k(h) = sum_i h[i] * (1-h[i+shift]) * dx (for valid i)
        # dg_k/dh[i] = (1-h[i+shift]) - h[i-shift] (with appropriate bounds)
        for i in range(n):

            j_bwd = i - shift

            if 0 <= j_bwd < n:

    h_new = project_feasible(h - step_size * grad, n)
    return h_new

        grad += fwd - bwd

    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
