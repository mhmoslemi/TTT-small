import numpy as np
import time


def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)


def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h


def binary_c(h, n):
    hi = h.astype(np.int16)
    ci = (1 - hi).astype(np.int16)
    return np.correlate(hi, ci, mode="full")[::-1].astype(np.int16)


def make_binary(p, seed):
    n = len(p)
    T = n // 2
    rng = np.random.RandomState(seed)
    order = np.argsort(-(p + 1e-7 * rng.random(n)))
    h = np.zeros(n, dtype=np.int8)
    h[order[:T]] = 1
    return h


def rand_onezero(h, n, rng):
    for _ in range(25):
        i = int(rng.randint(n))
        j = int(rng.randint(n))
        if i != j:
            if h[i] != h[j]:
                if h[i] == 1:
                    return j, i
                return i, j
    return None


def eval_swap(c, comp, h_rev, a, b, o, n, L, delta, tmp):
    delta.fill(0)
    delta[o - a:o - a + n] += comp
    delta[a:a + n] -= h_rev
    delta[b:b + n] += h_rev
    delta[o - b:o - b + n] -= comp
    delta[o] -= 2
    ip = b - a
    delta[o + ip] += 1
    delta[o - ip] += 1
    np.add(c, delta, out=tmp)
    return int(tmp.max())


def local_search(n, h0, deadline, seed):
    T = n // 2
    h = h0.astype(np.int8).copy()
    if h.sum() != T:
        h = make_binary(h.astype(float), seed)
    c = binary_c(h, n)
    M = int(c.max())
    o = n - 1
    L = 2 * n - 1
    comp = (1 - h).astype(np.int8)
    h_rev = h[::-1]
    delta = np.zeros(L, dtype=np.int16)
    tmp = np.empty(L, dtype=np.int16)
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_M = M
    active = None
    stall = 0
    pk = 2
    accepted = 0

    while time.time() < deadline:
        if active is None or rng.random() < 0.03:
            act = np.where(c == M)[0]
            if act.size < 5:
                act = np.where(c >= M - 1)[0]
            active = act

        cand = None
        if active.size and rng.random() < 0.65:
            idx = int(active[int(rng.randint(active.size))])
            d = idx - o
            if d != 0:
                other = 2 * o - idx
                if c[other] > c[idx] and rng.random() < 0.8:
                    d_use = -d
                else:
                    d_use = d
                m = abs(d_use)
                hi = n - m
                if hi > 0:
                    for _ in range(6):
                        pos = int(rng.randint(0, hi))
                        if d_use > 0:
                            if h[pos] == 1 and h[pos + m] == 0:
                                cand = (pos + m, pos)
                                break
                        else:
                            if h[pos + m] == 1 and h[pos] == 0:
                                cand = (pos, pos + m)
                                break
                    if cand is None:
                        if d_use > 0:
                            mask = (h[:hi] == 1) & (h[m:] == 0)
                            if mask.any():
                                pos = int(np.flatnonzero(mask)[int(rng.randint(mask.sum()))])
                                cand = (pos + m, pos)
                        else:
                            mask = (h[m:] == 1) & (h[:hi] == 0)
                            if mask.any():
                                pos = int(np.flatnonzero(mask)[int(rng.randint(mask.sum()))])
                                cand = (pos, pos + m)

        if cand is None:
            cand = rand_onezero(h, n, rng)

        if cand is None:
            stall += 20
        else:
            a, b = cand
            if h[b] == 1 and h[a] == 0:
                newM = eval_swap(c, comp, h_rev, a, b, o, n, L, delta, tmp)
                if newM < M:
                    h[a] = 1
                    h[b] = 0
                    comp[a] = 0
                    comp[b] = 1
                    np.copyto(c, tmp)
                    M = newM
                    active = None
                    stall = 0
                    pk = 2
                    accepted += 1
                    if M < best_M:
                        best_M = M
                        best_h = h.copy()
                elif newM == M and rng.random() < 0.25:
                    h[a] = 1
                    h[b] = 0
                    comp[a] = 0
                    comp[b] = 1
                    np.copyto(c, tmp)
                    M = newM
                    active = None
                    stall += 1
                    accepted += 1
                else:
                    stall += 1
            else:
                stall += 1

        if stall > 4000 or (M - best_M > 8 and rng.random() < 0.1):
            if M > best_M + 4 or rng.random() < 0.3:
                h = best_h.copy()
                c = binary_c(h, n)
                M = int(c.max())
            else:
                for _ in range(pk):
                    rp = rand_onezero(h, n, rng)
                    if rp is not None:
                        a, b = rp
                        h[a] = 1
                        h[b] = 0
                c = binary_c(h, n)
                M = int(c.max())
                pk = min(60, pk * 2)
            comp = (1 - h).astype(np.int8)
            h_rev = h[::-1]
            active = None
            stall = 0
            if M < best_M:
                best_M = M
                best_h = h.copy()

        if accepted and accepted % 4000 == 0:
            c2 = binary_c(h, n)
            M2 = int(c2.max())
            if M2 < M - 1 or M2 > M + 1:
                c = c2
                M = M2
            comp = (1 - h).astype(np.int8)
            h_rev = h[::-1]

    return best_h, best_M


def explore_n(n, base, deadline, seed):
    best_h = None
    best_M = 10**9

    if base is not None:
        if base.size == n:
            base_n = base.astype(float).copy()
        else:
            xold = np.linspace(0, 2, base.size)
            xnew = np.linspace(0, 2, n)
            base_n = np.interp(xnew, xold, base.astype(float))
    else:
        base_n = np.full(n, 0.5)

    count = 0
    while time.time() < deadline - 2:
        rem = deadline - time.time()
        if rem < 2:
            break
        rngp = np.random.RandomState(seed + count * 97)

        if count == 0:
            p = base_n.copy()
        else:
            mode = count % 4
            if mode == 1:
                p = base_n + 0.15 * rngp.normal(size=n)
            elif mode == 2:
                i = np.arange(n, dtype=float)
                lam = max(3, n // 3 + 1)
                s = 0.5 + 0.3 * np.sin(2.0 * np.pi * i / lam + rngp.uniform(0, 6.0))
                p = 0.55 * base_n + 0.45 * (s + 0.05 * rngp.normal(size=n))
            elif mode == 3:
                i = np.arange(n, dtype=float)
                l1 = max(3, n // 5 + 1)
                l2 = max(l1 + 2, n // 2)
                s = (0.5
                     + 0.25 * np.sin(2.0 * np.pi * i / l1 + rngp.uniform(0, 6.0))
                     + 0.20 * np.sin(2.0 * np.pi * i / l2 + rngp.uniform(0, 6.0)))
                p = 0.4 * (s + 0.05 * rngp.normal(size=n)) + 0.6 * base_n
            else:
                p = rngp.random(n)
            p = np.clip(p, 0.0, 1.0)

        h0 = make_binary(p, seed + count * 131)
        if count == 0:
            loc_dl = min(deadline, time.time() + max(8, min(60, 0.6 * rem)))
        else:
            loc_dl = min(deadline, time.time() + max(4, min(25, 0.25 * rem)))

        hloc, Mloc = local_search(n, h0, loc_dl, seed + 1000 + count)
        if hloc is not None and Mloc < best_M:
            best_M = Mloc
            best_h = hloc.copy()
        count += 1

    if best_h is None:
        return None, 1.0
    return best_h, float(best_M * (2.0 / n))


def refine_cont(h0, n, deadline, noise=0.0, seed=1):
    x = project_feasible(h0.astype(float), n)
    if noise > 0:
        rng = np.random.RandomState(seed)
        x = project_feasible(x + rng.normal(0.0, noise, n), n)

    best_x = x.copy()
    best_c = compute_c5(x, n)
    o = n - 1
    L = 2 * n - 1

    for _ in range(80):
        if time.time() > deadline:
            break
        h = x.copy()
        c = np.correlate(h, 1.0 - h, mode="full")[::-1].astype(float)
        m_best = float(c.max())
        improved = False
        h_rev = h[::-1]
        comp = 1.0 - h

        for p in range(n):
            if time.time() > deadline:
                break
            y = float(h[p])
            if y < 1e-5 or y > 1.0 - 1e-5:
                continue
            A = np.zeros(L, dtype=float)
            A[o - p:o - p + n] += comp
            A[p:p + n] -= h_rev

            yy = np.array([0.0, 1.0, y,
                           y - 0.2, y - 0.1, y - 0.05,
                           y + 0.05, y + 0.1, y + 0.2], dtype=float)
            np.clip(yy, 0.0, 1.0, out=yy)
            yy = np.unique(yy)
            if yy.size <= 1:
                continue

            u = yy - y
            vals = c[None, :] + u[:, None] * A
            vals[:, o] -= u * u
            mv = vals.max(axis=1)
            j = int(mv.argmin())

            if mv[j] < m_best - 1e-13:
                xn = float(yy[j])
                du = xn - y
                h[p] = xn
                comp[p] = 1.0 - xn
                c += du * A
                c[o] -= du * du
                m_best = float(mv[j])
                improved = True

        if not improved or time.time() > deadline:
            break

        x = project_feasible(h, n)
        cc = compute_c5(x, n)
        if cc < best_c:
            best_c = cc
            best_x = x.copy()

    return best_x, best_c
