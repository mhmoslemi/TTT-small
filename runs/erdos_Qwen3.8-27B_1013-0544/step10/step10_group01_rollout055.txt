def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d):
    vals = c_aff + d * a
    if has_diag:
        vals[diag_idx] = c_aff[diag_idx] + d * (1.0 - h0 - d)
    result = float(np.max(vals))
    if unaff_max > result:
        result = unaff_max
    return result

def coord_descent(h, n, max_passes, deadline, gs_iters=80, threshold=1e-6):
    h = h.copy()
    best_c = compute_c5(h, n)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-13 or h0 >= 1.0 - 1e-13:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(valid_f):
                a[valid_f] += (1.0 - h[jf[valid_f]])
            if np.any(valid_b):
                a[valid_b] -= h[jb[valid_b]]
            diag_pos = np.where(ms == (n - 1))[0]
            has_diag = len(diag_pos) > 0
            diag_idx = int(diag_pos[0]) if has_diag else -1
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            d_lo = -h0
            d_hi = 1.0 - h0
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
            f2 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-16:
                    break
                if f1 < f2:
                    d_hi = d2
                    d2 = d1
                    f2 = f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    f1 = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
                else:

                    d1 = d2
                    f1 = f2

            d_opt = (d_lo + d_hi) / 2.0

            f_curr = _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, 0.0)
            if f_opt < f_curr - 1e-15:

                improved = True
        if improved:

            if c_new < best_c:

        else:
            break
    final_c = compute_c5(h, n)
    return h, min(best_c, final_c)

def subgradient_step(h, n, top_k=10):
    """Compute a subgradient of the max correlation at h."""
    c_raw = np.correlate(h, 1.0 - h, mode='full')
    c = c_raw * (2.0 / n)
    k_vals = np.argsort(c)[::-1][:top_k]
    # Soft weights based on correlation values
    w = c[k_vals] - c[k_vals[-1]] + 0.01
    w = w ** 3  # Sharpen: focus more on the max
    w /= w.sum()
    
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx, k in enumerate(k_vals):
        m = n - 1 - k
        if m >= 0:
            if n - m > 0:
                grad[:n-m] += w[idx] * (1.0 - h[m:])
            if m < n:
                grad[m:] -= w[idx] * h[:n-m]
        else:
            if -m < n:
                grad[-m:] += w[idx] *(1.0 - h[:n+m])
            if n + m > 0:
                grad[:n+m] -= w[idx] * h[-m:]
    return dx * grad, c.max(), k_vals

    best_c = compute_c5(h, n)

        h = h_new
        
        c_val = compute_c5(h, n)
        if c_val < best_c:
            best_c = c_val
            best_h = h.copy()

Let me try a different approach: **dual ascent / fictitious play**

The problem can be seen as a zero-sum game:
- Player 1 chooses h (to minimize the max correlation)
- Player 2 chooses k (to maximize the correlation at that lag)

In fictitious play:
- Player 2 maintains a distribution p over lags, updated by putting more weight on recently played (maximizing) lags
- Player 1 responds by minimizing the expected correlation: sum_k p_k * c_k(h)
- This inner problem is a quadratic program (since c_k is quadratic in h)

The inner problem: minimize sum_k p_k * c_k(h) subject to 0<=h<=1, sum(h)=n/2
= minimize sum_k p_k * sum_i h_i * (1-h_{i+m_k}) * dx
= minimize sum_i h_i * dx - sum_k p_k * sum_i h_i * h_{i+m_k} * dx
= 1 - dx * sum_i h_i * (sum_k p_k * h_{i+m_k})

Let g(i) = sum_k p_k * h_{i+m_k} (the "expected" shifted version)
Then we want to maximize dx * sum_i h_i * g(i)

But g depends on h! So this is still a coupled problem. However, in the alternating scheme, we fix g (based on current h) and optimize h:

Given g: maximize sum_i h_i * g_i subject to 0<=h<=1, sum(h)=n/2
This is a knapsack-like problem! The optimal solution is:
- Sort g in decreasing order
- Set h_i = 1 for the top n/2 values of g_i
- Set h_i = 0 for the rest
(Or more precisely, set h_i = 1 where g_i is largest, which gives the maximum inner product)

Wait, but that gives a 0/1 solution. The issue is that g depends on h, so we need to iterate.

This is actually a fixed-point iteration:
1. Start with h^(0)
2. Compute g^(t) from h^(t): g_i = sum_k p_k * h^(t)_{i+m_k}
3. Update h^(t+1): set h_i = 1 where g_i is in top n/2, 0 otherwise
4. Update p: p_k ∝ p_k * exp(c_k(h^(t+1)) / τ) (softmax update)
5. Repeat

This is like a variant of "best response" in a game. It might converge to a Nash equilibrium or cycle.

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d):
