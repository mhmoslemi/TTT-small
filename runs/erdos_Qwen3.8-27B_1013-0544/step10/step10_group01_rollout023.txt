def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return c * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def subgradient_step(h, n, step, n_active=5):
    """Take a subgradient step using the top n_active shifts."""
    c = np.correlate(h, 1.0 - h, mode='full')
    # Get top n_active shifts
    top_k = np.argsort(c)[-n_active:]
    grad = np.zeros(n)
    for k in top_k:
        w = c[k] / max(c.max(), 1e-30)  # weight by value
        if w < 0.01:
            continue
        # Compute gradient of c[k] w.r.t. h
        # c[k] = sum_i h[i] * (1-h[i+k-(n-1)])
        # grad[p] = (1-h[p+k-(n-1)]) - h[p-k+(n-1)] (with boundary checks)
        jm = k - (n - 1)  # offset for forward index
        jp = (n - 1) - k  # offset for backward index
        # Forward: h[p] appears with coefficient (1-h[p+jm]) when p+jm in [0,n)
        # Backward: h[p] appears with coefficient -h[p-jp] when p-jp in [0,n)
        # More precisely:
        # For c[k]: indices i where 0<=i<n and 0<=i+k-(n-1)<n
        # i ranges from max(0, n-1-k) to min(n-1, n-1+k)
        i_lo = max(0, (n-1) - k)
        i_hi = min(n-1, (n-1) + k)
        # For each i in [i_lo, i_hi], j = i + k - (n-1)
        # grad[i] gets (1-h[j])
        # grad[j] gets -h[i]
        indices = np.arange(i_lo, i_hi + 1)
        j_indices = indices + k - (n - 1)
        # Avoid double-counting when i == j (i.e., k == n-1)
        if k == n - 1:
            # c[n-1] = sum h[i](1-h[i]), grad = 1 - 2*h[i]

        else:
            grad[indices] += w * (1.0 - h[j_indices])
            grad[j_indices] -= w * h[indices]
    h_new = h - step * grad
    return project_feasible(h_new, n)

def power_sum_smooth(h, n, p):
    """Power-sum smooth max: (sum c_k^p)^(1/p) * dx"""
    c = np.correlate(h, 1.0 - h, mode='full')
    c_pos = np.maximum(c, 0)
    if p > 100:
        return (c_pos.max() + np.log(np.sum(np.exp(c_pos/c_pos.max() * (p-1)))) / p * c_pos.max()) * (2.0/n)
    return (np.sum(c_pos**p))**(1.0/p) * (2.0/n)

def cd_optimize(h, n, max_passes, deadline, gs_iters=50, threshold=1e-6):

    h = h.copy()
    best_c = compute_c5(h, n)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    dx = 2.0 / n
    
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-13 or h0 >= 1.0 - 1e-13:
                continue

            m_hi = min(2 * n - 2, n - 1 + p)

            jb = p - ms + (n - 1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(valid_f):
                a[valid_f] += (1.0 - h[jf[valid_f]])
            if np.any(valid_b):
                a[valid_b] -= h[jb[valid_b]]
            diag_pos = np.where(ms == (n - 1))[0]
            has_diag = len(diag_pos) > 0
            diag_idx = int(diag_pos[0]) if has_diag else -1
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))

                vals = c_aff + d * a
                if has_diag:
                    vals[diag_idx] = c_aff[diag_idx] + d * (1.0 - h0 - d)
                result = float(np.max(vals))
                if unaff_max > result:
                    result = unaff_max
                return result * dx
            
            d_lo = -h0
            d_hi = 1.0 - h0
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            f1 = fval(d1)
            f2 = fval(d2)
            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-16:

            h[np.where(h > 1.0 - threshold)[0]] = 1.0
            h = project_feasible(h, n)
        else:
            break
    final_c = compute_c5(h, n)

def subgrad_optimize(h, n, deadline, max_iters=20000):
    """Subgradient method with adaptive step size."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = 0.01
    patience = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        h_new = subgradient_step(h, n, step)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_h = h_new.copy()
            best_c = c_new
            h = h_new
            patience = 0
            step = min(step * 1.05, 0.05)
        else:
            h = h_new  # still update h (momentum-like)
            patience += 1
            if patience > 50:
                step *= 0.95
                patience = 0
            if step < 1e-5:
                break
    return best_h, best_c

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def _cd_fval(c_aff, a, diag_idx, has_diag, unaff_max, h0, d):
    vals = c_aff + d * a
    if has_diag:
        vals = vals.copy()
        vals[diag_idx] = c_aff[diag_idx] + d * (1.0 - h0 - d)
    result = float(np.max(vals))
    if unaff_max > result:
        result = unaff_max
    return result

    dx = 2.0 / n

            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in range(n):
            if time.time() > deadline:

            h0 = h[p]

            a = np.zeros(len(ms))

def balanced_subgrad_step(h, n, step, top_k=8):

    c_max = c.max()
    if c_max < 1e-30:
        return h.copy()

    weights = np.zeros(n * 2 - 1)
    for k in sorted_idx:
        w = c[k] / c_max
        w = w * w
        weights[k] = w
    total_w = weights.sum()
    if total_w < 1e-30:
        return h.copy()
    weights /= total_w
    grad = np.zeros(n)
    for k in sorted_idx:
        w = weights[k]
        if w < 0.005:
            continue
        i_lo = max(0, (n - 1) - k)
        i_hi = min(n - 1, (n - 1) + k)
        if k == n - 1:

        else:

            grad[j_indices] -= w * h[indices]

def _smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    if m < 1e-30:
        return 0.0
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def _sum_eq(x, n):
    return x.sum() - n / 2.0

def _sum_eq_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alphas, maxiter_each=300):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n

    for alpha in alphas:

        except Exception:
