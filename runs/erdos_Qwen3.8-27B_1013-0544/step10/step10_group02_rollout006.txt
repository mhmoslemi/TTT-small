def _finish_projection(x, target):

def project_sum(h, target):

def add_candidate(pairs, seen, p, q):
    p = int(p)
    q = int(q)
    if p == q:
        return
    key = p * 20000 + q
    if key not in seen:
        seen.add(key)
        pairs.append((p, q))

def candidate_pairs(h, c, n, rng, count):
    pairs = []
    seen = set()
    center = n - 1
    topm = min(24, len(c))
    order = np.argsort(c)[-topm:]
    for k in order:
        s = int(k - center)
        if s == 0:
            continue
        if s > 0:
            L = n - s
            if L <= 0:
                continue
            terms = h[s:s + L] * (1.0 - h[:L])
            start = 0
        else:
            t = -s
            L = n - t
            if L <= 0:
                continue
            terms = h[:L] * (1.0 - h[t:t + L])
            start = t
        if L <= 6:
            idxs = np.arange(L)
            rng.shuffle(idxs)
        else:
            kp = L - 8 if L > 8 else 0
            idxs = np.argpartition(terms, kp)[kp:] if kp > 0 else np.arange(L)
            if L > 24:
                idxs = np.concatenate((idxs, [rng.randint(L)]))
        for pos in idxs[:5]:
            base = start + int(pos)
            shifted = base + s
            if 0 <= base < n and 0 <= shifted < n:
                add_candidate(pairs, seen, base, shifted)
                if rng.rand() < 0.3:
                    add_candidate(pairs, seen, shifted, base)
    high = np.flatnonzero(h > 0.5)
    low = np.flatnonzero(h < 0.5)
    for _ in range(max(0, count // 4)):
        if high.size and low.size:
            p = int(low[rng.randint(low.size)])
            q = int(high[rng.randint(high.size)])
            add_candidate(pairs, seen, p, q)
            if rng.rand() < 0.35:
                add_candidate(pairs, seen, q, p)
        else:
            p = int(rng.randint(n))
            q = int(rng.randint(n))
            add_candidate(pairs, seen, p, q)
    fills = 0
    while len(pairs) < count and fills < 10 * count:
        p = int(rng.randint(n))
        q = int(rng.randint(n))
        add_candidate(pairs, seen, p, q)
        fills += 1
    return pairs[:count]

def pair_search(h, c, deadline, rng, pool_size=80):
    n = len(h)
    h = project_sum(h, n / 2.0)
    c = corr_full(h)
    cur = float(np.max(c))
    best_h = h.copy()
    best_val = cur
    pool = candidate_pairs(h, c, n, rng, pool_size)
    pos = 0
    no_imp = 0
    acc = 0
    while time.time() < deadline:
        if pos >= len(pool):
            pool = candidate_pairs(h, c, n, rng, pool_size)
            pos = 0
            no_imp = 0
            if len(pool) == 0:
                break
        p, q = pool[pos]
        pos += 1
        h, c, imp = attempt_pair(h, c, p, q)
        if imp:
            cur = float(np.max(c))
            acc += 1
            no_imp = 0
            if cur < best_val - 1e-12:
                best_val = cur
                best_h = h.copy()
            if acc % 25 == 0:
                c = corr_full(h)
                cur = float(np.max(c))
                if cur < best_val - 1e-12:
                    best_val = cur
                    best_h = h.copy()
        else:
            no_imp += 1
        if no_imp > 3 * pool_size:
            break
    c = corr_full(h)
    cur = float(np.max(c))
    if cur <= best_val + 1e-10:
        best_h = h.copy()
        best_val = cur
    return best_h, best_val

def interp_h(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_sum(h_new, n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    bmax = max(1.0, float(budget_s))
    deadline = t0 + bmax - 3.0
    rng = np.random.RandomState(seed)
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        n = 700
        x = np.linspace(0.0, 2.0, n, endpoint=False)
        h_best = 0.5 + 0.18 * np.sin(4.0 * np.pi * x) + 0.10 * np.sin(10.0 * np.pi * x)
    if n < 50:
        h_best = np.full(700, 0.5, dtype=float)
        n = 700
    h_best = project_sum(h_best, n / 2.0)
    best_c5 = compute_c5(h_best)

    d1 = min(deadline, t0 + 0.45 * bmax)
    h1, _ = pair_search(h_best, corr_full(h_best), d1, rng, 90)
    c5 = compute_c5(h1)
    if c5 < best_c5:
        h_best, best_c5 = h1, c5

    dp = min(deadline - 120.0, time.time() + 70.0)
    if dp > time.time() + 5.0:
        h2, _ = pair_search(h_best, corr_full(h_best), dp, rng, 110)
        c5 = compute_c5(h2)
        if c5 < best_c5:
            h_best, best_c5 = h2, c5

    base_n = len(h_best)
    nt_list = [base_n - 60, base_n + 60, base_n - 120, base_n + 120, base_n - 200, base_n + 200, 600, 800]
    seen_n = set()
    for nt in nt_list:
        nt = int(nt)
        if 250 <= nt <= 950 and nt not in seen_n and nt != len(h_best):
            seen_n.add(nt)
            if time.time() > deadline - 120.0:
                break
            dt = min(deadline - 100.0, time.time() + 55.0)
            if dt <= time.time() + 3.0:
                continue
            ht = interp_h(h_best, len(h_best), nt)
            ht, _ = pair_search(ht, corr_full(ht), dt, rng, 65)
            c5 = compute_c5(ht)
            if c5 < best_c5:
                h_best, best_c5 = ht, c5

    special_bin = True
    for sc in (0.05, 0.10, 0.02, 0.15):
        if time.time() > deadline - 70.0:
            break
        n = len(h_best)
        if special_bin:
            hp = (h_best > 0.5).astype(float)
            hp = project_sum(hp, n / 2.0)
            special_bin = False
        else:
            hp = project_sum(h_best + rng.normal(0.0, sc, n), n / 2.0)
        dt = min(deadline - 35.0, time.time() + 45.0)
        if dt <= time.time() + 3.0:
            continue
        ht, _ = pair_search(hp, corr_full(hp), dt, rng, 70)
        c5 = compute_c5(ht)
        if c5 < best_c5:
            h_best, best_c5 = ht, c5

    if time.time() < deadline - 4.0:
        hf, _ = pair_search(h_best, corr_full(h_best), deadline, rng, 130)
        c5 = compute_c5(hf)
        if c5 < best_c5:
            h_best, best_c5 = hf, c5

    h_best = project_sum(h_best, len(h_best) / 2.0)
    final_c5 = compute_c5(h_best)
    try:
        ev = float(evaluate_erdos_solution(h_best))
        if np.isfinite(ev) and ev < final_c5:
            final_c5 = ev
    except Exception:
        pass
    return h_best, final_c5, len(h_best)
