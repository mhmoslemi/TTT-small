def project_feasible(h,n):
    h=np.asarray(h,dtype=float)
    h=np.clip(h,0.0,1.0)
    target=0.5*n
    for _ in range(200):
        diff=target-h.sum()
        if abs(diff)<1e-14: break
        if diff>0:
            room=1.0-h; tot=room.sum()
            if tot<=1e-15: break
            h=np.clip(h+(diff/tot)*room,0.0,1.0)
        else:
            tot=h.sum()
            if tot<=1e-15: break
            h=np.clip(h+(diff/tot)*h,0.0,1.0)
    diff=target-h.sum()
    if abs(diff)>1e-12:
        idx=np.where((h>1e-12)&(h<1.0-1e-12))[0]
        if idx.size:
            for _ in range(100):
                step=diff/idx.size
                h[idx]=np.clip(h[idx]+step,0.0,1.0)
                diff=target-h.sum()
                if abs(diff)<1e-13: break
        else:
            if diff>0:
                idx=np.argmax(1.0-h); h[idx]=min(1.0,h[idx]+diff)
            else:
                idx=np.argmax(h); h[idx]=max(0.0,h[idx]+diff)
    return h

def project_feasible(h, n):
    h = np.asarray(h, dtype=float)
    if h.ndim != 1 or len(h) != n:
        h = np.resize(h, n).astype(float)
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    for _ in range(200):
        diff = target - float(h.sum())
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = float(room.sum())
            if tot <= 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = float(h.sum())
            if tot <= 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) > 1e-12:
        idx = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if idx.size:
            for _ in range(100):
                step = diff / float(idx.size)
                h[idx] = np.clip(h[idx] + step, 0.0, 1.0)

                best_c = cc
                best_h = hx.copy()
    hx = project_feasible(hx, n)
    cc = compute_c5(hx, n)
    if cc < best_c:
        return hx, cc
    return best_h, best_c


def smooth_value_grad(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / n
    m = float(c.max())
    z = alpha * (c - m)
    z -= z.max()
    ez = np.exp(z)
    ss = float(ez.sum())
    val = (m + (np.log(ss) - np.log(float(c.size))) / alpha) * dx
    w = ez / ss
    g = np.zeros(n)
    for l in range(c.size):
        wl = float(w[l])
        if wl <= 1e-12:
            continue
        s = n - 1 - l
        if s == 0:
            g += wl * (1.0 - 2.0 * h)
        elif s > 0:
            g[:n - s] += wl * (1.0 - h[s:n])
            g[s:] -= wl * h[:n - s]
        else:
            t = -s
            g[t:] += wl * (1.0 - h[:n - t])
            g[:n - t] -= wl * h[t:]
    return val, g * dx


def smooth_phase(h, n, deadline, rng):
    hx = project_feasible(h, n).copy()
    best_h = hx.copy()
    best_c = compute_c5(hx, n)
    for alpha in (20.0, 100.0, 500.0, 2000.0, 8000.0):
        if time.time() > deadline:
            break
        hc = best_h.copy()
        v = np.zeros(n)
        fail = 0
        while time.time() < deadline:
            F, G = smooth_value_grad(hc, n, alpha)
            d = -(G - float(G.mean()))
            nn = float(np.linalg.norm(d))
            if nn < 1e-10:
                break
            d /= nn
            v = 0.75 * v + d
            vn = float(np.linalg.norm(v))
            if vn < 1e-10:
                break
            v /= vn
            ok = False
            for st in (0.05, 0.1, 0.25, 0.6):
                ht = project_feasible(hc + st * v, n)
                Ft, _ = smooth_value_grad(ht, n, alpha)
                if Ft <= F - 1e-13:
                    hc = ht
                    cc = compute_c5(hc, n)
                    if cc < best_c:
                        best_c = cc
                        best_h = hc.copy()
                    ok = True
                    break
            if ok:
                fail = 0
            else:
                v *= 0.5
                fail += 1
                if fail > 30:
                    break
    return best_h, best_c


def resample_h(h, n):
    old = len(h)
    x0 = np.linspace(0.0, 2.0, old)
    x1 = np.linspace(0.0, 2.0, n)
    return project_feasible(np.interp(x1, x0, h), n)


def perturb_h(h, n, kind, rng):
    h = h.copy()
    x = np.linspace(0.0, 2.0, n)
    if kind == 0:
        h += rng.normal(0.0, 0.02 + 0.08 * float(rng.random()), n)
    elif kind == 1:
        k = 1 + int(rng.randint(0, 4))
        h += (0.03 + 0.08 * float(rng.random())) * np.sin(np.pi * k * x + rng.uniform(0.0, 2.0 * np.pi))
    elif kind == 2:
        L = max(5, n // 10 + int(rng.randint(0, max(1, n // 4))))
        st = int(rng.randint(0, max(1, n - L + 1)))
        h[st:st + L] = 1.0 - h[st:st + L]
    elif kind == 3:
        h = 0.6 * h + 0.4 * (1.0 - np.abs(x - 1.0))
    else:
        h = np.roll(h, int(rng.randint(-max(1, n // 3), max(1, n // 3))))
    return project_feasible(h, n)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + max(60.0, float(budget_s) - 10.0)
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 700
        x = np.linspace(0.0, 2.0, n)
        h0 = 1.0 - np.abs(x - 1.0)
    best_h = project_feasible(h0, n)
    best_c = compute_c5(best_h, n)

    cd_deadline = min(deadline - 430.0, t0 + 460.0)
    if cd_deadline > t0 + 25.0:
        best_h, best_c = pair_cd(best_h, len(best_h), 10000, cd_deadline, rng, gs_iters=27)

    sm_deadline = min(deadline - 260.0, time.time() + 220.0)
    if sm_deadline > time.time() + 20.0:
        best_h, best_c = smooth_phase(best_h, len(best_h), sm_deadline, rng)

    tested_n = set()
    while time.time() < deadline - 120.0:
        ncur = len(best_h)
        kind = int(rng.randint(0, 5))
        ntry = ncur
        if time.time() < deadline - 170.0:
            for m in (350, 512, 700, 896):
                if m not in tested_n and 80 <= m <= 1000:
                    ntry = m
                    tested_n.add(m)
                    break
        if ntry == ncur:
            hp = perturb_h(best_h, ncur, kind, rng)
            hn, cn = pair_cd(hp, ncur, 18, deadline - 80.0, rng, gs_iters=23)
        else:
            hp = resample_h(best_h, ntry)
            hn, cn = pair_cd(hp, ntry, 18, deadline - 80.0, rng, gs_iters=23)
            if time.time() < deadline - 70.0:
                sn, sc = smooth_phase(hn, ntry, min(deadline - 60.0, time.time() + 30.0), rng)
                if sc < cn:
                    hn, cn = sn, sc
        if cn < best_c:
            best_h, best_c = hn, cn

    if time.time() < deadline - 8.0:
        hn, cn = pair_cd(best_h, len(best_h), 14, deadline - 8.0, rng, gs_iters=25)
        if cn < best_c:
            best_h, best_c = hn, cn
    if time.time() < deadline - 3.0:
        hn, cn = smooth_phase(best_h, len(best_h), deadline - 3.0, rng)
        if cn < best_c:
            best_h, best_c = hn, cn

    best_h = project_feasible(best_h, len(best_h))
    return best_h, compute_c5(best_h, len(best_h)), len(best_h)
