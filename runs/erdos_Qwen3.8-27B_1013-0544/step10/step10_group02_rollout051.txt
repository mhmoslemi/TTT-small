        h2, c2 = pnorm_optim(h_best, n, p_val, 300, deadline)
        if c2 < best_c - 1e-14:
            h_best, best_c = h2, c2
            h_best, best_c = exact_cd(h_best, n, 200, deadline, rng)

    n_list = []
    for delta in [-5, -3, -2, -1, 1, 2, 3, 5, -10, 10, -20, 20]:
        n_list.append(n + delta)
    n_list += [350, 400, 450, 500, 550, 600, 650, 750, 800, 850, 900]
    n_list = sorted(set(x for x in n_list if x >= 100))
    for n_try in n_list:
        if time.time() > deadline - 50:
            break
        h_int = interp_h(h_best, len(h_best), n_try)
        h_o, c_o = exact_cd(h_int, n_try, 250, deadline, rng)
        if c_o < best_c - 1e-10:
            h_best, best_c = h_o, c_o
        if time.time() < deadline - 25:
            h_o2, c_o2 = pnorm_optim(h_o, n_try, 100, 150, deadline)
            if c_o2 < best_c - 1e-10:
                h_best, best_c = h_o2, c_o2
            h_o3, c_o3 = exact_cd(h_best, len(h_best), 150, deadline, rng)
            if c_o3 < best_c - 1e-10:
                h_best, best_c = h_o3, c_o3

    n_cur = len(h_best)
    for r in range(50):
        if time.time() > deadline - 20:
            break
        mode = r % 5
        hp = structured_perturb(h_best, n_cur, rng, mode)
        h_o, c_o = exact_cd(hp, n_cur, 120, deadline, rng)
        if c_o < best_c - 1e-11:
            h_best, best_c = h_o, c_o
            if time.time() < deadline - 40:
                h_o2, c_o2 = pnorm_optim(h_o, n_cur, 200, 100, deadline)
                if c_o2 < best_c - 1e-11:
                    h_best, best_c = h_o2, c_o2
                    h_best, best_c = exact_cd(h_best, n_cur, 100, deadline, rng)

    n_final = len(h_best)
    h_f, c_f = exact_cd(h_best, n_final, 800, deadline, rng)
    if c_f < best_c - 1e-15:
        h_best, best_c = h_f, c_f

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
