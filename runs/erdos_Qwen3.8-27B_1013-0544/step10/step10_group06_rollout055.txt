```python
    h = h.copy()

    T_target = T / dx  # c[k] <= T_target in the unnormalized sense
    
    for it in range(max_iters):
        if time.time() > deadline:
            return h, False
        c = np.correlate(h, 1.0 - h, mode='full')
        k_star = c.argmax()
        if c[k_star] <= T_target + 1e-10:
            return h, True
        # Compute gradient of c[k_star] w.r.t. h
        L = k_star - (n-1)
        grad = np.zeros(n)
        if L >= 0:
            grad[:n-L] += (1.0 - h[L:n])
            grad[L:] -= h[:n-L]
        else:
            Lp = -L
            grad[Lp:] += (1.0 - h[:n-Lp])
            grad[:n-Lp] -= h[Lp:]
        # Projected gradient step
        # We want to decrease c[k_star], so move in -grad direction
        # But we also need to maintain sum(h) = n/2
        grad_centered = grad - grad.mean()
        # Adaptive step size
        viol = c[k_star] - T_target
        step = min(0.1, viol / (np.max(np.abs(grad_centered)) + 1e-10))
        h_new = h - step * grad_centered
        h_new = project_feasible(h_new, n)
        # Check if it actually decreased the max
        c_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        if c_new[k_star] < c[k_star] or c_new.max() < c.max():
            h = h_new
        else:
            step *= 0.5
            h_new = h - step * grad_centered
            h_new = project_feasible(h_new, n)
            h = h_new
    c_final = np.correlate(h, 1.0 - h, mode='full')
    return h, c_final.max() <= T_target + 1e-10
```

```python
def compute_c5(h, n):

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):

            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h = np.clip(h + diff / tot * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h = np.clip(h + diff / tot * h, 0.0, 1.0)
    return h

def coord_descent(h, n, max_passes, deadline, rng, nbisect=55):
    h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        order = rng.permutation(n)
        for p in order:
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-15 or h0 >= 1.0 - 1e-15:
                continue
            x_new = solve_coord(c, h, p, n, nbisect)
            if abs(x_new - h0) > 1e-14:
                h[p] = x_new
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def lse_grad(h, n, alpha):
    """Compute gradient of LSE surrogate (smooth max) of correlations."""
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    weights = np.exp(shifted - mx)
    weights /= weights.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-15:
            continue
        L = k - (n - 1)
        if L >= 0:
            grad[:n - L] += w * (1.0 - h[L:n])
            grad[L:] -= w * h[:n - L]
        else:
            Lp = -L
            grad[Lp:] += w * (1.0 - h[:n - Lp])
            grad[:n - Lp] -= w * h[Lp:]
    grad *= dx
    return grad, (m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha) * dx
```
