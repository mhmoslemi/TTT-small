    c = np.correlate(h, 1.0 - h, mode='full')

def solve_coord(c, h, p, n):
    h0 = h[p]
    m_min = max(0, n - 1 - p)
    m_max = min(2 * n - 2, 2 * n - 2 - p)
    ms = np.arange(m_min, m_max + 1)
    jf = p + ms - (n - 1)
    jb = p - ms + (n - 1)
    vf = (jf >= 0) & (jf < n)
    vb = (jb >= 0) & (jb < n)
    a = np.zeros(len(ms))
    if np.any(vf):
        a[vf] += (1.0 - h[jf[vf]])
    if np.any(vb):
        a[vb] -= h[jb[vb]]
    c_base = c[ms]
    diag_idx = np.where(ms == (n - 1))[0]
    has_diag = len(diag_idx) > 0
    unaff = 0.0
    if m_min > 0:
        unaff = float(np.max(c[:m_min]))
    if m_max < 2 * n - 2:
        unaff = max(unaff, float(np.max(c[m_max + 1:])))
    cur_max = float(np.max(c))
    neg_mask = a < -1e-14
    pos_mask = a > 1e-14
    zero_mask = np.abs(a) <= 1e-14
    L_lo, L_hi = unaff, cur_max
    for _ in range(50):
        if L_hi - L_lo < 1e-14:
            break
        L = (L_lo + L_hi) / 2.0
        x_lo, x_hi = 0.0, 1.0
        feasible = True
        if np.any(zero_mask) and np.any(c_base[zero_mask] > L + 1e-14):
            feasible = False
        if feasible and np.any(neg_mask):
            x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
        if feasible and np.any(pos_mask):
            x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
        if x_lo > x_hi + 1e-12:
            feasible = False
        if feasible and has_diag:
            c_diag = c_base[diag_idx[0]]
            a_d = a[diag_idx[0]]
            disc = a_d * a_d + 4.0 * (c_diag - L)
            if disc >= 0:
                sq = np.sqrt(disc)
                r1 = (a_d - sq) / 2.0
                r2 = (a_d + sq) / 2.0
                if x_lo > h0 + r1 and x_hi < h0 + r2:
                    feasible = False
        if feasible:
            L_hi = L
        else:
            L_lo = L
    L = L_hi
    x_lo, x_hi = 0.0, 1.0
    if np.any(neg_mask):
        x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
    if np.any(pos_mask):
        x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
    x_lo = max(x_lo, 0.0)
    x_hi = min(x_hi, 1.0)
    if has_diag:
        c_diag = c_base[diag_idx[0]]
        a_d = a[diag_idx[0]]
        disc = a_d * a_d + 4.0 * (c_diag - L)
        if disc >= 0:
            sq = np.sqrt(disc)
            r1 = (a_d - sq) / 2.0
            r2 = (a_d + sq) / 2.0
            if x_lo <= h0 + r1:
                x_hi = min(x_hi, h0 + r1)
            elif x_hi >= h0 + r2:
                x_lo = max(x_lo, h0 + r2)
            else:
                if abs(x_lo - (h0 + r1)) <= abs(x_hi - (h0 + r2)):
                    x_hi = min(x_hi, h0 + r1)
                else:
                    x_lo = max(x_lo, h0 + r2)
    if x_lo > x_hi:
        return h0
    return float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))

def exact_cd(h, n, max_passes, deadline):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    best_c = compute_c5(h, n)
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            h0 = h[p]
            if h0 < 1e-13 or h0 > 1.0 - 1e-13:
                continue
            xn = solve_coord(c, h, p, n)
            if abs(xn - h0) > 1e-13:
                delta = xn - h0
                m_min = max(0, n - 1 - p)
                m_max = min(2 * n - 2, 2 * n - 2 - p)
                ms = np.arange(m_min, m_max + 1)
                jf = p + ms - (n - 1)
                jb = p - ms + (n - 1)
                vf = (jf >= 0) & (jf < n)
                vb = (jb >= 0) & (jb < n)
                a = np.zeros(len(ms))
                if np.any(vf):
                    a[vf] += (1.0 - h[jf[vf]])
                if np.any(vb):
                    a[vb] -= h[jb[vb]]
                c[ms] += delta * a
                diag_i = np.where(ms == (n - 1))[0]
                if len(diag_i) > 0:
                    c[ms[diag_i[0]]] -= delta * delta
                h[p] = xn
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        cur_c = compute_c5(h, n)
        if cur_c < best_c:
            best_c = cur_c
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, best_c

def subgradient_cd(h, n, max_iter, deadline, top_k=50, alpha_init=0.001):
    """Subgradient method with projection, alternating with CD."""
    h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    for it in range(max_iter):
        if time.time() > deadline:
            break
        alpha = alpha_init / (1 + 0.01 * it)
        c = np.correlate(h, 1.0 - h, mode='full')
        # Top-k lags with exponential weighting
        c_sorted = np.sort(c)[::-1]
        idx = np.argsort(c)[::-1][:top_k]
        weights = np.exp(-(c[idx] - c_sorted[0]) / (dx * 0.01 + 1e-15))
        weights /= weights.sum()
        grad = np.zeros(n)
        for mi in range(top_k):
            m = idx[mi]
            w = weights[mi]
            if w < 1e-10:
                continue
            if m == n - 1:
                grad += w * (1.0 - 2.0 * h)
            else:
                offset = m - (n - 1)
                if offset >= 0:
                    grad[:n - offset] += w * (1.0 - h[offset:])
                    grad[offset:] -= w * h[:n - offset]
                else:
                    o = -offset
                    grad[o:] += w * (1.0 - h[:n - o])
                    grad[:n - o] -= w * h[o:]
        h = project_feasible(h - alpha * grad, n)
        cur_c = compute_c5(h, n)
        if cur_c < best_c:
            best_c = cur_c
        # Alternate with a few CD passes every 10 iterations
        if it % 10 == 0 and time.time() < deadline:
            h, best_c = exact_cd(h, n, 2, deadline)
    return h, best_c
