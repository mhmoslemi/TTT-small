def simulated_annealing(h, n, deadline, rng, T0=0.1, T1=0.001, max_iters=5000):

        h[p] = h_new_val

        if delta_c < 0 or rng.random() < np.exp(-delta_c / max(T, 1e-10)):

                best_h = h.copy()

        n_iters += 1
    return best_h, best_c

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:

        else:
            tot = h.sum()

    vals = c_aff + d * a
    if has_diag:
        vals[diag_idx] = c_aff[diag_idx] + d * (1.0 - h0 - d)
    r = float(np.max(vals))
    if unaff_max > r:

    return r

def coord_descent(h, n, max_passes, deadline, gs_iters=50):

    for _ in range(max_passes):

        c = np.correlate(h, 1.0 - h, mode='full')

            jb = p - ms + (n - 1)

            vb = (jb >= 0) & (jb < n)

def softmax_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    return m + np.log(np.sum(np.exp(alpha * (c - m)))) / alpha

    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    e = np.exp(alpha * (c - m))
    w = alpha * e / e.sum()
    # grad of c[k] w.r.t. x[p]
    # c[k] = sum_j x[j] * (1-x[j+k-(n-1)]) * dx
    # dc[k]/dx[p] = (1-x[p+k-(n-1)])*dx if valid - x[p-k+(n-1)]*dx if valid - 2*x[p]*dx if k=n-1
    grad = np.zeros(n)
    for p in range(n):
        g = 0.0
        for k in range(len(c)):
            s = k - (n-1)  # shift
            if 0 <= p + s < n:

            if 0 <= p - s < n:
                g -= w[k] * x[p-s] * dx
            if s == 0:
                g -= 2.0 * w[k] * x[p] * dx
        grad[p] = g
    return grad

    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)

    vals = c_aff + d * a
    if has_d:

    r = float(np.max(vals))
    return max(r, um)
