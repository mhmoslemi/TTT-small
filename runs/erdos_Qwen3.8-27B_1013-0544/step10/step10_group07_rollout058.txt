def project_feasible(h, n, max_iter=100):

    h = np.clip(h.copy(), 0.0, 1.0)

        if abs(diff) < 1e-12:

        if diff > 0:

            tot = room.sum()
            if tot < 1e-15:

        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)

def block_cd(h, n, block_size, max_passes, deadline, seed=42):

    h = h.copy()

            break

            # This is a 1D problem. Use bisection on the target value L.

                    else:
                        ok = False
                if ok:
                    L_hi = L
                else:
                    L_lo = L

                    cand2 = np.clip(h0, max(h0+r2, x_lo), x_hi) if max(h0+r2, x_lo) <= x_hi else None

                    else:
                        x_opt = h0
                else:
                    x_opt = h0

            c = np.correlate(h, 1.0-h, mode='full')
        if not improved:

def block_optimizer(h, n, p, block_size, deadline):
    """Optimize a block of block_size consecutive points starting at p."""
    # Extract block
    idx = np.arange(p, p + block_size)
    block = h[idx].copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    
    # For the block, we need to figure out how c changes
    # This is complex for block_size > 1, so let's use a simpler approach:
    # Just use Nelder-Mead on the block with the full c5 objective
    
    def obj(x):
        h_new = h.copy()
        h_new[idx] = np.clip(x, 0, 1)
        h_new = project_feasible(h_new, n, max_iter=10)
        return compute_c5(h_new, n)
    
    res = minimize(obj, block, method='Nelder-Mead',
                  options={'maxiter': block_size * 20, 'xatol': 1e-8, 'fatol': 1e-12})
    new_block = np.clip(res.x, 0, 1)
    
    # Enforce sum constraint
    diff = block.sum() - new_block.sum()
    if abs(diff) > 1e-10:
        h_trial = h.copy()
        h_trial[idx] = new_block
        h_trial = project_feasible(h_trial, n)
        # Check if the block is still reasonable
        if compute_c5(h_trial, n) < compute_c5(h, n) + 1e-12:
            return h_trial
    
    return h

In SCA, at each iteration:
1. Linearize the quadratic constraints at the current point
2. Solve the resulting convex problem (LP or QP)
3. Update

Specifically, U_k = h^T A_k h where A_k is the shift matrix. The linearization at h_0 is:
h^T A_k h ≈ h_0^T A_k h + h^T A_k (h - h_0) = h_0^T A_k h + h^T A_k h_0 - h_0^T A_k h_0
= 2*h^T A_k h_0 - h_0^T A_k h_0 (since A_k is symmetric)

So the linearized constraint is: 2*h^T A_k h_0 - h_0^T A_k h_0 ≥ t

This gives a linear constraint in h and t! So at each SCA iteration, we solve:
maximize t
subject to: 2*A_k h_0^T h - h_0^T A_k h_0 ≥ t for all k
           sum(h) = n/2
           0 ≤ h_i ≤ 1

def sca_step(h0, n, solver='ECOS'):
    """One step of successive convex approximation."""
    h = cp.Variable(n)
    t = cp.Variable()
    dx = 2.0 / n
    
    constraints = [h >= 0, h <= 1, cp.sum(h) == n/2.0]
    
    # Linearize each autocorrelation constraint
    # U_k = sum_j h[j]*h[j+k] >= (max_k U_k)/dx ... 
    # Actually, let's work with c_k directly
    # c_k = sum_j h[j]*(1-h[j+k]) = sum_j h[j] - U_k = n/2 - U_k (in discrete, times dx)
    # Wait: sum_j h[j] = n/2 (the constraint), so c_k = n/2 - U_k
    # C5 = max_k c_k * dx = (n/2 - min_k U_k) * dx
    # To minimize C5, maximize min_k U_k
    
    # SCA: maximize t subject to linearized U_k >= t
    # U_k(h) ≈ 2 * sum_j h[j] * h0[j+k] - sum_j h0[j] * h0[j+k]

        h0_part = h0[j_min:j_max+1]
        h_part = h[j_min:j_max+1]
        # Linearized: 2 * h_part @ h0_part - h0_part @ h0_part
        lin_val = 2.0 * h0_part @ h_part - float(h0_part @ h0_part)
        constraints.append(lin_val >= t)
    
    prob = cp.Problem(cp.Maximize(t), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver='CBC')
        except Exception:
            return h0.copy()
    
    if h.value is None:
        return h0.copy()
    
    return np.clip(h.value, 0, 1)

def sca_lp_step(h0, n, max_iter=None):
    """One SCA iteration: solve LP with linearized constraints."""
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    dx = 2.0 / n
    
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n/2.0]
    
    # Compute current c values and gradients
    c = np.correlate(h0, 1.0 - h0, mode='full')
    
    for m in range(2*n - 1):
        # Compute gradient of c_m w.r.t. h
        k = m - (n - 1)
        grad = np.zeros(n)
        # Forward: h[i+k] valid when 0 <= i+k < n, i.e., -k <= i < n-k
        if k >= 0:
            grad[:n-k] += (1.0 - h0[k:])  # h[i+k] = h0[i+k] for i in [0, n-k-1]
        else:
            grad[-k:] += (1.0 - h0[:n+k])  # h[i+k] = h0[i+k] for i in [-k, n-1]
        # Backward: h[i-k] valid when 0 <= i-k < n, i.e., k <= i < n+k
        if k >= 0:
            grad[k:] -= h0[:n-k]
        else:
            grad[:n+k] -= h0[-k:]
        
        # Linearized constraint: c[m] + (h - h0)^T grad <= t/dx
        c_m = c[m]
        h0_grad = float(np.dot(h0, grad))
        # c_m + h^T grad - h0^T grad <= t/dx
        # h^T grad <= t/dx - c_m + h0^T grad
        constraints.append(cp.dot(grad, h_var) <= t_var/dx - c_m + h0_grad)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=cp.ECOS)
    except Exception:
        try:
            prob.solve(solver=cp.CBC)
        except Exception:
            return h0.copy(), compute_c5(h0, n)
    
    if h_var.value is None:
        return h0.copy(), compute_c5(h0, n)
    
    h_new = np.clip(h_var.value, 0, 1)
    h_new = project_feasible(h_new, n)
    return h_new, compute_c5(h_new, n)

import numpy as np
from scipy.optimize import minimize
import time
try:
    import cvxpy as cp
    HAS_CP = True
except ImportError:
    HAS_CP = False

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n, max_iter=200):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(max_iter):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def fast_cd(h, n, max_passes, deadline):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            h0 = h[p]
            if h0 < 1e-14 or h0 > 1.0 - 1e-14:
                continue
            m_min = max(0, n - 1 - p)
            m_max = min(2 * n - 2, 2 * n - 2 - p)
            ms = np.arange(m_min, m_max + 1)
            k_vals = ms - (n - 1)
            jf = p + k_vals
            jb = p - k_vals
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(vf):
                a[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                a[vb] -= h[jb[vb]]
            diag_pos = np.where(ms == (n - 1))[0]
            has_diag = len(diag_pos) > 0
            c_base = c[ms].copy()
            if has_diag:
                a[diag_pos] = 0.0
            unaff = 0.0
            if m_min > 0:
                unaff = max(unaff, float(np.max(c[:m_min])))
            if m_max < 2 * n - 2:
                unaff = max(unaff, float(np.max(c[m_max + 1:])))
            cur_max = float(np.max(c))
            L_lo, L_hi = unaff, cur_max
            for _ in range(40):
                L = (L_lo + L_hi) / 2.0
                x_lo, x_hi = 0.0, 1.0
                ok = True
                neg_mask = a < -1e-15
                pos_mask = a > 1e-15
                zero_mask = np.abs(a) <= 1e-15
                if np.any(zero_mask) and np.any(c_base[zero_mask] > L + 1e-15):
                    ok = False
                if ok and np.any(neg_mask):
                    x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
                if ok and np.any(pos_mask):
                    x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
                if x_lo > x_hi + 1e-14:
                    ok = False
                if ok and has_diag:
                    disc = (1 - 2 * h0) ** 2 - 4 * (L - c_base[diag_pos[0]])
                    if disc >= 0:
                        r1 = ((1 - 2 * h0) - np.sqrt(disc)) / 2
                        r2 = ((1 - 2 * h0) + np.sqrt(disc)) / 2
                        if not (max(x_lo, 0.0) <= min(h0 + r1, x_hi) or max(h0 + r2, x_lo) <= min(1.0, x_hi)):
                            ok = False
                    else:
                        ok = False
                if ok:
                    L_hi = L
                else:
                    L_lo = L
            L = L_hi
            x_lo, x_hi = 0.0, 1.0
            neg_mask = a < -1e-15
            pos_mask = a > 1e-15
            if np.any(neg_mask):
                x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
            if np.any(pos_mask):
                x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
            x_lo, x_hi = max(x_lo, 0.0), min(x_hi, 1.0)
            if has_diag:
                disc = (1 - 2 * h0) ** 2 - 4 * (L - c_base[diag_pos[0]])
                if disc >= 0:
                    r1 = ((1 - 2 * h0) - np.sqrt(disc)) / 2
                    r2 = ((1 - 2 * h0) + np.sqrt(disc)) / 2
                    c1 = min(max(h0, x_lo), h0 + r1) if h0 + r1 >= x_lo else None
                    c2 = max(min(h0, x_hi), h0 + r2) if h0 + r2 <= x_hi else None
                    cands = [x for x in [c1, c2] if x is not None]
                    x_opt = min(cands, key=lambda x: abs(x - h0)) if cands else h0
                else:
                    x_opt = h0
            else:
                x_opt = (x_lo + x_hi) / 2.0
            x_opt = float(np.clip(x_opt, 0.0, 1.0))
            if abs(x_opt - h0) > 1e-12:
                delta = x_opt - h0
                c[ms] += delta * a
                if has_diag:
                    c[ms[diag_pos[0]]] -= delta * delta
                h[p] = x_opt
                improved = True
        h = project_feasible(h, n)
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
        if not improved:
            break
    return h, compute_c5(h, n)

def sca_step(h0, n, eps=0.005):
    if not HAS_CP:
        return h0.copy(), compute_c5(h0, n)
    c = np.correlate(h0, 1.0 - h0, mode='full')
    c_max = float(np.max(c))
    active = np.where(c > c_max - eps)[0]
    if len(active) < 2:
        active = np.argsort(c)[-20:]
    dx = 2.0 / n
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2.0]
    for m in active:
        k = m - (n - 1)
        grad = np.zeros(n)
        if k >= 0:
            grad[:n - k] += (1.0 - h0[k:])
            grad[k:] -= h0[:n - k]
        else:
            grad[-k:] += (1.0 - h0[:n + k])
            grad[:n + k] -= h0[-k:]
        h0_grad = float(np.dot(h0, grad))
        constraints.append(cp.dot(grad, h_var) <= t_var / dx - c[m] + h0_grad)
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    for solver in [cp.GLOP, cp.CBC, cp.GLPK, cp.ECOS]:
        try:
            prob.solve(solver=solver)
            break
        except Exception:
            continue
    if h_var.value is None:
        return h0.copy(), compute_c5(h0, n)
    h_new = np.clip(h_var.value, 0.0, 1.0)
    h_new = project_feasible(h_new, n)
    return h_new, compute_c5(h_new, n)

def pair_cd(h, n, max_passes, deadline):
    h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(0, n - 1, 2):
            h_old = (h[p], h[p + 1])
            s = n / 2.0 - h[p] - h[p + 1]
            a_lo, a_hi = max(0.0, s - 1.0), min(1.0, s)
            if a_hi - a_lo < 1e-12:
                continue
            h_orig = h.copy()
            h_orig[p] = 0.0
            h_orig[p + 1] = 0.0
            c_base = np.correlate(h_orig, 1.0 - h_orig, mode='full')
            k_all = np.arange(2 * n - 1) - (n - 1)
            hp = np.where((p - k_all >= 0) & (p - k_all < n), h_orig[p - k_all][(p - k_all >= 0) & (p - k_all < n)], 0.0)
            hp = np.zeros(2 * n - 1)
            pk = p - k_all
            pkv = (pk >= 0) & (pk < n)
            hp[pkv] = h_orig[pk[pkv]]
            hp2 = np.zeros(2 * n - 1)
            pk2 = p + 1 + k_all
            pk2v = (pk2 >= 0) & (pk2 < n)
            hp2[pk2v] = h_orig[pk2[pk2v]]
            hq1 = np.zeros(2 * n - 1)
            q1 = p + 1 - k_all
            q1v = (q1 >= 0) & (q1 < n)
            hq1[q1v] = h_orig[q1[q1v]]
            hq2 = np.zeros(2 * n - 1)
            q2 = p + 1 + k_all
            q2v = (q2 >= 0) & (q2 < n)
            hq2[q2v] = h_orig[q2[q2v]]
            A = 1.0 - hp2 - hp
            B = 1.0 - hq2 - hq1
            A0 = hp2
            B0 = hq2
            cross_lin = np.zeros(2 * n - 1)
            cross_quad = np.zeros(2 * n - 1)
            cross_quad[n] = 1.0
            cross_lin[n] = -(s + 1.0)
            if n >= 2:
                cross_quad[n - 2] = 1.0
                cross_lin[n - 2] = (1.0 - s)
            Q0 = c_base + A0 + B0 + s * B
            Q1 = A - B + cross_lin
            Q2 = cross_quad
            grid = np.linspace(a_lo, a_hi, 100)
            fvals = (Q0[None, :] + grid[:, None] * Q1[None, :] + grid[:, None] ** 2 * Q2[None, :]).max(axis=1)
            i0 = int(fvals.argmin())
            lo = grid[max(0, i0 - 1)]
            hi_g = grid[min(len(grid) - 1, i0 + 1)]
            phi = (1.0 + np.sqrt(5.0)) / 2.0
            a1 = hi_g - (hi_g - lo) / phi
            a2 = lo + (hi_g - lo) / phi
            f1 = np.max(Q0 + Q1 * a1 + Q2 * a1 ** 2)
            f2 = np.max(Q0 + Q1 * a2 + Q2 * a2 ** 2)
            for _ in range(40):
                if hi_g - lo < 1e-12:
                    break
                if f1 < f2:
                    hi_g, a2, f2 = a2, a1, f1
                    a1 = hi_g - (hi_g - lo) / phi
                    f1 = np.max(Q0 + Q1 * a1 + Q2 * a1 ** 2)
                else:
                    lo, a1, f1 = a1, a2, f2
                    a2 = lo + (hi_g - lo) / phi
                    f2 = np.max(Q0 + Q1 * a2 + Q2 * a2 ** 2)
            a_star = float(np.clip((lo + hi_g) / 2.0, a_lo, a_hi))
            h[p] = a_star
            h[p + 1] = s - a_star
            if abs(h[p] - h_old[0]) > 1e-12 or abs(h[p + 1] - h_old[1]) > 1e-12:
                improved = True
        h = project_feasible(h, n)
        if not improved:
            break
    return h, compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = np.full(500, 0.5)

    best_c = compute_c5(h_best, n)

    # Phase 1: Quick 1D CD

    # Phase 2: SCA iterations
    if HAS_CP and time.time() < deadline - 200:
        eps_list = [0.001, 0.003, 0.008, 0.02, 0.05]
        for eps in eps_list:
            if time.time() > deadline - 200:
                break
            for _ in range(8):

                h_s, c_s = sca_step(h_best, n, eps)
                if c_s < best_c - 1e-12:
                    h_best, best_c = h_s, c_s
                else:
                    break
            h_best, c_cd = fast_cd(h_best, n, 20, deadline - 200)
            if c_cd < best_c:
                h_best, best_c = h_best, c_cd

    # Phase 3: 1D CD + Pair CD
    h3, c3 = fast_cd(h_best, n, 100, deadline - 150)
    if c3 < best_c:
        h_best, best_c = h3, c3
    if time.time() < deadline - 100:
        h4, c4 = pair_cd(h_best, n, 50, deadline - 100)
        if c4 < best_c:

    # Phase 4: Multi-n
    n_list = [350, 400, 450, 500, 550, 600, 650, 750, 800, 900, 1000]
    for n_try in n_list:

            continue
        rem = deadline - time.time()

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
