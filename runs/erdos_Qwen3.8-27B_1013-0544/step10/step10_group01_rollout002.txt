def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = 0.5 * n
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h = np.clip(h + diff * room / tot, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h = np.clip(h + diff * h / tot, 0.0, 1.0)
    # exact tail adjustment
    diff = target - h.sum()
    for _ in range(1000):
        if abs(diff) < 1e-12: break
        free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if free.size == 0: break
        # choose index with largest slack in direction
        if diff > 0:
            idx = free[np.argmax(1.0 - h[free])]
            add = min(diff, 1.0 - h[idx] - 1e-15)
        else:
            idx = free[np.argmax(h[free])]
            add = max(diff, h[idx] + 1e-15)  # diff negative, set
        h[idx] += add
        diff = target - h.sum()
    return h

def pair_step_update(h,c,n,p,q,max_step,grid_pts,current_max,one_sided):
    if p == q: return None
    hp = h[p]; hq = h[q]
    if one_sided:
       dlo = max(0.0, -hp, hq - 1.0, -max_step)
       dhi = min(max_step, 1.0 - hp, hq)
    else:
       dlo = max(-hp, hq - 1.0, -max_step)
       dhi = min(1.0 - hp, hq, max_step)
    dlo += 1e-12; dhi -= 1e-12
    if dhi <= dlo + 1e-14: return None
    klo = min(max(-p, p - (n-1)), max(-q, q - (n-1)))
    kmax = max(min(p, n-1-p), min(q, n-1-q))
    L = kmax - klo + 1
    if L <= 0: return None
    k = np.arange(klo, kmax+1)
    A = np.zeros(L, dtype=float)
    f = p + k
    m = (f >= 0) & (f < n)
    if np.any(m): A[m] += 1.0 - h[f[m]]
    b = p - k
    m = (b >= 0) & (b < n)
    if np.any(m): A[m] -= h[b[m]]
    f = q + k
    m = (f >= 0) & (f < n)
    if np.any(m): A[m] -= 1.0 - h[f[m]]
    b = q - k
    m = (b >= 0) & (b < n)
    if np.any(m): A[m] += h[b[m]]
    start = n - 1 + klo
    c_base = c[start:start+L].copy()
    U = 0.0
    if start > 0:

import time
import numpy as np


def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float).copy(), 0.0, 1.0)
    target = 0.5 * n
    for _ in range(250):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff * room / tot, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + diff * h / tot, 0.0, 1.0)

    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0.0:
            idx = int(np.argmax(1.0 - h))
            add = min(diff, 1.0 - h[idx])
        else:
            idx = int(np.argmax(h))
            add = max(diff, -h[idx])
        if add == 0.0:
            break
        h[idx] = min(1.0, max(0.0, h[idx] + add))
    return h


def make_base(n, seed):
    rng = np.random.RandomState(seed)
    x = np.linspace(0.0, 2.0, n)
    h = 0.5 + 0.25 * np.cos(np.pi * (x - 1.0)) + 0.10 * np.sin(2.0 * np.pi * x)
    h = h + 0.04 * rng.normal(size=n)
    return project_feasible(h, n)

    xt = np.linspace(0.0, 2.0, n_target)

        return None
    grid_pts = max(3, int(grid_pts))

    hp = h[p]

def propose_active(h, c, n, active, rng):

        return None, None
    idx = int(active[int(rng.randint(active.size))])
    delta = idx - (n - 1)

        p = int(np.argmax(h))
        q = int(np.argmin(h))

            return None, None

    length = n - abs(delta)
    best_score = -1.0
    best_p = None
    best_q = None

    for _ in range(8):
        t = int(rng.randint(length))
        if delta >= 0:
            high = t
            low = t + delta
        else:
            d = -delta
            high = t + d
            low = t
        score = float(h[high]) * float(1.0 - h[low])
        if score > best_score:
            best_score = score
            best_p = low
            best_q = high

    if best_p is None or best_score < 1e-4:
        return None, None
    return best_p, best_q


def pair_optimize(h, n, deadline, seed, grid_pts=25, max_step=0.12, active_tol=0.35):
    if n <= 1:
        return h, compute_c5(h, n)

    h = project_feasible(h, n)
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = float(c.max())
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = cmax * dx

    rng = np.random.RandomState(seed)
    active = np.where(c >= cmax - active_tol)[0]
    if active.size == 0:
        active = np.array([int(np.argmax(c))])

    fails = 0
    recompute = 0

    while time.time() < deadline:
        p = None
        if active.size > 0 and rng.rand() < 0.85:
            p, q = propose_active(h, c, n, active, rng)
            one = True
            if p is None:
                p = int(rng.randint(n))
                q = (p + 1 + int(rng.randint(n - 1))) % n
                one = False
        else:
            p = int(rng.randint(n))
            q = (p + 1 + int(rng.randint(n - 1))) % n
            one = False

        if p is None:
            p = int(rng.randint(n))
            q = (p + 1) % n
            one = False

        new = pair_step_update(h, c, n, p, q, max_step, grid_pts, cmax, one)
        if new is not None:
            cmax = float(new)
            fails = 0
            recompute += 1
            if cmax * dx < best_c5:
                best_c5 = cmax * dx
                best_h = h.copy()

            active = np.where(c >= cmax - active_tol)[0]
            if active.size == 0:
                active = np.array([int(np.argmax(c))])

            if recompute % 20 == 0:
                c = np.correlate(h, 1.0 - h, mode='full')
                cmax = float(c.max())
                if cmax * dx < best_c5:
                    best_c5 = cmax * dx
                    best_h = h.copy()
                active = np.where(c >= cmax - active_tol)[0]
                if active.size == 0:
                    active = np.array([int(np.argmax(c))])
        else:
            fails += 1
            if fails > 120:
                if time.time() > deadline - 3:
                    break
                scale = min(0.08, 0.01 + 0.005 * (fails // 120))
                h = project_feasible(h + scale * rng.normal(size=n), n)
                c = np.correlate(h, 1.0 - h, mode='full')
                cmax = float(c.max())
                if cmax * dx < best_c5:
                    best_c5 = cmax * dx
                    best_h = h.copy()
                active = np.where(c >= cmax - active_tol)[0]
                if active.size == 0:
                    active = np.array([int(np.argmax(c))])
                fails = 0

                if cmax * dx > best_c5 + 0.004:
                    h = best_h.copy()
                    c = np.correlate(h, 1.0 - h, mode='full')
                    cmax = float(c.max())
                    active = np.where(c >= cmax - active_tol)[0]
                    if active.size == 0:
                        active = np.array([int(np.argmax(c))])

    h = project_feasible(h, n)
    cur = compute_c5(h, n)
    if cur < best_c5:
        best_h = h.copy()
        best_c5 = cur
    true_best = compute_c5(best_h, n)
    if true_best < best_c5:
        best_c5 = true_best
    else:
        best_c5 = true_best
        if true_best > compute_c5(h, n):
            best_h = h.copy()
    return best_h, float(best_c5)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + max(10.0, float(budget_s) - 5.0)

    try:
        h0 = np.asarray(initial_h_values, dtype=float).copy()
        if h0.size < 2:
            raise ValueError
        n0 = int(h0.size)
    except Exception:
        n0 = 700
        h0 = make_base(n0, seed)

    best_h = project_feasible(h0, n0)
    best_c = compute_c5(best_h, n0)
    best_n = n0

    d1 = min(deadline, t0 + max(40.0, min(520.0, 0.45 * float(budget_s))))
    if d1 > time.time() + 5.0:
        h1, c1 = pair_optimize(best_h, best_n, d1, seed, 25, 0.12, 0.35)
        if c1 < best_c:
            best_h, best_c, best_n = h1, c1, best_n

    if time.time() < deadline - 90.0 and best_n < 920:
        n2 = min(920, max(600, int(best_n * 1.25)))
        rem = deadline - time.time()
        alloc = min(rem * 0.40, 300.0)
        if alloc > 45.0:
            h2 = interpolate_candidate(best_h, n2)
            h2, c2 = pair_optimize(h2, n2, time.time() + alloc, seed + 101, 23, 0.10, 0.35)
            if c2 < best_c:
                best_h, best_c, best_n = h2, c2, n2

    if time.time() < deadline - 70.0 and best_n > 520:
        n3 = max(300, min(550, best_n - 150))
        rem = deadline - time.time()
        alloc = min(rem * 0.30, 220.0)
        if alloc > 40.0:
            h3 = interpolate_candidate(best_h, n3)
            h3, c3 = pair_optimize(h3, n3, time.time() + alloc, seed + 201, 23, 0.12, 0.40)
            if c3 < best_c:
                best_h, best_c, best_n = h3, c3, n3

    if time.time() < deadline - 60.0:
        rem = deadline - time.time()
        alloc = min(rem * 0.25, 180.0)
        if alloc > 35.0:
            hr = make_base(best_n, seed + 302)
            hr, cr = pair_optimize(hr, best_n, time.time() + alloc, seed + 303, 23, 0.16, 0.50)
            if cr < best_c:
                best_h, best_c = hr, cr

    rem = deadline - time.time()
    if rem > 25.0:
        hf, cf = pair_optimize(best_h, best_n, deadline - 2.0, seed + 401, 33, 0.03, 0.25)
        if cf < best_c:
            best_h, best_c = hf, cf

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, float(best_c), int(best_n)
