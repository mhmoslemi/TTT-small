def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def binary_project(h, n):
    """Project to binary with sum = n/2"""
    target = n // 2
    # Sort by value, set top 'target' to 1, rest to 0
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

    """Compute C5 for binary h"""
    dx = 2.0 / n
    c = np.correlate(b, 1.0 - b, mode='full')
    return float(c.max() * dx)

def binary_local_search(b, n, max_iters=5000, deadline=None):
    """Local search on binary h via swaps"""
    b = b.copy()
    best_c = binary_c5(b, n)
    ones = np.where(b == 1)[0]
    zeros = np.where(b == 0)[0]
    
    for it in range(max_iters):
        if deadline and time.time() > deadline:
            break
        c = np.correlate(b, 1.0 - b, mode='full')
        best_improvement = -np.inf

        for _ in range(n_tries):
            i = ones[np.random.randint(len(ones))]
            j = zeros[np.random.randint(len(zeros))]

def penalty_slsqp(h, n, alphas, mu=100.0, maxiter=300):
    """SLSQP with penalty for sum constraint instead of hard constraint"""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    
    def obj_pen(x, alpha):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        lse = mx + np.log(np.sum(np.exp(shifted - mx)))
        smooth_max = (m + lse / alpha) * (2.0 / n)
        penalty = mu * (x.sum() - n / 2.0) ** 2
        return smooth_max + penalty
    
    for alpha in alphas:

        except Exception:
            pass
    return best_h, best_c

            tot = room.sum()
            if tot < 1e-14:
                break
            frac = diff / tot
            h = np.clip(h + frac * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            frac = diff / tot
            h = np.clip(h + frac * h, 0.0, 1.0)
    return h

def smooth_objective(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    if alpha < 1000:
        # Use power mean for numerical stability at low alpha
        p = alpha
        return float((np.sum((c ** p)) ** (1.0/p) * (2.0/n)))
    shifted = alpha * (c - m)
    if shifted.max() > 500:
        shifted = 500.0
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return float((m + lse / alpha) * (2.0 / n))

    return x.sum() - n / 2.0

def _sum_eq_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alphas, maxiter_each=300):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac, 'args': (n,)}]
    for alpha in alphas:
        try:
            res = minimize(smooth_objective, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-15})
            if res.x is not None:
                new_h = project_feasible(res.x, n)
                new_c = compute_c5(new_h, n)
                if new_c < best_c:
                    best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def cd_optimize(h, n, max_passes=100, deadline=None, gs_iters=30):
    h = h.copy()
    best_c = compute_c5(h, n)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    n_corr = 2 * n - 1
    
    for _ in range(max_passes):
        if deadline and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in range(n):
            if deadline and time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(n_corr - 1, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(valid_f):
                a[valid_f] += (1.0 - h[jf[valid_f]])
            if np.any(valid_b):
                a[valid_b] -= h[jb[valid_b]]
            
            diag_pos = np.where(ms == (n - 1))[0]
            has_diag = len(diag_pos) > 0
            diag_idx = int(diag_pos[0]) if has_diag else -1
            c_aff = c[ms]
            
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < n_corr - 1:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            
            def fval(d):
                vals = c_aff + d * a
                if has_diag:
                    vals[diag_idx] -= d * d
                result = float(np.max(vals))
                if unaff_max > result:
                    result = unaff_max
                return result
            
            d_lo = -h0
            d_hi = 1.0 - h0
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            f1 = fval(d1)
            f2 = fval(d2)
            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-14:
                    break
                if f1 < f2:
                    d_hi = d2
                    d2 = d1
                    f2 = f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    f1 = fval(d1)
                else:
                    d_lo = d1
                    d1 = d2
                    f1 = f2
                    d2 = d_lo + gr * (d_hi - d_lo)
                    f2 = fval(d2)
            d_opt = (d_lo + d_hi) / 2.0
            f_opt = fval(d_opt)
            f_curr = fval(0.0)
            if f_opt < f_curr - 1e-13:
                h[p] = h0 + d_opt
                improved = True
        if improved:
            h = project_feasible(h, n)
            c_new = compute_c5(h, n)
            if c_new < best_c:
                best_c = c_new
        else:
            break
    return h, best_c

def binary_search(h, n, max_swaps=5000, deadline=None, rng=None):
    """Binary local search: project to binary, then do swap-based local search"""
    if rng is None:
        rng = np.random.RandomState(0)
    target = n // 2
    b = np.zeros(n)
    idx = np.argsort(h)[::-1]
    b[idx[:target]] = 1.0
    best_c = binary_c5_eval(b, n)
    ones = np.where(b == 1)[0]
    zeros = np.where(b == 0)[0]
    
    for _ in range(max_swaps):
        if deadline and time.time() > deadline:
            break
        c = np.correlate(b, 1.0 - b, mode='full')
        cur_max = c.max()
        best_improve = cur_max
        best_pair = None
        n_tries = min(200, len(ones) * len(zeros) // 2)
        for _ in range(n_tries):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            if i == j:
                continue
            b[i], b[j] = 0.0, 1.0
            new_max = float(np.max(np.correlate(b, 1.0 - b, mode='full')))
            b[i], b[j] = 1.0, 0.0
            if new_max < best_improve:
                best_improve = new_max
                best_pair = (i, j)
        if best_pair:
            b[best_pair[0]], b[best_pair[1]] = 0.0, 1.0
            best_c = binary_c5_eval(b, n)
            ones = np.where(b == 1)[0]
            zeros = np.where(b == 0)[0]
        else:
            break
    return b, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def _lse_val(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    shifted = np.minimum(shifted, 500.0)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return float((m + lse / alpha) * (2.0 / n))

def _sum_eq(x, n):
    return x.sum() - n / 2.0

def _sum_eq_jac(x, n):
    return np.ones(n)

def slsqp(h, n, alphas, maxiter=400):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac, 'args': (n,)}]
    for alpha in alphas:
        try:
            res = minimize(_lse_val, best_h, args=(n, alpha), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-15})
            if res.x is not None:
                nh = project_feasible(res.x, n)
                nc = compute_c5(nh, n)
                if nc < best_c:
                    best_h, best_c = nh, nc
        except Exception:
            pass
    return best_h, best_c

def _cd_fval(ca, a, di, hd, um, d):
    v = ca + d * a
    if hd:
        v[di] -= d * d
    r = float(np.max(v))
    if um > r:
        r = um
    return r

def cd(h, n, max_passes=50, deadline=None, gs_iters=30):
    h = h.copy()
    best_c = compute_c5(h, n)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    nc = 2 * n - 1
    for _ in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in range(n):
            if deadline is not None and time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-12 or h0 >= 1.0 - 1e-12:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(nc - 1, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(vf):
                a[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                a[vb] -= h[jb[vb]]
            dp = np.where(ms == (n - 1))[0]
            hd = len(dp) > 0
            di = int(dp[0]) if hd else -1
            ca = c[ms]
            um = 0.0
            if m_lo > 0:
                um = max(um, float(c[:m_lo].max()))
            if m_hi < nc - 1:
                um = max(um, float(c[m_hi + 1:].max()))
            dlo = -h0
            dhi = 1.0 - h0
            d1 = dhi - gr * (dhi - dlo)
            d2 = dlo + gr * (dhi - dlo)
            f1 = _cd_fval(ca, a, di, hd, um, d1)
            f2 = _cd_fval(ca, a, di, hd, um, d2)
            for _ in range(gs_iters):
                if dhi - dlo < 1e-14:
                    break
                if f1 < f2:
                    dhi = d2; d2 = d1; f2 = f1
                    d1 = dhi - gr * (dhi - dlo)
                    f1 = _cd_fval(ca, a, di, hd, um, d1)
                else:
                    dlo = d1; d1 = d2; f1 = f2
                    d2 = dlo + gr * (dhi - dlo)
                    f2 = _cd_fval(ca, a, di, hd, um, d2)
            dopt = (dlo + dhi) / 2.0
            fo = _cd_fval(ca, a, di, hd, um, dopt)
            fc = _cd_fval(ca, a, di, hd, um, 0.0)
            if fo < fc - 1e-13:
                h[p] = h0 + dopt
                improved = True
        if improved:
            h = project_feasible(h, n)
            cn = compute_c5(h, n)
            if cn < best_c:
                best_c = cn
        else:
            break
    return h, best_c

def thue_morse_init(n):
    t = np.zeros(n)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    target = n // 2
    n_ones = int(t.sum())
    if n_ones > target:
        idx = np.where(t == 1)[0]
        rng_sub = np.random.RandomState(42)
        remove = rng_sub.choice(n_ones, n_ones - target, replace=False)
        t[idx[remove]] = 0.0
    elif n_ones < target:
        idx = np.where(t == 0)[0]
        rng_sub = np.random.RandomState(43)
        add = rng_sub.choice(n - n_ones, target - n_ones, replace=False)
        t[idx[add]] = 1.0
    return t

    best_h = h.copy()
    best_c = compute_c5(h, n)
    cur_c = best_c

        delta = c_new - cur_c

            h = h_new
            cur_c = c_new
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
    return best_h, best_c
