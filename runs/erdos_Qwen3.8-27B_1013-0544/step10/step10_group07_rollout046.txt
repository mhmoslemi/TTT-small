    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def subgrad_cd(h, n, max_iter, step_scale, deadline):
    """Subgradient method targeting the max lag with projection."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    c = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(c))
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        cur_c = float(c.max())
        cur_c5 = cur_c * (2.0 / n)
        if cur_c5 < best_c:
            best_c = cur_c5
            best_h = h.copy()
        k_star = int(np.argmax(c))
        grad = np.zeros(n)
        for i in range(n):
            j1 = i + k_star - (n - 1)
            j2 = i - k_star + (n - 1)
            v1 = 1.0 if (j1 < 0 or j1 >= n) else (1.0 - h[j1])
            v2 = -1.0 if (j2 < 0 or j2 >= n) else -h[j2]
            grad[i] = v1 + v2
        step = step_scale / (1.0 + 0.01 * it)
        h_new = h - step * grad
        h_new = project_feasible(h_new, n)
        h = h_new
    return best_h, best_c

def smooth_max_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    if m < 0:
        return 0.0
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * (2.0 / n)

def power_mean_obj(x, n, p):
    c = np.correlate(x, 1.0 - x, mode='full')
    cp = np.clip(c, 0, None)
    return float(np.sum(cp ** p) ** (1.0 / p) * (2.0 / n))

def lbfgs_opt(h, n, P_list, maxiter, deadline):
    best_h, best_c = h.copy(), compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for P in P_list:
        if time.time() > deadline:
            break
        def obj(z):
            cp = np.correlate(z, 1.0 - z, mode='full')
            cp = np.clip(cp, 0, None)
            return float(np.sum(cp ** P) ** (1.0 / P) * (2.0 / n))
        def obj_grad(z):
            c = np.correlate(z, 1.0 - z, mode='full')
            cp = np.clip(c, 1e-15, None)
            S = np.sum(cp ** P)
            coeff = (S ** (1.0 / P - 1.0)) * (2.0 / n)
            grad = np.zeros(n)
            for k in range(len(c)):
                w = cp[k] ** (P - 1) * coeff
                offset = k - (n - 1)
                for i in range(n):
                    j1 = i + offset
                    if 0 <= j1 < n:
                        grad[i] += w * (1.0 - z[j1])
                    j2 = i - offset
                    if 0 <= j2 < n:
                        grad[i] += w * (-z[j2])
            return grad
        try:
            res = minimize(obj, best_h, jac=obj_grad, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c - 1e-12:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

def cd_1d_fast(h, n, max_passes, deadline):
    """1D coordinate descent with incremental correlation update."""
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            h0 = h[p]
            if h0 < 1e-12 or h0 > 1.0 - 1e-12:
                continue
            m_min = n - 1 - p
            m_max = 2 * n - 2 - p
            ms = np.arange(m_min, m_max + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            vf = (jf >= 0) & (jf < n)
            vb = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(vf):
                a[vf] += (1.0 - h[jf[vf]])
            if np.any(vb):
                a[vb] -= h[jb[vb]]
            c_base = c[ms].copy()
            diag_idx = np.where(ms == (n - 1))[0]
            has_diag = len(diag_idx) > 0
            if has_diag:
                c_diag = c_base[diag_idx[0]].copy()
                a[diag_idx] = 0.0
            else:
                c_diag = None
            unaff = 0.0
            if m_min > 0:
                unaff = max(unaff, float(np.max(c[:m_min])))
            if m_max < 2 * n - 2:
                unaff = max(unaff, float(np.max(c[m_max + 1:])))
            cur_max = float(np.max(c))
            neg_mask = a < -1e-14
            pos_mask = a > 1e-14
            zero_mask = np.abs(a) <= 1e-14
            L_lo, L_hi = unaff, cur_max
            for _ in range(50):
                L = (L_lo + L_hi) / 2.0
                x_lo, x_hi = 0.0, 1.0
                ok = True
                if np.any(zero_mask) and np.any(c_base[zero_mask] > L + 1e-14):
                    ok = False
                if ok and np.any(neg_mask):
                    x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
                if ok and np.any(pos_mask):
                    x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
                if x_lo > x_hi + 1e-13:
                    ok = False
                if ok and has_diag:
                    Cq = L - c_diag + h0 - h0 * h0
                    disc = 1.0 - 4.0 * Cq
                    if disc >= 0:
                        r1 = (1.0 - np.sqrt(disc)) / 2.0
                        r2 = (1.0 + np.sqrt(disc)) / 2.0
                        if not (max(x_lo, 0.0) < r1 and min(x_hi, 1.0) > r2):
                            ok = False
                if ok:
                    L_hi = L
                else:
                    L_lo = L
            L = L_hi
            x_lo, x_hi = 0.0, 1.0
            if np.any(neg_mask):
                x_lo = max(x_lo, float(np.max((L - c_base[neg_mask]) / a[neg_mask])))
            if np.any(pos_mask):
                x_hi = min(x_hi, float(np.min((L - c_base[pos_mask]) / a[pos_mask])))
            x_lo = max(x_lo, 0.0)
            x_hi = min(x_hi, 1.0)
            if has_diag:
                Cq = L - c_diag + h0 - h0 * h0
                disc = 1.0 - 4.0 * Cq
                if disc >= 0:
                    r1 = (1.0 - np.sqrt(disc)) / 2.0
                    r2 = (1.0 + np.sqrt(disc)) / 2.0
                    if max(x_lo, 0.0) < r1 and min(x_hi, 1.0) > r2:
                        x_hi = min(x_hi, r1) if h0 - r1 <= r2 - h0 else max(x_lo, r2)
                        x_lo = min(x_lo, r2) if h0 - r1 <= r2 - h0 else max(x_lo, r2)
            if x_lo > x_hi:
                continue
            xn = float(np.clip((x_lo + x_hi) / 2.0, 0.0, 1.0))
            if abs(xn - h0) > 1e-13:
                delta = xn - h0
                if np.any(vf):
                    c[m_min + np.where(vf)[0]] += delta * (1.0 - h[jf[vf]])
                if np.any(vb):
                    c[m_min + np.where(vb)[0]] += delta * (-h[jb[vb]])
                if has_diag:
                    c[n - 1] -= delta * delta
                h[p] = xn
                improved = True
        h = project_feasible(h, n)
        if not improved:
            break
        if abs(h.sum() - n / 2.0) > 1e-10:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h, compute_c5(h, n)

def pair_cd(h, n, max_passes, deadline):
    h = h.copy()
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(0, n - 1, 2):
            h_old = (h[p], h[p + 1])
            s = n / 2.0 - h[p] - h[p + 1]
            a_lo = max(0.0, s - 1.0)
            a_hi = min(1.0, s)
            if a_hi - a_lo < 1e-12:
                continue
            h_orig = h.copy()
            h_orig[p] = 0.0
            h_orig[p + 1] = 0.0
            c_base = np.correlate(h_orig, 1.0 - h_orig, mode='full')
            k_all = np.arange(2 * n - 1) - (n - 1)
            pk = p - k_all
            pk2 = p + 1 - k_all
            pkv = (pk >= 0) & (pk < n)
            pk2v = (pk2 >= 0) & (pk2 < n)
            hp = np.zeros(2 * n - 1)
            hp2 = np.zeros(2 * n - 1)
            if np.any(pkv):
                hp[pkv] = h_orig[pk[pkv]]
            if np.any(pk2v):
                hp2[pk2v] = h_orig[pk2[pk2v]]
            q_p = p + 1 + 1 - k_all
            q_p2 = p + 1 + k_all + 1
            qp1v = (q_p >= 0) & (q_p < n)
            qp2v = (q_p2 >= 0) & (q_p2 < n)
            hq1 = np.zeros(2 * n - 1)
            hq2 = np.zeros(2 * n - 1)
            if np.any(qp1v):
                hq1[qp1v] = h_orig[q_p[qp1v]]
            if np.any(qp2v):
                hq2[qp2v] = h_orig[q_p2[qp2v]]
            A = 1.0 - hp2 - hp
            B = 1.0 - hq2 - hq1
            A0 = hp2
            B0 = hq2
            cross_lin = np.zeros(2 * n - 1)
            cross_quad = np.zeros(2 * n - 1)
            cross_quad[n] = 1.0
            cross_lin[n] = -(s + 1.0)
            if n >= 2:
                cross_quad[n - 2] = 1.0
                cross_lin[n - 2] = (1.0 - s)
            Q0 = c_base + A0 + B0 + s * B
            Q1 = A - B + cross_lin
            Q2 = cross_quad
            grid = np.linspace(a_lo, a_hi, 100)
            vals = Q0[None, :] + grid[:, None] * Q1[None, :] + grid[:, None] ** 2 * Q2[None, :]
            fvals = vals.max(axis=1)
            i0 = int(fvals.argmin())
            lo = grid[max(0, i0 - 1)]
            hi = grid[min(len(grid) - 1, i0 + 1)]
            phi = (1.0 + np.sqrt(5.0)) / 2.0
            a1 = hi - (hi - lo) / phi
            a2 = lo + (hi - lo) / phi
            f1 = np.max(Q0 + Q1 * a1 + Q2 * a1 ** 2)
            f2 = np.max(Q0 + Q1 * a2 + Q2 * a2 ** 2)
            for _ in range(50):
                if hi - lo < 1e-14:
                    break
                if f1 < f2:
                    hi = a2; a2 = a1; f2 = f1
                    a1 = hi - (hi - lo) / phi
                    f1 = np.max(Q0 + Q1 * a1 + Q2 * a1 ** 2)
                else:
                    lo = a1; a1 = a2; f1 = f2
                    a2 = lo + (hi - lo) / phi
                    f2 = np.max(Q0 + Q1 * a2 + Q2 * a2 ** 2)
            a_star = float(np.clip((lo + hi) / 2.0, a_lo, a_hi))
            h[p] = a_star
            h[p + 1] = s - a_star
            if abs(h[p] - h_old[0]) > 1e-13 or abs(h[p + 1] - h_old[1]) > 1e-13:
                improved = True
        h = project_feasible(h, n)
        if not improved:
            break
    return h, compute_c5(h, n)

def sin_perturb(h, n, freqs, amplitudes, rng):
    x = np.arange(n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for f in freqs:
        for amp in amplitudes:
            phase = rng.uniform(0, 2 * np.pi)
            pert = amp * np.cos(2 * np.pi * f * x / n + phase)
            hp = project_feasible(h + pert, n)
            c = compute_c5(hp, n)
            if c < best_c:
                best_h, best_c = hp, c
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    except Exception:
        h_best = np.full(400, 0.5)
        n = 400
    h_best = project_feasible(h_best, n)
    best_c = compute_c5(h_best, n)

    # Phase 1: Subgradient warmup (fast global search)
    if time.time() < deadline - 600:
        h_sg, c_sg = subgrad_cd(h_best, n, 2000, 0.02, deadline)
        if c_sg < best_c:
            h_best, best_c = h_sg, c_sg

    # Phase 2: L-BFGS-B with power mean annealing
    if time.time() < deadline - 400:
        h_lb, c_lb = lbfgs_opt(h_best, n, [10, 50, 100, 200], 200, deadline)
        if c_lb < best_c:
            h_best, best_c = h_lb, c_lb

    # Phase 3: 1D CD
    if time.time() < deadline - 200:
        h_cd, c_cd = cd_1d_fast(h_best, n, 500, deadline)
        if c_cd < best_c:
            h_best, best_c = h_cd, c_cd

    # Phase 4: Pair CD
    if time.time() < deadline - 100:
        h_pc, c_pc = pair_cd(h_best, n, 100, deadline)
        if c_pc < best_c:
            h_best, best_c = h_pc, c_pc

    # Phase 5: Multi-n exploration
    n_list = [350, 500, 700, 900, 1200]
    for n_try in n_list:
        if time.time() > deadline - 150:
            break
        if n_try == n:
            continue
        x_old = np.linspace(0, 2, len(h_best), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_int = project_feasible(np.interp(x_new, x_old, h_best), n_try)
        h_o, c_o = cd_1d_fast(h_int, n_try, 300, deadline)
        h_o, c_o2 = pair_cd(h_o, n_try, 80, deadline)
        if c_o2 < c_o:
            h_o, c_o = h_o, c_o2
        if c_o < best_c - 1e-7:
            h_best, best_c, n = h_o, c_o, n_try

    # Phase 6: Structured perturbations + re-optimize
    freqs = [1, 2, 3, 5, 8, 13]
    amps = [0.005, 0.01, 0.02, 0.04]
    if time.time() < deadline - 200:
        h_best, best_c = sin_perturb(h_best, n, freqs, amps, rng)
        if time.time() < deadline - 100:
            h_cd2, c_cd2 = cd_1d_fast(h_best, n, 300, deadline)
            if c_cd2 < best_c:
                h_best, best_c = h_cd2, c_cd2

    # Phase 7: More subgradient rounds on best
    if time.time() < deadline - 100:
        h_sg2, c_sg2 = subgrad_cd(h_best, n, 1500, 0.005, deadline)
        if c_sg2 < best_c:
            h_best, best_c = h_sg2, c_sg2

    # Phase 8: Final 1D CD polish
    if time.time() < deadline - 5:
        h_f, c_f = cd_1d_fast(h_best, n, 200, deadline)
        if c_f < best_c:
            h_best, best_c = h_f, c_f

    # Phase 9: Final pair CD
    if time.time() < deadline - 2:
        h_pf, c_pf = pair_cd(h_best, n, 50, deadline)
        if c_pf < best_c:
            h_best, best_c = h_pf, c_pf

    h_best = project_feasible(h_best, len(h_best))
    best_c = compute_c5(h_best, len(h_best))
    return h_best, best_c, len(h_best)
