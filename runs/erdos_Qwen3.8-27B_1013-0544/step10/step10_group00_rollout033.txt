import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def _eval_1d(c, a, diag_idx, has_diag, unaff_max, h0, d):
    vals = c + d * a
    if has_diag:
        vals[diag_idx] = c[diag_idx] + d * (1.0 - h0 - d)
    result = float(np.max(vals))
    if unaff_max > result:
        result = unaff_max
    return result

def coord_descent(h, n, max_passes, deadline, gs_iters=60):
    h = h.copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-13 or h0 >= 1.0 - 1e-13:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(valid_f):
                a[valid_f] += (1.0 - h[jf[valid_f]])
            if np.any(valid_b):
                a[valid_b] -= h[jb[valid_b]]
            diag_pos = np.where(ms == (n - 1))[0]
            has_diag = len(diag_pos) > 0
            diag_idx = int(diag_pos[0]) if has_diag else -1
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            d_lo = -h0
            d_hi = 1.0 - h0
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            f1 = _eval_1d(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
            f2 = _eval_1d(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-15:
                    break
                if f1 < f2:
                    d_hi, d2, f2 = d2, d1, f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    f1 = _eval_1d(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
                else:
                    d_lo, d1, f1 = d1, d2, f2
                    d2 = d_lo + gr * (d_hi - d_lo)
                    f2 = _eval_1d(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            d_opt = (d_lo + d_hi) / 2.0
            f_opt = _eval_1d(c_aff, a, diag_idx, has_diag, unaff_max, h0, d_opt)
            f_curr = _eval_1d(c_aff, a, diag_idx, has_diag, unaff_max, h0, 0.0)
            # Check if pushing to boundary helps
            f_lo = _eval_1d(c_aff, a, diag_idx, has_diag, unaff_max, h0, d_lo)
            f_hi = _eval_1d(c_aff, a, diag_idx, has_diag, unaff_max, h0, d_hi)
            best_f = min(f_opt, f_lo, f_hi)
            if best_f < f_curr - 1e-12:
                if best_f == f_lo:
                    h[p] = 0.0
                elif best_f == f_hi:
                    h[p] = 1.0
                else:
                    h[p] = h0 + d_opt
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def lse_smoothed(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    lse = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (m + lse / alpha) * dx

def lse_gradient(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    k_max = int(np.argmax(c))
    shifted = alpha * (c - m)
    mx = shifted.max()
    exp_s = np.exp(shifted - mx)
    w = exp_s / exp_s.sum()
    comb_w = w.copy()
    comb_w[k_max] += 1.0
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = comb_w[k]
        if abs(wk) < 1e-15:
            continue
        m_lag = k - (n - 1)
        if m_lag == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif m_lag > 0:
            grad[:n - m_lag] += wk * (1.0 - h[m_lag:])
            grad[m_lag:] += wk * (-h[:n - m_lag])
        else:
            ma = -m_lag
            grad[ma:] += wk * (1.0 - h[:n - ma])
            grad[:n - ma] += wk * (-h[ma:])
    return grad * dx

def slsqp_polish(h, n, alphas, maxiter, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0, 'jac': lambda x: np.ones(n)}]
    for alpha in alphas:
        if time.time() > deadline:
            break
        def obj(x):
            return lse_smoothed(x, n, alpha)
        def grad(x):
            return lse_gradient(x, n, alpha)
        try:
            res = minimize(obj, best_h, jac=grad, method='SLSQP',

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def _eval_1d(c, a, diag_idx, has_diag, unaff_max, h0, d):
    vals = c + d * a
    if has_diag:
        vals[diag_idx] = c[diag_idx] + d * (1.0 - h0 - d)
    result = float(np.max(vals))
    if unaff_max > result:
        result = unaff_max
    return result

def coord_descent(h, n, max_passes, deadline, gs_iters=60):
    h = h.copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for p in range(n):
            if time.time() > deadline:
                break
            h0 = h[p]
            if h0 <= 1e-13 or h0 >= 1.0 - 1e-13:
                continue
            m_lo = max(0, n - 1 - p)
            m_hi = min(2 * n - 2, n - 1 + p)
            ms = np.arange(m_lo, m_hi + 1)
            jf = p + ms - (n - 1)
            jb = p - ms + (n - 1)
            valid_f = (jf >= 0) & (jf < n)
            valid_b = (jb >= 0) & (jb < n)
            a = np.zeros(len(ms))
            if np.any(valid_f):
                a[valid_f] += (1.0 - h[jf[valid_f]])
            if np.any(valid_b):
                a[valid_b] -= h[jb[valid_b]]
            diag_pos = np.where(ms == (n - 1))[0]
            has_diag = len(diag_pos) > 0
            diag_idx = int(diag_pos[0]) if has_diag else -1
            c_aff = c[ms]
            unaff_max = 0.0
            if m_lo > 0:
                unaff_max = max(unaff_max, float(c[:m_lo].max()))
            if m_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(c[m_hi + 1:].max()))
            d_lo = -h0
            d_hi = 1.0 - h0
            d1 = d_hi - gr * (d_hi - d_lo)
            d2 = d_lo + gr * (d_hi - d_lo)
            f1 = _eval_1d(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
            f2 = _eval_1d(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            for _ in range(gs_iters):
                if d_hi - d_lo < 1e-15:
                    break
                if f1 < f2:
                    d_hi, d2, f2 = d2, d1, f1
                    d1 = d_hi - gr * (d_hi - d_lo)
                    f1 = _eval_1d(c_aff, a, diag_idx, has_diag, unaff_max, h0, d1)
                else:
                    d_lo, d1, f1 = d1, d2, f2
                    d2 = d_lo + gr * (d_hi - d_lo)
                    f2 = _eval_1d(c_aff, a, diag_idx, has_diag, unaff_max, h0, d2)
            d_opt = (d_lo + d_hi) / 2.0
            f_opt = _eval_1d(c_aff, a, diag_idx, has_diag, unaff_max, h0, d_opt)
            f_curr = _eval_1d(c_aff, a, diag_idx, has_diag, unaff_max, h0, 0.0)
            f_lo = _eval_1d(c_aff, a, diag_idx, has_diag, unaff_max, h0, d_lo)
            f_hi = _eval_1d(c_aff, a, diag_idx, has_diag, unaff_max, h0, d_hi)
            best_f = min(f_opt, f_lo, f_hi)
            if best_f < f_curr - 1e-12:
                if best_f == f_lo:
                    h[p] = 0.0
                elif best_f == f_hi:
                    h[p] = 1.0
                else:
                    h[p] = h0 + d_opt
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
    return h, compute_c5(h, n)

def lse_smoothed(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    exp_s = np.exp(shifted - mx)
    lse = mx + np.log(exp_s.sum())
    return (m + lse / alpha) * dx

def lse_gradient(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    k_max = int(np.argmax(c))
    shifted = alpha * (c - m)
    mx = shifted.max()
    exp_s = np.exp(shifted - mx)
    w = exp_s / exp_s.sum()
    comb_w = w.copy()
    comb_w[k_max] += 1.0
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = comb_w[k]
        if abs(wk) < 1e-15:
            continue
        m_lag = k - (n - 1)
        if m_lag == 0:
            grad += wk * (1.0 - 2.0 * x)
        elif m_lag > 0:
            grad[:n - m_lag] += wk * (1.0 - x[m_lag:])
            grad[m_lag:] += wk * (-x[:n - m_lag])
        else:
            ma = -m_lag
            grad[ma:] += wk * (1.0 - x[:n - ma])
            grad[:n - ma] += wk * (-x[ma:])
    return grad * dx

def lse_grad_descent(h, n, alphas, deadline, steps_per_alpha=300):
    h = h.copy()
    for alpha in alphas:
        step = 0.005
        for _ in range(steps_per_alpha):
            if time.time() > deadline:
                break
            f_curr = lse_smoothed(h, n, alpha)
            grad = lse_gradient(h, n, alpha)
            gnorm = np.sqrt(np.sum(grad**2))
            if gnorm < 1e-12:
                break
            for _ in range(30):
                h_new = project_feasible(h - step * grad, n)
                f_new = lse_smoothed(h_new, n, alpha)
                if f_new < f_curr:
                    break
                step *= 0.5
            if f_new < f_curr:
                h = h_new
            else:
                step *= 0.1
    return h, compute_c5(h, n)

def _slsqp_obj(x, n, alpha):
    return lse_smoothed(x, n, alpha)

def _slsqp_grad(x, n, alpha):
    return lse_gradient(x, n, alpha)

def _cons_fun(x, n):
    return x.sum() - n / 2.0

def _cons_jac(x, n):
    return np.ones(n)

def slsqp_polish(h, n, alphas, maxiter, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline:
            break
        cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac, 'args': (n,)}]
        try:
            res = minimize(_slsqp_obj, best_h, args=(n, alpha),
                           jac=lambda x, an=alpha: _slsqp_grad(x, n, an),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-15})
            new_h = project_feasible(res.x, n)
            new_c = compute_c5(new_h, n)
            if new_c < best_c:
                best_h, best_c = new_h, new_c
        except Exception:
            pass
    return best_h, best_c

            best_h, best_c = h1, c1
    # Phase 2: LSE gradient descent (smooth to sharp)
    if time.time() < deadline - 350:
        h2, c2 = lse_grad_descent(best_h, n, [100, 300, 1000, 3000], deadline - 350, steps_per_alpha=200)
        if c2 < best_c:
            best_h, best_c = h2, c2
        h2, c2 = coord_descent(best_h, n, 60, deadline - 350, gs_iters=55)
        if c2 < best_c:
            best_h, best_c = h2, c2
    # Phase 3: SLSQP polish
    if time.time() < deadline - 250:
        h3, c3 = slsqp_polish(best_h, n, [2000, 8000], 150, deadline - 250)
        if c3 < best_c:
            best_h, best_c = h3, c3
        h3, c3 = coord_descent(best_h, n, 40, deadline - 250, gs_iters=55)
        if c3 < best_c:
            best_h, best_c = h3, c3
    # Phase 4: Multi-restart with diverse perturbations
    n_cur = len(best_h)
    x_orig = np.linspace(0, 2, n_cur)
    n_restarts = 14
    for r in range(n_restarts):
        if time.time() > deadline - 120:
            break
        scale = 0.015 + 0.04 * (r / max(1, n_restarts - 1))
        rmod = r % 7
        if rmod == 0:
            hp = best_h + rng.normal(0, scale, n_cur)
        elif rmod == 1:
            hp = best_h + rng.normal(0, 2.0 * scale, n_cur)
        elif rmod == 2:
            hp = best_h.copy()
            blk = max(10, n_cur // 8)
            start = rng.randint(0, max(1, n_cur - blk))
            hp[start:start + blk] = 1.0 - hp[start:start + blk]
        elif rmod == 3:
            shift = rng.randint(1, n_cur // 3)
            hp = np.roll(best_h, shift)
        elif rmod == 4:
            kern = np.ones(7) / 7.0
            hp = np.convolve(best_h, kern, mode='same') + rng.normal(0, 0.008, n_cur)
        elif rmod == 5:
            hp = best_h.copy()
            half = n_cur // 2
            hp[:half] = 1.0 - best_h[half:half + half]
            hp[half:] = best_h[:half]
        else:
            hp = best_h * (1.0 + 0.03 * rng.randn(n_cur))
        hp = project_feasible(hp, n_cur)
        h_r, c_r = coord_descent(hp, n_cur, 35, deadline - 120, gs_iters=50)
        if c_r < best_c:
            if time.time() < deadline - 120:
                h_r, c_r = lse_grad_descent(h_r, n_cur, [500, 2000, 10000], deadline - 120, steps_per_alpha=150)
                h_r, c_r = coord_descent(h_r, n_cur, 30, deadline - 120, gs_iters=50)
            if c_r < best_c:
                best_h, best_c = h_r, c_r
                x_orig = np.linspace(0, 2, n_cur)
    # Phase 5: n-sweep
    for n_try in [450, 550, 650, 750, 850, 950, 1100, 1300]:
        if time.time() > deadline - 50:
            break
        if n_try == len(best_h):
            continue
        x_new = np.linspace(0, 2, n_try)
        h_int = project_feasible(np.interp(x_new, x_orig, best_h), n_try)
        h_o, c_o = coord_descent(h_int, n_try, 50, deadline - 50, gs_iters=55)
        if c_o < best_c:
            if time.time() < deadline - 50:
                h_o, c_o = lse_grad_descent(h_o, n_try, [500, 2000, 10000], deadline - 50, steps_per_alpha=100)
                h_o, c_o = coord_descent(h_o, n_try, 30, deadline - 50, gs_iters=50)
            if c_o < best_c:
                best_h, best_c = h_o, c_o
                x_orig = np.linspace(0, 2, n_try)
    # Phase 6: Final polish
    n_fin = len(best_h)
    if time.time() < deadline - 8:
        h_f, c_f = slsqp_polish(best_h, n_fin, [5000, 20000], 200, deadline - 5)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
        h_f, c_f = coord_descent(best_h, n_fin, 60, deadline - 3, gs_iters=60)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
        h_f, c_f = lse_grad_descent(best_h, n_fin, [5000, 20000, 100000], deadline - 3, steps_per_alpha=100)
        if c_f < best_c:
            best_h, best_c = h_f, c_f
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
