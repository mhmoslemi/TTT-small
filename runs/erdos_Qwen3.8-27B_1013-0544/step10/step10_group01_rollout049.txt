def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * room, 0.0, 1.0)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.clip(h + (diff / tot) * h, 0.0, 1.0)
    return h

def topk_smooth_max(c, k, alpha):
    """Smooth approximation of max using top-k elements."""
    c_sorted = np.sort(c)[::-1]
    top = c_sorted[:k]
    m = top.max()
    shifted = alpha * (top - m)
    lse = m + np.log(np.sum(np.exp(shifted)))/alpha
    # Blend with global max for safety
    global_m = c.max()
    return max(lse, global_m)

            best_h, best_c = h2, c2

    # Phase 3: CD after SLSQP
    if time.time() < deadline - 200:
        h3, c3 = coord_descent(best_h, n, 50, deadline, gs_iters=80)
        if c3 < best_c:
            best_h, best_c = h3, c3

    # Phase 4: Multi-scale exploration
    if time.time() < deadline - 300:
        x_orig = np.linspace(0, 2, n)
        for n_try in [300, 450]:
            if time.time() > deadline - 120:
                break
            if n_try >= n:
                continue
            x_s = np.linspace(0, 2, n_try)
            h_s = project_feasible(np.interp(x_s, x_orig, best_h), n_try)
            h_s, c_s = coord_descent(h_s, n_try, 30, deadline, gs_iters=60)
            h_s, c_s = slsqp_topk(h_s, n_try, k=5, alphas=[500, 3000], maxiter_each=100)
            h_s, c_s = coord_descent(h_s, n_try, 25, deadline, gs_iters=70)
            h_back = project_feasible(np.interp(x_orig, x_s, h_s), n)
            h_back, c_back = coord_descent(h_back, n, 20, deadline, gs_iters=75)
            if c_back < best_c:
                best_h, best_c = h_back, c_back

    # Phase 5: Stuck-coordinate perturbations with reoptimization
    if time.time() < deadline - 150:
        n_cur = len(best_h)
        x = np.linspace(0, 2, n_cur)
        n_restarts = max(4, int((deadline - time.time() - 80) / 65))
        for r in range(n_restarts):
            if time.time() > deadline - 60:
                break
            if r % 4 == 0:
                hp = project_feasible(best_h + 0.006 * rng.randn(n_cur), n_cur)
            elif r % 4 == 1:
                hp = stuck_perturb(best_h, n_cur, rng, scale=0.04)
            elif r % 4 == 2:
                freq = 1.0 + 1.0 * (r // 4)
                hp = project_feasible(best_h + 0.015 * np.sin(2 * np.pi * freq * x), n_cur)
            else:
                # Flip-based: swap some 0/1 boundary values
                hp = best_h.copy()
                zeros = np.where(hp < 0.01)[0]
                ones = np.where(hp > 0.99)[0]
                if len(zeros) > 5 and len(ones) > 5:
                    sz = rng.choice(zeros, size=min(5, len(zeros)), replace=False)
                    so = rng.choice(ones, size=min(5, len(ones)), replace=False)
                    hp[sz] = 0.5
                    hp[so] = 0.5
                hp = project_feasible(hp, n_cur)
            h_o, c_o = coord_descent(hp, n_cur, 30, deadline - 15, gs_iters=65)
            h_o, c_o = slsqp_topk(h_o, n_cur, k=5, alphas=[1000, 5000], maxiter_each=80)
            h_o, c_o = coord_descent(h_o, n_cur, 20, deadline - 15, gs_iters=70)
            if c_o < best_c:
                best_h, best_c = h_o, c_o

    # Phase 6: Final intensive CD
    if time.time() < deadline - 10:
        h_f, c_f = coord_descent(best_h, len(best_h), 50, deadline, gs_iters=90)
        if c_f < best_c:
            best_h, best_c = h_f, c_f

    # Final check
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
