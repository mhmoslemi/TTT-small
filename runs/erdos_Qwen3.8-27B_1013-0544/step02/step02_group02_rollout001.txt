        cands.append(pat)
    return cands

def run_sa(h, n, rng, t0, budget_s, max_iters=5000000, t_start=0.05, t_end=1e-7):
    h = h.copy()
    h = np.where(h >= 0.5, 1.0, 0.0)
    h = project_feasible(h, n)
    T = t_start
    cool = (t_end / t_start) ** (1.0 / max_iters)
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    cur_c5 = best_c5
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    for it in range(max_iters):
        if (it & 0xFFF) == 0 and budget_s - (time.time() - t0) < 15:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        new_c5 = compute_c5(h, n)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c5 = new_c5
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h <= 0.5)[0]
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
        else:
            h[i], h[j] = 1.0, 0.0
        T *= cool
    # Continuous version of best
    best_cont = project_feasible(best_h, n)
    return best_cont, best_c5, best_h

def multi_start_sa(n, h_init, rng, t0, budget_s):
    cands = gen_candidates(n, rng, h_init)
    scored = []
    for c in cands:
        c = project_feasible(c, n)
        scored.append((compute_c5(c, n), c))
    scored.sort(key=lambda x: x[0])
    best_c5 = scored[0][0]
    best_h = scored[0][1]
    # SA on top candidates with time split
    top_k = min(4, len(scored))
    time_per = max(10, (budget_s - (time.time() - t0) - 20) / top_k)
    for score, hc in scored[:top_k]:
        remaining = budget_s - (time.time() - t0)
        if remaining < 15:
            break
        alloc = min(time_per, remaining - 10)
        if alloc < 5:
            break
        h_sa, c5_sa, h_bin = run_sa(hc, n, rng, t0, t0 + alloc, 5000000)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa
    return best_h, best_c5

def continuous_refine(h, n, t0, budget_s):
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    for alpha in [500, 2000, 10000, 50000, 200000]:
        if budget_s - (time.time() - t0) < 10:
            break
        ht = slsqp_opt(best_h, n, alpha, 5000, 1e-16)
        c = compute_c5(ht, n)
        if c < best_c5 - 1e-14:
            best_h, best_c5 = ht, c
    return best_h, best_c5

def targeted_search(h, n, rng, t0, budget_s):
    """Local search with targeted moves on continuous values."""
    h = h.copy()
    best_c5 = compute_c5(h, n)
    steps = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    for step in steps:
        if budget_s - (time.time() - t0) < 5:
            break
        for _ in range(200):
            if budget_s - (time.time() - t0) < 5:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            s = rng.choice([1.0, -1.0])
            hi = h[i] + s * step
            hj = h[j] - s * step
            if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                h[i], h[j] = hi, hj
                c = compute_c5(h, n)
                if c < best_c5:
                    best_c5 = c
                else:
                    h[i], h[j] = hi - s * step, hj + s * step
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - t0)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 80

    overall_best_h = None
    overall_best_c5 = 1.0
    overall_best_n = n_init

    # Determine which resolutions to try based on budget
    resolutions = []
    if n_init in [80, 160, 240]:
        resolutions.append(n_init)
    for n_try in [80, 160, 240]:
        if n_try not in resolutions:
            resolutions.append(n_try)

    # Allocate time: more to smaller n (faster iterations)
    total_time = budget_s - 30
    weights = [1.0 / r for r in resolutions]
    wsum = sum(weights)
    time_alloc = [total_time * w / wsum for w in weights]

    for idx_n, n in enumerate(resolutions):
        alloc = time_alloc[idx_n]
        if remaining() < 30:
            break
        # Adjust for already elapsed time
        alloc = min(alloc, remaining() - 20)
        if alloc < 15:
            break

        h_start = None
        if n == n_init and h_init is not None:
            h_start = h_init.copy()

        # Phase A: Multi-start SA
        h_sa, c5_sa = multi_start_sa(n, h_start, rng, t0, t0 + alloc * 0.55)
        if c5_sa < overall_best_c5:
            overall_best_c5 = c5_sa
            overall_best_h = h_sa.copy()
            overall_best_n = n

        # Phase B: Continuous refinement
        if remaining() > 15:
            h_ref, c5_ref = continuous_refine(h_sa, n, t0, alloc)
            if c5_ref < overall_best_c5:
                overall_best_c5 = c5_ref
                overall_best_h = h_ref.copy()
                overall_best_n = n

        # Phase C: Targeted local search
        if remaining() > 10:
            h_tgt, c5_tgt = targeted_search(overall_best_h, overall_best_n, rng, t0, alloc)
            if c5_tgt < overall_best_c5:
                overall_best_c5 = c5_tgt
                overall_best_h = h_tgt.copy()

    # Final SLSQP polish with high alpha
    if remaining() > 10:
        h_final, c5_final = continuous_refine(overall_best_h, overall_best_n, t0, budget_s)
        if c5_final < overall_best_c5:
            overall_best_c5 = c5_final
            overall_best_h = h_final

    overall_best_h = project_feasible(overall_best_h, overall_best_n)
    overall_best_c5 = compute_c5(overall_best_h, overall_best_n)
    return overall_best_h, overall_best_c5, overall_best_n
