```python
def coord_grad(h, j, n):
    grad = np.zeros(2*n-1)
    # d ranges from -(n-1) to n-1, index = d + (n-1)
    ds = np.arange(-(n-1), n)
    # First term: 1-h[j+d] where 0<=j+d<n
    jd = j + ds
    mask1 = (jd >= 0) & (jd < n)
    grad[mask1] += (1.0 - h[jd[mask1]])
    # Second term: -h[j-d] where 0<=j-d<n
    jdm = j - ds
    mask2 = (jdm >= 0) & (jdm < n)
    grad[mask2] -= h[jdm[mask2]]
    return grad
```

```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max(x, beta):
    m = np.max(x)
    if beta < 1:
        return m
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta


def project_h(h, target_sum):
    h = h.copy()
    for _ in range(50):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() > 0:
            h[free] += (target_sum - s) / free.sum()
        else:
            h += (target_sum - s) / len(h)
    return np.clip(h, 0.0, 1.0)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")


def coord_grad(h, j, n):
    grad = np.zeros(2 * n - 1)
    ds = np.arange(-(n - 1), n)
    jd = j + ds
    mask1 = (jd >= 0) & (jd < n)
    grad[mask1] += (1.0 - h[jd[mask1]])
    jdm = j - ds
    mask2 = (jdm >= 0) & (jdm < n)
    grad[mask2] -= h[jdm[mask2]]
    return grad


def obj_slsqp(h, beta, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, beta) * dx


def _con_fun(x):
    return np.sum(x) - len(x) / 2.0


def _con_jac(x):
    return np.ones(len(x))


def coord_descent_round(h, corr, n, passes=5):
    h = h.copy()
    for p in range(passes):
        for j in range(n):
            grad = coord_grad(h, j, n)
            lo = -h[j]
            hi = 1.0 - h[j]
            for _ in range(35):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1 = np.max(corr + m1 * grad)
                f2 = np.max(corr + m2 * grad)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
            delta = (lo + hi) / 2.0
            if abs(delta) > 1e-10:
                h[j] += delta
                corr = corr + delta * grad
    return h, corr


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    def tleft():
        return budget_s - (time.time() - start_time)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)

    n0 = len(h_init)
    target0 = n0 / 2.0
    h0 = project_h(h_init, target0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP warmup on n0
    n = n0
    dx = 2.0 / n
    target = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    constraints = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    h_cur = h0.copy()

    for beta in [100, 200, 400]:
        if tleft() < 60:
            break
        try:
            res = minimize(obj_slsqp, h_cur, args=(beta, dx),
                           method='SLSQP', bounds=bounds,
                           constraints=constraints,
                           options={'maxiter': 2000, 'ftol': 1e-14})
            h_new = project_h(res.x, target)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new.copy(), c5
                h_cur = h_new
        except Exception:
            pass

    # Phase 2: Coordinate descent on n0 (main novel phase)
    h_cd = best_h.copy()
    corr = compute_corr(h_cd)
    prev_c5 = best_c5

    max_rounds = 200
    for rd in range(max_rounds):

        h_cd = project_h(h_cd, target)
        corr = compute_corr(h_cd)
        c5 = np.max(corr) * dx

        if c5 < best_c5 - 1e-12:
            best_h, best_c5 = h_cd.copy(), c5
            h_cd = best_h.copy()
            corr = compute_corr(h_cd)
            prev_c5 = c5
        elif abs(c5 - prev_c5) < 1e-10 and rd > 5:
            # Perturb to escape local minimum
            intensity = 0.03 + 0.04 * (rd % 5) / 5.0
            n_pert = np.random.randint(5, min(15, n // 4 + 1))
            idx = np.random.choice(n, n_pert, replace=False)
            h_cd[idx] += np.random.randn(n_pert) * intensity
            h_cd = project_h(h_cd, target)
            corr = compute_corr(h_cd)
            prev_c5 = np.max(corr) * dx

    # Phase 3: SLSQP polish after coordinate descent
    if tleft() > 40:
        for beta in [500, 800]:
            if tleft() < 20:
                break
            try:
                res = minimize(obj_slsqp, best_h, args=(beta, dx),
                               method='SLSQP', bounds=bounds,
                               constraints=constraints,

                if c5 < best_c5:
                    best_h, best_c5 = h_new.copy(), c5
            except Exception:
            pass

    # Phase 4: Try different n values
    for n_try in [80, 90, 60, 100, 120]:
        if tleft() < 50:
            break
        if n_try == n0:
            continue
        dx_try = 2.0 / n_try
        target_try = n_try / 2.0

        # Interpolate from best solution
        x_old = np.linspace(0, 2, len(best_h), endpoint=False) + 1.0 / len(best_h)
        x_new = np.linspace(0, 2, n_try, endpoint=False) + 1.0 / n_try
        h_try = np.interp(x_new, x_old, best_h)
        h_try = project_h(h_try, target_try)

        bounds_try = Bounds(np.zeros(n_try), np.ones(n_try))
        constraints_try = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]

        # Quick SLSQP
        if tleft() > 40:

                               method='SLSQP', bounds=bounds_try,
                               constraints=constraints_try,
                               options={'maxiter': 1500, 'ftol': 1e-13})
                h_try = project_h(res.x, target_try)
            except Exception:
                pass

        # Coordinate descent
        if tleft() > 30:
            corr_try = compute_corr(h_try)
            for _ in range(15):
                if tleft() < 10:

                h_try, corr_try = coord_descent_round(h_try, corr_try, n_try, passes=3)
                h_try = project_h(h_try, target_try)
                corr_try = compute_corr(h_try)

        # More coordinate descent if promising
        if c5_try < best_c5 + 0.001 and tleft() > 40:
            corr_try = compute_corr(best_h)
            n = len(best_h)
            dx = 2.0 / n
            target = n / 2.0
            for _ in range(30):
                if tleft() < 15:

                best_h, corr_try = coord_descent_round(best_h, corr_try, n, passes=3)
                best_h = project_h(best_h, target)
                corr_try = compute_corr(best_h)
                c5_chk = np.max(corr_try) * dx
                if c5_chk < best_c5:
                    best_c5 = c5_chk

    # Final polish
    if tleft() > 15:
        n = len(best_h)
        dx = 2.0 / n
        target = n / 2.0
        corr = compute_corr(best_h)

    final_c5 = compute_c5(best_h)

    return best_h, final_c5, len(best_h)
```
