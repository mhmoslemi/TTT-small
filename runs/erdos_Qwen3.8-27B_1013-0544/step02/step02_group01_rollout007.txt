    target = n / 2.0
    x = project_feasible(h_init.astype(float), target)
    x_prev = x.copy()
    best_x = x.copy()
    best_v = compute_c5(x)
    j = np.arange(n)
    for t in range(max_iters):
        lr = lr0 * (decay ** t)
        if t == 0:
            y = x.copy()
        else:
            mu = (t - 1) / (t + 2.0)
            y = x + mu * (x - x_prev)
        y = project_feasible(y, target)
        corr = np.correlate(y, 1.0 - y, mode="full")
        top_idx = np.argsort(corr)[-top_k:]
        lags = top_idx - (n - 1)
        grad = np.zeros(n)
        for s in lags:
            js = j - s
            jp = j + s
            ms = (js >= 0) & (js < n)
            mp = (jp >= 0) & (jp < n)
            if ms.any():
                grad[ms] += (1.0 - y[js[ms]])
            if mp.any():
                grad[mp] += (-y[jp[mp]])
        grad = grad / top_k * (2.0 / n)
        x_new = project_feasible(y - lr * grad, target)
        v = compute_c5(x_new)
        if v < best_v:
            best_v = v
            best_x = x_new.copy()
        x_prev = x
        x = x_new
    return best_x, best_v


def binary_local_search(h, max_rounds=50):
    n = len(h)
    target = n / 2.0
    x = h.copy().astype(float)
    best_v = compute_c5(x)
    for _ in range(max_rounds):
        improved = False
        for i in range(n):
            old = x[i]
            for nv in [0.0, 1.0]:
                if abs(old - nv) < 1e-8:
                    continue
                x[i] = nv
                x = project_feasible(x, target)
                v = compute_c5(x)
                if v < best_v:
                    best_v = v
                    improved = True
                else:
                    x[i] = old
                    x = project_feasible(x, target)
        if not improved:
            break
    return x, best_v


def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.full(200, 0.5)

    best_c5 = compute_c5(h_best)

    # Phase 1: Adam with medium beta
    remaining = budget_s - (time.time() - t0)
    if remaining > 60:
        h_a, c_a = adam_optimize(h_best, 4000, 0.005, 200)
        if c_a < best_c5:
            h_best, best_c5 = h_a, c_a

    # Phase 2: Adam with higher beta
    remaining = budget_s - (time.time() - t0)
    if remaining > 40:
        h_a, c_a = adam_optimize(h_best, 3000, 0.002, 500)
        if c_a < best_c5:
            h_best, best_c5 = h_a, c_a

    # Phase 3: Nesterov top-k for exploration
    remaining = budget_s - (time.time() - t0)
    if remaining > 40:
        h_a, c_a = nesterov_topk(h_best, 5000, 0.04, 0.9997, top_k=8)
        if c_a < best_c5:
            h_best, best_c5 = h_a, c_a

    # Phase 4: Adam with very high beta for precision
    remaining = budget_s - (time.time() - t0)
    if remaining > 30:
        h_a, c_a = adam_optimize(h_best, 4000, 0.001, 1000)
        if c_a < best_c5:
            h_best, best_c5 = h_a, c_a

    # Phase 5: Binary local search
    remaining = budget_s - (time.time() - t0)
    if remaining > 30:
        h_b, c_b = binary_local_search(h_best, max_rounds=80)
        if c_b < best_c5:
            h_best, best_c5 = h_b, c_b

    # Phase 6: After binary search, refine with Adam
    remaining = budget_s - (time.time() - t0)
    if remaining > 30:
        h_a, c_a = adam_optimize(h_best, 3000, 0.001, 500)
        if c_a < best_c5:
            h_best, best_c5 = h_a, c_a

    # Phase 7: Try different resolutions
    for n_try in [180, 220, 250]:
        remaining = budget_s - (time.time() - t0)
        if remaining < 60:
            break
        h_interp = interp_to_n(h_best, n_try)
        h_a, c_a = adam_optimize(h_interp, 2500, 0.003, 300)
        if c_a < best_c5:
            h_best, best_c5 = h_a, c_a
        remaining = budget_s - (time.time() - t0)
        if remaining < 30:
            break
        h_a, c_a = adam_optimize(h_best, 2000, 0.001, 800)
        if c_a < best_c5:
            h_best, best_c5 = h_a, c_a

    # Phase 8: Random restarts
    for _ in range(8):
        remaining = budget_s - (time.time() - t0)
        if remaining < 25:
            break
        h_pert = h_best + 0.015 * np.random.randn(len(h_best))
        h_pert = project_feasible(h_pert, len(h_best) / 2.0)
        h_a, c_a = adam_optimize(h_pert, 1500, 0.003, 400)
        if c_a < best_c5:
            h_best, best_c5 = h_a, c_a

    # Phase 9: Final fine polish with very small lr
    remaining = budget_s - (time.time() - t0)
    if remaining > 15:
        h_a, c_a = adam_optimize(h_best, 5000, 0.0003, 1500)
        if c_a < best_c5:
            h_best, best_c5 = h_a, c_a

    # Phase 10: Ultra-fine with highest beta
    remaining = budget_s - (time.time() - t0)
    if remaining > 10:
        h_a, c_a = adam_optimize(h_best, 3000, 0.0001, 3000)
        if c_a < best_c5:
            h_best, best_c5 = h_a, c_a

    # Final: binary local search again
    remaining = budget_s - (time.time() - t0)
    if remaining > 15:
        h_b, c_b = binary_local_search(h_best, max_rounds=40)
        if c_b < best_c5:
            h_best, best_c5 = h_b, c_b

    return h_best, best_c5, len(h_best)
