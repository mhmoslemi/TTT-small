            h_new = project_h(res.x, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
        except Exception:
            pass

    # Phase 3: Perturbation + SLSQP refine (main search loop)
    attempt = 0
    no_improve = 0
    while time_left() > 50:
        attempt += 1
        h0 = best_h.copy()
        strat = attempt % 8

        if strat == 0:
            scale = 0.008 * (1.0 + max(0, 30 - no_improve) * 0.1)
            h0 += rng.randn(n_points) * scale
        elif strat == 1:
            # Targeted: reduce active lag
            corr = corr_arr(h0)
            max_idx = np.argmax(corr)
            lag = max_idx - (n_points - 1)
            grad = compute_gradient_at_lag(h0, lag)
            h0 -= 0.015 * grad
        elif strat == 2:
            i, j = rng.choice(n_points, 2, replace=False)
            d = 0.06 * rng.randn()
            h0[i] += d
            h0[j] -= d
        elif strat == 3:
            blen = rng.randint(3, 8)
            bs = rng.randint(0, n_points - blen)
            h0[bs:bs + blen] += rng.randn(blen) * 0.025
        elif strat == 4:
            # Flatten toward mean
            h0 = 0.93 * h0 + 0.07 * np.mean(h0)
        elif strat == 5:
            # Mirror mix
            h0 = 0.7 * h0 + 0.3 * h0[::-1]
            h0 += rng.randn(n_points) * 0.005
        elif strat == 6:
            # Multi-lag gradient
            corr = corr_arr(h0)
            top3 = np.argsort(corr)[-3:]
            grad = np.zeros(n_points)
            for m in top3:
                lag = m - (n_points - 1)
                grad += compute_gradient_at_lag(h0, lag)
            h0 -= 0.01 * grad / (np.linalg.norm(grad) + 1e-10)
        else:
            # Random block perturbation
            nb = rng.randint(2, 5)
            for _ in range(nb):
                st = rng.randint(0, n_points)
                ln = rng.randint(1, 5)
                h0[st:min(st + ln, n_points)] += rng.randn() * 0.02

        h0 = project_h(h0, target_sum)

        beta = 300 if attempt % 2 == 0 else 800
        def obj_pf(x, b=beta):
            return smooth_max(corr_arr(x), b) * dx
        try:
            res = minimize(obj_pf, h0, method='SLSQP', bounds=bounds,
                           constraints=constraints, options={'maxiter': 3000, 'ftol': 1e-14})
            h_new = project_h(res.x, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
                no_improve = 0
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

        # Periodic subgradient restart if stuck
        if no_improve >= 15 and time_left() > 80:
            h0 = best_h.copy()
            h0 = subgradient_minimax(h0, target_sum, dx, 500, 0.2)
            c5 = compute_c5(h0)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h0.copy()
            no_improve = 0

    # Phase 4: Final high-precision polish
    if time_left() > 15:
        for beta in [3000, 8000]:
            if time_left() < 8:
                break
            def obj_final(x, b=beta):
                return smooth_max(corr_arr(x), b) * dx
            try:
                res = minimize(obj_final, best_h, method='SLSQP', bounds=bounds,
                               constraints=constraints, options={'maxiter': 10000, 'ftol': 1e-16})
                h_new = project_h(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()
            except Exception:
                pass

    best_h = project_h(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_points
