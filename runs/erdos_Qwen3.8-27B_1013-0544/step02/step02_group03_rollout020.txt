import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-12:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def subgrad_full(h, n):
    """Compute subgradient of max_k corr[k] w.r.t. h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_worst = np.argmax(corr)
    lag = k_worst - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        idx = np.arange(n - lag)
        grad[idx] += (1.0 - h[idx + lag])
        grad[idx + lag] -= h[idx]
    else:
        la = -lag
        idx = np.arange(n - la)
        grad[idx + la] += (1.0 - h[idx])
        grad[idx] -= h[idx + la]
    return grad, k_worst, corr

def subgrad_multi(h, n, n_active=3):
    """Subgradient averaging over top n_active lags."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    # Find top n_active lags
    top_idx = np.argsort(corr)[-n_active:]
    grad = np.zeros(n)

            grad[idx] += (1.0 - h[idx + lag])
            grad[idx + lag] -= h[idx]
        else:
            la = -lag
            idx = np.arange(n - la)
            grad[idx + la] += (1.0 - h[idx])
            grad[idx] -= h[idx + la]
    grad /= n_active
    return grad, corr

    grad = np.zeros(len(h))

        return budget_s - (time.time() - t0)

    # Phase 1: SLSQP with analytic gradient, continuation on beta
    beta_schedule = [5, 20, 80, 300, 1000, 3000, 10000, 30000]
    for beta in beta_schedule:
        if time_left() < 70:
            break
        for trial in range(4):
            if time_left() < 35:
                break
            x0 = best_h.copy()
            if trial > 0:
                amp = 0.02 / trial
                x0 += np.random.randn(n) * amp
                x0 = project_feasible(x0, target_sum)
            try:
                res = minimize(objective_and_grad, x0, args=(dx, beta),
                               method='SLSQP', bounds=bounds,
                               constraints=cons, jac=True,
                               options={'maxiter': 8000, 'ftol': 1e-15})
                h_new = project_feasible(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5
            except Exception:
                pass

    # Phase 2: Momentum subgradient with periodic restart
    if time_left() > 120:
        h = best_h.copy()
        v = np.zeros(n)
        mu = 0.85
        base_lr = 0.004
        no_improve = 0
        for it in range(300000):
            if time_left() < 50:
                break
            corr = np.correlate(h, 1.0 - h, mode='full')
            k_worst = np.argmax(corr)
            grad = lag_gradient(h, k_worst, n)
            lr = base_lr / np.sqrt(1.0 + 0.003 * it)
            v = mu * v - lr * grad
            h_new = project_feasible(h + v, target_sum)
            if it % 5 == 0:
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    no_improve = 0
                else:
                    no_improve += 1
                if c5_new < compute_c5(h) + 1e-12:
                    h = h_new
                else:
                    v *= 0.3
            else:
                h = h_new
            if no_improve > 3000:
                h = best_h.copy()
                v = np.random.randn(n) * 0.01
                no_improve = 0

    # Phase 3: Binary heuristic - try rounded versions
    if time_left() > 60:
        h_bin = binary_project(best_h, target_sum)
        c5_bin = compute_c5(h_bin)
        if c5_bin < best_c5:
            best_h = h_bin.copy()
            best_c5 = c5_bin
        # SLSQP from binary point
        for beta in [500, 5000]:
            if time_left() < 25:
                break
            try:
                res = minimize(objective_and_grad, h_bin.copy(), args=(dx, beta),
                               method='SLSQP', bounds=bounds,
                               constraints=cons, jac=True,
                               options={'maxiter': 6000, 'ftol': 1e-15})
                h_new = project_feasible(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5
            except Exception:
                pass

    # Phase 4: Multi-resolution search
    if time_left() > 80:
        n_candidates = [n + 10, n + 20, n - 10, n + 30, int(n * 1.4), int(n * 0.7)]
        for n_try in n_candidates:
            if n_try < 30 or n_try % 2 != 0 or time_left() < 35:
                continue
            dx_t = 2.0 / n_try
            ts_t = n_try / 2.0
            _target_sum[0] = ts_t
            cons_t = [{'type': 'eq', 'fun': _constraint_fun, 'jac': _constraint_jac}]
            h_t = resample_h(best_h, n_try)
            h_t = project_feasible(h_t, ts_t)
            bnds_t = Bounds(np.zeros(n_try), np.ones(n_try))
            for beta in [500, 5000]:
                if time_left() < 15:
                    break
                try:
                    res = minimize(objective_and_grad, h_t, args=(dx_t, beta),
                                   method='SLSQP', bounds=bnds_t,
                                   constraints=cons_t, jac=True,
                                   options={'maxiter': 5000, 'ftol': 1e-14})
                    h_new = project_feasible(res.x, ts_t)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5
                        n = n_try
                        dx = dx_t
                        target_sum = ts_t
                        bounds = bnds_t
                        _target_sum[0] = ts_t
                except Exception:
                    pass

    # Phase 5: Final intensive SLSQP polish at best resolution
    _target_sum[0] = len(best_h) / 2.0
    n_final = len(best_h)
    dx_final = 2.0 / n_final
    ts_final = n_final / 2.0
    bounds_final = Bounds(np.zeros(n_final), np.ones(n_final))
    cons_final = [{'type': 'eq', 'fun': _constraint_fun, 'jac': _constraint_jac}]
    if time_left() > 20:
        for beta in [50000, 200000]:
            if time_left() < 10:
                break
            for trial in range(2):
                if time_left() < 6:
                    break
                x0 = best_h.copy()
                if trial > 0:
                    x0 += np.random.randn(n_final) * 0.002
                    x0 = project_feasible(x0, ts_final)
                try:
                    res = minimize(objective_and_grad, x0, args=(dx_final, beta),
                                   method='SLSQP', bounds=bounds_final,
                                   constraints=cons_final, jac=True,
                                   options={'maxiter': 15000, 'ftol': 1e-17})
                    h_new = project_feasible(res.x, ts_final)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5
                except Exception:
                    pass

    return best_h, best_c5, len(best_h)
