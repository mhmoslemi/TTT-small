import time

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = h.copy()
    for _ in range(50):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()

        if free.sum() > 0:
            h[free] += diff / free.sum()
        else:
            h += diff / len(h)

    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

    return smooth_max(corr, beta)

def slsqp_con_jac(x):

def slsqp_polish(h, target_sum, beta=200, maxiter=2000):
    """Polish with SLSQP."""
    n = len(h)
    bounds = Bounds(np.zeros(n), np.ones(n))
    constraints = [{'type': 'eq', 'fun': slsqp_con, 'jac': slsqp_con_jac}]
    try:
        res = minimize(slsqp_obj, h, args=(beta,),
                       method='SLSQP', bounds=bounds,

Let me refine: for a 2-coordinate move (i, j), with h[i] + h[j] = c (fixed), the feasible set is the line segment from (max(0,c-1), min(1,c)) to (min(1,c), max(0,c-1)) in (h[i], h[j]) space. The objective is piecewise-linear along this segment, so the minimum is at one of the endpoints or at a kink. The endpoints correspond to h[i]=max(0,c-1) or h[i]=min(1,c), which are the boundary points.

So the best 2-coordinate move is always to push to a boundary! This means I should try:
- h[i] = max(0, c-1), h[j] = c - h[i]
- h[i] = min(1, c), h[j] = c - h[i]

Where c = h[i] + h[j].

This is much more efficient than trying arbitrary deltas!

    For each pair (i,j), try the two boundary points of the feasible segment.

    best = compute_c5(h)

                h_i_1 = max(0.0, c - 1.0)

            if c5_s < best_c5:
                best_h, best_c5 = h_s, c5_s
                no_improve_rounds = 0

        # SLSQP at higher beta
        if tleft() > 50 and no_improve_rounds >= 1:
            h_p, c5_p = slsqp_polish(best_h, target_sum, 2000, 5000)
            if c5_p < best_c5:
                best_h, best_c5 = h_p, c5_p
                h_c, c5_c = two_boundary_descent(best_h, target_sum, max_sweeps=1000, tol=1e-15)
                if c5_c < best_c5:
                    best_h, best_c5 = h_c, c5_c
                no_improve_rounds = 0

        # Perturb and re-optimize if stuck
        if no_improve_rounds >= 2 and tleft() > 120:
            n_pert = np.random.randint(4, 12)
            idx = np.random.choice(n, n_pert, replace=False)
            h_pert = best_h.copy()
            h_pert[idx] += np.random.randn(n_pert) * (0.08 + 0.05 * no_improve_rounds)
            h_pert = project_h(h_pert, target_sum)
            h_new, c5_new = two_boundary_descent(h_pert, target_sum, max_sweeps=2000, tol=1e-15)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new
                no_improve_rounds = 0
            else:
                # Try annealing from perturbed
                h_a, c5_a = anneal_targeted(h_pert, target_sum, 3000, 0.02, 0.0001, 0.999)
                if c5_a < best_c5:
                    best_h, best_c5 = h_a, c5_a
                    h_c, c5_c = two_boundary_descent(best_h, target_sum, max_sweeps=1500, tol=1e-15)
                    if c5_c < best_c5:
                        best_h, best_c5 = h_c, c5_c
                    no_improve_rounds = 0

    # Phase 5: Final polish
    if tleft() > 25:
        h_p, c5_p = slsqp_polish(best_h, target_sum, 5000, 5000)
        if c5_p < best_c5:
            best_h, best_c5 = h_p, c5_p
        h_c, c5_c = two_boundary_descent(best_h, target_sum, max_sweeps=3000, tol=1e-16)
        if c5_c < best_c5:
            best_h, best_c5 = h_c, c5_c

    # Final verification
    best_h = project_h(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
