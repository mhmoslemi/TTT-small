import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        free = (h > 1e-8) & (h < 1.0 - 1e-8)
        if free.sum() > 0:
            h[free] += diff / free.sum()
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smax_grad(h, beta):
    """Compute smooth max value and gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(beta * shifted)
    sw = np.sum(exps)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    
    grad = np.zeros(n)
    j = np.arange(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-12:
            continue
        s = k - (n - 1)
        # d/dh[m] c_s = (1-h[m-s])*[0<=m-s<n] + (-h[m+s])*[0<=m+s<n]
        idx1 = (j - s >= 0) & (j - s < n)
        if idx1.any():
            grad[idx1] += wk * (1.0 - h[j[idx1] - s])
        idx2 = (j + s >= 0) & (j + s < n)
        if idx2.any():
            grad[idx2] += wk * (-h[j[idx2] + s])
    grad *= dx
    return val, grad

def slsqp_optimize(h0, beta, maxiter=500):
    n = len(h0)
    target = n / 2.0
    
    def obj(x):
        v, _ = smax_grad(x, beta)
        return v
    
    def objg(x):
        _, g = smax_grad(x, beta)
        return g
    
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h0, jac=objg, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h0.copy(), target)

def interp_to_n(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

def nesterov_subgrad(h, max_iters, lr0, decay, top_k):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    x = h.copy().astype(float)
    x_prev = x.copy()
    best_x = x.copy()
    best_v = compute_c5(x)
    j = np.arange(n)
    for t in range(max_iters):
        lr = lr0 * (decay ** t)
        if t == 0:
            y = x.copy()
        else:
            mu = (t - 1) / (t + 2)
            y = x + mu * (x - x_prev)
        y = project_feasible(y, target)
        corr = np.correlate(y, 1.0 - y, mode="full")
        top_idx = np.argsort(corr)[-top_k:]
        lags = top_idx - (n - 1)
        grad = np.zeros(n)
        for s in lags:
            idx1 = (j - s >= 0) & (j - s < n)
            if idx1.any():
                grad[idx1] += (1.0 - y[j[idx1] - s])
            idx2 = (j + s >= 0) & (j + s < n)
            if idx2.any():
                grad[idx2] += (-y[j[idx2] + s])
        grad = grad / len(lags) * dx
        x_new = project_feasible(y - lr * grad, target)
        v = compute_c5(x_new)
        if v < best_v:
            best_v = v
            best_x = x_new.copy()
        x_prev = x
        x = x_new
    return best_x, best_v

    np.random.seed(seed)
    t0 = time.time()
    
    def remaining():
        return budget_s - (time.time() - t0)

import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        free = (h > 1e-8) & (h < 1.0 - 1e-8)
        if free.sum() > 0:
            h[free] += diff / free.sum()
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smax_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(beta * shifted)
    sw = np.sum(exps)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    j = np.arange(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-13:
            continue
        s = k - (n - 1)
        i1 = (j - s >= 0) & (j - s < n)
        if i1.any():
            grad[i1] += wk * (1.0 - h[j[i1] - s])
        i2 = (j + s >= 0) & (j + s < n)
        if i2.any():
            grad[i2] += wk * (-h[j[i2] + s])
    grad *= dx
    return val, grad

def slsqp_optimize(h0, beta, maxiter=500, ftol=1e-16):
    n = len(h0)
    target = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(smax_grad, h0, args=(beta,), method='SLSQP',
                       bounds=bounds, constraints=cons, jac=True,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h0.copy(), target)

def interp_to_n(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

def nesterov_subgrad(h, max_iters, lr0, decay, top_k):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    x = h.copy().astype(float)
    x_prev = x.copy()
    best_x = x.copy()
    best_v = compute_c5(x)
    j = np.arange(n)
    for t in range(max_iters):
        lr = lr0 * (decay ** t)
        if t == 0:
            y = x.copy()
        else:
            mu = (t - 1) / (t + 2.0)
            y = x + mu * (x - x_prev)
        y = project_feasible(y, target)
        corr = np.correlate(y, 1.0 - y, mode="full")
        top_idx = np.argsort(corr)[-top_k:]
        lags = top_idx - (n - 1)
        grad = np.zeros(n)
        for s in lags:
            i1 = (j - s >= 0) & (j - s < n)
            if i1.any():
                grad[i1] += (1.0 - y[j[i1] - s])
            i2 = (j + s >= 0) & (j + s < n)
            if i2.any():
                grad[i2] += (-y[j[i2] + s])
        grad = grad / len(lags) * dx
        x_new = project_feasible(y - lr * grad, target)
        v = compute_c5(x_new)
        if v < best_v:
            best_v = v
            best_x = x_new.copy()
        x_prev = x
        x = x_new
    return best_x, best_v

def block_guide(h, n_blocks):
    """Generate a lower-resolution guide from h by block averaging."""
    n = len(h)
    block_sizes = np.zeros(n_blocks, dtype=int)
    block_sizes[:n // n_blocks] = n // n_blocks
    block_sizes[:n % n_blocks] += 1
    h_blocks = np.zeros(n_blocks)
    start = 0
    for b in range(n_blocks):
        end = start + block_sizes[b]
        h_blocks[b] = h[start:end].mean()
        start = end
    h_blocks = np.clip(h_blocks, 0.05, 0.95)
    target = n_blocks / 2.0
    return project_feasible(h_blocks, target)
