import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx), corr


def project_h(h, target_sum):
    h = h.copy().astype(float)
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() > 0:
            h[free] += diff / free.sum()
        else:
            break
    return np.clip(h, 0.0, 1.0)


def subgrad_c5(h, corr, beta_w=30.0):
    n = len(h)
    max_c = np.max(corr)
    weights = np.exp(beta_w * (corr - max_c))
    w_sum = weights.sum()
    if w_sum < 1e-30:
        weights = np.zeros(len(corr))
        weights[np.argmax(corr)] = 1.0
    else:
        weights /= w_sum
    grad = np.zeros(n)
    n_lags = len(corr)
    for k_idx in range(n_lags):
        w = weights[k_idx]
        if w < 1e-8:
            continue
        k = k_idx - (n - 1)
        if k >= 0:
            if k == 0:
                grad += w * (1.0 - 2.0 * h)
            else:
                grad[k:] += w * (1.0 - h[:n-k])
                grad[:n-k] += w * (-h[k:])
        else:
            m = -k
            grad[:n-m] += w * (1.0 - h[m:])
            grad[m:] += w * (-h[:n-m])
    return grad


def targeted_move(h, corr, n, target):
    h_new = h.copy()
    max_c = np.max(corr)
    active = np.where(corr > max_c - 0.02 * max_c)[0]
    for k_idx in active[:6]:
        k = k_idx - (n - 1)
        if k >= 0 and k < n:
            valid = n - k
            contrib = h[k:k+valid] * (1.0 - h[:valid])
            idx_max = np.argmax(contrib)
            i = idx_max
            if h[i+k] > 0.3 and h[i] < 0.7:
                h_new[i+k] -= 0.02 * np.random.random()
            elif h[i+k] > 0.5:
                h_new[i+k] -= 0.03 * np.random.random()
        elif k < 0:
            m = -k
            if m >= n:
                continue
            valid = n - m
            contrib = h[:valid] * (1.0 - h[m:m+valid])
            idx_max = np.argmax(contrib)
            i = idx_max
            if h[i] > 0.3 and h[i+m] < 0.7:
                h_new[i] -= 0.02 * np.random.random()
            elif h[i] > 0.5:
                h_new[i] -= 0.03 * np.random.random()
    return project_h(h_new, target)


def coordinate_bipolarize(h, target, n_iterations=200, time_limit=None, start_time=None):
    h = h.copy()
    c5, corr = compute_c5(h)
    for _ in range(n_iterations):
        if time_limit is not None and time.time() - start_time > time_limit:
            break
        i = np.random.randint(len(h))
        if h[i] > 0.5:
            h_new = h.copy()
            h_new[i] = 1.0
            h_new = project_h(h_new, target)
            c5_new, _ = compute_c5(h_new)
            if c5_new <= c5:
                h = h_new
                c5 = c5_new
        else:
            h_new = h.copy()
            h_new[i] = 0.0
            h_new = project_h(h_new, target)
            c5_new, _ = compute_c5(h_new)
            if c5_new <= c5:
                h = h_new
                c5 = c5_new
    return h, c5


def slsqp_phase(h, n, target, beta, maxiter, ftol, time_left_fn):
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    constraints = None

    def con_fun(x):
        return x.sum() - target

    def con_jac(x):
        return np.ones(n)

    constraints = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]

    def obj_fn(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = np.max(corr)
        return (m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta) * dx

    try:
        res = minimize(obj_fn, h, method='SLSQP', bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_new = project_h(res.x, target)
        c5_new, _ = compute_c5(h_new)
        return h_new, c5_new
    except Exception:
        return h.copy(), None


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    def tleft():
        return budget_s - (time.time() - start_time)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)

    n = len(h_init)
    target = n / 2.0
    best_h = project_h(h_init, target)
    best_c5, _ = compute_c5(best_h)

    # Phase 1: SLSQP warmup
    if tleft() > 60:
        h1, c5_1 = slsqp_phase(best_h, n, target, beta=80, maxiter=1500,
                               ftol=1e-12, time_left_fn=tleft)
        if c5_1 is not None and c5_1 < best_c5:
            best_h, best_c5 = h1, c5_1
        h1, c5_1 = slsqp_phase(best_h, n, target, beta=200, maxiter=2000,
                               ftol=1e-13, time_left_fn=tleft)
        if c5_1 is not None and c5_1 < best_c5:
            best_h, best_c5 = h1, c5_1

    # Phase 2: Subgradient descent with backtracking
    h = best_h.copy()
    c5, corr = compute_c5(h)
    alpha = 0.015
    no_improve = 0
    patience = 300
    stuck_count = 0

    while tleft() > 50:
        grad = subgrad_c5(h, corr, beta_w=25.0)
        h_new = h - alpha * grad
        h_new = project_h(h_new, target)
        c5_new, corr_new = compute_c5(h_new)

        if c5_new < c5 - 1e-10:
            h = h_new
            c5 = c5_new
            corr = corr_new
            no_improve = 0
            alpha = min(alpha * 1.03, 0.06)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
        else:
            no_improve += 1
            alpha *= 0.75
            if no_improve > patience:
                # Try targeted move
                h_t = targeted_move(h, corr, n, target)
                c5_t, _ = compute_c5(h_t)
                if c5_t < c5:
                    h = h_t
                    c5 = c5_t
                    corr = np.correlate(h, 1.0 - h, mode="full")
                    if c5 < best_c5:
                        best_h = h.copy()
                        best_c5 = c5
                # Add small perturbation
                n_pert = np.random.randint(3, min(12, n // 4 + 1))
                idx = np.random.choice(n, n_pert, replace=False)
                h[idx] += np.random.randn(n_pert) * 0.02
                h = project_h(h, target)
                c5, corr = compute_c5(h)
                no_improve = 0
                alpha = 0.015
                stuck_count += 1
                if stuck_count > 8:
                    break

    # Phase 3: Coordinate bipolarization
    if tleft() > 60:
        h_bp, c5_bp = coordinate_bipolarize(best_h, target,
                                            n_iterations=30000,
                                            time_limit=tleft() - 30,
                                            start_time=start_time)
        if c5_bp < best_c5:
            best_h, best_c5 = h_bp, c5_bp

    # Phase 4: More subgradient rounds from improved point
    if tleft() > 80:
        h = best_h.copy()
        c5, corr = compute_c5(h)
        alpha = 0.01
        no_improve = 0
        while tleft() > 30:
            grad = subgrad_c5(h, corr, beta_w=40.0)
            h_new = h - alpha * grad
            h_new = project_h(h_new, target)
            c5_new, corr_new = compute_c5(h_new)
            if c5_new < c5 - 1e-10:
                h = h_new
                c5 = c5_new
                corr = corr_new
                no_improve = 0
                alpha = min(alpha * 1.02, 0.04)
                if c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = c5
            else:
                no_improve += 1
                alpha *= 0.8
                if no_improve > 200:
                    break

    # Phase 5: Final SLSQP polish
    if tleft() > 20:
        h5, c5_5 = slsqp_phase(best_h, n, target, beta=500, maxiter=3000,
                               ftol=1e-14, time_left_fn=tleft)
        if c5_5 is not None and c5_5 < best_c5:
            best_h, best_c5 = h5, c5_5
        if tleft() > 15:
            h5, c5_5 = slsqp_phase(best_h, n, target, beta=1000, maxiter=2000,
                                   ftol=1e-15, time_left_fn=tleft)
            if c5_5 is not None and c5_5 < best_c5:
                best_h, best_c5 = h5, c5_5

    # Final verification
    best_h = project_h(best_h, target)
    best_c5, _ = compute_c5(best_h)
    return best_h, best_c5, n
