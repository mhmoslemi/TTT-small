    target = n / 2.0
    dx = 2.0 / n
    
    x = project_v(h_init, target)
    y = x.copy()
    t = 1.0
    best_c5 = compute_c5(x)
    best_x = x.copy()
    
    for k in range(max_iter):
        g = grad_sm(y, beta, dx)
        
        # Gradient step
        z = y - step_size * g
        
        # Project
        x_new = project_v(z, target)
        
        # Check improvement
        c5 = compute_c5(x_new)
        if c5 < best_c5:
            best_c5 = c5
            best_x = x_new.copy()
        
        # FISTA update
        t_new = (1 + np.sqrt(1 + 4*t*t)) / 2
        y = x_new + ((t - 1) / t_new) * (x_new - x)
        x = x_new
        t = t_new
    
    return best_x, best_c5

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_v(h, target_sum):
    lo = -h.max() - 1.0
    hi = 1.0 - h.min()
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def obj_val(h, beta, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    s = corr * beta * dx
    m = s.max()
    return float(m + np.log(np.sum(np.exp(s - m))) / beta)

def obj_grad(h, beta, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    s = corr * beta * dx
    m = s.max()
    w = np.exp(s - m)
    w /= w.sum()
    i_idx = np.arange(n)
    lags = np.arange(-(n-1), n)
    I = i_idx[:, None]
    L = lags[None, :]
    il = I - L
    il2 = I + L
    v1 = (il >= 0) & (il < n)
    v2 = (il2 >= 0) & (il2 < n)
    term = np.zeros((n, 2*n-1), dtype=np.float64)
    r1, c1 = np.where(v1)
    if len(r1) > 0:
        term[r1, c1] += 1.0 - h[il[r1, c1]]
    r2, c2 = np.where(v2)
    if len(r2) > 0:
        term[r2, c2] -= h[il2[r2, c2]]
    return ((w * dx)[None, :] * term).sum(axis=1)

def fista_run(x0, n, target, dx, beta, max_iter, t_limit):
    x = project_v(x0, target)
    y = x.copy()
    t = 1.0
    step = 1.0 / (beta * 0.5 + 1.0)
    best_c5 = compute_c5(x)
    best_x = x.copy()
    f_y = obj_val(y, beta, dx)
    for k in range(max_iter):
        if time.time() > t_limit:
            break
        g = obj_grad(y, beta, dx)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        z = y - step * g
        backtracked = False
        for _ in range(30):
            x_new = project_v(z, target)
            f_new = obj_val(x_new, beta, dx)
            diff = x_new - y
            if f_new <= f_y + np.dot(g, diff) + 0.5 * step * np.dot(diff, diff):
                backtracked = True
                break
            step *= 0.5
            z = y - step * g
        if not backtracked:
            step = max(step, 1e-10)
            x_new = project_v(y - step * g, target)
        c5 = compute_c5(x_new)
        if c5 < best_c5:
            best_c5 = c5
            best_x = x_new.copy()
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y = np.clip(y, -0.1, 1.1)
        x = x_new.copy()
        t = t_new
        f_y = obj_val(y, beta, dx)
    return best_x, best_c5

def slsqp_polish(h0, n, target, dx, beta, maxiter, t_limit):
    bounds = Bounds(np.zeros(n), np.ones(n))
    def con_fun(x):
        return np.sum(x) - target
    def con_jac(x):
        return np.ones(n)
    def obj(h):
        return obj_val(h, beta, dx)
    def obj_g(h):
        return obj_grad(h, beta, dx)
    try:
        res = minimize(obj, h0, jac=obj_g, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_v(res.x, target)
        c5 = compute_c5(h_new)
        return h_new, c5
    except Exception:
        return h0, compute_c5(h0)
