def smooth_max_grad(h, beta):
    """Gradient of smooth_max(corr(h), beta) w.r.t. h."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 500)
    p = np.exp(shifted) / np.sum(np.exp(shifted))
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-12:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            # dI_k/dh[m] = (1-h[m-lag]) for m in [lag, n-1]
            #             - h[m+lag] for m in [0, n-1-lag]
            grad[lag:] += pk * (1.0 - h[:-lag] if lag > 0 else 1.0 - h)
            grad[:n-lag] -= pk * h[lag:]
        else:
            lag2 = -lag
            grad[:n-lag2] += pk * (1.0 - h[lag2:])
            grad[lag2:] -= pk * h[:n-lag2]
    return grad

    h = h0.copy()
    h_prev = h0.copy()
    best_h = h0.copy()
    best_val = compute_c5(h0)
    
    for t in range(n_iter):
        # Nesterov step
        h_nest = h + momentum * (h - h_prev)
        h_nest = project_h(h_nest, target_sum)
        
        # Gradient

        if val < best_val:
            best_val = val
            best_h = h_new.copy()

        h = h_new

        beta = 500 if attempt % 2 == 0 else 800
        tl = min(15, tleft() - 10)
        if tl > 5:
            h_opt, v = nesterov_loop(h0, target, beta, tl, alpha_init=2.0)
            if v < best_c5:
                best_c5 = v
                best_h = h_opt.copy()
            best_h, best_c5, best_n = try_thresholds(h_opt, best_n / 2.0, best_h, best_c5, best_n)

    # Phase 4: Final SLSQP polish
    if tleft() > 15:
        n = best_n
        target = n / 2.0
        dx = 2.0 / n
        h_start = project_h(best_h, target)

        def obj_func(h):
            corr = np.correlate(h, 1.0 - h, mode="full")
            m = np.max(corr)
            sh = np.clip(3000 * (corr - m), -500, 500)
            return (m + np.log(np.sum(np.exp(sh))) / 3000) * dx

        def obj_grad(h):
            return smooth_max_grad(h, 3000) * dx

        bounds = Bounds(np.zeros(n), np.ones(n))
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
                 'jac': lambda x: np.ones(n)}]
        try:
            res = minimize(obj_func, h_start, jac=obj_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 5000, 'ftol': 1e-18})
            h_new = project_h(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h, best_c5 = h_new.copy(), c5_new
        except Exception:
            pass

    # Phase 5: One more Nesterov pass on best
    if tleft() > 10:
        n = best_n
        target = n / 2.0
        tl = min(15, tleft() - 3)
        if tl > 3:
            h_opt, v = nesterov_loop(best_h, target, 2000, tl, alpha_init=1.0)
            if v < best_c5:
                best_h, best_c5 = h_opt.copy(), v
            best_h, best_c5, best_n = try_thresholds(best_h, best_n / 2.0, best_h, best_c5, best_n)

    final_n = best_n
    best_h = project_h(best_h, final_n / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, final_n
