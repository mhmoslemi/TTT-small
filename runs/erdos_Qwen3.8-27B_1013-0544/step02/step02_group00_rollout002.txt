    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def _obj_fn(h_in):
    c = np.correlate(h_in, 1.0 - h_in, mode="full")
    return smooth_max(c, _opt_params['beta']) * _opt_params['dx']

def _con_fn(x_in):
    return np.sum(x_in) - _opt_params['target_sum']

def _con_jac(x_in):
    return np.ones(x_in.shape)

def _run_subgrad(cur, target_sum, n, max_iter, time_fn, min_time, best_ref, K=3):
    cur_c5, _ = compute_c5(cur)
    for it in range(max_iter):
        if time_fn() < min_time:
            break
        _, corr_vals = compute_c5(cur)
        sorted_idx = np.argsort(corr_vals)[::-1]
        active = sorted_idx[:K]
        # Only include lags within 5% of max
        thresh = corr_vals[sorted_idx[0]] * 0.95
        active = sorted_idx[corr_vals[sorted_idx] >= thresh][:K]
        if len(active) == 0:
            active = sorted_idx[:1]
        grad = vectorized_subgrad(cur, active, n)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-13:
            break
        grad /= gnorm
        step = 0.08 / (1.0 + 0.001 * it)
        improved = False
        for _ in range(12):
            h_trial = project_h(cur - step * grad, target_sum)
            c5_trial, _ = compute_c5(h_trial)
            if c5_trial < cur_c5 - 1e-15:
                cur = h_trial.copy()
                cur_c5 = c5_trial
                if cur_c5 < best_ref['c5']:
                    best_ref['c5'] = cur_c5
                    best_ref['h'] = cur.copy()
                    best_ref['n'] = n
                improved = True
                break
            step *= 0.5
        if not improved and it % 200 == 0 and it > 100:
            perturb = np.random.normal(0, 0.005, n)
            h_trial = project_h(cur + perturb, target_sum)
            c5_trial, _ = compute_c5(h_trial)
            if c5_trial < cur_c5:
                cur = h_trial.copy()
                cur_c5 = c5_trial
                if cur_c5 < best_ref['c5']:
                    best_ref['c5'] = cur_c5
                    best_ref['h'] = cur.copy()
                    best_ref['n'] = n
    return cur, cur_c5

def _run_slsqp(cur, target_sum, n, dx, time_fn, min_time, best_ref):
    bounds = Bounds(np.zeros(n), np.ones(n))
    constraints = [{'type': 'eq', 'fun': _con_fn, 'jac': _con_jac}]
    cur_c5, _ = compute_c5(cur)
    for beta in [50, 100, 200, 400, 800, 1600, 3000]:
        if time_fn() < min_time:
            break
        _opt_params['beta'] = beta
        _opt_params['dx'] = dx
        _opt_params['target_sum'] = target_sum
        try:
            res = minimize(_obj_fn, cur, method='SLSQP', bounds=bounds,
                           constraints=constraints,
                           options={'maxiter': 4000, 'ftol': 1e-14})
            h_new = project_h(res.x, target_sum)
            c5_new, _ = compute_c5(h_new)
            if c5_new < cur_c5 - 1e-14:
                cur = h_new.copy()
                cur_c5 = c5_new
                if cur_c5 < best_ref['c5']:
                    best_ref['c5'] = cur_c5
                    best_ref['h'] = cur.copy()
                    best_ref['n'] = n
        except Exception:
            pass
    return cur, cur_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    def time_left():
        return budget_s - (time.time() - start_time)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_base = np.array(initial_h_values, dtype=float).copy()
        n_base = len(h_base)
    else:
        n_base = 140
        h_base = np.full(n_base, 0.5)

    best_ref = {'c5': compute_c5(h_base)[0], 'h': h_base.copy(), 'n': n_base}

    n_trials = sorted(set([n_base, max(40, n_base - 20), n_base + 20, n_base + 40, max(30, n_base - 40)]))

    for n_points in n_trials:
        if time_left() < 80:
            break

        target_sum = n_points / 2.0
        dx = 2.0 / n_points

        if n_points == n_base:
            cur = project_h(h_base.copy(), target_sum)
        else:
            x_old = np.linspace(0, 2, best_ref['n'])
            x_new = np.linspace(0, 2, n_points)
            cur = project_h(np.interp(x_new, x_old, best_ref['h']), target_sum)

        cur_c5, _ = compute_c5(cur)
        if cur_c5 < best_ref['c5']:
            best_ref['c5'] = cur_c5
            best_ref['h'] = cur.copy()
            best_ref['n'] = n_points

        # Phase 1: Subgradient with backtracking
        cur, cur_c5 = _run_subgrad(cur, target_sum, n_points, 4000, time_left, 40, best_ref, K=4)

        # Phase 2: SLSQP polish
        if time_left() > 40:
            cur, cur_c5 = _run_slsqp(cur, target_sum, n_points, dx, time_left, 20, best_ref)

        # Phase 3: Fine subgradient
        if time_left() > 15:
            cur, cur_c5 = _run_subgrad(cur, target_sum, n_points, 3000, time_left, 10, best_ref, K=2)

    # Final high-precision SLSQP on best
    if time_left() > 20:
        n_f = best_ref['n']
        dx_f = 2.0 / n_f
        cur_f = best_ref['h'].copy()
        cur_f, _ = _run_slsqp(cur_f, n_f / 2.0, n_f, dx_f, time_left, 5, best_ref)

        # Final subgradient polish
        if time_left() > 10:
            cur_f, _ = _run_subgrad(cur_f, n_f / 2.0, n_f, 2000, time_left, 3, best_ref, K=2)

    return best_ref['h'], best_ref['c5'], best_ref['n']
