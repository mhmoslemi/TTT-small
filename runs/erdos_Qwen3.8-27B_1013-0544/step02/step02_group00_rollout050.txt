def project_feasible(h, target_sum):

def compute_gradient(h, active_lags):

    n = len(h)

            if 0 <= j + s < n:
                grad[j] += (1.0 - h[j + s])
            if 0 <= j - s < n:
                grad[j] -= h[j - s]
    return grad / len(active_lags)
