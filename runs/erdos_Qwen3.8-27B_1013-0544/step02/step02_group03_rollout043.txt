Actually, a cleaner approach: since the constraint is sum h[i] = target_sum, and we're only changing h[j], the feasible range for h[j] is:
[max(0, target_sum - sum_{i≠j} max(h[i], 0))] <= h[j] <= min(1, target_sum - sum_{i≠j} min(h[i], 1))

But this is complex. Let me just do:
1. For each j, compute the optimal h[j] in [0,1] ignoring the sum constraint
2. Project back to satisfy the sum constraint

This should work in practice.

```python
def coordinate_descent(h, target_sum, max_iter=1000):
    n = len(h)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    
    for it in range(max_iter):
        improved = False
        for j in range(n):
            # Compute a and b for each lag m
            # h[j] appears in result[m] when:
            # - i=j: valid when 0<=j+lag<n, i.e., -j <= m-n+1 < n-j
            # - i=j+lag (i.e., i+m-n+1=j): valid when 0<=j-lag<n
            
            # For each m in [0, 2n-2], lag = m - (n-1)
            # Term 1 (i=j): valid when 0 <= j+lag < n, i.e., -j <= lag < n-j
            #   a1 = 1 - h[j+lag] = 1 - h[j+m-n+1]
            # Term 2 (i=j-lag): valid when 0 <= j-lag < n, i.e., j-n < lag <= j
            #   a2 = -h[j-lag] = -h[j+m-n+1-... ]
            #   Wait, i = j-lag, so i+m-n+1 = j-lag+m-n+1. We need this = j.
            #   j-lag+m-n+1 = j => m = lag+n-1. That's circular.
            
            # Let me redo this. result[m] = sum_i h[i]*(1-h[i+m-n+1])
            # h[j] appears with coefficient:
            # - (1-h[j+m-n+1]) if i=j (and 0<=j+m-n+1<n, i.e., n-1-j <= m < n-1+j+n... wait)
            #   0 <= j+m-n+1 < n => n-1-j <= m <= 2n-2-j
            # - h[j+m-n+1-j]... no wait.
            #   If i+m-n+1 = j, then i = j-m+n-1. This i must satisfy 0<=i<n.
            #   0 <= j-m+n-1 < n => j-n+1 <= m <= j-1+n... 
            #   Actually: 0 <= j-m+n-1 means m <= j+n-1. And j-m+n-1 < n means m > j-1.
            #   So j < m <= j+n-1. But m ranges from 0 to 2n-2, and m > j-1 means m >= j (integer).
            #   And m <= j+n-1. Since m <= 2n-2, this is m in [j, min(j+n-1, 2n-2)].
            
            # So for h[j]:
            # Coefficient in result[m]:
            # c_m = (1 - h[j+m-n+1]) * [0 <= j+m-n+1 < n] + (-1) * h[j-m+n-1] * [0 <= j-m+n-1 < n]
            # Wait, the second term: if i = j-m+n-1, then result[m] includes h[j-m+n-1] * (1-h[j])
            # So the coefficient of h[j] in result[m] from this term is -h[j-m+n-1]
            
            # Let me just compute this numerically for each j.
            
            # For efficiency, let's compute the gradient-like quantities
            a = np.zeros(2*n-1)  # coefficient of h[j] in each result[m]
            b = np.zeros(2*n-1)  # constant term in each result[m]
            
            for m in range(2*n-1):
                lag = m - (n - 1)
                # Term from i=j: valid if 0 <= j+lag < n
                idx1 = j + lag
                # Term from i=j-lag (i.e., i+m-n+1 = j means i = j-m+n-1 = j-lag): 
                # valid if 0 <= j-lag < n
                idx2 = j - lag
                
                c = 0.0  # coefficient of h[j]
                bd = 0.0  # constant
                
                if 0 <= idx1 < n:
                    c += (1.0 - h[idx1])
                if 0 <= idx2 < n:
                    c += (-1.0) * h[idx2]
                    bd += h[idx2]
                # Wait, this isn't right either. Let me be more careful.
                
                # result[m] = sum_i h[i] * (1-h[i+m-n+1])
                # = sum_i h[i] - sum_i h[i]*h[i+m-n+1]
                
                # For the second sum, h[j] appears:
                # (a) as h[i] when i=j: contributes -h[j]*h[j+m-n+1]
                # (b) as h[i+m-n+1] when i+m-n+1=j: contributes -h[i]*h[j] = -h[j-m+n-1]*h[j]
                
                # For the first sum, h[j] appears when i=j: contributes +h[j]
                
                # So coefficient of h[j] in result[m]:
                # +1 (from first sum) - h[j+m-n+1] (from second sum, part a) - h[j-m+n-1] (from second sum, part b)
                
                # But we need to be careful about bounds.
                
                c = 1.0  # from first sum
                if 0 <= j + lag < n:  # j+m-n+1 = j+lag
                    c -= h[j + lag]
                if 0 <= j - lag < n:  # j-m+n-1 = j-lag  
                    c -= h[j - lag]
                
                a[m] = c
                # b[m] = result[m] - c * h[j]
                b[m] = 0.0  # We'll compute result[m] - a[m]*h[j] = (result[m] - c*h[j])
            
            # Actually, it's easier to just compute result[m] and then b[m] = result[m] - a[m] * h[j]
            corr_current = np.correlate(h, 1.0 - h, mode="full")
            for m in range(2*n-1):
                b[m] = corr_current[m] - a[m] * h[j]
            
            # Now minimize max_m (a[m]*t + b[m]) subject to 0 <= t <= 1
            # This is a 1D minimax problem.
            # The optimal t is where the max is minimized.
            # If all a[m] > 0, we want t as small as possible (t=0)
            # If all a[m] < 0, we want t as large as possible (t=1)
            # Otherwise, the optimum is where the "active" constraints cross.
            
            # The function f(t) = max_m (a[m]*t + b[m]) is convex in t.
            # We can find the minimum by:
            # - Check endpoints t=0 and t=1
            # - Find where the maximum switches
            
            # Simple approach: ternary search or bisection on the derivative
            # subgradient of f at t is a[m*] where m* = argmax_m (a[m]*t + b[m])
            
            # Even simpler: since f is convex piecewise linear, the minimum is at a breakpoint.
            # Breakpoints occur where a[m1]*t + b[m1] = a[m2]*t + b[m2], i.e.,
            # t = (b[m2] - b[m1]) / (a[m1] - a[m2])
            
            # But there are O(n^2) breakpoints. For n=70, that's ~5000, which is fine.
            # But we can be smarter: just do a few iterations of the subgradient method in 1D.
            
            # Let's just do a simple approach: try t=0, t=1, and the point where
            # the two worst lags cross.
            
            # Find current worst lags
            vals = a * h[j] + b  # = corr_current (sanity check)
            m_worst = np.argmax(vals)
            
            # New objective as function of t: a[t_idx] * t + b[t_idx] for each t_idx
            # We want to minimize max_t_idx (a[t_idx]*t + b[t_idx])
            
            # Try a few candidate values of t:
            candidates = [0.0, 1.0, h[j]]
            
            # Add crossing points with the current worst
            m_other = np.argsort(vals)[::-1][1]  # second worst
            if a[m_worst] != a[m_other]:
                t_cross = (b[m_other] - b[m_worst]) / (a[m_worst] - a[m_other])
                if 0 <= t_cross <= 1:
                    candidates.append(t_cross)
            
            # Also try: for the current worst m_worst, if a[m_worst] > 0, decreasing t helps
            # if a[m_worst] < 0, increasing t helps
            if a[m_worst] > 0:
                candidates.append(max(0, h[j] - 0.1))
            else:
                candidates.append(min(1, h[j] + 0.1))
            
            # Evaluate all candidates
            best_t = h[j]
            best_val = np.max(a * h[j] + b)
            for t in candidates:
                t = np.clip(t, 0, 1)
                val = np.max(a * t + b)
                if val < best_val - 1e-12:
                    best_val = val
                    best_t = t
            
            # Apply
            h[j] = best_t
            h = project_feasible(h, target_sum)
        
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-10:
            best_c5 = c5
        elif c5 > best_c5 + 1e-10:
            # This shouldn't happen with exact 1D minimization
            pass
    
    return h, best_c5
```

Wait, there's an issue. After I project back, the values of other h[i] change slightly, so my a and b vectors are no longer valid. But since the projection is a small perturbation (it redistributes a small amount), this should be OK in practice.

Actually, the issue is more fundamental: after changing h[j] and projecting, ALL coordinates change. So the next coordinate's a and b are computed from the updated h. This is fine - it's just that the 1D optimization for each coordinate isn't exactly optimal for the final projected solution. But it should still make progress on average.

Let me also think about the computational cost. For each coordinate j (n=70), we need to:
1. Compute a and b vectors (length 2n-1 = 139): O(n)
2. Find the optimal t: O(n) or O(n log n)
3. Project: O(n)

Total per sweep: O(n^2) = O(4900)
This is very fast! We can do many sweeps.

But wait, computing `corr_current = np.correlate(h, 1.0 - h, mode="full")` is O(n^2) for each j. That's O(n^3) per sweep. For n=70, that's ~343,000 operations per sweep. We can do thousands of sweeps easily.

Actually, we can avoid recomputing the full correlation each time. Since we're only changing h[j], the change in correlation is:
delta_result[m] = (new_h[j] - old_h[j]) * a[m]

So we can update the correlation incrementally!

```python
def coordinate_descent_v2(h, target_sum, max_sweeps=100):

                elif a[m_star] < 0:
                    t_new = min(1.0, t + 0.01)
                else:
                    break
                if abs(t_new - t) < 1e-10:
                    break
                t = t_new

                h = project_feasible(h, target_sum)

                # For now, trust the incremental update
```

Hmm, but the 1D subgradient method with fixed step size might not converge well. Let me think of a better way.

For the 1D problem: minimize f(t) = max_m (a[m]*t + b[m]) over t in [0,1].

This is a convex piecewise-linear function. The minimum occurs at:
- t = 0 (if f is decreasing at 0, i.e., all active a[m] < 0)
- t = 1 (if f is increasing at 1, i.e., all active a[m] > 0)
- A breakpoint where two (or more) lines cross at the minimum

The breakpoints are at t_{mn} = (b[n] - b[m]) / (a[m] - a[n]) for a[m] ≠ a[n].

For n=70, there are 2n-1 = 139 lines, so C(139,2) ≈ 9600 breakpoints. We can find all breakpoints in [0,1], sort them, and evaluate f at each. This is O(n^2) per coordinate, which is fine.

Actually, even simpler: we can use the fact that the minimum of a convex piecewise-linear function is where 0 is in the subdifferential, i.e., where there exist m1, m2 (possibly same) with a[m1] <= 0 <= a[m2] and f(t) = a[m1]*t + b[m1] = a[m2]*t + b[m2].

In practice, let me just do a bisection on the subgradient:
- At t, the subgradient is [min active a[m], max active a[m]] where active = {m : a[m]*t + b[m] = f(t)}
- If min active a[m] > 0: move left
- If max active a[m] < 0: move right
- Otherwise: we're at the optimum

This converges in O(log(1/eps)) iterations.

Let me implement this properly:

```python
def solve_1d_minimax(a, b, t_lo=0.0, t_hi=1.0):
    """Solve min_t max_m (a[m]*t + b[m]) for t in [t_lo, t_hi]"""
    for _ in range(60):  # enough for double precision
        t_mid = (t_lo + t_hi) / 2
        vals = a * t_mid + b
        m_star = np.argmax(vals)
        f_mid = vals[m_star]
        
        # Subgradient at t_mid: all a[m] where a[m]*t_mid + b[m] == f_mid (within tolerance)
        active = np.isclose(a * t_mid + b, f_mid, atol=1e-10)
        if not np.any(active):
            active = (m_star == np.arange(len(a)))
        
        min_grad = np.min(a[active])
        max_grad = np.max(a[active])
        
        if min_grad > 1e-10:
            # f is increasing at t_mid, move left
            t_hi = t_mid
        elif max_grad < -1e-10:
            # f is decreasing at t_mid, move right
            t_lo = t_mid
        else:
            # 0 is in subdifferential, we're at optimum
            break
    
    return (t_lo + t_hi) / 2
```

Actually, let me reconsider the overall strategy. The coordinate descent approach is promising because:
1. It's exact in 1D (finds the true optimum for each coordinate)
2. It's fast (O(n^2) per sweep)
3. It should converge to a local optimum

The main risk is getting stuck in a bad local optimum. To mitigate this, I'll combine it with:
1. SLSQP for smoothing the solution
2. Random restarts
3. Multi-resolution

Let me write the final code:

```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-12:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def con_sum(x, target_sum):
    return x.sum() - target_sum

def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, beta) * dx

def solve_1d_minimax(a, b):
    t_lo = 0.0
    t_hi = 1.0
    for _ in range(60):
        t_mid = (t_lo + t_hi) / 2.0
        vals = a * t_mid + b
        m_star = np.argmax(vals)
        f_mid = vals[m_star]
        active = np.isclose(a * t_mid + b, f_mid, atol=1e-10)
        if not np.any(active):
            active = np.zeros(len(a), dtype=bool)
            active[m_star] = True
        min_grad = np.min(a[active])
        max_grad = np.max(a[active])
        if min_grad > 1e-10:
            t_hi = t_mid
        elif max_grad < -1e-10:
            t_lo = t_mid
        else:
            break
    return (t_lo + t_hi) / 2.0

def coordinate_descent(h, target_sum, max_sweeps=200, dx=None):
    n = len(h)
    if dx is None:
        dx = 2.0 / n
    h = h.copy()
    h = project_feasible(h, target_sum)
    corr = np.correlate(h, 1.0 - h, mode="full")
    best_c5 = float(np.max(corr) * dx)
    n_full = 2 * n - 1
    m_arr = np.arange(n_full)
    lag = m_arr - (n - 1)
    
    for sweep in range(max_sweeps):
        stalled = True
        for j in range(n):
            # Compute a[m] = coefficient of h[j] in corr[m]
            a = np.ones(n_full)
            idx1 = j + lag
            valid1 = (idx1 >= 0) & (idx1 < n)
            a[valid1] -= h[idx1[valid1]]
            idx2 = j - lag
            valid2 = (idx2 >= 0) & (idx2 < n)
            a[valid2] -= h[idx2[valid2]]
            
            # b[m] = corr[m] - a[m] * h[j]
            b = corr - a * h[j]
            
            # Solve 1D minimax
            t_new = solve_1d_minimax(a, b)
            
            if abs(t_new - h[j]) > 1e-8:
                delta = t_new - h[j]
                h[j] = t_new
                corr += delta * a
                h = project_feasible(h, target_sum)
                stalled = False
        
        # Periodic recompute
        if sweep % 20 == 19:
            corr = np.correlate(h, 1.0 - h, mode="full")
        
        c5 = float(np.max(corr) * dx)
        if c5 < best_c5:
            best_c5 = c5
        if stalled and sweep > 10:
            break
    
    return h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    def remaining():
        return budget_s - (time.time() - t0)
    
    # Initialize
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)
    
    n = len(h_init)
    dx = 2.0 / n
    target_sum = n / 2.0
    
    best_h = project_feasible(h_init, target_sum)
    best_c5 = compute_c5(best_h)
    best_n = n
    
    # Phase 1: Coordinate descent on initial resolution
    if remaining() > 100:
        max_sweeps = min(200, int(remaining() / 2))
        h_cd, c5_cd = coordinate_descent(best_h, target_sum, max_sweeps=max_sweeps, dx=dx)
        if c5_cd < best_c5:
            best_h = h_cd.copy()
            best_c5 = c5_cd
    
    # Phase 2: Multi-resolution scan
    if remaining() > 200:
        n_trials = [55, 60, 64, 68, 72, 76, 80, 84, 88, 90, 95, 100, 110, 120]
        n_trials = [x for x in n_trials if x != n and x >= 40]
        for n_t in n_trials:
            if remaining() < 20:
                break
            h_t = resample_h(best_h, n_t)
            h_t = project_feasible(h_t, n_t / 2.0)
            dx_t = 2.0 / n_t
            # Quick coordinate descent
            max_sw = max(30, int(remaining() / 20))
            h_cd, c5_cd = coordinate_descent(h_t, n_t / 2.0, max_sweeps=max_sw, dx=dx_t)
            if c5_cd < best_c5:
                best_h = h_cd.copy()
                best_c5 = c5_cd
                best_n = n_t
                n = n_t
                dx = dx_t
                target_sum = n / 2.0
    
    # Phase 3: SLSQP refinement
    if remaining() > 60:
        n = best_n
        dx = 2.0 / n
        target_sum = n / 2.0
        bounds = Bounds(np.zeros(n), np.ones(n))
        cons = [{'type': 'eq', 'fun': con_sum, 'args': (target_sum,)}]
        h = best_h.copy()
        
        for beta in [200, 1000, 5000]:
            if remaining() < 20:
                break
            try:
                res = minimize(obj_smooth, h, args=(dx, beta), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 2000, 'ftol': 1e-14})
                h_new = project_feasible(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5
                    h = h_new.copy()
            except:
                pass
    
    # Phase 4: More coordinate descent
    if remaining() > 50:
        max_sweeps = int(remaining() / 3)
        h_cd, c5_cd = coordinate_descent(best_h, target_sum, max_sweeps=max_sweeps, dx=dx)
        if c5_cd < best_c5:
            best_h = h_cd.copy()
            best_c5 = c5_cd
    
    # Phase 5: Random restarts + coordinate descent
    if remaining() > 80:
        for restart in range(10):
            if remaining() < 20:
                break
            h_r = best_h.copy()
            h_r += np.random.randn(n) * 0.01
            h_r = project_feasible(h_r, target_sum)
            max_sw = max(50, int(remaining() / 3))
            h_cd, c5_cd = coordinate_descent(h_r, target_sum, max_sweeps=max_sw, dx=dx)
            if c5_cd < best_c5:
                best_h = h_cd.copy()
                best_c5 = c5_cd
```

```python
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        if abs(s - target_sum) < 1e-13:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)

def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

    return x.sum() - target_sum

def solve_1d_minimax(a, b):
    t_lo, t_hi = 0.0, 1.0
    for _ in range(64):
```

```python
            ht = resample_h(best_h, nt)
            ht = project_feasible(ht, nt / 2.0)
            dxt = 2.0 / nt
            hs, cs = coord_desc(ht, nt / 2.0, 200, dxt)
            if cs < best_c5:
                best_h, best_c5, best_n = hs, cs, nt
                n, dx, ts = nt, dxt, nt / 2.0

    # Phase 7: Final SLSQP polish
    if budget_s - (time.time() - t0) > 20:
        bnds = Bounds(np.zeros(n), np.ones(n))
        cons = [{'type': 'eq', 'fun': con_sum, 'args': (ts,)}]
        for beta in [10000, 40000]:
            if budget_s - (time.time() - t0) < 10:
                break
            try:
                r = minimize(obj_smooth, best_h, args=(dx, beta), method='SLSQP',
                             bounds=bnds, constraints=cons,
                             options={'maxiter': 5000, 'ftol': 1e-15})
                hn = project_feasible(r.x, ts)
                c5 = compute_c5(hn)
                if c5 < best_c5:
                    best_h, best_c5 = hn, c5
            except:
                pass

    # Phase 8: Final coordinate descent
    if budget_s - (time.time() - t0) > 15:
        h3, c3 = coord_desc(best_h, ts, 500, dx)
        if c3 < best_c5:
            best_h, best_c5 = h3, c3

    return best_h, best_c5, len(best_h)
```
