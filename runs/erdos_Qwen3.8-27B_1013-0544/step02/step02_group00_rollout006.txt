import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def periodic_c5(t, p):
    """Compute C5 for a periodic pattern t of period p."""
    # corr at lag k: sum_j t[j] * (1 - t[(j+k)%p])
    vals = []
    for k in range(p):
        s = 0.0
        for j in range(p):
            s += t[j] * (1.0 - t[(j+k) % p])
        vals.append(s)
    return (2.0 / p) * max(vals)

def periodic_obj(t, p):
    vals = np.array([sum(t[j] * (1.0 - t[(j+k) % p]) for j in range(p)) for k in range(p)])
    return (2.0 / p) * np.max(vals)

def optimize_periodic(p, n_target, start_time, budget_s):

    target_t_sum = p / 2.0  # sum of t should be p/2
    bounds = Bounds(np.zeros(p), np.ones(p))
    
    def obj(t):
        vals = np.array([sum(t[j] * (1.0 - t[(j+k) % p]) for j in range(p)) for k in range(p)])
        return (2.0 / p) * np.max(vals)
    
    def con(t):
        return np.sum(t) - target_t_sum
    
    constraints = [{'type': 'eq', 'fun': con}]
    
    best_t = None
    best_val = 1.0
    
    # Multiple starts
    starts = []
    # Uniform

    # Alternating

    starts.append(alt)
    # Sine
    for freq in [1, 2, 3]:
        s = 0.5 + 0.3 * np.sin(2 * np.pi * freq * np.arange(p) / p)
        starts.append(np.clip(s, 0.01, 0.99))
    # Random
    np.random.seed(p * 100)
    for _ in range(5):
        r = np.random.dirichlet(np.ones(p)) * p / 2
        starts.append(np.clip(r, 0.01, 0.99))
    
    for t0 in starts:
        if time.time() - start_time > budget_s - 10:
            break
        t0 = np.clip(t0, 0.0, 1.0)
        # Project to sum = p/2
        t0 = t0 / t0.sum() * target_t_sum
        t0 = np.clip(t0, 0.0, 1.0)
        # Iterative projection
        for _ in range(50):
            t0 = np.clip(t0, 0.0, 1.0)
            diff = target_t_sum - t0.sum()
            t0 += diff / p
        t0 = np.clip(t0, 0.0, 1.0)
        
        val0 = obj(t0)
        if val0 < best_val:
            best_val = val0
            best_t = t0.copy()
        
        try:
            res = minimize(obj, t0, method='SLSQP', bounds=bounds,
                          constraints=constraints,
                          options={'maxiter': 3000, 'ftol': 1e-15})
            if res.success or res.fun < best_val:
                t_res = np.clip(res.x, 0.0, 1.0)
                # Re-project

                    t_res = np.clip(t_res, 0.0, 1.0)
                    diff = target_t_sum - t_res.sum()
                    t_res += diff / p

                if val < best_val:
                    best_val = val
                    best_t = t_res.copy()
        except Exception:
            pass
    
    if best_t is not None:
        # Tile to n_target
        n_tiles = n_target // p
        h = np.tile(best_t, n_tiles)
        if len(h) < n_target:
            h = np.concatenate([h, best_t[:n_target % p]])
        return h[:n_target], best_val
    return None, 1.0

    # Add some non-divisor periods for testing (we'll handle the remainder)

        if p > 40:  # Don't try too large periods (less improvement expected)

        if h_periodic is not None:
            c5 = compute_c5(h_periodic)
            if c5 < best_c5:
                best_h = h_periodic.copy()
                best_c5 = c5
    
    # Phase 2: Non-periodic refinement using local search
    n = len(best_h)
    target_sum = n / 2.0
    cur = best_h.copy()
    cur_c5 = compute_c5(cur)
    
    # Subgradient + line search
    for iteration in range(5000):
        if time_left() < 10:
            break
        corr = np.correlate(cur, 1.0 - cur, mode="full")
        max_val = np.max(corr)
        # Find top active lags
        threshold = max_val * 0.95
        active_lags = np.where(corr >= threshold)[0]
        
        grad = np.zeros(n)
        for lag_idx in active_lags:
            lag = int(lag_idx) - (n - 1)
            if lag == 0:
                # corr at lag 0: sum h[i]*(1-h[i])
                grad += (1.0 - 2.0 * cur)
            else:
                for j in range(n):
                    i1 = j - lag
                    i2 = j + lag
                    if 0 <= i1 < n:
                        grad[j] += (1.0 - cur[i1])
                    if 0 <= i2 < n:
                        grad[j] -= cur[i2]

        if gnorm < 1e-12:
            break
        grad /= gnorm

        step = 0.1
        improved = False
        for _ in range(20):
            h_trial = project_h(cur - step * grad, target_sum)
            c5_trial = compute_c5(h_trial)
            if c5_trial < cur_c5 - 1e-15:
                cur = h_trial
                cur_c5 = c5_trial
                if cur_c5 < best_c5:
                    best_h = cur.copy()
                    best_c5 = cur_c5
                improved = True
                break
            step *= 0.5
        
        if not improved:
            # Try smaller fixed steps with random perturbations
            for _ in range(3):

                h_trial = project_h(cur + pert, target_sum)
                c5_trial = compute_c5(h_trial)
                if c5_trial < cur_c5 - 1e-15:
                    cur = h_trial
                    cur_c5 = c5_trial
                    if cur_c5 < best_c5:
                        best_h = cur.copy()
                        best_c5 = cur_c5

    if time_left() > 30:
        n_final = len(best_h)
        target_final = n_final / 2.0
        dx_final = 2.0 / n_final
        bounds_final = Bounds(np.zeros(n_final), np.ones(n_final))
        
        def final_obj(h_in):
            c = np.correlate(h_in, 1.0 - h_in, mode="full")
            # Smooth max with high beta
            m = np.max(c)
            x_shift = np.clip(500.0 * (c - m), -500, 500)
            return (m + np.log(np.sum(np.exp(x_shift)) / len(c)) / 500.0) * dx_final
        
        def final_con(x_in):
            return np.sum(x_in) - target_final
        
        def final_jac(x_in):
            return np.ones(x_in.shape)
        
        constraints_f = [{'type': 'eq', 'fun': final_con, 'jac': final_jac}]
        
        try:
            res = minimize(final_obj, best_h, method='SLSQP', bounds=bounds_final,
                          constraints=constraints_f,
                          options={'maxiter': 5000, 'ftol': 1e-14})
            h_new = project_h(res.x, target_final)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
        except Exception:
            pass
    
    return best_h, best_c5, len(best_h)

One more thought: the subgradient computation can be vectorized. For lag k (where k is the actual shift, can be negative):
- If k > 0: corr[k] = sum_{n=0}^{n-1-k} h[n+k]*(1-h[n])
  grad wrt h[j]: 
  - appears in first factor when n+k=j → n=j-k: gives (1-h[j-k]) if 0≤j-k<n-k, i.e., k≤j≤n-1
  - appears in second factor when n=j: gives -h[j+k] if j+k<n, i.e., j<n-k

  So grad[j] += (1-h[j-k]) for j in range(k, n)
  And grad[j] += -h[j+k] for j in range(0, n-k)

  Vectorized:
  grad[k:] += (1 - h[:-k])
  grad[:n-k] -= h[k:]

- If k < 0, let k' = -k: corr[-k'] = sum_{n=k'}^{n-1} h[n]*(1-h[n-k'])
  By symmetry: grad[j] += (1-h[j+k']) for j in range(0, n-k')
  And grad[j] += -h[j-k'] for j in range(k', n)
  
  Vectorized:
  grad[:n-k'] += (1 - h[k':])
  grad[k':] -= h[:n-k']

- If k = 0: corr[0] = sum h[i]*(1-h[i])
  grad[j] += (1-2*h[j])
  Vectorized: grad += (1-2*h)

This is O(n) per lag, and we only need a few active lags.

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def periodic_objective(t, p):
    vals = np.zeros(p)
    for k in range(p):
        vals[k] = np.sum(t * (1.0 - np.roll(t, k)))
    return (2.0 / p) * np.max(vals)

def optimize_periodic(p, n_target, start_time, budget_s):
    target_t_sum = p / 2.0
    bounds = Bounds(np.zeros(p), np.ones(p))

    def obj(t):
        return periodic_objective(t, p)

    def con(t):
        return np.sum(t) - target_t_sum

    def con_jac(t):
        return np.ones(p)

    constraints = [{'type': 'eq', 'fun': con, 'jac': con_jac}]

    best_t = None
    best_val = 1.0
    np.random.seed(p * 777)

    starts = [np.full(p, 0.5)]
    if p >= 2:
        starts.append(np.where(np.arange(p) % 2 == 0, 0.7, 0.3))
    if p >= 3:
        starts.append(np.where(np.arange(p) % 3 < 2, 0.65, 0.2))
    for freq in [1, 2, 3]:
        s = 0.5 + 0.3 * np.sin(2 * np.pi * freq * np.arange(p) / p)
        starts.append(np.clip(s, 0.05, 0.95))
    for _ in range(8):
        r = np.random.dirichlet(np.ones(p) * 2) * target_t_sum
        starts.append(np.clip(r, 0.01, 0.99))

    for t0 in starts:
        if time.time() - start_time > budget_s - 15:
            break
        t0 = np.clip(t0, 0.0, 1.0)
        for _ in range(100):
            t0 = np.clip(t0, 0.0, 1.0)
            diff = target_t_sum - t0.sum()
            if abs(diff) < 1e-14:
                break
            t0 += diff / p
        t0 = np.clip(t0, 0.0, 1.0)
        val0 = obj(t0)
        if val0 < best_val:
            best_val = val0
            best_t = t0.copy()
        try:
            res = minimize(obj, t0, method='SLSQP', bounds=bounds,
                          constraints=constraints, options={'maxiter': 5000, 'ftol': 1e-16})
            t_res = np.clip(res.x, 0.0, 1.0)
            for _ in range(100):
                t_res = np.clip(t_res, 0.0, 1.0)
                diff = target_t_sum - t_res.sum()
                if abs(diff) < 1e-14:
                    break
                t_res += diff / p
            t_res = np.clip(t_res, 0.0, 1.0)
            val = obj(t_res)
            if val < best_val:
                best_val = val
                best_t = t_res.copy()
        except Exception:
            pass

    if best_t is None:
        return None, 1.0
    h = np.tile(best_t, n_target // p + 1)[:n_target]
    return h, best_val

def subgradient_step(cur, target_sum, n):
    corr = np.correlate(cur, 1.0 - cur, mode="full")
    max_val = np.max(corr)
    threshold = max_val * 0.92
    active = np.where(corr >= threshold)[0]
    grad = np.zeros(n)
    for lag_idx in active:
        lag = int(lag_idx) - (n - 1)
        if lag == 0:
            grad += (1.0 - 2.0 * cur)
        elif lag > 0:
            grad[lag:] += (1.0 - cur[:-lag])
            grad[:n - lag] -= cur[lag:]
        else:
            k = -lag
            grad[:n - k] += (1.0 - cur[k:])
            grad[k:] -= cur[:n - k]
    return grad, max_val

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    def time_left():
        return budget_s - (time.time() - start_time)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()

    best_c5 = compute_c5(best_h)

    # Phase 1: Periodic pattern search
    n_base = len(best_h)
    # Find good (p, n) pairs
    candidates = []
    for p in range(3, 25):
        # Find smallest n that is a multiple of p and >= 40
        n_min = ((40 + p - 1) // p) * p
        for multiplier in range(1, 8):
            n_try = p * multiplier
            if n_try >= 40 and n_try <= 280:
                candidates.append((p, n_try))
                break
    # Also include divisors of n_base
    for p in range(3, 21):
        if n_base % p == 0:
            if (p, n_base) not in candidates:
                candidates.append((p, n_base))
    # Deduplicate, prioritize small p
    seen = set()
    unique_cands = []
    for p, n in sorted(candidates, key=lambda x: (x[0], x[1])):
        if p not in seen:
            seen.add(p)
            unique_cands.append((p, n))

    for p, n_try in unique_cands:
        if time_left() < 20:
            break
        h_per, _ = optimize_periodic(p, n_try, start_time, budget_s)
        if h_per is not None and len(h_per) == n_try:
            c5_try = compute_c5(h_per)
            if c5_try < best_c5:
                best_h = h_per.copy()
                best_c5 = c5_try
                n_base = n_try

    # Phase 2: Subgradient local search
    n = len(best_h)
    target_sum = n / 2.0
    cur = best_h.copy()
    cur_c5 = compute_c5(cur)

    for iteration in range(8000):
        if time_left() < 8:
            break
        grad, max_val = subgradient_step(cur, target_sum, n)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        grad /= gnorm
        step = 0.08 / (1.0 + 0.005 * iteration)

            h_trial = project_h(cur - step * grad, target_sum)

                cur = h_trial
                cur_c5 = c5_trial
                if cur_c5 < best_c5:
                    best_h = cur.copy()
                    best_c5 = cur_c5

                break
            step *= 0.5
        if not improved:
            # Random perturbation

                h_trial = project_h(cur + pert, target_sum)

                    cur_c5 = c5_trial

    # Phase 3: SLSQP polish
    if time_left() > 25:
        n_f = len(best_h)
        target_f = n_f / 2.0
        dx_f = 2.0 / n_f
        bounds_f = Bounds(np.zeros(n_f), np.ones(n_f))

        def final_con(x_in):
            return np.sum(x_in) - target_f

            return np.ones(x_in.shape)

            h_new = project_h(res.x, target_f)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
        except Exception:
            pass

    # Phase 4: Final subgradient refinement
    n = len(best_h)
    target_sum = n / 2.0
    cur = best_h.copy()
    cur_c5 = compute_c5(cur)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        grad /= gnorm
        step = 0.03 / (1.0 + 0.003 * iteration)
        improved = False
        for _ in range(10):

                    best_h = cur.copy()
                    best_c5 = cur_c5
                improved = True
                break
            step *= 0.5
        if not improved and iteration > 100:
            break

    return best_h, best_c5, len(best_h)
