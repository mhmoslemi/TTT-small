import time

    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_c(h, n):
    """Return array of all c_k values, indexed 0 to 2n-2."""
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0

def two_point_refine(h, n, t0, budget_s):

    best_c5 = compute_c5(h, n)

            if budget_s - (time.time() - t0) < 5:
                break

                    break

                                improved = True
                            else:

    return h, best_c5

def block_optimize(h, n, block_start, block_size, alpha, maxiter=500):
    """Optimize a block of h while fixing the rest."""
    end = block_start + block_size
    if end > n:
        return h
    h_block = h[block_start:end].copy()
    total_sum = h_block.sum()
    
    def obj(x):
        h_test = h.copy()
        h_test[block_start:end] = x
        c = np.correlate(h_test, 1 - h_test, mode='full') * (2.0/n)
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(shifted))) / alpha)
    
    bounds = [(0.0, 1.0)] * block_size
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - total_sum, 
             'jac': lambda x: np.ones(block_size)}]
    try:
        res = minimize(obj, h_block, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
        h_new = h.copy()
        h_new[block_start:end] = project_feasible_1d(res.x, total_sum)
        return h_new
    except:
        return h

def project_feasible_1d(h, target_sum):
    """Project 1D array to [0,1] with given sum."""
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
            room = np.where(mask, 1.0 - h, 0.0)
            s = room.sum()
            if s < 1e-14:
                break
            h += (diff/s) * room
        else:
            mask = h > 0.0
            s = h[mask].sum()
            if s < 1e-14:
                break
            h[mask] *= (1.0 + diff/s)
    return np.clip(h, 0.0, 1.0)

Now let me also think about the **line search** for subgradient descent. The standard approach is:
- Polyak step size: alpha_k = (f(h_k) - f*) / ||g_k||²
  But we don't know f*.
- Decreasing step: alpha_k = c / sqrt(k)
- Backtracking: start with alpha, halve until f decreases

For our problem, the function is not convex (it's a max of non-convex functions... wait, is c_k(h) convex or concave in h?

c_k(h) = dx * sum_i h[i](1-h[i+k]) = dx * sum h[i] - dx * sum h[i]h[i+k]

The first term is linear, the second term is -dx * h^T S_k h where S_k is a symmetric matrix. Since S_k has non-negative entries (it's a shift matrix), h^T S_k h ≥ 0, so -h^T S_k h ≤ 0 and is concave.

So c_k(h) = linear + concave = concave.

The maximum of concave functions is NOT necessarily concave or convex. So the subgradient method doesn't have the nice convergence guarantees.

Actually, for step 3, the QP subproblem is:
minimize max_k (c_k(h) + ∇c_k(h)^T d)  [linear approximation]
subject to: 0 ≤ h + d ≤ 1, sum(d) = 0, ||d|| ≤ Δ

This is a linear program (minimize t subject to ∇c_k^T d + c_k ≤ t, ...). But solving an LP at each iteration is expensive.

A simpler alternative: use the **Frank-Wolfe** (conditional gradient) method:
- At each iteration, solve: min_{d : sum(d)=0, ||d||_∞ ≤ 1} max_k ∇c_k(h)^T d
- This is: min_d max_k (∇c_k)^T d = min_d max_k (sum of active gradient components) * d
- Since max of linear is convex, this is a convex problem in d
- The solution is d that minimizes the "worst-case" directional derivative

The Frank-Wolfe update is: h_new = (1-t)*h + t*d* where t is the step size.

For our problem, the linearized subproblem is:
min_d max_k g_k^T d  subject to sum(d) = 0, -1 ≤ d_j ≤ 1
where g_k = ∇c_k(h) for active k.

This can be solved as an LP:
min t
s.t. g_k^T d ≤ t for all active k
     sum(d) = 0
     -1 ≤ d_j ≤ 1

For n=80 and, say, 10 active constraints, this is an LP with 81 variables and ~20 constraints. Very fast!

from scipy.optimize import linprog

def frank_wolfe_step(h, n, top_m=10):
    """One Frank-Wolfe step for minimizing max_k c_k."""
    c_all = compute_all_c(h, n)
    M = c_all.max()
    # Active set: lags where c_k is within 2% of max
    threshold = M - 0.02 * M
    active_idx = np.where(c_all >= threshold)[0]
    if len(active_idx) == 0:
        active_idx = np.array([np.argmax(c_all)])
    if len(active_idx) > top_m:
        # Keep top_m
        vals = c_all[active_idx]
        order = np.argsort(vals)[-top_m:]
        active_idx = active_idx[order]
    
    # Compute gradients for active lags
    dx = 2.0 / n
    G = np.zeros((len(active_idx), n))
    for i, k in enumerate(active_idx):
        ell = k - (n - 1)
        g = np.zeros(n)
        if ell >= 0:
            ji = np.arange(n - ell)
            g[ji] += dx * (1.0 - h[ji + ell])
            jj = np.arange(ell, n)
            g[jj] += dx * (-h[jj - ell])
        else:
            ellp = -ell
            ji = np.arange(ellp, n)
            g[ji] += dx * (1.0 - h[ji - ellp])
            jj = np.arange(n - ellp)
            g[jj] += dx * (-h[jj + ellp])
        G[i] = g
    
    # Solve LP: min t s.t. G @ d <= t, sum(d) = 0, -1 <= d <= 1
    # Variables: [d_0, ..., d_{n-1}, t]
    # c_minimize = [0, ..., 0, 1]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality: G @ d - t <= 0 → [G, -1] @ [d, t] <= 0
    A_ub = np.hstack([G, -np.ones((len(active_idx), 1))])
    b_ub = np.zeros(len(active_idx))
    
    # Equality: sum(d) = 0 → [1, ..., 1, 0] @ [d, t] = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [0.0]
    
    # Bounds: -1 <= d_j <= 1, t unbounded below (but will be >= 0 at optimum)
    bounds = [(-1.0, 1.0)] * n + [(None, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
        if res.status == 0:
            d = res.x[:n]
            return d
    except:
        pass
    
    # Fallback: negative average gradient of active lags, zero-sum
    g_avg = G.mean(axis=0)
    g_avg -= g_avg.mean()
    if np.linalg.norm(g_avg) > 1e-10:
        d = -g_avg / np.linalg.norm(g_avg, ord=np.inf)
        return np.clip(d, -1, 1)
    return np.zeros(n)

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_c(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = np.where(h < 1.0, 1.0 - h, 0.0)
            s = room.sum()
            if s < 1e-14:
                break
            h += (diff / s) * room
        else:
            mask = h > 0.0
            s = h[mask].sum()
            if s < 1e-14:
                break
            h[mask] *= (1.0 + diff / s)
    return np.clip(h, 0.0, 1.0)

def compute_grad_k(h, n, k):
    dx = 2.0 / n
    ell = k - (n - 1)
    g = np.zeros(n)
    if ell >= 0:
        ji = np.arange(n - ell)
        g[ji] += dx * (1.0 - h[ji + ell])
        jj = np.arange(ell, n)
        g[jj] += dx * (-h[jj - ell])
    else:
        ep = -ell
        ji = np.arange(ep, n)
        g[ji] += dx * (1.0 - h[ji - ep])
        jj = np.arange(n - ep)
        g[jj] += dx * (-h[jj + ep])
    return g

def slsqp_smooth(h, n, alpha, maxiter=2000, ftol=1e-15):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        shifts = alpha * (c - c.max())
        mx = shifts.max()
        if mx > 500:
            return float((c.max() + mx + np.log(np.sum(np.exp(shifts - mx)))) / alpha)
        return float(np.log(np.sum(np.exp(alpha * c))) / alpha)
    def obj_grad(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        p = np.exp(alpha * (c - c.max()))
        p /= p.sum()
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if p[k] < 1e-10:
                continue
            g += p[k] * compute_grad_k(x, n, k)
        return g
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2, 'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bounds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def fw_step(h, n, top_m=12):
    c_all = compute_all_c(h, n)
    M = c_all.max()
    thresh = M - 0.005 * M
    active = np.where(c_all >= thresh)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c_all)])
    if len(active) > top_m:
        idx = np.argsort(c_all[active])[-top_m:]
        active = active[idx]
    G = np.array([compute_grad_k(h, n, k) for k in active])
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.hstack([G, -np.ones((len(active), 1))])
    b_ub = np.zeros(len(active))
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [0.0]
    bounds = [(-1.0, 1.0)] * n + [(None, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
        if res.status == 0:
            return res.x[:n]
    except Exception:
        pass
    g_avg = G.mean(axis=0)
    g_avg -= g_avg.mean()
    norm = np.abs(g_avg).max()
    if norm > 1e-12:
        d = -g_avg / norm
        return np.clip(d, -1, 1)
    return np.zeros(n)

def fw_descent(h, n, t0, budget_s, n_iter=50000, top_m=12):
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    cur_c5 = best_c5
    step_size = 1.0
    stalled = 0
    for it in range(n_iter):
        if budget_s - (time.time() - t0) < 15:
            break
        if it % 10000 == 0 and it > 0:
            step_size = max(step_size * 0.6, 0.01)
        d = fw_step(h, n, top_m)
        h_try = project_feasible(h + step_size * d, n)
        c_try = compute_c5(h_try, n)
        if c_try < cur_c5 - 1e-14:
            h = h_try
            cur_c5 = c_try
            stalled = 0
            if c_try < best_c5:
                best_h = h.copy()
                best_c5 = c_try
            step_size = min(step_size * 1.02, 1.0)
        else:
            stalled += 1
            if stalled > 200:
                step_size *= 0.7
                stalled = 0
                if step_size < 0.005:
                    break
    return best_h, best_c5

def block_optimize(h, n, t0, budget_s, block_sizes=(4, 6, 8)):
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    for bs in block_sizes:
        if budget_s - (time.time() - t0) < 5:
            break
        for start in range(n - bs + 1):
            if budget_s - (time.time() - t0) < 3:
                break
            end = start + bs
            h_block = best_h[start:end].copy()
            tot = h_block.sum()
            def blk_obj(x):
                ht = best_h.copy()
                ht[start:end] = x
                return compute_c5(ht, n)
            def blk_obj2(x):
                ht = best_h.copy()
                ht[start:end] = x
                c = np.correlate(ht, 1 - ht, mode='full') * (2.0 / n)
                m = c.max()
                shifts = 50000 * (c - c.max())
                mx = shifts.max()
                if mx > 500:
                    return float(m + (mx + np.log(np.sum(np.exp(shifts - mx)))) / 50000)
                return float(m + np.log(np.sum(np.exp(shifts))) / 50000)
            bl = [(0.0, 1.0)] * bs
            ce = [{'type': 'eq', 'fun': lambda x, t=tot: x.sum() - t,
                   'jac': lambda x: np.ones(bs)}]
            try:
                res = minimize(blk_obj2, h_block, method='SLSQP', bounds=bl,
                             constraints=ce, options={'maxiter': 300, 'ftol': 1e-12})
                h_new = best_h.copy()
                h_new[start:end] = project_feasible(res.x, bs)
                c_new = compute_c5(h_new, n)
                if c_new < best_c5 - 1e-14:
                    best_h = h_new
                    best_c5 = c_new
            except Exception:
                pass
    return best_h, best_c5

def two_point_refine(h, n, t0, budget_s):
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    steps = [0.005, 0.002, 0.001, 0.0005]
    for ds in steps:
        if budget_s - (time.time() - t0) < 3:
            break
        improved = True
        while improved:
            improved = False
            if budget_s - (time.time() - t0) < 2:
                break
            for i in range(n):
                if budget_s - (time.time() - t0) < 1:
                    break
                for j in range(i + 1, n):
                    for s in [1.0, -1.0]:
                        hi = best_h[i] + s * ds
                        hj = best_h[j] - s * ds
                        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                            best_h[i] = hi
                            best_h[j] = hj
                            c = compute_c5(best_h, n)
                            if c < best_c5 - 1e-14:
                                best_c5 = c
                                improved = True
                            else:
                                best_h[i] = hi - s * ds
                                best_h[j] = hj + s * ds
    return best_h, best_c5
