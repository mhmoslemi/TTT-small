                    break
                for topk in [5, 0]:
                    if time.time() - t0 > budget_s - 8:
                        break
                    try:
                        res = minimize(obj_func, h_t, args=(dx_t, beta, topk),
                                       method='SLSQP', bounds=b_t, constraints=c_t,
                                       options={'maxiter': 4000, 'ftol': 1e-15})
                        h_new = project_feasible(res.x, ts_t)
                        c5 = compute_c5(h_new)
                        if c5 < best_c5:
                            best_h = h_new.copy()
                            best_c5 = c5
                            n, dx, ts, bounds, cons = n_try, dx_t, ts_t, b_t, c_t
                    except Exception:
                        pass

    # Phase 4: Diverse restarts
    if time.time() - t0 < budget_s - 30:
        for restart in range(4):
            if time.time() - t0 > budget_s - 10:
                break
            h_r = np.random.beta(3, 3, n)
            h_r = project_feasible(h_r, ts)
            try:
                res = minimize(obj_func, h_r, args=(dx, 200, 5),
                               method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 3000, 'ftol': 1e-14})
                h_new = project_feasible(res.x, ts)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5
            except Exception:
                pass

    # Phase 5: Ultimate polish at highest beta
    if time.time() - t0 < budget_s - 8:
        for beta in [50000, 200000]:
            if time.time() - t0 > budget_s - 4:
                break
            for topk in [3, 2, 0]:
                if time.time() - t0 > budget_s - 2:
                    break
                try:
                    res = minimize(obj_func, best_h, args=(dx, beta, topk),
                                   method='SLSQP', bounds=bounds, constraints=cons,
                                   options={'maxiter': 12000, 'ftol': 1e-16})
                    h_new = project_feasible(res.x, ts)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5
                except Exception:
                    pass

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
