            c5_f = compute_c5(h_f)
            if c5_f < best_c5:
                best_h, best_c5 = h_f.copy(), c5_f
        except Exception:
            pass

        # SLSQP with equality constraint for final refinement
        if time_left() > 5:
            args_s = (5000.0, target_final)
            con_args = (target_final,)
            constraints = [{'type': 'eq', 'fun': slsqp_con_fn, 'jac': slsqp_con_jac_fn,
                           'args': con_args}]
            try:
                res_s = minimize(slsqp_obj_fn, best_h, args=args_s, jac=slsqp_jac_fn,
                                 method='SLSQP', bounds=bounds_final,
                                 constraints=constraints,
                                 options={'maxiter': 5000, 'ftol': 1e-15})
                h_s = project_h(np.clip(res_s.x, 0, 1), target_final)
                c5_s = compute_c5(h_s)
                if c5_s < best_c5:
                    best_h, best_c5 = h_s.copy(), c5_s
            except Exception:
                pass

        # One more projected gradient sweep at very high beta
        if time_left() > 5:
            for it in range(5000):
                if time_left() < 3:
                    break
                val, grad = smooth_max_obj_grad(best_h, 3000.0, target_final, 1000.0)
                lr = 0.02 / (1.0 + 0.002 * it)
                h_trial = project_h(best_h - lr * grad, target_final)
                c5_trial = compute_c5(h_trial)
                if c5_trial < best_c5 - 1e-15:
                    best_h = h_trial.copy()
                    best_c5 = c5_trial
                else:
                    h_trial2 = project_h(best_h - lr * 0.3 * grad, target_final)
                    c5_t2 = compute_c5(h_trial2)
                    if c5_t2 < best_c5 - 1e-15:
                        best_h = h_trial2.copy()
                        best_c5 = c5_t2

    return best_h, best_c5, best_n
