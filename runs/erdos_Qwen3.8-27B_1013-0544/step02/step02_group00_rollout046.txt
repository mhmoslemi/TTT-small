            grad[:n - k] += wi * (1.0 - h[k:])
            grad[k:] -= wi * h[:n - k]
    grad *= dx
    return grad, smooth_max_val(corr, beta)


def nesterov_iter(h, velocity, lr, mu, target, beta):
    h_look = project_h(h + mu * velocity, target)
    grad_look, _ = compute_grad_c5(h_look, beta)
    velocity = mu * velocity + lr * grad_look
    h_new = project_h(h - velocity, target)
    return h_new, velocity


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    def time_left():
        return budget_s - (time.time() - start_time)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = np.full(140, 0.5)

    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)
    base_n = len(h_parent)

    # Multi-resolution Nesterov with annealing beta
    resolutions = sorted(set([base_n, base_n * 2, max(70, base_n // 2)]))
    resolutions = [r for r in resolutions if 40 <= r <= 400]

    for n_pts in resolutions:
        if time_left() < 80:
            break
        target = n_pts / 2.0

        if n_pts == base_n:
            h_cur = h_parent.copy()
        else:
            x_old = np.linspace(0, 2, base_n)
            x_new = np.linspace(0, 2, n_pts)
            h_cur = project_h(np.interp(x_new, x_old, h_parent), target)

        # Also try from best_h if different size
        if len(best_h) != n_pts:
            x_b = np.linspace(0, 2, len(best_h))
            x_n = np.linspace(0, 2, n_pts)
            h_alt = project_h(np.interp(x_n, x_b, best_h), target)
            c5_alt = compute_c5(h_alt)
            if c5_alt < compute_c5(h_cur):
                h_cur = h_alt

        # Nesterov with annealed beta
        velocity = np.zeros(n_pts)
        mu = 0.85
        lr = 0.005
        c5_cur = compute_c5(h_cur)
        if c5_cur < best_c5:
            best_h, best_c5 = h_cur.copy(), c5_cur

        stage_time = time_left() * 0.7  # use 70% of remaining for this resolution
        for it in range(200000):
            if time_left() < 20:
                break
            if time.time() - start_time > stage_time + 60 and it > 100:
                break

            # Anneal beta from low to high
            progress = min(1.0, it / 5000.0)
            beta = 30.0 + progress * 470.0

            h_new, velocity = nesterov_iter(h_cur, velocity, lr, mu, target, beta)
            c5_new = compute_c5(h_new)

            if c5_new < c5_cur - 1e-13:
                h_cur = h_new
                c5_cur = c5_new
                if c5_cur < best_c5:
                    best_h, best_c5 = h_cur.copy(), c5_cur
            else:
                lr *= 0.95
                if lr < 1e-7:
                    lr = 0.005
                    velocity *= 0.5

            # Periodic momentum check
            if it % 2000 == 0 and it > 0:
                velocity *= 0.5  # damp momentum to avoid oscillation

        # Resample best to current resolution for next phase
        if len(best_h) != n_pts and time_left() > 40:
            x_b = np.linspace(0, 2, len(best_h))
            x_n = np.linspace(0, 2, n_pts)
            best_at_res = project_h(np.interp(x_n, x_b, best_h), target)
            c5_br = compute_c5(best_at_res)
            if c5_br < best_c5:
                best_h, best_c5 = best_at_res, c5_br

    # Phase 2: Simulated annealing with structured perturbations
    if time_left() > 50:
        n_sa = len(best_h)
        target_sa = n_sa / 2.0
        h_sa = best_h.copy()
        c5_sa = best_c5
        T = 0.08

        for sa_it in range(50000):
            if time_left() < 8:
                break
            T *= 0.9997
            if T < 1e-5:
                break

            # 60% Fourier perturbations, 40% random pairs
            if sa_it % 5 < 3:
                freq = np.random.randint(1, max(2, n_sa // 3))
                phase = np.random.uniform(0, 2 * np.pi)
                amp = np.random.uniform(0.01, 0.06) * (1.0 + T * 5)
                pert = amp * np.sin(2 * np.pi * freq * np.arange(n_sa) / n_sa + phase)
                h_trial = project_h(h_sa + pert, target_sa)
            else:
                i, j = np.random.choice(n_sa, 2, replace=False)
                delta = np.random.uniform(-0.06, 0.06)
                h_trial = h_sa.copy()
                h_trial[i] += delta
                h_trial[j] -= delta
                h_trial = project_h(np.clip(h_trial, 0, 1), target_sa)

            c5_trial = compute_c5(h_trial)
            accept = c5_trial < c5_sa or (
                c5_trial - c5_sa < 0.05 and
                np.random.random() < np.exp(-(c5_trial - c5_sa) / max(T, 1e-8))
            )
            if accept:
                h_sa = h_trial
                c5_sa = c5_trial
                if c5_sa < best_c5:
                    best_h, best_c5 = h_sa.copy(), c5_sa

    # Phase 3: Final SLSQP polish with high beta
    if time_left() > 8:
        n_f = len(best_h)
        target_f = n_f / 2.0
        dx_f = 2.0 / n_f

        def obj_f(h_in):
            c = np.correlate(h_in, 1.0 - h_in, mode="full") * dx_f
            return smooth_max_val(c, 1500.0)

        def con_f(x_in):
            return np.sum(x_in) - target_f

        def jac_f(x_in):
            return np.ones(x_in.shape)

        try:
            res = minimize(obj_f, best_h, method='SLSQP',
                           bounds=Bounds(np.zeros(n_f), np.ones(n_f)),
                           constraints=[{'type': 'eq', 'fun': con_f, 'jac': jac_f}],
                           options={'maxiter': 5000, 'ftol': 1e-15})
            h_new = project_h(res.x, target_f)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new
        except Exception:
            pass

        # Final subgradient polish
        if time_left() > 5:
            h_pol = best_h.copy()
            c5_pol = best_c5
            for it in range(20000):
                if time_left() < 2:
                    break
                grad, _ = compute_grad_c5(h_pol, 800.0)
                gnorm = np.linalg.norm(grad)
                if gnorm < 1e-12:
                    break
                step = 0.008 / (1.0 + 0.002 * it)
                h_trial = project_h(h_pol - step * grad / gnorm, target_f)
                c5_trial = compute_c5(h_trial)
                if c5_trial < c5_pol - 1e-14:
                    h_pol = h_trial
                    c5_pol = c5_trial
                    if c5_pol < best_c5:
                        best_h, best_c5 = h_pol, c5_pol

    return best_h, float(best_c5), len(best_h)
