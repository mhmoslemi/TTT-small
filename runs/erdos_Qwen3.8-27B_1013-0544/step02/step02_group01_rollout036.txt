For the second term, first part: delta(i+s, j)*h[i] means i = j-s, valid if 0 <= j-s <= n-1-s.
Contribution = h[j-s] if s <= j <= n-1, else 0.

For the second term, second part: delta(i, j)*h[i+s] means i = j, valid if 0 <= j <= n-1-s.
Contribution = h[j+s] if 0 <= j <= n-1-s, else 0.

So: dC(s)/dh[j] = (1 if s <= j <= n-1) - (h[j-s] if s <= j <= n-1) - (h[j+s] if 0 <= j <= n-1-s)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)

            if free.sum() > 0:
                h[free] = h[free] + diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_val_grad(h, beta, top_k):

    n = len(h0)
    target = n / 2.0
    
    def obj(x):
        v, _ = smoothmax_val_grad(x, beta, top_k)
        viol = x.sum() - target
        return v + penalty * viol * viol
    
    def objg(x):
        _, g = smoothmax_val_grad(x, beta, top_k)
        viol = x.sum() - target
        g = g + 2.0 * penalty * viol * np.ones(n)
        return g
    
    bounds = Bounds(np.zeros(n), np.ones(n))
    try:
        res = minimize(obj, h0, jac=objg, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-9, 'maxls': 50})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h0, target)

def subgrad_step_impl(h, lr, top_k, target):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argsort(corr)[-top_k:]
    j = np.arange(n)
    grad = np.zeros(n)
    for k in idx:
        s = k - (n - 1)
        js = j - s
        jp = j + s

            grad[ms] += (1.0 - h[js[ms]])
        if mp.any():
            grad[mp] += (-h[jp[mp]])
    grad = grad / top_k * dx
    return project_feasible(h - lr * grad, target)

from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-14:
            free = (h > 1e-10) & (h < 1.0 - 1e-10)
            if free.sum() > 0:
                h[free] = h[free] + diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h


def smoothmax_val_grad(h, beta, top_k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    if top_k < len(corr):
        idx = np.argpartition(corr, -top_k)[-top_k:]
        c = corr[idx]
    else:
        idx = np.arange(len(corr))
        c = corr
    m = np.max(c)
    exps = np.exp(beta * (c - m))
    sw = np.sum(exps)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for ii in range(len(idx)):
        wk = w[ii]
        if wk < 1e-12:
            continue
        s = idx[ii] - (n - 1)
        if s >= 0:
            grad[s:n] += wk * (1.0 - h[0:n - s])
            grad[0:n - s] += wk * (-h[s:n])
        else:
            sp = -s
            grad[0:n - sp] += wk * (1.0 - h[sp:n])
            grad[sp:n] += wk * (-h[0:n - sp])
    grad *= dx
    return val, grad


def lbfgsb_optimize(h0, maxiter, beta, top_k, penalty):
    n = len(h0)
    target = n / 2.0

    def obj(x):
        v, _ = smoothmax_val_grad(x, beta, top_k)
        viol = x.sum() - target
        return v + penalty * viol * viol

    def objg(x):
        _, g = smoothmax_val_grad(x, beta, top_k)
        viol = x.sum() - target
        g = g + 2.0 * penalty * viol * np.ones(n)
        return g

    bounds = Bounds(np.zeros(n), np.ones(n))
    try:
        res = minimize(obj, h0, jac=objg, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10, 'maxls': 40})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h0, target)


def slsqp_optimize(h0, maxiter, beta, top_k):
    n = len(h0)
    target = n / 2.0

    def obj(x):
        v, _ = smoothmax_val_grad(x, beta, top_k)
        return float(v)

    def objg(x):
        _, g = smoothmax_val_grad(x, beta, top_k)
        return g

    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h0, jac=objg, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h0, target)


def nesterov_subgrad(h, max_iters, lr0, decay, top_k):
    n = len(h)
    target = n / 2.0
    x = h.copy().astype(float)
    x_prev = x.copy()
    best_x = x.copy()
    best_v = compute_c5(x)
    for t in range(max_iters):
        lr = lr0 * (decay ** t)
        if t == 0:
            y = x.copy()
        else:
            mu = (t - 1) / (t + 2.0)
            y = x + mu * (x - x_prev)
        y = project_feasible(y, target)
        corr = np.correlate(y, 1.0 - y, mode="full")
        top_idx = np.argsort(corr)[-top_k:]
        j = np.arange(n)
        grad = np.zeros(n)
        for ki in top_idx:
            s = ki - (n - 1)
            js = j - s
            jp = j + s
            ms = (js >= 0) & (js < n)
            mp = (jp >= 0) & (jp < n)
            if ms.any():
                grad[ms] += (1.0 - y[js[ms]])
            if mp.any():
                grad[mp] += (-y[jp[mp]])
        grad = grad / top_k * (2.0 / n)
        x_new = project_feasible(y - lr * grad, target)
        v = compute_c5(x_new)
        if v < best_v:
            best_v = v
            best_x = x_new.copy()
        x_prev = x
        x = x_new
    return best_x, best_v


def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

    target = len(h) / 2.0
    pert = h + sigma * rng.standard_normal(len(h))
