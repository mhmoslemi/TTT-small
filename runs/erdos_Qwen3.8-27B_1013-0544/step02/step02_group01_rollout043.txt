import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-12:
            free = (h > 1e-8) & (h < 1.0 - 1e-8)
            if free.sum() > 0:
                h[free] = h[free] + diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_slsqp(h, beta, maxiter, top_k):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    j_arr = np.arange(n)
    
    def obj(x):
        corr = compute_corr(x)
        top_idx = np.argsort(corr)[-top_k:]
        m = np.max(corr[top_idx])
        sub = corr[top_idx]
        exps = np.exp(beta * (sub - m))
        sw = np.sum(exps)
        return (m + np.log(sw) / beta) * dx
    
    def objg(x):
        corr = compute_corr(x)
        top_idx = np.argsort(corr)[-top_k:]
        m = np.max(corr[top_idx])
        sub = corr[top_idx]
        exps = np.exp(beta * (sub - m))
        sw = np.sum(exps)
        w = exps / sw
        grad = np.zeros(n)
        for idx_k in range(len(top_idx)):
            wk = w[idx_k] * dx
            if wk < 1e-12:
                continue
            s = top_idx[idx_k] - (n - 1)
            js = j_arr - s
            jp = j_arr + s
            ms = (js >= 0) & (js < n)
            mp = (jp >= 0) & (jp < n)
            if ms.any():
                grad[ms] += wk * (1.0 - x[js[ms]])
            if mp.any():
                grad[mp] += wk * (-x[jp[mp]])
        return grad
    
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=objg, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)

def binary_local_search(h, max_iters=5000, seed=42):
    rng = np.random.RandomState(seed)
    n = len(h)
    target = n // 2
    # Threshold to binary
    sorted_h = np.sort(h)
    thresh = sorted_h[target - 1] if target > 0 else 0
    # Use top n/2 as 1s
    top_idx = np.argsort(h)[-target:]
    x = np.zeros(n)
    x[top_idx] = 1.0
    best_c5 = compute_c5(x)
    best_x = x.copy()
    
    for it in range(max_iters):
        ones = np.where(x == 1)[0]
        zeros = np.where(x == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        x[i], x[j] = 0.0, 1.0
        c = compute_c5(x)
        if c < best_c5 - 1e-12:
            best_c5 = c
            best_x = x.copy()
        else:
            x[i], x[j] = 1.0, 0.0
    return best_x, best_c5

def nesterov_subgrad(h, max_iters, lr0, decay, top_k=5):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    x = h.copy().astype(float)
    x_prev = x.copy()
    best_x = x.copy()
    best_v = compute_c5(x)
    j = np.arange(n)
    for t in range(max_iters):
        lr = lr0 * (decay ** t)
        if t == 0:
            y = x.copy()
        else:
            mu = (t - 1) / (t + 2)
            y = x + mu * (x - x_prev)
        y = project_feasible(y, target)
        corr = compute_corr(y)
        top_idx = np.argsort(corr)[-top_k:]
        lags = top_idx - (n - 1)
        grad = np.zeros(n)
        for s in lags:
            js = j - s
            jp = j + s
            ms = (js >= 0) & (js < n)
            mp = (jp >= 0) & (jp < n)
            if ms.any():
                grad[ms] += (1.0 - y[js[ms]])
            if mp.any():
                grad[mp] += (-y[jp[mp]])
        grad = grad / top_k * dx
        x_new = project_feasible(y - lr * grad, target)
        v = compute_c5(x_new)
        if v < best_v:
            best_v = v
            best_x = x_new.copy()
        x_prev = x
        x = x_new
    return best_x, best_v

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

            if c_s < best_c5:

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-12:
            free = (h > 1e-8) & (h < 1.0 - 1e-8)
            if free.sum() > 0:
                h[free] = h[free] + diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h


def slsqp_smoothmax(h, beta, maxiter, top_k):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    j_arr = np.arange(n)

    def obj(x):
        corr = compute_corr(x)
        top_idx = np.argsort(corr)[-top_k:]
        m = np.max(corr[top_idx])
        sub = corr[top_idx]
        exps = np.exp(beta * (sub - m))
        sw = np.sum(exps)
        return (m + np.log(sw) / beta) * dx

    def objg(x):
        corr = compute_corr(x)
        top_idx = np.argsort(corr)[-top_k:]
        m = np.max(corr[top_idx])
        sub = corr[top_idx]
        exps = np.exp(beta * (sub - m))
        sw = np.sum(exps)
        w = exps / sw
        grad = np.zeros(n)
        for idx_k in range(len(top_idx)):

        return grad
