class OptProblem:
    def __init__(self, beta, dx, target_sum):
        self.beta = beta
        self.dx = dx
        self.target_sum = target_sum
    def objective(self, x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = np.max(corr)
        return (m + np.log(np.sum(np.exp(self.beta * (corr - m)))) / self.beta) * self.dx
    def cons_fun(self, x):
        return np.sum(x) - self.target_sum
    def cons_jac(self, x):
        return np.ones(x.shape[0])

def slsqp_polish(h, target_sum, dx, beta, maxiter):
    n = len(h)
    prob = OptProblem(beta, dx, target_sum)
    bounds = Bounds(np.zeros(n), np.ones(n))
    constraints = [{'type': 'eq', 'fun': prob.cons_fun, 'jac': prob.cons_jac}]
    try:
        res = minimize(prob.objective, h, method='SLSQP', bounds=bounds,
                       constraints=constraints, options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_h(res.x, target_sum)
        return h_new, compute_c5(h_new)
    except:
        return h, compute_c5(h)

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def corr_arr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def active_grad(h, r_star):
    n = len(h)
    k = r_star - (n - 1)
    grad = np.zeros(n)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    elif k > 0:
        for i in range(n):
            g = 0.0
            if i <= n - 1 - k:
                g += 1.0 - h[i + k]
            if i >= k:
                g -= h[i - k]
            grad[i] = g
    else:
        m = -k
        for i in range(n):
            g = 0.0
            if i >= m:
                g += 1.0
                g -= h[i - m]
            if i <= n - 1 - m:
                g -= h[i + m]
            grad[i] = g
    grad -= np.mean(grad)
    return grad

def project_h(h, target_sum):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        if free.sum() == 0:
            break
        h[free] += diff / free.sum()
    return np.clip(h, 0.0, 1.0)

def active_set_refine(h, target_sum, dx, n_iter, lr_init):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    stall = 0
    for it in range(n_iter):
        corr = corr_arr(best_h)
        r_star = np.argmax(corr)
        lr = lr_init / (1.0 + 0.002 * it)
        grad = active_grad(best_h, r_star)
        improved = False
        for sm in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_new = best_h - lr * sm * grad
            h_new = project_h(h_new, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h_new
                improved = True
                stall = 0
                break
        if not improved:
            stall += 1
            if stall > 100:
                # Try second-worst lag
                corr2 = corr.copy()
                corr2[r_star] = -np.inf
                r2 = np.argmax(corr2)
                grad2 = active_grad(best_h, r2)
                grad_mix = 0.5 * (grad + grad2)
                for sm in [0.5, 0.25, 0.1]:
                    h_new = best_h - lr * sm * grad_mix
                    h_new = project_h(h_new, target_sum)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5 - 1e-15:
                        best_c5 = c5_new
                        best_h = h_new
                        improved = True
                        stall = 0
                        break
            if stall > 500:
                break
    return best_h, best_c5

def topk_balance_grad(h, top_k):
    n = len(h)
    corr = corr_arr(h)
    top_indices = np.argsort(corr)[::-1][:top_k]
    grad = np.zeros(n)
    for idx in top_indices:
        grad += active_grad(h, idx)
    grad /= top_k
    grad -= np.mean(grad)
    return grad

def topk_balance_refine(h, target_sum, dx, n_iter, lr_init, top_k):

class OptProblem:
    def __init__(self, beta, dx, target_sum):
        self.beta = beta
        self.dx = dx
        self.target_sum = target_sum
    def objective(self, x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = np.max(corr)
        return (m + np.log(np.sum(np.exp(self.beta * (corr - m)))) / self.beta) * self.dx
    def cons_fun(self, x):
        return np.sum(x) - self.target_sum
    def cons_jac(self, x):
        return np.ones(x.shape[0])

def slsqp_polish(h, target_sum, dx, beta, maxiter):
    n = len(h)
    prob = OptProblem(beta, dx, target_sum)
    bounds = Bounds(np.zeros(n), np.ones(n))
    constraints = [{'type': 'eq', 'fun': prob.cons_fun, 'jac': prob.cons_jac}]
    try:
        res = minimize(prob.objective, h, method='SLSQP', bounds=bounds,
                       constraints=constraints, options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_h(res.x, target_sum)
        return h_new, compute_c5(h_new)
    except Exception:
        return h.copy(), compute_c5(h)

            for restart in range(3):
                remaining = budget_s - (time.time() - t0)
                if remaining < 30:
                    break
                h_alt, c5_alt = active_set_refine(h_alt, alt_target, alt_dx, 2000, 0.01)
                h_alt, c5_alt = topk_balance_refine(h_alt, alt_target, alt_dx, 1000, 0.005, 3)
                if c5_alt < best_c5:
                    best_h, best_c5 = h_alt, c5_alt
                    n_points = alt_n
                    dx = alt_dx
                    target_sum = alt_target
                else:
                    h_alt = project_h(h_alt + rng.randn(alt_n) * 0.05, alt_target)

    # Phase 5: Final intensive refinement
    remaining = budget_s - (time.time() - t0)
    if remaining > 40:
        best_h, best_c5 = active_set_refine(best_h, target_sum, dx, 5000, 0.003)
        best_h, best_c5 = topk_balance_refine(best_h, target_sum, dx, 3000, 0.002, 4)

    remaining = budget_s - (time.time() - t0)
    if remaining > 25:
        for beta in [2000, 5000]:
            remaining = budget_s - (time.time() - t0)
            if remaining < 12:
                break
            h_new, c5_new = slsqp_polish(best_h, target_sum, dx, beta, 10000)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new
            best_h, best_c5 = active_set_refine(best_h, target_sum, dx, 2000, 0.002)

    # Final tiny steps
    remaining = budget_s - (time.time() - t0)
    if remaining > 5:
        best_h, best_c5 = active_set_refine(best_h, target_sum, dx, 2000, 0.001)

    best_h = project_h(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_points
