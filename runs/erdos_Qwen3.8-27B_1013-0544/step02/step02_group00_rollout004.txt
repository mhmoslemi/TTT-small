                            if cur_c5 < best_c5:
                                best = cur.copy()
                                best_c5 = cur_c5

    return best, best_c5

Hmm, but the coordinate descent with project_h at each step is expensive because project_h does up to 100 iterations. Let me optimize this.

Actually, for coordinate descent, if I change one element h[j] by delta, the sum changes by delta. I need to compensate by changing other elements. The simplest way is to distribute the compensation to all other elements that are not at 0 or 1.

But actually, a better approach for coordinate descent is:
- Pick two indices i, j
- Move h[i] up by delta and h[j] down by delta (preserves sum)
- Check if both stay in [0,1]
- This is O(1) to check and O(n) to evaluate

This is much more efficient. Let me use this "pair exchange" approach.

Let me also think about the SLSQP call. The issue is that SLSQP with n=140 variables and 279 equality constraints (from the smooth max) is actually just one equality constraint (sum = target). So it's n=140 variables, 1 equality constraint, 280 bounds. This should be fast.

But the objective function (smooth_max of the correlation) is expensive to evaluate because np.correlate is O(n^2). For n=140, that's about 19600 operations per evaluation, which should be fine.

The issue with SLSQP is that it might get stuck. Let me use it briefly and then switch to the more aggressive methods.

Let me also think about what n_points to use. The parent has n=140. Using n=140 is probably good because:
- It's the resolution of the parent
- Finer resolutions might not help (the structure is determined at this scale)
- Coarser resolutions lose information

But I should also try n=70 (half) and n=280 (double) to see if they help.

Actually, wait. The problem says "Smaller sequences with less than 1k samples are preferred - they are faster to optimize and evaluate." So n=140 is fine.

Let me finalize my approach:

1. Start from initial_h_values (n=140)
2. Brief SLSQP (beta=200)
3. Vectorized subgradient with momentum (3000 steps)
4. Pair-exchange local search (20000 iterations)
5. Another SLSQP (beta=500)
6. Another subgradient round (2000 steps)
7. Final coordinate/pair search (10000 iterations)
8. Return best

For the pair-exchange, I need to be smart about which pairs to try. The key insight is: if lag L is the maximum correlation lag, then I want to reduce corr[L]. To do this, I should either:
- Decrease h[i] for some i where h[i] is large and h[i+L] is small (this decreases the product h[i]*(1-h[i+L]))

Wait, actually corr[L] = sum_i h[i] * (1-h[i+L]). To decrease this:
- Decrease h[i] where (1-h[i+L]) is large (i.e., h[i+L] is small)
- Or increase h[i+L] where h[i] is large

So a targeted pair exchange would be:
- Find the lag L with maximum correlation
- Find pairs (i, i+L) where h[i] is high and h[i+L] is low
- Exchange: increase h[i+L] and decrease h[i]

This is more targeted than random pairs. Let me implement this.

import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta


def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-8) & (h < 1.0 - 1e-8)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def subgrad_vectorized(h, top_k=4):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_indices = np.argsort(corr)[::-1][:top_k]
    grad = np.zeros(n)
    for lag_idx in top_indices:
        lag = int(lag_idx) - (n - 1)
        if lag == 0:
            grad[:] += (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += (1.0 - h[:n - lag])
            grad[:n - lag] -= h[lag:]
        else:
            la = -lag
            grad[:n - la] += (1.0 - h[la:])
            grad[la:] -= h[:n - la]
    return grad, corr


def optimize_subgrad(h, target_sum, n_steps=3000, top_k=4, step0=0.04, seed=42):
    n = len(h)
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    best = cur.copy()
    best_c5 = cur_c5
    rng = np.random.RandomState(seed)
    momentum = np.zeros(n)
    mu = 0.85
    for it in range(n_steps):
        grad, corr = subgrad_vectorized(cur, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        grad /= gnorm
        step = step0 / (1.0 + 0.005 * it)
        momentum = mu * momentum - step * grad
        h_trial = project_h(cur + momentum, target_sum)
        c5_trial = compute_c5(h_trial)
        if c5_trial < cur_c5 - 1e-15:
            cur = h_trial
            cur_c5 = c5_trial
            if cur_c5 < best_c5:
                best = cur.copy()
                best_c5 = c5_trial
        else:
            momentum *= 0.5
            h_trial = project_h(cur + momentum, target_sum)
            c5_trial = compute_c5(h_trial)
            if c5_trial < cur_c5 - 1e-15:
                cur = h_trial
                cur_c5 = c5_trial
                if cur_c5 < best_c5:
                    best = cur.copy()
                    best_c5 = cur_c5
    return best, best_c5


def targeted_pair_search(h, target_sum, n_iters=30000, seed=42):
    n = len(h)
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    best = cur.copy()
    best_c5 = cur_c5
    rng = np.random.RandomState(seed)
    for it in range(n_iters):
        corr = np.correlate(cur, 1.0 - cur, mode="full")
        max_lag_idx = int(np.argmax(corr))
        lag = max_lag_idx - (n - 1)
        if lag == 0:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
        elif lag > 0:
            i = rng.randint(0, n - lag)
            j = i + lag
        else:
            la = -lag
            i = rng.randint(la, n)
            j = i - la
        for mag in [0.06, 0.03, 0.015]:
            for direction in [1, -1]:
                h_trial = cur.copy()
                if direction == 1:
                    h_trial[i] -= mag
                    h_trial[j] += mag
                else:
                    h_trial[i] += mag
                    h_trial[j] -= mag
                if h_trial[i] < 0 or h_trial[i] > 1 or h_trial[j] < 0 or h_trial[j] > 1:
                    continue
                c5_trial = compute_c5(h_trial)
                if c5_trial < cur_c5 - 1e-14:
                    cur = h_trial
                    cur_c5 = c5_trial
                    if cur_c5 < best_c5:
                        best = cur.copy()
                        best_c5 = cur_c5
                    break
    return best, best_c5


def random_pair_search(h, target_sum, n_iters=15000, seed=43):
    n = len(h)
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    best = cur.copy()
    best_c5 = cur_c5
    rng = np.random.RandomState(seed)
    for it in range(n_iters):
        i, j = rng.choice(n, 2, replace=False)
        lag_diff = abs(i - j)
        if lag_diff == 0:
            continue
        mag = rng.uniform(0.01, 0.08)
        d = 1 if rng.rand() < 0.5 else -1
        h_trial = cur.copy()
        h_trial[i] += d * mag
        h_trial[j] -= d * mag
        if h_trial[i] < 0 or h_trial[i] > 1 or h_trial[j] < 0 or h_trial[j] > 1:
            continue
        c5_trial = compute_c5(h_trial)
        if c5_trial < cur_c5 - 1e-14:
            cur = h_trial
            cur_c5 = c5_trial
            if cur_c5 < best_c5:
                best = cur.copy()
                best_c5 = c5_trial
    return best, best_c5


def slsqp_round(h, target_sum, beta, maxiter):
    n = len(h)
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))

    def obj_fn(h_in):
        c = np.correlate(h_in, 1.0 - h_in, mode="full")
        return smooth_max(c, beta) * dx

    def con_fn(x_in):
        return np.sum(x_in) - target_sum

    def con_jac(x_in):
        return np.ones(len(x_in))

    cons = [{'type': 'eq', 'fun': con_fn, 'jac': con_jac}]
    try:
        res = minimize(obj_fn, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = project_h(res.x, target_sum)
        return h_new
    except Exception:
        return h
