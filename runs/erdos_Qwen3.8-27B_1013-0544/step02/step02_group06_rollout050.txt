```python
    f_high = np.max(a * t_high + b)

    if np.allclose(a, 0):
```

```python
        vals = a * mid + b
        f_val = np.max(vals)
        active = np.isclose(vals, f_val, atol=1e-10)
        min_slope = np.min(a[active])
        max_slope = np.max(a[active])
        if min_slope < 0 and max_slope > 0:
            break  # Found the minimum
        elif min_slope > 0:
            hi = mid  # All active slopes positive, need to decrease t
        else:
            lo = mid  # All active slopes negative, need to increase t
    return (lo + hi) / 2
```

```python
def project_box_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
```

```python
    best_n = 0

    h_init = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()

    ns = [70]
    if tl() > 350:
        ns.append(100)
    if tl() > 650:
        ns.append(140)

    for n in ns:
        if tl() < 80:
            break
        ts = n / 2.0
        bnds = [(0.0, 1.0)] * n

        starts = []
        if h_init is not None and len(h_init) == n:
            starts.append(project_box_sum(h_init.copy(), ts))
        if h_init is not None and len(h_init) != n and len(h_init) > 10:
            starts.append(interpolate_h(h_init, len(h_init), n))
        for _ in range(10):
            starts.append(make_start(n, rng, ts))
        for _ in range(3):
            h_b = (rng.rand(n) > 0.5).astype(float) + 0.05 * rng.randn(n)
            starts.append(project_box_sum(h_b, ts))

        for h_s in starts:
            if tl() < 45:
                break
            h_cur = h_s.copy()
            c_cur = compute_c5(h_cur)
            if c_cur < best_c5:
                best_h, best_c5, best_n = h_cur.copy(), c_cur, n

            for beta in [20, 50, 100, 200, 500, 1000, 2000]:
                if tl() < 25:
                    break
                lam = 5.0 + beta * 0.15
                try:
                    r = minimize(obj_lse, h_cur, args=(n, ts, beta, lam),
                                 method='L-BFGS-B', bounds=bnds,
                                 options={'maxiter': 300, 'ftol': 1e-14})
                    h_new = project_box_sum(r.x, ts)
                    c_new = compute_c5(h_new)
                    if c_new < c_cur:
                        h_cur, c_cur = h_new, c_new
                        if c_cur < best_c5:
                            best_h, best_c5, best_n = h_cur.copy(), c_cur, n
                except Exception:
                    pass

            if tl() > 25:
                h_cur = coord_sweep(h_cur, n, ts, n_passes=2)
                c_cur = compute_c5(h_cur)
                if c_cur < best_c5:
                    best_h, best_c5, best_n = h_cur.copy(), c_cur, n

            if tl() > 25:
                for k in (3, 5):
                    if tl() < 15:
                        break
                    for beta2 in (1000, 2500):
                        if tl() < 12:
                            break
                        try:
                            r = minimize(obj_topk, h_cur, args=(n, ts, beta2, k, 100.0),
                                         method='L-BFGS-B', bounds=bnds,
                                         options={'maxiter': 200, 'ftol': 1e-14})
                            h_new = project_box_sum(r.x, ts)
                            c_new = compute_c5(h_new)
                            if c_new < c_cur:
                                h_cur, c_cur = h_new, c_new
                                if c_cur < best_c5:
                                    best_h, best_c5, best_n = h_cur.copy(), c_cur, n
                        except Exception:
                            pass

        if best_n == n and tl() > 50:
            for beta in (3000, 6000, 12000):
                if tl() < 20:
                    break
                lam = 500.0
                try:
                    r = minimize(obj_lse, best_h, args=(n, ts, beta, lam),
                                 method='L-BFGS-B', bounds=bnds,
                                 options={'maxiter': 3000, 'ftol': 1e-16, 'gtol': 1e-14})
                    h_new = project_box_sum(r.x, ts)
                    c_new = compute_c5(h_new)
                    if c_new < best_c5:
                        best_h, best_c5 = h_new, c_new
                except Exception:
                    pass
            best_h = coord_sweep(best_h, n, ts, n_passes=3)
            best_c5 = compute_c5(best_h)

    # SLSQP final try
    if best_h is not None and tl() > 30:
        n = best_n
        ts = best_n / 2.0
        bnds = [(0.0, 1.0)] * n
        try:
            from scipy.optimize import minimize as _min
            res = _min(slsqp_fun, best_h, args=(n, ts), method='SLSQP', bounds=bnds,
                       options={'maxiter': 800, 'ftol': 1e-16})
            h_new = project_box_sum(res.x, ts)
            c_new = compute_c5(h_new)
            if c_new < best_c5:
                best_h, best_c5 = h_new, c_new
        except Exception:
            pass

    if best_h is not None:
        ts = best_n / 2.0
        best_h = project_box_sum(best_h, ts)
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
```
