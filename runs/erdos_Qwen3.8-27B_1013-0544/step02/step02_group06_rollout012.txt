    h = 0.5 + a1 * np.sin(w1 * x + p1) + a2 * np.sin(w2 * x + p2)
    h = np.clip(h, 0, 1)
    return project_box_sum(h, n / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h_init)
    else:
        n0 = 70
        h_init = np.full(n0, 0.5)

    ts0 = n0 / 2.0
    best_h = project_box_sum(h_init, ts0)
    best_c5 = compute_c5(best_h)

    def tl():
        return budget_s - (time.time() - t0)

    # Phase 1: Quick L-BFGS-B polish from initial
    for beta, lam in [(100, 200), (300, 500), (600, 1000)]:
        if tl() < 60:
            break
        try:
            h_new = lbfgs_polish(best_h, n0, ts0, beta, lam, 400, 1e-16)
            c = compute_c5(h_new)
            if c < best_c5:
                best_c5, best_h = c, h_new
        except Exception:
            pass

    # Phase 2: Active-set gradient descent with momentum
    for n in [n0, 60, 80]:
        if tl() < 120:
            break
        ts = n / 2.0
        # Starting points
        starts = []
        if n == n0:
            starts.append(best_h.copy())
        for tr in range(4):
            starts.append(gen_parametric_start(n, seed * 1000 + n * 10 + tr))
        # Random
        starts.append(project_box_sum(np.random.uniform(0, 1, n), ts))

        for si, h_s in enumerate(starts):
            if tl() < 60:
                break
            h_curr = h_s.copy()
            v = np.zeros(n)
            lr = 0.005
            mu = 0.85
            top_k = 12
            for it in range(3000):
                if tl() < 20:
                    break
                # Stochastic focusing: vary top_k and beta
                tk = top_k + (it // 500) * 2
                tk = min(tk, 30)
                bf = 30.0 + it * 0.1

                h_trial, _ = active_set_step(h_curr, n, ts, tk, bf, lr, mu)
                h_trial = project_box_sum(h_trial, ts)

                c5_trial = compute_c5(h_trial)
                c5_curr = compute_c5(h_curr)
                if c5_trial < c5_curr:
                    v = mu * v + lr * (h_curr - h_trial)
                    h_curr = h_trial
                    lr = min(lr * 1.005, 0.03)
                else:
                    v *= 0.5
                    lr *= 0.85

                # Periodic L-BFGS boost
                if it % 500 == 499 and tl() > 40 and compute_c5(h_curr) <= best_c5 + 0.002:
                    try:
                        h_b = lbfgs_polish(h_curr, n, ts, 500, 800, 200, 1e-15)
                        cb = compute_c5(h_b)
                        if cb < compute_c5(h_curr):
                            h_curr = h_b
                    except Exception:
                        pass

            c5_final = compute_c5(h_curr)
            if c5_final < best_c5:
                best_c5, best_h = c5_final, h_curr.copy()

    # Phase 3: Perturbation + re-optimization on best
    for it in range(30):
        if tl() < 80:
            break
        n_b = len(best_h)
        ts_b = n_b / 2.0
        h_pert = best_h.copy()
        strat = it % 5
        if strat == 0:
            h_pert += np.random.randn(n_b) * 0.03
        elif strat == 1:
            # Flip a block
            blen = np.random.randint(5, max(6, n_b // 5))
            bs = np.random.randint(0, n_b - blen)
            h_pert[bs:bs+blen] = 1.0 - h_pert[bs:bs+blen]
        elif strat == 2:
            # Sinusoidal perturbation
            x = np.arange(n_b) * 2.0 / n_b
            h_pert += 0.04 * np.sin(np.random.uniform(2, 8) * np.pi * x + np.random.uniform(0, 6.28))
        elif strat == 3:
            # Swap pairs
            for _ in range(np.random.randint(3, 8)):
                i, j = np.random.choice(n_b, 2, replace=False)
                h_pert[i], h_pert[j] = h_pert[j], h_pert[i]
        else:
            # Local random walk
            mask = np.random.random(n_b) < 0.3
            h_pert[mask] += np.random.randn(mask.sum()) * 0.05

        h_pert = project_box_sum(h_pert, ts_b)

        # Active-set descent from perturbed
        h_c = h_pert.copy()
        lr = 0.003
        for sub_it in range(2000):
            if tl() < 15:
                break
            tk = 15
            h_t, _ = active_set_step(h_c, n_b, ts_b, tk, 40.0, lr, 0.9)
            h_t = project_box_sum(h_t, ts_b)
            if compute_c5(h_t) < compute_c5(h_c):
                h_c = h_t
                lr = min(lr * 1.01, 0.02)
            else:
                lr *= 0.9

        # L-BFGS polish
        if tl() > 20:
            for beta, lam in [(800, 1500), (1500, 3000)]:
                if tl() < 15:
                    break
                try:
                    h_p = lbfgs_polish(h_c, n_b, ts_b, beta, lam, 500, 1e-16)
                    c = compute_c5(h_p)
                    if c < best_c5:
                        best_c5, best_h = c, h_p
                        h_c = h_p
                except Exception:
                    pass

    # Phase 4: Final high-precision polish
    if tl() > 30:
        n_f = len(best_h)
        ts_f = n_f / 2.0
        for beta in [3000, 8000]:
            if tl() < 15:
                break
            try:
                h_f = lbfgs_polish(best_h, n_f, ts_f, beta, 10000, 5000, 1e-18)
                c = compute_c5(h_f)
                if c < best_c5:
                    best_c5, best_h = c, h_f
            except Exception:
                pass

    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
