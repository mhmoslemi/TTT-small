    for _ in range(500):

        if abs(diff) < 1e-13:

            tot = room.sum()

            if tot < 1e-15:
                break
            h += (diff / tot) * h

def make_coef(h, n, i):
    c = np.zeros(2 * n - 1)
    ks = np.arange(max(0, i), min(2 * n - 2, i + n - 1) + 1)
    js = i + n - 1 - ks
    m = (js >= 0) & (js < n)
    c[ks[m]] += 1.0 - h[js[m]]
    ks2 = np.arange(max(0, n - 1 - i), min(2 * n - 2, n - 1 + i) + 1)
    js2 = i + ks2 - (n - 1)
    m2 = (js2 >= 0) & (js2 < n)
    c[ks2[m2]] -= h[js2[m2]]
    return c

        mx = v.max()

        return a[act].max(), a[act].min()
