```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = h.copy()
    for _ in range(50):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() > 0:
            h[free] += diff / free.sum()
        else:
            h += diff / len(h)
    return np.clip(h, 0.0, 1.0)

def smooth_max_and_grad(corr, beta):
    m = np.max(corr)
    shifted = beta * (corr - m)
    exps = np.exp(shifted)
    Z = np.sum(exps)
    p = exps / Z  # softmax weights, length 2n-1
    F = (m + np.log(Z) / beta)  # smooth max value (before *dx)
    return F, p

def compute_gradient(h, p, dx):

    n = len(h)
    # p is the softmax weights over correlations, length 2n-1
    # p[k] corresponds to lag k-(n-1)
    grad = np.zeros(n)
    for j in range(n):
        val = 0.0
        for k_offset in range(-(n-1), n):

            if 0 <= j_minus < n:
                term -= h[j_minus]
            val += p[k] * term
        grad[j] = dx * val
```

```python
def coordinate_optimize(h, j, target_sum_nots_needed):
    """Optimize h[j] holding others fixed. Returns optimal value of h[j]."""
    n = len(h)
    dx = 2.0 / n
    # For each lag k, c_k(t) = t * a_k + b_k
    # a_k = (1-h[j+k])_valid - h[j-k]_valid
    # b_k = corr_k - h[j] * a_k (using current corr)
    
    corr = np.correlate(h, 1.0 - h, mode="full")
    d = np.arange(-(n-1), n)  # lags
    
    a = np.zeros(2*n-1)
    for m, d_k in enumerate(d):
        jp = j + d_k
        jm = j - d_k
        a[m] = 0.0
        if 0 <= jp < n:
            a[m] += (1.0 - h[jp])
        if 0 <= jm < n:
            a[m] -= h[jm]
    
    b = corr - h[j] * a  # current value minus contribution of h[j]
    
    # Now find min of max_k (a_k * t + b_k) * dx over t in [0, 1]
    # Since dx is positive, minimize max_k (a_k * t + b_k)
    
    # Candidate points: t=0, t=1, and all crossing points in (0,1)
    candidates = [0.0, 1.0]
    
    # Find crossing points: a_i * t + b_i = a_j * t + b_j
    # t = (b_j - b_i) / (a_i - a_j)
    for i in range(2*n-1):
        for jj in range(i+1, 2*n-1):
            da = a[i] - a[jj]
            if abs(da) > 1e-12:
                t_cross = (b[jj] - b[i]) / da
                if 0 < t_cross < 1:
                    candidates.append(t_cross)
    
    # Evaluate at all candidates
    best_t = 0.0
    best_val = np.inf
    for t in candidates:
        t = np.clip(t, 0, 1)
        vals = a * t + b
        max_val = np.max(vals)
        if max_val < best_val:
            best_val = max_val
            best_t = t
    
    return best_t
```
