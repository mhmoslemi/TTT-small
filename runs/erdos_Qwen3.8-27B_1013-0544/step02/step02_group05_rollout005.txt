import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    if beta < 1e-6:
        return m
    shifted = beta * (x - m)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) > 1e-12 and s > 0:
        h += (target_sum - s) / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def binarize(h, target_ones):
    n = len(h)
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target_ones]] = 1.0
    return hb

def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, float(beta)) * dx

def obj_active(h, dx, beta, top_k):
    corr = np.correlate(h, 1.0 - h, mode="full")
    sorted_idx = np.argsort(corr)[::-1]
    top = corr[sorted_idx[:top_k]]
    m = np.max(top)
    val = m + np.log(np.sum(np.exp(beta * (top - m)))) / beta
    return val * dx

def con_sum(x, target_sum):
    return np.sum(x) - target_sum

def con_jac(x, target_sum):
    return np.ones(len(x))

def discrete_sa(h_start, target_ones, n_iter, t_init, t_final):
    n = len(h_start)
    h = binarize(h_start, target_ones)
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    curr_cost = compute_c5(h)
    best_h = h.copy()
    best_cost = curr_cost
    ratio = t_final / t_init
    for it in range(n_iter):
        T = t_init * (ratio ** (it / max(n_iter - 1, 1)))
        oi = ones[np.random.randint(len(ones))]
        zi = zeros[np.random.randint(len(zeros))]
        h[oi] = 0
        h[zi] = 1
        new_cost = compute_c5(h)
        delta = new_cost - curr_cost
        if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-12)):
            curr_cost = new_cost
            ones = np.where(h == 1)[0].copy()
            zeros = np.where(h == 0)[0].copy()
            if new_cost < best_cost:
                best_cost = new_cost
                best_h = h.copy()
        else:
            h[oi] = 1
            h[zi] = 0
    return best_h, best_cost
