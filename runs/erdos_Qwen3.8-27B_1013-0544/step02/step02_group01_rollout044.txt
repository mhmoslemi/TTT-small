        best_idx = np.argmin(pop_vals)
        h_cand = pop[best_idx]
        h_r, c_r = grad_descent_adaptive(h_cand, 1500, 30, 200, 0.04, 0.999)
        if c_r < best_c5:
            h_best, best_c5 = h_r, c_r
        # Update population with refined best
        pop[0] = project_feasible(h_best + rng.randn(n_pts) * 0.02, n_pts / 2.0)
        pop_vals[0] = compute_c5(pop[0])
        # Add diverse individuals
        for i in range(1, min(5, len(pop))):
            h_div = block_perturb(h_best, rng, rng.randint(3, 8), rng.randint(4, 12))
            pop[i] = h_div
            pop_vals[i] = compute_c5(h_div)
        ga_elapsed += time.time() - gt

    # Phase 4: Multi-resolution attempts
    for n_try in [100, 150, 250, 300]:
        if tl() < 120:
            break
        if n_try == n_pts:
            continue
        h_interp = interp_to_n(h_best, n_try)
        h_r, c_r = grad_descent_adaptive(h_interp, 2000, 20, 200, 0.04, 0.999)
        if c_r < best_c5:
            h_best, best_c5 = h_r, c_r
        if tl() > 80:
            h_s = slsqp_refine(h_r, 400, 200)
            c_s = compute_c5(h_s)
            if c_s < best_c5:
                h_best, best_c5 = h_s, c_s

    # Phase 5: Aggressive polishing rounds
    for rnd in range(5):
        if tl() < 40:
            break
        beta_s = 50 + rnd * 50
        h_r, c_r = grad_descent_adaptive(h_best, 3000, beta_s, beta_s * 4, 0.03, 0.9995)
        if c_r < best_c5:
            h_best, best_c5 = h_r, c_r
        if tl() > 25:
            h_s = slsqp_refine(h_best, 600, 300 + rnd * 100)
            c_s = compute_c5(h_s)
            if c_s < best_c5:
                h_best, best_c5 = h_s, c_s

    # Phase 6: Final ultra-high beta SLSQP
    if tl() > 20:
        h_f = slsqp_refine(h_best, 1000, 600)
        c_f = compute_c5(h_f)
        if c_f < best_c5:
            h_best, best_c5 = h_f, c_f

    return h_best, best_c5, len(h_best)
