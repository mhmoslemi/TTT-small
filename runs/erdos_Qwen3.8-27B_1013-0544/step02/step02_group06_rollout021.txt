```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def project_box_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h > 0) & (h < 1)
        if m.sum() == 0:
            break
        h[m] += d / m.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_1d_solve(a, b):
    """Minimize max_k(a[k]*x + b[k]) for x in [0,1] using golden section."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    for _ in range(60):
        m1 = hi - gr * (hi - lo)
        m2 = lo + gr * (hi - lo)
        f1 = float(np.max(a * m1 + b))
        f2 = float(np.max(a * m2 + b))
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def coordinate_descent(h, n, target, max_sweeps=500, tol=1e-10):
    for sweep in range(max_sweeps):
        max_change = 0.0
        for j in range(n):
            old_val = h[j]
            h[j] = 0.0
            c0 = np.correlate(h, 1.0 - h, mode="full")
            h[j] = 1.0
            c1 = np.correlate(h, 1.0 - h, mode="full")
            h[j] = old_val
            a = c1 - c0
            b = c0
            best_x = coord_1d_solve(a, b)
            h[j] = best_x
            max_change = max(max_change, abs(best_x - old_val))
        h = project_box_sum(h, target)
        if max_change < tol:
            break
    return h

    c = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(c, beta) * (2.0 / n) + lam * (h.sum() - target) ** 2
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def project_box_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        mask = (h > 0.0) & (h < 1.0)
        if mask.sum() == 0:
            break
        h[mask] += d / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_1d_solve(a, b):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    for _ in range(50):
        m1 = hi - gr * (hi - lo)
        m2 = lo + gr * (hi - lo)
        if float(np.max(a * m1 + b)) < float(np.max(a * m2 + b)):
            hi = m2
        else:
            lo = m1
    return (lo + hi) * 0.5

def coord_descent(h, n, target, max_sweeps=200, tol=1e-10):
    for _ in range(max_sweeps):
        max_change = 0.0
        for j in range(n):
            old = h[j]
            h[j] = 0.0
            c0 = np.correlate(h, 1.0 - h, mode="full")
            h[j] = 1.0
            c1 = np.correlate(h, 1.0 - h, mode="full")
            h[j] = old
            best_x = coord_1d_solve(c1 - c0, c0)
            h[j] = best_x
            max_change = max(max_change, abs(best_x - old))
        h = project_box_sum(h, target)
        if max_change < tol:
            break
    return h

def binary_coord_descent(h, n, target, max_sweeps=100):
    for _ in range(max_sweeps):
        max_change = 0.0
        for j in range(n):
            old = h[j]
            h[j] = 0.0
            c0 = np.correlate(h, 1.0 - h, mode="full")
            v0 = float(np.max(c0))
            h[j] = 1.0
            c1 = np.correlate(h, 1.0 - h, mode="full")
            v1 = float(np.max(c1))
            h[j] = 0.0 if v0 <= v1 else 1.0
            max_change = max(max_change, abs(h[j] - old))
        h = project_box_sum(h, target)
        if max_change < 1e-10:
            break
    return h

def obj_ln(h, n, target, beta, lam):
    c = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(c, beta) * (2.0 / n) + lam * (h.sum() - target) ** 2

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(70, 0.5)

    n = len(h0)
    target = n / 2.0
    bnds = [(0.0, 1.0)] * n

    best_h = project_box_sum(h0, target)
    best_c5 = compute_c5(best_h)

    def time_left():
        return budget_s - (time.time() - t0)

    # Phase 1: Coordinate descent (exact 1D subproblem per coordinate)
    if time_left() > 300:
        h_cd = coord_descent(best_h.copy(), n, target, max_sweeps=300)
        c = compute_c5(h_cd)
        if c < best_c5:
            best_h, best_c5 = h_cd, c

    # Phase 2: Binary coordinate descent for structure discovery
    if time_left() > 200:
        h_bin = binary_coord_descent(best_h.copy(), n, target, max_sweeps=50)
        c = compute_c5(h_bin)
        if c < best_c5:
            best_h, best_c5 = h_bin, c
        # Run continuous CD from binary solution
        h_cd2 = coord_descent(h_bin, n, target, max_sweeps=200)
        c = compute_c5(h_cd2)
        if c < best_c5:
            best_h, best_c5 = h_cd2, c

    # Phase 3: L-BFGS-B polish
    for beta, lam in [(200, 100), (500, 200), (1500, 500)]:
        if time_left() < 60:
            break
        try:
            r = minimize(obj_ln, best_h.copy(), args=(n, target, beta, lam),
                         method='L-BFGS-B', bounds=bnds,
                         options={'maxiter': 500, 'ftol': 1e-15})
            h_new = project_box_sum(r.x, target)
            c = compute_c5(h_new)
            if c < best_c5:
                best_h, best_c5 = h_new, c
        except Exception:
            pass

    # Phase 4: Diversified restarts with CD + polish
    for it in range(40):
        if time_left() < 180:
            break
        h = best_h.copy()
        s = it % 6
        if s == 0:
            h += np.random.randn(n) * 0.06
        elif s == 1:
            h += np.random.randn(n) * 0.12
        elif s == 2:
            bl = np.random.randint(5, 20)
            bs = np.random.randint(0, n - bl)
            h[bs:bs + bl] = 1.0 - h[bs:bs + bl]
        elif s == 3:
            w = np.random.uniform(1.0, 6.0) * np.pi
            ph = np.random.uniform(0, 2 * np.pi)
            h += 0.04 * np.sin(w * np.arange(n) * 2.0 / n + ph)
        elif s == 4:
            for _ in range(3):
                bl = np.random.randint(2, 8)
                bs = np.random.randint(0, n - bl)
                h[bs:bs + bl] = np.random.uniform(0, 1, bl)
        else:
            for _ in range(np.random.randint(3, 8)):
                i, j = np.random.choice(n, 2, replace=False)
                h[i], h[j] = h[j], h[i]

        h = project_box_sum(h, target)
        h_cd = coord_descent(h, n, target, max_sweeps=150)
        c = compute_c5(h_cd)

        if c < best_c5 and time_left() > 40:
            try:
                r = minimize(obj_ln, h_cd, args=(n, target, 500, 300),
                             method='L-BFGS-B', bounds=bnds,
                             options={'maxiter': 300, 'ftol': 1e-15})
                h_new = project_box_sum(r.x, target)
                c2 = compute_c5(h_new)
                if c2 < c:
                    c, h_cd = c2, h_new
            except Exception:
                pass

        if c < best_c5:
            best_h, best_c5 = h_cd.copy(), c

    # Phase 5: Final high-precision polish
    if time_left() > 25:
        for beta in [2000, 5000]:
            if time_left() < 15:
                break
            try:
                r = minimize(obj_ln, best_h, args=(n, target, beta, 2000.0),
                             method='L-BFGS-B', bounds=bnds,
                             options={'maxiter': 3000, 'ftol': 1e-16})
                h_new = project_box_sum(r.x, target)
                c = compute_c5(h_new)
                if c < best_c5:
                    best_h, best_c5 = h_new, c
            except Exception:
                pass

    # Phase 6: Final CD pass
    if time_left() > 30:
        h_final = coord_descent(best_h.copy(), n, target, max_sweeps=200)
        c = compute_c5(h_final)
        if c < best_c5:
            best_h, best_c5 = h_final, c

    best_h = project_box_sum(best_h, target)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
```
