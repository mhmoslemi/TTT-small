    dx = 2.0 / n

            js = j - s
            jp = j + s

            mp = (jp >= 0) & (jp < n)
            if ms.any():
                grad[ms] += (1.0 - y[js[ms]])
            if mp.any():
                grad[mp] += (-y[jp[mp]])

        x_new = np.clip(y - lr * grad, 0, 1)

        if v < best_v:

            best_x = x_new.copy()

        x = x_new

            e = d_i - d_p

            delta_max = min(1.0 - h[i], h[i+1])
            
            if delta_max - delta_min < 1e-15:

            a, b = delta_min, delta_max

            c = b - gr * (b - a)
            d = a + gr * (b - a)

                    fd = fc

                h[i+1] -= delta_opt
                corr = np.correlate(h, 1.0 - h, mode="full")
                c5 = compute_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
    
    return h, best_c5

def slsqp_refine(h, maxiter, beta):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    sw = np.sum(exps)
    w = exps / sw

    def objg(x):
        c = np.correlate(x, 1.0 - x, mode="full")
        mm = np.max(c)
        ex = np.exp(beta * (c - mm))
        s_ = np.sum(ex)
        ww = ex / s_
        val = (mm + np.log(s_) / beta) * dx
        grad = np.zeros(n)
        j = np.arange(n)
        for k in range(len(c)):
            wk = ww[k]
            if wk < 1e-12:
                continue
            s = k - (n - 1)
            js = j - s
            jp = j + s
            ms = (js >= 0) & (js < n)
            mp = (jp >= 0) & (jp < n)
            if ms.any():
                grad[ms] += wk * (1.0 - x[js[ms]])
            if mp.any():
                grad[mp] += wk * (-x[jp[mp]])
        return val, grad * dx

    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(objg, h, jac=True, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_sum(res.x, target)
    except Exception:
        return project_sum(h, target)


def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_sum(h_new, n_new / 2.0)


def binarize_smooth(h, alpha):

    target = n / 2.0
    h_b = np.where(h > 0.5, 1.0, 0.0)
    h_b = project_sum(h_b, target)
    return (1.0 - alpha) * h + alpha * h_b
