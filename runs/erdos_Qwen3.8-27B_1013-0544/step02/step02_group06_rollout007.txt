        if c5 < best_c5:

            best_c5 = c5
        h = h_new

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h > 0.0) & (h < 1.0)
        if m.sum() == 0:
            break
        h[m] += d / m.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, n, beta):
    c = np.correlate(h, 1.0 - h, mode="full")
    cmax = np.max(c)
    e = np.exp(beta * (c - cmax))
    w = e / e.sum()
    obj = (cmax + np.log(e.sum()) / beta) * (2.0 / n)
    grad = np.zeros(n)
    for j in range(len(c)):
        wk = w[j]
        if wk < 1e-13:
            continue
        shift = j - (n - 1)
        lo = max(0, shift)
        hi = min(n, n + shift)
        if lo < hi:
            idx = np.arange(lo, hi)
            grad[lo:hi] += wk * (1.0 - h[idx - shift])
        lo2 = max(0, -shift)
        hi2 = min(n, n - shift)
        if lo2 < hi2:
            idx2 = np.arange(lo2, hi2)
            grad[lo2:hi2] += wk * (-h[idx2 + shift])
    grad *= (2.0 / n)
    return obj, grad

def _lbfgs_fun(h, n, beta, ts, lam):
    obj, grad = smooth_max_and_grad(h, n, beta)
    obj += lam * (h.sum() - ts) ** 2
    grad += 2.0 * lam * (h.sum() - ts) / n
    return obj, grad

def adam_optimize(h, n, beta_fn, n_steps, ts, base_lr=0.015):
    m = np.zeros(n)
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    b1, b2, eps = 0.85, 0.97, 1e-8
    for t in range(n_steps):
        frac = t / max(1, n_steps - 1)
        beta = beta_fn(frac)
        obj, grad = smooth_max_and_grad(h, n, beta)
        grad += 20.0 * (h.sum() - ts) / n
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad ** 2
        m_hat = m / (1.0 - b1 ** (t + 1))
        v_hat = v / (1.0 - b2 ** (t + 1))
        lr = base_lr * (1.0 - 0.5 * frac)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_box_sum(h_new, ts)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = c5
        h = h_new
    return best_h, best_c5

def _lbfgs_optimize(h, n, ts, beta, lam, maxiter=500):
    bnds = [(0.0, 1.0)] * n
    def _f(x):
        val, _ = _lbfgs_fun(x, n, beta, ts, lam)
        return val
    def _g(x):
        _, g = _lbfgs_fun(x, n, beta, ts, lam)
        return g
    try:
        r = minimize(_f, h, jac=_g, method='L-BFGS-B', bounds=bnds,
                     options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10})
        return project_box_sum(r.x, ts)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    else:
        n = 70
        h_init = np.full(n, 0.5)

    ts = n / 2.0
    best_h = project_box_sum(h_init, ts)
    best_c5 = compute_c5(best_h)

    def remaining():
        return budget_s - (time.time() - t0)

    # Phase 1: ADAM with beta annealing from multiple starts
    starts = [best_h.copy()]
    for _ in range(5):
        x = np.linspace(0, 2.0, n, endpoint=False)
        a = np.random.uniform(0.15, 0.40)
        w = np.random.uniform(1.0, 4.0) * np.pi
        p = np.random.uniform(0, 2 * np.pi)
        a2 = np.random.uniform(-0.15, 0.15)
        w2 = np.random.uniform(3.0, 8.0) * np.pi
        p2 = np.random.uniform(0, 2 * np.pi)
        h_s = 0.5 + a * np.sin(w * x + p) + a2 * np.sin(w2 * x + p2)
        starts.append(project_box_sum(h_s, ts))

    schedules = [
        lambda f: 40 + 460 * f,
        lambda f: 100 + 900 * f,
        lambda f: 300 + 1700 * f,
    ]

    for si, h_s in enumerate(starts):
        if remaining() < 80:
            break
        for sched in schedules:
            if remaining() < 50:
                break
            h_opt, c5_opt = adam_optimize(h_s.copy(), n, sched, 400, ts, base_lr=0.02)
            if c5_opt < best_c5:
                best_h, best_c5 = h_opt, c5_opt

    # Phase 2: L-BFGS-B with exact gradient, increasing beta
    for beta, lam in [(200, 50), (500, 100), (1000, 200), (2000, 500), (5000, 1000)]:
        if remaining() < 40:
            break
        h_new = _lbfgs_optimize(best_h, n, ts, beta, lam, maxiter=800)
        c = compute_c5(h_new)
        if c < best_c5:
            best_h, best_c5 = h_new, c

    # Phase 3: Structured restarts with ADAM + L-BFGS polish
    it = 0
    while remaining() > 120:
        it += 1
        h = best_h.copy()
        strat = it % 7

        if strat == 0:
            scale = max(0.008, 0.04 * np.exp(-it / 15.0))
            h += np.random.randn(n) * scale
        elif strat == 1:
            corr = np.correlate(h, 1.0 - h, mode="full")
            top_lags = np.argsort(corr)[-3:]
            for tl_ in top_lags:
                k_shift = int(tl_) - (n - 1)
                for j in range(n):
                    i1 = j + k_shift
                    i2 = j - k_shift
                    step = 0.008
                    if 0 <= i1 < n and h[j] < 1.0 and h[i1] > 0.0:
                        h[j] += step
                        h[i1] -= step
        elif strat == 2:
            w = np.random.uniform(0.5, 3.0) * 2 * np.pi / n
            ph = np.random.uniform(0, 2 * np.pi)
            h += 0.025 * np.sin(w * np.arange(n) + ph)
        elif strat == 3:
            blen = np.random.randint(4, max(5, n // 5))
            bs = np.random.randint(0, n - blen)
            h[bs:bs + blen] += np.random.randn(blen) * 0.06
        elif strat == 4:
            idx = np.random.choice(n, size=n // 3, replace=False)
            h[idx] += np.random.randn(len(idx)) * 0.03
        elif strat == 5:
            # Smooth step perturbation on random interval
            c = np.random.randint(0, n)
            w = np.random.randint(5, n // 4)
            amp = np.random.uniform(0.02, 0.06) * np.random.choice([-1, 1])
            env = np.exp(-0.5 * ((np.arange(n) - c) / (w / 3.0)) ** 2)
            h += amp * env
        else:
            h += np.random.randn(n) * 0.012

        h = project_box_sum(h, ts)

        # ADAM short run
        if remaining() > 50:
            h_opt, c5_opt = adam_optimize(h, n, lambda f: 400 + 600 * f, 150, ts, base_lr=0.015)
            if c5_opt < best_c5:
                best_h, best_c5 = h_opt, c5_opt

        # L-BFGS polish
        if remaining() > 35 and c5_opt < best_c5 + 0.0005:
            h_pol = h_opt if c5_opt < best_c5 else h
            h_new = _lbfgs_optimize(h_pol, n, ts, 1500, 800, maxiter=400)
            c = compute_c5(h_new)
            if c < best_c5:
                best_h, best_c5 = h_new, c

    # Phase 4: Final high-precision polish
    for beta, lam in [(3000, 2000), (8000, 5000), (15000, 10000)]:
        if remaining() < 15:
            break
        h_new = _lbfgs_optimize(best_h, n, ts, beta, lam, maxiter=3000)
        c = compute_c5(h_new)
        if c < best_c5:
            best_h, best_c5 = h_new, c

    # Final ADAM micro-polish
    if remaining() > 10:
        h_final, c_final = adam_optimize(best_h, n, lambda f: 5000 + 5000 * f, 100, ts, base_lr=0.005)
        if c_final < best_c5:
            best_h, best_c5 = h_final, c_final

    best_h = project_box_sum(best_h, ts)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
