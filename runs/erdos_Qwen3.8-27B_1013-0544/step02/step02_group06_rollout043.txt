import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def project_box_sum(h, ts):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        d = ts - h.sum()
        if abs(d) < 1e-13:
            break
        mask = (h > 0.0) & (h < 1.0)
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_descent(h, n, ts, beta, n_sweeps):
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode="full").copy()
    for sweep in range(n_sweeps):
        idx = np.random.permutation(n) if sweep % 3 == 2 else np.arange(n)
        for i in idx:
            old = h[i]
            adj = np.zeros(2 * n - 1)
            adj[i:i + n] = 1.0 - 2.0 * h
            lo, hi = 0.0, 1.0
            for _ in range(28):
                t1 = lo + (hi - lo) / 3.0
                t2 = hi - (hi - lo) / 3.0
                f1 = smooth_max(corr + (t1 - old) * adj, beta)
                f2 = smooth_max(corr + (t2 - old) * adj, beta)
                if f1 < f2:
                    hi = t2
                else:
                    lo = t1
            h[i] = (lo + hi) / 2.0
            corr = corr + (h[i] - old) * adj
        h = project_box_sum(h, ts)
        corr = np.correlate(h, 1.0 - h, mode="full")
    return h

def lbfgs_obj(v, beta, lam, n, ts):
    c = np.correlate(v, 1.0 - v, mode="full")
    return smooth_max(c, beta) * (2.0 / n) + lam * (v.sum() - ts) ** 2

def make_sawtooth(n, periods, phase):
    x = np.arange(n) / float(n)
    return np.abs((x * periods + phase) % 1.0 - 0.5) * 2.0

def make_triangular(n, periods, phase):
    x = np.arange(n) / float(n)
    return 1.0 - np.abs(2.0 * ((x * periods + phase) % 1.0) - 1.0)

def make_sinusoid(n, freq, amp, phase):
    x = 2.0 * np.arange(n) / float(n)
    return 0.5 + amp * np.sin(np.pi * freq * x + phase)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    else:
        n = 70
        h0 = np.full(n, 0.5)

    ts = n / 2.0
    bnds = [(0.0, 1.0)] * n
    best_h = project_box_sum(h0, ts)
    best_c5 = compute_c5(best_h)

    # Generate diverse starting points
    starts = [best_h.copy()]
    for p in [1, 2, 3]:
        for ph in [0.0, 0.25, 0.5, 0.75]:
            starts.append(project_box_sum(make_sawtooth(n, p, ph), ts))
            starts.append(project_box_sum(make_triangular(n, p, ph), ts))
    for f in [1, 2, 3, 4, 5, 7, 9, 11]:
        for amp in [0.25, 0.35, 0.45]:
            starts.append(project_box_sum(make_sinusoid(n, f, amp, 0), ts))
            starts.append(project_box_sum(make_sinusoid(n, f, amp, 1.2), ts))
    for _ in range(12):
        starts.append(project_box_sum(np.random.uniform(0, 1, n), ts))
    # Multi-scale combinations
    for f1 in [2, 3, 5]:
        for f2 in [7, 11, 13]:
            x = 2.0 * np.arange(n) / float(n)
            hc = 0.5 + 0.25 * np.sin(2 * np.pi * f1 * x) + 0.15 * np.sin(2 * np.pi * f2 * x)
            starts.append(project_box_sum(hc, ts))

    # Quick evaluate all starts
    for hs in starts:
        c = compute_c5(hs)
        if c < best_c5:
            best_h, best_c5 = hs.copy(), c

    # Main optimization loop
    n_iter = 0
    while budget_s - (time.time() - t0) > 60:
        n_iter += 1
        if n_iter <= len(starts):
            h_cur = starts[n_iter - 1].copy()
        else:
            scale = max(0.005, 0.18 * np.exp(-n_iter / 25.0))
            h_cur = project_box_sum(best_h + np.random.randn(n) * scale, ts)

        # Curriculum: coordinate descent with increasing beta
        if budget_s - (time.time() - t0) > 20:
            beta_cd = min(40 + 25 * n_iter, 600)
            h_cur = coord_descent(h_cur, n, ts, beta_cd, 8)
            c_cd = compute_c5(h_cur)
            if c_cd < best_c5:
                best_h, best_c5 = h_cur.copy(), c_cd

        # L-BFGS-B polish
        if budget_s - (time.time() - t0) > 25:
            beta_lb = min(400 + 250 * n_iter, 5000)
            lam = min(150 * n_iter, 2000)
            try:
                r = minimize(lbfgs_obj, h_cur, args=(beta_lb, lam, n, ts),
                             method='L-BFGS-B', bounds=bnds,
                             options={'maxiter': 250, 'ftol': 1e-16})
                h_new = project_box_sum(r.x, ts)
                c_new = compute_c5(h_new)
                if c_new < best_c5:
                    best_h, best_c5 = h_new, c_new
                    h_cur = h_new
            except Exception:
                pass

        # After L-BFGS-B, refine with higher-beta coordinate descent
        if c_new if 'c_new' in dir() else 0 < best_c5 or n_iter % 3 == 0:
            if budget_s - (time.time() - t0) > 15:
                beta_cd2 = min(200 + 100 * n_iter, 1000)
                h_cur = coord_descent(best_h, n, ts, beta_cd2, 5)
                c2 = compute_c5(h_cur)
                if c2 < best_c5:
                    best_h, best_c5 = h_cur.copy(), c2

        if n_iter % 4 == 0:
            starts.append(best_h.copy())
            if len(starts) > 50:
                starts = starts[-50:]

    # Final high-precision polish
    if budget_s - (time.time() - t0) > 10:
        best_h = coord_descent(best_h, n, ts, 500, 20)
        best_c5 = compute_c5(best_h)

    if budget_s - (time.time() - t0) > 12:
        for bf in [3000, 6000]:
            if budget_s - (time.time() - t0) < 8:
                break
            try:
                r = minimize(lbfgs_obj, best_h, args=(bf, 4000, n, ts),
                             method='L-BFGS-B', bounds=bnds,
                             options={'maxiter': 3000, 'ftol': 1e-18})
                h_new = project_box_sum(r.x, ts)
                c = compute_c5(h_new)
                if c < best_c5:
                    best_h, best_c5 = h_new, c
            except Exception:
                pass

    if budget_s - (time.time() - t0) > 5:
        best_h = coord_descent(best_h, n, ts, 700, 25)
        best_c5 = compute_c5(best_h)

    best_h = project_box_sum(best_h, ts)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
