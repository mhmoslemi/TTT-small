    top_idx = np.argsort(corr)[::-1][:top_k]
    grad = np.zeros(n)
    for idx in top_idx:
        lag = int(idx) - (n - 1)
        if lag >= 0:
            grad[:n - lag] += (1.0 - h[lag:])
            grad[lag:] -= h[:n - lag]
        else:
            la = -lag
            grad[la:] += (1.0 - h[:n - la])
            grad[:n - la] -= h[la:]
    return grad, corr

def sml_obj(h_in, beta, dx):
    c = np.correlate(h_in, 1.0 - h_in, mode="full")
    return smooth_max(c, beta) * dx

def sum_eq(x_in, target):
    return np.sum(x_in) - target

def sum_eq_jac(x_in, target):
    return np.ones(x_in.shape)

def resample(h, n_new):
    if len(h) == n_new:
        return h.copy()
    return np.interp(np.linspace(0.0, 2.0, n_new), np.linspace(0.0, 2.0, len(h)), h)

def do_slsqp(h0, n, target, beta, bounds, maxiter, ftol):
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (target,)}]
    try:
        r = minimize(sml_obj, h0, args=(beta, 2.0 / n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(r.x, target)
    except Exception:
        return h0

def subgrad_desc(cur, target, n, n_iter, step, t0, budget):
    mom = np.zeros(n)
    best_local = compute_c5(cur)
    for it in range(n_iter):
        if budget - (time.time() - t0) < 3:
            break
        g, _ = subgradient_c5(cur, 4)
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        g /= gn
        mom = 0.85 * mom - step * g
        trial = project_h(cur + mom, target)
        c_try = compute_c5(trial)
        if c_try < best_local - 1e-14:
            cur = trial
            best_local = c_try
        else:
            mom *= 0.4
            trial = project_h(cur + mom, target)
            c_try = compute_c5(trial)
            if c_try < best_local - 1e-14:
                cur = trial
                best_local = c_try
            else:
                mom = np.zeros(n)
                # Try even smaller step
                trial = project_h(cur - 0.3 * step * g, target)
                c_try = compute_c5(trial)
                if c_try < best_local - 1e-14:
                    cur = trial
                    best_local = c_try
    return cur

def coord_descent(cur, target, n, sweeps, t0, budget):
    best_c5 = compute_c5(cur)
    for sw in range(sweeps):
        if budget - (time.time() - t0) < 2:
            break
        improved = False
        for j in range(n):
            if budget - (time.time() - t0) < 1:
                break
            for d in (0.08, -0.08, 0.04, -0.04, 0.02, -0.02):
                trial = cur.copy()
                trial[j] += d
                trial = project_h(trial, target)
                c_try = compute_c5(trial)
                if c_try < best_c5 - 1e-14:
                    cur = trial
                    best_c5 = c_try
                    improved = True
                    break
        if not improved and sw > 5:
            break
    return cur, best_c5

    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:

    resolutions = sorted(set([n_base]))

        if r >= n_base:

        bounds = Bounds(np.zeros(n_pts), np.ones(n_pts))

        cur = project_h(resample(best_h, n_pts), target)
        cur_c5 = compute_c5(cur)
        if cur_c5 < best_c5:
            best_h, best_c5, best_n = cur.copy(), cur_c5, n_pts

        # SLSQP phase
        for beta in [50, 100, 200, 500]:
            if budget_s - (time.time() - t0) < 12:

            cur = do_slsqp(cur, n_pts, target, beta, bounds, 3000, 1e-14)
            cur_c5 = compute_c5(cur)
            if cur_c5 < best_c5:

        # Subgradient phase
        if budget_s - (time.time() - t0) > 25:

            cur_c5 = compute_c5(cur)
            if cur_c5 < best_c5:

        # Coordinate descent
        if budget_s - (time.time() - t0) > 15:

            if cur_c5 < best_c5:

        # Perturbation and re-optimize
        if budget_s - (time.time() - t0) > 35:
            for att in range(4):

                    break

                hp = project_h(cur + np.random.randn(n_pts) * scale, target)
                hp = do_slsqp(hp, n_pts, target, 200, bounds, 1000, 1e-13)

                if cur_c5 < best_c5:

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_c5(h, top_k):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[::-1][:top_k]
