```python
def corr_grad(h, k):
    n = len(h)
    grad = np.zeros(n)
    # First term: (1-h[j-k]) for j where 0 <= j-k < n, i.e., k <= j < k+n
    # j-k ranges from 0 to n-1, so j ranges from k to k+n-1
    # But j must also be in [0, n-1], so j ranges from max(0,k) to min(n-1, k+n-1)
    # If k >= 0: j from k to n-1, j-k from 0 to n-1-k
    # If k < 0: j from 0 to n-1+k, j-k from -k to n-1
    
    if k >= 0:
        idx = np.arange(k, n)
        grad[idx] += (1.0 - h[np.arange(0, n-k)])
    else:
        idx = np.arange(0, n+k)
        grad[idx] += (1.0 - h[np.arange(-k, n)])
    
    # Second term: -h[j+k] for j where 0 <= j+k < n, i.e., -k <= j < n-k
    # j must be in [0, n-1], so j from max(0,-k) to min(n-1, n-k-1)
    # If k >= 0: j from 0 to n-k-1, j+k from k to n-1
    # If k < 0: j from -k to n-1, j+k from 0 to n-1+k
    
    if k >= 0:
        idx = np.arange(0, n-k)
        grad[idx] -= h[np.arange(k, n)]
    else:
        idx = np.arange(-k, n)
        grad[idx] -= h[np.arange(0, n+k)]
    
    return grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    if beta < 1:
        return m
    return m + np.log1p(np.sum(np.exp(beta * (x - m))) - 1.0) / beta

def project_box_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() == 0:
            break
        h[mask] += d / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_grad(h, k):
    n = len(h)
    grad = np.zeros(n)
    if k >= 0:
        grad[k:] += (1.0 - h[:n - k])
        grad[:n - k] -= h[k:]
    else:
        jk = -k
        grad[:n - jk] += (1.0 - h[jk:])
        grad[jk:] -= h[:n - jk]
    return grad

def obj_global(h, n, beta, lam, ts):
    c = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(c, beta) * (2.0 / n) + lam * (h.sum() - ts) ** 2

def obj_slsqp(h, n, beta):
    c = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(c, beta) * (2.0 / n)

def sum_fun(x, ts):
    return x.sum() - ts

def sum_jac(x):
    return np.ones(len(x))

def subgrad_descent(h, n, ts, n_steps, lr_init, top_k):
    h = h.copy()
    lr = lr_init
    best_local = compute_c5(h)
    for it in range(n_steps):
        corr = np.correlate(h, 1.0 - h, mode="full")
        top_idx = np.argsort(corr)[-top_k:]
        grad = np.zeros(n)
        for ki in top_idx:
            k_shift = int(ki) - (n - 1)
            grad += corr_grad(h, k_shift) / top_k
        h -= lr * grad
        h = project_box_sum(h, ts)
        c = compute_c5(h)
        if c < best_local:
            best_local = c
        if it % 20 == 19:
            lr *= 0.9
    return h, best_local

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    best_c5 = 1e10
    best_h = None
    best_n = 0

    n_list = [70, 100, 140]
    has_init = 'initial_h_values' in globals() and initial_h_values is not None

    for n in n_list:
        if budget_s - (time.time() - t0) < 80:
            break
        ts = n / 2.0
        bnds = [(0.0, 1.0)] * n

        if has_init and n == len(initial_h_values):
            h0 = np.array(initial_h_values, dtype=float).copy()
        else:
            h0 = np.full(n, 0.5)
            x = np.arange(n)
            h0 += 0.25 * np.sin(4.0 * np.pi * x / n) + 0.1 * np.sin(8.0 * np.pi * x / n)
            h0 = project_box_sum(h0, ts)

        # Phase A: subgradient descent
        h_s, c_s = subgrad_descent(h0, n, ts, 300, 0.03, 3)
        if c_s < best_c5:
            best_c5, best_h, best_n = c_s, h_s.copy(), n
        h_cur = h_s

        # Phase B: L-BFGS-B with increasing beta
        for beta, lam in [(50, 50), (150, 150), (400, 300), (800, 600), (1500, 1000)]:
            if budget_s - (time.time() - t0) < 40:
                break
            try:
                r = minimize(obj_global, h_cur, args=(n, beta, lam, ts),
                             method='L-BFGS-B', bounds=bnds,
                             options={'maxiter': 600, 'ftol': 1e-16, 'gtol': 1e-10})
                h_new = project_box_sum(r.x, ts)
                c = compute_c5(h_new)
                if c < best_c5:
                    best_c5, best_h, best_n = c, h_new.copy(), n
                    h_cur = h_new
            except Exception:
                pass

        # Phase C: SLSQP with exact sum constraint
        if budget_s - (time.time() - t0) > 50:
            cons = [{'type': 'eq', 'fun': sum_fun, 'args': (ts,), 'jac': sum_jac}]
            for beta in [200, 500]:
                if budget_s - (time.time() - t0) < 25:
                    break
                try:
                    r = minimize(obj_slsqp, h_cur, args=(n, beta),
                                 method='SLSQP', bounds=bnds, constraints=cons,
                                 options={'maxiter': 500, 'ftol': 1e-14})
                    h_new = project_box_sum(r.x, ts)
                    c = compute_c5(h_new)
                    if c < best_c5:
                        best_c5, best_h, best_n = c, h_new.copy(), n
                        h_cur = h_new
                except Exception:
                    pass

        # Phase D: Diverse restarts
        it = 0
        while budget_s - (time.time() - t0) > 80:
            it += 1
            h_r = h_cur.copy()
            strat = it % 6
            if strat == 0:
                scale = max(0.003, 0.04 * np.exp(-it / 25.0))
                h_r += np.random.randn(n) * scale
            elif strat == 1:
                corr = np.correlate(h_r, 1.0 - h_r, mode="full")
                k_star = int(np.argmax(corr)) - (n - 1)
                h_r -= 0.015 * corr_grad(h_r, k_star)
                h_r += np.random.randn(n) * 0.002
            elif strat == 2:
                blen = np.random.randint(3, max(4, n // 7))
                bs = np.random.randint(0, n - blen)
                h_r[bs:bs + blen] = 0.5 + 0.35 * np.random.randn(blen)
            elif strat == 3:
                for _ in range(np.random.randint(2, 7)):
                    i = np.random.randint(0, n // 2)
                    j = n - 1 - i
                    h_r[i], h_r[j] = h_r[j], h_r[i]
            elif strat == 4:
                w = np.random.uniform(1, 5) * 2.0 * np.pi / n
                ph = np.random.uniform(0, 2.0 * np.pi)
                h_r += 0.025 * np.sin(w * np.arange(n) + ph)
            else:
                # Multi-lag subgradient
                corr = np.correlate(h_r, 1.0 - h_r, mode="full")
                top3 = np.argsort(corr)[-3:]
                for ki in top3:
                    h_r -= 0.005 * corr_grad(h_r, int(ki) - (n - 1))
                h_r += np.random.randn(n) * 0.003

            h_r = project_box_sum(h_r, ts)
            for beta in [300, 700]:
                if budget_s - (time.time() - t0) < 35:
                    break
                try:
                    r = minimize(obj_global, h_r, args=(n, beta, 400, ts),
                                 method='L-BFGS-B', bounds=bnds,
                                 options={'maxiter': 300, 'ftol': 1e-15})
                    h_new = project_box_sum(r.x, ts)
                    c = compute_c5(h_new)
                    if c < best_c5:
                        best_c5, best_h, best_n = c, h_new.copy(), n
                        h_cur = h_new
                except Exception:
                    pass

    # Final high-precision polish
    if budget_s - (time.time() - t0) > 15:
        n = best_n
        ts = n / 2.0
        bnds = [(0.0, 1.0)] * n
        for beta in [3000, 8000]:
            if budget_s - (time.time() - t0) < 10:
                break
            try:
                r = minimize(obj_global, best_h, args=(n, beta, 5000, ts),
                             method='L-BFGS-B', bounds=bnds,
                             options={'maxiter': 8000, 'ftol': 1e-18, 'gtol': 1e-13})
                h_new = project_box_sum(r.x, ts)
                c = compute_c5(h_new)
                if c < best_c5:
                    best_c5, best_h = c, h_new
            except Exception:
                pass
        # SLSQP final
        if budget_s - (time.time() - t0) > 10:
            cons = [{'type': 'eq', 'fun': sum_fun, 'args': (ts,), 'jac': sum_jac}]
            try:
                r = minimize(obj_slsqp, best_h, args=(n, 3000),
                             method='SLSQP', bounds=bnds, constraints=cons,
                             options={'maxiter': 2000, 'ftol': 1e-16})
                h_new = project_box_sum(r.x, ts)
                c = compute_c5(h_new)
                if c < best_c5:
                    best_c5, best_h = c, h_new
            except Exception:
                pass

    best_h = project_box_sum(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
```
