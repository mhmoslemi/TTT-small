def compute_c5(h):

    dx = 2.0 / n

    return float(np.max(corr) * dx)


def smooth_max(corr, beta):
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -100, 100)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            break
        h[free] += diff / free.sum()
    return np.clip(h, 0.0, 1.0)


def gradient_smooth_max(h, beta, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -100, 100)
    w = np.exp(shifted)
    w /= w.sum()
    grad_h = np.zeros(n)
    n_corr = len(corr)
    for k in range(n_corr):
        L = k - (n - 1)
        wv = w[k]
        if wv < 1e-15:
            continue
        i_min = max(0, -L)
        i_max = min(n - 1, n - 1 - L)
        for i in range(i_min, i_max + 1):
            grad_h[i] += wv * (1.0 - h[i + L])
            grad_h[i + L] += wv * (-h[i])
    return grad_h * beta * dx


def slsqp_refine(h0, beta, bounds, n_points, target_sum, maxiter=3000):

        return x.sum() - target_sum

        return np.ones(n_points)

        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                      constraints=cons, jac=jac,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        return res.x
    except Exception:
        return h0

import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max(corr, beta):
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -100, 100)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            break
        h[free] += diff / free.sum()
    return np.clip(h, 0.0, 1.0)


def gradient_smooth_max(h, beta, dx):

    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)

        s = x.sum() - self.target_sum
        return f + self.penalty * s * s
    def grad(self, x):
        g = gradient_smooth_max(x, self.beta, self.dx)
        s = x.sum() - self.target_sum
        g += 2.0 * self.penalty * s
        return g

    try:
        res = minimize(prob.obj, h0, method='SLSQP', bounds=bounds,
                      constraints=cons, jac=prob.obj_grad,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        return res.x
    except Exception:
        return h0

    except Exception:

def fista_run(h0, beta, dx, target_sum, n_iter):
    n = len(h0)
    x = h0.copy()
    z = h0.copy()
    t_k = 1.0
    step = 1.0 / (beta * dx * 3.0 + 0.01)
    obj_prev = smooth_max(np.correlate(x, 1 - x, mode="full"), beta) * dx
    for it in range(n_iter):
        grad = gradient_smooth_max(z, beta, dx)
        y = z - step * grad
        y = project_h(y, target_sum)
        obj_y = smooth_max(np.correlate(y, 1 - y, mode="full"), beta) * dx
        bt = 0
        while obj_y > obj_prev + 1e-12 and bt < 15:
            step *= 0.5
            y = z - step * grad
            y = project_h(y, target_sum)
            obj_y = smooth_max(np.correlate(y, 1 - y, mode="full"), beta) * dx
            bt += 1
        if abs(obj_y - obj_prev) < 1e-15:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        z = y + ((t_k - 1.0) / t_new) * (y - x)
        z = project_h(z, target_sum)
        x = y.copy()
        t_k = t_new
        obj_prev = obj_y
        step = min(step * 1.5, 1.0 / (beta * dx * 3.0 + 0.01))
    return x

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max(corr, beta):
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -100, 100)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            break
        h[free] += diff / free.sum()
    return np.clip(h, 0.0, 1.0)


def gradient_smooth_max(h, beta, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -100, 100)
    w = np.exp(shifted)
    w /= w.sum()
    grad_h = np.zeros(n)
    n_corr = len(corr)
    for k in range(n_corr):
        L = k - (n - 1)
        wv = w[k]
        if wv < 1e-15:
            continue
        i_min = max(0, -L)
        i_max = min(n - 1, n - 1 - L)
        for i in range(i_min, i_max + 1):
            grad_h[i] += wv * (1.0 - h[i + L])

                h_new = project_h(h_new, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h, best_c5 = h_new.copy(), c5
                    c = h_new.copy()

    # Phase 4: Active-lag targeted descent
    elapsed = time.time() - start_time
    if budget_s - elapsed > 40:
        for _ in range(80):
            elapsed = time.time() - start_time
            if budget_s - elapsed < 10:
                break
            grad = active_lag_gradient(best_h, n)
            gn = np.linalg.norm(grad)
            if gn < 1e-12:
                break
            grad /= gn
            improved = False
            for step in [0.0002, 0.0005, 0.001, 0.002, 0.004, 0.008, 0.015]:
                c = best_h - step * grad
                c = project_h(c, target_sum)
                c5 = compute_c5(c)
                if c5 < best_c5 - 1e-13:
                    best_h, best_c5 = c.copy(), c5
                    improved = True
                    break
            if not improved:
                c = fista_run(best_h, 800, dx, target_sum, n_iter=400)
                c5 = compute_c5(c)
                if c5 < best_c5:
                    best_h, best_c5 = c.copy(), c5
                else:
                    # Try small random perturbations
                    found = False
                    for _r in range(20):
                        c2 = best_h + rng.randn(n) * 0.003
                        c2 = project_h(c2, target_sum)
                        c2 = fista_run(c2, 500, dx, target_sum, n_iter=200)
                        c5_2 = compute_c5(c2)
                        if c5_2 < best_c5:
                            best_h, best_c5 = c2.copy(), c5_2
                            found = True
                            break
                    if not found:
                        break

    # Phase 5: Final high-precision polish
    for beta in [5000, 10000, 20000]:
        elapsed = time.time() - start_time
        if budget_s - elapsed < 8:
            break
        h_new = slsqp_optimize(best_h, beta, dx, n, target_sum, maxiter=8000)
        h_new = project_h(h_new, target_sum)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_h, best_c5 = h_new.copy(), c5

    best_h = project_h(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
