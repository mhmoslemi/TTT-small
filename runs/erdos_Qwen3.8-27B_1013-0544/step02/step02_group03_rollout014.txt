```python
    dx = 2.0 / n

    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_smooth(h, dx, beta):

        grad[idx + la] += (1.0 - h[idx])
```

```python
    others = np.array([j for j in range(n) if j != i])
    # Only adjust those not at boundary
    adjustable = others[(h[others] > 0) & (h[others] < 1)]
    if len(adjustable) == 0:
        return None
    h[adjustable] -= diff / len(adjustable)
    h = np.clip(h, 0, 1)
    # Final projection
    h = project_feasible(h, target_sum)
    return h
```

```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx


def con_fun(h, target_sum, n):
    return h.sum() - target_sum


def con_jac(h, target_sum, n):
    return np.ones(n)


def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def subgrad_at_lag(h, lag, n):
    grad = np.zeros(n)
    if lag >= 0:
        idx = np.arange(n - lag)
        grad[idx] += (1.0 - h[idx + lag])
        grad[idx + lag] -= h[idx]
    else:
        la = -lag
        idx = np.arange(n - la)
        grad[idx + la] += (1.0 - h[idx])
        grad[idx] -= h[idx + la]
    return grad


def weighted_subgrad(h, n, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    e = np.clip(beta * (corr - m), -700.0, 700.0)
    w = np.exp(e)
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(len(corr)):
        if w[k] < 1e-12:
            continue
        lag = k - (n - 1)
        g = subgrad_at_lag(h, lag, n)
        grad += w[k] * g
    return grad


def do_slsqp_phase(h, n, dx, target_sum, bounds, beta_list, max_iter, ftol, time_left_fn):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for beta in beta_list:
        if time_left_fn() < 30:
            break
        cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum, n)}]
        for trial in range(3):
            if time_left_fn() < 15:
                break
            x0 = best_h.copy()
            if trial > 0:
                scale = 0.012 / trial
                x0 += np.random.randn(n) * scale
                x0 = project_feasible(x0, target_sum)
            try:
                res = minimize(obj_smooth, x0, args=(dx, beta), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': max_iter, 'ftol': ftol})
                h_new = project_feasible(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5
            except Exception:
                pass
    return best_h, best_c5


def subgrad_phase(h, n, dx, target_sum, max_iter, time_left_fn):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h = h.copy()
    for it in range(max_iter):
        if time_left_fn() < 40:
            break
        grad = weighted_subgrad(h, n, 30.0)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        lr = 0.003 / (1.0 + 0.001 * it)
        h_new = h - lr * (grad / gnorm) * gnorm
        h_new = project_feasible(h_new, target_sum)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = c5
            h = h_new
        else:
            h = best_h.copy()
    return best_h, best_c5


def coord_descent_phase(h, n, target_sum, n_iters, time_left_fn, rng):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    candidates = [0.0, 0.1, 0.25, 0.4, 0.5, 0.6, 0.8, 0.9, 1.0]
    for it in range(n_iters):
        if time_left_fn() < 20:
            break
        i = rng.randint(n)
        for cv in candidates:
            if abs(cv - h[i]) < 1e-8:
                continue
            h_try = h.copy()
            h_try[i] = cv
            h_try = project_feasible(h_try, target_sum)
            c5 = compute_c5(h_try)
            if c5 < best_c5 - 1e-10:
                best_h = h_try.copy()
                best_c5 = c5
                h = h_try
                break
    return best_h, best_c5
```
