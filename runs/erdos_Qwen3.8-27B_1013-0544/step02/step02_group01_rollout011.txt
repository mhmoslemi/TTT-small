import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-12:
            free = (h > 1e-8) & (h < 1.0 - 1e-8)
            if free.sum() > 0:
                h[free] = h[free] + diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    sw = np.sum(exps)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    j = np.arange(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-10:
            continue
        s = k - (n - 1)
        js = j - s
        jp = j + s
        ms = (js >= 0) & (js < n)

        if mp.any():

def nesterov_opt(h, max_iters, lr0, decay, top_k, time_limit):
    """Nesterov accelerated subgradient with top-k lags."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    x = project_feasible(h.copy(), target)
    x_prev = x.copy()
    best_x = x.copy()
    best_v = compute_c5(x)
    j = np.arange(n)
    t0 = time.time()
    
    for t in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        lr = lr0 * (decay ** t)
        if t == 0:
            y = x.copy()
        else:
            mu = (t - 1) / (t + 2)
            y = x + mu * (x - x_prev)
        y = project_feasible(y, target)
        
        corr = np.correlate(y, 1.0 - y, mode="full")
        # Adaptive top-k: use more lags when close to optimum
        k_use = min(top_k, len(corr))
        top_idx = np.argsort(corr)[-k_use:]
        lags = top_idx - (n - 1)
        
        grad = np.zeros(n)
        for s in lags:
            js = j - s
            jp = j + s
            ms = (js >= 0) & (js < n)
            mp = (jp >= 0) & (jp < n)
            if ms.any():
                grad[ms] += (1.0 - y[js[ms]])
            if mp.any():
                grad[mp] += (-y[jp[mp]])
        grad = grad / k_use * dx
        
        x_new = project_feasible(y - lr * grad, target)
        v = compute_c5(x_new)
        if v < best_v:
            best_v = v
            best_x = x_new.copy()
        x_prev = x
        x = x_new
    
    return best_x, best_v

def slsqp_refine(h, maxiter, beta):
    n = len(h)
    target = n / 2.0
    
    def obj(x):
        v, _ = smoothmax_val_grad(x, beta)
        return v
    
    def objg(x):
        _, g = smoothmax_val_grad(x, beta)
        return g
    
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=objg, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-12:
            free = (h > 1e-8) & (h < 1.0 - 1e-8)
            if free.sum() > 0:
                h[free] = h[free] + diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    sw = np.sum(exps)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    j = np.arange(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-10:
            continue
        s = k - (n - 1)
        js = j - s
        jp = j + s
        ms = (js >= 0) & (js < n)
        mp = (jp >= 0) & (jp < n)
        if ms.any():
            grad[ms] += wk * (1.0 - h[js[ms]])
        if mp.any():
            grad[mp] += wk * (-h[jp[mp]])
    grad *= dx
    return val, grad

def nesterov_opt(h, max_iters, lr0, decay, top_k, time_limit):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    x = project_feasible(h.copy(), target)
    x_prev = x.copy()
    best_x = x.copy()
    best_v = compute_c5(x)
    j = np.arange(n)
    t0 = time.time()
    for t in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        lr = lr0 * (decay ** t)
        if t == 0:
            y = x.copy()
        else:
            mu = (np.sqrt(1.0 + 4.0 * t * t) - 1.0) / 2.0
            y = x + (mu - 1.0) / (mu + 1.0) * (x - x_prev)
        y = project_feasible(y, target)
        corr = np.correlate(y, 1.0 - y, mode="full")
        k_use = min(top_k, len(corr))
        top_idx = np.argsort(corr)[-k_use:]
        lags = top_idx - (n - 1)
        grad = np.zeros(n)
        for s in lags:
            js = j - s
            jp = j + s
            ms = (js >= 0) & (js < n)
            mp = (jp >= 0) & (jp < n)
            if ms.any():
                grad[ms] += (1.0 - y[js[ms]])
            if mp.any():
                grad[mp] += (-y[jp[mp]])
        grad = grad / k_use * dx
        # Backtracking
        step = lr
        for _ in range(8):
            x_new = project_feasible(y - step * grad, target)
            v_new = compute_c5(x_new)
            norm_g = np.linalg.norm(grad)
            if norm_g < 1e-15 or v_new < best_v - 1e-14 * step * norm_g:
                break
            step *= 0.5
        x_new = project_feasible(y - step * grad, target)
        v_new = compute_c5(x_new)
        if v_new < best_v:
            best_v = v_new
            best_x = x_new.copy()
        x_prev = x
        x = x_new
    return best_x, best_v

def slsqp_refine(h, maxiter, beta):
    n = len(h)
    target = n / 2.0
    def obj(x):
        v, _ = smoothmax_val_grad(x, beta)
        return v
    def objg(x):
        _, g = smoothmax_val_grad(x, beta)
        return g
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=objg, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)

def slsqp_anneal(h, betas, iters_per_beta):
    """SLSQP with annealed beta: start low for exploration, high for exploitation."""
    curr = h.copy()
    for b in betas:
        curr = slsqp_refine(curr, maxiter=iters_per_beta, beta=b)
    return curr

def binary_local_search(h_init, max_iters, time_limit, seed=42):
    """Greedy pair-flip local search on binary vectors."""
    n = len(h_init)
    h = (h_init >= 0.5).astype(float)
    target = n // 2
    ones = np.where(h == 1)[0]
    excess = len(ones) - target
    rng = np.random.default_rng(seed)
    if excess > 0:
        for _ in range(excess):
            if len(ones) == 0: break
            idx = ones[rng.integers(len(ones))]
            h[idx] = 0.0
            ones = np.where(h == 1)[0]
    elif excess < 0:
        zeros = np.where(h == 0)[0]
        for _ in range(-excess):
            if len(zeros) == 0: break
            idx = zeros[rng.integers(len(zeros))]
            h[idx] = 1.0
            zeros = np.where(h == 0)[0]
    best_h = h.copy()
    best_v = compute_c5(h)
    t0 = time.time()
    improved = True
    while improved and time.time() - t0 < time_limit:
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_delta = 0.0
        best_pair = None
        # Sample random pairs if too many
        if len(ones) * len(zeros) > 50000:
            samples = min(50000, len(ones) * len(zeros))
            for _ in range(samples):
                i1 = ones[rng.integers(len(ones))]
                i0 = zeros[rng.integers(len(zeros))]
                h[i1] = 0.0; h[i0] = 1.0
                v = compute_c5(h)
                h[i1] = 1.0; h[i0] = 0.0
                delta = v - best_v
                if delta < best_delta:
                    best_delta = delta
                    best_pair = (i1, i0)
        else:
            for i1 in ones:
                for i0 in zeros:
                    h[i1] = 0.0; h[i0] = 1.0
                    v = compute_c5(h)
                    h[i1] = 1.0; h[i0] = 0.0
                    delta = v - best_v
                    if delta < best_delta:
                        best_delta = delta
                        best_pair = (i1, i0)
        if best_pair is not None and best_delta < -1e-15:
            i1, i0 = best_pair
            h[i1] = 0.0; h[i0] = 1.0
            best_v += best_delta
            best_h = h.copy()
            improved = True
    return best_h, best_v

def interp_to_n(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

def perturb_reopt(h, base_c5, sigma, time_limit, seed):
    """Perturb and re-optimize to escape local optima."""
    n = len(h)
    rng = np.random.default_rng(seed)
    h_pert = h + sigma * rng.standard_normal(n)
    h_pert = project_feasible(h_pert, n / 2.0)
    # Quick Nesterov
    h_r, c_r = nesterov_opt(h_pert, 2000, 0.03, 0.9998, top_k=8,
                            time_limit=min(time_limit * 0.5, 30))
    # Quick SLSQP
    if time_limit > 15:
        h_r = slsqp_refine(h_r, maxiter=200, beta=150)
        c_r = compute_c5(h_r)
    return h_r, c_r

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.full(200, 0.5)

    best_c5 = compute_c5(h_best)

    # Phase 1: Nesterov with backtracking on starting resolution
    if tl() > 80:
        h1, c1 = nesterov_opt(h_best, 6000, 0.04, 0.9998, top_k=8,
                              time_limit=tl() - 80)
        if c1 < best_c5:
            h_best, best_c5 = h1, c1

    # Phase 2: SLSQP with beta annealing
    if tl() > 60:
        betas = [50, 100, 200, 400]
        h2 = slsqp_anneal(h_best, betas, iters_per_beta=300)
        c2 = compute_c5(h2)
        if c2 < best_c5:
            h_best, best_c5 = h2, c2

    # Phase 3: Binary local search
    if tl() > 50:
        h_b, c_b = binary_local_search(h_best, max_iters=100,
                                       time_limit=tl() - 50, seed=seed)
        if c_b < best_c5:
            h_best, best_c5 = h_b, c_b

    # Phase 4: Multi-resolution exploration
    for n_try in [100, 140, 200, 300]:
        if tl() < 120:
            break
        h_interp = interp_to_n(h_best, n_try)
        if tl() > 80:
            h_n, c_n = nesterov_opt(h_interp, 4000, 0.025, 0.9998,
                                    top_k=10, time_limit=tl() - 80)
            if c_n < best_c5:
                h_best, best_c5 = h_n, c_n
                h_interp = h_best.copy()
        if tl() > 50:
            h_s = slsqp_anneal(h_interp, [80, 200, 400], iters_per_beta=250)
            c_s = compute_c5(h_s)
            if c_s < best_c5:
                h_best, best_c5 = h_s, c_s

    # Phase 5: Perturbation restarts to escape local optima
    if tl() > 80:
        for p in range(3):
            if tl() < 40:
                break
            sigma = 0.1 * (p + 1)
            h_p, c_p = perturb_reopt(h_best, best_c5, sigma,
                                     time_limit=tl() - 40, seed=seed + 100 + p)
            if c_p < best_c5:
                h_best, best_c5 = h_p, c_p

    # Phase 6: Binary local search at best resolution
    if tl() > 40:
        h_b2, c_b2 = binary_local_search(h_best, max_iters=100,
                                         time_limit=tl() - 40, seed=seed + 50)
        if c_b2 < best_c5:
            h_best, best_c5 = h_b2, c_b2

    # Phase 7: Final high-beta SLSQP polish
    if tl() > 30:
        h_f = slsqp_refine(h_best, maxiter=1200, beta=500)
        c_f = compute_c5(h_f)
        if c_f < best_c5:
            h_best, best_c5 = h_f, c_f

    return h_best, best_c5, len(h_best)
