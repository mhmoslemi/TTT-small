import numpy as np
from scipy.optimize import minimize, minimize_scalar, Bounds
import time

_TS = [35.0]
_BETA = [100.0]
_DX = [2.0 / 70.0]
_CD_H = None
_CD_J = 0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx


def smooth_max_and_grad(h, beta):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(c)
    sh = np.clip(beta * (c - m), -500, 500)
    e = np.exp(sh)
    tot = np.sum(e)
    w = e / tot
    sm = m + np.log(tot) / beta
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        s = k - n + 1
        lo1 = max(0, -s)
        hi1 = min(n - 1, n - 1 - s)
        if lo1 <= hi1:
            jr = np.arange(lo1, hi1 + 1)
            grad[jr] += wk * (1.0 - h[jr + s])
        lo2 = max(0, s)
        hi2 = min(n - 1, n - 1 + s)
        if lo2 <= hi2:
            jr = np.arange(lo2, hi2 + 1)
            grad[jr] -= wk * h[jr - s]
    return sm, grad


def project_h(h, ts):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = ts - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        nf = free.sum()
        if nf == 0:
            break
        h[free] += diff / nf
    return np.clip(h, 0.0, 1.0)


def _con_fun(x):
    return x.sum() - _TS[0]


def _con_jac(x):
    return np.ones(len(x))


def _obj_f(x):
    sm, _ = smooth_max_and_grad(x, _BETA[0])
    return sm * _DX[0]


def _obj_g(x):
    _, g = smooth_max_and_grad(x, _BETA[0])
    return g * _DX[0]


def _cd_obj(t):
    h = _CD_H.copy()
    h[_CD_J] = t
    sm, _ = smooth_max_and_grad(h, _BETA[0])
    return sm * _DX[0]


def _slsqp_optimize(h_start, n, ts, bounds, cons, maxiter, ftol):
    try:
        res = minimize(_obj_f, h_start, jac=_obj_g, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_new = project_h(res.x, ts)
        return h_new
    except Exception:
        return h_start

    rng = np.random.RandomState(seed)

        return budget_s - (time.time() - t0)
