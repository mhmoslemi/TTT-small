def upsample(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - t0)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 80
        h = np.full(n, 0.5)

    h = project_feasible(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, n)

    # Phase 1: SLSQP continuation with increasing alpha
    for alpha in [2000, 20000, 200000, 2000000, 10000000]:
        if remaining() < 15:
            break
        ht = slsqp_optimize(best_h, n, alpha, 5000, 1e-18)
        c = compute_c5(ht, n)
        if c < best_c5 - 1e-15:
            best_h, best_c5 = ht, c

    # Phase 2: SA with broad exploration
    if remaining() > 120:
        best_h, best_c5 = sa_run(best_h, n, t0, budget_s, rng,
                                  300000, 0.15, 1e-5, 0.6, 0.01, 50000, 12.0)

    # Phase 3: SLSQP polish
    for alpha in [200000, 2000000, 20000000]:
        if remaining() < 15:
            break
        ht = slsqp_optimize(best_h, n, alpha, 5000, 1e-18)
        c = compute_c5(ht, n)
        if c < best_c5 - 1e-15:
            best_h, best_c5 = ht, c

    # Phase 4: Pair refinement
    if remaining() > 40:
        best_h, best_c5 = pair_refine(best_h, n, t0, budget_s)

    # Phase 5: Second SA round (finer)
    if remaining() > 100:
        best_h, best_c5 = sa_run(best_h, n, t0, budget_s, rng,
                                  200000, 0.05, 1e-6, 0.2, 0.005, 100000, 10.0)

    # Phase 6: SLSQP + pair again
    for alpha in [2000000, 20000000]:
        if remaining() < 15:
            break
        ht = slsqp_optimize(best_h, n, alpha, 5000, 1e-18)
        c = compute_c5(ht, n)
        if c < best_c5 - 1e-15:
            best_h, best_c5 = ht, c
    if remaining() > 30:
        best_h, best_c5 = pair_refine(best_h, n, t0, budget_s)

    # Phase 7: Try upsampled resolution
    if remaining() > 150:
        n2 = n * 2
        h2 = upsample(best_h, n, n2)
        c2 = compute_c5(h2, n2)
        for alpha in [50000, 500000, 5000000]:
            if remaining() < 20:
                break
            ht2 = slsqp_optimize(h2, n2, alpha, 3000, 1e-17)
            c2 = compute_c5(ht2, n2)
            if c2 < best_c5 - 1e-15:
                best_h, best_c5, n = ht2, c2, n2
        if remaining() > 60:
            best_h, best_c5 = sa_run(best_h, n, t0, budget_s, rng,
                                      150000, 0.03, 1e-6, 0.15, 0.003, 50000, 10.0)
        for alpha in [5000000, 50000000]:
            if remaining() < 15:
                break
            ht = slsqp_optimize(best_h, n, alpha, 5000, 1e-18)
            c = compute_c5(ht, n)
            if c < best_c5 - 1e-15:
                best_h, best_c5 = ht, c

    # Phase 8: Random restarts
    while remaining() > 60:
        scale = rng.choice([0.005, 0.01, 0.02, 0.05])
        h_pert = best_h + rng.normal(0, scale, size=n)
        h_pert = project_feasible(h_pert, n)
        ht = slsqp_optimize(h_pert, n, 100000, 2000, 1e-16)
        c = compute_c5(ht, n)
        if c < best_c5 - 1e-15:
            best_h, best_c5 = ht, c

    # Final polish
    if remaining() > 15:
        ht = slsqp_optimize(best_h, n, 50000000, 5000, 1e-18)
        c = compute_c5(ht, n)
        if c < best_c5 - 1e-15:
            best_h, best_c5 = ht, c

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
