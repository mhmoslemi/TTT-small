from scipy.interpolate import interp1d

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-12:
            free = (h > 1e-8) & (h < 1.0 - 1e-8)
            if free.sum() > 0:
                h[free] = h[free] + diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
                # Final correction
                diff3 = target_sum - h.sum()
                if abs(diff3) > 1e-10:
                    h += diff3 / len(h)
                    h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = corr - m

    max_shifted = np.max(shifted)
    exps = np.exp(beta * (shifted - max_shifted))
    sw = np.sum(exps)
    w = exps / sw
    val = (m + max_shifted + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    j = np.arange(n)

    significant = w > 1e-8

        if s == 0:

def _nesterov_phase(h0, iters, lr0, decay, top_k):
    """Nesterov subgradient method"""
    n = len(h0)
    target = n / 2.0
    x = h0.copy().astype(float)
    x_prev = x.copy()
    best_x = x.copy()

            else:

                    grad[ms] += (1.0 - y[js[ms]])
                if mp.any():
                    grad[mp] += (-y[jp[mp]])

            best_x = x_new.copy()

        x = x_new
    return best_x, best_v

def _slsqp(h0, maxiter, beta):
    """SLSQP with smoothmax objective"""
    n = len(h0)
    target = n / 2.0
    dx = 2.0 / n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = np.max(corr)
        exps = np.exp(beta * (corr - m))
        return (m + np.log(np.sum(exps)) / beta) * dx
    
    def objg(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = np.max(corr)
        exps = np.exp(beta * (corr - m))
        sw = np.sum(exps)
        w = exps / sw
        grad = np.zeros(n)
        j = np.arange(n)
        sig = w > 1e-10
        for k in np.where(sig)[0]:
            wk = w[k]
            s = k - (n - 1)
            if s == 0:
                grad += wk * (1.0 - 2.0 * x)
            else:
                js = j - s
                jp = j + s
                ms = (js >= 0) & (js < n)
                mp = (jp >= 0) & (jp < n)
                if ms.any():
                    grad[ms] += wk * (1.0 - x[js[ms]])
                if mp.any():
                    grad[mp] += wk * (-x[jp[mp]])
        grad *= dx
        return grad
    
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h0, jac=objg, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return _project(res.x, target)
    except Exception:
        return _project(h0, target)

def _local_search(h0, n_trials=500):
    """Try random perturbations and keep improvements"""
    n = len(h0)
    target = n / 2.0
    best = h0.copy()
    best_v = _compute_c5(best)
    rng = np.random.RandomState(0)
    for _ in range(n_trials):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:

        candidate = best.copy()
        candidate[i] -= delta
        candidate[j] += delta
        candidate = _project(candidate, target)
        v = _compute_c5(candidate)
        if v < best_v:
            best = candidate
            best_v = v
    return best, best_v

Q_s(h) = sum_i h[i]h[i+s]. The matrix for this quadratic form is a (0,1) matrix with 1s at positions corresponding to the shift s. This matrix is symmetric and has eigenvalues in [-1, 1] (it's a partial permutation matrix). So Q_s is neither convex nor concave in general.

This means the problem is genuinely non-convex, and we're in the territory of bilevel or min-max optimization.

import numpy as np
from scipy.optimize import minimize, Bounds, LinearConstraint
import time


def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _project(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target - h.sum()
        if abs(diff2) > 1e-14:
            free = (h > 1e-10) & (h < 1.0 - 1e-10)
            if free.sum() > 0:
                h[free] += diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h


def _nesterov(h0, iters, lr0, decay, top_k):
    n = len(h0)
    target = n / 2.0
    x = h0.copy().astype(float)
    x_prev = x.copy()
    best_x = x.copy()
    best_v = _compute_c5(x)
    j = np.arange(n)
    for t in range(iters):
        lr = lr0 * (decay ** t)
        mu = (t - 1) / (t + 2) if t > 0 else 0.0
        y = _project(x + mu * (x - x_prev), target)
        corr = np.correlate(y, 1.0 - y, mode="full")
        top_idx = np.argsort(corr)[-top_k:]
        grad = np.zeros(n)
        for s_pos in top_idx:
            s = s_pos - (n - 1)
            if s == 0:
                grad += (1.0 - 2.0 * y)
            else:
                js = j - s
                jp = j + s
                ms = (js >= 0) & (js < n)
                mp = (jp >= 0) & (jp < n)
                if ms.any():
                    grad[ms] += (1.0 - y[js[ms]])
                if mp.any():
                    grad[mp] += (-y[jp[mp]])
        grad = grad / top_k
        x_new = _project(y - lr * grad, target)
        v = _compute_c5(x_new)
        if v < best_v:
            best_v = v
            best_x = x_new.copy()
        x_prev = x
        x = x_new
    return best_x, best_v


def _slsqp(h0, maxiter, beta):
    n = len(h0)
    target = n / 2.0
    dx = 2.0 / n
    j = np.arange(n)

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = np.max(corr)
        exps = np.exp(beta * (corr - m))
        return (m + np.log(np.sum(exps)) / beta) * dx

    def objg(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = np.max(corr)
        exps = np.exp(beta * (corr - m))
        sw = np.sum(exps)
        w = exps / sw
        grad = np.zeros(n)
        sig = w > 1e-10
        for k in np.where(sig)[0]:
            wk = w[k]
            s = k - (n - 1)
            if s == 0:
                grad += wk * (1.0 - 2.0 * x)
            else:
                js = j - s
                jp = j + s
                ms = (js >= 0) & (js < n)
                mp = (jp >= 0) & (jp < n)
                if ms.any():
                    grad[ms] += wk * (1.0 - x[js[ms]])
                if mp.any():
                    grad[mp] += wk * (-x[jp[mp]])
        grad *= dx
        return grad

    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [LinearConstraint(np.ones(n), target, target)]
    try:
        res = minimize(obj, h0, jac=objg, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return _project(res.x, target)
    except Exception:
        return _project(h0, target)

def _local_search(h0, n_trials, seed_val=0):
    n = len(h0)
    target = n / 2.0
    best = h0.copy()
    best_v = _compute_c5(best)
    rng = np.random.RandomState(seed_val)
    for _ in range(n_trials):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        delta = (best[i] - best[j]) * 0.5
        if abs(delta) < 1e-7:
            continue
        candidate = best.copy()
        candidate[i] -= delta
        candidate[j] += delta
        candidate = _project(candidate, target)
        v = _compute_c5(candidate)
        if v < best_v:
            best = candidate
            best_v = v
    return best, best_v


def _local_sweep(h0, seed_val=0):
    n = len(h0)
    target = n / 2.0
    best = h0.copy()
    best_v = _compute_c5(best)
    rng = np.random.RandomState(seed_val)
    for trial in range(3000):
        i = trial % n
        j = rng.randint(n)
        if i == j:
            continue
        for frac in [0.3, 0.5, 0.7]:
            delta = (best[i] - best[j]) * frac
            if abs(delta) < 1e-7:
                continue
            candidate = best.copy()
            candidate[i] -= delta
            candidate[j] += delta
            candidate = _project(candidate, target)
            v = _compute_c5(candidate)

                best = candidate
                best_v = v
    return best, best_v
