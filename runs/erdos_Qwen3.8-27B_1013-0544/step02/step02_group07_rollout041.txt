import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h_bisect(h, target_sum):
    """Exact projection onto {0<=h<=1, sum(h)=target_sum} using bisection."""
    y = h.copy()
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        projected = np.clip(y - mid, 0.0, 1.0)
        s = projected.sum()
        if abs(s - target_sum) < 1e-15:
            break
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(y - mid, 0.0, 1.0)

def get_subgradient(h, d):
    """Compute gradient of corr_d wrt h, where d is the physical lag."""
    n = len(h)
    grad = np.zeros(n)
    if d >= 0:
        for j in range(n - d):
            grad[j] += (1.0 - h[j + d])
        for j in range(d, n):
            grad[j] -= h[j - d]
    else:
        da = -d
        for j in range(da, n):
            grad[j] += (1.0 - h[j - da])
        for j in range(n - da):
            grad[j] -= h[j + da]
    return grad

def subgradient_optimize(h0, target_sum, budget_s, start_time, n_top=5):
    """Nesterov subgradient method with momentum."""
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    v = np.zeros(n)  # momentum
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    beta_momentum = 0.9
    alpha0 = 0.1  # initial step size
    t = 0
    
    while time.time() - start_time < budget_s:
        t += 1
        # Compute correlations
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5_val = np.max(corr) * dx
        
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
        
        # Get top lags
        top_idx = np.argsort(corr)[-n_top:]
        # Weights: softmax of top correlations
        top_corr = corr[top_idx]
        weights = np.exp(50 * (top_corr - top_corr.max())) 
        weights /= weights.sum()
        
        # Compute subgradient
        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            d = k - (n - 1)
            grad += w * get_subgradient(h, d)
        
        # Nesterov step
        v = beta_momentum * v - alpha0 * grad
        h_new = project_h_bisect(h + v, target_sum)
        
        # Check if we improved (simple acceptance: always accept for now)
        h = h_new
        
        # Decay step size
        alpha0 = max(0.001, alpha0 * 0.999)
    
    return best_h, best_c5
