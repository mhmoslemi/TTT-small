            idx1 = i + s_all
            valid1 = (idx1 >= 0) & (idx1 < n)
            coef[valid1] += 1.0 - h[idx1[valid1]]
            idx2 = i - s_all
            valid2 = (idx2 >= 0) & (idx2 < n)
            coef[valid2] -= h[idx2[valid2]]

            hi = 1.0
            # We're optimizing h[i] directly in [0,1]
            # corr_new[m] = corr[m] - h[i]*coef[m] + t*coef[m] = corr[m] + (t-h[i])*coef[m]
            # So let delta = t - h[i], t = h[i] + delta, delta in [-h[i], 1-h[i]]

            d_hi = 1.0 - h[i]
            
            if d_hi - d_lo < 1e-14:
                continue

                mid = (d_lo + d_hi) / 2.0
                vals = corr + mid * coef
                mv = np.max(vals)
                active = vals >= mv - 1e-12
                sg = coef[active].sum()
                if sg > 1e-12:
                    d_hi = mid
                elif sg < -1e-12:
                    d_lo = mid
                else:
                    break
            
            delta = (d_lo + d_hi) / 2.0
            if abs(delta) > 1e-14:
                h[i] = np.clip(h[i] + delta, 0.0, 1.0)
                corr += delta * coef
        
        # Re-project and recompute
        h = project_box_sum(h, ts)
        corr = np.correlate(h, 1.0 - h, mode="full")
        curr_c5 = np.max(corr) * dx
        if curr_c5 < best_c5:
            best_c5 = curr_c5
        elif best_c5 - curr_c5 < tol and prev_best - best_c5 < tol:
            break
    
    return h, best_c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -50, 50)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def project_box_sum(h, target):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() == 0:
            break
        h[mask] += d / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h
