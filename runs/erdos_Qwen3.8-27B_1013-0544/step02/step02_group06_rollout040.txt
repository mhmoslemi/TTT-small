import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        if mask.sum() == 0:
            break
        h[mask] += d / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def logsumexp_max(x, beta):
    m = np.max(x)
    e = np.exp(beta * (x - m))
    s = e.sum()
    return m + np.log(s) / beta, e / s

def obj_grad(h, beta, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    sm, softmax = logsumexp_max(corr, beta)
    grad = np.zeros(n)
    for k in range(-(n-1), n):
        w = softmax[k + n - 1]
        if w < 1e-10:
            continue
        if k >= 0:
            grad[:n-k] += w * (1.0 - h[k:])
            grad[k:] -= w * h[:n-k]
        else:
            kk = -k
            grad[kk:] += w * (1.0 - h[:n-kk])
            grad[:n-kk] -= w * h[kk:]
    obj = sm * dx + lam * (h.sum() - n / 2.0) ** 2
    grad = grad * dx + 2.0 * lam * (h.sum() - n / 2.0)
    return obj, grad

def run_lbgfb(h_init, n, beta, lam, maxiter, ftol):
    target = n / 2.0

    def fun(x):
        return obj_grad(x, beta, lam)[0]

        return obj_grad(x, beta, lam)[1]
    try:
        r = minimize(fun, h_init, jac=jac, method='L-BFGS-B',
                     bounds=bnds, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(r.x, target)
    except Exception:

    return float(np.max(corr) * dx)

def project_feasible(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        mask = (h > 1e-15) & (h < 1.0 - 1e-15)
        if mask.sum() == 0:
            break
        h[mask] += d / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def logsumexp_max(x, beta):
    m = np.max(x)
    e = np.exp(beta * (x - m))
    s = e.sum()
    return m + np.log(s) / beta, e / s

def obj_grad(h, beta, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    sm, softmax = logsumexp_max(corr, beta)
    grad = np.zeros(n)
    for k in range(-(n-1), n):
        w = softmax[k + n - 1]
        if w < 1e-10:
            continue
        if k >= 0:
            grad[k:] += w * (1.0 - h[:n-k])
            grad[:n-k] -= w * h[k:]
        else:
            kk = -k
            grad[:n-kk] += w * (1.0 - h[kk:])
            grad[kk:] -= w * h[:n-kk]
    obj = sm * dx + lam * (h.sum() - n / 2.0) ** 2
    grad = grad * dx + 2.0 * lam * (h.sum() - n / 2.0)
    return obj, grad

            h_new = run_lbgfb(h_interp, n_try, beta, lam, maxit)
            c = compute_c5(h_new)
            if c < best_c5:
                best_h, best_c5, n = h_new, c, n_try
                target = n / 2.0

    # Phase 4: Final high-precision polish
    for beta, lam, maxit in [(1000, 3000, 400), (2000, 5000, 350), (4000, 10000, 350)]:
        if tl() < 10:
            break
        h_new = run_lbgfb(best_h, n, beta, lam, maxit)
        c = compute_c5(h_new)
        if c < best_c5:
            best_h, best_c5 = h_new, c

    # Phase 5: Fine-grained local search
    if tl() > 8:
        step = 0.003
        for _ in range(500):
            if tl() < 3:
                break
            h = best_h + rng.randn(n) * step
            h = project_feasible(h, target)
            c = compute_c5(h)
            if c < best_c5:
                best_h, best_c5 = h, c
            else:
                step *= 0.995

    best_h = project_feasible(best_h, target)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
