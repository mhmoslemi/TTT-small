    m = np.max(corr)
    s = np.clip(beta * (corr - m), -500, 500)
    e = np.exp(s)
    idx = np.argsort(e)[::-1][:top_m]
    e_rest = e.sum() - e[idx].sum()
    e_full = e[idx].sum() + e_rest
    w = np.zeros(len(corr))

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smax_val(corr, beta):
    m = np.max(corr)
    s = np.clip(beta * (corr - m), -500, 500)
    return m + np.log(np.sum(np.exp(s)) / len(corr)) / beta

def smax_weights_top(corr, beta, top_m):
    m = np.max(corr)
    s = np.clip(beta * (corr - m), -500, 500)
    e = np.exp(s)
    idx = np.argsort(e)[::-1][:top_m]
    e_total = e.sum()
    w = np.zeros(len(corr))
    w[idx] = e[idx] / e_total
    return w, idx

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_smax(h, beta, top_m):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    w, idx = smax_weights_top(corr, beta, top_m)
    grad = np.zeros(n)
    for k_idx in idx:
        wk = w[k_idx]
        if wk < 1e-14:
            continue
        k = k_idx - (n - 1)
        if k >= 0:
            grad[:n-k] += wk * (1.0 - h[k:])
            grad[k:] -= wk * h[:n-k]
        else:
            a = -k
            grad[a:] += wk * (1.0 - h[:n-a])
            grad[:n-a] -= wk * h[a:]
    return grad

def nesterov_opt(h0, target, beta, max_iter, step0, top_m):
    h_curr = h0.copy()
    h_prev = h0.copy()
    best = compute_c5(h_curr)
    best_h = h_curr.copy()
    t = 1.0
    for it in range(max_iter):
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        alpha = (t - 1.0) / t_new
        t = t_new
        h_y = project_h(h_curr + alpha * (h_curr - h_prev), target)
        grad = grad_smax(h_y, beta, top_m)
        step = step0 / (1.0 + 0.02 * it)
        h_new = project_h(h_y - step * grad, target)
        corr_new = np.correlate(h_new, 1 - h_new, mode="full")
        obj_new = smax_val(corr_new, beta)
        corr_y = np.correlate(h_y, 1 - h_y, mode="full")
        obj_y = smax_val(corr_y, beta)
        if obj_new < obj_y:
            h_prev = h_curr
            h_curr = h_new
        else:
            step *= 0.3
            h_new = project_h(h_y - step * grad, target)
            corr_new2 = np.correlate(h_new, 1 - h_new, mode="full")
            if smax_val(corr_new2, beta) < obj_y:
                h_prev = h_curr
                h_curr = h_new
            else:
                h_prev = h_curr
        c5 = compute_c5(h_curr)
        if c5 < best:
            best = c5
            best_h = h_curr.copy()
    return best_h, best

def sb_obj(x, beta, dx, target, pen):
    return smax_val(np.correlate(x, 1 - x, mode="full"), beta) * dx + pen * (x.sum() - target) ** 2

def sb_grad(x, beta, dx, target, pen):
    g = grad_smax(x, beta, 30) * dx
    g += 2.0 * pen * (x.sum() - target)
    return g

def sa_optimize(h0, target, T0, T1, n_iter, rng):
