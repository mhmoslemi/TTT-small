    """One pass of coordinate descent."""
    dx = 2.0 / n
    for j in range(n):
        # Compute A_k and B_k for all k
        # f_k = sum of h[i+k]*(1-h[i]) for valid i
        # When h[j] changes, only terms where j=i or j=i+k are affected
        
        A = np.zeros(2*n-1)
        B = np.zeros(2*n-1)
        
        # For each shift k (from -(n-1) to n-1), determine if h[j] appears
        # and with what coefficient
        for shift in range(-(n-1), n):
            # h[j] appears as h[i+k] when i = j - shift, need 0 <= j-shift < n
            # h[j] appears as h[i] when i = j, need 0 <= j+shift < n
            coeff = 0.0
            const = 0.0
            # Sum all terms not involving h[j]
            for i in range(n):

                if 0 <= ik < n:

                        coeff += (1 - h[i])

This is the same as before: ΔO[k] = (new_val - old_val) * A_k

So O_new[k] = O[k] + (new_val - old_val) * A_k = A_k * new_val + (O[k] - old_val * A_k)

So B_k = O[k] - old_val * A_k, and we want to minimize max_k (A_k * new_val + B_k) = max_k (O[k] + (new_val - old_val) * A_k).

This is equivalent to minimizing max_k (O[k] + A_k * x) where x = new_val - old_val, subject to 0 ≤ old_val + x ≤ 1, i.e., -old_val ≤ x ≤ 1 - old_val.

OK so the coordinate descent is:
1. Compute A_k for the current j
2. Find x that minimizes max_k (O[k] + A_k * x) subject to -old_val ≤ x ≤ 1 - old_val
3. Update h[j] += x

def coord_step_vectorized(h, n, j):
    """Compute optimal value for h[j] given all other h fixed."""
    O = compute_overlaps(h)  # (2n-1,)
    
    # Compute A: coefficient of h[j] in each overlap
    shifts = np.arange(-(n-1), n)  # (2n-1,)
    # A[m] for shift = shifts[m]:
    # (1-h[j-shift]) if 0<=j-shift<n, else 0
    # -h[j+shift] if 0<=j+shift<n, else 0
    
    j_minus_shift = j - shifts  # (2n-1,)
    j_plus_shift = j + shifts   # (2n-1,)
    
    A = np.zeros(2 * n - 1)
    mask1 = (j_minus_shift >= 0) & (j_minus_shift < n)
    mask2 = (j_plus_shift >= 0) & (j_plus_shift < n)
    A[mask1] += (1.0 - h[j_minus_shift[mask1]])
    A[mask2] += (-h[j_plus_shift[mask2]])
    
    old_val = h[j]
    lo_x = -old_val
    hi_x = 1.0 - old_val
    
    # Find all intersection points
    # x_nm = (O[m] - O[n]) / (A[n] - A[m]) for A[n] != A[m]
    # Vectorized:
    # diff_A = A[None, :] - A[:, None]  # (M, M)
    # diff_O = O[None, :] - O[:, None]
    # candidates = diff_O / diff_A where |diff_A| > eps
    
    M = 2 * n - 1
    diff_A = A[None, :] - A[:, None]  # (M, M), diff_A[i,j] = A[j] - A[i]
    diff_O = O[None, :] - O[:, None]  # (M, M), diff_O[i,j] = O[j] - O[i]

    # x_ij = (O[j] - O[i]) / (A[j] - A[i])
    # This is the intersection of lines y = A[i]*x + O[i] and y = A[j]*x + O[j]
    
    valid = np.abs(diff_A) > 1e-14
    with np.errstate(divide='ignore', invalid='ignore'):
        x_candidates = np.where(valid, diff_O / diff_A, np.nan)
    
    # Filter to [lo_x, hi_x]
    x_candidates = x_candidates[~np.isnan(x_candidates)]
    x_candidates = x_candidates[(x_candidates >= lo_x) & (x_candidates <= hi_x)]
    
    # Also include boundaries
    x_candidates = np.concatenate([x_candidates, [lo_x, hi_x]])
    x_candidates = np.unique(x_candidates)
    
    # Evaluate max at each candidate
    # For each x, compute max_m (O[m] + A[m] * x)
    # Values shape: (num_candidates, M)
    values = O[None, :] + A[None, :] * x_candidates[:, None]
    max_vals = np.max(values, axis=1)
    
    best_idx = np.argmin(max_vals)
    return x_candidates[best_idx] + old_val

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

    target = n / 2.0
    lo, hi = -2.0, 2.0

        p = np.clip(h + mid, 0.0, 1.0)
        if np.sum(p) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def coord_descent_pass(h, n):
    """One full pass of exact coordinate descent. Returns updated h and improvement."""
    O = compute_overlaps(h)
    shifts = np.arange(-(n-1), n)
    total_improvement = 0.0
    M = 2 * n - 1
    
    for j in range(n):
        old_val = h[j]
        
        # Compute A
        j_ms = j - shifts  # j - shift for each m
        j_ps = j + shifts  # j + shift for each m
        A = np.zeros(M)
        mask1 = (j_ms >= 0) & (j_ms < n)
        mask2 = (j_ps >= 0) & (j_ps < n)
        A[mask1] += (1.0 - h[j_ms[mask1]])
        A[mask2] += (-h[j_ps[mask2]])
        
        lo_x = -old_val
        hi_x = 1.0 - old_val
        
        # Find intersection points
        diff_A = A[None, :] - A[:, None]
        diff_O = O[None, :] - O[:, None]
        valid = np.abs(diff_A) > 1e-14
        with np.errstate(divide='ignore', invalid='ignore'):
            x_cands = np.where(valid, diff_O / diff_A, np.nan)
        
        x_cands = x_cands[~np.isnan(x_cands)]
        x_cands = x_cands[(x_cands >= lo_x) & (x_cands <= hi_x)]
        x_cands = np.concatenate([x_cands, [lo_x, hi_x]])
        
        # Evaluate
        values = O[None, :] + A[None, :] * x_cands[:, None]
        max_vals = np.max(values, axis=1)

        new_val = x_cands[best_idx] + old_val

            h[j] = new_val
            # Update O
            O += A * (new_val - old_val)
            total_improvement += abs(new_val - old_val)

        c5 = compute_c5(h)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        p = np.clip(h + mid, 0.0, 1.0)
        if np.sum(p) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def coord_descent_pass(h, n):
    O = compute_overlaps(h)
    shifts = np.arange(-(n-1), n)
    M = 2 * n - 1
    total_change = 0.0
    
    for j in range(n):
        old_val = h[j]
        j_ms = j - shifts
        j_ps = j + shifts
        A = np.zeros(M)
        mask1 = (j_ms >= 0) & (j_ms < n)
        mask2 = (j_ps >= 0) & (j_ps < n)
        A[mask1] += (1.0 - h[j_ms[mask1]])
        A[mask2] += (-h[j_ps[mask2]])
        
        lo_x = -old_val
        hi_x = 1.0 - old_val
        
        if lo_x >= hi_x:
            continue
        
        # Find all intersection points
        diff_A = A[None, :] - A[:, None]
        diff_O = O[None, :] - O[:, None]
        valid = np.abs(diff_A) > 1e-14
        if not np.any(valid):
            continue
        with np.errstate(divide='ignore', invalid='ignore'):
            x_cands = np.where(valid, diff_O / diff_A, np.nan)
        
        x_cands = x_cands[~np.isnan(x_cands)]
        x_cands = x_cands[(x_cands >= lo_x) & (x_cands <= hi_x)]
        x_cands = np.concatenate([x_cands, [lo_x, hi_x]])
        
        values = O[None, :] + A[None, :] * x_cands[:, None]
        max_vals = np.max(values, axis=1)
        best_idx = np.argmin(max_vals)

            delta = new_val - old_val
            h[j] = new_val
            O += A * delta
            total_change += abs(delta)
    
    return h, total_change

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        p = np.clip(h + mid, 0.0, 1.0)
        if np.sum(p) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def coord_descent_pass(h, n):
    O = compute_overlaps(h)
    shifts = np.arange(-(n - 1), n)
    M = 2 * n - 1
    total_change = 0.0
    for j in range(n):
        old_val = h[j]
        j_ms = j - shifts
        j_ps = j + shifts
        A = np.zeros(M)
        m1 = (j_ms >= 0) & (j_ms < n)
        m2 = (j_ps >= 0) & (j_ps < n)
        A[m1] += (1.0 - h[j_ms[m1]])
        A[m2] += (-h[j_ps[m2]])
        lo_x, hi_x = 0.0, 1.0
        if old_val >= 1.0 - 1e-15 or old_val <= 1e-15:
            pass
        diff_A = A[None, :] - A[:, None]
        diff_O = O[None, :] - O[:, None]
        valid = np.abs(diff_A) > 1e-14
        with np.errstate(divide='ignore', invalid='ignore'):
            xc = np.where(valid, diff_O / diff_A, np.nan)
        xc = xc[~np.isnan(xc)]
        xc = xc[(xc >= lo_x) & (xc <= hi_x)]
        xc = np.concatenate([xc, [lo_x, hi_x, 0.5]])
        xc = np.unique(xc)
        vals = O[None, :] + A[None, :] * (xc[:, None] - old_val)
        vals[:, n - 1] = O[n - 1] + (xc - xc**2) - old_val * (1.0 - old_val)
        max_vals = np.max(vals, axis=1)
        best_idx = np.argmin(max_vals)
        new_val = xc[best_idx]
        if abs(new_val - old_val) > 1e-13:
            delta = new_val - old_val
            h[j] = new_val
            O += A * delta
            O[n - 1] = O[n - 1] - A[n - 1] * delta + (new_val - new_val**2) - old_val * (1.0 - old_val)
            total_change += abs(delta)
    return h, total_change

def optimize(h_init, n, time_limit=60.0, max_passes=500, tol=1e-13):
    t0 = time.time()
    h = project_feasible(h_init.astype(float).copy(), n)
    for _ in range(max_passes):
        if time.time() - t0 > time_limit:
            break
        h, imp = coord_descent_pass(h, n)
        h = project_feasible(h, n)
        if imp < tol:
            break
    return h, compute_c5(h)

def binary_round(h, n):
    h_bin = (h > 0.5).astype(float)
    s = np.sum(h_bin)
    target = n // 2
    if s > target:
        order = np.argsort(h_bin)[::-1]
        for idx in range(int(s - target)):
            h_bin[order[idx]] = 0.0
    elif s < target:
        order = np.argsort(h_bin)
        for idx in range(int(target - s)):
            h_bin[order[idx]] = 1.0
    for _ in range(200):
        improved = False
        O = compute_overlaps(h_bin)
        current_max = np.max(O)
        for j in range(n):
            new_val = 1.0 - h_bin[j]
            if new_val == h_bin[j]:
                continue
            shifts = np.arange(-(n - 1), n)
            j_ms = j - shifts
            j_ps = j + shifts
            A = np.zeros(2 * n - 1)
            m1 = (j_ms >= 0) & (j_ms < n)
            m2 = (j_ps >= 0) & (j_ps < n)
            A[m1] += (1.0 - h_bin[j_ms[m1]])
            A[m2] += (-h_bin[j_ps[m2]])
            new_O = O + A * (new_val - h_bin[j])
            new_max = np.max(new_O)
            if new_max < current_max - 1e-14:
                h_bin[j] = new_val
                O = new_O
                current_max = new_max
                improved = True
        if not improved:
            break
    return h_bin

def resample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.arange(n_old) * 2.0 / n_old
    x_new = np.arange(n_new) * 2.0 / n_new
    return np.interp(x_new, x_old, h)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and globals()['initial_h_values'] is not None:
        h_start = np.array(globals()['initial_h_values'], dtype=float)
    else:
        h_start = np.ones(45) * 0.5

    best_h = h_start.copy()
    best_c5 = compute_c5(best_h)
    n_init = len(h_start)

    # Phase 1: Optimize at native resolution
    tl1 = min(45.0, budget_s * 0.08)
    h_opt, c5_opt = optimize(h_start, n_init, time_limit=tl1)
    if c5_opt < best_c5:
        best_h, best_c5 = h_opt, c5_opt

    # Phase 2: Binary rounding attempt
    h_bin = binary_round(best_h, n_init)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_h, best_c5 = h_bin, c5_bin

    # Phase 3: Try higher resolutions
    n_candidates = sorted(set([n_init * 2, n_init * 3, 90, 120, 150, 180, 200, 240]))
    n_candidates = [x for x in n_candidates if x > n_init and x <= 500]
    for n_r in n_candidates:
        if time.time() - t0 > budget_s * 0.55:
            break
        tl = budget_s * 0.55 - (time.time() - t0)
        if tl < 10:
            break
        h_res = resample(best_h, n_r)
        h_opt, c5_opt = optimize(h_res, n_r, time_limit=min(tl * 0.8, 100.0))
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
        h_bin = binary_round(best_h, n_r)
        c5_bin = compute_c5(h_bin)
        if c5_bin < best_c5:
            best_h, best_c5 = h_bin, c5_bin

    # Phase 4: Perturbation restarts
    n_best = len(best_h)
    for trial in range(50):
        if time.time() - t0 > budget_s * 0.82:
            break
        tl = budget_s * 0.82 - (time.time() - t0)
        if tl < 8:
            break
        scale = 0.15 / (1.0 + trial * 0.3)
        h_pert = best_h + np.random.randn(n_best) * scale
        h_pert = project_feasible(np.clip(h_pert, 0, 1), n_best)
        h_opt, c5_opt = optimize(h_pert, n_best, time_limit=min(tl * 0.5, 25.0))
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 5: Final high-res push
    if time.time() - t0 < budget_s * 0.75:
        n_high = min(len(best_h) * 2, 500)
        if n_high > len(best_h):
            h_res = resample(best_h, n_high)
            tl = budget_s * 0.88 - (time.time() - t0)
            if tl > 20:
                h_opt, c5_opt = optimize(h_res, n_high, time_limit=tl)
                if c5_opt < best_c5:
                    best_h, best_c5 = h_opt, c5_opt

    # Phase 6: Final coord descent polish on best
    if time.time() - t0 < budget_s * 0.92:
        tl = budget_s * 0.95 - (time.time() - t0)
        if tl > 5:
            h_final, c5_final = optimize(best_h, len(best_h), time_limit=tl)
            if c5_final < best_c5:
                best_h, best_c5 = h_final, c5_final

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
