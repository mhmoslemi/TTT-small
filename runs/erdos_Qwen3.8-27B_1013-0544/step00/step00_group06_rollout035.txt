import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h):
    """Project h to satisfy 0<=h<=1 and sum(h)=n/2."""
    n = len(h)
    h = np.clip(h, 0, 1)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        # Need to increase sum: add to smallest values
        sorted_idx = np.argsort(h)
        for i in sorted_idx:
            room = 1.0 - h[i]
            add = min(room, diff)
            h[i] += add
            diff -= add
            if abs(diff) < 1e-12:
                break
    else:
        # Need to decrease sum: reduce from largest values
        sorted_idx = np.argsort(-h)
        for i in sorted_idx:
            room = h[i]
            rem = min(room, -diff)
            h[i] -= rem
            diff += rem
            if abs(diff) < 1e-12:
                break
    return h

def coordinate_descent(h, max_passes=100, tol=1e-10):
    """Coordinate descent to minimize C5."""
    n = len(h)
    h = h.copy()
    
    for p in range(max_passes):
        max_change = 0
        for j in range(n):
            h_old = h[j]
            corr = np.correlate(h, 1.0 - h, mode='full')
            
            # Compute coefficient of h[j] in each correlation value
            coeffs = np.zeros(2*n-1)
            for pos in range(2*n-1):

                c = 0.0
                if 0 <= j - q < n:

                if 0 <= j + q < n:
                    c -= h[j + q]
                coeffs[pos] = c

            def f(x):
                return np.max(base + x * coeffs)

            new_val = result.x
            h[j] = new_val
            max_change = max(max_change, abs(new_val - h_old))
        
        # Project to feasible set
        h = project_feasible(h)
        
        if max_change < tol:
            break

    n = len(h)

        return compute_c5(h)

                      constraints=constraints, options={'maxiter': maxiter, 'ftol': 1e-10})
        if res.success:

        pass

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h):
    n = len(h)
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            order = np.argsort(h)
            for i in order:
                add = min(1.0 - h[i], diff)
                h[i] += add
                diff -= add
                if diff <= 1e-14:
                    break
        else:
            order = np.argsort(-h)
            for i in order:
                rem = min(h[i], -diff)
                h[i] -= rem
                diff += rem
                if diff >= -1e-14:
                    break
    return h

def coordinate_descent(h, n_passes=100, tol=1e-10):
    h = h.copy()
    for p in range(n_passes):
        h_new = cd_one_pass(h)
        h_new = project_feasible(h_new)
        delta = np.max(np.abs(h_new - h))
        h = h_new
        if delta < tol:
            break
    return h

def slsqp_optimize(h, maxiter=500):
    n = len(h)
    def objective(x):
        return compute_c5(x)
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(objective, h, method='SLSQP', bounds=bounds,
                      constraints=constraints, options={'maxiter': maxiter, 'ftol': 1e-10})
        if res.fun < compute_c5(h):
            return np.clip(res.x, 0, 1)
    except:
        pass
    return h

    n = len(h)
    h = h.copy()
    for j in range(n):
        h_old = h[j]
        corr = np.correlate(h, 1.0 - h, mode='full')
        
        coeffs = np.zeros(2*n-1)
        lo1, hi1 = j, min(j+n, 2*n-1)

            coeffs[lo1:hi1] += (1.0 - h[idxs1])
