    """Compute gradient of the worst-case overlap."""
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")

    shift = k_star - (n - 1)

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_onto_bounded_sum(v, target_sum):
    """Project v onto {0 ≤ x ≤ 1, sum(x) = target_sum}."""
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2
        projected = np.clip(v - mid, 0, 1)
        s = np.sum(projected)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    mid = (lo + hi) / 2
    return np.clip(v - mid, 0, 1)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        n = 45
        h0 = np.zeros(n)
        h0[:22] = 1.0
        h0[22] = 0.5

    target_sum = n / 2.0
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    
    def elapsed():
        return time.time() - start_time

    h = h0.copy()
    lr = 0.05
    momentum = 0.9
    velocity = np.zeros(n)
    top_m = 5
    stuck_count = 0
    
    it = 0
    while elapsed() < budget_s * 0.85:
        it += 1
        corr = np.correlate(h, 1 - h, mode="full")
        # Get top m shifts
        top_indices = np.argsort(corr)[-top_m:]
        
        # Average gradient over top m shifts
        grad = np.zeros(n)
        for k_idx in top_indices:
            s = k_idx - (n - 1)
            # First term: (1 - h[j-s]) for valid j
            j_start = max(0, s)
            j_end = min(n, n + s)
            if j_start < j_end:
                grad[j_start:j_end] += (1 - h[j_start - s:j_end - s])
            # Second term: -h[j+s] for valid j
            j_start2 = max(0, -s)
            j_end2 = min(n, n - s)
            if j_start2 < j_end2:
                grad[j_start2:j_end2] += -h[j_start2 + s:j_end2 + s]

        h_new = h + velocity
        h_new = project_onto_bounded_sum(h_new, target_sum)

                velocity = np.zeros(n)
                lr = 0.05
                stuck_count = 0

            def obj(x):
                return compute_c5(x)

                           constraints=constraints, options={'maxiter': 100, 'ftol': 1e-10})

    js = max(0, s)
    je = min(n, n + s)
