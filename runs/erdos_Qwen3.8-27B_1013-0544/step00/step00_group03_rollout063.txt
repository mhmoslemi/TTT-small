def compute_c5_full(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx, corr

def project_box_sum(x, target_sum):

    # Bisection on lambda for projection onto {0<=x<=1, sum(x)=target_sum}
    lo, hi = -2.0, 2.0

        projected = np.clip(x - mid, 0.0, 1.0)
        s = np.sum(projected)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def subgradient(h, dx, tau=0.05):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    shifted = corr - max_corr
    exp_vals = np.exp(shifted / tau)
    weights = exp_vals / np.sum(exp_vals)
    
    grad = np.zeros(n)
    j_arr = np.arange(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-8:
            continue
        lag = k - (n - 1)
        g = np.zeros(n)
        if lag >= 0:
            mask1 = j_arr >= lag
            g[mask1] += 1.0 - h[j_arr[mask1] - lag]
            mask2 = j_arr <= n - 1 - lag
            g[mask2] += -h[j_arr[mask2] + lag]
        else:
            L = -lag
            mask1 = j_arr <= n - 1 - L
            g[mask1] += 1.0 - h[j_arr[mask1] + L]
            mask2 = j_arr >= L
            g[mask2] += -h[j_arr[mask2] - L]
        grad += w * g
    return grad * dx

    h = h0.copy()
    h = project_box_sum(h, target_sum)
    best_h = h.copy()
    best_c5, _ = compute_c5_full(h, dx)
    
    start = time.time()
    step = 0.1
    tau = 0.05

def compute_c5(h, dx):

def project_box_sum(x, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def subgrad(h, dx, tau):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    expv = np.exp((corr - mc) / tau)
    wts = expv / np.sum(expv)
    grad = np.zeros(n)
    ja = np.arange(n)
    for k in range(len(corr)):
        wt = wts[k]
        if wt < 1e-8:
            continue
        lag = k - (n - 1)
        g = np.zeros(n)
        if lag >= 0:
            m1 = ja >= lag
            g[m1] += 1.0 - h[ja[m1] - lag]
            m2 = ja <= n - 1 - lag
            g[m2] -= h[ja[m2] + lag]
        else:
            L = -lag
            m1 = ja <= n - 1 - L
            g[m1] += 1.0 - h[ja[m1] + L]
            m2 = ja >= L
            g[m2] -= h[ja[m2] - L]
        grad += wt * g
    return grad * dx

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_box_sum(x, tsum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        p = np.clip(x - mid, 0.0, 1.0)
        if p.sum() > tsum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def subgrad(h, dx, tau):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = corr.max()
    ev = np.exp((corr - mc) / tau)
    wts = ev / ev.sum()
    grad = np.zeros(n)
    ja = np.arange(n)
    for k in range(len(corr)):
        wt = wts[k]
        if wt < 1e-8:
            continue
        lag = k - n + 1
        g = np.zeros(n)
        if lag >= 0:
            m1 = ja >= lag
            g[m1] += 1.0 - h[ja[m1] - lag]
            m2 = ja <= n - 1 - lag
            g[m2] -= h[ja[m2] + lag]
        else:
            L = -lag
            m1 = ja <= n - 1 - L
            g[m1] += 1.0 - h[ja[m1] + L]
            m2 = ja >= L
            g[m2] -= h[ja[m2] - L]
        grad += wt * g
    return grad * dx

def grad_desc(h0, tsum, dx, tl, seed):
    n = len(h0)
    rng = np.random.RandomState(seed)
    h = project_box_sum(h0, tsum)
    bh = h.copy()
    bc5 = compute_c5(h, dx)
    t_start = time.time()
    step, tau, stall = 0.1, 0.05, 0
    while time.time() - t_start < tl:
        gr = subgrad(h, dx, tau)
        hn = project_box_sum(h - step * gr, tsum)
        c5n = compute_c5(hn, dx)
        if c5n < bc5 - 1e-12:
            h, bh, bc5 = hn, hn.copy(), c5n
            stall = 0
            step = min(step * 1.06, 0.4)
        else:
            step *= 0.7
            stall += 1
            if stall > 60:
                tau = max(tau * 0.5, 0.001)
                step = 0.05
                h = project_box_sum(h + rng.randn(n) * 0.06, tsum)
                stall = 0
                if tau <= 0.002:
                    break
        if step < 1e-7:
            break
    return bh, bc5

        npairs = min(len(ones) * len(zeros), 4000)
        imp = False
        for _ in range(npairs):

            oi = ones[rng.randint(len(ones))]
            oz = zeros[rng.randint(len(zeros))]
            if hb[oi] != 1.0 or hb[oz] != 0.0:

            hb[oi], hb[oz] = 0.0, 1.0
            c5 = compute_c5(hb, dx)
            if c5 < bc5 - 1e-12:
                best, bc5 = hb.copy(), c5
                imp = True
            else:
                hb[oi], hb[oz] = 1.0, 0.0
        if not imp:
            break
    return best, bc5
