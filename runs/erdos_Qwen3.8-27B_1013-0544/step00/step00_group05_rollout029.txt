    dx = 2.0 / n
    terms = np.correlate(h, 1-h, mode="full")

    active = np.where(terms >= max_val - 1e-10)[0]
    
    # Compute subgradient
    grad = np.zeros(n)
    for k in active:

                h[i] = min(h[i] + abs(per), 1.0)
            else:
                h[i] = max(h[i] + per, 0.0)
    return h

def local_search(h, max_rounds=1000, time_limit=None, rng=None):
    n = len(h)
    best_val = compute_c5(h)
    fracs = [0.25, 0.5, 0.75, 1.0]
    for _ in range(max_rounds):
        if time_limit is not None and time.time() > time_limit:
            break
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                hi = h[i]
                hj = h[j]
                best_move = None
                best_mv = best_val
                for d in (1, -1):
                    if d > 0:
                        md = min(1.0 - hi, hj)
                    else:
                        md = min(hi, 1.0 - hj)
                    if md < 1e-12:
                        continue
                    for f in fracs:
                        delta = f * md
                        if delta < 1e-12:
                            continue
                        h[i] = hi + d * delta
                        h[j] = hj - d * delta
                        v = compute_c5(h)
                        if v < best_mv:
                            best_mv = v
                            best_move = (d, delta)
                        h[i] = hi
                        h[j] = hj
                if best_move is not None:
                    d, delta = best_move
                    h[i] = hi + d * delta
                    h[j] = hj - d * delta
                    best_val = best_mv
                    improved = True
        if not improved:
            break
    return h, best_val

def binary_local_search(h, max_rounds=5000, time_limit=None, rng=None):
    """Local search that only considers binary (0/1) solutions."""
    n = len(h)
    # First, project to binary (round to nearest 0/1, adjust count)
    h_bin = (h > 0.5).astype(float)
    target = n // 2
    ones = int(np.sum(h_bin))
    if ones > target:
        # Remove excess ones (remove the ones that are "least committed")
        order = np.argsort(h)[::-1]  # most committed first
        for idx in order:
            if h_bin[idx] == 1 and ones > target:
                h_bin[idx] = 0
                ones -= 1
    elif ones < target:
        order = np.argsort(h)  # least committed to 0 first
        for idx in order:
            if h_bin[idx] == 0 and ones < target:
                h_bin[idx] = 1
                ones += 1
    
    best_val = compute_c5(h_bin)
    
    for _ in range(max_rounds):
        if time_limit is not None and time.time() > time_limit:
            break
        improved = False
        ones_idx = np.where(h_bin == 1)[0]
        zeros_idx = np.where(h_bin == 0)[0]
        for ii in range(len(ones_idx)):
            for jj in range(len(zeros_idx)):
                i = ones_idx[ii]
                j = zeros_idx[jj]
                h_bin[i] = 0
                h_bin[j] = 1
                v = compute_c5(h_bin)
                if v < best_val:
                    best_val = v
                    improved = True
                else:
                    h_bin[i] = 1
                    h_bin[j] = 0
        if not improved:
            break
    return h_bin, best_val

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def repair_h(h, target_sum, rng):
