import numpy as np
import time
from scipy.optimize import minimize

def project_feasible(p, S):
    """Project p onto {h: 0<=h[i]<=1, sum(h)=S} using bisection on water level."""
    n = len(p)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(p - mid, 0.0, 1.0))
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.clip(p - (lo + hi) / 2.0, 0.0, 1.0)

def c5_value(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def grad_at_lag(h, m):
    """Gradient of c_{n-1+m} = sum_i h[i]*(1-h[i+m]) w.r.t. h."""
    n = len(h)
    grad = np.zeros(n)
    if m >= 0:
        if m < n:
            grad[:n - m] += (1.0 - h[m:])
            grad[m:] -= h[:n - m]
    else:
        p = -m
        if p < n:
            grad[p:] += (1.0 - h[:n - p])
            grad[:n - p] -= h[p:]
    return grad

def smooth_obj_grad(h, dx, beta):
    """Compute smooth objective and gradient using log-sum-exp trick."""
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(c)
    exp_c = np.exp(beta * (c - c_max))
    Z = np.sum(exp_c)
    obj = (c_max + np.log(Z) / beta) * dx
    weights = exp_c / Z
    grad = np.zeros(n)
    sig = weights > 1e-4
    for idx in np.where(sig)[0]:
        m = idx - (n - 1)
        grad += weights[idx] * grad_at_lag(h, m)
    return obj, grad * dx

def projected_gradient_opt(h0, S, dx, time_limit, seed=42):
    """Projected gradient with adaptive beta and step size."""
    n = len(h0)
    rng = np.random.RandomState(seed)
    h = h0.copy()
    best_h = h.copy()
    best_val = c5_value(h, dx)
    lr = 0.1
    beta = 10.0
    no_improve = 0
    iter_count = 0

    while time.time() < time_limit:
        iter_count += 1
        # Anneal beta (increase sharpness of max approximation)
        beta = min(10.0 * (1.005 ** iter_count), 200.0)

        obj, g = smooth_obj_grad(h, dx, beta)
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-10:
            step = lr * g / g_norm * min(1.0, lr * g_norm)
        else:
            step = 0.0

        h_new = h - step
        h_new = project_feasible(h_new, S)
        val = c5_value(h_new, dx)

        if val < best_val - 1e-7:
            best_val = val
            best_h = h_new.copy()
            h = h_new
            lr = min(lr * 1.01, 1.0)
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                # Restart with perturbation
                h = best_h + rng.normal(0, 0.03, n)
                h = project_feasible(h, S)
                best_val = c5_value(best_h, dx)
                no_improve = 0
                lr = 0.05
            elif no_improve > 50:
                lr = max(lr * 0.95, 0.005)
            h = best_h.copy()

    # Binarization attempt
    h_bin = (best_h > 0.5).astype(float)
    h_bin = project_feasible(h_bin, S)
    val_bin = c5_value(h_bin, dx)
    if val_bin < best_val:
        best_val = val_bin
        best_h = h_bin.copy()
    # Also try threshold at 0.4
    h_bin2 = (best_h > 0.4).astype(float)
    h_bin2 = project_feasible(h_bin2, S)
    val_bin2 = c5_value(h_bin2, dx)
    if val_bin2 < best_val:
        best_val = val_bin2
        best_h = h_bin2.copy()

    return best_h, best_val

def make_initial(n, h_ref=None, seed=42):
    """Create initial h for given n, optionally interpolating from h_ref."""
    if h_ref is not None and len(h_ref) == n:
        return h_ref.copy()
    if h_ref is not None:
        n_ref = len(h_ref)
        x_old = np.arange(n_ref) / (2.0 * n_ref)
        x_new = np.arange(n) / (2.0 * n)
        return np.interp(x_new, x_old, h_ref, period=2.0)
    # Random initialization with good structure
    rng = np.random.RandomState(seed)
    h = rng.rand(n)
    h = (h < 0.5).astype(float)
    return project_feasible(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()

    # Check for initial_h_values
    h0 = None
