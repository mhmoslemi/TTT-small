    return m + t * np.log(np.sum(np.exp((x - m) / t)))

    if abs(d) < 1e-14:
        return h
    o = np.argsort(1.0 - h) if d > 0 else np.argsort(-h)
    rem = abs(d)
    s = 1.0 if d > 0 else -1.0

            break
        a = min((1.0 - h[i]) if s > 0 else h[i], rem)

    return val + pen * sv * sv

def project(h, n):
    target = n * 0.5
    h = np.clip(h.copy(), 0.0, 1.0)
    d = target - np.sum(h)
    if abs(d) < 1e-14:
        return h
    o = np.argsort(1.0 - h) if d > 0 else np.argsort(-h)
    rem = abs(d)
    s = 1.0 if d > 0 else -1.0
    for i in o:
        if rem <= 1e-14:
            break
        a = min((1.0 - h[i]) if s > 0 else h[i], rem)
        h[i] += s * a
        rem -= a
    return h

def coord_descent(h, n, max_rounds=80):
    h = project(h, n)
    best = compute_c5(h, n)
    for _ in range(max_rounds):
        improved = False
        for i in range(n):
            for delta in (-0.1, -0.05, -0.02, -0.005, 0.005, 0.02, 0.05, 0.1):
                h2 = h.copy()
                h2[i] += delta
                h2 = project(h2, n)
                c5 = compute_c5(h2, n)
                if c5 < best - 1e-12:
                    best = c5
                    h = h2
                    improved = True
                    break
        if not improved:
            break
    return h, best

def anneal_optimize(h0, n, time_budget, rng):
    t_start = time.time()
    bounds = [(0.0, 1.0)] * n
    h = project(h0, n)
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    temps = [0.3, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001]
    for t in temps:
        if time.time() - t_start > time_budget - 15:
            break
        pen = max(1000.0, 20.0 / t)
        try:
            res = minimize(obj, h, args=(n, t, pen), method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': 600, 'ftol': 1e-15, 'gtol': 1e-12, 'maxls': 80})
            h_new = project(res.x, n)
            c5_new = compute_c5(h_new, n)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)

    def rem():
        return max(0.0, budget_s - (time.time() - t0) - 5.0)

    n = len(initial_h_values)
    h0 = project(np.array(initial_h_values, dtype=float), n)

    # Phase 1: Optimize initial construction
    h_best, c5_best = anneal_optimize(h0, n, rem() * 0.35, rng)

    # Phase 2: Random restarts at same n
    n_restarts = 25
    for i in range(n_restarts):
        if rem() < 25:
            break
        noise = 0.03 + 0.12 * rng.random()
        h_start = project(h_best + rng.normal(0, noise, n), n)
        alloc = min(rem() * 0.5 / max(n_restarts - i, 1), 70.0)
        h_try, c5_try = anneal_optimize(h_start, n, alloc, rng)
        if c5_try < c5_best - 1e-12:
            c5_best = c5_try
            h_best = h_try

    # Phase 3: Coordinate descent refinement
    if rem() > 20:
        h_cd, c5_cd = coord_descent(h_best, n, max_rounds=100)
        if c5_cd < c5_best - 1e-12:
            c5_best = c5_cd
            h_best = h_cd

    # Phase 4: Try higher resolution
    try_n_list = [2 * n, 4 * n, 8 * n, 16 * n]
    for n_hi in try_n_list:
        if n_hi >= 1000 or rem() < 80:
            break
        rep = n_hi // len(h_best)
        if rep < 1:
            continue
        h_hi = np.tile(h_best, rep)[:n_hi]
        if len(h_hi) < n_hi:
            h_hi = np.concatenate([h_hi, np.full(n_hi - len(h_hi), 0.5)])
        h_hi = project(h_hi, n_hi)
        alloc = min(rem() - 30, 250.0)
        h_try, c5_try = anneal_optimize(h_hi, n_hi, alloc, rng)
        if rem() > 20:
            h_try, c5_try2 = coord_descent(h_try, n_hi, max_rounds=50)
            if c5_try2 < c5_try:
                c5_try = c5_try2
                h_try = h_try
        if c5_try < c5_best - 1e-12:
            c5_best = c5_try
            h_best = h_try

    # Phase 5: Final fine-tuning
    if rem() > 40:
        h_fin, c5_fin = anneal_optimize(h_best, len(h_best), min(rem() - 10, 120), rng)
        if c5_fin < c5_best - 1e-12:
            c5_best = c5_fin
            h_best = h_fin
        if rem() > 15:
            h_fin2, c5_fin2 = coord_descent(h_best, len(h_best), max_rounds=60)
            if c5_fin2 < c5_best - 1e-12:
                c5_best = c5_fin2
                h_best = h_fin2

    n_final = len(h_best)
    c5_final = compute_c5(h_best, n_final)
    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = project(h_best, n_final)
    c5_final = compute_c5(h_best, n_final)
    return (h_best, c5_final, n_final)
