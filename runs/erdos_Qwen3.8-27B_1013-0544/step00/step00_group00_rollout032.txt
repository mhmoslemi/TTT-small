from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    return np.max(corr) * dx

def water_filling(h, target_sum):
    """Project h onto {x: sum(x)=target_sum, 0<=x<=1}"""
    lo, hi = -1.0, 2.0
    for _ in range(200):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    mid = (lo + hi) / 2
    return np.clip(h - mid, 0, 1)

def obj_and_grad(h, eps_smooth=0.01):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    
    # Smooth max via log-sum-exp
    scaled = corr / eps_smooth
    max_s = np.max(scaled)
    log_sum_exp = max_s + np.log(np.sum(np.exp(scaled - max_s)))
    obj = eps_smooth * log_sum_exp * dx
    
    # Softmax weights
    weights = np.exp(scaled - max_s)
    weights /= np.sum(weights)
    
    # Gradient
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-10:
            continue
        l = k - (n - 1)
        # Part 1: j where 0 <= j+l < n
        j_start = max(0, -l)
        j_end = min(n, n - l)
        if j_start < j_end:
            grad[j_start:j_end] += w * (1 - h[j_start + l:j_end + l])
        # Part 2: j where 0 <= j-l < n
        j_start2 = max(0, l)
        j_end2 = min(n, n + l)
        if j_start2 < j_end2:
            grad[j_start2:j_end2] += w * (-h[j_start2 - l:j_end2 - l])
    
    grad *= dx
    return obj, grad

            o, _ = obj_and_grad(x, eps)
            return o

            _, g = obj_and_grad(x, eps)
            return g
