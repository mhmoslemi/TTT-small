        step *= 0.9999
        if step < 0.005:
            step = 0.05

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s * 0.95

    best_h = None
    best_c5 = float('inf')

    def update(h, c5):
        nonlocal best_h, best_c5
        if h is not None and c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

    # Try initial_h_values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
        hb = make_binary(h0, n0 / 2.0)
        t_lim = min(deadline, start + budget_s * 0.2)
        hb, c5b = local_search(hb, t_lim, rng)
        update(hb, c5b)

    # Thue-Morse starts for various n
    tm_ns = [95, 91, 93, 97, 99, 127, 129, 131]
    for tn in tm_ns:
        if time.time() > deadline:
            break
        htm = thue_morse(tn)
        htm = make_binary(htm, tn / 2.0)
        remaining = deadline - time.time()
        t_lim = min(deadline, time.time() + max(5, remaining / 6))
        htm, c5tm = local_search(htm, t_lim, rng)
        update(htm, c5tm)

    # Random starts for various n
    rand_ns = [94, 96, 98, 100, 102, 126, 128, 130, 198, 200, 250, 252]
    for tn in rand_ns:
        if time.time() > deadline:
            break
        hr = rng.random(tn)
        hr = make_binary(hr, tn / 2.0)
        remaining = deadline - time.time()
        t_lim = min(deadline, time.time() + max(5, remaining / 5))
        hr, c5r = local_search(hr, t_lim, rng)
        update(hr, c5r)

    # Continuous refinement on best
    if best_h is not None and time.time() < deadline:
        hc = best_h.copy()
        t_lim = min(deadline, time.time() + 30)
        hc, c5c = local_search_continuous(hc, t_lim, rng)
        update(hc, c5c)

    if best_h is None:
        # Fallback
        n_fb = 95
        h_fb = make_binary(rng.random(n_fb), n_fb / 2.0)
        best_c5 = compute_c5(h_fb)
        best_h = h_fb

    return best_h, best_c5, len(best_h)
