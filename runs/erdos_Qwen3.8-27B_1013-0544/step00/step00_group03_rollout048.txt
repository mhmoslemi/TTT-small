def compute_gradient(h):

    n = len(h)
    one_minus_h = 1.0 - h

def compute_gradient_soft(h, top_k=10):
    """Gradient using top-k active lags."""
    n = len(h)
    corr = compute_corr(h)
    # Get top k indices
    top_idx = np.argsort(corr)[-top_k:]
    
    grad = np.zeros(n)
    for idx in top_idx:
        k = idx - (n - 1)
        j_lo = max(0, -k)

        j_hi2 = min(n, n + k)
        if j_lo2 < j_hi2:

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_gradient_soft(h, top_k=8):
    n = len(h)
    corr = compute_corr(h)
    top_k = min(top_k, 2 * n - 1)
    top_idx = np.argsort(corr)[-top_k:]
    grad = np.zeros(n)
    for idx in top_idx:
        k = idx - (n - 1)
        j_lo = max(0, -k)
        j_hi = min(n, n - k)
        if j_lo < j_hi:
            grad[j_lo:j_hi] += (1.0 - h[j_lo + k:j_hi + k])
        j_lo2 = max(0, k)
        j_hi2 = min(n, n + k)
        if j_lo2 < j_hi2:
            grad[j_lo2:j_hi2] -= h[j_lo2 - k:j_hi2 - k]
    grad /= float(len(top_idx))
    return grad

def project(h, target_sum):
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def try_binary_snap(h):
    n = len(h)
    target = n // 2
    b = (h > 0.5).astype(float)
    n_ones = int(np.sum(b))
    if n_ones > target:
        ones_idx = np.where(b == 1.0)[0]
        order = np.argsort(h[ones_idx])
        for idx in order[:n_ones - target]:
            b[idx] = 0.0
    elif n_ones < target:
        zeros_idx = np.where(b == 0.0)[0]
        order = np.argsort(-h[zeros_idx])
        for idx in order[:target - n_ones]:
            b[idx] = 1.0
    return b

def optimize_from(h_init, target_sum, time_limit, seed=0):

    h = project(h_init.copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    step_size = 0.04
    min_step = 1e-9
    no_improve = 0
    max_no_improve = 300
    top_k = 5

    for it in range(200000):
        if time.time() - start > time_limit:
            break
        if step_size < min_step:
            break
        ak = min(25, top_k + it // 400)
        grad = compute_gradient_soft(h, top_k=ak)
        h_new = h - step_size * grad
        h_new = project(h_new, target_sum)
        c5_new = compute_c5(h_new)
        h = h_new
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

            no_improve = 0
        if it % 200 == 0:
            step_size = max(min_step, step_size * 0.998)
    return best_h, best_c5
