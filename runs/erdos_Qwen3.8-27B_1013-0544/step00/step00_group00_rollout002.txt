    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max_c5(h, dx, t):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_val = np.max(corr)
    shifted = t * (corr - max_val)
    return max_val + (1.0 / t) * np.log(np.sum(np.exp(shifted)))
