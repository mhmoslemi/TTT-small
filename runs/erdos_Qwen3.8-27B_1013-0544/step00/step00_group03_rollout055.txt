def compute_subgradient(h, n_points):
    """Compute subgradient of C5 at h."""
    dx = 2.0 / n_points
    corr = np.correlate(h, 1-h, mode="full")
    k = n_points - 1  # offset
    max_val = np.max(corr)
    
    # Find all lags achieving the max (within tolerance)
    threshold = max_val - 1e-10
    max_lags = np.where(corr >= threshold)[0]
    
    # Compute gradient as average over max lags
    grad = np.zeros(n_points)
    for i in max_lags:
        lag = i - k
        # ∂c[i]/∂h[j] = (1-h[j-lag]) * [0 <= j-lag < N] - h[j+lag] * [0 <= j+lag < N]
        # For j from 0 to N-1:
        # (1-h[j-lag]) valid when 0 <= j-lag < N, i.e., lag <= j < lag+N
        # -h[j+lag] valid when 0 <= j+lag < N, i.e., -lag <= j < N-lag
        
        # Vectorized:
        j = np.arange(n_points)
        # First term: (1-h[j-lag]) where 0 <= j-lag < N
        valid1 = (j - lag >= 0) & (j - lag < n_points)
        grad[valid1] += (1 - h[j[valid1] - lag]) / len(max_lags)
        
        # Second term: -h[j+lag] where 0 <= j+lag < N
        valid2 = (j + lag >= 0) & (j + lag < n_points)
        grad[valid2] -= h[j[valid2] + lag] / len(max_lags)
    
    return grad * dx

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    return float(np.max(corr) * dx)

def compute_subgradient(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    max_val = np.max(corr)
    
    # Find lags achieving max (within small tolerance for numerical stability)
    threshold = max_val - 1e-8
    max_lags = np.where(corr >= threshold)[0]
    count = len(max_lags)
    
    grad = np.zeros(n)
    for i in max_lags:
        lag = i - (n - 1)
        j = np.arange(n)
        # First term: (1-h[j-lag]) where 0 <= j-lag < n
        idx1 = j - lag
        valid1 = (idx1 >= 0) & (idx1 < n)
        grad[valid1] += (1 - h[idx1[valid1]])
        
        # Second term: -h[j+lag] where 0 <= j+lag < n
        idx2 = j + lag
        valid2 = (idx2 >= 0) & (idx2 < n)
        grad[valid2] -= h[idx2[valid2]]
    
    return (grad / count) * dx

    """Project h onto {0 <= x <= 1, sum(x) = target_sum} via water-filling."""
    n = len(h)
    # Bisection on threshold
    lo, hi = 0.0, 1.0

        projected = np.clip(h - mid, 0, 1)
        s = np.sum(projected)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    return float(np.max(corr) * dx)

def compute_subgradient(h):
    """Subgradient of C5(h) = max_k c_k * dx."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode="full")
    max_val = np.max(corr)
    
    # Find lags achieving (near) max
    tol = 1e-10
    indices = np.where(corr >= max_val - tol)[0]
    
    grad = np.zeros(n)
    j = np.arange(n)
    for i in indices:
        lag = i - (n - 1)
        # ∂c_k/∂h[j] = (1-h[j-lag])*[valid] - h[j+lag]*[valid]
        idx1 = j - lag
        m1 = (idx1 >= 0) & (idx1 < n)
        grad[m1] += (1.0 - h[idx1[m1]])
        idx2 = j + lag
        m2 = (idx2 >= 0) & (idx2 < n)
        grad[m2] -= h[idx2[m2]]
    
    return (grad / len(indices)) * dx

    """Project onto {0<=x<=1, sum(x)=target}."""

            lo = mid
        else:

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def subgradient(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    mv = np.max(corr)
    idx = np.where(corr >= mv - 1e-9)[0]
    g = np.zeros(n)
    j = np.arange(n)
    for i in idx:
        lag = i - (n - 1)
        a = j - lag
        m = (a >= 0) & (a < n)
        g[m] += 1.0 - h[a[m]]
        b = j + lag
        m2 = (b >= 0) & (b < n)
        g[m2] -= h[b[m2]]
    return (g / len(idx)) * dx

def project(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
