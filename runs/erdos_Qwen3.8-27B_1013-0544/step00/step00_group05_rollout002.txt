        if time.time() > time_limit:
            break
        n_pert = rng.randint(1, 4)
        idxs = rng.choice(n, n_pert, replace=False)
        old = h_cur[idxs].copy()
        h_cur[idxs] = np.clip(h_cur[idxs] + rng.randn(n_pert) * 0.2 * (T / T_init + 0.3), 0, 1)

        # Fix sum by adjusting a random other index
        other = rng.randint(n)
        while other in idxs:
            other = rng.randint(n)
        diff = target - h_cur.sum()
        h_cur[other] = np.clip(h_cur[other] - diff, 0, 1)

        if abs(h_cur.sum() - target) > 0.5:
            h_cur[idxs] = old
            h_cur[other] = h_cur[other] + diff if diff < 0 else h_cur[other]
            T *= cool
            continue

        c5_new = _compute_c5(h_cur)
        delta = c5_new - c5_cur
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-8)):
            c5_cur = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_cur.copy()
        T *= cool

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s

    # Get initial values from pre-imported global
    if 'initial_h_values' in dir() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = np.full(70, 0.5)

    n = len(h_init)
    target = n / 2.0
    best_h = _fix_sum(h_init, target)
    best_c5 = _compute_c5(best_h)

    # Phase 1: SLSQP with increasing temperature (sharpness)
    for T in [3.0, 8.0, 20.0, 50.0, 100.0, 200.0]:
        if time.time() > deadline - 120:
            break
        hx, c5 = _slsqp_refine(best_h, n, T, maxiter=300, ftol=1e-10)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hx

    # Phase 2: Simulated annealing
    t_left = deadline - time.time()
    if t_left > 100:
        n_iter = min(50000, int(t_left * 20))
        h_sa, c5_sa = _sa_optimize(best_h, n, 0.05, n_iter, deadline - 30, seed_offset=0)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa
        # Polish after SA
        if time.time() < deadline - 30:
            hx, c5 = _slsqp_refine(best_h, n, 100.0, maxiter=300, ftol=1e-10)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hx

    # Phase 3: Multi-restart with perturbations
    t_left = deadline - time.time()
    if t_left > 60:
        n_restarts = min(8, int(t_left // 10))
        for r in range(n_restarts):
            if time.time() > deadline - 20:
                break
            # Perturb best solution
            mask = np.random.random(n) < 0.25
            h_pert = best_h.copy()
            h_pert[mask] = np.clip(h_pert[mask] + np.random.randn(mask.sum()) * 0.3, 0, 1)
            h_pert = _fix_sum(h_pert, target)

            # SA from perturbed start
            h_candidate, c5_cand = _sa_optimize(h_pert, n, 0.03, 3000, deadline - 20, seed_offset=r + 100)
            # Polish
            if time.time() < deadline - 5:
                h_candidate, c5_cand = _slsqp_refine(h_candidate, n, 80.0, maxiter=150, ftol=1e-9)
            if c5_cand < best_c5:
                best_c5 = c5_cand
                best_h = h_candidate

    # Phase 4: Try binarization - push values to 0 or 1
    if time.time() < deadline - 10:
        # Threshold at various levels
        for thresh in [0.3, 0.4, 0.5, 0.6, 0.7]:
            h_bin = (best_h > thresh).astype(float)
            # Adjust to meet sum constraint
            s = h_bin.sum()
            needed = target
            if s < needed:
                # Set some 0s to 1s (those with highest h value)
                zeros = np.where(h_bin == 0)[0]
                if len(zeros) > 0:
                    order = np.argsort(-best_h[zeros])
                    for idx in order[:int(needed - s)]:
                        h_bin[zeros[idx]] = 1.0
            elif s > needed:
                ones = np.where(h_bin == 1)[0]
                order = np.argsort(best_h[ones])
                for idx in order[:int(s - needed)]:
                    h_bin[ones[idx]] = 0.0
            if abs(h_bin.sum() - target) < 1e-6:
                c5_bin = _compute_c5(h_bin)
                if c5_bin < best_c5:
                    best_c5 = c5_bin
                    best_h = h_bin
                    # Polish the binarized version
                    if time.time() < deadline - 5:
                        hx, c5 = _slsqp_refine(best_h, n, 50.0, maxiter=200, ftol=1e-10)
                        if c5 < best_c5:
                            best_c5 = c5
                            best_h = hx

    # Final polish
    if time.time() < deadline - 3:
        hx, c5 = _slsqp_refine(best_h, n, 100.0, maxiter=200, ftol=1e-11)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hx

    # Ensure exact sum
    best_h = _fix_sum(best_h, target)
    best_c5 = _compute_c5(best_h)

    return best_h, best_c5, n
