import numpy as np
import time

def repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    current = h.sum()
    if abs(current - target) < 1e-12:
        return h
    # Proportional adjustment
    if current > 0:
        h = h * (target / current)
    h = np.clip(h, 0.0, 1.0)
    # Fix remaining difference
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            room = np.where(h < 1.0, 1.0 - h, 0.0)
            total_room = room.sum()
            if total_room > 0:
                h += room * (diff / total_room)
            h = np.clip(h, 0.0, 1.0)
        else:
            space = np.where(h > 0.0, h, 0.0)
            total_space = space.sum()
            if total_space > 0:
                h -= space * (-diff / total_space)
            h = np.clip(h, 0.0, 1.0)
    # Final fix for floating point
    h = np.clip(h, 0.0, 1.0)
    r = target - h.sum()
    if abs(r) > 1e-12:
        # Add/remove from a single element
        idx = n // 2
        h[idx] = np.clip(h[idx] + r, 0.0, 1.0)
    return h

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return corr.max(), corr

def repair(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}."""
    h = np.clip(h, 0.0, 1.0)

    diff = target - h.sum()
