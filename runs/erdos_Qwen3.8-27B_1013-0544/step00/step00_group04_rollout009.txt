def binarize(h, n):

def upsample_binary(h, new_n):
    n = len(h)
    if new_n == n:
        return h.copy()
    # Linear interpolation upsampling
    old_positions = np.linspace(0, 2, n, endpoint=False)
    new_positions = np.linspace(0, 2, new_n, endpoint=False)
    new_h = np.interp(new_positions, old_positions, h, period=2)
    return binarize(new_h, new_n)

def soft_max(x, beta=100):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def smooth_objective(z, n, beta):
    """z are the parameters, h = sigmoid(z) adjusted for constraint"""
    h = 1.0 / (1.0 + np.exp(-z))
    # Adjust to satisfy sum constraint via scaling the bias
    # Actually, let's just compute the cost and add a constraint penalty
    corr = np.correlate(h, 1.0 - h, mode="full")
    cost = soft_max(corr, beta) * (2.0 / n)
    # Constraint penalty
    constraint_violation = np.sum(h) - n / 2
    cost += 100.0 * constraint_violation ** 2
    return cost

    # d(corr[k])/d(h[i]) = (1-h[i]) * [h[i+k] if valid] + h[i] * (1-h[i-k]) * ... 
    # This is complex, let's use numerical gradient
    eps = 1e-6

    idx = rng.choice(n, size=n_ones, replace=False)

    return h

def structured_initial(n, seed_val):
    """Create initial with golden-ratio spacing"""
    rng = np.random.RandomState(seed_val)
    n_ones = n // 2
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    positions = (np.arange(n_ones) * phi + 0.3) % 1.0
    indices = (positions * n).astype(int) % n
    h = np.zeros(n)
    h[indices] = 1.0
    # Fix duplicates
    seen = set()
    fill = 0
    for i in range(n_ones):
        while indices[i] in seen:
            indices[i] = (indices[i] + 1 + fill) % n
            fill += 1
        seen.add(indices[i])
    h = np.zeros(n)
    h[indices] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    h_init = None
    try:
        h_init = np.array(initial_h_values, dtype=float)
    except (NameError, UnboundLocalError):
        pass
    if h_init is None:
        n = 100
        h_init = np.zeros(n)
        h_init[0::2] = 1.0

    n0 = len(h_init)
    if n0 % 2 != 0:
        h_init = np.concatenate([h_init, np.zeros(1)])
        n0 += 1

    h0 = binarize(h_init, n0)
    best_h = h0.copy()
    best_cost = evaluate_c5(best_h)

    # Phase 1: Optimize at original size
    dl = t0 + budget_s * 0.20
    h_opt, c_opt = simulated_annealing(h0.copy(), 0.01, 0.0001, 200000, dl)
    if c_opt < best_cost:
        best_cost, best_h = c_opt, h_opt

    dl = t0 + budget_s * 0.32
    h_opt, c_opt = greedy_local_search(best_h.copy(), 40, dl, 5000)
    if c_opt < best_cost:
        best_cost, best_h = c_opt, h_opt

    # Phase 2: Multi-start at original size
    dl = t0 + budget_s * 0.42
    for trial in range(3):
        if time.time() > dl:
            break
        h_try = random_initial(n0, seed + 100 + trial * 7)
        h_try, c_try = simulated_annealing(h_try, 0.008, 0.0001, 80000, min(time.time() + (dl - time.time()) / 3, dl))
        if c_try < best_cost:
            best_cost, best_h = c_try, h_try

    # Phase 3: Try larger sizes
    target_sizes = []
    for mult in [2, 3, 4, 5, 6, 8]:
        s = n0 * mult
        if 40 <= s <= 800 and s % 2 == 0:
            target_sizes.append(s)
    target_sizes = sorted(set(target_sizes))

    deadline_total = t0 + budget_s * 0.86
    if target_sizes:
        remaining = deadline_total - time.time()
        per_size = max(remaining / len(target_sizes), 3.0)

        for n in target_sizes:
            if time.time() > deadline_total:
                break
            elapsed_size = time.time() - t0 + per_size * target_sizes.index(n)
            budget_here = min(per_size, deadline_total - time.time())
            if budget_here < 3:
                break

            h_work = upsample_binary(h_init, n)
            # Also try structured init
            h_struct = structured_initial(n, seed + n)
            c_struct = evaluate_c5(h_struct)
            if c_struct < evaluate_c5(h_work):
                h_work = h_struct

            d1 = time.time() + budget_here * 0.55
            h_sa, c_sa = simulated_annealing(h_work, 0.005, 0.00005, 100000, d1)

            d2 = time.time() + budget_here * 0.90
            h_g, c_g = greedy_local_search(h_sa, 25, d2, 2500)

            if c_g < best_cost:
                best_cost, best_h = c_g, h_g.copy()

    # Phase 4: Multi-start at best size found
    n_best = len(best_h)
    deadline_total = t0 + budget_s * 0.93
    if time.time() < deadline_total:
        n_trials_ms = 3
        time_each = (deadline_total - time.time()) / n_trials_ms
        for trial in range(n_trials_ms):
            if time.time() > deadline_total:
                break
            h_try = random_initial(n_best, seed + 500 + trial * 13)
            h_try, c_try = simulated_annealing(h_try, 0.004, 0.00005, 60000,
                                                time.time() + time_each * 0.7)
            h_try, c_try = greedy_local_search(h_try, 10, time.time() + time_each * 0.95, 2000)
            if c_try < best_cost:
                best_cost, best_h = c_try, h_try

    # Phase 5: Final intensive polish on best
    if time.time() < t0 + budget_s * 0.97:
        h_fin, c_fin = simulated_annealing(best_h.copy(), 0.001, 0.00001, 200000,
                                            t0 + budget_s * 0.97)
        if c_fin < best_cost:
            best_cost, best_h = c_fin, h_fin

    best_cost = evaluate_c5(best_h)
    return best_h, best_cost, len(best_h)
