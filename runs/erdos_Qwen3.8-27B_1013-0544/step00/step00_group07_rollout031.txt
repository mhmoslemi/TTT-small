    xs = np.linspace(0.0, 1.0, grid_size)
    n_corr = 2 * n - 1
    ks = np.arange(n_corr)
    s_vals = ks - (n - 1)
    
    for sweep in range(max_sweeps):
        max_change = 0.0
        corr = np.correlate(h, 1.0 - h, mode="full")
        
        for j in range(n):
            old_x = h[j]
            nks = j - s_vals
            jpks = j + s_vals
            
            valid_nk = (nks >= 0) & (nks < n)
            valid_jpk = (jpks >= 0) & (jpks < n)
            
            c1 = np.zeros(n_corr)
            c1[valid_nk] = 1.0 - h[nks[valid_nk]]
            c2 = np.zeros(n_corr)
            c2[valid_jpk] = h[jpks[valid_jpk]]
            coeff = c1 - c2
            
            delta = xs - old_x
            vals = corr[None, :] + coeff[None, :] * delta[:, None]
            f_vals = np.max(vals, axis=1)
            
            best_idx = np.argmin(f_vals)
            best_x = xs[best_idx]
            
            if abs(best_x - old_x) > 1e-12:
                h[j] = best_x
                corr = corr + coeff * (best_x - old_x)
                max_change = max(max_change, abs(best_x - old_x))
        
        if max_change < 1e-9:
            break
    
    return h

import numpy as np
import time

def project_feasible(h, n):
    """Euclidean projection onto {x: 0<=x<=1, sum(x)=n/2}"""
    target = n / 2.0
    h = np.asarray(h, dtype=float)
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def coordinate_descent(h, n, max_sweeps=300, grid_size=300):
    """Coordinate descent: for each j, find x in [0,1] minimizing max_k corr[k]."""
    h = h.copy().astype(float)
    xs = np.linspace(0.0, 1.0, grid_size)
    n_corr = 2 * n - 1
    ks = np.arange(n_corr)
    s_vals = ks - (n - 1)

    for sweep in range(max_sweeps):
        max_change = 0.0
        corr = np.correlate(h, 1.0 - h, mode="full")

        for j in range(n):
            old_x = h[j]
            nks = j - s_vals
            jpks = j + s_vals

            valid_nk = (nks >= 0) & (nks < n)
            valid_jpk = (jpks >= 0) & (jpks < n)

            c1 = np.zeros(n_corr)
            c1[valid_nk] = 1.0 - h[nks[valid_nk]]
            c2 = np.zeros(n_corr)
            c2[valid_jpk] = h[jpks[valid_jpk]]
            coeff = c1 - c2

            delta = xs - old_x
            vals = corr[None, :] + coeff[None, :] * delta[:, None]
            f_vals = np.max(vals, axis=1)

            best_idx = np.argmin(f_vals)
            best_x = xs[best_idx]

            if abs(best_x - old_x) > 1e-12:
                h[j] = best_x
                corr = corr + coeff * (best_x - old_x)
                max_change = max(max_change, abs(best_x - old_x))

    return h
