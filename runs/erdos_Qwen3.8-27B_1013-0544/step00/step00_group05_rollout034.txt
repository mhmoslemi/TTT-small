        elapsed = time.time() - start_time
        if elapsed > budget_s - 150:
            break
        _g_temp = temp
        try:
            res = minimize(obj_func, best_h, method='SLSQP', bounds=bounds,
                          constraints=[constraint], jac=grad_func,
                          options={'maxiter': 5000, 'ftol': 1e-16})
            h_cand = normalize_h(res.x, target_sum)
            c5_cand = float(np.max(compute_correlation(h_cand)))
            if c5_cand < best_c5 - 1e-13:
                best_c5 = c5_cand
                best_h = h_cand
        except Exception:
            pass

    # Phase 2: Multi-start with perturbations of varying scale
    for r in range(80):
        elapsed = time.time() - start_time
        if elapsed > budget_s - 80:
            break
        if r < 40:
            scale = 0.003 + 0.001 * r
        else:
            scale = 0.04 + 0.005 * (r - 40)
        h_start = best_h + np.random.randn(n) * scale
        h_start = normalize_h(h_start, target_sum)
        _g_temp = max(0.001, 0.004 * np.exp(-0.03 * r))
        try:
            res = minimize(obj_func, h_start, method='SLSQP', bounds=bounds,
                          constraints=[constraint], jac=grad_func,
                          options={'maxiter': 2000, 'ftol': 1e-15})
            h_cand = normalize_h(res.x, target_sum)
            c5_cand = float(np.max(compute_correlation(h_cand)))
            if c5_cand < best_c5 - 1e-13:
                best_c5 = c5_cand
                best_h = h_cand
        except Exception:
            pass

    # Phase 3: Subgradient refinement targeting the maximum shifts
    elapsed = time.time() - start_time
    if elapsed < budget_s - 10:
        h_cur = best_h.copy()
        lr = 0.005
        dx = 2.0 / n
        for step in range(500):
            elapsed = time.time() - start_time
            if elapsed > budget_s - 5:
                break
            c = compute_correlation(h_cur)
            temp_sub = 0.002
            scaled = c / temp_sub
            m = np.max(scaled)
            p = np.exp(scaled - m)
            p /= np.sum(p)
            grad = np.zeros(n)
            for k in range(2 * n - 1):
                if p[k] < 1e-10:
                    continue
                j_lo = max(0, n - 1 - k)
                j_hi = min(n - 1, k)
                for j in range(j_lo, j_hi + 1):
                    l = j + k - n + 1
                    grad[j] += p[k] * dx * (1.0 - h_cur[l])
                    grad[l] -= p[k] * dx * h_cur[j]
            h_new = h_cur - lr * grad
            h_new = normalize_h(h_new, target_sum)
            c5_new = float(np.max(compute_correlation(h_new)))
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_cur = h_new
            if step % 40 == 39:
                lr *= 0.75

    # Final verification
    best_h = normalize_h(best_h, target_sum)
    final_c5 = float(np.max(compute_correlation(best_h)))
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, n
