    n = len(h)

    max_corr = np.max(corr)
    exp_corr = np.exp(T * (corr - max_corr))
    weights = exp_corr / np.sum(exp_corr)

    # If k >= 0: grad[j] = dx * [(1-h[j-k]) * 1{k<=j<n} - h[j+k] * 1{j<n-k}]

def project_to_simplex(y, S):
    """Project y onto {x: 0≤x_i≤1, sum(x)=S}"""
    n = len(y)
    # Binary search on lambda
    lo, hi = 0.0, 1.0
    # We want sum(clip(y - lambda, 0, 1)) = S
    # If sum(y) > S, we need to decrease, so lambda > 0
    # If sum(y) < S, we need to increase, so lambda < 0
    for _ in range(100):
        mid = (lo + hi) / 2
        val = np.clip(y - mid, 0, 1)
        s = np.sum(val)
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2, 0, 1)

We want to minimize max_k (A_k - B_k).

However, a peaked h will have large O_k for other k values. So there's a trade-off.

This is essentially a question of finding the optimal "shape" of h to balance the trade-off between different shifts.

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr)

def project_simplex(y, S):
    n = len(y)
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def compute_softmax_grad(h, dx, T):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(corr)
    exp_c = np.exp(T * (corr - max_c))
    sw = np.sum(exp_c)
    w = exp_c / sw
    grad = np.zeros(n)
    for idx in range(len(corr)):
        k = idx - (n - 1)
        wi = w[idx]
        if wi < 1e-10:
            continue
        if k >= 0:
            j1 = np.arange(k, n)
            grad[j1] += wi * dx * (1.0 - h[j1 - k])
            if n - k > 0:
                j2 = np.arange(0, n - k)
                grad[j2] -= wi * dx * h[j2 + k]
        else:
            kp = -k
            if n - kp > 0:
                j1 = np.arange(0, n - kp)
                grad[j1] += wi * dx * (1.0 - h[j1 + kp])
            j2 = np.arange(kp, n)
            grad[j2] -= wi * dx * h[j2 - kp]
    return grad

def gradient_phase(h, dx, target, T_list, max_iter_per_T, time_fn, best_ref):
    n = len(h)
    cur = compute_c5(h, dx)
    for T in T_list:
        if time_fn() < 20:
            break
        lr = 0.12
        for it in range(max_iter_per_T):
            if time_fn() < 20:
                break
            grad = compute_softmax_grad(h, dx, T)
            h_new = project_simplex(h - lr * grad, target)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_ref[0]:
                best_ref[0] = c5_new
                best_ref[1] = h_new.copy()
            delta = c5_new - cur
            if delta <= 0 or np.random.random() < np.exp(-80.0 * max(delta, 0)):
                h = h_new
                cur = c5_new
            lr *= 1.0
        lr *= 0.5
    return h, cur, best_ref

def anneal_phase(h, dx, target, time_fn, best_ref, n_iter_limit=50000000):
    n = len(h)
    cur = compute_c5(h, dx)
    temp = 0.008
    cool = 0.99997
    count = 0
    while time_fn() > 5 and count < n_iter_limit:
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        d = np.random.uniform(-0.2, 0.2)
        vi, vj = h[i] + d, h[j] - d
        if vi < 0 or vi > 1 or vj < 0 or vj > 1:
            continue
        h[i], h[j] = vi, vj
        c5_new = compute_c5(h, dx)
        delta = c5_new - cur
        if delta <= 0 or np.random.random() < np.exp(-delta / temp):
            cur = c5_new
            if c5_new < best_ref[0]:
                best_ref[0] = c5_new
                best_ref[1] = h.copy()
        else:
            h[i], h[j] = vj, vi
        temp *= cool
        count += 1
    return h, cur, best_ref
