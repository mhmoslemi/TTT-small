import numpy as np
import time

def compute_c5(h):

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return np.max(c) * dx

def compute_grad(h, n):
    """Compute subgradient of C5 with respect to h."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    k_star = np.argmax(c)
    t = k_star - (n - 1)
    
    grad = np.zeros(n)
    
    if t == 0:
        grad = 1.0 - 2.0 * h
    elif t > 0:
        s = t
        # [i>=s] - h[i-s]*[i>=s] - h[i+s]*[i<=n-1-s]
        grad[s:] = 1.0 - h[:n-s]  # [i>=s] part: 1 - h[i-s]
        # Add the -h[i+s] term for i <= n-1-s
        grad[:n-s] -= h[s:]
    else:
        s = -t
        # [i<=n-1-s] - h[i+s]*[i<=n-1-s] - h[i-s]*[i>=s]

def compute_grad_smooth(h, n, epsilon=1e-6):
    """Compute smoothed subgradient (average over near-max shifts)."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(c)
    # Find all k where c[k] >= c_max - epsilon
    active = c >= (c_max - epsilon)
    # Weight by how close to max
    weights = np.zeros(len(c))
    weights[active] = c[active] - (c_max - epsilon) + 1.0  # some positive weight
    
    grad = np.zeros(n)
    for k in np.where(active)[0]:
        t = k - (n - 1)
        g = np.zeros(n)
        if t == 0:
            g = 1.0 - 2.0 * h
        elif t > 0:

            g[s:] += 1.0 - h[:n-s]
            g[:n-s] -= h[s:]
        else:
            s = -t
            g[:n-s] += 1.0 - h[s:]
            g[s:] -= h[:n-s]
        grad += g * weights[k]
    
    grad *= dx / np.sum(weights)
    return grad

    c = np.correlate(h, 1.0 - h, mode='full')

        if t_k >= 0:
