```python
import numpy as np
import time

_PARAM = {"n": 1, "x": np.array([], dtype=float)}


def _project(x, S):
    x = np.asarray(x, dtype=float).copy()
    x = np.clip(x, 0.0, 1.0)
    d = float(S - x.sum())

    if d > 0.0:
        cap = 1.0 - x
        csum = float(cap.sum())
        if csum > 0.0:
            x = x + cap * (d / csum)
    elif d < 0.0:
        cap = x
        csum = float(cap.sum())
        if csum > 0.0:
            x = x * (1.0 + d / csum)

    x = np.clip(x, 0.0, 1.0)
    d = float(S - x.sum())

    if d > 1e-14:
        idx = np.argsort(1.0 - x)[::-1]
        for i in idx:
            add = min(d, 1.0 - x[i])
            if add <= 1e-15:
                continue
            x[i] += add
            d -= add
            if d <= 1e-14:
                break
    elif d < -1e-14:
        idx = np.argsort(x)[::-1]
        for i in idx:
            sub = min(-d, x[i])
            if sub <= 1e-15:
                continue
            x[i] -= sub
            d += sub
            if d >= -1e-14:
                break

    return np.clip(x, 0.0, 1.0)


def _eval(h):
    n = len(h)
    if n <= 0:
        return 0.0
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def _grad_lag(h, s):
    n = len(h)
    g = np.zeros(n, dtype=float)

    if s == 0:
        return 1.0 - 2.0 * h

    if s > 0:
        g[: n - s] += 1.0 - h[s:]
        g[s:] -= h[:-s]
    else:
        t = -s
        g[t:] += 1.0 - h[:-t]
        g[: n - t] -= h[t:]

    return g


def _smooth_info(h, tau):
    n = len(h)
    q = np.correlate(h, 1.0 - h, mode="full")
    m = float(q.max())
    if tau <= 1e-6:
        tau = 1e-6

    w = np.exp((q - m) / tau)
    wsum = float(w.sum())
    obj = float(m + tau * np.log(wsum))

    g = np.zeros(n, dtype=float)
    if wsum > 0.0:
        idx = np.nonzero(w > 1e-4 * wsum)[0]
        for r in idx:
            wr = float(w[r] / wsum)
            if wr > 0.0:
                g += wr * _grad_lag(h, n - 1 - int(r))

    return obj, g, q


def _attempt_step(h, g, c, base_step, S):
    g = g - g.mean()
    gm = float(np.max(np.abs(g)))
    if gm < 1e-12:
        return None, 1e9

    g = g / gm
    for mult in (1.0, 0.5, 0.2, 0.05):
        h2 = _project(h - base_step * mult * g, S)
        c2 = _eval(h2)
        if c2 < c - 1e-12:
            return h2, c2

    return None, 1e9

                    cur_c = c2
                    if c2 < best_c:
                        best_h = cand.copy()
                        best_c = c2

    return best_h, best_c


def _optimize(h, duration, rng):
    duration = max(0.0, float(duration))
    t0 = time.time()
    end = t0 + duration

    n = len(h)
    S = 0.5 * n
    h = _project(h, S)
    c = _eval(h)

    best_h = h.copy()
    best_c = c

    if duration > 2.0:
        lim1 = min(end, t0 + 0.10 * duration)
        if time.time() < lim1:
            h, c = _local(h, c, lim1, rng)
            if c < best_c:
                best_h, best_c = h, c

        lim2 = min(end - max(1.0, 0.20 * duration), t0 + 0.55 * duration)
        if time.time() < lim2:
            h, c = _soft_opt(h, c, lim2, rng)
            if c < best_c:
                best_h, best_c = h, c

        h, c = _local(h, c, end, rng)
        if c < best_c:
            best_h, best_c = h, c

    return best_h, best_c


def _resample(h, n):
    n0 = len(h)
    if n0 == n:
        return _project(h, 0.5 * n)

    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n) + 0.5) * (2.0 / n)
    y = np.interp(x1, x0, h)
    return _project(y, 0.5 * n)


def _random_h(n, rng):
    return _project(0.5 + 0.4 * rng.standard_normal(n), 0.5 * n)


def _param_objective(p):
    n = _PARAM["n"]
    x = _PARAM["x"]
    h = 0.5 * np.ones(n, dtype=float)

    L = len(p) // 3
    for i in range(L):
        a = float(p[3 * i])
        f = float(p[3 * i + 1])
        ph = float(p[3 * i + 2])
        h += a * np.cos(2.0 * np.pi * f * x + ph)

    return _eval(_project(h, 0.5 * n))


def _param_de(n, K, maxiter, popsize, seed, maxfev):
    _PARAM["n"] = n
    _PARAM["x"] = (np.arange(n) + 0.5) * (2.0 / n)

    bounds = []
    for _ in range(K):
        bounds.append((-0.5, 0.5))
        bounds.append((0.25, 6.0))
        bounds.append((0.0, 2.0 * np.pi))

    try:
        from scipy.optimize import differential_evolution

        res = differential_evolution(
            _param_objective,
            bounds=bounds,
            seed=int(seed),
            maxiter=int(maxiter),
            maxfev=int(maxfev),
            popsize=int(popsize),
            tol=1e-5,
            polish=False,
        )
        p = res.x
    except Exception:
        return None

    x = _PARAM["x"]
    h = 0.5 * np.ones(n, dtype=float)
    for i in range(K):
        h += p[3 * i] * np.cos(2.0 * np.pi * p[3 * i + 1] * x + p[3 * i + 2])

    return _project(h, 0.5 * n)


def _process_candidate(h, dur, rng, bound_end, best_h, best_c, processed):
    if h is None:
        return best_h, best_c

    h = np.asarray(h, dtype=float)
    n = len(h)
    S = 0.5 * n
    h = _project(h, S)
    c = _eval(h)

    if c < best_c:
        best_h = h.copy()
        best_c = c

    dur = max(0.0, min(float(dur), bound_end - time.time()))

    if n in processed:
        return best_h, best_c

    if dur > 1.0:
        h2, c2 = _optimize(h, dur, rng)
        if c2 < best_c:
            best_h = h2.copy()
            best_c = c2

    processed.add(n)
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    seed = int(seed)
    B = max(60.0, float(budget_s))
    t0 = time.time()
    global_end = t0 + 0.97 * B

    rng = np.random.default_rng(seed)
    processed = set()

    h0 = None
    try:
        h0 = np.asarray(initial_h_values, dtype=float)
    except Exception:
        h0 = None

    if h0 is None:
        h0 = _random_h(95, rng)

    best_h = None
    best_c = 1e9

    dur_init = min(0.30 * B, 300.0)
    best_h, best_c = _process_candidate(
        h0, dur_init, rng, global_end, best_h, best_c, processed
    )

    if time.time() + 10.0 < global_end:
        K = 4 if B > 700 else 3
        maxiter = 50 if B > 900 else 30
        maxfev = int(min(8000, max(1000, 3.5 * B)))
        h_de = _param_de(128, K, maxiter, 8, seed + 101, maxfev)

        if h_de is not None:
            c_de = _eval(_project(h_de, 64.0))
            if c_de < best_c + 0.03:
                best_h, best_c = _process_candidate(
                    h_de, min(0.25 * B, 200.0), rng, global_end, best_h, best_c, processed
                )

    for n in (64, 128, 160, 192, 256):
        if time.time() + 5.0 >= global_end:
            break
        if n in processed:
            continue

        if best_h is not None:
            h_start = _resample(best_h, n)
        else:
            h_start = _random_h(n, rng)

        best_h, best_c = _process_candidate(
            h_start, min(0.10 * B, 80.0), rng, global_end, best_h, best_c, processed
        )

    if best_h is not None and time.time() + 5.0 < global_end:
        dur_final = min(0.25 * B, 220.0)
        best_h, best_c = _process_candidate(
            best_h, dur_final, rng, global_end, best_h, best_c, processed
        )

    if best_h is not None and time.time() + 5.0 < global_end:
        end_local = min(global_end, time.time() + min(60.0, 0.05 * B))
        best_h, best_c = _local(best_h, best_c, end_local, rng)

    n = len(best_h)
    S = 0.5 * n
    best_h = _project(best_h, S)
    best_c = _eval(best_h)

    return best_h, best_c, int(n)
```
