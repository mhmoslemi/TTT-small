import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    """Compute C5 value for a given h array."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def soft_max_value_and_grad(h, t):
    """Compute soft max of overlaps and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    overlaps = corr * dx
    
    m = np.max(overlaps)
    # Numerically stable log-sum-exp
    shifted = t * (overlaps - m)
    log_sum = np.log(np.sum(np.exp(shifted)))
    value = m + log_sum / t
    
    # Weights for each lag
    weights = np.exp(shifted) / np.sum(np.exp(shifted))
    
    # Gradient: weighted sum of gradients of each g_k
    grad = np.zeros(n)
    for k_idx in range(len(overlaps)):
        k = k_idx - (n - 1)  # actual lag
        m_abs = abs(k)
        if m_abs == 0:
            grad += weights[k_idx] * (1.0 - 2.0 * h)
        elif m_abs < n:
            # gradient[j] = (1-h[j+m])*I(0<=j<=n-1-m) - h[j-m]*I(m<=j<=n-1)
            # First term: (1-h[j+m]) for j in [0, n-1-m]
            first = np.zeros(n)
            first[:n-m_abs] = 1.0 - h[m_abs:]
            # Second term: -h[j-m] for j in [m, n-1]
            second = np.zeros(n)
            second[m_abs:] = -h[:n-m_abs]
            grad += weights[k_idx] * (first + second)

The idea:

3. Compute the gradient of that specific overlap
4. Take a step in the negative gradient direction
5. Project to the constraint set

Wait, I also want to make sure I'm handling the "full" correlation correctly. Let me double-check:

For h of length n, np.correlate(h, 1-h, mode='full') has length 2n-1.
The element at index i corresponds to lag (i - (n-1)).
The value is sum over valid j of h[j + lag] * (1-h[j]).

For lag = 0: sum_j h[j]*(1-h[j]) → this is always positive (since 0<=h<=1)
For lag = 1: sum_{j=0}^{n-2} h[j+1]*(1-h[j])
For lag = -1: sum_{j=1}^{n-1} h[j-1]*(1-h[j]) = sum_{j=0}^{n-2} h[j]*(1-h[j+1])

Note that lag=1 and lag=-1 give different values in general (unless h has specific symmetry).

The C5 value is the maximum over ALL these lags (including positive and negative).

OK, I think my code is correct. Let me finalize.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_constraint(h, target):
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def soft_max_val(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    overlaps = corr * dx
    m = np.max(overlaps)
    ex = np.exp(t * (overlaps - m))
    return m + np.log(np.mean(ex)) / t

def soft_max_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    overlaps = corr * dx
    m = np.max(overlaps)
    ex = np.exp(t * (overlaps - m))
    w = ex / ex.sum()
    grad = np.zeros(n)
    for k_idx in range(len(overlaps)):
        s = k_idx - (n - 1)
        ma = abs(s)
        if ma >= n:
            continue
        wi = w[k_idx]
        if wi < 1e-12:
            continue
        if ma == 0:
            grad += wi * (1.0 - 2.0 * h)
        else:
            grad[:n - ma] += wi * (1.0 - h[ma:])
            grad[ma:] -= wi * h[:n - ma]
    return grad

def penalized_obj(x, t, target, pen):
    val = soft_max_val(x, t)
    viol = np.sum(x) - target
    return val + pen * viol * viol

def penalized_grad(x, t, target, pen):
    g = soft_max_grad(x, t)
    viol = np.sum(x) - target
    g += 2.0 * pen * viol * np.ones(len(x))
    return g

def subgradient_step(h, target, lr):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    overlaps = corr * dx
    k_idx = int(np.argmax(overlaps))
    s = k_idx - (n - 1)
    ma = abs(s)
    if ma == 0:
        grad = 1.0 - 2.0 * h
    elif ma < n:
        grad = np.zeros(n)
        grad[:n - ma] = 1.0 - h[ma:]
        grad[ma:] += -h[:n - ma]
    else:
        return h
    h = project_to_constraint(h - lr * grad, target)
    return h

    t_start = time.time()
    rng = np.random.RandomState(seed)

    h_best = np.array(initial_h_values, dtype=float).copy()
    n = len(h_best)
    target = n * 0.5
    h_best = project_to_constraint(h_best, target)
    best_c5 = compute_c5(h_best)

    def elapsed():
        return time.time() - t_start

    # Phase 1: Subgradient warm-start
    lr = 0.1
    for i in range(3000):
        if elapsed() > budget_s * 0.15:
            break
        h_best = subgradient_step(h_best, target, lr)
        lr *= 0.9995
        c5 = compute_c5(h_best)
        if c5 < best_c5:

    # Phase 2: L-BFGS-B with soft max at increasing temperatures
    for t in [10.0, 30.0, 80.0, 200.0, 500.0, 1500.0]:
        if elapsed() > budget_s * 0.5:

    # Phase 3: Random restarts with L-BFGS-B
