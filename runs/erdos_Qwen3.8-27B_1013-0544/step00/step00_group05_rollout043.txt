   x=np.array(h,dtype=float)
   n=x.size
   if n==0: return x
   target = n*0.5
   np.clip(x,0,1,out=x)
   for _ in range(50):
      diff=target-x.sum()
      if abs(diff) < 1e-13: break
      if diff>0:
         slack=1-x; mask=slack>1e-14
         if not mask.any():
             x[np.arange(int(np.floor(target)))] = 1.0
             rem = target - int(np.floor(target))
             if rem > 0 and int(np.floor(target)) < n: x[int(np.floor(target))]=rem
             break
         x[mask] += diff * slack[mask] / slack[mask].sum()
      else:

         if not mask.any(): x[:]=0; break
         x[mask] += diff * x[mask] / x[mask].sum()

   diff = target - x.sum()

      if k>0: x[:k]=1.0

import numpy as np
import time
import scipy.optimize as so
import warnings

warnings.simplefilter("ignore")

_O = {}


def time_left(deadline):
    return max(0.0, deadline - time.time())


def set_opt(tau, topk):
    _O["tau"] = float(tau)
    _O["topk"] = topk


def repair(h):
    x = np.array(h, dtype=float, copy=True)
    n = x.size
    if n == 0:
        return x
    target = 0.5 * float(n)
    np.clip(x, 0.0, 1.0, out=x)

    for _ in range(40):
        diff = target - float(x.sum())
        if abs(diff) <= 1e-13:
            break
        if diff > 0.0:
            slack = 1.0 - x
            mask = slack > 1e-14
            if not np.any(mask):
                k = int(np.floor(target))
                x[:] = 0.0
                if k > 0:
                    x[:k] = 1.0
                rem = target - float(k)
                if rem > 0.0 and k < n:
                    x[k] = min(rem, 1.0)
                break
            x[mask] += diff * slack[mask] / float(slack[mask].sum())
        else:
            mask = x > 1e-14
            if not np.any(mask):
                x[:] = 0.0
                break
            x[mask] += diff * x[mask] / float(x[mask].sum())
        np.clip(x, 0.0, 1.0, out=x)

    diff = target - float(x.sum())
    if abs(diff) > 1e-10:
        k = int(np.floor(target))
        x[:] = 0.0
        if k > 0:
            x[:k] = 1.0
        rem = target - float(k)
        if rem > 0.0 and k < n:
            x[k] = min(rem, 1.0)
    elif diff > 0.0:
        order = np.argsort(1.0 - x)[::-1]
        for idx in order:
            if diff <= 1e-15:
                break
            room = 1.0 - x[idx]
            if room > 0.0:
                add = min(diff, room)
                x[idx] += add
                diff -= add
    elif diff < 0.0:
        order = np.argsort(x)[::-1]
        for idx in order:
            if diff >= -1e-15:
                break
            room = x[idx]
            if room > 0.0:
                add = max(diff, -room)
                x[idx] += add
                diff -= add
    return x


def c_values(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, mode="full") * dx


def exact_c(h):
    return float(np.max(c_values(h)))


def add_lag_grad(x, grad, k, scale):
    if scale == 0.0:
        return
    n = x.size
    s = int(k) - (n - 1)

    lo = s if s > 0 else 0
    hi = s + n - 1
    if hi >= n:
        hi = n - 1
    if lo <= hi:
        grad[lo:hi + 1] += scale * (1.0 - x[lo - s:hi + 1 - s])

    lo = -s if s < 0 else 0
    hi = n - 1 - s
    if hi >= n:
        hi = n - 1
    if lo <= hi:
        grad[lo:hi + 1] -= scale * x[lo + s:hi + 1 + s]


def weighted_grad(x, w):
    n = x.size
    grad = np.zeros(n, dtype=float)
    idxs = np.nonzero(w > 1e-13)[0]
    for k in idxs:
        add_lag_grad(x, grad, int(k), float(w[k]))
    return grad * (2.0 / float(n))


def active_grad(x, active):
    n = x.size
    grad = np.zeros(n, dtype=float)
    if active.size == 0:
        return grad
    for k in active:
        add_lag_grad(x, grad, int(k), 1.0)
    return grad * (2.0 / float(n) / float(active.size))


def smooth_eval(x):
    x = np.asarray(x, dtype=float)
    n = x.size
    dx = 2.0 / float(n)
    tau = float(_O.get("tau", 0.01))
    topk = _O.get("topk", None)

    c = np.correlate(x, 1.0 - x, mode="full") * dx
    cm = float(np.max(c))
    m = c.size

    if tau <= 0.0:
        w = np.zeros(m, dtype=float)
        w[int(np.argmax(c))] = 1.0
        return cm, c, w

    z = (c - cm) / tau
    z = z - np.max(z)
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        w = np.zeros(m, dtype=float)
        w[int(np.argmax(c))] = 1.0
        return cm, c, w

    w = e / s
    if topk is not None and 0 < int(topk) < m:
        k = int(topk)
        idx = np.argsort(w)[::-1][:k]
        w2 = np.zeros(m, dtype=float)
        w2[idx] = w[idx]
        ss = float(w2.sum())
        if ss > 0.0:
            w = w2 / ss

    return cm + tau * float(np.log(s)), c, w


def smooth_fun(x):
    val, c, w = smooth_eval(x)
    return val


def smooth_jac(x):
    val, c, w = smooth_eval(x)
    return weighted_grad(x, w)


def max_fun(x):
    return exact_c(x)


def max_jac(x):
    c = c_values(x)
    C = float(np.max(c))
    tol = max(1e-9, 1e-5 * C)
    active = np.where(c >= C - tol)[0]
    if active.size > 6:
        active = np.argsort(c)[::-1][:6]
    return active_grad(x, active)


def cons_fun(x):
    x = np.asarray(x, dtype=float)
    return np.array([float(x.sum()) - 0.5 * float(x.size)])


def cons_jac(x):
    return np.ones((1, x.size), dtype=float)


def run_slsqp(fun, jac, x0, maxiter):
    x0 = np.asarray(x0, dtype=float)
    n = x0.size
    bounds = [(0.0, 1.0) for _ in range(n)]
    cons = [{"type": "eq", "fun": cons_fun, "jac": cons_jac}]
    opts = {"maxiter": int(maxiter), "ftol": 1e-12}
    try:
        return so.minimize(
            fun,
            x0,
            method="SLSQP",
            jac=jac,
            bounds=bounds,
            constraints=cons,
            options=opts,
        )
    except Exception:
        return None


def resample_to(h0, n_new):
    h0 = np.asarray(h0, dtype=float)
    n0 = h0.size
    n_new = int(n_new)
    if n0 == n_new:
        return repair(h0)
    if n0 == 0:
        return repair(np.full(n_new, 0.5))

    dt0 = 2.0 / float(n0)
    x0 = (np.arange(n0) + 0.5) * dt0
    x1 = (np.arange(n_new) + 0.5) * (2.0 / float(n_new))
    y = np.interp(x1, x0, h0)
    return repair(y)


def jitter_start(h0, scale, seed):
    h0 = np.asarray(h0, dtype=float)
    rng = np.random.RandomState(int(seed))
    y = h0 + rng.normal(0.0, float(scale), h0.size)
    return repair(y)


def structured_start(n, seed, freq, amp=0.42):
    rng = np.random.RandomState(int(seed))
    n = int(n)
    t = (np.arange(n) + 0.5) * (2.0 / float(n))
    h = np.full(n, 0.5, dtype=float)
    for f in (freq, 2 * freq, 3 * freq):
        h += (float(amp) / float(f)) * 0.55 * np.sin(
            2.0 * np.pi * float(f) * t / 2.0 + rng.uniform(0.0, 6.283185307179586)
        )
    h += 0.012 * rng.normal(0.0, 1.0, n)
    return repair(h)


def random_start(n, seed):
    rng = np.random.RandomState(int(seed))
    n = int(n)
    t = (np.arange(n) + 0.5) * (2.0 / float(n))
    f = int(rng.randint(1, max(2, n // 10 + 1)))
    amp = 0.30 + 0.18 * float(rng.rand())
    h = 0.5 + amp * np.sin(2.0 * np.pi * float(f) * t / 2.0 + rng.uniform(0.0, 6.283185307179586))
    h += 0.05 * rng.normal(0.0, 1.0, n)
    return repair(h)


def exact_polish(x0, time_limit, seed):
    x = repair(x0)
    best_x = x.copy()
    best_c = exact_c(x)
    rng = np.random.RandomState(int(seed))
    deadline = time.time() + max(0.0, float(time_limit))
    step = 0.3
    no_imp = 0

    while time.time() < deadline:
        cvals = c_values(x)
        C = float(np.max(cvals))
        if C < best_c - 1e-15:
            best_c = C
            best_x = x.copy()
            no_imp = 0
        else:
            no_imp += 1

        if no_imp > 15:
            improved = False
            for sg in (0.02, 0.01, 0.005, 0.002):
                y = repair(x + rng.normal(0.0, float(sg), x.size))
                cy = exact_c(y)
                if cy < best_c:
                    best_c = cy
                    best_x = y.copy()
                    x = y
                    no_imp = 0
                    improved = True
                    break
            if not improved:
                break
            continue

        if no_imp > 5:
            active = np.argsort(cvals)[::-1][:3]
        else:
            active = np.where(cvals >= C - max(1e-8, 1e-5 * C))[0]

        grad = active_grad(x, active)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-14:
            break

        accepted = False
        for _ in range(15):
            y = repair(x - step * grad)
            cy = exact_c(y)
            if cy < C - 1e-12:
                x = y
                accepted = True
                break
            step *= 0.5
            if step < 1e-10:
                break

        if accepted:
            no_imp = 0
        else:
            no_imp += 1

        if rng.rand() < 0.12:
            y = repair(x + rng.normal(0.0, 0.004, x.size))
            cy = exact_c(y)
            if cy < best_c:
                best_c = cy
                best_x = y.copy()
                x = y
                no_imp = 0

    return best_x, best_c


def optimize_h0(x0, time_limit, seed):
    x = repair(x0)
    best_x = x.copy()
    best_c = exact_c(x)
    start = time.time()
    n = x.size

    if n <= 80:
        scale = 1.0
    elif n <= 140:
        scale = 0.75
    else:
        scale = 0.5

    base_stages = [
        (0.03, 220),
        (0.01, 300),
        (0.005, 320),
        (0.002, 320),
        (0.0008, 250),
    ]

    stages = []
    if time_limit > 20.0:
        stages.append((base_stages[0][0], max(20, int(base_stages[0][1] * scale))))
    stages.append((base_stages[1][0], max(20, int(base_stages[1][1] * scale))))
    if time_limit > 35.0:
        stages.append((base_stages[2][0], max(20, int(base_stages[2][1] * scale))))
    stages.append((base_stages[3][0], max(20, int(base_stages[3][1] * scale))))
    if time_limit > 70.0:
        stages.append((base_stages[4][0], max(20, int(base_stages[4][1] * scale))))

    for tau, mi in stages:
        if time.time() - start > time_limit * 0.82:
            break
        set_opt(tau, None)
        res = run_slsqp(smooth_fun, smooth_jac, x, mi)
        if res is not None:
            xr = repair(res.x)
            c = exact_c(xr)
            if c < best_c:
                best_c = c
                best_x = xr.copy()
            if c <= best_c + 2e-4:
                x = xr
            else:
                x = best_x

    if time.time() - start < time_limit * 0.92:
        set_opt(0.0, None)
        res = run_slsqp(max_fun, max_jac, best_x, max(30, int(80 * scale)))
        if res is not None:
            xr = repair(res.x)
            c = exact_c(xr)
            if c < best_c:
                best_c = c
                best_x = xr.copy()

    tl = time_limit - (time.time() - start)
    if tl > 2.0:
        xp, cp = exact_polish(best_x, min(8.0, max(0.5, tl * 0.7)), seed)
        if cp < best_c:
            best_c = cp
            best_x = xp

    return best_x, best_c


def run(seed=42, budget_s=1100, **kwargs):
    deadline = time.time() + max(1.0, float(budget_s))
    seed_val = int(seed)

    raw = globals().get("initial_h_values", None)
    init = None if raw is None else np.asarray(raw, dtype=float)

    if init is not None and init.size >= 10:
        base_n = int(init.size)
        if base_n > 200:
            base_n = 200
    else:
        base_n = 70

    ordered = []
    for n in (base_n, base_n + 10, base_n - 10, base_n + 20, base_n - 20, base_n + 30, 120, 150, 200):
        n = int(n)
        if 20 <= n <= 200 and n not in ordered:
            ordered.append(n)
    if len(ordered) > 6:
        ordered = ordered[:6]

    best_x = None
    best_c = 1e9
    first_n = True

    for n in ordered:
        if time_left(deadline) < 70.0:
            break

        if init is not None and init.size > 0:
            base = resample_to(init, n)
        else:
            base = repair(np.full(n, 0.5))

        limit = 7 if first_n else 3
        first_n = False

        starts = [base]
        for scale in (0.005, 0.02, 0.05, 0.10):
            if len(starts) >= limit:
                break
            starts.append(jitter_start(base, scale, seed_val + len(starts) * 17 + n))

        for fq in (1, 2, 3):
            if len(starts) >= limit:
                break
            starts.append(structured_start(n, seed_val + fq * 101 + n, fq))

        starts.append(random_start(n, seed_val + 777 + n))
        if len(starts) > limit:
            starts = starts[:limit]

        if n <= 80:
            per = 35.0
        elif n <= 140:
            per = 25.0
        else:
            per = 15.0
        per = min(per, max(6.0, time_left(deadline) * 0.15))

        for ii, h0 in enumerate(starts):
            if time_left(deadline) < 20.0:
                break
            x, c = optimize_h0(h0, per, seed_val + 1000 * n + ii)
            if c < best_c:
                best_c = c
                best_x = x

    if best_x is None:
        best_x = repair(np.full(20, 0.5))
        best_c = exact_c(best_x)

    best_x = repair(best_x)
    best_c = exact_c(best_x)
    return best_x, best_c, int(best_x.size)
