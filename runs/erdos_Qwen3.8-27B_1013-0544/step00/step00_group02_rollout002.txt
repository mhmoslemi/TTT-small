import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_objective_and_grad(h, mu):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    corr_max = np.max(corr)
    shifted = (corr - corr_max) / mu
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    val = mu * np.log(sum_exp) + corr_max
    weights = exp_shifted / sum_exp
    
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        w = weights[k_idx]
        if w < 1e-10:
            continue
        k = k_idx - (n - 1)
        if k >= 0:
            idx_i = np.arange(n - k)
            grad[idx_i] += w * (1.0 - h[idx_i + k])
            grad[idx_i + k] -= w * h[idx_i]
        else:
            idx_i = np.arange(-k, n)
            grad[idx_i] += w * (1.0 - h[idx_i + k])
            grad[idx_i + k] -= w * h[idx_i]
    return val, grad * dx

    """Project h onto {x : sum(x) = target_sum} while keeping 0<=x<=1"""
    n = len(h)

        if np.sum(mask) > 0:

        pk = p[ki]
        if pk < 1e-12:
            continue
        k = ki - (n - 1)
        jlo = max(0, -k)
        jhi = min(n - 1, n - 1 - k)
        if jlo <= jhi:
            grad[jlo:jhi + 1] += pk * (1.0 - h[jlo + k:jhi + k + 1])
        jlo2 = max(0, k)
        jhi2 = min(n - 1, n - 1 + k)
        if jlo2 <= jhi2:
            grad[jlo2:jhi2 + 1] -= pk * h[jlo2 - k:jhi2 - k + 1]
    return val, grad * dx

def project_feasible(h, target_sum):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hp) > target_sum:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - np.sum(hp)
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = hp < 1.0
        else:
            mask = hp > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            hp[mask] += diff / cnt
            hp = np.clip(hp, 0.0, 1.0)
    return hp

def slsqp_optimize(h0, mu, maxiter=500):
    n = len(h0)
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        v, _ = smooth_obj_grad(x, mu)
        return v

    def grad(x):
        _, g = smooth_obj_grad(x, mu)
        return g

    def cons_fun(x):
        return np.sum(x) - target_sum

    def cons_jac(x):
        return np.ones(n)

    constraints = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    result = minimize(obj, h0, method='SLSQP', jac=grad,
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
    return result.x

def projected_gradient(h, mu, n_iter=3000):
    n = len(h)
    target_sum = n / 2.0
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_val = compute_c5(h_curr)
    alpha = 0.01
    for it in range(n_iter):
        val, grad = smooth_obj_grad(h_curr, mu)
        alpha_new = alpha
        accepted = False
        for _ in range(40):
            h_trial = project_feasible(h_curr - alpha_new * grad, target_sum)
            trial_val = compute_c5(h_trial)
            if trial_val < val - 1e-10:
                h_curr = h_trial
                if trial_val < best_val:
                    best_val = trial_val
                    best_h = h_curr.copy()
                accepted = True
                break
            alpha_new *= 0.5
        if not accepted:
            alpha = alpha_new
        else:
            alpha = min(alpha * 1.2, 0.1)
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)

    def tleft():
        return budget_s - (time.time() - start_time)

    # Phase 1: SLSQP with annealing on mu
    mus_slsqp = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
    for mu in mus_slsqp:
        if tleft() < 90:
            break
        try:
            h_opt = slsqp_optimize(best_h, mu, maxiter=300)
            h_opt = np.clip(h_opt, 0.0, 1.0)
            h_opt = project_feasible(h_opt, n / 2.0)
            c5_val = compute_c5(h_opt)
            if c5_val < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_val
        except Exception:
            pass

    # Phase 2: Projected gradient refinement
    for mu in [0.02, 0.01, 0.005, 0.002, 0.001]:
        if tleft() < 60:
            break
        h_opt, c5_val = projected_gradient(best_h, mu, n_iter=3000)
        if c5_val < best_c5:
            best_h = h_opt
            best_c5 = c5_val

    # Phase 3: Random restarts at current n
    n_restarts = 30
    for r in range(n_restarts):
        if tleft() < 120:
            break
        scale = 0.03 + 0.02 * (r % 5)
        h_pert = best_h + rng.randn(n) * scale
        h_pert = project_feasible(h_pert, n / 2.0)
        for mu in [0.05, 0.01, 0.005, 0.002]:
            if tleft() < 45:
                break
            h_opt, c5_val = projected_gradient(h_pert, mu, n_iter=2000)
            if c5_val < best_c5:
                best_h = h_opt
                best_c5 = c5_val
            h_pert = best_h.copy()

    # Phase 4: Try larger n by interpolation
    for n_new in [90, 135, 180]:
        if tleft() < 300:
            break
        x_old = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
        x_new = np.linspace(0.0, 2.0, n_new, endpoint=False) + 1.0 / n_new
        h_ext = np.concatenate([best_h, [best_h[0]]])
        x_ext = np.concatenate([x_old, [x_old[0] + 2.0]])
        h_interp = np.interp(x_new, x_ext, h_ext)
        h_interp = project_feasible(h_interp, n_new / 2.0)

        for mu in [0.1, 0.03, 0.01, 0.005, 0.002]:
            if tleft() < 60:
                break
            h_opt, c5_val = projected_gradient(h_interp, mu, n_iter=3000)
            if c5_val < best_c5:
                best_h = h_opt
                best_c5 = c5_val
                n = n_new
                h_interp = best_h.copy()

    # Phase 5: Final polish with projected gradient
    if tleft() > 30:
        for mu in [0.001, 0.0005]:
            h_opt, c5_val = projected_gradient(best_h, mu, n_iter=2000)
            if c5_val < best_c5:
                best_h = h_opt
                best_c5 = c5_val

    # Ensure final solution is feasible
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
