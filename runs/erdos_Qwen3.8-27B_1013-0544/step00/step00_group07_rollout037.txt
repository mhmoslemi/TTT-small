    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def make_feasible(h, target_sum):
    """Project h to be in [0,1] with sum = target_sum."""
    h = np.clip(h, 0, 1)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    # Distribute among non-saturated
    mask = (h > 1e-10) & (h < 1 - 1e-10)
    if mask.sum() == 0:
        # All saturated, need to use boundary values
        mask = (h < 1) if diff > 0 else (h > 0)
    if mask.sum() > 0:
        step = min(abs(diff) / mask.sum(), 0.5)
        if diff > 0:
            h[mask] = np.clip(h[mask] + step, 0, 1)
        else:
            h[mask] = np.clip(h[mask] + step, 0, 1)
    # If still not exact, adjust on a sliding basis
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        # Adjust individual elements
        if diff > 0:
            for i in range(len(h)):
                if h[i] < 1:
                    adj = min(diff, 1 - h[i])
                    h[i] += adj
                    diff -= adj
                    if abs(diff) < 1e-12:
                        break
        else:
            for i in range(len(h)):
                if h[i] > 0:
                    adj = min(-diff, h[i])
                    h[i] -= adj
                    diff += adj
                    if abs(diff) < 1e-12:
                        break
    return h

def c5_score(h):

    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n)

    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    mask = (h > 1e-14) & (h < 1.0 - 1e-14)
    if mask.sum() > 0:
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)

        for i in np.where(h < 1.0)[0]:
            d = min(diff, 1.0 - h[i])

            diff -= d

    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    mask = (h > 1e-14) & (h < 1.0 - 1e-14)
    if mask.sum() > 0:
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if diff > 1e-14:
        for i in np.where(h < 1.0)[0]:
            d = min(diff, 1.0 - h[i]); h[i] += d; diff -= d
            if diff < 1e-14: break
    elif diff < -1e-14:

            d = min(-diff, h[i]); h[i] -= d; diff += d
            if diff > -1e-14: break
