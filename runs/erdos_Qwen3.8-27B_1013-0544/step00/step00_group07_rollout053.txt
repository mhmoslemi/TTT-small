```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(h, args):
    beta, dx, n, pen = args
    hc = np.clip(h, 0.0, 1.0)
    corr = np.correlate(hc, 1.0 - hc, mode='full')
    z = beta * dx * corr
    zmax = np.max(z)
    lse = zmax + np.log(np.sum(np.exp(z - zmax)))
    base = lse / beta
    integral = np.sum(hc) * dx
    penalty = pen * (integral - 1.0)**2
    return base + penalty

def smooth_grad(h, args):
    beta, dx, n, pen = args
    hc = np.clip(h, 0.0, 1.0)
    corr = np.correlate(hc, 1.0 - hc, mode='full')
    z = beta * dx * corr
    zmax = np.max(z)
    ez = np.exp(z - zmax)
    p = ez / np.sum(ez)
    
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        k = m - (n - 1)
        w = p[m] * dx
        if w < 1e-14:
            continue
        if k >= 0:
            if k < n:
                grad[k:] += w * (1.0 - hc[:n-k])
                grad[:n-k] -= w * hc[k:]
        else:
            s = -k
            grad[:n-s] += w * (1.0 - hc[s:])
            grad[s:] -= w * hc[:n-s]
    
    integral = np.sum(hc) * dx
    grad += 2.0 * pen * (integral - 1.0) * dx
    return grad

def project_h(h, n):
    """Project h to [0,1]^n with sum(h) = n/2."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = np.sum(h)
    if abs(s - target) < 1e-10:
        return h
    if s > target:
        # Scale down
        lo, hi = 0.0, 1.0
        for _ in range(60):
            mid = (lo + hi) / 2.0
            if np.sum(np.clip(h * mid, 0, 1)) > target:
                hi = mid
            else:
                lo = mid
        h = np.clip(h * lo, 0, 1)
    else:
        # Scale up
        lo, hi = 0.0, 1.0
        for _ in range(60):
            mid = (lo + hi) / 2.0
            h_test = h + mid * (1.0 - h)
            if np.sum(h_test) < target:
                lo = mid
            else:
                hi = mid
        h = h + lo * (1.0 - h)
    return np.clip(h, 0, 1)

def coord_descent(h, n, dx, max_iters=200, rng=None):
    """Coordinate descent with integral constraint."""
    c5 = compute_c5(h, dx)
    h = h.copy()
    
    for it in range(max_iters):
        improved = False
        # Random order
        order = rng.permutation(n) if rng is not None else np.arange(n)
        for i in order:
            old_val = h[i]
            # Try new values
            best_t = old_val
            best_c5 = c5
            
            for t in np.concatenate([np.linspace(0, 1, 11), [old_val]]):
                if abs(t - old_val) < 1e-10:
                    continue
                h[i] = t
                # Renormalize to maintain integral
                h_proj = project_h(h, n)
                c5_new = compute_c5(h_proj, dx)
                if c5_new < best_c5 - 1e-12:
                    best_c5 = c5_new
                    best_t = t
            
            if best_t != old_val:
                h[i] = best_t
                h = project_h(h, n)
                c5 = best_c5
                improved = True
            else:
                h[i] = old_val
        
        if not improved:
            break
    
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    
    def time_left():
        return budget_s - (time.time() - t_start)
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = np.ones(80) * 0.5
    
    n = len(h_init)
    dx = 2.0 / n
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(h, args):
    beta, dx, n, pen = args
    hc = np.clip(h, 0.0, 1.0)
    corr = np.correlate(hc, 1.0 - hc, mode='full')
    z = beta * dx * corr
    zmax = np.max(z)
    lse = zmax + np.log(np.sum(np.exp(z - zmax)))
    base = lse / beta
    integral = np.sum(hc) * dx
    return base + pen * (integral - 1.0) ** 2

def smooth_grad(h, args):
    beta, dx, n, pen = args
    hc = np.clip(h, 0.0, 1.0)
    corr = np.correlate(hc, 1.0 - hc, mode='full')
    z = beta * dx * corr
    zmax = np.max(z)
    ez = np.exp(z - zmax)
    p = ez / np.sum(ez)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        k = m - (n - 1)
        w = p[m] * dx
        if w < 1e-14:
            continue
        if k >= 0:
            if k < n:
                grad[k:] += w * (1.0 - hc[:n - k])
                grad[:n - k] -= w * hc[k:]
        else:
            s = -k
            grad[:n - s] += w * (1.0 - hc[s:])
            grad[s:] -= w * hc[:n - s]
    integral = np.sum(hc) * dx
    grad += 2.0 * pen * (integral - 1.0) * dx
    return grad

def project_h(h, n):

            else:

            best_c5 = c5_new
            best_t = t
    h[i] = best_t
    h[:] = project_h(h, n)
    return best_c5
```
