import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr)

def local_search(h, dx, time_budget=30.0):
    n = len(h)
    best_c5 = compute_c5(h, dx)
    t0 = time.time()
    while time.time() - t0 < time_budget:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_delta = 1e-12
        best_i = -1
        best_j = -1
        for i in ones:
            h[i] = 0.0
            for j in zeros:
                h[j] = 1.0
                c5 = compute_c5(h, dx)
                delta = c5 - best_c5
                if delta < best_delta:
                    best_delta = delta
                    best_i = i
                    best_j = j
                h[j] = 0.0
            h[i] = 1.0
        if best_i >= 0:
            h[best_i] = 0.0
            h[best_j] = 1.0
            best_c5 = best_c5 + best_delta
        else:
            break
    return h, best_c5

def make_binary(h_cont, n_target, seed=0):
    rng = np.random.RandomState(seed)
    n = len(h_cont)
    if n != n_target:
        if n > n_target:
            idx = np.linspace(0, n - 1, n_target).astype(int)
            h_cont = h_cont[idx]
        else:
            h_cont = np.pad(h_cont, (0, n_target - n), mode='edge')
        n = n_target
    h_bin = (h_cont > 0.5).astype(float)
    target = n // 2
    actual = int(np.sum(h_bin))
    if actual > target:
        ones = np.where(h_bin == 1.0)[0]
        order = np.argsort(h_cont[ones])
        for k in range(actual - target):
            h_bin[ones[order[k]]] = 0.0
    elif actual < target:
        zeros = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h_cont[zeros])
        for k in range(target - actual):
            h_bin[zeros[order[k]]] = 1.0
    return h_bin
