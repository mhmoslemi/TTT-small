def coord_descent_step(h, j, penalty):
    n = len(h)
    dx = 2.0 / n
    g = 1 - h
    
    # Compute correlation with h[j] = 0 and h[j] = 1
    h0 = h.copy(); h0[j] = 0
    h1 = h.copy(); h1[j] = 1
    
    corr0 = np.correlate(h0, 1-h0, mode='full') * dx
    corr1 = np.correlate(h1, 1-h1, mode='full') * dx
    
    a = corr0  # constant part
    b = corr1 - corr0  # coefficient of x
    
    s = np.sum(h) - h[j]  # sum of other coordinates
    target = n / 2
    
    def obj(x):
        max_corr = np.max(a + b * x)
        penalty_term = penalty * (s + x - target) ** 2
        return max_corr + penalty_term

        m2 = hi - (hi - lo) / 3
        if obj(m1) < obj(m2):
            hi = m2
        else:
            lo = m1
    
    return (lo + hi) / 2

from scipy.optimize import minimize

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)

def project_capped_simplex(h, target_sum):
    """Project h onto {0<=x<=1, sum(x)=target_sum}."""
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        x = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(x)
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2, 0.0, 1.0)

def coord_step(h, j, penalty, n, dx):
    """Optimize coordinate j of h."""
    k_range = np.arange(2 * n - 1)
    lag = k_range - (n - 1)
    
    i1 = j - lag
    valid1 = (i1 >= 0) & (i1 < n)
    i2 = j + lag
    valid2 = (i2 >= 0) & (i2 < n)
    
    delta = np.zeros(2 * n - 1)
    delta[valid1] = (1.0 - h[i1[valid1]])
    delta[valid2] -= h[i2[valid2]]
    
    # Current correlation (times dx)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    
    # New corr at x: corr_k + (x - h[j]) * delta_k * dx
    a = corr
    b = delta * dx
    
    s = np.sum(h) - h[j]
    target = n / 2.0
    hj = h[j]
    
    def obj(x):
        vals = a + b * (x - hj)
        return np.max(vals) + penalty * (s + x - target) ** 2
    
    # Ternary search on [0, 1]
    lo, hi = 0.0, 1.0
    for _ in range(80):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if obj(m1) <= obj(m2):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

                improved = True

            t0 = time.time()
            for _ in range(100):
                if time.time() - t0 > min(30.0, time_left() * 0.5):
                    break
                for j in range(best_n):
                    x_new = coord_step(h2, j, penalty, best_n, dx)
                    if abs(x_new - h2[j]) > 1e-14:
                        h2[j] = x_new
            h2 = project_capped_simplex(h2, best_n / 2.0)
            c5 = compute_c5(h2)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h2.copy(), best_n

    return best_h, best_c5, best_n
