    """Project h onto {x: 0<=x_i<=1, sum(x)=t}"""

    lo, hi = -1.0, 1.0

        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0, 1)
        s = proj.sum()
        if s > t:
            lo = mid
        else:
            hi = mid

def project_capped(h, t):

        if s > t:

        out[pos] = val
    return out
