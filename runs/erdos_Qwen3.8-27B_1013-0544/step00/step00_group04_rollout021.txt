def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

    """Use FFT for faster computation when n is large."""
    n = len(h)
    dx = 2.0 / n
    # Use FFT-based cross-correlation
    size = 2 * n
    f1 = np.fft.rfft(h, size)
    f2 = np.fft.rfft(1.0 - h, size)
    corr = np.fft.irfft(f1 * np.conj(f2), size)[:n]
    # Wait, this gives linear correlation, not full correlation
    # Let me use a different approach
    # Actually, for our purposes, we need the full correlation

    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def random_binary(n, seed=42):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    ones = rng.choice(n, n//2, replace=False)

def steepest_descent(h, max_iters=500, time_limit=30.0, seed=42):

    rng = np.random.RandomState(seed)
    h = h.copy()
    n = len(h)
    best_c5 = compute_c5(h)
    start = time.time()

        if time.time() - start > time_limit:
            break

def simulated_annealing(h, max_iters=5000, t_start=1.0, t_end=0.01, time_limit=60.0, seed=42):
    """Simulated annealing on binary sequences."""
    rng = np.random.RandomState(seed)
    h = h.copy()
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5

        if time.time() - start > time_limit:
            break
        
        T = t_start * (t_end / t_start) ** (it / max_iters)
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        
        i = ones[rng.randint(len(ones))]

        h[j] = 1.0
        new_c5 = compute_c5(h)

            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0

                    j = zeros[jj]
                    if i == j:
                        continue
                    h[i] = 0.0
                    h[j] = 1.0
                    c5 = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))
                    if c5 < best_val:
                        best_val = c5
                        best_swap = (i, j)
                    h[i] = 1.0
                    h[j] = 0.0
        else:
            n_try = min(25000, n_pairs)
            for _ in range(n_try):
                i = ones[rng.randint(n1)]
                j = zeros[rng.randint(n0)]
                if i == j:
                    continue
                h[i] = 0.0
                h[j] = 1.0
                c5 = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))
                if c5 < best_val:
                    best_val = c5
                    best_swap = (i, j)
                h[i] = 1.0
                h[j] = 0.0
        
        if best_swap is None:
            stuck += 1
            if stuck > 8:
                break
            i = ones[rng.randint(n1)]
            j = zeros[rng.randint(n0)]
            if i != j:
                h[i] = 0.0
                h[j] = 1.0
                best_c5 = compute_c5(h)
        else:
            stuck = 0
            i, j = best_swap
            h[i] = 0.0
            h[j] = 1.0
            best_c5 = best_val
    
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        h0 = random_binary(42, seed)
    
    n0 = len(h0)
    h0 = make_valid(h0, n0)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    
    # Phase 1: Steepest descent on initial and its complement
    elapsed = time.time() - t0
    t_left = budget_s - elapsed - 60
    if t_left > 30:
        h1, c5_1 = steepest_descent(h0, max_iters=5000, time_limit=min(t_left*0.5, 90), seed=seed)
        if c5_1 < best_c5:
            best_c5, best_h = c5_1, h1.copy()
        # Try complement
        h0c = 1.0 - h0
        h0c = make_valid(h0c, n0)
        h1c, c5_1c = steepest_descent(h0c, max_iters=5000, time_limit=min(t_left*0.3, 60), seed=seed+1)
        if c5_1c < best_c5:
            best_c5, best_h = c5_1c, h1c.copy()
    
    # Phase 2: Try different n with multiple starts
    elapsed = time.time() - t0
    t_left = budget_s - elapsed - 40
    if t_left > 20:
        n_candidates = [n0]
        for n in range(40, 141, 2):
            if n not in n_candidates and n % 2 == 0:
                n_candidates.append(n)
        n_candidates.sort(key=lambda x: (abs(x - n0), x))
        
        for n in n_candidates:
            elapsed = time.time() - t0
            t_left = budget_s - elapsed - 15
            if t_left < 12:
                break
            
            n_starts = min(3, max(1, int(t_left / 40)))
            for s in range(n_starts):
                elapsed = time.time() - t0
                t_left = budget_s - elapsed - 10
                if t_left < 8:
                    break
                
                if s == 0 and n == n0:
                    h_init = best_h.copy() if n == n0 else random_binary(n, seed + n)
                else:
                    h_init = random_binary(n, seed + s * 7919 + n * 31)
                
                t_this = min(30, t_left - 5)
                if t_this < 4:
                    break
                h_new, c5_new = steepest_descent(h_init, max_iters=3000, time_limit=t_this, seed=seed + s*13 + n)
                if c5_new < best_c5:
                    best_c5, best_h = c5_new, h_new.copy()
    
    # Phase 3: Final intensive polish
    elapsed = time.time() - t0
    t_left = budget_s - elapsed - 3
    if t_left > 10:
        h_fin, c5_fin = steepest_descent(best_h, max_iters=10000, time_limit=t_left, seed=seed + 9999)
        if c5_fin < best_c5:
            best_c5, best_h = c5_fin, h_fin.copy()
        # Try complement of best
        h_comp = 1.0 - best_h
        h_comp = make_valid(h_comp, len(best_h))
        elapsed = time.time() - t0
        t_left = budget_s - elapsed - 2
        if t_left > 5:
            h_cf, c5_cf = steepest_descent(h_comp, max_iters=5000, time_limit=t_left, seed=seed + 8888)
            if c5_cf < best_c5:
                best_c5, best_h = c5_cf, h_cf.copy()
    
    return best_h, best_c5, len(best_h)
