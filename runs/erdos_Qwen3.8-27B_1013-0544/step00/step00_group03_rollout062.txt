def project_to_feasible(h, target_sum):

    h = np.clip(h, 0, 1)

        if abs(diff) < 1e-12:

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            m = h < 1.0
            if not m.any():
                break
            h[m] += diff / m.sum()
            h = np.clip(h, 0.0, 1.0)
        else:
            m = h > 0.0
            if not m.any():
                break
            h[m] += diff / m.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def grad_lag(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        g[:n-k] += 1.0 - h[k:]
        g[k:] -= h[:n-k]
    else:
        kk = -k
        g[kk:] += 1.0 - h[:n-kk]
        g[:n-kk] -= h[kk:]
    return g

    n = len(h)

    vals = corr * dx

        wm = w[m]
        if wm < 1e-8:
            continue
        k = m - (n - 1)
        grad += wm * grad_lag(h, k)
    return grad * dx
