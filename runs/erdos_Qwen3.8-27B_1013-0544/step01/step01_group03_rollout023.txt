    x_max = np.max(x)

    dx = 2.0 / n

    return float(np.max(corr) * dx)

def smooth_max_val_and_grad(obj, tau):
    x = tau * obj
    x_max = np.max(x)
    exp_x = np.exp(x - x_max)
    sum_exp = np.sum(exp_x)
    val = (x_max + np.log(sum_exp)) / tau
    weights = exp_x / sum_exp
    return val, weights

def proj_feasible(h, target):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h_proj.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h_proj < 1.0 - 1e-12
        else:
            mask = h_proj > 1e-12
        cnt = mask.sum()
        if cnt > 0:
            h_proj[mask] += diff / cnt
    return np.clip(h_proj, 0.0, 1.0)

def corr_grad_at_shift(h, n, s):
    grad = np.zeros(n)
    if s >= 0:
        m = n - s
        if m <= 0:
            return 0.0, grad
        val = np.dot(h[:m], 1.0 - h[s:])
        grad[:m] += (1.0 - h[s:s + m])
        grad[s:n] += -h[:m]
    else:
        sp = -s
        m = n - sp
        if m <= 0:
            return 0.0, grad
        val = np.dot(h[sp:], 1.0 - h[:m])
        grad[sp:] += (1.0 - h[:m])
        grad[:m] += -h[sp:]
    return val, grad

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s * 0.93

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(120, 0.5)

    n0 = len(h_init)
    target0 = n0 / 2.0
    h_init = proj_feasible(h_init, target0)
    best_h = h_init.copy()
    best_c5 = evaluate_c5(best_h)

    def rem():
        return deadline - time.time()

    def subgrad(h_start, n, iters, M, lr0):
        if rem() < 5:
            return h_start, evaluate_c5(h_start)
        target = n / 2.0
        h = h_start.copy()
        bh = h.copy()
        bc5 = evaluate_c5(bh)
        momentum = np.zeros(n)
        for it in range(iters):
            if (it + 1) % 300 == 0 and rem() < 3:
                break
            corr = np.correlate(h, 1.0 - h, mode="full")
            top_idx = np.argsort(corr)[-M:]
            grad = np.zeros(n)
            for k in top_idx:
                s = k - (n - 1)
                _, g = corr_grad_at_shift(h, n, s)
                grad += g
            grad /= M
            lr = lr0 / (1.0 + 0.005 * it)
            beta = 0.3
            momentum = beta * momentum + (1 - beta) * grad
            h_new = h - lr * momentum
            h_new = proj_feasible(h_new, target)
            c5_new = evaluate_c5(h_new)
            if c5_new < bc5:
                bc5 = c5_new
                bh = h_new.copy()
            h = h_new
        return bh, bc5

    def slsqp(h_start, n, tau, maxiter):
        if rem() < 5:
            return h_start, evaluate_c5(h_start)
        target = n / 2.0
        bnds = Bounds(np.zeros(n), np.ones(n))

        def obj(x):
            corr = np.correlate(x, 1.0 - x, mode="full")
            val, _ = smooth_max_val_and_grad(corr, tau)
            return val

        def obj_grad(x):
            corr = np.correlate(x, 1.0 - x, mode="full")
            val, weights = smooth_max_val_and_grad(corr, tau)
            grad = np.zeros(n)
            thresh = 1e-8
            for k in range(2 * n - 1):
                if weights[k] > thresh:
                    s = k - (n - 1)
                    _, g = corr_grad_at_shift(x, n, s)
                    grad += weights[k] * g
            return val, grad

        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                 'jac': lambda x: np.ones(n)}]
        try:
            res = minimize(obj_grad, h_start, method='SLSQP', bounds=bnds,
                          constraints=cons, jac=True,
                          options={'maxiter': maxiter, 'ftol': 1e-12})
            x = proj_feasible(res.x, target)
            c5 = evaluate_c5(x)
            return x, c5
        except Exception:
            return h_start, evaluate_c5(h_start)

    # Phase 1: Subgradient with increasing M (focus then balance)
    for M, its, lr in [(3, 3000, 0.03), (5, 3000, 0.02), (10, 3000, 0.015), (20, 2000, 0.01)]:
        if rem() < 10:
            break
        h_sg, c5_sg = subgrad(best_h, n0, its, M, lr)
        if c5_sg < best_c5:
            best_h = h_sg
            best_c5 = c5_sg

    # Phase 2: SLSQP tau annealing
    tau_schedule = [5, 20, 50, 100, 300, 1000, 3000, 8000]
    for tau in tau_schedule:
        if rem() < 12:
            break
        h_new, c5_new = slsqp(best_h, n0, tau, 300)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 3: Multi-resolution exploration
    res_list = [60, 80, 100, 150, 200, 240]
    for tn in res_list:
        if rem() < 50:
            break
        target_t = tn / 2.0
        h_t = np.interp(np.linspace(0, 1, tn), np.linspace(0, 1, len(best_h)), best_h)
        h_t = proj_feasible(h_t, target_t)

        h_sg2, c5_sg2 = subgrad(h_t, tn, 1500, M=5, lr0=0.02)
        if c5_sg2 < best_c5:
            best_h = h_sg2
            best_c5 = c5_sg2

        for tau in [50, 200, 1000, 5000]:
            if rem() < 10:
                break
            hn = len(best_h)
            h_new, c5_new = slsqp(best_h, hn, tau, 200)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    # Phase 4: Perturbation and quick refinement
    hn = len(best_h)
    for trial in range(40):
        if rem() < 8:
            break
        scale = 0.03 + 0.15 * np.random.random()
        hp = best_h + np.random.normal(0, scale, hn)
        hp = proj_feasible(hp, hn / 2.0)
        h_new, c5_new = slsqp(hp, hn, 500, 150)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 5: Binary-like restarts
    if rem() > 30:
        for _ in range(3):
            if rem() < 15:
                break
            idx = np.argsort(best_h)
            h_bin = np.zeros(hn)
            h_bin[idx[int(np.ceil(hn / 2.0)):]] = 1.0
            # Add noise to break symmetry
            h_bin = h_bin + np.random.normal(0, 0.1, hn)
            h_bin = proj_feasible(h_bin, hn / 2.0)
            h_new, c5_new = slsqp(h_bin, hn, 1000, 300)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    # Phase 6: Final high-tau refinement
    for tau in [5000, 10000, 20000]:
        if rem() < 8:
            break
        hn = len(best_h)
        h_new, c5_new = slsqp(best_h, hn, tau, 250)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Final feasibility
    hn_final = len(best_h)
    best_h = proj_feasible(best_h, hn_final / 2.0)
    best_c5 = evaluate_c5(best_h)

    return best_h, best_c5, hn_final
