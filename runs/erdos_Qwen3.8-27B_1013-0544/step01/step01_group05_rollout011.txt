For the smooth max approach:

- This is always ≥ max(corr) * dx, and approaches it as T → ∞

_g_target = None

def project_feasible(x, target_sum):
    lo = float(x.min()) - 1.0
    hi = float(x.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = float(np.clip(x - mid, 0.0, 1.0).sum())
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def soft_max_val(h, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * _g_dx
    m = vals.max()
    return m + np.log(np.sum(np.exp(T * (vals - m)))) / T

def soft_max_grad(h, T):
