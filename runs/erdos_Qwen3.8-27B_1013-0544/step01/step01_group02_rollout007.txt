def compute_smoothed_gradient(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    weights = np.exp(shifted - mx)
    weights /= weights.sum()
    
    grad = np.zeros(n)
    n_lags = 2*n - 1
    for k in range(n_lags):
        if weights[k] < 1e-15:
            continue
        s = n - 1 - k
        for j in range(n):
            if 0 <= j+s < n:
                grad[j] += weights[k] * (1 - h[j+s])
            if 0 <= j-s < n:
                grad[j] -= weights[k] * h[j-s]
    return grad

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-15:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:n//2]] = 1.0
    return h_bin

def obj_smooth(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 700:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def sum_con(h, n):
    return np.sum(h) - n / 2.0

def sum_con_jac(h, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, bounds, cons, maxiter=5000):
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h

def binary_local_search(h_bin, n, rng, max_iters=100000, time_limit=None, t0=None):
    best = h_bin.copy()
    best_c = compute_c5(best, n)
    current = h_bin.copy()
    current_c = best_c
    
    for it in range(max_iters):
        if time_limit is not None and t0 is not None:
            if time.time() - t0 > time_limit:
                break
        
        i = rng.randint(n)
        j = rng.randint(n)
        if current[i] == current[j]:
            continue
        
        h_try = current.copy()
        h_try[i], h_try[j] = current[j], current[i]
        c = compute_c5(h_try, n)
        
        if c <= current_c + 1e-15:  # Accept equal or better (to allow exploration)
            current = h_try
            current_c = c
            if c < best_c - 1e-15:
                best = h_try
                best_c = c
    
    return best, best_c
