    return h

        (5000, 0.01, 0.9997, 1, 5, 0.0),
        (8000, 0.008, 0.9998, 3, 10, 0.3),
        (10000, 0.005, 0.99985, 5, 20, 0.5),
        (8000, 0.003, 0.9999, 10, 50, 0.7),
        (5000, 0.002, 0.9999, 20, 100, 0.8),
    ]
    for ci, (n_it, a0, dec, tk0, tk1, mom) in enumerate(configs):
        if elapsed() > budget_s - 200:
            break
        h_new, c5_new = subgrad_run(best_h, dx, target_sum, n_it, a0, dec, tk0, tk1, mom, seed=seed + ci)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 2: SLSQP polishing with increasing temperature
    for t_val in [10, 30, 80, 200]:
        if elapsed() > budget_s - 120:
            break
        h_new = slsqp_polish(best_h, dx, target_sum, t=t_val, maxiter=200)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 3: Local pair adjustment
    if elapsed() < budget_s - 80:
        h_new, c5_new = local_adjust(best_h.copy(), dx, target_sum, n_rounds=500, seed=seed)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 4: More subgradient from current best
    if elapsed() < budget_s - 60:
        for ci in range(3):
            if elapsed() > budget_s - 40:
                break
            h_new, c5_new = subgrad_run(best_h, dx, target_sum, 5000, 0.004, 0.9998, 1, 8, 0.4, seed=seed + 100 + ci)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    # Phase 5: SLSQP again
    if elapsed() < budget_s - 30:
        h_new = slsqp_polish(best_h, dx, target_sum, t=300, maxiter=500)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 6: Try n=80 (coarser)
    if elapsed() < budget_s - 15:
        n2 = 80
        dx2 = 2.0 / n2
        ts2 = n2 / 2.0
        h2 = downsample(best_h, n2)
        h2 = project_feasible(h2, ts2)
        c5_2 = compute_c5(h2, dx2)
        if c5_2 < best_c5:
            h2, c5_2 = subgrad_run(h2, dx2, ts2, 3000, 0.01, 0.9997, 1, 10, 0.3, seed=seed + 200)
            h2 = slsqp_polish(h2, dx2, ts2, t=50, maxiter=200)
            c5_2 = compute_c5(h2, dx2)
            if c5_2 < best_c5:
                best_h = h2
                best_c5 = c5_2
                best_n = n2

    # Phase 7: Try structured initializations
    if elapsed() < budget_s - 10:
        for pat in ['triangle', 'sine']:
            if elapsed() > budget_s - 5:
                break
            hs = make_structured_init(n, seed + 50, pat)
            hs, c5_s = subgrad_run(hs, dx, target_sum, 3000, 0.01, 0.9997, 1, 10, 0.3, seed=seed + 300)
            hs = slsqp_polish(hs, dx, target_sum, t=50, maxiter=100)
            c5_s = compute_c5(hs, dx)
            if c5_s < best_c5:
                best_h = hs
                best_c5 = c5_s

    # Final polish
    if elapsed() < budget_s - 2:
        h_new = slsqp_polish(best_h, dx, target_sum, t=500, maxiter=1000)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_feasible(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h, 2.0 / best_n)
    return best_h, best_c5, best_n
