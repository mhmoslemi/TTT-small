            h[mask] += diff / np.sum(mask)

    if elapsed() < budget_s - 80:
        for thresh in [0.08, 0.12, 0.18, 0.25, 0.35]:
            if elapsed() > budget_s - 40:
                break
            h_pushed = best_h.copy()
            h_pushed[h_pushed < thresh] = 0.0
            h_pushed[h_pushed > 1.0 - thresh] = 1.0
            h_pushed = fix_sum(h_pushed, target_sum)
            obj = lambda x: smooth_max_val(x, 200, dx)
            h_new = slsqp_run(h_pushed, dx, target_sum, obj, maxiter=500)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    # Phase 6: Random perturbations + reoptimize
    if elapsed() < budget_s - 40:
        for trial in range(80):
            if elapsed() > budget_s - 20:
                break
            h_trial = best_h.copy()
            n_pert = rng.randint(3, 12)
            idx = rng.choice(n, n_pert, replace=False)
            h_trial[idx] += rng.uniform(-0.3, 0.3, n_pert)
            h_trial = fix_sum(h_trial, target_sum)
            if trial % 3 == 0:
                obj = lambda x: smooth_max_val(x, 80, dx)
            elif trial % 3 == 1:
                obj = lambda x: topk_val(x, 80, dx, 5)
            else:
                obj = lambda x: topk_val(x, 80, dx, 10)
            h_new = slsqp_run(h_trial, dx, target_sum, obj, maxiter=350)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    # Phase 7: Try n=320 upsampling
    if elapsed() < budget_s - 120:
        n2 = 2 * n
        dx2 = 2.0 / n2
        ts2 = n2 / 2.0
        h2 = upsample(best_h, n2)
        h2 = fix_sum(h2, ts2)
        for t in [30, 100, 400]:
            if elapsed() > budget_s - 60:
                break
            obj = lambda x, tt=t: smooth_max_val(x, tt, dx2)
            h2_new = slsqp_run(h2, dx2, ts2, obj, maxiter=400)
            c5_2 = compute_c5(h2_new, dx2)
            if c5_2 < compute_c5(h2, dx2):
                h2 = h2_new
        c5_2 = compute_c5(h2, dx2)
        if c5_2 < best_c5:
            best_h = h2
            best_c5 = c5_2
            best_n = n2

    # Phase 8: Final high-t polish
    if elapsed() < budget_s - 10:
        dd = 2.0 / best_n
        ts = best_n / 2.0
        for t in [3000, 20000]:
            if elapsed() > budget_s - 5:
                break
            obj = lambda x, tt=t: smooth_max_val(x, tt, dd)
            h_final = slsqp_run(best_h, dd, ts, obj, maxiter=1500)
            c5_final = compute_c5(h_final, dd)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    dx_final = 2.0 / best_n
    best_h = fix_sum(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h, dx_final)

    return best_h, best_c5, best_n
