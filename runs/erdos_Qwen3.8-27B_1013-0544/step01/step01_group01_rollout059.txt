def project_h(h, target_sum):
    """Euclidean projection onto {x: 0<=x_i<=1, sum(x)=target_sum}"""
    h = np.asarray(h, dtype=float)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _smooth_max_weights(corr, beta):
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -100, 100)
    exps = np.exp(shifted)
    total = np.sum(exps)
    weights = exps / total
    val = m + np.log(total) / beta
    return val, weights

def _compute_obj(h, beta, penalty, target_sum):
    n = len(h)
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1.0 - h, mode="full")
    corr = corr_raw * dx
    sm_val, weights = _smooth_max_weights(corr, beta)
    sum_diff = h.sum() - target_sum
    return sm_val + penalty * sum_diff * sum_diff, weights

class _OptFunc:
    def __init__(self, beta, penalty, target_sum):
        self.beta = beta
        self.penalty = penalty
        self.target_sum = target_sum
    
    def __call__(self, h):
        val, _ = _compute_obj(h, self.beta, self.penalty, self.target_sum)
        return val
    
    def gradient(self, h):
        n = len(h)
        dx = 2.0 / n
        corr_raw = np.correlate(h, 1.0 - h, mode="full")
        corr = corr_raw * dx
        sm_val, weights = _smooth_max_weights(corr, self.beta)
        
        grad = np.zeros(n)
        shifts = np.arange(-(n-1), n)
        for i in range(2*n - 1):
            s = shifts[i]
            w = weights[i] * dx
            j_lo = max(0, s)
            j_hi = min(n, s + n)
            if j_lo < j_hi:
                grad[j_lo:j_hi] += w * (1.0 - h[j_lo - s:j_hi - s])
            j_lo2 = max(0, -s)
            j_hi2 = min(n, n - s)
            if j_lo2 < j_hi2:
                grad[j_lo2:j_hi2] -= w * h[j_lo2 + s:j_hi2 + s]
        
        sum_diff = h.sum() - self.target_sum
        grad += 2.0 * self.penalty * sum_diff
        
        return grad

def _do_optimize(h0, n, beta, penalty, target_sum, max_iter):
    opt = _OptFunc(beta, penalty, target_sum)
    bounds = [(0.0, 1.0)] * n
    res = minimize(opt, h0, jac=opt.gradient, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-8})
    return project_h(res.x, target_sum)
