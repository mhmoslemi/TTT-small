import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -3.0, 3.0
    for _ in range(80):
        tau = (lo + hi) / 2.0
        proj = np.clip(h - tau, 0.0, 1.0)
        if proj.sum() > target:
            lo = tau
        else:
            hi = tau
    tau = (lo + hi) / 2.0
    result = np.clip(h - tau, 0.0, 1.0)
    diff = target - result.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            room = 1.0 - result
            order = np.argsort(-room)
            for i in order:
                add = min(diff, room[i])
                if add <= 0:
                    break
                result[i] += add
                diff -= add
                if abs(diff) < 1e-12:
                    break
        else:
            order = np.argsort(result)
            for i in order:
                sub = min(-diff, result[i])
                if sub <= 0:
                    break
                result[i] -= sub
                diff += sub
                if abs(diff) < 1e-12:
                    break
    return result

def build_W(w, n):
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(n)[None, :]
    indices = j_idx - m_idx + n - 1
    return w[indices]

def smooth_obj(h, n, alpha):
    c = compute_c5_vec(h, n)
    mc = c.max()
    shifted = alpha * (c - mc)
    return mc + np.log(np.sum(np.exp(shifted))) / alpha

def smooth_grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    mc = c.max()
    shifted = alpha * (c - mc)
    w = np.exp(shifted)
    w = w / w.sum()
    dx = 2.0 / n
    W = build_W(w, n)
    s1 = W @ (1.0 - h)
    s2 = W.T @ h
    return dx * (s1 - s2)

def projected_grad_descent(h, n, alpha, steps, lr_init, lr_decay, t0, budget_s):
    h = h.copy()
    lr = lr_init
    best_c5 = compute_c5(h, n)
    best_h = h.copy()
    stall = 0
    for step in range(steps):
        if budget_s - (time.time() - t0) < 5:
            break
        g = smooth_grad(h, n, alpha)
        h_new = h - lr * g
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new, n)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            stall = 0
        else:
            stall += 1
            if stall % 20 == 0:
                lr *= lr_decay
        if lr < 1e-10:
            break
    return best_h, best_c5

def slsqp_opt(h, n, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        h_res = project_feasible(res.x, n)
        return h_res, compute_c5(h_res, n)
    except Exception:

def coordinate_descent(h, n, rng, max_iters, t0, budget_s):
