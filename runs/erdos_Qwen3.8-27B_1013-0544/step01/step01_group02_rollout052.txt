        b_new[i], b_new[j] = b[j], b[i]
        c_new = compute_c5(b_new, n)
        if c_new < best_c:
            b = b_new
            best_b = b_new.copy()
            best_c = c_new
            T = max(T * 0.95, 0.001)
        else:
            if rng.random() < np.exp(-(c_new - best_c) / T):
                b = b_new
        T = max(T * 0.998, 0.001)
    # Polish binary -> continuous
    h_cont = best_b + 0.05 * rng.randn(n)
    h_cont = project_feasible(h_cont, n)
    return h_cont

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - t0)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 80
        h0 = np.full(n, 0.5)

    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, n)

    # Phase 1: SLSQP continuation
    for alpha in [50, 100, 300, 1000, 3000, 10000, 30000, 100000]:
        if remaining() < 40:
            break
        h_try = slsqp(best_h, n, alpha, maxiter=5000)
        c = compute_c5(h_try, n)
        if c < best_c5:
            best_h, best_c5 = h_try, c

    # Phase 2: Coordinate descent
    if remaining() > 60:
        h_try = coord_descent(best_h, n, max_sweeps=10)
        c = compute_c5(h_try, n)
        if c < best_c5:
            best_h, best_c5 = h_try, c
        # SLSQP after coord descent
        if remaining() > 40:
            h_try = slsqp(best_h, n, 20000, maxiter=5000)
            c = compute_c5(h_try, n)
            if c < best_c5:
                best_h, best_c5 = h_try, c

    # Phase 3: Multi-start exploration
    if remaining() > 200:
        starts = []
        # Start 1: Gaussian perturbation
        starts.append(project_feasible(best_h + 0.15 * rng.randn(n), n))
        # Start 2: Binary + noise
        b = (best_h > 0.5).astype(float)
        starts.append(project_feasible(b + 0.1 * rng.randn(n), n))
        # Start 3: Sinusoidal structure
        x = np.arange(n)
        h3 = 0.5 + 0.3 * np.sin(2 * np.pi * 3 * x / n)
        starts.append(project_feasible(h3, n))
        # Start 4: Block structure
        h4 = np.zeros(n)
        block = n // 3
        h4[block // 2:2 * block // 2] = 1.0
        h4[block // 2 + n // 4:2 * block // 2 + n // 4] = 0.5
        starts.append(project_feasible(h4 + 0.1 * rng.randn(n), n))
        # Start 5: Perturbed best
        starts.append(project_feasible(best_h + 0.3 * rng.randn(n), n))

        for h_start in starts:
            if remaining() < 80:
                break
            h_try = slsqp(h_start, n, 500, maxiter=3000)
            h_try = slsqp(h_try, n, 5000, maxiter=3000)
            h_try = coord_descent(h_try, n, max_sweeps=5)
            h_try = slsqp(h_try, n, 20000, maxiter=3000)
            c = compute_c5(h_try, n)
            if c < best_c5:
                best_h, best_c5 = h_try, c

    # Phase 4: Binary skeleton SA + continuous polish
    if remaining() > 150:
        h_sa = binary_sa(best_h, n, rng, remaining, min_remaining=50)
        h_try = slsqp(h_sa, n, 1000, maxiter=3000)
        h_try = slsqp(h_try, n, 10000, maxiter=5000)
        h_try = coord_descent(h_try, n, max_sweeps=5)
        c = compute_c5(h_try, n)
        if c < best_c5:
            best_h, best_c5 = h_try, c

    # Phase 5: Iterated perturbation + coordinate descent + SLSQP
    while remaining() > 50:
        scale = rng.choice([0.01, 0.02, 0.05, 0.1, 0.2])
        h_pert = project_feasible(best_h + scale * rng.randn(n), n)
        h_try = coord_descent(h_pert, n, max_sweeps=3)
        if remaining() > 30:
            h_try = slsqp(h_try, n, 5000, maxiter=2000)
        c = compute_c5(h_try, n)
        if c < best_c5 - 1e-14:
            best_h, best_c5 = h_try, c

    # Phase 6: Fine 2-point swaps
    if remaining() > 20:
        for scale in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]:
            if remaining() < 15:
                break
            for _ in range(3000):
                if remaining() < 15:
                    break
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                for s in [1, -1]:
                    d = s * scale
                    hi, hj = best_h[i] + d, best_h[j] - d
                    if 0 <= hi <= 1 and 0 <= hj <= 1:
                        h_try = best_h.copy()
                        h_try[i], h_try[j] = hi, hj
                        c = compute_c5(h_try, n)
                        if c < best_c5:
                            best_h, best_c5 = h_try, c

    # Phase 7: Final polish
    if remaining() > 15:
        for alpha in [5000, 50000, 200000]:
            if remaining() < 15:
                break
            h_try = slsqp(best_h, n, alpha, maxiter=10000)
            c = compute_c5(h_try, n)
            if c < best_c5:
                best_h, best_c5 = h_try, c
        h_try = coord_descent(best_h, n, max_sweeps=3)
        c = compute_c5(h_try, n)
        if c < best_c5:
            best_h, best_c5 = h_try, c

    # Phase 8: Try alternate resolutions
    for alt_n in [64, 96, 128]:
        if remaining() < 40:
            break
        x_old = np.arange(n) + 0.5
        x_new = np.arange(alt_n) * (n / float(alt_n)) + 0.5
        h_new = np.interp(x_new, x_old, best_h)
        h_new = project_feasible(h_new, alt_n)
        h_try = slsqp(h_new, alt_n, 1000, maxiter=3000)
        h_try = slsqp(h_try, alt_n, 10000, maxiter=5000)
        h_try = coord_descent(h_try, alt_n, max_sweeps=5)
        c = compute_c5(h_try, alt_n)
        if c < best_c5:
            best_h, best_c5, n = h_try, c, alt_n

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
