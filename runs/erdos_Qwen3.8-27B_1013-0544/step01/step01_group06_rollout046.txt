                valid = np.arange(p, n)
                bad_mask = np.zeros(n, dtype=bool)
                bad_mask[valid] = h[valid] == 1
                bad_mask[p:] = bad_mask[p:] & (h[:n-p] == 0)
                bad_i = np.where(bad_mask)[0]

            if len(bad_i) == 0:
                continue
            i = rng.choice(bad_i)
            j_mask = np.array([j for j in zeros if j != i])
            if len(j_mask) == 0:
                continue
            j = rng.choice(j_mask)

            h[i] = 0
            h[j] = 1
            new_max = np.max(np.correlate(h, 1 - h, mode='full'))
            if new_max < current_max:
                found = True
                current_max = new_max
                c5 = new_max * (2.0 / n)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                break
            else:
                h[i] = 1
                h[j] = 0

        if found:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 12:
                n_swap = rng.randint(3, 10)
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                for _ in range(n_swap):
                    if len(ones) == 0 or len(zeros) == 0:
                        break
                    i = rng.choice(ones)
                    j = rng.choice(zeros)
                    h[i] = 0
                    h[j] = 1
                no_improve = 0

    return best_h, best_c5

def interpolate_to_n(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_ext = np.concatenate([h_old, [h_old[0]]])
    x_ext = np.linspace(0.0, 2.0, n_old + 1, endpoint=False)
    return np.interp(x_new, x_ext, h_ext)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        n0 = 200
        h0 = 0.5 * np.ones(n0)

    n = len(h0)
    target = n / 2.0
    h0 = project_simplex(h0, target)

    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    best_n = n

    # Phase 1: Adam at native resolution
    dl1 = t0 + budget_s * 0.30
    h_opt, c5_opt = adam_optimize(h0, target, dl1, rng, 100.0, 0.01)
    if c5_opt < best_c5:
        best_h, best_c5, best_n = h_opt, c5_opt, n

    # Phase 2: Binary local search
    dl2 = t0 + budget_s * 0.50
    h_bin = threshold_binary(best_h, best_n // 2)
    h_bin_opt, c5_bin = binary_local_search(h_bin, dl2, rng)
    if c5_bin < best_c5:
        best_h, best_c5, best_n = h_bin_opt, c5_bin, best_n

    # Phase 3: De-binary + optimize
    dl3 = t0 + budget_s * 0.70
    if time.time() < dl3:
        h_de = best_h + rng.randn(best_n) * 0.08
        h_de = project_simplex(h_de, best_n / 2.0)
        h_opt3, c5_opt3 = adam_optimize(h_de, best_n / 2.0, dl3, rng, 150.0, 0.005)
        if c5_opt3 < best_c5:
            best_h, best_c5, best_n = h_opt3, c5_opt3, best_n

    # Phase 4: Try other resolutions
    remaining = budget_s - (time.time() - t0) - 15
    for n_try in [150, 250, 300]:
        if remaining < 50:
            break
        if n_try == best_n:
            continue
        h_interp = interpolate_to_n(best_h, n_try)
        h_interp = project_simplex(h_interp, n_try / 2.0)
        phase_dl = time.time() + min(remaining * 0.35, 120.0)
        h_opt4, c5_opt4 = adam_optimize(h_interp, n_try / 2.0, phase_dl, rng, 100.0, 0.01)
        if c5_opt4 < best_c5:
            best_h, best_c5, best_n = h_opt4, c5_opt4, n_try
        h_b4 = threshold_binary(h_opt4, n_try // 2)
        if time.time() < t0 + budget_s - 30:
            h_b4, c5_b4 = binary_local_search(h_b4, min(t0 + budget_s - 30, time.time() + 40), rng)
            if c5_b4 < best_c5:
                best_h, best_c5, best_n = h_b4, c5_b4, n_try
        remaining = budget_s - (time.time() - t0) - 20

    # Phase 5: Final polish
    if time.time() < t0 + budget_s - 5:
        h_fin = best_h + rng.randn(best_n) * 0.02
        h_fin = project_simplex(h_fin, best_n / 2.0)
        h_fin, c5_fin = adam_optimize(h_fin, best_n / 2.0, t0 + budget_s - 3, rng, 300.0, 0.003)
        if c5_fin < best_c5:
            best_h, best_c5, best_n = h_fin, c5_fin, best_n

    best_h = project_simplex(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
