import numpy as np
import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def grad_corr_single(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n - k])
        if k > 0:
            g[:n - k] -= h[k:]
    else:
        p = -k
        g[:n - p] += (1.0 - h[p:])
        if p > 0:
            g[p:] -= h[:n - p]
    return g

def project_sum_box(h, target_sum):
    lo, hi = -1.0, 2.0
    for _ in range(55):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_grad_topk(h, k_top):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(corr)[-k_top:]
    grad = np.zeros(n)
    for i in idx:
        k = i - (n - 1)
        grad += grad_corr_single(h, k)
    return (grad / float(len(idx))) * (2.0 / n)

def compute_grad_smooth(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    s = np.sum(w)
    if s < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= s
    grad = np.zeros(n)
    active = np.where(w > 1e-8)[0]
    for i in active:
        k = i - (n - 1)
        grad += w[i] * grad_corr_single(h, k)
    return grad * (2.0 / n)

def optimize_nesterov(h0, target_sum, deadline, rng, k_top=5, use_smooth=False, alpha=100):
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    lr = 0.025
    mu = 0.9
    stall = 0
    pert_scale = 0.02
    iter_count = 0

    while time.time() < deadline:
        iter_count += 1
        if use_smooth:
            h_look = project_sum_box(h + mu * v, target_sum)
            grad = compute_grad_smooth(h_look, alpha)
        else:
            h_look = project_sum_box(h + mu * v, target_sum)
            grad = compute_grad_topk(h_look, k_top)

        v = mu * v - lr * grad
        h_new = project_sum_box(h + v, target_sum)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            h = h_new
            stall = 0
            if lr < 0.05:
                lr = min(lr * 1.012, 0.05)
        else:
            h = h_new
            stall += 1
            if stall > 25:
                lr *= 0.85
            if stall > 80:
                scale = min(pert_scale * (1.0 + 0.02 * stall), 0.12)
                h = project_sum_box(best_h + rng.randn(n) * scale, target_sum)
                v = np.zeros(n)
                lr = 0.025
                mu = 0.85
                stall = 0
                pert_scale *= 1.4
                if pert_scale > 0.2:
                    break

    return best_h, best_c5

def interpolate_to_n(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False) + 1.0 / n_new
    h_ext = np.append(h_old, h_old[0])
    x_ext = np.append(x_old, 2.0 + 1.0 / n_old)
    return np.interp(x_new, x_ext, h_ext)
