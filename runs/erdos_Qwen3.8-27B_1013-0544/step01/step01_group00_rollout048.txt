    grad = dx * (1.0 - A - B)
    
    # Add penalty gradient
    if mu > 0:
        grad += 2.0 * mu * (np.sum(h) - n * 0.5)

        h_up = interp_h(h_best, n_new)
        c5_up = compute_c5(h_up)
        temps_up = [0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
        if tl() > 250:
            h_new, c5_new = optimize_lse(h_up, temps_up, maxiter=200, mu_start=1e3)
            if c5_new < c5_up:
                h_up = h_new
                c5_up = c5_new
        if tl() > 150:
            h_new, c5_new = optimize_t(h_up, max_iter=800, ftol=1e-12)
            if c5_new < c5_up:
                h_up = h_new
                c5_up = c5_new
        if c5_up < c5:
            h_best = h_up.copy()
            c5 = c5_up

    # Phase 4: Additional LSE refinement at best resolution
    n_cur = len(h_best)
    if tl() > 200:
        h_ref = h_best.copy()
        temps_r = [0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005, 0.0002]
        h_new, c5_new = optimize_lse(h_ref, temps_r, maxiter=300, mu_start=5e3)
        if c5_new < c5:
            h_best = h_new.copy()
            c5 = c5_new

    # Phase 5: t-variable final polish
    if tl() > 120:
        h_new, c5_new = optimize_t(h_best, max_iter=1500, ftol=1e-14)
        if c5_new < c5:
            h_best = h_new.copy()
            c5 = c5_new

    # Phase 6: Random restarts
    n_best = len(h_best)
    for i in range(15):
        if tl() < 50:
            break
        scale = 0.03 + 0.01 * (i % 5)
        h_pert = h_best + rng.randn(n_best) * scale
        h_pert = project_h(h_pert, n_best * 0.5)
        temps_s = [0.04, 0.02, 0.01, 0.005, 0.002]
        if tl() > 35:
            h_new, c5_new = optimize_lse(h_pert, temps_s, maxiter=150, mu_start=1e4)
            if c5_new < c5:
                h_best = h_new.copy()
                c5 = c5_new

    # Phase 7: Final t-variable squeeze
    if tl() > 60:
        h_new, c5_new = optimize_t(h_best, max_iter=2000, ftol=1e-15)
        if c5_new < c5:
            h_best = h_new.copy()
            c5 = c5_new

    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = project_h(h_best, len(h_best) * 0.5)
    c5_final = compute_c5(h_best)
    return h_best, c5_final, len(h_best)
