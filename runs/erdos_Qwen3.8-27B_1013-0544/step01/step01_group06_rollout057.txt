        w_full = np.exp(alpha * (corr - m_v))

            grad[:n-p] += wk * (1.0 - h[p:])

import numpy as np
import time
from itertools import combinations

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_box_sum(h, target_sum):
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def solve_binary_blocks(m, s, deadline):
    n = m * s
    half = m // 2
    combos = list(combinations(range(m), half))
    N = len(combos)
    if N == 0:
        return None, np.inf
    patterns = np.zeros((N, m))
    for i, c in enumerate(combos):
        patterns[i, list(c)] = 1.0
    
    # Precompute P, wk, wk_neg
    P = np.zeros((n - 1, m, m), dtype=np.int32)
    wk = np.zeros((n - 1, m), dtype=np.float64)
    wk_neg = np.zeros((n - 1, m), dtype=np.float64)
    
    for k in range(1, n):
        for p in range(n - k):
            P[k-1, p // s, (p + k) // s] += 1
        for i in range(m):
            wk[k-1, i] = max(0, min((i+1)*s, n-k) - i*s)
            wk_neg[k-1, i] = max(0, (i+1)*s - max(i*s, k))
    
    max_corr = np.zeros(N)
    for k in range(1, n):
        if time.time() > deadline:
            return None, np.inf
        Pk = P[k-1].astype(np.float64)
        temp = Pk @ patterns.T  # (m, N)
        quad = (patterns * temp).sum(axis=0)  # (N,)
        pos = patterns @ wk[k-1] - quad
        neg = patterns @ wk_neg[k-1] - quad
        max_corr = np.maximum(max_corr, np.maximum(pos, neg))
    
    c5_all = max_corr * (2.0 / n)
    idx = np.argmin(c5_all)
    return patterns[idx], c5_all[idx]
