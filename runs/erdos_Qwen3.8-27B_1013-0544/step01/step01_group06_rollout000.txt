import numpy as np
import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr)) * (2.0 / n)

def project(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_grad(h, alpha=100.0):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    sw = np.sum(w)
    if sw < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= sw
    grad = np.zeros(n)
    active = np.where(w > 1e-8)[0]
    for idx in active:
        k = int(idx) - (n - 1)
        wi = w[idx]
        if k >= 0:
            grad[:n - k] += wi * (1.0 - h[k:])
            if k > 0:
                grad[k:] -= wi * h[:n - k]
        else:
            p = -k
            grad[p:] += wi * (1.0 - h[:n - p])
            if p > 0:
                grad[:n - p] -= wi * h[p:]
    return grad * (2.0 / n)

def optimize(h0, deadline, rng, alpha0=80.0, lr0=0.008, max_stall=10):
    n = len(h0)
    target = n * 0.5
    best = project(h0.astype(np.float64), target)
    best_c5 = compute_c5(best)
    lr = lr0
    alpha = alpha0
    pat = 0
    stall = 0
    while time.time() < deadline:
        g = compute_grad(best, alpha)
        h_new = project(best - lr * g, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best = h_new
            best_c5 = c5_new
            pat = 0
            if lr < 0.1:
                lr *= 1.015
        else:
            pat += 1
            lr *= 0.94
            if pat > 40:
                sz = 0.015 * (1.0 + 0.35 * stall)
                hp = best + rng.randn(n) * sz
                hp = project(hp, target)
                if compute_c5(hp) < best_c5:
                    best = hp
                    best_c5 = compute_c5(hp)
                alpha = min(alpha * 1.5, 800.0)
                lr = lr0
                pat = 0
                stall += 1
                if stall > max_stall:
                    break
    return best, best_c5

def swap_local_search(h, n_swaps=200, rng=None):
    """Try random pair swaps to reduce C5."""
    n = len(h)
    target = n * 0.5
    best = h.copy()
    best_c5 = compute_c5(best)
    if rng is None:
        rng = np.random.RandomState(0)
    for _ in range(n_swaps):
        i, j = rng.choice(n, 2, replace=False)
        if abs(best[i] - best[j]) < 1e-6:
            continue
        swap = best[i].copy()
        best[i] = best[j]
        best[j] = swap
        c5 = compute_c5(best)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
        else:
            best[j] = best[i]
            best[i] = swap
    return best, best_c5

def make_block_start(n, rng):
    h = np.zeros(n)
    num_blocks = rng.randint(2, 5)
    block_len = n // (2 * num_blocks)
    for i in range(num_blocks):
        start = i * 2 * block_len
        h[start:start + block_len] = 1.0
    return project(h, n * 0.5)

def make_sine_start(n, freq, rng):
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + 0.45 * np.sin(np.pi * freq * x + rng.uniform(0, 2 * np.pi))
    return project(np.clip(h, 0, 1), n * 0.5)

def make_random_start(n, rng):
    h = 0.5 + 0.3 * rng.randn(n)
    return project(h, n * 0.5)

def downsample(h, n_new):
    n_old = len(h)
    h_new = np.zeros(n_new)
    for i in range(n_new):
        s = int(i * n_old / n_new)
        e = int((i + 1) * n_old / n_new)
        if e > s:
            h_new[i] = np.mean(h[s:e])
    return project(h_new, n_new * 0.5)

def interpolate_up(h_old, n_new):
    n_old = len(h_old)
    x_old = np.arange(n_old) / n_old * 2.0
    x_new = np.arange(n_new) / n_new * 2.0
    h_ext = np.concatenate([h_old, [h_old[0]]])
    x_ext = np.concatenate([x_old, [2.0]])
    return project(np.interp(x_new, x_ext, h_ext), n_new * 0.5)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = 1.0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=np.float64)
        best_h = project(h_parent.copy(), len(h_parent) * 0.5)
        best_c5 = compute_c5(best_h)
    else:
        h_parent = None

    # Phase 1: Coarse optimization from parent (fast, n=40)
    if h_parent is not None:
        n_c = 40
        h_c = downsample(h_parent, n_c)
        dl = t0 + budget_s * 0.08
        h_coarse, c5_c = optimize(h_c, dl, rng, alpha0=50, lr0=0.012, max_stall=5)
        h_up = interpolate_up(h_coarse, len(best_h))
        c5_up = compute_c5(h_up)
        if c5_up < best_c5:
            best_h = h_up
            best_c5 = c5_up

    # Phase 2: Main optimization from best candidate
    n_main = len(best_h)
    dl2 = t0 + budget_s * 0.45
    h_opt, c5_opt = optimize(best_h, dl2, rng, alpha0=100, lr0=0.006, max_stall=12)
    if c5_opt < best_c5:
        best_h = h_opt
        best_c5 = c5_opt

    # Phase 3: Diverse alternatives
    elapsed = time.time() - t0
    remaining = budget_s - elapsed
    alt_starts = []
    n_alt = max(60, min(120, n_main))
    alt_starts.append(make_block_start(n_alt, rng))
    alt_starts.append(make_sine_start(n_alt, 2, rng))
    alt_starts.append(make_sine_start(n_alt, 3, rng))
    alt_starts.append(make_random_start(n_alt, rng))

    if remaining > 80:
        time_per_alt = max(30, (remaining - 60) / len(alt_starts))
        for hs in alt_starts:
            if time.time() - t0 > budget_s - 60:
                break
            dl_alt = time.time() + time_per_alt
            h_a, c5_a = optimize(hs, dl_alt, rng, alpha0=80, lr0=0.01, max_stall=8)
            if c5_a < best_c5:
                best_h = h_a
                best_c5 = c5_a

    # Phase 4: Swap-based local search on best
    elapsed = time.time() - t0
    if elapsed < budget_s - 30:
        best_h, best_c5 = swap_local_search(best_h, n_swaps=500, rng=rng)

    # Phase 5: Final gradient polish
    elapsed = time.time() - t0
    if elapsed < budget_s - 10:
        h_fin, c5_fin = optimize(best_h, t0 + budget_s - 3, rng, alpha0=200, lr0=0.003, max_stall=6)
        if c5_fin < best_c5:
            best_h = h_fin
            best_c5 = c5_fin

    # Phase 6: Final swap search
    if time.time() - t0 < budget_s - 5:

    best_h = project(best_h.astype(np.float64), len(best_h) * 0.5)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
