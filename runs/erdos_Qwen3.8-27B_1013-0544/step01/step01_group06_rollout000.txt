    best_h = None
    best_c5 = 1.0
    best_n = 0

    def track(h, c5):
        nonlocal best_h, best_c5, best_n
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            best_n = len(h)

    if h_init is not None:
        n = len(h_init)
        h_init = project_box_sum(h_init, n / 2.0)
        track(h_init, compute_c5(h_init))
    else:
        n = 100
        h_init = project_box_sum(0.5 + 0.15 * rng.randn(n), n / 2.0)
        track(h_init, compute_c5(h_init))

    # Phase 1: Optimize at initial resolution (40% budget)
    d1 = t0 + budget_s * 0.40
    h1, c5_1 = optimize_nesterov(h_init, n / 2.0, d1, rng, alpha_init=50.0)
    track(h1, c5_1)

    # Phase 2: Try different resolutions (45% budget)
    elapsed = time.time() - t0
    remaining = budget_s - elapsed
    resolutions = [100, 140, 160]
    for n_try in resolutions:
        if remaining < 60:
            break
        h_interp = interp_resample(best_h, n_try)
        h_interp = project_box_sum(h_interp, n_try / 2.0)
        phase_time = min(remaining * 0.30, 200.0)
        d2 = time.time() + phase_time
        h2, c5_2 = optimize_nesterov(h_interp, n_try / 2.0, d2, rng, alpha_init=40.0)
        track(h2, c5_2)
        remaining = budget_s - (time.time() - t0)

    # Phase 3: Lower resolution attempt (often gives better C5)
    remaining = budget_s - (time.time() - t0)
    if remaining > 80:
        for n_low in [60, 80]:
            if remaining < 50:
                break
            h_low = interp_resample(best_h, n_low)
            h_low = project_box_sum(h_low, n_low / 2.0)
            d3 = time.time() + min(remaining * 0.35, 150.0)
            hl, cl = optimize_nesterov(h_low, n_low / 2.0, d3, rng, alpha_init=40.0)
            track(hl, cl)
            remaining = budget_s - (time.time() - t0)

    # Phase 4: Final polish on best resolution
    remaining = budget_s - (time.time() - t0)
    if remaining > 20:
        n_best = best_n
        target = n_best / 2.0
        d4 = t0 + budget_s - 3
        hp, cp = optimize_nesterov(best_h, target, d4, rng, alpha_init=100.0)
        track(hp, cp)

    # Final projection and evaluation
    best_h = project_box_sum(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
