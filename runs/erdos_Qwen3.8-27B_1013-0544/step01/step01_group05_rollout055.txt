```python
        if tleft() < 100.0:
            break
        is_primary = (n_points == n_init)
        time_limit = 80.0 if is_primary else 60.0

        dx = 2.0 / n_points
        target_sum = n_points / 2.0
        n_half = n_points - 1

        if n_points == n_init and h_init is not None:
            h = project_feasible(h_init, target_sum)
        elif best_h is not None:
            x_old = np.linspace(0, 2, len(best_h), endpoint=False)
            x_new = np.linspace(0, 2, n_points, endpoint=False)
            h = project_feasible(np.interp(x_new, x_old, best_h), target_sum)
        else:
            h = np.full(n_points, 0.5)

        c5_curr = compute_c5(h, dx)
        if c5_curr < best_c5:
            best_c5 = c5_curr
            best_h = h.copy()
            best_n = n_points

        # Phase 1: Adam with log-sum-exp (curriculum in T)
        T_schedule = [1.0, 3.0, 10.0, 30.0, 100.0, 300.0, 1000.0, 5000.0]
        for T in T_schedule:
            if tleft() < time_limit:
                break
            h_curr = h.copy()
            beta1, beta2, eps = 0.8, 0.999, 1e-8
            m_adam = np.zeros(n_points)
            v_adam = np.zeros(n_points)
            t_adam = 0
            base_lr = 0.08 / (1.0 + T * 0.03)
            best_this_T = compute_c5(h_curr, dx)
            stall = 0
            max_iters = min(4000, max(200, int((tleft() - time_limit) / 0.002)))

            for it in range(max_iters):
                if tleft() < time_limit:
                    break
                t_adam += 1
                corr = np.correlate(h_curr, 1.0 - h_curr, mode='full')
                lags = corr * dx
                lags_max = lags.max()
                scaled = T * (lags - lags_max)
                scaled = np.clip(scaled, -50.0, 50.0)
                weights = np.exp(scaled)
                weights /= weights.sum()

                grad = np.zeros(n_points)
                for i in range(2 * n_points - 1):
                    if weights[i] > 1e-8:
                        grad += weights[i] * lag_gradient(h_curr, i - n_half)

                m_adam = beta1 * m_adam + (1.0 - beta1) * grad
                v_adam = beta2 * v_adam + (1.0 - beta2) * grad * grad
                m_hat = m_adam / (1.0 - beta1 ** t_adam)
                v_hat = v_adam / (1.0 - beta2 ** t_adam)
                update = base_lr * m_hat / (np.sqrt(v_hat) + eps)

                h_new = project_feasible(h_curr - update, target_sum)
                c5_new = compute_c5(h_new, dx)

                if c5_new < best_this_T:
                    best_this_T = c5_new
                    h_curr = h_new
                    stall = 0
                else:
                    stall += 1
                    if stall > 80:
                        delta = c5_new - best_this_T
                        temp = 0.005 * np.exp(-t_adam / 500.0) + 0.0005
                        if rng.random() < np.exp(-delta / temp):
                            h_curr = h_new
                            stall = 0
                        else:
                            src = best_h if (best_n == n_points and best_h is not None) else h
                            h_curr = project_feasible(src + rng.normal(0, 0.03, n_points), target_sum)
                            m_adam = np.zeros(n_points)
                            v_adam = np.zeros(n_points)
                            t_adam = 0
                            stall = 0

                h = h_curr
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    best_n = n_points

        # Phase 2: Hard-max active-set gradient
        if tleft() < time_limit + 20.0:
            continue
        h_curr = (best_h if best_n == n_points else h).copy()
        step = 0.003
        mu = 0.5
        mom = np.zeros(n_points)
        max_iters = min(3000, int(max(100, (tleft() - time_limit - 20.0) / 0.003)))

        for it in range(max_iters):
            if tleft() < time_limit:
                break
            corr = np.correlate(h_curr, 1.0 - h_curr, mode='full')
            c_max = float(np.max(corr))
            tol = max(0.005, c_max * 0.0015)
            active = np.where(corr >= c_max - tol)[0]
            if len(active) < 2:
                active = np.argsort(corr)[-3:]
            elif len(active) > 15:
                active = np.argsort(corr)[-15:]

            grad = np.zeros(n_points)
            for idx in active:
                grad += lag_gradient(h_curr, int(idx) - n_half)
            grad /= float(len(active))

            mom = mu * mom - step * grad
            h_new = project_feasible(h_curr + mom, target_sum)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                best_n = n_points
            h_curr = h_new
            step *= 0.9995
            if step < 1e-9:
                break
        if best_n == n_points:
            h = best_h.copy()

    # Phase 3: L-BFGS on smooth objective
    if tleft() > 50.0 and best_h is not None:
        n_pts = best_n
        dx = 2.0 / n_pts
        tgt = n_pts / 2.0
        T_lbfgs = 300.0

        def smooth_obj(x):
            hh = np.asarray(x)
            lags = np.correlate(hh, 1.0 - hh, mode='full') * dx
            ls = lags - lags.max()
            return float(lags.max() + np.log(np.exp(T_lbfgs * np.clip(ls, -50, 0)).sum()) / T_lbfgs)

        try:
            res = minimize(smooth_obj, best_h, method='L-BFGS-B',
                          bounds=[(0.001, 0.999)] * n_pts,
                          options={'maxiter': 2000, 'ftol': 1e-14})
            h_ref = project_feasible(res.x, tgt)
            c5_ref = compute_c5(h_ref, dx)
            if c5_ref < best_c5:
                best_c5 = c5_ref
                best_h = h_ref
        except Exception:
            pass

    # Phase 4: SLSQP
    if tleft() > 30.0 and best_h is not None:
        n_pts = best_n
        dx = 2.0 / n_pts
        tgt = n_pts / 2.0
        T_slq = 500.0

        def slq_obj(x):
            hh = np.asarray(x)
            lags = np.correlate(hh, 1.0 - hh, mode='full') * dx
            ls = lags - lags.max()
            return float(lags.max() + np.log(np.exp(T_slq * np.clip(ls, -50, 0)).sum()) / T_slq)

        try:
            res = minimize(slq_obj, best_h, method='SLSQP',
                          bounds=[(0.0, 1.0)] * n_pts,
                          constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - tgt}],
                          options={'maxiter': 1000, 'ftol': 1e-12})
            if res.x is not None:
                h_ref = project_feasible(res.x, tgt)
                c5_ref = compute_c5(h_ref, dx)
                if c5_ref < best_c5:
                    best_c5 = c5_ref
                    best_h = h_ref
        except Exception:
            pass

    # Phase 5: Fine perturbation search
    if tleft() > 5.0 and best_h is not None:
        n_pts = best_n
        dx = 2.0 / n_pts
        tgt = n_pts / 2.0
        for noise in [0.015, 0.008, 0.004, 0.002, 0.001]:
            if tleft() < 3.0:
                break
            for _ in range(60):
                if tleft() < 2.0:
                    break
                h_p = best_h + rng.normal(0, noise, n_pts)
                h_p = project_feasible(h_p, tgt)
                c5_p = compute_c5(h_p, dx)
                if c5_p < best_c5:
                    best_c5 = c5_p
                    best_h = h_p.copy()

    return (best_h, best_c5, best_n)
```
