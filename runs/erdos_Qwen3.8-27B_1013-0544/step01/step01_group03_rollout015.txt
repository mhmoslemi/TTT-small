```python
            if new_max < best_delta:
                best_delta = new_max
                best_p, best_q = p, q
    
    return best_p, best_q, best_delta
```

```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(obj, tau):
    x = tau * obj
    x_max = np.max(x)
    return float(x_max / tau + np.log(np.sum(np.exp(x - x_max))) / tau)

def fix_sum(h, target):
    h = np.array(h, dtype=float)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return np.clip(h, 0.0, 1.0)
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = mask.sum()
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def make_binary(h):
    n = len(h)
    target = n // 2
    order = np.argsort(-h)
    b = np.zeros(n)
    b[order[:target]] = 1.0
    return b

def kronecker_h(n, alpha):
    i = np.arange(n)
    frac = (i * alpha) % 1.0
    target = n // 2
    idx = np.argsort(frac)[:target]
    h = np.zeros(n)
    h[idx] = 1.0
    return h

def compute_delta(b, p, q, k, n):
    j1 = p + k - (n - 1)
    j2 = p - k + (n - 1)
    j3 = q + k - (n - 1)
    j4 = q - k + (n - 1)
    delta = np.zeros(2 * n - 1)
    v1 = (j1 >= 0) & (j1 < n)
    v2 = (j2 >= 0) & (j2 < n)
    v3 = (j3 >= 0) & (j3 < n)
    v4 = (j4 >= 0) & (j4 < n)
    delta[v1] -= (1.0 - b[j1[v1]])
    delta[v2] += b[j2[v2]]
    delta[v3] += (1.0 - b[j3[v3]])
    delta[v4] -= b[j4[v4]]
    return delta

def greedy_improve(b, corr, t0, time_limit, max_rounds=200):
    n = len(b)
    k = np.arange(2 * n - 1)
    current_max = float(np.max(corr))
    
    for _ in range(max_rounds):
        if time.time() - t0 > time_limit:
            break
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        best_val = current_max
        best_p, best_q = -1, -1
        
        for p in ones:
            dp = compute_delta(np.zeros(n), p, -1, k, n) if False else None
            # Compute delta_p once
            dp = np.zeros(2 * n - 1)
            j1 = p + k - (n - 1)
            j2 = p - k + (n - 1)
            v1 = (j1 >= 0) & (j1 < n)
            v2 = (j2 >= 0) & (j2 < n)
            dp[v1] = -(1.0 - b[j1[v1]])
            dp[v2] += b[j2[v2]]
            
            for q in zeros:
                dq = np.zeros(2 * n - 1)
                j3 = q + k - (n - 1)
                j4 = q - k + (n - 1)
                v3 = (j3 >= 0) & (j3 < n)
                v4 = (j4 >= 0) & (j4 < n)
                dq[v3] = (1.0 - b[j3[v3]])
                dq[v4] -= b[j4[v4]]
                
                nm = np.max(corr + dp + dq)
                if nm < best_val:
                    best_val = nm
                    best_p, best_q = p, q
        
        if best_p == -1:
            break
        
        b[best_p] = 0.0
        b[best_q] = 1.0
        corr = np.correlate(b, 1.0 - b, mode="full").astype(float)
        current_max = float(np.max(corr))
    
    return b, corr, current_max

def sa_binary(b, t0, time_limit, max_iter=50000, T_init=2.0, T_min=0.05, cooling=0.9999):

    b = b.copy()
    k = np.arange(2 * n - 1)
    corr = np.correlate(b, 1.0 - b, mode="full").astype(float)

    best_b = b.copy()

        delta[v3] += (1.0 - b[j3[v3]])
        delta[v4] -= b[j4[v4]]

            b[p] = 0.0

            current_obj = new_obj
            if current_obj < best_obj:
                best_obj = current_obj
                best_b = b.copy()
        
        T = max(T_min, T * cooling)
    
    return best_b, best_obj

def continuous_optimize(h_start, n_pts, t0, time_limit):
    target = n_pts / 2.0
    bnds = Bounds(np.zeros(n_pts), np.ones(n_pts))
    best_h = h_start.copy()
    best_c5 = evaluate_c5(best_h)
    
    for tau in [30, 100, 300, 1000, 3000, 10000]:
        if time.time() - t0 > time_limit:
            break
        
        def obj(x):
            c = np.correlate(x, 1.0 - x, mode="full")
            return smooth_max(c, tau)
        
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                 'jac': lambda x: np.ones(n_pts)}]
        try:
            res = minimize(obj, best_h, method='SLSQP', bounds=bnds,
                          constraints=cons, options={'maxiter': 500, 'ftol': 1e-14})
            x = fix_sum(res.x, target)
            c5 = evaluate_c5(x)
            if c5 < best_c5:
                best_h = x
                best_c5 = c5
        except Exception:
            pass
    
    return best_h, best_c5

def bisection_feasible(h_init, n_pts, C_test, t0, time_limit):
    """Check if C5 <= C_test is feasible, and find a solution if so."""
    target = n_pts / 2.0
    bnds = Bounds(np.zeros(n_pts), np.ones(n_pts))
    
    def obj(x):
        return 0.0
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                    'jac': lambda x: np.ones(n_pts)}]
    for k in range(2 * n_pts - 1):
        k_offset = n_pts - 1
        i_min = max(0, k_offset - k)
        i_max = min(n_pts - 1, k)
        idx1 = np.arange(i_min, i_max + 1)
        idx2 = idx1 + k - k_offset
        
        def make_constraint(kk, id1, id2):
            def cfun(x):
                return C_test / (2.0 / n_pts) - np.sum(x[id1] * (1.0 - x[id2]))
            return {'type': 'ineq', 'fun': cfun}
        
        constraints.append(make_constraint(k, idx1, idx2))
    
    try:
        res = minimize(obj, h_init, method='SLSQP', bounds=bnds,
                      constraints=constraints, options={'maxiter': 200})
        if res.success and np.isfinite(res.fun):
            x = fix_sum(res.x, target)
            c5 = evaluate_c5(x)
            if c5 <= C_test + 1e-10:
                return x, c5
    except Exception:
        pass
    return None, None
```
