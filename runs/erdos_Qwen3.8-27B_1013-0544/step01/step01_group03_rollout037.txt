f is increasing at t if all active D_m > 0 (i.e., min active D > 0).

- If min(D[active]) > 0: f is increasing, minimum is to the left → hi = mid
- Else: we're at (or very near) the minimum

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(u, tau):
    x = tau * u
    x_max = np.max(x)
    return float(x_max / tau + np.log(np.sum(np.exp(x - x_max))) / tau)

def project_onto_set(v, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        if np.clip(v - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2, 0, 1)

def compute_gradient(h, tau):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    x = tau * corr
    x_max = np.max(x)
    w = np.exp(x - x_max)
    w /= w.sum()
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-12:
            continue
        o = m - (n - 1)
        g = np.zeros(n)
        if o >= 0:
            g[:n - o] += 1.0 - h[o:]
            g[o:] -= h[:n - o]
        else:
            p = -o
            g[p:] += 1.0 - h[:n - p]
            g[:n - p] -= h[p:]
        grad += w[m] * g
    return grad

def pg_optimize(h_init, tau, target, n_iter=200, step_init=None):
    n = len(h_init)
    h = h_init.copy()
    if step_init is None:
        step = 1.0 / (tau * n)
    else:
        step = step_init
    best_obj = smooth_max(np.correlate(h, 1 - h, mode='full'), tau)
    h_best = h.copy()
    for it in range(n_iter):
        grad = compute_gradient(h, tau)
        obj_new = best_obj + 1.0
        for _ in range(20):
            h_new = project_onto_set(h - step * grad, target)
            obj_new = smooth_max(np.correlate(h_new, 1 - h_new, mode='full'), tau)
            if obj_new < best_obj:
                break
            step *= 0.5
        h = h_new
        if obj_new < best_obj:
            best_obj = obj_new
            h_best = h.copy()
        step *= 1.05
    return h_best

def coord_desc(h, n_sweeps=5):
    n = len(h)
    target = n / 2.0
    for _ in range(n_sweeps):
        for j in range(n):
            old = h[j]
            h[j] = 0.0
            B = np.correlate(h, 1.0 - h, mode="full").copy()
            h[j] = 1.0
            C1 = np.correlate(h, 1.0 - h, mode="full").copy()
            h[j] = old
            D = C1 - B
            lo, hi2 = 0.0, 1.0
            for _ in range(50):
                mid = (lo + hi2) / 2
                vals = B + D * mid
                vmax = np.max(vals)
                active = vals >= vmax - 1e-14
                dmax = D[active].max()
                dmin = D[active].min()
                if dmax < 0:
                    lo = mid
                elif dmin > 0:
                    hi2 = mid
                else:
                    break
            h[j] = (lo + hi2) / 2.0
        h = project_onto_set(h, target)
    return h

def slsqp_polish(h_init, target, maxiter=200):
    n = len(h_init)
    bnds = minimize.Bounds(np.zeros(n), np.ones(n))
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        return smooth_max(corr, 10000)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h_init, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_onto_set(res.x, target)
    except Exception:
        return h_init.copy()
