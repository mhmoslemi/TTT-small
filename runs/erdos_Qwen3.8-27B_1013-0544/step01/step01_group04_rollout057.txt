    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    ones = np.ones(n)
    def obj(x):
        return c5_val_grad(x, dx)[0]
    def jac(x):
        return c5_val_grad(x, dx)[1]
    cons = {'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: ones}
    try:
        res = minimize(obj, h0, jac=jac, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h0, n)

def param_to_h(p, m, n_eval):
    values = p[:m]
    bounds_arr = np.concatenate([[0.0], p[m:], [2.0]])
    h = np.zeros(n_eval)
    for i in range(m):
        s = int(np.floor(bounds_arr[i] / 2.0 * n_eval))
        e = int(np.floor(bounds_arr[i+1] / 2.0 * n_eval))
        if e > s:
            h[s:e] = values[i]
    return h

def param_objective(p, m, n_eval):
    h = param_to_h(p, m, n_eval)
    s = h.sum()
    if abs(s - n_eval / 2.0) > n_eval * 0.01:
        return 1.0 + abs(s - n_eval / 2.0)
    return compute_c5(h, 2.0 / n_eval)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0

    best_h = project_feasible(h0.copy(), n)
    best_c5 = compute_c5(best_h, dx)

    def elapsed():
        return time.time() - t0

    # Phase 1: SLSQP from initial (with analytic gradient - fast)
    if elapsed() < budget_s * 0.1:
        h_new = slsqp_fast(best_h, n, dx, max_iter=1000, ftol=1e-14)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 2: Diverse restarts with SLSQP
    phase2_end = budget_s * 0.50

    candidates = []
    candidates.append(1.0 - best_h)
    for s in [1, 2, 3, 5, 7, 11, 15, 20, 30, 45]:
        candidates.append(np.roll(best_h, s))
    x_arr = np.arange(n) / n * 2.0 * np.pi
    for freq in [1, 2, 3, 4, 6, 8, 12]:
        for amp in [0.05, 0.1, 0.15, 0.25]:
            candidates.append(best_h + amp * np.sin(freq * x_arr + rng.uniform(0, 2*np.pi)))
    for scale in [0.02, 0.04, 0.08, 0.12, 0.2, 0.35]:
        candidates.append(best_h + rng.randn(n) * scale)
    for _ in range(8):
        candidates.append(rng.rand(n))
    # Smooth random
    for _ in range(4):
        coeff = rng.randn(n)
        from numpy.fft import rfft, irfft
        mask = np.ones(len(coeff), dtype=bool)
        mask[3:] = False
        coeff_f = rfft(coeff)
        coeff_f[3:] = 0
        smooth = irfft(coeff_f, n)
        candidates.append(best_h + 0.1 * smooth)

    rng.shuffle(candidates)

    pool = [best_h.copy()]
    for h_try in candidates:
        if elapsed() > phase2_end:
            break
        h_try = project_feasible(h_try.copy(), n)
        h_new = slsqp_fast(h_try, n, dx, max_iter=500, ftol=1e-13)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5 - 1e-14:
            best_h, best_c5 = h_new, c5_new
            pool.append(h_new.copy())
        elif len(pool) < 8:
            pool.append(h_new.copy())

    # Phase 3: Parameterized (piecewise constant) search
    phase3_end = budget_s * 0.72
    n_eval = 400

    for m in [4, 5, 6, 7, 8, 10, 12, 15, 20]:
        if elapsed() > phase3_end:
            break
        for trial in range(4):
            if elapsed() > phase3_end:
                break
            # Generate diverse initial params
            if trial == 0:
                values = np.round(rng.rand(m))
                values = np.clip(values, 0.0, 1.0)
            elif trial == 1:
                values = np.concatenate([np.ones(m//2), np.zeros(m - m//2)])
                rng.shuffle(values)
            else:
                values = rng.rand(m)
            boundaries = np.sort(rng.uniform(0.05, 1.95, m-1))
            # Adjust values to approximately satisfy sum constraint
            widths = np.diff(np.concatenate([[0.0], boundaries, [2.0]]))
            cur_sum = np.sum(widths * values)
            if cur_sum > 0.01:
                values = values * (1.0 / (2.0 * cur_sum))
                values = np.clip(values, 0.0, 1.0)
            params = np.concatenate([values, boundaries])

            p_bounds = []
            for _ in range(m):
                p_bounds.append((0.0, 1.0))
            for i in range(m-1):
                p_bounds.append((0.01, 1.99))

            def p_obj(p):
                return param_objective(p, m, n_eval)
            def p_jac(p):
                # Numerical gradient
                eps = 1e-7
                grad = np.zeros(len(p))
                f0 = p_obj(p)
                for j in range(len(p)):
                    p[j] += eps
                    grad[j] = (p_obj(p) - f0) / eps
                    p[j] -= eps
                return grad

            cons = []
            # Ordering constraints: b_i < b_{i+1}
            for i in range(m-2):
                def ord_con(p, i=i):
                    return p[m+1+i] - p[m+i]
                cons.append({'type': 'ineq', 'fun': ord_con})
            # Sum constraint
            def sum_con(p):
                h = param_to_h(p, m, n_eval)
                return h.sum() - n_eval / 2.0
            cons.append({'type': 'eq', 'fun': sum_con})

            try:
                res = minimize(p_obj, params, jac=p_jac, method='SLSQP',
                              bounds=p_bounds, constraints=cons,
                              options={'maxiter': 200, 'ftol': 1e-10})
                p_opt = res.x
            except Exception:
                p_opt = params

            h_param = param_to_h(p_opt, m, n_eval)
            h_mapped = np.interp(np.linspace(0, 2, n, endpoint=False),
                                np.linspace(0, 2, n_eval, endpoint=False), h_param)
            h_mapped = project_feasible(h_mapped, n)
            c5_mapped = compute_c5(h_mapped, dx)

            if c5_mapped < best_c5 + 0.005:
                h_new = slsqp_fast(h_mapped, n, dx, max_iter=600, ftol=1e-14)
                c5_new = compute_c5(h_new, dx)
                if c5_new < best_c5 - 1e-14:
                    best_h, best_c5 = h_new, c5_new

    # Phase 4: Iterative local improvement with decreasing scale
    phase4_end = budget_s - 20
    no_improve = 0
    for i in range(200):
        if elapsed() > phase4_end:
            break
        if no_improve > 30:
            break
        # Mix: small perturbations from best, occasional from pool
        if i % 5 == 0 and len(pool) > 1:
            base = pool[rng.randint(1, len(pool))]
        else:
            base = best_h
        scale = 0.03 * (1.0 + 0.5 * np.cos(i * 0.3))
        scale = max(0.005, scale)
        h_try = base + rng.randn(n) * scale
        h_try = project_feasible(h_try, n)
        h_new = slsqp_fast(h_try, n, dx, max_iter=300, ftol=1e-13)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5 - 1e-14:
            best_h, best_c5 = h_new, c5_new
            pool.append(h_new.copy())
            no_improve = 0
        else:
            no_improve += 1

    # Final: try higher resolution if time allows
    if elapsed() < budget_s - 30:
        n2 = 2 * n
        dx2 = 2.0 / n2
        h_up = np.interp(np.linspace(0, 2, n2, endpoint=False),
                         np.linspace(0, 2, n, endpoint=False), best_h)
        h_up = project_feasible(h_up, n2)
        h_up = slsqp_fast(h_up, n2, dx2, max_iter=300, ftol=1e-13)
        c5_up = compute_c5(h_up, dx2)
        if c5_up < best_c5:
            best_h, best_c5 = h_up, c5_up

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return (best_h, best_c5, len(best_h))
