```python
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    # Softmax with numerical stability
    vals_shifted = vals - np.max(vals)
    weights = np.exp(alpha * vals_shifted)
    weights /= np.sum(weights)
    
    # Objective
    obj = (np.max(vals) + np.log(np.sum(weights)) / alpha)
    # Actually: f_alpha = max + (1/alpha)*log(sum exp(alpha*(v_k - max)))
    
    # Gradient: sum_k weights[k] * dx * grad_corr[k]
    # grad_corr[m] with respect to h[j]:
    # offset = m - (n-1)
    # grad[j] = (1-h[j-offset]) - h[j+offset] (with bounds)
    
    grad_h = np.zeros(n)
    for m in range(len(corr)):
        if weights[m] < 1e-15:
            continue
        offset = m - (n - 1)
        j = np.arange(n)
        # First term: 1-h[j-offset], valid where 0<=j-offset<n
        idx1 = j - offset
        mask1 = (idx1 >= 0) & (idx1 < n)
        t1 = np.zeros(n)
        t1[mask1] = 1.0 - h[idx1[mask1]]
        # Second term: h[j+offset], valid where 0<=j+offset<n
        idx2 = j + offset
        mask2 = (idx2 >= 0) & (idx2 < n)
        t2 = np.zeros(n)
        t2[mask2] = h[idx2[mask2]]
        grad_h += weights[m] * (t1 - t2)
    
    grad_h *= dx
    return obj, grad_h
```

This is O(n * 2n) = O(n²) per gradient evaluation, which for n=90 is about 16200 operations per gradient. With L-BFGS-B, we can do many iterations quickly.

But wait, I also need the sum constraint. I can handle it with a penalty:
f_total = f_α + λ * (sum(h) - n/2)²

Or I can project after each step.

Let me use a projected gradient approach:
1. Compute gradient of f_α
2. Take a step in the negative gradient direction
3. Project to feasible set (0 ≤ h ≤ 1, sum(h) = n/2)
4. Repeat with adaptive step size

This should be quite fast and effective.

Let me also think about the alpha schedule: start with small alpha (smooth) and increase it (sharper) as we converge. This is like simulated annealing for the max function.

Here's my refined plan:

```
Phase 1 (60% of time): Projected gradient with smooth objective
  - Start alpha = 5, increase to alpha = 50 over the phase
  - Use backtracking line search
  - Project after each step
  
Phase 2 (30% of time): Basin hopping
  - Take best solution, perturb with random noise
  - Polish with SLSQP or more projected gradient
  - Keep best
  
Phase 3 (10% of time): Final polishing
  - Very fine SLSQP on the best solution
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - target
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(free):
            h[free] -= diff / np.sum(free)
        else:
            h -= diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = np.sum(h) - target
    if abs(diff) > 1e-15:
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(free):
            h[free] -= diff / np.sum(free)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_and_grad(h, dx, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    v_max = np.max(vals)
    shifted = alpha * (vals - v_max)
    exp_vals = np.exp(shifted)
    log_sum = np.log(np.sum(exp_vals))
    obj = v_max + log_sum / alpha
    weights = exp_vals / np.sum(exp_vals)
    grad_h = np.zeros(n)
    j = np.arange(n)
    for m in range(len(corr)):
        if weights[m] < 1e-10:
            continue
        w = weights[m]
        offset = m - (n - 1)
        idx1 = j - offset
        mask1 = (idx1 >= 0) & (idx1 < n)
        idx2 = j + offset
        mask2 = (idx2 >= 0) & (idx2 < n)
        t1 = np.zeros(n)
        t2 = np.zeros(n)
        t1[mask1] = 1.0 - h[idx1[mask1]]
        t2[mask2] = h[idx2[mask2]]
        grad_h += w * (t1 - t2)
    grad_h *= dx
    return obj, grad_h

def projected_gradient_optimize(h, dx, alpha_start, alpha_end, n_steps, target_sum):
    n = len(h)
    best = h.copy()
    best_obj = compute_c5(best, dx)
    lr = 0.1
    for step in range(n_steps):
        progress = step / max(n_steps - 1, 1)
        alpha = alpha_start + (alpha_end - alpha_start) * progress
        obj, grad = smooth_obj_and_grad(h, dx, alpha)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        # Backtracking line search
        for _ in range(30):
            h_new = project_feasible(h - lr * grad, n)
            obj_new, _ = smooth_obj_and_grad(h_new, dx, alpha)
            if obj_new < obj - 1e-4 * lr * gnorm * gnorm:
                break
            lr *= 0.5
        else:
            lr = max(lr, 1e-6)
            h_new = project_feasible(h - lr * grad, n)
        h = h_new
        c5_actual = compute_c5(h, dx)
        if c5_actual < best_obj:
            best_obj = c5_actual
            best = h.copy()
        lr = min(lr * 1.1, 0.5)
    return best, best_obj

def slsqp_polish(h_start, n, dx, target_sum, max_iter=300):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    res = minimize(lambda x: compute_c5(x, dx), h_start, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-15})
    return project_feasible(res.x, n)

def lbfgs_penalise(h_start, n, dx, target_sum, alpha, max_iter=200):
    def obj_pen(x):
        o, g = smooth_obj_and_grad(x, dx, alpha)
        pen = 100.0 * (np.sum(x) - target_sum) ** 2
        return o + pen
    def obj_pen_grad(x):
        o, g = smooth_obj_and_grad(x, dx, alpha)
        g_pen = g + 200.0 * (np.sum(x) - target_sum) * np.ones(n)
        return o + 100.0 * (np.sum(x) - target_sum) ** 2, g_pen
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj_pen, h_start, jac=obj_pen_grad, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-10})
    return project_feasible(res.x, n)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)

    target_sum = n / 2.0

    def elapsed():
        return time.time() - start_time

    # Phase 1: Smooth projected gradient with increasing alpha
    phase1_end = budget_s * 0.40

    if elapsed() < phase1_end:

        c5_pol = compute_c5(h_pol, dx)

            best_h, best_c5 = h_pol, c5_pol

    # Phase 2: Basin hopping
    phase2_end = budget_s * 0.80
    attempt = 0
    while elapsed() < phase2_end:
        attempt += 1
        progress = (elapsed() - phase1_end) / max(phase2_end - phase1_end, 1)
        scale = 0.12 * (1.0 - progress) + 0.02
        h_try = best_h + rng.randn(n) * scale
        h_try = project_feasible(h_try, n)
        # Quick smooth optimization
        h_new, c5_new = projected_gradient_optimize(h_try, dx, 10.0, 60.0, 500, target_sum)
        # Polish
        if elapsed() < phase2_end:
            h_new = slsqp_polish(h_new, n, dx, target_sum, max_iter=200)
            c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

        h_try = project_feasible(h_try, n)

        c5_final = compute_c5(h_final, dx)
        if c5_final < best_c5:
            best_h, best_c5 = h_final, c5_final

    return (best_h, best_c5, n)
```
