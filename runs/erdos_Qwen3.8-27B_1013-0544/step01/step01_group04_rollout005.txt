            grad[:n-j] -= h[j:]
    return grad

def subgradient_step(h, n, dx, top_k=5, temperature=1.0):
    corrs = np.correlate(h, 1.0 - h, mode='full')
    scaled = corrs * dx
    # Softmax weights over top lags
    idx = np.argsort(scaled)[-top_k:]
    w = np.exp(scaled[idx] * temperature)
    w /= np.sum(w)
    grad = np.zeros(n)
    center = n - 1
    for idx_i, ki in enumerate(idx):
        k = ki - center
        grad += w[idx_i] * lag_gradient(h, k, n)
    grad *= dx
    grad -= np.mean(grad)
    return grad, np.max(scaled)

def subgradient_optimize(h, n, dx, max_iters=50000, base_lr=0.05, decay=0.0005, top_k=5):
    h = project_feasible(h.copy(), n)
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    patience = 0
    for it in range(max_iters):
        grad, _ = subgradient_step(h, n, dx, top_k=top_k, temperature=5.0)
        lr = base_lr / (1.0 + decay * it)
        h_new = h + lr * grad
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            patience = 0
        else:
            patience += 1
            if patience > 2000:
                break
    return best_h, best_c5

def slsqp_optimize(h_start, n, dx, max_iter=300):
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    constraint = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    def objective(x):
        return compute_c5(x, dx)
    res = minimize(objective, h_start, method='SLSQP',
                   bounds=bounds, constraints=constraint,
                   options={'maxiter': max_iter, 'ftol': 1e-13})
    return project_feasible(res.x, n)

def random_feasible(n, rng):
    h = rng.rand(n)
    return project_feasible(h, n)

def interpolate_upscale(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    old_x = np.linspace(0, 2, n_old, endpoint=False) + 1.0/n_old
    new_x = np.linspace(0, 2, n_new, endpoint=False) + 1.0/n_new
    h_new = np.interp(new_x, old_x, h)
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    dx0 = 2.0 / n0

    best_h = project_feasible(h0.copy(), n0)
    best_c5 = compute_c5(best_h, dx0)

    def elapsed():
        return time.time() - t0

    # Phase 1: Subgradient at n0
    p1_end = budget_s * 0.30
    best_h_p1, best_c5_p1 = subgradient_optimize(best_h, n0, dx0,
                                                  max_iters=100000, base_lr=0.04,
                                                  decay=0.0003, top_k=5)
    if best_c5_p1 < best_c5:
        best_h, best_c5 = best_h_p1, best_c5_p1

    # SLSQP polish at n0
    if elapsed() < p1_end + 60:
        h_pol = slsqp_optimize(best_h, n0, dx0, max_iter=300)
        c5_pol = compute_c5(h_pol, dx0)
        if c5_pol < best_c5:
            best_h, best_c5 = h_pol, c5_pol
        # Try a few random restarts at n0
        for _ in range(3):
            if elapsed() > p1_end + 120:
                break
            h_rand = random_feasible(n0, rng)
            h_r, c5_r = subgradient_optimize(h_rand, n0, dx0, max_iters=30000,
                                             base_lr=0.05, decay=0.0005, top_k=7)
            if c5_r < best_c5:
                best_h, best_c5 = h_r, c5_r

    # Phase 2: Upscale to 2*n0 and 4*n0
    for factor in [2, 4]:
        n_new = n0 * factor
        if n_new > 800:
            break
        dx_new = 2.0 / n_new
        p2_end = elapsed() + budget_s * 0.20

        # Start from interpolated version
        h_up = interpolate_upscale(best_h, n_new)
        c5_up = compute_c5(h_up, dx_new)

        h_sg, c5_sg = subgradient_optimize(h_up, n_new, dx_new,
                                           max_iters=50000, base_lr=0.03,
                                           decay=0.0004, top_k=5)
        if c5_sg < best_c5:
            best_h, best_c5 = h_sg.copy(), c5_sg
        elif c5_sg < c5_up:
            h_up = h_sg.copy()
            c5_up = c5_sg

        # SLSQP at new resolution
        if c5_up < best_c5 + 0.002:
            h_pol2 = slsqp_optimize(h_up, n_new, dx_new, max_iter=250)
            c5_pol2 = compute_c5(h_pol2, dx_new)
            if c5_pol2 < best_c5:
                best_h, best_c5 = h_pol2.copy(), c5_pol2
                h_up = h_pol2.copy()
                c5_up = c5_pol2

        # Try random restart at new resolution
        if elapsed() < p2_end:
            h_rand2 = random_feasible(n_new, rng)
            h_r2, c5_r2 = subgradient_optimize(h_rand2, n_new, dx_new,
                                               max_iters=20000, base_lr=0.04,
                                               decay=0.0005, top_k=7)
            if c5_r2 < best_c5:
                best_h, best_c5 = h_r2.copy(), c5_r2

    # Phase 3: Final polishing
    n_final = len(best_h)
    dx_final = 2.0 / n_final
    p3_start = elapsed()
    p3_budget = max(60, budget_s - p3_start - 10)

    # Multiple SLSQP attempts with tiny perturbations
    for i in range(10):
        if elapsed() > budget_s - 10:
            break
        if i == 0:
            h_try = best_h.copy()
        else:
            h_try = best_h + rng.randn(n_final) * 0.01
            h_try = project_feasible(h_try, n_final)
        h_new = slsqp_optimize(h_try, n_final, dx_final, max_iter=400)
        c5_new = compute_c5(h_new, dx_final)
        if c5_new < best_c5 - 1e-14:
            best_h = h_new
            best_c5 = c5_new

    # One more subgradient pass with very small steps
    if elapsed() < budget_s - 30:
        h_sg_f, c5_sg_f = subgradient_optimize(best_h, n_final, dx_final,
                                               max_iters=20000, base_lr=0.01,
                                               decay=0.001, top_k=3)
        if c5_sg_f < best_c5:
            best_h, best_c5 = h_sg_f, c5_sg_f
        h_final = slsqp_optimize(best_h, n_final, dx_final, max_iter=300)
        c5_final = compute_c5(h_final, dx_final)
        if c5_final < best_c5:
            best_h, best_c5 = h_final, c5_final

    return (best_h, best_c5, len(best_h))
