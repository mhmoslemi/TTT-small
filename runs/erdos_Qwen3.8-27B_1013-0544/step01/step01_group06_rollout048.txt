def try_block_swap(h, target_sum, rng, block_size=5, n_trials=50):

        c5 = compute_c5(h_trial)
        if c5 < best_c5:
            best_h = h_trial
            best_c5 = c5

def grad_corr_at_k(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        if k < n:
            g[k:] += (1.0 - h[:n - k])
            if k > 0:
                g[:n - k] -= h[k:]
    else:
        p = -k
        if p < n:
            g[:n - p] += (1.0 - h[p:])
            if p > 0:

    return g
