```python
import cvxpy as cp

    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')

def grad_all_shifts(h):
    """Compute gradient of each shift's correlation w.r.t. h. Returns (2n-1, n) array."""
    n = len(h)
    grads = np.zeros((2*n-1, n))
    for i in range(2*n-1):
        k = i - (n-1)
        g = np.zeros(n)
        if k >= 0:
            g[k:] += (1.0 - h[:n-k])
            g[:n-k] -= h[k:]
        else:
            p = -k
            g[p:] += (1.0 - h[:n-p])
            g[:n-p] -= h[p:]
        grads[i] = g
    return grads
```

This is much cleaner and should be faster to construct. The constraint `dx * (G @ h_var) - cp.ones(2*n-1) * t <= rhs` is a single vectorized constraint with 2n-1 rows.

Let me verify the formulation:
- G[i, j] = ∂(raw correlation at shift i) / ∂h[j]
- f_i(h) = dx * corr_i(h) where corr_i is the raw correlation
- Linearization: f_i(h_new) ≈ f_i(h) + dx * G[i,:] @ (h_new - h)
  = dx * corr[i] + dx * G[i,:] @ h_new - dx * G[i,:] @ h
- Constraint: f_i(h_new) <= t
  => dx * G[i,:] @ h_new <= t - dx * corr[i] + dx * G[i,:] @ h
  => dx * G[i,:] @ h_new - t <= dx * (G[i,:] @ h - corr[i])

Yes, that's correct. And rhs = dx * (G @ h - corr) is a vector of length 2n-1.

```python
import numpy as np
import time
import cvxpy as cp

def compute_c5(h):
    n = len(h)
    return np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n)

def project_onto_simplex_box(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def build_grad_matrix(n, h):
    G = np.zeros((2 * n - 1, n))
    for i in range(2 * n - 1):
        k = i - (n - 1)
        if k >= 0:
            G[i, k:] += (1.0 - h[:n-k])
            G[i, :n-k] -= h[k:]
        else:
            p = -k
            G[i, p:] += (1.0 - h[:n-p])
            G[i, :n-p] -= h[p:]
    return G

def solve_qp(h, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    G = build_grad_matrix(n, h)
    rhs = dx * (G @ h - corr)
    h_var = cp.Variable(n)
    t = cp.Variable()
    ones = cp.ones(2 * n - 1)
    constraints = [
        cp.sum(h_var) == n / 2.0,
        h_var >= 0, h_var <= 1,
        dx * (G @ h_var) - ones * t <= rhs
    ]
    prob = cp.Problem(cp.Minimize(t + lam * cp.sum_squares(h_var - h)), constraints)
    try:
        prob.solve(solver=cp.ECOS, max_iters=50000, abstol=1e-10)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=50000)
        except Exception:
            return h.copy()
    if h_var.value is not None:
        return np.clip(np.array(h_var.value), 0, 1)
    return h.copy()

def smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= s
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        if w[i] < 1e-8:
            continue
        k = i - (n - 1)
        g = np.zeros(n)
        if k >= 0:
            g[k:] += (1.0 - h[:n-k])
            g[:n-k] -= h[k:]
        else:
            p = -k
            g[p:] += (1.0 - h[:n-p])
            g[:n-p] -= h[p:]
        grad += w[i] * g
    return grad * dx

def grd_opt(h, target, deadline, rng, a0=80):
    best = h.copy()
    best_c = compute_c5(best)
    lr = 0.01
    alpha = a0
    stall = 0
    n = len(best)
    while time.time() < deadline:
        h_new = project_onto_simplex_box(best - lr * smooth_grad(best, alpha), target)
        c = compute_c5(h_new)
        if c < best_c - 1e-14:
            best, best_c, stall = h_new, c, 0
            lr = min(lr * 1.03, 0.05)
        else:
            stall += 1
            lr *= 0.93
            if stall > 60:
                hp = project_onto_simplex_box(best + 0.03 * rng.randn(n), target)
                if compute_c5(hp) < best_c:
                    best, best_c = hp, compute_c5(hp)
                alpha = min(alpha * 1.3, 400)
                lr = 0.008
                stall = 0
                if alpha >= 390:
                    break
    return best, best_c

def interp_to(h, n_new):
    n = len(h)
    xo = np.linspace(0, 2, n, endpoint=False) + 1.0 / n
    xn = np.linspace(0, 2, n_new, endpoint=False) + 1.0 / n_new
    return np.interp(xn, np.append(xo, 2.0 + 1.0 / n), np.append(h, h[0]))

def qp_optimize(h, target, deadline, rng, lam_init=1.0):
    best = h.copy()
    best_c = compute_c5(best)
    lam = lam_init
    no_imp = 0
    n = len(best)
    while time.time() < deadline:
        h_new = solve_qp(best, lam)
        h_new = project_onto_simplex_box(h_new, target)
        c = compute_c5(h_new)
        if c < best_c - 1e-10:
            best, best_c, no_imp = h_new.copy(), c, 0
            lam = max(lam * 0.7, 0.01)
        else:
            improved = False
            for beta in [0.5, 0.3, 0.15]:
                hm = project_onto_simplex_box(beta * h_new + (1 - beta) * best, target)
                cm = compute_c5(hm)
                if cm < best_c - 1e-10:
                    best, best_c = hm.copy(), cm
                    improved = True
                    break
            if not improved:
                no_imp += 1
                lam = min(lam * 1.5, 20.0)
                if no_imp > 40:
                    best = project_onto_simplex_box(best + 0.04 * rng.randn(n), target)
                    best_c = compute_c5(best)
                    no_imp = 0
                    lam = max(lam_init, 0.5)
                    if lam >= 19.0:
                        break
    return best, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:

    # Phase 2: Gradient polish
    rem = budget_s - (time.time() - t0)

        dl3 = t0 + budget_s - 80

        if c_pert < best_c:

        best, best_c = grd_opt(best, len(best) / 2.0, t0 + budget_s - 3, rng, a0=200)

    best = project_onto_simplex_box(best, len(best) / 2.0)
    best_c = compute_c5(best)
    return best, best_c, len(best)
```
