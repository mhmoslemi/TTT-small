import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-14:
        return h
    # Water-filling
    for _ in range(50):
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if mask.sum() == 0:
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
    return h

                return smooth_max(corr, beta) * dx + lam * (np.sum(h) - target) ** 2

The bound C₅ = max_k corr_k * dx where dx = 2/n. So for a given "shape" of h, using a larger n makes dx smaller, which could either help or hurt depending on how the max correlation scales.

For a smooth h, the correlation at each lag is approximately (2/n) * ∫ h(x)(1-h(x+k*2/n)) dx, which tends to the continuous integral as n → ∞. So larger n should converge to the true continuous value.

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if mask.sum() == 0:
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_penalty(h, dx, target, beta, lam):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, beta) * dx + lam * (np.sum(h) - target) ** 2

def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, beta) * dx

def con_fun_sum(x, target):
    return np.sum(x) - target

def con_jac_sum(x, n):
    return np.ones(n)

def time_remaining(start, budget):
    return budget - (time.time() - start)

def fourier_h(params, n, n_harm):
    i = np.arange(n).astype(float)
    h = np.full(n, 0.5)
    for m in range(n_harm):
        A = params[2 * m]
        phi = params[2 * m + 1]
        h += A * np.cos(2.0 * np.pi * (m + 1) * i / n + phi)
    return h

def obj_fourier(params, n, n_harm, beta):
    h = fourier_h(params, n, n_harm)
    h = project_h(h, n / 2.0)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, beta) * dx

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
        n = len(best_h)
    else:
        n = 70
        best_h = np.full(n, 0.5)

    best_h = project_h(best_h, n / 2.0)
    best_c5 = compute_c5(best_h)

    # Phase 1: Fourier parameterization (low-dimensional global search)
    n_harm = 4
    n_params = 2 * n_harm
    for n_try in [n, n + 10, n - 10, n + 20]:
        if time_remaining(start, budget_s) < 40 or n_try < 30:
            continue
        best_f_c5 = float('inf')
        for restart in range(25):
            if time_remaining(start, budget_s) < 15:
                break
            if restart == 0:
                p0 = np.zeros(n_params)
                p0[0] = 0.15
                p0[2] = 0.05
            elif restart == 1:
                p0 = np.zeros(n_params)
                p0[0] = -0.1
                p0[2] = 0.1
                p0[4] = 0.05
            else:
                p0 = np.random.randn(n_params) * 0.12
                p0[0::2] = np.abs(p0[0::2]) * 0.3
            res = minimize(obj_fourier, p0, args=(n_try, n_harm, 150),
                          method='Nelder-Mead',
                          options={'maxiter': 8000, 'xatol': 1e-10, 'fatol': 1e-12})
            h_f = fourier_h(res.x, n_try, n_harm)
            h_f = project_h(h_f, n_try / 2.0)
            c5_f = compute_c5(h_f)
            if c5_f < best_f_c5:
                best_f_c5 = c5_f
            if c5_f < best_c5:
                best_h, best_c5 = h_f.copy(), c5_f

    # Phase 2: Penalty + L-BFGS-B (main local optimization)
    n = len(best_h)
    dx = 2.0 / n
    target = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))

    for beta in [80, 150, 250, 400]:
        if time_remaining(start, budget_s) < 40:
            break
        for lam in [50, 500, 5000, 50000]:
            if time_remaining(start, budget_s) < 10:
                break
            res = minimize(obj_penalty, best_h, args=(dx, target, beta, lam),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 3000, 'gtol': 1e-13})
            h_new = project_h(res.x, target)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new.copy(), c5

    # Phase 3: SLSQP with equality constraint
    if time_remaining(start, budget_s) > 20:
        constraints = [{'type': 'eq', 'fun': con_fun_sum, 'jac': con_jac_sum,
                       'args': (target, n)}]
        for beta in [200, 400]:
            if time_remaining(start, budget_s) < 10:
                break
            res = minimize(obj_smooth, best_h, args=(dx, beta),
                          method='SLSQP', bounds=bounds, constraints=constraints,
                          options={'maxiter': 2000, 'ftol': 1e-14})
            h_new = project_h(res.x, target)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new.copy(), c5

    # Phase 4: Diversified restarts with L-BFGS-B
    if time_remaining(start, budget_s) > 30:
        total_restarts = 0
        for beta in [120, 200, 350]:
            if time_remaining(start, budget_s) < 20:
                break
            n_rest = 15
            for r in range(n_rest):
                if time_remaining(start, budget_s) < 8:
                    break
                total_restarts += 1
                h0 = best_h.copy()
                s = np.random.random()

                if s < 0.2:
                    scale = 0.005 + 0.03 * np.exp(-total_restarts / 8.0)
                    h0 += np.random.randn(n) * scale
                elif s < 0.4:
                    pert = np.random.randn(n)
                    pert -= pert.mean()
                    h0 += 0.02 * pert
                elif s < 0.55:
                    nsw = np.random.randint(2, 8)
                    for _ in range(nsw):
                        i, j = np.random.choice(n, 2, replace=False)
                        h0[i], h0[j] = h0[j], h0[i]
                elif s < 0.7:
                    blen = np.random.randint(2, max(3, n // 6))
                    bs = np.random.randint(0, max(1, n - blen))
                    h0[bs:bs + blen] = 1.0 - h0[bs:bs + blen]
                elif s < 0.85:
                    freq = np.random.randint(1, 8)
                    phase = np.random.uniform(0, 2 * np.pi)
                    h0 += 0.015 * np.cos(2 * np.pi * freq * np.arange(n) / n + phase)
                else:
                    # Blend with a Fourier solution
                    p_rand = np.random.randn(n_params) * 0.1
                    p_rand[0::2] = np.abs(p_rand[0::2]) * 0.2
                    h_four = fourier_h(p_rand, n, n_harm)
                    h_four = project_h(h_four, target)
                    alpha = np.random.uniform(0.2, 0.6)
                    h0 = alpha * best_h + (1 - alpha) * h_four

                h0 = project_h(h0, target)

                res = minimize(obj_penalty, h0, args=(dx, target, beta, 100000),
                              method='L-BFGS-B', bounds=bounds,
                              options={'maxiter': 800, 'gtol': 1e-11})
                h_new = project_h(res.x, target)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h, best_c5 = h_new.copy(), c5

    # Phase 5: Try different resolutions
    if time_remaining(start, budget_s) > 50:
        n_best = len(best_h)
        for n_try in [n_best - 15, n_best + 15, n_best + 30, n_best + 50]:
            if time_remaining(start, budget_s) < 25 or n_try < 30:
                continue
            x_old = np.linspace(0, 2, n_best, endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_interp = np.interp(x_new, x_old, best_h, period=2.0)
            h_interp = project_h(h_interp, n_try / 2.0)

            dx_n = 2.0 / n_try
            target_n = n_try / 2.0
            bounds_n = Bounds(np.zeros(n_try), np.ones(n_try))

            for lam in [5000, 50000]:
                if time_remaining(start, budget_s) < 8:
                    break
                res = minimize(obj_penalty, h_interp, args=(dx_n, target_n, 200, lam),
                              method='L-BFGS-B', bounds=bounds_n,
                              options={'maxiter': 1500, 'gtol': 1e-12})
                h_new = project_h(res.x, target_n)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h, best_c5 = h_new.copy(), c5

    # Phase 6: Final high-precision polish
    if time_remaining(start, budget_s) > 8:
        n = len(best_h)
        dx = 2.0 / n
        target = n / 2.0
        bounds = Bounds(np.zeros(n), np.ones(n))
        constraints = [{'type': 'eq', 'fun': con_fun_sum, 'jac': con_jac_sum,
                       'args': (target, n)}]

        for beta in [400, 600]:
            if time_remaining(start, budget_s) < 4:
                break
            res = minimize(obj_smooth, best_h, args=(dx, beta),
                          method='SLSQP', bounds=bounds, constraints=constraints,
                          options={'maxiter': 3000, 'ftol': 1e-15})
            h_new = project_h(res.x, target)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new.copy(), c5

        # One more L-BFGS-B with very high lambda
        if time_remaining(start, budget_s) > 3:
            res = minimize(obj_penalty, best_h, args=(dx, target, 600, 1000000),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': 3000, 'gtol': 1e-14})
            h_new = project_h(res.x, target)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new.copy(), c5

    return best_h, best_c5, len(best_h)
