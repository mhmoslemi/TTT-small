    """Build A and b for the LP: min t s.t. A@h <= t, 0<=h<=1, sum(h)=n/2"""
    two_n_minus_1 = 2 * n - 1
    A = np.zeros((two_n_minus_1, n))
    for m in range(two_n_minus_1):
        offset = m - (n - 1)
        for j in range(n):
            idx = j + offset
            if 0 <= idx < n:
                A[m, j] = g[idx]
    return A

def solve_lp_for_h(g, n):

    target_sum = n / 2.0
    A = build_lp_matrices(g, n)

    c = np.zeros(n + 1)

        elif strategy == 4:
            # Sparse large jump
            mask = np.random.random(n) < 0.12
            h0[mask] = np.clip(h0[mask] + np.random.randn(int(mask.sum())) * 0.15, 0.01, 0.99)
        else:
            # Top-k focused: perturb coordinates most affecting worst lags
            lags = compute_all_lags(h0)
            top_k_idx = np.argsort(lags)[-3:]
            for ki in top_k_idx:
                offset = ki - (n - 1)
                for j in range(n):
                    idx2 = j + offset
                    if 0 <= idx2 < n:
                        h0[j] += np.random.randn() * 0.015

        h0 = project_feasible(h0, target_sum)

        # Try SLSQP
        try:
            beta = 300.0 if attempt % 2 == 0 else 150.0
            res = minimize(obj_smooth, h0, args=(beta,), method='SLSQP',
                           bounds=bounds, constraints=constraints,
                           options={'maxiter': 400, 'ftol': 1e-12})
            h_new = project_feasible(res.x, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new
                no_improve = 0
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

        # Occasionally try LP
        if attempt % 4 == 0 and elapsed_ok(30):
            h_l = solve_lp_for_h(1.0 - h0, n)
            if h_l is not None:
                h_l = project_feasible(h_l, target_sum)
                for alpha in [0.3, 0.5, 0.7, 1.0]:
                    h_try = project_feasible((1 - alpha) * best_h + alpha * h_l, target_sum)
                    c5_try = compute_c5(h_try)
                    if c5_try < best_c5:
                        best_h, best_c5 = h_try, c5_try
                        no_improve = 0

    # Phase 4: Final high-precision polish
    if elapsed_ok(5):
        try:
            res = minimize(obj_smooth, best_h, args=(1000.0,), method='SLSQP',
                           bounds=bounds, constraints=constraints,
                           options={'maxiter': 3000, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new
        except Exception:
            pass

    # Phase 5: Coordinate descent micro-adjustment
    if elapsed_ok(3):
        for _ in range(200):
            if (time.time() - t0) > budget_s - 1:
                break
            improved = False
            for j in range(n):
                for delta in [0.002, -0.002, 0.005, -0.005, 0.001, -0.001]:
                    h_try = best_h.copy()
                    h_try[j] = np.clip(h_try[j] + delta, 0.0, 1.0)
                    h_try = project_feasible(h_try, target_sum)
                    c5_try = compute_c5(h_try)
                    if c5_try < best_c5 - 1e-7:
                        best_h = h_try
                        best_c5 = c5_try
                        improved = True
                        break
            if not improved:
                break

    return best_h, best_c5, n
