            best_h = h.copy()
            best_c5 = c5_new
            no_improve = 0
        else:
            h[i] -= delta
            h[j] += delta
            no_improve += 1
        if step % 20000 == 0 and budget_s - elapsed() > 80:
            try:
                h_try = slsqp_polish(h, n, 5000, bounds, cons, maxiter=500)
                c = compute_c5(h_try, n)
                if c < best_c5:
                    h, best_h, best_c5 = h_try, h_try.copy(), c
                    C0 = get_correlation(h, n)
                    no_improve = 0
            except Exception:
                pass

    # Phase 3: Restart with perturbation if time remains
    if budget_s - elapsed() > 60:
        for attempt in range(5):
            if budget_s - elapsed() < 40:
                break
            scale = rng.choice([0.01, 0.03, 0.05, 0.1, 0.2])
            h_pert = best_h + rng.normal(0, scale, size=n)
            h_pert = project_feasible(h_pert, n)
            # SLSQP from perturbed start
            for alpha in [1000, 10000]:
                if budget_s - elapsed() < 20:
                    break
                try:
                    h_try = slsqp_polish(h_pert, n, alpha, bounds, cons, maxiter=1500)
                    c = compute_c5(h_try, n)
                    if c < best_c5:
                        best_h, best_c5 = h_try, c
                        h_pert = h_try.copy()
                except Exception:
                    pass
            # Pair descent from new start
            if budget_s - elapsed() > 50:
                h_local = h_pert.copy()
                C0_local = get_correlation(h_local, n)
                for _ in range(100000):
                    if budget_s - elapsed() < 50:
                        break
                    ii = rng.randint(n)
                    jj = rng.randint(n)
                    if ii == jj:
                        continue
                    G = compute_G(h_local, ii, jj, n)
                    delta = optimal_delta(C0_local, G, h_local, ii, jj, n)
                    if abs(delta) < 1e-14:
                        continue
                    h_local[ii] += delta
                    h_local[jj] -= delta
                    C0_new = C0_local + delta * G
                    c5_new = float(C0_new.max())
                    if c5_new < best_c5 - 1e-15:
                        C0_local = C0_new
                        best_h = h_local.copy()
                        best_c5 = c5_new
                    else:
                        h_local[ii] -= delta
                        h_local[jj] += delta

    # Phase 4: Final high-precision SLSQP
    if budget_s - elapsed() > 8:
        for alpha in [10000, 50000, 200000]:
            if budget_s - elapsed() < 8:
                break
            try:
                h_try = slsqp_polish(best_h, n, alpha, bounds, cons, maxiter=3000, ftol=1e-18)
                c = compute_c5(h_try, n)
                if c < best_c5:
                    best_h, best_c5 = h_try, c
            except Exception:
                pass

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
