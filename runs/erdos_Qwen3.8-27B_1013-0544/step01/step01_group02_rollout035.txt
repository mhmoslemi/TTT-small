import time

    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-14:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    shifted = alpha * (c - c.max())
    mx = shifted.max()
    if mx > 700:
        val = c.max() + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    else:
        val = c.max() + np.log(np.sum(np.exp(shifted))) / alpha
    return val * dx

def sum_con(h, n):
    return np.sum(h) - n / 2.0

def sum_con_jac(h, n):
    return np.ones(n)

def binary_sa(b_init, n, t0, budget_s, rng, iters=30000):
    b = b_init.copy()
    best_b = b.copy()
    best_c = compute_c5(b, n)
    curr_c = best_c

import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-14:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    shifted = alpha * (c - c.max())
    mx = shifted.max()
    if mx > 700:
        val = c.max() + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    else:
        val = c.max() + np.log(np.sum(np.exp(shifted))) / alpha
    return val * dx

def sum_con(h, n):
    return np.sum(h) - n / 2.0

def sum_con_jac(h, n):
    return np.ones(n)

def compute_grad_smooth(h, n, alpha):
    c = np.correlate(h, 1 - h, mode='full')
    shifted = alpha * (c - c.max())
    w = np.exp(shifted - shifted.max())
    w = w / w.sum()
    offsets = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)
    for i in range(n):
        idx1 = i + offsets
        m1 = (idx1 >= 0) & (idx1 < n)
        term1 = np.dot(w[m1], 1.0 - h[idx1[m1]])
        idx2 = i - offsets
        m2 = (idx2 >= 0) & (idx2 < n)
        term2 = np.dot(w[m2], h[idx2[m2]])
        grad[i] = term1 - term2
    return grad

        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-20})
        return project_feasible(res.x, n)
    except Exception:
        return h

        val = compute_c5(h_new, n)
        if val < best_val:
            best = h_new.copy()
            best_val = val
        h = h_new
    return best, best_val

def binary_sa(b_init, n, t0, budget_s, rng, iters=20000):
    b = b_init.copy()
    best_b = b.copy()
    best_c = compute_c5(b, n)
    curr_c = best_c
    T = 0.05
    cool = (0.001 / T) ** (1.0 / max(iters, 1))
    ones = np.where(b == 1)[0]
    zeros = np.where(b == 0)[0]
    for it in range(iters):
        if budget_s - (time.time() - t0) < 60:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        b[i], b[j] = 0.0, 1.0
        new_c = compute_c5(b, n)
        delta = new_c - curr_c
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            curr_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_b = b.copy()
            ones = np.array([x for x in ones if x != i])
            zeros = np.array([x for x in zeros if x != j])
            if j not in zeros:
                zeros = np.append(zeros, j)
            if i not in ones:
                ones = np.append(ones, i)
        else:
            b[i], b[j] = 1.0, 0.0
        T *= cool
    return best_b, best_c
