def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_box_sum(h, target_sum):
    # Project h onto {x: 0<=x[i]<=1, sum(x)=target_sum}
    # Use bisection on the threshold
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

    # Generate Thue-Morse sequence of length n
    t = np.zeros(n, dtype=int)
    for i in range(1, n):

    return t.astype(float)

def van_der_corput(n, base=2):
    # Generate van der Corput sequence
    v = np.zeros(n)
    for i in range(n):
        f = 1.0
        x = i + 1
        val = 0.0
        while x > 0:
            f /= base
            val += f * (x % base)
            x //= base
        v[i] = val
    return v

def interp_to(h, n_new):

    if n_old == n_new:
        return h.copy()

        x_new = np.arange(n_new, dtype=np.float64) / n_new
        return np.interp(x_new, x_old, h)

        x_new = np.arange(n_new, dtype=np.float64) / n_new
        return np.interp(x_new, x_old, h)

def opt_grad(h, target, deadline, rng, alpha_init=100, top_k_frac=0.08, lr_init=0.004):
    n = len(h)
    best = project_box_sum(h.copy(), target)
    best_c5 = compute_c5(best)
    lr = lr_init
    alpha = alpha_init
    patience = 0
    stall = 0
    
    while time.time() < deadline:
        corr = np.correlate(best, 1.0 - best, mode='full')
        m = float(np.max(corr))
        top_k = max(5, int(len(corr) * top_k_frac))
        idx = np.argsort(corr)[-top_k:]
        w = np.zeros(len(corr))
        w[idx] = np.exp(alpha * (corr[idx] - m))
        sw = np.sum(w)
        if sw > 0:
            w /= sw
        
        grad = np.zeros(n)
        active = np.where(w > 1e-12)[0]
        for ai in active:
            k = int(ai) - (n - 1)
            g = np.zeros(n)
            if k >= 0:
                g[k:] += (1.0 - best[:n-k])
                if k > 0:
                    g[:n-k] -= best[k:]
            else:
                p = -k
                g[:n-p] += (1.0 - best[p:])
                g[p:] -= best[:n-p]
            grad += w[ai] * g
        grad *= (2.0 / n)
        
        h_new = project_box_sum(best - lr * grad, target)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5 - 1e-15:
            best = h_new
            best_c5 = c5_new
            patience = 0
            if lr < 0.025:
                lr *= 1.02
        else:
            patience += 1
            lr *= 0.95
            if patience > 40:
                sz = 0.008 + 0.004 * stall
                h_p = best + rng.randn(n) * sz
                h_p = project_box_sum(h_p, target)
                cp = compute_c5(h_p)
                if cp < best_c5:
                    best = h_p
                    best_c5 = cp
                alpha = min(alpha * 1.35, 350.0)
                lr = lr_init
                patience = 0
                stall += 1
                if stall > 12:
                    break
    return best, best_c5

def binarize_swap_search(h, deadline, rng):

        # Flip the ones with smallest h values

        to_flip = ones_pos[order[:n_ones - target]]
        h_bin[to_flip] = 0
    elif n_ones < target:
        zero_pos = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zero_pos])
        to_flip = zero_pos[order[:target - n_ones]]
        h_bin[to_flip] = 1

    best_c5 = compute_c5(best)
    T = 0.005
    no_improve = 0
    
    while time.time() < deadline:
        ones_idx = np.where(best == 1)[0]
        zeros_idx = np.where(best == 0)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i = int(ones_idx[rng.randint(len(ones_idx))])
        j = int(zeros_idx[rng.randint(len(zeros_idx))])
        new_h = best.copy()
        new_h[i] = 0
        new_h[j] = 1
        new_c5 = compute_c5(new_h)
        
        if new_c5 < best_c5:
            best = new_h
            best_c5 = new_c5

        else:

            T *= 0.5
            no_improve = 0
            if T < 0.0005:

        T = max(T * 0.9995, 0.0005)
