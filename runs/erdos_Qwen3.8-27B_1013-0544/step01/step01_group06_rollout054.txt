import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def get_lag_grad(h, k):
    """Gradient of the k-th lag value with respect to h."""
    n = len(h)
    g = np.zeros(n)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    elif k > 0:
        g[k:] += (1.0 - h[:n-k])
        g[:n-k] -= h[k:]
    else:
        p = -k
        g[:n-p] += (1.0 - h[p:])
        g[p:] -= h[:n-p]
    return g

def compute_objective_and_lags(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return corr.copy()

def project_sum_box(h, target):
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(h - lam, 0.0, 1.0)

def active_gradient(h, corr, top_m, alpha):
    """Compute gradient weighted by top-M active constraints with exponential weighting."""
    n = len(h)
    # Find top M indices
    if len(corr) <= top_m:
        top_idx = np.arange(len(corr))
    else:
        top_idx = np.argpartition(corr, -top_m)[-top_m:]
    
    m = corr[top_idx].max()
    w = np.exp(alpha * (corr[top_idx] - m))
    s = w.sum()
    if s < 1e-300:
        w = np.ones_like(corr[top_idx]) / len(corr[top_idx])
    else:
        w /= s
    
    grad = np.zeros(n)
    for i, idx in enumerate(top_idx):
        if w[i] > 1e-10:
            k = idx - (n - 1)
            grad += w[i] * get_lag_grad(h, k)
    return grad * (2.0 / n)

def optimize_local(h0, target_sum, deadline, rng, alpha=50, top_m=20):
    n = len(h0)
    h = project_sum_box(h0.copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = 0.008
    stall = 0
    restart_count = 0
    max_restarts = 10
    
    while time.time() < deadline:
        corr = compute_objective_and_lags(h)
        grad = active_gradient(h, corr, top_m, alpha)
        
        # Backtracking line search
        step = lr
        improved = False
        for _ in range(8):
            h_try = project_sum_box(h - step * grad, target_sum)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                h = h_try
                best_c5 = c5_try
                best_h = h.copy()
                improved = True
                break
            step *= 0.5
        
        if improved:
            stall = 0
            lr = min(lr * 1.03, 0.05)
            alpha = min(alpha * 1.005, 300)
        else:
            stall += 1
            alpha *= 1.3
            lr *= 0.85
            if stall >= 40:

                    break
                scale = 0.015 * (1 + 0.5 * restart_count)
                h = project_sum_box(best_h + rng.randn(n) * scale, target_sum)
                alpha = 50
                lr = 0.008
                stall = 0
    
    return best_h, best_c5

def try_binary_proj(h, target_sum, rng):
    """Try to improve by projecting to binary and repairing."""
    n = len(h)
    # Threshold at 0.5 with some randomness
    threshold = 0.5 + rng.randn() * 0.05
    h_bin = (h > threshold).astype(float)
    # Repair to satisfy sum constraint
    ones_count = int(h_bin.sum())
    if ones_count != target_sum:
        # Swap some values
        if ones_count > target_sum:
            # Need to change some 1s to 0s
            n_swap = ones_count - int(target_sum)
            one_positions = np.where(h_bin == 1)[0]
            rng.shuffle(one_positions)
            h_bin[one_positions[:n_swap]] = 0
        else:
            n_swap = int(target_sum) - ones_count
            zero_positions = np.where(h_bin == 0)[0]
            rng.shuffle(zero_positions)
            h_bin[zero_positions[:n_swap]] = 1
    return h_bin

        try_optimize(h150, n150, min(deadline4, t0 + budget_s - 60), "n150")

    # Phase 5: Final intensive polish at best resolution
    if best_h is not None and time.time() < t0 + budget_s - 10:
        n_best = len(best_h)
        h_polish = best_h.copy()
        deadline5 = t0 + budget_s - 5
        h_final, c5_final = optimize_local(
            h_polish, n_best / 2.0, deadline5, rng,
            alpha=80, top_m=8, lr_init=0.004
        )
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final

    # Final validation
    n_out = len(best_h)
    best_h = project_sum_box(best_h, n_out / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n_out
