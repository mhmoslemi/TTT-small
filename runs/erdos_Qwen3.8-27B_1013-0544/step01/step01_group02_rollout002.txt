    """Optimize h[i] and h[j] with h[i]+h[j] fixed"""
    s = h[i] + h[j]
    def c5_of_t(t):
        hh = h.copy()
        hh[i] = t
        hh[j] = s - t
        return compute_c5(hh, n)
    # Ternary search on [max(0, s-1), min(1, s)]
    lo = max(0.0, s - 1.0)
    hi = min(1.0, s)
    if hi - lo < 1e-10:

    # Since C5 is convex in t, use golden section search
    gr = (np.sqrt(5) - 1) / 2
    a, b = lo, hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = c5_of_t(c)
    fd = c5_of_t(d)

            d = c
            fd = fc

            c = d
            fc = fd
            d = a + gr * (b - a)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-14:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max(obj, alpha):
    m = obj.max()
    shifted = alpha * (obj - m)
    mx = shifted.max()
    if mx > 700:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def obj_lse(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return smooth_max(c, alpha)

def sum_con_fun(x, n):
    return np.sum(x) - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)

def active_set_obj(x, n, active_lags, dx, alpha):
    c = np.correlate(x, 1 - x, mode='full') * dx
    return smooth_max(c[active_lags], alpha)

def two_coord_optimize(h, n, i, j):
    """Exact 2-coordinate optimization via golden section search."""
    s = h[i] + h[j]
    lo = max(0.0, s - 1.0)

    if hi - lo < 1e-12:
        return h, compute_c5(h, n)
    
    def c5_of_t(t):
        hh = h.copy()
        hh[i] = t
        hh[j] = s - t
        return compute_c5(hh, n)
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = c5_of_t(c)
    fd = c5_of_t(d)
    for _ in range(80):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = c5_of_t(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = c5_of_t(d)
        if abs(b - a) < 1e-12:
            break
    t_opt = (a + b) / 2.0
    h_new = h.copy()
    h_new[i] = t_opt
    h_new[j] = s - t_opt
    c_opt = compute_c5(h_new, n)
    return h_new, c_opt
