import numpy as np
import time
from scipy.ndimage import gaussian_filter1d

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def compute_grad(h, alpha, top_k=30):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    m = np.max(corr)
    # Use top_k weights
    if top_k < len(corr):
        idx_top = np.argsort(corr)[-top_k:]
        w = np.zeros(len(corr))
        w[idx_top] = np.exp(alpha * (corr[idx_top] - m))
        w /= w.sum()
    else:
        w = np.exp(alpha * (corr - m))
        s = w.sum()
        if s < 1e-300:
            w = np.zeros_like(corr)
            w[np.argmax(corr)] = 1.0
        else:
            w /= s
    grad = np.zeros(n)
    for idx in np.where(w > 1e-12)[0]:
        wi = w[idx]
        k = idx - (n - 1)
        if k >= 0:
            grad[k:] += wi * (1.0 - h[:n-k])
            if k > 0:
                grad[:n-k] -= wi * h[k:]
        else:
            p = -k
            grad[:n-p] += wi * (1.0 - h[p:])
            if p > 0:
                grad[p:] -= wi * h[:n-p]
    return grad * dx

    lo, hi = -1.0, 2.0
    for _ in range(50):

        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:

        else:

    pert = rng.randn(n) * scale
    # Circular smoothing
    kernel_size = 2 * int(sigma * 3) + 1
    kernel = np.exp(-0.5 * (np.arange(kernel_size) - kernel_size//2)**2 / (sigma**2))
    kernel /= kernel.sum()
    pert = np.convolve(pert, kernel, mode='same')
    return pert

    h = project_simplex_box(h0.astype(float).copy(), target_sum)

    best_c5 = compute_c5(best_h)

    beta = 0.9
    lr = 0.015

    stall = 0

    return np.interp(x_new, x_ext, h_ext)

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def compute_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= s
    grad = np.zeros(n)
    for idx in np.where(w > 1e-10)[0]:
        wi = w[idx]
        k = idx - (n - 1)
        if k >= 0:
            grad[k:] += wi * (1.0 - h[:n-k])
            if k > 0:
                grad[:n-k] -= wi * h[k:]
        else:
            p = -k
            grad[:n-p] += wi * (1.0 - h[p:])
            if p > 0:
                grad[p:] -= wi * h[:n-p]
    return grad * dx

def project_simplex_box(h, target_sum):
    lo, hi = -1.5, 1.5
    for _ in range(50):

        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)
