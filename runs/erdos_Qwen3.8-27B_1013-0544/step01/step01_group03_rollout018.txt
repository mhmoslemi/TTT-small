import numpy as np
from scipy.optimize import minimize, Bounds
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(corr, tau):
    x = tau * corr
    x_max = np.max(x)
    return x_max / tau + np.log(np.sum(np.exp(x - x_max))) / tau

def smooth_max_grad(corr, h, tau):
    """Gradient of smooth_max(corr, tau) w.r.t. h"""
    n = len(h)
    x = tau * corr
    x_max = np.max(x)
    exp_x = np.exp(x - x_max)
    weights = exp_x / (tau * np.sum(exp_x))
    
    g = np.zeros(n)
    # For each lag m, compute contribution
    # out[m] = sum_i h[i] * (1-h[i+m-(n-1)])
    # grad w.r.t. h[p]:
    #   (1-h[p+m-(n-1)]) if 0<=p+m-(n-1)<n
    #   -h[p-(m-(n-1))] if 0<=p-(m-(n-1))<n
    for m in range(len(corr)):
        w = weights[m]
        if w < 1e-8:
            continue
        shift = m - (n - 1)
        # p + shift is valid: 0 <= p+shift < n → -shift <= p < n-shift
        p1_lo = max(0, -shift)
        p1_hi = min(n, n - shift)
        if p1_hi > p1_lo:
            g[p1_lo:p1_hi] += w * (1.0 - h[p1_lo + shift : p1_hi + shift])
        # p - shift is valid: 0 <= p-shift < n → shift <= p < n+shift
        p2_lo = max(0, shift)
        p2_hi = min(n, n + shift)
        if p2_hi > p2_lo:
            g[p2_lo:p2_hi] -= w * h[p2_lo - shift : p2_hi - shift]
    return g

def project_bounded_simplex(y, target):
    """Project y onto {h: sum(h)=target, 0<=h<=1}"""
    n = len(y)
    lo = np.max(y) - 1.0
    hi = np.min(y) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if abs(s - target) < 1e-10:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def fix_sum(h, target):
    h = np.array(h, dtype=float)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return np.clip(h, 0.0, 1.0)
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = mask.sum()
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def subgradient_optimize(h, target, n_steps=1000, tau=None, step_init=0.01, step_decay=0.999):
    """Subgradient method with projection"""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    step = step_init
    
    for t in range(n_steps):
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        if tau is None:
            m_star = np.argmax(corr)
            g = _single_corr_grad(h, m_star) * dx
        else:
            g = smooth_max_grad(corr, h, tau) * dx
        
        if np.max(np.abs(g)) < 1e-15:
            break
        
        h = project_bounded_simplex(h - step * g / np.max(np.abs(g)), target)
        step *= step_decay
    
    return best_h, best_c5

def _single_corr_grad(h, m):
    """Gradient of corr[m] w.r.t. h"""
    n = len(h)
    shift = m - (n - 1)
    g = np.zeros(n)
    p1_lo = max(0, -shift)
    p1_hi = min(n, n - shift)
    if p1_hi > p1_lo:
        g[p1_lo:p1_hi] += (1.0 - h[p1_lo + shift : p1_hi + shift])
    p2_lo = max(0, shift)
    p2_hi = min(n, n + shift)
    if p2_hi > p2_lo:
        g[p2_lo:p2_hi] -= h[p2_lo - shift : p2_hi - shift]
    return g

            try:

                              constraints=cons, options={'maxiter': 200, 'ftol': 1e-14})
                x = fix_sum(res.x, target)
                c5 = evaluate_c5(x)
                if c5 < best_c5:
                    best_h, best_c5 = x, c5
            except:
                pass

    for tn in [60, 80, 100, 140, 160]:

    total = float(np.sum(exp_x))
    g = np.zeros(n)
    for m in range(len(corr)):
        w = exp_x[m] / (tau * total)
        if w < 1e-10:
            continue
        shift = m - (n - 1)
        lo1 = max(0, -shift)
        hi1 = min(n, n - shift)
        if hi1 > lo1:
            g[lo1:hi1] += w * (1.0 - h[lo1 + shift:hi1 + shift])
        lo2 = max(0, shift)
        hi2 = min(n, n + shift)
        if hi2 > lo2:
            g[lo2:hi2] -= w * h[lo2 - shift:hi2 - shift]
    return g


def subgradient_optimize(h, target, n_steps, tau, step_init, step_decay, momentum=0.0):
    n = len(h)
    dx = 2.0 / n
    h = np.array(h, dtype=float)
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    step = step_init

    for t in range(n_steps):
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = float(np.max(corr)) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        if tau <= 1:
            m_star = int(np.argmax(corr))
            g = corr_lag_grad(h, m_star) * dx
        else:
            g = smooth_max_gradient(corr, h, tau) * dx

        g_norm = float(np.max(np.abs(g)))
        if g_norm < 1e-15:
            break

        if momentum > 0:
            v = momentum * v - step * g / g_norm
            h = project_bounded_simplex(h + v, target)
        else:
            h = project_bounded_simplex(h - step * g / g_norm, target)
        step *= step_decay

    c5_final = evaluate_c5(h)
    if c5_final < best_c5:
        best_h, best_c5 = h.copy(), c5_final
    return best_h, best_c5


class SumConstraint:
    def __init__(self, target):
        self.target = target

    def fun(self, x):
        return float(np.sum(x)) - self.target

    def jac(self, x):
        return np.ones(len(x))


class SmoothMaxObj:
    def __init__(self, tau):
        self.tau = tau

    def __call__(self, x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        return smooth_max_val(corr, self.tau)


def slsqp_optimize(h, target, tau, maxiter=200):
    n = len(h)
    bnds = Bounds(np.zeros(n), np.ones(n))
    obj_fn = SmoothMaxObj(tau)
    cons_obj = SumConstraint(target)
    cons = [{'type': 'eq', 'fun': cons_obj.fun, 'jac': cons_obj.jac}]
    try:
        res = minimize(obj_fn, h, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        x = fix_sum(res.x, target)
        return x
    except Exception:
        return h


def greedy_binary_init(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    ones = set(rng.choice(n, n // 2, replace=False).tolist())
    for i in ones:
        h[i] = 1.0
    best_c5 = evaluate_c5(h)
    for _ in range(200):
        improved = False
        best_flip = -1
        best_val = best_c5
        for flip in range(n):
            h[flip] = 1.0 - h[flip]
            c5 = evaluate_c5(h)
            if c5 < best_val - 1e-12:
                best_val = c5
                best_flip = flip
            h[flip] = 1.0 - h[flip]
        if best_flip >= 0:
            h[best_flip] = 1.0 - h[best_flip]
            best_c5 = best_val
            improved = True
        if not improved:
            break
    return h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)

    best_h = h_init.copy()
    best_c5 = evaluate_c5(best_h)
    n0 = len(best_h)
    target0 = n0 / 2.0

    def time_remaining():
        return budget_s - (time.time() - t0)

    # Phase 1: Subgradient from initial (smooth)
    if time_remaining() > 80:
        h_new, c5_new = subgradient_optimize(best_h, target0, 4000, tau=30,
                                             step_init=0.06, step_decay=0.9997, momentum=0.7)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 2: SLSQP polish
    if time_remaining() > 60:
        for tau in [200, 1000, 3000]:
            if time_remaining() < 15:
                break
            h_new = slsqp_optimize(best_h, target0, tau, 150)
            h_new = fix_sum(h_new, target0)
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new

    # Phase 3: Hard subgradient for exploration
    if time_remaining() > 60:
        h_new, c5_new = subgradient_optimize(best_h, target0, 6000, tau=1,
                                             step_init=0.02, step_decay=0.99995, momentum=0.4)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 4: Resolution sweep
    resolutions = [60, 80, 100, 140, 160, 200]
    for tn in resolutions:
        if time_remaining() < 60:
            break
        target_t = tn / 2.0
        h_t = np.interp(np.linspace(0, 1, tn), np.linspace(0, 1, len(best_h)), best_h)
        h_t = fix_sum(h_t, target_t)
        c5_t = evaluate_c5(h_t)

        if time_remaining() > 40:
            h_new, c5_new = subgradient_optimize(h_t, target_t, 2000, tau=20,
                                                 step_init=0.05, step_decay=0.9997, momentum=0.6)
            if c5_new < c5_t:
                h_t, c5_t = h_new, c5_new

        if time_remaining() > 25 and c5_t < best_c5 + 0.02:
            h_new = slsqp_optimize(h_t, target_t, 500, 120)
            h_new = fix_sum(h_new, target_t)
            c5_new = evaluate_c5(h_new)
            if c5_new < c5_t:
                h_t, c5_t = h_new, c5_new

        if c5_t < best_c5:
            best_h, best_c5 = h_t, c5_t

    # Phase 5: Random restarts with subgradient
    hn = len(best_h)
    for i in range(40):
        if time_remaining() < 15:
            break
        scale = 0.03 + 0.12 * np.random.random()
        hp = best_h + np.random.normal(0, scale, hn)
        hp = project_bounded_simplex(hp, hn / 2.0)
        h_new, c5_new = subgradient_optimize(hp, hn / 2.0, 1000, tau=20,
                                             step_init=0.06, step_decay=0.9995, momentum=0.6)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new
            if time_remaining() > 20:
                h_pol = slsqp_optimize(best_h, hn / 2.0, 1000, 100)
                h_pol = fix_sum(h_pol, hn / 2.0)
                c5_pol = evaluate_c5(h_pol)
                if c5_pol < best_c5:
                    best_h, best_c5 = h_pol, c5_pol

    # Phase 6: Binary-inspired restarts
    if time_remaining() > 80:
        for s in [seed + 100, seed + 200, seed + 300]:
            if time_remaining() < 30:
                break
            h_bin, c5_bin = greedy_binary_init(hn, s)
            # Smooth the binary solution
            h_smooth = h_bin.copy()
            # Interpolate with slight smoothing
            kernel = np.array([0.25, 0.5, 0.25])
            h_kernel = np.convolve(h_smooth, kernel, mode='same')
            h_kernel = project_bounded_simplex(h_kernel, hn / 2.0)
            h_new, c5_new = subgradient_optimize(h_kernel, hn / 2.0, 2000,
                                                 tau=20, step_init=0.05, step_decay=0.9997, momentum=0.6)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new
                if time_remaining() > 15:
                    h_pol = slsqp_optimize(best_h, hn / 2.0, 1000, 80)
                    h_pol = fix_sum(h_pol, hn / 2.0)
                    c5_pol = evaluate_c5(h_pol)
                    if c5_pol < best_c5:
                        best_h, best_c5 = h_pol, c5_pol

    # Phase 7: Final fine-tuning with very small steps
    if time_remaining() > 30:
        h_new, c5_new = subgradient_optimize(best_h, len(best_h) / 2.0, 5000,
                                             tau=10, step_init=0.008, step_decay=0.99997, momentum=0.85)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 8: Final SLSQP polish
    if time_remaining() > 15:
        h_pol = slsqp_optimize(best_h, len(best_h) / 2.0, 5000, 100)
        h_pol = fix_sum(h_pol, len(best_h) / 2.0)
        c5_pol = evaluate_c5(h_pol)
        if c5_pol < best_c5:
            best_h, best_c5 = h_pol, c5_pol

    # Final constraint fix
    best_h = fix_sum(best_h, len(best_h) / 2.0)
    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, len(best_h)
