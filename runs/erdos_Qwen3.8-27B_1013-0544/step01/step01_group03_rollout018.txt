Let me try a **different optimization strategy**: use the fact that the problem can be formulated as a min-max problem and solve it iteratively.

**Minimax formulation**: Minimize t subject to c_k(h) ≤ t for all k, 0 ≤ h ≤ 1, sum(h) = n/2.

This is a non-convex QCQP (quadratically constrained quadratic program) because c_k(h) is bilinear. But we can use a sequence of convex problems:

At each iteration, linearize c_k(h) around the current h:
c_k(h) ≈ c_k(h_old) + ∇c_k(h_old)ᵀ(h - h_old)

Then solve: minimize t s.t. c_k(h_old) + ∇c_k(h_old)ᵀ(h - h_old) ≤ t for all k, 0 ≤ h ≤ 1, sum(h) = n/2.

This is a linear program! And we can solve it efficiently.

This is the **cutting plane method** or **linearization approach** for the non-convex problem.

Actually, this is essentially the same as the subgradient method but solved optimally at each step.

Let me implement this:

```python
def llam_iteration(h, target, n):
    """One iteration of the linearization (LLM) approach."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    t = np.max(corr)
    
    # Build grad matrix: G[k, j] = dc_k/dh[j] at current h
    # corr has 2n-1 elements, indexed 0 to 2n-2
    # corr[k] = sum_{valid i} h[i] * (1-h[i + n-1-k])
    
    G = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        s = n - 1 - k
        G[k, :] = 1.0
        if s > 0:
            G[k, :n-s] -= h[s:]
            G[k, s:] -= h[:n-s]
        elif s < 0:
            ss = -s
            G[k, ss:] -= h[:n-ss]
            G[k, :n-ss] -= h[ss:]
        else:
            G[k, :] = 1.0 - 2.0 * h
    
    # Solve: min t s.t. corr[k] + G[k,:]·(h_new - h) <= t, 0<=h<=1, sum(h)=target
    # Variables: h_new (n) + t (1) = n+1 variables
    # Constraints: (2n-1) + n (bounds) + 1 (sum)
    
    from scipy.optimize import linprog
    # min [0, ..., 0, 1] · [h_new, t]
    # s.t. G[k,:]*h_new - t <= corr[k] - G[k,:]*h  for each k
    #      -I <= h_new <= I
    #      sum(h_new) = target
    
    c_lin = np.zeros(n+1)
    c_lin[-1] = 1.0
    
    # Inequality constraints: G[k,:]*h - t <= corr[k] - G[k,:]*h_old
    # => G[k,:]*h - t - (corr[k] - G[k,:]*h_old) <= 0
    A_ub = np.zeros((2*n-1, n+1))
    b_ub = np.zeros(2*n-1)
    for k in range(2*n-1):
        A_ub[k, :n] = G[k, :]
        A_ub[k, n] = -1.0
        b_ub[k] = corr[k] - G[k, :] @ h
    
    # Bounds: 0 <= h_new[j] <= 1
    bounds = [(0, 1)] * n + [(None, None)]
    
    # Equality: sum(h_new) = target
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    res = linprog(c_lin, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    return res.x[:n], res.x[n]
```

import numpy as np
from scipy.optimize import minimize, Bounds, linprog
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target):
    h = np.array(h, dtype=float)
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max(corr, tau):
    mx = corr.max()
    return float(mx + np.log(np.sum(np.exp(tau * (corr - mx)))) / tau)

def compute_corr_grad(h, k_idx=None):
    """Compute gradient of c_k wrt h. If k_idx is None, return all gradients."""
    n = len(h)
    if k_idx is not None:
        s = n - 1 - k_idx
        grad = np.ones(n)
        if s > 0:
            grad[:n-s] -= h[s:]
            grad[s:] -= h[:n-s]
        elif s < 0:
            ss = -s
            grad[ss:] -= h[:n-ss]
            grad[:n-ss] -= h[ss:]
        else:
            grad = 1.0 - 2.0 * h
        return grad
    else:
        G = np.zeros((2*n-1, n))
        for k in range(2*n-1):
            G[k] = compute_corr_grad(h, k)
        return G

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    n = len(h_init)

    h_init = project_feasible(h_init, target)
    
    best_h = h_init.copy()
    best_c5 = evaluate_c5(best_h)
    
    def rem():
        return budget_s - (time.time() - t0)
    
    # Phase 1: Subgradient method (fast, many iterations)

        if rem() < 300:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        k_max = np.argmax(corr)
        grad = compute_corr_grad(best_h, k_max)
        h_new = project_feasible(best_h - alpha * grad, target)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
        alpha *= decay
    
    # Phase 2: SLSQP with analytic gradient

    for tau in [50, 200, 1000, 5000]:
        if rem() < 60:

        def obj_s(x):
            corr = np.correlate(x, 1.0 - x, mode="full")
            return smooth_max(corr, tau)
        def obj_g(x):

            mx = corr.max()
            w = np.exp(tau * (corr - mx))

            for kk in range(2*n-1):

            h_new = project_feasible(res.x, target)
            c5 = evaluate_c5(h_new)
            if c5 < best_c5:
                best_h = h_new
                best_c5 = c5
        except:

    # Phase 3: LLM (Linearized Minimax)
    for it in range(50):
        if rem() < 30:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        t_cur = np.max(corr)
        G = compute_corr_grad(best_h)
        
        c_lin = np.zeros(n + 1)
        c_lin[-1] = 1.0
        A_ub = np.zeros((2*n-1, n+1))

        bounds_lp = [(0.0, 1.0)] * n + [(None, None)]

        A_eq[0, :n] = 1.0
        b_eq = np.array([target])
        
        try:

                         bounds=bounds_lp, method='highs')
            if res.success:

                c5 = evaluate_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new
                    best_c5 = c5
        except:
            break
    
    # Phase 4: Diverse restarts + SLSQP
    starts = []
    for trial in range(5):

                return smooth_max(corr, tau)

            try:
                res = minimize(obj_r, h_s, method='SLSQP', bounds=bnds,
                              constraints=cons_r, options={'maxiter': 150, 'ftol': 1e-13})
                h_new = project_feasible(res.x, target)
                c5 = evaluate_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new
                    best_c5 = c5

            except:

    # Phase 5: Final polish

            except:

    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, n
