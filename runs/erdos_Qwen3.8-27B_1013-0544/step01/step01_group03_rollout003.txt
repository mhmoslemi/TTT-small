    corr = np.correlate(h, 1.0 - h, mode="full")
    # Top k
    top_idx = np.argpartition(corr, -k)[-k:]
    top_vals = corr[top_idx]
    # Weighted sum (uniform weights)
    obj = top_vals.mean()
    s = h.sum()
    obj += penalty * (s - target_sum) ** 2
    return obj

def top_k_gradient(h, k, penalty, target_sum):

    corr = np.correlate(h, 1.0 - h, mode="full")

    grad = np.zeros(n)

        m_hi = min(n, n - s)
        grad[m_lo:m_hi] += (1 - h[m_lo+s:m_hi+s]) / k

        grad[m_lo2:m_hi2] -= h[m_lo2-s:m_hi2-s] / k
