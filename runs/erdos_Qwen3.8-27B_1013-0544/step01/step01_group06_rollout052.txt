import numpy as np
import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def project_box_sum(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def make_binary(h, target_sum, rng):
    h_bin = (h >= 0.5).astype(float)
    n1 = int(h_bin.sum())
    ts = int(target_sum)
    if n1 > ts:
        idx = np.where(h_bin == 1)[0]
        order = np.argsort(h[idx])
        h_bin[idx[order[:n1 - ts]]] = 0
    elif n1 < ts:
        idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h[idx])
        h_bin[idx[order[:ts - n1]]] = 1
    return h_bin

def random_binary(n, rng):
    h = np.zeros(n)
    pos = rng.permutation(n)
    h[pos[:n // 2]] = 1.0
    return h

def thue_morse_binary(n, rng):
    tm = np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)
    n1 = int(tm.sum())
    ts = n // 2
    if n1 != ts:
        if n1 > ts:
            idx = np.where(tm == 1)[0]
            perm = rng.permutation(len(idx))
            tm[idx[perm[:n1 - ts]]] = 0
        else:
            idx = np.where(tm == 0)[0]
            perm = rng.permutation(len(idx))
            tm[idx[perm[:ts - n1]]] = 1
    return tm

def local_search(h, n, deadline, rng, max_batch=300):
    best = h.copy()
    best_c5 = compute_c5(best)
    ones = np.where(best == 1)[0].copy()
    zeros = np.where(best == 0)[0].copy()
    no_improve = 0

    while time.time() < deadline:
        found = False
        for _ in range(max_batch):
            if time.time() > deadline:
                return best, best_c5
            i = rng.integers(len(ones))
            j = rng.integers(len(zeros))
            pi, pj = int(ones[i]), int(zeros[j])
            best[pi], best[pj] = 0.0, 1.0
            c5 = compute_c5(best)
            if c5 < best_c5:
                best_c5 = c5
                no_improve = 0
                found = True
                ones = np.where(best == 1)[0]
                zeros = np.where(best == 0)[0]
            else:
                best[pi], best[pj] = 1.0, 0.0

        if not found:
            no_improve += 1
            if no_improve >= 2:
                k = int(rng.integers(2, 5))
                for _ in range(k):
                    i = rng.integers(len(ones))
                    j = rng.integers(len(zeros))
                    best[ones[i]] = 0.0
                    best[zeros[j]] = 1.0
                ones = np.where(best == 1)[0]
                zeros = np.where(best == 0)[0]
                no_improve = 0
    return best, best_c5

def steepest_descent(h, n, deadline, max_rounds=30):
    best = h.copy()
    best_c5 = compute_c5(best)
    ones = np.where(best == 1)[0].copy()
    zeros = np.where(best == 0)[0].copy()

    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        best_swap = None
        best_c = best_c5
        n_ones = len(ones)
        n_zeros = len(zeros)
        step = max(1, min(n_ones, n_zeros) // 3)
        for ii in range(0, n_ones, step):
            for jj in range(0, n_zeros, step):
                if time.time() > deadline:
                    break
                pi, pj = int(ones[ii]), int(zeros[jj])
                best[pi], best[pj] = 0.0, 1.0
                c5 = compute_c5(best)
                if c5 < best_c:
                    best_c = c5
                    best_swap = (pi, pj)
                best[pi], best[pj] = 1.0, 0.0
            if time.time() > deadline:
                break

        if best_swap is not None:
            pi, pj = best_swap
            best[pi], best[pj] = 0.0, 1.0
            best_c5 = best_c
            ones = np.where(best == 1)[0]
            zeros = np.where(best == 0)[0]
        else:
            break
    return best, best_c5

def cont_optimize(h0, target_sum, deadline, rng):
    n = len(h0)
    h = project_box_sum(h0.astype(float).copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    lr = 0.004
    alpha = 120.0
    stall = 0

    while time.time() < deadline:
        corr = np.correlate(h, 1.0 - h, mode='full')
        m = np.max(corr)
        w = np.exp(alpha * (corr - m))
        s = w.sum()
        if s > 1e-300:
            w /= s
        else:
            w = np.zeros_like(corr)
            w[int(np.argmax(corr))] = 1.0

        grad = np.zeros(n)
        active = np.where(w > 1e-7)[0]
        for idx in active:
            k = int(idx) - (n - 1)
            if k >= 0:
                grad[k:] += w[idx] * (1.0 - h[:n - k])
                if k > 0:
                    grad[:n - k] -= w[idx] * h[k:]
            else:
                p = -k
                grad[:n - p] += w[idx] * (1.0 - h[p:])
                grad[p:] -= w[idx] * h[:n - p]
        grad *= (2.0 / n)

        h_new = project_box_sum(h - lr * grad, target_sum)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5 - 1e-15:
            best_h = h_new.copy()
            best_c5 = c5_new
            h = h_new
            stall = 0
            lr = min(lr * 1.015, 0.04)
        else:
            stall += 1
            h = h_new
            lr *= 0.93
            if stall > 80:
                h = project_box_sum(h + rng.randn(n) * 0.03, target_sum)
                alpha = min(alpha * 1.3, 600.0)
                lr = 0.004
                stall = 0

    return best_h, best_c5

def interpolate(h, n_old, n_new):
    x_old = np.arange(n_old) + 0.5
    x_new = np.arange(n_new) + 0.5
    h_ext = np.append(h, h[0])
    x_ext = np.append(x_old, n_old + 0.5)
    return np.interp(x_new, x_ext, h_ext)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent = np.array(initial_h_values, dtype=float)
    else:
        parent = None

    best_h = None
    best_c5 = float('inf')

    # Phase 1: Binary local search at parent resolution (40% time)
    n_parent = len(parent) if parent is not None else 200
    d1 = t0 + budget_s * 0.35

    if parent is not None:
        h_bin = make_binary(parent, n_parent / 2.0, rng)
        hb, cb = local_search(h_bin, n_parent, d1, rng)
        if cb < best_c5:
            best_c5 = cb
            best_h = hb.copy()
        hs, cs = steepest_descent(best_h, n_parent, t0 + budget_s * 0.4, max_rounds=20)
        if cs < best_c5:
            best_c5 = cs
            best_h = hs.copy()

    # Phase 2: Other resolutions (30% time)
    d2 = t0 + budget_s * 0.65
    tried_n = {n_parent}
    for n_try in [120, 160, 200, 240, 280]:
        if n_try in tried_n or time.time() > d2:
            continue
        tried_n.add(n_try)
        t_sub = min(time.time() + 50, d2)
        # Try both random and Thue-Morse init
        for init_fn in [random_binary, thue_morse_binary]:
            if time.time() > t_sub:
                break
            h_init = init_fn(n_try, rng)
            t_ls = min(time.time() + 30, d2)
            hb, cb = local_search(h_init, n_try, t_ls, rng)
            if cb < best_c5:
                best_c5 = cb
                best_h = hb.copy()
            if time.time() < d2 - 5:
                hs, cs = steepest_descent(hb, n_try, min(time.time() + 15, d2), max_rounds=10)
                if cs < best_c5:
                    best_c5 = cs
                    best_h = hs.copy()

    # Phase 3: Continuous optimization from best binary (20% time)
    d3 = t0 + budget_s * 0.85
    if best_h is not None:
        n_cur = len(best_h)
        t_cont = min(d3, time.time() + budget_s * 0.2)
        hc, cc = cont_optimize(best_h, n_cur / 2.0, t_cont, rng)
        if cc < best_c5:
            best_c5 = cc
            best_h = hc

    # Phase 4: Try higher resolution with interpolation (10% time)
    d4 = t0 + budget_s * 0.95
    if best_h is not None and time.time() < d4:
        n_new = len(best_h) + 40
        h_interp = interpolate(best_h, len(best_h), n_new)
        h_interp = project_box_sum(h_interp, n_new / 2.0)
        t_fine = d4
        hc, cc = cont_optimize(h_interp, n_new / 2.0, t_fine, rng)
        if cc < best_c5:
            best_c5 = cc
            best_h = hc

    # Final: one more polish at best resolution
    if best_h is not None and time.time() < t0 + budget_s - 10:
        n_fin = len(best_h)
        t_fin = t0 + budget_s - 5
        hf, cf = cont_optimize(best_h, n_fin / 2.0, t_fin, rng)
        if cf < best_c5:
            best_c5 = cf
            best_h = hf

    if best_h is None:
        n = 200
        best_h = random_binary(n, rng)
        best_c5 = compute_c5(best_h)

    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
