    cons = [{'type': 'eq', 'fun': slsqp_cons_fun, 'jac': slsqp_cons_jac}]
    try:
        res = minimize(slsqp_obj_fn, h, args=(tau,), method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_out = project_box_sum(res.x, n / 2.0)
        return h_out, evaluate_c5(h_out)
    except Exception:
        return h, evaluate_c5(h)

def make_structured(n, pattern):
    x = np.linspace(0, 2, n, endpoint=False)
    if pattern == 0:
        h = np.full(n, 0.5)
    elif pattern == 1:
        h = (1.0 + np.sin(np.pi * x)) / 2.0
    elif pattern == 2:
        h = (1.0 + np.sin(2 * np.pi * x)) / 2.0
    elif pattern == 3:
        phi = (1 + np.sqrt(5)) / 2
        h = ((x * phi).astype(float) % 1.0 < 0.5).astype(float)
    elif pattern == 4:
        h = (1.0 + np.cos(2 * np.pi * x)) / 2.0
    else:
        h = np.full(n, 0.5)
    return project_box_sum(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def rem():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(120, 0.5)

    n0 = len(h_init)
    best_h = h_init.copy()
    best_c5 = evaluate_c5(best_h)

    # Phase 1: Annealed gradient from initial
    tau_sched = [10, 30, 100, 300, 1000, 3000, 10000]
    for tau in tau_sched:
        if rem() < 15:
            break
        h_new, c5_new = gradient_phase(best_h, tau, 400, 0.5 / max(tau, 1), rem)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 2: Multi-resolution exploration
    try_ns = [80, 100, 120, 140, 160]
    for tn in try_ns:
        if rem() < 25:
            break
        if tn % 2 != 0:
            continue
        if len(best_h) == tn:
            h_t = best_h.copy()
        else:
            h_t = np.interp(np.linspace(0, 1, tn), np.linspace(0, 1, len(best_h)), best_h)
            h_t = project_box_sum(h_t, tn / 2.0)
        for tau in [100, 1000, 10000]:
            if rem() < 12:
                break
            h_new, c5_new = gradient_phase(h_t, tau, 300, 0.5 / max(tau, 1), rem)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
            h_t = h_new

    # Phase 3: Structured seeds at best resolution
    hn = len(best_h)
    for pat in range(5):
        if rem() < 20:
            break
        h_s = make_structured(hn, pat)
        # Blend with current best
        blend = 0.5 * h_s + 0.5 * best_h
        blend = project_box_sum(blend, hn / 2.0)
        for tau in [100, 1000]:
            if rem() < 10:
                break
            h_new, c5_new = gradient_phase(blend, tau, 250, 0.5, rem)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
            blend = h_new

    # Phase 4: SLSQP polish
    if rem() > 20:
        hn2 = len(best_h)
        h_pol, c5_pol = slsqp_polish(best_h, 20000, 300)
        if c5_pol < best_c5:
            best_h = h_pol
            best_c5 = c5_pol
        h_pol2, c5_pol2 = slsqp_polish(best_h, 50000, 200)
        if c5_pol2 < best_c5:
            best_h = h_pol2
            best_c5 = c5_pol2

    # Phase 5: Final high-tau gradient refinement
    if rem() > 10:
        hn3 = len(best_h)
        for tau in [50000, 100000]:
            if rem() < 8:
                break
            h_new, c5_new = gradient_phase(best_h, tau, 150, 0.05, rem)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    # Phase 6: Final SLSQP
    if rem() > 8:
        h_pol3, c5_pol3 = slsqp_polish(best_h, 100000, 100)
        if c5_pol3 < best_c5:
            best_h = h_pol3
            best_c5 = c5_pol3

    hn_final = len(best_h)
    best_h = project_box_sum(best_h, hn_final / 2.0)
    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, hn_final
