def evaluate_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_compute(corr, tau):
    x = tau * corr
    x_max = x.max()
    sm = x_max / tau + np.log(np.sum(np.exp(x - x_max))) / tau
    # Weights for gradient
    weights = np.exp(x - x_max) / np.sum(np.exp(x - x_max))
    return float(sm), weights

def project_feasible(h, target_sum):
    h = np.array(h, dtype=float)
    # First clip
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    # Bisection on uniform shift
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_test = np.clip(h + mid, 0.0, 1.0)
        if h_test.sum() < target_sum:

        else:

            h = np.clip(h, 0.0, 1.0)

    h_new = h - lr * grad
    h_new = project_feasible(h_new, n / 2.0)
    return h_new, float(np.max(corr) * (2.0/n))
