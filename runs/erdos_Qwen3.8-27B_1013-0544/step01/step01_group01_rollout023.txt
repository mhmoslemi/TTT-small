```python
    weights = exp_corr / np.sum(exp_corr)
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        wk = weights[k]
        if wk < 1e-12:
            continue
        # First term: (1-h[j-lag]) for valid j
        # j ranges such that 0 <= j-lag < n, i.e., lag <= j < n+lag
        j_start = max(0, lag)
        j_end = min(n, n + lag)
        if j_start < j_end:
            idx = np.arange(j_start, j_end)
            grad[idx] += wk * (1.0 - h[idx - lag])
        # Second term: -h[j+lag] for valid j
        # j ranges such that 0 <= j+lag < n, i.e., -lag <= j < n-lag
        j_start2 = max(0, -lag)
        j_end2 = min(n, n - lag)
        if j_start2 < j_end2:
            idx2 = np.arange(j_start2, j_end2)
            grad[idx2] -= wk * h[idx2 + lag]
    
    grad *= dx
    return grad
```

This is vectorized over j for each k, so it's O(2n-1) vector operations of average length n/2, total O(n^2) but with numpy overhead being small.

```python
import numpy as np
from scipy.optimize import minimize, Bounds, differential_evolution
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta


def project_h(h, target_sum):
    h = h.copy()
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
            if mask.sum() == 0:
                break
            h[mask] += diff / mask.sum()
        else:
            mask = h > 0.0
            if mask.sum() == 0:
                break
            h[mask] += diff / mask.sum()
    return h


def compute_gradient(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    exp_corr = np.exp(beta * (corr - np.max(corr)))
    weights = exp_corr / np.sum(exp_corr)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        wk = weights[k]
        if wk < 1e-12:
            continue
        js = max(0, lag)
        je = min(n, n + lag)
        if js < je:
            idx = np.arange(js, je)
            grad[idx] += wk * (1.0 - h[idx - lag])
        js2 = max(0, -lag)
        je2 = min(n, n - lag)
        if js2 < je2:
            idx2 = np.arange(js2, je2)
            grad[idx2] -= wk * h[idx2 + lag]
    grad *= dx
    return grad


def fista_optimize(h0, target_sum, beta, max_iter=3000, lr_init=0.5):
    n = len(h0)
    h = project_h(h0, target_sum)
    y = h.copy()
    t = 1.0
    lr = lr_init
    for it in range(max_iter):
        gy = compute_gradient(y, beta)
        step = gy / (np.linalg.norm(gy) + 1e-15)
        alpha = lr / (1.0 + 0.001 * it)
        h_new = y - alpha * gy
        h_new = project_h(h_new, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = h_new + (t / t_new - 1.0) * (h_new - h)
        y = project_h(y, target_sum)
        h = h_new
        t = t_new
        lr = lr_init / (1.0 + 0.002 * it)
    return h


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_points = len(h_init)
    else:
        n_points = 70
        h_init = np.full(n_points, 0.5)

    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_h = project_h(h_init, target_sum)
    best_c5 = compute_c5(best_h)

    def time_left():
        return budget_s - (time.time() - start_time)

    # Phase 1: FISTA with continuation on beta
    betas = [20, 50, 100, 200, 400, 800]
    for beta in betas:
        if time_left() < 30:
            break
        h_curr = best_h.copy()
        # Run FISTA
        h_fista = fista_optimize(h_curr, target_sum, beta, max_iter=2000, lr_init=0.8)
        c5 = compute_c5(h_fista)
        if c5 < best_c5:
            best_h, best_c5 = h_fista.copy(), c5

    # Phase 2: SLSQP polish
    if time_left() > 15:
        bounds = Bounds(np.zeros(n_points), np.ones(n_points))
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                        'jac': lambda x: np.ones(n_points)}]

        def obj_func(h):
            corr = np.correlate(h, 1.0 - h, mode="full")
            return smooth_max(corr, 200.0) * dx

        try:
            res = minimize(obj_func, best_h, method='SLSQP', bounds=bounds,
                           constraints=constraints,
                           options={'maxiter': 3000, 'ftol': 1e-14})
            h_new = project_h(res.x, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new.copy(), c5
        except Exception:
            pass

    # Phase 3: Simulated annealing restarts with FISTA
    attempt = 0
    no_improve = 0
    while time_left() > 40 and no_improve < 50:
        attempt += 1
        # Generate perturbed start
        strategy = attempt % 4
        h0 = best_h.copy()
        if strategy == 0:
            scale = 0.05 * np.exp(-attempt / 40.0)
            h0 += np.random.randn(n_points) * scale
        elif strategy == 1:
            n_pert = np.random.randint(5, 20)
            idx = np.random.choice(n_points, n_pert, replace=False)
            h0[idx] = np.clip(h0[idx] + np.random.randn(n_pert) * 0.15, 0, 1)
        elif strategy == 2:
            # Block swap
            i, j = np.random.choice(n_points, 2, replace=False)
            blen = np.random.randint(3, 10)
            h0[i:i+blen], h0[j:j+blen] = h0[j:j+blen].copy(), h0[i:i+blen].copy()
        else:
            # Anti-correlation push: increase where h is low near peak lags
            corr = np.correlate(h0, 1.0 - h0, mode="full")
            peak_lag = np.argmax(corr) - (n_points - 1)
            for j in range(n_points):
                i1 = j - peak_lag
                if 0 <= i1 < n_points and h0[j] < 0.5 and h0[i1] > 0.5:
                    h0[j] += 0.03
                    h0[i1] -= 0.03
            h0 += np.random.randn(n_points) * 0.01

        h0 = project_h(h0, target_sum)

        # FISTA on perturbed start
        h_fista = fista_optimize(h0, target_sum, 200, max_iter=1500, lr_init=0.5)
        c5 = compute_c5(h_fista)
        if c5 < best_c5:
            best_h, best_c5 = h_fista.copy(), c5
            no_improve = 0
        else:
            no_improve += 1

    # Phase 4: Differential evolution (global search) if time remains
    if time_left() > 80:
        def de_objective(x):
            h = project_h(x, target_sum)
            corr = np.correlate(h, 1.0 - h, mode="full")
            c5_val = np.max(corr) * dx
            penalty = 500.0 * (np.sum(x) - target_sum) ** 2
            return c5_val + penalty

        try:
            res = differential_evolution(
                de_objective, [(0.0, 1.0)] * n_points,
                seed=seed + 1, maxiter=150, popsize=12,
                tol=1e-10, mutation=(0.4, 1.0), recombination=0.7,
                polish=False, init='soff'
            )
            h_new = project_h(res.x, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = c5
        except Exception:
            pass

    # Phase 5: Final polish
    if time_left() > 5:
        bounds = Bounds(np.zeros(n_points), np.ones(n_points))
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                        'jac': lambda x: np.ones(n_points)}]

        def obj_func_final(h):
            corr = np.correlate(h, 1.0 - h, mode="full")
            return smooth_max(corr, 500.0) * dx

        try:
            res = minimize(obj_func_final, best_h, method='SLSQP', bounds=bounds,
                           constraints=constraints,
                           options={'maxiter': 5000, 'ftol': 1e-16})
            h_new = project_h(res.x, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new.copy(), c5
        except Exception:
            pass

    # Verify feasibility
    best_h = project_h(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n_points
```
