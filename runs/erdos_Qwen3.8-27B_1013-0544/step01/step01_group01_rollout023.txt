This is a key insight! The coordinate descent step can be done exactly:
- Compute the slope of f_k with respect to h[i] for each k: s_k = 1 - h[i+k] - h[i-k] (with appropriate handling of boundary terms)
- Compute the intercept: c_k = f_k excluding the h[i] terms
- We want to minimize max_k (c_k + s_k * h[i]) over h[i] ∈ [0,1]
- This is a 1D convex optimization problem (max of linear functions)
- The optimum is where the maximum is minimized, i.e., the "minimax" point

This can be solved by:
- Binary search: find the value v such that we want all c_k + s_k * h[i] ≤ v, and find the smallest v for which there exists h[i] ∈ [0,1] satisfying all constraints.
- Or: the optimum h[i] is where the upper envelope of the lines c_k + s_k * h[i] is minimized.

Actually, the minimum of the upper envelope of lines over [0,1] occurs at:
- h[i] = 0 if the envelope is decreasing at 0
- h[i] = 1 if the envelope is increasing at 1
- Otherwise, at a crossing point of two lines

This is very efficient! Each coordinate descent step is O(n^2) (to compute all f_k values) + O(n log n) (to find the minimax point). Or even O(n^2) total.

from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    if m == 0:
        return 0.0
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def coord_descent(h, n_sweeps=100, time_limit=None, start_time=None):
    """Exact coordinate descent for the Erdos problem."""
    n = len(h)
    h = h.copy()
    
    for sweep in range(n_sweeps):
        if time_limit and start_time:
            if time.time() - start_time > time_limit:
                break
        
        improved = False
        for i in range(n):
            old_val = h[i]
            
            # Compute corr with h[i]=0 and h[i]=1
            h[i] = 0.0
            corr0 = np.correlate(h, 1.0 - h, mode="full")
            h[i] = 1.0
            corr1 = np.correlate(h, 1.0 - h, mode="full")
            
            slopes = corr1 - corr0
            intercepts = corr0
            
            # Minimize max_p (intercepts[p] + slopes[p] * t) for t in [0,1]

            a, b = 0.0, 1.0

import numpy as np
from scipy.optimize import minimize, Bounds
import time

_opt_state = {}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def _obj_func(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    sm = smooth_max(corr, _opt_state['beta']) * _opt_state['dx']
    pen = _opt_state.get('penalty', 0.0)
    if pen > 0:
        sm += pen * (x.sum() - _opt_state['target']) ** 2
    return sm

def _con_fun(x):
    return x.sum() - _opt_state['target']

def _con_jac(x):
    return np.ones(len(x))

def coord_descent_sweep(h):
    n = len(h)
    one = np.ones(n)
    for i in range(n):
        old = h[i]
        h[i] = 0.0
        c0 = np.correlate(h, one - h, mode="full")
        h[i] = 1.0
        c1 = np.correlate(h, one - h, mode="full")
        h[i] = old
        slopes = c1 - c0
        intercepts = c0
        a, b = 0.0, 1.0
        for _ in range(50):
            m1 = a + (b - a) / 3.0
            m2 = a + 2.0 * (b - a) / 3.0
            if np.max(intercepts + slopes * m1) <= np.max(intercepts + slopes * m2):
                b = m2
            else:
                a = m1
        h[i] = (a + b) / 2.0
    return h

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = mask.sum()
        if cnt == 0:
            mask = (h > 0) | (h < 1)
            cnt = mask.sum()
            if cnt == 0:
                break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h
