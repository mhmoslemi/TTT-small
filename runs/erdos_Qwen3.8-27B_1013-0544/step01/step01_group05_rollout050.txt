                              constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                              options={'maxiter': 800, 'ftol': 1e-10})
                h_ref = project_feasible(res.x, target_sum)
                c5_ref = compute_c5(h_ref, dx)
                if c5_ref < best_c5:
                    best_c5 = c5_ref
                    best_h = h_ref.copy()
                    h_curr = h_ref.copy()
            except Exception:
                pass
            if tleft() < 100:
                break

    # Phase 2: Softmax-weighted gradient descent with momentum
    h = best_h.copy()
    step_size = 0.004
    mu = 0.65
    momentum = np.zeros(n_points)
    stag = 0
    best_stag_step = step_size

    while tleft() > 250:
        corr = np.correlate(h, 1.0 - h, mode='full')
        c = corr * dx
        c_max = c.max()
        shifted = c - c_max
        exp_vals = np.exp(shifted)
        p = exp_vals / exp_vals.sum()
        active = np.where(p > 0.03)[0]
        if len(active) < 2:
            active = np.argsort(c)[-5:]

        grad = np.zeros(n_points)
        for idx in active:
            L = int(idx) - n_half
            grad += p[idx] * grad_corr_lag(h, L)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-13:
            break

        momentum = mu * momentum - step_size * grad / gnorm
        h_new = project_feasible(h + momentum, target_sum)
        c5_new = compute_c5(h_new, dx)

        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            stag = 0
            best_stag_step = step_size
        else:
            stag += 1
            if stag % 15 == 0:
                step_size *= 0.87
                momentum *= 0.4
            if step_size < 1e-10:
                h = best_h + rng.normal(0, 0.04, n_points)
                h = project_feasible(h, target_sum)
                step_size = best_stag_step
                momentum = np.zeros(n_points)
                stag = 0
        h = h_new

    # Phase 3: Simulated annealing with mixed proposals
    if tleft() > 120:
        h = best_h.copy()
        temp = 0.0004
        no_improve = 0
        max_no_improve = 400
        for i in range(20000):
            if tleft() < 5:
                break
            r = rng.random()
            if r < 0.4:
                h_pert = h + rng.normal(0, 0.012, n_points)
            elif r < 0.8:
                corr = np.correlate(h, 1.0 - h, mode='full')
                c = corr * dx
                c_max = c.max()
                active = np.where(c >= c_max - 0.002)[0]
                if len(active) < 1:
                    active = np.array([int(np.argmax(c))])
                if len(active) > 8:
                    active = np.argsort(c)[-8:]
                grad = np.zeros(n_points)
                for idx in active:
                    L = int(idx) - n_half
                    grad += grad_corr_lag(h, L)
                grad /= len(active)
                h_pert = h - 0.008 * grad + rng.normal(0, 0.004, n_points)
            else:
                # Pair swap
                i1, i2 = rng.choice(n_points, 2, replace=False)
                h_pert = h.copy()
                t = rng.uniform(-0.05, 0.05)
                h_pert[i1] += t
                h_pert[i2] -= t
            h_pert = project_feasible(h_pert, target_sum)
            c5_pert = compute_c5(h_pert, dx)
            c5_curr = compute_c5(h, dx)
            delta = c5_pert - c5_curr
            if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-12)):
                h = h_pert
                if c5_pert < best_c5:
                    best_c5 = c5_pert
                    best_h = h_pert.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
            temp *= 0.9998
            if no_improve > max_no_improve:
                no_improve = 0
                temp = max(temp, 0.0001)
                h = best_h + rng.normal(0, 0.02, n_points)
                h = project_feasible(h, target_sum)

    # Phase 4: Fine gradient polish
    if tleft() > 15:
        h = best_h.copy()
        step_size = 0.00015
        momentum = np.zeros(n_points)
        mu2 = 0.35
        for _ in range(20000):
            if tleft() < 1:
                break
            corr = np.correlate(h, 1.0 - h, mode='full')
            c = corr * dx
            c_max = c.max()
            active = np.where(c >= c_max - 1e-5)[0]
            if len(active) < 1:
                active = np.array([int(np.argmax(c))])
            if len(active) > 5:
                active = np.argsort(c)[-5:]
            grad = np.zeros(n_points)
            for idx in active:
                L = int(idx) - n_half
                grad += grad_corr_lag(h, L)
            grad /= len(active)
            momentum = mu2 * momentum - step_size * grad
            h_new = project_feasible(h + momentum, target_sum)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new.copy()
            else:
                step_size *= 0.94
                if step_size < 1e-15:
                    break
            h = h_new

    # Phase 5: Final random restarts
    if tleft() > 20:
        for _ in range(300):
            if tleft() < 2:
                break
            h_pert = best_h + rng.normal(0, 0.003, n_points)
            h_pert = project_feasible(h_pert, target_sum)
            c5_pert = compute_c5(h_pert, dx)
            if c5_pert < best_c5:
                best_c5 = c5_pert
                best_h = h_pert.copy()
            # Quick local refine
            h = h_pert.copy()
            ss = 0.0001
            mom = np.zeros(n_points)
            for _ in range(200):
                if tleft() < 2:
                    break
                corr = np.correlate(h, 1.0 - h, mode='full')
                c = corr * dx
                c_max = c.max()
                active = np.where(c >= c_max - 1e-5)[0]
                if len(active) < 1:
                    active = np.array([int(np.argmax(c))])
                if len(active) > 4:
                    active = np.argsort(c)[-4:]
                g = np.zeros(n_points)
                for idx in active:
                    L = int(idx) - n_half
                    g += grad_corr_lag(h, L)
                g /= len(active)
                mom = 0.4 * mom - ss * g
                h = project_feasible(h + mom, target_sum)
                c5_t = compute_c5(h, dx)
                if c5_t < best_c5:
                    best_c5 = c5_t
                    best_h = h.copy()
                else:
                    ss *= 0.9

    return (best_h, best_c5, n_points)
