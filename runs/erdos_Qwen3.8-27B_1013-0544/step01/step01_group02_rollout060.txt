The challenge is that computing C5 after each swap is O(n^2) due to the correlation. But we can compute the change in C5 more efficiently.

When we swap h[i] and h[j] (say h[i] goes from 1 to 0 and h[j] from 0 to 1), the change in the overlap at shift k is:
- For shift k: the overlap changes by the contributions from positions i and j
- Specifically, at shift k:
  - Position i: was contributing h[i]*(1-h[i+k]) = 1*(1-h[i+k]) = 1-h[i+k], now contributes 0*(1-h[i+k]) = 0. Change: -(1-h[i+k]) = h[i+k]-1
  - Position j: was contributing h[j]*(1-h[j+k]) = 0, now contributes 1*(1-h[j+k]) = 1-h[j+k]. Change: 1-h[j+k]
  - Also, position i-k (if valid) was contributing h[i-k]*(1-h[i]) = 1 (since h[i] was 1, 1-h[i]=0... wait, h[i]-k * (1-h[i])). If h[i]=1, then (1-h[i])=0, so contribution was 0. Now h[i]=0, so contribution is h[i-k]*(1-0) = h[i-k]. Change: +h[i-k]
  - Similarly for position j-k: was h[j-k]*(1-h[j]) = h[j-k]*1 = h[j-k]. Now h[j]=1, so contribution is 0. Change: -h[j-k]

So the total change at shift k from swapping i (1→0) and j (0→1) is:
Δ_k = (h[i+k] - 1) + (1 - h[j+k]) + h[i-k] - h[j-k]
     = h[i+k] - h[j+k] + h[i-k] - h[j-k]

(with appropriate boundary conditions - terms are 0 if the index is out of range)

This gives us an O(1) update per shift per swap! So we can compute the change in the max overlap for a given swap in O(n) time (by checking all shifts), or we can maintain the full overlap vector and update it in O(n) time.

def binary_local_search_smart(h, n, max_iters=20000, time_limit=60):
    """Local search with smart swap selection."""
    t0 = time.time()
    h = (h > 0.5).astype(float)
    ones_count = h.sum()
    if ones_count != n / 2:
        idx_ones = np.where(h == 1)[0]
        idx_zeros = np.where(h == 0)[0]
        if ones_count > n / 2:
            extra = int(ones_count - n / 2)
            for idx in idx_ones[:extra]:
                h[idx] = 0
            for idx in idx_zeros[:extra]:
                h[idx] = 1
        else:
            extra = int(n / 2 - ones_count)
            for idx in idx_zeros[:extra]:
                h[idx] = 1
            for idx in idx_ones[:extra]:
                h[idx] = 0
    
    c = compute_overlaps(h, n)
    current_max = c.max()
    no_improve_count = 0
    
    for it in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        
        k_star = np.argmax(c)
        tau = k_star - (n - 1)
        
        # Score each 1-position: how many 0s does it have at shift tau?
        # Score each 0-position: how many 1s does it have at shift -tau?
        # We want to swap a 1 with high "outgoing 0 count" with a 0 with high "incoming 1 count"
        
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        
        # For each 1 at position i, count 0s at i+tau (for positive tau)
        # or at i-tau (for negative tau)
        if tau > 0:
            # h[i]=1 contributes to overlap at tau if h[i+tau]=0
            # Score for 1 at i: 1 if i+tau < n and h[i+tau]=0, else 0
            score_ones = np.zeros(n)
            valid = np.arange(n - tau)
            score_ones[valid] = 1 - h[tau:]
            # For 0 at j: we want h[j] to become 1, which helps if h[j-tau]=1 (it was "wasting" a 1)
            # Actually, if h[j] becomes 1, it removes the contribution from position j-tau (if h[j-tau]=1, 
            # then h[j-tau] was contributing 1*(1-h[j])=1 to the overlap at tau; now it contributes 0)
            score_zeros = np.zeros(n)
            valid = np.arange(tau, n)
            score_zeros[valid] = h[:n - tau]
        elif tau < 0:
            s = -tau
            score_ones = np.zeros(n)
            valid = np.arange(s, n)
            score_ones[valid] = 1 - h[:n - s]
            score_zeros = np.zeros(n)
            valid = np.arange(n - s)
            score_zeros[valid] = h[s:]
        else:
            score_ones = 1 - h  # all 1s contribute equally
            score_zeros = h
        
        # Try top-scoring swaps
        tried = set()
        improved = False
        
        # Try a few candidate swaps
        n_candidates = min(30, len(ones_idx) * len(zeros_idx))
        for _ in range(n_candidates):
            # Pick a random 1 with high score
            o_scores = score_ones[ones_idx]
            if o_scores.max() <= 0:
                break
            # Weighted random selection
            probs = o_scores / o_scores.sum()
            oi = np.random.choice(len(ones_idx), p=probs)
            i = ones_idx[oi]
            
            z_scores = score_zeros[zeros_idx]
            if z_scores.max() <= 0:
                break
            probs = z_scores / z_scores.sum()
            zi = np.random.choice(len(zeros_idx), p=probs)
            j = zeros_idx[zi]
            
            if i == j or (i, j) in tried:
                continue
            tried.add((i, j))
            
            # Full swap evaluation
            h[i], h[j] = 0, 1
            c_new = compute_overlaps(h, n)
            new_max = c_new.max()
            if new_max < current_max - 1e-12:
                current_max = new_max
                c = c_new
                improved = True
                no_improve_count = 0
                break
            else:
                h[i], h[j] = 1, 0
        
        if not improved:
            no_improve_count += 1
            if no_improve_count > 100:
                break
    
    return h, compute_c5(h, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            t = room.sum()
            if t < 1e-14:
                break
            h = h + (diff / t) * room
        else:
            t = h.sum()
            if t < 1e-14:
                break
            h = h + (diff / t) * h
    return np.clip(h, 0.0, 1.0)
