def find_optimal_delta(h, a, b, n):
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full')
    # Compute g(p,s) for p=a and p=b
    # s ranges from -(n-1) to n-1, indexed as k = s + (n-1) in [0, 2n-2]
    g_a = np.zeros(2*n-1)
    g_b = np.zeros(2*n-1)
    for s in range(-(n-1), n):
        k = s + (n-1)
        # g(p, s) = (1-h[p+s]) if p+s in [0,n) else 0, minus h[p-s] if p-s in [0,n) else 0
        if 0 <= a+s < n:
            g_a[k] += (1-h[a+s])
        if 0 <= a-s < n:
            g_a[k] -= h[a-s]
        if 0 <= b+s < n:
            g_b[k] += (1-h[b+s])
        if 0 <= b-s < n:
            g_b[k] -= h[b-s]
    d = g_a - g_b
    # Now minimize max_k (c[k] + delta * d[k]) over delta in [-h[a], 1-h[a]] ∩ [h[b]-1, h[b]]
    delta_lo = max(-h[a], h[b]-1)
    delta_hi = min(1-h[a], h[b])
    if delta_lo >= delta_hi:
        return 0.0
    # Convex piecewise linear minimization
    # Use bisection on subgradient
    def subgradient(delta):
        vals = c + delta * d
        max_val = vals.max()
        active = np.where(np.isclose(vals, max_val, atol=1e-10))[0]
        return d[active].sum()
    
    lo, hi = delta_lo, delta_hi
    for _ in range(50):
        mid = (lo + hi) / 2
        sg = subgradient(mid)
        if sg > 0:
            hi = mid
        else:
            lo = mid
    return (lo + hi) / 2

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_val(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def find_best_delta(h, a, b, c_arr, n):
    d = np.zeros(2 * n - 1)
    s_vals = np.arange(-(n - 1), n)
    for idx, s in enumerate(s_vals):
        ga = 0.0
        gb = 0.0
        t = a + s
        if 0 <= t < n:
            ga += 1.0 - h[t]
        t = a - s
        if 0 <= t < n:
            ga -= h[t]
        t = b + s
        if 0 <= t < n:
            gb += 1.0 - h[t]
        t = b - s
        if 0 <= t < n:
            gb -= h[t]
        d[idx] = ga - gb
    dl = max(-h[a], h[b] - 1.0)
    dh = min(1.0 - h[a], h[b])
    if dl >= dh - 1e-14:
        return 0.0
    lo, hi = dl, dh

        vals = c_arr + mid * d
        max_v = vals.max()
        active = vals >= max_v - 1e-13
        da = d[active]
        if da.min() <= 0.0 <= da.max():
            return mid
        elif da.min() > 0.0:
            hi = mid
        else:
            lo = mid
    return (lo + hi) * 0.5

def coord_descent(h, n, n_steps, rng, t0, time_limit):
    c_arr = np.correlate(h, 1 - h, mode='full')
    best_max = float(c_arr.max())
    for step in range(n_steps):

            break
        a = rng.randint(n)
        b = rng.randint(n)
        if a == b:
            continue
        delta = find_best_delta(h, a, b, c_arr, n)
        if abs(delta) > 1e-11:
            h[a] += delta
            h[b] -= delta
            c_arr = np.correlate(h, 1 - h, mode='full')
            nm = float(c_arr.max())
            if nm > best_max + 1e-11:
                h[a] -= delta
                h[b] += delta
                c_arr = np.correlate(h, 1 - h, mode='full')
            else:
                best_max = nm
    return h, best_max

def coord_sweep(h, n, n_sweeps, rng, t0, time_limit):
    c_arr = np.correlate(h, 1 - h, mode='full')
    best_max = float(c_arr.max())
    pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    for sw in range(n_sweeps):
        if time.time() - t0 > time_limit:
            break
        rng.shuffle(pairs)
        any_improve = False
        for a, b in pairs:

def obj_smooth(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def sum_constraint(x, n):
    return x.sum() - n / 2.0

def sum_constraint_jac(x, n):
    return np.ones(len(x))

def slsqp_refine(h, n, t0, time_limit):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac, 'args': (n,)}]
    best_h = h.copy()
    best_c5 = compute_c5_val(best_h, n)

            h_try = project_feasible(res.x, n)
            c5 = compute_c5_val(h_try, n)
            if c5 < best_c5:
                best_h, best_c5 = h_try, c5
        except Exception:
            pass
    return best_h, best_c5
