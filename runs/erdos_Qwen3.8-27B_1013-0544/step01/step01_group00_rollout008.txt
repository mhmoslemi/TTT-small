def solve_lp(h_init, g):

    h_var = cp.Variable(n)
    t_var = cp.Variable()

    # For each lag k from -(n-1) to (n-1):
    #   sum_i g[i+k] * h[i] <= t (for valid i)

    prob = cp.Problem(cp.Minimize(t_var), constraints)
