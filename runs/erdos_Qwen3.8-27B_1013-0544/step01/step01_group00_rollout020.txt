import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def proj_uniform(h, target_sum):
    h = h.copy()
    s = np.sum(h)
    diff = target_sum - s
    if abs(diff) < 1e-15:
        return np.clip(h, 0, 1)
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = mask.sum()
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0, 1)

def lse_obj_fn(h, temp, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    val = m + temp * np.log(np.sum(np.exp((corr - m) / temp)))
    s = np.sum(h) - n * 0.5
    val += penalty * s * s
    return val

def interp_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.arange(n_old) / float(n_old)
    x_new = np.arange(n_new) / float(n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return proj_uniform(h_new, n_new * 0.5)

def opt_lbfgsb(h0, temp, penalty, max_iter):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    res = minimize(lse_obj_fn, h0.copy(), args=(temp, penalty),
                   method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'gtol': 1e-8})
    h_new = np.clip(res.x, 0.0, 1.0)
    h_new = proj_uniform(h_new, n * 0.5)
    return h_new, compute_c5(h_new)

def t_obj_fn(v):
    return v[-1]

def t_ineq_fn(v):
    n = len(v) - 1
    return v[-1] - np.correlate(v[:n], 1.0 - v[:n], mode='full') * (2.0 / n)

    n = len(v) - 1
    return np.sum(v[:n]) - n * 0.5

def t_eq_jac_fn(v):
    n = len(v) - 1
    j = np.zeros(len(v))
    j[:n] = 1.0
    return j

    c5_cur = compute_c5(h0)
    v0 = np.append(h0.copy(), c5_cur + 0.01)
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.5)]
    cons = [
        {'type': 'ineq', 'fun': t_ineq_fn},
        {'type': 'eq', 'fun': t_eq_fn, 'jac': t_eq_jac_fn}

                   constraints=cons, options={'maxiter': max_iter, 'ftol': ftol})
    h_new = np.clip(res.x[:n], 0.0, 1.0)
    h_new = proj_uniform(h_new, n * 0.5)
    return h_new, compute_c5(h_new)

        h_new, c5_new = opt_tvar(h, max_iter=800, ftol=1e-10)
        if c5_new < c5_best:
            h_best = h_new.copy()
            c5_best = c5_new
    h = h_best
    c5 = c5_best

    # Phase 3: Second L-BFGS-B pass from best
    tl = budget_s - (time.time() - t0)
    if tl > 100:
        h_new, c5_new = opt_lbfgsb(h_best, 0.0003, 5000.0, 600)
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new.copy()
        h_new, c5_new = opt_lbfgsb(h_best, 0.0001, 20000.0, 800)
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new.copy()

    # Phase 4: SLSQP final polish
    tl = budget_s - (time.time() - t0)
    if tl > 120:
        h_new, c5_new = opt_tvar(h_best, max_iter=1000, ftol=1e-11)
        if c5_new < c5_best:
            h_best = h_new.copy()
            c5_best = c5_new

    # Phase 5: Multi-resolution exploration
    for n_new in [n // 2, n * 2, n // 3]:
        tl = budget_s - (time.time() - t0)
        if tl < 200:
            break
        if n_new < 20:
            continue
        h_interp = interp_h(h_best, n_new)
        c5_interp = compute_c5(h_interp)
        # Optimize at new resolution with L-BFGS-B
        for t_i, p_i, m_i in [(0.02, 100.0, 300), (0.005, 500.0, 400), (0.001, 2000.0, 400)]:
            tl = budget_s - (time.time() - t0)
            if tl < 60:
                break
            h_try, c5_try = opt_lbfgsb(h_interp, t_i, p_i, m_i)
            if c5_try < c5_interp:
                h_interp, c5_interp = h_try, c5_try
        # SLSQP at new resolution
        tl = budget_s - (time.time() - t0)
        if tl > 80:
            h_try, c5_try = opt_tvar(h_interp, max_iter=500, ftol=1e-10)
            if c5_try < c5_interp:
                h_interp, c5_interp = h_try, c5_try
        # Interpolate back to original size
        h_back = interp_h(h_interp, n)
        c5_back = compute_c5(h_back)
        # Quick polish of back-interpolated
        tl = budget_s - (time.time() - t0)
        if tl > 60 and c5_back < c5_best + 0.003:
            h_pol, c5_pol = opt_lbfgsb(h_back, 0.001, 2000.0, 300)
            if c5_pol < c5_back:
                h_back, c5_back = h_pol, c5_pol
        if c5_back < c5_best:
            h_best = h_back.copy()
            c5_best = c5_back

    # Phase 6: Diverse random restarts
    for restart in range(25):
        tl = budget_s - (time.time() - t0)
        if tl < 50:
            break
        scale = 0.015 + 0.008 * (restart % 6)
        h_p = h_best + rng.randn(n) * scale
        h_p = proj_uniform(h_p, n * 0.5)
        h_new, c5_new = opt_lbfgsb(h_p, 0.005, 500.0, 300)
        if c5_new < c5_best:
            h_best = h_new.copy()
            c5_best = c5_new
        elif c5_new < c5_best + 0.0005:
            tl = budget_s - (time.time() - t0)
            if tl > 40:
                h_new2, c5_new2 = opt_tvar(h_new, max_iter=400, ftol=1e-10)
                if c5_new2 < c5_best:
                    h_best = h_new2.copy()
                    c5_best = c5_new2

    # Phase 7: Structured perturbations (shift, flip)
    for perm_type in ['shift', 'flip', 'shift2']:
        tl = budget_s - (time.time() - t0)
        if tl < 40:
            break
        if perm_type == 'shift':
            h_p = np.roll(h_best, 1)
        elif perm_type == 'flip':
            h_p = h_best[::-1].copy()
        else:
            h_p = np.concatenate([h_best[n//3:], h_best[:n//3]])
        h_p = proj_uniform(h_p, n * 0.5)
        h_new, c5_new = opt_lbfgsb(h_p, 0.002, 1000.0, 300)
        if c5_new < c5_best:
            h_best = h_new.copy()
            c5_best = c5_new

    # Phase 8: Final SLSQP polish
    tl = budget_s - (time.time() - t0)
    if tl > 60:
        h_new, c5_new = opt_tvar(h_best, max_iter=1200, ftol=1e-11)
        if c5_new < c5_best:
            h_best = h_new.copy()
            c5_best = c5_new

    # Final L-BFGS-B micro-polish
    tl = budget_s - (time.time() - t0)
    if tl > 30:
        h_new, c5_new = opt_lbfgsb(h_best, 0.00005, 50000.0, 500)
        if c5_new < c5_best:
            h_best = h_new.copy()
            c5_best = c5_new

    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = proj_uniform(h_best, len(h_best) * 0.5)
    c5_final = compute_c5(h_best)
    return h_best, c5_final, len(h_best)
