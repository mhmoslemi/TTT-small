import time

def evaluate_c5(h):

    dx = 2.0 / n

    return float(np.max(corr) * dx)

    x_max = np.max(x)
    return float(x_max / tau + np.log(np.sum(np.exp(x - x_max))) / tau)

    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-14:

    if diff > 0:

    else:
        mask = h > 0.0
    cnt = mask.sum()
    if cnt > 0:

    return np.clip(h, 0.0, 1.0)

def optimize_slsqp(h_start, n_pts, tau, maxiter=500, ftol=1e-14):
    target = n_pts / 2.0
    bnds = Bounds(np.zeros(n_pts), np.ones(n_pts))

        corr = np.correlate(x, 1.0 - x, mode="full")

             'jac': lambda x: np.ones(n_pts)}]
    res = minimize(obj, h_start, method='SLSQP', bounds=bnds,
                  constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

    """Local search: try small adjustments to individual elements"""
    target = n_pts / 2.0
    best_c5 = evaluate_c5(h)
    h = h.copy()
    
    for round in range(max_rounds):
        improved = False
        for i in range(n_pts):
            for delta in [0.05, -0.05, 0.1, -0.1, 0.2, -0.2]:

                h_new[i] += delta

            return np.ones(n_pts)

        try:

                          constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})

            return h_start

            c5_new = evaluate_c5(h_new)
            if c5_new <= evaluate_c5(h) + 1e-12:
                h = h_new
            else:
                break
        return h

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n

def smooth_max(obj, tau):
    x = tau * obj
    x_max = np.max(x)
    return float(x_max / tau + np.log(np.sum(np.exp(x - x_max))) / tau)

def fix_sum(h, target):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h.copy()
    if diff > 0:
        mask = h < 1.0 - 1e-12
    else:
        mask = h > 1e-12
    cnt = mask.sum()
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)
