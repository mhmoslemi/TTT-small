```python
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

        s = h.sum()

        h[mask] += amt
        h = np.clip(h, 0.0, 1.0)

        grad[i] = (val - base) / eps
```

```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta


def corr_of(h):
    return np.correlate(h, 1.0 - h, mode="full")


def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).astype(float)
    for _ in range(2000):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def obj_smooth(h, dx, beta):
    return smooth_max(corr_of(h), beta) * dx


def con_fun(x, target_sum):
    return np.sum(x) - target_sum


def con_jac(x, target_sum):
    return np.ones(len(x))


def num_grad(h, dx, beta):
    n = len(h)
    eps = 1e-6
    base = smooth_max(corr_of(h), beta) * dx
    grad = np.zeros(n)
    for i in range(n):
        h2 = h.copy()
        h2[i] += eps
        grad[i] = (smooth_max(corr_of(h2), beta) * dx - base) / eps
    return grad


def do_subgradient(h, dx, target_sum, n_iters, lr_start):
    n = len(h)
    h = h.copy()
    momentum = np.zeros(n)
    lr = lr_start
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(n_iters):
        beta = 50 + 250 * it / max(n_iters, 1)
        grad = num_grad(h, dx, beta)
        momentum = 0.9 * momentum - lr * grad
        h_new = h + momentum
        h_new = project_h(h_new, target_sum)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = c5
        h = h_new
        lr *= 0.99
    return best_h, best_c5


def do_coord_descent(h, dx, target_sum, n_sweeps, grid_size):
    n = len(h)
    h = h.copy()
    grid = np.linspace(0.0, 1.0, grid_size)
    best_c5 = compute_c5(h)
    for sweep in range(n_sweeps):
        improved = False
        for i in range(n):
            old = h[i]
            best_t = old
            best_obj = compute_c5(h)
            for t in grid:
                if abs(t - old) < 1e-10:
                    continue
                h[i] = t
                obj = compute_c5(h)
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    best_t = t
            h[i] = best_t
            if best_t != old:
                improved = True
        h = project_h(h, target_sum)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
        if not improved:
            break
    return h, best_c5


def do_pair_coord_descent(h, dx, target_sum, n_sweeps, grid_size):
    n = len(h)
    h = h.copy()
    grid = np.linspace(0.0, 1.0, grid_size)
    best_c5 = compute_c5(h)
    for sweep in range(n_sweeps):
        improved = False
        step = (sweep % 2) + 1
        for i in range(0, n, step):
            for j in range(i + 1, n, step):
                s = h[i] + h[j]
                best_ti, best_tj = h[i], h[j]
                best_obj = compute_c5(h)
                for t in grid:
                    t2 = s - t
                    if t2 < -1e-12 or t2 > 1.0 + 1e-12:
                        continue
                    t2 = np.clip(t2, 0.0, 1.0)
                    h[i] = t
                    h[j] = t2
                    obj = compute_c5(h)
                    if obj < best_obj - 1e-12:
                        best_obj = obj
                        best_ti, best_tj = t, t2
                h[i], h[j] = best_ti, best_tj
                if best_ti != h[i] or best_tj != h[j]:
                    improved = True
        h = project_h(h, target_sum)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
        if not improved:
            break
    return h, best_c5


def do_slsqp(h, dx, target_sum, beta, maxiter, ftol):
    n = len(h)
    bounds = Bounds(np.zeros(n), np.ones(n))
    constraints = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum,)}]
    try:
        res = minimize(obj_smooth, h, args=(dx, beta), method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       maxiter=maxiter, ftol=ftol)
        h_new = project_h(res.x, target_sum)
        return h_new
    except Exception:
        return h


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
        n_points = len(best_h)
    else:
        n_points = 70
        best_h = np.full(n_points, 0.5)

    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_h = project_h(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    n = n_points

    def time_left():
        return budget_s - (time.time() - start_time)

    # Phase 1: Subgradient with momentum
    if time_left() > 500:
        h1, c5_1 = do_subgradient(best_h, dx, target_sum, 200, 0.015)
        if c5_1 < best_c5:
            best_h, best_c5 = h1, c5_1

    # Phase 2: Single coordinate descent
    if time_left() > 400:
        h2, c5_2 = do_coord_descent(best_h, dx, target_sum, 30, 50)
        if c5_2 < best_c5:
            best_h, best_c5 = h2, c5_2

    # Phase 3: SLSQP with increasing beta
    for beta in [80, 150, 300, 600]:
        if time_left() < 50:
            break
        h3 = do_slsqp(best_h, dx, target_sum, beta, 400, 1e-13)
        c5_3 = compute_c5(h3)
        if c5_3 < best_c5:
            best_h, best_c5 = h3, c5_3

    # Phase 4: Pair coordinate descent
    if time_left() > 300:
        h4, c5_4 = do_pair_coord_descent(best_h, dx, target_sum, 8, 35)
        if c5_4 < best_c5:
            best_h, best_c5 = h4, c5_4

    # Phase 5: Diverse restarts with various strategies
    attempt = 0
    no_improve = 0
    while time_left() > 25:
        attempt += 1
        no_improve += 1
        h0 = best_h.copy()
        strat = attempt % 6

        if strat == 0:
            scale = 0.04 / (1.0 + attempt * 0.08)
            h0 += np.random.randn(n) * scale
        elif strat == 1:
            # Lag-focused perturbation
            corr = corr_of(h0)
            top_lags = np.argsort(corr)[-5:]
            affected = set()
            for k in top_lags:
                offset = k - (n - 1)
                for j in range(n):
                    m = j + offset
                    if 0 <= m < n:
                        affected.add(j)
                    m2 = j - offset
                    if 0 <= m2 < n:
                        affected.add(m2)
            idx = list(affected)
            h0[idx] += np.random.randn(len(idx)) * 0.06
        elif strat == 2:
            # Swap-based
            for _ in range(np.random.randint(2, 8)):
                a, b = np.random.choice(n, 2, replace=False)
                h0[a], h0[b] = h0[b], h0[a]
        elif strat == 3:
            # Block randomization
            blen = np.random.randint(3, max(4, n // 6))
            bs = np.random.randint(0, max(1, n - blen))
            h0[bs:bs + blen] = np.random.random(blen) * 0.5 + 0.25
        elif strat == 4:
            # Antisymmetric perturbation (preserves sum better)
            h0 += np.random.randn(n) * 0.03
            h0[np.arange(0, n, 2)] -= 0.01
            h0[np.arange(1, n, 2)] += 0.01
        else:
            # Large jump
            mp = max(2, n // 10)
            idx = np.random.choice(n, mp, replace=False)
            h0[idx] = np.random.random(mp)

        h0 = project_h(h0, target_sum)
        beta_r = 80 + 60 * (attempt % 5)
        h_r = do_slsqp(h0, dx, target_sum, beta_r, 250, 1e-12)
        c5_r = compute_c5(h_r)
        if c5_r < best_c5:
            best_h, best_c5 = h_r.copy(), c5_r
            no_improve = 0

        # Also try coord descent from this point occasionally
        if attempt % 4 == 0 and time_left() > 40:
            hc, c5c = do_coord_descent(h_r, dx, target_sum, 5, 30)
            if c5c < best_c5:
                best_h, best_c5 = hc.copy(), c5c
                no_improve = 0

    # Phase 6: Final high-precision polish
    if time_left() > 15:
        h_f = do_slsqp(best_h, dx, target_sum, 800, 1000, 1e-15)
        c5_f = compute_c5(h_f)
        if c5_f < best_c5:
            best_h, best_c5 = h_f, c5_f

    if time_left() > 5:
        hc, c5c = do_coord_descent(best_h, dx, target_sum, 10, 60)
        if c5c < best_c5:
            best_h, best_c5 = hc, c5c

    return best_h, best_c5, n_points
```
