    h_new[j] += best_delta
    h_new[l] -= best_delta
    h_new = np.clip(h_new, 0.0, 1.0)

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target_sum):
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-15:
        h = h + diff / len(h)
    return np.clip(h, 0.0, 1.0)

def pair_step_inplace(h, corr, j, l, n):
    """Try to improve h by adjusting pair (j, l). Returns (improved, new_corr_max, delta_used)"""
    hj, hl = h[j], h[l]
    delta_min = max(-hj, hl - 1.0)
    delta_max = min(1.0 - hj, hl)
    
    if delta_max - delta_min < 1e-12:
        return False, np.max(corr), 0.0
    
    # Compute sensitivity
    k_arr = np.arange(2 * n - 1)
    s = np.zeros(2 * n - 1)
    
    q = j + (n - 1) - k_arr
    m = (q >= 0) & (q < n)
    if np.any(m):
        s[m] += 1.0 - h[q[m]]
    
    q = l + (n - 1) - k_arr
    m = (q >= 0) & (q < n)
    if np.any(m):
        s[m] -= 1.0 - h[q[m]]
    
    q = j - (n - 1) + k_arr
    m = (q >= 0) & (q < n)
    if np.any(m):
        s[m] -= h[q[m]]
    
    q = l - (n - 1) + k_arr
    m = (q >= 0) & (q < n)
    if np.any(m):
        s[m] += h[q[m]]
    
    # Find optimal delta
    current_max = np.max(corr)
    candidates = [delta_min, delta_max, 0.0]
    
    # Add crossing of top lags
    top_8 = np.argsort(corr)[-8:][::-1]
    for a in range(8):
        for b in range(a + 1, 8):
            ka, kb = int(top_8[a]), int(top_8[b])
            ds = s[ka] - s[kb]
            if abs(ds) > 1e-12:
                d = (corr[kb] - corr[ka]) / ds
                if delta_min <= d <= delta_max:
                    candidates.append(d)
    
    best_delta = 0.0
    best_val = current_max
    for d in candidates:
        d = float(max(delta_min, min(delta_max, d)))
        val = np.max(corr + d * s)
        if val < best_val:
            best_val = val
            best_delta = d
    
    if best_val >= current_max - 1e-12 or abs(best_delta) < 1e-12:
        return False, current_max, 0.0
    
    # Apply
    h[j] += best_delta
    h[l] -= best_delta
    h[j] = np.clip(h[j], 0.0, 1.0)
    h[l] = np.clip(h[l], 0.0, 1.0)
    return True, best_val, best_delta

def coordinate_descent_round(h, target_sum, n_pairs=None, seed_offset=0):
    n = len(h)
    if n_pairs is None:
        n_pairs = n * 8  # Try 8n random pairs per round

    l_arr = rng.randint(0, n, size=n_pairs)
    mask = j_arr != l_arr
    j_arr = j_arr[mask]
    l_arr = l_arr[mask]
    
    corr = np.correlate(h, 1 - h, mode="full")
    current_max = np.max(corr)
    n_improved = 0
    
    for idx in range(len(j_arr)):
        j, l = int(j_arr[idx]), int(l_arr[idx])
        improved, new_max, _ = pair_step_inplace(h, corr, j, l, n)
        if improved:
            corr = np.correlate(h, 1 - h, mode="full")
            current_max = np.max(corr)
            n_improved += 1
    
    return h, n_improved, current_max

def upsample(h, new_n):
    old_n = len(h)

            j = int(j_arr[idx])
            l = int(l_arr[idx])
            
            hj = h[j]
            hl = h[l]

            dmax = min(1.0 - hj, hl)
            
            if dmax - dmin < 1e-11:
                continue
            
            # Sensitivity

            if m.any():
