```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def smooth_max_weights(corr, alpha):
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    s = np.sum(w)
    if s < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= s
    return w

def grad_corr_single(h, k, n):
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n-k])
        if k > 0:
            g[:n-k] -= h[k:]
    else:
        p = -k
        g[:n-p] += (1.0 - h[p:])
        if p > 0:
            g[p:] -= h[:n-p]
    return g

def compute_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    w = smooth_max_weights(corr, alpha)
    grad = np.zeros(n)
    active = np.where(w > 5e-8)[0]
    for idx in active:
        k = int(idx) - (n - 1)
        grad += w[idx] * grad_corr_single(h, k, n)
    return grad * (2.0 / n)

def project_box_sum(h, target_sum):
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(h - lam, 0.0, 1.0)
```

Actually, I just realized something important. Let me reconsider the problem structure.

The correlation `np.correlate(h, 1-h, mode='full')` has length 2n-1. The indices go from 0 to 2n-2, corresponding to lags from -(n-1) to +(n-1).

For the C5 value, we take the max of all these correlations times dx = 2/n.

Now, here's a key observation: for lag 0, corr[0] = sum(h[i] * (1-h[i])) = sum(h[i]) - sum(h[i]^2) = n/2 - sum(h[i]^2).

Since h[i] ∈ [0,1], h[i]^2 ≤ h[i], so sum(h[i]^2) ≤ sum(h[i]) = n/2.
Thus corr[0] ≥ 0, with equality iff h is binary.

For a binary h, corr[0] = 0. This is the minimum possible value at lag 0.

So binary solutions have the advantage that lag 0 gives 0 contribution. This strongly suggests that the optimal (or near-optimal) solution is binary!

If the solution is binary, then the problem reduces to: find a binary sequence of length n with exactly n/2 ones that maximizes min_k autocorr(k) (equivalently, minimizes max_k [(n/2) - autocorr(k)]).

This is a combinatorial optimization problem. For the discrete case, we could use:
1. Local search (swap-based)
2. Simulated annealing
3. Genetic algorithms

But since we're working with the continuous relaxation and the gradient method, the solution should naturally converge to (or near) binary values.

Actually, you know what, let me try a completely different approach for the optimization: **simulated annealing with gradient-guided moves**.

The idea:
1. Start from the current best
2. At each step, generate a "move" (perturbation)
3. Accept the move with probability exp(-delta/T) where delta is the increase in C5
4. Decrease T over time (annealing schedule)

The "moves" can be:
- Gradient step (deterministic descent direction)
- Random flip of a few positions
- Gaussian perturbation

This combines the best of both worlds: gradient guidance for efficient descent, and stochastic acceptance for escaping local minima.

```python
def optimize_sa(h0, target_sum, deadline, rng, alpha=100):
    n = len(h0)
    h = project_box_sum(h0.copy(), target_sum)
    best = compute_c5(h)
    T = 0.01  # initial temperature
    T_min = 1e-5
    cooling = 0.995
    
    while time.time() < deadline:
        # Generate a candidate move
        mode = rng.randint(0, 3)
        if mode == 0 and rng.rand() < 0.7:
            # Gradient-guided step
            g = compute_grad(h, alpha)
            step = rng.uniform(0.01, 0.05)
            h_try = project_box_sum(h - step * g, target_sum)
        elif mode == 1:
            # Random Gaussian perturbation
            size = rng.uniform(0.01, 0.06)
            h_try = project_box_sum(h + rng.randn(n) * size, target_sum)
        else:
            # Swap-based move
            h_try = h.copy()
            cnt = rng.randint(1, 5)
            for _ in range(cnt):
                i = rng.randint(n)
                j = rng.randint(n)
                h_try[i], h_try[j] = h_try[j], h_try[i]
            h_try = project_box_sum(h_try, target_sum)
        
        c5_try = compute_c5(h_try)
        delta = c5_try - compute_c5(h)
        
        # Accept if better, or with probability exp(-delta/T)
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, T_min)):
            h = h_try
            if c5_try < best:
                best = c5_try
        
        T = max(T * cooling, T_min)
    
    return h, best
```

```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def smooth_max_weights(corr, alpha):
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    s = np.sum(w)
    if s < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= s
    return w

def grad_corr_single(h, k, n):
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n-k])
        if k > 0:
            g[:n-k] -= h[k:]
    else:
        p = -k
        g[:n-p] += (1.0 - h[p:])
        if p > 0:
            g[p:] -= h[:n-p]
    return g

def compute_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    w = smooth_max_weights(corr, alpha)
    grad = np.zeros(n)
    active = np.where(w > 5e-8)[0]
    for idx in active:
        k = int(idx) - (n - 1)
        grad += w[idx] * grad_corr_single(h, k, n)
    return grad * (2.0 / n)

def project_box_sum(h, target_sum):
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def make_perturbation(n, rng, size):
    mode = rng.randint(0, 4)
    if mode == 0:
        return rng.randn(n) * size
    elif mode == 1:
        x = np.arange(n, dtype=float)
        f = rng.uniform(1.0, 6.0)
        phase = rng.uniform(0, 6.283)
        return size * 1.5 * np.sin(2.0 * np.pi * f * x / n + phase)
    elif mode == 2:
        pert = np.zeros(n)
        cnt = max(n // 6, 4)
        idx = rng.choice(n, size=cnt, replace=False)
        pert[idx] = rng.uniform(-size * 3, size * 3, size=cnt)
        return pert
    else:
        x = np.arange(n, dtype=float)
        f1, f2 = rng.uniform(1, 4), rng.uniform(4, 10)
        return size * (np.sin(2*np.pi*f1*x/n) + 0.5*np.sin(2*np.pi*f2*x/n))

def optimize(h0, target_sum, deadline, rng, alpha=100, max_iters=None):
    n = len(h0)
    h = project_box_sum(h0.copy(), target_sum)
    h_prev = h.copy()
    t = 1.0
    best = compute_c5(h)
    n_scaled = max(n, 100)
    lr = 15.0 / n_scaled
    stalls = 0
    a_cur = alpha
    best_lr = lr
    iters = 0
    max_it = max_iters if max_iters else 999999

    while time.time() < deadline and iters < max_it:
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta = (t - 1.0) / t_new
        y = h + beta * (h - h_prev)
        g = compute_grad(y, a_cur)
        h_new = project_box_sum(y - lr * g, target_sum)
        c5_new = compute_c5(h_new)
        iters += 1

        if c5_new < best - 1e-14:
            best = c5_new
            stalls = 0
            best_lr = lr
            if c5_new < best * 0.9995:
                lr = min(lr * 1.03, 30.0 / n_scaled)
        else:
            stalls += 1
            if stalls % 8 == 0:
                lr *= 0.9

        x_prev = h
        h = h_new
        h_prev = x_prev
        t = t_new

        if stalls > 30:
            h_prev = h.copy()
            t = 1.0
            pert_size = 3.0 / np.sqrt(n)
            pert = make_perturbation(n, rng, pert_size)
            h = project_box_sum(h + pert, target_sum)
            h_prev = h.copy()
            a_cur = min(a_cur * 1.35, 380.0)
            lr = best_lr
            stalls = 0

    return h, best

def interpolate_to_n(h_old, n_old, n_new):
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0 / n_new
    h_ext = np.append(h_old, h_old[0])
    x_ext = np.append(x_old, 2.0 + 1.0 / n_old)
    return np.interp(x_new, x_ext, h_ext)

def try_binary(h, n, deadline, rng):
    target = n // 2
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    h_opt, c5_opt = optimize(h_bin, float(target), deadline, rng, alpha=200, max_iters=2000)
    return h_opt, c5_opt

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = 0.5 + 0.3 * rng.randn(200)

    n0 = len(h0)
    h0 = project_box_sum(h0, n0 * 0.5)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Optimize at initial resolution
    dl1 = t0 + min(budget_s * 0.28, 220.0)
    best_h, best_c5 = optimize(h0, n0 * 0.5, dl1, rng, alpha=80)

    # Phase 1b: Try binary rounding
    if time.time() < t0 + budget_s * 0.35:
        h_b, c5_b = try_binary(best_h, len(best_h),
                                min(time.time() + 25, t0 + budget_s * 0.35), rng)
        if c5_b < best_c5:
            best_h, best_c5 = h_b, c5_b

    # Phase 2: Gradual resolution increase
    cur_n = len(best_h)
    target_ns = [300, 400, 500, 650, 800]
    remaining = budget_s - (time.time() - t0) - 40

    for n_try in target_ns:
        if remaining < 50 or time.time() > t0 + budget_s - 50:
            break
        if n_try <= cur_n:
            continue
        h_i = interpolate_to_n(best_h, cur_n, n_try)
        h_i = project_box_sum(h_i, n_try * 0.5)
        alloc = min(remaining * 0.28, 130.0)
        dl = time.time() + alloc
        h_o, c5_o = optimize(h_i, n_try * 0.5, dl, rng, alpha=120)
        if c5_o < best_c5:
            best_h, best_c5, cur_n = h_o, c5_o, n_try
        remaining = budget_s - (time.time() - t0) - 25

    # Phase 3: Multi-restart at best resolution
    remaining = budget_s - (time.time() - t0)
    if remaining > 40:
        n_best = len(best_h)
        n_restarts = min(4, int(remaining // 35))
        for r in range(n_restarts):
            if time.time() > t0 + budget_s - 15:
                break
            pert_size = (2.5 + r) / np.sqrt(n_best)
            pert = make_perturbation(n_best, rng, pert_size)
            h_p = project_box_sum(best_h + pert, n_best * 0.5)
            c5_p = compute_c5(h_p)
            dl = min(time.time() + 25, t0 + budget_s - 10)
            h_o, c5_o = optimize(h_p, n_best * 0.5, dl, rng, alpha=150)
            if c5_o < best_c5:
                best_h, best_c5 = h_o, c5_o

    # Phase 4: Final gradient polish
    remaining = budget_s - (time.time() - t0)
    if remaining > 10:
        best_h, best_c5 = optimize(best_h, len(best_h) * 0.5,
                                    t0 + budget_s - 2, rng, alpha=250, max_iters=3000)

    best_h = project_box_sum(best_h, len(best_h) * 0.5)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
```
