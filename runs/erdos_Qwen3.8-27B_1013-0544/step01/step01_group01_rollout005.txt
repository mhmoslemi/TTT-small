import numpy as np
from scipy.optimize import minimize
import time


def project_to_simplex_box(x, target_sum):
    """Project x onto {h: 0<=h[i]<=1, sum(h)=target_sum} via bisection on Lagrange multiplier."""
    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)


def grad_one_lag(h, lag_idx):
    """Exact gradient of corr[lag_idx] w.r.t. h. lag_idx is index into full correlation output."""
    n = len(h)
    shift = lag_idx - (n - 1)
    j = np.arange(n)
    g = np.zeros(n)
    m1 = (j + shift >= 0) & (j + shift < n)
    g[m1] += 1.0 - h[j[m1] + shift]
    m2 = (j - shift >= 0) & (j - shift < n)
    g[m2] -= h[j[m2] - shift]
    return g


def projected_gradient_step(h, target, top_k, lr, K):
    """One projected gradient step minimizing top-K correlations."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argsort(corr)[-K:][::-1]
    g = np.zeros(len(h))
    for ki in idx:
        g += grad_one_lag(h, ki)
    g /= K
    g -= g.mean()
    h_new = project_to_simplex_box(h - lr * g, target)
    return h_new, corr, idx


def optimize_at_n(h_init, n, target, budget_fn, max_iters=15000):
    """Optimize h using projected gradient with adaptive top-K and backtracking."""
    h = project_to_simplex_box(h_init, target)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    K = 5
    lr = 0.04
    stall = 0

    for it in range(max_iters):
        if budget_fn() < 30:
            break

        h_new, corr, top_idx = projected_gradient_step(h, target, top_idx, lr, K)
        c5_new = compute_c5(h_new)

        if c5_new <= best_c5:
            best_h, best_c5, h = h_new.copy(), c5_new, h_new
            stall = 0
            lr = min(0.06, lr * 1.01)
        else:
            improved = False
            for fct in [0.5, 0.25, 0.1, 0.05]:
                ht, _, _ = projected_gradient_step(h, target, top_idx, lr * fct, K)
                ct = compute_c5(ht)
                if ct < best_c5:
                    best_h, best_c5, h = ht.copy(), ct, ht
                    improved = True
                    break
            if not improved:
                stall += 1
                lr = max(0.003, lr * 0.85)
                if stall > 150:
                    h = best_h + np.random.randn(len(h)) * (0.05 + 0.05 * min(stall / 200.0, 1.0))
                    h = project_to_simplex_box(h, target)
                    lr = 0.03
                    stall = 0
                    K = 5

        if it > 0 and it % 100 == 0:
            sc = np.sort(corr)[::-1]
            if K < len(corr) and sc[min(K, len(sc)-1)] > 0.96 * sc[0]:
                K = min(20, K + 1)
            elif sc[min(K, len(sc)-1)] < 0.88 * sc[0]:
                K = max(2, K - 1)

    lr_fine = 0.002
    for _ in range(2000):

        c5_new = compute_c5(h_new)

            best_h, best_c5 = h_new.copy(), c5_new
        else:
            lr_fine *= 0.8
            if lr_fine < 1e-5:
                break

    return best_h, best_c5


def slsqp_polish(h, n, target, maxiter=300):
    """SLSQP polish using smooth max approximation."""
    dx = 2.0 / n

    def obj(hh, beta=100.0):
        c = np.correlate(hh, 1.0 - hh, mode="full")
        m = c.max()
        return (m + np.log(np.sum(np.exp(beta * (c - m))) + 1e-300) / beta) * dx

    def obj_grad(hh, beta=100.0):
        c = np.correlate(hh, 1.0 - hh, mode="full")
        m = c.max()
        weights = np.exp(beta * (c - m))
        weights /= weights.sum()
        g = np.zeros(n)
        for k in range(len(c)):
            if weights[k] > 1e-10:
                g += weights[k] * grad_one_lag(hh, k)
        return g * dx

    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=[(0, 1)] * n, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = project_to_simplex_box(res.x, target)
        return h_new
    except Exception:
        return h
