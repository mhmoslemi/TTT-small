    """Project x onto {sum(h) = target, lo <= h <= hi}."""

    # h_i = clip(x_i - lambda, lo, hi)
    # Need sum(h_i) = target

        h = np.clip(x - lam, lo, hi)
        return h.sum()
    
    lo_lam = x.min() - hi
    hi_lam = x.max() - lo
    
    for _ in range(100):

        if sum_at_lambda(mid) > target:
            lo_lam = mid
        else:

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

    corr = np.correlate(h, 1.0 - h, mode="full")

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_obj_grad(h, tau):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    x = tau * corr
    x_max = np.max(x)
    exps = np.exp(x - x_max)
    obj = (x_max + np.log(exps.sum())) / tau
    w = exps / exps.sum()
    grad = np.zeros(n)
    for di in range(2 * n - 1):
        d = di - (n - 1)
        wd = w[di]
        if wd < 1e-14:
            continue
        s1 = max(0, -d)
        e1 = min(n, n - d)
        if s1 < e1:
            grad[s1:e1] += wd * (1.0 - h[s1 + d:e1 + d])
        s2 = max(0, d)
        e2 = min(n, n + d)
        if s2 < e2:
            grad[s2:e2] -= wd * h[s2 - d:e2 - d]
    return obj, grad
