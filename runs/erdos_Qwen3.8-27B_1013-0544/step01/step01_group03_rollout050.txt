def binary_local_search(h, max_iter=10000):
    n = len(h)
    # Round to binary
    h_bin = (h > 0.5).astype(float)
    # Fix sum
    ones_count = h_bin.sum()
    if ones_count > n/2:
        # Need to flip some 1s to 0s
        idx = np.where(h_bin == 1)[0]
        # Flip the n/2 - ... positions
        excess = int(ones_count - n/2)
        h_bin[idx[:excess]] = 0
    elif ones_count < n/2:
        idx = np.where(h_bin == 0)[0]
        deficit = int(n/2 - ones_count)
        h_bin[idx[:deficit]] = 1
    
    c5 = evaluate_c5(h_bin)
    
    for it in range(max_iter):
        # Random swap
        i = np.random.randint(n)
        j = np.random.randint(n)
        if h_bin[i] == h_bin[j]:
            continue
        # Try swap
        h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
        new_c5 = evaluate_c5(h_bin)
        if new_c5 > c5:
            # Revert
            h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
        else:
            c5 = new_c5
    
    return h_bin, c5

import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def fix_sum(h, target):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = mask.sum()
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def compute_grad(h, weights):
    n = len(h)
    d = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)
    omh = 1.0 - h
    for i in range(n):
        idx1 = i + d
        v1 = (idx1 >= 0) & (idx1 < n)
        idx2 = i - d
        v2 = (idx2 >= 0) & (idx2 < n)
        grad[i] = np.dot(weights[v1], omh[idx1[v1]]) - np.dot(weights[v2], h[idx2[v2]])
    return grad

def softmax_weights(corr, tau):
    x = tau * corr
    x -= x.max()
    p = np.exp(x)
    p /= p.sum()
    return p

def softmax_optimize(h, n_pts, tau, max_it, lr_init=0.5):
    tgt = n_pts / 2.0
    h_cur = h.copy()
    lr = lr_init
    best_h = h_cur.copy()
    best_c5 = evaluate_c5(h_cur)
    for it in range(max_it):
        corr = get_corr(h_cur)
        p = softmax_weights(corr, tau)
        grad = compute_grad(h_cur, p)
        h_new = np.clip(h_cur - lr * grad, 0.0, 1.0)
        h_new = fix_sum(h_new, tgt)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        if c5_new <= evaluate_c5(h_cur) + 1e-10:
            h_cur = h_new
            lr = min(lr * 1.02, 0.5)
        else:
            lr *= 0.4
            if lr < 0.01:
                lr = 0.05
        if it % 200 == 199:
            lr = max(lr * 0.7, 0.01)
    return best_h, best_c5
