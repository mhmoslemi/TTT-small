def project_onto_simplex_box(v, target_sum):
    # Find τ such that sum(clip(v - τ, 0, 1)) = target_sum
    lo, hi = v.min() - 1, v.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(v - mid, 0, 1))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2, 0, 1)

This is the exact Euclidean projection onto the intersection of the box [0,1]^N and the hyperplane {sum = target_sum}.

```python
def subgradient_minimax(h0, n_iter=10000, lr_schedule=None):
    N = len(h0)
    h = h0.copy()
    # Number of shifts
    K = 2 * N - 1
    y = np.ones(K) / K  # uniform distribution over shifts
    
    for t in range(n_iter):
        # Compute all correlations
        corr = np.correlate(h, 1.0 - h, mode='full')
        
        # Update y (gradient ascent on simplex)
        lr_y = 1.0 / np.sqrt(t + 1)
        y = y * np.exp(lr_y * corr / max(np.max(corr), 1e-10))
        y = y / y.sum()
        
        # Compute expected gradient
        # g[j] = sum_k y_k * df_k/dh[j]
        # df_k/dh[j] = (1-h[j-k]) * I[0<=j-k<N] - h[j+k] * I[0<=j+k<N]
        # where k is the mathematical shift = corr_index - (N-1)
        
        g = np.zeros(N)
        for k_idx in range(K):
            if y[k_idx] < 1e-10:
                continue
            k = k_idx - (N - 1)  # mathematical shift
            # (1-h[j-k]) for valid j-k
            # h[j+k] for valid j+k
            if k >= 0:
                # j-k >= 0 means j >= k
                g[k:] += y[k_idx] * (1.0 - h[:N-k])
                # j+k < N means j < N-k
                g[:N-k] += y[k_idx] * (-h[k:])
            else:
                # k < 0, let k' = -k > 0
                kp = -k
                # j-k = j+kp >= 0 always (since j >= 0 and kp > 0), valid if j+kp < N
                g[:N-kp] += y[k_idx] * (1.0 - h[kp:])
                # j+k = j-kp < N means j >= kp (since j >= 0), valid if j-kp >= 0
                g[kp:] += y[k_idx] * (-h[:N-kp])
        
        # Projected gradient step
        lr_h = lr_schedule(t) if lr_schedule else 0.1 / np.sqrt(t + 1)
        h = h - lr_h * g
        h = project_onto_simplex_box(h, N * 0.5)
    
    return h
```

```python
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))


def project_onto_constraint(h, target_sum):
    """Project h onto {x: 0<=x<=1, sum(x)=target_sum} using bisection on threshold."""
    h = h.copy()
    # Find threshold tau such that sum(clip(h - tau, 0, 1)) = target_sum
    lo = h.max() - target_sum / len(h) + 1.0  # ensure sum < target at lo...

        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)


def compute_expected_gradient(h, y):
    """Compute sum_k y[k] * grad_k(h) where grad_k is gradient of corr_k w.r.t. h."""
    N = len(h)
    K = 2 * N - 1
    g = np.zeros(N)
    for k_idx in range(K):
        w = y[k_idx]
        if w < 1e-14:
            continue
        k = k_idx - (N - 1)
        # First part: (1-h[j-k]) for j where 0 <= j-k < N
        # j ranges from max(0, k) to min(N-1, N-1+k)
        j_lo = max(0, k)
        j_hi = min(N - 1, N - 1 + k)
        if j_hi >= j_lo:
            g[j_lo:j_hi+1] += w * (1.0 - h[j_lo - k : j_hi - k + 1])
        # Second part: -h[j+k] for j where 0 <= j+k < N
        # j ranges from max(0, -k) to min(N-1, N-1-k)
        j_lo2 = max(0, -k)
        j_hi2 = min(N - 1, N - 1 - k)
        if j_hi2 >= j_lo2:
            g[j_lo2:j_hi2+1] += -w * h[j_lo2 + k : j_hi2 + k + 1]
    return g

    """Subgradient minimax with exponential weights on shifts."""
    N = len(h0)
    K = 2 * N - 1
    h = h0.copy()
    y = np.ones(K) / K
    eta = eta0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    
    for t in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = np.max(corr)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
        
        # Update y with exponential weights
        # Normalize corr for numerical stability
        corr_shifted = corr - np.max(corr)
        y = y * np.exp(eta * corr_shifted)
        y_sum = y.sum()
        if y_sum < 1e-300:
            y_max = np.argmax(corr)
            y = np.zeros(K)
            y[y_max] = 1.0
        else:
            y /= y_sum
        
        # Compute expected gradient
        g = compute_expected_gradient(h, y)
        
        # Step size

        # Gradient step and project
        h = h - lr * g
        h = project_onto_constraint(h, N * 0.5)
        
        # Decay eta
        eta *= decay
    
    return best_h, best_c5
```

```python
def binary_local_search(h0, max_steps=1000, verbose=False):
    """Local search on binary sequence: try all moves, accept the best."""
    N = len(h0)
    K = 2 * N - 1
    h = h0.copy()
    h = (h > 0.5).astype(float)  # Binarize
    # Ensure exactly N/2 ones
    # (Should already be the case if h0 is from projection)
    
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    corr = np.correlate(h, 1.0 - h, mode='full')  # raw (without dx)
    dx = 2.0 / N
    
    for step in range(max_steps):
        ones_mask = h > 0.5
        zeros_mask = ~ones_mask
        ones_idx = np.where(ones_mask)[0]
        zeros_idx = np.where(zeros_mask)[0]
        
        if len(ones_idx) == 0 or len.zeros_idx) == 0:
            break
        
        # Compute B matrix: B[j, k] = (1-h[j-k])*I[0≤j-k<N] - h[j+k]*I[0≤j+k<N]
        # for all positions j (not just zeros, for simplicity)
        B = np.zeros((N, K))
        for k_idx in range(K):
            k = k_idx - (N - 1)
            # First part: (1-h[j-k]) for valid j-k
            j_lo = max(0, k)
            j_hi = min(N - 1, N - 1 + k)
            if j_hi >= j_lo:
                B[j_lo:j_hi+1, k_idx] = 1.0 - h[j_lo - k : j_hi - k + 1]
            # Second part: -h[j+k] for valid j+k
            j_lo2 = max(0, -k)
            j_hi2 = min(N - 1, N - 1 - k)
            if j_hi2 >= j_lo2:
                B[j_lo2:j_hi2+1, k_idx] -= h[j_lo2 + k : j_hi2 + k + 1]
        
        # For each zero position j and one position i, compute the new max correlation
        # New corr_k = corr_k + B[j, k] - B[i, k] (+ cross term)
        # D[j_idx, i_idx, k] = corr[k] + B[j, k] - B[i, k]
        
        B_zeros = B[zeros_idx]  # shape (n_zeros, K)
        B_ones = B[ones_idx]    # shape (n_ones, K)
        
        # D has shape (n_zeros, n_ones, K)
        D = corr[None, None, :] + B_zeros[:, None, :] - B_ones[None, :, :]
        
        # Add cross terms: for move i→j, cross +1 at k=j-i, -1 at k=i-j
        # k_idx = k + (N-1)
        for zi in range(len(zeros_idx)):
            for oi in range(len(ones_idx)):
                ji = zeros_idx[zi] - ones_idx[oi]
                ki = ones_idx[oi] - zeros_idx[zi]
                if 0 <= zi < len(zeros_idx):
                    pass
                # cross +1 at shift k=ji (if ji != 0)
                if ji != 0:
                    k_idx = ji + (N - 1)
                    if 0 <= k_idx < K:
                        D[zi, oi, k_idx] += 1.0
                if ki != 0:
                    k_idx = ki + (N - 1)
                    if 0 <= k_idx < K:
                        D[zi, oi, k_idx] -= 1.0
        
        # Find the best move
        D_max = D.max(axis=2)  # shape (n_zeros, n_ones)
        min_val = D_max.min()
        
        if min_val < corr.max() - 1e-10:
            zi, oi = np.unravel_index(D_max.argmin(), D_max.shape)
            j = zeros_idx[zi]
            i = ones_idx[oi]
            h[i] = 0.0
            h[j] = 1.0
            corr = D[zi, oi]  # Updated correlation
            c5_new = corr.max() * dx
            if c5_new < best_c5:
                best_c5 = c5_new
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))


def project_onto_constraint(h, target_sum):
    h = h.copy()
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)


def compute_expected_gradient(h, y):
    N = len(h)
    K = 2 * N - 1
    g = np.zeros(N)
    for k_idx in range(K):
        w = y[k_idx]
        if w < 1e-14:
            continue
        k = k_idx - (N - 1)
        j_lo = max(0, k)
        j_hi = min(N - 1, N - 1 + k)
        if j_hi >= j_lo:
            g[j_lo:j_hi + 1] += w * (1.0 - h[j_lo - k:j_hi - k + 1])
        j_lo2 = max(0, -k)
        j_hi2 = min(N - 1, N - 1 - k)
        if j_hi2 >= j_lo2:
            g[j_lo2:j_hi2 + 1] += -w * h[j_lo2 + k:j_hi2 + k + 1]
    return g


def subgradient_minimax(h0, n_iter, eta0=2.0, decay=0.9995, seed=42, momentum=0.9):
    N = len(h0)
    K = 2 * N - 1
    h = h0.copy()
    y = np.ones(K) / K
    eta = eta0
    vel = np.zeros(N)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for t in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = np.max(corr) * (2.0 / N)
        if c5_val < best_c5 - 1e-12:
            best_c5 = c5_val
            best_h = h.copy()
        corr_shifted = corr - np.max(corr)
        y = y * np.exp(eta * corr_shifted)
        ys = y.sum()
        if ys < 1e-300:
            y = np.zeros(K)
            y[np.argmax(corr)] = 1.0
        else:
            y /= ys
        g = compute_expected_gradient(h, y)
        lr = 0.3 / np.sqrt(t + 1)
        vel = momentum * vel + (1 - momentum) * g
        h = h - lr * vel
        h = project_onto_constraint(h, N * 0.5)
        eta *= decay
    return best_h, best_c5


def binary_local_search(h0, max_steps=2000):
    N = len(h0)
    K = 2 * N - 1
    h = (h0 > 0.5).astype(float)
    n_ones = int(np.sum(h > 0.5))
    if n_ones != N // 2:
        idx = np.where(h > 0.5)[0]
        if n_ones > N // 2:
            h[idx[-(n_ones - N // 2):]] = 0.0
        else:
            idx2 = np.where(h < 0.5)[0]
            h[idx2[:N // 2 - n_ones]] = 1.0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(max_steps):
        ones_idx = np.where(h > 0.5)[0]
        zeros_idx = np.where(h < 0.5)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        B = np.zeros((N, K))
        for k_idx in range(K):
            k = k_idx - (N - 1)
            j_lo = max(0, k)
            j_hi = min(N - 1, N - 1 + k)
            if j_hi >= j_lo:
                B[j_lo:j_hi + 1, k_idx] = 1.0 - h[j_lo - k:j_hi - k + 1]
            j_lo2 = max(0, -k)
            j_hi2 = min(N - 1, N - 1 - k)
            if j_hi2 >= j_lo2:
                B[j_lo2:j_hi2 + 1, k_idx] -= h[j_lo2 + k:j_hi2 + k + 1]
        Bz = B[zeros_idx]
        Bo = B[ones_idx]
        D = corr[None, None, :] + Bz[:, None, :] - Bo[None, :, :]
        JI = zeros_idx[:, None].astype(float) - ones_idx[None, :].astype(float)
        kp = (JI + (N - 1)).astype(int)
        km = (-JI + (N - 1)).astype(int)
        vp = (JI != 0) & (kp >= 0) & (kp < K)
        zm, om = np.where(vp)
        D[zm, om, kp[zm, om]] += 1.0
        vm = (JI != 0) & (km >= 0) & (km < K)
        zm2, om2 = np.where(vm)
        D[zm2, om2, km[zm2, om2]] -= 1.0
        Dmax = D.max(axis=2)
        if Dmax.min() < corr.max() - 0.5:
            zi, oi = np.unravel_index(Dmax.argmin(), Dmax.shape)
            h[ones_idx[oi]] = 0.0
            h[zeros_idx[zi]] = 1.0
            corr = D[zi, oi].copy()
            c5_new = corr.max() * dx
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            break
    return best_h, best_c5


def t_objective(v):
    return v[-1]


def t_ineq(v):
    n = len(v) - 1
    h = v[:n]
    t = v[-1]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return t - corr


def t_eq_fun(v):
    n = len(v) - 1
    return np.sum(v[:n]) - n * 0.5


def t_eq_jac(v):
    n = len(v) - 1
    j = np.zeros(len(v))
    j[:n] = 1.0
    return j


def optimize_t_var(h0, max_iter=500, ftol=1e-12):
    n = len(h0)
    c5_cur = compute_c5(h0)
    v0 = np.append(h0.copy(), c5_cur + 0.01)
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.5)]
    cons = [
        {'type': 'ineq', 'fun': t_ineq},
        {'type': 'eq', 'fun': t_eq_fun, 'jac': t_eq_jac}
    ]
    res = minimize(t_objective, v0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': ftol})
    h_new = np.clip(res.x[:n], 0.0, 1.0)
    h_new = project_onto_constraint(h_new, n * 0.5)
    return h_new, compute_c5(h_new)


def smooth_obj(h, temp):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    if temp < 1e-10:
        return m
    return m + temp * np.log(np.sum(np.exp((corr - m) / temp)))


def sum_eq_fun(h):
    return np.sum(h) - len(h) * 0.5


def sum_eq_jac(h):
    return np.ones(len(h))


def optimize_smooth(h0, temp, max_iter=200, ftol=1e-13):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac}]
    res = minimize(smooth_obj, h0.copy(), args=(temp,), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': ftol})
    h_new = np.clip(res.x, 0.0, 1.0)
    h_new = project_onto_constraint(h_new, n * 0.5)
    return h_new, compute_c5(h_new)


def make_binary_random(N, seed=0):
    rng = np.random.RandomState(seed)
    idx = rng.choice(N, size=N // 2, replace=False)
    h = np.zeros(N)
    h[idx] = 1.0
    return h


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()

    def time_left():
        return budget_s - (time.time() - t0)

    N = len(initial_h_values)
    h_init = np.array(initial_h_values, dtype=float)
    h_init = project_onto_constraint(h_init, N * 0.5)

    h_best = h_init.copy()
    c5_best = compute_c5(h_best)

    # Phase 1: Subgradient minimax from initial
    if time_left() > 250:

        h_new, c5_new = subgradient_minimax(h_best, n_iter, eta0=2.0, decay=0.9995, seed=seed, momentum=0.85)
        if c5_new < c5_best:
            h_best, c5_best = h_new, c5_new

    # Phase 2: Binary local search
    if time_left() > 250:
        h_bin = (h_best > 0.5).astype(float)
        h_bin_c5 = compute_c5(h_bin)
        if h_bin_c5 < c5_best + 0.02:
            h_bls, c5_bls = binary_local_search(h_bin, max_steps=1500)
            if c5_bls < c5_best:
                # Soften the binary solution for continuous optimization
                h_soft = np.clip(h_bls + 0.05 * (np.random.RandomState(seed).randn(N) * 0.3), 0, 1)
                h_soft = project_onto_constraint(h_soft, N * 0.5)
                h_sub, c5_sub = subgradient_minimax(h_soft, 3000, eta0=1.5, decay=0.9997, seed=seed + 7)
                if c5_sub < c5_best:
                    h_best, c5_best = h_sub, c5_sub
                else:
                    h_best, c5_best = h_bls, c5_bls

    # Phase 3: SLSQP polish
    if time_left() > 120:
        h_new, c5_new = optimize_t_var(h_best, max_iter=500, ftol=1e-12)
        if c5_new < c5_best:
            h_best, c5_best = h_new, c5_new

    # Phase 4: Smooth refinement
    for temp in [0.01, 0.005, 0.002, 0.001, 0.0005]:
        if time_left() < 80:

        h_new, c5_new = optimize_smooth(h_best, temp, max_iter=200, ftol=1e-13)
```
