```python
def minimize_convex_plinear(a, c, lo, hi):
    """Minimize max_i (a[i]*x + c[i]) over x in [lo, hi]."""
    # Check endpoints
    def g(x):
        return np.max(a * x + c)
    
    # Binary search for where subgradient crosses 0
    # Subgradient at x is [min active slope, max active slope]
    # Active at x: indices where a[i]*x + c[i] == g(x)
    
    best_val = min(g(lo), g(hi))
    best_x = lo if g(lo) <= g(hi) else hi
    
    # Try binary search
    for _ in range(50):
        mid = (lo + hi) / 2
        vals = a * mid + c
        max_val = np.max(vals)
        active = np.abs(vals - max_val) < 1e-10
        min_slope = a[active].min()
        max_slope = a[active].max()
        if min_slope > 0:
            hi = mid  # function is increasing, go left
        elif max_slope < 0:
            lo = mid  # function is decreasing, go right
        else:
            break  # 0 is in subgradient, found optimum
        best_val = min(best_val, max_val)
        best_x = mid if max_val < best_val else best_x
    
    return best_x, best_val
```

def coordinate_descent_pass(h, n, max_sweeps=10):
    dx = 2.0 / n
    target_sum = n / 2.0
    h = h.copy()
    
    for sweep in range(max_sweeps):
        improved = False
        order = np.random.permutation(n)
        for j in order:
            # Compute a_m for all lags
            s_vals = np.arange(-(n-1), n)  # shifts
            a = np.zeros(2*n - 1)
            for idx, s in enumerate(s_vals):
                a[idx] = 0.0
                if 0 <= j + s < n:
                    a[idx] += (1 - h[j + s])
                if 0 <= j - s < n:
                    a[idx] -= h[j - s]
            
            corr = np.correlate(h, 1 - h, mode="full")
            lo = -h[j]
            hi = 1 - h[j]
            
            # Minimize max_m (a_m * delta + corr[m]) over delta in [lo, hi]

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    """Project h onto {0 <= h[i] <= 1, sum(h) = n/2} using bisection."""
    target = n / 2.0
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    mid = (lo + hi) / 2.0
    return np.clip(h - mid, 0.0, 1.0)

def coord_update(j, h, corr):
    """Find optimal delta for coordinate j. Returns new h[j]."""
    n = len(h)
    # Compute a_m for all lags
    # a_m = (1-h[j+s])*I(0<=j+s<n) - h[j-s]*I(0<=j-s<n) where s = m-(n-1)
    s_vals = np.arange(-(n-1), n)  # shifts for each m
    a = np.zeros(2*n - 1)
    # For shift s: j+s valid means 0 <= j+s < n
    js = j + s_vals  # j + s for each s
    valid_js = (js >= 0) & (js < n)
    a[valid_js] += (1.0 - h[js[valid_js]])
    
    js2 = j - s_vals  # j - s for each s
    valid_js2 = (js2 >= 0) & (js2 < n)
    a[valid_js2] -= h[js2[valid_js2]]
    
    # Minimize max_m (a[m]*delta + corr[m]) over delta in [-h[j], 1-h[j]]
    lo = -h[j]
    hi = 1.0 - h[j]
    if lo >= hi:
        return h[j]
    
    # Bisection on subgradient
    best_delta = 0.0
    best_val = np.max(a * 0.0 + corr)  # = max(corr)
    
    for _ in range(40):
        mid = (lo + hi) / 2.0
        vals = a * mid + corr
        max_val = np.max(vals)
        if max_val < best_val:
            best_val = max_val
            best_delta = mid
        active = np.abs(vals - max_val) < 1e-12
        min_slope = a[active].min()
        max_slope = a[active].max()
        if min_slope > 0:
            hi = mid
        elif max_slope < 0:
            lo = mid
        else:
            break
    
    return h[j] + best_delta
