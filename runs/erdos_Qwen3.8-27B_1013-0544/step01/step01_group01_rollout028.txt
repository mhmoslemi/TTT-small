    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) > 1e-12 and s > 0:
        h += (target_sum - s) / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_subgradient(h, active_lags, dx):
    """Compute subgradient of C5 w.r.t. h components."""
    n = len(h)
    g = np.zeros(n)
    for lag in active_lags:

            # - h[j + lag] * 1_{j+lag in [0,n-1]}

            if 0 <= ir < n:

    return dx * g

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) > 1e-12:
        h += (target_sum - s) / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def proj_onto_box_sum(h, target_sum):
    """Water-filling projection onto {0<=h<=1, sum h = target_sum}."""
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        projected = np.clip(h - mid, 0.0, 1.0)
        s = projected.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    return np.clip(h - lam, 0.0, 1.0)

def get_active_lags(h, tol):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    mx = np.max(corr)
    active = np.where(corr >= mx - tol)[0]
    return active - (n - 1)

def compute_subgradient(h, active_lags, dx):
    n = len(h)
    g = np.zeros(n)
    for lag in active_lags:
        for j in range(n):
            il = j - lag
            if 0 <= il < n:
                g[j] += (1.0 - h[il])
            ir = j + lag
            if 0 <= ir < n:
                g[j] -= h[ir]
    return dx * g

def subgradient_descent(h, n_points, dx, target_sum, max_iter, tol, step_init):
    best_c5 = compute_c5(h)
    best_h = h.copy()
    upper_bound = best_c5 + 0.02
    step = step_init
    no_improve = 0
    for it in range(max_iter):
        active = get_active_lags(best_h, tol)
        if len(active) == 0:
            active = np.array([np.argmax(np.correlate(best_h, 1.0 - best_h, mode="full")) - (n_points - 1)])
        g = compute_subgradient(best_h, active, dx)
        g_proj = g - g.mean()
        g_norm = np.sqrt(np.sum(g_proj ** 2))
        if g_norm < 1e-12:
            break
        gap = max(upper_bound - best_c5, 1e-4)
        alpha = min(step, gap / (g_norm ** 2 + 1e-12))
        alpha = max(alpha, 1e-9)
        h_new = proj_onto_box_sum(best_h - alpha * g_proj, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
            no_improve = 0
            step = min(step * 1.05, 0.08)
        else:
            no_improve += 1
            step *= 0.92
        if no_improve > 80:
            break
    return best_h, best_c5

def slsqp_polish(h, n_points, dx, target_sum, beta, max_iter):
    bounds = Bounds(np.zeros(n_points), np.ones(n_points))
    one_vec = np.ones(n_points)
    def obj_func(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        return smooth_max(corr, beta) * dx
    def con_fun(x):
        return np.sum(x) - target_sum
    def con_jac(x):
        return one_vec
    constraints = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj_func, h, method='SLSQP', bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': max_iter, 'ftol': 1e-13})
        h_new = proj_onto_box_sum(res.x, target_sum)
        return h_new, compute_c5(h_new)
    except Exception:
        return h, compute_c5(h)

def try_optimize(h_init, n_points, dx, target_sum, start_time, budget_s):
    best_h = proj_onto_box_sum(h_init, target_sum)
    best_c5 = compute_c5(best_h)

    def time_left():
        return budget_s - (time.time() - start_time)

    # Phase 1: Subgradient descent
    if time_left() > 30:
        best_h, best_c5 = subgradient_descent(best_h, n_points, dx, target_sum,
                                               max_iter=300, tol=0.015, step_init=0.01)

    # Phase 2: SLSQP polish with increasing beta
    for beta in [40, 80, 150]:
        if time_left() < 20:
            break
        h_new, c5_new = slsqp_polish(best_h, n_points, dx, target_sum, beta, 1500)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 3: Diverse restarts
    attempt = 0
    no_improve_total = 0
    rng = np.random.RandomState(7 + n_points)
    while time_left() > 25 and no_improve_total < 60:
        attempt += 1
        strategy = attempt % 7
        h0 = best_h.copy()
        scale_base = 0.01 + 0.04 * np.exp(-attempt / 30.0)

        if strategy == 0:
            n_p = max(3, n_points // 6)
            idx = rng.choice(n_points, n_p, replace=False)
            h0[idx] += rng.randn(n_p) * scale_base
        elif strategy == 1:
            corr = np.correlate(h0, 1.0 - h0, mode="full")
            top_idx = np.argsort(corr)[-3:]
            for k in top_idx:
                lag = k - (n_points - 1)
                for j in range(n_points):
                    if 0 <= j - lag < n_points:
                        h0[j] -= 0.015 * (1.0 - h0[j - lag])
                    if 0 <= j + lag < n_points:
                        h0[j] -= 0.015 * h0[j + lag]
            h0 += rng.randn(n_points) * 0.004
        elif strategy == 2:
            n_swaps = rng.randint(1, 5)
            for _ in range(n_swaps):
                i, j = rng.choice(n_points, 2, replace=False)
                d = h0[i] - h0[j]
                h0[i] -= d
                h0[j] += d
        elif strategy == 3:
            blen = rng.randint(3, max(4, n_points // 5))
            bs = rng.randint(0, max(1, n_points - blen))
            h0[bs:bs + blen] = rng.random(blen) * 0.6 + 0.2
        elif strategy == 4:
            active = get_active_lags(h0, 0.01)
            g = compute_subgradient(h0, active, dx)
            g_proj = g - g.mean()
            h0 = h0 - 0.015 * g_proj
            h0 += rng.randn(n_points) * 0.003
        elif strategy == 5:
            mask = rng.random(n_points) < 0.12
            h0[mask] = rng.random(int(mask.sum()))
        else:
            half = n_points // 2
            shift = rng.randint(1, half)
            temp = h0.copy()
            for i in range(n_points):
                h0[(i + shift) % n_points] = temp[i]
            h0 += rng.randn(n_points) * 0.005

        h0 = proj_onto_box_sum(h0, target_sum)

        # Subgradient refinement
        h0, _ = subgradient_descent(h0, n_points, dx, target_sum,
                                     max_iter=80, tol=0.01, step_init=0.008)
        # SLSQP polish
        h_new, c5_new = slsqp_polish(h0, n_points, dx, target_sum, 100, 600)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new
            no_improve_total = 0
        else:
            no_improve_total += 1

    return best_h, best_c5
