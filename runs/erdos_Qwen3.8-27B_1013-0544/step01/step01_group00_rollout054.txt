    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-10
    else:
        mask = h > 1e-10
    cnt = np.sum(mask)
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def t_objective(v):
    return v[-1]

def t_ineq(v):
    n = len(v) - 1
    h = v[:n]
    t = v[-1]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return t - corr

    n = len(v) - 1
    return np.sum(v[:n]) - n * 0.5

def t_eq_jac(v):

def smooth_obj(h, temp):

    m = np.max(corr)
    shifted = np.clip((corr - m) / temp, -50, 0)

    h_new = np.interp(x_new, x_old, h_old)

def optimize_t_var(h0, max_iter=400, ftol=1e-10):

    h_new = np.clip(res.x[:n], 0.0, 1.0)
    h_new = project_h(h_new, n * 0.5)

def projected_gradient_lse(h0, temp, n_iter=500, lr=0.5, momentum=0.9):
    """Projected gradient with momentum on LSE objective."""
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)
    target = n * 0.5
    
    for i in range(n_iter):
        # Compute gradient
        corr = compute_corr(h)
        m = np.max(corr)
        shifted = np.clip((corr - m) / temp, -50, 0)
        w = np.exp(shifted)
        w /= np.sum(w)
        
        # Gradient: g[j] = dx * (1 - sum_k w[k]*h[j-k] - sum_k w[k]*h[j+k])
        dx = 2.0 / n
        g = np.zeros(n)
        for j in range(n):
            s = 0.0
            for k in range(len(w)):
                ik = j - k
                if 0 <= ik < n:
                    s += w[k] * h[ik]
                ik2 = j + k
                if 0 <= ik2 < n:
                    s += w[k] * h[ik2]
            g[j] = dx * (1.0 - s)
        
        # Momentum update
        v = momentum * v - lr * g
        h_new = h + v
        h_new = project_h(h_new, target)
        
        # Check improvement
        obj_new = smooth_obj(h_new, temp)
        obj_old = smooth_obj(h, temp)
        if obj_new < obj_old:
            h = h_new
        else:
            # Reduce step size
            lr *= 0.9
            v = v * 0.5
    
    h = project_h(h, target)
    return h, compute_c5(h)

        h = np.clip(h, 0.0, 1.0)

                   options={'maxiter': max_iter, 'ftol': ftol})
    h_new = np.clip(res.x, 0.0, 1.0)
    h_new = project_h(h_new, n * 0.5)
    return h_new, compute_c5(h_new)

def gen_block_h(n, rng, n_blocks=None):
    if n_blocks is None:
        n_blocks = rng.randint(3, max(4, n // 8))
    h = np.zeros(n)
    for i in range(n_blocks):
        start = int(i * n / n_blocks)
        end = int((i + 1) * n / n_blocks)
        if rng.rand() > 0.45:
            h[start:end] = rng.uniform(0.6, 1.0)
        else:
            h[start:end] = rng.uniform(0.0, 0.35)
    return project_h(h, n * 0.5)

def gen_binary_h(n, rng, density=0.5):
    """Generate a random binary-like step function."""
    n_switches = rng.randint(2, max(3, n // 15))
    switches = sorted(rng.choice(range(1, n), size=n_switches, replace=False))
    h = np.zeros(n)
    h[switches[0]:] = 1.0
    for s in switches[1:]:
        h[s:] = 1.0 - h[s]
    h = np.where(h > 0.5, rng.uniform(0.8, 1.0), rng.uniform(0.0, 0.2))
    return project_h(h, n * 0.5)

def gen_shifted_h(h_base, n, rng):
    """Create variations by shifting and perturbing a base function."""
    shift = rng.randint(-5, 6)
    h_shifted = np.roll(h_base, shift)
    return project_h(h_shifted, n * 0.5)

def pgd_lse(h0, temp, n_iter=200, lr_init=0.05, momentum=0.85):
    """Projected gradient with momentum on LSE objective (vectorized gradient)."""
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    v = np.zeros(n)
    target = n * 0.5
    lr = lr_init
    obj_prev = lse_obj(h, temp)
    
    for i in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        m = np.max(corr)
        shifted = np.clip((corr - m) / temp, -60, 0)
        w = np.exp(shifted)
        sw = np.sum(w)
        w = w / sw
        
        # Vectorized gradient computation using FFT
        N = 1 << (4 * n).bit_length()
        H = np.fft.rfft(h, N)
        W = np.fft.rfft(w, N)
        Wh = np.fft.irfft(H * W, N)
        Wf = np.fft.rfft(w[::-1], N)
        Whf = np.fft.irfft(H * Wf, N)
        
        # g[j] = dx * (1 - Wh[j] - Whf[j]) (approximate, valid for interior points)
        g = dx * (1.0 - Wh[:n] - Whf[:n])
        
        v = momentum * v - lr * g
        h_new = h + v
        h_new = project_h(h_new, target)
        
        obj_new = lse_obj(h_new, temp)
        if obj_new < obj_prev:
            h = h_new
            obj_prev = obj_new
            lr = min(lr * 1.02, 0.5)
        else:
            lr *= 0.8
            v *= 0.5
            if lr < 1e-6:
                break
    
    h = project_h(h, target)
    return h, compute_c5(h)

def refine_solution(h, n, time_limit):
    """Multi-stage refinement of a solution on grid of size n."""
    h = project_h(h, n * 0.5)
    c5 = compute_c5(h)
    
    if time_limit > 60:
        h_new, c5_new = optimize_t_var(h, max_iter=400, ftol=1e-10)
        if c5_new < c5:
            h, c5 = h_new, c5_new
    
    if time_limit > 50:
        h_new, c5_new = optimize_smooth(h, 0.005, max_iter=200, ftol=1e-11)
        if c5_new < c5:
            h, c5 = h_new, c5_new
    
    if time_limit > 40:
        h_new, c5_new = optimize_t_var(h, max_iter=300, ftol=1e-11)
        if c5_new < c5:
            h, c5 = h_new, c5_new
    
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    def elapsed():
        return time.time() - t0
    
    def time_left():
        return budget_s - elapsed()
    
    # Start from parent
    h_parent = np.array(initial_h_values, dtype=float)
    n_parent = len(h_parent)
    h = project_h(h_parent, n_parent * 0.5)
    c5 = compute_c5(h)
    h_best = h.copy()
    c5_best = c5
    
    # Phase 1: Quick SLSQP on parent
    if time_left() > 80:
        h_new, c5_new = optimize_t_var(h, max_iter=400, ftol=1e-10)
        if c5_new < c5_best:
            h_best, c5_best = h_new.copy(), c5_new
    
    # Phase 2: Smooth SLSQP refinement
    if time_left() > 60:
        h_new, c5_new = optimize_smooth(h_best, 0.01, max_iter=250, ftol=1e-11)
        if c5_new < c5_best:
            h_best, c5_best = h_new.copy(), c5_new
        h_new, c5_new = optimize_smooth(h_best, 0.003, max_iter=250, ftol=1e-12)
        if c5_new < c5_best:
            h_best, c5_best = h_new.copy(), c5_new
    
    # Phase 3: Coarse grid exploration with diverse starts
    coarse_configs = [(30, 4), (40, 4), (50, 4), (60, 3), (80, 3)]
    for n_c, n_starts in coarse_configs:
        if time_left() < 150:
            break
        for s in range(n_starts):
            if time_left() < 80:
                break
            # Generate diverse starts
            if s == 0:
                h_start = interp_h(h_best, n_c)
            elif s == 1:
                h_start = gen_block_h(n_c, rng)
            elif s == 2:
                h_start = gen_binary_h(n_c, rng)
            else:
                h_start = gen_block_h(n_c, rng, n_blocks=rng.randint(3, 8))
            
            # Optimize at coarse resolution
            h_c, c5_c = optimize_t_var(h_start, max_iter=300, ftol=1e-10)
            if c5_c < 0.42 and time_left() > 40:
                h_c, c5_c = optimize_smooth(h_c, 0.005, max_iter=150, ftol=1e-11)
            
            # If coarse solution is good, interpolate and refine at parent resolution
            if c5_c < c5_best + 0.01 and time_left() > 60:
                h_full = interp_h(h_c, n_parent)
                h_f, c5_f = optimize_t_var(h_full, max_iter=300, ftol=1e-10)
                if c5_f < c5_best and time_left() > 30:
                    h_f, c5_f = optimize_smooth(h_f, 0.003, max_iter=150, ftol=1e-12)
                    if c5_f < c5_best:
                        h_best, c5_best = h_f.copy(), c5_f
    
    # Phase 4: Projected gradient on LSE at parent resolution
    if time_left() > 100:
        for temp in [0.005, 0.002, 0.001]:
            if time_left() < 50:
                break
            h_new, c5_new = pgd_lse(h_best, temp, n_iter=200, lr_init=0.03, momentum=0.8)
            if c5_new < c5_best:
                h_best, c5_best = h_new.copy(), c5_new
    
    # Phase 5: Final refinement
    if time_left() > 80:
        h_new, c5_new = optimize_t_var(h_best, max_iter=600, ftol=1e-11)
        if c5_new < c5_best:
            h_best, c5_best = h_new.copy(), c5_new
    
    if time_left() > 50:
        h_new, c5_new = optimize_smooth(h_best, 0.001, max_iter=300, ftol=1e-12)
        if c5_new < c5_best:
            h_best, c5_best = h_new.copy(), c5_new
    
    if time_left() > 50:
        h_new, c5_new = optimize_t_var(h_best, max_iter=500, ftol=1e-11)
        if c5_new < c5_best:
            h_best, c5_best = h_new.copy(), c5_new
    
    # Phase 6: Try slightly different resolutions
    if time_left() > 150:
        for n_try in [120, 150, 200]:
            if time_left() < 80:
                break
            h_interp = interp_h(h_best, n_try)
            h_t, c5_t = optimize_t_var(h_interp, max_iter=350, ftol=1e-10)
            if c5_t < 0.40 and time_left() > 30:
                h_t, c5_t = optimize_smooth(h_t, 0.003, max_iter=150, ftol=1e-12)
            if c5_t < c5_best:
                # Interpolate back to parent resolution
                h_back = interp_h(h_t, n_parent)
                h_b, c5_b = optimize_t_var(h_back, max_iter=200, ftol=1e-10)
                if c5_b < c5_best:
                    h_best, c5_best = h_b.copy(), c5_b
    
    # Phase 7: Final polish
    if time_left() > 40:
        h_new, c5_new = optimize_t_var(h_best, max_iter=400, ftol=1e-12)
        if c5_new < c5_best:
            h_best, c5_best = h_new.copy(), c5_new
    
    # Ensure clean output
    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = project_h(h_best, len(h_best) * 0.5)
    c5_final = compute_c5(h_best)
    return h_best, c5_final, len(h_best)
