    corr = np.correlate(h, 1 - h, mode="full")

    return np.clip(h, 0.0, 1.0)

    best_c5 = compute_c5(h, dx)
    for _ in range(n_rounds):
        # Pick a random pair
        i, j = np.random.choice(n, 2, replace=False)
        c = h[i] + h[j]  # sum to preserve
        # Optimize: minimize smooth_max over h[i] in [max(0,c-1), min(1,c)]
        lo = max(0.0, c - 1.0)
        hi = min(1.0, c)
        
        if lo >= hi:
            continue
            
        def obj_1d(x):
            h_new = h.copy()
            h_new[i] = x
            h_new[j] = c - x
            return smooth_max_obj(h_new, t, dx)
        
        from scipy.optimize import minimize_scalar
        res = minimize_scalar(obj_1d, bounds=(lo, hi), method='bounded', 
                            options={'xatol': 1e-10})

            h[j] = c - res.x

import numpy as np
import time
from scipy.optimize import minimize, minimize_scalar, fmin

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_obj(h, t, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    return (m + (1.0 / t) * np.log(np.sum(np.exp(corr - m)))) * dx

def fix_sum(h, target_sum):
    h = h.copy()
    h += (target_sum - np.sum(h)) / len(h)
    return np.clip(h, 0.0, 1.0)

def slsqp_opt(h, dx, ts, t=50, maxiter=800):
    n = len(h)
    try:
        res = minimize(
            lambda x: smooth_max_obj(x, t, dx), h.copy(),
            method='SLSQP', bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - ts}],
            options={'maxiter': maxiter, 'ftol': 1e-13})
        return fix_sum(res.x, ts)
    except Exception:
        return h

def pair_descent(h, dx, ts, n_rounds=200, t=50):
    n = len(h)
    best_c5 = compute_c5(h, dx)
    for _ in range(n_rounds):
        i, j = np.random.choice(n, 2, replace=False)
        c = h[i] + h[j]
        lo = max(0.0, c - 1.0)
        hi = min(1.0, c)
        if hi - lo < 1e-12:
            continue
        def obj_1d(x):
            h_new = h.copy()
            h_new[i], h_new[j] = x, c - x
            return smooth_max_obj(h_new, t, dx)
        try:
            res = minimize_scalar(obj_1d, bounds=(lo, hi), method='bounded',
                                  options={'xatol': 1e-10})
            if res.fun < obj_1d(h[i]) - 1e-14:
                h[i], h[j] = res.x, c - res.x
                nc = compute_c5(h, dx)
                if nc < best_c5:
                    best_c5 = nc
        except Exception:
            pass
    return h, best_c5

def nm_group_descent(h, dx, ts, gs=10, rounds=60, t=30):

    best_c5 = compute_c5(h, dx)

        c_sum = target_sum - (np.sum(h) - np.sum(h[idx]))
        sub = h[idx].copy()
        def sobj(x):
            h_s = h.copy()
            h_s[idx] = np.clip(x, 0, 1)
            h_s[idx] += (c_sum - np.sum(h_s[idx])) / gs
            h_s[idx] = np.clip(h_s[idx], 0, 1)
            return smooth_max_obj(h_s, t, dx)
        try:
            r = fmin(sobj, sub.copy(), maxiter=40, maxfun=400, disp=False, xtol=1e-9)
            h_new = h.copy()
            h_new[idx] = np.clip(r, 0, 1)
            h_new[idx] += (c_sum - np.sum(h_new[idx])) / gs
            h_new[idx] = np.clip(h_new[idx], 0, 1)
            nc = compute_c5(h_new, dx)
            if nc < best_c5:
                h, best_c5 = h_new, nc
        except Exception:
            pass
    return h, best_c5

def binary_sa(h, dx, ts, iters=15000, T0=0.04, T1=0.003):
    n = len(h)
    hb = (h > 0.5).astype(float)
    ones_ct = int(np.sum(hb))
    tgt = int(ts)
    if ones_ct > tgt:
        oi = np.where(hb == 1)[0]
        order = np.argsort(h[oi])
        for k in range(ones_ct - tgt):
            hb[oi[order[k]]] = 0
    elif ones_ct < tgt:
        zi = np.where(hb == 0)[0]
        order = np.argsort(-h[zi])
        for k in range(tgt - ones_ct):
            hb[zi[order[k]]] = 1
    best_c5 = compute_c5(hb, dx)
    cur = hb.copy()
    cur_c5 = best_c5
    for it in range(iters):
        T = T0 * (T1 / T0) ** (it / max(iters, 1))
        oi = np.where(cur == 1)[0]
        zi = np.where(cur == 0)[0]
        a = oi[np.random.randint(len(oi))]
        b = zi[np.random.randint(len(zi))]
        cur[a], cur[b] = 0, 1
        nc = compute_c5(cur, dx)
        d = nc - cur_c5
        if d < 0 or np.random.random() < np.exp(-d / max(T, 1e-9)):
            cur_c5 = nc
            if nc < best_c5:
                best_c5 = nc
        else:
            cur[a], cur[b] = 1, 0
    return cur, best_c5

    return np.interp(x_new, x_old, h)
