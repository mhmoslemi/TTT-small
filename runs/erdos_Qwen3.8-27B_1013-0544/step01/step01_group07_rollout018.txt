import numpy as np

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_grad(h, dx, t):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    max_c = np.max(corr)
    shifted = t * (corr - max_c)
    shifted = shifted - np.max(shifted)
    weights = np.exp(shifted)
    sw = np.sum(weights)
    if sw < 1e-300:
        weights = np.zeros_like(weights)
        weights[np.argmax(corr)] = 1.0
    else:
        weights /= sw
    one_m_h = 1.0 - h
    grad = np.zeros(n)
    for k_idx in range(len(weights)):
        w = weights[k_idx]
        if w < 1e-10:
            continue
        d = k_idx - (n - 1)
        if d == 0:
            grad += w * (1.0 - 2.0 * h)
        elif d > 0:
            grad[d:] += w * one_m_h[:n - d]
            grad[:n - d] -= w * h[d:]
        else:
            nd = -d
            grad[nd:] += w * one_m_h[:n - nd]
            grad[:n - nd] -= w * h[nd:]
    return grad * dx

def project_capped_simplex(v, target_sum):
    lo = -v.max() - 1.0
    hi = 1.0 - v.min() + 1.0

        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def binary_round(h, target_sum):
    h_bin = np.round(h).astype(float)

    if current > target_sum:
        excess = current - target_sum

def grad_descent(h, dx, target_sum, t, n_steps, max_time, t0):
    n = len(h)
    c5 = compute_c5(h, dx)
    lr = 2.0 / (t * dx)
    for step in range(n_steps):
        if time.time() - t0 >= max_time:
            break
        grad = smooth_max_grad(h, dx, t)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-10:
            break
        h_new = project_capped_simplex(h - lr * grad, target_sum)
        c5_new = compute_c5(h_new, dx)
        if c5_new < c5:
            h = h_new
            c5 = c5_new
            lr = min(lr * 1.1, 5.0)
        else:
            lr *= 0.7
    return h, c5
