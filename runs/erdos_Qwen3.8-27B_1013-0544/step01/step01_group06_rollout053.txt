import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1 - h, mode='full')

def compute_c5(h):
    return float(np.max(compute_corr(h))) * (2.0 / len(h))

def grad_at_lag(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        g[:n-k] += (1.0 - h[k:])
        if k > 0:
            g[k:] -= h[:n-k]
    else:
        p = -k
        g[p:] += (1.0 - h[:n-p])
        g[:n-p] -= h[p:]
    return g

def project(h, target_sum):
    """Project h onto {x: 0<=x_i<=1, sum(x)=target_sum}."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    thr = (lo + hi) / 2.0
    result = np.clip(h - thr, 0.0, 1.0)
    # Ensure exact sum (handle floating point)
    correction = target_sum - result.sum()
    if abs(correction) > 1e-12:
        # Distribute among non-boundary elements
        mask = (result > 1e-10) & (result < 1.0 - 1e-10)
        if mask.sum() > 0:
            result[mask] += correction / mask.sum()
            result = np.clip(result, 0.0, 1.0)
    return result

def smooth_max_curr(corr, alpha):
    """Smooth max of correlation vector."""
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha

def smooth_max_grad(h, alpha, top_k=5):
    """Gradient of smooth max correlation w.r.t. h."""
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    w_sum = w.sum()
    w = w / w_sum
    
    # Focus on top lags for efficiency
    top_idx = np.argsort(w)[::-1][:top_k]
    w_top = w[top_idx]
    w_top_sum = w_top.sum()
    if w_top_sum > 0:
        w_top = w_top / w_top_sum
    
    grad = np.zeros(n)
    for idx, weight in zip(top_idx, w_top):
        k = idx - (n - 1)
        grad += weight * grad_at_lag(h, k)
    
    return grad

def nesterov_optimize(h0, target_sum, deadline, rng, alpha=100):
    """Nesterov accelerated gradient for smooth max problem."""
    n = len(h0)
    h = h0.astype(float).copy()
    y = h.copy()  # Nesterov momentum variable
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    beta = 0.9  # momentum
    step = 0.01
    
    while time.time() < deadline:
        # Gradient at y
        grad = smooth_max_grad(y, alpha)
        
        # Step
        h_new = project(y - step * grad, target_sum)
        
        # Nesterov update
        t = 1.0 / (1.0 + beta)
        y = project(t * h_new + (1.0 - t) * (h_new + (h_new - h)), target_sum)
        # Actually standard Nesterov: y_new = h_new + (t_new - 1) * (h_new - h_old)
        
        # Simpler: just use momentum
        y = h_new + beta * (h_new - h)
        
        new_c5 = compute_c5(h_new)
        if new_c5 < best_c5 - 1e-14:
            best_h = h_new.copy()
            best_c5 = new_c5
        
        h = h_new
        # Decay step size
        step *= 0.999
        if step < 0.001:
            break
    
    return best_h, best_c5

def binary_local_search(h_bin, deadline, rng, max_stall=3000):
    """Local search on binary h."""
    n = len(h_bin)
    dx = 2.0 / n
    h = h_bin.astype(float).copy()
    best_c5 = compute_c5(h)
    
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    
    stall = 0
    while time.time() < deadline and stall < max_stall:
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h[i], h[j] = h[j], h[i]
        new_c5 = compute_c5(h)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            stall = 0
        else:
            h[i], h[j] = h[j], h[i]
            stall += 1
    
    return h, best_c5

def to_binary(h, n):
    """Convert continuous h to binary with exactly n/2 ones."""
    h_bin = (h > 0.5).astype(float)
    target = n // 2
    current = int(h_bin.sum())
    if current > target:
        excess = current - target
        one_idx = np.where(h_bin == 1)[0]
        # Remove ones with lowest h value
        order = np.argsort(h[one_idx])
        h_bin[one_idx[order[:excess]]] = 0
    elif current < target:
        deficit = target - current
        zero_idx = np.where(h_bin == 0)[0]
        # Add zeros with highest h value
        order = np.argsort(-h[zero_idx])
        h_bin[zero_idx[order[:deficit]]] = 1
    return h_bin

import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1 - h, mode='full')

def compute_c5(h):
    return float(np.max(compute_corr(h))) * (2.0 / len(h))

def grad_at_lag(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        g[:n-k] += (1.0 - h[k:])
        if k > 0:
            g[k:] -= h[:n-k]
    else:
        p = -k
        g[p:] += (1.0 - h[:n-p])
        g[:n-p] -= h[p:]
    return g

def project(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    thr = (lo + hi) / 2.0
    return np.clip(h - thr, 0.0, 1.0)

def smooth_max_weighted_grad(h, alpha, top_k=5):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    w_sum = w.sum()
    top_idx = np.argsort(w)[::-1][:top_k]
    w_top = w[top_idx]
    w_tot = w_top.sum()
    if w_tot < 1e-100:
        w_top = np.ones(top_k) / top_k
    else:
        w_top /= w_tot
    grad = np.zeros(n)
    for idx, wt in zip(top_idx, w_top):
        k = idx - (n - 1)
        grad += wt * grad_at_lag(h, k)
    return grad

def gradient_optimize(h0, target_sum, deadline, rng, alpha_init=40):
    n = len(h0)

        gn = np.linalg.norm(grad)

            grad /= gn

        new_c5 = compute_c5(h_new)

    return best_h, best_c5

def to_binary(h, n):
    h_bin = (h > 0.5).astype(float)
    target = n // 2
    cur = int(h_bin.sum())
    if cur > target:
        idx = np.where(h_bin == 1)[0]
        order = np.argsort(h[idx])
        h_bin[idx[order[:cur - target]]] = 0
    elif cur < target:
        idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h[idx])
        h_bin[idx[order[:target - cur]]] = 1
    return h_bin
