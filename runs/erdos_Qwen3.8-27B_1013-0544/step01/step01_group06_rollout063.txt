import numpy as np
import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def compute_corr(h):
    return np.correlate(h, 1 - h, mode='full')

def project_onto_simplex_box(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def grad_corr_single(h, k_idx, n):
    """Gradient of correlation at index k_idx with respect to h."""
    k = k_idx - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        # corr[k_idx] = sum_{j=0}^{n-1-k} h[j+k] * (1-h[j])
        # d/dh[j] for j in [0, n-1-k]: -(h[j+k]) (from 1-h[j] term)
        # d/dh[j] for j in [k, n-1]: (1-h[j-k]) (from h[j] term)
        g[k:] += (1.0 - h[:n-k])
        if k > 0:
            g[:n-k] -= h[k:]
    else:
        p = -k
        # corr[k_idx] = sum_{j=p}^{n-1} h[j] * (1-h[j-p])
        # d/dh[j] for j in [p, n-1]: (1-h[j-p])
        # d/dh[j] for j in [0, n-1-p]: -h[j+p]
        g[p:] += (1.0 - h[:n-p])
        if p > 0:
            g[:n-p] -= h[p:]
    return g

def compute_smooth_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= s
    grad = np.zeros(n)
    active = np.where(w > 1e-8)[0]
    for idx in active:
        grad += w[idx] * grad_corr_single(h, idx, n)
    return grad * (2.0 / n)

def optimize_adam(h0, target_sum, deadline, rng, alpha_init=50.0):
    """Adam-based optimization with progressive alpha."""
    n = len(h0)
    h = h0.astype(float).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Adam state
    m_state = np.zeros(n)
    v_state = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    lr = 0.01
    
    alpha = alpha_init
    no_improve = 0
    best_alpha = alpha
    
    while time.time() < deadline:
        t += 1
        grad = compute_smooth_grad(h, alpha)
        
        # Adam update
        m_state = beta1 * m_state + (1 - beta1) * grad
        v_state = beta2 * v_state + (1 - beta2) * grad**2
        m_hat = m_state / (1 - beta1**t)
        v_hat = v_state / (1 - beta2**t)
        
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_onto_simplex_box(h_new, target_sum)
        
        new_c5 = compute_c5(h_new)
        
        if new_c5 < best_c5 - 1e-15:
            best_h = h_new.copy()
            best_c5 = new_c5
            no_improve = 0
        else:
            no_improve += 1
        
        h = h_new
        
        # Adaptive alpha: increase when stuck
        if no_improve % 50 == 0 and no_improve > 20:

                alpha = alpha_init

            lr = max(lr * 0.9, 0.001)
        
        # Reset lr if we made progress
        if no_improve == 0 and lr < 0.02:
            lr = min(lr * 1.05, 0.02)
    
    return best_h, best_c5

def snap_to_binary(h, target_sum):
    """Snap h values to 0 or 1, maintaining sum constraint."""
    n = len(h)
    n_ones = int(target_sum)
    # Sort by value, take top n_ones as 1s
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:n_ones]] = 1.0
    return h_bin

def binary_local_search(h_bin, deadline, rng, max_iters=5000):

        new_c5 = compute_c5(h)

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_onto_simplex_box(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def grad_corr_single(h, k_idx, n):
    k = k_idx - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n-k])
        if k > 0:
            g[:n-k] -= h[k:]
    else:
        p = -k
        g[p:] += (1.0 - h[:n-p])
        if p > 0:
            g[:n-p] -= h[p:]
    return g

def compute_smooth_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= s
    grad = np.zeros(n)
    active = np.where(w > 1e-8)[0]
    for idx in active:
        grad += w[idx] * grad_corr_single(h, idx, n)
    return grad * (2.0 / n)

def snap_to_binary(h, target_sum):
    n = len(h)
    n_ones = int(round(target_sum))
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:n_ones]] = 1.0
    return h_bin

def binary_local_search(h_bin, deadline, rng, max_iters=1000000):
    """Simulated annealing local search for binary sequences."""
    n = len(h_bin)
    h = h_bin.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    temp = 0.05
    n_ones = int(h.sum())
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    it = 0
    stall = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # Choose a random swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h[i] = 0
        h[j] = 1
        new_c5 = compute_c5(h)
        
        delta = new_c5 - best_c5
        if delta < 0 or rng.random() < np.exp(-delta / temp):
            # Accept
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            best_h = h.copy()
            best_c5 = new_c5
            stall = 0
        else:
            # Reject
            h[i] = 1
            h[j] = 0
        
        # Cooldown
        if it % 100 == 0:
            temp = max(temp * 0.995, 0.001)
        
        # Restart if stuck
        if it % 10000 == 0:
            # Perturb
            for _ in range(5):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h[i], h[j] = 0, 1
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
            temp = 0.05
    
    return best_h, best_c5

def optimize_adam(h0, target_sum, deadline, rng, alpha_init=50.0):
    n = len(h0)
    h = h0.astype(float).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    m_state = np.zeros(n)
    v_state = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    lr = 0.01
    alpha = alpha_init
    no_improve = 0
    
    while time.time() < deadline:
        t += 1
        grad = compute_smooth_grad(h, alpha)
        
        m_state = beta1 * m_state + (1 - beta1) * grad
        v_state = beta2 * v_state + (1 - beta2) * grad**2
        m_hat = m_state / (1 - beta1**t)
        v_hat = v_state / (1 - beta2**t)
        
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_onto_simplex_box(h_new, target_sum)
        h = h_new
        
        new_c5 = compute_c5(h)
        
        if new_c5 < best_c5 - 1e-15:
            best_h = h.copy()
            best_c5 = new_c5
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 50 and no_improve % 20 == 0:
            alpha = min(alpha * 1.3, 500.0)
        if no_improve > 300:
            h = best_h + rng.randn(n) * 0.03
            h = project_onto_simplex_box(h, target_sum)
            no_improve = 0
            alpha = alpha_init
            m_state = np.zeros(n)
            v_state = np.zeros(n)
            t = 0
        
        if no_improve == 0:
            lr = min(lr * 1.01, 0.02)
        elif no_improve > 100:
            lr = max(lr * 0.95, 0.002)
    
    return best_h, best_c5

    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False) + 1.0/n_new

    x_ext = np.append(x_old, 2.0 + 1.0/n_old)
    return np.interp(x_new, x_ext, h_ext)

import numpy as np

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def compute_corr_full(h):
    return np.correlate(h, 1 - h, mode='full')

def project_onto_simplex_box(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def grad_corr_single(h, k_idx, n):
    k = k_idx - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n-k])
        if k > 0:
            g[:n-k] -= h[k:]
    else:
        p = -k
        g[p:] += (1.0 - h[:n-p])
        if p > 0:
            g[:n-p] -= h[p:]
    return g

def compute_smooth_grad(h, alpha):
    n = len(h)
    corr = compute_corr_full(h)
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= s
    grad = np.zeros(n)
    active = np.where(w > 1e-8)[0]
    for idx in active:
        grad += w[idx] * grad_corr_single(h, idx, n)
    return grad * (2.0 / n)
