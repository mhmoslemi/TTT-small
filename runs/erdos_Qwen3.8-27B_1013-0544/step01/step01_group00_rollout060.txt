    weights: array same shape as np.correlate(h, 1-h, 'full')
    """
    n = len(h)

    # In numpy's full output, index 0 corresponds to k=-(n-1), index n-1 to k=0, index 2n-2 to k=n-1

    # d(corr[k])/dh[j] = (1-h[j-k]) * I[0<=j-k<n] - h[j+k] * I[0<=j+k<n]

    # corr_full[idx] corresponds to lag k = idx - (n-1)

        w = weights[idx]
        if abs(w) < 1e-15:

        # Term 1: (1-h[j-k]) for j in [max(0,k), min(n, n+k)) -- wait

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def compute_corr(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return corr

def softmax(corr, tau):
    m = np.max(corr)
    exps = np.exp(np.clip((corr - m) / tau, -500, 500))
    s = np.sum(exps)
    return exps / s, m + tau * np.log(s)

def corr_gradient(h, weights):
    """
    Gradient of sum_k weights[k] * corr_k w.r.t. h.
    weights has length 2N-1, corresponding to lags k = -(N-1), ..., 0, ..., N-1.
    Note: this is the gradient of the UN-scaled correlation (multiply by dx for the scaled version).
    Actually let's include dx here.
    """
    N = len(h)
    dx = 2.0 / N
    grad = np.zeros(N)
    
    for idx in range(2 * N - 1):
        k = idx - (N - 1)
        w = weights[idx] * dx
        if abs(w) < 1e-15:
            continue
        if k >= 0:
            # d(corr_k)/dh[j] = (1-h[j-k]) for j in [k, N-1] - h[j+k] for j in [0, N-1-k]
            grad[k:] += w * (1.0 - h[:N - k])
            grad[:N - k] -= w * h[k:]
        else:
            k_abs = -k
            # d(corr_k)/dh[j] = (1-h[j+k_abs]) for j in [0, N-1-k_abs] - h[j-k_abs] for j in [k_abs, N-1]
            grad[:N - k_abs] += w * (1.0 - h[k_abs:])
            grad[k_abs:] -= w * h[:N - k_abs]
    
    return grad

def project_feasible(h, target):
    """Project h onto {0 <= h_i <= 1, sum(h_i) = target}."""
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def gradient_descent_step(h, tau, lr):
    """One step of gradient descent on smooth max of correlations."""
    corr = compute_corr(h)
    weights, smooth_val = softmax(corr, tau)
    grad = corr_gradient(h, weights)
    # Gradient step
    h_new = h - lr * grad
    # Project to feasible set
    h_new = project_feasible(h_new, len(h) / 2.0)
    return h_new, smooth_val

def momentum_gradient_descent(h0, tau_schedule, lr_schedule, n_iters, target):
    """Momentum-based gradient descent with adaptive parameters."""
    N = len(h0)
    h = h0.copy()
    v = np.zeros(N)  # momentum
    beta = 0.9
    
    history = []
    for i in range(n_iters):
        tau = tau_schedule(i)
        lr = lr_schedule(i)
        
        corr = compute_corr(h)
        weights, smooth_val = softmax(corr, tau)
        grad = corr_gradient(h, weights)
        
        v = beta * v + (1 - beta) * grad
        h = h - lr * v
        h = project_feasible(h, target)
        
        c5 = compute_c5(h)
        history.append(c5)
    
    return h, history

            grad[k:] += w * (1.0 - h[:N - k])
            grad[:N - k] -= w * h[k:]
        else:
            ka = -k
            grad[:N - ka] += w * (1.0 - h[ka:])
            grad[ka:] -= w * h[:N - ka]
    return grad

def project_feasible(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h + mid, 0.0, 1.0)) > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def interp_up(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 1, n_old, endpoint=False)
    x_new = np.linspace(0, 1, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new / 2.0)

def gd_optimize(h0, n_iters, tau0, tau1, lr0, lr1, check_interval=10):
    N = len(h0)
    target = N / 2.0
    h = h0.copy()
    v = np.zeros(N)
    beta = 0.9
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    
    for i in range(n_iters):
        t_frac = i / max(n_iters - 1, 1)
        tau = tau0 * (tau1 / tau0) ** t_frac
        lr = lr0 * (lr1 / lr0) ** t_frac
        
        corr = compute_corr(h)
        w = softmax_weights(corr, tau)
        grad = corr_grad(h, w)
        
        v = beta * v + (1.0 - beta) * grad
        h = h - lr * v
        h = project_feasible(h, target)
        
        if i % check_interval == 0 or i == n_iters - 1:
            c5_new = compute_c5(h)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
    
    return best_h, best_c5

def t_obj(v):
    return v[-1]

def t_con_ineq(v):
    n = len(v) - 1
    h = v[:n]
    t = v[-1]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return t - corr

def t_con_eq(v):
    n = len(v) - 1
    return np.sum(v[:n]) - n * 0.5

def t_con_eq_jac(v):
    n = len(v) - 1
    j = np.zeros(len(v))
    j[:n] = 1.0
    return j

def slsqp_polish(h0, max_iter=200):
    n = len(h0)
    c5 = compute_c5(h0)
    v0 = np.append(h0.copy(), c5 + 0.01)
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.5)]
    cons = [
        {'type': 'ineq', 'fun': t_con_ineq},
        {'type': 'eq', 'fun': t_con_eq, 'jac': t_con_eq_jac}

                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-12})
    h = np.clip(res.x[:n], 0.0, 1.0)
    h = project_feasible(h, n / 2.0)
    return h, compute_c5(h)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def compute_corr(h):
    N = len(h)
    dx = 2.0 / N
    return np.correlate(h, 1.0 - h, mode='full') * dx

def softmax_weights(corr, tau):
    m = np.max(corr)
    exps = np.exp(np.clip((corr - m) / tau, -500, 0))
    s = np.sum(exps)
    return exps / s

def corr_grad(h, weights):
    N = len(h)
    dx = 2.0 / N
    grad = np.zeros(N)
    for idx in range(2 * N - 1):
        k = idx - (N - 1)
        w = weights[idx] * dx
        if w < 1e-14:
            continue
        if k >= 0:
            grad[k:] += w * (1.0 - h[:N - k])
            grad[:N - k] -= w * h[k:]
        else:
            ka = -k
            grad[:N - ka] += w * (1.0 - h[ka:])
            grad[ka:] -= w * h[:N - ka]
    return grad

def project_feasible(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h + mid, 0.0, 1.0)) > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def interp_up(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 1, n_old, endpoint=False)
    x_new = np.linspace(0, 1, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new / 2.0)

def gd_optimize(h0, n_iters, tau0, tau1, lr0, lr1, beta=0.9):
    N = len(h0)
    target = N / 2.0
    h = h0.copy()
    v = np.zeros(N)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for i in range(n_iters):
        t_frac = i / max(n_iters - 1, 1)
        tau = tau0 * (tau1 / tau0) ** t_frac
        lr = lr0 * (lr1 / lr0) ** t_frac
        corr = compute_corr(h)
        w = softmax_weights(corr, tau)
        grad = corr_grad(h, w)
        v = beta * v + (1.0 - beta) * grad
        h = h - lr * v
        h = project_feasible(h, target)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def t_obj(v):
    return v[-1]

def t_con_ineq(v):
    n = len(v) - 1
    h = v[:n]
    t = v[-1]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return t - corr

def t_con_eq(v):
    n = len(v) - 1
    return np.sum(v[:n]) - n * 0.5

def t_con_eq_jac(v):
    n = len(v) - 1
    j = np.zeros(len(v))
    j[:n] = 1.0
    return j

def slsqp_polish(h0, max_iter=200):
    n = len(h0)
    c5 = compute_c5(h0)
    v0 = np.append(h0.copy(), c5 + 0.01)
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.5)]
    cons = [
        {'type': 'ineq', 'fun': t_con_ineq},
        {'type': 'eq', 'fun': t_con_eq, 'jac': t_con_eq_jac}
    ]
    res = minimize(t_obj, v0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-12})
    h = np.clip(res.x[:n], 0.0, 1.0)
    h = project_feasible(h, n / 2.0)
    return h, compute_c5(h)
