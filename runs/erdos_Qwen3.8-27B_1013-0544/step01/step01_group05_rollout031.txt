            h_new = project_feasible(h + v, target_sum)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            elif it % 200 == 0 and it > 0:
                lr *= 0.92
                v *= 0.5
            h = h_new
        h = best_h.copy()

    # Phase 2: Hard active set fine-tuning with tight tolerance
    h = best_h.copy()
    v = np.zeros(n)
    lr = 0.0003
    for it in range(40000):
        if tleft() < 80.0:
            break
        grad, c_max = hard_grad(h, n, n_half, k=6, toler=max(0.002, c_max * 0.003))
        v = 0.5 * v - lr * grad
        h_new = project_feasible(h + v, target_sum)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        else:
            lr *= 0.94
            v *= 0.5
            if lr < 1e-14:
                break
        h = h_new

    # Phase 3: SLSQP polish
    if tleft() > 60.0:
        def obj_func(x):
            return compute_c5(x, dx)
        def eq_func(x):
            return float(np.sum(x)) - target_sum

        try:
            res = minimize(obj_func, best_h, method='SLSQP',
                           bounds=[(0.0, 1.0)] * n,
                           constraints=[{'type': 'eq', 'fun': eq_func}],
                           options={'maxiter': 5000, 'ftol': 1e-20})
            h_ref = project_feasible(res.x, target_sum)
            c5_ref = compute_c5(h_ref, dx)
            if c5_ref < best_c5:
                best_c5 = c5_ref
                best_h = h_ref
        except Exception:
            pass

    # Phase 4: SLSQP from perturbed points
    if tleft() > 30.0:
        def obj_func2(x):
            return compute_c5(x, dx)
        def eq_func2(x):
            return float(np.sum(x)) - target_sum

        for attempt in range(25):
            if tleft() < 5.0:
                break
            pert = 0.004 * (0.85 ** attempt)
            h_pert = best_h + rng.normal(0, pert, n)
            h_pert = project_feasible(h_pert, target_sum)
            c5_pert = compute_c5(h_pert, dx)
            # Only polish if perturbation didn't worsen too much
            if c5_pert < best_c5 + 0.002:
                try:
                    res = minimize(obj_func2, h_pert, method='SLSQP',
                                   bounds=[(0.0, 1.0)] * n,
                                   constraints=[{'type': 'eq', 'fun': eq_func2}],
                                   options={'maxiter': 3000, 'ftol': 1e-20})
                    h_ref = project_feasible(res.x, target_sum)
                    c5_ref = compute_c5(h_ref, dx)
                    if c5_ref < best_c5:
                        best_c5 = c5_ref
                        best_h = h_ref
                except Exception:
                    pass

    # Phase 5: Final ultra-fine gradient step
    if tleft() > 10.0:
        h = best_h.copy()
        v = np.zeros(n)
        lr = 1e-5
        for it in range(20000):
            if tleft() < 2.0:
                break
            grad, c_max = hard_grad(h, n, n_half, k=4, toler=1e-4)
            v = 0.3 * v - lr * grad
            h_new = project_feasible(h + v, target_sum)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            else:
                lr *= 0.9
                v *= 0.3
                if lr < 1e-16:
                    break
            h = h_new

    return (best_h, best_c5, n)
