import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(x, target_sum):
    lo = float(x.min()) - 1.0
    hi = float(x.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = float(np.clip(x - mid, 0.0, 1.0).sum())
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def softmax_obj_and_grad(h, dx, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_shift = np.max(corr)
    vals = T * (corr - c_shift)
    exp_vals = np.exp(vals)
    Z = np.sum(exp_vals)
    obj = (np.log(Z) + T * c_shift) / T * dx
    # Gradient
    n = len(h)
    n_half = n - 1
    weights = exp_vals / Z
    grad = np.zeros(n)
    for idx in range(len(corr)):
        w = weights[idx]
        L = idx - n_half
        if w < 1e-10:
            continue
        if L == 0:
            grad += w * (1.0 - 2.0 * h)
        elif L > 0:
            # d c_L / d h(j) = (1-h(j+L)) for j < n-L, minus h(j-L) for j >= L
            g = np.zeros(n)
            g[:n - L] += (1.0 - h[L:])
            g[L:] -= h[:n - L]
            grad += w * g
        else:
            L2 = -L
            g = np.zeros(n)
            g[L2:] += (1.0 - h[:n - L2])
            g[:n - L2] -= h[L2:]
            grad += w * g
    return obj * 1.0, grad * dx

def active_set_grad(h, corr):
    n = len(h)
    n_half = n - 1
    c_max = np.max(corr)
    tol = max(0.01, c_max * 0.001)
    active = np.where(corr >= c_max - tol)[0]
    if len(active) < 3:
        active = np.argsort(corr)[-3:]
    elif len(active) > 20:
        active = np.argsort(corr)[-20:]
    grad = np.zeros(n)
    for idx in active:
        L = int(idx) - n_half
        if L == 0:
            grad += (1.0 - 2.0 * h)
        elif L > 0:
            g = np.zeros(n)
            g[:n - L] += (1.0 - h[L:])
            g[L:] -= h[:n - L]
            grad += g
        else:
            L2 = -L
            g = np.zeros(n)
            g[L2:] += (1.0 - h[:n - L2])
            g[:n - L2] -= h[L2:]
            grad += g
    return grad / len(active)
