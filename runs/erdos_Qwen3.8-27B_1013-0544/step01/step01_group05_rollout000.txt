                h_ref = project_feasible(res.x, target_sum)
                c5_ref = compute_c5(h_ref, dx)
                if c5_ref < best_c5:
                    best_c5 = c5_ref
                    best_h = h_ref
        except Exception:
            pass

    # Phase 5: SLSQP from perturbed best
    if tleft() > 40.0:
        for attempt in range(3):
            if tleft() < 15.0:
                break
            h_start = best_h + rng.normal(0, 0.002, n_points)
            h_start = project_feasible(h_start, target_sum)
            def obj_func(x):
                return compute_c5(x, dx)
            def eq_func(x):
                return float(np.sum(x) - target_sum)
            try:
                res = minimize(obj_func, h_start, method='SLSQP',
                              bounds=[(0.0, 1.0)] * n_points,
                              constraints=[{'type': 'eq', 'fun': eq_func}],
                              options={'maxiter': 3000, 'ftol': 1e-20})
                if res.x is not None:
                    h_ref = project_feasible(res.x, target_sum)
                    c5_ref = compute_c5(h_ref, dx)
                    if c5_ref < best_c5:
                        best_c5 = c5_ref
                        best_h = h_ref
            except Exception:
                pass

    # Phase 6: Ultra-fine refinement
    if tleft() > 10.0:
        h = best_h.copy()
        step_size = 1e-6
        for _ in range(10000):
            if tleft() < 2.0:
                break
            corr = np.correlate(h, 1.0 - h, mode='full')
            c_max = float(np.max(corr))
            act = np.where(corr >= c_max - 1e-7)[0]
            if len(act) == 0:
                act = np.array([np.argmax(corr)])
            if len(act) > 3:
                act = np.argsort(corr)[-3:]
            grad = active_gradient(h, act, n_half)
            h_new = project_feasible(h - step_size * grad, target_sum)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
            else:
                step_size *= 0.9
                if step_size < 1e-14:
                    break

    return (best_h, best_c5, n_points)
