    For shift k >= 0: f_k = sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])

The cutting-plane method would be:
- Maintain a set of cuts: f_k_j(h) ≤ C for j = 1, ..., m
- At each step, solve: min C s.t. all cuts, 0≤h≤1, sum h = n/2
- Evaluate the actual f_k at the solution, add new cuts for the worst shifts
- Repeat

from scipy.optimize import linprog

def solve_lip(h, C, cuts, target_sum):
    """Solve LP: min C s.t. cuts(h, C) and box/sum constraints"""
    n = len(h)
    # Variables: h[0], ..., h[n-1], C
    # Objective: minimize C -> c = [0,...,0,1]
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Inequality constraints: A @ [h; C] <= b
    # From cuts: for each cut j, grad_j · h - C <= f_k_j(h0_j) - grad_j · h0_j
    # From box: h[i] <= 1 -> h[i] <= 1
    # From box: -h[i] <= 0
    # From sum: sum h = target (equality)
    
    # Collect all inequality constraints
    A_rows = []
    b_vals = []
    
    # Cuts
    for (grad_j, h0_j, f_val) in cuts:
        row = np.zeros(n + 1)
        row[:n] = grad_j
        row[n] = -1.0  # -C
        A_rows.append(row)
        b_vals.append(f_val - np.dot(grad_j, h0_j))
    
    # h[i] <= 1
    for i in range(n):
        row = np.zeros(n + 1)
        row[i] = 1.0
        A_rows.append(row)
        b_vals.append(1.0)
    
    # -h[i] <= 0
    for i in range(n):
        row = np.zeros(n + 1)
        row[i] = -1.0
        A_rows.append(row)
        b_vals.append(0.0)
    
    A = np.array(A_rows)
    b = np.array(b_vals)
    
    # Equality: sum h = target
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target_sum])
    
    bounds = [(0, 1)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    return res

        d3 = time.time() + min(remaining * 0.4, 200.0)
        h2, c5_2 = optimize(best_h.copy(), len(best_h) / 2.0, d3, rng,
                             lr0=0.01, K0=3, temp0=10.0)
        if c5_2 < best_c5:
            best_h, best_c5 = h2, c5_2

    # Phase 4: Final fine polish
    remaining = budget_s - (time.time() - t0)
    if remaining > 15 and best_h is not None:
        d4 = t0 + budget_s - 3
        h3, c5_3 = optimize(best_h.copy(), len(best_h) / 2.0, d4, rng,
                             lr0=0.005, K0=2, temp0=15.0)
        if c5_3 < best_c5:
            best_h, best_c5 = h3, c5_3

    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
