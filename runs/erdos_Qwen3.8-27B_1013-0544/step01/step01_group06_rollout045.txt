def project_to_constraints(h, target_sum):
    """Project onto {0 <= h_i <= 1, sum(h) = target_sum}"""

    n = len(h)

import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1 - h, mode='full')

def compute_c5(h):
    n = len(h)
    return np.max(compute_corr(h)) * (2.0 / n)

def project_box_sum(h, target_sum):
    lo, hi = -1.5, 1.5
    for _ in range(45):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_single_lag(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n - k])
        g[:n - k] -= h[k:]
    else:
        p = -k
        g[p:] += (1.0 - h[:n - p])
        g[:n - p] -= h[p:]
    return g

def compute_topk_grad(h, topk, alpha):
    n = len(h)
    corr = compute_corr(h)
    if topk >= len(corr):
        topk_idx = np.arange(len(corr))
    else:
        topk_idx = np.argpartition(corr, -topk)[-topk:]
    topk_vals = corr[topk_idx]
    m = topk_vals.max()
    w = np.exp(alpha * (topk_vals - m))
    w /= w.sum()
    grad = np.zeros(n)
    for idx, wi in zip(topk_idx, w):
        k = idx - (n - 1)
        grad += wi * grad_single_lag(h, k)
    grad *= (2.0 / n)
    return grad, topk_idx

def threshold_step(h, target_sum, rng, strength=0.3):
