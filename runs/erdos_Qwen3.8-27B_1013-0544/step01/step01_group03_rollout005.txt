import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val(obj, tau):
    x = tau * obj
    x_max = np.max(x)
    return x_max / tau + np.log(np.sum(np.exp(x - x_max))) / tau

def fix_sum(h, target):
    h = np.array(h, dtype=float)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def _lbfgs_obj(h, tau, target, penalty):
    corr = np.correlate(h, 1.0 - h, mode="full")
    x = tau * corr
    x_max = np.max(x)
    val = x_max / tau + np.log(np.sum(np.exp(x - x_max))) / tau
    val += penalty * (h.sum() - target) ** 2
    return val

def _fourier_obj(params, n, tau, n_modes):
    hft = np.zeros(n // 2 + 1, dtype=complex)
    hft[0] = n / 2.0
    for k in range(n_modes):
        idx = k + 1
        if idx <= n // 2:
            hft[idx] = params[2 * k] + 1j * params[2 * k + 1]
    h = np.fft.irfft(hft, n=n)
    viol = np.maximum(0.0, h - 1.0) + np.maximum(0.0, -h)
    h_c = np.clip(h, 0.0, 1.0)
    h_c = fix_sum(h_c, n / 2.0)
    corr = np.correlate(h_c, 1.0 - h_c, mode="full")
    x = tau * corr
    x_max = np.max(x)
    return x_max / tau + np.log(np.sum(np.exp(x - x_max))) / tau + 30.0 * np.sum(viol ** 2)

def _fourier_reconstruct(params, n, n_modes):
    hft = np.zeros(n // 2 + 1, dtype=complex)
    hft[0] = n / 2.0
    for k in range(n_modes):
        idx = k + 1
        if idx <= n // 2:
            hft[idx] = params[2 * k] + 1j * params[2 * k + 1]
    h = np.fft.irfft(hft, n=n)
    return np.clip(h, 0.0, 1.0)

def _de_obj(h, target):
    h = np.clip(h, 0.0, 1.0)
    h = fix_sum(h, target)
    return evaluate_c5(h)

def _subgradient_step(h, target, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_star = int(np.argmax(corr))
    k_mod = k_star % n
    grad = np.zeros(n)
    for j in range(n):
        grad[j] = (1.0 - h[(j + k_mod) % n]) - h[(j - k_mod) % n]
    h_new = h - alpha * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = fix_sum(h_new, target)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(70, 0.5)

    best_c5 = evaluate_c5(best_h)
    n = len(best_h)
    target = n / 2.0
    bounds_list = [(0.0, 1.0)] * n

    # Phase 1: L-BFGS-B annealing from initial point
    for tau in [5, 15, 50, 150, 400, 1200, 3000]:
        if budget_s - (time.time() - t0) < 10:
            break
        res = minimize(_lbfgs_obj, best_h, args=(tau, target, 100.0),
                      method='L-BFGS-B', bounds=bounds_list,
                      options={'maxiter': 400, 'gtol': 1e-9})
        h_new = fix_sum(res.x, target)
        c5 = evaluate_c5(h_new)
        if c5 < best_c5:
            best_h = h_new
            best_c5 = c5

    # Phase 2: Subgradient refinement (non-smooth max)
    if budget_s - (time.time() - t0) > 30:
        n_sg = len(best_h)
        t_sg = n_sg / 2.0
        h_sg = best_h.copy()
        c5_sg = best_c5
        for it in range(5000):
            if budget_s - (time.time() - t0) < 5:
                break
            alpha = 0.005 / (1.0 + 0.01 * it)
            h_sg = _subgradient_step(h_sg, t_sg, alpha)
            if it % 200 == 0:
                c5_sg = evaluate_c5(h_sg)
                if c5_sg < c5 - 0.001:
                    pass  # keep going
        c5_sg = evaluate_c5(h_sg)
        if c5_sg < best_c5:
            best_h = h_sg
            best_c5 = c5_sg
        # Polish subgradient result
        for tau in [100, 400, 1500]:
            if budget_s - (time.time() - t0) < 5:
                break
            res = minimize(_lbfgs_obj, best_h, args=(tau, target, 100.0),
                          method='L-BFGS-B', bounds=bounds_list,
                          options={'maxiter': 200, 'gtol': 1e-9})
            h_new = fix_sum(res.x, target)
            c5 = evaluate_c5(h_new)
            if c5 < best_c5:
                best_h = h_new
                best_c5 = c5

    # Phase 3: Fourier-space optimization
    if budget_s - (time.time() - t0) > 30:
        n_ft = len(best_h)
        t_ft = n_ft / 2.0
        for n_modes in [4, 6, 8, 12, 16, 24]:
            if budget_s - (time.time() - t0) < 12:
                break
            hft = np.fft.rfft(best_h)
            params0 = np.zeros(2 * n_modes)
            for k in range(n_modes):
                idx = k + 1
                if idx <= n_ft // 2:
                    params0[2 * k] = hft[idx].real
                    params0[2 * k + 1] = hft[idx].imag
            for tau in [30, 100, 400]:
                if budget_s - (time.time() - t0) < 8:
                    break
                res = minimize(_fourier_obj, params0, args=(n_ft, tau, n_modes),
                              method='L-BFGS-B', options={'maxiter': 250, 'gtol': 1e-6})
                h_new = _fourier_reconstruct(res.x, n_ft, n_modes)
                h_new = fix_sum(h_new, t_ft)
                c5 = evaluate_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new
                    best_c5 = c5
                params0 = res.x

    # Phase 4: Differential evolution at reduced resolution
    if budget_s - (time.time() - t0) > 80:
        for n_small in [30, 40, 50]:
            if budget_s - (time.time() - t0) < 40:
                break
            t_s = n_small / 2.0
            h_int = np.interp(np.linspace(0, 1, n_small),
                              np.linspace(0, 1, len(best_h)), best_h)
            h_int = fix_sum(h_int, t_s)
            try:
                de_res = differential_evolution(
                    _de_obj, bounds=[(0.0, 1.0)] * n_small,
                    args=(t_s,), seed=seed, maxiter=80,
                    popsize=12, tol=1e-5, polish=True, disp=False)
                h_de = fix_sum(de_res.x, t_s)
                h_up = np.interp(np.linspace(0, 1, len(best_h)),
                                 np.linspace(0, 1, n_small), h_de)
                h_up = fix_sum(h_up, len(best_h) / 2.0)
                c5_up = evaluate_c5(h_up)
                if c5_up < best_c5:
                    best_h = h_up
                    best_c5 = c5_up
            except Exception:
                pass

    # Phase 5: Random restarts
    if budget_s - (time.time() - t0) > 20:
        n_r = len(best_h)
        t_r = n_r / 2.0
        for i in range(50):
            if budget_s - (time.time() - t0) < 8:
                break
            scale = 0.05 + 0.25 * np.random.random()
            hp = best_h + np.random.normal(0, scale, n_r)
            hp = np.clip(hp, 0.0, 1.0)
            hp = fix_sum(hp, t_r)
            tau_r = float([30, 80, 200, 600, 2000][np.random.randint(5)])
            res = minimize(_lbfgs_obj, hp, args=(tau_r, t_r, 80.0),
                          method='L-BFGS-B', bounds=[(0.0, 1.0)] * n_r,
                          options={'maxiter': 120, 'gtol': 1e-7})
            h_new = fix_sum(res.x, t_r)
            c5 = evaluate_c5(h_new)
            if c5 < best_c5:
                best_h = h_new
                best_c5 = c5

    # Phase 6: Structured initializations (sine, cosine, Thue-Morse)
    if budget_s - (time.time() - t0) > 20:
        n_s = len(best_h)
        t_s = n_s / 2.0
        x_grid = np.linspace(0, 2, n_s)
        for trial in range(10):
            if budget_s - (time.time() - t0) < 8:
                break
            freq1 = np.random.randint(1, 5)
            freq2 = np.random.randint(1, 5)
            amp1 = 0.15 + 0.25 * np.random.random()
            amp2 = 0.1 + 0.2 * np.random.random()
            ph1 = np.random.random() * 2 * np.pi
            ph2 = np.random.random() * 2 * np.pi
            h_str = 0.5 + amp1 * np.sin(np.pi * freq1 * x_grid + ph1) \
                    + amp2 * np.cos(np.pi * freq2 * x_grid + ph2)
            h_str = np.clip(h_str, 0.0, 1.0)
            h_str = fix_sum(h_str, t_s)
            tau_st = float([50, 150, 500][np.random.randint(3)])
            res = minimize(_lbfgs_obj, h_str, args=(tau_st, t_s, 80.0),
                          method='L-BFGS-B', bounds=[(0.0, 1.0)] * n_s,
                          options={'maxiter': 200, 'gtol': 1e-8})
            h_new = fix_sum(res.x, t_s)
            c5 = evaluate_c5(h_new)
            if c5 < best_c5:
                best_h = h_new
                best_c5 = c5

    # Phase 7: Multi-resolution refinement
    if budget_s - (time.time() - t0) > 30:
        for n_try in [100, 140, 160, 200]:
            if budget_s - (time.time() - t0) < 15:
                break
            t_try = n_try / 2.0
            h_t = np.interp(np.linspace(0, 1, n_try),
                            np.linspace(0, 1, len(best_h)), best_h)
            h_t = fix_sum(h_t, t_try)
            for tau in [50, 200, 800]:
                if budget_s - (time.time() - t0) < 8:

                res = minimize(_lbfgs_obj, h_t, args=(tau, t_try, 80.0),
                              method='L-BFGS-B', bounds=[(0.0, 1.0)] * n_try,
                              options={'maxiter': 200, 'gtol': 1e-8})
                h_new = fix_sum(res.x, t_try)
                c5 = evaluate_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new
                    best_c5 = c5
                h_t = h_new

    # Phase 8: Final high-precision polish
    n_f = len(best_h)
    t_f = n_f / 2.0
    for tau in [3000, 8000, 20000]:
        if budget_s - (time.time() - t0) < 6:
            break
        res = minimize(_lbfgs_obj, best_h, args=(tau, t_f, 200.0),

        h_new = fix_sum(res.x, t_f)
        c5 = evaluate_c5(h_new)
        if c5 < best_c5:
            best_h = h_new
            best_c5 = c5

    # Final constraint enforcement
    n_final = len(best_h)
    best_h = fix_sum(best_h, n_final / 2.0)
    best_c5 = evaluate_c5(best_h)

    return best_h, best_c5, n_final
