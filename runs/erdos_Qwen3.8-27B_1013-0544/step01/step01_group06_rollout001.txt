import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1 - h, mode='full')

def compute_c5(h):
    n = len(h)
    return np.max(compute_corr(h)) * (2.0 / n)

def soft_weights(corr, alpha):
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= s
    return w

def grad_corr_at_h(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        g[:n - k] += (1.0 - h[k:])
        if k > 0:
            g[k:] -= h[:n - k]
    else:
        p = -k
        g[p:] += (1.0 - h[:n - p])
        if p > 0:
            g[:n - p] -= h[p:]
    return g

def compute_smooth_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    w = soft_weights(corr, alpha)
    grad = np.zeros(n)
    thresh = max(1e-7, 0.5 / len(corr))
    active = np.where(w > thresh)[0]
    for idx in active:
        k = idx - (n - 1)
        grad += w[idx] * grad_corr_at_h(h, k)
    return grad * (2.0 / n)

def project_box_sum(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def interpolate_res(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    return np.interp(x_new, x_old, h_old)

def binary_push(h, target_sum, rng):
    n = len(h)
    h2 = h.copy()
    mask = rng.rand(n) < 0.3
    h2[mask] = np.where(h2[mask] > 0.5, 1.0, 0.0)
    h2 = project_box_sum(h2, target_sum)
    return h2

def optimize(h0, target_sum, deadline, rng, a_start=60, a_end=500, lr0=0.012):
    n = len(h0)
    h = project_box_sum(h0.astype(float).copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    m1 = np.zeros(n)
    m2 = np.zeros(n)
    t = 0
    lr = lr0
    patience = 0
    stall = 0
    n_iter_est = max(10, int((deadline - time.time()) * 80.0))
    while time.time() < deadline:
        t += 1
        frac = min(1.0, t / (n_iter_est * 0.7))
        alpha = a_start + (a_end - a_start) * (frac ** 0.6)
        grad = compute_smooth_grad(h, alpha)
        m1 = 0.85 * m1 + 0.15 * grad
        m2 = 0.999 * m2 + 0.001 * grad * grad
        bc1 = 1 - 0.85 ** t
        bc2 = 1 - 0.999 ** t
        step = m1 / bc1 / (np.sqrt(m2 / bc2) + 1e-8)
        h_new = h - lr * step
        h_new = project_box_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            h = h_new
            best_h = h.copy()
            best_c5 = c5_new
            patience = 0
            if lr < 0.04:
                lr = min(lr * 1.01, 0.04)
        else:
            patience += 1
            if patience > 50:
                # Try multiple perturbation strategies
                best_pert = h.copy()
                best_pert_c5 = best_c5
                # Strategy 1: Gaussian noise
                h_p = project_box_sum(h + rng.randn(n) * 0.04, target_sum)
                if compute_c5(h_p) < best_pert_c5:
                    best_pert = h_p
                    best_pert_c5 = compute_c5(h_p)
                # Strategy 2: Binary push
                h_b = binary_push(h, target_sum, rng)
                cb = compute_c5(h_b)
                if cb < best_pert_c5:
                    best_pert = h_b
                    best_pert_c5 = cb
                # Strategy 3: Sinusoidal perturbation

                h_s = project_box_sum(h + 0.05 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase), target_sum)
                cs = compute_c5(h_s)

                    best_pert_c5 = cs
                if best_pert_c5 < best_c5:
                    h = best_pert
                    best_h = h.copy()
                    best_c5 = best_pert_c5
                    m1 = np.zeros(n)
                    m2 = np.zeros(n)
                    t = 0
                    lr = lr0
                patience = 0
                stall += 1
                if stall > 15:
                    break
    return best_h, best_c5
