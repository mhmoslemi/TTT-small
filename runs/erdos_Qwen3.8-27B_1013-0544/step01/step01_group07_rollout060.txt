def compute_gradient(h, t, dx, n):
    # corr has length 2n-1
    corr = np.correlate(h, 1-h, mode="full")
    # weights for smooth max
    max_corr = np.max(corr)
    exp_vals = np.exp(t * (corr - max_corr))
    weights = exp_vals / np.sum(exp_vals)
    
    # Gradient of each corr[k] with respect to h
    # corr[n-1+k] = sum_i h[i] * (1-h[i+k]) for valid i
    grad = np.zeros(n)
    
    for k in range(-(n-1), n):
        if weights[n-1+k] < 1e-15:
            continue
        # d/dh[j] of corr[n-1+k]
        # = (1-h[j+k]) for j where j+k is valid (j in "first" part)
        # - h[j-k] for j where j-k is valid (j in "second" part)
        
        if k >= 0:
            # First part: j from 0 to n-1-k, contribution (1-h[j+k])
            first_part = np.zeros(n)
            first_part[:n-k] = 1 - h[k:]
            # Second part: j from k to n-1, contribution -h[j-k]
            second_part = np.zeros(n)
            second_part[k:] = -h[:n-k]
        else:
            kk = -k
            # First part: j from 0 to n-1-kk, contribution (1-h[j-kk])
            first_part = np.zeros(n)
            first_part[:n-kk] = 1 - h[:-kk]
            # Second part: j from kk to n-1, contribution -h[j+kk]
            second_part = np.zeros(n)
            second_part[kk:] = -h[:n-kk]
        
        grad += weights[n-1+k] * dx * (first_part + second_part)
    
    return grad
```

def project(h, target_sum):
    """Project h onto {x: 0<=x<=1, sum(x)=target_sum}"""
    # Bisection on lambda
    lo, hi = -max(h) - 1, 1 - min(h) + 1
    for _ in range(80):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0, 1)
        s = np.sum(proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2, 0, 1)

def smooth_max_grad(h, t, dx):
    """Gradient of smooth max of correlations"""
    n = len(h)
    corr = np.correlate(h, 1-h, mode="full")
    max_c = np.max(corr)
    exp_val = np.exp(t * (corr - max_c))
    w = exp_val / np.sum(exp_val)
    
    grad = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-12:
            continue
        lag = k - (n-1)
        # d corr[k] / d h[j]
        # corr[k] = sum_{i: 0<=i<n, 0<=i+lag<n} h[i]*(1-h[i+lag])
        # d/dh[j]: for j in first position: (1-h[j+lag]) * I(0<=j+lag<n)
        #           for j in second position: -h[j-lag] * I(0<=j-lag<n)
        
        if lag >= 0:
            # h[j] appears in first position when j in [0, n-1-lag], contribution (1-h[j+lag])
            # h[j] appears in second position when j in [lag, n-1], contribution -h[j-lag]
            grad[:n-lag] += w[k] * (1 - h[lag:])
            grad[lag:] += w[k] * (-h[:n-lag])
        else:
            m = -lag
            # h[j] appears in first position when j in [0, n-1-m], contribution (1-h[j-m])
            # h[j] appears in second position when j in [m, n-1], contribution -h[j+m]
            grad[:n-m] += w[k] * (1 - h[:-m])
            grad[m:] += w[k] * (-h[:n-m])
    
    return grad * dx
```

                grad += wkx * (1.0 - 2.0 * h)
            else:
                grad[lag:] += wkx * (1.0 - h[:n - lag])
                grad[:n - lag] -= wkx * h[lag:]
        else:
            m = -lag
            grad[:n - m] += wkx * (1.0 - h[m:])
            grad[m:] -= wkx * h[:n - m]
    return value, grad

def pg_optimize(h, target_sum, dx, t_phases, iters_per_phase, base_step=1.0):
    n = len(h)
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c5 = compute_c5(h_curr, dx)
    for t in t_phases:
        step = base_step / max(t, 1.0)
        for it in range(iters_per_phase):
            val, grad = smooth_max_and_grad(h_curr, t, dx)
            h_new = project_feasible(h_curr - step * grad, target_sum)
            val_new, _ = smooth_max_and_grad(h_new, t, dx)
            if val_new < val + 1e-15:
                h_curr = h_new
                step = min(step * 1.03, 5.0 / max(t, 1.0))
            else:
                step *= 0.5
            if (it + 1) % 200 == 0:
                c5_now = compute_c5(h_curr, dx)
                if c5_now < best_c5:
                    best_c5 = c5_now
                    best_h = h_curr.copy()
    c5_final = compute_c5(h_curr, dx)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h_curr.copy()
    return best_h, best_c5

def pairwise_opt(h, target_sum, dx, n_pairs, rng):
    n = len(h)
    best_c5 = compute_c5(h, dx)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _ in range(n_pairs):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        s = h[i] + h[j]
        if s < 1e-10:
            continue
        lo_t = max(0.0, s - 1.0)
        hi_t = min(1.0, s)
        if hi_t - lo_t < 1e-10:
            continue
        a, b = lo_t, hi_t
        c = b - gr * (b - a)
        d = a + gr * (b - a)
        h[i], h[j] = c, s - c
        fc = compute_c5(h, dx)
        h[i], h[j] = d, s - d
        fd = compute_c5(h, dx)
        for _ in range(20):
            if fc < fd:
                b, d, fd = d, c, fc
                c = b - gr * (b - a)
                h[i], h[j] = c, s - c
                fc = compute_c5(h, dx)
            else:
                a, c, fc = c, d, fd
                d = a + gr * (b - a)
                h[i], h[j] = d, s - d
                fd = compute_c5(h, dx)
        if fc <= fd:
            h[i], h[j] = c, s - c
        else:
            h[i], h[j] = d, s - d
        c5_now = compute_c5(h, dx)
        if c5_now < best_c5:
            best_c5 = c5_now
    return h, best_c5

def upsample(h, new_n):
    n = len(h)
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)
