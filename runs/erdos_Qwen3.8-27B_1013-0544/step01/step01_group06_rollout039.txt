    n = len(h)
    corr = compute_corr(h)
    # Smooth max: (1/alpha) * log(sum(exp(alpha * corr)))
    # Gradient: (sum(exp(alpha * corr) * grad_corr)) / (sum(exp(alpha * corr)))
    m = np.max(corr)
