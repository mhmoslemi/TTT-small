                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]

def subgradient_step(h, alpha, n_targets=5):
    """Take a subgradient step to reduce max correlation."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    
    # Find top n_targets maximizers
    top_k = np.argpartition(corr, -n_targets)[-n_targets:]
    
    # Compute subgradient (average of active gradients)
    subgrad = np.zeros(n)
    for k_idx in top_k:
        # k_idx is the index in the "full" correlation array
        # For full mode, index k_idx corresponds to shift = k_idx - (n-1)
        shift = k_idx - (n - 1)
        for j in range(n):
            # Contribution from j as h[i] term: (1-h[i+shift]) where i=j, valid if 0<=j+shift<n
            if 0 <= j + shift < n:
                subgrad[j] += (1 - h[j + shift])
            # Contribution from j as h[i+shift] term: -h[i] where i+shift=j, i=j-shift, valid if 0<=j-shift<n
            if 0 <= j - shift < n:
                subgrad[j] += (-h[j - shift])
    
    subgrad /= len(top_k)
    
    # Update
    h_new = h - alpha * subgrad
    h_new = np.clip(h_new, 0, 1)
    h_new = fix_sum(h_new, n / 2)
    return h_new

```python
def compute_flip_delta(h, i, j, n):
    """Compute delta in corr_full when flipping h[i]:1->0 and h[j]:0->1. Both must be valid."""
    assert h[i] == 1 and h[j] == 0
    delta = np.zeros(2*n-1)
    
    # i: 1 -> 0
    # m in [n-1-i, 2n-2-i]: delta[m] += h[i+m-(n-1)] - 1
    m_start = n - 1 - i
    m_end = 2*n - 2 - i  # inclusive
    if m_end >= 0 and m_start < 2*n-1:
        idx = np.arange(max(m_start, 0), min(m_end + 1, 2*n-1))
        positions = idx - (n-1) + i  # = i + m - (n-1)
        delta[idx] += h[positions] - 1.0
    
    # m in [i, i+n-1]: delta[m] += h[i-m+(n-1)]
    m_start2 = i
    m_end2 = i + n - 1
    if m_end2 >= 0 and m_start2 < 2*n-1:
        idx2 = np.arange(max(m_start2, 0), min(m_end2 + 1, 2*n-1))
        positions2 = i - idx2 + (n-1)  # = i - m + (n-1)
        delta[idx2] += h[positions2]
    
    # j: 0 -> 1 (negate the above with i -> j)
    # m in [n-1-j, 2n-2-j]: delta[m] += 1 - h[j+m-(n-1)]
    m_start3 = n - 1 - j
    m_end3 = 2*n - 2 - j
    if m_end3 >= 0 and m_start3 < 2*n-1:
        idx3 = np.arange(max(m_start3, 0), min(m_end3 + 1, 2*n-1))
        positions3 = idx3 - (n-1) + j
        delta[idx3] += 1.0 - h[positions3]
    
    # m in [j, j+n-1]: delta[m] += -h[j-m+(n-1)]
    m_start4 = j
    m_end4 = j + n - 1
    if m_end4 >= 0 and m_start4 < 2*n-1:
        idx4 = np.arange(max(m_start4, 0), min(m_end4 + 1, 2*n-1))
        positions4 = j - idx4 + (n-1)
        delta[idx4] -= h[positions4]
    
    return delta
```
