            c5 = compute_c5(h4)
            if c5 < best_c5:
                best_c5, best_h = c5, h4
                h4b = slsqp_optimize(h4, n_pts, target_sum, 400.0, 500)
                c5b = compute_c5(h4b)
                if c5b < best_c5:
                    best_c5, best_h = c5b, h4b

    # Phase 5: Try other n_points values
    for n_try in [50, 60, 80, 90, 100, 120, 140, 160]:
        if time_left() < 25:
            break
        if n_try == best_n:
            continue
        target_sum = n_try / 2.0
        starts = make_starts(n_try, best_h, target_sum)
        for h0 in starts[:4]:
            if time_left() < 12:
                break
            h5 = slsqp_optimize(h0, n_try, target_sum, 100.0, 400)
            c5 = compute_c5(h5)
            if c5 < best_c5:
                best_c5, best_h, best_n = c5, h5, n_try
                h5b = slsqp_optimize(h5, n_try, target_sum, 300.0, 400)
                c5b = compute_c5(h5b)
                if c5b < best_c5:
                    best_c5, best_h = c5b, h5b

    # Phase 6: Simulated annealing
    if best_h is not None and time_left() > 25:
        n_pts = best_n
        target_sum = n_pts / 2.0
        h_sa = best_h.copy()
        c5_sa = best_c5
        T = 0.004
        n_steps = min(3000, int((time_left() - 10) / 0.02))
        for it in range(max(n_steps, 100)):
            if time_left() < 5:
                break
            n_p = np.random.randint(1, 5)
            idx = np.random.choice(n_pts, n_p, replace=False)
            h_new = h_sa.copy()
            h_new[idx] += np.random.randn(n_p) * 0.015
            h_new = project_h(h_new, target_sum)
            c5_new = compute_c5(h_new)
            delta = c5_new - c5_sa
            if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-10)):
                h_sa = h_new
                c5_sa = c5_new
                if c5_sa < best_c5:
                    best_c5, best_h = c5_sa, h_sa.copy()
            T = max(T * 0.9985, 5e-7)
        # Gradient polish after SA
        if time_left() > 10:
            h_pol = projected_grad_descent(h_sa, n_pts, target_sum, 200.0, 200, time_left)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5, best_h = c5_pol, h_pol

    # Phase 7: Final high-precision SLSQP
    if best_h is not None and time_left() > 12:
        n_pts = best_n
        target_sum = n_pts / 2.0
        h_fin = slsqp_optimize(best_h, n_pts, target_sum, 800.0, 2000)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5:
            best_c5, best_h = c5_fin, h_fin
        if time_left() > 8:
            h_fin2 = projected_grad_descent(best_h, n_pts, target_sum, 800.0, 100, time_left)
            c5_fin2 = compute_c5(h_fin2)
            if c5_fin2 < best_c5:
                best_c5, best_h = c5_fin2, h_fin2

    return best_h, best_c5, best_n
