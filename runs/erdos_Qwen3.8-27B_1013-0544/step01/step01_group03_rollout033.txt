Let me try a different approach: **direct coordinate descent on the max**.

At each step:
1. Compute corr = get_corr(h)
2. Find k* = argmax corr
3. Compute the gradient of corr[k*] with respect to h
4. Take a step in the direction that reduces corr[k*]
5. Project to feasible set

For the linear correlation, corr[k] = sum_i h[i] * (1-h[i+k]) (for the appropriate range of i).

The gradient with respect to h[j] is:
- From the first factor: (1-h[j+k]) for valid j
- From the second factor: -h[j-k] for valid j

So ∂corr[k]/∂h[j] = (1-h[j+k]) - h[j-k] (with appropriate boundary conditions)

This means we can do a gradient step specifically on the "worst" lag, which is more targeted than the soft-max.

But we need to be careful: reducing one lag might increase another. So we should use a weighted combination of the "hot" lags.

Let me implement a "top-K" approach:
1. Find the K lags with the highest correlation
2. Compute a weighted average of their gradients
3. Take a step in that direction
4. Project

This is like a stochastic subgradient method but focused on the "worst" cases.

Let me code this up:

```python
def gradient_step(h, k_hot, step_size):
    """Take a gradient step to reduce correlation at lag k_hot."""
    n = len(h)
    corr = get_corr(h)
    # Gradient of corr[k_hot] w.r.t. h
    # corr[k] in np.correlate mode "full" at index k_full:
    # corr_full[m] = sum_i h[i] * (1-h)[i + m - (n-1)]
    # Let shift = m - (n-1), so corr_full[n-1+shift] = sum_i h[i]*(1-h[i+shift])
    
    shift = k_hot - (n - 1)
    # d/dh[j] of sum_i h[i]*(1-h[i+shift])
    # = sum_i [delta(i,j) * (1-h[i+shift]) - delta(i+shift, j) * h[i]]
    # = (1-h[j+shift]) - h[j-shift]  (for valid indices)
    
    grad = np.zeros(n)
    for i in range(n):
        i1 = i + shift
        if 0 <= i1 < n:
            # From first factor: h[i]*(1-h[i+shift]), derivative w.r.t h[i] is (1-h[i+shift])
            grad[i] += (1.0 - h[i1])
            # From second factor: h[i]*(1-h[i+shift]), derivative w.r.t h[i+shift] is -h[i]
            grad[i1] -= h[i]
    
    # Project gradient step
    h_new = h - step_size * grad
    return h_new
```

```python
    grad = np.zeros(n)
```

```python
        grad[j_valid1] += (1.0 - h[j_valid1 + s])
```

```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def fix_sum(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    for _ in range(100):
        if diff > 1e-12:
            room = 1.0 - h
            tr = room.sum()
            if tr < 1e-12:
                break
            h += room * (diff / tr)
        elif diff < -1e-12:
            av = h.copy()
            ta = av.sum()
            if ta < 1e-12:
                break
            h -= av * (-diff / ta)
        else:
            break
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
    return h

def smooth_max(obj, tau):
    x = tau * obj
    x_max = x.max()
    return float(x_max / tau + np.log(np.sum(np.exp(x - x_max))) / tau)

def soft_max_objective(h, tau):
    corr = get_corr(h)
    return smooth_max(corr, tau)

def soft_max_grad(h, tau):
    """Gradient of soft_max(corr(h), tau) w.r.t. h."""
    n = len(h)
    corr = get_corr(h)
    x = tau * corr
    x_max = x.max()
    w = np.exp(x - x_max)
    w /= w.sum()
    # sum_k w[k] * d(corr[k])/dh
    grad = np.zeros(n)
    # For efficiency, only compute for significant weights
    threshold = 1e-6
    active = np.where(w > threshold)[0]
    for k in active:
        s = k - (n - 1)
        # corr at shift s: sum_{i valid} h[i]*(1-h[i+s])
        # d/dh[j] = (1-h[j+s])*[valid] - h[j-s]*[valid]
        if s >= 0:
            j1 = np.arange(0, n - s)
            grad[j1] += w[k] * (1.0 - h[j1 + s])
            j2 = np.arange(s, n)
            grad[j2] -= w[k] * h[j2 - s]
        else:
            j1 = np.arange(-s, n)
            grad[j1] += w[k] * (1.0 - h[j1 + s])
            j2 = np.arange(0, n + s)
            grad[j2] -= w[k] * h[j2 - s]
    return tau * grad
```

Let me now think about the hot-lag approach more carefully. The key idea is:

At each iteration:
1. Compute corr = get_corr(h)
2. Identify the "hot" lags (top K by correlation value)
3. Compute the gradient to reduce the weighted sum of hot-lag correlations
4. Take a step and project

The projection is the tricky part. We need:
- 0 ≤ h[i] ≤ 1
- sum h = n/2

A simple projection: clip to [0,1], then adjust sum. But this might not be optimal.

A better approach: use the "water-filling" algorithm for the sum constraint after clipping.

Actually, for the sum constraint, the optimal projection is:
1. Clip to [0,1]
2. If sum > target, find a threshold θ such that sum(min(max(h-θ, 0), 1)) = target
3. Apply: h_new = min(max(h - θ, 0), 1)

This is the Euclidean projection onto the intersection of [0,1]^n and the hyperplane sum = target.

Let me implement this:

```python
def project_onto_feasible(h, target):
    """Project h onto {x: 0<=x_i<=1, sum x_i = target}."""
    h = h.copy()
    lo, hi = 0.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        projected = np.clip(h - mid, 0.0, 1.0)
        s = projected.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-12:
            break
    projected = np.clip(h - (lo + hi) / 2, 0.0, 1.0)
    # Fix any remaining discrepancy
    diff = target - projected.sum()
    if abs(diff) > 1e-10:
        mask = (projected > 0) & (projected < 1)
        if mask.sum() > 0:
            projected[mask] += diff / mask.sum()
            projected = np.clip(projected, 0.0, 1.0)
    return projected
```

This is the proper Euclidean projection. Let me verify: we want to minimize ||x - h||² subject to 0≤x_i≤1 and sum x_i = target. The Lagrangian gives x_i = clip(h_i - λ, 0, 1) for some λ. We find λ by bisection so that sum x_i = target.

```python
def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_feasible(h, target):
    h = h.astype(float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        m = (proj > 1e-12) & (proj < 1.0 - 1e-12)
        if m.any():
            proj[m] += diff / m.sum()
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def smooth_max_val(obj, tau):
    x = tau * obj
    xm = x.max()
    return float(xm / tau + np.log(np.sum(np.exp(x - xm))) / tau)

def smooth_max_weights(obj, tau):
    x = tau * obj
    xm = x.max()
    w = np.exp(x - xm)
    w /= w.sum()
    return w

def corr_grad(h, s):
    """Gradient of corr at shift s w.r.t. h. Returns array of length n."""
    n = len(h)
    grad = np.zeros(n)
    if s >= 0:
        j1 = np.arange(n - s)
        grad[j1] += 1.0 - h[j1 + s]
        if s > 0:
            j2 = np.arange(s, n)
            grad[j2] -= h[j2 - s]
    else:
        j1 = np.arange(-s, n)
        grad[j1] += 1.0 - h[j1 + s]
        j2 = np.arange(n + s)
        grad[j2] -= h[j2 - s]
    return grad

def smooth_max_gradient(h, tau):
    n = len(h)
    corr = get_corr(h)
    w = smooth_max_weights(corr, tau)
    grad = np.zeros(n)
    active = np.where(w > 1e-8)[0]
    for k in active:
        s = k - (n - 1)
        grad += w[k] * corr_grad(h, s)
    return tau * grad

def slsqp_optimize(h, tau, maxiter=300):
    n = len(h)
    target = n / 2.0
    bnds = Bounds(np.zeros(n), np.ones(n))
    
    def obj(x):
        return smooth_max_val(get_corr(x), tau)
    
    def objg(x):
        return smooth_max_gradient(x, tau)
    
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h.copy(), jac=objg, method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_feasible(res.x, target)
    except Exception:
        return h.copy()

def hot_lag_descend(h, n_iters=3000, top_k=5):
    n = len(h)
    target = n / 2.0
    h = project_feasible(h.copy(), target)
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    step = 0.02
    stale = 0
    
    for it in range(n_iters):
        corr = get_corr(h)
        # Weight by excess over mean
        mean_c = corr[corr > 0].mean() if (corr > 0).any() else 0
        excess = np.maximum(corr - mean_c, 0)
        total_excess = excess.sum()
        if total_excess < 1e-10:
            break
        w = excess / total_excess
        active = np.where(w > 0.01)[0]
        if len(active) == 0:
            active = np.argsort(corr)[-top_k:]
        # Also include the absolute top
        top = np.argsort(corr)[-top_k:]
        active = np.unique(np.concatenate([active, top]))
        
        grad = np.zeros(n)
        for k in active:
            s = k - (n - 1)
            grad += w[k] * corr_grad(h, s)
        
        # Backtracking line search
        improved = False
        for mult in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02]:
            alpha = step * mult
            h_new = project_feasible(h - alpha * grad, target)
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                h = h_new
                best_c5 = c5_new
                best_h = h.copy()
                stale = 0
                improved = True
                break
        if not improved:
            step *= 0.6
            stale += 1
            if stale > 500:
                break
    
    return best_h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_feasible(h, target):
    h = h.astype(float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        m = (proj > 1e-12) & (proj < 1.0 - 1e-12)
        if m.any():
            proj[m] += diff / m.sum()
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def smooth_max_val(obj, tau):
    x = tau * obj
    xm = x.max()
    return float(xm / tau + np.log(np.sum(np.exp(x - xm))) / tau)

def smooth_max_weights(obj, tau):
    x = tau * obj
    xm = x.max()
    w = np.exp(x - xm)
    w /= w.sum()
    return w

def corr_grad(h, s):
    n = len(h)
    grad = np.zeros(n)
    if s >= 0:
        i_vals = np.arange(n - s)
        grad[i_vals] += 1.0 - h[i_vals + s]
        grad[i_vals + s] -= h[i_vals]
    else:
        i_vals = np.arange(-s, n)
        grad[i_vals] += 1.0 - h[i_vals + s]
        grad[i_vals + s] -= h[i_vals]
    return grad

def soft_max_grad(h, tau):
    n = len(h)
    corr = get_corr(h)
    w = smooth_max_weights(corr, tau)
    grad = np.zeros(n)
    active = np.where(w > 1e-8)[0]
    for k in active:
        s = k - (n - 1)
        grad += w[k] * corr_grad(h, s)
    return grad

def slsqp_optimize(h, tau, maxiter=300):
    n = len(h)
    target = n / 2.0
    bnds = Bounds(np.zeros(n), np.ones(n))
    def obj(x):
        return smooth_max_val(get_corr(x), tau)
    def objg(x):
        return soft_max_grad(x, tau)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h.copy(), jac=objg, method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_feasible(res.x, target)
    except Exception:
        return h.copy()

def hot_lag_descend(h, n_iters=3000, top_k=5, use_momentum=True):
    n = len(h)
    target = n / 2.0
    h = project_feasible(h.copy(), target)
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    step = 0.01
    v = np.zeros(n)
    stale = 0
    for it in range(n_iters):
        corr = get_corr(h)
        top_idx = np.argsort(corr)[-top_k:]
        grad = np.zeros(n)
        for k in top_idx:
            s = k - (n - 1)
            grad += corr_grad(h, s)
        grad /= top_k
        if use_momentum:
            v = 0.85 * v + 0.15 * grad
            direction = v
        else:
            direction = grad
        improved = False
        for mult in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02]:
            alpha = step * mult
            h_new = project_feasible(h - alpha * direction, target)
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5 - 1e-13:
                h = h_new
                best_c5 = c5_new
                best_h = h.copy()
                stale = 0
                improved = True
                break
        if not improved:
            step *= 0.7
            stale += 1
            if stale > 400:
                break
    return best_h, best_c5

def weighted_lag_descend(h, n_iters=3000, threshold_frac=0.3):
    n = len(h)
    target = n / 2.0
    h = project_feasible(h.copy(), target)
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    step = 0.005
    stale = 0
    for it in range(n_iters):
        corr = get_corr(h)
        corr_max = corr.max()
        threshold = corr_max * (1.0 - threshold_frac)
        active = np.where(corr > threshold)[0]
        if len(active) == 0:
            active = np.argsort(corr)[-3:]
        weights = np.zeros(len(corr))
        weights[active] = (corr[active] - threshold)
        tw = weights.sum()
        if tw < 1e-12:
            weights[active] = 1.0 / len(active)
        else:
            weights /= tw
        grad = np.zeros(n)
        for k in active:
            s = k - (n - 1)
            grad += weights[k] * corr_grad(h, s)
        improved = False
        for mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            alpha = step * mult
            h_new = project_feasible(h - alpha * grad, target)
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5 - 1e-13:
                h = h_new
                best_c5 = c5_new
                best_h = h.copy()
                stale = 0
                improved = True
                break
        if not improved:
            step *= 0.7
            stale += 1
            if stale > 400:
                break
    return best_h, best_c5

    t0 = time.time()
    def rem():
        return budget_s - (time.time() - t0)
```
