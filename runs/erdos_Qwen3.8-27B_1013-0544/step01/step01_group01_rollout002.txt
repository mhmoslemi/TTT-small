```python
from scipy.optimize import minimize, Bounds, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def project_polytope(h, target_sum):
    """Project h onto {x: 0<=x[i]<=1, sum(x)=target_sum} using bisection on lambda."""
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        projected = np.clip(h - mid, 0.0, 1.0)
        s = projected.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0.0, 1.0)

def compute_grad_active_lag(h, lag, n):
    """Compute gradient of correlation at specific lag."""
    grad = np.zeros(n)
    if lag > 0:
        # f = sum_{j=0}^{n-lag-1} h[j+lag] * (1-h[j])
        # d/dh[m] = (1-h[m-lag]) if lag<=m<=n-1, minus h[m+lag] if 0<=m<=n-lag-1
        for m in range(lag, n):
            grad[m] += (1.0 - h[m - lag])
        for m in range(0, n - lag):
            grad[m] -= h[m + lag]
    elif lag < 0:

            grad[m] -= h[m - k]
    else:
```

```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta


def project_polytope(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0.0, 1.0)


def grad_lag(h, lag, n):
    grad = np.zeros(n)
    if lag > 0:
        grad[lag:] += (1.0 - h[:n - lag])
        grad[:n - lag] -= h[lag:]
    elif lag < 0:
        k = -lag
        grad[:n - k] += (1.0 - h[k:])
        grad[k:] -= h[:n - k]
    else:
        grad[:] = 1.0 - 2.0 * h
    return grad


def grad_projection_step(h, n, target_sum, threshold_frac=0.01):
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = corr.max()
    active = np.where(corr >= max_val - threshold_frac)[0]
    lags = [int(i) - (n - 1) for i in active]
    grad = np.zeros(n)
    for lag in lags:
        grad += grad_lag(h, lag, n)
    gn = np.linalg.norm(grad)
    if gn < 1e-12:
        return h, 0.0
    c5_cur = compute_c5(h)
    step_size = 1.0
    best_h_try = h
    best_c5_try = c5_cur
    for _ in range(25):
        h_try = project_polytope(h - (step_size / (n * gn)) * grad, target_sum)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5_try:
            best_c5_try = c5_try
            best_h_try = h_try
            if c5_try < c5_cur - 1e-10:
                break
        step_size *= 0.6
    return best_h_try, best_c5_try


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    else:
        n = 70
        h_init = np.full(n, 0.5)

    dx = 2.0 / n
    target_sum = n / 2.0
    best_h = project_polytope(h_init, target_sum)
    best_c5 = compute_c5(best_h)

    def time_left():
        return budget_s - (time.time() - start_time)

    bounds = Bounds(np.zeros(n), np.ones(n))

    def obj_func(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        return smooth_max(corr, 120.0) * dx

    def con_fun(x):
        return np.sum(x) - target_sum

    constraints = [{'type': 'eq', 'fun': con_fun, 'jac': lambda x: np.ones(n)}]

    # Phase 1: SLSQP from initial point
    try:
        res = minimize(obj_func, best_h, method='SLSQP', bounds=bounds,
                       constraints=constraints, options={'maxiter': 1500, 'ftol': 1e-13})
        h_new = project_polytope(res.x, target_sum)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_h, best_c5 = h_new.copy(), c5
    except Exception:
        pass

    # Phase 2: Gradient projection (main workhorse)
    h = best_h.copy()
    for _ in range(20000):
        if time_left() < 60:

        h, c5_try = grad_projection_step(h, n, target_sum, 0.005)
        if c5_try < best_c5:

            for _ in range(n_sw):

        else:

            else:
                break

    return best_h, best_c5, n
```
