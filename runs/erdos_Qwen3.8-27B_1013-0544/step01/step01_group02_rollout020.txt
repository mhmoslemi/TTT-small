def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    sm = shifted.max()
    exp_s = np.exp(shifted - sm)
    w = exp_s / exp_s.sum()
    val = m + (sm + np.log(exp_s.sum())) / alpha
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-10:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif lag > 0:

            #        for i in [lag,n-1], d/dh[i] = -h[i-lag]
            grad[:n-lag] += wk * (1.0 - h[lag:])
            grad[lag:] += wk * (-h[:n-lag])

            # grad: for i in [t,n-1], d/dh[i] = (1-h[i-t])  [h[i+t] term, i+t = j -> j=i+t... wait]
            # Let me redo: O(-t) = sum_{i=0}^{n-1-t} h[i+t]*(1-h[i])

            #   h[i+t] contributes when j = i+t, i.e., i = j-t. Valid for j-t in [0,n-1-t], i.e., j in [t, n-1]

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_box_sum(y, target):
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(120):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_and_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    sm = shifted.max()
    exp_s = np.exp(shifted - sm)
    w = exp_s / exp_s.sum()
    val = m + (sm + np.log(exp_s.sum())) / alpha
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-10:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            s = lag
            grad[s:] += wk * (1.0 - h[:n - s])
            grad[:n - s] += wk * (-h[s:])
        else:
            t = -lag
            grad[:n - t] += wk * (1.0 - h[t:])
            grad[t:] += wk * (-h[:n - t])
    return val, grad
