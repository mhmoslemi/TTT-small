            c5_new = compute_c5(h_new)
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h_new.copy()

    # Phase 6: Random restarts for diversity
    n_fin = len(h_best)
    for i in range(8):
        if tl() < 50:
            break
        scale = 0.03 + 0.02 * i
        h_pert = h_best + rng.randn(n_fin) * scale
        h_pert = np.clip(h_pert, 0.0, 1.0)
        h_pert = project_h(h_pert, n_fin * 0.5)
        for temp in [0.005, 0.001]:
            if tl() < 25:
                break
            res = minimize(obj_smooth_max, h_pert, args=(temp, 500.0, n_fin * 0.5),
                           method='L-BFGS-B', bounds=[(0.0, 1.0)] * n_fin,
                           options={'maxiter': 3000, 'ftol': 1e-18})
            h_new = project_h(res.x, n_fin * 0.5)
            c5_new = compute_c5(h_new)
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h_new.copy()
            h_pert = h_new

    # Phase 7: Final polishing
    n_fin = len(h_best)
    for temp in [0.0001, 0.00005]:
        if tl() < 15:
            break
        res = minimize(obj_smooth_max, h_best.copy(),
                       args=(temp, 1000.0, n_fin * 0.5),
                       method='L-BFGS-B', bounds=[(0.0, 1.0)] * n_fin,
                       options={'maxiter': 5000, 'ftol': 1e-18})
        h_new = project_h(res.x, n_fin * 0.5)
        c5_new = compute_c5(h_new)
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new.copy()

    # Final barrier push
    if tl() > 10:
        n_fin = len(h_best)
        dx_fin = 2.0 / n_fin
        corr_fin = np.correlate(h_best, 1.0 - h_best, mode='full') * dx_fin
        t_final = float(np.max(corr_fin)) - 0.0003
        res = minimize(obj_barrier, h_best.copy(),
                       args=(t_final, 1000.0, n_fin * 0.5),
                       method='L-BFGS-B', bounds=[(0.0, 1.0)] * n_fin,
                       options={'maxiter': 3000, 'ftol': 1e-18})
        h_new = project_h(res.x, n_fin * 0.5)
        c5_new = compute_c5(h_new)
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new.copy()

    h_best = project_h(h_best, len(h_best) * 0.5)
    c5_final = compute_c5(h_best)
    return h_best, c5_final, len(h_best)
