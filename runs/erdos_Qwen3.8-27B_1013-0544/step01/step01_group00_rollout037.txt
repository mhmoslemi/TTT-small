```python
                h, c5 = h3, c53
                h_best = h.copy()

    # Phase 4: Subgradient from best binary
    if t_left() > 200:
        h4, c54 = subgrad_opt(h_bin_best, n_iters=5000, seed=seed)
        if c54 < c5:
            h, c5 = h4, c54
            h_best = h.copy()
        if t_left() > 120:
            h5, c55 = opt_t_var(h_bin_best, max_iter=500, ftol=1e-12)
            if c55 < c5:
                h, c5 = h5, c55
                h_best = h.copy()

    # Phase 5: Multi-start continuous from perturbed best
    for i in range(3):
        if t_left() < 100:
            break
        h_p = h_best + rng.randn(n) * 0.03
        h_p = project_h(np.clip(h_p, 0, 1), n * 0.5)
        h6, c56 = opt_t_var(h_p, max_iter=400, ftol=1e-12)
        if c56 < c5:
            h, c5 = h6, c56
            h_best = h.copy()

    # Phase 6: Final SLSQP + smooth
    if t_left() > 100:
        h7, c57 = opt_t_var(h_best, max_iter=600, ftol=1e-13)
        if c57 < c5:
            h, c5 = h7, c57
            h_best = h.copy()

    for temp in [0.001, 0.0004]:
        if t_left() < 60:
            break
        h8, c58 = opt_smooth(h_best, temp, max_iter=150)
        if c58 < c5:
            h, c5 = h8, c58
            h_best = h.copy()

    if t_left() > 80:
        h9, c59 = opt_t_var(h_best, max_iter=600, ftol=1e-13)
        if c59 < c5:
            h, c5 = h9, c59
            h_best = h.copy()

    h_best = project_h(h_best, n * 0.5)
    c5_final = compute_c5(h_best)
    return h_best, c5_final, n
```
