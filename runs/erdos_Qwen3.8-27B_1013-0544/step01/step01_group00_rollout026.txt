    """Project h onto {x : 0 ≤ x[i] ≤ 1, sum(x) = target_sum}"""
    # This is the Euclidean projection onto the intersection of [0,1]^n and the hyperplane sum(x) = target_sum
    # Use bisection on the offset
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        projected = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(projected)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    projected = np.clip(h - (lo+hi)/2, 0.0, 1.0)
    # Final adjustment for numerical precision
    diff = target_sum - np.sum(projected)
    if abs(diff) > 1e-10:
        # Distribute among non-saturated variables
        mask = (projected > 1e-10) & (projected < 1 - 1e-10)
        if np.sum(mask) > 0:
            projected[mask] += diff / np.sum(mask)
            projected = np.clip(projected, 0.0, 1.0)
    return projected

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))


def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - proj.sum()
    if abs(diff) > 1e-12:
        mask = (proj > 1e-8) & (proj < 1.0 - 1e-8)
        cnt = mask.sum()
        if cnt > 0:
            proj = proj.copy()
            proj[mask] += diff / cnt
            proj = np.clip(proj, 0.0, 1.0)
    return proj


def smoothed_min_ac(h, temp):
    ac = np.correlate(h, h, 'full')
    m = ac.min()
    return m - temp * np.log(np.sum(np.exp(-(ac - m) / temp)))


def smoothed_min_ac_grad(h, temp):
    n = len(h)
    ac = np.correlate(h, h, 'full')
    m = ac.min()
    w = np.exp(-(ac - m) / temp)
    w /= w.sum()
    grad = np.zeros(n)
    for d in range(2 * n - 1):
        wd = w[d]
        if wd < 1e-14:
            continue
        s = d - (n - 1)
        if s >= 0:
            grad[:n - s] += wd * h[s:]
            grad[s:] += wd * h[:n - s]
        else:
            sp = -s
            grad[sp:] += wd * h[:n - sp]
            grad[:n - sp] += wd * h[sp:]
    return grad


def penalized_obj(h, temp, penalty):
    n = len(h)
    val = -smoothed_min_ac(h, temp)
    c = np.sum(h) - n * 0.5
    val += penalty * c * c
    return val


def penalized_grad(h, temp, penalty):
    n = len(h)
    g = -smoothed_min_ac_grad(h, temp)
    c = np.sum(h) - n * 0.5
    g = g + 2.0 * penalty * c
    return g

    n = len(h0)
    res = minimize(penalized_obj, h0, jac=penalized_grad, args=(temp, penalty),
                   method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-10, 'maxls': 50})
    h_new = np.clip(res.x, 0.0, 1.0)
    h_new = project_feasible(h_new, n * 0.5)
    return h_new, compute_c5(h_new)


def t_objective(v):
    return v[-1]


def t_ineq(v):
    n = len(v) - 1
    h = v[:n]
    t = v[-1]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return t - corr


def t_eq(v):
    n = len(v) - 1
    return np.sum(v[:n]) - n * 0.5


def t_eq_jac(v):
    n = len(v) - 1
    j = np.zeros(len(v))
    j[:n] = 1.0
    return j


def optimize_t_var(h0, max_iter=400, ftol=1e-12):
    n = len(h0)
    c5_cur = compute_c5(h0)
    v0 = np.append(h0.copy(), c5_cur + 0.01)
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.5)]
    cons = [
        {'type': 'ineq', 'fun': t_ineq},
        {'type': 'eq', 'fun': t_eq, 'jac': t_eq_jac}
    ]
    res = minimize(t_objective, v0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': ftol})
    h_new = np.clip(res.x[:n], 0.0, 1.0)
    h_new = project_feasible(h_new, n * 0.5)
    return h_new, compute_c5(h_new)


def interp_h(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new * 0.5)


def projected_gradient_max_ac(h0, temp, n_iter, lr0=0.5):
    """Projected gradient ascent on smoothed min autocorrelation"""
    n = len(h0)
    target = n * 0.5

    best_c5 = compute_c5(h)

        alpha = lr0 / np.sqrt(t + 1)

        h_new = project_feasible(h_new, target)

        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
