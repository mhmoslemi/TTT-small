    return v[-1]

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))


def get_corr(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode='full') * dx


def project_h(h, target):
    h = h.copy()
    if abs(np.sum(h) - target) < 1e-14:
        return np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)


def soft_max_val(corr, T):
    m = np.max(corr)
    return m + T * np.log(np.sum(np.exp((corr - m) / T)))


def soft_max_weights(corr, T):
    m = np.max(corr)
    w = np.exp((corr - m) / T)
    return w / np.sum(w)


def grad_soft_max(h, w, dx):
    n = len(h)
    j = np.arange(n)
    m_idx = np.arange(n)
    idx1 = m_idx[:, None] - j[None, :] + (n - 1)
    FT = (w[idx1] * (1.0 - h)[None, :]).sum(axis=1)
    idx2 = j[None, :] - m_idx[:, None] + (n - 1)
    ST = (w[idx2] * h[None, :]).sum(axis=1)
    return dx * (FT - ST)


def projected_gradient(h0, T, max_iter=5000):

    dx = 2.0 / n
    target = n * 0.5
    h = project_h(h0, target)
    corr = get_corr(h)
    obj = soft_max_val(corr, T)
    c5 = float(np.max(corr))
    best_h = h.copy()
    best_c5 = c5
    lr = 0.02
    stall = 0

    for it in range(max_iter):
        w = soft_max_weights(corr, T)
        grad = grad_soft_max(h, w, dx)
        gn = np.linalg.norm(grad)
        if gn < 1e-12:
            break
        improved = False
        for _ in range(40):
            h_new = project_h(h - lr * grad, target)
            if np.allclose(h_new, h, atol=1e-14):
                break
            corr_new = get_corr(h_new)
            obj_new = soft_max_val(corr_new, T)
            if obj_new < obj - 1e-16:
                h = h_new
                corr = corr_new
                obj = obj_new
                c5_new = float(np.max(corr_new))
                if c5_new < best_c5:
                    c5 = c5_new
                    best_c5 = c5_new
                    best_h = h.copy()
                improved = True
                break
            lr *= 0.5
        if not improved:
            stall += 1
            if stall > 50:
                break
            lr *= 0.5
        else:
            stall = 0
            lr = min(lr * 1.1, 0.5)

    return best_h, best_c5


def t_obj(v):
    return v[-1]


def t_ineq(v):
    n = len(v) - 1
    h = v[:n]
    t = v[-1]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return t - corr


def t_eq(v):
    n = len(v) - 1
    return np.sum(v[:n]) - n * 0.5


def t_eq_jac(v):
    n = len(v) - 1
    jj = np.zeros(len(v))
    jj[:n] = 1.0
    return jj


def slsqp_mm(h0, max_iter=400):
    n = len(h0)
    c5_cur = compute_c5(h0)
    v0 = np.append(h0.copy(), c5_cur + 0.01)
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.5)]
    cons = [
        {'type': 'ineq', 'fun': t_ineq},
        {'type': 'eq', 'fun': t_eq, 'jac': t_eq_jac}
    ]
    res = minimize(t_obj, v0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-12})
    h_new = np.clip(res.x[:n], 0.0, 1.0)
    h_new = project_h(h_new, n * 0.5)
    return h_new, compute_c5(h_new)


def interp_h(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return project_h(h_old, n_new * 0.5)
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_h(h_new, n_new * 0.5)


def sharpen_and_optimize(h_in, target, T, rng, time_left_fn):
    n = len(h_in)
    h_sh = 1.0 * (h_in > 0.5).astype(float)
    h_sh = project_h(h_sh, target)
    h_blend = 0.5 * h_sh + 0.5 * h_in
    h_blend = project_h(h_blend, target)
    h_new, c5_new = projected_gradient(h_blend, T, max_iter=3000)
    return h_new, c5_new
