    n = len(h0)
    h = h0.copy()
    best_f = smooth_max_val(np.correlate(h, 1.0-h, mode="full"), tau)
    
    for it in range(maxiter):

def make_binary_seed(n, mode=0):
    """Create structured binary seeds."""
    if mode == 0:
        # Every other (simple)
        h = np.zeros(n)
        h[::2] = 1.0
    elif mode == 1:
        # Golden ratio
        phi = (1 + np.sqrt(5)) / 2.0
        h = np.zeros(n)
        for i in range(n):
            h[i] = 1.0 if (i * phi) % 1.0 < 0.5 else 0.0
    elif mode == 2:
        # Quadratic residues
        p = 1
        while p < n:
            p += 1
        h = np.zeros(n)
        for i in range(n):
            h[i] = 1.0 if (i * i) % p < p / 2 else 0.0
    elif mode == 3:
        # Block pattern
        h = np.zeros(n)
        block = n // 4
        h[:block] = 1.0
        h[n//2:n//2+block] = 1.0
    elif mode == 4:
        # Random
        idx = np.random.permutation(n)[:n//2]
        h = np.zeros(n)
        h[idx] = 1.0
    else:
        h = np.full(n, 0.5)
    # Fix to have exactly n/2 ones
    cnt = int(h.sum())
    if cnt > n // 2:
        ones = np.where(h == 1.0)[0]
        extra = cnt - n // 2
        for i in range(extra):
            h[ones[i]] = 0.0
    elif cnt < n // 2:
        zeros = np.where(h == 0.0)[0]
        need = n // 2 - cnt
        for i in range(need):
            h[zeros[i]] = 1.0
    return h

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val(x, t):
    tx = t * x
    tx_max = np.max(tx)
    return (tx_max + np.log(np.sum(np.exp(tx - tx_max)))) / t

def fix_sum(h, target):
    h = np.array(h, dtype=float)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return np.clip(h, 0.0, 1.0)
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = mask.sum()
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def compute_grad(h, tau):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    tc = tau * (corr - np.min(corr))
    tc_max = np.max(tc)
    w = np.exp(tc - tc_max)
    w /= w.sum()
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-13:
            continue
        l = m - (n - 1)
        if l >= 0:
            grad[l:] += wm * (1.0 - h[:n - l])
            grad[:n - l] -= wm * h[l:]
        else:
            nl = -l
            grad[:n - nl] += wm * (1.0 - h[nl:])
            grad[nl:] -= wm * h[:n - nl]
    return grad

        grad = compute_grad(h, tau)

            h_try = np.clip(h_try, 0.0, 1.0)

            corr_try = np.correlate(h_try, 1.0 - h_try, mode="full")
            f_try = smooth_max_val(corr_try, tau)

                f_cur = f_try

        else:

def coord_search(h, target, n_passes=40):

    f_cur = evaluate_c5(h)
    for p in range(n_passes):

                        f_cur = f_try
