    # idx_f[i,m] = i + shifts[m]
    # idx_b[i,m] = i - shifts[m]
    
    i_idx = np.arange(n)[:, None]
    s_idx = shifts[None, :]
    idx_f = i_idx + s_idx  # shape (n, 2n-1)
    idx_b = i_idx - s_idx
    
    # Get h values (0 where out of bounds)
    valid_f = (idx_f >= 0) & (idx_f < n)
    valid_b = (idx_b >= 0) & (idx_b < n)

    h_b = np.where(valid_b, h[np.clip(idx_b, 0, n-1)], 0.0)
    
    # Weighted sum

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val(corr, tau):
    x = np.asarray(corr, dtype=float) / tau
    xm = np.max(x)
    return float(tau * (xm + np.log(np.sum(np.exp(x - xm)))))

def fix_sum(h, target):
    h = np.array(h, dtype=float)
    diff = target - h.sum()
    if abs(diff) < 1e-15:
        return np.clip(h, 0.0, 1.0)
    if diff > 0:
        mask = h < 1.0 - 1e-12
    else:
        mask = h > 1e-12
    cnt = mask.sum()
    if cnt == 0:
        return np.clip(h, 0.0, 1.0)
    h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def compute_smooth_grad(h, weights):
    """Gradient of smooth_max(corr, tau) w.r.t. h.
    weights: softmax(corr/tau), length 2n-1, sums to 1."""
    n = len(h)
    shifts = np.arange(2 * n - 1) - (n - 1)
    i_idx = np.arange(n)[:, None]
    s_idx = shifts[None, :]
    idx_f = i_idx + s_idx
    idx_b = i_idx - s_idx
    valid_f = (idx_f >= 0) & (idx_f < n)
    valid_b = (idx_b >= 0) & (idx_b < n)
    h_f = np.where(valid_f, h[np.clip(idx_f, 0, n - 1)], 0.0)
    h_b = np.where(valid_b, h[np.clip(idx_b, 0, n - 1)], 0.0)
    w = weights[None, :]
    forward = (h_f * w).sum(axis=1)
    backward = (h_b * w).sum(axis=1)
    return 1.0 - forward - backward

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    def rem():
        return budget_s - (time.time() - t0)
    
    # Initialize
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float).copy()
    else:
        h = np.full(70, 0.5)
    
    n = len(h)
    target = n / 2.0
    h = fix_sum(h, target)
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # === Phase 1: Subgradient descent with momentum ===
    n_iters = 8000
    tau_seq = np.geomspace(50, 0.5, n_iters)
    lr_seq = np.geomspace(0.8, 0.005, n_iters)
    
    momentum = np.zeros(n)
    beta = 0.7
    
    for it in range(n_iters):
        if rem() < 30:
            break
        tau = tau_seq[it]
        lr = lr_seq[it]
        
        corr = np.correlate(h, 1.0 - h, mode="full")
        x = corr / tau
        xm = np.max(x)
        weights = np.exp(x - xm) / np.sum(np.exp(x - xm))
        
        grad = compute_smooth_grad(h, weights)
        
        # Momentum
        momentum = beta * momentum - lr * grad
        h_new = h + momentum
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = fix_sum(h_new, target)
        
        c5 = evaluate_c5(h_new)
        if c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = c5
        h = h_new
    
    # === Phase 2: SLSQP refinement ===
    def slsqp_optimize(h_start, n_pts, tau, maxiter=500):
        target_t = n_pts / 2.0
        bnds = Bounds(np.zeros(n_pts), np.ones(n_pts))
        
        def obj(x):
            corr = np.correlate(x, 1.0 - x, mode="full")
            return smooth_max_val(corr, tau)
        
        def obj_grad(x):
            corr = np.correlate(x, 1.0 - x, mode="full")
            xi = corr / tau
            xm = np.max(xi)
            w = np.exp(xi - xm) / np.sum(np.exp(xi - xm))
            return compute_smooth_grad(x, w)
        
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_t,
                 'jac': lambda x: np.ones(n_pts)}]
        
        try:
            res = minimize(obj, h_start, jac=obj_grad, method='SLSQP',
                          bounds=bnds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-14})
            x = fix_sum(res.x, target_t)
            c5 = evaluate_c5(x)
            return x, c5
        except Exception:
            return h_start, None
    
    # Decreasing tau schedule
    for tau in [100, 30, 10, 3, 1, 0.5, 0.3, 0.2, 0.1]:
        if rem() < 15:
            break
        hn = len(best_h)
        h_new, c5_new = slsqp_optimize(best_h, hn, tau, 400)
        if c5_new is not None and c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
    
    # === Phase 3: Try different resolutions ===
    for tn in [80, 100, 140, 160]:
        if rem() < 30:
            break
        target_t = tn / 2.0
        h_t = np.interp(np.linspace(0, 1, tn, endpoint=False), 
                       np.linspace(0, 1, len(best_h), endpoint=False), best_h)
        h_t = fix_sum(h_t, target_t)
        for tau in [10, 3, 1, 0.5, 0.2]:
            if rem() < 15:
                break
            h_new, c5_new = slsqp_optimize(h_t, tn, tau, 300)
            if c5_new is not None and c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
            h_t = h_new
    
    # === Phase 4: Random perturbations + SLSQP ===
    hn = len(best_h)
    for trial in range(30):
        if rem() < 15:
            break
        scale = 0.05 + 0.2 * np.random.random()
        hp = best_h + np.random.normal(0, scale, hn)
        hp = np.clip(hp, 0.0, 1.0)
        hp = fix_sum(hp, hn / 2.0)
        for tau in [10, 1, 0.3]:
            if rem() < 10:
                break
            h_new, c5_new = slsqp_optimize(hp, hn, tau, 200)
            if c5_new is not None and c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
            hp = h_new
    
    # === Phase 5: Fine polish ===
    if rem() > 15:
        hn = len(best_h)
        for tau in [0.1, 0.05, 0.03]:
            if rem() < 10:
                break
            h_new, c5_new = slsqp_optimize(best_h, hn, tau, 500)
            if c5_new is not None and c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
    
    # Final fix
    best_h = fix_sum(best_h, len(best_h) / 2.0)
    best_c5 = evaluate_c5(best_h)
    
    return best_h, best_c5, len(best_h)

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_and_weights(corr, tau):
    x = np.asarray(corr, dtype=float) / tau
    xm = np.max(x)
    ex = np.exp(x - xm)
    s = ex.sum()
    w = ex / s
    val = tau * (xm + np.log(s))
    return val, w

def smooth_grad(h, weights):
    n = len(h)
    shifts = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = weights[m]
        if wm < 1e-15:
            continue
        s = shifts[m]
        lo1, hi1 = max(0, -s), min(n, n - s)
        if hi1 > lo1:
            grad[lo1:hi1] += wm * (1.0 - h[lo1 + s:hi1 + s])
        lo2, hi2 = max(0, s), min(n, n + s)
        if hi2 > lo2:
            grad[lo2:hi2] -= wm * h[lo2 - s:hi2 - s]
    return grad

def fix_sum(h, target):
    h = np.array(h, dtype=float)
    diff = target - h.sum()
    if abs(diff) < 1e-15:
        return np.clip(h, 0.0, 1.0)
    if diff > 0:
        mask = h < 1.0 - 1e-12
    else:
        mask = h > 1e-12
    cnt = mask.sum()
    if cnt == 0:
        return np.clip(h, 0.0, 1.0)
    h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def slsqp_opt(h0, tau, maxiter=500):
    n = len(h0)
    tgt = n / 2.0
    bnds = Bounds(np.zeros(n), np.ones(n))

    def fun(x):
        v, _ = smooth_max_and_weights(np.correlate(x, 1.0 - x, mode="full"), tau)
        return v

    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        _, w = smooth_max_and_weights(corr, tau)
        return smooth_grad(x, w)

    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - tgt,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(fun, h0, jac=jac, method='SLSQP', bounds=bnds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        x = fix_sum(res.x, tgt)
        return x, evaluate_c5(x)
    except Exception:
        x = fix_sum(h0, tgt)
        return x, evaluate_c5(x)

def subgrad_descent(h0, n_iters, t0, budget_s):
    n = len(h0)
    tgt = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    momentum = np.zeros(n)
    beta = 0.7
    tau_arr = np.geomspace(80, 0.3, n_iters)
    lr_arr = np.geomspace(0.8, 0.005, n_iters)
    for it in range(n_iters):
        if budget_s - (time.time() - t0) < 40:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        _, w = smooth_max_and_weights(corr, tau_arr[it])
        g = smooth_grad(h, w)

def active_set_descent(h0, n_iters, t0, budget_s):
    n = len(h0)
    tgt = n / 2.0
    h = h0.copy()
