```python
    #            - I(0 ≤ j+(k-(n-1)) ≤ n-1) * h[j+(k-(n-1))]
    
    # Let's use s = k - (n-1) for s in [-(n-1), ..., n-1]
    # sens[j][s] = I(0 ≤ j-s ≤ n-1) * (1-h[j-s]) - I(0 ≤ j+s ≤ n-1) * h[j+s]
    
    # This is O(n^2) to compute all sensitivities
    sens = np.zeros((n, 2*n-1))
    for j in range(n):
        for s in range(-(n-1), n):
            idx = s + (n-1)
            val = 0.0
            if 0 <= j - s <= n - 1:
                val += (1.0 - h[j - s])
            if 0 <= j + s <= n - 1:
                val -= h[j + s]
            sens[j, idx] = val
    return sens
```

Wait, but the sensitivities change as h changes! After I update h[j], the sensitivities for other coordinates change. So I need to either:
(a) Recompute sensitivities after each update (O(n^2) per update, O(n^3) per sweep)
(b) Use the sensitivities computed at the start of the sweep (inexact coordinate descent)

Option (b) is simpler and still converges (it's just slower). Let me use that.

For n=120:

- Each 1D solve: O(n * grid_size) = 120 * 200 = 24000 operations
- Per sweep: 120 * 24000 = 2.88M operations
- Per second: probably ~100 million operations in numpy... actually the 1D solves can be vectorized.

Actually, let me vectorize the 1D solve:
- For coordinate j, with sensitivity sens[j] (length 2n-1) and current corr (length 2n-1):
  - For t in [0, 1]: obj(t) = max(corr + (t - h[j]) * sens[j])
  - We want to minimize this over t
  
  This is: min_t max_k (corr[k] + (t - h[j]) * sens[j][k])
  = min_t max_k ((corr[k] - h[j]*sens[j][k]) + t * sens[j][k])
  
  Let c_k = corr[k] - h[j]*sens[j][k] and b_k = sens[j][k]
  Then: min_t max_k (c_k + t*b_k) for t ∈ [0, 1]
  
  This is a 1D linear programming problem. The optimal t is where the upper envelope of the lines c_k + t*b_k is minimized.
  
  I can solve this by:
  1. Finding the t that minimizes the max
  2. Using the fact that the optimum is at an intersection of two lines or at a boundary
  
  Efficient method: 
  - Compute all intersection points t_i where line_i = line_j
  - Filter to t_i in [0, 1]
  - Evaluate the objective at each candidate point
  - Take the minimum
  
  But there are O(n^2) intersection points for 2n-1 lines, which is O(n^2) = O(57600) for n=120. That's fine.
  
  Actually, a simpler and faster approach: ternary search on t ∈ [0,1] since the function is convex (max of linear = convex).
  
  Ternary search: O(log(1/eps) * n) evaluations, each O(n). For eps=1e-6, that's about 20 * 240 = 4800 operations. Very fast!

Let me code this up properly:

```python
def solve_1d_convex(c, b):
    """Minimize max_k (c[k] + t * b[k]) over t in [0,1]"""
    def obj(t):
        return np.max(c + t * b)
    
    # Ternary search
    lo, hi = 0.0, 1.0
    for _ in range(60):  # enough for machine precision
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        if obj(m1) < obj(m2):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2
```

This is clean and fast. 60 iterations of ternary search gives precision ~ (2/3)^60 ≈ 2.7e-11, which is more than enough.

```python
def pixel_coordinate_descent(h, n_sweeps, time_budget):
    n = len(h)
    target = n / 2.0
    t0 = time.time()
    
    corr = np.correlate(h, 1.0 - h, mode="full")
    
    for sweep in range(n_sweeps):
        if time.time() - t0 > time_budget:
            break
        # Recompute sensitivities (using current h as approximation)
        sens = np.zeros((n, 2*n-1))
        for j in range(n):
            s_range = np.arange(-(n-1), n)
            # sens[j] at lag s: I(0<=j-s<=n-1)*(1-h[j-s]) - I(0<=j+s<=n-1)*h[j+s]
            valid1 = (j >= 0) & (j < n) & (j - s_range >= 0) & (j - s_range < n)
            valid2 = (j >= 0) & (j < n) & (j + s_range >= 0) & (j + s_range < n)
            sens[j, :] = np.where(valid1, 1.0 - h[j - s_range[valid1]], 0.0) - \
                         np.where(valid2, h[j + s_range[valid2]], 0.0)
        
        for j in range(n):
            if time.time() - t0 > time_budget - 1:
                break
            # 1D subproblem
            # minimize max_k (corr[k] + (t - h[j]) * sens[j][k]) for t in [0,1]
            c = corr - h[j] * sens[j]
            b = sens[j]
            t_opt = solve_1d_convex(c, b)
            
            # Update
            delta = t_opt - h[j]
            h[j] = t_opt
            corr += delta * sens[j]
        
        # Restore sum constraint
        h = fix_sum(h, target)
        corr = np.correlate(h, 1.0 - h, mode="full")
    
    return h
```

```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target):
    h = np.array(h, dtype=float)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return np.clip(h, 0.0, 1.0)
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = mask.sum()
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(obj, tau):
    x = tau * obj
    x_max = np.max(x)
    return float(x_max / tau + np.log(np.sum(np.exp(x - x_max))) / tau)

def solve_1d_convex(c, b):
    """Minimize max_k (c[k] + t*b[k]) for t in [0,1] using ternary search"""
    def obj(t):
        return np.max(c + t * b)
    lo, hi = 0.0, 1.0
    for _ in range(80):
        m1 = lo + (hi - lo) * 0.33333
        m2 = hi - (hi - lo) * 0.33333
        if obj(m1) <= obj(m2):
            hi = m2
        else:
            lo = m1
    return (lo + hi) * 0.5

def compute_sensitivities(h, n):
    """Compute sensitivity of each corr lag to each coordinate"""
    s_range = np.arange(-(n-1), n)
    sens = np.zeros((n, 2*n-1))
    for j in range(n):
        valid1 = (j - s_range >= 0) & (j - s_range < n)
        valid2 = (j + s_range >= 0) & (j + s_range < n)
        part1 = np.zeros(2*n-1)
        part2 = np.zeros(2*n-1)
        part1[valid1] = 1.0 - h[j - s_range[valid1]]
        part2[valid2] = h[j + s_range[valid2]]
        sens[j, :] = part1 - part2
    return sens

def coordinate_descent(h, time_budget):
    """Cyclical coordinate descent with exact 1D subproblems"""
    n = len(h)
    target = n / 2.0
    t0 = time.time()
    h = h.copy()
    
    corr = np.correlate(h, 1.0 - h, mode="full")
    best_c5 = float(np.max(corr) * (2.0/n))
    no_improve = 0
    
    while time.time() - t0 < time_budget:
        sens = compute_sensitivities(h, n)
        for j in range(n):
            if time.time() - t0 > time_budget - 0.5:
                break
            c = corr - h[j] * sens[j]
            b = sens[j]
            t_opt = solve_1d_convex(c, b)
            delta = t_opt - h[j]
            if abs(delta) > 1e-10:
                h[j] = t_opt
                corr = corr + delta * sens[j]
        
        # Restore constraint
        h = fix_sum(h, target)
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5 = float(np.max(corr) * (2.0/n))
        
        if c5 < best_c5 - 1e-8:
            best_c5 = c5
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 5:
            break
    
    return h, best_c5

        return budget_s - (time.time() - t0)
```

Wait, let me just use the definition from numpy docs:
out[k] = sum_n a[n] * conj(v[n-k])  (this is the cross-correlation definition)

Actually, numpy's correlate is defined as:
(C_a,v)[k] = sum_n a[n+k] * conj(v[n])

where the sum is over all valid n.

So for a = h, v = 1-h:
(C_h, 1-h)[k] = sum_n h[n+k] * (1-h[n])

The full output has k ranging from -(n-1) to (n-1), stored at indices 0 to 2n-2.
Index in output = k + (n-1) where k is the lag.

So output[idx] where idx = k + (n-1):
= sum_n h[n+k] * (1-h[n]) for valid n (0 ≤ n ≤ n-1 and 0 ≤ n+k ≤ n-1)

Now, if I change h[j] by δ:
- Terms where n+k = j, i.e., n = j-k: contribution is (1-h[j-k]) * δ, valid if 0 ≤ j-k ≤ n-1
- Terms where n = j: contribution is h[j+k] * (-δ), valid if 0 ≤ j+k ≤ n-1

So the change in output at index idx (lag k = idx - (n-1)):

```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target):
    h = np.array(h, dtype=float)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return np.clip(h, 0.0, 1.0)
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = mask.sum()
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(obj, tau):
    x = tau * obj
    x_max = np.max(x)
    return float(x_max / tau + np.log(np.sum(np.exp(x - x_max))) / tau)

def solve_1d(c, b):
    """Minimize max_k (c[k] + t*b[k]) for t in [0,1]"""
    lo, hi = 0.0, 1.0
    for _ in range(80):
        m1 = lo + (hi - lo) * 0.3333333
        m2 = hi - (hi - lo) * 0.3333333
        if np.max(c + m1 * b) <= np.max(c + m2 * b):
            hi = m2
        else:
            lo = m1
    return (lo + hi) * 0.5

def solve_1d_pen(c, b, sum_other, target, alpha):
    """Minimize max_k (c[k]+t*b[k]) + alpha*(sum_other+t-target)^2 for t in [0,1]"""
    lo, hi = 0.0, 1.0
    for _ in range(80):
        m1 = lo + (hi - lo) * 0.3333333
        m2 = hi - (hi - lo) * 0.3333333
        o1 = np.max(c + m1 * b) + alpha * (sum_other + m1 - target)**2
        o2 = np.max(c + m2 * b) + alpha * (sum_other + m2 - target)**2
        if o1 <= o2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) * 0.5

def compute_sens(h, n):
    """Sensitivity of each corr lag to each coordinate"""
    s = np.arange(-(n-1), n)
    out = np.zeros((n, 2*n-1))
    for j in range(n):
        v1 = (s >= j - (n-1)) & (s <= j)  # j-s in [0, n-1] means s in [j-(n-1), j]
        v2 = (s >= -j) & (s <= n - 1 - j)  # j+s in [0, n-1] means s in [-j, n-1-j]
        p1 = np.zeros(2*n-1)
        p2 = np.zeros(2*n-1)
        p1[v1] = 1.0 - h[j - s[v1]]
        p2[v2] = h[j + s[v2]]
        out[j] = p1 - p2
    return out

def coord_descent(h, t_budget, alpha=0.0):
    n = len(h)
    target = n / 2.0
    t0 = time.time()
    h = h.copy()
    best = evaluate_c5(h)
    
    while time.time() - t0 < t_budget:
        sens = compute_sens(h, n)
        corr = np.correlate(h, 1.0 - h, mode="full")
        for j in range(n):
            if time.time() - t0 > t_budget - 0.2:
                break
            c = corr - h[j] * sens[j]
            b = sens[j]
            if alpha > 0:
                so = h.sum() - h[j]
                t_new = solve_1d_pen(c, b, so, target, alpha)
            else:
                t_new = solve_1d(c, b)
            delta = t_new - h[j]
            h[j] = t_new
            corr = corr + delta * sens[j]

        if alpha == 0:
            h = fix_sum(h, target)
        c5 = evaluate_c5(h)
        if c5 < best:
            best = c5
    
    h = fix_sum(h, target)
    return h, evaluate_c5(h)

def slsqp_refine(h, tau, maxiter=400):
    n = len(h)
    target = n / 2.0
    bnds = Bounds(np.zeros(n), np.ones(n))
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")

                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return fix_sum(res.x, target)
    except Exception:
        return h
```
