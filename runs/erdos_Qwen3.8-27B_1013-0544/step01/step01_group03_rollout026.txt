import numpy as np
from scipy.optimize import minimize, Bounds
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def fix_sum(h, target):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = mask.sum()
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(vals, tau):
    x = tau * vals
    x_max = np.max(x)
    return float(x_max / tau + np.log(np.sum(np.exp(x - x_max))) / tau)

def grad_overlap_k(h, k):
    n = len(h)
    grad = np.zeros(n)
    if k >= 0:
        grad[k:] = 1.0
        grad[k:] -= h[:n - k]
        grad[:n - k] -= h[k:]
    else:
        kk = -k
        grad[:n - kk] = 1.0
        grad[:n - kk] -= h[kk:n]
        grad[kk:] -= h[0:n - kk]
    return grad

def softmax_weights(overlaps, temperature):
    x = overlaps / temperature
    x_max = np.max(x)
    w = np.exp(x - x_max)
    w /= w.sum()
    return w

def projected_step(h, grad, lr, target):
    h_new = h - lr * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    return fix_sum(h_new, target)

def active_gradient(h, overlaps, temp, top_frac=0.05):
    n = len(h)
    w = softmax_weights(overlaps, temp)
    n_active = max(5, int(0.05 * len(w)))
    top_idx = np.argsort(overlaps)[::-1][:n_active]
    w_top = w[top_idx]
    w_top /= w_top.sum()
    grad = np.zeros(n)
    for idx, wt in zip(top_idx, w_top):
        k = int(idx) - (n - 1)
        grad += wt * grad_overlap_k(h, k)
    return grad

def do_slsqp(h_start, tau, maxiter=300):
    n = len(h_start)
    target = n / 2.0
    bnds = Bounds(np.zeros(n), np.ones(n))
    
    def obj_fn(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        return smooth_max_val(corr, tau)
    
    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        xc = tau * corr
        xm = np.max(xc)
        w = np.exp(xc - xm)
        w /= w.sum()
        g = np.zeros(n)
        n_active = max(3, int(0.03 * len(w)))
        top_idx = np.argsort(corr)[::-1][:n_active]
        w_top = w[top_idx]
        w_top /= w_top.sum()
        for idx in top_idx:
            k = int(idx) - (n - 1)
            g += (tau * w_top[list(top_idx).index(idx)]) * grad_overlap_k(x, k)
        return g
    
    def con_fun(x):
        return np.sum(x) - target
    
    def con_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj_fn, h_start, jac=obj_grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        x = fix_sum(res.x, target)
        return x, evaluate_c5(x)
    except Exception:
        return h_start, None

def binary_hill_climb(h_bin, n, n_trials=3000, seed=0):
    np.random.seed(seed)
    best = h_bin.copy()
    best_c5 = evaluate_c5(best)
    ones = np.where(best == 1)[0]
    zeros = np.where(best == 0)[0]
    patience = 0
    for it in range(n_trials):
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        if i == j:
            continue
        best[i], best[j] = best[j], best[i]
        c5 = evaluate_c5(best)
        if c5 < best_c5:
            best_c5 = c5
            patience = 0
            ones = np.where(best == 1)[0]
            zeros = np.where(best == 0)[0]
        else:
            best[i], best[j] = best[j], best[i]
            patience += 1
            if patience > 2000:
                break
    return best, best_c5

    for phase in range(4):

                best_h = h_new.copy()
            h_cur = h_new

    # Phase 2: SLSQP with smooth max (various tau)
    for tau in [10, 30, 100, 300, 500]:

        x, c5 = do_slsqp(best_h, tau, 300)
        if c5 is not None and c5 < best_c5:

            best_c5 = c5

        h_back = np.interp(np.linspace(0, 1, n),

            best_c5 = c5_back
        # Local refine
        for tau in [50, 200]:

            x, c5 = do_slsqp(h_back, tau, 200)

                best_c5 = c5
            h_back = x

    # Phase 4: Multi-start with structured initializations
    if rem() > 60:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        starts = []
        # Sine-based starts
        for freq in [1, 2, 3, 5]:

            starts.append(h_s)
        # Random starts
        for _ in range(3):
            h_r = np.random.uniform(0, 1, n_cur)
            h_r = fix_sum(h_r, target_cur)
            starts.append(h_r)
        # Best and complement-like
        starts.append(best_h.copy())
        
        for h_start in starts:
            if rem() < 25:
                break
            for tau in [50, 200]:
                if rem() < 12:
                    break
                x, c5 = do_slsqp(h_start, tau, 200)
                if c5 is not None and c5 < best_c5:
                    best_h = x.copy()
                    best_c5 = c5

    # Phase 5: Active gradient again from new best (different temp)
    if rem() > 30:
        h_cur = best_h.copy()
        for temp in [0.005, 0.01, 0.001]:
            if rem() < 15:
                break
            for it in range(300):
                if rem() < 5:

                overlaps = compute_overlaps(h_cur)

                    best_c5 = c5_new

    # Phase 6: Final SLSQP fine-tuning

        for tau in [1000, 2000]:
            if rem() < 10:
                break
            x, c5 = do_slsqp(best_h, tau, 500)

                best_h = x.copy()
                best_c5 = c5

    best_h = fix_sum(best_h, len(best_h) / 2.0)
    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, len(best_h)
