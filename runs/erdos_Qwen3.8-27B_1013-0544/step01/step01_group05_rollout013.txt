    sh = _G_beta * (corr - mc)
    ev = np.exp(sh)
    w = ev / np.sum(ev)
    n = len(x)
    nh = n - 1
    g = np.zeros(n)
    for i in range(len(corr)):
        if w[i] < 1e-12:
            continue
        g += w[i] * grad_single(x, i - nh, _G_dx)
    return g

def _eq(x):
    return np.sum(x) - _G_target

def _obj_hard(x):
    return compute_c5(x, _G_dx)

def upsample(h, factor=2):
    n = len(h)
    out = np.empty(n * factor)
    out[0::factor] = h
    out[1::factor] = h
    return out

def lse_projected_grad(h, target, dx, beta, step, mom_arr, mu, n_iter, rng, seed_val):
    n = len(h)
    nh = n - 1
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    step_size = step
    momentum = mom_arr.copy()
    stag = 0
    b = beta
    for _ in range(n_iter):
        corr = get_corr(h, dx)
        mc = np.max(corr)
        sh = b * (corr - mc)
        ev = np.exp(sh)
        weights = ev / np.sum(ev)
        grad = np.zeros(n)
        for idx in range(len(corr)):
            w = weights[idx]
            if w < 1e-12:
                continue
            L = idx - nh
            grad += w * grad_single(h, L, dx)
        momentum = mu * momentum - step_size * grad
        h_new = project_feasible(h + momentum, target)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            stag = 0
        else:
            stag += 1
            if stag % 25 == 0:
                step_size *= 0.87
                b = min(b * 1.35, 400.0)
            if step_size < 1e-9:
                h = best_h + rng.normal(0, 0.02, n)
                h = project_feasible(h, target)
                momentum = np.zeros(n)
                step_size = step * 0.7
                stag = 0
        h = h_new
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    global _G_dx, _G_target, _G_beta
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return (start_time + budget_s - 3.0) - time.time()

    # Phase 1: Optimize at n=70
    n1 = 70
    dx1 = 2.0 / n1
    target1 = n1 / 2.0
    _G_dx = dx1
    _G_target = target1

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except NameError:
        h0 = np.full(n1, 0.5)
    h0 = project_feasible(h0, target1)
    best_h1 = h0.copy()
    best_c5_1 = compute_c5(h0, dx1)

    # LSE projected gradient at n=70
    if tleft() > 350:
        iters1 = int(min(2000, tleft() * 8))
        best_h1, best_c5_1 = lse_projected_grad(
            h0, target1, dx1, 20.0, 0.005, np.zeros(n1), 0.35, iters1, rng, seed)

    # SLSQP LSE at n=70
    if tleft() > 300:
        _G_beta = 250.0
        try:
            res = minimize(_obj_lse, best_h1, jac=_grad_lse, method='SLSQP',
                          bounds=[(0.0, 1.0)] * n1,
                          constraints=[{'type': 'eq', 'fun': _eq}],
                          options={'maxiter': 3000, 'ftol': 1e-15})
            if res.x is not None:
                hr = project_feasible(res.x, target1)
                c5r = compute_c5(hr, dx1)
                if c5r < best_c5_1:
                    best_c5_1 = c5r
                    best_h1 = hr
        except Exception:
            pass

    # Phase 2: Upsample to n=140 and optimize
    n2 = 140
    dx2 = 2.0 / n2
    target2 = n2 / 2.0
    _G_dx = dx2
    _G_target = target2

    h2 = upsample(best_h1, 2)
    h2 = project_feasible(h2, target2)
    best_h2 = h2.copy()
    best_c5_2 = compute_c5(h2, dx2)

    # LSE projected gradient at n=140
    if tleft() > 180:
        iters2 = int(min(3000, tleft() * 6))
        best_h2, best_c5_2 = lse_projected_grad(
            h2, target2, dx2, 50.0, 0.003, np.zeros(n2), 0.35, iters2, rng, seed)

    # SLSQP LSE at n=140
    if tleft() > 80:
        _G_beta = 350.0
        try:
            res = minimize(_obj_lse, best_h2, jac=_grad_lse, method='SLSQP',
                          bounds=[(0.0, 1.0)] * n2,
                          constraints=[{'type': 'eq', 'fun': _eq}],
                          options={'maxiter': 2500, 'ftol': 1e-15})
            if res.x is not None:
                hr = project_feasible(res.x, target2)
                c5r = compute_c5(hr, dx2)
                if c5r < best_c5_2:
                    best_c5_2 = c5r
                    best_h2 = hr
        except Exception:
            pass

    # Hard max fine-tuning at n=140
    if tleft() > 40:
        h = best_h2.copy()
        step_size = 0.0004
        momentum = np.zeros(n2)
        nh2 = n2 - 1
        while tleft() > 10:
            corr = get_corr(h, dx2)
            c_max = float(np.max(corr))
            tol = max(1e-6, c_max * 0.0002)
            active = np.where(corr >= c_max - tol)[0]
            if len(active) < 2:
                active = np.argsort(corr)[-2:]
            elif len(active) > 8:
                active = np.argsort(corr)[-8:]
            grad = np.zeros(n2)
            for idx in active:
                L = int(idx) - nh2
                grad += grad_single(h, L, dx2)
            grad /= float(len(active))
            momentum = 0.4 * momentum - step_size * grad
            h_new = project_feasible(h + momentum, target2)
            c5_new = compute_c5(h_new, dx2)
            if c5_new < best_c5_2:
                best_c5_2 = c5_new
                best_h2 = h_new.copy()
            else:
                step_size *= 0.92
                if step_size < 1e-11:
                    break
            h = h_new

    # Final SLSQP hard max
    if tleft() > 20:
        _G_dx = dx2
        _G_target = target2
        try:
            res = minimize(_obj_hard, best_h2, method='SLSQP',
                          bounds=[(0.0, 1.0)] * n2,
                          constraints=[{'type': 'eq', 'fun': _eq}],
                          options={'maxiter': 3000, 'ftol': 1e-15})
            if res.x is not None:
                hr = project_feasible(res.x, target2)
                c5r = compute_c5(hr, dx2)
                if c5r < best_c5_2:
                    best_c5_2 = c5r
                    best_h2 = hr
        except Exception:
            pass

    # Random perturbations
    if tleft() > 10:
        for _ in range(200):
            if tleft() < 2:
                break
            hp = best_h2 + rng.normal(0, 0.004, n2)
            hp = project_feasible(hp, target2)
            c5p = compute_c5(hp, dx2)
            if c5p < best_c5_2:
                best_c5_2 = c5p
                best_h2 = hp.copy()

    return (best_h2, best_c5_2, n2)
