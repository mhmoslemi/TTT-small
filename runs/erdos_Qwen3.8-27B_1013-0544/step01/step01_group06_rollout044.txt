def subgradient(h):
    """Compute subgradient of C5 at h (using exact argmax)"""
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    # Find the active set (lags with corr near max)
    max_corr = np.max(corr)
    # Use a small tolerance to include near-active lags
    active = np.where(corr >= max_corr - 0.01 * max_corr)[0]
    w = np.zeros(len(corr))
    w[active] = 1.0 / len(active)
    
    grad = np.zeros(n)
    for idx in active:
        k = idx - (n - 1)
        if k >= 0:
            grad[k:] += (1.0 - h[:n-k]) / len(active)
            if k > 0:
                grad[:n-k] -= h[k:] / len(active)
        else:
            p = -k
            grad[:n-p] += (1.0 - h[p:]) / len(active)
            if p > 0:
                grad[p:] -= h[:n-p] / len(active)
    return grad * (2.0 / n)

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_box_sum(h, target_sum):
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= s
    grad = np.zeros(n)
    for idx in range(len(corr)):
        wi = w[idx]
        if wi < 1e-10:
            continue
        k = idx - (n - 1)
        if k >= 0:
            grad[k:] += wi * (1.0 - h[:n - k])
            if k > 0:
                grad[:n - k] -= wi * h[k:]
        else:
            p = -k
            grad[:n - p] += wi * (1.0 - h[p:])
            if p > 0:
                grad[p:] -= wi * h[:n - p]
    return grad * (2.0 / n)

def sharp_haptic_optimize(h0, target_sum, deadline, rng, alpha_start=200, alpha_max=600):
    n = len(h0)
    h = project_box_sum(h0.copy(), target_sum)
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    lr = 0.006
    alpha = alpha_start
    beta = 0.9
    patience = 0
    stall = 0
    iters = 0

    while time.time() < deadline:
        y = h + lr * v
        g = smooth_max_grad(y, alpha)
        v = beta * v - lr * g
        h_new = project_box_sum(h + v / lr, target_sum) if lr > 0 else h
        h_new = project_box_sum(h + v, target_sum)
        c5 = compute_c5(h_new)
        iters += 1

        if c5 < best_c5 - 1e-14:
            best_h = h_new.copy()
            best_c5 = c5
            h = h_new
            patience = 0
            if iters % 20 == 0 and lr < 0.025:
                lr *= 1.03
        else:
            h = best_h.copy()
            v *= 0.5
            patience += 1
            lr *= 0.96
            if patience > 40:
                mode = stall % 3
                if mode == 0:
                    sigma = 0.08 + 0.04 * stall
                    h = project_box_sum(best_h + rng.randn(n) * sigma, target_sum)
                elif mode == 1:
                    h_sharp = project_box_sum(2.0 * best_h - 0.5, target_sum)
                    h = project_box_sum(h_sharp + rng.randn(n) * 0.05, target_sum)
                else:
                    h_flat = project_box_sum(0.5 + 0.5 * (best_h - 0.5), target_sum)
                    h = project_box_sum(h_flat + rng.randn(n) * 0.08, target_sum)
                v = np.zeros(n)
                alpha = min(alpha * 1.5, alpha_max)
                lr = 0.006
                patience = 0
                stall += 1
                if stall > 15:
                    break
        if iters % 50 == 0 and alpha < alpha_max:
            alpha = min(alpha + 10.0, alpha_max)

    return best_h, best_c5

def interpolate_resize(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, np.append(x_old, 2.0), np.append(h_old, h_old[0]))

def make_periodic_binary(n, period, rng):
    x = np.arange(n)
    h = np.where((x % period) < period // 2, 1.0, 0.0)
    h += rng.randn(n) * 0.03
    return project_box_sum(h, n / 2.0)

def make_sine_init(n, freq, rng):
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + 0.35 * np.sin(2.0 * np.pi * freq * x + rng.uniform(0, 2 * np.pi))
    return project_box_sum(h, n / 2.0)

    total_cand = len(candidates)
    per_cand = max(20, (d2_end - time.time()) / total_cand)

        if time.time() > d2_end - 5:

        cand = project_box_sum(cand, target)
        dl = min(d2_end, time.time() + per_cand)
        h_opt, c5_opt = sharp_haptic_optimize(cand, target, dl, rng, alpha_start=150 + 50 * (i % 4), alpha_max=500)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
            best_n = n

    # Phase 3: Try different resolutions (20% budget)
