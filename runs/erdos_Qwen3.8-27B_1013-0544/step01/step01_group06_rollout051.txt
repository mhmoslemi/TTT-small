def compute_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    s = np.sum(w)
    if s > 1e-300:
        w /= s
    else:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    grad = np.zeros(n)
    active = np.where(w > 1e-10)[0]
    for idx in active:
        k = int(idx - (n - 1))
        grad += w[idx] * grad_lag(h, k)
    grad *= (2.0 / n)
    return grad

def optimize_with_momentum(h_start, target_sum, deadline, alpha_init, lr, momentum, max_stall=5):
    n = len(h_start)
    h_curr = h_start.copy()
    v = np.zeros(n)
    best = h_start.copy()
    best_c5 = compute_c5(best)
    alpha = alpha_init
    patience = 0
    stall = 0
    rng_local = np.random.RandomState(int(time.time() * 10000) % 100000)
    
    while time.time() < deadline:
        grad = compute_grad(h_curr, alpha)
        v = momentum * v + (1 - momentum) * grad
        h_new = h_curr - lr * v
        h_new = project_simplex_box(h_new, target_sum)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5 - 1e-13:
            best = h_new
            best_c5 = c5_new
            h_curr = h_new
            patience = 0
            lr = min(lr * 1.02, 0.05)
            alpha = min(alpha * 1.01, 300.0)
        else:
            h_curr = h_new
            patience += 1
            lr *= 0.98
            if patience > 80:
                # Structured perturbation
                block_size = rng_local.randint(5, 20)
                start = rng_local.randint(0, n - block_size)
                h_curr = best + 0.0
                h_curr[start:start+block_size] += rng_local.randn(block_size) * 0.05
                h_curr = project_simplex_box(h_curr, target_sum)
                v = np.zeros(n)
                lr = lr * 0.5 if lr > 0.005 else 0.005
                alpha = alpha_init
                patience = 0
                stall += 1
                if stall >= max_stall:
                    break
    
    return best, best_c5

def binary_flip_search(h_bin, n_flips, seed=42):
    n = len(h_bin)
    h_curr = h_bin.copy()
    best_c5 = compute_c5(h_curr)
    ones = np.where(h_curr == 1)[0]
    zeros = np.where(h_curr == 0)[0]
    rng = np.random.RandomState(seed)
    
    for _ in range(n_flips):
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue
        h_curr[i] = 0
        h_curr[j] = 1
        c5_try = compute_c5(h_curr)
        if c5_try < best_c5:
            best_c5 = c5_try
            ones = np.where(h_curr == 1)[0]
            zeros = np.where(h_curr == 0)[0]
        else:
            h_curr[i] = 1
            h_curr[j] = 0
    return h_curr, best_c5

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')

def project_simplex_box(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:

def grad_lag(h, k):
