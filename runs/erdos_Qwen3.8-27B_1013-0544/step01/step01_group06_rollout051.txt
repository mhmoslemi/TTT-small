import numpy as np

def compute_c5_and_corr(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(corr) * (2.0 / n)
    return c5, corr

def project_onto_constraint(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum}."""
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def grad_smooth_max(h, alpha):
    """Compute gradient of smooth max of correlations * dx."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    s = np.sum(w)
    if s < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= s
    
    grad = np.zeros(n)
    active = np.where(w > 1e-10)[0]
    for idx in active:
        k = idx - (n - 1)
        wk = w[idx]
        if k >= 0:
            if k == 0:
                grad += wk * ((1 - h) - h)
            else:
                grad[k:] += wk * (1 - h[:n-k])
                grad[:n-k] -= wk * h[k:]
        else:

                grad += wk * ((1 - h) - h)
            else:
                grad[:n-p] += wk * (1 - h[p:])
                grad[p:] -= wk * h[:n-p]
    
    return grad * dx

    """Adam update with projection."""
    m_arr[:] = beta1 * m_arr + (1 - beta1) * grad
    v_arr[:] = beta2 * v_arr + (1 - beta2) * grad**2
    m_hat = m_arr / (1 - beta1**t)
    v_hat = v_arr / (1 - beta2**t)
    h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
    h_new = project_onto_constraint(h_new, target_sum)
    return h_new

def compute_c5_and_corr(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(corr) * (2.0 / n)
    return c5, corr

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def grad_smooth_max(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    s = np.sum(w)
    if s < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= s
    grad = np.zeros(n)
    active = np.where(w > 1e-10)[0]
    for idx in active:
        k = idx - (n - 1)
        wk = w[idx]
        if k >= 0:
            if k == 0:
                grad += wk * (1 - 2 * h)
            else:
                grad[k:] += wk * (1 - h[:n-k])
                grad[:n-k] -= wk * h[k:]
        else:
            p = -k
            if p == 0:
                grad += wk * (1 - 2 * h)
            else:
                grad[:n-p] += wk * (1 - h[p:])
                grad[p:] -= wk * h[:n-p]
    return grad * dx
