def binary_local_search(h_binary, n, dx, max_iter=100000):
    """Local search on binary sequences using swap moves."""
    ones = np.where(h_binary == 1)[0]
    zeros = np.where(h_binary == 0)[0]
    
    def evaluate(h):
        corr = np.correlate(h, 1.0-h, mode='full')
        return np.max(corr)
    
    best_corr = evaluate(h_binary)
    best_h = h_binary.copy()
    
    for it in range(max_iter):
        # Try all single swaps
        best_improvement = 0
        best_swap = None
        for i in range(len(ones)):
            for j in range(len(zeros)):
                # Swap: set ones[i] to 0, set zeros[j] to 1
                h_test = best_h.copy()
                h_test[ones[i]] = 0
                h_test[zeros[j]] = 1
                corr = evaluate(h_test)
                improvement = best_corr - corr
                if improvement > best_improvement:
                    best_improvement = improvement
                    best_swap = (i, j)
        
        if best_swap is None:
            break  # Local minimum
        
        i, j = best_swap
        best_h[ones[i]] = 0
        best_h[zeros[j]] = 1
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        best_corr = best_corr - best_improvement
    
    return best_h

def compute_c5(h, dx):

def project_feasible(h, n):
    target = n / 2.0
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - target
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            h[free] -= diff / np.sum(free)
        else:
            h -= diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = np.sum(h) - target
    if abs(diff) > 1e-12:

        if np.any(free):
            h[free] -= diff / np.sum(free)
            h = np.clip(h, 0.0, 1.0)

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    for _ in range(200):

        diff = np.sum(h) - target

def smooth_max_grad(h, dx, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = alpha * corr * dx
    max_s = np.max(scaled)
    w = np.exp(scaled - max_s)
    w /= np.sum(w)
    
    grad = np.zeros(n)
    for k in range(len(w)):
        if w[k] < 1e-12:
            continue
        offset = k  # c[k] = sum_i h[i+k] * (1-h[i])
        for m in range(n):
            g = 0.0
            # h[m] appears as h[i+k] when i = m-k, need 0<=m-k<n and 0<=(m-k)+k<n=0<=m<n
            if 0 <= m - offset < n:
                g += (1.0 - h[m - offset])
            # h[m] appears in (1-h[i]) when i = m, need 0<=m<n and 0<=m+offset<n
            if 0 <= m + offset < n:
                g += (-h[m + offset])
            grad[m] += w[k] * g
    return grad * dx

        # Adaptive learning rate
