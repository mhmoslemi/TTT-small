        val = logsumexp_c5(h, dx, T)
        val += lambda_ * (np.sum(h) - target_sum) + 0.5 * mu * (np.sum(h) - target_sum)**2
        return val

    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - target
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            h[free] -= diff / np.sum(free)
        else:
            h -= diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = np.sum(h) - target
    if abs(diff) > 1e-14:
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            h[free] -= diff / np.sum(free)
            h = np.clip(h, 0.0, 1.0)
    # Force exact sum
    diff = np.sum(h) - target
    if abs(diff) > 1e-12:
        idx = np.argsort(h)[n//2]  # middle value
        h[idx] -= diff
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_c5(h, dx, T):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(T * (corr - m)))) / T

def lse_c5_grad(h, dx, T):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    vals = T * corr
    vals -= np.max(vals)
    w = np.exp(vals)
    w /= np.sum(w)
    
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        if w[j] < 1e-14:
            continue
        shift = j - (n - 1)
        m1 = max(0, -shift)
        m2 = min(n, n - shift)
        if m2 > m1:
            grad[m1:m2] += w[j] * (1.0 - h[m1 + shift:m2 + shift])
        m1 = max(0, shift)
        m2 = min(n, n + shift)
        if m2 > m1:
            grad[m1:m2] -= w[j] * h[m1 - shift:m2 - shift]
    grad *= dx
    return grad

        for i in range(n):
            x[i] += eps
            c1 = compute_c5(x, dx)
            x[i] -= eps
            g[i] = (c1 - c0) / eps
        return g

                   bounds=bounds, constraints=constraint,
                   options={'maxiter': max_iter, 'ftol': 1e-15})

Wait, I have a problem. The numerical gradient in `slsqp_polish` is O(n) function evaluations per gradient, and each function evaluation is O(n²) for the correlation. So each gradient is O(n³), and for n=90 that's about 730,000 operations per gradient. With SLSQP doing maybe 100 iterations, that's 73 million operations. Should be manageable within the time budget.

But actually, I can do better. Let me compute the analytical subgradient of the max function. The subgradient of max_k c_k at a point where multiple k are at the max is the average of the gradients of the active k's. I can identify the active set and compute the exact gradient.

Let me add that:

```python
def max_c5_grad(h, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    cmax = np.max(c)
    # Active set: all k where c_k is within epsilon of max
    active = np.where(c >= cmax - 1e-10)[0]
    grad = np.zeros(n)
    for j in active:
        shift = j - (n - 1)
        m1 = max(0, -shift)
        m2 = min(n, n - shift)
        if m2 > m1:
            grad[m1:m2] += (1.0 - h[m1 + shift:m2 + shift])
        m1 = max(0, shift)
        m2 = min(n, n + shift)
        if m2 > m1:
            grad[m1:m2] -= h[m1 - shift:m2 - shift]
    grad /= len(active)
    grad *= dx
    return grad
```

This is much faster - O(n * |active|) which is typically O(n²) since |active| is usually small.
