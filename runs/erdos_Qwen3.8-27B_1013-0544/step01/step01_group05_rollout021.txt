import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(x, target_sum):
    lo = float(x.min()) - 1.0
    hi = float(x.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = float(np.clip(x - mid, 0.0, 1.0).sum())
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def smooth_max(obj, beta):
    m = np.max(obj)
    return m + np.log(np.sum(np.exp(obj - m))) / beta

def smooth_max_grad(obj, beta):
    m = np.max(obj)
    shifted = obj - m
    weights = np.exp(shifted * beta) / np.sum(np.exp(shifted * beta))
    return m + np.log(np.sum(np.exp(shifted))) / beta, weights

def compute_c5_and_grad_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    obj = corr * dx
    m = np.max(obj)
    shifted = obj - m
    exp_shifted = np.exp(shifted * beta)
    weights = exp_shifted / np.sum(exp_shifted)
    val = m + np.log(np.sum(exp_shifted)) / beta
    
    # Gradient of smooth_max(corr * dx) wrt h
    # d/dh smooth_max = dx * sum_k weights[k] * d(corr[k])/dh
    # corr[k] = sum_j h[j] * (1-h[j+k]) for valid j
    # d(corr[k])/dh[j] = (1-h[j+k]) - h[j] if j+k valid, else ...
    
    n = len(h)
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        if weights[k_idx] < 1e-10:
            continue
        L = k_idx - (n - 1)  # shift
        if L == 0:
            # corr = sum_j h[j]*(1-h[j]), d/dh[j] = 1-2h[j]
            grad += weights[k_idx] * dx * (1.0 - 2.0 * h)
        elif L > 0:
            # corr = sum_{j=0}^{n-1-L} h[j]*(1-h[j+L])

            j_lo = slice(0, n - L)
            j_hi = slice(n - L, n)
            grad[j_lo] += weights[k_idx] * dx * (1.0 - h[j_hi])
            grad[j_hi] += weights[k_idx] * dx * (-h[j_lo])

            # d/dh[j] for j in [0, L2-1]: -h[j+L2]
            j_lo = slice(0, L2)
            j_hi = slice(L2, n)
            grad[j_hi] += weights[k_idx] * dx * (1.0 - h[j_lo])
            grad[j_lo] += weights[k_idx] * dx * (-h[j_hi])
    
    return val, grad

def active_set_grad(h, dx, tol_frac=0.01, max_active=10, min_active=2):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    tol = max(1e-6, c_max * tol_frac)
    active = np.where(corr >= c_max - tol)[0]
    if len(active) < min_active:
        active = np.argsort(corr)[-min_active:]
    elif len(active) > max_active:
        active = np.argsort(corr)[-max_active:]
    
    grad = np.zeros(n)
    for k_idx in active:
        L = k_idx - (n - 1)
        if L == 0:
            grad += (1.0 - 2.0 * h)
        elif L > 0:
            j_lo = slice(0, n - L)
            j_hi = slice(n - L, n)
            grad[j_lo] += (1.0 - h[j_hi])
            grad[j_hi] += (-h[j_lo])
        else:
            L2 = -L
            j_lo = slice(0, L2)
            j_hi = slice(L2, n)
            grad[j_hi] += (1.0 - h[j_lo])
            grad[j_lo] += (-h[j_hi])
    grad /= len(active)
    return grad * dx

def project_feasible(x, target_sum):
    lo = float(x.min()) - 1.0

def smooth_max_obj_and_grad(h, dx, beta):

For each coordinate j, given all other coordinates fixed, find the optimal value of h[j]. Since the objective is max_k f_k(h) where each f_k is bilinear in h (one factor is h[j], the other depends on j and k), the optimal h[j] can be found by checking the breakpoints.

For a fixed k (shift L), f_k as a function of h[j]:
- If L = 0: f_k = ... + h[j](1-h[j]) + ... = -h[j]² + h[j] + const. Maximum at h[j] = 1/2.
- If L > 0 and j < n-L: f_k = ... + h[j](1-h[j+L]) + ... = h[j] + const (linear in h[j]!). 
  Wait, that's if j+L is valid. The coefficient of h[j] is (1-h[j+L]).
- If L > 0 and j ≥ n-L: f_k = ... + h[j-L](1-h[j]) + ... = -h[j]*h[j-L] + const (linear in h[j] with coefficient -h[j-L]).
- Similarly for L < 0.

So for each individual f_k, the dependence on h[j] is either linear or quadratic (for k=0). The max of these is a piecewise linear (or piecewise quadratic) function of h[j].

For coordinate descent, I could:
1. Fix all h except h[j]
2. Find h[j] that minimizes max_k f_k(h)
3. This is a 1D problem: minimize a piecewise-linear function over [0,1]

The minimizer of a max of linear functions is at a "kink" point, which is where two of the linear functions cross. So I can find all the kink points, evaluate the objective at each, and pick the best.

def project_feasible(x, target_sum):
    lo = float(x.min()) - 1.0
    hi = float(x.max()) + 1.0

        if s > target_sum:
            lo = mid
        else:
            hi = mid

def active_set_grad(h, dx, tol_frac, max_active):
    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode='full')
    obj_vals = corr * dx
    c_max = np.max(obj_vals)
    tol = max(1e-8, c_max * tol_frac)
    active = np.where(obj_vals >= c_max - tol)[0]
    if len(active) < 2:
        active = np.argsort(obj_vals)[-2:]
    elif len(active) > max_active:
        active = np.argsort(obj_vals)[-max_active:]
    
    grad = np.zeros(n)
