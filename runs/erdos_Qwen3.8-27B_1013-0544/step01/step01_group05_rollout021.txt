    other_sum = sum of all coordinates except j"""

    n_half = n - 1

    B = np.zeros(2*n - 1)

        # Replacing h[j] with variable x:
        # A_k = sum of terms not involving h[j]
        # B_k = coefficient of h[j]

        # - j is a valid first index: j in valid range for L
        # - j-L is a valid index (j appears as second factor)

                # Actually: the full term at i=j is h[j]*(1-h[j+L]) = x*(1-h[j+L])
                # So B_k += (1-h[j+L]) and A_k doesn't get this term
            else:
                pass
            # h[j] as second factor: i+L = j, so i = j-L, valid if 0 <= j-L <= n-L-1
            if 0 <= j - L <= n - 1 - L:

            for i in valid_first:

                    A[k] += h[i] * (1 - h[i + L])

            # h[j] as first factor: j in [L2, n-1]
            if L2 <= j <= n - 1:
