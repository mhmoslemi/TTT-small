import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) > 1e-12:
        h += (target_sum - s) / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def lag_grad(h, lag, n):
    """Gradient of correlation at given lag w.r.t. h. O(n)"""
    g = np.zeros(n)
    if lag >= 0:
        # Valid i: 0 <= i < n-lag, and 0 <= i+lag < n
        m = n - lag
        if m > 0:
            g[:m] += (1.0 - h[lag:])      # d/dh[i+lag]
            g[lag:] -= h[:m]               # d/dh[i]
    else:
        lp = -lag
        m = n - lp
        if m > 0:
            g[lp:] += (1.0 - h[:m])        # d/dh[i]  
            g[:m] -= h[lp:]                 # d/dh[i+lp]
    return g

    """Gradient of smooth_max(corr, beta) * dx w.r.t. h"""

        g += w[k] * lag_grad(h, lag, n)
    return g * (2.0 / n)

    w = e / e.sum()
    g = np.zeros(n)
    for idx in range(2 * n - 1):
        lag = idx - (n - 1)
        g += w[idx] * lag_grad(h, lag, n)
    g *= dx
    if sum_penalty > 0:
        s = h.sum() - n / 2.0
        obj_val += sum_penalty * s * s
        g += 2.0 * sum_penalty * s
    return obj_val, g

def optimize_local(h, n, beta, sum_pen, maxiter, gtol):
    """L-BFGS-B local optimization with sum penalty."""
    bounds = Bounds(np.zeros(n), np.ones(n))
    res = minimize(lambda x: smooth_max_obj_grad(x, beta, n, sum_pen)[0],
                   h, jac=lambda x: smooth_max_obj_grad(x, beta, n, sum_pen)[1],
                   method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'gtol': gtol})
    return project_h(res.x, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    def time_left():
        return budget_s - (time.time() - start_time)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
        n = len(best_h)
    else:
        n = 70
        best_h = np.full(n, 0.5)

    best_h = project_h(best_h, n / 2.0)
    best_c5 = compute_c5(best_h)

    # Phase 1: Beta annealing with increasing precision
    anneal_schedule = [(5, 0, 300, 1e-8), (15, 1, 300, 1e-9), (50, 10, 400, 1e-10),
                       (150, 100, 500, 1e-10), (500, 500, 500, 1e-11), (2000, 1000, 800, 1e-12)]
    for beta, sp, miter, gt in anneal_schedule:
        if time_left() < 40:
            break
        try:
            h_new = optimize_local(best_h, n, beta, sp, miter, gt)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new
        except Exception:
            pass

    # Phase 2: Diversification with lag-targeted perturbations
    no_improve = 0
    max_stall = 40
    while time_left() > 40 and no_improve < max_stall:
        h0 = best_h.copy()
        corr = np.correlate(h0, 1.0 - h0, mode="full")
        top_idx = np.argsort(corr)[-5:]

        # Subgradient steps on top lags
        for idx in top_idx:
            lag = idx - (n - 1)
            g = lag_grad(h0, lag, n)
            norm = np.max(np.abs(g)) + 1e-12
            h0 -= 0.01 * g / norm
            h0 = project_h(h0, n / 2.0)

        # Structured random perturbation
        strategy = np.random.randint(5)
        if strategy == 0:
            mask = np.random.random(n) < 0.2
            h0[mask] += np.random.randn(int(mask.sum())) * 0.05
        elif strategy == 1:
            idx_a = np.random.choice(n, n // 4, replace=False)
            idx_b = np.random.permutation(idx_a)
            h0[idx_a], h0[idx_b] = h0[idx_b].copy(), h0[idx_a].copy()
        elif strategy == 2:
            h0 += np.random.randn(n) * 0.03
        elif strategy == 3:
            # Shift part of the function
            shift = np.random.randint(2, 5)
            h0 = np.roll(h0, shift)
            h0[:shift] = np.random.random(shift) * 0.3 + 0.3
        else:
            # Mirror perturbation
            h0[:n//2] += 0.02 * h0[n//2:]
            h0[n//2:] -= 0.02 * h0[:n//2]

        h0 = project_h(h0, n / 2.0)

        # Polish with moderate beta
        try:
            h_new = optimize_local(h0, n, 300, 500, 300, 1e-11)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new
                no_improve = 0
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

    # Phase 3: High-precision final polish
    for beta, sp in [(5000, 5000), (10000, 10000)]:
        if time_left() < 15:
            break
        try:
            h_new = optimize_local(best_h, n, beta, sp, 1000, 1e-13)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new
        except Exception:
            pass

    # Phase 4: Try different n if time permits
    if time_left() > 60:
        for n_try in [50, 100]:
            if time_left() < 30:
                break
            h_try = np.full(n_try, 0.5)
            # Initialize with a structured pattern
            x = np.linspace(0, 2, n_try)
            h_try = 0.5 + 0.3 * np.sin(3 * np.pi * x / 2)
            h_try = project_h(h_try, n_try / 2.0)
            c5_try = compute_c5(h_try)

            for beta, sp, miter in [(20, 10, 300), (200, 200, 400), (1000, 2000, 500)]:
                if time_left() < 15:
                    break
                try:
                    h_try = optimize_local(h_try, n_try, beta, sp, miter, 1e-11)
                    c5_try = compute_c5(h_try)
                except Exception:
                    break
            if c5_try < best_c5:
                best_h, best_c5, n = h_try, c5_try, n_try

    return best_h, best_c5, n
