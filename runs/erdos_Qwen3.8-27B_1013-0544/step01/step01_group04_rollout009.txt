def smooth_max_gradient(h, n, tau=0.01):

    dx = 2.0 / n

    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    for _ in range(50):
        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - target
        if abs(diff) < 1e-14:
            break
