Let me try yet another approach: **Subgradient method with randomized smoothing**.

For the max function, we can use the softmax approximation:
max_k g_k ≈ (1/β) * log(sum_k exp(β * g_k))

For large β, this approximates the max. We can then use gradient descent on this smooth approximation.

The gradient of (1/β) * log(sum_k exp(β * g_k)) with respect to h is:
sum_k p_k * grad(g_k) where p_k = exp(β*g_k) / sum_j exp(β*g_j)

This is a convex combination of the gradients of individual g_k, weighted by their "importance" (softmax weights). This is actually a better approach than the active-set method!

Let me implement this:

```python
def softmax_grad(h, dx, beta):
    n = len(h)
    n_half = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    g = corr * dx  # all lag values
    
    # Softmax weights
    g_shifted = g - g.max()  # for numerical stability
    weights = np.exp(beta * g_shifted)
    weights /= weights.sum()
    
    # Gradient is weighted sum of individual gradients
    grad = np.zeros(n)
    for k in range(len(g)):
        if weights[k] > 1e-10:
            L = k - n_half
            grad += weights[k] * compute_grad_for_lag(h, L)
    
    return grad, g.max() * dx
```

import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(x, target_sum):
    lo = float(x.min()) - 1.0
    hi = float(x.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = float(np.clip(x - mid, 0.0, 1.0).sum())
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def grad_lag(h, L):
    n = len(h)
    L = abs(L)
    if L == 0:
        return 1.0 - 2.0 * h
    g = np.zeros(n)
    g[L:] += 1.0 - h[:n - L]
    g[:n - L] += -h[L:]
    return g

def softmax_gradient(h, nh, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    g = corr
    gm = float(g.max())
    gs = g - gm
    w = np.exp(beta * gs)
    ws = float(w.sum())
    if ws < 1e-30:
        w = np.zeros(len(g))
        w[int(np.argmax(g))] = 1.0
    else:
        w /= ws
    gr = np.zeros(n)
    for k in range(len(g)):
        if w[k] > 1e-9:
            gr += w[k] * grad_lag(h, k - nh)
    return gr

def active_set_gradient(h, nh, tol_frac=0.001, tol_abs=0.005, max_act=10):
    corr = np.correlate(h, 1.0 - h, mode='full')
    cm = float(corr.max())
    tol = max(tol_abs, cm * tol_frac)
    act = np.where(corr >= cm - tol)[0]
    if len(act) < 2:
        act = np.argsort(corr)[-2:]
    elif len(act) > max_act:
        act = np.argsort(corr)[-max_act:]
    gr = np.zeros(len(h))
    for idx in act:
        gr += grad_lag(h, int(idx) - nh)
    gr /= float(len(act))
    return gr

def _slsqp_obj(x, dx):
    return compute_c5(x, dx)

def _slsqp_con(x, ts):
    return np.sum(x) - ts
