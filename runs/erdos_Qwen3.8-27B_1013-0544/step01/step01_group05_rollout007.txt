    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def lag_grad(h, L):
    n = len(h)
    omh = 1.0 - h
    g = np.zeros(n)
    if L == 0:
        g[:] = omh[:] - h[:]
    elif L > 0:
        g[L:] += omh[:n - L]
        g[:n - L] -= h[L:]
    else:
        L2 = -L
        g[:n - L2] += omh[L2:]
        g[L2:] -= h[:n - L2]
    return g

def smooth_obj(x):
    dx = _global_dx
    T = _global_T
    corr = np.correlate(x, 1.0 - x, mode='full')
    c = corr * dx
    cm = float(np.max(c))
    return cm + float(np.log(np.sum(np.exp(T * (c - cm)))) / T)

def smooth_grad(x):
    dx = _global_dx
    T = _global_T
    n = len(x)
    n_half = n - 1
    corr = np.correlate(x, 1.0 - x, mode='full')
    c = corr * dx
    cm = float(np.max(c))
    w = np.exp(T * (c - cm))
    w /= w.sum()
    grad = np.zeros(n)
    for p in range(len(corr)):
        if w[p] < 1e-10:
            continue
        L = p - n_half
        grad += w[p] * lag_grad(x, L)
    return grad * dx

def eq_fun(x):
    return np.sum(x) - _global_target

def eq_grad(x):
    return np.ones(len(x))

def active_grad(h, dx, n, top_k=10):
    n_half = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    tol = max(1e-5, c_max * 3e-4)
    active = np.where(corr >= c_max - tol)[0]
    if len(active) < 3:
        active = np.argsort(corr)[-max(3, top_k // 2):]
    elif len(active) > top_k:
        active = np.argsort(corr)[-top_k:]
    grad = np.zeros(n)
    for idx in active:
        L = int(idx) - n_half
        grad += lag_grad(h, L)
    return grad / float(len(active))

def run(seed=42, budget_s=1100, **kwargs):
    global _global_dx, _global_n, _global_target, _global_T
    start_time = time.time()
    rng = np.random.RandomState(seed)

    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    _global_dx = dx
    _global_n = n_points
    _global_target = target_sum

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except NameError:
        h0 = np.full(n_points, 0.5)

    h0 = project_feasible(h0, target_sum)
    best_h = h0.copy()
    best_c5 = compute_c5(h0, dx)

    def tleft():
        return start_time + budget_s - 5.0 - time.time()

    # Phase 1: Smooth max SLSQP at increasing temperatures
    temps = [5, 15, 40, 100, 300, 800, 2000, 6000]
    for T in temps:
        if tleft() < 90:
            break
        _global_T = T
        x0 = best_h.copy()
        try:
            res = minimize(smooth_obj, x0, jac=smooth_grad, method='SLSQP',
                          bounds=[(0.0, 1.0)] * n_points,
                          constraints=[{'type': 'eq', 'fun': eq_fun, 'jac': eq_grad}],
                          options={'maxiter': 1500, 'ftol': 1e-14})
            if res.x is not None:
                h_new = project_feasible(res.x, target_sum)
                c5_new = compute_c5(h_new, dx)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new
        except Exception:
            pass

    # Phase 2: Active-set projected gradient with momentum
    if tleft() > 80:
        h = best_h.copy()
        step_size = 0.003
        momentum = np.zeros(n_points)
        mu = 0.65
        stag = 0
        while tleft() > 70:
            grad = active_grad(h, dx, n_points, top_k=12)
            momentum = mu * momentum - step_size * grad
            h_new = project_feasible(h + momentum, target_sum)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                stag = 0
            else:
                stag += 1
                if stag % 40 == 0:
                    step_size *= 0.88
                if step_size < 1e-9:
                    h = best_h + rng.normal(0, 0.03, n_points)
                    h = project_feasible(h, target_sum)
                    momentum = np.zeros(n_points)
                    step_size = 0.002
                    stag = 0
            h = h_new

    # Phase 3: Diverse starts with local refinement
    if tleft() > 50:
        starts = []
        # Cosine patterns
        x_grid = np.linspace(0, 2, n_points)
        for freq in [1, 2, 3, 5]:
            for phase in [0, np.pi / 2, np.pi]:
                h_s = 0.5 + 0.35 * np.cos(np.pi * freq * x_grid + phase)
                h_s = np.clip(h_s, 0, 1)
                h_s = project_feasible(h_s, target_sum)
                starts.append(h_s)
        # Block / pulse patterns
        for bw in [8, 14, 20, 30]:
            h_b = np.zeros(n_points)
            c = n_points // 2
            h_b[c - bw:c + bw] = 1.0
            h_b = project_feasible(h_b, target_sum)
            starts.append(h_b)
        # Random smoothed
        for _ in range(4):
            h_r = rng.rand(n_points)
            h_r = np.convolve(h_r, np.ones(7) / 7, mode='same')
            h_r = project_feasible(h_r, target_sum)
            starts.append(h_r)

        for s in starts:
            if tleft() < 15:
                break
            h = s.copy()
            ss = 0.002
            mom = np.zeros(n_points)
            for _ in range(400):
                if tleft() < 5:
                    break
                grad = active_grad(h, dx, n_points, top_k=6)
                mom = 0.5 * mom - ss * grad
                h_new = project_feasible(h + mom, target_sum)
                c5_new = compute_c5(h_new, dx)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                elif c5_new > compute_c5(h, dx) + 5e-8:
                    ss *= 0.85
                    if ss < 1e-11:
                        break
                h = h_new

    # Phase 4: SLSQP direct on piecewise-linear max (active set)
    if tleft() > 40:
        h = best_h.copy()
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = float(np.max(corr))
        tol = max(5e-4, c_max * 1e-3)
        active = np.where(corr >= c_max - tol)[0]
        if len(active) > 15:
            active = np.argsort(corr)[-15:]
        active_list = [int(a) - (n_points - 1) for a in active]

        def pw_obj(x):
            c = np.array([float(np.dot(lag_grad(x, L), x)) * dx +
                         float(np.dot(x, 1.0 - x)) * dx - float(lag_grad(x, L).dot(x)) * dx
                         for L in active_list])
            # Simpler: just compute actual c values
            corr2 = np.correlate(x, 1.0 - x, mode='full')
            n_half = len(x) - 1
            vals = [corr2[L + n_half] * dx for L in active_list if -n_half <= L <= n_half]
            return float(np.max(vals))

        def pw_grad(x):
            n = len(x)
            g = np.zeros(n)
            for L in active_list:
                g += lag_grad(x, L)
            return g * dx / len(active_list)

        try:
            res = minimize(pw_obj, h, jac=pw_grad, method='SLSQP',
                          bounds=[(0.0, 1.0)] * n_points,
                          constraints=[{'type': 'eq', 'fun': eq_fun, 'jac': eq_grad}],
                          options={'maxiter': 2000, 'ftol': 1e-14})
            if res.x is not None:
                h_new = project_feasible(res.x, target_sum)
                c5_new = compute_c5(h_new, dx)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new
        except Exception:
            pass

    # Phase 5: Ultra-fine active-set
    if tleft() > 10:
        h = best_h.copy()
        ss = 0.0001
        for _ in range(3000):
            if tleft() < 1:
                break
            grad = active_grad(h, dx, n_points, top_k=5)
            h_new = project_feasible(h - ss * grad, target_sum)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            else:
                ss *= 0.93
                if ss < 1e-13:
                    break
            h = h_new

    return (best_h, best_c5, n_points)
