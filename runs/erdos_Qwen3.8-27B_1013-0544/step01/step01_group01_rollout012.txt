    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    # Smooth max
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    w = exps / exps.sum()  # length 2n-1
    
    val = (m + np.log(exps.sum()) / beta) * dx

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val(corr, beta):
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta)

def proj_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_grad_lbfgs(h, n, dx, target_sum, beta, penalty):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    w = exps / exps.sum()
    val = (m + np.log(exps.sum()) / beta) * dx + penalty * (h.sum() - target_sum)**2
    # gradient
    j_idx = np.arange(n)
    u_idx = np.arange(n)
    idx1 = (n - 1) + j_idx[:, None] - u_idx[None, :]
    mask1 = (idx1 >= 0) & (idx1 < 2*n - 1)
    w1 = np.where(mask1, w[np.clip(idx1, 0, 2*n-2)], 0.0)
    sum1 = w1 @ (1.0 - h)
    idx2 = (n - 1) + u_idx[None, :] - j_idx[:, None]
    mask2 = (idx2 >= 0) & (idx2 < 2*n - 1)
    w2 = np.where(mask2, w[np.clip(idx2, 0, 2*n-2)], 0.0)
    sum2 = w2 @ h
    grad = dx * (sum1 - sum2) + 2 * penalty * (h.sum() - target_sum)
    return val, grad

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    def time_left():
        return budget_s - (time.time() - start_time)
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
        best_h = h_init.copy()
        best_c5 = compute_c5(h_init)
        best_n = n_init
    
    # n values to try
    n_list = sorted(set([best_n if best_n > 0 else 70, 80, 100, 120, 140, 160, 200, 250]))
    
    total_budget_per_n = budget_s / len(n_list)
    
    for n_points in n_list:
        alloc = min(total_budget_per_n * 1.5, time_left() - 10)
        if alloc < 15:
            continue
        
        dx = 2.0 / n_points
        target_sum = n_points / 2.0
        
        # Initializations
        inits = []
        
        # Upscale from best
        if best_h is not None and len(best_h) != n_points:
            from scipy.interpolate import interp1d
            old_x = np.linspace(0, 2, len(best_h), endpoint=False)
            new_x = np.linspace(0, 2, n_points, endpoint=False)
            h_up = interp1d(old_x, best_h, kind='linear', fill_value='extrapolate')(new_x)
            inits.append(proj_h(h_up, target_sum))

        inits.append(np.full(n_points, 0.5))

        h_block = np.tile([1.0, 0.0], n_points // 2 + 1)[:n_points]
        inits.append(proj_h(h_block, target_sum))

        for freq in [3, 5, 7, 11]:
            x = np.linspace(0, 2, n_points, endpoint=False)
            h_s = 0.5 + 0.45 * np.sin(np.pi * freq * x)
            inits.append(proj_h(h_s, target_sum))
        
        # Random
        inits.append(proj_h(np.random.random(n_points), target_sum))
        
        # Optimize each initialization with L-BFGS-B
        penalty = 500.0
        for init_i, h0 in enumerate(inits):
            if time_left() < 10:
                break
            remain_for_n = alloc - (time.time() - start_time)  # not quite right
            if time.time() - start_time > budget_s - 20:
                break

                if time.time() - start_time > budget_s - 15:
                    break

                    xc = np.clip(x, 1e-10, 1 - 1e-10)
                    v, _ = obj_grad_lbfgs(xc, n_points, dx, target_sum, beta, penalty)
                    return v

                corr = np.correlate(xc, 1.0-xc, mode='full')

    return best_h, best_c5, best_n

import numpy as np
from scipy.optimize import minimize
import time

_con_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val(corr, beta):
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta

def proj_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def _con_fun(x):
    return x.sum() - _con_target

def _con_jac(x):
    return np.ones(len(x))

def obj_grad_lbfgs(h, n, dx, target_sum, beta, penalty):
    hc = np.clip(h, 1e-12, 1.0 - 1e-12)
    corr = np.correlate(hc, 1.0 - hc, mode='full')
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    w = exps / exps.sum()
    val = (m + np.log(exps.sum()) / beta) * dx + penalty * (hc.sum() - target_sum) ** 2
    j_idx = np.arange(n)
    u_idx = np.arange(n)
    idx1 = (n - 1) + j_idx[:, None] - u_idx[None, :]
    mask1 = (idx1 >= 0) & (idx1 < 2 * n - 1)
    w1 = np.where(mask1, w[np.clip(idx1, 0, 2 * n - 2)], 0.0)
    sum1 = w1 @ (1.0 - hc)
    idx2 = (n - 1) + u_idx[None, :] - j_idx[:, None]
    mask2 = (idx2 >= 0) & (idx2 < 2 * n - 1)
    w2 = np.where(mask2, w[np.clip(idx2, 0, 2 * n - 2)], 0.0)
    sum2 = w2 @ hc
    grad = dx * (sum1 - sum2) + 2.0 * penalty * (hc.sum() - target_sum)
    return val, grad

def refine_lbfgs(h0, n, dx, target_sum, beta, penalty, max_iter, time_limit):
    t0 = time.time()
    def _obj(x):
        v, _ = obj_grad_lbfgs(x, n, dx, target_sum, beta, penalty)
        return v
    def _grad(x):
        _, g = obj_grad_lbfgs(x, n, dx, target_sum, beta, penalty)
        return g
    try:
        res = minimize(_obj, h0, jac=_grad, method='L-BFGS-B',
                       options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-10, 'maxls': 40})
        return proj_h(res.x, target_sum)
    except Exception:
        return proj_h(h0, target_sum)

def refine_slsqp(h0, n, dx, target_sum, beta, max_iter):
    global _con_target
    _con_target = target_sum
    def _obj(x):
        xc = np.clip(x, 1e-12, 1 - 1e-12)
        corr = np.correlate(xc, 1.0 - xc, mode='full')
        return smooth_max_val(corr, beta) * dx
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(_obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-13})
        return proj_h(res.x, target_sum)
    except Exception:
        return proj_h(h0, target_sum)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    def time_left():
        return budget_s - (time.time() - start_time)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        best_h = h_init.copy()
        best_c5 = compute_c5(h_init)
        best_n = len(h_init)

    n_candidates = [70, 80, 100, 120, 140, 160, 200, 250, 300]
    if best_n > 0:
        n_candidates = sorted(set([best_n] + n_candidates))

    for n_points in n_candidates:
        if time_left() < 40:
            break
        dx = 2.0 / n_points
        target_sum = n_points / 2.0

        inits = []
        if best_h is not None and len(best_h) != n_points:
            old_x = np.linspace(0, 2, len(best_h), endpoint=False)
            new_x = np.linspace(0, 2, n_points, endpoint=False)
            h_up = np.interp(new_x, old_x, best_h)
            inits.append(proj_h(h_up, target_sum))
        inits.append(proj_h(np.full(n_points, 0.5), target_sum))
        h_blk = np.tile([1.0, 0.0], n_points // 2 + 1)[:n_points]
        inits.append(proj_h(h_blk, target_sum))
        for freq in [3, 5, 7, 11]:
            x = np.linspace(0, 2, n_points, endpoint=False)
            h_s = 0.5 + 0.45 * np.sin(np.pi * freq * x)
            inits.append(proj_h(h_s, target_sum))
        for _ in range(3):
            inits.append(proj_h(np.random.random(n_points), target_sum))

        for h0 in inits:
            if time_left() < 20:
                break
            h_ref = refine_lbfgs(h0.copy(), n_points, dx, target_sum, 80.0, 500.0, 300, time_left())
            # Second pass with higher beta
            if time_left() > 30:
                h_ref2 = refine_lbfgs(h_ref.copy(), n_points, dx, target_sum, 150.0, 1000.0, 200, time_left())
                c5_2 = compute_c5(h_ref2)
                if c5_2 < compute_c5(h_ref):
                    h_ref = h_ref2
            c5_val = compute_c5(h_ref)
            if c5_val < best_c5:
                best_h = h_ref.copy()
                best_c5 = c5_val
                best_n = n_points

    if best_h is not None and time_left() > 40:
        n = best_n
        dx = 2.0 / n
        target_sum = n / 2.0
        for beta in [100, 150, 200, 300]:
            if time_left() < 20:
                break
            h_new = refine_lbfgs(best_h.copy(), n, dx, target_sum, float(beta), 1000.0, 300, time_left())
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new

    if best_h is not None and time_left() > 25:
        n = best_n
        dx = 2.0 / n
        target_sum = n / 2.0
        h_new = refine_slsqp(best_h.copy(), n, dx, target_sum, 200, 500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new

    if best_h is not None and time_left() > 15:
        n = best_n
        dx = 2.0 / n
        target_sum = n / 2.0

        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new

    if best_h is not None:
        best_h = proj_h(best_h, best_n / 2.0)
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
