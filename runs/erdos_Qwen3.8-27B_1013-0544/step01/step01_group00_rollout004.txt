```python
def project_h_l2(h, target):
    """Minimize ||h - h0||^2 subject to 0<=h<=1, sum(h)=target"""
    h0 = h.copy()
    h = h.copy()
    for _ in range(20):
        h = np.clip(h0 + (target - np.sum(h)) / len(h), 0, 1)
    # Final adjustment
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0, 1)
    return h
```

One more consideration: the `np.correlate` with `mode='full'` gives a vector of length 2n-1. For n=180, that's 359 values. The maximum is what we care about.

For the subgradient, I need to be careful about which lag is the argmax. The correlation at lag 0 is ∑h_i(1-h_i), which is always non-negative. For large lags, the overlap is small and the correlation approaches ∑h_i (for very large positive lags where there's no overlap, it's just ∑h_i * 1 = sum(h) = n/2... wait no).

Actually for lag k with |k| ≥ n, there's no overlap, so the correlation would be 0 (since the sum is over empty set). But in the "full" mode, the maximum lag is n-1, where there's exactly one overlapping pair.

For lag n-1: corr = h[n-1] * (1-h[0]) (just one term)
For lag 0: corr = ∑ h[i]*(1-h[i])

The maximum is likely at small lags (0, ±1, ±2, etc.) since there's more overlap.

```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))


def project_h(h, target):
    h = h.copy()
    for _ in range(30):
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        h = h + diff / len(h)
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h


def interp_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0.0, 1.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old, period=1.0)
    return project_h(h_new, n_new * 0.5)


def get_subgrad(h, k_index):
    n = len(h)
    dx = 2.0 / n
    k = k_index - (n - 1)
    g = np.zeros(n)
    if k > 0:
        g[k:] += (1.0 - h[:n - k])
        g[:n - k] += -h[k:]
    elif k < 0:
        p = -k
        g[:n - p] += (1.0 - h[p:])
        g[p:] += -h[:n - p]
    else:
        g[:] = 1.0 - 2.0 * h
    return g * dx


def cyclic_subgrad(h0, max_iter=2000, alpha0=0.04, mu=0.85):
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    stall = 0
    for it in range(max_iter):
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        c5 = float(np.max(corr))
        if c5 < best_c5 - 1e-10:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 500:
                break
        k_max = int(np.argmax(corr))
        g = get_subgrad(h, k_max)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-12:
            break
        v = mu * v + (1.0 - mu) * g
        alpha = alpha0 / (1.0 + 0.003 * it)
        h_new = h - alpha * v
        h_new = project_h(h_new, n * 0.5)
        c5_new = compute_c5(h_new)
        if c5_new > c5 + 0.01:
            alpha *= 0.3
            h_new = h - alpha * v
            h_new = project_h(h_new, n * 0.5)
        h = h_new
    return best_h, best_c5


def t_objective(v):
    return v[-1]


def make_t_ineq(n):
    dx = 2.0 / n
    def t_ineq(v):
        h = v[:n]
        t = v[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        return t - corr
    return t_ineq


def t_eq_fun(v):
    n = len(v) - 1
    return np.sum(v[:n]) - n * 0.5


def t_eq_jac(v):
    n = len(v) - 1
    j = np.zeros(len(v))
    j[:n] = 1.0
    return j


def sum_eq_fun(h):
    return np.sum(h) - len(h) * 0.5


def sum_eq_jac(h):
    return np.ones(len(h))


def smooth_obj(h, temp):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    return m + temp * np.log(np.sum(np.exp((corr - m) / temp)))


def optimize_t_var(h0, max_iter=400, ftol=1e-10):
    n = len(h0)
    c5_cur = compute_c5(h0)
    v0 = np.append(h0.copy(), c5_cur + 0.02)
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.5)]
    cons = [
        {'type': 'ineq', 'fun': make_t_ineq(n)},
        {'type': 'eq', 'fun': t_eq_fun, 'jac': t_eq_jac}
    ]
    res = minimize(t_objective, v0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': ftol})
    h_new = np.clip(res.x[:n], 0.0, 1.0)
    h_new = project_h(h_new, n * 0.5)
    return h_new, compute_c5(h_new)


def optimize_smooth(h0, temp, max_iter=300, ftol=1e-12):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac}]
    res = minimize(smooth_obj, h0.copy(), args=(temp,), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': ftol})
    h_new = np.clip(res.x, 0.0, 1.0)
    h_new = project_h(h_new, n * 0.5)
    return h_new, compute_c5(h_new)
```
