def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_h(h, target):
    h = h.copy()
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return np.clip(h, 0.0, 1.0)
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = np.sum(mask)
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def interp_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.arange(n_old) / n_old
    x_new = np.arange(n_new) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_h(h_new, n_new * 0.5)

def lse_obj_and_grad(d, temp):
    n1 = len(d)
    n = n1 + 1
    h = np.empty(n)
    h[:n1] = 0.5 + d
    h[n1] = 0.5 - np.sum(d)

    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    obj = m + temp * np.log(np.sum(np.exp((corr - m) / temp)))
    exp_vals = np.exp((corr - m) / temp)
    w = exp_vals / np.sum(exp_vals)
    grad_h = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-18:

        lag = k - (n - 1)
