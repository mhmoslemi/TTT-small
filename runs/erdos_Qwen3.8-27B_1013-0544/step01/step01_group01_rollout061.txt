def projected_gradient_step(h, target_sum, step_size, n_top_lags):
    n = len(h)
    # Compute all correlations
    corrs = np.array([np.sum(h[:n-k] * (1-h[k:])) for k in range(n)])
    # Find top lags
    top_idx = np.argsort(corrs)[-n_top_lags:]
    weights = np.softmax(corrs[top_idx] * 100)  # focus on top lags
    
    # Compute gradient
    grad = np.zeros(n)
    for idx, k in enumerate(top_idx):
        for j in range(n):
            g = 0.0
            if j + k < n:
                g += (1 - h[j+k])
            if j - k >= 0:
                g -= h[j-k]
            grad[j] += weights[idx] * g
    
    # Projected gradient step
    h_new = h - step_size * grad
    # Project onto feasible set
    h_new = np.clip(h_new, 0, 1)
    h_new += (target_sum - h_new.sum()) / n
    h_new = np.clip(h_new, 0, 1)
    return h_new

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - h.sum()
        if abs(diff) > 1e-12:
            mask = (h > 1e-8) & (h < 1.0 - 1e-8)
            if mask.sum() > 0:
                h[mask] += diff / mask.sum()
                h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

    n = len(h)
    corrs = np.zeros(n)
    for k in range(n):

        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        else:
            grad[:n-k] += w * (1.0 - h[k:])
            grad[k:] -= w * h[:n-k]

import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - h.sum()
        if abs(diff) > 1e-12:
            mask = (h > 1e-8) & (h < 1.0 - 1e-8)
            if mask.sum() > 0:
                h[mask] += diff / mask.sum()
                h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta


def full_gradient(h, lags, weights):
    n = len(h)
    grad = np.zeros(n)
    for w, k in zip(weights, lags):
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            grad[:n - k] += w * (1.0 - h[k:])
            grad[k:] -= w * h[:n - k]
        else:
            kk = -k
            grad[kk:] += w * (1.0 - h[:n - kk])
            grad[:n - kk] -= w * h[kk:]
    return grad


def obj_func(h, n, beta):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, beta) * (2.0 / n)


def con_fun(x, target_sum):
    return np.sum(x) - target_sum


def con_jac(x, target_sum):
    return np.ones(len(x))

        h_start = np.array(initial_h_values, dtype=float).copy()
    else:
        n = 70
        h_start = np.full(n, 0.5)

    n = len(h_start)
    target_sum = n / 2.0
    best_h = project_feasible(h_start.copy(), target_sum)
    best_c5 = compute_c5(best_h)

    # Phase 1: Subgradient with momentum and line search
    step = 0.004
    vel = np.zeros(n)
    momentum = 0.85
    for it in range(15000):
        if budget_s - (time.time() - start_time) < 200:
            break
        corr_full = np.correlate(best_h, 1.0 - best_h, mode="full")
        top_k = min(8, 2 * n - 1)
        top_idx = np.argsort(corr_full)[-top_k:]
        top_lags = (top_idx - (n - 1)).astype(int)
        w = np.exp(corr_full[top_idx] * 120.0)
        w /= w.sum()
        grad = full_gradient(best_h, top_lags, w)
        momentum = 0.85 + 0.1 * (it / 15000.0)
        vel = momentum * vel - step * grad
        alpha = 1.0
        improved = False
        for _ in range(5):
            h_try = project_feasible(best_h + alpha * vel, target_sum)
            c_try = compute_c5(h_try)
            if c_try < best_c5:
                best_h, best_c5 = h_try, c_try
                improved = True
                break
            alpha *= 0.5
        if not improved:
            step = max(step * 0.7, 1e-6)
            vel = np.zeros(n)
        if it % 400 == 399:
            idx = np.random.choice(n, 6, replace=False)
            best_h[idx] += np.random.randn(6) * 0.12
            best_h = project_feasible(best_h, target_sum)
            vel = np.zeros(n)

    # Phase 2: Simulated annealing with targeted moves
    n_sa = 8000
    for it in range(n_sa):
        if budget_s - (time.time() - start_time) < 120:
            break
        frac = it / n_sa
        T = 0.00003 + 0.003 * (1.0 - frac) ** 3.0
        h_new = best_h.copy()
        mt = np.random.randint(5)
        if mt == 0:
            i, j = np.random.choice(n, 2, replace=False)
            d = np.random.uniform(0.005, 0.15)
            d = min(d, best_h[i], 1.0 - best_h[j])
            if d > 1e-7:
                h_new[i] -= d
                h_new[j] += d
        elif mt == 1:
            corr_full = np.correlate(best_h, 1.0 - best_h, mode="full")
            wk = int(np.argmax(corr_full))
            lag = wk - (n - 1)
            if lag > 0:
                scores = best_h[:n - lag] * (1.0 - best_h[lag:])
                i = int(np.argmax(scores))
                j = i + lag
                d = min(best_h[i] * 0.2, 1.0 - best_h[j]) * np.random.uniform(0.5, 2.0)
            elif lag < 0:
                kk = -lag
                scores = best_h[kk:] * (1.0 - best_h[:n - kk])
                i = int(np.argmax(scores)) + kk
                j = i - kk
                d = min(best_h[i] * 0.2, 1.0 - best_h[j]) * np.random.uniform(0.5, 2.0)
            else:
                i = int(np.argmax(best_h * (1.0 - best_h)))
                j = int(np.random.choice(n))
                d = min(best_h[i] * 0.1, 1.0 - best_h[j])
            if d > 1e-7:
                h_new[i] -= d
                h_new[j] += d
        elif mt == 2:
            i = np.random.randint(1, n - 1)
            blend = np.random.uniform(0.2, 0.6)
            h_new[i] = blend * best_h[i] + (1 - blend) * 0.5 * (best_h[i - 1] + best_h[i + 1])
        elif mt == 3:
            i1 = np.random.randint(0, n // 2)
            i2 = np.random.randint(n // 2, n)
            j1 = int(np.random.choice(n))
            j2 = int(np.random.choice(n))
            d1 = min(best_h[i1] * 0.15, 1.0 - best_h[j1])
            d2 = min(best_h[i2] * 0.15, 1.0 - best_h[j2])
            if d1 > 1e-7:
                h_new[i1] -= d1
                h_new[j1] += d1
            if d2 > 1e-7:
                h_new[i2] -= d2
                h_new[j2] += d2
        else:
            block = np.random.randint(3, max(4, n // 8))
            bs = np.random.randint(0, n - block)
            shift = np.random.randint(1, block)
            h_new[bs:bs + block] = np.roll(best_h[bs:bs + block], shift)

        h_new = project_feasible(h_new, target_sum)
        c_new = compute_c5(h_new)
        delta = c_new - best_c5
        if delta <= 0 or np.random.random() < np.exp(-delta / max(T, 1e-14)):
            best_h = h_new
            if c_new < best_c5:
                best_c5 = c_new

    # Phase 3: SLSQP refinement
    if budget_s - (time.time() - start_time) > 30:
        try:
            bounds = Bounds(np.zeros(n), np.ones(n))
            cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum,)}]
            res = minimize(obj_func, best_h, args=(n, 200.0), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 3000, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target_sum)
            c_new = compute_c5(h_new)
            if c_new < best_c5:
                best_h, best_c5 = h_new, c_new
        except Exception:
            pass

    # Phase 4: Random and structured restarts
    if budget_s - (time.time() - start_time) > 80:
        restarts = []
        x_grid = np.linspace(0, 2, n)
        for freq in [1, 2, 3, 4, 5, 7, 10]:
            h_struct = 0.5 + 0.35 * np.sin(np.pi * freq * x_grid)
            restarts.append(project_feasible(h_struct, target_sum))
        for _ in range(8):
            h_rand = np.random.random(n)
            restarts.append(project_feasible(h_rand, target_sum))
        for h0 in restarts:
            if budget_s - (time.time() - start_time) < 15:
                break
            try:
                bounds = Bounds(np.zeros(n), np.ones(n))
                cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (target_sum,)}]
                res = minimize(obj_func, h0, args=(n, 100.0), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 300, 'ftol': 1e-11})
                h_new = project_feasible(res.x, target_sum)
                c_new = compute_c5(h_new)
                if c_new < best_c5:
                    best_h, best_c5 = h_new, c_new
            except Exception:
                pass

    # Phase 5: Multi-n exploration
    if budget_s - (time.time() - start_time) > 100:
        for n_try in [60, 80, 90, 100, 120, 55, 65]:
            if budget_s - (time.time() - start_time) < 40:
                break
            if n_try == n:
                continue
            try:
                x_old = np.linspace(0, 2, n)
                x_new_pts = np.linspace(0, 2, n_try)
                h_interp = np.interp(x_new_pts, x_old, best_h)
                ts = n_try / 2.0
                h_interp = project_feasible(h_interp, ts)
                bounds2 = Bounds(np.zeros(n_try), np.ones(n_try))
                cons2 = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (ts,)}]
                res = minimize(obj_func, h_interp, args=(n_try, 120.0),
                               method='SLSQP', bounds=bounds2, constraints=cons2,
                               options={'maxiter': 800, 'ftol': 1e-13})
                h_new = project_feasible(res.x, ts)
                c_new = compute_c5(h_new)
                if c_new < best_c5:
                    best_h, best_c5, n = h_new, c_new, n_try
            except Exception:
                pass

    # Phase 6: Final high-precision polish
    if budget_s - (time.time() - start_time) > 15:
        try:
            bounds = Bounds(np.zeros(n), np.ones(n))
            cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (n / 2.0,)}]
            res = minimize(obj_func, best_h, args=(n, 500.0), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 5000, 'ftol': 1e-16})
            h_new = project_feasible(res.x, n / 2.0)
            c_new = compute_c5(h_new)
            if c_new < best_c5:
                best_h, best_c5 = h_new, c_new
        except Exception:
            pass

    # Final verification
    best_h = project_feasible(best_h, n / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
