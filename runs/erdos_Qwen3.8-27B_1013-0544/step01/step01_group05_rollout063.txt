import numpy as np
from scipy.optimize import minimize
import time

_G_dx = None
_G_n = None
_G_nh = None
_G_temp = None
_G_ts = None

def project_feasible(x, target_sum):
    lo = float(x.min()) - 1.0
    hi = float(x.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = float(np.clip(x - mid, 0.0, 1.0).sum())
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_grad_lag(h, L):
    n = len(h)
    omh = 1.0 - h
    grad = np.zeros(n)
    if L == 0:
        grad[:] = omh[:] - h[:]
    elif L > 0:
        grad[L:] += omh[:n - L]
        grad[:n - L] -= h[L:]
    else:
        L2 = -L
        grad[:n - L2] += omh[L2:]
        grad[L2:] -= h[:n - L2]
    return grad

def _sob(x):
    c = np.correlate(x, 1.0 - x, mode='full')
    cm = float(np.max(c))
    t = _G_temp
    s = (c - cm) / t
    return (cm + t * float(np.log(np.sum(np.exp(s))))) * _G_dx

def _sog(x):
    c = np.correlate(x, 1.0 - x, mode='full')
    cm = float(np.max(c))
    t = _G_temp
    s = (c - cm) / t
    w = np.exp(s)
    w /= w.sum()
    g = np.zeros(_G_n)
    for i in range(len(c)):
        if w[i] > 1e-8:
            L = i - _G_nh
            g += w[i] * compute_grad_lag(x, L)
    return g * _G_dx

    _G_n = n

    # Phase 1: Projected gradient with softmax-weighted active set
    h = h0.copy()
    step = 0.005
    mu = 0.55
    mom = np.zeros(n)
    t1_limit = budget_s - 350
    while time.time() - start < t1_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = float(np.max(corr))
        T_act = max(c_max * 0.08, 0.01)
        s_act = (corr - c_max) / T_act
        w_act = np.exp(s_act)
        top_idx = np.argsort(w_act)[-20:]
        w_top = w_act[top_idx]
        w_top /= w_top.sum()
        grad = np.zeros(n)
        for i in range(len(top_idx)):
