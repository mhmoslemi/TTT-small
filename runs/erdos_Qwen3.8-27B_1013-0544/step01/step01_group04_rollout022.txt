import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_c5_smooth(h, dx, T=50.0):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    return m + np.log(np.mean(np.exp(T * (corr - m)))) / T

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = np.sum(h) - target
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(free):
            h[free] -= diff / np.sum(free)
        else:
            h -= diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def optimize_smooth(h_start, n, dx, T, max_iter=200):
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    ones = np.ones(n)

    def obj(x):
        return compute_c5_smooth(x, dx, T)

    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: ones}]
    try:
        res = minimize(obj, h_start, method='SLSQP', bounds=bounds,
                       constraints=constraints, options={'maxiter': max_iter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h_start, n)

def optimize_direct(h_start, n, dx, max_iter=300):
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    ones = np.ones(n)
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: ones}]
    try:
        res = minimize(lambda x: compute_c5(x, dx), h_start, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h_start, n)
