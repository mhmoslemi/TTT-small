import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_h(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = np.sum(mask)
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def interp_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 1, n_old, endpoint=False)
    x_new = np.linspace(0, 1, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_h(h_new, n_new * 0.5)

def smooth_lse(h, temp):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    return m + temp * np.log(np.sum(np.exp((corr - m) / temp)))

def sum_eq_fun(h):
    return np.sum(h) - len(h) * 0.5

def sum_eq_jac(h):
    return np.ones(len(h))

def optimize_smooth_slsqp(h0, temp, max_iter=200):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac}]
    
    def obj(h):
        return smooth_lse(h, temp)
    
    res = minimize(obj, h0.copy(), method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-12})
    h_new = project_h(res.x, n * 0.5)
    return h_new, compute_c5(h_new)

        j[:n] = 1.0
        return j

        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        top_idx = np.argsort(corr)[-n_top:]
        t = np.max(corr) + 0.001

            h_v = v[:n]
            t_v = v[-1]
            corr_v = np.correlate(h_v, 1.0 - h_v, mode='full') * dx
            return t_v - corr_v[top_idx]
        
        cons = [{'type': 'ineq', 'fun': ineq},
                {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
        
        v0 = np.append(h, t)

        if c5_new < compute_c5(h):
