```python
def project_to_simplex_sum(h, target):
    """Project h to {0 <= h[i] <= 1, sum(h) = target}"""
    h = np.clip(h, 0, 1)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # Water filling
    for _ in range(10):
        if diff > 0:
            mask = h < 1.0
            cnt = mask.sum()
            if cnt == 0: break
            step = min(diff / cnt, 1.0 - h[mask].min()) if (1.0 - h[mask]).min() > 0 else 0
            h[mask] += min(diff / cnt, (1.0 - h[mask]).min() + 1e-15)
        else:
            mask = h > 0.0
            cnt = mask.sum()
            if cnt == 0: break
            step = min(abs(diff) / cnt, h[mask].min())
            h[mask] -= step
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
    return np.clip(h, 0, 1)
```

import numpy as np
from scipy.optimize import minimize, Bounds
import time

_g_tau = 1000.0
_g_target = 60.0

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
            cnt = mask.sum()
            if cnt == 0:
                h[:int(target)] = 1.0
                break
            step = min(diff / cnt, 1.0 - h[mask].min() + 1e-13)
            h[mask] += step
        else:
            mask = h > 1e-13
            cnt = mask.sum()
            if cnt == 0:
                break
            step = min(-diff / cnt, h[mask].min() + 1e-13)
            h[mask] -= step
    return np.clip(h, 0.0, 1.0)

def cd_sweeps(h, n_sweeps, t0, budget_s):
    n = len(h)
    target = n / 2.0
    lags = 2 * n - 1
    shifts = np.arange(lags) - (n - 1)
    h = h.copy()
    result = np.correlate(h, 1.0 - h, mode="full")
    best_c5 = evaluate_c5(h)
    for _ in range(n_sweeps):
        if budget_s - (time.time() - t0) < 3:
            break
        h_cur = h.copy()
        res = result.copy()
        order = np.random.permutation(n)
        for i in order:
            ir = i + shifts
            vr = (ir >= 0) & (ir < n)
            hr = np.where(vr, h_cur[np.clip(ir, 0, n - 1)], 0.0)
            il = i - shifts
            vl = (il >= 0) & (il < n)
            hl = np.where(vl, h_cur[np.clip(il, 0, n - 1)], 0.0)
            a = res - h_cur[i] * (1.0 - hr) + h_cur[i] * hl
            b = (1.0 - hr) - hl
            lo, hi = 0.0, 1.0
            for _ in range(45):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                if np.max(a + b * m1) <= np.max(a + b * m2):
                    hi = m2
                else:
                    lo = m1
            h_cur[i] = (lo + hi) * 0.5
            res = a + b * h_cur[i]
        h = project_sum(h_cur, target)
        result = np.correlate(h, 1.0 - h, mode="full")
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
    return h, best_c5

def _g_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    t = _g_tau
    xv = t * corr
    xm = np.max(xv)
    return float(xm / t + np.log(np.sum(np.exp(xv - xm))) / t)

def _g_con_fun(x):
    return np.sum(x) - _g_target

def _g_con_jac(x):
    return np.ones(len(x))

def slsqp_polish(h, tau, maxiter=300):
    global _g_tau, _g_target
    _g_tau = tau
    _g_target = len(h) / 2.0
    n = len(h)
    bnds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': _g_con_fun, 'jac': _g_con_jac}]
    try:
        res = minimize(_g_obj, h, method='SLSQP', bounds=bnds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_sum(res.x, n / 2.0)
    except Exception:
        return h

def make_binary(n):
    h = np.zeros(n)
    idx = np.random.choice(n, size=int(n / 2), replace=False)
    h[idx] = 1.0
    return h

def make_smooth_random(n):
    h = np.random.beta(2, 2, n)
    return project_sum(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(120, 0.5)

    best_h = h_init.copy()
    best_c5 = evaluate_c5(best_h)

    # Phase 1: CD from initial
    best_h, best_c5 = cd_sweeps(best_h, 500, t0, budget_s)

    # Phase 2: Resolution sweep
    n0 = len(best_h)
    for tn in [60, 80, 100, 140, 160, 200, 240]:
        if budget_s - (time.time() - t0) < 20:
            break
        if tn == n0:
            continue
        h_t = np.interp(np.linspace(0, 1, tn), np.linspace(0, 1, len(best_h)), best_h)
        h_t = project_sum(h_t, tn / 2.0)
        h_t, c5_t = cd_sweeps(h_t, 250, t0, budget_s)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t

    # Phase 3: Binary restarts
    n_best = len(best_h)
    for _ in range(20):
        if budget_s - (time.time() - t0) < 12:
            break
        h_r = make_binary(n_best)
        h_r, c5_r = cd_sweeps(h_r, 150, t0, budget_s)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r

    # Phase 4: Smooth random restarts
    for _ in range(8):
        if budget_s - (time.time() - t0) < 12:
            break
        h_r = make_smooth_random(n_best)
        h_r, c5_r = cd_sweeps(h_r, 120, t0, budget_s)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r

    # Phase 5: Perturbations around best
    for _ in range(10):
        if budget_s - (time.time() - t0) < 12:
            break
        scale = 0.03 + 0.10 * np.random.random()
        h_p = best_h + np.random.normal(0, scale, n_best)
        h_p = project_sum(h_p, n_best / 2.0)
        h_p, c5_p = cd_sweeps(h_p, 120, t0, budget_s)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p

    # Phase 6: SLSQP polish
    if budget_s - (time.time() - t0) > 25:
        for tau in [500, 2000, 10000, 50000]:
            if budget_s - (time.time() - t0) < 12:
                break
            h_p = slsqp_polish(best_h, tau, 200)
            c5_p = evaluate_c5(h_p)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_p
            best_h, best_c5 = cd_sweeps(best_h, 150, t0, budget_s)

    # Phase 7: Final intensive CD
    if budget_s - (time.time() - t0) > 5:
        best_h, best_c5 = cd_sweeps(best_h, 800, t0, budget_s)

    best_h = project_sum(best_h, len(best_h) / 2.0)
    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, len(best_h)
