import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    """Project y onto {h: 0<=h<=1, sum(h)=n/2}."""
    target = n / 2.0
    # Bisection on lambda: find lambda s.t. sum(clip(y - lambda, 0, 1)) = target
    lo, hi = y.min() - 1, y.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.clip(y - mid, 0, 1)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
        if abs(s - target) < 1e-12:
            break
    return np.clip(y - (lo + hi) / 2, 0, 1)

def smooth_max(corr_vals, tau):
    """Compute smooth max and softmax weights."""
    x = tau * corr_vals
    x_max = x.max()
    exp_x = np.exp(x - x_max)
    weights = exp_x / exp_x.sum()
    val = (x_max + np.log(exp_x.sum())) / tau
    return val, weights

    grad = np.zeros(n)
    for m in range(n):

        v2 = (m + k_vals >= 0) & (m + k_vals < n)
        grad[m] = np.sum(weights[v1] * (1.0 - h[m - k_vals[v1]])) \
                 - np.sum(weights[v2] * h[m + k_vals[v2]])
    return grad

import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, target):
    """Project y onto {h: 0<=h<=1, sum(h)=target} via bisection."""
    lo = y.min() - target / len(y) - 1.0
    hi = y.max() + target / len(y) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        proj = np.clip(y - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)

def compute_grad(h, n, tau):
    """Compute gradient of smooth max correlation."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    k_vals = np.arange(-(n-1), n)
    x = tau * corr
    x_max = x.max()
    exp_x = np.exp(x - x_max)
    w = exp_x / exp_x.sum()
    
    grad = np.zeros(n)
    for m in range(n):
        v1 = (m - k_vals >= 0) & (m - k_vals < n)
        v2 = (m + k_vals >= 0) & (m + k_vals < n)
        grad[m] = np.sum(w[v1] * (1.0 - h[m - k_vals[v1]])) \
                 - np.sum(w[v2] * h[m + k_vals[v2]])
    return grad

    best_c5 = evaluate_c5(h)
    tau = tau_start
    stalled = 0
    target = n / 2.0
    
    for it in range(max_iters):
        if time_limit is not None and t0 is not None:

        g = compute_grad(best_h, n, tau)

            if tau < 50000:
                tau *= 3.0
            else:
                tau *= 0.3

                break

import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, target):
    n = len(y)
    lo = y.min() - target * 2.0 / n - 1.0
    hi = y.max() + target * 2.0 / n + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        proj = np.clip(y - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(y - lam, 0.0, 1.0)

def compute_grad_vec(h, w):
    n = len(h)
    first = np.convolve(1.0 - h, w, 'full')
    second = np.correlate(w, h, 'full')
    ft = first[n - 1:2 * n - 1]
    st = second[n - 1:2 * n - 1][::-1]
    return ft - st

def soft_max_params(corr, tau):
    x = tau * corr
    x_max = x.max()
    exp_x = np.exp(x - x_max)
    return exp_x / exp_x.sum()

def pg_optimize(h, n, tau_init, max_iters, t_start, time_limit):
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    target = n / 2.0
    tau = tau_init
    t_k = 1.0
    x_prev = best_h.copy()
    stalled = 0
    stall_count = 0

    for it in range(max_iters):
        if time.time() - t_start > time_limit:
            break
        y = best_h + (t_k - 1.0) / (t_k + 1.0) * (best_h - x_prev)
        y = np.clip(y, 0.0, 1.0)
        corr = np.correlate(y, 1.0 - y, mode='full') * (2.0 / n)
        w = soft_max_params(corr, tau)
        g = compute_grad_vec(y, w)

        lr = 0.5
        improved = False
        for _ in range(35):
            cand = project_feasible(y - lr * g, target)
            c5_val = evaluate_c5(cand)

                best_h = cand
                best_c5 = c5_val
                improved = True
                break
            lr *= 0.5

        if improved:

            t_k = 1.0
            if tau < 80000.0:
                tau *= 2.5
            else:
                tau *= 0.4
            if stalled > 30:
                break
    return best_h, best_c5
