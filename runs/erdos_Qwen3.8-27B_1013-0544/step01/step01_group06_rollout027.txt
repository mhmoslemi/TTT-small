            grad[:n - p] -= w * h[p:]
    return smax * dx, grad * dx

def optimize_momentum(h0, n, deadline, rng, alpha_init=60, alpha_max=8000):
    target_sum = n / 2.0
    h = project_box_simplex(h0.astype(float).copy(), target_sum)
    v = np.zeros(n)
    beta = 0.85
    lr = 0.015
    alpha = alpha_init
    best_h = h.copy()
    best_c5 = compute_c5(h)
    patience = 0
    step = 0
    n_restarts = 0

    while time.time() < deadline:
        smax, grad = smooth_max_grad(h, alpha)
        v = beta * v - lr * grad
        h_new = h + v
        h_new = project_box_simplex(h_new, target_sum)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5 - 1e-13:
            best_h = h_new.copy()
            best_c5 = c5_new
            h = h_new
            patience = 0
            if lr < 0.06:
                lr *= 1.008
        else:
            h = h_new
            patience += 1
            if patience % 5 == 0:
                lr *= 0.93
            if patience > 100:
                pert = 0.04 + 0.04 * min(n_restarts * 0.5, 1.0)
                h = best_h + rng.randn(n) * pert
                h = project_box_simplex(h, target_sum)
                v = np.zeros(n)
                lr = 0.015
                alpha = min(alpha * 1.4, alpha_max)
                patience = 0
                n_restarts += 1
                if n_restarts > 20:
                    break

        step += 1
        if step % 50 == 0:
            alpha = min(alpha * 1.08, alpha_max)
        if step % 500 == 0:
            beta = min(beta + 0.005, 0.95)

    return best_h, best_c5

def local_search_01(n, deadline, rng):
    n_half = n // 2
    ones = rng.choice(n, n_half, replace=False)
    h = np.zeros(n)
    h[ones] = 1.0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    ones_set = set(ones.tolist())
    zeros_set = set(range(n)) - ones_set
    no_improve = 0
    restarts = 0

    while time.time() < deadline - 2:
        ones_list = sorted(ones_set)
        zeros_list = sorted(zeros_set)
        best_swap = None
        best_swap_c5 = best_c5

        for i in ones_list:
            h[i] = 0.0
            for j in zeros_list:
                h[j] = 1.0
                c5 = compute_c5(h)
                h[j] = 0.0
                if c5 < best_swap_c5 - 1e-12:
                    best_swap_c5 = c5
                    best_swap = (i, j)
            h[i] = 1.0

        if best_swap is not None:
            i, j = best_swap
            h[i] = 0.0
            h[j] = 1.0
            ones_set.discard(i)
            zeros_set.discard(j)
            ones_set.add(j)
            zeros_set.add(i)
            best_c5 = best_swap_c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 3:
                no_improve = 0
                restarts += 1
                if restarts > 30:
                    break
                new_ones = rng.choice(n, n_half, replace=False)
                h[:] = 0.0
                h[new_ones] = 1.0
                ones_set = set(new_ones.tolist())
                zeros_set = set(range(n)) - ones_set
                c5 = compute_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()

    return best_h, best_c5

def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False) + 1.0 / n_new
    h_ext = np.append(h_old, h_old[0])
    x_ext = np.append(x_old, 2.0 + 1.0 / n_old)
    return np.interp(x_new, x_ext, h_ext)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    has_parent = 'initial_h_values' in globals() and initial_h_values is not None
    h_parent = np.array(initial_h_values, dtype=float) if has_parent else None

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    def elapsed():
        return time.time() - t0

    def remaining():
        return budget_s - elapsed()

    # Phase A: Local search on 0/1 at n=40 (fast, finds good structure)
    if remaining() > 200:
        ls_time = min(50.0, remaining() * 0.12)
        h_ls, c5_ls = local_search_01(40, t0 + ls_time, rng)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls
            best_n = 40
        # Interpolate and polish
        for n_target in [128, 160, 200]:
            if remaining() < 100:
                break
            h_interp = interpolate_h(h_ls, 40, n_target)
            h_interp = project_box_simplex(h_interp, n_target / 2.0)
            dl = t0 + budget_s - max(100, remaining() * 0.4)
            h_opt, c5_opt = optimize_momentum(h_interp, n_target, dl, rng, alpha_init=80)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                best_n = n_target

    # Phase B: Optimize from parent
    if has_parent and remaining() > 150:
        n_p = len(h_parent)
        dl = t0 + budget_s - max(120, remaining() * 0.35)
        h_opt, c5_opt = optimize_momentum(h_parent, n_p, dl, rng, alpha_init=100, alpha_max=6000)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = n_p

    # Phase C: Multiple structured starts at various resolutions
    resolutions = [128, 160, 200, 256]
    for n in resolutions:
        if remaining() < 80:
            break
        x = np.linspace(0, 2, n, endpoint=False) + 1.0 / n
        inits = []
        for freq in [1.0, 1.5, 2.0]:
            h_s = 0.5 + 0.35 * np.sin(2.0 * np.pi * freq * x)
            inits.append(project_box_simplex(h_s, n / 2.0))
        # Block init
        h_blk = np.zeros(n)
        h_blk[n // 4:3 * n // 4] = 1.0
        inits.append(project_box_simplex(h_blk, n / 2.0))
        # From best so far
        if best_h is not None and len(best_h) != n:
            inits.append(project_box_simplex(interpolate_h(best_h, len(best_h), n), n / 2.0))
        elif best_h is not None:
            inits.append(best_h.copy())

        time_per = max(20.0, min(80.0, (remaining() - 60) / max(1, len(inits))))
        for h_init in inits:
            if remaining() < 60:
                break
            dl = time.time() + time_per
            h_opt, c5_opt = optimize_momentum(h_init, n, dl, rng, alpha_init=50, alpha_max=5000)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                best_n = n
            time_per = max(20.0, time_per * 0.85)

    # Phase D: Final polish with high alpha
    if best_h is not None and remaining() > 20:
        dl = t0 + budget_s - 3
        h_opt, c5_opt = optimize_momentum(best_h, best_n, dl, rng, alpha_init=800, alpha_max=20000)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    if best_h is None:
        n = 200
        best_h = np.full(n, 0.5)
        best_c5 = compute_c5(best_h)
        best_n = n

    best_h = project_box_simplex(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
