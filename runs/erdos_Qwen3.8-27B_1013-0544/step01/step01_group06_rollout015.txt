import numpy as np
import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * (2.0 / n)

def corr_max(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')))

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(-h, kind='stable')
    result = np.zeros(n)
    result[idx[:target]] = 1.0
    return result

def sa_search(h, deadline, rng, T_init=0.03):
    n = len(h)
    h_cur = to_binary(h, n)
    ones = np.where(h_cur == 1.0)[0]
    zeros = np.where(h_cur == 0.0)[0]
    curr = corr_max(h_cur)
    best_h = h_cur.copy()
    best = curr
    T = T_init
    stall = 0
    while time.time() < deadline:
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h_cur[i] = 0.0
        h_cur[j] = 1.0
        new = corr_max(h_cur)
        d = new - curr
        if d <= 0 or rng.rand() < np.exp(-d / max(T, 1e-15)):
            curr = new
            ones = np.where(h_cur == 1.0)[0]
            zeros = np.where(h_cur == 0.0)[0]
            if curr < best:
                best_h = h_cur.copy()
                best = curr
                stall = 0
            else:
                stall += 1
        else:
            h_cur[j] = 0.0
            h_cur[i] = 1.0
            stall += 1
        T = max(T * 0.9997, 0.0005)
        if stall > 600:
            n_pert = rng.randint(8, 25)
            for _ in range(n_pert):
                ii = ones[rng.randint(len(ones))]
                jj = zeros[rng.randint(len(zeros))]
                h_cur[ii] = 0.0
                h_cur[jj] = 1.0
            ones = np.where(h_cur == 1.0)[0]
            zeros = np.where(h_cur == 0.0)[0]
            curr = corr_max(h_cur)
            T = T_init * 0.5
            stall = 0
    return best_h, best * (2.0 / n)

def hill_climb(h, deadline, rng, max_rounds=80):
    n = len(h)
    h_cur = h.copy()
    curr = corr_max(h_cur)
    best_h = h_cur.copy()
    best = curr
    rounds = 0
    while time.time() < deadline and rounds < max_rounds:
        improved = False
        ones = np.where(h_cur == 1.0)[0]
        zeros = np.where(h_cur == 0.0)[0]
        best_d = 0.0
        bi, bj = -1, -1
        for i in ones:
            h_cur[i] = 0.0
            base = h_cur.copy()
            for j in zeros:
                h_cur = base.copy()
                h_cur[j] = 1.0
                nm = corr_max(h_cur)
                if nm < curr + best_d - 1e-12:
                    best_d = nm - curr
                    bi, bj = i, j
            h_cur[i] = 1.0
            if time.time() > deadline:
                break
        if bi >= 0:
            h_cur[bi] = 0.0
            h_cur[bj] = 1.0
            curr += best_d
            if curr < best:
                best_h = h_cur.copy()
                best = curr
            rounds += 1
            improved = True
        else:
            if time.time() < deadline - 15:
                for _ in range(rng.randint(10, 30)):
                    ii = ones[rng.randint(len(ones))]
                    jj = zeros[rng.randint(len(zeros))]
                    h_cur[ii] = 0.0
                    h_cur[jj] = 1.0
                curr = corr_max(h_cur)
                rounds = 0
            else:
                break
    return best_h, best * (2.0 / n)

def two_opt_search(h, deadline, rng):
    """Try pairs of swaps: flip two 1s to 0 and two 0s to 1."""
    n = len(h)
    h_cur = h.copy()
    curr = corr_max(h_cur)
    best_h = h_cur.copy()
    best = curr
    stall = 0
    while time.time() < deadline:
        ones = np.where(h_cur == 1.0)[0]
        zeros = np.where(h_cur == 0.0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        i1 = ones[rng.randint(len(ones))]
        i2 = ones[rng.randint(len(ones))]
        if i1 == i2:
            continue
        j1 = zeros[rng.randint(len(zeros))]
        j2 = zeros[rng.randint(len(zeros))]
        if j1 == j2:
            continue
        h_cur[i1] = 0.0
        h_cur[i2] = 0.0
        h_cur[j1] = 1.0
        h_cur[j2] = 1.0
        new = corr_max(h_cur)
        if new < curr:
            curr = new
            if curr < best:
                best_h = h_cur.copy()
                best = curr
                stall = 0
            else:
                stall += 1
        else:
            h_cur[i1] = 1.0
            h_cur[i2] = 1.0
            h_cur[j1] = 0.0
            h_cur[j2] = 0.0
            stall += 1
        if stall > 2000:
            for _ in range(rng.randint(5, 20)):
                ii = np.where(h_cur == 1.0)[0]
                jj = np.where(h_cur == 0.0)[0]
                h_cur[ii[rng.randint(len(ii))]] = 0.0
                h_cur[jj[rng.randint(len(jj))]] = 1.0
            curr = corr_max(h_cur)
            stall = 0
    return best_h, best * (2.0 / n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        n = 200
        h0 = np.zeros(n)
        h0[:n // 2] = 1.0

    n0 = len(h0)
    h_bin = to_binary(h0, n0)
    best_h = h_bin.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: SA at initial resolution (25% time)
    dl1 = t0 + budget_s * 0.25
    h1, c5_1 = sa_search(best_h, dl1, rng, T_init=0.04)
    if c5_1 < best_c5:
        best_h, best_c5 = h1, c5_1

    # Phase 2: Hill climbing (20% time)
    dl2 = t0 + budget_s * 0.45
    h2, c5_2 = hill_climb(best_h, dl2, rng, max_rounds=60)
    if c5_2 < best_c5:
        best_h, best_c5 = h2, c5_2

    # Phase 3: Two-opt for more global moves (15% time)
    dl3 = t0 + budget_s * 0.60
    h3, c5_3 = two_opt_search(best_h, dl3, rng)
    if c5_3 < best_c5:
        best_h, best_c5 = h3, c5_3
    h3b, c5_3b = hill_climb(best_h, dl3 + budget_s * 0.05, rng, max_rounds=30)
    if c5_3b < best_c5:
        best_h, best_c5 = h3b, c5_3b

    # Phase 4: Multi-resolution exploration
    elapsed = time.time() - t0
    remaining = budget_s - elapsed - 10
    if remaining > 30:
        for ntr in [150, 250, 300]:
            if remaining < 25:
                break
            x_old = np.linspace(0.0, 2.0, len(best_h), endpoint=False)
            x_new = np.linspace(0.0, 2.0, ntr, endpoint=False)
            hi = np.interp(x_new, x_old, best_h, period=2.0)
            hi = to_binary(hi, ntr)
            phase_dl = t0 + budget_s * 0.88
            ht, c5t = sa_search(hi, phase_dl, rng, T_init=0.03)
            ht2, c5t2 = hill_climb(ht, phase_dl, rng, max_rounds=20)
            if c5t2 < c5t:
                ht, c5t = ht2, c5t2
            if c5t < best_c5:
                best_h, best_c5 = ht, c5t
            remaining = budget_s - (time.time() - t0) - 10

    # Phase 5: Final hill climb polish
    if time.time() < t0 + budget_s - 3:
        hf, c5f = hill_climb(best_h, t0 + budget_s - 1, rng, max_rounds=40)
        if c5f < best_c5:
            best_h, best_c5 = hf, c5f

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
