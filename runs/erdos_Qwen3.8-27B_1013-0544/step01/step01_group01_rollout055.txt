def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_and_grad(x, beta):
    m = np.max(x)
    shifted = beta * (x - m)
    exp_shifted = np.exp(shifted - np.max(shifted))  # numerical stability
    weights = exp_shifted / exp_shifted.sum()
    value = m + np.log(exp_shifted.sum()) / beta
    return value, weights

def objective_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    value, weights = smooth_max_and_grad(corr, beta)
    
    grad = np.zeros(n)
    for m in range(2*n-1):
        w = weights[m]
        if w < 1e-12:
            continue
        k = m - (n-1)
        # d(corr[m])/d(h[j]) = (1-h[j-k]) * 1{0≤j-k<n} - h[j+k] * 1{0≤j+k<n}
        if k >= 0:
            # j-k ≥ 0 → j ≥ k; j-k ≤ n-1 → j ≤ n-1
            # j+k ≤ n-1 → j ≤ n-1-k
            # First term: (1-h[j-k]) for j = k, ..., n-1
            grad[k:] += w * dx * (1 - h[:n-k])
            # Second term: -h[j+k] for j = 0, ..., n-1-k
            grad[:n-k] -= w * dx * h[k:]
        else:
            k2 = -k
            # j-k = j+k2 ≥ 0 → j ≥ 0; j+k2 ≤ n-1 → j ≤ n-1-k2
            # j+k = j-k2 ≥ 0 → j ≥ k2; j-k2 ≤ n-1 → j ≤ n-1
            # First term: (1-h[j+k2]) for j = 0, ..., n-1-k2
            grad[:n-k2] += w * dx * (1 - h[k2:])
            # Second term: -h[j-k2] for j = k2, ..., n-1
            grad[k2:] -= w * dx * h[:n-k2]
    
    return value, grad

def project_feasible(h, target_sum):
    """Project onto {0≤h≤1, sum(h)=target_sum} via bisection"""
    lo, hi = -2.0, 2.0
    for _ in range(80):

        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val(x, beta):
    m = np.max(x)
    shifted = beta * (x - m)
    e = np.exp(shifted)
    return m + np.log(np.sum(e)) / beta

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0.0, 1.0)

def erdos_obj(x, n, beta, penalty):
    target_sum = n / 2.0
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    val = smooth_max_val(corr, beta)
    val += penalty * (x.sum() - target_sum) ** 2
    return val

    bounds = Bounds(np.zeros(n), np.ones(n))
    res = minimize(erdos_obj, h0, method='L-BFGS-B', bounds=bounds,
                   args=(n, beta, penalty),
                   options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10})
    h_new = project_feasible(res.x, target_sum)
    return h_new, compute_c5(h_new)
