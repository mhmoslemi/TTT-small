def project(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum}"""
    h = np.clip(h, 0.0, 1.0)

```python
def compute_smooth_gradient(h, weights):
    """Compute gradient of weighted sum of correlation lags."""
    n = len(h)
    grad = np.zeros(n)
    for j in range(len(weights)):
        w = weights[j]
        if w < 1e-8:
            continue
        k = j - (n - 1)
        # First term: (1-h[m-k]) for valid m
        lo = max(0, k)
        hi = min(n, n + k)
        if hi > lo:
            grad[lo:hi] += w * (1.0 - h[lo-k:hi-k])
        # Second term: -h[m+k] for valid m
        lo2 = max(0, -k)
        hi2 = min(n, n - k)
        if hi2 > lo2:
            grad[lo2:hi2] -= w * h[lo2+k:hi2+k]
    return grad, n
```

This uses numpy vectorization for each lag, so each lag is O(n) but with numpy overhead. For 239 lags and n=120, this should be fast enough (maybe ~0.1ms per call).

With 50000 iterations, that's 50000 * 0.1ms = 5 seconds. That's acceptable!

But we also need to compute the correlation each iteration, which is np.correlate(h, 1-h, "full") - this is O(n²) = O(14400) ≈ 0.01ms for n=120.

Total per iteration: ~0.1ms + 0.01ms ≈ 0.11ms. For 50000 iterations: ~5.5 seconds.

import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_grad(h, corr, tau):
    n = len(h)
    weights = np.exp((corr - corr.max()) / tau)
    weights /= weights.sum()
    grad = np.zeros(n)
    for j in range(len(corr)):
        w = weights[j]
        if w < 1e-10:
            continue
        k = j - (n - 1)
        lo = max(0, k)
        hi = min(n, n + k)
        if hi > lo:
            grad[lo:hi] += w * (1.0 - h[lo-k:hi-k])
        lo2 = max(0, -k)
        hi2 = min(n, n - k)
        if hi2 > lo2:
            grad[lo2:hi2] -= w * h[lo2+k:hi2+k]
    return grad
