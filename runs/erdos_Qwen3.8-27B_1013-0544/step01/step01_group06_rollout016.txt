        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_corr_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    s = np.sum(w)
    if s < 1e-300:
        w = np.zeros(len(corr))
        w[np.argmax(corr)] = 1.0
    else:
        w /= s
    grad = np.zeros(n)
    active = np.where(w > 1e-12)[0]
    for idx in active:
        k = idx - (n - 1)
        wk = w[idx]
        if k == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif k > 0:
            grad[k:] += wk * (1.0 - h[:n-k])
            grad[:n-k] -= wk * h[k:]
        else:
            p = -k
            grad[:n-p] += wk * (1.0 - h[p:])
            grad[p:] -= wk * h[:n-p]
    return grad * (2.0 / n)

def adam_optimize(h0, target_sum, deadline, rng, alpha=200, lr_init=0.003):
    n = len(h0)
    h = project_simplex(h0.astype(float), target_sum)
    best_h = h.copy()
    best_val = compute_c5(best_h)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = lr_init
    patience = 0
    best_lr = lr_init
    stall_count = 0
    
    while time.time() < deadline - 1:
        t += 1
        grad = smooth_corr_grad(h, alpha)
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * grad**2
        t1 = 1 - 0.9**t
        t2 = 1 - 0.999**t
        h_new = h - lr * (m / t1) / (np.sqrt(v / t2) + 1e-8)
        h_new = project_simplex(h_new, target_sum)
        val = compute_c5(h_new)
        
        if val < best_val - 1e-14:
            best_h = h_new
            best_val = val
            h = h_new
            patience = 0
            lr = min(lr * 1.005, 0.04)
            best_lr = lr
        else:
            patience += 1
            lr *= 0.97
            if patience > 60:
                stall_count += 1
                size = 0.005 + 0.01 * min(stall_count, 5)
                h = best_h + rng.randn(n) * size
                h = project_simplex(h, target_sum)
                m[:] = 0.0
                v[:] = 0.0
                t = 0
                lr = max(best_lr, 0.002)
                patience = 0
                alpha = min(alpha * 1.2, 800.0)
    
    return best_h, best_val
