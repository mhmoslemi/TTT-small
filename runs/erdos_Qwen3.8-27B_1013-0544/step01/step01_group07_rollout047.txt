    k_star = np.argmax(corr)

        # = sum_{n=0}^{N-1-lag} h[n+lag] - sum_{n=0}^{N-1-lag} h[n+lag]*h[n]

        # If i = n+lag for some n in [0, N-1-lag], i.e., i in [lag, N-1]:
        #   contributes 1 - h[i-lag]
        # If i = n for some n in [0, N-1-lag], i.e., i in [0, N-1-lag]:

        for i in range(n - lag):
            g[i] -= h[i + lag]

            best_h, best_c5 = nh, nc5

    # Phase 8: Final SLSQP polish
    if time.time() - t0 < budget_s - 10:
        cd = 2.0 / best_n
        ct = best_n / 2.0
        h_fin = slsqp_opt(best_h, cd, ct, 500, 3000)
        c5_fin = compute_c5(h_fin, cd)
        if c5_fin < best_c5:
            best_h, best_c5 = h_fin, c5_fin

    # Ensure valid output
    best_h = np.clip(best_h, 0.0, 1.0)
    dx_f = 2.0 / best_n
    best_c5 = compute_c5(best_h, dx_f)

    return best_h, best_c5, best_n
