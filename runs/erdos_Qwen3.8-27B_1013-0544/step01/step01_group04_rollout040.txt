```python
def compute_gradient(h, n, dx, tau):
    c = np.correlate(h, 1-h, mode='full') * dx
    # softmax weights
    c_shifted = c - np.max(c)  # for numerical stability
    p = np.exp(tau * c_shifted) / np.sum(np.exp(tau * c_shifted))
    
    # Compute gradient: ∇_j F = dx * Σ_k p_k * ∂c_k/∂h_j
    # ∂c_k/∂h_j = (1-h[j-shift_k]) * I[0≤j-shift_k<n] - h[j+shift_k] * I[0≤j+shift_k<n]
    # where shift_k = k - (n-1)
    
    grad = np.zeros(n)
    for k in range(2*n - 1):
        shift = k - (n - 1)
        # First term: (1-h[j-shift]) for j where 0 ≤ j-shift < n
        j_min = max(0, shift)  # j-shift ≥ 0 => j ≥ shift
        j_max = min(n, n + shift)  # j-shift < n => j < n+shift
        for j in range(j_min, j_max):
            if p[k] > 0:
                grad[j] += p[k] * (1 - h[j - shift])
        # Second term: -h[j+shift] for j where 0 ≤ j+shift < n
        j_min2 = max(0, -shift)  # j+shift ≥ 0 => j ≥ -shift
        j_max2 = min(n, n - shift)  # j+shift < n => j < n-shift
        for j in range(j_min2, j_max2):
            if p[k] > 0:
                grad[j] += -p[k] * h[j + shift]
    
    return dx * grad
```

```python
def compute_gradient_vec(h, n, dx, tau):
    c = np.correlate(h, 1-h, mode='full') * dx
    c_max = np.max(c)
    p = np.exp(tau * (c - c_max)) / np.sum(np.exp(tau * (c - c_max)))
    
    grad = np.zeros(n)
    for k in range(2*n - 1):
        if p[k] < 1e-15:
            continue
        shift = k - (n - 1)
        # First term: add p[k] * (1-h[j-shift]) for j in [max(0,shift), min(n, n+shift))
        j_lo = max(0, shift)
        j_hi = min(n, n + shift)
        if j_hi > j_lo:
            grad[j_lo:j_hi] += p[k] * (1 - h[j_lo - shift : j_hi - shift])
        # Second term: subtract p[k] * h[j+shift] for j in [max(0,-shift), min(n, n-shift))
        j_lo2 = max(0, -shift)
        j_hi2 = min(n, n - shift)
        if j_hi2 > j_lo2:
            grad[j_lo2:j_hi2] -= p[k] * h[j_lo2 + shift : j_hi2 + shift]
    
    return dx * grad
```

```python
    return project_feasible(h, n)

def structured_init(n, rng):
    h = np.zeros(n)
    block = max(1, n // 6)
    for i in range(0, n, 2 * block):
        h[i:i + block] = 1.0
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    dx = 2.0 / n

    best_h = project_feasible(h0, n)
    best_c5 = compute_c5(best_h, dx)

    def elapsed():
        return time.time() - start_time

    # Phase 1: Optimize at original resolution with multi-tau gradient descent
    phase1_end = budget_s * 0.40
    starting_points = [best_h]
    for tau in [30, 80, 200, 500]:
        if elapsed() > phase1_end:
            break
        time_left = phase1_end - elapsed()
        per_call = max(8.0, time_left * 0.35)
        for sp in starting_points[:2]:
            if elapsed() > phase1_end:
                break
            h_new, c5_new = pgd(sp, n, dx, tau, per_call, rng)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new
            starting_points.insert(0, h_new)
            if len(starting_points) > 3:
                starting_points.pop()

    # Phase 2: Upsample to 2n and optimize
    phase2_end = budget_s * 0.68
    n2 = 2 * n
    dx2 = 2.0 / n2
    if elapsed() < phase2_end:
        h_up = upsample_linear(best_h, n2)
        h_up = project_feasible(h_up, n2)
        for tau in [80, 300]:
            if elapsed() > phase2_end:
                break
            time_per = max(8.0, (phase2_end - elapsed()) * 0.4)
            h_new, c5_new = pgd(h_up, n2, dx2, tau, time_per, rng)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new
            h_up = h_new

    # Phase 3: Try 3n resolution
    phase3_end = budget_s * 0.82
    n3 = 3 * n
    dx3 = 2.0 / n3
    if elapsed() < phase3_end and n3 <= 500:
        h_up3 = upsample_linear(best_h, n3)
        h_up3 = project_feasible(h_up3, n3)
        for tau in [100, 400]:
            if elapsed() > phase3_end:
                break
            time_per = max(8.0, (phase3_end - elapsed()) * 0.4)
            h_new, c5_new = pgd(h_up3, n3, dx3, tau, time_per, rng)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new
            h_up3 = h_new

    # Phase 4: Final high-tau polishing at best resolution
    n_final = len(best_h)
    dx_final = 2.0 / n_final
    phase4_end = budget_s - 25
    for tau in [800, 2000, 5000]:
        if elapsed() > phase4_end:
            break
        time_per = max(5.0, (phase4_end - elapsed()) * 0.45)
        h_new, c5_new = pgd(best_h, n_final, dx_final, tau, time_per, rng)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 5: Very fine polishing
    if elapsed() < budget_s - 10:
        time_left = budget_s - 15 - elapsed()
        if time_left > 5:
            h_new, c5_new = pgd(best_h, n_final, dx_final, 10000, time_left, rng)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new

    # Final feasibility check
    best_h = project_feasible(best_h, n_final)
    best_c5 = compute_c5(best_h, dx_final)

    return (best_h, best_c5, n_final)
```
