import numpy as np
import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def compute_corr(h):
    return np.correlate(h, 1 - h, mode='full')

def project_simplex_box(h, target_sum):
    """Project h onto {x: 0<=x<=1, sum(x)=target_sum}"""
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def grad_corr_at_lag(h, k):
    """Gradient of c(k) = sum_i h[i]*(1-h[i+k]) with respect to h"""
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        # c(k) = sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])
        g[:n-k] += (1.0 - h[k:])  # d/dh[i] of h[i]*(1-h[i+k]) = 1-h[i+k]
        if k > 0:
            g[k:] -= h[:n-k]  # d/dh[i+k] of h[i]*(1-h[i+k]) = -h[i]
    else:
        p = -k
        # c(-p) = sum_{i=p}^{n-1} h[i]*(1-h[i-p])
        # Same as c(p) by symmetry of correlate
        g[p:] += (1.0 - h[:n-p])
        g[:n-p] -= h[p:]
    return g

def exact_subgradient(h, top_m=5):
    """Exact subgradient using top-m active lags"""
    n = len(h)
    corr = compute_corr(h)
    # Find top m indices
    idx = np.argsort(corr)[-top_m:]
    weights = corr[idx] - corr[idx[0]] + 1.0  # soft weights
    weights /= weights.sum()
    grad = np.zeros(n)
    for i, k_idx in zip(weights, idx):
        k = k_idx - (n - 1)
        grad += i * grad_corr_at_lag(h, k)
    return grad * (2.0 / n)

def adam_optimize(h0, target_sum, deadline, alpha=100):
    n = len(h0)
    h = h0.copy()
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = 0.02
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    while time.time() < deadline:
        t += 1
        grad = exact_subgradient(h, top_m=5)
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad**2
        m_hat = m / (1 - b1**t)
        v_hat = v / (1 - b2**t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_simplex_box(h_new, target_sum)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = c5
        h = h_new
        lr = 0.02 / (1 + 0.001 * t)
    
    return best_h, best_c5

    starts.append(project_simplex_box(make_sine_start(n, 1.0, 2, 0.0), target))
    starts.append(project_simplex_box(make_sine_start(n, 1.5, 3, 0.7), target))
    starts.append(project_simplex_box(make_sine_start(n, 2.0, 2, 1.3), target))
    starts.append(project_simplex_box(rng.rand(n), target))
    starts.append(project_simplex_box(make_block_start(n), target))
    starts.append(project_simplex_box(make_tri_start(n), target))
    return starts

def refine_resolution(h_best, n_new, deadline, rng, target_sum):
    x_old = np.linspace(0.0, 2.0, len(h_best), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_interp = np.interp(x_new, x_old, h_best)
    h_interp = project_simplex_box(h_interp, target_sum)
    return optimize_adam(h_interp, target_sum, deadline, 0.03, 10, 80, rng)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h_init = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)

    best_h = None
    best_c5 = float('inf')

    # Phase 1: Multi-start at multiple resolutions (70% of budget)
    phase1_end = t0 + budget_s * 0.70
    resolutions = [80, 120, 160, 200, 250]

    for n in resolutions:
        if time.time() > phase1_end:
            break
        target = n / 2.0
        starts = generate_starts(n, rng, h_init)
        n_starts = len(starts)
        time_left = phase1_end - time.time()
        per_start = max(15.0, time_left / (n_starts * 1.3))

        for sp in starts:
            if time.time() > phase1_end:
                break
            deadline = min(time.time() + per_start, phase1_end)
            h_opt, c5_opt = optimize_adam(sp, target, deadline, 0.05, 10, 50, rng)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

        # Continue from best at this resolution
        if best_h is not None and len(best_h) == n:
            deadline = min(time.time() + 25, phase1_end)
            h_opt, c5_opt = optimize_adam(best_h, target, deadline, 0.02, 12, 100, rng)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Phase 2: Increase resolution of best solution (15% of budget)
    phase2_end = t0 + budget_s * 0.85
    if best_h is not None and time.time() < phase2_end:
        n_cur = len(best_h)
        for n_try in [200, 260, 320, 400]:
            if n_try <= n_cur or time.time() > phase2_end:
                continue
            deadline = min(time.time() + 40, phase2_end)
            h2, c5_2 = refine_resolution(best_h, n_try, deadline, rng, n_try / 2.0)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h2
                n_cur = n_try

    # Phase 3: Final polish (15% of budget)
    if best_h is not None and time.time() < t0 + budget_s - 5:
        deadline = t0 + budget_s - 2
        best_h, best_c5 = optimize_adam(best_h, len(best_h) / 2.0, deadline,
                                        0.008, 15, 150, rng)

    best_h = project_simplex_box(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
