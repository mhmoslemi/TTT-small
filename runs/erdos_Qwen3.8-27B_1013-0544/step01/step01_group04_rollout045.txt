        else:

            h_try = best_h + rng.randn(n) * scale
            h_try = project_feasible(h_try, n)
        h_new = slsqp_run(h_try, n, dx, 0, 400, 1e-14)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5 - 1e-14:
            best_h = h_new
            best_c5 = c5_new
        # Also try soft-max with high alpha from this point
        if attempt % 3 == 0 and elapsed() < phase3_end:
            h_new2 = slsqp_run(h_new, n, dx, 2000, 200, 1e-14)
            c5_new2 = compute_c5(h_new2, dx)
            if c5_new2 < best_c5 - 1e-14:
                best_h = h_new2
                best_c5 = c5_new2

    # Phase 4: Resolution increase to 2n
    phase4_end = budget_s * 0.82
    if elapsed() < phase4_end:
        n2 = 2 * n
        dx2 = 2.0 / n2
        _global_target = n2 / 2.0
        h_up = np.empty(n2)
        h_up[0::2] = best_h
        h_up[1::2] = best_h
        h_up = project_feasible(h_up, n2)
        for alpha in [50, 200, 500]:
            if elapsed() > phase4_end:
                break
            h_new2 = slsqp_run(h_up, n2, dx2, alpha, 150, 1e-14)
            c5_new2 = compute_c5(h_new2, dx2)
            if c5_new2 < best_c5 - 1e-14:
                best_h = h_new2
                best_c5 = c5_new2
                h_up = h_new2.copy()
            else:
                h_up = project_feasible(h_new2, n2)
        # SLSQP free at 2n
        if elapsed() < phase4_end:
            h_new2 = slsqp_run(h_up, n2, dx2, 0, 300, 1e-14)
            c5_new2 = compute_c5(h_new2, dx2)
            if c5_new2 < best_c5 - 1e-14:
                best_h = h_new2
                best_c5 = c5_new2

    # Phase 5: Final polishing
    while elapsed() < budget_s - 15:
        n_f = len(best_h)
        dx_f = 2.0 / n_f
        _global_target = n_f / 2.0
        h_try = best_h + rng.randn(n_f) * 0.008
        h_try = project_feasible(h_try, n_f)
        h_new = slsqp_run(h_try, n_f, dx_f, 0, 300, 1e-15)
        c5_new = compute_c5(h_new, dx_f)
        if c5_new < best_c5 - 1e-14:
            best_h = h_new
            best_c5 = c5_new

    final_h = project_feasible(best_h, len(best_h))
    final_c5 = compute_c5(final_h, 2.0 / len(final_h))
    return (final_h, final_c5, len(final_h))
