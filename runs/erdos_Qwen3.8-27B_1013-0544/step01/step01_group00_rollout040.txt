```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_h(h, target):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def t_objective(v):
    return v[-1]

def t_ineq(v):
    n = len(v) - 1
    h = v[:n]
    t = v[-1]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return t - corr

def t_eq_fun(v):
    n = len(v) - 1
    return np.sum(v[:n]) - n * 0.5

def t_eq_jac(v):
    n = len(v) - 1
    j = np.zeros(len(v))
    j[:n] = 1.0
    return j

def lse_obj(h, temp):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    return m + temp * np.log(np.sum(np.exp((corr - m) / temp)))

def lse_grad(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    exp_vals = np.exp((corr - m) / temp)
    weights = exp_vals / np.sum(exp_vals)
    v = 1.0 - h
    h_rev = h[::-1]
    grad = np.zeros(n)
    for j in range(n):
        k_lo = max(0, n - 1 - j)
        part1 = np.dot(weights[k_lo:k_lo + n], v)
        part2 = np.dot(weights[j:j + n], h_rev)
        grad[j] = dx * (part1 - part2)
    return grad

def sum_eq_fun(h):
    return np.sum(h) - len(h) * 0.5

def sum_eq_jac(h):
    return np.ones(len(h))

def optimize_t_var(h0, max_iter=400, ftol=1e-10):
    n = len(h0)
    c5_cur = compute_c5(h0)
    v0 = np.append(h0.copy(), c5_cur + 0.005)
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.5)]
    cons = [
        {'type': 'ineq', 'fun': t_ineq},
        {'type': 'eq', 'fun': t_eq_fun, 'jac': t_eq_jac}
    ]
    res = minimize(t_objective, v0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': ftol})
    h_new = np.clip(res.x[:n], 0.0, 1.0)
    h_new = project_h(h_new, n * 0.5)
    return h_new, compute_c5(h_new)

def optimize_lse(h0, temp, max_iter=400):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq_fn, 'jac': sum_eq_jac}]
    res = minimize(lse_obj, h0.copy(), args=(temp,), jac=lse_grad, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12})
    h_new = np.clip(res.x, 0.0, 1.0)
    h_new = project_h(h_new, n * 0.5)
    return h_new, compute_c5(h_new)

def sum_eq_fn(h):
    return np.sum(h) - len(h) * 0.5

def interp_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 1, n_old, endpoint=False)
    x_new = np.linspace(0, 1, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_h(h_new, n_new * 0.5)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - t0)

    h = np.array(initial_h_values, dtype=float)
    n = len(h)
    h = project_h(h, n * 0.5)
    c5 = compute_c5(h)
    h_best = h.copy()
    c5_best = c5

    # Phase 1: Quick t-var to get close
    if time_left() > 150:
        h_new, c5_new = optimize_t_var(h, max_iter=300, ftol=1e-10)
        if c5_new < c5_best:
            h, c5 = h_new, c5_new
            h_best, c5_best = h.copy(), c5

    # Phase 2: LSE with analytical gradient, decreasing temperature
    temps = [0.04, 0.02, 0.01, 0.005, 0.003, 0.002, 0.001, 0.0005, 0.0003, 0.0001]
    for temp in temps:
        if time_left() < 100:
            break
        h_new, c5_new = optimize_lse(h, temp, max_iter=500)
        if c5_new < c5_best:
            h, c5 = h_new, c5_new
            h_best, c5_best = h.copy(), c5

    # Phase 3: t-var polish
    if time_left() > 150:
        h_new, c5_new = optimize_t_var(h_best, max_iter=600, ftol=1e-11)
        if c5_new < c5_best:
            h_best, c5_best = h_new.copy(), c5_new
            h = h_new
            c5 = c5_new

    # Phase 4: Multi-start with diverse perturbations
    perturbations = []
    # Gaussian
    perturbations.append(h_best + rng.randn(n) * 0.03)
    # Larger Gaussian
    perturbations.append(h_best + rng.randn(n) * 0.06)
    # Fourier perturbation
    hf = np.fft.rfft(h_best - n * 0.5 / n)
    hf *= (1.0 + rng.randn(len(hf)) * 0.15)
    perturbations.append(np.fft.irfft(hf, n) + n * 0.5 / n)
    # Smooth bump
    h_bump = h_best.copy()
    center = rng.randint(10, n - 10)
    amp = rng.uniform(-0.08, 0.08)
    width = rng.uniform(8, 25)
    for j in range(n):
        d = min(abs(j - center), n - abs(j - center))
        h_bump[j] += amp * np.exp(-d * d / (2 * width * width))
    perturbations.append(h_bump)
    # Swap perturbation
    h_sw = h_best.copy()
    n_swaps = 8
    for _ in range(n_swaps):
        i1, i2 = rng.choice(n, 2, replace=False)
        h_sw[i1], h_sw[i2] = h_sw[i2], h_sw[i1]
    perturbations.append(h_sw)
    # Small smooth noise
    perturbations.append(h_best + rng.randn(n) * 0.015)
    # Low-freq perturbation
    h_low = h_best.copy()
    freq = rng.uniform(1, 3)
    phase = rng.uniform(0, 2 * np.pi)
    h_low += 0.04 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
    perturbations.append(h_low)
    # Another Fourier with different scale
    hf2 = np.fft.rfft(h_best - n * 0.5 / n)
    hf2[3:] *= (1.0 + rng.randn(len(hf2) - 3) * 0.25)
    perturbations.append(np.fft.irfft(hf2, n) + n * 0.5 / n)

    for i, h_p in enumerate(perturbations):
        if time_left() < 100:
            break
        h_p = project_h(h_p, n * 0.5)
        # Quick LSE then t-var
        h_new, c5_new = optimize_lse(h_p, 0.01, max_iter=200)
        if time_left() > 80:
            h_new, c5_new = optimize_t_var(h_new, max_iter=300, ftol=1e-10)
        if c5_new < c5_best:
            h_best, c5_best = h_new.copy(), c5_new
            h, c5 = h_new, c5

    # Phase 5: Final polish
    if time_left() > 120:
        h_new, c5_new = optimize_t_var(h_best, max_iter=800, ftol=1e-12)
        if c5_new < c5_best:
            h_best, c5_best = h_new.copy(), c5_new

    if time_left() > 80:
        h_new, c5_new = optimize_lse(h_best, 0.0005, max_iter=400)
        if c5_new < c5_best:
            h_best, c5_best = h_new.copy(), c5_new
        if time_left() > 60:
            h_new, c5_new = optimize_t_var(h_best, max_iter=500, ftol=1e-12)
            if c5_new < c5_best:
                h_best, c5_best = h_new.copy(), c5_new

    # Phase 6: Try higher resolution if time allows
    for n2 in [240, 300]:
        if time_left() < 250:
            break
        h2 = interp_h(h_best, n2)
        c5_2 = compute_c5(h2)
        h_cur2, c5_cur2 = h2, c5_2
        if time_left() > 150:
            h_cur2, c5_cur2 = optimize_lse(h_cur2, 0.02, max_iter=300)
        if time_left() > 120:
            h_cur2, c5_cur2 = optimize_lse(h_cur2, 0.005, max_iter=300)
        if time_left() > 100:
            h_cur2, c5_cur2 = optimize_t_var(h_cur2, max_iter=400, ftol=1e-11)
        if time_left() > 60 and c5_cur2 < c5_best:
            h_new, c5_new = optimize_lse(h_cur2, 0.002, max_iter=200)
            if c5_new < c5_cur2:
                h_cur2, c5_cur2 = h_new, c5_new
            if time_left() > 40:
                h_new, c5_new = optimize_t_var(h_cur2, max_iter=300, ftol=1e-11)
                if c5_new < c5_cur2:
                    h_cur2, c5_cur2 = h_new, c5_new
        if c5_cur2 < c5_best:
            h_best, c5_best = h_cur2.copy(), c5_cur2

    # Final validation
    h_best = project_h(h_best, len(h_best) * 0.5)
    c5_final = compute_c5(h_best)
    return h_best, c5_final, len(h_best)
```
