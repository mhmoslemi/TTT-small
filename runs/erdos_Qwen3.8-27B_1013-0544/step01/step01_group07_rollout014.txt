import numpy as np

from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def get_weights(corr, t):
    max_c = np.max(corr)
    shifted = t * (corr - max_c)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    if s < 1e-300:
        return np.zeros_like(corr)
    return exp_vals / s

def compute_grad(h, weights, dx):
    n = len(h)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-14:
            continue
        offset = m - (n - 1)
        i1 = max(0, -offset)
        i2 = min(n, n - offset)
        if i1 < i2:
            grad[i1:i2] += w * (1.0 - h[i1 + offset:i2 + offset])
        i3 = max(0, offset)
        i4 = min(n, n + offset)
        if i3 < i4:
            grad[i3:i4] -= w * h[i3 - offset:i4 - offset]
    return grad * dx

def project_box_sum(h, target_sum):
    n = len(h)
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target_sum - np.sum(proj)
    if abs(diff) > 1e-13:
        mask = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        if np.sum(mask) > 0:
            proj[mask] += diff / np.sum(mask)
            proj = np.clip(proj, 0.0, 1.0)
    return proj

    kk = 1.0

        g = compute_grad(y, weights, dx)
        x_new = project_box_sum(y - eta * g, target_sum)

        kk = kk_new

    c5 = compute_c5(h, dx)

        best_h = h.copy()
    return best_h, best_c5

def slsqp_optimize(h, dx, target_sum, t=50, maxiter=400):
    n = len(h)
    def obj(x):
        c = np.correlate(x, 1 - x, mode="full")
        mc = np.max(c)
        sh = t * (c - mc)
        return (mc + (1.0 / t) * np.log(np.sum(np.exp(sh)))) * dx
    def obj_grad(x):
        c = np.correlate(x, 1 - x, mode="full")
        w = get_weights(c, t)
        return compute_grad(x, w, dx)
    def constraint(x):
        return np.sum(x) - target_sum
    def constraint_jac(x):
        return np.ones(n)
    try:
        res = minimize(obj, h.copy(), jac=obj_grad, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': constraint, 'jac': constraint_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        h_new = project_box_sum(res.x, target_sum)
        return h_new, compute_c5(h_new, dx)
    except Exception:
        return h, compute_c5(h, dx)

def pair_descent(h, dx, target_sum, n_rounds=200, seed=42):
    n = len(h)
    rng = np.random.RandomState(seed)
    best_c5 = compute_c5(h, dx)

        s = h[i] + h[j]
        d_lo = max(-s / 2.0, s / 2.0 - 1.0)
        d_hi = min(s / 2.0, 1.0 - s / 2.0)
        if d_lo >= d_hi - 1e-12:
            continue
        def eval_d(d):
            h[i] = s / 2.0 + d
            h[j] = s / 2.0 - d
            return compute_c5(h, dx)
        a, b = d_lo, d_hi
        c_ = b - gr * (b - a)
        d_ = a + gr * (b - a)

        fd = eval_d(d_)

            else:

                c_ = d_
                fc = fd
                d_ = a + gr * (b - a)
                fd = eval_d(d_)

            eval_d(c_)
        else:
            eval_d(d_)
        c5_now = compute_c5(h, dx)
