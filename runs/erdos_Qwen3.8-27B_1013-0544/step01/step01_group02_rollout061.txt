def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-14:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def cd_optimize(h, n, n_sweeps, rng, recompute_interval=5):
    """Coordinate descent with exact 1D subproblem."""
    c = np.correlate(h, 1.0 - h, mode='full')
    for sweep in range(n_sweeps):
        order = rng.permutation(n)
        for count, i in enumerate(order):
            # Compute gradient of c w.r.t. h[i]
            m_all = np.arange(2 * n - 1)
            idx1 = i + m_all - (n - 1)
            idx2 = i - (m_all - (n - 1))
            v1 = (idx1 >= 0) & (idx1 < n)
            v2 = (idx2 >= 0) & (idx2 < n)
            grad = np.zeros(2 * n - 1)
            grad[v1] += 1.0 - h[idx1[v1]]
            grad[v2] -= h[idx2[v2]]

                    a = mid

                c += delta * grad

        if (sweep + 1) % recompute_interval == 0:
            c = np.correlate(h, 1.0 - h, mode='full')

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-14:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def coord_grad(h, i, n):
    """Compute ∂c/∂h[i] where c = correlate(h, 1-h, 'full')."""
    m_all = np.arange(2 * n - 1)
    idx1 = i + m_all - (n - 1)
    idx2 = i - (m_all - (n - 1))
    v1 = (idx1 >= 0) & (idx1 < n)
    v2 = (idx2 >= 0) & (idx2 < n)
    grad = np.zeros(2 * n - 1)
    grad[v1] += 1.0 - h[idx1[v1]]
    grad[v2] -= h[idx2[v2]]
    return grad

def solve_1d(c, grad, lo, hi):
    """Find delta in [lo, hi] minimizing max(c + delta * grad)."""
    if hi - lo < 1e-15:
        return lo
    a, b = lo, hi
    for _ in range(40):
        mid = 0.5 * (a + b)
        vals = c + mid * grad
        mx = vals.max()
        S = np.where(np.abs(vals - mx) < 1e-10)[0]
        if grad[S].sum() > 0:
            b = mid
        else:
            a = mid
    return 0.5 * (a + b)

def cd_optimize(h, n, n_sweeps, rng):
    """Coordinate descent."""
    c = np.correlate(h, 1.0 - h, mode='full')
    for sweep in range(n_sweeps):
        order = rng.permutation(n)
        for i in order:
            grad = coord_grad(h, i, n)
            delta = solve_1d(c, grad, -h[i], 1.0 - h[i])
            if abs(delta) > 1e-13:
                h[i] += delta
                c += delta * grad
        if (sweep + 1) % 10 == 0:
            c = np.correlate(h, 1.0 - h, mode='full')
    return h

def perturb(h, n, rng, strength=1.0):

    elif ptype == 2:

        i = rng.randint(0, n - length)
        j = rng.randint(0, n - length)
        if i != j:

    elif ptype == 3:
        # Block complement
        length = rng.randint(3, min(15, n // 3 + 1))
        i = rng.randint(0, n - length)
        h_new[i:i+length] = 1.0 - h_new[i:i+length]
    else:

        for i in range(1, n):
            noise[i] = 0.7 * noise[i-1] + 0.3 * rng.normal(0, scale)
        h_new += noise
    return h_new

Let me try a different approach: **subgradient method with projection**.

At each iteration:
1. Compute c = correlate(h, 1-h, 'full')
2. Find m* = argmax c
3. Compute g = ∂c[m*]/∂h (the gradient of the active constraint)
4. Take a step: h ← h - α * g
5. Project onto {0 ≤ h ≤ 1, sum(h) = n/2}

This is O(n) per step (for the gradient) + O(n) for the projection. Much faster per step than CD!

But subgradient methods converge slowly (O(1/√T) rate). However, combined with a decreasing step size, they can make progress.

For the subgradient step:
```python
def subgradient_step(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    m_star = c.argmax()
    # Compute grad of c[m_star] w.r.t. h
    # ∂c[m]/∂h[i] = (1-h[i+m-(n-1)])*[valid] - h[i-m+(n-1)]*[valid]
    i_all = np.arange(n)
    idx1 = i_all + m_star - (n - 1)
    idx2 = i_all - (m_star - (n - 1))
    v1 = (idx1 >= 0) & (idx1 < n)
    v2 = (idx2 >= 0) & (idx2 < n)
    g = np.zeros(n)
    g[v1] += 1.0 - h[idx1[v1]]
    g[v2] -= h[idx2[v2]]
    return g
```

This is O(n) to compute. Then:
```python
h_new = h - alpha * g
h_new = project_feasible(h_new, n)
```

The projection is O(n) per iteration of the inner loop.

Total per subgradient step: O(n) ≈ 80 operations. In Python/NumPy, this is maybe 10-50 microseconds. So 1000 steps take 10-50 milliseconds. Very fast!

But the problem is that subgradient methods oscillate and don't converge to the optimum. They're good for finding the right "region" but not for precise optimization.

Let me use a **bundle method** or **cutting plane** approach:
- Maintain a set of linear cuts (from previous gradients)
- Solve the cutting plane subproblem (find the point that minimizes the max of all cuts)
- Add a new cut
- The cutting plane subproblem is a QP (or LP if we ignore the box constraints)

For n=80, the cutting plane subproblem with K cuts is:
minimize max_{j=1..K} (g_j * h + b_j) + (box and sum constraints)

This is a linear program (or QP with the sum constraint). We can solve it with scipy or a custom solver.

Actually, let me think about this differently. The 

max's cut is g_j · h + c_j where g_j is the gradient at the j-th iterate and c_j = c[m*_j] - g_j · h_j.

The subproblem is:
minimize t
subject to g_j · h + c_j ≤ t for j = 1, ..., K
           0 ≤ h ≤ 1
           sum(h) = n/2

This is a linear program! (The variable t is free, and all constraints are linear.)

I can solve this with scipy.optimize.linprog or with a custom simplex.

But for K cuts and n=80 variables, the LP has n+1 = 81 variables and K + n + 1 ≈ K + 81 constraints. For K = 100, that's 181 constraints and 81 variables. This is very fast to solve (milliseconds).
