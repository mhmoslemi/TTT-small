For the full correlation array of length 2n-1, index m corresponds to lag L = m - (n-1).

corr[m] = sum_{i: valid} h[i] * (1-h[i+L])

    Both if L ≤ j ≤ n-1-L (when 2L ≤ n-1)

            grad[L:] += -h[:n-L]

            grad[L2:] += (1.0 - h[:n-L2])

                    h_hi = h_ref
            except Exception:
                pass

    # Phase 6: Active-set fine gradient (final polish)
    if tl() > 8:
        n_f = len(best_h)
        dx_f = 2.0 / n_f
        target_f = n_f / 2.0
        n_half_f = n_f - 1
        h = best_h.copy()
        ss = 1e-4
        mom = np.zeros(n_f)
        mu = 0.35
        stall = 0

        while tl() > 2:
            corr = np.correlate(h, 1.0 - h, mode='full')
            c_max = float(np.max(corr))
            tol = max(1e-9, c_max * 3e-5)
            act = np.where(corr >= c_max - tol)[0]
            if len(act) == 0:
                act = np.array([int(np.argmax(corr))])
            elif len(act) > 10:
                act = np.argsort(corr)[-10:]

            grad = np.zeros(n_f)
            for idx in act:
                L = int(idx) - n_half_f
                grad += compute_grad_lag(h, L)
            grad /= float(len(act))

            mom = mu * mom - ss * grad
            h_new = project_feasible(h + mom, target_f)
            c5_new = compute_c5(h_new, dx_f)

            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                stall = 0
            else:
                stall += 1
                if stall % 40 == 0:
                    ss *= 0.7
                    mom *= 0.3
                if ss < 1e-14:
                    break
            h = h_new

    return (best_h, best_c5, len(best_h))
