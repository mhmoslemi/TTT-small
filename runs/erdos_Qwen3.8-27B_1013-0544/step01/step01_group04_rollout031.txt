    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - target
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            h[free] -= diff / np.sum(free)
        else:
            h -= diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = np.sum(h) - target
    if abs(diff) > 1e-12:
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            h[free] -= diff / np.sum(free)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_value(h, dx, tau):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    vals = np.exp(corr * tau - m * tau)
    return (m + np.log(np.sum(vals)) / tau) * dx

def smooth_max_gradient(h, n, dx, tau):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    weights = np.exp(corr * tau - m * tau)
    weights /= np.sum(weights)
    
    grad = np.zeros(n)

        # Term 2: -h[j+k] where j+k is valid index
        js2 = max(0, -k)
        je2 = min(n-1, n-1-k)
        if js2 <= je2:

    return grad * dx

                    'jac': lambda x: np.ones(n)}]

                   bounds=bounds, constraints=constraints,
                   options={'maxiter': max_iter, 'ftol': 1e-14})
    return project_feasible(res.x, n)

from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float, copy=True)
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - target
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            h[free] -= diff / np.sum(free)
        else:
            h -= diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = np.sum(h) - target
    if abs(diff) > 1e-10:
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            h[free] -= diff / np.sum(free)
            h = np.clip(h, 0.0, 1.0)
    return h

def _smo_obj(x, n, dx, tau, target, pen):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    vals = np.exp(tau * (corr - m))
    smooth = m + np.log(np.sum(vals)) / tau
    viol = np.sum(x) - target
    return smooth * dx + pen * viol * viol

def _smo_grad(x, n, dx, tau, target, pen):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    exps = np.exp(tau * (corr - m))
    wsum = np.sum(exps)
    weights = exps / wsum
    grad = np.zeros(n)
    ks = np.arange(-(n - 1), n)
    for i in range(len(ks)):
        w = weights[i]
        if w < 1e-14:
            continue
        k = ks[i]
        a1 = max(0, k)
        a2 = min(n - 1, n - 1 + k)
        if a1 <= a2:
            grad[a1:a2+1] += w * (1.0 - x[a1-k:a2-k+1])
        b1 = max(0, -k)
        b2 = min(n - 1, n - 1 - k)
        if b1 <= b2:
            grad[b1:b2+1] += w * (-x[b1+k:b2+k+1])
    grad *= dx
    viol = np.sum(x) - target
    grad += 2.0 * pen * viol
    return grad

def _lbfgs(h0, n, dx, tau, target, iters=300):
    h = h0.astype(float).copy()
    pen = 1.0
    bnds = [(0.0, 1.0)] * n
    for _ in range(6):
        res = minimize(_smo_obj, h, jac=_smo_grad,
                       args=(n, dx, tau, target, pen),
                       method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': iters, 'ftol': 1e-15, 'gtol': 1e-9})
        h = res.x
        if abs(np.sum(h) - target) < 0.005:
            break
        pen *= 8.0
    return project_feasible(h, n)

def _lbfgs_multi(h0, n, dx, tau, target, ntry=2, scale=0.05, rng=None, iters=200):
    best = h0.copy()
    bc = compute_c5(best, dx)
    for i in range(ntry):
        if i == 0:
            h_try = best.copy()
        elif rng is not None:
            h_try = project_feasible(best + rng.randn(n) * scale, n)
        else:
            break
        hn = _lbfgs(h_try, n, dx, tau, target, iters)
        cn = compute_c5(hn, dx)
        if cn < bc:
            best = hn.copy()
            bc = cn
    return best, bc
