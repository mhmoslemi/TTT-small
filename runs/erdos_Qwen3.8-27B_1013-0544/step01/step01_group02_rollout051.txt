def initial_binary_from_continuous(h, n):
    s = (h >= 0.5).astype(int)
    count = s.sum()
    target = n // 2
    if count > target:
        ones = np.where(s == 1)[0]
        order = np.argsort(h[ones])
        for idx in order[:count - target]:
            s[ones[idx]] = 0
    elif count < target:
        zeros = np.where(s == 0)[0]
        order = np.argsort(-h[zeros])
        for idx in order[:target - count]:
            s[zeros[idx]] = 1
    return s

def vdc_binary(n, rng):
    """Generate binary pattern from Van der Corput sequence."""
    points = np.zeros(n)
    for i in range(n):
        v = 0.0
        d = 1.0
        ii = i + 1
        while ii > 0:
            v += (ii % 2) * d
            ii //= 2
            d /= 2
        points[i] = v
    thresh = np.sort(points)[n // 2]
    s = (points >= thresh).astype(int)
    while s.sum() > n // 2:
        idx = rng.choice(np.where(s == 1)[0])
        s[idx] = 0
    while s.sum() < n // 2:
        idx = rng.choice(np.where(s == 0)[0])
        s[idx] = 1
    return s

def random_binary(n, rng):
    s = np.zeros(n, dtype=int)
    s[rng.choice(n, n // 2, replace=False)] = 1
    return s

def polish_continuous(h, n, t0, deadline):
    """Multi-alpha SLSQP polishing."""
    best = h.copy()
    best_c = compute_c5(best, n)
    for alpha in [30, 100, 300, 1000, 3000, 10000]:
        if time.time() > deadline - 5:
            break
        h_try = slsqp_optimize(best, n, alpha, maxiter=3000, ftol=1e-18)
        c = compute_c5(h_try, n)
        if c < best_c - 1e-14:
            best, best_c = h_try, c
    return best, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_default = len(h_init)
    except Exception:
        n_default = 80
        h_init = np.full(n_default, 0.5)
    
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h, n_default)
    best_n = n_default
    
    def try_solution(h, n):
        nonlocal best_h, best_c5, best_n
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c5:
            best_h, best_c5, best_n = h, c, n
        return h, c
    
    # Phase 1: Binary SA at default n with multiple inits
    s0 = initial_binary_from_continuous(h_init, n_default)
    s_a, c_a = sa_binary(n_default, s0, 200000, rng, t0, t0 + 40)
    s_r, c_r = sa_binary(n_default, random_binary(n_default, rng), 100000, rng, t0, t0 + 40)
    s_v, c_v = sa_binary(n_default, vdc_binary(n_default, rng), 100000, rng, t0, t0 + 40)
    
    for s_cand, c_cand in [(s_a, c_a), (s_r, c_r), (s_v, c_v)]:
        if c_cand * (2.0 / n_default) < best_c5 or c_cand < 16:
            h_cand = s_cand.astype(float)
            h_pol, c_pol = polish_continuous(h_cand, n_default, t0, t0 + budget_s * 0.4)
            try_solution(h_pol, n_default)
    
    # Phase 2: Polish original continuous init
    if budget_s - (time.time() - t0) > 100:
        h_pol2, c_pol2 = polish_continuous(h_init, n_default, t0, t0 + budget_s * 0.4)
        try_solution(h_pol2, n_default)
    
    # Phase 3: Try other n values
    for n_try in [64, 72, 88, 96, 100, 120]:
        if budget_s - (time.time() - t0) < 60:
            break
        for s_init in [vdc_binary(n_try, rng), random_binary(n_try, rng)]:
            if time.time() > t0 + budget_s * 0.7:
                break
            s_cand, c_cand = sa_binary(n_try, s_init, 50000, rng, t0, t0 + budget_s * 0.7)
            h_cand = s_cand.astype(float)
            h_pol, c_pol = polish_continuous(h_cand, n_try, t0, t0 + budget_s * 0.8)
            try_solution(h_pol, n_try)
    
    # Phase 4: Perturbation-based local search on best
    if budget_s - (time.time() - t0) > 40:
        n = best_n
        for _ in range(300):
            if time.time() > t0 + budget_s - 20:
                break
            scale = rng.choice([0.003, 0.008, 0.015, 0.03])
            h_pert = best_h + rng.normal(0, scale, size=n)
            h_pert = project_feasible(h_pert, n)
            h_opt = slsqp_optimize(h_pert, n, alpha=1000, maxiter=1500, ftol=1e-16)
            c_opt = compute_c5(h_opt, n)
            if c_opt < best_c5 - 1e-13:
                best_h, best_c5 = h_opt, c_opt
    
    # Phase 5: Final high-alpha polish
    if budget_s - (time.time() - t0) > 15:
        for alpha in [5000, 15000, 30000]:
            if time.time() > t0 + budget_s - 5:
                break
            h_opt = slsqp_optimize(best_h, best_n, alpha, maxiter=5000, ftol=1e-18)
            c_opt = compute_c5(h_opt, best_n)
            if c_opt < best_c5:
                best_h, best_c5 = h_opt, c_opt
    
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return best_h, best_c5, best_n
