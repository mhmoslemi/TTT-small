```python
    G = np.zeros((n_lags, n))

        c_obj[-1] = 1.0

    shifted = alpha * (c - m)

    if mx > 700:
```

For the full correlation:
c[k] = sum_i h[i] * (1-h)[i + k - (n-1)]

Let s = k - (n-1) be the shift. Then:
c[k] = sum_{i: 0<=i<n, 0<=i+s<n} h[i] * (1 - h[i+s])

The gradient w.r.t. h[j]:
∂c[k]/∂h[j] = sum over terms where j appears.

h[j] appears in:
1. Term i=j (if j+s is valid): contributes (1-h[j+s])
2. Term i=j-s (if j-s is valid and j is valid as i+s): contributes -h[j-s]

So:
∂c[k]/∂h[j] = (1-h[j+s]) * [j valid AND j+s valid] + (-h[j-s]) * [j-s valid AND j valid]

Where "j valid" means 0<=j<n, "j+s valid" means 0<=j+s<n, etc.

And the "valid range" for i is: max(0,-s) <= i <= min(n-1, n-1-s).

So:
∂c[k]/∂h[j] = (1-h[j+s]) * [max(0,-s) <= j <= min(n-1,n-1-s) AND 0<=j+s<n]
             + (-h[j-s]) * [max(0,-s) <= j-s <= min(n-1,n-1-s) AND 0<=j<n]

Since j is always in [0,n-1] (it's an index), the condition "0<=j<n" is always true.
And "0<=j+s<n" is equivalent to -s <= j <= n-1-s.
And "max(0,-s) <= j-s <= min(n-1,n-1-s)" is equivalent to max(0,-s)+s <= j <= min(n-1,n-1-s)+s.

Let me simplify:
Condition 1: max(0,-s) <= j <= min(n-1, n-1-s) AND 0 <= j+s <= n-1
  The second part: -s <= j <= n-1-s
  Combined: max(0, -s, -s) <= j <= min(n-1, n-1-s, n-1-s) = min(n-1, n-1-s)
  So: max(0, -s) <= j <= min(n-1, n-1-s)
  Which is just the valid range for i. So condition 1 is: j is in the valid i range.

Condition 2: max(0,-s) <= j-s <= min(n-1, n-1-s) AND 0 <= j <= n-1
  The first part: max(0,-s)+s <= j <= min(n-1,n-1-s)+s

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-15:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def compute_lag_overlaps(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_lag_gradients(h, n):
    n_lags = 2 * n - 1
    G = np.zeros((n_lags, n))
    for k in range(n_lags):
        s = k - (n - 1)
        i_min = max(0, -s)
        i_max = min(n - 1, n - 1 - s)
        for i in range(i_min, i_max + 1):
            G[k, i] += (1.0 - h[i + s])
            G[k, i + s] += (-h[i])
    return G

def benders_optimize(h, n, max_iters=80, time_limit=60):
    h_cur = h.copy()
    t_start = time.time()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur, n)
    for _ in range(max_iters):
        if time.time() - t_start > time_limit:
            break
        overlaps = compute_lag_overlaps(h_cur, n)
        G = compute_lag_gradients(h_cur, n)
        n_lags = len(overlaps)
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_ub = np.zeros((n_lags, n + 1))
        A_ub[:, :n] = G
        A_ub[:, n] = -1.0
        b_ub = G @ h_cur - overlaps
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        bounds_lp = [(0, 1)] * n + [(0, 1)]
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds_lp, method='highs')
        if not res.success:
            break
        h_new = project_feasible(res.x[:n], n)
        c5_new = compute_c5(h_new, n)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        c5_cur = compute_c5(h_cur, n)
        if abs(c5_new - c5_cur) < 1e-12:
            h_cur = h_new
            break
        h_cur = h_new
    return best_h, best_c5

def obj_smooth(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 700:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def sum_con(h, n):
    return np.sum(h) - n / 2.0

def sum_con_jac(h, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_try = project_feasible(res.x, n)
        return h_try, compute_c5(h_try, n)
    except Exception:
        return h, compute_c5(h, n)

def lag_targeted_perturbation(h, n, rng, max_rounds=80, time_limit=60):
    t_start = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    for _ in range(max_rounds):
        if time.time() - t_start > time_limit:
            break
        overlaps = compute_lag_overlaps(best_h, n)
        top_k = max(3, len(overlaps) // 20)
        top_lags = np.argsort(overlaps)[-top_k:]
        G = compute_lag_gradients(best_h, n)
        grad_max = np.zeros(n)
        for lag in top_lags:
            w = 1.0 + 10.0 * (overlaps[lag] / (overlaps.max() + 1e-15))
            grad_max += w * G[lag]
        gn = np.max(np.abs(grad_max)) + 1e-15
        for step_scale in [0.1, 0.05, 0.02, 0.01, 0.005]:
            h_new = best_h - step_scale * grad_max / gn
            h_new = project_feasible(h_new, n)
            c5_new = compute_c5(h_new, n)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
                break
    return best_h, best_c5

def simulated_annealing(h, n, rng, max_rounds=2000, time_limit=60):
    t_start = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    cur_h = h.copy()
    cur_c5 = best_c5
    T = 0.01
    T_min = 0.0001
    cooling = (T_min / T) ** (1.0 / max(1, max_rounds))
    for _ in range(max_rounds):
        if time.time() - t_start > time_limit:
            break
        T *= cooling
        scale = rng.choice([0.01, 0.02, 0.05, 0.1])
        h_new = cur_h + rng.normal(0, scale, size=n)
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new, n)
        delta = c5_new - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / (T + 1e-15)):
            cur_h, cur_c5 = h_new, c5_new
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new
    return best_h, best_c5

def create_structured_start(n, seed, pattern):
    rng = np.random.RandomState(seed)
    if pattern == 0:
        h = np.ones(n) * 0.5
        h[:n // 3] = 1.0
        h[2 * n // 3:] = 0.0
    elif pattern == 1:
        x = np.linspace(0, 2, n)
        h = 0.5 + 0.4 * np.sin(np.pi * x)
    elif pattern == 2:
        x = np.linspace(0, 2, n)
        h = 1.0 - np.abs(x - 1.0)
    elif pattern == 3:
        x = np.linspace(0, 2, n)
        h = 0.3 + 0.7 * np.exp(-((x - 0.5) ** 2) / 0.1) + 0.7 * np.exp(-((x - 1.5) ** 2) / 0.1)
    elif pattern == 4:
        x = np.linspace(0, 2, n)
        h = 0.5 + 0.35 * np.cos(2 * np.pi * x)
    else:
        h = rng.uniform(0, 1, n)
    return project_feasible(np.clip(h, 0, 1), n)

def fine_swaps(h, n, rng, max_iter=20000, time_limit=60):
    t_start = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    deltas = [0.001, 0.002, 0.003, 0.005, 0.008, 0.01, 0.015, 0.02]
    count = 0
    while time.time() - t_start < time_limit and count < max_iter:
        count += 1
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        for d in deltas:
            improved = False
            for s in [1, -1]:
                di = s * d
                hi = best_h[i] + di
                hj = best_h[j] - di
                if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                    h_try = best_h.copy()
                    h_try[i] = hi
                    h_try[j] = hj
                    c = compute_c5(h_try, n)
                    if c < best_c5:
                        best_h = h_try
                        best_c5 = c
                        improved = True
                        break
            if improved:
                break
    return best_h, best_c5

        if n_alt == n0:
            continue
        if budget_s - (time.time() - t0) < 40:
            break
        h_alt = create_structured_start(n_alt, seed + 500, 1)
        h_opt3, c5_opt3 = optimize_for_n(h_alt, n_alt, seed + 500,
                                          budget_s - (time.time() - t0) - 10)
        if c5_opt3 < best_overall_c5:
            best_overall_h, best_overall_c5, best_overall_n = h_opt3, c5_opt3, n_alt

    # Final polish
    if budget_s - (time.time() - t0) > 5:
        h_fin, c5_fin = slsqp_optimize(best_overall_h, best_overall_n, 20000, maxiter=5000)
        if c5_fin < best_overall_c5:
            best_overall_h, best_overall_c5 = h_fin, c5_fin

    best_overall_h = project_feasible(best_overall_h, best_overall_n)
    best_overall_c5 = compute_c5(best_overall_h, best_overall_n)
    return best_overall_h, best_overall_c5, best_overall_n
```
