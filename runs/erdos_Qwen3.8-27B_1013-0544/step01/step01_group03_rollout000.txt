import numpy as np
from scipy.optimize import minimize, Bounds
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val(obj, tau):
    x = np.asarray(obj, dtype=float)
    x_max = np.max(x)
    return float(x_max + np.log(np.sum(np.exp(x - x_max))) / tau)

def fix_sum(h, target):
    h = np.array(h, dtype=float)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return np.clip(h, 0.0, 1.0)
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = mask.sum()
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def adam_optimize(h0, target, tau, lr, max_iters, t0, deadline):
    n = len(h0)
    dx = 2.0 / n
    x = h0.astype(float).copy()
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    best_x = x.copy()
    best_val = smooth_max_val(np.correlate(x, 1 - x, mode="full") * dx, tau)
    m_shifts = np.arange(2 * n) - (n - 1)

    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = np.correlate(x, 1.0 - x, mode="full")
        scaled = corr * dx
        x_max = np.max(scaled)
        exp_s = np.exp(scaled - x_max)
        weights = exp_s / np.sum(exp_s)

        grad = np.zeros(n)
        for k in range(2 * n):
            wk = weights[k]
            if wk < 1e-10:
                continue
            mv = m_shifts[k]
            j1s = max(0, -mv)
            j1e = min(n, n - mv)
            if j1s < j1e:
                grad[j1s:j1e] += wk * (1.0 - x[j1s + mv:j1e + mv])
            j2s = max(0, mv)
            j2e = min(n, n + mv)
            if j2s < j2e:
                grad[j2s:j2e] -= wk * x[j2s - mv:j2e - mv]
        grad *= dx

        it1 = it + 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        m_hat = m / (1 - b1 ** it1)
        v_hat = v / (1 - b2 ** it1)
        x_new = x - lr * m_hat / (np.sqrt(v_hat) + eps)
        x_new = np.clip(x_new, 0.0, 1.0)
        x_new = fix_sum(x_new, target)
        x = x_new
        if (it + 1) % 50 == 0:
            val = smooth_max_val(np.correlate(x, 1 - x, mode="full") * dx, tau)
            if val < best_val:
                best_val = val
                best_x = x.copy()

    val = smooth_max_val(np.correlate(x, 1 - x, mode="full") * dx, tau)
    if val < best_val:
        best_val = val
        best_x = x.copy()
    return best_x

def slsqp_refine(h0, target, maxiter=200):
    n = len(h0)
    bnds = Bounds(np.zeros(n), np.ones(n))

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        dx = 2.0 / n
        x_max = np.max(corr)
        return float(x_max + np.log(np.sum(np.exp(corr - x_max))) / 50.0)

    def jac(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        dx = 2.0 / n
        scaled = corr * dx
        x_max = np.max(scaled)
        exp_s = np.exp(scaled - x_max)
        weights = exp_s / np.sum(exp_s)
        g = np.zeros(n)
        m_shifts = np.arange(2 * n) - (n - 1)
        for k in range(2 * n):
            wk = weights[k]
            if wk < 1e-10:
                continue
            mv = m_shifts[k]
            j1s = max(0, -mv)
            j1e = min(n, n - mv)
            if j1s < j1e:
                g[j1s:j1e] += wk * (1.0 - x[j1s + mv:j1e + mv])
            j2s = max(0, mv)
            j2e = min(n, n + mv)
            if j2s < j2e:
                g[j2s:j2e] -= wk * x[j2s - mv:j2e - mv]
        return g * dx

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': jac_cons}]
    try:
        res = minimize(obj, h0, method='SLSQP', jac=jac, bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return fix_sum(res.x, target)
    except Exception:
        return h0

def jac_cons(x):
    return np.ones(len(x))

def binary_anneal(n, t0, deadline, rng):
    x = np.zeros(n)
    ones = np.random.choice(n, n // 2, replace=False)
    x[ones] = 1.0
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    best_c5 = np.max(corr)
    best_x = x.copy()
    temp = 0.05
    for step in range(50000):
        if time.time() > deadline:
            break
        if step % 500 == 499:
            temp *= 0.95
        i = rng.randint(n)
        j = rng.randint(n)
        if x[i] == x[j]:
            continue
        x[i], x[j] = x[j], x[i]
        corr2 = np.correlate(x, 1.0 - x, mode="full") * dx
        c5_2 = np.max(corr2)
        delta = c5_2 - best_c5
        if delta < 0 or rng.random() < np.exp(-delta / (temp + 1e-12)):
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_x = x.copy()
        else:
            x[i], x[j] = x[j], x[i]
    return best_x, best_c5

def upsample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    old_x = np.linspace(0, 1, n_old, endpoint=False)
    new_x = np.linspace(0, 1, n_new, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=1.0)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):

    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:

    best_h = h_init.copy()

    def rem():

    n0 = len(best_h)

        c5_new = evaluate_c5(h_new)
