import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(x, target_sum):
    lo = float(x.min()) - 1.0
    hi = float(x.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = float(np.clip(x - mid, 0.0, 1.0).sum())
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def soft_grad(h, n_half, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    weights = np.exp(-alpha * (c_max - corr) / max(c_max, 1e-10))
    total = float(np.sum(weights))
    weights /= total
    grad = np.zeros(n)
    for idx in range(len(corr)):
        w = weights[idx]
        if w < 1e-5:
            continue
        L = int(idx) - n_half
        omh = 1.0 - h
        if L == 0:
            grad += w * (omh - h)
        elif L > 0:
            grad[L:] += w * omh[:n - L]
            grad[:n - L] -= w * h[L:]
        else:
            L2 = -L
            grad[:n - L2] += w * omh[L2:]
            grad[L2:] -= w * h[:n - L2]
    return grad

def active_grad(h, n_half, K=5):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    tol = max(0.005, c_max * 0.001)
    active = np.where(corr >= c_max - tol)[0]
    if len(active) < 2:
        active = np.argsort(corr)[-K:]
    elif len(active) > K:
        active = np.argsort(corr)[-K:]
    grad = np.zeros(n)
    for idx in active:
        L = int(idx) - n_half
        omh = 1.0 - h
        if L == 0:
            grad += omh - h
        elif L > 0:
            grad[L:] += omh[:n - L]
            grad[:n - L] -= h[L:]
        else:
            L2 = -L
            grad[:n - L2] += omh[L2:]
            grad[L2:] -= h[:n - L2]
    grad /= float(len(active))
    return grad

def local_refine(h, best_c5, dx, target_sum, n_half, rng, max_steps, ss_init, label):
    h = h.copy()
    ss = ss_init
    mu = 0.5
    mom = np.zeros(len(h))
    K = 5
    for _ in range(max_steps):
        grad = active_grad(h, n_half, K)
        mom = mu * mom - ss * grad
        h_new = project_feasible(h + mom, target_sum)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            h = h_new
            ss = min(ss * 1.02, 0.01)
        else:
            ss *= 0.93
            if ss < 1e-9:
                break
        h = h_new
    return h, best_c5
