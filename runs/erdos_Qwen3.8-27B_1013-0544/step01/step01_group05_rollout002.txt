        c5_new = compute_c5(h_new, dx)

        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            stag = 0
        else:
            stag += 1
            if stag % 25 == 0:
                ss *= 0.85
            if ss < 1e-9:
                h = best_h + rng.normal(0, 0.04, n)
                h = project_feasible(h, target)
                ss = 0.002
                mom = np.zeros(n)
                stag = 0
        h = h_new

    # Phase 3: SLSQP polish
    if tleft() > 60:
        def slsqp_obj(x):
            return compute_c5(x, dx)
        def slsqp_eq(x):
            return np.sum(x) - target
        try:
            res = minimize(slsqp_obj, best_h, method='SLSQP',
                          bounds=[(0.0, 1.0)] * n,
                          constraints=[{'type': 'eq', 'fun': slsqp_eq}],
                          options={'maxiter': 4000, 'ftol': 1e-15})
            if res.x is not None:
                h_ref = project_feasible(res.x, target)
                c5_ref = compute_c5(h_ref, dx)
                if c5_ref < best_c5:
                    best_c5 = c5_ref
                    best_h = h_ref
        except Exception:
            pass

    # Phase 4: Random restarts with local refinement
    for _ in range(80):
        if tleft() < 15:
            break
        h_pert = best_h + rng.normal(0, 0.008, n)
        h_pert = project_feasible(h_pert, target)
        h_loc, c5_loc = local_refine(h_pert, n, dx, target, tleft, 150, 0.001)
        if c5_loc < best_c5:
            best_c5 = c5_loc
            best_h = h_loc

    # Phase 5: Try different n_points
    for n_try in [50, 60, 80, 100, 56, 64, 90]:
        if tleft() < 40:
            break
        dx_try = 2.0 / n_try
        target_try = n_try / 2.0
        h_candidate, c5_candidate = try_n_points(best_h, n_try, dx_try, target_try, tleft, rng)
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h_candidate
            best_n = n_try

    # Final local refinement on best
    if tleft() > 10:
        n_final = best_n
        dx_final = 2.0 / n_final
        target_final = n_final / 2.0
        h_fin, c5_fin = local_refine(best_h, n_final, dx_final, target_final, tleft, 300, 0.0005)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin

    return (best_h, best_c5, best_n)
