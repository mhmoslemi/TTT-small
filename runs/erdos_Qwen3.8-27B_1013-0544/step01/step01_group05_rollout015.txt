from scipy.optimize import minimize

def project_feasible(x, target_sum):
    lo = float(x.min()) - 1.0
    hi = float(x.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = float(np.clip(x - mid, 0.0, 1.0).sum())
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def smooth_max_grad(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    half = n - 1
    scaled = beta * corr * dx
    scaled -= scaled.max()
    weights = np.exp(scaled)
    weights /= weights.sum()
    grad = np.zeros(n)

        w = weights[k]
        if w < 1e-10:
            continue

        else:
            grad[j:] += w * (1.0 - h[:n - j])
            grad[:n - j] -= w * h[j:]
    return dx * grad
