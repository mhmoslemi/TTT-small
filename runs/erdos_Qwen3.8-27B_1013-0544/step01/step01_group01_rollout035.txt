        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            h += delta / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += delta / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def coord_descent(h, max_sweeps=200, time_limit=None, start_time=None):
    n = len(h)
    target_sum = n / 2.0
    h = project_to_sum(h.copy(), target_sum)
    best_c5 = compute_c5(h)
    best_h = h.copy()

    for sweep in range(max_sweeps):
        if time_limit is not None and time.time() - start_time > time_limit - 40:
            break
        improved = False
        for i in range(n):
            h_old = h[i]
            corr = np.correlate(h, 1.0 - h, mode="full")
            A = np.zeros(2 * n - 1)
            m_lo1 = max(0, i - n + 1)
            m_hi1 = min(i, 2 * n - 2)
            if m_hi1 >= m_lo1:
                A[m_lo1:m_hi1 + 1] += (1.0 - h[i - np.arange(m_lo1, m_hi1 + 1)])
            m_lo2 = 0
            m_hi2 = min(n - 1 - i, 2 * n - 2)
            if m_hi2 >= m_lo2:
                A[m_lo2:m_hi2 + 1] -= h[i + np.arange(m_lo2, m_hi2 + 1)]
            if np.max(np.abs(A)) < 1e-15:
                continue

            def f_obj(t):
                return np.max(corr + t * A)

            res = minimize_scalar(f_obj, bounds=(-h_old, 1.0 - h_old),
                                  method='bounded', options={'xatol': 1e-11})
            new_val = h_old + res.x
            if abs(new_val - h_old) > 1e-11:
                h[i] = new_val
                improved = True

        h = project_to_sum(h, target_sum)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if not improved:
            break

    return best_h, best_c5


def slsqp_polish(h, time_limit=None, start_time=None):
    n = len(h)
    target_sum = n / 2.0
    h = project_to_sum(h, target_sum)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    bounds = Bounds(np.zeros(n), np.ones(n))

    def con_fun(x):
        return np.sum(x) - target_sum

    def con_jac(x):
        return np.ones(n)

    constraints = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]

    for beta in [150.0, 300.0]:
        if time_limit is not None and time.time() - start_time > time_limit - 30:
            break

        def obj_func(x):
            corr = np.correlate(x, 1.0 - x, mode="full")
            return smooth_max(corr, beta) * (2.0 / n)

        try:
            res = minimize(obj_func, best_h, method='SLSQP', bounds=bounds,
                           constraints=constraints,
                           options={'maxiter': 500, 'ftol': 1e-14})
            h_new = project_to_sum(res.x, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
        except Exception:
            pass

    return best_h, best_c5


def make_initial(n, rng):
    """Create a diverse initial random h."""
    h = rng.random(n) * 0.6 + 0.1
    return project_to_sum(h, n / 2.0)


def make_structured(n, rng, style=0):
    """Create structured initial patterns."""
    if style == 0:
        t = np.linspace(0, 1, n)
        h = 0.5 + 0.3 * np.sin(2 * np.pi * t * 3 + rng.random() * 2 * np.pi)
    elif style == 1:
        t = np.linspace(0, 1, n)
        h = 0.5 + 0.25 * np.cos(2 * np.pi * t * 5) + 0.1 * np.sin(2 * np.pi * t * 11)
    elif style == 2:
        h = 0.5 + 0.2 * rng.standard_normal(n)
    else:
        h = 0.5 + 0.3 * rng.standard_normal(n)
    return project_to_sum(np.clip(h, 0, 1), n / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    def time_left():
        return budget_s - (time.time() - start_time)

    def update_best(h, c5, n):
        nonlocal best_h, best_c5, best_n
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            best_n = n

    # Get initial values
    init_70 = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        init_70 = np.array(initial_h_values, dtype=float).copy()

    # Try multiple n values
    n_candidates = [70]
    if init_70 is not None:
        n_candidates = [50, 60, 70, 80, 90, 100, 120]

    for n in n_candidates:
        if time_left() < 30:
            break

        # Build initial guesses for this n
        inits = []
        if init_70 is not None:
            if n == len(init_70):
                inits.append(init_70.copy())
            else:
                x_old = np.linspace(0, 2, len(init_70))
                x_new = np.linspace(0, 2, n)
                interp = np.interp(x_new, x_old, init_70)
                inits.append(project_to_sum(interp, n / 2.0))

        # Add random inits
        for _ in range(3):
            inits.append(make_initial(n, rng))
        for s in range(3):
            inits.append(make_structured(n, rng, style=s))

        for h0 in inits:
            if time_left() < 20:
                break

            # Coordinate descent
            h_cd, c5_cd = coord_descent(h0, max_sweeps=300,
                                        time_limit=budget_s, start_time=start_time)
            if c5_cd < best_c5:
                update_best(h_cd, c5_cd, n)

            # SLSQP polish
            if time_left() > 15:
                h_sp, c5_sp = slsqp_polish(h_cd, time_limit=budget_s, start_time=start_time)
                if c5_sp < best_c5:
                    update_best(h_sp, c5_sp, n)

    # Additional round: perturb the best and re-optimize
    if best_h is not None and time_left() > 40:
        n = best_n
        for attempt in range(20):
            if time_left() < 20:
                break
            scale = 0.03 + 0.05 * np.exp(-attempt / 8.0)
            h0 = best_h + rng.standard_normal(n) * scale
            h0 = project_to_sum(h0, n / 2.0)
            h_cd, c5_cd = coord_descent(h0, max_sweeps=150,
                                        time_limit=budget_s, start_time=start_time)
            if c5_cd < best_c5:
                update_best(h_cd, c5_cd, n)
            if time_left() > 15:
                h_sp, c5_sp = slsqp_polish(h_cd, time_limit=budget_s, start_time=start_time)
                if c5_sp < best_c5:
                    update_best(h_sp, c5_sp, n)

    if best_h is None:
        best_h = project_to_sum(np.full(70, 0.5), 35.0)
        best_c5 = compute_c5(best_h)
        best_n = 70

    return best_h, best_c5, best_n
