    grad = np.zeros(n)

        # dc_k/dh[i] for i in [0, n-1-k]: (1 - h[i+k])

            grad[i] += (1.0 - h[i + k])
        for i in range(k, n):
            grad[i] -= h[i - k]

        # c_k = sum_{i=j}^{n-1} h[i] * (1 - h[i-j]) = sum_{i=0}^{n-1-j} h[i+j] * (1-h[i])
        j = -k
        for i in range(j, n):
            grad[i] += (1.0 - h[i - j])
        for i in range(n - j):
            grad[i] -= h[i + j]
    return grad

Actually, one more optimization: instead of recomputing c (full correlation) after each coordinate update, I can update it incrementally. When h[i] changes from old to new:
c_k += (new - old) * B_k

                if 0 <= i - k < n:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def get_correlations(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy().clip(0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            t = room.sum()
            if t < 1e-12:
                break
            h += (diff / t) * room
        else:
            t = h.sum()
            if t < 1e-12:
                break
            h += (diff / t) * h
    return h.clip(0.0, 1.0)

def _sum_eq(h, n):
    return h.sum() - n / 2.0

def _sum_eq_jac(n):
    return np.ones(n)

def _smooth_obj(h, n, alpha):
    c = get_correlations(h, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def coord_descent(h, n, top_m=20, max_sweeps=500):
    c = get_correlations(h, n)
    nshift = n - 1
    for sweep in range(max_sweeps):
        improved = False
        for i in range(n):
            B = np.zeros(2 * n - 1)
            for idx in range(2 * n - 1):
                k = idx - nshift
                if 0 <= i + k < n:
                    B[idx] += (1.0 - h[i + k])
                if 0 <= i - k < n:
                    B[idx] -= h[i - k]
            top_idx = np.argsort(c)[-top_m:]
            cands_list = [0.0, 1.0]
            for a_pos in range(len(top_idx)):
                for b_pos in range(a_pos + 1, len(top_idx)):
                    ia, ib = int(top_idx[a_pos]), int(top_idx[b_pos])
                    dB = B[ia] - B[ib]
                    if abs(dB) > 1e-12:
                        t = h[i] + (c[ib] - c[ia]) / dB
                        if 0.0 < t < 1.0:
                            cands_list.append(t)
            cands_arr = np.array(cands_list)
            f_vals = np.max(c[None, :] + (cands_arr[:, None] - h[i]) * B[None, :], axis=1)
            best_idx = np.argmin(f_vals)
            if f_vals[best_idx] < c.max() - 1e-10:
                delta = cands_arr[best_idx] - h[i]
                h[i] = cands_arr[best_idx]
                c += delta * B
                improved = True
        if not improved:
            break
    return h, float(c.max())

def subgradient_optimize(h, n, n_iter=50000):
    c = get_correlations(h, n)
    nshift = n - 1
    v = np.zeros(n)
    lr = 0.003
    mom = 0.9
    best_c = c.max()
    for it in range(n_iter):
        max_val = c.max()
        top_idx = np.argsort(c)[-5:]
        grad = np.zeros(n)
        for m in top_idx:
            k = int(m) - nshift
            for i in range(n):
                g = 0.0
                if 0 <= i + k < n:
                    g += (1.0 - h[i + k])
                if 0 <= i - k < n:
                    g -= h[i - k]
                grad[i] += g
        grad /= len(top_idx)
        v = mom * v + (1 - mom) * grad
        h = project_feasible(h - lr * v, n)
        if it % 200 == 0:
            c = get_correlations(h, n)
            cv = c.max()
            if cv < best_c:
                best_c = cv
            if cv > best_c * 1.02:
                lr = max(0.0003, lr * 0.7)
            elif cv < best_c * 0.99:
                lr = min(0.01, lr * 1.3)
    return h, float(get_correlations(h, n).max())
