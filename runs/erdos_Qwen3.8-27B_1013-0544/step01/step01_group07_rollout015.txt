            # Decrease temperature gradually
            t = max(t_min, t * 0.999)
        else:
            # Revert and reduce momentum or lr
            v *= 0.5

    return best_h, best_c5

import numpy as np
import time

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h, dx):
    return float(np.max(compute_corrs(h)) * dx)

def fix_sum(h, target_sum):
    h = h.astype(float).copy()
    diff = target_sum - np.sum(h)
    h += diff / len(h)
    return np.clip(h, 0.0, 1.0)

def antisym_project(h):
    n = len(h)
    half = n // 2
    h_ant = np.zeros(n)
    for i in range(half):
        h_ant[i] = 0.5 * (h[i] + 1.0 - h[i + half])
    h_ant[half:] = 1.0 - h_ant[:half]
    return np.clip(h_ant, 0.0, 1.0)

def antisym_round(h):
    n = len(h)
    half = n // 2
    h_b = np.zeros(n)
    for i in range(half):
        val = 0.5 * (h[i] + 1.0 - h[i + half])
        h_b[i] = 1.0 if val >= 0.5 else 0.0
    h_b[half:] = 1.0 - h_b[:half]
    return h_b

def binary_antisym_local_search(h, dx, max_rounds=500, timeout=None, t0=None):
    n = len(h)
    half = n // 2
    h_b = antisym_round(h)
    best_c5 = compute_c5(h_b, dx)
    best_h = h_b.copy()
    for r in range(max_rounds):
        if timeout is not None and t0 is not None and time.time() - t0 > timeout:
            break
        improved = False
        for i in range(half):
            h_b[i] = 1.0 - h_b[i]
            h_b[i + half] = 1.0 - h_b[i + half]
            c5 = compute_c5(h_b, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_b.copy()
                improved = True
            else:
                h_b[i] = 1.0 - h_b[i]
                h_b[i + half] = 1.0 - h_b[i + half]
        if not improved:
            break
    return best_h, best_c5

def weighted_gradient(h, w, n):
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-14:
            continue
        s = m - (n - 1)
        if s >= 0:
            if s < n:
                grad[s:] += wm * (1.0 - h[:n - s])
                grad[:n - s] -= wm * h[s:]
        else:
            ss = -s
            if ss < n:
                grad[:n - ss] += wm * (1.0 - h[ss:])
                grad[ss:] -= wm * h[:n - ss]
    return grad

def gradient_descent(h, dx, target_sum, t_init=3.0, t_min=0.05,
                     lr_init=0.005, max_steps=20000, momentum=0.9,
                     timeout=None, t0=None):
    n = len(h)
    h = h.astype(float).copy()
    v = np.zeros(n)
    t = t_init
    lr = lr_init
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    for step in range(max_steps):
        if timeout is not None and t0 is not None and time.time() - t0 > timeout:
            break
        corrs = compute_corrs(h)
        max_c = np.max(corrs)
        w = np.exp((corrs - max_c) / t)
        w_sum = np.sum(w)
        if w_sum < 1e-300:
            break
        w /= w_sum
        grad = weighted_gradient(h, w, n)
        v = momentum * v - lr * grad
        h_new = h + v
        h_new = np.clip(h_new, 0.0, 1.0)
        diff = target_sum - np.sum(h_new)
        h_new += diff / n
        h_new = np.clip(h_new, 0.0, 1.0)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            h = h_new
            best_c5 = c5_new
            best_h = h.copy()
            t = max(t_min, t * 0.9995)
        else:
            lr *= 0.98
            v *= 0.5
            t = max(t_min, t * 0.999)
    return best_h, best_c5

def simulated_annealing_antisym(h0, dx, t0_param=0.5, t_end=0.005,
                                 n_iter=20000, timeout=None, t0=None, seed_val=0):
    n = len(h0)
    half = n // 2
    rng = np.random.RandomState(seed_val)
    h = antisym_round(h0)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    curr_c5 = best_c5
    for i in range(n_iter):
        if timeout is not None and t0 is not None and time.time() - t0 > timeout:
            break
        curr_t = t0_param * (t_end / t0_param) ** (i / max(n_iter, 1))
        j = rng.randint(0, half)
        h[j] = 1.0 - h[j]
        h[j + half] = 1.0 - h[j + half]
        new_c5 = compute_c5(h, dx)
        delta = new_c5 - curr_c5
        if delta < 0 or (curr_t > 1e-8 and rng.random() < np.exp(-delta / (curr_t * 0.005))):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[j] = 1.0 - h[j]
            h[j + half] = 1.0 - h[j + half]
    return best_h, best_c5

def upsample_antisym(h, new_n):
    n = len(h)
    half_old = n // 2
    half_new = new_n // 2
    h_old = h[:half_old]
    x_old = np.linspace(0, 1.0, half_old)
    x_new = np.linspace(0, 1.0, half_new)
    h_new_half = np.interp(x_new, x_old, h_old)
    h_new = np.zeros(new_n)
    h_new[:half_new] = h_new_half
    h_new[half_new:] = 1.0 - h_new_half
    return h_new
