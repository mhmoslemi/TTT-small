def fix_sum(h, target):
    h = np.array(h, dtype=float).copy()
    diff = target - h.sum()
    if abs(diff) < 1e-10:
        return np.clip(h, 0.0, 1.0)

    else:

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target):
    h = np.array(h, dtype=float).copy()
    diff = target - h.sum()
    if abs(diff) < 1e-10:
        return np.clip(h, 0.0, 1.0)
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = mask.sum()
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def _obj_func(x, tau):
    corr = np.correlate(x, 1.0 - x, mode="full")
    xs = tau * corr
    xm = np.max(xs)
    return float((xm + np.log(np.sum(np.exp(xs - xm)))) / tau)

def _con_func(x, target):
    return float(np.sum(x) - target)

def _con_jac(x, target):
    return np.ones_like(x)

def _neg_subgrad(h, m):
    n = len(h)
    g = np.zeros(n)
    if m >= 0:
        g[:n - m] += (1.0 - h[m:n])
        g[m:] -= h[:n - m]
    else:
        mm = -m
        g[mm:] += (1.0 - h[:n - mm])
        g[:n - mm] -= h[mm:]
    return -g

def _topk_direction(h, k=3):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argsort(corr)[-k:]
    d = np.zeros(n)
    for ki in idx:
        m = ki - (n - 1)
        d += _neg_subgrad(h, m)
    return d / len(idx)

def _slsqp_opt(h_start, n_pts, tau, maxiter, ftol):
    target = n_pts / 2.0
    bnds = Bounds(np.zeros(n_pts), np.ones(n_pts))
    try:
        res = minimize(_obj_func, h_start, args=(tau,),
                       method='SLSQP', bounds=bnds,
                       constraints=[{'type': 'eq', 'fun': _con_func,
                                     'jac': _con_jac, 'args': (target,)}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        x = fix_sum(res.x, target)
        return x, evaluate_c5(x)
    except Exception:
        return h_start, None

def _subgrad_phase(h, n, target, max_iters, base_alpha, time_limit, t0):
    best = h.copy()
    best_c5 = evaluate_c5(best)
    momentum = np.zeros(n)
    beta = 0.85
    for it in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        d = _topk_direction(best, 3)
        momentum = beta * momentum + d
        mn = np.linalg.norm(momentum)
        if mn > 1e-10:
            momentum = momentum / mn
        alpha = base_alpha / (1.0 + 0.005 * it)
        h_try = best + alpha * momentum
        h_try = np.clip(h_try, 0.0, 1.0)
        h_try = fix_sum(h_try, target)
        c5_try = evaluate_c5(h_try)
        if c5_try < best_c5:
            best = h_try
            best_c5 = c5_try
        else:
            alpha *= 0.9
    return best, best_c5

def _local_search(h, n, target, n_iter, t0, time_limit):
    best = h.copy()
    best_c5 = evaluate_c5(best)
    rng = np.random
    for _ in range(n_iter):
        if time.time() - t0 > time_limit:
            break
        i, j = rng.choice(n, 2, replace=False)
        if abs(best[i] - best[j]) < 1e-8:
            continue
        h_try = best.copy()
        h_try[i], h_try[j] = h_try[j], h_try[i]
        c5_try = evaluate_c5(h_try)
        if c5_try < best_c5:
            best = h_try
            best_c5 = c5_try
        # Also try small nudge
        if rng.random() < 0.5:
            h_try2 = best.copy()
            h_try2[i] += 0.05 * (best[j] - best[i])
            h_try2[j] -= 0.05 * (best[j] - best[i])
            h_try2 = np.clip(h_try2, 0.0, 1.0)
            h_try2 = fix_sum(h_try2, target)
            c5_try2 = evaluate_c5(h_try2)
            if c5_try2 < best_c5:
                best = h_try2
                best_c5 = c5_try2
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s

    def _rem():
        return deadline - time.time()

        h_new, c5_new = _slsqp_opt(best_h, n0, tau, 500, 1e-15)
        if c5_new is not None and c5_new < best_c5:

            best_c5 = c5_new

    # Phase 3: Multi-resolution exploration
    for tn in [80, 100, 128, 160, 200, 240]:

        tgt = tn / 2.0
        h_t = np.interp(np.linspace(0, 1, tn),

        h_t = fix_sum(h_t, tgt)
        for tau in [100, 300, 800, 1500]:
            if _rem() < 50:
                break
            h_new, c5_new = _slsqp_opt(h_t, tn, tau, 350, 1e-14)
            if c5_new is not None and c5_new < best_c5:

                best_c5 = c5_new
            h_t = h_new if h_new is not h_t else h_t
        # Subgradient at this resolution
        if _rem() > 40:
            h_t, c5_t = _subgrad_phase(h_t, tn, tgt, 100, 0.03,

            if _rem() < 30:

    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, hn_final
