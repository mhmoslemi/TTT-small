= sum_{valid n} h_base[n+k]*(1-h_base[n])

  - t * sum_{valid n} h_base[n+k]*[n==i]
  - t^2 * sum_{valid n} [n+k==i]*[n==i]

Term 1: corr_base[k] (correlation of h_base with 1-h_base)
Term 2: [n+k==i] means n=i-k. Valid if 0≤i-k<n (and 0≤n<n which is 0≤i-k<n). So this is (1-h_base[i-k]) if 0≤i-k<n. Note: if i-k==i, i.e., k=0, then h_base[i-k]=h_base[i]=0, so this becomes 1.
Term 3: [n==i] means n=i. Valid if 0≤i+k<n. So this is h_base[i+k] if 0≤i+k<n. If i+k==i, i.e., k=0, then h_base[i+k]=h_base[i]=0.
Term 4: [n+k==i]*[n==i] means n=i and k=0. So this is 1 if k=0, else 0.

corr[k](t) = corr_base[k] + t*(1-h_base[i-k])*[0≤i-k<n] - t*h_base[i+k]*[0≤i+k<n] - t^2*[k==0]

- For k=0: Term 2 gives (1-0)=1, Term 3 gives 0. So corr[0](t) = corr_base[0] + t - t^2
- For k≠0: 
  - If 0≤i-k<n: Term 2 = (1-h_base[i-k])
  - If 0≤i+k<n: Term 3 = -h_base[i+k]
  - Term 4 = 0

import numpy as np
import time

def project_feasible(x, target_sum):
    lo = float(x.min()) - 1.0
    hi = float(x.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = float(np.clip(x - mid, 0.0, 1.0).sum())
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def prepare_coord(h, i, n):
    h_base = h.copy()
    h_base[i] = 0.0
    corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
    B = np.zeros(2 * n - 1)
    A = np.zeros(2 * n - 1)
    nhal = n - 1
    for j in range(2 * n - 1):
        L = j - nhal
        b_val = 0.0
        n1 = i - L
        if 0 <= n1 < n:
            b_val += 1.0 - h_base[n1]
        n2 = i + L
        if 0 <= n2 < n:
            b_val -= h_base[n2]
        B[j] = b_val
        if L == 0:
            A[j] = -1.0
    return corr_base, A, B

def coord_step(h, i, dx, n_grid, n_refine):
    n = len(h)
    corr_base, A, B = prepare_coord(h, i, n)
    # Coarse grid search
    t_grid = np.linspace(0.0, 1.0, n_grid)
    t2 = t_grid * t_grid
    vals = corr_base[None, :] + A[None, :] * t2[:, None] + B[None, :] * t_grid[:, None]
    f_vals = np.max(vals, axis=1) * dx
    idx_min = int(np.argmin(f_vals))
    a = t_grid[max(0, idx_min - 1)]
    b = t_grid[min(n_grid - 1, idx_min + 1)]
    # Golden section refinement
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = float(np.max(corr_base + A * c * c + B * c)) * dx
    fd = float(np.max(corr_base + A * d * d + B * d)) * dx
    for _ in range(n_refine):
        if b - a < 1e-10:
            break
        if fc < fd:
            b, d, fd = d, c, fc
            c = b - gr * (b - a)
            fc = float(np.max(corr_base + A * c * c + B * c)) * dx
        else:
            a, c, fc = c, d, fd
            d = a + gr * (b - a)
            fd = float(np.max(corr_base + A * d * d + B * d)) * dx
    return 0.5 * (a + b)

def init_correlation(h):
    return np.correlate(h, 1.0 - h, mode='full')

def update_correlation_for_change(corr, h, i, delta, n):
    """Update corr in-place when h[i] changes by delta."""
    old = h[i]
    new = old + delta
    nhal = n - 1
    for j in range(2 * n - 1):
        L = j - nhal
        if L == 0:
            corr[j] += delta * (1.0 - new - old)
        else:
            dc = 0.0
            n1 = i - L
            if 0 <= n1 < n:
                dc += delta * (1.0 - h[n1])
            n2 = i + L
            if 0 <= n2 < n:
                dc -= delta * h[n2]
            corr[j] += dc
    h[i] = new

def get_optimal_t(corr, i, n, dx, n_grid=400):
    """Find optimal h[i] given current correlation (with h[i] at current value)."""
    # Need to compute B and A for coordinate i
    # B[j] = (1-h[i-L_j])*[0≤i-L_j<n] - h[i+L_j]*[0≤i+L_j<n] for L_j≠0
    #        = (1-0) - 0 = 1 for L_j=0 (since h[i]=0 in base, but we're using current corr)
    
    # Actually, this approach doesn't work directly because the correlation already
    # includes the current h[i]. I need to "undo" the current h[i] contribution first.
    
    # Alternative: compute the "base" correlation (with h[i]=0) from the current corr:
    # corr_base[j] = corr[j] - (current contribution of h[i] to corr[j])
    # current contribution = old * B[j] + old^2 * A[j] (using old = h[i])
    
    old = h[i]
    nhal = n - 1
    A = np.zeros(2 * n - 1)
    B = np.zeros(2 * n - 1)
    for j in range(2 * n - 1):
        L = j - nhal
        if L == 0:
            A[j] = -1.0
            B[j] = 1.0  # (1-h_base[i]) - h_base[i] = 1 - 0 = 1
        else:
            b_val = 0.0
            n1 = i - L
            if 0 <= n1 < n:
                b_val += 1.0 - h[n1]
            n2 = i + L
            if 0 <= n2 < n:
                b_val -= h[n2]
            B[j] = b_val
    
    # corr_base[j] = corr[j] - old*B[j] - old^2*A[j]
    corr_base = corr - old * B - old * old * A
    
    # Now find t that minimizes max_j (corr_base[j] + B[j]*t + A[j]*t^2) * dx
    t_grid = np.linspace(0.0, 1.0, n_grid)
    t2 = t_grid * t_grid
    vals = corr_base[None, :] + B[None, :] * t_grid[:, None] + A[None, :] * t2[:, None]
    f_vals = np.max(vals, axis=1) * dx
    idx_min = int(np.argmin(f_vals))
    
    # Golden section refinement
    a = t_grid[max(0, idx_min - 1)]
    b = t_grid[min(n_grid - 1, idx_min + 1)]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = float(np.max(corr_base + B * c + A * c * c)) * dx
    fd = float(np.max(corr_base + B * d + A * d * d)) * dx
    for _ in range(40):
        if b - a < 1e-10:
            break
        if fc < fd:
            b, d, fd = d, c, fc
            c = b - gr * (b - a)
            fc = float(np.max(corr_base + B * c + A * c * c)) * dx
        else:
            a, c, fc = c, d, fd
            d = a + gr * (b - a)
            fd = float(np.max(corr_base + B * d + A * d * d)) * dx
    return 0.5 * (a + b)
