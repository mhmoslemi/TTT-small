```python
def line_search_pair(h, n, i, j):
    C = np.correlate(h, 1-h, mode='full')
    
    # Compute slopes b[idx]
    k = np.arange(-(n-1), n)  # lag values
    
    # dC/dh[i]:
    valid_a = (i - k >= 0) & (i - k < n)  # i-k valid
    valid_b = (i + k >= 0) & (i + k < n)  # i+k valid
    dCi = np.zeros(2*n-1)
    dCi[valid_a] += (1 - h[i - k[valid_a]])
    dCi[valid_b] -= h[i + k[valid_b]]
    
    # dC/dh[j]:
    valid_c = (j - k >= 0) & (j - k < n)
    valid_d = (j + k >= 0) & (j + k < n)
    dCj = np.zeros(2*n-1)
    dCj[valid_c] += (1 - h[j - k[valid_c]])
    dCj[valid_d] -= h[j + k[valid_d]]
    
    b = dCi - dCj
    
    # Quadratic terms
    q = np.zeros(2*n-1)
    idx_a = (i - j) + (n - 1)  # k₀ = i-j
    idx_b = (j - i) + (n - 1)  # k₁ = j-i
    if 0 <= idx_a <= 2*n-2:
        q[idx_a] = 1
    if 0 <= idx_b <= 2*n-2:
        q[idx_b] = 1
    
    # Bounds on δ
    delta_lo = max(-h[i], h[j] - 1)
    delta_hi = min(1 - h[i], h[j])
    
    if delta_hi - delta_lo < 1e-14:
        return 0.0
    
    # Bisection
    for _ in range(80):
        delta_mid = (delta_lo + delta_hi) / 2
        vals = C + b * delta_mid + q * delta_mid**2
        max_val = vals.max()
        active = vals >= max_val - 1e-12
        grad = b[active] + 2 * q[active] * delta_mid
        if grad.max() < 0:
            delta_lo = delta_mid
        elif grad.min() > 0:
            delta_hi = delta_mid
        else:
            break
    
    delta_opt = (delta_lo + delta_hi) / 2
    return delta_opt
```

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-15:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(h, n, alpha):

    c = np.correlate(h, 1 - h, mode='full') * dx

    shifted = alpha * (c - m)
    mx = shifted.max()

    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

    return np.sum(h) - n / 2.0

    return np.ones(n)

def compute_pair_slopes(h, n, i, j):
    """Compute b and q for the pair (i,j) line search."""
    k = np.arange(-(n-1), n, dtype=float)
    
    valid_a = (i - k >= 0) & (i - k < n)
    valid_b = (i + k >= 0) & (i + k < n)
    dCi = np.zeros(2*n-1)
    ia = i - k[valid_a].astype(int)
    dCi[valid_a] += (1.0 - h[ia])
    ib = i + k[valid_b].astype(int)
    dCi[valid_b] -= h[ib]
    
    valid_c = (j - k >= 0) & (j - k < n)
    valid_d = (j + k >= 0) & (j + k < n)
    dCj = np.zeros(2*n-1)
    jc = j - k[valid_c].astype(int)
    dCj[valid_c] += (1.0 - h[jc])
    jd = j + k[valid_d].astype(int)
    dCj[valid_d] -= h[jd]
    
    b = dCi - dCj
    
    q = np.zeros(2*n-1)
    idx_a = (i - j) + (n - 1)
    idx_b = (j - i) + (n - 1)
    if 0 <= idx_a <= 2*n-2:
        q[idx_a] = 1.0
    if 0 <= idx_b <= 2*n-2:
        q[idx_b] = 1.0
    
    return b, q

def line_search_pair(h, n, i, j):

    b, q = compute_pair_slopes(h, n, i, j)
    
    delta_lo = max(-h[i], h[j] - 1.0)
    delta_hi = min(1.0 - h[i], h[j])
    
    if delta_hi - delta_lo < 1e-14:
        return 0.0
    
    for _ in range(70):
        delta_mid = (delta_lo + delta_hi) / 2.0
        vals = C + b * delta_mid + q * delta_mid * delta_mid
        max_val = vals.max()
        active = vals >= max_val - 1e-14
        grad = b[active] + 2.0 * q[active] * delta_mid
        gmax = grad.max()
        gmin = grad.min()
        if gmax < 0:
            delta_lo = delta_mid
        elif gmin > 0:
            delta_hi = delta_mid
        else:
            break
    
    return (delta_lo + delta_hi) / 2.0
