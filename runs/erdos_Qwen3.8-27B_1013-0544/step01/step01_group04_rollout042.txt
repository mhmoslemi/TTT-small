import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - target
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            h[free] -= diff / np.sum(free)
        else:
            h -= diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = np.sum(h) - target
    if abs(diff) > 1e-14:
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            h[free] -= diff / np.sum(free)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_and_grad(h, dx, alpha, mu, target_sum):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = alpha * dx * corr
    max_s = np.max(scaled)
    exp_s = np.exp(scaled - max_s)
    sum_exp = np.sum(exp_s)
    weights = exp_s / sum_exp  # softmax weights, length 2n-1
    
    f = (max_s + np.log(sum_exp)) / alpha
    
    # Gradient
    # weights[k] corresponds to lag l = k - (n-1)
    # dF/dh[j] = dx * sum_l weights[l+(n-1)] * [(1-h[j-l])*I(0<=j-l<n) + (-h[j+l])*I(0<=j+l<n)]
    
    grad_F = np.zeros(n)
    for j in range(n):
        for k in range(2*n-1):
            l = k - (n - 1)
            w = weights[k]
            if w < 1e-15:
                continue
            total = 0.0
            if 0 <= j - l < n:
                total += (1.0 - h[j - l])
            if 0 <= j + l < n:
                total += (-h[j + l])
            grad_F[j] += w * total
    grad_F *= dx
    
    # Penalty gradient
    penalty = mu * (np.sum(h) - target_sum) ** 2
    grad_penalty = 2 * mu * (np.sum(h) - target_sum) * np.ones(n)
    
    return f + penalty, grad_F + grad_penalty

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - target
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            h[free] -= diff / np.sum(free)
        else:
            h -= diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = np.sum(h) - target
    if abs(diff) > 1e-14:
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            h[free] -= diff / np.sum(free)
            h = np.clip(h, 0.0, 1.0)
    return h

def sigmoid(x):
    x = np.clip(x, -500.0, 500.0)
    return 1.0 / (1.0 + np.exp(-x))

def logit(x):
    x = np.clip(x, 1e-10, 1.0 - 1e-10)
    return np.log(x / (1.0 - x))

def smooth_obj_z(x, dx, alpha, mu, target_sum):
    n = len(x)
    h = sigmoid(x)
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = alpha * dx * corr
    max_s = np.max(scaled)
    exp_s = np.exp(scaled - max_s)
    sum_exp = np.sum(exp_s)
    f = (max_s + np.log(sum_exp)) / alpha
    penalty = mu * (np.sum(h) - target_sum) ** 2
    return f + penalty

def smooth_grad_z(x, dx, alpha, mu, target_sum):
    n = len(x)
    h = sigmoid(x)
    hprime = h * (1.0 - h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = alpha * dx * corr
    max_s = np.max(scaled)
    exp_s = np.exp(scaled - max_s)
    sum_exp = np.sum(exp_s)
    weights = exp_s / sum_exp
    
    cumw = np.concatenate([[0.0], np.cumsum(weights)])
    S1 = cumw[n:2*n] - cumw[0:n]
    
    conv2 = np.convolve(h, weights, 'full')
    S2 = conv2[n-1:2*n-1]
    
    weights_rev = weights[::-1]
    conv3 = np.convolve(h, weights_rev, 'full')
    S3 = conv3[n-1:2*n-1]
    
    grad_h = dx * (S1 - S2 - S3)
    grad_penalty = 2.0 * mu * (np.sum(h) - target_sum)
    return (grad_h + grad_penalty) * hprime

def slsqp_obj(x, dx):
    return compute_c5(x, dx)

def slsqp_con(x, target_sum):
    return np.sum(x) - target_sum
