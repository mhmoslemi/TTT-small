    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    for _ in range(50):  # iterative projection
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target_sum) < 1e-10:
            break
        h = h + (target_sum - s) / len(h)
    return np.clip(h, 0.0, 1.0)

def obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted -= np.max(shifted)
    exp_v = np.exp(shifted)
    w = exp_v / exp_v.sum()
    obj = (m + np.log(exp_v.sum()) / beta) * dx
    
    grad = np.zeros(n)
    lags = np.arange(-(n-1), n)
    for j in range(len(corr)):

            continue

        if mask2.any():
            idx2 = np.where(mask2)[0]
            grad[idx2] -= w[j] * h[valid2[mask2]]
    grad *= dx
    return obj, grad

        room_up = (1.0 - h) * (h < 1.0)
        room_dn = h * (h > 0.0)
        diff = target_sum - s

            total_room = room_up.sum()
            if total_room > 0:

            else:
                h += diff / len(h)
        else:
            total_room = room_dn.sum()
            if total_room > 0:
                h += diff * room_dn / total_room
            else:
                h += diff / len(h)
    return np.clip(h, 0.0, 1.0)

def obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    sh = beta * (corr - m)
    sh -= sh.max()
    ev = np.exp(sh)
    sw = ev.sum()
    w = ev / sw
    obj = (m + np.log(sw) / beta) * dx
    
    grad = np.zeros(n)
    lags = np.arange(-(n-1), n)
    idx = np.arange(n)
    for j in range(len(corr)):
        wj = w[j]
        if wj < 1e-14:
            continue
        lag = lags[j]
        # term1: (1-h[m-lag]) for m where 0<=m-lag<n
        v1 = idx - lag
        mask1 = (v1 >= 0) & (v1 < n)
        if mask1.any():
            grad[mask1] += wj * (1.0 - h[v1[mask1]])
        # term2: -h[m+lag] for m where 0<=m+lag<n
        v2 = idx + lag
        mask2 = (v2 >= 0) & (v2 < n)
        if mask2.any():
            grad[mask2] -= wj * h[v2[mask2]]
    grad *= dx
    return obj, grad
