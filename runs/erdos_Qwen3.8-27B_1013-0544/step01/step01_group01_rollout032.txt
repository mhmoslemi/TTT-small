def coordinate_descent_step(h, n, j, target_sum):
    """Optimize h[j] given all other coordinates fixed."""
    v = 1.0 - h
    corr = np.correlate(h, v, mode="full")
    
    # For coordinate j, find which lags involve it
    # result[k] = sum_m h[m] * v[m + k - n + 1]
    # h[j] appears when m = j, so v[j + k - n + 1] must be valid
    # j + k - n + 1 ranges from 0 to n-1
    # So k ranges from n-1-j to n-1+j
    
    # The contribution of h[j] to result[k] is h[j] * v[j + k - n + 1]
    # Let's compute the base (without h[j]) and the slope
    base = corr.copy()
    # Remove h[j]'s contribution from each relevant k
    k_min = n - 1 - j
    k_max = n - 1 + j
    for k in range(k_min, k_max + 1):
        idx = j + k - n + 1  # index in v
        base[k] -= h[j] * v[idx]
    
    # Now result[k] = base[k] + h[j] * v[idx_k]
    # Objectives: for each k, f_k(t) = base[k] + t * v[idx_k]
    # We want min_t max_k f_k(t) subject to t in [0,1]
    
    # The slope for each k is v[idx_k] = 1 - h[idx_k]
    slopes = np.zeros(2*n - 1)
    for k in range(k_min, k_max + 1):
        idx = j + k - n + 1
        slopes[k] = v[idx]
    
    # f_k(t) = base[k] + slopes[k] * t
    # g(t) = max_k (base[k] + slopes[k] * t)
    # g is convex piecewise linear
    # Minimum is where the subgradient contains 0
    
    # Simple approach: ternary search or bisection on subgradient
    def g(t):
        return np.max(base + slopes * t)
    
    def subgrad(t):
        vals = base + slopes * t
        max_val = np.max(vals)
        active = vals >= max_val - 1e-12
        return np.sum(slopes[active])
    
    # Bisection for zero subgradient
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2
        sg = subgrad(mid)
        if sg > 0:
            hi = mid
        else:
            lo = mid
    
    t_opt = (lo + hi) / 2
    
    # Also check endpoints
    best_t = min([0, 1, t_opt], key=lambda t: g(t))

import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -50, 50)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def project(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h


def find_optimal_t(base, slopes):
    def subgrad(t):
        vals = base + slopes * t
        mx = np.max(vals)
        active = vals >= mx - 1e-10
        return np.sum(slopes[active])
    sg0 = subgrad(0.0)
    if sg0 >= 0:
        return 0.0
    sg1 = subgrad(1.0)
    if sg1 <= 0:
        return 1.0
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        if subgrad(mid) > 0:
            hi = mid
        else:
            lo = mid
    return (lo + hi) / 2


def coord_descent(h, n, target, max_sweeps=200, tl=None):
    h = h.copy()
    for sweep in range(max_sweeps):
        if tl is not None and tl() < 5:
            break
        for j in range(n):
            v = 1.0 - h
            corr = np.correlate(h, v, mode="full")
            k_min = n - 1 - j
            k_max = n - 1 + j
            base = corr.copy()
            slopes = np.zeros(2 * n - 1)
            for k in range(k_min, k_max + 1):
                idx = j + k - n + 1
                base[k] -= h[j] * v[idx]
                slopes[k] = v[idx]
            h[j] = find_optimal_t(base, slopes)
        h = project(h, target)
    return h


def slsqp_optimize(h0, n, target, beta, maxiter):
    bounds = Bounds(np.zeros(n), np.ones(n))

    def obj(h):
        return smooth_max(np.correlate(h, 1.0 - h, mode="full"), beta) * (2.0 / n)

    def con_f(h):
        return h.sum() - target

    def con_j(h):
        return np.ones(n)

    res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': con_f, 'jac': con_j}],
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return project(res.x, target)


def generate_inits(n, target, seed, parent=None):
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.copy())
        else:
            xp = np.linspace(0, 2, len(parent))
            xn = np.linspace(0, 2, n)
            inits.append(project(np.interp(xn, xp, parent), target))
    else:
        inits.append(np.full(n, 0.5))
    # Block: 1 on [0,0.5] and [1,1.5]
    hb = np.zeros(n)
    q = n // 4
    hb[:q] = 1.0
    hb[n // 2: n // 2 + q] = 1.0
    inits.append(project(hb, target))
    # Random
    inits.append(project(np.random.random(n), target))
    # Alternating blocks
    bs = max(2, n // 8)
    ha = np.zeros(n)
    for s in range(0, n, 2 * bs):
        ha[s:min(s + bs, n)] = 1.0
    inits.append(project(ha, target))
    # Periodic
    period = max(3, n // 6)
    hp = np.tile(np.random.random(period), n // period + 1)[:n]
    inits.append(project(hp, target))
    # Three blocks
    h3 = np.zeros(n)
    h3[:n // 6] = 1.0
    h3[n // 3: n // 3 + n // 6] = 1.0
    h3[2 * n // 3: 2 * n // 3 + n // 6] = 1.0
    inits.append(project(h3, target))
    return inits


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()

    def tl():
        return budget_s - (time.time() - t_start)

    parent = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent = np.array(initial_h_values, dtype=float).copy()

    best_h = parent.copy() if parent is not None else np.full(70, 0.5)
    best_c5 = compute_c5(best_h)
    best_n = len(best_h)

    # Candidate n values, parent first
    n_cands = []
    if parent is not None:
        n_cands.append(len(parent))
    for n in [60, 70, 80, 90, 100, 120, 140, 160]:
        if n not in n_cands:
            n_cands.append(n)

    for n in n_cands:
        if tl() < 30:
            break
        target = n / 2.0
        inits = generate_inits(n, target, seed, parent)
        local_best = inits[0].copy()
        local_c5 = compute_c5(local_best)

        for h0 in inits:
            if tl() < 12:
                break
            h_work = project(h0, target)
            # SLSQP with increasing smoothness
            for beta in [30, 100, 300]:
                if tl() < 8:
                    break
                try:
                    h_new = slsqp_optimize(h_work, n, target, beta, 600)
                    c5 = compute_c5(h_new)
                    if c5 < local_c5:
                        local_c5 = c5
                        local_best = h_new.copy()
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_new.copy()
                        best_n = n
                    h_work = h_new.copy()
                except Exception:
                    pass

            # Coordinate descent refinement
            if tl() > 10 and local_c5 < 0.39:
                h_cd = coord_descent(local_best, n, target, max_sweeps=150, tl=tl)
                c5 = compute_c5(h_cd)
                if c5 < local_c5:
                    local_c5 = c5
                    local_best = h_cd.copy()
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_cd.copy()
                    best_n = n

    # Final high-precision polish
    if tl() > 15 and best_h is not None:
        target = best_n / 2.0
        for beta in [600, 1500, 3000]:
            if tl() < 8:
                break
            try:
                h_new = slsqp_optimize(best_h, best_n, target, beta, 2000)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()
            except Exception:
                pass

    # One more coordinate descent on final best
    if tl() > 10 and best_h is not None:
        target = best_n / 2.0
        h_cd = coord_descent(best_h, best_n, target, max_sweeps=100, tl=tl)
        c5 = compute_c5(h_cd)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_cd.copy()

    # Small perturbation search around best
    if tl() > 20:
        target = best_n / 2.0
        for _ in range(10):
            if tl() < 8:
                break
            scale = 0.01
            h_pert = best_h + np.random.randn(best_n) * scale
            h_pert = project(h_pert, target)
            try:
                h_new = slsqp_optimize(h_pert, best_n, target, 200, 400)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()
            except Exception:
                pass

    return best_h, best_c5, best_n
