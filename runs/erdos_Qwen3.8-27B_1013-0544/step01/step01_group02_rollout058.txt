            h += (diff / tot) * room

            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h

def subgrad_lag(h, n, p):
    g = np.zeros(n)
    if p >= n - 1:

    else:
        j = n - 1 - p
