    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    # Shifted for numerical stability
    max_c = np.max(corr)
    shifted = t * (corr - max_c)
    exp_shifted = np.exp(shifted)

    # Now compute gradient
    # For lag d (index k = n-1+d in corr output):
    # grad h[j] += w_k * ((1-h[j-d]) * [d<=j<=n-1] - h[j+d] * [0<=j<=n-1-d]) for d>=0

    lo, hi = -2.0, 2.0  # wide enough range

import time
from scipy.optimize import minimize
from functools import partial

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

    max_c = np.max(corr)
    shifted = t * (corr - max_c)
    val = max_c + (1.0 / t) * np.log(np.sum(np.exp(shifted)))
    return val * dx

def smooth_max_grad(h, t, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_c = np.max(corr)
    shifted = t * (corr - max_c)
    exp_s = np.exp(shifted)
    total = np.sum(exp_s)
    weights = exp_s / total
    
    grad = np.zeros(n)
    for s in range(n):
        if s == 0:

import numpy as np
import time
from scipy.optimize import minimize
from functools import partial

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_obj(h, t, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_c = np.max(corr)
    shifted = t * (corr - max_c)
    val = max_c + (1.0 / t) * np.log(np.sum(np.exp(shifted)))
    return val * dx

def smooth_max_grad(h, t, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_c = np.max(corr)
    shifted = t * (corr - max_c)
    exp_s = np.exp(shifted)
    total = np.sum(exp_s)
    weights = exp_s / total
    grad = np.zeros(n)
    for s in range(n):
        if s == 0:
            w_s = weights[n - 1]
        else:
            w_s = weights[n - 1 + s] + weights[n - 1 - s]
        if w_s < 1e-15:
            continue
        if n - s > 0:
            grad[s:] += w_s * (1.0 - h[:n - s])
            grad[:n - s] -= w_s * h[s:]
    grad *= dx
    return grad

def fix_sum(h, target_sum):
    h = h.astype(float).copy()
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-14:
        h = h + diff / len(h)
    return np.clip(h, 0.0, 1.0)

def project_onto_set(y, target_sum):
    lo, hi = -5.0, 5.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(y - lam, 0.0, 1.0)

    con_fun = partial(constraint_eq, target_sum=target_sum)
    con_jac = partial(constraint_eq_jac, target_sum=target_sum)
    try:
        res = minimize(
            obj_fn, h.copy(), jac=grad_fn,
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-15}
        )
        return fix_sum(res.x, target_sum)
    except Exception:
