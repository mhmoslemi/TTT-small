def compute_gradient(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    
    shifted = beta * (corr - np.max(corr))
    weights = np.exp(shifted)
    weights /= weights.sum()
    
    # For each position m, grad[m] = sum over lags k of weights[k] * dcorr[k]/dh[m]
    # dcorr[k]/dh[m] = (1-h[m-k]) * [valid] - h[m+k] * [valid]
    
    # Part 1: sum_k weights[k] * (1-h[m-k])
    # This is like a convolution of (1-h) with a window defined by weights
    # Let's use padding
    one_minus_h = 1.0 - h
    # For position m, we need sum_k weights[k] * one_minus_h[m-k]
    # where k ranges over -(n-1)..(n-1), stored at index k+(n-1)
    # This is a cross-correlation
    
    # Actually, let me just build the full correlation matrix
    # grad[m] = sum_{k} w_k * (1-h[m-k]) - sum_{k} w_k * h[m+k]
    
    # First sum: this is the convolution of one_minus_h with w (reversed)
    # In numpy: np.correlate(one_minus_h, w, 'same')? No, the lengths don't match.
    
    # Let me just do it with explicit indexing
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        k = k_idx - (n - 1)  # actual lag
        w_k = weights[k_idx]
        if w_k < 1e-15:
            continue
        # Term 1: (1-h[m-k]) for m-k in [0,n)
        lo1 = max(0, k)
        hi1 = min(n, n + k)
        if lo1 < hi1:
            grad[lo1:hi1] += w_k * one_minus_h[lo1-k:hi1-k]
        # Term 2: -h[m+k] for m+k in [0,n)
        lo2 = max(0, -k)
        hi2 = min(n, n - k)
        if lo2 < hi2:
            grad[lo2:hi2] -= w_k * h[lo2+k:hi2+k]
    
    return grad * dx

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        h += (target_sum - s) / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def gradient_smooth_max(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    M = np.max(corr)
    shifted = beta * (corr - M)
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum < 1e-300:
        return np.zeros(n)
    weights /= wsum
    one_minus_h = 1.0 - h
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        w_k = weights[k_idx]
        if w_k < 1e-16:
            continue
        k = k_idx - (n - 1)
        lo1 = max(0, k)
        hi1 = min(n, n + k)
        if lo1 < hi1:
            grad[lo1:hi1] += w_k * one_minus_h[lo1 - k:hi1 - k]
        lo2 = max(0, -k)
        hi2 = min(n, n - k)
        if lo2 < hi2:
            grad[lo2:hi2] -= w_k * h[lo2 + k:hi2 + k]
    return grad * dx

def smooth_max_obj(h, beta):
    corr = np.correlate(h, 1.0 - h, mode="full")
    M = np.max(corr)
    return M + np.log(np.sum(np.exp(beta * (corr - M)))) / beta

def grad_descent(h, target_sum, beta, max_iters, base_lr):
    n = len(h)
    momentum = np.zeros(n)
    mu = 0.85
    for it in range(max_iters):
        g = gradient_smooth_max(h, beta)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        momentum = mu * momentum + (1 - mu) * g
        m_norm = np.linalg.norm(momentum)
        if m_norm < 1e-14:
            break
        lr = base_lr / (1.0 + 0.05 * it)
        direction = momentum / m_norm
        f0 = smooth_max_obj(h, beta)
        alpha = lr
        improved = False
        for _ in range(20):
            h_try = project_h(h - alpha * direction, target_sum)
            f_try = smooth_max_obj(h_try, beta)
            if f_try < f0:
                h = h_try
                improved = True
                break
            alpha *= 0.5
        if not improved:
            break
    return h

def local_search_01(h, max_rounds=50):
    n = len(h)
    h = (h > 0.5).astype(float)
    best_c5 = compute_c5(h)
    for _ in range(max_rounds):
        ones = list(np.where(h > 0.5)[0])
        zeros = list(np.where(h < 0.5)[0])
        if not ones or not zeros:
            break
        improved = False
        n_trials = min(len(ones) * len(zeros), 300)

            h[j] = 1.0

                if not ones or not zeros:
                    break
            else:
                h[i] = 1.0
                h[j] = 0.0
        if not improved:
            break
    return h, best_c5
