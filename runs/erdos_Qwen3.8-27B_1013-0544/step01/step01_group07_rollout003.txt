import time
import cvxpy as cp
from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_all_corr(h):
    return np.correlate(h, 1 - h, mode="full")

def project_onto_constraint(h, target_sum):
    """Project h onto {0 ≤ h ≤ 1, sum(h) = target_sum}"""
    h = np.clip(h, 0.0, 1.0)
    # Adjust to meet sum constraint
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        # Distribute proportionally
        if diff > 0:
            # Need to increase: add to entries < 1
            room = 1.0 - h
            if np.sum(room) > 0:
                h = h + diff * room / np.sum(room)
            else:
                h = h + diff / len(h)
        else:
            # Need to decrease: subtract from entries > 0
            floor = h
            if np.sum(floor) > 0:
                h = h + diff * floor / np.sum(floor)
            else:
                h = h + diff / len(h)
    return np.clip(h, 0.0, 1.0)

def subgradient_step(h, dx, target_sum, step_size, top_k=5):
    """One subgradient descent step"""
    corr = np.correlate(h, 1 - h, mode="full")
    # Find top_k lags
    top_indices = np.argsort(corr)[-top_k:]
    # Average subgradient over top lags
    # Subgradient of corr[k] w.r.t. h: 
    # corr[k] = sum_i h[i] * (1-h[i+k-(n-1)])
    # d(corr[k])/d(h[j]) = (1-h[j+k-(n-1)]) - h[j-(n-1)-k]  (with appropriate bounds)
    
    n = len(h)
    # Actually, let's use numerical subgradient for the max
    # The subgradient of max_k corr[k] is the average of subgradients at active k's
    
    # corr[k] = sum_i h[i] * (1-h[i+k-(n-1)])
    # Let's denote m = n-1
    # corr[k] = sum_i h[i] * (1-h[i+k-m]) where 0 ≤ i < n and 0 ≤ i+k-m < n
    # = sum_i h[i] * (1-h[i+k-m]) for valid indices

    # If j appears as h[i] term: i=j, need 0 ≤ j+k-m < n → m-k ≤ j < n+k-m... 
    # This is getting complicated. Let me just use the numerical approach.
    
    # Alternative: use the fact that corr = np.correlate(h, 1-h)
    # The gradient of corr[k] w.r.t. h[j] can be computed as:
    # corr[k] involves pairs (i, i+k-(n-1))
    # So d(corr[k])/d(h[j]) = sum over all i where i=j: (1-h[i+k-(n-1)])
    #                          + sum over all i where i+k-(n-1)=j: (-h[i])
    
    grad = np.zeros(n)
    m = n - 1
    for k in top_indices:
        # corr[k] = sum_i h[i] * (1-h[i+k-m])
        # First part: d/dh[j] of h[j] * (1-h[j+k-m]) = (1-h[j+k-m]) if j+k-m in [0,n)
        # Second part: d/dh[j] of h[i] * (1-h[i+k-m]) where i+k-m = j, i.e., i = j+m-k
        #   = -h[j+m-k] if j+m-k in [0,n)
        
        for j in range(n):
            idx1 = j + k - m  # where h[j] is the first factor
            idx2 = j + m - k  # where h[j] is the second factor (1-h[...])
            
            if 0 <= idx1 < n:
                grad[j] += (1 - h[idx1])
            if 0 <= idx2 < n:
                grad[j] += (-h[idx2])
    
    grad /= len(top_indices)
    
    # Take step
    h_new = h - step_size * grad
    h_new = project_onto_constraint(h_new, target_sum)
    return h_new

def qp_active_set(h, dx, target_sum, top_k=10, max_iter=50):
    """Minimize sum of squares of top correlations using QP"""
    n = len(h)
    for _ in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        top_indices = np.argsort(corr)[-top_k:]
        
        # Build QP: min sum_{k in top} (corr_k)^2 subject to 0≤h≤1, sum=h_target
        # corr_k = sum_i h[i]*(1-h[i+k-(n-1)])
        # This is quadratic in h, so we can build it as a QP
        
        # For each active k, build the matrix A_k such that corr_k = h^T A_k h + b_k^T h
        # Actually corr_k = sum_i h[i] - sum_i h[i]*h[i+k-m]
        # = sum_i h[i] - sum_i h[i]*h[i+k-m]
        
        # The linear part is sum_i h[i] = 1^T h
        # The quadratic part is -sum_i h[i]*h[i+k-m]
        
        # So corr_k = 1^T h - sum_{i: i+k-m valid} h[i]*h[i+k-m]
        
        # For the QP, we need:
        # min sum_k (1^T h - sum_i h[i]h[i+k-m_k])^2
        # This is 4th order, not quadratic. 
        # 
        # Instead, let's linearize: at current h, corr_k ≈ corr_k(h) + g_k^T (h - h_old)
        # where g_k is the gradient of corr_k at h_old.
        # Then minimize sum_k (corr_k(h) + g_k^T (h-h_old))^2
        # This IS quadratic in h.
        
        # Compute gradients
        m = n - 1
        G = np.zeros((top_k, n))
        for idx, k in enumerate(top_indices):
            for j in range(n):
                idx1 = j + k - m
                idx2 = j + m - k
                if 0 <= idx1 < n:
                    G[idx, j] += (1 - h[idx1])
                if 0 <= idx2 < n:
                    G[idx, j] += (-h[idx2])
        
        # Linearized corr_k ≈ corr_k + G[k,:] @ (h - h_old)
        # = G[k,:] @ h + (corr_k - G[k,:] @ h_old)
        
        c_vec = np.zeros(top_k)
        for idx, k in enumerate(top_indices):
            c_vec[idx] = corr[k] - G[idx] @ h
        
        # min sum_k (G[k,:] @ h + c_k)^2 s.t. 0≤h≤1, 1^T h = target_sum
        # = min h^T (G^T G) h + 2 (G^T c)^T h + ||c||^2
        # P = G^T G, q = G^T c
        
        P = G.T @ G
        q = G.T @ c_vec
        
        h_cvx = cp.Variable(n)
        obj = cp.quad_form(h_cvx, P, assume_PSD=True) + q @ h_cvx
        cons = [h_cvx >= 0, h_cvx <= 1, cp.sum(h_cvx) == target_sum]
        prob = cp.Problem(cp.Minimize(obj), cons)
        
        try:
            prob.solve(solver=cp.SCS, max_iters=2000)
            if h_cvx.value is not None:
                h_new = np.clip(h_cvx.value, 0.0, 1.0)
                h_new = project_onto_constraint(h_new, target_sum)
                c5_new = compute_c5(h_new, dx)
                c5_old = compute_c5(h, dx)
                if c5_new < c5_old:
                    h = h_new
            else:
                break
        except Exception:
            break
    
    return h

def make_binary_pattern(n, period, duty=0.5, seed=0):

    return h

import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_onto_constraint(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            room = 1.0 - h
            s = np.sum(room)
            if s > 1e-12:
                h = h + diff * room / s
            else:
                h = h + diff / len(h)
        else:
            floor = h
            s = np.sum(floor)
            if s > 1e-12:
                h = h + diff * floor / s
            else:
                h = h + diff / len(h)
    return np.clip(h, 0.0, 1.0)

def subgradient_step(h, dx, target_sum, step_size, top_k=5):
    corr = np.correlate(h, 1 - h, mode="full")
    top_indices = np.argsort(corr)[-top_k:]
    n = len(h)
    m = n - 1
    grad = np.zeros(n)
    for k in top_indices:
        for j in range(n):
            idx1 = j + k - m
            idx2 = j + m - k
            if 0 <= idx1 < n:
                grad[j] += (1.0 - h[idx1])
            if 0 <= idx2 < n:
                grad[j] += (-h[idx2])
    grad /= len(top_indices)
    h_new = h - step_size * grad
    return project_onto_constraint(h_new, target_sum)

def qr_active_set(h, dx, target_sum, top_k=12, max_iter=40):
    n = len(h)
    m = n - 1
    h = h.copy()
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        top_indices = np.argsort(corr)[-top_k:]
        G = np.zeros((top_k, n))
        for idx, k in enumerate(top_indices):
            for j in range(n):
                idx1 = j + k - m
                idx2 = j + m - k
                if 0 <= idx1 < n:
                    G[idx, j] += (1.0 - h[idx1])
                if 0 <= idx2 < n:
                    G[idx, j] += (-h[idx2])
        c_vec = np.zeros(top_k)
        for idx, k in enumerate(top_indices):
            c_vec[idx] = corr[k] - G[idx] @ h
        P = G.T @ G
        q = G.T @ c_vec
        P = (P + P.T) / 2.0
        h_cvx = cp.Variable(n)
        obj = cp.quad_form(h_cvx, P, assume_PSD=True) + q @ h_cvx
        cons = [h_cvx >= 0, h_cvx <= 1, cp.sum(h_cvx) == target_sum]
        prob = cp.Problem(cp.Minimize(obj), cons)
        try:
            prob.solve(solver=cp.SCS, max_iters=3000)
            if h_cvx.value is not None:
                h_new = np.clip(h_cvx.value, 0.0, 1.0)
                h_new = project_onto_constraint(h_new, target_sum)
                c5_new = compute_c5(h_new, dx)
                c5_old = compute_c5(h, dx)
                if c5_new < c5_old:
                    h = h_new
                else:
                    h = 0.7 * h + 0.3 * h_new
                    h = project_onto_constraint(h, target_sum)
            else:
                break
        except Exception:
            break
    return h

def smooth_max_obj(h, t, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    max_c = np.max(corr)
    return (max_c + (1.0 / t) * np.log(np.sum(np.exp(t * (corr - max_c))))) * dx

def slsqp_refine(h, dx, target_sum, t=100, maxiter=800):
    n = len(h)
    try:
        res = minimize(
            lambda x: smooth_max_obj(x, t, dx),
            h.copy(),
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            options={'maxiter': maxiter, 'ftol': 1e-12}
        )
        return project_onto_constraint(res.x, target_sum)
    except Exception:
        return h

def group_descent(h, dx, target_sum, group_size=6, n_rounds=100, t=50, seed=0):
    rng = np.random.RandomState(seed)
    n = len(h)
    best_c5 = compute_c5(h, dx)
    for r in range(n_rounds):
        idx = rng.choice(n, group_size, replace=False)
        sub_h = h[idx].copy()
        fixed_sum = target_sum - (np.sum(h) - np.sum(h[idx]))
        h_orig = h
        def sub_obj(x):
            h_sub = h_orig.copy()
            h_sub[idx] = x
            return smooth_max_obj(h_sub, t, dx)
        try:
            res = minimize(sub_obj, sub_h.copy(), method='SLSQP',
                          bounds=[(0.0, 1.0)] * group_size,
                          constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - fixed_sum}],
                          options={'maxiter': 60, 'ftol': 1e-10})
            h_new = h.copy()
            h_new[idx] = np.clip(res.x, 0.0, 1.0)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                h = h_new
                best_c5 = c5_new
        except Exception:
            pass
    return h, best_c5

def make_structured_binary(n, pattern_type, seed=0):
    h = np.zeros(n)
    x = np.arange(n)
    if pattern_type == 'alternating':
        h[x % 2 == 0] = 1.0
    elif pattern_type == 'period3':
        h[x % 3 < 1] = 1.0
    elif pattern_type == 'period4':
        h[x % 4 < 2] = 1.0
    elif pattern_type == 'random':
        rng = np.random.RandomState(seed)
        ones = rng.choice(n, n // 2, replace=False)
        h[ones] = 1.0
    elif pattern_type == 'gaussian':
        center = n / 2.0
        width = n / 4.0
        h = 0.5 + 0.45 * np.cos(2 * np.pi * x / n)
        h = np.clip(h, 0, 1)
        h = project_onto_constraint(h, n / 2.0)
    num_ones = np.sum(h == 1)
    target = n / 2
    if pattern_type != 'gaussian':
        if num_ones > target:
            ones_idx = np.where(h == 1)[0]
            rng = np.random.RandomState(seed)
            to_remove = rng.choice(ones_idx, int(num_ones - target), replace=False)
            h[to_remove] = 0.0
        elif num_ones < target:
            zeros_idx = np.where(h == 0)[0]
            rng = np.random.RandomState(seed)
            to_add = rng.choice(zeros_idx, int(target - num_ones), replace=False)
            h[to_add] = 1.0
    return h
