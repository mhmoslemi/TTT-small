def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def soft_max_obj(h, dx, t):
    corr = np.correlate(h, 1 - h, mode="full") * dx
    m = np.max(corr)
    shifted = t * (corr - m)
    return m + (1.0 / t) * np.log(np.sum(np.exp(shifted)))

def fix_sum(h, target_sum):
    h = h.copy()
    for _ in range(30):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def optimize_lbfgs(h, dx, target_sum, t, maxiter=500):
    n = len(h)

        hc = np.clip(x, 0.0, 1.0)
        return soft_max_obj(hc, dx, t) + lam * (np.sum(hc) - target_sum) ** 2
