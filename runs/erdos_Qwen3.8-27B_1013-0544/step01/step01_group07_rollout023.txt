    target_sum = n / 2.0
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = fix_sum(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)

    # Phase 1: Alternating LP from initial
    h_alt, c5_alt = alternating_lp(best_h, n, dx, max_iter=50, damping=1.0, t0=t0, budget_s=budget_s)
    if c5_alt < best_c5:
        best_h = h_alt
        best_c5 = c5_alt

    # Phase 2: SLSQP polish
    if elapsed() < budget_s - 80:
        for t in [10, 30, 80, 200]:
            if elapsed() > budget_s - 50:
                break
            h_new = slsqp_polish(best_h, dx, target_sum, t=t, maxiter=400)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    # Phase 3: Alternating LP with damping for finer convergence
    if elapsed() < budget_s - 80:
        for damp in [0.9, 0.7, 0.5]:
            if elapsed() > budget_s - 50:
                break
            h_d, c5_d = alternating_lp(best_h, n, dx, max_iter=30, damping=damp, t0=t0, budget_s=budget_s)
            if c5_d < best_c5:
                best_h = h_d
                best_c5 = c5_d
                h_new = slsqp_polish(best_h, dx, target_sum, t=100, maxiter=300)
                c5_new = compute_c5(h_new, dx)
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new

    # Phase 4: Group LP coordinate descent
    if elapsed() < budget_s - 60:
        h_g, c5_g = group_lp_descent(best_h, n, dx, group_size=5, n_rounds=50, t0=t0, budget_s=budget_s)
        if c5_g < best_c5:
            best_h = h_g
            best_c5 = c5_g

    # Phase 5: More alternating LP rounds
    if elapsed() < budget_s - 50:
        h_alt2, c5_alt2 = alternating_lp(best_h, n, dx, max_iter=40, damping=1.0, t0=t0, budget_s=budget_s)
        if c5_alt2 < best_c5:
            best_h = h_alt2
            best_c5 = c5_alt2

    # Phase 6: Random perturbations + LP
    if elapsed() < budget_s - 40:
        for trial in range(8):
            if elapsed() > budget_s - 25:
                break
            scale = 0.05 + 0.03 * trial
            h_pert = best_h + np.random.randn(n) * scale
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = fix_sum(h_pert, target_sum)
            h_r, c5_r = alternating_lp(h_pert, n, dx, max_iter=25, damping=1.0, t0=t0, budget_s=budget_s)
            if c5_r < best_c5:
                best_h = h_r
                best_c5 = c5_r
                if elapsed() < budget_s - 20:
                    h_new = slsqp_polish(best_h, dx, target_sum, t=100, maxiter=200)
                    c5_new = compute_c5(h_new, dx)
                    if c5_new < best_c5:
                        best_h = h_new
                        best_c5 = c5_new

    # Phase 7: Final SLSQP polish with high t
    if elapsed() < budget_s - 10:
        for t in [200, 500]:
            if elapsed() > budget_s - 5:
                break
            h_final = slsqp_polish(best_h, dx, target_sum, t=t, maxiter=2000)
            c5_final = compute_c5(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final

    # Final group descent polish
    if elapsed() < budget_s - 5:
        h_g2, c5_g2 = group_lp_descent(best_h, n, dx, group_size=4, n_rounds=20, t0=t0, budget_s=budget_s)
        if c5_g2 < best_c5:
            best_h = h_g2
            best_c5 = c5_g2

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = fix_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h, dx)

    return best_h, best_c5, n
