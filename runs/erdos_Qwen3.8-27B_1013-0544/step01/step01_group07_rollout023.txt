def projected_grad_descent(h0, dx, target_sum, t_init=10, t_max=500, n_iters=5000,
                           lr_init=0.05, momentum=0.9, time_limit=None, t0=None):
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)  # momentum
    lr = lr_init
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    for it in range(n_iters):
        if time_limit is not None and time.time() - t0 > time_limit:
            break
        
        # Increase temperature gradually
        t = min(t_max, t_init * (1 + it / 100.0))
        
        corr = compute_corr(h)
        weights = get_weights(corr, t)
        g = grad_smooth_max(h, weights, dx)
        
        # Nesterov-like momentum step
        v_old = v.copy()
        v = momentum * v - lr * g
        h_candidate = project(h + v, target_sum)
        
        # Check if this improves
        c5_cand = compute_c5(h_candidate, dx)
        if c5_cand < best_c5:
            best_c5 = c5_cand
            best_h = h_candidate.copy()
            h = h_candidate
        else:
            # Reduce learning rate
            lr *= 0.9
            v = v_old

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1 - h, mode="full")

def get_weights(corr, t):
    max_c = np.max(corr)
    shifted = t * (corr - max_c)
    exp_shifted = np.exp(shifted)
    return exp_shifted / np.sum(exp_shifted)

def project(x, target_sum):
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def grad_smooth_max(h, weights, dx):
    n = len(h)
    idx = np.arange(n)
    lags = np.arange(-(n-1), n)
    diff_idx = idx[:, None] - lags[None, :]
    valid1 = (diff_idx >= 0) & (diff_idx < n)
    sum_idx = idx[:, None] + lags[None, :]
    valid2 = (sum_idx >= 0) & (sum_idx < n)
    h1 = np.where(valid1, h[np.clip(diff_idx, 0, n-1)], 0.0)
    h2 = np.where(valid2, h[np.clip(sum_idx, 0, n-1)], 0.0)
    grad = (weights[None, :] * (1.0 - h1)).sum(axis=1) - (weights[None, :] * h2).sum(axis=1)
    return grad * dx

def pgd_optimize(h0, dx, target_sum, t0_val, t1_val, n_iters, lr_init, time_limit, t_start):

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def get_weights(corr, t):
    mc = np.max(corr)
    s = t * (corr - mc)
    e = np.exp(s)
    return e / e.sum()

def project(x, ts):
    lo, hi = -1.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(x - mid, 0.0, 1.0)) > ts:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def grad_sm(h, w, dx):
    n = len(h)
    idx = np.arange(n)
    lags = np.arange(-(n-1), n)
    d = idx[:, None] - lags[None, :]
    s = idx[:, None] + lags[None, :]
    m1 = (d >= 0) & (d < n)
    m2 = (s >= 0) & (s < n)
    h1 = np.where(m1, h[np.clip(d, 0, n-1)], 0.0)
    h2 = np.where(m2, h[np.clip(s, 0, n-1)], 0.0)
    return ((w[None, :] * (1.0 - h1)).sum(1) - (w[None, :] * h2).sum(1)) * dx

                h_s = make_structured(n, freq, amp)
                h_s = project(h_s, ts)
                h_new, c_new = pgd(h_s, dx, ts, 10, 100, 5000, 0.01, 0.9, t0 + budget_s - 50)
                if c_new < bc:
                    bh, bc = h_new, c_new

    # Phase 4: Random perturbations of best + PGD refinement
    for _ in range(20):
        if time.time() > t0 + budget_s - 30:
            break
        hp = bh.copy()
        k = np.random.randint(3, 10)
        idx_p = np.random.choice(n, k, replace=False)
        hp[idx_p] += np.random.randn(k) * 0.12
        hp = project(hp, ts)
        h_new, c_new = pgd(hp, dx, ts, 50, 500, 6000, 0.006, 0.9, t0 + budget_s - 30)
        if c_new < bc:
            bh, bc = h_new, c_new

    # Phase 5: Final high-temperature polish
    if time.time() < t0 + budget_s - 5:
        h_new, c_new = pgd(bh, dx, ts, 500, 3000, 20000, 0.002, 0.94, t0 + budget_s - 5)
        if c_new < bc:
            bh, bc = h_new, c_new

    bh = np.clip(bh, 0, 1)
    bh = project(bh, ts)
    bc = compute_c5(bh, dx)
    return bh, bc, n
