def project_feasible(h, n):

    h = np.clip(h.copy(), 0.0, 1.0)

        if abs(diff) < 1e-14:

            if rsum < 1e-15:

        else:

            if hsum < 1e-15:
                break
            h = np.maximum(h + diff * h / hsum, 0.0)

def compute_deriv(h, i, n):
    """Compute ∂(correlation)/∂h[i] as array of length 2n-1."""
    dx = 2.0 / n
    d = np.zeros(2 * n - 1)
    # For k >= 0 (m = n-1+k, k=0..n-1):
    #   d[k] = dx * [(1-h[i-k])*I(k<=i) - h[i+k]*I(i<=n-1-k)]
    for k in range(n):
        m = n - 1 + k  # index in full correlation array
        dk = 0.0
        if k <= i and i - k >= 0:
            dk += 1.0 - h[i - k]
        if i <= n - 1 - k and i + k < n:
            dk -= h[i + k]

        dk = 0.0

            dk += 1.0 - h[i + kp]
        if kp <= i and i - kp >= 0:
            dk -= h[i - kp]

    return d

def solve_1d_minimax(c, delta_d, lo, hi):
    """Solve min_{x in [lo,hi]} max_m (c[m] + x * delta_d[m])."""
    if lo >= hi:
        return lo
    a, b = lo, hi
    for _ in range(40):
        mid = 0.5 * (a + b)
        vals = c + mid * delta_d
        fmid = vals.max()
        active = vals >= fmid - 1e-13
        if not active.any():
            active = vals >= fmid - 1e-10
        slopes = delta_d[active]
        if slopes.max() < -1e-14:
            a = mid
        elif slopes.min() > 1e-14:
            b = mid
        else:
            break
    x = 0.5 * (a + b)
    return max(lo, min(hi, x))

    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()

    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

    return np.ones(n)

def paired_coord_descent(h, n, rng, n_steps, time_limit, t0):
    """Run paired coordinate descent for n_steps or until time_limit exceeded."""
    best_c5 = compute_c5(h, n)
    c = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
    for step in range(n_steps):
        if time.time() - t0 > time_limit:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        di = compute_deriv(h, i, n)
        dj = compute_deriv(h, j, n)
        dd = di - dj
        lo = max(-h[i], h[j] - 1.0)
        hi = min(1.0 - h[i], h[j])
        if lo >= hi:
            continue
        delta = solve_1d_minimax(c, dd, lo, hi)
        if abs(delta) < 1e-15:

        h[i] += delta
        h[j] -= delta
        # Update c
        c += delta * dd
        # Recompute c5
        c5 = c.max()
        if c5 < best_c5:
            best_c5 = c5
    return h, best_c5
