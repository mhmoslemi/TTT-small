- Case 2: h[i-l]*(1-h[i]) where h[i-l] is fixed → linear in h[i]

2. For all other m: c[m](t) = A[m] + B[m]*t (linear)

    for rd in range(max_rounds):
        if time_limit is not None and time.time() > time_limit:
            break
        improved = False
        order = rng_cd.permutation(n)
        for i in order:
            c = np.correlate(h, 1 - h, mode='full')
            m_len = len(c)
            lags = np.arange(m_len) - (n - 1)
            A = np.zeros(m_len)
            B = np.zeros(m_len)
            C = np.zeros(m_len)
            hi = h[i]
            for mi in range(m_len):
                l = lags[mi]
                j_start = max(0, -l)
                j_end = min(n - 1, n - 1 - l)
                A[mi] = c[mi]
                if j_start <= i <= j_end and 0 <= i + l < n:
                    A[mi] -= hi * (1.0 - h[i + l])
                    B[mi] += (1.0 - h[i + l])
                    if l == 0:
                        C[mi] += -1.0
                if l != 0 and j_start <= i - l <= j_end and 0 <= i < n:
                    j2 = i - l
                    A[mi] -= h[j2] * (1.0 - hi)
                    B[mi] += -h[j2]
            def obj_1d(t):
                return float(np.max(A + B * t + C * t * t))
            res = minimize_scalar(obj_1d, bounds=(0, 1), method='bounded',
                                  options={'xatol': 1e-12})
            if abs(res.x - hi) > 1e-11:
                h[i] = res.x
                improved = True
        h = project_feasible(h, n)
        if not improved:
            break
    return h

def slsqp_phase(h, n, alphas, bounds, cons, time_limit, maxiter=10000):
    h_local = h.copy()
    c5_local = compute_c5(h_local, n)
    for alpha in alphas:
        if time.time() > time_limit:
            break
        try:
            res = minimize(obj_smooth, h_local, args=(n, alpha), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-18})
            h_try = project_feasible(res.x, n)
            c = compute_c5(h_try, n)
            if c < c5_local - 1e-15:
                h_local = h_try
                c5_local = c
        except Exception:
            pass
    return h_local, c5_local

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        n_init = 80
        h_init = np.full(n_init, 0.5)

    best_h = project_feasible(h_init, n_init)
    best_c5 = compute_c5(best_h, n_init)

    n_candidates = sorted(set([n_init] + [60, 80, 100, 120, 160]),
                          key=lambda x: abs(x - n_init))
    n_candidates = [n for n in n_candidates if n <= 200][:4]

    time_fraction = [0.40, 0.25, 0.20, 0.15]
    time_limit = t0 + budget_s * 0.85

    for idx, n in enumerate(n_candidates):
        tl = time_limit if idx < len(n_candidates) - 1 else time_limit
        if time.time() > tl:
            break

        if n == n_init:
            h0 = project_feasible(h_init, n)
        else:
            x_coarse = np.linspace(0, 2, n_init)
            x_fine = np.linspace(0, 2, n)
            h0 = np.interp(x_fine, x_coarse, h_init)
            h0 = project_feasible(h0, n)

        x = np.linspace(0, 2, n)
        patterns = [h0.copy()]
        for freq in [1, 2, 3]:
            for amp in [0.1, 0.2, 0.3, 0.4]:
                p = 0.5 + amp * np.cos(np.pi * freq * x)
                patterns.append(project_feasible(p, n))
        h0 = min(patterns, key=lambda p: compute_c5(p, n))
        c5 = compute_c5(h0, n)
        if c5 < best_c5:
            best_c5, best_h = c5, h0.copy()

        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]

        # Phase 1: SLSQP continuation
        alphas = [20, 50, 100, 300, 1000, 5000]
        h_local, c5_local = slsqp_phase(h0, n, alphas, bounds, cons, tl, maxiter=8000)
        if c5_local < best_c5:
            best_c5, best_h = c5_local, h_local.copy()

        # Phase 2: Coordinate descent
        if time.time() < tl:
            h_cd = coord_descent(h_local, n, max_rounds=40, time_limit=tl)
            c5_cd = compute_c5(h_cd, n)
            if c5_cd < best_c5:
                best_c5, best_h = c5_cd, h_cd.copy()

        # Phase 3: SLSQP again with high alpha
        if time.time() < tl - 5:
            alphas2 = [5000, 20000]
            h_local2, c5_local2 = slsqp_phase(h_cd, n, alphas2, bounds, cons, tl, maxiter=15000)
            if c5_local2 < best_c5:
                best_c5, best_h = c5_local2, h_local2.copy()

        # Phase 4: More coordinate descent
        if time.time() < tl - 3:
            h_cd2 = coord_descent(h_local2, n, max_rounds=30, time_limit=tl)
            c5_cd2 = compute_c5(h_cd2, n)
            if c5_cd2 < best_c5:
                best_c5, best_h = c5_cd2, h_cd2.copy()

    # Final polish on best solution
    n_best = len(best_h)
    bounds = [(0.0, 1.0)] * n_best
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n_best,)}]
    final_tl = t0 + budget_s - 5
    for alpha in [50000, 200000, 1000000]:
        if time.time() > final_tl:
            break
        try:
            res = minimize(obj_smooth, best_h, args=(n_best, alpha), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 30000, 'ftol': 1e-20})
            h_try = project_feasible(res.x, n_best)
            c = compute_c5(h_try, n_best)
            if c < best_c5:
                best_c5, best_h = c, h_try
        except Exception:
            pass

    # Final coordinate descent
    if time.time() < t0 + budget_s - 3:
        best_h = coord_descent(best_h, n_best, max_rounds=20, time_limit=t0 + budget_s - 2)
        best_c5 = compute_c5(best_h, n_best)

    best_h = project_feasible(best_h, n_best)
    best_c5 = compute_c5(best_h, n_best)
    return best_h, best_c5, n_best
