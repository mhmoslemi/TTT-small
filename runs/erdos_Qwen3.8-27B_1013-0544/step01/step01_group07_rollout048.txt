def group_lp_solve(h, idx, target_sum):
    """Solve LP subproblem for group idx, return new values or None."""
    N = len(h)
    m = len(idx)
    A = np.zeros((2 * N - 1, m))
    for ll in range(m):
        i = idx[ll]
        j_start = max(0, i)
        j_end = min(2 * N - 2, i + N - 1)
        if j_end >= j_start:
            A[j_start:j_end + 1, ll] = 1.0 - 2.0 * h[j_start - i:j_end - i + 1]
    corr = np.correlate(h, 1.0 - h, mode="full")
    b = corr - A @ h[idx]
    target_gs = float(np.sum(h[idx]))
    c_obj = np.zeros(m + 1)
    c_obj[-1] = 1.0
    A_ub = np.hstack([A, -np.ones((2 * N - 1, 1))])
    b_ub = -b
    A_eq = np.zeros((1, m + 1))
    A_eq[0, :m] = 1.0
    b_eq = np.array([target_gs])
    bounds = [(0.0, 1.0)] * m + [(None, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return np.clip(res.x[:m], 0.0, 1.0)
    except Exception:
        pass
    return None

def group_lp_descent(h, target_sum, group_size=8, n_rounds=500, seed=42,
                     time_limit=100.0, t_start=None):
    N = len(h)
    rng = np.random.RandomState(seed)
    best_c5 = compute_c5(h, 2.0 / N)
    t_start = t_start if t_start is not None else time.time()
    stall = 0
    for r in range(n_rounds):
        if time.time() - t_start > time_limit:
            break
        idx = rng.choice(N, group_size, replace=False)
        new_vals = group_lp_solve(h, idx, target_sum)
        if new_vals is not None:
            h[idx] = new_vals
            c5_new = compute_c5(h, 2.0 / N)
            if c5_new < best_c5:
                best_c5 = c5_new
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        if stall > 500:
            break
    return h, best_c5

def sequential_lp_sweep(h, target_sum, group_size=8, n_passes=5,
                        time_limit=100.0, t_start=None):
    N = len(h)
    best_c5 = compute_c5(h, 2.0 / N)
    t_start = t_start if t_start is not None else time.time()
    stride = max(1, group_size // 2)
    for p in range(n_passes):
        if time.time() - t_start > time_limit:
            break
        for start in range(0, N - group_size + 1, stride):
            idx = np.arange(start, start + group_size)
            new_vals = group_lp_solve(h, idx, target_sum)
            if new_vals is not None:
                h[idx] = new_vals
                c5_new = compute_c5(h, 2.0 / N)
                if c5_new < best_c5:
                    best_c5 = c5_new
    return h, best_c5

def slsqp_refine(h, dx, target_sum, t=100, maxiter=300):
    N = len(h)
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        mc = np.max(corr)
        sh = np.clip(t * (corr - mc), -50, 50)
        return (mc + (1.0 / t) * np.log(np.sum(np.exp(sh)))) * dx
    try:
        res = minimize(obj, h.copy(), method='SLSQP',
                       bounds=[(0.0, 1.0)] * N,
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return fix_sum(res.x, target_sum)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def elapsed():
        return time.time() - t0

    h0 = np.array(initial_h_values, dtype=float).copy()
    N = len(h0)
    dx = 2.0 / N
    target_sum = N / 2.0
    h0 = fix_sum(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)
    best_n = N

    # Phase 1: Aggressive random group LP descent (main workhorse)
    remaining = budget_s - 50
    h1, c5_1 = group_lp_descent(best_h, target_sum, group_size=8,
                                 n_rounds=2000, seed=seed,
                                 time_limit=remaining * 0.35, t_start=t0)
    if c5_1 < best_c5:
        best_h, best_c5 = h1.copy(), c5_1

    # Phase 2: Sequential sweep for smoothing
    rem = budget_s - elapsed() - 30
    if rem > 5:
        h1, c5_1 = sequential_lp_sweep(best_h, target_sum, group_size=8,
                                        n_passes=10, time_limit=rem * 0.3, t_start=t0)
        if c5_1 < best_c5:
            best_h, best_c5 = h1.copy(), c5_1

    # Phase 3: Different group sizes for diversity
    for gs in [6, 12, 14]:
        rem = budget_s - elapsed() - 20
        if rem < 5:
            break
        h1, c5_1 = group_lp_descent(best_h, target_sum, group_size=gs,
                                     n_rounds=500, seed=seed + gs * 17,
                                     time_limit=min(rem * 0.2, 30), t_start=t0)
        if c5_1 < best_c5:
            best_h, best_c5 = h1.copy(), c5_1

    # Phase 4: SLSQP refinement
    rem = budget_s - elapsed() - 20
    if rem > 10:
        for t_val in [30, 100, 500]:
            if elapsed() > budget_s - 15:
                break
            h_new = slsqp_refine(best_h, dx, target_sum, t=t_val, maxiter=400)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_h, best_c5 = h_new.copy(), c5_new

    # Phase 5: More LP descent after SLSQP
    rem = budget_s - elapsed() - 15
    if rem > 5:
        h1, c5_1 = group_lp_descent(best_h, target_sum, group_size=8,
                                     n_rounds=1000, seed=seed + 500,
                                     time_limit=min(rem * 0.5, 40), t_start=t0)
        if c5_1 < best_c5:
            best_h, best_c5 = h1.copy(), c5_1

    # Phase 6: Try n=320
    rem = budget_s - elapsed() - 10
    if rem > 30:
        N2 = 2 * N
        dx2 = 2.0 / N2
        ts2 = N2 / 2.0
        x_old = np.linspace(0, 2, N + 1)[:-1]
        x_new = np.linspace(0, 2, N2 + 1)[:-1]
        h2 = np.interp(x_new, x_old, best_h).astype(float)
        h2 = fix_sum(h2, ts2)
        c5_2 = compute_c5(h2, dx2)
        h2, c5_2 = group_lp_descent(h2, ts2, group_size=6,
                                     n_rounds=500, seed=seed + 900,
                                     time_limit=min(rem * 0.6, 60), t_start=t0)
        h2, c5_2 = sequential_lp_sweep(h2, ts2, group_size=6, n_passes=5,
                                        time_limit=min(rem * 0.3, 20), t_start=t0)
        if c5_2 < best_c5:
            best_h, best_c5, best_n = h2.copy(), c5_2, N2

    # Phase 7: Final polish
    rem = budget_s - elapsed() - 3
    if rem > 3:
        Nf = len(best_h)
        tsf = Nf / 2.0
        h1, c5_1 = group_lp_descent(best_h, tsf, group_size=6,
                                     n_rounds=500, seed=seed + 999,
                                     time_limit=max(0.5, rem * 0.8), t_start=t0)
        if c5_1 < best_c5:
            best_h, best_c5 = h1.copy(), c5_1

    best_h = fix_sum(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, best_c5, len(best_h)
