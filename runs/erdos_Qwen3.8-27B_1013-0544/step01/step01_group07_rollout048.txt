```python
    corr = np.correlate(h, g_vec, mode="full")
    k_star = np.argmax(corr)
    # Compute subgradient of c_{k_star} w.r.t. h
    # c_k = sum_i h[i]*(1-h[i+k-(n-1)])
    # dc_k/dh[j] = (1-h[j+k-(n-1)]) - h[j-k+(n-1)] (with bounds)
    shift = k_star - (n - 1)
    grad = np.zeros(n)
    # First term: (1-h[j+shift]) for valid j+shift
    valid1 = (np.arange(n) + shift >= 0) & (np.arange(n) + shift < n)
    grad[valid1] += (1 - h[np.arange(n)[valid1] + shift])
    # Second term: -h[j-shift] for valid j-shift
    valid2 = (np.arange(n) - shift >= 0) & (np.arange(n) - shift < n)
    grad[valid2] -= h[np.arange(n)[valid2] - shift]
    
    h_new = h - eta * grad
    h_new = project_hypersimplex(h_new, n/2)
    return h_new
```
