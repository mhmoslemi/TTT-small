import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target_sum):
    h = h.copy()
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        h += diff / len(h)
    return np.clip(h, 0.0, 1.0)

def pair_golden(h, i, j):
    s = h[i] + h[j]
    t_lo = max(0.0, s - 1.0)
    t_hi = min(1.0, s)
    if t_hi - t_lo < 1e-12:
        return
    gr = 0.6180339887498949
    a, b = t_lo, t_hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    h[i], h[j] = c, s - c
    fc = np.max(np.correlate(h, 1.0 - h, mode="full"))
    h[i], h[j] = d, s - d
    fd = np.max(np.correlate(h, 1.0 - h, mode="full"))
    for _ in range(45):
        if b - a < 1e-13:
            break
        if fc <= fd:
            b, d, fd = d, c, fc
            c = b - gr * (b - a)
            h[i], h[j] = c, s - c
            fc = np.max(np.correlate(h, 1.0 - h, mode="full"))
        else:
            a, c, fc = c, d, fd
            d = a + gr * (b - a)
            h[i], h[j] = d, s - d
            fd = np.max(np.correlate(h, 1.0 - h, mode="full"))
    t_opt = (a + b) * 0.5
    h[i] = t_opt
    h[j] = s - t_opt

def smooth_max(h, t, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    c = np.max(corr)
    return (c + np.log(np.sum(np.exp(t * (corr - c)))) / t) * dx

def slsqp(h, dx, target_sum, t=50, maxiter=500):
    try:
        res = minimize(
            lambda x: smooth_max(x, t, dx),
            h.copy(), method='SLSQP',
            bounds=[(0.0, 1.0)] * len(h),
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        return fix_sum(res.x, target_sum)
    except:
        return h

def block_optimize(h, idx, dx, target_sum, t=40, maxiter=80):
    n = len(h)
    g = len(idx)
    sub_sum = target_sum - (np.sum(h) - np.sum(h[idx]))
    def obj(x):
        ht = h.copy()
        ht[idx] = x
        return smooth_max(ht, t, dx)
    try:
        res = minimize(obj, h[idx].copy(), method='SLSQP',
                       bounds=[(0.0, 1.0)] * g,
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - sub_sum}],
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        h[idx] = np.clip(res.x, 0, 1)
    except:
        pass

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.arange(old_n) * (2.0 / old_n)
    x_new = np.arange(new_n) * (2.0 / new_n)
    return np.interp(x_new, x_old, h)

def pair_descent(h, n_pairs, rng):
    n = len(h)
    for _ in range(n_pairs):
        i = rng.randint(n)
        j = rng.randint(n)
        if i != j:
            pair_golden(h, i, j)
