```python
def binary_local_search(h_binary, max_iters=200, time_limit=30):
    n = len(h_binary)
    h_b = h_binary.copy()
    corr = np.correlate(h_b, 1.0 - h_b, mode='full') * (2.0/n)
    current_max = float(np.max(corr))
    
    t_start = time.time()
    for it in range(max_iters):
        if time.time() - t_start > time_limit:
            break
        
        ones = np.where(h_b == 1.0)[0]
        zeros = np.where(h_b == 0.0)[0]
        n_ones = len(ones)
        n_zeros = len(zeros)
        
        if n_ones == 0 or n_zeros == 0:
            break
        
        # Precompute flip vectors
        ones, zeros, f10, f01 = compute_flip_vectors(h_b)
        
        # Find best swap
        # new_max[i,j] = max(corr + f10[i] + f01[j])
        new_max = np.max(corr[None, None, :] + f10[:, None, :] + f01[None, :, :], axis=2)
        flat_idx = np.argmin(new_max)
        i_idx, j_idx = np.unravel_index(flat_idx, new_max.shape)
        best_val = new_max[i_idx, j_idx]
```

```python
    # So corr[m] = sum_j (1-h[j]) * h[j + m - n + 1]
    # The "effective lag" is m - n + 1.
    # If I define lag = m - n + 1, then corr[m] = sum_j (1-h[j]) * h[j + lag]
    
    # ∂corr[m]/∂h[j] = (1-h[j]) * δ_{j+lag, valid} ... hmm this is getting complicated.
    
    # Let me just use the definition directly.
    # corr[m] = sum_i h[i] * (1-h[i - m + n - 1])
    # Let p = m - n + 1 (the "lag"), so corr[m] = sum_i h[i] * (1-h[i + p])
    # where h[i+p] = 0 if i+p is out of [0, n-1]
    
    # ∂corr[m]/∂h[j]:
    # Term 1: from h[j] in the first factor: (1-h[j+p]) if j+p is valid
    # Term 2: from h[j] in the second factor (h[i+p] = h[j] means i = j-p): -h[j-p] if j-p is valid
    # So ∂corr[m]/∂h[j] = (1-h[j+p]) * I(j+p in [0,n-1]) - h[j-p] * I(j-p in [0,n-1])
    
    p = k_star - n + 1  # the actual lag
    for j in range(n):
        g = 0.0
        if 0 <= j + p < n:
            g += (1.0 - h[j + p])
        if 0 <= j - p < n:
            g -= h[j - p]
        grad[j] = g
    
    # Step
    h_new = h - step_size * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_h(h_new, n * 0.5)
    return h_new
```

This is O(n) per step, which is very fast. I can take many steps.

But the issue with the subgradient method is that it only addresses one lag at a time. If multiple lags are near the maximum, I need to address all of them. A better approach is:

**Averaged subgradient**: Take the weighted average of gradients at the active lags, where the weights are proportional to how close each lag is to the maximum.

```python
def averaged_subgradient_step(h, weights, step_size):
    n = len(h)
    grad = np.zeros(n)
    for k_star, w in weights:
        p = k_star - n + 1
        for j in range(n):
            g = 0.0
            if 0 <= j + p < n:
                g += (1.0 - h[j + p])
            if 0 <= j - p < n:
                g -= h[j - p]
            grad[j] += w * g
    h_new = h - step_size * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_h(h_new, n * 0.5)
    return h_new
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))


def project_h(h, target):
    h = h.copy()
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return np.clip(h, 0.0, 1.0)
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = np.sum(mask)
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)


def t_objective(v):
    return v[-1]


def t_ineq(v):
    n = len(v) - 1
    h = v[:n]
    t = v[-1]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return t - corr


def t_eq_fun(v):
    n = len(v) - 1
    return np.sum(v[:n]) - n * 0.5


def t_eq_jac(v):
    n = len(v) - 1
    j = np.zeros(len(v))
    j[:n] = 1.0
    return j


def optimize_t_var(h0, max_iter=400, ftol=1e-10):
    n = len(h0)
    c5_cur = compute_c5(h0)
    v0 = np.append(h0.copy(), c5_cur + 0.02)
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.5)]
    cons = [
        {'type': 'ineq', 'fun': t_ineq},
        {'type': 'eq', 'fun': t_eq_fun, 'jac': t_eq_jac}
    ]
    res = minimize(t_objective, v0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': ftol})
    h_new = np.clip(res.x[:n], 0.0, 1.0)
    h_new = project_h(h_new, n * 0.5)
    return h_new, compute_c5(h_new)


def smooth_obj(h, temp):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    return m + temp * np.log(np.sum(np.exp((corr - m) / temp)))


def sum_eq_fun(h):
    return np.sum(h) - len(h) * 0.5


def sum_eq_jac(h):
    return np.ones(len(h))


def optimize_smooth(h0, temp, max_iter=300, ftol=1e-12):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac}]
    res = minimize(smooth_obj, h0.copy(), args=(temp,), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': ftol})
    h_new = np.clip(res.x, 0.0, 1.0)
    h_new = project_h(h_new, n * 0.5)
    return h_new, compute_c5(h_new)


def binary_local_search(h_bin, max_iters=200, time_limit=30.0):
    n = len(h_bin)
    h_b = h_bin.copy()
    dx = 2.0 / n
    corr = np.correlate(h_b, 1.0 - h_b, mode='full') * dx
    current_max = float(np.max(corr))
    t_start = time.time()
    for it in range(max_iters):
        if time.time() - t_start > time_limit:
            break
        ones = np.where(h_b == 1.0)[0]
        zeros = np.where(h_b == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        f10 = np.zeros((len(ones), 2 * n - 1))
        for idx in range(len(ones)):
            i = ones[idx]
            dh = np.zeros(n)
            dh[i] = -1.0
            d1h = np.zeros(n)
            d1h[i] = 1.0
            f10[idx] = (np.correlate(dh, 1.0 - h_b, 'full') +
                        np.correlate(h_b, d1h, 'full')) * dx
        f01 = np.zeros((len(zeros), 2 * n - 1))
        for idx in range(len(zeros)):
            j = zeros[idx]
            dh = np.zeros(n)
            dh[j] = 1.0
            d1h = np.zeros(n)
            d1h[j] = -1.0
            f01[idx] = (np.correlate(dh, 1.0 - h_b, 'full') +
                        np.correlate(h_b, d1h, 'full')) * dx
        new_max = np.max(corr[None, None, :] + f10[:, None, :] + f01[None, :, :], axis=2)
        flat_idx = np.argmin(new_max)
        i_idx, j_idx = np.unravel_index(flat_idx, new_max.shape)
        if new_max[i_idx, j_idx] >= current_max - 1e-12:
            break
        h_b[ones[i_idx]] = 0.0
        h_b[zeros[j_idx]] = 1.0
        corr = np.correlate(h_b, 1.0 - h_b, mode='full') * dx
        current_max = float(np.max(corr))
    return h_b, current_max


def subgradient_optimize(h0, max_iters=5000, time_limit=30.0):
    n = len(h0)
    h = h0.copy()
    dx = 2.0 / n
    lr = 0.5 / n
    t_start = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(max_iters):
        if time.time() - t_start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        c5_cur = float(np.max(corr))
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h.copy()
        thresh = c5_cur - 0.005
        active = np.where(corr >= thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        weights = corr[active] - thresh + 0.001
        weights = weights / np.sum(weights)
        grad = np.zeros(n)
        for k_m, w in zip(active, weights):
            p = int(k_m) - n + 1
            for j in range(n):
                g = 0.0
                if 0 <= j + p < n:
                    g += (1.0 - h[j + p])
                if 0 <= j - p < n:
                    g -= h[j - p]
                grad[j] += w * g
        h = h - lr * grad
        h = np.clip(h, 0.0, 1.0)
        h = project_h(h, n * 0.5)
        lr *= 0.999
    return best_h, best_c5


def interp_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.arange(n_old) / n_old
    x_new = np.arange(n_new) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_h(h_new, n_new * 0.5)
```
