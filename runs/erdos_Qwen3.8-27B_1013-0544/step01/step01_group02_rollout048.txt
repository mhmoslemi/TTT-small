    for _ in range(60):  # enough for double precision
        mid = (lo + hi) / 2
        c_mid = a + b * mid
        max_idx = np.argmax(c_mid)
        max_val = c_mid[max_idx]
        # Check if all functions achieving max have b > 0 (go left) or b < 0 (go right)
        active = c_mid >= max_val - 1e-12
        if np.all(b[active] > 1e-12):
            hi = mid  # f is increasing at mid, go left
        elif np.all(b[active] < -1e-12):
            lo = mid  # f is decreasing at mid, go right
        else:
            break  # 0 is in the subgradient, mid is optimal
    return (lo + hi) / 2

    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-12:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-12:
                break
            h += (diff / total) * h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-13:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-13:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def find_opt_v(a, b):
    lo, hi = 0.0, 1.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        vals = a + b * mid
        max_val = np.max(vals)
        active = vals >= max_val - 1e-14
        b_act = b[active]
        if np.all(b_act > 1e-14):
            hi = mid
        elif np.all(b_act < -1e-14):
            lo = mid
        else:
            break
    return (lo + hi) * 0.5

def coord_descent(h, n, max_sweeps=5000, time_limit=None, t0=None, rng=None):
    dx = 2.0 / n
    offset = n - 1
    m_num = 2 * n - 1
    m_idx = np.arange(m_num)
    best_c5 = compute_c5(h, n)
    for _ in range(max_sweeps):
        if time_limit is not None and time.time() - t0 > time_limit:
            break
        c = np.correlate(h, 1 - h, mode='full') * dx
        order = rng.permutation(n)
        improved = False
        for i in order:
            idx1 = i + m_idx - offset
            idx2 = i - m_idx + offset
            v1 = (idx1 >= 0) & (idx1 < n)
            v2 = (idx2 >= 0) & (idx2 < n)
            b = np.zeros(m_num)
            b[v1] = 1.0 - h[idx1[v1]]
            b[v2] -= h[idx2[v2]]
            a = c - b * h[i]
            v_new = find_opt_v(a, b)
            delta = v_new - h[i]
            if abs(delta) > 1e-12:
                h[i] = v_new
                c += dx * delta * b
                improved = True
        h = project_feasible(h, n)
        c5 = compute_c5(h, n)
        if c5 < best_c5:
            best_c5 = c5
        if not improved:
            break
    return h, best_c5

def obj_smooth(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    return float(m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha)

def sum_con(h, n):
    return np.sum(h) - n / 2.0

def sum_con_jac(h, n):
    return np.ones(n)
