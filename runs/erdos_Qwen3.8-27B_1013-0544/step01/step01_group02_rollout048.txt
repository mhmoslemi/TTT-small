            if i == j:
                continue
            improved = False
            for delta in deltas:
                for s in [1, -1]:
                    di = s * delta
                    hi = best_h[i] + di
                    hj = best_h[j] - di
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h_try = best_h.copy()
                        h_try[i] = hi
                        h_try[j] = hj
                        c_try = compute_c5(h_try, n)
                        if c_try < d:
                            best_h = h_try
                            d = c_try
                            improved = True
                            break
                if improved:
                    break

    best_h = project(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
