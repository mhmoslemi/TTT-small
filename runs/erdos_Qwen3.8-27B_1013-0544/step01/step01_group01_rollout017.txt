import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    return m + (1.0 / beta) * np.log(np.sum(np.exp(beta * (x - m))))

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        h = np.clip(h + diff / len(h), 0.0, 1.0)

def _obj(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    sm = smooth_max(corr, _gs['beta'])
    s = h.sum() - _gs['target']
    return sm * _gs['dx'] + _gs['penalty'] * s * s

    return h.sum() - _gs['target']

    return np.ones(len(h))
