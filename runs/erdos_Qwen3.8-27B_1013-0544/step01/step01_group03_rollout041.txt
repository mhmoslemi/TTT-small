def fix_sum(h, target):

    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = mask.sum()
    if cnt > 0:
        h = h.copy()
        h[mask] += diff / cnt

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def fix_sum(h, target):
    h = np.clip(np.array(h, dtype=float), 0.0, 1.0)
    for _ in range(30):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(obj, tau):
    x = tau * obj
    xm = np.max(x)
    return float(xm / tau + np.log(np.sum(np.exp(x - xm))) / tau)

def slsqp_solve(h0, tau, maxiter=400, ftol=1e-14):
    n = len(h0)
    target = n / 2.0
    bnds = Bounds(np.zeros(n), np.ones(n))
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode="full")
        return smooth_max(c, tau)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bnds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        x = fix_sum(res.x, target)
        return x, evaluate_c5(x)
    except:
        return h0, None

def grad_top_lags(h, n_top=5, sharpness=40):
    """Compute weighted gradient from top lags."""
    n = len(h)
    ov = compute_overlaps(h)
    idx = np.argsort(ov)[::-1][:n_top]
    vals = ov[idx]
    w = np.exp((vals - vals[0]) * sharpness)
    w /= w.sum()
    g = np.zeros(n)
    js = np.arange(n)
    for wi, ii in zip(w, idx):
        d = ii - (n - 1)
        if d >= 0:
            m1 = js < (n - d)
            m2 = js >= d
            g[m1] += wi * (1.0 - h[js[m1] + d])
            g[m2] -= wi * h[js[m2] - d]
        else:
            dd = -d
            m1 = js >= dd
            m2 = js < (n - dd)
            g[m1] += wi * (1.0 - h[js[m1] - dd])
            g[m2] -= wi * h[js[m2] + dd]
    return g * (2.0 / n), ov

def anneal_h(h0, n_iters, t_start, t_end, focus=3, noise_base=0.03):
    n = len(h0)
    target = n / 2.0
    h = fix_sum(h0, target)
    best = h.copy()
    best_c5 = evaluate_c5(h)
    for it in range(n_iters):
        t = t_start * (t_end / t_start) ** (it / max(n_iters - 1, 1))
        g, ov = grad_top_lags(h, n_top=focus, sharpness=30)
        c5 = ov.max()
        scale = 0.05 + 0.1 * (t / t_start)
        h_new = h - g * scale + np.random.normal(0, noise_base * (t / t_start + 0.3), n)
        h_new = fix_sum(h_new, target)
        c5_new = evaluate_c5(h_new)
        if c5_new <= c5 or np.random.random() < np.exp(-(c5_new - c5) / max(t, 1e-8)):
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best = h.copy()
    return best, best_c5

def msequence(order=7):
    n = (1 << order) - 1
    reg = 1
    seq = np.zeros(n)
    for i in range(n):
        seq[i] = reg & 1
        fb = ((reg >> 3) ^ reg) & 1
        reg = ((reg >> 1) | (fb << (order - 1))) & ((1 << order) - 1)
    return seq

def to_balanced_binary(seq, n_target):
    """Take a binary sequence and balance/resize to n_target with exactly n_target//2 ones."""
    n = len(seq)
    # Resize by interpolation (nearest neighbor)
    if n != n_target:
        idx = np.round(np.linspace(0, n - 1, n_target)).astype(int)
        seq = seq[idx]
    # Balance
    target = n_target // 2
    cur_ones = int(seq.sum())
    if cur_ones > target:
        ones_idx = np.where(seq > 0.5)[0]
        seq = seq.copy()
        seq[ones_idx[cur_ones - target:]] = 0
    elif cur_ones < target:
        zeros_idx = np.where(seq < 0.5)[0]
        seq = seq.copy()
        seq[zeros_idx[:target - cur_ones]] = 1
    return seq
