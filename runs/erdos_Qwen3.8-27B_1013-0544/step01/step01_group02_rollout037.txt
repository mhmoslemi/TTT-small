A simple approach: binary search on t using the sign of the "derivative" (the slope of the upper envelope).

Even simpler: since it's convex, just use ternary search. 50 iterations give precision 3^(-50) which is way more than enough.

But we can do even better: the optimal t is either 0, 1, or at a point where two lines cross and both are on the upper envelope. The crossing point of lines k1 and k2 is t = (B_k1 - B_k2)/(A_k2 - A_k1).

For n=80, there are 159 lines. The upper envelope has at most 159 segments. We can:
1. Compute all pairwise crossing points (159*158/2 ≈ 12500 points)
2. Filter to those in [0,1]
3. Evaluate the upper envelope at each candidate point
4. Return the best

Actually that's overkill. Ternary search with 40 iterations is fast enough (40 evaluations of max over 159 values = 6360 ops per coordinate, 80 coordinates = 508800 ops per sweep). Very fast.

    # A_k = (1-h[j+L])*[0<=j+L<n] - h[j-L]*[0<=j-L<n]

            c = np.correlate(h, 1 - h, mode='full')
            return float(np.max(c))
        lo_s, hi_s = lo, hi
        for _ in range(50):
            m1 = lo_s + (hi_s - lo_s) / 3.0
            m2 = hi_s - (hi_s - lo_s) / 3.0
            if f(m1) < f(m2):
                hi_s = m2
            else:
                lo_s = m1
        h[j] = (lo_s + hi_s) / 2.0
        h[j2] = s - h[j]
        done.add(j)
        done.add(j2)
    return h

def random_pair_cd(h, n, rng, n_sweeps=20):
    h = h.copy()
    for _ in range(n_sweeps):
        indices = list(range(n))
        rng.shuffle(indices)
        for i in range(0, n - 1, 2):
            j = indices[i]
            j2 = indices[i + 1]
            s = h[j] + h[j2]
            lo = max(0.0, s - 1.0)
            hi = min(1.0, s)
            if hi - lo < 1e-12:
                continue
            def f(t):
                h[j] = t
                h[j2] = s - t
                c = np.correlate(h, 1 - h, mode='full')
                return float(np.max(c))
            lo_s, hi_s = lo, hi
            for _ in range(50):
                m1 = lo_s + (hi_s - lo_s) / 3.0
                m2 = hi_s - (hi_s - lo_s) / 3.0
                if f(m1) < f(m2):
                    hi_s = m2
                else:
                    lo_s = m1
            h[j] = (lo_s + hi_s) / 2.0
            h[j2] = s - h[j]
    return h

def slsqp_refine(h, n, bounds, cons, alpha, maxiter):
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def full_optimize(h, n, rng, bounds, cons, time_limit):
    t_start = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(best_h, n)
    
    # SLSQP continuation
    alphas = [20, 50, 100, 300, 1000, 3000, 10000]
    for alpha in alphas:
        if time_limit - (time.time() - t_start) < 10:
            break
        best_h = slsqp_refine(best_h, n, bounds, cons, alpha, 1200)
        best_c5 = compute_c5(best_h, n)
    
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    def time_left():
        return budget_s - (time.time() - t0)
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 80
        h0 = np.full(n, 0.5)
    
    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    
    # Phase 1: SLSQP continuation
    alphas = [20, 50, 100, 300, 1000, 3000, 10000]
    for alpha in alphas:
        if time_left() < 15:
            break
        best_h = slsqp_refine(best_h, n, bounds, cons, alpha, 1500)
        best_c5 = compute_c5(best_h, n)
    
    # Phase 2: Pair coordinate descent with structured partnerings
    partners = [
        (np.arange(n) + n // 2) % n,
        (np.arange(n) + 1) % n,
        (np.arange(n) + n // 3) % n,
    ]
    for round_idx in range(3):
        if time_left() < 15:
            break
        for partner in partners:
            if time_left() < 10:
                break
            h_cd = best_h.copy()
            for _ in range(25):
                h_cd = pair_optimize_sweep(h_cd, n, partner)
            c = compute_c5(h_cd, n)
            if c < best_c5:
                best_h, best_c5 = h_cd, c
                if time_left() > 15:
                    best_h = slsqp_refine(best_h, n, bounds, cons, 5000, 1000)
                    best_c5 = compute_c5(best_h, n)
    
    # Phase 3: Random pair CD
    if time_left() > 20:
        for _ in range(5):
            if time_left() < 15:
                break
            h_rp = random_pair_cd(best_h, n, rng, n_sweeps=30)
            c = compute_c5(h_rp, n)
            if c < best_c5:
                best_h, best_c5 = h_rp, c
                if time_left() > 15:
                    best_h = slsqp_refine(best_h, n, bounds, cons, 5000, 1000)
                    best_c5 = compute_c5(best_h, n)
    
    # Phase 4: Random restarts with local optimization
    while time_left() > 30:
        scale = rng.choice([0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7])
        h_pert = best_h + rng.normal(0, scale, size=n)
        h_pert = project_feasible(h_pert, n)
        h_try = slsqp_refine(h_pert, n, bounds, cons, 1000, 800)
        c = compute_c5(h_try, n)
        if c < best_c5:
            best_h, best_c5 = h_try, c
            if time_left() > 20:
                h_rp = random_pair_cd(best_h, n, rng, n_sweeps=15)
                c2 = compute_c5(h_rp, n)
                if c2 < best_c5:
                    best_h, best_c5 = h_rp, c2
                best_h = slsqp_refine(best_h, n, bounds, cons, 10000, 1500)
                best_c5 = compute_c5(best_h, n)
    
    # Phase 5: Try different n values
    for test_n in [40, 60, 100]:
        if time_left() < 60:
            break
        x = np.linspace(0, 2, test_n, endpoint=False) + 1.0 / test_n
        h_test = 0.5 + 0.3 * np.sin(np.pi * x)
        h_test = project_feasible(h_test, test_n)
        bounds_t = [(0.0, 1.0)] * test_n
        cons_t = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (test_n,)}]
        for alpha in [50, 500, 5000]:
            if time_left() < 15:
                break
            h_test = slsqp_refine(h_test, test_n, bounds_t, cons_t, alpha, 1000)
        c_test = compute_c5(h_test, test_n)
        if c_test < best_c5:
            best_h, best_c5, n = h_test, c_test, test_n
            bounds = bounds_t
            cons = cons_t
            # Do pair CD on the new n
            if time_left() > 20:
                h_rp = random_pair_cd(best_h, n, rng, n_sweeps=20)
                c2 = compute_c5(h_rp, n)
                if c2 < best_c5:
                    best_h, best_c5 = h_rp, c2
    
    # Phase 6: Final fine optimization (pair CD + SLSQP)
    if time_left() > 15:
        for _ in range(3):
            if time_left() < 10:
                break
            h_rp = random_pair_cd(best_h, n, rng, n_sweeps=30)
            c = compute_c5(h_rp, n)
            if c < best_c5:
                best_h, best_c5 = h_rp, c
        if time_left() > 10:
            best_h = slsqp_refine(best_h, n, bounds, cons, 10000, 3000)
            best_c5 = compute_c5(best_h, n)
    
    # Final feasibility check
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
