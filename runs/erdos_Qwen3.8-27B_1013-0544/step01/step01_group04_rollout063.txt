    corr = np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    for _ in range(50):
        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - target
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-8) & (h < 1.0 - 1e-8)
        if np.sum(free) > 0:
            h[free] -= diff / np.sum(free)
        else:
            h -= diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = np.sum(h) - target
    if abs(diff) > 1e-14:
        # Distribute among closest-to-boundary indices
        idx = np.argsort(np.abs(h - 0.5))[::-1]
        for i in idx:
            if h[i] > 0.5:
                h[i] -= min(h[i] - 0, abs(diff))
                diff *= -1 if h[i] == 0 else 1
            else:
                h[i] += min(1.0 - h[i], abs(diff))
                diff *= -1 if h[i] == 1 else 1
            if abs(diff) < 1e-14:
                break
    return h

Actually, let me compute the gradient more carefully. For a fixed lag k (where k = lag_idx - (n-1)):

$g_k = \sum_{i}^{valid} h(i+k) \cdot (1 - h(i))$

where the sum is over valid i such that both i and i+k are in [0, n-1].

$\frac{\partial g_k}{\partial h(j)} = \sum_{i: i+k=j} (1-h(i)) + \sum_{i: i=j} (-h(i+k))$

$= (1 - h(j-k)) \cdot \mathbb{1}[j-k \in valid] + (-h(j+k)) \cdot \mathbb{1}[j+k \in valid]$

So the gradient at position j for lag k is:
$dx \cdot [(1-h(j-k)) \cdot \mathbb{1}_{0 \leq j-k < n} - h(j+k) \cdot \mathbb{1}_{0 \leq j+k < n}]$

And the total gradient (averaged over active lags) is the sum of these.

def compute_grad_max(h, dx, n, tol_frac=0.02):
    """Compute gradient of max_k g_k(h) using active set."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    if max_val <= 0:
        return np.zeros(n)
    
    active = np.where(corr >= max_val * (1 - tol_frac))[0]
    offset = n - 1
    grad = np.zeros(n)
    
    for lag_idx in active:
        k = lag_idx - offset
        for j in range(n):
            g = 0.0
            if 0 <= j - k < n:
                g += (1.0 - h[j - k])
            if 0 <= j + k < n:
                g -= h[j + k]
            grad[j] += g * dx
    
    # Normalize by number of active lags
    grad /= len(active)
    return grad

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_box_sum(x, target):
    n = len(x)
    lo, hi = -2.0, 3.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def grad_active(h, dx, tol_frac=0.1):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    if max_val < 1e-15:
        return np.zeros(n)
    active = np.where(corr >= max_val * (1.0 - tol_frac))[0]
    offset = n - 1
    grad = np.zeros(n)
    for idx in active:
        k = idx - offset
        if k >= 0:
            grad[k:] += (1.0 - h[:n - k])
            grad[:n - k] -= h[k:]
        else:
            kk = -k
            grad[:n - kk] += (1.0 - h[kk:])
            grad[kk:] -= h[:n - kk]
    grad *= dx / len(active)
    return grad

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_box_sum(x, target):
    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(70):

        else:

def grad_active(h, dx, tf=0.1):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mv = float(np.max(corr))
    if mv < 1e-15:
        return np.zeros(n)
    active = np.where(corr >= mv * (1.0 - tf))[0]
    off = n - 1
    g = np.zeros(n)
    for idx in active:
        k = idx - off
        if k >= 0:
            g[k:] += 1.0 - h[:n - k]
            g[:n - k] -= h[k:]
        else:
            kk = -k
            g[:n - kk] += 1.0 - h[kk:]
            g[kk:] -= h[:n - kk]
    return g * dx / len(active)
