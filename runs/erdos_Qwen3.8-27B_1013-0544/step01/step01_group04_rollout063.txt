import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - target
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(free):
            h[free] -= diff / np.sum(free)
        else:
            h -= diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = np.sum(h) - target
    if abs(diff) > 1e-14:
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(free):
            h[free] -= diff / np.sum(free)
            h = np.clip(h, 0.0, 1.0)
    # Exact fix
    diff = np.sum(h) - target
    if abs(diff) > 1e-15:
        h[-1] -= diff
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_step(h, dx, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    p = np.argmax(corr)
    k = p - (n - 1)  # actual lag
    # Compute subgradient
    grad = np.zeros(n)
    # Term 1: (1-h[j-k]) * dx for j where 0 <= j-k < n
    lo1 = max(0, k)  # j >= k
    hi1 = min(n, n + k)  # j-k < n => j < n+k
    for j in range(lo1, hi1):
        grad[j] += (1.0 - h[j - k]) * dx
    # Term 2: -h[j+k] * dx for j where 0 <= j+k < n
    lo2 = max(0, -k)  # j >= -k => j+k >= 0
    hi2 = min(n, n - k)  # j+k < n
    for j in range(lo2, hi2):
        grad[j] -= h[j + k] * dx
    return h - alpha * grad, k, np.max(corr) * dx

def de_evolve(h_init, n, dx, n_pop, F_init, F_final, Cr, max_time, rng, start_time):
    # Initialize population
    pop = np.zeros((n_pop, n))
    pop[0] = h_init
    for i in range(1, n_pop):
        pop[i] = project_feasible(rng.rand(n) * 0.5 + h_init * 0.5, n)
    
    # Evaluate
    fitness = np.array([compute_c5(pop[i], dx) for i in range(n_pop)])
    best_idx = np.argmin(fitness)
    best_h = pop[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    gen = 0
    while time.time() - start_time < max_time:
        gen += 1
        progress = min(1.0, gen / 100.0)  # normalize
        F = F_init * (1 - progress) + F_final * progress
        # Actually, let's use time-based decay
        elapsed = time.time() - start_time
        F = F_init + (F_final - F_init) * min(1.0, elapsed / max_time)
        
        for i in range(n_pop):
            # Mutation
            a, b, c = rng.choice(n_pop, 3, replace=False)
            while a == b or a == c or b == c:
                a, b, c = rng.choice(n_pop, 3, replace=False)
            v = pop[a] + F * (pop[b] - pop[c])
            # Crossover
            r = rng.randint(n)
            mask = rng.rand(n) < Cr
            mask[r] = True
            trial = np.where(mask, v, pop[i])
            # Repair
            trial = project_feasible(trial, n)
            # Selection
            f_trial = compute_c5(trial, dx)
            if f_trial < fitness[i]:
                pop[i] = trial
                fitness[i] = f_trial
                if f_trial < best_c5:
                    best_c5 = f_trial
                    best_h = trial.copy()
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_box_sum(h, target):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        tau = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - tau, 0.0, 1.0))
        if s > target:
            lo = tau
        else:
            hi = tau
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def slsqp_minimize(h0, n, dx, max_iter=500):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    
    def objective(x):
        return compute_c5(x, dx)
    
    def obj_grad(x):
        # Subgradient using argmax
        k = int(np.argmax(np.correlate(x, 1.0 - x, mode='full'))) - (n - 1)
        grad = np.zeros(n)
        if k >= 0:
            grad[k:] += (1.0 - x[:n-k]) * dx
            grad[:n-k] -= x[k:] * dx
        else:
            kp = -k
            grad[:n-kp] += (1.0 - x[kp:]) * dx
            grad[kp:] -= x[:n-kp] * dx
        return grad
    
    res = minimize(objective, h0, jac=obj_grad, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-14})
    h_new = project_box_sum(res.x, target)
    return h_new

def de_evolve(h_init, n, dx, n_pop, budget, rng):
    target = n / 2.0
    start_t = time.time()
    
    # Initialize population
    pop = np.zeros((n_pop, n))
    pop[0] = h_init.copy()
    for i in range(1, n_pop):
        # Mix of random and perturbed
        if rng.rand() < 0.5:
            pop[i] = project_box_sum(h_init + rng.randn(n) * 0.3, target)
        else:
            pop[i] = project_box_sum(rng.rand(n), target)
    
    fit = np.array([compute_c5(pop[i], dx) for i in range(n_pop)])
    best_i = int(np.argmin(fit))
    best_h = pop[best_i].copy()
    best_f = fit[best_i]
    
    gen = 0
    total_budget = budget
    while time.time() - start_t < total_budget - 5:
        gen += 1
        progress = (time.time() - start_t) / total_budget
        F = 0.8 - 0.4 * progress  # decay from 0.8 to 0.4
        Cr = 0.8 - 0.3 * progress  # decay from 0.8 to 0.5
        
        for i in range(n_pop):
            # Mutation: DE/rand/1
            idx = rng.choice(n_pop, 3, replace=False)
            a, b, c = idx
            v = pop[a] + F * (pop[b] - pop[c])
            
            # Crossover: binomial
            r = rng.randint(n)
            mask = rng.rand(n) < Cr
            mask[r] = True
            trial = np.where(mask, v, pop[i])
            trial = project_box_sum(trial, target)
            
            f_trial = compute_c5(trial, dx)
            if f_trial < fit[i]:
                pop[i] = trial
                fit[i] = f_trial
                if f_trial < best_f:
                    best_f = f_trial
                    best_h = trial.copy()
    
    return best_h, best_f, pop, fit

Let me also think about whether the `np.correlate` is computing what we want. The problem says:
C₅ = max_k ∫ h(x)(1-h(x+k)) dx

In the discretization, this is:
max_k sum_i h[i] * (1-h[i+k]) * dx (for valid i)

And `np.correlate(h, 1-h, mode='full')` computes:
c[m] = sum_i h[i] * (1-h[i+m-(n-1)])

Wait, let me be more precise. The numpy documentation says:
`np.correlate(a, v)[k] = sum_n a[n+k] * conj(v[n])`

So with a=h and v=1-h:
`c[k] = sum_n h[n+k] * (1-h[n])`

where the sum is over all n such that both indices are valid.

For k from -(n-1) to (n-1) (in mode='full', indexed 0 to 2n-2):
- The actual lag is k_actual = k - (n-1)
- c[k] = sum over valid n of h[n + k_actual] * (1-h[n])

This matches: for a given shift k_actual, we compute sum h[n+k_actual] * (1-h[n]), which is exactly ∫ h(x+k)(1-h(x)) dx discretized.

So yes, `np.max(np.correlate(h, 1-h, mode='full')) * dx` gives C₅. ✓

            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new
    if time.time() < p2_end:
        h_new = slsqp_optimize(best_h, n, dx, max_iter=500, smooth_m=3)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 3: Try n2 = 2n
    n2 = 2 * n
    dx2 = 2.0 / n2
    target2 = n2 / 2.0
    h_up = project_box_sum(np.repeat(best_h, 2), target2)
    p3_end = t_start + budget_s - 25
    if time.time() < p3_end - 40:
        p3_de = min(130.0, p3_end - time.time() - 50)
        if p3_de > 20:
            bh2, bf2, pop2, fit2 = de_evolve(h_up, n2, dx2, 35, p3_de, rng)
            if bf2 < best_c5:
                best_h, best_c5 = bh2, bf2
            if time.time() < p3_end - 15:
                h_new2 = slsqp_optimize(best_h, n2, dx2, max_iter=300, smooth_m=3)
                c5_new2 = compute_c5(h_new2, dx2)
                if c5_new2 < best_c5:
                    best_h, best_c5 = h_new2, c5_new2

    # Phase 4: Final polish with m=1 (exact max gradient)
    nf = len(best_h)
    dxf = 2.0 / nf
    if time.time() < p3_end:
        h_new = slsqp_optimize(best_h, nf, dxf, max_iter=600, smooth_m=1)
        c5_new = compute_c5(h_new, dxf)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new
        if time.time() < p3_end - 5:
            h_new2 = slsqp_optimize(best_h, nf, dxf, max_iter=400, smooth_m=2)
            c5_new2 = compute_c5(h_new2, dxf)
            if c5_new2 < best_c5:
                best_h, best_c5 = h_new2, c5_new2

    # Final feasibility
    nf = len(best_h)
    best_h = project_box_sum(best_h, nf / 2.0)
    best_c5 = compute_c5(best_h, 2.0 / nf)
    return (best_h, best_c5, nf)
