import time

def fft_correlate(h):
    """Compute np.correlate(h, 1-h, mode='full') using FFT."""
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N *= 2
    H = np.fft.rfft(h, N)
    B = np.fft.rfft(1.0 - h, N)
    corr = np.fft.irfft(H * np.conj(B))[:2 * n - 1]
    return corr

def compute_c5(h):
    n = len(h)
    corr = fft_correlate(h)
    return np.max(corr) * (2.0 / n)

def smooth_max_grad(h, alpha):
    """Compute the gradient of the smooth max of the correlation."""
    n = len(h)
    corr = fft_correlate(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    exp_vals = np.exp(shifted)
    weights = exp_vals / np.sum(exp_vals)
    
    # Compute gradient
    grad = np.zeros(n)
    active = np.where(weights > 1e-8)[0]
    
    for idx in active:
        w = weights[idx]
        lag = idx - (n - 1)
        # corr at lag: sum_{i} h[i] * (1 - h[i+lag]) where 0<=i<n and 0<=i+lag<n
        # d/dh[j]:
        #   (1 - h[j+lag]) if 0 <= j+lag < n  [from h[j] factor]
        #   -h[j-lag] if 0 <= j-lag < n        [from (1-h[j+lag]) factor, i=j-lag]
        
        if lag >= 0:
            # j ranges 0..n-1-lag for first term
            grad[:n - lag] += w * (1.0 - h[lag:n])
            # j ranges lag..n-1 for second term
            grad[lag:] -= w * h[:n - lag]
        else:
            p = -lag
            # j ranges p..n-1 for first term
            grad[p:] += w * (1.0 - h[:n - p])
            # j ranges 0..n-1-p for second term
            grad[:n - p] -= w * h[p:]
    
    return grad * (2.0 / n)

def project_box_sum(h, target_sum):
    """Project h onto {x: 0<=x<=1, sum(x)=target_sum}."""
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def adam_smooth_max(h0, target_sum, deadline, rng, alpha_init=100, lr_init=0.01):
    """Adam optimizer with smooth max objective."""
    n = len(h0)
    h = project_box_sum(h0.astype(float).copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    v_grad = np.zeros(n)

        grad = smooth_max_grad(h, alpha)
        
        m_grad = beta1 * m_grad + (1 - beta1) * grad
        v_grad = beta2 * v_grad + (1 - beta2) * grad ** 2
        m_hat = m_grad / (1 - beta1 ** t)

        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_box_sum(h_new, target_sum)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        
        h = h_new
        
        # Adaptive alpha: increase when stuck
        if t % 100 == 0:
            alpha = min(alpha * 1.1, 500.0)
        
        # Adaptive lr: decrease if not improving
        if t % 50 == 0:
            lr *= 0.95
    
    return best_h, best_c5

import time

def fft_correlate(h):

    N = 1
    while N < 2 * n - 1:
        N *= 2
    H = np.fft.rfft(h, N)
    B = np.fft.rfft(1.0 - h, N)
    return np.fft.irfft(H * np.conj(B))[:2 * n - 1]

def compute_c5(h):
    n = len(h)
    corr = fft_correlate(h)
    return np.max(corr) * (2.0 / n)

def compute_grad_smooth_max(h, alpha):
    n = len(h)
    corr = fft_correlate(h)
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -50, 50)
    exp_v = np.exp(shifted)
    w = exp_v / np.sum(exp_v)
    
    grad = np.zeros(n)
    active = np.where(w > 1e-10)[0]
    
    for idx in active:
        wi = w[idx]
        lag = idx - (n - 1)
        if lag >= 0:
            grad[:n - lag] += wi * (1.0 - h[lag:n])
            grad[lag:] -= wi * h[:n - lag]
        else:
            p = -lag
            grad[p:] += wi * (1.0 - h[:n - p])
            grad[:n - p] -= wi * h[p:]
    
    return grad * (2.0 / n)

def project_box_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(h - mid, 0, 1)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def optimize(h0, target, deadline, rng, alpha0=100.0):
    n = len(h0)
    h = project_box_sum(h0.astype(float).copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = 0.01
    alpha = alpha0
    b1, b2, eps = 0.85, 0.999, 1e-8
    stall = 0
    
    while time.time() < deadline:
        t += 1
        g = compute_grad_smooth_max(h, alpha)
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g

import numpy as np
import time

def fft_corr(h):
    n = len(h)
    N = 1 << (2 * n - 1).bit_length()
    H = np.fft.rfft(h, N)
    B = np.fft.rfft(1.0 - h, N)
    return np.fft.irfft(H * np.conj(B))[:2 * n - 1]

def c5_val(h):
    n = len(h)
    return float(np.max(fft_corr(h)) * 2.0 / n)

def smooth_grad(h, alpha):
    n = len(h)
    c = fft_corr(h)
    m = c.max()
    e = np.exp(np.clip(alpha * (c - m), -50, 50))
    w = e / e.sum()
    g = np.zeros(n)
    active = np.where(w > 1e-10)[0]
    for i in active:
        wi = w[i]
        lag = i - n + 1
        if lag >= 0:
            g[:n - lag] += wi * (1.0 - h[lag:])
            g[lag:] -= wi * h[:n - lag]
        else:
            p = -lag
            g[p:] += wi * (1.0 - h[:n - p])
            g[:n - p] -= wi * h[p:]
    return g * 2.0 / n

def proj(h, tgt):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > tgt:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def opt(h0, tgt, dl, rng, a0=100.0, lr0=0.03):
    n = len(h0)
    h = proj(h0.astype(float).copy(), tgt)
    bh, bc = h.copy(), c5_val(h)
    mu = np.zeros(n)
    nu = np.zeros(n)
    t = 0
    lr = lr0
    a = a0
    b1, b2, ep = 0.9, 0.999, 1e-8
    stall = 0
    best_stall_lr = lr

    while time.time() < dl:
        t += 1
        g = smooth_grad(h, a)
        mu = b1 * mu + (1 - b1) * g
        nu = b2 * nu + (1 - b2) * g * g
        mhat = mu / (1 - b1 ** t)
        vhat = nu / (1 - b2 ** t)
        update = lr * mhat / (np.sqrt(vhat) + ep)
        h_new = proj(h - update, tgt)
        v_new = c5_val(h_new)

        if v_new < bc - 1e-14:
            bh, bc = h_new.copy(), v_new
            h = h_new
            stall = 0
            if lr < lr0 * 1.5:
                lr = min(lr * 1.02, lr0 * 2.0)
        else:
            stall += 1
            # Backtrack
            accepted = False
            for _ in range(4):
                lr *= 0.5
                h_new = proj(h - lr * mhat / (np.sqrt(vhat) + ep), tgt)
                v_new = c5_val(h_new)
                if v_new < bc:
                    bh, bc = h_new.copy(), v_new
                    h = h_new
                    stall = 0
                    accepted = True
                    break
            if not accepted:
                h = proj(h - lr * mhat / (np.sqrt(vhat) + ep), tgt)
            if stall > 35:
                pert = 0.02 + 0.01 * min(stall // 35, 3.0)
                h = proj(h + rng.randn(n) * pert, tgt)
                a = min(a * 1.4, 500.0)
                lr = lr0
                mu, nu = np.zeros(n), np.zeros(n)
                t = 0
                stall = 0
                continue

        if t % 200 == 0:
            a = min(a * 1.06, 400.0)
            lr = min(lr * 0.97, 0.1)

    return bh, bc

def ierp(h, nn):
    no = len(h)
    return np.interp(np.arange(nn) / nn, np.arange(no) / no, np.append(h, h[:1]))

def make_pattern(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    # Create a periodic pattern with random phase
    period = n // 4 + rng.randint(1, 5)
    duty = 0.3 + 0.3 * rng.random()
    for i in range(n):
        phase = (i % period) / period
        h[i] = 1.0 if phase < duty else 0.0
    h += rng.randn(n) * 0.1
    return proj(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = make_pattern(200, seed)

    n = len(h0)
    tgt = n / 2.0
    h0 = proj(h0, tgt)
    bh = h0.copy()
    bc = c5_val(bh)

    # Phase 1: Optimize at initial resolution with moderate alpha
    dl = t0 + budget_s * 0.25
    bh, bc = opt(bh, tgt, dl, rng, a0=60.0, lr0=0.03)

    # Phase 2: Try a different starting point at same resolution
    rem = budget_s - (time.time() - t0)
    if rem > 60:
        h_alt = make_pattern(n, seed + 1)
        h_alt = ierp(bh, n) * 0.3 + h_alt * 0.7
        h_alt = proj(h_alt, tgt)
        dl2 = time.time() + min(rem * 0.2, 150.0)
        h_alt_opt, c5_alt = opt(h_alt, tgt, dl2, rng, a0=80.0, lr0=0.03)
        if c5_alt < bc:
            bh, bc = h_alt_opt, c5_alt

    # Phase 3: Progressive resolution refinement
    rem = budget_s - (time.time() - t0)
    for nt in [350, 500, 800, 1200]:
        if rem < 30:
            break
        h_interp = ierp(bh, nt)
        h_interp = proj(h_interp, nt / 2.0)
        alloc = min(rem * 0.3, 200.0)
        dl3 = time.time() + alloc
        h_opt, c5_opt = opt(h_interp, nt / 2.0, dl3, rng, a0=50.0, lr0=0.02)
        if c5_opt < bc:
            bh, bc = h_opt, c5_opt
        rem = budget_s - (time.time() - t0)

    # Phase 4: Second pass at best resolution with higher alpha (exploitation)
    rem = budget_s - (time.time() - t0)
    if rem > 20:
        bn = len(bh)
        bh, bc = opt(bh, bn / 2.0, t0 + budget_s - 3, rng, a0=150.0, lr0=0.015)

    # Phase 5: Final attempt at slightly higher resolution if time permits
    rem = budget_s - (time.time() - t0)
    if rem > 25:
        bn = len(bh)
        bn2 = bn + 50
        h_interp = ierp(bh, bn2)
        h_interp = proj(h_interp, bn2 / 2.0)
        bh2, bc2 = opt(h_interp, bn2 / 2.0, t0 + budget_s - 1, rng, a0=100.0, lr0=0.02)
        if bc2 < bc:
            bh, bc = bh2, bc2

    bh = proj(bh, len(bh) / 2.0)
    bc = c5_val(bh)
    return bh, bc, len(bh)
