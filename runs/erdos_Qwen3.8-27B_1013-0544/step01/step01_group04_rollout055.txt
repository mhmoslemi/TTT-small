                best_c5 = c5_new

    # Phase 5: Pair swap local search
    if elapsed() < budget_s * 0.93:
        n_best = len(best_h)
        dx_best = 2.0 / n_best
        h_swapped, c5_swapped = pair_swap_optimize(best_h, n_best, dx_best, 500, rng)
        if c5_swapped < best_c5 - 1e-14:
            best_h = h_swapped
            best_c5 = c5_swapped
        # Polish after swaps
        h_new = do_slsqp(best_h, n_best, dx_best, 500, 400, 1e-13)
        c5_new = compute_c5(h_new, dx_best)
        if c5_new < best_c5 - 1e-14:
            best_h = h_new
            best_c5 = c5_new

    # Phase 6: Final high-alpha polish
    if elapsed() < budget_s - 10:
        n_fin = len(best_h)
        dx_fin = 2.0 / n_fin
        for alpha in [500, 1000, 2000]:
            if elapsed() > budget_s - 10:
                break
            h_new = do_slsqp(best_h, n_fin, dx_fin, alpha, 500, 1e-14)
            c5_new = compute_c5(h_new, dx_fin)
            if c5_new < best_c5 - 1e-14:
                best_h = h_new
                best_c5 = c5_new
        # Also try true (non-smooth) objective
        if elapsed() < budget_s - 5:
            h_new = do_slsqp_true(best_h, n_fin, dx_fin, 300, 1e-14)
            c5_new = compute_c5(h_new, dx_fin)
            if c5_new < best_c5 - 1e-14:
                best_h = h_new
                best_c5 = c5_new

    return (best_h, best_c5, len(best_h))
