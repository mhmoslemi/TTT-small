from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-15:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def obj_topk(h, n, alpha, top_k):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    if top_k is not None and top_k < len(c):
        idx = np.argsort(c)[-top_k:]
        c = c[idx]
    m = c.max()
    shifted = alpha * (c - m)

    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

    return np.sum(x) - n / 2.0

    return np.ones(n)

    return [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac, 'args': (n,)}]
