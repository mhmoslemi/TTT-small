            grad = np.zeros(n)
            for idx in active:
                L = int(idx) - n_half
                grad += compute_grad_lag(h, L)
            grad /= float(len(active))
            h_new = project_feasible(h - step * grad, target)
            c_new = compute_c5(h_new, dx)
            if c_new < compute_c5(h, dx):
                h = h_new
                try_update(h, dx, n)
            else:
                step *= 0.95
                if step < 1e-10:
                    break

    # Phase 6: Try n=60
    if tleft() > 60.0:
        n2 = 60
        dx2 = 2.0 / n2
        target2 = n2 / 2.0
        h2 = np.full(n2, 0.5)
        # Initialize from best by interpolation
        if best_h is not None:
            h2 = np.interp(np.linspace(0, 1, n2), np.linspace(0, 1, n), best_h)
            h2 = project_feasible(h2, target2)
        try_update(h2, dx2, n2)
        # Quick gradient descent
        temp2 = 0.03
        for _ in range(2000):
            if tleft() < 40.0:
                break
            grad2 = softmax_gradient(h2, dx2, temp2)
            h2_new, _ = projected_line_search(h2, grad2, target2, compute_c5(h2, dx2), dx2, max_iter=10)
            try_update(h2_new, dx2, n2)
            h2 = h2_new
            temp2 *= 0.995
            if temp2 < 0.002:
                temp2 = 0.002

    # Final SLSQP on best
    if tleft() > 20.0 and best_h is not None:
        n_best = best_n
        dx_best = 2.0 / n_best
        target_best = n_best / 2.0
        def final_obj(x):
            return compute_c5(x, dx_best)
        def final_eq(x):
            return np.sum(x) - target_best
        try:
            res = minimize(final_obj, best_h, method='SLSQP',
                          bounds=[(0.0, 1.0)] * n_best,
                          constraints=[{'type': 'eq', 'fun': final_eq}],
                          options={'maxiter': 2000, 'ftol': 1e-16})
            if res.x is not None:
                h_ref = project_feasible(res.x, target_best)
                final_c = compute_c5(h_ref, dx_best)
                if final_c < best_c5:
                    best_c5 = final_c
                    best_h = h_ref
        except Exception:
            pass

    return (best_h, best_c5, best_n)
