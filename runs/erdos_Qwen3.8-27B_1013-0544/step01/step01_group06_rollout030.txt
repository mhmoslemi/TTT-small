import numpy as np
import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def compute_corr(h):
    return np.correlate(h, 1 - h, mode='full')

def smooth_max(corr, alpha):
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha

def smooth_max_grad(corr, alpha):
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    w /= np.sum(w)
    return w

def project_simplex(h, target):
    n = len(h)
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.sum(np.maximum(h - mid, 0.0)) > target:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-12:
            break
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def nesterov_step(h, grad, vel, lr, momentum, target_sum):
    vel_new = momentum * vel - lr * grad
    h_new = project_simplex(h + vel_new, target_sum)
    return h_new, vel_new

    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = (lo+hi)*0.5
        if np.sum(np.clip(h-mid, 0, 1)) > target:
            lo = mid
        elif np.sum(np.clip(h-mid, 0, 1)) < target:
            hi = mid

    n = len(h)
    corr = np.correlate(h, 1-h, 'full')
    m = corr.max()
    w = np.exp(alpha*(corr-m))
    s = w.sum()

        w[corr.argmax()] = 1.0

def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1-h, 'full').max() * (2.0/n)

        s = np.sum(np.clip(h-mid, 0, 1))
        if s > target: lo = mid
        else: hi = mid

    m = corr.max()

            grad[p:] -= w[ki]*h[:n-p]

def compute_c5(h):
