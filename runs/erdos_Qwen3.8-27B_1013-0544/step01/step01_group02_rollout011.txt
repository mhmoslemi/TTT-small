def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    """Project h to satisfy 0<=h<=1 and sum(h)=n/2"""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-15:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    if shifted.max() > 500:
        return m + (shifted.max() + np.log(np.sum(np.exp(shifted - shifted.max())))) / alpha
    return m + np.log(np.sum(np.exp(shifted))) / alpha

    return np.sum(h) - n / 2.0

        elif h_parent is not None:

            x_parent = np.linspace(0, 2, len(h_parent), endpoint=False) + 1.0 / len(h_parent)
            x_new = np.linspace(0, 2, n, endpoint=False) + 1.0 / n
            hp = np.interp(x_new, x_parent, h_parent)
            hp = project_feasible(hp, n)

        cands = make_candidates(n, rng, hp)
        # Deduplicate
        seen = set()
        unique_cands = []
        for c in cands:
            key = tuple(c.astype(int))
            if key not in seen:
                seen.add(key)
                unique_cands.append(c)

        best_local_c5 = compute_c5(fix_binary_sum((hp > 0.5).astype(float) if hp is not None else np.zeros(n), n), n) if hp is not None else 1.0
        best_local_h = None

        for h_cand in unique_cands:
            if time.time() - phase2_start > n_time:
                break
            h_bin = fix_binary_sum(h_cand, n)
            c_bin = compute_c5(h_bin, n)
            if c_bin < best_local_c5:
                best_local_c5 = c_bin
                best_local_h = h_bin.copy()

            # Binary local search with block moves
            h_ls, c_ls = binary_2swap(h_bin, n, max_iters=8000, rng=rng, block_size=1)
            if c_ls < best_local_c5:
                best_local_c5 = c_ls
                best_local_h = h_ls.copy()

            # Try block moves on the best so far
            h_bs, c_bs = binary_2swap(best_local_h, n, max_iters=4000, rng=rng, block_size=3)
            if c_bs < best_local_c5:
                best_local_c5 = c_bs
                best_local_h = h_bs.copy()

        # Annealing on best binary
        if best_local_h is not None and remaining() > 60:
            h_ann, c_ann = anneal_binary(best_local_h, n, rng, max_iters=8000)
            if c_ann < best_local_c5:
                best_local_c5 = c_ann
                best_local_h = h_ann.copy()

        # SLSQP refinement on best binary for this n
        if best_local_h is not None:
            h_ref = best_local_h.copy()
            for alpha in [100, 1000, 10000, 100000]:
                if remaining() < 15:
                    break
                h_ref = slsqp(h_ref, n, alpha=alpha, maxiter=1500)
                c_ref = compute_c5(h_ref, n)
                if c_ref < best_c5:
                    best_c5 = c_ref
                    best_h = h_ref.copy()
                    best_n = n
                if c_ref < best_local_c5:
                    best_local_c5 = c_ref

            if best_local_c5 < best_c5:
                best_c5 = best_local_c5
                best_h = best_local_h.copy()
                best_n = n

    # Phase 3: Iterated perturbation + SLSQP on best
    while remaining() > 50:
        scale = rng.choice([0.005, 0.01, 0.02, 0.05, 0.1, 0.2])
        h_pert = best_h + rng.normal(0, scale, size=best_n)
        h_pert = project_feasible(h_pert, best_n)
        alpha = rng.choice([500, 2000, 10000, 50000])
        h_try = slsqp(h_pert, best_n, alpha=int(alpha), maxiter=1000)
        c = compute_c5(h_try, best_n)
        if c < best_c5 - 1e-14:
            best_h, best_c5 = h_try, c

    # Phase 4: Final high-alpha SLSQP polish
    if remaining() > 10:
        for alpha in [50000, 200000]:
            if remaining() < 5:
                break
            h_try = slsqp(best_h, best_n, alpha=alpha, maxiter=5000, ftol=1e-18)
            c = compute_c5(h_try, best_n)
            if c < best_c5:
                best_h, best_c5 = h_try, c

    # Final feasibility check
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)

    return best_h, best_c5, best_n
