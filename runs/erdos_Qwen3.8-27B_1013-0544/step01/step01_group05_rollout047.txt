    exps = np.exp(scaled - max_s)
    total = float(np.sum(exps))
    c5_s = (max_s + np.log(total)) / beta
    w = exps / total
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-12:
            continue
        L = k - half
        if L == 0:
            grad += wk * (omh - h)
        elif L > 0:
            grad[L:] += wk * omh[:n - L]
            grad[:n - L] -= wk * h[L:]
        else:
            L2 = -L
            grad[:n - L2] += wk * omh[L2:]
            grad[L2:] -= wk * h[:n - L2]
    return c5_s, grad

def active_set_gradient(h, dx, n_active=15, tol_frac=0.02):
    """Compute gradient using only active (near-max) lags."""
    n = len(h)
    half = n - 1
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full')
    c_max = float(np.max(corr))
    tol = c_max * tol_frac
    active = np.where(corr >= c_max - tol)[0]
    if len(active) < 3:
        active = np.argsort(corr)[-3:]
    elif len(active) > n_active:
        active = np.argsort(corr)[-n_active:]
    
    grad = np.zeros(n)
    for idx in active:
        L = int(idx) - half
        if L == 0:
            grad += omh - h
        elif L > 0:
            grad[L:] += omh[:n - L]
            grad[:n - L] -= h[L:]
        else:
            L2 = -L
            grad[:n - L2] += omh[L2:]
            grad[L2:] -= h[:n - L2]
    grad /= len(active)
    return grad * dx

def run_projected_gradient(h, dx, target_sum, step0, max_iter, time_limit, t_start,
                          method='lse', beta0=5.0, beta_end=50.0, n_active=15):
    """Run projected gradient with LSE or active-set gradient."""
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    mom = np.zeros(n)
    step = step0
    mu = 0.7
    
    for it in range(max_iter):
        if time.time() - t_start > time_limit:
            break
        
        if method == 'lse':
            beta = beta0 + (beta_end - beta0) * (it / max_iter)
            _, grad = smooth_c5_grad(h, dx, beta)
        else:
            grad = active_set_gradient(h, dx, n_active=n_active)
        
        mom = mu * mom - step * grad
        h_new = project_feasible(h + mom, target_sum)
        c5_new = compute_c5(h_new, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            step = min(step * 1.02, step0 * 2)
        else:
            step *= 0.9
            if step < 1e-10:
                break
        
        h = h_new
    
    return best_h, best_c5

def run_nelder_mead(h, dx, target_sum, time_limit, t_start):

    n = len(h)
    
    def obj(x):

        return compute_c5(x, dx)

    result = project_feasible(result, target_sum)
    return result, compute_c5(result, dx)
