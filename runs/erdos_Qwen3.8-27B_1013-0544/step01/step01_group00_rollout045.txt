    grad_lse = np.zeros(n)
    # For each j, the gradient contribution is:
    # dx * (sum_k weights[k] - sum_{k: valid} weights[k]*h[j-lag_k] - sum_{k: valid} weights[k]*h[j+lag_k])
    # where lag_k = k - mid
    # h[j-lag_k] valid when 0 <= j-lag_k < n, i.e., j-n+1 <= lag_k <= j
    # h[j+lag_k] valid when 0 <= j+lag_k < n, i.e., -j <= lag_k <= n-1-j
    # Also need the "j in range_k" condition which for this problem means
    # the pair (j, j-lag_k) or (j, j+lag_k) must be in the overlap region
    # For lag_k >= 0: overlap is i in [lag_k, n-1], so j in [lag_k, n-1] means lag_k <= j
    #   and j-lag_k in [0, n-1-lag_k] means j <= n-1+lag_k (always true since j <= n-1)
    #   So for lag_k >= 0: valid if lag_k <= j, and h[j-lag_k] is always in [0, n-1]
    # For lag_k < 0: let l = -lag_k > 0, overlap is i in [0, n-1-l], so j in [0, n-1-l] means n-1-j >= l
    #   and j-lag_k = j+l in [0, n-1] needs j+l <= n-1, i.e., j <= n-1-l (same condition)
    #   So for lag_k < 0 (l > 0): valid if j <= n-1-l, i.e., j+l <= n-1
    
    total_w = np.sum(weights)
    for j in range(n):
        # First sum: weights[k] * h[j-lag_k] for valid k
        s1 = 0.0
        # lag_k ranges from -(n-1) to n-1, k = mid + lag_k ranges from 0 to 2n-2
        # h[j-lag_k] valid when 0 <= j-lag_k < n
        # For lag_k >= 0 (k >= mid): need lag_k <= j (from range condition) and j-lag_k >= 0 (same)
        #   lag_k from 0 to j: k from mid to mid+j
        for lag in range(0, min(j, n - 1) + 1):
            s1 += weights[mid + lag] * h[j - lag]
        # For lag_k < 0 (k < mid): let l = -lag_k, need j+l <= n-1
        #   l from 1 to min(n-1-j, n-1): k from mid-l to mid-1
        for l in range(1, min(n - j, n) ):
            s1 += weights[mid - l] * h[j + l]
        
        s2 = 0.0
        # h[j+lag_k] valid when 0 <= j+lag_k < n
        # For lag_k >= 0: need j+lag_k <= n-1, i.e., lag_k <= n-1-j, and lag_k <= j (range)
        #   lag_k from 0 to min(j, n-1-j): k from mid to mid+min(j,n-1-j)
        for lag in range(0, min(j, n - 1 - j) + 1):
            s2 += weights[mid + lag] * h[j + lag]
        # For lag_k < 0 (l > 0): need j-l >= 0, i.e., l <= j, and j <= n-1-l (range)
        #   l from 1 to min(j, n-1-j): k from mid-l to mid-1
        for l in range(1, min(j, n - 1 - j) + 1):
            s2 += weights[mid - l] * h[j - l]
        
        grad_lse[j] = dx * (total_w - s1 - s2)
    
    return grad_lse + grad_penalty


def t_objective(v):
    return v[-1]


def t_ineq(v):
    n = len(v) - 1
    h = v[:n]
    t = v[-1]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return t - corr


def t_eq_fun(v):
    n = len(v) - 1
    return np.sum(v[:n]) - n * 0.5


def t_eq_jac(v):
    n = len(v) - 1
    j = np.zeros(len(v))
    j[:n] = 1.0
    return j


def optimize_t_var(h0, max_iter=200, ftol=1e-10):
    n = len(h0)
    c5_cur = compute_c5(h0)
    v0 = np.append(h0.copy(), c5_cur + 0.01)
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.5)]
    cons = [
        {'type': 'ineq', 'fun': t_ineq},
        {'type': 'eq', 'fun': t_eq_fun, 'jac': t_eq_jac}
    ]
    res = minimize(t_objective, v0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': ftol})
    h_new = np.clip(res.x[:n], 0.0, 1.0)
    h_new = project_h(h_new, n * 0.5)
    return h_new, compute_c5(h_new)


def optimize_lbfgs(h0, temp, penalty, max_iter=200, tol=1e-10):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    res = minimize(lse_penalty_obj, h0.copy(),
                   jac=lambda h: lse_penalty_grad(h, temp, penalty, n),
                   args=(temp, penalty, n), method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': max_iter, 'gtol': tol, 'ftol': tol})
    h_new = np.clip(res.x, 0.0, 1.0)
    h_new = project_h(h_new, n * 0.5)
    return h_new, compute_c5(h_new)


def interp_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_h(h_new, n_new * 0.5)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    def time_left():
        return budget_s - elapsed()

    # Initialize from parent
    h = np.array(initial_h_values, dtype=float)
    n = len(h)
    h = project_h(h, n * 0.5)
    c5 = compute_c5(h)
    h_best = h.copy()
    c5_best = c5

    # Phase 1: L-BFGS-B with decreasing temperature and increasing penalty
    temps = [0.05, 0.03, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    for temp in temps:
        if time_left() < 120:
            break
        for penalty in [10.0, 50.0, 200.0]:
            if time_left() < 60:
                break
            h_new, c5_new = optimize_lbfgs(h, temp, penalty, max_iter=100, tol=1e-8)
            if c5_new < c5:
                h, c5 = h_new, c5_new
                if c5 < c5_best:
                    h_best, c5_best = h.copy(), c5

    # Phase 2: SLSQP t-variable refinement
    if time_left() > 180:
        h_new, c5_new = optimize_t_var(h, max_iter=300, ftol=1e-10)
        if c5_new < c5:
            h, c5 = h_new, c5_new
            if c5 < c5_best:
                h_best, c5_best = h.copy(), c5

    # Phase 3: More L-BFGS-B refinement at lower temperatures
    for temp in [0.001, 0.0005, 0.0003, 0.0001]:
        if time_left() < 100:
            break
        for penalty in [50.0, 200.0, 500.0]:
            if time_left() < 50:
                break
            h_new, c5_new = optimize_lbfgs(h, temp, penalty, max_iter=150, tol=1e-10)
            if c5_new < c5:
                h, c5 = h_new, c5_new
                if c5 < c5_best:
                    h_best, c5_best = h.copy(), c5

    # Phase 4: Random perturbations + re-optimize
    n_pert = 8
    for i in range(n_pert):
        if time_left() < 80:
            break
        scale = 0.01 + 0.005 * i
        h_p = h_best + rng.randn(len(h_best)) * scale
        h_p = np.clip(h_p, 0.0, 1.0)
        h_p = project_h(h_p, len(h_best) * 0.5)
        h_new, c5_new = optimize_lbfgs(h_p, 0.005, 100.0, max_iter=100, tol=1e-8)
        if c5_new < c5_best:
            h_best, c5_best = h_new.copy(), c5_new
            h, c5 = h_new, c5_new

    # Phase 5: Try different resolutions
    for n_new in [120, 240, 360]:
        if time_left() < 200:
            break
        h_interp = interp_h(h_best, n_new)
        c5_cur = compute_c5(h_interp)
        h_cur = h_interp.copy()
        # L-BFGS-B at new resolution
        for temp in [0.02, 0.01, 0.005]:
            if time_left() < 80:
                break
            h_new, c5_new = optimize_lbfgs(h_cur, temp, 100.0, max_iter=100, tol=1e-8)
            if c5_new < c5_cur:
                h_cur, c5_cur = h_new, c5_new
        if c5_cur < c5_best:
            h_best, c5_best = h_cur.copy(), c5_cur
            h, c5 = h_cur, c5_cur

    # Phase 6: Final SLSQP t-variable polish on best
    if time_left() > 100:
        h_final = project_h(h_best.copy(), len(h_best) * 0.5)
        h_new, c5_new = optimize_t_var(h_final, max_iter=200, ftol=1e-11)
        if c5_new < c5_best:
            h_best, c5_best = h_new.copy(), c5_new

    # Final cleanup
    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = project_h(h_best, len(h_best) * 0.5)
    c5_final = compute_c5(h_best)
    if c5_final < c5_best:
        h_best, c5_best = h_best, c5_final

    return h_best, c5_best, len(h_best)
