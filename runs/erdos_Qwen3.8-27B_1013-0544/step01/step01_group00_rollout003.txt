from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def corr_values(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_h(h, target):
    h = h.copy()
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return np.clip(h, 0.0, 1.0)
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = np.sum(mask)
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def sum_eq_fun(h):
    return np.sum(h) - len(h) * 0.5

def sum_eq_jac(h):
    return np.ones(len(h))

def t_ineq(v):
    n = len(v) - 1
    h = v[:n]
    t = v[-1]
    corr = corr_values(h)
    return t - corr

def t_eq_fun(v):
    n = len(v) - 1
    return np.sum(v[:n]) - n * 0.5

def t_eq_jac(v):
    n = len(v) - 1
    j = np.zeros(len(v))
    j[:n] = 1.0
    return j

def t_objective(v):
    return v[-1]

def optimize_t_var(h0, max_iter=400, ftol=1e-10):
    n = len(h0)
    c5_cur = compute_c5(h0)
    v0 = np.append(h0.copy(), c5_cur + 0.02)
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.5)]
    cons = [
        {'type': 'ineq', 'fun': t_ineq},
        {'type': 'eq', 'fun': t_eq_fun, 'jac': t_eq_jac}
    ]
    res = minimize(t_objective, v0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': ftol})
    h_new = np.clip(res.x[:n], 0.0, 1.0)
    h_new = project_h(h_new, n * 0.5)

def simulated_annealing(h0, t_start, t_end, n_steps, rng):
    n = len(h0)
    h = h0.copy()
    c5_cur = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_cur
    
    for step in range(n_steps):
        T = t_start * (t_end / t_start) ** (step / n_steps)
        # Generate a move
        if rng.random() < 0.7:
            # Push-pull
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            delta = rng.uniform(0.05, 0.3) * rng.choice([-1, 1])
            h[i] += delta
            h[j] -= delta
        else:
            # Block perturbation
            start = rng.randint(n - 5)
            length = rng.randint(3, 8)
            delta = rng.normal(0, 0.1, length)
            h[start:start+length] += delta
            # Compensate elsewhere
            comp = -delta.sum() / (n - length)
            mask = np.ones(n, dtype=bool)
            mask[start:start+length] = False
            h[mask] += comp
        
        h = project_h(h, n * 0.5)
        c5_new = compute_c5(h)
        delta_c5 = c5_new - c5_cur
        
        if delta_c5 < 0 or rng.random() < np.exp(-delta_c5 / max(T, 1e-8)):
            c5_cur = c5_new
            if c5_cur < best_c5:
                best_c5 = c5_cur

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def corr_all(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode='full') * dx


def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = mask.sum()
    if cnt > 0:
        h = h.copy()
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)


def smooth_obj(h, temp):
    c = corr_all(h)
    m = np.max(c)
    return m + temp * np.log(np.sum(np.exp((c - m) / temp)))


def sum_eq(h):
    return np.sum(h) - len(h) * 0.5


def sum_jac(h):
    return np.ones(len(h))


def t_obj(v):
    return v[-1]


def t_ineq(v):
    n = len(v) - 1
    return v[-1] - corr_all(v[:n])


def t_eq(v):
    n = len(v) - 1
    return np.sum(v[:n]) - n * 0.5


def t_eq_jac(v):
    n = len(v) - 1
    j = np.zeros(len(v))
    j[:n] = 1.0
    return j


def slsqp_tvar(h0, max_iter=400, ftol=1e-10):
    n = len(h0)
    c5_cur = compute_c5(h0)
    v0 = np.append(h0.copy(), c5_cur + 0.03)
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.5)]
    cons = [{'type': 'ineq', 'fun': t_ineq},
            {'type': 'eq', 'fun': t_eq, 'jac': t_eq_jac}]
    res = minimize(t_obj, v0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': ftol})
    h_new = project_h(res.x[:n], n * 0.5)
    return h_new, compute_c5(h_new)


def slsqp_smooth(h0, temp, max_iter=300, ftol=1e-11):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_jac}]
    res = minimize(smooth_obj, h0.copy(), args=(temp,), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': ftol})
    h_new = project_h(res.x, n * 0.5)
    return h_new, compute_c5(h_new)


def simulated_annealing(h0, t_start, t_end, n_steps, rng):
    n = len(h0)
    h = h0.copy()
    c5_cur = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_cur
    for step in range(n_steps):
        T = t_start * (t_end / t_start) ** (step / max(n_steps - 1, 1))
        h_old = h.copy()
        r = rng.random()
        if r < 0.6:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = rng.uniform(0.03, 0.25) * rng.choice([-1.0, 1.0])
            h[i] += d
            h[j] -= d
        elif r < 0.85:
            start = rng.randint(max(n - 10, 1))
            length = rng.randint(2, min(10, n - start))
            d = rng.normal(0, 0.08, length)
            h[start:start + length] += d
            comp_mask = np.ones(n, dtype=bool)
            comp_mask[start:start + length] = False
            h[comp_mask] += -d.sum() / comp_mask.sum()
        else:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = rng.uniform(0.02, 0.15)
            h[i] += d
            h[j] -= d
            k = rng.randint(n)
            if k != i and k != j:
                h[k] += rng.uniform(-0.05, 0.05)
                h[(i + j + k) % n] -= rng.uniform(-0.05, 0.05)
        h = project_h(h, n * 0.5)
        c5_new = compute_c5(h)
        dc = c5_new - c5_cur
        if dc < 0 or rng.random() < np.exp(-dc / max(T, 1e-10)):
            c5_cur = c5_new
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h.copy()
        else:
            h = h_old
    return best_h, best_c5


def block_refine(h0, block_size, n_passes, rng):
    n = len(h0)
    h = h0.copy()
    c5 = compute_c5(h)
    for p in range(n_passes):
        order = rng.permutation((n + block_size - 1) // block_size)
        for b in order:
            start = b * block_size
            end = min(start + block_size, n)
            idx = slice(start, end)
            v0 = h[idx].copy()
            blen = end - start
            bnds = [(0.0, 1.0)] * blen
            res = minimize(
                lambda v: smooth_obj(np.concatenate([h[:start], v, h[end:]]), 0.005),
                v0, method='Nelder-Mead',
                bounds=bnds, options={'maxiter': 80, 'fatol': 1e-9}
            )
            h_new = h.copy()
            h_new[idx] = np.clip(res.x, 0.0, 1.0)
            h_new = project_h(h_new, n * 0.5)
            c5_new = compute_c5(h_new)
            if c5_new < c5:
                h = h_new
                c5 = c5_new
    return h, c5

    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - t0)

    h = np.array(initial_h_values, dtype=float)

            c5_best = c5_cand

    # Phase 1: Initial SLSQP polish
