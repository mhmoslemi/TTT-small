import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - target
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(free):
            h[free] -= diff / np.sum(free)
        else:
            h -= diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = np.sum(h) - target
    if abs(diff) > 1e-13:
        if diff > 0:
            order = np.argsort(-h)
            rem = diff
            for i in order:
                adj = min(h[i], rem)
                h[i] -= adj
                rem -= adj
                if rem < 1e-15:
                    break
        else:
            order = np.argsort(h)
            rem = -diff
            for i in order:
                adj = min(1.0 - h[i], rem)
                h[i] += adj
                rem -= adj
                if rem < 1e-15:
                    break
    return h

def smooth_obj_grad(h, dx, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    a = alpha * corr * dx
    amax = np.max(a)
    ea = np.exp(a - amax)
    se = np.sum(ea)
    w = ea / se
    f = (amax + np.log(se)) / alpha
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-16:
            continue
        L = k - (n - 1)
        M = abs(L)
        if M == 0:
            g += wk * (1.0 - 2.0 * h)
        elif L > 0:
            g[:n-M] += wk * (1.0 - h[M:])
            g[M:] -= wk * h[:n-M]
        else:
            g[M:] += wk * (1.0 - h[:n-M])
            g[:n-M] -= wk * h[M:]
    return f, g

def lfgb_opt(h0, n, dx, alpha, tgt, pen, nit):
    def fo(x):
        v, _ = smooth_obj_grad(x, dx, alpha)
        return v + pen * (np.sum(x) - tgt) ** 2
    def go(x):
        _, g = smooth_obj_grad(x, dx, alpha)
        return g + 2.0 * pen * (np.sum(x) - tgt)
    bnd = [(0.0, 1.0)] * n
    r = minimize(fo, h0, method='L-BFGS-B', jac=go, bounds=bnd,
                 options={'maxiter': nit, 'ftol': 1e-15, 'gtol': 1e-10})
    return r.x

def slsqp_polish(h0, n, dx, tgt, nit):
    bnd = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - tgt,
             'jac': lambda x: np.ones(n)}]
    r = minimize(lambda x: compute_c5(x, dx), h0,
                 method='SLSQP', bounds=bnd, constraints=cons,
                 options={'maxiter': nit, 'ftol': 1e-14})
    return r.x

    t = np.zeros(n)
    for i in range(n):
        t[i] = bin(i).count('1') % 2
    return t

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - target
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(free):
            h[free] -= diff / np.sum(free)
        else:
            h -= diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = np.sum(h) - target
    if abs(diff) > 1e-13:
        if diff > 0:
            order = np.argsort(-h)
            rem = diff
            for i in order:
                adj = min(h[i], rem)
                h[i] -= adj
                rem -= adj
                if rem < 1e-15:
                    break
        else:
            order = np.argsort(h)
            rem = -diff
            for i in order:
                adj = min(1.0 - h[i], rem)
                h[i] += adj
                rem -= adj
                if rem < 1e-15:
                    break
    return h

def smooth_obj_grad(h, dx, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    a = alpha * corr * dx
    amax = np.max(a)
    ea = np.exp(a - amax)
    se = np.sum(ea)
    w = ea / se
    f = (amax + np.log(se)) / alpha
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if wk < 1e-16:
            continue
        L = k - (n - 1)
        M = abs(L)
        if M == 0:
            g += wk * (1.0 - 2.0 * h)
        elif L > 0:
            g[:n - M] += wk * (1.0 - h[M:])
            g[M:] -= wk * h[:n - M]
        else:
            g[M:] += wk * (1.0 - h[:n - M])
            g[:n - M] -= wk * h[M:]
    return f, g

def lfgb_opt(h0, n, dx, alpha, tgt, pen, nit):

        v, _ = smooth_obj_grad(x, dx, alpha)
        return v + pen * (np.sum(x) - tgt) ** 2
    def go(x):
        _, g = smooth_obj_grad(x, dx, alpha)
        return g + 2.0 * pen * (np.sum(x) - tgt)
    bnd = [(0.0, 1.0)] * n
    r = minimize(fo, h0, method='L-BFGS-B', jac=go, bounds=bnd,
                 options={'maxiter': nit, 'ftol': 1e-15, 'gtol': 1e-10})
    return r.x

def slsqp_polish(h0, n, dx, tgt, nit):
    bnd = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - tgt,
             'jac': lambda x: np.ones(n)}]
    r = minimize(lambda x: compute_c5(x, dx), h0,
                 method='SLSQP', bounds=bnd, constraints=cons,
                 options={'maxiter': nit, 'ftol': 1e-14})
    return r.x

def gen_thue_morse(n):
    t = np.zeros(n)
    for i in range(n):
        t[i] = bin(i).count('1') % 2
    return t

def adaptive_lfgb(h0, n, dx, alpha, tgt, max_pen=10000.0):
