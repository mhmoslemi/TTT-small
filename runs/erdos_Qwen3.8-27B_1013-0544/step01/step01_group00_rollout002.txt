import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-10
    else:
        mask = h > 1e-10
    cnt = np.sum(mask)
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def lse_c5(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    return m + temp * np.log(np.sum(np.exp((corr - m) / temp)))

def lse_obj(h, temp, mu):
    n = len(h)
    return lse_c5(h, temp) + mu * (np.sum(h) - n * 0.5) ** 2

def optimize_lbfgs(h0, temp, mu=200.0, max_iter=3000):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    res = minimize(lse_obj, h0.copy(), args=(temp, mu), method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-10})
    h_new = project_h(res.x, n * 0.5)
    return h_new, compute_c5(h_new)

def coord_descent(h, temp, n_sweeps=5, mu=200.0):
    n = len(h)
    h = project_h(h, n * 0.5)
    for _ in range(n_sweeps):
        for i in range(n):
            h_work = h.copy()
            def sub_obj(val):
                h_work[i] = val
                return lse_obj(h_work, temp, mu)
            res = minimize_scalar(sub_obj, bounds=(0.0, 1.0), method='bounded',
                                 options={'xatol': 1e-7})
            h[i] = res.x
        h = project_h(h, n * 0.5)
    return h, compute_c5(h)

def interp_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old))
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_h(h_new, n_new * 0.5)
