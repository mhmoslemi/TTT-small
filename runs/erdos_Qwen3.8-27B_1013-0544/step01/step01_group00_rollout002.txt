        c_max = np.max(corr)

        active = np.where(corr > c_max - 0.01)[0]

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))


def project_feasible(h):
    n = len(h)
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0.0, 1.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new)


def slsqp_tvar(h0, max_iter=500, ftol=1e-12):
    n = len(h0)
    c5_cur = compute_c5(h0)
    v0 = np.append(h0.copy(), c5_cur + 0.005)
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

    def t_obj(v):
        return v[-1]

    def t_ineq(v):
        hh = v[:n]
        t = v[-1]
        corr = np.correlate(hh, 1.0 - hh, mode='full') * (2.0 / n)
        return t - corr

    def t_eq_fun(v):
        return np.sum(v[:n]) - n * 0.5

    def t_eq_jac(v):
        j = np.zeros(len(v))
        j[:n] = 1.0
        return j

    cons = [
        {'type': 'ineq', 'fun': t_ineq},
        {'type': 'eq', 'fun': t_eq_fun, 'jac': t_eq_jac}
    ]
    res = minimize(t_obj, v0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': ftol})
    h_new = np.clip(res.x[:n], 0.0, 1.0)
    h_new = project_feasible(h_new)
    return h_new, compute_c5(h_new)


def power_sum_obj(v, p, mu):
    n = len(v[:p + 1])
    hh = v[:n]
    dx = 2.0 / n
    corr = np.correlate(hh, 1.0 - hh, mode='full') * dx
    pos_corr = np.maximum(corr, 0.0)
    obj = np.sum(pos_corr ** p) ** (1.0 / p)
    obj += mu * (np.sum(hh) - n * 0.5) ** 2
    return obj


def power_sum_optimize(h0, p=40, mu=500.0, max_iter=200):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n

    def ps_obj(v):
        return power_sum_obj(v, p, mu)

    res = minimize(ps_obj, h0.copy(), method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-8})
    h_new = np.clip(res.x, 0.0, 1.0)
    h_new = project_feasible(h_new)
    return h_new, compute_c5(h_new)


def smooth_logsumexp_optimize(h0, temp, max_iter=300):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n

    def lse_obj(v):
        dx = 2.0 / n
        corr = np.correlate(v, 1.0 - v, mode='full') * dx
        m = np.max(corr)
        return m + temp * np.log(np.sum(np.exp((corr - m) / temp))) + \
               100.0 * (np.sum(v) - n * 0.5) ** 2

    def lse_eq(v):
        return np.sum(v) - n * 0.5

    def lse_eq_jac(v):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': lse_eq, 'jac': lse_eq_jac}]
    res = minimize(lse_obj, h0.copy(), method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-12})
    h_new = np.clip(res.x, 0.0, 1.0)
    h_new = project_feasible(h_new)
    return h_new, compute_c5(h_new)


def coord_polish(h, n_steps=200, alpha=0.01):
    n = len(h)
    h = h.copy()
    rng = np.random.RandomState(12345)
    for _ in range(n_steps):
        c5_cur = compute_c5(h)
        idx = rng.randint(n)
        old_val = h[idx]
        for delta in [alpha, -alpha, alpha * 2, -alpha * 2, alpha * 0.5]:
            h[idx] = np.clip(old_val + delta, 0.0, 1.0)
            h = project_feasible(h)
            c5_new = compute_c5(h)
            if c5_new < c5_cur:
                c5_cur = c5_new
                break
            else:
                h[idx] = old_val
        h = project_feasible(h)
    return h, compute_c5(h)
