import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project(h, target):
    """Project onto {sum=h_target, 0<=h<=1} via bisection."""
    h = np.asarray(h, dtype=float)
    n = len(h)
    # Bisection on L2 projection
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        hp = np.clip(h + mid, 0.0, 1.0)
        if hp.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2, 0.0, 1.0)

def smooth_max_and_grad(corr, tau):
    """Compute smooth max value and gradient w.r.t. corr."""
    x = tau * corr
    x_max = x.max()
    exp_x = np.exp(x - x_max)
    softmax = exp_x / exp_x.sum()
    val = (x_max + np.log(exp_x.sum())) / tau
    grad = softmax / tau
    return val, grad

def compute_h_gradient(h, tau):
    """Compute gradient of smooth_max(correlate(h,1-h,'full')) w.r.t. h."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    val, g = smooth_max_and_grad(corr, tau)  # g = gradient w.r.t. corr
    
    # grad_h[j] = sum_m g[m] * [(1-h[j+k_m])*I(0<=j+k_m<n) - h[j-k_m]*I(0<=j-k_m<n)]
    # where k_m = m - (n-1)
    
    grad_h = np.zeros(n)
    for j in range(n):
        # Term A: sum of g[m] for m where 0 <= j + k_m < n, i.e., n-1-j <= m <= 2n-2-j
        m_lo_A = max(0, n - 1 - j)
        m_hi_A = min(2 * n - 2, 2 * n - 2 - j)
        
        # Term B: sum of g[m] * h[j+k_m] for same range
        # j+k_m = j + m - (n-1), so index in h is j+m-n+1
        idx_B = (j + np.arange(m_lo_A, m_hi_A + 1)) - (n - 1)
        
        # Term C: sum of g[m] * h[j-k_m] where 0 <= j-k_m < n, i.e., j-n+1 <= m <= j+n-1
        m_lo_C = max(0, j - n + 1)
        m_hi_C = min(2 * n - 2, j + n - 1)
        # j-k_m = j - m + (n-1), so index in h is j-m+n-1
        idx_C = j - np.arange(m_lo_C, m_hi_C + 1) + (n - 1)
        
        grad_h[j] = g[m_lo_A:m_hi_A + 1].sum() - (g[m_lo_A:m_hi_A + 1] * h[idx_B]).sum() - (g[m_lo_C:m_hi_C + 1] * h[idx_C]).sum()
    
    return val, grad_h

def pgd_optimize(h_init, tau, target, max_iter=2000, lr_0=0.01):
    """Projected gradient descent."""
    n = len(h_init)
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    val, grad = compute_h_gradient(h, tau)
    c5 = val * (2.0 / n)  # Note: smooth max approximates max(corr), C5 = max(corr)*dx
    
    for it in range(max_iter):
        # Adaptive step size
        lr = lr_0 / (1.0 + it * 0.01)
        
        # Gradient step
        h_new = h - lr * grad
        
        # Project
        h_new = project(h_new, target)
        
        # Evaluate
        val_new, grad_new = compute_h_gradient(h_new, tau)
        c5_new = evaluate_c5(h_new)
        
        if val_new > val + 1e-10:
            # Decrease step size
            lr_0 *= 0.5
            if lr_0 < 1e-8:
                break
            continue
        
        h = h_new
        val = val_new
        grad = grad_new
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
    
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    def rem():
        return budget_s - (time.time() - t0)
    
    # Get initial solution
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(100, 0.5)
    
    best_c5 = evaluate_c5(best_h)
    n = len(best_h)
    target = n / 2.0
    
    # Phase 1: PGD at current resolution with annealing
    for tau in [10, 30, 100, 300, 1000, 3000]:
        if rem() < 30:
            break
        h_new, c5_new = pgd_optimize(best_h, tau, target, max_iter=1000, lr_0=0.005)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
    
    # Phase 2: Try structured seeds at same resolution
    seeds = []
    t = np.linspace(0, 2, n)
    seeds.append(0.5 + 0.3 * np.sin(np.pi * t))  # sine
    seeds.append(0.5 + 0.3 * np.cos(np.pi * t))  # cosine
    seeds.append(0.5 + 0.2 * np.sin(2 * np.pi * t))  # double sine
    seeds.append(0.5 + 0.3 * np.sin(0.5 * np.pi * t) * np.cos(0.5 * np.pi * t))
    
    for s in seeds:
        if rem() < 20:
            break
        s = project(s, target)
        h_new, c5_new = pgd_optimize(s, 100, target, max_iter=800, lr_0=0.005)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
    
    # Phase 3: Resolution refinement
    for tn in [200, 300]:
        if rem() < 60:
            break
        if tn % 2 != 0:
            tn -= 1
        target_n = tn / 2.0
        h_fine = np.interp(np.linspace(0, 1, tn), np.linspace(0, 1, n), best_h)
        h_fine = project(h_fine, target_n)
        
        for tau in [30, 100, 300, 1000]:
            if rem() < 30:
                break
            h_new, c5_new = pgd_optimize(h_fine, tau, target_n, max_iter=800, lr_0=0.003)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
                n = tn
                target = target_n
            h_fine = h_new
    
    # Phase 4: Random perturbations
    for _ in range(20):
        if rem() < 15:
            break
        scale = 0.05 + 0.1 * np.random.random()
        hp = best_h + np.random.normal(0, scale, n)
        hp = project(hp, target)
        for tau in [100, 500, 2000]:
            if rem() < 10:
                break
            h_new, c5_new = pgd_optimize(hp, tau, target, max_iter=600, lr_0=0.003)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
            hp = h_new
    
    # Phase 5: Final fine-tuning with SLSQP
    if rem() > 20:
        # Use SLSQP for final polish
        x0 = best_h.copy()
        constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
                        'jac': lambda x: np.ones(n)}]
        bounds = list(zip(np.zeros(n), np.ones(n)))
        
        def objective(x):
            corr = np.correlate(x, 1.0 - x, mode="full")
            x_tau = 5000 * corr
            x_max = x_tau.max()
            return (x_max + np.log(np.sum(np.exp(x_tau - x_max)))) / 5000
        
        try:
            from scipy.optimize import minimize as sp_min
            res = sp_min(objective, x0, method='SLSQP', bounds=bounds,
                        constraints=constraints, options={'maxiter': 300, 'ftol': 1e-12})
            res_x = project(res.x, target)
            c5_res = evaluate_c5(res_x)
            if c5_res < best_c5:
                best_h = res_x
                best_c5 = c5_res
        except:
            pass
    
    # Final check
    best_h = project(best_h, target)
    best_c5 = evaluate_c5(best_h)
    
    return best_h, best_c5, len(best_h)

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project(h, target):
    h = np.asarray(h, dtype=float)
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        if np.clip(h + mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_grad(corr, tau):
    x = tau * corr
    x_max = x.max()
    exp_x = np.exp(x - x_max)
    s = exp_x.sum()
    return (x_max + np.log(s)) / tau, exp_x / s / tau

def compute_grad(h, tau):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    val, g = smooth_max_grad(corr, tau)
    grad = np.zeros(n)
    for j in range(n):
        mlo = max(0, n - 1 - j)
        mhi = min(2 * n - 2, 2 * n - 2 - j)
        if mhi >= mlo:
            idx = j + np.arange(mlo, mhi + 1) - (n - 1)
            w = g[mlo:mhi + 1]
            grad[j] += w.sum() - (w * h[idx]).sum()
        mlo2 = j
        mhi2 = min(2 * n - 2, j + n - 1)

            w2 = g[mlo2:mhi2 + 1]
            grad[j] -= (w2 * h[idx2]).sum()
    return val, grad

def pgd_optimize(h0, tau, target, max_iter=2000, lr0=0.005, momentum=0.9):

    v = np.zeros(n)

    best_c5 = evaluate_c5(h)
    val, grad = compute_grad(h, tau)
    lr = lr0
    for it in range(max_iter):
        v = momentum * v - lr * grad
        h_new = project(h + v, target)
        val_new, grad_new = compute_grad(h_new, tau)

        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        lr = min(lr * 1.05, lr0)

            if rem() < 25:
                break
            h_new, c5_new = pgd_optimize(h_fine, tau, tf, 1000, 0.003, 0.85)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new
                n, target = nf, tf
            h_fine = h_new

    # Phase 5: Random restarts
    for _ in range(30):
        if rem() < 8:
            break
        scale = 0.03 + 0.12 * np.random.random()
        hp = best_h + np.random.normal(0, scale, n)
        hp = project(hp, target)
        for tau in [50, 300, 1500]:
            if rem() < 6:
                break
            h_new, c5_new = pgd_optimize(hp, tau, target, 600, 0.003, 0.8)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new
            hp = h_new

    # Phase 6: Final fine PGD
    if rem() > 10:
        for tau in [5000, 10000]:
            if rem() < 8:
                break
            h_new, c5_new = pgd_optimize(best_h, tau, target, 1000, 0.002, 0.9)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new

    best_h = project(best_h, target)
    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, len(best_h)
