def project_feasible(h, target):
    """Project h onto {0<=x<=1, sum(x)=target} via Euclidean projection."""
    n = len(h)
    # First clip
    h = np.clip(h, 0.0, 1.0)
    # Now find lambda such that sum(clip(h - lambda, 0, 1)) = target
    # clip(h - lambda, 0, 1) = min(max(h - lambda, 0), 1)
    # For a given lambda:
    #   h_i - lambda > 1 → contributes 1 (when lambda < h_i - 1, i.e., lambda < h_i - 1)
    #   0 <= h_i - lambda <= 1 → contributes h_i - lambda
    #   h_i - lambda < 0 → contributes 0 (when lambda > h_i)
    
    # The function f(lambda) = sum(clip(h - lambda, 0, 1)) is decreasing in lambda
    # f(0) = sum(h) (after clipping, so sum is between 0 and n)
    # We need f(lambda) = target
    
    # Bisection
    lo, hi = -1.0, 2.0  # lambda range
    for _ in range(60):
        mid = (lo + hi) / 2
        val = np.sum(np.clip(h - mid, 0.0, 1.0))
        if val > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    return np.clip(h - lam, 0.0, 1.0)

            p1 = np.arange(m, n)
            grad[p1] += wk * (1.0 - x[p1 - m])
            p2 = np.arange(0, n - m)
            grad[p2] += wk * (-x[p2 + m])
    return grad

def project_feasible(h, target):
    h = np.clip(h, 0.0, 1.0)
    if abs(h.sum() - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        val = np.sum(np.clip(h - mid, 0.0, 1.0))
        if val > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    return np.clip(h - lam, 0.0, 1.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

        n0 = len(h_init)

        n0 = 120

    best_c5 = evaluate_c5(best_h)

        return budget_s - (time.time() - t0)
    
    def make_obj_grad(tau, n_pts):
        def obj(x):
            c = np.correlate(x, 1.0 - x, mode="full")
            v, _ = smooth_max_and_weights(c, tau)
            return v
        def objg(x):
            c = np.correlate(x, 1.0 - x, mode="full")
            _, w = smooth_max_and_weights(c, tau)
            return compute_corr_grad(x, w)
        return obj, objg
    
    def slqp_opt(h_start, n_pts, tau, maxiter=400):
        tgt = n_pts / 2.0
        obj, objg = make_obj_grad(tau, n_pts)
        bnds = Bounds(np.zeros(n_pts), np.ones(n_pts))
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - tgt,
                 'jac': lambda x: np.ones(n_pts)}]
        try:
            res = minimize(obj, h_start, jac=objg, method='SLSQP',
                          bounds=bnds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-14})
            x = project_feasible(res.x, tgt)
            c5 = evaluate_c5(x)
            return x, c5
        except Exception:
            return h_start, None
    
    def proj_grad_opt(h_start, n_pts, tau, n_iter=2000):
        tgt = n_pts / 2.0
        h = h_start.copy()
        alpha = 0.01
        for i in range(n_iter):
            c = np.correlate(h, 1.0 - h, mode="full")
            cm = np.max(c)
            _, w = smooth_max_and_weights(c, tau)
            g = compute_corr_grad(h, w)
            # Backtracking
            for _ in range(30):

                c_new = np.max(np.correlate(h_new, 1.0 - h_new, mode="full"))
                if c_new <= cm + 1e-14:
                    break
                alpha *= 0.5
            else:
                h_new = project_feasible(h - alpha * g, tgt)
            h = h_new
            if alpha < 0.005:
                alpha = 0.02
        c5 = evaluate_c5(h)
        return h, c5

            best_h = h_new
            best_c5 = c5_new
