    h = h0.copy()
    y = h0.copy()  # Nesterov momentum
    t_k = 1.0
    
    for k in range(n_iter):
        # Evaluate at y
        corr = compute_corr(y)
        max_idx = np.argmax(corr)
        max_val = corr[max_idx]
        
        # Compute subgradient at y for the max lag
        # grad[j] = dx * [(1-h[y_j - k])*I(0<=j-k<=n-1) - h[j+k]*I(0<=j+k<=n-1)]

        lag = max_idx - (n - 1)
        dx = 2.0 / n
        
        grad = np.zeros(n)
        for j in range(n):
            if 0 <= j - lag <= n - 1:
                grad[j] += (1.0 - y[j - lag])
            if 0 <= j + lag <= n - 1:
                grad[j] -= y[j + lag]
        grad *= dx
        
        # Update

        h[add] = 1
    dx = 2.0 / n
    corr = compute_corr(h)
    best_val = np.max(corr) * dx
    for _ in range(n_iter):
        zero_pos = np.where(h == 0)[0]
        one_pos = np.where(h == 1)[0]
        if len(zero_pos) == 0 or len(one_pos) == 0:
            break
        j = zero_pos[rng.randint(len(zero_pos))]
        jp = one_pos[rng.randint(len(one_pos))]
        # Compute delta for each lag
        for k in range(2 * n - 1):
            lag = k - (n - 1)
            delta = 0.0
            if 0 <= j + lag <= n - 1:
                delta += h[j + lag]
            if 0 <= j - lag <= n - 1:
                delta += 1.0 - h[j - lag]
            if 0 <= jp + lag <= n - 1:
                delta -= h[jp + lag]
            if 0 <= jp - lag <= n - 1:
                delta -= 1.0 - h[jp - lag]
            corr[k] += delta
        new_max = np.max(corr) * dx
        if new_max < best_val:
            h[j] = 1
            h[jp] = 0
            best_val = new_max
        else:
            for k in range(2 * n - 1):
                lag = k - (n - 1)
                delta = 0.0
                if 0 <= j + lag <= n - 1:
                    delta += h[j + lag]
                if 0 <= j - lag <= n - 1:
                    delta += 1.0 - h[j - lag]
                if 0 <= jp + lag <= n - 1:
                    delta -= h[jp + lag]
                if 0 <= jp - lag <= n - 1:
                    delta -= 1.0 - h[jp - lag]
                corr[k] -= delta
    return h, best_val

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = np.arange(n_old) / n_old
    x_new = np.arange(n_new) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project_h(h_new, n_new * 0.5)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - t0)

    h = np.array(initial_h_values, dtype=float)
    n = len(h)
    h = project_h(h, n * 0.5)
    c5 = compute_c5(h)
    h_best = h.copy()
    c5_best = c5

    # Phase 1: Boosting with top-5 lags (sharp focus)
    if time_left() > 250:
        h_b, c5_b = boost_round(h_best, top_k=5, n_rounds=15, max_iter=300, ftol=1e-11)
        if c5_b < c5_best:
            h_best, c5_best = h_b, c5_b

    # Phase 2: t-variable polish
    if time_left() > 200:
        h_t, c5_t = slsqp_t_var(h_best, max_iter=800, ftol=1e-12, margin=0.005)
        if c5_t < c5_best:
            h_best, c5_best = h_t, c5_t

    # Phase 3: Boosting with different top_k values
    for k_val in [3, 8, 12, 20]:
        if time_left() < 150:
            break
        h_b, c5_b = boost_round(h_best, top_k=k_val, n_rounds=10, max_iter=250, ftol=1e-11, beta=30.0)
        if c5_b < c5_best:
            h_best, c5_best = h_b, c5_b

    # Phase 4: Another t-variable polish
    if time_left() > 150:
        h_t, c5_t = slsqp_t_var(h_best, max_iter=1000, ftol=1e-13, margin=0.003)
        if c5_t < c5_best:
            h_best, c5_best = h_t, c5_t

    # Phase 5: Binary local search for diversification
    if time_left() > 200:
        h_bin, c5_bin = binary_local_search(h_best, 5000, rng)
        # Use binary as starting point for continuous optimization
        h_cont = project_h(h_bin.astype(float), n * 0.5)
        h_t, c5_t = slsqp_t_var(h_cont, max_iter=600, ftol=1e-11, margin=0.02)
        if c5_t < c5_best:
            h_best, c5_best = h_t, c5_t
        c5_b2 = boost_round(h_best, top_k=5, n_rounds=10, max_iter=200, ftol=1e-11)
        if c5_b2[1] < c5_best:
            h_best, c5_best = c5_b2[0], c5_b2[1]

    # Phase 6: Random perturbations + re-optimize
    for i in range(15):
        if time_left() < 60:
            break
        scale = 0.005 + 0.003 * i
        h_p = h_best + rng.randn(n) * scale
        h_p = project_h(h_p, n * 0.5)
        h_t, c5_t = slsqp_t_var(h_p, max_iter=300, ftol=1e-11, margin=0.01)
        if c5_t < c5_best:
            h_best, c5_best = h_t, c5_t
        if time_left() > 120:
            h_b, c5_b = boost_round(h_best, top_k=5, n_rounds=8, max_iter=200, ftol=1e-11)
            if c5_b < c5_best:
                h_best, c5_best = h_b, c5_b

    # Phase 7: Try coarser resolution for global structure
    for n_coarse in [90, 60]:
        if time_left() < 200:
            break
        h_c = interp_to_n(h_best, n_coarse)
        h_t, c5_t = slsqp_t_var(h_c, max_iter=600, ftol=1e-12, margin=0.01)
        if c5_t < c5_best:
            h_up = interp_to_n(h_t, n)
            h_t2, c5_t2 = slsqp_t_var(h_up, max_iter=500, ftol=1e-12, margin=0.005)
            if c5_t2 < c5_best:
                h_best, c5_best = h_t2, c5_t2
            else:
                h_best, c5_best = h_up, compute_c5(h_up)
                if c5_best > c5_t:
                    h_best = h_up
                    c5_best = compute_c5(h_best)

    # Phase 8: Final intensive polish
    if time_left() > 100:
        h_b, c5_b = boost_round(h_best, top_k=3, n_rounds=8, max_iter=400, ftol=1e-13, beta=100.0)
        if c5_b < c5_best:
            h_best, c5_best = h_b, c5_b
    if time_left() > 60:
        h_t, c5_t = slsqp_t_var(h_best, max_iter=1200, ftol=1e-14, margin=0.002)
        if c5_t < c5_best:
            h_best, c5_best = h_t, c5_t

    # Final validation
    h_best = project_h(h_best, n * 0.5)
    c5_final = compute_c5(h_best)
    return h_best, c5_final, n
