    return np.clip(h, 0.0, 1.0)

def obj_smooth(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 700:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def sum_con(h, n):
    return np.sum(h) - n / 2.0

def sum_con_jac(h, n):
    return np.ones(n)

def slsqp(h, n, alpha, maxiter=2000, ftol=1e-16):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def upsample(h, n_old, n_new):
    """Upsample from n_old to n_new points using linear interpolation."""
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=2)
    return project_feasible(h_new, n_new)

def coordinate_descent(h, n, max_iters=5000, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    best_c5 = compute_c5(h, n)
    h = h.copy()
    deltas = [0.001, 0.002, 0.005, 0.01, 0.02, 0.05]
    
    for it in range(max_iters):
        j = rng.randint(n)
        k = rng.randint(n)
        if j == k:
            continue
        for d in deltas:
            for s in [1, -1]:
                di = s * d
                hi = h[j] + di
                hk = h[k] - di
                if 0 <= hi <= 1 and 0 <= hk <= 1:
                    h_try = h.copy()
                    h_try[j] = hi
                    h_try[k] = hk
                    c5 = compute_c5(h_try, n)
                    if c5 < best_c5 - 1e-15:
                        h = h_try
                        best_c5 = c5
                        break
            else:
                continue
            break
    return h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def all_overlaps(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(3000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-15:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def obj_lse(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 700:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def obj_power(h, n, p):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    c_pos = np.maximum(c, 0)
    return float(np.sum(c_pos ** p) ** (1.0 / p))

def sum_con(h, n):
    return np.sum(h) - n / 2.0

def sum_con_jac(h, n):
    return np.ones(n)

def slsqp(h, n, alpha, maxiter=2000, ftol=1e-16):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(obj_lse, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def lbfgsb(h, n, alpha, maxiter=5000):
    """L-BFGS-B with soft penalty for sum constraint"""
    target = n / 2.0
    penalty = 1e6
    
    def obj_pen(x):
        hp = np.clip(x, 0, 1)
        val = obj_lse(hp, n, alpha)
        val += penalty * (np.sum(hp) - target) ** 2
        return val
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_pen, h, method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-8})
        return project_feasible(res.x, n)
    except Exception:
        return h

def coord_desc_adaptive(h, n, max_iters=20000, rng=None):
    """Coordinate descent with adaptive step sizes."""
    if rng is None:
        rng = np.random.RandomState(0)
    best_c5 = compute_c5(h, n)
    h = h.copy()
    base_deltas = [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2]
    
    for it in range(max_iters):
        j = rng.randint(n)
        k = rng.randint(n)
        if j == k:
            continue
        # Scale deltas by current "roughness"
        scale = 1.0 if best_c5 > 0.39 else (0.5 if best_c5 > 0.381 else 0.2)
        for d in sorted(base_deltas, key=lambda x: abs(x - 0.02 * scale)):
            for s in [1, -1]:
                di = s * d * scale
                hi = h[j] + di
                hk = h[k] - di
                if 0 <= hi <= 1 and 0 <= hk <= 1:
                    h_try = h.copy()
                    h_try[j] = hi

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(3000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-15:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def obj_lse(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 700:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    c_pos = np.maximum(c, 1e-30)
    return float(np.sum(c_pos ** p) ** (1.0 / p))
