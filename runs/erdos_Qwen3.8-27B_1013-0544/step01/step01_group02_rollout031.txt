import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    """Project onto {0<=h<=1, sum(h)=n/2}"""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
        else:
            mask = h > 0.0
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_subgradient(h, n, top_m=5):
    """Compute subgradient of C5 using top_m active constraints."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    # Get the top_m values
    m = min(top_m, len(c))
    top_indices = np.argsort(c)[-m:]
    
    # For each active lag k, compute gradient
    # c[k_full] corresponds to lag k = k_full - (n-1)
    grad = np.zeros(n)
    for kf in top_indices:
        k = kf - (n - 1)  # actual lag
        # c_k = dx * sum_i h[i] * (1 - h[i+k]) for valid i
        # d(c_k)/d(h[j]) = dx * [(1-h[j+k]) if j+k in range] - dx * [h[j-k] if j-k in range]
        for j in range(n):
            g = 0.0
            if 0 <= j + k < n:
                g += (1.0 - h[j + k])
            if 0 <= j - k < n:
                g -= h[j - k]
            grad[j] += dx * g
    grad /= m
    return grad, c

def subgradient_optimize(h0, n, max_iter=5000, top_m=5, base_lr=0.01):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    
    for it in range(max_iter):
        grad, c = compute_subgradient(h, n, top_m)
        # Adaptive step size
        lr = base_lr / (1.0 + it * 0.001)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()
            h = h_new
        else:
            # Halve learning rate
            base_lr *= 0.95
            h = project_feasible(h - lr * 0.5 * grad, n)
    
    return best_h, best_c5

- Keep only breakpoints in the feasible interval
- Evaluate f(δ) at each candidate point
- Return the best

    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        s = h.sum()
        if abs(s - target) < 1e-14:
            break
        diff = target - s
        if diff > 0:
            mask = h < 1.0
            cnt = mask.sum()
            if cnt == 0: break
            h[mask] = np.minimum(1.0, h[mask] + diff / cnt)
        else:
            mask = h > 0.0
            cnt = mask.sum()
            if cnt == 0: break
            h[mask] = np.maximum(0.0, h[mask] + diff / cnt)
    return np.clip(h, 0.0, 1.0)

def solve_alm_lp(h_repl, n):
    """Solve the linearized LP for ALM.
    min t s.t. sum_i h[i]*(1-h_repl[i+lag])*dx <= t for all lags
               sum h[i]*dx = 1, 0<=h[i]<=1
    """
    dx = 2.0 / n
    n_lags = 2 * n - 1
    
    # Objective: minimize t (last variable)
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # coefficient of t
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each lag: sum_i h[i] * coeff_i - t <= 0
    A_ub = np.zeros((n_lags, n + 1))
    b_ub = np.zeros(n_lags)
    
    for lag in range(-(n-1), n):
        row_idx = lag + (n - 1)
        for i in range(n):
            j = i + lag
            if 0 <= j < n:
                A_ub[row_idx, i] = dx * (1.0 - h_repl[j])
        A_ub[row_idx, n] = -1.0  # coefficient of t
    
    # Equality: sum(h) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Bounds: 0 <= h[i] <= 1, 0 <= t <= 2 (t is at most 2*dx*n = 2*2 = 4... let's use 1.0 as upper bound since C5 should be < 1)
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        return res.x[:n]
    else:
        return h_repl.copy()

def obj_smooth(h, n, alpha):

    if alpha < 1e-10:

    if max_shifted > 500:
        log_sum = max_shifted + np.log(np.sum(np.exp(shifted - max_shifted)))
    else:
        log_sum = np.log(np.sum(np.exp(shifted)))
    return float(m + log_sum / alpha)

        return np.sum(x) - n / 2.0

        return np.ones(n)

                break

                if time.time() - t0 > budget_s * 0.9:
                    break
                # Try updating h[i] and adjusting a random other coordinate
                # For efficiency, try the "best" other coordinate
                # Simple approach: try each j in a random subset
                for _ in range(5):
                    j = rng.randint(n)
                    if j == i: continue
                    # Optimize over delta: h[i] += delta, h[j] -= delta
                    # Feasible: h[i]+delta in [0,1], h[j]-delta in [0,1]
                    d_lo = max(-best_h[i], best_h[j] - 1.0)
                    d_hi = min(1.0 - best_h[i], best_h[j])
                    if d_lo >= d_hi: continue

                    # c_k(delta) = c_k + delta * a_k where a_k = d(c_k)/d(h[i]) - d(c_k)/d(h[j])
                    dx = 2.0 / n
                    a_k = np.zeros(2*n-1)
                    for lag in range(-(n-1), n):
                        kf = lag + (n-1)
                        gi = 0.0
                        gj = 0.0
                        if 0 <= i - lag < n: gi += (1.0 - best_h[i - lag])
                        if 0 <= i + lag < n: gi -= best_h[i + lag]
                        if 0 <= j - lag < n: gj += (1.0 - best_h[j - lag])
                        if 0 <= j + lag < n: gj -= best_h[j + lag]
                        a_k[kf] = dx * (gi - gj)
                    
                    c_curr = np.correlate(best_h, 1-best_h, mode='full') * dx
                    
                    # f(delta) = max_k (c_curr[k] + delta * a_k[k])
                    # Minimize over delta in [d_lo, d_hi]
                    # Find breakpoints
                    candidates = [d_lo, d_hi]
                    # Add breakpoints (subsample for efficiency)
                    for k1 in range(0, 2*n-1, 3):  # subsample
                        for k2 in range(k1+1, 2*n-1, 3):
                            da = a_k[k1] - a_k[k2]
                            if abs(da) < 1e-10: continue
                            t_bp = (c_curr[k2] - c_curr[k1]) / da
                            if d_lo < t_bp < d_hi:
                                candidates.append(t_bp)
                    
                    best_d = 0.0
                    best_val = float('inf')
                    for d in candidates:
                        d = max(d_lo, min(d_hi, d))
                        val = float(np.max(c_curr + d * a_k))
                        if val < best_val:
                            best_val = val
                            best_d = d
                    
                    if best_d != 0.0:
                        h_try = best_h.copy()
                        h_try[i] += best_d
                        h_try[j] -= best_d
                        c_new = compute_c5(h_try, n)
                        if c_new < best_c5 - 1e-15:
                            best_h = h_try
                            best_c5 = c_new
    
    # Phase 4: Final SLSQP polish
    if time.time() - t0 < budget_s * 0.95:
        try:
            res = minimize(obj_smooth, best_h, args=(n, 50000), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 5000, 'ftol': 1e-20})
            h_try = project_feasible(res.x, n)
            c = compute_c5(h_try, n)
            if c < best_c5:
                best_c5 = c
                best_h = h_try.copy()
        except Exception:
            pass
    
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n

import numpy as np
from scipy.optimize import minimize, linprog

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        s = h.sum()
        if abs(s - target) < 1e-13:
            break
        diff = target - s
        if diff > 0:
            mask = h < 1.0
            cnt = int(mask.sum())
            if cnt == 0: break
            h[mask] = np.minimum(1.0, h[mask] + diff / cnt)
        else:
            mask = h > 0.0
            cnt = int(mask.sum())
            if cnt == 0: break
            h[mask] = np.maximum(0.0, h[mask] + diff / cnt)
    return np.clip(h, 0.0, 1.0)

def solve_alm_lp(h_repl, n):
    dx = 2.0 / n
    n_lags = 2 * n - 1
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    A_ub = np.zeros((n_lags, n + 1))
    b_ub = np.zeros(n_lags)
    for lag in range(-(n-1), n):
        row = lag + (n - 1)
        for i in range(n):
            j = i + lag
            if 0 <= j < n:
                A_ub[row, i] = dx * (1.0 - h_repl[j])
        A_ub[row, n] = -1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = [(0.0, 1.0)] * n + [(0.0, 5.0)]
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n]
    return h_repl.copy()

    """Minimize a convex function on [lo, hi] using ternary search."""
    for _ in range(n_iter):
        if hi - lo < 1e-12:
            break
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if f_eval(m1) < f_eval(m2):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0
