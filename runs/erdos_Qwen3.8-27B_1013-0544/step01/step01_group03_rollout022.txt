L(h) = (1/τ) log(∑_k exp(τ · corr_k))

where corr_k = ∑_i h_i · (1 - h_{i+m_k}) for valid indices, m_k = k - (n-1)

∂L/∂h_j = (1/τ) · ∑_k τ · exp(τ·corr_k) / ∑_l exp(τ·corr_l) · ∂corr_k/∂h_j
= ∑_k p_k · ∂corr_k/∂h_j

where p_k = exp(τ·corr_k) / ∑_l exp(τ·corr_l)

Now, corr_k = ∑_i h_i - ∑_i h_i · h_{i+m_k} (for valid i)

∂corr_k/∂h_j = 1 - ∂/∂h_j (∑_i h_i · h_{i+m_k})

For the second term:
∑_i h_i · h_{i+m} with respect to h_j:
- When i = j: contributes h_{j+m} (if j+m is valid)
- When i + m = j, i.e., i = j - m: contributes h_{j-m} (if j-m is valid, i.e., i >= 0)

So ∂(∑_i h_i · h_{i+m})/∂h_j = h_{j+m} (if 0 ≤ j+m < n) + h_{j-m} (if 0 ≤ j-m < n)

Therefore:
∂corr_k/∂h_j = 1 - h_{j+m_k} (if valid) - h_{j-m_k} (if valid)

And:
∂L/∂h_j = ∑_k p_k · (1 - h_{j+m_k} (if valid) - h_{j-m_k} (if valid))
= ∑_k p_k - ∑_k p_k · h_{j+m_k} (if valid) - ∑_k p_k · h_{j-m_k} (if valid)
= 1 - ∑_k p_k · h_{j+m_k} (if valid) - ∑_k p_k · h_{j-m_k} (if valid)

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val_grad(h, tau):
    """Value and gradient of (1/tau)*log(sum exp(tau*corr_k))"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    z = tau * corr
    z_max = np.max(z)
    exp_z = np.exp(z - z_max)
    p = exp_z / np.sum(exp_z)
    val = (z_max + np.log(np.sum(exp_z))) / tau
    
    # Gradient: sum_k p_k * d(corr_k)/dh_j
    # d(corr_k)/dh_j = 1 - h[j+m_k](if valid) - h[j-m_k](if valid)
    # grad_j = 1 - sum_k p_k*h[j+m_k](valid) - sum_k p_k*h[j-m_k](valid)
    grad = np.ones(n)
    for k in range(len(p)):
        pk = p[k]
        if pk < 1e-14:
            continue
        m = k - n + 1
        if m >= 0:
            grad[:n-m] -= pk * h[m:]
            grad[m:] -= pk * h[:n-m]
        else:
            pa = -m
            grad[pa:] -= pk * h[:n-pa]       # h[j+m] = h[j-pa], valid j>=pa
            grad[:n-pa] -= pk * h[pa:]       # h[j-m] = h[j+pa], valid j<n-pa
    return val, grad

def fix_sum(h, target):
    h = np.array(h, dtype=float).copy()
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return np.clip(h, 0.0, 1.0)
    if diff > 0:
        mask = h < 1.0 - 1e-10
    else:
        mask = h > 1e-10
    cnt = mask.sum()
    if cnt == 0:
        return np.clip(h, 0.0, 1.0)
    h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

        r = np.clip(0.5 + 0.25 * np.sin(2 * np.pi * f1 * xgrid) + 0.15 * np.sin(2 * np.pi * f2 * xgrid), 0, 1)
        restarts.append(r)
    # Blocky random
    for bs in [2, 3, 4, 6, 8, 12]:
        nb = hn // bs + 1
        r = np.repeat(np.random.random(nb), bs)[:hn]
        restarts.append(r)
    # Smoothed random
    for _ in range(6):
        r = np.random.random(hn)
        ker = np.ones(7) / 7.0
        r = np.convolve(r, ker, mode='same')
        restarts.append(r)

    for r in restarts:
        if rem() < 12:
            break
        r = fix_sum(r, tgt)
        for tau in [100, 500, 2000]:
            if rem() < 8:
                break
            h_new, c5_new = run_optimize(r, tau, 200, 200.0)
            if c5_new is not None and c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
            r = h_new

    # Phase 4: Shift-specific optimization (polish the active constraint)
    if rem() > 30:
        for _ in range(30):
            if rem() < 10:
                break
            corr = np.correlate(best_h, 1.0 - best_h, mode="full")
            k_star = int(np.argmax(corr))
            m_star = k_star - hn + 1
            h_new = optimize_shift(best_h, m_star, 300)
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
            else:
                # Try a random different shift to escape
                k_alt = np.random.randint(0, len(corr))
                m_alt = k_alt - hn + 1
                h_new = optimize_shift(best_h, m_alt, 200)
                c5_new = evaluate_c5(h_new)
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new

    # Phase 5: Final high-tau L-BFGS-B polish
    for tau in [5000, 15000, 30000, 50000]:
        if rem() < 15:
            break
        h_new, c5_new = run_optimize(best_h, tau, 500, 500.0)
        if c5_new is not None and c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 6: One more round of shift-specific after high-tau
    if rem() > 15:
        for _ in range(20):
            if rem() < 5:
                break
            corr = np.correlate(best_h, 1.0 - best_h, mode="full")
            k_star = int(np.argmax(corr))
            m_star = k_star - hn + 1
            h_new = optimize_shift(best_h, m_star, 400)
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    # Final constraint enforcement
    n_final = len(best_h)
    best_h = fix_sum(best_h, n_final / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = evaluate_c5(best_h)

    return best_h, best_c5, n_final
