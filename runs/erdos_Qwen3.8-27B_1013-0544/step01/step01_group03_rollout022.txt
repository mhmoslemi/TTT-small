```python
    # For each i: grad[i] = sum_k w[k] * (B[i+k-(n-1)] - h[i-k+(n-1)])
    # = sum_k w[k] * B[i+k-(n-1)] - sum_k w[k] * h[i-k+(n-1)]
    
    # First sum: for fixed i, k ranges over values where 0<=i+k-(n-1)<n
    # i.e., (n-1-i) <= k <= (n-1+i), intersected with [0, 2n-2]
    
    # Let me use np.convolve:
    # np.convolve(w, B, mode='full')[k] = sum_j w[j] * B[k-j]
    # We want sum_k w[k] * B[i+k-(n-1)]
    # Let m = k: sum_m w[m] * B[i+m-(n-1)]
    # = sum_m B[i+m-(n-1)] * w[m]
    # In convolution notation: (B_shifted * w)[i] where B_shifted[m] = B[m-(n-1)]
    
    # OK I'll just use a Toeplitz matrix approach or direct computation.
    
    # Direct vectorized approach:
    # Create a matrix where M[i,k] = B[i+k-(n-1)] (with zeros for invalid)
    # This is a Toeplitz-like matrix
    idx = np.arange(n)[:, None] + np.arange(2*n-1)[None, :] - (n-1)  # shape (n, 2n-1)
    valid1 = (idx >= 0) & (idx < n)
    B_ext = np.zeros(n)
    B_ext[:] = B  # B is already length n
    term1 = np.where(valid1, B[idx.clip(0, n-1)], 0) @ w
    
    idx2 = np.arange(n)[:, None] - np.arange(2*n-1)[None, :] + (n-1)
    valid2 = (idx2 >= 0) & (idx2 < n)
    term2 = np.where(valid2, h[idx2.clip(0, n-1)], 0) @ w
    
    grad = term1 - term2
    return grad
```

```python
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_obj_grad(h, tau):
    n = len(h)
    B = 1.0 - h
    corr = np.correlate(h, B, mode="full")
    
    x = tau * corr
    x_max = np.max(x)
    exp_x = np.exp(x - x_max)
    Z = np.sum(exp_x)
    w = exp_x / Z
    F = (x_max + np.log(Z)) / tau
    
    i_idx = np.arange(n)
    diff = i_idx[:, None] - i_idx[None, :] + (n - 1)
    M = w[diff]
    
    term1 = M @ B
    term2 = M.T @ h
    grad = term1 - term2
    return F, grad

def project_sum(h, target):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def pgd_step(h, tau, step):
    F, grad = compute_obj_grad(h, tau)
    n = len(h)
    target = n / 2.0
    h_new = project_sum(h - step * grad, target)
    F_new, _ = compute_obj_grad(h_new, tau)
    return h_new, F, F_new
```
