def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_lse(obj, tau):
    """Log-sum-exp smooth max"""
    x = tau * obj
    x_max = np.max(x)
    return float(x_max / tau + np.log(np.sum(np.exp(x - x_max))) / tau)

def smooth_max_power(obj, p):
    """Power mean smooth max"""
    return float(np.mean(obj**p)**(1.0/p))

def fix_sum(h, target, tol=1e-10):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < tol:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = mask.sum()
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    S_j = h.sum() - h[j]
    t_star = target - S_j  # ideal value for h[j]

    # c_k = sum_i h_i(1-h_{i+k}) dx
    # Terms with h_j: i=j gives h_j(1-h_{j+k}), i=j-k gives h_{j-k}(1-h_j)

    # In np.correlate(h, 1-h, mode="full"), the result has length 2n-1
    # Index 0 corresponds to lag -(n-1), index n-1 to lag 0, index 2n-2 to lag (n-1)
    
    # Let's just compute the full correlation with h_j = 0, and the coefficient
    # c_k = corr_base_k + h_j * coeff_k
    # where corr_base_k = c_k - h_j * (1 - h_{j+k} - h_{j-k} + ... )

    b_consts = corr0  # = c_k(0)
    
    # 1D optimization: min_t max_k (a_k*t + b_k) + lam*(t - t_star)^2
    t_vals = np.linspace(0, 1, 200)

        obj_vals[i] = np.max(c_vals) + lam * (t - t_star)**2
    
    best_t = t_vals[np.argmin(obj_vals)]

    return np.sum(x) - _constraint_target[0]

    return np.ones(len(x))

def fix_sum(h, target):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-12
    else:
        mask = h > 1e-12
    cnt = mask.sum()
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def coord_optimize(h, j, lam=50.0):
    """Optimize single coordinate h[j] via 1D search."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    t_star = target - (h.sum() - h[j])
    
    h0 = h.copy()
    h0[j] = 0.0
    h1 = h.copy()
    h1[j] = 1.0
    
    corr0 = np.correlate(h0, 1.0 - h0, mode="full") * dx
    corr1 = np.correlate(h1, 1.0 - h1, mode="full") * dx
    
    a_coeffs = corr1 - corr0
    b_consts = corr0
    
    t_vals = np.linspace(0, 1, 300)

    # c_vals = b_consts[None,:] + t_vals[:,None] * a_coeffs[None,:]

    c_mat = b_consts[None, :] + t_vals[:, None] * a_coeffs[None, :]
    obj_vals = np.max(c_mat, axis=1) + lam * (t_vals - t_star) ** 2
    best_t = t_vals[np.argmin(obj_vals)]

    n = len(h)
    h = h.copy()
    for sweep in range(n_sweeps):
        if random_order:
            order = np.random.permutation(n)
        else:
            order = range(n)
        for j in order:
            h[j] = coord_optimize(h, j, lam)
    return h

    target = n / 2.0
    _constraint_target[0] = target
    bnds = Bounds(np.zeros(n), np.ones(n))
    dx = 2.0 / n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full") * dx
        return smooth_max_lse(corr, tau)
    
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bnds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        x = fix_sum(res.x, target)
        return x, evaluate_c5(x)
    except Exception:
        return h, None

Actually, I can optimize this. Instead of computing two full correlations, I can compute the difference analytically:

When I change h[j] from its current value to t:
- The term h_j(1-h_{j+k}) changes from h[j]*(1-h[j+k]) to t*(1-h[j+k]): change = (t - h[j])*(1-h[j+k])
- The term h_{j-k}(1-h_j) changes from h[j-k]*(1-h[j]) to h[j-k]*(1-t): change = (t - h[j])*h[j-k]... 

Wait, let me be more careful. c_k = Σ_i h_i(1-h_{i+k}) dx.

When h[j] changes from old to new:
- If j is in the sum (i=j): contribution changes from old*(1-h[j+k]) to new*(1-h[j+k])
  Change: (new - old)*(1-h[j+k])
- If j+k is in the sum as i+k (i=j-k): contribution changes from h[j-k]*(1-old) to h[j-k]*(1-new)
  Change: h[j-k]*(old - new) = -(new - old)*h[j-k]

Total change in c_k: (new - old) * [(1-h[j+k]) - h[j-k]] (with appropriate boundary handling)

So c_k(new) = c_k(old) + (new - old) * [(1-h[j+k]) - h[j-k]] * dx

This means I only need ONE correlation (the current one), plus a simple vector operation!

Let me rewrite:

```python
def coord_optimize_v2(h, j, lam=50.0):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    t_star = target - (h.sum() - h[j])
    
    # Current correlation
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    
    # For changing h[j] from h[j] to t:
    # c_k changes by (t - h[j]) * [(1-h_{j+k}) - h_{j-k}] * dx
    # where indices are clamped to [0, n-1]
    
    # In np.correlate(h, 1-h, mode="full"), result has length 2n-1
    # Index n-1 corresponds to lag 0
    # Index n-1+k corresponds to lag k (for k = -(n-1), ..., n-1)
    # Lag k: c_k = sum_i h[i](1-h[i+k]) for valid i
    
    # When we change h[j]:
    # - h[j] appears as h[i] when i=j, affecting lag k (term: h[j]*(1-h[j+k]))
    # - h[j] appears as h[i+k] when i=j-k, affecting lag k (term: h[j-k]*(1-h[j]))
    
    # For lag k (corresponding to index n-1+k in the full correlation):
    # Term 1 (h[j] as h[i]): exists if 0 <= j < n and 0 <= j+k < n
    #   contribution: (t - h[j]) * (1 - h[j+k]) * dx
    # Term 2 (h[j] as h[i+k], i=j-k): exists if 0 <= j-k < n and 0 <= j < n
    #   contribution: -h[j-k] * (t - h[j]) * dx = (t - h[j]) * (-h[j-k]) * dx
    
    # Wait, I need to be careful with the correlate indexing.
    # np.correlate(a, v, mode="full") computes:
    # c_n[i] = sum_j a[j+i] * conj(v[j])  (or similar, depending on convention)
    
    # Actually, let me just use the two-point evaluation (h0 and h1) which is simpler
    # and still fast enough for n=120.
    
    h0 = h.copy()
    h0[j] = 0.0
    h1 = h.copy()
    h1[j] = 1.0
    
    corr0 = np.correlate(h0, 1.0 - h0, mode="full") * dx
    corr1 = np.correlate(h1, 1.0 - h1, mode="full") * dx
    
    a_coeffs = corr1 - corr0  # slope of c_k as function of h[j]
    b_consts = corr0  # c_k when h[j] = 0
    
    # 1D: min_t max_k (a_k*t + b_k) + lam*(t - t_star)^2
    t_vals = np.linspace(0, 1, 500)
    c_mat = b_consts[None, :] + t_vals[:, None] * a_coeffs[None, :]
    obj_vals = np.max(c_mat, axis=1) + lam * (t_vals - t_star) ** 2
    best_t = float(t_vals[np.argmin(obj_vals)])
    return best_t
```

_con_target = [0.0]
_obj_tau = [100.0]
_obj_n = [120]

def constraint_fun(x):
    return np.sum(x) - _con_target[0]

def constraint_jac(x):
    return np.ones(len(x))

def smooth_max_obj(x):
    n = _obj_n[0]
    dx = 2.0 / n
    tau = _obj_tau[0]
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    xv = tau * corr
    xm = np.max(xv)
    return float(xm / tau + np.log(np.sum(np.exp(xv - xm))) / tau)

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-12
    else:
        mask = h > 1e-12
    cnt = mask.sum()
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def coord_optimize(h, j, lam=50.0):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    t_star = target - (h.sum() - h[j])
    h0 = h.copy()
    h0[j] = 0.0
    h1 = h.copy()
    h1[j] = 1.0
    corr0 = np.correlate(h0, 1.0 - h0, mode="full") * dx
    corr1 = np.correlate(h1, 1.0 - h1, mode="full") * dx
    a_c = corr1 - corr0
    b_c = corr0
    t_vals = np.linspace(0.0, 1.0, 400)
    c_mat = b_c[None, :] + t_vals[:, None] * a_c[None, :]
    obj_vals = np.max(c_mat, axis=1) + lam * (t_vals - np.clip(t_star, 0, 1)) ** 2
    return float(t_vals[np.argmin(obj_vals)])

def coord_sweep(h, n_sweeps=3, lam=50.0, rnd=False):
    n = len(h)
    h = h.copy()
    for _ in range(n_sweeps):
        order = np.random.permutation(n) if rnd else range(n)
        for j in order:
            h[j] = coord_optimize(h, j, lam)
    return h

def slsqp_refine(h, tau, maxiter=300):
    n = len(h)
    target = n / 2.0
    _con_target[0] = target
    _obj_tau[0] = tau
    _obj_n[0] = n
    bnds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    try:
        res = minimize(smooth_max_obj, h, method='SLSQP', bounds=bnds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        x = fix_sum(res.x, target)
        return x, evaluate_c5(x)
    except Exception:
        return h, None
