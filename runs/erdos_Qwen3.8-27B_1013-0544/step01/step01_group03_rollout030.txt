        T = T_start * (T_end / T_start) ** (step / max(n_steps, 1))
        if not ones or not zeros:
            break
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        c5_new = evaluate_c5(h)
        delta = c5_new - current_c5
        if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-12)):
            current_c5 = c5_new
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c5

def continuous_sa(h, target, n_steps, T_start, T_end):
    n = len(h)
    h_cur = h.copy()
    cur_c5 = evaluate_c5(h_cur)
    best_h = h_cur.copy()
    best_c5 = cur_c5
    for step in range(n_steps):
        T = T_start * (T_end / T_start) ** (step / max(n_steps, 1))
        n_p = np.random.randint(3, 12)
        idx = np.random.choice(n, n_p, replace=False)
        scale = 0.03 + 0.07 * (T / T_start)
        noise = np.random.randn(n_p) * scale
        h_new = h_cur.copy()
        h_new[idx] += noise
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = fix_sum(h_new, target)
        c5_new = evaluate_c5(h_new)
        delta = c5_new - cur_c5
        if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-12)):
            h_cur = h_new
            cur_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_cur.copy()
    return best_h, best_c5

def slsqp_opt(h_start, n_pts, tau, maxiter=1000, ftol=1e-14):
    target = n_pts / 2.0
    bnds = Bounds(np.zeros(n_pts), np.ones(n_pts))
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        return smooth_max(corr, tau)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n_pts)}]
    try:
        res = minimize(obj, h_start, method='SLSQP', bounds=bnds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        x = fix_sum(res.x, target)
        return x, evaluate_c5(x)
    except Exception:
        return h_start, None

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)
    def rem():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(70, 0.5)

    n = len(best_h)
    target = n / 2.0
    best_c5 = evaluate_c5(best_h)

    # Phase 1: Binary SA - explore discrete structure
    if rem() > 250:
        idx_top = np.argsort(best_h)[::-1][:int(target)]
        h_bin = np.zeros(n)
        h_bin[idx_top] = 1.0
        for trial in range(4):
            if rem() < 250:
                break
            n_steps = 300000
            T_s = 0.04 * (1 + 0.3 * trial)
            h_b, c5_b = binary_sa(h_bin, n_steps, T_s, 0.001)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()
            h_bin = h_b

    # Phase 2: Continuous SA - explore continuous landscape
    if rem() > 200:
        n_steps = 800000
        h_s, c5_s = continuous_sa(best_h, target, n_steps, 0.008, 1e-5)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s

    # Phase 3: SLSQP annealing with increasing tau
    tau_sched = [5, 20, 50, 100, 300, 1000, 3000, 8000]
    for t in tau_sched:
        if rem() < 20:
            break
        h_new, c5_new = slsqp_opt(best_h, n, t, maxiter=1500)
        if c5_new is not None and c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 4: Multi-resolution exploration
    for tn in [50, 60, 70, 80, 100, 120, 140, 160]:
        if rem() < 30:
            break
        if tn == n:
            continue
        h_t = np.interp(np.linspace(0, 1, tn), np.linspace(0, 1, n), best_h)
        h_t = fix_sum(h_t, tn / 2.0)
        for t in [100, 500, 2000, 8000]:
            if rem() < 15:
                break
            h_new, c5_new = slsqp_opt(h_t, tn, t, maxiter=600)
            if c5_new is not None and c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
                n = tn
                target = tn / 2.0
            h_t = h_new

    # Phase 5: Binary SA at best resolution
    if rem() > 80:
        idx_top = np.argsort(best_h)[::-1][:int(target)]
        h_bin = np.zeros(n)
        h_bin[idx_top] = 1.0
        for trial in range(3):
            if rem() < 50:
                break
            h_b, c5_b = binary_sa(h_bin, 200000, 0.03, 0.0005)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()
            h_bin = h_b

    # Phase 6: Continuous SA polish
    if rem() > 60:
        h_s, c5_s = continuous_sa(best_h, target, 400000, 0.003, 1e-6)
        if c5_s < best_c5:
            best_h = h_s
            best_c5 = c5_s

    # Phase 7: High-tau SLSQP final polish
    for t in [5000, 15000, 40000, 80000]:
        if rem() < 12:
            break
        h_new, c5_new = slsqp_opt(best_h, n, t, maxiter=2000, ftol=1e-15)
        if c5_new is not None and c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 8: Short SA micro-polish
    if rem() > 15:
        h_s, c5_s = continuous_sa(best_h, target, 100000, 0.001, 1e-7)
        if c5_s < best_c5:
            best_h = h_s
            best_c5 = c5_s

    # Final constraint fix
    best_h = fix_sum(best_h, len(best_h) / 2.0)
    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, len(best_h)
