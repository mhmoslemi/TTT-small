class AdamState:
    def __init__(self, n):
        self.m = np.zeros(n)
        self.v = np.zeros(n)
        self.t = 0

def adam_step(h, grad, state, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    state.t += 1
    state.m = beta1 * state.m + (1-beta1) * grad
    state.v = beta2 * state.v + (1-beta2) * grad**2
    m_hat = state.m / (1 - beta1**state.t)
    v_hat = state.v / (1 - beta2**state.t)
    return h - lr * m_hat / (np.sqrt(v_hat) + eps)

def compute_c5(h):
    n = len(h)
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n)

    for _ in range(50):

        if s > target_sum:
            lo = mid
        else:

    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    if alpha < 1:

            w = np.zeros(len(corr))
            w[np.argmax(corr)] = 1.0

        else:
            p = -k
            grad[:n-p] += wi * (1.0 - h[p:])
