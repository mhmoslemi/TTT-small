```python
def compute_c5_all(h, n):
    """Returns all correlation values (not just max)."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return c

def compute_c5(h, n):
    c = compute_c5_all(h, n)
    return float(np.max(c))

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += (diff / total) * room
            h = np.clip(h, 0.0, 1.0)
        else:
            total = h.sum()
            if total < 1e-14:
                break
            h += (diff / total) * h
            h = np.clip(h, 0.0, 1.0)
    # Final exact fix
    h = np.clip(h, 0.0, 1.0)

            idx = np.where(h < 1.0)[0]
            per = diff / n_pos
            h[idx] = np.clip(h[idx] + per, 0.0, 1.0)
    return h

    c = compute_c5_all(h, n)
    c = np.maximum(c, 0)  # only positive values matter
    return float(np.sum(c**p)**(1.0/p))

def smooth_max_obj(h, n, alpha):

    c = compute_c5_all(h, n)
    m = c.max()
    shifted = alpha * (c - m)
    # Avoid overflow
    log_sum = np.log(np.sum(np.exp(shifted - shifted.max()))) + shifted.max()
    return float(m + log_sum / alpha)

def active_set_obj(h, n, active_lags, alpha):

    c = compute_c5_all(h, n)

    m = c.max()
    # Penalize active lags more
    penalty = alpha * np.sum(np.maximum(c_active - m, 0)**2)
    return float(m + penalty)

    best = h.copy()
    best_c = compute_c5(h, n)
    c = compute_c5_all(h, n)
    for it in range(max_iter):
        # Find the lag with maximum correlation
        k_star = np.argmax(c)
        # Compute gradient of c_{k_star} with respect to h
        # c_{k_star} = dx * sum_i h[i] * (1 - h[i + offset])
        # where offset depends on k_star position in the full correlation
        # For full mode, index k_star corresponds to lag = k_star - (n-1)
        lag = k_star - (n - 1)
        grad = np.zeros(n)
        dx = 2.0 / n
        if lag >= 0:
            for i in range(n - lag):
                # d/dh[i] of h[i]*(1-h[i+lag]) = 1 - h[i+lag]
                grad[i] += dx * (1 - h[i + lag])
                # d/dh[i+lag] of h[i]*(1-h[i+lag]) = -h[i]
                grad[i + lag] += dx * (-h[i])
        else:

            for i in range(lag_abs, n):
                # c_{-lag_abs} = dx * sum_i h[i+lag_abs]*(1-h[i])
                # d/dh[i] of (1-h[i])*h[i+lag_abs] = -h[i+lag_abs]
                grad[i] += dx * (-h[i + lag_abs])

            best_c = c_new

        else:
            step_size *= 0.9
            h = best.copy()
            c = compute_c5_all(h, n)
```

```python
def power_sum_obj_and_grad(h, n, p):
    c = compute_c5_all(h, n)
    c = np.maximum(c, 0)
    val = np.sum(c**p)**(1.0/p)
    # Gradient
    dx = 2.0 / n
    grad = np.zeros(n)
    c_p_minus_1 = c**(p-1)
    factor = (1.0/p) * np.sum(c**p)**(1.0/p - 1.0) * p  # = (sum c^p)^(1/p - 1)
    for k in range(len(c)):
        w = factor * c_p_minus_1[k]
        if w < 1e-15:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            idx1 = np.arange(n - lag)
            idx2 = idx1 + lag
            grad[idx1] += w * dx * (1 - h[idx2])
            grad[idx2] += w * dx * (-h[idx1])
        else:
            m = -lag
            idx1 = np.arange(n - m)
            idx2 = idx1 + m
            grad[idx2] += w * dx * (1 - h[idx1])
            grad[idx1] += w * dx * (-h[idx2])
    return val, grad
```
