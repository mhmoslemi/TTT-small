import numpy as np
import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * (2.0 / n)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_box_sum(h, target_sum):
    n = len(h)
    lo, hi = -1.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    return np.clip(h - lam, 0.0, 1.0)

def get_corr_grad(h, k):
    """Gradient of corr at shift k with respect to h."""
    n = len(h)
    g = np.zeros(n)
    # k is the shift (can be negative)
    # corr[k] = sum_{i: 0<=i<n, 0<=i+k<n} h[i]*(1-h[i+k])
    # d/dh[p] = (1-h[p+k])*[0<=p+k<n] + (-h[p-k])*[0<=p-k<n]
    # Left part: p+k in [0,n-1] => p in [-k, n-1-k]
    pl = max(0, -k)
    pr = min(n, n - k)
    if pr > pl:
        g[pl:pr] += (1.0 - h[pl+k:pr+k])
    # Right part: p-k in [0,n-1] => p in [k, n-1+k]
    plr = max(0, k)
    prr = min(n, n + k)
    if prr > plr:
        g[plr:prr] += (-h[plr-k:prr-k])
    return g

def smooth_max_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    s = w.sum()
    if s < 1e-300:
        idx = np.argmax(corr)
        w = np.zeros_like(corr)
        w[idx] = 1.0
    else:
        w = w / s * alpha
    grad = np.zeros(n)
    active = np.where(w > 1e-8)[0]
    for j in active:
        k = j - (n - 1)
        grad += w[j] * get_corr_grad(h, k)
    return grad * dx

def threshold_penalty_grad(h, t, dx):
    """Gradient of sum_k max(0, corr[k]*dx - t)^2"""
    n = len(h)
    corr = compute_corr(h)
    c = corr * dx
    mask = c > t
    if not np.any(mask):
        return np.zeros(n), 0.0
    # penalty = sum_{k in mask} (c[k] - t)^2
    # d penalty / d h = 2 * sum_{k in mask} (c[k]-t) * dx * d corr[k] / dh
    grad = np.zeros(n)
    for j in np.where(mask)[0]:
        k_shift = j - (n - 1)
        coeff = 2.0 * (c[j] - t) * dx
        grad += coeff * get_corr_grad(h, k_shift)
    val = np.sum((c[mask] - t)**2)
    return grad, val

def optimize_smmax(h0, target_sum, deadline, rng, alpha_start=50, alpha_end=300):
    n = len(h0)
    h = project_box_sum(h0.copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = 0.01
    alpha = alpha_start
    patience = 0
    
    t_start = time.time()
    t_end = deadline
    total_time = max(t_end - t_start, 1.0)
    
    while time.time() < t_end:
        frac = (time.time() - t_start) / total_time
        alpha = alpha_start + (alpha_end - alpha_start) * frac
        
        grad = smooth_max_grad(h, alpha)
        norm_g = np.linalg.norm(grad)
        if norm_g < 1e-10:
            break
        
        # Backtracking line search
        step = lr
        h_new = h
        c5_new = best_c5
        for _ in range(20):
            h_try = project_box_sum(h - step * grad, target_sum)
            c5_try = compute_c5(h_try)
            if c5_try < c5_new - 1e-12:
                h_new = h_try
                c5_new = c5_try
                break
            step *= 0.5
        else:
            patience += 1
            if patience > 50:
                # Perturb
                h = project_box_sum(h + rng.randn(n) * 0.05, target_sum)
                patience = 0
                lr = 0.01
            continue
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        h = h_new
        lr = min(lr * 1.1, 0.1)
        patience = 0
    
    return best_h, best_c5

def optimize_threshold(h0, target_sum, deadline, rng, t_start=None):

def local_search(h0, target_sum, deadline, rng):
    """Local search: try swaps and single moves to improve C5."""
    n = len(h0)
    h = project_box_sum(h0.copy(), target_sum)

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * (2.0 / n)

def project_box_sum(h, target_sum):
    lo, hi = -1.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    return np.clip(h - lam, 0.0, 1.0)

def smooth_max_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w = w / s * alpha
    grad = np.zeros(n)
    offset = n - 1
    active = np.where(w > 1e-8)[0]
    for j in active:
        k = j - offset
        pl = max(0, -k)
        pr = min(n, n - k)
        if pr > pl:
            grad[pl:pr] += w[j] * (1.0 - h[pl+k:pr+k])
        plr = max(0, k)
        prr = min(n, n + k)
        if prr > plr:
            grad[plr:prr] += w[j] * (-h[plr-k:prr-k])
    return grad * dx

            best_h, best_c5 = h1, c1

    # Phase 2: Threshold method
    if rem() > 80:
        d2 = min(rem() * 0.25, 220.0)
        h2, c2 = threshold_opt(best_h, len(best_h) / 2.0, t0 + d2, rng)
        if c2 < best_c5:
            best_h, best_c5 = h2, c2

    # Phase 3: Local search on binary rounding
    if rem() > 60:
        d3 = min(rem() * 0.3, 250.0)
        h3, c3 = local_search_binary(best_h, len(best_h) / 2.0, t0 + d3, rng)
        if c3 < best_c5:
            best_h, best_c5 = h3, c3

    # Phase 4: Try lower resolution n=100
    if rem() > 80:
        n_lo = 100
        h_lo = interpolate_to_n(best_h, len(best_h), n_lo)
        h_lo = project_box_sum(h_lo, n_lo / 2.0)
        d4 = min(rem() * 0.15, 100.0)
        h4, c4 = optimize_smmax(h_lo, n_lo / 2.0, t0 + d4, rng, 80, 350)
        h4b, c4b = local_search_binary(h4, n_lo / 2.0, t0 + d4 + 30, rng)
        c4_min = min(c4, c4b)
        if c4_min < best_c5:
            h_short = h4b if c4b < c4 else h4
            h_short = project_box_sum(h_short, n_lo / 2.0)
            best_h, best_c5 = h_short, c4_min

    # Phase 5: Go back to n=200 if we found something at n=100
    if rem() > 40 and len(best_h) != n:
        h_up = interpolate_to_n(best_h, len(best_h), n)
        h_up = project_box_sum(h_up, target_sum)
        d5 = min(rem() * 0.4, 150.0)
        h5, c5 = optimize_smmax(h_up, target_sum, t0 + d5, rng, 150, 500)
        if c5 < best_c5:
            best_h, best_c5 = h5, c5
        else:
            best_h, best_c5 = h_up, compute_c5(h_up)

    # Phase 6: Final local search + gradient polish
    if rem() > 30:
        d6 = min(rem() * 0.4, 120.0)
        h6, c6 = local_search_binary(best_h, len(best_h) / 2.0, t0 + d6, rng)
        if c6 < best_c5:
            best_h, best_c5 = h6, c3 if False else c6

    if rem() > 15:
        d7 = t0 + budget_s - 3
        h7, c7 = optimize_smmax(best_h, len(best_h) / 2.0, d7, rng, 200, 600)
        if c7 < best_c5:
            best_h, best_c5 = h7, c7

    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
