def compute_c5(h):

    return np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n)

def grad_c5_smooth(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    m = np.max(corr)
    logits = alpha * (corr - m)
    logits -= np.max(logits)
    w = np.exp(logits)
    s = np.sum(w)
    if s < 1e-300:
        w = np.zeros(len(corr))
        w[np.argmax(corr)] = 1.0
    else:
        w /= s
    grad = np.zeros(n)
    active = np.where(w > 1e-8)[0]
    for j in active:
        k = j - (n - 1)
        if k >= 0:
            grad[:n-k] += w[j] * (1.0 - h[k:])
            grad[k:] -= w[j] * h[:n-k]
        else:
            p = -k
            grad[p:] += w[j] * (1.0 - h[:n-p])
            grad[:n-p] -= w[j] * h[p:]

    if best_h is not None and time.time() < t0 + budget_s - 10:
        n_best = len(best_h)
        dl3 = t0 + budget_s - 5
        h_opt, c5_opt = optimize_nesterov(best_h, n_best / 2.0, dl3, rng,
                                           alpha_start=150, alpha_end=800)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Ensure feasibility
    if best_h is not None:
        best_h = project_simplex(best_h, len(best_h) / 2.0)
        best_c5 = compute_c5(best_h)
        return best_h, best_c5, len(best_h)
    else:
        # Fallback
        n = 100
        h = project_simplex(0.5 + 0.2 * rng.randn(n), n / 2.0)
        c5 = compute_c5(h)
        return h, c5, n
