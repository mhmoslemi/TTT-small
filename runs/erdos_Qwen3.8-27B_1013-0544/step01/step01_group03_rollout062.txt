        if proj.sum() > target:

def subgradient(h, n_active=10):
    """Compute subgradient of max_k corr[k] w.r.t. h"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[::-1][:n_active]
    
    grad = np.zeros(n)

        lag = j - (n - 1)
        g = np.zeros(n)

                g[p:n] += -h[0:n-p]  # second term

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target):
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        proj = np.clip(h + mid, 0, 1)
        if proj.sum() > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2, 0, 1)

def smooth_max(corr, tau):
    x = tau * corr
    x_max = np.max(x)
    return float(x_max / tau + np.log(np.sum(np.exp(x - x_max))) / tau)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(120, 0.5)

    best_h = h_init.copy()
    best_c5 = evaluate_c5(best_h)
    n = len(best_h)

    def remaining():
        return budget_s - (time.time() - t0)

    # Phase 1: SLSQP from initial
    if remaining() > 30:
        h_new, c5_new = slsqp_refine(best_h, n, tau=500, maxiter=200)
        if c5_new is not None and c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 2: Binary local search
    if remaining() > 30:
        h_bin = snap_to_binary(best_h)
        h_bin, _ = binary_local_search(h_bin, max_iters=200, n_samples=3000, seed=seed)
        c5_b = evaluate_c5(h_bin)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_bin.copy()

    # Phase 3: Continuous refinement from best binary
    if remaining() > 30:
        h_soft = best_h + np.random.normal(0, 0.05, n)
        h_soft = project_feasible(h_soft, n / 2.0)
        h_new, c5_new = slsqp_refine(h_soft, n, tau=2000, maxiter=300)
        if c5_new is not None and c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 4: Iterative binary + continuous
    for rnd in range(15):
        if remaining() < 40:
            break
        h_bin = snap_to_binary(best_h)
        h_bin, _ = binary_local_search(h_bin, max_iters=100, n_samples=2000, seed=seed + rnd * 7)
        c5_b = evaluate_c5(h_bin)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_bin.copy()
        h_soft = best_h + np.random.normal(0, 0.03, n)
        h_soft = project_feasible(h_soft, n / 2.0)
        h_new, c5_new = slsqp_refine(h_soft, n, tau=3000, maxiter=200)
        if c5_new is not None and c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 5: Projected gradient fine-tuning
    if remaining() > 20:
        h_pg, c5_pg = pg_optimize(best_h, n / 2.0, max_iters=1500, lr0=0.01, momentum=0.9)
        if c5_pg < best_c5:
            best_h = h_pg
            best_c5 = c5_pg

    # Phase 6: Try other resolutions
    for tn in [100, 140]:
        if remaining() < 35:
            break
        h_t = np.interp(np.linspace(0, 1, tn), np.linspace(0, 1, n), best_h)
        h_t = project_feasible(h_t, tn / 2.0)
        h_bin_t = snap_to_binary(h_t)
        h_bin_t, _ = binary_local_search(h_bin_t, max_iters=80, n_samples=2000, seed=seed + 100)
        c5_bt = evaluate_c5(h_bin_t)
        h_new, c5_new = slsqp_refine(h_t, tn, tau=2000, maxiter=150)
        if c5_new is not None and c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
            n = tn
        elif c5_bt < best_c5:
            best_h = h_bin_t
            best_c5 = c5_bt
            n = tn

    # Phase 7: Final high-tau SLSQP
    if remaining() > 12:
        n = len(best_h)
        h_new, c5_new = slsqp_refine(best_h, n, tau=10000, maxiter=300)
        if c5_new is not None and c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    n = len(best_h)
    best_h = project_feasible(best_h, n / 2.0)
    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, n
