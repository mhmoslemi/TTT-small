  If k >= 0: grad[0:n-k] += wk * (1 - h[k:n])
  If k < 0: grad[-k:] += wk * (1 - h[:n+k])

- Second term: for j in [max(0,k), min(n, n+k)): grad[j] -= wk * h[j-k]
  If k >= 0: j ranges from k to n-1, so h[j-k] = h[0:n-k]
  If k < 0: j ranges from 0 to n-1+k, so h[j-k] = h[-k: n-k+k] = h[-k: n]
  
  So:
  If k >= 0: grad[k:] -= wk * h[0:n-k]
  If k < 0: grad[0:n+k] -= wk * h[-k:]

This is O(n) per lag, and there are O(n) lags, so total O(n^2). But with vectorized NumPy operations, it's fast.

```python
def smooth_max_and_grad(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    e = np.exp(beta * (corr - m))
    w = e / e.sum()
    val = (m + np.log(e.sum()) / beta) * dx
    
    n_full = 2 * n - 1
    lag_offset = n - 1
    grad = np.zeros(n)
    for idx in range(n_full):
        wk = w[idx]
        if wk < 1e-14:
            continue
        k = idx - lag_offset
        if k >= 0:
            # First term: grad[0:n-k] += wk * (1 - h[k:n])
            grad[:n-k] += wk * (1.0 - h[k:])
            # Second term: grad[k:] -= wk * h[0:n-k]
            grad[k:] -= wk * h[:n-k]
        else:
            kk = -k
            # First term: grad[kk:] += wk * (1 - h[:n-kk])
            grad[kk:] += wk * (1.0 - h[:n-kk])
            # Second term: grad[:n-kk] -= wk * h[kk:]
            grad[:n-kk] -= wk * h[kk:]
    grad *= dx
    return val, grad
```

Let me also consider: for the binary local search, I can make it much more efficient by precomputing the correlation and only updating the affected terms when I make a swap.

If I swap h[i] and h[j], the correlation changes only at lags |i-j| and -|i-j|. So I can update in O(n) time instead of O(n^2).

For a swap of h[i] and h[j] (where h[i]=1, h[j]=0 in a binary sequence):
- After swap: h[i]=0, h[j]=1
- The change in corr at lag k = j-i (if j > i):
  Before: h[i]*(1-h[i+k]) + h[j]*(1-h[j+k]) = 1*(1-h[j]) + 0*(1-h[i]) = 1*(1-0) + 0 = 1
  After: h[i]*(1-h[i+k]) + h[j]*(1-h[j+k]) = 0*(1-h[j]) + 1*(1-h[i]) = 0 + 1*(1-1) = 0
  Wait, this isn't right because the correlation is a sum over ALL i, not just the swapped positions.

Let me think more carefully. The correlation at lag k is:
corr_k = sum_{all valid i} h[i] * (1-h[i+k])

If I change h[i] from 1 to 0 and h[j] from 0 to 1, the changes are:
- Terms involving h[i]: 
  - As the "first" element: h[i]*(1-h[i+k]) changes from 1*(1-h[i+k]) to 0
  - As the "second" element (position i-k in the sum): h[i-k]*(1-h[i]) changes from h[i-k]*1 to h[i-k]*0
- Terms involving h[j]:
  - As the "first" element: h[j]*(1-h[j+k]) changes from 0 to 1*(1-h[j+k])
  - As the "second" element: h[j-k]*(1-h[j]) changes from h[j-k]*0 to h[j-k]*1

So the net change at lag k is:
Δcorr_k = [-(1-h[i+k]) - h[i-k]] + [(1-h[j+k]) + h[j-k]]
= (1-h[j+k] + h[j-k]) - (1-h[i+k] + h[i-k])

This is O(1) per lag! So updating all lags after a swap is O(n) (since there are O(n) lags).

This makes the binary local search very efficient: each swap evaluation is O(n), and I can try many swaps per second.

```python
def binary_local_search(h_binary, n, iterations=1000):
    """Local search on binary sequences using swap moves."""
    h = h_binary.astype(float).copy()
    corr = np.correlate(h, 1.0 - h, mode="full")
    current_max = np.max(corr)
    
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    
    for it in range(iterations):
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        
        # Compute change in max correlation
        # After swap: h[i]: 1->0, h[j]: 0->1
        new_corr = corr.copy()
        k = j - i  # relative shift
        # Update all lags affected by changing h[i] and h[j]
        # Actually, changing h[i] affects all lags (since h[i] appears in corr_k for all k)
        # Wait, h[i] appears in corr_k as:
        #   - h[i] * (1-h[i+k]) if i+k is valid (as "first" element)
        #   - h[i-k] * (1-h[i]) if i-k is valid (as "second" element)
        # So changing h[i] from 1 to 0 affects:
        #   corr_k for k where i+k is valid: change by -(1-h[i+k])
        #   corr_k for k where i-k is valid: change by +h[i-k]
        
        # This is O(n) per swap, which is fine.
        # Let me just recompute the correlation after the swap for simplicity
        # (it's only O(n^2) = 4900 for n=70, which is fast)
        
        h[i] = 0
        h[j] = 1
        new_corr = np.correlate(h, 1.0 - h, mode="full")
        new_max = np.max(new_corr)
        
        if new_max <= current_max:
            corr = new_corr
            current_max = new_max
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
        else:
            h[i] = 1
            h[j] = 0
    
    return h
```

```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2
        h_p = np.clip(h - mid, 0.0, 1.0)
        if h_p.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0.0, 1.0)

def smooth_max_and_grad(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    e = np.exp(beta * (corr - m))
    s = e.sum()
    w = e / s
    val = (m + np.log(s) / beta) * dx
    
    n_full = 2 * n - 1
    lag_offset = n - 1
    grad = np.zeros(n)
    for idx in range(n_full):
        wk = w[idx]
        if wk < 1e-14:
            continue
        k = idx - lag_offset
        if k >= 0:
            grad[:n-k] += wk * (1.0 - h[k:])
            grad[k:] -= wk * h[:n-k]
        else:
            kk = -k
            grad[kk:] += wk * (1.0 - h[:n-kk])
            grad[:n-kk] -= wk * h[kk:]
    grad *= dx
    return val, grad

def optimize_continuous(h0, n, target_sum, dx, beta, mu, maxiter):
    bounds = Bounds(np.zeros(n), np.ones(n))
    
    def fun(h):
        c = np.correlate(h, 1.0 - h, mode="full")
        return smooth_max(c, beta) * dx + mu * (h.sum() - target_sum) ** 2
    
    def grad(h):
        _, g = smooth_max_and_grad(h, n, beta)
        g += 2.0 * mu * (h.sum() - target_sum)
        return g
    
    try:
        res = minimize(fun, h0, jac=grad, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
        return project_feasible(res.x, target_sum)
    except Exception:
        return project_feasible(h0, target_sum)

def binary_local_search(h_bin, n, max_iter=500, rng=None):
    h = h_bin.astype(float).copy()
    corr = np.correlate(h, 1.0 - h, mode="full")
    cur_max = np.max(corr)
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    
    for _ in range(max_iter):
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i] = 0.0
        h[j] = 1.0
        new_corr = np.correlate(h, 1.0 - h, mode="full")
        new_max = np.max(new_corr)
        if new_max <= cur_max + 1e-12:
            corr = new_corr
            cur_max = new_max
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
        else:
            h[i] = 1.0
            h[j] = 0.0
    return h

def make_binary(n, rng):
    h = (rng.random(n) < 0.5).astype(float)
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    target = n // 2
    while len(ones) > target:
        i = int(rng.choice(ones))
        h[i] = 0
        ones.remove(i)
        zeros.append(i)
    while len(ones) < target:
        i = int(rng.choice(zeros))
        h[i] = 1
        zeros.remove(i)
        ones.append(i)
    return h

def thue_morse_seq(n):
    res = np.zeros(n)
    for i in range(n):
        c = 0
        x = i
        while x:
            c += x & 1
            x >>= 1
        res[i] = c % 2
    return res

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    start = time.time()
    
    def tleft():
        return budget_s - (time.time() - start)
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)
    
    n = len(h_init)
    target_sum = n / 2.0
    dx = 2.0 / n
    
    best_h = project_feasible(h_init, target_sum)
    best_c5 = compute_c5(best_h)
    
    # Phase 1: Quick continuous refinement
    if tleft() > 30:
        h1 = optimize_continuous(best_h, n, target_sum, dx, 50, 1000, 300)
        c5_1 = compute_c5(h1)
        if c5_1 < best_c5:
            best_h, best_c5 = h1, c5_1

        elif trial % 50 == 25:

            h_bin = (best_h > 0.5).astype(float)
            # Make sure sum is right
            ones = list(np.where(h_bin == 1)[0])
            zeros = list(np.where(h_bin == 0)[0])
            target = n // 2
            while len(ones) > target:
                i = int(rng.choice(ones))
                h_bin[i] = 0
                ones.remove(i)
                zeros.append(i)
            while len(ones) < target:
                i = int(rng.choice(zeros))
                h_bin[i] = 1
                zeros.remove(i)
                ones.append(i)
        else:
            # Thue-Morse based
            tm = thue_morse_seq(n)
            # Randomly flip a few bits
            n_flips = rng.randint(3, 10)
            for _ in range(n_flips):
                idx = int(rng.randint(n))
                tm[idx] = 1 - tm[idx]
            h_bin = tm
        
        # Binary local search
```

```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta


def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(55):
        mid = (lo + hi) / 2
        h_p = np.clip(h - mid, 0.0, 1.0)
        if h_p.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0.0, 1.0)


def smooth_max_and_grad(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    e = np.exp(beta * (corr - m))
    s = e.sum()
    w = e / s
    val = (m + np.log(s) / beta) * dx
    n_full = 2 * n - 1
    lag_offset = n - 1
    grad = np.zeros(n)
    for idx in range(n_full):
        wk = w[idx]
        if wk < 1e-14:
            continue
        k = idx - lag_offset
        if k >= 0:
            grad[:n - k] += wk * (1.0 - h[k:])
            grad[k:] -= wk * h[:n - k]
        else:
            kk = -k
            grad[kk:] += wk * (1.0 - h[:n - kk])
            grad[:n - kk] -= wk * h[kk:]
    grad *= dx
    return val, grad


def optimize_continuous(h0, n, target_sum, dx, beta, mu, maxiter):
    bounds = Bounds(np.zeros(n), np.ones(n))

    def fun(h):
        c = np.correlate(h, 1.0 - h, mode="full")
        return smooth_max(c, beta) * dx + mu * (h.sum() - target_sum) ** 2

    def grad(h):
        _, g = smooth_max_and_grad(h, n, beta)
        g += 2.0 * mu * (h.sum() - target_sum)
        return g

    try:
        res = minimize(fun, h0, jac=grad, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
        return project_feasible(res.x, target_sum)
    except Exception:
        return project_feasible(h0.copy(), target_sum)

def make_binary(n, rng):

    ones = list(np.where(h == 1)[0])

def thue_morse_seq(n):
    res = np.zeros(n)
    for i in range(n):
        c = 0
        x = i
        while x:
            c += x & 1
            x >>= 1
        res[i] = c % 2
    return res
```
