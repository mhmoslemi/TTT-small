import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1 - h, mode='full')

def compute_c5(h):
    n = len(h)
    return float(np.max(compute_corr(h)) * (2.0 / n))

def smooth_max(corr, alpha):
    m = np.max(corr)
    v = np.clip(alpha * (corr - m), -50, 50)
    return m + np.log(np.sum(np.exp(v))) / alpha

def smooth_max_weights(corr, alpha):
    m = np.max(corr)
    v = np.clip(alpha * (corr - m), -50, 50)
    w = np.exp(v)
    return w / np.sum(w)

def project_box_sum(h, target_sum):
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_smooth_max(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    w = smooth_max_weights(corr, alpha)
    grad = np.zeros(n)
    n_corr = 2 * n - 1
    for k_idx in range(n_corr):
        wk = w[k_idx]
        if wk < 1e-12:
            continue
        k = k_idx - (n - 1)
        if k >= 0:
            n1 = n - k
            grad[:n1] += wk * (1.0 - h[k:])
            if k > 0:
                grad[k:] -= wk * h[:n1]
        else:
            p = -k
            n1 = n - p
            grad[:n1] += wk * (1.0 - h[p:])
            if p > 0:
                grad[p:] -= wk * h[:n1]
    return grad * (2.0 / n)

def coord_optimize(h, i):
    n = len(h)
    # Compute A_k and B_k for coordinate i
    # A_k = (1-h[i+k])*[valid] - h[i-k]*[valid]
    # B_k = corr[k] - A_k * h[i]
    h_i = h[i]
    corr = compute_corr(h)
    n_corr = 2 * n - 1
    A = np.zeros(n_corr)
    B = np.zeros(n_corr)
    for k_idx in range(n_corr):
        k = k_idx - (n - 1)
        a = 0.0
        if 0 <= i + k < n:
            a += 1.0 - h[i + k]
        if 0 <= i - k < n:
            a -= h[i - k]
        A[k_idx] = a
        B[k_idx] = corr[k_idx] - a * h_i
    
    # Minimize max_k (A_k * x + B_k) for x in [0,1]
    # Use grid + refinement
    xs = np.linspace(0, 1, 50)
    vals = np.max(A[None, :] * xs[:, None] + B[None, :], axis=1)
    best_idx = np.argmin(vals)
    
    # Refine around best
    if best_idx == 0:
        best_x = 0.0
    elif best_idx == len(xs) - 1:
        best_x = 1.0
    else:
        lo = xs[best_idx - 1]
        hi = xs[best_idx + 1]
        # Golden section
        gr = (np.sqrt(5) - 1) / 2
        for _ in range(30):
            x1 = hi - gr * (hi - lo)
            x2 = lo + gr * (hi - lo)
            v1 = np.max(A * x1 + B)
            v2 = np.max(A * x2 + B)
            if v1 < v2:
                hi = x2
            else:
                lo = x1
            if hi - lo < 1e-12:
                break
        best_x = (lo + hi) / 2
    
    return best_x

def coordinate_descent_pass(h, rng):
    n = len(h)
    order = np.arange(n)
    rng.shuffle(order)
    for i in order:
        h[i] = coord_optimize(h, i)
    return h

def gradient_descent(h, target_sum, deadline, rng, alpha_init=50):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    alpha = alpha_init
    lr = 0.02
    stall = 0

    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_smooth_max(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    w = smooth_max_weights(corr, alpha)
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        wk = w[k_idx]
        if wk < 1e-12:
            continue
        k = k_idx - (n - 1)
        if k >= 0:
            n1 = n - k
            grad[:n1] += wk * (1.0 - h[k:])
            if k > 0:
                grad[k:] -= wk * h[:n1]
        else:
            p = -k
            n1 = n - p
            grad[:n1] += wk * (1.0 - h[p:])
            if p > 0:
                grad[p:] -= wk * h[:n1]
    return grad * (2.0 / n)

def coord_optimize(h, i, corr):
    n = len(h)
    h_i = h[i]
    n_corr = 2 * n - 1
    k0_idx = n - 1
    A = np.zeros(n_corr)
    B = np.zeros(n_corr)
    B[k0_idx] = corr[k0_idx] - h_i + h_i * h_i
    for k_idx in range(n_corr):
        if k_idx == k0_idx:
            continue
        k = k_idx - (n - 1)
        a = 0.0
        if 0 <= i + k < n:
            a += 1.0 - h[i + k]
        if 0 <= i - k < n:
            a -= h[i - k]
        A[k_idx] = a
        B[k_idx] = corr[k_idx] - a * h_i
    xs = np.linspace(0.0, 1.0, 80)
    quad = B[k0_idx] + xs - xs * xs
    lin = np.max(A[None, :] * xs[:, None] + B[None, :], axis=1)
    total = np.maximum(quad, lin)
    best_idx = int(np.argmin(total))
    if best_idx == 0:
        return 0.0
    if best_idx == len(xs) - 1:
        return 1.0
    lo = xs[best_idx - 1]
    hi = xs[best_idx + 1]
    xs_f = np.linspace(lo, hi, 150)
    quad_f = B[k0_idx] + xs_f - xs_f * xs_f
    lin_f = np.max(A[None, :] * xs_f[:, None] + B[None, :], axis=1)
    total_f = np.maximum(quad_f, lin_f)
    return float(xs_f[int(np.argmin(total_f))])

def cd_pass(h, corr, rng):
    n = len(h)
    order = np.arange(n)
    rng.shuffle(order)
    for i in order:
        x_opt = coord_optimize(h, i, corr)
        diff = x_opt - h[i]
        if abs(diff) > 1e-14:
            n_corr = 2 * n - 1
            k0_idx = n - 1
            # Update corr incrementally
            for k_idx in range(n_corr):
                if k_idx == k0_idx:
                    corr[k_idx] += diff - 2.0 * h[i] * diff - diff * diff
                else:
                    k = k_idx - (n - 1)
                    a = 0.0
                    if 0 <= i + k < n:
                        a += 1.0 - h[i + k]
                    if 0 <= i - k < n:
                        a -= h[i - k]
                    corr[k_idx] += a * diff
        h[i] = x_opt
    return h, corr

def gradient_descent(h, target_sum, deadline, rng, alpha_init=50):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    alpha = alpha_init
    lr = 0.02
    stall = 0
    while time.time() < deadline:
        grad = grad_smooth_max(best_h, alpha)
        h_new = best_h - lr * grad
        h_new = project_box_sum(h_new, target_sum)
        new_c5 = compute_c5(h_new)
        if new_c5 < best_c5 - 1e-13:
            best_h = h_new
            best_c5 = new_c5
            stall = 0
            if lr < 0.08:
                lr *= 1.04
        else:
            stall += 1
            lr *= 0.93
            if stall > 50:
                scale = min(0.01 + 0.005 * (stall // 50), 0.06)
                h_p = best_h + rng.randn(n) * scale
                h_p = project_box_sum(h_p, target_sum)
                pc5 = compute_c5(h_p)
                if pc5 < best_c5:
                    best_h = h_p
                    best_c5 = pc5
                alpha = min(alpha * 1.3, 400.0)
                lr = 0.02
                stall = 0
                if scale >= 0.055:
                    break
    return best_h, best_c5

def interp_to(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0 / n_new
    h_ext = np.concatenate([h_old, h_old[:1]])
    x_ext = np.concatenate([x_old, [2.0 + 1.0 / n_old]])
    return np.interp(x_new, x_ext, h_ext)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    remaining_total = budget_s

    if 'initial_h_values' in dir() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float)
    else:
        n_init = 100
        h = np.clip(0.5 + 0.3 * rng.randn(n_init), 0.0, 1.0)
    n = len(h)
    target_sum = n / 2.0
    h = project_box_sum(h, target_sum)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    best_n = n

    deadline = t0 + remaining_total - 10

    # Phase 1: Gradient descent (coarse optimization)
    p1_end = t0 + min(remaining_total * 0.25, 200.0)
    best_h, best_c5 = gradient_descent(best_h, target_sum, p1_end, rng, alpha_init=50)
    best_n = len(best_h)

    # Phase 2: Coordinate descent polish
    n_cd = len(best_h)
    ts = n_cd / 2.0
    h_cd = best_h.copy()
    corr_cd = compute_corr(h_cd)
    cd_deadline = min(deadline, time.time() + min(remaining_total * 0.4, 350.0))
    prev_c5 = compute_c5(h_cd)
    no_improve = 0
    while time.time() < cd_deadline:
        h_cd, corr_cd = cd_pass(h_cd, corr_cd, rng)
        h_cd = project_box_sum(h_cd, ts)
        corr_cd = compute_corr(h_cd)
        c5_now = compute_c5(h_cd)
        if c5_now < best_c5 - 1e-12:
            best_h = h_cd.copy()
            best_c5 = c5_now
            no_improve = 0
        else:
            no_improve += 1
        if abs(prev_c5 - c5_now) < 1e-13 and no_improve > 5:
            break
        if no_improve > 30:
            # Perturb and continue
            h_cd = best_h + rng.randn(n_cd) * 0.005
            h_cd = project_box_sum(h_cd, ts)
            corr_cd = compute_corr(h_cd)
            no_improve = 0
        prev_c5 = c5_now

    # Phase 3: Multi-resolution exploration
    remaining = deadline - time.time()
    if remaining > 30:
        base_n = best_n
        ts_base = base_n / 2.0
        for n_try in [80, 50, 120]:
            if time.time() > deadline - 15:
                break
            h_low = interp_to(best_h, base_n, n_try)
            h_low = project_box_sum(h_low, n_try / 2.0)
            sub_dead = time.time() + min(remaining * 0.25, 60.0)
            h_opt, c5_opt = gradient_descent(h_low, n_try / 2.0, sub_dead, rng, alpha_init=40)
            # Interpolate back
            h_back = interp_to(h_opt, n_try, base_n)
            h_back = project_box_sum(h_back, ts_base)
            # Quick CD polish
            corr_b = compute_corr(h_back)
            for _ in range(30):
                if time.time() > deadline - 10:
                    break
                h_back, corr_b = cd_pass(h_back, corr_b, rng)
                h_back = project_box_sum(h_back, ts_base)
                corr_b = compute_corr(h_back)
            c5_back = compute_c5(h_back)
            if c5_back < best_c5:
                best_h = h_back
                best_c5 = c5_back
                best_n = base_n
            remaining = deadline - time.time()

    # Phase 4: Final polish
    remaining = deadline - time.time()
    if remaining > 8:
        n_fin = len(best_h)
        ts_fin = n_fin / 2.0
        sub_dead = time.time() + remaining * 0.4
        best_h, best_c5 = gradient_descent(best_h, ts_fin, sub_dead, rng, alpha_init=120)
        if time.time() < deadline - 3:
            h_cd2 = best_h.copy()
            corr2 = compute_corr(h_cd2)
            p2_dead = min(deadline - 1, time.time() + (deadline - time.time()))
            while time.time() < p2_dead:
                h_cd2, corr2 = cd_pass(h_cd2, corr2, rng)
                h_cd2 = project_box_sum(h_cd2, ts_fin)
                corr2 = compute_corr(h_cd2)
                c5_2 = compute_c5(h_cd2)
                if c5_2 < best_c5 - 1e-13:
                    best_h = h_cd2.copy()
                    best_c5 = c5_2
                else:
                    break

    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
