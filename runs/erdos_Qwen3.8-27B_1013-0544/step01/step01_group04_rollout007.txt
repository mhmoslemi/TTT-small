    return obj, grad * dx

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - target
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(free):
            h[free] -= diff / np.sum(free)
        else:
            h -= diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = np.sum(h) - target
    if abs(diff) > 1e-14:
        free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if len(free) > 0:
            h[free] -= diff / len(free)
            h = np.clip(h, 0.0, 1.0)
    return h

def lse_obj_grad(h, dx, epsilon):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = epsilon * corr * dx
    v_max = vals.max()
    shifted = vals - v_max
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    obj = (v_max + np.log(sum_exp)) / epsilon
    p = exp_shifted / sum_exp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lag = k - (n - 1)
        j1 = max(0, -lag)
        j2 = min(n - 1, n - 1 - lag)
        if j1 <= j2:
            grad[j1:j2+1] += pk * (1.0 - h[j1+lag:j2+lag+1])
        j3 = max(0, lag)
        j4 = min(n - 1, n - 1 + lag)
        if j3 <= j4:
            grad[j3:j4+1] -= pk * h[j3-lag:j4-lag+1]
    return obj, grad * dx

def lse_slsqp(h_start, n, dx, epsilon, max_iter=300, ftol=1e-13):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    def fun(x):
        return lse_obj_grad(x, dx, epsilon)[0]
    def jac(x):
        return lse_obj_grad(x, dx, epsilon)[1]
    res = minimize(fun, h_start, jac=jac, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': ftol})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_feasible(res.x, n)
    return project_feasible(h_start, n)

def pg_step(h, dx, epsilon, step_size, n):
    obj, grad = lse_obj_grad(h, dx, epsilon)
    h_new = h - step_size * grad
    h_new = project_feasible(h_new, n)
    return h_new

    t0 = time.time()
