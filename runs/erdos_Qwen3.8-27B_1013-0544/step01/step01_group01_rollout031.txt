For L > 0:

For L < 0, let M = -L > 0:

        # -h[j+M] for j in [0, n-1-M]
        grad[:n-M] -= h[M:]
        grad *= dx
    return grad

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr_values(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def smooth_max(x, beta):
    m = np.max(x)
    shifted = beta * (x - m)
    # Clip to avoid overflow
    shifted = np.clip(shifted, -50, 50)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_of_lag(h, lag):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    if lag == 0:
        return (1.0 - 2.0 * h) * dx
    elif lag > 0:
        grad[:n-lag] += (1.0 - h[lag:])
        grad[lag:] -= h[:n-lag]
    else:
        m = -lag
        grad[m:] += (1.0 - h[:n-m])
        grad[:n-m] -= h[m:]
    return grad * dx

def active_set_descent(h, target_sum, max_iter=500, tol=1e-12):

    h = project_feasible(h.copy(), target_sum)
    best_c5 = compute_c5(h)
    h_best = h.copy()
    
    for it in range(max_iter):
        corrs = compute_corr_values(h)
        c5 = np.max(corrs)

            best_c5 = c5
            h_best = h.copy()

        threshold = c5 - max(1e-8, 0.01 * c5)
        active_mask = corrs > threshold

            active_idx = np.argmax(corrs)
            active_mask = np.zeros(len(corrs), dtype=bool)
            active_mask[active_idx] = True
        
        # Combined gradient (average over active lags for stability)
        lags = np.where(active_mask)[0] - (n - 1)
        grad = np.zeros(n)
        for lag in lags:
            grad += grad_of_lag(h, int(lag))
        grad /= len(lags)

        grad -= np.mean(grad)

        improved = False

                break

                if c5_new < c5:

    return np.sum(x) - target_sum

def con_jac_sum(x):
    n = len(x)

                h = np.zeros(n)
                idx = np.random.choice(n, n // 2, replace=False)
                h[idx] = 1.0
                h += np.random.randn(n) * 0.02
            else:
                # Structured: alternating blocks
                block = n // 10
                h = np.tile(np.array([0.7, 0.3, 0.6, 0.4, 0.55, 0.45, 0.65, 0.35, 0.5, 0.5]), n // 10 + 1)[:n]
                h += np.random.randn(n) * 0.03

            h = project_feasible(h, target_sum)

            if time_left() > 15:
                h_as, _ = active_set_descent(h, target_sum, max_iter=150)
                h = h_as

            if time_left() > 12:
                try:
                    h_new = slsqp_polish(h, target_sum, 150, 500)
                    c5_new = compute_c5(h_new)
                    c5_cur = compute_c5(h)
                    if c5_new < c5_cur:
                        h = h_new
                except Exception:
                    pass

            c5 = compute_c5(h)
            if c5 < best_c5:
                best_h, best_c5, best_n = h.copy(), c5, n

    # Phase 3: Intensive polish on best solution
    if best_h is not None and time_left() > 15:
        n = best_n
        target_sum = n / 2.0
        for polish in range(4):
            if time_left() < 8:
                break
            h_as, c5_as = active_set_descent(best_h, target_sum, max_iter=200)
            c5 = compute_c5(h_as)
            if c5 < best_c5:
                best_h, best_c5 = h_as.copy(), c5
            if time_left() > 8:
                try:
                    beta = 300 + 200 * polish
                    h_new = slsqp_polish(best_h, target_sum, beta, 1500)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_h, best_c5 = h_new.copy(), c5
                except Exception:
                    pass

    best_h = project_feasible(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
