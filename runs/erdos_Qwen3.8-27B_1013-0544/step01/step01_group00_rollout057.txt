    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)

    g = np.zeros(n)
    for m_idx in active:
        s = m_idx - (n - 1)
        # term1: (1-h[i+s]) for valid i
        if s >= 0:
            term1 = np.zeros(n)
            term1[:n-s] = 1.0 - h[s:]
        else:
            term1 = np.zeros(n)
            term1[-s:] = 1.0 - h[:n+s]
        # term2: h[i-s] for valid i
        if s >= 0:
            term2 = np.zeros(n)
            term2[s:] = h[:n-s]
        else:
            term2 = np.zeros(n)
            term2[:n+s] = h[-s:]
        g += dx * (term1 - term2)
    g /= len(active)
    return g, m, active

    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))


def project_h(h, target):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = np.sum(mask)
    if cnt == 0:
        return h
    h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)


def t_objective(v):
    return v[-1]


def t_ineq(v):
    n = len(v) - 1
    h = v[:n]
    t = v[-1]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return t - corr


def t_eq_fun(v):
    n = len(v) - 1
    return np.sum(v[:n]) - n * 0.5


def t_eq_jac(v):
    n = len(v) - 1
    j = np.zeros(len(v))
    j[:n] = 1.0
    return j


def optimize_t_var(h0, max_iter=300, ftol=1e-10):
    n = len(h0)
    c5_cur = compute_c5(h0)
    v0 = np.append(h0.copy(), c5_cur + 0.01)
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.5)]
    cons = [
        {'type': 'ineq', 'fun': t_ineq},
        {'type': 'eq', 'fun': t_eq_fun, 'jac': t_eq_jac}
    ]
    res = minimize(t_objective, v0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': ftol})
    h_new = np.clip(res.x[:n], 0.0, 1.0)
    h_new = project_h(h_new, n * 0.5)
    return h_new, compute_c5(h_new)


def smooth_obj(h, temp):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip((corr - m) / temp, -500, 0)
    sse = np.sum(np.exp(shifted))
    return m + temp * np.log(max(sse, 1e-300))


def sum_eq_fun(h):
    return np.sum(h) - len(h) * 0.5


def sum_eq_jac(h):
    return np.ones(len(h))


def optimize_smooth(h0, temp, max_iter=200, ftol=1e-12):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac}]
    res = minimize(smooth_obj, h0.copy(), args=(temp,), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': ftol})
    h_new = np.clip(res.x, 0.0, 1.0)
    h_new = project_h(h_new, n * 0.5)
    return h_new, compute_c5(h_new)


def interp_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.arange(n_old)
    x_new = np.linspace(0, n_old - 1, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_h(h_new, n_new * 0.5)


def fix_binary_sum(h_bin, target):
    h_bin = h_bin.copy()
    ones = int(np.sum(h_bin == 1.0))
    if ones > target:
        idx = np.where(h_bin == 1.0)[0]
        to_rem = idx[:ones - target]
        h_bin[to_rem] = 0.0
    elif ones < target:
        idx = np.where(h_bin == 0.0)[0]
        to_add = idx[:target - ones]
        h_bin[to_add] = 1.0
    return h_bin


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - t0)

    # Initialize
    h = np.array(initial_h_values, dtype=float)
    n = len(h)
    h = project_h(h, n * 0.5)
    c5 = compute_c5(h)
    h_best = h.copy()

    # Phase 1: t-variable SLSQP polish
    if time_left() > 150:
        h_new, c5_new = optimize_t_var(h, max_iter=500, ftol=1e-10)
        if c5_new < c5:
            h_best = h_new.copy()
            c5 = c5_new

    # Phase 2: Mixed binary/continuous simulated annealing
    if time_left() > 350:
        # Start from thresholded best
        h_sa = (h_best > 0.4).astype(float)
        h_sa = fix_binary_sum(h_sa, n // 2)
        c5_sa = compute_c5(h_sa)
        h_sa_best = h_sa.copy()
        c5_sa_best = c5_sa

        # Also track continuous best from SA
        h_cont_best = h_best.copy()
        c5_cont_best = c5

        T = 0.008
        T_min = 0.00003
        n_iter = 150000
        cool = (T_min / T) ** (1.0 / n_iter)
        refine_every = 15000

        for it in range(n_iter):
            if time_left() < 200:
                break

            # Choose move type
            if rng.random() < 0.55:
                # Discrete swap
                i = rng.randint(n)
                j = rng.randint(n)
                if h_sa[i] == h_sa[j]:
                    T *= cool
                    continue
                h_sa[i], h_sa[j] = h_sa[j], h_sa[i]
                c5_new = compute_c5(h_sa)
                delta = c5_new - c5_sa
                if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
                    c5_sa = c5_new
                    if c5_sa < c5_sa_best:
                        c5_sa_best = c5_sa
                        h_sa_best = h_sa.copy()
                else:
                    h_sa[i], h_sa[j] = h_sa[j], h_sa[i]
            else:
                # Continuous perturbation
                n_pert = rng.randint(2, 8)
                idx = rng.choice(n, size=n_pert, replace=False)
                h_cont = h_sa_best + rng.randn(n_pert) * 0.15
                h_cont = np.clip(h_cont, 0.0, 1.0)
                h_cont_full = h_sa_best.copy()
                h_cont_full[idx] = h_cont
                h_cont_full = project_h(h_cont_full, n * 0.5)
                c5_new = compute_c5(h_cont_full)
                delta = c5_new - c5_cont_best
                if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
                    c5_cont_best = c5_new
                    h_cont_best = h_cont_full.copy()

            T *= cool

            # Periodic SLSQP refinement
            if (it + 1) % refine_every == 0 and time_left() > 300:
                # Refine continuous best
                h_ref, c5_ref = optimize_t_var(h_cont_best, max_iter=250, ftol=1e-9)
                if c5_ref < c5_cont_best:
                    c5_cont_best = c5_ref
                    h_cont_best = h_ref.copy()
                # Update binary from continuous
                h_sa = (h_cont_best > 0.4).astype(float)
                h_sa = fix_binary_sum(h_sa, n // 2)
                c5_sa = compute_c5(h_sa)

        # Take the best from SA phase
        if c5_sa_best < c5:
            c5 = c5_sa_best
            h_best = h_sa_best.copy()
        if c5_cont_best < c5:
            c5 = c5_cont_best
            h_best = h_cont_best.copy()

    # Phase 3: Subgradient + projection (helps escape shallow local minima)
    if time_left() > 200:
        h_sg = h_best.copy()
        c5_sg = compute_c5(h_sg)
        best_sg = c5_sg
        best_sg_h = h_sg.copy()
        n_sg_steps = 3000
        for step in range(n_sg_steps):
            if time_left() < 150:
                break
            # Compute subgradient
            dx = 2.0 / n
            corr = np.correlate(h_sg, 1.0 - h_sg, mode='full') * dx
            m = np.max(corr)
            active = np.where(np.abs(corr - m) < 1e-7)[0]
            if len(active) == 0:
                active = [int(np.argmax(corr))]
            g = np.zeros(n)
            for midx in active:
                s = midx - (n - 1)
                term1 = np.zeros(n)
                term2 = np.zeros(n)
                if s >= 0:
                    term1[:n - s] = 1.0 - h_sg[s:]
                    term2[s:] = h_sg[:n - s]
                else:
                    term1[-s:] = 1.0 - h_sg[:n + s]
                    term2[:n + s] = h_sg[-s:]
                g += dx * (term1 - term2)
            g /= len(active)
            alpha = 0.005 / (1.0 + step * 0.0005)
            h_sg = project_h(h_sg - alpha * g, n * 0.5)
            c5_sg = compute_c5(h_sg)
            if c5_sg < best_sg:
                best_sg = c5_sg
                best_sg_h = h_sg.copy()

        if best_sg < c5:
            c5 = best_sg
            h_best = best_sg_h.copy()

    # Phase 4: Smooth SLSQP refinement
    for temp in [0.004, 0.002, 0.001, 0.0004]:
        if time_left() < 70:
            break
        h_new, c5_new = optimize_smooth(h_best, temp, max_iter=250, ftol=1e-12)
        if c5_new < c5:
            h_best = h_new.copy()
            c5 = c5_new

    # Phase 5: Final t-variable polish
    if time_left() > 50:
        h_new, c5_new = optimize_t_var(h_best, max_iter=600, ftol=1e-11)
        if c5_new < c5:
            h_best = h_new.copy()
            c5 = c5_new

    # Phase 6: Resolution increase
    for n_new in [240, 360]:
        if time_left() < 120:
            break
        h_interp = interp_h(h_best, n_new)
        c5_cur = compute_c5(h_interp)
        h_cur = h_interp.copy()

        h_new, c5_new = optimize_t_var(h_cur, max_iter=350, ftol=1e-10)
        if c5_new < c5:
            h_cur = h_new.copy()
            c5 = c5_new
            h_best = h_new.copy()

        for temp in [0.003, 0.001]:
            if time_left() < 80:
                break
            h_new, c5_new = optimize_smooth(h_cur, temp, max_iter=200, ftol=1e-12)
            if c5_new < c5:
                h_cur = h_new.copy()
                c5 = c5_new
                h_best = h_new.copy()

        if time_left() > 60:
            h_new, c5_new = optimize_t_var(h_cur, max_iter=400, ftol=1e-11)
            if c5_new < c5:
                h_best = h_new.copy()
                c5 = c5_new

    # Final validation
    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = project_h(h_best, len(h_best) * 0.5)
    c5_final = compute_c5(h_best)
    return h_best, c5_final, len(h_best)
