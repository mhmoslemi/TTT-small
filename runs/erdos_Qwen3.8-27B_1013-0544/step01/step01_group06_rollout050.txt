import numpy as np
import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, 'full')
    return float(np.max(corr) * (2.0 / n))

def project(h, target_sum):
    lo, hi = -1.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, 'full')
    m = np.max(corr)
    z = alpha * (corr - m)
    e = np.exp(z)
    se = np.sum(e)
    if se < 1e-300:
        obj = m * (2.0 / n)
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        obj = (m + np.log(se) / alpha) * (2.0 / n)
        w = e / se
    grad = np.zeros(n)
    active = w > 1e-12
    for idx in np.where(active)[0]:
        k = int(idx) - (n - 1)
        wi = w[idx]
        if k >= 0:
            e2 = n - k
            grad[:e2] += wi * (1.0 - h[k:e2])
            if k > 0:
                grad[k:] -= wi * h[:e2]
        else:
            p = -k
            grad[p:] += wi * (1.0 - h[:n - p])
            if p > 0:
                grad[:n - p] -= wi * h[p:]
    grad *= (2.0 / n)
    return obj, grad

def exact_subgrad(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, 'full')
    max_val = np.max(corr)
    active = np.where(corr >= max_val - 1e-10)[0]
    grad = np.zeros(n)
    for idx in active:
        k = int(idx) - (n - 1)
        if k >= 0:
            e2 = n - k
            grad[:e2] += (1.0 - h[k:e2])
            if k > 0:
                grad[k:] -= h[:e2]
        else:
            p = -k
            grad[p:] += (1.0 - h[:n - p])
            if p > 0:
                grad[:n - p] -= h[p:]
    grad /= len(active)
    grad *= (2.0 / n)
    return grad

def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    pos = (np.arange(n_new) + 0.5) * n_old / n_new
    idx0 = np.floor(pos).astype(int) % n_old
    idx1 = (idx0 + 1) % n_old
    frac = pos - np.floor(pos)
    return h_old[idx0] * (1.0 - frac) + h_old[idx1] * frac

def adam_phase(h, target_sum, deadline, lr, alpha, max_iter=50000):
    n = len(h)
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t = 0
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    stall = 0
    while time.time() < deadline:
        t += 1
        obj, grad = obj_grad(h, alpha)
        m_adam = 0.9 * m_adam + 0.1 * grad
        v_adam = 0.999 * v_adam + 0.001 * grad * grad
        m_hat = m_adam / (1.0 - 0.9 ** t)
        v_hat = v_adam / (1.0 - 0.999 ** t)
        h_new = project(h - lr * m_hat / (np.sqrt(v_hat) + 1e-8), target_sum)
        c5 = compute_c5(h_new)
        if c5 < best_c5 - 1e-15:
            best_h = h_new.copy()
            best_c5 = c5
            stall = 0
        else:
            stall += 1
            if stall > 200:
                h += np.random.randn(n) * 0.005
                h = project(h, target_sum)
                m_adam = np.zeros(n)
                v_adam = np.zeros(n)
                t = 0
                stall = 0
                if compute_c5(h) > best_c5 + 0.01:
                    h = best_h.copy()
                    break
        h = h_new
        if t >= max_iter:
            break
    return best_h, best_c5

def exact_subgrad_phase(h, target_sum, deadline, lr_init=0.005, max_iter=30000):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    lr = lr_init
    stall = 0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        grad = exact_subgrad(best_h)
        h_new = project(best_h - lr * grad, target_sum)
        c5 = compute_c5(h_new)
        if c5 < best_c5 - 1e-15:
            best_h = h_new
            best_c5 = c5
            lr = min(lr * 1.05, 0.05)
            stall = 0
        else:
            stall += 1
            lr *= 0.9
            if stall > 50:
                break
    return best_h, best_c5

def coord_descent(h, target_sum, deadline, rng, n_iter=5000):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    deltas = [0.002, 0.005, 0.01, 0.02, -0.002, -0.005, -0.01, -0.02]
    for _ in range(n_iter):
        if time.time() > deadline:
            break
        j = rng.randint(n)
        best_local = best_h[j]
        best_local_c5 = best_c5
        for d in deltas:
            candidate = best_h.copy()
            candidate[j] = best_h[j] + d
            candidate = project(candidate, target_sum)
            c5 = compute_c5(candidate)
            if c5 < best_local_c5:
                best_local_c5 = c5
                best_local = candidate[j]
                best_h = candidate
        best_c5 = best_local_c5
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    np.random.seed(seed)

    if 'initial_h_values' in dir() or 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, dtype=float)
    else:
        n = 200
        h_init = 0.5 + 0.2 * rng.randn(n)
        h_init = project(h_init, n / 2.0)

    n = len(h_init)
    target_sum = n / 2.0
    best_h = project(h_init.copy(), target_sum)
    best_c5 = compute_c5(best_h)

    # Phase 1: Adam optimization with annealed alpha
    phases = [
        (300, 0.05),
        (100, 0.03),
        (300, 0.02),
        (800, 0.01),
    ]
    elapsed = time.time() - t0
    remaining = budget_s - elapsed - 120
    time_per_phase = remaining / len(phases)

    for i, (alpha, lr) in enumerate(phases):
        phase_deadline = time.time() + time_per_phase
        best_h, best_c5 = adam_phase(best_h, target_sum, phase_deadline, lr, alpha)

    # Phase 2: Exact subgradient polish
    remaining = budget_s - (time.time() - t0)
    if remaining > 60:
        best_h, best_c5 = exact_subgrad_phase(best_h, target_sum,
                                               time.time() + min(remaining * 0.4, 200),
                                               lr_init=0.003)

    # Phase 3: Coordinate descent
    remaining = budget_s - (time.time() - t0)
    if remaining > 40:
        best_h, best_c5 = coord_descent(best_h, target_sum,
                                        time.time() + min(remaining * 0.5, 150), rng)

    # Phase 4: Final Adam polish with high alpha
    remaining = budget_s - (time.time() - t0)
    if remaining > 30:
        best_h, best_c5 = adam_phase(best_h, target_sum,
                                      time.time() + min(remaining * 0.6, 100),
                                      lr=0.005, alpha=1500)

    # Phase 5: One more exact subgradient pass
    remaining = budget_s - (time.time() - t0)
    if remaining > 15:
        best_h, best_c5 = exact_subgrad_phase(best_h, target_sum,
                                               time.time() + min(remaining * 0.7, 60),
                                               lr_init=0.002)

    # Try a perturbed restart if time remains
    remaining = budget_s - (time.time() - t0)
    if remaining > 80:
        h_pert = best_h + rng.randn(len(best_h)) * 0.02
        h_pert = project(h_pert, target_sum)
        h_try, c5_try = adam_phase(h_pert, target_sum,
                                    time.time() + min(remaining * 0.5, 200),
                                    lr=0.03, alpha=200)
        if c5_try < best_c5:
            best_h, best_c5 = h_try, c5_try
            remaining2 = budget_s - (time.time() - t0)
            if remaining2 > 20:
                best_h, best_c5 = exact_subgrad_phase(best_h, target_sum,
                                                       time.time() + min(remaining2 * 0.7, 60),
                                                       lr_init=0.003)

    best_h = project(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
