Also, I should be more careful about the gradient. Let me verify:

For `np.correlate(h, 1-h, mode="full")`, the output is:
output[k] = sum_m h[m] * (1-h)[m + k - (len(1-h)-1)]

Wait, actually numpy's correlate is defined as:
(a * v)[k] = sum_m a[m+k] * conj(v[m])

So `np.correlate(h, 1-h)` computes:
result[k] = sum_m h[m+k] * (1-h)[m]

For mode="full", k ranges from -(len(1-h)-1) to len(h)-1, i.e., from -(n-1) to n-1.
The output array has length 2n-1, and the index in the array corresponds to:
array[idx] = result[idx - (n-1)] for idx = 0, ..., 2n-2

So array[idx] = sum_m h[m + idx - (n-1)] * (1-h)[m]

For idx = n-1 + lag (i.e., lag = idx - (n-1)):
array[idx] = sum_m h[m + lag] * (1-h)[m]

This is valid when 0 <= m <= n-1 and 0 <= m+lag <= n-1.

If lag >= 0: m ranges from 0 to n-1-lag
array[idx] = sum_{m=0}^{n-1-lag} h[m+lag] * (1-h[m])

If lag < 0, let la = -lag: m ranges from la to n-1
array[idx] = sum_{m=la}^{n-1} h[m-la] * (1-h[m])

Now the gradient:
For lag >= 0:
∂/∂h[j] of sum_{m=0}^{n-1-lag} h[m+lag] * (1-h[m]):
- h[j] appears as h[m+lag] when m = j-lag, valid when 0 <= j-lag <= n-1-lag, i.e., lag <= j <= n-1
  Contribution: (1-h[j-lag])
- h[j] appears as (1-h[m]) when m = j, valid when 0 <= j <= n-1-lag
  Contribution: -h[j+lag]

So: ∂/∂h[j] = (1-h[j-lag]) * [lag <= j <= n-1] - h[j+lag] * [0 <= j <= n-1-lag]

For lag < 0, la = -lag:
array[idx] = sum_{m=la}^{n-1} h[m-la] * (1-h[m])
∂/∂h[j]:
- h[j] appears as h[m-la] when m = j+la, valid when la <= j+la <= n-1, i.e., 0 <= j <= n-1-la
  Contribution: (1-h[j+la])
- h[j] appears as (1-h[m]) when m = j, valid when la <= j <= n-1
  Contribution: -h[j-la]

So: ∂/∂h[j] = (1-h[j+la]) * [0 <= j <= n-1-la] - h[j-la] * [la <= j <= n-1]

In vectorized form:

- grad[la:] += (-h[:n-la])  [from second term, j-la ranges from 0 to n-1-la]

import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def get_correlations(h):
    return np.correlate(h, 1.0 - h, mode="full")


def smooth_max_val(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta


def smooth_max_weights(x, beta):
    m = np.max(x)
    e = np.exp(beta * (x - m))
    return e / np.sum(e)


def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) > 1e-12 and s > 0:
        h = h + (target_sum - s) / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_smooth_max_c5(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = get_correlations(h)
    weights = smooth_max_weights(corr, beta)
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        w = weights[idx] * dx
        if w < 1e-14:
            continue
        lag = idx - (n - 1)
        if lag >= 0:
            if n - lag > 0:
                grad[lag:] += w * (1.0 - h[:n - lag])
                grad[:n - lag] += w * (-h[lag:n])
        else:
            la = -lag
            if n - la > 0:
                grad[:n - la] += w * (1.0 - h[la:n])
                grad[la:] += w * (-h[:n - la])
    return grad


def gradient_descend(h, n, target_sum, max_iters, beta, start_time, budget_s):
    h_cur = h.copy()
    best_local = compute_c5(h_cur)
    best_local_h = h_cur.copy()
    lr = 0.15
    stall = 0
    for it in range(max_iters):
        if budget_s - (time.time() - start_time) < 5:
            break
        grad = grad_smooth_max_c5(h_cur, beta)
        grad_proj = grad - np.mean(grad)
        h_new = h_cur - lr * grad_proj
        h_new = project_h(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_local - 1e-10:
            best_local = c5_new
            best_local_h = h_new.copy()
            h_cur = h_new
            lr = min(lr * 1.05, 0.4)
            stall = 0
        else:
            h_cur = h_new
            lr *= 0.8
            stall += 1
            if stall > 20 or lr < 1e-7:
                break
    return best_local_h, best_local


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)

    n_points = len(h_init)
    best_h = project_h(h_init, n_points / 2.0)
    best_c5 = compute_c5(best_h)

    def time_left():
        return budget_s - (time.time() - start_time)

    # Phase 1: Gradient descent (multiple beta values)
    for beta in [50.0, 100.0, 200.0, 400.0]:
        if time_left() < 10:
            break
        h_gd, c5_gd = gradient_descend(best_h, n_points, n_points / 2.0,
                                       400, beta, start_time, budget_s)
        if c5_gd < best_c5:
            best_h, best_c5 = h_gd, c5_gd

    # Phase 2: SLSQP with smooth max
    def make_obj(dx_val, beta_val):
        def obj(h):
            corr = get_correlations(h)
            return smooth_max_val(corr, beta_val) * dx_val
        return obj

    def make_con_fun(ts):
        def cf(x):
            return np.sum(x) - ts
        return cf

    def make_con_jac(n):
        def cj(x):
            return np.ones(n)
        return cj

    dx = 2.0 / n_points
    bounds = Bounds(np.zeros(n_points), np.ones(n_points))

    for beta in [100.0, 300.0]:
        if time_left() < 10:
            break
        obj_fn = make_obj(dx, beta)

            res = minimize(obj_fn, best_h, method='SLSQP', bounds=bounds,

                           options={'maxiter': 500, 'ftol': 1e-14})
            h_new = project_h(res.x, n_points / 2.0)

                else:

                        grad_dir[la:] += w * (-h_work[:n_points - la])

            grad_dir -= np.mean(grad_dir)
            step = 0.08 * (1.0 - sub * 0.01)

            h_try = project_h(h_try, n_points / 2.0)
            c5_try = compute_c5(h_try)
            if c5_try < compute_c5(h_work):

            else:

                h_try = project_h(h_try, n_points / 2.0)
                if compute_c5(h_try) < compute_c5(h_work):
                    h_work = h_try

        c5_work = compute_c5(h_work)

        x_old = np.linspace(0, 2.0, n_points, endpoint=False) + dx / 2

        h_cand = project_h(h_cand, cn / 2.0)

            dx = 2.0 / cn
            bounds = Bounds(np.zeros(n_points), np.ones(n_points))

    # Phase 5: Final polish - alternate gradient descent and SLSQP
    for it in range(10):

    return best_h, best_c5, n_points
