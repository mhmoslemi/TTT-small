import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    return float(m + np.log(np.sum(np.exp(beta * (x - m)))) / beta)

def project_feasible(h, target_sum):
    n = len(h)
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def slsqp_optimize(h, n, target_sum, beta, maxiter, ftol, time_limit):
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        return smooth_max(corr, beta) * dx
    
    def csum(x):
        return x.sum() - target_sum
    
    def csum_jac(x):
        return np.ones(n)
    
    res = minimize(obj, h, method='SLSQP', bounds=bounds,
                  constraints=[{'type': 'eq', 'fun': csum, 'jac': csum_jac}],
                  options={'maxiter': maxiter, 'ftol': ftol})
    h_new = project_feasible(res.x, target_sum)
    return h_new, compute_c5(h_new)

5. Choose the t that minimizes the max

Finding the upper envelope: I can just evaluate all 2n-1 lines at 2n-2 candidate t values and keep track of the max. But I need to know the t values first...

Alternative simple approach: Since the function is convex (max of linear functions is convex), I can use ternary search or binary search on the derivative!

The derivative of max_k (C_k + t*g_k) with respect to t is:
- At any t, the derivative is g_k* where k* is the argmax (or a convex combination if there's a tie)
- The derivative is non-decreasing in t (since the max of lines has an increasing left-derivative)

So I can binary search for the point where the derivative crosses 0:
- If at t, the active line has slope < 0, the minimum is to the right
- If the active line has slope > 0, the minimum is to the left
- If there's a tie with both positive and negative slopes, we're at a vertex (minimum)

This gives O(log(precision) * n) per coordinate, which is very fast!

Let me implement this:

```python
def min_max_lines(C, g, lb=0.0, ub=1.0, iters=50):
    """Find t in [lb, ub] that minimizes max_k (C[k] + t*g[k])"""
    lo, hi = lb, ub
    for _ in range(iters):
        mid = (lo + hi) / 2.0
        vals = C + mid * g
        kmin = np.argmin(vals)  # Wait, we want to minimize the max
        # Actually: f(t) = max(vals). f'(t) from the left = min(g[k] for k in argmax)
        # f'(t) from the right = max(g[k] for k in argmax)
        max_val = np.max(vals)
        active = np.where(vals >= max_val - 1e-12)[0]
        slope_left = np.min(g[active])
        slope_right = np.max(g[active])
        
        if slope_left > 0:
            hi = mid
        elif slope_right < 0:
            lo = mid
        else:
            # We're at a vertex (slope changes from negative to positive)
            break
    
    return (lo + hi) / 2.0
```

This is O(iters * n) = O(50 * 139) ≈ 7000 ops per coordinate. Per sweep: 70 * 7000 = 490,000 ops. Very fast!

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    return float(m + np.log(np.sum(np.exp(beta * (x - m)))) / beta)

def project_feasible(h, target_sum):
    n = len(h)
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def min_plot_lines(C, g, lb=0.0, ub=1.0):
    """Minimize max_k(C[k] + t*g[k]) over t in [lb, ub]"""
    lo, hi = lb, ub
    for _ in range(60):

        vals = C + mid * g
        max_val = np.max(vals)
        active = np.where(vals >= max_val - 1e-10)[0]

        slope_right = np.max(g[active])
        if slope_left > 1e-12:
            hi = mid
        elif slope_right < -1e-12:
            lo = mid
        else:
            break

    best_t = t

        if lb <= cand <= ub:
            v = np.max(C + cand * g)
            if v < best_val:
                best_val = v
                best_t = cand
    return best_t, float(best_val)

def coordinate_descent_step(h, n, target_sum):
    """One full sweep of coordinate descent"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    current_c5 = float(np.max(corr) * dx)
    
    for i in range(n):
        # Compute coefficient of h[i] in each corr[k]
        # corr[k] = sum_j h[j]*(1-h[j+k-(n-1)])
        # h[i] appears when j=i (coeff: 1-h[i+lag]) or j+k-(n-1)=i i.e. j=i-lag (coeff: -h[i-lag])
        # where lag = k-(n-1)
        
        lags = np.arange(-(n-1), n)  # all possible lags, length 2n-1
        # For each lag, the coefficient of h[i] in corr[lag + (n-1)]:
        # term 1: j=i, valid if 0 <= i+lag < n, contrib: (1-h[i+lag])
        # term 2: j=i-lag, valid if 0 <= i-lag < n, contrib: -h[i-lag]
        
        valid1 = (i + lags >= 0) & (i + lags < n)
        valid2 = (i - lags >= 0) & (i - lags < n)
        
        g = np.zeros(2*n - 1)
        g[valid1] += 1.0 - h[i + lags[valid1]]
        g[valid2] -= h[i - lags[valid2]]
        
        # C_k = corr[k] - h[i] * g[k]
        C = corr - h[i] * g
        
        # Find optimal t
        t_opt, val_opt = min_plot_lines(C, g, 0.0, 1.0)
        
        if abs(t_opt - h[i]) > 1e-12:
            h[i] = t_opt
            h = project_feasible(h, target_sum)
            # Recompute corr
            corr = np.correlate(h, 1.0 - h, mode="full")
    
    new_c5 = compute_c5(h)
    return h, new_c5

            corr = np.correlate(x, 1.0 - x, mode="full")

      - Evaluating at the critical point of the quadratic (t=0.5)

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    return float(m + np.log(np.sum(np.exp(beta * (x - m)))) / beta)

def project_feasible(h, target_sum):
    n = len(h)
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _obj_1d(h, i, t, n):
    """Objective with h[i] replaced by t, returns max correlation * dx"""
    h[0] = t
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _cd_sweep(h, n, target_sum, n_sweeps=10):
    """Coordinate descent with golden section search"""
    work = h.copy()
    dx = 2.0 / n
    for _ in range(n_sweeps):
        improved = False
        for i in range(n):
            old_val = work[i]
            # Golden section search on [0,1] for work[i]
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a, b = 0.0, 1.0
            c = b - gr * (b - a)
            d = a + gr * (b - a)
            work[i] = c
            fc = _obj_1d(work, i, c, n)
            work[i] = d
            fd = _obj_1d(work, i, d, n)
            for _ in range(40):
                if fc < fd:
                    b = d
                    d = c
                    fd = fc
                    c = b - gr * (b - a)
                    work[i] = c
                    fc = _obj_1d(work, i, c, n)
                else:
                    a = c
                    c = d
                    fc = fd
                    d = a + gr * (b - a)
                    work[i] = d
                    fd = _obj_1d(work, i, d, n)
            t_opt = (a + b) / 2.0
            work[i] = t_opt
            work = project_feasible(work, target_sum)
            if abs(work[i] - old_val) > 1e-10:
                improved = True
        if not improved:
            break
    return work, compute_c5(work)

def _slsqp(h, n, target_sum, beta, maxiter, ftol):
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        return smooth_max(corr, beta) * dx
    def csum(x):
        return x.sum() - target_sum
    def csum_j(x):
        return np.ones(n)
    res = minimize(obj, h, method='SLSQP', bounds=bounds,
                  constraints=[{'type': 'eq', 'fun': csum, 'jac': csum_j}],
                  options={'maxiter': maxiter, 'ftol': ftol})
    h_new = project_feasible(res.x, target_sum)
    return h_new, compute_c5(h_new)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)

    n = len(h_init)
    target_sum = n / 2.0

    h = project_feasible(h_init, target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    def time_left():
        return budget_s - (time.time() - start_time)

    # Phase 1: SLSQP from parent
    if time_left() > 90:
        h_new, c5 = _slsqp(h, n, target_sum, 250, 5000, 1e-15)
        if c5 < best_c5:
            best_h, best_c5 = h_new, c5

    # Phase 2: Coordinate descent (exact objective)
    while time_left() > 90:
        h, c5 = _cd_sweep(best_h, n, target_sum, n_sweeps=5)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 3: SLSQP polish on CD result
    if time_left() > 60:
        h_new, c5 = _slsqp(best_h, n, target_sum, 350, 5000, 1e-15)
        if c5 < best_c5:
            best_h, best_c5 = h_new, c5

    # Phase 4: Structured restarts + CD
    attempt = 0
    no_improve = 0
    while time_left() > 45:
        attempt += 1
        strategy = attempt % 7
        h0 = best_h.copy()

        if strategy == 0:
            h0 += np.random.randn(n) * 0.02
        elif strategy == 1:
            h0 = np.roll(best_h, np.random.choice([-3, -2, -1, 1, 2, 3]))
        elif strategy == 2:
            h0 = best_h[::-1].copy()
        elif strategy == 3:
            i1 = np.random.randint(0, n // 2)
            i2 = np.random.randint(i1 + 5, n)
            h0[i1:i2] = h0[i1:i2][::-1]
        elif strategy == 4:
            blen = np.random.randint(3, max(4, n // 5))
            i1 = np.random.randint(0, n - 2 * blen)
            i2 = i1 + blen
            h0[i1:i1+blen], h0[i2:i2+blen] = h0[i2:i2+blen].copy(), h0[i1:i1+blen].copy()
        elif strategy == 5:
            freq = np.random.choice([1, 2, 3])
            h0 += 0.015 * np.sin(2 * np.pi * freq * np.arange(n) / n)
        else:
            mask = np.random.random(n) < 0.15
            h0[mask] += np.random.randn(int(mask.sum())) * 0.05

        h0 = project_feasible(h0, target_sum)
        h_new, c5 = _cd_sweep(h0, n, target_sum, n_sweeps=3)
        if c5 < best_c5:
            best_h, best_c5 = h_new, c5
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                break

    # Phase 5: Final ultra-precise SLSQP
    if time_left() > 25:
        h_new, c5 = _slsqp(best_h, n, target_sum, 500, 10000, 1e-17)
        if c5 < best_c5:
            best_h, best_c5 = h_new, c5

    # Phase 6: One more CD pass
    if time_left() > 15:
        h_new, c5 = _cd_sweep(best_h, n, target_sum, n_sweeps=10)
        if c5 < best_c5:
            best_h, best_c5 = h_new, c5

    return best_h, best_c5, n
