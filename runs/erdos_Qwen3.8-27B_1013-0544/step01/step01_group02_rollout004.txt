def compute_b(best_h, n, i, j):
    b = np.zeros(2*n - 1)
    # For index i, contribution to lag k:
    # coeff_i(k) = (1 - h[i+k]) if i+k in range, else 0
    #            - h[i-k] if i-k in range, else 0
    # b[k] = coeff_i(k) - coeff_j(k)
    
    # Vectorize: create arrays for all k
    # k ranges from -(n-1) to (n-1)
    ks = np.arange(-(n-1), n)
    
    # For i: valid k where i+k in [0,n-1] -> k in [-i, n-1-i]
    #         valid k where i-k in [0,n-1] -> k in [i-(n-1), i]
    ci = np.zeros(2*n-1)
    valid_plus = (i + ks >= 0) & (i + ks < n)
    ci[valid_plus] += (1 - best_h[i + ks[valid_plus]])
    valid_minus = (i - ks >= 0) & (i - ks < n)
    ci[valid_minus] -= best_h[i - ks[valid_minus]]
    
    cj = np.zeros(2*n-1)
    valid_plus_j = (j + ks >= 0) & (j + ks < n)
    cj[valid_plus_j] += (1 - best_h[j + ks[valid_plus_j]])
    valid_minus_j = (j - ks >= 0) & (j - ks < n)
    cj[valid_minus_j] -= best_h[j - ks[valid_minus_j]]
    
    return ci - cj

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-14:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = c - m
    mx = shifted.max()
    if mx > 700:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def sum_con(h, n):
    return np.sum(h) - n / 2.0

def sum_con_jac(h, n):
    return np.ones(n)

def compute_b_vec(h, n, i, j):
    ks = np.arange(-(n - 1), n)
    ci = np.zeros(2 * n - 1)
    vp = (i + ks >= 0) & (i + ks < n)
    ci[vp] += (1 - h[i + ks[vp]])
    vm = (i - ks >= 0) & (i - ks < n)
    ci[vm] -= h[i - ks[vm]]
    cj = np.zeros(2 * n - 1)
    vpj = (j + ks >= 0) & (j + ks < n)
    cj[vpj] += (1 - h[j + ks[vpj]])
    vmj = (j - ks >= 0) & (j - ks < n)
    cj[vmj] -= h[j - ks[vmj]]
    return ci - cj

def pair_descent(h, n, rng, n_pairs, t0, budget_s):
    best_h = h.copy()
    c = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = float(np.max(c) * 2.0 / n)
    for it in range(n_pairs):
        if time.time() - t0 > budget_s - 10:

        if it % 2000 == 1999 and time.time() - t0 > budget_s - 15:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        b = compute_b_vec(best_h, n, i, j)
        delta_lo = max(-best_h[i], best_h[j] - 1.0)
        delta_hi = min(1.0 - best_h[i], best_h[j])
        if delta_lo >= delta_hi - 1e-12:
            continue
        lo, hi = delta_lo, delta_hi
        for _ in range(50):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            if np.max(c + m1 * b) < np.max(c + m2 * b):
                hi = m2
            else:
                lo = m1
        delta_opt = (lo + hi) / 2.0
        if abs(delta_opt) < 1e-14:
            continue
        h_try = best_h.copy()
        h_try[i] = np.clip(h_try[i] + delta_opt, 0.0, 1.0)

def slsqp_step(h, n, alpha, bounds, cons, maxiter=5000):
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20})
        h_try = project_feasible(res.x, n)
        c = compute_c5(h_try, n)
        return h_try, c

        return h, compute_c5(h, n)
