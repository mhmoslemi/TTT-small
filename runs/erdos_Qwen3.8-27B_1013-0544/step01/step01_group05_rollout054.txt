        h_init = np.array(initial_h_values, dtype=float).copy()
    except NameError:
        h_init = None

    n_list = [70, 50, 60, 40, 80]
    if h_init is not None and len(h_init) not in n_list:
        n_list.insert(0, len(h_init))

    for n in n_list:
        if tleft() < 35:
            break
        dx = 2.0 / n
        target_sum = n / 2.0
        n_half = n - 1

        inits = []
        if h_init is not None and n == len(h_init):
            inits.append(project_feasible(h_init, target_sum))

        x = np.linspace(0, 2, n, endpoint=False)
        for freq in [2, 3, 4, 5, 7]:
            for amp in [0.3, 0.45, 0.6]:
                h_f = 0.5 + amp * np.sin(2.0 * np.pi * freq * x / 2.0)
                inits.append(project_feasible(h_f, target_sum))

        for _ in range(3):
            inits.append(project_feasible(rng.random(n), target_sum))

        for _ in range(2):
            h_b = (rng.random(n) > 0.5).astype(float)
            inits.append(project_feasible(h_b + 0.05 * rng.randn(n), target_sum))

        # Deduplicate
        unique_inits = []
        for h in inits:
            if len(unique_inits) >= 6:
                break
            if all(np.max(np.abs(h - h2)) > 0.04 for h2 in unique_inits):
                unique_inits.append(h)

        for h0 in unique_inits:
            if tleft() < 15:
                break
            h_cur = h0.copy()
            for beta in [8.0, 25.0, 60.0, 150.0, 400.0]:
                if tleft() < 8:
                    break
                h_cur, c5_cur = run_slsqp(h_cur, n, dx, target_sum, beta, 600)
                if c5_cur < best_c5:
                    best_c5 = c5_cur
                    best_h = h_cur.copy()
                    best_n = n

            if tleft() > 10:
                h_pg, c5_pg = pg_refine(h_cur, n, dx, target_sum, n_half,
                                        tleft, [0.003, 0.001, 0.0003, 0.0001])
                if c5_pg < best_c5:
                    best_c5 = c5_pg
                    best_h = h_pg
                    best_n = n

    # Final SLSQP polish with very high beta (≈ raw max)
    if best_h is not None and tleft() > 20:
        n = best_n
        dx = 2.0 / n
        target_sum = n / 2.0
        h_final, c5_final = run_slsqp(best_h, n, dx, target_sum, 800.0, 2000)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final

        # One more PG pass
        if tleft() > 8:
            n_half = n - 1
            h_pg, c5_pg = pg_refine(best_h, n, dx, target_sum, n_half,
                                    tleft, [0.0005, 0.0002, 0.00005])
            if c5_pg < best_c5:
                best_c5 = c5_pg
                best_h = h_pg

    if best_h is None:
        n = 70
        dx = 2.0 / n
        best_h = project_feasible(np.full(n, 0.5), n / 2.0)
        best_c5 = compute_c5(best_h, dx)
        best_n = n

    return (best_h, best_c5, best_n)
