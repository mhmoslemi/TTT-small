    """Projection onto {0<=h<=1, sum(h)=target_sum}"""

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_grad_lag(h, L):
    n = len(h)
    omh = 1.0 - h
    grad = np.zeros(n)
    if L == 0:
        grad[:] = omh[:] - h[:]
    elif L > 0:
        grad[L:] += omh[:n - L]
        grad[:n - L] -= h[L:]
    else:
        L2 = -L
        grad[:n - L2] += omh[L2:]
        grad[L2:] -= h[:n - L2]
    return grad

def gradient_refine(h, dx, target_sum, n_active=8, steps=500, step_size=0.001, mu=0.5, time_limit=None, start_time=None):
    """Refine h using projected gradient with active set"""
    n = len(h)
    n_half = n - 1
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    momentum = np.zeros(n)
    ss = step_size
    
    for _ in range(steps):
        if time_limit is not None:
            if time.time() - start_time > time_limit:
                break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = float(np.max(corr))
        tol = max(1e-6, c_max * 1e-4)
        active = np.where(corr >= c_max - tol)[0]
        if len(active) < 2:
            active = np.argsort(corr)[-2:]
        elif len(active) > n_active:
            active = np.argsort(corr)[-n_active:]

        h = h_new

One idea: **iterative water-filling**. 
- Start with some h
- Compute all overlaps: c_k = Σ h[i]·(1-h[i+k]) · dx for all k
- Find k_max = argmax c_k
- Try to reduce c_{k_max} while not increasing others
- This is like a "max-min" approach

Another idea: **Use the dual problem**. The max can be written as:
C₅ = max_k f_k(h) = max_{p ∈ simplex} Σ_k p_k · f_k(h)

So min_h max_k f_k(h) = min_h max_p Σ p_k f_k(h) ≤ max_p min_h Σ p_k f_k(h)

The inner problem min_h Σ p_k f_k(h) subject to constraints might be easier to solve for fixed p. Then we can alternately optimize p and h.

Σ p_k f_k(h) = Σ p_k · dx · Σ_i h[i](1-h[i+k])
= dx · Σ_i h[i] - dx · Σ_i,k p_k h[i]h[i+k]
= 1 - dx · Σ_i,k p_k h[i]h[i+k]  (using the constraint Σ h[i]·dx = 1)

So minimizing this is equivalent to maximizing Σ_i,k p_k h[i]h[i+k].

For fixed p, this is: maximize h^T A_p h where A_p = Σ_k p_k · (shift matrix at lag k), subject to 0≤h≤1, Σh = n/2.

This is a quadratic program (QP)! And if A_p is positive semidefinite, it's a convex QP (actually we want to maximize a concave function, so we need -A_p to be PSD for it to be convex... hmm, it's not straightforward).

Actually, maximizing h^T A h is generally non-convex. But we can use the fact that it's a quadratic program and use various QP solvers.

This alternating approach (fix p, optimize h; fix h, optimize p) might work well!

For the p update: given h, the optimal p puts all weight on the k that maximizes f_k(h). So p becomes a point mass at argmax. That's the same as what we're already doing with the active set gradient.

So the dual approach doesn't give us anything new. Let me stick with the primal approach.

_g_target = None

    return float(np.sum(x)) - _g_target

def project_feasible(x, target_sum):
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = float(np.clip(x - mid, 0.0, 1.0).sum())
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def get_active_grad(h, n_act=10):
    n = len(h)
    nh = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    cm = float(np.max(corr))
    tol = max(1e-8, cm * 5e-4)
    act = np.where(corr >= cm - tol)[0]
    if len(act) < 2:
        act = np.argsort(corr)[-2:]
    elif len(act) > n_act:
        act = np.argsort(corr)[-n_act:]
    omh = 1.0 - h
    g = np.zeros(n)
    for idx in act:
        L = int(idx) - nh
        if L == 0:
            g[:] += omh - h
        elif L > 0:
            g[L:] += omh[:n - L]
            g[:n - L] -= h[L:]
        else:
            L2 = -L
            g[:n - L2] += omh[L2:]
            g[L2:] -= h[:n - L2]
    return g / float(len(act))
