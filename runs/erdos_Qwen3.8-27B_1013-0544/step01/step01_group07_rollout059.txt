import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target_sum):
    """Project onto {0 <= h <= 1, sum(h) = target_sum} using bisection on lambda."""
    n = len(h)
    # We need sum(clip(h - lam, 0, 1)) = target_sum
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def compute_gradient(h, t, dx):
    """Compute gradient of smooth max objective at h."""
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    # corr has length 2n-1, indexed by m from 0 to 2n-2
    # lag k = m - (n-1), ranging from -(n-1) to (n-1)
    p = np.exp(t * corr / dx - np.max(t * corr / dx))
    p /= np.sum(p)
    
    # Gradient: ∂F/∂h[a] = dx * sum_k p_k * [(1-h[a+k])*1_{a+k<n} - h[a-k]*1_{a-k>=0}]
    # Using correlation structure:
    # Term1[a] = sum over valid k of p_k * (1-h[a+k])
    # Term2[a] = sum over valid k of p_k * h[a-k]
    
    # p is indexed by m = k + (n-1), so p[m] corresponds to lag k = m-(n-1)
    # Term1[a] = sum_m p[m] * (1-h[a + m - (n-1)]) where a + m - (n-1) is in [0, n-1]
    # Let j = a + m - (n-1), so m = j - a + (n-1)
    # Term1[a] = sum_j (1-h[j]) * p[j - a + (n-1)]
    # This is the cross-correlation of (1-h) with p, evaluated at position a
    # np.correlate(1-h, p, mode='full')[a + (n-1)] ... hmm
    
    # Let me just do it explicitly for clarity (n is small enough)
    grad = np.zeros(n)
    one_minus_h = 1 - h
    
    for m in range(2 * n - 1):
        if p[m] < 1e-10:
            continue
        k = m - (n - 1)  # lag
        if k >= 0:
            # h[i] * (1-h[i+k]) for i = 0 to n-1-k
            # ∂/∂h[a]: (1-h[a+k]) if a <= n-1-k, minus h[a-k] if a >= k
            # Positive part: a in [0, n-1-k]
            # Negative part: a in [k, n-1]
            valid_pos = slice(0, n - k)
            grad[valid_pos] += p[m] * one_minus_h[k:]  # (1-h[a+k]) for a in [0, n-k-1]
            if k > 0:
                valid_neg = slice(k, n)
                grad[valid_neg] -= p[m] * h[:n-k]  # h[a-k] for a in [k, n-1]
        else:
            # k < 0, let k2 = -k > 0
            # h[i] * (1-h[i-k2]) for i = k2 to n-1
            # ∂/∂h[a]: (1-h[a-k2]) if a >= k2, minus h[a+k2] if a <= n-1-k2
            k2 = -k
            valid_pos = slice(k2, n)
            grad[valid_pos] += p[m] * one_minus_h[:n-k2]  # (1-h[a-k2]) for a in [k2, n-1]
            valid_neg = slice(0, n - k2)
            grad[valid_neg] -= p[m] * h[k2:]  # h[a+k2] for a in [0, n-k2-1]
    
    return dx * grad

def projected_subgradient(h, target_sum, dx, t=50, n_iter=5000, step_scale=0.01):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    h = h.copy()
    step_size = step_scale
    
    for it in range(n_iter):

        # Step
        h = h - step_size * grad
        # Project
        h = fix_sum(h, target_sum)
        # Evaluate
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        # Adapt step size (simple: decrease if not improving)
        if it % 100 == 99:
            step_size *= 0.95
    
    return best_h, best_c5

                improved += 1
                if improved % 20 == 0:
                    corr = np.correlate(h, 1 - h, mode="full")
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
        if improved == 0:
            break
    return h, best_c5

def compute_grad(h, p, dx):
    n = len(h)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        pm = p[m]
        if pm < 1e-12:
            continue
        k = m - (n - 1)
        if k >= 0:
            grad[:n - k] += pm * (1.0 - h[k:])
            if k > 0:
                grad[k:] -= pm * h[:n - k]
        else:
            k2 = -k
            grad[k2:] += pm * (1.0 - h[:n - k2])
            grad[:n - k2] -= pm * h[k2:]
    return dx * grad

def subgradient_descent(h, target_sum, dx, n_iter=2000, t=40, init_step=0.004, decay=0.9997):
    n = len(h)
    step = init_step
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    h = h.copy()
    for it in range(n_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        exponents = t * corr / dx
        exponents -= np.max(exponents)
        p = np.exp(exponents)
        p /= np.sum(p)
        grad = compute_grad(h, p, dx)
        h = fix_sum(h - step * grad, target_sum)
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        step *= decay
    return best_h, best_c5

def swap_search(h, target_sum, dx, n_trials=200):
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h, dx)
    ones_idx = np.where(h > 0.4)[0]
    zeros_idx = np.where(h < 0.6)[0]
    for _ in range(n_trials):
        if len(ones_idx) < 2 or len(zeros_idx) < 2:
            break
        i = ones_idx[np.random.randint(len(ones_idx))]
        j = zeros_idx[np.random.randint(len(zeros_idx))]
        if i == j:
            continue
        h[i], h[j] = h[j], h[i]
        c5_new = compute_c5(h, dx)
        if c5_new > best_c5:
            h[i], h[j] = h[j], h[i]
        else:
            best_c5 = c5_new
            ones_idx = np.where(h > 0.4)[0]
            zeros_idx = np.where(h < 0.6)[0]
    return h, best_c5

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0, 2, old_n + 1)[:-1]
    x_new = np.linspace(0, 2, new_n + 1)[:-1]
    return np.interp(x_new, x_old, h)

def perturb_and_optimize(h, target_sum, dx, n_pert, seed_offset):
    rng = np.random.RandomState(seed_offset)
    n = len(h)
    h_new = h.copy()
    idx = rng.choice(n, n_pert, replace=False)
    h_new[idx] += rng.uniform(-0.3, 0.3, n_pert)
    h_new = fix_sum(h_new, target_sum)
    h_new, c5_1 = subgradient_descent(h_new, target_sum, dx, n_iter=500, t=40, init_step=0.005, decay=0.999)
    h_new, c5_2 = coordinate_descent(h_new, target_sum, dx, max_rounds=30)
    return h_new, c5_2

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = fix_sum(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)
    best_n = n

    # Phase 1: Coordinate descent (exact 1D)
    if elapsed() < budget_s - 60:
        best_h, best_c5 = coordinate_descent(best_h, target_sum, dx, max_rounds=150)

    # Phase 2: Subgradient refinement
    if elapsed() < budget_s - 60:
        h2, c5_2 = subgradient_descent(best_h, target_sum, dx, n_iter=3000, t=30, init_step=0.003, decay=0.9998)
        if c5_2 < best_c5:
            best_h, best_c5 = h2, c5_2
        best_h, best_c5 = coordinate_descent(best_h, target_sum, dx, max_rounds=100)

    # Phase 3: Swap search for near-binary structure
    if elapsed() < budget_s - 60:
        best_h, best_c5 = swap_search(best_h, target_sum, dx, n_trials=500)
        best_h, best_c5 = coordinate_descent(best_h, target_sum, dx, max_rounds=50)

    # Phase 4: Perturbation + restart
    trial = 0
    while elapsed() < budget_s - 150 and trial < 60:
        trial += 1
        h_t, c5_t = perturb_and_optimize(best_h, target_sum, dx,
                                          n_pert=np.random.randint(3, 10),
                                          seed_offset=seed + trial * 7919)
        if c5_t < best_c5:
            best_h, best_c5 = h_t, c5_t
            best_h, best_c5 = coordinate_descent(best_h, target_sum, dx, max_rounds=50)

    # Phase 5: Try other resolutions
    for n_try in [80, 120, 200, 240]:
        if elapsed() > budget_s - 50:
            break
        dx_t = 2.0 / n_try
        ts_t = n_try / 2.0
        h_t = upsample(best_h, n_try)
        h_t = fix_sum(h_t, ts_t)
        c5_t = compute_c5(h_t, dx_t)
        h_t, c5_t = coordinate_descent(h_t, ts_t, dx_t, max_rounds=50)
        if c5_t < best_c5 and elapsed() < budget_s - 20:
            h_t, c5_t = subgradient_descent(h_t, ts_t, dx_t, n_iter=1000, t=30, init_step=0.003)
            h_t, c5_t = coordinate_descent(h_t, ts_t, dx_t, max_rounds=40)
        if c5_t < best_c5:
            best_h, best_c5, best_n = h_t, c5_t, n_try

    # Phase 6: Final polish with extended coordinate descent
    if elapsed() < budget_s - 5:
        dx_f = 2.0 / best_n
        ts_f = best_n / 2.0
        best_h, best_c5 = coordinate_descent(best_h, ts_f, dx_f, max_rounds=200)
        # One more subgradient pass
        if elapsed() < budget_s - 2:
            best_h, c5_sg = subgradient_descent(best_h, ts_f, dx_f, n_iter=2000, t=60, init_step=0.002, decay=0.9999)
            if c5_sg < best_c5:
                best_c5 = c5_sg
            best_h, best_c5 = coordinate_descent(best_h, ts_f, dx_f, max_rounds=50)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = fix_sum(best_h, best_n / 2.0)
    dx_final = 2.0 / best_n
    best_c5 = compute_c5(best_h, dx_final)
    return best_h, best_c5, best_n
