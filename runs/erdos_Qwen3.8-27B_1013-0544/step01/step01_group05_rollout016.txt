def tleft():
    return (_global_start + _global_budget - 3.0) - time.time()

def project_feasible(x, target_sum):
    lo = float(x.min()) - 1.0
    hi = float(x.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = float(np.clip(x - mid, 0.0, 1.0).sum())
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def grad_lag(h, L):
    n = len(h)
    omh = 1.0 - h
    grad = np.zeros(n)
    if L == 0:
        grad[:] = omh[:] - h[:]
    elif L > 0:
        grad[L:] += omh[:n - L]
        grad[:n - L] -= h[L:]
    else:
        L2 = -L
        grad[:n - L2] += omh[L2:]
        grad[L2:] -= h[:n - L2]
    return grad

def active_grad(h, dx, n_half, n_active=6):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    tol = max(0.005, c_max * 0.003)
    active = np.where(corr >= c_max - tol)[0]
    if len(active) < n_active:
        active = np.argsort(corr)[-n_active:]
    elif len(active) > n_active * 3:
        active = np.argsort(corr)[-n_active * 3:]
    grad = np.zeros(len(h))
    for idx in active:
        L = int(idx) - n_half
        grad += grad_lag(h, L)
    grad /= float(len(active))
    return grad * dx
