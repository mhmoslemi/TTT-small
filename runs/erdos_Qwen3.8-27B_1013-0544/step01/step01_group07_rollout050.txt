        # Compute p = softmax(t * corr)

        p = np.exp(tcorr - tcorr.max())
        p /= p.sum()

        # where k ranges over valid lags

        lags = np.arange(n_corr) - (n-1)  # from -(n-1) to n-1

            if p[k_idx] < 1e-15:
                continue
            pi = p[k_idx]
            # (1-z[i+lag]) - z[i-lag]
            # valid i for z[i+lag]: 0 <= i+lag <= n-1, i.e., -lag <= i <= n-1-lag
            # valid i for z[i-lag]: 0 <= i-lag <= n-1, i.e., lag <= i <= n-1+lag
            # i+lag valid: i in [max(0,-lag), min(n-1, n-1-lag)]
            # i-lag valid: i in [max(0,lag), min(n-1, n-1+lag)]

            # For the second term: sum over i where i-lag is valid
            i_start2 = max(0, lag)
            i_end2 = min(n-1, n-1+lag)
            if i_start2 <= i_end2:
                grad[i_start2:i_end2+1] -= pi * z[i_start2-lag:i_end2-lag+1]
        
        # Step size (need Lipschitz constant or use backtracking)
        # For now, use a fixed step size
        step = step_size_schedule(step)

        # proximal step
        x_new = prox_constraint(x_new, target_sum)

        step = step_scale / (4.0 * n)
        for _ in range(n_inner):

            x_new = z - step * grad
            x_new = prox_constraint(x_new, target_sum)
            t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2)) / 2.0
            z = x_new + ((t_mom - 1.0) / t_next) * (x_new - x)
            t_mom = t_next
            x = x_new
        val = compute_c5(x, dx)
        if val < best_val:
            best_val = val
            best_x = x.copy()
        x = best_x.copy()

def binary_sa(h, dx, n_iter, rng, t_init=0.5, t_min=0.001):
    n = len(h)
    h = np.where(h >= 0.5, 1.0, 0.0).astype(float)
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones != target:
        ones_idx = np.where(h == 1.0)[0]
        zeros_idx = np.where(h == 0.0)[0]
        if n_ones > target:
            flip = rng.choice(ones_idx, n_ones - target, replace=False)
            h[flip] = 0.0
        else:
            flip = rng.choice(zeros_idx, target - n_ones, replace=False)
            h[flip] = 1.0
        ones_idx = np.where(h == 1.0)[0]
        zeros_idx = np.where(h == 0.0)[0]
    
    corr = np.correlate(h, 1.0 - h, mode="full")
    best_c5 = float(np.max(corr) * dx)
    cur_c5 = best_c5
    t_cur = t_init
    ones_idx = np.where(h == 1.0)[0]
    zeros_idx = np.where(h == 0.0)[0]
    n_c = len(corr)  # 2n-1
    
    for it in range(n_iter):
        if t_cur < t_min:
            break
        i = int(rng.choice(ones_idx))
        j = int(rng.choice(zeros_idx))
        
        # Compute delta R(lag) for all lags
        # lag = k - (n-1) for k in range(n_c)
        # Delta for lag:
        #   -(1-h[i+lag]) if 0<=i+lag<n
        #   +h[i-lag] if 0<=i-lag<n
        #   +(1-h[j+lag]) if 0<=j+lag<n
        #   -h[j-lag] if 0<=j-lag<n
        
        dR = np.zeros(n_c)
        lags = np.arange(n_c) - (n - 1)
        
        # For position i (1->0):
        # As "m" (first factor): contributes -(1-h[i+lag]) where 0<=i+lag<n
        mask_i_plus = (i + lags >= 0) & (i + lags < n)
        dR[mask_i_plus] -= (1.0 - h[i + lags[mask_i_plus]])
        # As "m+lag" (second factor): contributes +h[i-lag] where 0<=i-lag<n
        mask_i_minus = (i - lags >= 0) & (i - lags < n)
        dR[mask_i_minus] += h[i - lags[mask_i_minus]]
        
        # For position j (0->1):
        mask_j_plus = (j + lags >= 0) & (j + lags < n)
        dR[mask_j_plus] += (1.0 - h[j + lags[mask_j_plus]])
        mask_j_minus = (j - lags >= 0) & (j - lags < n)
        dR[mask_j_minus] -= h[j - lags[mask_j_minus]]
        
        new_corr = corr + dR
        new_max = float(np.max(new_corr))
        new_c5 = new_max * dx
        
        if new_c5 < cur_c5 or rng.random() < np.exp((cur_c5 - new_c5) / t_cur):
            # Accept
            h[i] = 0.0
            h[j] = 1.0
            corr = new_corr
            cur_c5 = new_c5
            ones_idx = np.where(h == 1.0)[0]
            zeros_idx = np.where(h == 0.0)[0]
            if new_c5 < best_c5:
                best_c5 = new_c5
            t_cur *= 0.999
        else:
            # Reject
            t_cur *= 0.9995
        
        if it % 1000 == 0 and len(ones_idx) == 0:
            break
    
    return h, best_c5

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0, 2.0, old_n + 1)[:-1]
    x_new = np.linspace(0, 2.0, new_n + 1)[:-1]
    return np.interp(x_new, x_old, h)

    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h
