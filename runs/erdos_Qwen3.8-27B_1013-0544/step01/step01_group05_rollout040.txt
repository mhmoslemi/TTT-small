import numpy as np

import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(x, target_sum):
    lo = float(x.min()) - 1.0
    hi = float(x.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = float(np.clip(x - mid, 0.0, 1.0).sum())
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def lse_obj_grad(h, dx, beta, penalty):
    """Log-sum-exp objective with penalty for sum constraint. Returns (obj, grad)."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(corr)
    shifted = beta * (corr - c_max)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    
    # Objective value (log-sum-exp approximation of max)
    obj = c_max + np.log(exp_shifted.sum()) / beta
    
    # Penalty for sum constraint
    sum_dev = np.sum(h) - n / 2.0
    obj += penalty * sum_dev ** 2
    
    # Gradient
    N_half = n - 1
    grad = np.zeros(n)
    
    for m in range(n):  # m >= 0
        wk = w[N_half + m] * dx
        if wk < 1e-15:
            continue
        grad[:n-m] += wk * (1.0 - h[m:])
        grad[m:] -= wk * h[:n-m]
    
    for p in range(1, n):  # m = -p
        wk = w[N_half - p] * dx
        if wk < 1e-15:
            continue
        grad[p:] += wk * (1.0 - h[:n-p])
        grad[:n-p] -= wk * h[p:]
    
    # Add penalty gradient
    grad += 2.0 * penalty * sum_dev
    
    return obj, grad

            best_n = n
        
        h_curr = h_start.copy()
        for beta in [15.0, 30.0, 60.0, 120.0, 250.0]:
            if time.time() - t0 > budget_s - 30:
                break
            result = _lbfgs_optimize(h_curr, dx, target, beta, 1e5, 400, t0, budget_s)
            if result is not None:
                c5_r = compute_c5(result, dx)
                h_curr = result
                if c5_r < best_c5:
                    best_c5 = c5_r
                    best_h = result.copy()
                    best_n = n
        
        if time.time() - t0 > budget_s - 20:
            break
        result = _slsqp_optimize(h_curr, dx, target, 800, t0, budget_s)
        if result is not None:
            c5_r = compute_c5(result, dx)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = result.copy()
                best_n = n
                h_curr = result
    
    # Random restarts
    n = best_n
    dx = 2.0 / n
    target = n / 2.0
    for i in range(40):
        if time.time() - t0 > budget_s - 8:
            break
        scale = 0.02 + 0.04 * rng.random()
        h_r = best_h + rng.normal(0, scale, n)
        h_r = project_feasible(h_r, target)
        result = _lbfgs_optimize(h_r, dx, target, 80.0, 1e5, 200, t0, budget_s)
        if result is not None:
            c5_r = compute_c5(result, dx)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = result.copy()
    
    # Final polish
    if time.time() - t0 < budget_s - 4:
        h_pol, c5_pol = _polish_active(best_h, dx, target, n, t0, budget_s)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol
    
    return (best_h, best_c5, best_n)
