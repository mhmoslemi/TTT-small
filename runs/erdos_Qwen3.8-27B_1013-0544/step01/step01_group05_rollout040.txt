from scipy.optimize import minimize
import time

_g_dx = None
_g_target = None
_g_n = None

def project_feasible(x, target_sum):
    lo = float(x.min()) - 1.0

    for _ in range(60):

        s = float(np.clip(x - mid, 0.0, 1.0).sum())

            lo = mid
        else:
            hi = mid

    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def corr_vals(h, dx):
    return np.correlate(h, 1.0 - h, mode='full') * dx

    g = np.zeros(n)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[:n-L] += (1.0 - h[L:])
        g[L:] -= h[:n-L]
    else:
        L2 = -L
        g[L2:] += (1.0 - h[:n-L2])
        g[:n-L2] -= h[L2:]
    return g

    n = len(h)
    nh = n - 1
    cv = corr_vals(h, dx)
    mx = float(np.max(cv))
    s = T * (cv - mx)
    e = np.exp(s)
    tot = float(np.sum(e))
    w = e / tot
    obj = mx + np.log(tot) / T

        if w[k] < 1e-10:

        grad += w[k] * grad_g(h, L, n) * dx
    return obj, grad

    return compute_c5(x, _g_dx)

def _smooth_grad(x):

    nh = n - 1
    best = h.copy()

    step = 0.006

    mu = 0.55
    stall = 0
    while time.time() - t_start < t_budget:

        cm = float(np.max(cv))
        na = max(3, min(12, n // 8))
        act = np.argsort(cv)[-na:]
        g = np.zeros(n)
        for idx in act:
            L = int(idx) - nh
            g += grad_g(h, L, n)
        g /= float(len(act))
        mom = mu * mom - step * g
        h_new = project_feasible(h + mom, target)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:

            stall = 0
