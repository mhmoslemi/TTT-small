    scaled = corr * dx * tau
    max_s = np.max(scaled)
    exps = np.exp(scaled - max_s)
    log_sum = max_s + np.log(np.sum(exps))

def project_feasible(h, n):
    target = n / 2.0
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - target
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        if np.any(free):
            h[free] -= diff / np.sum(free)
        else:
            h = h - diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = np.sum(h) - target
    if abs(diff) > 1e-15:
        h = h - diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_val(h, dx, tau):
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = corr * dx * tau
    m = np.max(scaled)
    return (m + np.log(np.sum(np.exp(scaled - m)))) / tau

def soft_max_grad(h, dx, tau):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = corr * dx * tau
    m = np.max(scaled)
    exps = np.exp(scaled - m)
    s = np.sum(exps)
    weights = exps / s
    g = np.zeros(n)
    for k_idx in range(2 * n - 1):
        w = weights[k_idx]
        if w < 1e-14:
            continue
        w *= dx
        k = k_idx - (n - 1)
        if k >= 0:
            g[k:] += (1.0 - h[:n - k]) * w
            g[:n - k] += (-h[k:]) * w
        else:
            kp = -k
            g[:n - kp] += (1.0 - h[kp:]) * w
            g[kp:] += (-h[:n - kp]) * w
    return g

def grad_single_shift(h, k_idx, n):
    dx = 2.0 / n
    k = k_idx - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        g[k:] = (1.0 - h[:n - k]) * dx
        g[:n - k] += (-h[k:]) * dx
    else:
        kp = -k
        g[:n - kp] = (1.0 - h[kp:]) * dx
        g[kp:] += (-h[:n - kp]) * dx
    return g

def run_lbfgsb(h_start, n, dx, tau, penalty, max_iter=500, ftol=1e-16):
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    def fun(x):
        return soft_max_val(x, dx, tau) + penalty * (np.sum(x) - target) ** 2
    def jac(x):
        return soft_max_grad(x, dx, tau) + 2.0 * penalty * (np.sum(x) - target)
    try:
        res = minimize(fun, h_start, jac=jac, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': max_iter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h_start, n)

def run_slsqp(h_start, n, dx, max_iter=500, ftol=1e-14):
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    def fun(x):
        return compute_c5(x, dx)
    def con(x):
        return np.sum(x) - target
    try:

                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': max_iter, 'ftol': ftol, 'iprint': 0})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h_start, n)

def subgradient_descent(h, n, dx, n_steps=5000, init_step=0.01):
    """Projected subgradient on the worst shift."""
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    alpha = init_step

        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
        alpha *= 0.999  # Slowly decrease step size
    return best_h, best_c5
