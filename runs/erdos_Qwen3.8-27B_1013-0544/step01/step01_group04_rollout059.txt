            h_try = h_pol + rng.randn(n2) * 0.015
            h_try = project_feasible(h_try, n2)
            h_new = slsqp_optimize(h_try, n2, max_iter=250)
            c5_new = compute_c5(h_new, dx2)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
    
    best_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return (best_h, best_c5, len(best_h))
