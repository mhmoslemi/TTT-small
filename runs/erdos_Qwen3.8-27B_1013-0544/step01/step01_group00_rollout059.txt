import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))


def project_h(h, target):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
        else:
            mask = h > 1e-13
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def t_ineq(v):
    n = len(v) - 1
    h = v[:n]
    t = v[-1]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return t - corr


def t_eq_fun(v):
    n = len(v) - 1
    return np.sum(v[:n]) - n * 0.5


def t_eq_jac(v):
    n = len(v) - 1
    j = np.zeros(len(v))
    j[:n] = 1.0
    return j


def t_obj(v):
    return v[-1]


def lse_obj(h, temp):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    return m + temp * np.log(np.sum(np.exp((corr - m) / temp)))


def pen_obj(x, temp, pen, target):
    return lse_obj(x, temp) + pen * (np.sum(x) - target) ** 2


def alum_obj(x, temp, lam, mu, target):
    s = np.sum(x) - target
    return lse_obj(x, temp) + lam * s + 0.5 * mu * s * s


def interp_h(h_old, n_new):

    h_new = np.interp(x_new, x_old, h_old)

    h_new = np.clip(res.x[:n], 0.0, 1.0)
    h_new = project_h(h_new, n * 0.5)
    return h_new, compute_c5(h_new)


def opt_alum(h0, temp, target, max_iter=1000):

    h = h0.copy()
    bounds = [(0.0, 1.0)] * n
    lam = 0.0
    mu = 50.0

                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': max_iter, 'ftol': 1e-16})
