        if s == 0:
            h_try = h0.copy()
        else:
            h_try = h0 + rng.randn(n) * 0.02 * (s + 1)
            h_try = project_box_sum(h_try, target_sum)
        alpha = alpha_base + 40 * s
        max_it = max(50, int((deadline - time.time()) / max(n_starts - s, 1) / 2.0))
        tl = min(deadline - time.time() - 2, time_per_start)
        if tl < 5:
            break
        h_opt, c5_opt = optimize_lbfgsb(h_try, target_sum, alpha, penalty, max_it, tl)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
    return best_h, best_c5

def optimize_momentum(h0, target_sum, deadline, rng, alpha=150):
    n = len(h0)
    h = h0.copy()
    vel = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = 0.02
    beta = 0.9
    stall = 0
    for it in range(50000):
        if time.time() >= deadline:
            break
        h_y = h + beta * vel
        corr = compute_corr(h_y)
        m = np.max(corr)
        shifted = np.clip(alpha * (corr - m), -500, 500)
        w = np.exp(shifted)
        sw = np.sum(w)
        w /= sw
        grad = np.zeros(n)
        active = np.where(w > 1e-10)[0]
        for k_idx in active:
            wi = w[k_idx]
            k = k_idx - (n - 1)
            if k >= 0:
                grad[:n-k] += wi * (1.0 - h_y[k:])
                if k > 0:
                    grad[k:] -= wi * h_y[:n-k]
            else:
                p = -k
                grad[p:] += wi * (1.0 - h_y[:n-p])
                grad[:n-p] -= wi * h_y[p:]
        grad *= (2.0 / n)
        vel = beta * vel - lr * grad
        h_new = h + vel
        h_new = project_box_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-13:
            best_h = h_new.copy()
            best_c5 = c5_new
            stall = 0
            lr = min(lr * 1.01, 0.08)
        else:
            stall += 1
            lr *= 0.97
            if stall > 80:
                h = best_h + rng.randn(n) * 0.008
                h = project_box_sum(h, target_sum)
                vel = np.zeros(n)
                stall = 0
                lr = 0.015
                alpha = min(alpha * 1.4, 500)
        h = h_new
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = 0.5 + 0.3 * rng.randn(200)

    n0 = len(h0)
    ts0 = n0 / 2.0
    h0 = project_box_sum(h0, ts0)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    # Phase 1: L-BFGS-B at initial resolution
    if time.time() - t0 < budget_s * 0.25:
        dl = t0 + budget_s * 0.25
        h_opt, c5_opt = optimize_with_restarts(h0, ts0, dl, rng, alpha_base=100, penalty=200, n_starts=3)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 2: Interpolate to higher resolution and optimize
    for n_target in [400, 500]:
        if time.time() - t0 >= budget_s * 0.82:
            break
        h_interp = interpolate_to_n(best_h, n_target)
        ts = n_target / 2.0
        h_interp = project_box_sum(h_interp, ts)
        frac = 0.55 if n_target == 400 else 0.78
        dl = t0 + budget_s * frac
        if dl - time.time() < 20:
            continue
        # First try L-BFGS-B
        h_opt, c5_opt = optimize_lbfgsb(h_interp, ts, alpha=150, penalty=200,
                                         max_iter=300, time_limit=dl - time.time() - 10)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
        # Then refine with momentum
        if time.time() < dl - 15:
            h_opt2, c5_opt2 = optimize_momentum(best_h, ts, dl, rng, alpha=200)
            if c5_opt2 < best_c5:
                best_h, best_c5 = h_opt2, c5_opt2

    # Phase 3: Final high-alpha polish
    remaining = budget_s - (time.time() - t0)
    if remaining > 20:
        ts = len(best_h) / 2.0
        h_opt, c5_opt = optimize_momentum(best_h, ts, t0 + budget_s - 5, rng, alpha=350)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Final: try L-BFGS-B with high alpha on best solution
    remaining = budget_s - (time.time() - t0)
    if remaining > 10:
        ts = len(best_h) / 2.0
        h_opt, c5_opt = optimize_lbfgsb(best_h, ts, alpha=300, penalty=500,
                                         max_iter=200, time_limit=remaining - 3)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
