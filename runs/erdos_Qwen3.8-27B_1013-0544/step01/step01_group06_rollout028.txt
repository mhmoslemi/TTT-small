def project_onto_simplex_box(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):

        if s > target_sum:

        else:

    m = np.max(corr)

    if s < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= s
    return w

        g[k:] -= h[:n - k]

    if remaining > 15:
        h_fin, c5_fin = pgd_optimize(best_h, len(best_h) / 2.0,
                                     t0 + budget_s - 5, rng, alpha_init=250)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin

    best_h = project_simplex(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
