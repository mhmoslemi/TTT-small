def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_smooth_grad(h, t):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_c = np.max(corr)
    weights = np.exp(t * (corr - max_c))
    total_w = np.sum(weights)
    if total_w < 1e-300:
        return np.zeros(n), max_c * (2.0/n)
    weights /= total_w
    
    grad = np.zeros(n)
    n_shifts = len(corr)
    for idx in range(n_shifts):
        w = weights[idx]
        if w < 1e-10:
            continue
        k = idx - (n - 1)
        if k > 0:
            grad[k:] += w * (1.0 - h[:-k])
            grad[:n-k] -= w * h[k:]
        elif k < 0:
            m = -k
            grad[:n-m] += w * (1.0 - h[m:])
            grad[m:] -= w * h[:-m]
        else:
            grad += w * (1.0 - 2.0 * h)
    return grad, max_c * (2.0/n)

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_smooth_grad(h, t):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_c = np.max(corr)
    shifted = t * (corr - max_c)
    weights = np.exp(shifted)
    total = np.sum(weights)
    if total < 1e-300:
        return np.zeros(n), max_c
    weights /= total
    threshold = 1.0 / (n * n)
    active = np.where(weights > threshold)[0]
    grad = np.zeros(n)
    for idx in active:
        w = weights[idx]
        k = idx - (n - 1)
        if k > 0:
            grad[k:] += w * (1.0 - h[:-k])
            grad[:n - k] -= w * h[k:]
        elif k < 0:
            m = -k
            grad[:n - m] += w * (1.0 - h[m:])
            grad[m:] -= w * h[:-m]
        else:
            grad += w * (1.0 - 2.0 * h)
    return grad, max_c

def grad_descent(h0, dx, target_sum, n_iter, lr_init, t_init, t_final, momentum=0.9, t_check=500):
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)
    lr = lr_init
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    for it in range(n_iter):
        frac = float(it) / max(n_iter, 1)
        t_val = t_init + (t_final - t_init) * frac
        grad, _ = compute_smooth_grad(h, t_val)
        v = momentum * v - lr * grad
        h_new = h + v
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = repair_sum(h_new, target_sum)
        h = h_new
        if (it + 1) % t_check == 0:
            c5 = compute_c5(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                lr = min(lr * 1.05, 0.5)
            else:
                lr = max(lr * 0.93, 1e-7)
    c5 = compute_c5(h, dx)
    if c5 < best_c5:
        best_h, best_c5 = h, c5
    return best_h, best_c5

def slsqp_refine(h, dx, target_sum, maxiter=400, t=50.0):
    n = len(h)
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = np.max(corr)
        s = np.sum(np.exp(t * (corr - m)))
        return (m + np.log(s) / t) * dx
    try:
        res = minimize(obj, h.copy(), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                       options={'maxiter': maxiter, 'ftol': 1e-11})
        return repair_sum(res.x, target_sum)
    except Exception:
        return h.copy()

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0, 2, old_n + 1)[:-1]
    x_new = np.linspace(0, 2, new_n + 1)[:-1]
    return np.interp(x_new, x_old, h)
