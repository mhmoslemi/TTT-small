def fix_sum(h, target):

        return np.clip(h, 0.0, 1.0)

    else:

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target):
    h = np.array(h, dtype=float)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return np.clip(h, 0.0, 1.0)
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = mask.sum()
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def project_feasible(y, target):
    lo, hi = -2.0, 3.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def compute_grad(h, tau):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    tc = tau * corr
    tc -= tc.max()
    w = np.exp(tc)
    w /= w.sum()
    grad = np.zeros(n)
    for shift in range(-(n - 1), n):
        k = shift + n - 1
        wk = w[k]
        if wk < 1e-14:
            continue
        lo = max(0, -shift)
        hi = min(n, n - shift)
        if lo < hi:
            grad[lo:hi] += wk * (1.0 - h[lo + shift:hi + shift])
        lo2 = max(0, shift)
        hi2 = min(n, n + shift)
        if lo2 < hi2:
            grad[lo2:hi2] -= wk * h[lo2 - shift:hi2 - shift]
    return grad

def proj_grad(h0, target, tau, lr, steps, mom):
    h = h0.copy()
    v = np.zeros_like(h)
    best = h.copy()
    bc5 = evaluate_c5(h)
    for _ in range(steps):
        g = compute_grad(h, tau)
        v = mom * v - lr * g
        h = project_feasible(h + v, target)
        c5 = evaluate_c5(h)
        if c5 < bc5:
            bc5 = c5
            best = h.copy()
    return best, bc5

def slsqp_opt(h0, n, tau, miter):
    tgt = n / 2.0
    bnds = Bounds(np.zeros(n), np.ones(n))
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode="full")
        tc = tau * (c - c.max())
        return float(c.max() / tau + np.log(np.sum(np.exp(tc))) / tau)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - tgt, 'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(obj, h0, method='SLSQP', bounds=bnds, constraints=cons,
                     options={'maxiter': miter, 'ftol': 1e-13})
        x = fix_sum(r.x, tgt)
        return x, evaluate_c5(x)
    except Exception:
        return h0, None

        best_pair = None
        c5_best = c5
        for _ in range(n_try):
            i = zeros[np.random.randint(len(zeros))]
            j = ones[np.random.randint(len(ones))]
            h[i], h[j] = 1.0, 0.0
            nc5 = evaluate_c5(h)
            if nc5 < c5_best - 1e-15:
                best_imp = c5_best - nc5
                best_pair = (i, j)
                c5_best = nc5
            else:
                h[i], h[j] = 0.0, 1.0
        if best_pair is None:
            break
        c5 = c5_best
        if c5 < bc5:
            bc5 = c5
            best = h.copy()
    return best, bc5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(60, 0.5)

    best_h = h_init.copy()
    best_c5 = evaluate_c5(best_h)

    def rem():
        return budget_s - (time.time() - t0)

    n = len(h_init)
    tgt = n / 2.0

    # Phase 1: Projected gradient with momentum
    for tau in [20, 60, 150, 400, 1000]:
        if rem() < 15:
            break
        for lr in [0.01, 0.005, 0.002]:
            if rem() < 8:
                break
            hh, cc = proj_grad(best_h, tgt, tau, lr, 250, 0.9)
            if cc < best_c5:
                best_h, best_c5 = hh, cc

    # Phase 2: SLSQP polish
    for tau in [100, 400, 1000]:
        if rem() < 12:
            break
        hh, cc = slsqp_opt(best_h, n, tau, 250)
        if cc is not None and cc < best_c5:
            best_h, best_c5 = hh, cc

    # Phase 3: Binary SA + greedy from continuous solution
    if rem() > 50:
        hb = to_bin(best_h, n)
        hb, cc = bin_sa(hb, 8000, 0.008, 0.00005, 25)
        if cc < best_c5:
            best_h, best_c5 = hb.astype(float), cc
        hb2, cc2 = bin_greedy(hb, 50, 300)
        if cc2 < best_c5:
            best_h, best_c5 = hb2.astype(float), cc2

    # Phase 4: Random binary starts with SA
    for restart in range(8):
        if rem() < 40:
            break
        hr = random_binary(n)
        hr, cc = bin_sa(hr, 5000, 0.006, 0.00003, 20)
        if cc < best_c5:
            best_h, best_c5 = hr.astype(float), cc
        hr2, cc2 = bin_greedy(hr, 40, 200)
        if cc2 < best_c5:
            best_h, best_c5 = hr2.astype(float), cc2

    # Phase 5: Multi-resolution exploration
    for tn in [40, 50, 60, 80, 100, 140, 160, 200]:
        if rem() < 35:
            break
        tt = tn / 2.0
        ht = np.interp(np.linspace(0, 1, tn), np.linspace(0, 1, len(best_h)), best_h)
        ht = fix_sum(ht, tt)
        for tau in [40, 150, 400]:
            if rem() < 12:
                break
            hh, cc = proj_grad(ht, tt, tau, 0.006, 180, 0.9)
            if cc < best_c5:
                best_h, best_c5, ht = hh, cc, hh
        if rem() > 12:
            hh, cc = slsqp_opt(ht, tn, 400, 200)
            if cc is not None and cc < best_c5:
                best_h, best_c5 = hh, cc
        if rem() > 18:
            hbt = to_bin(ht, tn)
            hbt, ccb = bin_sa(hbt, 4000, 0.006, 0.00004, 20)
            if ccb < best_c5:
                best_h, best_c5 = hbt.astype(float), ccb
            hbt2, ccb2 = bin_greedy(hbt, 35, 150)
            if ccb2 < best_c5:
                best_h, best_c5 = hbt2.astype(float), ccb2

    # Phase 6: Final polish
    if rem() > 12:
        nf = len(best_h)
        for tau in [500, 2000]:
            if rem() < 8:
                break
            hh, cc = slsqp_opt(best_h, nf, tau, 250)
            if cc is not None and cc < best_c5:
                best_h, best_c5 = hh, cc

    # Final constraint enforcement
    nf = len(best_h)
    best_h = fix_sum(best_h, nf / 2.0)
    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, nf
