def coord_descent_linear(h, active_lags, n, max_iter=100):
    """
    Coordinate descent for the linearized objective.
    active_lags: list/array of shifts s (can be negative)
    """
    h = h.copy()
    target = n / 2.0
    
    for _ in range(max_iter):
        # Compute coefficients c_j
        c = np.zeros(n)
        for s in active_lags:
            # h[j] appears as h[i] with i=j: contributes (1-h[j+s]) if j+s in [0,n-1]
            lo = max(0, -s)
            hi = min(n, n - s)
            # For i in [lo, hi-1]: h[i] contributes (1-h[i+s])
            # So for j = i: c[j] += (1 - h[j+s])
            # And h[j] appears as h[i+s] with i = j-s: contributes -h[j-s] if j-s in [0,n-1]
            
            for j in range(n):
                if 0 <= j + s < n:
                    c[j] += (1.0 - h[j + s])
                if 0 <= j - s < n:
                    c[j] -= h[j - s]
        
        # Sort and threshold
        sorted_idx = np.argsort(c)
        h_new = np.zeros(n)
        h_new[sorted_idx[:int(target)]] = 1.0
        
        if np.allclose(h_new, h):
            break
        h = h_new
    
    return h

        h_new = do_slsqp(best_h, hn, tgt2, tau, 400, 1e-15)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 7: Coordinate-level hill climbing polish
    if rem() > 5:
        hn = len(best_h)
        tgt2 = hn / 2.0
        for _ in range(5):
            if rem() < 3:
                break
            improved = False
            for i in range(hn):
                if rem() < 1:
                    break
                for delta in [0.005, -0.005, 0.002, -0.002]:
                    h_try = best_h.copy()
                    h_try[i] = np.clip(h_try[i] + delta, 0, 1)
                    h_try = fix_sum(h_try, tgt2)
                    c5_try = evaluate_c5(h_try)
                    if c5_try < best_c5 - 1e-10:
                        best_h = h_try
                        best_c5 = c5_try
                        improved = True
                        break
            if not improved:
                break

    # Final constraint enforcement
    best_h = fix_sum(best_h, len(best_h) / 2.0)
    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, len(best_h)
