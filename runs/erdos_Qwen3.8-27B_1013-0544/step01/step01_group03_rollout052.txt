def fix_sum(h, target):

            v = make_block_init(n0, [0.8, 0.2, 0.7, 0.3, 0.6, 0.4])
        elif i == 6:
            x = np.linspace(0, 2, n0, endpoint=False)
            v = fix_sum(0.5 + 0.2 * np.sin(np.pi * x), target0)
        else:
            v = fix_sum(np.clip(best_h + 0.1 * np.random.randn(n0), 0, 1), target0)
        for tau in [5.0, 1.0, 0.3]:
            if time.time() > deadline - 100:
                break
            h_new, c5_new = nesterov_optimize(v, target0, tau, 3000, 0.008, t0, deadline - 100)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new

    # Phase 3: Multi-resolution zoom
    for n_hi in [160, 200, 240]:
        if time.time() > deadline - 80:
            break
        target_hi = n_hi / 2.0
        x_low = np.linspace(0, 1, len(best_h))
        x_hi = np.linspace(0, 1, n_hi)
        h_hi = np.interp(x_hi, x_low, best_h)
        h_hi = fix_sum(h_hi, target_hi)
        for tau in [5.0, 1.5, 0.5]:
            if time.time() > deadline - 60:
                break
            h_new, c5_new = nesterov_optimize(h_hi, target_hi, tau, 4000, 0.006, t0, deadline - 60)
            if c5_new < best_c5 * (n0 / n_hi) * (n_hi / n0):
                # Interpolate back to compare fairly
                x_back = np.linspace(0, 1, n0)
                h_back = np.interp(x_back, x_hi, h_new)
                h_back = fix_sum(h_back, target0)
                c5_back = evaluate_c5(h_back)
                if c5_back < best_c5:
                    best_h, best_c5 = h_back, c5_back
                # Also keep the high-res version for final answer
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new
                    target0 = n_hi / 2.0

    # Phase 4: SLSQP polish at current best resolution
    n_final = len(best_h)
    target_final = n_final / 2.0
    if time.time() < deadline - 30:
        h_pol, c5_pol = slsqp_polish(best_h, n_final, target_final, 300, t0, deadline - 15)
        if c5_pol < best_c5:
            best_h, best_c5 = h_pol, c5_pol

    # Phase 5: Final Nesterov with very low tau
    if time.time() < deadline - 5:
        n_final = len(best_h)
        target_final = n_final / 2.0
        h_new, c5_new = nesterov_optimize(best_h, target_final, 0.05, 3000, 0.003, t0, deadline)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    best_h = fix_sum(best_h, len(best_h) / 2.0)
    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, len(best_h)
