def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) < 1e-12:
        return h
    h = h + residual / len(h)
    h = np.clip(h, 0.0, 1.0)
    residual2 = target_sum - np.sum(h)
    if abs(residual2) > 1e-10:
        if residual2 > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)
        for i in idx:
            if residual2 > 0:
                add = min(residual2, 1.0 - h[i])
                h[i] += add
                residual2 -= add
            else:
                sub = min(-residual2, h[i])
                h[i] -= sub
                residual2 += sub
            if abs(residual2) < 1e-12:
                break
    return h

def smooth_max_value(h, t, dx, target_sum, penalty_weight=200.0):
    h_c = np.clip(h, 0.0, 1.0)

    m = np.max(corr)

    lse = (m + np.log(np.sum(np.exp(shifted)))) / t
    penalty = penalty_weight * (np.sum(h) - target_sum) ** 2
    return lse * dx + penalty

            i, j = rng.choice(n, 2, replace=False)
            for delta in [step, -step, step * 0.5, -step * 0.5, step * 0.25, -step * 0.25]:
                h_new = h.copy()
                h_new[i] += delta

                        best_c5 = c5_new
                        improved = True
                        break
        if not improved:
            step *= 0.7
            if step < 1e-5:
                break
    return h, best_c5
