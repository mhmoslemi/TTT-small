import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def get_corr(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_h(h, target):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = np.sum(mask)
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def smooth_max(corr, temp):
    m = np.max(corr)
    if temp < 1e-14:
        return m
    return m + temp * np.log(np.sum(np.exp((corr - m) / temp)))

def lse_obj(v, temp, pen):
    h = v
    corr = get_corr(h)
    s = np.sum(h) - len(h) * 0.5
    return smooth_max(corr, temp) + pen * s * s

def t_obj(v):
    return v[-1]

def t_ineq(v):
    n = len(v) - 1
    h = v[:n]
    t = v[-1]
    corr = get_corr(h)
    return t - corr

def t_eq(v):
    n = len(v) - 1
    return np.sum(v[:n]) - n * 0.5

    n = len(v) - 1
    j = np.zeros(len(v))
