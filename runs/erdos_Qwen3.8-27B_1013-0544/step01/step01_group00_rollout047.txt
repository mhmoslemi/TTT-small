    """Project h onto {x: 0<=x<=1, sum(x)=target} using bisection on threshold."""

    x_new = np.arange(n_new) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new * 0.5)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    
    def time_left():
        return budget_s - (time.time() - t0)
    
    # Start from initial_h_values
    h = np.array(initial_h_values, dtype=float)
    n = len(h)
    h = project_feasible(h, n * 0.5)
    c5 = compute_c5(h)
    h_best = h.copy()
    
    # Phase 1: FISTA at current resolution with temperature annealing
    temps = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    for temp in temps:
        if time_left() < 80:
            break
        n_iter = min(5000, max(1000, int(time_left() * 8)))
        h_new, c5_new = fista_projected(h_best, temp, n_iter=n_iter, lr_init=0.005)
        if c5_new < c5:
            h_best = h_new
            c5 = c5_new
    
    # Phase 2: SLSQP polish
    if time_left() > 120:
        h_new, c5_new = optimize_t_var(h_best, max_iter=800, ftol=1e-12)
        if c5_new < c5:
            h_best = h_new
            c5 = c5_new
    
    # Phase 3: Diverse restarts at current resolution
    rng = np.random.RandomState(seed)
    for trial in range(8):
        if time_left() < 60:
            break
        if trial % 2 == 0:
            h_try = random_init(n, seed + 500 + trial)
        else:
            h_try = structured_init(n, seed + 600 + trial, n_blocks=3 + trial % 4)
        h_new, c5_new = fista_projected(h_try, 0.01, n_iter=3000, lr_init=0.003)
        if c5_new < c5 and time_left() > 40:
            h_new2, c5_new2 = fista_projected(h_new, 0.003, n_iter=2000, lr_init=0.002)
            if c5_new2 < c5:
                h_best = h_new2
                c5 = c5_new2
    
    # Phase 4: Resolution upgrade
    for n_new in [270, 360]:
        if time_left() < 250:
            break
        h_interp = interp_to_resolution(h_best, n_new)
        c5_cur = compute_c5(h_interp)
        h_cur = h_interp.copy()
        
        for temp in [0.02, 0.01, 0.005, 0.002, 0.001]:
            if time_left() < 100:
                break
            n_iter = min(4000, max(1500, int(time_left() * 5)))
            h_new, c5_new = fista_projected(h_cur, temp, n_iter=n_iter, lr_init=0.003)
            if c5_new < c5:
                h_best = h_new
                c5 = c5_new
                h_cur = h_new
            else:
                h_cur = h_best.copy()
        
        # SLSQP polish at new resolution (only if n is manageable)
        if time_left() > 150 and n_new <= 270:
            h_new, c5_new = optimize_t_var(h_best, max_iter=1000, ftol=1e-12)
            if c5_new < c5:
                h_best = h_new
                c5 = c5_new
        elif time_left() > 120:
            # For larger n, use another FISTA pass with very low temp
            h_new, c5_new = fista_projected(h_best, 0.0005, n_iter=5000, lr_init=0.002)
            if c5_new < c5:
                h_best = h_new
                c5 = c5_new
    
    # Phase 5: Final SLSQP polish
    if time_left() > 180:
        h_new, c5_new = optimize_t_var(h_best, max_iter=2000, ftol=1e-13)
        if c5_new < c5:
            h_best = h_new
            c5 = c5_new
    elif time_left() > 60:
        h_new, c5_new = fista_projected(h_best, 0.0003, n_iter=5000, lr_init=0.002)
        if c5_new < c5:
            h_best = h_new
            c5 = c5_new
    
    # Final cleanup
    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = project_feasible(h_best, len(h_best) * 0.5)
    c5_final = compute_c5(h_best)
    return h_best, c5_final, len(h_best)
