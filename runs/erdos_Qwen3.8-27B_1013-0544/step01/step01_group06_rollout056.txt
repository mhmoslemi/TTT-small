import numpy as np
import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def smooth_max(corr, alpha):
    m = np.max(corr)
    shifted = corr - m
    # Use log-sum-exp for numerical stability
    amax = np.max(shifted * alpha)
    if amax > 50:
        return m + (np.log(np.sum(np.exp(shifted * alpha - amax))) + amax) / alpha
    return m + np.log(np.sum(np.exp(shifted * alpha))) / alpha

def smooth_max_grad(corr, alpha):
    m = np.max(corr)
    shifted = (corr - m) * alpha
    amax = np.max(shifted)
    exp_shifted = np.exp(shifted - amax)
    s = np.sum(exp_shifted)
    w = exp_shifted / s
    return w  # weights for the softmax

def grad_corr_single(h, k, n):
    g = np.zeros(n)
    if k >= 0:
        g[:n-k] += (1.0 - h[k:])
        if k > 0:
            g[k:] -= h[:n-k]
    else:
        p = -k
        g[p:] += (1.0 - h[:n-p])
        if p > 0:
        g[:n-p] -= h[p:]
    return g

def compute_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    w = smooth_max_grad(corr, alpha)
    grad = np.zeros(n)
    # Only process significant weights for efficiency
    active = np.where(w > 1e-8)[0]
    for idx in active:
        k = idx - (n - 1)
        grad += w[idx] * grad_corr_single(h, k, n)
    return grad * (2.0 / n)

def project_simplex_box(h, target_sum):
    lo, hi = -1.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def optimize_nesterov(h0, target_sum, deadline, rng, alpha_init=100):
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)  # momentum
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    lr = 0.01
    alpha = alpha_init
    beta = 0.9  # momentum
    patience = 0
    stall = 0
    
    while time.time() < deadline:
        grad = compute_grad(h, alpha)
        v = beta * v + (1 - beta) * grad  # or v = beta * v + grad
        h_new = h - lr * v
        h_new = project_simplex_box(h_new, target_sum)
        
        # Backtracking linesearch
        new_c5 = compute_c5(h_new)
        if new_c5 < best_c5 - 1e-12:
            h = h_new
            best_h = h.copy()
            best_c5 = new_c5
            patience = 0
            if lr < 0.1:
                lr *= 1.02
        else:
            patience += 1
            lr *= 0.9
            if patience > 50:
                # Perturb and restart
                pert = h + rng.randn(n) * 0.05
                h_new = project_simplex_box(pert, target_sum)
                pc5 = compute_c5(h_new)
                if pc5 < best_c5:
                    h = h_new
                    best_h = h_new.copy()
                    best_c5 = pc5
                v = np.zeros(n)
                lr = 0.01
                alpha = min(alpha * 1.5, 500)
                patience = 0
                stall += 1
                if stall > 10:
                    break
    
    return best_h, best_c5

    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

    for _ in range(50):

        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def smooth_weights(corr, alpha):
    m = np.max(corr)
    s = (corr - m) * alpha
    amax = np.max(s)
    e = np.exp(s - amax)
    return e / e.sum()

def compute_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    w = smooth_weights(corr, alpha)
    grad = np.zeros(n)
    active = np.where(w > 1e-10)[0]
    for idx in active:
        k = idx - (n - 1)
        if k >= 0:
            grad[:n-k] += w[idx] * (1.0 - h[k:])
            if k > 0:
                grad[k:] -= w[idx] * h[:n-k]
        else:
            p = -k
            grad[p:] += w[idx] * (1.0 - h[:-p])
            grad[:n-p] -= w[idx] * h[p:]
    return grad * (2.0 / n)

    lr = 0.01
    alpha = alpha_init
    beta = 0.85
    patience = 0
    stall = 0

            best_c5 = c5_trial

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def project_simplex_box(h, target_sum):
    lo, hi = -1.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def symmetrize(h):
    n = len(h)
    h = 0.5 * (h + h[::-1].copy())
    return h

def smooth_weights(corr, alpha):
    m = np.max(corr)
    s = (corr - m) * alpha
    amax = np.max(s)
    e = np.exp(s - amax)
    return e / np.sum(e)

def compute_grad(h, weights):
    n = len(h)
    grad = np.zeros(n)
    active = np.where(weights > 1e-10)[0]
    for idx in active:
        k = int(idx - (n - 1))
        w = weights[idx]
        if k >= 0:
            grad[:n-k] += w * (1.0 - h[k:])
            if k > 0:
                grad[k:] -= w * h[:n-k]
        else:
            p = -k
            grad[p:] += w * (1.0 - h[:-p])
            grad[:n-p] -= w * h[p:]
    return grad * (2.0 / n)
