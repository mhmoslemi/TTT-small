def compute_gradient_shifted(h, dx, weights):
    """Compute gradient of weighted sum of shift values.
    
    weights: array of length 2n-1, same as correlation output.
    weights[k] corresponds to shift k = idx - (n-1).
    """
    n = len(h)
    grad = np.zeros(n)
    
    for idx in range(2 * n - 1):
        w = weights[idx]
        if abs(w) < 1e-15:
            continue
        k = idx - (n - 1)
        
        if k == 0:
            grad += w * dx * (1.0 - 2.0 * h)
        elif k > 0:
            # g_k = dx * sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])
            # grad[j] = dx * (1_{j<=n-1-k} - h[j+k]*1_{j<=n-1-k} - h[j-k]*1_{j>=k})
            upper = n - 1 - k
            for j in range(n):
                val = 0.0
                if j <= upper:
                    val = 1.0 - h[j + k]
                if j >= k:
                    val -= h[j - k]
                grad[j] += w * dx * val
        else:
            kk = -k
            # g_{-kk} = dx * sum_{i=kk}^{n-1} h[i]*(1-h[i-kk])
            # grad[j] = dx * (1_{j>=kk} - h[j-kk]*1_{j>=kk} - h[j+kk]*1_{j+kk<=n-1})
            for j in range(n):
                val = 0.0
                if j >= kk:
                    val = 1.0 - h[j - kk]
                if j + kk <= n - 1:
                    val -= h[j + kk]
                grad[j] += w * dx * val
    
    return grad

Now let me think about the overall algorithm:

**Algorithm: Smoothed max with adaptive epsilon + momentum**

1. Start from initial_h_values
2. For decreasing epsilon values:
   a. Compute smooth objective: f_ε(h) = ε * log(Σ exp(g_k/ε))
   b. Gradient: ∇f_ε = (1/Σ exp) * Σ exp(g_k/ε) * ∇g_k
   c. Use Nesterov momentum to optimize
   d. Track best C5 found
3. Final polishing with SLSQP on the best found

The smooth max gradient is:
∇f_ε = Σ_k p_k * ∇g_k where p_k = exp(g_k/ε) / Σ_j exp(g_j/ε)

This is a convex combination of the individual gradients, weighted by how "close to max" each shift is.

As ε → 0, p_k concentrates on the argmax shifts, and we recover the subgradient of the true max.

Let me also add a **Barzili-Borwein step size** adaptive mechanism:
step_size = <s, ys> / <ys, ys> where s = h_new - h_old, ys = g_new - g_old

This often gives better convergence than fixed or backtracking stepsizes.

    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

    return np.max(compute_all_shifts(h, dx))

    target = n / 2.0
    h = h.copy()
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - target

        else:

    h = np.clip(h, 0.0, 1.0)
    diff = np.sum(h) - target
    if abs(diff) > 1e-14:

        if np.any(free):

            h = np.clip(h, 0.0, 1.0)

def compute_gradient(h, dx, weights):
    n = len(h)
    grad = np.zeros(n)
    # k=0
    w0 = weights[n-1]
    if abs(w0) > 1e-15:
        grad += w0 * dx * (1.0 - 2.0 * h)
    for k in range(1, n):
        # Positive shift k
        wk = weights[n-1+k]
        if abs(wk) > 1e-15:
            g = np.zeros(n)
            g[:n-k] += 1.0 - h[k:]
            g[k:] += -h[:n-k]
            grad += wk * dx * g
        # Negative shift -k
        wk_neg = weights[n-1-k]
        if abs(wk_neg) > 1e-15:
            g = np.zeros(n)
            g[k:] += 1.0 - h[:n-k]
            g[:n-k] += -h[k:]
            grad += wk_neg * dx * g
    return grad

def smooth_max_gradient(h, dx, epsilon):
    shifts = compute_all_shifts(h, dx)
    # Softmax weights
    max_s = np.max(shifts)
    log_weights = (shifts - max_s) / epsilon
    log_weights -= np.max(log_weights)
    exp_w = np.exp(log_weights)
    p = exp_w / np.sum(exp_w)
    grad = compute_gradient(h, dx, p)
    f_val = max_s + epsilon * np.log(np.sum(exp_w))
    return f_val, grad, shifts, p

def nesterov_optimize(h_start, n, dx, max_iter=500, eps_start=0.05, eps_end=0.001):
    h = h_start.copy()
    y = h.copy()  # Nesterov lookahead
    momentum = 0.9
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    n_total = max_iter
    for it in range(n_total):
        # Anneal epsilon
        progress = it / max(1, n_total - 1)
        eps = eps_start * (eps_end / eps_start) ** progress
        
        # Nesterov: evaluate gradient at y (lookahead point)
        f_y, grad_y, shifts_y, p_y = smooth_max_gradient(y, dx, eps)
        
        # Step
        t_k = (1 + np.sqrt(1 + 4 * momentum**2)) / 2
        step = 2.0 / (t_k + 1)  # approximate step size
        
        h_new = y - step * grad_y
        h_new = project_feasible(h_new, n)
        
        # Update momentum (Nesterov)
        beta_k = (1 + np.sqrt(1 + 4 * momentum**2)) / 2
        y = h_new + ((momentum - 1) / beta_k) * (h_new - h)
        y = project_feasible(y, n)
        
        h = h_new
        
        c5_val = compute_c5(h, dx)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
    
    return best_h, best_c5

def slsqp_polish(h_start, n, dx, target_sum, max_iter=200):
    bounds = [(0.0, 1.0)] * n
    
    def objective(x):
        return compute_c5(x, dx)
    
    def objective_grad(x):
        shifts = compute_all_shifts(x, dx)
        max_val = np.max(shifts)
        active = shifts > max_val - 1e-8
        n_act = np.sum(active)
        if n_act == 0:
            active = shifts > max_val - 1e-4
            n_act = np.sum(active)
        weights = np.zeros(2*n-1)
        weights[active] = 1.0 / n_act
        return compute_gradient(x, dx, weights)
    
    constraint = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                  'jac': lambda x: np.ones(n)}]
    
    res = minimize(objective, h_start, jac=objective_grad,
                   method='SLSQP', bounds=bounds, constraints=constraint,
                   options={'maxiter': max_iter, 'ftol': 1e-14})
    return project_feasible(res.x, n)
