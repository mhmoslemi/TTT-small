        method='SLSQP',
        bounds=bounds,
        constraints=constraint,

    corr = np.correlate(h, 1.0-h, mode='full') * dx
    max_c = np.max(corr)
    # Log-sum-exp for numerical stability
    shifted = corr - max_c
    log_sum = np.log(np.sum(np.exp(beta * shifted))) + beta * max_c
    return log_sum / beta

def c5_smooth_grad(h, dx, beta=100):

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - target
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            h[free] -= diff / np.sum(free)
        else:
            h -= diff / n
    h = np.clip(h, 0.0, 1.0)
    return h

def c5_smoothed(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    mc = np.max(corr)
    return mc + np.log(np.sum(np.exp(beta * (corr - mc)))) / beta

def c5_smoothed_grad(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    mc = np.max(corr)
    p = np.exp(beta * (corr - mc))
    p /= p.sum()
    r = np.zeros(2 * n - 1)
    for i in range(2 * n - 1):
        r[i] = p[i] + p[2 * (n - 1) - i]
    S = np.zeros(n)
    for i in range(2 * n - 1):
        m = i - (n - 1)
        if m >= 0:
            S[:n - m] += r[i] * h[m:]
        else:
            S[-m:] += r[i] * h[:n + m]
    return dx * (1.0 - S)

def slsqp_smoothed(h_start, n, dx, target_sum, beta, max_iter=400, ftol=1e-12):
    bounds = [(0.0, 1.0)] * n
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)},)
    res = minimize(c5_smoothed, h_start,
                   jac=lambda x: c5_smoothed_grad(x, dx, beta),
                   args=(dx, beta),
                   method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': ftol})
    return project_feasible(res.x, n)

def slsqp_direct(h_start, n, dx, target_sum, max_iter=400):
    bounds = [(0.0, 1.0)] * n
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},)
    res = minimize(lambda x: compute_c5(x, dx), h_start,
                   method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12})
    return project_feasible(res.x, n)

def targeted_gradient_step(h, n, dx, worst_k, lr, rng):
    """One step reducing the correlation at worst_k specifically."""
    s = worst_k - (n - 1)
    grad = np.zeros(n)
    for j in range(n):
        g = 1.0
        if 0 <= j + s < n:
            g -= h[j + s]
        if 0 <= j - s < n:
            g -= h[j - s]
        grad[j] = g
    grad *= dx
    grad -= np.mean(grad)
    h_new = h + lr * (-grad)
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_feasible(h_new, n)
    return h_new

def targeted_gradient_descent(h_start, n, dx, max_iter=500, lr_init=0.005, decay=0.995):
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    lr = lr_init
    for _ in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        worst_k = np.argmax(corr)
        h = targeted_gradient_step(h, n, dx, worst_k, lr, None)
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            lr = lr_init
        else:
            lr *= decay
            if lr < 1e-6:
                break
    return best_h, best_c5
