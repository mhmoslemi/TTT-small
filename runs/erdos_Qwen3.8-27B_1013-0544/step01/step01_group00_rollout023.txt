```python
def lse_gradient(h, T, dx):
    n = len(h)
    corr = np.correlate(h, 1.0-h, mode='full') * dx
    # softmax weights
    shift = corr - np.max(corr)
    w = np.exp(T * shift)
    w /= np.sum(w)
    # w has length 2n-1, indexed by m
    # k = m - (n-1)
    
    # grad[j] = sum_m w[m] * [(1-h[j-(m-n+1)]) * 1_{valid} - h[j+(m-n+1)] * 1_{valid}]
    
    grad = np.zeros(n)
    for m in range(2*n-1):
        k = m - (n-1)
        # First part: (1-h[j-k]) for valid j-k in [0,n)
        # j ranges from max(0,k) to min(n-1, n-1+k)
        if k >= 0:
            j_start = k
            j_end = n  # exclusive
        else:
            j_start = 0
            j_end = n + k + 1
        if j_end > j_start:
            grad[j_start:j_end] += w[m] * (1.0 - h[j_start-k:j_end-k])
        
        # Second part: -h[j+k] for valid j+k in [0,n)
        # j ranges from max(0,-k) to min(n-1, n-1-k)
        if k >= 0:
            jj_start = 0
            jj_end = n - k
        else:
            jj_start = -k
            jj_end = n
        if jj_end > jj_start:
            grad[jj_start:jj_end] -= w[m] * h[jj_start+k:jj_end+k]
    
    return grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_feasible(h, target):
    """Project h onto {h: 0<=h<=1, sum(h)=target} via bisection on tau."""
    h = h.copy()
    # Bisection on tau: find tau such that sum(clip(h-tau, 0, 1)) = target
    lo, hi = -1.0, 2.0  # tau range
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2, 0.0, 1.0)

def lse_value_and_grad(h, T, dx, power=1.0):
    """Compute weighted LSE value and gradient."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    # Apply power weighting to emphasize worst cases
    if power != 1.0:
        m = np.max(corr)
        corr_w = corr.copy()
        corr_w = m - (m - corr) ** power  # this doesn't make sense
    # Actually let's just use the standard LSE
    shift = corr - np.max(corr)
    exp_T = np.exp(T * shift)
    w = exp_T / np.sum(exp_T)
    lse_val = (np.max(corr) + np.log(np.sum(exp_T))) / T
    
    # Gradient
    grad = np.zeros(n)
    for m_idx in range(2*n-1):
        k = m_idx - (n-1)
        wm = w[m_idx]
        # Part 1: wm * (1-h[j-k]) for j where 0<=j-k<n
        if k >= 0:
            js, je = k, n
        else:
            js, je = 0, n+k
        if je > js:
            grad[js:je] += wm * (1.0 - h[js-k:je-k])
        # Part 2: -wm * h[j+k] for j where 0<=j+k<n
        if k >= 0:
            js2, je2 = 0, n-k
        else:
            js2, je2 = -k, n
        if je2 > js2:
            grad[js2:je2] -= wm * h[js2+k:je2+k]
    
    return lse_val, grad

def projected_gradient(h0, T, dx, max_iter=2000, tol=1e-10):
    """Projected gradient descent with adaptive step size."""
    n = len(h0)
    target = n * 0.5
    h = project_feasible(h0.copy(), target)
    lr = 1.0 / (T * 4 * dx + 1e-10)  # initial step size
    
    for it in range(max_iter):
        val, grad = lse_value_and_grad(h, T, dx)
        # Gradient step
        h_new = h - lr * grad
        h_new = project_feasible(h_new, target)
        
        # Check convergence
        step_norm = np.linalg.norm(h_new - h)
        if step_norm < tol:
            break
        
        # Simple adaptive: if LSE increased, reduce lr; else increase slightly
        val_new, _ = lse_value_and_grad(h_new, T, dx)
        if val_new > val:
            lr *= 0.8
        else:
            lr = min(lr * 1.05, 1.0 / (T * dx))
        
        h = h_new
    
    return h

def optimize_lse(h0, T, dx, max_iter=1000):
    """Optimize LSE objective via projected gradient."""
    n = len(h0)
    target = n * 0.5
    h = project_feasible(h0.copy(), target)
    
    # Use backtracking line search
    lr = 1.0 / (T * 2 * dx + 1e-10)
    val, grad = lse_value_and_grad(h, T, dx)
    
    for it in range(max_iter):
        h_new = project_feasible(h - lr * grad, target)
```

```python
        if wm < 1e-15:
            continue
        k = m_idx - (n - 1)
        if k >= 0:
            js, je = k, n
        else:
            js, je = 0, n + k
        if je > js:
            grad[js:je] += wm * (1.0 - h[js-k:je-k])
        if k >= 0:
            js2, je2 = 0, n - k
        else:
            js2, je2 = -k, n
        if je2 > js2:
            grad[js2:je2] -= wm * h[js2+k:je2+k]
    return lse_val, grad


def optimize_lse_momentum(h0, T, dx, max_iter=5000, momentum=0.85):
    n = len(h0)
    target = n * 0.5
    h = project_feasible(h0.copy(), target)
    h_prev = h.copy()
    lr = 1.0 / (T * 4.0 * dx)
    val, grad = lse_grad(h, T, dx)
    for it in range(max_iter):
        h_raw = h - lr * grad + momentum * (h - h_prev)
        h_new = project_feasible(h_raw, target)
        val_new, grad_new = lse_grad(h_new, T, dx)
        bt = 0
        while val_new > val + 1e-14 and bt < 12:
            lr *= 0.6
            h_raw = h - lr * grad + momentum * (h - h_prev)
            h_new = project_feasible(h_raw, target)
            val_new, grad_new = lse_grad(h_new, T, dx)
            bt += 1
        if bt >= 12:
            break
        h_prev = h
        h = h_new
        val = val_new
        grad = grad_new
        step = np.linalg.norm(h - h_prev)
        if step < 1e-13:
            break
    return h


def optimize_slsqp_tvar(h0, max_iter=500, ftol=1e-11):
    n = len(h0)
    c5_cur = compute_c5(h0)

    def obj(v):
        return v[-1]

    def ineq(v):
        h = v[:n]
        t = v[-1]
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        return t - corr

    def eq_fun(v):
        return np.sum(v[:n]) - n * 0.5

    def eq_jac(v):
        j = np.zeros(len(v))
        j[:n] = 1.0
        return j

    v0 = np.append(h0.copy(), c5_cur + 0.05)
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [
        {'type': 'ineq', 'fun': ineq},
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}
    ]
    res = minimize(obj, v0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': ftol})
    h_new = project_feasible(res.x[:n], n * 0.5)
    return h_new, compute_c5(h_new)


def interp_h(h_old, n_new):
    x_old = np.linspace(0, 1, len(h_old), endpoint=False)
    x_new = np.linspace(0, 1, n_new, endpoint=False)
    return project_feasible(np.interp(x_new, x_old, h_old), n_new * 0.5)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - t0)

    h = np.array(initial_h_values, dtype=float)
    n = len(h)
    dx = 2.0 / n
    target = n * 0.5
    h = project_feasible(h, target)
    c5 = compute_c5(h)
    h_best = h.copy()

    # Phase 1: LSE momentum, decreasing temperature
    temps = [50, 100, 300, 1000, 3000, 10000]
    for T in temps:
        if time_left() < 180:
            break
        h_new = optimize_lse_momentum(h, T, dx, max_iter=4000, momentum=0.85)
        c5_new = compute_c5(h_new)
        if c5_new < c5:
            h, c5 = h_new, c5_new
            h_best = h.copy()

    # Phase 2: SLSQP t-variable polish
    if time_left() > 200:
        h_new, c5_new = optimize_slsqp_tvar(h, max_iter=600, ftol=1e-11)
        if c5_new < c5:
            h, c5 = h_new, c5_new
            h_best = h.copy()
        # Follow-up LSE from SLSQP result
        for T in [2000, 5000, 20000]:
            if time_left() < 100:
                break
            h_new = optimize_lse_momentum(h, T, dx, max_iter=3000, momentum=0.8)
            c5_new = compute_c5(h_new)
            if c5_new < c5:
                h, c5 = h_new, c5_new
                h_best = h.copy()

    # Phase 3: Diverse restarts
    for i in range(14):
        if time_left() < 40:
            break
        if i < 5:
            h_p = h_best + rng.randn(n) * (0.003 + 0.004 * i)
        elif i < 9:
            h_p = h_best.copy()
            n_flip = max(2, n // 25)
            idx = rng.choice(n, n_flip, replace=False)
            h_p[idx] = 1.0 - h_p[idx]
        else:
            x = np.linspace(0, 4 * np.pi, n)
            freq = 1 + 0.5 * i
            h_p = h_best + 0.04 * np.sin(freq * x + rng.randn() * 2 * np.pi)
        h_p = project_feasible(h_p, target)
        h_new = optimize_lse_momentum(h_p, 2000, dx, max_iter=2500, momentum=0.85)
        c5_new = compute_c5(h_new)
        if c5_new < c5:
            h, c5 = h_new, c5_new
            h_best = h.copy()

    # Phase 4: Resolution increase
    for n_new in [360, 720, 1440]:
        if time_left() < 280:
            break
        h_interp = interp_h(h_best, n_new)
        dx_new = 2.0 / n_new
        target_new = n_new * 0.5
        h_cur = h_interp
        c5_cur = compute_c5(h_cur)
        for T in [300, 1000, 5000]:
            if time_left() < 150:
                break
            h_new = optimize_lse_momentum(h_cur, T, dx_new, max_iter=3000, momentum=0.85)
            c5_new = compute_c5(h_new)
            if c5_new < c5:
                h_cur = h_new
                c5 = c5_new
                h_best = h_new.copy()
        if time_left() > 200:
            h_new, c5_new = optimize_slsqp_tvar(h_cur, max_iter=500, ftol=1e-11)
            if c5_new < c5:
                h_cur = h_new
                c5 = c5_new
                h_best = h_new.copy()
            h_new = optimize_lse_momentum(h_cur, 10000, 2.0/len(h_cur),
                                          max_iter=2000, momentum=0.8)
            c5_new = compute_c5(h_new)
            if c5_new < c5:
                h_best = h_new
                c5 = c5_new

    # Phase 5: Final SLSQP polish
    if time_left() > 60:
        h_new, c5_new = optimize_slsqp_tvar(h_best, max_iter=500, ftol=1e-12)
        if c5_new < c5:
            h_best = h_new
            c5 = c5_new

    h_best = project_feasible(h_best, len(h_best) * 0.5)
    c5_final = compute_c5(h_best)
    return h_best, c5_final, len(h_best)
```
