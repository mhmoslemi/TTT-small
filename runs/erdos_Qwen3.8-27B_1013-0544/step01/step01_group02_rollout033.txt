        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-14:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def obj_active(h, n, alpha, active):
    dx = 2.0 / n
    c_full = np.correlate(h, 1 - h, mode='full') * dx
    c = c_full[active]
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def sum_con(h, n):
    return np.sum(h) - n / 2.0

def sum_con_jac(h, n):
    return np.ones(n)

def get_active(h, n, gap=0.003, top_k=40):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    active = np.where(c >= m - gap)[0]
    if len(active) > top_k:
        active = np.argsort(c)[::-1][:top_k]
    return active

def slsqp_step(h, n, alpha, bounds, cons, active=None, maxiter=2000):
    if active is None:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
    else:
        res = minimize(obj_active, h, args=(n, alpha, active), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
    return res.x

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 80
        h0 = np.full(n, 0.5)

    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, n)
    best_n = n

    def elapsed():
        return time.time() - t0

    def remaining():
        return budget_s - elapsed()

    # Phase 1: Continuation SLSQP (full objective)
    alphas = [10, 25, 50, 100, 200, 400, 800, 1600, 3200, 6400, 12800]
    for alpha in alphas:
        if remaining() < 20:
            break
        bounds = [(0.0, 1.0)] * best_n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (best_n,)}]
        try:
            x = slsqp_step(best_h, best_n, alpha, bounds, cons, maxiter=3000)
            h_try = project_feasible(x, best_n)
            c = compute_c5(h_try, best_n)
            if c < best_c5 - 1e-15:
                best_h, best_c5 = h_try, c
        except Exception:
            pass

    # Phase 2: Active-set optimization (iterative)
    for alpha in [200, 500, 1000, 2000, 5000, 10000]:
        if remaining() < 20:
            break
        for iteration in range(4):
            if remaining() < 10:
                break
            active = get_active(best_h, best_n, gap=0.002, top_k=30)
            bounds = [(0.0, 1.0)] * best_n
            cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (best_n,)}]
            try:
                x = slsqp_step(best_h, best_n, alpha, bounds, cons, active=active, maxiter=3000)
                h_try = project_feasible(x, best_n)
                c = compute_c5(h_try, best_n)
                if c < best_c5 - 1e-15:
                    best_h, best_c5 = h_try, c
            except Exception:
                pass

    # Phase 3: Multi-n exploration
    for n_trial in [100, 120, 64, 96]:
        if remaining() < 100:
            break
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_trial, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h, period=2.0)
        h_interp = project_feasible(h_interp, n_trial)
        c_interp = compute_c5(h_interp, n_trial)
        local_h = h_interp.copy()
        local_c5 = c_interp
        local_n = n_trial

        bounds_l = [(0.0, 1.0)] * local_n
        cons_l = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (local_n,)}]
        for alpha in [50, 200, 800, 3000, 8000, 15000]:
            if remaining() < 15:
                break
            try:
                x = slsqp_step(local_h, local_n, alpha, bounds_l, cons_l, maxiter=2500)
                h_try = project_feasible(x, local_n)
                c = compute_c5(h_try, local_n)
                if c < local_c5 - 1e-15:
                    local_h, local_c5 = h_try, c
            except Exception:
                pass
        # Active-set refinement at this n
        for alpha in [2000, 5000, 10000]:
            if remaining() < 10:
                break
            active_l = get_active(local_h, local_n, gap=0.002, top_k=30)
            try:
                x = slsqp_step(local_h, local_n, alpha, bounds_l, cons_l, active=active_l, maxiter=2000)
                h_try = project_feasible(x, local_n)
                c = compute_c5(h_try, local_n)
                if c < local_c5 - 1e-15:
                    local_h, local_c5 = h_try, c
            except Exception:
                pass

        if local_c5 < best_c5:
            best_h, best_c5, best_n = local_h, local_c5, local_n

    # Phase 4: Perturbation + re-optimization (escape local optima)
    for attempt in range(8):
        if remaining() < 40:
            break
        scales = [0.02, 0.05, 0.1, 0.15, 0.25, 0.35, 0.5, 0.7]
        scale = scales[min(attempt, len(scales)-1)]
        h_pert = best_h + rng.normal(0, scale, size=best_n)
        h_pert = project_feasible(h_pert, best_n)
        c_pert = compute_c5(h_pert, best_n)
        bounds_p = [(0.0, 1.0)] * best_n
        cons_p = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (best_n,)}]
        for alpha in [200, 1000, 5000]:
            if remaining() < 10:
                break
            try:
                x = slsqp_step(h_pert, best_n, alpha, bounds_p, cons_p, maxiter=1500)
                h_try = project_feasible(x, best_n)
                c = compute_c5(h_try, best_n)
                if c < c_pert - 1e-15:
                    h_pert, c_pert = h_try, c
            except Exception:
                break
        if c_pert < best_c5 - 1e-15:
            best_h, best_c5 = h_pert, c_pert

    # Phase 5: Coordinate descent (two-swap with golden section)
    if remaining() > 40:
        n_cd = best_n
        dx_cd = 2.0 / n_cd
        max_swaps = 30000
        swap_count = 0
        no_improve = 0
        while remaining() > 20 and swap_count < max_swaps and no_improve < 3000:
            swap_count += 1
            i = rng.randint(n_cd)
            j = rng.randint(n_cd)
            if i == j:
                no_improve += 1
                continue
            hi = best_h[i]
            hj = best_h[j]
            d_lo = max(-hi, hj - 1.0) + 1e-10
            d_hi = min(1.0 - hi, hj) - 1e-10
            if d_hi <= d_lo:
                no_improve += 1
                continue
            # Evaluate at a few strategic points including golden section points
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            span = d_hi - d_lo
            candidates = [0.0, span * 0.5, span * (1 - gr), span * gr,
                          span * 0.25, span * 0.75, 0.01, -0.01, 0.02, -0.02]
            best_d = 0.0
            best_c = compute_c5(best_h, n_cd)
            for d in candidates:
                if d < d_lo or d > d_hi:
                    continue
                h_try = best_h.copy()
                h_try[i] = hi + d
                h_try[j] = hj - d
                c = compute_c5(h_try, n_cd)
                if c < best_c - 1e-15:
                    best_c = c
                    best_d = d
            if best_d != 0.0:
                best_h[i] = hi + best_d
                best_h[j] = hj - best_d
                best_c5 = best_c
                no_improve = 0
            else:
                no_improve += 1

    # Phase 6: Final high-alpha polish
    for alpha in [5000, 10000, 20000, 50000]:
        if remaining() < 8:
            break
        bounds_f = [(0.0, 1.0)] * best_n
        cons_f = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (best_n,)}]
        try:
            x = slsqp_step(best_h, best_n, alpha, bounds_f, cons_f, maxiter=5000)
            h_try = project_feasible(x, best_n)
            c = compute_c5(h_try, best_n)
            if c < best_c5:
                best_h, best_c5 = h_try, c
        except Exception:
            pass

    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return best_h, best_c5, best_n
