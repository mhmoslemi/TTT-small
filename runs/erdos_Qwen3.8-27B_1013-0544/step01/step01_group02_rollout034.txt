    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:n//2]] = 1.0
    return b

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            t = room.sum()
            if t < 1e-15: break
            h += (diff / t) * room
        else:
            t = h.sum()
            if t < 1e-15: break
            h += (diff / t) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def sum_con(h, n):
    return np.sum(h) - n / 2.0

def sum_con_jac(h, n):
    return np.ones(n)

def binary_ls(h, n, rng, max_iters=5000, time_budget=60.0):
    """Steepest descent local search on binary sequences."""
    h = to_binary(h, n).copy()
    best_c5 = compute_c5(h, n)
    best_h = h.copy()
    t_start = time.time()
    stalls = 0
    
    for it in range(max_iters):
        if time.time() - t_start > time_budget:
            break
        
        best_move = None
        best_c = best_c5
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        
        for i in ones:
            for j in zeros:
                h[i] = 0.0; h[j] = 1.0
                c = compute_c5(h, n)
                h[i] = 1.0; h[j] = 0.0
                if c < best_c - 1e-15:
                    best_c = c
                    best_move = (i, j)
        
        if best_move:
            i, j = best_move
            h[i], h[j] = 0.0, 1.0
            best_c5 = best_c
            best_h = h.copy()
            stalls = 0
        else:
            stalls += 1
            if stalls >= 15:
                # Random kick
                for _ in range(rng.randint(2, 5)):
                    ones = np.where(h == 1.0)[0]
                    zeros = np.where(h == 0.0)[0]
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                    h[i], h[j] = 0.0, 1.0
                best_c5 = compute_c5(h, n)
                stalls = 0
    
    return best_h, best_c5

def simulated_annealing(h, n, rng, time_budget=120.0):
    """SA on binary sequences with exactly n/2 ones."""
    h = to_binary(h, n).copy()
    cur_c5 = compute_c5(h, n)
    best_h = h.copy()
    best_c5 = cur_c5
    t_start = time.time()
    T = 0.05
    T_min = 0.001
    cooling = 0.9995
    
    while T > T_min and time.time() - t_start < time_budget:
        # Generate move: swap a random 1 with a random 0
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h[i], h[j] = 0.0, 1.0
        new_c5 = compute_c5(h, n)
        delta = new_c5 - cur_c5
        
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0  # undo
        
        T *= cooling
    
    return best_h, best_c5

def slsqp_polish(h, n, time_budget=60.0):
    """SLSQP on continuous relaxation."""
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    best_h = h.copy()
    best_c5 = compute_c5(best_h, n)

            c = compute_c5(h_try, n)
            if c < best_c5:
                best_h = h_try
                best_c5 = c
        except:
            pass
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:n//2]] = 1.0
    return b

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            t = room.sum()
            if t < 1e-15:
                break
            h += (diff / t) * room
        else:
            t = h.sum()
            if t < 1e-15:
                break
            h += (diff / t) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def obj_topk_smooth(h, n, k, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    cs = np.sort(c)[::-1][:k]
    m = cs.max()
    shifted = alpha * (cs - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def sum_con(h, n):
    return np.sum(h) - n / 2.0

def sum_con_jac(h, n):
    return np.ones(n)

def binary_ls(h, n, rng, max_iters=3000, time_budget=25.0):
    h = to_binary(h, n).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    t0 = time.time()
    stalls = 0
    for _ in range(max_iters):
        if time.time() - t0 > time_budget:
            break
        best_c = best_c5
        best_move = None
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        for i in ones:
            for j in zeros:
                h[i], h[j] = 0.0, 1.0
                c = compute_c5(h, n)
                h[i], h[j] = 1.0, 0.0
                if c < best_c - 1e-15:
                    best_c = c
                    best_move = (i, j)
        if best_move:
            i, j = best_move
            h[i], h[j] = 0.0, 1.0
            best_c5 = best_c
            best_h = h.copy()
            stalls = 0
        else:
            stalls += 1
            if stalls >= 8:
                for _ in range(rng.randint(2, 5)):
                    ones = np.where(h == 1.0)[0]
                    zeros = np.where(h == 0.0)[0]
                    if len(ones) == 0 or len(zeros) == 0:
                        break
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                    h[i], h[j] = 0.0, 1.0
                best_c5 = compute_c5(h, n)
                stalls = 0
    return best_h, best_c5

def sa_binary(h, n, rng, time_budget=60.0):
    h = to_binary(h, n).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    cur_c5 = best_c5
    t0 = time.time()
    T = 0.04
    best_T = T
    best_best_c5 = best_c5
    while T > 0.0005 and time.time() - t0 < time_budget:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        new_c5 = compute_c5(h, n)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
        T *= 0.9996
        if T < 0.005 and cur_c5 > best_best_c5 + 0.02:
            T = 0.03
    return best_h, best_c5

def slsqp_multi(h, n, time_budget=60.0):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    best_h = project_feasible(h.copy(), n).astype(float)
    best_c5 = compute_c5(best_h, n)
    t0 = time.time()
    for alpha in [100, 300, 1000, 3000, 10000, 30000]:
        if time.time() - t0 > time_budget:
            break
        try:
            res = minimize(obj_smooth, best_h, args=(n, alpha), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 3000, 'ftol': 1e-18})
            h_try = project_feasible(res.x, n)
            c = compute_c5(h_try, n)
            if c < best_c5 - 1e-15:
                best_h, best_c5 = h_try, c
        except Exception:
            pass
    # Also try top-k objective
    for k in [2, 3]:
        if time.time() - t0 > time_budget:
            break
        for alpha in [500, 2000]:
            if time.time() - t0 > time_budget:
                break
            try:
                res = minimize(obj_topk_smooth, best_h, args=(n, k, alpha),
                             method='SLSQP', bounds=bounds, constraints=cons,
                             options={'maxiter': 2000, 'ftol': 1e-18})
                h_try = project_feasible(res.x, n)
                c = compute_c5(h_try, n)
                if c < best_c5 - 1e-15:
                    best_h, best_c5 = h_try, c
            except Exception:
                pass
    return best_h, best_c5
