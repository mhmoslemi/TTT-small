def frank_wolfe_step(h, grad, n):
    """Linear minimization oracle over {v: 0<=v<=1, sum(v)=n/2}"""
    # Optimal: set v[i]=1 for the n/2 smallest grad[i]
    threshold = np.partition(grad, n//2)[n//2]
    v = (grad <= threshold).astype(float)
    # Handle ties
    if v.sum() != n // 2:
        # Adjust for ties at threshold
        tied = np.where(grad == threshold)[0]
        current_ones = int(v.sum())
        need = n // 2 - current_ones
        if need > 0:
            v[tied[:need]] = 1.0
    return v

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_box_sum(h, target_sum):
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_gradient(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= s
    grad = np.zeros(n)
    active = np.where(w > 1e-8)[0]
    for idx in active:
        k = int(idx) - (n - 1)
        if k >= 0:
            grad[:n-k] += w[idx] * (1.0 - h[k:])
            if k > 0:
                grad[k:] -= w[idx] * h[:n-k]
        else:
            p = -k
            grad[p:] += w[idx] * (1.0 - h[:n-p])
            if p > 0:
                grad[:n-p] -= w[idx] * h[p:]
    return grad * (2.0 / n)

def fw_step(grad, n):
    """Frank-Wolfe linear minimization oracle."""
    target = n // 2
    # Set v[i]=1 for the 'target' smallest grad[i]
    sorted_idx = np.argsort(grad)
    v = np.zeros(n)
    v[sorted_idx[:target]] = 1.0
    return v

    n = len(h0)
    h = project_box_sum(h0.astype(float).copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    stall = 0

        total_iters += 1

            s = fw_step(grad, n)
            t = 0.3  # FW mixing parameter
            h_fw = (1 - t) * h + t * s
            h_fw = project_box_sum(h_fw, target_sum)
            c5_fw = compute_c5(h_fw)

                best_h = h_fw.copy()
                best_c5 = c5_fw
                h = h_fw
            fw_count += 1
            continue

        y = project_box_sum(h + beta * v, target_sum)
        grad = compute_gradient(y, alpha)
        v = beta * v - lr * grad
        h_new = project_box_sum(h + v, target_sum)
        c5_new = compute_c5(h_new)

            best_h = h_new.copy()
            best_c5 = c5_new

    return best_h, best_c5

def make_binary_init(n, rng, pattern='random'):
    h = np.zeros(n)
    if pattern == 'random':
        idx = rng.choice(n, n//2, replace=False)
        h[idx] = 1.0
    elif pattern == 'block':
        # Alternating blocks

        for i in range(0, n, 2*bsize):

            h[rng.choice(ones, min(-diff, len(ones)), replace=False)] = 0.0

def topm_gradient(h, alpha, m=10):
    n = len(h)

    vm = np.max(vals)

            break
        n = len(h0)
        ts = n / 2.0
        h0 = project_box_sum(h0, ts)
        if name == 'parent':
            dl = t0 + min(budget_s * 0.30, 300.0)
        else:
            dl = time.time() + min(70.0, (budget_s - 300.0) / 10.0)
        ho, co = optimize_h(h0, ts, dl, rng, alpha_init=30, topm=8)
        update(ho, co, n)
        hb, cb = try_binary_round(ho, n)
        update(hb, cb, n)

    # Phase 2: interpolate best to higher res
    if best_h is not None:
        for nt in [200, 250, 300]:
            if time.time() - t0 > budget_s - 120:
                break
            if nt <= best_n:
                continue
            hi = interp_to(best_h, best_n, nt)
            hi = project_box_sum(hi, nt / 2.0)
            dl = time.time() + min(100.0, budget_s - (time.time() - t0) - 60)
            ho, co = optimize_h(hi, nt / 2.0, dl, rng, alpha_init=40, topm=6)
            update(ho, co, nt)
            hb, cb = try_binary_round(ho, nt)
            update(hb, cb, nt)

    # Phase 3: final polish
    rem = budget_s - (time.time() - t0)
    if rem > 40 and best_h is not None:
        hl, cl = optimize_h(best_h, len(best_h) / 2.0, t0 + budget_s - 5, rng,
                            alpha_init=100, topm=5)
        update(hl, cl, len(hl))
        hb, cb = try_binary_round(best_h, len(best_h))
        update(hb, cb, len(best_h))

    if best_h is None:
        best_h = make_init(100, rng, 'random')
        best_c5 = compute_c5(best_h)
        best_n = 100

    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
