import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)


def coord_sweep(h, target_sum, max_sweeps=50):
    n = len(h)
    dx = 2.0 / n
    for _ in range(max_sweeps):
        improved = False
        for i in range(n):
            curr = np.correlate(h, 1.0 - h, mode="full") * dx
            h_old = h[i]
            slopes = np.zeros(2 * n - 1)
            for m in range(2 * n - 1):
                lag = m - (n - 1)
                a = 0.0
                if 0 <= i - lag < n:
                    a += (1.0 - h[i - lag]) * dx
                if 0 <= i + lag < n:
                    a -= h[i + lag] * dx
                slopes[m] = a
            intercepts = curr - slopes * h_old
            a_lo, a_hi = 0.0, 1.0
            for _ in range(55):
                m1 = a_lo + (a_hi - a_lo) / 3.0
                m2 = a_hi - (a_hi - a_lo) / 3.0
                f1 = np.max(slopes * m1 + intercepts)
                f2 = np.max(slopes * m2 + intercepts)
                if f1 < f2:
                    a_hi = m2
                else:
                    a_lo = m1
            x_opt = (a_lo + a_hi) / 2.0
            if abs(x_opt - h[i]) > 1e-10:
                h[i] = x_opt
                improved = True
        h = project_sum(h, target_sum)
        if not improved:
            break
    return h


def interpolate_h(h_in, n_in, n_out):
    x_in = np.linspace(0, 2, n_in)
    x_out = np.linspace(0, 2, n_out)
    return np.interp(x_out, x_in, h_in)


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)

    n0 = len(h_init)
    target_sum = n0 / 2.0

    best_h = project_sum(h_init.copy(), target_sum)
    best_c5 = compute_c5(best_h)

    def time_left():
        return budget_s - (time.time() - start_time)

    # Phase 1: Coordinate descent from initial
    best_h = coord_sweep(best_h.copy(), target_sum, max_sweeps=80)
    best_c5 = compute_c5(best_h)

    # Phase 2: Multi-start with diverse perturbations
    no_improve = 0
    attempt = 0
    while time_left() > 60 and no_improve < 40:
        attempt += 1
        h0 = best_h.copy()
        strategy = attempt % 6

        if strategy == 0:
            scale = max(0.005, 0.2 * np.exp(-attempt / 35.0))
            h0 += np.random.randn(n0) * scale
        elif strategy == 1:
            nb = np.random.randint(3, 9)
            bs = n0 // nb
            blocks = []
            for i in range(nb):
                blocks.append(h0[i * bs:min((i + 1) * bs, n0)])
            np.random.shuffle(blocks)
            idx = 0
            for b in blocks:
                h0[idx:idx + len(b)] = b
                idx += len(b)
        elif strategy == 2:
            n_flips = np.random.randint(5, 18)
            idxs = np.random.choice(n0, n_flips, replace=False)
            h0[idxs] = 1.0 - h0[idxs]
        elif strategy == 3:
            noise = np.random.randn(n0)
            kernel = np.exp(-0.5 * np.arange(7) ** 2 * 0.3)
            noise = np.convolve(noise, kernel, mode='same')
            h0 += 0.12 * noise
        elif strategy == 4:
            corr = np.correlate(h0, 1.0 - h0, mode="full")
            top_k = np.argsort(corr)[-4:]
            for k in top_k:
                lag = k - (n0 - 1)
                for j in range(n0):
                    i1, i2 = j - lag, j + lag
                    if 0 <= i1 < n0 and 0 <= i2 < n0:
                        h0[j] += 0.008 * (h0[i1] * (1 - h0[i2]) - (1 - h0[i1]) * h0[i2])
        else:
            mask = np.random.random(n0) < 0.25
            h0[mask] = np.random.random(int(mask.sum())) * 0.6 + 0.2

        h0 = project_sum(h0, target_sum)
        h_new = coord_sweep(h0, target_sum, max_sweeps=30)
        c5 = compute_c5(h_new)
        if c5 < best_c5 - 1e-12:
            best_h, best_c5 = h_new.copy(), c5
            no_improve = 0
        else:
            no_improve += 1

    # Phase 3: Try different resolutions
    if time_left() > 120:
        for n_try in [n0 + 15, n0 - 15, n0 + 30]:
            if time_left() < 60 or n_try < 20:
                break
            ts = n_try / 2.0
            h_interp = interpolate_h(best_h, n0, n_try)
            h_interp = project_sum(h_interp, ts)
            h_try = coord_sweep(h_interp, ts, max_sweeps=40)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-12:
                best_h, best_c5, n0 = h_try.copy(), c5_try, n_try
            # Also try random init at this resolution
            if time_left() > 40:
                h_rand = project_sum(np.random.random(n_try) * 0.8 + 0.1, ts)
                h_rand = coord_sweep(h_rand, ts, max_sweeps=25)
                c5_r = compute_c5(h_rand)
                if c5_r < best_c5 - 1e-12:
                    best_h, best_c5, n0 = h_rand.copy(), c5_r, n_try

    # Phase 4: Additional multi-start at best resolution
    n_best = len(best_h)
    ts_best = n_best / 2.0
    attempt2 = 0
    no_imp2 = 0
    while time_left() > 40 and no_imp2 < 20:
        attempt2 += 1
        h0 = best_h.copy()
        scale = max(0.003, 0.1 * np.exp(-attempt2 / 20.0))
        h0 += np.random.randn(n_best) * scale
        if attempt2 % 3 == 0:
            idxs = np.random.choice(n_best, 8, replace=False)
            h0[idxs] = 1.0 - h0[idxs]
        h0 = project_sum(h0, ts_best)
        h_new = coord_sweep(h0, ts_best, max_sweeps=25)
        c5 = compute_c5(h_new)
        if c5 < best_c5 - 1e-12:
            best_h, best_c5 = h_new.copy(), c5
            no_imp2 = 0
        else:
            no_imp2 += 1

    # Phase 5: Final SLSQP polish
    if time_left() > 15:
        n_fin = len(best_h)
        ts_fin = n_fin / 2.0
        beta = 600.0

        def obj_fin(h):
            corr = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n_fin)
            m = np.max(corr)
            return m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta

        def eq_fun(x):
            return np.sum(x) - ts_fin

        try:
            res = minimize(obj_fin, best_h, method='SLSQP',
                           bounds=Bounds(np.zeros(n_fin), np.ones(n_fin)),
                           constraints=[{'type': 'eq', 'fun': eq_fun}],
                           options={'maxiter': 3000, 'ftol': 1e-15})
            h_new = project_sum(res.x, ts_fin)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new.copy(), c5
        except Exception:
            pass

    # Phase 6: One more coordinate descent pass
    if time_left() > 5:
        h_new = coord_sweep(best_h.copy(), len(best_h) / 2.0, max_sweeps=30)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_h = h_new

    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
