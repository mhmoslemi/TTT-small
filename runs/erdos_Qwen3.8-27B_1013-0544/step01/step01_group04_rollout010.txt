def project_feasible(h, n):
    target = n / 2.0
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - target
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            h[free] -= diff / np.sum(free)

            h -= diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = np.sum(h) - target
    if abs(diff) > 1e-14:

def subgradient_step(h, n, dx, top_m=5):
    """Compute subgradient using top-m active constraints and take a step."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    shifts = np.arange(-(n-1), n)
    # Find top-m shifts by correlation value
    idx = np.argsort(corr)[::-1][:top_m]
    active_shifts = shifts[idx]
    
    grad = np.zeros(n)
    for s in active_shifts:
        for j in range(n):
            if 0 <= j + s < n:
                grad[j] += (1.0 - h[j + s])
            if 0 <= j - s < n:
                grad[j] -= h[j - s]
    grad /= top_m
    return grad, corr

def subgradient_descent(h_init, n, dx, max_iter=5000, alpha0=0.1, beta=0.01, top_m=5, momentum=0.9):
    h = h_init.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    for t in range(1, max_iter + 1):
        grad, corr = subgradient_step(h, n, dx, top_m)
        alpha = alpha0 / (1.0 + beta * t) ** 0.5
        v = momentum * v - alpha * grad
        h_new = h + v
        h = project_feasible(h_new, n)
        c5 = np.max(corr) * dx
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

                h_sl = slsqp_optimize(h_sg, n, dx, max_iter=400)
                c5_sl = compute_c5(h_sl, dx)
                if c5_sl < c5_best_local:
                    c5_best_local = c5_sl
                    h_best_local = h_sl.copy()

        # Try 0-1 rounding
        if elapsed() < t_deadline - 30:
            h_r = try_01_rounding(h_best_local, n, dx, rng)
            h_r2 = slsqp_optimize(h_r, n, dx, max_iter=400)
            c5_r2 = compute_c5(h_r2, dx)
            if c5_r2 < c5_best_local:
                c5_best_local = c5_r2
                h_best_local = h_r2.copy()

        # SLSQP from random perturbations
        for _ in range(3):
            if elapsed() > t_deadline:
                break
            h_p = project_simplex_box(h_best_local + rng.randn(n) * 0.04, n)
            h_p2 = slsqp_optimize(h_p, n, dx, max_iter=300)
            c5_p2 = compute_c5(h_p2, dx)
            if c5_p2 < c5_best_local:
                c5_best_local = c5_p2
                h_best_local = h_p2.copy()

        if c5_best_local < best_c5:
            best_c5 = c5_best_local
            best_h = h_best_local.copy()

    # Final intensive polishing on best solution
    t_polish_end = budget_s - 15
    if best_h is not None:
        n_f = len(best_h)
        dx_f = 2.0 / n_f
        for i in range(20):
            if elapsed() > t_polish_end:
                break
            if i == 0:
                h_try = best_h.copy()
            else:
                scale = 0.02 * max(1.0, 1.0 - i / 20.0)
                h_try = project_simplex_box(best_h + rng.randn(n_f) * scale, n_f)
            h_new = slsqp_optimize(h_try, n_f, dx_f, max_iter=600)
            c5_new = compute_c5(h_new, dx_f)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h_new.copy()

    return (best_h, best_c5, len(best_h))
