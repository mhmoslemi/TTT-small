        if time.time() - start_time > time_limit:
            break
        i = rng.randint(n)
        best_delta = 0.0
        best_f_i = best_f
        for delta in [-0.1, -0.05, -0.02, 0.02, 0.05, 0.1, -0.01, 0.01, -0.2, 0.2]:
            h_try = h_curr.copy()
            h_try[i] += delta
            h_try = project_feasible(h_try, n)
            f_try = compute_c5(h_try, dx)
            if f_try < best_f_i - 1e-15:
                best_f_i = f_try
                best_delta = delta
        if best_delta != 0.0:
            h_curr[i] += best_delta
            h_curr = project_feasible(h_curr, n)
            best_f = best_f_i
    return h_curr, best_f

def generate_structural_start(n, rng):
    h = np.zeros(n)
    num_blocks = rng.randint(2, 5)
    block_sizes = np.sort(rng.choice(n, size=num_blocks, replace=False))
    for i in range(num_blocks):
        start = block_sizes[i]
        width = max(1, n // (2 * num_blocks))
        h[start:start + width] = 1.0
    h = project_feasible(h, n)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0

    best_h = project_feasible(h0.copy(), n)
    best_c5 = compute_c5(best_h, dx)

    def time_left(frac):
        return (time.time() - start_time) < budget_s * frac

    # Phase 1: Smooth max gradient optimization at n
    betas = [3.0, 5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 500.0]
    for beta in betas:
        if not time_left(0.45):
            break
        h_curr = best_h.copy()
        h_curr = pg_optimize(h_curr, n, dx, beta, 3000, start_time, budget_s * 0.45)
        c5_curr = compute_c5(h_curr, dx)
        if c5_curr < best_c5:
            best_h = h_curr
            best_c5 = c5_curr

    # Phase 2: Restarts with perturbations at n
    if time_left(0.65):
        for restart in range(30):
            if not time_left(0.65):
                break
            scale = 0.05 + 0.1 * (1.0 - restart / 30.0)
            h_try = best_h + rng.randn(n) * scale
            h_try = project_feasible(h_try, n)
            for beta in [50.0, 200.0]:
                h_try = pg_optimize(h_try, n, dx, beta, 2000, start_time, budget_s * 0.65)
                c5_try = compute_c5(h_try, dx)
                if c5_try < best_c5:
                    best_h = h_try
                    best_c5 = c5_try

    # Phase 3: Coordinate refinement at n
    if time_left(0.75):
        best_h, best_c5 = coord_refine(best_h, n, dx, 5000, start_time, budget_s * 0.75)

    # Phase 4: Try upsampling to 2n
    n2 = 2 * n
    dx2 = 2.0 / n2
    h_up = np.zeros(n2)
    h_up[0::2] = best_h
    h_up[1::2] = best_h
    h_up = project_feasible(h_up, n2)
    c5_up = compute_c5(h_up, dx2)

    if time_left(0.9) and c5_up < best_c5 + 0.01:
        for beta in [10.0, 30.0, 100.0, 300.0]:
            if not time_left(0.9):
                break
            h_up = pg_optimize(h_up, n2, dx2, beta, 1500, start_time, budget_s * 0.9)
            c5_up = compute_c5(h_up, dx2)
            if c5_up < best_c5:
                best_h = h_up.copy()
                best_c5 = c5_up

    # Phase 5: Final polishing
    n_final = len(best_h)
    dx_final = 2.0 / n_final
    if time_left(0.98):
        beta_final = 500.0
        best_h = pg_optimize(best_h, n_final, dx_final, beta_final, 3000, start_time, budget_s * 0.98)
        best_c5 = compute_c5(best_h, dx_final)

    best_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return (best_h, best_c5, len(best_h))
