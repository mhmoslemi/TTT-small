    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    for _ in range(1000):
        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - target
        if abs(diff) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] -= diff / np.sum(mask)
        else:
            h -= diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = np.sum(h) - target
    if abs(diff) > 1e-12:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(mask):
            h[mask] -= diff / len(mask)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_grad(x, n, dx, beta):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = beta * (corr - m) * dx
    exps = np.exp(shifted)
    tot = np.sum(exps)
    w = exps / tot
    f = m * dx + np.log(tot) / beta
    grad = np.zeros(n)
    for p in range(n):
        js = np.arange(max(0, p - n + 1), n)
        k1 = js - p + n - 1
        k2 = p - js + n - 1
        grad[p] = dx * (np.dot(w[k1], 1.0 - x[js]) - np.dot(w[k2], x[js]))
    return f, grad

def slsqp_smooth(h0, n, dx, target, beta, maxiter=300):
    try:
        res = minimize(
            lambda x: smooth_obj_grad(x, n, dx, beta),
            h0, jac=True, method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
