def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_h(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt > 0:
                h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
        else:
            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt > 0:
                h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    return h

def gradient_of_max_corr(h, n):
    """Compute subgradient of max_k C(k) at h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = np.argmax(corr)
    shift = k_star - (n - 1)
    grad = np.zeros(n)
    for j in range(n):
        jp = j + shift
        jm = j - shift
        if 0 <= jp < n:
            grad[j] += (1.0 - h[jp])
        if 0 <= jm < n:
            grad[j] -= h[jm]
    return grad, shift, k_star

def nesterov_projected_subgrad(h0, n_iters=2000, t0=1.0, time_limit=150):
    """Nesterov accelerated projected subgradient for min max_k C(k)."""
    n = len(h0)
    target = n * 0.5
    h = h0.copy()
    y = h0.copy()
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    t_start = time.time()
    
    for t in range(1, n_iters + 1):
        if time.time() - t_start > time_limit:
            break
        grad, shift, k_star = gradient_of_max_corr(y, n)
        alpha = t0 / (t + t0)
        h_new = y - alpha * grad
        h_new = project_h(h_new, target)
        # Nesterov momentum
        beta = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = h_new + (beta - 1.0) / beta * (h_new - h)
        y = project_h(y, target)
        h = h_new
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

```python
def gradient_top_k(h, n, K=5):
    """Compute weighted gradient using top K active constraints."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Get top K indices
    top_k = np.argsort(corr)[-K:][::-1]  # descending
    # Weights: proportional to (corr[k] - corr[k_second] + epsilon)
    # Or just uniform
    weights = np.ones(len(top_k)) / len(top_k)
    
    grad = np.zeros(n)
    for w, k in zip(weights, top_k):
        shift = k - (n - 1)
        for j in range(n):
            jp = j + shift
            jm = j - shift
            if 0 <= jp < n:
                grad[j] += w * (1.0 - h[jp])
            if 0 <= jm < n:
                grad[j] -= w * h[jm]
    return grad
```

This is more expensive (K times more) but gives a smoother descent direction.

However, the inner loop over j is O(n) in Python, which is slow. Let me vectorize it:

```python
def gradient_top_k_vec(h, n, K=5):
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_k = np.argsort(corr)[-K:]
    weights = np.ones(len(top_k)) / len(top_k)
    
    grad = np.zeros(n)
    for w, k in zip(weights, top_k):
        shift = k - (n - 1)
        # Vectorized gradient computation
        # For each j: grad[j] += w * [(1-h[j+shift]) * (0<=j+shift<n) - h[j-shift] * (0<=j-shift<n)]
        jp = np.arange(n) + shift
        jm = np.arange(n) - shift
        mask_p = (jp >= 0) & (jp < n)
        mask_m = (jm >= 0) & (jm < n)
        grad[mask_p] += w * (1.0 - h[jp[mask_p]])
        grad[mask_m] -= w * h[jm[mask_m]]
    return grad
```

This is O(K*n) per call, which is fast.

import numpy as np

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))


def project_h(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt > 0:
                h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
        else:
            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt > 0:
                h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    return h


def gradient_single(h, n, shift):
    j = np.arange(n)
    jp = j + shift
    jm = j - shift
    mp = (jp >= 0) & (jp < n)
    mm = (jm >= 0) & (jm < n)
    grad = np.zeros(n)
    grad[mp] += (1.0 - h[jp[mp]])
    grad[mm] -= h[jm[mm]]
    return grad


def gradient_top_k(h, n, K=5):
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-K:]
    weights = np.ones(K) / K
    grad = np.zeros(n)
    for w, k in zip(weights, top_idx):
        shift = k - (n - 1)
        grad += w * gradient_single(h, n, shift)
    return grad

def slsqp_t_var(h0, max_iter=500, ftol=1e-11):
    n = len(h0)
    c5_cur = compute_c5(h0)
    v0 = np.append(h0.copy(), c5_cur + 0.01)
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.5)]

    def obj(v):
        return v[-1]

    def ineq(v):
        hh = v[:n]
        t = v[-1]
        dx = 2.0 / n
        corr = np.correlate(hh, 1.0 - hh, mode='full') * dx
        return t - corr

    def eq_fun(v):
        return np.sum(v[:n]) - n * 0.5

    def eq_jac(v):
        jj = np.zeros(len(v))
        jj[:n] = 1.0
        return jj

    cons = [{'type': 'ineq', 'fun': ineq}, {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    res = minimize(obj, v0, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': ftol})
    h_new = np.clip(res.x[:n], 0.0, 1.0)
    h_new = project_h(h_new, n * 0.5)
    return h_new, compute_c5(h_new)


def smooth_lse_objective(h, temp):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    return m + temp * np.log(np.sum(np.exp((corr - m) / temp)))


def slsqp_smooth(h0, temp, max_iter=400, ftol=1e-12):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n

    def obj(h):
        return smooth_lse_objective(h, temp)

    def eq_fun(h):
        return np.sum(h) - n * 0.5

    def eq_jac(h):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    res = minimize(obj, h0.copy(), method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': ftol})
    h_new = np.clip(res.x, 0.0, 1.0)
    h_new = project_h(h_new, n * 0.5)
    return h_new, compute_c5(h_new)
