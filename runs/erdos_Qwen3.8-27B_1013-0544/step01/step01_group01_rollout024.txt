    return np.correlate(h, 1.0 - h, mode="full")

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20):

        diff = target_sum - s
        if abs(diff) < 1e-12:

        if mask.sum() == 0:
            h += diff / len(h)

        else:

            h = np.clip(h, 0, 1)

    corr = compute_corr(h)

Actually, one more important observation: the subgradient computation for the active lag can be vectorized with numpy for speed!

For lag k ≥ 0:
corr_k = sum_{i=0}^{n-1-k} h[i] * (1-h[i+k])

grad[:n-k] += (1 - h[k:])

import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")


def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(30):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() == 0:
            h = h + diff / len(h)
            h = np.clip(h, 0, 1)
        else:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0, 1)
    return h


def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta


def subgradient_for_lag(h, lag, n):
    """Vectorized subgradient of correlation at a given lag."""
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[:m] += (1.0 - h[lag:lag + m])
            grad[lag:lag + m] -= h[:m]
    else:
        m = n + lag
        if m > 0:
            grad[-m:] += (1.0 - h[:m])
            grad[:m] -= h[-m:]
    return grad


def nesterov_subgradient(h, target_sum, n_iters, n_top=3, base_step=1.0, momentum=0.9):
    """Nesterov-accelerated subgradient method."""
    n = len(h)
    v = np.zeros(n)
    c5 = compute_c5(h)
    h_curr = h.copy()
    
    for it in range(n_iters):
        step = base_step / (1.0 + 0.02 * it)
        # Nesterov lookahead
        h_look = h_curr + momentum * v
        corr = compute_corr(h_look)
        top_idx = np.argsort(corr)[-n_top:]
        grad = np.zeros(n)
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            grad += subgradient_for_lag(h_look, lag, n)
        # Update
        v = momentum * v - step * grad
        h_curr = project_h(h_curr + v, target_sum)
        c5_new = compute_c5(h_curr)
        if c5_new < c5:
            c5 = c5_new
    return h_curr, c5


def slsqp_polish(h, target_sum, beta, maxiter=500, ftol=1e-13):
    """SLSQP polish with smooth max objective."""
    n = len(h)
    dx = 2.0 / n
    
    def obj(x):
        corr = compute_corr(x)
        return smooth_max(corr, beta) * dx
    
    bounds = Bounds(np.zeros(n), np.ones(n))
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
                   'jac': lambda x: np.ones(n)}]
    res = minimize(obj, h, method='SLSQP', bounds=bounds,
                   constraints=constraints, options={'maxiter': maxiter, 'ftol': ftol})
    return project_h(res.x, target_sum)
