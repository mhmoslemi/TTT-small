def compute_grad_lag(h, L):

    omh = 1.0 - h
    grad = np.zeros(n)
    if L == 0:
        grad[:] = omh[:] - h[:]
    elif L > 0:
        grad[L:] += omh[:n - L]
        grad[:n - L] -= h[L:]
    else:

        grad[:n - L2] += omh[L2:]
        grad[L2:] -= h[:n - L2]
