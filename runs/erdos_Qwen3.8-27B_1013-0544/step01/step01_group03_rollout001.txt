    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val(obj, tau):
    x = tau * obj
    x_max = np.max(x)
    return float(x_max / tau + np.log(np.sum(np.exp(x - x_max))) / tau)

    k_star is the lag index (can be negative)."""

        # Second term: -h[j+k_star] if 0 <= j+k_star < n
        idx2 = j + k_star
        if 0 <= idx2 < n:
            sg[j] -= h[idx2]
    return sg

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val(obj, tau):
    x = tau * obj
    x_max = np.max(x)
    return float(x_max / tau + np.log(np.sum(np.exp(x - x_max))) / tau)

def project_to_feasible(x, target):
    lo, hi = -x.max() - 1.0, 1.0 - x.min() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(x + mid, 0.0, 1.0)
        s = h.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

    """Vectorized subgradient of correlation at lag k."""
    sg = np.zeros(n)
    j = np.arange(n)
    # First term: (1-h[j-k]) where 0 <= j-k < n
    idx1 = j - k
    valid1 = (idx1 >= 0) & (idx1 < n)
    sg[valid1] += (1.0 - h[idx1[valid1]])
    # Second term: -h[j+k] where 0 <= j+k < n
    idx2 = j + k
    valid2 = (idx2 >= 0) & (idx2 < n)
    sg[valid2] -= h[idx2[valid2]]
    return sg

def smooth_grad(h, tau):
    """Gradient of smooth max of correlations."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    x = tau * corr
    x_max = np.max(x)
    weights = np.exp(x - x_max)
    weights /= weights.sum()
    # Gradient is weighted sum of subgradients
    sg = np.zeros(n)
    for m in range(len(corr)):
        if weights[m] > 1e-10:
            k = m - (n - 1)
            sg += weights[m] * subgrad_vec(h, k, n)
    return sg / tau, corr

def subgrad_descent(h, n, target, n_iter, alpha0, momentum):
    """Subgradient descent with momentum."""
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    v = np.zeros(n)
    
    for it in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        # Use top-m lags for stability
        m = min(5, len(corr))
        top_idx = np.argsort(corr)[-m:]
        sg = np.zeros(n)
        for mi in top_idx:
            k = int(mi) - (n - 1)
            sg += subgrad_vec(h, k, n)
        sg /= m
        
        alpha = alpha0 / (1.0 + 0.005 * it)
        v = momentum * v - alpha * sg
        h_new = h - v
        h_new = project_to_feasible(h_new, target)
        
        c5 = evaluate_c5(h_new)
        if c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = c5
        h = h_new
    
    return best_h, best_c5

    """Gradient descent on smooth max approximation."""
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    v = np.zeros(n)

            best_c5 = c5

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val(obj, tau):
    x = tau * obj
    x_max = np.max(x)
    return float(x_max / tau + np.log(np.sum(np.exp(x - x_max))) / tau)

def project_to_feasible(x, target):
    lo = -x.max() - 1.0
    hi = 1.0 - x.min() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(x + mid, 0.0, 1.0)
        s = h.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_vec(h, k, n):
    sg = np.zeros(n)
    j = np.arange(n)
    idx1 = j - k
    v1 = (idx1 >= 0) & (idx1 < n)
    sg[v1] += (1.0 - h[idx1[v1]])
    idx2 = j + k
    v2 = (idx2 >= 0) & (idx2 < n)
    sg[v2] -= h[idx2[v2]]
    return sg

def weighted_subgrad(h, weights, n):
    sg = np.zeros(n)
    j = np.arange(n)
    for m in range(len(weights)):
        w = weights[m]
        if w < 1e-12:
            continue
        k = m - (n - 1)
        idx1 = j - k
        v1 = (idx1 >= 0) & (idx1 < n)
        sg[v1] += w * (1.0 - h[idx1[v1]])
        idx2 = j + k
        v2 = (idx2 >= 0) & (idx2 < n)
        sg[v2] -= w * h[idx2[v2]]
    return sg

def fista_smooth(h, n, target, n_iter, tau, lr):
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    x = h.copy()
    y = h.copy()
    t = 1.0
    corr = np.correlate(y, 1.0 - y, mode="full")
    exp_term = tau * (corr - corr.max())
    weights = np.exp(exp_term)
    weights /= weights.sum()
    for it in range(n_iter):
        grad = weighted_subgrad(y, weights, n)
        x_new = project_to_feasible(y - lr * grad / tau, target)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y = np.clip(y, 0.0, 1.0)
        x = x_new
        t = t_new
        if it % 50 == 0 or it == n_iter - 1:
            corr = np.correlate(y, 1.0 - y, mode="full")
            exp_term = tau * (corr - corr.max())
            weights = np.exp(exp_term)
            weights /= weights.sum()
        c5 = evaluate_c5(x)
        if c5 < best_c5:
            best_h = x.copy()
            best_c5 = c5
    return best_h, best_c5

def subgrad_momentum(h, n, target, n_iter, alpha0, mom, topm):
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    v = np.zeros(n)
    for it in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        m = min(topm, len(corr))
        top_idx = np.argsort(corr)[-m:]
        sg = np.zeros(n)
        for mi in top_idx:
            k = int(mi) - (n - 1)
            sg += subgrad_vec(h, k, n)
        sg /= m
        alpha = alpha0 / (1.0 + 0.003 * it)
        v = mom * v - alpha * sg
        h_new = project_to_feasible(h - v, target)
        c5 = evaluate_c5(h_new)
        if c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = c5
        h = h_new
    return best_h, best_c5

def slsqp_pen_obj(x, tau, target, pen):
    corr = np.correlate(x, 1.0 - x, mode="full")
    return smooth_max_val(corr, tau) + pen * (np.sum(x) - target) ** 2

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)

    best_h = h_init.copy()
    best_c5 = evaluate_c5(best_h)

    def rem():
        return budget_s - (time.time() - t0)

    # Phase 1: Subgradient with momentum on initial resolution
    n0 = len(h_init)
    if rem() > 60:
        h1, c51 = subgrad_momentum(h_init, n0, n0 / 2.0, 4000, 0.04, 0.9, 5)
        if c51 < best_c5:
            best_h, best_c5 = h1, c51

    # Phase 2: FISTA with smooth max
    for tau in [3, 10, 30, 100]:
        if rem() < 25:
            break
        h2, c52 = fista_smooth(best_h, len(best_h), len(best_h) / 2.0, 1500, tau, 0.15)
        if c52 < best_c5:
            best_h, best_c5 = h2, c52

    # Phase 3: SLSQP with penalty (no explicit constraints needed)
    if rem() > 30:
        n = len(best_h)
        target = n / 2.0
        bnds = Bounds(np.zeros(n), np.ones(n))
        for tau in [10, 50, 200, 1000]:
            if rem() < 12:
                break
            try:
                res = minimize(slsqp_pen_obj, best_h, args=(tau, target, 100.0),
                               method='L-BFGS-B', bounds=bnds,
                               options={'maxiter': 300, 'ftol': 1e-14})
                h3 = project_to_feasible(res.x, target)
                c53 = evaluate_c5(h3)
                if c53 < best_c5:
                    best_h, best_c5 = h3, c53
            except Exception:
                pass

    # Phase 4: Resolution sweep
    for tn in [60, 80, 100, 140, 160, 200]:
        if rem() < 40:
            break
        if tn % 2 != 0:
            continue
        h_t = np.interp(np.linspace(0, 1, tn, endpoint=False),
                        np.linspace(0, 1, len(best_h), endpoint=False), best_h)
        h_t = project_to_feasible(h_t, tn / 2.0)
        if rem() > 25:
            h4a, c54a = subgrad_momentum(h_t, tn, tn / 2.0, 2000, 0.04, 0.9, 5)
            if c54a < best_c5:
                best_h, best_c5 = h4a, c54a
        if rem() > 15:
            for tau in [10, 50, 200]:
                if rem() < 10:
                    break
                h4b, c54b = fista_smooth(best_h, len(best_h), len(best_h) / 2.0, 1000, tau, 0.12)
                if c54b < best_c5:
                    best_h, best_c5 = h4b, c54b

    # Phase 5: Binary relaxation
    for _ in range(4):
        if rem() < 25:
            break
        n = len(best_h)
        target = n / 2.0
        h_bin = (best_h > 0.5).astype(float)
        ones = int(h_bin.sum())
        if ones > int(target):
            order = np.argsort(h_bin + 0.01 * np.random.random(n))
            h_bin[order[:ones - int(target)]] = 0.0
        elif ones < int(target):
            order = np.argsort(-best_h + 0.01 * np.random.random(n))
            h_bin[order[:int(target) - ones]] = 1.0
        h_bin = project_to_feasible(h_bin, target)
        h5, c55 = subgrad_momentum(h_bin, n, target, 2500, 0.05, 0.85, 5)
        if c55 < best_c5:
            best_h, best_c5 = h5, c55

    # Phase 6: Random restarts from best
    for _ in range(10):
        if rem() < 15:
            break
        n = len(best_h)
        target = n / 2.0
        scale = 0.03 + 0.1 * np.random.random()
        h_r = best_h + scale * np.random.randn(n)
        h_r = project_to_feasible(h_r, target)
        h6, c56 = subgrad_momentum(h_r, n, target, 1500, 0.04, 0.9, 5)
        if c56 < best_c5:
            best_h, best_c5 = h6, c56

    # Phase 7: Final polish with high-tau FISTA
    if rem() > 15:
        for tau in [200, 1000, 3000]:
            if rem() < 8:
                break
            h7, c57 = fista_smooth(best_h, len(best_h), len(best_h) / 2.0, 1500, tau, 0.08)
            if c57 < best_c5:
                best_h, best_c5 = h7, c57

    # Final projection and evaluation
    best_h = project_to_feasible(best_h, len(best_h) / 2.0)
    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, len(best_h)
