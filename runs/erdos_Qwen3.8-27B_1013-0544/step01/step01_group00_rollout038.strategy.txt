        h = np.minimum(h, 1.0 - 1e-6)
        min_per_point = np.min(h)
    cnt = np.sum(max_per_point if diff>0 else (1.0-min_per_point))
    adjust = diff / max(1.0, cnt)
    for i in range(n):
        if diff>0 and h[i] < 1.0:
            h[i] = np.clip(h[i] + adjust/max_per_point, 0, 1.0)
        elif diff<0 and h[i] > 0:
            h[i] = np.clip(h[i] + adjust/max_per_point, 0, 1.0)
    return project_to_integral(h)  # Recursive until satisfaction

def repair_feasible(h):
    # Turn infeasible to mostly feasible
    while True:
        # Apply smoothing
        h = np.maximum(np.minimum(h, 1.0), 0.0)
        total = np.sum(h)
        # If not satisfied, adjust direction for integral
        if abs(total - n/2) < 1e-4:
            return h
        diff = (n/2) - total
        if diff>0:
            adjust = diff / (n*0.05)
            h += adjust
        else:
            adjust = diff / (n*0.05)
            h -= adjust
        h = np.clip(h, 0.0, 1.0)
        # Stochastic component for diversity
        if np.random.random() < 0.5:
            # Random walk addition
            h += 0.02 * np.random.randn(len(h))

def generate_initial_candidate():
    # For the initial grid
    n = 180 if 'initial_h_values' in globals() else 90
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
        h = project_to_integral(h)
    else:
        h = np.zeros(n)
        # Alternative initialization for smaller grid
        # High-low pattern
        k = np.exp(-np.log(0.7) * np.arange(n) / (n/3))
        h = np.exp(-np.log(0.9) * np.arange(n) / (n/3))
        h = (h + k) / (2 * (2-0.3))
        desired = n/2
        current = np.sum(h) * 2/n
        scaling = desired/current
        h *= scaling
        h = np.clip(h, 0, 1)
    return h

def direct_optimization(h):
    """ DIRECT global optimizer with integral maintenance"""
    from scipy.optimize import dual_annealing
    bounds = [(0,1)] * len(h)
    cons = {'type':'eq', 'fun':lambda x: np.sum(x) - n/2}
    # Simple objective
    def objective(x):
        try:
            x_val = np.clip(x, 0, 1)
            corr = np.correlate(x_val, 1-x_val, mode='full')
            return np.abs(np.max(corr))
        except:
            return np.inf
    
    x0 = h
    x_opt, f_opt = dual_annealing(objective, bounds, constraints=cons)
    return x_opt, f_opt

def lbfgsb_optimization(h):
    """L-BFGS-B local optimizer"""
    from scipy.optimize import minimize
    bounds = [(0,1)] * len(h)
    cons = {'type':'eq', 'fun':lambda x: np.sum(x) - n/2}
    def objective(x):
        try:
            corr = np.correlate(x, 1-x, mode='full')
            return np.min(corr)  # Better to minimize min correlation
        except:
            return np.inf
    
    x0 = h
    x_opt, f_opt = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, constraints=cons, 
        options={'maxcor':10, 'maxfun':500, 'maxiter':500, 'disp':False})
    # Check if constraints still satisfied
    x_opt = repair_feasible(x_opt) if abs(np.sum(x_opt)-n/2) > 0.5 else x_opt
    return x_opt, np.max(np.correlate(x_opt, 1-x_opt, mode='full')) * (2/len(x_opt))

def create_fine_scale(h, scale_up):
    """ Scale up while maintaining feasibility"""
    if scale_up/len(h) < 1.1:
        n_new = int(len(h) * scale_up)
        x_new = np.arange(n_new) / n_new
        x_old = np.arange(len(h)) / len(h)
        # Linear interpolation in logs for initialization
        log_x_old = np.log(x_old+1)
        log_x_new = np.log(x_new+1)
        # Convert to log-log to preserve integral
        log_vals = np.log(h)
        interp_log = np.interp(log_x_new, np.c_[log_x_old, np.log(x_old+1)], log_vals.reshape(-1))
        h_new = np.exp(interp_log)
        # Adjust integral
        h_new = project_to_integral(h_new)
    else:
        h_new = np.zeros(int(scale_up))
        np.random.random(len(h_new))
        h_new = project_to_integral(h_new)
    return h_new

def smart_selection(current_h, new_h, budget):
    """Partition budget between two candidates"""
    time_needed = budget*(np.min([compute_objective(current_h), compute_objective(new_h)])/np.max([compute_objective(current_h), compute_objective(new_h)]))**0.5
    return time_needed < budget/2  # Simplified heuristic
