def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full') * dx

def get_corr(h):

    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, target_sum):
    """Project h onto {x: sum(x)=target_sum, 0<=x<=1}"""
    # Water-filling / bisection
    lo, hi = -max(h) - 1, max(h) + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def subgradient_of_corr(h, shift, dx, n):
    """Compute gradient of corr at a given shift with respect to h."""
    grad = np.zeros(n)
    # f_shift = dx * sum_i h[i] * (1 - h[i+shift])
    # df/dh[j] = dx * [(1-h[j+shift])*I(valid) - h[j-shift]*I(valid)]
    if shift >= 0:

        # j in [0, n-1-shift]
        js = np.arange(0, n - shift)
        grad[js] += dx * (1.0 - h[js + shift])
        if shift > 0:
            js2 = np.arange(shift, n)
            grad[js2] -= dx * h[js2 - shift]
    else:

        js = np.arange(0, n - s)
        grad[js + s] += dx * (1.0 - h[js])
        if shift < 0:
            js2 = np.arange(0, n - s)
            grad[js2] -= dx * h[js2 + s]
    return grad

def subgradient_max(h, alpha=0.1):
    """Compute subgradient of max_k corr_k using softmax weighting."""
    n = len(h)
    dx = 2.0 / n
    corr = get_corr(h)
    m = np.max(corr)

    weights /= np.sum(weights)

    active = weights > 1e-6

        h = h_new

    """Compute subgradient of weighted sum of correlations.
    h: array of length n
    weights: array of length 2n-1 (one per correlation lag)

    """
    n = len(h)

def compute_c5(h):

def get_corr(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, target_sum):
    """Project h onto {x: sum(x)=target_sum, 0<=x<=1} using bisection."""
    lo = h.min() - target_sum / len(h) - 1.0
    hi = h.max() - target_sum / len(h) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_subgrad(h, weights):
    """Compute subgradient of sum_k weights[k]*corr_k w.r.t. h."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        w = weights[m]
        if abs(w) < 1e-12:
            continue
        shift = m - (n - 1)
        if shift >= 0:
            length = n - shift
            i = np.arange(length)
            grad[i] += dx * w * (1.0 - h[i + shift])
            grad[i + shift] -= dx * w * h[i]
        else:
            s = -shift
            length = n - s
            i = np.arange(length)
            grad[i + s] += dx * w * (1.0 - h[i])
            grad[i] -= dx * w * h[i + s]
    return grad

        return v[-1]
    
    def ineq(v):
        h = v[:n]
        t = v[-1]
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        return t - c
    
    def eq_fun(v):
        return np.sum(v[:n]) - n * 0.5
    
    def eq_jac(v):
        j = np.zeros(len(v))
        j[:n] = 1.0
        return j
    
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [
        {'type': 'ineq', 'fun': ineq},
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}
    ]
    
    res = minimize(obj, v0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-9})
    h_new = project_feasible(res.x[:n], n * 0.5)
    return h_new, compute_c5(h_new)

def lkbfgs_softmax(h0, temp, max_iter=500):
    """L-BFGS-B with log-sum-exp smoothed objective."""
    n = len(h0)
    dx = 2.0 / n
    
    def obj(h):
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        m = np.max(c)
        return m + temp * np.log(np.sum(np.exp((c - m) / temp)))

        return np.sum(h) - n * 0.5

    # Use SLSQP instead or penalty
    # Let's use a penalty approach with L-BFGS-B

        c = np.correlate(h, 1.0 - h, mode='full') * dx
        m = np.max(c)
        val = m + temp * np.log(np.sum(np.exp((c - m) / temp)))
        val += mu * (np.sum(h) - n * 0.5) ** 2
        return val

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def get_corr(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, target_sum):
    lo = h.min() - 2.0
    hi = h.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_subgrad(h, weights):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-10:
            continue
        shift = m - (n - 1)
        if shift >= 0:
            length = n - shift
            i = np.arange(length)
            grad[i] += dx * w * (1.0 - h[i + shift])
            grad[i + shift] -= dx * w * h[i]
        else:
            s = -shift
            length = n - s
            i = np.arange(length)
            grad[i + s] += dx * w * (1.0 - h[i])
            grad[i] -= dx * w * h[i + s]
    return grad

                h_cur, c5_cur = h_new, c5_new
        for temp in [0.01, 0.005]:
            if time_left() < 80:
                break
            h_new, c5_new = lkbfgs_smooth(h_cur, temp, 200)
            if c5_new < c5_cur:
                h_cur, c5_cur = h_new, c5_new
        if c5_cur < best_c5:
            best_c5 = c5_cur
            h_best = h_cur.copy()

    # Final cleanup
    h_best = project_feasible(h_best, len(h_best) * 0.5)
    best_c5 = compute_c5(h_best)
    return h_best, best_c5, len(h_best)
