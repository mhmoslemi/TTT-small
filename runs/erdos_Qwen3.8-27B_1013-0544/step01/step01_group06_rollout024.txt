import numpy as np

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def get_grad_at_lag(h, k):
    """Gradient of corr_k = sum_i h[i]*(1-h[i+k]) w.r.t. h"""
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        # corr = sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])
        # d/dh[j] for j in [0, n-1-k]: (1-h[j+k])
        # d/dh[j] for j in [k, n-1]: -h[j-k]
        g[:n-k] += (1.0 - h[k:])
        g[k:] -= h[:n-k]
    else:
        p = -k
        # corr = sum_{i=p}^{n-1} h[i]*(1-h[i-p])
        # d/dh[j] for j in [p, n-1]: (1-h[j-p])
        # d/dh[j] for j in [0, n-1-p]: -h[j+p]
        g[p:] += (1.0 - h[:n-p])
        g[:n-p] -= h[p:]
    return g

def smooth_grad(h, alpha):
    """Compute smoothed gradient of max_k corr_k * dx"""
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    m = np.max(corr)
    # Use log-sum-exp for stability
    shifted = alpha * (corr - m)
    # Clip to avoid overflow
    shifted = np.clip(shifted, -700, 700)
    w = np.exp(shifted)
    s = np.sum(w)
    if s < 1e-300:
        idx = np.argmax(corr)
        w = np.zeros_like(corr)
        w[idx] = 1.0
    else:
        w /= s
    
    # Only use significant weights

    grad = np.zeros(n)
    for idx in active:
        k = idx - (n - 1)
        grad += w[idx] * get_grad_at_lag(h, k)

                        best_c5 = c5_pert
                        improved = True
                        break
                lr = 0.06
                stall = 0
                if not improved:
                    restarts += 1
                if restarts > 10:
                    break
    return best_h, best_c5

def optimize_sa(h_init, target_sum, deadline, rng, T_init=0.05):
    """Simulated annealing with gradient guidance"""
    n = len(h_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    cur_h = h_init.copy()
    cur_c5 = best_c5
    T = T_init
    x = np.arange(n) / n * 2.0
    stall = 0

    while time.time() < deadline:
        if T > 0.001:
            T *= 0.995
        # Propose move: gradient direction + noise
        grad = smooth_grad(cur_h, 80.0)
        gnorm = np.sqrt(np.sum(grad ** 2)) + 1e-15
        step = 0.03 / gnorm
        noise = (T * 0.1 + 0.01) * rng.randn(n)
        cur_cand = cur_h - step * grad + noise
        cur_cand = project_box_sum(cur_cand, target_sum)
        cur_cand_c5 = compute_c5(cur_cand)
        delta = cur_cand_c5 - cur_c5
        if delta < 0 or (T > 0.005 and rng.rand() < np.exp(-delta / (T + 1e-15))):
            cur_h = cur_cand
            cur_c5 = cur_cand_c5
            if cur_c5 < best_c5:
                best_h = cur_cand.copy()
                best_c5 = cur_c5
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        if stall > 300:
            # Structured kick
            freq = rng.uniform(1, 8)
            phase = rng.uniform(0, 6.28)
            cur_h = best_h + 0.04 * np.sin(freq * np.pi * x + phase)
            cur_h = project_box_sum(cur_h, target_sum)
            cur_c5 = compute_c5(cur_h)
            T = max(T, 0.02)
            stall = 0
    return best_h, best_c5

def gen_start(n, rng):
    x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    kind = rng.randint(0, 6)
    if kind == 0:
        h = 0.5 + 0.3 * np.sin(np.pi * x)
    elif kind == 1:
        h = 0.5 + 0.25 * np.cos(2.0 * np.pi * x + rng.uniform(0, 6.28))
    elif kind == 2:
        h = 0.5 + 0.2 * np.sin(3.0 * np.pi * x)
    elif kind == 3:
        h = np.where((x % 1.0) < 0.5, 0.7, 0.3)
    elif kind == 4:
        h = 0.5 + 0.15 * np.sin(4.0 * np.pi * x + 0.5) * np.cos(2.0 * np.pi * x)
    else:
        h = 0.5 + 0.15 * rng.randn(n)
    return project_box_sum(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = gen_start(200, rng)

    n = len(h0)
    ts = n / 2.0
    h0 = project_box_sum(h0, ts)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Gradient optimization at initial resolution
    dl1 = t0 + budget_s * 0.35
    best_h, best_c5 = optimize(best_h, ts, dl1, rng, alpha_init=40.0)

    # Phase 2: Resolution sweep - try n=50,70,100,140 for potentially better bounds
    for n_try in [50, 70, 100, 140, 250]:
        rem = budget_s - (time.time() - t0)
        if rem < 45 or n_try == len(best_h):
            continue
        h_s = interpolate_to_n(best_h, n_try)
        h_s = project_box_sum(h_s, n_try / 2.0)
        phase_time = min(rem * 0.22, 180.0)
        dl = time.time() + phase_time
        h_o, c5_o = optimize(h_s, n_try / 2.0, dl, rng, alpha_init=35.0)
        if c5_o < best_c5:
            best_h, best_c5 = h_o, c5_o

    # Phase 3: SA polish at best resolution
    rem = budget_s - (time.time() - t0)
    if rem > 50:
        bn = len(best_h)
        dl_sa = time.time() + min(rem * 0.25, 200.0)
        h_sa, c5_sa = optimize_sa(best_h, bn / 2.0, dl_sa, rng, T_init=0.04)
        if c5_sa < best_c5:
            best_h, best_c5 = h_sa, c5_sa

    # Phase 4: Diverse restarts
    bn = len(best_h)
    for _ in range(5):
        rem = budget_s - (time.time() - t0)
        if rem < 35:
            break
        h_d = gen_start(bn, rng)
        h_b = 0.55 * best_h + 0.45 * h_d
        h_b = project_box_sum(h_b, bn / 2.0)
        dl = time.time() + min(rem * 0.18, 120.0)
        h_o, c5_o = optimize(h_b, bn / 2.0, dl, rng, alpha_init=25.0)
        if c5_o < best_c5:
            best_h, best_c5 = h_o, c5_o

    # Phase 5: Final gradient polish
    rem = budget_s - (time.time() - t0)
    if rem > 10:
        best_h, best_c5 = optimize(best_h, len(best_h) / 2.0,
                                   t0 + budget_s - 2, rng, alpha_init=120.0)

    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
