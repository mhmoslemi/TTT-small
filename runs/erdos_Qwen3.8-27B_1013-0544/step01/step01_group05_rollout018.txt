                break
            d = -d / gn
            h_new, f_new = line_search(h, d, target_sum, dx)
            h = h_new
            if f_new < best_local:
                best_local = f_new
                best_h = h.copy()
            if d @ d < 1e-20:  # converged
                break
    return best_h, best_local

def slsqp_polish(h0, target_sum, dx):
    n = len(h0)
    def obj(x):
        return compute_c5(x, dx)
    def eq(x):
        return np.sum(x) - target_sum
    try:
        res = minimize(obj, h0, method='SLSQP',
                      bounds=[(0.0, 1.0)] * n,
                      constraints=[{'type': 'eq', 'fun': eq}],
                      options={'maxiter': 5000, 'ftol': 1e-14})
        if res.x is not None:
            h = project_feasible(res.x, target_sum)
            return h, compute_c5(h, dx)
    except Exception:
        pass
    return h0, compute_c5(h0, dx)

_global_t0 = 0.0
_global_budget = 1100.0

def project_feasible(x, target_sum):
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = float(np.clip(x - mid, 0.0, 1.0).sum())
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def grad_at_lag(h, L):
    n = len(h)
    omh = 1.0 - h
    grad = np.zeros(n)
    if L == 0:
        grad[:] = omh - h
    elif L > 0:
        grad[L:] += omh[:n - L]
        grad[:n - L] -= h[L:]
    else:
        L2 = -L
        grad[:n - L2] += omh[L2:]
        grad[L2:] -= h[:n - L2]
    return grad

def softmax_direction(h, tau, n):
    dhalf = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    shifted = corr - np.max(corr)
    w = np.exp(tau * shifted)
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(len(w)):
        if w[k] > 1e-7:
            grad += w[k] * grad_at_lag(h, k - dhalf)
    return grad

def line_search(h, direction, target_sum, dx):

    return best_h, best_f
