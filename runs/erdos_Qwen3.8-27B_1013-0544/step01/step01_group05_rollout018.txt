def water_filling_step(h, corr, n, target_sum):
    """Try to equalize the top correlation values by adjusting h"""
    c_max = corr.max()
    # Find all lags within a small tolerance of the max
    tol = c_max * 0.01
    active = np.where(corr >= c_max - tol)[0]
    
    if len(active) < 2:
        # Just one active lag, use gradient
        h_half = n - 1
        L = active[0] - h_half
        grad = compute_grad_lag(h, L, n)
        return -grad
    
    # Multiple active lags: find direction that equalizes them
    # We want to find d such that c_k(h + d) ≈ c_max for all active k
    # Linearize: c_k(h) + ∇c_k(h) · d ≈ c_max
    # For active k with c_k < c_max: we want to increase c_k, so ∇c_k · d < 0
    # For active k with c_k = c_max: we want to keep c_k, so ∇c_k · d ≈ 0
    # This is a least-squares problem
    
    h_half = n - 1
    # Build the Jacobian for active lags
    # J[k, j] = ∂c_k/∂h_j
    J = np.zeros((len(active), n))
    residuals = np.zeros(len(active))
    for i, idx in enumerate(active):
        L = idx - h_half
        J[i] = compute_grad_lag(h, L, n)
        residuals[i] = c_max - corr[idx]  # how far below max
    
    # Solve J @ d = residuals (least squares)
    # But we also want small d, so use ridge regression
    d, _, _, _ = np.linalg.lstsq(J + 1e-4 * np.eye(n), residuals, rcond=None)
    
    return d
