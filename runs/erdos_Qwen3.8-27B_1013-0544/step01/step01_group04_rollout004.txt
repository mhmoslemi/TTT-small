def de_run(pop, fitness, n, dx, target_sum, pop_size, elapsed_fn, time_limit, seed):
    rng = np.random.RandomState(seed)
    best_idx = int(np.argmin(fitness))
    best_h = pop[best_idx].copy()
    best_c5 = fitness[best_idx]
    gen = 0
    max_gens = 20000
    while gen < max_gens and elapsed_fn() < time_limit:
        gen += 1
        progress = min(1.0, gen / max_gens)
        F = 0.8 - 0.5 * progress
        Cr = 0.85 - 0.45 * progress
        for i in range(pop_size):
            if elapsed_fn() >= time_limit:
                break
            excl = np.ones(pop_size, dtype=bool)
            excl[i] = False
            idxs = np.where(excl)[0]
            a, b, c = rng.choice(idxs, 3, replace=False)
            v = pop[a] + F * (pop[b] - pop[c])
            cross = rng.rand(n) < Cr
            j0 = rng.randint(n)
            cross[j0] = True
            trial = np.where(cross, v, pop[i])
            trial = project_feasible(trial, n)
            f_trial = compute_c5(trial, dx)
            if f_trial <= fitness[i]:
                pop[i] = trial
                fitness[i] = f_trial
                if f_trial < best_c5 - 1e-15:
                    best_c5 = f_trial
                    best_h = trial.copy()
        if gen % 100 == 0:
            worst_idx = int(np.argmax(fitness))
            if fitness[worst_idx] > best_c5 + 1e-8:
                pop[worst_idx] = best_h
                fitness[worst_idx] = best_c5
    return best_h, best_c5

def slsqp_polish(h_start, n, dx, target_sum, max_iter=400):
    bounds = [(0.0, 1.0)] * n
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},)
    try:
        res = minimize(
            lambda x: compute_c5(x, dx),
            h_start,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-13}
        )
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h_start, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0

    best_h = project_feasible(h0.copy(), n)
    best_c5 = compute_c5(best_h, dx)

    def elapsed():
        return time.time() - t0

    # Phase 1: DE at original resolution
    pop_size = 80
    pop = np.zeros((pop_size, n))
    pop[0] = best_h
    for i in range(1, pop_size):
        r = rng.rand()
        if r < 0.3:
            pop[i] = np.clip(best_h + rng.randn(n) * 0.25, 0, 1)
        elif r < 0.6:
            pop[i] = np.clip(rng.randn(n) * 0.3 + 0.4, 0, 1)
        elif r < 0.8:
            # Block structure
            block = int(n / (2 + int(rng.randint(4))))
            vals = np.clip(rng.randn(n // block + 1) * 0.4 + 0.5, 0, 1)
            pop[i] = np.repeat(vals, block)[:n]
        else:
            # Sine-based
            freq = rng.uniform(1, 6)
            phase = rng.uniform(0, 2 * np.pi)
            pop[i] = np.clip(0.5 + 0.4 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase), 0, 1)
        pop[i] = project_feasible(pop[i], n)

    fitness = np.array([compute_c5(pop[i], dx) for i in range(pop_size)])

    phase1_end = budget_s * 0.50
    best_h1, best_c5_1 = de_run(pop, fitness, n, dx, target_sum, pop_size, elapsed, phase1_end, seed)
    if best_c5_1 < best_c5:
        best_h = best_h1
        best_c5 = best_c5_1

    # Phase 2: Multiple SLSQP polishes
    phase2_end = budget_s * 0.70
    scales = [0.0, 0.01, 0.03, 0.05, 0.08, 0.12, 0.02, 0.06, 0.04, 0.1]
    for s in scales:
        if elapsed() >= phase2_end:
            break
        if s == 0.0:
            h_try = best_h.copy()
        else:
            h_try = best_h + rng.randn(n) * s
            h_try = project_feasible(h_try, n)
        h_new = slsqp_polish(h_try, n, dx, target_sum, max_iter=600)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5 - 1e-14:
            best_h = h_new
            best_c5 = c5_new

    # Second DE round with tighter perturbation around current best
    if elapsed() < budget_s * 0.80:
        pop_size2 = 60
        pop2 = np.zeros((pop_size2, n))
        pop2[0] = best_h
        for i in range(1, pop_size2):
            s = rng.uniform(0.05, 0.2)
            pop2[i] = np.clip(best_h + rng.randn(n) * s, 0, 1)
            pop2[i] = project_feasible(pop2[i], n)
        fitness2 = np.array([compute_c5(pop2[i], dx) for i in range(pop_size2)])
        phase2b_end = budget_s * 0.80
        best_h2, best_c5_2 = de_run(pop2, fitness2, n, dx, target_sum, pop_size2, elapsed, phase2b_end, seed + 500)
        if best_c5_2 < best_c5:
            best_h = best_h2
            best_c5 = best_c5_2
        # Final SLSQP
        if elapsed() < budget_s * 0.85:
            h_new = slsqp_polish(best_h, n, dx, target_sum, max_iter=800)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    # Phase 3: Upsample to 2n
    if elapsed() < budget_s * 0.87:
        n2 = 2 * n
        dx2 = 2.0 / n2
        target_sum2 = n2 / 2.0
        # Two upsample strategies
        h_rep = project_feasible(np.repeat(best_h, 2), n2)
        h_idx = np.zeros(n2)
        h_idx[0::2] = best_h
        h_idx[1::2] = 0.5 * (best_h[1:] + best_h[:-1])
        h_idx = project_feasible(h_idx, n2)
        c5_rep = compute_c5(h_rep, dx2)
        c5_idx = compute_c5(h_idx, dx2)
        h_up = h_rep if c5_rep <= c5_idx else h_idx

        pop_size3 = 50
        pop3 = np.zeros((pop_size3, n2))
        pop3[0] = h_up
        for i in range(1, pop_size3):
            s = rng.uniform(0.03, 0.12)
            pop3[i] = np.clip(h_up + rng.randn(n2) * s, 0, 1)
            pop3[i] = project_feasible(pop3[i], n2)
        fitness3 = np.array([compute_c5(pop3[i], dx2) for i in range(pop_size3)])
        phase3_end = budget_s * 0.93
        best_h3, best_c5_3 = de_run(pop3, fitness3, n2, dx2, target_sum2, pop_size3, elapsed, phase3_end, seed + 1000)
        if best_c5_3 < best_c5:
            best_h = best_h3
            best_c5 = best_c5_3
        if elapsed() < budget_s * 0.97:
            nf = len(best_h)
            df = 2.0 / nf
            h_new = slsqp_polish(best_h, nf, df, nf / 2.0, max_iter=500)
            c5_new = compute_c5(h_new, df)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    # Final feasibility
    n_final = len(best_h)
    dx_final = 2.0 / n_final
    best_h = project_feasible(best_h, n_final)
    best_c5 = compute_c5(best_h, dx_final)

    return (best_h, best_c5, n_final)
