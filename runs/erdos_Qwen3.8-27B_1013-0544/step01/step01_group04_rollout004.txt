import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - target
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            h[free] -= diff / np.sum(free)
        else:
            h -= diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = np.sum(h) - target
    if abs(diff) > 1e-14:
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            h[free] -= diff / np.sum(free)
            h = np.clip(h, 0.0, 1.0)
    return h

def project_simplex_box(h, target, n):
    """Project h onto {0 <= x <= 1, sum(x) = target} using bisection on lambda."""
    lo, hi = -max(h) - 1, max(h) - min(h) + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        projected = np.clip(h - mid, 0, 1)
        s = np.sum(projected)
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def compute_smooth_gradient(h, beta, dx):
    """Compute gradient of smooth max approximation."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    # Softmax weights
    max_corr = np.max(corr)
    exp_corr = np.exp(beta * (corr - max_corr))
    w = exp_corr / np.sum(exp_corr)
    
    # Compute gradient
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-15:
            continue
        d = m - (n - 1)
        # First term: (1-h[j+d]) for valid j+d
        one_minus_h = 1.0 - h
        if d >= 0:
            # j+d < n => j < n-d
            grad[:n-d] += w[m] * dx * one_minus_h[d:]
            # Second term: -h[j-d] for valid j-d, i.e., j >= d
            grad[d:] -= w[m] * dx * h[:n-d]
        else:
            # j+d >= 0 => j >= -d
            grad[-d:] += w[m] * dx * one_minus_h[:n+d]
            # j-d < n => j < n+d
            grad[:n+d] -= w[m] * dx * h[-d:]
    return grad

    m = np.max(corr)
    return m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta

def smooth_max_grad(h, beta, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    w = np.exp(beta * (corr - m))
    w /= w.sum()
    grad = np.zeros(n)
    omh = 1.0 - h
    for m_idx in range(2 * n - 1):
        wm = w[m_idx]
        if wm < 1e-14:
            continue
        d = m_idx - (n - 1)
        if d >= 0:
            grad[:n - d] += wm * dx * omh[d:]
            grad[d:] -= wm * dx * h[:n - d]
        else:
            grad[-d:] += wm * dx * omh[:n + d]
            grad[:n + d] -= wm * dx * h[-d:]
    return grad

def slsqp_optimize(h_start, n, dx, target, max_iter=300):
    bounds = [(0.0, 1.0)] * n
    constraint = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target}]
    res = minimize(
        lambda x: compute_c5(x, dx),
        h_start,
        method='SLSQP',
        bounds=bounds,
        constraints=constraint,
        options={'maxiter': max_iter, 'ftol': 1e-14}
    )
    return project_feasible(res.x, n)

def generate_random_step(n, rng):
    n_steps = rng.randint(2, 7)
    edges = np.sort(rng.choice(np.arange(1, n), size=min(n_steps - 1, n - 1), replace=False))
    boundaries = np.unique(np.concatenate([[0], edges, [n]]))
    h = np.zeros(n)
    for i in range(len(boundaries) - 1):
        h[boundaries[i]:boundaries[i + 1]] = rng.uniform(0.0, 1.0)
    return project_feasible(h, n)

def smooth_grad_descent(h, n, dx, target, beta_init=5.0, step_init=0.05,
                        max_iter=500, beta_max=400.0, time_budget=None, start_time=None):
    beta = beta_init
    step = step_init
    no_improve = 0
    best_local = h.copy()
    best_local_f = smooth_max_obj(h, beta, dx)
    for it in range(max_iter):
        if time_budget is not None and start_time is not None:
            if time.time() - start_time > time_budget:
                break
        grad = smooth_max_grad(h, beta, dx)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        accepted = False
        for ls in range(50):
            h_new = project_simplex_box(h - step * grad, target)
            f_new = smooth_max_obj(h_new, beta, dx)
            f_old = smooth_max_obj(h, beta, dx)
            if f_new < f_old - 1e-14 * step * gnorm:
                h = h_new
                accepted = True
                break
            step *= 0.75
        if not accepted:
            step *= 0.5
        f_act = smooth_max_obj(h, beta, dx)
        if f_act < best_local_f - 1e-14:
            best_local = h.copy()
            best_local_f = f_act
            no_improve = 0
        else:
            no_improve += 1
        if no_improve == 80:
            no_improve = 0
            step = min(step * 1.5, 0.15)
        if no_improve >= 200:
            beta = min(beta * 2.0, beta_max)
            no_improve = 0
            step = step_init
            if beta >= beta_max:
                break
    return best_local

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0

    best_h = project_feasible(h0.copy(), n)
    best_c5 = compute_c5(best_h, dx)

    def elapsed():
        return time.time() - start_time

    # Phase 1: Smooth gradient descent on initial (45% budget)
    t1 = budget_s * 0.45
    h_opt = smooth_grad_descent(best_h, n, dx, target,
                                beta_init=3.0, step_init=0.08,
                                max_iter=1000, beta_max=500.0,
                                time_budget=t1, start_time=start_time)
    c5_opt = compute_c5(h_opt, dx)
    if c5_opt < best_c5:
        best_h = h_opt
        best_c5 = c5_opt

    # Phase 2: Random restarts with smooth gradient (25% budget)
    t2 = budget_s * 0.70
    attempts = 0
    while elapsed() < t2:
        if attempts < 8:
            h_try = best_h + rng.randn(n) * (0.08 + 0.04 * rng.rand())
        elif attempts < 16:
            h_try = best_h + rng.randn(n) * 0.15
        else:
            h_try = generate_random_step(n, rng)
        h_try = project_feasible(h_try, n)
        h_try = smooth_grad_descent(h_try, n, dx, target,
                                    beta_init=10.0, step_init=0.04,
                                    max_iter=300, beta_max=300.0,
                                    time_budget=max(0, t2 - elapsed()),
                                    start_time=start_time)
        c5_try = compute_c5(h_try, dx)
        if c5_try < best_c5 - 1e-15:
            best_h = h_try.copy()
            best_c5 = c5_try
        attempts += 1

    # Phase 3: Upsampled optimization (15% budget)
    n2 = 2 * n
    dx2 = 2.0 / n2
    target2 = n2 / 2.0
    h_up = np.interp(np.linspace(0, 2, n2, endpoint=False),
                     np.linspace(0, 2, n, endpoint=False), best_h)
    h_up = project_feasible(h_up, n2)
    t3 = budget_s * 0.85
    if elapsed() < t3:
        h_up = smooth_grad_descent(h_up, n2, dx2, target2,
                                   beta_init=10.0, step_init=0.03,
                                   max_iter=400, beta_max=300.0,
                                   time_budget=max(0, t3 - elapsed()),
                                   start_time=start_time)
        c5_up = compute_c5(h_up, dx2)
        if c5_up < best_c5 - 1e-15:
            best_h = h_up.copy()
            best_c5 = c5_up

    # Phase 4: SLSQP polishing (remaining budget)
    n_final = len(best_h)
    dx_final = 2.0 / n_final
    target_final = n_final / 2.0
    t4 = budget_s - 8
    iters = 0
    while elapsed() < t4:
        if iters == 0:
            h_try4 = best_h.copy()
        else:
            scale = 0.015 * (0.5 + rng.rand())
            h_try4 = best_h + rng.randn(n_final) * scale
            h_try4 = project_feasible(h_try4, n_final)
        h_new4 = slsqp_optimize(h_try4, n_final, dx_final, target_final, max_iter=250)
        c5_new4 = compute_c5(h_new4, dx_final)
        if c5_new4 < best_c5 - 1e-15:
            best_h = h_new4
            best_c5 = c5_new4
        iters += 1
        if elapsed() > t4 * 0.5 and iters > 3:
            break

    return (best_h, best_c5, len(best_h))
