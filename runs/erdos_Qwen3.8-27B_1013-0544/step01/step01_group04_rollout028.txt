import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - target
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-9) & (h < 1.0 - 1e-9)
        if np.any(free):
            h[free] -= diff / np.count_nonzero(free)
        else:
            h -= diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = np.sum(h) - target
    if abs(diff) > 1e-13:
        free = (h > 1e-9) & (h < 1.0 - 1e-9)
        if np.any(free):
            h[free] -= diff / np.count_nonzero(free)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(corr, dx, temp):
    vals = corr * dx
    m = np.max(vals)
    shifted = np.clip((vals - m) / temp, -50, 50)
    return m + temp * np.log(np.sum(np.exp(shifted)))

def slsqp_smooth(h_start, n, dx, tsum, temp, max_iter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - tsum}]
    def obj(x):
        h = np.clip(x, 0, 1)
        c = np.correlate(h, 1.0 - h, mode='full')
        return smooth_max_obj(c, dx, temp)
    try:
        res = minimize(obj, h_start, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': ftol, 'disp': False})
        return project_feasible(res.x, n)
    except:
        return project_feasible(h_start, n)

def slsqp_exact(h_start, n, dx, tsum, max_iter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - tsum}]
    try:
        res = minimize(lambda x: compute_c5(x, dx), h_start, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': ftol, 'disp': False})
        return project_feasible(res.x, n)
    except:
        return project_feasible(h_start, n)
