    corr = np.correlate(h, 1 - h, mode="full")

    lo = v.min() - 2.0
    hi = v.max() + 2.0

        h = np.clip(v - mid, 0.0, 1.0)

        else:

def smooth_max_grad(h, dx, t):

    corr = np.correlate(h, 1 - h, mode="full")
    m = corr.max()
    exp_c = np.exp(corr - m)

    grad = np.zeros(N)
