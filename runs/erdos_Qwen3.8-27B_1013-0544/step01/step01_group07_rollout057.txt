```python
import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def projection_onto_constraint(h, target_sum):
    """Project h onto {x: 0≤x≤1, sum(x)=target_sum}"""
    h = np.clip(h, 0, 1)
    # Water-filling to achieve target_sum
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # Need to increase: increase elements not at 1
            mask = h < 1.0
            if not np.any(mask):
                break
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0, 1)
        else:
            # Need to decrease: decrease elements not at 0
            mask = h > 0.0
            if not np.any(mask):
                break
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0, 1)
    return h

def compute_autocorr(h):
    """Compute A(l) for all lags l."""
    return np.correlate(h, h, mode='full')

def subgradient_step(h, step_size, target_sum):
    """One step of projected subgradient ascent on min_l A(l)."""
    n = len(h)
    auto = compute_autocorr(h)
    l_star = np.argmin(auto)
    
    # Gradient of A(l_star) w.r.t. h
    l = l_star - (n - 1)  # actual lag
    grad = np.zeros(n)
    if l == 0:
        grad = 2 * h
    else:
        # dA(l)/dh[i] = h[i-l] * 1(0<=i-l<n) + h[i+l] * 1(0<=i+l<n)
        for i in range(n):
            g = 0
            if 0 <= i - l < n:
                g += h[i - l]
            if 0 <= i + l < n:
                g += h[i + l]
            grad[i] = g
    
    # We want to maximize A(l_star), so move in gradient direction
    h_new = h + step_size * grad
    h_new = projection_onto_constraint(h_new, target_sum)
    return h_new

def multi_subgradient_step(h, step_size, target_sum, alpha=0.3):
    """Weighted subgradient using all near-minimum lags."""
    n = len(h)
    auto = compute_autocorr(h)
    min_val = np.min(auto)
    # Weight lags by how close they are to the minimum
    weights = np.exp(-alpha * (auto - min_val))
    weights /= np.sum(weights)
    
    # Compute weighted gradient
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-10:
            continue
        l = m - (n - 1)
        if l == 0:
            grad += w * 2 * h
        else:
            for i in range(n):
                g = 0
                if 0 <= i - l < n:
                    g += h[i - l]
                if 0 <= i + l < n:
                    g += h[i + l]
                grad[i] += w * g
    
    h_new = h + step_size * grad
    h_new = projection_onto_constraint(h_new, target_sum)
    return h_new
```

def shifted(h, l):
    """Return array where result[i] = h[i-l] if 0<=i-l<n, else 0"""
    result = np.zeros(len(h))
    if l >= 0:
        result[l:] = h[:len(h)-l]
    else:
        result[:len(h)+l] = h[-l:]
    return result
