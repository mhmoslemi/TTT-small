        max_c = np.max(corr)
        exp_c = np.exp(temperature * (corr - max_c))
        sum_exp = np.sum(exp_c)
        obj = max_c + np.log(sum_exp) / temperature
        w = exp_c / sum_exp
        # Gradient
        grad = np.zeros(n)
        n_half = n - 1
        for k in range(len(corr)):
            if w[k] < 1e-8:
                continue
            m = k - n_half

            jm = j + m  # need to vectorize

        # Vectorized:

        x0 = np.linspace(0, 2, n, endpoint=False)
        x1 = np.linspace(0, 2, n2, endpoint=False)
        h_up = np.interp(x1, x0, best_h, period=2.0)
        h_up = project_feasible(h_up, n2)
        c5_up = compute_c5(h_up, dx2)
        if c5_up < best_c5:
            best_h = h_up
            best_c5 = c5_up
            n = n2
            dx = dx2
            target_sum = target_sum2
        while elapsed() < budget_s - 40:
            h_try = best_h + rng.randn(n) * 0.02
            h_try = project_feasible(h_try, n)
            h_new = slsqp_optimize(h_try, n, dx, target_sum, 0.5, max_iter=200)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5 - 1e-15:
                best_h = h_new
                best_c5 = c5_new

    # Final safety check
    best_h = project_feasible(best_h, len(best_h))
    n_final = len(best_h)
    dx_final = 2.0 / n_final
    best_c5 = compute_c5(best_h, dx_final)

    return (best_h, best_c5, n_final)
