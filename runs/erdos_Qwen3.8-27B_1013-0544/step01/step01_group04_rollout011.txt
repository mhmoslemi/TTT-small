    x_old = np.linspace(0, 2, n, endpoint=False)
    x_new = np.linspace(0, 2, n2, endpoint=False)
    # Fractional index into h
    frac_idx = (x_new / 2.0) * n
    i_lo = np.floor(frac_idx).astype(int) % n
    i_hi = (i_lo + 1) % n
    weight = frac_idx - np.floor(frac_idx)
    h_up = h[i_lo] * (1.0 - weight) + h[i_hi] * weight
    return h_up

def de_optimize(pop, objs, n, dx, F_start, F_end, Cr, time_limit, rng, min_improvement=0.0):
    start = time.time()
    pop_size = len(pop)
    best_idx = np.argmin(objs)
    best_h = pop[best_idx].copy()
    best_obj = objs[best_idx]
    
    gen = 0
    no_improve = 0
    while time.time() - start < time_limit:
        gen += 1
        progress = min(1.0, gen / 500.0)
        F = F_start + (F_end - F_start) * progress
        Cr_t = Cr * (1.0 - 0.5 * progress)
        
        gen_best = best_obj
        for i in range(pop_size):
            if time.time() - start >= time_limit:
                break
            
            # Select 3 distinct parents excluding i
            excl = [j for j in range(pop_size) if j != i]
            r1, r2, r3 = rng.choice(len(excl), 3, replace=False)
            a, b, c = excl[r1], excl[r2], excl[r3]
            
            v = pop[a] + F * (pop[b] - pop[c])
            
            j_rand = rng.randint(n)
            mask = rng.rand(n) < Cr_t
            mask[j_rand] = True
            trial = np.where(mask, v, pop[i])
            
            trial = project_feasible(trial, n)
            t_obj = compute_c5(trial, dx)
            
            if t_obj <= objs[i]:
                pop[i] = trial
                objs[i] = t_obj
            if t_obj < gen_best:
                gen_best = t_obj
        
        if gen_best < best_obj - min_improvement:
            best_obj = gen_best
            best_h = pop[np.argmin(objs)].copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                # Refresh some population members
                for i in range(pop_size // 4):
                    idx = rng.randint(pop_size)
                    pop[idx] = best_h + rng.randn(n) * 0.1
                    pop[idx] = project_feasible(pop[idx], n)
                    objs[idx] = compute_c5(pop[idx], dx)
                no_improve = 0
    
    # Return best
    final_best_idx = np.argmin(objs)
    if objs[final_best_idx] < best_obj:
        best_h = pop[final_best_idx].copy()
        best_obj = objs[final_best_idx]
    
    return best_h, best_obj

def local_refine(h_start, n, dx, target_sum, time_limit, rng):
    """Local refinement using L-BFGS-B with smooth max approximation."""
    start = time.time()
    best_h = h_start.copy()
    best_obj = compute_c5(h_start, dx)
    
    bounds = [(0.0, 1.0)] * n
    
    for alpha in [30, 60, 120, 240]:
        if time.time() - start >= time_limit:
            break
        
        for restart in range(3):
            if time.time() - start >= time_limit:
                break
            
            if restart == 0:
                x0 = best_h.copy()
            else:
                scale = 0.08 * (1.0 - restart / 3.0) + 0.01
                x0 = best_h + rng.randn(n) * scale
                x0 = project_feasible(x0, n)
            
            def obj_func(x):
                h = x  # Already in [0,1] due to bounds
                # Soft sum correction
                diff = np.sum(h) - target_sum
                h_adj = h - diff / n
                h_adj = np.clip(h_adj, 0.0, 1.0)
                corr = np.correlate(h_adj, 1.0 - h_adj, mode='full')
                m = np.max(corr)
                return (m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha) * dx
            
            res = minimize(obj_func, x0, method='L-BFGS-B', bounds=bounds,
                         options={'maxiter': 300, 'ftol': 1e-16, 'gtol': 1e-10})
            h_new = project_feasible(res.x, n)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_obj:
                best_h = h_new
                best_obj = c5_new
    
    return best_h, best_obj

    h = np.clip(h, 0.0, 1.0)
    for _ in range(40):

        if abs(diff) < 1e-13:

    if elapsed() < budget_s - 8:
        best_h, best_c5 = local_refine(best_h, nf, df, nf / 2.0,
                                        budget_s - 8 - elapsed(), rng)

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return (best_h, best_c5, len(best_h))
