            mom_k = np.zeros(n)
            for it in range(3000):
                if tl() < 50:
                    break
                g = topk_gradient(h_work, kk)
                mom_k = 0.8 * mom_k + 0.2 * g
                h_try = project_h(h_work - alpha_k * mom_k, n * 0.5)
                c5_try = compute_c5(h_try)
                if c5_try < compute_c5(h_work):
                    h_work = h_try
                    alpha_k = min(alpha_k * 1.02, 0.2)
                    if c5_try < best_c5:
                        best = h_work.copy()
                        best_c5 = c5_try
                else:
                    alpha_k *= 0.85
            if tl() > 60:
                h_s, c5_s = slsqp_opt(h_work, 200, 1e-12)
                if c5_s < best_c5:
                    best = h_s.copy()
                    best_c5 = c5_s

    # Phase 4: Random restarts with subgradient + SLSQP
    n_restarts = 25
    for r in range(n_restarts):
        if tl() < 70:
            break
        scales = [0.03, 0.05, 0.08, 0.12, 0.18]
        scale = scales[r % len(scales)]
        h_r = project_h(best + rng.randn(n) * scale, n * 0.5)
        # Quick subgradient warm-up
        temp_r = 0.02
        alpha_r = 0.04
        mom_r = np.zeros(n)
        for it in range(200):
            g = smooth_gradient(h_r, temp_r)
            mom_r = 0.8 * mom_r + 0.2 * g
            h_r = project_h(h_r - alpha_r * mom_r, n * 0.5)
            alpha_r *= 0.998
        # SLSQP polish
        h_r, c5_r = slsqp_opt(h_r, 300, 1e-12)
        if c5_r < best_c5:
            best = h_r.copy()
            best_c5 = c5_r

    # Phase 5: Final SLSQP with tight tolerance
    if tl() > 40:
        h_f, c5_f = slsqp_opt(best, 600, 1e-13)
        if c5_f < best_c5:
            best = h_f.copy()
            best_c5 = c5_f

    # Phase 6: Try higher resolution if time permits
    if tl() > 180:
        n_new = 300
        xo = np.arange(n) / n
        xn = np.arange(n_new) / n_new
        h_int = project_h(np.interp(xn, xo, best), n_new * 0.5)
        h_h, c5_h = slsqp_opt(h_int, 400, 1e-12)
        if c5_h < best_c5:
            # Subgradient at higher resolution
            temp_h = 0.01
            alpha_h = 0.03
            mom_h = np.zeros(n_new)
            for it in range(3000):
                if tl() < 50:
                    break
                g = smooth_gradient(h_h, temp_h)
                mom_h = 0.85 * mom_h + 0.15 * g
                h_try = project_h(h_h - alpha_h * mom_h, n_new * 0.5)
                c5_try = compute_c5(h_try)
                if c5_try < c5_h:
                    h_h = h_try
                    c5_h = c5_try
                    alpha_h = min(alpha_h * 1.01, 0.15)
                    if c5_h < best_c5:
                        best = h_h.copy()
                        best_c5 = c5_h
                else:
                    alpha_h *= 0.9
                    if int((1.0 - alpha_h / 0.03) * 20) % 40 == 0:
                        temp_h *= 0.8
            if tl() > 60:
                h_h, c5_h = slsqp_opt(h_h, 400, 1e-13)
                if c5_h < best_c5:
                    best = h_h.copy()
                    best_c5 = c5_h

    best = np.clip(best, 0.0, 1.0)
    best = project_h(best, len(best) * 0.5)
    return best, compute_c5(best), len(best)
