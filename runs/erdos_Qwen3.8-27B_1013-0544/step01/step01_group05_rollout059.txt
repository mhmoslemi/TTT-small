def compute_c5(h, dx):

def compute_lag_grad(h, L):
    n = len(h)
    omh = 1.0 - h
    grad = np.zeros(n)
    if L == 0:
        grad[:] = omh[:] - h[:]
    elif L > 0:
        grad[L:] += omh[:n - L]
        grad[:n - L] -= h[L:]
    else:
        L2 = -L
        grad[:n - L2] += omh[L2:]
        grad[L2:] -= h[:n - L2]
    return grad

    _global_target_sum = target_sum

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except NameError:
        h0 = np.full(n_points, 0.5)
    h0 = project_feasible(h0, target_sum)
    best_h = h0.copy()
    best_c5 = compute_c5(h0, dx)

    # Phase 1: Smooth max gradient with annealed beta
    h = h0.copy()
    betas = [1.0, 3.0, 8.0, 20.0, 50.0, 100.0, 200.0]
    for beta in betas:
        if time.time() > _global_deadline - 200.0:
            break
        step = 0.02 / (1.0 + 0.08 * beta)
        n_it = 400
        h, c5_v = smooth_refine(h, dx, target_sum, n_it, step, beta)
        if c5_v < best_c5:
            best_c5 = c5_v
            best_h = h.copy()

    # Phase 2: Basin hopping
    if time.time() < _global_deadline - 100.0:
        h = best_h.copy()
        sigma = 0.12
        for i in range(600):
            if time.time() > _global_deadline - 100.0:
                break
            h_pert = h + rng.normal(0, sigma, n_points)
            h_pert = project_feasible(h_pert, target_sum)
            h_ref, c5_ref = smooth_refine(h_pert, dx, target_sum, 80, 0.004, 40.0)
            if c5_ref < best_c5:
                best_c5 = c5_ref
                best_h = h_ref.copy()
                h = h_ref.copy()
                sigma = min(0.2, sigma * 1.08)
            else:
                sigma = max(0.025, sigma * 0.96)
            sigma *= 0.999

    # Phase 3: Active-set gradient
    if time.time() < _global_deadline - 50.0:
        h = best_h.copy()
        best_h, best_c5 = active_refine(h, dx, target_sum, 3000, 0.005, 0.65)

    # Phase 4: SLSQP
    if time.time() < _global_deadline - 25.0:
        try:
            res = minimize(_slsqp_obj, best_h, method='SLSQP',
                          bounds=[(0.0, 1.0)] * n_points,
                          constraints=[{'type': 'eq', 'fun': _slsqp_eq}],
                          options={'maxiter': 3000, 'ftol': 1e-14})
            if res.x is not None:
                h_ref = project_feasible(res.x, target_sum)
                c5_ref = compute_c5(h_ref, dx)
                if c5_ref < best_c5:
                    best_c5 = c5_ref
                    best_h = h_ref
        except Exception:
            pass

    # Phase 5: Fine active-set
    if time.time() < _global_deadline - 5.0:
        h = best_h.copy()
        best_h, best_c5 = active_refine(h, dx, target_sum, 5000, 0.0004, 0.4)

    # Phase 6: Local search with many small perturbations
    if time.time() < _global_deadline - 1.0:
        for _ in range(2000):
            if time.time() > _global_deadline:
                break
            h_pert = best_h + rng.normal(0, 0.003, n_points)
            h_pert = project_feasible(h_pert, target_sum)
            h_ref, c5_ref = active_refine(h_pert, dx, target_sum, 50, 0.0005, 0.5)
            if c5_ref < best_c5:
                best_c5 = c5_ref
                best_h = h_ref.copy()

    return (best_h, best_c5, n_points)
