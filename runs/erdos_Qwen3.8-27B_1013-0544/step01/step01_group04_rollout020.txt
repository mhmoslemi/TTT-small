from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - target
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(free):
            h[free] -= diff / np.sum(free)
        else:
            h -= diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = np.sum(h) - target
    if abs(diff) > 1e-13:
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(free):
            h[free] -= diff / np.sum(free)
            h = np.clip(h, 0.0, 1.0)
    return h

def soft_max(corr, tau):
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(corr - m))) / tau

def soft_max_weights(corr, tau):
    m = np.max(corr)
    e = np.exp(corr - m)
    return e / np.sum(e)

def smooth_obj(x, n, tau):
    corr = np.correlate(x, 1.0 - x, mode='full')
    return soft_max(corr, tau)

def smooth_grad(x, n, tau, dx):
    corr = np.correlate(x, 1.0 - x, mode='full')
    w = soft_max_weights(corr, tau)
    grad = np.zeros(n)
    i_idx = np.arange(n)
    for d in range(-(n - 1), n):
        p = n - 1 + d
        j1 = i_idx + d
        j2 = i_idx - d
        m1 = (j1 >= 0) & (j1 < n)
        m2 = (j2 >= 0) & (j2 < n)
        grad[m1] += w[p] * (1.0 - x[j1[m1]])
        grad[m2] -= w[p] * x[j2[m2]]
    return grad * dx

def slsqp_polish(h, n, dx, max_iter=500, ftol=1e-15):

                      'jac': lambda x: np.ones(n)}],
        options={'maxiter': max_iter, 'ftol': ftol}
    )
    return project_feasible(res.x, n)

            lambda x: smooth_obj(x, n, tau),

            method='L-BFGS-B', bounds=bounds,
            options={'maxiter': max_iter_each, 'ftol': 1e-14, 'gtol': 1e-10}
        )
        h = project_feasible(res.x, n)
    return h

    j2 = i_idx - d

    grad[m2] -= h[j2[m2]]
    h_new = h - step_size * grad / dx
    return project_feasible(h_new, n)
