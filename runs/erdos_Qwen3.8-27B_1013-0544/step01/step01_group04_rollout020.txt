def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - target
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            h[free] -= diff / np.sum(free)
        else:

        if np.any(free):

    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(c)
    c_shifted = c - c_max

    term2 = np.zeros(n)
    for j in range(n):
        m = min(n - j, len(w))

def adam_optimize(h_start, n, dx, target, tau_start, tau_end, n_steps, lr=0.01):
    h = h_start.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    best_h = h.copy()
    best_val = compute_c5(h, dx)
    
    for t in range(n_steps):
        progress = t / max(n_steps - 1, 1)
        tau = tau_start * (tau_end / tau_start) ** progress
        
        grad = compute_gradient(h, tau, dx)
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(t+1))
        v_hat = v / (1 - beta2**(t+1))
        
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_feasible(h_new, n)
        
        val = compute_c5(h, dx)
        if val < best_val:
            best_val = val
            best_h = h.copy()
    
    return best_h, best_val

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - target
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            h[free] -= diff / np.sum(free)
        else:
            h -= diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = np.sum(h) - target
    if abs(diff) > 1e-14:
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            h[free] -= diff / np.sum(free)
    h = np.clip(h, 0.0, 1.0)
    return h

def compute_gradient(h, tau, dx):
    n = len(h)
    v = 1.0 - h
    c = np.correlate(h, v, mode='full')
    c_max = np.max(c)
    exp_tau = np.exp(tau * (c - c_max))
    w = exp_tau / np.sum(exp_tau)
    conv_wv = np.convolve(w, v, mode='full')
    term1 = conv_wv[:n]
    term2 = np.zeros(n)
    for j in range(n):
        m = min(n - j, len(w))
        term2[j] = np.dot(w[:m], h[j:j+m])
    return dx * (term1 - term2)

def adam_optimize(h_start, n, dx, tau_start, tau_end, n_steps, lr, time_limit=None, t0=0):
    h = h_start.copy()
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    best_h = h.copy()
    best_val = compute_c5(h, dx)
    for t in range(n_steps):
        if time_limit is not None and (t0 + time.time() - t0) > time_limit:
            break
        progress = t / max(n_steps - 1, 1)
        tau = tau_start * (tau_end / tau_start) ** progress
        lr_t = lr * (1.0 - 0.5 * progress)
        grad = compute_gradient(h, tau, dx)
        m_adam = beta1 * m_adam + (1 - beta1) * grad
        v_adam = beta2 * v_adam + (1 - beta2) * (grad * grad)
        m_hat = m_adam / (1.0 - beta1 ** (t + 1))
        v_hat = v_adam / (1.0 - beta2 ** (t + 1))
        h_new = h - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        h = project_feasible(h_new, n)
        val = compute_c5(h, dx)
        if val < best_val:
            best_val = val
            best_h = h.copy()
    return best_h, best_val

def bit_flip_local_search(h, n, dx, max_sweeps=50):
    h = h.copy()
    target = n / 2.0
    binary = (h > 0.5).astype(float)
    c5 = compute_c5(binary, dx)
    for sweep in range(max_sweeps):
        improved = False
        for p in range(n):
            if binary[p] > 0.5:
                new_h = binary.copy()
                new_h[p] = 0.0
            else:
                new_h = binary.copy()
                new_h[p] = 1.0
            new_sum = np.sum(new_h)
            if abs(new_sum - target) > 1e-10:
                continue
            new_c5 = compute_c5(new_h, dx)
            if new_c5 < c5 - 1e-14:
                binary = new_h
                c5 = new_c5
                improved = True
        if not improved:
            break
    return binary, c5
