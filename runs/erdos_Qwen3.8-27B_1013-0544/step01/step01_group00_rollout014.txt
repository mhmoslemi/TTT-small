from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def compute_c5_arr(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_h(h, target):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = np.sum(mask)
    if cnt > 0:
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def solve_1d_minimax(coeffs, consts, lb=0.0, ub=1.0):
    """Solve min_{t in [lb,ub]} max_k (consts[k] + coeffs[k]*t) via bisection."""
    lo, hi = lb, ub
    for _ in range(50):
        mid = (lo + hi) * 0.5
        vals = consts + coeffs * mid
        mx = vals.max()
        active = vals >= mx - 1e-12
        if not np.any(active):
            active = np.zeros(len(vals), dtype=bool)
            active[vals.argmax()] = True
        min_s = coeffs[active].min()
        max_s = coeffs[active].max()
        if min_s > 1e-14:
            hi = mid
        elif max_s < -1e-14:
            lo = mid
        else:
            return mid
    return (lo + hi) * 0.5

def minimax_coord_descent(h, n_sweeps=10):
    n = len(h)
    h = h.copy()
    c5_best = compute_c5(h)
    
    for sweep in range(n_sweeps):
        m_vals = np.arange(2 * n - 1) - (n - 1)
        for j in range(n):
            corr = np.correlate(h, 1.0 - h, mode='full')
            valid1 = (j - m_vals >= 0) & (j - m_vals < n)
            valid2 = (j + m_vals >= 0) & (j + m_vals < n)
            
            # Current contribution of h[j] to each lag
            contrib = np.zeros(2 * n - 1)
            contrib[valid1] += h[j] * (1.0 - h[j - m_vals[valid1]])
            contrib[valid2] += h[j + m_vals[valid2]] * (1.0 - h[j])
            
            # New contribution if h[j] = t:
            # t*(1-h[j-m]) + h[j+m]*(1-t) = t*(1-h[j-m]-h[j+m]) + h[j+m]
            coeff = np.zeros(2 * n - 1)
            coeff[valid1] += (1.0 - h[j - m_vals[valid1]])
            coeff[valid2] += (-h[j + m_vals[valid2]])
            
            const = np.zeros(2 * n - 1)
            const[valid2] += h[j + m_vals[valid2]]
            
            base = corr - contrib + const
            
            t_opt = solve_1d_minimax(coeff, base)
            h[j] = t_opt
        
        h = np.clip(h, 0.0, 1.0)
        h = project_h(h, n * 0.5)
        c5_cur = compute_c5(h)
        if c5_cur < c5_best:
            c5_best = c5_cur
    
    return h, c5_best

            if c5_new < c5_best:

    return v[-1]

The function f(t) = max_k (base_k + coeff_k * t) is convex (pointwise max of linear functions). The bisection on the subgradient is correct for finding the minimum of a convex function.

The subgradient of f at t is the convex hull of {coeff_k : k ∈ A(t)} where A(t) = {k : base_k + coeff_k * t = f(t)} is the active set.

f is minimized at t* iff 0 ∈ ∂f(t*), i.e., iff min_{k∈A(t*)} coeff_k ≤ 0 ≤ max_{k∈A(t*)} coeff_k.

So the bisection is:
- If all active coeff_k > 0: f is increasing at t, so minimum is to the left → hi = mid
- If all active coeff_k < 0: f is decreasing at t, so minimum is to the right → lo = mid
- Otherwise: 0 is in the subdifferential, we found the minimum

This is correct!

    return t - corr

def t_eq(v):
    n = len(v) - 1
    return np.sum(v[:n]) - n * 0.5

def t_eq_jac(v):
    n = len(v) - 1
    j = np.zeros(len(v))
    j[:n] = 1.0
    return j

def slsqp_tvar(h0, max_iter=300):
    n = len(h0)
    c5_cur = compute_c5(h0)
    v0 = np.append(h0.copy(), c5_cur + 0.05)
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.5)]
    cons = [
        {'type': 'ineq', 'fun': t_ineq},
        {'type': 'eq', 'fun': t_eq, 'jac': t_eq_jac}
    ]
    res = minimize(t_obj, v0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-10})
    h_new = project_h(res.x[:n], n * 0.5)
    return h_new, compute_c5(h_new)

def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.arange(n_old) / n_old
    x_new = np.arange(n_new) / n_new
    return project_h(np.interp(x_new, x_old, h_old), n_new * 0.5)
