def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-15:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    mc = c.max()
    shifted = alpha * (c - mc)
    return float(mc + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    mc = c.max()
    shifted = alpha * (c - mc)
    exp_s = np.exp(shifted)
    sm = exp_s / exp_s.sum()
    grad = np.zeros(n)
    omh = 1.0 - h
    for m in range(2 * n - 1):
        w = sm[m]
        if w < 1e-12:

        d = m - (n - 1)
        t1 = np.zeros(n)
        if d >= 0:
            t1[d:] = omh[:n - d]

        t2 = np.zeros(n)

        if c < best_c5:
            best_h, best_c5 = h_new, c

    # Final coordinate descent polish
    if elapsed() < budget_s - 5:
        best_h, best_c5 = coord_descent(best_h, n, max_rounds=2)

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)

    return best_h, best_c5, n
