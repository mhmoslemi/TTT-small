_global_dx = None
_global_target_sum = None

def project_feasible(x, target_sum):
    lo = float(x.min()) - 1.0
    hi = float(x.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = float(np.clip(x - mid, 0.0, 1.0).sum())
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def smooth_grad(h, T):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    w = np.exp(T * (corr - max_val))
    ws = np.sum(w)
    if ws < 1e-300:
        return np.zeros(n)
    w /= ws
    n_half = n - 1

            grad[:n-L] -= wk * h[L:]
        else:
            L2 = -L
            grad[:n-L2] += wk * (1.0 - h[L2:])
            grad[L2:] -= wk * h[:n-L2]

            grad[L2:] -= wk * h[:n-L2]

import numpy as np
from scipy.optimize import minimize
import time

_global_dx = None
_global_target_sum = None

def project_feasible(x, target_sum):
    lo = float(x.min()) - 1.0
    hi = float(x.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = float(np.clip(x - mid, 0.0, 1.0).sum())
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def lagg_grad(h, L):
    n = len(h)
    g = np.zeros(n)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += (1.0 - h[:n-L])
        g[:n-L] -= h[L:]
    else:
        L2 = -L
        g[:n-L2] += (1.0 - h[L2:])
        g[L2:] -= h[:n-L2]
    return g

def smooth_max_grad(h, T):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    w = np.exp(T * (corr - max_val))
    ws = np.sum(w)
    if ws < 1e-300:
        return np.zeros(n)
    w /= ws
    n_half = n - 1
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-7:
            continue
        L = k - n_half
        grad += wk * lagg_grad(h, L)
    return grad

def active_smooth_grad(h, T_act):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    w = np.exp(T_act * (corr - c_max))
    ws = np.sum(w)
    if ws < 1e-300:
        return np.zeros(n)
    w /= ws
    n_half = n - 1
    grad = np.zeros(n)

            prev_c5 = c5_new

def run_slsqp(h, target_sum, dx, n_points, tleft_fn, best_state):

        return best_state
