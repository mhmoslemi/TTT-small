def compute_gradient(h, tau):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    x = corr / tau
    x_max = x.max()
    p = np.exp(x - x_max)
    p /= p.sum()
    
    # p has length 2n-1, indexed by full correlation output
    # p[m] corresponds to lag k = m - (n-1)
    
    grad = np.zeros(n)
    for m in range(2*n - 1):
        k = m - (n - 1)  # lag
        for j in range(n):
            # d(corr_k)/d(h[j]) = (1-h[j+k])*[0<=j+k<n] - h[j-k]*[0<=j-k<n]
            contrib = 0.0
            if 0 <= j + k < n:
                contrib += (1 - h[j+k])
            if 0 <= j - k < n:
                contrib -= h[j-k]
            grad[j] += p[m] * contrib
    grad /= tau
    return grad

```python
import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val(corr, tau):
    x = corr / tau
    x_max = x.max()
    return float(tau * (x_max + np.log(np.exp(x - x_max).sum())))

def project_constrained(v, target):
    lo, hi = v.min() - 1, v.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.clip(v - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2, 0, 1)

def compute_gradient(h, tau):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    x = corr / tau
    x_max = x.max()
    p = np.exp(x - x_max)
    p /= p.sum()
    
    grad = np.zeros(n)
    if p[n-1] > 1e-15:
        grad += p[n-1] * (1.0 - 2.0 * h)
    
    for m in range(1, n):
        kp = p[n-1+m]
        km = p[n-1-m]
        if kp > 1e-15:
            t1 = np.zeros(n)
            t1[:n-m] = 1.0 - h[m:]
            t2 = np.zeros(n)
            t2[m:] = h[:n-m]
            grad += kp * (t1 - t2)
        if km > 1e-15:
            t1 = np.zeros(n)
            t1[m:] = 1.0 - h[:n-m]
            t2 = np.zeros(n)
            t2[:n-m] = h[m:]
            grad += km * (t1 - t2)
    
    grad /= tau
    return grad

def nesterov_opt(h0, n, tau_start, tau_end, n_iter, lr, t_start, budget_s):
    target = n / 2.0
    h = project_constrained(h0.copy(), target)
    v = np.zeros(n)
    momentum = 0.9
    
    for i in range(n_iter):
        if budget_s - (time.time() - t_start) < 3:
            break
        tau = tau_start * (tau_end / tau_start) ** (i / max(n_iter - 1, 1))
        y = h + momentum * v
        y = project_constrained(y, target)
        grad = compute_gradient(y, tau)
        step = lr / (i + 1) ** 0.5  # or use backtracking
        h_new = project_constrained(y - step * grad, target)
        v = h_new - h
        h = h_new
    
    return h

def fix_sum(h, target):
    h = np.array(h, dtype=float)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return np.clip(h, 0, 1)
    mask = h < 1.0 if diff > 0 else h > 0.0
    cnt = mask.sum()
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0, 1)

def generate_seeds(n, rng, n_seeds=10):
    seeds = []
    half = n // 2
    # Random binary
    for _ in range(n_seeds // 2):
        idx = rng.choice(n, half, replace=False)
        b = np.zeros(n)
        b[idx] = 1.0
        seeds.append(b)
    # Block patterns
    for n_blocks in [2, 3, 4, 5, 7]:
        if len(seeds) >= n_seeds:
            break
        b = np.zeros(n)
        block_size = n // n_blocks
        for bi in range(n_blocks):
            start = bi * block_size
            end = start + block_size if bi < n_blocks - 1 else n
            if bi % 2 == 0:
                b[start:end] = 1.0
        if b.sum() > half:
            ones = np.where(b == 1)[0]
            rng.shuffle(ones)
            b[ones[int(b.sum()) - half:]] = 0
        elif b.sum() < half:
            zeros = np.where(b == 0)[0]
            rng.shuffle(zeros)
            b[zeros[:half - int(b.sum())]] = 1
        seeds.append(b)
    while len(seeds) < n_seeds:
        idx = rng.choice(n, half, replace=False)
        b = np.zeros(n)
        b[idx] = 1.0
        seeds.append(b)
    return seeds[:n_seeds]

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    
    def rem():
        return budget_s - (time.time() - t0)
    
    # Get initial values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(120, 0.5)
    
    best_h = h_init.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Phase 1: Binary optimization at multiple resolutions
    for n in [60, 80, 100]:
        if rem() < 30:
            break
        seeds = generate_seeds(n, rng, n_seeds=6)
        for b in seeds:
            if rem() < 10:
                break
            # Binary local search (bit-pair swaps)
            c5 = evaluate_c5(b)
            for _ in range(200):
                if rem() < 5:
                    break
                improved = False
                for _ in range(10):
                    i, j = rng.choice(n, 2, replace=False)
                    if b[i] == b[j]:
                        continue
                    b[i], b[j] = 1 - b[i], 1 - b[j]
                    c5_new = evaluate_c5(b)
                    if c5_new < c5:
                        c5 = c5_new
                        improved = True
                    else:
