import numpy as np

from scipy.optimize import minimize, minimize_scalar

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_constraint(h, target_sum):
    """Project h onto {x: 0<=x_i<=1, sum(x)=target_sum}"""
    n = len(h)
    # Bisection on lambda
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h_proj = np.clip(h - mid, 0, 1)
        s = np.sum(h_proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo+hi)/2, 0, 1)
    # Fine-tune
    diff = target_sum - np.sum(h_proj)
    if abs(diff) > 1e-10:
        # Distribute to coordinates not at boundary
        mask = (h_proj > 1e-8) & (h_proj < 1 - 1e-8)
        if np.sum(mask) > 0:
            h_proj[mask] += diff / np.sum(mask)
            h_proj = np.clip(h_proj, 0, 1)
    return h_proj

def solve_1d(corr0, corr1, n_grid=200):
    """Solve min_t max_k (corr0[k] + t*(corr1[k]-corr0[k])) for t in [0,1]"""
    a = corr1 - corr0
    b = corr0
    t_grid = np.linspace(0, 1, n_grid)
    vals = np.max(b[None, :] + t_grid[:, None] * a[None, :], axis=1)
    idx = np.argmin(vals)
    return t_grid[idx]

def coordinate_descent(h, dx, target_sum, n_rounds=50, n_grid=200):
    n = len(h)
    for r in range(n_rounds):
        improved = 0
        for j in range(n):
            old = h[j]
            h[j] = 0.0
            corr0 = np.correlate(h, 1-h, mode="full") * dx
            h[j] = 1.0
            corr1 = np.correlate(h, 1-h, mode="full") * dx
            h[j] = old
            t_opt = solve_1d(corr0, corr1, n_grid)
            if abs(t_opt - old) > 1e-6:
                improved += 1
            h[j] = t_opt
        h = project_to_constraint(h, target_sum)
        if improved == 0:
            break
    return h

def smooth_max_obj(h, t, dx):
    corr = np.correlate(h, 1-h, mode="full") * dx
    m = np.max(corr)
    return m + (1.0/t) * np.log(np.sum(np.exp(corr - m)))

def lbfgs_optimize(h, dx, target_sum, t=50, maxiter=2000):
    n = len(h)
    def obj(x):
        corr = np.correlate(x, 1-x, mode="full") * dx
        m = np.max(corr)
        smooth = m + (1.0/t) * np.log(np.sum(np.exp(corr - m)))
        penalty = 100.0 * (np.sum(x) - target_sum)**2
        return smooth + penalty
    
    def grad(x):
        corr = np.correlate(x, 1-x, mode="full") * dx
        m = np.max(corr)
        exp_vals = np.exp(corr - m)
        weights = exp_vals / np.sum(exp_vals)
        # Gradient of smooth max w.r.t. x
        # corr depends on x linearly in each coordinate
        # d(corr_k)/dx[j] = (1-x[j+k]) - (x[j-k] + x[j+k]) ... complex
        # Use numerical gradient or just use L-BFGS-B without gradient
        penalty_grad = 200.0 * (np.sum(x) - target_sum) * np.ones(n)
        return penalty_grad  # approximate
    
    # Just use L-BFGS-B without gradient (finite differences)
    res = minimize(obj, h.copy(), method='L-BFGS-B',
                   bounds=[(0,1)]*n, options={'maxiter': maxiter, 'ftol': 1e-12})
    h_new = project_to_constraint(res.x, target_sum)
    return h_new

def upsample(h, new_n):

    x_new = np.linspace(0, 2, new_n + 1)[:-1]
    h_new = np.interp(x_new, x_old, h)
    return h_new

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_to_constraint(h, target_sum):
    n = len(h)
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h_p = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - np.sum(h_p)
    if abs(diff) > 1e-9:
        mask = (h_p > 1e-8) & (h_p < 1.0 - 1e-8)
        cnt = np.sum(mask)
        if cnt > 0:
            h_p[mask] += diff / cnt
            h_p = np.clip(h_p, 0.0, 1.0)
    return h_p

def solve_1d_subproblem(corr0, corr1, n_grid=256):
    a = corr1 - corr0
    b = corr0
    t_grid = np.linspace(0.0, 1.0, n_grid)
    vals = b[None, :] + t_grid[:, None] * a[None, :]
    obj = np.max(vals, axis=1)
    idx = np.argmin(obj)
    return t_grid[idx]

def coordinate_descent(h, dx, target_sum, n_rounds=50, n_grid=256, rng=None):
    n = len(h)
    order = np.arange(n)
    for r in range(n_rounds):
        if rng is not None:
            rng.shuffle(order)
        for j in order:
            old = h[j]
            h[j] = 0.0
            corr0 = np.correlate(h, 1.0 - h, mode="full") * dx
            h[j] = 1.0
            corr1 = np.correlate(h, 1.0 - h, mode="full") * dx
            t_opt = solve_1d_subproblem(corr0, corr1, n_grid)
            h[j] = t_opt
        h = project_to_constraint(h, target_sum)
    return h

def smooth_max_obj(h, t, dx):
    corr = np.correlate(h, 1 - h, mode="full") * dx
    m = np.max(corr)
    return m + (1.0 / t) * np.log(np.sum(np.exp(corr - m)))

def slsqp_refine(h, dx, target_sum, t=80, maxiter=500):
    n = len(h)
    try:
        res = minimize(
            lambda x: smooth_max_obj(x, t, dx),
            h.copy(),
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            options={'maxiter': maxiter, 'ftol': 1e-10}
        )
        h_new = project_to_constraint(res.x, target_sum)
        return h_new
    except Exception:
        return h

def upsample_to_n(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0, 2, old_n + 1)[:-1]
    x_new = np.linspace(0, 2, new_n + 1)[:-1]
    return np.interp(x_new, x_old, h)

def make_structured_init(n, mode=0):
    x = np.linspace(0, 2, n)
    if mode == 0:
        h = (1.0 + np.cos(2.0 * np.pi * x * 3.0)) / 2.0
    elif mode == 1:
        h = np.where(x % (4.0 / n * 4) < 2.0 / n * 2, 1.0, 0.0)
    else:
        h = 0.5 * np.ones(n)
    return h
