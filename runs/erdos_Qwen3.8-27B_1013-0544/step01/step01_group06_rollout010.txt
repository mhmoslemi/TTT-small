import numpy as np
import time

def compute_c5(h):
    n = len(h)
    return np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n)

def sm_grad(corr, alpha):
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    s = w.sum()
    w /= s
    return m + np.log(s) / alpha, w

def g_shift(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += 1.0 - h[:n - k]
        if k:
            g[:n - k] -= h[k:]
    else:
        p = -k
        g[:n - p] += 1.0 - h[p:]
        if p:
            g[p:] -= h[:n - p]
    return g

def obj_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    _, w = sm_grad(corr, alpha)
    g = np.zeros(n)
    for i in np.where(w > 1e-8)[0]:
        g += w[i] * g_shift(h, i - (n - 1))
    return g * dx

def proj(h, ts):
    lo, hi = -1.5, 1.5
    for _ in range(55):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0, 1)) > ts:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def adam_opt(h0, ts, deadline, rng, a_s=20, a_e=300, lr=0.01, max_rest=6):

    h = h0.astype(np.float64).copy()

    m = np.zeros(n)
    v = np.zeros(n)

        alpha = a_s * (a_e / a_s) ** frac
        g = obj_grad(h, alpha)
        t += 1

        mh = m / (1 - 0.9 ** t)
