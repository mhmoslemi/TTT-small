    h = h.copy()
    best = compute_c5(h, dx)
    start = time.time()
    for p in range(max_passes):
        if time_limit is not None and time.time() - start > time_limit:
            break
        improved = False
        for i in range(n):
            # Try adjusting h[i] up or down, compensating at a random other index
            for delta in [0.005, -0.005, 0.01, -0.01, 0.02, -0.02]:
                if h[i] + delta < 0 or h[i] + delta > 1:
                    continue
                # Find best j to compensate
                h[i] += delta
                h = project_feasible(h, n)
                c5_new = compute_c5(h, dx)
                if c5_new < best - 1e-14:
                    best = c5_new
                    improved = True
                    break
                else:
                    h[i] -= delta
                    h = project_feasible(h, n)
        if not improved:
            break
    return h, best
