        g[:n - L2] -= h[L2:]
    return g

def _softmax_grad(h, dx, n, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = beta * corr * dx
    sm = float(scaled.max())
    ev = np.exp(scaled - sm)
    w = ev / float(ev.sum())
    g = np.zeros(n)
    nh = n - 1
    for k in np.where(w > 1e-10)[0]:
        g += w[k] * _grad_one(h, int(k) - nh)
    return g

def _obj_nm(x):
    return compute_c5(project_feasible(x, _g_target), _g_dx)

def _con_slsqp(x):
    return float(np.sum(x)) - _g_target
