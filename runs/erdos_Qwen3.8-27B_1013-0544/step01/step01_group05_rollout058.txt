    exps = np.exp(beta * (corr - max_c))
    sw = float(np.sum(exps))
    weights = exps / sw
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-14:
            continue
        L = int(k) - n_half
        if L == 0:
            grad += w * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += w * (1.0 - h[:n-L])
            grad[:n-L] -= w * h[L:]
        else:
            L2 = -L
            grad[:n-L2] += w * (1.0 - h[L2:])
            grad[L2:] -= w * h[:n-L2]
    return grad * dx

def grad_exact(h, dx, tol_frac):
    corr = np.correlate(h, 1.0 - h, mode='full')
    n = len(h)
    n_half = n - 1
    c_max = float(np.max(corr))
    tol = max(1e-12, c_max * tol_frac)
    active = np.where(corr >= c_max - tol)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(corr))])
    grad = np.zeros(n)
    for k in active:
        L = int(k) - n_half
        if L == 0:
            grad += (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += (1.0 - h[:n-L])
            grad[:n-L] -= h[L:]
        else:
            L2 = -L
            grad[:n-L2] += (1.0 - h[L2:])
            grad[L2:] -= h[:n-L2]
    grad /= float(len(active))
    return grad * dx

def slsqp_obj(x):
    h_p = project_feasible(x, _target_sum)
    corr = np.correlate(h_p, 1.0 - h_p, mode='full')
    max_c = float(np.max(corr))
    exps = np.exp(_slsqp_beta * (corr - max_c))
    return (max_c + np.log(float(np.sum(exps))) / _slsqp_beta) * _dx

def slsqp_eq(x):
    return np.sum(x) - _target_sum

def run(seed=42, budget_s=1100, **kwargs):
    global _dx, _target_sum, _n_points
    start_time = time.time()
    rng = np.random.RandomState(seed)

    n_points = 70
    _n_points = n_points
    _dx = 2.0 / n_points
    _target_sum = n_points / 2.0
    deadline = start_time + budget_s - 8.0

    # Get initial values
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except NameError:
        h0 = np.full(n_points, 0.5)
    h0 = project_feasible(h0, _target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(h0, _dx)

    # Phase 1: FISTA with annealed beta (smooth max)
    # This uses log-sum-exp for a differentiable surrogate of the max
    h_curr = h0.copy()
    h_prev = h0.copy()
    t_val = 1.0

    beta_schedule = [8, 20, 50, 120, 300, 700, 1500]
    for beta in beta_schedule:
        step = 0.004 / (1.0 + 0.1 * beta / 50.0)
        n_iter = max(200, int(3000.0 / (1.0 + beta / 100.0)))
        for it in range(n_iter):
            if time.time() > deadline - 200.0:
                break
            grad = grad_smooth(h_curr, _dx, beta)
            t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
            mom = (t_val - 1.0) / t_next
            y = h_curr + mom * (h_curr - h_prev)
            h_new = project_feasible(y - step * grad, _target_sum)
            c5_new = compute_c5(h_new, _dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_prev = h_curr
            h_curr = h_new
            t_val = t_next
        t_val = 1.0
        h_prev = h_curr.copy()

    best_c5 = compute_c5(best_h, _dx)

    # Phase 2: Projected gradient with exact subgradient (polishing)
    h_curr = best_h.copy()
    step = 0.0008
    momentum = np.zeros(n_points)
    mu = 0.4
    stall = 0

    while time.time() < deadline - 150.0:
        grad = grad_exact(h_curr, _dx, 0.003)
        momentum = mu * momentum - step * grad
        h_new = project_feasible(h_curr + momentum, _target_sum)
        c5_new = compute_c5(h_new, _dx)
        h_curr = h_new
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1
            if stall % 80 == 0:
                step *= 0.82
                if step < 1e-9:
                    h_curr = best_h + rng.normal(0, 0.008, n_points)
                    h_curr = project_feasible(h_curr, _target_sum)
                    momentum = np.zeros(n_points)
                    step = 0.0004
                    stall = 0

    # Phase 3: Try structured starting points (sinusoidal with various freqs/phases)
    freqs = [2, 3, 4, 5, 7, 9, 13]
    for freq in freqs:
        if time.time() > deadline - 60.0:
            break
        for phase in [0.0, np.pi / 3.0, 2.0 * np.pi / 3.0]:
            if time.time() > deadline - 40.0:
                break
            h_sin = 0.5 + 0.45 * np.sin(2.0 * np.pi * freq * np.arange(n_points) / n_points + phase)
            h_sin = project_feasible(h_sin, _target_sum)
            c5_s = compute_c5(h_sin, _dx)
            if c5_s < best_c5:
                # Quick FISTA polish from this start
                h_curr = h_sin.copy()
                h_prev = h_sin.copy()
                tv = 1.0
                for beta2 in [50, 200, 600]:
                    step2 = 0.002
                    for it2 in range(800):
                        if time.time() > deadline - 30.0:
                            break
                        g2 = grad_smooth(h_curr, _dx, beta2)
                        tn = (1.0 + np.sqrt(1.0 + 4.0 * tv * tv)) / 2.0
                        mc = (tv - 1.0) / tn
                        y2 = h_curr + mc * (h_curr - h_prev)
                        h_new2 = project_feasible(y2 - step2 * g2, _target_sum)
                        c5_2 = compute_c5(h_new2, _dx)
                        if c5_2 < best_c5:
                            best_c5 = c5_2
                            best_h = h_new2.copy()
                        h_prev = h_curr
                        h_curr = h_new2
                        tv = tn
                    tv = 1.0
                    h_prev = h_curr.copy()

    # Phase 4: SLSQP with smooth objective from best solution
    if time.time() < deadline - 40.0:
        _slsqp_beta = 500.0
        try:
            res = minimize(slsqp_obj, best_h, method='SLSQP',
                          bounds=[(0.0, 1.0)] * n_points,
                          constraints=[{'type': 'eq', 'fun': slsqp_eq}],
                          options={'maxiter': 5000, 'ftol': 1e-15})
            if res.x is not None:
                h_ref = project_feasible(res.x, _target_sum)
                c5_ref = compute_c5(h_ref, _dx)
                if c5_ref < best_c5:
                    best_c5 = c5_ref
                    best_h = h_ref
        except Exception:
            pass

    # Phase 5: Final fine polishing with exact gradient
    h_curr = best_h.copy()
    step = 0.00015
    momentum = np.zeros(n_points)
    mu3 = 0.2
    for it in range(3000):
        if time.time() > deadline:
            break
        grad = grad_exact(h_curr, _dx, 0.001)
        momentum = mu3 * momentum - step * grad
        h_new = project_feasible(h_curr + momentum, _target_sum)
        c5_new = compute_c5(h_new, _dx)
        h_curr = h_new
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        elif it > 200 and it % 150 == 0:
            step *= 0.7
            if step < 1e-11:
                break

    # Phase 6: Very fine local search - try small coordinate perturbations
    if time.time() < deadline:
        for coord in range(n_points):
            if time.time() > deadline:
                break
            for delta in [-0.02, -0.01, -0.005, 0.005, 0.01, 0.02]:
                h_test = best_h.copy()
                h_test[coord] += delta
                h_test = project_feasible(h_test, _target_sum)
                c5_test = compute_c5(h_test, _dx)
                if c5_test < best_c5:
                    best_c5 = c5_test
                    best_h = h_test.copy()

    return (best_h, best_c5, n_points)
