def subgradient_minmax(h, n, dx, n_iter=1000, step_scale=1.0):

    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    for t in range(n_iter):
        # Compute all overlaps
        corr = np.correlate(h, 1.0 - h, mode='full')
        overlaps = corr * dx
        
        # Find the active lag (max overlap)
        k_star = np.argmax(overlaps)
        # Convert to shift: shift = k_star - (n-1)
        s = k_star - (n - 1)
        
        # Compute subgradient of overlap at shift s
        # d(overlap_s)/d(h[j]) = (1 - h[j+s]) - h[j-s]
        grad = np.zeros(n)
        # (1 - h[j+s]) term: for j in [0, n-1-s] if s>=0, or j in [-s, n-1] if s<0
        if s >= 0:

        else:
            grad[-s:] += (1.0 - h[:n+s])  # (1-h[j+s]) for j=-s..n-1

        else:
            grad[:n+s] -= h[-s:]  # -h[j-s] for j=0..n-1+s

        h = h_new
