import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
            total = (1.0 - h[mask]).sum()
            if total < 1e-12:
                break
            h[mask] += (diff / total) * (1.0 - h[mask])
        else:
            mask = h > 1e-12
            total = h[mask].sum()
            if total < 1e-12:
                break
            h[mask] += (diff / total) * h[mask]
    return np.clip(h, 0.0, 1.0)

def obj_smooth(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def sum_con(h, n):
    return np.sum(h) - n / 2.0

def sum_con_jac(h, n):
    return np.ones(n)

def slsqp_step(h, n, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        h_try = project_feasible(res.x, n)
        c = compute_c5(h_try, n)
        return h_try, c
    except Exception:
        return h, compute_c5(h, n)

def coord_descent(h, n, t0, budget_s, max_rounds=60):
    best_h = h.copy()
    best_c5 = compute_c5(best_h, n)
    for _ in range(max_rounds):
        if budget_s - (time.time() - t0) < 5:
            break
        improved = False
        for i in range(n):
            if budget_s - (time.time() - t0) < 1:
                break
            orig = best_h[i]
            for scale in [0.005, 0.02, -0.02, 0.05, -0.05, 0.1, -0.1, 0.01, -0.01, 0.15, -0.15]:
                h_try = best_h.copy()
                h_try[i] = orig + scale
                h_try = project_feasible(h_try, n)
                c = compute_c5(h_try, n)
                if c < best_c5 - 1e-13:
                    best_h = h_try
                    best_c5 = c
                    orig = best_h[i]
                    improved = True
                    break
        if not improved:
            break
    return best_h, best_c5

def swap_descent(h, n, t0, budget_s, max_rounds=40, n_random=200):
    best_h = h.copy()
    best_c5 = compute_c5(best_h, n)
    rng = np.random.RandomState(999)
    deltas = [0.005, 0.01, 0.02, 0.05, 0.1]
    for _ in range(max_rounds):
        if budget_s - (time.time() - t0) < 5:
            break
        improved = False
        for _ in range(n_random):
            if budget_s - (time.time() - t0) < 1:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            for d in deltas:
                for s in [1, -1]:
                    di = s * d
                    hi = best_h[i] + di
                    hj = best_h[j] - di
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h_try = best_h.copy()
                        h_try[i] = hi
                        h_try[j] = hj
                        c = compute_c5(h_try, n)
                        if c < best_c5 - 1e-13:
                            best_h = h_try
                            best_c5 = c
                            improved = True
                            break
                if improved:
                    break
        if not improved:
            break
    return best_h, best_c5
