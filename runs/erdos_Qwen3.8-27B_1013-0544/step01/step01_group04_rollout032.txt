def project_feasible(h, n):
    target = n / 2.0

        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - target
        if abs(diff) < 1e-14:

        bounds=bounds,

def coord_descent_sweep(h, n, dx, n_sweeps=50, n_grid=30):

    target = n / 2.0

            old_val = h[i]
            # Try grid of values
            grid = np.linspace(0.0, 1.0, n_grid)
            best_x = old_val
