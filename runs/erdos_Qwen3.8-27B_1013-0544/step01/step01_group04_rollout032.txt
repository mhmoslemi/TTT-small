def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    """Project h onto {x: sum(x) = target_sum, 0 <= x <= 1}"""
    n = len(h)
    # Bisection on lambda
    lo, hi = -max(h), 1 - min(h) + 1
    for _ in range(100):

        proj = np.clip(h - mid, 0, 1)
        if np.sum(proj) > target_sum:

        else:

import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = np.sum(result) - target_sum
    if abs(diff) > 1e-12:
        free = (result > 1e-10) & (result < 1.0 - 1e-10)
        if np.any(free):
            result[free] -= diff / np.sum(free)
            result = np.clip(result, 0.0, 1.0)
    return result

def smooth_grad(h, dx, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr / alpha - mc / alpha
    w = np.exp(shifted)
    w /= np.sum(w)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        s = k - (n - 1)
        a = max(0, s)
        b = min(n, n + s)
        if a < b:
            grad[a:b] += w[k] * (1.0 - h[a - s:b - s])
        c = max(0, -s)
        d = min(n, n - s)
        if c < d:
            grad[c:d] -= w[k] * h[c + s:d + s]
    return grad * dx

def smooth_obj(h, dx, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr / alpha - mc / alpha
    return (mc + alpha * np.log(np.sum(np.exp(shifted)))) * dx

def slsqp_smooth(h0, n, dx, target_sum, alpha, max_iter=500):
    bounds = [(0.0, 1.0)] * n
    def f(x):
        return smooth_obj(x, dx, alpha)
    def g(x):
        return smooth_grad(x, dx, alpha)
    def c_eq(x):
        return np.sum(x) - target_sum
    def cg(x):
        return np.ones(n)
    try:
        res = minimize(f, h0, method='SLSQP', jac=g, bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': c_eq, 'jac': cg}],
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        return project_feasible(res.x, target_sum)
    except Exception:
        return project_feasible(h0, target_sum)

def slsqp_direct(h0, n, dx, target_sum, max_iter=500):
    bounds = [(0.0, 1.0)] * n
    def f(x):
        return compute_c5(x, dx)
    def c_eq(x):
        return np.sum(x) - target_sum
    try:
        res = minimize(f, h0, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': c_eq}],
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        return project_feasible(res.x, target_sum)
    except Exception:
        return project_feasible(h0, target_sum)

def subgradient_loop(h, n, dx, target_sum, alpha, eta, steps):
    for _ in range(steps):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        shifted = corr / alpha - mc / alpha
        w = np.exp(shifted)
        w /= np.sum(w)
        grad = np.zeros(n)
        for k in range(2 * n - 1):
