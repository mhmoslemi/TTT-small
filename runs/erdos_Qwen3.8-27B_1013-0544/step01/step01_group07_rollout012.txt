   def compute_c5(h, dx):
       corr = np.correlate(h, 1 - h, mode="full")
       return float(np.max(corr) * dx)

   def smooth_max(h, t, dx, mu, target_sum):
       c = np.correlate(h, 1 - h, mode="full")
       tc = t * c * dx
       tc_m = np.max(tc)
       val = (tc_m + np.log(np.sum(np.exp(tc - tc_m)))) / t
       if mu > 0:
           s = np.sum(h) - target_sum
           val += mu * s * s
       return val

   def smooth_max_grad(h, t, dx, mu, target_sum):
       n = len(h)
       c = np.correlate(h, 1 - h, mode="full")
       tc = t * c * dx
       tc_m = np.max(tc)
       w = np.exp(tc - tc_m)
       ws = w.sum()
       if ws < 1e-30:
           w = np.ones(len(c)) / len(c)
       else:
           w /= ws
       
       lags = np.arange(len(c)) - (n - 1)
       grad = np.zeros(n)
       for j in range(n):
           jm = j - lags
           jp = j + lags
           m1 = (jm >= 0) & (jm < n)
           m2 = (jp >= 0) & (jp < n)
           v = np.zeros(len(c))
           v[m1] = 1.0 - h[jm[m1]]
           v[m2] -= h[jp[m2]]
           grad[j] = w @ v
       
       if mu > 0:
           s = np.sum(h) - target_sum
           grad += 2.0 * mu * s
       
       return grad
