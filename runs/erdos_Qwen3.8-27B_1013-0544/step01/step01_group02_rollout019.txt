            h *= 1.0 + diff / s

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    shifted = alpha * (c - c.max())
    ms = shifted.max()
    if ms > 500:

    return c.max() + np.log(np.sum(np.exp(shifted))) / alpha

    c = np.correlate(h, 1.0 - h, mode='full') * dx

            if s < 1e-15:

    return float(c.max() + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_grad(x, n, alpha):

    n_full = len(c)
    lags = np.arange(n_full) - (n - 1)
    grad = np.zeros(n)
    for m in range(n_full):
        if w[m] < 1e-12:
            continue
        k = lags[m]
        ws = w[m]
        js = max(0, k)
        je = min(n, n + k)
        if js < je:
            grad[js:je] += ws * (1.0 - x[js - k:je - k])
        js2 = max(0, -k)
        je2 = min(n, n - k)
        if js2 < je2:
            grad[js2:je2] -= ws * x[js2 + k:je2 + k]
    return grad * dx
