import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-15:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 700:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def sum_con(h, n):
    return np.sum(h) - n / 2.0

def sum_con_jac(h, n):
    return np.ones(n)

def pair_descent(h, n, rng, t_end, n_grid=31):
    best = h.copy()
    best_c5 = compute_c5(best, n)
    while time.time() < t_end:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        s = best[i] + best[j]
        lo = max(0.0, s - 1.0)
        hi = min(1.0, s)
        if hi - lo < 1e-10:
            continue
        best_val = best[i]
        best_c5_local = best_c5
        grid = np.linspace(lo, hi, n_grid)
        for v in grid:
            if abs(v - best_val) < 1e-8:
                continue
            best[i] = v
            best[j] = s - v
            c5 = compute_c5(best, n)
            if c5 < best_c5_local:
                best_c5_local = c5
                best_val = v
        best[i] = best_val
        best[j] = s - best_val
        best_c5 = best_c5_local
    return best, best_c5

def sa_pairs(h, n, rng, t_end):
    best = h.copy()
    best_c5 = compute_c5(best, n)
    T = 0.002
    T_min = 1e-8
    cool_rate = 0.99995
    while time.time() < t_end:
        T = max(T_min, T * cool_rate)
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        s = best[i] + best[j]
        lo = max(0.0, s - 1.0)
        hi = min(1.0, s)
        if hi - lo < 1e-10:
            continue
        v1 = rng.uniform(lo, hi)
        old_i, old_j = best[i], best[j]
        best[i] = v1
        best[j] = s - v1
        c5_new = compute_c5(best, n)
        delta = c5_new - best_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            best_c5 = c5_new
        else:
            best[i] = old_i
            best[j] = old_j
    return best, best_c5
