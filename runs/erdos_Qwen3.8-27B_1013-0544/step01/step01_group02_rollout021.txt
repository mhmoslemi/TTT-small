But wait, is C₅ really convex in h[i] when h[j] = s - h[i]?

c_k = dx * sum_m h[m] * (1 - h[m+k-n+1])

- Terms where h[m] = h[i]: coefficient is (1-h[m+k-n+1])
- Terms where h[m+k-n+1] = h[i]: coefficient is -h[m]
- Same for h[j]

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-15:
                break
            h += (diff / total) * h

def obj_smooth(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 700:
        val = (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    else:
        val = np.log(np.sum(np.exp(shifted))) / alpha

def sum_con(h, n):
