            if ts < 1e-13: break
            h += d / ts * h

    mc = c.max()
    s = alpha * (c - mc)
    s = s - s.max()
    e = np.exp(s)
    return e / e.sum()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def get_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            r = 1.0 - h
            ts = r.sum()
            if ts < 1e-14:
                break
            h += d / ts * r
        else:
            ts = h.sum()
            if ts < 1e-14:
                break
            h += d / ts * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(c, alpha):
    mc = c.max()
    s = alpha * (c - mc)
    mx = s.max()
    if mx > 700:
        return mc + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha
    return mc + np.log(np.sum(np.exp(s))) / alpha

def grad_smooth_max_wrt_h(h, n, alpha):
    dx = 2.0 / n
    c = get_corr(h, n)
    mc = c.max()
    s = alpha * (c - mc)
    mx = s.max()
    e = np.exp(s - mx)
    w = e / e.sum()
    grad = np.zeros(n)
    for k_idx in range(len(c)):
        k = k_idx - (n - 1)
        wk = w[k_idx] * alpha * dx
        if wk < 1e-14:
            continue
        if k >= 0:
            grad[:n - k] += wk * (1.0 - h[k:])
            grad[k:] -= wk * h[:n - k]
        else:
            kk = -k
            grad[kk:] += wk * (1.0 - h[:n - kk])
            grad[:n - kk] -= wk * h[kk:]
    return grad

def obj_penalty(x, n, alpha, mu):
    h = np.clip(x, 0.0, 1.0)
    dx = 2.0 / n
    c = get_corr(h, n) * dx
    return smooth_max_val(c, alpha) + mu * (h.sum() - n / 2.0) ** 2

def obj_penalty_grad(x, n, alpha, mu):
    h = np.clip(x, 0.0, 1.0)
    dx = 2.0 / n
    c = get_corr(h, n) * dx
    val = smooth_max_val(c, alpha) + mu * (h.sum() - n / 2.0) ** 2
    g = grad_smooth_max_wrt_h(h, n, alpha)
    g += 2.0 * mu * (h.sum() - n / 2.0)
    return val, g

def projected_grad_step(h, n, alpha):
    dx = 2.0 / n
    c = get_corr(h, n)
    g = grad_smooth_max_wrt_h(h, n, alpha)
    # Project gradient to null space of sum constraint
    g = g - g.mean()
    return g

def lgbfsq_run(n, h_start, alpha, mu, maxiter, t0, budget_s):
    bounds = [(0.0, 1.0)] * n
    def fun(x):
        return obj_penalty(x, n, alpha, mu)
    def jac(x):
        _, g = obj_penalty_grad(x, n, alpha, mu)
        return g
    res = minimize(fun, h_start, jac=jac, method='L-BFGS-B',
                   bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
    return res.x

def pg_refine(h, n, alpha, lr, max_iter, t0, budget_s):
    h = h.copy()
    for it in range(max_iter):
        if budget_s - (time.time() - t0) < 3:
            break
        g = projected_grad_step(h, n, alpha)
        if np.max(np.abs(g)) < 1e-10:
            break
        h_new = h - lr * g
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new, n)
        c5_old = compute_c5(h, n)
        if c5_new < c5_old:
            h = h_new
        else:
            lr = max(lr * 0.5, 1e-4)
    return h

def two_swap_local(h, n, t0, budget_s, n_iters=5000):
    rng = np.random.RandomState(12345)
    h = h.copy()
    best = compute_c5(h, n)
    deltas = [0.001, 0.003, 0.005, 0.01, 0.02, 0.03]
    for _ in range(n_iters):
        if budget_s - (time.time() - t0) < 5:
            break
        i, j = rng.randint(n, 2)
        if i == j:
            continue
        for d in deltas:
            for s in [1.0, -1.0]:
                di = s * d
                hi = h[i] + di
                hj = h[j] - di
                if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                    h[i], h[j] = hi, hj
                    c = compute_c5(h, n)
                    if c < best:
                        best = c
                    else:
                        h[i], h[j] = hi - di, hj + di
    return h, best

def coord_descent(h, n, t0, budget_s, passes=5):
    h = h.copy()
    best = compute_c5(h, n)
    for _ in range(passes):
        if budget_s - (time.time() - t0) < 5:
            break
        improved = False
        for i in range(n):
            if budget_s - (time.time() - t0) < 3:
                break
            orig = h[i]
            target = n / 2.0 - h.sum() + orig
            candidates = [0.0, 1.0, target]
            candidates = [v for v in candidates if 0.0 <= v <= 1.0]
            candidates += list(np.linspace(0, 1, 8))
            best_v = orig
            best_c = best
            for v in candidates:
                if abs(v - orig) < 1e-4:
                    continue
                h[i] = v
                h = project_feasible(h, n)
                c = compute_c5(h, n)
                if c < best_c:
                    best_c = c
                    best_v = h[i]
            h[i] = best_v
            h = project_feasible(h, n)
            if best_c < best:
                best = best_c
                improved = True
    return h, best
