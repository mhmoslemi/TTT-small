def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))


def project_h(h, target):
    h = h.copy()
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return np.clip(h, 0.0, 1.0)
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = np.sum(mask)
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)


def coord_optimize(h, max_sweeps=50, tol=1e-7):
    """Coordinate descent for minimizing max correlation."""
    n = len(h)
    np_pts = n
    best_c5 = compute_c5(h)
    
    for sweep in range(max_sweeps):
        max_change = 0.0
        for i in range(n):
            old_val = h[i]

            # k = f - i + np - 1, for f in 0..np-1
            # k ranges from (0-i+np-1) to (np-1-i+np-1) = (np-1-i) to (2np-2-i)
            f_indices = np.arange(np_pts)
            k_forward = f_indices - i + np_pts - 1
            # Filter valid k (0 <= k < 2np-1)
            valid_f = (k_forward >= 0) & (k_forward < 2*np_pts - 1)
            coeffs[k_forward[valid_f]] += (1.0 - h[f_indices[valid_f]])
            
            # Backward: i - k + np - 1 = b, valid when 0 <= b < np
            # k = i - b + np - 1, for b in 0..np-1
            b_indices = np.arange(np_pts)
            k_backward = i - b_indices + np_pts - 1
            valid_b = (k_backward >= 0) & (k_backward < 2*np_pts - 1)

                    lo = mid
                else:
                    lo = hi = mid
                    break
            
            new_val = float(np.clip((lo + hi) / 2.0, 0.0, 1.0))
            
            # Update h and corr
            h[i] = new_val
            corr += (new_val - old_val) * coeffs
            max_change = max(max_change, abs(new_val - old_val))
        
        # Project
        h = project_h(h, n * 0.5)
        # Recompute corr after projection (since projection may change multiple entries)
        corr = np.correlate(h, 1.0 - h, mode='full')
        
        new_c5 = float(np.max(corr) * 2.0 / n)
        if new_c5 < best_c5:
            best_c5 = new_c5
        if max_change < tol:
            break
    
    return h, best_c5

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))


def project_h(h, target):
    h = h.copy()
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return np.clip(h, 0.0, 1.0)
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = np.sum(mask)
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)


def coord_optimize(h, max_sweeps=50, tol=1e-7, order='sequential'):
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h)
    corr = np.correlate(h, 1.0 - h, mode='full')

    for sweep in range(max_sweeps):
        max_change = 0.0
        if order == 'random':
            indices = np.random.permutation(n)
        else:
            indices = range(n)

        for i in indices:
            old_val = h[i]
            coeffs = np.zeros(2 * n - 1)
            f_idx = np.arange(n)
            k_fwd = f_idx - i + n - 1
            vf = (k_fwd >= 0) & (k_fwd < 2 * n - 1)
            coeffs[k_fwd[vf]] += (1.0 - h[f_idx[vf]])
            b_idx = np.arange(n)
            k_bwd = i - b_idx + n - 1
            vb = (k_bwd >= 0) & (k_bwd < 2 * n - 1)
            coeffs[k_bwd[vb]] -= h[b_idx[vb]]

            a = corr - old_val * coeffs
            lo, hi = 0.0, 1.0
            for _ in range(30):
                mid = (lo + hi) * 0.5
                vals = a + mid * coeffs
                g = np.max(vals)
                active = vals >= g - 1e-9
                sg = np.mean(coeffs[active])
                if sg > 1e-12:
                    hi = mid
                elif sg < -1e-12:
                    lo = mid
                else:
                    lo = hi = mid
                    break

            new_val = float(np.clip((lo + hi) * 0.5, 0.0, 1.0))
            h[i] = new_val
            corr += (new_val - old_val) * coeffs
            mc = abs(new_val - old_val)
            if mc > max_change:
                max_change = mc

        h = project_h(h, n * 0.5)
        corr = np.correlate(h, 1.0 - h, mode='full')
        new_c5 = float(np.max(corr) * 2.0 / n)
        if new_c5 < best_c5:
            best_c5 = new_c5
        if max_change < tol:
            break

    return h, best_c5


def smooth_obj(h, temp):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    return m + temp * np.log(np.sum(np.exp((corr - m) / temp)))


def sum_eq_fun(h):
    return np.sum(h) - len(h) * 0.5


def sum_eq_jac(h):
    return np.ones(len(h))


def optimize_smooth(h0, temp, max_iter=300, ftol=1e-12):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac}]
    res = minimize(smooth_obj, h0.copy(), args=(temp,), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': ftol})
    h_new = np.clip(res.x, 0.0, 1.0)
    h_new = project_h(h_new, n * 0.5)
    return h_new, compute_c5(h_new)


def t_objective(v):
    return v[-1]


def t_ineq(v):
    n = len(v) - 1
    h = v[:n]
    t = v[-1]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return t - corr


def t_eq_fun(v):
    n = len(v) - 1
    return np.sum(v[:n]) - n * 0.5


def t_eq_jac(v):
    n = len(v) - 1
    j = np.zeros(len(v))
    j[:n] = 1.0
    return j


def optimize_t_var(h0, max_iter=400, ftol=1e-10):
    n = len(h0)
    c5_cur = compute_c5(h0)
    v0 = np.append(h0.copy(), c5_cur + 0.01)
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.5)]
    cons = [
        {'type': 'ineq', 'fun': t_ineq},
        {'type': 'eq', 'fun': t_eq_fun, 'jac': t_eq_jac}
    ]
    res = minimize(t_objective, v0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': ftol})
    h_new = np.clip(res.x[:n], 0.0, 1.0)
    h_new = project_h(h_new, n * 0.5)
    return h_new, compute_c5(h_new)


def interp_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.arange(n_old) / n_old
    x_new = np.arange(n_new) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_h(h_new, n_new * 0.5)
