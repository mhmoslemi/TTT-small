from scipy.optimize import minimize

_g_dx = None
_g_ts = None

def project_feasible(x, target_sum):
    lo = float(x.min()) - 1.0
    hi = float(x.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = float(np.clip(x - mid, 0.0, 1.0).sum())
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def active_grad(h, tol_frac, max_active):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    tol = max(0.005, c_max * tol_frac)
    active = np.where(corr >= c_max - tol)[0]
    if len(active) < 1:
        active = np.array([int(np.argmax(corr))])
    elif len(active) > max_active:
        active = np.argsort(corr)[-max_active:]
    n_half = n - 1
    grad = np.zeros(n)
    for idx in active:
        L = int(idx) - n_half
        if L == 0:
            grad[:] += (1.0 - 2.0 * h[:])
        elif L > 0:
            grad[L:] += (1.0 - h)[:n - L]
            grad[:n - L] -= h[L:]
        else:
            L2 = -L
            grad[:n - L2] += (1.0 - h)[L2:]
            grad[L2:] -= h[:n - L2]
    grad /= float(len(active))
    return c_max * dx, grad

def pg_refine(h, dx, ts, n_iter, ss_start, ss_end, tol_frac, max_act, mu):
    n = len(h)
    bh, bc = h.copy(), compute_c5(h, dx)
    hc, mom = h.copy(), np.zeros(n)
    ss = ss_start
    decay = (ss_end / max(ss_start, 1e-15)) ** (1.0 / max(1, n_iter))
    for _ in range(n_iter):
        _, grad = active_grad(hc, tol_frac, max_act)
        mom = mu * mom - ss * grad

        c5n = compute_c5(hn, dx)
        if c5n < bc:
            bc, bh = c5n, hn.copy()
        hc = hn
        ss *= decay
    return bh, bc

            h = hp

        temp *= 0.9992
