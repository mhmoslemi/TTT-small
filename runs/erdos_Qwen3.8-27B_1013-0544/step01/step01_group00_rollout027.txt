def _lse_obj(h, temp):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    return m + temp * np.log(np.sum(np.exp(np.clip((corr - m) / temp, -500, 0))))

def _sum_eq(h):
    return np.sum(h) - len(h) * 0.5

def _sum_jac(h):
    return np.ones(len(h))

def opt_lse(h0, temp, max_iter=200, ftol=1e-12):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
    res = minimize(_lse_obj, h0.copy(), args=(temp,), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': ftol})
    hn = project_h(res.x, n * 0.5)
    return hn, compute_c5(hn)

def interp_h(h, n_new):
    x0 = np.linspace(0, 1, len(h), endpoint=False)
    x1 = np.linspace(0, 1, n_new, endpoint=False)
    return project_h(np.interp(x1, x0, h), n_new * 0.5)

def anneal_step(h, delta, rng):
    """One step of simulated annealing: modify delta fraction of entries."""
    n = len(h)
    k = max(1, int(delta * n))
    idx = rng.choice(n, size=k, replace=False)
    h_new = h.copy()
    h_new[idx] += rng.randn(k) * 0.1
    return project_h(h_new, n * 0.5)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    def tl(): return budget_s - (time.time() - t0)
    
    h = np.array(initial_h_values, dtype=float)
    n = len(h)

    c5 = compute_c5(h)

        if c5n < c5: h, c5, h_best = hn, c5n, hn.copy()
    
    # Phase 2: LSE smooth
    for temp in [0.03, 0.01, 0.005, 0.002, 0.001, 0.0005]:
        if tl() < 60: break
        hn, c5n = opt_lse(h, temp)
        if c5n < c5: h, c5, h_best = hn, c5n, hn.copy()
    
    # Phase 3: t-var again
    if tl() > 100:
        hn, c5n = opt_t(h, 600, 1e-11)
        if c5n < c5: h, c5, h_best = hn, c5n, hn.copy()
    
    # Phase 4: Simulated annealing
    T = 0.001
    for i in range(5000):
        if tl() < 60: break
        h_prop = anneal_step(h_best, 0.05, rng)
        c5_prop = compute_c5(h_prop)
        delta = c5_prop - c5
        if delta < 0 or rng.random() < np.exp(-delta / T):
            h_best, c5 = h_prop, c5_prop
        T *= 0.999
    
    # Phase 5: Refine after annealing
    if tl() > 80:
        hn, c5n = opt_t(h_best, 500, 1e-10)
        if c5n < c5: h_best, c5 = hn, c5n
    if tl() > 50:
        hn, c5n = opt_lse(h_best, 0.001)
        if c5n < c5: h_best, c5 = hn, c5n
    
    # Phase 6: Random restarts
    for i in range(15):
        if tl() < 50: break
        hp = project_h(h_best + rng.randn(n) * 0.02, n * 0.5)
        hn, c5n = opt_t(hp, 200, 1e-10)
        if c5n < c5: h_best, c5 = hn, c5n
    
    # Phase 7: Multi-scale
    for n_s in [90, 60]:
        if tl() < 120: break
        hs = interp_h(h_best, n_s)
        hs, _ = opt_t(hs, 400, 1e-10)
        for temp in [0.01, 0.003]:
            if tl() < 40: break
            hs, _ = opt_lse(hs, temp, 150, 1e-12)
        if tl() > 40: hs, _ = opt_t(hs, 400, 1e-11)
        h_up = interp_h(hs, n)

            h_best, c5 = h_up, compute_c5(h_up)
            if tl() > 50:
                hn, c5n = opt_t(h_best, 300, 1e-10)
                if c5n < c5: h_best, c5 = hn, c5n
    
    # Final
    h_best = project_h(h_best, n * 0.5)

Let me try yet another approach: **Use the fact that the problem is min-max to apply a "mirror descent" or "multiplicative weights" algorithm.**

The problem is: min_h max_k f_k(h) where f_k(h) = ∫ h(x)(1-h(x+k)) dx.

By von Neumann's minimax theorem (if the problem were convex-concave), we could solve it as a game. But f_k(h) is not convex in h (it's quadratic with negative cross terms), so minimax doesn't directly apply.

However, we can still use an iterative approach:
1. Maintain weights w_k over shifts k
2. Update h to minimize sum_k w_k * f_k(h) (a weighted sum of quadratics)
3. Update w to concentrate on large f_k(h)

Step 2: min_h sum_k w_k * f_k(h) subject to constraints. This is a quadratic program!
sum_k w_k * f_k(h) = sum_k w_k * sum_i h[i](1-h[i+k])dx
= dx * sum_i h[i] * (1 - sum_k w_k h[i+k])
= dx * sum_i h[i] - dx * sum_i h[i] * (Wh)[i]

where W is the weighted shift operator. This is a quadratic form:
= dx * h·1 - dx * h^T W h

The matrix W might not be PSD, so this quadratic program might be non-convex. But we can still try to solve it.

Actually, let me expand more carefully:
f_k(h) = dx * sum_i h[i] - dx * sum_i h[i] * h[i+m_k] (for valid i, where m_k is the shift)

So sum_k w_k * f_k(h) = dx * n * 0.5 - dx * sum_k w_k * sum_i h[i] h[i+m_k]
= constant - dx * h^T A_w h

where A_w = sum_k w_k * A_k (weighted sum of shift matrices).

If A_w is PSD, then -h^T A_w h is concave, and minimizing it is equivalent to maximizing h^T A_w h, which is a maximize-convex problem (non-convex).

If A_w is not PSD, the problem is still non-convex.

So this approach doesn't directly help with convexity. But we can still use it as a subproblem in the outer iteration.
