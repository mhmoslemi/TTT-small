            for tau in [100, 500, 2000]:
                if budget_s - (time.time() - t0) < 6:
                    break
                if n_best <= 200:
                    h_new, c5_new = slsqp_optimize(h_b, tau, maxiter=200)
                else:
                    h_new, c5_new = lbfgsb_optimize(h_b, tau, maxiter=200)
                if c5_new is not None and c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new
                h_b = h_new

    if budget_s - (time.time() - t0) > 30:
        hn = len(best_h)
        n_trials = min(int((budget_s - (time.time() - t0)) / 25), 12)
        for _ in range(n_trials):
            if budget_s - (time.time() - t0) < 10:
                break
            scale = 0.02 + 0.12 * rng.random()
            hp = best_h + rng.normal(0, scale, hn)
            hp = project_sum(hp, hn / 2.0)
            for tau in [200, 1000, 5000]:
                if budget_s - (time.time() - t0) < 5:
                    break
                if hn <= 200:
                    h_new, c5_new = slsqp_optimize(hp, tau, maxiter=200)
                else:
                    h_new, c5_new = lbfgsb_optimize(hp, tau, maxiter=200)
                if c5_new is not None and c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new
                hp = h_new

    if budget_s - (time.time() - t0) > 8:
        for tau in [10000, 50000, 200000]:
            if budget_s - (time.time() - t0) < 4:
                break
            hn = len(best_h)
            if hn <= 250:
                h_new, c5_new = slsqp_optimize(best_h, tau, maxiter=200, ftol=1e-14)
            else:
                h_new, c5_new = lbfgsb_optimize(best_h, tau, maxiter=200)
            if c5_new is not None and c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    best_h = project_sum(best_h, len(best_h) / 2.0)
    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, len(best_h)
