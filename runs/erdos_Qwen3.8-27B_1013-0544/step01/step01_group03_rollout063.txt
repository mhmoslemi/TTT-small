import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_gradient(h, tau):
    n = len(h)
    n_c = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode="full")
    x = tau * corr
    x_max = x.max()
    exp_x = np.exp(x - x_max)
    w = exp_x / exp_x.sum()
    
    grad = np.zeros(n)
    shifts = np.arange(n_c) - (n - 1)
    
    # Process in order of |shift| for cache efficiency
    # For each m, |shift| = |m - (n-1)|
    for m in range(n_c):
        wm = w[m]
        if wm < 1e-12:
            continue
        s = abs(m - (n - 1))
        if s == 0:
            grad += wm * (1.0 - 2.0 * h)
        else:
            grad[:n-s] += wm * (1.0 - h[s:])
            grad[s:] -= wm * h[:n-s]
    
    return grad / tau

def project_constraint(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(40):
        mid = (lo + hi) / 2.0
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def fista_optimize(h0, n, tau, target, max_iter=1000, lr=None):
    """FISTA for projected gradient on smooth max of correlations."""
    if lr is None:
        lr = 1.0 / (tau * 2.0)  # conservative
    h = h0.copy()
    y = h0.copy()
    t_mom = 1.0
    best = h.copy()
    best_val = evaluate_c5(h)
    
    for it in range(max_iter):
        grad = compute_gradient(y, tau)
        h_old = h.copy()
        h_new = y - lr * grad
        h_new = project_constraint(h_new, target)
        
        val = evaluate_c5(h_new)
        if val < best_val:
            best = h_new.copy()
            best_val = val
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        y = h_new + (t_mom - 1.0) / t_new * (h_new - h_old)
        h = h_new
        t_mom = t_new
    
    return best, best_val

def binary_local_search(h, n, target, max_iters=10000):
    """Local search on binary projection."""
    # Project to binary
    idx = np.argsort(h)
    h_bin = np.zeros(n)
    h_bin[idx[n - int(target):]] = 1.0
    # Ensure exactly target ones
    n_ones = int(target)
    h_bin = np.zeros(n)
    h_bin[idx[n - n_ones:]] = 1.0
    
    best_c5 = evaluate_c5(h_bin)
    best_h = h_bin.copy()
    
    # Compute current lags
    corr = np.correlate(h_bin, 1.0 - h_bin, mode="full")
    max_lag = corr.max()
    
    for it in range(max_iters):
        # Find a random swap that might help
        ones = np.where(h_bin == 1.0)[0]
        zeros = np.where(h_bin == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        
        # Try swap
        h_bin[i] = 0.0
        h_bin[j] = 1.0
        corr_new = np.correlate(h_bin, 1.0 - h_bin, mode="full")
        new_max = corr_new.max()
        
        if new_max < best_c5 / (2.0 / n) - 1e-15:  # improvement in max corr
            best_c5 = new_max * (2.0 / n)
            best_h = h_bin.copy()
            max_lag = new_max
        else:
            # Undo
            h_bin[i] = 1.0
            h_bin[j] = 0.0
    
    return best_h, best_c5

result[-1] = sum_{n=1}^{2} h[n-1]*(1-h[n]) = h[0]*(1-h[1]) + h[1]*(1-h[2]) = a(1-b) + b(1-c)

d/dh[a] = (1-b)
d/dh[b] = -a + (1-c) = 1-a-c
d/dh[c] = -b

My formula for negative lag -k:
grad[:n-k] += w*(1-h[k:]) -> grad[:2] += w*(1-h[1:]) = w*(1-[b,c]) = [w(1-b), w(1-c)]
grad[k:] -= w*h[:n-k] -> grad[1:] -= w*h[:2] = -w*[a,b]

Total: grad = [w(1-b), w(1-c)-wa, -wb] = [1-b, 1-a-c, -b] ✓

So the correct formulas are:
- Positive lag k (k>0):
  grad[k:] += w * (1 - h[:n-k])
  grad[:n-k] -= w * h[k:]

- Negative lag -k (k>0):
  grad[:n-k] += w * (1 - h[k:])
  grad[k:] -= w * h[:n-k]

- Lag 0:
  grad += w * (1 - 2*h)

Now let me map this to the numpy correlation indexing:
result[m] has lag k = m - (n-1)
- If m > n-1: positive lag k = m-(n-1) > 0
- If m < n-1: negative lag k = m-(n-1) < 0, |k| = n-1-m
- If m = n-1: lag 0

So in my loop:
for m in range(2*n-1):

    if k == 0:

    elif k > 0:

        grad[:n-k] -= w[m] * h[k:]
    else:  # k < 0
        kk = -k  # positive
        grad[:n-kk] += w[m] * (1 - h[kk:])
        grad[kk:] -= w[m] * h[:n-kk]

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

def compute_gradient(h, tau):
    n = len(h)
    n_c = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode="full")
    x = tau * corr
    x_max = x.max()
    exp_x = np.exp(x - x_max)
    w = exp_x / exp_x.sum()
    
    grad = np.zeros(n)
    # Lag 0
    grad += w[n-1] * (1.0 - 2.0 * h)
    
    # Positive lags k=1..n-1 (index m = n-1+k in corr)
    for k in range(1, n):
        wm = w[n-1+k]
        if wm < 1e-14:
            continue
        grad[k:] += wm * (1.0 - h[:n-k])
        grad[:n-k] -= wm * h[k:]
    
    # Negative lags k=1..n-1 (index m = n-1-k in corr)
    for k in range(1, n):
        wm = w[n-1-k]
        if wm < 1e-14:
            continue
        grad[:n-k] += wm * (1.0 - h[k:])
        grad[k:] -= wm * h[:n-k]

def directed_binary_search(h_init, n, max_iters=20000):
    target = n // 2
    idx = np.argsort(-h_init)
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    
    best_c5 = evaluate_c5(h)
    best_h = h.copy()

    max_val = corr.max() * (2.0/n)
    
    for it in range(max_iters):
        # Find the worst lag
        corr = np.correlate(h, 1.0 - h, mode="full")
        worst_m = np.argmax(corr)
        worst_k = worst_m - (n-1)  # the actual shift
        
        # Find positions involved in the worst overlap
        # For positive shift k: h[i]=1, h[i+k]=0
        # For negative shift -k: h[i]=1, h[i-k]=0 (equivalently h[i+k]=1, h[i]=0)
        
        if worst_k > 0:
            # h[i]=1 and h[i+k]=0 for i in [0, n-1-k]
            candidates_a = [i for i in range(n-worst_k) if h[i]==1 and h[i+worst_k]==0]
            candidates_b = [i+worst_k for i in range(n-worst_k) if h[i]==1 and h[i+worst_k]==0]
            # Also try: h[i]=0 and h[i+k]=1 (to potentially reduce other lags)
            candidates_c = [i for i in range(n-worst_k) if h[i]==0 and h[i+worst_k]==1]
            candidates_d = [i+worst_k for i in range(n-worst_k) if h[i]==0 and h[i+worst_k]==1]

            candidates_b = [i-kk for i in range(kk, n) if h[i]==1 and h[i-kk]==0]

    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def fista_optimize(h0, n, tau, target, max_iter=500):
    h = project_constraint(h0.astype(float).copy(), target)
    y = h.copy()
    t_mom = 1.0
    best = h.copy()
    best_val = evaluate_c5(h)
    lr = 1.0 / (2.5 * tau)
    for it in range(max_iter):
        grad = compute_gradient(y, tau)
        h_old = h.copy()
        h = project_constraint(y - lr * grad, target)
        val = evaluate_c5(h)
        if val < best_val:
            best = h.copy()
            best_val = val
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        y = h + (t_mom - 1.0) / t_new * (h - h_old)
        t_mom = t_new
    return best, best_val

def binary_local_search(h_init, n, max_iters=30000, temp=0.01):
    target = n // 2
    idx = np.argsort(-h_init)
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = float(corr.max()) * dx
    best_c5 = max_val
    best_h = h.copy()
    for it in range(max_iters):
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        corr_new = np.correlate(h, 1.0 - h, mode="full")
        new_max = float(corr_new.max()) * dx
        if new_max < max_val or np.random.random() < np.exp((max_val - new_max) / (temp * dx)):
            if new_max < best_c5:
                best_c5 = new_max
                best_h = h.copy()
            max_val = new_max
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c5

def directed_search(h_init, n, max_iters=20000):
    target = n // 2
    idx = np.argsort(-h_init)
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = float(corr.max()) * dx
    best_c5 = max_val
    best_h = h.copy()
    for it in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode="full")
        worst_m = int(np.argmax(corr))
        worst_k = worst_m - (n - 1)
        if worst_k == 0:
            break
        ak = abs(worst_k)
        good = []
        if worst_k > 0:
            for i in range(n - ak):
                if h[i] == 1.0 and h[i + ak] == 0.0:
                    good.append((i, i + ak))
        else:
            for i in range(ak, n):
                if h[i] == 1.0 and h[i - ak] == 0.0:
                    good.append((i, i - ak))
        if not good:
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[np.random.randint(len(ones))]
            j = zeros[np.random.randint(len(zeros))]
        else:
            pair = good[np.random.randint(len(good))]
            i, j = pair
            if np.random.random() < 0.5:
                pass
            else:
                i, j = j, i
        h[i] = 1.0 - h[i]
        h[j] = 1.0 - h[j]
        corr_new = np.correlate(h, 1.0 - h, mode="full")
        new_max = float(corr_new.max()) * dx
        if new_max < max_val:
            max_val = new_max
            if new_max < best_c5:
                best_c5 = new_max
                best_h = h.copy()
        else:
            h[i] = 1.0 - h[i]
            h[j] = 1.0 - h[j]
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(80, 0.5)
    best_h = h_init.copy()
    best_c5 = evaluate_c5(best_h)
    n0 = len(best_h)

    def rem():
        return budget_s - (time.time() - t0)

    # Phase 1: FISTA annealing on initial resolution
    n = n0
    target = n / 2.0
    for tau in [50, 150, 400, 1000, 3000, 8000]:
        if rem() < 20:
            break
        h_new, c5_new = fista_optimize(best_h, n, tau, target, max_iter=600)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 2: Multi-resolution exploration
    res_list = [60, 80, 100, 120, 140, 160, 200]
    for tn in res_list:
        if rem() < 25:
            break
        target_t = tn / 2.0
        h_t = np.interp(np.linspace(0, 1, tn), np.linspace(0, 1, len(best_h)), best_h)
        h_t = project_constraint(h_t, target_t)
        for tau in [150, 800, 3000]:
            if rem() < 12:
                break
            h_new, c5_new = fista_optimize(h_t, tn, tau, target_t, max_iter=400)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
            h_t = h_new

    # Phase 3: Binary + directed search on best resolution
    n_cur = len(best_h)
    if rem() > 30:
        h_bin, c5_b = binary_local_search(best_h, n_cur, max_iters=20000, temp=0.02)
        if c5_b < best_c5:
            best_h = h_bin
            best_c5 = c5_b
    if rem() > 25:
        h_dir, c5_d = directed_search(best_h, n_cur, max_iters=15000)
        if c5_d < best_c5:
            best_h = h_dir
            best_c5 = c5_d

    # Phase 4: FISTA refinement after combinatorial search
    if rem() > 20:
        for tau in [500, 2000, 8000]:
            if rem() < 12:
                break
            h_new, c5_new = fista_optimize(best_h, n_cur, tau, n_cur / 2.0, max_iter=500)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    # Phase 5: More binary searches with perturbation
    for rs in range(6):
        if rem() < 25:
            break
        scale = 0.05 + 0.1 * np.random.random()
        hp = np.clip(best_h + np.random.normal(0, scale, n_cur), 0, 1)
        hp = project_constraint(hp, n_cur / 2.0)
        h_bin, c5_b = binary_local_search(hp, n_cur, max_iters=15000, temp=0.015)
        if c5_b < best_c5:
            best_h = h_bin
            best_c5 = c5_b
        if rem() < 15:
            break
        h_dir, c5_d = directed_search(best_h, n_cur, max_iters=10000)
        if c5_d < best_c5:
            best_h = h_dir
            best_c5 = c5_d

    # Phase 6: Final FISTA fine-tuning
    if rem() > 15:
        for tau in [5000, 15000]:
            if rem() < 10:
                break
            h_new, c5_new = fista_optimize(best_h, n_cur, tau, n_cur / 2.0, max_iter=400)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    # Phase 7: Try a higher resolution interpolation of best
    if rem() > 20:
        for trn in [120, 160, 200]:
            if rem() < 15:
                break
            if trn == n_cur:
                continue
            target_t = trn / 2.0
            h_t = np.interp(np.linspace(0, 1, trn), np.linspace(0, 1, n_cur), best_h)
            h_t = project_constraint(h_t, target_t)
            for tau in [1000, 5000]:
                if rem() < 10:
                    break
                h_new, c5_new = fista_optimize(h_t, trn, tau, target_t, max_iter=300)
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new
                h_t = h_new

    # Final constraint fix
    hn = len(best_h)
    best_h = project_constraint(best_h, hn / 2.0)
    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, hn
