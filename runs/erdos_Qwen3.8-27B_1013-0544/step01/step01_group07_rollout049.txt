import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target_sum):
    """Fix sum by distributing correction, with clipping."""
    n = len(h)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_coeffs(h, i, N):
    """Compute D_m and E_m for the 1D subproblem at coordinate i."""
    # D_m = A_m - B_m where:
    # A_m = (1-h)[i - m + N - 1] if valid
    # B_m = h[i + m - (N-1)] if valid
    # E_m = C_m - D_m * old where C_m is current corr[m]
    
    corr = np.correlate(h, 1 - h, mode="full")
    old = h[i]
    
    # For each m in 0..2N-2:
    # A_m: j = i - m + N - 1, valid if 0 <= j < N
    # B_m: j = i + m - (N-1) = i + m - N + 1, valid if 0 <= j < N
    #      (this is the index into h for the first factor: h[j + m - (N-1)] = h[i] means j = i - m + N - 1)
    #      Wait, I need to re-derive.
    
    # output[m] = Σ_j h[j + m - (N-1)] * (1-h[j])
    # h[i] appears in first factor when j + m - (N-1) = i → j = i - m + N - 1
    # h[i] appears in second factor when j = i (contributing h[i + m - (N-1)] * (1-h[i]))
    
    # Let's index: m goes from 0 to 2N-2
    ms = np.arange(2 * N - 1)
    
    # A: index into (1-h) is i - m + N - 1
    idx_a = i - ms + N - 1
    valid_a = (idx_a >= 0) & (idx_a < N)
    A = np.zeros(2 * N - 1)
    A[valid_a] = 1.0 - h[idx_a[valid_a]]
    
    # B: h[i + m - (N-1)] = h[i + m - N + 1]
    idx_b = i + ms - N + 1
    valid_b = (idx_b >= 0) & (idx_b < N)
    B = np.zeros(2 * N - 1)
    B[valid_b] = h[idx_b[valid_b]]
    
    D = A - B
    E = corr - D * old
    
    return D, E

def solve_1d(D, E, t_min=0.0, t_max=1.0, n_iter=50):
    """Solve min_t max_m (E_m + D_m * t) for t in [t_min, t_max] using ternary search."""
    a, b = t_min, t_max
    for _ in range(n_iter):
        t1 = a + (b - a) / 3
        t2 = b - (b - a) / 3
        f1 = np.max(E + D * t1)
        f2 = np.max(E + D * t2)
        if f1 < f2:
            b = t2
        else:
            a = t1
    return (a + b) / 2

def coord_descent(h, dx, target_sum, n_sweeps=500, n_iter_1d=40, shuffle=True):

    best_c5 = compute_c5(h, dx)
    rng = np.random.RandomState(0)

            order = rng.permutation(N)
        else:
            order = range(N)
        
        for i in order:
            D, E = corr_coeffs(h, i, N)
            t_opt = solve_1d(D, E, 0.0, 1.0, n_iter=n_iter_1d)
            h[i] = t_opt

        c5 = compute_c5(h, dx)
        if c5 < best_c5:

SLSQP uses gradient information and can take large steps. Coordinate descent only moves one coordinate at a time, which can be slow in high dimensions. However, for this problem:
- The objective is the max of linear (almost) functions
- The constraint is a single equality (sum) + bounds
- The problem is "separable" in the sense that each coordinate appears in only a few correlations

Actually, each coordinate h[i] appears in exactly 2N-1 correlations (all of them, since changing h[i] affects every lag). So the coordinates are NOT separable. This makes coordinate descent potentially slow.

But in practice, the "effective" dimension is much smaller because only a few correlations are near the maximum. The coordinate descent should still work, just might need more sweeps.

Let me also add **block updates** for efficiency: update pairs of coordinates (i, i+1) or (i, i+N/2) simultaneously. This helps with the sum constraint and allows larger effective steps.

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target_sum, max_iter=500):
    n = len(h)
    for _ in range(max_iter):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def solve_1d(h, i, N, n_grid):
    corr = np.correlate(h, 1.0 - h, mode="full")
    old = h[i]
    ms = np.arange(2 * N - 1)
    idx_a = i - ms + N - 1
    valid_a = (idx_a >= 0) & (idx_a < N)
    A = np.zeros(2 * N - 1)
    A[valid_a] = 1.0 - h[idx_a[valid_a]]
    idx_b = i + ms - N + 1
    valid_b = (idx_b >= 0) & (idx_b < N)
    B = np.zeros(2 * N - 1)
    B[valid_b] = h[idx_b[valid_b]]
    D = A - B
    E = corr - D * old
    Z = corr[N - 1] - old + old * old
    ts = np.linspace(0.0, 1.0, n_grid)
    vals = E + D[:, None] * ts[None, :]
    vals[N - 1, :] = Z + ts - ts * ts
    max_vals = np.max(vals, axis=0)
    return ts[np.argmin(max_vals)]

def coord_descent(h, dx, target_sum, n_sweeps, n_grid, seed_val):
    N = len(h)
    rng = np.random.RandomState(seed_val)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    for sweep in range(n_sweeps):
        order = rng.permutation(N)
        for i in order:
            h[i] = solve_1d(h, i, N, n_grid)
        h = fix_sum(h, target_sum)
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if best_c5 < target_sum * dx * 0.38:
            break
    return best_h, best_c5

def upsample_to(h, new_N):
    old_N = len(h)
    if new_N == old_N:
        return h.copy()
    x_old = (np.arange(old_N) + 0.5) / old_N
    x_new = (np.arange(new_N) + 0.5) / new_N
    return np.interp(x_new, x_old, h)

def block_descent(h, dx, target_sum, n_sweeps, block_size, n_grid, seed_val):
    N = len(h)
    rng = np.random.RandomState(seed_val)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    for sweep in range(n_sweeps):
        order = rng.permutation(N)
        for start in range(0, N, block_size):
            idxs = [order[(start + k) % N] for k in range(min(block_size, N - start))]
            sub_sum = np.sum(h[idxs])
            ts_grid = np.linspace(0.0, 1.0, n_grid)
            best_local = np.inf
            best_alloc = h[idxs].copy()
            nb = len(idxs)
            if nb == 1:
                h[idxs[0]] = solve_1d(h, idxs[0], N, n_grid)
                continue
            for t0 in ts_grid[::max(1, n_grid // 40)]:
                rem = sub_sum - t0
                if rem < -0.01 or rem > (nb - 1) * 1.01:
                    continue
                alloc = np.full(nb, rem / (nb - 1))
                alloc[0] = t0
                if np.any(alloc < -0.01) or np.any(alloc > 1.01):
                    continue
                alloc = np.clip(alloc, 0, 1)
                if abs(alloc.sum() - sub_sum) > 0.01:
                    continue
                test_h = h.copy()
                test_h[idxs] = alloc
                c5_t = compute_c5(test_h, dx)
                if c5_t < best_local:
                    best_local = c5_t
                    best_alloc = alloc.copy()
            h[idxs] = best_alloc
        h = fix_sum(h, target_sum)
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def random_binary(N, seed_val):
    rng = np.random.RandomState(seed_val)
    h = np.zeros(N)
    ones = rng.choice(N, N // 2, replace=False)
    h[ones] = 1.0
    return h

    t0 = time.time()

    best_c5 = compute_c5(best_h, 2.0 / n0)
    best_N = n0

    resolutions = [80, 120, 160, 200, 240, 320]
    # Sort by priority: start with parent size, then neighbors
    resolutions.sort(key=lambda x: (abs(x - n0), x))

    for N in resolutions:
        if remaining() < 40:
            break
        if N == n0:
            h = best_h.copy()
        elif N < n0:
            h = upsample_to(h0, N)
        else:

        dx = 2.0 / N
        target_sum = N / 2.0
        h = fix_sum(h, target_sum)

        # Phase 1: coarse coordinate descent
        n_sweeps1 = min(300, max(30, int(remaining() * 0.4 / (2 * N))))
        h, c5 = coord_descent(h, dx, target_sum, n_sweeps1, 100, seed + N)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_N = N

        if remaining() < 30:

        # Phase 2: fine coordinate descent
        n_sweeps2 = min(500, max(50, int(remaining() * 0.3 / (2.5 * N))))
        h, c5 = coord_descent(h, dx, target_sum, n_sweeps2, 300, seed + N + 5000)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_N = N

        # Phase 3: block descent
        if remaining() > 60:
            h, c5 = block_descent(h, dx, target_sum,
                                  min(100, int(remaining() * 0.2 / N)),
                                  3, 50, seed + N + 9000)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                best_N = N

    # Try random restarts at best_N
    if remaining() > 50:
        N = best_N
        dx = 2.0 / N
        target_sum = N / 2.0
        for trial in range(20):
            if remaining() < 30:

    best_h = fix_sum(best_h, best_N / 2.0)
    best_c5 = compute_c5(best_h, 2.0 / best_N)

    return best_h, best_c5, best_N
