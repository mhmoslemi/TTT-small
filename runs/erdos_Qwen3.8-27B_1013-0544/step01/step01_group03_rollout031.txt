def fix_sum(h, target):
    h = np.array(h, dtype=float)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return np.clip(h, 0.0, 1.0)
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = mask.sum()
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def soft_max(corr, tau):
    x = tau * corr
    x_max = np.max(x)
    return float(x_max / tau + np.log(np.sum(np.exp(x - x_max))) / tau)

def fix_sum(h, target):
    h = np.array(h, dtype=float).copy()
    diff = target - h.sum()
    if abs(diff) < 1e-15:
        return np.clip(h, 0.0, 1.0)
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = mask.sum()
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def obj_func(x, tau):
    corr = np.correlate(x, 1.0 - x, mode="full")
    return soft_max(corr, tau)

def obj_func_penalty(x, tau, penalty):
    corr = np.correlate(x, 1.0 - x, mode="full")
    sm = soft_max(corr, tau)
    s = np.sum(x) - len(x) / 2.0
    return sm + penalty * s * s

def constraint_func(x, target):
    return np.sum(x) - target

def constraint_jac(x, target):
    return np.ones(len(x))

def thue_morse(n):
    t = np.array([0])
    while len(t) < n:
        t = np.concatenate([t, 1.0 - t])
    return t[:n].copy()

def balanced_random(n, rng):
    half = n // 2
    idx = rng.permutation(n)[:half]
    seq = np.zeros(n)
    seq[idx] = 1.0
    return seq

def optimize_slqp(h_start, n_pts, tau, maxiter):
    target = n_pts / 2.0
    bnds = Bounds(np.zeros(n_pts), np.ones(n_pts))
    cons = [{'type': 'eq', 'fun': constraint_func, 'jac': constraint_jac, 'args': (target,)}]
    try:
        res = minimize(obj_func, h_start, args=(tau,), method='SLSQP',
                      bounds=bnds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-13})
        x = fix_sum(res.x, target)
        return x, evaluate_c5(x)
    except Exception:
        return h_start, None

def optimize_lbfgs_h(h_start, n_pts, tau, penalty, maxiter):
    target = n_pts / 2.0
    bnds = Bounds(np.zeros(n_pts), np.ones(n_pts))
    try:
        res = minimize(obj_func_penalty, h_start, args=(tau, penalty), method='L-BFGS-B',
                      bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-15})
        x = fix_sum(res.x, target)
        return x, evaluate_c5(x)
    except Exception:
        return h_start, None

def coordinate_descent(h_init, target, n_sweeps, time_limit, t0):
    h = h_init.copy()
    n = len(h)
    best_c5 = evaluate_c5(h)
    for sweep in range(n_sweeps):
        if time.time() - t0 > time_limit:
            break
        improved = False
        for i in range(n):
            if time.time() - t0 > time_limit:
                break
            old = h[i]
            curr_c5 = evaluate_c5(h)
            best_val = old
            for nv in [0.0, 1.0, old - 0.15, old - 0.07, old + 0.07, old + 0.15]:
                nv = max(0.0, min(1.0, nv))
                if abs(nv - old) < 1e-9:
                    continue
                h[i] = nv
                h = fix_sum(h, target)
                c = evaluate_c5(h)
                if c < curr_c5 - 1e-13:
                    curr_c5 = c
                    best_val = nv
                h[i] = old
            if best_val != old:
                h[i] = best_val
                h = fix_sum(h, target)
                improved = True
        final_c5 = evaluate_c5(h)
        if final_c5 < best_c5:
            best_c5 = final_c5
        if not improved:
            break
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):

        if c5_new is not None and c5_new < best_c5:

            best_c5 = c5_new

    # Phase 2: Multi-resolution exploration
    n_list = [100, 140, 160, 200, 240]
    for tn in n_list:
        if time.time() - t0 > budget_s - 200:

        h_t = np.interp(np.linspace(0, 1, tn), np.linspace(0, 1, len(best_h)), best_h)
        h_t = fix_sum(h_t, tn / 2.0)
        for tau in [50, 200, 1000, 5000]:
            if time.time() - t0 > budget_s - 50:

            h_new, c5_new = optimize_slqp(h_t, tn, tau, 250)
            if c5_new is not None and c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
            h_t = h_new

    # Phase 3: Structured initializations
    for n_s in [100, 120, 160]:
        if time.time() - t0 > budget_s - 150:

        seeds_list = [thue_morse(n_s), balanced_random(n_s, rng),

        for h_s in seeds_list:
            h_s = fix_sum(h_s, n_s / 2.0)
            for tau in [100, 500, 3000]:
                if time.time() - t0 > budget_s - 30:

                h_new, c5_new = optimize_slqp(h_s, n_s, tau, 200)
                if c5_new is not None and c5_new < best_c5:

                    best_c5 = c5_new
                h_s = h_new

    # Phase 4: Random restarts (main exploration)
    hn = len(best_h)
    n_restarts = 80

        if time.time() - t0 > budget_s - 80:
            break

        hp = best_h + rng.normal(0, scale, hn)

            best_c5 = c5_new
        # Follow up with SLSQP if improved

            h_new2, c5_new2 = optimize_slqp(h_new if c5_new is not None else hp,
                                             hn, 2000, 150)
            if c5_new2 is not None and c5_new2 < best_c5:
                best_h = h_new2
                best_c5 = c5_new2
        hp = best_h if c5_new is not None and c5_new < best_c5 + 0.001 else hp

    # Phase 5: Ultra-fine SLSQP tuning
    if time.time() - t0 < budget_s - 40:
        hn_f = len(best_h)
        for tau in [5000, 20000, 50000, 100000]:
            if time.time() - t0 > budget_s - 15:
