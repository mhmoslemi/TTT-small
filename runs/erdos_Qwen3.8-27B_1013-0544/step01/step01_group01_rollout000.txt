            res = minimize(wg, h0, method='SLSQP', bounds=bounds, jac=True,
                           constraints=cons, options={'maxiter': 600, 'ftol': 1e-12})
            h_new = project_h(res.x, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new.copy(), c5
                no_improve = 0
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

        # Every few attempts, try L-BFGS-B with penalty
        if attempt % 4 == 0 and time_left() > 80:
            try:
                lam = 50.0
                def pen_wrapper(x):
                    corr = compute_corr(x)
                    obj = smooth_max_val(corr, 100.0) * dx
                    s = np.sum(x)
                    penalty = lam * (s - target_sum) ** 2
                    g = compute_gradient(x, 100.0) * dx + 2 * lam * (s - target_sum) * np.ones(n_points)
                    return obj + penalty, g
                res2 = minimize(pen_wrapper, h0, method='L-BFGS-B', bounds=bounds,
                                jac=True, options={'maxiter': 400, 'ftol': 1e-12})
                h_new2 = project_h(res2.x, target_sum)
                c5_2 = compute_c5(h_new2)
                if c5_2 < best_c5:
                    best_h, best_c5 = h_new2.copy(), c5_2
                    no_improve = 0
            except Exception:
                pass

    # Phase 3: Final polish with very high beta
    if time_left() > 20:
        for beta in [300.0, 500.0, 800.0]:
            if time_left() < 15:
                break
            wg = make_obj_grad(beta, dx)
            try:
                res = minimize(wg, best_h, method='SLSQP', bounds=bounds, jac=True,
                               constraints=cons, options={'maxiter': 3000, 'ftol': 1e-15})
                h_new = project_h(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h, best_c5 = h_new.copy(), c5
            except Exception:
                pass

    # Phase 4: Try nearby n values via interpolation
    if time_left() > 60:
        for n_try in [64, 76, 84]:
            if time_left() < 30:
                break
            try:
                ts_try = n_try / 2.0
                _target_sum[n_try] = ts_try
                dx_try = 2.0 / n_try
                x_old = np.linspace(0, 2, n_points)
                x_new = np.linspace(0, 2, n_try)
                h_interp = np.interp(x_new, x_old, best_h)
                h_interp = project_h(h_interp, ts_try)
                bounds_try = Bounds(np.zeros(n_try), np.ones(n_try))
                cons_try = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
                wg_try = make_obj_grad(120.0, dx_try)
                res = minimize(wg_try, h_interp, method='SLSQP', bounds=bounds_try,
                               jac=True, constraints=cons_try,
                               options={'maxiter': 800, 'ftol': 1e-13})
                h_new = project_h(res.x, ts_try)
                c5_try = compute_c5(h_new)
                if c5_try < best_c5:
                    best_h, best_c5 = h_new.copy(), c5_try
                    n_points = n_try
            except Exception:
                pass

    return best_h, best_c5, n_points
