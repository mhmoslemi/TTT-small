def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def softmax_max(obj, tau):
    x = obj / tau
    x_max = np.max(x)
    return float(tau * (x_max + np.log(np.sum(np.exp(x - x_max)))))

def fix_sum(h, target):
    h = np.array(h, dtype=float)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return np.clip(h, 0.0, 1.0)
    if diff > 0:
        mask = h < 1.0 - 1e-6
        if mask.sum() == 0:
            return np.clip(h, 0.0, 1.0)
        h[mask] += diff / mask.sum()
    else:
        mask = h > 1e-6
        if mask.sum() == 0:
            return np.clip(h, 0.0, 1.0)
        h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def penalty_obj(h, target, tau, penalty):
    corr = np.correlate(h, 1.0 - h, mode="full")
    base = softmax_max(corr, tau)
    sum_diff = h.sum() - target
    return base + penalty * sum_diff * sum_diff

    c5 = evaluate_c5(h_new)
    return h_new, c5

        return np.clip(h, 0.0, 1.0)
    if diff > 0:
        mask = h < 1.0 - 1e-8
    else:
        mask = h > 1e-8
    cnt = mask.sum()
    if cnt == 0:
        return np.clip(h, 0.0, 1.0)
    h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def optimize_step(h_start, target, tau, penalty, maxiter=500):
    n = len(h_start)
    bnds = [(0.0, 1.0)] * n
    args = (tau, penalty, target)
    try:
        res = minimize(
            softmax_max_obj, h_start, args=args,
            jac=lambda x, *a: softmax_max_grad(x, *a),
            method='L-BFGS-B', bounds=bnds,
            options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-8, 'maxls': 40}
        )
        h_new = fix_sum(res.x, target)
        c5 = evaluate_c5(h_new)
        return h_new, c5
    except Exception:
        return h_start, evaluate_c5(h_start)

def interp_h(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    return fix_sum(np.interp(x_new, x_old, h), n_new / 2.0)

def make_seeds(n, seed):
    seeds = [np.full(n, 0.5)]
    # Block patterns
    for nb in [2, 3, 4, 5]:
        h = np.zeros(n)
        bs = n // (2 * nb)
        for i in range(nb):
            s = i * 2 * bs
            if s + bs <= n:
                h[s:s+bs] = 1.0
        seeds.append(fix_sum(h, n/2))
    # Sinusoidal
    x = np.linspace(0, 2, n)
    for f in [1, 2, 3]:
        h = 0.5 + 0.35 * np.sin(np.pi * f * x)
        seeds.append(fix_sum(np.clip(h, 0, 1), n/2))
    # Random
    rng = np.random.RandomState(seed)
    for _ in range(3):
        h = rng.uniform(0, 1, n)
        seeds.append(fix_sum(h, n/2))
    return seeds

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    def rem():
        return budget_s - (time.time() - t0)
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(60, 0.5)
    
    best_c5 = evaluate_c5(best_h)
    
    # Phase 1: Low-res multi-seed
    n1 = 40
    seeds_1 = make_seeds(n1, seed)
    # Also include parent interpolation
    seeds_1.append(interp_h(best_h, n1))
    
    for h_seed in seeds_1:
        if rem() < 40:
            break
        target = n1 / 2.0
        h_cur = h_seed.copy()
        for tau in [3000, 1000, 500, 200, 100, 50, 20, 10]:
            if rem() < 15:
                break
            h_new, c5_new = optimize_step(h_cur, target, tau, 50.0, maxiter=400)
            h_cur = h_new
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
    
    # Phase 2: Medium resolution
    n2 = 72
    h_med = interp_h(best_h, n2)
    target = n2 / 2.0
    for tau in [3000, 1000, 400, 200, 100, 50, 25, 10, 5]:
        if rem() < 25:
            break
        h_new, c5_new = optimize_step(h_med, target, tau, 100.0, maxiter=400)
        h_med = h_new
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
    
    # Phase 3: Try several target resolutions
    for n3 in [56, 80, 96, 120]:
        if rem() < 50:
            break
        h_3 = interp_h(best_h, n3)
        target = n3 / 2.0
        for tau in [2000, 800, 300, 150, 80, 40, 20]:
            if rem() < 20:
                break
            h_new, c5_new = optimize_step(h_3, target, tau, 100.0, maxiter=350)
            h_3 = h_new
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
    
    # Phase 4: Random restarts at best resolution
    hn = len(best_h)
    target = hn / 2.0
    rng = np.random.RandomState(seed + 777)
    for i in range(80):
        if rem() < 20:
            break
        scale = 0.05 + 0.25 * rng.random()
        hp = best_h + rng.normal(0, scale, hn)
        hp = np.clip(hp, 0, 1)
        hp = fix_sum(hp, target)
        for tau in [500, 150, 50]:
            if rem() < 12:
                break
            h_new, c5_new = optimize_step(hp, target, tau, 100.0, maxiter=250)
            hp = h_new
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
    
    # Phase 5: Very fine-tuning at current resolution
    if rem() > 30:
        hn = len(best_h)
        target = hn / 2.0
        for tau in [20, 10, 5, 3, 2, 1]:
            if rem() < 12:
                break
            h_new, c5_new = optimize_step(best_h, target, tau, 500.0, maxiter=500)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
    
    # Phase 6: Try one more resolution above
    if rem() > 40:
        n_hi = len(best_h) + 20
        if n_hi % 2 == 0 and n_hi <= 200:
            h_hi = interp_h(best_h, n_hi)
            target = n_hi / 2.0
            for tau in [1000, 400, 150, 60, 25]:
                if rem() < 15:
                    break
                h_new, c5_new = optimize_step(h_hi, target, tau, 100.0, maxiter=300)
                h_hi = h_new
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new
    
    # Final
    hn_final = len(best_h)
    best_h = fix_sum(best_h, hn_final / 2.0)
    best_c5 = evaluate_c5(best_h)
    
    # Verify with pre-imported evaluator if available
    if 'evaluate_erdos_solution' in globals() and evaluate_erdos_solution is not None:
        try:
            val = evaluate_erdos_solution(best_h)
            if isinstance(val, (int, float)) and val < best_c5:
                best_c5 = val
        except Exception:
            pass
    
    return best_h, best_c5, hn_final
