import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not mask.any():
            break
        step = diff / mask.sum()
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    return h

def smooth_max_obj(h, beta):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(beta*(corr-m))))/beta)  # *dx is constant factor

def gradient_smooth_max(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    w = np.exp(beta * (corr - m))
    w_sum = w.sum()
    w = w / w_sum
    
    grad = np.zeros(n)
    n_corr = len(corr)
    for idx in range(n_corr):
        k = idx - (n - 1)
        wi = w[idx] / beta
        if abs(wi) < 1e-10:
            continue
        if k >= 0:
            # grad[j] += wi*(1-h[j-k]) for j=k..n-1
            # grad[j] -= wi*h[j+k] for j=0..n-1-k
            grad[k:] += wi * (1.0 - h[:n-k])
            grad[:n-k] -= wi * h[k:]
        else:
            kp = -k
            # grad[j] += wi*(1-h[j+kp]) for j=0..n-1-kp
            # grad[j] -= wi*h[j-kp] for j=kp..n-1
            grad[:n-kp] += wi * (1.0 - h[kp:])
            grad[kp:] -= wi * h[:n-kp]
    
    return grad, np.max(corr) + np.log(w_sum)/beta

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not mask.any():
            break
        step = diff / mask.sum()
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    return h


def grad_smooth_max(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exp_vals = np.exp(beta * (corr - m))
    w_sum = exp_vals.sum()
    w = exp_vals / w_sum
    grad = np.zeros(n)
    for idx in range(len(corr)):
        k = idx - (n - 1)
        wi = w[idx] / beta
        if wi < 1e-11:
            continue
        if k >= 0:
            grad[k:] += wi * (1.0 - h[:n - k])
            grad[:n - k] -= wi * h[k:]
        else:
            kp = -k
            grad[:n - kp] += wi * (1.0 - h[kp:])
            grad[kp:] -= wi * h[:n - kp]
    return grad, m + np.log(w_sum) / beta


def optimize_projected(h, target_sum, time_limit, betas=None):
    if betas is None:
        betas = [30, 60, 100, 160, 250, 400]
    start = time.time()
    best_local = h.copy()
    best_c5_local = compute_c5(h)
    for beta in betas:
        if time.time() - start > time_limit:
            break
        step = 2.0 / beta
        for it in range(300):
            if time.time() - start > time_limit:
                break
            grad, smax = grad_smooth_max(h, beta)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            good = False
            for _ in range(25):
                h_try = project_feasible(h - step * grad, target_sum)
                c5_try = compute_c5(h_try)
                if c5_try <= best_c5_local:
                    h = h_try
                    if c5_try < best_c5_local:
                        best_c5_local = c5_try
                        best_local = h_try.copy()
                    good = True
                    break
                step *= 0.6
            if not good:
                step *= 0.5
                if step < 1e-10:
                    break
            else:
                step = min(step * 1.3, 4.0 / beta)
    return best_local, best_c5_local


def optimize_slsqp(h, target_sum, time_limit, beta=150):
    n = len(h)
    start = time.time()

    def obj(x):
        return compute_corr_smooth_max(x, beta) * (2.0 / n)

    def compute_corr_smooth_max(x, b):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = np.max(corr)
        return m + np.log(np.sum(np.exp(b * (corr - m)))) / b

    # Can't define nested, use module-level trick
    from scipy.optimize import minimize as minf
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(len(x))}]
    bnds = [(0, 1)] * n

    def obj_fn(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = np.max(corr)
        return (m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta) * (2.0 / n)

    try:
        res = minf(obj_fn, h, method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': 500, 'ftol': 1e-13})
        if time.time() - start < time_limit:
            h_new = project_feasible(res.x, target_sum)
            c5_new = compute_c5(h_new)
            return h_new, c5_new
    except Exception:
        pass
    return h, compute_c5(h)


def make_binary_init(n, rng):
    positions = rng.choice(n, n // 2, replace=False)
    h = np.zeros(n)
    h[positions] = 1.0
    return h


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()

    def time_left():
        return budget_s - (time.time() - start_time)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)

    n_init = len(h_init)
    target_init = n_init / 2.0
    h_init = project_feasible(h_init, target_init)

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    best_n = n_init

    # Track best per-n solutions
    solutions = {n_init: (best_h.copy(), best_c5)}

    # Phase 1: Optimize at initial n with projected gradient
    if time_left() > 80:
        t_alloc = min(120, time_left() * 0.25)
        h_opt, c5_opt = optimize_projected(h_init, target_init, t_alloc)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = n_init
        solutions[n_init] = (h_opt, c5_opt)

    # Phase 2: Try larger n values
    for n in [140, 200, 280]:
        if time_left() < 60:
            break
        target_sum = n / 2.0

        # Multiple initializations
        inits = []

        # Upsampled from best
        h_up = np.interp(np.linspace(0, 1, n, endpoint=False),
                         np.linspace(0, 1, n_init, endpoint=False), best_h)
        inits.append(project_feasible(h_up, target_sum))

        # Binary random
        h_bin = make_binary_init(n, rng)
        inits.append(h_bin)

        # Smooth binary (slightly perturbed)
        h_sb = h_bin + rng.randn(n) * 0.05
        inits.append(project_feasible(h_sb, target_sum))

        # Structured: blocks
        h_struct = np.zeros(n)
        block = max(2, n // 20)
        for i in range(0, n, 2 * block):
            h_struct[i:i + block] = 1.0
        inits.append(project_feasible(h_struct, target_sum))

        for h0 in inits:
            if time_left() < 40:
                break
            t_alloc = min(60, time_left() * 0.3)
            h_opt, c5_opt = optimize_projected(h0, target_sum, t_alloc)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                best_n = n
            if n not in solutions or c5_opt < solutions[n][1]:
                solutions[n] = (h_opt, c5_opt)

    # Phase 3: SLSQP polish on best solution
    if time_left() > 30:
        n = best_n
        target_sum = n / 2.0
        h_pol, c5_pol = optimize_slsqp(best_h, target_sum, min(60, time_left() * 0.5))
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol

    # Phase 4: Cross-n refinement - try SLSQP on other n values
    for n, (h_sol, c5_sol) in solutions.items():
        if time_left() < 20:
            break
        if n == best_n:
            continue
        target_sum = n / 2.0
        h_pol, c5_pol = optimize_slsqp(h_sol, target_sum, min(30, time_left() * 0.4))
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol
            best_n = n

    # Phase 5: Simulated annealing polish
    n = best_n
    target_sum = n / 2.0
    h = best_h.copy()
    temperature = 0.002

    while time_left() > 3:
        n_pert = rng.randint(1, 5)
        idx = rng.choice(n, n_pert, replace=False)
        h[idx] += rng.randn(n_pert) * 0.04
        h = project_feasible(h, target_sum)
        new_c5 = compute_c5(h)
        if new_c5 < best_c5 or (new_c5 - best_c5 < temperature and rng.random() < 0.3):
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h = best_h.copy()
        temperature *= 0.9995

    # Final: enforce exact feasibility
    best_h = project_feasible(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
