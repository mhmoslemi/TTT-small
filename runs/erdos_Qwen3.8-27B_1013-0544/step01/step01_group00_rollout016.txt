import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))


def project_h(h, target):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    mask = (h < 1.0) if diff > 0 else (h > 0.0)
    cnt = np.sum(mask)
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)


def smooth_obj_and_grad(h, temp, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    corr_t = corr / temp
    max_ct = np.max(corr_t)
    exp_ct = np.exp(corr_t - max_ct)
    p = exp_ct / np.sum(exp_ct)
    f = temp * (max_ct + np.log(np.sum(exp_ct)))
    target_sum = n * 0.5
    sum_err = np.sum(h) - target_sum
    f += penalty * sum_err * sum_err
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        pk = p[k_idx]
        if pk < 1e-14:
            continue
        k = k_idx - (n - 1)
        i_lo = max(0, -k)
        i_hi = min(n, n - k)
        if i_hi > i_lo:
            grad[i_lo:i_hi] += pk * dx * (1.0 - h[i_lo + k:i_hi + k])
        j_lo = max(0, k)
        j_hi = min(n, n + k)
        if j_hi > j_lo:
            grad[j_lo:j_hi] -= pk * dx * h[j_lo - k:j_hi - k]
    grad += 2.0 * penalty * sum_err
    return f, grad

            pass  # keep current h, try next temp

    # Phase B: SLSQP t-variable refinement
    if elapsed() < budget_s - 120:
        h_new, c5_new = optimize_t_var(h_best, max_iter=500, ftol=1e-11)
        if c5_new < c5:
            h_best = h_new
            c5 = c5_new

    # Phase C: Coordinate descent
    if elapsed() < budget_s - 80:
        h_cd = coord_descent_sweep(h_best, n_sweeps=3)
        c5_cd = compute_c5(h_cd)
        if c5_cd < c5:
            h_best = h_cd
            c5 = c5_cd

    # Phase D: More L-BFGS-B with very low temp
    if elapsed() < budget_s - 100:
        h_new = optimize_lbfgsb(h_best, 0.0003, penalty=500.0, max_iter=600)
        c5_new = compute_c5(h_new)
        if c5_new < c5:
            h_best = h_new
            c5 = c5_new

    # Phase E: Random restarts
    for i in range(8):
        if elapsed() > budget_s - 60:
            break
        scale = 0.03 + 0.01 * i
        h_p = h_best + rng.randn(n) * scale
        h_p = project_h(h_p, n * 0.5)
        h_p = optimize_lbfgsb(h_p, 0.005, penalty=100.0, max_iter=300)
        c5_new = compute_c5(h_p)
        if c5_new < c5:
            h_best = h_p
            c5 = c5_new
            # Refine the improved solution
            if elapsed() < budget_s - 80:
                h_p2 = optimize_lbfgsb(h_best, 0.001, penalty=200.0, max_iter=400)
                c5_p2 = compute_c5(h_p2)
                if c5_p2 < c5:
                    h_best = h_p2
                    c5 = c5_p2

    # Phase F: Final SLSQP polish
    if elapsed() < budget_s - 30:
        h_new, c5_new = optimize_t_var(h_best, max_iter=600, ftol=1e-12)
        if c5_new < c5:
            h_best = h_new
            c5 = c5_new

    # Phase G: Final coordinate descent
    if elapsed() < budget_s - 10:
        h_cd = coord_descent_sweep(h_best, n_sweeps=2)
        c5_cd = compute_c5(h_cd)
        if c5_cd < c5:
            h_best = h_cd
            c5 = c5_cd

    h_best = project_h(h_best, len(h_best) * 0.5)
    c5_final = compute_c5(h_best)
    return h_best, c5_final, len(h_best)
