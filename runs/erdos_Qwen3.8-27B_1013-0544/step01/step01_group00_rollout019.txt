    h = v[:n]
    t = v[-1]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx

def optimize_slsqp(h0, max_iter=400):

    c5_cur = compute_c5(h0)

def block_perturb(h, rng, n_blocks=3, min_bs=5, max_bs=20):

    h_p = h.copy()

        start = rng.randint(0, max(1, n - bs))
        shift = rng.uniform(-0.3, 0.3)
        h_p[start:start + bs] += shift

    return project_h(h_p, n * 0.5)

    h_new = np.interp(x_new, x_old, h_old)

    def time_left():
        return budget_s - elapsed()

    h = np.array(initial_h_values, dtype=float)
    n = len(h)
    h = project_h(h, n * 0.5)
    c5 = compute_c5(h)
    h_best = h.copy()

    # Phase 1: Nesterov PGD with temperature annealing
    temps = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    for temp in temps:
        if time_left() < 80:
            break
        max_it = 3000 if temp > 0.01 else 2000
        h_new, c5_new = optimize_pgd(h, temp, max_iter=max_it)
        if c5_new < c5:
            h, c5 = h_new, c5_new
            h_best = h.copy()

    # Phase 2: SLSQP epigraph polish
    if time_left() > 120:
        h_new, c5_new = optimize_slsqp(h, max_iter=500)
        if c5_new < c5:
            h, c5 = h_new, c5_new
            h_best = h.copy()

    # Phase 3: More PGD after SLSQP
    for temp in [0.001, 0.0005, 0.0002]:
        if time_left() < 60:
            break
        h_new, c5_new = optimize_pgd(h, temp, max_iter=2000)
        if c5_new < c5:
            h, c5 = h_new, c5_new
            h_best = h.copy()

    # Phase 4: Stochastic restarts with block perturbations
    n_restarts = 15
    scales = [0.15, 0.1, 0.05, 0.03, 0.02]
    for i in range(n_restarts):
        if time_left() < 60:
            break
        scale = scales[i % len(scales)]
        h_p = h_best + rng.randn(n) * scale * 0.3
        h_p = block_perturb(h_p, rng, n_blocks=rng.randint(2, 5))
        h_new, c5_new = optimize_pgd(h_p, temp=0.01, max_iter=800)
        if c5_new < c5:
            h, c5 = h_new, c5_new
            h_best = h.copy()
        else:
            h_new2, c5_new2 = optimize_slsqp(h_p, max_iter=200)
            if c5_new2 < c5:
                h, c5 = h_new2, c5_new2
                h_best = h.copy()

    # Phase 5: Try higher resolution
    if time_left() > 200:
        n_new = 360
        h_interp = interp_to(h_best, n_new)
        c5_interp = compute_c5(h_interp)
        h_cur = h_interp
        c5_cur = c5_interp
        for temp in [0.05, 0.01, 0.005, 0.002, 0.001]:
            if time_left() < 80:
                break
            h_new, c5_new = optimize_pgd(h_cur, temp, max_iter=2000)
            if c5_new < c5_cur:
                h_cur, c5_cur = h_new, c5_new
        if time_left() > 60:
            h_new, c5_new = optimize_slsqp(h_cur, max_iter=400)
            if c5_new < c5_cur:
                h_cur, c5_cur = h_new, c5_new
        if c5_cur < c5:
            h_best = h_cur.copy()
            c5 = c5_cur

    # Phase 6: Final SLSQP polish
    if time_left() > 40:
        h_new, c5_new = optimize_slsqp(h_best, max_iter=600)
        if c5_new < c5:
            h_best, c5 = h_new, c5_new

    # Phase 7: Final PGD squeeze
    if time_left() > 30:
        h_new, c5_new = optimize_pgd(h_best, temp=0.0001, max_iter=1500)
        if c5_new < c5:
            h_best, c5 = h_new, c5_new

    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = project_h(h_best, len(h_best) * 0.5)
    c5_final = compute_c5(h_best)
    return h_best, c5_final, len(h_best)
