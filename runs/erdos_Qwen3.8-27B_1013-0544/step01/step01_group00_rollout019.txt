    dx = 2.0 / len(h)

def project_h(h, target):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
    else:
        mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

    c5_cur = compute_c5(h0)
    v0 = np.append(h0.copy(), c5_cur + 0.01)
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.5)]
    cons = [{'type': 'ineq', 'fun': t_ineq}, {'type': 'eq', 'fun': t_eq, 'jac': t_eq_jac}]
    res = minimize(t_obj, v0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': ftol})
    h_new = project_h(res.x[:n], n * 0.5)
    return h_new, compute_c5(h_new)


def binary_sa(h_init, n_steps=5000, temp0=0.3, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    n = len(h_init)
    h = (h_init >= 0.5).astype(float)
    ntarget = n // 2
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if len(ones) > ntarget:
        for idx in rng.choice(ones, len(ones) - ntarget, replace=False):
            h[idx] = 0
    elif len(ones) < ntarget:
        for idx in rng.choice(zeros, ntarget - len(ones), replace=False):
            h[idx] = 1
    c5_cur = compute_c5(h)
    h_best = h.copy()
    c5_best = c5_cur
    for step in range(n_steps):
        T = max(temp0 * (1.0 - step / n_steps) ** 2, 1e-10)
        i = rng.randint(n)
        j = rng.randint(n)
        if h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        c5_new = compute_c5(h)
        if c5_new < c5_cur or rng.random() < np.exp(-(c5_new - c5_cur) / T):
            c5_cur = c5_new
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
    return h_best


def make_init(n, rng, kind):
    if kind == 'random':
        h = (rng.rand(n) < 0.5).astype(float)
    elif kind == 'alt':
        h = np.tile([1.0, 0.0], (n + 1) // 2)[:n]
    elif kind == 'sin':
        x = np.arange(n) / n
        h = (np.sin(2 * np.pi * x) > 0).astype(float)
    else:
        h = np.zeros(n)
        bs = n // 4
        h[:bs] = 1.0
        h[2 * bs:3 * bs] = 1.0
        target = n // 2
        cur = int(np.sum(h))
        if cur < target:
            z = np.where(h == 0)[0]
            h[z[:target - cur]] = 1.0
        elif cur > target:
            o = np.where(h == 1)[0]
            h[o[:cur - target]] = 0.0
    return project_h(h, n * 0.5)


def coord_descent(h0, theta, n_passes=50):
    n = len(h0)
    h = h0.copy()
    target_sum = n * 0.5
    for _ in range(n_passes):
        for i in range(n):
            old = h[i]
            best_obj = softmax_loss(h, theta)
            for tv in [0.0, 1.0, 0.25, 0.75, old]:
                if tv == old:
                    continue
                h[i] = tv
                obj = softmax_loss(h, theta)
                if obj < best_obj - 1e-12:
                    best_obj = obj
            h = project_h(h, target_sum)
        h = project_h(h, target_sum)
    return h, compute_c5(h)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - t0)

    h_best = np.array(initial_h_values, dtype=float)
    n_best = len(h_best)
    h_best = project_h(h_best, n_best * 0.5)
    c5_best = compute_c5(h_best)

    # Phase 1: Binary SA at multiple resolutions for global exploration
    for n_try in [40, 60, 80, 100]:
        if time_left() < 60:
            break
        for trial in range(4):
            if time_left() < 25:
                break
            kind = ['random', 'random', 'sin', 'alt'][trial % 4]
            h_bin = make_init(n_try, rng, kind)
            h_sa = binary_sa(h_bin, n_steps=4000, temp0=0.3, rng=rng)
            c5_sa = compute_c5(h_sa)
            if c5_sa < c5_best:
                c5_best = c5_sa
                h_best = interp_h(h_sa, n_best)
                h_best = project_h(h_best, n_best * 0.5)

    # Phase 2: L-BFGS-B refinement at moderate resolution
    n_mid = 120
    h_mid = interp_h(h_best, n_mid)
    if time_left() > 100:
        for theta in [0.05, 0.02, 0.01, 0.005, 0.002]:
            if time_left() < 40:
                break
            h_new, c5_new = optimize_lbfgsb(h_mid, theta, 1000.0, max_iter=2000)
            h_mid = h_new
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = interp_h(h_mid, n_best)
                h_best = project_h(h_best, n_best * 0.5)

    # Phase 3: L-BFGS-B at full resolution
    h_cur = h_best.copy()
    for theta in [0.03, 0.01, 0.005, 0.002, 0.001]:
        if time_left() < 50:
            break
        h_new, c5_new = optimize_lbfgsb(h_cur, theta, 1000.0, max_iter=3000)
        h_cur = h_new
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new.copy()

    # Phase 4: t-variable SLSQP (exact constraint)
    if time_left() > 120:
        h_new, c5_new = optimize_t_var(h_best, max_iter=600, ftol=1e-11)
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new

    # Phase 5: Coordinate descent for fine tuning
    if time_left() > 80:
        h_new, c5_new = coord_descent(h_best, 0.005, n_passes=30)
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new

    # Phase 6: Another SLSQP pass after coordinate descent
    if time_left() > 80:
        h_new, c5_new = optimize_t_var(h_best, max_iter=500, ftol=1e-12)
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new

    # Phase 7: Try higher resolution
    for n_try in [n_best + 60, n_best + 120]:
        if time_left() < 150:
            break
        h_hi = interp_h(h_best, n_try)
        for theta in [0.02, 0.01, 0.005, 0.002]:
            if time_left() < 40:
                break
            h_new, c5_new = optimize_lbfgsb(h_hi, theta, 1000.0, max_iter=2000)
            h_hi = h_new
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h_new.copy()
        if time_left() > 60:
            h_new, c5_new = optimize_t_var(h_hi, max_iter=400, ftol=1e-11)
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h_new.copy()

    # Final SLSQP polish
    if time_left() > 30:
        h_new, c5_new = optimize_t_var(h_best, max_iter=400, ftol=1e-12)
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new

    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = project_h(h_best, len(h_best) * 0.5)
    c5_final = compute_c5(h_best)
    return h_best, c5_final, len(h_best)
