    n = len(h)
    u = 1.0 - h
    R = np.correlate(h, u, mode='full')
    T = temp * n / 2.0
    R_shifted = R - R.max()
    W = np.exp(R_shifted / T)
    W /= W.sum()
    
    grad = np.zeros(n)

        w = W[n-1+m]

        grad[:n-m] += w * u[m:]

    if abs(diff) < 1e-14:

    # Distribute the difference
    if diff > 0:
        # Need to increase: add to elements that are < 1
        headroom = 1.0 - h
        total_headroom = headroom.sum()
        if total_headroom > 0:
            h += headroom * (diff / total_headroom)
    else:

        if total_excess > 0:
            h += h * (diff / total_excess)
    return np.clip(h, 0.0, 1.0)

    dx = 2.0 / n

    R = np.correlate(h, u, mode='full')

    S *= dx

At each step, pick one index j, and optimize h[j] while keeping all other components fixed. The objective as a function of h[j] is:

- h[j] * (1-h[j+m]) if j+m is in range (j appears in first factor)
- h[j-m] * (1-h[j]) if j-m is in range (j appears in second factor, through 1-h[j])

So R[m] as a function of h[j] = a*h[j] + b*(1-h[j])*c + ... where a, b, c are constants (other h values).
= a*h[j] + c - c*h[j] + ... = (a-c)*h[j] + const

So R[m] IS linear in h[j]! And the max of linear functions is convex in h[j]. So we can find the optimal h[j] by checking where the maximum changes.

This is a great property! We can use **cyclical coordinate descent**:
1. For j = 0, 1, 2, ..., n-1, 0, 1, 2, ...:
   a. Compute R[m] for all m (with current h)
   b. For each m, determine the coefficient of h[j]: coeff_m = (2/n) * dR[m]/dh[j]
   c. The function f(t) = max_m (R[m] + coeff_m * (t - h[j])) for t ∈ [0,1]
   d. Find t* = argmin_{t∈[0,1]} f(t)
   e. Set h[j] = t*
2. After each full cycle, project to satisfy the sum constraint

Step (d) is finding the minimum of a convex piecewise-linear function on [0,1]. This can be done by:
- Evaluating f at t=0 and t=1
- Finding all "breakpoints" where the argmax changes
- The minimum is at a breakpoint or at an endpoint

The breakpoints occur where R[m] + coeff_m * t = R[m'] + coeff_m' * t for some m, m'.
This gives t = (R[m'] - R[m]) / (coeff_m - coeff_m'), which we can enumerate.

But enumerating all pairs (m, m') is O(n²) which for n=180 gives O(n⁴) total per coordinate. That's 180^4 ≈ 10^9, too slow.

Alternative: since f is piecewise linear and convex, we can use a ternary search or golden section search on [0,1] (in O(log(1/eps)) evaluations).

    # Compute coeff[m] = dR[m]/dh[j] for all m
    # R[m] = dx * sum_i h[i] * u[i + m - (n-1)]
    # dR[m]/dh[j] = dx * (contribution of h[j])
    
    # h[j] appears in R[m] when:
    # - As h[i] with i=j: requires j + m - (n-1) to be valid, i.e., 0 ≤ j+m-(n-1) < n
    #   coefficient: u[j+m-(n-1)] = 1 - h[j+m-(n-1)]
    # - As part of u[i + m - (n-1)] = 1 - h[i + m - (n-1)] with i+m-(n-1) = j
    #   i.e., i = j - m + (n-1): requires 0 ≤ j-m+(n-1) < n
    #   coefficient: -h[j-m+(n-1)]
    
    coeff = np.zeros(2*n-1)
    # First contribution: h[j] as the h factor
    idx1 = j + np.arange(2*n-1) - (n-1)  # index into u: j + m - (n-1) for m=0..2n-2
    valid1 = (idx1 >= 0) & (idx1 < n)
    coeff[valid1] += u[idx1[valid1]]
    
    # Second contribution: h[j] in u = 1-h
    idx2 = j - np.arange(2*n-1) + (n-1)  # index into h: j - m + (n-1) for m=0..2n-2
    valid2 = (idx2 >= 0) & (idx2 < n)
    coeff[valid2] -= h[idx2[valid2]]
    
    coeff *= dx
    
    # Now f(t) = max_m (R[m] + coeff[m] * (t - h[j]))
    # = max_m (R[m] - coeff[m]*h[j]) + t * max... no, it's max_m (R[m] + coeff[m]*t - coeff[m]*h[j])
    # Let B[m] = R[m] - coeff[m]*h[j]
    # f(t) = max_m (B[m] + coeff[m]*t)
    
    B = R - coeff * h[j]
    
    # Minimize f(t) over t ∈ [0, 1]
    from scipy.optimize import minimize_scalar
    def f(t):
        return np.max(B + coeff * t)
    
    res = minimize_scalar(f, bounds=(0, 1), method='bounded', options={'xatol': 1e-10})
    return res.x

def coordinate_descent(h0, n_cycles=20, temp_unused=0):
    n = len(h0)
    h = h0.copy()
    target = n * 0.5
    best_h = h.copy()
    best_val = compute_c5(h)
    
    for cycle in range(n_cycles):
        for j in range(n):
            h[j] = coordinate_descent_step(h, j)
        
        # Project to constraint
        h = project_h(h, target)
        
        val = compute_c5(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()
    
    return best_h, best_val

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))


def project_h(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(20):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def t_objective(v):
    return v[-1]


def t_ineq(v):
    n = len(v) - 1
    h = v[:n]
    t = v[-1]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return t - corr


def t_eq(v):
    n = len(v) - 1
    return np.sum(v[:n]) - n * 0.5


def t_eq_jac(v):
    n = len(v) - 1
    j = np.zeros(len(v))
    j[:n] = 1.0
    return j


def optimize_tvar(h0, max_iter=400, ftol=1e-12):

    c5_cur = compute_c5(h0)
    v0 = np.append(h0.copy(), c5_cur + 0.01)
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [
        {'type': 'ineq', 'fun': t_ineq},
        {'type': 'eq', 'fun': t_eq, 'jac': t_eq_jac}
    ]
    res = minimize(t_objective, v0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': ftol})
    h_new = np.clip(res.x[:n], 0.0, 1.0)

def smooth_grad(h, temp):
    n = len(h)
    dx = 2.0 / n
    u = 1.0 - h
    R = np.correlate(h, u, mode='full')
    T = temp * n / 2.0
    Rs = R - R.max()
    ev = np.exp(Rs / T)
    W = ev / ev.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = W[k]
        if w < 1e-16:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            m = lag
            grad[:n - m] += w * u[m:]
            grad[m:] -= w * h[:n - m]
        else:
            p = -lag
            grad[p:] += w * u[:n - p]
            grad[:n - p] -= w * h[p:]
    grad *= dx
    return grad


def grad_desc_ls(h0, temp, max_iters=200):
    n = len(h0)

    h = project_h(h0.copy(), target)
    best_h = h.copy()

        else:

def coord_opt(h, j):
    n = len(h)
    dx = 2.0 / n
    u = 1.0 - h
    R = np.correlate(h, u, mode='full') * dx
    k_vals = np.arange(2 * n - 1)
    idx1 = j + k_vals - (n - 1)
    valid1 = (idx1 >= 0) & (idx1 < n)
    coeff = np.zeros(2 * n - 1)

    idx2 = j - k_vals + (n - 1)
    valid2 = (idx2 >= 0) & (idx2 < n)
    coeff[valid2] -= h[idx2[valid2]]
    coeff *= dx
    B = R - coeff * h[j]
    def f1d(t):
        return float(np.max(B + coeff * t))
    res = minimize_scalar(f1d, bounds=(0.0, 1.0), method='bounded',
