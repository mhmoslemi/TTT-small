import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))


def project_feasible(h, target_sum):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max_val(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    return m + (1.0 / beta) * np.log(np.sum(np.exp(beta * (corr - m))))


def smooth_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    exp_v = np.exp(beta * (corr - m))
    p = exp_v / exp_v.sum()
    grad = np.zeros(n)
    for j in range(n):
        s_lo = max(-(n-1), -j)
        s_hi = min(n-1, n-1-j)
        if s_lo <= s_hi:
            k_lo = s_lo + (n-1)
            k_hi = s_hi + (n-1)
            s_vals = np.arange(s_lo, s_hi+1)
            grad[j] += np.dot(p[k_lo:k_hi+1], 1.0 - h[j+s_vals])
        s_lo2 = max(-(n-1), j-(n-1))
        s_hi2 = min(n-1, j)
        if s_lo2 <= s_hi2:
            k_lo2 = s_lo2 + (n-1)
            k_hi2 = s_hi2 + (n-1)
            s_vals = np.arange(s_lo2, s_hi2+1)
            grad[j] -= np.dot(p[k_lo2:k_hi2+1], h[j-s_vals])
    return grad * dx


def sum_eq_fun(h):
    return np.sum(h) - len(h) * 0.5


def sum_eq_jac(h):
    return np.ones(len(h))


def opt_smooth_slsqp(h0, beta, max_iter=400, ftol=1e-12):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac}]
    res = minimize(smooth_max_val, h0.copy(), args=(beta,), method='SLSQP',
                   jac=lambda x: smooth_max_grad(x, beta),
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': ftol})
    h_new = project_feasible(res.x, n * 0.5)
    return h_new, compute_c5(h_new)


def opt_t_var_slsqp(h0, max_iter=400, ftol=1e-11):
    n = len(h0)
    c5_cur = compute_c5(h0)

    def t_obj(v):
        return v[-1]

    def t_ineq(v):
        nn = len(v) - 1
        hh = v[:nn]
        dxx = 2.0 / nn
        corr = np.correlate(hh, 1.0 - hh, mode='full') * dxx
        return v[-1] - corr

    def t_eq(v):
        return np.sum(v[:-1]) - (len(v)-1) * 0.5

    def t_eq_j(v):
        j = np.zeros(len(v))
        j[:-1] = 1.0
        return j

    v0 = np.append(h0.copy(), c5_cur + 0.01)
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.5)]
    cons = [{'type': 'ineq', 'fun': t_ineq},
            {'type': 'eq', 'fun': t_eq, 'jac': t_eq_j}]
    res = minimize(t_obj, v0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': ftol})
    h_new = project_feasible(res.x[:n], n * 0.5)
    return h_new, compute_c5(h_new)


def coordinate_descent(h0, target_sum, max_sweeps=80):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5

    for sweep in range(max_sweeps):
        for j in range(n):
            h_save = h[j]
            h[j] = 0.0
            c0 = np.correlate(h, 1.0 - h, mode='full') * dx
            h[j] = 1.0
            c1 = np.correlate(h, 1.0 - h, mode='full') * dx
            delta = c1 - c0

            t_grid = np.linspace(0.0, 1.0, 33)
            f_vals = np.max(c0[None, :] + t_grid[:, None] * delta[None, :], axis=1)
            t_best = t_grid[np.argmin(f_vals)]
            h[j] = t_best

        h = project_feasible(h, target_sum)
        c5_new = compute_c5(h)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
        elif sweep > 10:
            h = best_h.copy()
            break
        c5 = compute_c5(h)

    return best_h, best_c5


def subgradient_opt(h0, target_sum, max_iter=8000, tol_init=0.01, momentum=0.85):
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    step = 0.005
    tol = tol_init

    for it in range(max_iter):
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        cmax = np.max(corr)
        active = corr >= (cmax - tol)
        n_act = max(1, np.sum(active))

        grad = np.zeros(n)
        for j in range(n):
            s_lo = max(-(n-1), -j)
            s_hi = min(n-1, n-1-j)
            if s_lo <= s_hi:
                k_range = np.arange(s_lo + (n-1), s_hi + (n-1) + 1)
                act = active[k_range]
                s_vals = np.arange(s_lo, s_hi+1)
                if np.any(act):
                    grad[j] += np.sum((1.0 - h[j+s_vals]) * act)
            s_lo2 = max(-(n-1), j-(n-1))
            s_hi2 = min(n-1, j)
            if s_lo2 <= s_hi2:
                k_range2 = np.arange(s_lo2 + (n-1), s_hi2 + (n-1) + 1)
                act2 = active[k_range2]
                s_vals2 = np.arange(s_lo2, s_hi2+1)
                if np.any(act2):
                    grad[j] -= np.sum(h[j-s_vals2] * act2)
        grad *= dx / n_act

        v = momentum * v - step * grad
        h_new = project_feasible(h + v, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5:
            h = h_new
            c5 = c5_new
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            step = min(step * 1.01, 0.02)
        else:
            step *= 0.6
        if it % 2000 == 0 and it > 0:
            tol *= 0.7
    return best_h, best_c5

    x_new = np.arange(n_new) / n_new

    return project_feasible(h_new, n_new * 0.5)

    h_best = project_feasible(np.clip(h_best, 0.0, 1.0), len(h_best) * 0.5)
    c5_final = compute_c5(h_best)
    return h_best, c5_final, len(h_best)
