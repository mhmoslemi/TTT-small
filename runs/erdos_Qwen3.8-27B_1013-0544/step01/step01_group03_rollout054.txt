import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_onto_constraint(h, target_sum):
    """Project h onto {x: 0<=x<=1, sum(x)=target_sum}"""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(h, tau):
    corr = np.correlate(h, 1.0 - h, mode="full")
    x = tau * corr
    x_max = x.max()
    return float(x_max / tau + np.log(np.sum(np.exp(x - x_max))) / tau)

import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_onto_constraint(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        mask = h < 1.0 if diff > 0 else h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_grad(h, tau):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    x = tau * corr
    x_max = x.max()
    obj = x_max + np.log(np.sum(np.exp(x - x_max))) / tau
    exp_x = np.exp(x - x_max)
    p = exp_x / exp_x.sum()
    first_term = np.convolve(p, 1.0 - h, mode='full')[:n]
    second_term = np.correlate(h, p, mode='full')[:n]
    grad = first_term - second_term
    return obj, grad

def grad_optimize(h, target_sum, tau, lr, steps, momentum=0.85):
    n = len(h)
    h = h.copy()
    v = np.zeros(n)
    for _ in range(steps):
        _, grad = smooth_max_obj_grad(h, tau)
        v = momentum * v - lr * grad
        h = project_onto_constraint(h + v, target_sum)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(120, 0.5)

    n = len(best_h)
    target_sum = n / 2.0
    best_c5 = evaluate_c5(best_h)

    def rem():
        return budget_s - (time.time() - t0)

    # Phase 1: Multi-scale gradient descent with momentum on initial n
    schedule = [
        (30, 0.15, 4000),
        (100, 0.08, 4000),
        (400, 0.04, 4000),
        (1500, 0.02, 4000),
        (6000, 0.01, 4000),
        (25000, 0.005, 4000),
        (100000, 0.002, 5000),
        (500000, 0.001, 5000),
    ]
    for tau, lr, steps in schedule:
        if rem() < 15:
            break
        h = grad_optimize(best_h, target_sum, tau, lr, steps)
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_h = h

    # Phase 2: Try different resolutions

        h_t = project_onto_constraint(h_t, tsum)
        for tau, lr, steps in [(200, 0.05, 2000), (2000, 0.02, 2000), (20000, 0.005, 3000)]:

            target_sum = tsum
