```python
def project_onto_constraint(v, target):
    """Project v onto {x: sum(x) = target, 0 <= x[i] <= 1}"""
    # Use bisection on the additve constant
    lo, hi = v.min() - target, v.max()
    for _ in range(100):

        projected = np.clip(v + mid, 0, 1)
        if projected.sum() > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo+hi)/2, 0, 1)
```

```python
def subgradient_step(h, step_size, n_active=5):
    n = len(h)
    corr = compute_corr(h)
    max_val = np.max(corr)
    # Find active lags (within 1% of max)
    threshold = max_val * 0.99
    active = np.where(corr >= threshold)[0]
    # For robustness, use top n_active
    if len(active) > n_active:
        top_vals = np.sort(corr)[::-1][:n_active]
        threshold = top_vals[-1]
        active = np.where(corr >= threshold)[0]
    
    # Compute subgradient: average of subgradients of active c_k
    # subgrad of c_k at h[i] = (1-h[i+k]) - h[i-k] (with boundary handling)
    # In the "full" correlation indexing: corr index = i + (n-1) corresponds to lag i
    # corr[k + n - 1] = sum_j h[j](1-h[j+k]) for lag k
    
    grad = np.zeros(n)
    for k_idx in active:
        k = k_idx - (n - 1)  # actual lag
        for i in range(n):
            g = 0.0
            # Term where i is the "first" position: h[i](1-h[i+k])
            # Valid if 0 <= i+k < n
            if 0 <= i + k < n:
                g += (1 - h[i + k])
            # Term where i is the "second" position: h[i-k](1-h[i])
            # Valid if 0 <= i-k < n
            if 0 <= i - k < n:
                g -= h[i - k]
            grad[i] += g
    
    grad /= len(active)
    return h - step_size * grad
```

```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_sum(v, target):
    lo = -v.max() - 1.0
    hi = -v.min() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(v + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)

def subgradient_direction(h, n_active=10):
    n = len(h)
    corr = compute_corr(h)
    max_val = np.max(corr)
    if max_val < 1e-10:
        return np.zeros(n)
    # Top n_active lags
    top_indices = np.argsort(corr)[::-1][:n_active]
    grad = np.zeros(n)
    for k_idx in top_indices:
        k = k_idx - (n - 1)
        if k >= 0:
            grad[:n-k] += (1.0 - h[k:])
            grad[k:] -= h[:n-k]
        else:
            ka = -k
            grad[ka:] += (1.0 - h[:n-ka])
            grad[:n-ka] -= h[ka:]
    grad /= n_active
    return grad

def continuous_optimize(h, t0, deadline, n_steps=5000):
    """Projected subgradient + SLSQP refinement."""
    n = len(h)
    target = n / 2.0
    h = project_sum(h, target)
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Subgradient phase
    step = 0.1
    for it in range(n_steps):
        if time.time() > deadline:
            break
        grad = subgradient_direction(h, n_active=10)
        h_new = h - step * grad
        h_new = project_sum(h_new, target)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        h = h_new
        if it % 200 == 0:
            step *= 0.95
    
    # SLSQP refinement
    if time.time() < deadline - 10:
        def smooth_max(obj, tau):
            x = tau * obj
            x_max = np.max(x)
            return float(x_max / tau + np.log(np.sum(np.exp(x - x_max))) / tau)
        
        def obj(x):
            corr = compute_corr(x)
            return smooth_max(corr, 50.0)
        
        bnds = Bounds(np.zeros(n), np.ones(n))
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                 'jac': lambda x: np.ones(n)}]
        try:
            res = minimize(obj, best_h, method='SLSQP', bounds=bnds,
                          constraints=cons, options={'maxiter': 200, 'ftol': 1e-10})
            x = project_sum(res.x, target)
            c5 = evaluate_c5(x)
            if c5 < best_c5:
                best_h = x
                best_c5 = c5
        except:
            pass
    
    return best_h, best_c5
```

Let me also add one more idea: **iterative proportional fitting (IPF)** or **coordinate descent**.

Coordinate descent: At each step, pick a position i and optimize h[i] while keeping all other positions fixed.

Given all h[j] for j ≠ i, the optimal h[i] to minimize max_k c_k is... well, it's complicated because c_k depends on h[i] in a specific way.

For a given i, c_k depends on h[i] only when i is in the valid range for lag k. Specifically:
- c_k includes the term h[i](1-h[i+k]) if 0 ≤ i+k < n
- c_k includes the term h[i-k](1-h[i]) if 0 ≤ i-k < n (this is in c_k, not c_{k'})

So for a fixed i, c_k is linear in h[i] (for each k). The max of linear functions is convex in h[i]. So the optimal h[i] is either 0, 1, or a point where two c_k's are equal and at the maximum.

This suggests a coordinate descent approach:
1. For each i (in some order), compute the optimal h[i] given all other h[j]
2. Iterate

The optimal h[i] can be found by:
- Computing c_k as a function of h[i] = t
- Finding t ∈ [0,1] that minimizes max_k c_k(t)

Since c_k(t) is linear in t (for each k), max_k c_k(t) is the upper envelope of lines, which is piecewise linear and convex. The minimum is at a point where two lines cross (or at a boundary).

This is a 1D optimization problem that can be solved exactly (or approximately) for each coordinate.

```python
def coordinate_descent_step(h, i):
    """Optimize h[i] given all other coordinates fixed."""
    n = len(h)
    # Compute c_k as a function of h[i] = t
    # c_k = A_k * t + B_k where A_k and B_k depend on other h values
    
    # For lag k (in the "full" correlation indexing):
    # c_{k-(n-1)} = sum_j h[j](1-h[j+k])
    # The terms involving h[i]:
    # - j = i: h[i](1-h[i+k]) = t * (1-h[i+k]) [if i+k valid]
    # - j+k = i, j = i-k: h[i-k](1-h[i]) = h[i-k]*(1-t) [if i-k valid]
    
    # So c_lag = sum_{j≠j_i} h[j](1-h[j+lag]) + [contribution from h[i]]
    # The contribution from h[i] to c_lag:
    # If 0 ≤ i+lag < n: t * (1-h[i+lag])
    # If 0 ≤ i-lag < n: h[i-lag] * (1-t)
    # Total linear coefficient of t: (1-h[i+lag]) - h[i-lag] (where valid)
    # Constant term: the rest
    
    # Let's compute for all lags
    h_i = h[i]
    h[i] = 0.0  # Temporarily set to 0
    base_corr = compute_corr(h)  # c_k when h[i] = 0
    
    # Now compute the coefficient of t for each lag
    coeff = np.zeros(2*n-1)
    for lag in range(-(n-1), n):
        c = 0.0
        if 0 <= i + lag < n:
            c += (1.0 - h[i + lag])  # from h[i](1-h[i+lag])
        if 0 <= i - lag < n:
            c -= h[i - lag]  # from h[i-lag](1-h[i]) = h[i-lag] - h[i-lag]*t, so coeff of t is -h[i-lag]
        coeff[lag + n - 1] = c
    
    # c_k(t) = base_corr[k] + coeff[k] * t
    # We want to find t ∈ [0,1] minimizing max_k (base_corr[k] + coeff[k]*t)
    
    # This is a 1D convex problem. Use bisection or ternary search.
    # Since it's piecewise linear, the minimum is at a breakpoint or boundary.
    # Breakpoints occur where two lines cross: base[a] + coeff[a]*t = base[b] + coeff[b]*t
    # t = (base[b] - base[a]) / (coeff[a] - coeff[b])
    
    # For simplicity, use ternary search (works for convex functions)
    lo, hi = 0.0, 1.0
    for _ in range(50):
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        v1 = np.max(base_corr + coeff * m1)
        v2 = np.max(base_corr + coeff * m2)
        if v1 < v2:
            hi = m2
        else:
            lo = m1
    
    t = (lo + hi) / 2
    h[i] = t
    return h, np.max(base_corr + coeff * t)
```

```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_sum(v, target):
    lo = -v.max() - 1.0
    hi = -v.min() + 1.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.clip(v + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)

def binary_sa(h_init, n_iter, t0, deadline, seed):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    n1 = n // 2
    h = np.zeros(n)
    h[np.argsort(h_init)[::-1][:n1]] = 1.0
    curr_corr = compute_corr(h)
    curr_max = float(np.max(curr_corr))
    best_h = h.copy()
    best_max = curr_max
    temp = 0.4 * curr_max
    min_temp = 0.005 * curr_max
    for it in range(n_iter):
        if time.time() > deadline:
            break
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1.0 - h)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_corr = compute_corr(h)
        new_max = float(np.max(new_corr))
        delta = new_max - curr_max
        if delta < 0 or (temp > min_temp and rng.random() < np.exp(-max(delta,0.0) / temp)):
            curr_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        if it % 80 == 0 and temp > min_temp:
            temp *= 0.975
    return best_h, best_max * (2.0 / n)

def coord_descent(h, n_sweeps, t0, deadline):
    n = len(h)
    target = n / 2.0
    h = project_sum(h, target).copy()
    corr = compute_corr(h)
    best_h = h.copy()
    best_c5 = float(np.max(corr)) * (2.0 / n)
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        for i in range(n):
            old_val = h[i]
            coeff = np.zeros(2 * n - 1)
            for L in range(-(n - 1), n):
                c = 0.0
                ip = i + L
                if 0 <= ip < n:
                    c += (1.0 - h[ip])
                im = i - L
                if 0 <= im < n:
                    c -= h[im]
                coeff[L + n - 1] = c
            base2 = corr - coeff * old_val
            lo_t, hi_t = 0.0, 1.0
            for _ in range(40):
                m1 = lo_t + (hi_t - lo_t) / 3.0
                m2 = hi_t - (hi_t - lo_t) / 3.0
                if np.max(base2 + coeff * m1) < np.max(base2 + coeff * m2):
                    hi_t = m2
                else:
                    lo_t = m1
            t_opt = (lo_t + hi_t) / 2.0
            if abs(t_opt - old_val) > 1e-10:
                corr = corr + (t_opt - old_val) * coeff
                h[i] = t_opt
        h = project_sum(h, target)
        corr = compute_corr(h)
        c5 = float(np.max(corr)) * (2.0 / n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def slsqp_refine(h, t0, deadline, tau=200.0, maxiter=200):
    n = len(h)
    target = n / 2.0
    h = project_sum(h, target).copy()

    def obj(x):
        corr = compute_corr(x)
        xt = tau * corr
        xm = np.max(xt)
        return float(xm / tau + np.log(np.sum(np.exp(xt - xm))) / tau)

    bnds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
        x = project_sum(res.x, target)
        c5 = evaluate_c5(x)
        return x, c5
    except Exception:
        return h, evaluate_c5(h)

def _cons_fun(x):
    return np.sum(x) - len(x) / 2.0

def _cons_jac(x):
    return np.ones(len(x))

def make_structured_init(n, variant, seed):
    rng = np.random.RandomState(seed)
    if variant == 0:
        x = np.linspace(0, 2, n, endpoint=False)
        h = 0.5 + 0.35 * np.cos(5.0 * np.pi * x) + 0.25 * np.cos(11.0 * np.pi * x)
    elif variant == 1:
        h = np.zeros(n)
        block = max(2, n // 16)
        for start in range(0, n, block * 2):
            end = min(start + block, n)
            h[start:end] = 1.0
    elif variant == 2:
        h = (rng.rand(n) > 0.5).astype(float)
    elif variant == 3:
        x = np.linspace(0, 2, n, endpoint=False)
        h = 0.5 + 0.4 * np.cos(3.0 * np.pi * x + 0.5) + 0.3 * np.sin(7.0 * np.pi * x)
    elif variant == 4:
        h = np.zeros(n)
        step = max(2, n // 12)
        for k in range(n // step + 1):
            pos = k * step
            if pos < n:
                h[pos] = 1.0
    else:
        h = rng.rand(n)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 15

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = make_structured_init(120, 0, seed)

    n0 = len(h_init)
    best_h = h_init.copy()
    best_c5 = evaluate_c5(best_h)

    # Phase 1: Binary SA with diverse starts
    for trial in range(6):

        if trial == 0:
            h_seed = h_init
        else:

        h_bin, c5_bin = binary_sa(h_seed, 100000, t0, deadline - 200, seed + trial * 7)
        if c5_bin < best_c5:
            best_h = h_bin
            best_c5 = c5_bin

    # Phase 2: Coordinate descent refinement
    h_cd, c5_cd = coord_descent(best_h, 80, t0, deadline - 100)

    # Phase 3: SLSQP polish with multiple tau
    for tau in [50.0, 200.0, 1000.0]:
        if time.time() > deadline - 30:

        h_sl, c5_sl = slsqp_refine(best_h, t0, deadline - 30, tau=tau, maxiter=150)

    # Phase 4: More coordinate descent rounds
```
