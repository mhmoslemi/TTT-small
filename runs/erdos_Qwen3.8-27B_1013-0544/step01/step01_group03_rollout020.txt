import numpy as np
from scipy.optimize import minimize, Bounds
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val(obj, tau):
    x = tau * obj
    x_max = np.max(x)
    return float(x_max / tau + np.log(np.sum(np.exp(x - x_max))) / tau)

def project_polytope(h, target):
    h = np.array(h, dtype=float).copy()
    for _ in range(200):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-13:
            break
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_grad(h, w_softmax, n):
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        l = abs(lag)
        wk = w_softmax[k] * dx
        if wk < 1e-14:
            continue
        if l == 0:
            grad += wk * (1.0 - 2.0 * h)
        else:
            m = n - l
            grad[:m] += wk * (1.0 - h[l:l + m])
            grad[l:] += -wk * h[:m]
    return grad

def pg_step(h, tau, lr, n):
    target = n / 2.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    x = tau * dx * corr
    x_max = np.max(x)
    w = np.exp(x - x_max)
    w /= w.sum()
    grad = compute_grad(h, w, n)
    h_new = h - lr * grad
    return project_polytope(h_new, target)

def pg_anneal(h_start, n_pts, tau_seq, lr_seq):
    h = h_start.copy()
    for tau, lr in zip(tau_seq, lr_seq):
        for _ in range(50):
            h = pg_step(h, tau, lr, n_pts)
    return h

def slsqp_solve(h_start, n_pts, tau, maxiter=250):
    target = n_pts / 2.0
    dx = 2.0 / n_pts
    bnds = Bounds(np.zeros(n_pts), np.ones(n_pts))

    def obj(x):
        return smooth_max_val(np.correlate(x, 1.0 - x, mode="full") * dx, tau)

    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        xx = tau * dx * corr
        xm = np.max(xx)
        w = np.exp(xx - xm)
        w /= w.sum()
        return compute_grad(x, w, n_pts)

    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n_pts)}]
    try:
        res = minimize(obj, h_start, jac=obj_grad, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        x = project_polytope(res.x, target)
        return x, evaluate_c5(x)
    except Exception:
        return h_start, None

            h_t, c5_t = slsqp_solve(h_t, tn, tau, 200)
            if c5_t is not None and c5_t < best_c5:
                best_h, best_c5 = h_t, c5_t

    # Phase 5: Diverse restarts
    hn = len(best_h)
    target = hn / 2.0
    starts = [best_h.copy(), 1.0 - best_h, best_h[::-1].copy()]
    # Add random structured seeds
    for _ in range(4):
        starts.append(project_polytope(
            np.clip(0.5 + 0.3 * np.random.randn(hn), 0, 1), target))
    # Add block patterns
    for freq in [2, 3, 4, 5]:
        x = np.linspace(0, 2 * np.pi * freq, hn, endpoint=False)
        starts.append(project_polytope(0.5 + 0.4 * np.sin(x), target))

    for s in starts:
        if rem() < 20:
            break
        h_s = project_polytope(s, target)
        for tau in [500, 2000, 5000]:
            if rem() < 10:
                break
            h_new, c5_new = slsqp_solve(h_s, hn, tau, 250)
            if c5_new is not None and c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new
            h_s = new if c5_new is not None else h_s

    # Phase 6: Random perturbations around best
    for _ in range(40):
        if rem() < 8:
            break
        scale = 0.03 + 0.15 * np.random.random()
        hp = best_h + np.random.normal(0, scale, hn)
        hp = project_polytope(np.clip(hp, 0, 1), target)
        h_new, c5_new = slsqp_solve(hp, hn, 1000, 150)
        if c5_new is not None and c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 7: Final fine-tuning with very high tau
    if rem() > 12:
        for tau in [8000, 15000, 30000]:
            if rem() < 8:
                break
            h_new, c5_new = slsqp_solve(best_h, len(best_h), tau, 350)
            if c5_new is not None and c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new

    # Final: PG polish
    if rem() > 5:
        hn_f = len(best_h)
        for tau, lr in [(20000, 0.0005), (50000, 0.0002)]:
            for _ in range(100):
                best_h = pg_step(best_h, tau, lr, hn_f)
        c5_f = evaluate_c5(best_h)
        if c5_f < best_c5:
            best_c5 = c5_f

    best_h = project_polytope(best_h, len(best_h) / 2.0)
    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, len(best_h)
