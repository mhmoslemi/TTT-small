    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) > 1e-12 and s > 0:
        # Water-filling approach
        h = (h - h.min()) / (h.max() - h.min() + 1e-15)

            if abs(s - target_sum) < 1e-12:

import time

    corr = corr_of_h(h)
    m = np.max(corr)
    e = np.exp(beta * (corr - m))
    weights = e / e.sum()

        if je2 > js2:
            idx2 = np.arange(js2, je2)
            grad[idx2] += w * (-h[idx2 + lag])
    return grad * dx

    def __init__(self, beta):
        self.beta = beta

    def __call__(self, h):
        return smooth_max(corr_of_h(h), self.beta) * (2.0 / len(h))

    def grad(self, h):
        return smooth_max_grad(h, self.beta)


def sum_con_fun(h):
    return h.sum() - len(h) / 2.0


def sum_con_jac(h):
    return np.ones(len(h))

    return [{'type': 'eq', 'fun': sum_con_fun, 'jac': sum_con_jac}]


def get_active_lags(h, threshold_frac=0.005):
    corr = corr_of_h(h)
    m = np.max(corr)
    threshold = m * (1.0 - threshold_frac)
    active = np.where(corr >= threshold)[0]
    return active, m


def targeted_perturb(h, active_lags, n, intensity=0.02):
    h_new = h.copy()
    for k_idx in active_lags[:5]:
        lag = k_idx - (n - 1)
        for j in range(n):
            if 0 <= j - lag < n:
                h_new[j] -= intensity * (1.0 - h[j - lag])
            if 0 <= j + lag < n:
                h_new[j] -= intensity * h[j + lag]
    return h_new

    np.random.seed(seed)
    start_time = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_points = len(h_init)
