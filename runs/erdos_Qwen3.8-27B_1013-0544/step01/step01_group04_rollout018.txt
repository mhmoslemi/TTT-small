from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def c5_smoothed(h, dx, t):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = corr * dx * t
    max_s = np.max(scaled)
    exp_s = np.exp(scaled - max_s)
    Z = np.sum(exp_s)
    return (max_s + np.log(Z)) / t * dx  # This is the log-sum-exp / t * dx

def c5_smoothed_grad(h, dx, t):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = corr * dx * t
    max_s = np.max(scaled)
    exp_s = np.exp(scaled - max_s)
    weights = exp_s / np.sum(exp_s)
    
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        w = weights[k_idx]
        if w < 1e-14:
            continue
        k = k_idx - (n - 1)
        if k >= 0:
            grad[k:] += w * (1.0 - h[:n-k])
            grad[:n-k] -= w * h[k:]
        else:
            kk = -k
            grad[:n-kk] += w * (1.0 - h[kk:])
            grad[kk:] -= w * h[:n-kk]
    grad *= dx
    return grad

    n = len(h)
    val = c5_smoothed(h, dx, t)
    penalty = lam * (np.sum(h) - n/2.0)**2
    return val + penalty

    g = c5_smoothed_grad(h, dx, t)
    g += 2.0 * lam * (np.sum(h) - n/2.0)
    return g

    """Smooth approximation of C5 using log-sum-exp."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')

    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    dx0 = 2.0 / n0

    best_h = project_feasible(h_init.copy(), n0)
    best_c5 = compute_c5(best_h, dx0)

    def elapsed():
        return time.time() - start_time

    # Phase 1: Optimize at original resolution with multiple temperatures
    phase1_end = budget_s * 0.35
    # Sweep temperatures from low to high
    temps = [3, 8, 20, 50, 150, 400]
    for t in temps:
        if elapsed() > phase1_end:
            break
        h_try = try_optimize(best_h, n0, dx0, t, 100.0, 300, budget_s - elapsed())
        c5_try = compute_c5(h_try, dx0)
        if c5_try < best_c5:
            best_h = h_try
            best_c5 = c5_try

    # Random restarts at n0
    scales = [0.15, 0.10, 0.06, 0.03, 0.02, 0.01]
    for scale in scales:
        if elapsed() > phase1_end:
            break
        for _ in range(2):
            if elapsed() > phase1_end:
                break
            h_pert = best_h + rng.randn(n0) * scale
            h_pert = project_feasible(h_pert, n0)
            h_try = try_optimize(h_pert, n0, dx0, 50, 200.0, 250, budget_s - elapsed())
            c5_try = compute_c5(h_try, dx0)
            if c5_try < best_c5:
                best_h = h_try
                best_c5 = c5_try

    # Also try complement
    h_comp = 1.0 - best_h
    h_comp = project_feasible(h_comp, n0)
    c5_comp = compute_c5(h_comp, dx0)
    if c5_comp < best_c5:
        best_h = h_comp
        best_c5 = c5_comp
    h_try_c = try_optimize(h_comp, n0, dx0, 50, 200.0, 250, budget_s - elapsed())
    c5_try_c = compute_c5(h_try_c, dx0)
    if c5_try_c < best_c5:
        best_h = h_try_c
        best_c5 = c5_try_c

    # Phase 2: Upsample and optimize at higher resolution
    phase2_end = budget_s * 0.72
    if elapsed() < phase2_end:
        n2 = 2 * n0
        dx2 = 2.0 / n2
        x_old = np.linspace(0.0, 1.0, n0, endpoint=False)
        x_new = np.linspace(0.0, 1.0, n2, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = project_feasible(h_up, n2)
        c5_up = compute_c5(h_up, dx2)
        if c5_up < best_c5:
            best_h = h_up.copy()
            best_c5 = c5_up

        for t in [10, 30, 100, 300]:
            if elapsed() > phase2_end:
                break
            h_try = try_optimize(h_up, n2, dx2, t, 500.0, 250, budget_s - elapsed())
            c5_try = compute_c5(h_try, dx2)
            if c5_try < best_c5:
                best_h = h_try
                best_c5 = c5_try
            h_up = h_try if c5_try < c5_up else h_up
            c5_up = min(c5_up, c5_try)

        for scale in [0.08, 0.04, 0.02]:
            if elapsed() > phase2_end:
                break
            for _ in range(2):
                if elapsed() > phase2_end:
                    break
                h_pert = h_up + rng.randn(n2) * scale
                h_pert = project_feasible(h_pert, n2)
                h_try = try_optimize(h_pert, n2, dx2, 50, 500.0, 250, budget_s - elapsed())
                c5_try = compute_c5(h_try, dx2)
                if c5_try < best_c5:
                    best_h = h_try
                    best_c5 = c5_try
                if c5_try < c5_up:
                    h_up = h_try
                    c5_up = c5_try

    # Phase 3: Final polishing at best resolution
    n_final = len(best_h)
    dx_final = 2.0 / n_final
    phase3_end = budget_s - 40

    # High-temperature L-BFGS-B (approximates exact max)
    for t in [300, 800, 2000]:
        if elapsed() > phase3_end:
            break
        h_try = try_optimize(best_h, n_final, dx_final, t, 5000.0, 400, budget_s - elapsed())
        c5_try = compute_c5(h_try, dx_final)
        if c5_try < best_c5:
            best_h = h_try
            best_c5 = c5_try

    # Tiny perturbations + re-optimize
    while elapsed() < phase3_end:
        h_pert = best_h + rng.randn(n_final) * 0.005
        h_pert = project_feasible(h_pert, n_final)
        h_try = try_optimize(h_pert, n_final, dx_final, 800, 10000.0, 300, budget_s - elapsed())
        c5_try = compute_c5(h_try, dx_final)
        if c5_try < best_c5:
            best_h = h_try
            best_c5 = c5_try

    # Final SLSQP polish for exact max
    if elapsed() < budget_s - 15 and n_final <= 200:
        from scipy.optimize import SLSQP
        def exact_obj(x):
            return compute_c5(x, dx_final)
        def exact_obj_grad(x):
            # Subgradient: gradient of the active (max) lag
            corr = np.correlate(x, 1.0 - x, mode='full')
            max_idx = np.argmax(corr)
            k = max_idx - (n_final - 1)
            g = np.zeros(n_final)
            if k >= 0:
                g[k:] += (1.0 - x[:n_final - k])
                g[:n_final - k] -= x[k:]
            else:
                kk = -k
                g[:n_final - kk] += (1.0 - x[kk:])
                g[kk:] -= x[:n_final - kk]
            return g * dx_final
        try:
            res = minimize(exact_obj, best_h, jac=exact_obj_grad, method='SLSQP',
                          bounds=[(0.0, 1.0)] * n_final,
                          constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n_final / 2.0}],
                          options={'maxiter': 200, 'ftol': 1e-12})
            h_new = project_feasible(res.x, n_final)
            c5_new = compute_c5(h_new, dx_final)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
        except Exception:
            pass

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return (best_h, best_c5, len(best_h))
