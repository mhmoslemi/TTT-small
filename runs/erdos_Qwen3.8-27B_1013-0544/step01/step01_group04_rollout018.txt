    h_sorted = np.sort(h)[::-1]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - target
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            h[free] -= diff / np.count_nonzero(free)
        else:
            h -= diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = np.sum(h) - target
    if abs(diff) > 1e-14:
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            h[free] -= diff / np.count_nonzero(free)
            h = np.clip(h, 0.0, 1.0)
    return h

def binarize(h, n):
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n, dtype=np.int8)
    h_bin[idx[:n//2]] = 1
    return h_bin

def rebalance(hb, n, rng):
    s = int(hb.sum())
    if s > n//2:
        oi = np.where(hb==1)[0]
        fl = rng.choice(oi, s - n//2, replace=False)
        hb[fl] = 0
    elif s < n//2:
        zi = np.where(hb==0)[0]
        fl = rng.choice(zi, n//2 - s, replace=False)
        hb[fl] = 1
    return hb

def sa_search(h_bin, n, iters, T0, T1, rng, kick_interval=0):
    dx = 2.0 / n
    h = h_bin.astype(np.float64)
    cur = compute_c5(h, dx)
    best = cur
    best_h = h.copy()
    for it in range(iters):
        if kick_interval > 0 and it > 0 and it % kick_interval == 0:
            n_k = max(4, n // 10)
            kp = rng.choice(n, n_k, replace=False)
            for p in kp:
                h_bin[p] ^= 1
            rebalance(h_bin, n, rng)
            h = h_bin.astype(np.float64)
            cur = compute_c5(h, dx)
            continue
        T = T0 * (T1 / T0) ** (it / max(iters - 1, 1))
        oi = np.where(h_bin == 1)[0]
        zi = np.where(h_bin == 0)[0]
        a = oi[rng.randint(len(oi))]
        b = zi[rng.randint(len(zi))]
        h_bin[a], h_bin[b] = h_bin[b], h_bin[a]
        h[a], h[b] = h[b], h[a]
        nc = compute_c5(h, dx)
        d = nc - cur
        if d <= 0 or rng.rand() < np.exp(-d / max(T, 1e-15)):
            cur = nc
            if nc < best:
                best = nc
                best_h = h.copy()
        else:
            h_bin[a], h_bin[b] = h_bin[b], h_bin[a]
            h[a], h[b] = h[b], h[a]
    return best_h, best

def slsqp_polish(h0, n, maxit=400):
    dx = 2.0 / n
    bnds = [(0.0, 1.0)] * n
    def f(x):
        return compute_c5(x, dx)
    def g(x):
        return x.sum() - n / 2.0
    r = minimize(f, h0, method='SLSQP', bounds=bnds,
                 constraints=[{'type': 'eq', 'fun': g}],
                 options={'maxiter': maxit, 'ftol': 1e-15})
    return project_feasible(r.x, n)
