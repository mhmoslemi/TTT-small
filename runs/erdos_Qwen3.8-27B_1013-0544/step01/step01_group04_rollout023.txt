    # = dx/eps * (sum_l w_l - sum_l w_l*h[j-l] - sum_l w_l*h[j+l])
    # 
    # sum_l w_l*h[j-l] = correlate(h, w, 'same') but need to be careful
    # Actually, let's just compute it directly for n=90 (it's fast)
    
    grad = np.zeros(n)
    for j in range(n):
        val = 0.0
        for i, lag in enumerate(lags):
            val += weights[i] * dx
            # (1-h[j-lag]) term: valid if 0 <= j-lag < n
            if 0 <= j - lag < n:
                val -= weights[i] * dx * h[j - lag]
            # -h[j+lag] term: valid if 0 <= j+lag < n
            if 0 <= j + lag < n:
                val -= weights[i] * dx * h[j + lag]
        grad[j] = val / epsilon
    
    return grad
