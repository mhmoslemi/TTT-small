def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(corr, t):
    max_c = np.max(corr)
    if t < 1:
        return max_c
    shifted = t * (corr - max_c)
    return max_c + np.log(np.sum(np.exp(shifted))) / t

            for idx_i in idx:
                if n_ones < n // 2 and h_val[idx_i] == 0:
                    h_val[idx_i] = 1
                    n_ones += 1
                elif n_ones > n // 2 and h_val[idx_i] == 1:
                    h_val[idx_i] = 0
                    n_ones -= 1
                if n_ones == n // 2:
                    break
        c5 = compute_c5(h_val, dx)
        return h_val, c5
    return None, None

def binary_greedy_search(h, max_rounds=50):
    """Greedy: try all swaps, pick best improvement per round."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    for round in range(max_rounds):
        if best_c5 <= 0:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_delta = 0
        best_swap = None
        
        # Vectorized: try all swaps
        # For each (i in ones, j in zeros), compute resulting c5
        # This is O(n² * n) = O(n³) total
        improved = False
        for i in ones:
            for j in zeros:
                h[i] = 0
                h[j] = 1
                c5 = compute_c5(h, dx)
                delta = best_c5 - c5
                if delta > best_delta:
                    best_delta = delta
                    best_swap = (i, j)
                h[i] = 1
                h[j] = 0
        
        if best_swap is not None:
            i, j = best_swap
            h[i] = 0
            h[j] = 1
            best_c5 = compute_c5(h, dx)
        else:
            break
    
    return h, best_c5

def sa_binary(h, n_iters=20000, t_start=0.05, t_end=0.0005):
    """Simulated annealing on binary space."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    curr_c5 = best_c5
    
    rng = np.random.RandomState(12345)
    
    for it in range(n_iters):
        temp = t_start * (t_end / t_start) ** (it / max(n_iters, 1))
        
        # Pick random 1 and random 0 to swap
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h[i] = 0
        h[j] = 1
        new_c5 = compute_c5(h, dx)
        
        delta = new_c5 - curr_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-10)):
            curr_c5 = new_c5
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = h.copy()
        else:
            h[i] = 1
            h[j] = 0
    
    return best_h, best_c5

def gradient_descent_smooth_max(h, target_sum, t_schedule, lr_schedule, max_iter):
    """Gradient descent on smooth max with projection."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    for it in range(max_iter):
        t = t_schedule(it)
        lr = lr_schedule(it)
        
        corr = np.correlate(h, 1 - h, mode="full")
        # Compute weights for smooth max
        max_c = np.max(corr)
        shifted = t * (corr - max_c)
        weights = np.exp(shifted)
        weights /= weights.sum()
        
        # Gradient of smooth max wrt h
        # d/dh[i] of corr[k] for each k... this is complex
        # Instead, use finite difference for the gradient
        grad = np.zeros(n)
        eps = 1e-4
        for i in range(n):
            h[i] += eps
            corr_plus = np.correlate(h, 1 - h, mode="full")
            h[i] -= 2 * eps
            corr_minus = np.correlate(h, 1 - h, mode="full")
            h[i] += eps
            grad[i] = np.dot(weights, corr_plus - corr_minus) / (2 * eps)
        
        # Project gradient onto constraint sum(h) = target_sum
        grad_proj = grad - np.mean(grad)
        
        h = h - lr * grad_proj
        # Clip and fix sum
        h = np.clip(h, 0, 1)
        diff = target_sum - np.sum(h)
        h += diff / n
        h = np.clip(h, 0, 1)
        
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

Let me use option 2: compute the delta in the correlation caused by a swap (i→0, j→1).

When h[i] goes 1→0 and h[j] goes 0→1:

The correlation at shift k (in numpy's "full" mode, k=0..2n-2) changes by:
For the "full" correlation out[k] = Σ_m h[m] * (1-h)[m-k+n-1]:
- h[i] appears as the first factor when m=i: contributes (1-h)[i-k+n-1] to out[k] (if valid)
  When h[i] goes 1→0, this contribution goes away: delta = -(1-h)[i-k+n-1]... but wait, (1-h) also changes at position j.

This gets complicated because BOTH h[i] and h[j] change, and (1-h) is the flipped version.

Let me think differently. Let's define:
out[k] = Σ_m h[m] * (1-h[m-k+n-1])

Let p = m-k+n-1 (the index into the second array).
out[k] = Σ_m h[m] * (1-h[p]) where p = m-k+n-1.

When h[i]: 1→0:
- In the sum, the term m=i: h[i]*(1-h[i-k+n-1]) goes from 1*(1-h[i-k+n-1]) to 0 = change of -(1-h[i-k+n-1])
  (if i-k+n-1 is in [0,n-1])
- Also, wherever p=i (i.e., m-k+n-1=i, m=i+k-n+1), the term h[m]*(1-h[i]) changes:
  (1-h[i]) goes 0→1, so the term goes from h[m]*0 to h[m]*1 = change of +h[m]
  (if m=i+k-n+1 is in [0,n-1])

When h[j]: 0→1:
- In the sum, the term m=j: h[j]*(1-h[j-k+n-1]) goes from 0 to 1*(1-h[j-k+n-1]) = change of +(1-h[j-k+n-1])
  (if j-k+n-1 is in [0,n-1])
- Also, wherever p=j (m=j+k-n+1), the term h[m]*(1-h[j]) changes:
  (1-h[j]) goes 1→0, so the term goes from h[m]*1 to h[m]*0 = change of -h[m]
  (if m=j+k-n+1 is in [0,n-1])

So the total change in out[k] is:
Δout[k] = [contribution from h[i] 1→0] + [contribution from h[j] 0→1]

For h[i] 1→0:
- Term at m=i: -(1-h[i-k+n-1]) if 0 ≤ i-k+n-1 ≤ n-1, i.e., k-n+1 ≤ i ≤ k, i.e., n-1-i ≤ k ≤ n-1+i
  Let's define: valid when p_i = i-k+n-1 ∈ [0,n-1], i.e., n-1-i ≤ k ≤ n-1+i
  Contribution: -(1-h[p_i]) where p_i = i-k+n-1
  
- Term at p=i (m = i+k-n+1): +h[m] if 0 ≤ i+k-n+1 ≤ n-1, i.e., n-1-i ≤ k ≤ n-1+i
  Wait, m = i+k-n+1, valid when 0 ≤ i+k-n+1 ≤ n-1, i.e., n-1-i ≤ k ≤ n-1-i+n-1 = 2n-2-i
  Hmm, that's k ∈ [n-1-i, 2n-2-i]. But k ranges from 0 to 2n-2.
  So valid when max(0, n-1-i) ≤ k ≤ min(2n-2, 2n-2-i)
  
  Actually, let me just denote: m' = i + k - (n-1). If 0 ≤ m' ≤ n-1, then the contribution is +h[m'].
  m' = i + k - n + 1, so k = m' - i + n - 1.
  For m' ∈ [0, n-1]: k ∈ [n-1-i, 2n-2-i].

For h[j] 0→1:
- Term at m=j: +(1-h[j-k+n-1]) if 0 ≤ j-k+n-1 ≤ n-1, i.e., n-1-j ≤ k ≤ n-1+j
- Term at p=j (m = j+k-n+1): -h[m] if 0 ≤ j+k-n+1 ≤ n-1, i.e., n-1-j ≤ k ≤ 2n-2-j

So Δout[k] is non-zero only for k in the union of these ranges, which is at most 4 intervals per swap.

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")

def fix_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(h, t, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    max_c = np.max(corr)
    shifted = t * (corr - max_c)
    log_sum = max_c + np.log(np.sum(np.exp(shifted))) / t
    return log_sum * dx

def solve_ilp(n, time_limit, seed_val=42):
    import cvxpy as cp
    dx = 2.0 / n
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()
    cons = [cp.sum(h) == n // 2, t >= 0]
    u_vars = {}
    for s in range(1, n):
        for i in range(n - s):
            uv = cp.Variable()
            u_vars[(i, s)] = uv
            cons += [uv <= h[i], uv <= h[i+s], uv >= h[i] + h[i+s] - 1, uv >= 0]
    for s in range(1, n):
        sum_pos = cp.sum(h[s:])
        sum_neg = cp.sum(h[:n-s])

        cons.append((sum_neg - sum_u) * dx <= t)

def slsqp_refine(h, target_sum, t, maxiter=500):
    n = len(h)
    dx = 2.0 / n
    def obj(x):
        return smooth_max_obj(x, t, dx)
    try:
        res = minimize(obj, h.copy(), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return fix_sum(res.x, target_sum)
    except Exception:
        return h

def quadratic_residue_init(n, p):
    h = np.zeros(n)

        return h
    reps = n // p
    qr = set((k * k) % p for k in range(p))
    for block in range(reps):
        for k in range(p):
            if k in qr:
                h[block * p + k] = 1.0
    return fix_sum(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    h0 = np.array(initial_h_values, dtype=float).copy()
    n0 = len(h0)
    dx0 = 2.0 / n0
    target0 = n0 / 2.0
    h0 = np.clip(h0, 0.0, 1.0)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx0)
    best_n = n0

    # Phase 1: ILP at n=80
    if elapsed() < budget_s - 700:
        h80, c580 = solve_ilp(80, time_limit=min(350, int(budget_s - 750 - elapsed())))
        if h80 is not None:
            h80, c580 = binary_greedy(h80, max_rounds=25)
            if c580 < best_c5:
                best_h, best_c5, best_n = h80, c580, 80

    # Phase 2: ILP at n=100
    if elapsed() < budget_s - 500:
        h100, c5100 = solve_ilp(100, time_limit=min(200, int(budget_s - 550 - elapsed())))
        if h100 is not None:
            h100, c5100 = binary_greedy(h100, max_rounds=15)
            if c5100 < best_c5:
                best_h, best_c5, best_n = h100, c5100, 100

    # Phase 3: ILP at n=120
    if elapsed() < budget_s - 350:
        h120, c5120 = solve_ilp(120, time_limit=min(150, int(budget_s - 400 - elapsed())))
        if h120 is not None:
            h120, c5120 = binary_greedy(h120, max_rounds=10)
            if c5120 < best_c5:
                best_h, best_c5, best_n = h120, c5120, 120

    # Phase 4: SLSQP on n=160 from initial_h_values
    if elapsed() < budget_s - 250:
        h_cur = h0.copy()
        ts = target0
        for t in [20, 50, 100, 200, 400]:
            if elapsed() > budget_s - 200:
                break
            h_new = slsqp_refine(h_cur, ts, t, maxiter=400)
            c5_new = compute_c5(h_new, dx0)
            if c5_new < best_c5:
                best_h, best_c5, best_n = h_new.copy(), c5_new, n0
            h_cur = h_new

    # Phase 5: Binary greedy on n=160
    if elapsed() < budget_s - 150:
        h_bin = np.where(h_cur >= 0.5, 1.0, 0.0)
        n_ones = int(np.sum(h_bin))
        if n_ones != n0 // 2:
            order = np.argsort(np.abs(h_cur - 0.5))
            for idx in order:
                if n_ones == n0 // 2:
                    break
                if n_ones < n0 // 2 and h_bin[idx] == 0:
                    h_bin[idx] = 1.0
                    n_ones += 1
                elif n_ones > n0 // 2 and h_bin[idx] == 1:
                    h_bin[idx] = 0.0
                    n_ones -= 1
        c5_bin = compute_c5(h_bin, dx0)
        if c5_bin < best_c5 + 0.05:
            h_bin, c5_bin = binary_greedy(h_bin, max_rounds=15)
            if c5_bin < best_c5:
                best_h, best_c5, best_n = h_bin, c5_bin, n0

    # Phase 6: SA on best binary solution
    if elapsed() < budget_s - 80:
        n_sa = best_n
        h_sa = best_h.copy()
        if np.max(h_sa - np.round(h_sa)) > 1e-6:
            h_sa = np.where(h_sa >= 0.5, 1.0, 0.0)
        n_ones = int(np.sum(h_sa))
        tgt = n_sa // 2
        if n_ones != tgt:
            order = np.argsort(np.abs(h_sa - 0.5))
            for idx in order:
                if n_ones == tgt:
                    break
                if n_ones < tgt and h_sa[idx] == 0:
                    h_sa[idx] = 1.0
                    n_ones += 1
                elif n_ones > tgt and h_sa[idx] == 1:
                    h_sa[idx] = 0.0
                    n_ones -= 1
        n_iters = min(30000, max(2000, int((budget_s - 90 - elapsed()) * 8)))
        h_sa_r, c5_sa = sa_binary(h_sa, n_iters, seed_val=seed + 7)
        if c5_sa < best_c5:
            best_h, best_c5, best_n = h_sa_r, c5_sa, n_sa

    # Phase 7: SLSQP polish on best
    if elapsed() < budget_s - 15:
        n_f = best_n
        dx_f = 2.0 / n_f
        ts_f = n_f / 2.0
        h_f = best_h.copy()
        h_f = fix_sum(h_f, ts_f)
        h_f = slsqp_refine(h_f, ts_f, t=300, maxiter=1000)
        c5_f = compute_c5(h_f, dx_f)
        if c5_f < best_c5:
            best_h, best_c5 = h_f, c5_f

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    dx_final = 2.0 / best_n
    best_c5 = compute_c5(best_h, dx_final)

    return best_h, best_c5, best_n
