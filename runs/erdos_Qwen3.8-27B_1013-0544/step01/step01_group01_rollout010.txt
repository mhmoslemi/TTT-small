def pair_step(h, j, m, dx):
    """Optimize the pair (j, m) by moving h[j] up and h[m] down (or vice versa)."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    
    # Compute net coefficient: change in corr[k] per unit t where h[j]+=t, h[m]-=t
    net_coeff = np.zeros(2*n-1)
    for k in range(2*n-1):
        lag = k - (n-1)
        c_j = 0.0
        if 0 <= j-lag <= n-1:
            c_j += (1.0 - h[j-lag])
        if 0 <= j+lag <= n-1:
            c_j += -h[j+lag]
        c_m = 0.0
        if 0 <= m-lag <= n-1:
            c_m += (1.0 - h[m-lag])
        if 0 <= m+lag <= n-1:
            c_m += -h[m+lag]
        net_coeff[k] = c_j - c_m
    
    # Now minimize max_k (corr[k] + net_coeff[k] * t) subject to bounds on t
    # t bounds: h[j]+t in [0,1] and h[m]-t in [0,1]
    t_lo = max(-h[j], h[m]-1.0)
    t_hi = min(1.0-h[j], h[m])
    if t_lo >= t_hi:
        return h
    
    # Use scipy minimize_scalar
    from scipy.optimize import minimize_scalar
    def obj_1d(t):
        vals = corr + net_coeff * t
        return np.max(vals)
    
    res = minimize_scalar(obj_1d, bounds=(t_lo, t_hi), method='bounded', maxiter=100)
    t_opt = res.x
    
    h_new = h.copy()
    h_new[j] += t_opt
    h_new[m] -= t_opt
    return h_new

    mask2 = (m + lag >= 0) & (m + lag <= n-1)
    c_m[mask2] += -h[m + lag[mask2]]
    
    return c_j - c_m

    if t_hi - t_lo < 1e-14:
        return 0.0, np.max(corr)
    best_t = 0.0
    best_obj = float(np.max(corr))
    # Try a few candidate t values via bounded minimization
    res = minimize_scalar(
        lambda t: np.max(corr + nc * t),
        bounds=(t_lo, t_hi), method='bounded', maxiter=200
    )
    if res.fun < best_obj:
        best_t = res.x
        best_obj = res.fun
    return best_t, best_obj


def coord_descent(h, n_steps, rng, n_candidates=4):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    for _ in range(n_steps):
        best_t = 0.0
        best_j, best_m = -1, -1
        best_obj = float(np.max(corr))
        for _ in range(n_candidates):
            j = rng.randint(n)
            m = rng.randint(n)
            if j == m:
                continue
            t, obj_val = pair_step(h, j, m, corr)
            if obj_val < best_obj:
                best_obj = obj_val
                best_t = t
                best_j, best_m = j, m
        if best_j >= 0:
            h[best_j] += best_t
            h[best_m] -= best_t
            corr = np.correlate(h, 1.0 - h, mode="full")
    return h


def obj_slsqp(x, n_points, beta):
    dx = 2.0 / n_points
    corr = np.correlate(x, 1.0 - x, mode="full")
    return smooth_max(corr, beta) * dx


def obj_slsqp_grad(x, n_points, beta):
    # Not implementing gradient; SLSQP will use finite differences
    return obj_slsqp(x, n_points, beta)


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_points = len(h_init)
    else:
        n_points = 70
        h_init = np.full(n_points, 0.5)

    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_h = project_h(h_init, target_sum)
    best_c5 = compute_c5(best_h)

    def time_left():
        return budget_s - (time.time() - start_time)

    bounds = Bounds(np.zeros(n_points), np.ones(n_points))

    # Build constraint (no lambda)
    def _con_fun(x):
        return x.sum() - target_sum

    def _con_jac(x):
        return np.ones(n_points)

    constraints = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]

    # Phase 1: SLSQP from parent
    beta1 = 200.0
    try:
        res = minimize(
            lambda x: obj_slsqp(x, n_points, beta1), best_h,
            method='SLSQP', bounds=bounds, constraints=constraints,
            options={'maxiter': 2000, 'ftol': 1e-14}
        )
        h_new = project_h(res.x, target_sum)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_h, best_c5 = h_new.copy(), c5
    except Exception:
        pass

    # Phase 2: Coordinate descent polish
    if time_left() > 30:
        try:
            best_h = coord_descent(best_h, 3000, rng, n_candidates=5)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    # Phase 3: Iterative diversify + optimize
    attempt = 0
    no_improve = 0
    while time_left() > 40:
        attempt += 1
        h0 = best_h.copy()
        s = attempt % 5

        if s == 0:
            scale = 0.015 + 0.04 * np.exp(-attempt / 40.0)
            n_p = max(5, n_points // 4)
            idx = rng.choice(n_points, n_p, replace=False)
            h0[idx] += rng.randn(n_p) * scale
        elif s == 1:
            n_swaps = rng.randint(2, 7)
            for _ in range(n_swaps):
                i, j = rng.choice(n_points, 2, replace=False)
                h0[i], h0[j] = h0[j], h0[i]
        elif s == 2:
            blen = rng.randint(4, max(5, n_points // 5))
            bs = rng.randint(0, max(1, n_points - blen))
            h0[bs:bs + blen] += rng.randn(blen) * 0.04
        elif s == 3:
            # Target the active lag direction
            corr = np.correlate(h0, 1.0 - h0, mode="full")
            active_k = int(np.argmax(corr))
            lag = active_k - (n_points - 1)
            perturb = np.zeros(n_points)
            for j in range(n_points):
                i1 = j + lag
                i2 = j - lag
                if 0 <= i1 < n_points:
                    perturb[j] += 0.01 * (1.0 - h0[i1])
                if 0 <= i2 < n_points:
                    perturb[j] += 0.01 * h0[i2]
            h0 += perturb
        else:
            # Sparse large perturbation
            mask = rng.random(n_points) < 0.15
            n_mask = int(mask.sum())
            if n_mask > 0:
                h0[mask] = rng.random(n_mask) * 0.4 + 0.1

        h0 = project_h(h0, target_sum)

        # SLSQP
        beta = 300.0
        try:
            res = minimize(
                lambda x: obj_slsqp(x, n_points, beta), h0,
                method='SLSQP', bounds=bounds, constraints=constraints,
                options={'maxiter': 800, 'ftol': 1e-12}
            )
            h_new = project_h(res.x, target_sum)
        except Exception:
            continue

        # Coordinate descent polish
        try:
            h_new = coord_descent(h_new, 500, rng, n_candidates=4)
        except Exception:
            pass

        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_h, best_c5 = h_new.copy(), c5
            no_improve = 0
        else:
            no_improve += 1

        # Occasionally try a different n_points
        if attempt % 8 == 0 and time_left() > 100 and n_points == 70:
            n_alt = 50
            h_alt = np.full(n_alt, 0.5)
            dx_alt = 2.0 / n_alt
            ts_alt = n_alt / 2.0
            bounds_alt = Bounds(np.zeros(n_alt), np.ones(n_alt))

            def _con_fun_alt(x):
                return x.sum() - ts_alt

            def _con_jac_alt(x):
                return np.ones(n_alt)

            constraints_alt = [{'type': 'eq', 'fun': _con_fun_alt, 'jac': _con_jac_alt}]
            try:
                res = minimize(
                    lambda x: obj_slsqp(x, n_alt, 300.0), h_alt,
                    method='SLSQP', bounds=bounds_alt, constraints=constraints_alt,
                    options={'maxiter': 2000, 'ftol': 1e-13}
                )
                h_alt = project_h(res.x, ts_alt)
                h_alt = coord_descent(h_alt, 1000, rng, n_candidates=4)
                c5_alt = compute_c5(h_alt)
                if c5_alt < best_c5:
                    best_h = h_alt.copy()
                    best_c5 = c5_alt
                    n_points = n_alt
                    target_sum = ts_alt
                    dx = dx_alt
                    bounds = bounds_alt
                    constraints = constraints_alt
            except Exception:
                pass

    # Final heavy coordinate descent polish
    if time_left() > 5:
        try:
            n_final_steps = int((time_left() - 2) * 400)
            n_final_steps = max(1000, min(n_final_steps, 20000))
            best_h = coord_descent(best_h, n_final_steps, rng, n_candidates=5)
            best_c5 = compute_c5(best_h)
        except Exception:
            pass

    return best_h, best_c5, n_points
