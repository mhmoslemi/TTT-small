import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_simplex_box(h, target_sum):
    """Project onto {x : 0 <= x[i] <= 1, sum(x) = target_sum}"""
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_smooth_objective_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    m = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m))
    s = np.sum(exp_vals)
    w = exp_vals / s
    obj = (m + np.log(s) / alpha) * (2.0 / n)
    
    # Compute gradient
    # For each lag position idx in "full" correlation (0 to 2n-2), 
    # the actual lag k = idx - (n-1)
    grad = np.zeros(n)
    active = np.where(w > 1e-12)[0]
    for idx in active:
        k = idx - (n - 1)
        wi = w[idx]
        if k >= 0:
            # corr contribution: sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])
            # d/dh[j]:
            #   j in [0, n-1-k]: += (1-h[j+k])
            #   j in [k, n-1]:   += -h[j-k]
            vec = wi * (2.0 / n)
            grad[:n-k] += vec * (1.0 - h[k:])
            if k > 0:
                grad[k:] -= vec * h[:n-k]
        else:

            vec = wi * (2.0 / n)
            grad[p:] += vec * (1.0 - h[:n-p])
            if p > 0:
                grad[:n-p] -= vec * h[p:]

    best_c5 = compute_c5(best_h)

        v = beta2 * v + (1 - beta2) * grad**2

import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_simplex_box(h, target_sum):
    lo, hi = -2.0, 2.0

        if s > target_sum:

        else:
            hi = mid

def compute_smooth_grad(h, alpha, top_k=80):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    m = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m))
    s = np.sum(exp_vals)
    w = exp_vals / s
    
    # Top-k by weight
    if top_k < len(w):
        idx = np.argpartition(w, -top_k)[-top_k:]
    else:
        idx = np.arange(len(w))

            if p > 0:
                grad[:n-p] -= wi * h[p:]
    
    grad *= (2.0 / n)
    return grad

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    v = np.zeros(n)
    t = 0
    lr = lr_init
    alpha = alpha_init
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    patience = 0
    stall_count = 0
    best_patience = 0
    
    while time.time() < deadline:
        t += 1
        grad = compute_smooth_grad(h, alpha)
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step
        h_new = project_simplex_box(h_new, target_sum)
        
        new_c5 = compute_c5(h_new)
        h = h_new

            best_h = h.copy()
            best_c5 = new_c5

            alpha = min(alpha * 1.01, 300.0)
        else:
            patience += 1
            if patience > 30:
                lr *= 0.95

                patience = 0

                    break

    x_ext = np.concatenate([x_old, [2.0 + 1.0 / n_old]])
    return np.interp(x_new, x_ext, h_ext)

        h_interp = interpolate_to_n(best_h, best_n, n_try)

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    return np.max(np.correlate(h, 1-h, mode='full')) * (2.0/n)

def project(h, ts):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo+hi)*0.5
        if np.sum(np.clip(h-mid, 0, 1)) > ts: lo = mid
        else: hi = mid
    return np.clip(h-(lo+hi)*0.5, 0, 1)

def sgrad(h, alpha, tk=100):
    n = len(h)
    c = np.correlate(h, 1-h, mode='full')
    m = c.max()
    e = np.exp(alpha*(c-m))
    w = e/e.sum()
    idx = np.argpartition(w, -tk)[-tk:] if tk < len(w) else np.arange(len(w))
    idx = idx[w[idx] > 1e-10]
    g = np.zeros(n)
    for ii in idx:
        wi = w[ii]; k = ii-(n-1)
        if k >= 0:
            g[:n-k] += wi*(1-h[k:])
            if k > 0: g[k:] -= wi*h[:n-k]
        else:
            p = -k; g[p:] += wi*(1-h[:n-p])
            if p > 0: g[:n-p] -= wi*h[p:]
    return g*(2.0/n)

def optimize(h0, tgt, dl, rng, a0=30, lr0=0.01):
    n = len(h0)
    h = project(h0.astype(float), tgt)
    bh, bc = h.copy(), compute_c5(h)
    M = np.zeros(n); V = np.zeros(n); t = 0
    b1, b2, ep = 0.9, 0.999, 1e-8
    lr, al = lr0, a0
    pat = 0; st = 0
    while time.time() < dl:
        t += 1
        g = sgrad(h, al)
        M = b1*M+(1-b1)*g; V = b2*V+(1-b2)*g*g
        h = project(h - lr*(M/(1-b1**t))/(np.sqrt(V/(1-b2**t))+ep), tgt)
        c = compute_c5(h)
        if c < bc - 1e-13:
            bh, bc, pat = h.copy(), c, 0
            al = min(al*1.02, 500)
        else:
            pat += 1
            if pat > 40: lr *= 0.94
            if pat > 180:
                sz = 0.02+0.02*st
                h = project(bh+rng.randn(n)*sz, tgt)
                M, V, t = np.zeros(n), np.zeros(n), 0
                lr, al, pat, st = lr0, max(20, al*0.5), 0, st+1
                if st > 12: break
    return bh, bc

def interp(h, no, nn):
    xo = np.linspace(0,2,no,endpoint=False)+1/no
    xn = np.linspace(0,2,nn,endpoint=False)+1/nn
    return np.interp(xn, np.append(xo,2+1/no), np.append(h,h[0]))

def run(seed=42, budget_s=1100, **kwargs):

    rng = np.random.RandomState(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        hp = np.array(initial_h_values, dtype=float)
    else:
        hp = 0.5+0.3*rng.randn(200)
    n = len(hp); tgt = n/2.0

    bh, bc, bn = hp.copy(), compute_c5(hp), n
    
    # Main optimization
    d = t0 + budget_s*0.38
    ho, co = optimize(hp, tgt, d, rng)
    if co < bc: bh, bc, bn = ho, co, len(ho)
    
    # Restart from best with perturbation
    rem = budget_s - (time.time()-t0)
    if rem > 60:
        hs = project(bh + 0.04*rng.randn(bn), bh.sum())
        d = time.time() + min(rem*0.2, 180)
        ho, co = optimize(hs, bn/2.0, d, rng, a0=50, lr0=0.008)
        if co < bc: bh, bc = ho, co
    
    # Try different resolutions
    for nt in [120, 160, 240]:
        rem = budget_s - (time.time()-t0)
        if rem < 80: break
        hi = project(interp(bh, bn, nt), nt/2.0)
        d = time.time() + min(rem*0.25, 200)
        ho, co = optimize(hi, nt/2.0, d, rng)
        if co < bc: bh, bc, bn = ho, co, nt
    
    # Final polish
    rem = budget_s - (time.time()-t0)
    if rem > 25:
        ho, co = optimize(bh, bn/2.0, t0+budget_s-3, rng, a0=200, lr0=0.002)
        if co < bc: bh, bc = ho, co
    
    bh = project(bh, bn/2.0)
    bc = compute_c5(bh)
    return bh, bc, bn

    def obj(h):

        m = corr.max()
        sm = m + np.log(np.sum(np.exp(alpha*(corr-m))))/alpha
        pen = penalty * (h.sum() - target_sum)**2
        return sm * (2.0/n) + pen
    
    def grads(h):

        corr = np.correlate(h, 1-h, mode='full')
        m = corr.max()
        e = np.exp(alpha*(corr-m))
        w = e/e.sum()

    n = len(h)

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    return np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n)

def project(h, ts):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(h - mid, 0, 1)) > ts:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def sgrad(h, alpha, tk=100):
    n = len(h)
    c = np.correlate(h, 1 - h, mode='full')
    m = c.max()
    e = np.exp(alpha * (c - m))
    w = e / e.sum()
    if tk < len(w):
        idx = np.argpartition(w, -tk)[-tk:]
    else:
        idx = np.arange(len(w))
    idx = idx[w[idx] > 1e-10]
    g = np.zeros(n)
    for ii in idx:
        wi = w[ii]
        k = ii - (n - 1)
        if k >= 0:
            g[:n-k] += wi * (1 - h[k:])
            if k > 0:
                g[k:] -= wi * h[:n-k]
        else:
            p = -k
            g[p:] += wi * (1 - h[:n-p])
            if p > 0:
                g[:n-p] -= wi * h[p:]
    return g * (2.0 / n)

def active_grad(h, tol=1e-8):
    n = len(h)
    c = np.correlate(h, 1 - h, mode='full')
    mx = c.max()
    active = np.where(c >= mx - tol)[0]
    g = np.zeros(n)
    for ii in active:
        k = ii - (n - 1)
        if k >= 0:
            g[:n-k] += (1 - h[k:])
            if k > 0:
                g[k:] -= h[:n-k]
        else:
            p = -k
            g[p:] += (1 - h[:n-p])
            if p > 0:
                g[:n-p] -= h[p:]
    g /= max(len(active), 1)
    return g * (2.0 / n)

def adam_opt(h0, tgt, dl, rng, a0=30, lr0=0.01, alpha_max=500):
    n = len(h0)
    h = project(h0.astype(float), tgt)
    bh, bc = h.copy(), compute_c5(h)
    M = np.zeros(n)
    V = np.zeros(n)
    t = 0
    b1, b2, ep = 0.9, 0.999, 1e-8
    lr, al = lr0, a0
    pat = 0
    st = 0
    while time.time() < dl:
        t += 1
        g = sgrad(h, al)
        M = b1 * M + (1 - b1) * g
        V = b2 * V + (1 - b2) * g * g
        mh = M / (1 - b1**t)
        vh = V / (1 - b2**t)
        h = project(h - lr * mh / (np.sqrt(vh) + ep), tgt)
        c = compute_c5(h)
        if c < bc - 1e-13:
            bh, bc, pat = h.copy(), c, 0
            al = min(al * 1.02, alpha_max)
        else:
            pat += 1
            if pat > 40:
                lr *= 0.94
            if pat > 180:
                sz = 0.02 + 0.015 * st
                h = project(bh + rng.randn(n) * sz, tgt)
                M, V, t = np.zeros(n), np.zeros(n), 0
                lr, al, pat = lr0, max(20, al * 0.5), 0
                st += 1
                if st > 15:
                    break
    return bh, bc

def active_opt(h0, tgt, dl, rng, lr0=0.002):

    bh, bc = h.copy(), compute_c5(h)

def interp(h, no, nn):
    xo = np.linspace(0, 2, no, endpoint=False) + 1.0 / no
    xn = np.linspace(0, 2, nn, endpoint=False) + 1.0 / nn
    he = np.append(h, h[0])
    xe = np.append(xo, 2.0 + 1.0 / no)
    return np.interp(xn, xe, he)

def local_search(h, tgt, rng, n_iters=200):
    """Try small coordinate perturbations."""
    n = len(h)
    bh, bc = h.copy(), compute_c5(h)
    for _ in range(n_iters):

        delta = rng.choice([-0.05, -0.02, -0.01, 0.01, 0.02, 0.05])

        h_try = project(h_try, tgt)
        c_try = compute_c5(h_try)
        if c_try < bc - 1e-14:
            bh, bc = h_try, c_try
    return bh, bc
