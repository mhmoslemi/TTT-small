    for k in range(len(vals)):
        if w[k] < 1e-10:
            continue
        L = int(k) - n_half
        if L == 0:
            grad += w[k] * (omh - x)
        elif L > 0:
            grad[L:] += w[k] * omh[:n - L]
            grad[:n - L] -= w[k] * x[L:]
        else:
            L2 = -L
            grad[:n - L2] += w[k] * omh[L2:]
            grad[L2:] -= w[k] * x[:n - L2]
    if mu > 1e-12:
        grad += 2.0 * mu * (float(np.sum(x)) - _g_target)
    return grad

def _exact_obj(x):
    return compute_c5(x, _g_dx)

def _eq_fun(x):
    return float(np.sum(x)) - _g_target

def _eq_jac(x):
    return np.ones(_g_n)

def _grad_subgrad(h):
    n = len(h)
    n_half = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * _g_dx
    idx = int(np.argmax(vals))
    L = idx - n_half
    omh = 1.0 - h
    grad = np.zeros(n)
    if L == 0:
        grad[:] = omh - h
    elif L > 0:
        grad[L:] += omh[:n - L]
        grad[:n - L] -= h[L:]
    else:
        L2 = -L
        grad[:n - L2] += omh[L2:]
        grad[L2:] -= h[:n - L2]
    return grad

def _local_refine(h, target_sum, budget_fn, max_iter=200):
    """Gradient polish from a starting point."""
    c5_curr = compute_c5(h, _g_dx)
    step = 0.0005
    mom = np.zeros(len(h))
    for _ in range(max_iter):
        if budget_fn() < 5.0:
            break
        g = _grad_subgrad(h)
        mom = 0.5 * mom - step * g
        h_new = project_feasible(h + mom, target_sum)
        c5_new = compute_c5(h_new, _g_dx)
        if c5_new < c5_curr:
            h = h_new
            c5_curr = c5_new
            step = min(step * 1.05, 0.003)
        else:
            step *= 0.8
            mom *= 0.5
            if step < 1e-12:
                break
    return h, c5_curr

def run(seed=42, budget_s=1100, **kwargs):
    global _g_dx, _g_target, _g_n, _g_T, _g_mu
    start_time = time.time()
    rng = np.random.RandomState(seed)

    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    _g_dx = dx
    _g_target = target_sum
    _g_n = n_points

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except NameError:
        h0 = np.full(n_points, 0.5)

    h0 = project_feasible(h0, target_sum)
    best_h = h0.copy()
    best_c5 = compute_c5(h0, dx)

    def tleft():
        return (start_time + budget_s - 3.0) - time.time()

    # Phase 1: L-BFGS-B with smooth max, temperature annealing
    for T_val in [5, 10, 25, 50, 100, 200, 400, 800]:
        if tleft() < 100:
            break
        _g_T = T_val
        _g_mu = 50.0
        try:
            res = minimize(_smooth_obj, best_h, jac=_smooth_grad,
                         method='L-BFGS-B',
                         bounds=[(0.0, 1.0)] * n_points,
                         options={'maxiter': 800, 'ftol': 1e-14, 'gtol': 1e-8,
                                  'maxfun': 2000})
            if res.x is not None:
                h_ref = project_feasible(res.x, target_sum)
                c5_ref = compute_c5(h_ref, dx)
                if c5_ref < best_c5:
                    best_c5 = c5_ref
                    best_h = h_ref
        except Exception:
            pass

    # Phase 2: SLSQP with exact objective
    if tleft() > 80:
        try:
            res = minimize(_exact_obj, best_h, method='SLSQP',
                          bounds=[(0.0, 1.0)] * n_points,
                          constraints=[{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}],
                          options={'maxiter': 8000, 'ftol': 1e-15})
            if res.x is not None:
                h_ref = project_feasible(res.x, target_sum)
                c5_ref = compute_c5(h_ref, dx)
                if c5_ref < best_c5:
                    best_c5 = c5_ref
                    best_h = h_ref
        except Exception:
            pass

    # Phase 3: Simulated annealing with local refinement
    if tleft() > 60:
        h_sa = best_h.copy()
        c5_sa = best_c5
        T_sa = 0.0002
        sigma = 0.006
        for it in range(50000):
            if tleft() < 20:
                break
            h_pert = h_sa + rng.normal(0, sigma, n_points)
            h_pert = project_feasible(h_pert, target_sum)
            c5_pert = compute_c5(h_pert, dx)

            accept = False
            if c5_pert < c5_sa:
                accept = True
            else:
                delta = c5_pert - c5_sa
                if delta < 5.0 * T_sa and rng.random() < np.exp(-delta / T_sa):
                    accept = True

            if accept:
                h_sa = h_pert
                c5_sa = c5_pert
                if c5_sa < best_c5:
                    best_c5 = c5_sa
                    best_h = h_sa.copy()
                    h_sa, c5_sa = _local_refine(h_sa, target_sum, tleft, max_iter=150)
                    if c5_sa < best_c5:
                        best_c5 = c5_sa
                        best_h = h_sa.copy()
            else:
                h_sa = h_pert
                c5_sa = c5_pert

            T_sa *= 0.99985
            sigma = max(0.002, 0.006 * (T_sa / 0.0002))

    # Phase 4: Multiple SLSQP restarts from perturbed best
    if tleft() > 40:
        for restart in range(5):
            if tleft() < 30:
                break
            h_init = best_h + rng.normal(0, 0.003, n_points)
            h_init = project_feasible(h_init, target_sum)
            try:
                res = minimize(_exact_obj, h_init, method='SLSQP',
                              bounds=[(0.0, 1.0)] * n_points,
                              constraints=[{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}],
                              options={'maxiter': 3000, 'ftol': 1e-15})
                if res.x is not None:
                    h_ref = project_feasible(res.x, target_sum)
                    c5_ref = compute_c5(h_ref, dx)
                    if c5_ref < best_c5:
                        best_c5 = c5_ref
                        best_h = h_ref
            except Exception:
                pass

    # Phase 5: Final fine gradient polish
    h_fin = best_h.copy()
    step = 0.0001
    for _ in range(20000):
        if tleft() < 1.0:
            break
        g = _grad_subgrad(h_fin)
        h_new = project_feasible(h_fin - step * g, target_sum)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            step = min(step * 1.03, 0.001)
        else:
            step *= 0.85
            if step < 1e-13:
                break
        h_fin = h_new

    return (best_h, best_c5, n_points)
