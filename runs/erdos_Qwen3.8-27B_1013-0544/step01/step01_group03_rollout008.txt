def fix_sum(h, target):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # Project onto sum constraint within [0,1]
    # Simple approach: add diff uniformly to non-saturated elements
    for _ in range(10):
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        if mask.sum() == 0:
            break
        step = diff / mask.sum()
        # Don't overshoot
        if diff > 0:
            max_step = (1.0 - h[mask]).min()
            step = min(step, max_step)
        else:
            min_step = h[mask].min()
            step = max(step, -min_step)
        h[mask] += step
        diff = target - h.sum()
        if abs(diff) < 1e-10:
            break
    return np.clip(h, 0.0, 1.0)

    m = np.max(x)
    return m + np.log(np.sum(np.exp(x - m))) / tau

    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    # Softmax probabilities
    shifted = tau * corr
    m = np.max(shifted)
    exp_shifted = np.exp(shifted - m)
    p = exp_shifted / np.sum(exp_shifted)

    # For each shift k (indexed as k in range [-(n-1), n-1]):
    # corr index = n-1+k

def evaluate_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if mask.sum() == 0:
            break
        step = diff / mask.sum()
        if diff > 0:
            step = min(step, (1.0 - h[mask]).min() - 1e-12)
        else:
            step = max(step, -(h[mask].min() - 1e-12))
        if abs(step) < 1e-15:
            break
        h[mask] += step
    return np.clip(h, 0.0, 1.0)

def soft_max_val(x, tau):
    m = np.max(x)
    return float(m + np.log(np.sum(np.exp(x - m))) / tau)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def smooth_max_grad(h, tau):
    n = len(h)
    corr = compute_corr(h)
    shifted = tau * corr
    m = np.max(shifted)
    exp_s = np.exp(shifted - m)
    p = exp_s / np.sum(exp_s)
    grad = np.zeros(n)
    for k in range(-(n-1), n):
        pk = p[n - 1 + k]
        if pk < 1e-14:
            continue
        jlo = max(0, k)

            grad[jlo:jhi] += pk * (1.0 - h[jlo - k:jhi - k])

                best_h, best_c5 = h_new, c5_new
            corr_f = compute_corr(h_new)
            p_f = np.exp(10.0 * corr_f)
            p_f /= p_f.sum()

    # Phase 7: Very high tau for final sharpening
    if rem() > 10:
        h_new = optimize_smooth_max(best_h, 5000, 5.0, 200)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Final fix
    best_h = fix_sum(best_h, len(best_h) / 2.0)
    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, len(best_h)
