Key insight: The previous approach uses SLSQP which handles the max function by numerical differentiation. The issue is that the max function is non-smooth, and numerical differentiation of a non-smooth function can be inaccurate, leading to poor convergence.

**Better approach: Subgradient method with projection.**

At each iteration:
1. Compute all g_k(h) values
2. Find k* = argmax g_k(h)
3. Compute the gradient of g_{k*}(h) (which is smooth)
4. Take a step in the negative gradient direction
5. Project back to the feasible set (box + sum constraint)

This is a projected subgradient method. It's simple, robust, and can be made to converge. The step size can be adaptively chosen.

def subgradient_step(h, dx, step_size, n):
    """One step of projected subgradient method."""
    # Compute all overlap values
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    k_star = np.argmax(corr)
    
    # Compute gradient of g_{k_star}(h)
    # g_k(h) = sum over valid (i,j) pairs where j-i = k of h[i]*(1-h[j]) * dx
    # d/dh[i] g_k = dx * sum_j [(1-h[j]) if j = i+k, -h[j] if i = j+k]

    # Numerical gradient of the max term:
    # Use central differences on the active term
    grad = np.zeros(n)
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] = min(1.0, h[i] + 1e-5)
        h_minus = h.copy()
        h_minus[i] = max(0.0, h[i] - 1e-5)
        # Only need the active term
        corr_p = np.correlate(h_plus, 1.0 - h_plus, mode='full')[k_star] * dx
        corr_m = np.correlate(h_minus, 1.0 - h_minus, mode='full')[k_star] * dx

    # Gradient step
    h_new = h - step_size * grad
    
    # Project to feasible set
    h_new = project_feasible(h_new, n)
    return h_new

2. Sample λ candidates: x_i = m + σ * N(0, C)
3. Evaluate fitness
4. Update m using the best μ candidates
5. Update C based on the sampled points
6. Update σ

    corr = np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = np.sum(h) - target
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)

        else:
            h -= diff / n

def gradient_active_term(h, dx, k, n):
    """Gradient of the k-th correlation term."""
    offset = k - (n - 1)
    grad = np.zeros(n)
    # First factor: (1-h[i+offset]) for valid i
    i1s = max(0, -offset)
    i1e = min(n-1, n-1-offset)
    if i1s <= i1e:
        grad[i1s:i1e+1] += (1.0 - h[i1s+offset:i1e+offset]) * dx
    # Second factor: -h[i-offset] for valid i  
    i2s = max(0, offset)
    i2e = min(n-1, n-1+offset)
    if i2s <= i2e:
        grad[i2s:i2e+1] -= h[i2s-offset:i2e-offset] * dx
    return grad

def subgradient_direction(h, dx, n, top_k=3):
    """Weighted subgradient using top-k active terms."""
    corr = np.correlate(h, 1.0 - h, mode='full') * dx

    top_indices = np.argsort(corr)[-top_k:]

    weights /= np.sum(weights)
    
    grad = np.zeros(n)

        grad += weights[i] * gradient_active_term(h, dx, k, n)

    grad = np.zeros(n)
    for i in range(top_k):
        grad += w[i] * gradient_active_term(h, dx, int(top_idx[i]), n)
    return grad, float(np.max(corr))

def slsqp_optimize(h_start, n, dx, target_sum, max_iter=500, ftol=1e-14):
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        return compute_c5(x, dx)
    def con(x):
        return np.sum(x) - target_sum
    try:
        res = minimize(obj, h_start, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': max_iter, 'ftol': ftol})
        h_out = res.x
    except Exception:
        h_out = h_start.copy()
    return project_feasible(h_out, n)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - start_time

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0

    best_h = project_feasible(h0.copy(), n)
    best_c5 = compute_c5(best_h, dx)

    # Phase 1: Subgradient-guided SLSQP at native resolution (50% budget)
    phase1_end = start_time + budget_s * 0.50
    att = 0
    while elapsed() < phase1_end:
        if att == 0:
            h_try = best_h.copy()
        else:
            sg, _ = subgradient_direction(best_h, dx, n, top_k=4)
            sg_norm = np.linalg.norm(sg)
            if sg_norm > 1e-10:
                sg_unit = sg / sg_norm
            else:
                sg_unit = rng.randn(n)
                sg_unit /= np.linalg.norm(sg_unit)

            r = rng.rand()
            if r < 0.35:
                scale = rng.uniform(0.03, 0.18)
                h_try = best_h - scale * sg_unit * np.sqrt(n)
            elif r < 0.6:
                scale = rng.uniform(0.02, 0.1)
                h_try = best_h - scale * sg_unit * np.sqrt(n) + rng.randn(n) * 0.06
            elif r < 0.8:
                h_try = best_h.copy()
                n_blks = rng.randint(1, 4)
                for _ in range(n_blks):
                    blk = rng.randint(3, max(4, n // 4))
                    st = rng.randint(0, n - blk)
                    h_try[st:st + blk] += rng.randn(blk) * 0.15
            elif r < 0.92:
                scale = rng.uniform(0.08, 0.25)
                h_try = best_h + rng.randn(n) * scale
            else:
                h_try = 0.5 + rng.randn(n) * 0.25

            h_try = project_feasible(h_try, n)

        h_new = slsqp_optimize(h_try, n, dx, target, max_iter=500, ftol=1e-13)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5 - 1e-15:
            best_h = h_new
            best_c5 = c5_new
        att += 1

    # Phase 2: Higher resolution (30% budget)
    if elapsed() < start_time + budget_s * 0.80:
        for factor in [2, 3]:
            if elapsed() >= start_time + budget_s * 0.80:
                break
            n2 = factor * n
            if n2 > 500:
                continue
            dx2 = 2.0 / n2
            target2 = n2 / 2.0
            x_old = np.linspace(0, 2, n, endpoint=False)
            x_new = np.linspace(0, 2, n2, endpoint=False)
            h_up = np.interp(x_new, x_old, best_h)
            h_up = project_feasible(h_up, n2)
            h_up = slsqp_optimize(h_up, n2, dx2, target2, max_iter=300)
            c5_up = compute_c5(h_up, dx2)

            max_time_phase2 = (start_time + budget_s * 0.80 - elapsed()) / 2.0
            phase2_end = elapsed() + max(min(max_time_phase2, 120), 10)
            att2 = 0
            while elapsed() < phase2_end and att2 < 40:
                sg2, _ = subgradient_direction(h_up, dx2, n2, top_k=5)
                sg2_norm = np.linalg.norm(sg2)
                if sg2_norm > 1e-10:
                    sg2_unit = sg2 / sg2_norm
                else:
                    sg2_unit = rng.randn(n2)
                    sg2_unit /= (np.linalg.norm(sg2_unit) + 1e-15)
                scale = rng.uniform(0.02, 0.1)
                h_try2 = h_up - scale * sg2_unit * np.sqrt(n2) + rng.randn(n2) * 0.03
                h_try2 = project_feasible(h_try2, n2)
                h_new2 = slsqp_optimize(h_try2, n2, dx2, target2, max_iter=400)
                c5_new2 = compute_c5(h_new2, dx2)
                if c5_new2 < best_c5 - 1e-15:
                    best_h = h_new2
                    best_c5 = c5_new2
                    h_up = h_new2.copy()
                elif c5_new2 < c5_up - 1e-15:
                    h_up = h_new2.copy()
                    c5_up = c5_new2
                att2 += 1

    # Phase 3: Final intensive polishing (remaining budget)
    n_f = len(best_h)
    dx_f = 2.0 / n_f
    target_f = n_f / 2.0

    # Long polish on current best
    h_pol = slsqp_optimize(best_h, n_f, dx_f, target_f, max_iter=3000, ftol=1e-16)
    c5_pol = compute_c5(h_pol, dx_f)
    if c5_pol < best_c5:
        best_h = h_pol
        best_c5 = c5_pol

    phase3_end = start_time + budget_s - 20
    att3 = 0
    while elapsed() < phase3_end:
        if att3 == 0:
            h_new3 = slsqp_optimize(best_h, n_f, dx_f, target_f, max_iter=2000, ftol=1e-16)
        else:
            sg3, _ = subgradient_direction(best_h, dx_f, n_f, top_k=5)
            sg3_norm = np.linalg.norm(sg3)
            if sg3_norm > 1e-10:
                h_try3 = best_h - 0.04 * sg3 + rng.randn(n_f) * 0.015
            else:
                h_try3 = best_h + rng.randn(n_f) * 0.02
            h_try3 = project_feasible(h_try3, n_f)
            h_new3 = slsqp_optimize(h_try3, n_f, dx_f, target_f, max_iter=800, ftol=1e-15)

        c5_new3 = compute_c5(h_new3, dx_f)
        if c5_new3 < best_c5 - 1e-16:
            best_h = h_new3
            best_c5 = c5_new3
        att3 += 1

    return (best_h, best_c5, n_f)
