def project_feasible(h, target_sum):
    """Project h onto {x: 0<=x<=1, sum(x)=target_sum}"""
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    current_sum = h.sum()
    diff = current_sum - target_sum
    
    if abs(diff) < 1e-15:
        return h
    
    if diff > 0:
        # Need to decrease sum by diff
        # Sort by value descending, decrease from top
        order = np.argsort(-h)
        remaining = diff
        for i in order:
            dec = min(h[i], remaining)
            h[i] -= dec
            remaining -= dec
            if remaining <= 1e-15:
                break
    else:

        # Sort by value ascending, increase from bottom
        order = np.argsort(h)
        remaining = -diff
        for i in order:
            inc = min(1.0 - h[i], remaining)
            h[i] += inc
            remaining -= inc
            if remaining <= 1e-15:
                break
    
    return h

        return h_new, c5_new

        return h_best.copy(), compute_c5(h_best)


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 12

    best_h = None
    best_c5 = float('inf')

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
        target = n_parent / 2.0
        h_parent = project_feasible(h_parent, target)
        best_h = h_parent.copy()
        best_c5 = compute_c5(h_parent)

    n_options = [70, 50, 100, 140]
    if best_h is not None and len(best_h) not in n_options:
        n_options = [len(best_h)] + n_options

    for n in n_options:
        if time.time() > deadline - 80:
            break
        target_sum = n / 2.0
        inits_per_n = 5

        for ia in range(inits_per_n):
            if time.time() > deadline - 40:
                break

            if ia == 0 and best_h is not None and len(best_h) == n:
                h0 = best_h + np.random.randn(n) * 0.03
                h0 = project_feasible(h0, target_sum)
            elif ia == 1:
                h0 = cosine_init(n, 1, 0.49)
            elif ia == 2:
                h0 = cosine_init(n, 2, 0.42)
            elif ia == 3:
                h0 = two_tone_init(n, 1, 0.28, 3, 0.22)
            elif ia == 4:
                h0 = three_tone_init(n, 1, 0.2, 2, 0.2, 5, 0.15)
            else:
                h0 = 0.5 + 0.35 * np.random.randn(n)
                h0 = project_feasible(h0, target_sum)

            h_opt, c5_opt = subgradient_optimize(h0, n, target_sum, deadline)
            if c5_opt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_opt

    if time.time() < deadline - 5 and best_h is not None:
        h_new, c5_new = slsqp_polish(best_h, deadline)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    if time.time() < deadline and best_h is not None:
        n = len(best_h)
        target_sum = n / 2.0
        for _ in range(3):
            if time.time() > deadline:
                break
            h_small = best_h + np.random.randn(n) * 0.01
            h_small = project_feasible(h_small, target_sum)
            h_opt, c5_opt = subgradient_optimize(h_small, n, target_sum, deadline, max_iter=5000)
            if c5_opt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_opt
        if time.time() < deadline:
            h_new, c5_new = slsqp_polish(best_h, deadline)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
