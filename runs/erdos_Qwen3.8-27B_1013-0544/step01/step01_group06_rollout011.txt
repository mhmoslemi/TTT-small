```python
import numpy as np
import time

def fft_correlate(a, b):
    """Fast cross-correlation using FFT. Returns np.correlate(a,b,mode='full')."""
    n, m = len(a), len(b)
    N = n + m - 1
    Nfft = 1 << (N - 1).bit_length()
    fa = np.fft.rfft(a, Nfft)
    fb = np.fft.rfft(b[::-1], Nfft)
    return np.fft.irfft(fa * fb, Nfft)[:N]

def compute_c5(h):
    n = len(h)
    corr = fft_correlate(h, 1.0 - h)
    return float(np.max(corr)) * (2.0 / n)

def project(h, target):
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(h - mid, 0, 1)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo+hi), 0, 1)

def subgrad_at(h, w):
    """Compute sum_l w[l] * d(corr_full[l])/dh"""
    n = len(h)
    grad = np.zeros(n)
    active = np.where(w > 1e-8)[0]
    for idx in active:
        s = idx - (n - 1)  # shift in [-(n-1), n-1]
        wv = w[idx]
        # term 1: (1-h[j-s]) for valid j-s
        # j-s in [0,n-1] means s <= j <= n-1+s
        j_start = max(0, s)
        j_end = min(n, n + s)
        if j_start < j_end:
            grad[j_start:j_end] += wv * (1.0 - h[j_start-s:j_end-s])
        # term 2: -h[j+s] for valid j+s
        # j+s in [0,n-1] means -s <= j <= n-1-s
        j_start2 = max(0, -s)
        j_end2 = min(n, n - s)
        if j_start2 < j_end2:
            grad[j_start2:j_end2] -= wv * h[j_start2+s:j_end2+s]
    return grad * (2.0 / n)
```
