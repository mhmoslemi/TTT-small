        h_up = project_simplex(h_up, n_up / 2.0)
        d3 = t0 + budget_s * 0.90
        h_res, c5_res = nesterov_optimize(h_up, n_up / 2.0, d3, rng, top_k=3, alpha=80)
        results.append((h_res, c5_res, n_up))
        if c5_res < best_c5:
            best_h, best_c5 = h_res, c5_res

    remaining = budget_s - (time.time() - t0) - 3

    # Phase 4: Final polish
    if best_h is not None and remaining > 10:
        n_final = len(best_h)
        best_h, best_c5 = nesterov_optimize(best_h, n_final / 2.0,
                                            t0 + budget_s - 2, rng, top_k=3, alpha=100)

    if best_h is None:
        # Fallback: random
        n = 200
        best_h = project_simplex(0.5 + 0.1 * rng.randn(n), n / 2.0)
        best_c5 = compute_c5(best_h)

    # Final feasibility check
    n_out = len(best_h)
    best_h = project_simplex(best_h, n_out / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n_out
