```python
from scipy.optimize import minimize, Bounds
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max(obj, tau):
    x = tau * obj
    x_max = np.max(x)
    return float(x_max / tau + np.log(np.sum(np.exp(x - x_max))) / tau)

def project(h, target):
    """Project h onto {sum(h) = target, 0 <= h <= 1}."""
    h = h.copy()
    # Binary search for lambda
    lo, hi = -h.max() - 1, -h.min() + 1
    for _ in range(60):
        mid = (lo + hi) / 2
        projected = np.clip(h - mid, 0, 1)
        s = projected.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def subgradient_for_lag(h, k_idx, n):
    """Subgradient of the k_idx-th correlation value w.r.t. h."""
    lag = (n - 1) - k_idx
    grad = np.zeros(n)
    k = abs(lag)
    if k == 0:
        return 1.0 - 2.0 * h
    if lag > 0:
        # c = sum_{i=k}^{n-1} h[i]*(1-h[i-k])
        idx1 = np.arange(k, n)
        idx2 = np.arange(0, n - k)
        grad[idx1] += (1.0 - h[idx2])
        grad[idx2] -= h[idx1]

        idx2 = np.arange(k, n)
        grad[idx1] += (1.0 - h[idx2])
        grad[idx2] -= h[idx1]
    return grad
```

```python
from scipy.optimize import minimize, Bounds
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max_val(corr, tau):
    x = tau * corr
    x_max = np.max(x)
    return float((x_max + np.log(np.sum(np.exp(x - x_max)))) / tau)

def smooth_max_val_and_weights(corr, tau):
    x = tau * corr
    x_max = np.max(x)
    shifted = x - x_max
    exp_s = np.exp(shifted)
    val = (x_max + np.log(exp_s.sum())) / tau
    w = exp_s / exp_s.sum()
    return val, w

def lag_subgrad(h, k_idx, n):
    lag = (n - 1) - k_idx
    k = abs(lag)
    if k == 0:
        return 1.0 - 2.0 * h
    g = np.zeros(n)
    if lag > 0:
        g[k:] += (1.0 - h[:n-k])
        g[:n-k] -= h[k:]
    else:
        g[:n-k] += (1.0 - h[k:])
        g[k:] -= h[:n-k]
    return g

    h = h.copy()
    lo, hi = -h.max() - 2.0, -h.min() + 2.0

        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
```

```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max_val(corr, tau):
    x = tau * corr
    x_max = np.max(x)
    return float((x_max + np.log(np.sum(np.exp(x - x_max)))) / tau)

def smooth_max_weights(corr, tau):
    x = tau * corr
    x_max = np.max(x)
    shifted = x - x_max
    e = np.exp(shifted)
    return e / e.sum()

def lag_subgrad(h, k_idx, n):
    lag = (n - 1) - k_idx
    k = abs(lag)
    if k == 0:
        return 1.0 - 2.0 * h
    g = np.zeros(n)
    if lag > 0:
        g[k:] += (1.0 - h[:n-k])
        g[:n-k] -= h[k:]
    else:
        g[:n-k] += (1.0 - h[k:])
        g[k:] -= h[:n-k]
    return g

def project(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float)
    lo, hi = -h.max() - 2.0, -h.min() + 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    r = np.clip(h - 0.5*(lo+hi), 0.0, 1.0)
    diff = target - r.sum()
    if abs(diff) > 1e-13:
        mask = (r < 1.0) if diff > 0 else (r > 0.0)
        if mask.sum() > 0:
            r[mask] += diff / mask.sum()
            r = np.clip(r, 0.0, 1.0)
    return r

def _obj_lbfgs(x, tau, mu):
    corr = np.correlate(x, 1.0 - x, mode='full')
    return smooth_max_val(corr, tau) + mu * (np.sum(x) - x.size*0.5)**2

def _jac_lbfgs(x, tau, mu):
    n = x.size
    target = 0.5 * n
    corr = np.correlate(x, 1.0 - x, mode='full')
    w = smooth_max_weights(corr, tau)
    g = np.zeros(n)
    for i in np.where(w > 1e-10)[0]:
        g += w[i] * lag_subgrad(x, i, n)
    g += 2.0 * mu * (np.sum(x) - target)
    return g

                h_new = project(res.x, target)
```
