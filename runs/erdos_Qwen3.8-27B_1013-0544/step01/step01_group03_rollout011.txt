from scipy.optimize import minimize

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_and_grad(corr, tau):
    x = tau * corr
    x_max = np.max(x)
    shifted = x - x_max
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    val = x_max / tau + np.log(exp_shifted.sum()) / tau
    return val, w

def compute_grad(w, h):
    n = len(h)
    grad = np.zeros(n)
    for j in range(n):
        s1 = 0.0
        s2 = 0.0
        for m in range(2 * n - 1):
            k = m - (n - 1)
            p = j - k
            if 0 <= p < n:
                s1 += w[m] * (1.0 - h[p])
            q = j + k
            if 0 <= q < n:
                s2 -= w[m] * h[q]
        grad[j] = s1 + s2
    return grad

    for _ in range(80):

        h_proj = np.clip(h - mid, 0.0, 1.0)
        if h_proj.sum() > target_sum:

            hi = mid

            h_proj = np.clip(h_proj, 0.0, 1.0)

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_grad(corr, h, tau):
    """Return (smooth_max_value, gradient_wrt_h)"""
    n = len(h)
    x = tau * corr
    x_max = np.max(x)
    exp_x = np.exp(x - x_max)
    w = exp_x / exp_x.sum()
    val = x_max / tau + np.log(exp_x.sum()) / tau
    
    # Gradient via convolution
    conv1 = np.convolve(1.0 - h, w, 'full')
    conv2 = np.convolve(h, w[::-1], 'full')
    grad = (conv1 - conv2)[n-1:n-1+n]
    
    return val, grad

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0

            hi = mid

The gradient grad[j] is a weighted average of terms like (1-h[p]) and h[q], each in [0,1]. The weights sum to 1. So |grad[j]| is at most about 2 (sum of two weighted averages each in [0,1]).

The function value (smooth max of corr) is at most about n/2 (since each corr entry is at most n/2 = sum(h)/2 roughly... actually corr[k] = sum h[j+k](1-h[j]), which is at most min(sum(h), sum(1-h)) = min(n/2, n/2) = n/2).

So the condition number is roughly (n/2) / 2 = n/4. For n=120, that's about 30.

A reasonable learning rate would be something like 1/(tau * L) where L is the Lipschitz constant of the gradient. The gradient is a convolution, so its Lipschitz constant is bounded by the L1 norm of the convolution kernel, which is at most 2 (since w sums to 1 and we have two convolutions). But in terms of the function, the Hessian involves second derivatives of the smooth max, which can be large when tau is small.

import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_grad(corr, h, tau):
    n = len(h)
    x = tau * corr
    x_max = np.max(x)
    exp_x = np.exp(x - x_max)
    w = exp_x / exp_x.sum()
    val = x_max / tau + np.log(exp_x.sum()) / tau
    conv1 = np.convolve(1.0 - h, w, 'full')
    conv2 = np.convolve(h, w[::-1], 'full')
    grad = (conv1 - conv2)[n - 1:n - 1 + n]
    return val, grad

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - h_proj.sum()
    if abs(diff) > 1e-10:
        mask = (h_proj > 1e-6) & (h_proj < 1.0 - 1e-6)
        if mask.sum() > 0:
            h_proj[mask] += diff / mask.sum()
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def fix_sum(h, target):
    h = np.array(h, dtype=float)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return np.clip(h, 0.0, 1.0)
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = mask.sum()
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def pgd_optimize(h_start, target_sum, tau_schedule, rem_func):
    h = h_start.copy()
    n = len(h)
    best_local = h.copy()
    best_c = np.max(np.correlate(h, 1.0 - h, "full"))
    for tau, n_iters in tau_schedule:
        lr = 0.5 / tau
        for it in range(n_iters):
            if rem_func() < 2:
                return best_local
            corr = np.correlate(h, 1.0 - h, "full")
            _, grad = smooth_max_grad(corr, h, tau)
            h_new = project_feasible(h - lr * grad, target_sum)
            c_new = np.max(np.correlate(h_new, 1.0 - h_new, "full"))
            if c_new < best_c:
                best_local = h_new.copy()
                best_c = c_new
            h = h_new
            lr *= 0.995
    return best_local

def binary_local_search(h_cont, rem_func, max_iter=20000):
    n = len(h_cont)
    target = n // 2
    h_bin = (h_cont > 0.5).astype(float)
    n_ones = int(h_bin.sum())
    sorted_idx = np.argsort(h_cont)
    if n_ones > target:
        for idx in sorted_idx[::-1]:
            if n_ones > target and h_bin[idx] == 1.0:
                h_bin[idx] = 0.0
                n_ones -= 1
    elif n_ones < target:
        for idx in sorted_idx:
            if n_ones < target and h_bin[idx] == 0.0:
                h_bin[idx] = 1.0
                n_ones += 1
    c = np.max(np.correlate(h_bin, 1.0 - h_bin, "full"))
    for _ in range(max_iter):
        if rem_func() < 2:
            break
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j or h_bin[i] == h_bin[j]:
            continue
        h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
        c_new = np.max(np.correlate(h_bin, 1.0 - h_bin, "full"))
        if c_new <= c:
            c = c_new
        else:
            h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
    return h_bin, c * (2.0 / n)

def make_initial(n, seed_val):
    np.random.seed(seed_val)
    h = np.random.random(n)
    h = fix_sum(h, n / 2.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):

    t0 = time.time()

    def rem():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:

    else:
        best_h = make_initial(120, seed)

    n = len(best_h)
    target_sum = n / 2.0
    best_c5 = evaluate_c5(best_h)

    tau_sched = [(50, 150), (25, 200), (10, 250), (5, 250), (3, 250),
                 (2, 300), (1.5, 300), (1.0, 350), (0.7, 350),
                 (0.5, 400), (0.3, 400), (0.2, 400)]

    h_opt = pgd_optimize(best_h, target_sum, tau_sched, rem)

        if rem() < 25:
            break
        if tn % 2 != 0:
            continue
        target_t = tn / 2.0
        h_t = np.interp(np.linspace(0, 1, tn), np.linspace(0, 1, len(best_h)), best_h)
        h_t = fix_sum(h_t, target_t)

    return best_h, best_c5, len(best_h)
