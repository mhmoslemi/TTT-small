    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_and_grad(corr, tau):
    x = tau * corr
    x_max = np.max(x)
    shifted = x - x_max
    exp_shifted = np.exp(shifted)
    log_sum = np.log(np.sum(exp_shifted))
    sm = x_max / tau + log_sum / tau
    weights = exp_shifted / np.sum(exp_shifted)
    return sm, weights

def corr_grad_and_value(h, tau):
    """Compute smooth max of correlations and its gradient."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    sm, weights = smooth_max_and_grad(corr, tau)
    
    # Gradient computation
    g = np.zeros(n)
    for m in range(2 * n - 1):
        wm = weights[m]
        if wm < 1e-14:

        k = m - (n - 1)

    args = (tau, target)

        res = minimize(objective_func, h0, args=args, method='L-BFGS-B',
                       bounds=bnds, jac=True,
                       options={'maxiter': maxiter, 'ftol': tol, 'gtol': 1e-10})
        return fix_sum(res.x, target)
    except Exception:
        return h0

def run_slsqp(h0, tau, target, maxiter=300):
    n = len(h0)
    bnds = Bounds(np.zeros(n), np.ones(n))
    args = (tau, target)
    cons = [{'type': 'eq', 'fun': sum_con_fun, 'jac': sum_jac_fun, 'args': (target,)}]
    try:
        res = minimize(objective_slsqp, h0, args=args, method='SLSQP',
                       bounds=bnds, jac=True, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return fix_sum(res.x, target)
    except Exception:
        return h0

def sum_con_fun(x, target):
    return x.sum() - target

def sum_jac_fun(x, target):
    return np.ones(len(x))

def projected_subgrad(h, tau, n_iter, step_init, decay):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    for it in range(n_iter):
        step = step_init * (decay ** it)
        _, g = corr_grad_and_value(h, tau)
        h_new = project_onto_simplex(h - step * g, target)
        c5 = evaluate_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(120, 0.5)

    n = len(h_init)
    target = n / 2.0
    best_h = h_init.copy()
    best_c5 = evaluate_c5(best_h)

    # Phase 1: Continuation method with L-BFGS-B
    tau_list = [2, 5, 15, 50, 150, 500, 1500, 5000, 15000]
    for tau in tau_list:
        if budget_s - (time.time() - t0) < 15:
            break
        h_new = run_lbfgsb(best_h, tau, target, maxiter=400)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 2: Projected subgradient for exploration
    for tau in [5, 20, 100, 500]:
        if budget_s - (time.time() - t0) < 10:
            break
        h_new, c5_new = projected_subgrad(best_h, tau, 800, 0.8, 0.997)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 3: SLSQP polish with exact constraint
    if budget_s - (time.time() - t0) > 20:
        for tau in [500, 2000, 8000]:
            if budget_s - (time.time() - t0) < 10:
                break
            h_new = run_slsqp(best_h, tau, target, maxiter=400)
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new

    # Phase 4: Resolution sweep
    for tn in [60, 80, 100, 140, 160, 200]:
        if budget_s - (time.time() - t0) < 40:
            break
        tt = tn / 2.0
        h_t = np.interp(np.linspace(0, 1, tn), np.linspace(0, 1, n), best_h)
        h_t = fix_sum(h_t, tt)
        for tau in [200, 1000, 5000]:
            if budget_s - (time.time() - t0) < 15:
                break
            h_t = run_lbfgsb(h_t, tau, tt, maxiter=250)
            c5_t = evaluate_c5(h_t)
            if c5_t < best_c5:
                best_h, best_c5 = h_t, c5_t
                n = tn
        if budget_s - (time.time() - t0) > 20:
            h_t, c5_t = projected_subgrad(h_t, 500, 300, 0.5, 0.995)
            if c5_t < best_c5:
                best_h, best_c5 = h_t, c5_t
                n = tn

    # Phase 5: Multiple random restarts
    for trial in range(40):
        if budget_s - (time.time() - t0) < 12:
            break
        scale = 0.03 + 0.25 * np.random.random()
        hp = np.clip(best_h + np.random.normal(0, scale, n), 0.0, 1.0)
        hp = fix_sum(hp, n / 2.0)
        for tau in [200, 1000, 4000]:
            if budget_s - (time.time() - t0) < 8:
                break
            hp = run_lbfgsb(hp, tau, n / 2.0, maxiter=200)
            c5_r = evaluate_c5(hp)
            if c5_r < best_c5:
                best_h, best_c5 = hp, c5_r

    # Phase 6: Final SLSQP polish
    if budget_s - (time.time() - t0) > 15:
        for tau in [5000, 20000]:
            if budget_s - (time.time() - t0) < 8:
                break
            h_new = run_slsqp(best_h, tau, n / 2.0, maxiter=500)
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new

    # Phase 7: Very fine L-BFGS-B
    if budget_s - (time.time() - t0) > 5:
        h_new = run_lbfgsb(best_h, 50000, n / 2.0, maxiter=1000, tol=1e-16)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    best_h = fix_sum(best_h, len(best_h) / 2.0)
    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, len(best_h)
