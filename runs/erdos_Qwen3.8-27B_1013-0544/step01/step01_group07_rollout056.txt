Now let me think about the exact coordinate descent more carefully. The key question is: when I change h[i] to v, the new result[m] = old_result[m] + (v-h[i])*a_m where a_m = ∂result[m]/∂h[i].

This is because result[m] is linear in h[i] (for fixed other coordinates). So the exact coordinate descent is correct.

Now, one subtlety: when I update h[i] in coordinate descent, the previously computed a_m values for other coordinates become stale. But since I recompute corr at the start of each sweep, and within a sweep I update coordinates one by one, the a_m values computed for coordinate j (j > i) don't account for the change in h[i]. This is the standard "Gauss-Seidel" style coordinate descent, which is fine.

Actually wait, within a sweep, after I update h[i], I should use the updated h for computing a_m for subsequent coordinates j > i. Let me make sure my code does this. Yes, since I'm modifying h[i] in place and then computing a for the next coordinate using the current h, this is correct.

But I'm not updating corr in place! I'm computing corr at the start of the sweep and using it throughout. That's a problem because after updating h[i], corr is stale.

Two options:
1. Update corr incrementally: when h[i] changes from old to new, corr[m] changes by (new-old)*a_m(i) for all m. This is O(n) per coordinate update.
2. Recompute corr after each coordinate update: O(n²) per coordinate, O(n³) per sweep. Too slow.
3. Use the stale corr (like I was doing): this is an approximation. The bisection finds the v that minimizes the max of the linear approximation, which may not be exact but should be close.

After updating h[i] from old_h to new_h:
corr[m] += (new_h - old_h) * a_m(i) for all m

This is O(n) per coordinate, so O(n²) per sweep. For n=160, that's 25600 per sweep. Very fast.

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def subgrad_at(h, m_star, n):
    s = m_star - (n - 1)
    g = np.zeros(n)
    idx1 = np.arange(n) - s
    mask1 = (idx1 >= 0) & (idx1 < n)
    g[mask1] += (1.0 - h[idx1[mask1]])
    idx2 = np.arange(n) + s
    mask2 = (idx2 >= 0) & (idx2 < n)
    g[mask2] -= h[idx2[mask2]]
    return g

def project_polytope(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def subgrad_optimize(h, dx, target_sum, n_iter, alpha0, momentum=0.9):
    n = len(h)
    h = h.copy().astype(float)
    h_prev = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    for k in range(n_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        max_val = np.max(corr)
        active = np.where(corr >= max_val - 1e-9)[0]
        g = np.zeros(n)
        for m in active:
            g += subgrad_at(h, m, n)
        g /= len(active)
        alpha = alpha0 / np.sqrt(k + 1)
        h_new = h - alpha * g + momentum * (h - h_prev)
        h_new = project_polytope(h_new, target_sum)
        h_prev = h
        h = h_new
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

            mask1 = (idx1 >= 0) & (idx1 < n)

            a = np.zeros(m_full)
            a[mask1] = 1.0 - h[idx1[mask1]]
            a[mask2] -= h[idx2[mask2]]
            c = corr - h[i] * a
            pos = a > 1e-12
            neg = a < -1e-12
            zero = ~pos & ~neg
            lo_t = np.max(np.where(a >= 0, c, c + a))
            hi_t = np.min(np.max(c), np.max(c + a))
            if lo_t > hi_t:
                hi_t = lo_t
            for _ in range(40):
                mid_t = 0.5 * (lo_t + hi_t)
                ok = True
                if np.any(zero) and np.max(c[zero]) > mid_t:
                    ok = False

                    lower = 0.0

                        lower = max(lower, float(np.max((mid_t - c[neg]) / a[neg])))
                    if upper < lower - 1e-12 or lower > 1.0 + 1e-12 or upper < -1e-12:
                        ok = False
                if ok:

            target_t = hi_t

def make_block_init(n, n_blocks, seed_val):
    rng = np.random.RandomState(seed_val)
    h = np.zeros(n)
    block_size = n // n_blocks
    n_ones = n_blocks // 2
    ones_blocks = rng.choice(n_blocks, n_ones, replace=False)
    for b in ones_blocks:
        start = b * block_size
        end = start + block_size
        h[start:end] = 1.0
    return project_polytope(h, n // 2)

def make_periodic_init(n, period, frac, seed_val):
    rng = np.random.RandomState(seed_val)
    h = np.zeros(n)
    pattern_len = period
    n_on = int(period * frac)
    pattern = np.zeros(pattern_len)
    pattern[:n_on] = 1.0
    rng.shuffle(pattern)
    n_repeats = n // pattern_len
    for r in range(n_repeats):
        h[r * pattern_len:(r + 1) * pattern_len] = pattern
    if n % pattern_len > 0:
        h[n_repeats * pattern_len:n] = pattern[:n % pattern_len]
    return project_polytope(h, n / 2.0)
