def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def compute_c5_raw(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr)

def make_binary(h_cont, n):
    """Convert continuous h to binary with exactly n/2 ones."""
    n = len(h_cont)
    # Sort by value, take top n/2 as 1s
    order = np.argsort(-h_cont)
    h = np.zeros(n)
    h[order[:n//2]] = 1.0
    return h

    for round_idx in range(max_rounds):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]

                h_try[i] = 0

    """Weighted subgradient over top shifts."""
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    # Get top-k shifts
    k = min(10, n)
    top_idx = np.argsort(-corr)[:k]
    # Weights based on softmax
    top_vals = corr[top_idx]
    w = np.exp(alpha * (top_vals - top_vals[0]))
    w /= w.sum()
    
    g = np.zeros(n)
    for idx, wi in zip(top_idx, w):
        shift = idx - (n - 1)
        gs = np.zeros(n)
        if shift >= 0:
            gs[shift:] += (1 - h[:n - shift])
            if shift > 0:
                gs[:n - shift] -= h[shift:]
        else:
            p = -shift
            gs[:n - p] += (1 - h[p:])
            gs[p:] -= h[:n - p]
        g += wi * gs
    return g

def compute_c5(h):

        else:
            hi = mid
