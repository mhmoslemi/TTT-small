    stall = 0
    stall_cyc = 0

    while time.time() < deadline:
        corr = compute_corr(h)
        mx = corr.max()

        w = np.exp(alpha * (corr - mx))
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros_like(corr)
            w[corr.argmax()] = 1.0
        else:
            w /= ws

        threshold = 0.02 / len(w)
        indices = np.where(w > threshold)[0]
        grad = np.zeros(n)
        for idx in indices:
            if w[idx] > 1e-6:
                grad += w[idx] * grad_at_lag(h, idx)
        grad *= (2.0 / n)

        vel = momentum * vel + (1 - momentum) * grad
        eff_lr = lr / (1.0 - momentum)

        improved = False
        cur_lr = eff_lr
        for _ in range(6):
            h_new = project_sum_box(h - cur_lr * vel, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-13:
                h = h_new
                best_h = h.copy()
                best_c5 = c5_new
                stall = 0
                lr = min(cur_lr * (1.0 - momentum) * 1.08, 0.08)
                improved = True
                break
            cur_lr *= 0.6

        if not improved:
            stall += 1
            if stall >= 12:
                ptype = rng.randint(6)
                if ptype == 0:
                    h_p = h + rng.randn(n) * 0.035
                elif ptype == 1:
                    blen = rng.randint(4, max(5, n // 6))
                    start = rng.randint(0, n - blen)
                    h_p = h.copy()
                    h_p[start:start + blen] = 1.0 - h[start:start + blen]
                elif ptype == 2:
                    blen = rng.randint(8, max(9, n // 4))
                    start = rng.randint(0, n - blen)
                    h_p = h.copy()
                    h_p[start:start + blen] = np.roll(h[start:start + blen], rng.randint(1, blen))
                elif ptype == 3:
                    t = np.arange(n)
                    freq = rng.choice([1, 2, 3, 4])
                    h_p = h + 0.04 * np.sin(2 * np.pi * freq * t / n + rng.uniform(0, 2 * np.pi))
                elif ptype == 4:
                    t = np.arange(n)
                    freq = rng.randint(5, 20)
                    h_p = h + 0.025 * np.sin(2 * np.pi * freq * t / n + rng.uniform(0, 2 * np.pi))
                else:
                    blen = rng.randint(3, max(4, n // 8))
                    h_p = h.copy()
                    for _ in range(3):
                        start = rng.randint(0, n - blen)
                        h_p[start:start + blen] = 1.0 - h_p[start:start + blen]

                h_p = project_sum_box(h_p, target_sum)
                c5_p = compute_c5(h_p)
                if c5_p < best_c5:
                    best_h = h_p.copy()
                    best_c5 = c5_p
                    h = h_p.copy()

                alpha = min(alpha * 1.25, 200.0)
                momentum = min(momentum + 0.05, 0.8)
                lr = 0.015
                vel = np.zeros(n)
                stall = 0
                stall_cyc += 1
                if stall_cyc > 15:
                    break

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    inits = []
    n_base = 200

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float)
        n_base = len(h_parent)
        inits.append(h_parent)

    n_r = n_base
    h_bin = np.zeros(n_r)
    ones_pos = rng.choice(n_r, size=n_r // 2, replace=False)
    h_bin[ones_pos] = 1.0
    inits.append(h_bin)

    h_alt = np.zeros(n_r)
    t_arr = np.arange(n_r)
    h_alt[:] = 0.5 + 0.3 * np.sin(np.pi * t_arr / n_r) + 0.15 * np.sin(2 * np.pi * t_arr / n_r)
    h_alt = project_sum_box(h_alt, n_r / 2.0)
    inits.append(h_alt)

    if inits:
        h_sm = smooth_h(inits[0], 2.5)
        h_sm = project_sum_box(h_sm, n_base / 2.0)
        inits.append(h_sm)

    best_overall = None
    best_c5_overall = 1e10
    best_n_overall = n_base

    n_inits = len(inits)
    phase1_budget = budget_s * 0.45
    time_per_init = phase1_budget / n_inits

    for i in range(n_inits):
        if time.time() - t0 > t0 + phase1_budget:
            break
        h_i = inits[i]
        n_i = len(h_i)
        ts_i = n_i / 2.0
        deadline_i = min(t0 + (i + 1) * time_per_init, t0 + phase1_budget)
        h_opt, c5_opt = optimize_inner(h_i, ts_i, deadline_i, rng)
        if c5_opt < best_c5_overall:
            best_c5_overall = c5_opt
            best_overall = h_opt.copy()
            best_n_overall = n_i

    remaining = budget_s - (time.time() - t0)
    if remaining > 50 and best_overall is not None:
        try_res = [128, 256, 320]
        for n_try in try_res:
            if remaining < 35:
                break
            if n_try == best_n_overall:
                continue
            x_old = np.linspace(0, 2, best_n_overall, endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_interp = np.interp(x_new, x_old, best_overall)
            h_interp = project_sum_box(h_interp, n_try / 2.0)
            d_try = time.time() + min(remaining * 0.3, 220.0)
            h_opt, c5_opt = optimize_inner(h_interp, n_try / 2.0, d_try, rng)
            if c5_opt < best_c5_overall:
                best_c5_overall = c5_opt
                best_overall = h_opt.copy()
                best_n_overall = n_try
            remaining = budget_s - (time.time() - t0)

    remaining = budget_s - (time.time() - t0)
    if remaining > 12 and best_overall is not None:
        best_overall, best_c5_overall = optimize_inner(
            best_overall, len(best_overall) / 2.0, t0 + budget_s - 3, rng)

    best_overall = project_sum_box(best_overall, len(best_overall) / 2.0)
    best_c5_overall = compute_c5(best_overall)

    return best_overall, best_c5_overall, len(best_overall)
