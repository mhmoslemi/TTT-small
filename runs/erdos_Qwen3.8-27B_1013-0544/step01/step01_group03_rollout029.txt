```python
def subgradient_step(h, n, step_size):
    """One subgradient step to reduce max overlap"""
    dx = 2.0 / n
    # Compute all overlaps
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full")
    overlaps = corr * dx
    k_max = np.argmax(overlaps)
    
    # Convert correlate index to lag
    # correlate(h, v, "full")[c] = sum_i h[i] * v[i - c + n - 1]
    # where v = 1-h
    # So c corresponds to lag = c - (n-1)
    lag = k_max - (n - 1)
    
    # Compute gradient of g(lag) w.r.t. h
    # g(lag) = sum_i h[i] * (1 - h[i+lag]) for valid i
    # For the discrete correlate, it's a bit different due to the "full" mode
    # Let me just use numerical gradient for simplicity, or compute analytically
    
    grad = np.zeros(n)
    if lag >= 0:
        # g = sum_{i=0}^{n-1-lag} h[i] * (1 - h[i+lag])
        # ∂g/∂h[j] = (1-h[j+lag]) for j < n-lag (from first term)
        #            - h[j-lag] for j >= lag (from second term)
        for i in range(n - lag):
            grad[i] += (1 - h[i + lag])
            grad[i + lag] -= h[i]
    else:
        llag = -lag
        # g = sum_{i=llag}^{n-1} h[i] * (1 - h[i-llag])
        for i in range(llag, n):
            grad[i] += (1 - h[i - llag])
            grad[i - llag] -= h[i]
    
    # Update
    h_new = h - step_size * grad
    # Project: clip to [0,1], fix sum
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = fix_sum(h_new, n / 2.0)
    return h_new
```

```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target):
    """Fix sum of h to target while keeping h in [0,1]"""
    h = np.array(h, dtype=float).copy()
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return np.clip(h, 0.0, 1.0)
    if diff > 0:
        # Need to increase; prioritize positions with most room
        room = 1.0 - h
        mask = room > 1e-10
        if mask.sum() == 0:
            return np.clip(h, 0.0, 1.0)
        # Distribute proportionally to room
        total_room = room[mask].sum()
        h[mask] += diff * room[mask] / total_room
    else:
        # Need to decrease; prioritize positions that can give the most
        mask = h > 1e-10
        if mask.sum() == 0:
            return np.clip(h, 0.0, 1.0)
        total = h[mask].sum()
        h[mask] += diff * h[mask] / total
    return np.clip(h, 0.0, 1.0)

def compute_overlaps(h):
    """Compute all overlap values"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def top_k_gravity(h, top_k):
    """Compute weighted gradient using top-k worst lags"""
    n = len(h)
    overlaps = compute_overlaps(h)
    # Get top-k indices
    idx = np.argsort(overlaps)[::-1][:top_k]
    weights = overlaps[idx]
    weights = weights / weights.sum()
    
    # Compute gradient as weighted sum
    grad = np.zeros(n)
    for w, ci in zip(weights, idx):
        lag = ci - (n - 1)
        if lag >= 0:
            i = np.arange(n - lag)
            grad[i] += w * (1 - h[i + lag])
            grad[i + lag] -= w * h[i]
        else:
            llag = -lag
            i = np.arange(llag, n)
            grad[i] += w * (1 - h[i - llag])
            grad[i - llag] -= w * h[i]
    return grad

def subgradient_optimize(h, n, time_limit, top_k=5):
    """Optimize using subgradient with momentum and adaptive step size"""
    t_start = time.time()
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    v = np.zeros(n)  # momentum
    mom = 0.9
    alpha = 1.0 / top_k
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    patience = 0
    max_patience = 500
    
    while time.time() - t_start < time_limit:
        grad = top_k_gravity(h, top_k)
        v = mom * v - alpha * grad
        h_new = h + v
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = fix_sum(h_new, target)
        
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            alpha = min(alpha * 1.02, 2.0)
            patience = 0
        else:
            h = h_new  # still accept for momentum, but track best
            alpha *= 0.95
            patience += 1
            if patience > max_patience:
                h = best_h.copy()
                alpha = 0.1
                patience = 0
                mom = 0.8
    
    return best_h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        mask = room > 1e-8
        if mask.sum() == 0: return h
        h[mask] += diff * room[mask] / room[mask].sum()
    else:
        mask = h > 1e-8
        if mask.sum() == 0: return h
        h[mask] += diff * h[mask] / h[mask].sum()
    return np.clip(h, 0.0, 1.0)

def get_grad_topk(h, top_k=5):
    """Compute weighted gradient from top-k worst lags"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    # Get top-k correlation values
    idx = np.argpartition(corr, -top_k)[-top_k:]
    w = corr[idx]
    w = w / (w.sum() + 1e-15)
    
    grad = np.zeros(n)
    for wi, ci in zip(w, idx):
        lag = ci - (n - 1)
        if lag >= 0:
            i = np.arange(n - lag)
            grad[i] += wi * (1.0 - h[i + lag])
            grad[i + lag] -= wi * h[i]
        else:
            m = -lag
            i = np.arange(m, n)
            grad[i] += wi * (1.0 - h[i - m])
            grad[i - m] -= wi * h[i]
    return grad / top_k  # normalize

def optimize_subgrad(h, time_limit, top_k=5, mom=0.85):
    """Subgradient optimization with momentum and adaptive step size"""
    t0 = time.time()
    n = len(h)
    target = n / 2.0
    h = h.copy()
    v = np.zeros(n)
    alpha = 0.5
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    no_improve = 0
    
    while time.time() - t0 < time_limit:
        g = get_grad_topk(h, top_k)
        v = mom * v - alpha * g
        h_new = h + v
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = fix_sum(h_new, target)
        
        c5 = evaluate_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
            h = h_new
            alpha = min(alpha * 1.01, 5.0)
            no_improve = 0
        else:
            h = h_new
            alpha *= 0.98
            no_improve += 1
            if no_improve > 2000:
                # Restart from best with perturbation
                h = best_h + 0.05 * np.random.randn(n)
                h = fix_sum(h, target)
                v = np.zeros(n)
                alpha = 0.3
                no_improve = 0
    
    return best_h, best_c5
```

```python
    """Compute gradient of smooth max of correlations"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    # Softmax weights
    x = corr / tau
    x_max = x.max()
    w = np.exp(x - x_max)
    w = w / w.sum()
    
    # Gradient = weighted sum of individual gradients
    grad = np.zeros(n)
    for ci in range(len(corr)):
        if w[ci] < 1e-8:
```

```python
def smooth_grad_fast(h, tau):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    x = corr / tau
    x_max = x.max()
    w = np.exp(x - x_max)
    w = w / w.sum()
    
    grad = np.zeros(n)
    # Positive lags: lag = 1, 2, ..., n-1 (corresponding to c = n, n+1, ..., 2n-2)
    for lag in range(1, n):
        wi = w[n + lag - 1]  # c = n-1+lag → wait
        # c = lag + (n-1) for positive lag
        wi = w[n - 1 + lag]
        if wi < 1e-10:
            continue
        grad[:n-lag] += wi * (1.0 - h[lag:n])
        grad[lag:n] -= wi * h[:n-lag]
    
    # Lag 0: c = n-1
    wi = w[n-1]
    grad += wi * (1.0 - 2.0 * h)
    
    # Negative lags: lag = -1, -2, ..., -(n-1)
    for lag in range(1, n):
        m = lag
        # c = n-1-m
        wi = w[n - 1 - m]
        if wi < 1e-10:
            continue
        grad[m:n] += wi * (1.0 - h[:n-m])
        grad[:n-m] -= wi * h[m:n]
    
    return grad
```
