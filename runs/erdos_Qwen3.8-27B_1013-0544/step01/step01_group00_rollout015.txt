    dx = 2.0 / n
    h = project_h(h.copy(), n * 0.5)

                                bounds=bounds_c, maxiter=60, seed=seed,
                                initial_temp=0.3, restart_temp_ratio=0.7)
        h_da = project_h(da_res.x, n_c * 0.5)
        # Local refine at coarse
        h_c, c5_c = optimize_local(h_da, temps_5, mu=3000.0, max_iter=200)
        # Also try without dual_annealing (just local from this start)
        h_c2, c5_c2 = optimize_local(s, temps_5, mu=3000.0, max_iter=200)
        if c5_c2 < c5_c:
            h_c, c5_c = h_c2, c5_c2
        # Upgrade to full resolution if competitive
        if c5_c < c5_best + 0.005 and time_left() > 150:
            h_up = interp_h(h_c, n_best)
            h_ref, c5_ref = optimize_local(h_up, temps_7, mu=2000.0, max_iter=250)
            if c5_ref < c5_best:
                h_best, c5_best = h_ref, c5_ref

    # Phase 3: Multi-start perturbation around best
    n_pert = 15
    for i in range(n_pert):
        if time_left() < 60:
            break
        scale = 0.003 + 0.008 * i
        h_p = h_best + rng.randn(n_best) * scale
        h_p = project_h(h_p, n_best * 0.5)
        h_new, c5_new = optimize_local(h_p, temps_4, mu=3000.0, max_iter=200)
        if c5_new < c5_best:
            h_best, c5_best = h_new, c5_new

    # Phase 4: Try n=96 as final resolution (might give lower discrete C5)
    if time_left() > 120:
        h_try = interp_h(h_best, 96)
        h_96, c5_96 = optimize_local(h_try, temps_7, mu=3000.0, max_iter=300)
        if c5_96 < c5_best:
            h_best, c5_best, n_best = h_96, c5_96, 96

    # Phase 5: Final polish
    if time_left() > 80:
        h_new, c5_new = optimize_local(h_best, temps_2, mu=10000.0, max_iter=500)
        if c5_new < c5_best:
            h_best, c5_best = h_new, c5_new

    # One more perturbation round if time allows
    if time_left() > 100:
        for i in range(5):
            if time_left() < 50:
                break
            scale = 0.002 + 0.005 * i
            h_p = h_best + rng.randn(n_best) * scale
            h_p = project_h(h_p, n_best * 0.5)
            h_new, c5_new = optimize_local(h_p, temps_2, mu=5000.0, max_iter=300)
            if c5_new < c5_best:
                h_best, c5_best = h_new, c5_new

    h_best = project_h(h_best, n_best * 0.5)
    c5_final = compute_c5(h_best)
    return h_best, c5_final, n_best
