def project_sum_box(v, target_sum):
    """Project v onto {x: 0<=x<=1, sum(x)=target_sum}."""
    # Water-filling
    lo, hi = -10, 10
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(v - mid, 0, 1))
        if s > target_sum:
            lo = mid
        else:
            hi = mid

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))


def project_h(h, target):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = np.sum(mask)
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)


def project_sum_box(v, target_sum):
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)


def lse_and_grad(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    shifted = (corr - m) / temp
    exp_sh = np.exp(shifted)
    sum_exp = np.sum(exp_sh)
    lse = m + temp * np.log(sum_exp)
    weights = exp_sh / sum_exp
    grad = np.zeros(n)
    omh = 1.0 - h
    for mi in range(len(weights)):
        w = weights[mi]
        if w < 1e-12:
            continue
        L = mi - (n - 1)
        if L >= 0:
            if L < n:
                grad[:n - L] += w * omh[L:]
                grad[L:] += -w * h[:n - L]
        else:
            Lp = -L
            if Lp < n:
                grad[Lp:] += w * omh[:n - Lp]
                grad[:n - Lp] += -w * h[Lp:]
    grad *= dx
    return lse, grad


def lse_pen_obj(h, temp, pen):
    val, g = lse_and_grad(h, temp)
    sv = np.sum(h) - len(h) * 0.5
    val += pen * sv * sv
    g += 2.0 * pen * sv
    return val, g


def lse_pen_wrapper(h, temp, pen):
    return lse_pen_obj(h, temp, pen)


def lse_pen_grad_wrapper(h, temp, pen):
    return lse_pen_obj(h, temp, pen)


def t_obj(v):
    return v[-1]


def t_ineq(v):
    n = len(v) - 1
    h = v[:n]
    t = v[-1]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return t - corr


def t_eq(v):
    n = len(v) - 1
    return np.sum(v[:n]) - n * 0.5


def t_eq_jac(v):
    n = len(v) - 1
    j = np.zeros(len(v))
    j[:n] = 1.0
    return j


def slsqp_tvar(h0, max_iter=500, ftol=1e-11):
    n = len(h0)
    c5_cur = compute_c5(h0)
    v0 = np.append(h0.copy(), c5_cur + 0.01)
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.5)]
    cons = [
        {'type': 'ineq', 'fun': t_ineq},
        {'type': 'eq', 'fun': t_eq, 'jac': t_eq_jac}
    ]
    res = minimize(t_obj, v0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': ftol})
    h_new = project_h(res.x[:n], n * 0.5)
    return h_new, compute_c5(h_new)


def subgradient_step(h, grad, alpha, target_sum):
    v = h - alpha * grad
    return project_sum_box(v, target_sum)


def subgradient_method(h0, max_iter=2000, tol=1e-10):
    n = len(h0)
    target = n * 0.5
    h = h0.copy()
    c5_prev = compute_c5(h)
    alpha = 0.01
    h_prev = h.copy()
    g_prev = None
    for it in range(max_iter):
        temp = 0.001
        _, grad = lse_and_grad(h, temp)
        sv = np.sum(h) - target
        grad += sv
        g_norm = np.linalg.norm(grad)
        if g_norm < tol:
            break
        if g_prev is not None:
            s = h - h_prev
            y = grad - g_prev
            sy = np.dot(s, y)
            yy = np.dot(y, y)
            if yy > 1e-16 and sy > 1e-16:
                alpha_bb = np.dot(s, s) / sy
                alpha = np.clip(alpha_bb, 0.001, 0.1)
        h_prev = h.copy()
        g_prev = grad.copy()
        h = subgradient_step(h, grad, alpha, target)
        c5_new = compute_c5(h)
        if c5_new < c5_prev:
            c5_prev = c5_new
        else:
            alpha *= 0.8
            h = subgradient_step(h, grad, alpha, target)
    return h, compute_c5(h)

    h_new = np.interp(x_new, x_old, h_old)
    return project_h(h_new, n_new * 0.5)

                pass

        h_fin, c5_fin = slsqp_tvar(h_best, max_iter=600, ftol=1e-12)
        if c5_fin < c5:
            h_best = h_fin
            c5 = c5_fin

    h_best = project_h(h_best, len(h_best) * 0.5)
    c5_final = compute_c5(h_best)
    return h_best, c5_final, len(h_best)
