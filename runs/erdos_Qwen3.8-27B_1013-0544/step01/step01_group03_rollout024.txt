        elif d < 0:
            for i in range(-d, n):
                if h[i] == 1 and h[i + d] == 0:
                    contributing.append(i)
        improved = False
        n_try = min(30, max(5, len(contributing)))
        if contributing:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            for _ in range(n_try):
                i = contributing[rng.randint(len(contributing))]
                mode = rng.randint(3)
                if mode == 0 and len(zeros) > 0:
                    j = zeros[rng.randint(len(zeros))]
                    if j == i:
                        continue
                    h[i] = 0
                    h[j] = 1
                elif mode == 1 and len(ones) > 1:
                    j = ones[rng.randint(len(ones))]
                    if j == i:
                        continue
                    k_neg = i + d
                    if 0 <= k_neg < n and h[k_neg] == 0:
                        h[j] = 0
                        h[k_neg] = 1
                    else:
                        continue
                else:
                    j = zeros[rng.randint(len(zeros))] if len(zeros) > 0 else i
                    if j == i:
                        continue
                    h[i] = 0
                    h[j] = 1
                new = evaluate_c5(h)
                if new < curr:
                    curr = new
                    improved = True
                    if new < best:
                        best = new
                        best_h = h.copy()
                else:
                    if mode == 0:
                        h[i] = 1
                        h[j] = 0
                    elif mode == 1:
                        k_neg = i + d
                        if 0 <= k_neg < n:
                            h[j] = 1
                            h[k_neg] = 0
                    else:
                        h[i] = 1
                        h[j] = 0
        if not improved:
            stall += 1
            if stall > 50 and rng.random() < 0.1:
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h[i] = 0
                h[j] = 1
                curr = evaluate_c5(h)
                stall = 0
        else:
            stall = 0
    return best_h, best

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def rem():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(120, 0.5)

    best_h = h_init.copy()
    best_c5 = evaluate_c5(best_h)

    # Phase 1: Binary search at multiple resolutions
    n_list = [60, 80, 100, 120, 140, 160]
    for n_pts in n_list:
        if rem() < 25:
            break
        n_ones = n_pts // 2
        seeds = []
        # From initial
        if len(h_init) == n_pts:
            hb = (h_init > 0.5).astype(float)
        else:
            hi = np.interp(np.linspace(0, 1, n_pts), np.linspace(0, 1, len(h_init)), h_init)
            hb = (hi > 0.5).astype(float)
        seeds.append(_fix_binary_count(hb, n_ones))
        # Periodic
        for p in [3, 4, 5, 6, 7]:
            seeds.append(_gen_periodic(n_pts, p, np.random))
        # Random
        hr = np.zeros(n_pts)
        hr[np.random.choice(n_pts, n_ones, replace=False)] = 1
        seeds.append(hr)

        for si in range(len(seeds)):
            if rem() < 10:
                break
            ns = int(min(3000, max(500, rem() * 20.0 / len(seeds))))
            ho, co = _binary_anneal(seeds[si], ns, seed + si * 7)
            if co < best_c5:
                best_h, best_c5 = ho, co
            if rem() < 10:
                break
            ng = int(min(2000, max(300, rem() * 15.0 / len(seeds))))
            ho, co = _guided_binary(best_h if best_h.shape[0] == n_pts else seeds[si],
                                     ng, seed + 100 + si)
            if co < best_c5:
                best_h, best_c5 = ho, co

    # Phase 2: Continuous refinement with annealed smooth max
    hn = len(best_h)
    target = hn / 2.0
    bnds = Bounds(np.zeros(hn), np.ones(hn))
    for tau in [20, 50, 100, 300, 1000, 3000, 8000]:
        if rem() < 12:
            break
        def _obj(x, _tau=tau):
            c = np.correlate(x, 1.0 - x, mode="full")
            return _smooth_max(c, _tau)
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                 'jac': lambda x: np.ones(hn)}]
        try:
            res = minimize(_obj, best_h, method='SLSQP', bounds=bnds,
                          constraints=cons, options={'maxiter': 400, 'ftol': 1e-13})
            x = _fix_sum(res.x, target)
            c5 = evaluate_c5(x)
            if c5 < best_c5:
                best_h, best_c5 = x, c5
        except Exception:
            pass

    # Phase 3: Multi-resolution continuous
    for n2 in [100, 120, 140, 160, 200]:
        if rem() < 20:
            break
        if n2 == hn:
            continue
        tgt2 = n2 / 2.0
        hi2 = np.interp(np.linspace(0, 1, n2), np.linspace(0, 1, hn), best_h)
        hi2 = _fix_sum(hi2, tgt2)
        bnds2 = Bounds(np.zeros(n2), np.ones(n2))
        for tau in [100, 500, 2000]:
            if rem() < 10:
                break
            def _obj2(x, _tau=tau):
                c = np.correlate(x, 1.0 - x, mode="full")
                return _smooth_max(c, _tau)
            cons2 = [{'type': 'eq', 'fun': lambda x: np.sum(x) - tgt2,
                      'jac': lambda x: np.ones(n2)}]
            try:
                res = minimize(_obj2, hi2, method='SLSQP', bounds=bnds2,
                              constraints=cons2, options={'maxiter': 250, 'ftol': 1e-13})
                x = _fix_sum(res.x, tgt2)
                c5 = evaluate_c5(x)
                if c5 < best_c5:
                    best_h, best_c5 = x, c5
                    hi2 = x
            except Exception:
                pass

    # Phase 4: Final binary polish + continuous
    if rem() > 15:
        hn = len(best_h)
        tb = (best_h > 0.5).astype(float)
        tb = _fix_binary_count(tb, hn // 2)
        nf = int(min(4000, max(500, int(rem() * 1.5))))
        ho, co = _guided_binary(tb, nf, seed + 500)
        if co < best_c5:
            best_h, best_c5 = ho, co
        if rem() > 10:
            target = hn / 2.0
            bnds = Bounds(np.zeros(hn), np.ones(hn))
            for tau in [300, 1500, 5000]:
                if rem() < 5:
                    break
                def _obj4(x, _tau=tau):
                    c = np.correlate(x, 1.0 - x, mode="full")
                    return _smooth_max(c, _tau)
                cons4 = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                          'jac': lambda x: np.ones(hn)}]
                try:
                    res = minimize(_obj4, best_h, method='SLSQP', bounds=bnds,
                                  constraints=cons4, options={'maxiter': 300, 'ftol': 1e-13})
                    x = _fix_sum(res.x, target)
                    c5 = evaluate_c5(x)
                    if c5 < best_c5:
                        best_h, best_c5 = x, c5
                except Exception:
                    pass

    # Final fix
    best_h = np.clip(best_h, 0, 1)
    diff = len(best_h) / 2.0 - best_h.sum()
    if abs(diff) > 1e-10:
        mask = (best_h < 0.999) if diff > 0 else (best_h > 0.001)
        if mask.sum() > 0:
            best_h[mask] += diff / mask.sum()
        best_h = np.clip(best_h, 0, 1)
    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, len(best_h)
