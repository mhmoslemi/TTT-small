import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1 - h, mode="full")

def project_onto_constraints(h, target_sum):
    n = len(h)
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(h_proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - np.sum(h_proj)
    if abs(diff) > 1e-10:
        interior = (h_proj > 1e-8) & (h_proj < 1.0 - 1e-8)
        if np.sum(interior) > 0:
            h_proj[interior] += diff / np.sum(interior)
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def smooth_max_obj(x, t, dx):
    corr = np.correlate(x, 1 - x, mode="full")
    max_c = np.max(corr)
    shifted = t * (corr - max_c)
    return (max_c + (1.0 / t) * np.log(np.sum(np.exp(shifted)))) * dx

def subgradient_direction(h, top_k=3):
    n = len(h)
    corr = compute_corr(h)
    sorted_idx = np.argsort(corr)[::-1]
    active = sorted_idx[:top_k]
    g = np.zeros(n)
    for idx in active:
        lag = idx - (n - 1)
        l = abs(lag)
        if l == 0:
            g += 2.0 * h - 1.0
        else:
            g[l:] += (1.0 - h[:-l])
            g[:n - l] -= h[l:]
    return g

def slsqp_optimize(h, dx, target_sum, t=50, maxiter=500):
    n = len(h)
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        return smooth_max_obj(x, t, dx)

    def con_fun(x):
        return np.sum(x) - target_sum

    try:
        res = minimize(
            obj, h.copy(),
            method='SLSQP',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': con_fun}],
            options={'maxiter': maxiter, 'ftol': 1e-12}
        )
        return project_onto_constraints(res.x, target_sum)
    except Exception:
        return h

def group_descent(h, dx, target_sum, group_size=8, n_rounds=40, t=50):
    n = len(h)
    best_c5 = compute_c5(h, dx)
    bounds = [(0.0, 1.0)] * group_size
    for _ in range(n_rounds):
        idx = np.random.choice(n, group_size, replace=False)
        fixed_sum = target_sum - (np.sum(h) - np.sum(h[idx]))
        sub_h = h[idx].copy()

        def sub_obj(x):
            h_sub = h.copy()
            h_sub[idx] = x
            return smooth_max_obj(h_sub, t, dx)

        def sub_con(x):
            return np.sum(x) - fixed_sum

        try:
            res = minimize(
                sub_obj, sub_h.copy(),
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': sub_con}],
                options={'maxiter': 100, 'ftol': 1e-10}
            )
            h_new = h.copy()
            h_new[idx] = np.clip(res.x, 0.0, 1.0)
            h_new = project_onto_constraints(h_new, target_sum)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                h = h_new
                best_c5 = c5_new
        except Exception:
            pass
    return h, best_c5

def subgradient_phase(h, dx, target_sum, n_steps=4000, alpha_init=0.02):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    alpha = alpha_init
    patience = 0
    for step in range(n_steps):
        g = subgradient_direction(best_h, top_k=3)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-12:
            break
        g = g / g_norm
        h_new = project_onto_constraints(best_h - alpha * g, target_sum)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5 - 1e-10:
            best_h = h_new
            best_c5 = c5_new
            alpha = min(alpha * 1.005, 0.08)
            patience = 0
        else:
            alpha = alpha * 0.97
            patience += 1
            if patience > 500:
                break
    return best_h, best_c5

def make_binary_pattern(n, seed_val):
    rng = np.random.RandomState(seed_val)
    h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h
