            c = compute_corr(x)
            max_c = np.max(c[top_idx])
            return (max_c + (1.0/t) * np.log(np.sum(np.exp(t * (c[top_idx] - max_c))))) * dx
        
        try:
            res = minimize(active_obj, h.copy(), method='SLSQP',
                           bounds=[(0.0, 1.0)] * n,
                           constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                           options={'maxiter': 200, 'ftol': 1e-12})
            h_new = fix_sum(res.x, target_sum)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                h = h_new
                best_c5 = c5_new
        except Exception:
            pass
    return h, best_c5

def thue_morse_seq(n):
    """Generate Thue-Morse inspired sequence of length n with n/2 ones"""
    power = int(np.log2(n))
    m = 2 ** (power + 1)
    t = np.zeros(m, dtype=np.float64)
    for i in range(1, m):
        t[i] = t[i >> 1] ^ (i & 1)
    # Take first n or repeat
    if m >= n:
        h = t[:n].copy()
    else:
        h = np.tile(t, int(n / m) + 1)[:n]
    # Fix count
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones > target:
        one_pos = np.where(h == 1)[0]
        remove = np.random.choice(one_pos, n_ones - target, replace=False)
        h[remove] = 0
    elif n_ones < target:
        zero_pos = np.where(h == 0)[0]
        add = np.random.choice(zero_pos, target - n_ones, replace=False)
        h[add] = 1
    return h

def make_random_binary(n, seed_val):
    rng = np.random.RandomState(seed_val)
    h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h

def make_block_pattern(n, seed_val):
    """Alternating blocks of varying sizes"""
    rng = np.random.RandomState(seed_val)
    h = np.zeros(n)
    pos = 0
    bit = 1
    while pos < n:
        blen = rng.choice([2, 3, 4, 5, 6, 7])
        end = min(pos + blen, n)
        h[pos:end] = bit
        pos = end
        bit = 1 - bit
    # Fix count
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones > target:
        one_pos = np.where(h == 1)[0]
        remove = rng.choice(one_pos, n_ones - target, replace=False)
        h[remove] = 0
    elif n_ones < target:
        zero_pos = np.where(h == 0)[0]
        add = rng.choice(zero_pos, target - n_ones, replace=False)
        h[add] = 1
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    def elapsed():
        return time.time() - t0
    
    # Start from initial values
    n = len(initial_h_values)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = np.clip(np.array(initial_h_values, dtype=float), 0.0, 1.0)
    h0 = fix_sum(h0, target_sum)
    
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)
    best_n = n
    
    # Phase 1: Multi-temperature SLSQP on initial solution
    for t in [5, 10, 30, 80, 200, 500]:
        if elapsed() > budget_s - 300:
            break
        h_new = slsqp_opt(best_h, dx, target_sum, t=t, maxiter=600)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
    
    # Phase 2: Active set refinement
    if elapsed() < budget_s - 250:
        h_new, c5_new = active_set_refine(best_h, dx, target_sum, n_iters=20, n_active=5)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
    
    # Phase 3: Binary local search on rounded and continuous versions
    if elapsed() < budget_s - 200:
        # Round to binary
        h_bin = (best_h > 0.5).astype(float)
        h_bin = fix_sum(h_bin, target_sum)
        h_bin, c5_bin = binary_local_search(h_bin, dx, target_sum, n_iters=3000)
        if c5_bin < best_c5:
            best_h = h_bin
            best_c5 = c5_bin
    
    if elapsed() < budget_s - 150:
        # Simulated annealing
        h_sa = (best_h > 0.5).astype(float)
        h_sa = fix_sum(h_sa, target_sum)
        h_sa, c5_sa = simulated_annealing(h_sa, dx, target_sum, n_iters=8000, t_start=0.008, t_end=0.0005)
        if c5_sa < best_c5:
            best_h = h_sa
            best_c5 = c5_sa
        # Follow up with local search
        h_sa2, c5_sa2 = binary_local_search(h_sa, dx, target_sum, n_iters=3000)
        if c5_sa2 < best_c5:
            best_h = h_sa2
            best_c5 = c5_sa2
    
    # Phase 4: Try diverse starting points
    if elapsed() < budget_s - 100:
        starts = []
        # Thue-Morse
        starts.append(thue_morse_seq(n))
        # Random binary
        for s in range(5):
            starts.append(make_random_binary(n, seed + 7000 + s * 111))
        # Block patterns
        for s in range(5):
            starts.append(make_block_pattern(n, seed + 9000 + s * 222))
        
        for h_start in starts:
            if elapsed() > budget_s - 60:
                break
            c5_s = compute_c5(h_start, dx)
            if c5_s < best_c5 + 0.02:
                # Promising start, optimize
                h_opt = slsqp_opt(h_start, dx, target_sum, t=30, maxiter=400)
                c5_opt = compute_c5(h_opt, dx)
                if c5_opt < best_c5:
                    best_h = h_opt
                    best_c5 = c5_opt
                # Binary local search on the rounded version
                h_round = (h_opt > 0.5).astype(float)
                h_round = fix_sum(h_round, target_sum)
                h_round, c5_round = binary_local_search(h_round, dx, target_sum, n_iters=2000)
                if c5_round < best_c5:
                    best_h = h_round
                    best_c5 = c5_round
    
    # Phase 5: Re-continuous refinement after binary search
    if elapsed() < budget_s - 60:
        for t in [50, 150, 400]:
            if elapsed() > budget_s - 30:
                break
            h_new = slsqp_opt(best_h, dx, target_sum, t=t, maxiter=800)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
    
    # Phase 6: Final active set + binary polish
    if elapsed() < budget_s - 30:
        h_new, c5_new = active_set_refine(best_h, dx, target_sum, n_iters=10, n_active=3)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
    
    if elapsed() < budget_s - 15:
        h_bin = (best_h > 0.5).astype(float)
        h_bin = fix_sum(h_bin, target_sum)
        h_bin, c5_bin = binary_local_search(h_bin, dx, target_sum, n_iters=5000)
        if c5_bin < best_c5:
            best_h = h_bin
            best_c5 = c5_bin
    
    # Final: one more SLSQP polish
    if elapsed() < budget_s - 5:
        h_new = slsqp_opt(best_h, dx, target_sum, t=800, maxiter=1500)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
    
    # Ensure valid
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = fix_sum(best_h, best_n / 2.0)
    dx_final = 2.0 / best_n
    best_c5 = compute_c5(best_h, dx_final)
    
    return best_h, best_c5, best_n
