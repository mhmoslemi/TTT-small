import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = x - m
    return m + np.log(np.sum(np.exp(beta * shifted))) / beta

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        s = h.sum()
        if abs(s - target_sum) < 1e-13:
            break
        h = h + (target_sum - s) / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def make_periodic(base, n):
    p = len(base)
    reps = n // p
    h = np.tile(base, reps)
    if len(h) < n:
        h = np.concatenate([h, base[:n - len(h)]])
    return h

class C5Obj:
    def __init__(self, n, beta):
        self.n = n
        self.beta = beta
    def evaluate(self, h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        m = np.max(corr)
        sm = m + np.log(np.sum(np.exp(self.beta * (corr - m)))) / self.beta
        return sm * (2.0 / self.n)
    def c_fun(self, h):
        return h.sum() - self.n / 2.0
    def c_jac(self, h):
        return np.ones(self.n)

class PeriodicC5:
    def __init__(self, n, p):
        self.n = n
        self.p = p
        self.reps = n // p
        self.target = (n / 2.0) / self.reps
    def evaluate(self, base):
        h = make_periodic(base, self.n)
        corr = np.correlate(h, 1.0 - h, mode="full")
        m = np.max(corr)
        sm = m + np.log(np.sum(np.exp(100.0 * (corr - m)))) / 100.0
        return sm * (2.0 / self.n)
    def c_fun(self, base):
        return base.sum() - self.target
    def c_jac(self, base):
        return np.ones(self.p)

def solve_slsqp(obj, con_fun, con_jac, x0, bounds, maxiter=500, ftol=1e-12):
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bounds,
                      constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                      options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return x0

def optimize_periodic(n, p, rng, time_limit=20.0):
    start = time.time()
    if n % p != 0:
        return None, 1.0
    prob = PeriodicC5(n, p)
    bounds = [(0.0, 1.0)] * p
    best_c5 = 1.0
    best_base = np.full(p, prob.target / p)
    n_starts = min(30, max(10, int((time_limit - 0.1) * 4)))
    for i in range(n_starts):
        if time.time() - start > time_limit:
            break
        x0 = np.random.uniform(0.1, 0.9, p)
        x0 = x0 * (prob.target / x0.sum())
        x0 = np.clip(x0, 0, 1)
        res_x = solve_slsqp(prob.evaluate, prob.c_fun, prob.c_jac, x0, bounds, 2000, 1e-14)
        h = make_periodic(res_x, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_base = res_x.copy()
    h = make_periodic(best_base, n)
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()

    h_global = None
    if 'initial_h_values' in globals() and initial_h_values is not None:

    def time_left():
        return budget_s - (time.time() - start_time)

    best_c5 = 1.0
    best_h = None

    def try_update(h, c5):
        nonlocal best_c5, best_h
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 1: Optimize 5-periodic and 10-periodic patterns
    for p, n_list in [(5, [50, 100, 150]), (10, [50, 100]), (2, [50, 100]), (3, [90, 120])]:

            best_h = h_new.copy()

        else:
            no_improve += 1

    if time_left() > 25:
        n_current = len(best_h)
        n_candidates = [x for x in [n_current-5, n_current+5, n_current+10,
