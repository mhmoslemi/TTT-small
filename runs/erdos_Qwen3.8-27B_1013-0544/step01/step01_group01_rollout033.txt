    shifted = beta * (x - m)

        if c5_tmp < c5_best:

        h_best, c5_tmp = try_slqp(h_best, n, target_sum, dx, 300, 500)
        if c5_tmp < c5_best:
            c5_best = c5_tmp

    # Phase 3: Diversified iteration
    no_improve = 0
    attempt = 0
    while tl() > 25 and no_improve < 60:
        attempt += 1
        h0 = h_best.copy()
        strat = attempt % 6

        if strat == 0:
            h0 = targeted_perturb(h0, n, 0.02 + 0.02 * np.exp(-attempt / 40.0))
        elif strat == 1:
            h0 += np.random.randn(n) * (0.03 + 0.05 * np.exp(-attempt / 50.0))
        elif strat == 2:
            for _ in range(np.random.randint(3, 8)):
                i, j = np.random.choice(n, 2, replace=False)
                h0[i], h0[j] = h0[j], h0[i]
        elif strat == 3:
            blen = np.random.randint(4, max(5, n // 5))
            bs = np.random.randint(0, n - blen)
            h0[bs:bs + blen] = np.random.random(blen) * 0.5 + 0.25
        elif strat == 4:
            mask = np.random.random(n) < 0.15
            h0[mask] = np.random.random(int(mask.sum()))
        else:
            # Occasionally try from a fresh random start
            rb = fix_binary_sum(make_random_binary(n, np.random), target_sum)
            h0 = project_h(rb + np.random.randn(n) * 0.15, target_sum)

        h0 = project_h(h0, target_sum)

        # Optimize
        h0, c5_new = try_slqp(h0, n, target_sum, dx, 120, 250)
        if c5_new < c5_best:
            h_best, c5_best = h0.copy(), c5_new
            no_improve = 0
        else:
            no_improve += 1

        # Every 15 attempts, also try L-BFGS-B
        if attempt % 15 == 3 and tl() > 40:
            h_tmp, c5_tmp = try_lbfgs(h_best, n, target_sum, dx, 150, 400)
            if c5_tmp < c5_best:
                h_best, c5_best = h_tmp, c5_tmp
                no_improve = 0

    # Phase 4: Targeted lag reduction
    if tl() > 20:
        for _ in range(3):
            if tl() < 10:
                break
            h0 = targeted_perturb(h_best, n, 0.015)
            h0, c5_new = try_slqp(h0, n, target_sum, dx, 300, 500)
            if c5_new < c5_best:
                h_best, c5_best = h0, c5_new

    # Phase 5: Final high-precision polish
    if tl() > 10:
        h_best, c5_final = try_slqp(h_best, n, target_sum, dx, 800, 3000)
        if c5_final < c5_best:
            c5_best = c5_final
        h_final, c5_final2 = try_lbfgs(h_best, n, target_sum, dx, 1000, 500)
        if c5_final2 < c5_best:
            h_best, c5_best = h_final, c5_final2

    # Verify
    h_best = project_h(h_best, target_sum)
    c5_best = compute_c5(h_best)
    return h_best, c5_best, n
