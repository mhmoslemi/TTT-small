    idx = jp - L
    m = (idx >= 0) & (idx < n)
    a[m] += h[idx[m]]
    return a

def find_optimal_delta(c, a, hj, hjp):
    dmin = max(-hj, hjp - 1.0)
    dmax = min(1.0 - hj, hjp)
    if dmin >= dmax - 1e-15:
        return 0.0
    lo, hi = dmin, dmax
    for _ in range(40):
        mid = 0.5 * (lo + hi)
        vals = c + a * mid
        mx = vals.max()
        active = vals >= mx - 1e-12
        if a[active].min() > 1e-14:
            hi = mid
        elif a[active].max() < -1e-14:
            lo = mid
        else:
            break
    return 0.5 * (lo + hi)

def coord_descent_loop(h, n, rng, max_iters, start_time, time_limit):
    c = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = c.max() * (2.0 / n)
    best_h = h.copy()
    stagnation = 0
    for _ in range(max_iters):
        if time.time() > time_limit:
            break
        j = rng.randint(n)
        jp = rng.randint(n)
        if j == jp:
            continue
        a = compute_coefficients(h, n, j, jp)
        delta = find_optimal_delta(c, a, h[j], h[jp])
        if abs(delta) < 1e-14:
            stagnation += 1
            continue
        h[j] = np.clip(h[j] + delta, 0.0, 1.0)
        h[jp] = np.clip(h[jp] - delta, 0.0, 1.0)
        c = c + a * delta
        cur = c.max() * (2.0 / n)

            best_c5 = cur
            best_h = h.copy()
            stagnation = 0
        else:
            stagnation += 1
        if stagnation > 2000:

            best_h = h.copy()

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    scaled = alpha * (c - m)

def slsqp_polish(h, n, bounds, cons, time_limit):
    for alpha in [500, 2000, 8000]:
