    """Project g onto {h : 0 <= h_i <= 1, sum(h) = target_sum}"""
    n = len(g)
    # Binary search for lambda such that sum(clip(g_i + lambda, 0, 1)) = target_sum
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h = np.clip(g + mid, 0, 1)
        s = np.sum(h)
        if s > target_sum:

    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:

def compute_c5(h, dx):

    return float(np.max(corr) * dx)

def project(h, target_sum):
    """Project h onto {x: 0<=x_i<=1, sum(x)=target_sum}"""
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2
        h_proj = np.clip(h + mid, 0, 1)
        if np.sum(h_proj) > target_sum:
            hi = mid
        else:
            lo = mid
    h_proj = np.clip(h + (lo + hi) / 2, 0, 1)
    # Fine correction
    diff = target_sum - np.sum(h_proj)
    # Add to coordinates that aren't at boundary

    if np.sum(free) > 0:
        h_proj[free] += diff / np.sum(free)
        h_proj = np.clip(h_proj, 0, 1)
    else:

        h_proj = np.clip(h_proj, 0, 1)

def coordinate_descent(h, dx, target_sum, n_rounds=20):
    """Coordinate descent with 1D golden section search."""
    n = len(h)
    best_c5 = compute_c5(h, dx)
    
    for rnd in range(n_rounds):
        improved = False
        for i in range(n):
            old = h[i]
            if old < 1e-6 or old > 1.0 - 1e-6:
                continue
            # Try a few perturbations
            best_t = old
            best_f = best_c5
            for scale in [0.05, 0.02, 0.01, 0.005]:
                for sign in [-1.0, 1.0]:
                    h_try = h.copy()
                    h_try[i] = np.clip(old + sign * scale, 0.0, 1.0)
                    h_try = project_onto_box_sum(h_try, target_sum)
                    f = compute_c5(h_try, dx)
                    if f < best_f:
                        best_f = f
                        best_t = h_try[i]
                        h = h_try.copy()
                        improved = True
        if best_f < best_c5:
            best_c5 = best_f
        if not improved:
            break
    
    return h, best_c5

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_onto_box_sum(h, target_sum):
    n = len(h)
    lo, hi = -3.0, 3.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        h_p = np.clip(h + mid, 0.0, 1.0)
        if h_p.sum() > target_sum:
            hi = mid
        else:
            lo = mid
    h_p = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

        if free.sum() > 0:
            h_p[free] += diff / float(free.sum())
        else:
            h_p += diff / n
        h_p = np.clip(h_p, 0.0, 1.0)
