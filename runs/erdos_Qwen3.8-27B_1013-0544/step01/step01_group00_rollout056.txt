import numpy as np
from scipy.optimize import linprog, minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_h(h, target):
    h = h.copy()
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt > 0:
                h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt > 0:
                h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def block_lp(h, free_mask):
    n = len(h)
    dx = 2.0 / n
    free_idx = np.where(free_mask)[0]
    m = len(free_idx)
    if m == 0:
        return h, compute_c5(h)
    
    fixed_sum = np.sum(h[~free_mask])
    target_sum = n * 0.5 - fixed_sum
    
    # Check feasibility
    if target_sum < -1e-10 or target_sum > m + 1e-10:
        return h, compute_c5(h)
    target_sum = np.clip(target_sum, 0.0, float(m))
    
    # Compute baseline (all free vars = 0)
    h_base = h.copy()
    h_base[free_mask] = 0.0
    corr_base = np.correlate(h_base, 1.0 - h_base, mode='full') * dx
    
    # Compute coefficients for each free variable
    # For k != n-1: coefficient is exact (linear)
    # For k = n-1: we use the relaxation x(1-x) <= x, so coeff = 1 for each free var
    n_k = 2 * n - 1
    coeff = np.zeros((n_k, m))
    
    for j in range(m):
        h_test = h_base.copy()
        h_test[free_idx[j]] = 1.0
        corr_test = np.correlate(h_test, 1.0 - h_test, mode='full') * dx
        coeff[:, j] = corr_test - corr_base
    
    # For k = n-1, replace coefficient with 1.0 (relaxation x(1-x) <= x)
    # The true coeff at k=n-1 for variable j is: (1 - 2*h_base[j]) but h_base[j]=0, so it's 1
    # Actually wait - when h_base[j] = 0, the true contribution is h[j]*(1-h[j]) = h[j] - h[j]^2
    # The linear part is h[j] with coefficient 1. The quadratic part is -h[j]^2.
    # For the relaxation, we use coefficient 1 (which is the true linear coefficient).
    # So coeff[n-1, j] should already be 1 from our computation!
    # Let me verify: corr_test[n-1] - corr_base[n-1] when h_test[free_idx[j]]=1
    # = (1 - 0) - (0 - 0) + ... hmm
    # At k=n-1: corr[n-1] = sum_i h[i](1-h[i])
    # corr_base[n-1] = sum of h_base[i](1-h_base[i]) for all i
    # corr_test[n-1] = corr_base[n-1] + 1*(1-0) = corr_base[n-1] + 1
    # So coeff[n-1, j] = 1. 
    # But the TRUE contribution is h[j] - h[j]^2, so for h[j]=t, the contribution is t - t^2.
    # The linear approximation at 0 gives coefficient 1, which is correct for the tangent.
    # And since t - t^2 <= t for t in [0,1], using coefficient 1 is a valid upper bound.
    # So coeff[n-1, j] = 1 is exactly what we computed, and it's a valid relaxation. 
    
    # Build LP
    c_vec = np.zeros(m + 1)
    c_vec[-1] = 1.0
    
    A_ub = np.zeros((n_k, m + 1))
    A_ub[:, :m] = -coeff
    A_ub[:, m] = 1.0
    b_ub = -corr_base
    
    A_eq = np.zeros((1, m + 1))
    A_eq[0, :m] = 1.0
    b_eq = np.array([target_sum])
    
    bounds = [(0.0, 1.0)] * m + [(-np.inf, np.inf)]
    
    res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        h_new = h.copy()
        h_new[free_mask] = np.clip(res.x[:m], 0.0, 1.0)
        h_new = project_h(h_new, n * 0.5)
        return h_new, compute_c5(h_new)
    else:
        return h, compute_c5(h)

def alternating_lp(h, n_rounds=100, time_limit=60, t_start=0):
    n = len(h)
    h_best = h.copy()
    c5_best = compute_c5(h_best)
    
    even_mask = np.zeros(n, dtype=bool)
    even_mask[0::2] = True
    odd_mask = ~even_mask
    
    for it in range(n_rounds):
        if time.time() - t_start > time_limit:
            break
        h_new, c5_new = block_lp(h_best, even_mask)
        if c5_new < c5_best - 1e-12:
            h_best, c5_best = h_new, c5_new
        h_new, c5_new = block_lp(h_best, odd_mask)
        if c5_new < c5_best - 1e-12:
            h_best, c5_best = h_new, c5_new
    
    return h_best, c5_best

def block_lp_consecutive(h, block_size, time_limit=30, t_start=0):
    n = len(h)
    h_best = h.copy()
    c5_best = compute_c5(h_best)
    n_blocks = n // block_size
    remainder = n % block_size
    
    for it in range(50):
        if time.time() - t_start > time_limit:
            break
        for b in range(n_blocks):
            mask = np.zeros(n, dtype=bool)
            start = b * block_size
            end = start + block_size
            mask[start:end] = True
            h_new, c5_new = block_lp(h_best, mask)
            if c5_new < c5_best - 1e-12:
                h_best, c5_best = h_new, c5_new
    
    return h_best, c5_best

def t_objective(v):
    return v[-1]

def t_ineq(v):
    n = len(v) - 1
    h = v[:n]
    t = v[-1]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return t - corr

def t_eq_fun(v):
    n = len(v) - 1
    return np.sum(v[:n]) - n * 0.5

def t_eq_jac(v):
    n = len(v) - 1
    j = np.zeros(len(v))
    j[:n] = 1.0
    return j

def optimize_t_var(h0, max_iter=300, ftol=1e-11):
    n = len(h0)
    c5_cur = compute_c5(h0)
    v0 = np.append(h0.copy(), c5_cur + 0.01)
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.5)]
    cons = [
        {'type': 'ineq', 'fun': t_ineq},
        {'type': 'eq', 'fun': t_eq_fun, 'jac': t_eq_jac}
    ]
    res = minimize(t_objective, v0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': ftol})
    h_new = np.clip(res.x[:n], 0.0, 1.0)
    h_new = project_h(h_new, n * 0.5)
    return h_new, compute_c5(h_new)

    rng = np.random.RandomState(seed)
    
    def time_left():
        return budget_s - (time.time() - t0)

    h = np.array(initial_h_values, dtype=float)
    n = len(h)

        if c5_new < c5:
            h_best, c5 = h_new, c5_new
    
    # Phase 3: More alternating LP with different splits
    if time_left() > 200:
        # Use a different split: first half / second half
        half = n // 2
        mask1 = np.zeros(n, dtype=bool)
        mask1[:half] = True
        mask2 = ~mask1
        
        h_cur = h_best.copy()
        c5_cur = c5
        for it in range(100):
            if time_left() < 100:
                break
            h_new, c5_new = block_lp(h_cur, mask1)
            if c5_new < c5_cur - 1e-12:
                h_cur, c5_cur = h_new, c5_new
            h_new, c5_new = block_lp(h_cur, mask2)
            if c5_new < c5_cur - 1e-12:
                h_cur, c5_cur = h_new, c5_new
        if c5_cur < c5:
            h_best, c5 = h_cur, c5_cur
    
    # Phase 4: SLSQP t-variable polish
    if time_left() > 80:
        h_new, c5_new = optimize_t_var(h_best, max_iter=300, ftol=1e-11)
        if c5_new < c5:
            h_best, c5 = h_new, c5_new

        h_p = h_best + rng.randn(n) * scale
        h_p = project_h(h_p, n * 0.5)
        h_new, c5_new = alternating_lp(h_p, n_rounds=50, time_limit=40, t_start=time.time())

import numpy as np
from scipy.optimize import linprog, minimize
import time


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))


def project_h(h, target):
    h = h.copy()
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt > 0:
                h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt > 0:
                h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)


def block_lp(h, free_mask):
    n = len(h)
    dx = 2.0 / n
    free_idx = np.where(free_mask)[0]
    m = len(free_idx)
    if m == 0:
        return h, compute_c5(h)
    fixed_sum = np.sum(h[~free_mask])
    target_sum = n * 0.5 - fixed_sum
    if target_sum < -1e-10 or target_sum > m + 1e-10:
        return h, compute_c5(h)
    target_sum = float(np.clip(target_sum, 0.0, m))
    h_base = h.copy()
    h_base[free_mask] = 0.0
    corr_base = np.correlate(h_base, 1.0 - h_base, mode='full') * dx
    n_k = 2 * n - 1
    coeff = np.zeros((n_k, m))
    for j in range(m):
        h_test = h_base.copy()
        h_test[free_idx[j]] = 1.0
        corr_test = np.correlate(h_test, 1.0 - h_test, mode='full') * dx
        coeff[:, j] = corr_test - corr_base
    c_vec = np.zeros(m + 1)
    c_vec[-1] = 1.0
    A_ub = np.zeros((n_k, m + 1))
    A_ub[:, :m] = -coeff
    A_ub[:, m] = 1.0
    b_ub = -corr_base
    A_eq = np.zeros((1, m + 1))
    A_eq[0, :m] = 1.0
    b_eq = np.array([target_sum])
    bounds = [(0.0, 1.0)] * m + [(-np.inf, np.inf)]
    res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        h_new = h.copy()
        h_new[free_mask] = np.clip(res.x[:m], 0.0, 1.0)
        h_new = project_h(h_new, n * 0.5)
        return h_new, compute_c5(h_new)
    return h, compute_c5(h)


def alternating_lp(h, masks, n_rounds=100, time_limit=60, t_start=0):
    n = len(h)
    h_best = h.copy()
    c5_best = compute_c5(h_best)
    for it in range(n_rounds):
        if time.time() - t_start > time_limit:
            break
        converged = True
        for mask in masks:
            h_new, c5_new = block_lp(h_best, mask)
            if c5_new < c5_best - 1e-12:
                h_best, c5_best = h_new, c5_new
                converged = False
        if converged:
            break
    return h_best, c5_best


def t_objective(v):
    return v[-1]


def t_ineq(v):
    n = len(v) - 1
    h = v[:n]
    t = v[-1]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return t - corr


def t_eq_fun(v):
    n = len(v) - 1
    return np.sum(v[:n]) - n * 0.5


def t_eq_jac(v):
    n = len(v) - 1
    j = np.zeros(len(v))
    j[:n] = 1.0
    return j


def optimize_t_var(h0, max_iter=300, ftol=1e-11):
    n = len(h0)
    c5_cur = compute_c5(h0)
    v0 = np.append(h0.copy(), c5_cur + 0.01)
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.5)]
    cons = [
        {'type': 'ineq', 'fun': t_ineq},
        {'type': 'eq', 'fun': t_eq_fun, 'jac': t_eq_jac}
    ]
    res = minimize(t_objective, v0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': ftol})
    h_new = np.clip(res.x[:n], 0.0, 1.0)
    h_new = project_h(h_new, n * 0.5)
    return h_new, compute_c5(h_new)


def smooth_obj(h, temp):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    return m + temp * np.log(np.sum(np.exp((corr - m) / temp)))


def sum_eq_fun(h):
    return np.sum(h) - len(h) * 0.5


def sum_eq_jac(h):
    return np.ones(len(h))


def optimize_smooth(h0, temp, max_iter=200):
    bounds = [(0.0, 1.0)] * len(h0)
    cons = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac}]
    res = minimize(smooth_obj, h0.copy(), args=(temp,), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12})
    h_new = np.clip(res.x, 0.0, 1.0)
    h_new = project_h(h_new, len(h0) * 0.5)
    return h_new, compute_c5(h_new)
