import numpy as np
import time

def compute_c5(h):
    n = len(h)
    return np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n)

def project_box_sum(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    m = corr.max()
    z = corr - m
    w = np.exp(alpha * z)
    sw = w.sum()
    if sw < 1e-300:
        w = np.zeros_like(corr)
        w[corr.argmax()] = 1.0
    else:
        w /= sw
    active = np.where(w > 1e-8)[0]
    grad = np.zeros(n)
    for idx in active:
        k = idx - (n - 1)
        wi = w[idx]
        if k >= 0:
            grad[k:] += wi * (1.0 - h[:n - k])
            if k > 0:
                grad[:n - k] -= wi * h[k:]
        else:
            p = -k
            grad[:n - p] += wi * (1.0 - h[p:])
            if p > 0:
                grad[p:] -= wi * h[:n - p]
    return grad * (2.0 / n)

def local_opt(h0, target_sum, deadline, rng, alpha=100.0, lr0=0.05):
    h = project_box_sum(h0, target_sum)
    best = h.copy()
    best_c5 = compute_c5(best)
    lr = lr0
    patience = 0
    while time.time() < deadline:
        g = smooth_max_grad(h, alpha)
        h_new = h - lr * g
        h_new = project_box_sum(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best = h_new
            best_c5 = c5_new
            h = h_new
            lr = min(lr * 1.06, 0.3)
            patience = 0
        else:
            patience += 1
            lr *= 0.92
            if patience > 50:
                break
    return best, best_c5

def antisym_opt(h0_full, deadline, rng):
    n = len(h0_full)
    nh = n // 2
    half = (h0_full[:nh] + (1.0 - h0_full[nh:2 * nh])) / 2.0
    half = np.clip(half, 0.0, 1.0)
    best_half = half.copy()

    def to_full(hh):
        h = np.empty(n)
        h[:nh] = hh
        h[nh:] = 1.0 - hh
        return h

    best_c5 = compute_c5(to_full(best_half))
    lr = 0.03
    alpha = 60.0
    patience = 0
    while time.time() < deadline:
        hf = to_full(best_half)
        gf = smooth_max_grad(hf, alpha)
        g_half = gf[:nh] - gf[nh:]
        new_half = best_half - lr * g_half
        new_half = np.clip(new_half, 0.0, 1.0)
        c5_check = compute_c5(to_full(new_half))
        if c5_check < best_c5 - 1e-15:
            best_half = new_half
            best_c5 = c5_check
            patience = 0
            lr = min(lr * 1.06, 0.3)
            if alpha < 250:
                alpha *= 1.003
        else:
            patience += 1
            lr *= 0.93
            if patience > 70:
                alpha = min(alpha * 1.3, 300.0)
                lr = 0.03
                patience = 0
    return to_full(best_half), best_c5
