import numpy as np
import time

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    return np.max(get_corr(h)) * (2.0 / n)

def smooth_grad(h, alpha):
    n = len(h)
    corr = get_corr(h)
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -500.0, 500.0)
    w = np.exp(shifted)
    s = np.sum(w)
    if s < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= s
    grad = np.zeros(n)
    for ki in range(len(w)):
        if w[ki] < 1e-11:
            continue
        k = ki - (n - 1)
        wk = w[ki]
        if k >= 0:
            grad[k:] += wk * (1.0 - h[:n - k])
            if k > 0:
                grad[:n - k] -= wk * h[k:]
        else:
            p = -k
            grad[:n - p] += wk * (1.0 - h[p:])
            if p > 0:
                grad[p:] -= wk * h[:n - p]
    return grad * (2.0 / n)

def project(h, target):
    lo, hi = -1.5, 1.5
    for _ in range(45):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def optimize_alpha(h0, target, deadline, rng, alpha0, alpha_max, lr_init, mom_init):
    n = len(h0)
    x = project(h0.astype(float).copy(), target)
    x_prev = x.copy()
    best_x = x.copy()
    best_val = compute_c5(best_x)
    mom = mom_init
    lr = lr_init
    alpha = alpha0
    stall = 0
    max_stall = 40 + int(20 * rng.rand())
    pert_count = 0

    while time.time() < deadline:
        # Momentum extrapolation
        y = x + mom * (x - x_prev)
        g = smooth_grad(y, alpha)
        new_x = x - lr * g
        new_x = project(new_x, target)
        val = compute_c5(new_x)

        if val < best_val - 1e-15:
            best_x = new_x.copy()
            best_val = val
            stall = 0
            if lr < 0.04:
                lr *= 1.015
        else:
            stall += 1
            if stall > max_stall:
                pert_count += 1
                eps = 0.008 + 0.012 * min(pert_count, 5)
                x_p = best_x + rng.randn(n) * eps
                x_p = project(x_p, target)
                v_p = compute_c5(x_p)
                if v_p < best_val:
                    best_x = x_p
                    best_val = v_p
                x = best_x.copy()
                x_prev = x.copy()
                lr = lr_init * (0.7 + 0.3 * rng.rand())
                alpha = min(alpha * (1.2 + 0.3 * rng.rand()), alpha_max)
                mom = mom_init + 0.1 * (rng.rand() - 0.5)
                stall = 0
                max_stall = int(max_stall * 1.3)
                if pert_count > 12:
                    break

        x_prev = x
        x = new_x
        alpha = min(alpha * 1.003, alpha_max)
        # Slightly decay momentum for stability
        mom = max(0.7, mom - 0.0001)

    return best_x, best_val

def make_block_init(n, rng):
    """Create a block-structured initial guess."""
    h = np.zeros(n)
    # Random block boundaries
    n_blocks = 3 + int(rng.randint(0, 3))
    boundaries = sorted(rng.choice(range(1, n), size=n_blocks - 1, replace=False))
    boundaries = [0] + list(boundaries) + [n]
    for i in range(n_blocks):
        val = rng.rand()
        h[boundaries[i]:boundaries[i + 1]] = val
    return project(h, n / 2.0)

def make_sinusoidal_init(n, rng):
    """Sinusoidal initial guess."""
    x = np.linspace(0, 2, n, endpoint=False)
    freq = 1.0 + rng.rand()
    phase = rng.rand() * 2 * np.pi
    h = 0.5 + 0.25 * np.sin(2 * np.pi * freq * x / 2.0 + phase)
    return project(h, n / 2.0)

    x_new = (np.arange(n_new) + 0.5) / n_new * 2.0
    return np.interp(x_new, x_old, h_old, period=2.0)
