                adj = min(diff, 1.0 - h[i])
                h[i] += adj
                diff -= adj
            else:
                adj = max(diff, -h[i])
                h[i] += adj
                diff -= adj
    return h

def run_slsqp(h_start, n, dx, beta, max_iter):
    global _G_DX, _G_BETA, _G_TARGET
    _G_DX = dx
    _G_BETA = beta
    _G_TARGET = n / 2.0
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': _constraint_eq, 'jac': _constraint_jac}]
    try:
        res = minimize(_objective, h_start, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h_start, n)

def block_perturb(h, n, rng):
    h_new = h.copy()
    n_blocks = rng.randint(1, 4)
    for _ in range(n_blocks):
        start = rng.randint(0, max(1, n - 5))
        length = rng.randint(1, 5)
        h_new[start:start + length] = 1.0 - h_new[start:start + length]
    return project_feasible(h_new, n)

def binary_round(h, n):
    h_bin = np.round(h)
    return project_feasible(h_bin, n)

def run(seed=42, budget_s=1100, **kwargs):
    global _G_DX, _G_BETA, _G_TARGET, _G_N
    start_time = time.time()
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    dx = 2.0 / n
    _G_N = n

    best_h = project_feasible(h0.copy(), n)
    best_c5 = compute_c5(best_h, dx)

    # Phase 1: Coarse-to-fine at n (adaptive soft-max + hard)
    betas = [5, 10, 20, 50, 100, 200, 500, 1000, 2000, 5000]
    phase1_end = budget_s * 0.35
    count = 0
    while time.time() - start_time < phase1_end:
        el = time.time() - start_time
        progress = min(el / phase1_end, 1.0)
        base_idx = int(progress * (len(betas) - 1))
        beta_idx = max(0, min(len(betas) - 1, base_idx + int(rng.randn())))
        beta = betas[beta_idx]

        r = rng.random()
        if r < 0.65:
            h_try = best_h.copy()
        elif r < 0.85:
            scale = 0.08 * (1.0 - progress) + 0.005
            h_try = project_feasible(best_h + rng.randn(n) * scale, n)
        else:
            h_try = block_perturb(best_h, n, rng)

        h_new = run_slsqp(h_try, n, dx, beta, max_iter=350)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5 - 1e-15:
            best_h = h_new
            best_c5 = c5_new

        # Try binary rounding occasionally
        if count > 3 and count % 7 == 0:
            h_bin = binary_round(best_h, n)
            c5_bin = compute_c5(h_bin, dx)
            if c5_bin < best_c5:
                h_ref = run_slsqp(h_bin, n, dx, 100, max_iter=300)
                c5_ref = compute_c5(h_ref, dx)
                if c5_ref < best_c5:
                    best_h = h_ref
                    best_c5 = c5_ref
        count += 1

    # Phase 2: DE global search
    de_end = budget_s * 0.50
    if time.time() - start_time < de_end:
        _G_DX = dx
        _G_N = n
        bounds_de = []
        for i in range(n):
            lo = max(0.0, best_h[i] - 0.35)
            hi = min(1.0, best_h[i] + 0.35)
            bounds_de.append((lo, hi))
        try:
            de_res = differential_evolution(
                _de_objective, bounds_de, seed=seed,
                maxiter=12, popsize=10, tol=1e-7,
                mutation=(0.4, 1.0), recombination=0.8, polish=False
            )
            h_de = project_feasible(de_res.x, n)
            c5_de = compute_c5(h_de, dx)
            if c5_de < best_c5:
                h_ref = run_slsqp(h_de, n, dx, 200, max_iter=500)
                c5_ref = compute_c5(h_ref, dx)
                if c5_ref < best_c5:
                    best_h = h_ref
                    best_c5 = c5_ref
                elif c5_de < best_c5:
                    best_h = h_de
                    best_c5 = c5_de
        except Exception:
            pass

    # Phase 3: Upsample to 2n
    up_end = budget_s * 0.78
    if time.time() - start_time < up_end:
        n2 = 2 * n
        dx2 = 2.0 / n2
        _G_N = n2
        h_up = np.repeat(best_h, 2)
        h_up = project_feasible(h_up, n2)
        c5_up = compute_c5(h_up, dx2)
        if c5_up < best_c5:
            best_h = h_up.copy()
            best_c5 = c5_up

        betas2 = [10, 30, 100, 300, 1000, 3000]
        count2 = 0
        while time.time() - start_time < up_end:
            el = time.time() - start_time
            p2_start = budget_s * 0.50
            progress2 = 0.3 + 0.7 * min(1.0, (el - p2_start) / max(up_end - p2_start, 1.0))
            beta_idx = min(int(progress2 * len(betas2)), len(betas2) - 1)
            beta = betas2[beta_idx]

            if count2 == 0 or rng.random() < 0.7:
                h_try2 = best_h.copy()
            else:
                h_try2 = project_feasible(best_h + rng.randn(n2) * 0.04, n2)

            h_new2 = run_slsqp(h_try2, n2, dx2, beta, max_iter=250)
            c5_new2 = compute_c5(h_new2, dx2)
            if c5_new2 < best_c5 - 1e-15:
                best_h = h_new2
                best_c5 = c5_new2
            count2 += 1

    # Phase 4: Final polishing at best resolution
    n_final = len(best_h)
    dx_final = 2.0 / n_final
    _G_N = n_final
    polish_end = budget_s - 12.0
    while time.time() - start_time < polish_end:
        beta = 0 if rng.random() < 0.5 else 5000
        h_new = run_slsqp(best_h, n_final, dx_final, beta, max_iter=450)
        c5_new = compute_c5(h_new, dx_final)
        if c5_new < best_c5 - 1e-15:
            best_h = h_new
            best_c5 = c5_new

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return (best_h, best_c5, len(best_h))
