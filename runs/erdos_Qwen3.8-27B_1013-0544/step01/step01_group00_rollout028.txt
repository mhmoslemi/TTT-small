def project_h(h, target_sum):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):

        if abs(diff) < 1e-12:

            mask = h < 1.0
        else:

            break

def lse_obj_and_grad(h, temp, lam):
    n = len(h)
    dx = 2.0 / n

    w = exp_vals / np.sum(exp_vals)
    lse = c_max + temp * np.log(np.sum(exp_vals))
    sum_dev = np.sum(h) - n * 0.5
