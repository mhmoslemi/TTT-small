def smooth_max_obj_and_grad(h, tau):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    # smooth max value
    x = tau * corr
    x_max = np.max(x)
    log_sum = np.log(np.sum(np.exp(x - x_max)))
    obj = (x_max + log_sum) / tau
    
    # softmax weights
    w = np.exp(x - x_max)
    w /= np.sum(w)
    
    # gradient: sum_k w_k * dcorr[k]/dh[j]
    # dcorr[k]/dh[j] = (1-h[j+k-(n-1)]) - h[j-k+(n-1)] (for valid indices)
    grad = np.zeros(n)
    for k in range(len(corr)):
        if w[k] < 1e-15:
            continue
        shift = k - (n - 1)
        # Contribution from h[j] as the first factor: (1-h[j+shift]) for j+shift in [0,n)
        # Contribution from h[j] as the second factor: -h[j-shift] for j-shift in [0,n)
        # First: j+shift in [0,n) => j in [-shift, n-shift)
        lo1 = max(0, -shift)
        hi1 = min(n, n - shift)
        if lo1 < hi1:
            # 1-h[j+shift] for j in [lo1, hi1)
            grad[lo1:hi1] += w[k] * (1.0 - h[lo1+shift:hi1+shift])
        # Second: j-shift in [0,n) => j in [shift, shift+n)
        lo2 = max(0, shift)
        hi2 = min(n, shift + n)
        if lo2 < hi2:
            grad[lo2:hi2] -= w[k] * h[lo2-shift:hi2-shift]
    
    return obj, grad

        if j_lo < j_hi:
            grad[j_lo:j_hi] += w[k] * (1.0 - h[j_lo + s:j_hi + s])
        j_lo2 = max(0, s)
        j_hi2 = min(n, s + n)
        if j_lo2 < j_hi2:
            grad[j_lo2:j_hi2] -= w[k] * h[j_lo2 - s:j_hi2 - s]
    return obj, grad

def smoo_obj(x, tau):
    return smooth_max_obj_grad(x, tau)[0]

def smoo_grad(x, tau):
    return smooth_max_obj_grad(x, tau)[1]

def sum_fun(x, target):
    return np.sum(x) - target

def sum_jac(x, target):
    return np.ones(len(x))

def fix_sum(h, target):
    h = np.array(h, dtype=float).copy()
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return np.clip(h, 0.0, 1.0)
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = mask.sum()
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def slsqp_opt(h_start, n_pts, tau, maxiter=300, ftol=1e-14):
    target = n_pts / 2.0
    bnds = Bounds(np.zeros(n_pts), np.ones(n_pts))
    cons = [{'type': 'eq', 'fun': sum_fun, 'args': (target,), 'jac': sum_jac}]
    try:
        res = minimize(smoo_obj, h_start, args=(tau,), jac=smoo_grad,
                       method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        x = fix_sum(res.x, target)
        c5 = evaluate_c5(x)
        return x, c5
    except Exception:
        x = fix_sum(h_start, target)
        return x, evaluate_c5(x)

def binary_sa(h_init, t_start=0.5, t_end=0.005, n_iter=3000, rng=None):
    if rng is None:
        rng = np.random.RandomState(42)
    n = len(h_init)
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    current_c5 = best_c5
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for i in range(n_iter):
        t = t_start * (t_end / t_start) ** (i / float(n_iter))
        if len(ones) == 0 or len(zeros) == 0:
            break
        i1 = ones[rng.randint(len(ones))]
        i0 = zeros[rng.randint(len(zeros))]
        h[i1], h[i0] = h[i0], h[i1]
        # Update ones/zeros
        if h[i1] == 1:
            ones[ones == i1] = i0
            zeros[zeros == i0] = i1
        else:
            ones[ones == i0] = i1
            zeros[zeros == i1] = i0
        new_c5 = evaluate_c5(h)
        delta = new_c5 - current_c5
        if delta < 0 or rng.random() < np.exp(-delta / (t + 1e-10)):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i1], h[i0] = h[i0], h[i1]
            if h[i1] == 1:
                zeros[zeros == i1] = i0
                ones[ones == i0] = i1
            else:
                zeros[zeros == i0] = i1
                ones[ones == i1] = i0
    return best_h, best_c5

def pair_swap_search(h, n_iter=2000, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    n = len(h)
    best_c5 = evaluate_c5(h)
    for _ in range(n_iter):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        old_i, old_j = h[i], h[j]
        h[i], h[j] = old_j, old_i
        new_c5 = evaluate_c5(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
        else:
            h[i], h[j] = old_i, old_j
    return h, best_c5

def make_binary_seed(n, period, duty, rng):
    h = np.zeros(n)
    for i in range(n):
        if (i % period) < duty:
            h[i] = 1.0
    # Adjust to get exactly n/2 ones
    ones = np.sum(h)
    target = n // 2
    if ones > target:
        idx = np.where(h == 1)[0]
        rng.shuffle(idx)
        for i in range(int(ones - target)):
            h[idx[i]] = 0.0
    elif ones < target:
        idx = np.where(h == 0)[0]
        rng.shuffle(idx)
        for i in range(int(target - ones)):
            h[idx[i]] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    def rem():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(60, 0.5)

    best_h = h_init.copy()
    best_c5 = evaluate_c5(best_h)
    n0 = len(best_h)

    # Phase 1: Annealing from initial with analytical gradient
    tau_schedule = [5, 20, 80, 400, 2000, 10000, 50000, 200000]
    for tau in tau_schedule:
        if rem() < 15:
            break
        n_iter = 400 if tau < 1000 else 300
        h_new, c5_new = slsqp_opt(best_h, n0, tau, maxiter=n_iter, ftol=1e-15)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 2: Multi-resolution exploration
    res_list = [80, 100, 120, 140, 160, 180, 200]
    for tn in res_list:
        if rem() < 30:
            break
        if tn % 2 != 0:
            continue
        target = tn / 2.0
        if tn == n0:
            h_t = best_h.copy()
        else:
            h_t = np.interp(np.linspace(0, 1, tn), np.linspace(0, 1, n0), best_h)
            h_t = fix_sum(h_t, target)
        for tau in [100, 2000, 20000]:
            if rem() < 15:
                break
            h_new, c5_new = slsqp_opt(h_t, tn, tau, maxiter=250, ftol=1e-14)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
                n0 = tn
            h_t = h_new

    # Phase 3: Binary SA from best solution
    if rem() > 60:
        h_bin = (best_h > 0.5).astype(float)
        h_bin = fix_sum(h_bin, n0 / 2.0)
        h_bin = (h_bin >= 0.5).astype(float)
        # Ensure exactly n/2 ones
        ones_count = int(h_bin.sum())
        if ones_count > n0 // 2:
            idx = np.where(h_bin == 1)[0]
            rng.shuffle(idx)
            h_bin[idx[:ones_count - n0 // 2]] = 0.0
        elif ones_count < n0 // 2:
            idx = np.where(h_bin == 0)[0]
            rng.shuffle(idx)
            h_bin[idx[:n0 // 2 - ones_count]] = 1.0
        h_new, c5_new = binary_sa(h_bin, t_start=0.3, t_end=0.001, n_iter=3000, rng=rng)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
        # Also relax the binary solution
        h_relax = h_new.copy()
        for tau in [500, 5000, 50000]:
            if rem() < 10:
                break
            h_new2, c5_new2 = slsqp_opt(h_relax, n0, tau, maxiter=300)
            if c5_new2 < best_c5:
                best_h = h_new2
                best_c5 = c5_new2
            h_relax = h_new2

    # Phase 4: Diverse starting points
    starts = []
    for period in range(3, 12):
        for duty in range(1, period):
            if period % 2 == 0 and duty == period // 2:
                h_p = make_binary_seed(n0, period, duty, rng)
                starts.append(h_p)
    for s in range(15):
        h_r = make_binary_seed(n0, 2, 1, rng)
        # Mix with best
        h_r = 0.5 * h_r + 0.5 * best_h
        h_r = fix_sum(h_r, n0 / 2.0)
        starts.append(h_r)

    for sp in starts:
        if rem() < 12:
            break
        for tau in [500, 5000]:
            if rem() < 8:
                break
            h_new, c5_new = slsqp_opt(sp, n0, tau, maxiter=200, ftol=1e-13)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
            sp = h_new

    # Phase 5: Pair swap local search
    if rem() > 20:
        h_sw, c5_sw = pair_swap_search(best_h.copy(), n_iter=3000, rng=rng)
        if c5_sw < best_c5:
            best_h = h_sw
            best_c5 = c5_sw
        # Refine after swaps
        for tau in [5000, 50000]:
            if rem() < 8:
                break
            h_new, c5_new = slsqp_opt(best_h, n0, tau, maxiter=300, ftol=1e-15)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    # Phase 6: Random perturbations around best
    for _ in range(20):
        if rem() < 12:
            break
        scale = 0.03 + 0.15 * rng.random()
        hp = best_h + rng.normal(0, scale, n0)
        hp = np.clip(hp, 0.0, 1.0)
        hp = fix_sum(hp, n0 / 2.0)
        for tau in [2000, 20000]:
            if rem() < 8:
                break
            h_new, c5_new = slsqp_opt(hp, n0, tau, maxiter=250, ftol=1e-14)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
            hp = h_new

    # Phase 7: Final high-precision refinement
    for tau in [50000, 200000, 1000000]:
        if rem() < 10:
            break
        h_new, c5_new = slsqp_opt(best_h, n0, tau, maxiter=500, ftol=1e-16)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Final constraint fix and evaluation
    best_h = fix_sum(best_h, n0 / 2.0)
    best_c5 = evaluate_c5(best_h)

    return best_h, best_c5, n0
