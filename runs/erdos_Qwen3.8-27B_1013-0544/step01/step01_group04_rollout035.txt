def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = np.sum(h) - target
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            # Need to decrease sum by diff
            room = h  # can decrease each entry by at most h[i]
            total_room = np.sum(room)
            if total_room < diff:
                # Can't decrease enough, set all to 0 (shouldn't happen)
                h = np.zeros(n)
                break

        elif s == 3:
            h_try = binary_round(best_h)
        elif s == 4:
            h_try = smooth_perturb(best_h, n, rng, 0.12)
        else:
            h_try = block_perturb(best_h, n, rng, rng.randint(10, 40), (0.08, 0.35))
        h_new = slsqp(h_try, n, dx, max_iter=350)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5 - 1e-12:
            best_h, best_c5 = h_new, c5_new
            no_improve = 0
        else:
            no_improve += 1

    # Phase 2: Smooth max with annealing (20%)
    p2_end = start_time + budget_s * 0.68
    for T in [40, 25, 15, 10, 7, 5, 4]:
        if time.time() > p2_end:
            break
        h_new = slsqp(best_h.copy(), n, dx, max_iter=200, T=T)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5 - 1e-12:
            best_h, best_c5 = h_new, c5_new
        if time.time() < p2_end:
            h_p = smooth_perturb(best_h, n, rng, 0.06)
            h_new2 = slsqp(h_p, n, dx, max_iter=200, T=T)
            c5_new2 = compute_c5(h_new2, dx)
            if c5_new2 < best_c5 - 1e-12:
                best_h, best_c5 = h_new2, c5_new2

    # Phase 3: Binary search + resolution (18%)
    p3_end = start_time + budget_s * 0.86
    if time.time() < p3_end:
        h_bin = binary_round(best_h)
        h_bs, _ = binary_local_search(h_bin, n, dx, rng, 40000)
        h_pol = slsqp(h_bs, n, dx, max_iter=400)
        c5_pol = compute_c5(h_pol, dx)
        if c5_pol < best_c5 - 1e-12:
            best_h, best_c5 = h_pol, c5_pol

    # Additional binary search from random starts
    if time.time() < p3_end:
        for _ in range(2):
            if time.time() > p3_end:
                break
            h_r = np.zeros(n)
            idx = rng.choice(n, size=n // 2, replace=False)
            h_r[idx] = 1.0
            h_bs2, _ = binary_local_search(h_r, n, dx, rng, 25000)
            h_pol2 = slsqp(h_bs2, n, dx, max_iter=300)
            c5_2 = compute_c5(h_pol2, dx)
            if c5_2 < best_c5 - 1e-12:
                best_h, best_c5 = h_pol2, c5_2

    # Downsample optimization
    if time.time() < p3_end:
        h_down = downsample_optimize(best_h, n, n // 2, dx, rng, 6)
        if h_down is not None:
            c5_d = compute_c5(h_down, dx)
            if c5_d < best_c5 - 1e-12:
                best_h, best_c5 = h_down, c5_d

    # Phase 4: Fine polishing (12%)
    p4_end = start_time + budget_s - 25
    while time.time() < p4_end:
        if tl() < 35:
            break
        h_try = smooth_perturb(best_h, n, rng, 0.015)
        h_new = slsqp(h_try, n, dx, max_iter=500)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5 - 1e-12:
            best_h, best_c5 = h_new, c5_n if False else c5_new

    # Extra: try a few more smooth max at low T
    if time.time() < p4_end:
        for T in [3, 2]:
            if time.time() > p4_end:
                break
            h_new = slsqp(best_h.copy(), n, dx, max_iter=300, T=T)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5 - 1e-12:
                best_h, best_c5 = h_new, c5_new

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, dx)
    return (best_h, best_c5, n)
