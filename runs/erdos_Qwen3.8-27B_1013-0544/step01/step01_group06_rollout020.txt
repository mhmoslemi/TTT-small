        if no_improve % 30 == 0 and no_improve > 0:
            alpha = min(alpha * 1.4, 2000.0)
        if no_improve % 60 == 0 and no_improve >= 60:
            lr = max(lr * 0.85, 0.001)
        
        if no_improve < 5:
            grad = compute_smooth_grad(h, alpha)
        else:
            blend = min(no_improve / 50.0, 1.0)
            grad_s = compute_smooth_grad(h, alpha)
            grad_d = compute_direct_grad(h)
            grad = (1 - blend) * grad_s + blend * grad_d
        
        if use_adam:
            t_count += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad ** 2
            m_hat = m / (1 - beta1 ** t_count)
            v_hat = v / (1 - beta2 ** t_count)
            step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        else:
            step = lr * grad
        
        h_new = project_onto_simplex_box(h - step, target_sum)
        new_c5 = compute_c5(h_new)
        
        if new_c5 < best_c5 - 1e-15:
            h = h_new
            best_h = h_new.copy()
            best_c5 = new_c5
            no_improve = 0
            fail_count = 0
            if lr < 0.06:
                lr = min(lr * 1.03, 0.06)
        else:
            h = h_new
            no_improve += 1
            if no_improve > 40:
                fail_count += 1
                pert_size = 0.03 + 0.02 * min(fail_count, 5)
                h_pert = project_onto_simplex_box(h + rng.randn(n) * pert_size, target_sum)
                c5_pert = compute_c5(h_pert)
                if c5_pert < best_c5:
                    h = h_pert
                    best_h = h_pert.copy()
                    best_c5 = c5_pert
                    no_improve = 0
                    if use_adam:
                        m = np.zeros(n)
                        v = np.zeros(n)
                        t_count = 0
                    lr = 0.015
                    alpha = alpha_init
                else:
                    no_improve = 0
                    if fail_count > 15:
                        break
    
    return best_h, best_c5

def interpolate_to_n(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False) + 1.0 / n_new
    h_ext = np.append(h_old, h_old[0])
    x_ext = np.append(x_old, 2.0 + 1.0 / n_old)
    return np.interp(x_new, x_ext, h_ext)

def try_random_init(n, target_sum, rng):
    patterns = []
    x = np.linspace(0, 2, n, endpoint=False)
    patterns.append(0.5 + 0.3 * np.sin(np.pi * x))
    patterns.append(0.5 + 0.3 * np.sin(2 * np.pi * x))
    patterns.append(0.5 + 0.2 * np.sin(np.pi * x) + 0.2 * np.sin(2 * np.pi * x))
    patterns.append(0.5 + 0.4 * np.cos(np.pi * x) * np.cos(2 * np.pi * x))
    patterns.append(np.full(n, 0.5))
    for p in patterns:
        p = p + 0.05 * rng.randn(n)
        p = project_onto_simplex_box(p, target_sum)
        patterns[p is p] = p  # no-op, just keeping
    return patterns

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.full(200, 0.5)
        x = np.linspace(0, 2, 200, endpoint=False)
        h0 = h0 + 0.2 * np.sin(np.pi * x) + 0.05 * rng.randn(200)
    
    n = len(h0)
    target_sum = n / 2.0
    h0 = project_onto_simplex_box(h0, target_sum)
    
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    best_n = n
    
    # Phase 1: Main optimization at initial resolution
    deadline1 = t0 + budget_s * 0.40
    best_h, best_c5 = optimize_h(h0, target_sum, deadline1, rng, alpha_init=35, lr=0.02)
    best_n = len(best_h)
    
    # Phase 2: Second pass with higher alpha for polish
    elapsed = time.time() - t0
    remaining = budget_s - elapsed
    if remaining > 60:
        polish_deadline = t0 + budget_s * 0.55
        h_polish, c5_polish = optimize_h(best_h, len(best_h) / 2.0, polish_deadline, rng, alpha_init=100, lr=0.008)
        if c5_polish < best_c5:
            best_h = h_polish
            best_c5 = c5_polish
            best_n = len(best_h)
    
    # Phase 3: Try higher resolutions
    remaining = budget_s - (time.time() - t0)
    resolutions = [300, 400]
    for n_try in resolutions:
        if remaining < 80:
            break
        h_interp = interpolate_to_n(best_h, best_n, n_try)
        h_interp = project_onto_simplex_box(h_interp, n_try / 2.0)
        phase_dl = time.time() + min(remaining * 0.38, 250.0)
        h_opt, c5_opt = optimize_h(h_interp, n_try / 2.0, phase_dl, rng, alpha_init=30, lr=0.015)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
            best_n = n_try
        remaining = budget_s - (time.time() - t0)
    
    # Phase 4: Even higher resolution if time permits
    remaining = budget_s - (time.time() - t0)
    if remaining > 120:
        n_try = 500
        h_interp = interpolate_to_n(best_h, best_n, n_try)
        h_interp = project_onto_simplex_box(h_interp, n_try / 2.0)
        phase_dl = t0 + budget_s - 60
        h_opt, c5_opt = optimize_h(h_interp, n_try / 2.0, phase_dl, rng, alpha_init=30, lr=0.012)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
            best_n = n_try
    
    # Phase 5: Final polish at best resolution with high alpha
    remaining = budget_s - (time.time() - t0)
    if remaining > 25:
        best_h, best_c5 = optimize_h(best_h, len(best_h) / 2.0, t0 + budget_s - 3, rng, alpha_init=300, lr=0.004)
    
    best_h = project_onto_simplex_box(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, len(best_h)
