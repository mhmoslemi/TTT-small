import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_lags(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return c * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-14:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def lag_grad_single(h, n, lag):
    dx = 2.0 / n
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = dx * (1.0 - 2.0 * h)
    elif lag > 0:
        grad[:n - lag] += dx * (1.0 - h[lag:])
        grad[lag:] -= dx * h[:n - lag]
    else:
        nl = -lag
        grad[nl:] += dx * (1.0 - h[:nl])
        grad[:n - nl] -= dx * h[nl:]
    return grad

def active_sum_obj(h, n, lag_list):
    l = compute_lags(h, n)
    idx = np.array([int(lag) + n - 1 for lag in lag_list])
    return float(np.sum(l[idx]))

def active_sum_grad(h, n, lag_list):
    grad = np.zeros(n)
    for lag in lag_list:
        grad += lag_grad_single(h, n, int(lag))
    return grad

def smooth_max_obj(h, n, alpha):
    l = compute_lags(h, n)
    m = l.max()
    shifted = alpha * (l - m)
    mx = shifted.max()
    if mx > 500:
        logsum = mx + np.log(np.sum(np.exp(shifted - mx)))
    else:
        logsum = np.log(np.sum(np.exp(shifted)))
    return float(m + logsum / alpha)

def smooth_max_grad(h, n, alpha):
    l = compute_lags(h, n)
    m = l.max()
    shifted = alpha * (l - m)
    mx = shifted.max()
    exp_s = np.exp(shifted - mx)

    for k in range(len(l)):
        wk = w[k]
        if wk > 1e-12:
            grad += wk * lag_grad_single(h, n, k - (n - 1))
    return grad

    return np.sum(x) - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)
