import numpy as np

from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target_sum):
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
    return np.clip(h, 0.0, 1.0)

def block_obj(x, h_full, idx, target_sum, dx):
    ht = h_full.copy()
    ht[idx] = np.clip(x, 0.0, 1.0)
    ht = fix_sum(ht, target_sum)
    return compute_c5(ht, dx)

def block_descent(h, dx, target_sum, block_size, n_rounds, seed):
    rng = np.random.RandomState(seed)
    n = len(h)
    best_c5 = compute_c5(h, dx)
    for _ in range(n_rounds):

        idx = np.arange(start, start + block_size)
        h_sub = h[idx].copy()

        h_sub = h[idx].copy()
        try:
            res = minimize(block_obj, h_sub, method='Nelder-Mead',
                          args=(h, idx, target_sum, dx),
                          options={'maxiter': 150, 'xatol': 1e-8, 'fatol': 1e-12})
            h_new = h.copy()
            h_new[idx] = np.clip(res.x, 0.0, 1.0)
            h_new = fix_sum(h_new, target_sum)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                h = h_new
                best_c5 = c5_new
                corr = np.correlate(h, 1 - h, mode="full")
                top_lags = np.argsort(corr)[-5:]
        except Exception:
            pass
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = fix_sum(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)
    best_n = n

    # Phase 1: Broad block Nelder-Mead
    if elapsed() < budget_s - 250:
        nr = int(min(150, max(30, (budget_s - elapsed() - 250) / 0.04)))
        h_new, c5_new = block_descent(best_h, dx, target_sum, 6, nr, seed)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 2: Targeted block descent
    if elapsed() < budget_s - 200:
        h_new, c5_new = targeted_block(best_h, dx, target_sum, 60, seed + 1)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 3: Simulated annealing for exploration
    if elapsed() < budget_s - 180:
        h_new, c5_new = sa_optimize(best_h, dx, target_sum,
                                     int(min(12000, (budget_s - elapsed() - 180) / 0.003)),
                                     0.07, 0.001, seed + 2)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 4: Fine block descent (smaller blocks)
    if elapsed() < budget_s - 130:
        h_new, c5_new = block_descent(best_h, dx, target_sum, 4, 100, seed + 3)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 5: Local perturbations
    if elapsed() < budget_s - 100:
        h_new, c5_new = local_perturb(best_h, dx, target_sum, seed + 4, 800)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 6: Try n=320 upsample
    if elapsed() < budget_s - 80:
        n2 = 2 * n
        dx2 = 2.0 / n2
        ts2 = n2 / 2.0
        h2 = upsample(best_h, n2)
        h2 = fix_sum(h2, ts2)
        h2, c5_2 = block_descent(h2, dx2, ts2, 4, 60, seed + 5)
        if c5_2 < best_c5:
            best_h, best_c5, best_n = h2, c5_2, n2

    # Phase 7: SLSQP polish with increasing t
    if elapsed() < budget_s - 25:
        dd = 2.0 / best_n
        ts = best_n / 2.0
        for t in [30, 100, 300, 800]:
            if elapsed() > budget_s - 5:
                break
            h_new = slsqp_polish(best_h, dd, ts, t, 500)
            c5_new = compute_c5(h_new, dd)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new

    # Phase 8: Final fine-tuning
    if elapsed() < budget_s - 3:
        dd = 2.0 / best_n
        ts = best_n / 2.0
        h_new, c5_new = local_perturb(best_h, dd, ts, seed + 10, 500)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h, 2.0 / best_n)
    return best_h, best_c5, best_n
