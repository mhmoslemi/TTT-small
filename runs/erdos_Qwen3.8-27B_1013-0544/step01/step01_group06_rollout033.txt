import numpy as np
import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_simplex(h, target_sum):
    """Project h onto {x : 0 <= x[i] <= 1, sum(x) = target_sum}"""
    lo, hi = -1.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max(corr, alpha):
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha

def smooth_max_grad(corr, alpha):
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= s
    return w

    for idx in range(len(corr)):
        wj = w[idx]
        if wj < 1e-13:
            continue
        k = idx - (n - 1)
        if k >= 0:
            if k < n:
                grad[k:] += wj * (1.0 - h[:n - k])
                grad[:n - k] -= wj * h[k:]
        else:
            p = -k
            if p < n:
                grad[:n - p] += wj * (1.0 - h[p:])
                grad[p:] -= wj * h[:n - p]
    return grad * (2.0 / n)

def smooth_max_val(corr, alpha):
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha

def interpolate_h(h_old, n_old, n_new):
    x_old = (np.arange(n_old) + 0.5) / n_old * 2.0
    x_new = (np.arange(n_new) + 0.5) / n_new * 2.0
    return np.interp(x_new, x_old, h_old)

def make_block_init(n, rng):
    """Create a 50-50 block pattern with pseudo-random block sizes."""
    h = np.zeros(n)
    i = 0
    while i < n:
        blen = rng.randint(2, 8)
        blen = min(blen, n - i)
        if rng.rand() < 0.5:
            h[i:i+blen] = 1.0
        i += blen
    return project_simplex(h, n / 2.0)

def make_sinusoidal_init(n, rng):
    """Create a sinusoidal-based initial with random frequency."""
    x = np.linspace(0, 2, n, endpoint=False)
    f1 = rng.randint(1, 4)
    f2 = rng.randint(2, 7)
    p1 = rng.uniform(0, 2 * np.pi)
    p2 = rng.uniform(0, 2 * np.pi)
    amp = rng.uniform(0.2, 0.4)
    h = 0.5 + amp * np.sin(2 * np.pi * f1 * x + p1) + amp * 0.5 * np.sin(2 * np.pi * f2 * x + p2)
    return project_simplex(h, n / 2.0)

def optimize_momentum(h, target_sum, deadline, rng, alpha_init=60, max_stalls=20):
    n = len(h)
    h = project_simplex(h.astype(float).copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    v = np.zeros(n)
    lr = 0.01
    mu = 0.85
    alpha = alpha_init
    no_improve = 0
    stalls = 0

    while time.time() < deadline and stalls < max_stalls:
        grad = grad_c5(best_h, alpha)
        v = mu * v + (1.0 - mu) * grad
        h_new = project_simplex(best_h - lr * v, target_sum)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5 - 1e-14:
            best_h = h_new
            best_c5 = c5_new
            no_improve = 0
            if lr < 0.06:
                lr = min(lr * 1.04, 0.06)
            mu = min(mu + 0.001, 0.95)
        else:
            no_improve += 1
            lr *= 0.96
            mu = max(mu * 0.995, 0.5)

            if no_improve > 40:
                # Structured perturbation
                ptype = rng.randint(4)
                if ptype == 0:
                    scale = 0.015 + 0.01 * stalls
                    h_pert = best_h + rng.randn(n) * scale
                elif ptype == 1:
                    h_pert = best_h.copy()
                    bs = rng.randint(3, max(4, n // 6))
                    s = rng.randint(0, n - bs)
                    h_pert[s:s+bs] = 1.0 - h_pert[s:s+bs]
                elif ptype == 2:
                    freq = rng.randint(1, 6)
                    ph = rng.uniform(0, 2 * np.pi)
                    h_pert = best_h + 0.04 * np.sin(2 * np.pi * freq * np.arange(n) / n + ph)
                else:
                    # Two-point swap
                    i1 = rng.randint(n)
                    i2 = rng.randint(n)
                    h_pert = best_h.copy()
                    h_pert[i1], h_pert[i2] = h_pert[i2], h_pert[i1]

                h_pert = project_simplex(h_pert, target_sum)
                c5_pert = compute_c5(h_pert)

                if c5_pert < best_c5:
                    best_h = h_pert
                    best_c5 = c5_pert
                    v = np.zeros(n)
                    lr = 0.01
                    mu = 0.85
                else:
                    alpha = min(alpha * 1.15, 400.0)
                    v = np.zeros(n)
                    lr = 0.008
                    mu = 0.8

                no_improve = 0
                stalls += 1

    return best_h, best_c5

def optimize_annealing(h, target_sum, deadline, rng, alpha_init=80, max_stalls=30):
    """Gradient + annealing hybrid with more aggressive exploration."""
    n = len(h)
    h = project_simplex(h.astype(float).copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    v = np.zeros(n)
    lr = 0.015
    mu = 0.8
    alpha = alpha_init
    no_improve = 0
    stalls = 0
    temp = 0.02

    while time.time() < deadline and stalls < max_stalls:
        if temp > 0.005 and rng.rand() < 0.15:
            # Annealing step
            h_cand = best_h + rng.randn(n) * temp
            h_cand = project_simplex(h_cand, target_sum)
            c5_cand = compute_c5(h_cand)
            if c5_cand < best_c5:
                best_h = h_cand
                best_c5 = c5_cand
            temp *= 0.97
        else:
            grad = grad_c5(best_h, alpha)
            v = mu * v + (1.0 - mu) * grad
            h_new = project_simplex(best_h - lr * v, target_sum)
            c5_new = compute_c5(h_new)

            if c5_new < best_c5 - 1e-14:
                best_h = h_new
                best_c5 = c5_new
                no_improve = 0
                if lr < 0.05:
                    lr = min(lr * 1.03, 0.05)
            else:
                no_improve += 1
                lr *= 0.95
                mu = max(mu * 0.998, 0.4)

                if no_improve > 35:
                    # Multi-scale perturbation
                    h_pert = best_h.copy()
                    for _ in range(rng.randint(1, 4)):
                        bs = rng.randint(2, max(3, n // 8))
                        s = rng.randint(0, n - bs)
                        h_pert[s:s+bs] += rng.randn(bs) * 0.03
                    h_pert = project_simplex(h_pert, target_sum)
                    c5_pert = compute_c5(h_pert)

                    if c5_pert < best_c5:
                        best_h = h_pert
                        best_c5 = c5_pert
                        v = np.zeros(n)
                        lr = 0.015
                        mu = 0.8
                    else:
                        alpha = min(alpha * 1.2, 500.0)
                        v = np.zeros(n)
                        lr = 0.006

                    no_improve = 0
                    stalls += 1

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    # Get initial values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = make_block_init(200, rng)

    n_init = len(h_init)
    best_h = project_simplex(h_init.copy(), n_init / 2.0)
    best_c5 = compute_c5(best_h)
    best_n = n_init

    # Phase 1: Optimize at initial resolution (45% of budget)
    deadline1 = t0 + budget_s * 0.45
    best_h, best_c5 = optimize_momentum(best_h, best_n / 2.0, deadline1, rng, alpha_init=60)
    best_n = len(best_h)

    # Phase 2: Try alternative initializations at same n (15%)
    remaining = budget_s - (time.time() - t0)
    if remaining > 60:
        deadline2 = t0 + budget_s * 0.60
        alts = [
            make_block_init(best_n, rng),
            make_sinusoidal_init(best_n, rng),
            project_simplex(0.5 + 0.3 * rng.randn(best_n), best_n / 2.0),
        ]
        for alt in alts:
            if time.time() > deadline2:
                break
            phase_dl = min(time.time() + 50, deadline2 - 5)
            h_opt, c5_opt = optimize_momentum(alt, best_n / 2.0, phase_dl, rng, alpha_init=50)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt

    # Phase 3: Try different resolutions (20%)
    remaining = budget_s - (time.time() - t0)
    for n_try in [100, 300, 150]:
        if remaining < 50:
            break
        if n_try == best_n:
            continue
        h_interp = interpolate_h(best_h, best_n, n_try)
        h_interp = project_simplex(h_interp, n_try / 2.0)
        phase_dl = min(t0 + budget_s * 0.82, time.time() + min(remaining * 0.4, 120))
        h_opt, c5_opt = optimize_annealing(h_interp, n_try / 2.0, phase_dl, rng, alpha_init=50)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
            best_n = n_try
        remaining = budget_s - (time.time() - t0)

    # Phase 4: Final polish at best resolution (remaining time)
    final_dl = t0 + budget_s - 3
    if time.time() < final_dl:
        best_h, best_c5 = optimize_momentum(best_h, best_n / 2.0, final_dl, rng, alpha_init=150, max_stalls=10)

    # Ensure constraint is met
    best_h = project_simplex(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
