**Pair swap coordinate descent:**
- Pick two indices i, j
- Try h[i] + δ, h[j] - δ (maintains sum)
- Choose δ to minimize the objective
- This is a 1D optimization problem

The objective as a function of δ is:
f(δ) = max_k corr_k(h + δ(e_i - e_j))

Each corr_k is linear in δ (since correlation is linear in h). So f(δ) is the max of linear functions, which is convex in δ. We can minimize it by finding where the subgradient contains 0.

For a given pair (i, j), the change in corr_k when we add δ to h[i] and subtract δ from h[j]:
Δcorr_k = δ * [∂corr_k/∂h[i] - ∂corr_k/∂h[j]]

So corr_k(δ) = corr_k(0) + δ * d_k where d_k = ∂corr_k/∂h[i] - ∂corr_k/∂h[j]

We want to minimize max_k (corr_k(0) + δ * d_k) over δ such that:
- 0 ≤ h[i] + δ ≤ 1
- 0 ≤ h[j] - δ ≤ 1

This is a 1D convex minimization problem! We can solve it in O(n) time by finding the intersection points.

```python
def pair_swap_fast(h, i, j, n):
    """Fast pair swap: precompute the directional derivatives"""
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full')  # length 2n-1
    
    # Compute d_k = ∂corr_k/∂h[i] - ∂corr_k/∂h[j] for all k
    # ∂corr_k/∂h[m]:
    # corr_k corresponds to lag = k - (n-1)
    # ∂corr[lag]/∂h[m] = (1-h[m+lag])·[0≤m+lag<n] - h[m-lag]·[0≤m-lag<n]
    
    delta_corr = np.zeros(2*n-1)
    for k_idx in range(2*n-1):
        lag = k_idx - (n - 1)
        gi = 0.0
        gj = 0.0
        if 0 <= i + lag < n:
            gi += (1.0 - h[i + lag])
        if 0 <= i - lag < n:
            gi -= h[i - lag]
        if 0 <= j + lag < n:
            gj += (1.0 - h[j + lag])
        if 0 <= j - lag < n:
            gj -= h[j - lag]
        delta_corr[k_idx] = gi - gj
    
    # Now corr_k + d * delta_corr_k is the correlation at delta
    # We want to minimize max_k (corr_k + d * delta_corr_k)
    # subject to delta_lo <= d <= delta_hi
    
    delta_lo = max(-h[i], h[j] - 1.0)
    delta_hi = min(1.0 - h[i], h[j])
    
    if delta_hi - delta_lo < 1e-12:
        return 0.0, float(np.max(corr))
    
    # Minimize max_k (corr_k + d * delta_corr_k) over d in [delta_lo, delta_hi]
    # This is a convex PL function. Use minimize_scalar with bounded method.
    
    def obj(d):
        return float(np.max(corr + d * delta_corr))
    
    res = minimize_scalar(obj, bounds=(delta_lo, delta_hi), method='bounded',
                          options={'xatol': 1e-10})
    return float(res.x), float(res.fun)
```

```python
import numpy as np
import time
from scipy.optimize import minimize, minimize_scalar

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_subgrad(h, n):
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full')
    max_val = np.max(corr)
    threshold = max_val - 1e-7
    active = np.where(corr >= threshold)[0]
    grad = np.zeros(n)
    for k_idx in active:
        lag = k_idx - (n - 1)
        if lag > 0:
            grad[:n - lag] += (1.0 - h[lag:n])
            grad[lag:] -= h[:n - lag]
        elif lag < 0:
            L = -lag
            grad[L:] += (1.0 - h[:n - L])
            grad[:n - L] -= h[L:]
        else:
            grad += (1.0 - 2.0 * h)
    grad /= len(active)
    return grad, float(max_val)

def project_box_sum(x, target):
    n = len(x)
    lo = -float(np.max(x)) - 0.5
    hi = 1.0 - float(np.min(x)) + 0.5
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)

def find_opt_delta(corr, delta_corr, d_lo, d_hi):
    """Bisection to minimize max_k(corr_k + d*delta_corr_k) over d in [d_lo, d_hi]"""
    if d_hi - d_lo < 1e-14:
        return 0.0
    for _ in range(60):
        d_mid = (d_lo + d_hi) / 2.0
        vals = corr + d_mid * delta_corr
        k_max = int(np.argmax(vals))
        if delta_corr[k_max] > 1e-12:
            d_hi = d_mid
        elif delta_corr[k_max] < -1e-12:
            d_lo = d_mid
        else:
            break
    return (d_lo + d_hi) / 2.0

def compute_delta_corr(h, i, j, n):
    """Compute d_k = ∂corr_k/∂h[i] - ∂corr_k/∂h[j] for all k (length 2n-1)"""
    omh = 1.0 - h
    result = np.zeros(2 * n - 1)
    base = n - 1
    # For index i: (1-h[i+lag]) at k=base+lag where 0<=i+lag<n, and -h[i-lag] where 0<=i-lag<n
    lo1 = max(-i, -(n - 1))
    hi1 = min(n - 1 - i, n - 1)
    if lo1 <= hi1:
        idx = np.arange(lo1, hi1 + 1) + base
        result[idx] += omh[i + lo1: i + hi1 + 1]
    lo2 = max(i - n + 1, -(n - 1))
    hi2 = min(i, n - 1)
    if lo2 <= hi2:
        idx = np.arange(lo2, hi2 + 1) + base
        result[idx] -= h[i - hi2: i - lo2 + 1][::-1]
    # For index j (subtract)
    lo1 = max(-j, -(n - 1))
    hi1 = min(n - 1 - j, n - 1)
    if lo1 <= hi1:
        idx = np.arange(lo1, hi1 + 1) + base
        result[idx] -= omh[j + lo1: j + hi1 + 1]
    lo2 = max(j - n + 1, -(n - 1))
    hi2 = min(j, n - 1)
    if lo2 <= hi2:
        idx = np.arange(lo2, hi2 + 1) + base
        result[idx] += h[j - hi2: j - lo2 + 1][::-1]
    return result

def pair_swap_step(h, i, j, n, dx):
    """Optimal pair swap: returns (new_h, new_c5, improved)"""
    hi, hj = h[i], h[j]
    d_lo = max(-hi, hj - 1.0)
    d_hi = min(1.0 - hi, hj)
    if d_hi - d_lo < 1e-12:
        return h, compute_c5(h, dx), False
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full')
    dc = compute_delta_corr(h, i, j, n)
    old_max = float(np.max(corr))
    opt_d = find_opt_delta(corr, dc, d_lo, d_hi)
    new_max = float(np.max(corr + opt_d * dc))
    if new_max < old_max - 1e-13:
        h_new = h.copy()
        h_new[i] = np.clip(hi + opt_d, 0.0, 1.0)
        h_new[j] = np.clip(hj - opt_d, 0.0, 1.0)
        return h_new, new_max * dx, True
    return h, old_max * dx, False

def pair_swap_phase(h, n, dx, target, time_limit, seed_val=42):
    rng = np.random.RandomState(seed_val)
    t_start = time.time()
    best_h = h.copy()

def sturmian_binary(nn, alpha):
    h = np.zeros(nn)
    for i in range(nn):
        if (i * alpha) % 1.0 < 0.5:
            h[i] = 1.0
    return h

def smooth_max_obj(x, t_val, dx):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = t_val * (corr - m)
    shifted[shifted > 50] = 50
    shifted[shifted < -50] = -50
    return (m + (1.0 / t_val) * np.log(np.sum(np.exp(shifted)))) * dx

def sum_eq_fun(x, target_val):
    return float(np.sum(x) - target_val)

def slsqp_polish(h, n, dx, target, t_val, maxiter):
    try:
        res = minimize(
            lambda x: smooth_max_obj(x, t_val, dx),
            h.copy(), method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': lambda x: sum_eq_fun(x, target)}],
            options={'maxiter': maxiter, 'ftol': 1e-12}
        )
        return project_box_sum(res.x, target)
    except Exception:
        return h.copy()

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)

    h_init = np.array(initial_h_values, dtype=float).copy()
    n = len(h_init)
    dx = 2.0 / n
    target = n / 2.0
    h_init = project_box_sum(h_init, target)

    best_h = h_init.copy()
    best_c5 = compute_c5(h_init, dx)
    best_n = n

    # Phase 1: Subgradient on initial (80s)
    h1, c5_1 = subgradient_phase(best_h, n, dx, target, 80, seed)
    if c5_1 < best_c5:
        best_h, best_c5, best_n = h1, c5_1, n

    # Phase 2: Pair swap local search (500s)
    time_left = budget_s - (time.time() - t0) - 250
    h2, c5_2 = pair_swap_phase(best_h, n, dx, target, time_left, seed + 1)
    if c5_2 < best_c5:
        best_h, best_c5, best_n = h2, c5_2, n

    # Phase 3: Try alternative initializations (150s)
    for trial in range(6):
        if time.time() - t0 > budget_s - 120:
            break
        alpha = 0.5 + 0.15 * trial
        for nn in [n, n // 2]:
            if nn < 40 or time.time() - t0 > budget_s - 110:
                continue
            dd = 2.0 / nn
            tt = nn / 2.0
            h_s = sturmian_binary(nn, alpha)
            h_s = project_box_sum(h_s, tt)
            h_s, c5_s = subgradient_phase(h_s, nn, dd, tt, 15, seed + 50 + trial)
            if time.time() - t0 < budget_s - 100:
                h_s, c5_s = pair_swap_phase(h_s, nn, dd, tt, 25, seed + 100 + trial)
            c5_s = compute_c5(h_s, dd)
            if c5_s < best_c5:
                best_h, best_c5, best_n = h_s, c5_s, nn

    # Phase 4: SLSQP polish (60s)
    if time.time() - t0 < budget_s - 50:
        dd = 2.0 / best_n
        tt = best_n / 2.0
        for t_val in [80, 300]:
            if time.time() - t0 > budget_s - 30:
                break
            h_p = slsqp_polish(best_h, best_n, dd, tt, t_val, 500)
            c5_p = compute_c5(h_p, dd)
            if c5_p < best_c5:
                best_h, best_c5 = h_p, c5_p

    # Phase 5: Final pair swaps on best (remaining - 10s)
    remaining = budget_s - (time.time() - t0) - 10
    if remaining > 30:
        dd = 2.0 / best_n
        tt = best_n / 2.0
        h5, c5_5 = pair_swap_phase(best_h, best_n, dd, tt, remaining, seed + 999)
        if c5_5 < best_c5:
            best_h, best_c5 = h5, c5_5

    # Final: ensure validity
    best_h = np.clip(best_h, 0.0, 1.0)
    dd_final = 2.0 / best_n
    best_h = project_box_sum(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h, dd_final)

    return best_h, best_c5, best_n
```
