        ts = best_n / 2.0
        for t in [100, 300]:
            if elapsed() > budget_s - 40:
                break
            h_new = slsqp_refine(best_h, dd, ts, t=t, maxiter=200)
            c5_new = compute_c5(h_new, dd)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new
            # Follow up with CD
            h_new, c5_new = coord_descend(best_h, best_n, ts, n_sweeps=200)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new

    # Phase 5: More perturbations at final best_n
    if elapsed() < budget_s - 30:
        rng2 = np.random.RandomState(seed + 7777)
        for trial in range(30):
            if elapsed() > budget_s - 20:
                break
            h_trial = best_h.copy()
            n_pert = rng2.randint(2, 8)
            idx = rng2.choice(best_n, n_pert, replace=False)
            h_trial[idx] += rng2.uniform(-0.15, 0.15, n_pert)
            h_trial = fix_sum(h_trial, best_n / 2.0)
            nsw = max(50, int((budget_s - 25 - elapsed()) / 2))
            h_trial, c5_trial = coord_descend(h_trial, best_n, best_n / 2.0, n_sweeps=nsw)
            if c5_trial < best_c5:
                best_h, best_c5 = h_trial, c5_trial

    # Phase 6: Final heavy CD
    if elapsed() < budget_s - 10:
        dd = 2.0 / best_n
        h_new, c5_new = coord_descend(best_h, best_n, best_n / 2.0, n_sweeps=500)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    best_h = np.clip(best_h, 0.0, 1.0)
    dd = 2.0 / best_n
    best_c5 = compute_c5(best_h, dd)
    return best_h, best_c5, best_n
