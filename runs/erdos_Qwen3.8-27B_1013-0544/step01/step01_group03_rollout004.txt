So minimizing max_k corr[k] is the same as maximizing min_k ∑_i h[i]*h[i+k].

For the binary case, ∑_i h[i]*h[i+k] is just the number of positions where both h[i]=1 and h[i+k]=1.

This confirms that we want a binary sequence where the autocorrelation is as uniform and high as possible.

import numpy as np
from scipy.optimize import minimize, Bounds
import time

_op = {'tau': 1.0, 'target': 0.0, 'pen': 0.0}

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(obj, tau):
    x = obj / tau
    x_max = np.max(x)
    return float(tau * (x_max + np.log(np.sum(np.exp(x - x_max)))))

def sobj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    return smooth_max(corr, _op['tau']) + _op['pen'] * (np.sum(x) - _op['target']) ** 2

def golden_binary(n, target):
    h = np.zeros(n)
    phi = (1 + np.sqrt(5)) / 2
    used = set()
    for i in range(target):
        pos = int(i * n * phi) % n
        while pos in used:
            pos = (pos + 1) % n
        h[pos] = 1.0
        used.add(pos)
    return h

def binary_sa(h0, n_iter, T_start, T_end, seed):
    rng = np.random.RandomState(seed)
    h = h0.copy()
    curr = evaluate_c5(h)
    best = h.copy()
    best_c = curr
    alpha = (T_end / T_start) ** (1.0 / max(n_iter, 1))
    T = T_start
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for it in range(n_iter):
        i_out = ones[rng.randint(len(ones))]
        i_in = zeros[rng.randint(len(zeros))]
        h[i_out] = 0
        h[i_in] = 1
        new_c = evaluate_c5(h)
        if new_c <= curr or rng.random() < np.exp(-(new_c - curr) / T):
            curr = new_c
            ones[ones == i_out] = i_in
            zeros[zeros == i_in] = i_out
            if new_c < best_c:
                best_c = new_c
                best = h.copy()
        else:
            h[i_out] = 1
            h[i_in] = 0
        T *= alpha
    return best, best_c

def double_swap_sa(h0, n_iter, T_start, T_end, seed):

            h[i1] = 1
            h[i2] = 0

    return best, best_c

            best_c5 = c_f
            best_n = n_f

    # Ensure sum constraint

    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = evaluate_c5(best_h)

    return best_h, best_c5, best_n
