Let me try a different approach: **Use Nesterov's accelerated gradient** combined with a **more careful active set selection**.

The key idea of Nesterov's method:
- Maintain a "lookahead" point y_k = x_k + (t_k-1)/t_{k+1} * (x_k - x_{k-1})
- Compute gradient at y_k
- Update: x_{k+1} = y_k - (1/L) * grad(y_k)
- t_{k+1} = (1 + sqrt(1 + 4t_k²))/2

This gives O(1/k²) convergence rate vs O(1/k) for standard gradient.

But the non-smoothness of the max complicates things. Let me use a smoothed approximation instead.

**Smoothed max approach**: Instead of max(c_k), use the log-sum-exp:
f(x) = (1/β) * log(sum(exp(β*c_k)))

As β → ∞, this approaches max. For moderate β (say 10-50), it's a smooth approximation that we can optimize with standard methods.

The gradient of f is:
∇f = (1/sum) * sum(c_k' * exp(β*c_k))

where c_k' is the gradient of the k-th correlation.

This is weighted average of gradients, with weights proportional to exp(β*c_k). The lags with higher correlation get more weight.

This is actually what the active-set method approximates! But the log-sum-exp gives a smoother, more principled version.

def smoothed_max_c5(h, dx, beta=20.0):
    corr = np.correlate(h, 1.0 - h, mode='full')
    corr_scaled = beta * corr  # scale for numerical stability
    max_corr = float(np.max(corr_scaled))
    log_sum = max_corr + np.log(np.sum(np.exp(corr_scaled - max_corr)))
    return (log_sum / beta) * dx

def smoothed_max_grad(h, dx, beta=20.0):

import time

def project_feasible(x, target_sum):
    lo = float(x.min()) - 1.0
    hi = float(x.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = float(np.clip(x - mid, 0.0, 1.0).sum())
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_grad_lag(h, L):
    n = len(h)
    omh = 1.0 - h
    grad = np.zeros(n)
    if L == 0:
        grad[:] = omh[:] - h[:]
    elif L > 0:
        grad[L:] += omh[:n - L]
        grad[:n - L] -= h[L:]
    else:
        L2 = -L
        grad[:n - L2] += omh[L2:]
        grad[L2:] -= h[:n - L2]
    return grad

def active_gradient(h, top_k=8, rel_tol=0.002):
    n = len(h)
    n_half = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    thresh = c_max * (1.0 - rel_tol)
    active = np.where(corr >= thresh)[0]
    if len(active) < 1:
        active = np.array([int(np.argmax(corr))])
    elif len(active) > top_k:
        active = np.argsort(corr)[-top_k:]
    grad = np.zeros(n)
    for idx in active:
        L = int(idx) - n_half
        grad += compute_grad_lag(h, L)
    grad /= float(len(active))
    return grad

def gradient_descend(h, dx, target_sum, max_time, start_time):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(h, dx)
    ss = 0.003
    mom = np.zeros(n)
    mu = 0.65
    stag = 0
    while (time.time() - start_time) < max_time:
        grad = active_gradient(h, top_k=10, rel_tol=0.003)
        mom = mu * mom - ss * grad
        h_new = project_feasible(h + mom, target_sum)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best = h_new.copy()
            stag = 0
        else:
            stag += 1
            if stag % 30 == 0:
                ss *= 0.82
            if ss < 1e-10:
                break
        h = h_new
    return best, best_c5

def simulated_annealing(h, dx, target_sum, max_time, start_time, rng):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(h, dx)
    T = 0.005
    T_min = 5e-7
    cooling = 0.9998
    h_cur = h.copy()
    c5_cur = best_c5
    no_improve = 0
    total_iters = 0
    
    while (time.time() - start_time) < max_time:
        total_iters += 1
        # Adaptive perturbation
        base_sigma = 0.015
        sigma = base_sigma * (0.5 + T / 0.005 + 0.5 * rng.random())
        sigma = min(sigma, 0.06)
        
        # Occasionally large jump

            sigma = min(0.12, sigma * 4.0)
        
        h_prop = h_cur + rng.normal(0, sigma, n)
        h_prop = project_feasible(h_prop, target_sum)
        c5_prop = compute_c5(h_prop, dx)

        if delta < 0 or (T > T_min and rng.random() < np.exp(-delta / T)):
            h_cur = h_prop
            c5_cur = c5_prop
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best = h_cur.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

                    best = h_cur.copy()
                    no_improve = 0
            T = max(T_min, T * 0.8)
    
    return h_cur, c5_cur, best, best_c5

def slsqp_optimize(h_init, dx, target_sum, maxiter=5000):
    n = len(h_init)
    def obj(x):
        return compute_c5(x, dx)
    def eq(x):
        return float(np.sum(x) - target_sum)
    try:
        res = minimize(obj, h_init, method='SLSQP',
                      bounds=[(0.0, 1.0)] * n,
                      constraints=[{'type': 'eq', 'fun': eq}],
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None:
            h_ref = project_feasible(np.clip(res.x, 0.0, 1.0), target_sum)
            return h_ref, compute_c5(h_ref, dx)
    except Exception:
        pass
    return h_init, compute_c5(h_init, dx)

                grad = active_gradient(h_cur, top_k=6, rel_tol=0.002)
                h_cur = project_feasible(h_cur - 0.0008 * grad, target_sum)
                c5_cur = compute_c5(h_cur, dx)
                if c5_cur < best_c5:
                    best_c5 = c5_cur
                    best = h_cur.copy()
                    no_improve = 0
            T = max(T_min, T * 0.7)
    return h_cur, c5_cur, best, best_c5

def slsqp_optimize(h_init, dx, target_sum, maxiter=5000, max_time=60.0):
    n = len(h_init)
    t0 = time.time()
    def obj(x):
        if (time.time() - t0) > max_time:
            return compute_c5(x, dx) + 1.0
        return compute_c5(x, dx)
    def eq(x):
        return float(np.sum(x) - target_sum)
    try:
        res = minimize(obj, h_init, method='SLSQP',
                      bounds=[(0.0, 1.0)] * n,
                      constraints=[{'type': 'eq', 'fun': eq}],
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None:
            h_ref = project_feasible(np.clip(res.x, 0.0, 1.0), target_sum)
            return h_ref, compute_c5(h_ref, dx)
    except Exception:
        pass
    return h_init, compute_c5(h_init, dx)

def fine_polish(h, dx, target_sum, duration):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(h, dx)
    ss = 0.0001
    mom = np.zeros(n)
    t0 = time.time()
    while (time.time() - t0) < duration:
        grad = active_gradient(h, top_k=5, rel_tol=0.001)
        mom = 0.3 * mom - ss * grad
        h_new = project_feasible(h + mom, target_sum)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best = h_new.copy()
        else:
            ss *= 0.88
            if ss < 1e-13:
                break
        h = h_new
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except NameError:
        h0 = np.full(n_points, 0.5)
    h0 = project_feasible(h0, target_sum)
    best_h = h0.copy()
    best_c5 = compute_c5(h0, dx)
    def tleft():
        return max(0.0, (start_time + budget_s - 3.0) - time.time())
    # Phase 1: Gradient from initial
    if tleft() > 550.0:
        dur = min(180.0, tleft() - 500.0)
        h1, c5_1 = gradient_descend(h0, dx, target_sum, dur)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1
    # Phase 2: SLSQP
    if tleft() > 350.0:
        h_s, c5_s = slsqp_optimize(best_h, dx, target_sum, maxiter=5000, max_time=50.0)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s
    # Phase 3: SA
    if tleft() > 200.0:
        dur = min(200.0, tleft() - 190.0)
        h_sa, c5_sa, h_sa_best, c5_sa_best = simulated_annealing(best_h, dx, target_sum, dur, rng)
        if c5_sa_best < best_c5:
            best_c5 = c5_sa_best
            best_h = h_sa_best
        sa_state = h_sa.copy()
    else:
        sa_state = best_h.copy()
    # Phase 4: SLSQP from SA best
    if tleft() > 80.0:
        h_s2, c5_s2 = slsqp_optimize(best_h, dx, target_sum, maxiter=5000, max_time=45.0)
        if c5_s2 < best_c5:
            best_c5 = c5_s2
            best_h = h_s2
    # Phase 5: SLSQP from SA current if different
    if tleft() > 60.0 and not np.allclose(sa_state, best_h, atol=1e-5):
        h_s3, c5_s3 = slsqp_optimize(sa_state, dx, target_sum, maxiter=3000, max_time=40.0)
        if c5_s3 < best_c5:
            best_c5 = c5_s3
            best_h = h_s3
    # Phase 6: Multi-restart
    if tleft() > 100.0:
        for trial in range(3):
            if tleft() < 50.0:
                break
            h_pert = best_h + rng.normal(0, 0.04, n_points)
            h_pert = project_feasible(h_pert, target_sum)
            dur_sa = min(30.0, (tleft() - 30.0) / 2.0)
            if dur_sa > 5.0:
                _, _, h_sb, c5_sb = simulated_annealing(h_pert, dx, target_sum, dur_sa, rng)
                if c5_sb < best_c5:
                    best_c5 = c5_sb
                    best_h = h_sb
            if tleft() > 25.0:
                h_sp, c5_sp = slsqp_optimize(best_h, dx, target_sum, maxiter=2000, max_time=20.0)
                if c5_sp < best_c5:
                    best_c5 = c5_sp
                    best_h = h_sp
    # Phase 7: Fine polish
    if tleft() > 20.0:
        dur_p = max(3.0, tleft() - 10.0)
        h_f, c5_f = fine_polish(best_h, dx, target_sum, dur_p)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f
    # Final SLSQP
    if tleft() > 20.0:
        h_sf, c5_sf = slsqp_optimize(best_h, dx, target_sum, maxiter=5000, max_time=15.0)
        if c5_sf < best_c5:
            best_c5 = c5_sf
            best_h = h_sf
    return (best_h, best_c5, n_points)
