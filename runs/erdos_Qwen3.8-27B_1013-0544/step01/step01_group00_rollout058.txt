import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_h(h, target):
    h = h.copy()
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return np.clip(h, 0.0, 1.0)
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = mask.sum()
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def interp_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 1, n_old, endpoint=False)
    x_new = np.linspace(0, 1, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_h(h_new, n_new * 0.5)

def smooth_obj_and_grad(h, temp):
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full') * dx
    m = np.max(corr)
    shifted = (corr - m) / temp
    shifted -= np.max(shifted)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    obj = m + temp * (np.max(shifted) + np.log(sum_exp))
    
    # Weights (softmax)
    w = exp_s / sum_exp  # length 2n-1
    
    # Gradient
    grad = np.zeros(n)
    for m_idx in range(n):
        # First term: sum_{i=0}^{n-1} w[n-1+m_idx-i] * (1-h[i])
        idx1 = np.arange(n-1+m_idx, m_idx-1, -1)
        first = np.dot(w[idx1], one_minus_h)
        # Second term: sum_{i=0}^{n-1} w[n-1+i-m_idx] * h[i]  
        idx2 = np.arange(n-1-m_idx, 2*n-1-m_idx)
        second = np.dot(w[idx2], h)
        grad[m_idx] = first - second
    grad *= dx
    return obj, grad

def lbgfs_opt(h0, temp, penalty=100.0, max_iter=300):
    n = len(h0)
    target = n * 0.5
    
    def obj_grad(x):
        o, g = smooth_obj_and_grad(x, temp)
        c = x.sum() - target
        o += penalty * c * c
        g = g + 2.0 * penalty * c * np.ones(n)
        return o, g
    
    res = minimize(obj_grad, h0, method='L-BFGS-B', jac=True,
                   bounds=[(0.0, 1.0)]*n,
                   options={'maxiter': max_iter, 'gtol': 1e-7, 'ftol': 1e-10})
    h_new = np.clip(res.x, 0.0, 1.0)
    h_new = project_h(h_new, target)
    return h_new, compute_c5(h_new)

def t_objective(v):
    return v[-1]

def t_ineq(v):
    n = len(v) - 1
    h = v[:n]
    t = v[-1]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return t - corr

def t_eq_fun(v):
    n = len(v) - 1
    return np.sum(v[:n]) - n * 0.5

def t_eq_jac(v):
    n = len(v) - 1
    j = np.zeros(len(v))
    j[:n] = 1.0
    return j

def optimize_t_var(h0, max_iter=400):
    n = len(h0)
    c5_cur = compute_c5(h0)
    v0 = np.append(h0.copy(), c5_cur + 0.02)
    bounds = [(0.0, 1.0)]*n + [(0.0, 1.5)]
    cons = [{'type': 'ineq', 'fun': t_ineq},
            {'type': 'eq', 'fun': t_eq_fun, 'jac': t_eq_jac}]
    res = minimize(t_objective, v0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-10})
    h_new = np.clip(res.x[:n], 0.0, 1.0)
    h_new = project_h(h_new, n * 0.5)
    return h_new, compute_c5(h_new)

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))


def project_h(h, target):
    h = h.copy()
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return np.clip(h, 0.0, 1.0)
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = mask.sum()
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)


def interp_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 1, n_old, endpoint=False)
    x_new = np.linspace(0, 1, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_h(h_new, n_new * 0.5)


def smooth_obj_and_grad(h, temp):
    n = len(h)
    dx = 2.0 / n
    omm = 1.0 - h
    corr = np.correlate(h, omm, mode='full') * dx
    m = np.max(corr)
    shifted = (corr - m) / temp
    shifted -= np.max(shifted)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    obj = m + temp * (np.max(shifted) + np.log(sum_exp))
    w = exp_s / sum_exp
    grad = np.zeros(n)
    for mi in range(n):
        idx1 = np.arange(n - 1 + mi, mi - 1, -1)
        idx2 = np.arange(n - 1 - mi, 2 * n - 1 - mi)
        grad[mi] = np.dot(w[idx1], omm) - np.dot(w[idx2], h)
    grad *= dx
    return obj, grad


def lbgfs_opt(h0, temp, penalty=200.0, max_iter=300):
    n = len(h0)
    target = n * 0.5

    def obj_grad(x):
        o, g = smooth_obj_and_grad(x, temp)
        c = x.sum() - target
        o += penalty * c * c
        g = g + 2.0 * penalty * c
        return o, g

    res = minimize(obj_grad, h0, method='L-BFGS-B', jac=True,
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': max_iter, 'gtol': 1e-8, 'ftol': 1e-12})
    h_new = np.clip(res.x, 0.0, 1.0)
    h_new = project_h(h_new, target)
    return h_new, compute_c5(h_new)


def t_obj(v):
    return v[-1]


def t_ineq(v):
    n = len(v) - 1
    h = v[:n]
    t = v[-1]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return t - corr


def t_eq(v):
    n = len(v) - 1
    return np.sum(v[:n]) - n * 0.5


def t_eq_j(v):
    n = len(v) - 1
    j = np.zeros(len(v))
    j[:n] = 1.0
    return j


def optimize_t_var(h0, max_iter=400, ftol=1e-11):
    n = len(h0)
    c5_cur = compute_c5(h0)
    v0 = np.append(h0.copy(), c5_cur + 0.01)
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.2)]
    cons = [{'type': 'ineq', 'fun': t_ineq},
            {'type': 'eq', 'fun': t_eq, 'jac': t_eq_j}]
    res = minimize(t_obj, v0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': ftol})
    h_new = np.clip(res.x[:n], 0.0, 1.0)
    h_new = project_h(h_new, n * 0.5)
    return h_new, compute_c5(h_new)
