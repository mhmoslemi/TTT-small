import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(3000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-14:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def binary_max_corr(h, n):
    c = np.correlate(h, 1 - h, mode='full')
    return float(c.max())

def sum_con(h, n):
    return np.sum(h) - n / 2.0

def obj_smooth(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 700:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

        idx = zeros[rng.randint(len(zeros))]
