import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-15:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 700:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def sum_con(h, n):
    return np.sum(h) - n / 2.0

def sum_con_jac(h, n):
    return np.ones(n)

def coord_sweep(h, n, n_passes=30, skip_cycle=True):
    h = h.copy()
    for p in range(n_passes):
        skip = (p % n) if skip_cycle else 0
        for j in range(n):
            if j == skip:
                continue
            S = n / 2.0 - (h.sum() - h[j] - h[skip])
            lo = max(0.0, S - 1.0)
            hi = min(1.0, S)
            if lo >= hi - 1e-13:
                continue
            gr = (np.sqrt(5) - 1) / 2
            a, b = lo, hi
            c1 = b - gr * (b - a)
            d1 = a + gr * (b - a)
            h[j] = c1; h[skip] = S - c1
            fc = compute_c5(h, n)
            h[j] = d1; h[skip] = S - d1
            fd = compute_c5(h, n)
            for _ in range(45):
                if b - a < 1e-11:
                    break
                if fc < fd:
                    b = d1; d1 = c1; fd = fc
                    c1 = b - gr * (b - a)
                    h[j] = c1; h[skip] = S - c1
                    fc = compute_c5(h, n)
                else:
                    a = c1; c1 = d1; fc = fd
                    d1 = a + gr * (b - a)
                    h[j] = d1; h[skip] = S - d1
                    fd = compute_c5(h, n)
            h[j] = (a + b) / 2.0
            h[skip] = np.clip(S - h[j], 0.0, 1.0)
    return h

def slsqp_polish(h, n, alphas=None):
    if alphas is None:
        alphas = [50, 200, 1000, 5000]
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    best = h.copy()
    best_c = compute_c5(best, n)
    for alpha in alphas:
        try:
            res = minimize(obj_smooth, best, args=(n, alpha), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 2000, 'ftol': 1e-17})
            h_try = project_feasible(res.x, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c:
                best, best_c = h_try, c_try
        except Exception:
            pass
    return best, best_c

def block_init(n, nblocks, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    remaining = n // 2
    boundaries = sorted(rng.choice(range(1, n), size=nblocks - 1, replace=False))
    boundaries = [0] + list(boundaries) + [n]
    for i in range(nblocks):
        seg = boundaries[i + 1] - boundaries[i]
        fill = remaining // (nblocks - i) if i < nblocks - 1 else remaining
        fill = min(fill, seg)
        h[boundaries[i]:boundaries[i] + fill] = 1.0
        remaining -= fill
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):

        h = project_feasible(h, n)

    # Random restarts at best_n

        if budget_s - (time.time() - t0) < 30:
            break

        if c_pert < best_c5:
            best_h, best_c5 = h_pert.copy(), c_pert

    # Block structure restarts
    for nb in [2, 3, 4, 5]:

        for attempt in range(3):

            c_blk = compute_c5(h_blk, n)
            if c_blk < best_c5:

        best_h = coord_sweep(best_h, best_n, n_passes=30)
        best_h, best_c5 = slsqp_polish(best_h, best_n, [500, 2000, 8000])

    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return best_h, best_c5, best_n
