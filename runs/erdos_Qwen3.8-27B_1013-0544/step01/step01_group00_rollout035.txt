The projection onto {0<=h<=1, sum(h)=target} is:

- This is a 1D bisection problem

Actually, the projection is:
P(h) = clip(s, 0, 1) where s = h - lambda * 1, and lambda is chosen so that sum(P(h)) = target.

This is a standard water-filling problem. I can solve it with bisection on lambda.

Let me also think about the step size. Barzilai-Borwein (BB) step size:
- alpha = ||s_k||^2 / <s_k, y_k> (BB1) or ||s_k||^2 / ||y_k||^2 * <s_k, y_k> / ||y_k||^2 (BB2)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def compute_corr(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_box_sum(h, target):
    """Project h onto {0<=h<=1, sum(h)=target}."""
    # Water-filling: find lambda s.t. sum(clip(h-lambda, 0, 1)) = target
    s = np.sort(h)[::-1]
    cumsum = np.cumsum(s)
    n = len(h)
    # Find k such that (cumsum[k] - k*lambda) / (k+1) ... 
    # Actually, the standard approach:
    # u = sorted h in decreasing order
    # For each k (1-indexed), theta_k = (sum_{i=1}^k u_i - target) / k
    # The solution is: h_proj[i] = max(0, min(1, h[i] - theta))
    # where theta = theta_k for the correct k
    
    u = np.sort(h)[::-1]
    rho = np.arange(1, n+1)
    theta_candidates = (np.cumsum(u) - target) / rho
    # Find the largest k such that theta_k < u_k (0-indexed: theta_candidates[k-1] < u[k-1])
    # And theta_k >= u_{k+1} (theta_candidates[k-1] >= u[k])
    # Simpler: find k such that u[k-1] > theta_candidates[k-1] >= u[k] (with u[n]=0 convention)
    
    # Use the standard algorithm:
    # Start with all active, remove those that violate
    # Or just bisection on lambda:
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        val = np.sum(np.clip(h - mid, 0, 1))
        if val > target:
            lo = mid
        else:
            hi = mid
    lambda_star = (lo + hi) / 2
    return np.clip(h - lambda_star, 0, 1)

def active_lags_gradient(h, active_indices):
    """Compute gradient of sum_{k in active} corr[k] w.r.t. h."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in active_indices:
        m = k - n + 1  # offset
        # grad[i] += dx * [(1-h[i-m]) * I(0<=i-m<n) - h[i+m] * I(0<=i+m<n)]
        for i in range(n):
            im = i - m
            if 0 <= im < n:
                grad[i] += (1.0 - h[im])
            ip = i + m
            if 0 <= ip < n:
                grad[i] -= h[ip]
    grad *= dx
    return grad

def active_lags_gradient(h, active_indices):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in active_indices:
        m = k - n + 1
        # First term: (1-h[i-m]) for valid i-m
        # This is (1-h) shifted by m positions
        if m >= 0:
            # i-m >= 0 means i >= m; i-m < n means i < n+m (always true since i < n)
            # So for i in [m, n): grad[i] += (1-h[i-m])
            grad[m:] += (1.0 - h[:n-m])
        else:
            # i-m >= 0 means i >= m (always true since m<0, i>=0); i-m < n means i < n+m
            # So for i in [0, n+m): grad[i] += (1-h[i-m]) = (1-h[i+|m|])
            abs_m = -m
            grad[:n-abs_m] += (1.0 - h[abs_m:])
        
        # Second term: -h[i+m] for valid i+m
        if m >= 0:
            # i+m < n means i < n-m
            grad[:n-m] -= h[m:]
        else:
            # i+m >= 0 means i >= -m = |m|; i+m < n means i < n (always true)
            abs_m = -m
            grad[abs_m:] -= h[:n-abs_m]
    grad *= dx
    return grad

def lse_penalty_obj(v, temp, mu):
    n = len(v)
    h = v
    corr = compute_corr(h)
    m = np.max(corr)
    lse = m + temp * np.log(np.sum(np.exp((corr - m) / temp)))
    penalty = mu * (np.sum(h) - n * 0.5) ** 2
    return lse + penalty

def lse_penalty_grad(v, temp, mu):
    n = len(v)
    h = v
    corr = compute_corr(h)
    m = np.max(corr)
    w = np.exp((corr - m) / temp)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_box_sum(h, target):
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def gradient_corr_sum(h, active_indices):
    """Gradient of sum_{k in active} corr[k] w.r.t. h."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in active_indices:
        m = k - n + 1
        if m >= 0:
            grad[m:] += (1.0 - h[:n - m])
            grad[:n - m] -= h[m:]
        else:
            p = -m
            grad[:n - p] += (1.0 - h[p:])
            grad[p:] -= h[:n - p]
    return grad * dx

def lse_grad(h, temp, active_indices=None):
    """Gradient of log-sum-exp of corr over active_indices (or all lags)."""
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    if active_indices is not None:
        c = corr[active_indices]
        idx = active_indices
    else:
        c = corr
        idx = np.arange(2 * n - 1)
    m = np.max(c)
    shifted = (c - m) / temp
    w = np.exp(np.clip(shifted, -50, 50))
    w_sum = np.sum(w)
    if w_sum < 1e-30:
        return np.zeros(n)
    w /= w_sum
    
    grad = np.zeros(n)
    for i, k in enumerate(idx):
        wi = w[i]
        if wi < 1e-8:
            continue
        m_off = k - n + 1
        if m_off >= 0:
            grad[m_off:] += wi * (1.0 - h[:n - m_off])
            grad[:n - m_off] -= wi * h[m_off:]
        else:
            p = -m_off
            grad[:n - p] += wi * (1.0 - h[p:])
            grad[p:] -= wi * h[:n - p]
    return grad * dx

def nesterov_pg(h0, n_iters=10000, top_m=15, temp=0.01, momentum=0.9):
    """Nesterov accelerated projected gradient descent."""
    n = len(h0)
    target = n * 0.5
    h = h0.copy()
    h_prev = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    alpha = 0.1
    active = np.argsort(compute_corr(h))[::-1][:top_m]
    
    for it in range(n_iters):
        if it % 500 == 0:
            corr = compute_corr(h)
            c5 = float(np.max(corr))
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            active = np.argsort(corr)[::-1][:top_m]
        
        # Nesterov momentum step
        y = h + momentum * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)  # keep in box for gradient evaluation
        
        # Gradient at y
        g = lse_grad(y, temp, active)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-12:
            break
        
        # Gradient step + projection
        h_new = project_box_sum(h - alpha * g, target)
        
        # Backtracking if needed
        c_new = compute_corr(h_new)[active]
        c_old = compute_corr(h)[active]
        m_new = np.max(c_new) + temp * np.log(np.sum(np.exp((c_new - np.max(c_new)) / temp)))
        m_old = np.max(c_old) + temp * np.log(np.sum(np.exp((c_old - np.max(c_old)) / temp)))
        
        if m_new > m_old:
            alpha *= 0.5
            h_new = project_box_sum(h - alpha * g, target)
        
        h_prev = h
        h = h_new
        
        c5 = float(np.max(compute_corr(h)))
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

def t_objective(v):

def t_ineq(v):
    n = len(v) - 1
    h = v[:n]
    t = v[-1]
    dx = 2.0 / n

    return t - corr

    return np.sum(v[:n]) - n * 0.5

def t_eq_jac(v):
    n = len(v) - 1
    j = np.zeros(len(v))

def smooth_obj(v, temp):
    n = len(v)
    h = v
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    shifted = (corr - m) / temp
