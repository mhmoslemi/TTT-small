        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-14:

            h += (diff / total) * h

    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = c - m
    mx = shifted.max()
    if mx > 700:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def subgradient_step(h, n, step, top_k=5, sharpness=10):
    """Take one projected subgradient step."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = min(top_k, 2 * n - 1)
    idx = np.argsort(c)[::-1][:m]
    w = c[idx] - c[idx[-1]] + 0.01
    w = np.exp(w * sharpness)
    w /= w.sum()
    
    s = np.zeros(n)
    for wi, mi in zip(w, idx):
        lag = int(mi) - (n - 1)
        if lag == 0:
            contrib = 1.0 - 2.0 * h
        elif lag > 0:
            contrib = np.zeros(n)
            contrib[lag:] += (1.0 - h[:-lag])
            contrib[:n-lag] -= h[lag:]
        else:
            la = -lag
            contrib = np.zeros(n)
            contrib[la:] += (1.0 - h[:-la])
            contrib[:n-la] -= h[la:]
        s += wi * contrib
    
    h_new = h - step * s * dx
    return project_feasible(h_new, n)

def subgradient_optimize(h, n, steps=3000, step0=0.01, top_k=5, sharpness=10):
    """Subgradient method with geometric decay."""
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    step = step0
    
    for it in range(steps):
        h = subgradient_step(h, n, step, top_k, sharpness)
        c5 = compute_c5(h, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        step = step0 * (0.999 ** it)
        if step < 1e-6:
            break
    
    return best_h, best_c5

def coord_descent_fast(h, n, sweeps=10, precision=25):
    h = h.copy()
    c = np.correlate(h, 1 - h, mode='full')  # shape (2n-1,)
    
    for s in range(sweeps):
        c = np.correlate(h, 1 - h, mode='full')
        for j in range(n):
            # Precompute sensitivity of each c[m] to h[j]
            # dc[m]/dh[j] = (1-h[j-lag_m])*[0<=j-lag_m<n] - h[j+lag_m]*[0<=j+lag_m<n]
            # where lag_m = m - (n-1)
            
            # For m = 0 to 2n-2, lag_m = m - n + 1
            # left_idx = j - lag_m = j - m + n - 1
            # right_idx = j + lag_m = j + m - n + 1
            
            m_vals = np.arange(2 * n - 1)
            lags = m_vals - (n - 1)
            left = j - lags  # = j - m + n - 1
            right = j + lags  # = j + m - n + 1
            
            sens = np.zeros(2 * n - 1)
            left_valid = (left >= 0) & (left < n)
            right_valid = (right >= 0) & (right < n)
            
            sens[left_valid] += (1.0 - h[left[left_valid]])
            sens[right_valid] -= h[right[right_valid]]
            
            # Now c[m] = c[m]_current + (new_hj - old_hj) * sens[m]
            old_hj = h[j]
            
            # Ternary search for optimal h[j]
            lo, hi = 0.0, 1.0
            for _ in range(precision):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                c1 = c + (m1 - old_hj) * sens
                c2 = c + (m2 - old_hj) * sens
                if c1.max() < c2.max():
                    hi = m2
                else:
                    lo = m1
            new_hj = (lo + hi) / 2.0
            new_hj = np.clip(new_hj, 0.0, 1.0)
            
            # Update c and h
            c += (new_hj - old_hj) * sens
            h[j] = new_hj
        
        h = project_feasible(h, n)
    
    return h

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-14:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = c - m
    mx = shifted.max()
    if mx > 700:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def sum_con(h, n):
    return np.sum(h) - n / 2.0

def sum_con_jac(h, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=3000, ftol=1e-18):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def coord_descent_fast(h, n, sweeps=20, precision=25):
    h = h.copy()
    c_arr = np.correlate(h, 1 - h, mode='full')
    m_vals = np.arange(2 * n - 1)
    lags = m_vals - (n - 1)
    for _ in range(sweeps):
        c_arr = np.correlate(h, 1 - h, mode='full')
        for j in range(n):
            left = j - lags
            right = j + lags
            sens = np.zeros(2 * n - 1)
            lv = (left >= 0) & (left < n)
            rv = (right >= 0) & (right < n)
            sens[lv] += (1.0 - h[left[lv]])
            sens[rv] -= h[right[rv]]
            old = h[j]
            lo, hi = 0.0, 1.0
            for _ in range(precision):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                if (c_arr + (m1 - old) * sens).max() < (c_arr + (m2 - old) * sens).max():
                    hi = m2
                else:
                    lo = m1
            new_val = np.clip((lo + hi) / 2.0, 0.0, 1.0)
            c_arr += (new_val - old) * sens
            h[j] = new_val
        h = project_feasible(h, n)
    return h

def subgradient_optimize(h, n, steps=3000, step0=0.01, top_k=5, sharpness=8.0):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    dx = 2.0 / n
    for it in range(steps):
        step = step0 * (0.9995 ** it)
        if step < 1e-7:
            break
        c = np.correlate(h, 1 - h, mode='full') * dx
        m = min(top_k, 2 * n - 1)
        idx = np.argsort(c)[::-1][:m]
        w = c[idx] - c[idx[-1]] + 0.01
        w = np.exp(w * sharpness)
        w /= w.sum()
        s = np.zeros(n)
        for wi, mi in zip(w, idx):
            lag = int(mi) - (n - 1)
            if lag == 0:
                contrib = 1.0 - 2.0 * h
            elif lag > 0:
                contrib = np.zeros(n)
                contrib[lag:] += (1.0 - h[:-lag])
                contrib[:n - lag] -= h[lag:]
            else:
                la = -lag
                contrib = np.zeros(n)
                contrib[la:] += (1.0 - h[:-la])
                contrib[:n - la] -= h[la:]
            s += wi * contrib
        h = project_feasible(h - step * s * dx, n)
        c5 = compute_c5(h, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def upsample(h, n_old, n_new):
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)
