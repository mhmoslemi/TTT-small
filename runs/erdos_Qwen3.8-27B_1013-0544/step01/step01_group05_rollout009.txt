import numpy as np
from scipy.optimize import minimize

def project_feasible(x, target_sum):
    lo = float(x.min()) - 1.0
    hi = float(x.max()) + 1.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = float(np.clip(x - mid, 0.0, 1.0).sum())
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def grad_gk(h, L):
    n = len(h)
    grad = np.zeros(n)
    if L == 0:
        grad[:] = 1.0 - 2.0 * h
    elif L > 0:
        grad[:n - L] += (1.0 - h[L:])
        grad[L:] -= h[:n - L]
    else:
        L2 = -L
        grad[L2:] += (1.0 - h[:n - L2])
        grad[:n - L2] -= h[L2:]
    return grad

def lse_gradient(h, dx, beta):
    n = len(h)
    n_half = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    exp_vals = np.exp(beta * (corr - m))
    weights = exp_vals / exp_vals.sum()
    grad = np.zeros(n)
    for k in range(len(corr)):
        if weights[k] > 1e-8:
            L = k - n_half
            grad += weights[k] * grad_gk(h, L)
    return grad

def active_set_gradient(h, dx, top_k=8):
    n = len(h)
    n_half = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(corr)
    tol = max(1e-4, c_max * 0.002)
    active = np.where(corr >= c_max - tol)[0]
    if len(active) < 2:
        active = np.argsort(corr)[-2:]
    elif len(active) > top_k:
        active = np.argsort(corr)[-top_k:]
    grad = np.zeros(n)
    for idx in active:
        L = int(idx) - n_half
        grad += grad_gk(h, L)
    grad /= float(len(active))
    return grad
