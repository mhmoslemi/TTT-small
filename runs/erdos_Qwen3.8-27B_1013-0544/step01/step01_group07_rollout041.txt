```python
def lse_obj_and_grad(h, t, dx, mu=0):
    n = len(h)
    c = np.correlate(h, 1 - h, mode='full')
    max_c = np.max(c)
    shifted = t * (c - max_c)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    p = exp_shifted / sum_exp
    LSE = (max_c + (1.0/t) * np.log(sum_exp)) * dx
    
    # Gradient
    S1 = np.convolve(1 - h, p, mode='full')[:n]
    S2 = np.correlate(h, p, mode='full')[:n]
    grad = dx * (S1 - S2)
    
    if mu > 0:
        sum_h = np.sum(h)
        target = n / 2.0
        LSE += mu * (sum_h - target) ** 2
        grad += 2.0 * mu * (sum_h - target)
    
    return LSE, grad
```

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target_sum):
    h = h.copy()
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-14:
        h += diff / len(h)
    return np.clip(h, 0.0, 1.0)

def lse_obj_and_grad(x, t, dx, mu):
    n = len(x)
    c = np.correlate(x, 1 - x, mode='full')
    max_c = np.max(c)
    shifted = t * (c - max_c)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    p = exp_shifted / sum_exp
    val = (max_c + (1.0/t) * np.log(sum_exp)) * dx
    
    S1 = np.convolve(1 - x, p, mode='full')[:n]
    S2 = np.correlate(x, p, mode='full')[:n]
    grad = dx * (S1 - S2)
    
    if mu > 0:
        s = np.sum(x)
        diff = s - n / 2.0
        val += mu * diff * diff

        res = minimize(f, h, jac=g, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10, 'factr': 1e7})
        return res.x
    except:
        return h.copy()

def binary_local_search(h, dx, n_iters=1000):
    n = len(h)
    # Threshold to binary
    h_b = (h > 0.5).astype(float)
    # Ensure exactly n/2 ones
    n_ones = int(np.sum(h_b))
    if n_ones > n // 2:
        # Remove excess ones (those closest to 0.5)
        ones = np.where(h_b == 1)[0]
        order = np.argsort(h[ones])  # lowest h values first
        for i in range(n_ones - n // 2):
            h_b[ones[order[i]]] = 0
    elif n_ones < n // 2:
        zeros = np.where(h_b == 0)[0]
        order = np.argsort(-h[zeros])  # highest h values first
        for i in range(n // 2 - n_ones):
            h_b[zeros[order[i]]] = 1
    
    best_c5 = compute_c5(h_b, dx)
    rng = np.random.RandomState(42)
    
    for _ in range(n_iters):
        # Pick a random 1 and a random 0 to swap
        ones = np.where(h_b == 1)[0]
        zeros = np.where(h_b == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h_b[i] = 0
        h_b[j] = 1
        c5_new = compute_c5(h_b, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
        else:
            h_b[i] = 1
            h_b[j] = 0
    
    return h_b, best_c5
