import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def make_binary(h, n):
    """Convert to binary with exactly n//2 ones."""
    target = n // 2
    # Get indices sorted by value descending
    idx = np.argsort(-h, kind='stable')
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def sa_binary(h, n, dx, rng, time_budget, T_init=0.05, T_min=0.005, cool=0.999995):
    """Simulated annealing on binary sequence."""
    t_start = time.time()
    h = h.copy()
    
    # Track ones and zeros
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    n_ones = len(ones_list)
    n_zeros = len(zeros_list)
    
    corr = np.correlate(h, 1 - h, mode="full")
    current_max = float(np.max(corr))
    best_c5 = current_max * dx
    best_h = h.copy()
    T = T_init
    
    iters_no_improve = 0
    reheat_threshold = 100000
    
    while time.time() - t_start < time_budget:
        ii = rng.randint(n_ones)
        jj = rng.randint(n_zeros)
        i = ones_list[ii]
        j = zeros_list[jj]
        
        h[i] = 0.0
        h[j] = 1.0
        ones_list[ii] = j
        zeros_list[jj] = i
        
        corr_new = np.correlate(h, 1 - h, mode="full")
        new_max = float(np.max(corr_new))
        
        delta = (new_max - current_max) * dx
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            current_max = new_max
            new_c5 = new_max * dx
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                iters_no_improve = 0
            else:
                iters_no_improve += 1
        else:
            h[i] = 1.0
            h[j] = 0.0
            ones_list[ii] = i
            zeros_list[jj] = j
            iters_no_improve += 1
        
        T *= cool
        if T < T_min or iters_no_improve > reheat_threshold:
            T = T_init * 0.5
            iters_no_improve = 0
    
    return best_h, best_c5

For each k in the correlation:
corr[k] = sum_m h[m] * (1 - h[m + k - (n-1)])

The change in corr[k] is:

- Due to h[j]: 0→1

    Old: 0*(1-h[j+k-(n-1)]) = 0
    New: 1*(1-h[j+k-(n-1)]) = 1-h[j+k-(n-1)]
    Change: +(1-h[j+k-(n-1)])
  - If j is the "second" index (m+k-(n-1)=j, i.e., m = j-k+(n-1)):
    Old: h[m]*(1-0) = h[m]
    New: h[m]*(1-1) = 0
    Change: -h[m]

    idx = np.argsort(-h, kind='stable')

    return h

def sa_binary(h, n, dx, seed, time_budget):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    h = h.copy()
    
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    n_ones = len(ones_list)
    n_zeros = len(zeros_list)
    
    current_max = float(np.max(np.correlate(h, 1 - h, mode="full")))
    best_c5 = current_max * dx
    best_h = h.copy()
    
    T = 0.08
    cool = 0.999997
    
    while time.time() - t_start < time_budget:
        ii = rng.randint(n_ones)
        jj = rng.randint(n_zeros)
        i = ones_list[ii]
        j = zeros_list[jj]
        
        h[i] = 0.0
        h[j] = 1.0
        ones_list[ii] = j
        zeros_list[jj] = i
        
        new_max = float(np.max(np.correlate(h, 1 - h, mode="full")))
        delta = (new_max - current_max) * dx
        
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            current_max = new_max
            if new_max * dx < best_c5:
                best_c5 = new_max * dx
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
            ones_list[ii] = i
            zeros_list[jj] = j
        
        T *= cool
        if T < 0.002:
            T = 0.04
    
    return best_h, best_c5

def refine_nonbinary(h, n, dx, seed, time_budget):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    while time.time() - t_start < time_budget:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        if h[i] <= 0.0 or h[j] >= 1.0:
            continue
        
        eps = min(rng.random() * 0.03 + 0.002, h[i], 1.0 - h[j])
        old_i, old_j = h[i], h[j]
        h[i] -= eps
        h[j] += eps
        
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
        else:
            h[i] = old_i
            h[j] = old_j
    
    return h, best_c5
