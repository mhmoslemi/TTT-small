import numpy as np
from scipy.optimize import minimize, Bounds, LinearConstraint
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) > 1e-12 and s > 0:
        h += (target_sum - s) / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_refine(h, target_sum, n, beta=200, maxiter=500):
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))

    def obj(x):
        corr = get_corr(x)
        return smooth_max(corr, beta) * dx

    def obj_grad(x):
        corr = get_corr(x)
        w = np.exp(beta * (corr - corr.max()))
        wsum = w.sum()
        if wsum < 1e-300:
            return np.zeros(n)
        w /= wsum
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-12:
                continue
            s = k - (n - 1)
            if s >= 0:
                grad[s:] += w[k] * (1.0 - x[:n - s])
                grad[:n - s] -= w[k] * x[s:]
            else:
                t = -s
                grad[:n - t] += w[k] * (1.0 - x[t:])
                grad[t:] -= w[k] * x[:n - t]
        return grad * dx

    con = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
            'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bounds,
                       constraints=con, options={'maxiter': maxiter, 'ftol': 1e-13})
        return project_h(res.x, target_sum)
    except Exception:
        return h

def subgrad_step(h, corr, n, top_k=12):
    sorted_idx = np.argsort(corr)[::-1]
    active = sorted_idx[:top_k]
    shifts = active - (n - 1)
    grad = np.zeros(n)
    for s in shifts:
        if s >= 0:
            grad[s:] += (1.0 - h[:n - s])
            grad[:n - s] -= h[s:]
        else:
            t = -s
            grad[:n - t] += (1.0 - h[t:])
            grad[t:] -= h[:n - t]
    grad /= top_k
    return grad

def subgrad_momentum(h, target_sum, n, max_iters=800, top_k=12, mu=0.82):
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    for it in range(max_iters):
        corr = get_corr(h)
        grad = subgrad_step(h, corr, n, top_k)
        alpha = 0.025 / (1.0 + it / 150.0)
        if no_improve > 80:
            alpha *= 3.0
        v = mu * v - alpha * grad
        h_new = project_h(h + v, target_sum)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
            h = h_new
            no_improve = 0
        else:
            h = h_new
            no_improve += 1
            v *= 0.4
        if no_improve > 400:
            break
    return best_h, best_c5

def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    def time_left():
        return budget_s - (time.time() - start_time)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)

    n0 = len(h_init)
    best_h = project_h(h_init, n0 / 2.0)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP with moderate beta
    if time_left() > 120:
        ts = n0 // 2
        h1 = slsqp_refine(best_h, ts, n0, beta=80, maxiter=600)
        c5 = compute_c5(h1)
        if c5 < best_c5:
            best_h, best_c5 = h1.copy(), c5

    # Phase 2: Subgradient momentum
    if time_left() > 80:
        ts = n0 // 2
        h2, c5_2 = subgrad_momentum(best_h, ts, n0, max_iters=600, top_k=12, mu=0.82)
        if c5_2 < best_c5:
            best_h, best_c5 = h2, c5_2

    # Phase 3: SLSQP high beta
    if time_left() > 60:
        ts = n0 // 2
        h3 = slsqp_refine(best_h, ts, n0, beta=400, maxiter=800)
        c5 = compute_c5(h3)
        if c5 < best_c5:
            best_h, best_c5 = h3.copy(), c5

    # Phase 4: Diversified restarts with SLSQP
    n = n0
    ts = n // 2
    for attempt in range(40):
        if time_left() < 25:
            break
        strat = attempt % 5
        h = best_h.copy()
        if strat == 0:
            scale = 0.06 * np.exp(-attempt / 25.0)
            h += np.random.randn(n) * scale
        elif strat == 1:
            corr = get_corr(h)
            top_idx = np.argsort(corr)[::-1][:8]
            for ki in top_idx:
                s = ki - (n - 1)
                for j in range(n):
                    i1 = j - s
                    i2 = j + s
                    d = 0.0
                    if 0 <= i1 < n:
                        d += (1.0 - h[i1])
                    if 0 <= i2 < n:
                        d -= h[i2]
                    h[j] -= 0.02 * d * np.random.rand()
        elif strat == 2:
            blen = np.random.randint(4, max(5, n // 5))
            b1 = np.random.randint(0, n - blen)
            b2 = np.random.randint(0, n - blen)
            h[b1:b1+blen], h[b2:b2+blen] = h[b2:b2+blen].copy(), h[b1:b1+blen].copy()
        elif strat == 3:
            half = n // 2
            h[:half] += np.random.randn(half) * 0.05
            h[half:] -= np.random.randn(n - half) * 0.05
        else:
            n_pert = max(3, n // 8)
            idx = np.random.choice(n, n_pert, replace=False)
            h[idx] += np.random.randn(n_pert) * 0.08
        h = project_h(h, ts)
        try:
            hr = slsqp_refine(h, ts, n, beta=300, maxiter=400)
            c5 = compute_c5(hr)
            if c5 < best_c5:
                best_h, best_c5 = hr.copy(), c5
        except Exception:
            pass

    # Phase 5: Try higher resolution
    for n_try in [80, 100, 120]:
        if time_left() < 80:
            break
        ts_try = n_try // 2
        h_high = interpolate_h(best_h, n_try)
        h_high = project_h(h_high, ts_try)
        try:
            h_high = slsqp_refine(h_high, ts_try, n_try, beta=150, maxiter=600)
            c5 = compute_c5(h_high)
            if c5 < best_c5:
                h_high = slsqp_refine(h_high, ts_try, n_try, beta=500, maxiter=800)
                c5 = compute_c5(h_high)
                if c5 < best_c5:
                    best_h, best_c5 = h_high.copy(), c5
        except Exception:
            pass

    # Phase 6: Final high-precision polish
    if time_left() > 15:
        n = len(best_h)
        ts = n // 2
        try:
            hp = slsqp_refine(best_h, ts, n, beta=2000, maxiter=3000)
            c5 = compute_c5(hp)
            if c5 < best_c5:
                best_h, best_c5 = hp.copy(), c5
        except Exception:
            pass

    n = len(best_h)
    best_h = project_h(best_h, n / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
