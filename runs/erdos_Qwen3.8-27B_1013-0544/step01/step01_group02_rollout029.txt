def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            r = 1.0 - h
            if r.sum() < 1e-15:
                break
            h += (d / r.sum()) * r
        else:
            if h.sum() < 1e-15:
                break
            h += (d / h.sum()) * h
    return np.clip(h, 0.0, 1.0)
