import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-15:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    c_shifted = alpha * (c - m)
    mx = c_shifted.max()
    if mx > 700:
        return float(m + (mx + np.log(np.sum(np.exp(c_shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(c_shifted))) / alpha)

def sum_con(h, n):
    return np.sum(h) - n / 2.0

def sum_con_jac(h, n):
    return np.ones(n)

def make_binary_initial(n, rng):
    """Create a binary initial configuration with n/2 ones"""
    # Use a quasi-random (golden ratio) distribution
    golden = 0.618033988749895
    ones_positions = set()
    for i in range(n // 2):
        pos = int((i * golden * n) % n)
        while pos in ones_positions:
            pos = (pos + 1) % n
        ones_positions.add(pos)
    h = np.zeros(n)
    for pos in ones_positions:
        h[pos] = 1.0
    return h

def sa_binary(h, n, rng, t_end, max_iters=500000):
    """Simulated annealing on binary configuration"""
    h = h.astype(int).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h.astype(float), n)
    current_c5 = best_c5
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    T = 1.0
    T_min = 0.001
    cooling = (T_min / T) ** (1.0 / max_iters)
    
    for it in range(max_iters):
        if time.time() > t_end:
            break
        # Move: swap a 1 with a 0
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0
        h[j] = 1
        new_c5 = compute_c5(h.astype(float), n)
        delta = new_c5 - current_c5
        if delta < 0 or rng.random() < np.exp(-delta / T):
            current_c5 = new_c5
            # Update ones and zeros
            ones = ones[ones != i]
            ones = np.append(ones, j)
            zeros = zeros[zeros != j]
            zeros = np.append(zeros, i)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        else:
            h[i] = 1
            h[j] = 0
        T *= cooling
    
    return best_h, best_c5
