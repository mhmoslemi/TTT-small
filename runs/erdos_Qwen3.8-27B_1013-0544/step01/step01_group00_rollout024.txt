import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_h(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = np.sum(mask)
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def t_objective(v):
    return v[-1]

def t_ineq_full(v):
    n = len(v) - 1
    h = v[:n]
    t = v[-1]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return t - corr

def t_eq_fun(v):
    n = len(v) - 1
    return np.sum(v[:n]) - n * 0.5

def t_eq_jac(v):
    n = len(v) - 1
    j = np.zeros(len(v))
    j[:n] = 1.0
    return j

def optimize_t_var(h0, max_iter=500, ftol=1e-10):
    n = len(h0)
    c5_cur = compute_c5(h0)
    v0 = np.append(h0.copy(), c5_cur + 0.05)
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.5)]
    cons = [
        {'type': 'ineq', 'fun': t_ineq_full},
        {'type': 'eq', 'fun': t_eq_fun, 'jac': t_eq_jac}
    ]
    res = minimize(t_objective, v0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': ftol})
    h_new = np.clip(res.x[:n], 0.0, 1.0)
    h_new = project_h(h_new, n * 0.5)
    return h_new, compute_c5(h_new)

def pnorm_obj(h, p, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    val = (np.sum(np.maximum(corr, 0)**p)) ** (1.0/p)
    return val + lam * (np.sum(h) - n*0.5)**2

def optimize_pnorm(h0, p=30, lam=100.0, max_iter=200):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    res = minimize(pnorm_obj, h0.copy(), args=(p, lam), method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': max_iter, 'gtol': 1e-8})
    h_new = np.clip(res.x, 0.0, 1.0)
    h_new = project_h(h_new, n * 0.5)
    return h_new, compute_c5(h_new)

def interp_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 1, n_old, endpoint=False)

    dx = 2.0 / len(h)

    return float(np.max(corr))

def project_h(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-10
    else:
        mask = h > 1e-10
    cnt = np.sum(mask)
    if cnt > 0:
        adj = diff / cnt
        h[mask] = np.clip(h[mask] + adj, 0.0, 1.0)
    return h

    n = len(h)

    m = np.max(corr)
    val = m + temp * np.log(np.sum(np.exp((corr - m) / temp)))
    return val + lam * (np.sum(h) - n * 0.5) ** 2

        res = minimize(lse_obj, h_cur.copy(), args=(temp, lam),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': max_iter, 'gtol': 1e-10})
        h_new = np.clip(res.x, 0.0, 1.0)
        h_new = project_h(h_new, n * 0.5)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur:
            h_cur, c5_cur = h_new, c5_new
    return h_cur, c5_cur


def t_obj(v):
    return v[-1]


def t_ineq(v):
    n = len(v) - 1
    h = v[:n]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return v[-1] - corr


def t_eq(v):
    return np.sum(v[:-1]) - (len(v) - 1) * 0.5


def t_eq_jac(v):
    j = np.zeros(len(v))
    j[:-1] = 1.0
    return j


def optimize_tv(h0, max_iter=600, ftol=1e-12):
    n = len(h0)
    c0 = compute_c5(h0)
    v0 = np.append(h0.copy(), c0 + 0.04)
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [
        {'type': 'ineq', 'fun': t_ineq},
        {'type': 'eq', 'fun': t_eq, 'jac': t_eq_jac}
    ]
    res = minimize(t_obj, v0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': ftol})
    h_new = np.clip(res.x[:n], 0.0, 1.0)
    h_new = project_h(h_new, n * 0.5)
    return h_new, compute_c5(h_new)


def interp_to(h, n_new):
    x_old = np.linspace(0, 1, len(h), endpoint=False)
    x_new = np.linspace(0, 1, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_h(h_new, n_new * 0.5)


def binary_local_search(h_init, rng, n_iter=5000):
    n = len(h_init)
    h_bin = np.zeros(n)
    order = np.argsort(-h_init)
    h_bin[order[:n // 2]] = 1.0
    c5_cur = compute_c5(h_bin)
    best = h_bin.copy()
    for _ in range(n_iter):
        zeros = np.where(h_bin == 0)[0]
        ones = np.where(h_bin == 1)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        h_bin[i], h_bin[j] = 1.0, 0.0
        c5_new = compute_c5(h_bin)
        if c5_new < c5_cur:
            c5_cur = c5_new
            best = h_bin.copy()
        else:
            h_bin[i], h_bin[j] = 0.0, 1.0
            c5_cur = compute_c5(h_bin)
    return best, compute_c5(best)


def fourier_perturb(h, rng, n_modes, scale):
    n = len(h)
    x = np.arange(n)
    pert = np.zeros(n)
    for k in range(1, n_modes + 1):
        a = rng.randn() * scale / np.sqrt(k)
        b = rng.randn() * scale / np.sqrt(k)
        pert += a * np.sin(2 * np.pi * k * x / n) + b * np.cos(2 * np.pi * k * x / n)
    return project_h(h + pert, n * 0.5)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - t0)

    h = np.array(initial_h_values, dtype=float)
    n = len(h)
    h = project_h(h, n * 0.5)
    c5 = compute_c5(h)
    h_best, c5_best = h.copy(), c5

    # Phase 1: L-BFGS-B with LSE at parent resolution
    if tl() > 120:
        h_new, c5_new = optimize_lse(h_best,
                                     temps=[0.05, 0.02, 0.01, 0.005, 0.002, 0.001],
                                     lam=3000.0, max_iter=500)
        if c5_new < c5_best:
            h_best, c5_best = h_new, c5_new

    # Phase 2: t-var SLSQP polish
    if tl() > 80:
        h_new, c5_new = optimize_tv(h_best, max_iter=800, ftol=1e-13)
        if c5_new < c5_best:
            h_best, c5_best = h_new, c5_new

    # Phase 3: Binary local search for alternative starts
    if tl() > 150:
        h_bin, c5_bin = binary_local_search(h_best, rng, n_iter=8000)
        if c5_bin < c5_best:
            h_best, c5_best = h_bin, c5_bin
        # Refine binary solution continuously
        if tl() > 100:
            h_new, c5_new = optimize_lse(h_bin, [0.03, 0.01, 0.005], lam=3000.0, max_iter=300)
            if c5_new < c5_best:
                h_best, c5_best = h_new, c5_new
            if tl() > 60:
                h_new2, c5_new2 = optimize_tv(h_new, max_iter=500, ftol=1e-12)
                if c5_new2 < c5_best:
                    h_best, c5_best = h_new2, c5_new2

    # Phase 4: Resolution increase to 240
    if tl() > 250:
        h240 = interp_to(h_best, 240)
        c5_240 = compute_c5(h240)
        h_cur = h240 if c5_240 < c5_best else h_best
        n_cur = len(h_cur)
        h_new, c5_new = optimize_lse(h_cur, [0.04, 0.02, 0.01, 0.005], lam=3000.0, max_iter=500)
        if c5_new < c5_best:
            h_best, c5_best = h_new, c5_new
        if tl() > 100:
            h_new, c5_new = optimize_tv(h_best, max_iter=800, ftol=1e-13)
            if c5_new < c5_best:
                h_best, c5_best = h_new, c5_new

    # Phase 5: Multi-start exploration at best resolution
    n_cur = len(h_best)
    n_starts = 25
    for i in range(n_starts):
        if tl() < 50:
            break
        r = rng.rand()
        if r < 0.4:
            scale = 0.02 + 0.04 * rng.rand()
            h_p = h_best + rng.randn(n_cur) * scale
        elif r < 0.7:
            h_p = fourier_perturb(h_best, rng, n_modes=3 + int(4 * rng.rand()),
                                  scale=0.05 + 0.05 * rng.rand())
        else:
            h_bin2, _ = binary_local_search(h_best + rng.randn(n_cur) * 0.1, rng, n_iter=2000)
            h_p = h_bin2
        h_p = project_h(h_p, n_cur * 0.5)
        h_new, c5_new = optimize_lse(h_p, [0.02, 0.01, 0.005], lam=3000.0, max_iter=250)
        if c5_new < c5_best:
            h_best, c5_best = h_new, c5_new
            if tl() > 50:
                h_new2, c5_new2 = optimize_tv(h_best, max_iter=400, ftol=1e-12)
                if c5_new2 < c5_best:
                    h_best, c5_best = h_new2, c5_new2

    # Phase 6: Try higher resolution if time remains
    if tl() > 200 and len(h_best) <= 240:
        n_hi = 300 if tl() > 350 else 260
        h_hi = interp_to(h_best, n_hi)
        h_new, c5_new = optimize_lse(h_hi, [0.03, 0.015, 0.008], lam=3000.0, max_iter=400)
        if tl() > 80:
            h_new, c5_new = optimize_tv(h_new, max_iter=600, ftol=1e-13)
        if c5_new < c5_best:
            h_best, c5_best = h_new, c5_new

    # Final polish
    if tl() > 40:
        h_new, c5_new = optimize_tv(h_best, max_iter=600, ftol=1e-14)
        if c5_new < c5_best:
            h_best, c5_best = h_new, c5_new

    h_best = project_h(h_best, len(h_best) * 0.5)
    c5_final = compute_c5(h_best)
    return h_best, c5_final, len(h_best)
