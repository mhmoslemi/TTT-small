def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):

def slsqp_optimize(h_start, n, dx, max_iter=500):
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    constraint = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    try:
        res = minimize(
            lambda x: compute_c5(x, dx),
            h_start.copy(),
            method='SLSQP',
            bounds=bounds,
            constraints=constraint,
            options={'maxiter': max_iter, 'ftol': 1e-14}
        )
        return project_feasible(res.x, n)
    except Exception:
        return h_start.copy()

def make_thue_morse(n):
    tm = np.zeros(n, dtype=np.float64)
    for i in range(1, n):
        tm[i] = tm[i >> 1] + (i & 1)
    tm = (tm % 2).astype(np.float64)
    n_ones = int(tm.sum())
    target = n // 2
    if n_ones > target:
        idx = np.where(tm == 1)[0]
        tm[np.random.RandomState(0).choice(idx, n_ones - target, replace=False)] = 0
    elif n_ones < target:
        idx = np.where(tm == 0)[0]
        tm[np.random.RandomState(0).choice(idx, target - n_ones, replace=False)] = 1
    return tm

def make_block(n, period, ones_per):
    h = np.zeros(n)
    for i in range(n):
        if (i % period) < ones_per:
            h[i] = 1.0

    elif n_ones < target:
        idx = np.where(h == 0)[0]
        h[rng.choice(idx, target - n_ones, replace=False)] = 1
    return h
