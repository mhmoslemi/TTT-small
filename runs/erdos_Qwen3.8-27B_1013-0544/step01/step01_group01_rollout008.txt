                h4, c5_4 = grad_descent(best_h, 3000, 0.03, 0.85, target_sum, time_left, top_k=3)
                if c5_4 < best_c5:
                    best_h, best_c5 = h4, c5_4

    # Phase 5: Perturbed restart
    if time_left() > 80:
        h5 = best_h + np.random.randn(n_points) * 0.08
        h5 = project_h(h5, target_sum)
        h5, c5_5 = grad_descent(h5, 3000, 0.04, 0.85, target_sum, time_left, top_k=4)
        if c5_5 < best_c5:
            best_h, best_c5 = h5, c5_5
        if time_left() > 30:
            best_h, best_c5 = slsqp_polish(best_h, target_sum, time_left, maxiter=500, beta=60)

    # Phase 6: Try different n values
    for n_try in [64, 66, 68, 72, 74, 76, 80, 84]:
        if time_left() < 50:
            break
        n_cur = len(best_h)
        x_o = np.linspace(0, 2, n_cur)
        x_n = np.linspace(0, 2, n_try)
        h_i = np.interp(x_n, x_o, best_h)
        h_i = project_h(h_i, n_try / 2.0)
        ts_i = n_try / 2.0
        h_t, c5_t = grad_descent(h_i, 2000, 0.04, 0.85, ts_i, time_left, top_k=3)
        if time_left() > 30:
            h_t, c5_t = slsqp_polish(h_t, ts_i, time_left, maxiter=300, beta=40)
        if c5_t < best_c5:
            best_h, best_c5 = h_t, c5_t

    # Phase 7: More GD + binary search + polish
    if time_left() > 60:
        n_f = len(best_h)
        ts_f = n_f / 2.0
        h_g, c5_g = grad_descent(best_h, 4000, 0.03, 0.88, ts_f, time_left, top_k=5)
        if c5_g < best_c5:
            best_h, best_c5 = h_g, c5_g
        if time_left() > 40:
            hb2, c5_b2 = binary_local_search(best_h, ts_f, time_left, max_iters=2000)
            if c5_b2 < best_c5:
                best_h, best_c5 = hb2, c5_b2
                if time_left() > 30:
                    best_h, best_c5 = grad_descent(best_h, 2000, 0.03, 0.85, ts_f, time_left, top_k=3)

    # Phase 8: Final high-precision SLSQP
    if time_left() > 20:
        best_h, best_c5 = slsqp_polish(best_h, len(best_h)/2.0, time_left, maxiter=3000, beta=200)

    # Phase 9: One more GD pass
    if time_left() > 20:
        n_f = len(best_h)
        h_final, c5_final = grad_descent(best_h, 3000, 0.02, 0.9, n_f/2.0, time_left, top_k=5)
        if c5_final < best_c5:
            best_h, best_c5 = h_final, c5_final

    return best_h, best_c5, len(best_h)
