import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))


def project_h(h, target):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = float(np.sum(mask))
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)


def smooth_obj(h, temp):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    s = (corr - m) / temp
    return m + temp * np.log(np.sum(np.exp(s)))


def sum_eq_fun(h):
    return np.sum(h) - len(h) * 0.5


def sum_eq_jac(h):
    return np.ones(len(h))


def t_obj(v):
    return v[-1]


def t_ineq(v):
    n = len(v) - 1
    h = v[:n]
    t = v[-1]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return t - corr


def opt_smooth(h0, temp, max_iter=500):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac}]
    res = minimize(smooth_obj, h0.copy(), args=(temp,), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-14})
    h_new = np.clip(res.x, 0.0, 1.0)
    h_new = project_h(h_new, n * 0.5)
    return h_new, compute_c5(h_new)


def opt_tvar(h0, max_iter=500):
    n = len(h0)
    c5_cur = compute_c5(h0)
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.5)]
    cons = [{'type': 'ineq', 'fun': t_ineq},
            {'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac}]
    res = minimize(t_obj, np.append(h0.copy(), c5_cur + 0.01),
                   method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12})
    h_new = np.clip(res.x[:n], 0.0, 1.0)
    h_new = project_h(h_new, n * 0.5)
    return h_new, compute_c5(h_new)


def binary_push(h, strength):
    n = len(h)
    h_new = h.copy()
    below = h_new < 0.5
    above = h_new > 0.5
    h_new[below] *= (1.0 - strength)
    h_new[above] = 1.0 - (1.0 - h_new[above]) * (1.0 - strength)
    return project_h(np.clip(h_new, 0.0, 1.0), n * 0.5)


def subgrad_step(h, alpha0, n_iter, decay):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    alpha = alpha0
    for _ in range(n_iter):
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full') * dx
        r_star = np.argmax(corr)
        offset = r_star - (n - 1)
        grad = np.zeros(n)
        lo1, hi1 = max(0, offset), min(n, n + offset)
        if hi1 > lo1:
            grad[lo1:hi1] += (1.0 - h_cur[lo1 - offset:hi1 - offset]) * dx
        lo2, hi2 = max(0, -offset), min(n, n - offset)
        if hi2 > lo2:
            grad[lo2:hi2] -= h_cur[lo2 + offset:hi2 + offset] * dx
        h_cur -= alpha * grad
        h_cur = project_h(h_cur, n * 0.5)
        alpha *= decay
    return h_cur, compute_c5(h_cur)


def active_smooth_obj(h, top_idx):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    top_vals = corr[top_idx]
    m = np.max(top_vals)
    s = (top_vals - m) / 0.005
    return m + 0.005 * np.log(np.sum(np.exp(s))) + 0.01 * np.max(corr)


def opt_active(h0, top_k=10, max_iter=300):
    n = len(h0)
    dx = 2.0 / n
    corr = np.correlate(h0, 1.0 - h0, mode='full') * dx
    top_idx = np.argsort(corr)[-top_k:]

    def obj(h):
        return active_smooth_obj(h, top_idx)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac}]
    res = minimize(obj, h0.copy(), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-14})
    h_new = np.clip(res.x, 0.0, 1.0)
    h_new = project_h(h_new, n * 0.5)
    return h_new, compute_c5(h_new)


def interp_to(h_src, n_new):
    x_old = np.arange(len(h_src)) / len(h_src)
    x_new_arr = np.arange(n_new) / n_new
    h_new = np.interp(x_new_arr, x_old, h_src)
    return project_h(h_new, n_new * 0.5)

    t0 = time.time()
    rng = np.random.RandomState(seed)

    def time_left():

        h_new, c5_new = opt_smooth(h, 0.01, 400)
        if c5_new < c5:

            h_best = h.copy()
        h = binary_push(h_best, 0.2 + 0.05 * cycle)
        h_new, c5_new = opt_smooth(h, 0.005, 400)
        if c5_new < c5:
