def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        x = np.clip(h - mid, 0.0, 1.0)
        if np.sum(x) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def gradient_c5(h, t, dx):
    n = len(h)
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    max_corr = np.max(corr)
    exp_vals = np.exp(t * (corr - max_corr))
    w = exp_vals / np.sum(exp_vals)
    
    A = np.zeros(n)
    B = np.zeros(n)
    
    for m in range(2 * n - 1):
        if w[m] < 1e-12:
            continue
        d = m - (n - 1)
        if d == 0:
            A += w[m] * h
            B += w[m] * h
        elif d > 0:
            A[:n-d] += w[m] * h[d:]
            B[d:] += w[m] * h[:n-d]
        else:
            dd = -d
            A[dd:] += w[m] * h[:n-dd]
            B[:n-dd] += w[m] * h[dd:]
    
    grad = dx * (1.0 - A - B)
    c5_val = max_corr * dx
    return grad, c5_val
