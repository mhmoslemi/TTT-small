def coordinate_sweep(h, n):
    """One full coordinate descent sweep."""
    corr = compute_corr(h)  # shape (2n-1,)
    dx = 2.0 / n
    
    for j in range(n):
        # Compute a_k for all k
        a = np.zeros(2*n-1)
        for idx in range(2*n-1):
            k = idx - (n-1)
            a[idx] = grad_contribution(h, j, k, n)
        
        # b_k = f_k - a_k * h[j]
        # f_k = corr[idx] * dx (but corr is already the sum, not the integral)
        # Actually corr = np.correlate(h, 1-h, mode='full'), so corr[idx] is the sum
        # f_k = corr[idx] * dx is the integral
        # But for the optimization, we can work with the sums directly
        b = corr - a * h[j]
        
        # Minimize max_k (a_k * t + b_k) * dx over t ∈ [0,1]
        # The dx factor doesn't affect the minimizer
        t_opt = minimax_1d(a, b)

    if k >= 0:
        if j <= n-1-k:
            a += (1.0 - h[j+k])
        if j >= k and k > 0:
            a += -h[j-k]
    else:
        p = -k
        if j >= p:
            a += (1.0 - h[j-p])
        if j <= n-1-p and p > 0:
            a += -h[j+p]
    return a

def minimax_1d(a, b):
    """Find t ∈ [0,1] that minimizes max_k (a[k]*t + b[k])."""
    # Binary search on V
    lo = np.min(np.minimum(b, a + b))  # lower bound on the min max
    hi = np.max(np.maximum(b, a + b))   # upper bound
    
    for _ in range(60):

        # Check if exists t ∈ [0,1] with a_k*t + b_k <= mid for all k
        # For a_k > 0: t <= (mid - b_k) / a_k
        # For a_k < 0: t >= (mid - b_k) / a_k
        # For a_k == 0: need b_k <= mid
        
        pos = a > 1e-15
        neg = a < -1e-15
        zero = ~pos & ~neg

import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1 - h, mode='full')

def compute_c5(h):
    n = len(h)
    return np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n)

def project_simplex_box(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def coord_a(h, j, k, n):
    """Coefficient of h[j] in the sum for shift k."""
    a = 0.0
    if k >= 0:
        if j <= n - 1 - k:
            a += 1.0 - h[j + k]
        if k > 0 and j >= k:
            a -= h[j - k]
    else:
        p = -k
        if j >= p:
            a += 1.0 - h[j - p]
        if p > 0 and j <= n - 1 - p:
            a -= h[j + p]
    return a

def minimax_linear(a, b):
    """Find t in [0,1] minimizing max_k(a[k]*t + b[k])."""
    # Binary search on the optimal value V
    vals_0 = b  # values at t=0
    vals_1 = a + b  # values at t=1
    lo = min(np.min(vals_0), np.min(vals_1))
    hi = max(np.max(vals_0), np.max(vals_1))
    
    pos = a > 1e-12
    neg = a < -1e-12
    zero = ~(pos | neg)
    
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.any(b[zero] > mid):
            hi = mid
            continue
        t_lo = 0.0
        t_hi = 1.0
        if np.any(neg):
            t_lo = max(t_lo, float(np.max((mid - b[neg]) / a[neg])))
        if np.any(pos):
            t_hi = min(t_hi, float(np.min((mid - b[pos]) / a[pos])))
        if t_lo <= t_hi:
            hi = mid
        else:
            lo = mid
    
    V = (lo + hi) / 2.0
    t_lo = 0.0
    t_hi = 1.0
    if np.any(neg):
        t_lo = max(t_lo, float(np.max((V - b[neg]) / a[neg])))
    if np.any(pos):
        t_hi = min(t_hi, float(np.min((V - b[pos]) / a[pos])))
    return np.clip((t_lo + t_hi) / 2.0, 0.0, 1.0)

def coordinate_sweep(h, n):
    """One full coordinate descent sweep."""
    corr = compute_corr(h)
    for j in range(n):
        # Compute a vector for all shifts
        a = np.zeros(2 * n - 1)
        for idx in range(2 * n - 1):
            k = idx - (n - 1)
            a[idx] = coord_a(h, j, k, n)
        b = corr - a * h[j]
        t_opt = minimax_linear(a, b)
        h[j] = t_opt
        # Update corr incrementally? Or recompute?
        # For now, recompute at the end of the sweep
    corr = compute_corr(h)
    return h, corr

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    # Get initial solution
    if 'initial_h_values' in globals() and initial_h_values is not None:

    else:
        n = 200
        h0 = 0.5 + 0.3 * rng.randn(n)

    n = len(h0)
    best_h = project_simplex_box(h0.copy(), n / 2.0)
    best_c5 = compute_c5(best_h)
    
    # Coordinate descent phase
    h_work = best_h.copy()
    sweeps = 0
    stall = 0
    
    while time.time() < t0 + budget_s - 60:
        h_work, corr = coordinate_sweep(h_work, n)
        # Project to satisfy sum constraint
        h_proj = project_simplex_box(h_work, n / 2.0)
        c5 = compute_c5(h_proj)
        sweeps += 1

            best_c5 = c5

            stall = 0

                h_work = best_h + pert
                h_work = project_simplex_box(h_work, n / 2.0)
                stall = 0
                if compute_c5(h_work) < best_c5:

                    best_c5 = compute_c5(best_h)
    
    # Gradient polish
    remaining = budget_s - (time.time() - t0)
    if remaining > 30:
        # Do a few gradient steps with the corrected gradient
        h_work = best_h.copy()
        alpha = 200
        lr = 0.003
        for _ in range(200):

                break
            corr = compute_corr(h_work)
            m = np.max(corr)

                    a[p:] += (1.0 - h_work[:n-p])
                    if p > 0:
                        a[:n-p] -= h_work[p:]
                grad += w[idx] * a

            h_new = project_simplex_box(h_new, n / 2.0)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
            h_work = h_new
        
        # Try higher resolution
        remaining = budget_s - (time.time() - t0)
        if remaining > 60:
            for n_try in [300, 400]:
                if remaining < 30:
                    break
                # Interpolate
                x_old = np.linspace(0, 2, n, endpoint=False) + 1.0/n
                x_new = np.linspace(0, 2, n_try, endpoint=False) + 1.0/n_try
                h_interp = np.interp(x_new, np.append(x_old, 2.0 + 1.0/n),
                                     np.append(best_h, best_h[0]))
                h_interp = project_simplex_box(h_interp, n_try / 2.0)
                
                # Coordinate descent at higher res

                for _ in range(5):

                    h_proj = project_simplex_box(h_hi, n_try / 2.0)

                        best_c5 = c5_hi

- Then minimax_linear: 50 binary search steps, each O(399) with numpy → 50*399 = 20K operations

In Python, 4M simple operations take about 1-2 seconds. The numpy operations in minimax_linear are fast (vectorized). The bottleneck is the 399-iteration loop for computing `a`.

If I can vectorize that, the sweep would be much faster. Let me try a different approach:

For a given j, I can precompute what `a` looks like:
- a[n-1] = (1-h[j])*[j≤n-1] + (-h[j])*[j≥0] = 1-2h[j] (for k=0)
- For k≥1: a[n-1+k] = (1-h[j+k])*[j+k≤n-1] - h[j-k]*[j-k≥0]
- For k≤-1: a[n-1+k] = (1-h[j+k])*[j+k≥0] - h[j-k]*[j-k≤n-1]
  (where k is negative, so j+k = j-|k|, j-k = j+|k|)

Let me just write it as:
```python
def compute_a_fast(h, j, n):
    a = np.zeros(2*n - 1)
    # For all k from -(n-1) to n-1:
    # a[n-1+k] = (1-h[j+k]) if 0<=j+k<n else 0
    #           - h[j-k] if 0<=j-k<n else 0
    # (Note: for k=0, both terms apply, giving 1-2h[j])
    
    # First part: (1-h[j+k]) for valid j+k
    lo = max(0, -j)  # k >= -j so j+k >= 0
    hi = min(n-1, n-1-j)  # k <= n-1-j so j+k <= n-1
    if lo <= hi:
        k_vals = np.arange(lo, hi+1)
        a[n-1 + k_vals] += 1.0 - h[j + k_vals]
    
    # Second part: -h[j-k] for valid j-k
    lo2 = max(0, j-(n-1))  # k >= j-(n-1) so j-k <= n-1
    hi2 = min(n-1, j)  # k <= j so j-k >= 0
    if lo2 <= hi2:
        k_vals = np.arange(lo2, hi2+1)
        a[n-1 + k_vals] -= h[j - k_vals]
    
    return a
```

- Second part: lo2 = max(0, 1-3)=0, hi2 = min(3,1)=1

Now, the question is: will coordinate descent converge to a good solution? Let me think about potential issues:

1. **Cycling**: In general, coordinate descent can cycle (especially for non-convex problems). However, since each step minimizes a convex function (in the 1D sense), the objective should be non-increasing per coordinate update. So it shouldn't cycle in the objective value.

Wait, I need to be more careful. The objective is max_k f_k(h). When I update h[j] to minimize max_k f_k as a function of h[j] (with other coordinates fixed), the new value of max_k f_k is ≤ the old value. So yes, the objective is non-increasing. No cycling possible.

2. **Convergence to local minimum**: Since the problem is non-convex (the f_k are bilinear), coordinate descent can get stuck in local minima. Mitigation: multiple restarts with perturbation.

3. **Sum constraint violation**: The coordinate descent doesn't enforce sum(h) = n/2. I need to handle this.

For the sum constraint, I have a few options:
a) After each coordinate update, if the sum drifts too far, project.
b) Modify the 1D subproblem to include a Lagrange multiplier for the sum.
c) Parameterize to automatically satisfy the constraint.

Option (b) is interesting. If we add a penalty μ * (sum(h) - n/2)² to the objective, then in the 1D subproblem for h[j], we minimize:
max_k (a_k * t + b_k) + μ * (S - h[j] + t - n/2)²

where S = sum of all h[i] for i≠j. This is the max of linear functions plus a quadratic, which is still convex in t. We can solve it with a similar binary search approach (or directly with scipy.optimize.minimize_scalar).

But this adds complexity. Let me start with option (a): project periodically.

Actually, let me think about whether the sum constraint is really binding. If h is close to 0.5 everywhere, the sum is already close to n/2. The coordinate descent will tend to keep h values in [0,1] and the mean will naturally be close to 0.5 (since the f_k depend on products h[i]*h[i+k], which are maximized when all h are 0.5... wait, no, f_k = sum h[i](1-h[i+k]) is maximized when h is all 0.5 or all 0/1 in a specific pattern).

Hmm, actually f_k is NOT maximized at h=0.5. f_k = sum h[i] - sum h[i]h[i+k]. At h=0.5: f_k = 0.5(n-k) - 0.25(n-k) = 0.25(n-k). At h=1 (all ones): f_k = 0. At h alternating 0,1,0,1,...: f_k depends on the pattern.

The point is: the coordinate descent is minimizing max_k f_k, and the optimum should have mean ≈ 0.5 (by symmetry and the constraint). So the sum should be close to n/2 naturally, and the projection at the end should be a small correction.

Let me just go with: do coordinate sweeps, project at the end of each sweep, and track the best.
