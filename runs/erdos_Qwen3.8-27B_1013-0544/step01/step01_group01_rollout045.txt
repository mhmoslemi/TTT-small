import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def obj_power_mean(h, dx, p):
    corr = np.correlate(h, 1.0 - h, mode="full")
    cp = np.maximum(corr, 0.0)
    return (np.sum(cp ** p) / len(cp)) ** (1.0 / p) * dx


def obj_smooth_max(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    sm = m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta
    return sm * dx


def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h


def make_thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = bin(i).count('1') % 2
    return h


def make_balanced_binary(n, seed_val):
    rng = np.random.RandomState(seed_val)
    h = np.zeros(n)
    h[:n // 2] = 1.0
    rng.shuffle(h)
    return h


def binary_local_search(h, n, max_iter=30000, time_limit=None, t0=None):
    target_sum = n / 2.0
    h = np.round(h).astype(float)
    s = h.sum()
    while s > target_sum:
        idx = np.where(h == 1.0)[0]
        if len(idx) == 0:
            break
        h[np.random.choice(idx)] = 0.0
        s -= 1
    while s < target_sum:
        idx = np.where(h == 0.0)[0]
        if len(idx) == 0:
            break
        h[np.random.choice(idx)] = 1.0
        s += 1
    best = compute_c5(h)
    no_imp = 0
    for it in range(max_iter):
        if time_limit is not None and (time.time() - t0) > time_limit:

            best = c5
            no_imp = 0
        else:
            h[i], h[j] = h[j], h[i]
            no_imp += 1
        if no_imp > 8000:
            break
    return h, best


def continuous_refine(h, n, dx, target_sum, time_remaining):
    bounds = Bounds(np.zeros(n), np.ones(n))
    # Phase A: L-BFGS-B with power mean

        def obj_pm(x):
            return obj_power_mean(x, dx, 25.0)
        try:
            res = minimize(obj_pm, h, method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 500, 'ftol': 1e-14, 'gtol': 1e-12})
            h_new = project_h(res.x, target_sum)
            c5 = compute_c5(h_new)
            if c5 < compute_c5(h):
                h = h_new
        except Exception:
            pass
    # Phase B: SLSQP with smooth max

            return obj_smooth_max(x, dx, 150.0)

            return np.sum(x) - target_sum

            return np.ones(n)

            res = minimize(obj_sm, h, method='SLSQP', bounds=bounds,
