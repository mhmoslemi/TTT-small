        while time_remaining(start_time, budget_s) > 25:
            attempt += 1
            strategy = attempt % 5
            h0 = h_work.copy()
            corr = get_correlations(h0)
            top_k = np.argsort(corr)[-min(6, K):]

            if strategy == 0:
                # Targeted anti-lag perturbation
                for lag_idx in top_k[:3]:
                    lag = int(lag_idx) - (n - 1)
                    for j in range(n):
                        i1 = j - lag
                        if 0 <= i1 < n:
                            h0[j] -= 0.015 * (1.0 - h0[i1])
                        i2 = j + lag
                        if 0 <= i2 < n:
                            h0[j] -= 0.015 * h0[i2]
            elif strategy == 1:
                # Moderate Gaussian
                scale = 0.03 + 0.06 * np.exp(-attempt / 30.0)
                h0 += np.random.randn(n) * scale
            elif strategy == 2:
                # Local block shuffle
                blen = np.random.randint(4, max(5, n // 6))
                i = np.random.randint(0, n - blen)
                j = np.random.randint(0, n - blen)
                if j < i:
                    i, j = j, i
                seg = h0[i:j + blen].copy()
                h0[i:j + blen] = np.roll(seg, np.random.randint(1, blen))
            elif strategy == 3:
                # Sparse large perturbation
                mask = np.random.random(n) < 0.12
                h0[mask] = np.random.random(int(mask.sum()))
            else:
                # Structured restart
                freq = np.random.uniform(0.5, 3.0)
                phase = np.random.uniform(0, 2 * np.pi)
                h0 = 0.5 + 0.25 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
                h0 += np.random.randn(n) * 0.05

            h0 = project_h(h0, target_sum)
            for beta_val in [80.0, 500.0]:
                if time_remaining(start_time, budget_s) < 12:
                    break
                corr = get_correlations(h0)
                active_idx = np.argsort(corr)[-K:]
                h0 = try_slsqp(h0, bounds, constraints, obj_active,
                               (beta_val, K, n, active_idx), 250, 1e-13)
                h0 = project_h(h0, target_sum)
                c5_cur = compute_c5(h0)
                if c5_cur < best_c5:
                    best_h, best_c5, best_n = h0.copy(), c5_cur, n
                    h_work = h0.copy()

        # Phase 4: Ultra-high precision final polish
        if time_remaining(start_time, budget_s) > 8:
            for beta_val in [3000.0, 10000.0]:
                if time_remaining(start_time, budget_s) < 5:
                    break
                corr = get_correlations(h_work)
                active_idx = np.argsort(corr)[-K:]
                h_work = try_slsqp(h_work, bounds, constraints, obj_active,
                                   (beta_val, K, n, active_idx), 3000, 1e-16)
                h_work = project_h(h_work, target_sum)
                c5_cur = compute_c5(h_work)
                if c5_cur < best_c5:
                    best_h, best_c5, best_n = h_work.copy(), c5_cur, n

    # Final verification
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, best_n
