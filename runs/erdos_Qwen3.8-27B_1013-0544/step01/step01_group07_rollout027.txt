def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    """Project h onto {x: sum(x)=target_sum, 0<=x<=1} using water-filling."""
    n = len(h)
    # Find lambda such that sum(clip(h + lambda, 0, 1)) = target_sum
    lo, hi = -h.max() - 1, 1 - h.min()
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h + mid, 0, 1))
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2, 0, 1)

def compute_gradient_smooth_max(h, t):
    """Compute gradient of (1/t)*log(sum(exp(t*corr[k])))"""
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")  # length 2n-1
    
    # Softmax weights
    shifted = corr - corr.max()
    exp_vals = np.exp(t * shifted)
    w = exp_vals / np.sum(exp_vals)
    
    # Gradient
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        k = idx - (n - 1)  # actual lag
        wk = w[idx]
        if abs(wk) < 1e-15:
            continue
        if k >= 0:
            grad[k:] += wk * (1 - h[:n-k])
            grad[:n-k] -= wk * h[k:]
        else:
            s = -k
            grad[:n-s] += wk * (1 - h[s:])
            grad[s:] -= wk * h[:n-s]
    
    return grad

        h2 = coordinate_sweep(h2, target_sum2, n_candidate=30, n_sweeps=25,
                              time_fn=elapsed, time_limit=budget_s - 60)
        c5_2 = compute_c5(h2, dx2)
        
        if c5_2 < best_c5:
            best_h = h2.copy()
            best_c5 = c5_2
            best_n = n2
    
    # Phase 5: SLSQP polish on best
    if elapsed() < budget_s - 30:
        dd = 2.0 / best_n
        ts = best_n / 2.0
        for t in [100, 300, 800]:
            if elapsed() > budget_s - 10:
                break
            h_new = slsqp_optimize(best_h, dd, ts, t=t, maxiter=500)
            c5_new = compute_c5(h_new, dd)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
    
    # Final coordinate descent
    if elapsed() < budget_s - 5:
        dd = 2.0 / best_n
        ts = best_n / 2.0
        h_cd = coordinate_sweep(best_h, ts, n_candidate=80, n_sweeps=15,
                                time_fn=elapsed, time_limit=budget_s - 2)
        c5_cd = compute_c5(h_cd, dd)
        if c5_cd < best_c5:
            best_h = h_cd.copy()
            best_c5 = c5_cd
            best_n = best_n
    
    # Ensure valid output
    best_h = np.clip(best_h, 0.0, 1.0)
    dx_final = 2.0 / best_n
    best_h = project_to_sum(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h, dx_final)
    
    return best_h, best_c5, best_n
