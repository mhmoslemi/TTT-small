```python
    corr = np.correlate(h, 1 - h, mode="full")
    max_c = np.max(corr)
    shifted = t * (corr - max_c)
    shifted -= np.max(shifted)  # numerical stability
    exps = np.exp(shifted)
    weights = exps / np.sum(exps)
```

```python
import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(h, dx, t):
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    s = t * (corr - m)
    s -= np.max(s)
    return (m + np.log(np.exp(s).sum()) / t) * dx

def smooth_max_grad(h, dx, t):
    n = len(h)
    corr = np.correlate(h, 1-h, mode="full")
    m = np.max(corr)
    s = t * (corr - m)
    s -= np.max(s)
    e = np.exp(s)
    w = e / e.sum()
    
    grad = np.zeros(n)
    half = n - 1
    for k_idx in range(2*n-1):
        if w[k_idx] < 1e-10:
            continue
        k = k_idx - half
        wi = w[k_idx]
        for i in range(n):
            v = 0.0
            if 0 <= i+k < n:
                v += 1.0 - h[i+k]
            if 0 <= i-k < n:
                v -= h[i-k]
            grad[i] += wi * v
    return grad * dx

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if mask.sum() == 0:
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def adam_optimize(h, dx, target_sum, t_init=5.0, t_final=500.0, 
                  n_steps=2000, lr=0.05, seed=42):
    n = len(h)
    rng = np.random.RandomState(seed)
    h = h.copy()
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    for step in range(n_steps):
        # Anneal temperature
        frac = step / n_steps
        t = t_init * (t_final / t_init) ** frac
        
        obj = smooth_max(h, dx, t)
        grad = smooth_max_grad(h, dx, t)
        
        # Adam update
        m = b1 * m + (1-b1) * grad
        v = b2 * v + (1-b2) * grad**2
        m_hat = m / (1 - b1**(step+1))
        v_hat = v / (1 - b2**(step+1))
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_feasible(h, target_sum)
        
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5
```

```python
    corr = np.correlate(h, 1-h, mode="full")

    for k_idx in range(2*n-1):
        if w[k_idx] < 1e-10:
            continue

        wi = w[k_idx]
        
        # Subgradient for shift k:
        # sg[i] = (1-h[i+k])*1{0<=i+k<n} - h[i-k]*1{0<=i-k<n}
        sg = np.zeros(n)
        if k >= 0:
            sg[:n-k] += (1.0 - h[k:])
            sg[k:] -= h[:n-k]
        else:
            kp = -k
            sg[kp:] += (1.0 - h[:n-kp])
            sg[:n-kp] -= h[kp:]
        
        grad += wi * sg
    
    return grad * dx
```

        grad = smooth_max_grad(h, dx, t)
        
        m1 = b1 * m1 + (1 - b1) * grad
        m2 = b2 * m2 + (1 - b2) * grad**2
        m1_hat = m1 / (1 - b1**(step+1))
        m2_hat = m2 / (1 - b2**(step+1))
        
        h = h - step_lr * m1_hat / (np.sqrt(m2_hat) + eps)
        h = project_feasible(h, target_sum)
        
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Periodic restart within adam: if stuck, add small noise
        if step % 500 == 499:
            c5_now = compute_c5(h, dx)
            if c5_now > best_c5 * 1.001:
                h = best_h.copy()
                h += np.random.randn(n) * 0.01
                h = project_feasible(h, target_sum)
    
    return best_h, best_c5

def lbfgs_polish(h, dx, target_sum, t=200, maxiter=500):
    n = len(h)
    try:
        res = minimize(
            lambda x: smooth_max_val(x, dx, t),
            h.copy(),
            jac=lambda x: smooth_max_grad(x, dx, t),
            method='L-BFGS-B',
            bounds=[(0.0, 1.0)] * n,
            # L-BFGS-B doesn't support equality constraints directly
            # We'll handle the sum constraint via penalty
            options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10}
        )
        h_new = project_feasible(res.x, target_sum)
        return h_new
    except Exception:
        return h
