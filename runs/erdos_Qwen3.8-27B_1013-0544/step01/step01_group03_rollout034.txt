from scipy.optimize import minimize, Bounds

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_onto_constraint(h, target_sum):
    """Project h onto {0 <= h <= 1, sum(h) = target_sum} via bisection on lambda."""
    h = np.array(h, dtype=float)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        clipped = np.clip(h - mid, 0.0, 1.0)
        s = clipped.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    clipped = np.clip(h - (lo + hi) / 2, 0.0, 1.0)
    # Final adjustment
    diff = target_sum - clipped.sum()
    if abs(diff) > 1e-12:
        mask = (clipped > 0) & (clipped < 1)
        if mask.sum() > 0:
            clipped[mask] += diff / mask.sum()
            clipped = np.clip(clipped, 0.0, 1.0)
    return clipped

def compute_all_lags(h):
    """Compute all u_k values (the correlation values times dx)."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def subgradient_step(h, target_sum, lr):
    """One subgradient step for minimizing max_k u_k."""
    vals = compute_all_lags(h)
    k_max = np.argmax(vals)
    n = len(h)
    # Subgradient of u_{k_max} w.r.t. h
    # u_k = sum_i h[i] * (1-h[i+j]) * dx where j = k - (n-1)
    j = k_max - (n - 1)
    subgrad = np.zeros(n)
    if j >= 0:
        for i in range(n - j):
            subgrad[i] += (1.0 - h[i + j]) * dx
            subgrad[i + j] += (-h[i]) * dx  # derivative of -h[i]*h[i+j] w.r.t. h[i+j]

    h_new = project_onto_constraint(h_new, target_sum)
    return h_new

def subgradient_descent(h_init, target_sum, max_iter=5000, time_limit=None, t0=None):
    """Subgradient descent with polynomial step size."""
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Adaptive step size
    prev_val = best_c5
    step = 0.1
    
    for it in range(max_iter):
        if time_limit is not None:
            if time.time() - t0 > time_limit:
                break
        
        h = subgradient_step(h, target_sum, step)
        c5 = evaluate_c5(h)
        
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
        
        # Adaptive step: decrease if no improvement
        if c5 > prev_val * 1.01:
            step *= 0.9
        else:
            step *= 1.01
        step = min(step, 0.5)
        prev_val = c5
    
    return best_h, best_c5

def make_cosine_init(n, a=0.5, b=0.0, c=0.0):
    """Create cosine-based initialization."""
    x = np.linspace(0, 2, n, endpoint=False) + 1.0/n  # midpoints

    h = np.clip(h, 0.0, 1.0)
    h = project_onto_constraint(h, n/2.0)
    return h

def slsqp_polish(h, target_sum, maxiter=200):
    """Polish with SLSQP using smoothed max."""
    n = len(h)
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        tau = 500
        x_tau = tau * corr
        x_max = np.max(x_tau)
        return float(x_max / tau + np.log(np.sum(np.exp(x_tau - x_max))) / tau)
    
    bnds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bnds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        x = project_onto_constraint(res.x, target_sum)
        c5 = evaluate_c5(x)
        return x, c5
    except:
        return h, evaluate_c5(h)

The key issue with subgradient methods is slow convergence. The rate is O(1/√T) for the standard method. To get from C5=0.5 (constant) to C5=0.38, we need significant improvement, and subgradient might not be fast enough.

Better approaches:
1. **Use the smooth max with very high temperature** (essentially exact max) in SLSQP, starting from a good initialization.
2. **Use a trust-region method** that handles the nonsmoothness.
3. **Use the "max of linear functions" structure**: at any point h0, the active lag k* gives a linear constraint. We can maintain a set of active constraints and solve a QP.

Actually, here's a better idea: **Use the fact that near the optimum, only a few lags are active (close to the max).** We can:
1. Compute all lags
2. Identify the top few (say top 5) lags
3. Solve: minimize max(u_1, ..., u_5) subject to constraints
   This is equivalent to: minimize t subject to u_k ≤ t for k=1,...,5
   But u_k is bilinear, so this is a non-convex QP.
   
   However, we can linearize: at the current point, u_k(h) ≈ u_k(h0) + ∇u_k(h0)·(h-h0)
   Then solve: minimize t subject to u_k(h0) + ∇u_k(h0)·(h-h0) ≤ t, 0≤h≤1, sum=h0
   This is an LP!
   
   Then take a damped step towards the LP solution.

This is essentially the **bundle method** for nonsmooth optimization, which has much better convergence than plain subgradient.

def bundle_iteration(h, target_sum, n_active=5, damp=0.5):
    n = len(h)
    dx = 2.0 / n
    vals = np.correlate(h, 1-h, mode="full") * dx
    # Get top lags
    top_k = np.argsort(vals)[-n_active:]
    
    # Linearize around h

def project_constraint(h, target):

    """Compute averaged subgradient from top-k active lags."""
