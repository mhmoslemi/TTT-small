            break
        step = diff / cnt
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_descent_step(h, n, target):
    """One pass of coordinate descent.
    For each i, compute the linear coefficients a_d and b_d,
    then find t in [0,1] minimizing max_d (a_d*t + b_d).
    """
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode="full")
    # d ranges from -(n-1) to (n-1), index j = d + (n-1) ranges from 0 to 2n-2
    
    for i in range(n):
        # Compute a_d and b_d for all d
        # c(d) = a_d * h[i] + b_d
        # h[i] appears in c(d) as:
        #   - h[n+d] when n = i-d (if valid): contributes h[i]*(1-h[i-d])
        #   - h[n] when n = i (if valid): contributes h[i+d]*(1-h[i])
        
        # For each d (represented by j = d + n - 1):
        # a_d = (1-h[i-d]) if 0<=i-d<n else 0  [from h[n+d]=h[i] term]
        #       minus h[i+d] if 0<=i<n and 0<=i+d<n else 0  [from h[n]=h[i] term]
        # b_d = rest of c(d)
        
        # Actually, let me compute this differently.
        # c(d) = sum_{n in S(d)} h[n+d]*(1-h[n])
        # The terms involving h[i]:
        #   Term 1: n=i-d, if i-d in S(d): h[i]*(1-h[i-d])
        #   Term 2: n=i, if i in S(d): h[i+d]*(1-h[i])
        
        # So c(d) = h[i]*(1-h[i-d])*I1(d) + h[i+d]*(1-h[i])*I2(d) + rest(d)
        # where I1(d) = 1 if 0<=i-d<n and 0<=i<n (the second always holds)
        #        I2(d) = 1 if 0<=i<n and 0<=i+d<n (the first always holds)
        
        # = h[i]*(1-h[i-d])*I1(d) + h[i+d]*I2(d) - h[i]*h[i+d]*I2(d) + rest(d)
        # = h[i]*[(1-h[i-d])*I1(d) - h[i+d]*I2(d)] + [h[i+d]*I2(d) + rest(d)]
        
        # a_d = (1-h[i-d])*I1(d) - h[i+d]*I2(d)
        # b_d = h[i+d]*I2(d) + rest(d)
        
        # But computing rest(d) for all d is expensive. Instead, let's use:
        # b_d = c(d) - h[i]*a_d  (since c(d) = a_d*h[i] + b_d)
        
        # So: b_d = c(d) - a_d*h[i]
        
        # This means we don't need to compute rest(d) separately!
        
        # Compute a for all d:
        # I1(d) = 1 if 0 <= i-d < n, i.e., d in [i-n+1, i]
        # I2(d) = 1 if 0 <= i+d < n, i.e., d in [-i, n-1-i]
        
        # d ranges from -(n-1) to (n-1)
        # Let's use j = d + (n-1) as the index into corr array, j from 0 to 2n-2
        
        total_d = 2 * n - 1
        a = np.zeros(total_d)
        
        for j in range(total_d):
            d = j - (n - 1)
            I1 = 1 if (0 <= i - d < n) else 0
            I2 = 1 if (0 <= i + d < n) else 0
            if I1:
                a[j] += (1 - h[i - d])
            if I2:
                a[j] -= h[i + d]
        
        # Now b_d = c(d) - a_d * h[i]
        b = corr[:total_d] - a * h[i]
        
        # Find t in [0,1] minimizing max_j (a[j]*t + b[j])
        # Use ternary search since it's convex
        lo, hi = 0.0, 1.0
        for _ in range(30):
            m1 = lo + (hi - lo) / 3
            m2 = hi - (hi - lo) / 3
            f1 = np.max(a * m1 + b)
            f2 = np.max(a * m2 + b)
            if f1 < f2:
                hi = m2
            else:
                lo = m1
        t_opt = (lo + hi) / 2
        
        # Update h[i]
        h[i] = t_opt
    
    # Fix sum
    h = fix_sum(h, target)
    return h

    h = np.array(h, dtype=float).copy()

        h = np.clip(h, 0.0, 1.0)
    return h

def coord_pass(h, focus=None, focus_frac=0.0):
    """One pass of coordinate descent. If focus is given (array of bool),
    only update coordinates where focus is True, plus a fraction of others."""
    n = len(h)
    h = h.copy()

    N = 2 * n - 1
    j = np.arange(N)

    if focus is None:
        order = np.arange(n)
    else:

        focused = np.where(focus)[0]
        others = np.where(~focus)[0]
        np.random.shuffle(others)
        n_extra = max(1, int(focus_frac * n))
        order = np.concatenate([focused, others[:n_extra]])

        d = j - (n - 1)
        I1 = (j >= i) & (j <= i + n - 1)
        I2 = (j >= n - 1 - i) & (j <= 2 * n - 2 - i)
