- h[j+k] is a shifted version of h

def compute_weighted_grad(h, w, dx):
    n = len(h)
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for k_idx in range(2 * n - 1):
        if w[k_idx] < 1e-10:
            continue
        lag = k_idx - (n - 1)
        wk = w[k_idx] * dx
        # Contribution: wk * [(1-h[j-lag]) - h[j+lag]] for valid j
        # (1-h[j-lag]): valid when 0 <= j-lag < n, i.e., lag <= j < n+lag
        # h[j+lag]: valid when 0 <= j+lag < n, i.e., -lag <= j < n-lag
        if lag >= 0:
            # (1-h[j-lag]) valid for j in [lag, n-1]
            grad[lag:] += wk * one_minus_h[:n-lag]
            # h[j+lag] valid for j in [0, n-1-lag]
            grad[:n-lag] -= wk * h[lag:]
        else:
            # lag < 0, let ll = -lag
            ll = -lag
            # (1-h[j+ll]) valid for j in [0, n-1-ll]
            grad[:n-ll] += wk * one_minus_h[ll:]
            # h[j-ll] valid for j in [ll, n-1]
            grad[ll:] -= wk * h[:n-ll]
    return grad

import numpy as np
from scipy.optimize import minimize, Bounds

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_and_fix(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - h.sum()
        if abs(diff) > 1e-10:
            if diff > 0:
                order = np.argsort(1.0 - h)[::-1]
                for i in order:
                    add = min(diff, 1.0 - h[i])
                    h[i] += add
                    diff -= add
                    if diff < 1e-12:
                        break
            else:
                order = np.argsort(h)
                for i in order:
                    rem = min(-diff, h[i])
                    h[i] -= rem
                    diff += rem
                    if diff > -1e-12:
                        break
    return h


def weighted_gradient(h, w, dx):
    n = len(h)
    grad = np.zeros(n)
    omh = 1.0 - h
    for k_idx in range(2 * n - 1):
        if w[k_idx] < 1e-8:
            continue
        lag = k_idx - (n - 1)
        wk = w[k_idx] * dx
        if lag >= 0:
            if n - lag > 0:
                grad[lag:] += wk * omh[:n - lag]
                grad[:n - lag] -= wk * h[lag:]
        else:
            ll = -lag
            if n - ll > 0:
                grad[:n - ll] += wk * omh[ll:]
                grad[ll:] -= wk * h[:n - ll]
    return grad


def make_smooth_obj(h, beta, w=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    if w is None:
        exp_b = np.exp(beta * (corr - m))
        return (m + np.log(np.sum(exp_b)) / beta) * dx

def make_smooth_grad(h, beta, w=None):

    if w is None:
        m = np.max(corr)
        w = np.exp(beta * (corr - m))
        w /= np.sum(w)
    return weighted_gradient(h, w, dx)
