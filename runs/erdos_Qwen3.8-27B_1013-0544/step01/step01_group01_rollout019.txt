            if n_dec == 0:

        idx = np.argsort(corr)[-top_k:]
        corr_t = corr[idx]

        corr_t = corr

        if j1s < j1e:

        j2s = max(0, lag)
        j2e = min(n, n + lag)
        if j2s < j2e:
            grad[j2s:j2e] -= w[ii] * dx * h[j2s - lag:j2e - lag]
    return grad

def _obj(x):
    return smooth_max_obj(x, _g_beta, _g_top_k)

def _grad(x):
    return smooth_max_grad(x, _g_beta, _g_top_k)

def _time_remaining(start_time, budget_s):
    return budget_s - (time.time() - start_time)

def run(seed=42, budget_s=1100, **kwargs):
    global _g_beta, _g_top_k
    np.random.seed(seed)
    start_time = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)

    n = len(h_init)
    target_sum = n / 2.0
    best_h = project_feasible(h_init.copy(), target_sum)
    best_c5 = compute_c5(best_h)
    bounds = [(0.0, 1.0)] * n

    # Phase 1: Systematic L-BFGS-B with increasing sharpness
    for beta in [10, 20, 40, 80, 150, 300]:
        if _time_remaining(start_time, budget_s) < 20:
            break
        for top_k in [None, 25, 12, 6, 4]:
            if _time_remaining(start_time, budget_s) < 10:
                break
            _g_beta = beta
            _g_top_k = top_k
            h_curr = best_h.copy()
            try:
                res = minimize(_obj, h_curr, jac=_grad, method='L-BFGS-B',
                             bounds=bounds,
                             options={'maxiter': 400, 'ftol': 1e-15,
                                      'gtol': 1e-10, 'maxfun': 6000})
                h_new = project_feasible(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h, best_c5 = h_new.copy(), c5
            except Exception:
                pass

    # Phase 2: Focused refinement on top lags with high beta
    for beta in [200, 500, 1000]:
        if _time_remaining(start_time, budget_s) < 10:
            break
        for top_k in [5, 3, 2]:
            if _time_remaining(start_time, budget_s) < 5:
                break
            _g_beta = beta
            _g_top_k = top_k
            try:
                res = minimize(_obj, best_h, jac=_grad, method='L-BFGS-B',
                             bounds=bounds,
                             options={'maxiter': 2000, 'ftol': 1e-15,
                                      'gtol': 1e-11, 'maxfun': 15000})
                h_new = project_feasible(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h, best_c5 = h_new.copy(), c5
            except Exception:
                pass

    # Phase 3: Diverse random restarts with local optimization
    no_improve = 0
    attempt = 0
    while _time_remaining(start_time, budget_s) > 25 and no_improve < 50:
        attempt += 1
        h0 = best_h.copy()
        strategy = attempt % 6

        if strategy == 0:
            scale = 0.04 + 0.12 * np.exp(-attempt / 50.0)
            h0 += np.random.randn(n) * scale
        elif strategy == 1:
            corr = compute_all_corrs(best_h)
            top_idx = np.argsort(corr)[-5:]
            for ki in top_idx:
                lag = int(ki) - (n - 1)
                mask = np.zeros(n, dtype=bool)
                for j in range(n):
                    if 0 <= j - lag < n:
                        mask[j] = True
                    if 0 <= j + lag < n:
                        mask[j] = True
                if mask.any():
                    h0[mask] += np.random.randn(int(mask.sum())) * 0.04
        elif strategy == 2:
            bl = np.random.randint(3, max(4, n // 4))
            b1 = np.random.randint(0, max(1, n - bl))
            h0[b1:b1+bl] = np.random.random(bl) * 0.6 + 0.2
        elif strategy == 3:
            freq = np.random.randint(1, 5)
            phase = np.random.uniform(0, 2.0 * np.pi)
            t = np.arange(n) / n * 2.0 * np.pi
            h0 += 0.04 * np.sin(freq * t + phase)
        elif strategy == 4:
            idx = np.random.choice(n, max(3, n // 8), replace=False)
            h0[idx] = np.clip(h0[idx] + np.random.randn(len(idx)) * 0.08, 0, 1)
        else:
            h0 = best_h[np.random.permutation(n)]

        h0 = project_feasible(h0, target_sum)

        improved_this_attempt = False
        for beta in [60, 150, 400]:
            if _time_remaining(start_time, budget_s) < 8:
                break
            _g_beta = beta
            _g_top_k = 8 if beta < 200 else 4
            try:
                res = minimize(_obj, h0, jac=_grad, method='L-BFGS-B',
                             bounds=bounds,
                             options={'maxiter': 300, 'ftol': 1e-14,
                                      'gtol': 1e-9, 'maxfun': 4000})
                h_new = project_feasible(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h, best_c5 = h_new.copy(), c5
                    h0 = h_new.copy()
                    no_improve = 0
                    improved_this_attempt = True
                else:
                    no_improve += 1
                    break
            except Exception:
                no_improve += 1
                break
        if not improved_this_attempt:
            pass

    # Phase 4: Very high precision final polish
    if _time_remaining(start_time, budget_s) > 8:
        for beta in [800, 1500, 3000]:
            if _time_remaining(start_time, budget_s) < 4:
                break
            _g_beta = beta
            _g_top_k = 2
            try:
                res = minimize(_obj, best_h, jac=_grad, method='L-BFGS-B',
                             bounds=bounds,
                             options={'maxiter': 5000, 'ftol': 1e-16,
                                      'gtol': 1e-12, 'maxfun': 20000})
                h_new = project_feasible(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h, best_c5 = h_new.copy(), c5
            except Exception:
                pass

    # Phase 5: Try interpolation to n=100 if time permits
    if _time_remaining(start_time, budget_s) > 60:
        n2 = 100
        target_sum2 = n2 / 2.0
        x1 = np.linspace(0, 2, n, endpoint=False)
        x2 = np.linspace(0, 2, n2, endpoint=False)
        h_interp = np.interp(x2, x1 + (2.0/n), np.concatenate([best_h, [best_h[0]]]))
        h_interp = project_feasible(h_interp, target_sum2)
        c5_interp = compute_c5(h_interp)

        bounds2 = [(0.0, 1.0)] * n2
        if c5_interp < best_c5 or _time_remaining(start_time, budget_s) > 80:
            for beta in [30, 80, 200]:
                if _time_remaining(start_time, budget_s) < 10:
                    break
                for top_k in [20, 10, 5]:
                    if _time_remaining(start_time, budget_s) < 5:
                        break
                    _g_beta = beta
                    _g_top_k = top_k
                    try:
                        res = minimize(_obj, h_interp, jac=_grad, method='L-BFGS-B',
                                     bounds=bounds2,
                                     options={'maxiter': 500, 'ftol': 1e-14,
                                              'gtol': 1e-10, 'maxfun': 8000})
                        h_new = project_feasible(res.x, target_sum2)
                        c5_new = compute_c5(h_new)
                        if c5_new < best_c5:
                            best_h, best_c5, n = h_new.copy(), c5_new, n2
                            h_interp = h_new.copy()
                    except Exception:
                        pass

    return best_h, best_c5, n
