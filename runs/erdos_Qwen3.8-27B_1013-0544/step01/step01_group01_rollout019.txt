def project_onto_polytope(h, target):
    # Find alpha such that sum(clip(h - alpha, 0, 1)) = target
    # clip(h - alpha, 0, 1) = max(0, min(1, h - alpha))
    # As alpha increases, each coordinate decreases linearly until it hits 0
    # We can use bisection
    
    lo, hi = h.min() - target, h.max()  # rough bounds
    # Actually: sum(clip(h-alpha,0,1)) is decreasing in alpha
    # At alpha = h.min()-1: all coords are >= 1, sum = n
    # At alpha = h.max(): all coords <= 0, sum = 0
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    
    for _ in range(50):
        mid = (lo + hi) / 2
        val = np.clip(h - mid, 0, 1).sum()
        if val > target:
            lo = mid
        else:
            hi = mid
    
    alpha = (lo + hi) / 2
    return np.clip(h - alpha, 0, 1)
```

def compute_lag_gradient(h, lag):
    n = len(h)
    grad = np.zeros(n)
    # Valid i: 0 <= i < n and 0 <= i+lag < n
    if lag >= 0:
        i_range = range(0, n - lag)  # i from 0 to n-lag-1
    else:
        i_range = range(-lag, n)  # i from -lag to n-1
    
    i_vals = np.array(list(i_range))
    j_vals = i_vals + lag
    
    # Term 1: j is first index (i=j), contribution (1-h[j+lag])
    # This is for all j in i_range, contribution to grad[j] is (1-h[j+lag])
    grad[i_vals] += (1 - h[j_vals])
    
    # Term 2: j is second index (i+lag=j), contribution -h[i]
    # j = i+lag for i in i_range, so contribution to grad[j] is -h[i]
    grad[j_vals] -= h[i_vals]
    
    return grad
```

def active_set_gradient(h, epsilon=0.01):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_c = corr.max()
    active = np.where(corr > max_c - epsilon)[0]
    
    grad = np.zeros(n)
    for k in active:
        lag = k - (n - 1)
        # Compute gradient for this lag
        if lag >= 0:
            i_vals = np.arange(0, n - lag)
        else:
            i_vals = np.arange(-lag, n)
        j_vals = i_vals + lag
        grad[i_vals] += (1 - h[j_vals])
        grad[j_vals] -= h[i_vals]
    
    grad /= len(active)
    return grad, active, max_c

                best_c5 = c5_now

        if step % 100 == 50 and time.time() - start < time_limit - 5:
            # Build smooth objective from active set
            def obj(h_in):
                corr = np.correlate(h_in, 1.0 - h_in, mode="full")
                m = corr.max()
                weights = np.exp(5.0 * (corr - m))

import numpy as np
from scipy.optimize import minimize, Bounds
import time

_target_sum = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max(corr, beta):
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta


def project_polytope(h, target):
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def con_fun(x):
    return x.sum() - _target_sum


def con_jac(x):
    return np.ones_like(x)


def objective_smooth(h, beta):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, beta) * (2.0 / len(h))


def objective_periodic(theta, p, n, beta):
    reps = n // p
    h = np.tile(theta, reps)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, beta) * (2.0 / n)


def compute_subgradient(h, epsilon):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_c = corr.max()
    active = np.where(corr > max_c - epsilon)[0]
    grad = np.zeros(n)
    for k in active:
        lag = k - (n - 1)
        if lag >= 0:
            i_vals = np.arange(0, n - lag)
        else:
            i_vals = np.arange(-lag, n)
        j_vals = i_vals + lag
        grad[i_vals] += (1.0 - h[j_vals])
        grad[j_vals] -= h[i_vals]
    grad /= max(1, len(active))
    return grad

            h_new = project_polytope(res.x, n / 2.0)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = c5
                h = h_new.copy()
        except Exception:
            pass
        # Subgradient escape between SLSQP calls
        if bi < len(betas) - 1 and time.time() - start < time_limit - 5:
            grad = compute_subgradient(h, 0.2 * (2.0 / n))
