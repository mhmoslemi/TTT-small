    val = cmax + temp * np.log(np.sum(exps))
    w = sm / temp
    g = np.zeros(n)
    for m in range(2 * n - 1):

        wm = w[m]
        if wm < 1e-18:
            continue
        if k == 0:
            g += wm * (1.0 - 2.0 * h)
        elif k > 0:
            nk = n - k
            g[k:] += wm * (1.0 - h[:nk])
            g[:nk] -= wm * h[k:]
        else:
            l = -k
            nl = n - l
            g[:nl] += wm * (1.0 - h[l:])
            g[l:] -= wm * h[:nl]
    return val, g

def heavy_ball(h0, temp, n_iter, target, momentum=0.9, L_init=None):
    x = h0.copy()
    y = h0.copy()
    L = L_init if L_init else 10.0
    best_c5 = compute_c5(x)
    best_x = x.copy()
    for i in range(n_iter):
        val, g = lse_vg(y, temp)
        step = 1.0 / L
        x_new = project_box_sum(y - step * g, target)
        vn, _ = lse_vg(x_new, temp)
        bt = 0
        while vn > val + np.dot(g, x_new - y) + 0.5 / step * np.sum((x_new - y)**2) and bt < 10:
            L *= 2.0
            step = 1.0 / L
            x_new = project_box_sum(y - step * g, target)
            vn, _ = lse_vg(x_new, temp)
            bt += 1
        y = project_box_sum(x_new + momentum * (x_new - x), target)
        x = x_new
        c5 = compute_c5(x)
        if c5 < best_c5:
            best_c5 = c5
            best_x = x.copy()
    return best_x, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_box_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target) < 1e-14:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def lse_vg(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    exps = np.exp((corr - cmax) / temp)
    sm = exps / np.sum(exps)
    val = cmax + temp * np.log(np.sum(exps))

def t_obj(v):
    return v[-1]

def t_ineq(v):
    n = len(v) - 1
    h = v[:n]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return v[-1] - corr

def t_eq(v):
    return np.sum(v[:-1]) - len(v) * 0.5

def t_eq_jac(v):
    j = np.zeros(len(v))
    j[:-1] = 1.0
    return j

def slsqp_tvar(h0, max_iter=500, ftol=1e-12):
