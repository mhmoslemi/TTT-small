    for k in active_idx:
        w = weights[k]
        lag = int(k - (n - 1))
        if lag >= 0:
            idx1 = np.arange(0, n - lag)
            if len(idx1) > 0:
                grad[idx1] += w * (1.0 - h[idx1 + lag])
            idx2 = np.arange(lag, n)
            if len(idx2) > 0:
                grad[idx2] -= w * h[idx2 - lag]
        else:
            m = -lag
            idx1 = np.arange(m, n)
            if len(idx1) > 0:
                grad[idx1] += w * (1.0 - h[idx1 - m])
            idx2 = np.arange(0, n - m)
            if len(idx2) > 0:
                grad[idx2] -= w * h[idx2 + m]
    return grad * dx, np.max(c) * dx

def nesterov_projected_gradient(h_start, n, dx, budget, rng, alpha=50):
    """Nesterov accelerated projected gradient descent."""
    target = n / 2.0
    best_h = h_start.copy()
    best_c5 = compute_c5(best_h, dx)
    
    h = best_h.copy()
    h_prev = h.copy()
    beta = 1.0
    step = 1.0 / (alpha * dx * np.sqrt(2 * n))  # Initial step size
    
    start = time.time()
    no_improve = 0
    
    while time.time() - start < budget:
        # Nesterov momentum
        beta_new = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        y = h - (beta / beta_new) * (h - h_prev)
        beta = beta_new
        
        # Compute gradient at y
        y_clipped = np.clip(y, 0.0, 1.0)
        grad, _ = c5_smooth_grad(y_clipped, dx, alpha)
        
        # Gradient step
        h_new = y - step * grad
        
        # Project
        h_new = project_feasible(h_new, n)
        
        # Check improvement
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5 - 1e-15:
            best_h = h_new.copy()
            best_c5 = c5_new
            no_improve = 0
        else:
            no_improve += 1
        
        h_prev = h
        h = h_new
        
        # Adaptive step size
        if no_improve > 50:
            step *= 0.9
            no_improve = 0
        elif c5_new < best_c5:
            step = min(step * 1.05, 1.0)
    
    return best_h, best_c5
