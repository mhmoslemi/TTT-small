from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1-h, mode='full')) * dx)

def project_feasible(h, n):
    """Exact projection onto {0<=h<=1, sum(h)=n/2}."""
    target = n / 2.0
    h = np.asarray(h, dtype=float)
    lo, hi = -h.max() - 0.01, 1.0 - h.min() + 0.01
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2, 0.0, 1.0)

def find_optimal_t(B, A):
    """Find t in [0,1] minimizing max_k (B[k] + t*A[k])."""
    # Check boundaries
    max_at_0 = B.max()
    max_at_1 = (B + A).max()
    # At t=0: if all active A >= 0, t*=0
    active_0 = np.where(np.abs(B - max_at_0) < 1e-14)[0]
    if A[active_0].min() >= -1e-14:
        return 0.0
    # At t=1: if all active A <= 0, t*=1
    active_1 = np.where(np.abs(B + A - max_at_1) < 1e-14)[0]
    if A[active_1].max() <= 1e-14:
        return 1.0
    # Bisection
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2
        vals = B + mid * A
        mv = vals.max()
        active = np.where(np.abs(vals - mv) < 1e-12)[0]
        if A[active].min() < 0 and A[active].max() > 0:
            break
        elif A[active].max() > 0:
            hi = mid
        else:
            lo = mid
    return (lo + hi) / 2

def coordinate_descent(h, n, n_sweeps=500):
    h = h.copy()
    for _ in range(n_sweeps):
        for i in range(n):
            old = h[i]
            h[i] = 0.0
            c0 = np.correlate(h, 1-h, mode='full')
            h[i] = 1.0
            c1 = np.correlate(h, 1-h, mode='full')
            h[i] = old
            B = c0
            A = c1 - c0
            t_opt = find_optimal_t(B, A)
            h[i] = t_opt
        h = project_feasible(h, n)
    return h

def subgradient_step(h, n, lr):
    c = np.correlate(h, 1-h, mode='full')
    k_star = np.argmax(c)
    shift = k_star - (n - 1)
    g = np.zeros(n)
    for i in range(n):
        grad = 0.0
        if 0 <= i + shift < n:
            grad += (1.0 - h[i + shift])
        if 0 <= i - shift < n:
            grad -= h[i - shift]
        g[i] = grad
    h_new = project_feasible(h - lr * g, n)
    return h_new

def slsqp_polish(h, n, maxiter=2000):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1-x, mode='full') * dx
        # Smooth max
        alpha = 100.0
        m = c.max()
        return float(m + np.log(np.sum(np.exp((c - m) * alpha))) / alpha)
    res = minimize(obj, h, method='SLSQP',
                   bounds=[(0,1)]*n,
                   constraints=[{'type':'eq', 'fun': lambda x: x.sum()-n/2, 'jac': lambda x: np.ones(n)}],
                   options={'maxiter': maxiter, 'ftol': 1e-16})
    return project_feasible(res.x, n)
