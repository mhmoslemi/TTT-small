def compute_gradient(h, dx, t):
    n = len(h)
    corr_raw = np.correlate(h, 1 - h, mode="full")
    corr = corr_raw * dx
    max_c = np.max(corr)
    shifted = t * (corr - max_c)
    shifted = np.clip(shifted, -400, 400)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    obj = max_c + (1.0/t) * np.log(sum_exp)
    weights = exp_vals / sum_exp
    
    grad = np.zeros(n)
    center = n - 1
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-14:
            continue
        s = k - center
        # d(c_k)/d(h[j]) = (1-h[j-s])[0<=j-s<n] - h[j+s][0<=j+s<n]
        # First part: (1-h[j-s]) for j in [s, s+n)
        lo = max(0, s)
        hi = min(n, s + n)
        if lo < hi:
            grad[lo:hi] += w * (1.0 - h[lo - s:hi - s])
        # Second part: -h[j+s] for j in [-s, n-s)
        lo2 = max(0, -s)
        hi2 = min(n, n - s)
        if lo2 < hi2:
            grad[lo2:hi2] -= w * h[lo2 + s:hi2 + s]
    
    grad *= dx
    return obj, grad

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_polytope(y, target_sum):
    n = len(y)
    lo, hi = -5.0, 5.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(y - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def soft_max_grad(h, dx, t):
    n = len(h)
    corr_raw = np.correlate(h, 1 - h, mode="full")
    corr = corr_raw * dx
    max_c = np.max(corr)
    shifted = t * (corr - max_c)
    shifted = np.clip(shifted, -400, 400)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    obj = max_c + (1.0 / t) * np.log(sum_exp)
    weights = exp_vals / sum_exp
    grad = np.zeros(n)
    center = n - 1
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-14:
            continue
        s = k - center
        lo = max(0, s)
        hi = min(n, s + n)
        if lo < hi:
            grad[lo:hi] += w * (1.0 - h[lo - s:hi - s])
        lo2 = max(0, -s)
        hi2 = min(n, n - s)
        if lo2 < hi2:
            grad[lo2:hi2] -= w * h[lo2 + s:hi2 + s]
    grad *= dx
    return obj, grad

def pgd_optimize(h_init, dx, target_sum, t, n_iters, lr_init):

    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    lr = lr_init
    stall = 0
    for it in range(n_iters):
        obj, grad = soft_max_grad(h, dx, t)
        alpha = lr
        obj_curr = obj
        improved = False
        for _ in range(25):
            h_trial = project_polytope(h - alpha * grad, target_sum)
            obj_trial, _ = soft_max_grad(h_trial, dx, t)
            if obj_trial < obj_curr - 1e-12:
                improved = True
                break
            alpha *= 0.5
        if improved:
            h = h_trial
            c5 = compute_c5(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
            lr = min(lr * 1.03, 2.0)
        else:
            lr = max(lr * 0.7, 0.005)
            stall += 1
        if stall > 300:
            break
    return best_h, best_c5

def binary_local_search(h, dx, target_sum, max_rounds=80):
    n = len(h)
    h_bin = (h >= 0.5).astype(float)
    diff = int(round(target_sum - np.sum(h_bin)))
    if diff > 0:
        order = np.argsort(h_bin)
        for i in range(diff):
            h_bin[order[n - 1 - i]] = 1.0
    elif diff < 0:
        order = np.argsort(h_bin)
        for i in range(-diff):
            h_bin[order[i]] = 0.0
    best_c5 = compute_c5(h_bin, dx)
    for rnd in range(max_rounds):
        improved = False
        for i in range(n):
            old = h_bin[i]
            h_bin[i] = 1.0 - old
            c5 = compute_c5(h_bin, dx)
            if c5 < best_c5 - 1e-14:
                best_c5 = c5
                improved = True
            else:
                h_bin[i] = old
        if not improved:
            break
    return h_bin, best_c5

def make_binary_pattern(n, seed_val):
    rng = np.random.RandomState(seed_val)
    h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h
