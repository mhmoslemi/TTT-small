import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val(h, t, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    max_c = np.max(corr)
    shifted = t * (corr - max_c)
    return (max_c + (1.0/t) * np.log(np.sum(np.exp(shifted)))) * dx

    n = len(h)
    max_c = np.max(corr)
    shifted = t * (corr - max_c)
    exp_s = np.exp(shifted)

    # = 1 - sum_m weights[m]*h[j+k_m] - sum_m weights[m]*h[j-k_m]

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val(h, t, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    s = t * (corr - m)
    return (m + np.log(np.sum(np.exp(s))) / t) * dx

def smooth_max_grad(h, t, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    n = len(h)
    m = np.max(corr)
    s = t * (corr - m)
    e = np.exp(s)
    se = np.sum(e)
    w = e / se
    grad = np.ones(n)
    for i in range(len(corr)):
        k = i - (n - 1)
        wi = w[i]
        if wi < 1e-13:
            continue
        ak = abs(k)
        if ak == 0:
            grad -= 2.0 * wi * h
        elif ak < n:
            grad[:n - ak] -= wi * h[ak:]
            grad[ak:] -= wi * h[:n - ak]
    return grad * dx

def project(h, ts):
    lo, hi = -2.0, 2.0
    for _ in range(80):

        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > ts:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def pgd(h, dx, ts, t, max_iter, init_lr=None):

    f_cur = smooth_max_val(h, t, dx)

    step = init_lr
    stale = 0
    for it in range(max_iter):
        g = smooth_max_grad(h, t, dx)
        for _ in range(40):

            ft = smooth_max_val(trial, t, dx)

            stale = 0

            stale += 1

    return best_h, best_c5
