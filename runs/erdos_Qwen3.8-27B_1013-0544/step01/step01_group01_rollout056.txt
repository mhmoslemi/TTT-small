```python
from scipy.optimize import minimize, minimize_scalar, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    # Use log-sum-exp trick
    shifted = beta * (x - m)
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    return m + np.log(np.sum(np.exp(shifted)) + 1e-300) / beta
```

```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_grad(corr, beta):
    """Compute soft max and weights for gradient."""
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    weights = exp_vals / sum_exp
    sm_val = m + np.log(sum_exp) / beta
    return sm_val, weights

def objective_and_grad(h, mu, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    sm_val, weights = smooth_max_grad(corr, beta)
    obj = sm_val * dx + mu * (h.sum() - n/2.0) ** 2
    
    # Gradient
    g = 1.0 - 2.0 * h
    grad_corr = np.correlate(weights, g, mode='valid')
    grad = dx * grad_corr + 2.0 * mu * (h.sum() - n/2.0) * np.ones(n)
    return obj, grad
```

Wait, I need to check: is np.correlate(w, g, mode='valid') giving me the right thing?

np.correlate(a, v, mode='valid') returns the values of the correlation only for indices where the two signals overlap completely. The length of the output is max(M, N) - min(M, N) + 1 where M and N are the lengths of a and v.

If a has length 2n-1 and v has length n, then the 'valid' output has length (2n-1) - n + 1 = n.

np.correlate(a, v, mode='valid')[i] = ∑_j a[i+j] * v[j] for j=0..n-1

So with a = weights (length 2n-1) and v = g = 1-2h (length n):
result[i] = ∑_{j=0}^{n-1} weights[i+j] * g[j]

And I showed that ∂f/∂h[i] = dx * ∑_{j=0}^{n-1} w_{i+j} * (1-2*h[j])

So result[i] = ∂(soft_max)/∂h[i] (without the dx factor).

This looks correct!
