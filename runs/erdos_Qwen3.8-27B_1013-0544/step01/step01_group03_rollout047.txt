        mask = x < 1.0 if diff > 0 else x > 0.0
        cnt = mask.sum()
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def grad_corr_at(h, k):
    """Gradient of corr[k] (index k in full correlation) w.r.t. h."""
    n = len(h)
    g = np.zeros(n)
    m = abs(k - (n - 1))
    if m == 0:
        return 1.0 - 2.0 * h
    if k > n - 1:
        # corr[k] = sum_{i=0}^{n-1-m} h[i] * (1-h[i+m])
        # d/dh[j] = (1-h[j+m]) for j <= n-1-m, minus h[j-m] for j >= m
        for j in range(n - m):
            g[j] += (1.0 - h[j + m])
        for j in range(m, n):
            g[j] -= h[j - m]
    else:
        # corr[k] = sum_{i=0}^{n-1-m} (1-h[i]) * h[i+m]
        # d/dh[j] = (1-h[j-m]) for j >= m, minus h[j+m] for j <= n-1-m
        for j in range(m, n):
            g[j] += (1.0 - h[j - m])
        for j in range(n - m):
            g[j] -= h[j + m]
    return g

def nesterov_smooth_max(h0, n, tau, t0, time_limit, alpha=0.01, max_iter=20000):
    """Nesterov accelerated projected gradient for smooth max."""
    target = n / 2.0
    x = h0.copy()
    x_prev = h0.copy()
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    t = 1.0
    
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + (t - 1.0) / t_new * (x - x_prev)
        y = np.clip(y, 0.0, 1.0)  # Rough clip for gradient computation
        
        # Compute gradient of smooth max at y
        corr = np.correlate(y, 1.0 - y, mode="full")
        x_tau = tau * corr
        x_max = np.max(x_tau)
        weights = np.exp(x_tau - x_max)
        w_sum = weights.sum()
        weights /= w_sum
        
        # Only sum gradients for lags with significant weight
        # Use top lags for efficiency
        top_k = np.argsort(weights)[-min(5, len(weights)):]
        grad = np.zeros(n)
        for k in top_k:
            grad += weights[k] * grad_corr_at(y, k)
        grad *= (1.0 / tau)  # Chain rule for smooth max
        
        # Actually, the gradient of smooth_max(corr, tau) = (1/tau) * sum_k softmax_k * corr_k
        # d/dh = (1/tau) * sum_k softmax_k * d(corr_k)/dh + (1/tau) * sum_k (d softmax_k / dh) * corr_k
        # For simplicity, just use the first term (ignore the softmax gradient)
        # This is an approximation but works in practice
        
        x_prev = x
        x = y - alpha * grad
        x = project_feasible(x, n)
        
        c5 = evaluate_c5(x)
        if c5 < best_c5:
            best_c5 = c5
            best_h = x.copy()
        
        t = t_new
    
    return best_h, best_c5

def binary_sa(n, t0, time_limit, seed):
    """Simulated annealing for binary vectors."""
    rng = np.random.RandomState(seed)
    target = n // 2
    h = np.zeros(n)
    perm = rng.permutation(n)
    h[perm[:target]] = 1.0
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    cur_c5 = best_c5
    
    T = 0.08
    T_factor = 0.99995
    
    while T > 0.005:
        if time.time() - t0 > time_limit:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = h[j], h[i]
        c5_new = evaluate_c5(h)
        delta = c5_new - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
        T *= T_factor
    
    return best_h, best_c5

def slsqp_optimize(h0, n, tau, t0, time_limit, maxiter=500):
    """SLSQP with smooth max."""
    target = n / 2.0
    bnds = Bounds(np.zeros(n), np.ones(n))
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        return smooth_max_val(corr, tau)
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bnds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
        x = fix_sum(res.x, target)
        c5 = evaluate_c5(x)
        return x, c5
    except Exception:
        return h0, None
