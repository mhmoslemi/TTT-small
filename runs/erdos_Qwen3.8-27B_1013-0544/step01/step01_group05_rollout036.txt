```python
            idx1 = np.arange(0, n - L)
            grad[idx1] += (1.0 - h[idx1 + L])
            idx2 = np.arange(L, n)
            grad[idx2] -= h[idx2 - L]
        else:
            L2 = -L
            idx1 = np.arange(L2, n)
            grad[idx1] += (1.0 - h[idx1 - L2])
            idx2 = np.arange(0, n - L2)
            grad[idx2] -= h[idx2 + L2]
    grad /= float(len(active_indices))
    return grad

def binary_local_search(h_bin, dx, max_iterations=5000, time_limit=None, start_time=None):
    """Local search on binary solutions"""
    n = len(h_bin)
    h = h_bin.astype(float).copy()
    best_c5 = float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx
    best_max_corr = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
    
    ones = np.where(h_bin == 1)[0]
    zeros = np.where(h_bin == 0)[0]
    
    for it in range(max_iterations):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        
        improved = False
        # Shuffle for diversity
        ones_order = np.random.permutation(ones)
        zeros_order = np.random.permutation(zeros)
        
        for i in ones_order:
            for j in zeros_order:
                if i == j:
                    continue
                # Swap: h[i] 1->0, h[j] 0->1
                h[i] = 0.0
                h[j] = 1.0
                corr = np.max(np.correlate(h, 1.0 - h, mode='full'))
                if corr < best_max_corr - 1e-10:
                    best_max_corr = corr
                    best_c5 = corr * dx
                    # Update ones and zeros sets
                    ones = ones[ones != i]
                    zeros = zeros[zeros != j]
                    ones = np.append(ones, j)
                    zeros = np.append(zeros, i)
                    improved = True
                else:
                    # Revert
                    h[i] = 1.0
                    h[j] = 0.0
        
        if not improved:
            break
    
    return h, best_c5
```

```python
def binary_swap_delta(h, i, j):
    """Compute delta in correlation when swapping h[i]:1->0 and h[j]:0->1"""
    n = len(h)
    full_len = 2 * n - 1
    delta = np.zeros(full_len)
    
    # m ranges from -(n-1) to (n-1), stored as indices 0 to 2n-2
    # m = idx - (n-1)
    
    # Term 1: -(1-h[i-m]) for 0 <= i-m < n, i.e., i-n+1 <= m <= i
    m_min1 = max(-(n-1), i - (n-1))
    m_max1 = min(n-1, i)
    for m in range(m_min1, m_max1 + 1):
        delta[m + (n-1)] -= (1.0 - h[i - m])
    
    # Term 2: +(1-h[j-m]) for 0 <= j-m < n, i.e., j-n+1 <= m <= j
    m_min2 = max(-(n-1), j - (n-1))
    m_max2 = min(n-1, j)
    for m in range(m_min2, m_max2 + 1):
        delta[m + (n-1)] += (1.0 - h[j - m])
    
    # Term 3: +h[i+m] for 0 <= i+m < n, i.e., -i <= m <= n-1-i
    m_min3 = max(-(n-1), -i)
    m_max3 = min(n-1, n-1-i)
    for m in range(m_min3, m_max3 + 1):
        delta[m + (n-1)] += h[i + m]
    
    # Term 4: -h[j+m] for 0 <= j+m < n, i.e., -j <= m <= n-1-j
    m_min4 = max(-(n-1), -j)
    m_max4 = min(n-1, n-1-j)
    for m in range(m_min4, m_max4 + 1):
        delta[m + (n-1)] -= h[j + m]
    
    # Cross terms
    # m=0: -2
    delta[0 + (n-1)] -= 2.0
    # m=i-j: +1
    if -(n-1) <= i-j <= n-1:
        delta[(i-j) + (n-1)] += 1.0
    # m=j-i: +1
    if -(n-1) <= j-i <= n-1:
        delta[(j-i) + (n-1)] += 1.0
    
    return delta
```

For binary h, this simplifies:
- (1-h[i-m]) = 1 if h[i-m]=0, 0 if h[i-m]=1
- h[i+m] = 1 if h[i+m]=1, 0 if h[i+m]=0

So the delta is always an integer! And for binary h:
- Term 1: -1 for each m where h[i-m]=0 (and valid)
- Term 2: +1 for each m where h[j-m]=0 (and valid)
- Term 3: +1 for each m where h[i+m]=1 (and valid)
- Term 4: -1 for each m where h[j+m]=1 (and valid)

This is nice. The delta values are integers.

```python
import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(x, target_sum):
    lo = float(x.min()) - 1.0
    hi = float(x.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = float(np.clip(x - mid, 0.0, 1.0).sum())
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def active_gradient(h, n, active_idx):
    n_half = n - 1
    grad = np.zeros(n)
    for idx in active_idx:
        L = int(idx) - n_half
        if L == 0:
            grad[:] += (1.0 - 2.0 * h)
        elif L > 0:
            i1 = np.arange(0, n - L)
            grad[i1] += (1.0 - h[i1 + L])
            i2 = np.arange(L, n)
            grad[i2] -= h[i2 - L]
        else:
            L2 = -L
            i1 = np.arange(L2, n)
            grad[i1] += (1.0 - h[i1 - L2])
            i2 = np.arange(0, n - L2)
            grad[i2] -= h[i2 + L2]
    grad /= float(len(active_idx))
    return grad

def bin_swap_delta(h, i, j):
    n = len(h)
    nf = 2 * n - 1
    delta = np.zeros(nf)
    nh = n - 1
    for m in range(max(-nh, i - nh), min(nh, i) + 1):
        delta[m + nh] -= (1.0 - h[i - m])
    for m in range(max(-nh, j - nh), min(nh, j) + 1):
        delta[m + nh] += (1.0 - h[j - m])
    for m in range(max(-nh, -i), min(nh, nh - i) + 1):
        delta[m + nh] += h[i + m]
    for m in range(max(-nh, -j), min(nh, nh - j) + 1):
        delta[m + nh] -= h[j + m]
    delta[nh] -= 2.0
    if abs(i - j) <= nh:
        delta[i - j + nh] += 1.0
        delta[j - i + nh] += 1.0
    return delta

def binary_search(h_bin, dx, start_time, t_budget):
    n = len(h_bin)
    h = h_bin.astype(float).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float).copy()
    cur_max = float(np.max(corr))
    best_max = cur_max
    best_h = h.copy()
    ones = list(np.where(h_bin == 1)[0])
    zeros = list(np.where(h_bin == 0)[0])
    while time.time() - start_time < t_budget:
        best_swap = None
        best_nm = cur_max
        for ii in range(len(ones)):
            i = ones[ii]
            for jj in range(len(zeros)):
                j = zeros[jj]
                d = bin_swap_delta(h, i, j)
                nm = float(np.max(corr + d))
                if nm < best_nm - 1e-12:
                    best_nm = nm
                    best_swap = (ii, jj, d)
        if best_swap is None:
            break
        ii, jj, d = best_swap
        i = ones[ii]
        j = zeros[jj]
        h[i] = 0.0
        h[j] = 1.0
        corr += d
        cur_max = best_nm
        ones[ii] = j
        zeros[jj] = i
        if cur_max < best_max:
            best_max = cur_max
            best_h = h.copy()
    return best_h, best_max * dx

def smooth_optimize(h, dx, target_sum, start_time, t_budget):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    h = h.copy()
    ss = 0.004
    mu = 0.6
    mom = np.zeros(n)
    while time.time() - start_time < t_budget:
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = float(np.max(corr))
        tol = max(0.02, c_max * 0.001)
        act = np.where(corr >= c_max - tol)[0]
        if len(act) < 2:
            act = np.argsort(corr)[-3:]
        elif len(act) > 12:
            act = np.argsort(corr)[-12:]
        g = active_gradient(h, n, act)
        mom = mu * mom - ss * g
        h_new = project_feasible(h + mom, target_sum)
        c5n = compute_c5(h_new, dx)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new.copy()
        else:
            ss *= 0.9
            if ss < 1e-8:
                break
        h = h_new
    return best_h, best_c5

def slsqp_refine(h, dx, target_sum, start_time, t_budget):
    n = len(h)
    def obj(x):
        return compute_c5(x, dx)
    def eq(x):
        return np.sum(x) - target_sum
    try:
```

```python
    all_m = all_m[mask]
    signs = signs[mask]
    
    delta = np.zeros(2*n-1)
    np.add.at(delta, all_m + nh, signs)
    
    # Cross terms
    delta[nh] -= 2.0
    if abs(i - j) <= nh:
        delta[i - j + nh] += 1.0
        delta[j - i + nh] += 1.0
    
    return delta
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_g_dx = 0.0
_g_target = 0.0

def project_feasible(x, target_sum):
    lo = float(x.min()) - 1.0
    hi = float(x.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = float(np.clip(x - mid, 0.0, 1.0).sum())
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def _obj(x):
    return compute_c5(x, _g_dx)

def _eq(x):
    return np.sum(x) - _g_target

def active_gradient(h, n, active_idx):
    nh = n - 1
    grad = np.zeros(n)
    for idx in active_idx:
        L = int(idx) - nh
        if L == 0:
            grad[:] += (1.0 - 2.0 * h)
        elif L > 0:
            i1 = np.arange(0, n - L)
            grad[i1] += (1.0 - h[i1 + L])
            i2 = np.arange(L, n)
            grad[i2] -= h[i2 - L]
        else:
            L2 = -L
            i1 = np.arange(L2, n)
            grad[i1] += (1.0 - h[i1 - L2])
            i2 = np.arange(0, n - L2)
            grad[i2] -= h[i2 + L2]
    grad /= float(len(active_idx))
    return grad

def bin_delta(h, i, j):
    n = len(h)
    nh = n - 1
    delta = np.zeros(2 * n - 1)
    for m in range(max(-nh, i - nh), min(nh, i) + 1):
        if h[i - m] == 0.0:
            delta[m + nh] -= 1.0
    for m in range(max(-nh, j - nh), min(nh, j) + 1):
        if h[j - m] == 0.0:
            delta[m + nh] += 1.0
    for m in range(max(-nh, -i), min(nh, nh - i) + 1):
        if h[i + m] == 1.0:
            delta[m + nh] += 1.0
    for m in range(max(-nh, -j), min(nh, nh - j) + 1):
        if h[j + m] == 1.0:
            delta[m + nh] -= 1.0
    delta[nh] -= 2.0
    if abs(i - j) <= nh:
        delta[i - j + nh] += 1.0
        delta[j - i + nh] += 1.0
    return delta

def binary_search(h_bin, dx, start_time, t_budget):
    n = len(h_bin)
    h = h_bin.astype(float).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float).copy()
    cur_max = float(np.max(corr))
    best_max = cur_max
    best_h = h.copy()
    ones = list(np.where(h_bin == 1)[0])
    zeros = list(np.where(h_bin == 0)[0])
    while time.time() - start_time < t_budget:
        best_swap = None
        best_nm = cur_max
        for ii in range(len(ones)):
            i = ones[ii]
            for jj in range(len(zeros)):
                j = zeros[jj]
                d = bin_delta(h, i, j)
                nm = float(np.max(corr + d))
                if nm < best_nm - 1e-12:
                    best_nm = nm
                    best_swap = (ii, jj, d)
        if best_swap is None:
            break
        ii, jj, d = best_swap
        i = ones[ii]
        j = zeros[jj]
        h[i] = 0.0
        h[j] = 1.0
        corr = corr + d
        cur_max = best_nm
        ones[ii] = j
        zeros[jj] = i
        if cur_max < best_max:
            best_max = cur_max
            best_h = h.copy()
    return best_h, best_max * dx

def smooth_opt(h, dx, tsum, st, tb):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    h = h.copy()
    ss = 0.005
    mu = 0.55
    mom = np.zeros(n)
    while time.time() - st < tb:
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = float(np.max(corr))
        tol = max(0.015, c_max * 0.0008)
        act = np.where(corr >= c_max - tol)[0]
        if len(act) < 2:
            act = np.argsort(corr)[-3:]
        elif len(act) > 10:
            act = np.argsort(corr)[-10:]
        g = active_gradient(h, n, act)
        mom = mu * mom - ss * g
        h_new = project_feasible(h + mom, tsum)
        c5n = compute_c5(h_new, dx)
        if c5n < best_c5:
            best_c5 = c5n
            best_h = h_new.copy()
        else:
            ss *= 0.88
            if ss < 1e-9:
                break
        h = h_new
    return best_h, best_c5

def slsqp_polish(h, dx, tsum, st, tb):
    global _g_dx, _g_target
    _g_dx = dx
    _g_target = tsum
    n = len(h)
    try:

                       bounds=[(0.0, 1.0)] * n,

            c5 = compute_c5(h_ref, dx)

    except Exception:

    return h, compute_c5(h, dx)

def quantize_to_binary(h, n):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    half = n // 2
    if n_ones > half:
        idx_one = np.where(h_bin == 1)[0]
        order = np.argsort(h[idx_one])
        for k in range(n_ones - half):
            h_bin[idx_one[order[k]]] = 0.0
    elif n_ones < half:
        idx_zero = np.where(h_bin == 0)[0]
        order = np.argsort(h[idx_zero])[::-1]
        for k in range(half - n_ones):
            h_bin[idx_zero[order[k]]] = 1.0
    return h_bin

def run(seed=42, budget_s=1100, **kwargs):
    global _g_dx, _g_target
    start = time.time()
    rng = np.random.RandomState(seed)
    n = 70
    dx = 2.0 / n
    tsum = n / 2.0
    _g_dx = dx
    _g_target = tsum

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except NameError:
        h0 = np.full(n, 0.5)
    h0 = project_feasible(h0, tsum)
    best_h = h0.copy()
    best_c5 = compute_c5(h0, dx)

    def tl():
        return start + budget_s - 8.0 - time.time()

    # Phase 1: Smooth optimization
    if tl() > 180:
        hs, cs = smooth_opt(h0, dx, tsum, start, min(tl(), 160))
        if cs < best_c5:
            best_c5 = cs
            best_h = hs

    # Phase 2: SLSQP
    if tl() > 80:
        hs, cs = slsqp_polish(best_h, dx, tsum, start, min(tl(), 70))
        if cs < best_c5:
            best_c5 = cs
            best_h = hs

    # Phase 3: Binary local search
    if tl() > 120:
        hb = quantize_to_binary(best_h, n)
        hb2, cb2 = binary_search(hb, dx, start, min(tl(), 100))
        if cb2 < best_c5:
            best_c5 = cb2
            best_h = hb2

    # Phase 4: Smooth from binary
    if tl() > 80:
        hs, cs = smooth_opt(best_h, dx, tsum, start, min(tl(), 70))
        if cs < best_c5:
            best_c5 = cs
            best_h = hs

    # Phase 5: SLSQP again
    if tl() > 40:
        hs, cs = slsqp_polish(best_h, dx, tsum, start, min(tl(), 35))
        if cs < best_c5:
            best_c5 = cs
            best_h = hs

    # Phase 6: Binary search from refined
    if tl() > 80:
        hb = quantize_to_binary(best_h, n)
        hb2, cb2 = binary_search(hb, dx, start, min(tl(), 70))
        if cb2 < best_c5:
            best_c5 = cb2
            best_h = hb2

    # Phase 7: Smooth from new binary
    if tl() > 40:
        hs, cs = smooth_opt(best_h, dx, tsum, start, min(tl(), 35))
        if cs < best_c5:
            best_c5 = cs
            best_h = hs

    # Phase 8: Multi-start random binary
    if tl() > 60:
        for _ in range(20):
            if tl() < 3:
                break
            hr = np.where(rng.rand(n) < 0.5, 1.0, 0.0)
            n1 = int(np.sum(hr))
            if n1 > n // 2:
                idx = np.where(hr == 1)[0]
                rng.shuffle(idx)
                for k in range(n1 - n // 2):
                    hr[idx[k]] = 0.0
            elif n1 < n // 2:
                idx = np.where(hr == 0)[0]
                rng.shuffle(idx)
                for k in range(n // 2 - n1):
                    hr[idx[k]] = 1.0
            hb, cb = binary_search(hr, dx, start, min(tl(), 2))
            if cb < best_c5:
                best_c5 = cb
                best_h = hb

    # Phase 9: Fine smooth from absolute best
    if tl() > 30:
        hs, cs = smooth_opt(best_h, dx, tsum, start, min(tl(), 25))
        if cs < best_c5:
            best_c5 = cs
            best_h = hs

    # Phase 10: Final SLSQP
    if tl() > 15:
        hs, cs = slsqp_polish(best_h, dx, tsum, start, min(tl(), 12))
        if cs < best_c5:
            best_c5 = cs
            best_h = hs

    # Final binary check
    if tl() > 10:
        hb = quantize_to_binary(best_h, n)
        hb2, cb2 = binary_search(hb, dx, start, min(tl(), 8))
        if cb2 < best_c5:
            best_c5 = cb2
            best_h = hb2

    return (best_h, best_c5, n)
```
