import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(x, target_sum):
    lo = float(x.min()) - 1.0
    hi = float(x.max()) + 1.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        s = float(np.clip(x - mid, 0.0, 1.0).sum())
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def grad_corr_at_lag(h, L):
    n = len(h)
    grad = np.zeros(n)
    if L == 0:
        grad[:] = 1.0 - 2.0 * h
    elif L > 0:
        grad[L:] += (1.0 - h[:n - L])
        grad[:n - L] -= h[L:]
    else:
        L2 = -L
        grad[:n - L2] += (1.0 - h[L2:])
        grad[L2:] -= h[:n - L2]
    return grad

def _sm_obj(x, dx, temp, target_sum, pw):
    corr = np.correlate(x, 1.0 - x, mode='full')
    scaled = corr * dx
    m = np.max(scaled)
    obj = m + np.log(np.sum(np.exp(scaled - m))) / temp
    obj += pw * (np.sum(x) - target_sum) ** 2
    return obj

def _sm_grad(x, dx, temp, target_sum, pw):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    scaled = corr * dx
    m = np.max(scaled)
    exp_v = np.exp(scaled - m)
    s_exp = np.sum(exp_v)
    weights = exp_v / (s_exp * temp)
    n_half = n - 1
    grad = np.zeros(n)
    for idx in range(len(weights)):
        w = weights[idx]
        if w < 1e-13:
            continue
        L = idx - n_half
        grad += w * grad_corr_at_lag(x, L)
    grad += 2.0 * pw * (np.sum(x) - target_sum)
    return grad

def _active_grad_opt(h, dx, target_sum, n_iter, t_end, alpha0=0.01):
    n = len(h)
    n_half = n - 1
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    h = h.copy()
    alpha = alpha0
    mu = 0.3
    momentum = np.zeros(n)
    stall = 0
    for _ in range(n_iter):
        if time.time() > t_end:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = float(np.max(corr))
        tol = max(1e-5, c_max * 0.001)
        active = np.where(corr >= c_max - tol)[0]
        if len(active) > 12:
            active = np.argsort(corr)[-12:]
        elif len(active) < 2:
            active = np.argsort(corr)[-2:]
        grad = np.zeros(n)
        for idx in active:
            L = int(idx) - n_half
            grad += grad_corr_at_lag(h, L)
        grad /= float(len(active))
        momentum = mu * momentum - alpha * grad
        h_new = project_feasible(h + momentum, target_sum)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1
            if stall % 40 == 0:
                alpha *= 0.85
            if alpha < 1e-9:
                h_new = project_feasible(h + np.random.normal(0, 0.03, n), target_sum)
                c5_new = compute_c5(h_new, dx)
                alpha = alpha0 * 0.5
                momentum = np.zeros(n)
                stall = 0
        h = h_new
    return best_h, best_c5
