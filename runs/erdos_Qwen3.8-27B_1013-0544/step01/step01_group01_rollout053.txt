import numpy as np
from scipy.optimize import minimize, Bounds, minimize_scalar
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta


def obj_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, beta) * dx


def obj_1d(t, h, i, dx, n):
    h[i] = t
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def con_eq(h, target_sum):
    return np.sum(h) - target_sum


def con_eq_jac(h, n):
    return np.ones(n)


def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h


def coord_descent(h, target_sum, dx, n, max_rounds=30):
    for _ in range(max_rounds):
        improved = False
        for i in range(n):
            h_old = h[i]
            c_old = compute_c5(h)
            res = minimize_scalar(obj_1d, args=(h, i, dx, n),
                                  bounds=(0.0, 1.0), method='bounded',
                                  options={'maxiter': 200, 'xatol': 1e-10})
            if res.fun < c_old - 1e-12:
                h[i] = res.x
                improved = True
            else:
                h[i] = h_old
        h = project_h(h, target_sum)
        if not improved:
            break
    return h


def bottleneck_perturb(h, n, strength=0.1):
    h_new = h.copy()
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_star = int(np.argmax(corr)) - (n - 1)
    if 0 < k_star < n:
        for i in range(n - k_star):
            contrib = h_new[i] * (1.0 - h_new[i + k_star])
            if contrib > 0.05:
                h_new[i] *= (1.0 - strength)
                h_new[i + k_star] = min(1.0, h_new[i + k_star] + strength * h_new[i])
    elif -n < k_star < 0:
        kk = -k_star
        for i in range(kk, n):
            contrib = h_new[i] * (1.0 - h_new[i - kk])
            if contrib > 0.05:
                h_new[i] *= (1.0 - strength)
                h_new[i - kk] = min(1.0, h_new[i - kk] + strength * h_new[i])
    return h_new

    start_time = time.time()

    best_h = None
    best_c5 = 1.0
    best_n = None

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_ref = np.array(initial_h_values, dtype=float)
        n_ref = len(h_ref)
    else:
        h_ref = None
        n_ref = 70

    n = n_ref
    dx = 2.0 / n
    target_sum = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))

    h0 = project_h(h_ref.copy(), target_sum) if h_ref is not None else np.full(n, 0.5)
    best_h, best_c5, best_n = h0.copy(), compute_c5(h0), n

        except Exception:
            pass

    # Coordinate descent polish

        h_cd = coord_descent(h_cd, target_sum, dx, n, max_rounds=30)

            except Exception:
                pass

        dx_f = 2.0 / n_f
        target_sum_f = n_f / 2.0

            h_f = project_h(res_f.x, target_sum_f)

        if time.time() - start_time < budget_s - 5:

            h_f2 = coord_descent(h_f2, target_sum_f, dx_f, n_f, max_rounds=40)

            if c5_f2 < best_c5:

        best_c5 = compute_c5(best_h)
        best_n = 70
