            best_h, best_c5 = h_opt.copy(), c5_opt
    
    # Phase 5: Local search
    remaining = budget_s - (time.time() - t0)
    if remaining > 30:
        d5 = t0 + budget_s - 25
        best_h, best_c5 = local_search(best_h, d5, rng)
    
    # Phase 6: Scipy polish
    remaining = budget_s - (time.time() - t0)
    if remaining > 30:
        d6 = t0 + budget_s - 5
        h_sp, c5_sp = scipy_polish(best_h, d6, alpha=400)
        if c5_sp < best_c5:
            best_h, best_c5 = h_sp, c5_sp
        # Another round of local search after scipy
        if time.time() - t0 < budget_s - 15:
            best_h, best_c5 = local_search(best_h, t0 + budget_s - 5, rng)
    
    best_h = project_simplex_box(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, len(best_h)
