Let me also consider: what if I use a **second-order method**? The Hessian of the smooth-max is:
H = α * Cov(f_k) + ∑ w_k * H_k

where Cov is the covariance matrix of the f_k values (weighted by w_k) and H_k is the Hessian of f_k.

The Hessian H_k of f_k = ∑_i h[i](1-h[i+k]) is:
H_k[i,j] = -1 if j = i+k (for valid i) or i = j+k (for valid j)

So H_k is a sparse matrix with -1 on the k-th and -k-th diagonals. This could be useful for a quasi-Newton method.

But implementing a full quasi-Newton might be too complex. Let me stick with first-order methods but make them more effective.

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def compute_corr(h):
    return np.correlate(h, 1 - h, mode='full')

def grad_full(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= s
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx in range(len(corr)):
        if w[idx] < 1e-12:
            continue
        k = idx - (n - 1)
        if k >= 0:
            grad[:n-k] += w[idx] * (1.0 - h[k:])
            grad[k:] -= w[idx] * h[:n-k]
        else:
            p = -k
            grad[:n-p] -= w[idx] * h[p:]
            grad[p:] += w[idx] * (1.0 - h[:n-p])
    return grad * dx

def grad_single_lag(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        g[:n-k] += (1.0 - h[k:])
        g[k:] -= h[:n-k]
    else:
        p = -k
        g[:n-p] -= h[p:]
        g[p:] += (1.0 - h[:n-p])
    return g * (2.0 / n)

def project_box_sum(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(65):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def optimize_smooth(h0, target_sum, deadline, rng, alpha_init=50, momentum_init=0.3):
    n = len(h0)
    best = project_box_sum(h0.copy(), target_sum)
    best_c5 = compute_c5(best)
    v = np.zeros(n)
    y = best.copy()
    alpha = alpha_init
    lr = 0.01
    mu = momentum_init
    patience = 0
    stall = 0
    best_alpha = alpha

    while time.time() < deadline:
        grad = grad_full(y, alpha)
        y_new = y - lr * (grad - mu * (grad - grad_full(best, alpha) if np.allclose(y, best) else 0))
        # Nesterov step
        grad_best = grad_full(best, alpha)
        step = grad_best + mu * (grad_best - grad) if not np.allclose(y, best) else grad_best
        y_new = y - lr * step
        y_new = project_box_sum(y_new, target_sum)
        new_c5 = compute_c5(y_new)

        if new_c5 < best_c5 - 1e-15:
            improved = True
            best = y_new
            best_c5 = new_c5
            patience = 0
            lr = min(lr * 1.02, 0.08)
        else:
            improved = False
            patience += 1
            lr *= 0.9

        y = project_box_sum(y - lr * grad_full(best, alpha), target_sum)

        if patience > 50:
            # Restart with perturbation
            pert = 0.02 + 0.01 * stall
            h_pert = best + rng.randn(n) * pert
            h_pert = project_box_sum(h_pert, target_sum)
            pc5 = compute_c5(h_pert)
            if pc5 < best_c5:
                best = h_pert
                best_c5 = pc5
            else:
                # Try structured perturbation: attack max lag
                corr = compute_corr(best)
                k_max = np.argmax(corr) - (n - 1)
                g_attack = grad_single_lag(best, k_max)
                h_atk = project_box_sum(best - 0.01 * g_attack, target_sum)
                ac5 = compute_c5(h_atk)
                if ac5 < best_c5:
                    best = h_atk
                    best_c5 = ac5
            alpha = min(alpha * 1.25, 500.0)
            lr = 0.01
            y = best.copy()
            patience = 0
            stall += 1
            if stall > 12:
                break

        # Slowly increase alpha
        if patience % 20 == 0 and patience > 0:
            alpha = min(alpha * 1.05, 500.0)

    return best, best_c5

def optimize_multi_alpha(h0, target_sum, deadline, rng):
    n = len(h0)
    best = h0.copy()
    best_c5 = compute_c5(best)
    remaining = deadline - time.time()
    if remaining < 5:
        return best, best_c5

    alphas = [30, 60, 120, 200]
    for a in alphas:
        if time.time() > deadline - 5:
            break
        sub_deadline = min(time.time() + remaining * 0.3, deadline - 5)
        h_opt, c5_opt = optimize_smooth(best, target_sum, sub_deadline, rng, alpha_init=a)
        if c5_opt < best_c5:
            best = h_opt
            best_c5 = c5_opt
        remaining = deadline - time.time()

    return best, best_c5

def make_smooth_init(n, freq, phase, rng):
    x = np.linspace(0, 2, n, endpoint=False) + 1.0 / n
    h = 0.5 + 0.3 * np.sin(np.pi * freq * x + phase) + 0.1 * rng.randn(n)
    return project_box_sum(np.clip(h, 0, 1), n / 2.0)

def make_indicator_init(n, rng):
    h = np.zeros(n)
    positions = rng.choice(n, size=n // 2, replace=False)
    h[positions] = 1.0
    return h

def make_alternating_init(n, rng):
    h = np.zeros(n)
    block = max(2, n // 20)
    for i in range(0, n, 2 * block):
        h[i:i+block] = 1.0
    h = h + 0.1 * rng.randn(n)
    return project_box_sum(np.clip(h, 0, 1), n // 2.0)

def interpolate_resample(h_old, n_old, n_new):
    x_old = (np.arange(n_old) + 0.5) / n_old * 2.0
    x_new = (np.arange(n_new) + 0.5) / n_new * 2.0
    h_ext = np.concatenate([h_old, h_old[:1]])
    x_ext = np.concatenate([x_old, [2.0]])
    return np.interp(x_new, x_ext, h_ext)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    # Get initial values if available
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)

    else:
        n_init = 100
        h_init = make_smooth_init(n_init, 2, 0, rng)

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    best_n = n_init

    # Phase 1: Optimize at initial resolution
    deadline1 = t0 + min(budget_s * 0.35, 400.0)
    best_h, best_c5 = optimize_multi_alpha(h_init, best_n / 2.0, deadline1, rng)
    best_n = len(best_h)

    # Phase 2: Try other resolutions
    elapsed = time.time() - t0
    remaining = budget_s - elapsed
    resolutions = [80, 100, 120, 150, 200, 250]
    tried_n = set([best_n])
    for n_try in resolutions:
        if n_try in tried_n:
            continue
        tried_n.add(n_try)
        if remaining < 60:
            break
        # Interpolate from best to new resolution
        h_interp = interpolate_resample(best_h, best_n, n_try)
        h_interp = project_box_sum(h_interp, n_try / 2.0)
        phase_dead = min(time.time() + remaining * 0.3, t0 + budget_s - 20)
        h_opt, c5_opt = optimize_multi_alpha(h_interp, n_try / 2.0, phase_dead, rng)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
            best_n = n_try
        remaining = budget_s - (time.time() - t0)

    # Phase 3: Try diverse initializations at promising resolutions
    remaining = budget_s - (time.time() - t0)
    if remaining > 80:
        for n_try in [best_n, max(60, best_n // 2), min(300, best_n + 50)]:
            if remaining < 40:
                break
            for init_func in [make_smooth_init, make_indicator_init, make_alternating_init]:

                                  rng.uniform(0, 2*np.pi) if init_func == make_smooth_init else 0, rng)
                sub_dead = min(time.time() + remaining * 0.25, t0 + budget_s - 15)
                h_opt, c5_opt = optimize_smooth(h_div, n_try / 2.0, sub_dead, rng, alpha_init=60)
                if c5_opt < best_c5:
                    best_h = h_opt
                    best_c5 = c5_opt
                    best_n = n_try
                remaining = budget_s - (time.time() - t0)

    # Phase 4: Final polish
    remaining = budget_s - (time.time() - t0)
    if remaining > 20:
        h_pol, c5_pol = optimize_smooth(best_h, best_n / 2.0, min(time.time() + remaining * 0.9, t0 + budget_s - 3), rng, alpha_init=200)

            best_c5 = c5_pol

    best_h = project_box_sum(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
