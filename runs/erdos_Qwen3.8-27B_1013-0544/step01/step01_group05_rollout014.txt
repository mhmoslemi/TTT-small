                           options={'maxiter': 2000, 'ftol': 1e-14})
            if res.x is not None:
                h_ref = project_feasible(res.x, target_sum)
                c5_ref = compute_c5(h_ref, dx)
                if c5_ref < best_c5:
                    best_c5 = c5_ref
                    best_h = h_ref.copy()
        except Exception:
            pass

    # Phase 3: Active-set projected gradient for hard max
    if tleft() > 80:
        h = best_h.copy()
        step_size = 0.004
        mu = 0.5
        momentum = np.zeros(n_points)
        stag = 0
        while tleft() > 60:
            corr = np.correlate(h, 1.0 - h, mode='full')
            grad = active_set_grad(h, corr, n_points, top_k=15)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-12:
                break
            momentum = mu * momentum - step_size * grad / max(gnorm, 1e-10)
            h_new = project_feasible(h + momentum, target_sum)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                stag = 0
                step_size = min(step_size * 1.08, 0.01)
            else:
                stag += 1
                if stag % 25 == 0:
                    step_size *= 0.88
                if step_size < 1e-8:
                    h = best_h + rng.normal(0, 0.025, n_points)
                    h = project_feasible(h, target_sum)
                    step_size = 0.003
                    momentum = np.zeros(n_points)
                    stag = 0
            h = h_new

    # Phase 4: Multi-restart exploration
    if tleft() > 50:
        n_restarts = 5
        for r in range(n_restarts):
            if tleft() < 15:
                break
            if r == 0:
                h = best_h + rng.normal(0, 0.04, n_points)
            elif r == 1:
                h = best_h + rng.normal(0, 0.1, n_points)
            elif r == 2:
                # Structured perturbation
                h = best_h.copy()
                mask = rng.random(n_points) < 0.3
                h[mask] = np.clip(h[mask] + rng.normal(0, 0.1, mask.sum()), 0, 1)
            else:
                h = best_h + rng.normal(0, 0.06, n_points)
            h = project_feasible(h, target_sum)
            c5_t = compute_c5(h, dx)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h.copy()

            # Short p-norm refinement from this start
            p = 64
            step_size = 0.008
            mu2 = 0.5
            momentum = np.zeros(n_points)
            for it in range(300):
                if tleft() < 15:
                    break
                grad = pnorm_grad(h, p)
                gnorm = np.linalg.norm(grad)
                if gnorm < 1e-12:
                    break
                momentum = mu2 * momentum - step_size * grad / max(gnorm, 1e-10)
                h_new = project_feasible(h + momentum, target_sum)
                c5_new = compute_c5(h_new, dx)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h = h_new
                if it % 80 == 79:
                    step_size *= 0.8

    # Phase 5: SLSQP on hard max (final polish)
    if tleft() > 40:
        def hard_obj(x):
            return compute_c5(x, dx)
        def hard_eq(x):
            return np.sum(x) - target_sum
        try:
            res = minimize(hard_obj, best_h, method='SLSQP',
                           bounds=[(0.0, 1.0)] * n_points,
                           constraints=[{'type': 'eq', 'fun': hard_eq}],
                           options={'maxiter': 3000, 'ftol': 1e-15})
            if res.x is not None:
                h_ref = project_feasible(res.x, target_sum)
                c5_ref = compute_c5(h_ref, dx)
                if c5_ref < best_c5:
                    best_c5 = c5_ref
                    best_h = h_ref.copy()
        except Exception:
            pass

    # Phase 6: Ultra-fine active-set polish
    if tleft() > 8:
        h = best_h.copy()
        step_size = 0.0008
        mu = 0.35
        momentum = np.zeros(n_points)
        while tleft() > 2:
            corr = np.correlate(h, 1.0 - h, mode='full')
            grad = active_set_grad(h, corr, n_points, top_k=8)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-13:
                break
            momentum = mu * momentum - step_size * grad / max(gnorm, 1e-10)
            h_new = project_feasible(h + momentum, target_sum)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            else:
                step_size *= 0.92
                if step_size < 1e-11:
                    break
            h = h_new

    # Phase 7: Random micro-perturbations
    if tleft() > 5:
        for _ in range(200):
            if tleft() < 1:
                break
            h_p = best_h + rng.normal(0, 0.003, n_points)
            h_p = project_feasible(h_p, target_sum)
            c5_p = compute_c5(h_p, dx)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_p.copy()

    return (best_h, best_c5, n_points)
