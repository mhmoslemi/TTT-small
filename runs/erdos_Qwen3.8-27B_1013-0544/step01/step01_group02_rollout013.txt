                best_h, best_c5 = h_try, c

    # Phase 2: SLSQP polishing with increasing alpha
    alphas = [50, 100, 200, 500, 1000, 3000, 10000]
    for alpha in alphas:
        if time_left() < 20:
            break
        h_try = try_slsqp(best_h, n, alpha, maxiter=3000)
        c = compute_c5(h_try, n)
        if c < best_c5 - 1e-14:
            best_h, best_c5 = h_try, c
        # Follow with more coordinate descent
        if time_left() > 30:
            h_try = coord_descent_simple(best_h, n, 40, rng, max_sweeps=2, tol=1e-13)
            c = compute_c5(h_try, n)
            if c < best_c5 - 1e-14:
                best_h, best_c5 = h_try, c

    # Phase 3: Random restarts with perturbations
    restart_scales = [0.05, 0.1, 0.15, 0.2, 0.08, 0.12, 0.03, 0.25]
    for scale in restart_scales:
        if time_left() < 60:
            break
        h_pert = best_h + rng.normal(0, scale, size=n)
        h_pert = project_feasible(h_pert, n)
        # Coordinate descent on perturbed
        h_try = coord_descent_simple(h_pert, n, 30, rng, max_sweeps=3, tol=1e-13)
        # SLSQP polish
        if time_left() > 30:
            h_try = try_slsqp(h_try, n, 1000, maxiter=2000)
        c = compute_c5(h_try, n)
        if c < best_c5 - 1e-14:
            best_h, best_c5 = h_try, c

    # Phase 4: Binary-pushing coordinate descent
    # Try to push coordinates toward 0 or 1
    if time_left() > 30:
        h_bin = best_h.copy()
        for _ in range(50):
            if time_left() < 10:
                break
            i = rng.randint(n)
            target = 0.0 if h_bin[i] < 0.5 else 1.0
            old_c5 = compute_c5(h_bin, n)
            h_bin[i] = target
            h_bin = project_feasible(h_bin, n)
            c = compute_c5(h_bin, n)
            if c < old_c5 - 1e-14:
                if c < best_c5 - 1e-14:
                    best_h, best_c5 = h_bin, c
            else:
                h_bin[i] = best_h[i]
                h_bin = project_feasible(h_bin, n)

    # Phase 5: Fine coordinate descent with small steps
    if time_left() > 20:
        for _ in range(100):
            if time_left() < 10:
                break
            i = rng.randint(n)
            for delta in [0.001, 0.002, 0.005, 0.01, 0.02, -0.001, -0.002, -0.005, -0.01, -0.02]:
                new_val = best_h[i] + delta
                if 0.0 <= new_val <= 1.0:
                    h_try = best_h.copy()
                    h_try[i] = new_val
                    h_try = project_feasible(h_try, n)
                    c = compute_c5(h_try, n)
                    if c < best_c5 - 1e-14:
                        best_h, best_c5 = h_try, c
                        break

    # Final SLSQP polish
    if time_left() > 10:
        for alpha in [5000, 20000]:
            if time_left() < 5:
                break
            h_try = try_slsqp(best_h, n, alpha, maxiter=5000)
            c = compute_c5(h_try, n)
            if c < best_c5 - 1e-14:
                best_h, best_c5 = h_try, c

    # Final coordinate descent
    if time_left() > 5:
        h_try = coord_descent_simple(best_h, n, 50, rng, max_sweeps=2, tol=1e-14)
        c = compute_c5(h_try, n)
        if c < best_c5 - 1e-14:
            best_h, best_c5 = h_try, c

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
