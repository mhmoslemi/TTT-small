        if abs(new_max - old_max) < 1e-14:
            break
    
    h_final = project_sum(h, n)
    return h_final, compute_c5(h_final, n)

def _solve_1d_pen(A, B, penalty, offset, lo=0.0, hi=1.0, iters=30):
    for _ in range(iters):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(A + m1 * B) + penalty * (m1 + offset) ** 2
        f2 = np.max(A + m2 * B) + penalty * (m2 + offset) ** 2
        if f1 <= f2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def slsqp_polish(h, n, alpha=5000, maxiter=3000):
    dx = 2.0 / n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        if mx > 500:
            return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
        return m + np.log(np.sum(np.exp(shifted))) / alpha
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        h_try = project_sum(res.x, n)
        c = compute_c5(h_try, n)
        return h_try, c
    except:
        return h.copy(), compute_c5(h, n)
