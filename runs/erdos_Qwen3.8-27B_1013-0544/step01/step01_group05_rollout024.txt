    lo = -1.0
    hi = 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = float(np.clip(x - mid, 0.0, 1.0).sum())
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

    corr = np.correlate(x, 1.0 - x, mode='full')

    max_s = float(np.max(scaled))
    shifted = scaled - max_s
    return (max_s + np.log(np.sum(np.exp(shifted)))) / (alpha * dx)
