    elif k > 0:
        g[:n-k] += (1.0 - h[k:])
        g[k:] -= h[:n-k]
    else:

        g[p:] += (1.0 - h[:n-p])
        g[:n-p] -= h[p:]

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    return np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n)

def project_simplex_box(h, target_sum):
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_corr_at_k(h, k):
    n = len(h)
    g = np.zeros(n)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    elif k > 0:
        g[:n-k] -= h[k:]
        g[k:] += (1.0 - h[:n-k])
    else:
        p = -k
        g[:n-p] += (1.0 - h[p:])
        g[p:] -= h[:n-p]
    return g

def smooth_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    m = np.max(corr)
    obj = (m + np.log(np.sum(np.exp(corr - m))) / alpha) * (2.0 / n)
    w = np.exp(alpha * (corr - m))
    s = np.sum(w)
    g = np.zeros(n)
    if s > 1e-300:
        w /= s
        active = np.where(w > 0.001)[0]
    else:
        active = np.array([np.argmax(corr)])
        w = np.zeros_like(corr)
        w[active[0]] = 1.0
    for idx in active:
        g += w[idx] * grad_corr_at_k(h, idx - (n - 1))
    return obj, g * (2.0 / n)

def optimize(h0, target_sum, deadline, rng, alpha_start=60, alpha_end=1000, lr_init=0.02):
    n = len(h0)
    h = project_simplex_box(h0.copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t = 0
    lr = lr_init
    b1, b2, eps = 0.9, 0.999, 1e-8
    patience = 0
    steps = 0
    while time.time() < deadline:
        t += 1
        steps += 1
        alpha = alpha_start * (alpha_end / alpha_start) ** min(1.0, steps / 1500.0)
        _, grad = smooth_obj_grad(h, alpha)
        m_adam = b1 * m_adam + (1 - b1) * grad
        v_adam = b2 * v_adam + (1 - b2) * grad**2
        m_hat = m_adam / (1 - b1**t)
        v_hat = v_adam / (1 - b2**t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_simplex_box(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-13:
            best_h = h_new.copy()
            best_c5 = c5_new
            patience = 0
            if lr < 0.08:
                lr *= 1.003
        else:
            patience += 1
            if patience > 80 and lr > 0.003:
                lr *= 0.95
            if patience > 400:

                h = project_simplex_box(h + scale * rng.randn(n), target_sum)
                m_adam[:] = 0
                v_adam[:] = 0
                t = 0
                lr = lr_init
                patience = 0
    return best_h, best_c5

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    return np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n)

def project_simplex_box(h, target_sum):
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_corr_at_k(h, k):
    n = len(h)
    g = np.zeros(n)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    elif k > 0:
        g[:n-k] -= h[k:]
        g[k:] += (1.0 - h[:n-k])
    else:
        p = -k
        g[:n-p] += (1.0 - h[p:])
        g[p:] -= h[:n-p]
    return g

def smooth_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    m = np.max(corr)
    obj = (m + np.log(np.sum(np.exp(corr - m))) / alpha) * (2.0 / n)
    w = np.exp(alpha * (corr - m))
    s = np.sum(w)
    g = np.zeros(n)
    if s > 1e-300:
        w /= s
        active = np.where(w > 0.001)[0]
    else:
        active = np.array([np.argmax(corr)])
        w = np.zeros_like(corr)
        w[active[0]] = 1.0
    for idx in active:
        g += w[idx] * grad_corr_at_k(h, idx - (n - 1))
    return obj, g * (2.0 / n)

def optimize(h0, target_sum, deadline, rng, alpha_start=60, alpha_end=1000, lr_init=0.02):
    n = len(h0)
    h = project_simplex_box(h0.copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t = 0
    lr = lr_init
    b1, b2, eps = 0.9, 0.999, 1e-8
    patience = 0
    steps = 0
    while time.time() < deadline:
        t += 1
        steps += 1
        alpha = alpha_start * (alpha_end / alpha_start) ** min(1.0, steps / 1200.0)
        _, grad = smooth_obj_grad(h, alpha)
        m_adam = b1 * m_adam + (1.0 - b1) * grad
        v_adam = b2 * v_adam + (1.0 - b2) * grad**2
        m_hat = m_adam / (1.0 - b1**t)
        v_hat = v_adam / (1.0 - b2**t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_simplex_box(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-13:
            best_h = h_new.copy()
            best_c5 = c5_new
            h = h_new
            patience = 0
            if lr < 0.06:
                lr *= 1.004
        else:
            h = h_new
            patience += 1
            if patience > 60 and lr > 0.002:
                lr *= 0.94
            if patience > 350:
                scale = 0.03 * (1.0 + min(patience / 1500.0, 1.0))
                h = project_simplex_box(h + scale * rng.randn(n), target_sum)
                c5_check = compute_c5(h)
                if c5_check < best_c5:
                    best_h = h.copy()
                    best_c5 = c5_check
                m_adam[:] = 0
                v_adam[:] = 0
                t = 0
                lr = lr_init
                patience = 0
    return best_h, best_c5

def pairwise_search(h, target_sum, deadline, rng):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    while time.time() < deadline:
        improved = False
        one_idx = np.where(best_h > 0.6)[0]
        zero_idx = np.where(best_h < 0.4)[0]
        if len(one_idx) == 0 or len(zero_idx) == 0:
            break
        rng.shuffle(one_idx)
        rng.shuffle(zero_idx)
        trials = min(3, len(zero_idx))
        for i in one_idx:
            for j_idx in range(trials):
                j = zero_idx[j_idx % len(zero_idx)]
                old_i, old_j = best_h[i], best_h[j]
                best_h[i] = 0.0
                best_h[j] = 1.0
                best_h = project_simplex_box(best_h, target_sum)
                c5 = compute_c5(best_h)
                if c5 < best_c5 - 1e-13:
                    best_c5 = c5
                    improved = True
                    one_idx = np.where(best_h > 0.6)[0]
                    zero_idx = np.where(best_h < 0.4)[0]
                    if len(zero_idx) == 0:
                        break
                    break
                else:
                    best_h[i] = old_i
                    best_h[j] = old_j
                    best_h = project_simplex_box(best_h, target_sum)
        if not improved:
            break
    return best_h, best_c5

def interpolate(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.clip(0.5 + 0.3 * rng.randn(200), 0, 1)
    n = len(h0)
    target_sum = n / 2.0
    h0 = project_simplex_box(h0, target_sum)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Main Adam optimization at native resolution (40% budget)
    d1 = t0 + budget_s * 0.40
    best_h, best_c5 = optimize(h0, target_sum, d1, rng, 80, 1200, 0.02)

    # Phase 2: Pairwise move local search (15% budget)
    elapsed = time.time() - t0
    if budget_s - elapsed > 80:
        d2 = t0 + budget_s * 0.55
        best_h, best_c5 = pairwise_search(best_h, target_sum, d2, rng)

    # Phase 3: Coarser resolution exploration to escape local minima (20% budget)
    elapsed = time.time() - t0
    if budget_s - elapsed > 100:
        h_coarse = interpolate(best_h, 100)
        h_coarse = project_simplex_box(h_coarse, 50.0)
        d3 = t0 + budget_s * 0.75
        h_copt, c5_copt = optimize(h_coarse, 50.0, d3, rng, 100, 2000, 0.02)
        if c5_copt < best_c5 * 0.999:
            # Interpolate back and refine
            h_back = interpolate(h_copt, n)
            h_back = project_simplex_box(h_back, target_sum)
            remaining = budget_s - (time.time() - t0) - 10
            if remaining > 60:
                h_back, c5_back = optimize(h_back, target_sum, t0 + budget_s - 10, rng, 200, 2000, 0.01)
                if c5_back < best_c5:
                    best_h = h_back
                    best_c5 = c5_back

    # Phase 4: Final polish with high alpha (remaining)
    remaining = budget_s - (time.time() - t0)
    if remaining > 40:
        best_h, best_c5 = optimize(best_h, target_sum, t0 + budget_s - 5, rng, 500, 4000, 0.008)

    # Phase 5: Quick pairwise refinement
    remaining = budget_s - (time.time() - t0)
    if remaining > 20:
        best_h, best_c5 = pairwise_search(best_h, target_sum, t0 + budget_s - 3, rng)

    best_h = project_simplex_box(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
