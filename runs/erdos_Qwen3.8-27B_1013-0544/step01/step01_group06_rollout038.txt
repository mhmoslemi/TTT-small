def project_box_sum(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_smooth_max_c5(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    scaled = np.clip(alpha * (corr - m), -500, 0)
    w = np.exp(scaled)
    wsum = np.sum(w)
    if wsum < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= wsum
    grad = np.zeros(n)
    for s in np.where(w > 1e-10)[0]:
        ws_val = w[s]
        k = s - (n - 1)
        if k >= 0:
            grad[k:] += ws_val * (1.0 - h[:n-k])
            if k < n:
                grad[:n-k] += ws_val * (-h[k:])
        else:
            p = -k
            grad[:n-p] += ws_val * (1.0 - h[p:])
            grad[p:] += ws_val * (-h[:n-p])
    return grad * (2.0 / n)

    n = len(h0)
    x = project_box_sum(h0.copy(), target_sum)
    best_x = x.copy()
    best_c5 = compute_c5(x)
    v = np.zeros(n)
    stall = 0
    
    while time.time() < deadline:
        # Nesterov lookahead
        x_look = x - lr * v
        x_look = project_box_sum(x_look, target_sum)
        g = grad_smooth_max_c5(x_look, alpha)
        v = beta * v + (1 - beta) * g
        x_new = x - lr * v
        x_new = project_box_sum(x_new, target_sum)
        c5 = compute_c5(x_new)
        
        if c5 < best_c5 - 1e-15:
            best_x = x_new.copy()
            best_c5 = c5
            x = x_new.copy()
            stall = 0
            if lr < 0.05:
                lr *= 1.02
        else:
            stall += 1
            lr *= 0.9
            if stall > 100:
                # Perturb and restart momentum
                x = best_x + rng.randn(n) * 0.005
                x = project_box_sum(x, target_sum)
                v = np.zeros(n)
                lr = 0.01
                alpha = min(alpha * 1.5, 500)
                stall = 0
                if alpha > 400:
                    break
    
    return best_x, best_c5
