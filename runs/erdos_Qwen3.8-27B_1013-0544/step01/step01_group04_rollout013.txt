                best_h = h_new
                best_c5 = c5_new

    # Phase 5: Try different resolutions
    for n_try in [n * 2, n // 2 * 2 if n > 40 else n, n + 10, n - 10]:
        if time_left() < 80:
            break
        if n_try < 20 or n_try > 400:
            continue
        dx_try = 2.0 / n_try
        target_try = n_try / 2.0
        # Resize best_h to n_try via interpolation
        if n_try == n:
            h_try = best_h.copy()
        elif n_try > n:
            x_old = np.linspace(0, 2, n, endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_try = np.interp(x_new, x_old, np.concatenate([best_h, [best_h[0]]]))
        else:
            x_old = np.linspace(0, 2, n, endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_try = np.interp(x_new, x_old, np.concatenate([best_h, [best_h[0]]]))
        h_try = project_feasible(h_try, n_try)
        c5_try = compute_c5(h_try, dx_try)

        # Optimize at this resolution
        for temp in [0.05, 0.01, 0.003]:
            if time_left() < 40:
                break
            h_try = lbfgs_optimize(h_try, dx_try, target_try, temp, penalty, maxiter=150)
            c5_try = compute_c5(h_try, dx_try)

        if time_left() > 20:
            h_try = slsqp_polish(h_try, dx_try, target_try, maxiter=200)
            c5_try = compute_c5(h_try, dx_try)

        if c5_try < best_c5:
            best_h = h_try
            best_c5 = c5_try

    # Phase 6: Final polishing
    n_final = len(best_h)
    dx_final = 2.0 / n_final
    target_final = n_final / 2.0
    if time_left() > 20:
        best_h = slsqp_polish(best_h, dx_final, target_final, maxiter=500)
        best_c5 = compute_c5(best_h, dx_final)

    best_h = project_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return (best_h, final_c5, len(best_h))
