from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    for _ in range(50):
        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - target
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            h[free] -= diff / np.sum(free)
        else:
            h -= diff / n
    return np.clip(h, 0.0, 1.0)

def overlap_gradient(h, s, dx, n):
    grad = np.zeros(n)
    if s >= 0:
        grad[:n-s] += (1.0 - h[s:])
        grad[s:] -= h[:n-s]
    else:
        t = -s
        grad[t:] += (1.0 - h[:n-t])
        grad[:n-t] -= h[t:]
    return grad * dx

def subgradient_step(h, dx, n, lr, momentum, v):

    overlaps = np.correlate(h, 1.0 - h, mode='full') * dx
    k_star = np.argmax(overlaps)
    s = k_star - n + 1
    grad = overlap_gradient(h, s, dx, n)
    v = momentum * v + grad
    h_new = h - lr * v
    h_new = project_feasible(h_new, n)
    return h_new, v, overlaps[k_star]

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - target
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(free):
            h[free] -= diff / np.sum(free)
        else:
            h = h - diff / n
    return np.clip(h, 0.0, 1.0)

def overlap_gradient(h, s, dx, n):
    grad = np.zeros(n)
    if s >= 0:
        if s < n:
            grad[:n-s] += (1.0 - h[s:])
            grad[s:] -= h[:n-s]
    else:
        t = -s
        if t < n:
            grad[t:] += (1.0 - h[:n-t])
            grad[:n-t] -= h[t:]
    return grad * dx

def subgradient_descent(h, dx, n, lr0, n_steps, momentum, time_limit):
    """Subgradient descent with momentum on max overlap."""
    h_cur = h.copy()
    v = np.zeros(n)
    best = compute_c5(h_cur, dx)
    start = time.time()
    
    for step in range(n_steps):
        if time.time() - start > time_limit:
            break
        lr = lr0 / (1.0 + 0.01 * step)
        overlaps = np.correlate(h_cur, 1.0 - h_cur, mode='full') * dx
        k_star = np.argmax(overlaps)
        s = k_star - n + 1
        grad = overlap_gradient(h_cur, s, dx, n)
        v = momentum * v + (1.0 - momentum) * grad
        h_cur = h_cur - lr * v
        h_cur = project_feasible(h_cur, n)
        c5_cur = compute_c5(h_cur, dx)

        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(free):
            h[free] -= diff / np.sum(free)
        else:
            h = h - diff / n
    return np.clip(h, 0.0, 1.0)

def overlap_gradient(h, s, dx, n):
    grad = np.zeros(n)
    if s >= 0:
        if s < n:
            grad[:n-s] += (1.0 - h[s:])
            grad[s:] -= h[:n-s]
    else:
        t = -s
        if t < n:
            grad[t:] += (1.0 - h[:n-t])
            grad[:n-t] -= h[t:]
    return grad * dx

def smooth_obj(h, dx, alpha):
    overlaps = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(overlaps)
    return m + np.log(np.sum(np.exp(alpha * (overlaps - m)))) / alpha

def smooth_grad(h, dx, alpha):
    overlaps = np.correlate(h, 1.0 - h, mode='full') * dx
    n = len(h)
    m = np.max(overlaps)
    expv = np.exp(alpha * (overlaps - m))
    w = expv / np.sum(expv)
    grad = np.zeros(n)
    for k in range(len(overlaps)):
        if w[k] < 1e-8:
            continue
        s = k - n + 1
        grad += w[k] * overlap_gradient(h, s, dx, n)
    return grad

def smooth_slsqp(h, dx, n, alpha, max_iter=200):
    target = n / 2.0
    def obj(x):
        return smooth_obj(x, dx, alpha)
    def obj_grad(x):
        return smooth_grad(x, dx, alpha)
    res = minimize(
        obj, h, jac=obj_grad, method='SLSQP',
        bounds=[(0.0, 1.0)] * n,
        constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                      'jac': lambda x: np.ones(n)}],
        options={'maxiter': max_iter, 'ftol': 1e-12}
    )
    return project_feasible(res.x, n)

def slsqp_polish(h, dx, n, max_iter=300):
    target = n / 2.0
    res = minimize(
        lambda x: compute_c5(x, dx),
        h, method='SLSQP',
        bounds=[(0.0, 1.0)] * n,
        constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target}],
        options={'maxiter': max_iter, 'ftol': 1e-12}
    )
    return project_feasible(res.x, n)

def subgradient_descent(h, dx, n, lr0, n_steps, momentum, top_k, time_limit):
    h_cur = h.copy()
    v = np.zeros(n)
    best = compute_c5(h_cur, dx)
    start = time.time()
    for step in range(n_steps):
        if time.time() - start > time_limit:
            break
        lr = lr0 / (1.0 + 0.005 * step)
        overlaps = np.correlate(h_cur, 1.0 - h_cur, mode='full') * dx
        idx = np.argsort(-overlaps)[:top_k]
        grad = np.zeros(n)
        for k_idx in idx:
            s = k_idx - n + 1
            grad += overlap_gradient(h_cur, s, dx, n)
        grad /= len(idx)
        v = momentum * v + (1.0 - momentum) * grad
        h_cur = h_cur - lr * v
        h_cur = project_feasible(h_cur, n)
        c5_cur = compute_c5(h_cur, dx)
        if c5_cur < best:
            best = c5_cur
    return h_cur, best

def binary_local_search(h, n, dx, rng, max_iters=5000, time_limit=30.0):
    order = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[order[:int(n/2)]] = 1.0
    best_c5 = compute_c5(h_bin, dx)
    best_h = h_bin.copy()
    start = time.time()
    for _ in range(max_iters):
        if time.time() - start > time_limit:
            break
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h_bin[i] = 0
        h_bin[j] = 1
        c5_new = compute_c5(h_bin, dx)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_bin.copy()
        else:
            h_bin[i] = 1
            h_bin[j] = 0
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    dx = 2.0 / n
    best_h = project_feasible(h0.copy(), n)
    best_c5 = compute_c5(best_h, dx)

    def elapsed():
        return time.time() - start_time

    # Phase 1: Smooth SLSQP with increasing alpha (helps escape smoothness issues)
    t1 = budget_s * 0.20
    for alpha in [5.0, 10.0, 20.0, 40.0, 80.0]:
        if elapsed() > t1:
            break
        h_new = smooth_slsqp(best_h.copy(), dx, n, alpha, max_iter=150)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 2: Subgradient descent for global exploration
    t2 = budget_s * 0.40
    for lr0, steps, mom, tk in [(0.06, 800, 0.9, 3), (0.04, 1000, 0.95, 5),
                                 (0.08, 600, 0.85, 1), (0.03, 1500, 0.97, 2)]:
        if elapsed() > t2:
            break
        rem = t2 - elapsed()
        if rem < 5:
            break
        h_sg, c5_sg = subgradient_descent(best_h, dx, n, lr0, steps, mom, tk, rem)
        if c5_sg < best_c5:
            best_h = h_sg
            best_c5 = c5_sg
        # Quick polish after subgradient
        if elapsed() < t2:
            h_p = smooth_slsqp(best_h.copy(), dx, n, 40.0, max_iter=80)
            c5_p = compute_c5(h_p, dx)
            if c5_p < best_c5:
                best_h = h_p
                best_c5 = c5_p

    # Phase 3: SLSQP with random restarts
    t3 = budget_s * 0.55
    for i in range(40):
        if elapsed() > t3:
            break
        if i == 0:
            h_start = best_h.copy()
        else:
            scale = 0.06 + 0.04 * np.exp(-0.15 * i)
            h_start = best_h + rng.randn(n) * scale
            h_start = project_feasible(h_start, n)
        h_new = slsqp_polish(h_start, dx, n, max_iter=200)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 4: Binary local search
    t4 = budget_s * 0.68
    for i in range(8):
        if elapsed() > t4:
            break
        if i == 0:
            h_start = best_h
        elif i % 2 == 1:
            h_start = best_h + rng.randn(n) * 0.08
            h_start = project_feasible(h_start, n)
        else:
            # Try a different random starting point for binary
            h_start = np.clip(best_h + rng.randn(n) * 0.2, 0, 1)
            h_start = project_feasible(h_start, n)
        rem = t4 - elapsed()
        if rem < 3:
            break
        h_bin, c5_bin = binary_local_search(h_start, n, dx, rng, max_iters=4000, time_limit=rem * 0.8)
        if c5_bin < best_c5:
            best_h = h_bin.copy()
            best_c5 = c5_bin
            h_p = slsqp_polish(best_h.copy(), dx, n, max_iter=200)
            c5_p = compute_c5(h_p, dx)
            if c5_p < best_c5:
                best_h = h_p
                best_c5 = c5_p

    # Phase 5: Higher resolution (n=180)
    t5 = budget_s * 0.85
    n2 = 2 * n
    dx2 = 2.0 / n2
    h_up = np.repeat(best_h, 2)
    h_up = project_feasible(h_up, n2)
    c5_up = compute_c5(h_up, dx2)
    if c5_up < best_c5:
        best_h = h_up.copy()
        best_c5 = c5_up

    if elapsed() < t5:
        for i in range(15):
            if elapsed() > t5:
                break
            if i == 0:
                h_start = best_h.copy() if len(best_h) == n2 else h_up.copy()
            else:
                h_start = (best_h if len(best_h) == n2 else h_up) + rng.randn(n2) * 0.04
                h_start = project_feasible(h_start, n2)
            h_new = smooth_slsqp(h_start, dx2, n2, 30.0, max_iter=100)
            c5_new = compute_c5(h_new, dx2)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
            if elapsed() < t5:
                h_new2 = slsqp_polish(h_new, dx2, n2, max_iter=150)
                c5_new2 = compute_c5(h_new2, dx2)
                if c5_new2 < best_c5:
                    best_h = h_new2
                    best_c5 = c5_new2

    # Phase 6: Final polish
    n_f = len(best_h)
    dx_f = 2.0 / n_f
    while elapsed() < budget_s - 10:
        h_start = best_h + rng.randn(n_f) * 0.012
        h_start = project_feasible(h_start, n_f)
        h_new = slsqp_polish(h_start, dx_f, n_f, max_iter=300)
        c5_new = compute_c5(h_new, dx_f)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
        if best_c5 < 0.3808:
            break

    return (best_h, best_c5, len(best_h))
