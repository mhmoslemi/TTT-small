import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    """Project h onto {x: sum(x) = target_sum, 0 <= x <= 1}"""
    h = h.copy()
    # Water-filling / bisection
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2, 0, 1)
    return h

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def smooth_max_grad(corr, beta):
    """Compute smooth max and its gradient w.r.t. corr values."""
    m = np.max(corr)
    shifted = corr - m
    exp_vals = np.exp(beta * shifted)
    weights = exp_vals / np.sum(exp_vals)
    return m + np.log(np.sum(exp_vals)) / beta, weights

def smooth_max_obj_grad(h, beta):
    """Compute smooth max objective and its gradient w.r.t. h."""
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    val, weights = smooth_max_grad(corr, beta)
    
    # Gradient of each corr[k] w.r.t. h
    # corr[m] = sum_i h[i] * (1 - h[i - (m - n + 1)])
    # Let lag = m - (n-1), ranges from -(n-1) to (n-1)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-15:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += w * (1 - 2 * h)
        elif lag > 0:
            # corr[k] = sum_{i=lag}^{n-1} h[i] * (1 - h[i-lag])
            # d/dh[j]: (1-h[j-lag]) for j >= lag, -h[j+lag] for j <= n-1-lag
            for j in range(lag, n):
                grad[j] += w * (1 - h[j - lag])
            for j in range(0, n - lag):
                grad[j] += w * (-h[j + lag])
        else:
            m2 = -lag  # positive
            # corr[k] = sum_{i=0}^{n-1-m2} h[i] * (1 - h[i+m2])
            for j in range(0, n - m2):
                grad[j] += w * (1 - h[j + m2])
            for j in range(m2, n):
                grad[j] += w * (-h[j - m2])
    
    return val * dx, grad * dx

def slsqp_optimize(h, beta, max_iter=3000):
    n = len(h)
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        return smooth_max_obj_grad(x, beta)[0]

    def grad_fn(x):
        return smooth_max_obj_grad(x, beta)[1]

    def cons_fun(x):
        return np.sum(x) - target_sum

    def cons_jac(x):
        return np.ones(n)

    constraints = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj, h, jac=grad_fn, method='SLSQP', bounds=bounds,
                       constraints=constraints, options={'maxiter': max_iter, 'ftol': 1e-14})
        return project_h(res.x, target_sum)
    except Exception:
        return project_h(h, target_sum)


def subgradient_refine(h, max_iters=5000):

    dx = 2.0 / n
    target_sum = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = 0.05
    for it in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_star = np.argmax(corr)
        lag = k_star - (n - 1)
        grad = np.zeros(n)
        if lag == 0:
            grad = 1.0 - 2.0 * h
        elif lag > 0:
            grad[lag:] += (1.0 - h[:n - lag])

        else:

        else:
