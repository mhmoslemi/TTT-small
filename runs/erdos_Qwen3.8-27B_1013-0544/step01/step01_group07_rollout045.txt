            c5_t2 = compute_c5(h_t2, dx_t)
            if c5_t2 < best_c5:
                best_h, best_c5, best_n = h_t2, c5_t2, n_try
                n, dx, target_sum = n_try, dx_t, ts_t
    
    # Phase 4: Random restarts with coordinate descent
    for trial in range(40):
        if elapsed() > budget_s - 30:
            break
        scale = 0.08 + 0.02 * trial
        h_r = best_h + np.random.randn(n) * scale
        h_r = project_sum(h_r, target_sum)
        h_r = coord_descent(h_r, n_sweeps=80)
        h_r = project_sum(h_r, target_sum)
        c5_r = compute_c5(h_r, dx)
        if c5_r < best_c5:
            best_h, best_c5 = h_r, c5_r
    
    # Phase 5: Final SLSQP polish with very high t
    if elapsed() < budget_s - 10:
        h_f = slsqp_refine(best_h, dx, target_sum, t=2000, maxiter=3000)
        c5_f = compute_c5(h_f, dx)
        if c5_f < best_c5:
            best_h, best_c5 = h_f, c5_f
    
    best_h = project_sum(np.clip(best_h, 0.0, 1.0), len(best_h) / 2.0)
    best_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return best_h, best_c5, len(best_h)
