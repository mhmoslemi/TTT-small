import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target_sum):
    """Project h to have exact sum = target_sum, staying in [0,1]"""
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        # Distribute among non-boundary elements
        interior = (h > 1e-6) & (h < 1 - 1e-6)
        if np.sum(interior) > 0:
            h[interior] += diff / np.sum(interior)
            h = np.clip(h, 0.0, 1.0)
        else:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(h, t, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    max_c = np.max(corr)
    shifted = t * (corr - max_c)
    return (max_c + (1.0 / t) * np.log(np.sum(np.exp(shifted)))) * dx

def slsqp_optimize(h, dx, target_sum, t=50, maxiter=500):
    n = len(h)
    try:
        res = minimize(
            lambda x: smooth_max_obj(x, t, dx),
            h.copy(),
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            options={'maxiter': maxiter, 'ftol': 1e-12}
        )
        return fix_sum(res.x, target_sum)
    except Exception:
        return h

But if I fix h to be the current iterate and linearize:
h^T C_k h ≈ 2 h₀^T C_k h - h₀^T C_k h₀ (first-order Taylor)

Then the constraint becomes linear in h, and the whole problem is an LP!

So the algorithm would be:
1. Start with h₀
2. Repeat:
   a. Compute active set A = {k : corr[k] > corr_max - ε}
   b. Linearize corr[k] around current h for k ∈ A
   c. Solve LP: min t s.t. t ≥ dx * (linearized corr[k]) for k ∈ A, 0 ≤ h ≤ 1, Σh = n/2
   d. Update h
3. Return best h

This is essentially a cutting-plane / outer approximation method. It might converge to a local optimum but could be effective.

def fix_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        interior = (h > 0.01) & (h < 0.99)
        n_int = np.sum(interior)
        if n_int > 0:
            h[interior] += diff / n_int
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)

def smooth_max_obj(h, t, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    max_c = np.max(corr)
    shifted = t * (corr - max_c)
    return (max_c + (1.0 / t) * np.log(np.sum(np.exp(shifted)))) * dx

def slsqp_optimize(h, dx, target_sum, t=50, maxiter=500):
    n = len(h)
    try:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                 'jac': lambda x: np.ones(n)}]
        res = minimize(
            lambda x: smooth_max_obj(x, t, dx),
            h.copy(),
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': maxiter, 'ftol': 1e-10}
        )
        return fix_sum(res.x, target_sum)
    except Exception:
        return h

    """Project y onto {h: 0≤h[i]≤1, sum(h)=target_sum}"""
    n = len(y)
    # Bisection on lambda
    lo = -1.0 - np.max(y)
    hi = 2.0 - np.min(y)
    for _ in range(60):
        mid = (lo + hi) / 2
        h = np.clip(y + mid, 0.0, 1.0)
        s = np.sum(h)
        if s < target_sum:
            lo = mid
        else:
            hi = mid

    diff = target_sum - np.sum(h)

    """Take projected gradient steps on smooth max objective"""
    n = len(h)
    h_cur = h.copy()
    for step in range(n_steps):
        # Compute smooth max gradient
        corr = np.correlate(h_cur, 1 - h_cur, mode="full")
        max_c = np.max(corr)
        weights = np.exp(t * (corr - max_c))
        weights /= weights.sum()

        grad = np.zeros(n)
        for k in range(len(corr)):

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        interior = (h > 0.01) & (h < 0.99)
        n_int = np.sum(interior)
        if n_int > 0:
            h[interior] += diff / n_int
        else:
            h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(h, t, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    max_c = np.max(corr)
    shifted = t * (corr - max_c)
    return (max_c + (1.0 / t) * np.log(np.sum(np.exp(shifted)))) * dx

def slsqp_optimize(h, dx, target_sum, t=50, maxiter=500):
    n = len(h)
    try:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                 'jac': lambda x: np.ones(n)}]
        res = minimize(
            lambda x: smooth_max_obj(x, t, dx),
            h.copy(),
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': maxiter, 'ftol': 1e-10}
        )
        return fix_sum(res.x, target_sum)
    except Exception:
        return h

def project_onto_set(y, target_sum):
    n = len(y)
    lo = -1.0 - np.max(y)
    hi = 2.0 - np.min(y)

        s = np.sum(h)
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(y + (lo + hi) / 2, 0.0, 1.0)
    diff = target_sum - np.sum(h)

    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    max_c = np.max(corr)
    shifted = t * (corr - max_c)
    weights = np.exp(shifted)
    weights /= weights.sum()
    
    significant = np.where(weights > 1e-3)[0]
    grad = np.zeros(n)
    for k in significant:
        m = k - n + 1
        w = weights[k]
        if m >= 0:
            grad[m:] += w * (1.0 - h[:n - m])
            grad[:n - m] += w * (-h[m:])
        else:
            p = -m
            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] += w * (-h[:n - p])
    grad *= dx
    return grad

def proj_grad_descent(h, dx, target_sum, t=100, n_steps=100, alpha_init=0.01):
    cur = h.copy()
    alpha = alpha_init
    best_f = smooth_max_obj(cur, t, dx)
    for i in range(n_steps):
        grad = compute_smooth_grad(cur, dx, target_sum, t)
        new = project_onto_set(cur - alpha * grad, target_sum)
        f_new = smooth_max_obj(new, t, dx)
        if f_new < best_f:
            cur = new
            best_f = f_new
            if i % 20 == 0 and i > 0:
                alpha = min(alpha * 1.2, 0.1)
        else:
            alpha *= 0.7
    return cur

def perturb_interval(h, target_sum, rng):
    n = len(h)

    corr = np.correlate(h, 1 - h, mode="full")
    max_c = np.max(corr)
    shifted = t * (corr - max_c)
    return (max_c + (1.0 / t) * np.log(np.sum(np.exp(shifted)))) * dx

def obj_func(x):
    return smooth_max_obj(x, _opt_state['t'], _opt_state['dx'])

def constraint_func(x):
    return np.sum(x) - _opt_state['target_sum']

def constraint_jac(x):
    return np.ones(len(x))

def slsqp_optimize(h, dx, target_sum, t=50, maxiter=500):
    n = len(h)
    _opt_state['t'] = t
    _opt_state['dx'] = dx
    _opt_state['target_sum'] = target_sum
    try:
        res = minimize(
            obj_func, h.copy(), method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': constraint_func, 'jac': constraint_jac}],
            options={'maxiter': maxiter, 'ftol': 1e-10}
        )
        return fix_sum(res.x, target_sum)
    except Exception:
        return h

def project_onto_set(y, target_sum):
    lo = -2.0 - np.max(y)
    hi = 3.0 - np.min(y)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(y + mid, 0.0, 1.0)
        if np.sum(h) < target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(y + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        interior = (h > 1e-4) & (h < 1.0 - 1e-4)
        if np.sum(interior) > 0:
            h[interior] += diff / np.sum(interior)
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_smooth_grad(h, dx, t):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    max_c = np.max(corr)
    shifted = t * (corr - max_c)
    weights = np.exp(shifted)
    weights /= weights.sum()
    significant = np.where(weights > 1e-4)[0]
    grad = np.zeros(n)
    for k in significant:
        m = k - n + 1
        w = weights[k]
        if m >= 0:
            grad[m:] += w * (1.0 - h[:n - m])
            if n - m > 0:
                grad[:n - m] += w * (-h[m:])
        else:
            p = -m
            grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] += w * (-h[:n - p])
    grad *= dx
    return grad

def proj_grad_descent(h, dx, target_sum, t=100, n_steps=200, alpha_init=0.01):
    cur = h.copy()
    alpha = alpha_init
    best_f = smooth_max_obj(cur, t, dx)
    for i in range(n_steps):
        grad = compute_smooth_grad(cur, dx, t)
        new = project_onto_set(cur - alpha * grad, target_sum)
        f_new = smooth_max_obj(new, t, dx)
        if f_new < best_f - 1e-15:
            cur = new
            best_f = f_new
            if i % 50 == 0 and i > 0:
                alpha = min(alpha * 1.1, 0.05)
        else:
            alpha *= 0.7
            if alpha < 1e-5:
                alpha = alpha_init * 0.5
    return cur

def perturb_noise(h, target_sum, rng, scale=0.3):
    n = len(h)
    h_new = h.copy()
    k = rng.randint(3, min(20, n))
    idx = rng.choice(n, k, replace=False)
    h_new[idx] += rng.normal(0, scale, k)
    return fix_sum(h_new, target_sum)

def perturb_interval(h, target_sum, rng):
    n = len(h)
    h_new = h.copy()
    length = rng.randint(4, min(30, n // 2))
    a = rng.randint(0, n - length)
    h_new[a:a + length] = 1.0 - h_new[a:a + length]
    return fix_sum(h_new, target_sum)

def perturb_structured(h, target_sum, rng):
    n = len(h)
    h_new = h.copy()
    freq = rng.randint(2, 12)
    phase = rng.uniform(0, 2 * np.pi)
    amp = rng.uniform(0.1, 0.4)
    x = np.arange(n)
    h_new += amp * np.sin(2.0 * np.pi * freq * x / n + phase)
    return fix_sum(h_new, target_sum)

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0, 2.0, old_n)
    x_new = np.linspace(0, 2.0, new_n)
    return np.interp(x_new, x_old, h)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = fix_sum(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)
    best_n = n

    # Phase 1: Projected gradient with annealing
    for t in [5, 10, 20, 50, 100, 200]:
        if time.time() - t0 > budget_s - 700:
            break
        h_new = proj_grad_descent(best_h, dx, target_sum, t=t,
                                   n_steps=300, alpha_init=0.02 / (1.0 + t * 0.005))
        c5 = compute_c5(h_new, dx)
        if c5 < best_c5:
            best_h = h_new
            best_c5 = c5

    # Phase 2: SLSQP polish
    for t in [20, 50, 150, 400]:
        if time.time() - t0 > budget_s - 550:
            break
        h_new = slsqp_optimize(best_h, dx, target_sum, t=t, maxiter=600)
        c5 = compute_c5(h_new, dx)
        if c5 < best_c5:
            best_h = h_new
            best_c5 = c5
        h_pg = proj_grad_descent(best_h, dx, target_sum, t=t * 2,
                                  n_steps=80, alpha_init=0.005)
        c5_pg = compute_c5(h_pg, dx)
        if c5_pg < best_c5:
            best_h = h_pg
            best_c5 = c5_pg

    # Phase 3: Basin hopping with diverse perturbations
    n_hops = 0
    while time.time() - t0 < budget_s - 350 and n_hops < 250:
        n_hops += 1
        choice = rng.randint(0, 4)
        if choice == 0:
            h_trial = perturb_noise(best_h, target_sum, rng,
                                     scale=rng.uniform(0.15, 0.5))
        elif choice == 1:
            h_trial = perturb_interval(best_h, target_sum, rng)
        elif choice == 2:
            h_trial = perturb_structured(best_h, target_sum, rng)
        else:
            h_trial = best_h.copy()
            k = rng.randint(5, 30)
            idx = rng.choice(n, k, replace=False)
            h_trial[idx] = rng.uniform(0, 1, k)
            h_trial = fix_sum(h_trial, target_sum)

        h_trial = proj_grad_descent(h_trial, dx, target_sum, t=80,
                                     n_steps=60, alpha_init=0.01)
        h_new = slsqp_optimize(h_trial, dx, target_sum, t=100, maxiter=300)
        c5 = compute_c5(h_new, dx)
        if c5 < best_c5:
            best_h = h_new
            best_c5 = c5

    # Phase 4: Try n=320
    if time.time() - t0 < budget_s - 150:
        n2 = 2 * n
        dx2 = 2.0 / n2
        ts2 = n2 / 2.0
        h2 = upsample(best_h, n2)
        h2 = fix_sum(h2, ts2)

        h2 = proj_grad_descent(h2, dx2, ts2, t=50, n_steps=150, alpha_init=0.01)
        c5_2 = compute_c5(h2, dx2)

        if time.time() - t0 < budget_s - 100:
            for t in [30, 100, 250]:
                if time.time() - t0 > budget_s - 80:
                    break
                h2_new = slsqp_optimize(h2, dx2, ts2, t=t, maxiter=400)
                c5_test = compute_c5(h2_new, dx2)
                if c5_test < c5_2:
                    h2 = h2_new
                    c5_2 = c5_test
                h2 = proj_grad_descent(h2, dx2, ts2, t=t + 50,
                                        n_steps=50, alpha_init=0.005)
                c5_2 = compute_c5(h2, dx2)

        if c5_2 < best_c5:
            best_h = h2
            best_c5 = c5_2
            best_n = n2

    # Phase 5: Additional basin hopping at best resolution
    if time.time() - t0 < budget_s - 50:
        dd = 2.0 / best_n
        ts = best_n / 2.0
        n_extra = 50
        for _ in range(n_extra):
            if time.time() - t0 > budget_s - 30:
                break
            h_trial = perturb_noise(best_h, ts, rng,
                                     scale=rng.uniform(0.1, 0.3))
            h_trial = proj_grad_descent(h_trial, dd, ts, t=100,
                                         n_steps=40, alpha_init=0.008)
            h_new = slsqp_optimize(h_trial, dd, ts, t=200, maxiter=200)
            c5 = compute_c5(h_new, dd)
            if c5 < best_c5:
                best_h = h_new
                best_c5 = c5

    # Final polish
    if time.time() - t0 < budget_s - 5:
        dd = 2.0 / best_n
        ts = best_n / 2.0
        h_final = slsqp_optimize(best_h, dd, ts, t=500, maxiter=1500)
        c5_final = compute_c5(h_final, dd)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final

    best_h = np.clip(best_h, 0.0, 1.0)
    dx_final = 2.0 / best_n
    best_c5 = compute_c5(best_h, dx_final)

    return best_h, best_c5, best_n
