def compute_c5(h, dx):

    return float(np.max(corr) * dx)

def compute_grad_smooth_max(h, T, dx):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    
    # Weights (softmax over correlations)
    max_corr = np.max(corr)
    shifted = T * (corr - max_corr)
    exp_shifted = np.exp(shifted - np.max(shifted))
    weights = exp_shifted / np.sum(exp_shifted)
    
    # Grad
    grad = np.zeros(n)
    for k in range(len(corr)):
        lag = k - (n - 1)
        w = weights[k] * dx
        if w < 1e-14:
            continue
        if lag == 0:
            grad += w * (1 - 2 * h)
        elif lag > 0:
            grad[:n-lag] += w * (1 - h[lag:])
            grad[lag:] -= w * h[:n-lag]
        else:
            m = -lag
            grad[m:] += w * (1 - h[:n-m])
            grad[:n-m] -= w * h[m:]
    
    return grad, max_corr

def compute_smooth_max_value_and_grad(h, T, dx):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    overlaps = corr * dx
    
    # Softmax weights
    max_ov = np.max(overlaps)
    shifted = T * (overlaps - max_ov)
    shifted -= np.max(shifted)  # for numerical stability
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    
    # Smooth max value
    smooth_val = max_ov + (1.0/T) * np.log(np.sum(exp_shifted))
    
    # Gradient
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-12:

        lag = k - (n - 1)
        if lag == 0:

        elif lag > 0:

        else:
            m = -lag
            grad[m:] += w * dx * (1 - h[:n-m])
            grad[:n-m] -= w * dx * h[m:]
