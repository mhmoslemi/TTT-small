The terms involving h[i] are:

- When j+s = i, i.e., j = i-s: term is h[i-s] * (1 - h[i]) if i-s is valid

def grad_corr_k_vec(h, k):
    n = len(h)
    shift = k - (n - 1)
    g = np.zeros(n)
    lo1 = max(0, -shift)
    hi1 = min(n, n - shift)

    lo2 = max(0, shift)
    hi2 = min(n, n + shift)
    if lo2 < hi2:
        g[lo2:hi2] += (-h[lo2-shift:hi2-shift])

def compute_gradient(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    e = np.exp(beta * (corr - m))
    e_sum = e.sum()
    w = e / e_sum
    grad = np.zeros(n)
    for k in range(len(corr)):
        if w[k] > 1e-12:
            grad += w[k] * grad_corr_k_vec(h, k)
    grad *= dx
    return grad

def project_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    lo, hi = -abs(diff) - 0.5, abs(diff) + 0.5
    for _ in range(200):
        mid = (lo + hi) / 2
        hp = np.clip(h - mid, 0.0, 1.0)
        if hp.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            room = 1.0 - h
            idx = np.argsort(-room)
            for i in idx:
                add = min(diff, room[i])
                h[i] += add
                diff -= add
                if diff < 1e-14:
                    break
        else:
            idx = np.argsort(h)
            for i in idx:
                rem = min(-diff, h[i])
                h[i] -= rem
                diff += rem
                if diff > -1e-14:
                    break
    return h

def gd_step(h, target_sum, beta, lr, deadline):
    """Single gradient descent step with backtracking line search"""
    n = len(h)
    grad = compute_gradient(h, beta)
    grad -= np.mean(grad)
    c5_curr = compute_c5(h)
    for ls in range(50):
        step = lr * (0.6 ** ls)
        if step < 1e-12:
            return h, lr, False
        h_new = project_sum(h - step * grad, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5_curr:
            return h_new, min(lr * 1.15, 20.0), True
    return h, lr * 0.5, False

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def grad_corr_k_vec(h, k):
    n = len(h)
    shift = k - (n - 1)
    g = np.zeros(n)
    lo1 = max(0, shift)
    hi1 = min(n, n + shift)
    if lo1 < hi1:
        g[lo1:hi1] += (1.0 - h[lo1 - shift:hi1 - shift])
    lo2 = max(0, -shift)
    hi2 = min(n, n - shift)
    if lo2 < hi2:
        g[lo2:hi2] += (-h[lo2 + shift:hi2 + shift])
    return g

def compute_gradient(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    e = np.exp(beta * (corr - m))
    e_sum = e.sum()
    w = e / e_sum
    grad = np.zeros(n)
    for k in range(len(corr)):
        if w[k] > 1e-12:
            grad += w[k] * grad_corr_k_vec(h, k)
    grad *= dx
    return grad

def project_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-15:
        return h
    lo, hi = -abs(diff) - 0.5, abs(diff) + 0.5
    for _ in range(200):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        if hp.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            room = 1.0 - h
            idx = np.argsort(-room)
            for i in idx:
                add = min(diff, room[i])
                h[i] += add
                diff -= add
                if diff < 1e-14:
                    break
        else:
            idx = np.argsort(h)
            for i in idx:
                rem = min(-diff, h[i])
                h[i] -= rem
                diff += rem
                if diff > -1e-14:
                    break
    return h

def gd_inner(h, target_sum, beta, max_iter, lr_init, deadline):
    n = len(h)
    h = h.copy()
    lr = lr_init
    best_c5 = compute_c5(h)
    best_h = h.copy()
    stagnant = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        grad = compute_gradient(h, beta)
        grad -= np.mean(grad)
        c5_curr = compute_c5(h)
        improved = False
        for ls in range(50):
            step = lr * (0.65 ** ls)
            if step < 1e-14:
                break
            h_new = project_sum(h - step * grad, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < c5_curr - 1e-16:
                h = h_new
                c5_curr = c5_new
                improved = True
                break
        if improved:
            if c5_curr < best_c5:
                best_c5 = c5_curr
                best_h = h.copy()
                stagnant = 0
            lr = min(lr * 1.12, 50.0)
        else:
            stagnant += 1
            lr *= 0.5
            if lr < 1e-11 or stagnant > 200:
                break
    return best_h, best_c5

def slsqp_obj(h, target_sum, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    e = np.exp(beta * (corr - m))
    return (m + np.log(e.sum()) / beta) * dx

def slsqp_con_fun(x, target_sum):
    return x.sum() - target_sum

def slsqp_con_jac(x):
    return np.ones(len(x))

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_points = len(h_init)
    else:
        n_points = 70
        h_init = np.full(n_points, 0.5)

    target_sum = n_points / 2.0
    best_h = project_sum(h_init, target_sum)
    best_c5 = compute_c5(best_h)
    deadline = start_time + budget_s - 25

    # Phase 1: Coarse-to-fine gradient descent
    beta_schedule = [5, 10, 30, 80, 200]
    for beta in beta_schedule:
        if time.time() > deadline:
            break
        h, c5 = gd_inner(best_h, target_sum, beta, 5000, 2.0 / (beta + 1.0), deadline)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 2: Fine gradient descent
    for beta in [500, 1500, 5000]:
        if time.time() > deadline:
            break
        h, c5 = gd_inner(best_h, target_sum, beta, 8000, 0.005, deadline)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 3: Diverse restarts
    no_improve = 0
    attempt = 0
    while time.time() < deadline and no_improve < 200:
        attempt += 1
        h0 = best_h.copy()
        strat = attempt % 8
        scale = 0.02 + 0.06 * np.exp(-attempt / 80.0)

        if strat == 0:
            h0 += np.random.randn(n_points) * scale
        elif strat == 1:
            blen = np.random.randint(2, max(3, n_points // 5))
            bs = np.random.randint(0, n_points - blen)
            h0[bs:bs + blen] = 1.0 - h0[bs:bs + blen]
        elif strat == 2:
            mask = np.random.random(n_points) < 0.25
            h0[mask] = np.random.random(int(mask.sum()))
        elif strat == 3:
            corr = np.correlate(h0, 1.0 - h0, mode="full")
            for k in np.argsort(corr)[-2:]:
                h0 -= 0.15 * grad_corr_k_vec(h0, int(k))
        elif strat == 4:
            n1 = np.random.randint(2, max(3, n_points // 4))
            s1 = np.random.randint(0, n_points - n1)
            h0[s1:s1 + n1] = np.concatenate([h0[s1:s1 + n1][::-1]])
        elif strat == 5:
            for _ in range(np.random.randint(1, 4)):
                i, j = np.random.choice(n_points, 2, replace=False)
                h0[i], h0[j] = h0[j], h0[i]
        elif strat == 6:
            h0 += np.random.randn(n_points) * 0.12
        else:
            n_p = max(2, n_points // 6)
            idx = np.random.choice(n_points, n_p, replace=False)
            h0[idx] += np.random.randn(n_p) * 0.08

        h0 = project_sum(h0, target_sum)
        h_new, c5_new = gd_inner(h0, target_sum, 100, 800, 0.5, deadline)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new
            no_improve = 0
            for bf in [500, 2000]:
                if time.time() > deadline:
                    break
                hf, cf = gd_inner(best_h, target_sum, bf, 3000, 0.02, deadline)
                if cf < best_c5:
                    best_h, best_c5 = hf, cf
        else:
            no_improve += 1

    # Phase 4: Try different n_points
    for n_try in [50, 100, 140]:
        if time.time() > deadline:
            break
        n_t = n_try
        ts = n_t / 2.0
        x_old = np.linspace(0.0, 2.0, n_points + 1)[:-1]
        x_new = np.linspace(0.0, 2.0, n_t + 1)[:-1]
        h_interp = np.interp(x_new, x_old, best_h)
        h_try = project_sum(h_interp, ts)
        for beta in [50, 300, 1000]:
            if time.time() > deadline:
                break
            h_pol, c5_pol = gd_inner(h_try, ts, beta, 3000, 0.3, deadline)
            h_try = h_pol
            if c5_pol < best_c5:
                best_h, best_c5 = h_pol, c5_pol
                n_points = n_t
                target_sum = ts

    # Phase 5: SLSQP polish
    if time.time() < start_time + budget_s - 15:
        try:
            bounds = Bounds(np.zeros(n_points), np.ones(n_points))
            cons = [{'type': 'eq', 'fun': slsqp_con_fun, 'jac': slsqp_con_jac}]

            def con_wrapped(x):
                return slsqp_con_fun(x, target_sum)

            def jac_wrapped(x):
                return slsqp_con_jac(x)

            cons = [{'type': 'eq', 'fun': con_wrapped, 'jac': jac_wrapped}]
            res = minimize(slsqp_obj, best_h, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 3000, 'ftol': 1e-16})
            h_new = project_sum(res.x, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new
        except Exception:
            pass

    # Phase 6: Ultra-fine final polish
    if time.time() < start_time + budget_s - 5:
        h, c5 = gd_inner(best_h, target_sum, 10000, 20000, 0.0005,
                         start_time + budget_s - 5)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    return best_h, best_c5, n_points
