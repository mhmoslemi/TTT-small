def smooth_max_grad(h, alpha):

    n = len(h)

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    return np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n)

def smooth_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    s = w.sum()
    if s < 1e-300:
        return np.zeros(n)
    w /= s
    grad = np.zeros(n)
    for idx in np.where(w > 1e-6)[0]:
        k = idx - (n - 1)
        if k >= 0:
            grad[k:] += w[idx] * (1.0 - h[:n-k])
            if k > 0:
                grad[:n-k] -= w[idx] * h[k:]
        else:
            p = -k
            grad[:n-p] += w[idx] * (1.0 - h[p:])
            if p > 0:
                grad[p:] -= w[idx] * h[:n-p]
    return grad * (2.0 / n)

def proj(h, t):
    lo, hi = -1.5, 1.5
    for _ in range(55):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > t:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def optimize(h0, t, deadline, rng, a_min=30, a_max=1200):
    n = len(h0)
    h = proj(h0.astype(float).copy(), t)
    bc5 = compute_c5(h)
    lr = 0.004
    mom = 0.75
    v = np.zeros(n)
    stall = 0
    it = 0
    while time.time() < deadline:
        progress = min(1.0, it / 350.0)
        alpha = a_min + (a_max - a_min) * progress
        g = smooth_grad(h, alpha)
        v = mom * v + (1.0 - mom) * g
        h2 = proj(h - lr * v, t)
        c5 = compute_c5(h2)
        if c5 < bc5 - 1e-15:
            h = h2
            bc5 = c5
            stall = 0
            if lr < 0.06:
                lr *= 1.012
        else:
            h = h2
            stall += 1
            lr *= 0.95
            if stall > 45:
                amp = 0.012 + 0.006 * min(stall / 15.0, 5.0)
                hp = proj(h + rng.randn(n) * amp, t)
                if compute_c5(hp) < bc5:
                    h = hp
                    bc5 = compute_c5(h)
                lr = 0.004
                stall = 0
                v[:] = 0
        it += 1
    return h, bc5

def interp(h_src, n_new):
    n_src = len(h_src)
    x1 = np.linspace(0.0, 2.0, n_src, endpoint=False) + 1.0 / n_src
    x2 = np.linspace(0.0, 2.0, n_new, endpoint=False) + 1.0 / n_new
    return np.interp(x2, np.append(x1, 2.0 + 1.0 / n_src), np.append(h_src, h_src[0]))

def polish(h, t, deadline, rng):
    n = len(h)
    bc5 = compute_c5(h)
    no_imp = 0
    while time.time() < deadline and no_imp < 40:
        corr = np.correlate(h, 1 - h, mode='full')
        top = np.argsort(corr)[-4:]
        improved = False
        for ki in top:
            if improved:
                break
            k = ki - (n - 1)
            if k >= 0:
                c = h[:n-k] * (1.0 - h[k:])
                mi = int(np.argmax(c))
                for d in [0.02, 0.04, 0.07, 0.1, 0.15, 0.2]:
                    i, j = mi, mi + k
                    if h[i] >= d and h[j] <= 1.0 - d:
                        h[i] -= d
                        h[j] += d
                        c5 = compute_c5(h)
                        if c5 < bc5 - 1e-15:
                            bc5 = c5
                            improved = True
                            break
                        h[i] += d
                        h[j] -= d
            else:
                p = -k
                c = h[p:] * (1.0 - h[:n-p])
                mi = int(np.argmax(c))
                for d in [0.02, 0.04, 0.07, 0.1, 0.15, 0.2]:
                    i, j = mi + p, mi
                    if h[i] >= d and h[j] <= 1.0 - d:
                        h[i] -= d
                        h[j] += d
                        c5 = compute_c5(h)
                        if c5 < bc5 - 1e-15:
                            bc5 = c5
                            improved = True
                            break
                        h[i] += d
                        h[j] -= d
        if not improved:
            for _ in range(80):
                i, j = rng.randint(n, size=2)
                if i == j:
                    continue
                d = rng.uniform(0.03, 0.2)
                if h[i] >= d and h[j] <= 1.0 - d:
                    h[i] -= d
                    h[j] += d
                    if compute_c5(h) < bc5 - 1e-15:
                        bc5 = compute_c5(h)
                        improved = True
                        break
                    h[i] += d
                    h[j] -= d
            if not improved:
                no_imp += 1
                h = proj(h + rng.randn(n) * 0.01, t)
    return h, bc5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        n = 200
        x = np.linspace(0, 2, n, endpoint=False)
        h0 = 0.5 + 0.3 * np.sin(np.pi * x) + 0.1 * np.sin(2 * np.pi * x + 0.5)

    n = len(h0)
    t = n / 2.0
    best_h = proj(h0, t)
    best_c5 = compute_c5(best_h)

    # Phase 1: Main gradient optimization at original resolution (45% budget)
    d1 = t0 + budget_s * 0.45
    best_h, best_c5 = optimize(best_h, t, d1, rng, a_min=30, a_max=1200)

    # Phase 2: Try alternative resolutions (20% budget)
    d2 = t0 + budget_s * 0.65
    for nn in [100, 150]:
        if time.time() > d2:
            break
        ha = interp(best_h, nn)
        ha = proj(ha, nn / 2.0)
        hd, cd = optimize(ha, nn / 2.0, min(d2, time.time() + 70), rng, a_min=50, a_max=900)
        if cd < best_c5:
            best_h, best_c5 = hd, cd

    # Phase 2b: Second pass at best resolution with higher alpha (15% budget)
    d2b = t0 + budget_s * 0.80
    if time.time() < d2b:
        best_h, best_c5 = optimize(best_h, len(best_h) / 2.0, d2b, rng, a_min=200, a_max=2000)

    # Phase 3: Local polish (remaining time minus buffer)
    d3 = t0 + budget_s - 8
    best_h, best_c5 = polish(best_h, len(best_h) / 2.0, d3, rng)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
