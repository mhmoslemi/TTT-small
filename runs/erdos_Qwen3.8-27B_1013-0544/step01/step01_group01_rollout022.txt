import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_h(h, target_sum):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target_sum - s
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - h
            tr = room.sum()
            if tr > 1e-14:
                h += diff * room / tr
        else:
            room = h
            tr = room.sum()
            if tr > 1e-14:
                h += diff * room / tr
        h = np.clip(h, 0.0, 1.0)
        s2 = h.sum()
        if abs(s2 - target_sum) > 1e-14:
            h += (target_sum - s2) / len(h)
            h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta


def subgradient(h, n):
    """Compute subgradient of max_k corr_k(h) w.r.t. h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = corr.max()
    # Find all k that achieve (or are close to) the max
    threshold = max_val - 1e-8
    active = np.where(corr >= threshold)[0]
    grad = np.zeros(n)
    for k in active:
        l = k - (n - 1)
        i_start = max(0, -l)
        i_end = min(n, n - l)
        if i_end > i_start:
            i_vals = np.arange(i_start, i_end)
            j1 = i_vals + l
            j2 = i_vals
            grad[j1] += (1.0 - h[j2])
            grad[j2] -= h[j1]
    # Average over active set
    grad /= len(active)
    return grad, max_val, active


def nesterov_subgrad(h, target_sum, num_iters, lr_init, time_check):
    """Nesterov-accelerated subgradient descent."""
    n = len(h)
    y = h.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    beta = 0.9
    
    for t in range(num_iters):
        if time_check() < 30:
            break
        lr = lr_init / (1.0 + 0.001 * t)
        
        # Nesterov lookahead
        gamma = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta * t * t)) / (2.0 * beta * t + 2.0) if t > 0 else 0.0
        grad, _, _ = subgradient(y, n)
        
        y = best_h + gamma * (y - best_h)
        
        grad_y, _, _ = subgradient(y, n)
        v = beta * v - lr * grad_y
        h_new = best_h + v
        h_new = project_h(h_new, target_sum)
        
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_h = h_new
            best_c5 = c5
    
    return best_h, best_c5


def binary_local_search(b, target, max_iters, time_check):
    """Local search on binary sequence with exactly target ones."""
    n = len(b)
    b = b.copy()
    
    def get_max_overlap(b):
        corr = np.correlate(b.astype(float), 1.0 - b, mode='full')
        return corr.max()
    
    best_max = get_max_overlap(b)
    ones = np.where(b == 1)[0]
    zeros = np.where(b == 0)[0]
    
    for _ in range(max_iters):
        if time_check() < 20:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        b[i] = 0
        b[j] = 1
        mx = get_max_overlap(b)
        if mx < best_max:
            best_max = mx
            ones = np.where(b == 1)[0]
            zeros = np.where(b == 0)[0]
        else:
            b[i] = 1
            b[j] = 0
    
    return b, best_max


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    def time_left():
        return budget_s - (time.time() - start_time)

    if 'initial_h_values' in globals() and initial_h_values is not None:

        n_points = 70
        h_init = np.full(n_points, 0.5)

    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_h = project_h(h_init, target_sum)
    best_c5 = compute_c5(best_h)

    # Phase 1: Nesterov subgradient descent
    if time_left() > 100:
        best_h, best_c5 = nesterov_subgrad(best_h, target_sum, 10000, 0.01,
                                           lambda: time_left())

    # Phase 2: SLSQP with smooth max (multiple betas)
    bounds = Bounds(np.zeros(n_points), np.ones(n_points))

    def con_fun(x):
        return np.sum(x) - target_sum

    def con_jac(x):
        return np.ones(n_points)

    constraints = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]

    for beta_val in [80, 120, 200, 300]:
        if time_left() < 40:
            break

        def obj_func(h, b=beta_val):
            corr = np.correlate(h, 1.0 - h, mode="full")
            return smooth_max(corr, b) * dx

        try:
            res = minimize(obj_func, best_h, method='SLSQP', bounds=bounds,
                           constraints=constraints,
                           options={'maxiter': 1500, 'ftol': 1e-14})
            h_new = project_h(res.x, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new, c5
        except Exception:
            pass

    # Phase 3: Binary local search on a rounded version
    if time_left() > 60:
        b_init = (best_h > 0.5).astype(float)
        # Fix sum
        ones_count = int(b_init.sum())
        if ones_count > target_sum:
            # Remove excess ones (those furthest from 0.5 in h)
            one_indices = np.where(b_init == 1)[0]
            h_vals = best_h[one_indices]
            order = np.argsort(h_vals)
            for idx in order[:int(ones_count - target_sum)]:
                b_init[one_indices[idx]] = 0
        elif ones_count < target_sum:
            zero_indices = np.where(b_init == 0)[0]
            h_vals = best_h[zero_indices]
            order = np.argsort(-h_vals)
            for idx in order[:int(target_sum - ones_count)]:
                b_init[zero_indices[idx]] = 1

        b_searched, b_max = binary_local_search(b_init, target_sum, 50000,
                                                lambda: time_left())
        # Convert binary back to continuous and polish
        h_from_bin = b_searched.astype(float)
        h_from_bin = project_h(h_from_bin, target_sum)
        if time_left() > 30:
            try:

                    corr = np.correlate(h, 1.0 - h, mode="full")
                    return smooth_max(corr, 200.0) * dx
                res = minimize(obj_bin, h_from_bin, method='SLSQP', bounds=bounds,
                               constraints=constraints,
                               options={'maxiter': 1000, 'ftol': 1e-13})
                h_new = project_h(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h, best_c5 = h_new, c5
            except Exception:
                pass

    # Phase 4: Coordinate descent on active set
    if time_left() > 30:
        for it in range(20000):
            if time_left() < 15:
                break
            corr = np.correlate(best_h, 1.0 - best_h, mode="full")
            max_idx = np.argmax(corr)
            l = max_idx - (n_points - 1)
            
            # Identify coordinates that affect the max correlation
            i_start = max(0, -l)
            i_end = min(n_points, n_points - l)
            active_coords = np.unique(np.concatenate(
                [np.arange(i_start, i_end), np.arange(i_start, i_end) + l]))
            
            improved = False
            np.random.shuffle(active_coords)
            for j in active_coords:
                if time_left() < 15:
                    break
                old_val = best_h[j]
                # Try a few perturbations
                for delta in [0.01, -0.01, 0.03, -0.03, 0.005, -0.005]:
                    new_val = old_val + delta
                    if new_val < 0 or new_val > 1:
                        continue
                    best_h[j] = new_val
                    best_h = project_h(best_h, target_sum)
                    c5 = compute_c5(best_h)
                    if c5 < best_c5:
                        best_c5 = c5
                        improved = True
                        break
                    else:
                        best_h[j] = old_val
                        best_h = project_h(best_h, target_sum)
                if improved:
                    break
            if not improved and it % 500 == 499:
                # Add small noise and continue
                mask = np.random.random(n_points) < 0.1
                best_h[mask] += np.random.randn(int(mask.sum())) * 0.01

    # Phase 5: Final SLSQP polish with high beta
    if time_left() > 10:
        try:
            def obj_final(h):
                corr = np.correlate(h, 1.0 - h, mode="full")
                return smooth_max(corr, 500.0) * dx
            res = minimize(obj_final, best_h, method='SLSQP', bounds=bounds,
                           constraints=constraints,
                           options={'maxiter': 3000, 'ftol': 1e-15})
            h_new = project_h(res.x, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new, c5
        except Exception:
            pass

    # Phase 6: Try different n values if time remains
    if time_left() > 60:
        for n_alt in [50, 60, 80, 100]:
            if time_left() < 30:
                break
            n_pts = n_alt
            dx_alt = 2.0 / n_pts
            target_alt = n_pts / 2.0
            # Initialize by interpolating best_h

            x_new = np.linspace(0, 2, n_pts)

            h_alt = project_h(h_alt, target_alt)
            
            c5_alt = compute_c5(h_alt)
            if c5_alt < best_c5:
                best_h, best_c5, n_points = h_alt, c5_alt, n_pts
                dx = dx_alt
                target_sum = target_alt
                bounds = Bounds(np.zeros(n_points), np.ones(n_points))

    return best_h, best_c5, n_points
