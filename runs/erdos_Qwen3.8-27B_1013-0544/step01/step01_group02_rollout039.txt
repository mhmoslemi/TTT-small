    for step in range(num_steps):
        # Compute g_k
        c = np.correlate(h, 1-h, mode='full') * (2.0/n)
        max_val = c.max()
        
        # Compute softmax weights
        shifted = alpha * (c - max_val)
        p = np.exp(shifted) / np.exp(shifted).sum()
        
        # Compute gradient: sum_k p_k * grad g_k

        grad = compute_gradient(h, n, p)
        
        # Take step
        h_new = h - step_size * grad
        
        # Project
        h_new = project_feasible(h_new, n)
        
        h = h_new
    return h

- Otherwise, at a vertex of the upper envelope

To find the minimum, we can:
1. Sort lines by slope
2. Compute the upper envelope (vertices where the active line changes)
3. Find the vertex with t closest to where the slope crosses 0

The upper envelope of m lines can be computed in O(m log m) time.

But even simpler: since f(t) is convex, we can use a bisection/ternary search on t. Each evaluation is O(m). With 50 iterations of bisection, that's 50*m = 50*159 ≈ 8000 operations per coordinate. Times 80 = 640,000 per pass. Very fast!

Let me use bisection on the subgradient:
- f(t) is convex, so f'(t) is non-decreasing
- f'(t) = slope of the active line at t
- We want to find t where f'(t) crosses 0 (or where 0 is in the subdifferential)

At a given t, the subdifferential is [min slope among active lines, max slope among active lines].
Active lines at t are those with b[m] + s[m]*t = f(t) (within numerical tolerance).

Algorithm:
1. Evaluate f(0) and find active lines. If min slope among active ≥ 0, return t=0.
2. Evaluate f(1) and find active lines. If max slope among active ≤ 0, return t=1.
3. Bisect: at t = (lo+hi)/2, find active lines. If min slope < 0, the minimum is to the right, so lo = t. If max slope > 0, the minimum is to the left, so hi = t. If 0 is in [min slope, max slope], we've found the minimum.

This converges in O(log(1/ε) * m) time per coordinate.

def find_min_of_max_lines(b, slopes, lo=0.0, hi=1.0, tol=1e-12):

    m_len = len(b)
    
    def eval_and_subdiff(t):
        vals = b + slopes * t
        max_val = vals.max()
        # Active lines
        active = vals >= max_val - 1e-12
        if not active.any():
            # Numerical issue, use all
            active = np.ones(m_len, dtype=bool)
        return max_val, slopes[active].min(), slopes[active].max()
    
    # Check endpoints
    _, min_s0, max_s0 = eval_and_subdiff(lo)
    if min_s0 >= -1e-12:
        return lo
    
    _, min_s1, max_s1 = eval_and_subdiff(hi)
    if max_s1 <= 1e-12:
        return hi
    
    # Bisection
    for _ in range(80):
        mid = (lo + hi) / 2.0
        _, min_s, max_s = eval_and_subdiff(mid)
        if min_s < -1e-12:
            lo = mid  # minimum is to the right
        elif max_s > 1e-12:
            hi = mid  # minimum is to the left
        else:
            return mid  # 0 is in subdifferential
    return (lo + hi) / 2.0

def compute_c5(h, n):

    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def project_feasible(h, n):
    h = np.array(h, dtype=float)
    target = n / 2.0
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def find_min_max_lines(b, slopes):
    def get_subdiff(t):
        vals = b + slopes * t
        mx = vals.max()
        active = vals >= mx - 1e-9
        if not np.any(active):
            return slopes.min(), slopes.max()
        return slopes[active].min(), slopes[active].max()
    ms0, _ = get_subdiff(0.0)
    if ms0 >= -1e-12:
        return 0.0
    _, ms1 = get_subdiff(1.0)
    if ms1 <= 1e-12:
        return 1.0
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        ms_lo, ms_hi = get_subdiff(mid)
        if ms_lo < -1e-12:
            lo = mid
        elif ms_hi > 1e-12:
            hi = mid
        else:
            return mid
    return (lo + hi) * 0.5

def coord_pass_lam(h, n, lam):
    h = h.copy()
    dx = 2.0 / n
    d_all = np.arange(2 * n - 1) - (n - 1)
    for j in range(n):
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        vp = (j + d_all >= 0) & (j + d_all < n)
        vm = (j - d_all >= 0) & (j - d_all < n)
        slopes = np.zeros(2 * n - 1)
        slopes[vp] += (1.0 - h[j + d_all[vp]])
        slopes[vm] -= h[j - d_all[vm]]
        slopes *= dx
        s = slopes + lam
        b = c - slopes * h[j]
        h[j] = find_min_max_lines(b, s)
    return h

    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    mx = c.max()
    shifted = alpha * (c - mx)
    mx2 = shifted.max()
    if mx2 > 500:
        return mx + (mx2 + np.log(np.sum(np.exp(shifted - mx2)))) / alpha
    return mx + np.log(np.sum(np.exp(shifted))) / alpha

def sum_con(h, n):
    return h.sum() - n / 2.0

def sum_con_jac(h, n):
    return np.ones(n)
