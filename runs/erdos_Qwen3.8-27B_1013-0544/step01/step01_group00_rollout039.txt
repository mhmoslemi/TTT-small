def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_h(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = np.sum(mask)
    if cnt > 0:
        h = h.copy()
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

    h = v[:n]
    t = v[-1]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return t - corr

    n = len(v) - 1

            c5 = c5_cd
            h_best = h_cd.copy()

    # Phase 4: Random restarts with multi-stage polish
    for i in range(25):
        if time_left() < 50:
            break
        scale = 0.002 + 0.006 * rng.random()
        h_p = h_best + rng.randn(n) * scale
        h_p = project_h(h_p, n * 0.5)
        h_new, c5_new = optimize_t_var(h_p, max_iter=500, ftol=1e-13)
        if c5_new < c5:
            c5 = c5_new
            h_best = h_new.copy()
        elif c5_new < c5 + 0.001 and time_left() > 100:
            h_s, c5_s = optimize_smooth(h_new, 0.003, max_iter=300, ftol=1e-14)
            if c5_s < c5:
                c5 = c5_s
                h_best = h_s.copy()
            elif time_left() > 80:
                h_f, c5_f = optimize_t_var(h_s, max_iter=400, ftol=1e-14)
                if c5_f < c5:
                    c5 = c5_f
                    h_best = h_f.copy()

    # Phase 5: Final SLSQP polish
    if time_left() > 60:
        h_new, c5_new = optimize_t_var(h_best, max_iter=800, ftol=1e-14)
        if c5_new < c5:
            c5 = c5_new
            h_best = h_new.copy()

    # Phase 6: Resolution increase
    for n_new in [360, 540]:
        if time_left() < 200:
            break
        h_interp = interp_h(h_best, n_new)
        c5_cur = compute_c5(h_interp)
        if c5_cur > c5 + 0.004:
            continue
        h_cur = h_interp
        h_new, c5_new = optimize_t_var(h_cur, max_iter=800, ftol=1e-13)
        if c5_new < c5:
            c5 = c5_new
            h_best = h_new.copy()
            h_cur = h_new
        for temp in [0.005, 0.002, 0.001]:
            if time_left() < 60:
                break
            h_new, c5_new = optimize_smooth(h_cur, temp, max_iter=400, ftol=1e-14)
            if c5_new < c5:
                c5 = c5_new
                h_best = h_new.copy()
                h_cur = h_new
        if time_left() > 120:
            h_new, c5_new = optimize_t_var(h_cur, max_iter=800, ftol=1e-14)
            if c5_new < c5:
                c5 = c5_new
                h_best = h_new.copy()

    # Final validation
    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = project_h(h_best, len(h_best) * 0.5)
    c5_final = compute_c5(h_best)
    return h_best, c5_final, len(h_best)
