    g = np.zeros(n)

    w /= w.sum()

    for i, k_idx in enumerate(top_idx):
        if w[i] > 1e-12:
            grad += w[i] * grad_at_shift(h, k_idx, n)
    return grad * (2.0 / n), np.max(corr) * (2.0 / n)

def optimize_core(h0, target, deadline, rng, alpha_init=50, n_top_init=7):
    n = len(h0)
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    step = 0.004
    momentum = np.zeros(n)
    alpha = alpha_init
    n_top = n_top_init
    stall = 0
    iter_ct = 0

    while time.time() < deadline - 2:
        iter_ct += 1
        grad, _ = subgrad_direction(h, alpha, n_top)
        gn = np.linalg.norm(grad)
        if gn < 1e-12:
            break
        momentum = 0.88 * momentum - step * grad
        h_new = project_onto_simplex_box(h + momentum, target)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5 - 1e-14:
            h = h_new
            best_c5 = c5_new
            best_h = h.copy()
            stall = 0
            step = min(step * 1.008, 0.02)
        else:
            stall += 1
            if stall > 30:
                step *= 0.95
            if stall > 150:
                h = project_onto_simplex_box(h + rng.randn(n) * 0.012, target)
                momentum = np.zeros(n)
                step = 0.004
                alpha = min(alpha * 1.4, 250.0)
                n_top = max(4, n_top - 1)
                stall = 0
            elif stall > 60:
                g_single, _ = subgrad_direction(h, 500, 1)
                h_try = project_onto_simplex_box(h - 0.003 * g_single, target)
                if compute_c5(h_try) < best_c5:
                    h = h_try
                    best_c5 = compute_c5(h)
                    best_h = h.copy()

        if iter_ct % 800 == 0:
            n_top = max(4, n_top - 1)
            alpha = min(alpha * 1.15, 300.0)

    return best_h, best_c5

def decode_ctrl(h_ctrl, n_full):
    nc = len(h_ctrl)
    x_c = np.linspace(0.0, 2.0, nc, endpoint=False)
    x_f = np.linspace(0.0, 2.0, n_full, endpoint=False)
    h_ext = np.append(h_ctrl, h_ctrl[0]
)
    x_ext = np.append(x_c, 2.0)
    return np.interp(x_f, x_ext, h_ext)

def ctrl_obj(h_ctrl, n_full):
    h = decode_ctrl(h_ctrl, n_full)
    h = project_onto_simplex_box(h, n_full / 2.0)
    return compute_c5(h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in dir() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = 0.5 + 0.25 * rng.randn(200)

    n = len(h_parent)
    target = n / 2.0
    h_parent = project_onto_simplex_box(h_parent, target)

    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Low-dimensional optimization via control points
    n_ctrl = 40
    x_c = np.linspace(0.0, 2.0, n_ctrl, endpoint=False)
    x_f = np.linspace(0.0, 2.0, n, endpoint=False)
    h_ctrl_init = np.interp(x_c, x_f, h_parent)

    phase_deadline = t0 + budget_s * 0.35
    for trial in range(4):
        if time.time() > phase_deadline - 20:
            break
        if trial == 0:
            h_c = h_ctrl_init.copy()
        elif trial == 1:
            h_c = h_ctrl_init + rng.randn(n_ctrl) * 0.04
        elif trial == 2:
            # Try anti-periodic structure
            half = n_ctrl // 2
            h_c[:half] = 0.5 + 0.3 * rng.randn(half)
            h_c[half:] = 1.0 - h_c[:half]
        else:
            # Try more square-wave like
            h_c = 0.5 + 0.35 * np.sin(3.0 * np.pi * x_c / 2.0) + 0.1 * rng.randn(n_ctrl)

        h_c = np.clip(h_c, 0.0, 1.0)
        res = minimize(ctrl_obj, h_c, args=(n,), method='Nelder-Mead',
                       options={'maxiter': 800, 'xatol': 1e-7, 'fatol': 1e-8})
        h_full = decode_ctrl(res.x, n)
        h_full = project_onto_simplex_box(h_full, target)
        c5 = compute_c5(h_full)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_full.copy()

    # Phase 2: Full-resolution subgradient optimization
    remaining = budget_s - (time.time() - t0)
    if remaining > 30:
        h_opt, c5_opt = optimize_core(best_h, target, t0 + budget_s * 0.75, rng,
                                      alpha_init=60, n_top_init=6)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt

    # Phase 3: Try different initializations at full resolution
    remaining = budget_s - (time.time() - t0)
    if remaining > 40:
        # Anti-periodic initialization
        half_n = n // 2
        h_anti = np.zeros(n)
        h_anti[:half_n] = 0.5 + 0.25 * rng.randn(half_n)
        h_anti[half_n:] = 1.0 - h_anti[:half_n]
        h_anti = project_onto_simplex_box(h_anti, target)
        dl = t0 + budget_s * 0.9
        h_occ, c5_occ = optimize_core(h_anti, target, dl, rng, alpha_init=50, n_top_init=5)
        if c5_occ < best_c5:
            best_h = h_occ
            best_c5 = c5_occ
        remaining = budget_s - (time.time() - t0)

    # Phase 4: Final polish
    if remaining > 15:
        h_fin, c5_fin = optimize_core(best_h, target, t0 + budget_s - 3, rng,
                                      alpha_init=120, n_top_init=4)
        if c5_fin < best_c5:
            best_h = h_fin
            best_c5 = c5_fin

    best_h = project_onto_simplex_box(best_h, target)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
