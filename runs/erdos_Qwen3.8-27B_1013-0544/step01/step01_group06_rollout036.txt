import numpy as np
import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_box_sum(h, target_sum):
    lo, hi = -1.5, 1.5
    for _ in range(40):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def grad_lag(h, k_idx):
    n = len(h)
    k = k_idx - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n - k])
        if k > 0:
            g[:n - k] -= h[k:]
    else:
        p = -k
        g[:n - p] += (1.0 - h[p:])
        if p > 0:
            g[p:] -= h[:n - p]
    return g

def nesterov_optimize(h0, target_sum, deadline, rng, alpha_init=50, n_restarts=0):
    n = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    m = np.zeros(n)  # momentum
    beta = 0.9
    lr = 0.01
    alpha = alpha_init
    patience = 0
    stall = 0
    
    while time.time() < deadline:
        # Nesterov lookahead
        y = best_h + beta * m
        corr = np.correlate(y, 1 - y, mode='full')
        m_val = np.max(corr)
        weights = np.exp(alpha * (corr - m_val))
        wsum = np.sum(weights)
        if wsum < 1e-300:
            weights = np.zeros_like(corr)
            weights[np.argmax(corr)] = 1.0
        else:
            weights /= wsum
        
        active = np.where(weights > 1e-8)[0]
        grad = np.zeros(n)
        for idx in active:
            grad += weights[idx] * grad_lag(y, idx)
        
        m = (best_h - y) / beta if beta != 1 else grad
        # Actually standard Nesterov:
        # y_{k+1} = x_k + beta_k * (x_k - x_{k-1})
        # x_{k+1} = proj(y_{k+1} - lr * grad(y_{k+1}))
        
        h_new = y - lr * grad
        h_new = project_box_sum(h_new, target_sum)
        new_c5 = compute_c5(h_new)
        
        if new_c5 < best_c5:
            m = h_new - best_h  # this is the velocity
            best_h = h_new
            best_c5 = new_c5
            patience = 0
            lr = min(lr * 1.005, 0.03)
        else:
            patience += 1
            lr *= 0.98
            if patience > 80:
                sz = 0.02 + 0.01 * stall
                h_pert = best_h + rng.randn(n) * sz
                h_pert = project_box_sum(h_pert, target_sum)
                pc5 = compute_c5(h_pert)
                if pc5 < best_c5:
                    best_h = h_pert
                    best_c5 = pc5
                m[:] = 0
                lr = 0.01
                alpha = min(alpha * 1.2, 300.0)
                patience = 0
                stall += 1
                if stall > 12:
                    break
    
    return best_h, best_c5

def interpolate(h_old, n_old, n_new):
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    return np.interp(x_new, x_old, h_old)

        remaining = budget_s - (time.time() - t0)

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_box_sum(h, target_sum):
    lo, hi = -1.5, 1.5
    for _ in range(45):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def grad_lag(h, k_idx):
    n = len(h)
    k = k_idx - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n - k])
        if k > 0:
            g[:n - k] -= h[k:]
    else:
        p = -k
        g[:n - p] += (1.0 - h[p:])
        if p > 0:
            g[p:] -= h[:n - p]
    return g

def smooth_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    wsum = np.sum(w)
    if wsum < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= wsum
    grad = np.zeros(n)
    active = np.where(w > 1e-10)[0]
    for idx in active:
        grad += w[idx] * grad_lag(h, idx)
    return grad * (2.0 / n)

def nesterov_opt(h0, target_sum, deadline, rng, alpha=50, lr_init=0.012):
    n = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    x = best_h.copy()
    v = np.zeros(n)
    lr = lr_init
    patience = 0
    stall = 0
    t_k = 1.0

    while time.time() < deadline:
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        beta_k = (1.0 - t_k) / t_k
        y = x + beta_k * v
        grad = smooth_grad(y, alpha)
        new_x = y - lr * grad
        new_x = project_box_sum(new_x, target_sum)
        new_c5 = compute_c5(new_x)

        if new_c5 < best_c5:
            v = new_x - x
            x = new_x
            best_h = new_x.copy()
            best_c5 = new_c5
            patience = 0
            if lr < 0.025:
                lr *= 1.004
        else:
            lr *= 0.97
            patience += 1
            v *= 0.7
            if patience > 50:
                sz = 0.03 + 0.015 * stall
                x = best_h + rng.randn(n) * sz
                x = project_box_sum(x, target_sum)
                v[:] = 0.0
                lr = lr_init
                alpha = min(alpha * 1.3, 350.0)
                patience = 0
                stall += 1
                if stall > 14:
                    break

    return best_h, best_c5

def interpolate(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    return np.interp(x_new, x_old, h_old)

def make_structured_init(n, rng, typ):
    h = np.zeros(n)
    if typ == 0:
        h[:] = 0.5 + 0.3 * np.cos(2.0 * np.pi * np.arange(n) / n + rng.uniform(0, 6.28))
