    # The function is convex piecewise-linear.
    # Minimum is where the subgradient contains 0, or at a boundary.
    
    # At t, the active constraint is argmax_k (p[k]*t + q[k])
    # The slope is the p[k] of the active constraint (if unique)
    
    # Binary search on the slope: find t where slope crosses 0

    val_hi = np.max(p * hi + q)

    # Check if this point is in [lo, hi] and is actually on the upper envelope
    
    # More efficient: just do a ternary search or binary search on the slope
    # Since the function is convex, we can binary search
    
    def f(t):

        if slope < 0:
            lo_t = mid

            hi_t = mid
        else:
            return mid
    return (lo_t + hi_t) / 2

def _slope_at(p, q, t):
    """Return the subgradient (slope) of max_k(p[k]*t + q[k]) at t."""
    vals = p * t + q

    active = np.where(np.abs(vals - max_val) < 1e-12)[0]
    # The subgradient is the convex hull of {p[k] : k in active}
    # For bisection, we can use the mean or just check if 0 is in the range
    return np.min(p[active]), np.max(p[active])

def solve_1d_minimax(p, c, u_lo, u_hi):
    """Minimize max_k (c[k] + u*p[k]) for u in [u_lo, u_hi]."""
    # c and p are arrays of length (2n-1)
    # The function g(u) = max_k(c[k] + u*p[k]) is convex piecewise-linear.
    
    # Bisection on u
    lo, hi = u_lo, u_hi
    for _ in range(80):
        mid = (lo + hi) / 2
        vals = c + mid * p
        max_val = np.max(vals)
        active = np.where(np.abs(vals - max_val) < 1e-12)[0]
        if len(active) == 0:
            active = np.array([np.argmax(vals)])
        min_p = np.min(p[active])
        max_p = np.max(p[active])
        if min_p >= -1e-14:
            lo = mid
        elif max_p <= 1e-14:
            hi = mid
        else:
            break
    u_opt = (lo + hi) / 2
    return u_opt

def pair_descent_step(h, i, j):

    n = len(h)
    s = h[i] + h[j]

    # where L ranges from -(n-1) to (n-1)
    # and I1 = (0 <= i-L < n), I2 = (0 <= i+L < n), etc.
    
    # In numpy correlate, the index k in corr corresponds to lag L = k - (n-1)
    # So I need to compute p for each k = 0, ..., 2n-2
    
    # Let's vectorize:
    # For each lag L, the indices i-L, i+L, j-L, j+L must be checked.
    
    # lags = np.arange(-(n-1), n)  # length 2n-1
    # For each lag L:
    #   I1 = (0 <= i-L < n)
    #   I2 = (0 <= i+L < n)
    #   I3 = (0 <= j-L < n)
    #   I4 = (0 <= j+L < n)
    
    # Vectorized:
    lags = np.arange(-(n-1), n, dtype=float)
    
    i_minus_L = i - lags  # array
    i_plus_L = i + lags
    j_minus_L = j - lags
    j_plus_L = j + lags

    I2 = (i_plus_L >= 0) & (i_plus_L < n)
    I3 = (j_minus_L >= 0) & (j_minus_L < n)
    I4 = (j_plus_L >= 0) & (j_plus_L < n)
    
    # Get values (handle out-of-bounds with 0)
    def safe_get(arr, idx, mask):
        result = np.zeros(len(idx))
        valid = mask & (idx >= 0) & (idx < len(arr))
        result[valid] = arr[idx[valid].astype(int)]
        return result

    h_i_plus_L = safe_get(h, i_plus_L, I2)
    h_j_minus_L = safe_get(h, j_minus_L, I3)
    h_j_plus_L = safe_get(h, j_plus_L, I4)

Let me also think about whether the 1D minimax solver is correct. The function g(u) = max_k (c_k + u * p_k) is convex in u (max of linear functions). We want to minimize it over [u_lo, u_hi].

The subgradient of g at u is {p_k : c_k + u*p_k = g(u)}. If 0 is in this set (or in its convex hull), then u is optimal.

In the bisection:
- At mid, compute vals = c + mid * p
- Find all k that achieve the max
- If min(p[active]) >= 0: the function is non-decreasing at mid, so the minimum is at or before mid → hi = mid
- If max(p[active]) <= 0: the function is non-increasing at mid, so the minimum is at or after mid → lo = mid
- Otherwise: 0 is in the subgradient, mid is optimal → break

This is correct.

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def solve_1d_minimax(p, c, u_lo, u_hi):
    if u_hi - u_lo < 1e-14:
        return (u_lo + u_hi) * 0.5
    lo, hi = u_lo, u_hi
    for _ in range(64):
        mid = (lo + hi) * 0.5
        vals = c + mid * p
        max_val = np.max(vals)
        active = np.where(vals >= max_val - 1e-9)[0]
        if np.min(p[active]) > 1e-10:
            hi = mid
        elif np.max(p[active]) < -1e-10:
            lo = mid
        else:
            break
    return (lo + hi) * 0.5

def fix_sum(h, target_sum):
    h = h.copy()
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return np.clip(h, 0, 1)
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return np.clip(h, 0, 1)
    h[idx] += diff / len(idx)
    return np.clip(h, 0, 1)

def pair_descent_timed(h, corr, remaining_s, rng):
    n = len(h)
    n_full = 2 * n - 1
    lags = np.arange(-(n - 1), n, dtype=int)
    t_start = time.time()
    count = 0
    while time.time() - t_start < remaining_s:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            j = (j + 1) % n
        s = h[i] + h[j]
        u_lo = max(0.0, s - 1.0) - h[i]
        u_hi = min(1.0, s) - h[i]
        if u_hi - u_lo < 1e-12:
            continue
        i_mL = i - lags
        i_pL = i + lags
        j_mL = j - lags
        j_pL = j + lags
        I1 = (i_mL >= 0) & (i_mL < n)
        I2 = (i_pL >= 0) & (i_pL < n)
        I3 = (j_mL >= 0) & (j_mL < n)
        I4 = (j_pL >= 0) & (j_pL < n)
        h_iL = np.zeros(n_full)
        h_iL[I1] = h[i_mL[I1]]
        h_iLp = np.zeros(n_full)
        h_iLp[I2] = h[i_pL[I2]]
        h_jL = np.zeros(n_full)
        h_jL[I3] = h[j_mL[I3]]
        h_jLp = np.zeros(n_full)
        h_jLp[I4] = h[j_pL[I4]]
        p = (1.0 - h_iL)*I1.astype(float) - h_iLp*I2.astype(float) \
            - (1.0 - h_jL)*I3.astype(float) + h_jLp*I4.astype(float)
        u_opt = solve_1d_minimax(p, corr, u_lo, u_hi)
        new_max = np.max(corr + u_opt * p)
        if new_max < np.max(corr) - 1e-13:
            h[i] = np.clip(h[i] + u_opt, 0.0, 1.0)
            h[j] = np.clip(h[j] - u_opt, 0.0, 1.0)
            corr = corr + u_opt * p
        count += 1
        if count % 500 == 0:
            corr = np.correlate(h, 1 - h, mode="full").astype(float)
    corr = np.correlate(h, 1 - h, mode="full").astype(float)
    return h, corr

def make_binary_from_alpha(n, alpha, seed_val):
    rng = np.random.RandomState(seed_val)
    h = np.zeros(n)
    for i in range(n):
        if (i * alpha / n) % 1.0 < 0.5:
            h[i] = 1.0
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones > target:
        ones = np.where(h == 1)[0]
        h[rng.choice(ones, n_ones - target, replace=False)] = 0
    elif n_ones < target:
        zeros = np.where(h == 0)[0]
        h[rng.choice(zeros, target - n_ones, replace=False)] = 1
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    h0 = np.array(initial_h_values, dtype=float).copy()
    best_h = h0.copy()
    n_best = len(h0)
    best_c5 = compute_c5(best_h, 2.0 / n_best)

    n = n_best
    dx = 2.0 / n
    target_sum = n / 2.0
    h = fix_sum(h0.copy(), target_sum)
    corr = np.correlate(h, 1 - h, mode="full").astype(float)

    # Phase 1: Main pair descent on initial (50% budget)
    t1 = min(budget_s * 0.5, budget_s - elapsed() - 10)
    if t1 > 10:
        h, corr = pair_descent_timed(h, corr, t1, rng)
        c5_now = compute_c5(h, dx)
        if c5_now < best_c5:
            best_h, best_c5 = h.copy(), c5_now

    # Phase 2: Structured binary starts (25% budget)
    if elapsed() < budget_s - 60:
        alphas = [(1+5**0.5)/2, (1+17**0.5)/2, np.pi, np.e, 2.414, 0.618]
        time_per = (budget_s * 0.25) / len(alphas)
        for ai, alpha in enumerate(alphas):
            if elapsed() > budget_s - 40:
                break
            h_try = make_binary_from_alpha(n, alpha, seed + ai * 7)
            c_try = np.correlate(h_try, 1 - h_try, mode="full").astype(float)
            t_alloc = min(time_per, budget_s - elapsed() - 20)
            if t_alloc < 2:
                continue
            h_try, c_try = pair_descent_timed(h_try, c_try, t_alloc, rng)
            c5_try = compute_c5(h_try, dx)
            if c5_try < best_c5:
                best_h, best_c5 = h_try.copy(), c5_try

    # Phase 3: Try n=80 (20% budget)
    if elapsed() < budget_s - 40:
        n80 = 80
        dx80 = 2.0 / n80
        ts80 = n80 / 2.0
        t3 = min(budget_s * 0.2, budget_s - elapsed() - 20)
        if t3 > 5:
            # Downsample best to n=80
            h80 = np.interp(np.linspace(0, 1, n80, endpoint=False),
                            np.linspace(0, 1, len(best_h), endpoint=False), best_h)
            h80 = fix_sum(h80, ts80)
            c80 = np.correlate(h80, 1 - h80, mode="full").astype(float)
            h80, c80 = pair_descent_timed(h80, c80, t3 * 0.6, rng)
            c5_80 = compute_c5(h80, dx80)
            if c5_80 < best_c5:
                best_h, best_c5, n_best = h80.copy(), c5_80, n80
            # Golden binary at n=80
            if elapsed() < budget_s - 30:
                ga = (1 + 5**0.5) / 2
                hg80 = make_binary_from_alpha(n80, ga, seed + 999)
                cg80 = np.correlate(hg80, 1 - hg80, mode="full").astype(float)
                hg80, cg80 = pair_descent_timed(hg80, cg80, budget_s - elapsed() - 25, rng)
                c5_g80 = compute_c5(hg80, dx80)
                if c5_g80 < best_c5:
                    best_h, best_c5, n_best = hg80.copy(), c5_g80, n80

    # Final verification
    dx_final = 2.0 / n_best
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h, dx_final)
    return best_h, best_c5, n_best
