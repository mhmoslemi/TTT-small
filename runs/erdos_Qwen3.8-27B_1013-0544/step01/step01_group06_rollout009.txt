def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_box_simplex(h, target_sum):
    lo, hi = -1.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def smooth_max_grad(h, alpha, top_m=5):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    m_max = np.max(corr)
    if top_m < len(corr):
        idx = np.argpartition(corr, -top_m)[-top_m:]
        sub = corr[idx]
        w = np.exp(alpha * (sub - sub.max()))
        s = w.sum()
        if s < 1e-300:
            w = np.zeros(top_m)
            w[np.argmax(sub)] = 1.0
        else:
            w /= s
    else:
        w = np.exp(alpha * (corr - m_max))
        s = w.sum()
        if s < 1e-300:
            w = np.zeros(len(corr))
            w[np.argmax(corr)] = 1.0
        else:
            w /= s
        idx = np.where(w > 1e-12)[0]
    grad = np.zeros(n)
    dx = 2.0 / n
    for j in range(len(idx)):
        wi = w[j] if top_m < len(corr) else w[idx[j]]
        if wi < 1e-12:
            continue
        k = idx[j] - (n - 1) if top_m < len(corr) else idx[j] - (n - 1)
        if k >= 0:
            g = np.zeros(n)
            g[k:] += (1.0 - h[:n - k])
            if k > 0:
                g[:n - k] -= h[k:]
        else:
            p = -k
            g[:n - p] += (1.0 - h[p:])
            if p > 0:
                g[p:] -= h[:n - p]
        grad += wi * g
    return grad * dx

def grad_corr_exact(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n - k])
        if k > 0:
            g[:n - k] -= h[k:]
    else:
        p = -k
        g[:n - p] += (1.0 - h[p:])
        if p > 0:
            g[p:] -= h[:n - p]
    return g

def active_set_grad(h, top_m=3):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    m_max = np.max(corr)
    idx = np.argpartition(corr, -top_m)[-top_m:]
    vals = corr[idx]
    w = np.exp(100.0 * (vals - vals.max()))
    s = w.sum()
    if s < 1e-300:
        w = np.zeros(top_m)
        w[np.argmax(vals)] = 1.0
    else:
        w /= s
    grad = np.zeros(n)
    dx = 2.0 / n
    for j in range(top_m):
        k = idx[j] - (n - 1)
        grad += w[j] * grad_corr_exact(h, k)
    return grad * dx

def optimize_gradient(h0, target_sum, deadline, rng, alpha=300, top_m=5):
    n = len(h0)
    best_h = project_box_simplex(h0.astype(float).copy(), target_sum)
    best_c5 = compute_c5(best_h)
    lr = 0.008
    patience = 0
    stall_count = 0
    alpha_cur = alpha

    while time.time() < deadline:
        grad = smooth_max_grad(best_h, alpha_cur, top_m)
        step = lr
        improved = False
        for _ in range(15):
            h_try = project_box_simplex(best_h - step * grad, target_sum)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_h = h_try
                best_c5 = c5_try
                improved = True
                break
            step *= 0.5
        if improved:
            patience = 0
            if lr < 0.15:
                lr = min(lr * 1.04, 0.15)
        else:
            patience += 1
            lr *= 0.85
            if patience > 40:
                size = 0.03 + 0.02 * min(stall_count, 5)
                h_pert = project_box_simplex(best_h + rng.randn(n) * size, target_sum)
                c5_pert = compute_c5(h_pert)
                if c5_pert < best_c5:
                    best_h = h_pert
                    best_c5 = c5_pert
                alpha_cur = min(alpha_cur * 1.4, 800.0)
                lr = 0.008
                patience = 0
                stall_count += 1
                if stall_count > 12:
                    break
    return best_h, best_c5

def optimize_active_set(h0, target_sum, deadline, rng, top_m=3):
    n = len(h0)
    best_h = project_box_simplex(h0.astype(float).copy(), target_sum)
    best_c5 = compute_c5(best_h)
    lr = 0.005
    patience = 0
    stall_count = 0

    while time.time() < deadline:
        grad = active_set_grad(best_h, top_m)
        step = lr
        improved = False
        for _ in range(15):
            h_try = project_box_simplex(best_h - step * grad, target_sum)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_h = h_try
                best_c5 = c5_try
                improved = True
                break
            step *= 0.5
        if improved:
            patience = 0
            if lr < 0.1:
                lr = min(lr * 1.05, 0.1)
        else:
            patience += 1
            lr *= 0.85
            if patience > 50:
                size = 0.04 + 0.02 * min(stall_count, 4)
                h_pert = project_box_simplex(best_h + rng.randn(n) * size, target_sum)
                c5_pert = compute_c5(h_pert)
                if c5_pert < best_c5:
                    best_h = h_pert
                    best_c5 = c5_pert
                lr = 0.005
                patience = 0
                stall_count += 1
                if stall_count > 8:
                    break
    return best_h, best_c5

def quantize_and_refine(h, target_sum, deadline, rng):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    levels = [0.0, 0.25, 0.5, 0.75, 1.0]
    for lv in levels:
        h_q = np.clip(np.round(h / lv + 0.5) * lv if lv > 0 else h, 0, 1)
        h_q = project_box_simplex(h_q, target_sum)
        c5_q = compute_c5(h_q)
        if c5_q < best_c5:
            best_h = h_q
            best_c5 = c5_q
    h_b = (h > 0.5).astype(float)
    h_b = project_box_simplex(h_b, target_sum)
    c5_b = compute_c5(h_b)
    if c5_b < best_c5:
        best_h = h_b
        best_c5 = c5_b
    if time.time() < deadline:
        best_h, best_c5 = optimize_gradient(best_h, target_sum, deadline, rng, alpha=400, top_m=4)
    return best_h, best_c5

def interpolate_to_n(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False) + 1.0 / n_new
    h_ext = np.concatenate([h_old, h_old[:1]])
    x_ext = np.concatenate([x_old, [2.0 + 1.0 / n_old]])
    return np.interp(x_new, x_ext, h_ext)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        n = 100
        h_init = np.clip(0.5 + 0.3 * rng.randn(n), 0, 1)
        h_init = project_box_simplex(h_init, n / 2.0)

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    best_n = len(best_h)

    n = len(h_init)
    target = n / 2.0

    # Phase 1: Gradient optimization at initial resolution
    d1 = t0 + min(budget_s * 0.30, 300.0)
    h1, c5_1 = optimize_gradient(h_init, target, d1, rng, alpha=250, top_m=5)
    if c5_1 < best_c5:
        best_h, best_c5, best_n = h1, c5_1, n

    # Phase 2: Quantize and refine
    d2 = t0 + min(budget_s * 0.50, 500.0)
    if time.time() < d2:
        h2, c5_2 = quantize_and_refine(best_h, target, d2, rng)
        if c5_2 < best_c5:
            best_h, best_c5, best_n = h2, c5_2, len(h2)

    # Phase 3: Active set optimization
    d3 = t0 + min(budget_s * 0.65, 650.0)
    if time.time() < d3:
        h3, c5_3 = optimize_active_set(best_h, len(best_h) / 2.0, d3, rng, top_m=3)
        if c5_3 < best_c5:
            best_h, best_c5, best_n = h3, c5_3, len(h3)

    # Phase 4: Try higher resolution
    remaining = budget_s - (time.time() - t0) - 20
    for n_try in [300, 400]:
        if remaining < 80:
            break
        h_interp = interpolate_to_n(best_h, best_n, n_try)
        h_interp = project_box_simplex(h_interp, n_try / 2.0)
        d_phase = min(time.time() + remaining * 0.4, t0 + budget_s - 30)
        h_opt, c5_opt = optimize_gradient(h_interp, n_try / 2.0, d_phase, rng, alpha=200, top_m=5)
        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, n_try
        remaining = budget_s - (time.time() - t0) - 20

    # Phase 5: Final polish
    remaining = budget_s - (time.time() - t0)
    if remaining > 30:
        h_f, c5_f = optimize_gradient(best_h, len(best_h) / 2.0, t0 + budget_s - 3, rng, alpha=500, top_m=3)
        if c5_f < best_c5:
            best_h, best_c5 = h_f, c5_f

    best_h = project_box_simplex(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
