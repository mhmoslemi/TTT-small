def compute_c5(h):

def project_simplex_box(h, target_sum):
    """Project h onto {x : 0 <= x[i] <= 1, sum(x) = target_sum}"""
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid

def smooth_max(corr, alpha):
    m = np.max(corr)
    if alpha < 1:
        return m
    shifted = corr - m
    # Use log-sum-exp for stability
    max_shifted = np.max(shifted)
    log_sum = max_shifted + np.log(np.sum(np.exp(shifted - max_shifted)))
    return m + log_sum / alpha

def smooth_max_grad(corr, alpha):
    """Return (value, weights) where weights sum to 1 and are concentrated on max"""
    m = np.max(corr)
    if alpha < 1:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
        return m, w
    shifted = alpha * (corr - m)
    max_shifted = np.max(shifted)
    exp_shifted = np.exp(shifted - max_shifted)
    w = exp_shifted / np.sum(exp_shifted)
    value = m + np.log(np.sum(exp_shifted)) / alpha
    return value, w

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_simplex_box(h, target_sum):
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def smooth_max_weights(corr, alpha):
    m = np.max(corr)
    shifted = alpha * (corr - m)
    max_s = np.max(shifted)
    exp_s = np.exp(shifted - max_s)
    w = exp_s / np.sum(exp_s)
    return w

def grad_corr(h, k, n):
    g = np.zeros(n)
    if k >= 0:
        g[:n - k] += (1.0 - h[k:])
        if k > 0:
            g[k:] -= h[:n - k]
    else:
        p = -k
        g[p:] += (1.0 - h[:n - p])
        g[:n - p] -= h[p:]
    return g

def grad_smooth_max(h, alpha, n_max=40):
    n = len(h)
    corr = compute_corr(h)
    w = smooth_max_weights(corr, alpha)
    g = np.zeros(n)
    active = np.where(w > 1e-7)[0]
    if len(active) == 0:
        active = np.array([np.argmax(w)])
    active = active[np.argsort(-w[active])[:n_max]]
    for idx in active:
        k = idx - (n - 1)
        g += w[idx] * grad_corr(h, k, n)
    return g * (2.0 / n)

def optimize_gradient(h0, target_sum, deadline, rng):
    n = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    lr = 0.02
    momentum = np.zeros(n)
    beta = 0.85
    alpha = 30.0
    stall = 0
    total_stall = 0

    while time.time() < deadline:
        g = grad_smooth_max(best_h, alpha)
        momentum = beta * momentum - lr * g
        h_new = best_h + momentum
        h_new = project_simplex_box(h_new, target_sum)
        new_c5 = compute_c5(h_new)

        if new_c5 < best_c5 - 1e-12:
            best_h = h_new
            best_c5 = new_c5
            stall = 0
            if alpha < 200:
                alpha *= 1.03
        else:
            # Try without momentum
            h_new2 = best_h - lr * g
            h_new2 = project_simplex_box(h_new2, target_sum)
            new_c52 = compute_c5(h_new2)
            if new_c52 < best_c5 - 1e-12:
                best_h = h_new2
                best_c5 = new_c52
                stall = 0
                momentum *= 0.5
            else:
                stall += 1
                total_stall += 1
                lr *= 0.88
                if stall > 20:
                    # Structured perturbation
                    size = 0.03 + 0.01 * min(total_stall / 50.0, 3.0)
                    h_pert = best_h + rng.randn(n) * size
                    h_pert = project_simplex_box(h_pert, target_sum)
                    c5_pert = compute_c5(h_pert)
                    if c5_pert < best_c5:
                        best_h = h_pert
                        best_c5 = c5_pert
                    alpha = min(alpha * 1.4, 400.0)
                    lr = 0.02
                    momentum = np.zeros(n)
                    stall = 0
                    if alpha > 350 and total_stall > 200:
                        break

    return best_h, best_c5

def binary_local_search(h_cont, max_rounds=300, rng=None):
    n = len(h_cont)
    if rng is None:
        rng = np.random.RandomState(0)
    h = (h_cont > 0.5).astype(float)
    target_ones = n // 2
    n_ones = int(np.sum(h))
    if n_ones > target_ones:
        ones_idx = np.where(h == 1)[0]
        order = ones_idx[np.argsort(h_cont[ones_idx])]
        h[order[:n_ones - target_ones]] = 0
    elif n_ones < target_ones:
        zeros_idx = np.where(h == 0)[0]
        order = zeros_idx[np.argsort(-h_cont[zeros_idx])]
        h[order[:target_ones - n_ones]] = 1

    corr = compute_corr(h)
    best_max = np.max(corr)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]

    for _ in range(max_rounds):
        improved = False
        for attempt in range(80):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0.0
            h[j] = 1.0
            new_corr = compute_corr(h)
            new_max = np.max(new_corr)
            if new_max < best_max - 1e-10:
                best_max = new_max
                corr = new_corr
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                improved = True
                break
            else:
                h[i] = 1.0
                h[j] = 0.0
        if not improved:
            # Try accepting equal solutions to escape plateaus
            for _ in range(20):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h[i] = 0.0
                h[j] = 1.0
                new_max = np.max(compute_corr(h))
                if new_max <= best_max:
                    best_max = min(best_max, new_max)
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                    improved = True
                    break
                else:
                    h[i] = 1.0
                    h[j] = 0.0
            if not improved:
                break
    return h, best_max * (2.0 / n)

def interpolate_to_n(h_old, n_old, n_new):
    x_old = (np.arange(n_old) + 0.5) / n_old * 2.0
    x_new = (np.arange(n_new) + 0.5) / n_new * 2.0
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        n0 = 200
        h0 = np.clip(0.5 + 0.1 * rng.randn(n0), 0, 1)
        h0 = project_simplex_box(h0, n0 / 2.0)

    n0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    best_n = n0

    # Phase 1: Gradient optimization at initial resolution
    deadline1 = t0 + budget_s * 0.25
    best_h, c5_1 = optimize_gradient(best_h, best_n / 2.0, deadline1, rng)
    if c5_1 < best_c5:
        best_c5 = c5_1

    # Phase 2: Binary local search from continuous solution
    elapsed = time.time() - t0
    if budget_s - elapsed > 80:
        h_bin, c5_bin = binary_local_search(best_h, max_rounds=500, rng=rng)
        if c5_bin < best_c5:
            best_h = h_bin
            best_c5 = c5_bin

    # Phase 3: Multi-resolution exploration
    elapsed = time.time() - t0
    remaining = budget_s - elapsed
    resolutions = [150, 250, 300, 400]
    for n_try in resolutions:
        if remaining < 80:
            break
        h_interp = interpolate_to_n(best_h, best_n, n_try)
        h_interp = project_simplex_box(h_interp, n_try / 2.0)
        phase_dead = time.time() + min(remaining * 0.22, 180.0)
        h_opt, c5_opt = optimize_gradient(h_interp, n_try / 2.0, phase_dead, rng)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
            best_n = n_try
        # Also try binary search at this resolution
        if time.time() < t0 + budget_s - 60:
            h_b, c5_b = binary_local_search(h_opt, max_rounds=200, rng=rng)
            if c5_b < best_c5:
                best_h = h_b
                best_c5 = c5_b
                best_n = n_try
        elapsed = time.time() - t0
        remaining = budget_s - elapsed

    # Phase 4: Restart from perturbed best
    elapsed = time.time() - t0
    if budget_s - elapsed > 60:
        h_pert = best_h + rng.randn(best_n) * 0.04
        h_pert = project_simplex_box(h_pert, best_n / 2.0)
        deadline4 = t0 + budget_s - 20
        h_opt, c5_opt = optimize_gradient(h_pert, best_n / 2.0, deadline4, rng)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt

    # Phase 5: Final polish
    remaining = budget_s - (time.time() - t0)
    if remaining > 15:
        h_opt, c5_opt = optimize_gradient(best_h, best_n / 2.0, t0 + budget_s - 3, rng)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt

    best_h = project_simplex_box(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
