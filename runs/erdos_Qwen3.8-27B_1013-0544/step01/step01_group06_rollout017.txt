    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = 0.5 * np.ones(200)

    target_sum = len(best_h) / 2.0
    best_h = project_box_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    # Phase 1: Main Adam optimization with low initial alpha
    d1 = t0 + budget_s * 0.38
    best_h, best_c5 = optimize_adam(best_h, target_sum, d1, rng, alpha_init=18)

    # Phase 2: Downscale-upscale to escape local minimum
    remaining = budget_s - (time.time() - t0)
    if remaining > 120:
        n_low = max(50, len(best_h) // 4)
        h_low = interpolate_to_n(best_h, len(best_h), n_low)
        h_low = project_box_sum(h_low, n_low / 2.0)
        d2a = time.time() + min(remaining * 0.2, 130)
        h_low_opt, _ = optimize_adam(h_low, n_low / 2.0, d2a, rng, alpha_init=12)
        h_up = interpolate_to_n(h_low_opt, n_low, len(best_h))
        h_up = project_box_sum(h_up, target_sum)
        d2b = time.time() + min(remaining * 0.15, 100)
        h_up_opt, c5_up = optimize_adam(h_up, target_sum, d2b, rng, alpha_init=14)
        if c5_up < best_c5:
            best_h, best_c5 = h_up_opt, c5_up

    # Phase 3: Upscale to 2x resolution for finer control
    remaining = budget_s - (time.time() - t0)
    if remaining > 80:
        n_new = len(best_h) * 2
        h_new = interpolate_to_n(best_h, len(best_h), n_new)
        h_new = project_box_sum(h_new, n_new / 2.0)
        d3 = time.time() + min(remaining * 0.55, 280)
        h3, c5_3 = optimize_adam(h_new, n_new / 2.0, d3, rng, alpha_init=14)
        if c5_3 < best_c5:
            best_h, best_c5 = h3, c5_3

    # Phase 4: Final polish at best resolution
    remaining = budget_s - (time.time() - t0)
    if remaining > 20:
        best_h, best_c5 = optimize_adam(best_h, len(best_h) / 2.0,
                                         t0 + budget_s - 3, rng, alpha_init=10)

    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
