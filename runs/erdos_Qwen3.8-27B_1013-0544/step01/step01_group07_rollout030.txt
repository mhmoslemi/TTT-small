    return best_local

def targeted_coord_descent(h, dx, target_sum, n_rounds, t):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best, dx)
    for rnd in range(n_rounds):
        corr = np.correlate(best, 1 - best, mode="full")
        center = n - 1
        top_lags = np.argsort(corr)[-5:]
        influence = np.zeros(n)
        for idx in top_lags:
            k = idx - center
            w = corr[idx]
            if k >= 0:
                influence[:n-k] += w
                influence[k:] += w
            else:
                d = -k
                influence[d:] += w
                influence[:n-d] += w
        probs = influence + 0.01
        probs /= probs.sum()
        group_size = 6
        idx_sel = np.random.choice(n, group_size, replace=False, p=probs)
        sub_h = best[idx_sel].copy()
        fixed_sum = target_sum - np.sum(best) + np.sum(sub_h)
        def sub_obj(x):
            h_sub = best.copy()
            h_sub[idx_sel] = x
            return smooth_max_obj(h_sub, t, dx)
        try:
            res = minimize(sub_obj, sub_h.copy(), method='SLSQP',
                          bounds=[(0.0, 1.0)] * group_size,
                          constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - fixed_sum}],
                          options={'maxiter': 40, 'ftol': 1e-10})
            h_new = best.copy()
            h_new[idx_sel] = np.clip(res.x, 0.0, 1.0)
            h_new = fix_sum(h_new, target_sum)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best = h_new
                best_c5 = c5_new
        except Exception:
            pass
    return best, best_c5

def make_thue_morse(n):
    t = np.zeros(n)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    return t

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = fix_sum(np.clip(h0, 0.0, 1.0), target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)
    best_n = n

    # Phase 1: Subgradient with momentum (adaptive temp)
    for t in [3, 5, 10, 20, 40, 80]:
        if elapsed() > budget_s * 0.30:
            break
        h_sg = subgradient_descent(best_h, dx, target_sum, t,
                                   n_iter=300, alpha_init=0.3 / t)
        c5_sg = compute_c5(h_sg, dx)
        if c5_sg < best_c5:
            best_h = h_sg
            best_c5 = c5_sg

    # Phase 2: SLSQP polish with increasing smoothness
    for t in [5, 15, 30, 60, 120, 250]:
        if elapsed() > budget_s * 0.55:
            break
        h_new = slsqp_step(best_h, dx, target_sum, t, maxiter=400)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 3: Targeted coordinate descent
    if elapsed() < budget_s * 0.75:
        n_rounds = 80
        h_new, c5_new = targeted_coord_descent(best_h, dx, target_sum,
                                               n_rounds=n_rounds, t=40)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 4: Try Thue-Morse and other structured binaries
    if elapsed() < budget_s * 0.88:
        candidates = []
        # Thue-Morse
        tm = make_thue_morse(n)
        candidates.append(tm)
        # Negated Thue-Morse
        candidates.append(1.0 - tm)
        # Random binaries (a few)
        for trial in range(8):
            rng = np.random.RandomState(seed + trial * 131)
            ones = rng.choice(n, n // 2, replace=False)
            h_bin = np.zeros(n)
            h_bin[ones] = 1.0
            candidates.append(h_bin)

        for h_cand in candidates:
            if elapsed() > budget_s * 0.88:
                break
            h_cand = fix_sum(np.clip(h_cand, 0.0, 1.0), target_sum)
            h_opt = slsqp_step(h_cand, dx, target_sum, t=15, maxiter=300)
            c5_opt = compute_c5(h_opt, dx)
            if c5_opt < best_c5:
                h_opt2 = slsqp_step(h_opt, dx, target_sum, t=80, maxiter=400)
                c5_opt2 = compute_c5(h_opt2, dx)
                if c5_opt2 < best_c5:
                    best_h = h_opt2
                    best_c5 = c5_opt2
                else:
                    best_h = h_opt
                    best_c5 = c5_opt

    # Phase 5: Final high-temperature polish
    if elapsed() < budget_s - 15:
        for t in [200, 500, 1000]:
            if elapsed() > budget_s - 15:
                break
            h_new = slsqp_step(best_h, dx, target_sum, t, maxiter=2000)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    # Phase 6: One more subgradient pass on final result
    if elapsed() < budget_s - 5:
        h_sg = subgradient_descent(best_h, dx, target_sum, t=50,
                                   n_iter=200, alpha_init=0.05)
        h_sg = slsqp_step(h_sg, dx, target_sum, t=200, maxiter=1000)
        c5_sg = compute_c5(h_sg, dx)
        if c5_sg < best_c5:
            best_h = h_sg
            best_c5 = c5_sg

    best_h = fix_sum(np.clip(best_h, 0.0, 1.0), target_sum)
    best_c5 = compute_c5(best_h, dx)
    return best_h, best_c5, best_n
