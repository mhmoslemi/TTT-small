            ones[oi] = i

            h[j] = 1.0

    best_c5 = compute_c5(best_h, dx)

    def rem():
        return budget_s - (time.time() - t0)

    # Phase 1: SLSQP polish from initial
    h1 = slsqp(best_h, n, dx, 300)
    c51 = compute_c5(h1, dx)
    if c51 < best_c5:
        best_h, best_c5 = h1, c51

    # Phase 2: SA from diverse binary starts
    starts = [to_bin(best_h, n)]
    for _ in range(4):
        hb = np.zeros(n)
        hb[rng.permutation(n)[:n // 2]] = 1.0
        starts.append(hb)

    for hs in starts:
        if rem() < 40:
            break
        alloc = min(rem() * 0.10, 80.0)
        n_it = max(int(alloc * 55000), 10000)
        hb, c5 = sa_bin(hs, n, dx, rng, n_it)
        if c5 < best_c5:
            best_c5 = c5
            hp = slsqp(hb, n, dx, 300)
            c5p = compute_c5(hp, dx)
            if c5p < best_c5:
                best_h, best_c5 = hp, c5p
            else:
                best_h = hb

    # Phase 3: L-BFGS-B continuous refinement
    if rem() > 20:
        h3 = lfb_opt(best_h, n, dx, 300, lam=200.0)
        c53 = compute_c5(h3, dx)
        if c53 < best_c5:
            best_h, best_c5 = h3, c53

    # Phase 4: Iterated SA + ILS + polish cycles
    cycle = 0
    while rem() > 35:
        cycle += 1
        hb = to_bin(best_h, n)
        progress = (budget_s - rem()) / budget_s
        T0_val = 0.006 * (1.0 - 0.7 * progress) + 0.0005
        T1_val = 0.00001
        alloc = min(rem() * 0.4, 180.0)
        n_it = max(int(alloc * 55000), 10000)
        hsa, c5s = sa_bin(hb, n, dx, rng, n_it, T0_val, T1_val)
        improved = False
        if c5s < best_c5:
            best_c5 = c5s
            improved = True
            hp = slsqp(hsa, n, dx, 400)
            c5p = compute_c5(hp, dx)
            if c5p < best_c5:
                best_h, best_c5 = hp, c5p
            else:
                best_h = hsa
        # Try ILS every other cycle
        if not improved and cycle % 2 == 1 and rem() > 50:
            n_outer = max(int(min(rem() * 0.2, 60.0) * 50), 50)
            h_ils, c5_ils = ilsa(hb, n, dx, rng, n_outer)
            if c5_ils < best_c5:
                best_c5 = c5_ils
                hp = slsqp(h_ils, n, dx, 300)
                c5p = compute_c5(hp, dx)
                if c5p < best_c5:
                    best_h, best_c5 = hp, c5p
                else:
                    best_h = h_ils
        if not improved and rem() > 20 and cycle % 3 == 0:
            hl = lfb_opt(best_h, n, dx, 200, lam=200.0)
            c5l = compute_c5(hl, dx)
            if c5l < best_c5:
                best_h, best_c5 = hl, c5l

    return (best_h, best_c5, n)
