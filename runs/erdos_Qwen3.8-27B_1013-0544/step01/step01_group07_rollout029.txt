import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def lse_obj(h, t, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    m = corr.max()
    return (m + np.log(np.sum(np.exp((corr - m) / t))) * t) * dx

def lse_grad(h, t, dx):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    m = corr.max()
    w = np.exp((corr - m) / t)
    w /= w.sum()
    grad = np.zeros(n)
    lags = np.arange(2 * n - 1) - (n - 1)
    for j in range(2 * n - 1):
        wj = w[j]
        if wj < 1e-12:
            continue
        lag = lags[j]
        lo1 = max(0, lag)
        hi1 = min(n, n + lag)
        if hi1 > lo1:
            grad[lo1:hi1] += wj * (1.0 - h[lo1 - lag:hi1 - lag])
        lo2 = max(0, -lag)
        hi2 = min(n, n - lag)
        if hi2 > lo2:
            grad[lo2:hi2] -= wj * h[lo2 + lag:hi2 + lag]
    return grad * dx

def project_feasible(v, target_sum):
    n = len(v)
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        p = np.clip(v - mid, 0.0, 1.0)
        if p.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    p = np.clip(v - mid, 0.0, 1.0)
    diff = target_sum - p.sum()
    if abs(diff) > 1e-14:
        free = (p > 1e-12) & (p < 1.0 - 1e-12)
        if free.sum() > 0:
            p[free] += diff / free.sum()
            p = np.clip(p, 0.0, 1.0)
    return p

def pg_optimize(h, t, dx, target_sum, n_iters, lr0, momentum, lr_decay):
    n = len(h)
    h_cur = h.copy()
    v = np.zeros(n)
    lr = lr0
    best_local = h_cur.copy()
    best_c5 = compute_c5(h_cur, dx)
    for it in range(n_iters):
        h_la = np.clip(h_cur + momentum * v, 0.0, 1.0)
        g = lse_grad(h_la, t, dx)
        v = momentum * v - lr * g
        h_new = project_feasible(h_cur + v, target_sum)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_local = h_new.copy()
        h_cur = h_new
        lr *= lr_decay
    return best_local, best_c5

def slsqp_refine(h, t, dx, target_sum, maxiter=500):
    n = len(h)
    try:
        res = minimize(
            lambda x: lse_obj(x, t, dx),
            h.copy(),
            jac=lambda x: lse_grad(x, t, dx),
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = project_feasible(res.x, target_sum)
        return h_new, compute_c5(h_new, dx)
    except Exception:
        return h, compute_c5(h, dx)
