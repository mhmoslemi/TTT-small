    n = len(h)

def compute_c5(h, dx):

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):

        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target_sum:
            lo = mid
        else:

    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")

    shifted = t * (corr - m)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    w = exp_s / sum_exp  # shape (2n-1,)
    
    val = (m + (1.0/t) * np.log(sum_exp))

    # A[i,j] = w[i-j+n-1] (Toeplitz, first column is w[n-1], w[n-2], ..., w[0])

    #   = w[n-1], w[n-2], ..., w[0]

    #   = w[n-1], w[n], ..., w[2n-2]
    col_A = w[n-1::-1]  # w[n-1], w[n-2], ..., w[0]

    for _ in range(80):

        s = np.sum(np.clip(h + mid, 0.0, 1.0))

    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    shifted = t * (corr - m)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    w = exp_s / sum_exp
    
    val = m + (1.0 / t) * np.log(sum_exp)
    
    # Build Toeplitz matrix D where D[i,j] = w[i-j+n-1]
    # First column (j=0): D[i,0] = w[i+n-1] for i=0..n-1
    # First row (i=0): D[0,j] = w[n-1-j] for j=0..n-1
    col_D = w[n-1:2*n-1]  # w[n-1], w[n], ..., w[2n-2]
    row_D = w[n-1::-1]    # w[n-1], w[n-2], ..., w[0]
    
    # Build D as Toeplitz
    D = np.zeros((n, n))
    for i in range(n):

            D[i, j] = w[i - j + n - 1]

    Dh = D @ h
    DTh = D.T @ (1.0 - h)
    grad = (DTh - Dh) / t
    
    return val, grad

from scipy.optimize import minimize

def compute_c5(h, dx):

def project_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def smax_value_and_grad(h, t):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    shifted = t * (corr - m)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    w = exp_s / sum_exp
    val = m + (1.0 / t) * np.log(sum_exp)
    ii = np.arange(n)[:, None]
    jj = np.arange(n)[None, :]
    D = w[ii - jj + n - 1]
    Dh = D @ h
    DTh = D.T @ (1.0 - h)
    grad = (DTh - Dh) / t
    return val, grad

    n = len(h)
    h = project_feasible(h.copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    val, grad = smax_value_and_grad(h, t)
    lr = lr_init
    prev_h = None

            val_trial, _ = smax_value_and_grad(h_trial, t)

    n = len(h)
    def obj(x):
        val, _ = smax_value_and_grad(x, t)
        return val
    def constr(x):
        return np.sum(x) - target_sum
    def constr_jac(x):
        return np.ones(n)
    cons = {'type': 'eq', 'fun': constr, 'jac': constr_jac}
    try:
        res = minimize(obj, h.copy(), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n, constraints=[cons],
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_feasible(res.x, target_sum)
    except Exception:
        return h

    x_new = np.linspace(0, 2, new_n + 1)[:-1]
    h_new = np.interp(x_new, x_old, h)
    return h_new
