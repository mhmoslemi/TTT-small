    return float(m + np.log(np.sum(np.exp(shifted))))) / 1.0

def sum_con(h, n):
    return np.sum(h) - n / 2.0

def sum_con_jac(h, n):
    return np.ones(n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - t0)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 80
        h0 = np.full(n, 0.5)
    h0 = project_feasible(h0, n)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h, n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]

    # Phase 1: Continuation SLSQP
    alphas = [20, 50, 100, 200, 500, 1000, 2000, 5000]
    for alpha in alphas:
        if remaining() < 20:
            break
        for attempt in range(2):
            if remaining() < 15:
                break
            try:
                start = best_h.copy()
                if attempt == 1:
                    start = project_feasible(start + rng.normal(0, 0.03, n), n)
                res = minimize(obj_smooth, start, args=(n, alpha), method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 2000, 'ftol': 1e-16})
                h_try = project_feasible(res.x, n)
                c = compute_c5(h_try, n)
                if c < best_c5 - 1e-14:
                    best_h, best_c5 = h_try, c
            except Exception:
                pass

    # Phase 2: Active-set focused refinement
    for _ in range(4):
        if remaining() < 30:
            break
        c_all = compute_all_c(best_h, n)
        c_max_val = c_all.max()
        thresh = c_max_val * 0.97
        active = np.where(c_all >= thresh)[0]
        if len(active) < 3:
            active = np.argsort(c_all)[-8:]
        elif len(active) > 35:
            active = np.argsort(c_all)[-35:]
        for alpha in [500, 2000, 8000]:
            if remaining() < 15:
                break
            try:
                res = minimize(obj_focused, best_h,
                              args=(n, tuple(active), alpha), method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 1500, 'ftol': 1e-15})
                h_try = project_feasible(res.x, n)
                c = compute_c5(h_try, n)
                if c < best_c5 - 1e-14:
                    best_h, best_c5 = h_try, c
            except Exception:
                pass

    # Phase 3: Null-space perturbations
    if remaining() > 80:
        for _ in range(8):
            if remaining() < 50:
                break
            d = rng.normal(0, 1, n)
            d -= d.mean()
            norm = np.linalg.norm(d)
            if norm > 0:
                d = d / norm
            for scale in [0.1, 0.2, 0.4]:
                if remaining() < 30:
                    break
                h_pert = project_feasible(best_h + scale * d, n)
                try:
                    res = minimize(obj_smooth, h_pert, args=(n, 3000), method='SLSQP',
                                  bounds=bounds, constraints=cons,
                                  options={'maxiter': 1000, 'ftol': 1e-14})
                    h_try = project_feasible(res.x, n)
                    c = compute_c5(h_try, n)
                    if c < best_c5 - 1e-14:
                        best_h, best_c5 = h_try, c
                except Exception:
                    pass

    # Phase 4: Coordinate descent with dense evaluation
    if remaining() > 60:
        for _ in range(8):
            if remaining() < 40:
                break
            for i in range(n):
                if remaining() < 40:
                    break
                orig = best_h[i]
                lo = max(0.0, orig - 0.4)
                hi = min(1.0, orig + 0.4)
                cands = np.linspace(lo, hi, 17)
                best_local = best_c5
                best_local_h = best_h.copy()
                for v in cands:
                    if abs(v - orig) < 0.01:
                        continue
                    test = best_h.copy()
                    test[i] = v
                    test = project_feasible(test, n)
                    c = compute_c5(test, n)
                    if c < best_local:
                        best_local = c
                        best_local_h = test.copy()
                if best_local < best_c5 - 1e-10:
                    best_h = best_local_h
                    best_c5 = best_local

    # Phase 5: Pairwise swaps with multiple scales
    if remaining() > 30:
        deltas = [0.001, 0.002, 0.005, 0.01, 0.02, 0.05]
        count = 0
        while remaining() > 30 and count < 60000:
            count += 1
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            for d in deltas:
                for s in [1, -1]:
                    di = s * d
                    hi = best_h[i] + di
                    hj = best_h[j] - di
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h_try = best_h.copy()
                        h_try[i] = hi
                        h_try[j] = hj
                        c = compute_c5(h_try, n)
                        if c < best_c5:
                            best_h = h_try
                            best_c5 = c
                            break
                else:
                    continue
                break

    # Final polish
    if remaining() > 8:
        for alpha in [10000, 30000, 100000]:
            if remaining() < 8:
                break
            try:
                res = minimize(obj_smooth, best_h, args=(n, alpha), method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 5000, 'ftol': 1e-17})
                h_try = project_feasible(res.x, n)
                c = compute_c5(h_try, n)
                if c < best_c5:
                    best_h, best_c5 = h_try, c
            except Exception:
                pass

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
