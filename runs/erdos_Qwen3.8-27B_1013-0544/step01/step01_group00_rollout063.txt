```python
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx  # length 2n-1
    
    # F = (sum corr^p)^(1/p)
    # dF/dh = (sum corr^p)^(1/p-1) * sum_k corr^(p-1) * dcorr_k/dh
    ps = np.sum(corr**p)
    if ps < 1e-30:
        return np.zeros(n)
```

```python
    """Project h onto {x : 0<=x<=1, sum(x)=target_sum}."""
```

4. **Nesterov acceleration**: Faster convergence for convex (or nearly convex) problems

5. **Two-scale optimization**: Optimize at very low resolution (e.g., n=20) to get the "shape", then progressively increase resolution.

6. **Different objective smoothing**: 
   - Instead of soft-max, use "top-k average": average of the k largest correlation values
   - This is a piecewise-linear function that's easier to handle

Let me try the L-BFGS-B approach with the reparameterization:

Define h = 0.5 + u, where u is free but we need sum(u) = 0 and |u| ≤ 0.5.
For the sum constraint, we can use: u = z - mean(z), where z is the free variable.
Then h = 0.5 + z - mean(z), sum(h) = n*0.5 + sum(z) - n*mean(z) = n*0.5. ✓
And h ∈ [0,1] ⟺ z - mean(z) ∈ [-0.5, 0.5]

The constraint |z - mean(z)| ≤ 0.5 is a bit awkward for L-BFGS-B (which needs bound constraints on each variable independently).

Alternative: just use the penalty method.
minimize soft_max(h) + λ * (sum(h) - n/2)^2
subject to 0 ≤ h ≤ 1

With L-BFGS-B, the bounds are handled natively, and the sum constraint is in the objective.

For large λ, this enforces the constraint strongly. We can increase λ over time.

Let me try this:

```python
def lbfgs_obj(h, temp, lam):
```

But L-BFGS-B uses numerical or analytical gradients. For the analytical gradient:
∇(soft_max + λ*(sum-n/2)^2) = ∇soft_max + 2λ*(sum-n/2)*1

The gradient of soft_max is what I computed above. The gradient of the penalty is just a constant vector. So:

```python
def lbfgs_grad(h, temp, lam):
    g = sm_grad(h, temp)
    g += 2.0 * lam * (np.sum(h) - len(h) * 0.5)
    return g
```

```python
def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_h(h, target_sum):
    h = h.copy()
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def sm_grad(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    w = np.exp((corr - m) / temp)
    w /= np.sum(w)
    grad = np.zeros(n)
    grad[:] = w[n-1] * dx * (1.0 - 2.0 * h)
    for L in range(1, n):
        wp = w[n-1+L]
        wm = w[n-1-L]
        s = wp + wm
        grad[L:] += dx * (wp - s * h[:n-L])
        grad[:n-L] += dx * (wm - s * h[L:])
    return grad

def sm_obj(h, temp):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    return m + temp * np.log(np.sum(np.exp((corr - m) / temp)))

    return sm_obj(h, temp) + lam * (np.sum(h) - len(h) * 0.5)**2

    g += 2.0 * lam * (np.sum(h) - len(h) * 0.5)
    return g
```
