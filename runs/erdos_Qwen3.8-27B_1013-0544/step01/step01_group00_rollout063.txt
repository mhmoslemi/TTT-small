from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_to_feasible(h, target_sum):
    """Project h to satisfy 0<=h<=1 and sum(h)=target_sum."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def softmax_obj_and_grad(h, T, lam):
    """Soft-max objective with penalty, returns (value, gradient)."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    
    # Soft-max value and gradient
    c_scaled = corr / T
    c_max = np.max(c_scaled)
    exp_vals = np.exp(c_scaled - c_max)
    p = exp_vals / np.sum(exp_vals)
    soft_val = T * (c_max + np.log(np.sum(exp_vals)))
    
    # Gradient of soft-max w.r.t. h
    v = 1.0 - h
    indices = np.arange(n)[:, None] + (n - 1) - np.arange(n)[None, :]
    mask = (indices >= 0) & (indices < 2 * n - 1)
    P = np.where(mask, p[indices], 0.0)
    grad_sm = dx * (P @ v) - dx * h * p[n - 1]
    
    # Penalty gradient
    diff = np.sum(h) - n * 0.5
    grad_pen = 2.0 * lam * diff * np.ones(n)
    
    val = soft_val + lam * diff * diff
    grad = grad_sm + grad_pen
    return val, grad

    bounds = [(0.0, 1.0)] * n

        return val, grad
    
    res = minimize(obj_and_grad, h0.copy(), method='L-BFGS-B',
                   jac=True, bounds=bounds,
                   options={'maxiter': max_iter, 'gtol': 1e-8, 'ftol': 1e-10})
    h_new = project_to_feasible(res.x, n * 0.5)
    return h_new

    p = exp_c / np.sum(exp_c)
    
    # grad[j] = sum_m p[m] * d(corr[m])/dh[j]
    # = dx * sum_m p[m] * (1-h[j-m+n-1]) * 1[0<=j-m+n-1<n]

def softmax_gradient(h, T):
    n = len(h)
    dx = 2.0 / n
    m_total = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode='full') * dx

    c -= np.max(c)
    exp_c = np.exp(c)
    p = exp_c / np.sum(exp_c)
    
    j_idx = np.arange(n)
    k_idx = np.arange(n)
    
    # First term: m = j + n - 1 - k
    m1 = j_idx[:, None] + (n - 1) - k_idx[None, :]
    mask1 = (m1 >= 0) & (m1 < m_total)

import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_feasible(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def sm_obj_grad(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    mt = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c = corr / T
    cm = np.max(c)
    ec = np.exp(c - cm)
    p = ec / np.sum(ec)
    val = T * (cm + np.log(np.sum(ec))) + lam * (np.sum(h) - n*0.5)**2
    
    j_idx = np.arange(n)
    k_idx = np.arange(n)
    m1 = j_idx[:, None] + (n-1) - k_idx[None, :]
    mask1 = (m1 >= 0) & (m1 < mt)
    m2 = k_idx[None, :] - j_idx[:, None] + (n-1)
    mask2 = (m2 >= 0) & (m2 < mt)
    
    v = 1.0 - h
    t1 = dx * (v[None, :] * np.where(mask1, p[m1], 0.0)).sum(axis=1)
    t2 = dx * (h[None, :] * np.where(mask2, p[m2], 0.0)).sum(axis=1)
    diff = np.sum(h) - n * 0.5
    grad = t1 - t2 + 2.0 * lam * diff
    
    return val, grad

def optimize_sm(h0, T, lam, max_iter=500):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    res = minimize(sm_obj_grad, h0.copy(), method='L-BFGS-B', jac=True,

def project_feasible(h, target):
    h = np.clip(h, 0.0, 1.0).copy()

        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break

            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

    m2 = j[None, :] - j[:, None] + (n - 1)
    mk2 = (m2 >= 0) & (m2 < mt)
