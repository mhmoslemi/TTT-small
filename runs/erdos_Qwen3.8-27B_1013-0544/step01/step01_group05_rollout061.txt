import numpy as np
from scipy.optimize import minimize
import time

_global_dx = 0.0
_global_target = 0.0

def project_feasible(x, target_sum):
    lo = float(x.min()) - 1.0
    hi = float(x.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = float(np.clip(x - mid, 0.0, 1.0).sum())
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def lse_grad(h, beta, dx):
    n = len(h)
    nh = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = float(np.max(corr))
    sh = beta * (corr - m)
    et = np.exp(sh)
    se = float(np.sum(et))
    w = et / se
    grad = np.zeros(n)
    omh = 1.0 - h
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-10:
            continue
        L = k - nh
        if L == 0:
            grad += wk * (omh - h)
        elif L > 0:
            grad[L:] += wk * omh[:n - L]
            grad[:n - L] -= wk * h[L:]
        else:
            L2 = -L
            grad[:n - L2] += wk * omh[L2:]
            grad[L2:] -= wk * h[:n - L2]
    grad *= dx
    obj = (m + np.log(se) / beta) * dx
    return obj, grad

def active_grad(h, dx, tol_frac=0.01):
    n = len(h)
    nh = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = float(np.max(corr))
    tol = max(0.01, cmax * tol_frac)
    act = np.where(corr >= cmax - tol)[0]
    if len(act) < 3:
        act = np.argsort(corr)[-3:]
    elif len(act) > 20:
        act = np.argsort(corr)[-20:]
    grad = np.zeros(n)
    omh = 1.0 - h
    for k in act:
        L = int(k) - nh
        if L == 0:
            grad += (omh - h)
        elif L > 0:
            grad[L:] += omh[:n - L]
            grad[:n - L] -= h[L:]
        else:
            L2 = -L
            grad[:n - L2] += omh[L2:]
            grad[L2:] -= h[:n - L2]
    grad /= float(len(act))
    grad *= dx
    return grad, cmax * dx

def _s_obj(x):
    return compute_c5(x, _global_dx)

def _s_eq(x):
    return np.sum(x) - _global_target

    mu = 0.6
    mom = np.zeros(n)
    no_improve = 0
    it = 0
    while time.time() < deadline - 250:

        beta = 5.0 * (80.0) ** p

        _, grad = lse_grad(h, beta, dx)
        mom = mu * mom - alpha * grad
        h_new = project_feasible(h + mom, target)
        c5 = compute_c5(h_new, dx)

        if c5 < best_c5:

            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        h = h_new

        if no_improve > 800:

            mom = np.zeros(n)
            alpha = 0.005
            no_improve = 0

            if alpha < 0.0005:
