import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(x, target_sum):
    lo = float(x.min()) - 1.0
    hi = float(x.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = float(np.clip(x - mid, 0.0, 1.0).sum())
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_grad_corr(h, L, n):
    omh = 1.0 - h
    grad = np.zeros(n)
    if L == 0:
        grad[:] = omh[:] - h[:]
    elif L > 0:
        grad[L:] += omh[:n - L]
        grad[:n - L] -= h[L:]
    else:
        L2 = -L
        grad[:n - L2] += omh[L2:]
        grad[L2:] -= h[:n - L2]
    return grad

def softmax_gradient(h, T, n):
    corr = compute_corr(h)
    max_c = float(np.max(corr))
    shifted = corr - max_c
    lw = shifted / T
    max_lw = float(np.max(lw))
    ew = np.exp(lw - max_lw)
    w = ew / np.sum(ew)
    n_half = n - 1
    grad = np.zeros(n)
    active = np.where(w > 1e-8)[0]
    for k in active:
        L = int(k) - n_half
        grad += w[k] * compute_grad_corr(h, L, n)
    return grad

def active_gradient(h, n, tol_frac=0.001, max_active=12):
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    tol = max(0.01, c_max * tol_frac)
    active = np.where(corr >= c_max - tol)[0]
    if len(active) < 1:
        active = np.array([int(np.argmax(corr))])
    elif len(active) > max_active:
        active = np.argsort(corr)[-max_active:]
    n_half = n - 1
    grad = np.zeros(n)
    for idx in active:
        L = int(idx) - n_half
        grad += compute_grad_corr(h, L, n)
    return grad / float(len(active))

def generate_start(n, rng, mode=0):
    if mode == 0:
        h = rng.uniform(0, 1, n)
    elif mode == 1:
        h = 0.5 + 0.3 * np.sin(2 * np.pi * rng.randint(1, 4) * np.arange(n) / n + rng.uniform(0, 6.28))
    elif mode == 2:
        h = np.zeros(n)
        h[rng.choice(n, n // 2, replace=False)] = 0.8
        h[rng.choice(n - n // 2, n - n // 2 - n // 2, replace=False)] = 0.2
    elif mode == 3:
        h = 0.5 + 0.15 * rng.normal(0, 1, n)
    else:
        h = np.concatenate([rng.uniform(0.3, 0.7, n // 2), rng.uniform(0.3, 0.7, n - n // 2)])
        h = h[rng.permutation(n)]
    return h

    rng = np.random.RandomState(seed)

        return (start_time + budget_s - 3.0) - time.time()

    except NameError:
        h_init = None

    n_options = [50, 60, 70, 80, 100]
    best_h = None
    best_c5 = float('inf')
    best_n = 70

    if h_init is not None:
        best_n = len(h_init)
        dx_init = 2.0 / best_n
        _tmp = project_feasible(h_init, best_n / 2.0)
        _c = compute_c5(_tmp, dx_init)
        if _c < best_c5:
            best_c5 = _c
            best_h = _tmp.copy()

    for n_points in n_options:

            break

        target_sum = n_points / 2.0
