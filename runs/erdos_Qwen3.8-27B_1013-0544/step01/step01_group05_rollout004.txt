import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(x, target_sum):
    lo = float(np.min(x)) - 1.5
    hi = float(np.max(x)) + 1.5
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x - mid, 0.0, 1.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_grad_lag(h, L):
    n = len(h)
    grad = np.zeros(n)
    if L == 0:
        grad[:] = (1.0 - h) - h
    elif L > 0:
        grad[L:] += (1.0 - h[:n - L])
        grad[:n - L] -= h[L:]
    else:
        L2 = -L
        grad[:n - L2] += (1.0 - h[L2:])
        grad[L2:] -= h[:n - L2]
    return grad

def active_set_grad(h, corr, n_half, tol_frac, max_active):
    n = len(h)
    c_max = float(np.max(corr))
    tol = max(0.02, c_max * tol_frac)
    active = np.where(corr >= c_max - tol)[0]
    if len(active) < 2:
        active = np.argsort(corr)[-2:]
    elif len(active) > max_active:
        active = np.argsort(corr)[-max_active:]
    grad = np.zeros(n)
    for idx in active:
        L = int(idx) - n_half
        grad += compute_grad_lag(h, L)
    grad /= float(len(active))
    return grad

def lse_obj(x, dx, beta, penalty, n_half, n_pts):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = beta * corr * dx
    m = float(np.max(vals))
    shifted = vals - m
    e = np.exp(shifted)
    s = float(np.sum(e))
    obj = (m + np.log(s)) / beta + penalty * (np.sum(h) - n_pts * 0.5) ** 2
    return obj

def lse_jac(x, dx, beta, penalty, n_half, n_pts):
    h = np.clip(x, 0.0, 1.0)
    n = n_pts
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = beta * corr * dx
    m = float(np.max(vals))
    shifted = vals - m
    e = np.exp(shifted)
    total = float(np.sum(e))
    w = e / total
    grad = np.zeros(n)
    for k in range(len(w)):
        if w[k] < 1e-10:
            continue
        L = k - n_half
        if L == 0:
            grad += w[k] * dx * ((1.0 - h) - h)
        elif L > 0:
            grad[L:] += w[k] * dx * (1.0 - h[:n - L])
            grad[:n - L] -= w[k] * dx * h[L:]
        else:
            L2 = -L
            grad[:n - L2] += w[k] * dx * (1.0 - h[L2:])
            grad[L2:] -= w[k] * dx * h[:n - L2]
    grad += 2.0 * penalty * (np.sum(h) - n_pts * 0.5)
    return grad
