from scipy.optimize import minimize

def project_feasible(x, target_sum):
    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(64):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(x - mid, 0.0, 1.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def lag_grad(h, L):
    n = len(h)
    if L == 0:
        return 1.0 - 2.0 * h
    grad = np.zeros(n)
    if L > 0:
        nv = n - L
        grad[:nv] += 1.0 - h[L:]
        grad[L:] -= h[:nv]
    else:
        L2 = -L
        nv = n - L2
        grad[L2:] += 1.0 - h[:nv]
        grad[:nv] -= h[L2:]
    return grad

def lse_obj_and_grad(h, T, n_half):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    c_max = np.max(corr)
    if T < 1e-6:
        # Use active-set mean
        tol = max(1e-4, c_max * 0.001)
        active = np.where(corr >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        obj = float(np.mean(corr[active]))
        g = np.zeros(n)
        for idx in active:
            g += lag_grad(h, int(idx) - n_half)
        g /= len(active)
        return obj, g
    ex = np.exp(T * (corr - c_max))
    w = ex / np.sum(ex)
    obj = float(c_max + np.log(np.sum(ex)) / T)
    g = np.zeros(n)
    for k in range(len(corr)):
        if w[k] > 1e-10:
            g += w[k] * lag_grad(h, k - n_half)
    return obj, g

def slsqp_smooth(h0, T, target_sum, n_half, maxiter=3000):
    n = len(h0)
    def obj(h):
        return lse_obj_and_grad(h, T, n_half)[0]
    def grad(h):
        return lse_obj_and_grad(h, T, n_half)[1]
    def eq(h):
        return np.sum(h) - target_sum
    res = minimize(obj, h0, jac=grad, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': eq}],
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    if res.success or res.x is not None:
        return project_feasible(res.x, target_sum)
    return h0

import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(x, target_sum):
    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(64):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.clip(x - mid, 0.0, 1.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def lag_grad(h, L):
    n = len(h)
    if L == 0:
        return 1.0 - 2.0 * h
    grad = np.zeros(n)
    if L > 0:
        nv = n - L
        grad[:nv] += 1.0 - h[L:]
        grad[L:] -= h[:nv]
    else:
        L2 = -L
        nv = n - L2
        grad[L2:] += 1.0 - h[:nv]
        grad[:nv] -= h[L2:]
    return grad

def lse_func(h, T, n_half):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    c_max = np.max(corr)
    if T < 1e-6:
        tol = max(1e-4, c_max * 0.002)
        active = np.where(corr >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        obj = float(np.mean(corr[active]))
        g = np.zeros(n)
        for idx in active:
            g += lag_grad(h, int(idx) - n_half)
        g /= len(active)
        return obj, g
    ex = np.exp(T * (corr - c_max))
    w = ex / np.sum(ex)
    obj = float(c_max + np.log(np.sum(ex)) / T)
    g = np.zeros(n)
    for k in range(len(corr)):
        if w[k] > 1e-12:
            g += w[k] * lag_grad(h, k - n_half)
    return obj, g

def slsqp_smooth(h0, T, target_sum, n_half, maxiter=3000):
    n = len(h0)
    def obj(h):
        return lse_func(h, T, n_half)[0]
    def grad(h):
        return lse_func(h, T, n_half)[1]
    def eq(h):
        return np.sum(h) - target_sum
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': eq}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None:
            return project_feasible(res.x, target_sum)
    except Exception:
        pass
    return h0

def active_set_step(h, target_sum, n_half, ss, mu, top_k, momentum):
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cm = float(np.max(corr))
    tol = max(1e-5, cm * 0.001)
    act = np.where(corr >= cm - tol)[0]
    if len(act) < 3:
        act = np.argsort(corr)[-3:]
    elif len(act) > top_k:
        act = np.argsort(corr)[-top_k:]
    g = np.zeros(len(h))
    for idx in act:
        g += lag_grad(h, int(idx) - n_half)
    g /= len(act)
    momentum[:] = mu * momentum - ss * g
    h_new = project_feasible(h + momentum, target_sum)
    return h_new, momentum
