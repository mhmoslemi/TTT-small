        # Second term: -h[p+k] for p in [max(0,-k), min(n-1, n-1-k)]
        # i.e., p+k in [0, n-1], so p in [max(0,-k), min(n-1, n-1-k)]
        
        # First contribution: w[j]*dx*(1-h[p-k]) for valid p
        p_start1 = max(0, k)
        p_end1 = min(n, n + k)  # exclusive
        if p_end1 > p_start1:
            grad[p_start1:p_end1] += w[j] * dx * (1 - h[p_start1-k:p_end1-k])
        
        # Second contribution: -w[j]*dx*h[p+k] for valid p
        p_start2 = max(0, -k)
        p_end2 = min(n, n - k)  # exclusive
        if p_end2 > p_start2:
            grad[p_start2:p_end2] += -w[j] * dx * h[p_start2+k:p_end2+k]
    
    return grad, m

def project_h(h, target):
    """Project h to [0,1]^n with sum = target."""
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:

        cnt = np.sum(mask)
        if cnt == 0:
            break
        adj = min(diff / cnt, 0.5)  # Don't overshoot
        if diff > 0:
            adj = min(diff / cnt, 1.0 - np.max(h[mask]))
        else:
            adj = max(diff / cnt, -np.min(h[mask]))
        h[mask] += adj
        h = np.clip(h, 0.0, 1.0)
    return h

def exact_subgradient(h, temp=0.001):
    """Compute exact subgradient of max(corr) w.r.t. h using softmax."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    w = np.exp((corr - m) / temp)
    w /= np.sum(w)

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))


def project_h(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def exact_subgradient(h, temp, top_k=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    if top_k is not None and top_k < len(corr):
        idx = np.argpartition(corr, -top_k)[-top_k:]
        w = np.zeros(len(corr))
        w[idx] = np.exp((corr[idx] - m) / temp)
        w[idx] /= np.sum(w[idx])
    else:
        w = np.exp((corr - m) / temp)
        w /= np.sum(w)
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        if w[j] < 1e-12:
            continue
        k = j - (n - 1)
        lo1 = max(0, k)
        hi1 = min(n, n + k)
        if hi1 > lo1:
            grad[lo1:hi1] += w[j] * dx * (1.0 - h[lo1 - k:hi1 - k])
        lo2 = max(0, -k)
        hi2 = min(n, n - k)
        if hi2 > lo2:
            grad[lo2:hi2] += -w[j] * dx * h[lo2 + k:hi2 + k]
    return grad, m

    return v[-1]


def t_ineq(v):
    n = len(v) - 1
    h = v[:n]
    t = v[-1]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return t - corr

    return np.sum(v[:-1]) - (len(v) - 1) * 0.5


def t_eq_jac(v):
    n = len(v) - 1
    j = np.zeros(len(v))
    j[:n] = 1.0
    return j


def slsqp_t_var(h0, max_iter=500, t_margin=0.01, ftol=1e-13):
    n = len(h0)
    c5 = compute_c5(h0)
    v0 = np.append(h0.copy(), c5 + t_margin)
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.5)]
    cons = [
        {'type': 'ineq', 'fun': t_ineq},
        {'type': 'eq', 'fun': t_eq, 'jac': t_eq_jac}
    ]
    res = minimize(t_obj, v0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': ftol})
    h_new = project_h(res.x[:n], n * 0.5)
    return h_new, compute_c5(h_new)


def nesterov_subgrad(h0, iters, alpha0, decay, temp, top_k, momentum0, momentum_decay):
    n = len(h0)
    h = h0.copy()
    y = h0.copy()
    v = np.zeros(n)
    momentum = momentum0
    alpha = alpha0
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    for it in range(iters):
        grad, m = exact_subgradient(y, temp, top_k)
        v = momentum * v - alpha * grad
        h_new = project_h(y + v, n * 0.5)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        mu = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * momentum * momentum))
        y = h_new + (momentum / mu) * (h_new - h)
        h = h_new
        alpha *= decay
        momentum = momentum0 * (momentum_decay ** it)
    return best_h, best_c5
