            grad[:n-lag] += (1 - h[lag:n])  # h[j+lag] for j=0..n-1-lag → h[lag..n-1]
        if lag < n:
            grad[lag:] += (-h[:n-lag])  # -h[j-lag] for j=lag..n-1 → -h[0..n-1-lag]
    else:
        k = -lag
        # val = sum_{i=k}^{n-1} h[i-k] * (1 - h[i])
        # d/dh[j]:
        #   j in [0, n-1-k]: (1 - h[j+k])
        #   j in [k, n-1]: -h[j-k]
        if n - k > 0:
            grad[:n-k] += (1 - h[k:])  # (1-h[j+k]) for j=0..n-1-k → (1-h[k..n-1])
        if k < n:
            grad[k:] += (-h[:n-k])  # -h[j-k] for j=k..n-1 → -h[0..n-1-k]
    
    return grad, corr, max_idx

def projected_subgradient(h, target, max_iters=500, initial_step=0.01):

    n = len(h)
    dx = 2.0 / n
    h = np.array(h, dtype=float).copy()

    best_h = h.copy()
    best_val = float(np.max(np.correlate(h, 1-h, mode="full")) * dx)
    
    step = initial_step
    stall = 0
    
    for it in range(max_iters):
        grad, corr, _ = compute_max_corr_gradient(h)
        val = float(np.max(corr) * dx)
        
        if val < best_val - 1e-8:
            best_val = val
            best_h = h.copy()

        else:
            stall += 1
        
        if stall > 50:
            step *= 0.5
            if step < 1e-6:

        alpha = step

            h_new = fix_sum(h_new, target)

            if new_val < val + 1e-12:
                break
            alpha *= 0.5
        
        h = h_new
        
    return best_h, best_val

def slsqp_optimize(h_start, n_pts, tau, maxiter=200):
    """SLSQP optimization with smooth max."""
    target = n_pts / 2.0
    dx = 2.0 / n_pts
    
    def smooth_max(corr, tau):
        x = tau * corr * dx  # scale by dx to get the actual C5 values
        x_max = np.max(x)
        return float(x_max + np.log(np.sum(np.exp(x - x_max))) / tau)
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        return smooth_max(corr, tau)
    
    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        x_scaled = tau * corr * dx
        x_max = np.max(x_scaled)
        weights = np.exp(x_scaled - x_max)
        weights /= np.sum(weights)
        # d/dx of smooth_max = sum_k weights[k] * d(corr[k]*dx)/dx
        # d(corr[k])/dx[j] is the gradient of the k-th correlation w.r.t. x[j]
        # We only need the gradient of the weighted sum
        grad = np.zeros(n_pts)
        for k_idx in range(len(corr)):
            if weights[k_idx] < 1e-10:
                continue
            lag = k_idx - (n_pts - 1)
            if lag >= 0:
                grad[:n_pts-lag] += weights[k_idx] * dx * (1 - x[lag:n_pts])
                grad[lag:] += weights[k_idx] * dx * (-x[:n_pts-lag])
            else:
                kk = -lag
                grad[:n_pts-kk] += weights[k_idx] * dx * (1 - x[kk:])
                grad[kk:] += weights[k_idx] * dx * (-x[:n_pts-kk])
        return grad
    
    bnds = Bounds(np.zeros(n_pts), np.ones(n_pts))
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n_pts)}]
    
    try:
        res = minimize(obj, h_start, jac=obj_grad, method='SLSQP', bounds=bnds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
        x = fix_sum(res.x, target)
        c5 = evaluate_c5(x)
        return x, c5
    except Exception:
        return h_start, None

    best_h = h_init.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Phase 1: Projected gradient at initial resolution with momentum
    n = len(best_h)
    target = n / 2.0
    h_pg, c5_pg = projected_gradient_optimize(best_h, target, 3000, 5.0, 50.0, momentum=0.85)
    if c5_pg < best_c5:
        best_h = h_pg
        best_c5 = c5_pg
    
    # Phase 2: SLSQP polish
    if rem() > 20:
        h_new, c5_new = slsqp_polish(best_h, maxiter=300)
        if c5_new is not None and c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
    
    # Phase 3: Multi-resolution search
    resolutions = [60, 80, 100, 120, 140, 160, 200]
    for tn in resolutions:
        if rem() < 30:
            break
        if tn % 2 != 0:
            continue
        t_target = tn / 2.0
        h_t = np.interp(np.linspace(0, 1, tn), np.linspace(0, 1, len(best_h)), best_h)
        h_t = fix_sum(h_t, t_target)
        if t_target != int(t_target):
            continue
        
        h_pg, c5_pg = projected_gradient_optimize(h_t, t_target, 2000, 5.0, 40.0, momentum=0.85)
        if c5_pg < best_c5:
            best_h = h_pg
            best_c5 = c5_pg
        elif rem() > 15:
            h_new, c5_new = slsqp_polish(h_pg, maxiter=200)
            if c5_new is not None and c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
    
    # Phase 4: Binary and structured restarts
    hn = len(best_h)
    if rem() > 30:
        for trial in range(6):
            if rem() < 20:
                break
            if trial == 0:
                idx = np.argsort(best_h)
                h_bin = np.zeros(hn)
                h_bin[idx[hn // 2:]] = 1.0
            elif trial == 1:
                h_bin = make_periodic_binary(hn, 10)
            elif trial == 2:
                h_bin = make_periodic_binary(hn, 14)
            else:
                pos = np.random.choice(hn, hn // 2, replace=False)
                h_bin = np.zeros(hn)
                h_bin[pos] = 1.0
            
            h_pg, c5_pg = projected_gradient_optimize(h_bin, hn / 2.0, 2000, 10.0, 60.0, momentum=0.8)
            if c5_pg < best_c5:
                best_h = h_pg
                best_c5 = c5_pg
            if rem() > 15:
                start = h_pg if c5_pg < best_c5 else h_bin
                h_new, c5_new = slsqp_polish(start, maxiter=200)
                if c5_new is not None and c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new
    
    # Phase 5: Random perturbations from best
    for _ in range(15):
        if rem() < 15:
            break
        scale = 0.03 + 0.1 * np.random.random()
        hp = best_h + np.random.normal(0, scale, hn)
        hp = np.clip(hp, 0.0, 1.0)
        hp = fix_sum(hp, hn / 2.0)
        h_pg, c5_pg = projected_gradient_optimize(hp, hn / 2.0, 1500, 5.0, 50.0, momentum=0.85)
        if c5_pg < best_c5:
            best_h = h_pg
            best_c5 = c5_pg
        elif rem() > 15:
            h_new, c5_new = slsqp_polish(h_pg, maxiter=150)
            if c5_new is not None and c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
    
    # Phase 6: Final high-precision polish
    if rem() > 15:
        hn_f = len(best_h)
        h_new, c5_new = slsqp_polish(best_h, maxiter=400)
        if c5_new is not None and c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
    
    # Final: re-optimize at best resolution with more iterations
    if rem() > 20:
        hn_f = len(best_h)
        h_pg, c5_pg = projected_gradient_optimize(best_h, hn_f / 2.0, 2000, 2.0, 30.0, momentum=0.9)
        if c5_pg < best_c5:
            best_h = h_pg
            best_c5 = c5_pg
    
    # Final constraint fix
    hn_final = len(best_h)
    best_h = fix_sum(best_h, hn_final / 2.0)
    best_c5 = evaluate_c5(best_h)
    
    return best_h, best_c5, hn_final
