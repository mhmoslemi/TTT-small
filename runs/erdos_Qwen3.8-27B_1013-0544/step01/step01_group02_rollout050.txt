Since sum h[i] = n/2, this is n/2 - sum h[i]^2. To minimize this, we want sum h[i]^2 to be as large as possible, which means h should be as binary as possible (concentrated at 0 and 1).

For other lags, the situation is more complex. The optimal solution likely has values very close to 0 or 1, with a few values in between to handle the "boundary" effects.

1. Start with the initial solution

3. Optimize the binary pattern using local search (flip individual bits or pairs)
4. Then do a small continuous refinement around the best binary pattern

            h += (diff / total) * room

            if total < 1e-13:

    m = c.max()
    c_shifted = c - m
    return float(m + np.log(np.sum(np.exp(c_shifted))) / alpha)

    return np.sum(h) - n / 2.0

Standard autocorrelation: R(k) = sum h[i] * h[i+k]
Our correlation: C(k) = sum h[i] * (1-h[i+k]) = sum h[i] - sum h[i]*h[i+k]

For the range where both i and i+k are valid (i.e., the "central" part), C(k) = (number of ones in the overlap region) - R(k)

So minimizing C(k) is equivalent to maximizing R(k) for the "central" correlations. But for edge correlations (where the overlap region is smaller), it's different.

        c = compute_c5(h_try, n)
        if c < best_c - 1e-15:
            best = h_try
            best_c = c
    return best, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 80
        h0 = np.full(n, 0.5)

    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'args': (n,)}]

    def remaining():
        return budget_s - (time.time() - t0)

    # Phase 1: SLSQP continuation from initial
    alphas = [10, 30, 100, 300, 1000, 3000, 10000, 30000]
    for alpha in alphas:
        if remaining() < 20:
            break
        try:
            res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 5000, 'ftol': 1e-20})
            h_try = project_feasible(res.x, n)
            c = compute_c5(h_try, n)
            if c < best_c5 - 1e-15:
                best_h, best_c5 = h_try, c
        except Exception:
            pass

    # Phase 2: Binary pattern optimization
    if remaining() > 120:
        b = to_binary(best_h, n)
        b, c_bin = binary_flip_search(b, n, rng, remaining)
        if c_bin < best_c5:
            best_h = b.copy()
            best_c5 = c_bin
        for alpha in [5000, 20000, 50000]:
            if remaining() < 20:
                break
            try:
                res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 5000, 'ftol': 1e-20})
                h_try = project_feasible(res.x, n)
                c = compute_c5(h_try, n)
                if c < best_c5 - 1e-15:
                    best_h, best_c5 = h_try, c
            except Exception:
                pass

    # Phase 3: Multi-restart with diverse strategies
    while remaining() > 60:
        r = rng.random()
        total_budget = budget_s - 60
        frac = max(0.0, remaining() - 60) / max(1.0, total_budget)
        if r < 0.35:
            scale = (0.01 + 0.15 * frac) * rng.choice([0.5, 1.0, 2.0])
            h_pert = best_h + rng.normal(0, scale, size=n)
        elif r < 0.55:
            b = to_binary(best_h, n)
            n_flips = rng.randint(1, 6)
            for _ in range(n_flips):
                i = rng.randint(n)
                j = rng.randint(n)
                if i != j and b[i] != b[j]:
                    b[i], b[j] = b[j], b[i]
            h_pert = b
        elif r < 0.75:
            # Blend binary and continuous
            b = to_binary(best_h, n)
            w = rng.uniform(0.5, 1.0)
            h_pert = w * b + (1.0 - w) * best_h
        else:
            h_pert = rng.random(n)
        h_pert = project_feasible(h_pert, n)
        alpha = rng.choice([5000, 10000, 30000, 100000])
        try:
            res = minimize(smooth_obj, h_pert, args=(n, alpha), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 3000, 'ftol': 1e-18})
            h_try = project_feasible(res.x, n)
            c = compute_c5(h_try, n)
            if c < best_c5 - 1e-14:
                best_h, best_c5 = h_try, c
        except Exception:
            pass

    # Phase 4: Fine local search with adaptive step
    if remaining() > 15:
        count = 0
        best_delta = 0.005
        while remaining() > 8 and count < 120000:
            count += 1
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            for d in [best_delta * 0.5, best_delta, best_delta * 2.0]:
                improved = False
                for s in [1, -1]:
                    di = s * d
                    hi = best_h[i] + di
                    hj = best_h[j] - di
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h_try = best_h.copy()
                        h_try[i] = hi
                        h_try[j] = hj
                        c = compute_c5(h_try, n)
                        if c < best_c5:
                            best_h = h_try
                            best_c5 = c
                            best_delta = min(best_delta, d)
                            improved = True
                            break
                if improved:
                    break
        # Reset delta occasionally
        best_delta = max(0.001, best_delta)

    # Phase 5: Final high-precision SLSQP
    if remaining() > 10:
        for alpha in [200000, 1000000]:
            if remaining() < 8:
                break
            try:
                res = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 30000, 'ftol': 1e-20})
                h_try = project_feasible(res.x, n)
                c = compute_c5(h_try, n)
                if c < best_c5:
                    best_h, best_c5 = h_try, c
            except Exception:
                pass

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
