import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1 - h, mode='full')

def compute_c5(h):
    n = len(h)
    return np.max(compute_corr(h)) * (2.0 / n)

def project_box(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_shift(h, k):
    n = len(h)
    g = np.zeros(n)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    elif k > 0:
        g[:n - k] += 1.0 - h[k:]
        g[k:] -= h[:n - k]
    else:
        p = -k
        g[p:] += 1.0 - h[:n - p]
        g[:n - p] -= h[p:]
    return g

def smooth_grad(h, alpha, top_k=30):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    if top_k < len(corr):
        top_idx = np.argsort(corr)[-top_k:]
    else:
        top_idx = np.arange(len(corr))
    shifted = corr[top_idx] - m
    w = np.exp(alpha * shifted)
    s = np.sum(w)
    if s < 1e-300:
        w[:] = 0.0
        w[np.argmax(shifted)] = 1.0
    else:
        w /= s
    dx = 2.0 / n
    g = np.zeros(n)
    for i in range(len(top_idx)):
        if w[i] < 1e-10:
            continue
        g += w[i] * grad_shift(h, top_idx[i] - (n - 1))
    return g * dx

def optimize(h0, target_sum, deadline, rng, alpha0=120):

        v = mu * v - lr * g
        hc = project_box(h + v, target_sum)
        c5 = compute_c5(hc)

            best = hc.copy()
            best_c5 = c5

            pat = 0

                lr = 0.006
                alpha = max(50.0, alpha * 0.5)
                pat = 0
                stall += 1
                if stall > 12:
                    break
    return best, best_c5

def greedy_binary(n, rng):
    h = np.zeros(n)
    corr = np.zeros(2 * n - 1)
    k_vals = np.arange(-(n - 1), n)
    for _ in range(n // 2):

        best_max = np.inf
        zeros = np.where(h == 0)[0]
        for p in zeros:

            j2 = p - k_vals
            v1 = (j1 >= 0) & (j1 < n)
            v2 = (j2 >= 0) & (j2 < n)
            delta = np.zeros(2 * n - 1)

        delta = np.zeros(2 * n - 1)
        delta[v1] += 1.0 - h[j1[v1]]
        delta[v2] -= h[j2[v2]]
        corr += delta
    return h

def interpolate(h, n_new):
    n = len(h)
    if n == n_new:
        return h.copy()
    x0 = np.linspace(0.0, 1.0, n, endpoint=False) + 0.5 / n
    x1 = np.linspace(0.0, 1.0, n_new, endpoint=False) + 0.5 / n_new
    h_ext = np.concatenate([h, h[:1]])
    x_ext = np.concatenate([x0, [1.0 + 0.5 / n]])
    return np.interp(x1, x_ext, h_ext)
