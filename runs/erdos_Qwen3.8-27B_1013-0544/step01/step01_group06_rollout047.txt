import numpy as np
import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def binarize(h, n):
    """Convert h to binary with exactly n/2 ones."""
    # Sort h, take top n/2 as 1s
    threshold_idx = n // 2
    sorted_idx = np.argsort(h)[::-1]
    result = np.zeros(n)
    result[sorted_idx[:threshold_idx]] = 1.0
    return result

def binarize_with_ties(h, n):
    """Binarize, breaking ties randomly."""
    n_ones = n // 2
    # Add small random perturbation to break ties
    perturbed = h + np.random.RandomState(42).randn(n) * 1e-6
    sorted_idx = np.argsort(perturbed)[::-1]
    result = np.zeros(n)
    result[sorted_idx[:n_ones]] = 1.0
    return result

def swap_search(h, deadline, rng, n_iter=50000):
    """Binary local search with exchange moves."""
    n = len(h)
    n_ones = n // 2
    h = binarize(h, n)
    best_c5 = compute_c5(h)
    current_c5 = best_c5
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    it = 0
    no_improve = 0
    T = 0.01  # temperature for simulated annealing
    
    while time.time() < deadline and it < n_iter:
        it += 1
        # Random swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h[i] = 0
        h[j] = 1
        new_c5 = compute_c5(h)
        
        if new_c5 < current_c5:
            current_c5 = new_c5
            no_improve = 0
            if new_c5 < best_c5:
                best_c5 = new_c5
        else:
            # Simulated annealing: accept with probability exp(-delta/T)
            delta = new_c5 - current_c5
            if rng.random() < np.exp(-delta / max(T, 1e-10)):
                current_c5 = new_c5
            else:
                # Revert
                h[i] = 1
                h[j] = 0
                no_improve += 1
        
        # Update ones/zeros sets
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        # Cool down
        if no_improve > 100:
            T *= 0.95
            no_improve = 0

            T = 0.01

            for _ in range(5):

                j = rng.randint(n)
                h[i], h[j] = h[j], h[i]
            h = binarize(h, n)
            current_c5 = compute_c5(h)
    
    return h, best_c5

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr)) * (2.0 / n)

def compute_c5_with_lag(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr)) * (2.0 / n), int(np.argmax(corr))

def project_binary(h, n):
    """Project to binary vector with exactly n/2 ones."""
    n_ones = n // 2
    # Use the values to determine which positions get 1
    # Top n_ones values become 1
    idx = np.argsort(h)[::-1]
    result = np.zeros(n)
    result[idx[:n_ones]] = 1.0
    return result

def gradient_step(h, n, alpha=100):
    """Compute smooth max gradient and return gradient direction."""
    corr = np.correlate(h, 1 - h, mode='full')
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    s = w.sum()
    w = w / s
    
    # Gradient of smooth max
    grad = np.zeros(n)
    n_corr = len(corr)
    for idx in range(n_corr):
        if w[idx] < 1e-10:
            continue
        k = idx - (n - 1)
        # grad_corr_single
        if k >= 0:
            g = np.zeros(n)
            g[k:] += (1.0 - h[:n-k])
            if k > 0:
                g[:n-k] -= h[k:]
            grad += w[idx] * g
        else:
            p = -k
            g = np.zeros(n)
            g[:n-p] += (1.0 - h[p:])
            if p > 0:
                g[p:] -= h[:n-p]
            grad += w[idx] * g
    
    return grad * (2.0 / n)

def exact_subgradient(h, n):
    """Compute subgradient using only the worst lag."""
    corr = np.correlate(h, 1 - h, mode='full')
    idx = int(np.argmax(corr))
    k = idx - (n - 1)
    
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n-k])
        if k > 0:
            g[:n-k] -= h[k:]
    else:
        p = -k
        g[:n-p] += (1.0 - h[p:])
        if p > 0:
            g[p:] -= h[:n-p]
    
    return g * (2.0 / n)

def binary_sa_search(h, n, deadline, rng, n_restarts=10):
    """Simulated annealing on binary vectors."""
    n_ones = n // 2
    h = project_binary(h, n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    current_c5 = best_c5
    
    for restart in range(n_restarts):
        if time.time() > deadline:
            break
        
        if restart > 0:
            # Perturb: flip random pairs
            n_flips = max(3, n // 20)
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            for _ in range(n_flips):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h[i], h[j] = h[j], h[i]
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        
        T = 0.005
        no_improve = 0
        max_no_improve = 2000
        
        while time.time() < deadline:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            
            h[i], h[j] = 0.0, 1.0
            new_c5 = compute_c5(h)
            
            delta = new_c5 - current_c5
            if delta < 0 or rng.random() < np.exp(-delta / T):
                current_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h[i], h[j] = 1.0, 0.0
                no_improve += 1
            
            if no_improve >= max_no_improve:
                T *= 0.9
                no_improve = 0
                if T < 1e-7:
                    break
    
    return best_h, best_c5

def gradient_refine(h, n, deadline, rng, alpha=100, lr=0.01):
    """Gradient-based refinement with projection."""
    target_sum = n / 2.0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    patience = 0
    
    while time.time() < deadline:
        grad = gradient_step(best_h, n, alpha)
        h_new = best_h - lr * grad
        # Project to [0,1] with sum constraint
        h_new = np.clip(h_new, 0, 1)
        # Adjust sum
        diff = target_sum - h_new.sum()
        # Distribute adjustment
        if abs(diff) > 1e-10:
            # Add/subtract uniformly to non-boundary elements
            mask = (h_new > 1e-6) & (h_new < 1 - 1e-6)
            if mask.sum() > 0:
                h_new[mask] += diff / mask.sum()
                h_new = np.clip(h_new, 0, 1)
        
        new_c5 = compute_c5(h_new)
        if new_c5 < best_c5:
            best_h = h_new
            best_c5 = new_c5
            patience = 0
            if lr < 0.1:
                lr = min(lr * 1.05, 0.1)
        else:
            patience += 1
            lr *= 0.85
            if patience > 50:
                break
    
    return best_h, best_c5

def project_onto_simplex_box(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def interpolate_to_n(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False) + 1.0 / n_new
    h_ext = np.append(h_old, h_old[0])
    x_ext = np.append(x_old, 2.0 + 1.0 / n_old)
    return np.interp(x_new, x_ext, h_ext)

            new_c5 = compute_c5(h)
            tried += 1
            if new_c5 < best_swap_c5:
                best_swap_c5 = new_c5
                best_i, best_j = i, j
            else:
                h[i], h[j] = 1.0, 0.0
        
        if best_i >= 0:
            h[best_i], h[best_j] = 0.0, 1.0
            best_c5 = best_swap_c5
            best_h = h.copy()
        else:
            # No improvement: try a perturbation
            n_pert = 2 + rng.randint(3)
            for _ in range(n_pert):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h[i], h[j] = 0.0, 1.0
            h = project_binary(h, n)
            
    return best_h, best_c5

def sa_binary_search(h, n, deadline, rng):
    """Simulated annealing with binary swaps."""
    n_ones = n // 2
    h = project_binary(h, n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    current_c5 = best_c5
    T = 0.003
    no_improve = 0
    patience_limit = 3000
    
    it = 0
    while time.time() < deadline:
        it += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        new_c5 = compute_c5(h)
        
        delta = new_c5 - current_c5
        if delta < 0 or (T > 1e-10 and rng.random() < np.exp(-delta / T)):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i], h[j] = 1.0, 0.0
            no_improve += 1
        
        if no_improve >= patience_limit:
            T *= 0.85
            no_improve = 0
            if T < 1e-8:
                # Restart with perturbation
                for _ in range(3 + rng.randint(5)):
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                    h[i], h[j] = 0.0, 1.0
                h = project_binary(h, n)
                current_c5 = compute_c5(h)
                T = 0.003
    
    return best_h, best_c5

def gradient_refine(h, n, deadline, rng, alpha=120, lr=0.02):
    target_sum = n / 2.0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    patience = 0
    
    while time.time() < deadline:
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        m = np.max(corr)
        w = np.exp(alpha * (corr - m))
        s = w.sum()
        if s > 1e-300:
            w /= s
        else:
            w = np.zeros_like(corr)
            w[np.argmax(corr)] = 1.0
        
        grad = np.zeros(n)
        n_corr = len(corr)
        active = np.where(w > 1e-8)[0]
        for idx in active:
            k = int(idx) - (n - 1)
            if k >= 0:
                g = np.zeros(n)
                g[k:] += (1.0 - best_h[:n - k])
                if k > 0:
                    g[:n - k] -= best_h[k:]
            else:
                p = -k
                g = np.zeros(n)
                g[:n - p] += (1.0 - best_h[p:])
                if p > 0:
                    g[p:] -= best_h[:n - p]
            grad += w[idx] * g
        grad *= (2.0 / n)
        
        h_new = best_h - lr * grad
        h_new = project_onto_simplex_box(h_new, target_sum)
        new_c5 = compute_c5(h_new)
        
        if new_c5 < best_c5 - 1e-14:
            best_h = h_new
            best_c5 = new_c5
            patience = 0
            if lr < 0.08:
                lr = min(lr * 1.04, 0.08)
        else:
            patience += 1
            lr *= 0.9
            if patience > 60:
                break
    
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8
    rng = np.random.RandomState(seed)
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        n0 = 200
        h0 = np.zeros(n0)
        h0[rng.choice(n0, n0 // 2, replace=False)] = 1.0
    
    n0 = len(h0)
    best_c5 = float('inf')
    best_h = None
    best_n = n0
    
    # Try n0 first (primary), then other resolutions
    resolutions = [n0]
    for nt in [100, 150, 300]:
        if nt != n0:
            resolutions.append(nt)
    resolutions.sort(reverse=True)  # primary first
    
    time_budget_each = (budget_s - 60) / len(resolutions)
    
    for n in resolutions:
        if time.time() > deadline - 20:
            break
        
        if n == n0:
            h_init = h0.copy()
        else:
            h_init = interpolate_to_n(best_h if best_h is not None else h0,
                                       best_n if best_h is not None else n0, n)
        
        target_sum = n / 2.0
        h_init = project_onto_simplex_box(h_init, target_sum)
        
        # Phase 1: Short gradient refinement
        t1 = time.time() + min(time_budget_each * 0.15, 45)
        h_cur, c5_cur = gradient_refine(h_init, n, t1, rng, alpha=100, lr=0.015)
        
        # Phase 2: Binarize and check
        h_bin = project_binary(h_cur, n)
        c5_bin = compute_c5(h_bin)
        if c5_bin < c5_cur:
            h_cur = h_bin
            c5_cur = c5_bin
        
        # Phase 3: Focused swap search (main phase)
        t3 = time.time() + min(time_budget_each * 0.45, 300)
        h_cur, c5_cur = focused_swap_search(h_cur, n, t3, rng, K=40)
        
        # Phase 4: SA binary search
        t4 = time.time() + min(time_budget_each * 0.30, 200)
        h_sa, c5_sa = sa_binary_search(h_cur, n, t4, rng)
        if c5_sa < c5_cur:
            h_cur = h_sa
            c5_cur = c5_sa
        
        # Phase 5: Final gradient + rebinarize
        if time.time() < deadline - 10:
            t5 = min(time.time() + 30, deadline - 5)
            h_gr, c5_gr = gradient_refine(h_cur, n, t5, rng, alpha=150, lr=0.01)
            h_bin2 = project_binary(h_gr, n)
            c5_bin2 = compute_c5(h_bin2)
            if c5_bin2 < c5_cur:
                h_cur = h_bin2
                c5_cur = c5_bin2
            elif c5_gr < c5_cur:
                h_cur = h_gr
                c5_cur = c5_gr
        
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()
            best_n = n
    
    # Final projection
    best_h = project_onto_simplex_box(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, best_n
