        proj = np.clip(h - tau, 0.0, 1.0)

            hi = tau
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def obj_grad(h, n, target_sum, beta, penalty):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    weights = np.exp(beta * (corr - m))
    wsum = np.sum(weights)
    weights /= wsum
    val = (m + np.log(wsum) / beta) * dx
    val += penalty * (np.sum(h) - target_sum) ** 2
    g = np.zeros(n)
    for m_idx in range(2 * n - 1):
        w = weights[m_idx]
        if w < 1e-15:
            continue
        k = m_idx - (n - 1)
        for j in range(n):
            jk = j + k
            jmk = j - k
            if 0 <= jk < n:
                g[j] += w * (1.0 - h[jk])
            if 0 <= jmk < n:
                g[j] -= w * h[jmk]
    g *= dx
    g += 2.0 * penalty * (np.sum(h) - target_sum)
    return val, g
