    n = len(h)
    grad = np.zeros(n)
    n1 = n - 1
    for k_idx in range(2 * n - 1):
        w = weights[k_idx]
        if w < 1e-14:
            continue
        lag = k_idx - n1
        i_min = max(0, -lag)
        i_max = min(n - 1, n - 1 - lag)
        
        # Term 1: j = i + lag, i in [i_min, i_max]
        j_start = max(0, i_min + lag)
        j_end = min(n - 1, i_max + lag)
        if j_start <= j_end:
            indices = np.arange(j_start, j_end + 1)
            grad[indices] += w * (1.0 - h[indices - lag])
        
        # Term 2: j = i, i in [i_min, i_max]
        if i_min <= i_max:
            indices = np.arange(i_min, i_max + 1)
            grad[indices] -= w * h[indices + lag]
    
    return grad

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_value(corr, beta):
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta

def smooth_max_weights(corr, beta):
    m = np.max(corr)
    w = np.exp(beta * (corr - m))
    w /= np.sum(w)
    return w

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(-h)
            for i in order:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
                if diff <= 1e-12:
                    break
        else:
            order = np.argsort(h)
            for i in order:
                rem = min(-diff, h[i])
                h[i] -= rem
                diff += rem
                if diff >= -1e-12:
                    break
    return h

def compute_gradient(h, weights):
    n = len(h)
    grad = np.zeros(n)
    n1 = n - 1
    for k_idx in range(2 * n - 1):
        w = weights[k_idx]
        if w < 1e-14:
            continue
        lag = k_idx - n1
        i_min = max(0, -lag)
        i_max = min(n - 1, n - 1 - lag)
        if i_min > i_max:
            continue
        # Term 1: j = i + lag, i in [i_min, i_max]
        j_start = max(0, i_min + lag)
        j_end = min(n - 1, i_max + lag)
        if j_start <= j_end:
            grad[j_start:j_end+1] += w * (1.0 - h[j_start-lag:j_end+1-lag])
        # Term 2: j = i, i in [i_min, i_max]
        grad[i_min:i_max+1] -= w * h[i_min+lag:i_max+1+lag]
    return grad

def adam_optimize(h0, target_sum, n_iters, lr, beta_param):
    n = len(h0)
    h = h0.copy()
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for t in range(1, n_iters + 1):
        corr = np.correlate(h, 1.0 - h, mode="full")
        weights = smooth_max_weights(corr, beta_param)
        grad = compute_gradient(h, weights)
        
        m_adam = b1 * m_adam + (1 - b1) * grad
        v_adam = b2 * v_adam + (1 - b2) * grad**2
        m_hat = m_adam / (1 - b1**t)
        v_hat = v_adam / (1 - b2**t)
        
        h -= lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_h(h, target_sum)
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

def coordinate_descent(h0, target_sum, n_passes=50, beta_param=100):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)

        weights = smooth_max_weights(corr, beta_param)
        
        for j in range(n):
            # Find optimal h[j] by minimizing smooth max
            # corr_k as function of h[j] is affine: corr_k = a_k * h[j] + b_k
            # where a_k = d(corr_k)/d(h[j])
            
            # Compute a_k for all k
            a = np.zeros(2*n-1)
            n1 = n-1
            for k_idx in range(2*n-1):
                lag = k_idx - n1
                i_min = max(0, -lag)
                i_max = min(n-1, n-1-lag)
                if i_min > i_max:
                    continue
                # j appears as h[i+lag] when i = j-lag
                if i_min <= j-lag <= i_max:
                    a[k_idx] += 1.0 - h[j-lag]  # Wait, this is wrong
                # Actually, a[k_idx] = d(corr_k)/d(h[j]) with current h fixed except h[j]
                # We need the coefficient of h[j] in corr_k
                # corr_k = sum_i h[i](1-h[i+lag])
                # h[j] appears as h[i] when i=j: contributes h[j](1-h[j+lag])
                # h[j] appears as h[i+lag] when i+lag=j, i=j-lag: contributes h[j-lag](1-h[j])
                # So the coefficient of h[j] (linear part after expanding):
                # From first: (1-h[j+lag]) if j is valid i (i_min<=j<=i_max and j+lag valid)
                # From second: -h[j-lag] if j-lag is valid i (i_min<=j-lag<=i_max)
                # Total coefficient: (1-h[j+lag])*I[valid] - h[j-lag]*I[valid]
                
                # Let me recompute:
                coeff = 0.0
                # h[j] as h[i] where i=j: valid if i_min<=j<=i_max and 0<=j+lag<n
                if i_min <= j <= i_max and 0 <= j+lag < n:
                    coeff += (1.0 - h[j+lag])
                # h[j] as h[i+lag] where i=j-lag: valid if i_min<=j-lag<=i_max and 0<=j<n
                if i_min <= j-lag <= i_max and 0 <= j < n:
                    coeff -= h[j-lag]
                a[k_idx] = coeff
                # Wait, I was computing a[k_idx] for each k_idx, but this should be
                # the derivative d(corr_k)/d(h[j]) which is just the coefficient since
                # corr_k is linear in h[j]
            
            # Now minimize smooth_max(a[h[j]] + b) where b = corr - a*h[j] (current corr minus the h[j] contribution)
            # Actually: corr_k(h') = corr_k(h) - a[k]*h[j] + a[k]*h'[j] = corr_k(h) + a[k]*(h'[j] - h[j])
            # So we want to minimize smooth_max(corr + a*(x - h[j])) over x in [0,1]
            
            current_hj = h[j]
            # Scan x in [0, 1] to find minimum of smooth_max(corr + a*(x - current_hj))
            # This is a 1D convex function in x
            # Use golden section or ternary search
            
            lo, hi = 0.0, 1.0
            for _ in range(50):  # 50 iterations of golden section
                m1 = lo + 0.382 * (hi - lo)
                m2 = hi - 0.382 * (hi - lo)
                f1 = smooth_max_value(corr + a * (m1 - current_hj), beta_param)
                f2 = smooth_max_value(corr + a * (m2 - current_hj), beta_param)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
            h[j] = (lo + hi) / 2
        
        h = project_h(h, target_sum)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

def binary_sa(h0, target_sum, max_iter, temp_init=2.0):
    n = len(h0)
    h = (h0 > 0.5).astype(float)
    n_ones = int(h.sum())
    if n_ones > target_sum:
        idx = np.where(h == 1)[0]
        rem = np.random.choice(idx, n_ones - target_sum, replace=False)
        h[rem] = 0
    elif n_ones < target_sum:
        idx = np.where(h == 0)[0]
        add = np.random.choice(idx, target_sum - n_ones, replace=False)
        h[add] = 1
    
    best_h = h.copy()
    best_val = float(np.max(np.correlate(h, 1 - h, mode="full")))
    curr_val = best_val
    temp = temp_init
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(max_iter):
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        
        h[i], h[j] = 1.0 - h[i], 1.0 - h[j]
        # Actually, swap: h[i] was 1, now 0; h[j] was 0, now 1

import numpy as np
from scipy.optimize import minimize, Bounds
import time

_G = {}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_value(corr, beta):
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta

def smooth_max_weights(corr, beta):
    m = np.max(corr)
    w = np.exp(beta * (corr - m))
    w /= np.sum(w)
    return w

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(-h)
            for i in order:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
                if diff <= 1e-14:
                    break
        else:
            order = np.argsort(h)
            for i in order:
                rem = min(-diff, h[i])
                h[i] -= rem
                diff += rem
                if diff >= -1e-14:
                    break
    return h

def _con_fun(x):
    return np.sum(x) - _G['target_sum']

def _con_jac(x):
    return np.ones(len(x))

def _obj_slsqp(h):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max_value(corr, _G['beta']) * _G['dx']

def compute_gradient(h, weights):
    n = len(h)
    grad = np.zeros(n)
    n1 = n - 1
    for k_idx in range(2 * n - 1):
        w = weights[k_idx]
        if w < 1e-14:
            continue
        lag = k_idx - n1
        i_min = max(0, -lag)
        i_max = min(n - 1, n - 1 - lag)
        if i_min > i_max:
            continue
        j_start = max(0, i_min + lag)
        j_end = min(n - 1, i_max + lag)
        if j_start <= j_end:
            grad[j_start:j_end + 1] += w * (1.0 - h[j_start - lag:j_end + 1 - lag])
        grad[i_min:i_max + 1] -= w * h[i_min + lag:i_max + 1 + lag]
    return grad

def adam_optimize(h0, target_sum, n_iters, lr, beta_param):
    n = len(h0)
    h = h0.copy()
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for t in range(1, n_iters + 1):
        corr = np.correlate(h, 1.0 - h, mode="full")
        weights = smooth_max_weights(corr, beta_param)
        grad = compute_gradient(h, weights)
        m_adam = b1 * m_adam + (1.0 - b1) * grad
        v_adam = b2 * v_adam + (1.0 - b2) * grad * grad
        m_hat = m_adam / (1.0 - b1 ** t)
        v_hat = v_adam / (1.0 - b2 ** t)
        h -= lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_h(h, target_sum)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def coord_descent(h0, target_sum, n_passes, beta_param):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n1 = n - 1
    for p in range(n_passes):
        corr = np.correlate(h, 1.0 - h, mode="full")
        for j in range(n):
            coeff = np.zeros(2 * n - 1)
            for k_idx in range(2 * n - 1):
                lag = k_idx - n1
                i_min = max(0, -lag)
                i_max = min(n - 1, n - 1 - lag)
                if i_min > i_max:
                    continue
                c = 0.0
                if i_min <= j <= i_max and 0 <= j + lag < n:
                    c += 1.0 - h[j + lag]
                if i_min <= j - lag <= i_max:
                    c -= h[j - lag]
                coeff[k_idx] = c
            hj = h[j]
            lo, hi_val = 0.0, 1.0
            for _ in range(45):
                m1 = lo + 0.382 * (hi_val - lo)
                m2 = hi_val - 0.382 * (hi_val - lo)
                f1 = smooth_max_value(corr + coeff * (m1 - hj), beta_param)
                f2 = smooth_max_value(corr + coeff * (m2 - hj), beta_param)
                if f1 < f2:
                    hi_val = m2
                else:
                    lo = m1
            h[j] = (lo + hi_val) / 2.0
        h = project_h(h, target_sum)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def binary_sa(h0, target_sum, max_iter, temp_init):
    n = len(h0)
    h = (h0 > 0.5).astype(float)
    n_ones = int(h.sum())
    if n_ones > int(target_sum):
        idx = np.where(h == 1)[0]
        rem = np.random.choice(idx, n_ones - int(target_sum), replace=False)
        h[rem] = 0
    elif n_ones < int(target_sum):
        idx = np.where(h == 0)[0]
        add = np.random.choice(idx, int(target_sum) - n_ones, replace=False)
        h[add] = 1
    best_h = h.copy()
    best_val = float(np.max(np.correlate(h, 1 - h, mode="full")))
    curr_val = best_val
    temp = temp_init
    for it in range(max_iter):
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_val = float(np.max(np.correlate(h, 1 - h, mode="full")))
        delta = new_val - curr_val
        if delta <= 0 or np.random.random() < np.exp(-delta / max(temp, 0.01)):
            curr_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        temp = max(temp * 0.9995, 0.05)
    return best_h, best_val * (2.0 / n)

def binary_hill_climb(h0, target_sum, max_iter):
    n = len(h0)
    h = (h0 > 0.5).astype(float)
    n_ones = int(h.sum())
    if n_ones > int(target_sum):
        idx = np.where(h == 1)[0]
        rem = np.random.choice(idx, n_ones - int(target_sum), replace=False)
        h[rem] = 0
    elif n_ones < int(target_sum):
        idx = np.where(h == 0)[0]
        add = np.random.choice(idx, int(target_sum) - n_ones, replace=False)
        h[add] = 1
    best_h = h.copy()
    best_val = float(np.max(np.correlate(h, 1 - h, mode="full")))
    improved = True
    it = 0
    while improved and it < max_iter:
        improved = False
        it += 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        for i in np.random.choice(ones, min(len(ones), 10), replace=False):
            for j in np.random.choice(zeros, min(len(zeros), 10), replace=False):
                h[i] = 0.0
                h[j] = 1.0
                new_val = float(np.max(np.correlate(h, 1 - h, mode="full")))
                if new_val < best_val:
                    best_val = new_val
                    best_h = h.copy()
                    improved = True
                else:
                    h[i] = 1.0
                    h[j] = 0.0
    return best_h, best_val * (2.0 / n)

def _time_left(start_time, budget_s):
    return budget_s - (time.time() - start_time)
