So there's a tension: at small k, we want h to be "spiky" (0 or 1), but at large k, we want it to be "smooth" (correlated with its shifts).

This is exactly the kind of structure that a carefully optimized non-trivial h would have.

import numpy as np
from scipy.optimize import minimize, Bounds
import time

# Global state for objective (avoids closures)
_g_n = 70
_g_beta = 100
_g_lam = 0.0
_g_target = 35.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_obj(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta) * dx

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def _objective(h):
    return smooth_max_obj(h, _g_n, _g_beta) + _g_lam * (np.sum(h) - _g_target)

def _run_lbfgs(h0, n, target_sum, beta, bounds, maxiter, ftol):
    _g_n = n
    _g_beta = beta
    _g_lam = 0.0
    _g_target = target_sum
    h = h0.copy()
    for dual_iter in range(30):
        try:
            res = minimize(_objective, h, method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-7})
            h = project_h(res.x, target_sum)
        except Exception:
            break
        viol = np.sum(h) - target_sum
        if abs(viol) < 1e-11:
            break
        _g_lam -= 0.05 * viol
        if abs(_g_lam) > 5.0:
            _g_lam = np.sign(_g_lam) * 5.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    def tl():
        return budget_s - (time.time() - start_time)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)

    n0 = len(h_init)
    best_h = project_h(h_init.copy(), n0 / 2.0)
    best_c5 = compute_c5(best_h)

    n_list = sorted(set([n0, 60, 70, 80, 100]))

    for n in n_list:
        if tl() < 70:
            break
        target_sum = n / 2.0
        bounds = Bounds(np.zeros(n), np.ones(n))

        inits = []
        if n == n0:
            inits.append(best_h.copy())

        x = np.linspace(0, 2, n, endpoint=False)
        inits.append(np.full(n, 0.5))
        inits.append(0.5 + 0.25 * np.sin(np.pi * x))
        inits.append(0.5 + 0.2 * np.sin(2.0 * np.pi * x / 2.0) + 0.15 * np.cos(4.0 * np.pi * x / 2.0))
        inits.append(0.5 + 0.3 * np.sin(3.0 * np.pi * x / 2.0 + 0.5))
        inits.append(np.clip(0.5 + 0.25 * np.random.randn(n), 0.05, 0.95))
        inits.append(np.clip(0.3 + 0.4 * np.random.random(n), 0.0, 1.0))

        for h0 in inits:
            if tl() < 15:
                break
            h0 = project_h(h0, target_sum)

            for beta in [20, 50, 100, 200, 400]:
                if tl() < 8:
                    break
                h0 = _run_lbfgs(h0, n, target_sum, beta, bounds, 200, 1e-13)
                c5 = compute_c5(h0)
                if c5 < best_c5:
                    best_h, best_c5 = h0.copy(), c5

    # Random restarts around best
    n = len(best_h)
    target_sum = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))

    restart = 0
    while tl() > 30:
        restart += 1
        h0 = best_h.copy()
        scale = 0.01 + 0.04 * np.exp(-restart / 15.0)

        if restart % 4 == 1:
            h0 += np.random.randn(n) * scale
        elif restart % 4 == 2:

            bs = np.random.randint(0, max(1, n - blen))
            h0[bs:bs + blen] += np.random.randn(blen) * scale * 2
        elif restart % 4 == 3:
            corr = np.correlate(h0, 1.0 - h0, mode="full")
            top = np.argsort(corr)[-3:]
            for k in top:
                if k < n:
                    lag = k

                        h0[i + lag] += 0.01 * (1.0 - h0[i])
                        h0[i] += 0.01 * h0[i + lag]
                else:
                    lag = k - (n - 1)
                    for i in range(n - lag):
                        h0[i] += 0.01 * (1.0 - h0[i + lag])
                        h0[i + lag] += 0.01 * h0[i]
        else:
            h0[np.random.choice(n, n // 4, replace=False)] += np.random.randn(n // 4) * scale

            c5 = compute_c5(best_h)
            if c5 < best_c5:
                best_c5 = c5

    return best_h, best_c5, n
