import numpy as np
import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project(h, target_sum):
    lo, hi = -1.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def corr_grad_single(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n - k])
        if k > 0:
            g[:n - k] -= h[k:]
    else:
        p = -k
        g[:n - p] += (1.0 - h[p:])
        g[p:] -= h[:n - p]
    return g

def get_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    s = np.sum(w)
    w /= s
    grad = np.zeros(n)
    active = np.where(w > 1e-10)[0]
    for idx in active:
        k = idx - (n - 1)
        grad += w[idx] * corr_grad_single(h, k)
    return grad * (2.0 / n)

def nesterov_opt(h0, target_sum, deadline, rng, alpha_init=50, lr_init=0.006):
    n = len(h0)
    h = project(h0.copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    alpha = alpha_init
    momentum = 0.85
    v = np.zeros(n)
    stall = 0
    max_stall = 60
    stall_budget = 0

    while time.time() < deadline:
        y = h + momentum * v
        grad = get_grad(y, alpha)
        v = momentum * v - lr * grad
        h_new = project(y + v, target_sum)
        c5 = compute_c5(h_new)

        if c5 < best_c5 - 1e-13:
            best_h = h_new.copy()
            best_c5 = c5
            stall = 0
            if lr < 0.03:
                lr = min(lr * 1.015, 0.04)
            momentum = min(momentum + 0.001, 0.95)
        else:
            stall += 1
            if stall > 15:
                lr *= 0.93
            if stall > max_stall:
                stall_budget += 1
                size = 0.015 + 0.01 * min(stall_budget, 5)
                h_pert = best_h + rng.randn(n) * size
                h = project(h_pert, target_sum)
                v = np.zeros(n)
                alpha = min(alpha * 1.4, 500.0)
                lr = lr_init * 0.6
                momentum = 0.8
                stall = 0
                max_stall += 40
                if stall_budget > 12:
                    break

        h = h_new
        alpha = min(alpha * 1.0008, 600.0)

    return best_h, best_c5

def interp_h(h_old, n_old, n_new):
    x_old = np.arange(n_old) / n_old
    x_new = np.arange(n_new) / n_new
    h_ext = np.concatenate([h_old, [h_old[0]]])
    x_ext = np.concatenate([x_old, [1.0]])
    return np.interp(x_new, x_ext, h_ext)

def make_block_init(n, rng, n_blocks):
    h = np.zeros(n)
    boundaries = np.sort(rng.choice(range(1, n), size=n_blocks - 1, replace=False))
    boundaries = np.concatenate([[0], boundaries, [n]])
    for i in range(n_blocks):
        val = rng.choice([0.0, 0.3, 0.5, 0.7, 1.0])
        h[boundaries[i]:boundaries[i+1]] = val
    return project(h, n / 2.0)

def make_periodic_init(n, rng):
    freq = rng.randint(2, 8)
    phase = rng.random() * 2 * np.pi
    x = np.arange(n) / n * 2.0
    h = 0.5 + 0.4 * np.sin(2 * np.pi * freq * x / 2.0 + phase)
    h = np.clip(h, 0, 1)
    return project(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float)
    else:
        h_parent = None

    best_h = None
    best_c5 = 1.0
    best_n = 0

    n_list = [100, 150, 200, 250]
    if h_parent is not None:
        n_parent = len(h_parent)
        if n_parent not in n_list:
            n_list = [n_parent] + [x for x in n_list if x != n_parent]
        n_list.sort()

    total_budget = budget_s
    per_phase = total_budget / (len(n_list) + 2)

    # Phase 1: Optimize each resolution
    for n in n_list:
        if time.time() - t0 > total_budget - 60:
            break
        phase_deadline = min(t0 + total_budget * 0.6, time.time() + per_phase * 1.5)
        deadline = min(phase_deadline, t0 + total_budget - 80)

        if best_h is not None and best_n != n:
            h_init = project(interp_h(best_h, best_n, n), n / 2.0)
        elif h_parent is not None and n == len(h_parent):
            h_init = project(h_parent.copy(), n / 2.0)
        else:
            h_init = make_block_init(n, rng, rng.randint(4, 12))

        h_opt, c5_opt = nesterov_opt(h_init, n / 2.0, deadline, rng,
                                      alpha_init=50, lr_init=0.006)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
            best_n = n

    # Phase 2: Diverse restarts at best resolution
    remaining = total_budget - (time.time() - t0)
    if remaining > 50:
        n_restart = best_n
        n_restarts = int(remaining / 40)
        n_restarts = max(2, min(n_restarts, 8))
        for r in range(n_restarts):
            if time.time() - t0 > total_budget - 40:
                break
            if r % 3 == 0:
                h_init = make_block_init(n_restart, rng, rng.randint(3, 15))
            elif r % 3 == 1:
                h_init = make_periodic_init(n_restart, rng)
            else:
                h_init = project(best_h + rng.randn(n_restart) * 0.05, n_restart / 2.0)
            d_r = min(time.time() + 35, t0 + total_budget - 30)
            h_r, c5_r = nesterov_opt(h_init, n_restart / 2.0, d_r, rng,
                                      alpha_init=80, lr_init=0.004)
            if c5_r < best_c5:
                best_h = h_r
                best_c5 = c5_r
                best_n = n_restart

    # Phase 3: Final polish
    remaining = total_budget - (time.time() - t0)
    if remaining > 20:
        polish_deadline = t0 + total_budget - 3
        best_h, best_c5 = nesterov_opt(best_h, best_n / 2.0, polish_deadline,
                                        rng, alpha_init=200, lr_init=0.002)

    best_h = project(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
