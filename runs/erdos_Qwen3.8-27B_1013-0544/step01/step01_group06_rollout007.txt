    m = np.max(corr)

    if s < 1e-300:

        w[np.argmax(corr)] = 1.0
    else:
        w = e / s

def grad_for_shift(h, k, n):
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n - k])
        if k > 0:
            g[:n - k] -= h[k:]
    else:
        p = -k
        g[:n - p] += (1.0 - h[p:])
        if p > 0:
            g[p:] -= h[:n - p]
    return g

    best_h, best_c5 = adam_optimize(h0, target_sum, d1, rng, alpha_lo=100, alpha_hi=2500)

    # Phase 2: Binary-structured restarts (25% budget)
    d2 = t0 + budget_s * 0.60
    nrestart = 4
    for i in range(nrestart):
        if time.time() > d2:
            break
        rd = time.time() + (d2 - time.time()) / (nrestart - i)
        if i % 2 == 0:
            h_pert = binary_perturb(best_h, n, rng)
        else:
            h_pert = frequency_perturb(best_h, n, rng)
        h_pert = project_box_sum(h_pert, target_sum)
        h_opt, c5_opt = adam_optimize(h_pert, target_sum, min(rd, d2), rng,
                                       alpha_lo=200, alpha_hi=3000)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt

    # Phase 3: Higher resolution attempt (20% budget)
    d3 = t0 + budget_s * 0.80
    if time.time() < d3:
        n2 = 400
        h_interp = interpolate_vals(best_h, n, n2)
        h_interp = project_box_sum(h_interp, n2 / 2.0)
        h_opt, c5_opt = adam_optimize(h_interp, n2 / 2.0, d3, rng,
                                       alpha_lo=200, alpha_hi=3000)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt

    # Phase 4: Final high-alpha polish (remaining time)
    d4 = t0 + budget_s - 5
    if time.time() < d4:
        nf = len(best_h)
        best_h, best_c5 = adam_optimize(best_h, nf / 2.0, d4, rng,
                                         alpha_lo=1000, alpha_hi=5000)

    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
