        h0 = best_h.copy()
        s = attempt % 9

        if s == 0:
            h0 += np.random.randn(n) * 0.04
        elif s == 1:
            f = np.random.uniform(0.5, 5.0)
            ph = np.random.uniform(0, 2 * np.pi)
            h0 += 0.03 * np.sin(2 * np.pi * f * np.arange(n) / n + ph)
        elif s == 2:
            L = np.random.randint(5, 15)
            st = np.random.randint(0, max(1, n - L))
            h0[st:st + L] = 1.0 - h0[st:st + L]
        elif s == 3:
            d = np.random.randint(2, n // 3)
            h0 = np.roll(h0, d)
        elif s == 4:
            h0 = 0.7 * best_h + 0.3 * np.random.random(n)
        elif s == 5:
            # Gradient of worst lag
            corr = np.correlate(h0, 1.0 - h0, mode="full")
            worst = int(np.argmax(corr))
            lag = worst - (n - 1)
            dx = 2.0 / n
            grad = np.zeros(n)
            for j in range(n):
                js = j + lag
                jm = j - lag
                if 0 <= js < n:
                    grad[j] += (1.0 - h0[js]) * dx
                if 0 <= jm < n:
                    grad[j] -= h0[jm] * dx
            h0 -= 0.05 * grad
            h0 += np.random.randn(n) * 0.01
        elif s == 6:
            for _ in range(np.random.randint(2, 8)):
                a, b = np.random.choice(n, 2, replace=False)
                h0[a], h0[b] = h0[b], h0[a]
        elif s == 7:
            # Targeted: swap mass at worst and 2nd worst lags
            corr = np.correlate(h0, 1.0 - h0, mode="full")
            sorted_idx = np.argsort(corr)[-2:]
            for idx in sorted_idx:
                lag = idx - (n - 1)
                for i in range(n):
                    j = i + lag
                    if 0 <= j < n:
                        if h0[i] > 0.5 and h0[j] < 0.5:
                            h0[i] -= 0.03
                            h0[j] += 0.03
                        elif h0[i] < 0.5 and h0[j] > 0.5:
                            h0[i] += 0.03
                            h0[j] -= 0.03
        else:
            h0 = 0.5 * best_h + 0.5 * np.random.random(n)

        h0 = project_h(h0, target)
        _g_mode = 'smooth'
        _g_beta = 300.0
        try:
            r = minimize(obj_func, h0, method='SLSQP', bounds=bounds,
                         constraints=cons, options={'maxiter': 1500, 'ftol': 1e-14})
            h_new = project_h(r.x, target)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new, c5
                no_improve = 0
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

        if no_improve > 12 and attempt % 4 == 0 and time.time() - t0 < budget_s * 0.82:
            _g_mode = 'max'
            try:
                r = minimize(obj_func, best_h, method='Nelder-Mead',
                             options={'maxiter': 400, 'xatol': 1e-8, 'fatol': 1e-10})
                h_new = project_h(r.x, target)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h, best_c5 = h_new, c5
                    no_improve = 0
            except Exception:
                pass

    # Phase 4: Final polish
    if time.time() - t0 < budget_s - 10:
        _g_mode = 'smooth'
        _g_beta = 5000.0
        try:
            r = minimize(obj_func, best_h, method='SLSQP', bounds=bounds,
                         constraints=cons, options={'maxiter': 10000, 'ftol': 1e-18})
            h_new = project_h(r.x, target)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new, c5
        except Exception:
            pass
        _g_mode = 'topk'
        _g_topk = 1
        try:
            r = minimize(obj_func, best_h, method='SLSQP', bounds=bounds,
                         constraints=cons, options={'maxiter': 5000, 'ftol': 1e-16})
            h_new = project_h(r.x, target)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new, c5
        except Exception:
            pass

    return best_h, best_c5, n
