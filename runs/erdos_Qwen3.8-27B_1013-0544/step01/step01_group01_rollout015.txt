def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    return float(m + np.log(np.sum(np.exp(beta * (x - m)))) / beta)

def compute_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    w = np.exp(beta * (corr - m))
    sw = w.sum()
    if sw < 1e-300:
        return np.zeros(n)
    w /= sw
    grad = np.zeros(n)
    for mi in range(2 * n - 1):
        wm = w[mi]
        if wm < 1e-14:
            continue
        k = mi - (n - 1)
        if k >= 0:
            grad[k:] += wm * (1.0 - h[:n - k])
            grad[:n - k] -= wm * h[k:]
        else:
            kp = -k
            grad[:n - kp] += wm * (1.0 - h[kp:])
            grad[kp:] -= wm * h[:n - kp]
    return grad * dx

def project_h(h, target_sum):
    h = h.astype(float)
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

    return val
