                best_h = pert
                best_n = n_best

def project_feasible(x, target_sum):
    lo = float(x.min()) - 1.5
    hi = float(x.max()) + 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = float(np.clip(x - mid, 0.0, 1.0).sum())
        if s > target_sum:
            lo = mid
        else:
            hi = mid

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx
