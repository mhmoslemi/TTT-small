    """
    Solve: minimize max_j (slopes[j]*t + intercepts[j]) subject to 0<=t<=1
    Optionally include a quadratic term: quad_coef*t^2 + t + quad_const
    (the k=0 term contributes -t^2 + t + const)
    
    Uses bisection on the value c.
    """
    # Binary search on c
    # Lower bound: min of max at t=0 and t=1
    f0 = np.max(intercepts)  # at t=0
    f1 = np.max(slopes + intercepts)  # at t=1
    if quad_coef is not None:
        f0 = max(f0, quad_const)  # -0 + 0 + const
        f1 = max(f1, -1 + 1 + quad_const)  # -1 + 1 + const = quad_const
    
    lo = min(f0, f1) - 1.0
    hi = max(f0, f1) + 1.0
    
    # Check if value c is feasible
    def check(c):
        # For each linear constraint: slopes[j]*t + intercepts[j] <= c
        # If slopes[j] > 0: t <= (c - intercepts[j]) / slopes[j]

        # If slopes[j] == 0: need intercepts[j] <= c

                return False

                else:
                    return False

        return True

    for _ in range(60):

        if check(mid):
            hi = mid
        else:
            lo = mid
    
    t_opt = hi  # feasible value

    # Let's just use t_opt = hi and find the corresponding t
    # Actually, we want the t that minimizes the max, which is at the "tightest" constraint
    
    # For the optimal c, find feasible t
    t_lo = 0.0
    t_hi = 1.0
    c = hi
    for j in range(len(slopes)):
        if slopes[j] > 1e-12:
            t_hi = min(t_hi, (c - intercepts[j]) / slopes[j])
        elif slopes[j] < -1e-12:
            t_lo = max(t_lo, (c - intercepts[j]) / slopes[j])
    
    if quad_coef is not None:
        d = 1.0 - 4.0 * (c - quad_const)
        if d >= 0:

        tol = max(0.005, c_max * 0.001)
        active = np.where(corr >= c_max - tol)[0]
        if len(active) < 2:
            active = np.argsort(corr)[-2:]
        elif len(active) > 15:
            active = np.argsort(corr)[-15:]
        
        grad = np.zeros(n)
        for idx in active:
            L = int(idx) - n_half
            grad += compute_grad_lag(h, L)
        grad /= float(len(active))
        
        momentum = mu * momentum - step_size * grad
        h_new = project_feasible(h + momentum, target_sum)
        c5_new = compute_c5(h_new, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            stag = 0
        else:
            stag += 1
            if stag % 50 == 0:
                step_size *= 0.88
            if step_size < 1e-9:
                break
        h = h_new
    
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    def tleft():
        return (start_time + budget_s - 5.0) - time.time()
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    except NameError:
        h_parent = None
    
    n_candidates = [70, 90, 100, 120]
    
    for n_points in n_candidates:
        if tleft() < 80.0:
            break
        
        dx = 2.0 / n_points
        target_sum = n_points / 2.0
        
        starts = []
        
        if h_parent is not None:
            if len(h_parent) == n_points:
                starts.append(h_parent.copy())
            else:
                xo = np.linspace(0, 2, len(h_parent))
                xn = np.linspace(0, 2, n_points)
                starts.append(np.interp(xn, xo, h_parent))
        
        if best_h is not None and best_n != n_points:
            xo = np.linspace(0, 2, best_n)
            xn = np.linspace(0, 2, n_points)
            starts.append(np.interp(xn, xo, best_h))
        
        x = np.linspace(0, 2, n_points)
        for freq in [1, 2, 3, 4, 5, 6, 7, 8]:
            for phase in [0.0, np.pi * 0.5, np.pi]:
                h_f = 0.5 + 0.35 * np.sin(2.0 * np.pi * freq * x / 2.0 + phase)
                starts.append(h_f.copy())
        
        for _ in range(4):
            starts.append(rng.uniform(0, 1, n_points))
        
        for _ in range(3):
            starts.append(rng.choice([0.0, 1.0], size=n_points))
        
        for h_start in starts:
            if tleft() < 60.0:
                break
            
            h_start = project_feasible(h_start, target_sum)
            c5_s = compute_c5(h_start, dx)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_start.copy()
                best_n = n_points
            
            if tleft() < 50.0:
                break
            
            # Phase A: Coordinate descent
            h_cd = h_start.copy()
            h_cd, c5_cd = coord_descent(h_cd, n_points, target_sum, dx,
                                         max_sweeps=200, time_limit_fn=tleft)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()
                best_n = n_points
            
            if tleft() < 40.0:
                break
            
            # Phase B: Active set gradient from CD result
            h_as = h_cd.copy()
            h_as, c5_as = active_set_gradient(h_as, n_points, target_sum, dx, tleft)
            if c5_as < best_c5:
                best_c5 = c5_as
                best_h = h_as.copy()
                best_n = n_points
            
            if tleft() < 35.0:
                break
            
            # Phase C: SLSQP refinement
            def obj_sl(x):
                return compute_c5(x, dx)
            def eq_fun(x):
                return np.sum(x) - target_sum
            def eq_jac(x):
                return np.ones(n_points)
            
            try:
                res = minimize(obj_sl, h_cd.copy(), method='SLSQP',
                              bounds=[(0.0, 1.0)] * n_points,
                              constraints=[{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}],
                              options={'maxiter': 2000, 'ftol': 1e-14})
                if res.x is not None:
                    h_res = project_feasible(res.x, target_sum)
                    c5_res = compute_c5(h_res, dx)
                    if c5_res < best_c5:
                        best_c5 = c5_res
                        best_h = h_res.copy()
                        best_n = n_points
            except Exception:
                pass
            
            if tleft() < 30.0:
                break
            
            # Phase D: More coordinate descent from SLSQP result
            if tleft() > 30.0:
                h_cd2 = best_h.copy() if best_n == n_points else h_cd.copy()
                h_cd2, c5_cd2 = coord_descent(h_cd2, n_points, target_sum, dx,
                                               max_sweeps=100, time_limit_fn=tleft)
                if c5_cd2 < best_c5:
                    best_c5 = c5_cd2
                    best_h = h_cd2.copy()
                    best_n = n_points
    
    # Final polish with coordinate descent on best solution
    if best_h is not None and tleft() > 10.0:
        n_final = best_n
        dx_final = 2.0 / n_final
        ts_final = n_final / 2.0
        h_final, c5_final = coord_descent(best_h.copy(), n_final, ts_final, dx_final,
                                           max_sweeps=500, time_limit_fn=tleft)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()
            best_n = n_final
    
    return (best_h, best_c5, best_n)
