import numpy as np
import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= s
    grad = np.zeros(n)
    active = np.where(w > 1e-10)[0]
    for k_idx in active:
        k = k_idx - (n - 1)
        wk = w[k_idx]
        if k >= 0:
            grad[k:] += wk * (1.0 - h[:n - k])
            if k > 0:
                grad[:n - k] -= wk * h[k:]
        else:
            p = -k
            grad[:n - p] += wk * (1.0 - h[p:])
            if p > 0:
                grad[p:] -= wk * h[:n - p]
    return grad * (2.0 / n)

def exact_subgrad_step(h, target_sum, lr):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    k_idx = np.argmax(corr)
    k = k_idx - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n - k])
        if k > 0:
            g[:n - k] -= h[k:]
    else:
        p = -k
        g[:n - p] += (1.0 - h[p:])
        if p > 0:
            g[p:] -= h[:n - p]
    return project(h - lr * g * (2.0 / n), target_sum)

def interpolate_h(h_old, n_old, n_new):
    x_old = (np.arange(n_old) + 0.5) / n_old * 2.0
    x_new = (np.arange(n_new) + 0.5) / n_new * 2.0
    h_ext = np.append(h_old, h_old[0])

    return np.interp(x_new, x_ext, h_ext)
