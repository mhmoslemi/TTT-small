def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')

def grad_corr_single(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n - k])
        if k > 0:

    else:
        p = -k
        g[:n - p] += (1.0 - h[p:])
        if p > 0:

    return g

def gradient_smooth_max(h, corr, top_k=5):
    """Compute gradient of smooth max using top_k lags"""
    n = len(h)
    vals = corr.copy()
    # Find top_k indices
    idx = np.argpartition(-vals, top_k - 1)[:top_k]
    idx = idx[np.argsort(-vals[idx])]
    # Weight by exponential
    max_val = vals[idx[0]]
    weights = np.exp(10 * (vals[idx] - max_val))
    weights /= weights.sum()
    grad = np.zeros(n)
    for i, k_idx in enumerate(idx):
        k = k_idx - (n - 1)
        grad += weights[i] * grad_corr_single(h, k)
    return grad * (2.0 / n)
