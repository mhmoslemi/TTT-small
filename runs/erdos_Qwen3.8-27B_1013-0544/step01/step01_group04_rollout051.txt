    h[i] = t
    corr = np.correlate(h, 1.0 - h, mode='full')
    val = np.max(corr) * dx
    return val

    """Coordinate descent for the Erdős problem"""
    target = n / 2.0
    best_c5 = compute_c5(h, dx)
    n_improve = 0

        improved = False
        # Random order for diversity

        for i in order:
            old_val = h[i]
            # Minimize over h[i] ∈ [0,1]
            def obj(t):
                h[i] = t
                corr = np.correlate(h, 1.0 - h, mode='full')
                return np.max(corr) * dx

            h_work = best_h.copy()
        else:
            scale = 0.1 * (1.0 - restart / 20.0) + 0.02
            h_work = best_h + rng.randn(n) * scale
            h_work = project_feasible(h_work, n)
        h_cd, c5_cd = coord_descent(h_work, n, dx, rng, p1_end - t0, t0)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 2: L-BFGS-B with smoothed max (25% time)
    p2_end = budget_s * 0.60
    if elapsed() < p2_end:
        h_lf, c5_lf = lfgsb_optimize(best_h, n, dx, p2_end - t0, t0)
        if c5_lf < best_c5:
            best_c5 = c5_lf
            best_h = h_lf

    # Phase 3: SLSQP polishing (15% time)
    p3_end = budget_s * 0.78
    while elapsed() < p3_end:
        h_try = best_h + rng.randn(n) * 0.01
        h_try = project_feasible(h_try, n)
        h_sl = slsqp_once(h_try, n, dx, max_iter=400)
        c5_sl = compute_c5(h_sl, dx)
        if c5_sl < best_c5:
            best_c5 = c5_sl
            best_h = h_sl

    # Phase 4: Upsample to 2n and optimize (20% time)
    p4_end = budget_s * 0.95
    if elapsed() < p4_end:
        n2 = 2 * n
        dx2 = 2.0 / n2
        h_up = np.repeat(best_h, 2).astype(float)
        h_up = project_feasible(h_up, n2)
        c5_up = compute_c5(h_up, dx2)

        # Coordinate descent at higher res
        h_up2, c5_up2 = coord_descent(h_up, n2, dx2, rng, p4_end - 120 - t0, t0)
        if c5_up2 < best_c5:
            best_c5 = c5_up2
            best_h = h_up2

        # SLSQP on higher res
        if elapsed() < p4_end:
            h_up3 = slsqp_once(h_up2, n2, dx2, max_iter=200)
            c5_up3 = compute_c5(h_up3, dx2)
            if c5_up3 < best_c5:
                best_c5 = c5_up3
                best_h = h_up3

    # Final polish
    if elapsed() < budget_s - 5:
        n_f = len(best_h)
        dx_f = 2.0 / n_f
        h_f = project_feasible(best_h + rng.randn(n_f) * 0.005, n_f)
        h_f = slsqp_once(h_f, n_f, dx_f, max_iter=500)
        c5_f = compute_c5(h_f, dx_f)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return (best_h, best_c5, len(best_h))
