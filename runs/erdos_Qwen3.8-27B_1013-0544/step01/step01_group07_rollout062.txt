def lse_obj(h, t, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    return (m + (1.0/t) * np.log(np.sum(np.exp(t*(corr - m))))) * dx

    h += (target - np.sum(h)) / len(h)
    return np.clip(h, 0.0, 1.0)

    """Project onto {h: sum(h)=target, 0<=h<=1}"""
    # This is a quadratic projection
    n = len(h)
    # Simple approach: bisection on lambda
    # sum(clip(h - lambda, 0, 1)) = target
    lo, hi = h.min() - 1, h.max() + 1

        s = np.sum(np.clip(h - mid, 0, 1))

        else:

    corr = np.correlate(h, 1 - h, mode="full")
    max_idx = np.argmax(corr)

Let me also think about the subgradient more carefully.

For the full correlation:
result[k] = sum_{i} a[i] * v[i + k - (n-1)] where the product is only counted when 0 <= i+k-(n-1) < n

Let me define offset = k - (n-1), so:

def proj_sum(h, target):
    """Project h onto {x: sum(x)=target, 0<=x[i]<=1} using bisection"""
    n = len(h)
    lo = h.min() - 2.0
    hi = h.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
