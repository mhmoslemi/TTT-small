```python
def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_sum(h, target_sum):
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(60):
        mu = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mu, 0.0, 1.0))
        if s > target_sum:
            lo = mu
        else:
            hi = mu
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def coord_descent_pass(h, n, target_sum, rng, mu=0.0):

    order = rng.permutation(n)
    lags = np.arange(-(n-1), n)
    corr_len = 2 * n - 1
    
    for j in order:
        x_curr = h[j]
        corr_current = np.correlate(h, 1 - h, mode="full")
        
        jp = j + lags
        jm = j - lags
        mask_p = (jp >= 0) & (jp <= n - 1)
        mask_m = (jm >= 0) & (jm <= n - 1)
```

```python
def compute_gradient(h, n, t):
    corr = np.correlate(h, 1-h, mode="full")
    dx = 2.0/n
    # Softmax weights
    shifted = t * corr * dx - np.max(t * corr * dx)
    w = np.exp(shifted) / np.sum(np.exp(shifted))
    
    # Gradient
    grad = np.zeros(n)
    lags = np.arange(-(n-1), n)
    for j in range(n):
        jp = j + lags
        jm = j - lags
        mask_p = (jp >= 0) & (jp <= n-1)
        mask_m = (jm >= 0) & (jm <= n-1)
        h_jp = np.zeros(len(lags))
        h_jp[mask_p] = h[jp[mask_p]]
        h_jm = np.zeros(len(lags))
        h_jm[mask_m] = h[jm[mask_m]]
        a = (1-h_jp)*mask_p - h_jm*mask_m
        grad[j] = dx * np.dot(w, a)
    return grad
```

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_sum(h, target_sum):
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mu = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mu, 0.0, 1.0))
        if s > target_sum:
            lo = mu
        else:
            hi = mu
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _compute_a(h, j, n, lags):
    jp = j + lags
    jm = j - lags
    mask_p = (jp >= 0) & (jp <= n - 1)
    mask_m = (jm >= 0) & (jm <= n - 1)
    h_jp = np.zeros(2 * n - 1)
    h_jp[mask_p] = h[jp[mask_p]]
    h_jm = np.zeros(2 * n - 1)
    h_jm[mask_m] = h[jm[mask_m]]
    return (1.0 - h_jp) * mask_p - h_jm * mask_m, mask_p, mask_m

def coord_minimize(h, j, n, lags, corr_current, mu=0.0):
    x_curr = h[j]
    a, _, _ = _compute_a(h, j, n, lags)
    b = corr_current - a * x_curr
    a_eff = a + mu
    lo_b, hi_b = 0.0, 1.0
    for _ in range(55):
        m1 = lo_b + (hi_b - lo_b) / 3.0
        m2 = hi_b - (hi_b - lo_b) / 3.0
        if np.max(a_eff * m1 + b) < np.max(a_eff * m2 + b):
            hi_b = m2
        else:
            lo_b = m1
    return (lo_b + hi_b) / 2.0

def coord_descent_pass(h, n, target_sum, rng, mu=0.0):
