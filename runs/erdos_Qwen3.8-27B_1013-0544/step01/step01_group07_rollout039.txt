The subdifferential of f at h is the convex hull of {∇f_k : f_k = f}. So a valid subgradient is:

Then projected subgradient:
h_new = P_C(h - lr * g)
where P_C is projection onto {h : 0 ≤ h ≤ 1, sum h = target_sum}

The projection is:
1. Clip to [0,1]
2. Adjust to satisfy sum constraint (add constant, then re-clip, iterate)

This is much simpler than SLSQP! And with Nesterov momentum, it can converge fast.

def project_onto_constraint(h, target_sum):
    """Project h onto {x : 0 ≤ x ≤ 1, sum x = target_sum}."""
    # Simple approach: iteratively adjust
    h = h.copy()
    for _ in range(20):
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break

        if np.sum(free) == 0:
            h = np.clip(h, 0.0, 1.0)
            break

    max_c = np.max(corr)

    return (max_c + (1.0 / t) * np.log(np.sum(np.exp(shifted)))) * dx

def compute_subgrad(h, dx, tau, top_lags=20):
    """Compute weighted subgradient of max_k f_k(h)."""
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    max_c = np.max(corr)
    shifted = (corr - max_c) / tau
    shifted = np.clip(shifted, -50, 0)
    weights = np.exp(shifted)
    total_w = np.sum(weights)
    if total_w < 1e-30:
        weights = np.zeros_like(corr)
        weights[np.argmax(corr)] = 1.0
        total_w = 1.0
    weights /= total_w
    
    # Only process top lags
    top_idx = np.argsort(weights)[::-1][:top_lags]
    
    grad = np.zeros(n)
    for k in top_idx:
        if weights[k] < 1e-10:
            continue
        lag = k - (n - 1)
        w = weights[k] * dx
        if lag >= 0:
            if n - lag > 0:
                grad[:n-lag] += w * (1.0 - h[lag:])
            if lag < n:
                grad[lag:] -= w * h[:n-lag]
        else:
            L = -lag
            if n - L > 0:
                grad[L:] += w * (1.0 - h[:n-L])
                grad[:n-L] -= w * h[L:]
    
    c5 = max_c * dx
    return grad, c5

def nesterov_subgrad(h, dx, target_sum, n_iters, tau, lr_init, momentum):
    """Nesterov accelerated projected subgradient descent."""
    n = len(h)
    h_cur = h.copy()
    v = np.zeros(n)
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur, dx)
    
    for it in range(n_iters):
        lr = lr_init / (1.0 + it * 0.001)
        
        # Nesterov lookahead
        y = h_cur + momentum * v
        
        # Compute subgradient at y
        grad, c5_y = compute_subgrad(y, dx, tau)
        
        # Update velocity
        v = momentum * v - lr * grad
        
        # Update position
        h_new = h_cur + v
        h_new = project_constraint(h_new, target_sum)
        
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        
        h_cur = h_new
    
    return best_h, best_c5
