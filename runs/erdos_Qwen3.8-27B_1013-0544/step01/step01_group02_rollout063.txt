def smooth_max_obj(h, n, alpha):
    """Smooth approximation of C5."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()

    mx = shifted.max()

        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha)
    val = np.sum(np.exp(s))
    return float(m + np.log(val) / alpha)

def grad_smooth(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 500:
        w = np.exp(s - mx)
        w /= w.sum()
    else:
        w = np.exp(s)
        w /= w.sum()
    # w is the softmax weights over the 2n-1 correlation values
    # grad of smooth_obj = sum_k w[k] * grad_R(k) * dx
    # where R(k) = correlate(h, 1-h, full)[k]
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-10:
            continue
        lag = k - (n - 1)
        g = np.zeros(n)
        if lag >= 0:
            g[lag:] += (1.0 - h[:-lag])
            g[:n-lag] -= h[lag:]
        else:
            L = -lag
            g[:n-L] += (1.0 - h[L:])
            g[L:] -= h[:n-L]
        grad += w[k] * g * dx
    return grad

def fista(h_init, n, alpha, max_iter=5000, tol=1e-12):
    """FISTA for min smooth_obj(h) s.t. 0<=h<=1, sum(h)=n/2"""

    y = h.copy()

    L = 4.0 * alpha / n  # rough Lipschitz estimate
    prev_f = smooth_obj(y, n, alpha)

        x_new = y - g / L
        x_new = project_feasible(x_new, n)
        f_new = smooth_obj(x_new, n, alpha)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x_new + ((t - 1.0) / t_new) * (x_new - h)
        y = np.clip(y, 0.0, 1.0)
        h = x_new
        t = t_new
