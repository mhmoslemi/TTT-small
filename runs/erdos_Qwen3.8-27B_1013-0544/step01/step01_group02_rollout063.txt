def two_coord_sweep_fast(h, n, n_sweeps=5, deadline=None):
    dx = 2.0 / n
    h = h.copy()
    
    for sweep in range(n_sweeps):
        if deadline and time.time() > deadline:
            break
        c = np.correlate(h, 1-h, mode='full')
        cur_max = c.max()
        improved = False
        
        # Precompute shift effects for all positions
        # For position p, the "plus" effect (as first factor) at shift s: (1-h[p+s]) if valid
        # The "minus" effect (as second factor) at shift s: h[p-s] if valid
        plus_eff = np.zeros((n, 2*n-1))  # plus_eff[p, k] = (1-h[p+s_k])_valid
        minus_eff = np.zeros((n, 2*n-1))  # minus_eff[p, k] = h[p-s_k]_valid
        
        for p in range(n):
            for k in range(2*n-1):
                s = k - (n-1)
                if 0 <= p+s < n:
                    plus_eff[p, k] = 1.0 - h[p+s]
                if 0 <= p-s < n:
                    minus_eff[p, k] = h[p-s]
        
        # For pair (i,j): effect = plus_eff[i] - minus_eff[i] - plus_eff[j] + minus_eff[j]
        for i in range(n):
            ei = plus_eff[i] - minus_eff[i]
            for j in range(i+1, n):
                effect = ei - plus_eff[j] + minus_eff[j]
                lo = max(-h[i], h[j]-1.0)
                hi = min(1.0-h[i], h[j])
                if lo >= hi:
                    continue
                # Minimize max(c + delta*effect) over delta in [lo, hi]
                # Check boundaries
                best_d = 0.0
                best_v = cur_max
                for d in (lo, hi):
                    v = (c + d*effect).max()
                    if v < best_v:
                        best_v = v
                        best_d = d
                # Check equalization of top 2
                new_c = c + best_d * effect
                top = np.argpartition(new_c, -2)[-2:]
                k1, k2 = top
                if abs(effect[k1] - effect[k2]) > 1e-12:
                    d_eq = (c[k2] - c[k1]) / (effect[k1] - effect[k2])
                    if lo < d_eq < hi:

                        if v < best_v:

                if best_d != 0.0 and best_v < cur_max - 1e-12:
                    h[i] += best_d
                    h[j] -= best_d
                    cur_max = best_v
                    improved = True
                    # Update c
                    c = c + best_d * effect
        
        if not improved:
            break

    ev = np.exp(sh - mx)
    w = ev / ev.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        s = k - (n - 1)
        wk = w[k] * dx
        if wk < 1e-14:
            continue
        a, b = max(0, -s), min(n, n - s)
        c2, d2 = max(0, s), min(n, n + s)
        if b > a:
            g[a:b] += wk * (1.0 - h[a+s:b+s])
        if d2 > c2:
            g[c2:d2] -= wk * h[c2-s:d2-s]
    return g

def slsqp(h, n, alpha, mi, dl):
    def obj(x):
        return smo_obj(x, n, alpha)
    def jac(x):
        return smo_grad(x, n, alpha)
    res = minimize(obj, h, jac=jac, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
                                 'jac': lambda x: np.ones(n)}],
                   options={'maxiter': mi, 'ftol': 1e-16})
    if time.time() > dl:
        return h
    return project_feasible(res.x, n)

def tc_sweep(h, n, nsw, dl):
    h = h.copy()
    for sw in range(nsw):
        if dl and time.time() > dl:
            break
        c = np.correlate(h, 1 - h, mode='full')
        mx = c.max()
        imp = False
        pe = np.zeros((n, 2*n-1))
        me = np.zeros((n, 2*n-1))
        for p in range(n):
            for k in range(2*n-1):
                s = k - (n-1)
                if 0 <= p+s < n:
                    pe[p, k] = 1.0 - h[p+s]
                if 0 <= p-s < n:
                    me[p, k] = h[p-s]
        for i in range(n):
            ei = pe[i] - me[i]
            for j in range(i+1, n):
                ef = ei - pe[j] + me[j]
                lo = max(-h[i], h[j]-1.0)
                hi = min(1.0-h[i], h[j])
                if lo >= hi:
                    continue
                bd, bv = 0.0, mx
                for d in (lo, hi):
                    v = (c + d*ef).max()
                    if v < bv:
                        bv, bd = v, d
                nc = c + bd*ef
                topk = min(5, 2*n-1)
                tidx = np.argpartition(nc, -topk)[-topk:]
                cands = [lo, hi, 0.0]
                for a2 in range(len(tidx)):
                    for b2 in range(a2+1, len(tidx)):
                        k1, k2 = tidx[a2], tidx[b2]
                        if abs(ef[k1]-ef[k2]) > 1e-12:
                            de = (c[k2]-c[k1])/(ef[k1]-ef[k2])
                            if lo < de < hi:
                                cands.append(de)
                for d in cands:
                    v = (c + d*ef).max()
                    if v < bv - 1e-14:
                        bv, bd = v, d
                if bd != 0.0 and bv < mx - 1e-12:
                    h[i] += bd
                    h[j] -= bd
                    c = c + bd*ef
                    mx = bv
                    imp = True
        if not imp:
            break
    return h

def optimize_path(h, n, dl, rng):
    h = project_feasible(h, n)
    for alpha in [100, 300, 1000, 3000, 10000]:
        if dl - time.time() < 12:
            break
        h = slsqp(h, n, alpha, 1200, dl)
    if dl - time.time() > 10:
        h = tc_sweep(h, n, 8, dl)
    if dl - time.time() > 8:
        h = slsqp(h, n, 10000, 800, dl)
    if dl - time.time() > 5:
        h = tc_sweep(h, n, 5, dl)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s
    rng = np.random.RandomState(seed)

    try:
        best_h = np.array(initial_h_values, dtype=float).copy()
        n = len(best_h)
    except Exception:
        n = 80
        best_h = np.full(n, 0.5)

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)

    # Phase 1: Optimize from initial
    if dl - time.time() > 40:
        h1 = optimize_path(best_h, n, dl, rng)
        c1 = compute_c5(h1, n)
        if c1 < best_c5:
            best_h, best_c5 = h1, c1

    # Phase 2: Multi-start with diverse initializations
    starts = []
    xb = np.linspace(0, 2, n, endpoint=False)
    # Binary-like
    hb = np.zeros(n)
    hb[rng.choice(n, n//2, replace=False)] = 1.0
    starts.append(project_feasible(hb + rng.normal(0, 0.12, n), n))
    # Sinusoidal variants
    for f in [1, 2, 3, 4, 5, 6, 7, 8]:
        hs = 0.5 + 0.25*np.sin(2*np.pi*f*xb/2.0 + rng.uniform(0, 6.28))
        starts.append(project_feasible(hs, n))
    # Random smooth
    for _ in range(4):
        cr = rng.uniform(0.1, 0.4)
        hr = project_feasible(rng.uniform(0.2, 0.8, n), n)
        starts.append(hr)
    # Block patterns
    for nb in [2, 3, 4, 5]:
        hblk = np.zeros(n)
        for b in range(nb):
            sl = slice(b*n//nb, (b+1)*n//nb)
            hblk[sl] = rng.uniform(0.3, 0.7)
        starts.append(project_feasible(hblk, n))

    for h_s in starts:
        rem = dl - time.time()
        if rem < 25:
            break
        h_cur = optimize_path(h_s, n, dl, rng)
        c = compute_c5(h_cur, n)
        if c < best_c5:
            best_h, best_c5 = h_cur, c

    # Phase 3: Iterated perturbation + local optimization
    attempt = 0
    while dl - time.time() > 25:
        attempt += 1
        sc = rng.choice([0.005, 0.01, 0.02, 0.03, 0.05, 0.08, 0.12, 0.2, 0.3, 0.4])
        hp = project_feasible(best_h + rng.normal(0, sc, n), n)
        hp = optimize_path(hp, n, dl, rng)
        c = compute_c5(hp, n)
        if c < best_c5:
            best_h, best_c5 = hp, c

    # Phase 4: Final aggressive TC sweep + SLSQP
    if dl - time.time() > 5:
        best_h = tc_sweep(best_h, n, 20, dl)
    if dl - time.time() > 5:
        best_h = slsqp(best_h, n, 20000, 3000, dl)
    if dl - time.time() > 3:
        best_h = tc_sweep(best_h, n, 10, dl)

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
