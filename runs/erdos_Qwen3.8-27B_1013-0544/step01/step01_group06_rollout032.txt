```python
        if s == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[:n-s] += (1.0 - h[s:])
            g[s:] -= h[:n-s]
    else:
        p = -s
        # c(-p) = sum_{j=0}^{n-1-p} h[j+p](1-h[j])
        # grad[j] = (1-h[j-p])*1[j>=p] - h[j+p]*1[j<n-p]
        g[p:] += (1.0 - h[:n-p])
        g[:n-p] -= h[p:]
    return g
```

Let me verify the s=0 case: c(0) = Σ h[i](1-h[i]) = Σ h[i] - h[i]²
∂c(0)/∂h[j] = 1 - 2h[j]. ✓

And the s>0 case matches what I had before. The s<0 case is the new formula.

Now, in the `np.correlate(h, 1-h, mode='full')` output:
- Index 0 corresponds to s = 0 - (n-1) = -(n-1)
- Index n-1 corresponds to s = 0
- Index 2n-2 corresponds to s = n-1

So k_idx = n-1+s.

Let me make sure my gradient function uses this convention. Yes, `s = k_idx - (n-1)` gives the correct shift.

Now let me write the complete solution. I'll aim for clarity and efficiency.

One more optimization idea: instead of computing the full correlation (O(n²)) at every step, I can compute only the correlations at the "active" shifts. But for n ≤ 300, the full correlation is fast enough (numpy's correlate is optimized in C).

Let me also think about whether I should use the log-sum-exp with a very large alpha (essentially the max) or a moderate alpha (smooth approximation). 

At the start, a moderate alpha (like 50-100) gives a smooth landscape that's easier to navigate. As we get closer to the optimum, increasing alpha makes the objective sharper and helps us converge to the true max.

But there's a trade-off: with too large alpha, the gradient is dominated by a single shift and we might oscillate.

My strategy: start with alpha=80, topk_frac=0.5, and gradually increase alpha while decreasing topk_frac.

Let me finalize the code now. I'll keep it concise and under 300 lines.

```python
import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    return np.max(compute_corr(h)) * (2.0 / len(h))

def grad_single_shift(h, k_idx):
    n = len(h)
    s = k_idx - (n - 1)
    g = np.zeros(n)
    if s == 0:
        g[:] = 1.0 - 2.0 * h
    elif s > 0:
        g[:n-s] += (1.0 - h[s:])
        g[s:] -= h[:n-s]
    else:
        p = -s
        g[p:] += (1.0 - h[:n-p])
        g[:n-p] -= h[p:]
    return g

def project_box_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def get_weights(corr, alpha, topk_frac):
    n = len(corr)
    m = corr.max()
    w = np.exp(alpha * (corr - m))
    if topk_frac < 1.0:
        k_count = max(1, int(topk_frac * n))
        thresh_idx = n - k_count
        threshold = np.partition(w, thresh_idx)[thresh_idx]
        w[w < threshold] = 0.0
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= s
    return w

def weighted_grad(h, weights):
    n = len(h)
    g = np.zeros(n)
    active = np.where(weights > 1e-8)[0]
    for idx in active:
        g += weights[idx] * grad_single_shift(h, idx)
    return g * (2.0 / n)

def smooth_max_obj(h, alpha):
    corr = compute_corr(h)
    m = corr.max()
    return (m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha) * (2.0 / len(h))

def optimize_gradient(h0, target, deadline, rng, alpha0=100, topk0=0.5):
    n = len(h0)
    h = project_box_sum(h0.astype(float).copy(), target)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = 0.004
    alpha = alpha0
    topk = topk0
    stall = 0
    restarts = 0
    no_improve = 0

    while time.time() < deadline and restarts < 15:
        corr = compute_corr(best_h)
        w = get_weights(corr, alpha, topk)
        g = weighted_grad(best_h, w)

        step = lr
        accepted = False
        for _ in range(10):
            h_new = project_box_sum(best_h - step * g, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-16:
                best_h = h_new
                best_c5 = c5_new
                no_improve = 0
                accepted = True
                break
            step *= 0.5

        if accepted:
            stall = 0
            lr = min(lr * 1.025, 0.1)
            topk = max(topk * 0.999, 0.03)
            alpha = min(alpha * 1.004, 800.0)
        else:
            stall += 1
            lr *= 0.7
            if stall > 30:
                size = 0.02 + 0.015 * restarts
                h_p = best_h + rng.randn(n) * size
                h_p = project_box_sum(h_p, target)
                c5_p = compute_c5(h_p)
                if c5_p < best_c5:
                    best_h = h_p
                    best_c5 = c5_p
                    no_improve = 0
                alpha = max(40.0, alpha * 0.5)
                topk = min(1.0, topk * 2.5)
                lr = 0.004
                stall = 0
                restarts += 1
                no_improve += 1
                if no_improve > 3 and restarts > 8:
                    break
    return best_h, best_c5

def optimize_scipy(h0, target, deadline, rng):
    n = len(h0)
    h = project_box_sum(h0.astype(float).copy(), target)
    best_c5 = compute_c5(h)
    best_h = h.copy()

    for alpha in [50, 100, 200, 400]:
        if time.time() > deadline:
            break
        def obj(x):
            return smooth_max_obj(x, alpha)
        def obj_grad(x):
            corr = compute_corr(x)
            m = corr.max()
            w = np.exp(alpha * (corr - m))
            w /= w.sum()
            return weighted_grad(x, w)

        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                 'jac': lambda x: np.ones(n)}]
        bounds = [(0.0, 1.0)] * n

        try:
            res = scipy_minimize(obj, h, jac=obj_grad, method='SLSQP',
                                 bounds=bounds, constraints=cons,
                                 options={'maxiter': 200, 'ftol': 1e-12})
            if res.success:
                h_new = project_box_sum(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new
        except Exception:
            pass
    return best_h, best_c5

def local_search(h, target, deadline, rng, max_rounds=5000):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    accepted = 0

    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        delta = rng.uniform(0.01, 0.05) * rng.choice([-1, 1])
        h_new = best_h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        if h_new[i] < 0 or h_new[i] > 1 or h_new[j] < 0 or h_new[j] > 1:
            continue
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_h = h_new
            best_c5 = c5_new
            accepted += 1
    return best_h, best_c5

def interp_to_n(h_old, n_new):
    n_old = len(h_old)
    x_old = np.arange(n_old + 1) / n_old
    h_ext = np.concatenate([h_old, [h_old[0]]])
    x_new = (np.arange(n_new) + 0.5) / n_new
    return np.interp(x_new, x_old, h_ext)

def make_candidates(n, rng, h_parent):
    cands = []
    if h_parent is not None:
        if len(h_parent) == n:
            cands.append(h_parent.astype(float).copy())
        else:
            cands.append(interp_to_n(h_parent, n))
    x = np.arange(n) / n
    for f in [1, 2, 3]:
        for p in np.linspace(0, 2*np.pi, 6, endpoint=False):
            cands.append(project_box_sum(0.5 + 0.35*np.sin(2*np.pi*f*x + p), n/2.0))
    for _ in range(4):
        cands.append(project_box_sum(0.3 + 0.4*rng.rand(n), n/2.0))
    for nb in [4, 6, 8, 12]:
        step = n // nb
        for off in range(min(step, 3)):
            h = np.zeros(n)
            for b in range(nb):
                lo_i = b * step + off
                hi_i = min((b+1) * step + off, n)
                if b % 2 == 0:
                    h[lo_i:hi_i] = 1.0
            cands.append(project_box_sum(h, n/2.0))
    return cands

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h_parent = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float)

    best_h = None
    best_c5 = np.inf

    n_list = [100, 150, 200, 250, 300]
    if h_parent is not None:
        if len(h_parent) not in n_list:
            n_list.insert(0, len(h_parent))
        n_list = sorted(set(n_list))

    time_budget = budget_s * 0.75
    n_res = len(n_list)

    for idx, n in enumerate(n_list):
        elapsed = time.time() - t0
        if elapsed > budget_s * 0.82:
            break
        time_left = budget_s * 0.82 - elapsed
        alloc = min(time_left / (n_res - idx), 250.0)
        if alloc < 15:
            break

        cands = make_candidates(n, rng, h_parent)
        best_init = None
        best_init_c5 = np.inf
        for c in cands:
            c = project_box_sum(c, n/2.0)
            v = compute_c5(c)
            if v < best_init_c5:
                best_init_c5 = v
                best_init = c
            if v < best_c5:
                best_c5 = v
                best_h = c.copy()

        dl1 = time.time() + alloc * 0.6
        h1, c5_1 = optimize_gradient(best_init, n/2.0, dl1, rng,
                                      alpha0=80 + 30*idx, topk0=0.4 + 0.08*idx)

        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()

        if time.time() < t0 + budget_s * 0.82:
            dl2 = min(t0 + budget_s * 0.82 - 5, time.time() + alloc * 0.25)
            h2, c5_2 = optimize_scipy(h1, n/2.0, dl2, rng)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h2.copy()

    remaining = budget_s - (time.time() - t0) - 5
    if remaining > 15 and best_h is not None:
        n_b = len(best_h)
        dl = time.time() + remaining * 0.5
        h_ls, c5_ls = local_search(best_h, n_b/2.0, dl, rng)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

        remaining2 = budget_s - (time.time() - t0) - 3
        if remaining2 > 10:
            h_fin, c5_fin = optimize_gradient(best_h, n_b/2.0,
                                              time.time() + remaining2, rng,
                                              alpha0=300, topk0=0.1)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin.copy()

    if best_h is None:
        n = 100
        best_h = project_box_sum(0.5 + 0.1*rng.randn(n), n/2.0)
        best_c5 = compute_c5(best_h)

    best_h = project_box_sum(best_h, len(best_h)/2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
```
