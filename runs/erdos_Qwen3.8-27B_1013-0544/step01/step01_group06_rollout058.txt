def compute_smooth_grad(h, alpha):

    corr = np.correlate(h, 1 - h, mode='full')

def compute_smooth_grad(h, alpha):

    n = len(h)

    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -100, 0)
    w = np.exp(shifted)
    sw = np.sum(w)
    if sw < 1e-300:
        return np.zeros(n)
    w /= sw
    dx = 2.0 / n
    grad = np.zeros(n)
    # Vectorized computation
    # For lag k (index k+(n-1) in corr array):
    #   d/dh[j] = (1-h[j+k]) for j < n-k, plus -h[j-k] for j >= k
    # Let's compute using convolutions
    # grad_from_first = for each k, add (1-h[k:]) to positions [0, n-k) weighted by w[k]
    # grad_from_second = for each k, subtract h[:n-k] from positions [k, n) weighted by w[k]

    pos_w = w[n-1:]  # weights for k=0,1,...,n-1
    # For negative lags k=-(n-1),...,0 (corr indices 0,1,...,n-1):
    neg_w = w[:n]    # weights for k=-(n-1),...,0 (reversed)
    
    # For positive lag k: contributes (1-h[k:]) to grad[0:n-k] and -h[0:n-k] to grad[k:n]
    # For negative lag -k (k>0): contributes (1-h[0:n-k]) to grad[k:n] and -h[k:] to grad[0:n-k]
    
    # This is essentially a convolution-like operation. Let me just loop.

        k = idx - (n-1)

            grad[:n-k] += widx * (1.0 - h[k:])
            # Positions k to n-1: subtract h[0:n-k]
            grad[k:] -= widx * h[:n-k]
        else:

            grad[p:] += widx * (1.0 - h[:n-p])
