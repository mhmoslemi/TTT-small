```python
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    # Softmax weights
    max_c = np.max(corr)
    log_weights = t * (corr - max_c)
    w = np.exp(log_weights)
    w /= np.sum(w)
    # Gradient: grad[i] = dx * sum_k w[k] * ((1-h[i-shift_k])*[valid] - h[i+shift_k]*[valid])
    # shift_k = k - (n-1)
    shifts = np.arange(len(corr)) - (n - 1)  # shape (2n-1,)
    grad = np.zeros(n)
    for i in range(n):
        # First term: sum_k w[k] * (1-h[i-shift_k]) where 0<=i-shift_k<n
        valid1 = (i - shifts >= 0) & (i - shifts < n)

        if np.any(valid2):
            grad[i] -= np.sum(w[valid2] * h[i + shifts[valid2]])
```

```python
import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_constrained(y, target_sum):
    lo, hi = -np.max(y) - 1.0, np.min(y) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s < target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

def smooth_max_obj(h, t, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    return (mc + (1.0 / t) * np.log(np.sum(np.exp(t * (corr - mc)))) * 1.0) * dx

def smooth_grad(h, t, dx):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    e = np.exp(t * (corr - mc))
    w = e / e.sum()
    shifts = np.arange(len(corr)) - (n - 1)
    grad = np.zeros(n)
    for i in range(n):
        v1 = (i - shifts)
        m1 = (v1 >= 0) & (v1 < n)
        if m1.any():
            grad[i] += w[m1] @ (1.0 - h[v1[m1]])
        v2 = (i + shifts)
        m2 = (v2 >= 0) & (v2 < n)
        if m2.any():
            grad[i] -= w[m2] @ h[v2[m2]]
    return grad * dx

def gd_smooth(h, dx, t, target_sum, iters=3000, lr=0.01):
    n = len(h)
    best_local = h.copy()
    best_c5 = compute_c5(h, dx)
    beta1, beta2 = 0.9, 0.999
    m = np.zeros(n)
    v = np.zeros(n)
    for it in range(1, iters + 1):
        g = smooth_grad(h, t, dx)
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g * g
        mh = m / (1 - beta1 ** it)
        vh = v / (1 - beta2 ** it)
        h_raw = h - lr * mh / (np.sqrt(vh) + 1e-8)
        h = project_constrained(h_raw, target_sum)
        if it % 100 == 0:
            c5 = compute_c5(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_local = h.copy()
    return best_local, best_c5

def subgrad_step(h, dx, target_sum, alpha, K=3):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    top_k = np.argsort(corr)[-K:]
    shifts = np.arange(len(corr)) - (n - 1)
    grad = np.zeros(n)
    for idx in top_k:
        s = shifts[idx]
        if s >= 0:
            if s < n:
                grad[s:] += (1.0 - h[:n - s])
                grad[:n - s] -= h[s:]
        else:
            ss = -s
            if ss < n:
                grad[:n - ss] += (1.0 - h[ss:])
                grad[ss:] -= h[:n - ss]
    h_new = h - alpha * dx * grad / K
    return project_constrained(h_new, target_sum)

def local_refine(h, dx, target_sum, maxiter=500):
    n = len(h)
    def obj(x):
        return compute_c5(np.clip(x, 0, 1), dx)
    try:
        res = minimize(obj, h.copy(), method='Nelder-Mead',
                       options={'maxiter': maxiter, 'xatol': 1e-6, 'fatol': 1e-8})
        h_new = project_constrained(res.x, target_sum)
        return h_new
    except:
        return h

def slsqp_refine(h, dx, target_sum, t=100, maxiter=300):
    n = len(h)
    try:
        res = minimize(
            lambda x: smooth_max_obj(x, t, dx),
            h.copy(), method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            options={'maxiter': maxiter, 'ftol': 1e-10}
        )
        return project_constrained(res.x, target_sum)
    except:
        return h

def make_structured_init(n, seed_val):
    rng = np.random.RandomState(seed_val)
    # Try different structures
    choice = rng.randint(4)
    if choice == 0:
        # Block structure
        h = np.zeros(n)
        block = n // 4
        h[:block] = 1.0
        h[2*block:3*block] = 1.0
    elif choice == 1:
        # Random binary
        h = (rng.rand(n) < 0.5).astype(float)
    elif choice == 2:
        # sinusoidal threshold
        x = np.linspace(0, 2, n, endpoint=False)
        h = (np.sin(2 * np.pi * 3 * x / 2 + rng.rand() * 2) > 0).astype(float)
    else:
        # Gaussian mixture threshold
        centers = rng.uniform(0, 2, 4)
        x = np.linspace(0, 2, n, endpoint=False)
        vals = sum(np.exp(-((x - c) / 0.2)**2) for c in centers)
        h = (vals > np.median(vals)).astype(float)
    return project_constrained(h, n / 2.0)
```

```python
import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_constrained(y, target_sum):
    lo, hi = -np.max(y) - 1.0, np.min(y) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s < target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

def smooth_max_obj(h, t, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    return (mc + (1.0 / t) * np.log(np.sum(np.exp(t * (corr - mc))))) * dx

def smooth_grad(h, t, dx):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    e = np.exp(t * (corr - mc))
    w = e / e.sum()
    shifts = np.arange(len(corr)) - (n - 1)
    grad = np.zeros(n)
    for i in range(n):
        v1 = i - shifts
        m1 = (v1 >= 0) & (v1 < n)
        if m1.any():
            grad[i] += w[m1] @ (1.0 - h[v1[m1]])
        v2 = i + shifts
        m2 = (v2 >= 0) & (v2 < n)
        if m2.any():
            grad[i] -= w[m2] @ h[v2[m2]]
    return grad * dx

def gd_adam(h, dx, t, target_sum, iters=3000, lr=0.005):
    n = len(h)
    best_local = h.copy()
    best_c5 = compute_c5(h, dx)
    b1, b2 = 0.9, 0.999
    m = np.zeros(n)
    v = np.zeros(n)
    for it in range(1, iters + 1):
        g = smooth_grad(h, t, dx)
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        mh = m / (1 - b1 ** it)
        vh = v / (1 - b2 ** it)
        h = project_constrained(h - lr * mh / (np.sqrt(vh) + 1e-8), target_sum)
        if it % 200 == 0:
            c5 = compute_c5(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_local = h.copy()
    return best_local, best_c5

def subgrad_step(h, dx, target_sum, alpha, K=3):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    top_k = np.argsort(corr)[-K:]
    shifts = np.arange(len(corr)) - (n - 1)
    grad = np.zeros(n)
    for idx in top_k:
        s = shifts[idx]
        if s >= 0:
            if s < n:
                grad[s:] += (1.0 - h[:n - s])
                grad[:n - s] -= h[s:]
        else:
            ss = -s
            if ss < n:
                grad[:n - ss] += (1.0 - h[ss:])
                grad[ss:] -= h[:n - ss]
    return project_constrained(h - alpha * dx * grad / K, target_sum)

def binary_local_search(h, dx, target_sum, n_flips=2000):
    n = len(h)
    h_bin = (h >= 0.5).astype(float)
    n_ones = int(round(target_sum))
    # Fix count
    if h_bin.sum() > n_ones:
        idx_ones = np.where(h_bin == 1)[0]
        to_remove = idx_ones[:int(h_bin.sum()) - n_ones]
        h_bin[to_remove] = 0
    elif h_bin.sum() < n_ones:
        idx_zeros = np.where(h_bin == 0)[0]
        to_add = idx_zeros[:n_ones - int(h_bin.sum())]
        h_bin[to_add] = 1
    best_c5 = compute_c5(h_bin, dx)
    best_h = h_bin.copy()
    for _ in range(n_flips):
        # Pick a 1 to remove and a 0 to add
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i_rm = np.random.choice(ones)
        i_ad = np.random.choice(zeros)
        h_bin[i_rm] = 0
        h_bin[i_ad] = 1
        c5 = compute_c5(h_bin, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_bin.copy()
        else:
            h_bin[i_rm] = 1
            h_bin[i_ad] = 0
    return best_h, best_c5

def slsqp_refine(h, dx, target_sum, t=200, maxiter=500):
    n = len(h)
    try:
        res = minimize(
            lambda x: smooth_max_obj(x, t, dx),
            h.copy(), method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            options={'maxiter': maxiter, 'ftol': 1e-10}
        )
        return project_constrained(res.x, target_sum)
    except:
        return h

def make_random_structured(n, seed_val):
    rng = np.random.RandomState(seed_val)
    choice = rng.randint(5)
    x = np.linspace(0, 2, n, endpoint=False)
    if choice == 0:
        h = (rng.rand(n) < 0.5).astype(float)
    elif choice == 1:
        freq = rng.choice([2, 3, 4, 5])
        phase = rng.rand() * 2 * np.pi
        h = (np.sin(2 * np.pi * freq * x / 2.0 + phase) > 0).astype(float)
    elif choice == 2:
        h = np.zeros(n)
        quarters = n // 4
        h[:quarters] = 1.0
        h[2*quarters:3*quarters] = 1.0
    elif choice == 3:
        k = rng.randint(2, 6)
        h = (x % (2.0/k) < 1.0/k).astype(float)
    else:
        centers = rng.uniform(0, 2, rng.randint(2, 5))
        sigma = 0.15
        vals = sum(np.exp(-((x - c) / sigma)**2) for c in centers)
        h = (vals > np.median(vals)).astype(float)
    return project_constrained(h, n / 2.0)
```
