    # Need sum_i clip(v_i - theta, 0, 1) = k

    lo, hi = v.min() - 1, v.max() + 1

        s = x.sum()

            lo = mid
        else:

    x = np.clip(v - (lo+hi)/2, 0, 1)

def compute_c5(h, dx):

def smooth_max_and_grad(h, t):
    corr = np.correlate(h, 1.0 - h, mode="full")
    n = len(h)
    max_c = np.max(corr)
    shifted = t * (corr - max_c)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    value = max_c + np.log(sum_exp) / t
    weights = exp_vals / sum_exp
    grad = np.zeros(n)
    for j in range(len(corr)):
        w = weights[j]
        if w < 1e-12:
            continue
        lag = j - (n - 1)
        if lag >= 0:
            idx = np.arange(lag, n)
            grad[idx] += w * (1.0 - h[idx - lag])
            if n - 1 - lag >= 0:
                idx2 = np.arange(0, n - lag)
                grad[idx2] -= w * h[idx2 + lag]
        else:

            if n - 1 - llag >= 0:

                grad[idx] += w * (1.0 - h[idx + llag])
            idx2 = np.arange(llag, n)
            grad[idx2] -= w * h[idx2 - llag]
    return value, grad / t

def fw_oracle(g, n, target_sum):
    """Solve min g^T h s.t. 0<=h<=1, sum(h)=target_sum."""
    k = int(round(target_sum))
    sorted_idx = np.argsort(g)
    h = np.zeros(n)
    h[sorted_idx[:k]] = 1.0
    return h

def project_hypersimplex(v, k):
    """Project v onto {x: 0<=x<=1, sum(x)=k} using sorting."""
    n = len(v)
    sv = np.sort(v)[::-1]
    # Find threshold
    cumsum = np.cumsum(sv)
    # k* = max{j: sv[j] - (cumsum[j]-k)/(j+1) > 0}  (0-indexed: j from 0 to n-1)
    # Actually: k* = max{j: sv[j] > (cumsum[j] - k) / (j+1)} where j is 1-indexed count
    rho = np.arange(1, n + 1)
    cond = sv - (cumsum - k) / rho > 0
    if not np.any(cond):
        return np.full(n, k / n)
    k_star = int(np.max(np.where(cond)))  # 1-indexed
    theta = (cumsum[k_star - 1] - k) / k_star
    return np.clip(v - theta, 0.0, 1.0)

        return smooth_max_and_grad(np.clip(x, 0, 1), t)[1]

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_and_grad(h, t):
    corr = np.correlate(h, 1.0 - h, mode="full")
    n = len(h)
    max_c = np.max(corr)
    shifted = t * (corr - max_c)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    value = max_c + np.log(sum_exp) / t
    weights = exp_vals / sum_exp
    grad = np.zeros(n)
    for j in range(len(corr)):
        w = weights[j]
        if w < 1e-14:
            continue
        lag = j - (n - 1)
        if lag >= 0:
            m_start = lag
            m_end = n - 1
            grad[m_start:m_end+1] += w * (1.0 - h[m_start:m_end+1 - lag])
            m2_end = n - 1 - lag
            if m2_end >= 0:
                grad[0:m2_end+1] -= w * h[lag:lag+m2_end+1]
        else:
            llag = -lag
            m2_end = n - 1 - llag
            if m2_end >= 0:
                grad[0:m2_end+1] += w * (1.0 - h[llag:llag+m2_end+1])
            grad[llag:n] -= w * h[0:n-llag]
    return value, grad

def grad_single_corr(h, k_idx):
    """Gradient of corr[k_idx] w.r.t. h."""
    n = len(h)
    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        grad[lag:n] += (1.0 - h[0:n-lag])
        grad[0:n-lag] -= h[lag:n]
    else:
        llag = -lag
        grad[0:n-llag] += (1.0 - h[llag:n])
        grad[llag:n] -= h[0:n-llag]
    return grad

def project_hypersimplex(v, k):
    n = len(v)
    sv = np.sort(v)[::-1]
    cumsum = np.cumsum(sv)
    rho = np.arange(1, n + 1, dtype=float)
    cond = sv - (cumsum - k) / rho > 0
    if not np.any(cond):
        return np.full(n, k / n)
    k_star = int(np.max(np.where(cond)))
    theta = (cumsum[k_star - 1] - k) / k_star
    return np.clip(v - theta, 0.0, 1.0)

def frank_wolfe(h, t, n_iters, target_sum, t0_ref):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h, 2.0 / n)
    for it in range(n_iters):
        val, grad = smooth_max_and_grad(h, t)
        sorted_idx = np.argsort(grad)
        h_fw = np.zeros(n)
        h_fw[sorted_idx[:int(target_sum)]] = 1.0
        gamma = 2.0 / (it + 2)
        h = (1.0 - gamma) * h + gamma * h_fw
        h = project_hypersimplex(h, target_sum)
        c5_now = compute_c5(h, 2.0 / n)
        if c5_now < best_c5:
            best_c5 = c5_now
            best_h = h.copy()
        if time.time() - t0_ref > 200:
            break
    return best_h, best_c5

def adam_optimize(h, t, n_iters, target_sum, t0_ref, lr=0.1, check_interval=500):
    n = len(h)
    dx = 2.0 / n
    m = np.zeros(n)
    u = np.zeros(n)
    beta1, beta2 = 0.85, 0.999
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    for it in range(n_iters):
        val, grad = smooth_max_and_grad(h, t)
        m = beta1 * m + (1.0 - beta1) * grad
        u = beta2 * u + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        u_hat = u / (1.0 - beta2 ** (it + 1))
        h_new = h - lr * m_hat / (np.sqrt(u_hat) + 1e-8)
        h_new = project_hypersimplex(h_new, target_sum)
        h = h_new
        if (it + 1) % check_interval == 0:
            c5_now = compute_c5(h, dx)
            if c5_now < best_c5:
                best_c5 = c5_now
                best_h = h.copy()
        if time.time() - t0_ref > 600:
            break
    c5_final = compute_c5(h, dx)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    return best_h, best_c5

def slsqp_refine(h, target_sum, t, maxiter=500, t0_ref=None, max_time=60):
    n = len(h)
    def obj_func(x):
        v, _ = smooth_max_and_grad(x, t)
        return v
    def grad_func(x):
        _, g = smooth_max_and_grad(x, t)
        return g
    def con_func(x):
        return np.sum(x) - target_sum
    def con_grad(x):
        return np.ones(n)
    try:
        if t0_ref is not None and time.time() - t0_ref > max_time:
            return h
        res = minimize(
            obj_func, h.copy(), jac=grad_func,
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': con_func, 'jac': con_grad}],
            options={'maxiter': maxiter, 'ftol': 1e-11}
        )
        h_new = np.clip(res.x, 0.0, 1.0)
        h_new = project_hypersimplex(h_new, target_sum)
        return h_new
    except Exception:
        return h

def downsample_to(h, new_n):
    n = len(h)
    x_old = np.linspace(0, 2, n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return h_new

def make_binary(n, rng):
    h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0

def round_to_binary(h, target_sum):

    k = int(round(target_sum))
    sorted_idx = np.argsort(h)[::-1]
