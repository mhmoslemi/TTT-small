        if 0 <= j - offset < n:
            hb = h[j - offset]

        # Actually: R = x - x*ho + hb - hb*x = x*(1-ho-hb) + hb = x*coeff + hb
        # If coeff < 0: minimize at x=1, R = coeff + hb = 1-ho
        # If coeff > 0: minimize at x=0, R = hb
        # But we need to maintain sum constraint

        if coeff > 0:

            h[j] = max(0.0, h[j] - min(h[j], 0.1))
        elif coeff < 0:
            # Want to increase h[j]
            h[j] = min(1.0, h[j] + min(1.0 - h[j], 0.1))

def push_pull_step(h, target_lag, target_sum, step_size=0.05):
    n = len(h)
    offset = target_lag - (n - 1)
    
    # Compute dR/dh[j] for this specific lag
    dR = np.zeros(n)
    for j in range(n):
        term1 = 0.0
        if 0 <= j + offset < n:
            term1 = 1.0 - h[j + offset]
        term2 = 0.0
        if 0 <= j - offset < n:
            term2 = h[j - offset]
        dR[j] = term1 - term2
    
    # Find best pair to push-pull
    # Want to decrease R: increase h where dR < 0, decrease h where dR > 0
    neg_idx = np.where(dR < -1e-6)[0]
    pos_idx = np.where(dR > 1e-6)[0]
    
    if len(neg_idx) == 0 or len(pos_idx) == 0:
        return h
    
    # Greedy: find the pair (j in neg, j' in pos) that gives most improvement
    # Improvement from moving δ from j' to j: δ * (dR[j'] - dR[j])
    # Best: most negative dR[j] and most positive dR[j']
    j_best = neg_idx[np.argmin(dR[neg_idx])]
    jpos_best = pos_idx[np.argmax(dR[pos_idx])]
    
    # How much can we move?
    max_move = min(h[jpos_best], 1.0 - h[j_best], step_size)
    
    h[jpos_best] -= max_move
    h[j_best] += max_move
    
    return h

        if elapsed() > budget_s * 0.25:
            break
        h_new = slsqp_opt(best_h, dx, tsum, t_val, 600)
        h_new = proj_box_sum(h_new, tsum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 2: Push-pull on top lags with decreasing step
    if elapsed() < budget_s * 0.50:
        step = 0.10
        no_improve = 0
        for it in range(20000):
            if elapsed() > budget_s * 0.50:
                break
            corr = np.correlate(best_h, 1 - best_h, mode="full")
            top_lags = np.argsort(corr)[-3:]
            improved = False
            for lag in top_lags:
                offset = int(lag) - (n - 1)
                h_trial = best_h.copy()
                h_trial = push_pull_step(h_trial, offset, step)
                c5_trial = compute_c5(h_trial)
                if c5_trial < best_c5 - 1e-12:
                    best_h = h_trial
                    best_c5 = c5_trial
                    improved = True
                    break
            if not improved:
                step *= 0.85
                no_improve += 1
                if no_improve > 20:
                    break
            else:
                no_improve = 0

    # Phase 3: SLSQP refinement with high t
    if elapsed() < budget_s * 0.65:
        for t_val in [500, 1500, 3000, 5000]:
            if elapsed() > budget_s * 0.65:
                break
            h_new = slsqp_opt(best_h, dx, tsum, t_val, 1000)
            h_new = proj_box_sum(h_new, tsum)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    # Phase 4: Random restarts - perturb and reoptimize
    if elapsed() < budget_s * 0.78:
        n_restarts = 30
        for r in range(n_restarts):
            if elapsed() > budget_s * 0.78:
                break
            h_trial = best_h.copy()
            n_pert = np.random.randint(5, 15)
            idx = np.random.choice(n, n_pert, replace=False)
            scale = np.random.uniform(0.1, 0.4)
            h_trial[idx] += np.random.uniform(-scale, scale, n_pert)
            h_trial = proj_box_sum(h_trial, tsum)
            for t_val in [30, 150]:
                if elapsed() > budget_s * 0.78:
                    break
                h_trial = slsqp_opt(h_trial, dx, tsum, t_val, 300)
                h_trial = proj_box_sum(h_trial, tsum)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5:
                best_h = h_trial
                best_c5 = c5_trial
                # Refine the restart
                for t_val in [500, 2000]:
                    if elapsed() > budget_s * 0.78:
                        break
                    h_trial = slsqp_opt(h_trial, dx, tsum, t_val, 500)
                    h_trial = proj_box_sum(h_trial, tsum)
                    c5_trial = compute_c5(h_trial)
                    if c5_trial < best_c5:
                        best_h = h_trial
                        best_c5 = c5_trial

    # Phase 5: Try n=320
    if elapsed() < budget_s * 0.90:
        n2 = 320
        dx2 = 2.0 / n2
        tsum2 = n2 / 2.0
        h2 = upsample(best_h, n2)
        h2 = proj_box_sum(h2, tsum2)
        c5_2 = compute_c5(h2)
        for t_val in [20, 60, 200, 600]:
            if elapsed() > budget_s * 0.90:
                break
            h2 = slsqp_opt(h2, dx2, tsum2, t_val, 400)
            h2 = proj_box_sum(h2, tsum2)
            c5_2 = compute_c5(h2)
            if c5_2 < best_c5:
                best_h = h2.copy()
                best_c5 = c5_2
                best_n = n2

    # Phase 6: Final polish at best resolution
    if elapsed() < budget_s - 30:
        n_f = best_n
        dx_f = 2.0 / n_f
        tsum_f = n_f / 2.0
        for t_val in [2000, 5000, 10000]:
            if elapsed() > budget_s - 25:
                break
            h_new = slsqp_opt(best_h, dx_f, tsum_f, t_val, 2000)
            h_new = proj_box_sum(h_new, tsum_f)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    n_final = len(best_h)
    dx_final = 2.0 / n_final
    best_h = proj_box_sum(best_h, n_final / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n_final
