We want to minimize max_d overlap(d).

        for d in [0.02, -0.02, 0.05, -0.05]:
            ht = best.copy()
            ht[i] = np.clip(ht[i] + d, 0.0, 1.0)
            ht[j] = np.clip(ht[j] - d, 0.0, 1.0)
            c = compute_c5(ht, dx)
            if c < best_c5 - 1e-11:
                best = ht
                best_c5 = c
                break
    return best, best_c5

def anneal(h, dx, ts, n_iters, t_start, t_end):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(h, dx)
    cur = h.copy()
    cur_c5 = best_c5
    t0_an = time.time()
    for it in range(n_iters):
        if time.time() - t0_an > 80:
            break
        tc = t_start * (t_end / t_start) ** (it / max(n_iters, 1))
        np_ = np.random.randint(2, 7)
        idx = np.random.choice(n, np_, replace=False)
        ht = cur.copy()
        ht[idx] = np.clip(ht[idx] + np.random.randn(np_) * 0.08, 0.0, 1.0)
        ht = fix_sum(ht, ts)
        c = compute_c5(ht, dx)
        delta = c - cur_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / max(tc, 1e-8)):
            cur = ht
            cur_c5 = c
            if c < best_c5:
                best = ht.copy()
                best_c5 = c
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0_time = time.time()

    def remaining():
        return budget_s - (time.time() - t0_time)

    h0 = np.array(initial_h_values, dtype=float).copy()
    n0 = len(h0)
    dx0 = 2.0 / n0
    ts0 = n0 / 2.0
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = fix_sum(h0, ts0)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx0)
    best_n = n0
    best_dx = dx0

    # Phase 1: L-BFGS-B fast optimization
    for t in [20, 50, 150, 400]:
        if remaining() < 180:
            break
        ht = lbfgs_opt(best_h, best_dx, best_n / 2.0, t, 600)
        c = compute_c5(ht, best_dx)
        if c < best_c5:
            best_h, best_c5 = ht, c

    # Phase 2: SLSQP precision refinement
    for t in [100, 300, 800]:
        if remaining() < 140:
            break
        ht = slsqp_opt(best_h, best_dx, best_n / 2.0, t, 800)
        c = compute_c5(ht, best_dx)
        if c < best_c5:
            best_h, best_c5 = ht, c

    # Phase 3: 2-opt local search
    if remaining() > 80:
        best_h, best_c5 = local_search_2opt(best_h, best_dx, best_n / 2.0, 3000)

    # Phase 4: SLSQP re-optimization after local search
    if remaining() > 120:
        ht = slsqp_opt(best_h, best_dx, best_n / 2.0, 500, 1000)
        c = compute_c5(ht, best_dx)
        if c < best_c5:
            best_h, best_c5 = ht, c

    # Phase 5: Simulated annealing
    if remaining() > 120:
        best_h, best_c5 = anneal(best_h, best_dx, best_n / 2.0, 3000, 0.12, 0.003)
        if remaining() > 50:
            ht = lbfgs_opt(best_h, best_dx, best_n / 2.0, 200, 500)
            c = compute_c5(ht, best_dx)
            if c < best_c5:
                best_h, best_c5 = ht, c
            ht = slsqp_opt(best_h, best_dx, best_n / 2.0, 500, 800)
            c = compute_c5(ht, best_dx)
            if c < best_c5:
                best_h, best_c5 = ht, c

    # Phase 6: Try different resolutions
    for n_try in [n0 - 40, n0 - 20, n0 + 20, n0 + 40, n0 + 80]:
        if n_try < 40:
            continue
        if remaining() < 80:
            break
        dx_t = 2.0 / n_try
        ts_t = n_try / 2.0
        ht = upsample(best_h, n_try)
        ht = fix_sum(ht, ts_t)
        ht = lbfgs_opt(ht, dx_t, ts_t, 80, 400)
        ht = slsqp_opt(ht, dx_t, ts_t, 200, 500)
        c = compute_c5(ht, dx_t)
        if c < best_c5:
            best_h, best_c5, best_n, best_dx = ht, c, n_try, dx_t

    # Phase 7: Structured random restarts
    if remaining() > 60:
        for s in range(12):
            if remaining() < 30:
                break
            for n_r in [best_n, best_n + 40]:
                if remaining() < 20:
                    break
                dx_r = 2.0 / n_r
                ts_r = n_r / 2.0
                hr = make_fourier_binary(n_r, seed + s * 173 + n_r)
                hr = lbfgs_opt(hr, dx_r, ts_r, 40, 300)
                hr = slsqp_opt(hr, dx_r, ts_r, 150, 400)
                hr = lbfgs_opt(hr, dx_r, ts_r, 300, 300)
                c = compute_c5(hr, dx_r)
                if c < best_c5:
                    best_h, best_c5, best_n, best_dx = hr, c, n_r, dx_r

    # Phase 8: Final high-precision SLSQP
    if remaining() > 20:
        ts_f = best_n / 2.0
        ht = slsqp_opt(best_h, best_dx, ts_f, 1500, 3000)
        c = compute_c5(ht, best_dx)
        if c < best_c5:
            best_h, best_c5 = ht, c

    # Phase 9: One more L-BFGS pass
    if remaining() > 10:
        ht = lbfgs_opt(best_h, best_dx, best_n / 2.0, 1000, 1000)
        c = compute_c5(ht, best_dx)
        if c < best_c5:
            best_h, best_c5 = ht, c

    best_h = np.clip(best_h, 0.0, 1.0)
    best_dx = 2.0 / best_n
    best_c5 = compute_c5(best_h, best_dx)
    return best_h, best_c5, best_n
