Actually, for the subgradient approach, I can compute:
- corr = np.correlate(h, 1-h, mode='full') * dx
- Find k_max = argmax(corr)
- Compute the gradient of corr[k_max] with respect to h

The k_max-th element of the full correlation is:
corr_full[k_max] = sum_{i} h[i] * (1 - h[i - (k_max - (n-1))])
where the product is defined only when both indices are in [0, n-1].

Let s = k_max - (n-1). Then:
corr_full[k_max] = sum_{i: 0≤i≤n-1, 0≤i+s≤n-1} h[i] * (1 - h[i+s])

The gradient with respect to h[j]:
d/dh[j] = sum_{i} [delta(i,j)*(1-h[i+s]) + h[i]*(-delta(i+s,j))]
= (1 - h[j+s]) * [j+s in [0,n-1]] - h[j-s] * [j-s in [0,n-1]]

So the gradient is:
grad[j] = (1 - h[j+s]) if 0 ≤ j+s ≤ n-1, else 0

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def corr_values(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_h(h, target):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
    else:
        mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def max_corr_grad(h):
    """Gradient of the max correlation term."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    k_max = int(np.argmax(corr))
    s = k_max - (n - 1)
    grad = np.zeros(n)
    # For j in [0, n-1]:
    # grad[j] = (1 - h[j+s]) if 0 <= j+s <= n-1 else 0
    #           - h[j-s] if 0 <= j-s <= n-1 else 0
    # (times dx)
    if s >= 0:
        # j+s in [0, n-1] means j in [0, n-1-s]
        grad[:n-s] += (1.0 - h[s:n]) 
        # j-s in [0, n-1] means j in [s, n-1]
        grad[s:] -= h[:n-s]
    else:
        s2 = -s
        # j+s in [0, n-1] means j in [s2, n-1]
        grad[s2:] += (1.0 - h[:n-s2])
        # j-s in [0, n-1] means j in [0, n-1-s2]
        grad[:n-s2] -= h[s2:]
    grad *= dx
    return grad

def smooth_corr_grad(h, temp):
    """Gradient of the log-sum-exp smooth approximation."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    shifted = corr - np.max(corr)
    weights = np.exp(shifted / temp)
    weights /= weights.sum()
    # Weighted average of gradients
    grad = np.zeros(n)
    top_k = min(20, len(corr))
    top_idx = np.argsort(corr)[-top_k:]
    for k in top_idx:
        s = k - (n - 1)
        if s >= 0:
            g = np.zeros(n)
            g[:n-s] = (1.0 - h[s:n])
            g[s:] -= h[:n-s]
        else:
            s2 = -s
            g = np.zeros(n)
            g[s2:] = (1.0 - h[:n-s2])
            g[:n-s2] -= h[s2:]
        grad += weights[k] * g
    grad *= dx
    return grad

def subgradient_descent(h0, n_iter=2000, alpha0=0.1, decay=0.999, time_limit=30):
    """Projected subgradient method on the max correlation."""
    n = len(h0)
    h = h0.copy()
    c5 = compute_c5(h)
    alpha = alpha0
    t_start = time.time()
    for i in range(n_iter):
        if time.time() - t_start > time_limit:
            break
        grad = max_corr_grad(h)
        h_new = h - alpha * grad
        h_new = project_h(h_new, n * 0.5)
        c5_new = compute_c5(h_new)
        if c5_new < c5:
            h = h_new
            c5 = c5_new
        else:
            alpha *= decay
        alpha = max(alpha, 1e-6)
    return h, c5

def optimize_t_var(h0, max_iter=500, ftol=1e-11):
    n = len(h0)
    c5_cur = compute_c5(h0)
    v0 = np.append(h0.copy(), c5_cur + 0.02)
    
    def t_obj(v):
        return v[-1]
    
    def t_ineq(v):
        h = v[:n]
        t = v[-1]
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        return t - corr
    
    def t_eq(v):
        return np.sum(v[:n]) - n * 0.5
    
    def t_eq_jac(v):
        j = np.zeros(len(v))
        j[:n] = 1.0
        return j
    
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.5)]
    cons = [
        {'type': 'ineq', 'fun': t_ineq},
        {'type': 'eq', 'fun': t_eq, 'jac': t_eq_jac}

    res = minimize(t_obj, v0, method='SLSQP', bounds=bounds,

    h_new = np.clip(res.x[:n], 0.0, 1.0)
    h_new = project_h(h_new, n * 0.5)

def optimize_smooth(h0, temp, max_iter=300, ftol=1e-12):
    n = len(h0)
    dx = 2.0 / n
    
    def smooth_obj(h):
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        m = np.max(corr)
        return m + temp * np.log(np.sum(np.exp((corr - m) / temp)))
    
    def sum_eq(h):
        return np.sum(h) - n * 0.5
    
    def sum_jac(h):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_jac}]
    res = minimize(smooth_obj, h0.copy(), method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': ftol})
    h_new = np.clip(res.x, 0.0, 1.0)
    h_new = project_h(h_new, n * 0.5)
    return h_new, compute_c5(h_new)

def max_corr_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    k_max = int(np.argmax(corr))
    s = (n - 1) - k_max

    if s >= 0:

        grad[:n-s] += (1.0 - h[s:n])
        # j-s in [0, n-1] → j in [s, n-1]

        # j-s = j+s2 in [0, n-1] → j in [0, n-1-s2]

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))


def project_h(h, target):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
    else:
        mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)


def max_corr_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    k_max = int(np.argmax(corr))
    s = (n - 1) - k_max
    grad = np.zeros(n)
    if s >= 0:
        grad[:n - s] += (1.0 - h[s:n])
        grad[s:] -= h[:n - s]
    else:
        s2 = -s
        grad[s2:] += (1.0 - h[:n - s2])
        grad[:n - s2] -= h[s2:]
    grad *= dx
    return grad


def smooth_corr_grad(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    shifted = corr - np.max(corr)
    weights = np.exp(shifted / temp)
    weights /= weights.sum()
    top_k = min(15, len(corr))
    top_idx = np.argsort(corr)[-top_k:]
    grad = np.zeros(n)
    for k in top_idx:
        s = (n - 1) - k
        if s >= 0:
            g = np.zeros(n)
            g[:n - s] = (1.0 - h[s:n])
            g[s:] -= h[:n - s]
        else:
            s2 = -s
            g = np.zeros(n)
            g[s2:] = (1.0 - h[:n - s2])
            g[:n - s2] -= h[s2:]
        grad += weights[k] * g
    grad *= dx
    return grad

    n = len(h)
    if use_smooth:
        grad = smooth_corr_grad(h, temp)
    else:
        grad = max_corr_grad(h)
    h_new = h - alpha * grad
    return project_h(h_new, n * 0.5)


def subgradient_opt(h0, n_iter, alpha0, decay, temp, time_limit, use_smooth=True):
    n = len(h0)
    h = h0.copy()
    c5 = compute_c5(h)
    alpha = alpha0
    t_start = time.time()
    best_h = h.copy()
    best_c5 = c5
    for _ in range(n_iter):
        if time.time() - t_start > time_limit:
            break
        h_new = subgradient_step(h, alpha, temp, use_smooth)
        c5_new = compute_c5(h_new)
        if c5_new < c5:
            h = h_new
            c5 = c5_new
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            alpha *= decay
        alpha = max(alpha, 1e-7)
    return best_h, best_c5


def t_obj(v):
    return v[-1]


def make_t_ineq(n):
    def t_ineq(v):
        h = v[:n]
        t = v[-1]
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        return t - corr
    return t_ineq


def t_eq_fun(n):
    def f(v):
        return np.sum(v[:n]) - n * 0.5
    return f


def t_eq_jac(n):
    def f(v):
        j = np.zeros(len(v))
        j[:n] = 1.0
        return j
    return f


def optimize_t_var(h0, max_iter=500, ftol=1e-11):

    v0 = np.append(h0.copy(), c5_cur + 0.02)

    res = minimize(t_obj, v0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': ftol})
    h_new = np.clip(res.x[:n], 0.0, 1.0)
    h_new = project_h(h_new, n * 0.5)
    return h_new, compute_c5(h_new)

        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        m = np.max(corr)
        return m + temp * np.log(np.sum(np.exp((corr - m) / temp)))
    return f
