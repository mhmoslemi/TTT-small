```python
def project_box_sum(h, target):
    """Project h onto {0 <= x <= 1, sum(x) = target}"""
    n = len(h)
    lo, hi = 0.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo+hi)/2, 0.0, 1.0)
    # Final correction
    diff = target - proj.sum()
    # Distribute difference to non-saturated variables
    mask = (proj > 0) & (proj < 1)
    if mask.sum() > 0 and abs(diff) > 1e-14:
        proj[mask] += diff / mask.sum()
        proj = np.clip(proj, 0.0, 1.0)
    return proj
```

```python
    n = len(h)
```

```python
def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
```

```python
    return float(np.max(corr) * dx)

def smooth_max_val(corr, tau):
    x = tau * corr
    x_max = np.max(x)
    return float(x_max + np.log(np.sum(np.exp(x - x_max))) / tau)

def smooth_max_weights(corr, tau):
    x = tau * corr
    x_max = np.max(x)
    e = np.exp(x - x_max)
    s = e.sum()
    return e / s

def project_box_sum(h, target):
    n = len(h)
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    nu = (lo + hi) / 2.0
    proj = np.clip(h - nu, 0.0, 1.0)
    diff2 = target - proj.sum()
    if abs(diff2) > 1e-12:
        mask = (proj > 1e-12) & (proj < 1.0 - 1e-12)
        if mask.sum() > 0:
            proj[mask] += diff2 / mask.sum()
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def compute_grad(h, w):
    n = len(h)

        wl = w[l + n - 1]
        if wl < 1e-14:
            continue
        if l >= 0:
            grad[:n - l] += wl * (1.0 - h[l:])
            grad[l:] -= wl * h[:n - l]
        else:
            m = -l
            grad[m:] += wl * (1.0 - h[:n - m])
            grad[:n - m] -= wl * h[m:]
    return grad
```
