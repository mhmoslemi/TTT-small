        h_init = np.array(initial_h_values, dtype=float).copy()
    except NameError:
        h_init = np.full(70, 0.5)
    n_init = len(h_init)

    for n_points in [70, 50, 35, 100, 56]:
        if tleft() < 100:
            break

        dx = 2.0 / n_points
        target_sum = n_points / 2.0
        _g_dx = dx
        _g_target = target_sum
        _g_n = n_points

        if n_points == n_init:
            h0 = h_init.copy()
        else:
            x_old = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n_points)
            h0 = np.interp(x_new, x_old, h_init)
        h0 = project_feasible(h0, target_sum)

        # Phase A: Log-sum-exp SLSQP with analytical gradient, annealing
        alphas = [3.0, 8.0, 20.0, 50.0, 100.0, 200.0]
        for alpha in alphas:
            if tleft() < 50:
                break
            h0 = slsqp_lse(h0, alpha, 3000, n_points)
            c5 = compute_c5(h0, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h0.copy()
                best_n = n_points

        # Phase B: Active-set gradient descent with momentum
        if tleft() < 30:
            break
        h = h0.copy()
        step_size = 0.002
        momentum = np.zeros(n_points)
        mu = 0.6
        stag = 0
        for _ in range(4000):
            if tleft() < 30:
                break
            grad = active_set_grad(h, n_points)
            momentum = mu * momentum - step_size * grad
            h_new = project_feasible(h + momentum, target_sum)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                best_n = n_points
                stag = 0
            else:
                stag += 1
                if stag % 40 == 0:
                    step_size *= 0.87
                if step_size < 1e-9:
                    h = best_h + rng.normal(0, 0.02, n_points)
                    h = project_feasible(h, target_sum)
                    step_size = 0.0015
                    momentum = np.zeros(n_points)
                    stag = 0
            h = h_new

        # Phase C: Random restarts with SLSQP
        n_restarts = 8
        for r in range(n_restarts):
            if tleft() < 20:
                break
            if best_n == n_points:
                h_pert = best_h + rng.normal(0, 0.04, n_points)
            else:
                h_pert = h0 + rng.normal(0, 0.06, n_points)
            h_pert = project_feasible(h_pert, target_sum)
            h_pert = slsqp_lse(h_pert, 40.0, 2000, n_points)
            c5 = compute_c5(h_pert, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_pert.copy()
                best_n = n_points
                h0 = h_pert.copy()

    # Phase D: Fine polishing with active-set gradient
    if best_h is not None and tleft() > 15:
        n = best_n
        dx = 2.0 / n
        target_sum = n / 2.0
        h = best_h.copy()
        step_size = 0.0004
        momentum = np.zeros(n)
        for _ in range(6000):
            if tleft() < 3:
                break
            grad = active_set_grad(h, n)
            momentum = 0.4 * momentum - step_size * grad
            h_new = project_feasible(h + momentum, target_sum)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            else:
                step_size *= 0.95
                if step_size < 1e-12:
                    break
            h = h_new

    # Phase E: One final SLSQP polish on best
    if best_h is not None and tleft() > 20:
        n = best_n
        _g_dx = 2.0 / n
        _g_target = n / 2.0
        h_final = slsqp_lse(best_h, 300.0, 2000, n)
        c5_final = compute_c5(h_final, 2.0 / n)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final
            best_n = n

    return (best_h, best_c5, best_n)
