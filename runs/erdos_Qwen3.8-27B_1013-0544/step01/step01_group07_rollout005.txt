So it's a weighted average of the gradients of the individual overlap functions, weighted by their relative magnitudes.

This is nice because:
- It's smooth (C^1)
- The gradient is a convex combination of individual gradients

def sticky_gradient_descent(h, dx, target_sum, n_iters=5000, t=50.0, lr=0.01):
    n = len(h)
    v = np.zeros_like(h)
    h_curr = h.copy()
    best_c5 = compute_c5(h_curr, dx)
    best_h = h_curr.copy()
    
    for it in range(n_iters):
        # Compute gradient of smooth max
        corr = np.correlate(h_curr, 1 - h_curr, mode="full")
        # Softmax weights
        t_corr = t * corr * dx
        t_corr -= np.max(t_corr)
        weights = np.exp(t_corr)
        weights /= np.sum(weights)
        
        # Gradient = weighted sum of individual gradients
        grad = np.zeros_like(h_curr)
        # Only compute for lags with significant weight
        threshold = 1e-6
        active = weights > threshold
        active_indices = np.where(active)[0]
        
        for k_idx in active_indices:
            k = k_idx - (n - 1)
            w = weights[k_idx]
            for j in range(n):
                g = 0.0
                if 0 <= j + k < n:
                    g += h_curr[j + k]
                if 0 <= j - k < n:
                    g += h_curr[j - k]
                grad[j] += w * (1.0 - g)
        
        # Nesterov step
        v = 0.9 * v - lr * grad
        h_new = h_curr + v
        h_new = project_onto_constraint(h_new, target_sum)
        h_curr = h_new
        
        if it % 100 == 0:
            c5 = compute_c5(h_curr, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_curr.copy()
    
    return best_h, best_c5

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project(h, target_sum):
    lo = -np.max(h) - 1.0
    hi = 1.0 - np.min(h) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def smooth_max_obj(h, t, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    shifted = t * (corr - m)
    return (m + (1.0 / t) * np.log(np.sum(np.exp(shifted)))) * dx

def smooth_max_grad(h, t, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    m = np.max(corr)
    shifted = t * (corr - m)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    n = len(h)
    grad = np.zeros(n)
    thresh = 1e-7
    active = weights > thresh
    active_idx = np.where(active)[0]
    for k_idx in active_idx:
        k = k_idx - (n - 1)
        w = weights[k_idx]
        if k >= 0:
            grad[k:] += w * (1.0 - h[:n - k])
            grad[:n - k] -= w * h[k:]
        else:
            ma = -k
            grad[:n - ma] += w * (1.0 - h[ma:])
            grad[ma:] -= w * h[:n - ma]
    return grad * dx

def proj_grad_descent(h, dx, target_sum, n_iters=2000, t_start=10.0, t_end=300.0,
                      lr_init=0.05, beta=0.9, time_limit=30.0):
    n = len(h)
    t0 = time.time()
    v = np.zeros(n)
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c5 = compute_c5(h_curr, dx)
    lr = lr_init
    for it in range(n_iters):
        if time.time() - t0 > time_limit:
            break
        frac = it / max(n_iters - 1, 1)
        t = t_start * (t_end / t_start) ** frac
        grad = smooth_max_grad(h_curr, t, dx)
        v = beta * v - lr * grad
        h_new = h_curr + v
        h_new = project(h_new, target_sum)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        if c5_new > best_c5 + 0.01:
            lr *= 0.9
        elif it > 50 and it % 50 == 0:
            lr = min(lr * 1.05, 0.2)
        h_curr = h_new
    return best_h, best_c5

def slsqp_optimize(h, dx, target_sum, t=50, maxiter=500):
    n = len(h)
    try:
        res = minimize(
            lambda x: smooth_max_obj(x, t, dx),
            h.copy(),
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            options={'maxiter': maxiter, 'ftol': 1e-12}
        )
        h_new = project(res.x, target_sum)
        return h_new
    except Exception:
        return h

        nc = n0 // 2
        dxc = 2.0 / nc
        tsc = nc / 2.0
        hc = upsample(best_h, nc)
        hc = project(hc, tsc)
        hc = slsqp_optimize(hc, dxc, tsc, t=50, maxiter=500)
        hc, c5c = proj_grad_descent(hc, dxc, tsc,
                                    n_iters=2000, t_start=15, t_end=200,
                                    lr_init=0.03, beta=0.9,
                                    time_limit=min(40, remaining() * 0.5))
        if c5c < best_c5:
            best_c5, best_h, best_n = c5c, hc.copy(), nc

    # Phase 8: Final polish
    if remaining() > 30:
        df = 2.0 / best_n
        tsf = best_n / 2.0
        h_final = slsqp_optimize(best_h, df, tsf, t=500, maxiter=1500)
        c5_final = compute_c5(h_final, df)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()
        # One more gradient pass
        if remaining() > 15:
            h_final2, c5_f2 = proj_grad_descent(best_h, df, tsf,
                                                n_iters=1000, t_start=50, t_end=500,
                                                lr_init=0.01, beta=0.9, time_limit=10)
            if c5_f2 < best_c5:
                best_c5 = c5_f2
                best_h = h_final2.copy()

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project(best_h, best_n / 2.0)
    dfinal = 2.0 / best_n
    best_c5 = compute_c5(best_h, dfinal)

    return best_h, best_c5, best_n
