import numpy as np
import time

def compute_correlate(h):
    return np.correlate(h, 1 - h, mode='full')

def compute_c5(h):
    n = len(h)
    return np.max(compute_correlate(h)) * (2.0 / n)

def grad_corr_single(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n-k])
        if k > 0:
            g[:n-k] -= h[k:]
    else:
        p = -k
        g[:n-p] += (1.0 - h[p:])
        if p > 0:
            g[p:] -= h[:n-p]
    return g

def top_k_grad(h, K=5):
    """Average gradient over top-K correlation points."""
    n = len(h)
    corr = compute_correlate(h)
    # Get top K indices
    top_idx = np.argsort(corr)[-K:]
    grad = np.zeros(n)
    for idx in top_idx:
        k = idx - (n - 1)
        grad += grad_corr_single(h, k)
    grad /= K
    return grad * (2.0 / n)

    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

            g[p:] -= h[:n - p]
    return g

def project_simplex(h, target_sum):
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def interpolate_to_n(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0 / n_new
    return np.interp(x_new, x_old, h_old)

def optimize_adam(h0, target_sum, deadline, rng, K=3, lr=0.008, beta1=0.85, beta2=0.95, perturb_every=200):
    n = len(h0)
    h = project_simplex(h0.astype(float).copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    m = np.zeros(n)
    v = np.zeros(n)
    t_step = 0
    eps = 1e-8
    stall = 0

    while time.time() < deadline:
        t_step += 1
        corr = compute_correlate(h)
        top_idx = np.argpartition(corr, -K)[-K:]
        grad = np.zeros(n)
        for idx in top_idx:
            k = idx - (n - 1)
            grad += grad_at_lag(h, k, n)
        grad = grad / K * (2.0 / n)

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t_step)
        v_hat = v / (1.0 - beta2 ** t_step)

        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_simplex(h_new, target_sum)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5 - 1e-16:
            best_h = h_new.copy()
            best_c5 = c5_new
            h = h_new
            stall = 0
        else:
            h = best_h.copy()
            stall += 1
            if stall >= perturb_every:
                noise_scale = 0.005 + 0.005 * min(stall / perturb_every, 3.0)
                h_pert = best_h + rng.randn(n) * noise_scale
                h_pert = project_simplex(h_pert, target_sum)
                c5_pert = compute_c5(h_pert)
                if c5_pert < best_c5:
                    best_h = h_pert.copy()
                    best_c5 = c5_pert
                h = best_h.copy()
                m *= 0.3
                v *= 0.3
                stall = 0

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.full(200, 0.5)

    n = len(h0)
    target_sum = n / 2.0
    h0 = project_simplex(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Broad exploration K=8, moderate lr
    d1 = t0 + budget_s * 0.20
    h1, c5_1 = optimize_adam(h0, target_sum, d1, rng, K=8, lr=0.015)
    if c5_1 < best_c5:
        best_h, best_c5 = h1, c5_1

    # Phase 2: Focused K=4
    d2 = t0 + budget_s * 0.50
    h2, c5_2 = optimize_adam(best_h, target_sum, d2, rng, K=4, lr=0.008)
    if c5_2 < best_c5:
        best_h, best_c5 = h2, c5_2

    # Phase 3: Very focused K=2
    d3 = t0 + budget_s * 0.75
    h3, c5_3 = optimize_adam(best_h, target_sum, d3, rng, K=2, lr=0.004)
    if c5_3 < best_c5:
        best_h, best_c5 = h3, c5_3

    # Phase 4: Try a different resolution (interpolate to n*1.33)
    remaining = budget_s - (time.time() - t0)
    if remaining > 80:
        n_try = int(n * 4 / 3)
        if n_try != n:
            h_interp = interpolate_to_n(best_h, n_try)
            h_interp = project_simplex(h_interp, n_try / 2.0)
            d4 = time.time() + min(remaining * 0.4, 200)
            h4, c5_4 = optimize_adam(h_interp, n_try / 2.0, d4, rng, K=3, lr=0.008)
            if c5_4 < best_c5:
                best_h, best_c5 = h4, c5_4
                n = n_try
                target_sum = n / 2.0

    # Phase 5: Final fine polish K=1, very small lr
    remaining = budget_s - (time.time() - t0)
    if remaining > 20:
        d5 = t0 + budget_s - 3
        h5, c5_5 = optimize_adam(best_h, target_sum, d5, rng, K=1, lr=0.002, beta1=0.9, beta2=0.97, perturb_every=300)
        if c5_5 < best_c5:
            best_h, best_c5 = h5, c5_5

    best_h = project_simplex(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
