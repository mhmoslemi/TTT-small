import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_sum(h, target):
    # Project h onto {0<=h<=1, sum(h)=target}
    # Binary search on lambda such that sum(clamp(h-lambda, 0, 1)) = target
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def smooth_max_grad(h, t, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    max_c = np.max(corr)
    shifted = t * (corr - max_c)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    p = exp_s / sum_exp
    obj = (max_c + np.log(sum_exp) / t) * dx
    
    # Gradient
    n = len(h)
    grad = np.zeros(n)
    # For each shift s (from -(n-1) to n-1):
    # d(corr[n-1+s])/dh[i] = (1-h[i+s]) if 0<=i+s<n, minus h[i-s] if 0<=i-s<n
    for s in range(-(n-1), n):
        ps = p[n-1+s]
        if ps < 1e-10:
            continue
        # First part: (1-h[i+s]) for valid i
        i_lo = max(0, -s)
        i_hi = min(n, n - s)
        if i_lo < i_hi:
            grad[i_lo:i_hi] += ps * (1 - h[i_lo+s:i_hi+s])
        # Second part: -h[i-s] for valid i
        i_lo2 = max(0, s)
        i_hi2 = min(n, n + s)
        if i_lo2 < i_hi2:
            grad[i_lo2:i_hi2] += -ps * h[i_lo2-s:i_hi2-s]
    grad *= dx
    return obj, grad

def pgd_optimize(h, dx, target_sum, t=50, n_iter=5000, lr_init=0.01):
    """Projected gradient descent with backtracking"""
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_obj = compute_c5(h, dx)
    lr = lr_init
    
    for it in range(n_iter):
        obj, grad = smooth_max_grad(h, t, dx)
        
        # Backtracking line search
        step = lr
        for _ in range(20):
            h_new = project_sum(h - step * grad, target_sum)
            c5_new = compute_c5(h_new, dx)
            if c5_new < obj + 1e-10:

            step *= 0.5
        
        h = h_new
        if c5_new < best_obj:
            best_obj = c5_new
            best_h = h.copy()
        
        # Adaptive learning rate
        if it % 100 == 0:
            lr = lr * 0.95
        
    return best_h, best_obj

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_grad(h, t, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    sh = t * (corr - mc)
    es = np.exp(sh)
    se = np.sum(es)
    p = es / se
    n = len(h)
    gh = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-12:
            continue
        s = k - (n - 1)
        a = max(0, -s)
        b = min(n, n - s)
        if a < b:
            gh[a:b] += pk * (1.0 - h[a+s:b+s])
        a2 = max(0, s)
        b2 = min(n, n + s)
        if a2 < b2:
            gh[a2:b2] -= pk * h[a2-s:b2-s]
    gh *= dx
    return (mc + np.log(se) / t) * dx, gh

def pgd_adam(h, dx, target, t, n_iter, lr=0.05):
    n = len(h)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    v = np.zeros(n)
    best = h.copy()
    best_c5 = compute_c5(h, dx)
    cur = h.copy()
    for it in range(n_iter):
        obj, grad = smooth_max_grad(cur, t, dx)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad

        c5 = compute_c5(cur, dx)
        if c5 < best_c5:

            best = cur.copy()

import numpy as np

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_grad(h, t, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    mc = np.max(corr)
    sh = t * (corr - mc)
    es = np.exp(sh)
    se = np.sum(es)
    p = es / se
    n = len(h)
    gh = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-10:
            continue
        s = k - (n - 1)
        a = max(0, -s)
        b = min(n, n - s)
        if a < b:
            gh[a:b] += pk * (1.0 - h[a + s:b + s])
        a2 = max(0, s)
        b2 = min(n, n + s)
        if a2 < b2:
            gh[a2:b2] -= pk * h[a2 - s:b2 - s]
    gh *= dx
    return (mc + np.log(se) / t) * dx, gh

def pgd_adam(h, dx, target, t, n_iter, lr=0.05):
    n = len(h)
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    best = h.copy()
    best_c5 = compute_c5(h, dx)
    cur = h.copy()
    stall = 0
    for it in range(n_iter):
        _, grad = smooth_max_grad(cur, t, dx)
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        mhat = m / (1 - b1 ** (it + 1))
        vhat = v / (1 - b2 ** (it + 1))

            best_c5 = c5
            best = cur.copy()
            stall = 0
        else:
            stall += 1
            if stall > 2000 and it < n_iter - 1:
                step *= 0.5
    return best, best_c5

def subgrad_refine(h, dx, target, n_iter, lr0=0.005):
    n = len(h)
    cur = h.copy()
    best = h.copy()
    best_c5 = compute_c5(h, dx)
    avg = np.zeros(n)
    for it in range(n_iter):
        corr = np.correlate(cur, 1 - cur, mode="full")
        k = np.argmax(corr)
        s = k - (n - 1)
        grad = np.zeros(n)
        a = max(0, -s)
        b = min(n, n - s)
        if a < b:
            grad[a:b] += (1.0 - cur[a + s:b + s])
        a2 = max(0, s)
        b2 = min(n, n + s)
        if a2 < b2:
            grad[a2:b2] -= cur[a2 - s:b2 - s]
        lr = lr0 / np.sqrt(it + 1)
        cur = project_box_sum(cur - lr * grad * dx, target)
        avg = (it * avg + cur) / (it + 1)
        c5 = compute_c5(cur, dx)
        c5_avg = compute_c5(avg, dx)

            best_c5 = c5
            best = cur.copy()
        if c5_avg < best_c5:

            best = avg.copy()
    return best, best_c5
