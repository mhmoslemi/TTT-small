def project_capped(v, target, lo=0.0, hi=1.0):
    """Project v onto {x: lo<=x<=hi, sum(x)=target}"""
    n = len(v)
    # Binary search on tau
    a = -10.0
    b = 10.0

        s = np.sum(np.clip(v - mid, lo, hi))
        if s > target:
            a = mid
        else:
            b = mid

def pgd_smooth_max(h, dx, target_sum, t=50.0, max_iter=5000, alpha_init=0.1):
    """Projected gradient descent on smooth max of correlation"""
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_val = compute_c5(h, dx)
    
    alpha = alpha_init
    n_corr = 2 * n - 1
    lags = np.arange(n_corr) - n + 1
    
    for it in range(max_iter):
        corr = np.correlate(h, 1 - h, mode="full")
        max_c = np.max(corr)
        shifted = t * (corr - max_c)
        exp_s = np.exp(shifted)
        softmax = exp_s / np.sum(exp_s)
        
        # Compute gradient
        grad = np.zeros(n)
        for i in range(n):
            s = 0.0
            for m in range(n_corr):
                lag = lags[m]
                term = 0.0
                j1 = i - lag
                if 0 <= j1 < n:
                    term += (1.0 - h[j1])
                j2 = i + lag
                if 0 <= j2 < n:
                    term -= h[j2]
                s += softmax[m] * term
            grad[i] = s * dx  # multiply by dx since objective includes dx
        
        # Backtracking line search
        alpha_try = alpha
        val_0 = max_c * dx
        for _ in range(20):
            v = h - alpha_try * grad
            h_try = project_capped(v, target_sum)
            val_try = compute_c5(h_try, dx)
            if val_try < val_0:
                break
            alpha_try *= 0.5
        else:
            alpha = alpha_try
            break
        
        h = h_try
        alpha = min(alpha * 1.05, 1.0)
        
        val = compute_c5(h, dx)
        if val < best_val:
            best_val = val
            best_h = h.copy()
    
    return best_h, best_val

        s = np.sum(np.clip(v - mid, lo, hi))
        if s > target:
            a = mid
        else:
            b = mid
    return np.clip(v - (a + b) / 2.0, lo, hi)

def vectorized_gradient(h, softmax, n):
    n_corr = 2 * n - 1
    lags = np.arange(n_corr) - n + 1
    i_idx = np.arange(n)[:, None]
    lags_row = lags[None, :]
    idx1 = i_idx - lags_row
    idx2 = i_idx + lags_row
    valid1 = (idx1 >= 0) & (idx1 < n)
    valid2 = (idx2 >= 0) & (idx2 < n)
    h1 = np.where(valid1, np.take(h, np.clip(idx1, 0, n-1)), 0.0)
    h2 = np.where(valid2, np.take(h, np.clip(idx2, 0, n-1)), 0.0)
    part1 = (1.0 - h1) * valid1
    part2 = h2 * valid2
    return (part1 - part2).T.dot(softmax)

def pgd_optimize(h, dx, target_sum, t=50.0, max_iter=3000):
    n = len(h)
    h = h.copy().astype(float)
    best_h = h.copy()
    best_val = compute_c5(h, dx)
    alpha = 0.1 / t  # initial step size
    
    for it in range(max_iter):
        corr = get_correlation(h)
        max_c = np.max(corr)
        shifted = t * (corr - max_c)
        exp_s = np.exp(shifted)
        softmax = exp_s / np.sum(exp_s)
        
        grad = vectorized_gradient(h, softmax, n) * dx
        
        # Backtracking
        a = alpha
        val0 = max_c * dx
        for _ in range(15):
            v = h - a * grad
            h_try = project_capped(v, target_sum)
            val_try = compute_c5(h_try, dx)
            if val_try < val0 * 0.999:
                h = h_try
                break
            a *= 0.5
        else:
            break
        
        alpha = min(a * 1.2, 1.0)
        val = compute_c5(h, dx)
        if val < best_val:
            best_val = val
            best_h = h.copy()
    
    return best_h, best_val

def to_binary(h, target_sum):
    """Convert h to binary with exactly target_sum ones"""
    n = len(h)
    # Sort by h value, assign 1 to top target_sum
    order = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[order[:int(target_sum)]] = 1.0
    return h_bin

def binary_local_search(h_bin, dx, n_swaps=100000):

    n = len(h_bin)
    h = h_bin.copy()
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    best_c5 = compute_c5(h, dx)
    
    corr = get_correlation(h)
    
    for _ in range(n_swaps):
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        if i == j:
            continue
        # Try swap
        h[i] = 0
        h[j] = 1
        c5_new = np.max(get_correlation(h)) * dx
        if c5_new < best_c5:
            best_c5 = c5_new
            ones = np.delete(ones, np.where(ones == i)[0][0])
            zeros = np.delete(zeros, np.where(zeros == j)[0][0])
            ones = np.append(ones, j)
            zeros = np.append(zeros, i)
        else:
            h[i] = 1
            h[j] = 0
    
    return h, best_c5

def simulated_annealing(h_bin, dx, n_steps=500000, T0=0.5, T1=0.01):
    n = len(h_bin)

    zeros = np.where(h < 0.5)[0]
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    for step in range(n_steps):
        T = T0 * (T1 / T0) ** (step / n_steps)
        frac = step / n_steps
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        if i == j:
            continue
        c5_old = compute_c5(h, dx)
        h[i] = 0
        h[j] = 1
        c5_new = compute_c5(h, dx)
        delta = c5_new - c5_old
        if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-8)):
            # Accept
            ones = np.delete(ones, np.where(ones == i)[0][0])
            zeros = np.delete(zeros, np.where(zeros == j)[0][0])
            ones = np.append(ones, j)
            zeros = np.append(zeros, i)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            h[i] = 1
            h[j] = 0
    
    return best_h, best_c5
