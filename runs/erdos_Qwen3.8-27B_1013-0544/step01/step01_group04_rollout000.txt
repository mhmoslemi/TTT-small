    dx_final = 2.0 / n_final
    phase4_end = budget_s - 15
    for attempt in range(80):
        if elapsed() > phase4_end:
            break
        scale = 0.015 if attempt < 40 else 0.008
        h_try = best_h + rng.randn(n_final) * scale
        h_try = project_feasible(h_try, n_final)
        h_cd, c5_cd = coordinate_descent(h_try, n_final, dx_final, max_sweeps=100, grid_size=400)
        if c5_cd < best_c5:
            best_h, best_c5 = h_cd, c5_cd
    
    return (best_h, best_c5, len(best_h))
