    a, b = h[i], h[j]
    if direction == 1:

        delta_max = min(1.0 - a, b)

        h_new[i] += direction * delta
        h_new[j] -= direction * delta
        return compute_c5(h_new, 1.0)  # dx cancels for comparison

import time
import scipy.optimize as opt

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target_sum):
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        # Distribute among non-saturated coordinates
        free = np.where((h > 1e-6) & (h < 1 - 1e-6))[0]
        if len(free) == 0:
            h += diff / len(h)
        else:
            h[free] += diff / len(free)
        h = np.clip(h, 0.0, 1.0)
    return h

def pair_exchange_objective(h, i, j, direction, dx):
    """Objective for pair exchange: h[i] += dir*delta, h[j] -= dir*delta"""
    def obj(delta):
        h[i] += direction * delta
        h[j] -= direction * delta
        val = compute_c5(h, dx)
        h[i] -= direction * delta
        h[j] += direction * delta
        return val
    return obj

def do_pair_exchange(h, i, j, dx, rng):
    """Perform optimal pair exchange between i and j."""
    a, b = h[i], h[j]
    direction = 1 if rng.randint(2) == 0 else -1
    
    if direction == 1:
        d_min = max(0.0, -a)
        d_max = min(1.0 - a, b)
    else:
        d_min = max(0.0, -b)
        d_max = min(1.0 - b, a)
    
    if d_max - d_min < 1e-10:
        return 0.0
    
    a0, b0 = a, b
    def obj(delta):
        h[i] = a0 + direction * delta
        h[j] = b0 - direction * delta
        return compute_c5(h, dx)
    
    result = opt.minimize_scalar(obj, bounds=(d_min + 1e-8, d_max - 1e-8),
                                  method='bounded', options={'xatol': 1e-7})
    
    delta = result.x
    h[i] = a0 + direction * delta
    h[j] = b0 - direction * delta
    return delta

def exchange_descent(h, dx, n_rounds, n_pairs_per_round, rng):

    n = len(h)
    current_c5 = compute_c5(h, dx)

                grad[second] -= weights[k] * h[j]
    
    h_new = h - lr * grad
    h_new = fix_sum(h_new, target_sum)

    # Phase 1: Random pair exchanges (main workhorse)
    stall = 0
    max_stall_to_restart = 30
    for r in range(20000):
        if elapsed() > budget_s - 120:
            break
        n_pairs = n
        improved = False
        for _ in range(n_pairs):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            c5_new = pair_exchange(h, i, j, dx)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h.copy()
                improved = True
        if not improved:
            stall += 1
        else:
            stall = 0

        # Stuck: try group SLSQP
        if stall >= 3 and stall < max_stall_to_restart and elapsed() < budget_s - 60:
            gs = rng.randint(3, 8)
            idx = rng.choice(n, gs, replace=False)
            c5_new = group_slsqp(h, idx, dx, target, maxiter=200)
            c5_new = compute_c5(fix_sum(h, target), dx)
            h = fix_sum(h, target)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0

        # Very stuck: restart from perturbed best
        if stall >= max_stall_to_restart and elapsed() < budget_s - 40:
            h = best_h.copy()
            n_pert = rng.randint(2, 8)
            idx = rng.choice(n, n_pert, replace=False)
            h[idx] += rng.uniform(-0.15, 0.15, n_pert)
            h = fix_sum(h, target)
            stall = 0

    # Phase 2: Systematic sweep (all pairs in order)
    if elapsed() < budget_s - 60:
        for i in range(n):
            if elapsed() > budget_s - 30:
                break
            for j in range(i + 1, n):
                if elapsed() > budget_s - 20:
                    break
                pair_exchange(h, i, j, dx)
        c5_now = compute_c5(fix_sum(h, target), dx)
        h = fix_sum(h, target)
        if c5_now < best_c5:
            best_c5 = c5_now
            best_h = h.copy()

    # Phase 3: Second systematic sweep with different order
    if elapsed() < budget_s - 40:
        order = rng.permutation(n)
        for ii in range(n):
            if elapsed() > budget_s - 15:
                break
            i = order[ii]
            for jj in range(n):
                j = order[jj]
                if i == j:
                    continue
                if elapsed() > budget_s - 10:
                    break
                pair_exchange(h, i, j, dx)
        c5_now = compute_c5(fix_sum(h, target), dx)
        h = fix_sum(h, target)
        if c5_now < best_c5:
            best_c5 = c5_now
            best_h = h.copy()

    # Phase 4: Try n=80 (downsample and reoptimize)
    if elapsed() < budget_s - 20:
        n2 = n // 2
        dx2 = 2.0 / n2
        target2 = n2 / 2.0
        x_old = np.linspace(0, 2, n)
        x_new = np.linspace(0, 2, n2)
        h2 = np.interp(x_new, x_old, best_h)
        h2 = fix_sum(h2, target2)
        for _ in range(300):
            if elapsed() > budget_s - 5:
                break
            i = rng.randint(n2)
            j = rng.randint(n2)
            if i != j:
                pair_exchange(h2, i, j, dx2)
        c5_2 = compute_c5(fix_sum(h2, target2), dx2)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    final_n = len(best_h)
    final_dx = 2.0 / final_n
    best_h = fix_sum(best_h, final_n / 2.0)
    final_c5 = compute_c5(best_h, final_dx)

    return best_h, final_c5, final_n
