import numpy as np
import time
from scipy.optimize import minimize

# Global state for objective functions (to avoid lambdas/closures)
_g = {}

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def corr_of(h):
    return np.correlate(h, 1 - h, mode="full")

def smooth_max_obj(h, t, dx):
    corr = corr_of(h)
    max_c = np.max(corr)
    shifted = t * (corr - max_c)
    return (max_c + (1.0 / t) * np.log(np.sum(np.exp(shifted)))) * dx

def topk_equalize_obj(h, K, lam, dx):
    corr = corr_of(h)
    sorted_c = np.sort(corr)
    topk = sorted_c[-K:]
    return (np.mean(topk) + lam * np.var(topk)) * dx

def fix_sum(h, target_sum):
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
    return np.clip(h, 0.0, 1.0)

def set_globals(obj_type, t, K, lam, dx, target_sum):
    _g['obj_type'] = obj_type
    _g['t'] = t
    _g['K'] = K
    _g['lam'] = lam
    _g['dx'] = dx
    _g['target_sum'] = target_sum

def objective_wrapper(x):
    h = fix_sum(x, _g['target_sum'])
    if _g['obj_type'] == 'smooth':
        return smooth_max_obj(h, _g['t'], _g['dx'])
    else:
        return topk_equalize_obj(h, _g['K'], _g['lam'], _g['dx'])

def constraint_wrapper(x):
    return np.sum(x) - _g['target_sum']

    n = len(h)

    try:
        res = minimize(objective_wrapper, h.copy(), method='SLSQP',
                      bounds=[(0.0, 1.0)] * n,
                      constraints=[{'type': 'eq', 'fun': constraint_wrapper}],
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return fix_sum(res.x, target_sum)
    except Exception:
        return h

        h_sub[idx] = x

        else:
            return topk_equalize_obj(h_sub, K, lam, 2.0 / n)
