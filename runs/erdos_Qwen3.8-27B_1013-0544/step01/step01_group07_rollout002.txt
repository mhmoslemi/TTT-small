import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1 - h, mode="full")

def project_box_sum(h, target_sum):
    n = len(h)
    lo, hi = -5.0, 5.0
    def s(lam):
        return np.sum(np.clip(h - lam, 0.0, 1.0))
    for _ in range(30):
        if s(lo) < target_sum:
            lo -= 5.0
        if s(hi) > target_sum:
            hi += 5.0
        if s(lo) >= target_sum and s(hi) <= target_sum:
            break
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if s(mid) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def subgradient_step(h, k_star, alpha, target_sum):
    n = len(h)
    lag = k_star - (n - 1)
    sg = np.zeros(n)
    if lag >= 0:
        if lag < n:
            sg[:n - lag] += (1.0 - h[lag:])
            sg[lag:] -= h[:n - lag]
    else:
        la = -lag
        if la < n:
            sg[la:] += (1.0 - h[:n - la])
            sg[:n - la] -= h[la:]
    return project_box_sum(h - alpha * sg, target_sum)

def smooth_max_topk(h, t, k_top):
    corr = get_corr(h)
    if k_top < len(corr):
        top = np.partition(corr, -k_top)[-k_top:]
    else:
        top = corr
    mx = np.max(top)
    return mx + (1.0 / t) * np.log(np.sum(np.exp(t * (top - mx))))

def slsqp_refine(h, dx, target_sum, t, k_top, maxiter):
    n = len(h)
    def obj(x):
        return smooth_max_topk(x, t, k_top) * dx
    try:
        res = minimize(obj, h.copy(), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_box_sum(res.x, target_sum)
    except Exception:
        return h.copy()

def subgrad_phase(h, dx, target_sum, n_iters, alpha0, alpha_min, alpha_decay, patience_max):
    n = len(h)
    best = compute_c5(h, dx)
    alpha = alpha0
    patience = 0
    h_curr = h.copy()
    for it in range(n_iters):
        corr = get_corr(h_curr)
        k_star = int(np.argmax(corr))
        h_new = subgradient_step(h_curr, k_star, alpha, target_sum)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best:
            h_curr = h_new
            best = c5_new
            patience = 0
            alpha = min(alpha * 1.03, 0.5)
        else:
            patience += 1
            if patience >= patience_max:
                alpha *= alpha_decay
                patience = 0
                if alpha < alpha_min:
                    break
    return h_curr, best

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)
