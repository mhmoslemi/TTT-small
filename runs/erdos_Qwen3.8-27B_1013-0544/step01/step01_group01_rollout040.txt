```python
def sm_vg(h, beta):
    n = len(h)
    dx = 2.0 / n
    om = 1.0 - h
    c = np.correlate(h, om, mode="full")
    mx = c.max()
    ec = np.exp(beta * (c - mx))
    es = ec.sum()
    w = ec / es
    val = (mx + np.log(es) / beta) * dx
    g = np.zeros(n)
    for ki in range(len(c)):
        wi = w[ki]
        if wi < 1e-14:
            continue
        lag = ki - (n - 1)
        if lag > 0:
            g[lag:] += wi * om[:n - lag]
            g[:n - lag] -= wi * h[lag:]
        elif lag < 0:
            l2 = -lag
            g[:n - l2] += wi * om[l2:]
            g[l2:] -= wi * h[:n - l2]
        else:
            g += wi * (1.0 - 2.0 * h)
    return val, g * dx
```
