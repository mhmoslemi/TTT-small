def coordinate_descent_step(h, j, n):
    """Find optimal h[j] given all other coordinates fixed."""
    # For each lag m, corr[n-1+m] = a_m * h[j] + b_m
    # where a_m and b_m depend on other coordinates
    n_lags = 2 * n - 1
    a = np.zeros(n_lags)
    b = np.zeros(n_lags)
    
    for m in range(-(n-1), n):
        k = n - 1 + m
        # corr[k] = sum_i h[i] * (1 - h[i+m]) for valid i
        # Terms involving h[j]:
        # i = j: h[j] * (1 - h[j+m]) if 0 <= j+m < n
        # i = j-m: h[j-m] * (1 - h[j]) if 0 <= j-m < n
        coeff = 0.0
        const = 0.0
        for i in range(max(0, -m), min(n, n-m)):
            if i == j:
                # h[j] * (1 - h[i+m]) = h[j] * (1 - h[j+m])
                if 0 <= j + m < n:
                    coeff += (1 - h[j+m])
                # h[j] is the variable, so this contributes to coeff
            elif i + m == j:
                # h[i] * (1 - h[j]) = h[i] - h[i]*h[j]
                if 0 <= i < n:
                    coeff += -h[i]
                    const += h[i]
            else:
                # No h[j] dependence
                if 0 <= i + m < n:
                    const += h[i] * (1 - h[i+m])
        a[k] = coeff
        b[k] = const
    
    # Now minimize max_k (a[k]*x + b[k]) over x in [0,1]
    # Collect all critical points
    candidates = [0.0, 1.0]
    for i in range(n_lags):
        for j2 in range(i+1, n_lags):
            if a[i] != a[j2]:
                x = (b[j2] - b[i]) / (a[i] - a[j2])
                if 0 <= x <= 1:
                    candidates.append(x)
    
    best_x = 0.0
    best_val = np.inf
    for x in candidates:
        vals = a * x + b
        v = np.max(vals)
        if v < best_val:
            best_val = v
            best_x = x
    return best_x

from scipy.optimize import minimize, minimize_scalar, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) > 1e-10:
        h += (target_sum - s) / len(h)
        h = np.clip(h, 0.0, 1.0)
    # Second correction for clipping
    s = h.sum()
    if abs(s - target_sum) > 1e-10:
        # Iterative water-filling
        for _ in range(10):
            h += (target_sum - s) / len(h)
            h = np.clip(h, 0.0, 1.0)
            s = h.sum()
            if abs(s - target_sum) < 1e-12:
                break
    return h

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def obj_and_grad(h, beta=100):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    obj = smooth_max(corr, beta) * dx
    
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    weights = exps / np.sum(exps)
    
    grad = np.zeros(n)
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-12:
            continue
        lag = k - (n - 1)
        for j in range(n):
            g = 0.0
            if 0 <= j + lag < n:
                g += (1.0 - h[j + lag])
            if 0 <= j - lag < n:
                g += (-h[j - lag])
            grad[j] += w * g
    grad *= dx
    return obj, grad

def run_lqfgsb(h0, target_sum, beta=100, penalty=500.0, maxiter=500):
    n = len(h0)
    bounds = Bounds(np.zeros(n), np.ones(n))
    
    def obj(x):
        o, _ = obj_and_grad(x, beta)
        cv = (np.sum(x) - target_sum) / target_sum
        return o + penalty * cv * cv
    
    def grad(x):
        _, g = obj_and_grad(x, beta)
        cv = (np.sum(x) - target_sum) / target_sum
        g = g + 2 * penalty * cv / target_sum
        return g
    
    res = minimize(obj, h0, jac=grad, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8})
    return project_h(res.x, target_sum)

def coord_descent(h, max_sweeps=15):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    for sweep in range(max_sweeps):
        max_change = 0.0
        for j in range(n):
            old = h[j]
            corr = np.correlate(h, 1.0 - h, mode="full")
            n_lags = 2 * n - 1

import numpy as np
from scipy.optimize import minimize, minimize_scalar, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20):
        s = h.sum()
        if abs(s - target_sum) < 1e-13:
            break
        h += (target_sum - s) / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta


def grad_obj(x, beta):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    exps = np.exp(beta * (corr - m))
    weights = exps / np.sum(exps)
    grad = np.zeros(n)
    n_lags = 2 * n - 1
    for k in range(n_lags):
        w = weights[k]
        if w < 1e-12:
            continue
        lag = k - (n - 1)
        if lag == 0:
            g_vec = 1.0 - 2.0 * x
        elif lag > 0:
            g_vec = np.zeros(n)
            g_vec[:n - lag] += (1.0 - x[lag:])
            g_vec[lag:] += (-x[:n - lag])
        else:
            g_vec = np.zeros(n)
            g_vec[-lag:] += (1.0 - x[:n + lag])
            g_vec[:n + lag] += (-x[-lag:])
        grad += w * g_vec
    return grad * dx


def obj_lbfgsb(x, beta, penalty, target_sum):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    o = smooth_max(corr, beta) * dx
    cv = (np.sum(x) - target_sum) / target_sum
    return o + penalty * cv * cv


def grad_lbfgsb(x, beta, penalty, target_sum):
    g = grad_obj(x, beta)
    cv = (np.sum(x) - target_sum) / target_sum
    g = g + 2.0 * penalty * cv / target_sum
    return g


def f1d_max_linear(x, a, b):
    return np.max(a * x + b)


def coord_descent(h, max_sweeps=20):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    dir_order = 1
    for sweep in range(max_sweeps):
        max_change = 0.0
        indices = list(range(n))
        if dir_order < 0:
            indices.reverse()
        for j in indices:
            old = h[j]
            corr = np.correlate(h, 1.0 - h, mode="full")
            n_lags = 2 * n - 1
            a = np.zeros(n_lags)
            for k in range(n_lags):
                lag = k - (n - 1)
                ak = 0.0
                if 0 <= j + lag < n:
                    ak += (1.0 - h[j + lag])
                if 0 <= j - lag < n:
                    ak += (-h[j - lag])
                a[k] = ak
            b = corr - a * old
            res = minimize_scalar(f1d_max_linear, bounds=(0.0, 1.0),
                                  args=(a, b), method='bounded',
                                  options={'xatol': 1e-10})
            h[j] = res.x
            max_change = max(max_change, abs(res.x - old))
        h = project_h(h, target)
        dir_order = -dir_order
        if max_change < 1e-10:
            break
    return h


def run_lbfgsb(h0, target_sum, beta=100, penalty=1000.0, maxiter=300):
    n = len(h0)
    bounds = Bounds(np.zeros(n), np.ones(n))
    res = minimize(obj_lbfgsb, h0, jac=grad_lbfgsb, method='L-BFGS-B',
                   bounds=bounds, args=(beta, penalty, target_sum),
                   options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-7})
    return project_h(res.x, target_sum)


def run_slsqp(h0, target_sum, beta=200, maxiter=500):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n

    def obj_s(x):
        return smooth_max(np.correlate(x, 1.0 - x, mode="full"), beta) * (2.0 / n)

    def con_fun(x):
        return np.sum(x) - target_sum

    def con_jac(x):
        return np.ones(n)

    constraints = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    res = minimize(obj_s, h0, jac=grad_obj, method='SLSQP',
                   bounds=bounds, constraints=constraints,
                   args=(beta,), options={'maxiter': maxiter, 'ftol': 1e-13})
    return project_h(res.x, target_sum)


def structured_init(n, seed):
    rng = np.random.RandomState(seed)
    patterns = []
    x = np.linspace(0, 1, n)
    patterns.append(0.5 + 0.3 * np.sin(2 * np.pi * x))
    patterns.append(0.5 + 0.3 * np.sin(4 * np.pi * x))
    patterns.append(np.clip(0.8 * x + 0.1, 0, 1))
    patterns.append(np.clip(1.0 - 0.8 * x, 0, 1))
    patterns.append(0.5 + 0.4 * np.cos(2 * np.pi * 3 * x))
    patterns.append(np.where(np.arange(n) % 2 == 0, 0.7, 0.3))
    patterns.append(np.where(np.arange(n) % 3 == 0, 0.8, 0.35))
    block = max(3, n // 7)
    p = np.zeros(n)
    for i in range(0, n, block * 2):
        p[i:i + block] = 0.8
        p[i + block:i + 2 * block] = 0.3 if i + 2 * block <= n else 0.3
    patterns.append(p)
    patterns.append(rng.random(n) * 0.4 + 0.3)
    return patterns


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_points = len(h_init)
    else:
        n_points = 70
        h_init = np.full(n_points, 0.5)

    target_sum = n_points / 2.0
    best_h = project_h(h_init, target_sum)
    best_c5 = compute_c5(best_h)

    def time_left():
        return budget_s - (time.time() - start_time)

    # Phase 1: Initial refinement with L-BFGS-B
    if time_left() > 120:

    # Phase 4: Structured restarts

                break
            h0 = project_h(np.array(pat, dtype=float), target_sum)
            h_new = run_lbfgsb(h0, target_sum, beta=100, penalty=2000, maxiter=300)
            h_new = coord_descent(h_new, max_sweeps=20)
            c5 = compute_c5(h_new)
            if c5 < best_c5:

    # Phase 5: Diverse restarts with annealing
    attempt = 0
    temp = 0.002
    while time_left() > 30:
        attempt += 1
        temp = 0.002 * np.exp(-attempt / 50.0)

        strategy = attempt % 5
        if strategy == 0:
            scale = 0.04 + 0.08 * np.exp(-attempt / 40.0)
            n_p = max(5, n_points // 5)
            idx = np.random.choice(n_points, n_p, replace=False)
            h0 = best_h.copy()
            h0[idx] += np.random.randn(n_p) * scale
        elif strategy == 1:
            h0 = best_h.copy()
            n_blks = np.random.randint(2, 5)
            for _ in range(n_blks):
                blen = np.random.randint(2, max(3, n_points // 10))
                bs = np.random.randint(0, max(1, n_points - blen))
                h0[bs:bs + blen] += np.random.randn(blen) * 0.08
        elif strategy == 2:
            h0 = best_h.copy()
            i1 = np.random.randint(0, n_points - 1)
            i2 = np.random.randint(0, n_points - 1)
            h0[i1], h0[i2] = h0[i2], h0[i1]
            if i1 + 1 < n_points and i2 + 1 < n_points:
                h0[i1 + 1], h0[i2 + 1] = h0[i2 + 1], h0[i1 + 1]
        elif strategy == 3:
            # Subspace optimization: pick random subset
            h0 = best_h.copy()
            n_sub = np.random.randint(8, min(20, n_points))

            frac = np.random.uniform(0.2, 0.5)

            h0[mask] = 1.0 - h0[mask]

        h0 = project_h(h0, target_sum)

        if temp > 0.0005:

        c5 = compute_c5(h_new)

    return best_h, best_c5, n_points
