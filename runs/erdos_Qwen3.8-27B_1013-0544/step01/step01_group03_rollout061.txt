import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val(obj, tau):
    x = tau * obj
    x_max = np.max(x)
    return float(x_max / tau + np.log(np.sum(np.exp(x - x_max))) / tau)

def fix_sum(h, target):
    h = np.array(h, dtype=float)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return np.clip(h, 0.0, 1.0)
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = mask.sum()
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def binary_search(h_start, n, target, n_iter, t_start, t_end):
    """Simulated annealing on binary solutions."""
    rng = np.random.RandomState(42)
    # Start from thresholded version
    h = (h_start >= np.median(h_start)).astype(float)
    # Fix to have exactly target ones
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if len(ones) > target:
        excess = len(ones) - target
        remove = rng.choice(ones, excess, replace=False)
        h[remove] = 0
    elif len(ones) < target:
        deficit = target - len(ones)
        add = rng.choice(zeros, deficit, replace=False)
        h[add] = 1

    ones = np.where(h == 1)[0]

    for it in range(n_iter):
        T = t_start * (t_end / t_start) ** (it / n_iter)
        # Pick a random 1 and a random 0 to swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i == j:

        h[i], h[j] = h[j], h[i]
        c5_new = evaluate_c5(h)
        if c5_new < best_c5 or rng.random() < np.exp(-(c5_new - best_c5) / max(T, 1e-10)):
            best_h = h.copy()
            best_c5 = c5_new
        else:
            h[i], h[j] = h[j], h[i]  # revert
    
    return best_h, best_c5

```python
def smooth_max_grad(h, tau):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    # Compute softmax weights
    x = tau * corr
    x_max = np.max(x)
    p = np.exp(x - x_max) / np.sum(np.exp(x - x_max))
    # p has length 2n-1, indexed by m = 0, ..., 2n-2
    # k = m - (n-1), so k ranges from -(n-1) to n-1
    
    grad = np.zeros(n)
    for i in range(n):
        s1 = 0.0
        s2 = 0.0
        # First sum: k from i-(n-1) to n-1, term: p[k+(n-1)] * (1-h[i-k])
        k_min = i - (n-1)
        k_max = n - 1
        for k in range(k_min, k_max + 1):
            if 0 <= i - k < n:
                s1 += p[k + (n-1)] * (1.0 - h[i - k])
        # Second sum: k from -i to n-1-i, term: p[k+(n-1)] * h[i+k]
        k_min2 = -i
        k_max2 = n - 1 - i
        for k in range(k_min2, k_max2 + 1):
            if 0 <= i + k < n:
                s2 += p[k + (n-1)] * h[i + k]
        grad[i] = (s1 - s2) / tau
    return grad
```

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def evaluate_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * 2.0 / n)

def fix_sum(h, target):
    h = np.array(h, dtype=float)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return np.clip(h, 0.0, 1.0)
    mask = h < 1.0 if diff > 0 else h > 0.0
    cnt = mask.sum()
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(h, tau):
    corr = np.correlate(h, 1.0 - h, mode="full")
    x = tau * corr
    xm = np.max(x)
    return float(xm / tau + np.log(np.sum(np.exp(x - xm))) / tau)

def smooth_max_full(h, tau):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    x = tau * corr
    xm = np.max(x)
    obj = float(xm / tau + np.log(np.sum(np.exp(x - xm))) / tau)
    p = np.exp(x - xm) / np.sum(np.exp(x - xm))
    idx = np.arange(n)
    pi1 = idx[:, None] + (n - 1) - idx[None, :]
    pi2 = idx[None, :] - idx[:, None] + (n - 1)
    s1 = p[pi1] @ (1.0 - h)
    s2 = p[pi2] @ h
    return obj, (s1 - s2) / tau

def make_slsqp_obj(tau):
    return lambda x: smooth_max_obj(x, tau)

def make_slsqp_grad(tau):
    return lambda x: smooth_max_full(x, tau)[1]

def make_slsqp_cons(n):
    target = n / 2.0
    return [{'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': lambda x: np.ones(n)}]

def slsqp_opt(h0, n, tau, maxiter=300):
    bnds = Bounds(np.zeros(n), np.ones(n))
    cons = make_slsqp_cons(n)
    try:
        res = minimize(make_slsqp_obj(tau), h0, method='SLSQP', bounds=bnds,
                       jac=make_slsqp_grad(tau), constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        x = fix_sum(res.x, n / 2.0)
        return x, evaluate_c5(x)
    except Exception:
        return h0, None

def binary_sa(h_init, n, target, n_iter, rng, T0=0.4, T1=0.005):
    h = (h_init > 0.5).astype(float)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if len(ones) > target:
        h[rng.choice(ones, len(ones) - target, replace=False)] = 0
    elif len(ones) < target:
        h[rng.choice(zeros, target - len(ones), replace=False)] = 1
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    cur_c5 = best_c5
    for it in range(n_iter):
        T = T0 * (T1 / T0) ** (it / max(n_iter - 1, 1))
        a = ones[rng.randint(len(ones))]
        b = zeros[rng.randint(len(zeros))]
        h[a], h[b] = 0.0, 1.0
        c5n = evaluate_c5(h)
        if c5n < cur_c5 or rng.random() < np.exp(-(c5n - cur_c5) / max(T, 1e-12)):
            cur_c5 = c5n
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h.copy()
        else:
            h[a], h[b] = 1.0, 0.0
    return best_h, best_c5

    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()

    for tau in [5, 15, 50, 150, 500, 1500, 5000, 15000]:

    # Phase 5: Beatty/structured restarts
    if rem() > 40:
        hn = n_best
        target = hn / 2.0
