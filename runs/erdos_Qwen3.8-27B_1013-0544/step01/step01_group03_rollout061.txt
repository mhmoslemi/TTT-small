import numpy as np
from scipy.optimize import minimize, Bounds
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val(values, tau):
    x = values * tau
    x_max = np.max(x)
    return float(x_max / tau + np.log(np.sum(np.exp(x - x_max))) / tau)

def smooth_max_grad(corr, tau, x, n):
    """Compute gradient of smooth_max(corr, tau) w.r.t. x"""
    alpha = np.exp(tau * (corr - np.max(corr)))
    alpha_sum = alpha.sum()
    p = alpha / alpha_sum  # softmax weights
    
    # T1[m] = sum_k p[k] * (1-x[m+k-n+1])
    # T1[m] = np.correlate(1-x, p, 'full')[m+n-1]
    corr_1x_p = np.correlate(1.0 - x, p, 'full')
    T1 = corr_1x_p[n-1 : n-1+n]
    
    # T2[m] = sum_k p[k] * x[m-k+n-1]
    # T2[m] = T1 computed with p reversed
    p_rev = p[::-1]
    corr_1x_pred = np.correlate(1.0 - x, p_rev, 'full')
    T2 = corr_1x_pred[n-1 : n-1+n]
    
    return T1 - T2

def fix_sum(h, target):
    """Project h onto {x: sum(x)=target, 0<=x<=1} using bisection"""
    h = np.array(h, dtype=float)
    diff = target - h.sum()
    if abs(diff) < 1e-10:
        return np.clip(h, 0.0, 1.0)
    # Bisection for lambda
    if diff > 0:
        # Need to increase values, so increase those < 1
        lo, hi = -1.0, 1.0
    else:
        lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2
        projected = np.clip(h - mid, 0, 1)
        if projected.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(100, 0.5)
    
    best_h = h_init.copy()
    best_c5 = evaluate_c5(best_h)
    
    def remaining():
        return budget_s - (time.time() - t0)
    
    def optimize_h(h_start, tau, penalty_w=100.0, maxiter=2000):
        """Optimize using L-BFGS-B with analytical gradient"""
        n = len(h_start)
        target = n / 2.0
        
        def obj(x):
            corr = np.correlate(x, 1.0 - x, mode="full")
            sm = smooth_max_val(corr, tau)
            pen = penalty_w * (x.sum() - target) ** 2
            return sm + pen
        
        def grad(x):
            corr = np.correlate(x, 1.0 - x, mode="full")
            g = smooth_max_grad(corr, tau, x, n)
            g += 2.0 * penalty_w * (x.sum() - target)
            return g
        
        bnds = Bounds(np.zeros(n), np.ones(n))
        res = minimize(obj, h_start, method='L-BFGS-B', bounds=bnds,
                      jac=grad, options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10,
                                         'maxls': 50})
        x = fix_sum(res.x, target)
        return x
    
    # Phase 1: Annealing with increasing tau
    taus = [5, 10, 20, 50, 100, 200, 500, 1000, 2000, 5000]
    n0 = len(best_h)
    for tau in taus:
        if remaining() < 10:
            break
        h_new = optimize_h(best_h, tau, penalty_w=50.0, maxiter=500)
        h_new = fix_sum(h_new, n0 / 2.0)
        c5 = evaluate_c5(h_new)
        if c5 < best_c5:
            best_h = h_new
            best_c5 = c5
    
    # Phase 2: Try different resolutions
    for tn in [60, 80, 100, 120, 140, 160, 200]:
        if remaining() < 20:
            break
        target = tn / 2.0
        # Interpolate best_h to new resolution
        h_t = np.interp(np.linspace(0, 1, tn, endpoint=False), 
                        np.linspace(0, 1, len(best_h), endpoint=False), best_h)
        h_t = fix_sum(h_t, target)
        
        for tau in [50, 200, 1000]:
            if remaining() < 10:
                break
            h_new = optimize_h(h_t, tau, penalty_w=50.0, maxiter=300)
            h_new = fix_sum(h_new, target)
            c5 = evaluate_c5(h_new)
            if c5 < best_c5:
                best_h = h_new
                best_c5 = c5
            h_t = h_new
    
    # Phase 3: Random restarts
    hn = len(best_h)
    for _ in range(30):
        if remaining() < 10:
            break
        # Different types of perturbations
        rtype = np.random.randint(3)
        if rtype == 0:
            # Gaussian noise
            h_pert = best_h + np.random.normal(0, 0.1, hn)
        elif rtype == 1:
            # Sine perturbation
            freq = np.random.randint(1, 5)
            phase = np.random.uniform(0, 2*np.pi)
            h_pert = best_h + 0.1 * np.sin(2*np.pi*freq*np.arange(hn)/hn + phase)
        else:
            # Block perturbation
            h_pert = best_h.copy()
            block = hn // 5
            start = np.random.randint(0, hn - block)
            h_pert[start:start+block] += np.random.uniform(-0.2, 0.2)
        
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = fix_sum(h_pert, hn / 2.0)
        
        for tau in [100, 500]:
            if remaining() < 5:
                break
            h_new = optimize_h(h_pert, tau, penalty_w=50.0, maxiter=200)
            h_new = fix_sum(h_new, hn / 2.0)
            c5 = evaluate_c5(h_new)
            if c5 < best_c5:
                best_h = h_new
                best_c5 = c5
            h_pert = h_new
    
    # Phase 4: Final fine-tuning
    if remaining() > 5:
        hn = len(best_h)
        for tau in [2000, 5000, 10000]:
            if remaining() < 5:
                break
            h_new = optimize_h(best_h, tau, penalty_w=100.0, maxiter=500)
            h_new = fix_sum(h_new, hn / 2.0)
            c5 = evaluate_c5(h_new)
            if c5 < best_c5:
                best_h = h_new
                best_c5 = c5
    
    # Final projection
    best_h = fix_sum(best_h, len(best_h) / 2.0)
    best_c5 = evaluate_c5(best_h)
    
    return best_h, best_c5, len(best_h)

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val(values, tau):
    x = tau * values
    x_max = x.max()
    return float(x_max / tau + np.log(np.sum(np.exp(x - x_max))) / tau)

def compute_gradient(x, tau):
    N = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    xc = tau * corr
    xc_shifted = xc - xc.max()
    p = np.exp(xc_shifted)
    p /= p.sum()
    T1 = np.correlate(1.0 - x, p, mode='full')[N-1:2*N-1]
    T2 = np.convolve(x, p, mode='full')[N-1:2*N-1]
    return T1 - T2

def project_sum(h, target):
    h = np.array(h, dtype=float).copy()
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0

        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

            if remaining() < 12:
                break
            if tn == len(best_h):
                continue
            target = tn / 2.0
            h_t = np.interp(np.linspace(0, 2, tn, endpoint=False),
                           np.linspace(0, 2, len(best_h), endpoint=False), best_h)
            h_t = project_sum(h_t, target)
            for tau in [200, 1000, 3000]:
                if remaining() < 6:
                    break
                h_new = optimize_one(h_t, tau, 200.0, 300)
                h_new = project_sum(h_new, target)
                c5 = evaluate_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new
                    best_c5 = c5
                h_t = h_new

    # Phase 5: Final high-tau refinement
    if remaining() > 5:
        hn = len(best_h)
        for tau in [3000, 8000, 20000]:
            if remaining() < 4:
                break
            h_new = optimize_one(best_h, tau, 500.0, 600)
            h_new = project_sum(h_new, hn / 2.0)
            c5 = evaluate_c5(h_new)
            if c5 < best_c5:
                best_h = h_new
                best_c5 = c5

    # Final projection and evaluation
    best_h = project_sum(best_h, len(best_h) / 2.0)
    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, len(best_h)
