def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def softmax_weights(corr, alpha):
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    s = np.sum(w)
    if s < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= s
    return w

def grad_corr_at_shift(h, k):
    """Gradient of sum(h[i]*h[i+k]) w.r.t. h at shift k (for the product h*(1-h))"""
    n = len(h)
    g = np.zeros(n)
    # The term is sum_i h[i]*(1-h[i+k]) for valid i
    # d/dh[j] = (1-h[j+k]) if j+k valid + (-h[j-k]) if j-k valid
    if k >= 0:
        # sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])
        # d/dh[j] for j in [0, n-k-1]: (1-h[j+k])
        # d/dh[j] for j in [k, n-1]: -h[j-k]
        for j in range(n - k):
            g[j] += (1.0 - h[j + k])
        for j in range(k, n):
            g[j] -= h[j - k]
    else:
        p = -k
        for j in range(n - p):
            g[j] += (1.0 - h[j + p])
        for j in range(p, n):
            g[j] -= h[j - p]
    return g

def compute_c5(h):

            w[np.argmax(corr)] = 1.0

                    g[p:] -= best_h[:n-p]

        # ADAM update

Let me also add a trick: instead of full ternary search, I can often get a good answer in fewer iterations. The function f(x) = max(A_k + B_k*x) is piecewise linear with at most 2n-1 pieces. The optimal x is at a crossing of two adjacent pieces. I can find this more directly:

1. Find the line giving the max at x=0: k0 = argmax A
2. Find the line giving the max at x=1: k1 = argmax(A+B)
3. If k0 == k1, the max is at a boundary (x=0 if A[k0] <= A[k0]+B[k0], else x=1)
4. Otherwise, the optimal x is where line k0 and line k1 cross: x* = (A[k1]-A[k0])/(B[k0]-B[k1])
5. But this might not be the true optimum if other lines are above at x*
6. So we need to check: is max_k(A_k + B_k*x*) = A[k0] + B[k0]*x* = A[k1] + B[k1]*x*?
7. If not, iterate: find the new argmax at x*, update, repeat

This is the "cutting plane" approach and it converges in at most (2n-1) iterations, but usually much faster.

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_onto_simplex_box(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def coord_descent(h, deadline, target_sum):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    overlap = np.correlate(best_h, 1 - best_h, mode='full')
    iter_count = 0
    no_improve = 0
    rng_local = np.random.RandomState(12345)

    while time.time() < deadline:
        iter_count += 1
        sweep_improved = False

        for i in range(n):
            old_val = best_h[i]
            B = np.zeros(2 * n - 1)
            k_lo = max(-(n - 1), -i)
            k_hi = min(n - 1, n - 1 - i)
            if k_lo <= k_hi:
                ks = np.arange(k_lo, k_hi + 1)
                B[ks + n - 1] += (1.0 - best_h[i + ks])
            k_lo2 = max(-(n - 1), i - (n - 1))
            k_hi2 = min(n - 1, i)
            if k_lo2 <= k_hi2:
                ks2 = np.arange(k_lo2, k_hi2 + 1)
                B[ks2 + n - 1] -= best_h[i - ks2]

            A = overlap - old_val * B

            # Minimize max(A + B*x) over x in [0,1] via ternary search
            lo, hi = 0.0, 1.0

def adam_smooth_opt(h, deadline, target_sum, rng, alpha=200.0, lr=0.006):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    v_exp = np.zeros(n)
    t = 0
    patience = 0
    b1, b2, eps = 0.82, 0.999, 1e-8
    current_alpha = alpha

    while time.time() < deadline:
        t += 1
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        m = np.max(corr)
        w = np.exp(current_alpha * (corr - m))
        s = np.sum(w)
        if s < 1e-300:
            w = np.zeros_like(corr)
            w[np.argmax(corr)] = 1.0
        else:
            w /= s

                    g[k:] -= best_h[:n - k]

                    g[:n - p] += (1.0 - best_h[p:])
                if p > 0:
                    g[p:] -= best_h[:n - p]
            grad += w[idx] * g

        m_exp = b1 * m_exp + (1 - b1) * grad
        v_exp = b2 * v_exp + (1 - b2) * grad ** 2
