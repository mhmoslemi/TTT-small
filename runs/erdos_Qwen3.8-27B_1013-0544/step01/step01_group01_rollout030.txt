            grad[:n-ll] += one_minus_h[ll:]

- C_k1 + a_k1 * t* = C_k2 + a_k2 * t* = φ(t*)
- a_k1 < 0 < a_k2 (or a_k1 = 0, or a_k2 = 0)

An efficient approach: 
1. Sort the lines by slope (a_k)
2. Use a convex hull / upper envelope algorithm

1. For each pair (k1, k2) with a_k1 < a_k2, compute t = (C_k2 - C_k1)/(a_k1 - a_k2)

4. Return the best

def optimize_coordinate_v2(h, j, dx):
    n = len(h)
    n_lags = 2 * n - 1
    lags = np.arange(n_lags) - (n - 1)
    
    # a[k] for each lag
    a = np.ones(n_lags) * dx
    # Vectorized: for each lag, check if j+lag and j-lag are in bounds
    jq = j + lags  # j + lag for each lag
    jl = j - lags  # j - lag for each lag
    valid_q = (jq >= 0) & (jq < n)
    valid_l = (jl >= 0) & (jl < n)
    a[valid_q] -= h[jq[valid_q]]
    a[valid_l] -= h[jl[valid_l]]
    
    # C[k]
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    C = corr - h[j] * a
    
    # Find minimum of max_k(C[k] + a[k]*t) over t in [0,1]
    # Use the fact that the optimum is at a breakpoint
    # Vectorize: for all pairs (i,j2), compute intersection t
    # A[i,j2] = a[i] - a[j2], B[i,j2] = C[j2] - C[i]
    # t[i,j2] = B[i,j2] / A[i,j2] where A != 0
    
    # This creates a 139x139 matrix, which is fine
    A = a[:, None] - a[None, :]  # A[i,j] = a[i] - a[j]
    B = C[None, :] - C[:, None]  # B[i,j] = C[j] - C[i]
    
    # t[i,j] = B[i,j] / A[i,j]
    with np.errstate(divide='ignore', invalid='ignore'):
        T = B / A
    
    # Valid t: 0 < T < 1 and |A| > eps
    valid = (T > 0) & (T < 1) & (np.abs(A) > 1e-14)
    
    if valid.any():
        # For each valid (i,j), evaluate φ(T[i,j]) = max_k(C[k] + a[k]*T[i,j])
        # This is expensive if we do it for all valid pairs
        # Instead, just evaluate at a grid of t values that include the breakpoints
        t_candidates = np.unique(T[valid])
        # Add endpoints
        t_candidates = np.concatenate([[0.0], t_candidates, [1.0]])
        t_candidates = t_candidates[(t_candidates >= 0) & (t_candidates <= 1)]
        
        # Evaluate φ at each candidate
        # φ(t) = max_k(C[k] + a[k]*t)
        # Shape: (n_candidates, n_lags)
        vals = C[None, :] + a[None, :] * t_candidates[:, None]
        phi = vals.max(axis=1)
        idx = phi.argmin()
        best_t = t_candidates[idx]
    else:
        # No internal breakpoints, check endpoints
        if C.max() <= (C + a).max():
            best_t = 0.0
        else:
            best_t = 1.0
    
    return best_t

import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = h.copy()
    for _ in range(1000):
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() == 0:
            h = np.clip(h + diff / len(h), 0, 1)
            break
        h[mask] += diff / mask.sum()
    return h

def softmax_obj(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta) * dx

def coord_optimize(h, j, dx):
    """Find optimal h[j] given all other coordinates fixed. Returns new value."""
    n = len(h)
    n_lags = 2 * n - 1
    lags = np.arange(n_lags) - (n - 1)
    
    # Compute a[k] = coefficient of h[j] in f_k
    a = np.full(n_lags, dx)
    jq = j + lags
    jl = j - lags
    mq = (jq >= 0) & (jq < n)
    ml = (jl >= 0) & (jl < n)
    a[mq] -= h[jq[mq]]
    a[ml] -= h[jl[ml]]
    
    # Compute C[k] = f_k - h[j]*a[k]
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    C = corr - h[j] * a
    
    # Find t in [0,1] minimizing max_k(C[k] + a[k]*t)
    best_t = 0.0
    best_val = C.max()
    
    val1 = (C + a).max()
    if val1 < best_val:
        best_val = val1
        best_t = 1.0
    
    # Find all pairwise breakpoints in (0,1)
    A_mat = a[:, None] - a[None, :]
    B_mat = C[None, :] - C[:, None]
    with np.errstate(divide='ignore', invalid='ignore'):
        T = B_mat / A_mat
    valid = (T > 0) & (T < 1) & (np.abs(A_mat) > 1e-14)
    
    if valid.any():
        t_cands = np.unique(T[valid])
        t_cands = np.concatenate([[0.0], t_cands, [1.0]])
        vals = C[None, :] + a[None, :] * t_cands[:, None]
        phi = vals.max(axis=1)
        idx = phi.argmin()
        if phi[idx] < best_val:
            best_val = phi[idx]
            best_t = t_cands[idx]
    
    return float(best_t)

    start = time.time()

    else:

import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_h(h, target_sum):
    h = h.copy()
    for _ in range(1000):
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() == 0:
            h = np.clip(h + diff / len(h), 0, 1)
            break
        h[mask] += diff / mask.sum()
    return h


def softmax_obj(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta) * dx


def coord_optimize(h, j, dx):
    n = len(h)
    n_lags = 2 * n - 1
    lags = np.arange(n_lags) - (n - 1)
    a = np.full(n_lags, dx)
    jq = j + lags
    jl = j - lags
    mq = (jq >= 0) & (jq < n)
    ml = (jl >= 0) & (jl < n)
    a[mq] -= h[jq[mq]]
    a[ml] -= h[jl[ml]]
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    C = corr - h[j] * a
    best_t = 0.0
    best_val = C.max()
    val1 = (C + a).max()
    if val1 < best_val:
        best_val = val1
        best_t = 1.0
    if n_lags <= 500:
        A_mat = a[:, None] - a[None, :]
        B_mat = C[None, :] - C[:, None]
        with np.errstate(divide='ignore', invalid='ignore'):
            T = B_mat / A_mat
        valid = (T > 0) & (T < 1) & (np.abs(A_mat) > 1e-14)
        if valid.any():
            t_cands = np.unique(T[valid])
            t_cands = np.concatenate([[0.0], t_cands, [1.0]])
