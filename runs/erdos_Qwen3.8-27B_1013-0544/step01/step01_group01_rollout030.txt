```python
def compute_gradient(h, n, beta):
    """Compute gradient of smooth_max objective."""
    dx = 2.0 / n
    # Compute all correlations
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full")  # length 2n-1
    g = corr * dx  # the actual values
    
    # Softmax weights
    max_g = np.max(g)
    exp_g = np.exp(beta * (g - max_g))
    weights = exp_g / np.sum(exp_g)  # length 2n-1
    
    # Gradient: sum_k weights[k] * d(g_k)/d(h[j])
    # d(g_k)/d(h[j]) = dx * [(1-h[j+k']) * I(0<=j+k'<n) - h[j-k'] * I(0<=j-k'<n)]
    # where k' = k - (n-1)
    
    grad = np.zeros(n)
    for k_idx in range(2*n-1):
        k = k_idx - (n-1)  # actual lag
        w = weights[k_idx]
        if w < 1e-15:
            continue
        # Contribution to grad[j] for this k
        # For each j: (1-h[j+k]) if j+k in [0,n) minus h[j-k] if j-k in [0,n)
        # First term: (1-h[j+k]) for j in [0, n-1-k] if k>=0, or j in [-k, n-1] if k<0
        # Second term: -h[j-k] for j in [k, n-1] if k>=0, or j in [0, n-1+k] if k<0
        
        # Term 1: (1-h[j+k]) for valid j
        if k >= 0:
            # j ranges from 0 to n-1-k
            valid_j1 = slice(0, n-k)
            grad[valid_j1] += w * dx * (1.0 - h[k:n])
        else:
            # j+k >= 0 means j >= -k
            valid_j1 = slice(-k, n)
            grad[valid_j1] += w * dx * (1.0 - h[0:n+k])
        
        # Term 2: -h[j-k] for valid j
        if k >= 0:
            # j-k >= 0 means j >= k
            valid_j2 = slice(k, n)
            grad[valid_j2] -= w * dx * h[0:n-k]
        else:
            # j-k < n means j < n+k
            valid_j2 = slice(0, n+k)
            grad[valid_j2] -= w * dx * h[-k:n]
    
    return grad
```

```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_value(g, beta):
    m = np.max(g)
    return m + np.log(np.sum(np.exp(beta * (g - m)))) / beta

def smooth_max_gradient(g, beta):
    """Return (value, softmax_weights) for smooth_max."""
    m = np.max(g)
    shifted = beta * (g - m)
    exp_shifted = np.exp(shifted - np.max(shifted))  # numerical stability
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    value = m + np.log(sum_exp) / beta
    return value, weights

def project_polytope(y, target_sum):
    """Project y onto {h: 0≤h≤1, sum(h)=target_sum}."""
    n = len(y)
    # Bisection on lambda
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        projected = np.clip(y - mid, 0.0, 1.0)
        s = projected.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo+hi)/2, 0.0, 1.0)

def compute_grad_smooth(h, beta):
    """Gradient of smooth_max of correlation * dx."""
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full")
    g = corr * dx
    _, weights = smooth_max_gradient(g, beta)
    
    grad = np.zeros(n)
    n_lags = 2 * n - 1
    for k_idx in range(n_lags):
        k = k_idx - (n - 1)
        w = weights[k_idx]
        if w < 1e-14:
            continue
        # Term 1: (1-h[j+k]) for valid j
        if k >= 0:
            j_range = range(0, n - k)
            for j in j_range:
                grad[j] += w * (1.0 - h[j + k])
        else:
            j_range = range(-k, n)
            for j in j_range:
                grad[j] += w * (1.0 - h[j + k])
        # Term 2: -h[j-k] for valid j
        if k >= 0:
            j_range = range(k, n)
            for j in j_range:
                grad[j] -= w * h[j - k]
        else:
            j_range = range(0, n + k)
            for j in j_range:
                grad[j] -= w * h[j - k]
    
    return grad * dx
```

g_k = Σ_m h[m]*(1-h[m+k])*dx

Terms involving h[i] (=t) or h[j] (=c-t):

- If m+k=i (m=i-k): h[i-k]*(1-t)*dx (if i-k valid)
- If m=j: (c-t)*(1-h[j+k])*dx (if j+k valid)
- If m+k=j (m=j-k): h[j-k]*(1-(c-t))*dx = h[j-k]*(t-c+1)*dx (if j-k valid)

```python
    # R_k = g_k(h) - A_k*h[i] - B_k*h[j] + B_k*c
    # = g_k(h) - A_k*h[i] - B_k*(c - h[i]) + B_k*c
    # = g_k(h) - A_k*h[i] + B_k*h[i]
    # = g_k(h) + (B_k - A_k)*h[i]
    # = g_k(h) - S_k*h[i]
    
    R = g - h[i] * S
    
    # Now minimize max_k (R_k + t*S_k) for t in [lo_t, hi_t]
    # Use golden section search
    phi = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
    
    def f1d(t):
        return np.max(R + t * S)
    
    # Golden section
    a, b = lo_t, hi_t
    c_g = b - phi * (b - a)
    d_g = a + phi * (b - a)
    fc = f1d(c_g)
    fd = f1d(d_g)
    for _ in range(60):
        if fc < fd:
            b = d_g
            d_g = c_g
            fd = fc
            c_g = b - phi * (b - a)
            fc = f1d(c_g)
        else:
            a = c_g
            c_g = d_g
            fc = fd
            d_g = a + phi * (b - a)
            fd = f1d(d_g)
    
    t_opt = (a + b) / 2
    h_new = h.copy()
    h_new[i] = t_opt
    h_new[j] = c - t_opt
    return h_new
```

This is a nice exact 1D optimization for each pair. The golden section search converges in about 60 iterations to very high precision, and each evaluation is O(n) (just a dot product and max).

Total cost per pair update: O(n) * 60 = O(60n) ≈ 4200 for n=70. Very fast!

Now, the overall algorithm:
1. Initialize h
2. Repeat until time runs out:
   a. Pick a random pair (i, j)
   b. Optimize the pair exactly (golden section)

3. Periodically do a full SLSQP refinement
4. Try different n values

```python
            s -= (1.0 - h[j + k])
```

```python
    hi = y.max() + 1.0

def opt_pair(h, i, j):

    c = h[i] + h[j]
    lo_t = max(0.0, c - 1.0)
    hi_t = min(1.0, c)

        return h, compute_c5(h)
    dx = 2.0 / n
    omh = 1.0 - h
    g = np.correlate(h, omh, mode="full") * dx
    S = np.zeros(len(g))
    for ki in range(len(g)):
        k = ki - (n - 1)
        s = 0.0
        if 0 <= i + k < n:
            s += 1.0 - h[i + k]
        if 0 <= i - k < n:
            s -= h[i - k]
        if 0 <= j + k < n:
            s -= 1.0 - h[j + k]
        if 0 <= j - k < n:
            s += h[j - k]
        S[ki] = s * dx
    R = g - h[i] * S
    phi = (np.sqrt(5.0) - 1.0) * 0.5
    a, b = lo_t, hi_t
    cg = b - phi * (b - a)
    dg = a + phi * (b - a)
    fc = float(np.max(R + cg * S))
    fd = float(np.max(R + dg * S))
    for _ in range(50):
        if fc < fd:
            b = dg
            dg = cg
            fd = fc
            cg = b - phi * (b - a)
            fc = float(np.max(R + cg * S))
        else:
            a = cg
            cg = dg
            fc = fd
            dg = a + phi * (b - a)
            fd = float(np.max(R + dg * S))
    t_opt = (a + b) * 0.5
    h_new = h.copy()
    h_new[i] = t_opt
    h_new[j] = c - t_opt
    return h_new, compute_c5(h_new)
```

```python
import numpy as np
import time
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_obj(x, beta):
    dx = 2.0 / len(x)
    g = np.correlate(x, 1.0 - x, mode="full") * dx
    m = g.max()
    s = beta * (g - m)
    s -= s.max()
    return m + np.log(np.sum(np.exp(s))) / beta


def smooth_jac(x, beta):
    n = len(x)
    dx = 2.0 / n
    omh = 1.0 - x
    g = np.correlate(x, omh, mode="full") * dx
    m = g.max()
    s = beta * (g - m)
    s -= s.max()
    es = np.exp(s)
    w = es / es.sum()
    grad = np.zeros(n)
    for ki in range(len(g)):
        k = ki - (n - 1)
        wi = w[ki]
        if wi < 1e-14:
            continue
        if k >= 0:
            grad[:n - k] += wi * omh[k:]
            grad[k:] -= wi * x[:n - k]
        else:
            grad[-k:] += wi * omh[:n + k]
            grad[:n + k] -= wi * x[-k:]
    return grad * dx


def con_eq(x):
    return np.sum(x) - len(x) * 0.5


def con_eq_jac(x):
    return np.ones(len(x))

def pair_descent(h, n_iters, seed, bias_lag=False):

    rng = np.random.RandomState(seed)
    best = h.copy()
    best_c5 = compute_c5(best)
    for it in range(n_iters):
        if bias_lag and it % 100 == 0:
            corr = np.correlate(best, 1.0 - best, mode="full")
```
