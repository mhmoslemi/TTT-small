    """Gradient of f_k(h) = sum_j h(j)*(1-h(j+k)) with respect to h"""
    n = len(h)

        # d/dh(i): if i is in [0, n-1-k), contributes (1 - h(i+k))
        #           if i is in [k, n-1], contributes -h(i-k)
        g[:n-k] += (1.0 - h[k:])
        if k > 0:

    else:
        p = -k
        # f_k = sum_{j=p}^{n-1} h(j)*(1-h(j-p))
        # d/dh(i): if i is in [p, n-1], contributes (1 - h(i-p))
        #           if i is in [0, n-1-p), contributes -h(i+p)
        g[p:] += (1.0 - h[:n-p])
        g[:n-p] -= h[p:]
    return g

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')

def grad_f_k(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        g[:n-k] += (1.0 - h[k:])
        if k > 0:
            g[k:] -= h[:n-k]
    else:
        p = -k
        g[p:] += (1.0 - h[:n-p])
        g[:n-p] -= h[p:]
    return g

    corr = np.correlate(h, 1 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    # Clip to avoid overflow
    shifted = np.clip(shifted, -500, 0)
    w = np.exp(shifted)
    s = np.sum(w)
    if s < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= s
    n = len(h)
    grad = np.zeros(n)
    active = np.where(w > 1e-6)[0]

            break

            h[i] -= eps
            h[j] += eps
            new_c5 = compute_c5(h)
            delta = new_c5 - c5
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
                c5 = new_c5
                if c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = c5

                h[j] -= eps

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def grad_f_k(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        g[:n-k] += (1.0 - h[k:])
        if k > 0:
            g[k:] -= h[:n-k]
    else:
        p = -k
        g[p:] += (1.0 - h[:n-p])
        g[:n-p] -= h[p:]
    return g

def smooth_max_grad(h, alpha):
    corr = np.correlate(h, 1 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -500, 0)
    w = np.exp(shifted)
    s = np.sum(w)
    n = len(h)
    if s < 1e-300:
        k_idx = np.argmax(corr)
        return grad_f_k(h, k_idx - (n - 1)) * (2.0 / n)
    w /= s
    grad = np.zeros(n)
    active = np.where(w > 1e-6)[0]
    for k_idx in active:
        grad += w[k_idx] * grad_f_k(h, k_idx - (n - 1))
    return grad * (2.0 / n)

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def grad_f_k(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        g[:n - k] += (1.0 - h[k:])
        if k > 0:
            g[k:] -= h[:n - k]
    else:
        p = -k
        g[p:] += (1.0 - h[:n - p])
        g[:n - p] -= h[p:]
    return g

def smooth_max_grad(h, alpha):
    corr = np.correlate(h, 1 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -500, 0)
    w = np.exp(shifted)
    s = np.sum(w)
    n = len(h)
    if s < 1e-300:
        ki = np.argmax(corr)
        return grad_f_k(h, ki - (n - 1)) * (2.0 / n)
    w /= s
    grad = np.zeros(n)
    active = np.where(w > 1e-6)[0]
    for ki in active:
        grad += w[ki] * grad_f_k(h, ki - (n - 1))
    return grad * (2.0 / n)

def project_sb(h, target_sum):
    lo, hi = -1.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
