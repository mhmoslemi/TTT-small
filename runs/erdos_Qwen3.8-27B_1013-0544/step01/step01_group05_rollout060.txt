from scipy.optimize import minimize
import time

def project_feasible(x, target_sum):
    lo = float(x.min()) - 1.0
    hi = float(x.max()) + 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = float(np.clip(x - mid, 0.0, 1.0).sum())
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_for_lag(h, L):
    n = len(h)
    omh = 1.0 - h
    grad = np.zeros(n)
    if L == 0:
        grad[:] = omh[:] - h[:]
    elif L > 0:
        grad[L:] += omh[:n-L]
        grad[:n-L] -= h[L:]
    else:
        L2 = -L
        grad[:n-L2] += omh[L2:]
        grad[L2:] -= h[:n-L2]
    return grad

def soft_max_grad(h, corr, n_half, T, top_k=8):
    """Soft-max weighted gradient over top lags"""
    n = len(h)
    idx = np.argsort(corr)[-top_k:]
    vals = corr[idx]
    max_v = vals.max()
    w = np.exp((vals - max_v) / T)
    w /= w.sum()
    grad = np.zeros(n)
    for i in range(len(idx)):
        L = int(idx[i]) - n_half
        grad += w[i] * grad_for_lag(h, int(idx[i]) - n_half)
    return grad

def optimize_n(h, dx, target_sum, budget_s, seed):

    n = len(h)
    n_half = n - 1
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)

    h_curr = h.copy()

            best_h = h_new.copy()
        h_curr = h_new

                if c5_ref < best_c5:
                    best_c5 = c5_ref
                    best_h = h_ref
        except Exception:
            pass

import numpy as np

import time

def project_feasible(x, target_sum):
    lo = float(x.min()) - 1.0
    hi = float(x.max()) + 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5

            lo = mid
        else:

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_for_lag(h, L):
    n = len(h)
    omh = 1.0 - h
    g = np.zeros(n)
    if L == 0:
        g[:] = omh[:] - h[:]
    elif L > 0:
        g[L:] += omh[:n-L]
        g[:n-L] -= h[L:]
    else:

        g[:n-L2] += omh[L2:]
        g[L2:] -= h[:n-L2]
    return g

    g = np.zeros(n)

        g += w[i] * grad_for_lag(h, L)
    return g

    step0 = 0.005
    t0 = time.time()
    stag = 0
    
    while time.time() - t0 < budget:
        frac = (time.time() - t0) / max(budget, 1e-6)
        T = max(0.002, T0 * np.exp(-3.0 * frac))
        step = max(0.0002, step0 * np.exp(-2.0 * frac))
        
        corr = compute_corr(h)
        g = soft_max_grad(h, corr, n_half, T, top_k)
        momentum = mu * momentum - step * g
        h_new = project_feasible(h + momentum, ts)
        c5_new = compute_c5(h_new, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            stag = 0
        else:
            stag += 1
            if stag > 5000:
                h = best_h + rng.normal(0, 0.03, n)
                h = project_feasible(h, ts)
                momentum = np.zeros(n)
                stag = 0
        h = h_new
    
    return best_h, best_c5

def _sa_phase(h, dx, ts, budget, seed):
    n = len(h)
    n_half = n - 1
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    h_cur = h.copy()
    cur_c5 = best_c5
    T0_sa = 0.004
    noise0 = 0.015
    t0 = time.time()
    
    while time.time() - t0 < budget:
        frac = (time.time() - t0) / max(budget, 1e-6)
        T_sa = max(0.0001, T0_sa * np.exp(-4.0 * frac))
        noise = max(0.002, noise0 * np.exp(-3.0 * frac))
        
        corr = compute_corr(h_cur)
        idx = np.argsort(corr)[-6:]
        g = np.zeros(n)
        for idx_k in idx:
            L = int(idx_k) - n_half
            g += grad_for_lag(h_cur, L)
        g /= float(len(idx))
        
        prop = h_cur - 0.003 * g + rng.normal(0, noise, n)
        prop = project_feasible(prop, ts)
        prop_c5 = compute_c5(prop, dx)
        delta = prop_c5 - cur_c5
        
        if delta < 0 or rng.random() < np.exp(-max(0.0, delta) / max(T_sa, 1e-10)):
            h_cur = prop
            cur_c5 = prop_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h_cur.copy()
    
    return best_h, best_c5
