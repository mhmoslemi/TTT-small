    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()

    mx = s.max()
    # p_k = exp(s_k - mx) / sum(exp(s_k - mx))
    exp_s = np.exp(s - mx)
    p = exp_s / exp_s.sum()
    
    # Gradient is weighted sum of per-lag gradients
    grad = np.zeros(n)
    for idx in range(2*n-1):
        if p[idx] < 1e-10:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-15:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def obj_lse_val(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    exp_s = np.exp(s - mx)
    return float(m + (mx + np.log(exp_s.sum())) / alpha)

def _lag_grad(h, n, L):
    """Gradient of c[lag=L] w.r.t. h"""
    g = np.zeros(n)
    if L >= 0:
        if n - L > 0:
            g[:n-L] += (1.0 - h)[L:]
            g[L:] += -h[:n-L]
    else:
        La = -L
        if n - La > 0:
            g[La:] += (1.0 - h)[:n-La]
            g[:n-La] += -h[La:]
    return g

def obj_lse_jac(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    exp_s = np.exp(s - mx)
    p = exp_s / exp_s.sum()
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        if p[idx] < 1e-12:
            continue
        L = idx - (n - 1)
        grad += p[idx] * _lag_grad(h, n, L)
    return grad

def constraint_sum(x, n):
    return x.sum() - n / 2.0

def constraint_sum_jac(x, n):
    return np.ones(n)

def subgrad_top_k(h, n, k=5):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    top_idx = np.argsort(c)[-k:]
    grad = np.zeros(n)
    for idx in top_idx:
        L = idx - (n - 1)
        grad += _lag_grad(h, n, L)
    return grad / k

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - t0)

        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        n_init = 80
        h_init = np.full(n_init, 0.5)

    h_init = project_feasible(h_init, n_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h, n_init)
    best_n = n_init

    def slsqp_optimize(h_start, n, alphas, maxiter=3000, ftol=1e-18):
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': constraint_sum, 'jac': constraint_sum_jac, 'args': (n,)}]
        h_curr = h_start.copy()
        for alpha in alphas:
            if remaining() < 35:
                break
            try:
                res = minimize(obj_lse_val, h_curr, args=(n, alpha), jac=obj_lse_jac,
                              method='SLSQP', bounds=bounds, constraints=cons,
                              options={'maxiter': maxiter, 'ftol': ftol})
                h_try = project_feasible(res.x, n)
                c_try = compute_c5(h_try, n)
                if c_try < compute_c5(h_curr, n) - 1e-14:
                    h_curr = h_try
            except Exception:
                pass
        return h_curr

    def subgrad_loop(h_start, n, max_iter=1000, k=5, step0=0.1):
        h_curr = h_start.copy()
        step = step0
        c_curr = compute_c5(h_curr, n)
        no_improve = 0
        for _ in range(max_iter):
            if remaining() < 50:
                break
            grad = subgrad_top_k(h_curr, n, k=k)
            h_new = project_feasible(h_curr - step * grad, n)
            c_new = compute_c5(h_new, n)
            if c_new < c_curr - 1e-12:
                h_curr = h_new
                c_curr = c_new
                no_improve = 0
            else:
                no_improve += 1
                if no_improve % 15 == 0:
                    step *= 0.9
            if step < 1e-7:
                break
        return h_curr

    # Phase 1: SLSQP continuation at initial n

        h_test = slsqp_optimize(h_test, n_test, [5000, 50000], maxiter=2000)

    n = best_n

        h_pert = slsqp_optimize(h_pert, n, [500, 5000, 50000], maxiter=1500)

    # Phase 7: Coordinate descent fine-tuning
    if remaining() > 25:
        h_curr = best_h.copy()
        c_curr = best_c5
        for d in [0.002, 0.001, 0.003, 0.0005]:
            if remaining() < 10:
                break
            for _ in range(4000):
                if remaining() < 5:
                    break
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:

                for s in [1, -1]:
                    di = s * d
                    hi = h_curr[i] + di
                    hj = h_curr[j] - di
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:

                        h_try[i] = hi

                        c = compute_c5(h_try, n)

        if c_curr < best_c5:

    # Phase 8: Final high-precision SLSQP

        c_curr = compute_c5(h_curr, best_n)
        if c_curr < best_c5:
            best_h, best_c5 = h_curr.copy(), c_curr

    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
