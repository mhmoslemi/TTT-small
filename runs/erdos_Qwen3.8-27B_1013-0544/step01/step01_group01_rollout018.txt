def compute_gradient(h, beta):
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    m = np.max(corr)
    # softmax weights
    exp_vals = np.exp(beta * (corr - m))
    w = exp_vals / np.sum(exp_vals)
    
    grad = np.zeros(n)
    for j in range(n):
        s = 0.0
        for i in range(2*n - 1):
            lag = i - (n - 1)
            # First term: (1-h[j-lag]) if 0 <= j-lag < n
            idx1 = j - lag
            if 0 <= idx1 < n:
                s += w[i] * (1.0 - h[idx1])
            # Second term: -h[j+lag] if 0 <= j+lag < n
            idx2 = j + lag
            if 0 <= idx2 < n:
                s -= w[i] * h[idx2]
        grad[j] = dx * s
    return grad

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def compute_gradient(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exp_vals = np.exp(beta * (corr - m))
    w = exp_vals / np.sum(exp_vals)
    
    grad = np.zeros(n)
    for j in range(n):
        s = 0.0
        for i in range(2 * n - 1):
            lag = i - (n - 1)
            idx1 = j - lag
            if 0 <= idx1 < n:
                s += w[i] * (1.0 - h[idx1])
            idx2 = j + lag
            if 0 <= idx2 < n:
                s -= w[i] * h[idx2]

        else:
            hi = mid

    n = len(h)
    t0 = time.time()

            f_old = f_new
            break

        f_old = f_new

def generate_perturbation(h, n_points, strategy, attempt, rng):
    """Generate a perturbed version of h using the given strategy"""
    h0 = h.copy()
    
    if strategy == 0:
        # Gaussian perturbation
        scale = 0.02 + 0.08 * np.exp(-attempt / 100.0)
        h0 += rng.standard_normal(n_points) * scale
    elif strategy == 1:
        # Target high-correlation lags
        corr = compute_correlation(h0)
        top_idx = np.argsort(corr)[-min(3, n_points // 3):]
        for ti in top_idx:
            lag = ti - (n_points - 1)
            if lag == 0:
                continue
            for j in range(n_points):
                a = j - lag
                if 0 <= a < n_points:
                    # Push h[j] away from h[a] to reduce correlation at this lag
                    if h0[j] > h0[a]:
                        h0[j] += 0.03 * rng.random()
                    else:
                        h0[j] -= 0.03 * rng.random()
    elif strategy == 2:
        # Block shuffle
        blen = max(2, n_points // 10)
        bs = rng.integers(0, n_points - blen + 1)
        block = h0[bs:bs+blen].copy()
        rng.shuffle(block)
        h0[bs:bs+blen] = block
    elif strategy == 3:
        # Anti-periodic push
        p = rng.integers(3, max(4, n_points // 3))
        for j in range(p, n_points):
            diff = 1.0 - h0[j-p] - h0[j]
            h0[j] += 0.03 * diff * rng.standard_normal()
    elif strategy == 4:
        # Binary-like perturbation (push toward 0 or 1)
        mask = rng.random(n_points) < 0.3
        h0[mask] = np.where(h0[mask] > 0.5, 
                           np.minimum(1.0, h0[mask] + 0.1),
                           np.maximum(0.0, h0[mask] - 0.1))
    elif strategy == 5:
        # Gradient ascent + noise (anti-gradient to escape)
        grad = analytical_gradient(h0, 100.0)
        h0 = project_feasible(h0 + 0.5 * grad, target_sum=h0.sum())  # keep same sum
        h0 = h0 + rng.standard_normal(n_points) * 0.03
    elif strategy == 6:
        # Pair swap
        for _ in range(rng.integers(2, 6)):
            i, j = rng.choice(n_points, 2, replace=False)
            h0[i], h0[j] = h0[j] + rng.standard_normal()*0.01, h0[i] + rng.standard_normal()*0.01
    else:
        # Sinusoidal perturbation
        freq = rng.integers(2, max(3, n_points // 5))
        phase = rng.random() * 2 * np.pi
        h0 += 0.05 * np.sin(2 * np.pi * freq * np.arange(n_points) / n_points + phase)
    
    return h0
