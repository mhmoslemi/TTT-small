import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    if m == -np.inf:
        return 0.0
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) > 1e-10 and s > 0:
        h += (target_sum - s) / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def make_obj(dx, beta):
    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        return smooth_max(corr, beta) * dx
    return obj

def coord_descent(h, target_sum, dx, beta, n_iter=200):
    """Cyclic coordinate descent with penalty for sum constraint."""
    n = len(h)
    h = h.copy()
    lam = 0.0  # Lagrange multiplier for sum constraint
    for it in range(n_iter):
        for i in range(n):
            # Compute gradient w.r.t. h[i]
            # h[i] appears in various correlation terms
            # For simplicity, use numerical gradient
            eps = 1e-7
            h[i] = min(max(h[i] - eps, 0), 1)
            f_minus = smooth_max(np.correlate(h, 1.0 - h, mode="full"), beta) * dx
            h[i] = min(max(h[i] + 2*eps, 0), 1)
            f_plus = smooth_max(np.correlate(h, 1.0 - h, mode="full"), beta) * dx
            h[i] = min(max(h[i] - eps, 0), 1)
            grad = (f_plus - f_minus) / (2*eps)
            # Also add Lagrange term
            grad += lam
            # Update
            h[i] -= 0.1 * grad
            h[i] = min(max(h[i], 0.0), 1.0)
        # Update lambda to enforce sum constraint
        lam += 0.01 * (h.sum() - target_sum)
    return project_h(h, target_sum)

    s = h.sum()
    if abs(s - target_sum) > 1e-10:
        h += (target_sum - s) / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exps = np.clip(beta * (corr - m), -50, 50)
    w = np.exp(exps)
    w_sum = w.sum()
    w = w / w_sum
    val = m + np.log(w_sum) / beta
    lags = np.arange(-(n-1), n)
    i_idx = np.arange(n)
    pos1 = i_idx[:, None] + lags[None, :]
    pos2 = i_idx[:, None] - lags[None, :]
    mask1 = (pos1 >= 0) & (pos1 < n)
    mask2 = (pos2 >= 0) & (pos2 < n)
    safe1 = np.clip(pos1, 0, n-1)
    safe2 = np.clip(pos2, 0, n-1)
    term1 = mask1.astype(float) * (1.0 - h[safe1]) * w[None, :]
    term2 = mask2.astype(float) * h[safe2] * w[None, :]
    grad = term1.sum(axis=1) - term2.sum(axis=1)
    return val, grad

def _obj(x):
    n = _obj_state['n']
    beta = _obj_state['beta']
    dx = _obj_state['dx']
    corr = np.correlate(x, 1.0 - x, mode="full")

    exps = np.clip(beta * (corr - m), -50, 50)

    n = _obj_state['n']
    beta = _obj_state['beta']
    dx = _obj_state['dx']
    val, grad = smooth_max_and_grad(x, beta)
    return val * dx, grad * dx
