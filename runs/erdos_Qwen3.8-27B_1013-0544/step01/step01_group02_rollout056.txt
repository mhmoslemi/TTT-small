```python
def corr_grad(h, n, weights, dx):
    """Compute gradient of sum_m weights[m]*c[m]*dx with respect to h.
    weights has length 2n-1."""
    grad = np.zeros(n)
    # c[m] = sum_j h[j]*(1-h[j+d_m]) where d_m = m-(n-1)
    # dc[m]/dh[i] = (1-h[i+d_m])_valid - h[i-d_m]_valid
    # grad[i] = dx * sum_m weights[m] * ((1-h[i+d_m])_valid - h[i-d_m]_valid)
    
    for m in range(2*n - 1):
        d = m - (n - 1)
        w = weights[m] * dx
        if w == 0:
            continue
        # (1-h[i+d]) for valid i: i ranges from max(0,-d) to min(n-1, n-1-d)
        # h[i-d] for valid i: i ranges from max(0,d) to min(n-1, n-1+d)
        # Contribution to grad[i] from (1-h[i+d]):
        lo1 = max(0, -d)
        hi1 = min(n, n - d)  # exclusive upper bound
        if lo1 < hi1:
            grad[lo1:hi1] += w * (1.0 - h[lo1+d:hi1+d])
        # Contribution to grad[i] from -h[i-d]:
        lo2 = max(0, d)
        hi2 = min(n, n + d)  # exclusive upper bound
        if lo2 < hi2:
            grad[lo2:hi2] -= w * h[lo2-d:hi2-d]
    
    return grad
```

```python
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)

            if total < 1e-14:

            h += (diff / total) * h

    mx = shifted.max()

    grad = np.zeros(n)

        hi2 = min(n, n + d)
        if lo2 < hi2:
            grad[lo2:hi2] -= wm * h[lo2 - d:hi2 - d]
    return grad * dx
```

```python
    h = x[:n]
    n_c = 2*n - 1
    jac = np.zeros((n_c + 1, n + 1))  # +1 for the sum constraint
    # For the sum constraint: d(sum(h))/dh[i] = 1
    # For the lag constraints: d(c[k])/dh[i] = (1-h[i+d])_valid - h[i-d]_valid, and d(c[k])/dt = -1
    for k in range(n_c):
        d = k - (n-1)
        # d(c[k])/dh[i] for valid i
        lo1 = max(0, -d)
        hi1 = min(n, n-d)
        if lo1 < hi1:
            jac[k, lo1:hi1] += (1.0 - h[lo1+d:hi1+d]) * (2.0/n)
        lo2 = max(0, d)
        hi2 = min(n, n+d)
        if lo2 < hi2:
            jac[k, lo2:hi2] -= h[lo2-d:hi2-d] * (2.0/n)
        jac[k, n] = -1.0  # d/dt
    # Sum constraint (index n_c)
    jac[n_c, :n] = 1.0
    return jac
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tr = room.sum()
            if tr < 1e-14:
                break
            h += (diff / tr) * room
        else:
            t = h.sum()
            if t < 1e-14:
                break
            h += (diff / t) * h
    return np.clip(h, 0.0, 1.0)

def lse_obj(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 700:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def lse_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    exp_s = np.exp(shifted - mx)
    w = exp_s / exp_s.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        d = k - (n - 1)
        wk = w[k]
        if wk < 1e-16:
            continue
        lo1 = max(0, -d)
        hi1 = min(n, n - d)
        if lo1 < hi1:
            grad[lo1:hi1] += wk * (1.0 - h[lo1 + d:hi1 + d])
        lo2 = max(0, d)
        hi2 = min(n, n + d)
        if lo2 < hi2:
            grad[lo2:hi2] -= wk * h[lo2 - d:hi2 - d]
    return grad * dx

def sum_con(h, n):
    return np.sum(h) - n / 2.0

def sum_con_jac(h, n):
    return np.ones(n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 80

    best_h = None
    best_c5 = float('inf')
    best_n = n_init

    n_list = [n_init]
    if n_init <= 80:
        n_list = [80, 160]
    elif n_init <= 120:
        n_list = [n_init, 200]
    else:
        n_list = [n_init]

    for n in n_list:
        if budget_s - (time.time() - t0) < 50:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]

        inits = []
        if n == n_init and h_init is not None:
            inits.append(project_feasible(h_init, n))
        elif h_init is not None:
            x_old = np.linspace(0, 2, n_init, endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            hi = np.interp(x_new, x_old, h_init, period=2.0)
            inits.append(project_feasible(hi, n))

        inits.append(project_feasible(rng.uniform(0, 1, n), n))

        h_blk = np.zeros(n)
        blk = max(2, n // 14)
        for b in range(0, n, 2 * blk):
            h_blk[b:b + blk] = 1.0
        inits.append(project_feasible(h_blk, n))

        for f in [2, 3, 5]:
            hs = 0.5 + 0.4 * np.sin(2 * np.pi * f * np.arange(n) / n)
            inits.append(project_feasible(hs, n))

        for h_start in inits:
            if budget_s - (time.time() - t0) < 25:
                break
            c = compute_c5(h_start, n)
            if c < best_c5:

            h_cur = h_start.copy()
            alphas = [10, 30, 100, 300, 1000, 3000]
            for alpha in alphas:

                    res = minimize(lse_obj, h_cur, args=(n, alpha), jac=lse_grad,
                                   method='SLSQP', bounds=bounds, constraints=cons,

                    ht = project_feasible(res.x, n)

                    h_cur = ht
                except Exception:
                    pass

            if budget_s - (time.time() - t0) < 15:

            # Binary push
            hb = np.where(h_cur > 0.5, 1.0, 0.0)
            hb = project_feasible(hb, n)
            try:

                               method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 1500, 'ftol': 1e-15})
                ht = project_feasible(res.x, n)
                c = compute_c5(ht, n)
                if c < best_c5:

                h_cur = ht
            except Exception:
                pass

        # Perturbation loop for this n
        if budget_s - (time.time() - t0) > 40:
            hp = best_h.copy() if best_n == n else inits[0].copy()
            for _ in range(80):
                if budget_s - (time.time() - t0) < 20:

                hp2 = project_feasible(hp2, n)

                    res = minimize(lse_obj, hp2, args=(n, 500), jac=lse_grad,
                                   method='SLSQP', bounds=bounds, constraints=cons,
                                   options={'maxiter': 1000, 'ftol': 1e-14})
                    ht = project_feasible(res.x, n)
                    c = compute_c5(ht, n)
                    if c < best_c5:
                        best_c5, best_h, best_n = c, ht.copy(), n
                        hp = ht.copy()
                except Exception:
                    pass

    # Final polish
    if best_h is not None and budget_s - (time.time() - t0) > 8:
        n = best_n
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        for alpha in [3000, 10000, 30000]:

                break

                res = minimize(lse_obj, best_h, args=(n, alpha), jac=lse_grad,
                               method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 3000, 'ftol': 1e-17})
                ht = project_feasible(res.x, n)
                c = compute_c5(ht, n)
                if c < best_c5:
                    best_c5, best_h = c, ht.copy()
            except Exception:
                pass

    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return best_h, best_c5, best_n
```
