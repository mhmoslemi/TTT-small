    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def corr_full(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def project_feasible(h, n):
    t = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        d = t - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            r = 1.0 - h
            s = r.sum()
            if s < 1e-15: break
            h += (d / s) * r
        else:
            s = h.sum()
            if s < 1e-15: break
            h += (d / s) * h
    return np.clip(h, 0.0, 1.0)

def subgrad(h, n, tol=1.0):
    """Subgradient of max_k c[k] where c = correlate(h,1-h,'full')*dx."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = c.max()
    mask = c >= cmax - tol
    nb = max(1, mask.sum())
    g = np.zeros(n)
    for k in np.where(mask)[0]:
        s = int(k) - n + 1
        a = max(0, -s)
        b = min(n, n - s)
        if a < b:
            g[a:b] += 1.0 - h[a+s:b+s]
        a = max(0, s)
        b = min(n, n + s)
        if a < b:
            g[a:b] -= h[a-s:b-s]
    return (g * dx) / nb

def smooth_obj(x, n, alpha):
    c = corr_full(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 400:
        return m + 3.0/alpha
    return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)

def s_con(h, n):
    return h.sum() - n / 2.0

def s_jac(h, n):
    return np.ones(n)

def slsqp(h, n, alpha, iters=3000):
    try:
        r = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                    bounds=[(0.0,1.0)]*n,
                    constraints=[{'type':'eq','fun':s_con,'jac':s_jac,'args':(n,)}],
                    options={'maxiter':iters, 'ftol':1e-16})
        return project_feasible(r.x, n)
    except:
        return h

    h[rng.choice(n, n//2, replace=False)] = 1.0
    return h

    for i in range(n):
        if i % period < duty:
            h[i] = 1.0
    return project_feasible(h, n)
