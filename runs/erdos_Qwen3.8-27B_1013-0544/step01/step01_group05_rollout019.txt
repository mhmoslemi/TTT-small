    return float(np.max(np.correlate(h, 1.0-h, mode='full'))) * dx

def smooth_max_grad(h, dx, beta):
    """Returns (smooth_max_value, gradient) of smooth max of all f_k."""
    n = len(h)
    n_half = n - 1
    f = np.correlate(h, 1.0 - h, mode='full') * dx
    f_max = np.max(f)
    exp_bf = np.exp(beta * (f - f_max))
    w = exp_bf / exp_bf.sum()
    obj = f_max + np.log(np.sum(exp_bf)) / beta
    
    # Gradient: weighted sum of individual gradients
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-12:
            continue
        L = idx - n_half
        if L == 0:
            grad += w[idx] * dx * (1.0 - 2.0 * h)
        elif L > 0:
            grad[L:] += w[idx] * dx * (1.0 - h[:n-L])
            grad[:n-L] -= w[idx] * dx * h[L:]
        else:
            L2 = -L
            grad[:n-L2] += w[idx] * dx * (1.0 - h[L2:])
            grad[L2:] -= w[idx] * dx * h[:n-L2]
    return obj, grad

    """Coordinate descent on exact max."""
    n = len(h)
    h = h.copy()
    c5 = c5_of(h, dx)
    for it in range(max_iters):
        improved = False
        for i in range(n):
            # Try to optimize h[i] while keeping others fixed
            # After changing h[i], project to feasible set
            for delta in [0.01, -0.01, 0.005, -0.005, 0.05, -0.05]:
                h_test = h.copy()
                h_test[i] += delta
                h_test = project_feasible(h_test, target_sum)
                c5_test = c5_of(h_test, dx)
                if c5_test < c5 - tol:
                    h = h_test
                    c5 = c5_test
                    improved = True
                    break
        if not improved:
            break
    return h, c5
