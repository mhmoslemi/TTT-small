    if best_h is not None and time.time() < start_time + budget_s - 5:
        for _ in range(50):
            if time.time() > start_time + budget_s - 2:
                break
            h_pert = best_h + rng.normal(0, 0.003, n_points)
            h_pert = project_feasible(h_pert, target_sum)
            c5_pert = compute_c5(h_pert, dx)
            if c5_pert < best_c5:
                h_r, c5_r = active_set_refine(h_pert, dx, target_sum, 
                                              start_time + budget_s - 1, max_iter=2000)
                if c5_r < best_c5:
                    best_c5 = c5_r
                    best_h = h_r.copy()
    
    return (best_h, best_c5, best_n)
