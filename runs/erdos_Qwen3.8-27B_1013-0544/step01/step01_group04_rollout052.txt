import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def smooth_obj(h, dx, eps):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip((corr - m) / eps, -50.0, 50.0)
    return m + eps * np.log(np.sum(np.exp(shifted)))

def project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = np.sum(h) - target
        if abs(d) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            h[free] -= d / np.count_nonzero(free)
        else:
            h -= d / n
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

def slsqp_smooth(h0, n, dx, target, eps, maxiter=300):
    res = minimize(
        lambda x: smooth_obj(x, dx, eps),
        h0, method='SLSQP',
        bounds=[(0.0, 1.0)] * n,
        constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target}],
        options={'maxiter': maxiter, 'ftol': 1e-12}
    )
    return project(res.x, n)

def slsqp_exact(h0, n, dx, target, maxiter=500):
    res = minimize(
        lambda x: compute_c5(x, dx),
        h0, method='SLSQP',
        bounds=[(0.0, 1.0)] * n,
        constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target}],
        options={'maxiter': maxiter, 'ftol': 1e-14}
    )
    return project(res.x, n)

def binary_local_search(h, n, dx, max_rounds=5000):
    """Steepest descent: try all swaps, pick best."""
    hb = h.copy()
    c5 = compute_c5(hb, dx)
    target = n // 2
    ones = np.where(hb == 1)[0]
    zeros = np.where(hb == 0)[0]
    
    for round_ in range(max_rounds):
        best_delta = 1e-15
        best_swap = None
        for i in ones:
            for j in zeros:
                hb[i] = 0
                hb[j] = 1
                c5_new = compute_c5(hb, dx)
                delta = c5_new - c5
                if delta < -best_delta:
                    best_delta = -delta
                    best_swap = (i, j)
                hb[i] = 1
                hb[j] = 0
        if best_swap is None:
            break
        i, j = best_swap
        hb[i] = 0
        hb[j] = 1
        c5 += best_delta * (-1)  # c5 decreased by best_delta
        c5 = compute_c5(hb, dx)  # Recompute to avoid drift
        ones = np.where(hb == 1)[0]
        zeros = np.where(hb == 0)[0]
    
    return hb, c5

        # Steepest descent on swaps (limited rounds for time)
