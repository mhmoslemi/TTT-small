import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_c5_smooth(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    vals = beta * corr
    v_max = np.max(vals)
    if v_max > 400:
        return float(np.exp(v_max - 400) * (np.sum(np.exp(vals - v_max)) ** (1.0 / beta)) * beta / beta)
    shifted = vals - v_max
    log_sum = v_max + np.log(np.sum(np.exp(shifted)))
    return log_sum / beta

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    for _ in range(300):
        np.clip(h, 0.0, 1.0, out=h)
        diff = np.sum(h) - target
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(free):
            h[free] -= diff / np.sum(free)
        else:
            h -= diff / n
    np.clip(h, 0.0, 1.0, out=h)
    diff = np.sum(h) - target
    if abs(diff) > 1e-15:
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.sum(free) > 0:
            h[free] -= diff / np.sum(free)
            np.clip(h, 0.0, 1.0, out=h)
    return h

def smooth_slsqp(h_start, n, dx, target_sum, beta, max_iter=300):
    bounds = [(0.0, 1.0)] * n
    ones = np.ones(n)
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum, 'jac': lambda x: ones}
    def obj(x):
        v = compute_c5_smooth(x, dx, beta)
        return v if np.isfinite(v) else 1.0
    res = minimize(obj, h_start, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-14})
    return project_feasible(res.x, n)

def binary_round(h, n):
    h = h.copy()
    h[h < 0.01] = 0.0
    h[h > 0.99] = 1.0
    return project_feasible(h, n)
