from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-14:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def pair_swap_optimize(h, i, j, n):
    """Find optimal delta to swap between h[i] and h[j]."""
    dx = 2.0 / n
    g = 1.0 - h
    # For shift m (0 to 2n-2), the correlation value is:
    # out[m] = sum_k h[k] * g[k - m + n - 1]
    # If we change h[i] += delta and h[j] -= delta:
    #   g[i] -= delta, g[j] += delta
    # Change in out[m]:
    #   From h[i] term: delta * g[i - m + n - 1]
    #   From g[i] term: -delta * h[i + m - n + 1]  (the h value at position i+m-n+1)
    #   From h[j] term: -delta * g[j - m + n - 1]
    #   From g[j] term: +delta * h[j + m - n + 1]
    # So coefficient a[m] = g[i-m+n-1] - h[i+m-n+1] - g[j-m+n-1] + h[j+m-n+1]
    # (with appropriate bounds checking)
    
    m = np.arange(2 * n - 1)
    # Index calculations with bounds
    idx_h_i = i + m - (n - 1)  # position in h for the h[i+m-n+1] term
    idx_g_i = i - m + (n - 1)  # position in g for the g[i-m+n-1] term
    idx_h_j = j + m - (n - 1)
    idx_g_j = j - m + (n - 1)
    
    a = np.zeros(2 * n - 1)
    valid_gi = (idx_g_i >= 0) & (idx_g_i < n)
    valid_hi = (idx_h_i >= 0) & (idx_h_i < n)
    valid_gj = (idx_g_j >= 0) & (idx_g_j < n)
    valid_hj = (idx_h_j >= 0) & (idx_h_j < n)
    
    a[valid_gi] += g[idx_g_i[valid_gi]]
    a[valid_hi] -= h[idx_h_i[valid_hi]]
    a[valid_gj] -= g[idx_g_j[valid_gj]]
    a[valid_hj] += h[idx_h_j[valid_hj]]
    
    # Current correlation values
    out = np.correlate(h, g, mode='full')
    
    # Bounds on delta
    delta_min = max(-h[i], h[j] - 1.0)  # h[i]+delta >= 0, h[j]-delta <= 1
    delta_max = min(1.0 - h[i], h[j])      # h[i]+delta <= 1, h[j]-delta >= 0
    
    if delta_min >= delta_max:
        return h
    
    # Binary search on optimal t
    # For given t, feasible if: max(delta_min, max_{a<0}(t-out)/a) <= min(delta_max, min_{a>0}(t-out)/a)
    # and out[m] <= t for a[m] == 0
    
    # Simple approach: ternary search on delta (since the function is convex)
    # Or: find the minimum of max_m (out[m] + a[m]*delta) over delta in [delta_min, delta_max]

    # The minimum is at a breakpoint or boundary
    # Breakpoints are where out[m] + a[m]*delta = out[m'] + a[m']*delta
    # i.e., delta = (out[m'] - out[m]) / (a[m] - a[m'])
    
    # Efficient approach: ternary search (works for convex functions)
    lo, hi = delta_min, delta_max
    for _ in range(80):
        mid1 = lo + (hi - lo) / 3
        mid2 = hi - (hi - lo) / 3
        f1 = np.max(out + a * mid1)
        f2 = np.max(out + a * mid2)
        if f1 < f2:
            hi = mid2
        else:
            lo = mid1
    
    delta_opt = (lo + hi) / 2
    h_new = h.copy()
    h_new[i] += delta_opt
    h_new[j] -= delta_opt
    return h_new

    a = np.zeros(2 * n - 1)
    v = (idx_gi >= 0) & (idx_gi < n)
    a[v] += g[idx_gi[v]]
    v = (idx_hi >= 0) & (idx_hi < n)
    a[v] -= h[idx_hi[v]]
    v = (idx_gj >= 0) & (idx_gj < n)
    a[v] -= g[idx_gj[v]]
    v = (idx_hj >= 0) & (idx_hj < n)
    a[v] += h[idx_hj[v]]
    out = np.correlate(h, g, mode='full')
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if hi - lo < 1e-15:
        return h
    for _ in range(iters):
        mid1 = lo + (hi - lo) / 3.0
        mid2 = hi - (hi - lo) / 3.0
        if np.max(out + a * mid1) < np.max(out + a * mid2):
            hi = mid2
        else:
            lo = mid1
    d = (lo + hi) / 2.0
    h_new = h.copy()
    h_new[i] = np.clip(h[i] + d, 0.0, 1.0)
    h_new[j] = np.clip(h[j] - d, 0.0, 1.0)
    return h_new

def obj_smooth(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 700:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def sum_con(h, n):
    return np.sum(h) - n / 2.0

def sum_con_jac(h, n):
    return np.ones(n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 80
        h0 = np.full(n0, 0.5)

    best_h = None
    best_c5 = 1e9

    def optimize_from(h_start, n, max_time):
        nonlocal best_h, best_c5
        h_start = project_feasible(h_start, n)
        c_cur = compute_c5(h_start, n)
        h_cur = h_start.copy()
        if c_cur >= best_c5:
            return h_cur, c_cur

        # Phase 1: Random pair swaps
        t1_end = min(time.time() + 0.25 * max_time, t0 + budget_s - 200)
        while time.time() < t1_end:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            h_try = optimize_pair(h_cur, i, j, n)
            c = compute_c5(h_try, n)
            if c < c_cur - 1e-15:
                h_cur = h_try
                c_cur = c

        # Phase 2: Systematic sweeps
        t2_end = min(time.time() + 0.55 * max_time, t0 + budget_s - 100)
        no_improve = 0
        while time.time() < t2_end:
            pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
            rng.shuffle(pairs)
            improved = 0
            for i, j in pairs:
                h_try = optimize_pair(h_cur, i, j, n)
                c = compute_c5(h_try, n)
                if c < c_cur - 1e-15:
                    h_cur = h_try
                    c_cur = c
                    improved += 1
            if improved == 0:
                no_improve += 1
                if no_improve >= 4:
                    break
            else:
                no_improve = 0

        # Phase 3: SLSQP polish
        if budget_s - (time.time() - t0) > 30:
            bounds = [(0.0, 1.0)] * n
            cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
            for alpha in [2000, 5000, 20000]:
                if budget_s - (time.time() - t0) < 15:
                    break
                try:
                    res = minimize(obj_smooth, h_cur, args=(n, alpha), method='SLSQP',
                                  bounds=bounds, constraints=cons,
                                  options={'maxiter': 3000, 'ftol': 1e-18})
                    h_try = project_feasible(res.x, n)
                    c = compute_c5(h_try, n)
                    if c < c_cur:
                        h_cur = h_try
                        c_cur = c
                except Exception:
                    pass

        # Phase 4: Final pair sweeps with higher precision
        if budget_s - (time.time() - t0) > 10:
            pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
            rng.shuffle(pairs)
            for i, j in pairs:
                if budget_s - (time.time() - t0) < 5:
                    break
                h_try = optimize_pair(h_cur, i, j, n, iters=80)
                c = compute_c5(h_try, n)
                if c < c_cur - 1e-15:
                    h_cur = h_try
                    c_cur = c

        h_cur = project_feasible(h_cur, n)
        c_cur = compute_c5(h_cur, n)
        if c_cur < best_c5:
            best_c5 = c_cur
            best_h = h_cur.copy()
        return h_cur, c_cur

    # Primary: optimize from initial_h_values
    max_time1 = 0.75 * budget_s
    optimize_from(h0, n0, max_time1)

    # Secondary: try n=100 with a structured start if time remains
    if budget_s - (time.time() - t0) > 120:
        n2 = 100
        h2 = np.full(n2, 0.5)
        # Add some structure: slightly vary to break symmetry
        x = np.linspace(0, 2, n2)
        h2 = 0.5 + 0.1 * np.sin(2 * np.pi * x)
        h2 = project_feasible(h2, n2)
        max_time2 = budget_s - (time.time() - t0) - 20
        if max_time2 > 60:
            optimize_from(h2, n2, max_time2)

    # Tertiary: try perturbation of best if time remains
    if budget_s - (time.time() - t0) > 60 and best_h is not None:
        n_b = len(best_h)
        h3 = best_h + rng.normal(0, 0.05, size=n_b)
        h3 = project_feasible(h3, n_b)
        max_time3 = budget_s - (time.time() - t0) - 10
        if max_time3 > 30:
            optimize_from(h3, n_b, max_time3)

    if best_h is None:
        best_h = project_feasible(h0, n0)
        best_c5 = compute_c5(best_h, n0)

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h, len(best_h))
    return best_h, best_c5, len(best_h)
