    n = len(h)
    omh = 1.0 - h
    grad = np.zeros(n)

def project_feasible(x, target_sum):
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(x - mid, 0.0, 1.0).sum())
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def weighted_lse_grad(h, w, lags, dx):
    n = len(h)
    omh = 1.0 - h
    grad = np.zeros(n)
    nk = len(lags)
    for k in range(nk):
        wk = w[k]
        if wk < 1e-8:
            continue
        L = lags[k]
        if L == 0:
            grad += wk * (omh - h)
        elif L > 0:
            grad[L:] += wk * omh[:n - L]
            grad[:n - L] -= wk * h[L:]
        else:
            L2 = -L
            grad[:n - L2] += wk * omh[L2:]
            grad[L2:] -= wk * h[:n - L2]
    return grad * dx

    n = len(h)

            L2 = -L
            grad[:n - L2] += omh[L2:]
            grad[L2:] -= h[:n - L2]
    return grad * dx / float(len(idx))
