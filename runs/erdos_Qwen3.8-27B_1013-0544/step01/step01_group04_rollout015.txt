```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - target
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            h[free] -= diff / np.sum(free)
        else:
            h -= diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = np.sum(h) - target
    if abs(diff) > 1e-14:
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            h[free] -= diff / np.sum(free)
            h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_optimize(h_start, n, dx, target_sum, max_iter=300):
    bounds = [(0.0, 1.0)] * n
    constraint = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    def objective(x):
        return compute_c5(x, dx)
    
    res = minimize(
        objective, h_start,
        method='SLSQP',
        bounds=bounds,
        constraints=constraint,
        options={'maxiter': max_iter, 'ftol': 1e-12}
    )
    return project_feasible(res.x, n)
```

The number of alignments at shift m is the autocorrelation of the indicator set.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - target
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            h[free] -= diff / np.count_nonzero(free)
        else:
            h -= diff / n
    return np.clip(h, 0.0, 1.0)

def slsqp_optimize(h_start, n, dx, max_iter=400):
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    def obj(x):
        return compute_c5(x, dx)
    res = minimize(obj, h_start, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-12})
    return project_feasible(res.x, n)

def make_binary(h, n):
    target = n // 2
    h_bin = (h >= 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones > target:
        ones_idx = np.where(h_bin == 1)[0]
        order = h[ones_idx].argsort()
        for idx in order[:n_ones - target]:
            h_bin[ones_idx[idx]] = 0.0
    elif n_ones < target:
        zeros_idx = np.where(h_bin == 0)[0]
        order = h[zeros_idx].argsort()[::-1]
        for idx in order[:target - n_ones]:
            h_bin[zeros_idx[idx]] = 1.0
    return h_bin

def binary_sa(h_init, n, dx, n_iter, T_init, T_final, rng):

                best_h = h.copy()
        else:

            h[j] = 0.0
    return best_h, best_c5
```
