    h_new = 1.0 / (1.0 + np.exp(-10.0 * (res.x - 0.5)))
    h_new = project_h(h_new, n * 0.5)
    return h_new, compute_c5(h_new)


def worst_lag_weights(h, top_k=8, decay=1.0):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    sorted_idx = np.argsort(corr)[::-1]
    weights = np.ones(len(corr))
    for i in range(min(top_k, len(corr))):
        weights[sorted_idx[i]] = 1.0 + decay
    return weights


def coarsen_and_refine(h_best, n_coarse, budget_check, budget_s):
    t0 = time.time()
    h_c = interp_h(h_best, n_coarse)
    c5_c = compute_c5(h_c)
    h_c, c5_c = optimize_t_var(h_c, max_iter=300, ftol=1e-11)
    if time.time() - t0 > budget_s:
        return None
    for p in [16, 32, 64]:
        if time.time() - t0 > budget_s:
            break
        w1 = np.ones(len(np.correlate(h_c, 1.0 - h_c, mode='full')))
        h_c2, c5_c2 = optimize_pnorm(h_c, p, w1, max_iter=200, ftol=1e-13)
        if c5_c2 < c5_c:
            h_c, c5_c = h_c2, c5_c2
    h_hi = interp_h(h_c, len(h_best))
    h_hi, c5_hi = optimize_t_var(h_hi, max_iter=300, ftol=1e-11)
    return h_hi, c5_hi


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - t_start)

    h = np.array(initial_h_values, dtype=float)
    n = len(h)
    h = project_h(h, n * 0.5)
    c5 = compute_c5(h)
    h_best = h.copy()
    ones_w = np.ones(2 * n - 1)

    # Phase 1: Initial t-var polish
    if time_left() > 300:
        h_new, c5_new = optimize_t_var(h, max_iter=400, ftol=1e-11)
        if c5_new < c5:
            h, c5 = h_new, c5_new
            h_best = h.copy()

    # Phase 2: LSE with decreasing temp + worst-lag weights
    for temp in [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
        if time_left() < 120:
            break
        h_new, c5_new = optimize_smooth(h, temp, ones_w, max_iter=400, ftol=1e-13)
        if c5_new < c5:
            h, c5 = h_new, c5_new
            h_best = h.copy()
        if time_left() < 80:
            break
        w = worst_lag_weights(h, top_k=10, decay=1.5)
        h_new2, c5_new2 = optimize_smooth(h, temp, w, max_iter=300, ftol=1e-13)
        if c5_new2 < c5:
            h, c5 = h_new2, c5_new2
            h_best = h.copy()

    # Phase 3: p-norm optimization (different smooth max)
    for p in [16, 32, 64, 128]:
        if time_left() < 100:
            break
        h_new, c5_new = optimize_pnorm(h_best, p, ones_w, max_iter=300, ftol=1e-13)
        if c5_new < c5:
            h, c5 = h_new, c5_new
            h_best = h.copy()
            w = worst_lag_weights(h, top_k=8, decay=1.0)
            h_new2, c5_new2 = optimize_pnorm(h, p, w, max_iter=200, ftol=1e-13)
            if c5_new2 < c5:
                h, c5 = h_new2, c5_new2
                h_best = h.copy()

    # Phase 4: Sigmoid/L-BFGS-B (different geometry)
    if time_left() > 180:
        for temp in [0.02, 0.005, 0.002]:
            if time_left() < 80:
                break
            h_new, c5_new = optimize_sigmoid(h_best, temp, max_iter=400)
            if c5_new < c5:
                h, c5 = h_new, c5_new
                h_best = h.copy()

    # Phase 5: Coarsen-refine cycle
    for n_c in [90, 60, 120]:
        if time_left() < 150:
            break
        result = coarsen_and_refine(h_best, n_c, None, 120)
        if result is not None:
            h_cr, c5_cr = result
            if c5_cr < c5:
                h, c5 = h_cr, c5_cr
                h_best = h.copy()

    # Phase 6: Diverse restarts
    for i in range(20):
        if time_left() < 70:
            break
        mode = i % 4
        if mode == 0:
            h_p = h_best + rng.randn(n) * (0.008 + 0.003 * (i // 4))
        elif mode == 1:
            center = rng.randint(n // 4, 3 * n // 4)
            width = max(3, rng.randint(n // 10, n // 3))
            bump = np.exp(-0.5 * ((np.arange(n) - center).astype(float) / width) ** 2)
            h_p = h_best + 0.04 * bump * rng.choice([1.0, -1.0])
        elif mode == 2:
            bs = max(2, rng.randint(n // 12, n // 5))
            st = rng.randint(0, n - bs)
            h_p = h_best.copy()
            h_p[st:st + bs] += rng.randn(bs) * 0.04
        else:
            # Symmetric perturbation
            h_p = h_best.copy()
            half = n // 2
            pert = rng.randn(half) * 0.02
            h_p[:half] += pert
            h_p[half:] += pert[::-1]
        h_p = np.clip(h_p, 0.0, 1.0)
        h_p = project_h(h_p, n * 0.5)
        h_new, c5_new = optimize_t_var(h_p, max_iter=250, ftol=1e-11)
        if c5_new < c5:
            h, c5 = h_new, c5_new
            h_best = h.copy()

    # Phase 7: Final intensive polish
    if time_left() > 80:
        h_new, c5_new = optimize_t_var(h_best, max_iter=600, ftol=1e-12)
        if c5_new < c5:
            h_best = h_new
            c5 = c5_new
    if time_left() > 50:
        h_new, c5_new = optimize_smooth(h_best, 0.0003, ones_w, max_iter=300, ftol=1e-14)
        if c5_new < c5:
            h_best = h_new
            c5 = c5_new
    if time_left() > 30:
        w = worst_lag_weights(h_best, top_k=12, decay=2.0)
        h_new, c5_new = optimize_smooth(h_best, 0.0003, w, max_iter=200, ftol=1e-14)
        if c5_new < c5:
            h_best = h_new
            c5 = c5_new

    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = project_h(h_best, len(h_best) * 0.5)
    c5_final = compute_c5(h_best)
    return h_best, c5_final, len(h_best)
