def compute_corr(h):
    return np.correlate(h, 1 - h, mode='full')

def compute_c5(h):
    n = len(h)
    return np.max(compute_corr(h)) * (2.0 / n)

def smooth_max_val(corr, alpha):
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha

def smooth_max_weights(corr, alpha, top_k=None):
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    s = w.sum()
    w = w / s
    if top_k is not None and top_k < len(w):
        top_idx = np.argsort(w)[-top_k:]
        w_new = np.zeros_like(w)
        w_new[top_idx] = 1.0
        w_new[top_idx] /= top_k
        w = w_new
    return w

    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n-k])

    else:

        g[:n-p] += (1.0 - h[p:])
        if p > 0:
            g[p:] -= h[:n-p]
    return g

    corr = compute_corr(h)
    w = smooth_max_weights(corr, alpha, top_k)

            patience = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        n = 120
        x = np.linspace(0, 2, n, endpoint=False)
        h0 = 0.5 + 0.25 * np.sin(2 * np.pi * x) + 0.1 * np.sin(4 * np.pi * x)
        h0 = project_simplex_box(h0, n / 2.0)

    n = len(h0)
    target_sum = n / 2.0
    h0 = project_simplex_box(h0, target_sum)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Main optimization
    deadline1 = t0 + min(budget_s * 0.30, 300.0)
    best_h, best_c5 = optimize_phase(h0, target_sum, deadline1, rng,
                                      alpha=80, top_k=15, lr=0.01)
    n = len(best_h)

    # Phase 2: Multiple structured restarts
    remaining = budget_s - (time.time() - t0)
    n_restarts = max(2, min(6, int(remaining / 80)))
    for r in range(n_restarts):
        if time.time() - t0 > budget_s - 40:
            break
        if r == 0:
            h_pert = project_simplex_box(best_h + rng.randn(n) * 0.05, target_sum)
        else:
            h_pert = structured_perturb(best_h, rng, n)
        h_pert_c5 = compute_c5(h_pert)
        if h_pert_c5 > best_c5 + 0.02:
            alpha_use = 60
            lr_use = 0.006
        else:
            alpha_use = 100
            lr_use = 0.01
        deadline_r = time.time() + min(remaining * 0.18, 130.0)
        h_opt, c5_opt = optimize_phase(h_pert, target_sum, deadline_r, rng,
                                        alpha=alpha_use, top_k=12, lr=lr_use)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt

    # Phase 3: Upscaling
    remaining = budget_s - (time.time() - t0) - 20
    for n_new in [n + 60, n + 120]:
        if n_new > 600 or remaining < 50:
            break
        h_interp = interpolate_h(best_h, n, n_new)
        h_interp = project_simplex_box(h_interp, n_new / 2.0)
        deadline = min(time.time() + remaining * 0.4, t0 + budget_s - 20)
        h_opt, c5_opt = optimize_phase(h_interp, n_new / 2.0, deadline, rng,
                                        alpha=80, top_k=15, lr=0.005)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
            n = n_new
        remaining = budget_s - (time.time() - t0) - 15

    # Phase 4: Denser upscaling if time permits
    remaining = budget_s - (time.time() - t0) - 10
    if remaining > 60 and n <= 300:
        n_new = min(n + 80, 500)
        h_interp = interpolate_h(best_h, n, n_new)
        h_interp = project_simplex_box(h_interp, n_new / 2.0)
        deadline = min(time.time() + remaining * 0.5, t0 + budget_s - 15)
        h_opt, c5_opt = optimize_phase(h_interp, n_new / 2.0, deadline, rng,
                                        alpha=100, top_k=12, lr=0.004)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt

    # Phase 5: Final polish with high alpha, low top_k
    remaining = budget_s - (time.time() - t0)
    if remaining > 12:
        n_final = len(best_h)
        best_h, best_c5 = optimize_phase(best_h, n_final / 2.0,
                                          t0 + budget_s - 3, rng,
                                          alpha=300, top_k=6, lr=0.004)

    best_h = project_simplex_box(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
