def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def corr_k(h, k):
    """Compute correlation at a single shift k (0-indexed from n-1)."""
    n = len(h)
    if k >= 0:
        overlap = n - k
        return np.sum(h[:overlap] * (1 - h[k:]))
    else:
        p = -k
        overlap = n - p
        return np.sum(h[p:] * (1 - h[:overlap]))

def grad_corr_single(h, k):
    """Gradient of corr_k with respect to h."""
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n - k])
        if k > 0:
            g[:n - k] -= h[k:]
    else:
        p = -k
        g[:n - p] += (1.0 - h[p:])
        if p > 0:
            g[p:] -= h[:n - p]
    return g

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    return np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n)

def grad_corr_single(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n-k])
        if k > 0:
            g[:n-k] -= h[k:]
    else:
        p = -k
        g[:n-p] += (1.0 - h[p:])
        if p > 0:
            g[p:] -= h[:n-p]
    return g

def project_onto_simplex_box(h, ts):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0, 1)) > ts:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0, 1)

def smooth_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= s
    g = np.zeros(n)
    for idx in np.where(w > 1e-8)[0]:
        k = int(idx) - n + 1
        g += w[idx] * grad_corr_single(h, k)
    return g * (2.0 / n)

def interp(h, no, nn):
    if no == nn:
        return h.copy()
    xo = np.linspace(0, 2, no, endpoint=False) + 1.0 / no
    xn = np.linspace(0, 2, nn, endpoint=False) + 1.0 / nn
    return np.interp(xn, np.append(xo, 2.0 + 1.0 / no), np.append(h, h[0]))

def optimize(h, deadline, rng, a_hard=400, a_soft=30):
    n = len(h)
    ts = n / 2.0
    h = project_onto_simplex_box(h.astype(float).copy(), ts)
    best, bc = h.copy(), compute_c5(h)
    hc = h.copy()
    lr, stall, mode = 0.002, 0, 0

    while time.time() < deadline:
        a = a_hard if mode == 0 else a_soft
        g = smooth_grad(hc, a)
        hn = project_onto_simplex_box(hc - lr * g, ts)
        c = compute_c5(hn)

        if c < bc - 1e-14:
            if c < bc:
                best, bc = hn.copy(), c
            hc = hn
            lr = min(lr * 1.04, 0.03)
            stall = 0
        else:
            stall += 1
            lr *= 0.94
            if stall == 40:
                mode = 1 - mode
                lr = 0.002 if mode == 0 else 0.004
            if stall > 120:
                bs = rng.randint(2, max(3, n // 20))
                s = rng.randint(0, n - bs)
                hp = hc.copy()
                hp[s:s+bs] += rng.randn(bs) * 0.05
                hp = project_onto_simplex_box(hp, ts)
                cp = compute_c5(hp)
                t = max(0.02, 0.1 * np.exp(-(stall - 120) / 60.0))
                if cp < compute_c5(hc) or rng.random() < np.exp(-max(0, cp - bc) / t):
                    hc = hp
                    stall = 40
                else:
                    stall = 0
                    lr = 0.002
                    mode = 0
            if stall > 300:
                hc = best + rng.randn(n) * 0.015
                hc = project_onto_simplex_box(hc, ts)
                stall = 0
                lr = 0.002
                mode = 0

    return best, bc
