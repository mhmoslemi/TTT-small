    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx

    if abs(diff) < 1e-14:
        return np.clip(h, 0.0, 1.0)

        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = np.sum(mask)
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def smooth_obj(h, temp):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    a = (corr - m) / temp
    # a has max = 0, so exp(a) <= 1
    return m + temp * np.log(np.sum(np.exp(a)))

    h_bin[indices] = 1.0

    """Push h toward 0/1 values while preserving the sum constraint."""
    n = len(h)
    target = n * 0.5
    h_new = h.copy()
    h_new = np.where(h_new > 0.5, 1.0, 0.0)
    # Adjust to meet constraint
    ones = np.sum(h_new)
    if ones > target:
        # Need to turn some 1s to 0s
        excess = ones - target
        # Turn the "least confident" 1s to 0s (those closest to 0.5)
        one_indices = np.where(h_new == 1.0)[0]
        conf = h[one_indices]  # original values, higher means more confident
        order = np.argsort(conf)  # least confident first
        h_new[one_indices[order[:int(excess)]]] = 0.0
    elif ones < target:
        deficit = target - ones
        zero_indices = np.where(h_new == 0.0)[0]
        conf = 1.0 - h[zero_indices]  # original values, lower means less confident about being 0
        order = np.argsort(conf)
        h_new[zero_indices[order[:int(deficit)]]] = 1.0
    return project_h(h_new, target)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_h(h, target):
    h = h.copy()
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return np.clip(h, 0.0, 1.0)
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = np.sum(mask)
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def smooth_obj(h, temp):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    a = (corr - m) / temp
    return m + temp * np.log(np.sum(np.exp(a)))

def sum_fn(h):
    return np.sum(h) - len(h) * 0.5

def sum_jac(h):
    return np.ones(len(h))

def t_obj(v):
    return v[-1]

def t_ineq(v):
    n = len(v) - 1
    h = v[:n]
    t = v[-1]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return t - corr

def t_eq(v):
    n = len(v) - 1
    return np.sum(v[:n]) - n * 0.5

def t_eq_jac(v):
    n = len(v) - 1
    j = np.zeros(len(v))
    j[:n] = 1.0
    return j

def interp_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.arange(n_old) * (1.0 / n_old)
    x_new = np.arange(n_new) * (1.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_h(h_new, n_new * 0.5)

def opt_smooth(h0, temp, max_iter=500):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_fn, 'jac': sum_jac}]
    res = minimize(smooth_obj, h0.copy(), args=(temp,), method='L-BFGS-B',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-6})
    h_new = np.clip(res.x, 0.0, 1.0)
    h_new = project_h(h_new, n * 0.5)
    return h_new, compute_c5(h_new)

def opt_t(h0, max_iter=500):

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_h(h, target):
    h = h.copy()
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return np.clip(h, 0.0, 1.0)
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = np.sum(mask)
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    a = (corr - m) / temp
    base = m + temp * np.log(np.sum(np.exp(a)))

    return v[-1]

def t_ineq(v):
    n = len(v) - 1
    h = v[:n]
    t = v[-1]
    dx = 2.0 / n

    n = len(v) - 1

    n = len(v) - 1

    j[:n] = 1.0

                h_best, c5 = h_new, c5_new

    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = project_h(h_best, len(h_best) * 0.5)
    c5_final = compute_c5(h_best)
    return h_best, c5_final, len(h_best)
