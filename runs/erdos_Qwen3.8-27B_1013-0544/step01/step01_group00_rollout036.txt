    h = h.copy()

        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break

        else:

        if cnt == 0:
            break

        h = np.clip(h, 0.0, 1.0)
    return h

    n = len(h)
    dx = 2.0 / n
    corr = get_corr(h)
    m = np.max(corr)
    shifted = (corr - m) / temp
    shifted = np.clip(shifted, -500, 0)
    exp_s = np.exp(shifted)
    log_sum = np.log(np.sum(exp_s))
    obj = m + temp * log_sum
    w = exp_s / np.sum(exp_s)
    half = n - 1
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        s = k - half
        wk = w[k]

        if s == 0:

            lo2 = max(0, s)
            hi2 = min(n, n + s)
            grad[lo2:hi2] -= dx * wk * h[lo2 - s:hi2 - s]
    return obj, grad

    n = len(h)
    obj, grad = lsq_obj_and_grad(h, temp)
    cv = np.sum(h) - n * 0.5
    obj += lam * cv * cv
    grad = grad + 2.0 * lam * cv * np.ones(n)
    return obj, grad

    return v[-1]

    n = len(v) - 1
    h = v[:n]
    t = v[-1]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return t - corr

def t_eq(v):
    n = len(v) - 1
    return np.sum(v[:n]) - n * 0.5

def t_eq_jac(v):
    j = np.zeros(len(v))
    j[:-1] = 1.0
    return j

def optimize_lsq(h0, temp, lam, maxiter=1000):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    
    def obj(h):
        return penalty_wrapper(h, temp, lam)[0]
    
    def grad(h):
        return penalty_wrapper(h, temp, lam)[1]
    
    res = minimize(obj, h0, jac=grad, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'gtol': 1e-10, 'ftol': 1e-14})
    h_new = project_h(res.x, n * 0.5)
    return h_new, compute_c5(h_new)

def optimize_tvar(h0, maxiter=500, ftol=1e-12):

    c5_cur = compute_c5(h0)
    v0 = np.append(h0.copy(), c5_cur + 0.005)
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    cons = [

        {'type': 'eq', 'fun': t_eq, 'jac': t_eq_jac}

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))


def get_corr(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode='full') * dx


def project_h(h, target_sum):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def lsq_func(h, temp, lam):
    n = len(h)
    corr = get_corr(h)
    m = np.max(corr)
    shifted = np.clip((corr - m) / temp, -500, 0)
    exp_s = np.exp(shifted)
    obj = m + temp * np.log(np.sum(exp_s))
    cv = np.sum(h) - n * 0.5
    return obj + lam * cv * cv


def lsq_jac_func(h, temp, lam):
    n = len(h)
    dx = 2.0 / n
    corr = get_corr(h)
    m = np.max(corr)

    res = minimize(lsq_func, h0, jac=lsq_jac_func, args=(temp, lam),
                   method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'gtol': 1e-12, 'ftol': 1e-16})
    h_new = project_h(res.x, n * 0.5)
    return h_new, compute_c5(h_new)


def optimize_tvar(h0, maxiter=500, ftol=1e-12):
    n = len(h0)
    c5_cur = compute_c5(h0)
    v0 = np.append(h0.copy(), c5_cur + 0.005)
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    cons = [
        {'type': 'ineq', 'fun': t_ineq},
        {'type': 'eq', 'fun': t_eq, 'jac': t_eq_jac}
    ]
    res = minimize(t_obj, v0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    h_new = project_h(res.x[:n], n * 0.5)
    return h_new, compute_c5(h_new)


def interp_h(h_old, n_new):
    x_old = np.linspace(0, 1, len(h_old), endpoint=False)
    x_new = np.linspace(0, 1, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_h(h_new, n_new * 0.5)


def make_perturb(h_base, rng, i):
    n = len(h_base)
    scale = 0.003 + 0.004 * (i % 6)
    mode = i % 4
    if mode == 0:
        return h_base + rng.randn(n) * scale
    elif mode == 1:
        freq = rng.randint(1, 12)
        phase = rng.rand() * 2 * np.pi
        return h_base + scale * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
    elif mode == 2:
        h_p = h_base.copy()
        start = rng.randint(0, n // 2)
        length = max(3, rng.randint(n // 10, n // 4))
        h_p[start:min(n, start + length)] += rng.randn(length) * scale
        return h_p
    else:
        h_p = h_base.copy()
        mask = rng.rand(n) < 0.3
        h_p[mask] += rng.randn(np.sum(mask)) * scale
        return h_p


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - t0)

    h = np.array(initial_h_values, dtype=float)
    n = len(h)
    h = project_h(h, n * 0.5)
    c5_best = compute_c5(h)
    h_best = h.copy()
    lam = 200.0

    # Phase 1: L-BFGS-B with decreasing temperature
    temps1 = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5]
    for temp in temps1:
        if tl() < 80:
            break
        h_new, c5_new = optimize_lsq(h, temp, lam, maxiter=800)
        if c5_new < c5_best:
            h_best = h_new
            c5_best = c5_new
        h = h_new

    # Phase 2: SLSQP t-variable polish
    if tl() > 180:
        h_new, c5_new = optimize_tvar(h, maxiter=600, ftol=1e-13)
        if c5_new < c5_best:
            h_best = h_new
            c5_best = c5_new
            h = h_new

    # Phase 3: More L-BFGS-B at ultra-low temp
    for temp in [2e-4, 1e-4, 5e-5]:
        if tl() < 60:
            break
        h_new, c5_new = optimize_lsq(h, temp, lam, maxiter=1500)
        if c5_new < c5_best:
            h_best = h_new
            c5_best = c5_new
        h = h_new

    # Phase 4: Resolution exploration
    for n_new in [120, 150, 200, 240, 300]:
        if tl() < 120:
            break
        h_int = interp_h(h_best, n_new)
        c5_int = compute_c5(h_int)
        if c5_int < c5_best:
            h_best = h_int
            c5_best = c5_int
        h_cur = h_int
        for temp in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]:
            if tl() < 50:
                break
            h_new, c5_new = optimize_lsq(h_cur, temp, lam, maxiter=500)
            if c5_new < c5_best:
                h_best = h_new
                c5_best = c5_new
            h_cur = h_new
        if tl() > 80:
            h_new, c5_new = optimize_tvar(h_cur, maxiter=400, ftol=1e-13)
            if c5_new < c5_best:
                h_best = h_new
                c5_best = c5_new

    # Phase 5: Refine back at original resolution from best
    if len(h_best) != n and tl() > 100:
        h_back = interp_h(h_best, n)
        for temp in [0.01, 0.005, 0.002, 0.001, 0.0005]:
            if tl() < 40:
                break
            h_new, c5_new = optimize_lsq(h_back, temp, lam, maxiter=500)
            if c5_new < c5_best:
                h_best = h_new
                c5_best = c5_new
            h_back = h_new

    # Phase 6: Random restarts
    n_base = len(h_best)
    for i in range(25):
        if tl() < 40:
            break
        h_p = make_perturb(h_best, rng, i)
        h_p = project_h(h_p, n_base * 0.5)
        h_new, c5_new = optimize_lsq(h_p, 0.01, lam, maxiter=400)
        if c5_new < c5_best:
            h_best = h_new
            c5_best = c5_new
        if tl() > 30 and c5_new < c5_best:
            h_new2, c5_new2 = optimize_lsq(h_new, 0.001, lam, maxiter=500)
            if c5_new2 < c5_best:
                h_best = h_new2
                c5_best = c5_new2
            if tl() > 20:
                h_new3, c5_new3 = optimize_tvar(h_new2, maxiter=300, ftol=1e-13)
                if c5_new3 < c5_best:
                    h_best = h_new3
                    c5_best = c5_new3

    # Final polish
    if tl() > 30:
        h_new, c5_new = optimize_lsq(h_best, 1e-5, lam, maxiter=2000)
        if c5_new < c5_best:
            h_best = h_new
            c5_best = c5_new

    h_best = project_h(h_best, len(h_best) * 0.5)
    c5_final = compute_c5(h_best)
    return h_best, c5_final, len(h_best)
