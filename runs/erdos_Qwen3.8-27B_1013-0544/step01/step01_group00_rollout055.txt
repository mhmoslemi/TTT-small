def project_h(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        mask = h < 1.0
    else:

    cnt = np.sum(mask)
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

    x_new = np.linspace(0, 1, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_h(h_new, n_new * 0.5)

def compute_grad_g(h, p, dx):
    """Gradient of g(h) = sum_k p_k * corr_k * dx"""
    n = len(h)
    # a[m] = sum(p[m:m+n])
    p_cumsum = np.concatenate([[0.0], np.cumsum(p)])
    a = np.array([p_cumsum[m+n] - p_cumsum[m] for m in range(n)])
    
    # b[m] = sum_s pc[s] * h[m-s]
    b = np.zeros(n)
    for s in range(-(n-1), n):
        w = p[s + n - 1]
        if abs(w) < 1e-15:
            continue
        m_start = max(0, s)
        m_end = min(n, s + n)
        b[m_start:m_end] += w * h[m_start - s:m_end - s]
    
    # c[m] = sum_s pc[s] * h[m+s]
    c = np.zeros(n)
    for s in range(-(n-1), n):
        w = p[s + n - 1]
        if abs(w) < 1e-15:
            continue
        m_start = max(0, -s)
        m_end = min(n, n - s)
        c[m_start:m_end] += w * h[m_start + s:m_end + s]
    
    return dx * (a - b - c)

        h = h_new
