    """Local search on binary solution using pair flips."""
    n = len(h)
    h_bin = (h > 0.5).astype(float)
    # Ensure exactly n/2 ones
    ones = np.where(h_bin == 1)[0]
    zeros = np.where(h_bin == 0)[0]
    while len(ones) < n//2:
        h_bin[zeros.pop(0)] = 1
        ones = np.where(h_bin == 1)[0]
    while len(ones) > n//2:
        h_bin[ones.pop(0)] = 0
        ones = np.where(h_bin == 1)[0]
    
    def autocorr(hb):
        return np.correlate(hb, hb, mode='full')
    
    def min_ac(hb):
        ac = autocorr(hb)
        # Only consider lags 1 to n-1 (exclude lag 0 which is always n/2)
        return np.min(ac[1:n])  # or some other range
    
    # Local search
    best_h = h_bin.copy()
    best_score = min_ac(best_h)
    
    for it in range(max_iter):
        # Pick a random pair (one, zero) to swap
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        
        new_h = best_h.copy()
        new_h[i] = 0
        new_h[j] = 1
        score = min_ac(new_h)
        
        if score >= best_score or np.random.random() < np.exp((score - best_score) / temperature):
            best_h = new_h
            best_score = score
    
    return best_h

def sum_project(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = np.sum(mask)
    if cnt == 0:
        return h
    h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def smooth_penalty_obj(h, temp, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip((corr - m) / temp, -50, 50)
    lse = m + temp * np.log(np.sum(np.exp(shifted)))
    s = np.sum(h) - n * 0.5
    return lse + penalty * s * s

    h_new = sum_project(res.x, n * 0.5)
    return h_new, compute_c5(h_new)


def lbgfs_adaptive(h0, temp, max_iter=200):
    n = len(h0)
    h_cur = h0.copy()
    for pen in [500.0, 5000.0, 50000.0]:
        h_new, _ = lbgfs_smooth(h_cur, temp, pen, max_iter=max_iter)
        if abs(np.sum(h_new) - n * 0.5) < 1e-7:
            h_cur = h_new
            break
        h_cur = h_new
    return h_cur, compute_c5(h_cur)


def _t_ineq(v):
    n = len(v) - 1
    h = v[:n]
    t = v[-1]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return t - corr


def _t_eq(v):
    n = len(v) - 1
    return np.sum(v[:n]) - n * 0.5


def _t_obj(v):
    return v[-1]


def slsqp_tvar(h0, max_iter=600, ftol=1e-12):
    n = len(h0)
    c5 = compute_c5(h0)
    v0 = np.append(h0.copy(), c5 + 0.003)
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    cons = [{'type': 'ineq', 'fun': _t_ineq}, {'type': 'eq', 'fun': _t_eq}]
    res = minimize(_t_obj, v0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': ftol})
    h_new = sum_project(res.x[:n], n * 0.5)
    return h_new, compute_c5(h_new)


def make_anti_symmetric(n, h_half):
    h = np.zeros(n)
    h[:n // 2] = h_half
    h[n // 2:] = 1.0 - h_half
    return h


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - (time.time() - t0)

    h = np.array(initial_h_values, dtype=float)
    n = len(h)
    h = sum_project(h, n * 0.5)
    c5 = compute_c5(h)
    h_best = h.copy()

    # Phase 1: L-BFGS-B smooth refinement with temperature schedule
    temps = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    for temp in temps:
        if tleft() < 250:
            break
        h_new, c5_new = lbgfs_adaptive(h, temp, max_iter=300)
        if c5_new < c5:
            h, c5 = h_new, c5_new
            h_best = h.copy()

    # Phase 2: SLSQP t-variable refinement
    if tleft() > 250:
        h_new, c5_new = slsqp_tvar(h_best, max_iter=800, ftol=1e-13)
        if c5_new < c5:
            h_best = h_new.copy()
            c5 = c5_new

    # Phase 3: Further L-BFGS-B at very low temps
    for temp in [0.0001, 5e-05, 2e-05]:
        if tleft() < 200:
            break
        h_new, c5_new = lbgfs_smooth(h_best, temp, 100000.0, max_iter=500)
        if c5_new < c5:
            h_best, c5 = h_new, c5_new

    # Phase 4: SLSQP again after smooth refinement
    if tleft() > 250:
        h_new, c5_new = slsqp_tvar(h_best, max_iter=1000, ftol=1e-14)
        if c5_new < c5:
            h_best, c5 = h_new, c5_new

    # Phase 5: Diverse restarts
    for idx in range(12):
        if tleft() < 150:
            break
        if idx == 0:
            h_r = h_best + rng.randn(n) * 0.04
        elif idx == 1:
            h_r = h_best + rng.randn(n) * 0.08
        elif idx == 2:
            h_r = h_best + rng.randn(n) * 0.15
        elif idx == 3:
            h_half = 0.5 + 0.35 * rng.randn(n // 2)
            h_r = make_anti_symmetric(n, h_half)
        elif idx == 4:
            h_half = 0.5 + 0.4 * np.sin(2 * np.pi * 2 * np.arange(n // 2) / (n // 2)) + 0.1 * rng.randn(n // 2)
            h_r = make_anti_symmetric(n, h_half)
        elif idx == 5:
            h_half = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * np.arange(n // 2) / (n // 2)) + 0.15 * rng.randn(n // 2)
            h_r = make_anti_symmetric(n, h_half)
        elif idx == 6:
            h_r = 0.5 + 0.4 * np.sin(2 * np.pi * rng.randint(2, 6) * np.arange(n) / n) + 0.1 * rng.randn(n)
        elif idx == 7:
            h_r = 0.5 + 0.35 * rng.randn(n)
        elif idx == 8:
            h_r = np.clip(h_best + rng.choice([-1, 1], n) * 0.06, 0, 1)
        elif idx == 9:
            h_r = 0.5 + 0.45 * rng.randn(n)
        else:
            h_r = h_best + rng.randn(n) * (0.05 + 0.02 * idx)
        h_r = sum_project(h_r, n * 0.5)
        h_new, c5_new = lbgfs_adaptive(h_r, temp=0.01, max_iter=250)
        if c5_new < c5:
            h_best, c5 = h_new.copy(), c5_new
            h_new2, c5_new2 = lbgfs_smooth(h_new, 0.001, 50000.0, max_iter=300)
            if c5_new2 < c5:
                h_best, c5 = h_new2.copy(), c5_new2

    # Phase 6: Fine-tune perturbations from best
    for scale in [0.01, 0.02, 0.03, 0.015, 0.025]:
        if tleft() < 100:
            break
        h_p = sum_project(h_best + rng.randn(n) * scale, n * 0.5)
        h_new, c5_new = lbgfs_smooth(h_p, 0.003, 50000.0, max_iter=350)
        if c5_new < c5:
            h_best, c5 = h_new.copy(), c5_new

    # Phase 7: Final SLSQP polishing
    if tleft() > 80:
        h_new, c5_new = slsqp_tvar(h_best, max_iter=1200, ftol=1e-14)
        if c5_new < c5:
            h_best, c5 = h_new.copy(), c5_new

    # Final validation
    h_best = sum_project(h_best, n * 0.5)
    c5 = compute_c5(h_best)
    return h_best, c5, n
