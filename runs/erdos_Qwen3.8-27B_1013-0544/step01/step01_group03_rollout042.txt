```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-12
    else:
        mask = h > 1e-12
    cnt = mask.sum()
    if cnt == 0:
        return h
    h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def compute_c5_and_weights(h, tau):
    """Compute C5 smooth max value and softmax weights over lags."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    x = tau * corr
    x_max = np.max(x)
    val = x_max / tau + np.log(np.sum(np.exp(x - x_max))) / tau
    w = np.exp(x - x_max)
    w = w / w.sum()
    return val, w, corr

def gradient_smooth_max(h, w):
    """Gradient of smooth_max(corr(h), tau) w.r.t. h."""
    n = len(h)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-16:
            continue
        shift = k - (n - 1)
        # (1-h[j-shift]) for valid j
        js = max(0, shift)
        je = min(n, shift + n)
        if je > js:
            grad[js:je] += wk * (1.0 - h[js - shift:je - shift])
        # -h[j+shift] for valid j
        js2 = max(0, -shift)
        je2 = min(n, n - shift)
        if je2 > js2:
            grad[js2:je2] -= wk * h[js2 + shift:je2 + shift]
    return grad

def opt_al(h0, tau, verbose=False):
    """Augmented Lagrangian + L-BFGS-B optimizer."""
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    h = fix_sum(h, target)
    nu = 0.0
    rho = 1.0
    
    bnds = Bounds(np.zeros(n), np.ones(n))
    
    for outer in range(30):
        def obj(x):
            val, _, _ = compute_c5_and_weights(x, tau)
            s = x.sum() - target
            return val + 0.5 * rho * s * s + nu * s
        
        def grad(x):
            val, w, _ = compute_c5_and_weights(x, tau)
            g = gradient_smooth_max(x, w)
            s = x.sum() - target
            g += rho * s + nu
            return g
        
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                      bounds=bnds, options={'maxiter': 200, 'ftol': 1e-15})
        h = np.clip(res.x, 0.0, 1.0)
        
        s = h.sum() - target
        if abs(s) < 1e-10:
            break
        nu -= rho * s
        if abs(s) > 1e-6:
            rho = min(rho * 5, 1e6)
        else:
            rho = max(rho * 0.5, 1e-2)
    
    h = fix_sum(h, target)
    return h, evaluate_c5(h)

def opt_slsqp(h0, tau, maxiter=200):
    """SLSQP with smooth max."""
    n = len(h0)
    target = n / 2.0
    bnds = Bounds(np.zeros(n), np.ones(n))
    
    def obj(x):
        val, _, _ = compute_c5_and_weights(x, tau)
        return val
    
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bnds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        h = fix_sum(res.x, target)
        return h, evaluate_c5(h)
    except:
        return h0, None

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    def remaining():
```

```python
    return x.sum() - _g_target

def _slsqp_jac(x):
    return np.ones(_g_n)

def opt_slsqp(h0, tau, maxiter=300):
    global _g_tau, _g_n, _g_target
    _g_tau = float(tau)
    n = len(h0)
    _g_n = n
    _g_target = n / 2.0
    bnds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': _slsqp_fun, 'jac': _slsqp_jac}]
    try:
        res = minimize(_slsqp_obj, h0, method='SLSQP', bounds=bnds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        h = fix_sum(res.x, _g_target)
        return h, evaluate_c5(h)
    except Exception:
        return h0, None

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(120, 0.5)

    best_h = h_init.copy()
    best_c5 = evaluate_c5(best_h)

    # Phase 1: Anneal tau on initial resolution with AL+L-BFGS-B
    tau_list = [30, 80, 200, 500, 1500, 4000]
    for tau in tau_list:
        rem = budget_s - (time.time() - t0)
        if rem < 35:
            break
        h_new, c5_new = opt_al(best_h, tau, max_outer=20)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new
        # Second pass at same tau for refinement
        rem = budget_s - (time.time() - t0)
        if rem > 20 and tau >= 500:
            h_new2, c5_new2 = opt_al(best_h, tau * 2, max_outer=10)
            if c5_new2 < best_c5:
                best_h, best_c5 = h_new2, c5_new2

    # Phase 2: SLSQP polish at current resolution
    rem = budget_s - (time.time() - t0)
    if rem > 25:
        h_new, c5_new = opt_slsqp(best_h, 3000, maxiter=400)
        if c5_new is not None and c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 3: Try different resolutions
    for tn in [80, 100, 140, 160, 200]:
        rem = budget_s - (time.time() - t0)
        if rem < 45:
            break
        h_t = np.interp(np.linspace(0, 1, tn), np.linspace(0, 1, len(best_h)), best_h)
        h_t = fix_sum(h_t, tn / 2.0)
        for tau in [80, 400, 2000]:
            rem = budget_s - (time.time() - t0)
            if rem < 18:
                break
            h_new, c5_new = opt_al(h_t, tau, max_outer=15)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new
            h_t = h_new

    # Phase 4: SLSQP on best resolution
    rem = budget_s - (time.time() - t0)
    if rem > 20:
        h_new, c5_new = opt_slsqp(best_h, 5000, maxiter=300)
        if c5_new is not None and c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 5: Diverse random restarts
    hn = len(best_h)
    for it in range(50):
        rem = budget_s - (time.time() - t0)
        if rem < 10:
            break
        r = np.random.random()
        if r < 0.25:
            hp = best_h + np.random.normal(0, 0.04, hn)
        elif r < 0.5:
            hp = best_h + np.random.normal(0, 0.12, hn)
        elif r < 0.7:
            # Perturb only a random subset
            hp = best_h.copy()
            idx = np.random.choice(hn, size=hn // 3, replace=False)
            hp[idx] += np.random.normal(0, 0.2, len(idx))
        elif r < 0.85:
            # Binary-like
            hp = np.where(best_h > 0.4, 0.8, 0.2) + np.random.normal(0, 0.1, hn)
        else:
            # Complement exploration (leveraging h <-> 1-h symmetry)
            hp = 1.0 - best_h + np.random.normal(0, 0.05, hn)
        hp = fix_sum(hp, hn / 2.0)
        tau = [100, 500, 1500][it % 3]
        h_new, c5_new = opt_al(hp, tau, max_outer=12)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 6: Final SLSQP polish
    rem = budget_s - (time.time() - t0)
    if rem > 8:
        h_new, c5_new = opt_slsqp(best_h, 8000, maxiter=200)
        if c5_new is not None and c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new
        rem = budget_s - (time.time() - t0)
        if rem > 5:
            h_new, c5_new = opt_slsqp(best_h, 20000, maxiter=100)
            if c5_new is not None and c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new

    # Final constraint enforcement
    best_h = fix_sum(best_h, len(best_h) / 2.0)
    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, len(best_h)
```
