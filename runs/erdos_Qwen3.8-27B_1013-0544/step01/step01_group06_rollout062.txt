import numpy as np
import time
from scipy.optimize import minimize

def compute_corr(h):
    return np.correlate(h, 1 - h, mode='full')

def compute_c5(h):
    n = len(h)
    return np.max(compute_corr(h)) * (2.0 / n)

def smooth_max(corr, alpha):
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    w /= np.sum(w)
    # Gradient of smooth max w.r.t. h
    grad = np.zeros(n)
    active = np.where(w > 1e-8)[0]
    for idx in active:
        k = idx - (n - 1)
        if k >= 0:
            g = np.zeros(n)
            g[k:] += (1.0 - h[:n-k])
            g[:n-k] -= h[k:]
            grad += w[idx] * g
        else:
            p = -k
            g = np.zeros(n)
            g[:n-p] += (1.0 - h[p:])
            g[p:] -= h[:n-p]
            grad += w[idx] * g
    return grad * (2.0 / n)

def objective_and_grad(h, alpha, penalty):
    n = len(h)
    target = n / 2.0
    # Smooth max part
    val = smooth_max(compute_corr(h), alpha) * (2.0/n)
    g = smooth_max_grad(h, alpha)
    # Penalty for sum constraint
    s = np.sum(h)
    val += penalty * (s - target) ** 2
    g += 2 * penalty * (s - target) * np.ones(n)
    return val, g

def smooth_max_grad(h, alpha):

    corr = compute_corr(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -500, 500)

    w /= np.sum(w)

            g[:n-k] += (1.0 - h[k:])

import time
from scipy.optimize import minimize

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    return float(np.max(compute_corr(h)) * (2.0 / n))

def smooth_max(corr, alpha):
    m = np.max(corr)
    s = np.exp(alpha * (corr - m))
    s = np.clip(s, 0, 1e300)
    return m + np.log(np.sum(s)) / alpha

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    s = np.exp(np.clip(alpha * (corr - m), -500, 500))
    s /= np.sum(s)
    grad = np.zeros(n)
    for idx in np.where(s > 1e-5)[0]:
        k = idx - (n - 1)
        g = np.zeros(n)
        if k >= 0:
            g[:n-k] += (1.0 - h[k:])
            g[k:] -= h[:n-k]
        else:
            p = -k
            g[:n-p] += (1.0 - h[p:])
            g[p:] -= h[:n-p]
        grad += s[idx] * g
    return grad * (2.0 / n)

def project_box_sum(h, target):
    """Project h to [0,1]^n with sum = target using bisection."""
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def objective(h, alpha, penalty, target):
    n = len(h)
    dx = 2.0 / n
    val = smooth_max(compute_corr(h), alpha) * dx
    diff = np.sum(h) - target
    val += penalty * diff * diff
    g = smooth_max_grad(h, alpha)
    g += 2.0 * penalty * diff
    return val, g

def run_lbfgs(h0, target, alpha, penalty, deadline, maxiter=1500):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    t_dead = [deadline]
    
    def fun(x):
        if time.time() > t_dead[0]:
            return 1e10, np.zeros(n)
        return objective(x, alpha, penalty, target)
    
    res = minimize(fun, h0, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-9,
                            'maxls': 40, 'iprint': -1})
    return res.x

def subgrad_refine(h, target, deadline, max_iter=500):
    """Subgradient descent with momentum for the exact max."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    momentum = np.zeros(n)
    lr = 0.01
    beta = 0.9
    
    for i in range(max_iter):
        if time.time() > deadline:
            break
        corr = compute_corr(h)
        k_star = np.argmax(corr)
        k = k_star - (n - 1)
        g = np.zeros(n)
        if k >= 0:
            g[:n-k] += (1.0 - h[k:])
            g[k:] -= h[:n-k]
        else:
            p = -k
            g[:n-p] += (1.0 - h[p:])
            g[p:] -= h[:n-p]
        g *= dx
        
        momentum = beta * momentum + lr * g
        h_new = h - momentum
        h_new = project_box_sum(h_new, target)
        
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = c5
            h = h_new
            lr = min(lr * 1.02, 0.05)
        else:
            lr = lr * 0.95
            h = project_box_sum(h, target)
    
    return best_h, best_c5

    if rem() > 12:
        dl = t0 + budget_s - 2
        h_new, c5 = subgrad_refine(best_h, tgt_best, dl, 600)
        if c5 < best_c5:
            best_h = h_new
            best_c5 = c5

    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
