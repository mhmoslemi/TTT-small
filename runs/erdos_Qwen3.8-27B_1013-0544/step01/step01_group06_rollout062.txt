import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def compute_corr(h):
    return np.correlate(h, 1 - h, mode='full')

def project_onto_simplex_box(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_grad_smooth(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    s = np.sum(w)
    if s > 1e-300:
        w /= s
    else:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    
    h_grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-10:

        k = idx - (n - 1)
        if k >= 0:
            h_grad[:n-k] += wi * (1.0 - h[k:])
            if k > 0:
                h_grad[k:] -= wi * h[:n-k]
        else:
            p = -k
            h_grad[p:] += wi * (1.0 - h[:n-p])
            if p > 0:
                h_grad[:n-p] -= wi * h[p:]
    
    return h_grad * (2.0 / n)

def adam_optimize(h0, target_sum, deadline, rng, alpha=100, lr=0.01):

    h = project_onto_simplex_box(h0.astype(float).copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    alpha_current = alpha
    
    while time.time() < deadline:
        t += 1
        grad = compute_grad_smooth(h, alpha_current)
        
        m_v = beta1 * m_v + (1 - beta1) * grad
        v_v = beta2 * v_v + (1 - beta2) * grad**2
        m_hat = m_v / (1 - beta1**t)
        v_hat = v_v / (1 - beta2**t)
        
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_onto_simplex_box(h_new, target_sum)
        
        new_c5 = compute_c5(h_new)
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
        
        h = h_new
        # Slowly increase alpha to focus on the max

    return best_h, best_c5

def simulated_annealing(h0, target_sum, deadline, rng):

def compute_corr(h):
    return np.correlate(h, 1 - h, mode='full')

def compute_c5(h):
    n = len(h)
    return float(np.max(compute_corr(h)) * (2.0 / n))

def project(h, ts):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > ts:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def grad_smooth(h, a):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    e = np.exp(a * (corr - m))
    s = e.sum()
    w = e / s if s > 1e-300 else np.eye(2*n-1)[np.argmax(corr)]
    g = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        if k >= 0:
            g[:n-k] += wi * (1.0 - h[k:])
            if k > 0:
                g[k:] -= wi * h[:n-k]
        else:
            p = -k
            g[p:] += wi * (1.0 - h[:n-p])
            if p > 0:
                g[:n-p] -= wi * h[p:]
    return g * (2.0 / n)

                vv = np.zeros(n)
                stall = 0
    return bh, bc

def subgrad_polish(h0, ts, deadline, rng):
    n = len(h0)
    h = h0.copy()
    bh = h.copy()
    bc = compute_c5(h)
    lr = 0.02
    stall = 0

    while time.time() < deadline:
        corr = compute_corr(h)
        ks = np.argmax(corr)
        k = ks - (n - 1)
        g = np.zeros(n)
        if k >= 0:
            g[:n-k] += (1.0 - h[k:])
            if k > 0:
                g[k:] -= h[:n-k]
        else:
            p = -k
            g[p:] += (1.0 - h[:n-p])
            if p > 0:
                g[:n-p] -= h[p:]
        g *= (2.0 / n)
        h_new = project(h - lr * g, ts)
        c_new = compute_c5(h_new)
        if c_new < bc:
            h = h_new
            bc = c_new
            bh = h.copy()
            lr = min(lr * 1.005, 0.08)
            stall = 0
        else:
            lr *= 0.92
            stall += 1
            if stall > 100:
                lr = 0.01
                h = project(h + rng.randn(n) * 0.008, ts)
                stall = 0
    return bh, bc

def interp(h, nn):
    xo = np.arange(len(h), dtype=float) / len(h)
    xn = np.arange(nn, dtype=float) / nn
    return np.interp(xn, xo, h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = rng.rand(150)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    base_n = len(best_h)

    # Phase 1: Main Adam (broad exploration)
    d = t0 + budget_s * 0.30
    h, c = adam_opt(h0, base_n / 2.0, d, rng, a0=15, a1=250, lr0=0.04, lr1=0.005)
    if c < best_c5:
        best_h, best_c5 = h, c

    # Phase 2: Adam with higher alpha (focused refinement)
    rem = budget_s - (time.time() - t0)
    if rem > 30:
        d = time.time() + min(rem * 0.18, 120.0)
        h, c = adam_opt(best_h, len(best_h) / 2.0, d, rng, a0=100, a1=600, lr0=0.015, lr1=0.002)
        if c < best_c5:
            best_h, best_c5 = h, c

    # Phase 3: Subgradient polish on worst shift
    rem = budget_s - (time.time() - t0)
    if rem > 20:
        d = time.time() + min(rem * 0.15, 80.0)
        h, c = subgrad_polish(best_h, len(best_h) / 2.0, d, rng)
        if c < best_c5:
            best_h, best_c5 = h, c

    # Phase 4: Perturbed restart with different trajectory
    rem = budget_s - (time.time() - t0)
    if rem > 30:
        h_p = project(best_h + rng.randn(len(best_h)) * 0.02, len(best_h) / 2.0)
        d = time.time() + min(rem * 0.15, 80.0)
        h, c = adam_opt(h_p, len(best_h) / 2.0, d, rng, a0=40, a1=400, lr0=0.02, lr1=0.003)
        if c < best_c5:
            best_h, best_c5 = h, c

    # Phase 5: Higher resolution refinement
    rem = budget_s - (time.time() - t0)
    for nn in [300, 400]:
        if rem < 50:
            break
        hi = project(interp(best_h, nn), nn / 2.0)
        d = time.time() + min(rem * 0.22, 120.0)
        h, c = adam_opt(hi, nn / 2.0, d, rng, a0=30, a1=350, lr0=0.02, lr1=0.004)
        if c < best_c5:
            best_h, best_c5 = h, c
        rem = budget_s - (time.time() - t0)

    # Phase 6: Final subgradient polish
    rem = budget_s - (time.time() - t0)
    if rem > 15:
        h, c = subgrad_polish(best_h, len(best_h) / 2.0, time.time() + rem - 3, rng)
        if c < best_c5:
            best_h, best_c5 = h, c

    best_h = project(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
