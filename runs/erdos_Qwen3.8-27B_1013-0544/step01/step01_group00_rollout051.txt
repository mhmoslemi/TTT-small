```python
def smooth_gradient(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    M = np.max(corr)
    p = np.exp((corr - M) / temp)
    p = p / np.sum(p)  # softmax weights
    
    # Gradient of log-sum-exp: sum_j p_j * grad(c_j)
    # c_j = dx * sum_i h[i+lag_j] * (1-h[i]) where lag_j = j - (n-1)
    # grad(c_j)[m] = dx * [(1-h[m-lag_j]) * [valid(m-lag_j)] - h[m+lag_j] * [valid(m+lag_j)]]
    
    grad = np.zeros(n)
    lags = np.arange(2*n-1) - (n-1)  # lags from -(n-1) to (n-1)
    
    for j in range(2*n-1):
        lag = lags[j]
        if p[j] < 1e-10:
            continue
        # (1-h[m-lag]) for valid m-lag
        m1 = np.arange(n)
        idx1 = m1 - lag
        valid1 = (idx1 >= 0) & (idx1 < n)
        term1 = np.zeros(n)
        term1[valid1] = 1.0 - h[idx1[valid1]]
        
        # h[m+lag] for valid m+lag
        idx2 = m1 + lag
        valid2 = (idx2 >= 0) & (idx2 < n)
        term2 = np.zeros(n)
        term2[valid2] = h[idx2[valid2]]
        
        grad += p[j] * dx * (term1 - term2)
    
    return grad
```

```python
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_h(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = np.sum(mask)
    if cnt > 0:
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, temp):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    M = np.max(corr)
    return M + temp * np.log(np.sum(np.exp((corr - M) / temp)))

def smooth_grad(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    M = np.max(corr)
    p = np.exp((corr - M) / temp)
    p = p / p.sum()
    lags = np.arange(2 * n - 1) - (n - 1)
    m_idx = np.arange(n)
    diff1 = m_idx[:, None] - lags[None, :]
    valid1 = (diff1 >= 0) & (diff1 < n)
    idx1 = np.where(valid1, diff1, 0)
    term1 = (p[None, :] * (1.0 - h[idx1]) * valid1).sum(axis=1)
    diff2 = m_idx[:, None] + lags[None, :]
    valid2 = (diff2 >= 0) & (diff2 < n)
    idx2 = np.where(valid2, diff2, 0)
    term2 = (p[None, :] * h[idx2] * valid2).sum(axis=1)
    return dx * (term1 - term2)

    y = h0.copy()
    h = h0.copy()

        h = h_new
        t = t_new
        S_prev = S_new
        L = max(L / 1.3, 0.5)
```

```python
            h_new, c5_new = fista_smooth(h_cur, temp, max_iter=4000,
                                         time_limit=time_left() - 40, t0=t0)
            if c5_new < c5_cur:
                h_cur, c5_cur = h_new, c5_new
        if c5_cur < c5:
            h_best = h_cur.copy()
            c5 = c5_cur

    # Phase 7: Final SLSQP polish
    if time_left() > 80:
        h_new, c5_new = optimize_t_var(h_best, max_iter=300, ftol=1e-11)
        if c5_new < c5:
            h_best = h_new
            c5 = c5_new

    h_best = project_h(h_best, len(h_best) * 0.5)
    c5_final = compute_c5(h_best)
    return h_best, c5_final, len(h_best)
```
