    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    g = corr  # These are the correlation values (before dx multiplication)
    
    # Compute weights
    max_g = np.max(g)
    shifted = t * (g - max_g)
    weights = np.exp(shifted) / np.exp(np.max(shifted))
    weights = weights / np.sum(weights)
    
    # Compute gradient: sum_k w_k * grad_k
    # grad_k[j] = (1-h[j+k]) * 1[valid] - h[j-k] * 1[valid]
    grad = np.zeros(n)
    
    for m in range(len(corr)):
        k = m - (n - 1)
        w = weights[m]
        if w < 1e-10:
            continue
        
        # First term: (1-h[j+k]) for j in valid range
        # Valid range for i: max(0,-k) <= i <= min(n-1, n-1-k)
        i_lo = max(0, -k)

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_onto_simplex(y, target_sum):
    n = len(y)
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        h = np.clip(y - mid, 0.0, 1.0)
        s = np.sum(h)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) / 2, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_grad(h, t):
    """Compute smooth max value and gradient."""
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    
    max_g = np.max(corr)
    shifted = t * (corr - max_g)
    exp_shifted = np.exp(shifted - np.max(shifted))
    weights = exp_shifted / np.sum(exp_shifted)
    
    smooth_val = max_g + np.log(np.sum(exp_shifted)) / t
    
    # Gradient
    grad = np.zeros(n)
    for m in range(len(corr)):
        w = weights[m]
        if w < 1e-8:
            continue
        k = m - (n - 1)
        # First term: (1-h[j+k]) for j in valid range
        i_lo = max(0, -k)
        i_hi = min(n - 1, n - 1 - k)
        if i_lo <= i_hi:
            idx = np.arange(i_lo, i_hi + 1)
            grad[idx] += w * (1.0 - h[idx + k])
        # Second term: -h[j-k] for j in valid range
        j_lo = max(0, i_lo + k)
        j_hi = min(n - 1, i_hi + k)
        if j_lo <= j_hi:
            idx = np.arange(j_lo, j_hi + 1)
            grad[idx] -= w * h[idx - k]
    
    return smooth_val, grad

    y = h.copy()

        y_new = y - step * grad
        h_new = project_onto_simplex(y_new, target_sum)
        
        if it % 100 == 0:

            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        
        kk_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * kk * kk))
        y = h_new + ((kk - 1.0) / kk_new) * (h_new - h)
        h = h_new
        kk = kk_new
    
    c5_final = compute_c5(h, dx)
    if c5_final < best_c5:
        best_h = h.copy()
        best_c5 = c5_final
    
    return best_h, best_c5
