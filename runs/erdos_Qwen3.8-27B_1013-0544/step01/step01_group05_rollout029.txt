    lo = float(x.min()) - 1.0
    hi = float(x.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = float(np.clip(x - mid, 0.0, 1.0).sum())
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_grad_lag(h, L):
    n = len(h)
    omh = 1.0 - h
    grad = np.zeros(n)
    if L == 0:
        grad[:] = omh[:] - h[:]
    elif L > 0:
        grad[L:] += omh[:n - L]
        grad[:n - L] -= h[L:]
    else:
        L2 = -L
        grad[:n - L2] += omh[L2:]
        grad[L2:] -= h[:n - L2]
    return grad

    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = beta * corr
    m = np.max(scaled)
    return (m + np.log(np.sum(np.exp(scaled - m)))) / beta * dx

                          options={'maxiter': 3000, 'ftol': 1e-16, 'gtol': 1e-12})
            if res.x is not None:
                h_ref = project_feasible(res.x, target_sum)
                c5_ref = compute_c5(h_ref, dx)
                if c5_ref < best_c5:
                    best_c5 = c5_ref
                    best_h = h_ref.copy()
                    best_n = n_points
                h_curr = h_ref
        except Exception:
            pass

    # Phase 2: SLSQP with smooth objective, continuation
    h_curr = best_h.copy()
    for beta in [100, 500, 2000, 5000, 10000]:
        if tleft() < 40:
            break
        def sobj(x):
            return smooth_c5_obj(x, dx, beta)
        def sgr(x):
            return smooth_c5_grad(x, dx, beta)
        def seq(x):
            return np.sum(x) - target_sum
        try:
            res = minimize(sobj, h_curr, jac=sgr, method='SLSQP',
                          bounds=[(0.0, 1.0)] * n_points,
                          constraints=[{'type': 'eq', 'fun': seq}],
                          options={'maxiter': 3000, 'ftol': 1e-15})
            if res.x is not None:
                h_ref = project_feasible(res.x, target_sum)
                c5_ref = compute_c5(h_ref, dx)
                if c5_ref < best_c5:
                    best_c5 = c5_ref
                    best_h = h_ref.copy()
                    best_n = n_points
                h_curr = h_ref
        except Exception:
            pass

    # Phase 3: SLSQP with exact max objective
    if tleft() > 30:
        def eobj(x):
            return compute_c5(x, dx)
        def eeq(x):
            return np.sum(x) - target_sum
        try:
            res = minimize(eobj, best_h, method='SLSQP',
                          bounds=[(0.0, 1.0)] * n_points,
                          constraints=[{'type': 'eq', 'fun': eeq}],
                          options={'maxiter': 5000, 'ftol': 1e-16})
            if res.x is not None:
                h_ref = project_feasible(res.x, target_sum)
                c5_ref = compute_c5(h_ref, dx)
                if c5_ref < best_c5:
                    best_c5 = c5_ref
                    best_h = h_ref.copy()
                    best_n = n_points
        except Exception:
            pass

    # Phase 4: Random restarts with continuation
    restarts = [(0.02, 1000), (0.04, 500), (0.06, 500),
                (0.01, 2000), (0.08, 300), (0.03, 1000),
                (0.10, 200), (0.05, 800)]
    for scale, beta_r in restarts:
        if tleft() < 30:
            break
        h_p = best_h + rng.normal(0, scale, n_points)
        h_p = project_feasible(h_p, target_sum)
        for beta_try in [100, beta_r, max(2000, beta_r)]:
            def robj(x):
                return smooth_c5_obj(x, dx, beta_try)
            def rgr(x):
                return smooth_c5_grad(x, dx, beta_try)
            def req(x):
                return np.sum(x) - target_sum
            try:
                res = minimize(robj, h_p, jac=rgr, method='SLSQP',
                              bounds=[(0.0, 1.0)] * n_points,
                              constraints=[{'type': 'eq', 'fun': req}],
                              options={'maxiter': 2000, 'ftol': 1e-14})
                if res.x is not None:
                    h_p = project_feasible(res.x, target_sum)
            except Exception:
                break
        c5_p = compute_c5(h_p, dx)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p.copy()
            best_n = n_points

    # Phase 5: Active-set gradient fine-tuning
    if tleft() > 15:
        h_curr = best_h.copy()
        n_h = len(h_curr)
        n_half2 = n_h - 1
        dx_h = 2.0 / n_h
        tgt_h = n_h / 2.0
        step = 0.0005
        for it in range(3000):
            if tleft() < 2:
                break
            corr = np.correlate(h_curr, 1.0 - h_curr, mode='full')
            c_max = np.max(corr)
            tol = max(1e-8, c_max * 1e-6)
            active = np.where(corr >= c_max - tol)[0]
            if len(active) > 15:
                active = np.argsort(corr)[-15:]
            grad = np.zeros(n_h)
            for idx in active:
                L = int(idx) - n_half2
                grad += compute_grad_lag(h_curr, L)
            grad /= float(len(active))
            h_new = project_feasible(h_curr - step * grad, tgt_h)
            c5_new = compute_c5(h_new, dx_h)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                best_n = n_h
                h_curr = h_new
                step = min(step * 1.15, 0.01)
            else:
                step *= 0.85
                if step < 1e-8:
                    break

    # Phase 6: Try different n_points
    for n_alt in [50, 60, 80, 100]:
        if tleft() < 40:
            break
        dx_a = 2.0 / n_alt
        tgt_a = n_alt / 2.0
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        x_new = np.linspace(0, 2, n_alt, endpoint=False)
        h_a = np.interp(x_new, x_old, best_h)
        h_a = project_feasible(h_a, tgt_a)
        for beta_a in [100, 500, 2000]:
            def aobj(x):
                return smooth_c5_obj(x, dx_a, beta_a)
            def agr(x):
                return smooth_c5_grad(x, dx_a, beta_a)
            def aeq(x):
                return np.sum(x) - tgt_a
            try:
                res = minimize(aobj, h_a, jac=agr, method='SLSQP',
                              bounds=[(0.0, 1.0)] * n_alt,
                              constraints=[{'type': 'eq', 'fun': aeq}],
                              options={'maxiter': 2000, 'ftol': 1e-14})
                if res.x is not None:
                    h_a = project_feasible(res.x, tgt_a)
            except Exception:
                break
        c5_a = compute_c5(h_a, dx_a)
        if c5_a < best_c5:
            best_c5 = c5_a
            best_h = h_a.copy()
            best_n = n_alt

    # Phase 7: Final SLSQP polish from best
    if tleft() > 20:
        n_f = len(best_h)
        dx_f = 2.0 / n_f
        tgt_f = n_f / 2.0
        def fobj(x):
            return compute_c5(x, dx_f)
        def feq(x):
            return np.sum(x) - tgt_f
        try:
            res = minimize(fobj, best_h, method='SLSQP',
                          bounds=[(0.0, 1.0)] * n_f,
                          constraints=[{'type': 'eq', 'fun': feq}],
                          options={'maxiter': 5000, 'ftol': 1e-16})
            if res.x is not None:
                h_r = project_feasible(res.x, tgt_f)
                c5_r = compute_c5(h_r, dx_f)
                if c5_r < best_c5:
                    best_c5 = c5_r
                    best_h = h_r.copy()
                    best_n = n_f
        except Exception:
            pass

    return (best_h, best_c5, best_n)
