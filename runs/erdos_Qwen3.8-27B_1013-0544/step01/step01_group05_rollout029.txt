def _softmax_obj(x):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full') * _g_dx
    m = np.max(corr)
    a = _g_alpha
    e = np.exp(a * (corr - m))
    s = e.sum()
    val = m + np.log(s) / a
    sh = h.sum() * _g_dx
    val += _g_lam * (sh - 1.0) + 0.5 * _g_rho * (sh - 1.0) ** 2
    return val

def _softmax_grad(x):
    h = np.clip(x, 0.0, 1.0)
    n = len(h)
    dx = _g_dx
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    a = _g_alpha
    e = np.exp(a * (corr - m))
    s = e.sum()
    w = e / s
    nh = n - 1
    grad = np.zeros(n)
    for ki in range(2 * n - 1):
        wi = w[ki]
        if wi < 1e-12:
            continue
        L = ki - nh
        if L == 0:
            g = (1.0 - 2.0 * h) * dx
        elif L > 0:
            g = np.zeros(n)
            g[:n-L] += (1.0 - h[L:]) * dx
            g[L:] -= h[:n-L] * dx
        else:
            L2 = -L
            g = np.zeros(n)
            g[L2:] += (1.0 - h[:n-L2]) * dx
            g[:n-L2] -= h[L2:] * dx
        grad += wi * g
    sh = h.sum() * dx
    grad = grad + (_g_lam * dx + _g_rho * (sh - 1.0) * dx)
    return grad

def _exact_obj(x):
    return compute_c5(np.clip(x, 0.0, 1.0), _g_dx)

def _eq_con(x):
    return np.sum(x) - _g_target_sum

def _active_set_grad(h, dx, n, top_k=12):
    corr = np.correlate(h, 1.0 - h, mode='full')
    cm = np.max(corr)
    act = np.where(corr >= cm - 1e-6)[0]
    if len(act) < 1:
        act = np.array([np.argmax(corr)])
    if len(act) > top_k:
        act = np.argsort(corr)[-top_k:]
    nh = n - 1
    g = np.zeros(n)
    for idx in act:
        L = int(idx) - nh
        if L == 0:
            gi = (1.0 - 2.0 * h) * dx
        elif L > 0:
            gi = np.zeros(n)
            gi[:n-L] += (1.0 - h[L:]) * dx
            gi[L:] -= h[:n-L] * dx
        else:
            L2 = -L
            gi = np.zeros(n)
            gi[L2:] += (1.0 - h[:n-L2]) * dx
            gi[:n-L2] -= h[L2:]*dx
        g += gi
    return g / len(act)

def _fine_gradient_descent(h, dx, ts, n, budget_check, best_c5_ref):
    lh = h.copy()
    lc = compute_c5(lh, dx)
    ss = 5e-5
    mom = np.zeros(n)
    for it in range(2000):
        if budget_check():
            break
        g = _active_set_grad(lh, dx, n, top_k=10)
        mom = 0.4 * mom - ss * g
        h_new = project_feasible(lh + mom, ts)
        c_new = compute_c5(h_new, dx)
        if c_new < lc:
            lc = c_new
            lh = h_new
        else:
            ss *= 0.93
            if ss < 1e-9:
                break
    return lh, lc

def run(seed=42, budget_s=1100, **kwargs):
    global _g_dx, _g_alpha, _g_lam, _g_rho, _g_target_sum
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except NameError:
        h_init = None

    best_h = np.full(70, 0.5)
    best_c5 = compute_c5(best_h, 2.0 / 70)
    best_n = 70

    def tleft():
        return (t0 + budget_s - 5.0) - time.time()

    n_list = [70, 80, 60, 90, 100]
    for np_ in n_list:
        if tleft() < 60.0:
            break
        dx = 2.0 / np_
        ts = np_ / 2.0
        _g_dx = dx
        _g_target_sum = ts

        inits = []
        if h_init is not None:
            if len(h_init) == np_:
                inits.append(project_feasible(h_init.copy(), ts))
            else:
                xo = np.linspace(0, 2, len(h_init), endpoint=False)
                xn = np.linspace(0, 2, np_, endpoint=False)
                inits.append(project_feasible(np.interp(xn, xo, h_init), ts))
        x = np.linspace(0, 2, np_, endpoint=False)
        inits.append(project_feasible(0.5 + 0.25 * np.cos(np.pi * x), ts))
        inits.append(project_feasible(0.5 + 0.15 * np.cos(2*np.pi*x) + 0.15*np.cos(4*np.pi*x), ts))

        for ii in range(min(len(inits), 3)):
            if tleft() < 50.0:
                break
            lh = inits[ii].copy()
            lc = compute_c5(lh, dx)

            # Phase A: Augmented Lagrangian + L-BFGS-B with soft-max
            _g_lam = 0.0
            _g_rho = 3.0
            _g_alpha = 8.0
            hc = lh.copy()
            for ai in range(30):
                if tleft() < 45.0:
                    break
                res = minimize(_softmax_obj, hc, method='L-BFGS-B',
                               jac=_softmax_grad,
                               bounds=[(0.0, 1.0)] * np_,
                               options={'maxiter': 200, 'ftol': 1e-14})
                hn = np.clip(res.x, 0.0, 1.0)
                hn = project_feasible(hn, ts)
                cn = compute_c5(hn, dx)
                if cn < lc:
                    lc = cn
                    lh = hn.copy()
                hc = hn
                v = hn.sum() * dx - 1.0
                _g_lam += _g_rho * v
                if abs(v) > 1e-5:
                    _g_rho = min(_g_rho * 2.0, 500.0)
                _g_alpha = min(_g_alpha * 1.5, 200.0)

            # Phase B: SLSQP on exact objective
            if tleft() < 20.0:
                break
            try:
                res = minimize(_exact_obj, lh, method='SLSQP',
                               bounds=[(0.0, 1.0)] * np_,
                               constraints=[{'type': 'eq', 'fun': _eq_con}],
                               options={'maxiter': 5000, 'ftol': 1e-16})
                if res.x is not None:
                    hp = project_feasible(res.x, ts)
                    cp = compute_c5(hp, dx)
                    if cp < lc:
                        lc = cp
                        lh = hp
            except Exception:
                pass

            # Phase C: Fine projected gradient with tight active set
            if tleft() > 10.0:
                lh, lc = _fine_gradient_descent(lh, dx, ts, np_,
                                               lambda: tleft() < 8.0, lc)

            # Phase D: Random restarts around best
            if tleft() > 15.0:
                for _ in range(100):
                    if tleft() < 10.0:
                        break
                    pert = lh + rng.normal(0, 0.003, np_)
                    pert = project_feasible(pert, ts)
                    cp = compute_c5(pert, dx)
                    if cp < lc:
                        lc = cp
                        lh = pert.copy()
                    # quick local refine
                    h_r = lh.copy()
                    ss_r = 1e-4
                    mom_r = np.zeros(np_)
                    for _ in range(100):
                        if tleft() < 8.0:
                            break
                        g_r = _active_set_grad(h_r, dx, np_, top_k=6)
                        mom_r = 0.5 * mom_r - ss_r * g_r
                        h_r = project_feasible(h_r + mom_r, ts)
                        cr = compute_c5(h_r, dx)
                        if cr < lc:
                            lc = cr
                            lh = h_r.copy()
                        else:
                            ss_r *= 0.9
                            if ss_r < 1e-9:
                                break

            if lc < best_c5:
                best_c5 = lc
                best_h = lh.copy()
                best_n = np_

    # Final universal polish on best solution
    if tleft() > 5.0:
        dx_f = 2.0 / best_n
        ts_f = best_n / 2.0
        _g_dx = dx_f
        _g_target_sum = ts_f
        try:
            res = minimize(_exact_obj, best_h, method='SLSQP',
                           bounds=[(0.0, 1.0)] * best_n,
                           constraints=[{'type': 'eq', 'fun': _eq_con}],
                           options={'maxiter': 3000, 'ftol': 1e-16})
            if res.x is not None:
                hp = project_feasible(res.x, ts_f)
                cp = compute_c5(hp, dx_f)
                if cp < best_c5:
                    best_c5 = cp
                    best_h = hp
        except Exception:
            pass

    return (best_h, best_c5, best_n)
