        # term2: -h[j+k] for j in [max(0,-k), min(n, n-k))
        j1_lo = max(0, k)
        j1_hi = min(n, n + k)
        if j1_lo < j1_hi:
            grad[j1_lo:j1_hi] += wk * (1.0 - h[j1_lo-k:j1_hi-k])
        j2_lo = max(0, -k)
        j2_hi = min(n, n - k)
        if j2_lo < j2_hi:
            grad[j2_lo:j2_hi] -= wk * h[j2_lo+k:j2_hi+k]
    
    return grad

Wait, I realize this O(n²) gradient computation is fine for n=120 (that's only 14400 operations per gradient, and we'll do maybe 10000 iterations, so 144 million operations total - very fast).

But actually, there's an even simpler way to think about this. The gradient of the smooth max is:
grad = dx * (w * (1-h) convolved appropriately - w * h convolved appropriately)

Where * denotes the weighted sum over shifts. This is really a convolution!

If I define W as the weight vector (length 2n-1) corresponding to shifts -(n-1) to (n-1), then:
grad[j] = dx * (∑_k W[k] * (1-h[j-k]) - ∑_k W[k] * h[j+k])

The first sum is (W * (1-h))[j] in the convolution sense (with appropriate convention).
The second sum is (W * h)[j] but with a sign flip in the index.

Actually, ∑_k W[k] * h[j+k] = ∑_k W[k] * h[j+k] = (W_flipped * h)[j] where W_flipped[k] = W[-k].

This is getting complicated. For n=120, let me just use the O(n²) approach or even simpler: use finite differences for the gradient (since n is small).

Actually, the simplest and most robust approach for small n: use central finite differences. For each coordinate j, compute f(h + ε*e_j) and f(h - ε*e_j), then gradient ≈ (f+ - f-)/(2ε). This costs 2n evaluations per gradient, each of which is O(n) for the correlation. Total: O(n²) per gradient, same as the analytical approach but simpler to implement correctly.

But wait, for n=120, that's 240 evaluations of the smooth max per gradient step, each involving a correlation of length 120 (which is O(n) = O(120)). So 240*120 = 28800 operations per gradient. For 10000 iterations, that's 288 million - still fast.

Wait, let me just test this mentally. For a=[a0,a1,a2], v=[v0,v1,v2]:
full correlation has 5 elements.
c[0] = a[0]*v[2]  (shift = -2, only a[0] and v[2] overlap)
c[1] = a[0]*v[1] + a[1]*v[2]
c[2] = a[0]*v[0] + a[1]*v[1] + a[2]*v[2]
c[3] = a[1]*v[0] + a[2]*v[1]
c[4] = a[2]*v[0]

So c[i] for i=0,...,4 corresponds to shift k = i - (len(v)-1) = i-2, giving k=-2,-1,0,1,2.

c[k+(n-1)] = ∑_n a[n+k] * v[n] where both indices are valid.

For k≥0: n ranges from 0 to n-1-k, so c = ∑_{n=0}^{n-1-k} a[n+k] * v[n]
For k<0: n ranges from -k to n-1, so c = ∑_{n=-k}^{n-1} a[n+k] * v[n]

With a=h, v=1-h:
c_{k} = ∑_{n: valid} h[n+k] * (1-h[n])

∂c_k/∂h[j]:
- h[j] appears in h[n+k] when n+k=j, i.e., n=j-k. Valid if 0≤j-k<n and 0≤n+k<n (the latter is 0≤j<n, which is given). Also need n valid: 0≤j-k<n.
  Contribution: (1-h[j-k])
- h[j] appears in (1-h[n]) when n=j. Valid if 0≤j<n (given) and 0≤n+k=n+k=j+k<n, i.e., 0≤j+k<n.
  Contribution: -h[j+k] (derivative of (1-h[n]) w.r.t. h[n] is -1, multiplied by h[n+k]=h[j+k])

So: ∂c_k/∂h[j] = (1-h[j-k])*1[0≤j-k<n] - h[j+k]*1[0≤j+k<n]

And the smooth max gradient:
∂f/∂h[j] = dx * ∑_k w_k * [(1-h[j-k])*1[0≤j-k<n] - h[j+k]*1[0≤j+k<n]]

where w_k = exp(τ*dx*c_k) / ∑_m exp(τ*dx*c_m) (softmax weights)

This is O(n²) to compute (for each of n values of j, sum over n+1 values of k).

For n=120, that's 120*239 ≈ 29000 operations. Very fast.

A simple approach: add a constant to all elements (in the interior), then re-clip. Or use a "water-filling" approach: to increase the sum, add to the elements that are strictly between 0 and 1; to decrease, subtract from those.

Actually, a cleaner approach: use a Lagrange multiplier for the sum constraint. The Lagrangian is:
L = f(h) - λ*(∑h - n/2)

The gradient becomes: grad_f - λ * 1

And we just need to find λ such that after the step and projection, the sum is correct. This can be done with a bisection on λ.

import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(obj, tau):
    x = tau * obj
    x_max = np.max(x)
    return float(x_max / tau + np.log(np.sum(np.exp(x - x_max))) / tau)

def compute_smooth_max_and_weights(corr, dx, tau):
    x = tau * corr * dx
    x_max = np.max(x)
    shifted = x - x_max
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / exp_shifted.sum()
    val = (x_max + np.log(exp_shifted.sum())) / tau
    return val, weights

def projected_gradient_step(h, grad, step_size, target_sum):
    """Take a gradient step, project to [0,1], fix sum."""
    n = len(h)
    h_new = h - step_size * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    # Fix sum
    diff = target_sum - h_new.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h_new < 0.999
        else:
            mask = h_new > 0.001
        cnt = mask.sum()
        if cnt > 0:
            h_new[mask] += diff / cnt
            h_new = np.clip(h_new, 0.0, 1.0)
    return h_new

def compute_gradient(h, weights, dx):
    """Compute gradient of smooth max objective."""
    n = len(h)
    grad = np.zeros(n)
    for idx in range(len(weights)):
        k = idx - (n - 1)
        wk = dx * weights[idx]
        if abs(wk) < 1e-15:
            continue
        # term1: (1-h[j-k]) for j where 0<=j-k<n
        j1_lo = max(0, k)
        j1_hi = min(n, n + k)
        if j1_lo < j1_hi:
            grad[j1_lo:j1_hi] += wk * (1.0 - h[j1_lo-k:j1_hi-k])
        # term2: -h[j+k] for j where 0<=j+k<n
        j2_lo = max(0, -k)
        j2_hi = min(n, n - k)
        if j2_lo < j2_hi:
            grad[j2_lo:j2_hi] -= wk * h[j2_lo+k:j2_hi+k]
    return grad

        return budget_s - (time.time() - t0)

        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)
    
    n0 = len(h_init)
    best_h = h_init.copy()
    best_c5 = evaluate_c5(best_h)
    
    target = n0 / 2.0
    best_h = np.clip(best_h, 0, 1)
    # Fix sum
    diff = target - best_h.sum()
    mask = best_h < 0.999 if diff > 0 else best_h > 0.001
    cnt = mask.sum()
    if cnt > 0:
        best_h[mask] += diff / cnt
        best_h = np.clip(best_h, 0, 1)
    best_c5 = evaluate_c5(best_h)
    
    # Main optimization loop with projected gradient
    def optimize_pg(h_start, n_pts, tau_schedule, max_iters, target):
        """Projected gradient with adaptive tau and step size."""
        h = h_start.copy()
        dx = 2.0 / n_pts
        step_size = 1.0
        best_local_c5 = evaluate_c5(h)
        
        for tau in tau_schedule:
            for iter_i in range(max_iters):
                if rem() < 10:
                    break
                # Compute objective and weights
                corr = np.correlate(h, 1.0 - h, mode="full")
                obj_val, weights = compute_smooth_max_and_weights(corr, dx, tau)
                
                # Compute gradient
                grad = compute_gradient(h, weights, dx)
                
                # Backtracking line search
                alpha = step_size
                f0 = obj_val
                for _ in range(30):
                    h_try = projected_gradient_step(h, grad, alpha, target)
                    corr_try = np.correlate(h_try, 1.0 - h_try, mode="full")
                    f_try, _ = compute_smooth_max_and_weights(corr_try, dx, tau)
                    if f_try < f0 - 1e-4 * alpha * np.dot(grad, h - h_try):
                        h = h_try
                        break
                    alpha *= 0.5
                
                # Track progress
                c5 = evaluate_c5(h)
                if c5 < best_local_c5:
                    best_local_c5 = c5
                
                # Decay step size slowly
                step_size *= 0.999
            
        return h, best_local_c5
    
    # Phase 1: Optimize at current resolution
    tau_schedule_1 = [5, 10, 20, 50, 100, 300, 500, 1000]

        hp = np.clip(hp, 0, 1)

        mask = hp < 0.999 if diff > 0 else hp > 0.001

        if cnt > 0:

            hp = np.clip(hp, 0, 1)

    best_h = np.clip(best_h, 0, 1)
    diff = len(best_h)/2.0 - best_h.sum()
    if abs(diff) > 1e-14:

λ = (∑grad - (∑h - target)/α) / n

This is a "bregman projection" onto the simplex-like constraint {h ∈ [0,1]^n : ∑h = target}.

The problem "minimize t subject to c_k ≤ t for all k, 0≤h≤1, ∑h=n/2" is a linear program! (Because c_k is linear in h for each k... wait, is it?)

This is NOT linear in h because of the h[i]*h[i-k] term! So it's a quadratic program (QP), not an LP.

But wait, if we restrict to binary h (h ∈ {0,1}), then h[i]*(1-h[i-k]) = h[i] - h[i]*h[i-k] = h[i] - h[i] (if h[i]=h[i-k]=1) or h[i] (if h[i-k]=0). So for binary h, c_k = ∑ h[i]*(1-h[i-k])*dx which IS linear! But the binary constraint makes it a QP or MIQP.

Where Q_k and p_k are matrices/vectors that define the quadratic form for each k.
