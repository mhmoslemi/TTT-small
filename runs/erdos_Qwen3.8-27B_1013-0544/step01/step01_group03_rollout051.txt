import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def compute_c5(h):
    return float(np.max(compute_corr(h)))

def smooth_max_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    x = tau * corr
    x_max = np.max(x)
    exp_vals = np.exp(x - x_max)
    sum_exp = np.sum(exp_vals)
    val = x_max + np.log(sum_exp) / tau
    weights = exp_vals / sum_exp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-12:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            m = n - lag
            if m > 0:
                grad[:m] += w * (1.0 - h[lag:])
                grad[lag:] -= w * h[:m]
        else:
            L = -lag
            m = n - L
            if m > 0:
                grad[L:] += w * (1.0 - h[:m])
                grad[:L] -= w * h[L:]
    grad *= dx
    return val, grad

def project_box_hyperplane(h, target):
    h = np.array(h, dtype=float)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        mask = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        if mask.sum() > 0:
            proj[mask] += diff / mask.sum()
    return proj

def pgd_optimize(h, tau, max_iter, lr_init, target, t0, budget_s, alpha=0.02):
    h = h.copy()
    n = len(h)
    lr = lr_init
    best_local = h.copy()
    best_val = compute_c5(h)
    for it in range(max_iter):
        rem = budget_s - (time.time() - t0)
        if rem < 10:
            break
        val, grad = smooth_max_grad(h, tau)
        h_new = h - lr * grad
        h_new = project_box_hyperplane(h_new, target)
        val_new, _ = smooth_max_grad(h_new, tau)
        if val_new > val + 1e-15:
            lr *= 0.5
            if lr < 1e-10:
                break
        else:
            lr = min(lr * (1.0 + alpha), lr_init * 2)
        h = h_new
        if it % 200 == 0 or it == max_iter - 1:
            c5_now = compute_c5(h)
            if c5_now < best_val:
                best_val = c5_now
                best_local = h.copy()
    return best_local, best_val

def subgradient_refine(h, target, max_iter, t0, budget_s):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best = h.copy()
    best_c5 = compute_c5(h)
    lr = 0.008
    stuck = 0
    for it in range(max_iter):
        rem = budget_s - (time.time() - t0)
        if rem < 10:
            break
        corr = compute_corr(h)
        k_max = int(np.argmax(corr))
        lag = k_max - (n - 1)
        g = np.zeros(n)
        if lag >= 0:
            m = n - lag
            g[:m] = (1.0 - h[lag:])
            g[lag:] += -h[:m]
        else:
            L = -lag
            m = n - L
            g[L:] = (1.0 - h[:m])
            g[:L] = -h[L:]
        g *= dx
        h_new = h - lr * g
        h_new = project_box_hyperplane(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best = h_new.copy()
            stuck = 0
        else:
            stuck += 1
        h = h_new
        if stuck > 300:
            lr *= 0.7
            stuck = 0
        if lr < 1e-6:
            break
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float).copy()
    else:
        h = np.full(70, 0.5)

    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: PGD annealing (smooth to sharp)
    taus = [100, 500, 2000, 8000, 30000]
    for tau in taus:
        rem = budget_s - (time.time() - t0)
        if rem < 40:
            break
        lr_init = n / (10.0 * tau)
        n_iter = min(30000, int(rem * 50))
        best_h, best_c5 = pgd_optimize(best_h, tau, n_iter, lr_init, target, t0, budget_s)

    # Phase 2: Subgradient refinement at exact max
    rem = budget_s - (time.time() - t0)
    if rem > 50:
        best_h, best_c5 = subgradient_refine(best_h, target, min(8000, int(rem * 30)), t0, budget_s)

    # Phase 3: Random restarts with PGD
    rem = budget_s - (time.time() - t0)
    if rem > 60:
        n_starts = min(25, int(rem / 25))
        for rs in range(n_starts):
            rem = budget_s - (time.time() - t0)
            if rem < 25:
                break
            scale = 0.03 + 0.12 * np.random.random()
            h_p = best_h + np.random.normal(0, scale, n)
            h_p = project_box_hyperplane(h_p, target)
            tau_r = 10000
            lr_r = n / (10.0 * tau_r)
            h_p, c5_p = pgd_optimize(h_p, tau_r, min(3000, int(rem * 15)), lr_r, target, t0, budget_s)
            if c5_p < best_c5:
                best_h = h_p.copy()
                best_c5 = c5_p

    # Phase 4: Try higher resolution
    rem = budget_s - (time.time() - t0)
    for tn in [160, 200, 250]:
        rem = budget_s - (time.time() - t0)
        if rem < 50:
            break
        target_t = tn / 2.0
        h_t = np.interp(np.linspace(0, 1, tn, endpoint=False),
                        np.linspace(0, 1, n, endpoint=False), best_h)
        h_t = project_box_hyperplane(h_t, target_t)
        tau_t = 8000
        lr_t = tn / (10.0 * tau_t)
        h_t, c5_t = pgd_optimize(h_t, tau_t, min(5000, int(rem * 20)), lr_t, target_t, t0, budget_s)
        if c5_t < best_c5:
            best_h = h_t.copy()
            best_c5 = c5_t
            n = tn
            target = target_t

    # Phase 5: Final subgradient polish
    rem = budget_s - (time.time() - t0)
    if rem > 30:
        best_h, best_c5 = subgradient_refine(best_h, target, min(5000, int(rem * 20)), t0, budget_s)

    # Final: also try a second-level anneal on best
    rem = budget_s - (time.time() - t0)
    if rem > 30:
        for tau in [50000, 100000]:
            rem = budget_s - (time.time() - t0)
            if rem < 15:
                break
            lr_f = n / (10.0 * tau)
            best_h, c5_f = pgd_optimize(best_h, tau, min(5000, int(rem * 30)), lr_f, target, t0, budget_s)
            if c5_f < best_c5:
                best_c5 = c5_f

    best_h = project_box_hyperplane(best_h, target)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
