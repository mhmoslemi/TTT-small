from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s > 1e-12 and abs(s - target_sum) > 1e-10:
        h += (target_sum - s) / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    shifted = beta * (x - m)
    # Numerical stability
    max_shifted = np.max(shifted)
    if max_shifted > 500:
        shifted -= max_shifted
    return m + (np.log(np.sum(np.exp(shifted))) / beta)

def smooth_max_grad(x, beta):
    """Returns (value, gradient weights) for smooth max."""
    m = np.max(x)
    shifted = beta * (x - m)
    max_shifted = np.max(shifted)
    shifted -= max_shifted
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    return m + np.log(np.sum(exp_shifted)) / beta, weights

    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)

    m = np.max(x)
    shifted = np.clip(beta * (x - m), -500, 500)
    e = np.exp(shifted)
    s = e.sum()
    return m + np.log(s) / beta, e / s

import numpy as np
from scipy.optimize import minimize, Bounds

from functools import partial

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * 2.0 / n)

def project_h(h, ts):
    h = np.clip(h, 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) > 1e-10:
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smx_w(x, beta):
    m = x.max()
    s = np.clip(beta * (x - m), -500, 500)
    e = np.exp(s)
    return m + np.log(e.sum()) / beta, e / e.sum()

def smx_grad(h, beta):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode="full")
    _, w = smx_w(c, beta)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wi = w[k]
        if wi < 1e-12:
            continue
        d = abs(k - (n - 1))
        if d == 0:
            g += wi * (1 - 2 * h)
        elif d < n:
            g[d:] += wi * (1 - h[:n - d])
            g[:n - d] -= wi * h[d:]
    return g

    return smx_w(np.correlate(x, 1.0 - x, mode="full"), beta)[0] * dx

            if best_val != old_val:
                improved = True
            if best_local < best_c5:

        if not improved:
            break
    return hb, best_c5
