def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    if beta == 0:
        return m
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) > 1e-12 and s > 0:
        h += (target_sum - s) / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def binary_local_search(h_bin, max_iters=1000):
    """Local search on binary sequence via swaps"""
    n = len(h_bin)
    best = h_bin.copy().astype(float)
    best_c5 = compute_c5(best)

        improved = False
        ones = np.where(best == 1)[0]
        zeros = np.where(best == 0)[0]

        best_swap_val = best_c5

                candidate[i] = 0
                candidate[j] = 1

            best[best_swap[1]] = 1
            best_c5 = best_swap_val
            improved = True

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-10:
        return h
    # Bisection for the right shift
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        shifted = np.clip(h - mid, 0.0, 1.0)
        if shifted.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0.0, 1.0)

def binary_local_search(h_bin, max_iters=500, time_limit=None, start_time=None):
    n = len(h_bin)
    best = h_bin.copy().astype(float)
    best_c5 = compute_c5(best)
    
    for it in range(max_iters):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        
        ones = np.where(best.astype(int) == 1)[0]
        zeros = np.where(best.astype(int) == 0)[0]
        
        best_val = best_c5
        best_swap = None
        
        for i in ones:
            for j in zeros:
                if i == j:
                    continue
                # Swap
                old_i, old_j = best[i], best[j]
                best[i], best[j] = old_j, old_i
                c5 = compute_c5(best)
                best[i], best[j] = old_i, old_j
                if c5 < best_val - 1e-12:
                    best_val = c5
                    best_swap = (i, j)
        
        if best_swap is not None:
            i, j = best_swap
            best[i], best[j] = best[j], best[i]
            best_c5 = best_val
        else:
            break
    
    return best, best_c5

def continuous_optimize(h0, n_points, target_sum, max_iter=500, tol=1e-12):
    dx = 2.0 / n_points
    bounds = Bounds(np.zeros(n_points), np.ones(n_points))
    
    def con_fun(x):
        return np.sum(x) - target_sum
    
    constraints = [{'type': 'eq', 'fun': con_fun, 'jac': lambda x: np.ones(n_points)}]
    
    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        return smooth_max(corr, 150.0) * dx
    
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': max_iter, 'ftol': tol})
        h_new = project_h(res.x, target_sum)
        return h_new, compute_c5(h_new)
    except:
        return h0, compute_c5(h0)

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def project_h(h, target_sum):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        shifted = np.clip(h - mid, 0.0, 1.0)
        if shifted.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def min_max_linear(a, b, lo, hi):
    if hi - lo < 1e-14:
        t = lo
        return t, float(np.max(a * t + b))
    for _ in range(50):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = float(np.max(a * m1 + b))
        f2 = float(np.max(a * m2 + b))
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    t = (lo + hi) / 2.0
    return t, float(np.max(a * t + b))

def twocoord_optimize(h, n_points, target_sum, max_passes=50, deadline=None):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    n = n_points
    dx = 2.0 / n
    for p in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        perm = np.random.permutation(n)
        for idx_i in range(n):
            for idx_j in range(idx_i + 1, n):
                i, j = perm[idx_i], perm[idx_j]
                s = best_h[i] + best_h[j]
                lo = max(0.0, s - 1.0)
                hi = min(1.0, s)
                if hi - lo < 1e-12:
                    continue
                delta = 0.25 * (hi - lo)
                t1 = lo + delta
                t2 = lo + 2.0 * delta
                if t2 > hi:
                    t2 = hi
                    t1 = hi - delta
                if t2 - t1 < 1e-14:
                    t1, t2 = lo + 1e-6, hi - 1e-6
                old_i, old_j = best_h[i], best_h[j]
                best_h[i], best_h[j] = t1, s - t1
                c1 = compute_corr(best_h)
                best_h[i], best_h[j] = t2, s - t2
                c2 = compute_corr(best_h)
                best_h[i], best_h[j] = old_i, old_j
                a = (c2 - c1) / (t2 - t1)
                b = c1 - a * t1
                t_opt, val = min_max_linear(a, b, lo, hi)
                new_c5 = val * dx
                if new_c5 < best_c5 - 1e-13:
                    best_h[i] = t_opt
                    best_h[j] = s - t_opt
                    best_c5 = new_c5
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def binary_search(h_bin, max_iters=200, deadline=None):
    n = len(h_bin)
    best = h_bin.copy()
    best_c5 = compute_c5(best)
    for it in range(max_iters):
        if deadline is not None and time.time() > deadline:
            break
        ones = np.where(best == 1)[0]
        zeros = np.where(best == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_val = best_c5
        best_swap = None
        for i in ones:
            for j in zeros:
                best[i], best[j] = best[j], best[i]
                val = compute_c5(best)
                best[i], best[j] = best[j], best[i]

                    best_val = val
                    best_swap = (i, j)

            best[best_swap[0]], best[best_swap[1]] = best[best_swap[1]], best[best_swap[0]]
            best_c5 = best_val
        else:
            break
    return best, best_c5

def obj_func(x, beta, dx):
    return smooth_max(compute_corr(x), beta) * dx

def cons_fun(x, target_sum):
    return x.sum() - target_sum

def cons_jac(x, target_sum):
    return np.ones(len(x))

def slsqp_optimize(h0, n_points, target_sum, beta=200, maxiter=2000, ftol=1e-13):
    dx = 2.0 / n_points
    bounds = Bounds(np.zeros(n_points), np.ones(n_points))
    constraints = [{'type': 'eq', 'fun': cons_fun, 'args': (target_sum,),
                    'jac': cons_jac, 'args_jac': (target_sum,)}]
    try:
        res = minimize(obj_func, h0, args=(beta, dx), method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_new = project_h(res.x, target_sum)
        return h_new, compute_c5(h_new)
    except Exception:
        return h0.copy(), compute_c5(h0)

def subgradient_step(h, target_sum, lr):
    n = len(h)
    corr = compute_corr(h)
    k_star = int(np.argmax(corr))
    lag = k_star - (n - 1)
    grad = np.zeros(n)
    for i in range(n):
        i_plus = i + lag
        i_minus = i - lag
        if 0 <= i_plus < n:
            grad[i] += -(1.0 - h[i_plus])
        if 0 <= i_minus < n:
            grad[i] += h[i_minus]
    h_new = h - lr * grad
    h_new = project_h(h_new, target_sum)
    return h_new
