            h_pert = project_feasible(h_pert, target)
            res = minimize(_slsqp_obj, h_pert, method='SLSQP',
                          bounds=[(0.0, 1.0)] * n,
                          constraints=[{'type': 'eq', 'fun': _slsqp_con}],
                          options={'maxiter': 2000, 'ftol': 1e-15})
            if res.x is not None:
                h_ref = project_feasible(res.x, target)
                c5_ref = compute_c5(h_ref, dx)
                if c5_ref < best_c5:
                    best_c5 = c5_ref
                    best_h = h_ref
        except Exception:
            pass

    # Phase 4: Fine LSE refinement
    h = best_h.copy()
    bh, bc = fine_refine(h, dx, target, rng, 40.0)
    if bc < best_c5:
        best_c5 = bc
        best_h = bh

    # Phase 5: Final random perturbations + quick refine
    if tleft() > 10.0:
        for _ in range(50):
            if tleft() < 3.0:
                break
            h_p = best_h + rng.normal(0, 0.003, n)
            h_p = project_feasible(h_p, target)
            c5_p = compute_c5(h_p, dx)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_p.copy()
                bh, bc = fine_refine(h_p, dx, target, rng, 5.0)
                if bc < best_c5:
                    best_c5 = bc
                    best_h = bh

    return (best_h, best_c5, n)
