def project_feasible(x, target_sum):
    lo = float(x.min()) - 1.0
    hi = float(x.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = float(np.clip(x - mid, 0.0, 1.0).sum())
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def gradient_for_lag(h, L):
    n = len(h)
    grad = np.zeros(n)
    if L == 0:
        grad[:] = (1.0 - h) - h
    elif L > 0:
        omh = 1.0 - h
        grad[L:] += omh[:n - L]
        grad[:n - L] -= h[L:]
    else:
        L2 = -L
        omh = 1.0 - h
        grad[:n - L2] += omh[L2:]
        grad[L2:] -= h[:n - L2]
    return grad

def soft_max_gradient(h, beta):
    n = len(h)
    n_half = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = (corr - c_max) * beta
    shifted = np.clip(shifted, -500.0, 0.0)
    weights = np.exp(shifted)
    wsum = float(weights.sum())
    if wsum < 1e-300:
        weights = np.zeros_like(corr)
        weights[int(np.argmax(corr))] = 1.0
        wsum = 1.0
    weights /= wsum
    grad = np.zeros(n)
    active = np.where(weights > 1e-8)[0]
    for idx in active:
        L = int(idx) - n_half
        w = weights[idx]
        if L == 0:
            grad[:] += w * ((1.0 - h) - h)
        elif L > 0:

            grad[L:] += w * omh[:n - L]

        grad += gradient_for_lag(h, L)
