def project_feasible(x, target_sum):
    lo = float(x.min()) - 1.5
    hi = float(x.max()) + 1.5
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = float(np.clip(x - mid, 0.0, 1.0).sum())
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def gradient_at_lag(h, L):
    n = len(h)
    g = np.zeros(n)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += (1.0 - h[:n-L])
        g[:n-L] -= h[L:]
    else:
        L2 = -L
        g[:n-L2] += (1.0 - h[L2:])
        g[L2:] -= h[:n-L2]
    return g

def softmax_obj_grad(h, dx, tau):
    n = len(h)
    n_half = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_vals = corr * dx
    c_max = float(np.max(c_vals))

    w = np.exp(tau * shifted)

        w[int(np.argmax(c_vals))] = 1.0
        w_sum = 1.0
