from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target):
    h = np.array(h, dtype=float).copy()
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return np.clip(h, 0.0, 1.0)
    if diff > 0:
        mask = h < 0.999
    else:
        mask = h > 0.001
    cnt = mask.sum()
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(vals, tau):
    x = tau * vals
    xm = np.max(x)
    return float(xm / tau + np.log(np.sum(np.exp(x - xm))) / tau)

def softmin_val(vals, theta):

    return -smooth_max_val(-vals, 1.0/theta)

    return np.concatenate([x, x[::-1]])
