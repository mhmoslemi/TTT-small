def project_onto_feasible(h, target):
    """Project h onto {x : 0 ≤ x ≤ 1, sum(x) = target}"""
    # First, find lambda such that sum(clip(h - lambda, 0, 1)) = target
    # f(lambda) = sum(clip(h[i] - lambda, 0, 1)) is continuous, non-increasing in lambda
    # f(-inf) = n, f(+inf) = 0
    
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2, 0.0, 1.0)
    # Fix any tiny numerical error
    diff = target - proj.sum()
    if abs(diff) > 1e-12:

        mask = (proj > 1e-10) & (proj < 1 - 1e-10)

from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_vals(vals, tau):
    x = tau * np.asarray(vals)
    x_max = np.max(x)
    return x_max / tau + np.log(np.sum(np.exp(x - x_max))) / tau

def project_sum(h, target):
    """Project h onto {x: 0<=x<=1, sum(x)=target} using bisection on lambda"""
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - proj.sum()
    if abs(diff) > 1e-10:
        mask = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        cnt = mask.sum()
        if cnt > 0:
            proj[mask] += diff / cnt
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def subgrad_objective(h):
    """Return (c5_max, subgradient) for the max C_k"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    c5_vals = corr * dx
    k_star = int(np.argmax(c5_vals))
    lag = k_star - (n - 1)
    idx = np.arange(n)
    g = np.zeros(n)
    if lag >= 0:
        m1 = idx >= lag
        m2 = idx < (n - lag)
        g[m1] += (1.0 - h[idx[m1] - lag]) * dx
        g[m2] -= h[idx[m2] + lag] * dx
    else:
        l = -lag
        m1 = idx < (n - l)
        m2 = idx >= l
        g[m1] += (1.0 - h[idx[m1] + l]) * dx
        g[m2] -= h[idx[m2] - l] * dx
    return c5_vals[k_star], g

def lmbfgs_optimize(h0, n, tau, penalty, maxiter):
    """Optimize using L-BFGS-B with smooth max + penalty"""
    target = n / 2.0

        sm = smooth_max_vals(corr, tau)
        s = x.sum() - target
        return sm + penalty * s * s

    res = minimize(objective, h0, method='L-BFGS-B', bounds=bnds,
                   options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8})
    x = project_sum(res.x, target)
    c5 = evaluate_c5(x)
    return x, c5

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_vals(vals, tau):
    x = tau * np.asarray(vals, dtype=float)
    x_max = np.max(x)
    return float(x_max / tau + np.log(np.sum(np.exp(x - x_max))) / tau)

def project_sum(h, target):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - proj.sum()
    if abs(diff) > 1e-10:
        mask = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        cnt = mask.sum()
        if cnt > 0:
            proj[mask] += diff / cnt
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def make_objective(n, tau, penalty):
    target = n / 2.0
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full") * dx
        sm = smooth_max_vals(corr, tau)
        s = x.sum() - target
        return sm + penalty * s * s
    return obj

def lmbfgs_optimize(h0, tau, penalty, maxiter=500):
    n = len(h0)
    target = n / 2.0
    dx = 2.0 / n
    def objective(x):
        corr = np.correlate(x, 1.0 - x, mode="full") * dx
        sm = smooth_max_vals(corr, tau)
        s = x.sum() - target
        return sm + penalty * s * s
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(objective, h0, method='L-BFGS-B', bounds=bnds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
        x = project_sum(res.x, target)
        c5 = evaluate_c5(x)
        return x, c5
    except Exception:
        return h0, evaluate_c5(h0)

def subgrad_step(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    c5_vals = corr * dx
    k_star = int(np.argmax(c5_vals))
    lag = k_star - (n - 1)
    idx = np.arange(n)
    g = np.zeros(n)
    if lag >= 0:
        m1 = idx >= lag
        m2 = idx < (n - lag)
        if m1.any():
            g[m1] += (1.0 - h[idx[m1] - lag]) * dx
        if m2.any():
            g[m2] -= h[idx[m2] + lag] * dx
    else:
        l = -lag
        m1 = idx < (n - l)
        m2 = idx >= l
        if m1.any():
            g[m1] += (1.0 - h[idx[m1] + l]) * dx
        if m2.any():
            g[m2] -= h[idx[m2] - l] * dx
    return c5_vals[k_star], g

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)

    n0 = len(h_init)
    best_h = h_init.copy()
    best_c5 = evaluate_c5(best_h)

    def rem():
        return budget_s - (time.time() - t0)

    # Phase 1: Short subgradient exploration
    h_sg = best_h.copy()
    target = n0 / 2.0
    step = 0.08
    for it in range(2000):
        if rem() < 500:
            break
        c5_k, g = subgrad_step(h_sg)
        h_sg = h_sg - step * g
        h_sg = project_sum(h_sg, target)
        step *= 0.9992
        if it % 200 == 0:
            fc = evaluate_c5(h_sg)
            if fc < best_c5:
                best_c5 = fc
                best_h = h_sg.copy()

    # Phase 2: L-BFGS-B polish from best with increasing tau
    if rem() > 300:
        for tau in [10, 30, 100, 300, 1000, 3000, 10000]:
            if rem() < 50:

    # Phase 3: Multi-resolution optimization

            elif mode == 3:

            hp = project_sum(hp, target_h)
            for tau in [50, 500, 2000]:

    hn_final = len(best_h)
    best_h = project_sum(best_h, hn_final / 2.0)
    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, hn_final
