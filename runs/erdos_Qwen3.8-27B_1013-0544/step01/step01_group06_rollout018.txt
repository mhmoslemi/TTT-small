def compute_corr(h):
    return np.correlate(h, 1 - h, mode='full')

def compute_c5(h):
    n = len(h)
    return np.max(compute_corr(h)) * (2.0 / n)

def grad_at_shift(h, k):
    """Gradient of overlap at shift k w.r.t. h"""
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        # O(k) = sum_{i=0}^{n-k-1} h[i]*(1-h[i+k])
        # d/dh[i] = (1-h[i+k]) for i < n-k, minus h[i-k] for i >= k
        g[:n-k] += (1.0 - h[k:])
        if k > 0:
            g[k:] -= h[:n-k]
    else:
        p = -k
        # O(-p) = sum_{i=p}^{n-1} h[i]*(1-h[i-p])
        # d/dh[j] = (1-h[j-p]) for j >= p, minus h[j+p] for j < n-p
        g[p:] += (1.0 - h[:-p])
        g[:n-p] -= h[p:]
    return g

def smooth_max_grad(h, alpha):
    """Gradient using smooth max approximation"""
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    s = np.sum(w)
    if s < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= s
    
    g = np.zeros(n)
    # Only compute for significant weights to save time
    active = np.where(w > 1e-8)[0]
    for idx in active:
        k = idx - (n - 1)
        g += w[idx] * grad_at_shift(h, k)
    return g * (2.0 / n)

def project_box_sum(h, target_sum):
    """Project h onto {x : 0 <= x[i] <= 1, sum(x) = target_sum}"""
    lo, hi = -2.0, 2.0

        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

    n = len(h)
    corr = compute_corr(h)
    top_idx = np.argsort(corr)[-k_count:]

    for idx in top_idx:
        k = idx - (n - 1)
        g += grad_at_shift(h, k)
    return g / k_count * (2.0 / n)

def compute_corr(h):
    return np.correlate(h, 1 - h, mode='full')

def compute_c5(h):
    n = len(h)
    return float(np.max(compute_corr(h)) * (2.0 / n))

def grad_at_shift(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        if k < n:
            g[:n-k] += (1.0 - h[k:])
            if k > 0:
                g[k:] -= h[:n-k]
    else:
        p = -k
        if p < n:
            g[p:] += (1.0 - h[:-p])
            g[:n-p] -= h[p:]
    return g

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    s = np.sum(w)
    if s < 1e-300:
        idx = np.argmax(corr)
        return grad_at_shift(h, idx - (n-1)) * (2.0/n)
    w /= s
    g = np.zeros(n)
    active = np.where(w > 1e-6)[0]
    for idx in active:

        g += w[idx] * grad_at_shift(h, k)

The product h'[i] * (1 - h'[j]) = (h[i]+t) * (1 - h[j] + t)
= h[i](1-h[j]) + h[i]*t + t(1-h[j]) + t^2
= h[i](1-h[j]) + t(h[i] + 1 - h[j]) + t^2

The product h'[j] * (1 - h'[i]) = (h[j]-t) * (1 - h[i] - t)
= h[j](1-h[i]) - h[j]*t - t(1-h[i]) + t^2
= h[j](1-h[i]) - t(h[j] + 1 - h[i]) + t^2

If these are both terms in O(k) (which they are when k = j-i and both i and j are valid), the total t-dependent part is:
t(h[i] + 1 - h[j]) + t^2 - t(h[j] + 1 - h[i]) + t^2
= t(2h[i] - 2h[j]) + 2t^2
= 2(h[i]-h[j])t + 2t^2

So yes, the quadratic coefficient is +2, making O_{j-i} convex in t.

For other shifts k (where only one of i,j is in the valid range), O_k is linear in t.

Max of convex functions (quadratics with positive leading coeff) and linear functions is convex. So C5(t) IS convex! Golden section search will work.

def compute_corr(h):
    return np.correlate(h, 1 - h, mode='full')

def compute_c5(h):
    n = len(h)
    return float(np.max(compute_corr(h)) * (2.0 / n))

def grad_at_shift(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0 and k < n:
        g[:n-k] += (1.0 - h[k:])
        if k > 0:
            g[k:] -= h[:n-k]
    elif k < 0:
        p = -k
        if p < n:
            g[p:] += (1.0 - h[:-p])
            g[:n-p] -= h[p:]
    return g

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    s = np.sum(w)
    if s < 1e-300:
        idx = np.argmax(corr)
        return grad_at_shift(h, idx - (n-1)) * (2.0/n)
    w /= s
    g = np.zeros(n)
    active = np.where(w > 1e-6)[0]
    for idx in active:
        k = int(idx) - (n - 1)
        g += w[idx] * grad_at_shift(h, k)
    return g * (2.0 / n)

def project_box_sum(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

    h = project_box_sum(h0.astype(float).copy(), target_sum)
    v = np.zeros(n)

    lr = 0.005

    patience = 0
    stalls = 0

        v = beta * v - lr * g
        h_new = project_box_sum(h + v, target_sum)

            h = h_new
            best_h = h.copy()
            best_c5 = c5_new
            patience = 0

            v *= 0.7

                    h = best_h.copy()
                else:
                    h = best_h.copy()
                alpha = min(alpha * 1.3, 400.0)
                lr = 0.005
                v = np.zeros(n)

                    break
    return best_h, best_c5
