def grad_corr_single(h, k):

            h_r = rng.uniform(0.3, 0.7, size=n_try)
            h_r = project_onto_simplex_box(h_r, n_try / 2.0)
            starts.append((h_r, n_try))

    # Structured: half-1 half-0 with noise
    for n_try in [40, 60]:
        h_s = np.concatenate([np.ones(n_try // 2), np.zeros(n_try - n_try // 2)])
        h_s += rng.randn(n_try) * 0.1
        h_s = project_onto_simplex_box(h_s, n_try / 2.0)
        starts.append((h_s, n_try))

    # Sawtooth pattern
    for n_try in [50, 60]:
        x = np.linspace(0, 1, n_try, endpoint=False)
        h_s = 0.5 + 0.3 * np.sin(2 * np.pi * x * 3)
        h_s = project_onto_simplex_box(h_s, n_try / 2.0)
        starts.append((h_s, n_try))

    n_starts = len(starts)
    per_start_budget = (phase1_deadline - t0) / n_starts

    for idx, (h_start, n_s) in enumerate(starts):
        if time.time() > phase1_deadline:
            break
        sub_deadline = min(phase1_deadline, time.time() + per_start_budget)
        sa_deadline = time.time() + per_start_budget * 0.5

        # SA phase
        h_sa, c5_sa = simulated_annealing(h_start, n_s / 2.0, sa_deadline, rng,
                                          t_start=0.2, t_end=0.03)
        # PGD phase
        h_opt, c5_opt = pgd_refine(h_sa, n_s / 2.0, sub_deadline, rng,
                                   alpha_init=80, lr_init=0.01)
        # Brief CD
        cd_deadline = min(sub_deadline, time.time() + 5)
        if time.time() < cd_deadline:
            h_cd, c5_cd = coordinate_descent(h_opt, n_s / 2.0, cd_deadline, rng, n_sweeps=10)
            if c5_cd < c5_opt:
                h_opt, c5_opt = h_cd, c5_cd

        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = n_s

    # Phase 2: Multi-resolution refinement
    remaining = budget_s - (time.time() - t0)
    if remaining < 20:
        if best_h is not None:
            best_h = project_onto_simplex_box(best_h, len(best_h) / 2.0)
            best_c5 = compute_c5(best_h)
        return best_h, best_c5, len(best_h)

    if best_n <= 60:
        target_resolutions = [100, 150, 200]
    elif best_n <= 100:
        target_resolutions = [150, 200]
    elif best_n <= 150:
        target_resolutions = [200]
    else:
        target_resolutions = []

    for n_new in target_resolutions:
        if time.time() > t0 + budget_s - 25:
            break
        h_interp = interpolate_to_n(best_h, best_n, n_new)
        h_interp = project_onto_simplex_box(h_interp, n_new / 2.0)

        # Also try direct PGD from parent at this resolution
        candidates = [h_interp]
        if h_parent is not None and n_new != n_parent:
            h_pi = interpolate_to_n(h_parent, n_parent, n_new)
            h_pi = project_onto_simplex_box(h_pi, n_new / 2.0)
            candidates.append(h_pi)
        elif h_parent is not None and n_new == n_parent:
            candidates.append(h_parent.copy())

        for h_cand in candidates:
            if time.time() > t0 + budget_s - 20:
                break
            phase_dl = min(t0 + budget_s - 15, time.time() + remaining * 0.4)
            h_opt, c5_opt = pgd_refine(h_cand, n_new / 2.0, phase_dl, rng,
                                       alpha_init=100, lr_init=0.008)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                best_n = n_new
        remaining = budget_s - (time.time() - t0)

    # Phase 3: Final polish with high alpha
    remaining = budget_s - (time.time() - t0)
    if remaining > 10 and best_h is not None:
        final_dl = t0 + budget_s - 3
        best_h, best_c5 = pgd_refine(best_h, len(best_h) / 2.0, final_dl, rng,
                                     alpha_init=250, lr_init=0.004)

    best_h = project_onto_simplex_box(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
