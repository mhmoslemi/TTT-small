        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_step(h, n_pts, n_active=5, step_size=0.005):
    """One step of subgradient descent on max_k f_k(h)"""
    dx = 2.0 / n_pts
    corr = np.correlate(h, 1.0 - h, mode="full")
    # Get top k indices
    top_idx = np.argsort(corr)[-n_active:]
    # Compute subgradient (average of gradients of active f_k)
    grad = np.zeros(n_pts)
    for k_idx in top_idx:
        k = k_idx - (n_pts - 1)
        for j in range(n_pts):
            # ∂f_k/∂h[j] = dx * [(1-h[j+k]) if 0<=j+k<n else 0 - h[j-k] if 0<=j-k<n else 0]
            val = 0.0
            if 0 <= j + k < n_pts:
                val += (1.0 - h[j + k])
            if 0 <= j - k < n_pts:
                val -= h[j - k]
            grad[j] += val * dx
    grad /= n_active
    # Project gradient step
    h_new = h - step_size * grad
    return h_new

        # Build starting point
        if init_vals is not None:
            if n_points == len(init_vals):
                h_start = init_vals.copy()
            else:
                h_start = np.interp(np.linspace(0, 2, n_points),
                                    np.linspace(0, 2, len(init_vals)), init_vals)
        else:
            h_start = np.full(n_points, 0.5)
        h_start = project_h(h_start, target_sum)

        # Phase 1: Multi-stage SLSQP
        h_cur = h_start.copy()
        for beta_stage in [50, 120, 300, 800]:
            if time.time() - start_time > budget_s - 20:
                break
            maxiter = 3000 if n_points <= 80 else (2000 if n_points <= 120 else 1000)
            try:
                res = minimize(make_obj(beta_stage), h_cur, method='SLSQP',
                               bounds=bounds, constraints=constraints,
                               options={'maxiter': maxiter, 'ftol': 1e-14})
                h_new = project_h(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5
                    best_n = n_points
                h_cur = h_new.copy()
            except Exception:
                pass

        # Phase 2: Subgradient equalization
        if time.time() - start_time < budget_s - 15:
            h_sg = best_h.copy() if best_n == n_points else h_cur.copy()
            for it in range(800):
                if time.time() - start_time > budget_s - 12:
                    break
                corr = compute_corr(h_sg)
                max_val = corr.max()
                threshold = max_val - 0.005 / dx
                active = np.where(corr >= threshold)[0]
                if len(active) < 3:
                    active = np.argsort(corr)[-8:]
                grad = sg_subgradient(h_sg, n_points, active)
                step = 0.003 * max(0.1, 1.0 - it / 600.0)
                h_try = h_sg - step * grad
                h_try = project_h(h_try, target_sum)
                c5_try = compute_c5(h_try)
                h_sg = h_try
                if c5_try < best_c5:
                    best_h = h_try.copy()
                    best_c5 = c5_try
                    best_n = n_points

        # Phase 3: Coordinate perturbation
        if time.time() - start_time < budget_s - 8:
            h_c = best_h.copy() if best_n == n_points else h_cur.copy()
            deltas = [0.015, -0.015, 0.008, -0.008, 0.025, -0.025, 0.004, -0.004]
            it = 0
            while time.time() - start_time < budget_s - 8:
                j = it % n_points
                for d in deltas:
                    h_try = h_c.copy()
                    h_try[j] = np.clip(h_try[j] + d, 0.0, 1.0)
                    h_try = project_h(h_try, target_sum)
                    c5_try = compute_c5(h_try)
                    if c5_try < best_c5:
                        best_h = h_try.copy()
                        best_c5 = c5_try
                        best_n = n_points
                        h_c = h_try.copy()
                        break
                it += 1
                if it > n_points * 4:
                    break

        # Phase 4: Random restarts with SLSQP
        n_restarts = 5 if n_points <= 100 else 3
        for _ in range(n_restarts):
            if time.time() - start_time > budget_s - 12:
                break
            if best_n == n_points and best_h is not None:
                h0 = best_h.copy() + np.random.randn(n_points) * 0.06
            else:
                h0 = np.full(n_points, 0.5) + np.random.randn(n_points) * 0.25
            h0 = project_h(h0, target_sum)
            try:
                res = minimize(make_obj(400), h0, method='SLSQP',
                               bounds=bounds, constraints=constraints,
                               options={'maxiter': 1500, 'ftol': 1e-13})
                h_new = project_h(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5
                    best_n = n_points
            except Exception:
                pass

    # Final ultra-precise polish
    if time.time() - start_time < budget_s - 5 and best_h is not None:
        n_pts = best_n
        ts2 = n_pts / 2.0
        bds2 = Bounds(np.zeros(n_pts), np.ones(n_pts))
        cts2 = [{'type': 'eq', 'fun': lambda x: np.sum(x) - ts2,
                'jac': lambda x: np.ones(n_pts)}]
        try:
            def obj_final(h):
                corr = compute_corr(h)
                return smooth_max(corr, 3000.0) * (2.0 / n_pts)
            res = minimize(obj_final, best_h, method='SLSQP', bounds=bds2,
                           constraints=cts2, options={'maxiter': 10000, 'ftol': 1e-16})
            h_new = project_h(res.x, ts2)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new.copy(), c5
        except Exception:
            pass

    if best_h is None:
        best_h = np.full(70, 0.5)
        best_c5 = compute_c5(best_h)
        best_n = 70

    return best_h, best_c5, best_n
