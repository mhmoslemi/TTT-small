            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new.copy(), c5
        except Exception:
            pass

    # Phase 2: Nelder-Mead on true max (no smooth approximation)
    if remaining() > 40:
        bounds = Bounds(np.zeros(n_points), np.ones(n_points))
        try:
            res = minimize(partial(obj_true, n=n_points), best_h,
                         method='Nelder-Mead', bounds=bounds,
                         options={'maxiter': 8000, 'xatol': 5e-9, 'fatol': 1e-14,
                                  'maxfev': 40000, 'adaptive': True})
            h_new = project_h(res.x, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new.copy(), c5
        except Exception:
            pass

    # Phase 3: Penalty Nelder-Mead (robust to constraint issues)
    for mu in [5.0, 30.0, 100.0, 500.0]:
        if remaining() < 20:
            break
        bounds = Bounds(np.zeros(n_points), np.ones(n_points))
        try:
            res = minimize(partial(obj_penalty, beta=300, mu=mu, n=n_points),
                         best_h, method='Nelder-Mead', bounds=bounds,
                         options={'maxiter': 6000, 'xatol': 5e-9, 'fatol': 1e-14,
                                  'maxfev': 25000, 'adaptive': True})
            h_new = project_h(res.x, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new.copy(), c5
        except Exception:
            pass

    # Phase 4: Diverse restarts
    no_improve = 0
    attempt = 0
    while remaining() > 50 and no_improve < 60:
        attempt += 1
        h0 = best_h.copy()
        strategy = attempt % 8

        if strategy == 0:
            h0 += np.random.randn(n_points) * (0.004 + 0.05 * np.exp(-attempt / 40.0))
        elif strategy == 1:
            n_p = max(3, n_points // 4)
            idx = np.random.choice(n_points, n_p, replace=False)
            h0[idx] += np.random.randn(n_p) * 0.07
        elif strategy == 2:
            i, j = np.random.choice(n_points, 2, replace=False)
            h0[i], h0[j] = h0[j], h0[i]
            h0 += np.random.randn(n_points) * 0.002
        elif strategy == 3:
            n_flip = np.random.randint(2, max(3, n_points // 15))
            idx = np.random.choice(n_points, n_flip, replace=False)
            for ii in idx:
                h0[ii] = 1.0 - h0[ii]
        elif strategy == 4:
            corr = np.correlate(h0, 1.0 - h0, mode="full")
            top_k = np.argsort(corr)[-4:]
            for k_idx in top_k:
                lag = k_idx - (n_points - 1)
                for j in range(n_points):
                    ii = j - lag
                    if 0 <= ii < n_points:
                        h0[j] -= 0.004 * (1.0 - h0[ii])
                    ii = j + lag
                    if 0 <= ii < n_points:
                        h0[j] -= 0.004 * h0[ii]
        elif strategy == 5:
            freq = np.random.uniform(0.5, 10.0)
            phase = np.random.uniform(0, 2 * np.pi)
            amp = np.random.uniform(0.003, 0.015)
            x = np.arange(n_points) * 2.0 * np.pi / n_points
            h0 += amp * np.sin(freq * x + phase)
        elif strategy == 6:
            blen = np.random.randint(2, max(3, n_points // 10))
            b1 = np.random.randint(0, n_points - blen)
            b2 = np.random.randint(0, n_points - blen)
            if abs(b1 - b2) >= blen:
                tmp = h0[b1:b1 + blen].copy()
                h0[b1:b1 + blen] = h0[b2:b2 + blen]
                h0[b2:b2 + blen] = tmp
        else:
            n_res = np.random.randint(2, max(3, n_points // 6))
            idx = np.random.choice(n_points, n_res, replace=False)
            h0[idx] = np.random.random(n_res)

        h0 = project_h(h0, target_sum)
        bounds = Bounds(np.zeros(n_points), np.ones(n_points))
        cons = [{'type': 'eq', 'fun': partial(sum_con_fun, target=target_sum),
                 'jac': partial(sum_con_jac, target=target_sum)}]

        # Try SLSQP first
        try:
            res = minimize(partial(obj_smooth, beta=300, n=n_points), h0,
                         method='SLSQP', bounds=bounds, constraints=cons,
                         options={'maxiter': 600, 'ftol': 1e-14})
            h_new = project_h(res.x, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new.copy(), c5
                no_improve = 0
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

        # Occasionally try Nelder-Mead from the perturbed point
        if attempt % 7 == 0 and remaining() > 60:
            try:
                res = minimize(partial(obj_true, n=n_points), h0,
                             method='Nelder-Mead', bounds=bounds,
                             options={'maxiter': 5000, 'xatol': 5e-9,
                                      'fatol': 1e-14, 'maxfev': 20000})
                h_new = project_h(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h, best_c5 = h_new.copy(), c5
                    no_improve = 0
            except Exception:
                pass

    # Phase 5: Multi-resolution
    for n_try in [98, 140, 196]:
        if remaining() < 150:
            break
        x_old = np.linspace(0, 2, n_points, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_ext = np.concatenate([best_h[-1:], best_h, best_h[:1]])
        x_ext = np.concatenate([[-2.0 / n_points], x_old, [2.0 + 2.0 / n_points]])
        h_interp = np.interp(x_new, x_ext, h_ext)
        h_interp = np.clip(h_interp, 0, 1)
        ts_new = n_try / 2.0
        h_interp = project_h(h_interp, ts_new)

        c5_try = compute_c5(h_interp)
        if c5_try < best_c5:
            best_h = h_interp.copy()
            best_c5 = c5_try
            n_points = n_try
            target_sum = ts_new

        if remaining() > 80:
            bounds = Bounds(np.zeros(n_points), np.ones(n_points))
            cons = [{'type': 'eq', 'fun': partial(sum_con_fun, target=target_sum),
                     'jac': partial(sum_con_jac, target=target_sum)}]
            for beta in [80, 200, 500]:
                if remaining() < 30:
                    break
                try:
                    res = minimize(partial(obj_smooth, beta=beta, n=n_points), best_h,
                                 method='SLSQP', bounds=bounds, constraints=cons,
                                 options={'maxiter': 2000, 'ftol': 1e-15})
                    h_new = project_h(res.x, target_sum)
                    c5_new = compute_c5(h_new)
                    if c5_new < best_c5:
                        best_h, best_c5 = h_new.copy(), c5_new
                except Exception:
                    pass

    # Phase 6: Final high-precision polish
    for beta in [800, 1500, 3000]:
        if remaining() < 12:
            break
        bounds = Bounds(np.zeros(n_points), np.ones(n_points))
        cons = [{'type': 'eq', 'fun': partial(sum_con_fun, target=target_sum),
                 'jac': partial(sum_con_jac, target=target_sum)}]
        try:
            res = minimize(partial(obj_smooth, beta=beta, n=n_points), best_h,
                         method='SLSQP', bounds=bounds, constraints=cons,
                         options={'maxiter': 5000, 'ftol': 1e-16})
            h_new = project_h(res.x, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new.copy(), c5
        except Exception:
            pass

    # Final Nelder-Mead polish on true max
    if remaining() > 15:
        bounds = Bounds(np.zeros(n_points), np.ones(n_points))
        try:
            res = minimize(partial(obj_true, n=n_points), best_h,
                         method='Nelder-Mead', bounds=bounds,
                         options={'maxiter': 5000, 'xatol': 1e-10,
                                  'fatol': 1e-15, 'maxfev': 30000, 'adaptive': True})
            h_new = project_h(res.x, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new.copy(), c5
        except Exception:
            pass

    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_points
