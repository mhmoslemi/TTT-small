import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-15:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def make_binary(h_frac, n):
    """Convert fractional to binary with exactly n/2 ones"""
    h_bin = (h_frac >= 0.5).astype(float)
    n_ones = int(h_bin.sum())
    target = n // 2
    if n_ones > target:
        # Need to flip (n_ones - target) ones to zeros
        # Preferentially flip positions where h_frac is lowest
        ones = np.where(h_bin == 1)[0]
        order = np.argsort(h_frac[ones])
        for idx in order[:n_ones - target]:
            h_bin[ones[idx]] = 0
    elif n_ones < target:
        zeros = np.where(h_bin == 0)[0]
        order = np.argsort(-h_frac[zeros])
        for idx in order[:target - n_ones]:
            h_bin[zeros[idx]] = 1
    return h_bin

            if total < 1e-15: break

            for _ in range(n_try):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                x[i] = 0; x[j] = 1
                c = compute_c5(x, n)
                x[i] = 1; x[j] = 0
                if c < best_c5 - 1e-14:
                    best_c5 = c
                    best_move = (i, j)
        else:
            for i in ones:
                for j in zeros:
                    x[i] = 0; x[j] = 1
                    c = compute_c5(x, n)
                    x[i] = 1; x[j] = 0
                    if c < best_c5 - 1e-14:
                        best_c5 = c
                        best_move = (i, j)
        if best_move is None:
            break
        x[best_move[0]] = 0
        x[best_move[1]] = 1
        c5 = best_c5
    return x, c5

def sa_binary(x, n, rng, n_iters=100000, t_start=0.005, t_end=0.00005, time_limit=None, t0=None):
    x = x.copy()
    c5 = compute_c5(x, n)
    best_x = x.copy()
    best_c5 = c5
    for it in range(n_iters):
        if time_limit is not None and t0 is not None:
            if time.time() - t0 > time_limit:
                break
        T = t_start * (t_end / t_start) ** (it / max(n_iters-1, 1))
        ones = np.where(x == 1)[0]
        zeros = np.where(x == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        x[i] = 0; x[j] = 1
        new_c5 = compute_c5(x, n)
        if new_c5 < c5 or rng.random() < np.exp(-(new_c5 - c5) / max(T, 1e-12)):
            c5 = new_c5
            if c5 < best_c5:
                best_x = x.copy()
                best_c5 = c5
        else:
            x[i] = 1; x[j] = 0
    return best_x, best_c5

def obj_smooth(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 700:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def sum_con(x, n):
    return np.sum(x) - n / 2.0

def sum_con_jac(x, n):
    return np.ones(n)
