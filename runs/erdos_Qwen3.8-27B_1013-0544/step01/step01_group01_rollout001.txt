import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max_val(corr, beta):
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -60, 60)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def smooth_max_grad_h(h, beta, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -60, 60)
    exp_s = np.exp(shifted)
    w = exp_s / np.sum(exp_s)
    grad = np.zeros(n)
    idx = np.arange(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-10:
            continue
        lag = k - (n - 1)
        v1 = idx + lag
        m1 = (v1 >= 0) & (v1 < n)
        if np.any(m1):
            grad[m1] += wk * (1.0 - h[v1[m1]])
        v2 = idx - lag
        m2 = (v2 >= 0) & (v2 < n)
        if np.any(m2):
            grad[m2] -= wk * h[v2[m2]]
    return grad * dx


def project_h(h, target_sum):
    n = len(h)
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(2000):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-15:
            break
        h_new = np.clip(h + diff / n, 0.0, 1.0)
        if np.max(np.abs(h_new - h)) < 1e-17:
            break
        h = h_new
    return h


def top_lags_gradient(h, top_k_indices, dx):
    """Gradient of average of top-k correlations."""
    n = len(h)
    grad = np.zeros(n)
    idx = np.arange(n)
    for k in top_k_indices:
        lag = k - (n - 1)
        v1 = idx + lag
        m1 = (v1 >= 0) & (v1 < n)
        if np.any(m1):
            grad[m1] += (1.0 - h[v1[m1]])
        v2 = idx - lag
        m2 = (v2 >= 0) & (v2 < n)
        if np.any(m2):
            grad[m2] -= h[v2[m2]]
    return grad * dx / len(top_k_indices)


def con_fun(x, target_sum):
    return x.sum() - target_sum


def con_jac(x, n_points):
    return np.ones(n_points)

    bounds = Bounds(np.zeros(n_points), np.ones(n_points))
