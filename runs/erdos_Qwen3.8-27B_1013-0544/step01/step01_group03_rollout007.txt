def solve_lp(h0, n):
    """Solve the linearized subproblem."""
    from scipy.optimize import linprog
    
    # Variables: h[0], ..., h[n-1], t
    # Objective: minimize t → c = [0, ..., 0, 1]
    # Constraints:
    #   For each j: grad_j(h0)^T h + (f_j(h0) - grad_j(h0)^T h0) ≤ t
    #   i.e., grad_j(h0)^T h - t ≤ -(f_j(h0) - grad_j(h0)^T h0)
    #   i.e., grad_j(h0) h[0..n-1] + [-1]*t ≤ -(f_j(h0) - grad_j(h0)^T h0)
    #   sum(h) = n/2
    #   0 ≤ h ≤ 1
    
    corr0 = compute_corr(h0)
    f0 = corr0  # f_j(h0) for all j
    
    # Build A_ub, b_ub for: A_ub @ x ≤ b_ub
    # x = [h[0], ..., h[n-1], t]
    m = len(corr0)  # 2n-1
    A_ub = np.zeros((m, n+1))
    b_ub = np.zeros(m)
    
    for j in range(m):
        d = j - (n-1)
        g = corr_shift_gradient(h0, d)
        A_ub[j, :n] = g
        A_ub[j, n] = -1
        b_ub[j] = -(f0[j] - g @ h0)
    
    # Plus sum constraint (handled separately)
    # Bounds: 0 ≤ h[i] ≤ 1, t ≥ 0
    bounds = [(0, 1)] * n + [(0, None)]
    
    # Equality: sum(h) = n/2
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1
    b_eq = np.array([n/2.0])
    
    c_obj = np.zeros(n+1)
    c_obj[n] = 1  # minimize t
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[:n], res.x[n]
    return h0, None

    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def corr_shift_gradient(h, d):
    """Gradient of correlation at shift d w.r.t. h. Vectorized."""
    n = len(h)

from scipy.optimize import minimize, linprog, Bounds
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def corr_shift_gradient(h, d):
    n = len(h)
    grad = np.zeros(n)
    if d >= 0:
        if d < n:
            grad[:n-d] += (1.0 - h[d:])
        if d > 0:
            grad[d:] += (-h[:n-d])
    else:
        dd = -d
        if dd < n:
            grad[dd:] += (1.0 - h[:n-dd])
            grad[:n-dd] += (-h[dd:])
    return grad

def project_feasible(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

    """Project z onto {0≤h≤1, sum(h)=target} using bisection."""
    lo, hi = z.min() - target, z.max()  # wide bounds
