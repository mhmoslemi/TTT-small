From numpy docs: For mode='full', the output is:
c_k = sum_j a_{j+k} * conj(v_j)

where a is the first argument and v is the second. The output length is M+N-1 where M=len(a), N=len(v).

So np.correlate(h, 1-h, mode='full')[k] = sum_j h[j + (k - N + 1)] * (1-h)[j]
where N = len(1-h) = n.

For k = n-1 (the middle element): sum_j h[j] * (1-h)[j] = sum_j h[j] - sum_j h[j]^2

For k = n-1+L (L > 0): sum_j h[j+L] * (1-h)[j] = sum_j h[j+L] - sum_j h[j+L]*h[j]
This is the integral of h(x+L)(1-h(x)) which is what we want for shift L.

For k = n-1-L (L > 0): sum_j h[j-L] * (1-h)[j] = sum_j h[j-L] - sum_j h[j-L]*h[j]

    if time.time() - start_time < time_limit - 25:
        h = best_h.copy()
        momentum = np.zeros(n_points)
        ss = 0.004
        mu = 0.65
        stag = 0
        for _ in range(200000):
            if time.time() - start_time > time_limit - 25:
                break
            g = grad_for_active(h, n_points, dx, 0.002, 0.005, 12, 3)
            momentum = mu * momentum - ss * g
            h_new = project_feasible(h + momentum, target)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                stag = 0
                ss = min(ss * 1.03, 0.015)
            else:
                stag += 1
                if stag % 50 == 0:
                    ss *= 0.85
                if ss < 1e-9:
                    h = best_h + rng.normal(0, 0.025, n_points)
                    h = project_feasible(h, target)
                    ss = 0.004
                    momentum = np.zeros(n_points)
                    stag = 0
            h = h_new

    # Phase 3: Fine refinement
    if time.time() - start_time < time_limit - 5:
        h = best_h.copy()
        momentum = np.zeros(n_points)
        ss = 0.0004
        mu = 0.35
        for _ in range(200000):
            if time.time() - start_time > time_limit - 5:
                break
            g = grad_for_active(h, n_points, dx, 0.0004, 1e-5, 6, 2)
            momentum = mu * momentum - ss * g
            h_new = project_feasible(h + momentum, target)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            else:
                ss *= 0.9
                if ss < 1e-13:
                    break
            h = h_new

    # Phase 4: Random restarts with quick local refine
    if time.time() - start_time < time_limit - 10:
        for _ in range(50):
            if time.time() - start_time > time_limit - 10:
                break
            h_pert = best_h + rng.normal(0, 0.015, n_points)
            h_pert = project_feasible(h_pert, target)
            c5_p = compute_c5(h_pert, dx)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_pert.copy()
            # Quick refine
            h = h_pert.copy()
            ss2 = 0.0003
            mom2 = np.zeros(n_points)
            for _2 in range(500):
                if time.time() - start_time > time_limit - 10:
                    break
                g = grad_for_active(h, n_points, dx, 0.001, 2e-5, 5, 2)
                mom2 = 0.45 * mom2 - ss2 * g
                h = project_feasible(h + mom2, target)
                c5_t = compute_c5(h, dx)
                if c5_t < best_c5:
                    best_c5 = c5_t
                    best_h = h.copy()
                else:
                    ss2 *= 0.9
                    if ss2 < 1e-12:
                        break

    return best_h, best_c5

def _set_p(p_val):
    global _g_p
    _g_p = p_val

def run(seed=42, budget_s=1100, **kwargs):
    global _g_dx, _g_p, _g_target
    start_time = time.time()
    time_limit = budget_s - 5.0
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except NameError:
        h_init = None

    n_list = [70, 60, 80, 65, 75, 55, 85]
    if h_init is not None:
        n_init = len(h_init)
        n_list = [n_init] + [x for x in n_list if x != n_init]

    best_overall_h = None
    best_overall_c5 = float('inf')
    best_overall_n = 70

    for n_points in n_list:
        elapsed = time.time() - start_time
        if elapsed > time_limit - 70:
            break

        dx = 2.0 / n_points
        target = float(n_points) / 2.0
        _g_dx = dx
        _g_target = target

        # Generate diverse initializations
        if h_init is not None and n_points == len(h_init):
            inits = [project_feasible(h_init, target)]
        else:
            inits = []
            inits.append(project_feasible(np.full(n_points, 0.5), target))
            inits.append(project_feasible(rng.uniform(0, 1, n_points), target))
            # Step function
            step = np.zeros(n_points)
            step[:n_points // 2] = 1.0
            inits.append(project_feasible(step, target))
            # Sine
            x_grid = np.linspace(0, 2, n_points, endpoint=False)
            sine = 0.5 + 0.45 * np.sin(np.pi * x_grid)
            inits.append(project_feasible(sine, target))
            # Alternating blocks
            nb = max(1, n_points // 6)
            alt = np.zeros(n_points)
            for i in range(0, n_points, 2 * nb):
                end = min(i + nb, n_points)
                alt[i:end] = 1.0
            inits.append(project_feasible(alt, target))
            # Two separated blocks
            two_blk = np.zeros(n_points)
                two_blk[:n_points // 5] = 1.0
            two_blk[3 * n_points // 4:] = 1.0
            inits.append(project_feasible(two_blk, target))

        # Pick best initialization
        best_h0 = None
        best_c5_0 = float('inf')
        for h0 in inits:
            c5_0 = compute_c5(h0, dx)
            if c5_0 < best_c5_0:
                best_c5_0 = c5_0
                best_h0 = h0.copy()

        # Optimize
        best_h, best_c5 = optimize_single_n(best_h0, n_points, dx, target, rng, time_limit, start_time)

        # Update global best
        if best_c5 < best_overall_c5:
            best_overall_c5 = best_c5
            best_overall_h = best_h.copy()
            best_overall_n = n_points

    if best_overall_h is None:
        n_points = 70
        dx = 2.0 / n_points
        target = n_points / 2.0
        _g_dx = dx
        _g_target = target
        best_overall_h = np.full(n_points, 0.5)
        best_overall_c5 = compute_c5(best_overall_h, dx)
        best_overall_n = n_points

    return (best_overall_h, best_overall_c5, best_overall_n)
