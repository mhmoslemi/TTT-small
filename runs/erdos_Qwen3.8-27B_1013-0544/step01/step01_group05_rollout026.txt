import numpy as np
from scipy.optimize import minimize
import time

_G_dx = None
_G_target = None
_G_pen = None

def project_feasible(x, target_sum):
    lo = -1.0
    hi = 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(x - mid, 0.0, 1.0).sum())
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def smooth_obj_grad(h, dx, tau):
    n = len(h)
    nh = n - 1
    c = np.correlate(h, 1.0 - h, mode='full')
    cd = c * dx
    m = float(np.max(cd))
    ex = np.exp(tau * (cd - m))
    S = float(np.sum(ex))
    f = m + np.log(S) / tau
    w = ex / S
    grad = np.zeros(n)
    for k in range(len(cd)):
        wk = w[k]
        if wk < 1e-15:
            continue
        L = k - nh
        if L > 0:
            grad[L:] += wk * (1.0 - h[:n - L])
            grad[:n - L] -= wk * h[L:]
        elif L < 0:
            L2 = -L
            grad[:n - L2] += wk * (1.0 - h[L2:])
            grad[L2:] -= wk * h[:n - L2]
        else:
            grad[:] += wk * (1.0 - h)
    return f, grad * dx

def active_subgrad(h, dx, tol_frac=0.002, max_lags=20):
    n = len(h)
    nh = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    threshold = c_max - tol_frac * c_max
    active = np.where(corr >= threshold)[0]
    if len(active) < 2:
        active = np.argsort(corr)[-2:]
    elif len(active) > max_lags:
        active = np.argsort(corr)[-max_lags:]
    grad = np.zeros(n)

            L2 = -L
            grad[:n - L2] += (1.0 - h[L2:])
            grad[L2:] -= h[:n - L2]
        else:
            grad[:] += (1.0 - h)
    return grad / float(len(active)) * dx
