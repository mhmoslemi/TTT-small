                h_cur = h_r.copy()
                mom_r = np.zeros(n)
                ss_r = 0.003
                for _ in range(800):
                    rem = t0 + budget_s - 3.0 - time.time()
                    if rem < 5:
                        break
                    g = lse_grad(h_cur, 30.0, dx)
                    mom_r = 0.4 * mom_r - ss_r * g
                    h_new = project_feasible(h_cur + mom_r, ts)
                    c5n = c5_val(h_new, dx)
                    if c5n < bl_c5:
                        bl_c5, bl_h = c5n, h_new.copy()
                    h_cur = h_new
            attempts += 1

        if bl_c5 < best_c5:
            best_c5 = bl_c5
            best_h = bl_h.copy()
            best_n = n

    return (best_h, best_c5, best_n)
