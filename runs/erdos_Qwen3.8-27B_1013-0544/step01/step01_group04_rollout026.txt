```python
    n = len(h)
    h = h.copy()
    for t in range(n_iter):
        corr = np.correlate(h, 1.0-h, mode='full')
        max_corr = np.max(corr)
        # Active set
        active = corr >= max_corr - active_tol
        # Subgradient: average of gradients of active constraints
        grad = np.zeros(n)
        for p in range(len(corr)):
            if active[p]:
                q = p - (n-1)
                for i in range(n):
                    term1 = (1 - h[i+q]) if 0 <= i+q < n else 0
                    term2 = h[i-q] if 0 <= i-q < n else 0
                    grad[i] += term1 - term2
        grad /= np.sum(active)
        # Step
        h -= lr * grad
        h = project_feasible(h, n)
    return h
```

```python
def compute_subgradient(h, active_mask):
    n = len(h)
    grad = np.zeros(n)
    for p in np.where(active_mask)[0]:
        q = p - (n-1)
        # First term: (1-h[i+q]) for valid i
        if q >= 0:
            # i ranges 0 to n-1-q
            grad[:n-q] += (1 - h[q:n])
            # Second term: -h[i-q] for i=q to n-1
            grad[q:] -= h[:n-q]
        else:
            q_neg = -q
            # i ranges q_neg to n-1
            grad[q_neg:] += (1 - h[:n-q_neg])
            # Second term: -h[i+q_neg] for i=0 to n-1-q_neg
            grad[:n-q_neg] -= h[q_neg:]
    grad /= np.sum(active_mask)
    return grad
```
