- For the remaining coordinates, scale them to meet the constraint:

- If any h[j] goes outside [0,1], clip and redistribute

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-15:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha

def penalty_obj(x, n, alpha, mu):
    return smooth_obj(x, n, alpha) + mu * (np.sum(x) - n / 2.0) ** 2

def sum_con(h, n):
    return np.sum(h) - n / 2.0

def sum_con_jac(h, n):
    return np.ones(n)

def coord_step(h, n, i, trial_val):
    """Try setting h[i]=trial_val, reproject, return (h_new, c5)."""
    h_try = h.copy()
    h_try[i] = trial_val
    h_try = project_feasible(h_try, n)
    c = compute_c5(h_try, n)
    return h_try, c
