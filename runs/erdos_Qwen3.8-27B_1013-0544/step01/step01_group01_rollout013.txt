from scipy.optimize import minimize, Bounds, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_lags(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target_sum - s
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def gradient_for_lag(h, d):
    """Gradient of lag-d overlap. d is the lag (can be negative)."""
    n = len(h)
    grad = np.zeros(n)
    if d >= 0:
        # f = sum_{j=0}^{n-1-d} (1-h[j]) * h[j+d]
        if d > 0:
            # h[i] in (1-h[i]) term for i in [0, n-1-d]: -h[i+d]
            grad[:n-d] += -h[d:]
            # h[i] as h[j+d] for j=i-d, i in [d, n-1]: (1-h[i-d])
            grad[d:] += 1.0 - h[:-d]
        # d=0: f = sum h[i]*(1-h[i]), grad = 1-2h
        elif d == 0:
            grad = 1.0 - 2.0 * h
    else:
        l = -d
        # f = sum_{j=0}^{n-1-l} h[j] * (1-h[j+l])

        grad[:n-l] += 1.0 - h[l:]
        # h[i] in (1-h[i]) for i in [l, n-1]: -h[i-l]
        grad[l:] += -h[:-l]
    return grad

    m = np.max(lag_vals)

        grad[:n - l] += 1.0 - h[l:]
        grad[l:] -= h[:n - l]
    return grad

def make_binary(h, tsum, rng):
    n = len(h)
    hb = np.where(h > 0.5, 1.0, 0.0).copy()
    s = hb.sum()
    if s > tsum:
        excess = int(s - tsum)
        ones = np.where(hb == 1.0)[0]
        idx = rng.choice(ones, excess, replace=False)
        hb[idx] = 0.0
    elif s < tsum:
        deficit = int(tsum - s)
        zeros = np.where(hb == 0.0)[0]
        idx = rng.choice(zeros, deficit, replace=False)
        hb[idx] = 1.0
    return hb

def binary_swap_search(hb, tsum, max_iters=5000, rng=None, time_check=None, budget_left=10):
    n = len(hb)
    if rng is None:
        rng = np.random
    lags = compute_lags(hb)
    max_v = lags.max()
    ones = np.where(hb == 1.0)[0]
    zeros = np.where(hb == 0.0)[0]
    no_improve = 0
    for it in range(max_iters):
        if time_check is not None and time_check() < budget_left:
            break
        improved = False
        for _ in range(100):
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            hb[i], hb[j] = hb[j], hb[i]
            new_lags = compute_lags(hb)
            new_max = new_lags.max()
            if new_max < max_v - 1e-12:
                max_v = new_max
                lags = new_lags
                ones = np.where(hb == 1.0)[0]
                zeros = np.where(hb == 0.0)[0]
                improved = True
                no_improve = 0
                break
            else:
                hb[i], hb[j] = hb[j], hb[i]
        if not improved:
            no_improve += 1
            if no_improve > 30:
                break
    return hb, max_v

def grad_descent_momentum(h, tsum, max_iter=2000, top_k=3, beta=80.0,
                          lr_init=0.02, momentum=0.5, time_check=None, budget_left=10):
    n = len(h)
    h = h.copy()
    v = np.zeros(n)
    lr = lr_init
    for it in range(max_iter):
        if time_check is not None and time_check() < budget_left:
            break
        lags = compute_lags(h)
        idx = np.argsort(lags)[-top_k:]
        active = lags[idx]
        m = active.max()
        w = np.exp(beta * (active - m))
        w = w / w.sum()
        grad = np.zeros(n)
        for ii, ww in zip(idx, w):
            d = ii - (n - 1)
            grad += ww * gradient_lag(h, d)
        # Momentum update
        v = momentum * v - lr * grad
        h_try = project_h(h + v, tsum)
        new_val = compute_lags(h_try).max()
        cur_val = lags.max()
        if new_val < cur_val:
            h = h_try
            lr = min(lr * 1.05, 0.08)
        else:
            lr *= 0.5
            v = momentum * v
            h_try = project_h(h - lr * grad, tsum)
            new_val2 = compute_lags(h_try).max()
            if new_val2 < cur_val:
                h = h_try
            if lr < 1e-10:
                break
    return h

def multi_start_optimize(h0, tsum, n, rng, time_budget, time_check):
    best_h = project_h(h0, tsum)
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n

    # Phase 1: Binary local search from multiple starts
    for restart in range(40):
        if time_check() < budget_left if (budget_left := 120) else False:
            break
        if time_check() < 120:
            break
        if restart == 0:
            hb = make_binary(best_h, tsum, rng)
        elif restart < 5:
            hb = make_binary(best_h, tsum, rng)
            nflips = rng.randint(1, 4)
            for _ in range(nflips):
                pos = rng.randint(n)
                hb[pos] = 1.0 - hb[pos]
            s = hb.sum()
            if s > tsum:
                excess = int(s - tsum)
                ones = np.where(hb == 1.0)[0]
                idx = rng.choice(ones, min(excess, len(ones)), replace=False)
                hb[idx] = 0.0
            elif s < tsum:
                deficit = int(tsum - s)
                zeros = np.where(hb == 0.0)[0]
                idx = rng.choice(zeros, min(deficit, len(zeros)), replace=False)
                hb[idx] = 1.0
        else:
            # More aggressive random binary
            hb = (rng.random(n) < 0.5).astype(float)
            s = hb.sum()
            if s > tsum:
                excess = int(s - tsum)
                ones = np.where(hb == 1.0)[0]
                idx = rng.choice(ones, min(excess, len(ones)), replace=False)
                hb[idx] = 0.0
            elif s < tsum:
                deficit = int(tsum - s)
                zeros = np.where(hb == 0.0)[0]
                idx = rng.choice(zeros, min(deficit, len(zeros)), replace=False)
                hb[idx] = 1.0

        hb, _ = binary_swap_search(hb, tsum, max_iters=3000, rng=rng,
                                   time_check=time_check, budget_left=80)
        c5b = compute_c5(hb)
        if c5b < best_c5:
            best_c5 = c5b
            best_h = hb.copy()

    # Phase 2: Fractional refinement from best binary
    if time_check() > 60:
        hf = best_h.copy()
        hf = grad_descent_momentum(hf, tsum, max_iter=3000, top_k=3, beta=60.0,
                                   lr_init=0.015, momentum=0.4,
                                   time_check=time_check, budget_left=40)
        hf = project_h(hf, tsum)
        c5f = compute_c5(hf)
        if c5f < best_c5:
            best_h = hf
            best_c5 = c5f

    # Phase 3: Alternating binary-fractional
    for cycle in range(20):
        if time_check() < 40:
            break
        # Fractional to binary
        hb = make_binary(best_h, tsum, rng)
        hb, _ = binary_swap_search(hb, tsum, max_iters=2000, rng=rng,
                                   time_check=time_check, budget_left=30)
        # Binary to fractional
        hf = grad_descent_momentum(hb, tsum, max_iter=2000, top_k=3, beta=100.0,
                                   lr_init=0.01, momentum=0.3,
                                   time_check=time_check, budget_left=25)
        hf = project_h(hf, tsum)
        c5_new = compute_c5(hf)
        if c5_new < best_c5:
            best_h = hf
            best_c5 = c5_new

    # Phase 4: Final high-precision polish
    if time_check() > 15:
        hf = best_h.copy()
        hf = grad_descent_momentum(hf, tsum, max_iter=5000, top_k=2, beta=300.0,
                                   lr_init=0.005, momentum=0.2,
                                   time_check=time_check, budget_left=5)
        hf = project_h(hf, tsum)
        c5f = compute_c5(hf)
        if c5f < best_c5:
            best_h = hf
            best_c5 = c5f

    # Phase 5: Try SLSQP for final polish if time remains
    if time_check() > 20:
        try:
            from scipy.optimize import minimize, Bounds
            def obj(h):
                lags = compute_lags(h)
                m = lags.max()
                return float(m + np.log(np.sum(np.exp(150 * (lags - m)))) / 150.0)
            def obj_grad(h):
                lags = compute_lags(h)
                m = lags.max()
                w = np.exp(150 * (lags - m))
                w = w / w.sum()
                grad = np.zeros(len(h))
                for idx, ww in zip(np.argsort(lags)[-5:], w[np.argsort(lags)[-5:]):
                    d = idx - (len(h) - 1)
                    grad += ww * gradient_lag(h, d)
                return grad
            bounds = Bounds(np.zeros(n), np.ones(n))
            cons = [{'type': 'eq', 'fun': lambda x: x.sum() - tsum,
                     'jac': lambda x: np.ones(n)}]
            res = minimize(obj, best_h, jac=obj_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 500, 'ftol': 1e-14})
            h_s = project_h(res.x, tsum)
            c5_s = compute_c5(h_s)
            if c5_s < best_c5:
                best_h = h_s
                best_c5 = c5_s
        except Exception:
            pass

    return best_h, best_c5, n

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def tl():
        return budget_s - (time.time() - t0)

    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(70, 0.5)

    n = len(h0)
    tsum = n / 2.0

    best_overall_h = project_h(h0, tsum)
    best_overall_c5 = compute_c5(best_overall_h)
    best_overall_n = n

    # Try multiple n values
    n_candidates = [n]
    if n == 70:
        n_candidates = [70, 68, 72, 66, 74]
    elif n <= 100:
        n_candidates = [n, n - 2, n + 2]

    for nc in n_candidates:
        if tl() < 150:
            break
        seed_i = seed + nc
        rng_i = np.random.RandomState(seed_i)
        h_start = np.full(nc, 0.5)
        # Add some structure
        h_start[:nc // 2] = 0.7
        h_start[nc // 2:] = 0.3
        h_start = project_h(h_start, nc / 2.0)

        h_best, c5_best, n_used = multi_start_optimize(h_start, nc / 2.0, nc,
                                                       rng_i, budget_s, tl)
        # Compare normalized: c5 is already the actual C5 value
        if c5_best < best_overall_c5:
            best_overall_h = h_best
            best_overall_c5 = c5_best
            best_overall_n = n_used

    # Also try from the parent directly
    if 'initial_h_values' in globals() and initial_h_values is not None and tl() > 200:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        np_ = len(h_parent)
        ts_ = np_ / 2.0
        h_best, c5_best, _ = multi_start_optimize(h_parent, ts_, np_,
                                                  np.random.RandomState(seed + 999),
                                                  budget_s, tl)
        if c5_best < best_overall_c5:
            best_overall_h = h_best
            best_overall_c5 = c5_best
            best_overall_n = np_

    # Final safety: ensure constraints
    best_overall_h = project_h(best_overall_h, best_overall_n / 2.0)
    best_overall_c5 = compute_c5(best_overall_h)

    return best_overall_h, best_overall_c5, best_overall_n
