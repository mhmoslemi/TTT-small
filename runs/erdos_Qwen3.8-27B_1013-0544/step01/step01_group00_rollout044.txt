        h_interp = np.interp(x_full, x_low, h_de)
        h_interp = project_h(h_interp, n * 0.5)
        h_pol, c5_pol = optimize_t_var(h_interp, max_iter=400, ftol=1e-10)
        if c5_pol < c5_best:
            h_best = h_pol.copy()
            c5_best = c5_pol

    # Phase 3: Aggressive multi-start perturbations
    scales = [0.08, 0.12, 0.18, 0.25, 0.30]
    for scale in scales:
        if time_left() < 80:
            break
        h_p = h_best + rng.randn(n) * scale
        h_p = np.clip(h_p, 0.0, 1.0)
        h_p = project_h(h_p, n * 0.5)
        h_new, c5_new = optimize_t_var(h_p, max_iter=350, ftol=1e-10)
        if c5_new < c5_best:
            h_best = h_new.copy()
            c5_best = c5_new

    # Phase 4: Block perturbations
    for _ in range(6):
        if time_left() < 60:
            break
        blen = rng.randint(n // 6, n // 2)
        start = rng.randint(0, n - blen)
        h_p = h_best.copy()
        h_p[start:start + blen] += rng.randn(blen) * 0.12
        h_p = np.clip(h_p, 0.0, 1.0)
        h_p = project_h(h_p, n * 0.5)
        h_new, c5_new = optimize_t_var(h_p, max_iter=300, ftol=1e-10)
        if c5_new < c5_best:
            h_best = h_new.copy()
            c5_best = c5_new

    # Phase 5: Active set optimization (focus on max correlations)
    for tol in [1e-3, 5e-4, 2e-4, 1e-4]:
        if time_left() < 70:
            break
        h_new, c5_new = optimize_active(h_best, tol=tol, temp=0.003, max_iter=300)
        if c5_new < c5_best:
            h_best = h_new.copy()
            c5_best = c5_new

    # Phase 6: Log-sum-exp refinement at decreasing temps
    for temp in [0.008, 0.004, 0.002, 0.001, 0.0005]:
        if time_left() < 70:
            break
        h_new, c5_new = optimize_smooth(h_best, temp, max_iter=300, ftol=1e-12)
        if c5_new < c5_best:
            h_best = h_new.copy()
            c5_best = c5_new

    # Phase 7: Another round of perturbations + active set
    for scale in [0.06, 0.10, 0.15]:
        if time_left() < 60:
            break
        h_p = h_best + rng.randn(n) * scale
        h_p = np.clip(h_p, 0.0, 1.0)
        h_p = project_h(h_p, n * 0.5)
        h_new, c5_new = optimize_t_var(h_p, max_iter=300, ftol=1e-10)
        if c5_new < c5_best:
            h_best = h_new.copy()
            c5_best = c5_new
        if time_left() < 50:
            break
        h_new, c5_new = optimize_active(h_best, tol=2e-4, temp=0.002, max_iter=250)
        if c5_new < c5_best:
            h_best = h_new.copy()
            c5_best = c5_new

    # Phase 8: Final t-variable polish
    if time_left() > 50:
        h_new, c5_new = optimize_t_var(h_best, max_iter=600, ftol=1e-11)
        if c5_new < c5_best:
            h_best = h_new.copy()
            c5_best = c5_new

    # Phase 9: Downsample-upsample trick
    for n_ds in [80, 50]:
        if time_left() < 150:
            break
        x_full = np.arange(n) / n
        x_ds = np.arange(n_ds) / n_ds
        h_ds = np.interp(x_ds, x_full, h_best)
        h_ds = project_h(h_ds, n_ds * 0.5)
        h_opt, _ = optimize_t_var(h_ds, max_iter=400, ftol=1e-10)
        h_up = np.interp(x_full, x_ds, h_opt)
        h_up = project_h(h_up, n * 0.5)
        c5_up = compute_c5(h_up)
        if c5_up < c5_best:
            h_best = h_up.copy()
            c5_best = c5_up
            if time_left() > 60:
                h_new, c5_new = optimize_t_var(h_best, max_iter=400, ftol=1e-10)
                if c5_new < c5_best:
                    h_best = h_new.copy()
                    c5_best = c5_new

    # Final cleanup
    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = project_h(h_best, n * 0.5)
    c5_final = compute_c5(h_best)
    return h_best, c5_final, n
