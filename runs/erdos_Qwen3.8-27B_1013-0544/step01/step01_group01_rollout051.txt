            scale = max(0.002, 0.025 * np.exp(-attempt / 40.0))
            h0 += np.random.randn(n) * scale
        elif strat == 1:
            ov = overlap_vec(h0)
            ks = np.argmax(ov)
            s = ks - (n - 1)
            idx = np.arange(n)
            contrib = np.zeros(n)
            v1 = (idx + s >= 0) & (idx + s < n)
            contrib[v1] = h0[v1] * (1.0 - h0[idx[v1] + s])
            top = np.argsort(contrib)[-max(5, n // 8)]
            h0[top] += np.random.randn(len(top)) * 0.012
        elif strat == 2:
            ns = np.random.randint(2, 6)
            for _ in range(ns):
                i, j = np.random.choice(n, 2, replace=False)
                tmp = h0[i]
                h0[i] = h0[j] + np.random.randn() * 0.004
                h0[j] = tmp + np.random.randn() * 0.004
        elif strat == 3:
            bl = np.random.randint(4, max(5, n // 3))
            bs = np.random.randint(0, max(1, n - bl))
            h0[bs:bs + bl] += np.random.randn(bl) * 0.02
        elif strat == 4:
            for j in range(0, n - 1, 2):
                d = np.random.randn() * 0.008
                h0[j] += d
                h0[j + 1] -= d
        elif strat == 5:
            h0 += (0.5 - h0) * 0.04 + np.random.randn(n) * 0.003
        elif strat == 6:
            # Perturb to reduce peak concentration
            ov = overlap_vec(h0)
            top_lags = np.argsort(ov)[-3:]
            for kl in top_lags:
                s = kl - (n - 1)
                idx = np.arange(n)
                v1 = (idx + s >= 0) & (idx + s < n)
                for j in v1:
                    h0[j] -= 0.003 * h0[j + s]
        else:
            h0 += np.random.randn(n) * 0.002

        h0 = project_feasible(h0, target)

        # SLSQP from perturbed start
        try:
            res = minimize(_obj, h0, jac=_grad, method='SLSQP', bounds=bounds,
                           constraints=constraints, options={'maxiter': 1500, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new, c5
        except Exception:
            pass

        # L-BFGS-B with penalty
        if tl() > 40:
            _P['pen'] = 50.0
            try:
                res = minimize(_obj_pen, h0, jac=_grad_pen, method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': 3000, 'gtol': 1e-14, 'iprint': -1})
                h_new = project_feasible(res.x, target)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h, best_c5 = h_new, c5
            except Exception:
                pass

    # Final high-beta polish
    if tl() > 12:
        _P['beta'] = 500.0
        try:
            res = minimize(_obj, best_h, jac=_grad, method='SLSQP', bounds=bounds,
                           constraints=constraints, options={'maxiter': 5000, 'ftol': 1e-16})
            h_new = project_feasible(res.x, target)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new, c5
        except Exception:
            pass

    # Verify
    if tl() > 5:
        _P['beta'] = 800.0
        c5_check = compute_c5(best_h)
        if c5_check < best_c5:
            best_c5 = c5_check

    best_h = project_feasible(best_h, target)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
