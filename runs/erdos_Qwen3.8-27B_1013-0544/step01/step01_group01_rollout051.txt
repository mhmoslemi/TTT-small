    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_obj(h, beta, penalty_lambda, target_sum):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    sm = m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta
    penalty = penalty_lambda * (np.sum(h) - target_sum) ** 2

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10):
        s = h.sum()
        if abs(s - target_sum) < 1e-12:
            break
        h += (target_sum - s) / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_a_vectorized(j, h):
    n = len(h)
    lags = np.arange(-(n-1), n)
    a = np.zeros(2*n-1)
    idx1 = (j + lags >= 0) & (j + lags < n)
    a[idx1] = 1.0 - h[j + lags[idx1]]
    idx2 = (j - lags >= 0) & (j - lags < n)
    a[idx2] -= h[j - lags[idx2]]
    return a

def coord_descent(h, n_sweeps=500, lambda_pen=0.1, rng=None):

    if rng is None:
        rng = np.random.RandomState(42)
    target_sum = n / 2.0

    for sweep in range(n_sweeps):
        order = rng.permutation(n)
        max_change = 0.0
        sum_others = h.sum()  # will update as we go
        
        for j in order:
            a = compute_a_vectorized(j, h)
            b = corr - a * h[j]
            sum_others = sum_others - h[j]  # hmm, this gets complicated
            # Actually, let's just include the penalty in the objective
            # Penalty: lambda * (sum(h) - n/2)^2
            # d/dx of penalty = 2*lambda*(sum_others + x - n/2)
            # So the 1D problem is: min_x max_k(a_k*x + b_k) + lambda*(sum_others + x - n/2)^2 * dx
            
            # For golden section, just evaluate the full 1D objective
            s_o = h.sum() - h[j]  # sum of all other coordinates
            mu = target_sum - s_o  # ideal value for h[j] to satisfy constraint
            
            def obj_1d(x):
                val = np.max(a * x + b) * (2.0 / n)
                val += lambda_pen * (s_o + x - target_sum) ** 2
                return val
            
            phi = (1.0 + np.sqrt(5.0)) / 2.0
            lo, hi = 0.0, 1.0

                x2 = lo + (hi - lo) / phi
                if obj_1d(x1) < obj_1d(x2):

                h[j] = x_opt
                corr += delta * a
                max_change = max(max_change, abs(delta))

        h = project_h(h, target_sum)
        corr = np.correlate(h, 1.0 - h, mode="full").copy()
        
        if max_change < 1e-9:
            break
    
    return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20):
        s = h.sum()
        if abs(s - target_sum) < 1e-13:
            break
        h += (target_sum - s) / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_a_vec(j, h):
    n = len(h)
    lags = np.arange(-(n - 1), n)
    a = np.zeros(2 * n - 1)
    idx1 = (j + lags >= 0) & (j + lags < n)
    a[idx1] = 1.0 - h[j + lags[idx1]]
    idx2 = (j - lags >= 0) & (j - lags < n)
    a[idx2] -= h[j - lags[idx2]]
    return a

def coord_descent(h, n_sweeps=400, lam=1.0, rng=None, stall_limit=50):
    n = len(h)
    if rng is None:
        rng = np.random.RandomState(0)
    target = n / 2.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full").astype(float)
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    stall = 0

    for sweep in range(n_sweeps):
        order = rng.permutation(n)
        max_chg = 0.0
        for j in order:
            a = compute_a_vec(j, h)
            b = corr - a * h[j]
            s_o = h.sum() - h[j]
            def obj1d(x):
                return np.max(a * x + b) * dx + lam * (s_o + x - target) ** 2
            lo, hi = 0.0, 1.0
            for _ in range(50):
                x1 = hi - (hi - lo) / phi
                x2 = lo + (hi - lo) / phi
                if obj1d(x1) < obj1d(x2):
                    hi = x2
                else:
                    lo = x1
            x_opt = (lo + hi) * 0.5
            delta = x_opt - h[j]
            if abs(delta) > 1e-11:
                h[j] = x_opt
                corr += delta * a
                max_chg = max(max_chg, abs(delta))
        h = project_h(h, target)
        corr = np.correlate(h, 1.0 - h, mode="full").astype(float)
        if max_chg < 1e-9:
            stall += 1
            if stall > stall_limit:
                break
        else:
            stall = 0
    return h

def slsqp_refine(h, maxiter=1500, beta=150.0):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    bnds = [(0.0, 1.0)] * n
    def obj_fn(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = np.max(corr)
        shifted = beta * (corr - m)
        log_sum = np.log(np.sum(np.exp(shifted)) + 1e-300)
        return (m + log_sum / beta) * dx
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    res = minimize(obj_fn, h, method='SLSQP', bounds=bnds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
    return res.x

def make_init(n, kind, rng):
    if kind == 0:
        return np.full(n, 0.5)
    elif kind == 1:
        h = np.zeros(n)
        bs = max(1, n // 14)
        for i in range(0, n, 2 * bs):
            h[i:min(i + bs, n)] = 1.0
        return project_h(h, n / 2.0)
    elif kind == 2:
        x = np.linspace(0, 2, n, endpoint=False)
        h = 0.5 + 0.4 * np.sin(2 * np.pi * x)
        return project_h(np.clip(h, 0, 1), n / 2.0)
    elif kind == 3:
        h = np.linspace(0, 1, n)
        return project_h(h, n / 2.0)
    else:
        h = np.zeros(n)
        n_ones = n // 2
        idx = rng.choice(n, n_ones, replace=False)
        h[idx] = 1.0
        return project_h(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    def tleft():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Coordinate descent on initial
    if tleft() > 30:
        h1 = coord_descent(best_h.copy(), n_sweeps=500, lam=1.0, rng=rng)
        h1 = project_h(h1, len(h1) / 2.0)
        c5 = compute_c5(h1)
        if c5 < best_c5:
            best_h, best_c5 = h1, c5

    # Phase 2: SLSQP polish
    if tleft() > 25:
        try:
            hx = slsqp_refine(best_h.copy(), maxiter=2000, beta=100.0)
            hx = project_h(hx, len(hx) / 2.0)
            c5 = compute_c5(hx)
            if c5 < best_c5:
                best_h, best_c5 = hx.copy(), c5
        except Exception:
            pass

    # Phase 3: Try multiple n and initializations with coord descent
    n_list = [40, 50, 55, 60, 70, 80, 90, 100, 120]
    for n_t in n_list:
        if tleft() < 40:
            break
        target = n_t / 2.0
        for kind in range(5):
            if tleft() < 20:
                break
            h_try = make_init(n_t, kind, rng)
            h_cd = coord_descent(h_try, n_sweeps=300, lam=0.8, rng=rng)
            h_cd = project_h(h_cd, target)
            c5 = compute_c5(h_cd)
            if c5 < best_c5:
                best_h = h_cd.copy()
                best_c5 = c5
                # Quick SLSQP polish
                if tleft() > 20:
                    try:
                        hx = slsqp_refine(best_h.copy(), maxiter=500, beta=120.0)
                        hx = project_h(hx, len(hx) / 2.0)
                        c5p = compute_c5(hx)
                        if c5p < best_c5:
                            best_h, best_c5 = hx.copy(), c5p
                    except Exception:
                        pass

    # Phase 4: Targeted refinement of best solution
    if tleft() > 20:
        n_b = len(best_h)
        lam_vals = [0.1, 0.5, 2.0, 5.0]
        for lam_v in lam_vals:
            if tleft() < 10:
                break
            h_r = coord_descent(best_h.copy(), n_sweeps=300, lam=lam_v, rng=rng)
            h_r = project_h(h_r, n_b / 2.0)
            c5 = compute_c5(h_r)
            if c5 < best_c5:
                best_h, best_c5 = h_r, c5

    # Phase 5: Final high-precision SLSQP
    if tleft() > 15:
        try:
            hx = slsqp_refine(best_h.copy(), maxiter=3000, beta=300.0)
            hx = project_h(hx, len(hx) / 2.0)
            c5 = compute_c5(hx)
            if c5 < best_c5:
                best_h, best_c5 = hx.copy(), c5
        except Exception:
            pass

    # Phase 6: Final coordinate descent + SLSQP cycle
    if tleft() > 15:
        n_b = len(best_h)
        h_f = coord_descent(best_h.copy(), n_sweeps=400, lam=0.3, rng=rng)
        h_f = project_h(h_f, n_b / 2.0)
        c5 = compute_c5(h_f)
        if c5 < best_c5:
            best_h, best_c5 = h_f, c5
        if tleft() > 10:
            try:
                hx = slsqp_refine(best_h.copy(), maxiter=2000, beta=200.0)
                hx = project_h(hx, len(hx) / 2.0)
                c5 = compute_c5(hx)
                if c5 < best_c5:
                    best_h, best_c5 = hx.copy(), c5
            except Exception:
                pass

    best_h = project_h(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
