        eff_lr = lr / (1.0 - beta_m ** (it + 1))
        h_try = h - eff_lr * momentum
        h_try = project_feasible(h_try, target_sum)
        c5_try = compute_c5(h_try)

        if c5_try < best_c5:
            best_h, best_c5 = h_try.copy(), c5_try
            h = h_try
            lr = min(0.15, lr * 1.02)
        else:
            h_try2 = h - 0.3 * eff_lr * momentum
            h_try2 = project_feasible(h_try2, target_sum)
            c5_try2 = compute_c5(h_try2)
            if c5_try2 < best_c5:
                best_h, best_c5 = h_try2.copy(), c5_try2
                h = h_try2
            else:
                lr = max(0.005, lr * 0.85)

    # Phase 3: Global restarts
    attempt = 0
    while time_left() > 60:
        attempt += 1
        hr = best_h.copy()
        strategy = attempt % 7

        if strategy == 0:
            scale = 0.04 + 0.06 * np.exp(-attempt / 40.0)
            hr += np.random.randn(n) * scale
        elif strategy == 1:
            b1 = np.random.randint(0, n // 3)
            b2 = np.random.randint(2 * n // 3, n - 4)
            blen = np.random.randint(3, min(8, n // 5))
            tmp = hr[b1:b1+blen].copy()
            hr[b1:b1+blen] = hr[b2:b2+blen]
            hr[b2:b2+blen] = tmp
        elif strategy == 2:
            freq = np.random.uniform(1.0, 6.0)
            hr += 0.03 * np.sin(2 * np.pi * freq * np.arange(n) / n + np.random.random() * 6.28)
        elif strategy == 3:
            idx = np.random.choice(n, max(3, n // 8), replace=False)
            hr[idx] += np.random.randn(len(idx)) * 0.08
        elif strategy == 4:
            # Anti-correlation push: find worst lag and directly push
            corr = np.correlate(best_h, 1.0 - best_h, mode="full")
            k_worst = np.argmax(corr)
            sh = k_worst - (n - 1)
            if sh >= 0:
                L = n - sh
                hr[sh:] -= 0.03 * (1.0 - best_h[:L])
                hr[:L] += 0.03 * best_h[sh:]
            else:
                a = -sh
                L = n - a
                hr[:L] -= 0.03 * (1.0 - best_h[a:])
                hr[a:] += 0.03 * best_h[:L]
        elif strategy == 5:
            # Coarse re-randomize a block
            blen = np.random.randint(n // 5, n // 3)
            bs = np.random.randint(0, n - blen)
            hr[bs:bs+blen] = np.random.random(blen) * 0.6 + 0.1
        else:
            # Two-point swap with noise
            i, j = np.random.choice(n, 2, replace=False)
            diff = (hr[i] - hr[j]) * 0.5 + np.random.randn() * 0.02
            hr[i] -= diff
            hr[j] += diff

        hr = project_feasible(hr, target_sum)

        # Optimize from restart
        beta_r = 120 + 150 * np.random.random()
        mu_r = 500.0

        def obj_r(hh):
            return smooth_max_grad_val(hh, beta_r, dx, target_sum, mu_r)[0]

        def grad_r(hh):
            return smooth_max_grad_val(hh, beta_r, dx, target_sum, mu_r)[1]

        try:
            res = minimize(obj_r, hr, jac=grad_r, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': 800, 'ftol': 1e-15, 'gtol': 1e-10})
            h_new = project_feasible(res.x, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new.copy(), c5
        except Exception:
            pass

    # Phase 4: Final high-precision polish
    if time_left() > 20:
        for beta_f in [500, 800, 1200]:
            if time_left() < 10:
                break

            def obj_f(hh):
                return smooth_max_grad_val(hh, beta_f, dx, target_sum, 5000.0)[0]

            def grad_f(hh):
                return smooth_max_grad_val(hh, beta_f, dx, target_sum, 5000.0)[1]

            try:
                res = minimize(obj_f, best_h, jac=grad_f, method='L-BFGS-B',
                               bounds=bounds,
                               options={'maxiter': 5000, 'ftol': 1e-20, 'gtol': 1e-14})
                h_new = project_feasible(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h, best_c5 = h_new.copy(), c5
            except Exception:
                pass

    # Final verification
    best_h = project_feasible(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n
