def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def power_obj(h, p):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.sum(corr**p) ** (1.0/p)

def sum_constraint(h):
    return np.sum(h) - len(h) * 0.5

def project(h, target_sum):
    h = h.copy()
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return np.clip(h, 0, 1)
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = mask.sum()
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0, 1)

def optimize_power(h0, p, maxiter=500):
    n = len(h0)
    res = minimize(power_obj, h0.copy(), args=(p,), method='L-BFGS-B',
                   bounds=[(0,1)]*n,
                   constraints=[{'type':'eq', 'fun': sum_constraint, 'jac': lambda x: np.ones(n)}],
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    h_new = project(np.clip(res.x, 0, 1), n*0.5)
    return h_new, compute_c5(h_new)

def optimize_t(h0, maxiter=500):
    n = len(h0)
    c0 = compute_c5(h0)
    v0 = np.concatenate([h0.copy(), [c0 + 0.01]])

        t = v[-1]
        dx = 2.0/n
        corr = np.correlate(h, 1-h, mode='full') * dx

    n = len(v) - 1
    return np.array([np.sum(v[:n]) - n * 0.5])


def tvar_eq_jac(v):
    n = len(v) - 1
    j = np.zeros(n + 1)
    j[:n] = 1.0
    return j


def penalty_power_obj(h, p, mu):
    n = len(h)
    base = power_smooth_obj(h, p)
    s = np.sum(h) - n * 0.5
    return base + mu * s * s


def optimize_power_slsqp(h0, p, max_iter=300):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac}]
    res = minimize(power_smooth_obj, h0.copy(), args=(p,), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-14})
    h_new = project_to_constraint(res.x, n * 0.5)
    return h_new, compute_c5(h_new)


def optimize_tvar_slsqp(h0, temp, max_iter=400):
    n = len(h0)
    c0 = compute_c5(h0)
    v0 = np.concatenate([h0.copy(), [c0 + 0.02]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.5)]
    cons = [
        {'type': 'ineq', 'fun': tvar_ineq_smooth, 'args': (temp,)},
        {'type': 'eq', 'fun': tvar_eq, 'jac': tvar_eq_jac}
    ]
    res = minimize(tvar_obj, v0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-12})
    h_new = project_to_constraint(res.x[:n], n * 0.5)
    return h_new, compute_c5(h_new)


def optimize_lbfgs_penalty(h0, p, mu, max_iter=300):
    n = len(h0)
    res = minimize(penalty_power_obj, h0.copy(), args=(p, mu),
                   method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': max_iter, 'ftol': 1e-15})
    h_new = project_to_constraint(res.x, n * 0.5)
    return h_new, compute_c5(h_new)


def interp_to_resolution(h, n_new):
    x_old = np.arange(len(h)) / len(h)
    x_new = np.arange(n_new) / n_new
    h_new = np.interp(x_new, x_old, h)
    return project_to_constraint(h_new, n_new * 0.5)


def refine_at_resolution(h, n, t0, budget_s, rng):
    elapsed = time.time() - t0
    h_cur = h.copy()
    c5_cur = compute_c5(h_cur)
    # Power smoothing
    for p in [8, 32, 128, 512]:
        if time.time() - t0 > budget_s - 60:
            break
        h_new, c5_new = optimize_power_slsqp(h_cur, p, max_iter=200)
        if c5_new < c5_cur:
            h_cur = h_new
            c5_cur = c5_new
    # t-variable
    for temp in [50.0, 200.0, 500.0]:
        if time.time() - t0 > budget_s - 50:
            break
        h_new, c5_new = optimize_tvar_slsqp(h_cur, temp, max_iter=300)
        if c5_new < c5_cur:
            h_cur = h_new
            c5_cur = c5_new
    # L-BFGS-B penalty
    for p in [128, 512]:
        if time.time() - t0 > budget_s - 30:
            break
        for mu in [1000, 10000]:
            if time.time() - t0 > budget_s - 20:
                break
            h_new, c5_new = optimize_lbfgs_penalty(h_cur, p, mu, max_iter=150)
            if c5_new < c5_cur:
                h_cur = h_new
                c5_cur = c5_new
    return h_cur, c5_cur


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    def remaining():
        return budget_s - elapsed()

    h = np.array(initial_h_values, dtype=float)
    n = len(h)
    h = project_to_constraint(h, n * 0.5)
    h_best = h.copy()
    c5_best = compute_c5(h_best)

    # Phase 1: Coarse resolution search (n=60, 80)
    for n_coarse in [60, 80]:
        if remaining() < 200:
            break
        h_coarse = interp_to_resolution(h_best, n_coarse)
        h_ref, c5_ref = refine_at_resolution(h_coarse, n_coarse, t0, budget_s - 150, rng)
        if c5_ref < c5_best:
            h_best = interp_to_resolution(h_ref, n)
            c5_best = compute_c5(h_best)

    # Phase 2: Refine at current resolution
    if remaining() > 150:
        h_ref, c5_ref = refine_at_resolution(h_best, n, t0, budget_s - 80, rng)
        if c5_ref < c5_best:
            h_best = h_ref
            c5_best = c5_ref

    # Phase 3: Higher resolution
    for n_high in [240, 360, 500]:
        if remaining() < 200:
            break
        h_interp = interp_to_resolution(h_best, n_high)
        h_ref, c5_ref = refine_at_resolution(h_interp, n_high, t0, budget_s - 100, rng)
        if c5_ref < c5_best:
            h_best = h_ref
            c5_best = c5_ref

    # Phase 4: Multiple restarts at best resolution
    n_use = len(h_best)
    for i in range(8):
        if remaining() < 80:
            break
        scales = [0.03, 0.05, 0.08, 0.12]
        scale = scales[i % len(scales)]
        h_pert = h_best + rng.randn(n_use) * scale
        h_pert = project_to_constraint(h_pert, n_use * 0.5)
        h_new, c5_new = optimize_power_slsqp(h_pert, 32, max_iter=200)
        if c5_new < c5_best:
            h_best = h_new
            c5_best = c5_new
        h_new, c5_new = optimize_tvar_slsqp(h_best, 200.0, max_iter=300)
        if c5_new < c5_best:
            h_best = h_new
            c5_best = c5_new
        h_new, c5_new = optimize_lbfgs_penalty(h_best, 256, 5000, max_iter=200)
        if c5_new < c5_best:
            h_best = h_new
            c5_best = c5_new

    # Phase 5: Final polish
    if remaining() > 60:
        h_new, c5_new = optimize_tvar_slsqp(h_best, 500.0, max_iter=500)
        if c5_new < c5_best:
            h_best = h_new
            c5_best = c5_new
        h_new, c5_new = optimize_power_slsqp(h_best, 1024, max_iter=200)
        if c5_new < c5_best:
            h_best = h_new
            c5_best = c5_new

    h_best = project_to_constraint(h_best, len(h_best) * 0.5)
    c5_final = compute_c5(h_best)
    return h_best, c5_final, len(h_best)
