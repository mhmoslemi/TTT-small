def coord_optimize(h, i, corr, n, dx):

    # Compute a_k and b_k for each lag k
    # c_k = a_k * h[i] + b_k
    # a_k = d(c_k)/d(h[i]) = (1-h[i-k])*1[valid] - h[i+k]*1[valid]
    # b_k = c_k - a_k * h[i]
    
    lags = np.arange(-(n-1), n)
    ip = i + lags  # i+k
    im = i - lags  # i-k
    valid_p = (ip >= 0) & (ip < n)
    valid_m = (im >= 0) & (im < n)
    
    a = np.zeros(2*n-1)
    for idx in range(2*n-1):
        k = lags[idx]
        a[idx] = 0.0
        if valid_m[idx]:
            a[idx] += (1.0 - h[im[idx]])
        if valid_p[idx]:
            a[idx] -= h[ip[idx]]
    
    b = corr - a * h[i]
    
    # Minimize max_k (a_k * x + b_k) over x in [0, 1]
    # using ternary search
    lo, hi = 0.0, 1.0
    for _ in range(50):
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        f1 = np.max(a * m1 + b)
        f2 = np.max(a * m2 + b)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    
    x_opt = (lo + hi) / 2
    return x_opt
