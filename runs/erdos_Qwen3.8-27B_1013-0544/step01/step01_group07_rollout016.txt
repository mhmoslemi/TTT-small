def project_sum_box(z, target_sum, n):
    """Project z onto {0<=h<=1, sum(h)=target_sum}"""
    # Bisection on tau
    lo, hi = z.min() - 1, z.max() + 1
    for _ in range(100):
        tau = (lo + hi) / 2
        h = np.clip(z - tau, 0, 1)
        s = h.sum()
        if s > target_sum:
            lo = tau
        else:
            hi = tau
    return np.clip(z - (lo+hi)/2, 0, 1)

def subgradient_c5(h, dx, tol=1e-4):
    """Compute subgradient of max_k C[k]*dx"""
    v = 1 - h
    C = np.correlate(h, v, mode="full") * dx
    Cmax = C.max()
    # Active set: those within tol of max
    active = C > (Cmax - tol)
    n_active = active.sum()
    
    n = len(h)
    # Compute gradient for active lags
    # C[k] = sum_i h[i] * v[i+k-(n-1)] for valid indices
    # dC[k]/dh[j] = v[j+k-(n-1)] * I[valid] - h[j-k+(n-1)] * I[valid]
    
    grad = np.zeros(n)
    for k in np.where(active)[0]:
        for j in range(n):
            idx1 = j + k - (n - 1)
            idx2 = j - k + (n - 1)
            g = 0.0
            if 0 <= idx1 < n:
                g += v[idx1]
            if 0 <= idx2 < n:
                g -= h[idx2]
            grad[j] += g
    grad /= n_active
    return grad * dx, Cmax

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def soft_max_value(corr, t):
    m = corr.max()
    return m + (1.0/t) * np.log(np.sum(np.exp(t * (corr - m))))

def obj_lbfgs(h, t, dx, target_sum, lam=1000.0):
    corr = np.correlate(h, 1 - h, mode="full") * dx
    sm = soft_max_value(corr, t)
    pen = lam * (np.sum(h) - target_sum) ** 2
    return sm + pen

def project_sum_box(z, target_sum):
    n = len(z)

        tau = (lo + hi) / 2

        if h.sum() > target_sum:

        else:
            hi = tau

def grad_subgrad(h, dx, tol=1e-4):
    n = len(h)
    v = 1 - h
    C = np.correlate(h, v, mode="full") * dx
    Cmax = C.max()
    active = C >= (Cmax - tol)

    j = np.arange(n)

    idx2 = j[:, None] - k[None, :] + (n - 1)

    G[mask1] = v[idx1[mask1]]
    G[mask2] -= h[idx2[mask2]]
    
    grad = (G[:, active].sum(axis=1) / n_act) * dx
    return grad, Cmax

def lbfgs_optimize(h, dx, target_sum, t=50, maxiter=500, lam=1000.0):
    n = len(h)

        h.copy(),
        method='L-BFGS-B',
        bounds=[(0, 1)] * n,
        options={'maxiter': maxiter, 'ftol': 1e-14}
    )
    h_new = fix_sum(np.clip(res.x, 0, 1), target_sum)
    return h_new

def slsqp_optimize(h, dx, target_sum, t=100, maxiter=300):
    n = len(h)
    corr_fn = lambda x: soft_max_value(np.correlate(x, 1-x, mode="full")*dx, t)
    res = minimize(
        corr_fn, h.copy(),
        method='SLSQP',
        bounds=[(0, 1)] * n,
        constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
        options={'maxiter': maxiter, 'ftol': 1e-12}
    )
    return fix_sum(np.clip(res.x, 0, 1), target_sum)

def block_descent(h, dx, target_sum, block_size=4, n_rounds=50, t=100):

        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            h = h_new
            best_c5 = c5_new
    return h, best_c5

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def soft_max_value(corr, t):
    m = corr.max()
    if t < 1e-6:
        return m
    shifted = t * (corr - m)
    return m + (1.0 / t) * np.log(np.sum(np.exp(shifted)))

def fix_sum(h, target_sum):
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        h = h + diff / len(h)
    return np.clip(h, 0.0, 1.0)

def project_sum_box(z, target_sum):
    n = len(z)
    lo, hi = z.min() - 2.0, z.max() + 2.0
    for _ in range(100):
        tau = (lo + hi) / 2.0
        h = np.clip(z - tau, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = tau
        else:
            hi = tau
    return np.clip(z - (lo + hi) / 2.0, 0.0, 1.0)

def grad_subgrad(h, dx, tol=1e-4):
    n = len(h)
    v = 1 - h
    C = np.correlate(h, v, mode="full") * dx
    Cmax = C.max()
    active = C >= (Cmax - tol)
    n_act = max(1, active.sum())
    j = np.arange(n)
    k = np.arange(2 * n - 1)
    idx1 = j[:, None] + k[None, :] - (n - 1)
    idx2 = j[:, None] - k[None, :] + (n - 1)
    mask1 = (idx1 >= 0) & (idx1 < n)
    mask2 = (idx2 >= 0) & (idx2 < n)
    G = np.zeros((n, 2 * n - 1))
    G[mask1] = v[idx1[mask1]]
    G[mask2] -= h[idx2[mask2]]
    grad = (G[:, active].sum(axis=1) / n_act) * dx
    return grad, Cmax, set(np.where(active)[0].tolist())

def subgradient_optimize(h, dx, target_sum, n_iter=400, alpha_init=0.004, tol=1e-4):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    h = h.copy()
    stagnant = 0
    for it in range(n_iter):
        grad, c5_cur, _ = grad_subgrad(h, dx, tol)
        alpha = alpha_init / np.sqrt(it + 1)
        h_new = project_sum_box(h - alpha * grad, target_sum)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            stagnant = 0
        else:
            stagnant += 1
        h = h_new
        if stagnant > 150:
            break
    return best_h, best_c5

def obj_smooth(h, t, dx, target_sum, lam=5000.0):
    corr = np.correlate(h, 1 - h, mode="full") * dx
    sm = soft_max_value(corr, t)
    pen = lam * (h.sum() - target_sum) ** 2
    return sm + pen

def lbfgs_optimize(h, dx, target_sum, t=50, maxiter=400, lam=5000.0):
    n = len(h)
    res = minimize(
        lambda x: obj_smooth(x, t, dx, target_sum, lam),
        h.copy(), method='L-BFGS-B',
        bounds=[(0.0, 1.0)] * n,
        options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10}
    )
    return fix_sum(np.clip(res.x, 0.0, 1.0), target_sum)

def slsqp_optimize(h, dx, target_sum, t=200, maxiter=400):
    n = len(h)
    def obj(x):
        return soft_max_value(np.correlate(x, 1 - x, mode="full") * dx, t)
    res = minimize(
        obj, h.copy(), method='SLSQP',
        bounds=[(0.0, 1.0)] * n,
        constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - target_sum}],
        options={'maxiter': maxiter, 'ftol': 1e-14}
    )
    return fix_sum(np.clip(res.x, 0.0, 1.0), target_sum)

def block_descent(h, dx, target_sum, block_size=4, n_rounds=40, t=200):
    n = len(h)
    best_c5 = compute_c5(h, dx)
    for _ in range(n_rounds):
        if n <= block_size:
            break
        start = np.random.randint(0, n - block_size + 1)
        idx = np.arange(start, start + block_size)
        sub = h[idx].copy()
        fixed_sum = target_sum - (h.sum() - h[idx].sum())
        def sub_obj(x):
            htmp = h.copy()
            htmp[idx] = x
            return soft_max_value(np.correlate(htmp, 1 - htmp, mode="full") * dx, t)
        try:
            res = minimize(sub_obj, sub, method='SLSQP',
                           bounds=[(0.0, 1.0)] * block_size,
                           constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - fixed_sum}],
                           options={'maxiter': 40, 'ftol': 1e-12})
            h_new = h.copy()
            h_new[idx] = np.clip(res.x, 0.0, 1.0)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                h = h_new
                best_c5 = c5_new
        except Exception:
            pass
    return h, best_c5

def upsample(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0, 2, old_n + 1)[:-1]
    x_new = np.linspace(0, 2, new_n + 1)[:-1]
    return np.interp(x_new, x_old, h)
