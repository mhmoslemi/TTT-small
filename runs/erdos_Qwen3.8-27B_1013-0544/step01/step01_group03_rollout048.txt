            if c5_new is not None:

    # Phase 2: Structured seeds at multiple resolutions
    seed_configs = [
        (80, lambda n: periodic_binary_seed(n, 8, 4)),
        (100, lambda n: periodic_binary_seed(n, 10, 5)),
        (120, lambda n: periodic_binary_seed(n, 12, 6)),
        (100, lambda n: harmonic_seed(n, 3, 0.5)),
        (120, lambda n: harmonic_seed(n, 4, 0.45)),
        (140, lambda n: harmonic_seed(n, 3, 0.5)),
        (100, lambda n: m_sequence_seed(n)),
    ]

    for n_s, gen_fn in seed_configs:
        if rem() < 25:
            break
        h_s = gen_fn(n_s)
        for tau in [50, 200, 1000]:
            if rem() < 12:
                break
            h_new, c5_new = slsqp_optimize(h_s, n_s, tau, maxiter=350)
            if c5_new is not None and c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
            if c5_new is not None:
                h_s = h_new

    # Phase 3: Diverse random restarts from best
    hn = len(best_h)
    n_restarts = max(5, int(rem() // 20))
    for r in range(n_restarts):
        if rem() < 15:
            break
        scale = 0.05 + 0.2 * np.random.random()
        hp = best_h + np.random.normal(0, scale, hn)
        hp = np.clip(hp, 0.0, 1.0)
        hp = fix_sum(hp, hn / 2.0)
        for tau in [100, 500, 2000]:
            if rem() < 10:
                break
            h_new, c5_new = slsqp_optimize(hp, hn, tau, maxiter=250)
            if c5_new is not None and c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
            if c5_new is not None:
                hp = h_new
        # Also try different n
        if rem() > 30 and r % 2 == 0:
            hn_alt = hn + 20 * (1 if r % 4 == 0 else -1)
            hn_alt = max(40, min(200, hn_alt))
            if hn_alt % 2 == 0:
                hp2 = upsample(best_h, hn_alt)
                h_new, c5_new = slsqp_optimize(hp2, hn_alt, 500, maxiter=300)
                if c5_new is not None and c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new

    # Phase 4: Greedy lag reduction
    if rem() > 15:
        hn_g = len(best_h)
        h_g, c5_g = greedy_lag_reduce(best_h, hn_g, max_rounds=300)
        if c5_g < best_c5:
            best_h = h_g
            best_c5 = c5_g

    # Phase 5: Final high-temperature polish
    if rem() > 10:
        hn_f = len(best_h)
        for tau in [3000, 10000, 30000]:
            if rem() < 8:
                break
            h_new, c5_new = slsqp_optimize(best_h, hn_f, tau, maxiter=400)
            if c5_new is not None and c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    # Phase 6: Try to push toward binary (reduces lag-0)
    if rem() > 10:
        hn_b = len(best_h)
        target_b = hn_b / 2.0
        # Round most values to 0/1, keep a few fractional
        sorted_idx = np.argsort(best_h)
        h_bin_attempt = best_h.copy()
        # Keep the middle 10 values free, round the rest
        n_free = max(4, hn_b // 15)
        mid_start = (hn_b - n_free) // 2
        for i in range(hn_b):
            if i < mid_start or i >= mid_start + n_free:
                h_bin_attempt[i] = 1.0 if best_h[i] > 0.5 else 0.0
        h_bin_attempt = fix_sum(h_bin_attempt, target_b)
        h_new, c5_new = slsqp_optimize(h_bin_attempt, hn_b, 2000, maxiter=400)
        if c5_new is not None and c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Final fix
    hn_final = len(best_h)
    best_h = fix_sum(best_h, hn_final / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = evaluate_c5(best_h)

    return best_h, best_c5, hn_final
