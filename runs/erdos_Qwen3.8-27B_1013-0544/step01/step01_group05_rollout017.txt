    """Compute average gradient over active lags."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    tol = max(0.02, c_max * tol_frac)
    active = np.where(corr >= c_max - tol)[0]
    if len(active) < 2:
        active = np.argsort(corr)[-2:]
    elif len(active) > max_active:
        active = np.argsort(corr)[-max_active:]

        else:

            return h_ref, compute_c5(h_ref, dx)

def project_feasible(x, target_sum):

    hi = float(np.max(x)) + 1.0

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_subgrad(h, n, tol_abs, max_active):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    active = np.where(corr >= c_max - tol_abs)[0]
    if len(active) < 2:
        active = np.argsort(corr)[-max_active:]
    elif len(active) > max_active:
        active = np.argsort(corr)[-max_active:]
    n_half = n - 1
    omh = 1.0 - h

    for idx in active:
        L = abs(int(idx) - n_half)
        if L == 0:
            grad += omh - h
        else:
            grad[L:] += omh[:n - L]
            grad[:n - L] -= h[L:]
    grad /= float(len(active))
    return grad
