            if total < 1e-15:

            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-15:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def optimize_coord(h, c, j, d_j):
    lo = -h[j]
    hi = 1.0 - h[j]
    if lo >= hi - 1e-15:
        return h[j]
    lo_b = lo
    hi_b = hi
    best_delta = 0.0
    best_val = float(np.max(c))
    for _ in range(60):
        mid = 0.5 * (lo_b + hi_b)
        vals = c + mid * d_j
        max_val = float(np.max(vals))
        if max_val < best_val - 1e-15:
            best_val = max_val
            best_delta = mid
        thresh = max_val - 1e-11
        sg = d_j[vals >= thresh]
        if sg.size == 0:
            break
        if float(np.min(sg)) > 1e-10:
            hi_b = mid
        elif float(np.max(sg)) < -1e-10:
            lo_b = mid
        else:
            best_delta = mid
            break
        if hi_b - lo_b < 1e-14:
            break
    return float(np.clip(h[j] + best_delta, 0.0, 1.0))

def coord_descent(h, n, max_sweeps=80):
    c = np.correlate(h, 1 - h, mode='full')
    m = 2 * n - 1
    lag = np.arange(-(n-1), n)
    for sweep in range(max_sweeps):
        improved = False
        old_max = float(np.max(c))
        for j in range(n):
            idx1 = j - lag
            idx2 = j + lag
            v1 = (idx1 >= 0) & (idx1 < n)
            v2 = (idx2 >= 0) & (idx2 < n)
            d_j = np.zeros(m)
            d_j[v1] = 1.0 - h[idx1[v1]]
            d_j[v2] -= h[idx2[v2]]
            new_val = optimize_coord(h, c, j, d_j)
            if abs(new_val - h[j]) > 1e-14:
                h[j] = new_val
                c = np.correlate(h, 1 - h, mode='full')
                improved = True
        if not improved:
            break
    return h

def obj_smooth(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 700:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def sum_con(h, n):
    return np.sum(h) - n / 2.0

def sum_con_jac(h, n):
    return np.ones(n)

def gen_binary(n, rng):
    arr = np.zeros(n)
    ones = rng.choice(n, size=n // 2, replace=False)
    arr[ones] = 1.0
    return arr

def gen_periodic(n, freq, phase):
    x = np.arange(n)
    return 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)

def gen_block(n, rng):
    h = np.zeros(n)
    k = rng.randint(n // 4, n // 2 + 1)
    start = rng.randint(0, n - k)
    h[start:start + k] = 2.0 * k / n
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 80
        h0 = np.full(n, 0.5)

    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]

    # Phase 1: Coordinate descent on parent
    h_cd = coord_descent(best_h.copy(), n, max_sweeps=100)
    h_cd = project_feasible(h_cd, n)
    c5 = compute_c5(h_cd, n)
    if c5 < best_c5:
        best_h, best_c5 = h_cd, c5

    # Phase 2: Multi-start diverse initializations
    n_starts = 60
    for s in range(n_starts):
        if budget_s - (time.time() - t0) < 80:
            break
        strat = s % 6
        if strat == 0:
            h_s = gen_binary(n, rng)
        elif strat == 1:
            h_s = gen_periodic(n, rng.randint(1, 12), rng.uniform(0, 2 * np.pi))
        elif strat == 2:
            h_s = best_h + rng.normal(0, 0.15, size=n)
            h_s = np.clip(h_s, 0, 1)
        elif strat == 3:
            h_s = rng.uniform(0, 1, size=n)
        elif strat == 4:
            h_s = gen_block(n, rng)
        else:
            h_s = gen_binary(n, rng)
            h_s = 0.3 * h_s + 0.7 * np.clip(best_h + rng.normal(0, 0.05, size=n), 0, 1)

        h_s = project_feasible(h_s, n)
        h_cd = coord_descent(h_s, n, max_sweeps=60)
        h_cd = project_feasible(h_cd, n)
        c5_s = compute_c5(h_cd, n)
        if c5_s < best_c5:
            best_h, best_c5 = h_cd, c5_s

    # Phase 3: SLSQP polish
    for alpha in [300, 1000, 5000, 15000]:
        if budget_s - (time.time() - t0) < 40:
            break
        try:
            res = minimize(obj_smooth, best_h, args=(n, alpha), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 8000, 'ftol': 1e-18})
            h_try = project_feasible(res.x, n)
            c5_try = compute_c5(h_try, n)
            if c5_try < best_c5:
                best_h, best_c5 = h_try, c5_try
        except Exception:
            pass

    # Phase 4: Iterative perturb + coordinate descent
    no_improve_count = 0
    scales = [0.002, 0.005, 0.01, 0.02, 0.03, 0.05, 0.08, 0.1, 0.15, 0.2]
    trial = 0
    while budget_s - (time.time() - t0) > 50:
        scale = scales[min(trial % len(scales), len(scales) - 1)]
        h_pert = best_h + rng.normal(0, scale, size=n)
        h_pert = project_feasible(h_pert, n)
        h_cd = coord_descent(h_pert, n, max_sweeps=40)
        h_cd = project_feasible(h_cd, n)
        c5_try = compute_c5(h_cd, n)
        if c5_try < best_c5 - 1e-14:
            best_h, best_c5 = h_cd, c5_try
            no_improve_count = 0
        else:
            no_improve_count += 1
        trial += 1
        if no_improve_count > 40:
            break

    # Phase 5: Final coordinate descent + SLSQP
    if budget_s - (time.time() - t0) > 20:
        h_final = coord_descent(best_h.copy(), n, max_sweeps=100)
        h_final = project_feasible(h_final, n)
        c5_f = compute_c5(h_final, n)
        if c5_f < best_c5:
            best_h, best_c5 = h_final, c5_f

    if budget_s - (time.time() - t0) > 15:
        try:
            res = minimize(obj_smooth, best_h, args=(n, 20000), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 10000, 'ftol': 1e-19})
            h_try = project_feasible(res.x, n)
            c5_try = compute_c5(h_try, n)
            if c5_try < best_c5:
                best_h, best_c5 = h_try, c5_try
        except Exception:
            pass

    # Final CD polish
    if budget_s - (time.time() - t0) > 5:
        h_final = coord_descent(best_h.copy(), n, max_sweeps=50)
        h_final = project_feasible(h_final, n)
        c5_f = compute_c5(h_final, n)
        if c5_f < best_c5:
            best_h, best_c5 = h_final, c5_f

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
