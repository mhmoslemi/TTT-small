        mid = (lo + hi) * 0.5
        s = float(np.clip(x - mid, 0.0, 1.0).sum())

    corr = np.correlate(h, 1.0 - h, mode='full')

    return np.correlate(h, 1.0 - h, mode='full')

def compute_grad_lag(h, L):
    n = len(h)
    omh = 1.0 - h
    grad = np.zeros(n)
    if L == 0:
        grad[:] = omh[:] - h[:]
    elif L > 0:
        grad[L:] += omh[:n - L]
        grad[:n - L] -= h[L:]
    else:
        L2 = -L
        grad[:n - L2] += omh[L2:]
        grad[L2:] -= h[:n - L2]
    return grad

    scaled = corr * dx
    m = np.max(scaled)
    shifted = scaled - m
    vals = np.exp(beta * shifted)
    return m + np.log(np.sum(vals)) / beta
