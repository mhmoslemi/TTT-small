def compute_c5(h, dx):

    return float(np.max(corr) * dx)

def project_sum(h, target_sum):
    """Project h onto {0 <= h <= 1, sum(h) = target_sum}"""
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-15:
        # Simple adjustment: add/subtract uniformly from non-boundary elements
        free = (h > 0) & (h < 1)
        if np.sum(free) > 0:
            adj = diff / np.sum(free)
            h[free] += adj
            h = np.clip(h, 0.0, 1.0)
            # Final adjustment on remaining free
            diff2 = target_sum - np.sum(h)
            if abs(diff2) > 1e-15:
                free2 = (h > 0) & (h < 1)
                if np.sum(free2) > 0:
                    h[free2] += diff2 / np.sum(free2)
                    h = np.clip(h, 0.0, 1.0)
    # Exact fix
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        mask = (h > 0.01) & (h < 0.99) if diff > 0 else (h > 0.01) & (h < 0.99)
        if np.sum(mask) == 0:
            mask = np.ones(len(h), dtype=bool)
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0, 1)
    return h

    return grad * dx

                best_c5 = c5

def two_coord_min(h, i, j, c_curr):
    """Find optimal h[i], h[j] given all others fixed, preserving sum."""
    n = len(h)
    s = h[i] + h[j]
    
    # Compute A_i[L] and A_j[L] for all L
    # A_i[L] = (1-h[i-L])*(i-L valid) - h[i+L]*(i+L valid)
    # We need this for L in range(-(n-1), n)
    
    # For efficiency, compute for all L at once
    # A_i[L] for lag L:
    #   (1-h[i-L]) if 0 <= i-L < n else 0
    #   - h[i+L] if 0 <= i+L < n else 0
    
    # Let's compute alpha[L] and beta[L] directly
    # c[L] is stored in c_curr with offset: c_curr[L + n - 1] = c[L]
    
    # A_i as a function of L:
    # For a given i, A_i[L] = (1 - h[i-L]) * [0<=i-L<n] - h[i+L] * [0<=i+L<n]
    
    # Let's compute this vector for all L
    # L ranges from -(n-1) to n-1, index in c_curr is L + n - 1
    
    L_vals = np.arange(-(n-1), n)
    
    # For position i:
    # i - L is valid when 0 <= i-L < n, i.e., i-n+1 <= L <= i
    # i + L is valid when 0 <= i+L < n, i.e., -i <= L <= n-1-i
    
    valid_il = (L_vals >= i - n + 1) & (L_vals <= i)  # i-L valid
    valid_ipl = (L_vals >= -i) & (L_vals <= n - 1 - i)  # i+L valid
    
    # h[i-L] for valid i-L:
    # When L = i - k, we get h[k] for k in [0, n-1]
    # So h[i-L] = h at index (i - L)
    h_il = np.zeros(2*n-1)
    idx_il = i - L_vals
    mask_il = (idx_il >= 0) & (idx_il < n)
    h_il[mask_il] = h[idx_il[mask_il]]
    
    h_ipl = np.zeros(2*n-1)
    idx_ipl = i + L_vals
    mask_ipl = (idx_ipl >= 0) & (idx_ipl < n)
    h_ipl[mask_ipl] = h[idx_ipl[mask_ipl]]
    
    A_i = (1 - h_il) * mask_il - h_ipl * mask_ipl
    
    # Similarly for j
    h_jl = np.zeros(2*n-1)
    idx_jl = j - L_vals
    mask_jl = (idx_jl >= 0) & (idx_jl < n)
    h_jl[mask_jl] = h[idx_jl[mask_jl]]
    
    h_jpl = np.zeros(2*n-1)
    idx_jpl = j + L_vals
    mask_jpl = (idx_jpl >= 0) & (idx_jpl < n)
    h_jpl[mask_jpl] = h[idx_jpl[mask_jpl]]
    
    A_j = (1 - h_jl) * mask_jl - h_jpl * mask_jpl
    
    # alpha[L] = c[L] - h[i]*A_i[L] + (s-h[j])*A_j[L]
    alpha = c_curr - h[i]*A_i + (s - h[j])*A_j
    beta = A_i - A_j
    
    # Minimize max_L (alpha[L] + t*beta[L]) over t in [max(0,s-1), min(1,s)]
    t_lo = max(0.0, s - 1.0)
    t_hi = min(1.0, s)
    
    if t_lo >= t_hi:
        return t_lo  # or t_hi, they're the same
    
    # Bisection on subgradient
    for _ in range(40):
        t_mid = (t_lo + t_hi) / 2
        vals = alpha + t_mid * beta
        max_val = np.max(vals)
        active = (vals >= max_val - 1e-12)

            min_beta = np.min(beta[active])
            max_beta = np.max(beta[active])
            if min_beta <= 0 <= max_beta:
                return t_mid
            elif max_beta < 0:
                t_lo = t_mid  # decreasing, go right
            else:
                t_hi = t_mid  # increasing, go left
        else:
            return t_mid
    
    return (t_lo + t_hi) / 2

def project_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0)

    if abs(diff) < 1e-14:
        return h
    if diff > 0:

                return h
            h[idx] = 1.0

        order = np.argsort(-h)
