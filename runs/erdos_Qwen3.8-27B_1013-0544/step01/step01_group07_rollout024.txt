import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val(corr, t):
    mc = np.max(corr)
    return mc + (1.0 / t) * np.log(np.sum(np.exp(t * (corr - mc))))

def compute_grad(h, t, dx):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    mc = np.max(corr)
    w = np.exp(t * (corr - mc))
    w /= np.sum(w)
    v = 1.0 - 2.0 * h
    M = np.lib.stride_tricks.sliding_window_view(w, n)
    return dx * (M @ v)

def project_feasible(y, ts):
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > ts:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    d = ts - h.sum()
    if abs(d) > 1e-12:
        h = h + d / len(h)
    return np.clip(h, 0.0, 1.0)

def pgd(h, dx, ts, t, iters, lr0=0.05, momentum=0.85):
    n = len(h)
    h = h.astype(float).copy()
    best = h.copy()
    best_c = compute_c5(h, dx)
    lr = lr0
    mom = np.zeros(n)
    for it in range(iters):
        g = compute_grad(h, t, dx)
        mom = momentum * mom + (1.0 - momentum) * g
        y = h - lr * mom
        hn = project_feasible(y, ts)
        c = compute_c5(hn, dx)
        if c < best_c:
            best_c = c
            best = hn.copy()
            lr = min(lr * 1.03, 0.8)
        else:
            lr = max(lr * 0.75, 0.001)
        h = hn
    return best, best_c

def binary_local_search(h, dx, ts, n_iter=30000, tabu_size=80, rng_seed=42):
    n = len(h)
    rng = np.random.RandomState(rng_seed)
    h_bin = (h > 0.5).astype(float)
    # Enforce exactly n//2 ones
    order = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[order[:n // 2]] = 1.0
    best = h_bin.copy()
    best_c = compute_c5(h_bin, dx)
    tabu = np.full(n, -1000, dtype=int)
    temp = 0.5
    for it in range(n_iter):
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # Tabu check
        if it - tabu[i] < tabu_size or it - tabu[j] < tabu_size:
            continue
        h_bin[i] = 0.0
        h_bin[j] = 1.0
        tabu[i] = it
        tabu[j] = it
        c = compute_c5(h_bin, dx)
        if c < best_c:
            best_c = c
            best = h_bin.copy()
            temp = max(temp * 0.99, 0.01)
        else:
            # Simulated annealing: accept with probability
            delta = c - best_c
            if rng.random() < np.exp(-delta / max(temp, 1e-6)):
                temp = max(temp * 0.98, 0.01)
            else:
                h_bin[i] = 1.0
                h_bin[j] = 0.0
    return best, best_c

def slsqp_groups(h, dx, ts, n_groups=50, gs=4, t_smooth=50, maxiter=80):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    for _ in range(n_groups):
        idx = np.random.choice(n, gs, replace=False)
        sub = best_h[idx].copy()
        fs = ts - (best_h.sum() - sub.sum())

        def sobj(x):
            hs = best_h.copy()
            hs[idx] = np.clip(x, 0.0, 1.0)
            c = np.correlate(hs, 1 - hs, mode='full')
            return smooth_max_val(c, t_smooth) * dx

        def scon(x):
            return np.sum(x) - fs

        try:
            res = minimize(sobj, sub, method='SLSQP',
                          bounds=[(0.0, 1.0)] * gs,
                          constraints=[{'type': 'eq', 'fun': scon}],
                          options={'maxiter': maxiter, 'ftol': 1e-14})
            old_c5 = best_c5
            old_h = best_h.copy()
            best_h[idx] = np.clip(res.x, 0.0, 1.0)
            best_h = project_feasible(best_h, ts)
            c5_new = compute_c5(best_h, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
            else:
                best_h = old_h
                best_c5 = old_c5
        except Exception:
            pass
    return best_h, best_c5

def slsqp_full(h, dx, ts, t_smooth=200, maxiter=500):
    n = len(h)

    def fobj(x):
        x = np.clip(x, 0.0, 1.0)
        c = np.correlate(x, 1 - x, mode='full')
        return smooth_max_val(c, t_smooth) * dx

    def fcon(x):
        return np.sum(x) - ts

    try:
        res = minimize(fobj, h, method='SLSQP',
                      bounds=[(0.0, 1.0)] * n,
                      constraints=[{'type': 'eq', 'fun': fcon}],
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_feasible(res.x, ts)
        return h_new, compute_c5(h_new, dx)
    except Exception:
        return h, compute_c5(h, dx)

    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)

    best_c5 = compute_c5(best_h, dx)

    for t in [3, 5, 10, 20, 50, 100, 200, 400]:

        hp, cp = pgd(best_h, dx, ts, t, iters=400, lr0=0.05)

            best_h, best_c5 = hp, cp

    if time.time() - t0 < budget_s * 0.55:
        remaining = budget_s * 0.55 - (time.time() - t0)
        nit = int(remaining * 2000)
        nit = max(nit, 5000)
        hb, cb = binary_local_search(best_h, dx, ts, n_iter=nit, tabu_size=80, rng_seed=seed)
        if cb < best_c5:
            best_h, best_c5 = hb, cb
        # Also try a second binary search with different seed
        if time.time() - t0 < budget_s * 0.55 - 10:
            hb2, cb2 = binary_local_search(best_h, dx, ts, n_iter=nit // 2, tabu_size=60, rng_seed=seed + 1)
            if cb2 < best_c5:
                best_h, best_c5 = hb2, cb2

    # Phase 3: PGD refinement from best found
    if time.time() - t0 < budget_s * 0.75:
        for t in [10, 30, 80, 150, 300]:
            if time.time() - t0 > budget_s * 0.72:
                break
            hp, cp = pgd(best_h, dx, ts, t, iters=300, lr0=0.03)
            if cp < best_c5:
                best_h, best_c5 = hp, cp

    # Phase 4: SLSQP group descent
    if time.time() - t0 < budget_s * 0.90:
        ng = int((budget_s * 0.90 - (time.time() - t0)) * 3)
        ng = max(ng, 15)
        best_h, best_c5 = slsqp_groups(best_h, dx, ts, n_groups=ng, gs=4, t_smooth=60, maxiter=80)

    # Phase 5: Final full SLSQP
    if time.time() - t0 < budget_s - 15:
        best_h, best_c5 = slsqp_full(best_h, dx, ts, t_smooth=300, maxiter=600)

    # Final project and evaluate
    best_h = project_feasible(best_h, ts)
    best_c5 = compute_c5(best_h, dx)
    return best_h, best_c5, n
