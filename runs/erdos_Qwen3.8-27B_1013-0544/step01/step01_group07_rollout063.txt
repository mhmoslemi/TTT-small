def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def swap_delta(h, a, b, n):
    m = np.arange(-(n - 1), n)
    delta = np.zeros(2 * n - 1)
    am = a + m
    mask = (am >= 0) & (am < n)
    delta[mask] += h[am[mask]]
    apm = a - m
    mask = (apm >= 0) & (apm < n)
    delta[mask] += h[apm[mask]]
    bm = b + m
    mask = (bm >= 0) & (bm < n)
    delta[mask] -= h[bm[mask]]
    bpm = b - m
    mask = (bpm >= 0) & (bpm < n)
    delta[mask] -= h[bpm[mask]]
    return delta

            mp = (pm >= 0) & (pm < n)

            vec[p, mm] += h[pmm[mm]]
