            infl_i = np.zeros(n_corr)
            vp = (idx_p >= 0) & (idx_p < n)
            infl_i[vp] = 1.0 - h[idx_p[vp]]
            vm = (idx_mn >= 0) & (idx_mn < n)
            infl_i[vm] -= h[idx_mn[vm]]
            idx_p2 = m + shifts
            idx_mn2 = m - shifts
            infl_m = np.zeros(n_corr)
            vp2 = (idx_p2 >= 0) & (idx_p2 < n)
            infl_m[vp2] = 1.0 - h[idx_p2[vp2]]
            vm2 = (idx_mn2 >= 0) & (idx_mn2 < n)
            infl_m[vm2] -= h[idx_mn2[vm2]]
            diff_infl = infl_i - infl_m
            base = corr
            hi_val = h[i]
            a, b = lo, hi
            for _ in range(n_ternary):
                m1 = a + (b - a) / 3.0
                m2 = b - (b - a) / 3.0
                f1 = np.max(base + (m1 - hi_val) * diff_infl)
                f2 = np.max(base + (m2 - hi_val) * diff_infl)
                if f1 < f2:
                    b = m2
                else:
                    a = m1
            x_opt = (a + b) / 2.0
            h[i] = x_opt
            h[m] = s - x_opt
            c5_new = compute_c5(h, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
    return h, best_c5

def slsqp_optimize(h, dx, target_sum, t, maxiter):
    n = len(h)
    obj = partial(smooth_max_val, t=t, dx=dx)
    con = partial(sum_constraint, target_sum=target_sum)
    try:
        res = minimize(
            obj, h.copy(),
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': con}],
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        return project_box_sum(res.x, target_sum)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = project_box_sum(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)
    best_n = n

    # Phase 1: Random pair descent (main workhorse)
    elapsed = time.time() - t0
    if elapsed < budget_s - 300:
        n_iter1 = min(200000, int((budget_s - elapsed - 300) / 0.00035))
        h_new, c5_new = pair_descent(best_h, dx, n_iter1, rng, n_ternary=40)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 2: SLSQP polish
    if time.time() - t0 < budget_s - 250:
        for t in [50, 150, 400]:
            if time.time() - t0 > budget_s - 200:
                break
            h_new = slsqp_optimize(best_h, dx, target_sum, t=t, maxiter=800)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    # Phase 3: Cyclic pair descent (structured sweep)
    if time.time() - t0 < budget_s - 200:
        n_rounds = min(50, int((budget_s - (time.time() - t0) - 200) / (n * 0.0004)))
        n_rounds = max(n_rounds, 10)
        h_new, c5_new = cyclic_pair_descent(best_h, dx, n_rounds, rng, n_ternary=35)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 4: More random pair descent
    if time.time() - t0 < budget_s - 150:
        n_iter2 = min(150000, int((budget_s - (time.time() - t0) - 150) / 0.00035))
        h_new, c5_new = pair_descent(best_h, dx, n_iter2, rng, n_ternary=40)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 5: High-precision SLSQP
    if time.time() - t0 < budget_s - 80:
        h_new = slsqp_optimize(best_h, dx, target_sum, t=1000, maxiter=3000)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 6: Final pair descent polish
    if time.time() - t0 < budget_s - 10:
        n_iter3 = min(80000, int((budget_s - (time.time() - t0) - 10) / 0.00035))
        h_new, c5_new = pair_descent(best_h, dx, n_iter3, rng, n_ternary=45)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Ensure valid output
    best_h = np.clip(best_h, 0.0, 1.0)
    dx_final = 2.0 / best_n
    best_c5 = compute_c5(best_h, dx_final)
    return best_h, best_c5, best_n
