import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta


def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h


def make_cosine_init(n, m):
    i = np.arange(n)
    return 0.5 + 0.5 * np.cos(2.0 * np.pi * m * i / n)


def make_thue_morse(n):
    tm = [0]
    while len(tm) < n:
        tm = tm + [1 - x for x in tm]
    return np.array(tm[:n], dtype=float)


def balance_binary(h, target):
    h = h.copy()
    n_ones = int(h.sum())
    if n_ones > target:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        h[idx[:n_ones - target]] = 0
    elif n_ones < target:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        h[idx[:target - n_ones]] = 1
    return h


def obj_smooth(h, beta):
    corr = np.correlate(h, 1.0 - h, mode="full")
    n = len(h)
    dx = 2.0 / n
    return smooth_max(corr, beta) * dx


def obj_penalized(h, beta, target_sum, penalty):
    corr = np.correlate(h, 1.0 - h, mode="full")
    n = len(h)
    dx = 2.0 / n
    sm = smooth_max(corr, beta) * dx
    sv = h.sum() - target_sum
    return sm + penalty * sv * sv


def con_eq(h, target_sum):
    return h.sum() - target_sum


def con_eq_jac(h):
    return np.ones(len(h))


def binary_local_search(h, n, target_sum, n_iter, seed):
    """Simulated annealing on binary solutions."""
    rng = np.random.RandomState(seed)
    h = h.copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    one_idx = np.where(h == 1)[0]
    zero_idx = np.where(h == 0)[0]
    
    T_start = 0.1
    T_end = 0.001
    for it in range(n_iter):
        T = T_start * (T_end / T_start) ** (it / max(1, n_iter - 1))
        # Swap a random 1 and 0
        i = one_idx[rng.randint(len(one_idx))]
        j = zero_idx[rng.randint(len(zero_idx))]
        h[i], h[j] = h[j], h[i]
        c5_new = compute_c5(h)
        delta = c5_new - c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            c5 = c5_new
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
            # Update indices
            one_idx = np.where(h == 1)[0]
            zero_idx = np.where(h == 0)[0]
        else:
            h[i], h[j] = h[j], h[i]  # Revert
    
    return best_h, best_c5

        c5n = compute_c5(h)
        if c5n < c5 or rng.random() < np.exp(-(c5n - c5) / max(T, 1e-9)):
            c5 = c5n
            if c5n < best_c5:
                best_h, best_c5 = h.copy(), c5n
            ia = np.where(h == 1)[0]
            ja = np.where(h == 0)[0]
        else:
            h[i], h[j] = h[j], h[i]
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
        best_n = len(best_h)
        best_h = project_h(best_h, best_n / 2.0)
        best_c5 = compute_c5(best_h)

    n_list = [60, 65, 70, 75, 80, 85, 90, 100]
    if best_n > 0 and best_n in n_list:
        n_list.remove(best_n)
        n_list.insert(0, best_n)

    for n in n_list:
        if time.time() - t0 > budget_s - 80:
            break
        tsum = n / 2.0
        bnds = Bounds(np.zeros(n), np.ones(n))
        inits = []
        for m in [1, 2, 3]:
            inits.append(project_h(make_cosine_init(n, m), tsum))
        inits.append(balance_binary(make_thue_morse(n), int(tsum)))
        inits.append(balance_binary((np.random.random(n) > 0.5).astype(float), int(tsum)))
        if best_n == n and best_h is not None:
            inits.append(best_h.copy())
        # anti-palindromic
        half = n // 2
        hh = np.random.random(half)
        hap = np.zeros(n)
        hap[:half] = hh
        hap[half:n - half] = 1.0 - hh[::-1]
        if n % 2 == 1:
            hap[half] = 0.5
        inits.append(project_h(ha_ if False else hap, tsum))

        for hi in inits:
            if time.time() - t0 > budget_s - 50:
                break
            hc = hi.copy()
            for beta in [25, 50, 100, 200]:
                if time.time() - t0 > budget_s - 40:
                    break
                try:
                    r = minimize(obj_pena, hc, args=(beta, tsum, 200.0),
                                 method='L-BFGS-B', bounds=bnds,
                                 options={'maxiter': 200, 'ftol': 1e-15})
                    hc = project_h(r.x, tsum)
                    c5v = compute_c5(hc)
                    if c5v < best_c5:
                        best_h, best_c5, best_n = hc.copy(), c5v, n
                except Exception:
                    pass
            # SLSQP polish
            if time.time() - t0 > budget_s - 35:
                break
            try:
                r = minimize(obj_smooth, hc, args=(150.0,), method='SLSQP',
                             bounds=bnds,
                             constraints=[{'type': 'eq', 'fun': con_eq,
                                          'jac': con_jac, 'args': (tsum,)}],
                             options={'maxiter': 500, 'ftol': 1e-14})
                hn = project_h(r.x, tsum)
                c5v = compute_c5(hn)
                if c5v < best_c5:
                    best_h, best_c5, best_n = hn.copy(), c5v, n
            except Exception:
                pass

    # Binary SA on best and on parent
    if best_h is not None and time.time() - t0 < budget_s - 40:
        n = best_n
        tsum_i = int(n / 2)
        hb = balance_binary((best_h > 0.5).astype(float), tsum_i)
        hb, c5b = binary_sa(hb, 30000, seed + 100)
        if c5b < best_c5:
            best_h, best_c5, best_n = hb.copy(), c5b, n

    # Also try binary SA on a fresh Thue-Morse at best_n
    if best_n > 0 and time.time() - t0 < budget_s - 25:
        n = best_n
        tsum_i = int(n / 2)
        hb2 = balance_binary(make_thue_morse(n), tsum_i)
        hb2, c5b2 = binary_sa(hb2, 20000, seed + 200)
        if c5b2 < best_c5:
            best_h, best_c5, best_n = hb2.copy(), c5b2, n

    # Final high-precision SLSQP
    if best_h is not None and time.time() - t0 < budget_s - 8:
        n = best_n
        tsum = n / 2.0
        bnds = Bounds(np.zeros(n), np.ones(n))
        try:
            r = minimize(obj_smooth, best_h, args=(300.0,), method='SLSQP',
                         bounds=bnds,
                         constraints=[{'type': 'eq', 'fun': con_eq,
                                      'jac': con_jac, 'args': (tsum,)}],
                         options={'maxiter': 3000, 'ftol': 1e-16})
            hn = project_h(r.x, tsum)
            c5v = compute_c5(hn)
            if c5v < best_c5:
                best_h, best_c5 = hn.copy(), c5v
        except Exception:
            pass
    return best_h, best_c5, best_n
