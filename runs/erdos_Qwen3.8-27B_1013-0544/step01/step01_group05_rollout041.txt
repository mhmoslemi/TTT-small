_global_target = None
_global_T = None

def project_feasible(x, target_sum):
    lo = float(x.min()) - 1.0
    hi = float(x.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = float(np.clip(x - mid, 0.0, 1.0).sum())
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def soft_max_val(corr, dx, T):
    c = corr * dx
    m = c.max()
    shifted = np.clip(T * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(shifted))) / T

def active_grad(h, dx, target, T_act):
    n = len(h)
    n_half = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = corr.max()
    w = np.exp(np.clip(T_act * (corr - c_max), -500, 0))
    ws = w.sum()
    if ws > 0:
        w = w / ws
    else:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
