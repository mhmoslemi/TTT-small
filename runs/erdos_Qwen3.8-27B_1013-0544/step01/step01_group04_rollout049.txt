            c5_new = compute_c5(h_new, dx)

            if c5_new < best_c5 - 1e-10:
                best_h = h_new.copy()
                best_c5 = c5_new
                # Polish
                h_pol = _slsqp_run(h_new, n, dx, target_sum, 300)
                c5_pol = compute_c5(h_pol, dx)
                if c5_pol < best_c5:
                    best_c5 = c5_pol
                    best_h = h_pol.copy()
            elif c5_new < best_c5 + 0.002:
                h_pol = _slsqp_run(h_new, n, dx, target_sum, 200)
                c5_pol = compute_c5(h_pol, dx)
                if c5_pol < best_c5:
                    best_c5 = c5_pol
                    best_h = h_pol.copy()

            attempt += 1

    # Phase 5: Final deep polish
    if elapsed() < budget_s - 15:
        h_final = _slsqp_run(best_h, n, dx, target_sum, 800)
        c5_final = compute_c5(h_final, dx)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final

        h_mm = _mm_slsqp_run(best_h, n, dx, target_sum, 500)
        c5_mm = compute_c5(h_mm, dx)
        if c5_mm < best_c5:
            best_h = h_mm
            best_c5 = c5_mm
            h_final = _slsqp_run(best_h, n, dx, target_sum, 400)
            c5_final = compute_c5(h_final, dx)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, dx)
    return (best_h, best_c5, len(best_h))
