import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target_sum):
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-14:
        h = h + diff / len(h)
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(h, t, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    max_c = np.max(corr)
    shifted = t * (corr - max_c)
    return (max_c + (1.0/t) * np.log(np.sum(np.exp(shifted)))) * dx

def slsqp_refine(h, dx, target_sum, t, maxiter):
    n = len(h)
    try:
        res = minimize(
            lambda x: smooth_max_obj(x, t, dx),
            h.copy(),
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            options={'maxiter': maxiter, 'ftol': 1e-10}
        )
        return fix_sum(res.x, target_sum)
    except:
        return h.copy()

def binary_swap_search(h, n_iters, seed):
    """Local search on binary sequences via pair swaps"""
    n = len(h)
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    
    for _ in range(n_iters):
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        trial = best_h.copy()
        trial[i] = 0
        trial[j] = 1
        c5 = compute_c5(trial, dx)
        if c5 < best_c5:
            best_h = trial
            best_c5 = c5
    
    return best_h, best_c5

            zero_indices = np.where(h_bin == 0)[0]
            order = np.argsort(-h[zero_indices])[:deficit]

    # Update with momentum
    v = momentum * v - lr * grad  # Nesterov-like
    h_new = h + v
    # Project onto feasible set
    h_new = project(h_new, target_sum)
    return h_new, v

def compute_gradient(h, t, dx):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    max_c = np.max(corr)
    shifted = t * (corr - max_c)
    weights = np.exp(shifted) / np.sum(np.exp(shifted))
    
    # weights[m] corresponds to c_{m-(n-1)}
    grad = np.zeros(n)
    
    for idx in range(2*n - 1):
        w = weights[idx]
        if w < 1e-12:
            continue
        k = idx - (n - 1)  # actual lag
        
        if k >= 0:
            # d(c_k)/d(h[i]) = (1-h[i-k]) for i in [k, n-1]
            #                        - h[i+k]  for i in [0, n-k-1]
            # First term: i from k to n-1
            i_range1 = np.arange(k, n)
            grad[i_range1] += w * (1 - h[i_range1 - k])
            # Second term: i from 0 to n-k-1
            if n - k > 0:
                i_range2 = np.arange(0, n - k)
                grad[i_range2] -= w * h[i_range2 + k]
        else:
            k_neg = -k  # positive
            # d(c_{-k_neg})/d(h[i]) = (1-h[i+k_neg]) for i in [0, n-k_neg-1]
            #                               - h[i-k_neg]  for i in [k_neg, n-1]
            if n - k_neg > 0:
                i_range1 = np.arange(0, n - k_neg)
                grad[i_range1] += w * (1 - h[i_range1 + k_neg])
            i_range2 = np.arange(k_neg, n)
            grad[i_range2] -= w * h[i_range2 - k_neg]
    
    return grad * dx

            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
            options={'maxiter': maxiter, 'ftol': 1e-11}
        )
        return fix_sum(res.x, target_sum)
    except Exception:
        return h.copy()

def binary_swap_search(h, n_iters, seed):
    n = len(h)
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    no_improve = 0
    for _ in range(n_iters):
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        trial = best_h.copy()
        trial[i] = 0.0
        trial[j] = 1.0
        c5 = compute_c5(trial, dx)
        if c5 < best_c5:
            best_h = trial
            best_c5 = c5
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                break
    return best_h, best_c5

def make_binary(h, target_count):
    h_bin = np.zeros(len(h))
    top_indices = np.argsort(-h)[:target_count]
    h_bin[top_indices] = 1.0
    return h_bin

def upsample(h, new_n):
    x_old = np.linspace(0, 2, len(h) + 1)[:-1]
    x_new = np.linspace(0, 2, new_n + 1)[:-1]
    return np.interp(x_new, x_old, h)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    h0 = np.array(initial_h_values, dtype=float).copy()
    n0 = len(h0)
    dx0 = 2.0 / n0
    target0 = n0 / 2.0
    h0 = fix_sum(h0, target0)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx0)
    best_n = n0
    best_dx = dx0

    # Phase 1: Accelerated projected gradient with annealing on initial
    for t in [5, 10, 20, 50, 100, 200, 400]:
        if elapsed() > budget_s - 200:
            break
        h_new = accelerated_pg(best_h, t, best_dx, best_n / 2.0, n_iters=300, lr_init=0.02, momentum=0.9)
        c5_new = compute_c5(h_new, best_dx)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 2: SLSQP polish
    for t in [50, 150, 400]:
        if elapsed() > budget_s - 150:
            break
        h_new = slsqp_refine(best_h, best_dx, best_n / 2.0, t=t, maxiter=600)
        c5_new = compute_c5(h_new, best_dx)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 3: Binary search + continuous refinement at multiple sizes
    for n_trial in [best_n, best_n // 2, best_n // 2 + 20, best_n + 40]:
        if elapsed() > budget_s - 120 or n_trial < 40:
            continue
        dx_t = 2.0 / n_trial
        target_t = n_trial / 2.0

        if n_trial == best_n:
            h_start = best_h
        elif n_trial < best_n:
            h_start = best_h[:n_trial * (best_n // n_trial)].reshape(-1, best_n // n_trial).mean(axis=1)
        else:
            h_start = upsample(best_h, n_trial)
        h_start = fix_sum(h_start, target_t)

        # Binary initialization from continuous
        h_bin = make_binary(h_start, int(target_t))
        h_bin, c5_bin = binary_swap_search(h_bin, n_iters=3000, seed=seed + n_trial)

        # Continuous refinement from binary
        h_ref = h_bin.copy()
        for t in [5, 15, 40, 100, 250]:
            if elapsed() > budget_s - 60:
                break
            h_ref = accelerated_pg(h_ref, t, dx_t, target_t, n_iters=200, lr_init=0.02, momentum=0.9)
        h_ref = slsqp_refine(h_ref, dx_t, target_t, t=300, maxiter=400)
        c5_ref = compute_c5(h_ref, dx_t)
        if c5_ref < best_c5:
            best_h, best_c5, best_n, best_dx = h_ref, c5_ref, n_trial, dx_t

        # Fresh random binary
        if elapsed() < budget_s - 80:
            rng2 = np.random.RandomState(seed + 7000 + n_trial)
            h_rbin = np.zeros(n_trial)
            h_rbin[rng2.choice(n_trial, int(target_t), replace=False)] = 1.0
            h_rbin, _ = binary_swap_search(h_rbin, n_iters=2000, seed=seed + 9000 + n_trial)
            h_r = h_rbin.copy()
            for t in [10, 50, 150]:
                if elapsed() > budget_s - 40:
                    break
                h_r = accelerated_pg(h_r, t, dx_t, target_t, n_iters=200, lr_init=0.02, momentum=0.9)
            h_r = slsqp_refine(h_r, dx_t, target_t, t=200, maxiter=400)
            c5_r = compute_c5(h_r, dx_t)
            if c5_r < best_c5:
                best_h, best_c5, best_n, best_dx = h_r, c5_r, n_trial, dx_t

    # Phase 4: Multiple random restarts with APG
    for trial in range(8):
        if elapsed() > budget_s - 30:
            break
        n_r = np.random.choice([best_n, best_n // 2, best_n + 20])
        dx_r = 2.0 / n_r
        target_r = n_r / 2.0
        rng3 = np.random.RandomState(seed + 50000 + trial)
        h_rand = rng3.uniform(0, 1, n_r)
        h_rand = fix_sum(h_rand, target_r)
        # Push toward binary
        h_half = (h_rand > 0.5).astype(float)
        h_half = fix_sum(h_half, target_r)
        hx = 0.5 * h_rand + 0.5 * h_half
        hx = fix_sum(hx, target_r)
        for t in [10, 50, 150]:
            if elapsed() > budget_s - 15:
                break
            hx = accelerated_pg(hx, t, dx_r, target_r, n_iters=200, lr_init=0.02, momentum=0.9)
        hx = slsqp_refine(hx, dx_r, target_r, t=200, maxiter=300)
        c5_x = compute_c5(hx, dx_r)
        if c5_x < best_c5:
            best_h, best_c5, best_n, best_dx = hx, c5_x, n_r, dx_r

    # Final polish
    if elapsed() < budget_s - 5:
        h_final = slsqp_refine(best_h, best_dx, best_n / 2.0, t=500, maxiter=1500)
        c5_final = compute_c5(h_final, best_dx)
        if c5_final < best_c5:
            best_h, best_c5 = h_final, c5_final

    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = compute_c5(best_h, 2.0 / best_n)
    return best_h, final_c5, best_n
