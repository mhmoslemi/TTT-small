from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-10
    else:
        mask = h > 1e-10
    cnt = mask.sum()
    if cnt == 0:
        return h
    step = diff / cnt
    h[mask] += step
    return np.clip(h, 0.0, 1.0)

def smooth_obj_grad(h, tau, topk=None, lam=0.0):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    nc = len(corr)
    if topk is not None and topk < nc:
        idx = np.argsort(corr)[-topk:]
    else:
        idx = np.arange(nc)
    cs = corr[idx]
    x = tau * cs
    xm = float(x.max())
    sh = x - xm
    ws = np.exp(sh)
    wsum = float(ws.sum())
    obj = xm / tau + np.log(wsum) / tau
    if lam > 0:
        s = h.sum() - n/2.0
        obj += lam * s * s
    wt = np.zeros(nc)
    wt[idx] = ws / wsum
    g = np.zeros(n)
    for i in range(len(idx)):
        w = wt[idx[i]]
        if w < 1e-15:
            continue
        m = idx[i] - (n-1)
        if m >= 0:
            g[:n-m] += w * (1.0 - h[m:])
            g[m:] -= w * h[:n-m]
        else:
            p = -m
            g[p:] += w * (1.0 - h[:n-p])
            g[:n-p] -= w * h[p:]
    if lam > 0:
        s = h.sum() - n/2.0
        g += 2.0 * lam * s
    return obj, g

                    break

                    h = project_feasible(h, target)
        if not improved:
            break
    return h, c5

Let me also make sure I'm computing C_5 correctly. The formula is:
C_5 = max_k ∫ h(x)(1-h(x+k)) dx

In the discrete case:
C_5 = max_k (∑_i h[i] * (1-h[i+k]) * dx)

where the sum is over valid i (both indices in range). This is exactly what np.correlate(h, 1-h, mode="full") * dx gives (taking the max).

Wait, I need to be careful. np.correlate(h, 1-h, mode="full") computes:
result[k] = ∑_i h[i] * (1-h)[i + k - (len(h)-1)]

So result[k] = ∑_i h[i] * (1 - h[i + k - (n-1)])

Let m = k - (n-1). Then:
result[k] = ∑_i h[i] * (1 - h[i+m]) for valid i

This is the cross-correlation at lag m. The max over all k (equivalently, over all m from -(n-1) to (n-1)) gives the maximum overlap.

So C_5 = max(result) * dx. This is what evaluate_c5 computes. ✓

        obj += lam * s * s

            g[:n - p] -= w * h[p:]
    if lam > 0:

        g += 2.0 * lam * s
    return obj, g

def coord_descent(h, n_rounds=3):
    n = len(h)
    target = n / 2.0
    c5 = evaluate_c5(h)
    deltas = [0.02, -0.02, 0.05, -0.05, 0.01, -0.01, 0.1, -0.1]
    for _ in range(n_rounds):
        improved = False
        for i in range(n):
            old = h[i]
            for d in deltas:
                nv = old + d
                if nv < 0 or nv > 1:
                    continue
                h[i] = nv
                h = project_feasible(h, target)
                c5n = evaluate_c5(h)
                if c5n < c5:
                    c5 = c5n
                    improved = True
                    break
                else:
                    h[i] = old
                    h = project_feasible(h, target)
        if not improved:
            break
    return h, c5
