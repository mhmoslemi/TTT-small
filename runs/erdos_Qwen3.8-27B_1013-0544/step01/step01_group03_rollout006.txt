            best_c5 = c5_new

    # Phase 2: SLSQP polish
    for tau in [500, 2000, 10000, 50000]:
        if rem() < 25:
            break
        h_new, c5_new = slsqp_opt(best_h, n0, tau, 500)
        if c5_new is not None and c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 3: Trust-constr
    if rem() > 60:
        for tau in [1000, 5000, 20000]:
            if rem() < 20:
                break
            h_new, c5_new = trust_constr_opt(best_h, n0, tau, 300)
            if c5_new is not None and c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    # Phase 4: Higher resolution
    for tn in [160, 200, 240]:
        if rem() < 80:
            break
        target = tn / 2.0
        h_up = np.interp(np.linspace(0, 1, tn), np.linspace(0, 1, len(best_h)), best_h)
        h_up = project_sum(h_up, target)
        for tau in [500, 2000, 8000]:
            if rem() < 20:
                break
            h_new, c5_new = nesterov_pg(h_up, tau, 2000, 0.02)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
            h_up = h_new
        if rem() > 30:
            h_new, c5_new = slsqp_opt(best_h, len(best_h), 5000, 300)
            if c5_new is not None and c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    # Phase 5: Diverse restarts at best resolution
    hn = len(best_h)
    target_hn = hn / 2.0
    for trial in range(30):
        if rem() < 12:
            break
        if trial % 4 == 0:
            hp = np.random.uniform(0.1, 0.9, hn)
        elif trial % 4 == 1:
            hp = 0.5 + 0.3 * np.sin(2 * np.pi * (2 + trial) * np.linspace(0, 1, hn))
            hp = np.clip(hp, 0, 1)
        elif trial % 4 == 2:
            hp = best_h + np.random.normal(0, 0.08, hn)
        else:
            hp = best_h
            idx = np.random.choice(hn, size=hn // 5, replace=False)
            hp[idx] = np.clip(hp[idx] + np.random.uniform(-0.3, 0.3, len(idx)), 0, 1)
        hp = project_sum(hp, target_hn)
        h_new, c5_new = nesterov_pg(hp, 2000, 1500, 0.02)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
        if rem() > 15:
            h_new2, c5_new2 = slsqp_opt(h_new, hn, 5000, 200)
            if c5_new2 is not None and c5_new2 < best_c5:
                best_h = h_new2
                best_c5 = c5_new2

    # Phase 6: Final fine-tuning
    if rem() > 15:
        hn_final = len(best_h)
        for tau in [20000, 100000, 500000]:
            if rem() < 10:
                break
            h_new, c5_new = slsqp_opt(best_h, hn_final, tau, 300)
            if c5_new is not None and c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
        h_new, c5_new = nesterov_pg(best_h, 50000, 5000, 0.005)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Final projection
    hn_final = len(best_h)
    best_h = project_sum(best_h, hn_final / 2.0)
    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, hn_final
