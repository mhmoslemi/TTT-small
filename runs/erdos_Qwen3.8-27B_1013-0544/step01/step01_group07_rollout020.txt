The sum constraint makes coordinate descent tricky. If I change h[i], I need to adjust another coordinate to maintain the sum. This is like a "swap" operation.

Let me use a different approach for the polish: **paired coordinate descent**.

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def c5_fft(h):
    n = len(h)
    dx = 2.0 / n
    H = np.fft.rfft(h, n=2*n)
    ac = np.fft.irfft(H * np.conj(H), n=2*n)[:n]
    return float(np.max((np.sum(h) - ac[1:]) * dx))

def project_feasible(h, S):
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0, 1)

def pgd_solve(h, S, t_sched, alpha0, mom, t_stop):
    n = len(h)
    dx = 2.0 / n
    bh = h.copy()
    bc = c5_fft(h)
    h = h.astype(float).copy()
    vel = np.zeros(n)
    it = 0
    for temp, n_it in t_sched:
        for _ in range(n_it):
            if time.time() > t_stop:
                return bh, bc
            it += 1
            H = np.fft.rfft(h, n=2*n)
            ac = np.fft.irfft(H * np.conj(H), n=2*n)[:n]
            sh = np.sum(h)
            C = np.zeros(n)
            C[1:] = (sh - ac[1:]) * dx
            mc = C[1:].max()
            eg = np.exp(temp * (C[1:] - mc))
            w = np.zeros(n)
            w[1:] = eg / eg.sum()
            W = np.fft.rfft(w, n=n)
            c1 = np.fft.irfft(W * H, n=n)
            wr = np.zeros(n)
            wr[1:] = w[1:][::-1]
            Wr = np.fft.rfft(wr, n=n)
            c2 = np.fft.irfft(Wr * H, n=n)
            grad = dx * (1.0 - c1 - c2)
            a = alpha0 / np.sqrt(1.0 + 0.01 * it)
            vel = mom * vel - a * grad
            h = project_feasible(h + vel, S)
            c = c5_fft(h)
            if c < bc:
                bc = c
                bh = h.copy()
    return bh, bc
