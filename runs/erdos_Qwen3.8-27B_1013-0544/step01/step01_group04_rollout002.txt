import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_all_lags(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = np.sum(h) - target
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(free):
            h[free] -= diff / np.sum(free)
        else:
            h -= diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def active_gradient(h, n, dx, top_k=3):
    lags = compute_all_lags(h, dx)
    max_val = np.max(lags)
    threshold = max_val - 1e-10
    active = np.where(lags >= threshold)[0]
    if len(active) < top_k:
        active = np.argsort(lags)[-top_k:]
    else:
        active = active[:top_k]
    
    n_lag = 2 * n - 1
    ext_n = n - 1
    grad = np.zeros(n)
    
    for m in active:
        s = m - ext_n
        if s >= 0:
            lim = n - s
            if lim > 0:
                grad[:lim] += (1.0 - h[s:]) * dx
            if s < n:
                grad[s:] -= h[:lim] * dx
        else:
            s2 = -s
            lim2 = n - s2
            if s2 < n:
                grad[s2:] += (1.0 - h[:lim2]) * dx
                grad[:lim2] -= h[s2:] * dx
