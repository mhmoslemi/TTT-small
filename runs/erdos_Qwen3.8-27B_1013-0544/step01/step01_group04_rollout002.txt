            best_c5 = c5_new
            sa_accepts += 1
        elif c5_new - best_c5 < current_temp and rng.rand() < np.exp(-(c5_new - best_c5) / max(current_temp, 1e-10)):
            best_h = h_new
            sa_accepts += 1

    # Phase 2: trust-constr + SLSQP alternating with diverse starts
    phase2_end = budget_s * 0.80
    attempt = 0
    while elapsed() < phase2_end:
        if attempt % 3 == 0:
            h_try = best_h + rng.randn(n) * 0.03
        elif attempt % 3 == 1:
            h_try = block_perturb(best_h, n, rng, num_blocks=3, scale=0.2)
        else:
            h_try = binary_push(best_h, n, rng, frac=0.3)
        h_try = project_feasible(h_try, n)
        
        # Alternate between trust-constr and SLSQP
        if attempt % 2 == 0:
            h_new = trust_constr_optimize(h_try, n, dx, target_sum, max_iter=150)
        else:
            h_new = slsqp_optimize(h_try, n, dx, target_sum, max_iter=300)
        
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5 - 1e-14:
            best_h = h_new
            best_c5 = c5_new
        attempt += 1

    # Phase 3: Fine polishing with very small perturbations
    phase3_end = budget_s - 15
    attempt3 = 0
    while elapsed() < phase3_end:
        h_try = best_h + rng.randn(n) * 0.008
        h_try = project_feasible(h_try, n)
        h_new = slsqp_optimize(h_try, n, dx, target_sum, max_iter=500, ftol=1e-15)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5 - 1e-14:
            best_h = h_new
            best_c5 = c5_new
        attempt3 += 1

    # Final verification
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, dx)

    return (best_h, best_c5, n)
