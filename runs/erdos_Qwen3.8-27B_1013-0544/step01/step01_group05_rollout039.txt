_g_target = None

def project_feasible(x, target_sum):
    lo = -2.0
    hi = 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = float(np.clip(x - mid, 0.0, 1.0).sum())
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def corr_grad(h, L):
    n = len(h)
    g = np.zeros(n)
    if L == 0:
        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[L:] += (1.0 - h[:n - L])
        g[:n - L] -= h[L:]
    else:
        L2 = -L
        g[:n - L2] += (1.0 - h[L2:])
        g[L2:] -= h[:n - L2]
    return g

def lse_gradient(h, dx, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    mc = c.max()
    sh = alpha * (c - mc)
    es = np.exp(sh)
    se = es.sum()
    w = es / se
    n = len(h)
    nh = n - 1
    grad = np.zeros(n)
    idxs = np.where(w > 1e-7)[0]
    for k in idxs:
        L = k - nh
        grad += w[k] * corr_grad(h, L)
    return grad * dx
