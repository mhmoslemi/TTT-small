    x_max = np.max(x)

    h = np.array(h, dtype=float)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return np.clip(h, 0.0, 1.0)
    if diff > 0:

    else:

    cnt = mask.sum()

def project_onto_simplex(h, target):
    """Project h onto {0<=h<=1, sum(h)=target}"""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:

    else:

    if cnt > 0:

        # Need to ensure we don't go out of bounds

            step = min(step, max_step)
        else:
            min_step = h[mask].min()
            step = max(step, -min_step)
        h[mask] += step
    return np.clip(h, 0.0, 1.0)
