import numpy as np
from scipy.optimize import minimize, Bounds
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_sum(h, target):
    """Project h to have sum = target, maintaining [0,1] bounds."""
    h = h.copy()
    for _ in range(10):
        err = target - h.sum()
        if abs(err) < 1e-12:
            break
        if err > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += err / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def cd_sweep(h, n):
    """One full sweep of coordinate descent."""
    for i in range(n):
        old = h[i]
        # Compute c for h[i]=0 and h[i]=1
        h[i] = 0.0
        c0 = np.correlate(h, 1.0 - h, mode='full')
        h[i] = 1.0
        c1 = np.correlate(h, 1.0 - h, mode='full')
        h[i] = old
        
        B = c1 - c0
        A = c0  # c at t=0
        
        # Ternary search for t* in [0,1] minimizing max(A + t*B)
        lo, hi = 0.0, 1.0
        for _ in range(40):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            if np.max(A + m1 * B) < np.max(A + m2 * B):
                hi = m2
            else:
                lo = m1
        h[i] = (lo + hi) / 2.0
    
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    # Get initial values

    else:
        h_init = np.full(70, 0.5)
    
    n0 = len(h_init)
    best_h = h_init.copy()
    best_c5 = evaluate_c5(best_h)
    
    def rem():
        return budget_s - (time.time() - t0)
    
    # Phase 1: Coordinate descent from parent
    for sweep in range(200):
        if rem() < 30:
            break
        h = best_h.copy()
        h = cd_sweep(h, n0)
        h = project_sum(h, n0 / 2.0)
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5

        bnds = Bounds(np.zeros(n0), np.ones(n0))
        tau = 5000
        def obj(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            x_tau = tau * corr
            x_max = np.max(x_tau)
            return float(x_max / tau + np.log(np.sum(np.exp(x_tau - x_max))) / tau)
        def obj_grad(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            # gradient of smooth max
            x_tau = tau * corr
            x_max = np.max(x_tau)
            weights = np.exp(x_tau - x_max)
            weights /= weights.sum()
            # d(smooth_max)/d(corr[k]) = weights[k]
            # d(corr[k])/dx[j] ... complex, skip gradient
            return None
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                 'jac': lambda x: np.ones(n0)}]
        res = minimize(obj, best_h, method='SLSQP', bounds=bnds,
                      constraints=cons, options={'maxiter': 500, 'ftol': 1e-14})

        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:

            best_c5 = c5_new

        h_t = project_sum(h_t, target)
        for sweep in range(100):

            if c5 < best_c5:

                best_c5 = c5

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.array(h, dtype=float)
    for _ in range(20):
        err = target - h.sum()
        if abs(err) < 1e-14:
            return h
        if err > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            return h
        step = err / cnt
        h[mask] += step
        np.clip(h, 0.0, 1.0, out=h)
    return h

def cd_sweep(h, reverse=False):
    n = len(h)
    indices = range(n-1, -1, -1) if reverse else range(n)
    for i in indices:
        old = h[i]
        h[i] = 0.0
        c0 = np.correlate(h, 1.0 - h, mode='full')
        h[i] = 1.0
        c1 = np.correlate(h, 1.0 - h, mode='full')
        h[i] = old
        B = c1 - c0
        A = c0
        # Ternary search
        lo, hi = 0.0, 1.0
        for _ in range(45):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            if np.max(A + m1 * B) <= np.max(A + m2 * B):
                hi = m2
            else:
                lo = m1
        h[i] = (lo + hi) * 0.5
    return h

    x = tau * corr
    x_max = np.max(x)
    return float(x_max / tau + np.log(np.sum(np.exp(x - x_max))) / tau)

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.array(h, dtype=float)
    for _ in range(30):
        err = target - h.sum()
        if abs(err) < 1e-15:
            return h
        if err > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = mask.sum()
        if cnt == 0:
            return h
        h[mask] += err / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h

def cd_sweep(h, reverse=False):
    n = len(h)
    rng = range(n - 1, -1, -1) if reverse else range(n)
    for i in rng:
        old = h[i]
        h[i] = 0.0
        c0 = np.correlate(h, 1.0 - h, mode='full')
        h[i] = 1.0
        c1 = np.correlate(h, 1.0 - h, mode='full')
        h[i] = old
        B = c1 - c0
        A = c0
        lo, hi = 0.0, 1.0
        for _ in range(50):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            if np.max(A + m1 * B) <= np.max(A + m2 * B):
                hi = m2
            else:
                lo = m1
        h[i] = (lo + hi) * 0.5
    return h

def smooth_max_val(corr, tau):
    x = tau * corr
    x_max = np.max(x)
    return float(x_max / tau + np.log(np.sum(np.exp(x - x_max))) / tau)

def cd_until_stall(h, n, max_sweeps, tol=1e-12):
    best_c = evaluate_c5(h)
    stall = 0
    for s in range(max_sweeps):
        rev = (s % 2 == 1)
        h2 = cd_sweep(h, reverse=rev)
        h2 = project_sum(h2, n / 2.0)
        c2 = evaluate_c5(h2)
        if c2 < best_c - tol:
            h = h2
            best_c = c2
            stall = 0
        else:
            stall += 1
            if stall >= 30:
                break
    return h, best_c

        return budget_s - (time.time() - t0)

    # Phase 1: Coordinate descent from parent (main workhorse)
