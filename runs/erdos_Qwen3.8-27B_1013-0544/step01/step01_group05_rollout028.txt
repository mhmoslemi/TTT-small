from scipy.optimize import minimize
import time

_global_dx = None
_global_target_sum = None

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(x, target_sum):
    lo = float(x.min()) - 1.0
    hi = float(x.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = float(np.clip(x - mid, 0.0, 1.0).sum())
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def compute_grad_lag(h, L):
    n = len(h)
    omh = 1.0 - h
    grad = np.zeros(n)
    if L == 0:
        grad[:] = omh[:] - h[:]
    elif L > 0:
        grad[L:] += omh[:n - L]
        grad[:n - L] -= h[L:]
    else:
        L2 = -L
        grad[:n - L2] += omh[L2:]
        grad[L2:] -= h[:n - L2]
    return grad

def smooth_obj_grad(h, dx, beta):
    """Compute log-sum-exp smoothed objective and its gradient."""
    n = len(h)
    n_half = n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = corr * dx
    m = np.max(scaled)
    exp_term = np.exp(beta * (scaled - m))
    sum_exp = np.sum(exp_term)
    obj = (m + np.log(sum_exp)) / beta
    w = exp_term / sum_exp
    
    grad = np.zeros(n)
    for k in range(len(corr)):
        if w[k] < 1e-14:
            continue
        L = k - n_half
        grad += w[k] * compute_grad_lag(h, L) * dx
    return obj, grad

    return np.sum(h) - _global_target_sum

def _smooth_eq_init(h):
    return np.sum(h) - _global_target_sum

def slsqp_smooth(h0, dx, target_sum, beta, maxiter=3000):
    """Run SLSQP with smooth objective."""
    global _global_target_sum
    _global_target_sum = target_sum
    
    n = len(h0)
    
    def fun(x):
        return smooth_obj_grad(x, dx, beta)[0]
    
    def jac(x):
        return smooth_obj_grad(x, dx, beta)[1]
    
    res = minimize(fun, h0, jac=jac, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': _smooth_eq}],
                   options={'maxiter': maxiter, 'ftol': 1e-14})

        h_res = project_feasible(res.x, target_sum)
        return h_res, compute_c5(h_res, dx)
    return h0, compute_c5(h0, dx)

def active_set_gradient(h, dx, target_sum, step_size, n_iters, tol=1e-4, max_active=10):
    """Active-set projected gradient refinement."""
    n = len(h)
    n_half = n - 1
    best_c5 = compute_c5(h, dx)
    momentum = np.zeros(n)
    mu = 0.5
    
    for _ in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = float(np.max(corr))
        active = np.where(corr >= c_max - tol)[0]
        if len(active) < 2:
            active = np.argsort(corr)[-2:]
        elif len(active) > max_active:
            active = np.argsort(corr)[-max_active:]
        
        grad = np.zeros(n)
        for idx in active:
            L = int(idx) - n_half
            grad += compute_grad_lag(h, L)
        grad /= float(len(active))
        
        momentum = mu * momentum - step_size * grad

        c5_new = compute_c5(h_new, dx)

        else:
            step_size *= 0.9
            if step_size < 1e-12:

        h = h_new
    
    return h, best_c5

    hi = float(x.max()) + 1.0

        if s > target_sum:

        else:

    n = len(h)
    omh = 1.0 - h
    grad = np.zeros(n)
    if L == 0:
        grad[:] = omh[:] - h[:]
    elif L > 0:
        grad[L:] += omh[:n - L]
        grad[:n - L] -= h[L:]
    else:
        L2 = -L
        grad[:n - L2] += omh[L2:]
        grad[L2:] -= h[:n - L2]
