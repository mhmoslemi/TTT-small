import numpy as np
from scipy.optimize import minimize
import time

def project_onto_set(y, target_sum):
    """Project y onto {0<=p<=1, sum(p)=target_sum}"""
    n = len(y)
    # Bisection on lambda
    lo = y.min() - 1
    hi = y.max() + 1
    for _ in range(60):
        mid = (lo + hi) / 2
        p = np.clip(y - mid, 0, 1)
        s = p.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    p = np.clip(y - (lo+hi)/2, 0, 1)
    return p

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_all_corrs(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return corr

def gradient_at_lag(h, lag):
    """Compute gradient of c(lag) = sum h[i](1-h[i+lag]) * dx"""
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    # For index t: grad[t] = (1-h[t+lag]) * 1_{t+lag in [0,n-1]} - h[t-lag] * 1_{t-lag in [0,n-1]}
    t = np.arange(n)
    # First term: t+lag in [0, n-1]
    mask1 = (t + lag >= 0) & (t + lag < n)
    g[mask1] += (1.0 - h[t[mask1] + lag])
    # Second term: t-lag in [0, n-1]
    mask2 = (t - lag >= 0) & (t - lag < n)
    g[mask2] -= h[t[mask2] - lag]
    return g * dx

    best_h = h.copy()

    alpha = 1.0
    h = h.copy()

        corrs = compute_all_corrs(h)
        c5 = corrs.max()
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        if it > 100 and (best_c5 - c5) < tol:  # Actually check improvement
            pass  # Keep going
        
        # Find active lags (within epsilon of max)
        max_val = corrs.max()
        active = np.where(corrs >= max_val - 1e-6)[0]
        
        # Average gradient over active lags
        g = np.zeros(n)
        for m in active:
            lag = m - (n - 1)
            g += gradient_at_lag(h, lag)
        g /= len(active)
        
        # Backtracking line search
        accepted = False
        for _ in range(20):

from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_sum(y, target):
    """Project y onto {0<=p<=1} while adjusting sum to target."""
    n = len(y)
    y = np.clip(y, 0, 1)
    diff = target - y.sum()
    if abs(diff) < 1e-12:
        return y
    if diff > 0:
        mask = y < 1.0
    else:
        mask = y > 0.0
    cnt = mask.sum()
    if cnt == 0:
        return y
    y[mask] += diff / cnt
    return np.clip(y, 0, 1)

def logsumexp_corr(h, beta):
    """Compute logsumexp(beta * correlations) and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corrs = np.correlate(h, 1.0 - h, mode="full") * dx
    x = beta * corrs
    x_max = np.max(x)
    lse = x_max / beta + np.log(np.sum(np.exp(x - x_max))) / beta
    # Gradient: softmax(beta * corrs) * d(corr)/dh (each component)
    weights = np.exp(x - x_max) / np.sum(np.exp(x - x_max))
    # grad of lse wrt h = sum_k weights[k] * grad(c_k)/dh * dx
    # But dx is already included in corrs, so:
    # grad = sum_k weights[k] * grad_corr_k
    g = np.zeros(n)
    for k in range(2*n-1):
        w = weights[k]
        if w < 1e-15:
            continue
        lag = k - (n-1)
        # grad of corr[k] wrt h
        t = np.arange(n)
        mask1 = (t + lag >= 0) & (t + lag < n)
        g[mask1] += w * (1.0 - h[t[mask1] + lag])
        mask2 = (t - lag >= 0) & (t - lag < n)
        g[mask2] -= w * h[t[mask2] - lag]
    return lse, g * dx

def project_onto_box_sum(y, target):
    """Euclidean projection onto {p: 0<=p<=1, sum(p)=target}"""
    n = len(y)
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        p = np.clip(y - mid, 0.0, 1.0)
        if p.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

def lse_objective(x, beta, mu, target):
    """Log-sum-exp of correlations + sum penalty. Returns (value, gradient)."""
    n = len(x)
    dx = 2.0 / n
    corrs = np.correlate(x, 1.0 - x, mode="full") * dx
    x_beta = beta * corrs
    x_max = x_beta.max()
    shifted = x_beta - x_max
    exp_shifted = np.exp(shifted)
    log_sum = x_max + np.log(exp_shifted.sum())
    lse_val = log_sum / beta
    
    # Penalty
    sum_x = x.sum()
    penalty = mu * (sum_x - target) ** 2

        g[m2] -= w * x[idx[m2] - lag]
