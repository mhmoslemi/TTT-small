def project_feasible(h, target):
    """Project h onto {x in [0,1]^n : sum x = target}"""
    n = len(h)
    # Bisection on lambda
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0, 1)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo+hi)/2, 0, 1)
    # Final adjustment
    diff = target - proj.sum()
    if abs(diff) > 1e-10:
        # Distribute among non-saturated
        mask = (proj > 1e-10) & (proj < 1 - 1e-10)
        if mask.sum() > 0:
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, 0, 1)
    return proj

import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - result.sum()
    if abs(diff) > 1e-12:
        mask = (result > 1e-12) & (result < 1.0 - 1e-12)
        cnt = mask.sum()
        if cnt > 0:
            result[mask] += diff / cnt
            result = np.clip(result, 0.0, 1.0)
    return result

def compute_grad(h, tau):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    x = tau * corr
    x_max = x.max()
    w = np.exp(x - x_max)
    w /= w.sum()
    grad = np.zeros(n)
    for j in range(len(corr)):
        wj = w[j]
        if wj < 1e-15:
            continue
        s = j - n + 1
        lo1 = max(0, s)
        hi1 = min(n, n + s)
        if lo1 < hi1:
            grad[lo1:hi1] += wj * (1.0 - h[lo1 - s:hi1 - s])
        lo2 = max(0, -s)
        hi2 = min(n, n - s)
        if lo2 < hi2:
            grad[lo2:hi2] -= wj * h[lo2 + s:hi2 + s]
    return grad

def smooth_max(corr, tau):
    x = tau * corr
    return float((x.max() + np.log(np.sum(np.exp(x - x.max())))) / tau)

def pg_optimize(h_init, n_pts, tau_list, iters_per_tau, t_limit, t0):
    target = n_pts / 2.0
    h = project_feasible(h_init.astype(float), target)
    best = h.copy()
    best_c5 = evaluate_c5(h)
    alpha = 1.0
    
    for tau in tau_list:
        if time.time() - t0 > t_limit:
            break
        for it in range(iters_per_tau):
            if time.time() - t0 > t_limit:
                break
            grad = compute_grad(h, tau)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-12:
                break
            # Line search
            f_old = smooth_max(np.correlate(h, 1.0 - h, mode="full"), tau)
            a = alpha
            for _ in range(40):
                h_new = project_feasible(h - a * grad, target)
                f_new = smooth_max(np.correlate(h_new, 1.0 - h_new, mode="full"), tau)
                if f_new < f_old:
                    h = h_new
                    alpha = min(a * 1.5, 5.0)
                    break
                a *= 0.5
            else:
                break
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best = h.copy()
    return best, best_c5

def make_binary(n, seed):
    """Create a binary pattern with n/2 ones"""
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    idx = rng.choice(n, n//2, replace=False)
    h[idx] = 1.0
    return h

def make_sinusoidal(n, freq, amp, phase):
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + amp * np.sin(2 * np.pi * freq * x + phase)
    return np.clip(h, 0, 1)

def make_block(n, num_blocks):
    """Alternating blocks"""
    block_size = n // num_blocks
    h = np.zeros(n)
    for i in range(num_blocks):
        if i % 2 == 0:
            start = i * block_size
            end = start + block_size if i < num_blocks - 1 else n
            h[start:end] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    time_limit = budget_s - 10  # leave 10s margin
    
    # Get best starting point

        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = np.full(60, 0.5)
    
    best_h = h_parent.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Phase 1: Optimize parent at its resolution
    n_p = len(h_parent)
    h_opt, c5_opt = pg_optimize(h_parent, n_p, [10, 50, 200, 500, 1000], 200, time_limit, t0)
    if c5_opt < best_c5:
        best_h, best_c5 = h_opt, c5_opt
    
    # Phase 2: Try multiple resolutions starting from structured points
    resolutions = [40, 60, 80, 100, 120, 140, 160]
    
    for n_pts in resolutions:
        if time.time() - t0 > time_limit:
            break
        if n_pts % 2 != 0:

        # Multiple starting points
        starts = []
        
        # 1. Interpolated parent
        h_interp = np.interp(np.linspace(0, 1, n_pts), np.linspace(0, 1, len(best_h)), best_h)
        starts.append(h_interp)

            if c5_opt < best_c5:

    # Phase 3: Fine-tune best at its resolution
    n_best = len(best_h)
    if time.time() - t0 < time_limit:
        h_opt, c5_opt = pg_optimize(best_h, n_best, [100, 500, 1000, 2000], 300, time_limit, t0)
        if c5_opt < best_c5:

import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target):
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    r = np.clip(h - (lo + hi) / 2, 0, 1)
    d = target - r.sum()
    if abs(d) > 1e-14:
        m = (r > 1e-14) & (r < 1 - 1e-14)
        if m.any():
            r[m] += d / m.sum()
            r = np.clip(r, 0, 1)
    return r

def grad_sm(h, tau):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    x = tau * corr
    xm = x.max()
    w = np.exp(x - xm)
    w /= w.sum()
    c1 = np.convolve(1.0 - h, w, 'full')[n-1:2*n-1]
    c2 = np.convolve(h, w[::-1], 'full')[n-1:2*n-1]
    return c1 - c2

def obj_sm(h, tau):
    corr = np.correlate(h, 1.0 - h, mode="full")
    x = tau * corr
    return float((x.max() + np.log(np.sum(np.exp(x - x.max())))) / tau)

def pg_opt(h0, taus, iters, t0, tlim):
    n = len(h0)
    tgt = n / 2.0
    h = project_feasible(np.array(h0, dtype=float), tgt)
    best = h.copy()
    best_c = evaluate_c5(h)
    alpha = 1.0
    for tau in taus:
        if time.time() - t0 > tlim:
            break
        for _ in range(iters):
            if time.time() - t0 > tlim:
                break
            g = grad_sm(h, tau)
            gn = np.linalg.norm(g)
            if gn < 1e-12:
                break
            f0 = obj_sm(h, tau)
            a = alpha
            ok = False
            for _ in range(40):
                hn = project_feasible(h - a * g, tgt)
                fn = obj_sm(hn, tau)
                if fn < f0:
                    h = hn
                    ok = True
                    break
                a *= 0.5
            if not ok:
                break
            alpha = min(a * 1.2, 5.0)
            c = evaluate_c5(h)
            if c < best_c:
                best_c = c
                best = h.copy()
    return best, best_c
