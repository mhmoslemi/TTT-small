def project_to_constraint(h, target_sum):

    h = h.copy()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_to_constraint(h, target_sum):
    h = h.copy()
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        hp = np.clip(h + mid, 0.0, 1.0)
        if np.sum(hp) > target_sum:
            hi = mid
        else:
            lo = mid
    hp = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - np.sum(hp)
    if abs(diff) > 1e-10:
        mask = (hp > 0.001) & (hp < 0.999)
        cnt = np.sum(mask)
        if cnt > 0:
            hp[mask] += diff / cnt
            hp = np.clip(hp, 0.0, 1.0)
    return hp

def subgradient_c5(h, top_m=5):
    corr = compute_corr(h)
    n = len(h)
    max_c = np.max(corr)
    top_k = np.argsort(corr)[-top_m:]
    grad = np.zeros(n)
    dx = 2.0 / n
    j = np.arange(n)
    for k in top_k:
        shift = k - (n - 1)
        g = np.zeros(n)
        idx1 = j - shift
        v1 = (idx1 >= 0) & (idx1 < n)
        g[v1] += (1.0 - h[idx1[v1]])
        idx2 = j + shift
        v2 = (idx2 >= 0) & (idx2 < n)
        g[v2] -= h[idx2[v2]]
        grad += g
    grad *= dx / top_m
    return max_c, grad

def projected_subgradient(h, target_sum, n_iter, step_init, step_decay=0.9995):
    h_best = h.copy()
    c5_cur = compute_c5(h)
    alpha = step_init
    for it in range(n_iter):
        _, grad = subgradient_c5(h, top_m=5)
        h_new = project_to_constraint(h - alpha * grad, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur:
            c5_cur = c5_new
            h_best = h_new.copy()
        h = h_new
        alpha *= step_decay
    return h_best, c5_cur

    return v[-1]

def t_ineq(v):
    n = len(v) - 1
    h = v[:n]
    t = v[-1]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return t - corr

    n = len(v) - 1
    return np.sum(v[:n]) - n * 0.5

def t_eq_jac(v):
    n = len(v) - 1
    j = np.zeros(len(v))

    return j

    v0 = np.append(h0.copy(), c5_cur + t_margin)
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.5)]
    cons = [
        {'type': 'ineq', 'fun': t_ineq},
        {'type': 'eq', 'fun': t_eq_fun, 'jac': t_eq_jac}
    ]
    res = minimize(t_objective, v0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': ftol})
    h_new = np.clip(res.x[:n], 0.0, 1.0)
    h_new = project_to_constraint(h_new, n * 0.5)
    return h_new, compute_c5(h_new)

def smooth_obj(h, temp):
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = np.clip((corr - m) / temp, -50, 0)
    return m + temp * np.log(np.sum(np.exp(shifted)))

def sum_eq_fun(h):
    return np.sum(h) - len(h) * 0.5

def sum_eq_jac(h):
    return np.ones(len(h))

def optimize_smooth(h0, temp, max_iter=300, ftol=1e-12):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac}]
    res = minimize(smooth_obj, h0.copy(), args=(temp,), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': ftol})
    h_new = np.clip(res.x, 0.0, 1.0)
    h_new = project_to_constraint(h_new, n * 0.5)
    return h_new, compute_c5(h_new)

def interp_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 1, n_old, endpoint=False)
    x_new = np.linspace(0, 1, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_to_constraint(h_new, n_new * 0.5)

def binary_local_search(h_bin, n_iters, rng):
    n = len(h_bin)
    dx = 2.0 / n
    h = h_bin.copy()
    corr = compute_corr(h)
    c5_cur = np.max(corr)
    h_best = h.copy()
    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])
    s_range = np.arange(-(n - 1), n)
    for it in range(n_iters):
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        t1 = np.zeros(2 * n - 1)
        v1 = (j - s_range >= 0) & (j - s_range < n)
        t1[v1] = 1.0 - h[j - s_range[v1]]
        t2 = np.zeros(2 * n - 1)
        v2 = (i - s_range >= 0) & (i - s_range < n)
        t2[v2] = 1.0 - h[i - s_range[v2]]
        t3 = np.zeros(2 * n - 1)
        v3 = (i + s_range >= 0) & (i + s_range < n)
        t3[v3] = h[i + s_range[v3]]
        t4 = np.zeros(2 * n - 1)
        v4 = (j + s_range >= 0) & (j + s_range < n)
        t4[v4] = h[j + s_range[v4]]
        dc = t1 - t2 + t3 - t4
        new_corr = corr + dx * dc
        c5_new = np.max(new_corr)
        if c5_new < c5_cur - 1e-12:
            h[i] = 0.0
            h[j] = 1.0
            ones[i] = j
            zeros[j] = i
            corr = new_corr
            c5_cur = c5_new
            h_best = h.copy()
    return h_best, float(c5_cur)

def round_to_binary(h, n):
    target = n // 2
    h_bin = np.zeros(n)
    order = np.argsort(-h)
    h_bin[order[:target]] = 1.0
    return h_bin

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - t0)

    h = np.array(initial_h_values, dtype=float)
    n = len(h)
    h = project_to_constraint(h, n * 0.5)
    c5 = compute_c5(h)
    h_best = h.copy()

    # Phase 1: Subgradient
    if time_left() > 250:
        h_new, c5_new = projected_subgradient(h, n * 0.5, 5000, 0.04)
        if c5_new < c5:
            h, c5 = h_new, c5_new
            h_best = h.copy()

    # Phase 2: t-var SLSQP
    if time_left() > 250:
        h_new, c5_new = optimize_t_var(h, max_iter=500, ftol=1e-10)
        if c5_new < c5:
            h, c5 = h_new, c5_new
            h_best = h.copy()

    # Phase 3: Binary local search
    if time_left() > 300:
        h_bin = round_to_binary(h_best, n)
        h_b, c5_b = binary_local_search(h_bin, 300000, rng)
        if c5_b < c5:
            h_best = h_b.copy()
            c5 = c5_b
        # Refine from binary using t-var
        h_cont = project_to_constraint(h_b + rng.randn(n) * 0.01, n * 0.5)
        if time_left() > 200:
            h_new, c5_new = optimize_t_var(h_cont, max_iter=400, ftol=1e-10, t_margin=0.05)
            if c5_new < c5:
                h, c5 = h_new, c5_new
                h_best = h.copy()

    # Phase 4: Smooth optimization
    for temp in [0.03, 0.015, 0.008, 0.004, 0.002, 0.001]:
        if time_left() < 150:
            break
        h_new, c5_new = optimize_smooth(h_best, temp, max_iter=250, ftol=1e-12)
        if c5_new < c5:
            h, c5 = h_new, c5_new
            h_best = h.copy()

    # Phase 5: Subgradient round 2
    if time_left() > 200:
        h_new, c5_new = projected_subgradient(h_best, n * 0.5, 4000, 0.02)
        if c5_new < c5:
            h_best = h_new.copy()
            c5 = c5_new

    # Phase 6: Final t-var polish
    if time_left() > 100:
        h_new, c5_new = optimize_t_var(h_best, max_iter=500, ftol=1e-11, t_margin=0.01)
        if c5_new < c5:
            h_best = h_new.copy()
            c5 = c5_new

    # Phase 7: Random restarts
    for i in range(8):
        if time_left() < 120:
            break
        scale = 0.03 + 0.02 * i
        h_p = h_best + rng.randn(n) * scale
        h_p = project_to_constraint(h_p, n * 0.5)
        h_new, c5_new = optimize_t_var(h_p, max_iter=300, ftol=1e-10, t_margin=0.03)
        if c5_new < c5:
            h_best = h_new.copy()
            c5 = c5_new

    # Phase 8: Different resolution attempts
    for n_new in [120, 240, 300]:
        if time_left() < 300:
            break
        h_interp = interp_h(h_best, n_new)
        h_cur = h_interp
        c5_cur = compute_c5(h_cur)
        h_new, c5_new = optimize_t_var(h_cur, max_iter=350, ftol=1e-10)
        if c5_new < c5:
            h_best = h_new.copy()
            c5 = c5_new
            h_cur = h_new
        for temp in [0.02, 0.01, 0.005]:
            if time_left() < 150:
                break
            h_new, c5_new = optimize_smooth(h_cur, temp, max_iter=200)
            if c5_new < c5:
                h_best = h_new.copy()
                c5 = c5_new
                h_cur = h_new
        if time_left() > 200:
            h_new, c5_new = optimize_t_var(h_cur, max_iter=400, ftol=1e-11, t_margin=0.01)
            if c5_new < c5:
                h_best = h_new.copy()
                c5 = c5_new

    # Phase 9: Binary search at best resolution
    n_final = len(h_best)
    if time_left() > 150:
        h_bin = round_to_binary(h_best, n_final)
        h_b, c5_b = binary_local_search(h_bin, 200000, rng)
        if c5_b < c5:
            h_best = h_b.copy()
            c5 = c5_b

    h_best = project_to_constraint(h_best, len(h_best) * 0.5)
    c5_final = compute_c5(h_best)
    if c5_final > 0.381:
        h_best = np.array(initial_h_values, dtype=float)
        h_best = project_to_constraint(h_best, len(h_best) * 0.5)
        c5_final = compute_c5(h_best)
    return h_best, c5_final, len(h_best)
