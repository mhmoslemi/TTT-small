import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - target
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(free):
            h[free] -= diff / np.sum(free)
        else:
            h -= diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = np.sum(h) - target
    if abs(diff) > 1e-15:
        # Distribute among all
        h -= diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def circular_coord_descent(h, n, max_rounds=1000, n_grid=50):
    dx = 2.0 / n
    best_c5 = compute_c5(h, dx)
    h = h.copy()
    
    for round_num in range(max_rounds):
        improved = False
        for j in range(n):
            # Try changing h[j]
            old_val = h[j]
            # Grid of delta values
            deltas = np.linspace(-old_val, 1.0 - old_val, n_grid)
            best_delta = 0.0
            best_obj = best_c5
            for delta in deltas:
                h_new = h.copy()
                h_new[j] = old_val + delta
                h_new = project_feasible(h_new, n)
                c5_new = compute_c5(h_new, dx)
                if c5_new < best_obj - 1e-15:

                    best_delta = delta
            if best_delta != 0.0:

                h = project_feasible(h, n)
                improved = True
                if best_obj < best_c5:
                    best_c5 = best_obj
        if not improved:
            break
    return h, best_c5

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = np.sum(h) - target
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(free):
            h[free] -= diff / np.sum(free)
        else:
            h -= diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = np.sum(h) - target
    if abs(diff) > 1e-14:
        h -= diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def two_coord_descent(h, n, max_rounds, offsets):
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h, dx)
    c5_cur = best_c5
    k_vals = np.arange(-(n - 1), n)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    no_improve = 0

    for rnd in range(max_rounds):
        improved_any = False
        for j in range(n):
            for off in offsets:
                m = (j + off) % n
                if m == j:
                    continue
                lo = max(-h[j], h[m] - 1.0)
                hi = min(1.0 - h[j], h[m])
                if hi - lo < 1e-13:
                    continue
                corr = np.correlate(h, 1.0 - h, mode='full')
                f = corr * dx
                s_j = np.zeros(2 * n - 1)
                mk1 = (j - k_vals >= 0) & (j - k_vals < n)
                if np.any(mk1):
                    s_j[mk1] += (1.0 - h[j - k_vals[mk1]])
                mk2 = (j + k_vals >= 0) & (j + k_vals < n)
                if np.any(mk2):
                    s_j[mk2] += (-h[j + k_vals[mk2]])
                s_m = np.zeros(2 * n - 1)
                mk1 = (m - k_vals >= 0) & (m - k_vals < n)
                if np.any(mk1):
                    s_m[mk1] += (1.0 - h[m - k_vals[mk1]])
                mk2 = (m + k_vals >= 0) & (m + k_vals < n)
                if np.any(mk2):
                    s_m[mk2] += (-h[m + k_vals[mk2]])
                d_vec = s_j - s_m
                a, b = lo, hi
                c = b - gr * (b - a)
                d_pt = a + gr * (b - a)
                gc = np.max(f + c * d_vec)
                gd = np.max(f + d_pt * d_vec)
                for _ in range(30):
                    if gc < gd:
                        b = d_pt
                        d_pt = c
                        gd = gc
                        c = b - gr * (b - a)
                        gc = np.max(f + c * d_vec)
                    else:
                        a = c
                        c = d_pt
                        gc = gd
                        d_pt = a + gr * (b - a)
                        gd = np.max(f + d_pt * d_vec)
                delta_opt = (a + b) / 2.0
                val = np.max(f + delta_opt * d_vec)
                if val < c5_cur - 1e-13:
                    h[j] += delta_opt
                    h[m] -= delta_opt
                    c5_cur = val
                    improved_any = True
        if c5_cur < best_c5 - 1e-15:
            best_c5 = c5_cur
        no_improve = no_improve + 1 if not improved_any else 0
        if not improved_any or no_improve > 40:
            break
    return h, best_c5

def upsample_periodic(h, n_old, n_new):
    x_new = np.arange(n_new).astype(float) * n_old / n_new
    i0 = np.floor(x_new).astype(int) % n_old
    i1 = (i0 + 1) % n_old
    frac = x_new - np.floor(x_new)
    h_new = (1.0 - frac) * h[i0] + frac * h[i1]
    return h_new

def subgradient_descent(h, n, max_iter, alpha_init=0.05, mu=0.7):
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h, dx)
    c5_cur = best_c5
    k_vals = np.arange(-(n - 1), n)
    momentum = np.zeros(n)
    alpha = alpha_init

    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        f = corr * dx
        F = np.max(f)
        active = f >= F - max(1e-6, 0.01 * F)
        n_active = np.sum(active)
        if n_active == 0:
            break
        g = np.zeros(n)
        for idx in np.where(active)[0]:
            k = k_vals[idx]
            for j in range(n):
                s = 0.0
                jk = j - k
                if 0 <= jk < n:
                    s += (1.0 - h[jk])
                jk2 = j + k
                if 0 <= jk2 < n:
                    s += (-h[jk2])
                g[j] += s * dx
        g /= n_active
        momentum = mu * momentum - alpha * g
        h_new = project_feasible(h + momentum, n)
        c5_new = compute_c5(h_new, dx)
        if c5_new < c5_cur:
            h = h_new
            c5_cur = c5_new
            if c5_cur < best_c5:
                best_c5 = c5_cur
            alpha *= 1.01
        else:
            alpha *= 0.85
            momentum *= 0.5
        if alpha < 1e-8:
            break
    return h, best_c5
