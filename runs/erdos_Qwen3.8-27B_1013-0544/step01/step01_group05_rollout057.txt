                grad[:n - L] += (1.0 - h[L:])
                if L > 0:
                    grad[L:] -= h[:n - L]
            else:
                L2 = -L
                grad[L2:] += (1.0 - h[:n - L2])
                grad[:n - L2] -= h[L2:]
        grad /= float(len(active))
        h_new = project_feasible(h - step_size * grad, target_sum)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            no_improve = 0
        else:
            h = h_new
            no_improve += 1
            if no_improve > 80:
                step_size *= 0.8
                no_improve = 0
            if step_size < 1e-11:
                break
    return best_h, best_c5

def _slsqp_smooth(h, dx, target_sum, beta, maxiter=250):
    global _g_dx, _g_target, _g_beta
    _g_dx = dx
    _g_target = target_sum
    _g_beta = beta
    n = len(h)
    try:
        res = minimize(_obj_smooth, h, jac=_grad_smooth, method='SLSQP',
                      bounds=[(0.0, 1.0)] * n,
                      constraints=[{'type': 'eq', 'fun': _eq_constraint, 'jac': _eq_jac}],
                      options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        if res.x is not None:
            h_ref = project_feasible(res.x, target_sum)
            return h_ref, compute_c5(h_ref, dx)
    except Exception:
        pass
    return h, compute_c5(h, dx)

def _slsqp_direct(h, dx, target_sum, maxiter=200):
    global _g_dx, _g_target
    _g_dx = dx
    _g_target = target_sum
    n = len(h)
    try:
        res = minimize(_obj_direct, h, method='SLSQP',
                      bounds=[(0.0, 1.0)] * n,
                      constraints=[{'type': 'eq', 'fun': _eq_constraint, 'jac': _eq_jac}],
                      options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        if res.x is not None:
            h_ref = project_feasible(res.x, target_sum)
            return h_ref, compute_c5(h_ref, dx)
    except Exception:
        pass
    return h, compute_c5(h, dx)

def _update_best(h, c5, n):
    global _g_best_h, _g_best_c5, _g_best_n
    if c5 < _g_best_c5:
        _g_best_c5 = c5
        _g_best_h = h.copy()
        _g_best_n = n

def _optimize_one(n_points, h_start, deadline):
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    h = project_feasible(h_start.astype(float).copy(), target_sum)
    cur_c5 = compute_c5(h, dx)
    for beta in [5, 15, 50, 150, 500, 1500]:
        if time.time() > deadline - 30:
            break
        h, cur_c5 = _slsqp_smooth(h, dx, target_sum, beta, maxiter=250)
    if time.time() < deadline - 25:
        h, cur_c5 = _slsqp_direct(h, dx, target_sum, maxiter=200)
    h, cur_c5 = _active_set_refine(h, dx, target_sum, deadline)
    _update_best(h, cur_c5, n_points)

def run(seed=42, budget_s=1100, **kwargs):
    global _g_start_time, _g_budget_s, _g_best_h, _g_best_c5, _g_best_n
    _g_start_time = time.time()
    _g_budget_s = budget_s
    _g_best_h = None
    _g_best_c5 = np.inf
    _g_best_n = 0
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except NameError:
        h_init = np.full(70, 0.5)

    t70a = _g_start_time + (budget_s - 20) * 0.40
    t70b = _g_start_time + (budget_s - 20) * 0.55
    t80 = _g_start_time + (budget_s - 20) * 0.72
    t100 = _g_start_time + budget_s - 20

    # Main: n=70 from initial
    _optimize_one(70, h_init, t70a)

    # n=70 perturbations
    for k in range(4):
        if tleft() < 30:
            break
        size = 0.06 + 0.04 * k
        h_p = _g_best_h + rng.normal(0, size, 70)
        _optimize_one(70, h_p, min(t70b, _g_start_time + budget_s - 60))

    # n=80
    if tleft() > 50:
        h_80 = np.interp(np.linspace(0, 69, 80), np.arange(70), _g_best_h)
        _optimize_one(80, h_80, min(t80, _g_start_time + budget_s - 40))

    # n=100
    if tleft() > 50:
        h_100 = np.interp(np.linspace(0, 69, 100), np.arange(70), _g_best_h)
        _optimize_one(100, h_100, min(t100, _g_start_time + budget_s - 20))

    # Final polish on best
    if tleft() > 15 and _g_best_h is not None:
        n_best = _g_best_n
        dx_b = 2.0 / n_best
        ts_b = n_best / 2.0
        h_pol, c5_pol = _active_set_refine(_g_best_h, dx_b, ts_b,
                                           _g_start_time + budget_s - 5, n_iter=50000)
        _update_best(h_pol, c5_pol, n_best)

    if _g_best_h is None:
        _g_best_h = np.full(70, 0.5)
        _g_best_c5 = compute_c5(_g_best_h, 2.0 / 70)
        _g_best_n = 70

    return (_g_best_h, _g_best_c5, _g_best_n)
