def project_feasible(x, target_sum):
    lo = float(x.min()) - 1.0
    hi = float(x.max()) + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = float(np.clip(x - mid, 0.0, 1.0).sum())
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

    return (start_time + budget_s - 3.0) - time.time()

def smooth_max_grad(h, dx, tau):
    n = len(h)
    corr = compute_corr(h)
    vals = corr * dx
    m = np.max(vals)
    shifted = tau * (vals - m)
    exps = np.exp(shifted)
    sw = np.sum(exps)
    weights = exps / sw
    n_half = n - 1
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        w = weights[k_idx]
        if w < 1e-10:
            continue
        L = k_idx - n_half
        if L >= 0:
            if L < n:
                grad[L:] += w * (1.0 - h[:n - L])
            if n - 1 - L >= 0:
                grad[:n - L] -= w * h[L:]
        else:
            L2 = -L
            if L2 < n:
                grad[:n - L2] += w * (1.0 - h[L2:])
                grad[L2:] -= h[:n - L2]
    return grad * dx
