def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def project_h(h, target_sum):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - h
            order = np.argsort(-room)
            for idx in order:
                add = min(diff, room[idx])
                if add <= 0:
                    break
                h[idx] += add
                diff -= add
                if abs(diff) < 1e-14:
                    break
        else:
            order = np.argsort(-h)
            for idx in order:
                sub = min(-diff, h[idx])
                if sub <= 0:
                    break
                h[idx] -= sub
                diff += sub
                if abs(diff) < 1e-14:
                    break
    return h

def objective_func(h, n_points, beta):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, beta) * (2.0 / n_points)

def constraint_func(h, target_sum):
    return h.sum() - target_sum

    return np.ones(len(h))

def sla_step_binary(h, n, T, max_iter, rng):
    """Simulated annealing on binary h with sum = n/2"""
    cost = np.max(np.correlate(h, 1 - h, mode="full"))
    best_h = h.copy()
    best_cost = cost
    for it in range(max_iter):
        i, j = rng.choice(n, 2, replace=False)
        if h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        new_cost = np.max(np.correlate(h, 1 - h, mode="full"))
        delta = new_cost - cost
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-8)):
            cost = new_cost
            if cost < best_cost:
                best_cost = cost
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
        T *= 0.9999
    return best_h, best_cost

def adversarial_step(h, target_sum, n_top=3, step_size=0.001):
    """Take a step reducing the worst lags"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(-corr)[:n_top]
    grad = np.zeros(n)

        for i in range(n):
            if 0 <= i + lag < n:
                grad[i] += (1.0 - h[i + lag])
            if 0 <= i - lag < n:
                grad[i] -= h[i - lag]
    grad = grad - grad.mean()
    norm = np.linalg.norm(grad)
    if norm > 1e-12:
        h = project_h(h - step_size * grad / norm, target_sum)
    return h

For a given lag `lag` (which can be negative, zero, or positive), the correlation term is:
f_lag(h) = sum_{i: 0≤i<n, 0≤i+lag<n} h[i] * (1 - h[i+lag])

The product rule gives:
∂f_lag/∂h[j] = sum over all pairs (i, i+lag) that involve j:

def sca_optimize(h, max_iters=200, tol=1e-12):
    from scipy.optimize import linprog
    n = len(h)
    target_sum = n / 2.0
    dx = 2.0 / n
    n_lags = 2 * n - 1
    
    for it in range(max_iters):
        # Compute f_lag and gradients for all lags
        f_vals = np.zeros(n_lags)
        grads = np.zeros((n_lags, n))
        
        for k in range(n_lags):
            lag = k - (n - 1)
            # Compute f_lag and gradient
            f_val = 0.0
            grad = np.zeros(n)
            for i in range(n):
                if 0 <= i + lag < n:
                    f_val += h[i] * (1.0 - h[i + lag])
                    grad[i] += (1.0 - h[i + lag])
                if lag != 0 and 0 <= i - lag < n:
                    grad[i] -= h[i - lag]
            f_vals[k] = f_val
            grads[k] = grad
        
        # Build LP: min t
        # s.t. grads[k] @ x - t <= RHS[k] for k = 0..n_lags-1
        #      sum(x) = target_sum
        #      0 <= x[j] <= 1
        #      t >= 0 (implicit from objective)
        
        c = np.zeros(n + 1)
        c[n] = 1.0  # minimize t
        
        A_ub = np.zeros((n_lags, n + 1))
        b_ub = np.zeros(n_lags)
        for k in range(n_lags):
            A_ub[k, :n] = grads[k]
            A_ub[k, n] = -1.0
            b_ub[k] = np.dot(grads[k], h) - f_vals[k]
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target_sum])
        
        bounds_lp = [(0, 1)] * n + [(0, None)]
        
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds_lp, method='highs')
        
        if res.success:
            h_new = res.x[:n]
            t_new = res.x[n]
        else:
            break
        
        improvement = np.max(np.abs(h_new - h))
        h = h_new
        
        if improvement < tol:
            break
    
    return h

        else:

        f_vals[k] = np.sum(h[indices] * (1 - h[indices + lag]))
        # Gradient: (1-h[i+lag]) for i in valid range

        # i-lag is valid when 0 ≤ i-lag < n
        # Since i is in [i_start, i_end), i-lag is in [i_start-lag, i_end-lag)
        # For lag >= 0: i-lag ranges from -lag to n-2lag, valid part is max(0,-lag) to min(n, n-lag) = 0 to n-lag = i_start to i_end
        # Wait, this is the same set of indices! So the second term also applies to the same indices.
        # Actually no. Let me reconsider.
        # The second term -h[i-lag] applies to position j = i+lag (the second index in the pair).
        # So for position j, we need j-lag to be a valid i, i.e., 0 ≤ j-lag < n and j-lag is in the valid i range.
        # j = i+lag, so j ranges from i_start+lag to i_end-1+lag.
        # For lag >= 0: j ranges from lag to n-1
        # For lag < 0: j ranges from 0 to n-1+l, i.e., 0 to n-1+lag

            j_end = min(n, i_end + lag)
            j_indices = np.arange(j_start, j_end)
            grads[k, j_indices] -= h[j_indices - lag]

import numpy as np
from scipy.optimize import minimize, Bounds, linprog
import time
from functools import partial

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    if m < -700 / beta:
        return m
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def project_h(h, target_sum):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - h
            order = np.argsort(-room)
            for idx in order:
                add = min(diff, room[idx])
                if add <= 0: break
                h[idx] += add
                diff -= add
                if abs(diff) < 1e-14: break
        else:
            order = np.argsort(-h)
            for idx in order:
                sub = min(-diff, h[idx])
                if sub <= 0: break
                h[idx] -= sub
                diff += sub
                if abs(diff) < 1e-14: break
    return h

def objective_func(h, n_points, beta):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, beta) * (2.0 / n_points)

def constraint_func(h, target_sum):
    return h.sum() - target_sum

def constraint_jac(h, target_sum):
    return np.ones(len(h))

def sca_optimize(h, target_sum, max_iters=500, time_limit=60.0):
    n = len(h)
    n_lags = 2 * n - 1
    start = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    damping = 0.7
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        if it % 50 == 0 and it > 0:
            damping = max(0.3, damping * 0.95)
        
        # Compute f_vals and gradients
        f_vals = np.zeros(n_lags)
        grads = np.zeros((n_lags, n))
        for k in range(n_lags):
            lag = k - (n - 1)
            if lag >= 0:
                valid_i = np.arange(n - lag)
            else:
                valid_i = np.arange(-lag, n)
            f_vals[k] = np.sum(h[valid_i] * (1.0 - h[valid_i + lag]))
            if lag != 0:
                grads[k, valid_i] += (1.0 - h[valid_i + lag])
                grads[k, valid_i + lag] -= h[valid_i]
            else:
                grads[k, valid_i] += (1.0 - 2.0 * h[valid_i])
        
        # Build LP
        c = np.zeros(n + 1)
        c[n] = 1.0
        A_ub = np.zeros((n_lags, n + 1))
        b_ub = np.zeros(n_lags)
        for k in range(n_lags):
            A_ub[k, :n] = grads[k]
            A_ub[k, n] = -1.0
            b_ub[k] = np.dot(grads[k], h) - f_vals[k]
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target_sum])
        bounds_lp = [(0, 1)] * n + [(0, None)]
        
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds_lp, method='highs')
        if not res.success:
            break
        
        x_lp = res.x[:n]
        h_cand = (1 - damping) * h + damping * x_lp
        h_cand = project_h(h_cand, target_sum)
        c5_cand = compute_c5(h_cand)
        
        if c5_cand < best_c5:
            best_h = h_cand.copy()
            best_c5 = c5_cand
            h = h_cand
        else:
            # Try smaller damping
            h_cand2 = 0.5 * h + 0.5 * x_lp
            h_cand2 = project_h(h_cand2, target_sum)
            if compute_c5(h_cand2) < best_c5:
                h = h_cand2
                best_h = h.copy()
                best_c5 = compute_c5(h)
    
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float).copy()
    else:
        h = np.full(70, 0.5)
    
    n = len(h)
    target_sum = n / 2.0
    h = project_h(h, target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Phase 1: SLSQP refinement
    bounds = Bounds(np.zeros(n), np.ones(n))
    for beta in [50, 150, 300]:
        if time.time() - start_time > budget_s - 300:
            break
        obj = partial(objective_func, n_points=n, beta=beta)
        con_fun = partial(constraint_func, target_sum=target_sum)
        con_jac = partial(constraint_jac, target_sum=target_sum)
        constraints = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
        try:
            res = minimize(obj, best_h, method='SLSQP', bounds=bounds,
                         constraints=constraints,
                         options={'maxiter': 2000, 'ftol': 1e-14})
            h_new = project_h(res.x, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new.copy(), c5
        except Exception:
            pass
    
    # Phase 2: SCA optimization
    remaining = budget_s - (time.time() - start_time)
    if remaining > 60:
        best_h, best_c5 = sca_optimize(best_h, target_sum,
                                       max_iters=300,
                                       time_limit=min(remaining - 60, 400))
    
    # Phase 3: SLSQP polish from SCA result
    if time.time() - start_time < budget_s - 30:
        for beta in [500, 1000]:
            if time.time() - start_time > budget_s - 15:
                break
            obj = partial(objective_func, n_points=n, beta=beta)
            con_fun = partial(constraint_func, target_sum=target_sum)
            con_jac = partial(constraint_jac, target_sum=target_sum)
            constraints = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
            try:
                res = minimize(obj, best_h, method='SLSQP', bounds=bounds,
                             constraints=constraints,
                             options={'maxiter': 3000, 'ftol': 1e-15})
                h_new = project_h(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h, best_c5 = h_new.copy(), c5
            except Exception:
                pass
    
    # Phase 4: Try SCA again from polished result
    if time.time() - start_time < budget_s - 30:
        best_h, best_c5 = sca_optimize(best_h, target_sum,
                                       max_iters=200,
                                       time_limit=budget_s - (time.time() - start_time) - 15)
    
    return best_h, best_c5, len(best_h)

We want: min t

Since f_lag is bilinear (not convex), we can't solve this directly. Instead, at the current point h, we linearize:
f_lag(x) ≈ f_lag(h) + ∇f_lag(h) · (x - h)

The constraint becomes:
f_lag(h) + ∇f_lag(h) · x - ∇f_lag(h) · h ≤ t
∇f_lag(h) · x - t ≤ ∇f_lag(h) · h - f_lag(h)
