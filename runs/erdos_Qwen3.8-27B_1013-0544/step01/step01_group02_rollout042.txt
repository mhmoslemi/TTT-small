    """Update correlation c when h[i] changes by delta. O(n) update."""
    # c[k] = sum_j h[j] * (1-h[k-j]) for valid j, k-j

import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):

    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-12:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-12:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def pair_coord_descent(h, n, n_passes=50, verbose=False):
    """Pair coordinate descent to minimize C5."""
    dx = 2.0 / n
    best_c5 = compute_c5(h, n)
    
    # Coarse grid
    coarse_deltas = np.array([0.0, 0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5])
    
    for p in range(n_passes):
        improved = False
        for i in range(n):
            for j in range(i+1, n):
                max_d = min(h[i], 1.0 - h[j])
                if max_d < 1e-6:
                    continue
                # Try all deltas (positive and negative)
                best_delta = 0.0
                best_c = best_c5
                for d in coarse_deltas:
                    if d > max_d:
                        continue
                    for s in ([1, -1] if d > 0 else [1]):
                        di = s * d
                        if di < 0 and (-di) > h[j]:
                            continue
                        if di > 0 and di > h[i]:
                            continue
                        if di < 0 and (-di) > (1.0 - h[j]):
                            continue
                        if di > 0 and di > (1.0 - h[i]):
                            continue
                        # Check: h[i]+di in [0,1] and h[j]-di in [0,1]
                        hi = h[i] + di
                        hj = h[j] - di
                        if hi < 0 or hi > 1 or hj < 0 or hj > 1:
                            continue
                        h_try = h.copy()
                        h_try[i] = hi
                        h_try[j] = hj
                        c = compute_c5(h_try, n)
                        if c < best_c:
                            best_c = c
                            best_delta = di
                if best_delta != 0:
                    h[i] += best_delta
                    h[j] -= best_delta
                    best_c5 = best_c
                    improved = True
        if not improved and p > 5:
            break
    
    return h, best_c5

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-13:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-13:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)
