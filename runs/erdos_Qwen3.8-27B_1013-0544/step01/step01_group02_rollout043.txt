    """Local search on binary sequence: try all swaps, take best improving move."""
    h = h.copy().astype(int)

    ones = np.where(h == 1)[0]

                h[j] = 1

                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                    improved = True
                    break
                h[i] = 1
                h[j] = 0
            if improved:
                break

            break

            m2 = hi - (hi - lo) / 3.0
            v1 = np.max(c_full + m1 * d)
            v2 = np.max(c_full + m2 * d)
            if v1 < v2:
                hi = m2
            else:
                lo = m1
        
        delta = (lo + hi) / 2.0
        new_c = c_full + delta * d
        new_max = new_c.max()
        
        if new_max < best_c - 1e-13:
            h[i] += delta
            h[j] -= delta
            c_full = new_c
            best_c = new_max
    
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    def elapsed():
        return time.time() - t0
    
    def remaining():
        return budget_s - elapsed()
    
    # Get initial
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 80
        h0 = np.full(n0, 0.5)
    
    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, n0)
    
    bounds = [(0.0, 1.0)] * n0
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n0,)}]
    
    # Phase 1: SLSQP with smooth max, multiple alpha values
    alphas = [30, 80, 200, 500, 1200, 3000]
    for alpha in alphas:
        if remaining() < 20:
            break
        try:
            res = minimize(smooth_max_obj, best_h, args=(n0, alpha), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 2000, 'ftol': 1e-16})
            h_try = project_feasible(res.x, n0)
            c = compute_c5(h_try, n0)
            if c < best_c5 - 1e-14:
                best_h, best_c5 = h_try, c
        except Exception:
            pass
    
    # Phase 2: Pairwise local search (main optimization)
    if remaining() > 30:
        time_alloc = min(remaining() * 0.45, 400)
        best_h, _ = pairwise_optimize(best_h, n0, rng, 200000, time_alloc)
        best_h = project_feasible(best_h, n0)
        best_c5 = compute_c5(best_h, n0)
    
    # Phase 3: SLSQP refinement after pairwise
    if remaining() > 20:
        for alpha in [500, 2000, 5000]:
            if remaining() < 15:
                break
            try:
                res = minimize(smooth_max_obj, best_h, args=(n0, alpha), method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 3000, 'ftol': 1e-17})
                h_try = project_feasible(res.x, n0)
                c = compute_c5(h_try, n0)
                if c < best_c5 - 1e-14:
                    best_h, best_c5 = h_try, c
            except Exception:
                pass
    
    # Phase 4: More pairwise + perturbation restarts
    stagnation_count = 0
    while remaining() > 50:
        time_per_iter = min(remaining() * 0.15, 30)
        h_new, _ = pairwise_optimize(best_h, n0, rng, 50000, time_per_iter)
        h_new = project_feasible(h_new, n0)
        c_new = compute_c5(h_new, n0)
        
        if c_new < best_c5 - 1e-12:
            best_h, best_c5 = h_new, c_new
            stagnation_count = 0
        else:
            stagnation_count += 1
            if stagnation_count >= 3:
                # Perturb and retry
                scale = rng.choice([0.01, 0.02, 0.05, 0.1])
                h_pert = best_h + rng.normal(0, scale, size=n0)
                h_pert = project_feasible(h_pert, n0)
                h_new, _ = pairwise_optimize(h_pert, n0, rng, 30000, min(remaining() * 0.1, 20))
                h_new = project_feasible(h_new, n0)
                c_new = compute_c5(h_new, n0)
                if c_new < best_c5 - 1e-12:
                    best_h, best_c5 = h_new, c_new
                    stagnation_count = 0
    
    # Phase 5: Try different n values
    for n_alt in [100, 120, 160]:
        if remaining() < 60:
            break
        # Interpolate from best solution
        x_old = np.linspace(0, 2, n0)
        x_new = np.linspace(0, 2, n_alt)
        h_alt = np.interp(x_new, x_old, best_h)
        h_alt = project_feasible(h_alt, n_alt)
        
        bounds_alt = [(0.0, 1.0)] * n_alt
        cons_alt = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n_alt,)}]
        
        # Quick SLSQP
        if remaining() > 40:
            try:
                res = minimize(smooth_max_obj, h_alt, args=(n_alt, 500), method='SLSQP',
                              bounds=bounds_alt, constraints=cons_alt,
                              options={'maxiter': 1000, 'ftol': 1e-15})
                h_alt = project_feasible(res.x, n_alt)
            except Exception:
                pass
        
        # Pairwise optimization
        if remaining() > 20:
            h_alt, _ = pairwise_optimize(h_alt, n_alt, rng, 100000, min(remaining() * 0.3, 80))
            h_alt = project_feasible(h_alt, n_alt)
        
        c_alt = compute_c5(h_alt, n_alt)
        if c_alt < best_c5:
            best_h, best_c5 = h_alt, c_alt
    
    # Phase 6: Final polish
    n_final = len(best_h)
    if remaining() > 10:
        bounds_f = [(0.0, 1.0)] * n_final
        cons_f = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n_final,)}]
        try:
            res = minimize(smooth_max_obj, best_h, args=(n_final, 5000), method='SLSQP',
                          bounds=bounds_f, constraints=cons_f,
                          options={'maxiter': 5000, 'ftol': 1e-18})
            h_f = project_feasible(res.x, n_final)
            c_f = compute_c5(h_f, n_final)
            if c_f < best_c5:
                best_h, best_c5 = h_f, c_f
        except Exception:
            pass
    
    # Final pairwise polish
    if remaining() > 5:
        best_h, _ = pairwise_optimize(best_h, len(best_h), rng, 20000, min(remaining() - 2, 10))
        best_h = project_feasible(best_h, len(best_h))
        best_c5 = compute_c5(best_h, len(best_h))
    
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h, len(best_h))
    
    return best_h, best_c5, len(best_h)
