def project_feasible(x, target_sum):
    lo = float(x.min()) - 2.0
    hi = float(x.max()) + 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(x - mid, 0.0, 1.0).sum())
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max()) * dx

def compute_corr_array(h):
    return np.correlate(h, 1.0 - h, mode='full')

def corr_delta(h, i, j, n):
    """Compute change in corr[k] for all k when h[i]:1->0 and h[j]:0->1.
    Returns array of length 2n-1."""
    # L_k = k - (n-1) for k = 0, ..., 2n-2
    L = np.arange(2*n-1) - (n-1)  # lags from -(n-1) to n-1
    # Δcorr[k] = h[i-L_k] + h[i+L_k] - h[j-L_k] - h[j+L_k] (with h=0 out of bounds)
    iL = i - L
    iR = i + L
    jL = j - L
    jR = j + L
    # Get values with boundary handling
    def safe_get(arr, idx):
        out = np.zeros(len(idx))
        valid = (idx >= 0) & (idx < len(arr))
        out[valid] = arr[idx[valid]]
        return out
    delta = safe_get(h, iL) + safe_get(h, iR) - safe_get(h, jL) - safe_get(h, jR)
    return delta

import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(x, target_sum):
    lo = float(x.min()) - 2.0
    hi = float(x.max()) + 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(x - mid, 0.0, 1.0).sum())
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max()) * dx

def compute_corr_array(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_binary(h, n):
    b = (h >= 0.5).astype(float)
    n_ones = int(b.sum())
    if n_ones != n // 2:
        if n_ones > n // 2:
            idx = np.where(b == 1)[0]
            for k in range(n_ones - n // 2):
                b[idx[k]] = 0
        else:
            idx = np.where(b == 0)[0]
            for k in range(n // 2 - n_ones):
                b[idx[k]] = 1
    return b

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    n = 70
    dx = 2.0 / n
    target_sum = n / 2.0
    n_half = n - 1

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except NameError:
        h0 = np.full(n, 0.5)
    h0 = project_feasible(h0, target_sum)
    best_h = h0.copy()
    best_c5 = compute_c5(h0, dx)

    def tleft():
        return (start_time + budget_s - 3.0) - time.time()

    # Phase 1: Binary Simulated Annealing with incremental correlation update
    h_sa = make_binary(h0, n)
    corr_sa = compute_corr_array(h_sa)
    max_sa = float(corr_sa.max())
    c5_sa = max_sa * dx
    if c5_sa < best_c5:
        best_c5 = c5_sa
        best_h = h_sa.copy()

    L_arr = np.arange(2 * n - 1) - n_half
    n_sa_iters = 300000
    T_init = 0.06
    T_final = 0.00005
    best_sa_c5 = c5_sa
    best_sa_h = h_sa.copy()

    for it in range(n_sa_iters):
        if tleft() < 200:
            break
        T = T_init * (T_final / T_init) ** (it / n_sa_iters)
        # Pick random 1 and 0 positions
        i = rng.randint(n)
        while h_sa[i] == 0:
            i = rng.randint(n)
        j = rng.randint(n)
        while h_sa[j] == 1 or j == i:
            j = rng.randint(n)

        # Compute delta corr for all lags
        iL = i - L_arr
        iR = i + L_arr
        jL = j - L_arr
        jR = j + L_arr
        delta = np.zeros(2 * n - 1)
        v = (iL >= 0) & (iL < n)
        delta[v] += h_sa[iL[v]]
        v = (iR >= 0) & (iR < n)
        delta[v] += h_sa[iR[v]]
        v = (jL >= 0) & (jL < n)
        delta[v] -= h_sa[jL[v]]
        v = (jR >= 0) & (jR < n)
        delta[v] -= h_sa[jR[v]]

        new_corr = corr_sa + delta
        new_max = float(new_corr.max())
        d_obj = new_max - max_sa
        if d_obj <= 0 or rng.random() < np.exp(-d_obj / max(T, 1e-12)):
            h_sa[i] = 0
            h_sa[j] = 1
            corr_sa = new_corr
            max_sa = new_max
            c5_new = max_sa * dx
            if c5_new < best_sa_c5:
                best_sa_c5 = c5_new
                best_sa_h = h_sa.copy()
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_sa.copy()

        return np.sum(x) - target_sum

            if res.x is not None:

                cr = compute_c5(hr, dx)
                if cr < best_c5:
                    best_c5 = cr
                    best_h = hr
        except Exception:
            pass

    # Phase 3: Projected gradient with momentum
    if tleft() > 80:
        h = best_h.copy()
        step = 0.002
        mu = 0.65
        mom = np.zeros(n)
        stag = 0
        while tleft() > 30:
            cg = compute_corr_array(h)
            cm = float(cg.max())
            tol = max(0.01, cm * 0.001)
            act = np.where(cg >= cm - tol)[0]
            if len(act) < 3:
                act = np.argsort(cg)[-3:]
            elif len(act) > 10:
                act = np.argsort(cg)[-10:]
            grad = np.zeros(n)
            for ki in act:
                Lk = int(ki) - n_half
                if Lk >= 0:
                    if Lk == 0:
                        grad[:] += (1.0 - h[:]) - h[:]
                    else:
                        ix1 = np.arange(Lk, n)
                        grad[ix1] += (1.0 - h[ix1 - Lk])
                        ix2 = np.arange(0, n - Lk)
                        grad[ix2] -= h[ix2 + Lk]
                else:
                    L2 = -Lk
                    ix1 = np.arange(L2, n)
                    grad[ix1] += (1.0 - h[ix1 - L2])
                    ix2 = np.arange(0, n - L2)
                    grad[ix2] -= h[ix2 + L2]
            grad /= float(len(act))
            mom = mu * mom - step * grad
            hn = project_feasible(h + mom, target_sum)
            cn = compute_c5(hn, dx)
            if cn < best_c5:
                best_c5 = cn
                best_h = hn.copy()
                stag = 0
            else:
                stag += 1
                if stag % 25 == 0:
                    step *= 0.9
                if step < 1e-9:
                    h = best_h + rng.normal(0, 0.02, n)
                    h = project_feasible(h, target_sum)
                    step = 0.001
                    mom = np.zeros(n)
                    stag = 0
            h = hn

    # Phase 4: Multi-start SLSQP
    if tleft() > 40:
        for trial in range(15):
            if tleft() < 15:
                break
            h_try = best_h + rng.normal(0, 0.04, n)
            h_try = project_feasible(h_try, target_sum)
            ct = compute_c5(h_try, dx)
            if ct < best_c5:
                best_c5 = ct
                best_h = h_try.copy()
            try:
                res = minimize(obj_fn, h_try, method='SLSQP', bounds=bounds,
                              constraints=cons, options={'maxiter': 5000, 'ftol': 1e-15})
                if res.x is not None:
                    hr = project_feasible(res.x, target_sum)
                    cr = compute_c5(hr, dx)
                    if cr < best_c5:
                        best_c5 = cr
                        best_h = hr
            except Exception:
                pass

    # Phase 5: Rounding check
    if tleft() > 5:
        h_bin = make_binary(best_h, n)
        c5_bin = compute_c5(h_bin, dx)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
        # Also try best_sa_h rounded
        h_bin2 = make_binary(best_sa_h, n)
        c5_bin2 = compute_c5(h_bin2, dx)
        if c5_bin2 < best_c5:
            best_c5 = c5_bin2
            best_h = h_bin2.copy()

    # Phase 6: Final polish SLSQP
    if tleft() > 15:
        try:
            res = minimize(obj_fn, best_h, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': 15000, 'ftol': 1e-18})
            if res.x is not None:
                hr = project_feasible(res.x, target_sum)
                cr = compute_c5(hr, dx)
                if cr < best_c5:
                    best_c5 = cr
                    best_h = hr
        except Exception:
            pass

    return (best_h, best_c5, n)
