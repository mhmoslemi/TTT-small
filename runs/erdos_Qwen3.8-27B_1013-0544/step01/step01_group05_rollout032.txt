import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(x, target_sum):
    lo = float(x.min()) - 1.0
    hi = float(x.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = float(np.clip(x - mid, 0.0, 1.0).sum())
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def lse_gradient(h, dx, t):
    n = len(h)
    n_half = n - 1
    corr_raw = np.correlate(h, 1.0 - h, mode='full')
    corr = corr_raw * dx
    c_max = float(np.max(corr))
    exp_vals = np.exp(t * (corr - c_max))
    sum_exp = float(np.sum(exp_vals))
    weights = exp_vals / sum_exp
    grad = np.zeros(n)
    omh = 1.0 - h
    thresh = max(1e-9, 1.0 / (n * 20))
    active = np.where(weights > thresh)[0]
    for idx in active:
        L = int(idx) - n_half
        w = weights[idx]
        if L == 0:
            grad += w * (omh - h)
        elif L > 0:
            grad[L:] += w * omh[:n - L]
            grad[:n - L] -= w * h[L:]
        else:
            L2 = -L
            grad[:n - L2] += w * omh[L2:]
            grad[L2:] -= w * h[:n - L2]
    grad *= dx
    return c_max, grad

def lse_optimize(h_init, dx, target_sum, t_start, t_end, max_time, seed):

                t_lse = max(t_start * 0.5, t_lse * 0.4)
                momentum = 0.7

        h = h_new
    return best_h, best_c5

def slsqp_polish(h_init, dx, target_sum, max_iter=3000):
    n = len(h_init)
    def obj(x):
        return compute_c5(x, dx)
    def eq(x):
        return float(np.sum(x) - target_sum)
    try:
        res = minimize(obj, h_init, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': eq}],
                       options={'maxiter': max_iter, 'ftol': 1e-18})
        if res.x is not None:
            h_ref = project_feasible(res.x, target_sum)
            return h_ref, compute_c5(h_ref, dx)
    except Exception:
        pass
    return h_init, compute_c5(h_init, dx)
