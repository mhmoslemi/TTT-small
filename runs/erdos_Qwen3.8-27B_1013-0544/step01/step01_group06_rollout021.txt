```python
import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    return np.max(compute_corr(h)) * (2.0 / n)

def project_onto_simplex_box(h, target_sum):
    # Project h onto {x : 0 <= x <= 1, sum(x) = target_sum}
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def grad_corr_at_lag(h, k):
    """Gradient of corr at lag k with respect to h."""
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        # corr = sum_{i=0}^{n-1-k} h[i] * (1-h[i+k])
        g[:n-k] += (1.0 - h[k:])
        if k > 0:
            g[k:] -= h[:n-k]
    else:
        p = -k
        g[p:] += (1.0 - h[:n-p])
        if p > 0:
            g[:n-p] -= h[p:]
    return g

def smooth_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    s = w.sum()
    if s < 1e-300:
        idx = np.argmax(corr)
        w = np.zeros_like(corr)
        w[idx] = 1.0
    else:
        w /= s
    grad = np.zeros(n)
    active = np.where(w > 1e-8)[0]
    for idx in active:
        k = idx - (n - 1)
        grad += w[idx] * grad_corr_at_lag(h, k)
    return grad * (2.0 / n)

def interpolate_to_n(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, np.append(h_old, h_old[0]))

def optimize_with_backtrack(h0, target_sum, deadline, rng, alpha_init=40, alpha_max=600):
    n = len(h0)
    best_h = project_onto_simplex_box(h0.copy(), target_sum)
    best_c5 = compute_c5(best_h)
    alpha = alpha_init
    lr = 0.01
    patience = 0
    
    while time.time() < deadline:
        grad = smooth_grad(best_h, alpha)
        # Backtracking line search
        f0 = best_c5
        step = lr
        improved = False
        for _ in range(20):
            h_try = project_onto_simplex_box(best_h - step * grad, target_sum)
            c5_try = compute_c5(h_try)
            if c5_try < f0 - 1e-12:
                best_h = h_try
                best_c5 = c5_try
                patience = 0
                improved = True
                break
            step *= 0.5
        if not improved:
            # Try increasing alpha and perturbing
            alpha = min(alpha * 1.5, alpha_max)
            patience += 1
            if patience > 30:
                # Perturb
                pert = 0.05 + 0.05 * min(patience / 50.0, 1.0)
                h_pert = best_h + rng.randn(n) * pert
                h_pert = project_onto_simplex_box(h_pert, target_sum)
                c5_pert = compute_c5(h_pert)
                if c5_pert < best_c5:
                    best_h = h_pert
                    best_c5 = c5_pert
                patience = 0
                alpha = max(alpha_init, alpha * 0.5)
                lr = 0.01
        
        if best_c5 < 0.3805:
            break
    
    return best_h, best_c5
```

```python
def compute_pair_slope(h, i, j):
    n = len(h)
    s_vals = np.arange(2*n-1) - (n-1)  # s = -（n-1), ..., 0, ..., n-1
    
    # A(m, s) = (1-h[m+s]) if 0 <= m+s < n else 0
    def A(m, s_vals):
        idx = m + s_vals
        valid = (idx >= 0) & (idx < n)
        result = np.zeros(len(s_vals))
        result[valid] = 1.0 - h[idx[valid]]
        return result
    
    def B(m, s_vals):
        idx = m - s_vals
        valid = (idx >= 0) & (idx < n)
        result = np.zeros(len(s_vals))
        result[valid] = h[idx[valid]]
        return result
    
    return A(i, s_vals) - B(i, s_vals) - A(j, s_vals) + B(j, s_vals)
```

```python
def compute_corr(h):

def compute_c5(h):

    return float(np.max(compute_corr(h)) * (2.0 / n))

def project_onto_simplex_box(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):

        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:

def find_delta(corr, S, lo, hi):
    """Minimize max_k(corr[k] + delta*S[k]) for delta in [lo,hi] via ternary search."""
    a, b = lo, hi
    if b - a < 1e-12:
        return a
    for _ in range(60):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        if np.max(corr + m1 * S) < np.max(corr + m2 * S):
            b = m2
        else:
            a = m1
    return (a + b) / 2.0

def pairwise_descent(h, deadline, rng):
    n = len(h)
    h = h.astype(float).copy()
    c5 = compute_c5(h)
    stall = 0
    
    while time.time() < deadline:
        corr = compute_corr(h)
        c_max = np.max(corr)
        k_max = int(np.argmax(corr))
        s_max = k_max - (n - 1)
        
        best_improve = 0.0
        best_move = None
        
        # Try moves involving the active lag
        i_lo = max(0, -s_max)
        i_hi = min(n, n - s_max if s_max > 0 else n)
        n_active = i_hi - i_lo
        if n_active > 0:

                i = i_lo + rng.randint(n_active)
                if s_max != 0 and 0 <= i + s_max < n and rng.random() < 0.6:
                    j = i + s_max
                else:
                    j = rng.randint(n)
                if i == j:
                    continue
                S = pair_slope(h, i, j)
                lo = max(-h[i], h[j] - 1.0)
                hi = min(1.0 - h[i], h[j])
                if hi - lo < 1e-10:
                    continue
                d = find_delta(corr, S, lo, hi)
                new_max = float(np.max(corr + d * S))
                imp = c_max - new_max
                if imp > best_improve:
                    best_improve = imp
                    best_move = (i, j, d)
        
        # Random pairs

            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue

            lo = max(-h[i], h[j] - 1.0)

            new_max = float(np.max(corr + d * S))
            imp = c_max - new_max
            if imp > best_improve:
                best_improve = imp
                best_move = (i, j, d)
        
        if best_move is not None and best_improve > 1e-14:
            i, j, d = best_move
            h[i] += d
            h[j] -= d
            np.clip(h, 0.0, 1.0, out=h)
            new_c5 = compute_c5(h)
            if new_c5 < c5:

                stall = 0
            else:

            h = project_onto_simplex_box(h, n / 2.0)

def smooth_grad_descent(h, deadline, rng, alpha_init=30, alpha_max=500):
    n = len(h)
    target = n / 2.0
    h = project_onto_simplex_box(h.copy(), target)
    best_c5 = compute_c5(h)
    alpha = alpha_init
    lr = 0.008
    stall = 0
    
    while time.time() < deadline:
        corr = compute_corr(h)
        m = np.max(corr)
        w = np.exp(alpha * (corr - m))
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros_like(corr)
            w[int(np.argmax(corr))] = 1.0
        else:
            w /= ws
        
        # Compute gradient
        grad = np.zeros(n)
        active = np.where(w > 1e-10)[0]
        for idx in active:
            k = int(idx) - (n - 1)
            if k >= 0:
                grad[:n-k] += w[idx] * (1.0 - h[k:])
                if k > 0:
                    grad[k:] -= w[idx] * h[:n-k]
            else:
                p = -k
                grad[p:] += w[idx] * (1.0 - h[:n-p])
                if p > 0:
                    grad[:n-p] -= w[idx] * h[p:]
        grad *= (2.0 / n)
        
        # Backtracking
        f0 = best_c5
        step = lr
        improved = False
        for _ in range(15):
            h_try = project_onto_simplex_box(h - step * grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < f0 - 1e-13:
                h = h_try
                best_c5 = c5_try
                stall = 0
                improved = True
                break
            step *= 0.5
        
        if not improved:
            alpha = min(alpha * 1.4, alpha_max)
            stall += 1
            if stall > 40:
                h += rng.randn(n) * (0.03 + 0.02 * min(stall/100.0, 1.0))
                h = project_onto_simplex_box(h, target)
                best_c5 = compute_c5(h)
                stall = 0
                alpha = max(alpha_init, alpha * 0.4)
                lr = 0.008
        
        if best_c5 < 0.3800:
            break
    
    return h, best_c5
```
