```python
    """Minimize max_i (base[i] + rate[i]*t) over t in [lo, hi]."""
    # f(t) is convex piecewise linear
    # Use ternary search (works for convex functions)
    for _ in range(60):
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        f1 = np.max(base + m1 * rate)
        f2 = np.max(base + m2 * rate)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    t = (lo + hi) / 2
    return t, np.max(base + t * rate)
```

```python
                          options={'maxiter': 5000, 'ftol': 1e-18})
            ht = project_feasible(res.x, n)
            c = compute_c5(ht, n)
            if c < best_c5 - 1e-14:
                best_h, best_c5 = ht, c
        except Exception:
            pass
    if budget_s - (time.time() - t0) > 5:
        h_new, c_new = cd_random(best_h, n, rng, n_iter=5000, max_delta=0.02)
        if c_new < best_c5 - 1e-14:
            best_h, best_c5 = h_new, c_new

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
```
