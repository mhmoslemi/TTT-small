            k = get_worst_lag(best_h)
            h_new, c5 = do_opt(best_h, 200, 300, target_lag=k)
            if c5 is not None and c5 < best_c5:
                best_h, best_c5 = h_new, c5

    # Phase 7: Final fine-tune with high tau
    if rem() > 10:
        for tau in [2000, 5000, 10000]:
            if rem() < 6:
                break
            h_new, c5 = do_opt(best_h, tau, 400)
            if c5 is not None and c5 < best_c5:
                best_h, best_c5 = h_new, c5

    # Phase 8: Final L-BFGS polish
    if rem() > 8:
        h_new, c5 = do_opt_lbfgs(best_h, 5000, 1000)
        if c5 is not None and c5 < best_c5:
            best_h, best_c5 = h_new, c5

    best_h = fix_sum(best_h, len(best_h) / 2.0)
    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, len(best_h)
