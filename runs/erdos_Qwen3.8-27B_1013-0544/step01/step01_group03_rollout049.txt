```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target):
    h = np.array(h, dtype=float)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return np.clip(h, 0.0, 1.0)
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = mask.sum()
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(obj, tau):
    x = tau * obj
    x_max = np.max(x)
    return float(x_max / tau + np.log(np.sum(np.exp(x - x_max))) / tau)

    """Compute C5 for a binary sequence efficiently."""
    n = len(h)
    dx = 2.0 / n
    # For binary h, correlate(h, 1-h) at lag k = sum h[i+k]*h[i] complement
    # = sum h[i+k] - sum h[i+k]*h[i] in the overlap
    # Actually let's just compute it directly
    corr = np.correlate(h.astype(float), 1.0 - h.astype(float), mode="full")
    return float(np.max(corr) * dx)

def run_sa_binary(h0, n, t_start, t_end, n_iter, seed=42):
    """Simulated annealing on binary sequence."""
    rng = np.random.RandomState(seed)
    h = h0.copy()
    current_c5 = binary_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]

        idx1 = ones[rng.randint(len(ones))]
        idx0 = zeros[rng.randint(len(zeros))]
        
        h[idx1] = 0
        h[idx0] = 1
        new_c5 = binary_c5(h)
        
        delta = new_c5 - current_c5
        if delta < 0 or rng.random() < np.exp(-delta / T):
            current_c5 = new_c5
            # Update ones and zeros lists
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[idx1] = 1
            h[idx0] = 0
    
    return best_h, best_c5
```

```python
def smooth_max_and_grad(h, tau):

    corr = np.correlate(h, 1.0 - h, mode="full")
    x = tau * corr
    x_max = np.max(x)
```

```python
    weights = exp_shifted / exp_shifted.sum()
    value = x_max / tau + np.log(exp_shifted.sum()) / tau
    
    lags = np.arange(-(n-1), n)  # shape (2n-1,)
    J = np.arange(n)[:, None]     # shape (n, 1)
    E = lags[None, :]             # shape (1, 2n-1)
    
    diff1 = J - E   # shape (n, 2n-1): j - ell
    diff2 = J + E   # shape (n, 2n-1): j + ell
    
    mask1 = (diff1 >= 0) & (diff1 < n)
    mask2 = (diff2 >= 0) & (diff2 < n)
    
    # first[j,k] = (1-h[diff1[j,k]]) * weights[k] if mask1[j,k] else 0
    # We need h[diff1[j,k]] which is a fancy index
    # h[diff1] won't work directly because diff1 has invalid entries
    # Let's clip and use where
    idx1 = np.clip(diff1, 0, n-1)
    first = np.where(mask1, (1.0 - h[idx1]) * weights[None, :], 0.0)
    
    idx2 = np.clip(diff2, 0, n-1)
```

Let me also think about an alternative: **coordinate descent**.

In coordinate descent, we optimize one variable h[j] at a time while keeping others fixed. For a fixed j, the objective as a function of h[j] is:
f(h) = max_k g_k(h) where g_k depends on h[j] through:
- The term h[j] appears in g_k as: h[j]*(1-h[j-ell_k]) + h[j+ell_k]*(1-h[j]) (approximately)
  Wait, let me be more careful.

g_k(h) = sum_i h[i+ell_k]*(1-h[i])

h[j] appears in this sum:
- As h[i+ell_k] when i+ell_k = j, i.e., i = j-ell_k: contributes (1-h[j-ell_k]) * h[j]
- As h[i] when i = j: contributes h[j+ell_k] * (1-h[j]) = h[j+ell_k] - h[j+ell_k]*h[j]

So g_k as a function of t = h[j]:
g_k(t) = [contribution from other terms] + (1-h[j-ell_k])*t + h[j+ell_k]*(1-t)
       = [const] + (1-h[j-ell_k])*t + h[j+ell_k] - h[j+ell_k]*t
       = [const] + h[j+ell_k] + t*((1-h[j-ell_k]) - h[j+ell_k])

So g_k is LINEAR in t! That means f(h_j = t) = max_k (a_k + b_k * t) which is the max of linear functions, hence convex in t.

This means coordinate descent should work! For each coordinate j, we minimize a convex piecewise-linear function over [0,1], which has an analytical solution (the minimum is at a breakpoint or at the boundary).

We can compute all b_k in O(n) time (just look up the relevant h values).
We can compute all a_k in O(n) time if we already have the correlation values and just subtract the contribution of h[j].

Then finding the minimum of max_k(a_k + b_k*t) over [0,1] can be done in O(n log n) using a convex hull trick, or O(n²) by checking all breakpoints.

- Evaluate at t=0: f(0) = max_k a_k
- Evaluate at t=1: f(1) = max_k (a_k + b_k)

Actually, the simplest efficient approach is:
1. Compute all b_k in O(n)
2. The optimal t is where the weighted average of b_k (weighted by the "activity" of each k) is zero
3. Since f is convex piecewise linear, binary search on t

For binary search: evaluate f at t_mid, compute subgradient (set of active b_k), and move. This takes O(log(1/eps)) iterations, each O(n).

Let me implement this:

```python
def optimize_coordinate(h, j, n_corr):
    """Optimize h[j] given precomputed correlation contributions."""
    n = len(h)
    # Compute b_k for each lag
    lags = np.arange(-(n-1), n)
    b = np.zeros(2*n-1)
    for idx, ell in enumerate(lags):
        # b_k = (1-h[j-ell]) if 0<=j-ell<n else 0 - h[j+ell] if 0<=j+ell<n else 0
        val = 0.0
        if 0 <= j - ell < n:
            val += (1.0 - h[j - ell])
        if 0 <= j + ell < n:
            val -= h[j + ell]
        b[idx] = val
    
    # Compute a_k = g_k - b_k * h[j]
    # g_k = corr_full[k] (given the current h)
    # a_k = corr_full[k] - b_k * h[j]
    a = n_corr - b * h[j]
    
    # Now minimize f(t) = max_k (a_k + b_k * t) over t in [0,1]
    # f is convex piecewise linear
    # Use binary search on the subgradient
    
    h_copy = h.copy()
    
    lo, hi = 0.0, 1.0
    for _ in range(30):
        mid = (lo + hi) / 2
        vals = a + b * mid
        max_val = np.max(vals)
        # Subgradient: the set of b_k for which a_k + b_k*mid = max_val
        active = np.where(np.abs(vals - max_val) < 1e-10)[0]
        min_slope = np.min(b[active])
        max_slope = np.max(b[active])
        if min_slope > 1e-10:
            # f is increasing at mid, move left
            hi = mid
        elif max_slope < -1e-10:
            # f is decreasing at mid, move right
            lo = mid
        else:
            # mid is optimal
            break
    
    h[j] = (lo + hi) / 2
    return h
```

Wait, there's an issue. The a_k depends on h[j] (the current value), and when we change h[j], the other g_k values change too. But in coordinate descent, we fix all other variables and only change h[j]. So a_k should be computed as "g_k with h[j]=0" and then f(t) = max_k(a_k + b_k*t) is correct.

Let me verify: if h[j] = t, then:
g_k(t) = sum_i h[i+ell]*(1-h[i]) where h[j] = t
= [terms not involving h[j]] + [terms involving h[j]]
= g_k(h with h[j]=0) + b_k * t

where b_k = dc_k/dh[j] (the partial derivative, which is constant since g_k is linear in h[j]).

So: a_k = g_k(h with h[j]=0) = corr_full[k] (with h[j]=0)

If the current h[j] = t0, then:
corr_full[k] (with h[j]=t0) = a_k + b_k * t0
So: a_k = corr_full[k] - b_k * t0

This is what I had. Good.

Now, the binary search for the optimal t:
- f(t) = max_k(a_k + b_k*t) is convex piecewise linear
- Its subgradient at t is conv({b_k : a_k + b_k*t = f(t)})
- f is minimized where 0 ∈ subgradient, i.e., where there exist active k with b_k > 0 and b_k < 0 (or just b_k = 0)

The binary search works: if all active slopes are positive, f is increasing, so optimal is to the left. If all active slopes are negative, f is decreasing, so optimal is to the right.

This should converge in O(log(1/eps)) iterations, each taking O(n) time.

A full pass of coordinate descent is O(n * n * log(1/eps)) = O(n² * 30) for n coordinates. For n=400, that's 400*400*30 = 4.8M operations per pass. Very fast!

We can do many passes. Let me implement this.

Actually, I realize there's a subtlety: after changing h[j], the sum constraint ∑h = n/2 is violated. We need to handle this.

Options:
1. After each coordinate change, repair the sum (add a small correction to another coordinate)
2. Use a Lagrange multiplier for the sum constraint
3. Only make changes that preserve the sum (swap: decrease h[j] by δ and increase h[l] by δ)

Option 3 is cleanest for the binary case but more complex for the continuous case.

For the continuous case with the sum constraint, let's use a Lagrange multiplier approach:
Minimize f(h) + λ(∑h - n/2)

In coordinate descent, when optimizing h[j], the effective objective becomes:
f_j(t) = max_k(a_k + b_k*t) + λ*t

This is still convex piecewise linear (max of affine + affine = max of affine), so the same binary search works.

We update λ using a subgradient method: λ ← λ - η*(∑h - n/2)

This is the "dual ascent" approach. It should work well.

Let me implement the full algorithm:

```python
def coordinate_descent(h, n_max_iters=1000, lambda_init=0.0):
    """Coordinate descent with Lagrange multiplier for sum constraint."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    lam = lambda_init
    lrs = 1.0  # learning rate for lambda
    
    lags = np.arange(-(n-1), n)
    
    for it in range(n_max_iters):
        h_old = h.copy()
        for j in range(n):
            # Compute b_k (slope of g_k w.r.t. h[j])
            b = np.zeros(2*n-1)
            for idx, ell in enumerate(lags):
                val = 0.0
                if 0 <= j - ell < n:
                    val += (1.0 - h[j - ell])
                if 0 <= j + ell < n:
                    val -= h[j + ell]
                b[idx] = val
            
            # Compute a_k = corr[k] - b_k * h[j]
            corr = np.correlate(h, 1.0 - h, mode="full")
            a = corr - b * h[j]
            
            # Minimize max_k(a_k + (b_k+lam)*t) over t in [0,1]
            b_eff = b + lam
            lo, hi = 0.0, 1.0
            for _ in range(20):
                mid = (lo + hi) / 2
                vals = a + b_eff * mid
                max_val = np.max(vals)
                active = np.where(np.abs(vals - max_val) < 1e-10)[0]
                min_slope = np.min(b_eff[active])
                if min_slope > 1e-8:
                    hi = mid
                elif np.max(b_eff[active]) < -1e-8:
                    lo = mid
                else:
                    break
            h[j] = (lo + hi) / 2
        
        # Update lambda
        diff = h.sum() - target
        lam -= lrs * diff
        # Adaptive learning rate
        if abs(diff) < 0.01:
            lrs *= 0.9
    
    return h
```

Wait, this is O(n² * (2n-1)) per pass because for each j, we recompute the full correlation. That's O(n * n²) = O(n³) per pass. For n=400, that's 64M per pass. With 1000 passes, that's 64B. Too slow!

I need to avoid recomputing the full correlation each time. The key is that when we change h[j] from t0 to t1, the correlation values change by:
corr_new[k] = corr_old[k] + b_k * (t1 - t0)

where b_k is the slope we already computed! So we can update the correlation incrementally.
