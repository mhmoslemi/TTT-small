    """Gradient of smooth max with respect to corr values"""
    m = np.max(corr)
    shifted = t * (corr - m)
    shifted = np.clip(shifted, -50, 50)
    weights = np.exp(shifted) / np.sum(np.exp(shifted))
    return weights * t * dx  # d(smooth_max)/d(corr[m]) = weights[m] * t * dx

def grad_cwrt_h(h, t, dx):
    """Compute gradient of smooth_max_obj with respect to h"""
    n = len(h)
    corr = compute_corr(h)
    # weights for each lag
    m = np.max(corr)
    shifted = t * (corr - m)
    shifted = np.clip(shifted, -50, 50)
    weights = np.exp(shifted) / np.sum(np.exp(shifted))
    
    # For each lag k, the gradient of corr[k] w.r.t. h[j]:
    # corr[k] = sum_n h[n+k] * (1-h[n])  (numpy convention)
    # d(corr[k])/d(h[j]) = (1-h[j-k]) * [0<=j-k<n and 0<=j<n] - h[j+k] * [0<=j+k<n and 0<=j<n]
    # Wait, let me reconsider.
    # corr[k] = sum_{n: valid} h[n+k] * (1-h[n])
    # h[j] appears as h[n+k] when n = j-k: contributes (1-h[j-k]) if 0<=j-k<n
    # h[j] appears as h[n] when n = j: contributes -h[j+k] if 0<=j+k<n
    # Wait, the (1-h[n]) term: d/dh[j] of (1-h[n]) = -1 if n=j, so contribution is -h[j+k] if 0<=j+k<n
    # Hmm wait: corr[k] = sum_n h[n+k]*(1-h[n])
    # If j = n+k (i.e., n = j-k): term is h[j]*(1-h[j-k]), d/dh[j] = (1-h[j-k])
    # If j = n (i.e., n = j): term is h[j+k]*(1-h[j]), d/dh[j] = -h[j+k]
    
    # So: d(corr[k])/d(h[j]) = (1-h[j-k]) * [0<=j-k<n] - h[j+k] * [0<=j+k<n]
    # But we also need j to be in [0, n-1] for h[j] to exist.
    
    # Let me compute this efficiently using convolution/correlation.
    # (1-h[j-k]) for valid j-k: this is like a correlation of (1-h) with a delta at position k
    # Actually, let me think of it as:
    # grad[j] = sum_k weights[k] * t * dx * [(1-h[j-k])*[0<=j-k<n] - h[j+k]*[0<=j+k<n]]
    #         = t*dx * (sum_k weights[k]*(1-h[j-k])*[0<=j-k<n] - sum_k weights[k]*h[j+k]*[0<=j+k<n])
    
    # First term: sum_k weights[k] * (1-h[j-k]) for valid indices
    # This is like a "correlation" of weights with (1-h)
    # Second term: -sum_k weights[k] * h[j+k] for valid indices
    # This is like -"correlation" of weights with h (with sign flip)
    
    # Using np.correlate:
    # np.correlate(weights, 1-h, mode="full") gives sum_n weights[n+k]*(1-h[n])
    # Hmm, the indexing is tricky. Let me just do it directly for efficiency.
    
    # Actually, for n=160, direct computation is fast enough.
    # But let's use the correlation trick.
    
    # First term: for each j, sum_{k: 0<=j-k<n} weights[k] * (1-h[j-k])
    # Let i = j-k, so k = j-i, and 0<=i<n:
    # = sum_{i=0}^{n-1} weights[j-i] * (1-h[i])  [where weights[j-i] is 0 if j-i is out of range]
    # This is the cross-correlation of weights and (1-h), evaluated at lag j.
    # np.correlate(weights, (1-h), mode="full")[j+offset] gives this.
    
    # Full correlation of two length-n sequences has length 2n-1.
    # c[m] = sum_i A[i+m]*B[i] for m from -(n-1) to n-1
    # In numpy: np.correlate(A, B, mode="full")[m + (n-1)] = sum_i A[i+m]*B[i]
    
    # First term at position j: sum_i weights[j-i]*(1-h[i])
    # Let A = weights, B = (1-h)
    # We want sum_i A[j-i]*B[i] = sum_i A[i'+j]*B[i] where i' = -i... 
    # This is actually the cross-correlation at a specific lag.
    # c_full[m] = sum_i A[i+m]*B[i]
    # We want sum_i A[j-i]*B[i]. Let i' = j-i, so i = j-i':
    # = sum_{i'} A[i']*B[j-i'] = this is the correlation of B and A at lag j
    # = np.correlate(B, A, mode="full")[j + (n-1)] ... hmm, not quite.
    
    # Let me just verify: np.correlate(A, B, mode="full")[k] = sum_n A[n+k]*B[n]
    # So np.correlate((1-h), weights, mode="full")[k] = sum_n (1-h)[n+k]*weights[n]
    # At k=j: = sum_n (1-h)[n+j]*weights[n]  -- this isn't what we want.
    
    # What we want is: sum_i weights[j-i]*(1-h[i])
    # = sum_i weights[i'*( -1 ]*(1-h[i]) where i' = j-i
    # Let me define w_rev = weights[::-1]
    # Then sum_i w_rev[n-(j-n+1)]*... this is getting too complicated.
    
    # For n=160, let's just do the O(n²) computation directly.
    # Actually, O(n²) = 25600 operations per gradient eval, which is fine.
    
    grad = np.zeros(n)
    complement = 1.0 - h
    for k in range(-(n-1), n):
        wk = weights[k + n - 1]  # weight for lag k
        if wk < 1e-10:
            continue
        # d(corr[k])/d(h[j]) = (1-h[j-k])*[0<=j-k<n] - h[j+k]*[0<=j+k<n]
        # For term 1: j-k in [0, n-1], i.e., k<=j<=k+n-1
        # For term 2: j+k in [0, n-1], i.e., -k<=j<=n-1-k
        if k >= 0:
            # Term 1: j in [k, n-1], contribution (1-h[j-k])
            grad[k:n] += wk * complement[0:n-k]
            # Term 2: j in [0, n-1-k], contribution -h[j+k]
            grad[0:n-k] -= wk * h[k:n]
        else:
            kk = -k
            # Term 1: j in [0, n-1+kk], i.e., j-k = j+kk in [kk, n-1+kk] but must be < n
            # j+kk in [0, n-1], so j in [-kk, n-1-kk], intersected with [0, n-1] gives [0, n-1-kk]
            # contribution: (1-h[j+kk])
            grad[0:n-kk] += wk * complement[kk:n]
            # Term 2: j+kk... wait, for k<0, j+k = j-kk in [0, n-1], so j in [kk, n-1]
            # contribution: -h[j-kk]
            grad[kk:n] -= wk * h[0:n-kk]
    
    return grad * t * dx

def fix_sum(h, target_sum):
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-14:
        h = h + diff / len(h)
    return np.clip(h, 0.0, 1.0)

def project_feasible(h, target_sum):
    """Project h onto the feasible set: 0<=h<=1, sum(h)=target_sum"""
    h = np.clip(h, 0.0, 1.0)
    h = fix_sum(h, target_sum)
    return h

def gradient_descent(h, dx, target_sum, t, n_steps, lr_init=0.01):
    """Gradient descent with backtracking line search"""
    n = len(h)
    best_c5 = compute_c5(h, dx)
    h_curr = h.copy()
    lr = lr_init
    
    for step in range(n_steps):
        grad = grad_cwrt_h(h_curr, t, dx)
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break
        grad = grad / grad_norm
        
        # Backtracking line search
        alpha = lr
        f_curr = smooth_max(compute_corr(h_curr), t, dx)
        improved = False
        for _ in range(20):
            h_trial = h_curr - alpha * grad
            h_trial = project_feasible(h_trial, target_sum)
            f_trial = smooth_max(compute_corr(h_trial), t, dx)
            if f_trial < f_curr:
                h_curr = h_trial
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            lr *= 0.5
            if lr < 1e-8:
                break
        else:
            lr = min(lr * 1.1, 0.1)
        
        c5_curr = compute_c5(h_curr, dx)
        if c5_curr < best_c5:
            best_c5 = c5_curr
    
    return h_curr, best_c5

def coordinate_descent_focused(h, dx, target_sum, n_rounds=1000):
    """Coordinate descent focusing on the maximum lags"""
    n = len(h)
    best_c5 = compute_c5(h, dx)
    corr = compute_corr(h)

        corr = compute_corr(h)
        c5_curr = np.max(corr) * dx
        
        # Find the lag with maximum correlation
        m_idx = np.argmax(corr)
        m = m_idx - (n - 1)  # actual lag (can be negative)

        # c[m] = sum_n h[n+m]*(1-h[n])
        # d(c[m])/d(h[j]) = (1-h[j-m])*[0<=j-m<n] - h[j+m]*[0<=j+m<n]

            local_grad = np.zeros(n)
            local_grad[m:n] = 1.0 - h[0:n-m]  # (1-h[j-m])
            local_grad[0:n-m] -= h[m:n]  # -h[j+m]
        else:

            local_grad[0:n-kk] = 1.0 - h[kk:n]  # (1-h[j+kk]) for j in [0, n-1-kk]
            local_grad[kk:n] -= h[0:n-kk]  # -h[j-kk] for j in [kk, n-1]
        
        # Find the coordinate with the most negative gradient (most reducing)
        # We want to increase h[j] where grad < 0, or decrease h[j] where grad > 0
        # Best single coordinate move:
        best_gain = 0
        best_j = -1
        best_dir = 0
        
        for j in range(n):
            g = local_grad[j]
            if g < 0:
                # Increasing h[j] reduces c[m]
                # Max increase: 1 - h[j]
                # But also need to maintain sum, so we'd need to decrease something else
                gain = -g * min(1.0 - h[j], 0.5)  # conservative
            else:
                # Decreasing h[j] reduces c[m]
                gain = -g * min(h[j], 0.5)
            if gain > best_gain:
                best_gain = gain
                best_j = j
                best_dir = 1 if g < 0 else -1
        
        if best_j < 0:

            best_j = np.random.randint(n)
            g = local_grad[best_j]
            best_dir = -1 if g > 0 else 1
        
        # Try moving h[best_j] in the best_dir direction
        # Also need to compensate the sum by adjusting another coordinate
        # Simple approach: move h[best_j] and h[random_other] in opposite directions
        step_size = 0.1
        h_trial = h.copy()
        h_trial[best_j] += best_dir * step_size
        
        # Compensate sum
        diff = -best_dir * step_size
        # Distribute among other coordinates
        others = [j for j in range(n) if j != best_j]

        h_trial[others] += diff / (n - 1)

def pair_coordinate_descent(h, dx, target_sum, n_rounds=500):

        h_trial[i] = min(1.0, target_ij)
        h_trial[j] = max(0.0, target_ij - 1.0)

        c5_a = compute_c5(h_trial, dx)

        h_trial2 = project_feasible(h_trial2, target_sum)
        c5_b = compute_c5(h_trial2, dx)
        
        if c5_a < best_c5:
            h = h_trial
            best_c5 = c5_a
        elif c5_b < best_c5:
            h = h_trial2
            best_c5 = c5_b
    
    return h, best_c5

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")

    target_sum = n / 2.0
    h0 = fix_sum(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)
    best_n = n

    # Phase 1: SLSQP with increasing smoothness
    for t in [10, 30, 80, 200, 500, 1000]:
        if elapsed() > budget_s - 300:
            break
        h_new = slsqp_optimize(best_h, dx, target_sum, t=t, maxiter=600)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 2: Pair coordinate descent
    if elapsed() < budget_s - 200:
        h_new, c5_new = pair_descent(best_h, dx, target_sum, n_rounds=3000)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 3: Group descent
    if elapsed() < budget_s - 150:
        h_new, c5_new = group_descent(best_h, dx, target_sum, group_size=4, n_rounds=150, t=80)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 4: SLSQP refinement
    for t in [200, 500, 1000, 2000]:
        if elapsed() > budget_s - 100:
            break
        h_new = slsqp_optimize(best_h, dx, target_sum, t=t, maxiter=500)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 5: Stochastic local search
    if elapsed() < budget_s - 80:
        h_new, c5_new = anneal(best_h, dx, target_sum, n_steps=4000, t_init=0.003)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 6: SLSQP with random restarts
    n_trials = 0
    while elapsed() < budget_s - 60 and n_trials < 60:
        n_trials += 1
        h_t = slsqp_restart(best_h, dx, target_sum, t=100,
                            n_pert=np.random.randint(2, 6),
                            pert_mag=np.random.uniform(0.05, 0.15))
        c5_t = compute_c5(h_t, dx)
        if c5_t < best_c5:
            best_h, best_c5 = h_t, c5_t
            h_new, c5_new = pair_descent(best_h, dx, target_sum, n_rounds=500)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new

    # Phase 7: Final polish
    if elapsed() < budget_s - 30:
        h_new, c5_new = pair_descent(best_h, dx, target_sum, n_rounds=2000)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Final SLSQP
    if elapsed() < budget_s - 10:
        h_new = slsqp_optimize(best_h, dx, target_sum, t=2000, maxiter=1000)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    best_h = fix_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h, dx)
    return best_h, best_c5, best_n
