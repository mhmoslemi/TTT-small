                # Circular block shift
                blen = np.random.randint(3, min(9, n_points // 5 + 1))
                i = np.random.randint(0, n_points - blen + 1)
                shift = np.random.randint(-4, 5)
                if shift != 0:
                    block = h_new[i:i+blen].copy()
                    h_new[i:i+blen] = np.roll(block, shift)
            elif move_type < 0.9:
                # Multi-point Gaussian
                n_pts = np.random.randint(5, min(20, n_points // 2 + 1))
                idx = np.random.choice(n_points, n_pts, replace=False)
                h_new[idx] += np.random.randn(n_pts) * 0.06
            else:
                # Swap pairs
                n_swaps = np.random.randint(1, 4)
                for _ in range(n_swaps):
                    i, j = np.random.choice(n_points, 2, replace=False)
                    h_new[i], h_new[j] = h_new[j], h_new[i]

            h_new = project_h(h_new, target_sum)
            c5_new = compute_c5(h_new)

            delta = c5_new - c5_cur
            if delta < 0 or np.random.random() < np.exp(-min(200.0, delta / T)):
                h_cur = h_new
                c5_cur = c5_new
                if c5_cur < best_c5:
                    best_h = h_cur.copy()
                    best_c5 = c5_cur
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1

            T *= cooling
            if no_improve > 400:
                T = max(T, 0.008)
                no_improve = 0

    # Phase 3: Multi-start SLSQP from perturbed best
    if tleft() > 60:
        for attempt in range(5):
            if tleft() < 30:
                break
            h0 = best_h.copy()
            n_pert = np.random.randint(5, 15)
            idx = np.random.choice(n_points, n_pert, replace=False)
            h0[idx] += np.random.randn(n_pert) * 0.05
            h0 = project_h(h0, target_sum)
            beta = np.random.choice([150, 250, 400])
            try:
                res = minimize(obj_func, h0, args=(beta, dx),
                               method='SLSQP', bounds=bounds,
                               constraints=constraints,
                               options={'maxiter': 2000, 'ftol': 1e-13})
                h_new = project_h(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h, best_c5 = h_new.copy(), c5
            except Exception:
                pass

    # Phase 4: Final high-precision polish
    if tleft() > 20:
        for beta in [500, 1000, 2000]:
            if tleft() < 15:
                break
            try:
                res = minimize(obj_func, best_h, args=(beta, dx),
                               method='SLSQP', bounds=bounds,
                               constraints=constraints,
                               options={'maxiter': 5000, 'ftol': 1e-15})
                h_new = project_h(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h, best_c5 = h_new.copy(), c5
            except Exception:
                pass

    # Final verification and projection
    best_h = project_h(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n_points
