    if beta == 0:
        return m
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def smooth_max_grad_weights(x, beta):
    """Returns weights for the smooth max (log-sum-exp)."""
    m = np.max(x)
    w = np.exp(beta * (x - m))
    w /= w.sum()
    return w

def project_h(h, target_sum):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) > 1e-10:

            if abs(s - target_sum) < 1e-10:

            grad[i + lag] += (-h[i])

            grad[i] += (-h[i + k])
