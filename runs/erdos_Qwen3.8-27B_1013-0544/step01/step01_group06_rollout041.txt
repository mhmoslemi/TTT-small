    """Compute the change in correlation vector when h[a]:1->0 and h[b]:0->1."""
    n = len(h)
    # corr has length 2n-1, index n-1+k corresponds to lag k
    # corr[n-1+k] = sum_i h[i] * (1-h[i+k])
    # After swap: h[a]:1->0, h[b]:0->1

    #   -from h[a] change: position a in h: -(1-h[a+k]) if 0<=a+k<n
    #   -from (1-h[i+k]) change at i=a-k: h[a-k] * [(1-0)-(1-1)] = h[a-k] if 0<=a-k<n
    #   -from h[b] change: position b in h: +(1-h[b+k]) if 0<=b+k<n
    #   -from (1-h[i+k]) change at i=b-k: h[b-k] * [(1-1)-(1-0)] = -h[b-k] if 0<=b-k<n
    delta = np.zeros(2*n-1)

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * (2.0 / n)

def project_simplex_box(h, target_sum):
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def gradient_at_lag(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        m = n - k
        if m > 0:
            g[:m] += (1.0 - h[k:k+m])
        if k > 0 and m > 0:
            g[k:] -= h[:m]
    else:
        p = -k
        m = n - p
        if m > 0:
            g[p:] += (1.0 - h[:m])
            g[:m] -= h[p:]
    return g

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    mx = np.max(corr)
    w = np.exp(alpha * (corr - mx))
    sw = np.sum(w)
    if sw < 1e-300:
        k_star = int(np.argmax(corr))
        g = gradient_at_lag(h, k_star - (n - 1))
    else:
        w /= sw
        g = np.zeros(n)
        active = np.where(w > 1e-8)[0]
        for idx in active:
            g += w[idx] * gradient_at_lag(h, int(idx) - (n - 1))
    return g * (2.0 / n), mx * (2.0 / n)

def grad_optim(h0, target_sum, deadline, rng, alpha=100, lr_init=0.008):
    n = len(h0)
    best_h = project_simplex_box(h0.astype(float).copy(), target_sum)
    best_c5 = compute_c5(best_h)
    lr = lr_init
    stall = 0
    restarts = 0
    while time.time() < deadline:
        g, _ = smooth_grad(best_h, alpha)
        h_new = project_simplex_box(best_h - lr * g, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_h = h_new
            best_c5 = c5_new
            stall = 0
            if lr < 0.08:
                lr = min(lr * 1.04, 0.08)
        else:
            stall += 1
            lr *= 0.93
            if stall > 60:

                    h_pert = best_h + rng.randn(n) * (0.01 + 0.005 * restarts)
                    h_pert = project_simplex_box(h_pert, target_sum)
                    c5_pert = compute_c5(h_pert)
                    if c5_pert < best_c5:
                        best_h = h_pert
                        best_c5 = c5_pert
                alpha = min(alpha * 1.4, 600.0)
                lr = 0.008
                stall = 0
                restarts += 1
    return best_h, best_c5

def binary_round(h, target):
    n = len(h)
    h_bin = (h >= 0.5).astype(float)
    diff = int(target) - int(np.sum(h_bin))
    if diff > 0:
        zeros = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros])[:diff]
        h_bin[zeros[order]] = 1.0
    elif diff < 0:
        ones = np.where(h_bin == 1)[0]
        order = np.argsort(h[ones])[:(-diff)]
        h_bin[ones[order]] = 0.0
    return h_bin

def bin_local_search(h_start, deadline, rng, n_trials=20, sways_per_iter=200):

def interp_to(h_old, n_new):

    if n_old == n_new:

    x_old = np.linspace(0, 1, n_old, endpoint=False)
