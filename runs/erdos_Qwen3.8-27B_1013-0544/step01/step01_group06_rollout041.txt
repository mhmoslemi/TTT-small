def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    return float(np.max(corr_full(h)) * (2.0 / n))

def smooth_weights(corr, alpha):
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(corr)
        w[int(np.argmax(corr))] = 1.0
    else:
        w /= s
    return w

def grad_from_weights(h, w):
    n = len(h)
    grad = np.zeros(n)
    active = np.where(w > 1e-12)[0]
    for idx in active:
        k = idx - (n - 1)
        wv = w[idx]
        if k >= 0:
            grad[k:] += wv * (1.0 - h[:n-k])
            if k > 0:
                grad[:n-k] -= wv * h[k:]
        else:
            p = -k
            grad[:n-p] += wv * (1.0 - h[p:])
            grad[p:] -= wv * h[:n-p]
    return grad * (2.0 / n)

def project_bs(h, target):
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo+hi), 0.0, 1.0)

import numpy as np
import time

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    return float(np.max(corr_full(h)) * (2.0 / n))

def smooth_weights(corr, alpha):
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(corr)
        w[int(np.argmax(corr))] = 1.0
    else:
        w /= s
    return w

def grad_from_weights(h, w):
    n = len(h)
    grad = np.zeros(n)
    active = np.where(w > 1e-12)[0]
    for idx in active:
        k = int(idx - (n - 1))
        wv = w[idx]
        if k >= 0:
            grad[k:] += wv * (1.0 - h[:n - k])
            if k > 0:
                grad[:n - k] -= wv * h[k:]
        else:
            p = -k
            grad[:n - p] += wv * (1.0 - h[p:])
            grad[p:] -= wv * h[:n - p]
    return grad * (2.0 / n)

def project_bs(h, target):
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def grad_desc(h0, target, deadline, rng, alpha, lr, mom):

    h = project_bs(h0.astype(float).copy(), target)
    v = np.zeros(n)

    best_c5 = compute_c5(h)
    pat = 0
    stalls = 0

        c5 = compute_c5(h_new)

            best_c5 = c5

def coord_refine(h, target, n_sweeps, deadline):

    h = h.copy()

    for sw in range(n_sweeps):
