import time
from scipy.interpolate import interp1d


def project_simplex(h, target_sum):
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def eval_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * (2.0 / n)


def obj_grad(h, alpha):
    n = len(h)
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full')
    m = np.max(corr)
    exps = np.exp(alpha * (corr - m))
    s = np.sum(exps)
    val = (m + np.log(s) / alpha) * (2.0 / n)
    w = exps / s
    dx = 2.0 / n
    grad = np.zeros(n)
    for j in range(n):
        k_lo = max(j - n + 1, 1 - n)
        k_hi = min(j, n - 1)
        if k_lo <= k_hi:
            ks = np.arange(k_lo, k_hi + 1)
            grad[j] += np.dot(w[ks + n - 1], 1.0 - h[j - ks])
        k_lo2 = max(-j, 1 - n)
        k_hi2 = min(n - 1 - j, n - 1)
        if k_lo2 <= k_hi2:
            ks2 = np.arange(k_lo2, k_hi2 + 1)
            grad[j] -= np.dot(w[ks2 + n - 1], h[j + ks2])
    return val, grad * dx


def local_grad(h0, target, deadline, rng, alpha=40.0, lr=0.008):
    n = len(h0)
    h = project_simplex(h0.copy(), target)
    best_h = h.copy()
    best_c5 = eval_c5(h)
    stall = 0
    while time.time() < deadline:
        val, grad = obj_grad(h, alpha)
        if np.linalg.norm(grad) < 1e-12:
            break
        step = lr
        ok = False
        for _ in range(6):
            h_new = project_simplex(h - step * grad, target)
            v_new, _ = obj_grad(h_new, alpha)
            if v_new < val - 1e-14:
                h = h_new
                ok = True
                break
            step *= 0.4
        if not ok:
            h = project_simplex(h - step * grad, target)
        c5 = eval_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            stall = 0
        else:
            stall += 1
        if stall > 250:
            alpha = min(alpha * 1.15, 300.0)
            if stall > 900:
                h = project_simplex(h + rng.randn(n) * 0.04, target)
                stall = 250
    return best_h, best_c5


def de_search(n, target, deadline, rng, pop=35):
    F, CR = 0.7, 0.85
    pop_arr = np.zeros((pop, n))
    for i in range(pop):
        pop_arr[i] = project_simplex(0.5 + 0.25 * rng.randn(n), target)
    fit = np.array([eval_c5(pop_arr[i]) for i in range(pop)])
    bi = int(np.argmin(fit))
    best_h, best_f = pop_arr[bi].copy(), fit[bi]
    while time.time() < deadline:
        for i in range(pop):
            if time.time() > deadline:
                break
            idx = [x for x in range(pop) if x != i]
            a, b, c = rng.choice(idx, 3, replace=False)
            v = pop_arr[a].astype(float) + F * (pop_arr[b] - pop_arr[c])
            trial = pop_arr[i].copy()
            r = rng.randint(n)
            mask = rng.random(n) < CR
            mask[r] = True
            trial[mask] = v[mask]
            trial = project_simplex(trial, target)
            ft = eval_c5(trial)
            if ft <= fit[i]:
                pop_arr[i] = trial
                fit[i] = ft
                if ft < best_f:
                    best_f, best_h = ft, trial.copy()
    return best_h, best_f


def interp_h(h, n_new):
    n_old = len(h)
    xo = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    xn = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(xo, h, kind='cubic', fill_value='extrapolate')
    return np.clip(f(xn), 0.0, 1.0)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        hp = np.array(initial_h_values, dtype=float)
        ni = len(hp)
    else:
        ni = 200
        hp = project_simplex(0.5 + 0.1 * rng.randn(ni), ni / 2.0)

    best_h = hp.copy()
    best_c5 = eval_c5(hp)

    # Phase 1: DE at coarse resolution for diverse starting points
    n_de = 80
    dl1 = t0 + min(50.0, budget_s * 0.08)
    h_de, c5_de = de_search(n_de, n_de / 2.0, dl1, rng, pop=35)

    # Phase 2: Gradient on parent
    dl2 = t0 + min(300.0, budget_s * 0.4)
    h_g, c5_g = local_grad(hp, ni / 2.0, dl2, rng, alpha=40.0, lr=0.008)
    if c5_g < best_c5:
        best_h, best_c5 = h_g, c5_g

    # Phase 3: Interpolate DE result and optimize
    if time.time() < t0 + budget_s - 200:
        h_interp_de = project_simplex(interp_h(h_de, ni), ni / 2.0)
        dl3 = min(time.time() + 150.0, t0 + budget_s - 200)
        h_d, c5_d = local_grad(h_interp_de, ni / 2.0, dl3, rng, alpha=40.0, lr=0.006)
        if c5_d < best_c5:
            best_h, best_c5 = h_d, c5_d

    # Phase 4: Basin hopping at best resolution
    dl4 = t0 + budget_s - 100
    n_bh = len(best_h)
    target_bh = n_bh / 2.0
    h_bh = best_h.copy()
    c5_bh = best_c5
    pert = 0.06
    while time.time() < dl4:
        inner_dl = min(time.time() + 20.0, dl4)
        h_p = project_simplex(h_bh + rng.randn(n_bh) * pert, target_bh)
        h_l, c5_l = local_grad(h_p, target_bh, inner_dl, rng, alpha=30.0, lr=0.006)
        if c5_l < c5_bh:
            h_bh, c5_bh = h_l, c5_l
            pert = max(0.02, pert * 0.85)
        else:
            pert = min(0.2, pert * 1.1)
        if c5_bh < best_c5:
            best_h, best_c5 = h_bh.copy(), c5_bh

    # Phase 5: Try higher resolution
    if time.time() < t0 + budget_s - 40:
        for n_try in [300, 400]:
            if time.time() < t0 + budget_s - 40:
                h_interp = project_simplex(interp_h(best_h, n_try), n_try / 2.0)
                dl5 = min(time.time() + 100.0, t0 + budget_s - 40)
                h_hi, c5_hi = local_grad(h_interp, n_try / 2.0, dl5, rng, alpha=30.0, lr=0.005)
                if c5_hi < best_c5:
                    best_h, best_c5 = h_hi, c5_hi

    # Final polish
    remaining = t0 + budget_s - 3 - time.time()
    if remaining > 10:
        best_h, best_c5 = local_grad(best_h, len(best_h) / 2.0, t0 + budget_s - 3,
                                     rng, alpha=80.0, lr=0.003)

    best_h = project_simplex(best_h, len(best_h) / 2.0)
    best_c5 = eval_c5(best_h)
    return best_h, best_c5, len(best_h)
