    """Project v onto {h: 0<=h<=1, sum(h)=target_sum}"""

    # h[i] = clip(v[i] - lambda, 0, 1)

    for _ in range(100):

        h = np.clip(v - mid, 0, 1)

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def soft_max_obj_and_grad(h, t):
    n = len(h)
    corr = compute_corr(h)
    max_c = np.max(corr)
    shifted = t * (corr - max_c)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    soft_max = max_c + np.log(sum_exp) / t
    weights = exp_s / sum_exp
    
    # Compute gradient
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-12:
            continue
        m = k - (n - 1)
        if m >= 0:
            if n - m > 0:
                grad[:n-m] += w * (1.0 - h[m:])
            if m < n:
                grad[m:] -= w * h[:n-m]
        else:
            p = -m
            if n - p > 0:
                grad[p:] += w * (1.0 - h[:n-p])
                grad[:n-p] -= w * h[p:]
    
    return soft_max, grad

def project_constraint(v, target_sum):
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
    return np.clip(h, 0.0, 1.0)

def adam_optimize(h_init, t_schedule, lr=0.05, n_iter_per_t=2000):
    n = len(h_init)
    target_sum = n / 2.0
    h = h_init.copy()
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    best_h = h.copy()
    best_c5 = compute_c5(h, 2.0/n)
    
    for t_idx, t in enumerate(t_schedule):
        m_t = t_idx * n_iter_per_t
        for i in range(n_iter_per_t):
            it = m_t + i
            obj, grad = soft_max_obj_and_grad(h, t)
            
            m = 0.9 * m + 0.1 * grad
            v = 0.999 * v + 0.001 * grad**2
            m_hat = m / (1 - 0.9**(it+1))
            v_hat = v / (1 - 0.999**(it+1))
            
            h_new = h - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
            h = project_constraint(h_new, target_sum)
            
            c5 = compute_c5(h, 2.0/n)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    return best_h, best_c5

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def soft_max_and_grad(h, t):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_c = np.max(corr)
    shifted = t * (corr - max_c)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    soft_max = max_c + np.log(sum_exp) / t
    weights = exp_s / sum_exp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-14:
            continue
        m = k - (n - 1)
        if m >= 0:
            nm = n - m
            if nm > 0:
                grad[:nm] += w * (1.0 - h[m:])
                grad[m:] -= w * h[:nm]
        else:
            p = -m
            np_ = n - p
            if np_ > 0:
                grad[p:] += w * (1.0 - h[:np_])
                grad[:np_] -= w * h[p:]
    return soft_max, grad
