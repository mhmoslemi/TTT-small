import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-15:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def solve_1d_max(base, d):
    g0 = np.max(base)
    g1 = np.max(base + d)
    if np.max(d) <= 1e-12:
        return 0.0, g0
    if np.min(d) >= -1e-12:
        return 1.0, g1
    lo = min(g0, g1)
    hi = max(g0, g1)
    for _ in range(55):
        mid = (lo + hi) / 2
        pos = d > 1e-15
        neg = d < -1e-15
        zero = ~(pos | neg)
        feasible = True
        if zero.any() and base[zero].max() > mid:
            feasible = False
        else:
            U = 1.0
            if pos.any():
                U = min(U, np.min((mid - base[pos]) / d[pos]))
            L = 0.0
            if neg.any():
                L = max(L, np.max((mid - base[neg]) / d[neg]))
            if L > U or U < -1e-12 or L > 1.0 + 1e-12:
                feasible = False
        if feasible:
            hi = mid
        else:
            lo = mid
    v = hi
    pos = d > 1e-15
    neg = d < -1e-15
    U = 1.0
    if pos.any():
        U = min(U, np.min((v - base[pos]) / d[pos]))
    L = 0.0
    if neg.any():
        L = max(L, np.max((v - base[neg]) / d[neg]))
    x = max(0.0, min(1.0, (max(L, 0.0) + min(U, 1.0)) / 2))

def coord_descent(h, n, n_sweeps=200):
    h = h.copy()
    corr = np.correlate(h, 1 - h, mode='full')
    rev_h = h[::-1]
    dx = 2.0 / n
    best_c5 = corr.max() * dx
    for sweep in range(n_sweeps):
        improved = False
        for i in range(n):
            d = np.zeros(2 * n - 1)
            d[n - 1 - i: 2 * n - 1 - i] += (1.0 - h)
            d[i: n + i] -= rev_h
            base = corr - h[i] * d
            x_star, _ = solve_1d_max(base, d)
            new_corr = base + d * x_star
            new_max = new_corr.max()
            if new_max < corr.max() - 1e-12:
                h[i] = x_star
                rev_h = h[::-1]
                corr = new_corr
                improved = True
                c5v = corr.max() * dx
                if c5v < best_c5:
                    best_c5 = c5v
        if sweep % 5 == 4:
            h = project_feasible(h, n)
            corr = np.correlate(h, 1 - h, mode='full')
            rev_h = h[::-1]
            best_c5 = min(best_c5, corr.max() * dx)
        if not improved:
            break
    return h, best_c5

def binary_local_search(h, n, n_iters=50000, rng=None):
    h = h.copy().astype(float)
    corr = np.correlate(h, 1 - h, mode='full')
    rev_h = h[::-1]
    dx = 2.0 / n
    best_c5 = corr.max() * dx
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for it in range(n_iters):
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        d_i = np.zeros(2 * n - 1)
        d_i[n - 1 - i: 2 * n - 1 - i] += (1.0 - h)
        d_i[i: n + i] -= rev_h
        d_j = np.zeros(2 * n - 1)
        d_j[n - 1 - j: 2 * n - 1 - j] += (1.0 - h)
        d_j[j: n + j] -= rev_h
        new_corr = corr - d_i + d_j
        new_max = new_corr.max()
        if new_max < corr.max() - 1e-12:
            h[i] = 0
            h[j] = 1
            corr = new_corr
            rev_h = h[::-1]
            ones = ones[ones != i]
            ones = np.append(ones, j)
            zeros = zeros[zeros != j]
            zeros = np.append(zeros, i)
            c5v = corr.max() * dx
            if c5v < best_c5:
                best_c5 = c5v
    return h, best_c5

        t[i] = t[i >> 1] ^ (i & 1)
    return t
