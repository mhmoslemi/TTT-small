```python
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def all_correlations(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def gradient_f_k(h, k_lag, n):
    """Gradient of f_k(h) = sum_{valid i} h[i]*(1-h[i+k]) with respect to h.
    k_lag is the lag in the full correlation, converted to actual shift.
    For np.correlate(h, 1-h, mode='full'), index i corresponds to lag i-(n-1).
    """
    # Convert correlation index to lag
    lag = k_lag - (n - 1)
    # f = sum_{i} h[i] * (1 - h[i+lag]) where i and i+lag are valid
    # For lag >= 0: i ranges from 0 to n-1-lag
    # For lag < 0: i ranges from -lag to n-1
    grad = np.zeros(n)
    if lag >= 0:
        i_max = n - 1 - lag
        # ∂/∂h[j] of sum_{i=0}^{i_max} h[i](1-h[i+lag])
        # = sum_{i=0}^{i_max} [delta(i,j) - delta(i+lag, j) * h[i] - delta(i,j) * h[i+lag]]
        # For j in [0, i_max]: contribution from i=j: 1 - h[j+lag]
        # For j in [lag, i_max+lag]: contribution from i=j-lag: -h[j-lag]
        idx1 = np.arange(0, i_max + 1)
        grad[idx1] += 1.0
        grad[idx1] -= h[idx1 + lag]
        idx2 = np.arange(lag, i_max + 1 + lag)
        grad[idx2] -= h[idx2 - lag]
    else:
        # lag < 0, let l = -lag > 0
        l = -lag
        # f = sum_{i=l}^{n-1} h[i] * (1 - h[i-l])
        # For j in [l, n-1]: from i=j: 1 - h[j-l]
        # For j in [0, n-1-l]: from i=j+l: -h[j+l]
        idx1 = np.arange(l, n)
        grad[idx1] += 1.0
        grad[idx1] -= h[idx1 - l]
        idx2 = np.arange(0, n - l)
        grad[idx2] -= h[idx2 + l]
    return grad

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    # Binary search on offset
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = h_proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def subgradient_step(h, n, step_size, top_k=1):
    """One subgradient step for minimizing max_k f_k(h)."""
    corr = all_correlations(h, n)
    # Get top_k lags
    if top_k == 1:
        k_star = np.argmax(corr)
        grad = gradient_f_k(h, k_star, n)
    else:
        top_indices = np.argsort(corr)[-top_k:]
        grad = np.mean([gradient_f_k(h, k, n) for k in top_indices], axis=0)
    h_new = h - step_size * grad
    return project_feasible(h_new, n), corr
```

```python
    max_c = corr.max()
    shifted = alpha * (corr - max_c)
    exp_shifted = np.exp(shifted)
    log_sum = np.log(exp_shifted.sum())
    return float((max_c + log_sum / alpha) * (2.0 / n))

def smooth_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    max_c = corr.max()
    shifted = alpha * (corr - max_c)
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / exp_shifted.sum()
    grad = np.zeros(n)
    for w, k_idx in zip(weights, range(len(corr))):
        if w > 1e-10:
            grad += w * gradient_f_k(h, k_idx, n)
    return grad * dx
```

The inner loop over all k_idx (2n-1 values) is O(n) per gradient evaluation. For SLSQP with maxiter=2000, that's 2000 * O(n) = O(n²) total, which is fine.

gradient_f_k(h, k_idx, n) for a given lag:
- For lag ≥ 0: grad[lag:n] += (1-h[0:n-lag]); grad[0:n-lag] -= h[lag:n]
- For lag < 0 (l = -lag): grad[0:n-l] += (1-h[l:n]); grad[l:n] -= h[0:n-l]

I can compute all gradients at once using a loop over lags, which is O(n²) total. For n=200, that's 40000 operations, very fast.

Actually, let me just vectorize it properly. The gradient of the smooth objective is:
grad = dx * sum_k w_k * gradient_f_k(h, k, n)

where w_k are the softmax weights.

I can compute this as:
For positive lags k (k=1 to n-1): weight w_{n-1+k}
  grad[k:n] += w_{n-1+k} * (1-h[0:n-k])
  grad[0:n-k] -= w_{n-1+k} * h[k:n]

For negative lags (k=1 to n-1): weight w_{n-1-k}
  grad[0:n-k] += w_{n-1-k} * (1-h[k:n])
  grad[k:n] -= w_{n-1-k} * h[0:n-k]

For k=0: weight w_{n-1}
  grad[0:n] += w_{n-1} * (1-h)
  grad[0:n] -= w_{n-1} * h
  Net: grad += w_{n-1} * (1 - 2*h)

This is O(n²) total (sum of n-k for k=1 to n-1 is O(n²/2)). For n=200, that's ~20000 operations, very fast.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_obj(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    mc = corr.max()
    shifted = alpha * (corr - mc)
    exp_s = np.exp(shifted)
    return float((mc + np.log(exp_s.sum()) / alpha) * dx)

def smooth_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    mc = corr.max()
    shifted = alpha * (corr - mc)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    grad = np.zeros(n)
    w0 = w[n - 1]
    grad += w0 * (1.0 - 2.0 * h)
    for k in range(1, n):
        wp = w[n - 1 + k]
        if wp > 1e-12:
            grad[k:] += wp * (1.0 - h[:n - k])
            grad[:n - k] -= wp * h[k:]
        wn = w[n - 1 - k]
        if wn > 1e-12:
            grad[:n - k] += wn * (1.0 - h[k:])
            grad[k:] -= wn * h[:n - k]
    return grad * dx

def subgradient_momentum(h, n, iters, c0=0.4):
    best_h = h.copy()
    best_val = compute_c5(best_h, n)
    h_prev = h.copy()
    for t in range(1, iters + 1):
        step = c0 / (t ** 0.5)
        corr = np.correlate(h, 1 - h, mode='full')
        k_star = int(np.argmax(corr))
        lag = k_star - (n - 1)
        grad = np.zeros(n)
        if lag >= 0:
            im = n - 1 - lag
            grad[lag:]. += (1.0 - h[:im + 1])
            grad[:im + 1] -= h[lag:]
        else:
            l = -lag
            grad[:n - l] += (1.0 - h[l:])
            grad[l:] -= h[:n - l]
        h_new = h - step * grad + 0.85 * (h - h_prev)
        h_new = project_feasible(h_new, n)
        val = compute_c5(h_new, n)
        if val < best_val:
            best_h = h_new.copy()
            best_val = val
        h_prev = h
        h = h_new
    return best_h, best_val

def two_point_refine(h, n, max_iters, rng):
    best_h = h.copy()
    best_c5 = compute_c5(best_h, n)
    for _ in range(max_iters):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        for d in [0.005, 0.01, 0.02, 0.003, 0.015]:
            for s in [1, -1]:
                di = s * d
                hi = best_h[i] + di
                hj = best_h[j] - di
                if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                    trial = best_h.copy()
                    trial[i] = hi
                    trial[j] = hj
                    c = compute_c5(trial, n)
                    if c < best_c5:
                        best_h = trial
                        best_c5 = c
                        break
            else:
                continue
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - t0)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
        h_init = project_feasible(h_init, n_init)
    except Exception:
        h_init = None
        n_init = 80

    best_h = None
    best_c5 = float('inf')
    best_n = n_init

    n_candidates = sorted(set([n_init, 100, 120, 160, 200]))
    n_candidates = [x for x in n_candidates if x <= 300]

    for n in n_candidates:
        if remaining() < 50:
            break

        starts = []
        if n == n_init and h_init is not None:
            starts.append(h_init.copy())

        x = np.linspace(0, 2, n)
        starts.append(project_feasible(rng.uniform(0.2, 0.8, n), n))
        for freq in [1, 2, 3]:
            starts.append(project_feasible(0.5 + 0.35 * np.cos(np.pi * freq * x), n))
        for shift in range(3):
            h_block = np.zeros(n)
            blk = max(1, n // 10)
            for b in range(0, n, 2 * blk):
                s0 = (b + shift * blk) % n
                h_block[s0:min(s0 + blk, n)] = 1.0
            h_block = project_feasible(h_block, n)
            if h_block.max() > 0.01:
                starts.append(h_block)

        for h_start in starts:
            if remaining() < 25:
                break
            h_cur = project_feasible(h_start, n)

            # Phase 1: Subgradient with momentum
            if remaining() > 20:
                iters = int(min(20000, max(5000, remaining() * 1500)))
                h_cur, val = subgradient_momentum(h_cur, n, iters)
                if val < best_c5:
                    best_c5, best_h, best_n = val, h_cur.copy(), n

            # Phase 2: SLSQP with smooth objective (continuation on alpha)
            bounds = [(0.0, 1.0)] * n
            cons = [{'type': 'eq',
                     'fun': lambda h, _n=n: h.sum() - _n / 2.0,
                     'jac': lambda h, _n=n: np.ones(_n)}]
            for alpha in [100, 500, 2000, 8000]:
                if remaining() < 10:
                    break
                try:
                    res = minimize(
                        smooth_obj, h_cur, args=(n, alpha),
                        jac=lambda h, _n=n, _a=alpha: smooth_grad(h, _n, _a),
                        method='SLSQP', bounds=bounds, constraints=cons,
                        options={'maxiter': 1500, 'ftol': 1e-18})
                    h_try = project_feasible(res.x, n)
                    c = compute_c5(h_try, n)
                    if c < best_c5:
                        best_c5, best_h, best_n = c, h_try.copy(), n
                    if c < compute_c5(h_cur, n):
                        h_cur = h_try
                except Exception:
                    pass

    # Final polish with 2-point swaps on the best solution
    if best_h is not None and remaining() > 5:
        n_best = best_n
        max_iter = int(min(30000, remaining() * 3000))
        best_h, best_c5 = two_point_refine(best_h, n_best, max_iter, rng)

    # Final SLSQP polish
    if best_h is not None and remaining() > 5:
        n_best = best_n
        bounds = [(0.0, 1.0)] * n_best
        cons = [{'type': 'eq',
                 'fun': lambda h, _n=n_best: h.sum() - _n / 2.0,
                 'jac': lambda h, _n=n_best: np.ones(_n)}]
        try:
            res = minimize(
                smooth_obj, best_h, args=(n_best, 10000),
                jac=lambda h: smooth_grad(h, n_best, 10000),
                method='SLSQP', bounds=bounds, constraints=cons,
                options={'maxiter': 3000, 'ftol': 1e-18})
            h_try = project_feasible(res.x, n_best)
            c = compute_c5(h_try, n_best)
            if c < best_c5:
                best_c5, best_h = c, h_try
        except Exception:
            pass

    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c5 = compute_c5(best_h, n_init)
        best_n = n_init

    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return best_h, best_c5, best_n
```
