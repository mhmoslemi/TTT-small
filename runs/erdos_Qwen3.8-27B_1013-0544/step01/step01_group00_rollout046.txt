def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

    return v[-1]


def t_ineq(v):
    n = len(v) - 1
    h = v[:n]
    t = v[-1]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return t - corr


def t_eq_fun(v):
    n = len(v) - 1
    return np.sum(v[:n]) - n * 0.5


def t_eq_jac(v):
    n = len(v) - 1
    j = np.zeros(len(v))
    j[:n] = 1.0
    return j


def smooth_obj(h, temp):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    shifted = (corr - m) / temp
    shifted = np.clip(shifted, -500, 500)
    return m + temp * np.log(np.sum(np.exp(shifted)))


def sum_eq_fun(h):
    return np.sum(h) - len(h) * 0.5


def sum_eq_jac(h):
    return np.ones(len(h))


def project_h(h, target):

            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def optimize_smooth(h0, temp, max_iter=300, ftol=1e-12):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac}]

                   bounds=bounds, constraints=cons,

def pc_to_grid(params, n_pieces, n_grid):
    """Convert piecewise constant params to fine grid."""
    m = n_pieces
    log_widths = params[:m]
    heights = params[m:]
    # Softmax for widths
    exp_w = np.exp(log_widths - np.max(log_widths))
    widths = 2.0 * exp_w / np.sum(exp_w)
    # Integral constraint penalty: sum(h_i * w_i) should be 1
    integral = np.sum(heights * widths)
    # Build grid
    x = np.linspace(0, 2, n_grid, endpoint=False)
    h = np.zeros(n_grid)
    edges = np.concatenate([[0], np.cumsum(widths)])
    for i in range(m):
        mask = (x >= edges[i]) & (x < edges[i + 1])
        h[mask] = heights[i]
    h = np.clip(h, 0.0, 1.0)
    h = project_h(h, n_grid * 0.5)
    return h
