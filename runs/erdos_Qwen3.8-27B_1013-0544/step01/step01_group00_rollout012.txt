import numpy as np
from scipy.optimize import minimize, LinearConstraint
import time


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))


def project_h(h, target):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = np.sum(mask)
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)


def interp_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.arange(n_old) / n_old
    x_new = np.arange(n_new) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_h(h_new, n_new * 0.5)


def lse_objective(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip((corr - m) / temp, -50.0, 0.0)
    return m + temp * np.log(np.sum(np.exp(shifted)))


def lse_gradient(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip((corr - m) / temp, -50.0, 0.0)
    exp_vals = np.exp(shifted)
    sw = np.sum(exp_vals)
    if sw < 1e-300:
        return np.zeros(n)
    weights = exp_vals / sw
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        s = k - (n - 1)
        wk = weights[k] * dx
        if wk < 1e-15:
            continue
        js_start = max(0, s)
        js_end = min(n, s + n)
        if js_start < js_end:
            grad[js_start:js_end] += wk * (1.0 - h[js_start - s:js_end - s])
        jp_start = max(0, -s)
        jp_end = min(n, n - s)
        if jp_start < jp_end:
            grad[jp_start:jp_end] -= wk * h[jp_start + s:jp_end + s]
    return grad


def sum_eq_fun(h):
    return np.array([np.sum(h) - len(h) * 0.5])


def sum_eq_jac(h):
    return np.ones((1, len(h)))


def optimize_tc(h0, temp, maxiter=80):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    A_eq = np.ones((1, n))
    b_eq = np.array([n * 0.5])
    cons = [LinearConstraint(A_eq, b_eq, b_eq)]
    res = minimize(lse_objective, h0.copy(), jac=lambda x: lse_gradient(x, temp),
                   args=(temp,), method='trust-constr', bounds=bounds,
                   constraints=cons,
                   options={'maxiter': maxiter, 'gtol': 1e-9, 'x0_step': 0.1})
    h_new = np.clip(res.x, 0.0, 1.0)
    h_new = project_h(h_new, n * 0.5)
    return h_new, compute_c5(h_new)


def t_objective(v):
    return v[-1]


def t_ineq(v):
    n = len(v) - 1
    h = v[:n]
    t = v[-1]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return t - corr


def t_eq_fun(v):
    n = len(v) - 1
    return np.sum(v[:n]) - n * 0.5


def t_eq_jac(v):
    n = len(v) - 1
    j = np.zeros(len(v))
    j[:n] = 1.0
    return j


def optimize_slsqp_t(h0, maxiter=400):
    n = len(h0)
    c5_cur = compute_c5(h0)
    v0 = np.append(h0.copy(), c5_cur + 0.02)
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.5)]
    cons = [
        {'type': 'ineq', 'fun': t_ineq},
        {'type': 'eq', 'fun': t_eq_fun, 'jac': t_eq_jac}
    ]
    res = minimize(t_objective, v0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-10})
    h_new = np.clip(res.x[:n], 0.0, 1.0)
    h_new = project_h(h_new, n * 0.5)
    return h_new, compute_c5(h_new)


def coordinate_descent(h, max_iter=100, n_trials=15):
    n = len(h)
    h = h.copy()
    c5_cur = compute_c5(h)
    grid = np.linspace(-0.15, 0.15, n_trials)
    for _ in range(max_iter):
        improved = False
        for i in range(n):
            j = (i + 1) % n
            old_i, old_j = h[i], h[j]
            best_delta = 0.0
            best_val = c5_cur
            for d in grid:
                ni = old_i + d
                nj = old_j - d
                if 0.0 <= ni <= 1.0 and 0.0 <= nj <= 1.0:
                    h[i], h[j] = ni, nj
                    v = compute_c5(h)
                    if v < best_val:
                        best_val = v
                        best_delta = d
            if best_delta != 0.0:
                h[i], h[j] = old_i + best_delta, old_j - best_delta
                c5_cur = best_val
                improved = True
            else:
                h[i], h[j] = old_i, old_j
        if not improved:
            break
    return h, c5_cur


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - t0)

    h = np.array(initial_h_values, dtype=float)
    n = len(h)

    # Phase 2: Trust-constr with LSE at n=180
    temps = [0.05, 0.03, 0.01, 0.005, 0.002, 0.001]
    for temp in temps:
        if time_left() < 80:

        h_new, c5_new = optimize_tc(h_best, temp, maxiter=60)

            c5_best = c5_new

    # Phase 3: SLSQP t-variable again at n=180
    if time_left() > 120:
        h_new, c5_new = optimize_slsqp_t(h_best, maxiter=500)
        if c5_new < c5_best:

    for n_new in [240, 320]:
        if time_left() < 180:

        h_cur = h_interp.copy()
        c5_cur = compute_c5(h_cur)
        for temp in [0.04, 0.02, 0.01, 0.005, 0.002]:

            if c5_new < c5_best:

    if time_left() > 40:
