        val, grad = smooth_max_and_grad(h_curr, tau)
        v = momentum * v - step * grad
        h_new = project_box_sum(h_curr + v, target)
        new_val = evaluate_c5(h_new)
        if new_val > val + 0.5 * step * np.dot(grad, grad) + 1e-12:
            step *= 0.5
            h_new = project_box_sum(h_curr - step * grad, target)
            new_val = evaluate_c5(h_new)
        elif step < step_init * 0.5 and it % 100 == 99:
            step = min(step * 1.3, step_init)
        h_curr = h_new
        if new_val < best_c5:
            best_c5 = new_val
            best_h = h_new.copy()
    return best_h, best_c5

def topk_subgrad(h, target, k, step):
    n = len(h)
    dx = 2.0 / n
    comp = c5_components(h)
    top_idx = np.argsort(comp)[-k:]
    grad = np.zeros(n)
    for ki in top_idx:
        s = ki - (n - 1)
        if s >= 0:
            if s < n:
                grad[:n - s] += (1.0 - h[s:])
                grad[s:] += (-h[:n - s])
        else:
            ss = -s
            if ss < n:
                grad[ss:] += (1.0 - h[:n - ss])
                grad[:n - ss] += (-h[ss:])
    grad *= dx / (k + 0.1)
    return project_box_sum(h - step * grad, target)

def slsqp_polish(h, target, tau, rem_func):
    n = len(h)
    bnds = Bounds(np.zeros(n), np.ones(n))
    def obj(x):
        return smooth_max_and_grad(x, tau)[0]
    def obj_grad(x):
        return smooth_max_and_grad(x, tau)
    def obj_wrap(x):
        return obj(x)
    def grad_wrap(x):
        return obj_grad(x)[1]
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj_wrap, h, jac=grad_wrap, method='SLSQP',
                       bounds=bnds, constraints=cons,
                       options={'maxiter': 300, 'ftol': 1e-14})
        x = project_box_sum(res.x, target)
        return x, evaluate_c5(x)
    except Exception:
        return h, evaluate_c5(h)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(120, 0.5)
    n0 = len(h_init)
    target0 = float(n0) / 2.0
    best_h = h_init.copy()
    best_c5 = evaluate_c5(best_h)

    def rem():
        return budget_s - (time.time() - t0)

    # Phase 1: PGD with smooth max, decreasing temperature
    tau_schedule = [200, 100, 50, 20, 10, 5, 2, 1, 0.5, 0.2, 0.1]
    for tau in tau_schedule:
        if rem() < 25:
            break
        h_new, c5_new = pgd_smooth(best_h, target0, tau, 400, 0.05, 0.9, rem)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
        # Also try SLSQP polish at this tau
        if rem() > 15:
            h_s, c5_s = slsqp_polish(best_h, target0, tau, rem)
            if c5_s < best_c5:
                best_h = h_s
                best_c5 = c5_s

    # Phase 2: Top-k subgradient with adaptive step
    for k in [1, 2, 3, 5, 8]:
        if rem() < 10:
            break
        step = 0.02
        for _ in range(200):
            if rem() < 5:
                break
            h_new = topk_subgrad(best_h, target0, k, step)
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                best_h = h_new
                best_c5 = c5_new
                step = min(step * 1.1, 0.1)
            else:
                step *= 0.8
                if step < 1e-5:
                    break

    # Phase 3: Multi-resolution探索
    for tn in [80, 100, 160, 200, 240]:
        if rem() < 40:
            break
        if tn % 2 != 0:
            continue
        target_t = float(tn) / 2.0
        x_old = np.linspace(0, 1, len(best_h))
        x_new = np.linspace(0, 1, tn)
        h_t = np.interp(x_new, x_old, best_h)
        h_t = project_box_sum(h_t, target_t)
        for tau in [30, 10, 5, 2, 1, 0.5]:
            if rem() < 15:
                break
            h_new, c5_new = pgd_smooth(h_t, target_t, tau, 250, 0.05, 0.9, rem)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
            h_t = h_new
        if rem() > 10:
            h_s, c5_s = slsqp_polish(h_t, target_t, 1.0, rem)
            if c5_s < best_c5:
                best_h = h_s
                best_c5 = c5_s

    # Phase 4: Random perturbations around best
    hn = len(best_h)
    tgt = float(hn) / 2.0
    for trial in range(30):
        if rem() < 15:
            break
        scale = 0.03 + 0.12 * np.random.random()
        hp = best_h + np.random.normal(0, scale, hn)
        hp = project_box_sum(hp, tgt)
        for tau in [5, 2, 0.5]:
            if rem() < 8:
                break
            h_new, c5_new = pgd_smooth(hp, tgt, tau, 150, 0.03, 0.85, rem)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
                hp = h_new
                # Keep going from improved point
            else:
                hp = h_new

    # Phase 5: Final SLSQP polish at multiple temperatures
    if rem() > 10:
        for tau in [1, 0.5, 0.2]:
            if rem() < 8:
                break
            h_s, c5_s = slsqp_polish(best_h, tgt, tau, rem)
            if c5_s < best_c5:
                best_h = h_s
                best_c5 = c5_s

    # Final constraint enforcement
    nf = len(best_h)
    best_h = project_box_sum(best_h, float(nf) / 2.0)
    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, nf
