    """Project onto {x: 0<=x<=1, sum(x)=target_sum} via bisection on Lagrange multiplier."""
    lo, hi = -1.0, 1.0

        else:

    """Gradient of c(k) = sum h[i]*(1-h[i+k]) w.r.t. h."""
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        g[:n-k] += (1.0 - h[k:])

import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1 - h, mode='full')

def compute_c5(h):
    n = len(h)
    return float(np.max(compute_corr(h)) * (2.0 / n))

def project_sum_box(h, target_sum):
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def grad_lag(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        g[:n-k] += (1.0 - h[k:])
        if k > 0:
            g[k:] -= h[:n-k]
    else:
        p = -k
        g[p:] += (1.0 - h[:n-p])
        g[:n-p] -= h[p:]
    return g

def smooth_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    s = np.sum(w)
    if s < 1e-100:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= s
    grad = np.zeros(n)
    thresh = 1.0 / (n * 100.0)
    active = np.where(w > thresh)[0]
    for idx in active:
        k = idx - (n - 1)
        grad += w[idx] * grad_lag(h, k)
    return grad * (2.0 / n)

def gradient_optimize(h0, target_sum, deadline, rng, alpha_init, lr_init):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    alpha = alpha_init
    stall = 0

    while time.time() < deadline:
        g = smooth_grad(h, alpha)
        step = lr
        improved = False
        for _ in range(10):
            h_new = project_sum_box(h - step * g, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5 - 1e-15:
                h = h_new
                best_h = h.copy()
                best_c5 = c5
                improved = True
                break
            step *= 0.5
        if improved:
            stall = 0
            lr = min(lr * 1.04, 0.06)
        else:
            stall += 1
            h = project_sum_box(h - step * g, target_sum)
            if stall % 20 == 0:
                lr = max(lr * 0.85, 0.0008)
            if stall % 40 == 0:
                alpha = min(alpha * 1.35, 500.0)
            if stall > 130:
                h = best_h + rng.randn(n) * 0.012
                h = project_sum_box(h, target_sum)
                lr = lr_init * 0.8
                alpha = alpha_init
                stall = 0
    return best_h, best_c5

def binary_sa(h_start, n, deadline, rng):
    target = n // 2
    b = (h_start >= 0.5).astype(np.float64)
    cnt = int(np.sum(b))
    if cnt > target:
        idx = np.nonzero(b)[0]
        rng.shuffle(idx)
        b[idx[:cnt - target]] = 0
    elif cnt < target:
        idx = np.nonzero(1 - b)[0]
        rng.shuffle(idx)
        b[idx[:target - cnt]] = 1

    best_b = b.copy()
    best_c5 = compute_c5(b)
    T = 0.005
    ones = np.nonzero(b)[0]
    zeros = np.nonzero(1 - b)[0]

    while time.time() < deadline:
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        b[i] = 0
        b[j] = 1
        c5 = compute_c5(b)
        if c5 < best_c5:

            best_c5 = c5

            ones = np.nonzero(b)[0]
            zeros = np.nonzero(1 - b)[0]
        else:
            b[i] = 1
            b[j] = 0
        T = max(T * 0.9998, 0.0004)
    return best_b, best_c5
