||J_k||_2^2 = sum_j (∂corr_k/∂h_j)^2

For corr_k = sum_i h[i](1-h[i+k]), the gradient has at most n non-zero components, each bounded by 1 in absolute value. So ||J_k||_2^2 <= n.

Thus ||J||_2^2 <= sum_k n = (2n-1) * n ≈ 2n².

So ||first term||_2 <= alpha * 1 * 2n² = 2 * alpha * n².

For alpha=100, n=200: 2 * 100 * 40000 = 8,000,000.

That's the Lipschitz constant! So lr should be about 1/8e6 = 1.25e-7. That's incredibly small.

But wait, this is a very rough upper bound. In practice, the weights w_k are concentrated on a few k values (the ones with max corr), so the effective dimension is much smaller. And the J_k vectors for different k are not aligned, so the sum doesn't add up as badly.

In practice, the effective Lipschitz constant is probably alpha * n (not alpha * n²). For alpha=100, n=200: L ≈ 20000, lr ≈ 5e-5.

But the previous code used lr=0.005 and it was making progress (albeit slowly). This suggests that either:
1. The effective Lipschitz constant is much smaller than my estimate
2. The method is working in a subgradient-like fashion despite the large step size
3. The projection is acting as a natural step size limiter

I think option 3 is likely. The projection onto the simplex-box constraint effectively limits the step size, because after a large gradient step, the projection clips many values to 0 or 1, and the sum constraint further restricts the change.

So let me not worry too much about the theoretical step size and instead use:
- A moderate initial lr (like 0.01)
- Backtracking line search: if the objective increases, halve the step size
- This is robust and adapts to the local landscape

Let me finalize my approach:

def optimize_with_backtrack(h0, target, alpha, deadline, rng, max_iter=5000):

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def compute_corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def projection(h, target_sum):
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def grad_obj(h, alpha):
    n = len(h)
    corr = compute_corr_full(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted -= np.max(shifted)
    w = np.exp(shifted)
    ws = np.sum(w)
    if ws < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= ws
    f = 1.0 - h
    grad = np.zeros(n)
    widx = np.arange(2 * n - 1)
    for j in range(n):
        k_lo = max(-(n - 1), -j)
        k_hi = min(n - 1, n - 1 - j)
        A = 0.0
        if k_lo <= k_hi:
            ks = np.arange(k_lo, k_hi + 1)
            A = np.dot(w[ks + n - 1], f[j + ks])
        k_lo2 = max(-(n - 1), j - (n - 1))
        k_hi2 = min(n - 1, j)
        B = 0.0
        if k_lo2 <= k_hi2:
            ks2 = np.arange(k_lo2, k_hi2 + 1)
            B = np.dot(w[ks2 + n - 1], h[j - ks2])
        grad[j] = A - B
    return grad * (2.0 / n)

def optimize_phase(h0, target, alphas, deadline, rng, iters_per_alpha=2000):
    n = len(h0)
    h = h0.copy()
    best_h = h0.copy()
    best_c5 = compute_c5(h)
    for alpha in alphas:
        lr = 0.02
        no_improve = 0
        stall = 0
        for it in range(iters_per_alpha):
            if time.time() > deadline:
                return best_h, best_c5
            grad = grad_obj(h, alpha)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-10:
                break
            accepted = False
            for _ in range(20):
                h_new = projection(h - lr * grad, target)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-14:
                    h = h_new
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    no_improve = 0
                    accepted = True
                    break
                lr *= 0.5
            if not accepted:
                no_improve += 1
                if no_improve > 60:
                    h = best_h + rng.randn(n) * (0.02 + 0.01 * stall)
                    h = projection(h, target)
                    c5_h = compute_c5(h)
                    if c5_h < best_c5:
                        best_h = h.copy()
                        best_c5 = c5_h
                    stall += 1
                    no_improve = 0
                    lr = 0.02
                    if stall > 10:
                        break
            else:
                lr = min(lr * 1.08, 0.15)
    return best_h, best_c5

def interpolate_to_n(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False) + 1.0 / n_new
    h_ext = np.append(h_old, h_old[0])
    x_ext = np.append(x_old, 2.0 + 1.0 / n_old)
    return np.interp(x_new, x_ext, h_ext)

def polish(h, target, deadline, rng):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = 0.005
    while time.time() < deadline:
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            old = best_h[j]
            for dv in [step, -step, step * 0.5, -step * 0.5]:
                new_val = old + dv
                if new_val < 0 or new_val > 1:
                    continue
                cand = best_h.copy()
                cand[j] = new_val
                cand = projection(cand, target)
                c5_c = compute_c5(cand)
                if c5_c < best_c5 - 1e-12:
                    best_h = cand
                    best_c5 = c5_c
                    improved = True
                    break
        if not improved:
            step *= 0.5
            if step < 1e-5:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = 0.5 * np.ones(200)

    n = len(h0)
    target = n / 2.0
    h0 = projection(h0, target)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Continuation at initial resolution
    alphas_p1 = [30, 60, 100, 180, 300, 500]
    deadline1 = t0 + min(budget_s * 0.35, 400)
    best_h, best_c5 = optimize_phase(best_h, target, alphas_p1, deadline1, rng)

    # Phase 2: Try lower resolution (sometimes easier landscape)
    remaining = budget_s - (time.time() - t0)
    if remaining > 100:
        for n_try in [150, 100]:
            if time.time() - t0 > budget_s - 80:
                break
            h_interp = interpolate_to_n(best_h, n, n_try)
            h_interp = projection(h_interp, n_try / 2.0)
            dl = time.time() + min(60, budget_s - 60 - (time.time() - t0))
            h_opt, c5_opt = optimize_phase(h_interp, n_try / 2.0,
                                            [50, 100, 200, 400], dl, rng)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
                n = n_try
                target = n / 2.0

    # Phase 3: Try higher resolution
    remaining = budget_s - (time.time() - t0)
    if remaining > 80:
        for n_try in [250, 400]:
            if time.time() - t0 > budget_s - 60:
                break
            h_interp = interpolate_to_n(best_h, len(best_h), n_try)
            h_interp = projection(h_interp, n_try / 2.0)
            dl = time.time() + min(50, budget_s - 40 - (time.time() - t0))
            h_opt, c5_opt = optimize_phase(h_interp, n_try / 2.0,
                                            [80, 150, 300], dl, rng)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt

    # Phase 4: Multiple restarts at best resolution
    remaining = budget_s - (time.time() - t0)
    n_cur = len(best_h)
    target_cur = n_cur / 2.0
    n_restarts = max(1, int(remaining / 35))
    for r in range(n_restarts):
        if time.time() - t0 > budget_s - 15:
            break
        scale = 0.03 + 0.01 * r
        h_pert = best_h + rng.randn(n_cur) * scale
        h_pert = projection(h_pert, target_cur)
        dl = time.time() + min(25, budget_s - 10 - (time.time() - t0))
        h_opt, c5_opt = optimize_phase(h_pert, target_cur, [100, 200, 400], dl, rng)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt

    # Phase 5: Coordinate polish
    remaining = budget_s - (time.time() - t0)
    if remaining > 10:
        best_h, best_c5 = polish(best_h, len(best_h) / 2.0,
                                  t0 + budget_s - 3, rng)

    best_h = projection(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
