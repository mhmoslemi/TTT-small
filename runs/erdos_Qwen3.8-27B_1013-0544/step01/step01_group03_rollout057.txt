from scipy.optimize import minimize, Bounds, LinearConstraint
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(obj, tau):
    x = tau * obj
    x_max = np.max(x)
    return float(x_max / tau + np.log(np.sum(np.exp(x - x_max))) / tau)

    if abs(diff) < 1e-12:
        return h
    # Adjust non-bound entries
    interior = (h > 1e-8) & (h < 1 - 1e-8)
    if interior.sum() > 0:
        h[interior] += diff / interior.sum()
    else:
        # Fallback: adjust all
        h += diff / len(h)
    return np.clip(h, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(obj, tau):
    x = tau * obj
    x_max = np.max(x)
    return float(x_max / tau + np.log(np.sum(np.exp(x - x_max))) / tau)

def fix_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    interior = (h > 1e-10) & (h < 1.0 - 1e-10)
    if interior.sum() > 0:
        h[interior] += diff / interior.sum()
    else:
        h += diff / len(h)
    return np.clip(h, 0.0, 1.0)
