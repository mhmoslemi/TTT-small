import time

def compute_c5(h, n):

def compute_corr(h, n):

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += (d / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += (d / s) * h
    return np.clip(h, 0.0, 1.0)

def sum_con(h, n):
    return np.sum(h) - n / 2.0

def sum_con_jac(h, n):
    return np.ones(n)

def soft_max_val(h, n, alpha):
    c = compute_corr(h, n)
    m = c.max()
    s = np.sum(np.exp(alpha * (c - m)))
    return m + np.log(s) / alpha

def obj_sm(h, n, alpha):
    return soft_max_val(h, n, alpha)

def grad_sm(h, n, alpha):
    dx = 2.0 / n
    c = compute_corr(h, n)
    m = c.max()
    w = np.exp(alpha * (c - m))
    w /= w.sum()
    grad = np.zeros(n)
    for i in range(n):
        s1 = 0.0
        s2 = 0.0
        for m_idx in range(2 * n - 1):
            wm = w[m_idx]
            if wm < 1e-12:
                continue
            j1 = i + m_idx - n + 1
            if 0 <= j1 < n:
                s1 += wm * (1.0 - h[j1])
            j2 = i - m_idx + n - 1
            if 0 <= j2 < n:
                s2 += wm * h[j2]
        grad[i] = alpha * dx * (s1 - s2)
    return grad

def proj_gradient_descent(h, n, alpha, max_iter=2000, tol=1e-14):
    h = project_feasible(h.copy(), n)
    best_val = soft_max_val(h, n, alpha)
    best_h = h.copy()
    momentum = np.zeros(n)
    mu = 0.9
    step = 0.5
    for it in range(max_iter):
        g = grad_sm(h, n, alpha)
        momentum = mu * momentum - step * g
        h_new = project_feasible(h + momentum, n)
        val = soft_max_val(h_new, n, alpha)
        if val < best_val - 1e-16:
            best_val = val
            best_h = h_new.copy()
            h = h_new
        else:
            step *= 0.7
            mu = 0.5
            if step < 1e-8:
                break
    return best_h, compute_c5(best_h, n)

def focused_swap(h, n, max_iter=50000, seed=42):
    rng = np.random.RandomState(seed)
    best_c5 = compute_c5(h, n)
    h = h.copy()
    remaining = max_iter
    while remaining > 0 and True:
        c = compute_corr(h, n)
        k_best = c.argmax()
        deltas = [0.001, 0.003, 0.005, 0.01, 0.02, 0.05]
        improved = False
        batch = min(remaining, 200)
        for _ in range(batch):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            for d in deltas:
                for sgn in [1, -1]:
                    di = sgn * d

def slsqp_optimize(h, n, alphas, max_iter_each=2000, time_limit=60):
    t_start = time.time()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    best_h = project_feasible(h.copy(), n)
    best_c5 = compute_c5(best_h, n)
    for alpha in alphas:
        if time.time() - t_start > time_limit or (time.time() - t_start) < 0:
            break
        try:
            res = minimize(obj_sm, best_h, args=(n, alpha), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': max_iter_each, 'ftol': 1e-18})
            h_try = project_feasible(res.x, n)
            c = compute_c5(h_try, n)
            if c < best_c5:
                best_h, best_c5 = h_try, c
        except Exception:
            pass
    return best_h, best_c5
