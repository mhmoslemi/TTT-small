The autocorrelation R_h(k) = Σ_i h[i]h[i+k] is related to the periodogram (via FFT). Specifically, if we take the DFT of h, then R_h is the inverse DFT of |H|^2 (the periodogram).

So we want to maximize min_k R_h(k), which in the frequency domain means we want |H(f)|^2 to be as "flat" as possible (since a flat periodogram gives constant autocorrelation).

Wait, that's not quite right because R_h(k) = Σ_i h[i]h[i+k] is the cyclic autocorrelation, and its DFT is |H(f)|^2. So R_h(k) = (1/n) Σ_f |H(f)|^2 * exp(2πiftk).

If |H(f)|^2 is constant (= n/2 for binary h), then R_h(k) = 1/2 for all k≠0 and R_h(0) = n/2. But we can't have a constant periodogram for a binary sequence (that would require all |H(f)| to be equal, which is a Barker sequence property).

Actually, for a binary sequence with n/2 ones, the periodogram is:
|H(f)|^2 = |Σ_i h[i] exp(-2πifk/n)|^2

For a "good" sequence, we want this to be as flat as possible (except at f=0). This is related to the concept of "low autocorrelation" sequences.

So one approach: **optimize the frequency domain representation**. Define h in terms of its DFT coefficients, with the constraint that the inverse DFT is in [0,1] and sums to n/2. Then optimize for flatness of |H(f)|^2.

import numpy as np
import time
from scipy.optimize import minimize

def _compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)

def _smooth_max_grad(h, t):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = corr.max()
    tc = t * (corr - mc)
    tc -= tc.max()
    w = np.exp(tc)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(len(corr)):
        if w[k] < 1e-10:
            continue
        ks = (k - (n - 1)) % n
        g += w[k] * (1.0 - np.roll(h, -ks) - np.roll(h, ks))
    return mc, g

def _project(h, ts):
    lo, hi = -3.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        p = np.clip(h - mid, 0.0, 1.0)
        if p.sum() > ts:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def _fix_sum(h, ts):
    diff = ts - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
    return np.clip(h, 0.0, 1.0)

def _slsqp_obj(x, dx, t, n):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = corr.max()
    tc = t * (corr - mc)
    tc -= tc.max()
    return (mc + np.log(np.sum(np.exp(tc))) / t) * dx

def _slsqp_jac(x, dx, t, n):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mc = corr.max()
    tc = t * (corr - mc)
    tc -= tc.max()
    w = np.exp(tc)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(len(corr)):
        if w[k] < 1e-10:
            continue
        ks = (k - (n - 1)) % n
        g += w[k] * (1.0 - np.roll(x, -ks) - np.roll(x, ks))
    return g * dx

    return x.sum() - ts

                       jac=_slsqp_jac, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _constraint_eq, 'args': (ts,)}],
                       options={'maxiter': maxiter, 'ftol': 1e-10})
        return _project(res.x, ts)
    except Exception:
        return h

def _gradient_descent(h, dx, ts, max_iter, t0, t1, time_limit):
    n = len(h)
    beta = 0.9
    lr = 0.4
    mom = np.zeros(n)
    h_cur = h.copy()
    best_c5 = _compute_c5(h_cur, dx)
    best_h = h_cur.copy()
    stalld = 0
    for it in range(max_iter):
        if it > 0 and it % 100 == 0:
            if time.time() - time_limit > 0:
                break
        frac = it / max(1, max_iter)
        t = t0 + (t1 - t0) * frac
        mc, g = _smooth_max_grad(h_cur, t)
        g -= g.mean()
        mom = beta * mom + (1.0 - beta) * g
        h_new = h_cur - lr * mom
        h_new = _project(h_new, ts)
        c5_new = _compute_c5(h_new, dx)
        if c5_new < mc * dx:

            lr = min(lr * 1.02, 1.5)

                lr = 0.3
                stalld = 0
        if mc * dx < best_c5:

            best_h = h_cur.copy()
    return best_h, best_c5

    phases = rng.uniform(0, 2 * np.pi, n)
    h_complex = np.fft.ifft(np.exp(1j * phases))
    h_real = np.real(h_complex)
    h_real = (h_real - h_real.min()) / (h_real.max() - h_real.min() + 1e-15)
    h_real *= 0.6
    h_real += 0.2
    return _project(h_real, n / 2.0)
