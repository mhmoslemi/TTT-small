- This is: minimize ||h-x||² s.t. 0≤h≤1, sum(h)=target
- Solution: h[i] = clip(x[i] - λ, 0, 1) for some λ
- Find λ by bisection

```python
import numpy as np
import time
from scipy.linalg import toeplitz

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, target_sum, n):
    """Project x onto {h: 0<=h<=1, sum(h)=target_sum}"""
    # Find lambda such that sum(clip(x - lambda, 0, 1)) = target_sum
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    return np.clip(x - lam, 0.0, 1.0)

def smooth_gradient(h, t):
    n = len(h)
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full")
    
    max_c = np.max(corr)
    W = np.exp(t * (corr - max_c))
    W /= np.sum(W)
    
    # Build Toeplitz matrices
    fc1 = W[n-1::-1]  # W[n-1], W[n-2], ..., W[0]
    fr1 = W[n-1:]     # W[n-1], W[n], ..., W[2n-2]
    T1 = toeplitz(fc1, fr1)
    
    fc2 = W[n-1:]     # W[n-1], W[n], ..., W[2n-2]
    fr2 = W[n-1::-1]  # W[n-1], W[n-2], ..., W[0]
    T2 = toeplitz(fc2, fr2)
    
    return T1 @ (1.0 - h) - T2 @ h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    def elapsed():
        return time.time() - t0
    
    h0 = np.array(initial_h_values, dtype=float).copy()
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = np.clip(h0, 0.0, 1.0)
    
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)
    
    # Phase 1: Subgradient method with momentum
    h = best_h.copy()
    v = np.zeros(n)
    beta = 0.9  # momentum
    step_size = 0.1
    
    t_schedule = [1, 2, 5, 10, 20, 50, 100, 200, 500]
    
    for t_val in t_schedule:
        if elapsed() > budget_s - 200:
            break
        n_iter = 500
        for it in range(n_iter):
            if elapsed() > budget_s - 200:
                break
            g = smooth_gradient(h, t_val)
            # Nesterov-style update
```

```python
import numpy as np

from scipy.optimize import minimize
from scipy.linalg import toeplitz

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(x, target_sum):
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def smooth_grad(h, t):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_c = np.max(corr)
    W = np.exp(t * (corr - max_c))
    W /= np.sum(W)
    fc1 = W[n-1::-1]
    fr1 = W[n-1:]
    fc2 = W[n-1:]
    fr2 = W[n-1::-1]
    T1 = toeplitz(fc1, fr1)
    T2 = toeplitz(fc2, fr2)
    return T1 @ (1.0 - h) - T2 @ h

def smooth_obj(h, t):
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_c = np.max(corr)
    vals = t * (corr - max_c)
    vals[vals < -50] = -50
    return (max_c + np.log(np.sum(np.exp(vals))) / t)

def subgrad_method(h, target_sum, dx, n_iter, t_start, t_end, beta=0.9):
    n = len(h)
    h_curr = h.copy()
    v = np.zeros(n)
    lr0 = 0.5
    best = h_curr.copy()
    best_c5 = compute_c5(h_curr, dx)
    for it in range(n_iter):
        frac = it / max(n_iter - 1, 1)
        t_val = t_start * (t_end / t_start) ** frac
        lr = lr0 * (1.0 - 0.5 * frac)
        g = smooth_grad(h_curr, t_val)
        v = beta * v - lr * g
        h_new = project_feasible(h_curr + v, target_sum)
        c5 = compute_c5(h_new, dx)
        if c5 < best_c5:
            best = h_new.copy()
            best_c5 = c5
        h_curr = h_new
    return best, best_c5

def slsqp_refine(h, target_sum, dx, t, maxiter):
    n = len(h)
    def obj(x):
        return smooth_obj(x, t)
    def obj_grad(x):
        return smooth_grad(x, t)
    res = minimize(
        obj, h.copy(), jac=obj_grad,
        method='SLSQP',
        bounds=[(0.0, 1.0)] * n,
        constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                       'jac': lambda x: np.ones(n)}],
        options={'maxiter': maxiter, 'ftol': 1e-10}
    )
    h_new = project_feasible(res.x, target_sum)
    return h_new

        if c5 < best_c5:
```

At the optimum, the "bottleneck" lags (those achieving the max) should all be equal. So a good local search is:
1. Find the top m lags
2. Try to equalize them (make them all equal to the minimum possible)
3. This is a linear system: find h such that f_{k1}(h) = f_{k2}(h) = ... = f_{km}(h) = t (minimize t)

This is a QP (since each f_k is quadratic), but for the linearized version, it's a LP.

Actually, for the linearized problem (fixing all but a subset of variables), it becomes:
min_t t
s.t. f_k(h) ≤ t for k in active set
     0 ≤ h ≤ 1
     sum(h) = n/2

Since f_k is quadratic in h, this is a QP. But we can linearize f_k at the current h:
f_k(h + δ) ≈ f_k(h) + ∇f_k(h)·δ

Then the problem becomes:
min_t t
s.t. f_k(h) + ∇f_k(h)·δ ≤ t for k in active set
     0 ≤ h+δ ≤ 1
     sum(δ) = 0

This is a linear program! And it's small (n variables + 1, with ~n + m constraints). We can solve it with scipy or cvxpy.

Let me implement this "linearized active set" approach:

```python
from scipy.optimize import linprog

def active_set_qp(h, target_sum, dx, dx_step=0.01):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(corr)
    
    # Find active lags (within 5% of max)
    threshold = max_val * 0.95
    active = np.where(corr >= threshold)[0]
    
    # Linearize: minimize t s.t. f_k(h + step * g_k) <= t
    # where g_k is a step direction
    # Actually, let's do: find direction d that minimizes max_k (f_k(h) + step * grad_k · d)
    # This is: min_d max_k (c_k + step * grad_k · d)
    
    # Compute gradients of active lags
    grads = []
    for idx in active:
        k = idx - (n - 1)
        g = np.zeros(n)
        if k >= 0:
            g[:n-k] = (1.0 - h[k:])
            g[k:] -= h[:n-k]
        else:
            kp = -k
            g[kp:] = (1.0 - h[:n-kp])
            g[:n-kp] -= h[kp:]
        grads.append(g)
    
    # min t s.t. c_k + step * grads[k]·d <= t
    #            d · 1 = 0 (preserve sum)
    #            0 <= h + step * d <= 1
    
    m = len(active)
    # Variables: d (n) + t (1) = n+1 variables
    # Objective: minimize t → c_obj = [0...0, 1]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality constraints: c_k + step * grads[k]·d - t <= 0
    # → c_k + step * grads[k]·d - t <= 0
    # → [step * grads[k], -1] · [d, t] <= -c_k
    A_ub = np.zeros((m, n + 1))
    for i in range(m):
        A_ub[i, :n] = dx_step * grads[i]
        A_ub[i, n] = -1.0
    b_ub = -corr[active]
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [0.0]
    
    # Bounds: -h/step <= d <= (1-h)/step
    bounds = [(-h[i]/dx_step, (1-h[i])/dx_step) for i in range(n)] + [(0, None)]
    
    res = linprog(c_obj, A_ub, b_ub, A_eq, b_eq, bounds=bounds, method='highs')
    if res.success:

        h_new = project_feasible(h + dx_step * d, target_sum)
```
