def gradient_c(h, n, m):
    """Gradient of c[m] = correlate(h, 1-h, 'full')[m] with respect to h"""
    lag = m - (n - 1)
    grad = np.zeros(n)
    # First term: (1 - h[j - lag]) for valid j - lag
    if lag >= 0:
        # j - lag >= 0 means j >= lag, j - lag < n means j < n + lag
        # So j ranges from lag to n-1
        grad[lag:] = 1 - h[:n-lag]
    else:
        # j - lag >= 0 means j >= 0 (always true since j >= 0)
        # j - lag < n means j < n + lag, i.e., j < n + lag
        # Since lag < 0, n + lag < n
        # So j ranges from 0 to n + lag - 1
        grad[:n+lag] = 1 - h[-lag:]
    
    # Second term: -h[j + lag] for valid j + lag
    if lag >= 0:
        # j + lag < n means j < n - lag
        # j + lag >= 0 always true
        grad[:n-lag] -= h[lag:]
    else:
        # j + lag >= 0 means j >= -lag
        # j + lag < n means j < n - lag
        grad[-lag:] -= h[:n+lag]
    
    return grad

    xc = c * dx
    xc_shifted = xc - xc.max()
    exp_xc = np.exp(alpha * xc_shifted)

    lags = np.arange(-(n-1), n)  # length 2n-1

def optimize_2swap(h, n, i, j):
    """Find optimal δ for 2-swap h[i]+δ, h[j]-δ"""
    dx = 2.0 / n
    c = np.correlate(h, 1-h, mode='full')
    
    # Compute d_m = coefficient of δ in c[m]
    d = np.zeros(2*n-1)
    lags = np.arange(-(n-1), n)
    
    # d[m] = a_m[i] - a_m[j]
    # a_m[i] = (1-h[i-lag_m])*I[valid] - h[i+lag_m]*I[valid]
    # a_m[j] = (1-h[j-lag_m])*I[valid] - h[j+lag_m]*I[valid]
    
    for m in range(2*n-1):
        lag = lags[m]
        a_i = 0.0
        a_j = 0.0
        if 0 <= i - lag < n:
            a_i += 1 - h[i - lag]
        if 0 <= i + lag < n:
            a_i -= h[i + lag]
        if 0 <= j - lag < n:
            a_j += 1 - h[j - lag]
        if 0 <= j + lag < n:
            a_j -= h[j + lag]
        d[m] = a_i - a_j
    
    # Now minimize max_m (c[m] + d[m]*δ) over δ ∈ [-h[i], 1-h[j]]
    # This is a convex 1D problem
    
    lo = -h[i]
    hi = 1 - h[j]
    
    if lo > hi:
        return 0.0  # No feasible swap
    
    # Use ternary search (or subgradient)
    # Since the function is convex, ternary search works
    for _ in range(60):
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        f1 = np.max(c + d * m1)
        f2 = np.max(c + d * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    
    return (lo + hi) / 2
