from scipy.optimize import minimize
import time

    corr = np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(v, n):
    target = n * 0.5
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(v - lam, 0.0, 1.0)

def lse_objective_and_grad(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)

        w[np.argmax(corr)] = 1.0
        val = m
    else:

        exps = np.exp(shifted)
        s = np.sum(exps)

**Mirror Descent with Entropy Regularization:**

The problem can be written as:
min_h max_k f_k(h) where f_k(h) = ∑_i h[i]*(1-h[i+k])*dx

Mirror descent alternates between:
1. (Inner) For a given distribution p over k, minimize ∑_k p_k * f_k(h) + (1/η) * KL(h || h_prev) subject to constraints
2. (Outer) Update p to put more weight on the worst f_k

But the inner problem is still non-convex (bilinear). However, if we linearize f_k around h_prev, we get a convex problem:
f_k(h) ≈ f_k(h_prev) + ∇f_k(h_prev) · (h - h_prev)

Then the inner problem is:
min_h ∑_k p_k * [f_k(h_prev) + ∇f_k(h_prev) · (h - h_prev)] + (1/η) * KL(h || h_prev)

= min_h ∇g · h + (1/η) * ∑ h[i] * log(h[i]/h_prev[i])

where g = ∑_k p_k * ∇f_k(h_prev)

This is a linear program + entropy, which is convex! We can solve it with gradient descent on the dual, or directly.

The solution is h[i] ∝ h_prev[i] * exp(-η * g[i]) (normalized to satisfy the constraints).

With the box constraints 0≤h≤1 and sum constraint, we can use:
h[i] = clip(h_prev[i] * exp(-η * g[i]) + λ_offset, 0, 1)

where λ_offset is chosen to satisfy the sum constraint.

This is essentially **exponentiated gradient** (or multiplicative weights) applied to our problem!

    n = len(h)

            grad[:k] += wi * dx * (1.0 - h[ka:])

        return j

        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}

    return h_new, compute_c5(h_new)

    """SLSQP with only the top-K worst lags as constraints"""
    n = len(h0)
    dx = 2.0 / n
    
    # Find top-K lags
    corr = np.correlate(h0, 1.0 - h0, mode='full') * dx
    top_k_indices = np.argsort(corr)[-n_active:]
    c5_cur = compute_c5(h0)
    
    v0 = np.append(h0.copy(), c5_cur + 0.005)
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.5)]

        return v[-1]

        h = v[:n]
        t = v[-1]
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        return t - corr[top_k_indices]  # Only top-K constraints

        return np.sum(v[:n]) - n * 0.5
    
    def eq_jac(v):
        j = np.zeros(n + 1)
        j[:n] = 1.0
        return j
    
    cons = [

        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}
    ]

                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-13})
