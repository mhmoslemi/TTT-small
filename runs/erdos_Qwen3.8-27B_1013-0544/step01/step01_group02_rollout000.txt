        ones_idx = np.where(h_bin == 1)[0]
        zeros_idx = np.where(h_bin == 0)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i = ones_idx[rng.randint(len(ones_idx))]
        j = zeros_idx[rng.randint(len(zeros_idx))]
        h_bin[i] = 0
        h_bin[j] = 1
        c_new = compute_c5(h_bin, n)
        if c_new < c:
            c = c_new
        else:
            h_bin[i] = 1
            h_bin[j] = 0
    return h_bin, c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - t0)

    # Get initial values
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h0)
    except Exception:
        n_init = 80
        h0 = np.full(n_init, 0.5)

    best_h = h0.copy()
    best_n = n_init
    best_c5 = compute_c5(best_h, best_n)

    n_values = sorted(set([n_init, 64, 80, 96, 100, 120, 160]))

    for n in n_values:
        if remaining() < 80:
            break

        # Initialize
        if n == n_init:
            h = best_h.copy()
        else:
            candidates = [np.full(n, 0.5)]
            for freq in [2, 3, 5, 7]:
                candidates.append(0.5 + 0.3 * np.sin(2 * np.pi * freq * np.arange(n) / n))
            for freq in [1, 2]:
                candidates.append(0.5 + 0.2 * np.cos(2 * np.pi * freq * np.arange(n) / n + 1))
            best_init = None
            best_init_c = 1e9
            for p in candidates:
                p = project_feasible(p, n)
                c = compute_c5(p, n)
                if c < best_init_c:
                    best_init_c = c
                    best_init = p
            h = best_init.copy()

        c = compute_c5(h, n)

        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: sum_con(x, n), 'jac': lambda: sum_con_jac(x, n)}]

        # Phase 1: SLSQP continuation
        alphas = [10, 30, 100, 300, 1000, 3000, 10000, 30000]
        for alpha in alphas:
            if remaining() < 40:
                break
            try:
                res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 3000, 'ftol': 1e-17})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < c - 1e-14:
                    h, c = h_new, c_new
            except Exception:
                pass

        # Phase 2: Sub-gradient method
        if remaining() > 50:
            for step in range(2000):
                if remaining() < 25:
                    break
                grad, max_val = subgrad_max(h, n, top_k=10)
                gn = np.linalg.norm(grad)
                if gn < 1e-14:
                    break
                lr = 0.005 * (1 + 20 / (step + 1))
                for mult in [1.0, 0.5, 0.25, 0.1]:
                    h_try = project_feasible(h - lr * mult * grad / gn, n)
                    c_try = compute_c5(h_try, n)
                    if c_try < c - 1e-15:
                        h, c = h_try, c_try
                        break

        # Phase 3: SLSQP re-polish after sub-gradient
        if remaining() > 40:
            for alpha in [3000, 10000]:
                if remaining() < 20:
                    break
                try:
                    res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                                  bounds=bounds, constraints=cons,
                                  options={'maxiter': 3000, 'ftol': 1e-17})
                    h_new = project_feasible(res.x, n)
                    c_new = compute_c5(h_new, n)
                    if c_new < c:
                        h, c = h_new, c_new
                except Exception:
                    pass

        # Phase 4: Binary local search
        if remaining() > 40:
            h_bin, c_bin = binary_local_search(h, n, rng, max_iters=3000)
            if c_bin < c:
                # Try to refine from binary solution
                h = h_bin.astype(float)
                c = c_bin
                try:
                    res = minimize(obj_smooth, h, args=(n, 5000), method='SLSQP',
                                  bounds=bounds, constraints=cons,
                                  options={'maxiter': 2000, 'ftol': 1e-16})
                    h_new = project_feasible(res.x, n)
                    c_new = compute_c5(h_new, n)
                    if c_new < c:
                        h, c = h_new, c_new
                except Exception:
                    pass

        # Phase 5: Fine 2-point swaps
        if remaining() > 20:
            deltas = [0.001, 0.002, 0.005, 0.008, 0.01, 0.02]
            count = 0
            while remaining() > 15 and count < 20000:
                count += 1
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                for d in deltas:
                    improved = False
                    for s in [1, -1]:
                        di = s * d
                        hi = h[i] + di
                        hj = h[j] - di
                        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                            h_try = h.copy()
                            h_try[i] = hi
                            h_try[j] = hj
                            c_try = compute_c5(h_try, n)
                            if c_try < c - 1e-15:
                                h = h_try
                                c = c_try
                                improved = True
                                break
                    if improved:
                        break

        # Phase 6: Multi-restart with perturbation
        if remaining() > 60:
            for restart in range(3):
                if remaining() < 30:
                    break
                scale = rng.choice([0.02, 0.05, 0.1])
                h_pert = project_feasible(h + rng.normal(0, scale, size=n), n)
                c_pert = compute_c5(h_pert, n)
                if c_pert < c + 0.001:
                    # Polish this perturbation
                    for alpha in [1000, 5000]:
                        if remaining() < 15:
                            break
                        try:
                            res = minimize(obj_smooth, h_pert, args=(n, alpha), method='SLSQP',
                                          bounds=bounds, constraints=cons,
                                          options={'maxiter': 2000, 'ftol': 1e-16})
                            h_new = project_feasible(res.x, n)
                            c_new = compute_c5(h_new, n)
                            if c_new < c_pert:
                                h_pert, c_pert = h_new, c_new
                        except Exception:
                            pass
                    if c_pert < c:
                        h, c = h_pert, c_pert

        # Final SLSQP polish
        if remaining() > 15:
            try:
                res = minimize(obj_smooth, h, args=(n, 50000), method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 5000, 'ftol': 1e-18})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < c:
                    h, c = h_new, c_new
            except Exception:
                pass

        # Update global best
        final_c = compute_c5(h, n)
        if final_c < best_c5:
            best_h, best_c5, best_n = h.copy(), final_c, n

    # Final feasibility check
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)

    return best_h, best_c5, best_n
