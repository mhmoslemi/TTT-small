    return np.sum(x) - _g_target

def slsqp_polish(h, n, dx, target, max_iter=100):
    global _g_dx, _g_target
    _g_dx = dx
    _g_target = target
    try:
        res = minimize(_obj_func, h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': _con_func}],
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def gd_softmax(h, n, dx, T, max_iter, check_time):
    alpha = 1.0
    h_c = h.copy()
    c5s_c, _ = c5_soft_grad(h_c, dx, T)
    for _ in range(max_iter):
        if check_time():
            break
        c5s_c, grad = c5_soft_grad(h_c, dx, T)
        if np.max(np.abs(grad)) < 1e-12:
            break
        a = alpha
        for _2 in range(20):
            h_t = project_feasible(h_c - a * grad, n)
            c5s_t, _ = c5_soft_grad(h_t, dx, T)
            if c5s_t < c5s_c - 1e-14:
                break
            a *= 0.5
        h_c = project_feasible(h_c - a * grad, n)
        c5s_c, _ = c5_soft_grad(h_c, dx, T)
        alpha = max(a * 1.3, 1e-10)
    return h_c, compute_c5(h_c, dx)

def binary_swap_search(h, n, dx, target, max_iter, check_time, rng):
    h_b = np.where(h > 0.5, 1.0, 0.0)
    # Ensure exact count
    ones = np.sum(h_b == 1.0)
    if ones > target:
        idx = np.where(h_b == 1.0)[0]
        for i in range(int(ones - target)):
            h_b[idx[i]] = 0.0
    elif ones < target:
        idx = np.where(h_b == 0.0)[0]
        for i in range(int(target - ones)):
            h_b[idx[i]] = 1.0
    h_b = project_feasible(h_b, n)
    c5_c = compute_c5(h_b, dx)
    best_h_b = h_b.copy()
    best_c5 = c5_c
    T_sa = 0.01
    for it in range(max_iter):
        if check_time():
            break
        ones_idx = np.where(h_b == 1.0)[0]
        zeros_idx = np.where(h_b == 0.0)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i = ones_idx[rng.randint(len(ones_idx))]
        j = zeros_idx[rng.randint(len(zeros_idx))]
        h_b[i], h_b[j] = h_b[j], h_b[i]
        c5_new = compute_c5(h_b, dx)
        delta = c5_new - c5_c
        if delta < 0 or rng.rand() < np.exp(-delta / T_sa):
            c5_c = c5_new
            if c5_new < best_c5:
                best_h_b = h_b.copy()
                best_c5 = c5_new
        else:
            h_b[i], h_b[j] = h_b[j], h_b[i]
        T_sa = max(0.001, T_sa * 0.999)
    return best_h_b, best_c5
