from scipy.optimize import minimize

_opt_state = {}

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - target
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(free):
            h[free] -= diff / np.count_nonzero(free)
        else:
            h -= diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = np.sum(h) - target
    if abs(diff) > 1e-14:
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(free):

def soft_max_obj_grad(h, dx, beta):
    n = len(h)
    raw_corr = np.correlate(h, 1.0 - h, mode='full')
    c = raw_corr * dx
    max_c = np.max(c)
    exp_c = np.exp(c - max_c)
    sum_exp = np.sum(exp_c)
    w = exp_c / sum_exp
    obj = max_c + np.log(sum_exp) / beta
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-16:
            continue
        k = idx - (n - 1)
        if k == 0:

        elif k > 0:

            grad[:n - k] -= wi * h[k:]
        else:
            kk = -k
            grad[:n - kk] += wi * (1.0 - h[kk:])
            grad[kk:] -= wi * h[:n - kk]
    grad *= dx
    return obj, grad
