```python
import time

    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    
    def objective(x):
        return compute_c5(x, dx)
    
    def objective_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        k_star = np.argmax(corr)
        d = k_star - (n - 1)
        grad = np.ones(n)
        if d == 0:
            grad = 1.0 - 2.0 * x
        elif d > 0:
            grad[d:] -= x[:-d]
            grad[:-d] -= x[d:]
        else:
            s = -d
            grad[:n-s] -= x[s:]
            grad[s:] -= x[:n-s]
        return grad
    
    constraint = {'type': 'eq', 'fun': lambda x: np.sum(x) - target, 
                  'jac': lambda x: np.ones(n)}
    
    res = minimize(objective, h, jac=objective_grad, method='SLSQP',
                   bounds=bounds, constraints=constraint,
                   options={'maxiter': max_iter, 'ftol': 1e-14})
    return project_feasible(res.x, n)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def all_overlaps(h, dx):
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    for _ in range(1000):
        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - target
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nf = np.count_nonzero(free)
        if nf > 0:
            h[free] -= diff / nf
        else:
            h -= diff / n
    h = np.clip(h, 0.0, 1.0)
    return h

def smoothed_grad(h, n, sigma):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_v = np.max(corr)
    if sigma < 1e-10:
        # Use only the argmax
        weights = np.zeros(len(corr))
        weights[np.argmax(corr)] = 1.0
    else:
        weights = np.exp((corr - max_v) / sigma)
        sw = np.sum(weights)
        if sw < 1e-30:
            weights = np.zeros(len(corr))
            weights[np.argmax(corr)] = 1.0
        else:
            weights /= sw
    
    grad = np.zeros(n)
    for k in range(len(corr)):
        if weights[k] < 1e-8:
            continue
        d = k - (n - 1)
        g = np.ones(n)
        if d == 0:
            g = 1.0 - 2.0 * h
        elif d > 0:
            g[d:] -= h[:-d]
            g[:-d] -= h[d:]
        else:
            s = -d
            g[:n-s] -= h[s:]
            g[s:] -= h[:n-s]
        grad += weights[k] * g
    return grad, max_v

def subgrad_descent(h, n, max_steps, sigma, lr0, decay, rng):
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    v = np.zeros(n)
    beta = 0.9
    
    for step in range(max_steps):
        lr = lr0 / (1.0 + step * decay)
        grad, _ = smoothed_grad(h, n, sigma)
        v = beta * v + (1 - beta) * grad
        h_new = project_feasible(h - lr * v, n)
        c5 = compute_c5(h_new, dx)
        if c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = c5
        h = h_new
    return best_h, best_c5

def steepest_descent_01(h_bin, n, max_rounds, rng):
    dx = 2.0 / n
    best = h_bin.copy()
    best_c5 = compute_c5(best, dx)
    
    for rd in range(max_rounds):
        ones = np.where(best == 1)[0]
        zeros = np.where(best == 0)[0]
        best_gain = 0
        best_swap = None
        
        # Sample a subset of swaps for efficiency
        n_eval = min(500, len(ones) * len(zeros))
        for _ in range(n_eval):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            best[i] = 0
            best[j] = 1
            c5 = compute_c5(best, dx)
            best[i] = 1
            best[j] = 0
            if c5 < best_c5 - best_gain:
                best_gain = best_c5 - c5
                best_swap = (i, j)
        
        if best_swap is None:
            break
        i, j = best_swap
        best[i] = 0
        best[j] = 1
        best_c5 -= best_gain
    
    return best, best_c5

def slsqp_optimize(h, n, max_iter=200):
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return compute_c5(x, dx)
    
    def obj_grad(x):
        grad, _ = smoothed_grad(x, n, 1e-8)
        return grad
    
    constraint = {'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                  'jac': lambda x: np.ones(n)}
    
    res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                   bounds=bounds, constraints=constraint,
                   options={'maxiter': max_iter, 'ftol': 1e-15})
    return project_feasible(res.x, n)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - target
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nf = np.count_nonzero(free)
        if nf > 0:
            h[free] -= diff / nf
        else:
            h -= diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = np.sum(h) - target
    if abs(diff) > 1e-13:
        idxs = np.argsort(np.abs(h - 0.5))
        for idx in idxs:
            if abs(diff) < 1e-14:
                break
            adj = np.sign(-diff) * min(abs(diff), min(h[idx], 1.0 - h[idx]))
            if abs(adj) < 1e-15:
                continue
            h[idx] += adj
            diff -= adj
    return h

def compute_gradient(h, n, sigma):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    mv = np.max(corr)
    if sigma < 1e-10:
        w = np.zeros(len(corr))
        w[np.argmax(corr)] = 1.0
    else:
        w = np.exp((corr - mv) / sigma)
        sw = np.sum(w)
        if sw < 1e-30:
            w = np.zeros(len(corr))
            w[np.argmax(corr)] = 1.0
        else:
            w /= sw
    grad = np.zeros(n)
    for k in range(len(corr)):
        if w[k] < 1e-8:
            continue
        d = k - (n - 1)
        g = np.ones(n)
        if d == 0:
            g[:] = 1.0 - 2.0 * h
        elif d > 0:
            g[d:] -= h[:-d]
            g[:-d] -= h[d:]
        else:
            s = -d
            g[:n - s] -= h[s:]
            g[s:] -= h[:n - s]
        grad += w[k] * g
    return grad

def subgrad_desc(h, n, steps, sigma, lr0, decay, beta):

    bh = h.copy()
    bc5 = compute_c5(h, dx)
    v = np.zeros(n)
    for t in range(steps):
        lr = lr0 / (1.0 + t * decay)
        g = compute_gradient(h, n, sigma)
        v = beta * v + (1.0 - beta) * g
        hn = project_feasible(h - lr * v, n)
        c5 = compute_c5(hn, dx)
        if c5 < bc5 - 1e-15:
            bh = hn.copy()
            bc5 = c5
        h = hn
    return bh, bc5

def sd_01(hb, n, rounds, nevals, rng):
    dx = 2.0 / n
    best = hb.copy()
    bc5 = compute_c5(best, dx)
    for _ in range(rounds):
        ones = np.where(best == 1)[0]
        zeros = np.where(best == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        bg = 0.0
        bs = None
        ne = min(nevals, len(ones) * len(zeros))
        for _ in range(ne):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            best[i] = 0.0
            best[j] = 1.0
            c5 = compute_c5(best, dx)
            best[i] = 1.0
            best[j] = 0.0
            ga = bc5 - c5
            if ga > bg:
                bg = ga
                bs = (i, j)
        if bs is None:
            break
        i, j = bs
        best[i] = 0.0
        best[j] = 1.0
        bc5 = compute_c5(best, dx)
    return best, bc5

def pen_obj(x):
    n = len(x)
    dx = 2.0 / n
    return compute_c5(x, dx) + 1e4 * (np.sum(x) - n / 2.0) ** 2

def pen_grad(x):
    n = len(x)
    dx = 2.0 / n
    g = compute_gradient(x, n, 1e-8)
    sd = np.sum(x) - n / 2.0
    return g + 2e4 * sd * np.ones(n)

def slsqp_polish(h, n, iters=400):
    bounds = [(0.0, 1.0)] * n
    res = minimize(pen_obj, h, jac=pen_grad, method='SLSQP',
                   bounds=bounds, options={'maxiter': iters, 'ftol': 1e-16})
    return project_feasible(res.x, n)
```
