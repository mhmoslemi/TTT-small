import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def smooth_max_val(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def smooth_max_weights(x, beta):
    m = np.max(x)
    e = np.exp(beta * (x - m))
    return e / e.sum()

def project_exact(x, target_sum):
    lo, hi = -5.0, 5.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def gradient_c5_smooth(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    w = smooth_max_weights(corr, beta)
    grad = np.zeros(n)
    for lag_idx in range(2 * n - 1):
        wk = w[lag_idx]
        if wk < 1e-16:
            continue
        k = lag_idx - (n - 1)
        if k == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif k > 0:
            idx1 = np.arange(k, n)
            grad[idx1] += wk * (1.0 - h[idx1 - k])
            idx2 = np.arange(0, n - k)
            grad[idx2] -= wk * h[idx2 + k]
        else:
            m = -k
