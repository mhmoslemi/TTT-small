From numpy docs: `correlate(a, v, 'full')[k] = sum_n a[n+k] * v[n]` for the full mode, where the valid range of k ensures all indices are in bounds.

More precisely, for `correlate(a, v, 'full')` with a of length na and v of length nv:
result[k] = sum_n a[n + k - (nv-1)] * v[n] for k = 0, ..., na+nv-2

        c5_bin2 = evaluate_c5(h_ls)
        if c5_bin2 < best_c5:
            best_h = h_ls.copy()
            best_c5 = c5_bin2

    # Phase 3: SLSQP refinement from best
    if rem() > 40:
        for tau in [500, 2000, 8000]:
            if rem() < 20:
                break
            h_new, c5_new = slsqp_optimize(best_h, n0, tau, 400)
            if c5_new is not None and c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    # Phase 4: Try different resolutions
    for tn in [80, 100, 140, 160]:
        if rem() < 50:
            break
        target = tn / 2.0
        h_t = np.interp(np.linspace(0, 1, tn), np.linspace(0, 1, n0), best_h)
        h_t = fix_sum(h_t, target)
        # Quick PGD
        h_t = pgd_smooth_max(h_t, tn, 200, 0.3, 2000)
        c5_t = evaluate_c5(h_t)
        if c5_t < best_c5:
            # Refine with SLSQP
            h_t2, c5_t2 = slsqp_optimize(h_t, tn, 1000, 300)
            if c5_t2 is not None and c5_t2 < best_c5:
                best_h = h_t2
                best_c5 = c5_t2
            else:
                best_h = h_t
                best_c5 = c5_t
        # Also try binary at this resolution
        if rem() > 30:
            h_bt = np.zeros(tn)
            h_bt[np.argsort(h_t)[::-1][:int(target)]] = 1.0
            h_bt, c5_bt = simulated_anneal_binary(h_bt, tn, 0.003, 1e-7, 50000, seed + tn)
            if c5_bt < best_c5:
                h_bt2, c5_bt2 = binary_local_search(h_bt, tn, max_passes=10)
                if c5_bt2 < best_c5:
                    best_h = h_bt2
                    best_c5 = c5_bt2
                else:
                    best_h = h_bt
                    best_c5 = c5_bt

    # Phase 5: Final SLSQP polish
    if rem() > 15:
        hn = len(best_h)
        h_new, c5_new = slsqp_optimize(best_h, hn, 10000, 500)
        if c5_new is not None and c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Final constraint fix
    hn_final = len(best_h)
    best_h = fix_sum(best_h, hn_final / 2.0)
    best_c5 = evaluate_c5(best_h)

    return best_h, best_c5, hn_final
