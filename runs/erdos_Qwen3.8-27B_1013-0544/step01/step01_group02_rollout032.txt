import numpy as np
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_c5_vals(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return c * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-15:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 700:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def sum_con(h, n):
    return np.sum(h) - n / 2.0

def sum_con_jac(h, n):
    return np.ones(n)

def binary_sa(h, n, rng, max_iters=50000, t_start=5.0, t_end=0.01):
    """Simulated annealing on binary sequences."""
    dx = 2.0 / n
    current = h.copy()
    best = h.copy()
    best_energy = compute_c5(current, n)
    ones = np.where(current == 1)[0]
    zeros = np.where(current == 0)[0]
    
    for it in range(max_iters):
        T = t_start * (t_end / t_start) ** (it / max_iters)
        # Pick a random 1 and a random 0, swap them
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        # Compute change in energy (max of correlations)
        # Actually, computing the full C5 each time is expensive for large n
        # But for n=80, it's fast enough
        current[i], current[j] = current[j], current[i]
        new_energy = np.max(np.correlate(current, 1 - current, mode='full')) * dx
        
        if new_energy < current_e or rng.random() < np.exp(-(new_energy - current_e) / T):
            # Accept
            ones = np.where(current == 1)[0]
            zeros = np.where(current == 0)[0]
            current_e = new_energy
            if new_energy < best_energy:
                best = current.copy()
                best_energy = new_energy
        else:
            # Reject
            current[i], current[j] = current[j], current[i]
    
    return best, best_energy

    a = np.zeros(2 * n - 1)
    # a[k] for k = 0, ..., 2n-2, corresponding to shift s = k - (n-1)
    s_vals = np.arange(2 * n - 1) - (n - 1)  # shifts from -(n-1) to (n-1)

    idx2 = p + s_vals
    mask2 = (idx2 >= 0) & (idx2 <= n - 1)
    a[mask2] -= h[idx2[mask2]]

        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = None

    best_h = None
    best_c5 = float('inf')

    if h_init is not None:
        h_init = project_feasible(h_init, n_init)
        best_h = h_init.copy()
        best_c5 = compute_c5(best_h, n_init)

    n_list = [80, 60, 100, 70, 90]
    if n_init is not None and n_init not in n_list:
        n_list.insert(0, n_init)
    n_list = list(dict.fromkeys(n_list))

    time_per_n = max(80, int((budget_s - 150) / len(n_list)))

    for n in n_list:
        if time.time() > deadline:
            break
        phase_deadline = time.time() + time_per_n

        # Binary SA initialization
        h_bin = np.zeros(n)
        idx = rng.permutation(n)
        h_bin[idx[:n // 2]] = 1.0
        h_sa, c_sa = binary_sa(h_bin, n, rng, 40000, 3.0, 0.005)

        # Structured init
        h_struct = np.zeros(n)
        positions = np.linspace(0, n - 1, n // 2).astype(int)
        h_struct[positions] = 1.0
        h_sa2, c_sa2 = binary_sa(h_struct, n, rng, 25000, 2.0, 0.003)

        h_bin_best = h_sa if c_sa <= c_sa2 else h_sa2
        c_bin_best = c_sa if c_sa <= c_sa2 else c_sa2

        h_cont = project_feasible(h_bin_best, n)
        c_cont = compute_c5(h_cont, n)
        if c_cont < best_c5:
            best_c5 = c_cont
            best_h = h_cont.copy()

        # SLSQP from binary solution
        h_ref, c_ref = slsqp_refine(h_cont, n, [50, 200, 1000, 5000], t0, min(phase_deadline, deadline))
        if c_ref < best_c5:
            best_c5 = c_ref
            best_h = h_ref.copy()

        # Coordinate descent
        if time.time() < deadline - 20:
            h_cd, c_cd = coordinate_descent(best_h, n, t0, min(phase_deadline, deadline - 5), rng, n_sweeps=80)
            if c_cd < best_c5:
                best_c5 = c_cd
                best_h = h_cd.copy()

        # If n matches initial, also refine initial
        if n == n_init and h_init is not None and time.time() < phase_deadline:
            h_ref2, c_ref2 = slsqp_refine(h_init, n, [200, 1000, 5000, 20000], t0, deadline)
            if c_ref2 < best_c5:
                best_c5 = c_ref2
                best_h = h_ref2.copy()
            if time.time() < deadline - 10:
                h_cd2, c_cd2 = coordinate_descent(best_h, n, t0, deadline - 5, rng, n_sweeps=60)
                if c_cd2 < best_c5:
                    best_c5 = c_cd2
                    best_h = h_cd2.copy()

    # Final phases on best solution
    if best_h is not None and time.time() < deadline - 10:
        n_best = len(best_h)
        bounds = [(0.0, 1.0)] * n_best
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n_best,)}]

        # High-alpha SLSQP
        h_fin, c_fin = slsqp_refine(best_h, n_best, [10000, 50000], t0, deadline)
        if c_fin < best_c5:
            best_c5 = c_fin
            best_h = h_fin.copy()

        # More coordinate descent
        if time.time() < deadline - 10:
            h_cd3, c_cd3 = coordinate_descent(best_h, n_best, t0, deadline - 5, rng, n_sweeps=150)
            if c_cd3 < best_c5:
                best_c5 = c_cd3
                best_h = h_cd3.copy()

        # Random restart + SLSQP
        if time.time() < deadline - 30:
            for _ in range(5):
                if time.time() > deadline - 10:
                    break
                scale = rng.choice([0.01, 0.02, 0.05, 0.1])
                h_pert = best_h + rng.normal(0, scale, size=n_best)
                h_pert = project_feasible(h_pert, n_best)
                h_try, c_try = slsqp_refine(h_pert, n_best, [2000, 10000], t0, deadline)
                if c_try < best_c5:
                    best_c5 = c_try
                    best_h = h_try.copy()

        # Final polish
        if time.time() < deadline - 5:
            h_pol, c_pol = slsqp_refine(best_h, n_best, [20000, 100000], t0, deadline)
            if c_pol < best_c5:
                best_c5 = c_pol
                best_h = h_pol.copy()
            if time.time() < deadline - 2:
                h_cd4, c_cd4 = coordinate_descent(best_h, n_best, t0, deadline, rng, n_sweeps=50)
                if c_cd4 < best_c5:
                    best_c5 = c_cd4
                    best_h = h_cd4.copy()

    # Ensure feasibility
    if best_h is not None:
        n_final = len(best_h)
        best_h = project_feasible(best_h, n_final)
        best_c5 = compute_c5(best_h, n_final)
        return best_h, best_c5, n_final

    n_fb = 80
    h_fb = np.full(n_fb, 0.5)
    h_fb = project_feasible(h_fb, n_fb)
    return h_fb, compute_c5(h_fb, n_fb), n_fb
