from scipy.optimize import minimize, Bounds
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_all_lags(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx  # all lag values

def subgradient_at_max(h):
    """Compute subgradient of the max objective at current h."""
    n = len(h)
    dx = 2.0 / n
    lags = np.correlate(h, 1.0 - h, mode="full") * dx
    max_val = np.max(lags)
    # Find all lags within tolerance of max
    active = np.where(lags >= max_val - 1e-10)[0]
    
    # For each active lag k (0-indexed in the full correlation),
    # the lag in terms of shift is k - (n-1)
    # The subgradient contribution for shift s = k - (n-1):
    # ∂/∂h[j] of sum_i h[i]*(1-h[i+s]) * dx
    # = dx * ((1-h[j+s]) if j+s<n else 0) + dx * (-h[j-s]) if j-s>=0 else 0
    
    grad = np.zeros(n)
    for k in active:
        s = k - (n - 1)  # shift
        if s == 0:
            # q(0) = sum h[i]*(1-h[i]) * dx
            grad += dx * (1 - 2*h)
        elif s > 0:
            # q(s) = sum_{i=0}^{n-s-1} h[i]*(1-h[i+s]) * dx
            # ∂/∂h[j]: from first factor: (1-h[j+s]) for j < n-s
            #            from second factor: -h[j-s] for j >= s
            forward = np.zeros(n)
            forward[:n-s] = (1 - h[s:])
            backward = np.zeros(n)
            backward[s:] = -h[:n-s]
            grad += dx * (forward + backward)
        else:
            # s < 0, let s' = -s > 0
            # q(-s') = sum_{i=s'}^{n-1} h[i]*(1-h[i-s']) * dx = sum_{j=0}^{n-s'-1} h[j+s']*(1-h[j]) * dx
            # ∂/∂h[j]: from first factor: (1-h[j]) for j < n-s' (h[j+s'] is the first factor)
            #            from second factor: -h[j+s'] for j >= s'
            s_pos = -s
            forward = np.zeros(n)
            forward[:n-s_pos] = (1 - h[:n-s_pos])  # 1-h[j] for j < n-s'
            backward = np.zeros(n)
            backward[s_pos:] = -h[:n-s_pos]  # -h[j-s'] for j >= s'
            grad += dx * (forward + backward)
    
    # Average over active lags for a better subgradient
    if len(active) > 1:
        grad /= len(active)
    return grad, max_val

    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    # Distribute the difference to elements not at boundary
    if diff > 0:
        mask = h < 1.0 - 1e-10
    else:
        mask = h > 1e-10
    cnt = mask.sum()
    if cnt == 0:
        return h
    h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

    corr = np.correlate(h, 1.0 - h, mode="full")

def subgradient_descent(h, n_iter=2000, step0=0.1, decay=0.999):

    h = h.copy()
    step = step0

    for it in range(n_iter):

        if val < best_val:
            best_val = val
            best_copy = h.copy()

        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = fix_sum(h_new, target)
        val_new = evaluate_c5(h_new)
        if val_new < best_val:
            best_val = val_new
            best_copy = h_new.copy()
            h = h_new
        else:
            step *= decay
        if step < 1e-6:
            break
    return best_copy, best_val

    """Round h to nearest binary with correct sum."""
    n = len(h)
    target = int(n / 2)
    # Sort by value, set top 'target' to 1, rest to 0
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

    target = n_pts / 2.0
    bnds = Bounds(np.zeros(n_pts), np.ones(n_pts))

        x = tau * obj
        x_max = np.max(x)
        return float(x_max / tau + np.log(np.sum(np.exp(x - x_max))) / tau)

        corr = np.correlate(x, 1.0 - x, mode="full")
        return smooth_max(corr, tau)

        corr = np.correlate(x, 1.0 - x, mode="full")
        # Gradient of smooth max
        softmax = np.exp(tau * corr - tau * np.max(corr))
        softmax /= softmax.sum()
        # Gradient of corr w.r.t. x
        # This is complex, let's use numerical or just skip gradient
        return None

             'jac': lambda x: np.ones(n_pts)}]

        return x, c5

        return h, evaluate_c5(h)

        h[::2] = 1.0
    elif mode == 1:
        # Block: first half 1, second half 0
        h[:n//2] = 1.0

            h[idx[:cnt-target]] = 0
        elif cnt < target:
            idx = np.where(h == 0)[0]
            h[idx[:target-cnt]] = 1
    return h

def subgrad_step(h, step):
    """One subgradient step for the max objective."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    lags = corr * dx
    max_val = np.max(lags)
    active = np.where(lags >= max_val - 1e-9)[0]
    
    grad = np.zeros(n)
    for k in active:
        s = k - (n - 1)
        if s == 0:
            grad += dx * (1.0 - 2.0 * h)
        elif s > 0:
            f = np.zeros(n)
            f[:n-s] += (1.0 - h[s:]) * dx
            f[s:] += (-h[:n-s]) * dx
            grad += f
        else:
            p = -s
            f = np.zeros(n)
            f[p:] += (1.0 - h[:n-p]) * dx
            f[:n-p] += (-h[p:]) * dx
            grad += f
    
    if len(active) > 1:
        grad /= len(active)
    
    h_new = h - step * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = fix_sum(h_new, n / 2.0)
    return h_new
