import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def compute_all_corr(h):
    return np.correlate(h, 1 - h, mode='full')

def project_onto_simplex_box(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def grad_at_shift(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        g[:n-k] += (1.0 - h[k:])
        if k > 0:
            g[k:] -= h[:n-k]
    else:
        p = -k
        g[p:] += (1.0 - h[:n-p])
        if p > 0:
            g[:n-p] -= h[p:]
    return g

def optimize_subgrad(h0, target_sum, deadline, rng, max_iters=5000):
    """Subgradient optimization with focus on top correlations"""
    n = len(h0)
    best_h = project_onto_simplex_box(h0.astype(float).copy(), target_sum)
    best_c5 = compute_c5(best_h)
    
    momentum = np.zeros(n)
    beta = 0.9
    lr = 0.02
    top_k = 5  # focus on top k correlations
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        corr = compute_all_corr(best_h)
        # Get top k indices
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
        top_corr = corr[top_idx]
        
        # Weight by softmax of top correlations
        m = np.max(top_corr)
        w = np.exp(3.0 * (top_corr - m))
        w /= w.sum()
        
        # Compute weighted gradient
        grad = np.zeros(n)
        for idx, weight in zip(top_idx, w):
            k = idx - (n - 1)
            grad += weight * grad_at_shift(best_h, k)

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_onto_simplex_box(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def grad_at_shift(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        g[:n-k] += (1.0 - h[k:])
        if k > 0:
            g[k:] -= h[:n-k]
    else:
        p = -k
        g[p:] += (1.0 - h[:n-p])
        if p > 0:
            g[:n-p] -= h[p:]
    return g

```python
def compute_all_deltas(h, k_star):
    """Compute delta R(k_star) for all possible pair swaps (i:1->0, j:0->1)"""
    n = len(h)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    # For each i (currently 1), compute A[i] = h[i-k] (if valid) - h[i+k] (if valid)
    A = np.zeros(len(ones))
    for idx, i in enumerate(ones):
        ik = i - k_star
        ikp = i + k_star
        if 0 <= ik < n:
            A[idx] += h[ik]
        if 0 <= ikp < n:
            A[idx] -= h[ikp]
    
    # For each j (currently 0), compute B[j] = (1-h[j+k]) (if valid) - h[j-k] (if valid)
    B = np.zeros(len(zeros))
    for idx, j in enumerate(zeros):
        jk = j + k_star
        jkm = j - k_star
        if 0 <= jk < n:
            B[idx] += (1.0 - h[jk])
        if 0 <= jkm < n:
            B[idx] -= h[jkm]
    
    # Delta for pair (i_a, j_b) = A[a] + B[b]
    # We want to minimize this, so find the pair with minimum A[a] + B[b]
    # That's just min(A) + min(B)
    
    return A, B
```

                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                    if len(ones) > 0 and len(zeros) > 0:
                        h[int(rng.choice(ones))] = 0.0
                        h[int(rng.choice(zeros))] = 1.0
            elif strategy == 1:
                start = rng.randint(0, n - 12)
                length = rng.randint(12, min(50, n - start))
                end = start + length
                seg = h[start:end].copy()
                oi = np.where(seg == 1)[0]
                zi = np.where(seg == 0)[0]
                ns = min(len(oi), len(zi))
                for s in range(ns):
                    seg[oi[s]] = 0.0
                    seg[zi[s]] = 1.0
                h[start:end] = seg
            elif strategy == 2:
                h_cont = h.astype(float)
                freq = rng.randint(2, 12)
                phase = rng.uniform(0, 2 * np.pi)
                h_cont += 0.4 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
                h = binarize_keep_sum(h_cont, n)
            else:
                h = (rng.rand(n) < 0.5).astype(float)
                if np.sum(h) != n // 2:
                    h = binarize_keep_sum(h + 0.01 * rng.randn(n), n)
            no_improve = 0
            best_c5 = min(best_c5, compute_c5(h))
            if compute_c5(h) < best_c5:
                best_c5 = compute_c5(h)
                best_h = h.copy()

    return best_h, best_c5

def gradient_refine(h, target_sum, deadline, rng, max_iters=3000):
    n = len(h)
    best_h = project_onto_simplex_box(h.astype(float).copy(), target_sum)
    best_c5 = compute_c5(best_h)
    lr = 0.015
    momentum = np.zeros(n)
    beta = 0.9
    stall = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        top_idx = np.argpartition(corr, -5)[-5:]
        top_vals = corr[top_idx]
        m = top_vals.max()
        w = np.exp(4.0 * (top_vals - m))
        w /= w.sum()
        grad = np.zeros(n)
        for idx, weight in zip(top_idx, w):
            k = int(idx) - (n - 1)
            grad += weight * grad_at_shift(best_h, k)
        grad *= (2.0 / n)
        momentum = beta * momentum - lr * grad
        h_new = project_onto_simplex_box(best_h + momentum, target_sum)
        new_c5 = compute_c5(h_new)
        if new_c5 < best_c5:
            best_h = h_new
            best_c5 = new_c5
            lr = min(lr * 1.004, 0.06)
            stall = 0
        else:
            stall += 1
            lr *= 0.98
            if stall > 150:
                h_b = binarize_keep_sum(best_h + 0.08 * rng.randn(n), n)
                h_b = project_onto_simplex_box(h_b.astype(float), target_sum)
                if compute_c5(h_b) < best_c5:
                    best_h = h_b
                    best_c5 = compute_c5(best_h)
                lr = 0.015
                momentum = np.zeros(n)
                stall = 0
    return best_h, best_c5

def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0 / n_new
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = 0.5 + 0.3 * rng.randn(100)
        h_init = project_onto_simplex_box(h_init, 50.0)

    best_h = None
    best_c5 = float('inf')

    def try_solution(h_cand, n_cand, phase_frac_start, phase_frac_end):
        nonlocal best_h, best_c5
        dl = t0 + budget_s * phase_frac_end
        if time.time() > dl:
            return
        h_b = binarize_keep_sum(h_cand, n_cand)
        dl_search = min(t0 + budget_s * (phase_frac_start + 0.5 * (phase_frac_end - phase_frac_start)), dl)
        h_opt, c5_opt = binary_local_search(h_b, dl_search, rng)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        if time.time() < dl - 5:
            dl_grad = dl - 2
            h_g, c5_g = gradient_refine(h_opt, n_cand / 2.0, dl_grad, rng, max_iters=2000)
            if c5_g < best_c5:
                best_c5 = c5_g
                best_h = h_g.copy()

    n_init = len(h_init)

    # Phase 1: Optimize at initial resolution (binary + gradient)
    try_solution(h_init, n_init, 0.0, 0.3)

    # Phase 2: Try n=100 from random start
    if time.time() < t0 + budget_s * 0.4:
        h_rand = (rng.rand(100) < 0.5).astype(float)
        h_rand = binarize_keep_sum(h_rand + 0.01 * rng.randn(100), 100)
        try_solution(h_rand, 100, 0.3, 0.5)

    # Phase 3: Try other resolutions
    for n_try in [80, 120, 150]:
        remaining_frac = 1.0 - (time.time() - t0) / budget_s
        if remaining_frac < 0.1:
            break
        start_frac = 1.0 - remaining_frac
        end_frac = min(1.0 - 0.02, start_frac + 0.15)
        h_interp = interpolate_h(best_h, n_try)
        h_pert = h_interp + 0.2 * rng.randn(n_try)
        try_solution(h_pert, n_try, start_frac, end_frac)

    # Phase 4: Final polish on best
    if time.time() < t0 + budget_s - 10:
        n_best = len(best_h)
        h_b = binarize_keep_sum(best_h, n_best)
        h_final, c5_final = binary_local_search(h_b, t0 + budget_s - 3, rng, max_rounds=100000)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()
        if time.time() < t0 + budget_s - 2:
            h_g, c5_g = gradient_refine(best_h, len(best_h) / 2.0, t0 + budget_s - 1, rng, max_iters=1000)
            if c5_g < best_c5:
                best_c5 = c5_g
                best_h = h_g.copy()

    best_h = np.clip(best_h, 0, 1)
    best_h = project_onto_simplex_box(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
