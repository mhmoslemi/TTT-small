def project_feasible(h, n):
    target = n / 2.0
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - target
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            h[free] -= diff / np.sum(free)
        else:
            h -= diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = np.sum(h) - target
    if abs(diff) > 1e-14:

        if np.any(free):

            h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_optimize(h_start, n, dx, target_sum, max_iter=300):
    bounds = [(0.0, 1.0)] * n
    constraint = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return np.max(corr)
    
    res = minimize(
        obj,
        h_start,
        method='SLSQP',
        bounds=bounds,
        constraints=constraint,
        options={'maxiter': max_iter, 'ftol': 1e-12}
    )
    return project_feasible(res.x, n)

def bottleneck_gradient_step(h, n, dx, step_size=0.05):
    """Take a step in the negative subgradient direction of the max."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    
    # Find active lags (within tolerance)
    tol = 1e-6
    active = np.where(corr >= max_val - tol)[0]
    
    # Compute subgradient: average gradient over active lags
    # ∂c[k]/∂h[j] = 1 - h[j+shift] - h[j-shift] (shift = k - (n-1))
    #              = 1 - 2*h[j] (shift = 0)
    grad = np.zeros(n)
    for k in active:
        shift = k - (n - 1)
        if shift == 0:
            g = 1.0 - 2.0 * h
        else:
            g = np.ones(n)
            # h[j+shift] with out-of-range = 0
            if shift > 0:
                g[:n-shift] -= h[shift:]  # h[j+shift] for j=0..n-shift-1
                g[shift:] -= h[:n-shift]  # h[j-shift] for j=shift..n-1
            else:
                s = -shift
                g[s:] -= h[:n-s]  # h[j+shift] = h[j-s] for j=s..n-1
                g[:n-s] -= h[s:]   # h[j-shift] = h[j+s] for j=0..n-s-1
        grad += g
    grad /= len(active)
    
    # Step in negative gradient direction
    h_new = h - step_size * grad
    h_new = project_feasible(h_new, n)
    return h_new

        c5 = compute_c5(h_opt, dx)
        if c5 < best_c5:
            best_h, best_c5 = h_opt, c5
        # Bottleneck escape with varied tolerance
        if elapsed() < phase2_end:
            for tol_frac in [0.005, 0.02, 0.05]:
                if elapsed() > phase2_end:
                    break
                h_esc = bottleneck_step(h_opt, n, 0.08, tol_frac=tol_frac)
                h_esc2 = slsqp_optimize(h_esc, n, dx, target_sum, max_iter=400)
                c5_e = compute_c5(h_esc2, dx)
                if c5_e < best_c5:
                    best_h, best_c5 = h_esc2, c5_e
                    h_opt = h_esc2
                    c5 = c5_e
        # Soft max refinement
        if elapsed() < phase2_end:
            h_sm = slsqp_optimize(h_opt, n, dx, target_sum, max_iter=300,
                                  use_softmax=True, tau=150.0)
            c5_sm = compute_c5(h_sm, dx)
            if c5_sm < best_c5:
                best_h, best_c5 = h_sm, c5_sm
                h_opt = h_sm
                c5 = c5_sm
    
    # Phase 3: Try upsampling to 2n
    if elapsed() < budget_s * 0.92:
        n2 = 2 * n
        dx2 = 2.0 / n2
        target2 = n2 / 2.0
        h_up = np.interp(np.linspace(0, 2, n2), np.linspace(0, 2, n), best_h)
        h_up = project_feasible(h_up, n2)
        c5_up = compute_c5(h_up, dx2)
        if c5_up < best_c5:
            best_h, best_c5 = h_up, c5_up
            n_best, dx_best = n2, dx2
        else:
            n_best, dx_best = n, dx
        
        phase3_end = budget_s * 0.92
        while elapsed() < phase3_end:
            h_opt2 = slsqp_optimize(h_up.copy(), n2, dx2, target2, max_iter=300)
            c5_2 = compute_c5(h_opt2, dx2)
            if c5_2 < best_c5:
                best_h, best_c5 = h_opt2, c5_2
                n_best, dx_best = n2, dx2
            h_up = bottleneck_step(h_opt2, n2, 0.04)
            h_opt2 = slsqp_optimize(h_up, n2, dx2, target2, max_iter=250)
            c5_2b = compute_c5(h_opt2, dx2)
            if c5_2b < best_c5:
                best_h, best_c5 = h_opt2, c5_2b
                n_best, dx_best = n2, dx2
    
    # Final polish
    n_fin = len(best_h)
    dx_fin = 2.0 / n_fin
    while elapsed() < budget_s - 10:
        h_fin = slsqp_optimize(best_h, n_fin, dx_fin, n_fin / 2.0, max_iter=600)
        c5_fin = compute_c5(h_fin, dx_fin)
        if c5_fin >= best_c5 - 1e-12:
            break
        best_h, best_c5 = h_fin, c5_fin
    
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h, 2.0 / len(best_h))
    return (best_h, best_c5, len(best_h))
