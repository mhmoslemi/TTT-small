    for d_test in [hi, (lo+hi)/2]:
        v = np.max(c + d_test * b)
        if v < best_v:
            best_v = v
            best_d = d_test

            best_d = m

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx), c

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-15:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def upsample(h, n_new):
    n_old = len(h)
    x_old = np.arange(n_old) / n_old
    x_new = np.arange(n_new) / n_new
    return project_feasible(np.interp(x_new, x_old, h), n_new)

def compute_grads(h, n, eps=1e-8):
    n_corr = 2 * n - 1
    c = np.correlate(h, 1 - h, mode='full')
    grads = np.zeros((n, n_corr))
    for i in range(n):
        h_p = h.copy()
        h_m = h.copy()
        h_p[i] = min(1.0, h[i] + eps)
        h_m[i] = max(0.0, h[i] - eps)
        c_p = np.correlate(h_p, 1 - h_p, mode='full')
        c_m = np.correlate(h_m, 1 - h_m, mode='full')
        denom = h_p[i] - h_m[i]
        if denom > 0:
            grads[i] = (c_p - c_m) / denom
    return grads, c

def find_opt_delta(c, b, lo, hi):
    if lo >= hi:
        return lo
    a, bnd = lo, hi
    for _ in range(40):
        if bnd - a < 1e-14:
            break
        mid = (a + bnd) / 2
        vals = c + mid * b
        mx = vals.max()
        tol = 1e-9 * max(1.0, abs(mx))
        active = vals >= mx - tol
        mb = b[active].min()
        xb = b[active].max()
        if mb > 1e-8:
            bnd = mid
        elif xb < -1e-8:
            a = mid
        else:
            break
    best_d = (a + bnd) / 2
    best_v = np.max(c + best_d * b)
    for d in [lo, hi]:
        v = np.max(c + d * b)
        if v < best_v:
            best_v = v
            best_d = d
    return best_d

def coord_descent(h, n, max_sweeps=200, rng=None, deadline=None):
    h = h.copy()
    c = np.correlate(h, 1 - h, mode='full')
    best_c5 = c.max()
    for sweep in range(max_sweeps):
        if deadline and time.time() > deadline:
            break
        grads, c = compute_grads(h, n)
        improved = False
        idx = np.arange(n)
        rng.shuffle(idx)
        for ii in range(n):
            i = idx[ii]
            for j in range(n):
                if i == j:
                    continue
                b = grads[i] - grads[j]
                lo = max(-h[i], h[j] - 1.0)
                hi = min(1.0 - h[i], h[j])
                if lo >= hi - 1e-12:
                    continue
                d = find_opt_delta(c, b, lo, hi)
                if abs(d) < 1e-13:
                    continue
                hi_old, hj_old = h[i], h[j]
                h[i] += d
                h[j] -= d
                new_c = c + d * b
                if new_c.max() < best_c5 - 1e-14:
                    c = new_c
                    best_c5 = new_c.max()
                    improved = True
                else:
                    h[i], h[j] = hi_old, hj_old
        c = np.correlate(h, 1 - h, mode='full')
        best_c5 = c.max()
        if not improved:
            break
    return h, best_c5

def smooth_obj_fn(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    return m + np.log(np.sum(np.exp(shifted))) / alpha

def sum_constraint(x, n):
    return x.sum() - n / 2.0

def sum_jacobian(x, n):
    return np.ones(len(x))

def slsqp_polish(h, n, alpha=5000, maxiter=3000, deadline=None):
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_jacobian, 'args': (n,)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_obj_fn, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = project_feasible(res.x, n)
        return h_new
    except Exception:
        return h
