import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_correlations(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return c

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-15:
                break
            h *= (1.0 + diff / total)
    return np.clip(h, 0.0, 1.0)

def make_binary(h, n):
    """Convert to binary with exactly n/2 ones"""
    n_ones = n // 2
    # Threshold at 0.5, then adjust
    h_bin = (h >= 0.5).astype(float)
    current_ones = int(h_bin.sum())
    if current_ones > n_ones:
        # Need to flip some 1s to 0s
        # Flip the ones with smallest h value
        ones_idx = np.where(h_bin == 1)[0]
        sorted_by_h = ones_idx[np.argsort(h[ones_idx])]
        for idx in sorted_by_h[:current_ones - n_ones]:
            h_bin[idx] = 0
    elif current_ones < n_ones:
        # Need to flip some 0s to 1s
        zeros_idx = np.where(h_bin == 0)[0]
        sorted_by_h = zeros_idx[np.argsort(-h[zeros_idx])]
        for idx in sorted_by_h[:n_ones - current_ones]:
            h_bin[idx] = 1
    return h_bin

def binary_sa_search(h_start, n, budget_frac, rng, t0, total_budget):
    """Simulated annealing on binary space"""
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    temp = 0.01
    cool_rate = 0.999
    
    end_time = t0 + total_budget * budget_frac
    while time.time() < end_time:
        # Choose a random swap: pick a 1-position and a 0-position
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = h[j], h[i]
        c_new = compute_c5(h, n)
        delta = c_new - best_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-10)):
            if c_new < best_c5:
                best_h = h.copy()
                best_c5 = c_new
        else:
            h[i], h[j] = h[j], h[i]  # revert
        temp *= cool_rate
        if temp < 1e-6:
            temp = 0.005  # re-heat
    return best_h, best_c5

def obj_smooth(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 700:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def sum_con(h, n):
    return np.sum(h) - n / 2.0

def sum_con_jac(h, n):
    return np.ones(n)

    # Phase 1: Initial SLSQP polish

    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]

def compute_c5(h, n):

    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:

        if diff > 0:
            room = 1.0 - h

            if total < 1e-15:
                break
            h += (diff / total) * room
        else:

            if total < 1e-15:

        temp *= 0.9997
        if temp < 0.001:
            temp = 0.001
    return best_h, best_c5

            best_j = -1

                    h_try[i] += d
                    h_try[j] -= d
                    c = compute_c5(h_try, n)
                    if c < best_c - 1e-13:

                        best_delta = d
                        best_j = j

                best_h[i] += best_delta
                best_h[best_j] -= best_delta
                best_h = np.clip(best_h, 0, 1)
                best_c5 = best_c
                improved_any = True
    return best_h, best_c5

        if abs(diff) < 1e-14:

            if total < 1e-15:
