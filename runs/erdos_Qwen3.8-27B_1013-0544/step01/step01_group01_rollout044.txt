import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) > 1e-12:
        h = h + (target_sum - s) / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -50, 50)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def obj_smooth(h, beta):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, beta) * (2.0 / len(h))

def constraint_eq(h):
    return np.sum(h) - len(h) / 2.0

def constraint_jac(h):
    return np.ones(len(h))

def subgradient_active(h, lag):
    n = len(h)
    grad = np.zeros(n)
    for m in range(n):
        mk_minus = m - lag
        if 0 <= mk_minus < n:
            grad[m] += 1.0 - h[mk_minus]
        mk_plus = m + lag
        if 0 <= mk_plus < n:
            grad[m] -= h[mk_plus]
    return grad

def sg_refine(h, target_sum, max_iter=300, dx_budget=30):
    n = len(h)
    alpha = 0.04
    best_local = h.copy()
    best_val = compute_c5(best_local)
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        active_idx = np.argmax(corr)
        lag = active_idx - (n - 1)
        grad = subgradient_active(h, lag)
        h_new = h - alpha * grad
        h_new = project_h(h_new, target_sum)
        val = compute_c5(h_new)
        if val < best_val:
            best_local = h_new.copy()
            best_val = val
            h = h_new
        else:
            alpha *= 0.93
            if alpha < 0.0005:
                break
    return best_local, best_val

def slsqp_refine(h, target_sum, beta, maxiter=600):
    n = len(h)
    bounds = Bounds(np.zeros(n), np.ones(n))
    constraints = [{'type': 'eq', 'fun': constraint_eq, 'jac': constraint_jac}]
    try:
        def obj(h_in):
            return obj_smooth(h_in, beta)
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = project_h(res.x, target_sum)
        return h_new, compute_c5(h_new)
    except Exception:
        return h, compute_c5(h)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:

    best_h = h_parent.copy()
    best_n = len(h_parent)
    best_c5 = compute_c5(best_h)

    def time_left():
        return budget_s - (time.time() - start_time)

    n_candidates = [50, 60, 64, 70, 72, 80, 90, 100, 110, 120]

            break
