    best_h = project_feasible(h0.copy(), n)
    best_c5 = compute_c5(best_h, dx)

    # Phase 1: Diverse SLSQP direct at n
    t1 = t0 + budget_s * 0.35
    starts = gen_starts(n, rng, best_h)
    for sh in starts:
        if elapsed() > t1:
            break
        h_new = slsqp_direct(sh, n, dx, ts, max_iter=500)
        c5 = compute_c5(h_new, dx)
        if c5 < best_c5 - 1e-14:
            best_h, best_c5 = h_new, c5

    # Phase 2: Smooth SLSQP with annealing
    t2 = t0 + budget_s * 0.55
    for beta in [20, 40, 80, 160, 300]:
        if elapsed() > t2:
            break
        h_s = slsqp_smooth(best_h, n, dx, ts, beta=beta, max_iter=600)
        c5_s = compute_c5(h_s, dx)
        if c5_s < best_c5 - 1e-14:
            best_h, best_c5 = h_s, c5_s

    # Phase 3: Subgradient exploration at n
    t3 = t0 + budget_s * 0.72
    ss = 0.06
    stall = 0
    for _ in range(2000):
        if elapsed() > t3:
            break
        h_sg, v_sg = subgrad_step(best_h, n, dx, ss, active_tol=0.008)
        if v_sg < best_c5 - 1e-15:
            best_h, best_c5 = h_sg, v_sg
            ss = max(ss * 0.97, 0.005)
            stall = 0
        else:
            ss = max(ss * 0.8, 0.005)
            stall += 1
        if stall > 80 and int(_ % 40) == 0:
            h_r = slsqp_direct(best_h + rng.randn(n) * 0.05, n, dx, ts, max_iter=300)
            c5_r = compute_c5(h_r, dx)
            if c5_r < best_c5 - 1e-14:
                best_h, best_c5 = h_r, c5_r
                stall = 0
                ss = 0.05

    # Phase 4: Upsample to 2n
    t4 = t0 + budget_s * 0.90
    if elapsed() < t4:
        n2 = 2 * n
        dx2 = 2.0 / n2
        ts2 = n2 / 2.0
        h_up = np.interp(np.linspace(0, n - 1, n2), np.arange(n), best_h)
        h_up = project_feasible(h_up, n2)

        h_u = slsqp_direct(h_up, n2, dx2, ts2, max_iter=400)
        c5_u = compute_c5(h_u, dx2)
        if c5_u < best_c5 - 1e-14:
            best_h, best_c5 = h_u, c5_u

        h_u = slsqp_smooth(best_h, n2, dx2, ts2, beta=100, max_iter=500)
        c5_u = compute_c5(h_u, dx2)
        if c5_u < best_c5 - 1e-14:
            best_h, best_c5 = h_u, c5_u

        ss2 = 0.04
        stall2 = 0
        for _ in range(1500):
            if elapsed() > t4:
                break
            h_sg2, v_sg2 = subgrad_step(best_h, n2, dx2, ss2, active_tol=0.004)
            if v_sg2 < best_c5 - 1e-15:
                best_h, best_c5 = h_sg2, v_sg2
                ss2 = max(ss2 * 0.97, 0.003)
                stall2 = 0
            else:
                ss2 = max(ss2 * 0.8, 0.003)
                stall2 += 1
            if stall2 > 60 and int(_ % 30) == 0:
                h_r2 = slsqp_direct(best_h + rng.randn(n2) * 0.03, n2, dx2, ts2, max_iter=250)
                c5_r2 = compute_c5(h_r2, dx2)
                if c5_r2 < best_c5 - 1e-14:
                    best_h, best_c5 = h_r2, c5_r2
                    stall2 = 0
                    ss2 = 0.03

        h_u = slsqp_direct(best_h, n2, dx2, ts2, max_iter=600)
        c5_u = compute_c5(h_u, dx2)
        if c5_u < best_c5 - 1e-14:
            best_h, best_c5 = h_u, c5_u

    # Phase 5: Final polish
    if elapsed() < budget_s - 5:
        nf = len(best_h)
        dxf = 2.0 / nf
        tf = nf / 2.0
        h_f = slsqp_smooth(best_h, nf, dxf, tf, beta=200, max_iter=800)
        c5_f = compute_c5(h_f, dxf)
        if c5_f < best_c5 - 1e-14:
            best_h, best_c5 = h_f, c5_f
        h_f = slsqp_direct(best_h, nf, dxf, tf, max_iter=1000)
        c5_f = compute_c5(h_f, dxf)
        if c5_f < best_c5 - 1e-14:
            best_h, best_c5 = h_f, c5_f

    return (best_h, best_c5, len(best_h))
