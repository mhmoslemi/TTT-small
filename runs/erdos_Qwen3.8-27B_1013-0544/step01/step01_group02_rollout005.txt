        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 80
        h = np.full(n, 0.5)

    h = project_feasible(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, n)

    # Phase 1: L-BFGS-B with continuation
    for alpha in [10, 30, 100, 300, 1000, 3000, 10000, 30000]:
        if budget_s - (time.time() - t0) < 25:
            break
        try:
            res = minimize(_obj_smooth, best_h, args=(n, alpha),
                           jac=lambda x: _grad_smooth(x, n, alpha),
                           method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': 2000, 'ftol': 1e-22,
                                    'gtol': 1e-16, 'maxls': 50})
            ht = project_feasible(res.x, n)
            c = compute_c5(ht, n)
            if c < best_c5:
                best_h, best_c5 = ht, c
        except Exception:
            pass

    # Phase 2: Steepest 2-point descent (full sweep)
    for sweep in range(2000):
        if budget_s - (time.time() - t0) < 120:
            break
        G = get_grad_matrix(best_h, n)
        overlaps = np.correlate(best_h, 1 - best_h, mode='full') * (2.0 / n)
        cur_max = overlaps.max()

        best_imp = 0.0
        best_delta = 0.0
        best_pair = None

        for i in range(n):
            for j in range(i + 1, n):
                d = G[i] - G[j]
                lo = max(-best_h[i], best_h[j] - 1.0)
                hi = min(1.0 - best_h[i], best_h[j])
                if hi - lo < 1e-14:
                    continue
                delta, obj_val = _golden_min_1d(overlaps, d, lo, hi)
                imp = cur_max - obj_val
                if imp > best_imp:
                    best_imp = imp
                    best_delta = delta
                    best_pair = (i, j)

        if best_pair is not None and best_imp > 1e-14:
            i, j = best_pair
            best_h[i] += best_delta
            best_h[j] -= best_delta
            best_h = np.clip(best_h, 0.0, 1.0)
            if abs(best_h.sum() - n / 2.0) > 1e-12:
                best_h = project_feasible(best_h, n)
            new_c5 = compute_c5(best_h, n)
            if new_c5 < best_c5:
                best_c5 = new_c5
        else:
            break

    # Phase 3: Randomized 2-point moves with restarts
    for restart in range(30):
        if budget_s - (time.time() - t0) < 60:
            break
        scale = rng.choice([0.005, 0.01, 0.02, 0.05, 0.1])
        h_pert = best_h + rng.normal(0, scale, size=n)
        h_pert = project_feasible(h_pert, n)
        c_pert = compute_c5(h_pert, n)

        # Short steepest-2-point descent from perturbed point
        for _ in range(50):
            G = get_grad_matrix(h_pert, n)
            overlaps = np.correlate(h_pert, 1 - h_pert, mode='full') * (2.0 / n)
            cur_max = overlaps.max()
            best_imp = 0.0
            best_delta = 0.0
            best_pair = None
            for _trial in range(500):
                pi, pj = rng.choice(n, 2, replace=False)
                d = G[pi] - G[pj]
                lo = max(-h_pert[pi], h_pert[pj] - 1.0)
                hi = min(1.0 - h_pert[pi], h_pert[pj])
                if hi - lo < 1e-14:
                    continue
                delta, obj_val = _golden_min_1d(overlaps, d, lo, hi)
                imp = cur_max - obj_val
                if imp > best_imp:
                    best_imp = imp
                    best_delta = delta
                    best_pair = (pi, pj)
            if best_pair is not None and best_imp > 1e-14:
                pi, pj = best_pair
                h_pert[pi] += best_delta
                h_pert[pj] -= best_delta
                h_pert = np.clip(h_pert, 0.0, 1.0)
                if abs(h_pert.sum() - n / 2.0) > 1e-12:
                    h_pert = project_feasible(h_pert, n)
            else:
                break

        c_pert = compute_c5(h_pert, n)
        if c_pert < best_c5:
            best_h, best_c5 = h_pert.copy(), c_pert

    # Phase 4: Fine 2-point sweep with very small deltas (local refinement)
    for sweep in range(1000):
        if budget_s - (time.time() - t0) < 20:
            break
        improved = False
        for i in range(n):
            if budget_s - (time.time() - t0) < 10:
                break
            for j in range(i + 1, n):
                d_trial = 0.001 * (1.0 if best_h[i] < best_h[j] else -1.0)
                hi = best_h[i] + d_trial
                hj = best_h[j] - d_trial
                if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                    h_try = best_h.copy()
                    h_try[i] = hi
                    h_try[j] = hj
                    c = compute_c5(h_try, n)
                    if c < best_c5 - 1e-15:
                        best_h = h_try
                        best_c5 = c
                        improved = True
        if not improved:
            break

    # Phase 5: Final SLSQP polish
    if budget_s - (time.time() - t0) > 10:
        try:
            res = minimize(_obj_smooth, best_h, args=(n, 100000),
                           jac=lambda x: _grad_smooth(x, n, 100000),
                           method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': 5000, 'ftol': 1e-22, 'gtol': 1e-16})
            ht = project_feasible(res.x, n)
            c = compute_c5(ht, n)
            if c < best_c5:
                best_h, best_c5 = ht, c
        except Exception:
            pass

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
