            grad[:n + ell] -= w[k] * h[-ell:]
    grad *= dx
    grad += 2.0 * penalty * (np.sum(h) - target)
    return grad

def c5_objective(x, dx, beta, target, penalty):
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    val = smooth_max(corr, beta)
    diff = np.sum(h) - target
    val += penalty * diff * diff
    return val

def c5_objective_grad(x, dx, beta, target, penalty):
    h = np.clip(x, 0.0, 1.0)
    val = c5_objective(h, dx, beta, target, penalty)
    g = c5_gradient_vec(h, dx, beta, target, penalty)
    g[h <= 0.0] *= 0.0
    g[h >= 1.0] *= 0.0
    return val, g

def project_feasible(h, n):
    target = n / 2.0
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - target
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(free):
            h[free] -= diff / np.count_nonzero(free)
        else:
            h -= diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = np.sum(h) - target
    if abs(diff) > 1e-15:
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(free):
            h[free] -= diff / np.count_nonzero(free)
            h = np.clip(h, 0.0, 1.0)
    return h

def lbfgsb_optimize(h_start, n, dx, target, beta, max_iter=400, penalty=200.0):
    def obj(x):
        return c5_objective_grad(x, dx, beta, target, penalty)
    res = minimize(obj, h_start, method='L-BFGS-B', jac=True,
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-8, 'maxls': 50})
    return project_feasible(res.x, n)

def slsqp_optimize(h_start, n, dx, target, beta, max_iter=500):
    bounds = [(0.0, 1.0)] * n
    constraint = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                   'jac': lambda x: np.ones(n)}]
    res = minimize(lambda x: smooth_max(np.correlate(x, 1.0 - x, mode='full') * dx, beta),
                   h_start, method='SLSQP', bounds=bounds, constraints=constraint,
                   options={'maxiter': max_iter, 'ftol': 1e-14})
    return project_feasible(res.x, n)

def random_step_func(n_local, dx_local, rng_local):
    n_seg = rng_local.randint(4, 15)
    bps = np.sort(rng_local.uniform(0, 2, n_seg))
    bps = np.concatenate([[0.0], bps, [2.0]])
    x = (np.arange(n_local) + 0.5) * dx_local
    h_rf = np.zeros(n_local)
    for s in range(len(bps) - 1):
        mask = (x >= bps[s]) & (x < bps[s + 1])
        h_rf[mask] = rng_local.uniform(0, 1)
    return project_feasible(h_rf, n_local)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0

    best_h = project_feasible(h0.copy(), n)
    best_c5 = compute_c5(best_h, dx)

    def elapsed():
        return time.time() - start_time

    # Phase 1: L-BFGS-B with annealing beta at original resolution
    phase1_end = budget_s * 0.50
    betas = [3, 5, 10, 20, 40, 80, 150, 300, 600]
    for beta in betas:
        if elapsed() > phase1_end:
            break
        n_restarts = max(2, min(8, int(12 * (4.0 / beta) ** 0.4)))
        for r in range(n_restarts):
            if elapsed() > phase1_end:
                break
            if r == 0:
                h_try = best_h.copy()
            elif r == 1:
                h_try = project_feasible(best_h + rng.randn(n) * 0.04, n)
            else:
                if rng.rand() < 0.4:
                    h_try = project_feasible(best_h + rng.randn(n) * 0.12, n)
                else:
                    h_try = random_step_func(n, dx, rng)

            h_new = lbfgsb_optimize(h_try, n, dx, target, beta,
                                     max_iter=300, penalty=200.0)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5 - 1e-15:
                best_h = h_new
                best_c5 = c5_new

    # Phase 2: SLSQP polish at original resolution
    if elapsed() < budget_s * 0.68:
        for beta in [100, 300, 1000]:
            if elapsed() > budget_s * 0.68:
                break
            h_new = slsqp_optimize(best_h, n, dx, target, beta, max_iter=800)
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5 - 1e-15:
                best_h = h_new
                best_c5 = c5_new

    # Phase 3: Upsample to 2n and optimize
    if elapsed() < budget_s * 0.87:
        n2 = 2 * n
        dx2 = 2.0 / n2
        target2 = n2 / 2.0

        x_orig = (np.arange(n) + 0.5) * dx
        x_new = (np.arange(n2) + 0.5) * dx2
        h_interp = np.interp(x_new, x_orig, best_h)
        h_interp = project_feasible(h_interp, n2)
        h_rep = project_feasible(np.repeat(best_h, 2), n2)

        c5_interp = compute_c5(h_interp, dx2)
        c5_rep = compute_c5(h_rep, dx2)
        h_up = h_interp if c5_interp <= c5_rep else h_rep
        c5_up = min(c5_interp, c5_rep)

        if c5_up < best_c5:
            best_h = h_up.copy()
            best_c5 = c5_up

        betas2 = [20, 50, 150, 400]
        for beta in betas2:
            if elapsed() > budget_s * 0.87:
                break
            h_new = lbfgsb_optimize(h_up, n2, dx2, target2, beta,
                                     max_iter=250, penalty=200.0)
            c5_new = compute_c5(h_new, dx2)
            if c5_new < best_c5 - 1e-15:
                best_h = h_new
                best_c5 = c5_new
                h_up = h_new.copy()
            elif c5_new < c5_up - 1e-15:
                h_up = h_new.copy()
                c5_up = c5_new

    # Phase 4: Final SLSQP polish at best resolution
    n_final = len(best_h)
    dx_final = 2.0 / n_final
    target_final = n_final / 2.0
    phase4_end = budget_s - 15

    attempt = 0
    while elapsed() < phase4_end:
        if attempt % 3 == 0:
            h_try = best_h.copy()
        elif attempt % 3 == 1:
            h_try = project_feasible(best_h + rng.randn(n_final) * 0.01, n_final)
        else:
            h_try = project_feasible(best_h + rng.randn(n_final) * 0.03, n_final)

        h_new = slsqp_optimize(h_try, n_final, dx_final, target_final,
                                beta=500, max_iter=800)
        c5_new = compute_c5(h_new, dx_final)
        if c5_new < best_c5 - 1e-15:
            best_h = h_new
            best_c5 = c5_new
        attempt += 1

    best_h = project_feasible(best_h, n_final)
    best_c5 = compute_c5(best_h, dx_final)
    return (best_h, best_c5, n_final)
