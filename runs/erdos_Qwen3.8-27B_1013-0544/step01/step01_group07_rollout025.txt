import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_grad(h, t):
    """Compute smooth max value and its gradient."""
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    max_c = np.max(corr)
    shifted = t * (corr - max_c)
    weights = np.exp(shifted)
    sw = np.sum(weights)
    weights /= sw
    smooth_val = max_c + np.log(sw) / t
    
    # Gradient computation (vectorized O(n^2))
    N = 2 * n - 1
    j = np.arange(n)
    k = np.arange(N)
    raw1 = j[:, None].astype(float) - k[None, :].astype(float) + n - 1
    raw2 = j[:, None].astype(float) + k[None, :].astype(float) - n + 1
    valid1 = (raw1 >= 0) & (raw1 < n)
    valid2 = (raw2 >= 0) & (raw2 < n)
    idx1 = np.clip(np.round(raw1).astype(int), 0, n-1)
    idx2 = np.clip(np.round(raw2).astype(int), 0, n-1)
    h1 = np.where(valid1, h[idx1], 0.0)
    h2 = np.where(valid2, h[idx2], 0.0)
    grad = np.ones(n) - (weights[None, :] * h1).sum(axis=1) - (weights[None, :] * h2).sum(axis=1)
    
    return smooth_val, grad

def project_feasible(x, target_sum):
    """Project x onto {0<=h<=1, sum(h)=target_sum} using bisection on lambda."""
    # Find lambda such that sum(clip(x - lambda, 0, 1)) = target_sum
    lo, hi = -max(x) - 1, max(x) + 1
    for _ in range(60):
        mid = (lo + hi) / 2
        h_try = np.clip(x - mid, 0.0, 1.0)
        s = np.sum(h_try)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(x - (lo + hi) / 2, 0.0, 1.0)
    # Final correction for any tiny residual
    diff = target_sum - np.sum(h_proj)
    if abs(diff) > 1e-10:
        # Distribute the difference among non-boundary elements
        free = (h_proj > 1e-8) & (h_proj < 1 - 1e-8)
        if np.sum(free) > 0:
            h_proj[free] += diff / np.sum(free)
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def nesterov_optimize(h0, target_sum, t_init=5, t_final=500, 
                       n_iter=5000, lr_init=0.1, beta=0.9, 
                       time_limit=None, t0=None):

    n = len(h0)
    h = h0.copy()
    y = h0.copy()
    v = np.zeros(n)
    best_h = h0.copy()
    best_val = compute_c5(h0, 2.0/n)
    
    for i in range(n_iter):
        if time_limit is not None and t0 is not None:
            if time.time() - t0 > time_limit:
                break
        
        # Anneal t
        progress = i / max(1, n_iter - 1)
        t = t_init * (t_final / t_init) ** progress
        
        # Compute gradient at y
        _, grad_y = smooth_max_grad(y, t)
        
        # Nesterov step
        v_new = beta * v - lr_init * grad_y
        y_new = y + v_new
        
        # Project
        h_new = project_feasible(y_new, target_sum)
        
        # Check improvement
        val = compute_c5(h_new, 2.0/n)
        if val < best_val:

            best_h = h_new.copy()
        
        y = h_new
        v = v_new
    
    return best_h, best_val

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_value(h, t):
    corr = np.correlate(h, 1 - h, mode="full")
    max_c = np.max(corr)
    if t < 1000:
        shifted = t * (corr - max_c)
        return max_c + np.log(np.sum(np.exp(shifted))) / t
    return max_c

def smooth_max_grad(h, t):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    max_c = np.max(corr)
    shifted = t * (corr - max_c)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    
    # Vectorized gradient computation
    N = 2 * n - 1
    j = np.arange(n)[:, None].astype(float)
    k = np.arange(N)[None, :].astype(float)
    raw1 = j - k + n - 1
    raw2 = j + k - n + 1
    valid1 = (raw1 >= 0) & (raw1 < n)
    valid2 = (raw2 >= 0) & (raw2 < n)
    idx1 = np.clip(np.round(raw1).astype(int), 0, n - 1)
    idx2 = np.clip(np.round(raw2).astype(int), 0, n - 1)
    h1 = np.where(valid1, h[idx1], 0.0)
    h2 = np.where(valid2, h[idx2], 0.0)
    grad = np.ones(n) - (weights[None, :] * h1).sum(axis=1) - (weights[None, :] * h2).sum(axis=1)
    return grad

            best_h = y.copy()

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_grad(h, t):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    max_c = np.max(corr)
    shifted = t * (corr - max_c)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    N = 2 * n - 1
    j = np.arange(n)[:, None].astype(float)
    k = np.arange(N)[None, :].astype(float)
    raw1 = j + k - n + 1
    raw2 = j - k + n - 1
    valid1 = (raw1 >= 0) & (raw1 < n)
    valid2 = (raw2 >= 0) & (raw2 < n)
    idx1 = np.clip(np.round(raw1).astype(int), 0, n - 1)
    idx2 = np.clip(np.round(raw2).astype(int), 0, n - 1)
    h1 = np.where(valid1, h[idx1], 0.0)
    h2 = np.where(valid2, h[idx2], 0.0)
    grad = np.ones(n) - (weights[None, :] * h1).sum(axis=1) - (weights[None, :] * h2).sum(axis=1)
    return grad

def project_feasible(x, target_sum):
    lo, hi = x.min() - 2.0, x.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:

    lam = (lo + hi) / 2.0
    h = np.clip(x - lam, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        free = (h > 1e-6) & (h < 1.0 - 1e-6)
        if free.sum() > 0:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def nesterov_pg(h0, target_sum, t_init, t_final, n_iter, lr0, beta, time_limit, t_start):
    n = len(h0)
    dx = 2.0 / n
    y = h0.copy()

    best_h = h0.copy()
    best_c5 = compute_c5(h0, dx)
    stalled = 0
    for i in range(n_iter):
        if time.time() - t_start > time_limit:
            break
        frac = i / max(1, n_iter - 1)
        t = t_init * (t_final / t_init) ** frac
        lr = lr0 / (1.0 + 2.0 * frac)
        grad = smooth_max_grad(y, t)
        v = beta * v - lr * grad
        y_new = project_feasible(y + v, target_sum)
        c5_new = compute_c5(y_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = y_new.copy()
            stalled = 0
        else:
            stalled += 1
            if stalled > 2000 and frac > 0.3:
                y_new = project_feasible(y + 0.5 * v, target_sum)
        y = y_new
    return best_h, best_c5

def sa_refine(h0, target_sum, n_iter, time_limit, t_start, T0=0.08, T1=0.002):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h0, dx)
    for i in range(n_iter):
        if time.time() - t_start > time_limit:
            break

        T = T0 * (T1 / T0) ** frac
        n_pert = np.random.randint(1, 4)
        idx = np.random.choice(n, n_pert, replace=False)
        amp = 0.4 * (1.0 - 0.5 * frac)
        delta = np.random.uniform(-amp, amp, n_pert)
        h_new = h.copy()
        h_new[idx] += delta
        h_new = project_feasible(h_new, target_sum)
        c5_new = compute_c5(h_new, dx)
        c5_old = compute_c5(h, dx)
        d = c5_new - c5_old
        if d < 0 or np.random.rand() < np.exp(-d / max(T, 1e-8)):
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
    return best_h, best_c5

def slsqp_polish(h0, target_sum, t=300, maxiter=2000):
    n = len(h0)
    dx = 2.0 / n
    try:
        def obj(x):
            corr = np.correlate(x, 1 - x, mode="full")
            mc = np.max(corr)
            shifted = t * (corr - mc)
            return (mc + np.log(np.sum(np.exp(shifted))) / t) * dx
        res = minimize(obj, h0.copy(), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target_sum)
    except Exception:
        return h0
