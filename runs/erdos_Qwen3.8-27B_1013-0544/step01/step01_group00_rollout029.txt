```python
def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def sum_eq_fun(h):
    return np.sum(h) - len(h) * 0.5

def sum_eq_jac(h):
    return np.ones(len(h))

def project_h(h, target):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = np.sum(mask)
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

    m = np.max(corr)

    shifted = (corr - m) / temp
    return m + temp * np.log(np.sum(np.exp(shifted)))
```

```python
def coord_descent_step(h, m, target_sum):
    """Find optimal h[m] given all other h fixed, maintaining sum constraint."""
    n = len(h)
    dx = 2.0 / n
    others_sum = np.sum(h) - h[m]
    # h[m] is free (we'll project sum later), but for now let's just optimize
    # c_j as function of h[m]:
    # c_j = constant + h[m] * coeff_j
    # where coeff_j = (1-h[m+k_j])*[0<=m+k_j<n] - h[m-k_j]*[0<=m-k_j<n]
    
    coeffs = np.zeros(2*n-1)
    for j in range(2*n-1):
        k = j - (n-1)
        c_val = 0.0
        if 0 <= m+k < n:
            c_val += (1.0 - h[m+k])
        if 0 <= m-k < n:
            c_val -= h[m-k]
        coeffs[j] = c_val * dx
    
    # Current correlations (with current h[m]):
    corr = np.correlate(h, 1.0-h, mode='full') * dx
    # c_j(h[m]) = corr[j] - h[m]*coeffs[j] + h[m]*coeffs[j]... 
    # Wait, corr[j] was computed with the current h[m].
    # c_j(h[m]+δ) = corr[j] + δ * coeffs[j]
    # So c_j(h[m]) = corr[j] where h[m] is the current value.
    # c_j(v) for new value v: corr[j] + (v - h[m]) * coeffs[j]
    
    # We want to minimize max_j (corr[j] + (v-h[m])*coeffs[j]) over v in [0,1]
    # This is: minimize max_j (a_j * v + b_j) where a_j = coeffs[j], b_j = corr[j] - h[m]*coeffs[j]
    
    a = coeffs
    b = corr - h[m] * coeffs
    
    # Find v in [0,1] that minimizes max_j (a_j*v + b_j)
    # This is a 1D LP. The optimal v is either at a breakpoint or at a boundary.
    # Breakpoints are where two lines cross: a_i*v + b_i = a_j*v + b_j => v = (b_j-b_i)/(a_i-a_j)
    
    # For efficiency, find the v that minimizes the max:
    # The function f(v) = max_j(a_j*v+b_j) is convex in v.
    # Use ternary search or binary search on the subgradient.
    
    # Actually, since it's convex, we can use the fact that the minimum is where
    # the subgradient contains 0. The subgradient at v is the convex hull of {a_j : a_j*v+b_j = f(v)}.
    
    # Simple approach: ternary search on [0,1]
    lo, hi = 0.0, 1.0
    for _ in range(50):
        m1 = lo + (hi-lo)/3
        m2 = hi - (hi-lo)/3
        f1 = np.max(a*m1 + b)
        f2 = np.max(a*m2 + b)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    return (lo+hi)/2
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_h(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = np.sum(mask)
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def coord_optimize(h, m, n_grid=200):
    """Find optimal h[m] given all other h fixed."""
    n = len(h)
    dx = 2.0 / n
    # Compute coefficients and constants for the 1D problem
    # c_j(v) = b_j + a_j * v where v is the new value of h[m]
    # a_j = dx * [(1-h[m+k])*[0<=m+k<n] - h[m-k]*[0<=m-k<n]]
    # b_j = c_j(current) - h[m] * a_j = corr[j] - h[m]*a_j
    
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    a = np.zeros(2*n - 1)
    for j in range(2*n - 1):
        k = j - (n - 1)
        val = 0.0
        if 0 <= m + k < n:
            val += (1.0 - h[m + k])
        if 0 <= m - k < n:
            val -= h[m - k]
        a[j] = val * dx
    b = corr - h[m] * a
    
    # Minimize max_j (a_j * v + b_j) over v in [0, 1]
    v_grid = np.linspace(0, 1, n_grid)
    f_vals = np.max(a[:, None] * v_grid[None, :] + b[:, None], axis=0)
    best_v = v_grid[np.argmin(f_vals)]
    return best_v

def smooth_obj(h, temp, penalty_coeff=1000.0):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    shifted = (corr - m) / temp
    base = m + temp * np.log(np.sum(np.exp(shifted)))
    penalty = penalty_coeff * (np.sum(h) - len(h) * 0.5) ** 2
    return base + penalty

def optimize_smooth_lbfgsb(h0, temp, max_iter=500, ftol=1e-15):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    res = minimize(smooth_obj, h0.copy(), args=(temp,), method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': max_iter, 'ftol': ftol, 'gtol': 1e-6})
    h_new = np.clip(res.x, 0.0, 1.0)
    h_new = project_h(h_new, n * 0.5)
    return h_new, compute_c5(h_new)

def optimize_t_var(h0, max_iter=400, ftol=1e-10):
    n = len(h0)
    c5_cur = compute_c5(h0)
    v0 = np.append(h0.copy(), c5_cur + 0.02)
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.5)]
    
    def t_obj(v):
        return v[-1]
    
    def t_ineq(v):
        h = v[:n]
        t = v[-1]
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        return t - corr
    
    def t_eq(v):
        return np.sum(v[:n]) - n * 0.5
    
    def t_eq_jac(v):
        j = np.zeros(len(v))
        j[:n] = 1.0
        return j
    
    cons = [
        {'type': 'ineq', 'fun': t_ineq},
        {'type': 'eq', 'fun': t_eq, 'jac': t_eq_jac}
    ]
    res = minimize(t_obj, v0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': ftol})
    h_new = np.clip(res.x[:n], 0.0, 1.0)
    h_new = project_h(h_new, n * 0.5)
    return h_new, compute_c5(h_new)

def coord_descent(h, n_sweeps=50, n_grid=200, time_check=None, time_limit=100):

            break
        for m in range(n):
            h[m] = coord_optimize(h, m, n_grid)
        h = project_h(h, n * 0.5)
    return h
```

Wait, I realize there might be an issue with the coordinate descent. When I call `coord_optimize(h, m, n_grid)`, I'm computing the correlation `corr = np.correlate(h, 1.0-h, mode='full') * dx` which includes the current value of h[m]. Then I compute `a` and `b` based on this. The formula c_j(v) = b_j + a_j * v should give the correct correlation at lag j when h[m] = v and all other entries are unchanged.

Let me verify: 
- corr[j] = ∑_i h[i]*(1-h[i+k])*dx (with current h[m])
- When h[m] changes to v, the change is:
  - If m is a valid first index (m+k in range): the term h[m]*(1-h[m+k]) changes to v*(1-h[m+k]), change = (v-h[m])*(1-h[m+k])
  - If m is a valid second index (m-k in range): the term h[m-k]*(1-h[m]) changes to h[m-k]*(1-v), change = h[m-k]*(1-v - (1-h[m])) = h[m-k]*(h[m]-v) = -h[m-k]*(v-h[m])
  
So Δc_j = (v-h[m]) * [(1-h[m+k])*[valid] - h[m-k]*[valid]] * dx = (v-h[m]) * a_j

Therefore: c_j(v) = corr[j] + (v-h[m])*a_j = corr[j] - h[m]*a_j + v*a_j = b_j + a_j*v

Yes, this is correct!

Now, one concern: the coordinate descent might be slow because each sweep involves n calls to coord_optimize, each of which computes np.correlate (O(n^2)) and then does a vectorized grid evaluation (O(2n * n_grid)). For n=180, n_grid=200:
- np.correlate: O(180*359) ≈ 65K ops
- Grid evaluation: O(359*200) ≈ 72K ops (vectorized)
- Total per coordinate: ~137K ops
- Per sweep: 180 * 137K ≈ 24.6M ops
- Per 30 sweeps: ~740M ops

In numpy, this should take a few seconds (numpy can do ~100M ops/second for simple operations, but with the overhead of Python function calls and array creation, maybe 20-50M ops/second). So 30 sweeps might take 15-40 seconds. That's acceptable.

But wait, I'm also doing np.correlate inside coord_optimize for each coordinate. That's 180 calls to np.correlate per sweep. Each call is O(n^2) = O(32K) for the actual computation, but with function call overhead in Python, maybe 50-100μs each. So 180 * 100μs = 18ms per sweep just for correlates. That's fine.
