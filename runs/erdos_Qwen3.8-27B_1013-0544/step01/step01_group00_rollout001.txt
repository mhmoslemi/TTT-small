import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))


def t_objective(v):
    return v[-1]


def t_ineq(v):
    n = len(v) - 1
    h = v[:n]
    t = v[-1]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return t - corr


def t_eq_fun(v):
    n = len(v) - 1
    return np.sum(v[:n]) - n * 0.5


def t_eq_jac(v):
    n = len(v) - 1
    j = np.zeros(len(v))
    j[:n] = 1.0
    return j


def smooth_obj(h, temp):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    return m + temp * np.log(np.sum(np.exp((corr - m) / temp)))


def sum_eq_fun(h):
    return np.sum(h) - len(h) * 0.5


def sum_eq_jac(h):
    return np.ones(len(h))


def project_h(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = np.sum(mask)
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)


def interp_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 1, n_old, endpoint=False)
    x_new = np.linspace(0, 1, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_h(h_new, n_new * 0.5)


def optimize_t_var(h0, max_iter=400, ftol=1e-10):
    n = len(h0)
    c5_cur = compute_c5(h0)
    v0 = np.append(h0.copy(), c5_cur + 0.02)
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.5)]
    cons = [
        {'type': 'ineq', 'fun': t_ineq},
        {'type': 'eq', 'fun': t_eq_fun, 'jac': t_eq_jac}
    ]
    res = minimize(t_objective, v0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': ftol})
    h_new = np.clip(res.x[:n], 0.0, 1.0)
    h_new = project_h(h_new, n * 0.5)
    return h_new, compute_c5(h_new)


def optimize_smooth(h0, temp, max_iter=300, ftol=1e-12):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac}]
    res = minimize(smooth_obj, h0.copy(), args=(temp,), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': ftol})
    h_new = np.clip(res.x, 0.0, 1.0)
    h_new = project_h(h_new, n * 0.5)
    return h_new, compute_c5(h_new)


def polish_direct(h0, n_rounds=50):
    """Iteratively target the worst lag with small SLSQP steps."""
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    c5 = compute_c5(h)
    for _ in range(n_rounds):
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        idx = int(np.argmax(corr))
        
        def obj_worst(x):
            c = np.correlate(x, 1.0 - x, mode='full') * dx
            return c[idx] + 0.1 * np.max(c)
        
        res = minimize(obj_worst, h.copy(), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac}],
                       options={'maxiter': 50, 'ftol': 1e-14})
        h_try = project_h(res.x, n * 0.5)
        c5_try = compute_c5(h_try)
        if c5_try < c5 - 1e-10:
            h = h_try
            c5 = c5_try
        else:
            break
    return h, c5


def multiplicative_weights_polish(h0, n_iter=30, rho=0.5):
    """Exogenous optimization via multiplicative weights on lags."""
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    w = np.ones(len(corr))
    for it in range(n_iter):
        w = np.exp(rho * (corr - m))
        w /= np.sum(w)
        
        def obj_weighted(x):
            c = np.correlate(x, 1.0 - x, mode='full') * dx
            return np.dot(w, c)
        
        res = minimize(obj_weighted, h.copy(), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac}],
                       options={'maxiter': 100, 'ftol': 1e-13})
        h_new = project_h(res.x, n * 0.5)
        c5_new = compute_c5(h_new)
        if c5_new < compute_c5(h) + 1e-10:
            h = h_new
            corr = np.correlate(h, 1.0 - h, mode='full') * dx
            m = np.max(corr)
    return h, compute_c5(h)
