def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def lse_value(corr, temp):
    m = np.max(corr)
    if m < -1e30:
        return 0.0
    return m + temp * np.log(np.sum(np.exp((corr - m) / temp)))

def lse_weights(corr, temp):
    m = np.max(corr)
    w = np.exp((corr - m) / temp)
    s = np.sum(w)
    if s == 0:
        w = np.ones(len(corr)) / len(corr)
    else:
        w /= s
    return w

def project_feasible(h, target_sum):

    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-15:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(80):

        h_p = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(h_p)
        if s > target_sum:

        else:

        return v[-1]

        t = v[-1]

        return t - corr

        j = np.zeros(len(v))
        j[:n] = 1.0
        return j

    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]

        {'type': 'ineq', 'fun': ineq},
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}
    ]

                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-12})
    h_new = project_feasible(np.clip(res.x[:n], 0.0, 1.0), n * 0.5)
    return h_new, compute_c5(h_new)

import numpy as np
from scipy.optimize import minimize
from scipy.linalg import toeplitz
import time


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-15:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_p = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(h_p)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h_p = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - np.sum(h_p)
    if abs(diff) > 1e-12:
        mask = (h_p > 1e-10) & (h_p < 1.0 - 1e-10)
        if np.sum(mask) > 0:
            h_p[mask] += diff / np.sum(mask)
            h_p = np.clip(h_p, 0.0, 1.0)
    return h_p


def lse_grad(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    w = np.exp((corr - m) / temp)
    s = np.sum(w)
    if s < 1e-300:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w /= s
    col_A = w[:n][::-1]
    row_A = w[n - 1:]
    T_A = toeplitz(col_A, row_A)
    T_B = toeplitz(row_A, col_A)
    A = T_A @ h
    B = T_B @ h
    grad = dx * (1.0 - A - B)
    return grad, corr


def lse_val(corr, temp):
    m = np.max(corr)
    return float(m + temp * np.log(np.sum(np.exp((corr - m) / temp))))


def fista_optimize(h0, temp, max_iter=5000, step_size=None):

    target_sum = n * 0.5
    dx = 2.0 / n
    if step_size is None:

    h = h0.copy()
    z = h0.copy()
    t_k = 1.0
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    for it in range(max_iter):
        grad, corr_z = lse_grad(z, temp)
        f_z = lse_val(corr_z, temp)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        y = z + ((t_k - 1.0) / t_new) * (z - h)
        h = y - step_size * grad
        h = project_feasible(h, target_sum)
        corr_h = np.correlate(h, 1.0 - h, mode='full') * dx
        f_h = lse_val(corr_h, temp)
        if f_h > f_z + 0.01 * np.dot(grad, h - y) + 1e-12:
            h = y - step_size * grad
            h = project_feasible(h, target_sum)
            t_k = 1.0
        else:
            t_k = t_new
        z = h
        c5_cur = compute_c5(h)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h.copy()
    return best_h, best_c5


def subgrad_polish(h0, max_iter=5000, step_scale=1.0):
    n = len(h0)
    target_sum = n * 0.5
    dx = 2.0 / n
    h = h0.copy()
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    step = step_scale / np.sqrt(max_iter + 1)
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        k_star = int(np.argmax(corr))
        lag = k_star - (n - 1)
        grad = np.zeros(n)
        for p in range(n):
            g = 1.0
            if 0 <= p + lag < n:
                g -= h[p + lag]
            if 0 <= p - lag < n:
                g -= h[p - lag]
            grad[p] = g
        grad *= dx
        h = h - step * (it + 1) * grad / (it + 1)
        h = project_feasible(h, target_sum)
        c5_cur = compute_c5(h)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h.copy()
    return best_h, best_c5

            best_h = h_new.copy()
            best_c5 = c5_new
            h = h_new

    # Phase 2: Subgradient polish (fast, O(n) per step)
    if time_left() > 80:
        h_sub, c5_sub = subgrad_polish(best_h, max_iter=10000, step_scale=2.0)
        if c5_sub < best_c5:
            best_h = h_sub.copy()
            best_c5 = c5_sub

    # Phase 3: SLSQP polish
    if time_left() > 120:
        h_new, c5_new = slsqp_polish(best_h, max_iter=400)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new

    # Phase 4: Another FISTA round after polish
    for temp in [0.002, 0.001, 0.0005, 0.0002]:
        if time_left() < 60:
            break
        h_new, c5_new = fista_optimize(best_h, temp, max_iter=5000, step_size=5.0 / np.sqrt(n))
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new

    # Phase 5: Multi-start with perturbations
    n_starts = 12
    for i in range(n_starts):
        if time_left() < 50:
            break
        scale = 0.003 + 0.003 * (i % 6)
        h_pert = best_h + rng.randn(n) * scale
        h_pert = project_feasible(h_pert, n * 0.5)
        for temp in [0.005, 0.002]:
            if time_left() < 30:
                break
            h_new, c5_new = fista_optimize(h_pert, temp, max_iter=3000, step_size=5.0 / np.sqrt(n))
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                h_pert = h_new

    # Phase 6: Final SLSQP
    if time_left() > 60:
        h_new, c5_new = slsqp_polish(best_h, max_iter=500)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new

    # Final subgradient sweep
    if time_left() > 30:
        h_sub, c5_sub = subgrad_polish(best_h, max_iter=8000, step_scale=1.5)
        if c5_sub < best_c5:
            best_h = h_sub.copy()
            best_c5 = c5_sub

    best_h = project_feasible(best_h, len(best_h) * 0.5)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
