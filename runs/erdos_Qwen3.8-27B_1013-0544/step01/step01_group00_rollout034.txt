```python
def subgradient_step(h, step_size, eps=1e-4):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_val = np.max(corr)
    active = np.where(corr >= max_val - eps)[0]
    
    grad = np.zeros(n)
    for p in active:
        k = p - (n - 1)
        for m in range(n):
            g = 0.0
            if 0 <= m - k < n:
                g += (1.0 - h[m - k])
            if 0 <= m + k < n:
                g -= h[m + k]
            grad[m] += g * dx / len(active)
    
    h_new = h - step_size * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_h(h_new, n * 0.5)
    return h_new
```

This is a linear program at each step (since the linearized correlation is linear in h). LPs are much faster to solve than QPs, and cvxpy with CLP or GLPK should handle this quickly.

Wait, but the linearization of a bilinear function isn't great. Let me think...

corr_k = sum_i h[i] * (1-h[i+k]) = sum_i h[i] - sum_i h[i]*h[i+k]

The second term is bilinear. Linearizing around h0:
h[i]*h[i+k] ≈ h0[i]*h[i+k] + h[i]*h0[i+k] - h0[i]*h0[i+k]

So corr_k ≈ sum_i h[i] - sum_i (h0[i]*h[i+k] + h[i]*h0[i+k] - h0[i]*h0[i+k])
= sum_i h[i] - sum_i h0[i]*h[i+k] - sum_i h[i]*h0[i+k] + sum_i h0[i]*h0[i+k]

The first and last terms are constants. The middle terms are linear in h. So yes, the linearization is linear in h, and we get an LP!

LP: minimize t
subject to: for each k, (linear expression in h) <= t
           sum(h) = n/2
           0 <= h <= 1

This is a standard LP with n+1 variables and 2n-1 + n + n + 1 constraints. For n=180, that's 181 variables and about 720 constraints. This should be solvable in milliseconds with a good LP solver.

The algorithm would be:
1. Start from h0
2. Linearize corr around h0
3. Solve LP to get h1
4. Check if h1 is close to h0 (convergence) or if the objective improved
5. If not converged, set h0 = h1 (or a blend) and repeat

This is essentially a **semidefinite/linear programming relaxation** approach, similar to the "method of conditional expectations" or "linearization iteration" used in combinatorial optimization.

```python
import cvxpy as cp

def lp_linearization(h0, max_iter=100, tol=1e-8):
    n = len(h0)
    dx = 2.0 / n
    h_cur = h0.copy()
    
    for it in range(max_iter):
        # Compute current correlation
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full') * dx
        
        # Linearize: corr_k ≈ constant + linear terms in h
        # corr_k = sum_i h[i]*(1-h[i+k]) * dx
        # ≈ sum_i (h0[i] - h0[i]*h0[i+k]) * dx + sum_i (1-h0[i+k])*dh[i] * dx - sum_i h0[i+k]*dh[i]... 
        
        # Actually, let me be more careful.
        # f(h) = sum_i h[i]*(1-h[i+k])
        # ∇f at h0: for each m, ∂f/∂h[m] = (1-h0[m-k]) - h0[m+k] (with validity checks)
        # f(h0) = sum_i h0[i]*(1-h0[i+k])
        # Linear approximation: f(h) ≈ f(h0) + ∇f(h0) . (h - h0)
        
        # Build the LP
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        
        constraints = []
        constraints.append(cp.sum(h_var) == n * 0.5)
        constraints.append(h_var >= 0)
        constraints.append(h_var <= 1)
        
        for k_idx in range(2 * n - 1):
            k = k_idx - (n - 1)
            # Compute f(h0) at this lag
            f0 = 0.0
            grad_f = np.zeros(n)
            for i in range(n):
                # h[i] * (1 - h[i-k])
                if 0 <= i - k < n:
                    f0 += h_cur[i] * (1.0 - h_cur[i - k])
                    # gradient w.r.t. h[i]: (1 - h0[i-k])
                    grad_f[i] += (1.0 - h_cur[i - k])
                    # gradient w.r.t. h[i-k]: -h0[i]
                    grad_f[i - k] += (-h_cur[i])
            # Linear approximation: f(h) ≈ f0 + grad_f . (h - h0)
            # = f0 - grad_f . h0 + grad_f . h
            const = (f0 - np.dot(grad_f, h_cur)) * dx
            coeff = grad_f * dx
            # coeff . h + const <= t
            constraints.append(cp.matmul(coeff.T, h_var) + const <= t_var)
        
        objective = cp.Minimize(t_var)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.CLARABEL)  # or cp.SCS, cp.ECOS
        except:
            prob.solve(solver=cp.SCS)
        
        if h_var.value is None:
            break
        
        h_new = np.clip(h_var.value, 0.0, 1.0)
        h_new = project_h(h_new, n * 0.5)
        c5_new = compute_c5(h_new)
        
        if abs(c5_new - compute_c5(h_cur)) < tol:
            h_cur = h_new
            break
        
        # Blend to avoid oscillation
        h_cur = 0.5 * h_cur + 0.5 * h_new
        
    return h_cur, compute_c5(h_cur)
```

Let me also think about **another approach: use the structure of the problem more directly**.

The correlation at lag k is:
C(k) = sum_i h[i] * (1 - h[i+k])

We want to minimize max_k C(k).

Note that C(k) + C(-k) = sum_i h[i](1-h[i+k]) + sum_i h[i](1-h[i-k])
= sum_i h[i] - sum_i h[i]h[i+k] + sum_i h[i] - sum_i h[i]h[i-k]
= 2*sum(h) - sum_i h[i]h[i+k] - sum_i h[i]h[i-k]
= 2n/2 - 2*sum_i h[i]h[i+k] (by symmetry of the sum)
= n - 2*A(k)

where A(k) = sum_i h[i]h[i+k] is the autocorrelation of h at lag k.

So C(k) + C(-k) = n - 2*A(k), which means:
max_k C(k) >= max_k (n - 2*A(k)) / 2 = n/2 - min_k A(k)

So minimizing max_k C(k) is related to maximizing min_k A(k), i.e., making the autocorrelation as "flat" as possible (in the minimum sense).

This is exactly the problem of designing a "low autocorrelation" sequence! In signal processing, this is related to Zadoff-Chu sequences, m-sequences, Barker sequences, etc.

For the continuous case (h in [0,1]), the optimal solution might be related to a "perfect" sequence where A(k) is constant for all k ≠ 0.

For binary sequences with n/2 ones, the autocorrelation A(k) = n/2 * p(k) where p(k) is the fraction of positions where both h[i] and h[i+k] are 1. If the sequence is "random-like", p(k) ≈ 1/4 for k ≠ 0, so A(k) ≈ n/8.

Then C(k) ≈ n/2 - 2*n/8 = n/4 for large k, and C(0) = 0 (for binary).

So for a random binary sequence, max C(k) ≈ n/4, giving C5 ≈ (n/4) * (2/n) = 1/2.

But we can do better than random. The optimal binary sequence (if it exists) would have max C(k) < n/4.

For the continuous case, we can likely do even better by using fractional values.

Anyway, this theoretical insight doesn't directly give us an algorithm, but it suggests that the problem is related to sequence design and that there might be structured solutions.

```python
def subgradient_optimize(h0, n_steps=5000, lr0=0.01, eps=1e-3):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for step in range(n_steps):
        # Adaptive learning rate
        lr = lr0 / (1.0 + step / 500.0)
        
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        max_val = np.max(corr)
        active = np.where(corr >= max_val - eps)[0]
        
        grad = np.zeros(n)
        for p in active:
            k = p - (n - 1)
            # Compute gradient of corr[p] w.r.t. h
            for m in range(n):
                g = 0.0
                if 0 <= m - k < n:
                    g += (1.0 - h[m - k])
                if 0 <= m + k < n:
                    g -= h[m + k]
                grad[m] += g
        grad *= dx / len(active)
        
        h = h - lr * grad
        h = np.clip(h, 0.0, 1.0)
        h = project_h(h, n * 0.5)
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5
```

```python
def corr_gradient(h, k):
    n = len(h)
    grad = np.zeros(n)
    # Term 1: (1-h[m-k]) for m where 0<=m-k<n, i.e., k<=m<n+k
    m_start = max(0, k)
    m_end = min(n, n + k)
    if m_start < m_end:
        grad[m_start:m_end] += (1.0 - h[m_start - k:m_end - k])
    # Term 2: -h[m+k] for m where 0<=m+k<n, i.e., -k<=m<n-k
    m_start2 = max(0, -k)
    m_end2 = min(n, n - k)
    if m_start2 < m_end2:
        grad[m_start2:m_end2] -= h[m_start2 + k:m_end2 + k]
    return grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))


def t_objective(v):
    return v[-1]


def t_ineq(v):
    n = len(v) - 1
    h = v[:n]
    t = v[-1]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return t - corr


def t_eq_fun(v):
    n = len(v) - 1
    return np.sum(v[:n]) - n * 0.5


def t_eq_jac(v):
    n = len(v) - 1
    j = np.zeros(len(v))
    j[:n] = 1.0
    return j


def smooth_obj(h, temp):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip((corr - m) / temp, -500, 500)
    return m + temp * np.log(np.sum(np.exp(shifted)))


def sum_eq_fun(h):
    return np.sum(h) - len(h) * 0.5


def sum_eq_jac(h):
    return np.ones(len(h))


def project_h(h, target):
    h = h.copy()
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return np.clip(h, 0.0, 1.0)
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = np.sum(mask)
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)


def interp_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 1, n_old, endpoint=False)
    x_new = np.linspace(0, 1, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_h(h_new, n_new * 0.5)


def optimize_t_var(h0, max_iter=400, ftol=1e-10):
    n = len(h0)
    c5_cur = compute_c5(h0)
    v0 = np.append(h0.copy(), c5_cur + 0.02)
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.5)]
    cons = [
        {'type': 'ineq', 'fun': t_ineq},
        {'type': 'eq', 'fun': t_eq_fun, 'jac': t_eq_jac}
    ]
    res = minimize(t_objective, v0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': ftol})
    h_new = np.clip(res.x[:n], 0.0, 1.0)
    h_new = project_h(h_new, n * 0.5)
    return h_new, compute_c5(h_new)


def optimize_smooth(h0, temp, max_iter=300, ftol=1e-12):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac}]
    res = minimize(smooth_obj, h0.copy(), args=(temp,), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': ftol})
    h_new = np.clip(res.x, 0.0, 1.0)
    h_new = project_h(h_new, n * 0.5)
    return h_new, compute_c5(h_new)


def lp_linearize(h0, max_iter=50, tol=1e-7, blend=0.5):
    n = len(h0)
    dx = 2.0 / n
    h_cur = h0.copy()
    c5_cur = compute_c5(h_cur)
    
    for it in range(max_iter):
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full') * dx
        
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == n * 0.5,
            h_var >= 0,
            h_var <= 1
        ]
        
        for k_idx in range(2 * n - 1):
            k = k_idx - (n - 1)
            # Compute f(h0) and gradient at this lag
            f0 = 0.0
            grad_f = np.zeros(n)
            for i in range(n):
                if 0 <= i - k < n:
                    f0 += h_cur[i] * (1.0 - h_cur[i - k])
                    grad_f[i] += (1.0 - h_cur[i - k])
                    grad_f[i - k] += (-h_cur[i])
            const = (f0 - np.dot(grad_f, h_cur)) * dx
            coeff = grad_f * dx
            constraints.append(coeff @ h_var + const <= t_var)
        
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iter=10000)
        except Exception:
            try:
                prob.solve(solver=cp.CVXOPT, max_iter=10000)
            except Exception:
                break
        
        if h_var.value is None:
            break
        
        h_new = np.clip(h_var.value, 0.0, 1.0)
        h_new = project_h(h_new, n * 0.5)
        c5_new = compute_c5(h_new)
        
        if abs(c5_new - c5_cur) < tol:
            h_cur = h_new
            c5_cur = c5_new
            break
        
        h_cur = (1.0 - blend) * h_cur + blend * h_new
        c5_cur = compute_c5(h_cur)
    
    return h_cur, compute_c5(h_cur)
```

For a given k, the gradient is:

```python
def compute_lag_gradient(h, k, n):
    grad = np.zeros(n)
    # Positive term: (1-h[m-k]) for m in [max(0,k), min(n, n+k))
    m_start = max(0, k)
    m_end = min(n, n + k)
    if m_start < m_end:
        grad[m_start:m_end] += 1.0 - h[m_start - k:m_end - k]
    # Negative term: -h[m+k] for m in [max(0,-k), min(n, n-k))
    m_start2 = max(0, -k)
    m_end2 = min(n, n - k)
    if m_start2 < m_end2:
        grad[m_start2:m_end2] -= h[m_start2 + k:m_end2 + k]
    return grad
```

```python
def compute_lag_value(h, k, n):
    # sum_i h[i] * (1-h[i-k]) for i and i-k both in [0,n)
    m_start = max(0, k)
    m_end = min(n, n + k)
    if m_start >= m_end:
        return 0.0
    return np.sum(h[m_start:m_end] * (1.0 - h[m_start - k:m_end - k]))
```

```python
    return float(np.max(corr))

    return v[-1]


def t_ineq(v):
    n = len(v) - 1
    h = v[:n]
    t = v[-1]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return t - corr

    n = len(v) - 1
    return np.sum(v[:n]) - n * 0.5

    n = len(v) - 1
    j = np.zeros(len(v))
    j[:n] = 1.0
    return j


def smooth_obj(h, temp):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip((corr - m) / temp, -500, 500)
    return m + temp * np.log(np.sum(np.exp(shifted)))

    h = h.copy()

    else:

        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)
```
