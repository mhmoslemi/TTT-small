def compute_corr(h):
    return np.correlate(h, 1 - h, mode='full')

def project_box_sum(h, target_sum):
    """Project h onto {x : 0 <= x[i] <= 1, sum(x) = target_sum}"""
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def solve_lp_step(h0, n, target_sum, max_c5):

    # corr[k] ≈ c_k + g_k · h where c_k and g_k are computed at h0

    # A_ub: each row is [g_k, -1], b_ub: -c_k

    # Variables: n+1

    return np.correlate(h, 1 - h, mode='full')

def project_box_sum(h, target_sum):
    lo, hi = -2.0, 2.0

        else:

    """Compute gradients of all corr[idx] w.r.t. h, returning (2n-1) x n matrix."""
    n_corr = 2 * n - 1
    G = np.zeros((n_corr, n))

            G[idx, k:] += (1.0 - h0[:n-k])
            if k > 0:

        else:
            p = -k
            G[idx, :n-p] += (1.0 - h0[p:])
            if p > 0:
                G[idx, p:] -= h0[:n-p]
    return G

                  bounds=bounds, method='highs')
    
    if res.success:
        h_lp = res.x[:n]
        h_new = (1 - damping) * h0 + damping * h_lp
        return project_box_sum(h_new, target_sum)
    return None

import numpy as np
import time
from scipy.optimize import linprog

def compute_corr(h):
    return np.correlate(h, 1 - h, mode='full')

def compute_c5(h):
    n = len(h)
    return np.max(compute_corr(h)) * (2.0 / n)

def project_box_sum(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def smooth_max_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    s = np.sum(w)
    if s < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= s
    grad = np.zeros(n)
    for idx in range(len(w)):
        if w[idx] < 1e-10:
            continue
        k = idx - (n - 1)
        wi = w[idx]
        if k >= 0:
            grad[k:] += wi * (1.0 - h[:n - k])
            if k > 0:
                grad[:n - k] -= wi * h[k:]
        else:
            p = -k
            grad[:n - p] += wi * (1.0 - h[p:])
            if p > 0:
                grad[p:] -= wi * h[:n - p]
    return grad * dx

def lp_direction(h0, n, target_sum):
    corr0 = compute_corr(h0)
    n_corr = len(corr0)
    G = np.zeros((n_corr, n))
    for idx in range(n_corr):
        k = idx - (n - 1)
        if k >= 0:
            G[idx, k:] = (1.0 - h0[:n - k])
            if k > 0:
                G[idx, :n - k] = (1.0 - h0[:n - k]) - h0[k:]
        else:
            p = -k
            G[idx, :n - p] = (1.0 - h0[p:])
            if p > 0:
                G[idx, p:] = -h0[:n - p]
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    A_ub = np.hstack([G, -np.ones((n_corr, 1))])
    b_ub = G @ h0 - corr0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target_sum])
    bounds = [(0, 1)] * n + [(None, None)]
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    if res.success:
        return res.x[:n]
    return None

def lp_optimize(h, n, target_sum, deadline, rng):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    damping = 1.0
    stall = 0
    while time.time() < deadline:
        h_lp = lp_direction(h, n, target_sum)
        if h_lp is None:
            break
        direction = h_lp - h
        norm_d = np.linalg.norm(direction)
        if norm_d < 1e-8:
            break
        # Line search along direction
        found = False
        for alpha_try in [damping, damping * 0.5, damping * 0.25, damping * 0.1]:
            h_new = project_box_sum(h + alpha_try * direction, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                h = h_new
                best_h = h_new.copy()
                best_c5 = c5_new
                damping = min(1.0, alpha_try * 1.3)
                stall = 0
                found = True
                break
        if not found:
            damping *= 0.5
            stall += 1
            if damping < 0.01 or stall > 30:
                # Perturb and restart
                h = project_box_sum(best_h + rng.randn(n) * 0.03, target_sum)
                damping = 0.5
                stall = 0
    return best_h, best_c5

def grad_optimize(h, n, target_sum, deadline, rng, alpha_init=80.0):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    v = np.zeros(n)
    alpha = alpha_init
    lr = 0.008
    momentum = 0.75
    stall = 0
    while time.time() < deadline:
        g = smooth_max_grad(h, alpha)
        v = momentum * v - lr * g
        h_new = project_box_sum(h + v, target_sum)
        c5 = compute_c5(h_new)
        if c5 < best_c5 - 1e-12:
            best_h = h_new.copy()
            best_c5 = c5
            h = h_new
            stall = 0
            if lr < 0.04:
                lr *= 1.015
            alpha = min(alpha * 1.008, 600.0)
        else:
            h = h_new
            stall += 1
            lr *= 0.96
            if stall > 40:
                h = best_h + rng.randn(n) * (0.02 + 0.01 * stall / 40.0)
                h = project_box_sum(h, target_sum)
                v = np.zeros(n)
                lr = 0.008
                alpha = max(60.0, alpha * 0.7)
                stall = 0
    return best_h, best_c5

def interpolate_h(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False) + 1.0 / n_new
    h_ext = np.append(h_old, h_old[0])
    x_ext = np.append(x_old, 2.0 + 1.0 / n_old)
    return np.interp(x_new, x_ext, h_ext)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        n = 200
        h0 = np.ones(n) * 0.5

    n = len(h0)
    target_sum = n / 2.0
    h = project_box_sum(h0, target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: LP-based optimization
    deadline1 = t0 + budget_s * 0.35
    best_h, best_c5 = lp_optimize(h, n, target_sum, deadline1, rng)
    n = len(best_h)
    target_sum = n / 2.0

    # Phase 2: Gradient-based refinement
    deadline2 = t0 + budget_s * 0.65
    best_h, best_c5 = grad_optimize(best_h, n, target_sum, deadline2, rng, alpha_init=120.0)

    # Phase 3: Try resolution refinement
    remaining = budget_s - (time.time() - t0)
    for n_try in [300, 400, 500]:
        if remaining < 80:
            break
        h_interp = interpolate_h(best_h, n, n_try)
        h_interp = project_box_sum(h_interp, n_try / 2.0)
        alloc = min(remaining * 0.4, 150.0)
        dl3 = time.time() + alloc
        h_opt, c5_opt = grad_optimize(h_interp, n_try, n_try / 2.0, dl3, rng, alpha_init=100.0)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
            n = n_try
        # Also try LP at new resolution
        remaining = budget_s - (time.time() - t0)
        if remaining > 60:
            dl4 = time.time() + min(remaining * 0.25, 100.0)
            h_lp, c5_lp = lp_optimize(best_h, n, n / 2.0, dl4, rng)
            if c5_lp < best_c5:
                best_h = h_lp
                best_c5 = c5_lp
        remaining = budget_s - (time.time() - t0)

    # Phase 4: Final polish
    remaining = budget_s - (time.time() - t0)
    if remaining > 20:
        h_final, c5_final = grad_optimize(best_h, n, n / 2.0,
                                          t0 + budget_s - 3, rng, alpha_init=200.0)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final

    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
