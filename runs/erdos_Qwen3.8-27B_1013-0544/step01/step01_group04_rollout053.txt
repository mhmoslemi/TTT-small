np.correlate(a, b, 'full')[m] for m = 0, ..., 2n-2:
= sum_i a[i] * b[i + m - (n-1)]

where a[i] and b[j] are 0 when the index is out of [0, n-1].

So result[m] = sum_i h[i] * (1 - h[i + m - (n-1)])

Let s = m - (n-1). Then result[m] = sum_i h[i] * (1-h[i+s])

∂result[m]/∂h[j]:
= sum_i [δ_{ij}(1-h[i+s]) + h[i]*(-δ_{i+s,j})]
= (1-h[j+s]) - h[j-s] (with h[x]=0 outside [0,n-1])

This confirms my formula. Let me proceed.

Now, for the penalty term: penalty * (sum(h) - n/2)^2
Gradient: 2 * penalty * (sum(h) - n/2) * ones

So the total gradient is:
softmax_grad + 2 * penalty * (sum(h) - n/2) * ones

import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time
from functools import partial

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_c5_vec(h, dx):
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    for _ in range(500):
        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - target
        if abs(diff) < 1e-16:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(free):
            h[free] -= diff / np.sum(free)
        else:
            h -= diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = np.sum(h) - target
    if abs(diff) > 1e-16:
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(free):
            h[free] -= diff / np.sum(free)
            h = np.clip(h, 0.0, 1.0)
    return h

def _cons(x, target):
    return np.sum(x) - target

def _obj_true_max(x, dx):
    return compute_c5(x, dx)

def _obj_sm(x, dx, tau, target, penalty):
    h = np.clip(x, 0.0, 1.0)
    c = compute_c5_vec(h, dx)
    f = tau * logsumexp(c / tau)
    f += penalty * (np.sum(h) - target) ** 2
    return f

def _grad_sm(x, dx, tau, target, penalty):
    h = np.clip(x, 0.0, 1.0)
    n = len(h)
    c = compute_c5_vec(h, dx)
    p = np.exp(c / tau - logsumexp(c / tau))
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        pm = p[m]
        if pm < 1e-18:
            continue
        s = m - (n - 1)
        j_lo = max(0, -s)
        j_hi = min(n, n - s)
        if j_hi > j_lo:
            grad[j_lo:j_hi] += pm * dx * (1.0 - h[j_lo + s:j_hi + s])
        j_lo2 = max(0, s)
        j_hi2 = min(n, n + s)
        if j_hi2 > j_lo2:
            grad[j_lo2:j_hi2] -= pm * dx * h[j_lo2 - s:j_hi2 - s]
    grad += 2.0 * penalty * (np.sum(h) - target)
    return grad

def _obj_weighted(x, dx, w, target, penalty):
    h = np.clip(x, 0.0, 1.0)
    c = compute_c5_vec(h, dx)
    f = np.dot(w, c)
    f += penalty * (np.sum(h) - target) ** 2
    return f

def _grad_weighted(x, dx, w, target, penalty):
    h = np.clip(x, 0.0, 1.0)
    n = len(h)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-18:
            continue
        s = m - (n - 1)
        j_lo = max(0, -s)
        j_hi = min(n, n - s)
        if j_hi > j_lo:
            grad[j_lo:j_hi] += wm * dx * (1.0 - h[j_lo + s:j_hi + s])
        j_lo2 = max(0, s)
        j_hi2 = min(n, n + s)
        if j_hi2 > j_lo2:
            grad[j_lo2:j_hi2] -= wm * dx * h[j_lo2 - s:j_hi2 - s]
    grad += 2.0 * penalty * (np.sum(h) - target)
    return grad

def slsqp_optimize(h_start, n, dx, target, max_iter=500):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': partial(_cons, target=target)}]
    res = minimize(partial(_obj_true_max, dx=dx), h_start, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-13})
    return project_feasible(res.x, n)

def lbfgsb_optimize(h_start, n, dx, target, tau, max_iter=500, penalty=50.0):
    bounds = [(0.0, 1.0)] * n
    res = minimize(partial(_obj_sm, dx=dx, tau=tau, target=target, penalty=penalty),
                  h_start, jac=partial(_grad_sm, dx=dx, tau=tau, target=target, penalty=penalty),
                  method='L-BFGS-B', bounds=bounds,
                  options={'maxiter': max_iter, 'ftol': 1e-14})
    return project_feasible(res.x, n)

def weighted_optimize(h_start, n, dx, target, w, max_iter=300, penalty=50.0):
    bounds = [(0.0, 1.0)] * n
    res = minimize(partial(_obj_weighted, dx=dx, w=w, target=target, penalty=penalty),
                  h_start, jac=partial(_grad_weighted, dx=dx, w=w, target=target, penalty=penalty),
                  method='L-BFGS-B', bounds=bounds,
                  options={'maxiter': max_iter, 'ftol': 1e-14})
    return project_feasible(res.x, n)

    start_time = time.time()

    # Phase 1: Multiplicative weights + L-BFGS-B (global exploration)

    n_lags = 2 * n - 1
    w = np.ones(n_lags) / n_lags

        for _ in range(3):

            c5_now = compute_c5(h_mw, dx)

            c_vec = compute_c5_vec(h_mw, dx)

    # Phase 2: SLSQP from best + perturbations
    deadline2 = start_time + budget_s * 0.75

        if attempt == 0:
            h_try = best_h.copy()
        elif attempt % 4 == 1:
            h_try = best_h + rng.randn(n) * 0.05
            h_try = project_feasible(h_try, n)
        elif attempt % 4 == 2:
            # Block flip
            bs = rng.randint(3, 15)
            st = rng.randint(0, n - bs)
            h_try = best_h.copy()
            h_try[st:st+bs] = 1.0 - h_try[st:st+bs]
            h_try = project_feasible(h_try, n)
        elif attempt % 4 == 3:
            # Smoothed perturbation
            h_try = best_h + rng.randn(n) * 0.08
            h_try = project_feasible(h_try, n)
        
        h_new = slsqp_optimize(h_try, n, dx, target, max_iter=400)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5 - 1e-14:
            best_h = h_new
            best_c5 = c5_new
        attempt += 1
    
    # Phase 3: Final precision polish
    deadline3 = start_time + budget_s - 10
    while elapsed() < deadline3:
        h_new = slsqp_optimize(best_h, n, dx, target, max_iter=1000)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5 - 1e-15:
            best_h = h_new
            best_c5 = c5_new
        else:
            break
    
    return (best_h, best_c5, n)
