def compute_c5(h, dx):

def fix_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total > 1e-12:
            h += room * (diff / total)
        else:
            h += diff / len(h)
    else:

        total = room.sum()

def _coord_subproblem(slopes, b):
    """Minimize max(slopes*t + b) over t in [0,1]. Returns optimal t."""
    # Coarse

    hi = tc[min(len(tc) - 1, ib + 1)]

- If i = j + s for some s, then corr[m] includes h[i]*h[j] = h[j+s]*h[j], and the change when both i and j change is delta_i * (1-h[j]) + delta_j * (1-h[i]) + ... 

This is getting complicated. For simplicity, let me just do the sequential update: first turn off i, update corr, then turn on j, update corr. This is exact.

```python
def binary_ls_fast(h, dx, n_steps):
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1 - h, mode="full").copy()
    c5 = float(np.max(corr) * dx)
    s_vals = np.arange(-(n - 1), n)
    
    for _ in range(n_steps):
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(_rng.choice(ones))
        j = int(_rng.choice(zeros))
        
        # Turn off i: delta = -1
        js = i + s_vals; jm = i - s_vals
        vs = (js >= 0) & (js < n); vm = (jm >= 0) & (jm < n)
        a = np.zeros(2 * n - 1)
        a[vs] += 1.0 - h[js[vs]]
        a[vm] -= h[jm[vm]]
        corr_new = corr - a  # h[i] decreases by 1
        
        # Now h[i] is effectively 0, turn on j: delta = +1
        # Recompute a for j with h[i]=0
        h_tmp_i = h[i]
        h[i] = 0.0
        js = j + s_vals; jm = j - s_vals
        vs = (js >= 0) & (js < n); vm = (jm >= 0) & (jm < n)
        a2 = np.zeros(2 * n - 1)
        a2[vs] += 1.0 - h[js[vs]]
        a2[vm] -= h[jm[vm]]
        h[i] = h_tmp_i  # restore
        corr_new2 = corr_new + a2  # h[j] increases by 1
        
        new_max = float(np.max(corr_new2) * dx)
        if new_max <= c5 + 1e-14:
            corr = corr_new2
            h[i] = 0.0
            h[j] = 1.0
            c5 = new_max
    
    return h, c5
```

Actually wait, I think I can simplify. The coefficient `a` for coordinate i is:
a[m] = sum of contributions of h[i] to corr[m]

When h[i] changes by delta, corr changes by a * delta.

So for turning off i (delta = -1): corr -= a_i
For turning on j (delta = +1): corr += a_j

But a_j depends on the current h values. After turning off i, h[i] = 0, so a_j might be different.

For the specific case where i and j are not "neighbors" (|i-j| is large), the a vectors don't interact, and we can compute:
delta_corr = -a_i + a_j (both computed with original h)

For nearby i and j, there's a small error. But for the purpose of accepting/rejecting moves, this approximation is usually fine.

import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        room = 1.0 - h
        total = room.sum()
        if total > 1e-12:
            h += room * (diff / total)
        else:
            h += diff / len(h)
    else:
        room = h.copy()
        total = room.sum()
        if total > 1e-12:
            h -= room * (-diff / total)
        else:
            h += diff / len(h)
    return np.clip(h, 0.0, 1.0)

def _sub1d(slopes, intercepts):
    tc = np.linspace(0.0, 1.0, 16)
    fc = np.max(slopes[:, None] * tc[None, :] + intercepts[:, None], axis=0)
    ib = int(np.argmin(fc))
    lo = tc[max(0, ib - 1)]
    hi = tc[min(len(tc) - 1, ib + 1)]
    tf = np.linspace(lo, hi, 64)
    ff = np.max(slopes[:, None] * tf[None, :] + intercepts[:, None], axis=0)
    return float(tf[int(np.argmin(ff))])

def _coeff(h, i, n, s_vals):
    js = i + s_vals
    jm = i - s_vals
    vs = (js >= 0) & (js < n)
    vm = (jm >= 0) & (jm < n)
    a = np.zeros(2 * n - 1)
    a[vs] += 1.0 - h[js[vs]]
    a[vm] -= h[jm[vm]]
    return a

def coord_desc(h, target_sum, n_sweeps, lam_rate=0.12):
    n = len(h)
    lam = 0.0
    corr = np.correlate(h, 1 - h, mode="full").copy()
    s_vals = np.arange(-(n - 1), n)
    for sweep in range(n_sweeps):
        order = np.arange(n)
        _rng.shuffle(order)
        for i in order:
            old = h[i]
            a = _coeff(h, i, n, s_vals)
            b = corr - a * old
            t_new = _sub1d(a + lam, b)
            t_new = float(np.clip(t_new, 0.0, 1.0))
            if abs(t_new - old) > 1e-14:
                corr += a * (t_new - old)
                h[i] = t_new
        err = np.sum(h) - target_sum
        if abs(err) > 1e-8:
            lam = float(np.clip(lam + lam_rate * err / n, -20.0, 20.0))
    return h, lam

def binary_ls(h, dx, n_steps):
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1 - h, mode="full").copy()
    c5 = float(np.max(corr) * dx)
    s_vals = np.arange(-(n - 1), n)
    for _ in range(n_steps):
        oi = np.where(h == 1.0)[0]
        zi = np.where(h == 0.0)[0]
        if len(oi) == 0 or len(zi) == 0:
            break
        i = int(_rng.choice(oi))
        j = int(_rng.choice(zi))
        ai = _coeff(h, i, n, s_vals)
        aj = _coeff(h, j, n, s_vals)
        delta = aj - ai
        new_max = float(np.max(corr + delta) * dx)
        if new_max <= c5 + 1e-14:
            corr += delta
            h[i] = 0.0
            h[j] = 1.0
            c5 = new_max
    return h, c5
