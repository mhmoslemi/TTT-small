```python
import numpy as np

from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_obj(h, t, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    max_c = np.max(corr)
    shifted = t * (corr - max_c)
    return (max_c + (1.0 / t) * np.log(np.sum(np.exp(shifted)))) * dx

def smooth_max_grad(h, t, dx):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    max_c = np.max(corr)
    shifted = t * (corr - max_c)
    weights = np.exp(shifted)
    weights = weights / weights.sum()
    
    # wp[s] = weights[s + n - 1] for shift s = 0, 1, ..., n-1
    wp = weights[n-1:]
    # wn_rev[s'] for s' = 1, 2, ..., n-1: wn_rev[s'] = weights[n-1-s']
    # (shift s = -s')
    
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    
    # Term 1: for each s=0..n-1, add wp[s] * one_minus_h[s:] to grad[0:n-s]
    for s in range(n):
        if wp[s] > 1e-15:
            grad[:n-s] += wp[s] * one_minus_h[s:]
    
    # Term 2: for each s'=1..n-1, add wn_rev[s'] * one_minus_h[0:n-s'] to grad[s':n]
    for sp in range(1, n):
        w = weights[n-1-sp]
        if w > 1e-15:
            grad[sp:] += w * one_minus_h[:n-sp]
    
    # Term 3: for each s=0..n-1, subtract wp[s] * h[0:n-s] from grad[s:n]
    for s in range(n):
        if wp[s] > 1e-15:
            grad[s:] -= wp[s] * h[:n-s]
    
    # Term 4: for each s'=1..n-1, subtract wn_rev[s'] * h[s':n] from grad[0:n-s']
    for sp in range(1, n):
        w = weights[n-1-sp]
        if w > 1e-15:
            grad[:n-sp] -= w * h[sp:]
    
    return grad * dx

def fix_sum(h, target_sum):
    h = h.copy()
    diff = target_sum - np.sum(h)
    h += diff / len(h)
    return np.clip(h, 0.0, 1.0)

def slsqp_with_grad(h, dx, target_sum, t=50, maxiter=500):
    n = len(h)
    
    def obj(x):
        return smooth_max_obj(x, t, dx)
    
    def grad(x):
        return smooth_max_grad(x, t, dx)
    
    try:
        res = minimize(
            obj, h.copy(), jac=grad,
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                         'jac': lambda x: np.ones(n)}],
            options={'maxiter': maxiter, 'ftol': 1e-12}
        )
        h_new = fix_sum(res.x, target_sum)
        return h_new
    except Exception:
        return h
```

```python
def smooth_max_grad(h, t, dx):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    max_c = np.max(corr)
    shifted = t * (corr - max_c)
    w = np.exp(shifted)
    w = w / w.sum()
    
    grad = np.zeros(n)
    om_h = 1.0 - h
    
    for s in range(n):  # s = 0, 1, ..., n-1
        ws_pos = w[n-1-s]  # weight for shift +s
        if ws_pos > 1e-15:
            grad[:n-s] += ws_pos * om_h[s:]      # Term A
            grad[s:] -= ws_pos * h[:n-s]           # Term B
    
    for sp in range(1, n):  # s' = 1, 2, ..., n-1
        ws_neg = w[n-1+sp]  # weight for shift -s'
        if ws_neg > 1e-15:
            grad[sp:] += ws_neg * om_h[:n-sp]     # Term C
            grad[:n-sp] -= ws_neg * h[sp:]         # Term D
    
    return grad * dx
```

```python
import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_obj(h, t, dx, mu=0.0, target_sum=0.0):
    corr = np.correlate(h, 1 - h, mode="full")
    max_c = np.max(corr)
    shifted = t * (corr - max_c)
    val = (max_c + (1.0 / t) * np.log(np.sum(np.exp(shifted)))) * dx
    if mu > 0:
        val += mu * (np.sum(h) - target_sum) ** 2
    return val

def smooth_max_grad(h, t, dx, mu=0.0, target_sum=0.0):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    max_c = np.max(corr)
    shifted = t * (corr - max_c)
    w = np.exp(shifted)
    w = w / w.sum()
    grad = np.zeros(n)
    om_h = 1.0 - h
    for s in range(n):
        ws = w[n - 1 - s]
        if ws > 1e-15:
            grad[:n - s] += ws * om_h[s:]
            grad[s:] -= ws * h[:n - s]
    for sp in range(1, n):
        ws = w[n - 1 + sp]
        if ws > 1e-15:
            grad[sp:] += ws * om_h[:n - sp]
            grad[:n - sp] -= ws * h[sp:]
    grad *= dx
    if mu > 0:
        grad += 2.0 * mu * (np.sum(h) - target_sum)
    return grad

def fix_sum(h, target_sum):
    h = h.copy()
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        h += diff / len(h)
    return np.clip(h, 0.0, 1.0)

def slsqp_with_grad(h, dx, target_sum, t=50, maxiter=500):
    n = len(h)
    def obj(x):
        return smooth_max_obj(x, t, dx)
    def grad_fn(x):
        return smooth_max_grad(x, t, dx)
    def cons_fun(x):
        return np.sum(x) - target_sum
    def cons_jac(x):
        return np.ones(n)
    try:
        res = minimize(obj, h.copy(), jac=grad_fn, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return fix_sum(res.x, target_sum)
    except Exception:
        return h

def lbfgsb_with_penalty(h, dx, target_sum, t=50, mu=100.0, maxiter=2000):
    n = len(h)
    def obj(x):
        return smooth_max_obj(x, t, dx, mu=mu, target_sum=target_sum)
    def grad_fn(x):
        return smooth_max_grad(x, t, dx, mu=mu, target_sum=target_sum)
    try:
        res = minimize(obj, h.copy(), jac=grad_fn, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
        return fix_sum(res.x, target_sum)
    except Exception:
        return h

def alternating_lp(h, dx, target_sum, n_active=15, n_iters=80, time_limit=500):
    import cvxpy as cp
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    t0 = time.time()
    for it in range(n_iters):
        if time.time() - t0 > time_limit:
            break
        corr = np.correlate(best_h, 1 - best_h, mode="full")
        sorted_idx = np.argsort(corr)[::-1]
        active = sorted_idx[:n_active]
        A = np.zeros((len(active), n))
        for idx, m in enumerate(active):
            s = m - (n - 1)
            if s >= 0:
                A[idx, :n - s] = 1 - best_h[s:]
            else:
                A[idx, -s:] = 1 - best_h[:n + s]
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1,
                       cp.sum(h_var) == target_sum,
                       A @ h_var <= t_var]
        try:
            prob = cp.Problem(cp.Minimize(t_var), constraints)
            prob.solve(solver=cp.CBC)
        except Exception:
            try:
                prob.solve()
            except Exception:
                break
        if h_var.value is None:
            break
        new_h = np.clip(h_var.value, 0, 1)
        new_h = fix_sum(new_h, target_sum)
        new_c5 = compute_c5(new_h, dx)
        if new_c5 < best_c5 - 1e-9:
            best_h = new_h
            best_c5 = new_c5
        else:
            blended = 0.5 * best_h + 0.5 * new_h
            blended = fix_sum(blended, target_sum)
            blended_c5 = compute_c5(blended, dx)
            if blended_c5 < best_c5:
                best_h = blended
                best_c5 = blended_c5
            if n_active < 40:
                n_active += 5
            elif it > 10:
                break
    return best_h, best_c5

def gradient_descent_momentum(h, dx, target_sum, t=50, lr=0.001, n_iters=500, momentum=0.9):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    v = np.zeros(n)
    for it in range(n_iters):
        g = smooth_max_grad(h, t, dx)
        penalty = 50.0 * (np.sum(h) - target_sum)
        g += penalty
        v = momentum * v - lr * g
        h_new = h + v
        h_new = fix_sum(h_new, target_sum)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        h = h_new
        if it % 50 == 0:
            lr *= 0.8
    return best_h, best_c5

def project_to_n(h, new_n):
    old_n = len(h)
    x_old = np.linspace(0, 2, old_n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return fix_sum(h_new, new_n / 2.0)

    dx_final = 2.0 / best_n
    best_c5 = compute_c5(best_h, dx_final)
    return best_h, best_c5, best_n
```
