import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def t_objective(v):
    return v[-1]

def t_ineq(v):
    n = len(v) - 1
    h = v[:n]
    t = v[-1]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return t - corr

def t_eq_fun(v):
    n = len(v) - 1
    return np.sum(v[:n]) - n * 0.5

def t_eq_jac(v):
    n = len(v) - 1
    j = np.zeros(len(v))
    j[:n] = 1.0
    return j

def smooth_obj(h, temp):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    shifted = (corr - m) / temp
    max_shifted = np.max(shifted)
    return m + temp * (max_shifted + np.log(np.sum(np.exp(shifted - max_shifted))))

def sum_eq_fun(h):
    return np.sum(h) - len(h) * 0.5

def sum_eq_jac(h):
    return np.ones(len(h))

def project_h(h, target):
    h = h.copy()
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return np.clip(h, 0.0, 1.0)
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = np.sum(mask)
    if cnt > 0:
        h[mask] += diff / cnt
        np.clip(h, 0.0, 1.0, out=h)
        diff2 = target - np.sum(h)
        if abs(diff2) > 1e-10:
            if diff2 > 0:
                mask2 = h < 1.0
            else:
                mask2 = h > 0.0
            cnt2 = np.sum(mask2)
            if cnt2 > 0:
                h[mask2] += diff2 / cnt2
    return np.clip(h, 0.0, 1.0)

def interp_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.arange(n_old) / n_old
    x_new = np.arange(n_new) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_h(h_new, n_new * 0.5)

def optimize_t_var(h0, max_iter=500, ftol=1e-12, t_margin=0.003):
    n = len(h0)
    c5_cur = compute_c5(h0)
    v0 = np.append(h0.copy(), c5_cur + t_margin)
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.5)]
    cons = [
        {'type': 'ineq', 'fun': t_ineq},
        {'type': 'eq', 'fun': t_eq_fun, 'jac': t_eq_jac}
    ]
    res = minimize(t_objective, v0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': ftol})
    h_new = np.clip(res.x[:n], 0.0, 1.0)
    h_new = project_h(h_new, n * 0.5)
    return h_new, compute_c5(h_new)

def optimize_smooth(h0, temp, max_iter=300, ftol=1e-13):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac}]
    res = minimize(smooth_obj, h0.copy(), args=(temp,), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': ftol})
    h_new = np.clip(res.x, 0.0, 1.0)
    h_new = project_h(h_new, n * 0.5)
    return h_new, compute_c5(h_new)

def optimize_resolution(h_best, n_orig, n_new, budget_check):
    h_interp = interp_h(h_best, n_new)
    c5_cur = compute_c5(h_interp)
    h_cur = h_interp.copy()
    if budget_check() < 200:
        return None, None
    h_new, c5_new = optimize_t_var(h_cur, max_iter=500, ftol=1e-12)
    if c5_new < c5_cur:
        h_cur, c5_cur = h_new, c5_new
    for temp in [0.02, 0.01, 0.005, 0.002, 0.001]:
        if budget_check() < 120:
            break
        h_new, c5_new = optimize_smooth(h_cur, temp, max_iter=250, ftol=1e-13)
        if c5_new < c5_cur:
            h_cur, c5_cur = h_new, c5_new
    if budget_check() > 100:
        h_new, c5_new = optimize_t_var(h_cur, max_iter=600, ftol=1e-13)
        if c5_new < c5_cur:
            h_cur, c5_cur = h_new, c5_new
    if n_new != n_orig:
        h_back = interp_h(h_cur, n_orig)
        c5_back = compute_c5(h_back)
        return h_back, c5_back
    return h_cur, c5_cur
