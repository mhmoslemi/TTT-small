from scipy.optimize import linprog

def solve_direction_lp(h, active_indices, n):
    dx = 2.0 / n
    target = n / 2.0
    c_all = compute_corr(h)  # unscaled
    
    # Build LP
    # Variables: y_0,...,y_{n-1}, t
    # Minimize: t
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize t
    
    A_ub_rows = []
    b_ub_vals = []
    
    for m in active_indices:
        k = m - (n - 1)
        g_k = corr_grad(h, k)  # gradient of c_k (unscaled)
        A_k = dx * g_k
        b_k = dx * c_all[m] - A_k @ h
        # Constraint: A_k · y + b_k ≤ t → A_k · y - t ≤ -b_k
        row = np.concatenate([A_k, [-1.0]])
        A_ub_rows.append(row)
        b_ub_vals.append(-b_k)
    
    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub_vals)
    
    # Equality: sum(y) = target
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    bounds = [(0, 1)] * n + [(None, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
