import numpy as np

from scipy.optimize import minimize

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_obj(h, t, dx):
    corr = np.correlate(h, 1 - h, mode="full")
    max_c = np.max(corr)
    shifted = t * (corr - max_c)
    return (max_c + (1.0 / t) * np.log(np.sum(np.exp(shifted)))) * dx

def sum_constraint(h, target_sum):
    return np.sum(h) - target_sum

def fix_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-14:
        h = h + diff / len(h)
    return np.clip(h, 0.0, 1.0)

def slsqp_optimize(h, dx, target_sum, t=50, maxiter=500):
    n = len(h)
    try:
        res = minimize(smooth_obj, h.copy(), args=(t, dx), method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': sum_constraint, 'args': (target_sum,)}],
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return fix_sum(res.x, target_sum)
    except Exception:
        return h

def round_to_binary(h, target_ones):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones != target_ones:
        if n_ones > target_ones:
            idx = np.where(h_bin == 1)[0]
            order = np.argsort(h[idx])
            h_bin[idx[order[:n_ones - target_ones]]] = 0
        else:
            idx = np.where(h_bin == 0)[0]
            order = np.argsort(-h[idx])
            h_bin[idx[order[:target_ones - n_ones]]] = 1
    return h_bin

    rng = np.random.RandomState(seed)
    n = len(h)
    best_c5 = compute_c5(h, dx)

        j = zeros[rng.randint(len(zeros))]

        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(t, 1e-8)):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
        else:
            h[i], h[j] = h[j], h[i]
    return h, best_c5
