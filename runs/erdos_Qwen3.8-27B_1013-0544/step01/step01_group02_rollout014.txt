            T = max(T * 0.96, T_floor)
        if it < 40:
            sc = rng.uniform(0.15, 0.5)
        elif it < 100:
            sc = rng.uniform(0.05, 0.25)
        elif it < 200:
            sc = rng.uniform(0.02, 0.12)
        else:
            sc = rng.uniform(0.01, 0.06)
        hp = project_feasible(best_h + rng.normal(0, sc, n), n)
        a = int(3000 + 7000 * min(it / 80, 2.5))
        h, c = local_opt(hp, n, a, 1500)
        if c < best_c - 1e-15:
            best_h, best_c = h, c
            stall = 0
        else:
            stall += 1
            if T > T_floor and rng.random() < np.exp(-(c - best_c) / T):
                best_h = h.copy()
        if stall > 80:
            T = max(T * 1.5, 0.003)
            stall = 0

    # Phase 3: Re-refine
    for a in [5000, 20000, 50000]:
        if budget_s - elapsed() < 10:
            break
        h, c = local_opt(best_h, n, a, 5000)
        if c < best_c:
            best_h, best_c = h, c

    # Phase 4: Multi-start
    if budget_s - elapsed() > 80:
        x = np.linspace(0, 2, n)
        inits = [init_sine1(x, n), init_sine2(x, n), init_step(x, n), init_block(x, n)]
        inits.append(rng.uniform(0, 1, n))
        inits.append(rng.uniform(0, 1, n))
        for hi0 in inits:
            if budget_s - elapsed() < 20:
                break
            hi0 = project_feasible(hi0, n)
            h, c = local_opt(hi0, n, 5000, 2000)
            if c < best_c:
                best_h, best_c = h, c

    # Phase 5: Fine basin hopping
    while budget_s - elapsed() > 50:
        sc = rng.uniform(0.003, 0.03)
        hp = project_feasible(best_h + rng.normal(0, sc, n), n)
        h, c = local_opt(hp, n, 20000, 2000)
        if c < best_c - 1e-15:
            best_h, best_c = h, c

    # Phase 6: Coordinate descent
    if budget_s - elapsed() > 15:
        deltas = [0.0005, 0.001, 0.002, 0.003, 0.005, 0.01]
        cnt = 0
        while budget_s - elapsed() > 8 and cnt < 60000:
            cnt += 1
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            for d in deltas:
                ok = False
                for sgn in [1, -1]:
                    hi_val = best_h[i] + sgn * d
                    hj_val = best_h[j] - sgn * d
                    if 0.0 <= hi_val <= 1.0 and 0.0 <= hj_val <= 1.0:
                        t = best_h.copy()
                        t[i] = hi_val
                        t[j] = hj_val
                        c = compute_c5(t, n)
                        if c < best_c - 1e-16:
                            best_h = t
                            best_c = c
                            ok = True
                            break
                if ok:
                    break

    # Phase 7: Final polish
    if budget_s - elapsed() > 5:
        h, c = local_opt(best_h, n, 50000, 5000)
        if c < best_c:
            best_h, best_c = h, c
    if budget_s - elapsed() > 3:
        h, c = local_opt(best_h, n, 100000, 3000)
        if c < best_c:
            best_h, best_c = h, c

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
