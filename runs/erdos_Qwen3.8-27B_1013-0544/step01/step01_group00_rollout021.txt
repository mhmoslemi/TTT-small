            grad[ell:] -= w * h[:n-ell] * dx

    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - t0)

    h = np.array(initial_h_values, dtype=float)
    n = len(h)
    h = project_to_set(h, n * 0.5)
    c5 = compute_c5(h)
    h_best = h.copy()
    c5_best = c5

    # Phase 1: Coarse optimization at n=90
    if time_left() > 400:
        n_c = 90
        h_c = interp_between(h, n, n_c)
        h_c = project_to_set(h_c, n_c * 0.5)
        c5_c = compute_c5(h_c)
        for temp, mu in [(0.05, 30.0), (0.02, 80.0), (0.008, 150.0), (0.003, 300.0)]:
            if time_left() < 200:
                break
            h_c, c5_c2 = lse_optimize(h_c, temp, mu, maxiter=600)
            if c5_c2 < c5_c:
                h_c, c5_c = h_c, c5_c2
        # Interpolate back to n
        h_up = interp_between(h_c, n_c, n)
        h_up = project_to_set(h_up, n * 0.5)
        c5_up = compute_c5(h_up)
        if c5_up < c5_best:
            h_best = h_up.copy()
            c5_best = c5_up
            h = h_up
            c5 = c5_up

    # Phase 2: Fine L-BFGS-B optimization at n=180
    temps_fine = [0.03, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0003]
    for temp in temps_fine:
        if time_left() < 70:
            break
        mu = 200.0
        h_new, c5_new = lse_optimize(h, temp, mu, maxiter=1000)
        if c5_new < c5_best:
            h = h_new
            c5 = c5_new
            h_best = h.copy()
            c5_best = c5_new

    # Phase 3: Subgradient polishing with multiple scales
    if time_left() > 120:
        n_iter = 3000
        for i in range(n_iter):
            if time_left() < 30:
                break
            lr = 0.03 * (0.9995 ** i)
            h_sg, c5_sg = subgrad_step(h, lr)
            if c5_sg < c5_best - 1e-6:
                h = h_sg
                c5 = c5_sg
                h_best = h.copy()
                c5_best = c5_sg
        # Second pass with smaller steps
        if time_left() > 60:
            for i in range(2000):
                if time_left() < 20:
                    break
                lr = 0.01 * (0.9998 ** i)
                h_sg, c5_sg = subgrad_step(h, lr)
                if c5_sg < c5_best - 1e-7:
                    h = h_sg
                    c5 = c5_sg
                    h_best = h.copy()
                    c5_best = c5_sg

    # Phase 4: T-variable SLSQP polish
    if time_left() > 80:
        h_tp, c5_tp = polish_tvar(h_best, max_iter=800, ftol=1e-13)
        if c5_tp < c5_best:
            h_best = h_tp
            c5_best = c5_tp

    # Phase 5: Another L-BFGS-B round from polished solution
    if time_left() > 100:
        for temp in [0.002, 0.001, 0.0005, 0.0003]:
            if time_left() < 50:
                break
            h_new, c5_new = lse_optimize(h_best, temp, 500.0, maxiter=800)
            if c5_new < c5_best:
                h_best = h_new
                c5_best = c5_new
        if time_left() > 60:
            h_tp, c5_tp = polish_tvar(h_best, max_iter=600, ftol=1e-13)
            if c5_tp < c5_best:
                h_best = h_tp
                c5_best = c5_tp

    # Phase 6: Random perturbations from best
    if time_left() > 150:
        for trial in range(5):
            if time_left() < 80:
                break
            scale = 0.03
            h_p = h_best + rng.randn(n) * scale
            h_p = project_to_set(h_p, n * 0.5)
            h_new, c5_new = lse_optimize(h_p, 0.01, 200.0, maxiter=500)
            if c5_new < c5_best and time_left() > 40:
                h_tp2, c5_tp2 = polish_tvar(h_new, max_iter=400, ftol=1e-12)
                if c5_tp2 < c5_best:
                    h_best = h_tp2
                    c5_best = c5_tp2

    # Final: ensure constraints
    h_best = project_to_set(h_best, len(h_best) * 0.5)
    c5_final = compute_c5(h_best)
    return h_best, c5_final, len(h_best)
