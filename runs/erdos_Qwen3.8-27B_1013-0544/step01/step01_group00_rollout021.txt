import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))


def project_h(h, target):
    h = h.copy()
    for _ in range(30):
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        else:
            break
    return np.clip(h, 0.0, 1.0)


def smooth_c5_and_grad(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    shifted = corr - m
    exp_vals = np.exp(shifted / temp)
    logsum = m + temp * np.log(np.sum(exp_vals))
    weights = exp_vals / np.sum(exp_vals)
    lags = np.arange(-(n - 1), n)
    j = np.arange(n)
    idx1 = j[:, None] + lags[None, :]
    idx2 = j[:, None] - lags[None, :]
    valid1 = (idx1 >= 0) & (idx1 < n)
    valid2 = (idx2 >= 0) & (idx2 < n)
    first = np.where(valid1, weights[None, :] * (1.0 - h)[np.clip(idx1, 0, n - 1)], 0.0).sum(axis=1)
    second = np.where(valid2, weights[None, :] * h[np.clip(idx2, 0, n - 1)], 0.0).sum(axis=1)
    grad = dx * (first - second)
    return logsum, grad


def lbfgsb_smooth(h0, target, temp, max_iter=500, ftol=1e-14):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n

    def obj_fn(x):
        val, _ = smooth_c5_and_grad(x, temp)
        penalty = 1e5 * (np.sum(x) - target) ** 2
        return val + penalty

    def jac_fn(x):
        _, g = smooth_c5_and_grad(x, temp)
        g = g + 2e5 * (np.sum(x) - target)
        return g

    res = minimize(obj_fn, h0.copy(), jac=jac_fn, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': max_iter, 'ftol': ftol, 'gtol': 1e-8})
    h_new = project_h(res.x, target)
    return h_new, compute_c5(h_new)


def slqp_t_var(h0, max_iter=400, ftol=1e-11):
    n = len(h0)
    c5_cur = compute_c5(h0)
    t0 = c5_cur + 0.01
    v0 = np.append(h0.copy(), t0)

    def t_obj(v):
        return v[-1]

    def t_ineq(v):
        h = v[:n]
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        return v[-1] - corr

    def t_eq(v):
        return np.sum(v[:n]) - n * 0.5

    def t_eq_jac(v):
        jj = np.zeros(n + 1)
        jj[:n] = 1.0
        return jj

    bounds = [(0.0, 1.0)] * n + [(0.0, 1.5)]
    cons = [{'type': 'ineq', 'fun': t_ineq},
            {'type': 'eq', 'fun': t_eq, 'jac': t_eq_jac}]
    res = minimize(t_obj, v0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': ftol})
    h_new = project_h(res.x[:n], n * 0.5)
    return h_new, compute_c5(h_new)


def interp_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.arange(n_old, dtype=float)
    x_new = np.linspace(0, n_old - 1, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_h(h_new, n_new * 0.5)
