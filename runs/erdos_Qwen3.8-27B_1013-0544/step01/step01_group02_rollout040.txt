    return np.ones(n)

def optimize_n(h_init, n, rng, time_limit):
    deadline = time.time() + time_limit
    h_work = h_init.copy()
    cur_c5 = compute_c5(h_work, n)
    no_improve = 0
    best_local = h_work.copy()
    best_local_c5 = cur_c5
    use_random = rng.rand() < 0.5
    while time.time() < deadline - 2:
        if use_random:
            h_work = cl_cd_pass_random(h_work, n, rng, deadline)
        else:
            h_work = cl_cd_pass(h_work, n, deadline)
        h_work = project_feasible(h_work, n)
        new_c5 = compute_c5(h_work, n)
        if new_c5 < cur_c5 - 1e-14:
            cur_c5 = new_c5
            no_improve = 0
            if new_c5 < best_local_c5:
                best_local_c5 = new_c5
                best_local = h_work.copy()
        else:
            no_improve += 1
        if no_improve > (100 if n <= 120 else 50):
            scale = 0.02 + 0.03 * min(no_improve / 200.0, 1.0)
            h_work = best_local + rng.normal(0, scale, n)
            h_work = project_feasible(h_work, n)
            cur_c5 = compute_c5(h_work, n)
            no_improve = 0
            use_random = rng.rand() < 0.5
    return best_local, best_local_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h0)
    except Exception:
        n_parent = 80
        h0 = np.full(n_parent, 0.5)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h, n_parent)
    best_n = n_parent

    # Priority: parent n first, then explore others
    n_list = [n_parent]
    for cand in [40, 60, 100, 120, 160, 200, 240, 320, 400]:
        if cand != n_parent and 20 <= cand <= 512:
            n_list.append(cand)

    # Allocate time: 50% to parent, rest split among others
    remaining = budget_s - (time.time() - t0)
    time_parent = remaining * 0.45
    time_others = remaining * 0.45
    time_slsqp = remaining * 0.10
    n_others = [x for x in n_list if x != n_parent]
    time_per_other = time_others / max(len(n_others), 1)

    # Phase 1: Optimize parent n with multiple starts
    inits = [h0.copy()]
    for _ in range(3):
        h_r = h0 + rng.normal(0, 0.05, n_parent)
        inits.append(project_feasible(h_r, n_parent))
    # Add a structured init
    x = np.linspace(0, 2, n_parent)
    h_struct = 0.5 + 0.3 * np.sin(2 * np.pi * x)
    inits.append(project_feasible(h_struct, n_parent))

    for i, h_init in enumerate(inits):
        t_left = time_parent - (time.time() - t0 - (budget_s - remaining))
        if t_left < 5:
            break
        h_opt, c_opt = optimize_n(h_init, n_parent, rng, t_left)
        if c_opt < best_c5 - 1e-14:
            best_c5 = c_opt
            best_h = h_opt.copy()
            best_n = n_parent

    # Phase 2: Try other n values
    for n in n_others:
        elapsed = time.time() - t0
        if elapsed > budget_s - time_slsqp - 10:
            break
        t_alloc = min(time_per_other, budget_s - time_slsqp - 10 - elapsed)
        if t_alloc < 5:
            continue
        # Interpolate from best solution
        x_best = np.linspace(0, 2, best_n)
        x_new = np.linspace(0, 2, n)
        h_interp = np.interp(x_new, x_best, best_h)
        h_interp = project_feasible(np.clip(h_interp, 0, 1), n)
        inits2 = [h_interp]
        h_rand = project_feasible(rng.random(n), n)
        inits2.append(h_rand)
        # Alternating pattern
        h_alt = np.zeros(n)
        h_alt[::2] = 1.0
        if n % 2 == 1:
            h_alt[-1] = 0.0
        h_alt = project_feasible(h_alt, n)
        inits2.append(h_alt)

        for h_init in inits2[:2]:
            if time.time() > budget_s - time_slsqp - 5:
                break
            h_opt, c_opt = optimize_n(h_init, n, rng, budget_s - time_slsqp - 5 - time.time())
            if c_opt < best_c5 - 1e-14:
                best_c5 = c_opt
                best_h = h_opt.copy()
                best_n = n

    # Phase 3: SLSQP polish
    if budget_s - (time.time() - t0) > 15:
        n = best_n
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
        for alpha in [2000, 5000, 10000, 20000]:
            if budget_s - (time.time() - t0) < 10:
                break
            try:
                res = minimize(obj_smooth, best_h, args=(n, alpha), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 3000, 'ftol': 1e-18})
                h_try = project_feasible(res.x, n)
                c = compute_c5(h_try, n)
                if c < best_c5 - 1e-14:
                    best_h, best_c5 = h_try, c
            except Exception:
                pass

    # Phase 4: Final CD passes
    if budget_s - (time.time() - t0) > 5:
        deadline = t0 + budget_s - 2
        for _ in range(200):
            if time.time() > deadline:
                break
            best_h = cl_cd_pass(best_h, best_n, deadline)
            best_h = project_feasible(best_h, best_n)

    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)
    return best_h, best_c5, best_n
