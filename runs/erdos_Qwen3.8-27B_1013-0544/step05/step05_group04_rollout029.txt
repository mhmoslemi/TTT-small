import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 300:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _sum_con(x, n):
    return float(x.sum() - n / 2.0)

def _sum_jac(x, n):
    return np.ones(n)

def run_slsqp(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_jac, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def grad_guided_search(h, n, deadline, max_sweeps=500):
    best_c = compute_c5(h, n)
    h = h.copy()
    dx = 2.0 / n
    # Adaptive delta schedule
    deltas = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5]
    di = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        d = deltas[min(di, len(deltas) - 1)]
        improved = False
        n_imp = 0
        # Use gradient to prioritize: compute corr and find active lags
        corr = np.correlate(h, 1 - h, mode='full') * dx
        cmax = corr.max()
        # Active lags are those within 0.005 of max
        active = np.where(corr > cmax - 0.005)[0]
        
        for i in range(0, n, 1):
            if time.time() > deadline:
                return h, best_c
            hi_val = h[i]
            for j in range(i + 1, n):
                hj_val = h[j]
                # Try h[i]+=d, h[j]-=d
                if hi_val + d <= 1.0 + 1e-12 and hj_val - d >= -1e-12:
                    h[i] = hi_val + d
                    h[j] = hj_val - d
                    c = compute_c5(h, n)
                    if c < best_c - 1e-15:
                        best_c = c
                        improved = True
                        n_imp += 1
                    else:
                        h[i] = hi_val
                        h[j] = hj_val
                # Try h[i]-=d, h[j]+=d

                    h[i] = hi_val - d
                    h[j] = hj_val + d
                    c = compute_c5(h, n)

                        improved = True
