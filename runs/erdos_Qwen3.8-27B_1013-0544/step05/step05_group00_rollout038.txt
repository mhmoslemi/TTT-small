def compute_c5(h, n):

    c = np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_optimize(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    def cons(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': 1e-10})
        return project_feasible(res.x, n)
    except Exception:
        return h

    lags = np.arange(-(n-1), n)

    """Greedy local search: for each zero, find best swap and do it."""
    target = n // 2

            best_j = -1
            for j in ones:

                new_max = (corr + delta).max() * dx
                if new_max < best_gain:

                improved = True

                    best_c = cur_c
                    best_h = h.copy()
        if not improved:
            break
    return best_h, best_c

  ΔfL[m] at index k=m-i: increases by mask_L[m, m-i] (if valid)
  
  So Δg[m] at index (i-m): -mask_R[m, i-m]
  Δg[m] at index (m-i): -mask_L[m, m-i]

  Note: i-m in the lag array corresponds to position (i-m) + (n-1) in the 0-based index of the 2n-1 vector.
  And m-i corresponds to position (m-i) + (n-1).

Similarly for h[j] → 0 (decrease by 1):
  ΔfR[m] at index k=j-m: increases by mask_R[m, j-m] (goes from mask*(1-1)=0 to mask*(1-0)=mask)
  ΔfL[m] at index k=m-j: decreases by mask_L[m, m-j] (goes from mask*1=mask to mask*0=0)
  
  So Δg[m] at index (j-m): +mask_R[m, j-m]
  Δg[m] at index (m-j): +mask_L[m, m-j]

Total Δg[m] from swap (i:0→1, j:1→0):
  At index (i-m)+(n-1): -mask_R[m, i-m]
  At index (m-i)+(n-1): -mask_L[m, m-i]  
  At index (j-m)+(n-1): +mask_R[m, j-m]
  At index (m-j)+(n-1): +mask_L[m, m-j]

Each g[m] gets at most 4 point updates! So updating all g values is O(n) (for each m, 4 point updates). Total O(4n) = O(n) per swap!

This makes the total cost per pass: O(n * n_ones * n) for the search part (vectorized) + O(n * n_swaps) for updates. If n_swaps ≤ n, total is O(n³) + O(n²) ≈ O(n³).

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_g(h, n):
    """Compute g[m] = fR[m] - fL[m] for all m.
    g[m, k] = mask_R[m,k]*(1-h[m+lags[k]]) - mask_L[m,k]*h[m-lags[k]]
    where lags = arange(-(n-1), n)
    """
    lags = np.arange(-(n-1), n)  # shape (2n-1,)
    # For each m, m+lags gives indices into h
    idx_r = np.arange(n)[:, None] + lags[None, :]  # shape (n, 2n-1)
    idx_l = np.arange(n)[:, None] - lags[None, :]  # shape (n, 2n-1)
    
    mask_r = (idx_r >= 0) & (idx_r < n)
    mask_l = (idx_l >= 0) & (idx_l < n)
    
    # Clamp for safe indexing
    idx_r_clamped = np.clip(idx_r, 0, n-1)
    idx_l_clamped = np.clip(idx_l, 0, n-1)
    
    h_r = h[idx_r_clamped]  # shape (n, 2n-1)
    h_l = h[idx_l_clamped]
    
    fR = mask_r.astype(float) * (1.0 - h_r)
    fL = mask_l.astype(float) * h_l
    
    return fR - fL  # shape (n, 2n-1)

def update_g_swap(g, h, n, i, j):
    """Update g after h[i]: 0→1 and h[j]: 1→0.
    Returns updated g (in place) and updated h.
    """
    lags = np.arange(-(n-1), n)
    # For h[i] increasing by 1:
    #   g[m] at position where m+lags[k]=i, i.e., lags[k]=i-m, k = i-m+(n-1):
    #     fR[m,k] decreases by mask_R[m,k] (since 1-h goes from 1 to 0)
    #     So g[m,k] decreases by mask_R[m,k]
    #   g[m] at position where m-lags[k]=i, i.e., lags[k]=m-i, k = m-i+(n-1):
    #     fL[m,k] increases by mask_L[m,k] (since h goes from 0 to 1)
    #     So g[m,k] decreases by mask_L[m,k]
    # For h[j] decreasing by 1:
    #   g[m] at position where m+lags[k]=j, k = j-m+(n-1):
    #     fR[m,k] increases by mask_R[m,k]
    #     So g[m,k] increases by mask_R[m,k]
    #   g[m] at position where m-lags[k]=j, k = m-j+(n-1):
    #     fL[m,k] decreases by mask_L[m,k]
    #     So g[m,k] increases by mask_L[m,k]
    
    ms = np.arange(n)
    # Position i change: for each m, k1 = i-m+(n-1) for fR, k2 = m-i+(n-1) for fL
    k1_r = i - ms + (n-1)  # position in g vector where fR is affected
    k2_l = ms - i + (n-1)  # position in g vector where fL is affected
    valid1_r = (k1_r >= 0) & (k1_r < 2*n-1)
    valid2_l = (k2_l >= 0) & (k2_l < 2*n-1)
    
    # mask_R[m, k1_r[m]] = 1 if 0 <= m + lags[k1_r[m]] < n
    # lags[k1_r[m]] = i-m, so m + (i-m) = i, which is always in [0,n) since 0<=i<n
    # So mask_R is always 1 when valid1_r!
    g[ms[valid1_r], k1_r[valid1_r]] -= 1.0
    
    # mask_L[m, k2_l[m]] = 1 if 0 <= m - lags[k2_l[m]] < n
    # lags[k2_l[m]] = m-i, so m - (m-i) = i, always valid
    g[ms[valid2_l], k2_l[valid2_l]] -= 1.0
    
    # Position j change:
    k3_r = j - ms + (n-1)
    k4_l = ms - j + (n-1)
    valid3_r = (k3_r >= 0) & (k3_r < 2*n-1)
    valid4_l = (k4_l >= 0) & (k4_l < 2*n-1)
    
    g[ms[valid3_r], k3_r[valid3_r]] += 1.0
    g[ms[valid4_l], k4_l[valid4_l]] += 1.0
    
    h[i] = 1.0
    h[j] = 0.0
    return g, h

def vectorized_greedy_binary(h, n, deadline, max_passes=5000):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    corr = np.correlate(h, 1-h, mode='full').copy()
    dx = 2.0/n
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    g = compute_g(h, n)
    
    for p in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        zeros = np.where(h < 0.5)[0]
        ones = np.where(h >= 0.5)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        
        for i in zeros:
            if time.time() > deadline:
                break
            # Vectorized: compute max of corr + g[i] - g[j] for all j in ones
            base = corr + g[i]  # shape (2n-1,)
            vals = (base[None, :] - g[ones]).max(axis=1) * dx  # shape (n_ones,)
            min_idx = np.argmin(vals)
            if vals[min_idx] < cur_c - 1e-14:
                j = ones[min_idx]
                # Do the swap
                delta = g[i] - g[j]
                corr = corr + delta
                g, h = update_g_swap(g, h, n, int(i), int(j))
                cur_c = vals[min_idx]
                improved = True
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
        
        if not improved:
            break
    
    return best_h, best_c

def slsqp_optimize(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    def cons(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cons}],
                       options={'maxiter': maxiter, 'ftol': 1e-10})
        return project_feasible(res.x, n)
    except Exception:
        return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_g(h, n):
    lags = np.arange(-(n-1), n)
    ms = np.arange(n)[:, None]
    idx_r = ms + lags[None, :]
    idx_l = ms - lags[None, :]
    mask_r = (idx_r >= 0) & (idx_r < n)
    mask_l = (idx_l >= 0) & (idx_l < n)
    ir = np.clip(idx_r, 0, n-1)
    il = np.clip(idx_l, 0, n-1)
    fR = mask_r.astype(float) * (1.0 - h[ir])
    fL = mask_l.astype(float) * h[il]
    return fR - fL

def update_g(g, h, n, i, j):

    nm1 = n - 1
    k1 = i - ms + nm1
    v1 = (k1 >= 0) & (k1 < 2*n-1)

    h[i] = 1.0
    h[j] = 0.0

def second_order_vec(n, di, dj):

    corr[nm1] += -2.0
    if di != 0:
        corr[nm1 + di] += 1.0
    if dj != 0:
        corr[nm1 + dj] += 1.0
    return corr

    corr = np.correlate(h, 1-h, mode='full').astype(float).copy()
    dx = 2.0 / n
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    g = compute_g(h, n)
    
    for p in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        zeros = np.where(h < 0.5)[0]
        ones = np.where(h >= 0.5)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        order = rng.permutation(len(zeros)) if use_sa else np.arange(len(zeros))
        for ii in order:
            if time.time() > deadline:
                break
            i = zeros[ii]
            base = corr + g[i]
            if len(ones) > 200:
                chunks = [np.arange(0, len(ones), 50), np.arange(50, len(ones), 50)]
                vals = np.concatenate([(base[None,:] - g[ones[c[0]:c[0]+50]]).max(axis=1)*dx for c in [chunks[0]] if len(ones[c[0]:c[0]+50])>0])
            else:
                vals = (base[None,:] - g[ones]).max(axis=1) * dx
            min_idx = np.argmin(vals)
            accept = vals[min_idx] < cur_c - 1e-14
            if not accept and use_sa:

                    accept = True

                di = int(j - i)
                dj = int(i - j)
                so = second_order_vec(n, di, dj)
                corr = corr + delta + so
                update_g(g, h, n, int(i), int(j))
                cur_c = corr.max() * dx
                improved = True
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
        if not improved:
            break
    return best_h, best_c

def slsqp_opt(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        gv = np.correlate(x, 1-x, mode='full') * dx
        m = gv.max()
        sh = alpha * (gv - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type':'eq','fun':cons_fun}],
                       options={'maxiter': maxiter, 'ftol': 1e-10})
        return project_feasible(res.x, n)
    except Exception:
        return h
