                if d > 1e-10:
                    best_h[i_s] -= d
                    best_h[i_d] += d
                    nc = compute_c5(best_h, n)
                    if nc < best_c:
                        best_c = nc
                        stall = 0
                    else:
                        best_h[i_s] += d
                        best_h[i_d] -= d
                        stall += 1
                else:
                    stall += 1
            step *= 0.7
            if stall > 300:
                break
    
    return best_h, best_c

        for alpha in [500, 2000, 10000, 50000, 200000]:

                hp = best_h.copy()
                hp[st:st+bs] += rng.normal(0, 0.02, bs)
            hp = project_feasible(hp, n)
            alloc = min(35, deadline - time.time() - 20)

            if c_opt < best_c:

    if best_h is not None and time.time() < deadline - 2:
        n = len(best_h)
        best_h, best_c = grad_local_search(best_h, n, top_k=3, deadline=deadline-1)

        best_c = compute_c5(best_h, n)

        best_h = project_feasible(np.full(n_init, 0.5), n_init)
