Actually, let me be more careful. np.correlate(h, 1-h, mode='full') gives output of length 2n-1. For output index m (0-indexed), the lag is m - (n-1). For lag s = m - (n-1):
- If s ≥ 0: sum over i from 0 to n-1-s of h[i] * (1-h[i+s])
- If s < 0: sum over i from -s to n-1 of h[i] * (1-h[i+s]) = sum over i from 0 to n-1-|s| of h[i+|s|] * (1-h[i])

In either case, the number of terms is n - |s|.

So c_k = (∑ over valid i of h[i]*(1-h[i+k])) * dx

This is bilinear in h, so the constraint c_k ≤ t is NOT linear. It's a quadratic constraint. So this is a QCQP (quadratically constrained quadratic program), which is NP-hard in general.

```python
def lse_obj_and_grad(x, n, alpha):
    """Compute objective and gradient for log-sum-exp smoothed max overlap."""
    h_x = np.clip(x, 0, 1)
    dx = 2.0 / n
    c = np.correlate(h_x, 1 - h_x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    exp_shifted = np.exp(shifted - mx)
    sum_exp = exp_shifted.sum()
    weights = exp_shifted / sum_exp  # softmax weights
    
    f = m + (mx + np.log(sum_exp)) / alpha
    
    # Gradient
    grad = np.zeros(n)
    for i in range(len(c)):
        if weights[i] < 1e-12:
            continue
        s = i - (n - 1)
        lo = max(0, -s)
        hi = min(n - 1, n - 1 - s)
        idx = np.arange(lo, hi + 1)
        grad[idx] += weights[i] * (1 - h_x[idx + s]) * dx
        lo2 = max(0, lo + s)
        hi2 = min(n - 1, hi + s)
        idx2 = np.arange(lo2, hi2 + 1)
        grad[idx2] -= weights[i] * h_x[idx2 - s] * dx
    
    return f, grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def get_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-16:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _lse_obj(x, n, alpha):
    h_x = np.clip(x, 0, 1)
    dx = 2.0 / n
    c = np.correlate(h_x, 1 - h_x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha

def _lse_grad(x, n, alpha):
    h_x = np.clip(x, 0, 1)
    dx = 2.0 / n
    c = np.correlate(h_x, 1 - h_x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    exp_s = np.exp(shifted - mx)
    sum_exp = exp_s.sum()
    weights = exp_s / sum_exp
    grad = np.zeros(n)
    for i in range(len(c)):
        w = weights[i]
        if w < 1e-10:
            continue
        s = i - (n - 1)
        lo = max(0, -s)
        hi = min(n - 1, n - 1 - s)
        idx = np.arange(lo, hi + 1)
        grad[idx] += w * (1.0 - h_x[idx + s]) * dx
        lo2 = max(0, lo + s)
        hi2 = min(n - 1, hi + s)
        idx2 = np.arange(lo2, hi2 + 1)
        grad[idx2] -= w * h_x[idx2 - s] * dx
    return grad

def _sum_fun(x, n):
    return x.sum() - n / 2.0

def _sum_jac(x, n):
    return np.ones(n)

def slsqp_with_grad(h, n, alpha, maxiter=10000):
    def obj(x):
        return _lse_obj(x, n, alpha)
    def grad(x):
        return _lse_grad(x, n, alpha)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac, 'args': (n,)}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def aggressive_polish(h, n, deadline):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    c = get_corr(h, n)
    h = h.copy()
    steps = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    si = 3
    stall = 0
    while time.time() < deadline:
        k_star = int(np.argmax(c))
        s = k_star - (n - 1)
        lo = max(0, -s)
        hi = min(n - 1, n - 1 - s)
        idx = np.arange(lo, hi + 1)
        g = np.zeros(n)
        g[idx] += (1.0 - h[idx + s])
        lo2 = max(0, lo + s)
        hi2 = min(n - 1, hi + s)
        idx2 = np.arange(lo2, hi2 + 1)
        g[idx2] -= h[idx2 - s]
        if g.max() < 1e-15 and -g.min() < 1e-15:
            break
        g_ranked = np.argsort(g)
        candidates = []
        for so, do in [(0,0),(1,0),(2,0),(0,1),(1,1),(3,0),(2,1),(4,0)]:
            i_src = int(g_ranked[n - 1 - so])
            i_dst = int(g_ranked[do])
            if i_src != i_dst:
                candidates.append((i_src, i_dst))
        improved = False
        for i_src, i_dst in candidates:
            d = steps[si]
            d = min(d, h[i_src] * 0.99, (1.0 - h[i_dst]) * 0.99)
            if d < 1e-14:
                continue
            h[i_src] -= d
            h[i_dst] += d
            new_c = get_corr(h, n)
            nm = float(new_c.max())
            if nm < best_c - 1e-15:
                best_c = nm
                best_h = h.copy()
                c = new_c
                improved = True
                break
            else:
                h[i_src] += d
                h[i_dst] -= d
        if improved:
            stall = 0
            si = max(0, si - 1)
        else:
            stall += 1
            si = min(si + 1, len(steps) - 1)
            if stall > 1000:
                break
    return best_h, best_c

def sa_global(h, n, deadline, seed=42):
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    T = 0.003
    T_min = 1e-9
    cooling = 0.9995
    accept = 0
    total = 0
    while time.time() < deadline:
        r = rng.random()
        if r < 0.35:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = rng.uniform(0.001, 0.04)
            if h[i] - d < 0 or h[j] + d > 1:
                continue
            h[i] -= d
            h[j] += d
        elif r < 0.65:
            L = rng.randint(3, min(15, n // 10))
            i = rng.randint(0, n - L)
            j = rng.randint(0, n - L)
            if abs(i - j) <= L:
                continue
            seg_i = h[i:i+L].copy()
            seg_j = h[j:j+L].copy()
            h[i:i+L] = seg_j
            h[j:j+L] = seg_i
        else:
            c = get_corr(h, n)
            k_star = int(np.argmax(c))
            s = k_star - (n - 1)
            lo = max(0, -s)
            hi = min(n - 1, n - 1 - s)
            idx = np.arange(lo, hi + 1)
            contrib = h[idx] * (1.0 - h[idx + s])
            top_pos = idx[np.argmax(contrib)]
            i = int(top_pos)
            j = rng.randint(n)
            if i == j:
                continue
            d = rng.uniform(0.001, 0.03)
            if h[i] - d < 0 or h[j] + d > 1:
                continue
            h[i] -= d
            h[j] += d
        new_c = compute_c5(h, n)
        total += 1
        if new_c < best_c:
            best_c = new_c
            best_h = h.copy()
            accept += 1
        else:
            delta = new_c - best_c
            if delta < T * 3 and rng.random() < np.exp(-delta / max(T, T_min)):
                accept += 1
            else:
                pass
        T = max(T * cooling, T_min)
        if total % 500 == 0:
            rate = accept / max(1, total)
            if rate < 0.03:
                T = max(T, 0.002)
            elif rate > 0.4:
                T *= 0.7
            accept = 0
            total = 0
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):

    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

            if time.time() < deadline - 15:

                    best_h = ht2
    # Phase 5: Final polish
    if time.time() < deadline - 5:
        best_h, best_c = aggressive_polish(best_h, n_init, deadline - 2)
    best_h = project_feasible(best_h, n_init)
    best_c = compute_c5(best_h, n_init)
    return best_h, best_c, n_init
```
