import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s_room = room.sum()
            if s_room < 1e-14:
                break
            h = h + diff / s_room * room
        else:
            s_h = h.sum()
            if s_h < 1e-14:
                break
            h = h + diff / s_h * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha, lam):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    val = m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    val += lam * (x.sum() - n / 2.0) ** 2
    return val

def smooth_grad(x, n, alpha, lam):
    dx = 2.0 / n
    c_raw = np.correlate(x, 1 - x, mode='full')
    m_star = int(np.argmax(c_raw))
    k = m_star - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - x[:n - k])
        g[:n - k] += -x[k:]
    else:
        g[:n + k] += (1.0 - x[-k:])
        g[-k:] += -x[:n + k]
    g *= dx
    g += 2.0 * lam * (x.sum() - n / 2.0)
    return g

def lbfgs_optimize(h, n, alpha, lam, maxiter, deadline):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_obj, h, method='L-BFGS-B', jac=smooth_grad,
                       args=(n, alpha, lam), bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
        return project_feasible(res.x, n)
    except Exception:
        return h

def coord_descent(h, n, rounds, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(int(time.time() * 1000) % 100000)
    for r in range(rounds):
        if time.time() > deadline:
            break
        order = rng.permutation(n)
        for i in order:
            if time.time() > deadline:
                break
            old = h[i]
            # Try several candidate values
            for v in [0.0, 1.0, 0.5, old + 0.1, old - 0.1, old + 0.05, old - 0.05]:
                v = np.clip(v, 0.0, 1.0)
                if abs(v - old) < 1e-10:
                    continue
                h[i] = v
                h = project_feasible(h, n)
                c = compute_c5(h, n)
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
                    break
            h = best_h.copy()  # reset to best if no improvement
    return best_h, best_c
