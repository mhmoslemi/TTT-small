        m = c.max()
        sh = alpha * (c - m)
        sm = sh.max()
        val = m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        sm = sh.max()
        val = m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
        val += 1e4 * (x.sum() - n/2.0)**2
        return val
    def grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m_star = int(np.argmax(c))
        k = m_star - (n-1)
        g = np.zeros(n)
        if k >= 0:
            g[k:] += (1.0 - x[:n-k])
            g[:n-k] += -x[k:]
        else:
            g[:n+k] += (1.0 - x[-k:])
            g[-k:] += -x[:n+k]
        g *= dx
        g += 2e4 * (x.sum() - n/2.0)
        return g
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=grad,
                      bounds=[(0.0,1.0)]*n,
                      options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
        return project(res.x, n)
    except:
        return h

        if abs(d) < 1e-12:
            break
        if d > 0:
            r = 1.0 - h
            s = r.sum()
            if s < 1e-14: break
            h += d / s * r
        else:
            s = h.sum()
            if s < 1e-14: break
            h += d / s * h
    return np.clip(h, 0.0, 1.0)

def binary_ls(h, n, deadline, rng, max_iter=400000):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    for it in range(max_iter):
        if time.time() > deadline:
            return best_h, best_c
        r = rng.random()
        if r < 0.45 and ones and zeros:
            i = zeros[rng.randint(len(zeros))]
            j = ones[rng.randint(len(ones))]
            h[i], h[j] = 1.0, 0.0
            cn = compute_c5(h, n)
            if cn < best_c - 1e-15:
                best_c, best_h = cn, h.copy()
                ones.remove(j); ones.append(i)
                zeros.remove(i); zeros.append(j)
            else:
                h[i], h[j] = 0.0, 1.0
        elif r < 0.65 and len(ones) >= 2 and len(zeros) >= 2:
            ia = rng.choice(len(zeros), 2, replace=False)
            ja = rng.choice(len(ones), 2, replace=False)
            a, b = zeros[ia[0]], zeros[ia[1]]
            c2, d2 = ones[ja[0]], ones[ja[1]]
            h[a], h[b], h[c2], h[d2] = 1., 1., 0., 0.
            cn = compute_c5(h, n)
            if cn < best_c - 1e-15:
                best_c, best_h = cn, h.copy()
                for x in (c2, d2): ones.remove(x)
                for x in (a, b): ones.append(x)
                for x in (a, b): zeros.remove(x)
                for x in (c2, d2): zeros.append(x)
            else:
                h[a], h[b], h[c2], h[d2] = 0., 0., 1., 1.
        else:
            lo, hi = sorted(rng.randint(0, n, 2))
            if hi - lo < 3:
                continue
            h[lo:hi+1] = h[lo:hi+1][::-1]
            cn = compute_c5(h, n)
            if cn < best_c - 1e-15:
                best_c, best_h = cn, h.copy()
                ones = np.where(h == 1)[0].tolist()
                zeros = np.where(h == 0)[0].tolist()
            else:
                h[lo:hi+1] = h[lo:hi+1][::-1]
    return best_h, best_c

def ils_binary(h, n, deadline, rng, n_restarts=None):
    if n_restarts is None:
        n_restarts = max(3, int((deadline - time.time()) / 60))
    best_h, best_c = h.copy(), compute_c5(h, n)
    per = (deadline - time.time()) / max(n_restarts, 1)
    for r in range(n_restarts):
        t_end = min(time.time() + per * 0.85, deadline - 2)
        if t_end <= time.time():
            break
        hb, cb = binary_ls(h.copy(), n, t_end, rng)
        if cb < best_c:
            best_c, best_h = cb, hb.copy()
        # Perturb for next iteration
        h = best_h.copy()
        nperturb = rng.randint(3, 10)
        o = np.where(h == 1)[0].tolist()
        z = np.where(h == 0)[0].tolist()
        for _ in range(nperturb):
            if not o or not z:
                break
            i = z[rng.randint(len(z))]
            j = o[rng.randint(len(o))]
            h[i], h[j] = 1.0, 0.0
            o.remove(j); o.append(i)
            z.remove(i); z.append(j)
    return best_h, best_c
