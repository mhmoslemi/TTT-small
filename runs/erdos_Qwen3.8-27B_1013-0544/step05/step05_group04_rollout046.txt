import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def get_all_c(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _sum_c(x, n):
    return float(x.sum() - n / 2.0)

def _sum_j(x, n):
    return np.ones(n)

def slsqp_step(h, n, alpha, maxiter=1000):

    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]
    try:

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except:
        return h.copy()

    idx2 = np.clip(i_idx - s_val, 0, n-1)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def get_all_c(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _sum_c(x, n):
    return float(x.sum() - n / 2.0)

def _sum_j(x, n):
    return np.ones(n)

def slsqp_step(h, n, alpha, maxiter=1000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def compute_grad(h, n):
    i_idx = np.arange(n)[:, None]
    s_val = np.arange(-(n - 1), n)[None, :]
    idx1 = i_idx + s_val
    valid1 = (idx1 >= 0) & (idx1 < n)
    term1 = np.where(valid1, 1.0 - h[np.clip(idx1, 0, n - 1)], 0.0)
    idx2 = i_idx - s_val
    valid2 = (idx2 >= 0) & (idx2 < n)
    term2 = np.where(valid2, h[np.clip(idx2, 0, n - 1)], 0.0)
    return term1 - term2

def vectorized_search(h, n, delta, deadline, max_sweeps=100):
    dx = 2.0 / n
    C = get_all_c(h, n)
    best_c = float(C.max())
    grad = compute_grad(h, n)
    stalled = 0
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        best_improve = 1e-15
        best_move = None
        for i in range(n):
            if h[i] + delta <= 1.0:
                diff = grad[i, :] - grad
                new_max = (C[None, :] + dx * delta * diff).max(axis=1)
                valid = h >= delta
                if valid.any():
                    val = np.where(valid, new_max, np.inf)
                    j = int(val.argmin())
                    if val[j] < best_c - 1e-13:
                        imp = best_c - val[j]
                        if imp > best_improve:
                            best_improve = imp
                            best_move = (i, j, delta)
            if h[i] - delta >= 0.0:
                diff = grad[i, :] - grad
                new_max = (C[None, :] - dx * delta * diff).max(axis=1)
                valid = h <= 1.0 - delta
                if valid.any():
                    val = np.where(valid, new_max, np.inf)
                    j = int(val.argmin())
                    if val[j] < best_c - 1e-13:
                        imp = best_c - val[j]
                        if imp > best_improve:
                            best_improve = imp
                            best_move = (i, j, -delta)
        if best_move is None:
            stalled += 1
            if stalled >= 3:
                break
            continue
        stalled = 0
        i, j, dm = best_move
        h[i] += dm
        h[j] -= dm
        C = get_all_c(h, n)
        best_c = float(C.max())
        grad = compute_grad(h, n)
    return h, best_c

def optimize_at_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)
    for alpha in [5000, 50000, 200000]:
        if time.time() > deadline - 30:
            break
        ht = slsqp_step(h, n, alpha, maxiter=800)
        c = compute_c5(ht, n)
        if c < best_c - 1e-12:
            best_c = c
            h = ht
    deltas = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    for d in deltas:
        if time.time() > deadline - 15:
            break
        alloc = min(60, (deadline - time.time()) * 0.25)
        if alloc < 10:
            break
        h, best_c = vectorized_search(h, n, d, time.time() + alloc,
                                      max_sweeps=80)
    if time.time() < deadline - 10:
        ht = slsqp_step(h, n, 500000, maxiter=1500)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            h = ht
        if time.time() < deadline - 5:
            h, best_c = vectorized_search(h, n, 0.0001, deadline - 2,
                                          max_sweeps=60)
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    best_h = project_feasible(best_h, len(best_h))
