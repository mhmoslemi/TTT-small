import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def make_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    nb = np.zeros(n)
    nb[idx[:target]] = 1.0
    return nb

def binary_c5_delta(h, n, p, q):
    """Compute change in max C5 when swapping h[p]=1->0 and h[q]=0->1"""
    g = np.correlate(h, 1.0 - h, mode='full')
    cur_max = g.max()
    # After swap
    h[p], h[q] = 1.0, 0.0
    g_new = np.correlate(h, 1.0 - h, mode='full')
    h[p], h[q] = 0.0, 1.0
    return g_new.max() - cur_max, g_new.max(), cur_max

Wait, I need to be more careful:
g_l = sum_i h[i] * (1 - h[i+s_l]) where s_l = l - (n-1)

∂g_l/∂h[i] comes from two sources:
1. h[i] appears directly: term is h[i]*(1-h[i+s_l]), derivative w.r.t. h[i] is (1-h[i+s_l]) [if i+s_l is valid]
2. h[i] appears in h[i'+s_l] for i' = i-s_l: term is h[i-s_l]*(1-h[i]), derivative w.r.t. h[i] is -h[i-s_l] [if i-s_l is valid]

So: ∂g_l/∂h[i] = (1-h[i+s_l]) * [0≤i+s_l<n] - h[i-s_l] * [0≤i-s_l<n]

This is O(1) per (l, i) pair, O(n) per l, O(n²) total for the full gradient matrix. For n=250, that's 62500 operations, very fast.
