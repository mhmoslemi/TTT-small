    d_max = min(1.0 - h[p], h[q])
    if d_min >= d_max:
        return 0.0
    
    # Find minimum of f(d) = max_k (c_arr[k] + A[k]*d) on [d_min, d_max]
    # Using the "walk along the upper envelope" method
    
    def f_at(d):
        return float(np.max(c_arr + A * d))
    
    # Start at d=0 (clamped to [d_min, d_max])
    d = max(d_min, min(0.0, d_max))
    
    for _ in range(n):  # at most n steps to walk through all facets
        c_new = c_arr + A * d
        k_max = int(np.argmax(c_new))
        a_k = A[k_max]
        
        if abs(a_k) < 1e-15:
            break  # at minimum (flat facet)

            # Moving right (increasing d) will decrease f
            # Find next breakpoint to the right
            # c_arr[j] + A[j]*d = c_arr[k_max] + A[k_max]*d
            # d_next = (c_arr[j] - c_arr[k_max]) / (A[k_max] - A[j])
            diffs = c_arr - c_arr[k_max]  # c_arr[j] - c_arr[k_max]
            denom = A[k_max] - A  # A[k_max] - A[j]
            # We need d_next > d and denom != 0
            valid = np.abs(denom) > 1e-15
            valid &= (denom > 0)  # for d_next > d when a_k < 0... 
            # Wait: d_next = diffs[j] / denom[j]
            # We need d_next > d
            # Let me just compute all candidates
            with np.errstate(divide='ignore', invalid='ignore'):
                d_candidates = np.where(valid, diffs / np.where(valid, denom, 1.0), np.inf)
            valid2 = (d_candidates > d + 1e-15) & (d_candidates <= d_max)
            if np.any(valid2):
                d = min(d_candidates[valid2])
            else:
                d = d_max
        else:  # a_k > 0
            # Moving left (decreasing d) will decrease f
            with np.errstate(divide='ignore', invalid='ignore'):
                d_candidates = np.where(np.abs(denom) > 1e-15, diffs / np.where(np.abs(denom) > 1e-15, denom, 1.0), np.inf)
            valid2 = (d_candidates < d - 1e-15) & (d_candidates >= d_min)
            if np.any(valid2):
                d = max(d_candidates[valid2])
            else:
                d = d_min
        
        # Check if we've converged
        # At the new d, check if we're at a minimum
        c_new = c_arr + A * d
        k_max = int(np.argmax(c_new))
        # Check if subgradient contains 0
        max_val = c_new[k_max]
        active = np.where(np.abs(c_new - max_val) < 1e-12)[0]
        if np.any(A[active] < -1e-12) and np.any(A[active] > 1e-12):
            break  # 0 is in subgradient, we're at minimum
        if len(active) > 1 and np.all(np.abs(A[active]) < 1e-12):
            break
    
    # Verify and clamp
    d = max(d_min, min(d, d_max))
    
    # Check if this is actually better than d=0
    f_new = f_at(d)
    f_old = f_at(0.0)
    if f_new >= f_old:
        return 0.0
    return d

def subgradient_step(h, c_arr, n, step_size):
    """One subgradient step."""
    # Find active set
    max_c = c_arr.max()
    tol = 1e-6
    active = np.where(c_arr > max_c - tol)[0]
    
    # Compute subgradient (average gradient over active set)
    # ∂c[k]/∂h[p] = dx * [(1-h[p+t])*I(0<=p+t<n) - h[p-t]*I(0<=p-t<n)] where t = k-(n-1)

        t = shifts[k]
        # ∂c[k]/∂h[p] for all p

        g[mask2] -= h[np.arange(n)[mask2] - t]
        g += 0  # placeholder

    # Projected step
    h_new = h - step_size * g
    h_new = project_feasible(h_new, n)

        ht = slsqp_opt(best_h, n, a, maxiter=3000)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            best_h = ht.copy()
    
    if time.time() < deadline - 20:
        best_h, best_c = greedy_pair_phase(best_h, n, deadline - 20, max_iters=30000)
    
    if time.time() < deadline - 10:
        best_h, best_c = subgrad_polish(best_h, n, deadline - 10, iters=1500)
    
    if time.time() < deadline - 3:
        best_h, best_c = greedy_pair_phase(best_h, n, deadline - 3, max_iters=20000)
    
    if time.time() < deadline - 1:
        if n <= 150:
            best_h, best_c = fine_pair_polish(best_h, n, deadline - 1)
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 250
    
    best_h = None
    best_c = float('inf')
    
    if h_init is not None:
        # Primary optimization at native resolution
        alloc1 = (deadline - t0) * 0.5
        h_opt, c_opt = optimize_n(h_init, n_init, t0 + alloc1)
        best_h, best_c = h_opt, c_opt
    
        # Try coarser resolutions
        x_init = np.linspace(0, 2, n_init)
        for n_try in [100, 150]:
            if n_try >= n_init:
                continue
            if time.time() > deadline - 60:
                break
            alloc = min(80, (deadline - time.time()) * 0.25)
            if alloc < 30:
                break
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_init, h_init)
            h_interp = project_feasible(h_interp, n_try)
            h_opt, c_opt = optimize_n(h_interp, n_try, time.time() + alloc)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Restart from best with perturbations
    if best_h is not None and time.time() < deadline - 80:
        n = len(best_h)
        remaining = deadline - time.time() - 40
        n_restarts = max(2, int(remaining / 60))
        scales = [0.01, 0.02, 0.04, 0.06]
        for r in range(n_restarts):
            if time.time() > deadline - 35:
                break
            scale = scales[r % len(scales)]
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(45, (deadline - time.time() - 30) * 0.4)
            if alloc < 20:
                break
            h_opt, c_opt = optimize_n(hp, n, time.time() + alloc)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    
    # Final polish
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        best_h, best_c = greedy_pair_phase(best_h, n, deadline - 2, max_iters=30000)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    
    if best_h is None:
        n = 250
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)
    
    return best_h, best_c, len(best_h)
