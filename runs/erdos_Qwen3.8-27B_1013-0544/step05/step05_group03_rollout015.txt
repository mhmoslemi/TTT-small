def compute_c5_binary(hb, n):
    """Compute C5 for a binary h."""

    return np.correlate(hb, 1 - hb, mode='full')

    """Generate Thue-Morse binary sequence of length n."""
    h = np.zeros(n)

    zeros = np.where(hb == 0)[0]

def simulated_annealing(hb, n, deadline, T0=1.0, T1=0.01, cooldown=0.999):
    """Simulated annealing for binary optimization."""
    rng = np.random.RandomState(42)
    T = T0
    best_c = compute_c5_binary(hb, n)
    best_h = hb.copy()
    cur_c = best_c
    ones = np.where(hb == 1)[0]
    zeros = np.where(hb == 0)[0]
    
    while time.time() < deadline and T > T1:
        # Random swap
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        hb[j] = 0
        hb[i] = 1
        new_c = compute_c5_binary(hb, n)
        delta = new_c - cur_c
        if delta < 0 or rng.rand() < np.exp(-delta / T):
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = hb.copy()
            ones = np.where(hb == 1)[0]
            zeros = np.where(hb == 0)[0]
        else:
            hb[j] = 1
            hb[i] = 0
        T *= cooldown
    
    return best_h, best_c

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c)) * dx

def compute_corr_arr(h, n):

def project_binary(h, n):
    """Project to nearest binary with exactly n/2 ones."""
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

def thue_morse_binary(n):

    for i in range(n):
        h[i] = (bin(i).count('1') & 1)
    # Ensure exactly n/2 ones
    if h.sum() != n // 2:

        # Flip some to balance
        onecount = int(h.sum())
        target = n // 2
        if onecount > target:

            ones = np.where(h == 1)[0]
            rng = np.random.RandomState(0)

            zeros.append(j)

            hb[i] = 0.0
