    target_sum = n / 2.0
    h = h.copy()

            tot = h.sum()
            if tot < 1e-14:

            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=5000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return np.clip(res.x, 0.0, 1.0)
    except Exception:
        return h.copy()

def binary_sa(h_bin, n, deadline, rng, T0=0.5, T_end=0.01, max_iter=None):
    """Simulated annealing on binary vectors."""
    if max_iter is None:
        max_iter = 500000
    h = h_bin.copy().astype(float)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    cur_c = best_c
    
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        T = T0 * (T_end / T0) ** (it / max_iter)
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h[i], h[j] = 0.0, 1.0
        new_c = compute_c5(h, n)
        delta = new_c - cur_c
        
        if delta < 0 or (T > 0 and rng.rand() < np.exp(-delta / T)):
            cur_c = new_c
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h <= 0.5)[0]
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
    
    return best_h, best_c

    """Convert to binary with exact sum constraint."""
    h = h.copy()
    # Sort by value, assign 1 to top n/2

    b = np.zeros(n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return np.clip(res.x, 0.0, 1.0)
    except Exception:
        return h.copy()

def make_binary(h, n):
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:n//2]] = 1.0
    return b

def binary_sa(h_bin, n, deadline, rng, T0=0.5, T_end=0.001, max_iter=500000):
    h = h_bin.copy().astype(float)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    cur_c = best_c
    n_iter = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        T = T0 * (T_end / T0) ** (it / max_iter)
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j or h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        new_c = compute_c5(h, n)
        delta = new_c - cur_c
        if delta < 0 or (T > 1e-5 and rng.rand() < np.exp(-delta / T)):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
        n_iter += 1
    return best_h, best_c

def binary_hill_climb(h_bin, n, deadline, rng, max_passes=50):
    h = h_bin.copy().astype(float)
    best_c = compute_c5(h, n)
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        idx1 = np.where(h == 1)[0]
        idx0 = np.where(h == 0)[0]
        for i in idx1:
            if time.time() > deadline:
                break
            for j in idx0:
                h[i], h[j] = 0.0, 1.0
                c = compute_c5(h, n)
                if c < best_c - 1e-15:
                    best_c = c
                    improved = True
                    idx1 = np.where(h == 1)[0]
                    idx0 = np.where(h == 0)[0]
                    break
                else:
                    h[i], h[j] = 1.0, 0.0
        if not improved:
            break
    return h, best_c

def continuous_refine(h, n, deadline):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for alpha in [200, 2000, 20000, 200000, 2000000]:
        if time.time() > deadline:
            break
        ht = slsqp_opt(best_h, n, alpha, maxiter=2000)
        ht = project_feasible(ht, n)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht.copy()
    return best_h, best_c

    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()

    # Phase 3: Hill climb on best binary (exhaustive pair search for small n)

    # Phase 4: Continuous refinement from best binary
    if time.time() < deadline - 80:

    for r in range(n_restarts):

            elif need < 0:

        alloc = min(60, (deadline - time.time() - 20) * 0.3)

            best_c = c_opt
