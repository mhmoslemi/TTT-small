import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    exp_vals = np.exp(shifted - mx)
    log_sum = mx + np.log(exp_vals.sum())
    obj = m + log_sum / alpha
    w = exp_vals / exp_vals.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lo1 = max(0, k - n + 1)
        hi1 = min(n, k + 1)
        if lo1 < hi1:
            idx = np.arange(lo1, hi1)
            grad[lo1:hi1] += wk * (1.0 - x[idx - k + n - 1])
        lo2 = max(0, n - 1 - k)
        hi2 = min(n, 2 * n - 1 - k)
        if lo2 < hi2:
            idx = np.arange(lo2, hi2)
            grad[lo2:hi2] -= wk * x[idx + k - n + 1]
    return obj, grad

def lbbf_opt(h, n, alpha, maxiter, deadline):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_obj_grad, h, args=(n, alpha),
                       method='L-BFGS-B', jac=True, bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10, 'maxls': 50})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def local_refine(h, n, deadline, deltas):
    h = h.copy()
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(200):
            if time.time() > deadline:
                break
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                for j in range(i + 1, n):
                    hi = h[i] + d
                    hj = h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i] = hi; h[j] = hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] = hi - d; h[j] = hj + d
                    hi2 = h[i] - d
                    hj2 = h[j] + d
                    if 0.0 <= hi2 <= 1.0 and 0.0 <= hj2 <= 1.0:
                        h[i] = hi2; h[j] = hj2
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] = hi2 - d; h[j] = hj2 + d
            if not improved:
                break
    return h, best_c

def sa_smooth(h, n, deadline, rng, n_iter=5000, T0=0.002, Tf=0.00005, alpha=5000):
    h = h.copy()
    # Precompute initial obj
    dx = 2.0 / n
    def get_obj(x):
        c = np.correlate(x, 1-x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        return m + (mx + np.log(np.exp(sh - mx).sum())) / alpha
    
    best_c = compute_c5(h, n)
    best_h = h.copy()
    obj0 = get_obj(h)
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        T = T0 * (Tf / T0) ** (it / n_iter)
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        delta = rng.uniform(0.002, 0.02) * (1 if rng.rand() < 0.5 else -1)
        hi = h[i] + delta
        hj = h[j] - delta
        if not (0 <= hi <= 1 and 0 <= hj <= 1):

        h[i] = hi; h[j] = hj
        h = project_feasible(h, n)
        obj1 = get_obj(h)
        d_obj = obj1 - obj0
        if d_obj < 0 or (T > 0 and np.log(max(rng.rand(), 1e-300)) < -d_obj / T):
            obj0 = obj1
            c = compute_c5(h, n)
            if c < best_c:

        else:

            h = project_feasible(h, n)
            # recompute obj0 (project_feasible might have changed it)

        if ct < best_c:

    # Final L-BFGS-B
    if time.time() < deadline - 5:
        ht = lbbf_opt(best_h, n, 50000, 2000, deadline)
        c = compute_c5(ht, n)
        if c < best_c:

            best_h = ht.copy()
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)

        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break

            room = 1.0 - h

            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)
