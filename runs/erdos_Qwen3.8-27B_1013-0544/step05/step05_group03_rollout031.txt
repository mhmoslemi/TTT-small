def compute_max_corr(h, n):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max())

    return compute_max_corr(h, n) * (2.0 / n)

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb
