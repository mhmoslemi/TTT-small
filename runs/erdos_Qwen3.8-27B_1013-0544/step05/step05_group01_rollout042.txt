    # if alpha == 1, c_i^0 = 1
    if alpha > 1:

The Frank-Wolfe step:
1. Compute gradient g = ∇f(h)
2. Find s = argmin_{s∈P} g·s (the "linear minimization oracle")
3. Take step: h ← (1-γ)h + γ*s
4. Update step size γ

For our problem, the linear minimization oracle is:
min g·s s.t. ∑s = n/2, 0≤s≤1
The solution is: s_i = 1 if g_i is among the n/2 smallest values, 0 otherwise (approximately).
More precisely, sort g and set s_i = 1 for the n/2 smallest, 0 for the rest, with one value being fractional.

Wait, but the constraint is ∑s = n/2 and 0≤s≤1. The optimal s for min g·s is:
- Sort indices by g_i: i_1, i_2, ..., i_n (g_{i_1} ≤ g_{i_2} ≤ ... ≤ g_{i_n})
- Set s_{i_1} = ... = s_{i_{floor(n/2)}} = 1
- Set s_{i_{ceil(n/2)}} = n/2 - floor(n/2) (fractional, which is 0 if n is even)
- Set rest to 0

If n is even (which it should be since n/2 must be integer), then s is simply 0/1: the n/2 indices with smallest g get 1, the rest get 0.

So the Frank-Wolfe step is:
1. Compute g = ∇f(h)
2. Let S = {i : g_i is among n/2 smallest}
3. s = 1_S (indicator of S)
4. h ← (1-γ)h + γ*s

This is simple and efficient! And it handles the constraint naturally.

The step size γ can be:
- Fixed: γ = 2/(t+2) (decreasing)
- Line search: minimize f((1-γ)h + γ*s) over γ ∈ [0,1]

            h += (diff / tot) * h

    """Project v onto {h: sum(h) = target_sum, 0 <= h <= 1}"""
    n = len(v)

    lo = v.min() - 1.0

        else:
            tot = h.sum()
            if tot > 1e-14:
                h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def compute_gradient(h, n, alpha):
    """Compute gradient of power-max smooth objective."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx  # length 2n-1
    # Power max gradient w.r.t. c
    s = np.sum(c**alpha)
    if s < 1e-30:
        return np.zeros(n)
    coeff = s**((1.0 - alpha) / alpha)
    if alpha > 1.001:
        dc = coeff * np.power(c, alpha - 1)
    else:
        dc = coeff * np.ones_like(c)
    
    # Chain rule: grad[i] = sum over all lags
    grad = np.zeros(n)
    # Lag 0
    grad += dc[n-1] * (1 - 2*h)
    # Positive and negative lags
    for L in range(1, n):
        combined = dc[n-1+L] + dc[n-1-L]
        if abs(combined) < 1e-20:
            continue
        # (1-h[i-L]) for i in [L, n-1]
        grad[L:] += combined * (1 - h[:n-L])
        # -h[i+L] for i in [0, n-1-L]
        grad[:n-L] -= combined * h[L:]
    return grad

        s = np.zeros(n)
        s[idx[:n//2]] = 1.0

            best_c = c_new
            best_h = h_new.copy()
        
        # Check for stagnation
        if it > 100 and it % 500 == 0:
            # Try FR step size (farthest point)
            pass
        
        h = h_new
    
    return best_h, best_c

    n = len(v)

def compute_lse_gradient(h, n, alpha):
    """Gradient of LSE smooth max objective."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx  # length 2n-1, all >= 0
    M = c.max()
    shifted = alpha * (c - M)
    # Softmax weights
    max_shifted = shifted.max()
    exp_shifted = np.exp(shifted - max_shifted)
    weights = exp_shifted / exp_shifted.sum()  # length 2n-1
    
    grad = np.zeros(n)
    # Lag 0: dc[n-1]/dh[i] = 1 - 2*h[i]
    grad += weights[n-1] * (1 - 2.0 * h)
    # Other lags
    for L in range(1, n):
        w = weights[n-1+L] + weights[n-1-L]
        if w < 1e-18:
            continue
        grad[L:] += w * (1.0 - h[:n-L])
        grad[:n-L] -= w * h[L:]
    return grad

def lse_objective(h, n, alpha):

    c = np.correlate(h, 1 - h, mode='full') * dx
    M = c.max()
    shifted = alpha * (c - M)
    mx = shifted.max()
    if mx > 500:
        return float(M + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(M + np.log(np.sum(np.exp(shifted))) / alpha)

        if res.success or res.fun < lse_objective(h, n, alpha):
            return project_feasible(res.x, n)
    except:
        pass
    return h.copy()

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                return np.clip(h, 0, 1)
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                return np.clip(h, 0, 1)
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def project_simplex(v, target_sum):
    n = len(v)
    lo, hi = v.min() - 1.5, v.max() + 1.5
    for _ in range(60):
        mid = (lo + hi) * 0.5
        h = np.clip(v - mid, 0.0, 1.0)
        if h.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def lse_objective(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    M = c.max()
    shifted = alpha * (c - M)
    mx = shifted.max()
    if mx > 500:
        return float(M + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)

def compute_lse_gradient(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    M = c.max()
    shifted = alpha * (c - M)
    mx = shifted.max()
    exp_s = np.exp(shifted - mx)
    weights = exp_s / exp_s.sum()
    grad = np.zeros(n)
    grad += weights[n - 1] * (1.0 - 2.0 * h)
    for L in range(1, n):
        w = weights[n - 1 + L] + weights[n - 1 - L]
        if w < 1e-18:
            continue
        grad[L:] += w * (1.0 - h[:n - L])
        grad[:n - L] -= w * h[L:]
    return grad
