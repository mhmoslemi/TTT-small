def optimal_delta_grid(g_top, s_top, d_lo, d_hi):
    grid = np.linspace(d_lo, d_hi, 32)
    vals = np.max(g_top[None, :] + grid[:, None] * s_top[None, :], axis=1)
    idx = np.argmin(vals)
    if idx == 0:
        lo_g, hi_g = grid[0], grid[1]
    elif idx == len(grid) - 1:
        lo_g, hi_g = grid[-2], grid[-1]
    else:
        lo_g, hi_g = grid[idx - 1], grid[idx + 1]
    for _ in range(45):
        mid = (lo_g + hi_g) / 2.0
        act_val = np.max(g_top + mid * s_top)
        active = np.abs(g_top + mid * s_top - act_val) < 1e-10
        if active.sum() == 0:
            active = np.full(len(s_top), True)
        sa = s_top[active]
        if sa.min() > 1e-12:
            hi_g = mid
        elif sa.max() < -1e-12:
            lo_g = mid
        else:
            break
    d_star = (lo_g + hi_g) / 2.0
    d_star = np.clip(d_star, d_lo, d_hi)
    return d_star

def pair_descent_v2(h, n, deadline, K=16, max_sweeps=400):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    g = compute_g(h, n)
    n_shifts = 2 * n - 1
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        F = compute_F(h, n)
        n_updates = 0
        cur_max = g.max()
        update_count = 0
        for i in range(n):
            if time.time() > deadline:
                break
            Fi = F[i]
            for j in range(i + 1, n):
                s = Fi - F[j]
                d_lo = max(-h[i], h[j] - 1.0)
                d_hi = min(1.0 - h[i], h[j])
                if d_hi - d_lo < 1e-14:
                    continue
                g_rank = np.argsort(g)[::-1][:K]
                s_abs = np.abs(s)
                s_rank = np.argsort(s_abs)[::-1][:K]
                top_idx = np.unique(np.concatenate([g_rank, s_rank]))
                top_idx = top_idx[:2 * K]
                g_top = g[top_idx]
                s_top = s[top_idx]
                d_star = optimal_delta_grid(g_top, s_top, d_lo, d_hi)
                if d_star == 0.0:
                    continue
                new_max = np.max(g + d_star * s)
                if new_max < cur_max - 1e-13:
                    h[i] += d_star
                    h[j] -= d_star
                    g = g + d_star * s
                    cur_max = new_max
                    n_updates += 1
                    update_count += 1
                    if update_count % 50 == 0:
                        Fi = F[i]
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        g = compute_g(h, n)
        if n_updates == 0:
            stall += 1
            if stall > 8:
                break
        else:
            stall = 0
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 15
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: SLSQP warm start
    p1 = t0 + 0.12 * budget_s
    h = best_h.copy()
    for alpha in [2000, 20000, 200000, 2000000]:
        if time.time() > p1:
            break
        h = slsqp_optimize(h, n, alpha, p1, maxiter=150)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 2: Main pair descent
    p2 = t0 + 0.55 * budget_s
    h, c = pair_descent_v2(best_h.copy(), n, p2, K=16, max_sweeps=400)
    if c < best_c:
        best_c = c
        best_h = h.copy()

    # Phase 3: Random kicks to escape local minima
    p3 = t0 + 0.80 * budget_s
    for kick in range(30):
        if time.time() > p3:
            break
        h_kick = best_h.copy()
        n_p = rng.randint(3, 8)
        idx = rng.choice(n, size=n_p, replace=False)
        h_kick[idx] += rng.uniform(-0.04, 0.04, size=n_p)
        h_kick = project_feasible(h_kick, n)
        h_kick, c_kick = pair_descent_v2(h_kick, n, min(p3 - 2, time.time() + 20), K=16, max_sweeps=40)
        if c_kick < best_c:
            best_c = c_kick
            best_h = h_kick.copy()

    # Phase 4: SLSQP + pair descent polish
    if time.time() < dl - 40:
        h = slsqp_optimize(best_h.copy(), n, 10000000, dl - 40, maxiter=200)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    if time.time() < dl - 5:
        h, c = pair_descent_v2(best_h.copy(), n, dl - 2, K=16, max_sweeps=100)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
