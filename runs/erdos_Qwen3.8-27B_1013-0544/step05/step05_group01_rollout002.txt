def compute_c5(h, n):

    return float(np.max(c) * dx)

def compute_c_array(h, n):
    """Compute the full correlation array (before multiplying by dx)."""
    return np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h -= (-diff / tot) * h

def compute_delta_g(h, a, b):
    """Compute g[l] for transfer h[a]+=d, h[b]-=d.
    g[l] is the coefficient such that new_c[l] = c[l] + d*g[l].

    Lag k corresponds to index k in the full correlation, where
    lag = k - (n-1).

    g = np.zeros(2 * n - 1)
    for lag in range(-(n - 1), n):
        # a appears as first index at position a, second index at position a-lag
        # h[b+l] term: b+lag must be in [0, n-1]
        # h[a+l] term: a+lag must be in [0, n-1]

        # h[a-l] term: a-lag must be in [0, n-1]

    """Find d in [d_min, d_max] that minimizes max(c + d*g)."""
    # Check boundaries and d=0
    candidates = [d_min, d_max, 0.0]
    
    # Bisection on subgradient
    lo, hi = d_min, d_max
    for _ in range(50):
        mid = (lo + hi) / 2
        c_new = c + mid * g
        max_val = c_new.max()
        # Active lags
        active = c_new >= max_val - 1e-12
        g_active = g[active]
        if len(g_active) == 0:

        if g_active.min() > 1e-12:
            # All active have positive gradient, f is increasing, go left
            hi = mid
        elif g_active.max() < -1e-12:
            # All active have negative gradient, f is decreasing, go right
            lo = mid

        if val < best_val:

            best_d = d
