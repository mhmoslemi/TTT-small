def project_feasible(h, n):

            break

            room = 1.0 - h
            s = room.sum()
            if s < 1e-13: break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-13: break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    val = m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    val += 1e6 * (x.sum() - n/2.0)**2
    return val

def smooth_grad(x, n, alpha):
    dx = 2.0 / n
    c_raw = np.correlate(x, 1 - x, mode='full')
    c = c_raw * dx
    m = c.max()
    m_star = int(np.argmax(c_raw))
    k = m_star - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - x[:n-k])
        g[:n-k] += -x[k:]
    else:
        g[:n+k] += (1.0 - x[-k:])
        g[-k:] += -x[:n+k]
    g *= dx
    # softmax gradient
    sh = alpha * (c - m)
    sm = sh.max()
    probs = np.exp(sh - sm)
    probs /= probs.sum()
    g += probs
    # constraint gradient
    g += 2e6 * (x.sum() - n/2.0)
    return g

    def obj(x): return smooth_obj(x, n, alpha)
    def grad(x): return smooth_grad(x, n, alpha)

    except:

        zeros = list(np.where(h == 0)[0])

        best_move = None

            for j in ones:
                h[i], h[j] = 1, 0
                c = compute_c5(h, n)
                h[i], h[j] = 0, 1
                delta = c - best_c

        best_c += best_delta
        best_h = h.copy()

        if time.time() > p1 - 5:
            break
        h_new = lbfgs_smooth(best_h.copy(), n, alpha, 500)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()

    # Phase 2: Binary local search - best improvement (50% of time)
    p2 = t0 + 0.70 * (budget_s - 3)
    h_bin = make_binary(best_h, n)
    c_bin = compute_c5(h_bin, n)
    if c_bin < best_c:
        best_c = c_bin
        best_h = h_bin.copy()

    h_res, c_res = binary_local_search(h_bin, n, p2, rng)
    if c_res < best_c:
        best_c = c_res
        best_h = h_res.copy()

    # Phase 3: Block local search (10%)
    p3 = t0 + 0.80 * (budget_s - 3)
    h_blk, c_blk = block_local_search(best_h, n, p3, rng)
    if c_blk < best_c:
        best_c = c_blk
        best_h = h_blk.copy()

    # Phase 4: Second round of binary local search from best (7%)
    p4 = t0 + 0.87 * (budget_s - 3)
    h_res2, c_res2 = binary_local_search(best_h, n, p4, rng)
    if c_res2 < best_c:
        best_c = c_res2
        best_h = h_res2.copy()

    # Phase 5: Final L-BFGS polish from binary (3%)
    p5 = deadline
    if time.time() < p5 - 2:
        h_smooth = best_h.copy()
        h_smooth = lbfgs_smooth(h_smooth, n, 200000, 200)
        c_sm = compute_c5(h_smooth, n)
        if c_sm < best_c:
            best_c = c_sm
            best_h = h_smooth.copy()
        # Also try from slightly perturbed binary
        h_pert = best_h + 0.1 * rng.randn(n)
        h_pert = project_feasible(h_pert, n)
        h_pert = lbfgs_smooth(h_pert, n, 50000, 100)
        c_pert = compute_c5(h_pert, n)
        if c_pert < best_c:
            best_c = c_pert
            best_h = h_pert.copy()

    # Final feasibility check
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
