import time

    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            orig_sum = h.sum()
            if orig_sum < 1e-15:
                break
            h += (diff / orig_sum) * h
    return np.clip(h, 0.0, 1.0)

def vectorized_subgrad(h, n, tol_frac=0.02):
    """Compute the subgradient of the max correlation at h."""
    corr = compute_corr(h)
    mx = corr.max()
    # Active set: lags where corr is within tol_frac of max
    active_mask = corr >= mx * (1 - tol_frac) - 1e-10
    active_indices = np.where(active_mask)[0]
    
    grad = np.zeros(n)
    for m in active_indices:
        p = m - (n - 1)  # lag
        if p >= 0:
            grad[:n-p] += (1.0 - h[p:])
            grad[p:] -= h[:n-p]
        else:
            q = -p
            grad[q:] += (1.0 - h[:n-q])
            grad[:n-q] -= h[q:]
    
    return grad, mx, len(active_indices)

    """Run subgradient method to reduce max correlation."""
    h = h.copy()
    best_c = compute_c5(h, n)
    step = step_init
    for s in range(n_steps):
        grad, mx, n_active = vectorized_subgrad(h, n, tol_frac)
        h_new = h - step * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, n)
        c = compute_c5(h_new, n)
        if c < best_c:

            h = h_new
        step *= step_decay

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            orig_sum = h.sum()
            if orig_sum < 1e-15:
                break
            h += (diff / orig_sum) * h
    return np.clip(h, 0.0, 1.0)

def vectorized_subgrad(h, n, tol_frac=0.03):
    corr = compute_corr(h, n)
    mx = corr.max()
    active_mask = corr >= mx * (1.0 - tol_frac) - 1e-12
    active_indices = np.where(active_mask)[0]
    grad = np.zeros(n)
    for m in active_indices:
        p = int(m) - (n - 1)
        if p >= 0:
            grad[:n - p] += (1.0 - h[p:])
            grad[p:] -= h[:n - p]
        else:
            q = -p
            grad[q:] += (1.0 - h[:n - q])
            grad[:n - q] -= h[q:]
    return grad, mx, len(active_indices)

    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
