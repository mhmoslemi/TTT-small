import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def get_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h = h + (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h = h + (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, k):
    s = k - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        g[:n - s] += (1.0 - h[s:])
        g[s:] -= h[:n - s]
    else:
        g[-s:] += (1.0 - h[:n + s])
        g[:n + s] -= h[-s:]
    return g

def slsqp_opt(h, n, alpha, maxiter=3000):
    dx = 2.0 / n
    def obj(x):
        hx = np.clip(x, 0, 1)
        c = np.correlate(hx, 1 - hx, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def anneal(h, n, t_deadline, rng, T0=0.04):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    T = T0
    while time.time() < t_deadline and T > 0.0003:
        c = get_corr(h, n)
        curr = float(c.max())
        if curr < best_c - 1e-14:
            best_c, best_h = curr, h.copy()
        k_star = int(np.argmax(c))
        g = lag_gradient(h, n, k_star)
        if rng.random() < 0.7:
            i_src = int(np.argmax(g))
            w = np.maximum(0.0, -g + g.max() + 0.01)
            w /= w.sum()
            i_dst = int(rng.choice(n, p=w))
        else:
            i_src = int(rng.randint(n))
            i_dst = int(rng.randint(n))
        if i_src == i_dst:
            T *= 0.999
            continue
        delta = 0.06 * np.sqrt(T)
        delta = min(delta, h[i_src] * 0.8, (1.0 - h[i_dst]) * 0.8)
        if delta < 1e-7:
            T *= 0.998
            continue
        h[i_src] -= delta
        h[i_dst] += delta
        new_c = compute_c5(h, n)
        if new_c >= curr and rng.random() > np.exp(-(new_c - curr) / T):
            h[i_src] += delta
            h[i_dst] -= delta
        T *= 0.9993
    return best_h, best_c

def fine_polish(h, n, t_deadline):
    best_c = compute_c5(h, n)
