import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def smooth_max_c5(h, n, alpha):
    g = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    m = g.max()
    sh = g - m
    return m + (sh.max() + np.log(np.exp(sh - sh.max()).sum())) / alpha

def slsqp_refine(h0, n, alpha, maxiter, deadline):
    def obj(x):
        return smooth_max_c5(x, n, alpha)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h0

def subgrad_polish(h0, n, deadline, max_iter=100000):
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    lr = 0.1
    momentum = 0.9
    vel = np.zeros(n)
    dx = 2.0 / n
    
    for it in range(max_iter):
        if time.time() > deadline: break
        g = np.correlate(best_h, 1.0 - best_h, mode='full') * dx
        m = g.max()
        k = int(np.argmax(g))
        lag = k - (n - 1)
        grad = np.zeros(n)
        if lag >= 0:
            grad[lag:] += dx * (1.0 - best_h[:n - lag])
            if n - lag > 0:
                grad[:n - lag] -= dx * best_h[lag:]
        else:
            to = -lag
            if n - to > 0:
                grad[:n - to] += dx * (1.0 - best_h[to:])
            grad[to:] -= dx * best_h[:n - to]
        
        vel = momentum * vel - lr * grad
        h_new = project_feasible(best_h + vel, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        else:
            # Undo if worse
            best_h = project_feasible(best_h, n)
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def smooth_max_c5(h, n, alpha):
    g = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    m = g.max()
    sh = g - m
    return m + (sh.max() + np.log(np.exp(sh - sh.max()).sum())) / alpha

def slsqp_refine(h0, n, alpha, maxiter, deadline):
    def obj(x):
        return smooth_max_c5(x, n, alpha)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        if time.time() > deadline:
            return h0
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h0

def local_search_binary(h_bin, n, deadline, max_sweeps=5000, seed=123):

    rng = np.random.RandomState(seed)

            if time.time() > deadline: break

                    zeros = np.where(h_bin == 0)[0].tolist()
                    improved = True
                    break
                else:
                    h_bin[i] = 0.0; h_bin[j] = 1.0
            if improved: break
        if not improved: break
    return h_bin, best_c

    rng = np.random.RandomState(seed)

        else:

def subgrad_polish(h0, n, deadline, max_iter=200000):

    lr = 0.05; mom = 0.85; vel = np.zeros(n)
    dx = 2.0 / n
    for it in range(max_iter):
        if time.time() > deadline: break
        g = np.correlate(best_h, 1.0 - best_h, mode='full') * dx
        m = g.max(); k = int(np.argmax(g)); lag = k - (n - 1)
        grad = np.zeros(n)
        if lag >= 0:
            grad[lag:] += dx * (1.0 - best_h[:n-lag])
            if n-lag > 0: grad[:n-lag] -= dx * best_h[lag:]
        else:
            to = -lag
            if n-to > 0: grad[:n-to] += dx * (1.0 - best_h[to:])
            grad[to:] -= dx * best_h[:n-to]
        vel = mom * vel - lr * grad
        h_new = project_feasible(best_h + vel, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new; best_h = h_new
    return best_h, best_c
