    dx = 2.0 / n

        for j in range(lag, n):

    else:
        lp = -lag
        for j in range(lp, n):
            grad[j] += dx * (1 - h[j - lp])
        for j in range(n - lp):
            grad[j] -= dx * h[j + lp]
    return grad

    best_c = compute_c5(h, n)
    
    for d in deltas:

            corr = compute_corr(h, n)
            k_star = int(np.argmax(corr))
            lag = k_star - (n - 1)
            grad = grad_f_lag(h, n, lag)

            # Actually: we want to decrease f_lag, so move h in -grad direction

                    return h, best_c
                for idx_j in range(0, n_half):  # low grad

                        h[j] += d

                            h[j] -= d

                            break

                            h[i] -= d
                            h[j] += d
            if not improved:
                break
    
    return h, best_c

            grad[j] += dx * (1.0 - h[j + lag])
        for j in range(lag, n):
            grad[j] -= dx * h[j - lag]
    else:
        lp = -lag
        for j in range(lp, n):
            grad[j] += dx * (1.0 - h[j - lp])
        for j in range(n - lp):
            grad[j] -= dx * h[j + lp]
    return grad

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _sum_c(x, n):
    return float(x.sum() - n / 2.0)

def _sum_j(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, maxiter=1500):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_j, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def gradient_guided_pairs(h, n, deadline, deltas, top_k=25):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            corr = compute_corr(h, n)
            k_star = int(np.argmax(corr))
            lag = k_star - (n - 1)
            grad = grad_f_lag(h, n, lag)
            order = np.argsort(grad)
            lo = max(0, n - top_k)
            hi = min(n, top_k)
            for idx_i in range(lo, n):
                if time.time() > deadline:
                    return h, best_c
                i = int(order[idx_i])
                for idx_j in range(0, hi):
                    j = int(order[idx_j])
                    if i == j:
                        continue
                    if h[i] >= d and h[j] <= 1.0 - d:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                            break
                        else:
                            h[i] += d; h[j] -= d
                    if h[i] <= 1.0 - d and h[j] >= d:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                            break
                        else:
                            h[i] -= d; h[j] += d
                if improved:
                    break
            if not improved:
                break
    return h, best_c

def blind_pair_search(h, n, deadline, deltas, radius=None):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                if radius is None:
                    lo, hi = 0, n
                else:
                    lo = max(0, i - radius)
                    hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def optimize_candidate(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)
    alphas = [1000, 10000, 100000, 500000]
    for a in alphas:
        if time.time() > deadline - 15:
            break
        ht = slsqp(h, n, a, maxiter=1200)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            h = ht
    if time.time() < deadline - 10:
        h, best_c = gradient_guided_pairs(h, n, deadline - 10,
                                          [0.01, 0.005, 0.002, 0.001, 0.0005], top_k=30)
    if time.time() < deadline - 5:
        if n <= 200:
            h, best_c = blind_pair_search(h, n, deadline - 5,
                                          [0.001, 0.0005, 0.0002, 0.0001])
        else:
            h, best_c = blind_pair_search(h, n, deadline - 5,
                                          [0.001, 0.0005, 0.0002], radius=40)
    if time.time() < deadline - 3:
        ht = slsqp(h, n, 1000000, maxiter=2000)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            h = ht
    if time.time() < deadline - 1:
        h, best_c = gradient_guided_pairs(h, n, deadline - 1,
                                          [0.0005, 0.0002, 0.0001, 0.00005], top_k=20)
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def random_binary_init(n, rng):
    h = np.zeros(n)
    idx = rng.choice(n, size=n // 2, replace=False)
    h[idx] = 1.0
    return h

def m_sequence_start(n, rng):
    state = 1
    seq = []
    for _ in range(255):
        seq.append(state & 1)
        bit = ((state >> 0) ^ (state >> 4) ^ (state >> 5) ^ (state >> 6) ^ (state >> 7)) & 1
        state = (state >> 1) | (bit << 7)
    seq = np.array(seq[:n], dtype=float)
    if n > len(seq):
        seq = np.concatenate([seq, np.zeros(n - len(seq))])
    h = project_feasible(seq, n)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(100, 0.5)
        n_init = 100

    best_h = None
    best_c = float('inf')

    # Phase 1: Optimize the initial construction (most time)
    alloc1 = min(300, (deadline - t0) * 0.35)
    h_opt, c_opt = optimize_candidate(h_init, n_init, t0 + alloc1, rng)
    best_h, best_c = h_opt, c_opt

    # Phase 2: Try different n values via interpolation
    x_init = np.linspace(0, 2, n_init)
    for n_try in [200, 300]:
        if time.time() > deadline - 120:
            break
        if n_try == n_init:
            continue
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_init, h_init)
        h_interp = project_feasible(h_interp, n_try)
        alloc = min(120, (deadline - time.time()) * 0.25)
        if alloc < 40:
            break
        h_opt, c_opt = optimize_candidate(h_interp, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt

    # Phase 3: Multi-restart with perturbation on best solution
    if best_h is not None and time.time() < deadline - 80:
        n = len(best_h)
        n_restarts = max(3, int((deadline - time.time() - 40) / 50))
        for r in range(n_restarts):
            if time.time() > deadline - 30:
                break
            scale = 0.02 + 0.1 * (r / max(1, n_restarts - 1))
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(55, (deadline - time.time() - 25) * 0.4)
            if alloc < 20:
                break
            h_opt, c_opt = optimize_candidate(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 4: Try alternative structural starting points
    if time.time() < deadline - 60:
        n = len(best_h) if best_h is not None else n_init
        for init_func in [random_binary_init, m_sequence_start]:
            if time.time() > deadline - 30:
                break
            h_alt = init_func(n, rng)
            alloc = min(40, (deadline - time.time() - 15))
            if alloc < 15:
                break
            h_opt, c_opt = optimize_candidate(h_alt, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    # Phase 5: Final intensive polish on the best solution
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        if n <= 250:
            best_h, best_c = blind_pair_search(best_h, n, deadline - 2,
                                               [0.0005, 0.0002, 0.0001, 0.00005, 0.00002])
        else:
            best_h, best_c = blind_pair_search(best_h, n, deadline - 2,
                                               [0.0005, 0.0002, 0.0001], radius=50)

    if best_h is None:
        n = n_init
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
