import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def compute_fg(h, n, m_range):
    idx_f = np.arange(n)[:, None] + m_range[None, :]
    vf = (idx_f >= 0) & (idx_f < n)
    F = np.where(vf, 1.0 - h[np.clip(idx_f, 0, n - 1)], 0.0)
    idx_b = np.arange(n)[:, None] - m_range[None, :]
    vb = (idx_b >= 0) & (idx_b < n)
    G = np.where(vb, h[np.clip(idx_b, 0, n - 1)], 0.0)
    return F, G

def vec_binary_sa(h, n, deadline, max_sweeps=80000, sa_temp=0.3, seed=42, kick_every=500):
    m_range = np.arange(-(n - 1), n)
    two_n_1 = 2 * n - 1
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_c = float(corr.max())
    dx = 2.0 / n
    best_c = cur_c
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    F, G = compute_fg(h, n, m_range)
    T = sa_temp
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        order = list(zeros_list)
        rng.shuffle(order)
        for i in order:
            if h[i] != 0:
                continue
            base = F[i] - G[i]
            if len(ones_list) == 0:
                break
            j_arr = np.array(ones_list)
            deltas = base[None, :] - F[j_arr] + G[j_arr]
            pos1 = j_arr - i + (n - 1)
            pos2 = i - j_arr + (n - 1)
            v1 = (pos1 >= 0) & (pos1 < two_n_1)
            v2 = (pos2 >= 0) & (pos2 < two_n_1)
            new_c = corr[None, :] + deltas
            rows = np.arange(len(j_arr))
            new_c[rows[v1], pos1[v1]] += 1.0
            new_c[rows[v2], pos2[v2]] += 1.0
            new_max = new_c.max(axis=1)
            best_idx = int(np.argmin(new_max))
            best_val = float(new_max[best_idx])
            if best_val < cur_c - 1e-15:
                accept = True
            elif sa_temp > 0 and best_val < cur_c + 3.0:
                accept = rng.random() < np.exp(-(best_val - cur_c) / (T + 1e-30))
            else:
                accept = False
            if not accept:
                continue
            j = int(j_arr[best_idx])
            h[i] = 1.0
            h[j] = 0.0
            corr = new_c[best_idx].copy()
            cur_c = float(corr.max())
            F, G = compute_fg(h, n, m_range)
            ones_list.remove(j)
            zeros_list.remove(i)
            ones_list.append(i)
            zeros_list.append(j)
            improved = True
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
            T *= 0.9997
            break
        if not improved:
            no_improve += 1
            if no_improve >= kick_every:
                k = min(5, len(ones_list), len(zeros_list))
                for _ in range(k):
                    oi = list(ones_list)
                    zi = list(zeros_list)
                    a = oi[rng.randint(len(oi))]
                    b = zi[rng.randint(len(zi))]
                    h[a], h[b] = h[b], h[a]
                    ones_list.remove(a)
                    zeros_list.remove(b)
                    ones_list.append(b)
                    zeros_list.append(a)
                corr = np.correlate(h, 1 - h, mode='full')
                cur_c = float(corr.max())
                F, G = compute_fg(h, n, m_range)
                no_improve = 0
                T = sa_temp
            elif no_improve > 3000:
                break

    return best_h, best_c * dx

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h
