```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def compute_g_raw(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def make_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def binary_local_search(h_bin, n, deadline, seed=42):
    best_h = h_bin.copy()
    best_c = compute_c5(best_h, n)
    rng = np.random.RandomState(seed)
    no_improve = 0
    
    while time.time() < deadline:
        if no_improve > 10000:
            break
        g = compute_g_raw(best_h, n)
        k_star = int(np.argmax(g))
        lag = k_star - (n - 1)
        
        # Find contributing positions
        if lag >= 0 and lag < n:
            contrib = [(i, i + lag) for i in range(n - lag) 
                      if best_h[i] == 1 and best_h[i + lag] == 0]
        elif lag < 0 and -lag < n:
            to = -lag
            contrib = [(i + to, i) for i in range(n - to)
                      if best_h[i + to] == 1 and best_h[i] == 0]
        else:
            no_improve += 1
            continue
        
        if not contrib:
            no_improve += 1
            continue
        
        # Try random sample of contributing pairs
        n_try = min(30, len(contrib))
        sample = rng.choice(len(contrib), size=n_try, replace=False)
        
        improved = False
        for idx in sample:
            i, j = contrib[idx]
            best_h[i] = 0.0
            best_h[j] = 1.0
            c = compute_c5(best_h, n)
            if c < best_c - 1e-15:
                best_c = c
                no_improve = 0
                improved = True
                break
            else:
                best_h[i] = 1.0
                best_h[j] = 0.0
        
        if not improved:
            # Random swap
            ones = np.where(best_h == 1)[0]
            zeros = np.where(best_h == 0)[0]
            ii = zeros[int(rng.randint(len(zeros)))]
            jj = ones[int(rng.randint(len(ones)))]
            best_h[ii] = 1.0
            best_h[jj] = 0.0
            c = compute_c5(best_h, n)
            if c < best_c - 1e-15:
                best_c = c
                no_improve = 0
            else:
                best_h[ii] = 0.0
                best_h[jj] = 1.0
                no_improve += 1
    
    return best_h, best_c

def simulated_annealing_binary(h_bin, n, deadline, seed=777):
    best_h = h_bin.copy()
    best_c = compute_c5(best_h, n)
    cur_h = h_bin.copy()
    cur_c = best_c
    rng = np.random.RandomState(seed)
    
    ones = np.where(cur_h == 1)[0]
    zeros = np.where(cur_h == 0)[0]
    
    t_start = 0.3
    t_end = 0.001
    total_iters = 300000
    t0 = time.time()
    
    for it in range(total_iters):
        if time.time() > deadline:
            break
        frac = it / total_iters
        T = t_start * (t_end / t_start) ** frac
        
        ii = zeros[int(rng.randint(len(zeros)))]
        jj = ones[int(rng.randint(len(ones)))]
        
        cur_h[ii] = 1.0
        cur_h[jj] = 0.0
        new_c = compute_c5(cur_h, n)
        
        if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / max(T, 1e-15)):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
        else:
            cur_h[ii] = 0.0
            cur_h[jj] = 1.0
    
    return best_h, best_c

def pairwise_refine(h, n, deadline):
    """Try changing pairs (i,j) where h[i]=1, h[j]=0 to (a, 1-a)."""
    best_c = compute_c5(h, n)
    n_ones = int(h.sum())  # should be n/2
    
    # Get lists of 1s and 0s
    ones = np.where(h >= 0.999)[0]
    zeros = np.where(h <= 0.001)[0]
    
    # a values to try (from coarse to fine)
    a_coarse = np.arange(0.05, 1.0, 0.05)  # 0.05, 0.10, ..., 0.95
    
    while time.time() < deadline:
        improved = False
        g = compute_g_raw(h, n)
        k_star = int(np.argmax(g))
        lag = k_star - (n - 1)
        
        # Prioritize pairs that affect the worst lag
        if lag >= 0 and lag < n:
            # Pairs (i, i+lag) where h[i]=1, h[i+lag]=0
            prio_ones = []
            for i in range(n - lag):
                if h[i] >= 0.999 and h[i + lag] <= 0.001:
                    prio_ones.append(i)
        else:
            prio_ones = []
        
        # Build pair list: prioritized first, then random
        pairs = []
        for i in prio_ones:
            if lag >= 0:
                pairs.append((i, i + lag))
            else:
                pairs.append((i - lag, i))
        
        # Add some random pairs
        n_random = 200
        for _ in range(n_random):
            i = int(rng.randint(len(ones))) if len(ones) > 0 else 0
            j = int(rng.randint(len(zeros))) if len(zeros) > 0 else 0
            pairs.append((int(ones[i]), int(zeros[j])))
        
        for (i, j) in pairs:
            if time.time() > deadline:
                break
            if h[i] < 0.5 or h[j] > 0.5:
                continue  # skip if not a clear 1-0 pair
            
            for a in a_coarse:
                if a <= 0.01 or a >= 0.99:
                    continue
                h[i] = a
                h[j] = 1.0 - a
                c = compute_c5(h, n)
                if c < best_c - 1e-14:
                    best_c = c
                    improved = True
                    break
                else:
                    h[i] = 1.0
                    h[j] = 0.0
            
            # Update ones/zeros lists
            if improved:
                ones = np.where(h >= 0.999)[0]
                zeros = np.where(h <= 0.001)[0]
        
        if not improved:
```

**Idea: Projected gradient with backtracking**

For the smooth max objective f(h) = smoothmax(G(h), alpha):
1. Compute gradient: ∇f = (1/alpha) * sum_k exp((G[k]-m)*alpha) * ∇G[k] / sum_k exp((G[k]-m)*alpha)
   (softmax-weighted average of the gradients of G[k])
2. Take a step in the negative gradient direction
3. Project back to the feasible set
4. Repeat

The gradient of G[k] with respect to h[i]:
G[k] = sum_l h[l] * (1-h[l+k])
∂G[k]/∂h[i] = (1-h[i+k]) - h[i-k] (with appropriate bounds)

For the smooth max:
∂f/∂h[i] = sum_k p[k] * ((1-h[i+k]) - h[i-k])
where p[k] = exp(alpha*(G[k]-m)) / sum_j exp(alpha*(G[j]-m))

This is a weighted average of the "directions" that would increase each G[k].

The projected gradient step:
h_new = project(h - lr * grad)

This is simple and fast. Let me implement it with:
- Backtracking line search
- Momentum (Nesterov)
- Increasing alpha over time

This should be much faster than SLSQP since each iteration is O(n) (computing the gradient) + O(n) (projection).

Actually, computing the gradient requires summing over all K=2n-1 lags, each involving a computation over n positions. So it's O(n²) per gradient evaluation. But with the softmax weights, we can precompute the weights and then the gradient is:

grad[i] = sum_k p[k] * ((1-h[i+k]) - h[i-k])

This can be computed as:
grad[i] = sum_k p[k]*(1-h[i+k]) - sum_k p[k]*h[i-k]
         = (1 - convolve-like) - (convolve-like)

Actually:
sum_k p[k] * h[i+k] is a weighted convolution of h with p!
And sum_k p[k] * h[i-k] is also a weighted convolution (with reflected p).

So we can compute the gradient using FFT-based convolution in O(n log n)!

```python
def compute_smooth_grad(h, n, alpha):
    """Compute gradient of smooth max of G."""
    g = compute_g_raw(h, n) * (2.0 / n)
    m = g.max()
    shifted = alpha * (g - m)
    # Use stable log-sum-exp
    max_shifted = shifted.max()
    weights = np.exp(shifted - max_shifted)
    weights /= weights.sum()
    # weights[k] corresponds to lag k - (n-1)
    # G[k-(n-1)] = sum_i h[i] * (1-h[i+k-(n-1)])
    # dG[k]/dh[i] = (1-h[i+k]) - h[i-k] (with bounds)
    # sum_k weights[k] * dG[k]/dh[i]
    # = sum_k weights[k] * (1-h[i+k]) - sum_k weights[k] * h[i-k]
    # Let p[m] = weights[m + n - 1] for m in [-(n-1), n-1]
    # Then sum_k p[m] * h[i+m] is a convolution
    p = weights  # p[j] where j = n-1+m, m is the lag
    # sum_m p[n-1+m] * h[i+m] for valid i+m
    # This is like cross-correlation of h with p (reflected)
    
    # Actually, let's just do it directly with numpy
    # For each i: grad[i] = sum_k p[k] * ((1-h[i+lag_k]) - h[i-lag_k])
    # where lag_k = k - (n-1)
    
    # This is: (1 - correlation(h, p)) + correlation(h, p_flipped)
    # Hmm, let me think more carefully.
    
    # G_full[j] = sum_i h[i] * (1-h[i+j-(n-1)])
    # dG_full[j]/dh[i] = (1-h[i+j-(n-1)]) - h[i-(j-(n-1))] * (if i-(j-(n-1)) valid)
    # Wait, the second term: in G_full[j], h[i] appears in the second factor when
    # the "first" index is i - (j-(n-1)), i.e., G_full[j] includes term
    # h[i-((j)-(n-1))] * (1-h[i]) if i-((j)-(n-1)) >= 0
    # So d/dh[i] of that term = -h[i-((j)-(n-1))]
    
    # So dG_full[j]/dh[i] = (1-h[i+j-(n-1)]) (if 0<=i+j-(n-1)<n) 
    #                         - h[i-(j-(n-1))] (if 0<=i-(j-(n-1))<n)
    
    # sum_j weights[j] * dG_full[j]/dh[i]
    # = sum_j weights[j] * (1-h[i+j-(n-1)]) * [0<=i+j-(n-1)<n]
    #   - sum_j weights[j] * h[i-(j-(n-1))] * [0<=i-(j-(n-1))<n]
    
    # First term: sum_{j: 0<=i+j-(n-1)<n} weights[j] * (1-h[i+j-(n-1)])
    # = (sum_{j: valid} weights[j]) - sum_{j: valid} weights[j] * h[i+j-(n-1)]
    # The valid j range: n-1-i <= j <= n-1+i, intersected with [0, 2n-2]
    
    # This is getting complex. Let me just use the direct O(n²) approach for n=250.
    # O(n²) = 62500 operations per gradient, which is fine.
    
    lags = np.arange(2 * n - 1) - (n - 1)  # lags from -(n-1) to (n-1)
    grad = np.zeros(n)
    for i in range(n):
        s1 = 0.0
        s2 = 0.0
        for k in range(2 * n - 1):
            lag = lags[k]
            if 0 <= i + lag < n:
                s1 += weights[k] * (1.0 - h[i + lag])
            if 0 <= i - lag < n:
                s2 += weights[k] * h[i - lag]
        grad[i] = s1 - s2
    return grad
```

```python
def compute_smooth_grad(h, n, alpha):
    g = compute_g_raw(h, n) * (2.0 / n)
    m = g.max()
    shifted = alpha * (g - m)
    max_s = shifted.max()
    weights = np.exp(shifted - max_s)
    weights /= weights.sum()
    
    # Vectorized computation
    # For each i, grad[i] = sum_k weights[k] * (1-h[i+lag_k]) - sum_k weights[k] * h[i-lag_k]
    # where the sums are over valid indices only.
    
    # First sum: sum_k weights[k] * (1-h[i+lag_k]) for 0<=i+lag_k<n
    # = sum_k weights[k] * (1 - h[i+lag_k]) * [0<=i+lag_k<n]
    
    # Let's use matrix approach:
    # M[i,k] = 1 if 0 <= i + lags[k] < n, else 0
    # Then first_sum[i] = sum_k M[i,k] * weights[k] - sum_k M[i,k] * weights[k] * h[i+lags[k]]
    
    # This is complex to vectorize nicely. Let me use FFT.
    
    # Alternative: use the fact that for fixed i, the set of valid k's is a contiguous range.
    # lags[k] = k - (n-1), so i + lags[k] = i + k - (n-1)
    # 0 <= i + k - (n-1) < n means: (n-1-i) <= k <= (n-1-i+n-1) = (2n-2-i)
    # So k ranges from max(0, n-1-i) to min(2n-2, 2n-2-i)
    
    # Similarly for the second sum: 0 <= i - lags[k] < n
    # i - (k-(n-1)) = i - k + n - 1
    # 0 <= i - k + n - 1 < n means: i+n-1-n < k <= i+n-1, i.e., i-1 < k <= i+n-1
    # So k ranges from max(0, i) to min(2n-2, i+n-1)
    
    # For the first sum:
    # S1[i] = sum_{k=a}^{b} weights[k] - sum_{k=a}^{b} weights[k] * h[i+k-(n-1)]
    # where a = max(0, n-1-i), b = min(2n-2, 2n-2-i)
    
    # The first part is just a range sum of weights (can use cumulative sum).
    # The second part is: sum_{k=a}^{b} weights[k] * h[i+k-(n-1)]
    # Let j = k - (n-1): sum_{j=a-(n-1)}^{b-(n-1)} weights[j+(n-1)] * h[i+j]
    # = sum_{j=-(i)}^{n-1-i} weights[j+(n-1)] * h[i+j]
    # This is a weighted sum of h[0..n-1] with weights that depend on i.
    
    # This is essentially a cross-correlation between weights and h, evaluated at different offsets.
    # Specifically, it's the cross-correlation of h with the weight vector p where p[j] = weights[j+(n-1)] for j in [-(n-1), n-1].
    
    # We can compute this using FFT!
    
    # Let's construct the weight vector as a full correlation kernel:
    # p_full has length 2n-1, p_full[k] = weights[k]
    # The cross-correlation (h ⋆ p_full)[i] = sum_k h[i+k] * p_full[k] (with appropriate bounds)
    # Actually, np.correlate(h, p_full, mode='full') would give us this.
    
    # But the bounds are tricky. Let me just use the direct approach with cumsum for the constant part
    # and FFT for the weighted part.
    
    # Actually, for n=250, the direct O(n²) approach with numpy vectorization should be fast enough.
    # Let me create a matrix of valid indices.
    
    # For speed, let me precompute the valid index ranges:
    # For the first sum, valid k for position i: k in [n-1-i, 2n-2-i] ∩ [0, 2n-2]
    # For the second sum, valid k for position i: k in [i, i+n-1] ∩ [0, 2n-2]
    
    # I'll use a different approach: compute the gradient as:
    # grad[i] = sum_k weights[k] * (1-h[i+lag_k]) * valid1[i,k] - sum_k weights[k] * h[i-lag_k] * valid2[i,k]
    
    # Let me just do it with a loop over k (outer) and vectorize over i (inner):
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        w = weights[k]
        if abs(w) < 1e-15:
            continue
        if lag >= 0:
            # First part: h[i+lag] valid for i in [0, n-1-lag]
            if n - 1 - lag >= 0:
                grad[:n-lag] += w * (1.0 - h[lag:])
            # Second part: h[i-lag] valid for i in [lag, n-1]
            if lag < n:
                grad[lag:] -= w * h[:n-lag]
        else:
            to = -lag
            # First part: h[i+lag] = h[i-to] valid for i in [to, n-1]
            if to < n:
                grad[to:] += w * (1.0 - h[:n-to])
            # Second part: h[i-lag] = h[i+to] valid for i in [0, n-1-to]
            if n - 1 - to >= 0:
                grad[:n-to] -= w * h[to:]
    
    return grad
```
