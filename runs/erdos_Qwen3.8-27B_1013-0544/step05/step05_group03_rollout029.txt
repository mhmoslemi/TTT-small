    h = np.clip(h, 0.0, 1.0)

        if abs(diff) < 1e-14:
            break

            s = room.sum()

                break

        else:
            s = h.sum()
            if s < 1e-14:

    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

    return hb

def make_structured_binary(n, rng, mode=0):
    """Create structured initial binary solutions."""
    hb = np.zeros(n)
    target = n // 2
    if mode == 0:
        # Dense cluster in middle
        start = n // 4
        hb[start:start+target] = 1.0
    elif mode == 1:
        # Two blocks
        block = target // 2
        hb[:block] = 1.0
        hb[n//2:n//2+block] = 1.0

            hb[n//2+block] = 1.0
    elif mode == 2:

        for i in range(0, n, 2):
            if (i // 2) < target:
                hb[i] = 1.0
    elif mode == 3:
        # Quadratic residue pattern
        for i in range(n):
            if (i * i) % n < n // 2:
                hb[i] = 1.0
        # Adjust to have exactly target ones
        ones = np.where(hb == 1)[0]
        zeros = np.where(hb == 0)[0]
        if len(ones) > target:
            remove = rng.choice(ones - hb[ones], size=len(ones)-target, replace=False)

            hb[add] = 1
    else:

        return make_random_binary(n, rng)

    perm = rng.permutation(n)
    hb = hb[perm]
    return hb

    best_hb = hb.copy()
    T = T0
    current_c = best_c
    ones_list = list(ones)
    zeros_list = list(zeros)
    
    iter_no = 0
    while time.time() < deadline and iter_no < max_iters:
        iter_no += 1
        T *= cooling
        if T < 1e-10:
            T = 1e-10
        
        # Random pair swap
        i = zeros_list[rng.randint(len(zeros_list))]
        j = ones_list[rng.randint(len(ones_list))]
        
        # Apply swap
        hb[i] = 1.0
        hb[j] = 0.0
        c_new = compute_c5(hb, n)

        if delta < 0 or rng.random() < np.exp(-delta / T):
            current_c = c_new
            ones_list.remove(j)
            ones_list.append(i)
            zeros_list.remove(i)

                best_hb = hb.copy()
        else:

            hb[j] = 1.0

        return m + np.log(np.exp(sh).sum()) / alpha

            # d(c_lag)/dh[j] = (1-h[j-lag]) for j in [lag, n-1] - h[j+lag] for j in [0, n-1-lag]
            part1 = np.zeros(n)
            part1[lag:] = 1.0 - h[:n-lag]
            part2 = np.zeros(n)
            part2[:n-lag] = -h[lag:]
            grad += wk * (part1 + part2)
        else:
            ell = -lag
            # d(c_{-ell})/dh[j] = (1-h[j+ell]) for j in [0, n-1-ell] - h[j-ell] for j in [ell, n-1]
            part1 = np.zeros(n)
            part1[:n-ell] = 1.0 - h[ell:]
            part2 = np.zeros(n)
            part2[ell:] = -h[:n-ell]
            grad += wk * (part1 + part2)
    return grad

def slsqp_optimize(h, n, alpha, maxiter, deadline):
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
        m = g.max()
        shifted = alpha * (g - m)
        shifted -= shifted.max()
        return m + np.log(np.exp(shifted).sum()) / alpha
    bounds = [(0.0, 1.0)] * n
    def con_fun(x):
        return x.sum() - n / 2.0

        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h

    c = np.correlate(h, 1.0 - h, mode='full')
    f = c * dx
    m = f.max()
