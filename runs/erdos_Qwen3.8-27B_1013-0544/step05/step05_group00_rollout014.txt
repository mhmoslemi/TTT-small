def fast_swap_search(h, n, c, deadline, max_rounds=500):
    """
    Greedy swap search: at each step, find the best (i,j) swap.
    h: binary array, c: current correlation array (length 2n-1)
    """
    L = n - 1
    n_ones = h.sum()
    cur_max = c.max()
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for round in range(max_rounds):
        if time.time() > deadline:
            break
        
        best_improve = 0
        best_i = -1
        best_j = -1
        best_new_c = None
        
        # For efficiency, only check a subset of swaps
        # Strategy: focus on swaps that affect the maximum correlation
        max_lags = np.where(np.isclose(c, cur_max, atol=1e-10))[0]
        
        for i in zeros:

            di = np.zeros(2*n-1)

def anneal_binary(h, n, deadline, seed=42):
    """Simulated annealing on binary h."""
    L = n - 1
    c = np.correlate(h, 1-h, mode='full')
    cur_c = c.max()
    best_c = cur_c
    best_h = h.copy()
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    rng = np.random.RandomState(seed)
    T = 5.0
    cool = 0.999
    
    while time.time() < deadline:
        # Pick random swap
        i = rng.choice(zeros)
        j = rng.choice(ones)
        
        # Compute delta
        di = np.zeros(2*n-1)
        m1 = i - np.arange(2*n-1) + L
        v1 = (m1 >= 0) & (m1 < n)
        di[v1] = 1 - h[m1[v1]]
        m2 = i + np.arange(2*n-1) - L
        v2 = (m2 >= 0) & (m2 < n)
        di[v2] -= h[m2[v2]]
        
        dj = np.zeros(2*n-1)
        m1 = j - np.arange(2*n-1) + L
        v1 = (m1 >= 0) & (m1 < n)
        dj[v1] = -(1 - h[m1[v1]])
        m2 = j + np.arange(2*n-1) - L
        v2 = (m2 >= 0) & (m2 < n)
        dj[v2] += h[m2[v2]]
        
        new_c = c + di + dj
        new_max = new_c.max()
        delta_obj = new_max - cur_c
        
        if delta_obj < 0 or rng.random() < np.exp(-delta_obj / T):
            h[i] = 1; h[j] = 0
            c = new_c
            cur_c = new_max
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        
        T *= cool
    
    return best_h, best_c * 2/n

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def swap_delta(h, n, i, j):
    """Compute delta in correlation when h[i] 0->1 and h[j] 1->0."""
    L = n - 1
    m = np.arange(2*n-1)
    delta = np.zeros(2*n-1)
    # i contributes: +（1-h[i-m+L]) where valid
    p1 = i - m + L
    v1 = (p1 >= 0) & (p1 < n)
    delta[v1] += (1 - h[p1[v1]])
    # i removes from (1-h): -h[i+m-L] where valid
    p2 = i + m - L
    v2 = (p2 >= 0) & (p2 < n)
    delta[v2] -= h[p2[v2]]
    # j removes: -(1-h[j-m+L]) where valid
    p3 = j - m + L
    v3 = (p3 >= 0) & (p3 < n)
    delta[v3] -= (1 - h[p3[v3]])
    # j adds to (1-h): +h[j+m-L] where valid
    p4 = j + m - L
    v4 = (p4 >= 0) & (p4 < n)
    delta[v4] += h[p4[v4]]
    return delta

                best_c = cur_max
                best_h = h.copy()
        T *= cool_rate
