    # h[i] goes from 0 to 1, h[j] goes from 1 to 0
    # corr[k] = sum_m h[m] * (1-h[m-k+n-1])
    #

    # k ranges from 0 to 2n-2, index = i - k + n - 1 must be in [0, n-1]

    for idx in range(n):

        if 0 <= k < 2*n - 1:

def l5bfsb_step(h, n, deadline, alpha=100000, penalty=1000):
    dx = 2.0 / n
    target = n / 2.0
    
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha + penalty * (x.sum() - target)**2
    
    def grad(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        exp_sh = np.exp(sh - sm)
        w = exp_sh / exp_sh.sum()
        
        grad_corr = np.zeros(n)
        for k in range(2*n-1):
            wk = w[k]
            if wk < 1e-14:
                continue
            s = k - n + 1
            if s >= 0:
                if n - s > 0:
                    grad_corr[0:n-s] += wk * (1.0 - x[s:n])
                    grad_corr[s:n] -= wk * x[0:n-s]
            else:
                ss = -s
                grad_corr[ss:n] += wk * (1.0 - x[0:n-ss])
                grad_corr[0:n-ss] -= wk * x[ss:n]
        grad_corr *= dx
        
        grad_pen = 2 * penalty * (x.sum() - target)
        return grad_corr + grad_pen
    
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': 5000, 'ftol': 1e-18, 'gtol': 1e-12})
    return np.clip(res.x, 0, 1)
