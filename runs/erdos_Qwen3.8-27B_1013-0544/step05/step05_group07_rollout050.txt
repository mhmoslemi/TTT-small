        return c_new.max()
    
    # Ternary search on [0, 1]
    lo, hi = 0.0, 1.0
    for _ in range(40):
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        if f(m1) < f(m2):
            hi = m2
        else:
            lo = m1

    c_new[n-1] = c_vals[n-1] + (t_opt*(1-t_opt) - old*(1-old))
    return t_opt, c_new

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-13:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-13:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def eval_move(h, n, i, j, delta, lags):
    """Evaluate C5 after moving delta mass from i to j, using cached lags."""
    old_i, old_j = h[i], h[j]
    h[i] = old_i + delta
    h[j] = old_j - delta
    c = compute_lags(h, n)
    h[i], h[j] = old_i, old_j
    return c.max()

def adaptive_pair_move(h, n, i, j, lags, current_max):
    """Find optimal delta for moving mass from i to j (or j to i)."""
    old_i, old_j = h[i], h[j]

    lo = -old_j
    hi = 1.0 - old_i
    if hi - lo < 1e-12:
        return 0.0
    # Ternary search for min of max(lags after move)
    for _ in range(30):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        h[i], h[j] = old_i + m1, old_j - m1

        h[i], h[j] = old_i + m2, old_j - m2
        f2 = compute_lags(h, n).max()

        else:
            lo = m1

                zeros_set.add(o2)
            else:
                h_bin[o1], h_bin[o2] = 1.0, 1.0
                h_bin[z1], h_bin[z2] = 0.0, 0.0
    return best_h, best_c

def lbfgs_refine(h, n, alpha, maxiter):
    """L-BFGS-B with log-sum-exp smoothing."""
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        sm = sh.max()
        val = m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
        val += 1e4 * (x.sum() - n / 2.0) ** 2
        return val
    def grad(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m_star = int(np.argmax(c))
        k = m_star - (n - 1)
        g = np.zeros(n)
        if k >= 0:
            g[k:] += (1.0 - x[:n - k])
            g[:n - k] += -x[k:]
        else:
            g[:n + k] += (1.0 - x[-k:])
            g[-k:] += -x[:n + k]
        g += 2e4 * (x.sum() - n / 2.0)
        return g
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=grad,
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
        return project_feasible(res.x, n)
    except Exception:
        return h

def multi_start_search(h_init, n, deadline, rng):
    """Try multiple starting points and optimize each."""
    best_h = h_init.copy()
    best_c = compute_c5(h_init, n)
    
    starts = [h_init.copy()]
    # Binary projection
    idx_sorted = np.argsort(h_init)[::-1]
    hb = np.zeros(n)
    hb[idx_sorted[:n//2]] = 1.0
    starts.append(hb)
    # Random with structure
    for _ in range(3):
        hr = h_init.copy()
        hr += 0.15 * rng.randn(n)
        starts.append(project_feasible(hr, n))
    
    for s in starts:
        if time.time() > deadline:
            break
        c = compute_c5(s, n)
        if c < best_c:
            best_c, best_h = c, s.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)
    
    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(h0, n)
    
    # Phase 1: Quick L-BFGS (15% time)
    p1 = t0 + 0.15 * (budget_s - 5)
    for alpha in [5000, 50000, 500000]:
        if time.time() > p1 - 5:
            break
        h_new = lbfgs_refine(best_h.copy(), n, alpha, 200)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c, best_h = c_new, h_new.copy()
    
    # Phase 2: Adaptive pair-move local search (35% time)
    p2 = t0 + 0.50 * (budget_s - 5)
    h2, c2 = local_search_pairs(best_h.copy(), n, p2, rng, max_rounds=800)
    if c2 < best_c:
        best_c, best_h = c2, h2.copy()
    
    # Phase 3: Binary local search (20% time)
    p3 = t0 + 0.70 * (budget_s - 5)
    hb, cb = binary_locals(best_h.copy(), n, p3, rng, max_iter=200000)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()
    
    # Phase 4: Recovery from binary + fractional refinement (20% time)
    p4 = t0 + 0.90 * (budget_s - 5)
    # Start from best binary, add fractional perturbation
    idx_sorted = np.argsort(best_h)[::-1]
    hb2 = np.zeros(n)
    hb2[idx_sorted[:n//2]] = 1.0
    hb2, cb2 = binary_locals(hb2, n, p4 - 30, rng, max_iter=100000)
    if cb2 < best_c:
        best_c, best_h = cb2, hb2.copy()
    # Fractional refinement from binary
    hf = best_h.copy()
    hf += 0.1 * rng.randn(n)
    hf = project_feasible(hf, n)
    h4, c4 = local_search_pairs(hf, n, p4, rng, max_rounds=300)
    if c4 < best_c:
        best_c, best_h = c4, h4.copy()
    
    # Phase 5: Final polish (10% time)
    p5 = deadline
    h_new = lbfgs_refine(best_h.copy(), n, 2000000, 100)
    c_new = compute_c5(h_new, n)
    if c_new < best_c:
        best_c, best_h = c_new, h_new.copy()
    h5, c5 = local_search_pairs(best_h.copy(), n, p5, rng, max_rounds=200)
    if c5 < best_c:
        best_c, best_h = c5, h5.copy()
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
