       def obj(x):
           g = np.correlate(x, 1 - x, mode='full') * dx
           m = g.max()
           shifted = alpha * (g - m)
           sm = shifted.max()
           return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
       bounds = [(0.0, 1.0)] * n
       cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0}]
       try:
           res = minimize(obj, h, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
           return project_feasible(res.x, n)
       except:
           return h

   def pair_descent(h, n, deadline, K=6, max_sweeps=2000):
       best_h = h.copy()
       best_c = compute_c5(h, n)
       dm = np.arange(2*n-1) - (n-1)
       idx_all = np.arange(2*n-1)
       stall = 0
       for sweep in range(max_sweeps):
           if time.time() > deadline: break
           F = compute_F(h, n)
           g = compute_g(h, n)
           cur_max = g.max()
           top = np.argsort(g)[::-1][:K]
           n_improve = 0
           for i in range(n):
               if time.time() > deadline: break
               for j in range(i+1, n):
                   s = F[i] - F[j]
                   d_lo = max(-h[i], h[j]-1.0)
                   d_hi = min(1.0-h[i], h[j])
                   if d_hi - d_lo < 1e-14: continue
                   cands = [d_lo, d_hi]
                   for a in range(K):
                       for b in range(K):
                           if top[a] == top[b]: continue
                           ds = s[top[b]] - s[top[a]]
                           if abs(ds) > 1e-15:
                               dc = (g[top[a]] - g[top[b]]) / ds
                               if d_lo < dc < d_hi: cands.append(dc)
                   best_d = 0.0
                   best_v = cur_max
                   for d in cands:
                       d = float(np.clip(d, d_lo, d_hi))
                       v = float(np.max(g + d*s))
                       if v < best_v - 1e-13:
                           best_v = v
                           best_d = d
                   if best_d != 0.0:
                       h[i] += best_d
                       h[j] -= best_d
                       g += best_d * s
                       # Update F
                       v1 = (i + dm >= 0) & (i + dm < n)
                       F[i+dm[v1], idx_all[v1]] -= best_d
                       v2 = (i - dm >= 0) & (i - dm < n)
                       F[i-dm[v2], idx_all[v2]] -= best_d
                       v3 = (j + dm >= 0) & (j + dm < n)
                       F[j+dm[v3], idx_all[v3]] += best_d
                       v4 = (j - dm >= 0) & (j - dm < n)
                       F[j-dm[v4], idx_all[v4]] += best_d
                       cur_max = g.max()
                       top = np.argsort(g)[::-1][:K]
                       n_improve += 1
           h = project_feasible(h, n)
           c = compute_c5(h, n)
           if c < best_c:
               best_c = c
               best_h = h.copy()
           if n_improve == 0:
               stall += 1
               if stall > 3: break
           else:
               stall = 0
       return best_h, best_c

   def binary_sa(h_bin, n, deadline, T_init=0.02, T_f=0.0005, rng=None):
       if rng is None: rng = np.random.RandomState(99)
       h = h_bin.copy()
       target = n // 2
       # Ensure exactly target ones
       if h.sum() != target:
           order = np.argsort(h)[::-1]
           h = np.zeros(n)
           h[order[:target]] = 1.0
       g = compute_g(h, n)
       F = compute_F(h, n)
       dm = np.arange(2*n-1) - (n-1)
       idx_all = np.arange(2*n-1)
       best_c = float(g.max())
       best_h = h.copy()
       cur_c = best_c
       ones = np.where(h == 1)[0].tolist()
       zeros = np.where(h == 0)[0].tolist()
       T = T_init
       n_iter = 0
       while time.time() < deadline:
           a = ones[rng.randint(len(ones))]
           b = zeros[rng.randint(len(zeros))]
           # Compute new g
           delta_g = F[b] - F[a]
           new_g = g + delta_g
           new_c = float(new_g.max())
           if new_c < cur_c or rng.random() < np.exp(-(new_c - cur_c) / max(T, 1e-10)):
               # Accept
               h[a] = 0; h[b] = 1
               g = new_g
               cur_c = new_c
               # Update F
               v1 = (a + dm >= 0) & (a + dm < n)
               F[a+dm[v1], idx_all[v1]] += 1
               v2 = (a - dm >= 0) & (a - dm < n)
               F[a-dm[v2], idx_all[v2]] += 1
               v3 = (b + dm >= 0) & (b + dm < n)
               F[b+dm[v3], idx_all[v3]] -= 1
               v4 = (b - dm >= 0) & (b - dm < n)
               F[b-dm[v4], idx_all[v4]] -= 1
               # Update lists
               ones.remove(a); ones.append(b)
               zeros.remove(b); zeros.append(a)
               if cur_c < best_c:
                   best_c = cur_c
                   best_h = h.copy()
           # Cooling
           n_iter += 1
           if n_iter % 1000 == 0:
               T = max(T * 0.995, T_f)
       return best_h, best_c

   def binary_greedy(h_bin, n, deadline):
       """Try all single swaps, accept the best improvement."""
       h = h_bin.copy()
       target = n // 2
       if h.sum() != target:
           order = np.argsort(h)[::-1]
           h = np.zeros(n)
           h[order[:target]] = 1.0
       g = compute_g(h, n)
       F = compute_F(h, n)
       dm = np.arange(2*n-1) - (n-1)
       idx_all = np.arange(2*n-1)
       best_c = float(g.max())
       best_h = h.copy()
       cur_c = best_c
       ones = list(np.where(h == 1)[0])
       zeros = list(np.where(h == 0)[0])
       improved = True
       while improved and time.time() < deadline:
           improved = False
           best_delta = 0
           best_a = -1; best_b = -1
           for a in ones:
               for b in zeros:
                   delta_g = F[b] - F[a]
                   new_c = float((g + delta_g).max())
                   if new_c < cur_c - 1e-14:
                       if new_c < best_delta:
                           best_delta = new_c
                           best_a = a; best_b = b
           if best_a >= 0:
               h[best_a] = 0; h[best_b] = 1
               g = g + F[best_b] - F[best_a]
               # Update F
               v1 = (best_a + dm >= 0) & (best_a + dm < n)
               F[best_a+dm[v1], idx_all[v1]] += 1
               v2 = (best_a - dm >= 0) & (best_a - dm < n)
               F[best_a-dm[v2], idx_all[v2]] += 1
               v3 = (best_b + dm >= 0) & (best_b + dm < n)
               F[best_b+dm[v3], idx_all[v3]] -= 1
               v4 = (best_b - dm >= 0) & (best_b - dm < n)
               F[best_b-dm[v4], idx_all[v4]] -= 1
               ones.remove(best_a); ones.append(best_b)
               zeros.remove(best_b); zeros.append(best_a)
               cur_c = best_delta
               if cur_c < best_c:
                   best_c = cur_c
                   best_h = h.copy()
               improved = True
       return best_h, best_c
