from scipy.optimize import minimize

    c = np.correlate(h, 1 - h, mode='full') * dx

    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h

def lag_gradient(h, n, k):
    """Gradient of correlation at lag k with respect to h, times dx."""
    dx = 2.0 / n
    s = abs(k - (n - 1))
    g = np.zeros(n)
    if s == 0:
        g[:] = 1.0 - 2.0 * h
    else:
        g[s:] += (1.0 - h[:n-s])
        g[:n-s] -= h[s:]
    return g * dx

    return float(np.sum(x) - n / 2.0)

def sum_con_jac(x, n):
    return np.ones(n)

    """SLSQP with log-sum-exp smoothing of the max."""
    dx = 2.0 / n
    def obj(x):
        hx = np.clip(x, 0, 1)
        c = np.correlate(hx, 1 - hx, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        soft = (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
        return m + soft * 0.01  # blend: primarily max, plus smoothing

        shifted = alpha * (c - m)

        return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha

            if time.time() > deadline - 20:
                break
            scale = 0.01 + 0.07 * (r / max(1, n_restarts - 1))
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(35, (deadline - time.time() - 20) * 0.25)
            if alloc < 12:
                break
            h_opt, c_opt = optimize_sequence(hp, n, time.time() + alloc)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        h, c = smart_local(best_h, n, deadline - 2)
        if c < best_c:
            best_c = c
            best_h = h
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    if best_h is None:
        n = n_init
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)
    return best_h, best_c, len(best_h)
