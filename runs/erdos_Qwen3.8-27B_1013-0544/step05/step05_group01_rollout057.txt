import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_correlations(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float)
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_gradient(h, c_vals, alpha, dx):
    """Compute gradient of smooth max of correlations.
    c_vals: array of shape (2n-1,) with c_k values
    Returns gradient of shape (n,)
    """
    n = len(h)
    # Compute weights
    shifted = alpha * (c_vals - c_vals.max())
    max_shift = shifted.max()
    exp_vals = np.exp(shifted - max_shift)
    w = exp_vals / exp_vals.sum()
    
    # w is indexed from 0 to 2n-2, corresponding to k = -(n-1) to n-1
    # w[k + (n-1)] is the weight for lag k
    
    grad = np.zeros(n)
    for j in range(n):
        # First term: sum_{k: 0<=j+k<n} w[k+(n-1)] * (1 - h[j+k])
        k_lo = max(-(n-1), -j)
        k_hi = min(n-1, n-1-j)
        if k_lo <= k_hi:
            ks = np.arange(k_lo, k_hi + 1)
            grad[j] += np.sum(w[ks + (n-1)] * (1.0 - h[j + ks]))
        
        # Second term: -sum_{k: 0<=j-k<n} w[k+(n-1)] * h[j-k]
        k_lo2 = max(-(n-1), j - (n-1))
        k_hi2 = min(n-1, j)
        if k_lo2 <= k_hi2:
            ks2 = np.arange(k_lo2, k_hi2 + 1)
            grad[j] -= np.sum(w[ks2 + (n-1)] * h[j - ks2])
    
    return grad * dx

def adam_gradient_descent(h, n, deadline, rng, alpha_init=500, lr_init=0.01):
    """ADAM optimizer for the smooth max objective with projection."""
    h = project_feasible(h.copy(), n)
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    alpha = alpha_init
    lr = lr_init
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    t = 0
    patience = 0
    
    while time.time() < deadline:
        t += 1
        c_vals = compute_correlations(h, n)
        grad = compute_gradient(h, c_vals, alpha, dx)
        
        # ADAM update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            patience = 0
            # Increase alpha slowly
            alpha *= 1.02
        else:
            patience += 1
            if patience > 50:
                lr *= 0.9
                alpha *= 1.05
                patience = 0
        
        h = h_new
        
        # Occasionally reset ADAM state to escape local minima
        if t % 200 == 0 and patience > 30:
            m = np.zeros(n)
            v = np.zeros(n)
            lr = max(lr_init * 0.5, lr * 0.5)
    
    return best_h, best_c

    """
    coeffs: array of shape (2*m,) where coeffs[2k] = a_k, coeffs[2k+1] = b_k

import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_correlations(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float)
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def fourier_h(coeffs, n):
    x = np.linspace(0.0, 2.0, n)
    h = np.full(n, 0.5)
    m = len(coeffs) // 2
    for k in range(m):
        h += coeffs[2*k] * np.cos((k+1)*np.pi*x) + coeffs[2*k+1] * np.sin((k+1)*np.pi*x)
    return h

def fourier_obj(coeffs, n):
    h = fourier_h(coeffs, n)
    h = project_feasible(np.clip(h, 0.0, 1.0), n)
    return compute_c5(h, n)

def gradient_from_weights(h, w, dx):
    n = len(h)
    grad = np.zeros(n)
    for j in range(n):
        k_lo = max(-(n-1), -j)
        k_hi = min(n-1, n-1-j)
        if k_lo <= k_hi:
            ks = np.arange(k_lo, k_hi+1)
            grad[j] += np.sum(w[ks+(n-1)]*(1.0-h[j+ks]))
        k_lo2 = max(-(n-1), j-(n-1))
        k_hi2 = min(n-1, j)
        if k_lo2 <= k_hi2:
            ks2 = np.arange(k_lo2, k_hi2+1)
            grad[j] -= np.sum(w[ks2+(n-1)]*h[j-ks2])
    return grad * dx

def adam_optimize(h, n, deadline, alpha0=200.0, lr0=0.05):
    h = project_feasible(h.copy(), n)
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alpha = alpha0
    lr = lr0
    b1, b2, eps = 0.85, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    stall = 0
    while time.time() < deadline:
        t += 1
        cv = compute_correlations(h, n)
        sh = alpha * (cv - cv.max())
        exp_v = np.exp(sh - sh.max())
        w = exp_v / exp_v.sum()
        g = gradient_from_weights(h, w, dx)
        m = b1*m + (1-b1)*g
        v = b2*v + (1-b2)*g*g
        mh = m/(1-b1**t)
        vh = v/(1-b2**t)
        h2 = h - lr*mh/(np.sqrt(vh)+eps)
        h2 = project_feasible(h2, n)
        c2 = compute_c5(h2, n)
        if c2 < best_c - 1e-15:
            best_c = c2
            best_h = h2.copy()
            stall = 0
            alpha *= 1.01
        else:
            stall += 1
            if stall > 80:
                lr *= 0.85
                alpha *= 1.03
                stall = 0
            if stall > 200:
                m[:] = 0
                v[:] = 0
                stall = 0
        h = h2
    return best_h, best_c
