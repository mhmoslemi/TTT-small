import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = np.minimum(1.0, h + diff / s * room)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = np.maximum(0.0, h + diff / s * h)
    return np.clip(h, 0.0, 1.0)

def get_active_lags(c, threshold=0.0):
    """Get indices where c is close to max."""
    m = c.max()
    idx = np.where(c >= m - threshold)[0]
    return idx

def subgradient_step(h, n, lr, n_active_frac=0.2):
    """One subgradient descent step."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    # Get active lags (those contributing to max or close to it)
    idx = np.where(c >= m * (1 - n_active_frac))[0]
    # Compute subgradient: sum over active lags of (1-h[i+k] - h[i-k])
    grad = np.zeros(n)
    for mi in idx:
        k = mi - (n - 1)  # shift
        # d/dh[i] of sum_j h[j]*(1-h[j+k]) = (1-h[i+k]) - h[i-k]
        # for valid indices
        for i in range(n):
            ik = i + k
            ikm = i - k
            g = 0.0
            if 0 <= ik < n:
                g += 1 - h[ik]
            if 0 <= ikm < n:
                g -= h[ikm]
            grad[i] += g
    grad *= dx
    # Projected subgradient descent
    h_new = h - lr * grad
    h_new = project_feasible(h_new, n)
    return h_new, m

    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = np.minimum(1.0, h + diff / s * room)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = np.maximum(0.0, h + diff / s * h)
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth(h, n, alpha, maxiter, deadline):

    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx

def binary_exchange_search(h_init, n, deadline, seed):
    target = n // 2
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_max = int(corr.max())
    best_max = cur_max
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    j = np.arange(-(n - 1), n)
    ones_set = set(np.where(h == 1)[0].tolist())
    zeros_set = set(np.where(h == 0)[0].tolist())
    beta = 1.5
    total_iter = 0

    while time.time() < deadline:
        if not zeros_set or not ones_set:
            break
        a = zeros_set and list(zeros_set)[rng.randint(len(zeros_set))]
        b = ones_set and list(ones_set)[rng.randint(len(ones_set))]

        ia = a - j
        ma = (ia >= 0) & (ia < n)
        delta = np.zeros(2 * n - 1)
        delta[ma] += 1.0 - h[ia[ma]]
        ja = a + j
        mj = (ja >= 0) & (ja < n)
        delta[mj] -= h[ja[mj]]
        ib = b - j
        mb = (ib >= 0) & (ib < n)
        delta[mb] -= 1.0 - h[ib[mb]]
        jb = b + j
        mk = (jb >= 0) & (jb < n)
        delta[mk] += h[jb[mk]]
        delta[n - 1] = 0.0
        delta[n - 1 + a - b] += 1.0
        delta[n - 1 + b - a] += 1.0

        new_corr = corr + delta
        new_max = int(new_corr.max())

        if new_max < cur_max:
            accept = True
        elif new_max == cur_max:
            accept = rng.rand() < 0.5
        else:
            accept = rng.rand() < np.exp(-beta * (new_max - cur_max))

        if accept:
            h[a], h[b] = 1.0, 0.0
            corr = new_corr
            cur_max = new_max
            zeros_set.discard(a)
            zeros_set.add(b)
            ones_set.discard(b)
            ones_set.add(a)
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()

        total_iter += 1
        if total_iter % 2000 == 0:
            beta = min(15.0, beta * 1.005)

    return best_h, best_max * dx
