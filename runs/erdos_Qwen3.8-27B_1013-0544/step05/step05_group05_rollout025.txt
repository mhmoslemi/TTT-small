def compute_c5(h, n):

    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-13:
                break
            h = h + (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-13:
                break
            h = h + (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def smooth_objective(h, n, alpha):
    g = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    m = g.max()
    sh = g - m
    return m + (sh.max() + np.log(np.exp(sh - sh.max()).sum())) / alpha

def slsqp_step(h0, n, alpha, maxiter, deadline):
    def obj(x):
        return smooth_objective(x, n, alpha)
    def con(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h0, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con}],
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h0

def subgradient_polish(h0, n, deadline, max_iter=50000):
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    lr = 0.01
    dx = 2.0 / n
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = np.correlate(best_h, 1.0 - best_h, mode='full') * dx
        m = g.max()
        k = int(np.argmax(g))
        lag = k - (n - 1)
        grad = np.zeros(n)
        if lag >= 0:
            if n - lag > 0:
                grad[lag:] += dx * (1.0 - best_h[:n - lag])
                grad[:n - lag] -= dx * best_h[lag:]
        else:
            to = -lag
            if n - to > 0:
                grad[:n - to] += dx * (1.0 - best_h[to:])
            grad[to:] -= dx * best_h[:n - to]
        ng = np.linalg.norm(grad)
        if ng < 1e-14:
            break
        grad /= ng
        improved = False
        for f in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01, 0.005, 0.001]:
            step = lr * f
            ht = project_feasible(best_h - step * grad, n)
            ct = compute_c5(ht, n)
            if ct < best_c - 1e-13:
                best_h = ht
                best_c = ct
                improved = True
                break
        if not improved:
            lr *= 0.9
            if lr < 1e-14:
                break
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    g = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = g.max()
    sh = g - m
    return m + (sh.max() + np.log(np.exp(sh - sh.max()).sum())) / alpha

def slsqp_opt(h0, n, alpha, maxiter, deadline):
    def obj(x):
        return smooth_obj(x, n, alpha)
    def obj_grad(x):
        g = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
        m = g.max()
        sh = g - m
        w = np.exp(sh - sh.max())
        w /= w.sum()
        grad = np.zeros(2 * n - 1)
        for k in range(2 * n - 1):
            grad[k] = w[k] * (2.0 / n)
        lag = np.arange(2 * n - 1) - (n - 1)
        h_grad = np.zeros(n)
        for i in range(n):
            vals = []
            for k_idx in range(2 * n - 1):
                lag_val = k_idx - (n - 1)
                if lag_val >= 0:
                    j = i - lag_val
                    if 0 <= j < n:
                        vals.append(grad[k_idx] * (1.0 - x[j]))
                else:
                    j = i + (-lag_val)
                    if 0 <= j < n:
                        vals.append(grad[k_idx] * (-x[j]))
            h_grad[i] = sum(vals) if vals else 0.0
        return h_grad
    def con_fun(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h0, method='SLSQP', jac=obj_grad,
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con_fun}],
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return slsqp_no_grad(h0, n, alpha, maxiter, deadline)

def slsqp_no_grad(h0, n, alpha, maxiter, deadline):
    def obj(x):
        return smooth_obj(x, n, alpha)
    def con_fun(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h0, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con_fun}],
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h0

def subgrad_polish(h0, n, deadline, max_iter=100000):
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    lr = 0.05
    dx = 2.0 / n
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = np.correlate(best_h, 1.0 - best_h, mode='full') * dx
        m = g.max()
        k = int(np.argmax(g))
        lag = k - (n - 1)
        grad = np.zeros(n)
        if lag >= 0:
            if n - lag > 0:
                grad[lag:] += dx * (1.0 - best_h[:n - lag])
                grad[:n - lag] -= dx * best_h[lag:]
        else:
            to = -lag
            if n - to > 0:
                grad[:n - to] += dx * (1.0 - best_h[to:])
            grad[to:] -= dx * best_h[:n - to]
        ng = np.linalg.norm(grad)
        if ng < 1e-15:
            break
        grad /= ng
        improved = False
        for f in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.001, 0.0005]:
            step = lr * f
            ht = project_feasible(best_h - step * grad, n)
            ct = compute_c5(ht, n)
            if ct < best_c - 1e-14:
                best_h = ht
                best_c = ct
                improved = True
                break
        if not improved:
            lr *= 0.85
            if lr < 1e-15:
                break
    return best_h, best_c

def multi_lag_subgrad(h0, n, deadline, max_iter=50000, top_k=5):
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    lr = 0.02
    dx = 2.0 / n
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = np.correlate(best_h, 1.0 - best_h, mode='full') * dx
        m = g.max()
        top_idx = np.argsort(g)[::-1][:top_k]
        weights = g[top_idx] - m + 0.01
        weights = weights / weights.sum()
        grad = np.zeros(n)
        for wi, k in zip(weights, top_idx):
            lag = int(k) - (n - 1)
            if lag >= 0:
                if n - lag > 0:
                    grad[lag:] += wi * dx * (1.0 - best_h[:n - lag])
                    grad[:n - lag] -= wi * dx * best_h[lag:]
            else:
                to = -lag
                if n - to > 0:
                    grad[:n - to] += wi * dx * (1.0 - best_h[to:])
                grad[to:] -= wi * dx * best_h[:n - to]
        ng = np.linalg.norm(grad)
        if ng < 1e-15:
            break
        grad /= ng
        improved = False
        for f in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01, 0.005, 0.001]:
            step = lr * f
            ht = project_feasible(best_h - step * grad, n)
            ct = compute_c5(ht, n)
            if ct < best_c - 1e-14:
                best_h = ht
                best_c = ct
                improved = True
                break
        if not improved:
            lr *= 0.85
            if lr < 1e-15:
                break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
