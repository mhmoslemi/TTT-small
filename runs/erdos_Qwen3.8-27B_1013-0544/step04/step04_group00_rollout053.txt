def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_binary(h, n, rng):
    """Project h to binary with exactly n//2 ones."""
    target = n // 2
    h_bin = (h > 0.5).astype(float)
    ones = np.where(h_bin == 1)[0]
    zeros = np.where(h_bin == 0)[0]
    if len(ones) > target:
        excess = rng.choice(ones, len(ones) - target, replace=False)
        h_bin[excess] = 0
    elif len(ones) < target:
        need = target - len(ones)
        add = rng.choice(zeros, need, replace=False)
        h_bin[add] = 1
    return h_bin

def legendre_set(p):
    """Generate quadratic residues mod p (size (p-1)/2)."""
    result = set()
    for x in range(1, p):
        result.add(x * x % p)
    return sorted(result)

def systematic_hill_climb(h, n, deadline, max_sweeps=100):
    """Systematic pair-swap hill climbing."""
    best_c = compute_c5(h, n)
    target = n // 2
    ones = sorted(np.where(h == 1)[0].tolist())
    zeros = sorted(np.where(h == 0)[0].tolist())

            break
        improved = False
        new_ones = list(ones)
        new_zeros = list(zeros)
        for i in range(len(zeros)):
            if time.time() > deadline:
                break
            for j in range(len(ones)):
                zi = zeros[i]
                oj = ones[j]
                h[zi] = 1
                h[oj] = 0
                c = compute_c5(h, n)

                    improved = True
                    new_ones[j] = zi

                else:
                    h[zi] = 0
                    h[oj] = 1
        if not improved:

        ones = new_ones
        zeros = new_zeros

                    best_h = h.copy()
            else:

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def to_binary_ranks(h, n):
    """Project to binary using ranking (top n/2 become 1)."""
    target = n // 2
    h_b = np.zeros(n)
    idx = np.argsort(h)[::-1][:target]
    h_b[idx] = 1.0
    return h_b

def hill_climb(h, n, deadline, max_sweeps=200):
    """Systematic pair-swap hill climbing."""
    best_c = compute_c5(h, n)
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for iz in range(len(zeros)):
            if time.time() > deadline:
                break
            z = zeros[iz]
            for io in range(len(ones)):
                o = ones[io]
                h[z] = 1.0
                h[o] = 0.0
                c = compute_c5(h, n)
                if c < best_c:
                    best_c = c
                    improved = True
                    zeros[iz] = o
                    ones[io] = z
                else:
                    h[z] = 0.0
                    h[o] = 1.0
        if not improved:
            break
    
    return h, best_c

    best_c = compute_c5(h, n)
    cur_c = best_c
    T = T0
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    n_ones = n // 2
    kick_interval = 5000
    it = 0
    it_since_kick = 0
    
    while time.time() < deadline:
        batch = 500
        for _ in range(batch):
            io = rng.randint(n_ones)
            iz = rng.randint(n - n_ones)
            o = ones[io]
            z = zeros[iz]
            h[o] = 0.0
            h[z] = 1.0
            new_c = compute_c5(h, n)
            if new_c <= cur_c or rng.random() < np.exp(-(new_c - cur_c) / max(T, 1e-15)):
                cur_c = new_c
                ones[io] = z
                zeros[iz] = o
                if new_c < best_c:
                    best_c = new_c
                    best_h = h.copy()
            else:
                h[o] = 1.0
                h[z] = 0.0
        it += batch
        it_since_kick += batch
        T = max(T * cool, 1e-12)
        
        # Kick: random perturbation

            n_kick = int(n * kick_frac)
            kick_ones = rng.choice(ones, n_kick//2, replace=False)
            kick_zeros = rng.choice(zeros, n_kick//2, replace=False)

            ones = list(np.where(h == 1)[0])
            zeros = list(np.where(h == 0)[0])
            cur_c = compute_c5(h, n)

def coord_descent_continuous(h, n, deadline, max_sweeps=50):
    """Coordinate descent with discrete values {0, 0.2, 0.4, 0.6, 0.8, 1}."""
    best_c = compute_c5(h, n)
    target_sum = n / 2.0
    values = [0.0, 0.2, 0.4, 0.5, 0.6, 0.8, 1.0]
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            old_val = h[i]
            best_v = old_val
            best_cv = compute_c5(h, n)
            for v in values:
                if v == old_val:
                    continue
                # Check if changing h[i] to v keeps sum close to target
                new_sum = h.sum() - old_val + v
                if abs(new_sum - target_sum) > 0.5:
                    continue
                h[i] = v
                c = compute_c5(h, n)
                if c < best_cv:
                    best_cv = c
                    best_v = v
            if best_v != old_val:
                h[i] = best_v
                improved = True
            else:
                h[i] = old_val
        if best_cv < best_c:
            best_c = best_cv
