The dual (relaxed) is: min_{λ≥0, Σλ=1} max_{0≤h≤1, Σh=n/2} Σ_k λ_k c_k(h)

For a fixed λ, the inner problem is: minimize Σ_k λ_k c_k(h) = Σ_k λ_k (Σ h[i] - Σ h[i]h[i+k])
= constant - Σ_k λ_k Σ h[i]h[i+k]
= constant - h^T (Σ_k λ_k A_k) h

where A_k is the matrix with (A_k)_{i,j} = 1 if j = i+k (for valid i,j).

So the dual inner problem is: maximize h^T B h - Σ_k λ_k Σ_i h[i] (where B = Σ_k λ_k A_k)
subject to 0≤h≤1, Σh = n/2.

This is a concave maximization (since B is a sum of permutation matrices, it's symmetric and... hmm, is it positive semidefinite? Not necessarily).

This dual approach might not be directly useful, but it suggests that for a fixed set of "active" lags with weights λ, the optimal h is the one that maximizes a quadratic form subject to box and sum constraints. This is a non-convex QP in general.

However, if B happens to be positive semidefinite (or we can make it so by choosing λ), then the problem becomes a convex QP and we can solve it exactly.

This is getting too theoretical. Let me focus on practical improvements.

**Practical plan for improvement**:

The main issue with the previous code is likely that it doesn't spend enough time on the final polishing. The SLSQP phases take a lot of time (each call with n=480 variables can be slow), leaving less time for the subgradient polishing.

My strategy:
1. **Minimize SLSQP time**: Use only 1-2 SLSQP calls with moderate maxiter
2. **Maximize subgradient time**: Spend most of the budget on careful subgradient
3. **Adaptive step sizing**: Use a geometric sequence starting from the step that just worked
4. **Multiple gradient directions**: Alternate between top-1, top-3, top-5, top-10 lags
5. **Occasional "jumps"**: Take a larger step in a random direction to escape plateaus

Let me also add a key improvement: **after each successful step, remember the step size and use it as the starting point for the next iteration's step size search**. This is a form of adaptive step size.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    if abs(h.sum() - target) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def slqp_solve(h0, n, beta, maxiter):
    target = n / 2.0
    dx = 2.0 / n
    def obj(x):
        x = np.clip(x, 0, 1)
        c = np.correlate(x, 1.0 - x, mode="full")
        m = c.max()
        v = np.clip(beta * (c - m), -500, 300)
        return (m + np.log(np.exp(v).sum()) / beta) * dx
    def grad(x):
        x = np.clip(x, 0, 1)
        c = np.correlate(x, 1.0 - x, mode="full")
        m = c.max()
        v = np.clip(beta * (c - m), -500, 300)
        e = np.exp(v)
        se = e.sum()
        w = e / se
        g = np.zeros(n)
        for j in range(len(c)):
            wj = w[j]
            if wj < 1e-10:
                continue
            lag = j - (n - 1)
            if lag == 0:
                g += wj * (1.0 - 2.0 * x)
            elif lag > 0:
                g[lag:] += wj * (1.0 - x[:-lag])
                g[:n - lag] -= wj * x[lag:]
            else:
                m2 = -lag
                g[:n - m2] += wj * (1.0 - x[m2:])
                g[m2:] -= wj * x[:n - m2]
        return g * dx
    bounds = [(0, 1)] * n
    res = minimize(obj, h0, jac=grad, method='SLSQP', bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - target,
                                 'jac': lambda: np.ones(n)}],
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return project_sum(res.x, target)

def subgrad_polish(h, target, n, t0, budget, max_iter, n_active_start, use_adaptive):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    last_good_step = 0.01
    no_improve = 0
    for it in range(max_iter):
        if budget - (time.time() - t0) < 20:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        na = min(n_active_start + (it // 500) * 2, 20)
        top_idx = np.argsort(corr)[::-1][:na]
        g = np.zeros(n)
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            if lag == 0:
                g += (1.0 - 2.0 * best_h)
            elif lag > 0:
                g[lag:] += (1.0 - best_h[:-lag])
                g[:n - lag] -= best_h[lag:]
            else:
                m2 = -lag
                g[:n - m2] += (1.0 - best_h[m2:])
                g[m2:] -= best_h[:n - m2]
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        improved = False
        if use_adaptive:
            base = last_good_step
            muls = [1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.01, 2.0, 4.0]
        else:
            base = 0.01
            muls = [1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.01]
        for mult in muls:
            step = base * mult
            ht = project_sum(best_h - step * g, target)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-15:
                best_h = ht
                best_c5 = c5t
                last_good_step = step
                improved = True
                break
        if not improved:
            no_improve += 1
            if no_improve > 100 and it % 60 == 59:
                scale = 0.004 * max(0.3, 1.0 - it / 10000.0)
                ht = project_sum(best_h + np.random.randn(n) * scale, target)
                c5t = compute_c5(ht)
                if c5t < best_c5:
                    best_h = ht
                    best_c5 = c5t
                    no_improve = 0
            if no_improve > 1500:
                break
        else:
            no_improve = 0
    return best_h, best_c5

def coordinate_refine(h, target, n, t0, budget, max_passes):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    for p in range(max_passes):
        if budget - (time.time() - t0) < 10:
            break
        improved_any = False
        order = np.random.permutation(n) if p > 0 else np.arange(n)
        for i in order:
            if budget - (time.time() - t0) < 5:
                break
            hi = best_h[i]
            if hi < 1e-8 or hi > 1.0 - 1e-8:
                continue
            corr = np.correlate(best_h, 1.0 - h and 1.0 - best_h, mode="full")
            d = np.zeros(2 * n - 1)
            d[n - 1] = 1.0 - 2.0 * hi
            for k in range(1, n):
                if i + k < n:
                    d[n - 1 + k] = 1.0 - best_h[i + k]
                if i - k >= 0:
                    d[n - 1 - k] = -best_h[i - k]
            lo_d, hi_d = -hi, 1.0 - hi
            ng = 40
            deltas = np.linspace(lo_d, hi_d, ng)
            best_val = np.max(corr)
            best_delta = 0.0
            for dt in deltas:
                v = np.max(corr + dt * d)
                if v < best_val:
                    best_val = v
                    best_delta = dt
            if best_delta != 0.0:
                ht = best_h.copy()
                ht[i] = np.clip(hi + best_delta, 0.0, 1.0)
                ht = project_sum(ht, target)
                c5t = compute_c5(ht)
                if c5t < best_c5 - 1e-14:
                    best_h = ht
                    best_c5 = c5t
                    improved_any = True
        if not improved_any and p > 2:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in dir() and initial_h_values is not None:
        best_h = np.asarray(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(240, 0.5)

    n = len(best_h)
    target = n / 2.0
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP for initial polish (limited time)
    for beta in [300, 2000]:
        if budget_s - (time.time() - t0) < 250:
            break
        h_new = slqp_solve(best_h, n, beta, maxiter=3000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 2: Intensive adaptive subgradient
    best_h, best_c5 = subgrad_polish(best_h, target, n, t0, budget_s,
                                     max_iter=20000, n_active_start=5, use_adaptive=True)

    # Phase 3: Coordinate refinement pass
    best_h, best_c5 = coordinate_refine(best_h, target, n, t0, budget_s, max_passes=3)

    # Phase 4: More subgradient with different parameters
    best_h, best_c5 = subgrad_polish(best_h, target, n, t0, budget_s,
                                     max_iter=15000, n_active_start=3, use_adaptive=True)

    # Phase 5: Try alternative resolution n=400
    if budget_s - (time.time() - t0) > 200:
        x_old = np.linspace(0, 2, n)
        x_new = np.linspace(0, 2, 400)
        h400 = project_sum(np.interp(x_new, x_old, best_h), 200.0)
        h400 = slqp_solve(h400, 400, 500, maxiter=2000)
        c5_400 = compute_c5(h400)
        h400, c5_400 = subgrad_polish(h400, 200.0, 400, t0, budget_s,
                                      max_iter=8000, n_active_start=6, use_adaptive=True)
        if c5_400 < best_c5:
            best_h = h400
            best_c5 = c5_400
            n = 400
            target = 200.0

    # Phase 6: Try n=560
    if budget_s - (time.time() - t0) > 150:
        x_old = np.linspace(0, 2, n)
        x_new = np.linspace(0, 2, 560)
        h560 = project_sum(np.interp(x_new, x_old, best_h), 280.0)
        if budget_s - (time.time() - t0) > 100:
            h560 = slqp_solve(h560, 560, 300, maxiter=1500)
        c5_560 = compute_c5(h560)
        h560, c5_560 = subgrad_polish(h560, 280.0, 560, t0, budget_s,
                                      max_iter=5000, n_active_start=5, use_adaptive=True)
        if c5_560 < best_c5:
            best_h = h560
            best_c5 = c5_560
            n = 560
            target = 280.0

    # Phase 7: Final ultra-fine polish
    best_h, best_c5 = subgrad_polish(best_h, target, len(best_h), t0, budget_s,
                                     max_iter=30000, n_active_start=3, use_adaptive=True)

    # Coordinate refinement as final touch
    if budget_s - (time.time() - t0) > 15:
        best_h, best_c5 = coordinate_refine(best_h, target, len(best_h), t0, budget_s,
                                            max_passes=2)

    return best_h, best_c5, len(best_h)
