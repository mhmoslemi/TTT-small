def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_corr(h, n):
    """Return the full correlation array (before dx scaling)."""
    return np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max(c, alpha):
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

def slsqp_step(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        return smooth_max(np.correlate(x, 1 - x, mode='full') * dx, alpha)
    def grad_obj(x):
        # Approximate gradient
        eps = 1e-7
        g = np.zeros(n)
        c0 = smooth_max(np.correlate(x, 1 - x, mode='full') * dx, alpha)
        for i in range(n):
            x[i] += eps
            g[i] = (smooth_max(np.correlate(x, 1 - x, mode='full') * dx, alpha) - c0) / eps
            x[i] -= eps
        return g
    def cons_fun(x):
        return x.sum() - n / 2.0
    try:
        res = minimize(obj, h, method='SLSQP', bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cons_fun}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def binary_local_search(h, n, deadline, max_sweeps=300):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    best_c = compute_c5(h_bin, n)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        dx = 2.0 / n
        c_full = compute_corr(h_bin, n) * dx
        current_max = c_full.max()
        
        # Try top few lags
        improved = False
        top_lags = np.argsort(c_full)[::-1][:5]
        
        for max_lag in top_lags:
            if improved:
                break
            if time.time() > deadline:
                break
            s = int(max_lag) - (n - 1)
            ones = np.where(h_bin == 1)[0]
            zeros = np.where(h_bin == 0)[0]
            
            # Candidates that would reduce the overlap at this lag
            valid_i = [i for i in zeros if 0 <= i + s < n and h_bin[i + s] == 1]
            valid_j = [j for j in ones if 0 <= j + s < n and h_bin[j + s] == 0]
            
            if not valid_i or not valid_j:
                continue
            
            # Limit for speed
            valid_i = valid_i[:25]
            valid_j = valid_j[:25]
            
            best_c_new = current_max
            best_pair = None
            for i in valid_i:
                for j in valid_j:
                    h_bin[i], h_bin[j] = 1.0, 0.0
                    c_new = float(np.max(compute_corr(h_bin, n)) * dx)
                    h_bin[i], h_bin[j] = 0.0, 1.0
                    if c_new < best_c_new - 1e-15:
                        best_c_new = c_new
                        best_pair = (i, j)
            
            if best_pair:
                i, j = best_pair
                h_bin[i], h_bin[j] = 1.0, 0.0
                best_c = best_c_new
                improved = True
    
    return h_bin, best_c

def pair_refine_continuous(h, n, deadline):
    """Try small continuous adjustments to pairs."""
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    deltas = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    
    for d in deltas:
        if time.time() > deadline:
            break
        for sweep in range(10):
            if time.time() > deadline:
                break
            improved = False
            c_full = compute_corr(h, n) * dx
            max_lag = int(c_full.argmax())
            s = max_lag - (n - 1)
            
            # Find pairs where moving mass reduces the max
            for i in range(n):
                if time.time() > deadline:
                    break
                for j in range(i + 1, n, max(1, n // 50)):  # sample j values
                    for sign in (1, -1):
                        hi, hj = h[i] + sign * d, h[j] - sign * d
                        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                            h[i], h[j] = hi, hj
                            c_new = float(np.max(compute_corr(h, n)) * dx)
                            if c_new < best_c - 1e-15:
                                best_c = c_new
                                improved = True
                            else:
                                h[i], h[j] = h[i] - sign * d, h[j] + sign * d
            if not improved:
                break
    
    return h, best_c

def subgrad_polish(h, n, deadline, max_iter=3000):
    """Subgradient method focusing on the worst lag."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    lr = 0.5
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c_full = compute_corr(h, n) * dx
        m = c_full.max()
        ks = int(c_full.argmax())
        
        # Compute subgradient at the max lag
        # c(ks) = sum_i h[i] * (1 - h_ext[i + ks - (n-1)])
        # dc/dh[i] = (1 - h_ext[i + ks - (n-1)]) + h[i] * (-1) * (delta if i+s == i' for some i')
        # Actually: c(ks) = sum_i h[i] - sum_i h[i]*h_ext[i+s] where s = ks-(n-1)
        # dc/dh[i] = 1 - h_ext[i+s] (from the first term)
        #            - h_ext[i+s] (from the second term, derivative of h[i]*h[i+s] wrt h[i])
        #            - sum_j h[j] * (delta_{j+s, i}) (from the second term, derivative wrt h[i] when i = j+s)
        # = 1 - 2*h_ext[i+s] - h_ext[i-s] (where h_ext[x-s] means h[x] if x-s is valid, i.e., h[i-s] if 0<=i-s<n)
        
        s = ks - (n - 1)
        grad = np.zeros(n)
        for i in range(n):
            # Term 1: 1 - h_ext[i+s]
            t1 = 1.0
            if 0 <= i + s < n:
                t1 = 1.0 - h[i + s]
            # Term 2: -h_ext[i+s] (derivative of -h[i]*h[i+s] wrt h[i])
            t2 = 0.0
            if 0 <= i + s < n:
                t2 = -h[i + s]
            # Term 3: -h_ext[i-s] (derivative of -h[j]*h[j+s] wrt h[i] where i=j+s, so j=i-s)
            t3 = 0.0
            if 0 <= i - s < n:
                t3 = -h[i - s]
            grad[i] = t1 + t2 + t3
        
        ng = np.linalg.norm(grad)
        if ng < 1e-12:
            break
        
        # Line search
        found = False
        for factor in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02]:
            step = lr * factor / ng
            ht = project_feasible(h - step * grad, n)
            ct = compute_c5(ht, n)
            if ct < m - 1e-13:
                h = ht
                found = True
                break
        if not found:
            lr *= 0.9
            if lr < 1e-12:
                break
        else:
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
    
    return best_h, best_c

        ks = int(c_full.argmax())
        s = ks - (n - 1)
        grad = np.zeros(n)
        for i in range(n):
            hi_s = h[i + s] if 0 <= i + s < n else 0.0
            hi_minus_s = h[i - s] if 0 <= i - s < n else 0.0
            grad[i] = (1.0 - hi_s) - hi_minus_s
        ng = np.linalg.norm(grad)
        if ng < 1e-12:
            break
        found = False
        for factor in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01, 0.005]:
            ht = project_feasible(h - (lr * factor / ng) * grad, n)
            if compute_c5(ht, n) < m - 1e-13:
                h = ht
                found = True
                break
        if not found:
            lr *= 0.92
            if lr < 1e-12:
                break
        else:
            c = compute_c5(h, n)
            if c < best_c:
                best_c, best_h = c, h.copy()
    return best_h, best_c

def multi_start_binary(h, n, deadline, rng, n_starts=4):
    best_hb, best_cb = h.copy(), compute_c5(h, n)
    for k in range(n_starts):
        if time.time() > deadline:
            break
        if k == 0:
            start = h
        else:
            scale = 0.1 * (k + 1)
            start = project_feasible(h + scale * rng.randn(n), n)
        hb, cb = binary_local_search(start, n, deadline, max_sweeps=300)
        if cb < best_cb:
            best_cb, best_hb = cb, hb.copy()
    return best_hb, best_cb

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0, h0 = 200, np.full(200, 0.5)
    best_h, best_c, n = h0.copy(), compute_c5(h0, n0), n0

    # Phase 1: SLSQP from parent (20% budget)
    p1 = t0 + 0.20 * (budget_s - 5)
    for alpha in [2000, 10000, 50000, 200000, 1000000]:
        if time.time() > p1:
            break
        h = slsqp_step(best_h, n, alpha, 800, p1)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 2: Binary search with multiple starts (35% budget)
    p2 = t0 + 0.55 * (budget_s - 5)
    if time.time() < p2 - 10:
        hb, cb = multi_start_binary(best_h, n, p2 - 5, rng, n_starts=3)
        if cb < best_c:
            best_c, best_h = cb, hb.copy()

    # Phase 3: Subgradient polish (15% budget)
    p3 = t0 + 0.70 * (budget_s - 5)
    if time.time() < p3 - 5:
        h3, c3 = subgrad_polish(best_h, n, p3)
        if c3 < best_c:
            best_c, best_h = c3, h3.copy()

    # Phase 4: SLSQP from best solution (15% budget)
    p4 = t0 + 0.85 * (budget_s - 5)
    if time.time() < p4 - 5:
        h4 = project_feasible(best_h + 0.01 * rng.randn(n), n)
        for alpha in [50000, 200000, 1000000, 5000000]:
            if time.time() > p4:
                break
            h4 = slsqp_step(h4, n, alpha, 600, p4)
            c = compute_c5(h4, n)
            if c < best_c:
                best_c, best_h = c, h4.copy()

    # Phase 5: Try higher resolution (8% budget)
    if time.time() < dl - 20:
        for nh in [n + 50, n + 100, n + 150]:
            if time.time() > dl - 25 or nh > 500:
                break
            xo = np.linspace(0, 2, n + 1)[:-1]
            xn = np.linspace(0, 2, nh + 1)[:-1]
            hh = project_feasible(np.interp(xn, xo, best_h), nh)
            hh = slsqp_step(hh, nh, 50000, 400, dl - 25)
            hbh, cbh = binary_local_search(hh, nh, dl - 15, max_sweeps=100)
            hsub, csub = subgrad_polish(hbh, nh, dl - 10, max_iter=2000)
            for hc, cc in [(hbh, cbh), (hsub, csub)]:
                if cc < best_c:
                    best_c, best_h, n = cc, hc.copy(), nh

    # Phase 6: Final SLSQP + subgrad (7% budget)
    if time.time() < dl - 3:
        h6 = slsqp_step(best_h, n, 10000000, 500, dl - 3)
        c6 = compute_c5(h6, n)
        if c6 < best_c:
            best_c, best_h = c6, h6.copy()
        h6b, c6b = subgrad_polish(best_h, n, dl, max_iter=2000)
        if c6b < best_c:
            best_c, best_h = c6b, h6b.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
