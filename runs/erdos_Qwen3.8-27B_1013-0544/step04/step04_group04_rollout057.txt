    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 0) & (h < 1.0)
        if free.sum() == 0:
            h[:] = target_sum / len(h)
            break
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val_grad(h, beta, sum_penalty=0.0, target_sum=0.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    x_shift = np.clip(beta * (corr - m), -500, 500)
    exp_vals = np.exp(x_shift)
    sum_exp = np.sum(exp_vals)
    weights = exp_vals / sum_exp
    val = (m + np.log(sum_exp) / beta) * dx
    grad = np.zeros(n)
    active = np.where(weights > 1e-12)[0]
    for k in active:
        w = weights[k] * dx
        lag = int(k) - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif 0 < lag < n:
            grad[lag:] += w * (1.0 - h[:-lag])
            grad[:n - lag] -= w * h[lag:]
        elif -n < lag < 0:
                m2 = -lag
            grad[:n - m2] += w * (1.0 - h[m2:])
            grad[m2:] -= w * h[:n - m2]
    # Add sum penalty gradient
    if sum_penalty > 0:
        s = h.sum() - target_sum
        val += sum_penalty * s * s
        grad += 2.0 * sum_penalty * s
    return val, grad

def lbfqbs_optimize(h0, beta, sum_pen, target, maxiter=5000, time_fn=None, time_limit=None):

        h_res = project_h(h_res, n_res / 2.0)
        tgt_r = n_res / 2.0
        betas_r = [100, 500, 2000, 8000, 30000]
        maxit_r = 3000 if n_res <= 300 else 2000
        h_out, c5_out = optimize_at_resolution(h_res, tgt_r, betas_r, 50.0, maxit_r, t0, budget_s)
        if c5_out < best_c5:
            best_h = h_out
            best_c5 = c5_out

    # Phase 3: Multi-start with structured perturbations at best resolution
    n_b = len(best_h)
    tgt_b = n_b / 2.0
    n_restarts = 0
    while _time_left(t0, budget_s) > 50 and n_restarts < 10:
        if n_restarts == 0:
            h_try = best_h.copy()
        else:
            # Smooth Fourier perturbation
            x = np.arange(n_b)
            n_modes = 4
            freqs = np.random.randint(1, max(2, n_b // 6), size=n_modes)
            phases = np.random.uniform(0, 2.0 * np.pi, size=n_modes)
            amp = 0.015 * (1.0 + 0.3 * n_restarts)
            pert = np.zeros(n_b)
            for jj in range(n_modes):
                pert += amp * np.cos(2.0 * np.pi * freqs[jj] * x / n_b + phases[jj])
            pert -= pert.mean()
            h_try = project_h(best_h + pert, tgt_b)

        for beta in [3000, 15000, 60000]:
            if _time_left(t0, budget_s) < 15:
                break
            lam_r = 80.0
            h_new = lbfgsb_step(h_try, beta, lam_r, tgt_b, 4000)
            h_new = project_h(h_new, tgt_b)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
            h_try = h_new
        n_restarts += 1

    # Phase 4: Subgradient polish with full smooth-max gradient
    if _time_left(t0, budget_s) > 40:
        n_f = len(best_h)
        tgt_f = n_f / 2.0
        h_sg = best_h.copy()
        steps = [0.03, 0.015, 0.008, 0.004, 0.002, 0.001]
        no_imp = 0
        for it in range(20000):
            if _time_left(t0, budget_s) < 10:
                break
            _, g = smooth_max_and_grad(h_sg, 5000.0, 0.0, tgt_f)
            gn = np.linalg.norm(g)
            if gn < 1e-14:
                break
            g = g / gn
            found = False
            for st in steps:
                h_cand = project_h(h_sg - st * g, tgt_f)
                c5_cand = compute_c5(h_cand)
                if c5_cand < best_c5 - 1e-15:
                    h_sg = h_cand
                    best_h = h_cand.copy()
                    best_c5 = c5_cand
                    found = True
                    no_imp = 0
                    break
            if not found:
                no_imp += 1
                if no_imp > 500:
                    break

    # Phase 5: Final high-beta polish
    if _time_left(t0, budget_s) > 15:
        n_f2 = len(best_h)
        tgt_f2 = n_f2 / 2.0
        for beta in [150000, 400000]:
            if _time_left(t0, budget_s) < 8:
                break
            h_new = lbfgsb_step(best_h, beta, 150.0, tgt_f2, 5000)
            h_new = project_h(h_new, tgt_f2)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new

    return best_h, best_c5, len(best_h)
