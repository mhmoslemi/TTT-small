import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -400, 400)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def slsqp_optimize(h, target, n, beta, maxiter, time_fn):
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    
    def obj(h_in):
        c = np.correlate(h_in, 1.0 - h_in, mode="full")
        return smooth_max(c, beta) * dx
    
    def con(h_in):
        return h_in.sum() - target
    
    res = minimize(obj, h, method='SLSQP', bounds=bounds,
                  constraints=[{'type': 'eq', 'fun': con, 'jac': lambda x: np.ones(n)}],
                  options={'maxiter': maxiter, 'ftol': 1e-16})
    h_new = project_h(res.x, target)
    return h_new, compute_c5(h_new)

1. Start with a random or structured initial sequence
2. Apply iterative spectral methods (like the "split-step" or "alternating projections" method)
3. Post-process

1. Compute the current autocorrelation
2. In the "time domain": project h to [0,1] and normalize
3. In the "frequency domain": modify the spectrum to reduce the peak sidelobe
4. Repeat

3. **Alternating minimization**: Focus on top 1-3 lags
4. **Subgradient with momentum and adaptive step**
5. **Random restarts**: Perturb and re-optimize

The key insight I think is: the alternating minimization (focusing on 1-3 specific lags) is more effective than the smooth max approach because it directly addresses the active constraints.

- "Warm" phases: SLSQP (exploits local structure)
- "Cold" phases: subgradient or random (explores global structure)

import numpy as np
from scipy.optimize import minimize, Bounds
import time

# Globals for scipy
_g_n = 0
_g_target = 0.0
_g_beta = 100.0
_g_lags = []

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def _set_globals(n, lags=None, beta=None):
    global _g_n, _g_target, _g_lags, _g_beta
    _g_n = n
    _g_target = n / 2.0
    if lags is not None:
        _g_lags = lags
    if beta is not None:
        _g_beta = beta

def _obj_full(h):
    n = _g_n
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(c, _g_beta) * dx

def _obj_lags(h):

    lags = _g_lags
    vals = np.zeros(len(lags))

        else:
            m = -lag
            vals[idx] = np.sum(h[m:] * (1 - h[:n-m])) * dx
    return smooth_max(vals, _g_beta)

def _con(h):
    return h.sum() - _g_target

def _jac_con(h):

def slsqp_full(h, beta, maxiter):
    """SLSQP with smooth max over all lags."""
    _set_globals(len(h), beta=beta)
    bounds = Bounds(np.zeros(_g_n), np.ones(_g_n))
    try:
        res = minimize(_obj_full, h, method='SLSQP', bounds=bounds,
                      constraints=[{'type': 'eq', 'fun': _con, 'jac': _jac_con}],
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_h(res.x, _g_target)
        return h_new, compute_c5(h_new)
    except:
        return h, compute_c5(h)

def slsqp_lags(h, lags, beta, maxiter):
    """SLSQP with smooth max over specific lags."""
