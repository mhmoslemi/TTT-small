    n = len(h)

    except:
        pass

    c5_cur = compute_c5(h)
    stagnant = 0
    for _ in range(n_iters):
        if time.time() > deadline - 3:
            break
        corr, top_idx, grads = get_top_k_grads(h, top_k)
        corr_vals = corr[top_idx]
        h_new = solve_qp_sub(h, corr_vals, grads, n, mu)
        if h_new is None:
            break
        h_new = project_h(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur - 1e-12:
            h = h_new
            c5_cur = c5_new
            stagnant = 0
        else:
            improved = False
            for alpha in [0.7, 0.5, 0.3, 0.15, 0.05]:
                h_blend = project_h(alpha * h_new + (1.0 - alpha) * h, target)
                c5_blend = compute_c5(h_blend)
                if c5_blend < c5_cur - 1e-12:
                    h = h_blend
                    c5_cur = c5_blend
                    improved = True
                    stagnant = 0
                    break
            if not improved:
                stagnant += 1
                if stagnant > 8:
                    break
    return h, c5_cur

def _slsqp_obj(x):
    n = int(_n_box[0])
    beta = _beta_box[0]
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    vals = np.exp(np.clip(beta * (corr - m), -500, 500))
    return (m + np.log(np.sum(vals)) / beta) * (2.0 / n)

def _slsqp_grad(x):
    n = int(_n_box[0])
    beta = _beta_box[0]
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    vals = np.exp(np.clip(beta * (corr - m), -500, 500))
    s = np.sum(vals)
    weights = vals / s
    g = np.zeros(n)
    active = np.where(weights > 1e-6)[0]
    for k in active:
        lag = int(k) - (n - 1)
        w = weights[k] * dx
        if lag == 0:
            g += w * (1.0 - 2.0 * x)
        elif lag > 0:
            g[lag:] += w * (1.0 - x[:-lag])
            g[:n - lag] -= w * x[lag:]
        else:
            L = -lag
            g[:n - L] += w * (1.0 - x[L:])
            g[L:] -= w * x[:n - L]
    return g

def _slsqp_con(x):
    return x.sum() - _n_box[0] / 2.0

def _slsqp_con_jac(x):
    return np.ones(int(_n_box[0]))

def do_slsqp_polish(h, beta, maxiter, deadline):
    n = len(h)
    _n_box[0] = float(n)
    _beta_box[0] = beta
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_con, 'jac': _slsqp_con_jac}]
    c5_cur = compute_c5(h)
    try:
        res = minimize(_slsqp_obj, h, method='SLSQP', jac=_slsqp_grad,
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None:
            h_out = project_h(res.x, n / 2.0)
            c5_out = compute_c5(h_out)
            if c5_out < c5_cur:
                return h_out, c5_out
    except Exception:
        pass
    return h, c5_cur

def do_subgrad(h, top_k, deadline, max_iter):
    n = len(h)
    target = n / 2.0
    c5_cur = compute_c5(h)
    no_imp = 0
    steps = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    for _ in range(max_iter):
        if time.time() > deadline - 3:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        nk = min(top_k, max(3, n // 25))
        top_idx = np.argsort(corr)[::-1][:nk]
        grad = np.zeros(n)
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            if lag == 0:
                grad += 1.0 - 2.0 * h
            elif lag > 0:
                grad[lag:] += 1.0 - h[:n - lag]
                grad[:n - lag] -= h[lag:]
            else:
                L = -lag
                grad[:n - L] += 1.0 - h[L:]
                grad[L:] -= h[:n - L]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        found = False
        for step in steps:
            ht = project_h(h - step * grad, target)
            c5t = compute_c5(ht)
            if c5t < c5_cur - 1e-13:
                h = ht
                c5_cur = c5t
                found = True
                no_imp = 0
                break
        if not found:
            no_imp += 1
            if no_imp % 100 == 99:
                ht = project_h(h + np.random.randn(n) * 0.004, target)
                if compute_c5(ht) < c5_cur:
                    h = ht
                    c5_cur = compute_c5(h)
                    no_imp = 0
            if no_imp > 400:
                break
    return h, c5_cur

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(200, 0.5)

    best_h = project_h(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)

    # Phase 1: QP cutting-plane at native resolution
    h_work = best_h.copy()
    configs = [
        (8, 0.05, 40), (8, 0.01, 40), (12, 0.01, 40),
        (12, 0.005, 40), (16, 0.005, 40), (16, 0.002, 40),
        (20, 0.002, 35), (20, 0.001, 35), (25, 0.001, 30),
        (25, 0.0005, 30), (30, 0.0005, 25),
    ]
    for top_k, mu, n_it in configs:
        if time.time() > deadline - 30:
            break
        h_work, c5_work = do_qp_cutting(h_work, top_k, mu, n_it, deadline)
        if c5_work < best_c5:
            best_h, best_c5 = h_work.copy(), c5_work

    # Phase 2: Multi-resolution exploration
    for n_res in [200, 250, 300, 400]:
        if time.time() > deadline - 80:
            break
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_res)
        h_init = np.interp(x_new, x_old, best_h)
        h_init = project_h(h_init, n_res / 2.0)
        for top_k, mu, n_it in [(10, 0.01, 30), (15, 0.005, 30), (20, 0.001, 25)]:
            if time.time() > deadline - 40:
                break
            h_init, c5_init = do_qp_cutting(h_init, top_k, mu, n_it, deadline)
            if c5_init < best_c5:
                best_h, best_c5 = h_init.copy(), c5_init

    # Phase 3: SLSQP polish at best resolution
    if time.time() < deadline - 60:
        h_pol, c5_pol = do_slsqp_polish(best_h, 5000, 5000, deadline)
        if c5_pol < best_c5:
            best_h, best_c5 = h_pol, c5_pol

    # Phase 4: More QP after SLSQP
    if time.time() < deadline - 30:
        h_work = best_h.copy()
        for top_k, mu in [(15, 0.001), (20, 0.0005), (10, 0.001)]:
            if time.time() > deadline - 20:
                break
            h_work, c5_work = do_qp_cutting(h_work, top_k, mu, 20, deadline)
            if c5_work < best_c5:
                best_h, best_c5 = h_work.copy(), c5_work

    # Phase 5: Subgradient fine-tuning
    if time.time() < deadline - 20:
        h_sub, c5_sub = do_subgrad(best_h, 20, deadline - 10, 40000)
        if c5_sub < best_c5:
            best_h, best_c5 = h_sub, c5_sub

    # Phase 6: Final SLSQP with high beta
    if time.time() < deadline - 12:
        h_pol, c5_pol = do_slsqp_polish(best_h, 20000, 5000, deadline)
        if c5_pol < best_c5:
            best_h, best_c5 = h_pol, c5_pol

    # Phase 7: Very final QP
    if time.time() < deadline - 5:
        h_work = best_h.copy()
        h_work, c5_work = do_qp_cutting(h_work, 15, 0.0005, 15, deadline)
        if c5_work < best_c5:
            best_h, best_c5 = h_work.copy(), c5_work

    return best_h, best_c5, len(best_h)
