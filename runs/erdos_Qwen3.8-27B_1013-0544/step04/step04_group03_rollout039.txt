                if c_try < best_c5 - 1e-14:
                    cur = trial
                    best_c5 = c_try
                    improved = True
                    break
        if not improved and sw > 3:
            break
    return cur, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        n_init = 320
        h_parent = np.full(n_init, 0.5)
        h_parent[:n_init // 2] = 0.75
        h_parent[n_init // 2:] = 0.25
        h_parent = project_h(h_parent, n_init / 2.0)

    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)
    best_n = len(best_h)
    n_cur = best_n
    target = n_cur / 2.0

    # Phase 1: SLP with broad active set
    rem = budget_s - (time.time() - t0)
    if rem > 40:
        best_h, best_c5 = slp_optimize(best_h, n_cur, target, t0, budget_s,
                                       n_iter=80, active_frac=0.25, trust_init=0.15)

    # Phase 2: Subgradient escape
    rem = budget_s - (time.time() - t0)
    if rem > 20:
        best_h, best_c5 = subgrad_desc(best_h, target, n_cur, 4000, 0.02, t0, budget_s, top_k=10)

    # Phase 3: SLP with tighter active set
    rem = budget_s - (time.time() - t0)
    if rem > 40:
        best_h, best_c5 = slp_optimize(best_h, n_cur, target, t0, budget_s,
                                       n_iter=120, active_frac=0.10, trust_init=0.10)

    # Phase 4: Sharpen toward binary and re-optimize
    rem = budget_s - (time.time() - t0)
    if rem > 30:
        h_sharp = np.where(best_h > 0.5, np.clip(best_h * 1.6, 0, 1), best_h * 0.4)
        h_sharp = project_h(h_sharp, target)
        h_sharp, c_try = slp_optimize(h_sharp, n_cur, target, t0, budget_s,
                                       n_iter=60, active_frac=0.2, trust_init=0.12)
        if c_try < best_c5:
            best_h = h_sharp
            best_c5 = c_try

    # Phase 5: Multi-restart SLP
    rem = budget_s - (time.time() - t0)
    rng = np.random.default_rng(seed + 7)
    for att in range(10):
        rem = budget_s - (time.time() - t0)
        if rem < 25:
            break
        scale = 0.05 * (0.75 ** att)
        hp = best_h + rng.standard_normal(n_cur) * scale
        hp = project_h(hp, target)
        hp, c_try = slp_optimize(hp, n_cur, target, t0, budget_s,
                                  n_iter=50, active_frac=0.2, trust_init=0.12)
        if c_try < best_c5:
            best_h = hp
            best_c5 = c_try

    # Phase 6: Lower-resolution exploration then upscale
    rem = budget_s - (time.time() - t0)
    for n_low in [128, 192]:
        rem = budget_s - (time.time() - t0)
        if rem < 50:
            break
        target_low = n_low / 2.0
        h_low = project_h(resample_fn(best_h, n_low), target_low)
        h_low, _ = slp_optimize(h_low, n_low, target_low, t0, budget_s,
                                 n_iter=80, active_frac=0.25, trust_init=0.3)
        h_up = project_h(resample_fn(h_low, n_cur), target)
        h_up, c_up = slp_optimize(h_up, n_cur, target, t0, budget_s,
                                   n_iter=40, active_frac=0.1, trust_init=0.08)
        if c_up < best_c5:
            best_h = h_up
            best_c5 = c_up

    # Phase 7: Final SLP refinement with very tight active set
    rem = budget_s - (time.time() - t0)
    if rem > 20:
        best_h, best_c5 = slp_optimize(best_h, n_cur, target, t0, budget_s,
                                       n_iter=100, active_frac=0.05, trust_init=0.05)

    # Phase 8: Coordinate polish
    rem = budget_s - (time.time() - t0)
    if rem > 15:
        best_h, best_c5 = coord_polish(best_h, target, n_cur, 150, t0, budget_s)

    # Phase 9: Final SLP squeeze
    rem = budget_s - (time.time() - t0)
    if rem > 10:
        best_h, best_c5 = slp_optimize(best_h, n_cur, target, t0, budget_s,
                                       n_iter=50, active_frac=0.03, trust_init=0.03)

    return best_h, best_c5, best_n
