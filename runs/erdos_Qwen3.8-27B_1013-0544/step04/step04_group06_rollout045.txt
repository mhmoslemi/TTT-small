        g /= gn
        h_cand = project_sum(y - step * g + mom * (h_prev - y), target)
        c5_cand = compute_c5(h_cand)
        if c5_cand < best_c5 - 1e-16:
            best = h_cand.copy()
            best_c5 = c5_cand
            stalled = 0
            step = min(step * 1.05, 0.1)
            mom = min(mom + 0.002, 0.98)
        else:
            stalled += 1
            step = max(step * 0.85, 0.002)
            if stalled % 500 == 499:
                pert = np.random.randn(n) * (0.005 + 0.003 * (stalled // 500))
                h_cand = project_sum(best + pert, target)
                c5_cand = compute_c5(h_cand)
                if c5_cand < best_c5:
                    best = h_cand.copy()
                    best_c5 = c5_cand
                    y = best.copy()
                    h_prev = best.copy()
                    step = 0.03
                    stalled = 0
        h_prev = y
        y = h_cand
    return best, best_c5

def micro_polish(h, target, t0, budget, max_iter=20000):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    stalled = 0
    for it in range(max_iter):
        if budget - (time.time() - t0) < 1.5:
            break
        corr = np.correlate(best, 1.0 - best, mode="full")
        top_idx = np.argsort(corr)[::-1][:3]
        m = corr[top_idx[0]]
        # Weight by proximity to max
        w = np.ones(len(top_idx))
        for i in range(1, len(top_idx)):
            w[i] = 0.5 + 0.5 * (corr[top_idx[i]] / m) if m > 0 else 1.0
        g = np.zeros(n)
        for i, idx in enumerate(top_idx):
            g += w[i] * corr_gradient(best, idx, n)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g /= gn
        improved = False
        for step in [0.004, 0.002, 0.001, 0.0005, 0.0002, 0.00005]:
            ht = project_sum(best - step * g, target)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 5e-17:
                best = ht
                best_c5 = c5t
                improved = True
                stalled = 0
                break
        if not improved:
            stalled += 1
            if stalled > 2000:
                break
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.asarray(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(200, 0.5)
    best_c5 = compute_c5(best_h)
    n0 = len(best_h)
    target0 = n0 / 2.0

    # Phase 1: High-beta SLSQP at current resolution
    for beta in [500, 3000, 15000, 80000]:
        if tl() < 90:
            break
        h_new = slsqp_solve(best_h, beta, maxiter=2500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
        best_h, best_c5 = nesterov_polish(best_h, target0, 4, 800, t0, budget_s)

    # Phase 2: Try nearby resolutions
    resolutions = [200, 240, 280, 320, 360, 400, 440]
    for n_try in resolutions:
        if tl() < 200:
            break
        if n_try == n0:
            continue
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_interp = project_sum(np.interp(x_new, x_old, best_h), n_try / 2.0)
        for beta in [500, 5000, 40000]:
            if tl() < 70:
                break
            h_new = slsqp_solve(h_interp, beta, maxiter=1800)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
            h_interp = h_new
        best_h, best_c5 = nesterov_polish(best_h, len(best_h) / 2.0, 5, 600, t0, budget_s)

    # Phase 3: Diverse restarts at best resolution
    n_best = len(best_h)
    tgt = n_best / 2.0
    restarts = [
        best_h.copy(),
        project_sum(best_h + np.random.randn(n_best) * 0.03, tgt),
        project_sum(np.where(np.random.rand(n_best) < 0.15, 1.0 - best_h, best_h), tgt),
        project_sum(best_h + 0.02 * np.sin(np.linspace(0, 4 * np.pi, n_best)), tgt),
    ]
    for h_try in restarts:
        if tl() < 100:
            break
        for beta in [2000, 20000]:
            if tl() < 50:
                break
            h_new = slsqp_solve(h_try, beta, maxiter=1500)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
            h_try = h_new
        best_h, best_c5 = nesterov_polish(best_h, tgt, 4, 1000, t0, budget_s, mom=0.95)

    # Phase 4: Final intensive polish
    if tl() > 30:
        h_new = slsqp_solve(best_h, 100000, maxiter=5000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    best_h, best_c5 = nesterov_polish(best_h, tgt, 3, 5000, t0, budget_s, mom=0.96)
    best_h, best_c5 = micro_polish(best_h, tgt, t0, budget_s)

    # Final SLSQP squeeze if time allows
    if tl() > 20:
        h_new = slsqp_solve(best_h, 300000, maxiter=3000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
        best_h, best_c5 = micro_polish(best_h, tgt, t0, budget_s, max_iter=5000)

    return best_h, best_c5, len(best_h)
