def find_optimal_coord(h, i, active_lags, dx, n):
    # For each active lag L, compute a_L and b_L such that
    # contribution of h[i] to corr_L is a_L * h[i] + b_L
    # (the rest of corr_L is constant w.r.t. h[i])
    
    lines_a = []
    lines_b = []
    for L in active_lags:
        a = 0.0
        b = 0.0
        # h[i] as first factor: h[i] * (1-h[i+L])
        # valid if 0 <= i < n and 0 <= i+L < n
        if 0 <= i + L < n:
            a += (1.0 - h[i+L])
        # h[i] as second factor: h[i-L] * (1-h[i]) = h[i-L] - h[i-L]*h[i]
        # valid if 0 <= i-L < n and 0 <= i < n (i.e., L <= i < n and i-L >= 0)
        if 0 <= i - L < n:
            a += -h[i-L]
            b += h[i-L]
        a *= dx
        b *= dx
        lines_a.append(a)
        lines_b.append(b)
    
    # Find t in [0,1] minimizing max_k (a_k * t + b_k)
    lines_a = np.array(lines_a)
    lines_b = np.array(lines_b)
    
    # Candidates: 0, 1, and all crossing points in (0,1)
    candidates = [0.0, 1.0]
    for p in range(len(lines_a)):
        for q in range(p+1, len(lines_a)):
            da = lines_a[p] - lines_a[q]
            if abs(da) > 1e-15:
                t_cross = (lines_b[q] - lines_b[p]) / da

        vals = lines_a * t + lines_b
        v = vals.max()
        if v < best_val:

            best_t = t
    
    return best_t

            alpha *= 0.7
            if alpha < 1e-8:
                break
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.asarray(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(200, 0.5)
    best_c5 = compute_c5(best_h)
    n = len(best_h)
    target = n / 2.0

    # Phase 1: SLSQP warmup
    t_phase1 = t0 + min(90, 0.12 * budget_s)
    for beta in [300, 2000, 8000]:
        if time.time() > t_phase1:
            break
        h_new = slsqp_optimize(best_h, n, beta, maxiter=1500)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 2: Intensive coordinate descent (main phase)
    t_phase2 = t0 + 0.65 * budget_s
    best_h, best_c5 = coord_descent_phase(best_h, target, n, t_phase2, K=12, inner_passes=2)

    # Phase 3: Subgradient with momentum
    t_phase3 = t0 + 0.82 * budget_s
    best_h, best_c5 = subgrad_polish(best_h, target, n, t_phase3, K=6)

    # Phase 4: Final coordinate descent with larger K
    t_phase4 = t0 + 0.93 * budget_s
    best_h, best_c5 = coord_descent_phase(best_h, target, n, t_phase4, K=18, inner_passes=1)

    # Phase 5: Final micro subgradient polish
    t_end = t0 + budget_s - 5
    best_h, best_c5 = subgrad_polish(best_h, target, n, t_end, K=4)

    # Verify and return
    best_h = project_sum(best_h, target)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
