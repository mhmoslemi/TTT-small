def binary_sa(h, n, deadline, max_iter=50000):
    """Simulated annealing on binary sequences."""
    h = h.copy()
    target = n // 2
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    
    c = np.correlate(h, 1-h, mode='full')
    best_c5 = float(c.max() * 2.0 / n)
    best_h = h.copy()
    
    T = best_c5 * 0.05  # initial temperature
    T_min = 1e-10
    cool_rate = 0.999
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        peak_idx = int(np.argmax(c))
        k = peak_idx - (n - 1)
        
        # Random swap
        ai = np.random.randint(len(ones))
        bi = np.random.randint(len(zeros))
        a = ones[ai]
        b = zeros[bi]
        
        # Compute delta at all lags (vectorized)
        # delta_c[k'] = h[a+k'] - h[b+k'] + h[a-k'] - h[b-k'] + corrections
        # For all k' from -(n-1) to n-1:
        ks = np.arange(-(n-1), n)  # all possible lags
        a_plus = a + ks
        a_minus = a - ks
        b_plus = b + ks
        b_minus = b - ks
        
        delta = np.zeros(2*n-1)
        mask = (a_plus >= 0) & (a_plus < n)
        delta[mask] += h[a_plus[mask]]
        mask = (a_minus >= 0) & (a_minus < n)
        delta[mask] += h[a_minus[mask]]
        mask = (b_plus >= 0) & (b_plus < n)
        delta[mask] -= h[b_plus[mask]]
        mask = (b_minus >= 0) & (b_minus < n)
        delta[mask] -= h[b_minus[mask]]
        
        # Corrections
        if k == 0:  # this is for the delta formula, not the peak
            pass
        # delta correction: -2*delta(k',0) + delta(k',a-b) + delta(k',b-a)
        # k' = 0: index in delta array = n-1
        delta[n-1] -= 2.0
        idx_ab = (a-b) + (n-1)
        if 0 <= idx_ab < 2*n-1:
            delta[idx_ab] += 1.0
        idx_ba = (b-a) + (n-1)
        if 0 <= idx_ba < 2*n-1:
            delta[idx_ba] += 1.0
        
        new_c = c + delta
        new_max = float(np.max(new_c) * 2.0 / n)
        old_max = float(c.max() * 2.0 / n)
        
        # Accept/reject
        if new_max < old_max:
            accept = True
        else:
            d = new_max - old_max
            if T > T_min:
                accept = np.random.random() < np.exp(-d / T)
            else:
                accept = False
        
        if accept:

            ones[ai] = b
            zeros[bi] = a
            c = new_c
            if new_max < best_c5:

                best_h = h.copy()
        
        T *= cool_rate
    
    return best_h, best_c5

def slsqp_optimize(h, n, alpha, deadline, maxiter=500):
    """SLSQP with log-sum-exp smoothing."""
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1-x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        logsumexp = sm + np.log(np.exp(sh - sm).sum())
        return m + logsumexp / alpha
    
    def obj_grad(x):
        g = np.correlate(x, 1-x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
