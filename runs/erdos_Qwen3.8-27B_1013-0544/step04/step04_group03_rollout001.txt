from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_lags(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def resample_fn(h, n_new):
    if len(h) == n_new:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, len(h), endpoint=False)
    new_x = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(new_x, old_x, h)

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def block_obj(x_block, h_full, block_start, block_size, beta, n, target):
    # Reconstruct full h with the block replaced
    h = h_full.copy()
    h[block_start:block_start+block_size] = x_block
    c = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(c, beta) * (2.0 / n)

def block_sum_constraint(x_block, h_full, block_start, block_size, target):
    h = h_full.copy()
    h[block_start:block_start+block_size] = x_block
    return h.sum() - target

def block_sum_jac(x_block, h_full, block_start, block_size, target):
    return np.ones(x_block.shape)

def do_block_slsqp(h, n, target, beta, block_size, maxiter=3000):
    for start in range(0, n - block_size + 1, block_size // 2):
        x0 = h[start:start+block_size].copy()
        bounds = Bounds(np.zeros(block_size), np.ones(block_size))
        cons = [{'type': 'eq', 'fun': block_sum_constraint, 'jac': block_sum_jac,
                 'args': (h, start, block_size, target)}]
        try:
            r = minimize(block_obj, x0, args=(h, start, block_size, beta, n, target),
                         method='SLSQP', bounds=bounds, constraints=cons,
                         options={'maxiter': maxiter, 'ftol': 1e-14})
            h[start:start+block_size] = r.x
            h = project_h(h, target)
        except Exception:
            pass
    return h

def subgrad_desc(h, target, n, n_iter, step_init, t0, budget, top_k=10):
    cur = h.copy()
    mom = np.zeros(n)
    best_local = compute_c5(cur)
    stall = 0
    step = step_init
    for it in range(n_iter):
        if budget - (time.time() - t0) < 3:
            break
        corr = np.correlate(cur, 1.0 - cur, mode="full")
        top_idx = np.argsort(corr)[::-1][:top_k]
        grad = np.zeros(n)
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            if lag >= 0:
                if lag < n:
                    grad[:n - lag] += (1.0 - cur[lag:])
                    grad[lag:] -= cur[:n - lag]
            else:
                la = -lag
                grad[la:] += (1.0 - cur[:n - la])
                grad[:n - la] -= cur[la:]
        grad /= top_k
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad /= gn
        mom = 0.9 * mom - step * grad
        trial = project_h(cur + mom, target)
        c_try = compute_c5(trial)
        if c_try < best_local - 1e-15:
            cur = trial
            best_local = c_try
            stall = 0
            step = min(step * 1.03, 0.08)
        else:
            stall += 1
            step *= 0.75
            mom *= 0.5
            if stall > 300:
                break
    return cur, best_local

def do_slsqp(h0, n, target, beta, maxiter=8000, ftol=1e-16):
    bounds = Bounds(np.zeros(n), np.ones(n))
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode="full")
        return smooth_max(c, beta) * (2.0 / n)
    def sumc(x):
        return np.sum(x) - target
    def sumj(x):
        return np.ones(x.shape)
    cons = [{'type': 'eq', 'fun': sumc, 'jac': sumj}]
    try:
        r = minimize(obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(r.x, target)
    except Exception:
        return h0

def coord_descent(h, target, n, max_sweeps, t0, budget):
    cur = h.copy()
    best_c5 = compute_c5(cur)
    for sw in range(max_sweeps):
        if budget - (time.time() - t0) < 2:
            break
        improved = False
        for j in range(n):
            if budget - (time.time() - t0) < 1:
                break
            for d in [0.05, -0.05, 0.02, -0.02, 0.01, -0.01]:
                trial = cur.copy()
                trial[j] = np.clip(trial[j] + d, 0, 1)
                trial = project_h(trial, target)
                c_try = compute_c5(trial)
                if c_try < best_c5 - 1e-14:
                    cur = trial
                    best_c5 = c_try
                    improved = True
                    break
        if not improved and sw > 2:
            break
    return cur, best_c5

                grad[:n - la] -= cur[la:]
        grad /= top_k
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad /= gn
        mom = 0.85 * mom - step * grad
        trial = project_h(cur + mom, target)
        c_try = compute_c5(trial)
        if c_try < best_local - 1e-15:
            cur = trial
            best_local = c_try
            stall = 0
            step = min(step * 1.04, 0.07)
        else:
            stall += 1
            step *= 0.72
            mom *= 0.4
            trial = project_h(cur + mom, target)
            c_try = compute_c5(trial)
            if c_try < best_local - 1e-15:
                cur = trial
                best_local = c_try
                stall = 0
            else:
                mom = np.zeros(n)
        if stall > 400:
            break
    return cur, best_local

def coord_descent(h, target, n, max_sweeps, t0, budget):
    cur = h.copy()
    best_c5 = compute_c5(cur)
    for sw in range(max_sweeps):
        if budget - (time.time() - t0) < 2:
            break
        improved = False
        for j in range(n):
            if budget - (time.time() - t0) < 1:
                break
            for d in [0.05, -0.05, 0.02, -0.02, 0.008, -0.008]:
                trial = cur.copy()
                trial[j] = min(1.0, max(0.0, trial[j] + d))
                trial = project_h(trial, target)
                c_try = compute_c5(trial)
                if c_try < best_c5 - 1e-14:
                    cur = trial
                    best_c5 = c_try
                    improved = True
                    break
        if not improved and sw > 2:
            break
    return cur, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        n = 384
        best_h = 0.5 * np.ones(n) + 0.25 * np.sin(np.linspace(0, 2*np.pi, n, endpoint=False))
        best_h = project_h(best_h, n / 2.0)

    best_n = len(best_h)
    best_c5 = compute_c5(best_h)
    target = best_n / 2.0
    n_cur = best_n

    # Phase 1: Multi-resolution - optimize at lower res then upscale
    remaining = budget_s - (time.time() - t0)
    if remaining > 60:
        for n_low in [128, 192, 256]:
            remaining = budget_s - (time.time() - t0)
            if remaining < 20:
                break
            h_low = project_h(resample_fn(best_h, n_low), n_low / 2.0)
            for beta in [300, 1500]:
                remaining = budget_s - (time.time() - t0)
                if remaining < 8:
                    break
                h_low = do_slsqp(h_low, n_low, n_low / 2.0, beta, maxiter=5000)
            h_up = project_h(resample_fn(h_low, n_cur), target)
            # Refine at original resolution
            h_up = do_slsqp(h_up, n_cur, target, 2000, maxiter=4000)
            c_try = compute_c5(h_up)
            if c_try < best_c5:
                best_h = h_up
                best_c5 = c_try

    # Phase 2: Block SLSQP (key differentiator from previous)
    remaining = budget_s - (time.time() - t0)
    for bsize in [16, 32, 64, 128]:
        if remaining < 20:
            break
        for beta in [400, 1500, 5000]:
            remaining = budget_s - (time.time() - t0)
            if remaining < 10:
                break
            h_try = do_block_slsqp(best_h, n_cur, target, beta, bsize, maxiter=1200)
            c_try = compute_c5(h_try)
            if c_try < best_c5:
                best_h = h_try
                best_c5 = c_try

    # Phase 3: Global SLSQP with moderate beta
    for beta in [1000, 4000, 15000]:
        remaining = budget_s - (time.time() - t0)
        if remaining < 12:
            break
        best_h = do_slsqp(best_h, n_cur, target, beta, maxiter=6000)
        best_c5 = compute_c5(best_h)

    # Phase 4: Subgradient descent
    remaining = budget_s - (time.time() - t0)
    if remaining > 20:
        best_h, best_c5 = subgrad_desc(best_h, target, n_cur, 15000, 0.012, t0, budget_s, top_k=14)

    # Phase 5: Block SLSQP again after subgradient
    remaining = budget_s - (time.time() - t0)
    for bsize in [32, 96]:
        if remaining < 15:
            break
        for beta in [4000, 12000]:
            remaining = budget_s - (time.time() - t0)
            if remaining < 8:
                break
            h_try = do_block_slsqp(best_h, n_cur, target, beta, bsize, maxiter=1000)
            c_try = compute_c5(h_try)
            if c_try < best_c5:
                best_h = h_try
                best_c5 = c_try

    # Phase 6: Coordinate descent
    remaining = budget_s - (time.time() - t0)
    if remaining > 10:
        best_h, best_c5 = coord_descent(best_h, target, n_cur, 150, t0, budget_s)

    # Phase 7: Perturbation restarts with block refinement
    remaining = budget_s - (time.time() - t0)
    rng = np.random.default_rng(seed + 314)
    for att in range(10):
        if remaining < 20:
            break
        scale = 0.015 + 0.02 * (att % 3)
        hp = best_h + rng.standard_normal(n_cur) * scale
        hp = project_h(hp, target)
        hp = do_block_slsqp(hp, n_cur, target, 2000, 48, maxiter=800)
        hp = do_slsqp(hp, n_cur, target, 5000, maxiter=3000)
        hp, _ = subgrad_desc(hp, target, n_cur, 2500, 0.008, t0, budget_s, top_k=10)
        c_try = compute_c5(hp)
        if c_try < best_c5:
            best_h = hp.copy()
            best_c5 = c_try

    # Phase 8: Final high-beta SLSQP and coordinate polish
    remaining = budget_s - (time.time() - t0)
    if remaining > 10:
        best_h = do_slsqp(best_h, n_cur, target, 40000, maxiter=8000, ftol=1e-16)
        best_c5 = compute_c5(best_h)
    remaining = budget_s - (time.time() - t0)
    if remaining > 8:
        best_h, best_c5 = coord_descent(best_h, target, n_cur, 250, t0, budget_s)

    best_h = project_h(best_h, target)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
