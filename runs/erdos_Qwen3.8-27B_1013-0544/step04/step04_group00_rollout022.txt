```python
def swap_delta(h, f, i, j, n):
    """Delta in full correlation when h[i]:0->1, h[j]:1->0 (f=1-h)"""
    delta = np.zeros(2*n-1)
    
    # From h[i] += 1: add f[::-1] at offset i
    delta[i:i+n] += f[::-1]
    
    # From h[j] -= 1: subtract f[::-1] at offset j
    delta[j:j+n] -= f[::-1]
    
    # From f[i] -= 1: subtract h at offset n-1-i
    off = n-1-i
    delta[off:off+n] -= h
    
    # From f[j] += 1: add h at offset n-1-j
    off = n-1-j
    delta[off:off+n] += h
    
    return delta
```

```python
    """E[i] = delta from turning on i, D[j] = delta from turning off j"""
    E = np.zeros((n, 2 * n - 1))
    D = np.zeros((n, 2 * n - 1))

        E[i, n-1-i:n-1-i+n] -= h
    for j in range(n):
        D[j, j:j+n] -= fr
        D[j, n-1-j:n-1-j+n] += h
    return E, D

    f = 1.0 - h
    c = np.correlate(h, f, mode='full')
    best_c = float(np.max(c))
    changed = True
    while changed and time.time() < deadline:
        changed = False
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        E, D = precompute_swap(h, f, n)
        best_val = best_c
        best_pair = None
        D_ones = D[ones]  # shape (n/2, 2n-1)
        for i in zeros:
            if time.time() > deadline:
                break
            base = c + E[i]
            vals = np.max(base[None, :] + D_ones, axis=1)
            k = np.argmin(vals)
            if vals[k] < best_val - 1e-10:

                best_pair = (i, ones[k])

        h[i], h[j] = 1.0, 0.0
        c = c + E[i] + D[j]
        f = 1.0 - h
        best_c = best_val
        changed = True

        delta[n-1-i:n-1-i+n] -= h

            if new_c_max < best_c:
                best_c = new_c_max
                best_h = h.copy()
    
    return best_h, best_c * (2.0 / n)
```
