```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_lags(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def smooth_max_grad(x, beta):
    """Return (value, gradient) of smooth max"""
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    exp_vals = np.exp(x_shift)
    weights = exp_vals / np.sum(exp_vals)
    value = m + np.log(np.sum(exp_vals) / len(x)) / beta
    return value, weights / beta  # gradient w.r.t. x

def lag_gradient(corr, h, top_k_indices, n):
    """Compute approximate gradient of sum of top-k lags w.r.t. h"""
    grad = np.zeros(n)
    for idx in top_k_indices:
        lag = int(idx) - (n - 1)
        if lag >= 0:
            # c_lag = sum_{j=0}^{n-1-lag} h[j]*(1-h[j+lag])
            # dc/dh[i] = (1-h[i+lag]) for i in [0, n-1-lag]
            #           - h[i-lag] for i in [lag, n-1]
            grad[:n-lag] += (1.0 - h[lag:])
            grad[lag:] -= h[:n-lag]
        else:
            la = -lag
            # c_{-la} = sum_{j=la}^{n-1} h[j]*(1-h[j-la])
            # dc/dh[i] = (1-h[i-la]) for i in [la, n-1]
            #           - h[i+la] for i in [0, n-1-la]
            grad[la:] += (1.0 - h[:n-la])
            grad[:n-la] -= h[la:]
    return grad

def projected_grad_desc(h, target, n, n_iter, t0, budget, top_k=20, beta=500):
    """Projected gradient descent with Nesterov momentum"""
    cur = h.copy()
    vel = np.zeros(n)
    best_c5 = compute_c5(cur)
    best_h = cur.copy()
    step = 0.05
    momentum = 0.9
    stall = 0
    
    for it in range(n_iter):
        remaining = budget - (time.time() - t0)
        if remaining < 2:
            break
        
        # Compute lags and find top-k
        corr = np.correlate(cur, 1.0 - cur, mode="full")
        top_idx = np.argsort(corr)[::-1][:top_k]
        
        # Compute gradient
        grad = lag_gradient(corr, cur, top_idx, n)
        grad /= max(len(top_idx), 1)
        
        # Nesterov momentum
        vel = momentum * vel - step * grad
        trial = project_h(cur + vel, target)
        
        # Evaluate
        c_try = compute_c5(trial)
        if c_try < best_c5 - 1e-15:
            cur = trial
            best_c5 = c_try

            stall = 0
            step = min(step * 1.02, 0.15)
        else:
            stall += 1
            step *= 0.8
            vel *= 0.5

    return best_h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_c5_and_lags(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr)), corr

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def projected_grad(h, target, n, n_iter, t0, budget, top_k=20):

    vel = np.zeros(n)
    best_c5 = compute_c5(cur)
    best_h = cur.copy()
    step = 0.03
    momentum = 0.85
    stall = 0

            break
```

```python
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def resample_fn(h, n_new):
    if len(h) == n_new:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, len(h), endpoint=False)
    new_x = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(new_x, old_x, h)

def binary_snap(h, n):
    n_ones = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:n_ones]] = 1.0
    return h_bin

def binary_anneal(h_bin, t0, budget, n_iter, seed=42):
    n = len(h_bin)
    h = h_bin.copy()
    cur_lags = np.correlate(h, 1.0 - h, mode="full")
    cur_max = float(np.max(cur_lags))
    best_h = h.copy()
    best_max = cur_max
    rng = np.random.default_rng(seed)
    ones_set = list(np.where(h == 1)[0])
    zeros_set = list(np.where(h == 0)[0])
    for it in range(n_iter):
        remaining = budget - (time.time() - t0)
        if remaining < 3:
            break
        a = ones_set[rng.integers(len(ones_set))]
        b = zeros_set[rng.integers(len(zeros_set))]
        h[a] = 0.0
        h[b] = 1.0
        new_lags = np.correlate(h, 1.0 - h, mode="full")
        new_max = float(np.max(new_lags))
        if new_max < cur_max:
            cur_lags = new_lags
            cur_max = new_max
            ones_set.remove(a)
            zeros_set.remove(b)
            zeros_set.append(a)
            ones_set.append(b)
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
        else:
            frac = it / max(n_iter, 1)
            temp = (cur_max + 5.0) * 0.1 * (0.005) ** frac
            if rng.random() < np.exp(-(new_max - cur_max) / max(temp, 1e-15)):
                cur_lags = new_lags
                cur_max = new_max
                ones_set.remove(a)
                zeros_set.remove(b)
                zeros_set.append(a)
                ones_set.append(b)
            else:
                h[a] = 1.0
                h[b] = 0.0
    return best_h, best_max

def binary_local_search(h, t0, budget, n_passes=5):
    n = len(h)
    h = h.copy()
    cur_lags = np.correlate(h, 1.0 - h, mode="full")
    cur_max = float(np.max(cur_lags))
    best_h = h.copy()
    best_max = cur_max
    k_vals = np.arange(-(n - 1), n)
    for pas in range(n_passes):
        if budget - (time.time() - t0) < 5:
            break
        improved = False
        ones_list = np.where(h == 1)[0]
        for a in ones_list:
            if budget - (time.time() - t0) < 3:
                break
            if h[a] != 1:
                continue
            a_pk = a + k_vals
            a_mk = a - k_vals
            I1 = (a_pk >= 0) & (a_pk < n)
            I2 = (a_mk >= 0) & (a_mk < n)
            A = np.where(I1, h[np.clip(a_pk, 0, n-1)] - 1, 0.0) + np.where(I2, h[np.clip(a_mk, 0, n-1)], 0.0)
            zeros_now = np.where(h == 0)[0]
            if len(zeros_now) == 0:
                break
            b_idx = zeros_now[:, None]
            pk = b_idx + k_vals[None, :]
            mk = b_idx - k_vals[None, :]
            vpk = (pk >= 0) & (pk < n)
            vmk = (mk >= 0) & (mk < n)
            B_all = np.where(vpk, 1.0 - h[np.clip(pk, 0, n-1)], 0.0) - np.where(vmk, h[np.clip(mk, 0, n-1)], 0.0)
            new_lags_all = cur_lags[None, :] + A[None, :] + B_all
            new_maxs = np.max(new_lags_all, axis=1)
            best_bi = np.argmin(new_maxs)
            if new_maxs[best_bi] < cur_max - 1e-12:
                b = zeros_now[best_bi]
                h[a] = 0.0
                h[b] = 1.0
                cur_lags = np.correlate(h, 1.0 - h, mode="full")
                cur_max = float(np.max(cur_lags))
                improved = True
                if cur_max < best_max:
                    best_max = cur_max
                    best_h = h.copy()
        if not improved:
            break
    return best_h, best_max

def projected_grad(h, target, n, n_iter, t0, budget, top_k=20):
    cur = h.copy()
    vel = np.zeros(n)
    best_c5 = compute_c5(cur)
    best_h = cur.copy()
    step = 0.03
    momentum = 0.85
    stall = 0
    for it in range(n_iter):
        remaining = budget - (time.time() - t0)
        if remaining < 2:
            break
        corr = np.correlate(cur, 1.0 - cur, mode="full")
        top_idx = np.argsort(corr)[::-1][:top_k]
        grad = np.zeros(n)
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            if lag >= 0:
                grad[:n - lag] += (1.0 - cur[lag:])
                grad[lag:] -= cur[:n - lag]
            else:
                la = -lag
                grad[la:] += (1.0 - cur[:n - la])
                grad[:n - la] -= cur[la:]
        grad /= max(len(top_idx), 1)
        vel = momentum * vel - step * grad
        trial = project_h(cur + vel, target)
        c_try = compute_c5(trial)
        if c_try < best_c5 - 1e-15:
            cur = trial
            best_c5 = c_try
            best_h = cur.copy()
            stall = 0
            step = min(step * 1.01, 0.1)
        else:
            stall += 1
            step *= 0.85
            vel *= 0.5
            if stall > 300:
                break
    return best_h, best_c5

def slsqp_opt(h0, n, target, beta, maxiter=2000):
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode="full")
        return smooth_max(c, beta) * (2.0 / n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(obj, h0, method='SLSQP', bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_h(r.x, target)
    except Exception:
        return h0

def coord_descent(h, target, n, max_sweeps, t0, budget):
    cur = h.copy()
    best_c5 = compute_c5(cur)
    deltas = [0.02, -0.02, 0.01, -0.01, 0.005, -0.005]
    for sw in range(max_sweeps):
        if budget - (time.time() - t0) < 2:
            break
        improved = False
        for j in range(n):
            if budget - (time.time() - t0) < 1:
                break
            for d in deltas:
                trial = cur.copy()
                trial[j] += d
                trial = project_h(trial, target)
                c_try = compute_c5(trial)
                if c_try < best_c5 - 1e-14:
                    cur = trial
                    best_c5 = c_try
                    improved = True
                    break
        if not improved and sw > 3:
            break
    return cur, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        n_init = 384
        h_parent = np.zeros(n_init)
        h_parent[:n_init // 2] = 1.0
        h_parent = project_h(h_parent, n_init / 2.0)
    n_cur = len(h_parent)
    target = n_cur / 2.0
    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)
    best_n = n_cur
    # Phase 1: Binary annealing
    h_bin = binary_snap(best_h, n_cur)
    h_bin, bin_max = binary_anneal(h_bin, t0, budget_s, n_iter=400000, seed=seed)
    if bin_max < best_c5:
        best_h = h_bin.copy()
        best_c5 = bin_max
    # Phase 2: Binary local search (steepest descent)
    remaining = budget_s - (time.time() - t0)
    if remaining > 15:
        best_h, best_c5 = binary_local_search(best_h, t0, budget_s, n_passes=5)
    # Phase 3: Second annealing from perturbed binary
    remaining = budget_s - (time.time() - t0)
    if remaining > 60:
        rng = np.random.default_rng(seed + 200)
        h_pert = best_h + rng.standard_normal(n_cur) * 0.08
        h_pert = project_h(h_pert, target)
        h_bin2 = binary_snap(h_pert, n_cur)
        h_bin2, bin_max2 = binary_anneal(h_bin2, t0, budget_s, n_iter=200000, seed=seed + 2)
        if bin_max2 < best_c5:
            best_h = h_bin2.copy()
            best_c5 = bin_max2
        remaining = budget_s - (time.time() - t0)
        if remaining > 10:
            best_h, best_c5 = binary_local_search(best_h, t0, budget_s, n_passes=3)
    # Phase 4: Gradient refinement
    remaining = budget_s - (time.time() - t0)
    if remaining > 25:
        best_h, best_c5 = projected_grad(best_h, target, n_cur, 15000, t0, budget_s, top_k=20)
    # Phase 5: SLSQP
    remaining = budget_s - (time.time() - t0)
    for beta in [300, 1000]:
        remaining = budget_s - (time.time() - t0)
        if remaining < 20:
            break
        best_h = slsqp_opt(best_h, n_cur, target, beta, maxiter=2000)
        best_c5 = compute_c5(best_h)
    # Phase 6: More gradient + binary local search
    remaining = budget_s - (time.time() - t0)
    if remaining > 20:
        best_h, best_c5 = projected_grad(best_h, target, n_cur, 10000, t0, budget_s, top_k=12)
    remaining = budget_s - (time.time() - t0)
    if remaining > 10:
        h_b = binary_snap(best_h, n_cur)
        h_b, bm = binary_local_search(h_b, t0, budget_s, n_passes=4)
        if bm < best_c5:
            best_h = h_b.copy()
            best_c5 = bm
    # Phase 7: Try alternative resolutions
    remaining = budget_s - (time.time() - t0)
    for n_alt in [320, 420]:
        remaining = budget_s - (time.time() - t0)
        if remaining < 70:
            break
        t_alt = n_alt / 2.0
        h_alt = project_h(resample_fn(best_h, n_alt), t_alt)
        h_bin_alt = binary_snap(h_alt, n_alt)
        h_bin_alt, bm_alt = binary_anneal(h_bin_alt, t0, budget_s, n_iter=100000, seed=seed + n_alt)
        remaining = budget_s - (time.time() - t0)
        if remaining > 10:
            h_bin_alt, bm_alt = binary_local_search(h_bin_alt, t0, budget_s, n_passes=4)
        if bm_alt < best_c5:
            best_h = h_bin_alt.copy()
            best_c5 = bm_alt
            best_n = n_alt
            n_cur = n_alt
            target = t_alt
        remaining = budget_s - (time.time() - t0)
        if remaining > 20:
            h_alt = project_h(resample_fn(best_h, n_cur), target)
            h_alt, c_alt = projected_grad(h_alt, target, n_cur, 5000, t0, budget_s, top_k=15)
            if c_alt < best_c5:
                best_h = h_alt.copy()
                best_c5 = c_alt
    # Phase 8: Final polish
    remaining = budget_s - (time.time() - t0)
    if remaining > 15:
        best_h = slsqp_opt(best_h, best_n, best_n / 2.0, 5000, maxiter=3000)
        best_c5 = compute_c5(best_h)
    remaining = budget_s - (time.time() - t0)
    if remaining > 10:
        best_h, best_c5 = coord_descent(best_h, best_n / 2.0, best_n, 100, t0, budget_s)
    remaining = budget_s - (time.time() - t0)
    if remaining > 10:
        best_h, best_c5 = projected_grad(best_h, best_n / 2.0, best_n, 5000, t0, budget_s, top_k=8)
    return best_h, best_c5, best_n
```
