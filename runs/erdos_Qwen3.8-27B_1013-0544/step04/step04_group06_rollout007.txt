import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def corr_grad_single(h, idx, n):
    lag = int(idx) - (n - 1)
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        g[lag:] += 1.0 - h[:-lag]
        g[:n - lag] -= h[lag:]
    else:
        m2 = -lag
        g[:n - m2] += 1.0 - h[m2:]
        g[m2:] -= h[:n - m2]
    return g

def smooth_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    bc = np.clip(beta * (corr - m), -500, 500)
    e = np.exp(bc)
    se = e.sum()
    return (m + np.log(se / len(corr)) / beta) * dx

def smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    bc = np.clip(beta * (corr - m), -500, 500)
    e = np.exp(bc)
    se = e.sum()
    w = e / se
    g = np.zeros(n)
    for j in range(len(corr)):
        if w[j] < 1e-11:
            continue
        g += w[j] * corr_grad_single(h, j, n)
    return g * dx

def slsqp_step(h, n, beta, maxiter=2000):
    target = n / 2.0
    bounds = [(0, 1)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    res = minimize(lambda x: smooth_obj(np.clip(x, 0, 1), beta), h,
                   jac=lambda x: smooth_grad(np.clip(x, 0, 1), beta),
                   method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-16})
    return project_sum(res.x, target)

def subgrad_balanced(h, target, tol_frac, max_iter, t0, budget):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    base_lr = 0.04
    no_improve = 0
    for it in range(max_iter):
        if budget - (time.time() - t0) < 3:
            break
        corr = np.correlate(best, 1.0 - best, mode="full")
        m = np.max(corr)
        threshold = m - tol_frac * (2.0 / n)
        active = np.where(corr >= threshold)[0]
        g = np.zeros(n)
        for idx in active:
            g += corr_grad_single(best, idx, n)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        improved = False
        for lr in [base_lr, base_lr * 0.5, base_lr * 0.2, base_lr * 0.07, base_lr * 0.02]:
            h_try = project_sum(best - lr * g, target)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best = h_try
                best_c5 = c5_try
                improved = True
                break
        if improved:
            no_improve = 0
            base_lr = min(base_lr * 1.1, 0.06)
        else:
            no_improve += 1
            base_lr *= 0.85
            if no_improve > 500 and it % 250 == 0:
                pert = np.random.randn(n) * 0.015
                h_try = project_sum(best + pert, target)
                if compute_c5(h_try) < best_c5:
                    best = h_try
                    best_c5 = compute_c5(best)
                    no_improve = 0
                    base_lr = 0.04
            if base_lr < 1e-6 or no_improve > 2000:
                break
    return best, best_c5
