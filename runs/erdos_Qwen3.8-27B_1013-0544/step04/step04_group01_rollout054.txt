    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    step = 0.05
    momentum = np.zeros(n)
    beta = 0.9
    
    while time.time() < deadline:
        c = np.correlate(h, 1 - h, mode='full') * dx
        cmax = c.max()
        
        # Active set: lags within 1% of max
        threshold = cmax - 0.01 * max(cmax, dx)

            lo1 = max(0, shift)
            hi1 = min(n, n + shift)

                grad[lo1:hi1] += (1.0 - h[idx - shift])

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    if shifted.max() > 500:
        mx = shifted.max()
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, alpha, maxiter=4000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def gradient_search(h, n, deadline):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    step = 0.08
    mom = np.zeros(n)
    beta = 0.85
    stall = 0
    while time.time() < deadline:
        c = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
        cmax = c.max()
        threshold = cmax - 0.02 * max(cmax, 0.01)
        active = np.where(c >= threshold)[0]
        grad = np.zeros(n)
        for m in active:
            shift = m - (n - 1)
            lo1, hi1 = max(0, shift), min(n, n + shift)
            if lo1 < hi1:
                grad[lo1:hi1] += (1.0 - h[lo1 - shift:hi1 - shift])
            lo2, hi2 = max(0, -shift), min(n, n - shift)
            if lo2 < hi2:
                grad[lo2:hi2] -= h[lo2 + shift:hi2 + shift]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        mom = beta * mom - step * grad / gnorm
        h_new = project_feasible(h + mom, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c, best_h = c_new, h_new.copy()
            h = h_new
            step = min(step * 1.02, 0.15)
            stall = 0
        else:
            step *= 0.8
            mom *= 0.5
            stall += 1
            if stall > 30 or step < 1e-9:
                break
    return best_h, best_c

def pair_search(h, n, deadline, deltas, max_pass=200):
    h = h.copy()
    best_h = h.copy()

        if time.time() > deadline:
            break

            if time.time() > deadline:
                break
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break

                    hi, hj = h[i] + d, h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_h = h.copy()
                            improved = True
                        else:
                            h[i], h[j] = hi - d, hj + d
                    hi2, hj2 = h[i] - d, h[j] + d
                    if 0.0 <= hi2 <= 1.0 and 0.0 <= hj2 <= 1.0:
