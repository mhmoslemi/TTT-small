        return np.sum(x) - target

        return np.ones(n)

            elif -n < lag < 0:
                la = -lag
                grad[la:] += w * (1.0 - x[:n-la])
                grad[:n-la] -= w * x[la:]
        return grad
    
    # Use SLSQP as fallback since L-BFGS-B doesn't handle equality well
    # Instead, use a penalty approach
    penalty = 100.0
    def penalized_obj(x):
        base = obj(x)
        viol = abs(np.sum(x) - target)
        return base + penalty * viol**2
    
    bounds = [(0.0, 1.0)] * n
    try:
        # Do multiple rounds of L-BFGS-B with increasing penalty
        for pen in [10, 100, 1000]:
            def po(x):
                return obj(x) + pen * (np.sum(x) - target)**2
            r = minimize(po, x, method='L-BFGS-B', bounds=bounds,
                         options={'maxiter': maxiter//3, 'gtol': tol})
            x = project_h(r.x, target)
        return project_h(x, target)
    except Exception:
        return x

def resample(h, n_new):
    if len(h) == n_new:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, len(h), endpoint=False)
    new_x = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_interp = np.interp(new_x, old_x, h)
    return project_h(h_interp, n_new / 2.0)

def coordinate_polish(h, target, n, max_sweeps, t0, budget, deltas=None):
    if deltas is None:
        deltas = [0.02, -0.02, 0.01, -0.01, 0.005, -0.005, 0.002, -0.002]
    cur = h.copy()
    best_c5 = compute_c5(cur)
    for sw in range(max_sweeps):
        if budget - (time.time() - t0) < 2:
            break
        improved = False
        for j in range(n):
            if budget - (time.time() - t0) < 1:
                break
            for d in deltas:
                trial = cur.copy()
                trial[j] = np.clip(trial[j] + d, 0.0, 1.0)
                trial = project_h(trial, target)
                c_try = compute_c5(trial)
                if c_try < best_c5 - 1e-13:
                    cur = trial
                    best_c5 = c_try
                    improved = True
                    break
        if not improved:
            break
    return cur, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = np.full(384, 0.5)
        h_parent[:192] = 0.7
        h_parent[192:] = 0.3
        h_parent = project_h(h_parent, 192.0)
    
    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)
    best_n = len(best_h)
    
    # Phase 1: Binary optimization at parent resolution
    n_p = best_n
    remaining = budget_s - (time.time() - t0)
    if remaining > 60:
        h_bin = (best_h > 0.5).astype(float)
        h_bin = project_h(h_bin, n_p / 2.0)
        # General annealing
        h_bin, _ = efficient_binary_search(h_bin, n_p, t0, budget_s, n_swaps=80000)
        # Directed search
        h_bin, _ = directed_binary_search(h_bin, n_p, t0, budget_s, n_iterations=30000)
        c_bin = compute_c5(h_bin)
        if c_bin < best_c5:
            best_c5 = c_bin
        # Keep binary for refinement
        h_from_bin = h_bin.copy()
    
    # Phase 2: Continuous optimization from binary start
    remaining = budget_s - (time.time() - t0)
    if remaining > 30:
        # Slight perturbation to break symmetry
        rng = np.random.default_rng(seed + 1)
        h_work = h_from_bin + 0.01 * rng.standard_normal(n_p)
        h_work = project_h(h_work, n_p / 2.0)
        for p in [20, 40, 80, 160]:
            remaining = budget_s - (time.time() - t0)
            if remaining < 15:
                break
            h_work = slsqp_optimize(h_work, n_p, n_p/2.0, p, maxiter=4000, ftol=1e-16)
            c_try = compute_c5(h_work)
            if c_try < best_c5:
                best_h = h_work.copy()
                best_c5 = c_try
    
    # Phase 3: Try different resolutions
    remaining = budget_s - (time.time() - t0)
    for n_try in [256, 320, 448, 512]:
        if remaining < 45:
            break
        target_t = n_try / 2.0
        h_t = resample(best_h, n_try)
        # Binary search at this resolution
        h_bin_t = (h_t > 0.5).astype(float)
        h_bin_t = project_h(h_bin_t, target_t)
        remaining = budget_s - (time.time() - t0)
        if remaining > 20:
            h_bin_t, _ = directed_binary_search(h_bin_t, n_try, t0, budget_s, n_iterations=20000)
        # Continuous refinement
        h_t2 = h_bin_t + 0.01 * np.random.default_rng(seed + n_try).standard_normal(n_try)
        h_t2 = project_h(h_t2, target_t)
        for p in [30, 80]:
            remaining = budget_s - (time.time() - t0)
            if remaining < 10:
                break
            h_t2 = slsqp_optimize(h_t2, n_try, target_t, p, maxiter=3000, ftol=1e-15)
            c_try = compute_c5(h_t2)
            if c_try < best_c5:
                best_h = h_t2.copy()
                best_c5 = c_try
                best_n = n_try
                n_p = n_try
    
    # Phase 4: Multi-start perturbation at best resolution
    remaining = budget_s - (time.time() - t0)
    rng = np.random.default_rng(seed + 200)
    n_cur = best_n
    target = n_cur / 2.0
    for att in range(8):
        if remaining < 20:
            break
        scale = 0.02 + 0.01 * att
        hp = best_h + rng.standard_normal(n_cur) * scale
        hp = project_h(hp, target)
        # Binary snap and search
        hp_bin = (hp > 0.5).astype(float)
        hp_bin = project_h(hp_bin, target)
        remaining = budget_s - (time.time() - t0)
        if remaining > 10:
            hp_bin, _ = directed_binary_search(hp_bin, n_cur, t0, budget_s, n_iterations=10000)
        hp2 = hp_bin + 0.008 * rng.standard_normal(n_cur)
        hp2 = project_h(hp2, target)
        for p in [40, 100]:
            remaining = budget_s - (time.time() - t0)
            if remaining < 8:
                break
            hp2 = slsqp_optimize(hp2, n_cur, target, p, maxiter=3000, ftol=1e-15)
            c_try = compute_c5(hp2)
            if c_try < best_c5:
                best_h = hp2.copy()
                best_c5 = c_try
    
    # Phase 5: Final polish with coordinate descent
    remaining = budget_s - (time.time() - t0)
    if remaining > 10:
        best_h, best_c5 = coordinate_polish(best_h, best_n/2.0, best_n, 200, t0, budget_s)
    
    # Phase 6: Ultra-fine SLSQP
    remaining = budget_s - (time.time() - t0)
    if remaining > 10:
        for p in [200, 400]:
            remaining = budget_s - (time.time() - t0)
            if remaining < 8:
                break
            best_h = slsqp_optimize(best_h, best_n, best_n/2.0, p, maxiter=5000, ftol=1e-17)
            best_c5 = compute_c5(best_h)
    
    # Final coordinate polish
    remaining = budget_s - (time.time() - t0)
    if remaining > 5:
        best_h, best_c5 = coordinate_polish(best_h, best_n/2.0, best_n, 300, t0, budget_s,
                                            deltas=[0.005, -0.005, 0.002, -0.002, 0.001, -0.001])
    
    return best_h, best_c5, best_n
