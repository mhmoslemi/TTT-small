from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(np.clip(beta * (x - m), -500, 500))) / len(x)) / beta

def sml_obj(h_in, args):
    beta, dx = args
    return smooth_max(np.correlate(h_in, 1.0 - h_in, mode="full"), beta) * dx

def sum_eq(x, t):
    return np.sum(x) - t

def sum_eq_jac(x, t):
    return np.ones(x.shape)

def do_slsqp(h0, n, target, beta, maxiter):
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (target,)}]
    try:
        r = minimize(sml_obj, h0, args=(beta, 2.0/n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_h(r.x, target)
    except:
        return h0

def get_top_lags(h, num=20):
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    thr = mc - 0.005 * (mc + 1e-10)
    idx = np.where(corr >= thr)[0]
    if len(idx) > num:
        idx = idx[np.argsort(corr[idx])[::-1][:num]]

def subgrad_top(h, n, idx):
    g = np.zeros(n)
    for i in idx:
        lag = int(i) - (n - 1)
        if lag >= 0:
            g[:n - lag] += (1.0 - h[lag:])
            g[lag:] -= h[:n - lag]
        else:
            la = -lag
            g[la:] += (1.0 - h[:n - la])
            g[:n - la] -= h[la:]
    return g / max(len(idx), 1)

        accept = c_new < c_cur or rng.random() < np.exp(-(c_new - c_cur) / temp)
        if accept:

    return best_h, best_c5

def line_search_pair(h, n, target, j, k, t0, budget):

    # Try several directions
    best = h.copy()
    best_c = compute_c5(h)

    return best, best_c

    betas = [50, 100, 200, 500, 1000, 2000, 4000, 8000]
    for beta in betas:
        if time.time() - t0 > budget - 15:
            break
        h = do_slsqp(h, n, target, beta, 2500)
    return h

def sim_anneal(h0, n, target, n_iter, t0, budget, seed=42):
    h = h0.copy()
    best_h, best_c5 = h.copy(), compute_c5(h)
    rng = np.random.RandomState(seed)
    temp = 0.001
    cool_rate = 0.9999
    for it in range(n_iter):
        if time.time() - t0 > budget - 3:
            break
        if it % 1000 == 0:
            temp *= cool_rate
        j = rng.randint(n)
        k = rng.randint(n)
        if j == k:
            continue
        d = rng.uniform(-0.2, 0.2)
        trial = h.copy()
        trial[j] = np.clip(trial[j] + d, 0.0, 1.0)
        trial[k] = np.clip(trial[k] - d, 0.0, 1.0)
        trial = project_h(trial, target)
        c_new = compute_c5(trial)
        if c_new < best_c5 - 1e-15:
            best_h, best_c5 = trial.copy(), c_new
        c_cur = compute_c5(h)
        if c_new < c_cur or (c_new - c_cur) < temp and rng.random() < max(0, 1 - (c_new - c_cur) / temp):
            h = trial
    return best_h, best_c5

def local_refine(h0, n, target, n_sweeps, t0, budget, seed=99):
    h = h0.copy()
    best_c5 = compute_c5(h)
    rng = np.random.RandomState(seed)
    for sw in range(n_sweeps):
        if time.time() - t0 > budget - 3:
            break
        improved = False
        for _ in range(n):
            if time.time() - t0 > budget - 2:
                break
            j, k = rng.choice(n, 2, replace=False)
            for d in (0.03, -0.03, 0.06, -0.06, 0.1, -0.1, 0.15, -0.15, 0.25, -0.25):
                trial = h.copy()
                trial[j] = np.clip(trial[j] + d, 0.0, 1.0)
                trial[k] = np.clip(trial[k] - d, 0.0, 1.0)
                trial = project_h(trial, target)
                c = compute_c5(trial)
                if c < best_c5 - 1e-15:
                    h, best_c5, improved = trial, c, True
                    break
        if not improved:
            break
    return h, best_c5

def randomized_restart(h_best, n, target, t0, budget, n_restarts=6, seed=777):
    rng = np.random.RandomState(seed)
    best_h, best_c5 = h_best.copy(), compute_c5(h_best)
    for r in range(n_restarts):
        if time.time() - t0 > budget - 25:
            break
        h_pert = h_best.copy()
        n_flip = max(3, n // 8)
        flips = rng.choice(n, n_flip, replace=False)
        h_pert[flips] = 1.0 - h_pert[flips]
        h_pert = project_h(h_pert, target)
        h_pert = do_slsqp(h_pert, n, target, 500, 1500)
        h_pert, c = local_refine(h_pert, n, target, 5, t0, budget, seed=100+r)
        if c < best_c5 - 1e-15:
            best_h, best_c5 = h_pert.copy(), c
    return best_h, best_c5

def resample(h, n_new):
    if len(h) == n_new:
        return h.copy()
    return np.interp(np.linspace(0, 2, n_new), np.linspace(0, 2, len(h)), h)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = np.full(320, 0.5)

    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)

    n_parent = len(h_parent)
    # Prioritize parent resolution and a few smaller ones
    candidate_ns = [n_parent]
    for r in [128, 160, 200, 224, 256, 288, 320, 384, 448]:
        if r != n_parent and 80 <= r <= 600 and r not in candidate_ns:
            candidate_ns.append(r)
    # Sort: parent first, then smaller ones (faster), then larger
    candidate_ns.sort(key=lambda r: (0 if r == n_parent else 1, abs(r - n_parent)))

    for n_pts in candidate_ns:
        rem = budget_s - (time.time() - t0)
        if rem < 80:
            break
        target = n_pts / 2.0

        # Initialize from best
        cur = project_h(resample(best_h, n_pts), target)
        cur_c5 = compute_c5(cur)

        # Also try a random binary-ish init for diversity
        rem = budget_s - (time.time() - t0)
        if rem > 60:
            h_rand = (rng.rand(n_pts) > 0.5).astype(float)
            h_rand = project_h(h_rand, target)
            h_rand = do_slsqp(h_rand, n_pts, target, 200, 2000)
            cr = compute_c5(h_rand)
            if cr < cur_c5:
                cur, cur_c5 = h_rand, cr

        # Phase 1: Continuation SLSQP
        rem = budget_s - (time.time() - t0)
        if rem > 50:
            cur = continuation_slsqp(cur, n_pts, target, t0, budget_s)
            cur_c5 = compute_c5(cur)

        # Phase 2: Local refinement
        rem = budget_s - (time.time() - t0)
        if rem > 40:
            cur, cur_c5 = local_refine(cur, n_pts, target, 25, t0, budget_s)

        # Phase 3: Simulated annealing
        rem = budget_s - (time.time() - t0)
        if rem > 50:
            cur2, c2 = sim_anneal(cur, n_pts, target, 50000, t0, budget_s)
            if c2 < cur_c5 - 1e-15:
                cur, cur_c5 = cur2, c2
            # Local refine after SA
            rem = budget_s - (time.time() - t0)
            if rem > 30:
                cur, cur_c5 = local_refine(cur, n_pts, target, 15, t0, budget_s)

        # Phase 4: Randomized restarts
        rem = budget_s - (time.time() - t0)
        if rem > 60:
            cur, cur_c5 = randomized_restart(cur, n_pts, target, t0, budget_s,
                                              n_restarts=8, seed=seed + n_pts)

        # Phase 5: Final SLSQP polish
        rem = budget_s - (time.time() - t0)
        if rem > 20:
            cur = do_slsqp(cur, n_pts, target, 5000, 4000)
            cur = do_slsqp(cur, n_pts, target, 10000, 3000)
            cur_c5 = compute_c5(cur)

        if cur_c5 < best_c5 - 1e-15:
            best_h, best_c5 = cur.copy(), cur_c5

    # Final polish at best resolution
    rem = budget_s - (time.time() - t0)
    nf = len(best_h)
    tf = nf / 2.0
    if rem > 30:
        best_h, best_c5 = local_refine(best_h, nf, tf, 30, t0, budget_s)
    if rem > 20:
        best_h = do_slsqp(best_h, nf, tf, 8000, 5000)
        best_c5 = compute_c5(best_h)
    if rem > 10:
        best_h, best_c5 = local_refine(best_h, nf, tf, 20, t0, budget_s)
    if rem > 5:
        best_h = do_slsqp(best_h, nf, tf, 10000, 3000)
        best_c5 = compute_c5(best_h)

    best_h = project_h(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
