    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project_sum(h, target):
    """Project h onto {x: 0<=x<=1, sum(x)=target} using water-filling."""
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_corr(h):
    """Compute full correlation of h with 1-h using FFT for speed."""
    n = len(h)
    # Use direct correlation for n <= 1000 (FFT overhead not worth it)
    return np.correlate(h, 1.0 - h, mode="full")

def objective_and_gradient(h, beta):

    n = len(h)
    dx = 2.0 / n
    nc = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)

    shifted = beta * (corr - m)

    n = len(h)

    best_c5 = compute_c5(best_h)

        if budget - (time.time() - t0) < 30:
            break
        m = np.zeros(n)  # First moment
        v = np.zeros(n)  # Second moment
        t = 0
        lr = 0.01
        beta1, beta2, eps = 0.8, 0.999, 1e-8
        
        for it in range(iterations_per_beta):
            if budget - (time.time() - t0) < 30:
                break
            t += 1
            val, grad = objective_and_gradient(h, beta)
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad**2
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
            h = project_sum(h_new, target)
            
            # Track best
            c5_cur = compute_c5(h)
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h.copy()
        
        # Decay learning rate
        lr *= 0.5
    
    return best_h, best_c5

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = float(corr.max())
    s = np.clip(beta * (corr - m), -500.0, 0.0)
    e = np.exp(s)
    se = e.sum()
    val = (m + np.log(se) / beta) * dx
    w = e / se
    g = np.zeros(n)
    for j in range(len(corr)):
        wj = w[j]
        if wj < 1e-9:
            continue
        lag = j - (n - 1)
        if lag == 0:
            g += wj * (1.0 - 2.0 * h)
        elif lag > 0:
            g[:n - lag] += wj * (1.0 - h[lag:])
            g[lag:] -= wj * h[:n - lag]
        else:
            m2 = -lag
            g[m2:] += wj * (1.0 - h[:n - m2])
            g[:n - m2] -= wj * h[m2:]
    return val, g * dx

def adam_optimize(h_init, target, beta, n_iter, t0, budget, lr_start=0.008):
    n = len(h_init)
    h = h_init.copy()
    m1 = np.zeros(n)
    m2 = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    lr = lr_start
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    for it in range(n_iter):
        if time.time() - t0 > budget - 15:
            break
        val, grad = obj_and_grad(h, beta)
        t = it + 1
        m1 = b1 * m1 + (1.0 - b1) * grad
        m2 = b2 * m2 + (1.0 - b2) * grad * grad
        m1h = m1 / (1.0 - b1 ** t)
        m2h = m2 / (1.0 - b2 ** t)
        h = project_sum(h - lr * m1h / (np.sqrt(m2h) + eps), target)
        if it % 500 == 499:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            lr *= 0.92
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def subgrad_polish(h, target, n_active, max_iter, t0, budget):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    no_improve = 0
    for it in range(max_iter):
        if time.time() - t0 > budget - 2:
            break
        corr = np.correlate(best, 1.0 - best, mode="full")
        top_idx = np.argsort(corr)[::-1][:n_active]
        g = np.zeros(n)
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            if lag == 0:
                g += (1.0 - 2.0 * best)
            elif lag > 0:
                g[:n - lag] += (1.0 - best[lag:])
                g[lag:] -= best[:n - lag]
            else:
                m2 = -lag
                g[m2:] += (1.0 - best[:n - m2])
                g[:n - m2] -= best[m2:]
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        improved = False
        for step in [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0004, 0.0001]:
            ht = project_sum(best - step * g, target)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-15:
                best = ht
                best_c5 = c5t
                improved = True
                break
        if not improved:
            no_improve += 1
            if no_improve > 200 and it % 100 == 99:
                pert = np.random.randn(n) * 0.005
                ht = project_sum(best + pert, target)
                c5t = compute_c5(ht)
                if c5t < best_c5:
                    best = ht
                    best_c5 = c5t
                    no_improve = 0
            if no_improve > 2000:
                break
        else:
            no_improve = 0
    return best, best_c5

def binary_local_search(h, target, t0, budget, max_rounds=200):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    ones = np.where(best > 0.5)[0]
    zeros = np.where(best <= 0.5)[0]
    for rnd in range(max_rounds):
        if time.time() - t0 > budget - 5:
            break
        improved = False
        np.random.shuffle(ones)
        np.random.shuffle(zeros)
        n_try = min(len(ones), 50)
        for i in range(n_try):
            oi = ones[i]
            for j in range(min(len(zeros), 50)):
                zj = zeros[j]
                if oi == zj:
                    continue
                best[oi], best[zj] = best[zj], best[oi]
                c5t = compute_c5(best)
                if c5t < best_c5 - 1e-12:
                    best_c5 = c5t
                    improved = True
                    break
                else:
                    best[oi], best[zj] = best[zj], best[oi]
            if not improved and time.time() - t0 > budget - 10:
                break
        if not improved:
            break
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.asarray(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(240, 0.5)
    best_c5 = compute_c5(best_h)
    n = len(best_h)
    target = n / 2.0

    # Phase 1: Adam with beta annealing
    for beta in [30, 100, 400, 1500, 5000, 15000]:
        if time.time() - t0 > budget_s - 80:
            break
        h_new, c5_new = adam_optimize(best_h, target, beta, 10000, t0, budget_s, lr_start=0.008)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    best_h, best_c5 = subgrad_polish(best_h, target, 6, 5000, t0, budget_s)

    # Phase 2: Resolution changes
    for n_try in [360, 480, 600, 720]:
        if n_try == n:
            continue
        if time.time() - t0 > budget_s - 120:
            break
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = project_sum(np.interp(x_new, x_old, best_h), n_try / 2.0)
        tgt = n_try / 2.0
        for beta in [100, 1000, 8000]:
            if time.time() - t0 > budget_s - 50:
                break
            h_new, c5_new = adam_optimize(h_interp, tgt, beta, 6000, t0, budget_s, lr_start=0.006)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
            h_interp = h_new
        best_h, best_c5 = subgrad_polish(best_h, len(best_h) / 2.0, 6, 3000, t0, budget_s)
        n = len(best_h)
        target = n / 2.0

    # Phase 3: Diverse restarts
    n_best = len(best_h)
    target_best = n_best / 2.0
    h_starts = [project_sum(best_h + np.random.randn(n_best) * 0.04, target_best)]
    # Structured start: alternating blocks
    blk = max(1, n_best // 16)
    alt = np.zeros(n_best)
    for bi in range(n_best // blk + 1):
        s = bi * blk
        e = min(s + blk, n_best)
        alt[s:e] = 1.0 if bi % 2 == 0 else 0.0
    h_starts.append(project_sum(alt, target_best))
    # Random binary
    h_starts.append(project_sum((np.random.rand(n_best) > 0.5).astype(float), target_best))

    for h_try in h_starts:
        if time.time() - t0 > budget_s - 100:
            break
        for beta in [300, 3000]:
            if time.time() - t0 > budget_s - 50:
                break
            h_new, c5_new = adam_optimize(h_try, target_best, beta, 5000, t0, budget_s, lr_start=0.005)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
            h_try = h_new

    # Phase 4: Binary local search on rounded solution
    if time.time() - t0 < budget_s - 40:
        h_bin = np.where(best_h > 0.5, 1.0, 0.0)
        h_bin = project_sum(h_bin, target_best)
        h_bs, c5_bs = binary_local_search(h_bin, target_best, t0, budget_s, max_rounds=100)
        if c5_bs < best_c5:
            best_h = h_bs
            best_c5 = c5_bs

    # Phase 5: Final intensive subgradient polish
    n_final = len(best_h)
    target_final = n_final / 2.0
    best_h, best_c5 = subgrad_polish(best_h, target_final, 8, 50000, t0, budget_s)

    # Phase 6: Final Adam micro-tune
    if time.time() - t0 < budget_s - 20:
        h_new, c5_new = adam_optimize(best_h, target_final, 30000, 3000, t0, budget_s, lr_start=0.001)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 7: Micro-polish (very small steps)
    for it in range(15000):
        if time.time() - t0 > budget_s - 1:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        top_idx = np.argsort(corr)[::-1][:4]
        g = np.zeros(n_final)
        for idx in top_idx:
            lag = int(idx) - (n_final - 1)
            if lag == 0:
                g += (1.0 - 2.0 * best_h)
            elif lag > 0:
                g[:n_final - lag] += (1.0 - best_h[lag:])
                g[lag:] -= best_h[:n_final - lag]
            else:
                m2 = -lag
                g[m2:] += (1.0 - best_h[:n_final - m2])
                g[:n_final - m2] -= best_h[m2:]
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g /= gn
        improved = False
        for step in [0.004, 0.002, 0.001, 0.0004, 0.0001]:
            ht = project_sum(best_h - step * g, target_final)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-16:
                best_h = ht
                best_c5 = c5t
                improved = True
                break
        if not improved and it > 5000:
            break

    return best_h, best_c5, len(best_h)
