            step_init=0.02, step_decay=0.7, coord_every=150
        )
    
    # Phase 3: Try different resolutions
    resolutions = [200, 300, 500, 600]
    for nr in resolutions:
        if rem() < 50:
            break
        target_r = nr / 2.0
        dx_r = 2.0 / nr
        x_old = np.linspace(0, 2, n0, endpoint=False)
        x_new = np.linspace(0, 2, nr, endpoint=False)
        h_r = project_h(np.interp(x_new, x_old, best_h), target_r)
        c5_r = compute_c5(h_r)
        
        # SLSQP polish
        if rem() > 35:
            h_r, c5_r = slsqp_smooth(h_r, target_r, dx_r, 500, 2000, rem)
        # Subgradient
        if rem() > 20:
            h_r, c5_r = subgrad_optimize(
                h_r, target_r, dx_r, rem,
                max_time_even=rem() - 10, beta_start=20, beta_end=1000,
                max_iter=20000, n_lags=min(20, nr // 10), 
                momentum_start=0.8, momentum_end=0.92,
                step_init=0.015, step_decay=0.75, coord_every=120
            )
        
        if c5_r < best_c5:
            best_h = h_r
            best_c5 = c5_r
            n0 = nr
            target0 = target_r
            dx0 = dx_r
    
    # Phase 4: Final SLSQP + subgradient at best resolution
    n_f = len(best_h)
    target_f = n_f / 2.0
    dx_f = 2.0 / n_f
    
    if rem() > 15:
        best_h, best_c5 = slsqp_smooth(best_h, target_f, dx_f, 5000, 5000, rem)
    
    if rem() > 10:
        best_h, best_c5 = subgrad_optimize(
            best_h, target_f, dx_f, rem,
            max_time_even=rem() - 2, beta_start=100, beta_end=5000,
            max_iter=30000, n_lags=30, momentum_start=0.85, momentum_end=0.97,
            step_init=0.01, step_decay=0.8, coord_every=200
        )
    
    return best_h, best_c5, len(best_h)
