def compute_c5(h, n):

def project_feasible(h, n):

            s = h.sum()
            if s < 1e-13:

        if ct < best_c:
            best_c = ct
            best_h = ht.copy()
            per_attempt = max(5.0, (p3 - time.time()) / max(1, remaining_attempts - attempt - 1))

    # Phase 4: Multi-scale (15%)
    p4 = t0 + 0.85 * (budget_s - 5)
    for nh in [200, 300]:
        if time.time() > p4:
            break
        t_end = min(p4, time.time() + 50)
        ht, ct, nn = multi_scale_search(best_h, n, nh, t_end, seed + 555)
        if ct < best_c:
            best_c = ct
            best_h = ht.copy()
            n = nn

    # Phase 5: Final targeted LS + continuous (10%)
    if time.time() < dl - 30:
        t_end = min(dl - 30, time.time() + 60)
        hb = binarize(best_h, n, rng)
        ht, ct = targeted_ls(hb, n, t_end, max_sweeps=20)
        if ct < best_c:
            best_c = ct
            best_h = ht.copy()

    if time.time() < dl - 5:
        best_h, best_c = final_continuous(best_h, n, dl - 2)

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
