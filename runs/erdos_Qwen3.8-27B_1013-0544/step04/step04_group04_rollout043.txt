```python
            alpha_excl = np.concatenate([alpha_full[:n-1], alpha_full[n:]])  # (2n-2,)
            corr_excl = np.concatenate([corr[:n-1], corr[n:]])  # (2n-2,)
            # new_vals[c, k] = corr_excl[k] + d[c] * alpha_excl[k]
            new_vals = corr_excl[None, :] + d[:, None] * alpha_excl[None, :]  # (C, 2n-2)
            max_pl = np.max(new_vals, axis=1)  # (C,)
            
            # Quadratic term for k=n-1:
            quad = -candidates**2 + candidates + corr[n-1] - old + old**2  # (C,)
            
            total = np.maximum(max_pl, quad)
            best_idx = np.argmin(total)
            best_x = candidates[best_idx]
            best_val = total[best_idx]
            
            if best_val < float(np.max(corr)) - 1e-15 and abs(best_x - old) > 1e-12:
                h[i] = best_x
                d_best = best_x - old
                corr += d_best * alpha_full
                corr[n-1] += (best_x - best_x**2) - (old - old**2)
        
        # Project
        h = project_h(h, target)
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5 = float(np.max(corr) * dx)
        if c5 < best_c5:
            best_c5 = c5
    
    return h, best_c5
```

```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_descent(h, n_sweeps, time_fn, n_cand=50, seed=42):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    rng = np.random.RandomState(seed)
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(np.max(corr) * dx)
    no_improve = 0
    
    for sweep in range(n_sweeps):
        if time_fn() < 5:
            break
        sweep_best = best_c5
        order = rng.permutation(n)
        for i in order:
            old = h[i]
            # Compute alpha for positive lags
            alpha_p = np.zeros(n - 1)
            for p in range(1, n):
                v = 0.0
                if i >= p:
                    v += 1.0 - h[i - p]
                if i + p < n:
                    v -= h[i + p]
                alpha_p[p-1] = v
            # Full alpha (antisymmetric)
            alpha_full = np.zeros(2 * n - 1)
            alpha_full[n:] = alpha_p
            alpha_full[:n-1] = -alpha_p[::-1]
            # Candidates
            if old < 0.02:
                candidates = np.linspace(0.0, 0.5, n_cand)
            elif old > 0.98:
                candidates = np.linspace(0.5, 1.0, n_cand)
            else:
                candidates = np.linspace(0.0, 1.0, n_cand)
            d = candidates - old
            alpha_excl = np.concatenate([alpha_full[:n-1], alpha_full[n:]])
            corr_excl = np.concatenate([corr[:n-1], corr[n:]])
            new_vals = corr_excl[None, :] + d[:, None] * alpha_excl[None, :]
            max_pl = np.max(new_vals, axis=1)
            quad = -candidates**2 + candidates + corr[n-1] - old + old**2
            total = np.maximum(max_pl, quad)
            best_idx = np.argmin(total)
            best_x = candidates[best_idx]
            best_val = float(total[best_idx])
            # Refine
            if best_val < float(np.max(corr)) - 1e-12:
                lo = max(0.0, best_x - 3.0/n_cand)
                hi = min(1.0, best_x + 3.0/n_cand)
                fine = np.linspace(lo, hi, 30)
                df = fine - old
                nv = corr_excl[None, :] + df[:, None] * alpha_excl[None, :]
                mp = np.max(nv, axis=1)
                qf = -fine**2 + fine + corr[n-1] - old + old**2
                tf = np.maximum(mp, qf)
                fi = np.argmin(tf)
                if tf[fi] < best_val:
                    best_x = fine[fi]
                    best_val = float(tf[fi])
            if best_val < float(np.max(corr)) - 1e-12 and abs(best_x - old) > 1e-12:
                h[i] = best_x
                db = best_x - old
                corr += db * alpha_full
                corr[n-1] += (best_x - best_x**2) - (old - old**2)
                if best_val * dx < sweep_best - 1e-15:
                    sweep_best = best_val * dx
        # Project
        h = project_h(h, target)
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5 = float(np.max(corr) * dx)
        if c5 < best_c5:
            best_c5 = c5
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 30:
            break
    return h, best_c5

def subgrad_polish(h, n_iter, time_fn, top_k=5, steps=None):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    if steps is None:
        steps = [0.02, 0.01, 0.005, 0.002, 0.001]
    for it in range(n_iter):
        if time_fn() < 3:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[::-1][:top_k]
        grad = np.zeros(n)
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            if lag == 0:
                grad += (1.0 - 2.0 * h)
            elif 0 < lag < n:
                grad[lag:] += (1.0 - h[:-lag])
                grad[:n - lag] -= h[lag:]
            else:
                m2 = -lag
                grad[:n - m2] += (1.0 - h[m2:])
                grad[m2:] -= h[:n - m2]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        found = False
        for step in steps:
            ht = project_h(h - step * grad, target)
            c5t = compute_c5(ht)
            if c5t < compute_c5(h) - 1e-15:
                h = ht
                found = True
                break
        if not found:
            h = project_h(h + np.random.randn(n) * 0.01, target)
    return h, compute_c5(h)
```
