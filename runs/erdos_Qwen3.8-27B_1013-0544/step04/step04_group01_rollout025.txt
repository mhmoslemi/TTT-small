import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(3000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def subgradient_topk(h, n, top_k=3):
    """Compute weighted subgradient for top-k correlation values."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    n_full = len(c)
    # Select top indices
    if top_k >= n_full:
        sel = np.argsort(c)[::-1]
        w = np.full(top_k, 1.0 / top_k)
    else:
        sel = np.argsort(c)[::-1][:top_k]
        # Weight by excess above median
        med = np.median(c)
        vals = c[sel]
        w = np.maximum(vals - med, 1e-12)
        w = w / w.sum()
    grad = np.zeros(n)
    for idx, wt in zip(sel, w):
        u = idx - (n - 1)
        # Term 1: (1-h[j+u]) for valid j+u
        j1_lo = max(0, -u)
        j1_hi = min(n, n - u)
        if j1_lo < j1_hi:
            grad[j1_lo:j1_hi] += wt * (1.0 - h[j1_lo + u: j1_hi + u])
        # Term 2: -h[j-u] for valid j-u
        j2_lo = max(0, u)
        j2_hi = min(n, n + u)
        if j2_lo < j2_hi:
            grad[j2_lo:j2_hi] += wt * (-h[j2_lo - u: j2_hi - u])
    return grad * dx, c

def subgrad_descent(h, n, deadline, top_k=3, step0=0.3, momentum=0.8, decay=0.997):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    vel = np.zeros(n)
    step = step0
    stall = 0
    for _ in range(200000):
        if time.time() > deadline:
            break
        grad, c = subgradient_topk(h, n, top_k=min(top_k, len(c)))
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-13:
            break
        eff_step = min(step, 0.4 * best_c / (gnorm * gnorm + 1e-12))
        eff_step = max(eff_step, 1e-8)
        vel = momentum * vel - eff_step * grad
        h_new = project_feasible(h + vel, n)
        c_new = compute_c5(h_new, n)
        h = h_new
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 200:
                step *= decay
                vel *= 0.3
                stall = 0
    return best_h, best_c

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def logsumexp_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def slsqp_optimize(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(logsumexp_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def fine_local_search(h, n, deadline, steps):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for d in steps:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return best_h, best_c
                for j in range(n):
                    if i == j:
                        continue
                    hi = best_h[i] + d
                    hj = best_h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        best_h[i], best_h[j] = hi, hj
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-14:
                            best_c = c
                            improved = True
                        else:
                            best_h[i], best_h[j] = hi - d, hj + d
    return best_h, best_c

def structured_perturb(h, n, rng, mode=0):
    """Add structured perturbation to escape local optima."""
    h_new = h.copy()
    if mode == 0:
        # Low-frequency Fourier perturbation
        freqs = np.array([1, 2, 3, 4, 5])
        for f in freqs:
            amp = rng.uniform(0.02, 0.08)
            phase = rng.uniform(0, 2 * np.pi)
            h_new += amp * np.sin(2 * np.pi * f * np.arange(n) / n + phase)
    elif mode == 1:
        # Block swap perturbation
        block_size = max(5, n // 20)
        for _ in range(3):
            i = rng.randint(0, n - block_size)
            j = rng.randint(0, n - block_size)
            if abs(i - j) > block_size:
                h_new[i:i+block_size], h_new[j:j+block_size] = h[j:j+block_size].copy(), h[i:i+block_size].copy()
    else:
        # Random smooth perturbation
        noise = rng.normal(0, 0.05, n)
        # Smooth it
        kernel = np.ones(5) / 5
        noise = np.convolve(noise, kernel, mode='same')
        h_new += noise
    return project_feasible(h_new, n)
