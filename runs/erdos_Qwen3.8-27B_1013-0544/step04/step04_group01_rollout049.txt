```python
def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_overlaps(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return c

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
        else:
            mask = h > 1e-15
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)

def subgradient_max_overlap(h, n):
    """Compute subgradient of max_k overlap[k] with respect to h."""
    c = compute_overlaps(h, n)
    m = c.max()
    # Find all lags achieving the max (within tolerance)
    tol = 1e-10
    active = (c >= m - tol)
    # Subgradient is average of gradients at active lags
    # For lag k (0-indexed in full correlation, centered at n-1):
    # overlap[k] = sum_i h[i] * (1-h[i+k]) * dx
    # where k = (index - (n-1))
    # d(overlap[k])/d(h[j]) = sum_i [delta(i,j)*(1-h[i+k]) - delta(i+k,j)*h[i]] * dx
    #                         = (1-h[j+k])*dx - h[j-k]*dx (with bounds)
    dx = 2.0 / n
    n_lags = active.sum()
    grad = np.zeros(n)
    for idx in np.where(active)[0]:
        k = idx - (n - 1)  # actual lag
        # d(overlap[k])/d(h[j]) = (1-h[j+k])*dx - h[j-k]*dx
        # h[j+k] is defined for 0 <= j+k < n, else 0
        # h[j-k] is defined for 0 <= j-k < n, else 0
        j = np.arange(n)

        valid_jk = (j + k >= 0) & (j + k < n)
        h_jk[valid_jk] = h[j[valid_jk] + k]
        h_jmk = np.zeros(n)
        valid_jmk = (j - k >= 0) & (j - k < n)
        h_jmk[valid_jmk] = h[j[valid_jmk] - k]
        grad += (1 - h_jk) * dx - h_jmk * dx
    grad /= n_lags
    return grad, m, c

def projected_subgradient(h, n, deadline, max_iter=50000):
    """Projected subgradient descent with BB step size."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h_curr = h.copy()
    step = 0.01
    c_prev = best_c
    g_prev = None
    x_prev = None
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        grad, m, c = subgradient_max_overlap(h_curr, n)
        
        # Barzilai-Borwein step size update
        if g_prev is not None and x_prev is not None:
            dx_vec = h_curr - x_prev
            dg_vec = grad - g_prev
            s = np.dot(dx_vec, dx_vec)
            y = np.dot(dx_vec, dg_vec)
            if abs(y) > 1e-14:
                if s > 0:
                    step_bb = s / y
                    if step_bb > 0 and step_bb < 1.0:
                        step = 0.5 * step + 0.5 * step_bb
                    else:
                        step = max(0.001, step * 0.9)
                else:
                    step = max(0.001, step * 0.9)
        
        h_prev = h_curr.copy()
        g_prev = grad
        x_prev = h_prev
        
        # Gradient step
        h_curr = h_curr - step * grad
        h_curr = project_feasible(h_curr, n)
        
        c_curr = compute_c5(h_curr, n)
        if c_curr < best_c:
            best_c = c_curr
            best_h = h_curr.copy()
        
        # Decay step size if stuck
        if it > 100 and c_curr >= best_c:
            step = max(0.001, step * 0.99)
    
    return best_h, best_c
```

```python
        if res.fun < 1e10:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()
```

This is a good idea! By focusing SLSQP on only the active lags, we:
1. Simplify the objective (fewer terms in the max)
2. The gradient is more focused (only involves the active lags)
3. SLSQP can converge more precisely to the local optimum for this reduced problem

Then, after this focused optimization, we check if new lags have become active (their correlation has increased to the max level). If so, we add them to the active set and repeat.

This is essentially a **column generation** approach for the max problem!

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_overlaps(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_overlaps(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _sum_con(x, n):
    return x.sum() - n / 2.0

def _sum_con_jac(x, n):
    return np.ones(len(x))

def smooth_obj_full(x, n, alpha):
    c = compute_overlaps(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _make_focused_obj(n, active_lags, alpha):
    dx = 2.0 / n
    def obj(x):
        omx = 1.0 - x
        vals = np.empty(len(active_lags))
        for idx, k_idx in enumerate(active_lags):
            k = k_idx - (n - 1)
            if k >= 0:
                if k == 0:
                    vals[idx] = np.dot(x, omx) * dx
                else:
                    vals[idx] = np.dot(x[:n-k], omx[k:]) * dx
            else:
                kk = -k
                vals[idx] = np.dot(x[kk:], omx[:n-kk]) * dx
        m = vals.max()
        sh = alpha * (vals - m)
        mx = sh.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(sh))) / alpha)
    return obj

def slsqp_full(h, n, alpha, maxiter=4000, ftol=1e-18):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jack': None, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj_full, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.fun < 1e10:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

def slsqp_focused(h, n, active_lags, alpha, maxiter=4000, ftol=1e-18):
    obj = _make_focused_obj(n, active_lags, alpha)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.fun < 1e10:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

def find_active_lags(h, n, tol=1e-12):
    c = compute_overlaps(h, n)
    m = c.max()
    return np.where(c >= m - tol)[0]

def column_generation_optimize(h, n, deadline):
    """SLSQP with column generation: focus on active lags, recompute as needed."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alphas = [5e4, 2e5, 1e6, 5e6, 2e7, 1e8]
    for a in alphas:
        if time.time() > deadline:
            break
        active = find_active_lags(best_h, n, tol=1e-8)
        ht = slsqp_focused(best_h, n, active, a, maxiter=3000)
        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            best_h = ht
        # Check if new lags became active
        new_active = find_active_lags(best_h, n, tol=1e-10)
        if len(new_active) > len(active):
            ht = slsqp_focused(best_h, n, new_active, a, maxiter=2000)
            c = compute_c5(ht, n)
            if c < best_c - 1e-14:
                best_c = c
                best_h = ht
    return best_h, best_c

def micro_descent(h, n, deadline, radius=3):
    """Fine coordinate descent checking nearby pairs."""
    steps = [0.005, 0.002, 0.001, 0.0005, 0.0002]
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for d in steps:
        if time.time() > deadline:
            break
        for _pass in range(100):
            if time.time() > deadline:
                break
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return best_h, best_c
                for off in range(1, radius + 1):
                    for j in [(i + off) % n, (i - off) % n]:
                        if j == i:
                            continue
                        if best_h[i] + d <= 1.0 and best_h[j] - d >= 0.0:
                            best_h[i] += d; best_h[j] -= d
                            c = compute_c5(best_h, n)
                            if c < best_c - 1e-14:
                                best_c = c; improved = True
                            else:
                                best_h[i] -= d; best_h[j] += d
                        if best_h[i] - d >= 0.0 and best_h[j] + d <= 1.0:
                            best_h[i] -= d; best_h[j] += d
                            c = compute_c5(best_h, n)
                            if c < best_c - 1e-14:
                                best_c = c; improved = True
                            else:
                                best_h[i] += d; best_h[j] -= d
            if not improved:
                break
    return best_h, best_c

def gradient_guided_step(h, n, d, top_k=8):
    """One step: adjust coordinates guided by gradient of top-k lags."""
    c = compute_overlaps(h, n)
    m = c.max()
    top = np.argsort(c)[::-1][:top_k]
    dx_ = 2.0 / n
    grad = np.zeros(n)
    for idx in top:
        k = idx - (n - 1)
        j = np.arange(n)
        h_jk = np.zeros(n)
        v1 = (j + k >= 0) & (j + k < n)
        h_jk[v1] = h[j[v1] + k]
        h_jmk = np.zeros(n)
        v2 = (j - k >= 0) & (j - k < n)
        h_jmk[v2] = h[j[v2] - k]
        grad += ((1 - h_jk) - h_jmk) * dx_
    grad /= top_k
    h_new = h - d * grad
    h_new = project_feasible(h_new, n)
    return h_new
```
