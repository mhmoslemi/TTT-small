import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -400, 400)
    return (m + np.log(np.sum(np.exp(shifted)) / len(corr)) / beta) * dx

def weighted_subgrad(h, weight_beta):
    """Compute approximate subgradient of max correlation using softmax weights."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = np.clip(weight_beta * (corr - m), -400, 400)
    w = np.exp(shifted)
    w /= w.sum()
    
    grad = np.zeros(n)
    for idx in range(len(w)):
        if w[idx] < 1e-10:
            continue
        lag = idx - (n - 1)
        if lag == 0:
            grad += w[idx] * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[:n-lag] += w[idx] * (1.0 - h[lag:])
            grad[lag:] -= w[idx] * h[:n-lag]
        else:
            mm = -lag
            grad[:n-mm] += w[idx] * (1.0 - h[mm:])
            grad[mm:] -= w[idx] * h[:n-mm]
    return grad

def slsqp_opt(h, target, dx, betas, max_iter, time_fn):
    n = len(h)
    bounds = Bounds(np.zeros(n), np.ones(n))
    best = h.copy()
    best_c5 = compute_c5(best)
    for beta in betas:
        if time_fn() < 15:
            break
        def obj(h_in):
            return smooth_max_obj(h_in, dx, beta)
        def con(h_in):
            return h_in.sum() - target
        def jac(h_in):
            return np.ones(n)
        try:
            res = minimize(obj, best, method='SLSQP', bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': con, 'jac': jac}],
                          options={'maxiter': max_iter, 'ftol': 1e-18})
            h_new = project_h(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best = h_new
                best_c5 = c5_new
        except Exception:
            pass
    return best, best_c5

def subgrad_opt(h, target, time_fn, max_iters, weight_beta, step_list, 
                perturb_every, perturb_scale, momentum):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    vel = np.zeros(n)
    no_improve = 0
    for it in range(max_iters):
        if time_fn() < 3:
            break
        g = weighted_subgrad(best, weight_beta)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        vel = momentum * vel + (1.0 - momentum) * (g / gn)
        vn = np.linalg.norm(vel)
        if vn < 1e-14:
            break
        d = -vel / vn
        found = False
        for step in step_list:
            ht = project_h(best + step * d, target)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-15:
                best = ht
                best_c5 = c5t
                no_improve = 0
                found = True
                break
        if not found:
            no_improve += 1
            if no_improve >= perturb_every and it % perturb_every == perturb_every - 1:
                p = np.random.randn(n) * perturb_scale
                ht = project_h(best + p, target)
                c5t = compute_c5(ht)
                if c5t < best_c5:
                    best = ht
                    best_c5 = c5t
                    vel *= 0.5
                    no_improve = 0
            if no_improve > perturb_every * 15:
                break
    return best, best_c5

    candidates.append(project_h(1.0 - h, target))
    
    return candidates

def weighted_subgrad(h, weight_beta):

    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)

    # lag = 0
    w0 = w[n-1]
    if w0 > 1e-14:
        grad += w0 * (1.0 - 2.0 * h)
    # Combine positive and negative lags
    for m in range(1, n):
        wp = w[n-1+m]  # weight for lag +m
        wn = w[n-1-m]  # weight for lag -m

        if wp > 1e-14:
            grad[m:] += wp * (1.0 - h[:n-m])

    vel = np.zeros(n)
    no_improve = 0

            break
        g = weighted_subgrad(best, weight_beta)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        vel = momentum * vel + (1.0 - momentum) * (g / gn)
        vn = np.linalg.norm(vel)
        if vn < 1e-14:

        found = False
        for step in step_list:

                best_c5 = c5t
                no_improve = 0
                found = True

                    cand, c5_c = slsqp_opt(cand, target0, dx0,
                        [100, 500, 2000, 8000], 2000, rem)
                if rem() > 20:
                    cand, c5_c = subgrad_opt(cand, target0, rem,
                        max_iters=8000, weight_beta=120,
                        step_list=[0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002],
                        perturb_every=300, perturb_scale=0.008, momentum=0.88,
                        max_active=100)
                if c5_c < best_c5:
                    best_h = cand
                    best_c5 = c5_c

    # Phase 4: Try n=500 for better approximation
    if rem() > 200 and n0 == 400:
        n2 = 500
        target2 = n2 / 2.0
        dx2 = 2.0 / n2
        x_old = np.linspace(0, 2, n0)
        x_new = np.linspace(0, 2, n2)
        h2 = project_h(np.interp(x_new, x_old, best_h), target2)
        if rem() > 130:
            h2, c5_2 = slsqp_opt(h2, target2, dx2,
                [50, 200, 800, 3000, 10000], 3500, rem)
        if rem() > 40:
            h2, c5_2 = subgrad_opt(h2, target2, rem,
                max_iters=12000, weight_beta=80,
                step_list=[0.008, 0.004, 0.002, 0.001, 0.0005, 0.0002],
                perturb_every=350, perturb_scale=0.007, momentum=0.88,
                max_active=120)
        if c5_2 < best_c5:
            best_h = h2
            best_c5 = c5_2

    # Phase 5: Final focused polish
    n_f = len(best_h)
    target_f = n_f / 2.0
    dx_f = 2.0 / n_f
    if rem() > 30:
        best_h, best_c5 = slsqp_opt(best_h, target_f, dx_f,
            [100, 500, 2000, 8000, 30000], 5000, rem)
    if rem() > 8:
        best_h, best_c5 = subgrad_opt(best_h, target_f, rem,
            max_iters=20000, weight_beta=250,
            step_list=[0.003, 0.001, 0.0005, 0.0002, 0.0001, 0.00005],
            perturb_every=500, perturb_scale=0.003, momentum=0.92,
            max_active=80)

    return best_h, best_c5, len(best_h)
