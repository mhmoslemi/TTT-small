    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]

    """Simulated annealing on binary sequences."""
    rng = np.random.RandomState(seed)
    b = h.copy()
    
    # Initial correlation
    c = np.correlate(b, 1 - b, mode='full')
    current_max = c.max()
    best_max = current_max
    best_b = b.copy()
    
    # Temperature schedule
    t_start = 2.0
    t_end = 0.01
    n_iters = 5000000  # will be limited by deadline

    # t(t) = t_start * (t_end/t_start)^t  (t in [0,1])

                best_b = b.copy()
        else:
            # Revert
            b[i], b[j] = b[j], b[i]
        
        temp *= cool_rate

For k<0: c[k] = sum_{n=-k}^{n-1} h[n+k]*(1-h[n])

The maximum over all k (from -(n-1) to n-1) is what we minimize.

Note: c[k] for k>0 counts "1 at position n+k, 0 at position n" for n = 0 to n-1-k.
c[-k] for k>0 counts "1 at position n-k, 0 at position n" for n = k to n-1, which is the same as "0 at position m, 1 at position m+k" for m = 0 to n-1-k.

So c[k] counts "01" patterns at distance k (0 first, 1 second), and c[-k] counts "10" patterns at distance k (1 first, 0 second).

        best_delta = 0.0
        best_i = -1
        best_j = -1
        ones_idx = np.where(b == 1)[0]
        zeros_idx = np.where(b == 0)[0]
        n = len(b)
        k_shift = curr_max_idx - (n - 1)
        for i in ones_idx:
            if time.time() > deadline:
                return b
            for j in zeros_idx:
                b[i] = 0.0
                b[j] = 1.0
                c_new = np.correlate(b, 1 - b, mode='full')
                nv = c_new.max()
                b[i] = 1.0
                b[j] = 0.0
                if nv < curr - 1e-12:
                    delta = curr - nv
                    if delta > best_delta:
                        best_delta = delta
                        best_i = i
                        best_j = j
                        curr = nv
        if best_i >= 0:
            b[best_i] = 0.0
            b[best_j] = 1.0
            c = np.correlate(b, 1 - b, mode='full')
            curr_max_idx = c.argmax()
            curr = c.max()
            improved = True
    return b

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n = 200

    b = to_binary(h_init, n)
    best_b = b.copy()
    best_c = compute_c5(best_b, n)

    # Phase 1: SA (35% budget)
    alloc1 = (deadline - t0) * 0.35
    b1, c1 = sa_run(b, n, t0 + alloc1, seed, t_start=5.0, t_end=0.01)
    if c1 < best_c:
        best_c = c1
        best_b = b1
    # Quick local search
    if time.time() < deadline - 30:
        bl = local_search(best_b, n, min(deadline - 30, time.time() + 8))
        cl = compute_c5(bl, n)
        if cl < best_c:
            best_c = cl
            best_b = bl

    # Phase 2: SA restarts with perturbations (45% budget)
    remaining = deadline - time.time()
    n_restarts = max(4, int(remaining / 60))
    for r in range(n_restarts):
        if time.time() > deadline - 30:
            break
        b_pert = best_b.copy()
        k_swap = rng.randint(4, 12)
        ones = list(np.where(b_pert == 1)[0])
        zeros = list(np.where(b_pert == 0)[0])
        for _ in range(k_swap):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            b_pert[i] = 0.0; b_pert[j] = 1.0
            ones[ones.index(i)] = j
            zeros[zeros.index(j)] = i
        alloc_r = min(55, (deadline - time.time() - 30) * 0.28)
        if alloc_r < 8:
            break
        seed_r = seed + r * 7919 + 13
        b_r, c_r = sa_run(b_pert, n, time.time() + alloc_r, seed_r, t_start=3.0, t_end=0.008)
        if c_r < best_c:
            best_c = c_r
            best_b = b_r
        if time.time() < deadline - 25:
            bl = local_search(best_b, n, min(deadline - 25, time.time() + 5))
            cl = compute_c5(bl, n)
            if cl < best_c:
                best_c = cl
                best_b = bl

    # Phase 3: 2-swap search (15% budget)
    if time.time() < deadline - 5:
        b2, c2 = two_swap_local(best_b, n, min(deadline - 5, time.time() + 40), seed + 999)
        if c2 < best_c:
            best_c = c2
            best_b = b2
        bl = local_search(best_b, n, deadline - 3)
        cl = compute_c5(bl, n)
        if cl < best_c:
            best_c = cl
            best_b = bl

    # Phase 4: Final polish
    if time.time() < deadline - 1:
        bl = local_search(best_b, n, deadline - 1)
        cl = compute_c5(bl, n)
        if cl < best_c:
            best_c = cl
            best_b = bl

    return best_b, best_c, n
