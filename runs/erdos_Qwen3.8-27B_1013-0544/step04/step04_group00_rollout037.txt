```python
def compute_c5(h, n):

            room = 1.0 - h
            s = room.sum()

                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_step(h, n, alpha, maxiter, deadline):

        m = g.max()

        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    
    def cons_fun(x):
        return x.sum() - n / 2.0
```

```python
def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_box_sum(h, target):
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def lse_objective(x, n, alpha, target):

    g = compute_g(h, n)
    m = g.max()
    shifted = alpha * (g - m)
    shifted -= shifted.max()
    return m + np.log(np.exp(shifted).sum()) / alpha
```

```python
def lse_gradient(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1-h, 'full')
    g = c * dx
    m = g.max()
    shifted = alpha * (g - m)
    shifted -= shifted.max()
    w = np.exp(shifted)
    w /= w.sum()
    
    # w has length 2n-1, w[m] corresponds to lag j = m-(n-1)
    # grad[i] = dx * sum_m w[m] * [(1-h[i+j])*1{0<=i+j<n} - h[i-j]*1{0<=i-j<n}]
    
    # Build the gradient efficiently
    grad = np.zeros(n)
    # For each m (lag), add w[m] * (contribution to grad)
    # Contribution of lag j=m-(n-1) to grad:
    #   For valid i with i+j in [0,n): grad[i] += w[m] * (1-h[i+j])
    #   For valid i with i-j in [0,n): grad[i] -= w[m] * h[i-j]
    
    for m in range(2*n-1):
        j = m - (n-1)
        if w[m] < 1e-15:
            continue
        if j >= 0:
            # i+j in [0,n) means i in [0, n-j)
            # i-j in [0,n) means i in [j, n)
            if j == 0:
                grad += w[m] * (1 - h)  # (1-h[i+0]) for i in [0,n)
                grad -= w[m] * h  # h[i-0] for i in [0,n)
            else:
                # First term: i in [0, n-j), add w[m]*(1-h[i+j])
                grad[:n-j] += w[m] * (1 - h[j:])
                # Second term: i in [j, n), subtract w[m]*h[i-j]
                grad[j:] -= w[m] * h[:n-j]
        else:
            j_neg = -j
            # i+j in [0,n) means i in [j_neg, n)
            # i-j in [0,n) means i in [0, n-j_neg)
            # First term: i in [j_neg, n), add w[m]*(1-h[i+j]) = w[m]*(1-h[i-j_neg])
            grad[j_neg:] += w[m] * (1 - h[:j_neg])
            # Second term: i in [0, n-j_neg), subtract w[m]*h[i-j] = w[m]*h[i+j_neg]
            grad[:n-j_neg] -= w[m] * h[j_neg:]
    
    return dx * grad
```

```python
            break
        grad = lse_grad(h, n, alpha)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        obj_cur = lse_obj(h, n, alpha)
        accepted = False
        step = lr
        for _ in range(25):
            h_new = project_box_sum(h - step * grad, target)
            obj_new = lse_obj(h_new, n, alpha)
            if obj_new < obj_cur - 1e-15:
                h = h_new
                accepted = True
                break
            step *= 0.5
        if not accepted:
            lr *= 0.85
            stalling += 1
            if stalling > 30:
                break
        else:
            stalling = 0
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
    return best_h, best_c

def slsqp_step(h, n, alpha, maxiter, deadline):
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
        m = g.max()
        sh = alpha * (g - m)
        sh -= sh.max()
        return m + (sh.max() + np.log(np.exp(sh).sum())) / alpha
    def obj_grad(x):
        return lse_grad(x, n, alpha)
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_box_sum(res.x, n / 2.0)
    except Exception:
        try:
            res = minimize(obj, h, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
            return project_box_sum(res.x, n / 2.0)
        except Exception:
            return h

def subgrad_polish(h, n, deadline, max_iter=5000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.3
    no_imp = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = compute_g(h, n)
        m = g.max()
        ks = int(np.argmax(g))
        lag = ks - (n - 1)
        grad = np.zeros(n)
        dx = 2.0 / n
        if lag >= 0:
            for i in range(n - lag):
                grad[i] += (1 - h[i + lag]) * dx
                grad[i + lag] += (-h[i]) * dx
        else:
            ln = -lag
            for i in range(n - ln):
                grad[i + ln] += (1 - h[i]) * dx
                grad[i] += (-h[i + ln]) * dx
        ng = np.linalg.norm(grad)
        if ng < 1e-13:
            break
        step = lr / ng
        h_new = project_box_sum(h - step * grad, n / 2.0)
        c_new = compute_c5(h_new, n)
        if c_new < m - 1e-14:
            h = h_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            lr *= 0.92
            no_imp += 1
        if no_imp > 100:
            break
    return best_h, best_c

def binary_local_search(h, n, deadline, max_sweeps=80):
    target = int(n / 2)
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    best_c = compute_c5(h_bin, n)
    ones = set(np.where(h_bin == 1)[0].tolist())
    zeros = list(np.where(h_bin == 0)[0].tolist())
    rng = np.random.RandomState(7)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        rng.shuffle(zeros)
        for i in zeros:
            if time.time() > deadline:
                break
            candidates = list(ones)
            rng.shuffle(candidates)
            for j in candidates[:200]:
                h_bin[i], h_bin[j] = 1.0, 0.0
                c = compute_c5(h_bin, n)
                if c < best_c:
                    best_c = c
                    ones.add(i)
                    ones.remove(j)
                    zeros.remove(i)
                    zeros.append(j)
                    improved = True
                    break
                else:
                    h_bin[i], h_bin[j] = 0.0, 1.0
        if not improved:
            break
    return h_bin, best_c

def try_resolution(h_best, n_best, c_best, deadline, nh, rng):
    xo = np.linspace(0, 2, n_best)
    xn = np.linspace(0, 2, nh)
    hh = project_box_sum(np.interp(xn, xo, h_best), nh / 2.0)
    for alpha in [5000, 50000, 500000]:
        if time.time() > deadline:
            break
        hh = slsqp_step(hh, nh, alpha, 500, deadline)
    hh, c = pg_optimize(hh, nh, 100000, 3000, deadline)
    if c < c_best:
        return hh, c, nh
    return h_best, c_best, n_best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 3
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 100
        h0 = np.full(n0, 0.5)

    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    n = n0

    # Phase 1: Quick SLSQP to establish good starting point (20%)
    p1 = t0 + 0.20 * (budget_s - 3)
    h = best_h.copy()
    for alpha in [2000, 20000, 200000, 2000000]:
        if time.time() > p1:
            break
        h = slsqp_step(h, n, alpha, 600, p1)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    # Phase 2: Projected gradient with multiple alphas (25%)
    p2 = t0 + 0.45 * (budget_s - 3)
    for alpha in [5000, 50000, 500000, 5000000]:
        if time.time() > p2:
            break
        h_pg, c_pg = pg_optimize(best_h.copy(), n, alpha, 8000, p2)
        if c_pg < best_c:
            best_c = c_pg
            best_h = h_pg.copy()

    # Phase 3: SLSQP from best with high alpha (15%)
    p3 = t0 + 0.60 * (budget_s - 3)
    for alpha in [100000, 1000000, 10000000, 100000000]:
        if time.time() > p3:
            break
        h = slsqp_step(best_h.copy(), n, alpha, 1000, p3)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        # Also try from perturbed
        hp = project_box_sum(best_h + 0.02 * rng.randn(n), n / 2.0)
        h2 = slsqp_step(hp, n, alpha, 400, p3)
        c2 = compute_c5(h2, n)
        if c2 < best_c:
            best_c = c2
            best_h = h2.copy()

    # Phase 4: Multi-resolution attempt (15%)
    p4 = t0 + 0.75 * (budget_s - 3)
    if time.time() < p4 - 20:
        for nh in [200, 300]:
            if time.time() > p4 - 10:
                break
            h_try, c_try, n_try = try_resolution(best_h, n, best_c, p4 - 10, nh, rng)
            if c_try < best_c:
                best_c = c_try
                best_h = h_try.copy()
                n = n_try

    # Phase 5: Subgradient polish on exact max (10%)
    p5 = t0 + 0.85 * (budget_s - 3)
    h, c = subgrad_polish(best_h.copy(), n, p5, max_iter=8000)
    if c < best_c:
        best_c = c
        best_h = h.copy()

    # Phase 6: Binary local search (8%)
    p6 = t0 + 0.93 * (budget_s - 3)
    hb, cb = binary_local_search(best_h.copy(), n, p6, max_sweeps=60)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()

    # Phase 7: Final SLSQP polish (4%)
    if time.time() < dl - 3:
        h = slsqp_step(best_h.copy(), n, 500000000, 1500, dl - 3)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    best_h = project_box_sum(best_h, n / 2.0)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
```
