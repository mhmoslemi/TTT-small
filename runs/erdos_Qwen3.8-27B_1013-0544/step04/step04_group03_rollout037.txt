```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def corr_max(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")))

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def slsqp_obj(h, beta, n):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, beta) * (2.0 / n)

def sum_con(x, target):
    return np.sum(x) - target

def sum_jac(x, target):
    return np.ones(x.shape)

def slsqp(h0, n, target, beta, maxiter=8000):
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac, 'args': (target,)}]
    try:
        r = minimize(slsqp_obj, h0, args=(beta, n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_h(r.x, target)
    except Exception:
        return h0

def resample(h, n_new):
    old_x = np.linspace(0, 2, len(h), endpoint=False)
    new_x = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(new_x, old_x, h)

def binary_snap(h, n):
    n_ones = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:n_ones]] = 1.0
    return h_bin

def binary_ls(h_bin, n, t0, budget, min_time=5, max_stall=40):
    h = h_bin.astype(float)
    cur_cm = corr_max(h)
    best_h = h.copy()
    best_cm = cur_cm
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    stall = 0
    rng = np.random.default_rng(12345)
    while budget - (time.time() - t0) > min_time:
        best_delta = 0.0
        best_pair = None
        for a in ones:
            if budget - (time.time() - t0) < min_time:
                break
            for b in zeros:
                h[a], h[b] = 0.0, 1.0
                cm = corr_max(h)
                delta = cm - cur_cm
                if delta < best_delta - 1e-15:
                    best_delta = delta
                    best_pair = (a, b)
                h[a], h[b] = 1.0, 0.0
        if best_pair is not None:
            a, b = best_pair
            h[a], h[b] = 0.0, 1.0
            ones.remove(a)
            ones.append(b)
            zeros.remove(b)
            zeros.append(a)
            cur_cm += best_delta
            stall = 0
            if cur_cm < best_cm:
                best_cm = cur_cm
                best_h = h.copy()
        else:
            stall += 1
            if stall <= 8 and len(ones) > 3:
                for _ in range(3):
                    ai = rng.integers(len(ones))
                    bi = rng.integers(len(zeros))
                    a, b = ones[ai], zeros[bi]
                    h[a], h[b] = 0.0, 1.0
                    ones[ai] = b
                    zeros[bi] = a
                cur_cm = corr_max(h)
                stall = 0
            elif stall > max_stall:
                break
    return best_h, best_cm

def coord_desc(h, n, target, t0, budget, max_sweeps=150):
    cur = h.copy()
    best_c5 = compute_c5(cur)
    deltas = [0.05, -0.05, 0.02, -0.02, 0.01, -0.01, 0.005, -0.005, 0.002, -0.002]
    for sw in range(max_sweeps):
        if budget - (time.time() - t0) < 3:
            break
        improved = False
        for j in range(n):
            if budget - (time.time() - t0) < 1:
                return cur, best_c5
            for d in deltas:
                trial = cur.copy()
                trial[j] = np.clip(trial[j] + d, 0.0, 1.0)
                trial = project_h(trial, target)
                c = compute_c5(trial)
                if c < best_c5 - 1e-14:
                    cur = trial
                    best_c5 = c
                    improved = True
                    break
        if not improved and sw > 5:
            break
    return cur, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.asarray(initial_h_values, dtype=float).copy()
    else:
        n_init = 384
        h_parent = np.random.rand(n_init)
        h_parent = project_h(h_parent, n_init / 2.0)
    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)
    best_n = len(h_parent)
    # Phase 1: Binary local search at parent n
    n1 = best_n
    h_bin = binary_snap(best_h, n1)
    h_bin, cm_bin = binary_ls(h_bin, n1, t0, budget_s, min_time=15)
    c_bin = cm_bin * 2.0 / n1
    if c_bin < best_c5:
        best_h = h_bin.copy()
        best_c5 = c_bin
    remaining = budget_s - (time.time() - t0)
    if remaining > 20:
        h_s = slsqp(h_bin, n1, n1 / 2.0, 500, maxiter=6000)
        c_s = compute_c5(h_s)
        if c_s < best_c5:
            best_h = h_s
            best_c5 = c_s
    # Phase 2: Random binary starts
    remaining = budget_s - (time.time() - t0)
    rng = np.random.default_rng(seed + 7)
    n_starts = max(2, min(6, int(remaining / 80)))
    for si in range(n_starts):
        if budget_s - (time.time() - t0) < 35:
            break
        h_rand = np.zeros(n1)
        positions = rng.choice(n1, n1 // 2, replace=False)
        h_rand[positions] = 1.0
        h_r, cm_r = binary_ls(h_rand, n1, t0, budget_s, min_time=8, max_stall=25)
        c_r = cm_r * 2.0 / n1
        if c_r < best_c5:
            best_h = h_r.copy()
            best_c5 = c_r
            remaining = budget_s - (time.time() - t0)
            if remaining > 15:
                h_s = slsqp(h_r, n1, n1 / 2.0, 1000, maxiter=4000)
                c_s = compute_c5(h_s)
                if c_s < best_c5:
                    best_h = h_s
                    best_c5 = c_s
    # Phase 3: SLSQP with increasing beta
    remaining = budget_s - (time.time() - t0)
    for beta in [300, 1500, 6000, 25000]:
        if budget_s - (time.time() - t0) < 10:
            break
        best_h = slsqp(best_h, best_n, best_n / 2.0, beta, maxiter=8000)
        best_c5 = compute_c5(best_h)
    # Phase 4: Higher resolutions
    remaining = budget_s - (time.time() - t0)
    for n_hr in [448, 512]:
        if remaining < 50:
            break
        target_hr = n_hr / 2.0
        cur_hr = project_h(resample(best_h, n_hr), target_hr)
        h_bin_hr = binary_snap(cur_hr, n_hr)
        h_bin_hr, cm_hr = binary_ls(h_bin_hr, n_hr, t0, budget_s, min_time=8, max_stall=20)
        c_hr = cm_hr * 2.0 / n_hr
        if c_hr < best_c5:
            best_h = h_bin_hr.copy()
            best_c5 = c_hr
            best_n = n_hr
        remaining = budget_s - (time.time() - t0)
        if remaining > 15:
            h_s = slsqp(h_bin_hr, n_hr, target_hr, 500, maxiter=4000)
            c_s = compute_c5(h_s)
            if c_s < best_c5:
                best_h = h_s
                best_c5 = c_s
                best_n = n_hr
    # Phase 5: Final SLSQP polish
    remaining = budget_s - (time.time() - t0)
    for beta in [10000, 50000]:
        if budget_s - (time.time() - t0) < 5:
            break
        best_h = slsqp(best_h, best_n, best_n / 2.0, beta, maxiter=12000)
        best_c5 = compute_c5(best_h)
    # Phase 6: Coordinate descent
    remaining = budget_s - (time.time() - t0)
    if remaining > 5:
        best_h, best_c5 = coord_desc(best_h, best_n, best_n / 2.0, t0, budget_s, max_sweeps=200)
    return best_h, best_c5, best_n
```
