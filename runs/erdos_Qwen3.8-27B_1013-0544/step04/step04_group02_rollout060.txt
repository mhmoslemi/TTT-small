This is equivalent to: minimize max_j (corr[j] + δ * C[j]) where δ = h[i] - h[i]_old.

This is minimizing the max of linear functions of δ! The optimal δ is where the top two (or more) lines cross.

We can find this by:
1. For each j, the line is f_j(δ) = corr[j] + δ * C[j]
2. The envelope (max of all f_j) is a convex piecewise-linear function of δ
3. We want to minimize this over δ ∈ [-(h[i]_old), 1 - h[i]_old]

The minimum occurs either at a breakpoint (where two lines cross) or at a boundary.

To find the optimal δ:
- Compute all breakpoints: δ_jk = (corr[k] - corr[j]) / (C[j] - C[k]) for pairs where C[j] ≠ C[k]
- But there are O(n²) breakpoints, which is too many.
- Instead, use a ternary search or bisection on the convex function.

Actually, since the function is convex in δ, we can use bisection on the derivative (subgradient). The subgradient at δ is C[j*] where j* is an argmax of f_j(δ).

def coord_descent_fast(h, n, deadline, seed):
    np.random.seed(seed)
    ts = n / 2.0
    dx = 2.0 / n
    c5 = compute_c5(h)
    corr = np.correlate(h, 1.0 - h, mode='full')  # without dx
    
    n_full = 2 * n - 1
    
    for sweep in range(1000):
        if time.time() > deadline:
            break
        improved = False
        
        for i in range(n):
            old_val = h[i]
            
            # Compute C[j] = coefficient of delta in new_corr[j]
            # C[j] = (1 - h[i+j-(n-1)]) * [i+j-(n-1) in [0,n-1]] - h[n-1+i-j] * [n-1+i-j in [0,n-1]]
            C = np.zeros(n_full)
            
            # First term: i+j-(n-1) in [0, n-1] => j in [n-1-i, 2n-2-i]
            j_start1 = max(0, n - 1 - i)
            j_end1 = min(n_full - 1, 2 * n - 2 - i)
            if j_start1 <= j_end1:
                t = i + np.arange(j_start1, j_end1 + 1) - (n - 1)  # t = i+j-(n-1), range [0, ...]
                C[j_start1:j_end1 + 1] += (1.0 - h[t])
            
            # Second term: n-1+i-j in [0, n-1] => j in [i, n-1+i]
            j_start2 = max(0, i)
            j_end2 = min(n_full - 1, n - 1 + i)
            if j_start2 <= j_end2:
                m = n - 1 + i - np.arange(j_start2, j_end2 + 1)  # m = n-1+i-j
                C[j_start2:j_end2 + 1] += -h[m]
            
            # Now minimize max_j (corr[j] + delta * C[j]) over delta in [-old_val, 1-old_val]
            # Use ternary search
            lo = -old_val
            hi = 1.0 - old_val
            
            if hi - lo < 1e-12:
                continue
            
            # The function f(delta) = max_j(corr[j] + delta * C[j]) is convex in delta
            # Use golden section / ternary search
            gr = (np.sqrt(5.0) - 1.0) / 2.0  # golden ratio conjugate
            a1 = lo + (1 - gr) * (hi - lo)
            b1 = lo + gr * (hi - lo)
            fa = np.max(corr + a1 * C)
            fb = np.max(corr + b1 * C)
            
            for _ in range(40):
                if fa < fb:
                    hi = b1
                    b1 = a1
                    fb = fa
                    a1 = lo + (1 - gr) * (hi - lo)
                    fa = np.max(corr + a1 * C)
                else:
                    lo = a1
                    a1 = b1
                    fa = fb
                    b1 = lo + gr * (hi - lo)
                    fb = np.max(corr + b1 * C)
            
            opt_delta = (lo + hi) / 2.0
            new_val = old_val + opt_delta
            
            if abs(new_val - old_val) > 1e-10:
                h[i] = new_val
                # Update corr
                corr += opt_delta * C
                improved = True
        
        # Project to maintain feasibility
        h = project_feasible(h, ts)
        # Recompute corr after projection (since projection changed h)
        corr = np.correlate(h, 1.0 - h, mode='full')
        new_c5 = float(np.max(corr) * dx)
        
        if not improved:
            break
    
    return h, compute_c5(h)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_overlap(h, lag, n):
    if lag >= 0:
        m = n - lag
        if m <= 0:
            return np.zeros(n)
        g = np.zeros(n)
        g[:m] += (1.0 - h[lag:lag + m])
        g[lag:lag + m] -= h[:m]
    else:
        L = -lag
        m = n - L
        if m <= 0:
            return np.zeros(n)
        g = np.zeros(n)
        g[L:L + m] += (1.0 - h[:m])
        g[:m] -= h[L:L + m]
    return g

                    C[j1s:j1e + 1] += (1.0 - h[t])
                j2s = max(0, i)
                j2e = min(n_full - 1, n - 1 + i)
                if j2s <= j2e:
                    m = n - 1 + i - np.arange(j2s, j2e + 1)
                    C[j2s:j2e + 1] += -h[m]
                lo = -old_val
                hi = 1.0 - old_val
                if hi - lo < 1e-12:
                    continue
                a1 = lo + (1 - gr) * (hi - lo)
                b1 = lo + gr * (hi - lo)
                f_a = np.max(corr + a1 * C)
                f_b = np.max(corr + b1 * C)
                for _ in range(35):
                    if f_a < f_b:
                        hi = b1
                        b1 = a1
                        f_b = f_a
                        a1 = lo + (1 - gr) * (hi - lo)
                        f_a = np.max(corr + a1 * C)
                    else:
                        lo = a1
                        a1 = b1
                        f_a = f_b
                        b1 = lo + gr * (hi - lo)
                        f_b = np.max(corr + b1 * C)
                best_d = (lo + hi) / 2.0
            new_val = old_val + best_d
            if abs(new_val - old_val) > 1e-10:
                h[i] = new_val
                corr += best_d * C
                improved = True
        h = project_feasible(h, ts)
        corr = np.correlate(h, 1.0 - h, mode='full')
        if not improved:
            break
    return h, compute_c5(h)

def smooth_max_obj(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = corr.max()
    shifted = np.clip(beta * (corr - m), -700, 700)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def con_eq(h, ts):
    return h.sum() - ts

def con_eq_jac(h, ts):
    return np.ones_like(h)

def make_periodic(n, period, seed):
    np.random.seed(seed)
    base = np.random.rand(min(period, n))
    mn, mx = base.min(), base.max()
    if mx - mn > 1e-10:
        base = (base - mn) / (mx - mn) * 0.6 + 0.2
    else:
        base = np.full(len(base), 0.5)
    pattern = np.tile(base, n // len(base) + 1)[:n]
    return project_feasible(pattern, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent = np.array(initial_h_values, dtype=float).copy()
    else:
        parent = np.full(200, 0.5)
    pn = len(parent)

    # Phase 1: Nesterov from parent
    d1 = t0 + budget_s * 0.30
    h1, c1 = nesterov_optimize(parent.copy(), pn, d1, seed)
    if c1 < best_c5:
        best_h, best_c5, best_n = h1.copy(), c1, pn

    # Phase 2: Periodic starts + Nesterov
    d2 = t0 + budget_s * 0.50
    for p in [2, 3, 4, 5, 6, 7, 8, 10, 12, 15, 20, 25, 35]:
        if time.time() > d2:
            break
        h0 = make_periodic(pn, p, seed + p * 137)
        c0 = compute_c5(h0)
        if c0 < best_c5 + 0.10:
            h2, c2 = nesterov_optimize(h0, pn, d2, seed + p * 271)
            if c2 < best_c5:
                best_h, best_c5, best_n = h2.copy(), c2, pn

    # Phase 3: Coordinate descent polish
    d3 = t0 + budget_s * 0.65
    if best_h is not None and time.time() < d3:
        h3, c3 = coord_descent_fast(best_h.copy(), best_n, d3, seed + 500)
        if c3 < best_c5:
            best_h, best_c5, best_n = h3.copy(), c3, best_n

    # Phase 4: SLSQP with increasing smoothness
    d4 = t0 + budget_s * 0.88
    if best_h is not None and time.time() < d4:
        n = best_n
        dx = 2.0 / n
        ts = n / 2.0
        bounds = Bounds(np.zeros(n), np.ones(n))
        cons = [{'type': 'eq', 'fun': con_eq, 'jac': con_eq_jac, 'args': (ts,)}]
        for beta in [500, 5000, 50000, 500000]:
            if time.time() > d4:
                break
            x0 = project_feasible(best_h + np.random.randn(n) * 0.002, ts)
            try:
                res = minimize(smooth_max_obj, x0, args=(dx, beta), method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 8000, 'ftol': 1e-15})
                h_new = project_feasible(res.x, ts)
                c_new = compute_c5(h_new)
                if c_new < best_c5:
                    best_h, best_c5 = h_new, c_new
            except Exception:
                pass
        if time.time() < d4:
            try:
                res = minimize(smooth_max_obj, best_h, args=(dx, 1000000), method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 15000, 'ftol': 1e-16})
                h_new = project_feasible(res.x, ts)
                c_new = compute_c5(h_new)
                if c_new < best_c5:
                    best_h, best_c5 = h_new, c_new
            except Exception:
                pass

    # Phase 5: Final coordinate descent + Nesterov
    d5 = t0 + budget_s * 0.98
    if best_h is not None and time.time() < d5:
        h5, c5v = coord_descent_fast(best_h.copy(), best_n, d5, seed + 999)
        if c5v < best_c5:
            best_h, best_c5 = h5, c5v
        if time.time() < d5:
            h5b, c5v2 = nesterov_optimize(best_h.copy(), best_n, d5, seed + 1001,
                                           target_init=best_c5 * 0.9998)
            if c5v2 < best_c5:
                best_h, best_c5 = h5b, c5v2

    if best_h is None:
        best_h = np.full(100, 0.5)
        best_n = 100
    best_h = project_feasible(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
