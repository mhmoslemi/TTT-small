import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_c5_full(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def subgradient_step(h, n, lr, active_thresh=0.0):
    """One projected subgradient step."""
    c = np.correlate(h, 1 - h, mode='full')
    max_c = c.max()
    
    # Active set: lags within active_thresh of max
    active = c >= max_c - active_thresh
    
    if not active.any():
        active = np.zeros(len(c), dtype=bool)
        active[np.argmax(c)] = True
    
    # Compute subgradient for active lags (averaged)
    g = np.zeros(n)
    n_active = active.sum()
    
    for k in np.where(active)[0]:
        off = k - (n - 1)
        # ∂c[k]/∂h[j] = (1-h[j+off]) * 1[0<=j+off<=n-1] - h[j-off] * 1[0<=j-off<=n-1]
        if off >= 0:
            # j+off valid when j in [0, n-1-off]
            g[:n-off] += (1 - h[off:n])
            # j-off valid when j in [off, n-1]
            g[off:] -= h[:n-off]
        else:
            o = -off
            # j+off = j-o valid when j in [o, n-1]
            g[o:] += (1 - h[:n-o])
            # j-off = j+o valid when j in [0, n-1-o]
            g[:n-o] -= h[o:n]
    
    g /= n_active
    
    h_new = h - lr * g
    return project_feasible(h_new, n)

def subgradient_optimize(h, n, deadline, max_iter=100000):
    """Project subgradient with Polyak step size and iterate averaging."""
    dx = 2.0 / n
    target = 0.38  # optimistic target for Polyak step
    
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    # Compute initial upper bound
    upper = best_c
    
    # Iterate averaging buffer
    h_avg = h.copy()
    count = 0
    
    lr = 0.1
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        h = subgradient_step(h, n, lr, active_thresh=0.02)
        count += 1
        h_avg = ((count - 1) * h_avg + h) / count
        
        if it % 500 == 0:
            c = compute_c5(h, n)
            c_avg = compute_c5(h_avg, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
            if c_avg < best_c:
                best_c = c_avg
                best_h = h_avg.copy()
            
            # Polyak step size
            gap = compute_c5(h, n) - target
            if gap > 0.01:
                lr = min(0.5, gap / (1.0 + 0.1 * lr))
            elif gap < -0.01:
                lr = max(0.01, lr * 0.5)
    
    return best_h, best_c

```python
import cvxpy as cp

def slp_optimize(h, n, deadline, max_iter=100):
    """Successive Linear Programming for the Erdős problem."""
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        c = np.correlate(h, 1 - h, mode='full') * dx
        max_c = c.max()
        
        # Compute gradient of each c[k] with respect to h
        # c[k] = Σ_i h[i] * (1-h[i-k+n-1]) * dx
        # ∂c[k]/∂h[j] = dx * [(1-h[j-k+n-1]) if valid - h[j+k-(n-1)] if valid]
        
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n/2]
        obj = cp.Minimize(t_var)
        
        for k in range(len(c)):
            off = k - (n - 1)
            grad = np.zeros(n)
            if off >= 0:
                grad[:n-off] = (1 - h[off:n]) * dx
                grad[off:] -= h[:n-off] * dx
            else:
                o = -off
                grad[o:] = (1 - h[:n-o]) * dx
                grad[:n-o] -= h[o:n] * dx
            
            linearized = c[k] + grad @ (h_var - h)
            constraints.append(t_var >= linearized)
        
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.ECOS)  # or CBC
        except:
            break
        
        if h_var.value is None:
            break
        
        new_h = np.clip(h_var.value, 0, 1)
        new_h = project_feasible(new_h, n)
        new_c = compute_c5(new_h, n)
        
        if new_c < best_c:
            best_c = new_c
            best_h = new_h.copy()
        
        # Check convergence
        if abs(new_c - best_c) < 1e-10:
            break
        
        h = new_h
    
    return best_h, best_c

def compute_jacobian(h, n, dx, eps=1e-7):
    """Compute Jacobian of c = correlate(h, 1-h, 'full')*dx with respect to h."""
    n_k = 2*n - 1
    J = np.zeros((n_k, n))
    c0 = np.correlate(h, 1-h, mode='full') * dx
    for j in range(n):
        h_pert = h.copy()
        h_pert[j] += eps
        c_pert = np.correlate(h_pert, 1-h_pert, mode='full') * dx
        J[:, j] = (c_pert - c0) / eps
    return J
```

import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_c5_array(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def slp_optimize(h_init, n, deadline, max_slp_iters=50, damping=1.0):
    """Successive Linear Programming optimization."""
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    for it in range(max_slp_iters):
        if time.time() > deadline:
            break
        
        c = compute_c5_array(h, n)
        J = np.zeros((2*n-1, n))
        # Compute Jacobian numerically
        for j in range(n):
            h_p = h.copy()
            h_p[j] += 1e-8
            c_p = np.correlate(h_p, 1-h_p, mode='full') * dx
            J[:, j] = (c_p - c) / 1e-8
        
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n/2]
        for k in range(2*n-1):
            lin = c[k] + J[k] @ (h_var - h)
            constraints.append(t_var >= lin)
        
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            try:
                prob.solve(solver=cp.CBC)
            except Exception:
                break
        
        if h_var.value is None:
            break
        
        h_lp = np.clip(h_var.value, 0, 1)
        h_new = damping * h_lp + (1 - damping) * h
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        
        if abs(c_new - best_c) < 1e-12:
            break
        
        h = h_new
    
    return best_h, best_c

            if h.sum() < 1e-15: break
            h += (d / h.sum()) * h
    return np.clip(h, 0.0, 1.0)

def slp_optimize(h_init, n, deadline, max_iters=30, active_frac=0.3, damping=1.0):
    dx = 2.0 / n

    best_h = h.copy()

    h_var = cp.Variable(n)
    t_var = cp.Variable()

        if time.time() > deadline: break
        c_arr = np.correlate(h, 1-h, mode='full') * dx
        c_max = c_arr.max()
        
        # Active set: top fraction of lags
        thresh = c_max - (c_max - c_arr.min()) * active_frac
        active = np.where(c_arr >= thresh)[0]
        if len(active) < 5:
            # At least top 5
            top_k = np.argsort(c_arr)[-5:]
            active = top_k
        
        # Numerical Jacobian for active lags
        J_active = np.zeros((len(active), n))
        for col, j in enumerate(range(n)):
            hp = h.copy()
            hp[j] += 1e-8
            cp_arr = np.correlate(hp, 1-hp, mode='full') * dx
            J_active[:, col] = (cp_arr[active] - c_arr[active]) / 1e-8
        
        constraints = list(base_constraints)
        for idx, k in enumerate(active):
            lin = c_arr[k] + J_active[idx] @ (h_var - h)
            constraints.append(t_var >= lin)
        
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.ECOS)
        except:
            break
        if h_var.value is None: break
        
        h_new = damping * np.clip(h_var.value, 0, 1) + (1-damping) * h
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c

import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(8000):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            r = 1.0 - h
            if r.sum() < 1e-15:
                break
            h += (d / r.sum()) * r
        else:
            if h.sum() < 1e-15:
                break
            h += (d / h.sum()) * h
    return np.clip(h, 0.0, 1.0)

def slp_optimize(h_init, n, deadline, max_iters=40, active_frac=0.4):
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    base_cons = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2]
    prev_lp_val = None
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c_arr = np.correlate(h, 1 - h, mode='full') * dx
        c_max = c_arr.max()
        c_min = c_arr.min()
        span = c_max - c_min
        if span < 1e-12:
            break
        thresh = c_max - span * active_frac
        active = np.where(c_arr >= thresh)[0]
        if len(active) < 5:
            active = np.argsort(c_arr)[-8:]
        J_a = np.zeros((len(active), n))
        eps = 1e-8
        for j in range(n):
            hp = h.copy()
            hp[j] += eps
            cp_arr = np.correlate(hp, 1 - hp, mode='full') * dx
            J_a[:, j] = (cp_arr[active] - c_arr[active]) / eps
        constraints = list(base_cons)
        for idx, k in enumerate(active):
            lin = c_arr[k] + J_a[idx] @ (h_var - h)
            constraints.append(t_var >= lin)
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.CBC, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, verbose=False)
            except Exception:
                break
        if h_var.value is None:
            break
        h_lp = np.clip(h_var.value, 0.0, 1.0)
        h_new = project_feasible(h_lp, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
        if prev_lp_val is not None and np.max(np.abs(h_new - prev_lp_val)) < 1e-6:
            break
        h = h_new
        prev_lp_val = h_new.copy()
    return best_h, best_c

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 400:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def _sc(x, n):
    return float(np.sum(x) - n / 2.0)

def _scj(x, n):
    return np.ones(n)

def slsqp_run(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sc, 'jac': _scj, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def grad_informed_search(h, n, deadline, deltas=None):
    if deltas is None:
        deltas = [0.01, 0.005, 0.002, 0.001, 0.0005]
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(200):
            if time.time() > deadline:

            improved = False
            c_arr = np.correlate(h, 1 - h, mode='full') * dx
            c_max = c_arr.max()
            active = np.where(c_arr >= c_max - 0.02)[0]
            g = np.zeros(n)
            for k in active:
                hp = h.copy()
                eps = 1e-7
                for j in range(0, n, 4):
                    hp[j] += eps
                    cp = np.correlate(hp, 1 - hp, mode='full') * dx
                    g[j] += (cp[k] - c_arr[k]) / eps
                    hp[j] -= eps
            g /= max(1, len(active))
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                if abs(g[i]) < 1e-10:
                    continue
                for j in range(i + 1, n):
                    if abs(g[j]) < 1e-10:

                    if (g[i] > 0 and g[j] < 0) or (g[i] < 0 and g[j] > 0):
                        if g[i] > 0:
                            hi, hj = h[i] - d, h[j] + d
                        else:

                        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                            h[i], h[j] = hi, hj
                            c = compute_c5(h, n)
                            if c < best_c - 1e-15:

                                improved = True
                            else:

                                # restore
                                h[i] = hi + d if g[i] > 0 else hi - d
                                h[j] = hj - d if g[i] > 0 else hj + d
            if not improved:

    return h, best_c

def pair_search(h, n, deadline, deltas, radius=200):
    best_c = compute_c5(h, n)
    for d in deltas:
        if time.time() > deadline:
            break
        for _pass in range(500):
            if time.time() > deadline:
                return h, best_c
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                jlo = max(0, i - radius)
                jhi = min(n, i + radius + 1)
                for j in range(jlo, jhi):
                    if i == j:
                        continue
                    hi, hj = h[i] + d, h[j] - d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i], h[j] = hi - d, hj + d
            if not improved:
                break
    return h, best_c
