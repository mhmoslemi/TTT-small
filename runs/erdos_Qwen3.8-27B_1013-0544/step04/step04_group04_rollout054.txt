When h[i] changes from old to new (delta = new - old):

For the linear terms, the max is minimized at the point where the slopes of the active constraints straddle 0. This is equivalent to finding the point where the "upper envelope" of the lines has a flat point.

- The zero-lag term is t - t² (concave), added to a constant
- The other terms are linear: a_k * t + b_k
- f(t) = max(const₀ + t - t², max_k (a_k * t + b_k)) for t ∈ [0,1]

- The quadratic is max at t=1/2
- If the max of the linear terms at t=1/2 is less than the quadratic at t=1/2, the min is at t=1/2 (or where the linear catches up)
- Otherwise, the min is where the quadratic and the max linear term intersect

This can be solved in O(n) by:
1. Evaluating f at t=0, t=1, t=1/2, and at the intersection points
2. The intersection points are where const₀ + t - t² = a_k*t + b_k for some k

                corr = np.correlate(x, 1.0-x, mode="full")
                m = np.max(corr)

                penalty = mu * (x.sum() - target)**2

                        g[:n-m2] += wk*(1-x[m2:])

            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-12:
                h_cur = h_new
                c5_cur = c5_new
    return h_cur, c5_cur

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nf = free.sum()
        if nf == 0:
            break
        h[free] += d / nf
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj_grad(x, beta, mu, target, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    xs = np.clip(beta * (corr - m), -500, 0)
    ew = np.exp(xs)
    se = ew.sum()
    w = ew / se
    val = (m + np.log(se) / beta) * dx + mu * (x.sum() - target) ** 2
    # Gradient
    grad = np.zeros(n)
    active = w > 1e-12
    for k in np.where(active)[0]:
        wk = w[k] * dx
        lag = k - (n - 1)
        if lag == 0:
            grad += wk * (1.0 - 2.0 * x)
        elif 0 < lag < n:
            grad[lag:] += wk * (1.0 - x[:-lag])
            grad[:n - lag] += -wk * x[lag:]
        elif -n < lag < 0:
            m2 = -lag
            grad[:n - m2] += wk * (1.0 - x[m2:])
            grad[m2:] += -wk * x[:n - m2]
    grad += 2.0 * mu * (x.sum() - target)
    return val, grad

def lbfgs_opt(h0, betas, mus, maxiter_each, time_fn):
    n = len(h0)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    h_cur = h0.copy()
    c5_cur = compute_c5(h_cur)
    
    for mu in mus:
        for beta in betas:
            if time_fn() < 30:
                break
            def obj(x):
                v, _ = _smooth_obj_grad(x, beta, mu, target, n)
                return v
            def grad(x):
                _, g = _smooth_obj_grad(x, beta, mu, target, n)
                return g
            try:

                              jac=grad, options={'maxiter': maxiter_each,
                                                 'ftol': 1e-15, 'gtol': 1e-10})

                c5_new = compute_c5(h_new)

def perturb_and_refine(h, n_rounds, time_fn, seed=0):

    n = len(h)

            best = h_s.copy()
            best_c5 = c5_s
    return best, best_c5
