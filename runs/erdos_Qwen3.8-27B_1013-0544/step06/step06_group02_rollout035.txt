def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)

        if abs(diff) < 1e-14:

            if tot < 1e-14:

def lbfgs_phase(h, n, deadline, alpha=50000, mu=1.0):

        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()

        val += mu * (x.sum() - n/2.0) ** 2
        return float(val)

        g += 2.0 * mu * (x.sum() - n/2.0) * np.ones(n) / n * n

        g += 2.0 * mu * (x.sum() - n/2.0)
        return g

        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_c5(h, n, alpha):
    c = np.correlate(h, 1 - h, mode='full')
    cs = alpha * (c - c.max())
    w = np.exp(cs - cs.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-12:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            g[:nv] += wm * (1.0 - h[k:])
            g[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            g[kk:] += wm * (1.0 - h[:nv])
            g[:nv] += wm * (-h[kk:])
    return g

def smooth_obj(x, n, alpha, mu):
    c = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    if mu > 0:
        s = x.sum() - n / 2.0
        val += mu * s * s
    return float(val)

def smooth_grad(x, n, alpha, mu):
    g = grad_c5(x, n, alpha)
    if mu > 0:
        g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def adam_opt(h, n, deadline, alpha, lr):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    m1 = np.zeros(n)
    m2 = np.zeros(n)
    t = 0
    stall = 0
    b1, b2, eps = 0.85, 0.999, 1e-8
    while time.time() < deadline:
        t += 1
        g = grad_c5(h, n, alpha)
        m1 = b1 * m1 + (1 - b1) * g
        m2 = b2 * m2 + (1 - b2) * g * g
        c1 = 1 - b1 ** t
        c2 = 1 - b2 ** t
        h_new = project_feasible(h - lr * (m1 / c1) / (np.sqrt(m2 / c2) + eps), n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 3000:
                break
        h = h_new
    return best_h, best_c

def nesterov_anneal(h, n, deadline, alpha=1000, lr=0.002, beta=0.7):

    best_c = compute_c5(h, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_c5(h, n, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    cs = alpha * (c - c.max())
    w = np.exp(cs - cs.max())
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-12:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            g[:nv] += wm * (1.0 - h[k:])
            g[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            g[kk:] += wm * (1.0 - h[:nv])
            g[:nv] += wm * (-h[kk:])
    return g

def _smooth_obj(x, n, alpha, mu):
    c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    if mu > 0:
        s = x.sum() - n / 2.0
        val += mu * s * s
    return float(val)

def _smooth_jac(x, n, alpha, mu):
    g = grad_c5(x, n, alpha)
    if mu > 0:
        g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def adam_opt(h, n, deadline, alpha, lr):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    m1 = np.zeros(n)
    m2 = np.zeros(n)
    t = 0
    stall = 0
    b1, b2, eps = 0.85, 0.999, 1e-8
    while time.time() < deadline:
        t += 1
        g = grad_c5(h, n, alpha)
        m1 = b1 * m1 + (1 - b1) * g
        m2 = b2 * m2 + (1 - b2) * g * g
        c1 = 1 - b1 ** t
        c2 = 1 - b2 ** t
        h_new = project_feasible(h - lr * (m1 / c1) / (np.sqrt(m2 / c2) + eps), n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 3000:
                break
        h = h_new
    return best_h, best_c

def nesterov_anneal(h, n, deadline, alpha=1500, lr=0.003, beta=0.7):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    v = np.zeros(n)
    T = 0.015
    rng = np.random.RandomState(7)
    while time.time() < deadline:
        h_y = project_feasible(h + v, n)
        g = grad_c5(h_y, n, alpha)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g /= gn
        h_t = project_feasible(h - lr * g, n)
        c_t = compute_c5(h_t, n)
        c_cur = compute_c5(h, n)
        if c_t < best_c:
            v = beta * v + (h - h_t) / (1 + beta)
            h = h_t
            best_c = c_t
            best_h = h.copy()
        elif c_t < c_cur:
            v = beta * v + (h - h_t) / (1 + beta)
            h = h_t
        elif T > 0.001 and rng.random() < np.exp(-(c_t - best_c) / T):
            v = beta * v + (h - h_t) / (1 + beta)
            h = h_t
        else:
            v = beta * v
            lr *= 0.995
            if lr < 1e-7:
                break
        T *= 0.9997
    return best_h, best_c

def lbfgs_opt(h, n, deadline, alpha=100000, mu=0.5):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_smooth_obj, h, args=(n, alpha, mu),
                       jac=_smooth_jac, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 400, 'ftol': 1e-18, 'gtol': 1e-12})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return h.copy(), compute_c5(h, n)

def grad_pair(h, n, deadline, alpha=2000, top_k=60):
    best_c = compute_c5(h, n)
    for step in [0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005]:
        any_imp = False
        while time.time() < deadline:
            g = grad_c5(h, n, alpha)
            oi = np.argsort(g)[:top_k]
            od = np.argsort(-g)[:top_k]
            imp = False
            for i in oi:
                if time.time() > deadline:
                    return h, best_c
                if h[i] >= 1.0:
                    continue
                d = min(step, 1.0 - h[i])
                for j in od:
                    if i == j or h[j] < d:
                        continue
                    h[i] += d
                    h[j] -= d
                    c = compute_c5(h, n)
                    if c < best_c - 1e-14:
                        best_c = c
                        imp = True
                        any_imp = True
                        break
                    h[i] -= d
                    h[j] += d
                if imp:
                    break
            if not imp:
                break
        if not any_imp:
            break
    return h, best_c
