import numpy as np
from scipy.optimize import minimize
import time


def c5_vec(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)


def c5_val(h, n):
    return float(np.max(c5_vec(h, n)))


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    target = 0.5 * n
    h = np.clip(h, 0.0, 1.0)
    for _ in range(250):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0.0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        m = int(mask.sum())
        if m > 0:
            h[mask] += diff / float(m)
        else:
            h[0] = float(np.clip(h[0] + diff, 0.0, 1.0))
        h = np.clip(h, 0.0, 1.0)
    return h


def fallback_h(n=300):
    t = np.linspace(0.0, 2.0, n, endpoint=False)
    best = None
    best_c = 1e9
    triples = [

    for a1, a2, a3 in triples:

def smooth_max(c, alpha):
    m = float(c.max())
    sh = alpha * (c - m)
    return m + float(np.log(np.sum(np.exp(sh)))) / alpha


def lse_weights_full(c, alpha):
    m = float(c.max())
    sh = alpha * (c - m)
    w = np.exp(sh)
    s = float(w.sum())
    if s > 0.0:
        w /= s
    else:

        w[c == m] = 1.0 / max(1, int(np.sum(c == m)))

    off = n - 1
    for idx, wt in enumerate(w):
        if wt <= 1e-16:
            continue
        s = idx - off
        if s >= 0:
            m = n - s
            g[s:n] += wt * (1.0 - h[0:m])
            g[0:m] += -wt * h[s:n]
        else:
            s = -s
            m = n - s
            g[0:m] += wt * (1.0 - h[s:n])
            g[s:n] += -wt * h[0:m]
    return g


def grad_from_selected(h, n, idx, wsel):
    g = np.zeros(n, dtype=float)
    off = n - 1
    for k in range(len(idx)):
        wt = float(wsel[k])
        if wt <= 1e-16:
            continue
        s = int(idx[k]) - off
        if s >= 0:
            m = n - s
            g[s:n] += wt * (1.0 - h[0:m])
            g[0:m] += -wt * h[s:n]
        else:
            s = -s
            m = n - s
            g[0:m] += wt * (1.0 - h[s:n])
            g[s:n] += -wt * h[0:m]
    g -= g.mean()
    return g


def obj_lse_pen(x, n, alpha, mu):
    c = c5_vec(x, n)
    v = smooth_max(c, alpha)
    d = x.sum() - 0.5 * n
    return float(v + mu * d * d)


def jac_lse_pen(x, n, alpha, mu):
    c = c5_vec(x, n)
    w = lse_weights_full(c, alpha)
    g = grad_weights_full(x, n, w)
    d = float(x.sum() - 0.5 * n)
    if d != 0.0:
        g += 2.0 * mu * d
    return g


def lbfgs_optimize(h, n, deadline, max_iters=80):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = c5_val(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [
        (200.0, 1.0),
        (1000.0, 5.0),
        (5000.0, 20.0),
        (20000.0, 100.0),
        (100000.0, 500.0),
    ]
    for alpha, mu in configs:
        if time.time() > deadline - 5.0:
            break
        try:
            res = minimize(
                obj_lse_pen,
                h,
                jac=jac_lse_pen,
                args=(n, alpha, mu),
                method='L-BFGS-B',
                bounds=bounds,
                options={
                    'maxiter': max_iters,
                    'ftol': 1e-16,
                    'gtol': 1e-14,
                    'maxls': 30,
                }
            )
            hn = project_feasible(res.x, n)
            cn = c5_val(hn, n)
            if cn < best_c - 1e-15:
                best_c = cn
                h = hn
                best_h = hn.copy()
        except Exception:
            pass
    hn = project_feasible(best_h, n)
    cn = c5_val(hn, n)
    if cn <= best_c + 1e-12:
        return hn, cn
    return project_feasible(best_h, n), best_c


def pair_coeff(h, n, i, j):
    L = 2 * n - 1
    coeff = np.zeros(L, dtype=float)
    seg = 1.0 - h[::-1]
    coeff[i:i + n] += seg
    s = n - 1 - i
    coeff[s:s + n] -= h
    coeff[j:j + n] -= seg
    s2 = n - 1 - j
    coeff[s2:s2 + n] += h
    coeff *= (2.0 / n)
    return coeff


def best_delta_model(c, coeff, lo, hi):
    base = float(c.max())
    lo = float(lo)
    hi = float(hi)
    if hi - lo <= 1e-15:
        return 0.0, base

    a = lo
    b = hi
    found_delta = 0.0
    coeff_scale = float(np.max(np.abs(coeff))) if coeff.size else 1.0
    eps = 1e-12 + 1e-10 * max(1.0, coeff_scale)

    for _ in range(34):
        mid = 0.5 * (a + b)
        v = c + coeff * mid
        m = float(v.max())
        act = v >= (m - 1e-13 - 1e-12 * m)
        if not np.any(act):
            act = v >= (m - 1e-9)
        amin = float(coeff[act].min())
        amax = float(coeff[act].max())
        if amin <= eps and amax >= -eps:
            found_delta = mid
            break
        if amax <= eps:
            a = mid
        else:
            b = mid

    best_val = base
    best_delta = 0.0
    cand = (0.0, a, b, 0.5 * (a + b), found_delta)
    for d in cand:
        d = float(d)
        if d < lo - 1e-14 or d > hi + 1e-14:
            continue
        if d < lo:
            d = lo
        elif d > hi:
            d = hi
        val = float(np.max(c + coeff * d))
        if val < best_val:
            best_val = val
            best_delta = d
    return best_delta, best_val


def try_pair(h, n, i, j, c, C, best_c, max_step):
    if i == j:
        return False, None, None
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if max_step < 1e100:
        if lo < -max_step:
            lo = -max_step
        if hi > max_step:
            hi = max_step
    if hi - lo <= 1e-15:
        return False, None, None

    coeff = pair_coeff(h, n, i, j)
    delta, mval = best_delta_model(c, coeff, lo, hi)
    if mval >= C - 1e-13:
        return False, None, None

    h[i] += delta
    h[j] -= delta
    cnew = c5_vec(h, n)
    cn = float(cnew.max())
    if cn < best_c - 1e-14:
        return True, cnew, cn
    h[i] -= delta
    h[j] += delta

    fracs = (0.5, 0.25, 0.125, 0.0625)
    for f in fracs:
        d = delta * f
        if abs(d) < 1e-15:
            continue
        h[i] += d
        h[j] -= d
        cnew = c5_vec(h, n)
        cn = float(cnew.max())
        if cn < best_c - 1e-14:
            return True, cnew, cn
        h[i] -= d
        h[j] += d
    return False, None, None


def local_pair_search(h, n, deadline, max_step=0.1, tau=1e-5, top_k=80, rand_seed=1234):
    rng = np.random.RandomState(rand_seed)
    h = project_feasible(h, n).copy()
    c = c5_vec(h, n)
    C = float(c.max())
    best = C
    best_h = h.copy()
    stagnant = 0
    tau_cur = float(tau)
    top_k = int(min(top_k, 2 * n - 1))

    while time.time() < deadline:
        if float(h.min()) < -1e-9 or float(h.max()) > 1.0 + 1e-9:
            h = project_feasible(h, n)
            c = c5_vec(h, n)
            C = float(c.max())
            if C < best - 1e-15:
                best = C
                best_h = h.copy()

        C = float(c.max())
        if C < best - 1e-15:
            best = C
            best_h = h.copy()

        diff = C - c
        if top_k < c.size:
            idx = np.argpartition(diff, top_k - 1)[:top_k]
        else:
            idx = np.arange(c.size)

        wsel = np.exp(-diff[idx] / tau_cur)
        sw = float(wsel.sum())
        if sw <= 0.0:
            wsel = np.ones(len(idx), dtype=float)
            wsel /= len(idx)
        else:
            wsel /= sw

        g = grad_from_selected(h, n, idx, wsel)
        order_hi = np.argsort(g)[::-1]
        order_lo = np.argsort(g)
        klen = min(6, n)

        improved = False
        seen = set()
        for a in range(klen):
            i = int(order_hi[a])
            for b in range(klen):
                j = int(order_lo[b])
                if i == j:
                    continue
                if i < j:
                    aa, bb = i, j
                else:
                    aa, bb = j, i
                key = aa * n + bb
                if key in seen:
                    continue
                seen.add(key)
                ok, cnew, cn = try_pair(h, n, aa, bb, c, C, best, max_step)
                if ok:
                    c = cnew
                    C = cn
                    best = C
                    best_h = h.copy()
                    improved = True
                    break
            if improved:
                break

        if improved:
            stagnant = 0
            continue

        stagnant += 1
        random_count = 0
        if stagnant == 8:
            random_count = 60
        elif stagnant == 24:
            random_count = 150
        elif stagnant == 60:
            random_count = 350

        if random_count:
            for _ in range(random_count):
                if time.time() > deadline:
                    break
                i = int(rng.randint(n))
                j = int(rng.randint(n))
                if i == j:
                    continue
                if i < j:
                    aa, bb = i, j
                else:
                    aa, bb = j, i
                ok, cnew, cn = try_pair(h, n, aa, bb, c, C, best, max_step)
                if ok:
                    c = cnew
                    C = cn
                    best = C
                    best_h = h.copy()
                    improved = True
                    break
            if improved:
                stagnant = 0
                continue

        if stagnant == 30:
            tau_cur = min(tau_cur * 2.0, 1e-3)
        elif stagnant == 90:
            tau_cur = min(tau_cur * 3.0, 1e-3)
        if stagnant >= 160:
            break

    hf = project_feasible(best_h, n)
    cf = c5_val(hf, n)
    if cf > best + 1e-11:
        return best_h, best
    return hf, cf


def smooth_perturb(h, n, rng, amp):
    t = np.linspace(0.0, 2.0, n, endpoint=False)
    noise = np.zeros(n, dtype=float)
    for m in range(1, 7):
        noise += rng.randn() * np.cos(m * np.pi * t)
    h2 = h + amp * noise
    return project_feasible(h2, n)


def upsample_test(h, best_c, n, deadline):
    if time.time() > deadline - 130.0:
        return h, best_c, n
    n2 = 600 if n < 600 else 750
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
    h2 = np.interp(x_new, x_old, h)
    h2 = project_feasible(h2, n2)
    end = min(time.time() + 100.0, deadline - 20.0)
    h2, c2 = local_pair_search(h2, n2, end, max_step=0.05, tau=1e-5, top_k=90, rand_seed=4242)
    h2f = project_feasible(h2, n2)
    c2f = c5_val(h2f, n2)
    if c2f < best_c:
        return h2f, c2f, n2
    return h, best_c, n


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget_s = float(budget_s)
    deadline = t0 + max(60.0, budget_s) - 5.0

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = int(len(h0))
    except Exception:
        h0 = fallback_h(300)
        n0 = 300

    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_c = c5_val(h0, n0)

    hs = project_feasible(0.5 * (h0 + h0[::-1]), n0)
    cs = c5_val(hs, n0)
    if cs < best_c:
        best_h = hs
        best_c = cs

    if time.time() < deadline - 320.0:
        end = min(t0 + 180.0, deadline - 350.0)
        hc, cc = lbfgs_optimize(best_h, n0, end, max_iters=70)
        hf = project_feasible(hc, n0)
        cf = c5_val(hf, n0)
        if cf <= best_c + 1e-10:
            best_h, best_c = hf, cf

    steps = [0.5, 0.2, 0.1, 0.05, 0.02, 0.008, 0.003, 0.001]
    allocs = [80.0, 80.0, 80.0, 80.0, 80.0, 80.0, 80.0, 80.0]
    for idx, (ms, al) in enumerate(zip(steps, allocs)):
        if time.time() > deadline - 25.0:
            break
        end = min(time.time() + al, deadline - 20.0)
        hc, cc = local_pair_search(
            best_h,
            len(best_h),
            end,
            max_step=ms,
            tau=2e-5,
            top_k=90,
            rand_seed=1000 + idx
        )
        hf = project_feasible(hc, len(best_h))
        cf = c5_val(hf, len(best_h))
        if cf <= best_c + 1e-10:
            best_h, best_c = hf, cf

    hs = project_feasible(0.5 * (best_h + best_h[::-1]), len(best_h))
    cs = c5_val(hs, len(best_h))
    if cs < best_c + 1e-12 and time.time() < deadline - 70.0:
        best_h, best_c = hs, cs
        hc, cc = local_pair_search(
            best_h,
            len(best_h),
            min(time.time() + 60.0, deadline - 30.0),
            max_step=0.01,
            tau=5e-6,
            top_k=90,
            rand_seed=7777
        )
        hf = project_feasible(hc, len(best_h))
        cf = c5_val(hf, len(best_h))
        if cf <= best_c + 1e-10:
            best_h, best_c = hf, cf

    if time.time() < deadline - 110.0 and best_c > 0.38075:
        rng = np.random.RandomState(seed + 37)
        for amp in (0.02, 0.05, 0.08):
            if time.time() > deadline - 70.0:
                break
            hp = smooth_perturb(best_h, len(best_h), rng, amp)
            end = min(time.time() + 55.0, deadline - 45.0)
            hc, cc = local_pair_search(
                hp,
                len(hp),
                end,
                max_step=0.05,
                tau=1e-5,
                top_k=90,
                rand_seed=500 + int(1000 * amp)
            )
            hf = project_feasible(hc, len(hc))
            cf = c5_val(hf, len(hc))
            if cf < best_c - 1e-13:
                best_h, best_c = hf, cf

    if time.time() < deadline - 135.0:
        best_h, best_c, n0 = upsample_test(best_h, best_c, len(best_h), deadline)

    nf = int(len(best_h))
    hf = project_feasible(best_h, nf)
    cf = c5_val(hf, nf)
    if cf <= best_c + 1e-11:
        best_h, best_c = hf, cf

    nf = int(len(best_h))
    final_c = c5_val(best_h, nf)
    return best_h, float(final_c), nf
