def binary_pair_search(h, n, deadline, max_rounds=10):
    """Local search on near-binary solutions via pair flips."""
    h = h.copy()
    dx = 2.0 / n
    best_c = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)
    
    # Initial binarization: threshold at 0.5
    # But we need to maintain sum = n/2
    # Sort h and set top n/2 to 1, rest to 0? No, that changes the structure.
    # Instead, do soft pair flips.
    
    for round_idx in range(max_rounds):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        current_max = float(np.max(c))
        improved = False
        
        # Try pairs: one position to increase, one to decrease
        # Focus on positions near boundaries (0 or 1)
        lo_positions = np.where(h < 0.3)[0]
        hi_positions = np.where(h > 0.7)[0]
        
        rng = np.random.RandomState(round_idx * 12345 + 67890)
        rng.shuffle(lo_positions)
        rng.shuffle(hi_positions)
        
        n_lo = min(len(lo_positions), 50)
        n_hi = min(len(hi_positions), 50)
        
        for i_idx in range(n_lo):
            if time.time() > deadline:
                return h, best_c
            i = lo_positions[i_idx]
            for j_idx in range(n_hi):
                j = hi_positions[j_idx]
                if i == j:
                    continue
                # Flip: h[i] += 1-h[i], h[j] -= h[j] (make binary)
                di = 1.0 - h[i]
                dj = h[j]
                if abs(di - dj) > 0.01:
                    # Can't do a clean flip; try partial
                    d = min(di, dj) * 0.5
                else:
                    d = di  # Equal, clean binary flip
                
                h[i] += d
                h[j] -= d
                cval = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)
                if cval < best_c - 1e-14:
                    best_c = cval
                    improved = True
                    c = np.correlate(h, 1.0 - h, mode='full')
                    current_max = float(np.max(c))
                else:
                    h[i] -= d
                    h[j] += d
        
        if not improved:
            break
    
    return h, best_c

        improved = False
        for p in range(n):

            d_hi = 1.0 - h[p]
            if d_hi - d_lo < 1e-14:
                continue
            
            k_lo = max(0, min(p, n - 1 - p))
            k_hi = min(2 * n - 2, max(p + n - 1, 2 * n - 2 - p))
            
            um = 0.0
            if k_lo > 0:
                um = float(np.max(c[:k_lo]))
            if k_hi < 2 * n - 2:

            ks = np.arange(k_lo, k_hi + 1)
            idx_fwd = p + ks - (n - 1)
            idx_bwd = p - ks + (n - 1)
            mask_fwd = (idx_fwd >= 0) & (idx_fwd < n)
            mask_bwd = (idx_bwd >= 0) & (idx_bwd < n)
            s = np.zeros(len(ks))
            s[mask_fwd] += 1.0 - h[idx_fwd[mask_fwd]]
            s[mask_bwd] -= h[idx_bwd[mask_bwd]]
            
            # Find best d analytically (ignoring d² correction for now)
            # f(d) ≈ max(um, max_k(c[k] + d*s[k]))
            # Candidates: d_lo, d_hi, and d where c[k*] + d*s[k*] = um for the active k*
            
            candidates = [d_lo, d_hi, 0.0]
            
            # Find k with maximum c[k] + d*s[k] at d=0
            cur_vals = c[ks]
            k_active_idx = int(np.argmax(cur_vals))
            k_active = ks[k_active_idx]
            s_active = s[k_active_idx]
            
            if abs(s_active) > 1e-14:
                d_flat = (um - cur_vals[k_active_idx]) / s_active
                if d_lo <= d_flat <= d_hi:
                    candidates.append(d_flat)
            
            # Also check where the second-highest might become active
            # (skip for now, the 3-4 candidates should be sufficient)
            
            # Evaluate candidates
            best_d = 0.0
            best_val = current_max + penalty * (sumh - target) ** 2
            
            for d in candidates:
                if d < d_lo - 1e-14 or d > d_hi + 1e-14:
                    continue
                d = np.clip(d, d_lo, d_hi)
                if abs(d) < 1e-14:
                    continue
                new_c = c[ks] + d * s
                ci = (ks == (n - 1))
                if np.any(ci):
                    new_c[ci] += d * d
                val = max(um, float(np.max(new_c)))
                if penalty > 0:
                    val += penalty * (sumh + d - target) ** 2
                if val < best_val - 1e-14:
                    best_val = val
                    best_d = d
            
            if abs(best_d) > 1e-12:
                c[ks] += best_d * s
                ci = (ks == (n - 1))
                if np.any(ci):
                    c[n - 1] += best_d * best_d
                h[p] = np.clip(h[p] + best_d, 0.0, 1.0)
                sumh = float(h.sum())
                current_max = max(um, float(np.max(c[ks])))
                improved = True
        
        if not improved:
            break
    
    return h, current_max * dx

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h -= (-diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    sh = alpha * (c - m)
    mx = float(sh.max())
    if mx > 500:
        lse = mx + np.log(np.sum(np.exp(sh - mx)))
    else:
        lse = np.log(np.sum(np.exp(sh)))
    return float((m + lse / alpha) * dx)

def slsqp_optimize(h, n, alpha=500000, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda: np.ones(n)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def binary_pair_search(h, n, deadline, max_rounds=50):

    best_c = compute_c5(h, n)
    for rnd in range(max_rounds):

        c = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(c))
        improved = False
        lo_pos = np.where(h < 0.4)[0]
        hi_pos = np.where(h > 0.6)[0]
        rng = np.random.RandomState(rnd * 7919 + 31)
        rng.shuffle(lo_pos)
        rng.shuffle(hi_pos)
        n_lo = min(len(lo_pos), 60)
        n_hi = min(len(hi_pos), 60)
        for ii in range(n_lo):
            if time.time() > deadline:
                return h, best_c
            i = lo_pos[ii]
            for jj in range(n_hi):
                j = hi_pos[jj]
                if i == j:
                    continue
                d = min(1.0 - h[i], h[j]) * 0.8
                if d < 1e-4:
                    continue
                h[i] += d
                h[j] -= d
                cval = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
                if cval < cur_max - 1e-12:
                    improved = True
                    cur_max = cval
                    best_c = cur_max * dx
                    c = np.correlate(h, 1.0 - h, mode='full')
                else:
                    h[i] -= d
                    h[j] += d
        if not improved:
            break
    return h, best_c

def subgrad_squeeze(h, n, steps=5000, lr=0.005, deadline=None):
    h = h.copy()

    target = n / 2.0
    best_h = h.copy()

    for step in range(steps):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(c))
        cur_val = float(c[k_star]) * dx
        if cur_val < best_c:
            best_c = cur_val
            best_h = h.copy()
        idx1 = np.arange(n) + k_star - (n - 1)
        idx2 = np.arange(n) - k_star + (n - 1)
        m1 = (idx1 >= 0) & (idx1 < n)
        m2 = (idx2 >= 0) & (idx2 < n)

        if c_new < best_c:

            best_h = h_new.copy()
        h = h_new
