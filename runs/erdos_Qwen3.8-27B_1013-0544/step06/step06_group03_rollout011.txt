import time
import numpy as np
from scipy.optimize import minimize

GOLDEN = 0.6180339887498949

def project_feasible(x, n):
    x = np.clip(np.asarray(x, dtype=float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(250):
        d = target - float(x.sum())
        if abs(d) < 1e-13:
            break
        if d > 0.0:
            r = 1.0 - x
            s = float(r.sum())
            if s < 1e-15:
                break
            x += (d / s) * r
        else:
            s = float(x.sum())
            if s < 1e-15:
                break
            x += (d / s) * x
    d = target - float(x.sum())
    if abs(d) > 1e-15:
        if d > 0.0:
            f = 1.0 - x
            s = float(f.sum())
            if s > 1e-15:
                x += min(1.0, d / s) * f
        else:
            s = float(x.sum())
            if s > 1e-15:
                x += max(-1.0, d / s) * x
    return np.clip(x, 0.0, 1.0)

def compute_lags(h, n):
    P = np.zeros(n - 1)
    Q = np.zeros(n - 1)
    for m in range(1, n):
        L = n - m
        a = h[:L]
        b = h[m:m + L]
        ab = float(np.dot(a, b))
        P[m - 1] = float(a.sum()) - ab
        Q[m - 1] = float(b.sum()) - ab
    return P, Q, float(np.dot(h, 1.0 - h))

def make_pair_coeffs(h, i, j, n):
    CP = np.zeros(n - 1)
    CQ = np.zeros(n - 1)
    q = j - i - 1
    r = n - 1 - i
    if r:
        CP[:r] -= 1.0 - h[i + 1:i + 1 + r]
    r = n - 1 - j
    if r:
        CP[:r] += 1.0 - h[j + 1:j + 1 + r]
    if i:
        CP[:i] += h[:i][::-1]
    if j:
        CP[:j] -= h[:j][::-1]
    if i:
        CQ[:i] -= 1.0 - h[:i][::-1]
    if j:
        CQ[:j] += 1.0 - h[:j][::-1]
    r = n - 1 - i
    if r:
        CQ[:r] += h[i + 1:i + 1 + r]
    r = n - 1 - j
    if r:
        CQ[:r] -= h[j + 1:j + 1 + r]
    return CP, CQ, q, 2.0 * (h[i] - h[j])

def pair_value(d, P, Q, c0, CP, CQ, q, a0):
    v = c0 + a0 * d - 2.0 * d * d
    m = np.max(P + CP * d)
    u = P[q] + CP[q] * d + d * d
    if u > m:
        m = u
    if m > v:
        v = m
    m = np.max(Q + CQ * d)
    u = Q[q] + CQ[q] * d + d * d
    if u > m:
        m = u
    if m > v:
        v = m
    return float(v)

def golden_pair(P, Q, c0, CP, CQ, q, a0, aa, bb):
    a, b = aa, bb
    c = a + (1.0 - GOLDEN) * (b - a)
    d = a + GOLDEN * (b - a)
    fc = pair_value(c, P, Q, c0, CP, CQ, q, a0)
    fd = pair_value(d, P, Q, c0, CP, CQ, q, a0)
    for _ in range(16):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = a + (1.0 - GOLDEN) * (b - a)
            fc = pair_value(c, P, Q, c0, CP, CQ, q, a0)
        else:
            a = c
            c = d
            fc = fd
            d = a + GOLDEN * (b - a)
            fd = pair_value(d, P, Q, c0, CP, CQ, q, a0)
    return 0.5 * (a + b)

def line_search_pair(h, P, Q, c0, i, j, cnt):
    lo = -h[j]
    hi = h[i]
    if hi - lo < 1e-11:
        return 0.0, cnt, None
    CP, CQ, q, a0 = make_pair_coeffs(h, i, j, len(h))
    ng = 11
    bd = 0.0
    bv = float(cnt)
    for d in np.linspace(lo, hi, ng):
        if d == 0.0:
            continue
        v = pair_value(d, P, Q, c0, CP, CQ, q, a0)
        if v < bv - 1e-13:
            bv = v
            bd = d
    if bd != 0.0:
        step = (hi - lo) / (ng - 1)
        a = max(lo, bd - 1.5 * step)
        b = min(hi, bd + 1.5 * step)
        if b - a > 1e-12:
            d = golden_pair(P, Q, c0, CP, CQ, q, a0, a, b)
            v = pair_value(d, P, Q, c0, CP, CQ, q, a0)
            if v < bv:
                bd = d
                bv = v
    return bd, bv, (CP, CQ, q, a0)

def add_candidate(cand, seen, i, j):
    if i == j:
        return
    if i > j:
        i, j = j, i
    k = (i, j)
    if k not in seen:
        seen.add(k)
        cand.append(k)

def gen_candidates(h, n, P, Q, rng, nact=3, pa=4, nr=55):
    cand = []
    seen = set()
    for arr in (P, Q):
        for p in np.argsort(arr)[-nact:][::-1]:
            m = int(p) + 1
            L = n - m
            if L <= 2:
                continue
            o = np.argsort(np.abs(h[:L] - h[m:]))[-pa:]
            for ii in o:
                add_candidate(cand, seen, int(ii), int(ii) + m)

        i = int(rng.randint(n))
        j = int(rng.randint(n))
        add_candidate(cand, seen, i, j)
    return cand[:80]

def pair_descent(h, n, end, rng):
    h = project_feasible(h, n)
    P, Q, c0 = compute_lags(h, n)
    cnt = max(c0, float(np.max(P)), float(np.max(Q)))
    bh = h.copy()
    bc = float(cnt)
    moves = 0
    stagnant = 0
    while time.time() < end:
        if moves and moves % 30 == 0:
            P, Q, c0 = compute_lags(h, n)
            cnt = max(c0, float(np.max(P)), float(np.max(Q)))
            if cnt < bc:
                bc = cnt
                bh = h.copy()
        improved = False
        for i, j in gen_candidates(h, n, P, Q, rng):
            if time.time() > end:
                break
            d, nv, coef = line_search_pair(h, P, Q, c0, i, j, cnt)
            if coef is not None and abs(d) > 1e-12 and nv < cnt - 1e-11:
                CP, CQ, q, a0 = coef
                P += CP * d
                Q += CQ * d
                P[q] += d * d
                Q[q] += d * d
                c0 += a0 * d - 2.0 * d * d
                h[i] -= d
                h[j] += d
                h[i] = float(np.clip(h[i], 0.0, 1.0))
                h[j] = float(np.clip(h[j], 0.0, 1.0))
                cnt = float(nv)

        if not improved:

            if stagnant >= 3:
                break
        else:
            stagnant = 0
    bh = project_feasible(bh, n)
    P, Q, c0 = compute_lags(bh, n)

def binary_round(h, n):
    y = np.zeros(n)
    y[np.argsort(h)[-n // 2:]] = 1.0
    return y

def resample_h(h, n):
    h = np.asarray(h, dtype=float)
    if len(h) == n:
        return h.copy()
    x0 = np.linspace(0.0, 2.0, len(h), endpoint=False)
    x1 = np.linspace(0.0, 2.0, n, endpoint=False)
    return project_feasible(np.interp(x1, x0, h), n)

def get_active(P, Q, c0, K):
    half = max(1, K // 2)
    items = [(float(c0), 0, 0)]
    for idx in np.argsort(P)[-half:][::-1]:
        items.append((float(P[idx]), 1, int(idx) + 1))
    for idx in np.argsort(Q)[-half:][::-1]:
        items.append((float(Q[idx]), 2, int(idx) + 1))
    items.sort(reverse=True)
    return [(s, m) for _, s, m in items[:K]]

def _active_vals(x, n, active):
    vals = np.empty(len(active))
    c0 = float(np.dot(x, 1.0 - x))
    for k, sm in enumerate(active):
        s, m = sm
        if s == 0:
            vals[k] = c0
        else:
            L = n - m
            a = x[:L]
            b = x[m:m + L]
            ab = float(np.dot(a, b))
            vals[k] = float(a.sum() - ab if s == 1 else b.sum() - ab)
    return vals

def active_obj(x, n, active, alpha):
    vals = _active_vals(x, n, active)
    mx = float(vals.max())
    sh = alpha * (vals - mx)
    return float(mx + np.log(np.sum(np.exp(sh))) / alpha)

def active_jac(x, n, active, alpha):
    vals = _active_vals(x, n, active)
    mx = float(vals.max())
    w = np.exp(alpha * (vals - mx))
    w /= w.sum()
    g = np.zeros(n)
    for k, sm in enumerate(active):
        wk = float(w[k])
        if wk < 1e-15:
            continue
        s, m = sm
        if s == 0:
            g += wk * (1.0 - 2.0 * x)
        else:
            L = n - m
            a = x[:L]
            b = x[m:m + L]
            if s == 1:
                g[:L] += wk * (1.0 - b)
                g[m:] -= wk * a
            else:
                g[:L] -= wk * b
                g[m:] += wk * (1.0 - a)
    return g

def _sum_con(x, n):
    return float(x.sum() - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

    bounds = [(0.0, 1.0)] * n

                x = res.x

            else:

    if time.time() < deadline - 80.0:
        thr = binary_round(best_h, best_n)
        alloc = min(100.0, (deadline - time.time()) * 0.22)
        if alloc > 20.0:
            ht, ct = optimize_candidate(thr, best_n, time.time() + alloc, rng)
            cc = ct * 2.0 / best_n
            if cc < best_c:
                best_h = ht.copy()
                best_c = cc
    if time.time() < deadline - 80.0:
        hp = project_feasible(best_h + rng.normal(0.0, 0.12, best_n), best_n)
        alloc = min(90.0, (deadline - time.time()) * 0.22)
        if alloc > 20.0:
            ht, ct = optimize_candidate(hp, best_n, time.time() + alloc, rng)
            cc = ct * 2.0 / best_n
            if cc < best_c:
                best_h = ht.copy()
                best_c = cc
    for nt in (200, 400):
        if time.time() > deadline - 80.0:
            break
        alloc = min(110.0, (deadline - time.time()) * 0.28)
        if alloc < 30.0:
            break
        ht = resample_h(best_h, nt)
        ht, ct = optimize_candidate(ht, nt, time.time() + alloc, rng)
        cc = ct * 2.0 / nt
        if cc < best_c:
            best_h = ht.copy()
            best_c = cc
            best_n = nt
    if time.time() < deadline - 60.0:
        for per in (6, 12, 24):
            if time.time() > deadline - 50.0:
                break
            seq = ((np.arange(best_n) % per) < (per // 2)).astype(float)
            alloc = min(45.0, (deadline - time.time()) * 0.18)
            if alloc < 15.0:
                break
            ht, ct = optimize_candidate(project_feasible(seq, best_n), best_n, time.time() + alloc, rng)
            cc = ct * 2.0 / best_n
            if cc < best_c:
                best_h = ht.copy()
                best_c = cc
    if time.time() < deadline - 20.0:
        e = min(deadline - 3.0, time.time() + min(80.0, (deadline - time.time()) * 0.5))
        hp, ct = optimize_candidate(best_h, best_n, e, rng)
        cc = ct * 2.0 / best_n
        if cc < best_c:
            best_h = hp.copy()
            best_c = cc
    best_h = project_feasible(best_h, best_n)
    P, Q, c0 = compute_lags(best_h, best_n)
    return best_h, float(max(c0, np.max(P), np.max(Q)) * 2.0 / best_n), best_n
