import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def soft_max_val(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)

def grad_smooth_max(x, n, alpha):
    c = compute_c5_vec(x, n)
    sh = alpha * (c - c.max())
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - x[k:n])
            grad[k:n] += wm * (-x[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - x[0:nv])
            grad[0:nv] += wm * (-x[kk:n])
    return grad

def _obj(x, n, alpha, mu):
    return soft_max_val(x, n, alpha) + mu * (x.sum() - n / 2.0) ** 2

def _jac(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def _eq_con(x, n):
    return x.sum() - n / 2.0

def _eq_jac(x, n):
    return np.ones(n)

def _sls_obj(x, n, alpha):
    return soft_max_val(x, n, alpha)

def lbfgs_optimize(h, n, deadline, max_iters=300):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(500, 5.0), (2000, 10.0), (10000, 50.0), (50000, 200.0), (200000, 1000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 2:
            break
        try:
            res = minimize(_obj, h, jac=_jac, args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': max_iters, 'ftol': 1e-16, 'gtol': 1e-13})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def slsqp_optimize(h, n, deadline, alpha=500000, maxiter=1000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con, 'jac': _eq_jac, 'args': (n,)}]
    try:
        res = minimize(_sls_obj, h, jac=lambda x: grad_smooth_max(x, n, alpha),
                      args=(n, alpha), method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def nesterov_optimize(h, n, deadline, alpha=500, mu=50):
    h = project_feasible(h, n)
    v = np.zeros(n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    beta = 0.9
    lr = 0.05
    for it in range(20000):
        if time.time() > deadline:
            break
        g = grad_smooth_max(h, n, alpha)
        g += 2.0 * mu * (h.sum() - n / 2.0)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        g /= g_norm
        v = beta * v - lr * g
        h_new = project_feasible(h + v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            lr = min(0.1, lr * 1.01)
        else:
            lr *= 0.7
            v *= 0.5
            h = project_feasible(h, n)
        if lr < 1e-6:
            break
    return best_h, best_c
