def slsqp_phase(h, n, beta, deadline, maxsec):
    _BETA=float(beta); _N=int(n); _PEN=0.0
    target=n/2.0
    bounds=[(0.0,1.0)]*n
    cons=[{'type':'eq','fun':eq_fun,'jac':eq_jac}]
    h_cur = project_feasible(h,n)
    c_cur = compute_c5(h_cur,n)
    best_local=h_cur; c_best=c_cur
    maxiter = 80
    est = None
    while time.time() < deadline - max(5, 0.3*maxsec):
       start=time.time()
       try:
         res = minimize(smooth_obj, h_cur, method='SLSQP', bounds=bounds, constraints=cons, jac=smooth_jac,
                        options={'maxiter': maxiter, 'ftol': 1e-16, 'disp': False})
       except Exception:
         break
       elapsed=time.time()-start
       h_new=project_feasible(res.x,n)
       c_new=compute_c5(h_new,n)
       if c_new <= c_cur + 1e-12:
           h_cur=h_new; c_cur=c_new
       # update local best
       if c_new < c_best:
           c_best=c_new; best_local=h_new.copy()
       if res.nit < maxiter:
           break
       if est is None:
           est=elapsed/max(1,res.nit)
       else:
           est=0.7*est + 0.3*(elapsed/max(1,res.nit))
       # next maxiter to use about maxsec but not too large
       rem=deadline-time.time()
       if rem <= 10: break
       # choose target 0.8*maxsec capped rem
       target_time=min(maxsec*0.8, rem*0.8)
       maxiter_new = int(target_time / max(1e-6, est + 0.01))
       maxiter = max(25, min(250, maxiter_new))
       if maxiter_new < 25: break
    return best_local, c_best
```

def project_feasible(y,n):
    target = 0.5*float(n)
    x = np.asarray(y, dtype=float)
    if x.size == 0: return x
    x = np.clip(x, 0.0, 1.0)
    if abs(x.sum() - target) < 1e-14:
        return x
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = 0.5*(lo+hi)
        s = np.sum(np.clip(x + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    z = np.clip(x + 0.5*(lo+hi), 0.0, 1.0)
    for _ in range(20):
        diff = target - z.sum()
        if abs(diff) < 1e-13:
            break
        idx = np.where((z > 1e-12) & (z < 1.0 - 1e-12))[0]
        if idx.size == 0:
            z = z + diff/n
            z = np.clip(z, 0.0, 1.0)
            continue
        z[idx] = z[idx] + diff / float(idx.size)
        z = np.clip(z, 0.0, 1.0)
    z = np.clip(z, 0.0, 1.0)
    diff = target - z.sum()
    if diff != 0.0:
        z = z + diff / float(n)
        z = np.clip(z, 0.0, 1.0)
    return z

def fixed_jac(x):
    beta = float(_BETA)
    n = int(_N)
    c = np.correlate(x, 1.0 - x, mode='full')
    sel = np.take(c, _LIDX)
    sm = float(sel.max())
    sh = beta * (sel - sm)
    w = np.exp(sh)
    s = w.sum()
    if s <= 0.0:
        w = np.full(sel.shape, 1.0 / sel.size)
    else:
        w = w / s
    grad = np.zeros(n, dtype=float)
    center = n - 1
    for lag, wt in zip(_LIDX, w):
        if wt < 1e-15:
            continue
        m = int(lag) - center
        absm = abs(m)
        if absm >= n:
            continue
        size = n - absm
        if m >= 0:
            grad[m:m+size] += wt * (1.0 - x[0:size])
            grad[0:size] -= wt * x[m:m+size]
        else:
            p = -m
            grad[0:size] += wt * (1.0 - x[p:n])
            grad[p:n] -= wt * x[0:size]
    return grad
```

def perturb(h,n,rng,scale,steps=5):
    x=h.copy()
    for _ in range(int(rng.randint(1, steps+1))):
        i=int(rng.randint(n)); j=int(rng.randint(n))
        if i==j: continue
        pos = min(float(scale), float(x[i]), float(1.0 - x[j]))
        neg = min(float(scale), float(x[j]), float(1.0 - x[i]))
        if pos > 0.0:
            delta = float(rng.uniform(-neg, pos)) if (pos + neg) > 0 else 0.0
            x[i] -= delta
            x[j] += delta

        x = x + rng.normal(0.0, max(1e-6, scale*0.05), size=n)

def choose_lags(c, max_lags, window):
    m = float(np.max(c))
    cand = np.where(c >= m - float(window))[0]
    ml = int(max_lags)

def perturb(h,n,rng,scale,steps=5):
   x=h.copy()
   for _ in range(int(rng.randint(1, steps+1))):
      i=int(rng.randint(n)); j=int(rng.randint(n))

import numpy as np
import time
from scipy.optimize import minimize

_BETA = 1.0
_PEN = 0.0
_N = 1
_LIDX = np.arange(1)


def force_exact_sum(h, n):
    target = 0.5 * float(n)
    h = np.asarray(h, dtype=float)
    if not h.flags.writeable:
        h = h.copy()
    for _ in range(30):
        diff = target - float(np.sum(h))
        if abs(diff) < 1e-14:
            break
        idx = np.where((h > 1e-13) & (h < 1.0 - 1e-13))[0]
        if idx.size:
            h[idx] += diff / float(idx.size)
        else:
            h += diff / float(n)
        np.clip(h, 0.0, 1.0, out=h)
    return h


def project_feasible(y, n):
    target = 0.5 * float(n)
    x = np.asarray(y, dtype=float)
    if not x.flags.writeable:
        x = x.copy()
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.clip(x, 0.0, 1.0)
    if abs(float(np.sum(x)) - target) < 1e-14:
        return force_exact_sum(x, n)

    lo, hi = -1.0, 1.0
    for _ in range(45):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s > target:
            hi = mid
        else:
            lo = mid

    z = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(20):
        diff = target - float(np.sum(z))
        if abs(diff) < 1e-13:
            break
        idx = np.where((z > 1e-12) & (z < 1.0 - 1e-12))[0]
        if idx.size:
            z[idx] += diff / float(idx.size)
        else:
            z += diff / float(n)
        np.clip(z, 0.0, 1.0, out=z)
    return force_exact_sum(z, n)


def corr_counts(h):
    return np.correlate(h, 1.0 - h, mode="full")


def compute_c5(h, n):
    return float(np.max(corr_counts(h)) * (2.0 / float(n)))


def fallback_initial(n):
    i = np.arange(n)
    h = ((i % 4) < 2).astype(float)
    h += 0.08 * np.sin(2.0 * np.pi * i / 37.0)
    return project_feasible(h, n)


def choose_lags(c, max_lags, window):
    m = float(np.max(c))
    cand = np.where(c >= m - float(window))[0]
    ml = int(max_lags)
    if ml <= 0:
        ml = len(c)
    elif ml > len(c):
        ml = len(c)

    if cand.size > ml:
        cand = cand[np.argsort(c[cand])[-ml:]]
    else:
        top = np.argsort(c)[-ml:]
        cand = np.unique(np.concatenate((cand, top)))
        if cand.size > ml:
            cand = cand[np.argsort(c[cand])[-ml:]]
    return np.sort(np.asarray(cand, dtype=np.int64))


def fixed_obj(x):
    beta = float(_BETA)
    c = np.correlate(x, 1.0 - x, mode="full")
    sel = np.take(c, _LIDX)
    sm = float(np.max(sel))
    sh = beta * (sel - sm)
    return float((sm + np.log(np.exp(sh).sum())) / beta)


def fixed_jac(x):
    beta = float(_BETA)
    n = int(_N)
    c = np.correlate(x, 1.0 - x, mode="full")
    sel = np.take(c, _LIDX)
    sm = float(np.max(sel))
    sh = beta * (sel - sm)
    w = np.exp(sh)
    s = float(np.sum(w))
    if s <= 0.0:
        w = np.full(sel.shape, 1.0 / float(sel.size))
    else:
        w = w / s

    grad = np.zeros(n, dtype=float)
    center = n - 1
    for lag, wt in zip(_LIDX, w):
        if wt < 1e-15:
            continue
        m = int(lag) - center
        absm = abs(m)
        if absm >= n:
            continue
        size = n - absm
        if m >= 0:
            grad[m:m + size] += wt * (1.0 - x[0:size])
            grad[0:size] -= wt * x[m:m + size]
        else:
            p = -m
            grad[0:size] += wt * (1.0 - x[p:n])
            grad[p:n] -= wt * x[0:size]
    return grad


def pen_obj(x):
    return float(fixed_obj(x) + _PEN * (float(np.sum(x)) - 0.5 * float(_N)) ** 2)


def pen_jac(x):
    coeff = 2.0 * float(_PEN) * (float(np.sum(x)) - 0.5 * float(_N))
    return fixed_jac(x) + coeff


def eq_fun(x):
    return np.array([float(np.sum(x)) - 0.5 * float(_N)])


def eq_jac(x):
    return np.ones((1, int(_N)))


def next_chunk(elapsed, nit, rem, cap):
    if nit <= 0 or rem < 8.0:
        return 0
    est = max(1e-6, elapsed / float(nit))
    target = min(25.0, rem * 0.8)
    val = int(target / est)
    if val < 1:
        val = 1
    capi = int(cap)
    if val > capi:
        val = capi
    if val < 5 and rem > 60.0:
        val = min(capi, 5)
    return val


def lbfgs_active(h, n, beta, pen, cap, deadline, window, max_lags):
    global _BETA, _PEN, _N, _LIDX
    cur = project_feasible(h, n)
    ccur = compute_c5(cur, n)
    best = ccur
    best_h = cur.copy()

    _N = int(n)
    _BETA = float(beta)
    _PEN = float(pen)
    bounds = [(0.0, 1.0)] * n
    chunk = min(int(cap), 12)
    noimp = 0

    while time.time() < deadline - 4.0:
        if chunk <= 0:
            break
        cfull = corr_counts(cur)
        _LIDX = choose_lags(cfull, int(max_lags), float(window))
        start = time.time()
        try:
            res = minimize(
                pen_obj, cur,
                method="L-BFGS-B",
                bounds=bounds,
                jac=pen_jac,
                options={"maxiter": chunk, "ftol": 1e-16, "maxls": 40}
            )
        except Exception:
            break

        xnew = res.x
        if not np.all(np.isfinite(xnew)):
            break
        h2 = project_feasible(xnew, n)
        c2 = compute_c5(h2, n)

        improved = c2 < ccur - 1e-12
        if improved:
            cur = h2
            ccur = c2
        elif c2 <= ccur + 1e-10:
            cur = h2

        progress = improved
        if c2 < best:
            best = c2
            best_h = h2.copy()
            progress = True

        if progress:
            noimp = 0
        else:
            noimp += 1
            if noimp >= 3:
                break

        nit = int(getattr(res, "nit", 0) or 0)
        if nit < chunk:
            break
        chunk = next_chunk(time.time() - start, nit, deadline - time.time(), int(cap))

    return best_h, float(best)


def slsqp_active(h, n, beta, window, max_lags, cap, deadline):
    global _BETA, _PEN, _N, _LIDX

def perturb(h, n, rng, scale, steps=5):
    x = h.copy()
    for _ in range(int(rng.randint(1, steps + 1))):
        i = int(rng.randint(n))
        j = int(rng.randint(n))
        if i == j:
            continue
        pos = min(float(scale), float(x[i]), float(1.0 - x[j]))
        neg = min(float(scale), float(x[j]), float(1.0 - x[i]))
        if pos + neg > 0.0:
            delta = float(rng.uniform(-neg, pos))
            x[i] -= delta
            x[j] += delta
    if scale > 0.004:
        x += rng.normal(0.0, max(1e-6, 0.05 * float(scale)), size=n)
    return project_feasible(x, n)
