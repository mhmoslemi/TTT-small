def grad_from_weights(h,n,w):
    g=np.zeros(n); O=n-1; dx=2.0/n
    nz=np.nonzero(w > 1e-12)[0]
    for idx in nz:
      wt=float(w[idx]); tau=int(idx)-O
      if tau>=0:
        l=n-tau
        if l>0:
          g[:l] += wt*(1.0-h[tau:])
          g[tau:] -= wt*h[:l]
      else:
        t=-tau
        if t<n:
          g[t:] += wt*(1.0-h[:-t])
          g[:-t] -= wt*h[t:]
    return g*dx
```

def binary_delta_batch(x,pos,sign,lag_idx,n,O):
    pos=np.asarray(pos,int); tau=np.asarray(lag_idx,int)-O

import time
import numpy as np
from scipy.optimize import minimize


def box_sum_project(u, target):
    u = np.asarray(u, dtype=float)
    target = float(target)
    if u.size == 0:
        return u.copy()
    if np.all(u >= 0.0) and np.all(u <= 1.0) and abs(float(u.sum()) - target) < 1e-11:
        return u.copy()
    lo = float(np.min(u)) - 1.0
    hi = float(np.max(u)) + 1.0
    for _ in range(48):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(u - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(u - 0.5 * (lo + hi), 0.0, 1.0)


def c5_value(h, n):
    dx = 2.0 / float(n)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))


def c_array_ours(h, n):
    dx = 2.0 / float(n)
    return (np.correlate(h, 1.0 - h, mode="full") * dx)[::-1]


def binarize(h, n):
    target = int(n // 2)
    h = np.asarray(h, dtype=float)
    x = np.zeros(n, dtype=np.int8)
    if target > 0 and n:
        idx = np.argsort(h)[::-1]
        x[idx[:target]] = 1
    return x


def binary_c_int(x, n):
    xf = np.asarray(x, dtype=float)
    c = np.correlate(xf, 1.0 - xf, mode="full")[::-1]
    return np.rint(c).astype(np.int32)


def binary_delta_batch(x, pos_list, sign, lag_idx, n, O):
    pos = np.asarray(pos_list, dtype=np.int64)
    tau = np.asarray(lag_idx, dtype=np.int64) - O
    out = np.zeros((pos.size, tau.size), dtype=np.int32)
    if pos.size == 0 or tau.size == 0:
        return out
    for i, p in enumerate(pos):
        fv = (p + tau >= 0) & (p + tau < n)
        sv = (p - tau >= 0) & (p - tau < n)
        if sign > 0:
            if fv.any():
                out[i, fv] += (1 - x[p + tau[fv]]).astype(np.int32)
            if sv.any():
                out[i, sv] -= x[p - tau[sv]].astype(np.int32)
        else:
            if fv.any():
                out[i, fv] -= (1 - x[p + tau[fv]]).astype(np.int32)
            if sv.any():
                out[i, sv] += x[p - tau[sv]].astype(np.int32)
    return out


def full_single_delta(x, pos, sign, n, O):
    L = 2 * n - 1
    tau = np.arange(L, dtype=np.int64) - O
    d = np.zeros(L, dtype=np.int32)
    fv = (pos + tau >= 0) & (pos + tau < n)
    sv = (pos - tau >= 0) & (pos - tau < n)
    if sign > 0:
        if fv.any():
            d[fv] += (1 - x[pos + tau[fv]]).astype(np.int32)
        if sv.any():
            d[sv] -= x[pos - tau[sv]].astype(np.int32)
    else:
        if fv.any():
            d[fv] -= (1 - x[pos + tau[fv]]).astype(np.int32)
        if sv.any():
            d[sv] += x[pos - tau[sv]].astype(np.int32)
    return d


def binary_descend(x, c, deadline):
    x = np.asarray(x, dtype=np.int8).copy()
    c = np.asarray(c, dtype=np.int32).copy()
    n = x.size
    O = n - 1
    L = 2 * n - 1
    M = int(c.max())
    while time.time() < deadline:
        S = np.flatnonzero(c >= M - 2)
        if S.size == 0:
            break
        zeros = np.flatnonzero(x == 0)
        ones = np.flatnonzero(x == 1)
        if zeros.size == 0 or ones.size == 0:
            break
        dP = binary_delta_batch(x, zeros, 1, S, n, O)
        dQ = binary_delta_batch(x, ones, -1, S, n, O)
        col = np.full(L, -1, dtype=np.int64)
        col[S] = np.arange(S.size, dtype=np.int64)
        best_cost = M
        best_pi = -1
        best_qi = -1
        for pi, p in enumerate(zeros):
            exact = dP[pi][:, None] + dQ
            idx1 = ones - p + O
            idx2 = p - ones + O
            l1 = col[idx1]
            l2 = col[idx2]
            m1 = l1 >= 0
            if m1.any():
                exact[m1, l1[m1]] += 1
            m2 = l2 >= 0
            if m2.any():
                exact[m2, l2[m2]] += 1
            maxS = exact.max(axis=1)
            valid = maxS <= M - 1
            if valid.any():
                where = np.flatnonzero(valid)
                loc = where[np.argmin(maxS[where])]
                cost = int(maxS[loc])
                if cost < best_cost:
                    best_cost = cost
                    best_pi = pi
                    best_qi = int(loc)
        if best_pi < 0:
            break
        p = int(zeros[best_pi])
        q = int(ones[best_qi])
        d = full_single_delta(x, p, 1, n, O) + full_single_delta(x, q, -1, n, O)
        ab = (q - p) + O
        if 0 <= ab < L:
            d[ab] += 1
        ac = (p - q) + O
        if 0 <= ac < L:
            d[ac] += 1
        c += d
        x[p] = 1
        x[q] = 0
        newM = int(c.max())
        if newM >= M:
            c -= d
            x[p] = 0
            x[q] = 1
            break
        M = newM
    return x, c, M


def binary_multi_start(x0, n, deadline, seed):
    x0 = np.asarray(x0, dtype=np.int8)
    c0 = binary_c_int(x0, n)
    best_x = x0.copy()
    best_m = int(c0.max())
    rng = np.random.RandomState(seed)
    target = n // 2
    attempt = 0
    while time.time() < deadline and attempt < 100000:
        if attempt == 0:
            cur = x0.copy()
        elif attempt > 0 and rng.rand() < 0.07:
            cur = np.zeros(n, dtype=np.int8)
            idx = rng.choice(n, target, replace=False)
            cur[idx] = 1
        else:
            cur = best_x.copy()
            k = int(rng.randint(6, min(50, 8 + attempt // 2)))
            for _ in range(k):
                z = np.flatnonzero(cur == 0)
                o = np.flatnonzero(cur == 1)
                if z.size == 0 or o.size == 0:
                    break
                a = z[rng.randint(z.size)]
                b = o[rng.randint(o.size)]
                cur[a], cur[b] = 1, 0
        c = binary_c_int(cur, n)
        cur, c, M = binary_descend(cur, c, deadline)
        if M < best_m:
            best_m = M
            best_x = cur.copy()
        attempt += 1
    return best_x


def smooth_eval(h, n, alpha, K):
    c = c_array_ours(h, n)
    m = float(c.max())
    L = c.size
    if K <= 0 or K >= L:
        vals = c
        top = None
    else:
        top = np.argpartition(c, -K)[-K:]
        vals = c[top]
    sh = alpha * (vals - m)
    amax = float(sh.max())
    e = np.exp(sh - amax)
    s = float(e.sum())
    F = m + (amax + float(np.log(max(s, 1e-300)))) / alpha
    if top is None:
        w = e / s
    else:
        w = np.zeros(L, dtype=float)
        w[top] = e / s
    return c, m, F, w


def grad_from_weights(h, n, w):
    g = np.zeros(n, dtype=float)
    O = n - 1
    dx = 2.0 / float(n)
    nz = np.flatnonzero(w > 1e-12)
    for idx in nz:
        wt = float(w[idx])
        tau = int(idx) - O
        if tau >= 0:
            l = n - tau
            if l > 0:
                g[:l] += wt * (1.0 - h[tau:])
                g[tau:] -= wt * h[:l]
        else:
            t = -tau
            if t < n:
                g[t:] += wt * (1.0 - h[:-t])
                g[:-t] -= wt * h[t:]
    return g * dx


def smooth_refine(h, n, deadline):
    target = 0.5 * float(n)
    h = box_sum_project(h, target)
    best_h = h.copy()
    best_c = c5_value(h, n)
    stages = [
        (24, 160, 0.22, 220),
        (60, 110, 0.16, 280),
        (160, 65, 0.10, 260),
        (420, 30, 0.06, 220),
    ]
    for alpha, K, step0, iters in stages:
        if time.time() > deadline:
            break
        step = step0
        for _ in range(iters):
            if time.time() > deadline:
                break
            c, Ccur, F, w = smooth_eval(h, n, alpha, K)
            if Ccur < best_c - 1e-13:
                best_c = Ccur
                best_h = h.copy()
            g = grad_from_weights(h, n, w)
            gn = float(np.max(np.abs(g)))
            if gn < 1e-11:
                break
            g /= gn
            accepted = False
            for _ls in range(12):
                step -= 0.0
                hp = box_sum_project(h - step * g, target)
                cp, Fp, _, _ = smooth_eval(hp, n, alpha, K)
                Cp = float(cp.max())
                if Cp < Ccur - 1e-11 or (Fp < F - 1e-10 and Cp <= Ccur + 0.002):
                    h = hp
                    Ccur = Cp
                    F = Fp
                    accepted = True
                    if Cp < best_c - 1e-13:
                        best_c = Cp
                        best_h = h.copy()
                    break
                step *= 0.5
                if step < 1e-7:
                    break
            if not accepted:
                step *= 0.35
                if step < 1e-7:
                    break
    return best_h, best_c


def select_active(c, max_active, seed):
    C = float(c.max())
    near = np.flatnonzero(c >= C - 1e-5)
    top = np.argsort(c)[-max_active:]
    active = np.unique(np.concatenate((near, top)))
    if active.size > max_active:
        active = active[np.argsort(c[active])[-max_active:]]
    if seed is not None:
        rng = np.random.RandomState(int(seed) % (2**31 - 1))
        wide = np.argsort(c)[-max(2 * max_active, 40):]
        avail = np.setdiff1d(wide, active)
        if avail.size > 0 and active.size <= max_active + 8:
            add = rng.choice(avail, min(6, avail.size), replace=False)
            active = np.unique(np.concatenate((active, add)))
            if active.size > max_active:
                active = active[np.argsort(c[active])[-max_active:]]
    return active


def _active_cvals(h, n, active):
    dx = 2.0 / float(n)
    active = np.asarray(active, dtype=int)
    out = np.zeros(active.size, dtype=float)
    O = n - 1
    for r, idx in enumerate(active):
        tau = int(idx) - O
        if tau >= 0:
            if tau == 0:
                out[r] = float(np.dot(h, 1.0 - h))
            else:
                out[r] = float(np.dot(h[:n - tau], 1.0 - h[tau:]))
        else:
            t = -tau
            out[r] = float(np.dot(h[t:], 1.0 - h[:-t]))
    return out * dx


def _ajac_row(dc, n):
    dx = 2.0 / float(n)
    row = np.zeros(n + 1)
    row[:n] = -dc * dx
    row[n] = 1.0
    return row


def _slsqp_f(z, n):
    return float(z[n])


def _slsqp_fj(z, n):
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g


def _sum_eq(z, n):
    return float(z[:n].sum() - 0.5 * float(n))


def _sum_eq_jac(z, n):
    g = np.zeros(n + 1)
    g[:n] = 1.0
    return g


def _slsqp_ineq(z, n, active):
    h = z[:n]
    t = float(z[n])
    return t - _active_cvals(h, n, active)


def _slsqp_ineq_jac(z, n, active):
    h = z[:n]
    active = np.asarray(active, dtype=int)
    J = np.zeros((active.size, n + 1))
    O = n - 1
    dx = 2.0 / float(n)
    for r, idx in enumerate(active):
        tau = int(idx) - O
        dc = np.zeros(n)
        if tau >= 0:
            l = n - tau
            if l > 0:
                dc[:l] += 1.0 - h[tau:]
                dc[tau:] -= h[:l]
        else:
            t = -tau
            if t < n:
                dc[t:] += 1.0 - h[:-t]
                dc[:-t] -= h[t:]
        J[r, :n] = -dc * dx
        J[r, n] = 1.0
    return J


def slsqp_active_polish(h, n, deadline, seed, max_round=42):
    target = 0.5 * float(n)
    h = box_sum_project(h, target)
    best_h = h.copy()
    best_c = c5_value(h, n)
    rng = np.random.RandomState(seed)
    rounds = 0
    no_improve = 0
    while time.time() < deadline and rounds < max_round:
        if deadline - time.time() < 25:
            break
        c = c_array_ours(h, n)
        C = float(c.max())
        if C < best_c - 1e-13:
            best_c = C
            best_h = h.copy()
            no_improve = 0
        active = select_active(c, min(22, max(8, n // 12)), int(rng.randint(1, 2**31 - 1)))
        if active.size == 0:
            break
        t0 = C + 1e-7
        x0 = np.empty(n + 1)
        x0[:n] = h
        x0[n] = t0
        bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
        cons = [
            {"type": "eq", "fun": _sum_eq, "jac": _sum_eq_jac, "args": (n,)},
            {"type": "ineq", "fun": _slsqp_ineq, "jac": _slsqp_ineq_jac, "args": (n, active)},
        ]
        try:
            res = minimize(
                _slsqp_f,
                x0,
                args=(n,),
                jac=_slsqp_fj,
                method="SLSQP",
                bounds=bounds,
                constraints=cons,
                options={"maxiter": 55, "ftol": 1e-9, "disp": False},
            )
            if np.all(np.isfinite(res.x)):
                h2 = box_sum_project(res.x[:n], target)
                C2 = c5_value(h2, n)
                if C2 < best_c - 1e-13:
                    best_c = C2
                    best_h = h2.copy()
                    h = h2.copy()
                    no_improve = 0
                elif C2 <= C + 1e-5:
                    h = h2.copy()
                    no_improve += 1
                else:
                    no_improve += 1
        except Exception:
            no_improve += 1
        rounds += 1
        if rounds % 4 == 0:
            h = box_sum_project(h + 0.015 * rng.normal(size=n), target)
        if no_improve >= 8:
            break
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = max(60.0, float(budget_s))
    deadline = t0 + total - 8.0

    try:
        arr = np.asarray(initial_h_values, dtype=float).copy()
        n0 = int(arr.size)
    except Exception:
        n0 = 250
        arr = np.full(n0, 0.5, dtype=float)

    if n0 < 40 or n0 > 900 or (n0 % 2) != 0:
        old_pos = (np.arange(n0, dtype=float) + 0.5) * (2.0 / max(1, n0))
        new_pos = (np.arange(250, dtype=float) + 0.5) * (2.0 / 250.0)
        arr = np.interp(new_pos, old_pos, arr)
        n0 = 250

    n = n0
    target = 0.5 * float(n)

    h = box_sum_project(arr, target)
    best_h = h.copy()
    best_c = c5_value(h, n)

    hs = box_sum_project(0.5 * (h + h[::-1]), target)
    cs = c5_value(hs, n)
    if cs < best_c - 1e-13:
        best_h = hs.copy()
        best_c = cs
        h = hs.copy()

    h_orig = best_h.copy()

    d_bin1 = t0 + 0.30 * total
    if time.time() < d_bin1:
        x0 = binarize(h_orig, n)
        xb = binary_multi_start(x0, n, d_bin1, seed)
        cb = c5_value(xb.astype(float), n)
        if cb < best_c - 1e-13:
            best_c = cb
            best_h = xb.astype(float).copy()

    d_con = t0 + 0.62 * total
    if time.time() < d_con:
        h1, c1 = smooth_refine(best_h, n, d_con - 20)
        if c1 < best_c - 1e-13:
            best_c = c1
            best_h = h1.copy()
        if time.time() < d_con:
            h2, c2 = slsqp_active_polish(best_h, n, d_con, seed + 1)
            if c2 < best_c - 1e-13:
                best_c = c2
                best_h = h2.copy()
        if time.time() < d_con and c1 < best_c + 0.004:
            h3, c3 = slsqp_active_polish(h1, n, d_con, seed + 2)
            if c3 < best_c - 1e-13:
                best_c = c3
                best_h = h3.copy()

    d_bin2 = t0 + 0.82 * total
    if time.time() < d_bin2:
        x1 = binarize(best_h, n)
        xb = binary_multi_start(x1, n, d_bin2, seed + 101)
        cb = c5_value(xb.astype(float), n)
        if cb < best_c - 1e-13:
            best_c = cb
            best_h = xb.astype(float).copy()

    if time.time() < deadline - 5:
        h4, c4 = slsqp_active_polish(best_h, n, deadline - 4, seed + 300)
        if c4 < best_c - 1e-13:
            best_c = c4
            best_h = h4.copy()

    if time.time() < deadline - 2:
        h5, c5 = smooth_refine(best_h, n, deadline - 1)
        if c5 < best_c - 1e-13:
            best_c = c5
            best_h = h5.copy()

    hs = box_sum_project(0.5 * (best_h + best_h[::-1]), target)
    cs = c5_value(hs, n)
    if cs < best_c - 1e-13:
        best_c = cs
        best_h = hs.copy()

    best_h = box_sum_project(best_h, target)
    best_c = c5_value(best_h, n)
    return best_h, float(best_c), int(n)
