- For j >= lag': h[j] appears in h[j]*(1-h[j-lag']), contributing dx*(1-h[j-lag'])

So for lag < 0 (lag' = -lag):
∂f_lag/∂h[j] = -dx * h[j+lag'] * [j < n-lag'] + dx * (1-h[j-lag']) * [j >= lag']

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _smooth_max(vals, alpha):
    m = vals.max()
    shifted = alpha * (vals - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _full_obj(x, n, alpha):
    c = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    return _smooth_max(c, alpha)

def _active_obj(x, n, idx, alpha):
    c = np.correlate(x, 1 - x, mode='full') * (2.0 / n)
    return _smooth_max(c[idx], alpha)

def _sum_c(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_jac(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, idx=None, maxiter=5000, ftol=1e-16):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_c, 'jac': _sum_jac, 'args': (n,)}]
    if idx is not None:
        res = minimize(_active_obj, h, args=(n, idx, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    else:
        res = minimize(_full_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    return project_feasible(res.x, n)

def lag_grad(h, n, lag):
    grad = np.zeros(n)
    dx = 2.0 / n
    m = abs(lag)
    if m >= n:
        return grad
    if m == 0:
        return dx * (1.0 - 2.0 * h)
    if lag > 0:
        grad[:n - m] += (1.0 - h[m:])
        grad[m:] -= h[:n - m]
    else:
        grad[:n - m] -= h[m:]
        grad[m:] += (1.0 - h[:n - m])
    return dx * grad

def grad_local_search(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-05, 2e-05, 1e-05]
    for delta in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            c = compute_all_lags(h, n)
            k_star = int(np.argmax(c))
            lag = k_star - (n - 1)
            if lag == 0 or abs(lag) >= n:
                break
            g = lag_grad(h, n, lag)
            src = np.argsort(g)[-5:][::-1]
            tgt = np.argsort(g)[:5]
            best_gain = 0.0
            best_mv = None
            for s in src:
                for t in tgt:
                    if s == t:
                        continue
                    d = min(delta, h[s] * 0.4, 1.0 - h[t])
                    if d < 1e-12:
                        continue
                    h[s] -= d
                    h[t] += d
                    cn = compute_c5(h, n)
                    if best_c - cn > best_gain:
                        best_gain = best_c - cn
                        best_mv = (s, t, d)
                    h[s] += d
                    h[t] -= d
            if best_mv is not None and best_gain > 1e-16:
                s, t, d = best_mv
                h[s] -= d
                h[t] += d
                h = project_feasible(h, n)
                best_c = compute_c5(h, n)
                improved = True
    return h, best_c

def opt_n(h0, n, deadline):
    h = project_feasible(h0.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
