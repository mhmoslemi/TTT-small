import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * room, 0.0, 1.0)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * h, 0.0, 1.0)
    return h

def project_bisect(h, n):
    """Project onto {0<=x<=1, sum(x)=n/2} using bisection."""
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        projected = np.clip(h - mid, 0.0, 1.0)
        s = projected.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    projected = np.clip(h - (lo + hi) / 2, 0.0, 1.0)
    return projected

def compute_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def subgradient_method(h, n, deadline, max_iter=200000):
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    best_corr = compute_corr(h, n).max()
    
    step = 0.05
    momentum = np.zeros(n)
    mu = 0.85
    active_tol = 1.0  # lags within this of max are "active"
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        corr = compute_corr(h, n)
        max_corr = corr.max()
        
        if max_corr < best_corr:
            best_corr = max_corr
            best_c = max_corr * dx
            best_h = h.copy()
        
        # Active set
        active = np.where(corr >= max_corr - active_tol)[0]
        
        # Compute subgradient
        grad = np.zeros(n)
        m = np.arange(n)
        for k_idx in active:
            t = k_idx - (n - 1)
            mt = m + t
            mtt = m - t
            vt = (mt >= 0) & (mt < n)
            vtt = (mtt >= 0) & (mtt < n)
            grad[vt] += (1.0 - h[mt[vt]])
            grad[vtt] -= h[mtt[vtt]]
        grad /= len(active)
        
        # Momentum step
        momentum = mu * momentum - step * grad
        h_new = h + momentum
        
        # Project
        h_new = project_bisect(h_new, n)
        
        # Check improvement
        new_corr_max = compute_corr(h_new, n).max()
        if new_corr_max < max_corr * (1 - 1e-4):

        if it % 1000 == 0:
            active_tol = max(0.5, max_corr * 0.01)

def binary_swap_search(h, n, deadline, max_iters=50000):

    best_corr = compute_corr(h, n).max()
    best_h = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]

        corr = compute_corr(h, n)
        max_corr = corr.max()

        t = worst_lag - (n - 1)

                        h[b] = 1.0

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_bisect(h, n):
    target = n / 2.0
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            h = np.clip(h + (diff / s) * room, 0.0, 1.0) if s > 1e-14 else h
        else:
            s = h.sum()
            h = np.clip(h + (diff / s) * h, 0.0, 1.0) if s > 1e-14 else h
    return np.clip(h, 0.0, 1.0)

def get_subgradient(h, n, corr, top_k=20, sharpness=100):
    max_val = corr.max()
    # Get top-k indices
    top_idx = np.argsort(corr)[-top_k:]
    # Exponential weights
    w = np.exp(sharpness * (corr[top_idx] - max_val))
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(top_k) / top_k
    else:
        w /= w_sum
    
    grad = np.zeros(n)
    m = np.arange(n)
    for idx in range(top_k):
        k_idx = top_idx[idx]
        t = k_idx - (n - 1)
        mt = m + t
        mtt = m - t
        vt = (mt >= 0) & (mt < n)
        vtt = (mtt >= 0) & (mtt < n)
        wi = w[idx]
        if vt.any():
            grad[vt] += wi * (1.0 - h[mt[vt]])
        if vtt.any():
            grad[vtt] -= wi * h[mtt[vtt]]
    return grad

def projected_subgradient(h, n, deadline, max_iter=200000, top_k=20):
    dx = 2.0 / n
    best_h = h.copy()
    best_max = np.correlate(h, 1.0-h, mode='full').max()
    
    lr = 0.1
    mu = 0.85
    momentum = np.zeros(n)
    stagnation = 0
    best_stag_max = best_max

            break
        
        corr = np.correlate(h, 1.0-h, mode='full').astype(np.float64)
        max_val = corr.max()

            best_h = h.copy()
            stagnation = 0
            best_stag_max = max_val
        else:
            stagnation += 1

            momentum = np.zeros(n)
            rng = np.random.RandomState(it)
            h = project_bisect(h + 0.05 * (rng.rand(n) - 0.5), n)
            stagnation = 0
            continue

        momentum = mu * momentum - lr * grad

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def obj_grad(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        e = np.exp(sh - sm)
        w = e / e.sum()
        # Gradient of log-sum-exp part
        grad_lse = np.zeros(n)
        idx = np.where(w > 1e-10)[0]
        mm = np.arange(n)
        for k_idx in idx:
            t = k_idx - (n-1)
            mt = mm + t
            mtt = mm - t
            vt = (mt >= 0) & (mt < n)
            vtt = (mtt >= 0) & (mtt < n)
            grad_lse[vt] += w[k_idx] * (1.0 - x[mt[vt]])
            grad_lse[vtt] -= w[k_idx] * x[mtt[vtt]]
        # Gradient of the max part (approximate by the same)
        return dx * (grad_lse + w[np.argmax(g)] * get_subgradient_at(x, n, k_idx=np.argmax(g)))
    # Skip gradient for SLSQP - let it use numerical
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0}]
    res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-18})
    return project_feasible(res.x, n)

def get_subgradient_at(h, n, k_idx, sharpness=100):
    m = np.arange(n)
    t = k_idx - (n - 1)
    mt = m + t
    mtt = m - t
    vt = (mt >= 0) & (mt < n)
    vtt = (mtt >= 0) & (mtt < n)
    g = np.zeros(n)
    g[vt] += (1.0 - h[mt[vt]])
    g[vtt] -= h[mtt[vtt]]
    return g

def binary_targeted_search(h, n, deadline, max_sweeps=5000):
    dx = 2.0 / n
    h = h.copy()
    corr = np.correlate(h, 1.0-h, mode='full').astype(np.float64)
    max_corr = corr.max()
    best_corr = max_corr
    best_h = h.copy()
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        top_lags = np.argsort(corr)[-3:]
        
        for a in zeros:
            if time.time() > deadline:
                break
            if h[a] != 0:
                continue
            best_b = -1
            best_new_max = max_corr + 1  # must improve
            
            for b in ones:
                if h[b] != 1:
                    continue
                # Compute new max considering all top lags
                new_max = 0
                for k_idx in top_lags:
                    t = k_idx - (n-1)
                    # Effect of adding a
                    da = 0.0
                    if 0 <= a+t < n:
                        da += (1.0 - h[a+t])
                    if 0 <= a-t < n:
                        da -= h[a-t]
                    # Effect of removing b
                    db = 0.0
                    if 0 <= b+t < n:
                        db -= (1.0 - h[b+t])
                    if 0 <= b-t < n:
                        db += h[b-t]
                    new_val = corr[k_idx] + da + db
                    if new_val > new_max:
                        new_max = new_val
                    if new_val >= max_corr:
                        break
                else:
                    if new_max < best_new_max:
                        best_new_max = new_max
                        best_b = b
            
            if best_b >= 0:
                h[a] = 1.0
                h[best_b] = 0.0
                corr = np.correlate(h, 1.0-h, mode='full').astype(np.float64)
                max_corr = corr.max()
                if max_corr < best_corr:
                    best_corr = max_corr
                    best_h = h.copy()
                improved = True
                break
        
        if not improved:
            # Random perturbation
            rng = np.random.RandomState(sweep)
            a = rng.choice(zeros)
            b = rng.choice(ones)
            if h[a] == 0 and h[b] == 1:
                h[a] = 1.0
                h[b] = 0.0
                corr = np.correlate(h, 1.0-h, mode='full').astype(np.float64)
                max_corr = corr.max()
    
    return best_h, float(best_corr * dx)
