        ones = np.append(ones, j)

import numpy as np

import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    """Project h to satisfy 0<=h<=1 and sum(h)=n/2."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * room, 0.0, 1.0)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * h, 0.0, 1.0)
    return h

def slsqp_smooth_obj(x, n, alpha):
    """Smooth max objective using log-sum-exp."""
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    # Use log-sum-exp for numerical stability
    max_c = c.max()
    log_sum = max_c + np.log(np.exp(c - max_c).sum()) / alpha
    return log_sum

def slsqp_optimize(h, n, deadline, maxiter=5000, alpha=500):

            # Let's use a different approach
        return None  # Let SLSQP use numerical gradients

    best = h_start.copy()
