```python
def swap_delta_fast(h, i, j, n):
    """Fast computation of delta in correlate when h[i]:0->1, h[j]:1->0"""
    delta = np.zeros(2 * n - 1)
    # h[i] goes 0->1: contributes +1*(1-h[neighbor]) for all neighbors
    # h[j] goes 1->0: contributes -1*(1-h[neighbor]) for all neighbors
    # But also: when h[i] is the SECOND factor (1-h[.]), it goes from 1 to 0
    # when h[j] is the SECOND factor, it goes from 0 to 1
    
    # For h[i] as first factor (position i in h, matching with position i+k in (1-h)):
    # lags m where i is first: m - (n-1) + i ranges over valid second positions
    # i.e., b = m - (n-1) + i, need 0 <= b <= n-1
    # So m ranges from (n-1)-i to (n-1)-i+(n-1) = 2(n-1)-i
    lo = max(0, n - 1 - i)
    hi = min(2 * n - 2, 2 * n - 2 - i)
    if lo <= hi:
        m = np.arange(lo, hi + 1)
        b = i + m - (n - 1)
        # (1 - h[b]) but with old values (h[i]=0, h[j]=1)
        vals = 1.0 - h[b]
        # Fix if b hits i or j
        if i in b:
            vals[np.where(b == i)[0]] = 1.0  # 1 - old_h[i] = 1 - 0 = 1
        if j in b:
            vals[np.where(b == j)[0]] = 0.0  # 1 - old_h[j] = 1 - 1 = 0
        delta[m] += vals
    
    # For h[i] as second factor (1-h[i]): old value 1-0=1, new value 1-1=0
    # change = -h[a] where a is first factor, a = m - (n-1) - i + ... 
    # Actually: m = a + (n-1) + (b - i) ... no.
    # correlate[m] = sum_a h[a] * (1-h[a+m-(n-1)])
    # If b = i, then a + m - (n-1) = i, so a = i - m + (n-1)
    # Need 0 <= a <= n-1: 0 <= i-m+(n-1) <= n-1
    # => i <= m <= i + n - 1
    lo = max(0, i)
    hi = min(2 * n - 2, i + n - 1)
    if lo <= hi:
        m = np.arange(lo, hi + 1)
        a = i - m + (n - 1)
        vals = h[a]
        if i in a:
            vals[np.where(a == i)[0]] = 0.0  # old h[i] = 0
        if j in a:
            vals[np.where(a == j)[0]] = 1.0  # old h[j] = 1
        delta[m] -= vals
    
    # For h[j] as first factor: old h[j]=1, new h[j]=0
    # change = -1*(1-h[b]) for valid b
    lo = max(0, n - 1 - j)

        b = j + m - (n - 1)
        vals = 1.0 - h[b]
        if i in b:
            vals[np.where(b == i)[0]] = 1.0

            vals[np.where(b == j)[0]] = 0.0
        delta[m] -= vals

    # change = +h[a]
    lo = max(0, j)

            vals[np.where(a == i)[0]] = 0.0
        if j in a:
            vals[np.where(a == j)[0]] = 1.0
        delta[m] += vals
    
    return delta
```

```python
def projected_gradient_smooth(h, n, alpha, deadline, lr=0.01, max_iter=100000):
    """Minimize log-sum-exp of correlate using projected gradient."""
    dx = 2.0 / n
    h = np.clip(h.copy(), 0, 1)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        # Compute g = correlate * dx
        g = np.correlate(h, 1 - h, mode='full') * dx
        
        # Compute softmax weights
        g_shifted = g - g.max()
        p = np.exp(alpha * g_shifted) / np.exp(alpha * g_shifted).sum()
        
        # Compute gradient
        # grad[i] = dx/alpha * (S1[i] - S2[i])
        # S1[i] = sum_m p[m] * (1-h[b]) where b = i + m - (n-1), valid if 0<=b<n
        # S2[i] = sum_m p[m] * h[a] where a = i - m + (n-1), valid if 0<=a<n
        grad = np.zeros(n)
        for i in range(n):
            s1 = 0.0
            s2 = 0.0
            for m in range(2*n-1):
                b = i + m - (n-1)
                if 0 <= b < n:
                    s1 += p[m] * (1.0 - h[b])
                a = i - m + (n-1)
                if 0 <= a < n:
                    s2 += p[m] * h[a]
            grad[i] = dx / alpha * (s1 - s2)
        
        # Projected gradient step
        h_new = h - lr * grad
        # Project onto feasible set
        h_new = np.clip(h_new, 0, 1)
        # Adjust sum
        target = n / 2.0
        diff = target - h_new.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                room = 1.0 - h_new
                s = room.sum()
                if s > 1e-14:
                    h_new += diff / s * room
            else:
                s = h_new.sum()
                if s > 1e-14:
                    h_new += diff / s * h_new
            h_new = np.clip(h_new, 0, 1)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        
        h = h_new
    
    return best_h, best_c
```
