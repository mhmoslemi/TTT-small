import numpy as np
import time
import math
from scipy.optimize import minimize

_N = 0
_DX = 0.0
_OFF = 0
_TARGET = 0.0
_ALPHA = 1.0
_ONES = None
_BOUNDS = None
_CONS = None
_FIRST = None
_SECOND = None

_FULL_PLANS = (

_VERY_PLANS = (

_TINY_PLANS = (

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n))


def _fix_sum(x, n):
    target = 0.5 * n
    for _ in range(3):
        diff = target - float(np.sum(x))
        if abs(diff) < 1e-13:
            break
        m = np.minimum(x, 1.0 - x)
        idx = int(np.argmax(m))
        if m[idx] + 1e-12 < abs(diff):
            kk = min(8, n)
            idxs = np.argsort(m)[::-1][:kk]
            x[idxs] += diff / float(kk)
        else:
            x[idx] += diff
    return np.clip(x, 0.0, 1.0)


def _proj_feasible(y, n):
    y = np.asarray(y, dtype=float)
    if y.size != n:
        y = np.resize(y, n)
    if not np.all(np.isfinite(y)):
        y = np.full(n, 0.5)
    target = 0.5 * n
    lo = -float(np.max(y))
    hi = 1.0 - float(np.min(y))
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(y + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y + 0.5 * (lo + hi), 0.0, 1.0)
    return _fix_sum(x, n)


def _cons_fun(x):
    return float(np.sum(x) - _TARGET)


def _cons_jac(x):
    return _ONES.copy()


def _prepare_opt(n):
    global _N, _DX, _OFF, _TARGET, _ONES, _BOUNDS, _CONS
    _N = int(n)
    _DX = 2.0 / float(_N)
    _OFF = _N - 1
    _TARGET = 0.5 * _N
    _ONES = np.ones(_N, dtype=float)
    _ONES.setflags(write=False)
    _BOUNDS = [(0.0, 1.0)] * _N
    _CONS = [{"type": "eq", "fun": _cons_fun, "jac": _cons_jac}]


def _prepare_bin(n):
    _prepare_opt(n)
    global _FIRST, _SECOND
    ar = np.arange(_N, dtype=np.int32)
    _FIRST = [_OFF + i - ar for i in range(_N)]
    _SECOND = [_OFF + ar - i for i in range(_N)]


def _smooth_fgrad(x):
    x = np.asarray(x, dtype=float)
    g = np.correlate(x, 1.0 - x, mode="full") * _DX
    m = float(np.max(g))
    t = _ALPHA * (g - m)
    sm = float(np.max(t))
    e = np.exp(t - sm)
    s = float(np.sum(e))
    if not np.isfinite(s) or s <= 0.0:
        w = np.zeros_like(g)
        w[int(np.argmax(g))] = 1.0
        lse = 0.0
    else:
        w = e / s
        lse = sm + float(math.log(s))
    f = m + lse / _ALPHA

    c = np.cumsum(w)
    V = np.empty(_N, dtype=float)
    if _N == 1:
        V[0] = w[0]
    else:
        V[0] = c[_N - 1]
        V[1:] = c[_N:] - c[:_N - 1]

    conv1 = np.convolve(x, w, mode="full")[_OFF:_OFF + _N]
    conv2 = np.convolve(x, w[::-1], mode="full")[_OFF:_OFF + _N]
    grad = (V - conv1 - conv2) * _DX
    return f, grad


def _smooth_fun(x):
    return _smooth_fgrad(x)[0]


def _smooth_jac(x):
    return _smooth_fgrad(x)[1]


def _try_slsqp(x, alpha, maxiter):
    global _ALPHA
    _ALPHA = float(alpha)
    x = _proj_feasible(x, _N)
    try:
        res = minimize(
            _smooth_fun,
            x,
            jac=_smooth_jac,
            method="SLSQP",
            bounds=_BOUNDS,
            constraints=_CONS,
            options={"maxiter": int(max(20, int(maxiter))), "ftol": 1e-14},
        )
        if res is not None and hasattr(res, "x") and np.all(np.isfinite(res.x)):
            return _proj_feasible(res.x, _N)
    except Exception:
        pass
    return _proj_feasible(x, _N)


def _polish_slsqp(h_in, deadline, plans):
    n_h = int(len(h_in))
    _prepare_opt(n_h)
    h = _proj_feasible(h_in, n_h)
    c = compute_c5(h, n_h)
    best_h = h.copy()
    best_c = float(c)

    for a, mi in plans:
        if time.time() >= deadline:
            break
        x = _try_slsqp(h, float(a), int(mi))
        c = compute_c5(x, n_h)
        if not np.isfinite(c):
            c = best_c
            x = best_h
        if c < best_c - 1e-15:
            best_c = float(c)
            best_h = x.copy()
        if np.isfinite(c) and c < best_c + 1e-7:
            h = x
        else:
            h = best_h
    return best_h, best_c


def _perturb_start(h, n, rng, sigma, nswap, roll=0):
    x = h.copy()
    if roll:
        x = np.roll(x, int(roll))
    for _ in range(int(nswap)):
        a = int(rng.randint(n))
        b = int(rng.randint(n))
        if a != b:
            x[a], x[b] = x[b], x[a]
    if sigma > 0.0:
        x = x + float(sigma) * rng.normal(size=n)
    return _proj_feasible(x, n)


def _round_top(h, n):
    k = n // 2
    order = np.argsort(h)[::-1]
    out = np.zeros(n, dtype=float)
    out[order[:k]] = 1.0
    return out


def _swap_delta(h, i, j):
    delta = np.zeros(2 * _N - 1, dtype=float)
    delta[_FIRST[i]] += 1.0 - h
    delta[_FIRST[j]] -= 1.0 - h
    delta[_SECOND[i]] -= h
    delta[_SECOND[j]] += h
    delta[_OFF] -= 2.0
    delta[_OFF + i - j] += 1.0
    delta[_OFF + j - i] += 1.0
    return delta


def _binary_sa(h0, n, deadline, seed=0, t_start=2.0, t_end=0.01, patience=20000, kick=6):

    step = 0

    while time.time() < deadline:
        pi = rng.randint(len(zeros))
        pj = rng.randint(len(ones))

        nw = corr + delta
        nc = float(nw.max())

        best_h = _proj_feasible(best_h, best_n)
        best_c = compute_c5(best_h, best_n)

    return best_h, float(best_c), best_n
