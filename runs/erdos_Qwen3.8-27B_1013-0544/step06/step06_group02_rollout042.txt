    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

For the 1D minimax problem: minimize max_k (c_k + d * g_k) over d in [d_min, d_max] (where d_min, d_max come from box constraints).

The optimal d is at a "kink" where two (or more) of the lines c_k + d*g_k intersect at the maximum. This can be found by:
1. Sort the k values by g_k
2. The upper envelope changes at intersection points
3. Find the d where the upper envelope is minimized

A simpler approach: the optimal d is where the derivative of the max changes sign. At the optimum, there exist k1 and k2 such that:
c_{k1} + d*g_{k1} = c_{k2} + d*g_{k2} > c_k + d*g_k for all other k

So d = (c_{k2} - c_{k1}) / (g_{k1} - g_{k2})

We can try all pairs (k1, k2) but that's O(n²). A faster approach:
1. Start with d=0, the max is at k_max
2. The optimal d is in the direction that reduces c_{k_max}
3. At the optimum, either:
   a. d hits a box constraint, or
   b. another k becomes tied with k_max at the new max

Case (b): d = (c_k - c_{k_max}) / (g_{k_max} - g_k) for the k that first catches up.

We can find this by:
- If g_{k_max} > 0 (decrease h[i]): the new max either stays at k_max (if no other k catches up) or jumps to another k.
  - For each k with g_k > g_{k_max}: c_k + d*g_k > c_{k_max} + d*g_{k_max} when d < (c_{k_max} - c_k) / (g_k - g_{k_max})
    Wait, if d < 0: c_k + d*g_k > c_{k_max} + d*g_{k_max} iff (c_k - c_{k_max}) > d*(g_k - g_{k_max})
    If g_k - g_{k_max} > 0: d < (c_k - c_{k_max}) / (g_k - g_{k_max})
    If g_k - g_{k_max} < 0: d > (c_k - c_{k_max}) / (g_k - g_{k_max})
  
This is getting complicated. Let me just use a bisection/ternary search on d.

Actually, the function f(d) = max_k (c_k + d*g_k) is a convex piecewise-linear function of d. We can minimize it over an interval using a simple approach:
- The minimum is at a vertex of the upper envelope
- We can find it by evaluating f at the intersection points

2. d in the direction that reduces the current max, with magnitude chosen to make one other k tie with the current max

For candidate 2: if g[k_max] > 0, we want d < 0. The first k that catches up is:
For each k: d_k = (c_k - c[k_max]) / (g_k - g[k_max]) (if g_k != g[k_max])
We want the smallest |d_k| such that d_k < 0 and c_k + d_k*g_k = c[k_max] + d_k*g[k_max] and this is >= all other c_j + d_k*g_j.

def find_optimal_delta(c, g, dx, hi):
    """Find the optimal delta to add to h[i] to minimize max(c + dx*delta*g)."""
    t = c.max()
    k_max = np.argmax(c)
    g_max = g[k_max]
    
    if abs(g_max) < 1e-15:
        return 0.0
    
    # Direction: reduce c[k_max]
    if g_max > 0:
        # Need delta < 0
        d_box = -hi  # h[i] + delta >= 0
        d_dir = -1  # negative direction
    else:
        # Need delta > 0
        d_box = 1.0 - hi  # h[i] + delta <= 1
        d_dir = 1  # positive direction
    
    # Find the delta that makes another k tie with k_max
    # c_k + dx*delta*g_k = t + dx*delta*g_max
    # dx*delta*(g_k - g_max) = t - c_k
    # delta = (t - c_k) / (dx * (g_k - g_max))
    
    candidates = [0.0, d_box * 0.5, d_box]
    
    for k in range(len(c)):
        if k == k_max:
            continue
        if abs(g[k] - g_max) < 1e-15:
            continue
        d = (t - c[k]) / (dx * (g[k] - g_max))
        # Check it's in the right direction and within box
        if d_dir > 0 and d > 0 and d <= d_box:
            candidates.append(d)
        elif d_dir < 0 and d < 0 and d >= d_box:
            candidates.append(d)
    
    # Evaluate at each candidate
    best_d = 0.0
    best_val = t
    for d in candidates:
        if d_dir > 0 and d < -1e-15:
            continue
        if d_dir < 0 and d > 1e-15:
            continue
        if hi + d < -1e-12 or hi + d > 1 + 1e-12:
            continue
        new_max = np.max(c + dx * d * g)
        if new_max < best_val - 1e-15:
            best_val = new_max
            best_d = d
    
    return best_d

import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    return np.correlate(h, 1 - h, mode='full') * (2.0 / n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for li in range(2 * n - 1):
        if w[li] < 1e-14:
            continue
        k = li - (n - 1)
        wm = w[li] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def find_opt_delta(c, g, dx, hi):
    t = c.max()
    k_max = np.argmax(c)
    g_max = g[k_max]
    if abs(g_max) < 1e-15:
        return 0.0
    if g_max > 0:
        d_box = -hi
        d_dir = -1
    else:
        d_box = 1.0 - hi
        d_dir = 1
    cands = [0.0, 0.3 * d_box, 0.6 * d_box, 0.9 * d_box, d_box]
    for k in range(len(c)):
        if k == k_max:
            continue
        denom = g[k] - g_max
        if abs(denom) < 1e-15:
            continue
        d = (t - c[k]) / (dx * denom)
        if d_dir > 0 and 0 < d <= d_box:
            cands.append(d)
        elif d_dir < 0 and d_box <= d < 0:
            cands.append(d)
    best_d = 0.0
    best_val = t
    for d in cands:
        if d > 1e-15 and hi + d > 1.0 + 1e-12:
            continue
        if d < -1e-15 and hi + d < -1e-12:
            continue
        nv = np.max(c + dx * d * g)
        if nv < best_val - 1e-15:
            best_val = nv
            best_d = d
    return best_d

def coord_descent(h, n, deadline, max_rounds=10000):
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    dx = 2.0 / n
    c = compute_c5_vec(best_h, n)
    k_vals = np.arange(2 * n - 1) - (n - 1)
    consecutive_no_improve = 0
    for rd in range(max_rounds):
        if time.time() > deadline:
            break
        any_improve = False
        for i in range(n):
            hi = best_h[i]
            ik = i + k_vals
            ik2 = i - k_vals
            v1 = (ik >= 0) & (ik < n)
            v2 = (ik2 >= 0) & (ik2 < n)
            g = np.zeros(2 * n - 1)
            g[v1] = 1.0 - best_h[ik[v1]]
            g[v2] -= best_h[ik2[v2]]
            d = find_opt_delta(c, g, dx, hi)
            if abs(d) > 1e-14:
                best_h[i] = hi + d
                c = c + dx * d * g
                if c.max() < best_c - 1e-14:

                any_improve = True
        best_h = project_feasible(best_h, n)
        c = compute_c5_vec(best_h, n)
        cur = c.max()
        if cur < best_c - 1e-14:
            best_c = cur
        if not any_improve:

            if consecutive_no_improve > 3:

def nesterov_opt(h0, n, deadline, alpha=500, max_iters=20000):

def pair_polish(h, n, deadline, steps=None):
    if steps is None:
        steps = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    best_h = project_feasible(h, n)
    best_c = compute_c5(best_h, n)
    for s in steps:
        while time.time() < deadline:
            improved = False
            for i in range(0, n, 1):
                if time.time() > deadline:
                    return best_h, best_c
                hi = best_h[i]
                for j in range((i + 1) % n, (i + 1) % n + n):
                    jj = j % n
                    if jj == i:
                        continue
                    for d in [s, -s]:
                        ni, nj = hi + d, best_h[jj] - d
                        if ni < 0 or ni > 1 or nj < 0 or nj > 1:
                            continue
                        best_h[i] = ni
                        best_h[jj] = nj
                        c = compute_c5(best_h, n)

                        else:
                            best_h[i] = hi
                            best_h[jj] = best_h[jj] + (-d) - (nj - best_h[jj] + (-d))
                            best_h[i] = hi
                            best_h[jj] = nj + d
            if not improved:
                break
    return best_h, best_c
