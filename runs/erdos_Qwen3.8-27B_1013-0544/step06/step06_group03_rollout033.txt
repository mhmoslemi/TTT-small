def linear_active_set_solve(h, n, top_k=5, eps_max=0.15):
    """Solve the linearized active set problem."""
    c = compute_all_lags(h, n)
    top_idx = np.argsort(c)[-top_k:]
    
    # Build LP
    # Variables: eps[0..n-1], t
    # Objective: minimize t
    # Constraints:
    #   For each active lag k: grad_k @ eps <= t - c[k]
    #   -min(h[i], eps_max) <= eps[i] <= min(1-h[i], eps_max)
    #   sum(eps) = 0
    
    grad = np.zeros((len(top_idx), n))
    for i, idx in enumerate(top_idx):
        lag = idx - (n - 1)
        grad[i] = lag_gradient(h, n, lag)
    
    # LP: min t
    # s.t. grad[i] @ eps - t <= -c[top_idx[i]] for each i
    #      sum(eps) = 0
    #      lo[i] <= eps[i] <= hi[i]
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize t
    
    # Inequality constraints: grad[i] @ eps - t <= -c[top_idx[i]]
    # A_ub has shape (top_k, n+1)
    A_ub = np.zeros((len(top_idx), n + 1))
    for i in range(len(top_idx)):
        A_ub[i, :n] = grad[i]
        A_ub[i, n] = -1.0
    b_ub = -c[top_idx].copy()
    
    # Equality: sum(eps) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [0.0]
    
    lo = np.maximum(-h, -eps_max)
    hi = np.minimum(1 - h, eps_max)
    bounds = list(zip(lo, hi)) + [(None, None)]  # t is unbounded
    
    from scipy.optimize import linprog
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')

        h_new = project_feasible(h + eps, n)

    idx = np.argsort(c)[-top_k:]
    A = np.zeros((top_k, n + 1))
    b = np.zeros(top_k)
    for i, k in enumerate(idx):
        g = lag_gradient(h, n, k - (n - 1))
        A[i, :n] = g
        A[i, n] = -1.0
        b[i] = -c[k]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    lo = np.maximum(-h, -eps_max)
    hi = np.minimum(1.0 - h, eps_max)
    bnds = list(zip(lo, hi)) + [(None, None)]
    res = linprog(c_obj, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=[0.0],
                  bounds=bnds, method='highs')
    if res.success:
        return project_feasible(h + res.x[:n], n)
    return h.copy()

def fine_local_search(h, n, deadline):
    """Coordinate-wise perturbation search with sum preservation."""
    h_curr = h.copy()
    best_c = compute_c5(h_curr, n)
    best_h = h_curr.copy()
    deltas = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                # Try increasing h[i] and decreasing h[(i+1)%n] (cyclic transfer)
                for j_offset in [1, 2, 3, n//4, n//3, n//2]:
                    j = (i + j_offset) % n
                    if h_curr[i] + d > 1.0 or h_curr[j] - d < 0.0:
                        continue
                    h_curr[i] += d
                    h_curr[j] -= d
                    c_new = compute_c5(h_curr, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        best_h = h_curr.copy()
                        improved = True
                    else:
                        h_curr[i] -= d
                        h_curr[j] += d
                    if improved:
                        break
    return best_h, best_c

            if s < 1e-14:
                break
            h += (d / s) * r

            s = h.sum()
            if s < 1e-14:

def lag_gradient(h, n, lag):
    m = abs(lag)

        return np.zeros(n)

    elif lag < 0:
        g[m:] += 1.0 - h[:n - m]
