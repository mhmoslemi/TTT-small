def compute_c5(h, n):

    c = np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break

            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:

        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff * h / s
    return np.clip(h, 0.0, 1.0)

def subgradient_step(h, n, alpha, max_lags=20):
    """One subgradient step for the continuous relaxation."""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    # Find top lags
    top_lags = np.argsort(corr)[-max_lags:]
    # Subgradient: average of gradients over top lags
    g = np.zeros(n)
    center = n - 1
    for k in top_lags:
        offset = k - center  # this is the actual lag
        for i in range(n):

            if 0 <= i - offset < n:
                term2 = h[i - offset]
            g[i] += term1 - term2
    g /= max_lags
    h_new = h - alpha * g
    return project_feasible(h_new, n)

import time

def project_feasible(h, n):

            if s < 1e-14:
                break
            h = h + diff * room / s
        else:
            s = h.sum()
            if s < 1e-14:
                break

def compute_gradient(h, n, weights):
    g = np.zeros(n)
    center = n - 1
    m = 2 * n - 1

def binary_sa(h0, n, deadline, seed=42, p2=0.3):

    dx = 2.0 / n
    h = h0.copy()

    zeros = np.where(h == 0)[0].copy()

    best_val = cur_val

        t = max(0.0005, 0.008 * (0.0005 / 0.008) ** frac)

            else:

            h[a], h[c] = 0, 1

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    bounds = [(0.0, 1.0)] * n
    try:
        if time.time() > deadline:
            return h
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
