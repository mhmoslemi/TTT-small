                    h = best_h.copy()
                    c = np.correlate(h, 1 - h, mode='full').astype(float)
                    g = build_g_binary(h, n)
                    ones = np.where(h == 1)[0].copy()
                    zeros = np.where(h == 0)[0].copy()
                    cur_c = c.max() * dx

    return best_h, best_c

def _do_swap(g, h, i, j, n, ones, zeros):
    m = 2 * n - 1
    gi = g[i].copy()
    gj = g[j].copy()
    p_idx = np.arange(n)
    for _i in range(n):
        for _j in range(n):
            pass
    # Update g efficiently: flipping i and j changes g for all rows
    # g[row, col] depends on h values, so we need to rebuild
    # For efficiency, we update only the affected entries
    # Actually, let's just rebuild - for n=250 it's fast enough
    h[i], h[j] = 1.0, 0.0 if h[i] == 0 else 0.0, 1.0
    # Wait, i was 0->1, j was 1->0
    h[i] = 1.0
    h[j] = 0.0
    ii = np.where(zeros == i)[0]
    jj = np.where(ones == j)[0]
    if len(ii) > 0:
        zeros = np.delete(zeros, ii[0])
    if len(jj) > 0:
        ones = np.delete(ones, jj[0])
    zeros = np.append(zeros, j)
    ones = np.append(ones, i)
    g = build_g_binary(h, n)
    return g, h, ones, zeros

def continuous_refine(h, n, deadline, alpha=5.0):
    dx = 2.0 / n
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    step = 0.05
    stall = 0
    m = 2 * n - 1
    offset = n - 1

    while time.time() < deadline:
        c = np.correlate(h, 1 - h, mode='full')
        c_max = c.max() * dx
        # Soft gradient from top correlations
        top_idx = np.argsort(c)[-8:]
        vals = c[top_idx]
        scaled = (vals - vals.max()) / alpha
        weights = np.exp(scaled)
        weights /= weights.sum()

        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            s = k - offset
            a_idx = np.arange(n) - s
            b_idx = np.arange(n) + s
            valid_a = (a_idx >= 0) & (a_idx < n)
            valid_b = (b_idx >= 0) & (b_idx < n)
            t1 = np.zeros(n)
            t2 = np.zeros(n)
            t1[valid_a] = 1.0 - h[a_idx[valid_a]]
            t2[valid_b] = -h[b_idx[valid_b]]
            grad += w * (t1 + t2)

        p = np.argmin(grad)
        q = np.argmax(grad)
        if p == q:
            break

        eps = min(step, 1.0 - h[p], h[q])
        if eps < 1e-12:
            step *= 0.85
            stall += 1
            if stall > 50:
                break
            continue

        h_new = h.copy()
        h_new[p] += eps
        h_new[q] -= eps
        h_new = np.clip(h_new, 0, 1)

        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-16:
            h = h_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
            stall = 0
            step = min(0.3, step * 1.03)
        else:
            step *= 0.88
            stall += 1
            if stall > 100:
                # Try a different pair
                grad_sorted = np.argsort(grad)
                found = False
                for trial in range(10):
                    p2 = grad_sorted[trial]
                    q2 = grad_sorted[-1 - trial]
                    if p2 == q2:
                        continue
                    eps2 = min(step, 1.0 - h[p2], h[q2])
                    if eps2 < 1e-12:
                        continue
                    h_try = h.copy()
                    h_try[p2] += eps2
                    h_try[q2] -= eps2
                    h_try = np.clip(h_try, 0, 1)
                    c_try = compute_c5(h_try, n)
                    if c_try < best_c - 1e-16:
                        h = h_try
                        best_c = c_try
                        best_h = h_try.copy()
                        step = min(0.3, step * 1.03)
                        stall = 0
                        found = True
                        break
                if not found:
                    step *= 0.9
                    if step < 1e-10:
                        break
                    stall = 0

    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    m = 2 * n - 1
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        mx = c.max()
        shifted = alpha * (c - mx)
        sm = shifted.max()
        return mx + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = 0.5 * np.ones(n)
    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: Quick binary search from multiple starts (25% time)
    p1 = t0 + 0.25 * (budget_s - 5)
    inits = [h0]
    inits.append(thue_morse_init(n, seed))
    inits.append(kronecker_init(n, seed + 7))
    h_rand = np.zeros(n)
    h_rand[rng.choice(n, n // 2, replace=False)] = 1.0
    inits.append(h_rand)

    t_split1 = (p1 - time.time()) / len(inits)
    for idx, h_init in enumerate(inits):
        if time.time() > p1 - 5:
            break
        hb, cb = binary_local_search(h_init, n, time.time() + max(t_split1, 30),
                                     top_k_init=20, top_k_min=5, seed=seed + idx * 100)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()

    # Phase 2: Deep binary search on best (20% time)
    p2 = t0 + 0.45 * (budget_s - 5)
    if time.time() < p2 - 10:
        hb, cb = binary_local_search(best_h, n, p2 - 5,
                                     top_k_init=15, top_k_min=3, seed=seed + 500)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()

    # Phase 3: Continuous gradient refinement (25% time)
    p3 = t0 + 0.70 * (budget_s - 5)
    if time.time() < p3 - 10:
        # Round to binary first for a clean start
        h_bin = (best_h > 0.5).astype(float)
        # Fix count
        target = n // 2
        cnt = int(h_bin.sum())
        if cnt > target:
            idx1 = np.where(h_bin == 1)[0]
            rng2 = np.random.RandomState(seed + 999)
            rng2.shuffle(idx1)
            h_bin[idx1[:cnt - target]] = 0
        elif cnt < target:
            idx0 = np.where(h_bin == 0)[0]
            rng2 = np.random.RandomState(seed + 999)
            rng2.shuffle(idx0)
            h_bin[idx0[:target - cnt]] = 1

        # Continuous from binary
        hc, cc = continuous_refine(h_bin, n, time.time() + (p3 - time.time()) * 0.5)
        if cc < best_c:
            best_c = cc
            best_h = hc.copy()

        # Continuous from slightly perturbed binary
        hp = h_bin + 0.1 * np.random.RandomState(seed + 777).randn(n)
        hp = project_feasible(hp, n)
        if time.time() < p3 - 10:
            hc2, cc2 = continuous_refine(hp, n, p3 - 5)
            if cc2 < best_c:
                best_c = cc2
                best_h = hc2.copy()

    # Phase 4: SLSQP polish (10% time)
    p4 = t0 + 0.80 * (budget_s - 5)
    if time.time() < p4 - 5:
        h_s = project_feasible(best_h + 0.02 * rng.randn(n), n)
        for a in [50000, 200000]:
            if time.time() > p4 - 3:
                break
            h_s = slsqp_smooth(h_s, n, a, 800, p4 - 2)
            c = compute_c5(h_s, n)
            if c < best_c:
                best_c = c
                best_h = h_s.copy()

    # Phase 5: Final binary + continuous (remaining)
    if time.time() < dl - 10:
        # Quick binary polish
        p5a = time.time() + 0.4 * (dl - time.time() - 10)
        hb, cb = binary_local_search(best_h, n, p5a, top_k_init=10, top_k_min=3, seed=seed + 800)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
        # Final continuous
        if time.time() < dl - 5:
            hc, cc = continuous_refine(best_h, n, dl - 3, alpha=3.0)
            if cc < best_c:
                best_c = cc
                best_h = hc.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
