```python
def subgradient_polish(h, n, deadline, top_m=5, epsilon=0.01, max_iter=100000):
    dx = 2.0 / n
    h = h.astype(float).copy()
    best_h = h.copy()
    best_c = np.max(np.correlate(h, 1-h, mode='full')) * dx
    
    lr = 0.001
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        corr = np.correlate(h, 1-h, mode='full')
        cur_max = corr.max()
        
        # Find active shifts (within epsilon of max)
        active = np.where(corr > cur_max - epsilon * dx)[0]
        
        # Compute gradient of sum of active corr values
        grad = np.zeros(n)
        for k_idx in active:
            shift = k_idx - (n - 1)
            # d corr[k] / d h[m] = (1-h[m+shift]) * [m+shift valid] - h[m-shift] * [m-shift valid]
            # Term 1: (1-h[m+shift]) for m+shift in [0,n-1]
            if shift >= 0:
                # m+shift in [0,n-1] → m in [0, n-1-shift]
                grad[:n-shift] += (1 - h[shift:n])
                # m-shift in [0,n-1] → m in [shift, n-1]
                grad[shift:] -= h[:n-shift]
            else:
                s = -shift
                # m+shift = m-s in [0,n-1] → m in [s, n-1]
                grad[s:] += (1 - h[:n-s])
                # m-shift = m+s in [0,n-1] → m in [0, n-1-s]
                grad[:n-s] -= h[s:]
        
        # Gradient step
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        
        c_new = np.max(np.correlate(h_new, 1-h_new, mode='full')) * dx
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        
        h = h_new
        
        # Adaptive learning rate
        if c_new > cur_max * dx:
            lr *= 0.9
        else:
            lr = min(lr * 1.01, 0.01)
    
    return best_h, best_c
```

```python
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def build_g_fast(h, n):
    p = np.arange(n)[:, None]
    k = np.arange(2 * n - 1)[None, :]
    kp = k - p
    valid = (kp >= 0) & (kp < n) & (k != 2 * np.arange(n)[:, None])
    h_idx = np.clip(kp, 0, n - 1)
    g = np.where(valid, 1.0 - 2.0 * h[h_idx], 0.0)
    return g

def apply_swap(g, h, corr, n, i, j):
    """Swap h[i]: 0→1, h[j]: 1→0. Update corr and g incrementally."""
    # Update corr
    corr = corr + g[i] - g[j]
    # Update g: for each row p, g[p, p+i] and g[p, p+j] change
    p_idx = np.arange(n)
    col_i = p_idx + i
    col_j = p_idx + j
    valid_i = col_i < 2 * n - 1
    valid_j = col_j < 2 * n - 1
    # h[i] was 0, now 1: g[p, p+i] was 1-2*h[i]=1, now 1-2*1=-1, so change = -2
    # But only if p+i != 2*p (the excluded diagonal)
    mask_i = valid_i & (col_i != 2 * p_idx)
    g[p_idx[mask_i], col_i[mask_i]] -= 2.0
    # h[j] was 1, now 0: g[p, p+j] was 1-2*h[j]=-1, now 1-2*0=1, so change = +2
    mask_j = valid_j & (col_j != 2 * p_idx)
    g[p_idx[mask_j], col_j[mask_j]] += 2.0
    h[i], h[j] = 1.0, 0.0
    return g, h, corr

    corr = np.correlate(h, 1 - h, mode='full').astype(float)

    best_c = cur_c
    best_h = h.copy()
    g = build_g_fast(h, n)
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()

    improve_count = 0

def subgradient_polish(h_init, n, deadline, top_m=10, max_iter=200000):
    dx = 2.0 / n
    h = h_init.astype(float).copy()
    best_h = h.copy()
    corr = np.correlate(h, 1 - h, mode='full')
    best_c = float(corr.max() * dx)
    
    lr = 0.005
    epsilon = dx * 2  # consider shifts within 2*dx of max
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        cur_max = corr.max()
        active = np.where(corr > cur_max - epsilon)[0]
        
        # Compute gradient of sum of active correlations
        grad = np.zeros(n)
        for k_idx in active:
            shift = k_idx - (n - 1)
            if shift >= 0:
                grad[:n - shift] += (1.0 - h[shift:n])
                grad[shift:] -= h[:n - shift]
            else:
                s = -shift
                grad[s:] += (1.0 - h[:n - s])
                grad[:n - s] -= h[s:]
        
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        
        corr = np.correlate(h_new, 1 - h_new, mode='full')
        c_new = float(corr.max() * dx)
        
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        
        h = h_new
        
        # Adaptive lr
        if c_new > best_c + 1e-8:
            lr *= 0.95
        else:
            lr = min(lr * 1.005, 0.02)
        
        # Periodically re-center epsilon
        if it % 1000 == 0:
            epsilon = max(dx, cur_max * 0.01)
    
    return best_h, best_c

def block_swap_kick(h, n, rng):
    """Perform a random block swap: swap two contiguous blocks of equal length."""
    h = h.copy()
    for _ in range(3):
        block_len = rng.randint(3, max(4, n // 10))
        if block_len >= n:
            continue
        start1 = rng.randint(0, n - block_len)
        start2 = rng.randint(0, n - block_len)

        h[start1:start1+block_len], h[start2:start2+block_len] = \
            h[start2:start2+block_len].copy(), h[start1:start1+block_len].copy()
    return h
```
