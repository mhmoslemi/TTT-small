**Key insight for continuous optimization:**

The objective max_k c_k(h) is piecewise linear in each pair (h[i], h[j]) but not globally linear. However, if I use the log-sum-exp trick, I get a smooth objective.

The gradient of the smoothed objective w.r.t. h can be computed efficiently:
- p_k = softmax(α * c_k) (probability weights for each shift)
- grad_h[j] = ∑_k p_k * (∂c_k/∂h[j]) * dx

And ∂c_k/∂h[j] = (1-h[j-lag_k])*I[valid] - h[j+lag_k]*I[valid]

This can be computed as:
grad_h[j] = ∑_k p_k*(1-h[j-lag_k])*I[0≤j-lag_k<n] - ∑_k p_k*h[j+lag_k]*I[0≤j+lag_k<n]

The first term is a weighted average of (1-h) at different shifts, and the second is a weighted average of h at different shifts. These can be computed via weighted convolutions.

For n=250, I'll just compute this directly in a loop over k (2n-1 = 499 terms), each contributing O(n) work. Total: O(n²) per gradient evaluation, which is about 62500 operations. That's fast.

```python
def smooth_objective_and_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    max_c = corr.max()
    shifted = alpha * (corr - max_c)
    exp_vals = np.exp(shifted)
    log_sum = np.log(exp_vals.sum())
    obj = (max_c + log_sum) / alpha
    p = exp_vals / exp_vals.sum()
    # Gradient
    grad = np.zeros(n)
    nn1 = n - 1
    for k in range(2 * n - 1):
        if p[k] < 1e-15:
            continue
        lag = k - nn1
        # d(c_k)/d(h[j]) = (1-h[j-lag])*I[0<=j-lag<n] - h[j+lag]*I[0<=j+lag<n]
        # First term: for j such that 0<=j-lag<n, i.e., lag<=j<=lag+n-1
        j_lo = max(0, lag)
        j_hi = min(n - 1, lag + n - 1)
        if j_lo <= j_hi:
            grad[j_lo:j_hi+1] += p[k] * dx * (1.0 - h[j_lo - lag:j_hi - lag + 1])
        # Second term: for j such that 0<=j+lag<n, i.e., -lag<=j<=n-1-lag
        j_lo2 = max(0, -lag)
        j_hi2 = min(n - 1, n - 1 - lag)
        if j_lo2 <= j_hi2:
            grad[j_lo2:j_hi2+1] -= p[k] * dx * h[j_lo2 + lag:j_hi2 + lag + 1]
    return obj, grad
```

```python
def project_set(h, n):
    """Project h onto {x: 0<=x_i<=1, sum(x)=n/2}"""
    target = n / 2.0
    # Use bisection on threshold
    lo, hi = 0.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        projected = np.clip(h - mid, 0, 1)
        s = projected.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    projected = np.clip(h - (lo + hi) / 2, 0, 1)
    # Fix residual
    diff = target - projected.sum()
    # Distribute among non-saturated components
    mask = (projected > 0) & (projected < 1)
    if mask.sum() > 0:
        projected[mask] += diff / mask.sum()
    return np.clip(projected, 0, 1)
```

```python
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def lse_objective(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full').astype(float) * dx
    m = corr.max()
    s = alpha * (corr - m)
    e = np.exp(s)
    return (m + np.log(e.sum())) / alpha

def lse_gradient(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full').astype(float) * dx
    m = corr.max()
    s = alpha * (corr - m)
    e = np.exp(s)
    p = e / e.sum()
    grad = np.zeros(n)
    nn1 = n - 1
    for k in range(2 * n - 1):
        if p[k] < 1e-12:
            continue
        lag = k - nn1
        # d(corr[k])/d(h[j]) for valid j
        # corr[k] = sum_m h[m]*(1-h[m-lag])
        # d/dh[j] = (1-h[j-lag])*I[0<=j-lag<n] - h[j+lag]*I[0<=j+lag<n]
        # (with the caveat about double-counting when lag=0, but for the gradient it's fine
        #  because we're differentiating the expression h[m]*(1-h[m-lag]) w.r.t. h[j])
        # Actually, the gradient of h[m]*(1-h[m-lag]) w.r.t. h[j] is:
        # = (1-h[j-lag]) if m=j (first factor)
        # + h[m]*(-1) if m-lag=j (second factor), i.e., m=j+lag
        # So the total gradient at j is:
        # (1-h[j-lag])*I[j-lag valid] + (-h[j+lag])*I[j+lag valid]
        # But wait, when j = j+lag (lag=0), we get:
        # (1-h[j]) + (-h[j]) = 1-2h[j]
        # which is correct: d(h[j](1-h[j]))/dh[j] = 1-2h[j]
        # Good, so the formula works even for lag=0!
        
        j_lo = max(0, lag)
        j_hi = min(n - 1, n - 1 + lag)
        if j_lo <= j_hi:
            grad[j_lo:j_hi+1] += p[k] * (1.0 - h[j_lo - lag : j_hi - lag + 1])
        j_lo2 = max(0, -lag)
        j_hi2 = min(n - 1, n - 1 - lag)
        if j_lo2 <= j_hi2:
            grad[j_lo2:j_hi2+1] -= p[k] * h[j_lo2 + lag : j_hi2 + lag + 1]
    return grad
```

```python
    lr = 0.01
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)

            best_h = h.copy()
```
