def local_search_binary(h, n, deadline, seed=42):
    h = binarize(h, n)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float).copy()
    cur_max = corr.max()
    best_c = cur_max * dx
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    
    # Precompute: positions near the maximum
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    while time.time() < deadline:
        improved = False
        # Shuffle for diversity
        rng.shuffle(ones)
        rng.shuffle(zeros)
        
        for i in zeros:
            for j in ones:
                d = swap_delta_fast(h, i, j, n)
                # Fast check: if max position doesn't increase
                new_max = (corr + d).max()
                if new_max < cur_max - 1e-10:
                    h[i], h[j] = 1.0, 0.0
                    corr += d
                    cur_max = new_max
                    best_c = cur_max * dx
                    best_h = h.copy()
                    improved = True
                    # Update ones and zeros
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                    break
            if improved:
                break
        
        if not improved:
            break
    
    return best_h, best_c

import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def binarize(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]

    b[idx[:target]] = 1.0

def swap_delta(v, i, j, n):
    s = np.arange(-(n - 1), n)
    d = np.zeros(2 * n - 1)
    va = (s >= -i) & (s <= n - 1 - i)
    if va.any():
        ia = i + s[va]
        vals = 1.0 - v[ia]
        vals[ia == i] = 0.0
        vals[ia == j] = 1.0
        d[va] = vals
    vb = (s >= -j) & (s <= n - 1 - j)
    if vb.any():
        d[vb] -= (1.0 - v[j + s[vb]])
    ci = (s >= i - (n - 1)) & (s <= i) & (s != 0) & (s != (i - j))

        d[ci] -= v[i - s[ci]]
    cj = (s >= j - (n - 1)) & (s <= j) & (s != 0) & (s != (j - i))
    if cj.any():
        d[cj] += v[j - s[cj]]
    return d

    rng = np.random.RandomState(seed)

        ones = np.where(h == 1)[0].copy()
        zeros = np.where(h == 0)[0].copy()
        rng.shuffle(ones)
        rng.shuffle(zeros)
        improved = False
        for i in zeros:
            if time.time() > deadline:
                break
