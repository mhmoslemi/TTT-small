            s2 = np.dot(p[n1 + lags2], h[j_lo:i+1])
            
            grad[i] = dx * (s1 - s2)
        
        # Projected gradient step
        h_new = np.clip(h - lr * grad, 0, 1)
        h_new = project_feasible(h_new, n)
        
        c = compute_c5(h_new, n)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
        
        # Adaptive learning rate
        if c > best_c * 1.01:
            lr *= 0.5
        elif it % 100 == 99:
            lr = min(lr * 1.1, 0.1)
        
        h = h_new
    
    return best_h

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def corr_of(h):
    return np.correlate(h, 1 - h, mode='full')

def binarize(h, n):
    target = n // 2
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def make_thue_morse(n, seed=0):
    tm = np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)
    target = n // 2
    n_ones = int(tm.sum())
    rng = np.random.RandomState(seed)
    if n_ones > target:
        ones = np.where(tm == 1)[0]
        rng.shuffle(ones)
        tm[ones[:n_ones - target]] = 0
    elif n_ones < target:
        zeros = np.where(tm == 0)[0]
        rng.shuffle(zeros)
        tm[zeros[:target - n_ones]] = 1
    return tm

def swap_delta(h, i, j, n):
    n1 = n - 1
    delta = np.zeros(2 * n - 1)
    lo = max(0, n1 - i)
    hi = min(2 * n1, n1 + n1 - i)
    if lo <= hi:
        m = np.arange(lo, hi + 1)
        delta[m] += 1.0 - h[m - n1 + i]
    lo2 = max(0, i)
    hi2 = min(2 * n1, n1 + i)
    if lo2 <= hi2:
        m = np.arange(lo2, hi2 + 1)
        delta[m] -= h[n1 + i - m]
    lo = max(0, n1 - j)
    hi = min(2 * n1, n1 + n1 - j)
    if lo <= hi:
        m = np.arange(lo, hi + 1)
        delta[m] -= 1.0 - h[m - n1 + j]
    lo2 = max(0, j)
    hi2 = min(2 * n1, n1 + j)
    if lo2 <= hi2:
        m = np.arange(lo2, hi2 + 1)
        delta[m] += h[n1 + j - m]
    return delta

def local_search(h, n, deadline, max_sweeps=2000, seed=42):
    h = binarize(h, n).copy()
    dx = 2.0 / n
    corr = corr_of(h).copy()
    cur_c = float(corr.max()) * dx
    best_c = cur_c
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    no_improve = 0
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        rng.shuffle(ones)
        rng.shuffle(zeros)
        improved = False
        for i in zeros:
            if time.time() > deadline:
                break
            best_j = -1
            best_val = cur_c
            for j in ones[:30]:
                d = swap_delta(h, int(i), int(j), n)
                nc = float((corr + d).max()) * dx
                if nc < best_val:
                    best_val = nc
                    best_j = int(j)
            if best_j >= 0 and best_val < cur_c - 1e-15:
                d = swap_delta(h, int(i), best_j, n)
                h[i], h[best_j] = 1.0, 0.0
                corr = corr + d
                cur_c = best_val
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                improved = True
                break
        if not improved:
            no_improve += 1
            if no_improve >= 5:
                break
        else:
            no_improve = 0
    return best_h, best_c

def smooth_optimize(h, n, deadline, alpha_start=500, alpha_end=50000, max_iter=5000):
    dx = 2.0 / n
    n1 = n - 1
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.05
    total_iter = max_iter
    
    for it in range(total_iter):
        if time.time() > deadline:
            break
        # Progressive alpha
        alpha = alpha_start * (alpha_end / alpha_start) ** (it / max(1, total_iter - 1))
        
        corr = corr_of(h)
        g = corr * dx
        g_max = g.max()
        exp_g = np.exp(alpha * (g - g_max))
        p = exp_g / exp_g.sum()
        
        # a[m] = p[n1+m] for m=0,...,n-1
        a = p[n1:2*n1+1]  # length n
        # s1[i] = sum_{k=0}^{n-1-i} a[k] * (1-h[i+k]) = correlate(1-h, a, 'full')[n1+i]
        s1 = np.correlate(1 - h, a, mode='full')[n1:2*n1+1]
        # s2[i] = sum_{m
