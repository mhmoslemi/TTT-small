    c = np.correlate(h, 1.0 - h, mode='full')
    sc = alpha * c
    mc = sc.max()
    ec = np.exp(sc - mc)
    p = ec / ec.sum()

    nidx = 2 * n - 1
    for k in range(nidx):
        pk = p[k]
        if pk < 1e-9:
            continue
        ell = k - (n - 1)
        wk = pk * alpha * (2.0 / n)
        if ell == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif ell > 0:
            if n - ell > 0:
                grad[:n-ell] += wk * (1.0 - h[ell:])
            if ell < n:
                grad[ell:] -= wk * h[:n-ell]
        else:
            le = -ell
            if n - le > 0:
                grad[le:] += wk * (1.0 - h[:n-le])
            if le < n:
                grad[:n-le] -= wk * h[le:]
    
    h_new = project_feasible(h - lr * grad, n)
    return h_new

def compute_c5(h):

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)

            if r.sum() < 1e-15:
                break
            h += (d / r.sum()) * r

            if h.sum() < 1e-15:
                break
            h += (d / h.sum()) * h
    return np.clip(h, 0.0, 1.0)
