def smooth_obj(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    s = np.sum(np.exp(sh - mx))
    return float((m + (mx + np.log(s)) / alpha) * (2.0 / n))

        return x.sum() - n / 2.0

        return np.ones(n)

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

        if abs(diff) < 1e-12: break

            if tot < 1e-15: break

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12: break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def coord_descent(h, n, max_passes=400, deadline=None, rng=None):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    tl = 2 * n - 1
    for _pass in range(max_passes):

        improved = False
        order = np.arange(n)
        if rng is not None: rng.shuffle(order)
        for p in order:

            if d_hi - d_lo < 1e-14: continue

            k_hi = min(tl-1, max(p+n-1, tl-1-p))
            UM = 0.0
            if k_lo > 0: UM = float(np.max(c[:k_lo]))
            if k_hi < tl-1: UM = max(UM, float(np.max(c[k_hi+1:])))
            ks = np.arange(k_lo, k_hi+1)
            s = np.zeros(len(ks))
            fwd = (ks >= p) & (ks <= p+n-1)
            bwd = (ks >= n-1-p) & (ks <= tl-1-p)
            if np.any(fwd): s[fwd] += 1.0 - h[p - ks[fwd] + n - 1]
            if np.any(bwd): s[bwd] -= h[p + ks[bwd] - (n-1)]
            nh = h[p]
            ci = (ks == n-1)
            has_c = np.any(ci)
            best_val = np.inf; best_d = 0.0
            for stage in range(3):

                    grid = np.linspace(lo, hi, 14)
                for d in grid:

                    val = max(UM, float(np.max(nv)))

                h[p] = np.clip(h[p] + best_d, 0.0, 1.0)
                improved = True
        if not improved: break
    return h

def slsqp_opt(h, n, alpha=80000, maxiter=1200):
    def obj(x):
        c = np.correlate(x, 1.0-x, mode='full')
        m = c.max(); sh = alpha*(c-m); mx = sh.max()
        return float((m + (mx + np.log(np.sum(np.exp(sh-mx))))/alpha) * (2.0/n))
    def cf(x): return x.sum() - n/2.0
    def cj(x): return np.ones(n)
    try:
        res = minimize(obj, h, method='SLSQP', bounds=[(0.0,1.0)]*n,
                       constraints=[{'type':'eq','fun':cf,'jac':cj}],
                       options={'maxiter':maxiter,'ftol':1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def pair_swap(h, n, deadline, steps=None, radius=60):
    if steps is None: steps = [0.02, 0.01, 0.005, 0.002]
    best_c = compute_c5(h, n)
    for st in steps:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for _epoch in range(50):
                if time.time() > deadline: return h, best_c
                perm = np.random.permutation(n)
                for i in perm:
                    if time.time() > deadline: return h, best_c
                    lo = max(0, i-radius); hi = min(n, i+radius+1)
                    for j in range(lo, hi):
                        if i == j: continue
                        for d in (st, -st):
                            if 0.0 <= h[i]+d <= 1.0 and 0.0 <= h[j]-d <= 1.0:
                                h[i] += d; h[j] -= d
                                cval = compute_c5(h, n)
                                if cval < best_c - 1e-12: best_c = cval; improved = True
                                else: h[i] -= d; h[j] += d
    return h, best_c
