def project_onto_simplex_box(h, n):
    """Project h onto {x : 0 <= x <= 1, sum(x) = n/2}"""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Bisection on lambda: x_i = clip(h_i - lambda, 0, 1)
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        x = np.clip(h - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - (lo + hi) / 2, 0.0, 1.0)
    return x

def compute_gradient(h, n, alpha):
    """Compute gradient of logsumexp objective."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    # Softmax weights
    m = c.max()
    w = np.exp(alpha * (c - m))
    w /= w.sum()
    
    # Gradient of c[k] w.r.t. h[i]
    # c[k] = sum_j h[j] * (1 - h[j + (k-n+1)])
    # d c[k] / d h[i] = (1 - h[i + s_k]) - h[i - s_k]
    
    grad = np.zeros(n)
    for k in range(2*n-1):
        s = k - (n-1)
        # (1 - h[i+s]) for valid i: i+s in [0,n-1] => i in [-s, n-1-s]
        # h[i-s] for valid i: i-s in [0,n-1] => i in [s, n-1+s]
        if s >= 0:
            # i from 0 to n-1-s: (1-h[i+s])
            grad[:n-s] += w[k] * (1 - h[s:])
            # i from s to n-1: -h[i-s]
            grad[s:] -= w[k] * h[:n-s]
        else:
            s2 = -s
            # (1-h[i-s2]) = (1-h[i+s2]) ... wait
            # s is negative, so i+s < i
            # (1-h[i+s]) valid when i+s >= 0, i.e., i >= -s = s2
            # So for i from s2 to n-1: (1-h[i-s2])
            grad[s2:] += w[k] * (1 - h[:n-s2])
            # -h[i-s] = -h[i+s2] valid when i-s < n, i.e., i < n+s = n-s2
            # So for i from 0 to n-1-s2: -h[i+s2]
            grad[:n-s2] -= w[k] * h[s2:]
    
    grad *= dx
    return grad
