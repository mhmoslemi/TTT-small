                dr[n - 1 - m] -= (1 - h[p - m])
            for m in range(1, n - p):
                dr[n - 1 - m] += h[p + m]
            base = c + dr
            for q_idx in range(len(zeros)):
                if deadline and time.time() > delay_deadline(deadline):
                    break
                q = zeros[q_idx]
                da = np.zeros(2 * n - 1)
                for d in range(n - q):
                    da[d + n - 1] += (1 - h[q + d])
                for d in range(1, q + 1):
                    da[d + n - 1] -= h[q - d]
                for m in range(1, q + 1):
                    da[n - 1 - m] += (1 - h[q - m])
                for m in range(1, n - q):
                    da[n - 1 - m] -= h[q + m]
                new_max = (base + da).max()
                if new_max < best_c - 1e-12:
                    h[p] = 0
                    h[q] = 1
                    c = c + dr + da
                    best_c = new_max
                    best_h = h.copy()
                    ones[p_idx] = q
                    np.sort(ones)
                    zeros = np.delete(zeros, q_idx)
                    improved = True
                    break
        if not improved:
            break
    return best_h.astype(float), best_c * (2.0 / n)


def delay_deadline(deadline):
    return deadline if deadline else float('inf')


def local_search(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            if time.time() > deadline:
                break
            improved = False
            c = compute_all_lags(h, n)
            k_star = int(np.argmax(c))
            lag = k_star - (n - 1)
            if lag == 0:
                break
            m = abs(lag)
            for i in range(n - m):
                if time.time() > deadline:
                    break
                j = i + m
                if h[i] > h[j] + 1e-10:
                    delta = min(d, (h[i] - h[j]) * 0.5, h[i], 1.0 - h[j])
                    if delta < 1e-10:
                        continue
                    h[i] -= delta
                    h[j] += delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        improved = True
                    else:
                        h[i] += delta
                        h[j] -= delta
                elif h[j] > h[i] + 1e-10:
                    delta = min(d, (h[j] - h[i]) * 0.5, h[j], 1.0 - h[i])
                    if delta < 1e-10:
                        continue
                    h[i] += delta
                    h[j] -= delta
                    c_new = compute_c5(h, n)
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        improved = True
                    else:
                        h[i] -= delta
                        h[j] += delta
    return h, best_c


def optimize_pipeline(h_start, n, deadline):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    t = time.time()
    if t < deadline - 60:
        h_new = slsqp_smooth(best_h, n, alpha=10000, maxiter=3000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c, best_h = c_new, h_new
    if time.time() < deadline - 50:
        for top_m in [3, 5, 8]:
            if time.time() > deadline - 30:
                break
            h_new = active_set_slsqp(best_h, n, top_m, alpha=50000, maxiter=3000)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c, best_h = c_new, h_new
    if time.time() < deadline - 40:
        h_bin, c_bin = binary_fast_search(best_h, n, max_sweeps=100,
                                          deadline=deadline - 20)
        c_bin_check = compute_c5(h_bin, n)
        if c_bin_check < best_c:
            best_c, best_h = c_bin_check, h_bin
            h_new = slsqp_smooth(best_h, n, alpha=100000, maxiter=3000)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c, best_h = c_new, h_new
    if time.time() < deadline - 20:
        h_new = slsqp_smooth(best_h, n, alpha=200000, maxiter=4000, ftol=1e-16)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c, best_h = c_new, h_new
    if time.time() < deadline - 5:
        best_h, best_c = local_search(best_h, n, deadline - 1)
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    best_h = None
    best_c = float('inf')
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 300
    if h_init is not None:
        alloc = min(400, (deadline - t0) * 0.4)
        h_opt, c_opt = optimize_pipeline(h_init, n_init, t0 + alloc)
        best_h, best_c = h_opt, c_opt
    n_candidates = [200, 250, 400, 500]
    for n_try in n_candidates:
        if time.time() > deadline - 80:
            break
        if best_h is not None and n_try == len(best_h):
            continue
        remaining = deadline - time.time() - 50
        if remaining < 60:
            break
        alloc = min(120, remaining * 0.3)
        if best_h is not None:
            x_best = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_best, best_h)
        elif h_init is not None:
            x_init = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_init, h_init)
        else:
            h_start = np.full(n_try, 0.5)
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_pipeline(h_start, n_try, time.time() + alloc)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt
    if best_h is not None and time.time() < deadline - 60:
        n = len(best_h)
        remaining = deadline - time.time() - 30
        n_restarts = max(1, int(remaining / 60))
        for r in range(n_restarts):
            if time.time() > deadline - 30:
                break
            scale = 0.02 * (1.0 + 0.3 * r)
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(55, (deadline - time.time() - 25) * 0.4)
            if alloc < 20:
                break
            h_opt, c_opt = optimize_pipeline(hp, n, time.time() + alloc)
            if c_opt < best_c:
                best_c, best_h = c_opt, h_opt
    if best_h is not None and time.time() < deadline - 30:
        n = len(best_h)
        h_new = slsqp_smooth(best_h, n, alpha=500000, maxiter=5000, ftol=1e-16)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c, best_h = c_new, h_new
        best_h, best_c = local_search(best_h, n, deadline - 2)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    return best_h, best_c, len(best_h)
