    return float(np.max(c) * dx)

def compute_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full').astype(np.float64)

def swap_delta(h, i, j, n):
    """Compute the change in correlation vector when swapping h[i]:1->0 and h[j]:0->1."""
    # For each lag k (from -(n-1) to n-1), compute delta
    # Using the formula derived above
    lags = np.arange(-(n-1), n)  # length 2n-1
    # delta_c[lag] = h[i+lag] - h[j+lag] - 1 + h[i-lag] - h[j-lag]
    # where out-of-bounds = 0
    
    def safe_get(arr, idx):
        mask = (idx >= 0) & (idx < n)
        out = np.zeros_like(idx, dtype=float)
        out[mask] = arr[idx[mask]]
        return out
    
    idx_ip = i + lags
    idx_jp = j + lags
    idx_im = i - lags
    idx_jm = j - lags
    
    delta = (safe_get(h, idx_ip) - safe_get(h, idx_jp) - 1 + safe_get(h, idx_im) - safe_get(h, idx_jm))
    return delta

def binary_sa(h_init, n, deadline, seed=42):
    """Simulated annealing on binary vectors."""
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    
    # Binarize
    h = (h_init >= 0.5).astype(float)
    # Ensure exactly n/2 ones
    n_ones = int(h.sum())
    if n_ones > n // 2:
        idx = np.where(h == 1)[0]
        for i in rng.choice(idx, n_ones - n//2, replace=False):
            h[i] = 0
    elif n_ones < n // 2:
        idx = np.where(h == 0)[0]
        for i in rng.choice(idx, n//2 - n_ones, replace=False):
            h[i] = 1
    
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    cur_max = corr.max()
    best_h = h.copy()
    best_max = cur_max
    
    T = 5.0  # initial temperature (in correlation units)
    cool = 0.9997
    no_improve = 0
    
    while time.time() < deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        # Perform swap
        h[i] = 0
        h[j] = 1
        new_corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)

            corr = new_corr
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
                no_improve = 0

            h[j] = 0

def binarize(h, n, rng):
    """Convert to binary with exactly n/2 ones."""
    if (h == 1).sum() == n // 2 and (h == 0).sum() == n // 2:
        return h.astype(float).copy()
    h = h.copy()
    # Sort by value, take top n/2
    h_order = np.argsort(h)[::-1]

    result[h_order[:n//2]] = 1.0
