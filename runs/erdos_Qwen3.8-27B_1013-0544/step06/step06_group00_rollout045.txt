def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]

def lbbfgs_smooth(h, n, alpha, maxiter, pen):
    dx = 2.0 / n
    target = n / 2.0
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha + pen * (x.sum() - target) ** 2
    bounds = [(0.0, 1.0)] * n
    try:

                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14})

    h_lp, c_lp = lp_refine(best_h, n, t5, max_iter=100, tol=1e-7)
    if c_lp < best_c:
        best_c, best_h = c_lp, h_lp.copy()

    # Phase 6: Anneal for better base, then smooth
    t6 = t0 + 0.85 * total
    ha, ca = anneal_binary(best_h, n, t6, seed=seed + 1, n_iters=1500000)
    if ca < best_c:
        best_c, best_h = ca, ha.copy()
    h = best_h.copy()
    for alpha in [1e6, 1e8, 1e10]:
        if time.time() > t6 - 3:
            break
        h = lbbfgs_smooth(h, n, alpha, 3000, 1e4)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 7: Try other resolutions
    t7 = t0 + 0.93 * total
    for nh in [300, 500]:
        if time.time() > t7 - 8:
            break
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, nh)
        hh = project_feasible(np.interp(xn, xo, best_h), nh)
        for alpha in [1e5, 1e7, 1e9, 1e10]:
            if time.time() > t7 - 6:
                break
            hh = lbbfgs_smooth(hh, nh, alpha, 3000, 1e4)
            ch = compute_c5(hh, nh)
            if ch < best_c:
                best_c, best_h, n = ch, hh.copy(), nh

    # Phase 8: Final noise restart polish
    t8 = dl - 2
    rng = np.random.RandomState(seed)
    if time.time() < t8:
        for trial in range(4):
            if time.time() > t8 - 1:
                break
            h = best_h + 0.002 * rng.randn(n)
            h = project_feasible(h, n)
            for alpha in [1e9, 1e11]:
                if time.time() > t8 - 0.5:
                    break
                h = lbbfgs_smooth(h, n, alpha, 5000, 1e4)
                c = compute_c5(h, n)
                if c < best_c:
                    best_c, best_h = c, h.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
