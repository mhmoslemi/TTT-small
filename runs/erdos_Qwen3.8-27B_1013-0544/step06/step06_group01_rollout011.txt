def find_opt_delta(c, slopes, lo, hi, penalty, sum_h, n):
    """Find δ that minimizes max_k(c[k] + δ*s[k]) + penalty*(sum_h + δ - n/2)²"""
    for _ in range(60):
        mid = (lo + hi) / 2
        vals = c + mid * slopes
        max_val = vals.max()
        active = np.abs(vals - max_val) < 1e-9
        min_s = slopes[active].min()
        max_s = slopes[active].max()
        quad_grad = 2 * penalty * (sum_h + mid - n / 2.0)
        # subgradient = [min_s, max_s] + {quad_grad}
        # 0 in subgradient iff min_s + quad_grad <= 0 <= max_s + quad_grad
        if min_s + quad_grad <= 0 <= max_s + quad_grad:
            break
        elif min_s + quad_grad > 0:
            hi = mid  # function is increasing, go left
        else:
            lo = mid  # function is decreasing, go right
    return (lo + hi) / 2

def compute_slopes_vec(h, n, p):
    m = 2 * n - 1
    N1 = n - 1
    slopes = np.zeros(m)
    
    # Case 1: j ∈ [max(p, N1-p), N1+p]
    lo1 = max(p, N1 - p)
    hi1 = min(N1 + p, m - 1)  # N1+p ≤ 2n-2 = m-1 always since p ≤ n-1
    if lo1 <= hi1:
        js = np.arange(lo1, hi1 + 1)
        idx = p + js - N1  # ranges from p+lo1-N1 to p+hi1-N1
        slopes[lo1:hi1+1] += -h[idx]
    
    # Case 2: j ∈ [max(p, (p+N1)/2), p+N1]
    lo2 = max(p, int(np.ceil((p + N1) / 2.0)))
    hi2 = min(p + N1, m - 1)
    if lo2 <= hi2:
        js = np.arange(lo2, hi2 + 1)
        idx = p - js + N1  # ranges from p-hi2+N1 to p-lo2+N1
        slopes[lo2:hi2+1] += 1.0 - h[idx]
    
    return slopes

def exact_cd(h, c, n, penalty=10.0, max_passes=200, deadline=None):
    m = 2 * n - 1
    for _pass in range(max_passes):
        if deadline and time.time() > deadline:
            break
        improved = False
        for p in range(n):
            slopes = compute_slopes_vec(h, n, p)
            lo = -h[p]
            hi = 1.0 - h[p]
            if hi - lo < 1e-14:
                continue
            delta = find_opt_delta(c, slopes, lo, hi, penalty, h.sum(), n)
            if abs(delta) > 1e-10:
                c += delta * slopes
                h[p] = np.clip(h[p] + delta, 0.0, 1.0)
                improved = True
        if not improved:
            break
    return h, c

```python
def find_opt_pair_delta(c, S, lo, hi):
    for _ in range(50):
        mid = (lo + hi) / 2
        vals = c + mid * S
        max_val = vals.max()
        active = vals >= max_val - 1e-12
        min_s = S[active].min()
        max_s = S[active].max()
        if min_s <= 0 <= max_s:
            break
        elif min_s > 0:
            hi = mid
        else:
            lo = mid
    return (lo + hi) / 2
```
