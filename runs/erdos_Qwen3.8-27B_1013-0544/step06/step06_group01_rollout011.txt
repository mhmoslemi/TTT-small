For the Frank-Wolfe method:
- At each iteration, compute the subgradient g of C5 at current h
- Solve: min_{v: 0<=v<=1, sum(v)=n/2} g·v

  - (If n/2 is not integer, one entry gets a fractional value)
- Update: h ← (1-γ)h + γv, where γ is the step size (backtracking or gamma = 2/(t+2))
- Project to feasible set

This is O(n log n) per iteration for the sort, and we can do many iterations.

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = np.where(h < 1.0, 1.0 - h, 0.0)
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def fw_subgradient(h, n, tol=1e-8):
    """Compute subgradient of max_k c[k] w.r.t. h."""
    c = np.correlate(h, 1.0 - h, mode='full')
    maxc = c.max()
    active = np.where(c >= maxc - tol)[0]
    n_act = len(active)
    if n_act == 0:
        return np.zeros(n)
    grad = np.zeros(n)
    for k in active:
        L = k - (n - 1)
        if L >= 0:
            # Contrib 1: p in [L, n-1], add (1-h[p-L])
            p_lo, p_hi = L, n
            grad[p_lo:p_hi] += (1.0 - h[p_lo - L:p_hi - L])
            # Contrib 2: p in [0, n-L-1], subtract h[p+L]
            p_lo2, p_hi2 = 0, n - L
            if p_hi2 > p_lo2:
                grad[p_lo2:p_hi2] -= h[p_lo2 + L:p_hi2 + L]
        else:
            # Contrib 1: p in [0, n-1+L], add (1-h[p-L])
            p_lo, p_hi = 0, n + L
            if p_hi > p_lo:
                grad[p_lo:p_hi] += (1.0 - h[p_lo - L:p_hi - L])
            # Contrib 2: p in [-L, n-1], subtract h[p+L]
            p_lo2, p_hi2 = -L, n
            grad[p_lo2:p_hi2] -= h[p_lo2 + L:p_hi2 + L]
    return grad / n_act

def frank_wolfe(h, n, max_iter=5000, deadline=None):

    best_c = compute_c5(h, n)
    target = n / 2.0
    for t in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        g = fw_subgradient(h, n, tol=1e-6)
        # Solve: min g·v s.t. 0<=v<=1, sum(v)=target
        # v[i]=1 for smallest g[i], v[i]=0 for largest, one fractional
        order = np.argsort(g)
        v = np.zeros(n)
        v[order[:int(target)]] = 1.0
        frac = target - int(target)
        if frac > 1e-12:
            v[order[int(target)]] = frac
        gamma = 2.0 / (t + 2)
        h_new = (1 - gamma) * h + gamma * v
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            h = h_new
        else:
            # Try smaller step
            gamma2 = gamma / 2
            h_new2 = (1 - gamma2) * h + gamma2 * v
            h_new2 = project_feasible(h_new2, n)
            c_new2 = compute_c5(h_new2, n)
            if c_new2 < best_c:
                best_c = c_new2
                h = h_new2
            else:
                if gamma > 0.01:
                    continue
                break
    return h, best_c

    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    target = n / 2.0
    eta = eta0
    stall = 0
    for t in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        g = fw_subgrad(h, n, topk=3)
        g_mean = g.mean()
        log_h = np.log(np.maximum(h, 1e-12))
        log_h -= eta * (g - g_mean)
        log_h -= log_h.max()
        h_new = np.exp(log_h)
        s = h_new.sum()
        if s < 1e-15:
            break
        h_new = h_new / s * target
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            h = h_new
            stall = 0
            eta *= 0.99
        else:
            stall += 1
            eta *= 0.95
            if stall > 300:
                break
    return h, best_c

def coord_desc(h, n, deadline, max_passes=400, grid=40):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    best_c = float(np.max(c))
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for p in range(n):
            d_lo, d_hi = -h[p], 1.0 - h[p]
            if d_hi - d_lo < 1e-14:
                continue
            k_lo = max(0, min(p, n - 1 - p))
            k_hi = min(2 * n - 2, max(p + n - 1, 2 * n - 2 - p))
            um = 0.0
            if k_lo > 0:
                um = max(um, float(np.max(c[:k_lo])))
            if k_hi < 2 * n - 2:
                um = max(um, float(np.max(c[k_hi + 1:])))
            ks = np.arange(k_lo, k_hi + 1)
            s = np.zeros(len(ks))
            for idx in range(len(ks)):
                k = ks[idx]
                L = k - (n - 1)
                if L >= 0:
                    if L <= p <= n - 1:
                        s[idx] += 1.0 - h[p - L]
                    if 0 <= p <= n - L - 1:
                        s[idx] -= h[p + L]
                else:
                    if 0 <= p <= n - 1 + L:
                        s[idx] += 1.0 - h[p - L]
                    if -L <= p <= n - 1:
                        s[idx] -= h[p + L]
            deltas = np.linspace(d_lo, d_hi, grid)
            best_d = 0.0
            best_val = max(um, best_c)
            for d in deltas:
                if abs(d) < 1e-14:
                    continue
                new_max = max(um, float(np.max(c[ks] + d * s)))
                if k_lo <= n - 1 <= k_hi:
                    nc = c[n - 1] + d * (1.0 - 2.0 * h[p]) - d * d
                    new_max = max(new_max, nc)
                if new_max < best_val:
                    best_val = new_max
                    best_d = d
            if abs(best_d) > 1e-12:
                d = best_d
                for idx in range(len(ks)):
                    k = ks[idx]
                    L = k - (n - 1)
                    if L >= 0:
                        if L <= p <= n - 1:
                            c[k] += d * (1.0 - h[p - L])
                        if 0 <= p <= n - L - 1:
                            c[k] -= d * h[p + L]
                    else:
                        if 0 <= p <= n - 1 + L:
                            c[k] += d * (1.0 - h[p - L])
                        if -L <= p <= n - 1:
                            c[k] -= d * h[p + L]
                c[n - 1] -= d * d
                h[p] = np.clip(h[p] + d, 0.0, 1.0)
                best_c = max(um, float(np.max(c[ks])), c[n - 1])
                improved = True
        if not improved:
            break
    return h, best_c * (2.0 / n)

def slsqp_solve(h, n, alpha=100000, maxiter=2000):
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            return float((m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha) * (2.0 / n))
        return float((m + np.log(np.sum(np.exp(sh))) / alpha) * (2.0 / n))
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def optimize_full(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)
    remaining = deadline - time.time()
    # Phase 1: Frank-Wolfe
    t1 = time.time() + min(remaining * 0.3, 200)
    h_fw, c_fw = frank_wolfe(h, n, max_iter=15000, deadline=t1)
    if c_fw < best_c:
        h, best_c = h_fw, c_fw
    # Phase 2: Mirror descent
    if time.time() < deadline - 100:
        t2 = time.time() + min((deadline - time.time()) * 0.2, 100)
        h_md, c_md = mirror_descent(h, n, max_iter=8000, deadline=t2)
        if c_md < best_c:
            h, best_c = h_md, c_md
    # Phase 3: Coordinate descent
    if time.time() < deadline - 60:
        t3 = min(deadline - time.time() - 40, 150)
        h_cd, c_cd = coord_desc(h, n, deadline=time.time() + t3, max_passes=500, grid=50)
        h_cd = project_feasible(h_cd, n)
        c_cd = compute_c5(h_cd, n)
        if c_cd < best_c:
            h, best_c = h_cd, c_cd
    # Phase 4: SLSQP polish
    if time.time() < deadline - 30:
        for a in [100000, 500000]:
            if time.time() > deadline - 20:
                break
            h_s = slsqp_solve(h, n, alpha=a, maxiter=2000)
            c_s = compute_c5(h_s, n)
            if c_s < best_c:
                h, best_c = h_s, c_s
    # Phase 5: Final CD + FW
    if time.time() < deadline - 10:
        h_cd2, c_cd2 = coord_desc(h, n, deadline=time.time() + 30, max_passes=200, grid=60)
        h_cd2 = project_feasible(h_cd2, n)
        c_cd2 = compute_c5(h_cd2, n)
        if c_cd2 < best_c:
            h, best_c = h_cd2, c_cd2
        h_fw2, c_fw2 = frank_wolfe(h, n, max_iter=3000, deadline=deadline - 2)
        if c_fw2 < best_c:
            h, best_c = h_fw2, c_fw2
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n_init = 200
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    # Main optimization on initial n
    alloc1 = min(300, (deadline - t0) * 0.35)
    h_o, c_o = optimize_full(h_init, n_init, t0 + alloc1, rng)
    if c_o < best_c:
        best_c, best_h = c_o, h_o
    # Try other n values
    for n_try in [150, 200, 250]:
        if time.time() > deadline - 150:
            break
        if n_try == n_init:
            continue
        x0 = np.linspace(0, 2, n_init, endpoint=False)
        x1 = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x1, x0, h_init, period=2.0)
        h_interp = project_feasible(h_interp, n_try)
        alloc = min(150, (deadline - time.time()) * 0.2)
        if alloc < 50:
            break
        h_o, c_o = optimize_full(h_interp, n_try, time.time() + alloc, rng)
        if c_o < best_c:
            best_c, best_h = c_o, h_o
    # Multi-restart with structured patterns
    n = len(best_h)
    if time.time() < deadline - 80:
        remaining = deadline - time.time() - 40
        n_restart = max(3, int(remaining / 70))
        for r in range(n_restart):
            if time.time() > deadline - 50:
                break
            if r == 0:
                hp = project_feasible(best_h + rng.normal(0, 0.03, n), n)
            elif r == 1:
                x = np.linspace(0, 2, n, endpoint=False)
                hp = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x / 2.0)
                hp = project_feasible(hp, n)
            elif r == 2:
                x = np.linspace(0, 2, n, endpoint=False)
                hp = 0.5 + 0.2 * np.sin(2 * np.pi * 3 * x / 2.0)
                hp = project_feasible(hp, n)
            else:
                scale = 0.05 + 0.05 * (r / max(1, n_restart - 1))
                hp = project_feasible(best_h + rng.normal(0, scale, n), n)
            alloc = min(65, (deadline - time.time() - 30) * 0.35)
            if alloc < 30:
                break
            h_o, c_o = optimize_full(hp, n, time.time() + alloc, rng)
            if c_o < best_c:
                best_c, best_h = c_o, h_o
    # Final polish
    if time.time() < deadline - 20:
        n = len(best_h)
        h_f = slsqp_solve(best_h, n, alpha=1000000, maxiter=3000)
        c_f = compute_c5(h_f, n)
        if c_f < best_c:
            best_c, best_h = c_f, h_f
    if time.time() < deadline - 5:
        n = len(best_h)
        h_cd, c_cd = coord_desc(best_h, n, deadline=deadline - 2, max_passes=200, grid=80)
        h_cd = project_feasible(h_cd, n)
        c_cd = compute_c5(h_cd, n)
        if c_cd < best_c:
            best_c, best_h = c_cd, h_cd
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
