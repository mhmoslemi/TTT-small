```python
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    cur_max = corr.max()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    rng = np.random.RandomState(42)
    
    for it in range(max_iters):
        if time.time() > deadline:

        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Randomly try some swaps
        found = False
        for attempt in range(20):
            i = rng.choice(zeros)  # 0 → 1
            j = rng.choice(ones)   # 1 → 0

                ones = np.where(h == 1)[0]

                break

                h[j] = 1.0

def simulated_annealing(h, n, t_start, t_end, n_steps, deadline, seed=42):

    rng = np.random.RandomState(seed)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    cur_max = corr.max()
    best_max = cur_max
    best_h = h.copy()

    zeros = list(np.where(h == 0)[0])
```

```python
def compute_swap_delta(h, n, i, j):
    """Compute delta in correlation for swap (i: 0→1, j: 1→0).
    Returns array of length 2n-1 (the delta in the correlate output)."""
    delta = np.zeros(2*n-1)
    
    # h_new at i-k, i+k, j-k, j+k
    def h_new_at(pos):
        if pos == i:
            return 1.0
        if pos == j:
            return 0.0
        if 0 <= pos < n:
            return h[pos]
        return None  # out of range
    
    for k in range(-(n-1), n):
        m = n - 1 + k
        val = 0.0
        # Term 1: (1 - h_new[i-k]) if i-k in [0, n-1]
        pos = i - k
        if 0 <= pos < n:
            val += (1.0 - h_new_at(pos))
        # Term 2: -h_new[i+k] if i+k in [0, n-1]
        pos = i + k
        if 0 <= pos < n:
            val += -h_new_at(pos)
        # Term 3: -(1 - h_new[j-k]) if j-k in [0, n-1]
        pos = j - k
        if 0 <= pos < n:
            val += -(1.0 - h_new_at(pos))
        # Term 4: +h_new[j+k] if j+k in [0, n-1]
        pos = j + k
        if 0 <= pos < n:
            val += h_new_at(pos)
        
        delta[m] = val
    
    return delta
```
