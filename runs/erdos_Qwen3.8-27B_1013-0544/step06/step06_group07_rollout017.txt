def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def round_to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    return out

def get_delta_zeros(h, n, m_range):
    zeros = np.where(h == 0)[0]
    nz = len(zeros)
    dz = np.zeros((nz, 2 * n - 1))
    for idx, i in enumerate(zeros):
        im = i + m_range
        mask1 = (im >= 0) & (im < n)
        dz[idx, mask1] -= h[im[mask1]]
        im2 = i - m_range
        mask2 = (im2 >= 0) & (im2 < n)
        dz[idx, mask2] += 1.0 - h[im2[mask2]]
    return zeros, dz

def get_delta_ones(h, n, m_range):
    ones = np.where(h == 1)[0]
    no = len(ones)
    do = np.zeros((no, 2 * n - 1))
    for idx, j in enumerate(ones):
        jm = j + m_range
        mask1 = (jm >= 0) & (jm < n)
        do[idx, mask1] += h[jm[mask1]]
        jm2 = j - m_range
        mask2 = (jm2 >= 0) & (jm2 < n)
        do[idx, mask2] -= 1.0 - h[jm2[mask2]]
    return ones, do

def steepest_descent_binary(h, n, m_range, deadline, max_sweeps=100):
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max()
    best_c = cur_c
    best_h = h.copy()
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        zeros, dz = get_delta_zeros(h, n, m_range)
        ones, do = get_delta_ones(h, n, m_range)
        
        # Find best swap
        all_new = corr[None, None, :] + dz[:, None, :] + do[None, :, :]
        all_maxes = all_new.max(axis=2)
        min_idx = all_maxes.argmin()
        bi, bj = np.unravel_index(min_idx, all_maxes.shape)
        new_max = all_maxes[bi, bj]
        
        if new_max >= cur_c - 1e-10:
            break  # No improvement
        
        # Apply swap
        i = zeros[bi]
        j = ones[bj]
        h[i] = 1.0
        h[j] = 0.0
        corr = corr + dz[bi] + do[bj]
        cur_c = new_max
        
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    
    return best_h, best_c * (2.0 / n)

def ils_binary(h_init, n, m_range, deadline, max_iters=500):

def continuous_optimize(h_init, n, deadline):
    """Use SLSQP to optimize the continuous version."""
    dx = 2.0 / n
    target = n / 2.0
    
    def objective(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        return m
    
    def smooth_objective(x, alpha=1000):
        c = np.correlate(x, 1 - x, mode='full') * dx
        shifted = alpha * (c - c.max())
        sm = shifted.max()
        return c.max() + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    
    x0 = project_feasible(h_init + 0.05 * np.random.RandomState(0).randn(n), n)
    
    for alpha in [100, 500, 2000, 10000, 50000]:
        if time.time() > deadline:
            break
        try:
            res = minimize(smooth_objective, x0, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 500, 'ftol': 1e-15, 'alpha': alpha})
            x0 = project_feasible(res.x, n)
        except Exception:
            break
    
    # Final direct optimization
    if time.time() < deadline - 5:
        try:
            res = minimize(objective, x0, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 1000, 'ftol': 1e-15})
            x0 = project_feasible(res.x, n)
        except Exception:
            pass
    
    return x0, compute_c5(x0, n)

            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def round_to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    return out

def get_deltas(h, n, m_range):
    """Precompute delta arrays for all zero and one positions."""
    zeros = np.where(h == 0)[0]
    ones = np.where(h == 1)[0]
    nz, no = len(zeros), len(ones)
    dz = np.zeros((nz, 2 * n - 1))
    do = np.zeros((no, 2 * n - 1))
    for idx, i in enumerate(zeros):
        im = i + m_range
        m1 = (im >= 0) & (im < n)
        dz[idx, m1] -= h[im[m1]]
        im2 = i - m_range
        m2 = (im2 >= 0) & (im2 < n)
        dz[idx, m2] += 1.0 - h[im2[m2]]
    for idx, j in enumerate(ones):
        jm = j + m_range
        m1 = (jm >= 0) & (jm < n)
        do[idx, m1] += h[jm[m1]]
        jm2 = j - m_range
        m2 = (jm2 >= 0) & (jm2 < n)
        do[idx, m2] -= 1.0 - h[jm2[m2]]
    return zeros, dz, ones, do

def steepest_descent(h, n, m_range, deadline, max_sweeps=50):
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_max = corr.max()
    best_max = cur_max
    best_h = h.copy()
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        zeros, dz, ones, do = get_deltas(h, n, m_range)
        all_maxes = (corr[None, None, :] + dz[:, None, :] + do[None, :, :]).max(axis=2)
        min_idx = all_maxes.argmin()
        bi, bj = np.unravel_index(min_idx, all_maxes.shape)
        new_max = all_maxes[bi, bj]
        if new_max >= cur_max - 1e-10:
            break
        i, j = zeros[bi], ones[bj]
        h[i], h[j] = 1.0, 0.0
        corr += dz[bi] + do[bj]
        cur_max = new_max
        if cur_max < best_max:
            best_max = cur_max
            best_h = h.copy()
    dx = 2.0 / n
    return best_h, best_max * dx, corr

def ils_binary(h_init, n, m_range, deadline, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h_init.copy()
    corr_init = np.correlate(h_init, 1 - h_init, mode='full').astype(float)
    best_c = corr_init.max() * (2.0 / n)
    dx = 2.0 / n
    
    cur_h = h_init.copy()
    iteration = 0
    no_improve = 0
    
    while time.time() < deadline:
        # Local search
        cur_h, cur_c, _ = steepest_descent(cur_h, n, m_range, deadline, max_sweeps=30)
        if cur_c < best_c - 1e-10:
            best_c = cur_c
            best_h = cur_h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 20:
            break
        
        # Perturb: random swaps (more aggressive if stuck)
        npert = 3 + no_improve * 2 + rng.randint(5)
        for _ in range(min(npert, 30)):
            zeros = np.where(cur_h == 0)[0]
            ones = np.where(cur_h == 1)[0]
            if len(zeros) == 0 or len(ones) == 0:
                break
            i = zeros[rng.randint(len(zeros))]
            j = ones[rng.randint(len(ones))]
            cur_h[i], cur_h[j] = 1.0, 0.0
        iteration += 1
    
    return best_h, best_c
