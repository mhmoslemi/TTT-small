from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(h, n, alpha, mu=0.0):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    if mu > 0:
        val += mu * (h.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(h, n, alpha, mu=0.0):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)

        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:

            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])

        return g

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha

def smooth_max_grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-14:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def obj_pen(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m_v = c.max()
    sh = alpha * (c - m_v)
    mx = sh.max()
    val = m_v + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def grad_pen(x, n, alpha, mu):
    g = smooth_max_grad(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def _c_eq(x, n):
    return float(x.sum() - n / 2.0)

def _c_jac(x, n):
    return np.ones(n)

def run_lbfgs(h, n, alpha, mu, maxiter=300):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_pen, h, jac=grad_pen, args=(n, alpha, mu),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-17,
                                'gtol': 1e-14, 'maxfun': maxiter * 25, 'maxls': 80})
        hn = project_feasible(res.x, n)
        return hn, compute_c5(hn, n)
    except Exception:
        return h, compute_c5(h, n)

def run_slsqp(h, n, alpha, maxiter=500):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _c_eq, 'jac': _c_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_max_val, h, jac=lambda x: smooth_max_grad(x, n, alpha),
                       args=(n, alpha), method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        hn = project_feasible(res.x, n)
        return hn, compute_c5(hn, n)
    except Exception:
        return h, compute_c5(h, n)

def subgrad_method(h, n, deadline, lr_init=0.02, alpha=50000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = lr_init
    for it in range(50000):
        if time.time() > deadline:
            break
        g = smooth_max_grad(h, n, alpha)
        g = g - g.mean()  # center for sum constraint
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g = g / gn
        h_new = project_feasible(h - lr * g, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
            h = h_new
        else:
            lr *= 0.95
            if lr < 1e-14:
                break
    return best_h, best_c

def fast_local_search(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(777)
    steps = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    for step in steps:
        no_improve = 0
        while time.time() < deadline:
            improved = False
            for _ in range(2000):
                if time.time() > deadline:
                    return best_h, best_c
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                d = step if rng.rand() < 0.5 else -step
                if best_h[i] + d < 0 or best_h[i] + d > 1:
                    continue
                if best_h[j] - d < 0 or best_h[j] - d > 1:
                    continue
                best_h[i] += d
                best_h[j] -= d
                c = compute_c5(best_h, n)
                if c < best_c - 1e-16:
                    best_c = c
                    improved = True
                else:
                    best_h[i] -= d
                    best_h[j] += d
                if improved:
                    no_improve = 0
            if not improved:
                no_improve += 1
                if no_improve >= 2:
                    break
    return best_h, best_c

def systematic_pair_scan(h, n, deadline, step=0.005):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for rad in [n, n // 2, n // 4, 10]:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return best_h, best_c
                lo = max(0, i - rad)
                hi = min(n, i + rad + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    for d in [step, -step]:
                        if best_h[i] + d < 0 or best_h[i] + d > 1:
                            continue
                        if best_h[j] - d < 0 or best_h[j] - d > 1:
                            continue
                        best_h[i] += d
                        best_h[j] -= d
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            best_h[i] -= d
                            best_h[j] += d
            if not improved:
                break
    return best_h, best_c

def perturb_resoptimize(h, n, deadline, rng, n_kicks=4):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    per_kick = max(40, (deadline - time.time()) / (n_kicks + 0.5))
    scales = [0.15, 0.10, 0.05, 0.03, 0.02]
    for kick in range(n_kicks):
        if time.time() > deadline:
            break
        scale = scales[kick % len(scales)]
        hp = h + scale * rng.randn(n)
        hp = project_feasible(hp, n)
        t_sub = time.time() + per_kick
        for alpha in [50000, 500000]:
            if time.time() > t_sub - 5:
                break
            ho, co = run_lbfgs(hp, n, alpha, 1e6, maxiter=150)
            if time.time() < t_sub - 10:
                ho, co = run_slsqp(ho, n, alpha, maxiter=300)
            if co < best_c:
                best_c = co
                best_h = ho
                h = ho
                break
    return best_h, best_c
