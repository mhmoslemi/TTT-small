```python
import numpy as np
import time
import math
from scipy.optimize import minimize


def full_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx


def c5(h, n):
    return float(np.max(full_lags(h, n)))


def project_feasible(p, n):
    target = 0.5 * float(n)
    p = np.asarray(p, dtype=float)
    lo, hi = -10.0, 10.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(p + mid, 0.0, 1.0)) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(p + 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(100):
        diff = target - float(np.sum(h))
        if abs(diff) < 1e-14:
            break
        if diff > 0.0:
            slack = 1.0 - h
            tot = float(np.sum(slack))
            if tot < 1e-15:
                break
            h += (diff / tot) * slack
        else:
            tot = float(np.sum(h))
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)


def low_noise(p, n, rng, scale):
    x = np.asarray(p, dtype=float).copy()
    i = np.arange(n, dtype=float)
    for freq, amp in ((1, scale), (2, 0.5 * scale), (3, 0.3 * scale)):
        phase = 2.0 * math.pi * rng.rand()
        x = x + amp * np.cos((2.0 * math.pi * float(freq) / float(n)) * i + phase)
    return project_feasible(np.clip(x, 0.0, 1.0), n)


def top_active(c, n, k):
    m = len(c)
    if k <= 0:
        return []
    if k >= m:
        idx = np.argsort(c)[::-1]
    else:
        idx = np.argpartition(c, -k)[-k:]
        idx = idx[np.argsort(c[idx])[::-1]]
    act = []
    for ii in idx:
        lag = int(ii) - (n - 1)
        if lag < 0:
            act.append((-1, -lag))
        elif lag == 0:
            act.append((0, 0))
        else:
            act.append((1, lag))
    return act


def act_vals(h, n, act):
    out = np.empty(len(act), dtype=float)
    for k, (s, r) in enumerate(act):
        if r == 0:
            out[k] = float(np.dot(h, 1.0 - h))
        else:
            nr = n - r
            a = h[:nr]
            b = h[r:]
            ab = float(np.dot(a, b))
            if s == 1:
                out[k] = float(np.sum(b)) - ab
            else:
                out[k] = float(np.sum(a)) - ab
    return out


def act_smooth(h, n, act, beta):
    c = act_vals(h, n, act)
    k = len(c)
    m = float(np.max(c))
    sh = beta * (c - m)
    mx = float(np.max(sh))
    e = np.exp(sh - mx)
    z = float(np.sum(e))
    if z <= 0.0:
        w = np.full(k, 1.0 / max(1, k), dtype=float)
        return float(np.mean(c)), w
    w = e / z
    return float(m + (mx + math.log(z)) / beta), w


def act_grad(h, n, act, w):
    g = np.zeros(n, dtype=float)
    for (s, r), wt in zip(act, w):
        if wt <= 1e-18:
            continue
        if r == 0:
            g += wt * (1.0 - 2.0 * h)
        else:
            nr = n - r
            a = h[:nr]
            b = h[r:]
            if s == 1:
                g[:nr] -= wt * b
                g[r:] += wt
                g[r:] -= wt * a
            else:
                g[:nr] += wt
                g[:nr] -= wt * b
                g[r:] -= wt * a
    return g


def pen_obj(x, n, act, beta, mu):
    val, _ = act_smooth(x, n, act, beta)
    s = float(np.sum(x) - 0.5 * n)
    return val + 0.5 * mu * s * s


def pen_jac(x, n, act, beta, mu):
    _, w = act_smooth(x, n, act, beta)
    g = act_grad(x, n, act, w)
    s = float(np.sum(x) - 0.5 * n)
    return g + (mu * s) * np.ones(n, dtype=float)


def lbfgs_step(h, n, act, beta, mu, maxiter):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            pen_obj, h, args=(n, act, beta, mu),
            jac=pen_jac, method="L-BFGS-B", bounds=bounds,
            options={"maxiter": maxiter, "ftol": 1e-12, "maxcor": 10}
        )
        if res.x is not None:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()


def lbfgs_rounds(h, n, end, k=60, betas=(20.0, 100.0, 400.0), mu=20.0, maxiter=120, rounds=6):
    h = project_feasible(h.copy(), n)
    best = c5(h, n)
    for _ in range(rounds):
        if time.time() > end:
            break
        act = top_active(full_lags(h, n), n, k)
        if not act:
            break
        changed = False
        for beta in betas:
            if time.time() > end:
                break
            h2 = lbfgs_step(h, n, act, beta, mu, maxiter)
            c2 = c5(h2, n)
            if c2 < best - 1e-13:
                h, best, changed = h2, c2, True
        if not changed:
            break
    return h, best


def grad_polish(h, n, end, k=60, beta=300.0, maxit=400):
    h = project_feasible(h.copy(), n)
    best = c5(h, n)
    step = 0.05
    vel = np.zeros(n, dtype=float)
    mom = 0.75
    for _ in range(maxit):
        if time.time() > end:
            break
        act = top_active(full_lags(h, n), n, k)
        if not act:
            break
        _, w = act_smooth(h, n, act, beta)
        g = act_grad(h, n, act, w)
        d = -(g - float(np.mean(g)))
        sc = float(np.max(np.abs(d)))
        if sc < 1e-13:
            break
        d /= sc
        vel = (1.0 - mom) * d + mom * vel
        vsc = float(np.max(np.abs(vel)))
        if vsc < 1e-13:
            break
        vel /= vsc
        ok = False
        for f in (step, 0.5 * step, 0.25 * step, 0.125 * step, 0.0625 * step):
            hn = project_feasible(h + f * vel, n)
            cn = c5(hn, n)
            if cn < best - 1e-14:
                h, best = hn, cn
                step = min(0.25, 1.3 * f)
                ok = True
                break
        if not ok:
            step *= 0.45
            if step < 1e-5:
                break
    return h, best


def pair_polish(h, n, end, k=50, beta=300.0, rounds=8, pairs=30, scan=9, rng=None):
    h = project_feasible(h.copy(), n)
    best = c5(h, n)
    for _ in range(rounds):
        if time.time() > end:
            break
        act = top_active(full_lags(h, n), n, k)
        if not act:
            break
        _, w = act_smooth(h, n, act, beta)
        g = act_grad(h, n, act, w)
        order = np.argsort(g)
        lows = order[:16]
        highs = order[-16:]
        cand = []
        for ii in lows:
            ii = int(ii)
            if h[ii] >= 1.0:
                continue
            for jj in highs:
                jj = int(jj)
                if ii == jj or h[jj] <= 0.0:
                    continue
                cand.append((float(g[ii] - g[jj]), ii, jj))
        if rng is not None:
            for _ in range(10):
                ii = int(rng.randint(n))
                jj = int(rng.randint(n))
                if ii != jj and h[ii] < 1.0 and h[jj] > 0.0:
                    cand.append((float(g[ii] - g[jj]), ii, jj))
        cand.sort()
        changed = False
        for _, ii, jj in cand[:pairs]:
            if time.time() > end:
                break
            lo = -float(h[jj])
            hi = 1.0 - float(h[ii])
            if hi - lo < 1e-11:
                continue
            best_delta = 0.0
            local = best + 1.0
            for delta in np.linspace(lo, hi, scan):
                delta = float(delta)
                if abs(delta) < 1e-12:
                    continue
                if delta < lo - 1e-12 or delta > hi + 1e-12:
                    continue
                hp = h.copy()
                hp[ii] += delta
                hp[jj] -= delta
                if (hp[ii] < -1e-12 or hp[ii] > 1.0 + 1e-12 or
                        hp[jj] < -1e-12 or hp[jj] > 1.0 + 1e-12):
                    continue
                cc = c5(hp, n)
                if cc < local:
                    local, best_delta = cc, delta
            if best_delta != 0.0 and local < best - 1e-13:
                h[ii] += best_delta
                h[jj] -= best_delta
                h = project_feasible(h, n)
                best = c5(h, n)
                changed = True
        if not changed:
            break
    return h, best


def optimize_candidate(h, n, end, rng):
    h = project_feasible(h.copy(), n)
    best = c5(h, n)
    rem = end - time.time()
    if rem > 20.0:
        t1 = min(end - 20.0, time.time() + 0.45 * rem)
        if time.time() < t1:
            h2, c2 = lbfgs_rounds(h, n, t1, k=60, betas=(20.0, 100.0, 400.0),
                                  mu=20.0, maxiter=110, rounds=5)
            if c2 < best:
                h, best = h2, c2
    if time.time() < end - 6.0:
        h2, c2 = grad_polish(h, n, end - 3.0, k=60, beta=300.0, maxit=250)
        if c2 < best:
            h, best = h2, c2
    if time.time() < end - 2.0:
        h2, c2 = pair_polish(h, n, end - 1.0, k=50, beta=300.0,
                             rounds=6, pairs=25, scan=9, rng=rng)
        if c2 < best:
            h, best = h2, c2
    h = project_feasible(h, n)
    return h, c5(h, n)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 10.0
    rng = np.random.RandomState(seed)

    try:
        hp = np.array(initial_h_values, dtype=float).copy()
        nn = len(hp)
    except Exception:
        nn = 300
        hp = np.full(nn, 0.5, dtype=float)

    hp = project_feasible(hp, nn)
    pc = c5(hp, nn)
    cands = [(pc, -1, hp, nn)]

    nlist = []
    for n in (nn, 240, 300, 360, 400):
        n = int(n)
        if n >= 60 and n % 2 == 0 and n not in nlist:
            nlist.append(n)
    if not nlist:
        nlist = [300]

    grid_end = min(deadline - 120.0, t0 + 55.0)
    if time.time() < grid_end:
        for n in nlist:
            if time.time() > grid_end:
                break
            i = np.arange(n, dtype=float)
            phis = np.linspace(0.0, 2.0 * math.pi, 16, endpoint=False)
            for a in (0.20, 0.30, 0.35, 0.40, 0.45, 0.50):
                for ph in phis:
                    if time.time() > grid_end:
                        break
                    h = 0.5 + a * np.cos((2.0 * math.pi / float(n)) * i + float(ph))
                    h = project_feasible(np.clip(h, 0.0, 1.0), n)
                    cands.append((c5(h, n), len(cands), h, n))
            base = np.full(n, 0.5, dtype=float)
            for _ in range(8):
                if time.time() > grid_end:
                    break
                h = low_noise(base, n, rng, 0.15)
                cands.append((c5(h, n), len(cands), h, n))

    cands.sort()
    sel = []
    for item in cands:
        cval, idx, hval, nval = item
        if idx == -1:
            sel.append(item)
        elif len(sel) < 10:
            cnt = 0
            for it2 in sel:
                if it2[3] == nval:
                    cnt += 1
            if cnt < 3:
                sel.append(item)
    sel.sort()

    best_h = hp.copy()
    best_c = pc
    best_n = nn
    if len(cands) > 0 and cands[0][0] < best_c - 1e-15:
        best_h = cands[0][2].copy()
        best_c = float(cands[0][0])
        best_n = int(cands[0][3])

    optimized = 0
    for c0, idx0, h0, n0 in sel:
        if time.time() > deadline - 80.0:
            break
        rem = deadline - time.time() - 50.0
        if rem < 45.0:
            break
        if c0 > best_c + 0.02 and optimized >= 2:
            break
        alloc = max(40.0, min(210.0, rem * 0.35))
        end = time.time() + alloc
        try:
            h1, c1 = optimize_candidate(h0, n0, end, rng)
        except Exception:
            continue
        optimized += 1
        if c1 < best_c - 1e-14:
            best_h, best_c, best_n = h1.copy(), float(c1), int(n0)

    restarts = 0
    while time.time() < deadline - 90.0 and restarts < 6:
        rem = deadline - time.time() - 70.0
        if rem < 50.0:
            break
        scale = 0.005 * (1.0 + 0.6 * float(restarts))
        try:
            h0 = low_noise(best_h, best_n, rng, scale)
            alloc = max(35.0, min(120.0, rem * 0.40))
            h1, c1 = optimize_candidate(h0, best_n, time.time() + alloc, rng)
        except Exception:
            restarts += 1
            continue
        if c1 < best_c - 1e-14:
            best_h, best_c = h1.copy(), float(c1)
        restarts += 1

    if time.time() < deadline - 4.0:
        try:
            h2, c2 = grad_polish(best_h, best_n, deadline - 2.0, k=60, beta=300.0, maxit=200)
            if c2 < best_c - 1e-14:
                best_h, best_c = h2, float(c2)
        except Exception:
            pass

    if time.time() < deadline - 1.0:
        try:
            h2, c2 = pair_polish(best_h, best_n, deadline - 0.5, k=50, beta=300.0,
                                 rounds=4, pairs=20, scan=9, rng=rng)
            if c2 < best_c - 1e-14:
                best_h, best_c = h2, float(c2)
        except Exception:
            pass

    best_h = project_feasible(best_h, best_n)
    best_c = c5(best_h, best_n)
    return best_h, best_c, int(best_n)
```
