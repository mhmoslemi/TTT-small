from scipy.optimize import linprog

def solve_lp(h, n):
    y = 1.0 - h
    # Build the constraint matrix
    # For each shift s in [-(n-1), ..., n-1]:
    #   sum_m h[m] * y[m-s] <= t
    #   i.e., sum_m y[m-s] * h[m] + (-1)*t <= 0
    # 
    # c vector for linprog: we want to minimize t, so c = [0,0,...,0,1]
    
    num_shifts = 2*n - 1
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0  # minimize t
    
    A_ub = np.zeros((num_shifts, n + 1))
    b_ub = np.zeros(num_shifts)
    
    for s_idx, s in enumerate(range(-(n-1), n)):
        # shift s: sum_m h[m] * y[m-s] where 0<=m<n and 0<=m-s<n
        for m in range(max(0, s), min(n, n+s)):

        A_ub[s_idx, n] = -1.0  # coefficient of t
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n/2.0])

                  bounds=bounds, method='highs')

        return res.x[:n]
    return h

min_h max_s c_s(h) where c_s(h) = sum_m h[m]*(1-h[m-s])

This is a min-max of quadratic functions (in h). It's non-convex.

The alternating LP method solves:
min_h max_s sum_m h[m]*y[m-s] where y = 1-h_old

This is a relaxed version where y is fixed. The solution h_new minimizes the max of linear functions of h, which is convex. But the true objective at h_new is different because y should be 1-h_new, not 1-h_old.

So the method is like a linearization approach. It should converge to a stationary point of the original problem.

The objective f(h) = max_s c_s(h) has subgradient:
∂f/∂h = sum_{s active} ∂c_s/∂h / |S_active|

where S_active = {s : c_s(h) = max_s' c_{s'}(h)}

And ∂c_s/∂h[i] = (1-h[i-s]) + (-h[i+s]) for valid indices (from the two terms where h[i] appears).

Wait, let me be more careful:
c_s(h) = sum_m h[m] * (1-h[m-s])

∂c_s/∂h[i] = (1-h[i-s]) [from the term m=i in h[m]] + sum_{m: m-s=i} h[m] * (-1) [from the term (1-h[m-s]) where m-s=i, i.e., m=i+s]

import numpy as np
from scipy.optimize import linprog
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def full_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _apply_single_vec(c, h, i, d, n):
    lo1 = i
    hi1 = min(2*n-2, i+n-1)
    idxs = np.arange(lo1, hi1+1)
    c[lo1:hi1+1] += d * (1.0 - h[n-1+i-idxs])
    lo2 = max(0, n-1-i)
    hi2 = min(2*n-2, 2*n-2-i)
    idxs2 = np.arange(lo2, hi2+1)
    c[lo2:hi2+1] += -d * h[i+idxs2-(n-1)]

def sa_optimize(h, n, deadline, rng, t0=0.02):
    h = h.copy()
    c = full_corr(h, n)
    current_max = float(np.max(c))
    best_max = current_max
    best_h = h.copy()
    t_start = time.time()
    total_time = max(1, deadline - t_start)
    
    while time.time() < deadline:
        elapsed = time.time() - t_start
        progress = min(1.0, elapsed / total_time)
        T = t0 * np.exp(-4.0 * progress)
        T = max(T, 1e-5)
        
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d_max_pos = min(1.0 - h[i], h[j])
        d_max_neg = min(h[i], 1.0 - h[j])
        if d_max_pos < 1e-12 and d_max_neg < 1e-12:
            continue
        d = rng.uniform(-d_max_neg, d_max_pos)
        if abs(d) < 1e-12:
            continue
        
        c_old = c.copy()
        _apply_single_vec(c, h, i, d, n)
        _apply_single_vec(c, h, j, -d, n)
        h[i] += d
        h[j] -= d
        
        new_max = float(np.max(c))
        
        if new_max <= current_max:
            current_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
        elif rng.random() < np.exp(-(new_max - current_max) / T):
            current_max = new_max
        else:
            c[:] = c_old
            h[i] -= d
            h[j] += d
    
    return best_h, best_max * (2.0/n)

def solve_lp_step(h, n):
    y = 1.0 - h
    num_shifts = 2 * n - 1
    c_vec = np.zeros(n + 1)
    c_vec[n] = 1.0
    A_ub = np.zeros((num_shifts, n + 1))
    for s_idx in range(num_shifts):
        s = s_idx - (n - 1)
        m_lo = max(0, s)
        m_hi = min(n - 1, n - 1 + s)
        m_range = np.arange(m_lo, m_hi + 1)
        A_ub[s_idx, m_range] = y[m_range - s]
        A_ub[s_idx, n] = -1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    bounds = [(0, 1)] * n + [(None, None)]
    try:
        res = linprog(c_vec, A_ub=A_ub, b_ub=np.zeros(num_shifts),
                      A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            return project_feasible(res.x[:n], n), res.x[n]
    except Exception:
        pass
    return None, None

def alternating_lp(h, n, max_iters=30, deadline=None):
    h = h.copy()
    best_c = compute_c5(h, n)
    alpha = 1.0
    for it in range(max_iters):
        if deadline is not None and time.time() > deadline:
            break
        new_h, t_val = solve_lp_step(h, n)
        if new_h is None:
            break
        if alpha < 1.0:
            new_h = (1 - alpha) * h + alpha * new_h
        new_h = project_feasible(new_h, n)
        c_val = compute_c5(new_h, n)
        if c_val < best_c - 1e-10:
            best_c = c_val
            h = new_h
            alpha = min(1.0, alpha * 1.5)
        else:
            alpha *= 0.5
            if alpha < 0.01:
                break
    return h, best_c

def coordinate_descent(h, n, max_passes=100, deadline=None):
    h = h.copy()
    c = full_corr(h, n)
    current_max = float(np.max(c))
    for _pass in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for p in range(n):
            if h[p] == 0.0 and c[n-1+p] <= current_max:
                continue
            k_lo = max(0, p)
            k_hi = min(2*n-2, p+n-1)
            # Find best new value for h[p]
            # The affected correlation entries are in [k_lo, k_hi] (union of both terms)
            lo = min(p, max(0, n-1-p))
            hi = max(p+n-1, min(2*n-2, 2*n-2-p))
            # Binary search on d in [-h[p], 1-h[p]]
            d_lo = -h[p] + 1e-10
            d_hi = 1.0 - h[p] - 1e-10
            if d_hi - d_lo < 1e-12:
                continue
            for _ in range(30):
                d_mid = (d_lo + d_hi) / 2
                c_save = c.copy()
                _apply_single_vec(c, h, p, d_mid, n)
                h[p] += d_mid
                new_max = float(np.max(c))
                c[:] = c_save
                h[p] -= d_mid
                if new_max < current_max:
                    d_lo = d_mid
                else:
                    d_hi = d_mid
            best_d = (d_lo + d_hi) / 2
            if abs(best_d) > 1e-10:
                _apply_single_vec(c, h, p, best_d, n)
                h[p] += best_d
                new_max = float(np.max(c))
                if new_max < current_max:
                    current_max = new_max
                    improved = True
        if not improved:
            break
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

Wait, I have a bug in the coordinate_descent function. The binary search logic is wrong. Let me fix it.

Actually, the issue is that the function new_max(d) = max(c + d * s) is not necessarily unimodal. It's the maximum of a set of linear functions of d, so it IS convex in d! (max of linear functions is convex). So the sublevel set {d : new_max(d) <= current_max} is an interval, and we can do binary search on the boundaries.

Actually wait - I want to find d that MINIMIZES new_max(d). Since new_max is convex in d, the minimum is unique (or on an interval). The derivative from the left is negative and from the right is positive at the minimum. So I can do:
- If new_max(d_mid) < new_max(d_mid - eps), the minimum is to the right → d_lo = d_mid
- Otherwise, d_hi = d_mid

But this is the wrong direction. If new_max is convex and we're at d_mid:
- If the minimum is to the left, then new_max is decreasing at d_mid, so new_max(d_mid) > new_max(d_mid - small)
- If the minimum is to the right, then new_max is increasing at d_mid, so new_max(d_mid) > new_max(d_mid + small)

Actually, for a convex function, I should check if the slope is positive or negative. But I don't have an explicit formula for the slope.

Alternative: just use ternary search! For a unimodal function on [a, b]:
- Evaluate at a+delta and b-delta... no, ternary search evaluates at two interior points.

For ternary search on [lo, hi]:
- m1 = lo + (hi-lo)/3
- m2 = hi - (hi-lo)/3
- If f(m1) < f(m2): hi = m2
- Else: lo = m1
- Convergence: range shrinks by 2/3 each iteration

After 50 iterations, range = initial_range * (2/3)^50 ≈ initial_range * 1.6e-9. Very precise.

                else:

            best_d = (d_lo + d_hi) / 2.0
