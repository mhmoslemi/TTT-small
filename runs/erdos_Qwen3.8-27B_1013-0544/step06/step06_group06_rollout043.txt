def compute_c5(h, n):

    c = np.correlate(h, 1 - h, mode='full')

def project_sum(h, target):
    """Project h to have sum = target, keeping in [0,1]."""
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        h = np.clip(h, 0, 1)
        if diff > 0:
            room = 1 - h
            if room.sum() < 1e-14:
                break
            h = h + diff * room / room.sum()
        else:
            if h.sum() < 1e-14:
                break
            h = h + diff * h / h.sum()
    return np.clip(h, 0, 1)

def c5_soft_and_grad(h, n, alpha, lam=1000.0):
    """Compute smoothed C5 and gradient including sum penalty."""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    
    # Soft max
    s = alpha * corr
    s_max = s.max()
    e = np.exp(s - s_max)
    Z = e.sum()
    w = e / Z
    c5_smooth = (s_max + np.log(Z)) / alpha * dx
    
    # Gradient of c5_smooth
    lags = np.arange(2*n-1) - (n-1)
    grad = np.ones(n) * dx
    for j in range(2*n-1):
        wj = w[j] * dx
        if wj < 1e-14:
            continue
        lag = lags[j]
        if lag == 0:
            grad -= 2 * wj * h
        elif lag > 0:
            grad[lag:] -= wj * h[:-lag]
            grad[:n-lag] -= wj * h[lag:]
        else:
            nl = -lag
            grad[:n-nl] -= wj * h[nl:]
            grad[nl:] -= wj * h[:n-nl]
    
    # Add penalty gradient: lam * (sum(h) - n/2) * 2
    sdiff = h.sum() - n / 2.0
    grad += 2.0 * lam * sdiff
    
    # Penalty value

def c5_smooth_and_grad(h, n, alpha, lam=0.0):
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    
    # Soft max weights
    s = alpha * corr
    s_max = s.max()
    e = np.exp(s - s_max)
    Z = e.sum()
    w = e / Z
    c5_val = (s_max + np.log(Z)) / alpha * dx
    
    # Gradient
    lags = np.arange(2*n-1) - (n-1)
    grad = np.zeros(n)
    
    for j in range(2*n-1):
        wj = w[j] * dx
        if wj < 1e-14:
            continue
        lag = lags[j]
        if lag == 0:
            # d(output[j])/d(h[p]) = 1 - 2*h[p]
            grad += wj * (1 - 2*h)
        elif lag > 0:
            # (1-h[p-lag]) for p >= lag, minus h[p+lag] for p < n-lag
            grad[lag:] += wj * (1 - h[:-lag])
            grad[:n-lag] -= wj * h[lag:]
        else:
            nl = -lag
            # (1-h[p+nl]) for p < n-nl, minus h[p-nl] for p >= nl
            grad[:n-nl] += wj * (1 - h[nl:])
            grad[nl:] -= wj * h[:-nl]
    
    # Add sum penalty
    if lam > 0:
        sdiff = h.sum() - n/2.0
        grad += 2.0 * lam * sdiff
        c5_val += lam * sdiff * sdiff
    
    return c5_val, grad

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_sum(h, n):
    target = n / 2.0
    h = np.clip(h, 0, 1)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            if room.sum() < 1e-14:
                break
            h = h + diff * room / room.sum()
        else:
            if h.sum() < 1e-14:
                break
            h = h + diff * h / h.sum()
    return np.clip(h, 0, 1)

def c5_smooth_and_grad(h, n, alpha, lam=0.0):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    s = alpha * corr
    s_max = s.max()
    e = np.exp(s - s_max)
    Z = e.sum()
    w = e / Z
    c5_val = (s_max + np.log(Z)) / alpha * dx
    lags = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)
    active = w > 1e-10
    for j in np.where(active)[0]:
        wj = w[j] * dx
        lag = lags[j]
        if lag == 0:
            grad += wj * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += wj * (1.0 - h[:-lag])
            grad[:n - lag] -= wj * h[lag:]
        else:
            nl = -lag
            grad[:n - nl] += wj * (1.0 - h[nl:])
            grad[nl:] -= wj * h[:-nl]
    if lam > 0:
        sdiff = h.sum() - n / 2.0
        grad += 2.0 * lam * sdiff
        c5_val += lam * sdiff * sdiff
    return c5_val, grad

    bounds = [(0.0, 1.0)] * n
    res = minimize(lambda x: c5_smooth_and_grad(x, n, alpha, lam)[0], h,
                   jac=lambda x: c5_smooth_and_grad(x, n, alpha, lam)[1],
                   method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12, 'maxfun': 5000})
    return np.clip(res.x, 0, 1)

    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    lags = np.arange(2 * n - 1) - (n - 1)
    K = min(60, 2 * n - 1)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        top_k = np.argsort(corr)[-K:]
        iz = zeros[:, None]
        wo = lags[None, :]
        il = iz - wo
        v1 = (il >= 0) & (il < n)
        t1 = np.where(v1, 1.0 - h[np.clip(il, 0, n - 1)], 0.0)
        ipl = iz + wo
        v2 = (ipl >= 0) & (ipl < n)
        t2 = np.where(v2, h[np.clip(ipl, 0, n - 1)], 0.0)
        D01 = t1 - t2
        io = ones[:, None]
        jll = io - wo
        v3 = (jll >= 0) & (jll < n)
        t3 = np.where(v3, 1.0 - h[np.clip(jll, 0, n - 1)], 0.0)
        jpl = io + wo
        v4 = (jpl >= 0) & (jpl < n)
        t4 = np.where(v4, h[np.clip(jpl, 0, n - 1)], 0.0)
        D10 = -t3 + t4
        nk = corr[top_k][None, None, :] + D01[:, top_k][:, None, :] + D10[:, top_k][None, :, :]
        new_maxes = nk.max(axis=2)
        bi, bj = np.unravel_index(new_maxes.argmin(), new_maxes.shape)
        if new_maxes[bi, bj] < corr.max() - 1e-10:
            h[zeros[bi]] = 1.0
            h[ones[bj]] = 0.0
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            corr = np.correlate(h, 1 - h, mode='full').astype(float)
            cur_c = corr.max() * dx
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            if np.random.rand() < 0.15 and it < max_iter - 100:
                ri = np.random.randint(len(zeros))
                rj = np.random.randint(len(ones))
                h[zeros[ri]] = 1.0
                h[ones[rj]] = 0.0
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                corr = np.correlate(h, 1 - h, mode='full').astype(float)
                cur_c = corr.max() * dx
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
            else:
                break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    np.random.seed(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except NameError:
        n = 250
        h0 = np.zeros(n)
        h0[:n // 2] = 1.0
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    # Phase 1: L-BFGS-B continuous (35% time)
    p1 = t0 + 0.35 * (budget_s - 3)
    hw = best_h.copy()
    for alpha in [30, 60, 120, 250, 500]:
        if time.time() > p1 - 15:
            break
        hw = lbfgs_run(hw, n, alpha, 3000, 300, p1 - 5)
        hw = project_sum(hw, n)
        c = compute_c5(hw, n)
        if c < best_c:
            best_c = c
            best_h = hw.copy()
    # Phase 2: Binary local search (20% time)
    p2 = t0 + 0.55 * (budget_s - 3)
    if time.time() < p2 - 10:
        hb, cb = binary_local_search(best_h, n, p2 - 5, max_iter=50000)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
    # Phase 3: Multi-start continuous (25% time)
    p3 = t0 + 0.80 * (budget_s - 3)
    if time.time() < p3 - 10:
        starts = [best_h.copy()]
        hm = best_h.copy()
        hm[hm > 0.3] = 1.0
        hm[hm <= 0.3] = 0.0
        s = int(hm.sum())
        if s != n // 2:
            if s > n // 2:
                oi = np.where(hm == 1)[0]
                np.random.shuffle(oi)
                hm[oi[:s - n // 2]] = 0
            else:
                zi = np.where(hm == 0)[0]
                np.random.shuffle(zi)
                hm[zi[:n // 2 - s]] = 1
        starts.append(hm)
        for trial in range(4):
            if time.time() > p3 - 10:
                break
            hp = best_h + 0.08 * np.random.randn(n)
            hp = project_sum(hp, n)
            starts.append(hp)
        for st in starts[:6]:
            if time.time() > p3 - 8:
                break
            hw2 = lbfgs_run(st, n, 100, 3000, 200, p3 - 5)
            hw2 = project_sum(hw2, n)
            hw2 = lbfgs_run(hw2, n, 400, 3000, 200, p3 - 5)
            hw2 = project_sum(hw2, n)
            c = compute_c5(hw2, n)
            if c < best_c:
                best_c = c
                best_h = hw2.copy()
    # Phase 4: Final polish (20% time)
    if time.time() < deadline - 10:
        hw = best_h.copy()
        for alpha in [500, 1000, 2000]:
            if time.time() > deadline - 5:
                break
            hw = lbfgs_run(hw, n, alpha, 5000, 300, deadline - 3)
            hw = project_sum(hw, n)
            c = compute_c5(hw, n)
            if c < best_c:
                best_c = c
                best_h = hw.copy()
    best_h = project_sum(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
