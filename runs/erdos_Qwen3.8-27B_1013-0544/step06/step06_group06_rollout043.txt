```python
def compute_jacobian(h, N):
    """Compute Jacobian J[m,i] = ∂result[m]/∂h[i] for all m in [0,2N-2], i in [0,N-1]."""
    J = np.zeros((2*N-1, N))
    for i in range(N):
        # First term: (1-h[i-m+N-1]) for m where 0 ≤ i-m+N-1 ≤ N-1
        # => N-1-i ≤ m ≤ N-1+i
        m_lo = max(0, N-1-i)
        m_hi = min(2*N-2, N-1+i)
        if m_hi >= m_lo:
            m = np.arange(m_lo, m_hi+1)
            idx = i - m + (N-1)
            J[m, i] += (1.0 - h[idx])
        
        # Second term: -h[i+m-(N-1)] for m where 0 ≤ i+m-(N-1) ≤ N-1
        # => N-1-i ≤ m ≤ 2N-2-i
        m_lo2 = max(0, N-1-i)
        m_hi2 = min(2*N-2, 2*N-2-i)
        if m_hi2 >= m_lo2:
            m = np.arange(m_lo2, m_hi2+1)
            idx = i + m - (N-1)
            J[m, i] -= h[idx]
    
    return J
```

This is O(N²) total, which for N=250 is about 156,000 operations. Very fast.

min t
s.t.: J @ h_new - t ≤ b where b[m] = result[m] - J[m,:] @ h

In the standard form for scipy.optimize.linprog:
minimize: c_vec @ x where x = [h_new (N vars); t (1 var)]
c_vec = [0, ..., 0, 1]

Inequality constraints: A_ub @ x ≤ b_ub
- J[m,:] @ h_new - t ≤ result[m] - J[m,:] @ h for all m
  => A row: [J[m,0], ..., J[m,N-1], -1], b = result[m] - J[m,:] @ h
- h_new[i] ≤ 1: A row: [0,...,1(at i),...,0, 0], b = 1
- -h_new[i] ≤ 0: A row: [0,...,-1(at i),...,0, 0], b = 0

Equality constraints: A_eq @ x = b_eq
- sum(h_new) = N/2: A row: [1, 1, ..., 1, 0], b = N/2

Bounds: (0, 1) for h_new, (-inf, inf) for t

```python
import numpy as np
import time
from scipy.optimize import linprog

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, target):
    n = len(h)
    lo, hi = -1.5, 1.5
    for _ in range(100):
        mid = (lo + hi) / 2
        if np.clip(h - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def compute_jacobian(h, N):
    J = np.zeros((2*N-1, N))
    for i in range(N):
        m_lo = max(0, N-1-i)
        m_hi = min(2*N-2, N-1+i)
        if m_hi >= m_lo:
            m = np.arange(m_lo, m_hi+1)
            J[m, i] += (1.0 - h[i - m + (N-1)])
        m_lo2 = max(0, N-1-i)
        m_hi2 = min(2*N-2, 2*N-2-i)
        if m_hi2 >= m_lo2:
            m = np.arange(m_lo2, m_hi2+1)
            J[m, i] -= h[i + m - (N-1)]
    return J

def sequential_lp(h0, N, deadline, max_iter=200):
    dx = 2.0 / N
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h, N)
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        result = np.correlate(h, 1-h, mode='full')
        J = compute_jacobian(h, N)
        
        # LP: min t s.t. J @ h_new - t ≤ result - J @ h, 0≤h_new≤1, sum(h_new)=N/2
        c_vec = np.zeros(N+1)
        c_vec[N] = 1.0  # minimize t
        
        # Inequality constraints
        A_rows = []
        b_rows = []
        
        # J @ h_new - t ≤ result - J @ h
        b_lp = result - (J @ h)
        A_main = np.hstack([J, -np.ones((2*N-1, 1))])
        A_rows.append(A_main)
        b_rows.append(b_lp)
        
        # h_new ≤ 1
        A_bound1 = np.hstack([np.eye(N), np.zeros((N,1))])
        A_rows.append(A_bound1)
        b_rows.append(np.ones(N))
        
        # -h_new ≤ 0
        A_bound2 = np.hstack([-np.eye(N), np.zeros((N,1))])
        A_rows.append(A_bound2)
        b_rows.append(np.zeros(N))
        
        A_ub = np.vstack(A_rows)
        b_ub = np.concatenate(b_rows)
        
        # Equality: sum(h_new) = N/2
        A_eq = np.zeros((1, N+1))
        A_eq[0, :N] = 1.0
        b_eq = np.array([N/2.0])
        
        bounds = [(0, 1)] * N + [(-np.inf, np.inf)]
        
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        
        if res.status != 0:
            break
        
        h_new = res.x[:N]
        
        # Check improvement
        c_new = compute_c5(h_new, N)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        
        # Damping: if the step is too large, damp it
        step = h_new - h
        step_norm = np.linalg.norm(step)
        if step_norm > 0.1:
            # Try damped step
            for alpha in [0.5, 0.25, 0.1]:
                h_damped = project_feasible(h + alpha * step, N/2.0)
                c_damped = compute_c5(h_damped, N)
                if c_damped < compute_c5(h, N):
                    h = h_damped
                    break
            else:
                h = h_new
        else:
            h = h_new
        h = project_feasible(h, N/2.0)
    
    return best_h, best_c

def pgd_logsumexp(h0, N, deadline, alpha=500):
    """Projected gradient descent with log-sum-exp smoothing."""
    dx = 2.0 / N
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h, N)
    
    # FISTA
    y = h.copy()
    t_k = 1.0
    L = 2.0 * alpha * dx * N  # rough Lipschitz bound
    step = 1.0 / L
    
    for it in range(10000):
        if time.time() > deadline:
            break
        
        result = np.correlate(y, 1-y, mode='full')
        
        # Compute softmax weights
        a = alpha * result * dx
        a_shifted = a - a.max()
        w = np.exp(a_shifted)
        w /= w.sum()
        
        # Gradient: dx * sum_m w[m] * J[m, :]
        J = compute_jacobian(y, N)
        grad = dx * (J.T @ w)
        
        # Gradient step
        z = y - step * grad
        h_new = project_feasible(z, N/2.0)
        
        # FISTA update
        t_new = (1 + np.sqrt(1 + 4*t_k*t_k)) / 2
        y = h_new + ((t_k - 1) / t_new) * (h_new - h)
        y = project_feasible(y, N/2.0)
        t_k = t_new
        h = h_new
        
        c = compute_c5(h, N)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    return best_h, best_c
```

```python
        h_lp = res.x[:N]
        step = h_lp - h
        sn = np.linalg.norm(step)
        cands = [h_lp]
        if sn > 0.03:
            cands.append(h + 0.5 * step)
            cands.append(h + 0.25 * step)
            cands.append(h + 0.1 * step)
        cur_c = compute_c5(h, N)
        best_cand = h.copy()
        best_cand_c = cur_c
        for cand in cands:
            cand = project_feasible(cand, N / 2.0)
            cc = compute_c5(cand, N)
            if cc < best_cand_c - 1e-14:
                best_cand_c = cc
                best_cand = cand.copy()
        h = best_cand
        if best_cand_c < best_c - 1e-14:
            best_c = best_cand_c
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 40:
                h = project_feasible(best_h + 0.03 * rng.randn(N), N / 2.0)
                stall = 0
    return best_h, best_c

def pgd_phase(h0, N, deadline, alpha=500.0, rng_seed=0):
    rng = np.random.RandomState(rng_seed)
    dx = 2.0 / N
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h, N)
    y = h.copy()
    t_k = 1.0
    L = 4.0 * alpha * dx * N
    step = 1.0 / L
    stall = 0
    for it in range(100000):
        if time.time() > deadline:
            break
        result = np.correlate(y, 1 - y, mode='full')
        a = alpha * result * dx
        a_s = a - a.max()
        w = np.exp(a_s)
        w /= w.sum()
        J = compute_jacobian(y, N)
        grad = dx * (J.T @ w)
        z = y - step * grad
        h_new = project_feasible(z, N / 2.0)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = h_new + ((t_k - 1.0) / t_new) * (h_new - h)
        y = project_feasible(y, N / 2.0)
        t_k = t_new
        h = h_new
        c = compute_c5(h, N)
        if c < best_c - 1e-14:
            best_c = c
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 200:
                h = project_feasible(best_h + 0.02 * rng.randn(N), N / 2.0)
                y = h.copy()
                t_k = 1.0
                stall = 0
    return best_h, best_c

def binary_swap_refine(h0, N, deadline, rng_seed=0):
    rng = np.random.RandomState(rng_seed)
    dx = 2.0 / N
    target = N // 2
    idx = np.argsort(h0)[::-1]
    h = np.zeros(N)
    h[idx[:target]] = 1.0
    best_h = h.copy()
    best_c = compute_c5(h, N)
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h, N)
    ones = np.where(h >= 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    for it in range(1000000):
        if time.time() > deadline:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        top_idx = np.argsort(corr)[-5:]
        improved = False
        for m in top_idx:
            for trial in range(3):
                i = rng.randint(len(zeros))
                j = rng.randint(len(ones))
                zi = zeros[i]
                oj = ones[j]
                h[zi], h[oj] = h[oj], h[zi]
                cors = np.correlate(h, 1 - h, mode='full')
                cc = cors.max() * dx
                if cc < best_c - 1e-14:
                    best_c = cc
                    best_h = h.copy()
                    zi2 = np.where(h < 0.5)[0]
                    oj2 = np.where(h >= 0.5)[0]
                    zeros = zi2
                    ones = oj2
                    improved = True
                    break
                else:
                    h[zi], h[oj] = h[oj], h[zi]
        if improved:
            break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        N = len(h0)
    except Exception:
        N = 250
        h0 = np.zeros(N)
        h0[:N // 2] = 1.0
    best_h = h0.copy()
    best_c = compute_c5(best_h, N)

    # Phase 1: Sequential LP (45% time)
    p1 = t0 + 0.45 * (budget_s - 5)
    hb, cb = lp_phase(h0, N, p1, max_iter=8000, rng_seed=seed)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()

    # Phase 2: PGD continuous polish (15% time)
    p2 = t0 + 0.60 * (budget_s - 5)
    for alpha in [300.0, 1000.0, 3000.0]:
        if time.time() > p2 - 10:
            break
        hb, cb = pgd_phase(best_h, N, p2 - 3, alpha=alpha, rng_seed=seed + alpha)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()

    # Phase 3: More LP (25% time)
    p3 = t0 + 0.85 * (budget_s - 5)
    hb, cb = lp_phase(best_h, N, p3, max_iter=5000, rng_seed=seed + 100)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()

    # Phase 4: Final refinement (remaining)
    if time.time() < dl - 30:
        hb, cb = pgd_phase(best_h, N, dl - 20, alpha=5000.0, rng_seed=seed + 200)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
    if time.time() < dl - 5:
        hb, cb = lp_phase(best_h, N, dl - 2, max_iter=2000, rng_seed=seed + 300)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()

    # Final: also try quantized binary version and pick best
    hb_bin, cb_bin = binary_swap_refine(best_h, N, dl - 1, rng_seed=seed + 400)
    if cb_bin < best_c:
        best_c = cb_bin
        best_h = hb_bin.copy()

    best_h = project_feasible(best_h, N / 2.0)
    best_c = compute_c5(best_h, N)
    return best_h, best_c, N
```
