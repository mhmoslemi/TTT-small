def project_feasible(h,n):

        if diff>0:
            room=1.0-h; s=room.sum()
            if s<1e-15: h+=diff/n

            s=h.sum()
            if s<1e-15: h+=diff/n
            else: h += diff/s*h

    diff=target-h.sum()

            idxs=np.where(h>1e-12)[0]

def smooth_grad(x,n,alpha,dx):

    m=c.max(); z=alpha*(c-m); w=np.exp(z); sw=w.sum(); w /= sw

import numpy as np
from scipy.optimize import minimize
import time

CFG={'n':1,'alpha':1.0,'dx':2.0,'lam':100.0,'target':0.5}

def compute_c5(h,n):
    return float(np.max(np.correlate(h,1.0-h))*(2.0/n))

def project_feasible(h,n):
    h=np.array(h,dtype=float).copy()
    if not np.all(np.isfinite(h)): h=np.full(n,0.5)
    h=np.clip(h,0.0,1.0); t=0.5*n
    for _ in range(80):
        d=t-h.sum()
        if abs(d)<1e-13: break
        if d>0:
            r=1.0-h; s=r.sum(); h += d/s*r if s>1e-15 else d/n
        else:
            s=h.sum(); h += d/s*h if s>1e-15 else d/n
        h=np.clip(h,0,1)
    d=t-h.sum()
    if abs(d)>1e-13:
        idx=np.where((h>1e-12)&(h<1-1e-12))[0]
        if idx.size: h[idx]=np.clip(h[idx]+d/idx.size,0,1)
        d=t-h.sum()
        if abs(d)>1e-13:
            i=0 if d>0 else n-1
            if d>0: i=int(np.argmin(h)); h[i]=min(1.0,h[i]+d)
            else: i=int(np.argmax(h)); h[i]=max(0.0,h[i]+d)
    return h

def to_binary(h,n):
    hb=np.zeros(n); idx=np.argsort(h)[::-1]; hb[idx[:n//2]]=1.0; return hb

def _obj(x):
    n=CFG['n']; c=np.correlate(x,1.0-x)*CFG['dx']; m=float(c.max())
    if not np.isfinite(m): return 1e6
    w=np.exp(CFG['alpha']*(c-m)); sw=float(w.sum())
    return m+float(np.log(sw))/CFG['alpha']

def _grad(x):
    n=CFG['n']; alpha=CFG['alpha']; c=np.correlate(x,1.0-x)*CFG['dx']
    if not np.all(np.isfinite(c)): return np.zeros(n)
    m=c.max(); w=np.exp(alpha*(c-m)); sw=float(w.sum()); w/=sw
    L=2*n-1
    p=np.concatenate(([0.0],np.cumsum(w)))
    A=p[n:n+n]-p[:n]
    f=np.convolve(x,w,mode='full'); B=f[n-1:n-1+n]
    f2=np.convolve(x,w[::-1],mode='full'); D=f2[n-1:n-1+n]
    return CFG['dx']*(A-B-D)

def _sum(x): return float(x.sum()-CFG['target'])

def slsqp(h,n,alpha,iters):
    CFG.update(n=n,alpha=float(alpha),dx=2.0/n,target=0.5*n)
    h0=project_feasible(h,n)
    cons=[{'type':'eq','fun':_sum}]
    try:
        r=minimize(_obj,h0,jac=_grad,method='SLSQP',bounds=[(0.0,1.0)]*n,constraints=cons,options={'maxiter':iters,'ftol':1e-12,'disp':False})
        return project_feasible(r.x,n)
    except Exception:
        return h0

def build_idx(n):
    b=np.arange(n); pos=np.empty((n,n),int); neg=np.empty((n,n),int)
    for p in range(n):
        pos[p]=n-1+b-p; neg[p]=n-1+p-b
    return pos,neg

def compute_A(h,n):
    A=np.empty((n,2*n-1)); Lh=None
    for i in range(n):
        k1=np.arange(i-n+1,i+1); A[i,k1+n-1]=1.0-h[i-k1]
        k2=np.arange(-i,n-i); A[i,k2+n-1]-=h[i+k2]
    return A

def upd_A(A,p,pos,neg,d):
    A[:,pos[p]]-=d; A[:,neg[p]]-=d

import numpy as np
from scipy.optimize import minimize
import time

CFG = {'n': 1, 'alpha': 1.0, 'dx': 2.0, 'target': 0.5}

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h)) * (2.0 / n))

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    if not np.all(np.isfinite(h)):
        h = np.full(n, 0.5)
    h = np.clip(h, 0.0, 1.0)
    t = 0.5 * n
    for _ in range(80):
        d = t - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            r = 1.0 - h
            s = r.sum()
            h += d / s * r if s > 1e-15 else d / n
        else:
            s = h.sum()
            h += d / s * h if s > 1e-15 else d / n
        h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-13:
        idx = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if idx.size:
            h[idx] = np.clip(h[idx] + d / idx.size, 0.0, 1.0)
        d = t - h.sum()
        if abs(d) > 1e-13:
            if d > 0:
                i = int(np.argmin(h))
                h[i] = min(1.0, h[i] + d)
            else:
                i = int(np.argmax(h))
                h[i] = max(0.0, h[i] + d)
    return h

def to_binary(h, n):
    hb = np.zeros(n)
    idx = np.argsort(h)[::-1]
    hb[idx[:n // 2]] = 1.0
    return hb

def _obj(x):
    n = CFG['n']
    c = np.correlate(x, 1.0 - x) * CFG['dx']
    m = float(c.max())
    if not np.isfinite(m):
        return 1e6
    w = np.exp(CFG['alpha'] * (c - m))
    sw = float(w.sum())
    return m + float(np.log(sw)) / CFG['alpha']

def _grad(x):
    n = CFG['n']
    alpha = CFG['alpha']
    c = np.correlate(x, 1.0 - x) * CFG['dx']
    if not np.all(np.isfinite(c)):
        return np.zeros(n)
    m = float(c.max())
    w = np.exp(alpha * (c - m))
    sw = float(w.sum())
    w = w / sw
    p = np.concatenate(([0.0], np.cumsum(w)))
    A = p[n:n + n] - p[:n]
    f = np.convolve(x, w, mode='full')
    B = f[n - 1:n - 1 + n]
    f2 = np.convolve(x, w[::-1], mode='full')
    D = f2[n - 1:n - 1 + n]
    return CFG['dx'] * (A - B - D)

def _sum_eq(x):
    return float(x.sum() - CFG['target'])

def slsqp_polish(h, n, alpha, iters):
    CFG['n'] = n
    CFG['alpha'] = float(alpha)
    CFG['dx'] = 2.0 / n
    CFG['target'] = 0.5 * n
    h0 = project_feasible(h, n)
    cons = [{'type': 'eq', 'fun': _sum_eq}]
    try:
        r = minimize(_obj, h0, jac=_grad, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n, constraints=cons,
                     options={'maxiter': iters, 'ftol': 1e-12, 'disp': False})
        return project_feasible(r.x, n)
    except Exception:
        return h0

def build_idx(n):
    b = np.arange(n)
    pos = np.empty((n, n), dtype=int)
    neg = np.empty((n, n), dtype=int)
    for p in range(n):
        pos[p] = n - 1 + b - p
        neg[p] = n - 1 + p - b
    return pos, neg

def compute_A(h, n):
    A = np.empty((n, 2 * n - 1), dtype=float)
    for i in range(n):
        k1 = np.arange(i - n + 1, i + 1)
        A[i, k1 + n - 1] = 1.0 - h[i - k1]
        k2 = np.arange(-i, n - i)
        A[i, k2 + n - 1] -= h[i + k2]
    return A

def upd_A(A, p, pos, neg, d):
    A[:, pos[p]] -= d
    A[:, neg[p]] -= d

def anneal_binary(h, n, A, c, cur, dl, seed, pos, neg):
    rng = np.random.RandomState(seed)
    z = list(np.where(h == 0)[0])
    o = list(np.where(h == 1)[0])
    best_c = cur
    best_h = h.copy()
    T = 3.0
    cool = 0.999985
    center = n - 1
    acc = 0
    for it in range(400000):
        if it % 2000 == 0 and time.time() > dl - 8:
            break
        if not z or not o:
            break
        ai = rng.randint(0, len(z))
        bi = rng.randint(0, len(o))
        i = int(z[ai])
        j = int(o[bi])
        v = c + A[i] - A[j]
        v[center] -= 2.0
        v[center + j - i] += 1.0
        v[center + i - j] += 1.0
        nc = float(v.max())
        accept = False
        if nc <= cur:
            accept = True
        else:
            d = nc - cur
            if T > 1e-4 and d < 20.0 * T and rng.random() < np.exp(-d / T):
                accept = True
        if accept:
            h[i] = 1.0
            h[j] = 0.0
            c = v
            cur = nc
            upd_A(A, i, pos, neg, 1.0)
            upd_A(A, j, pos, neg, -1.0)
            z[ai] = j
            o[bi] = i
            acc += 1
            if acc % 4000 == 0:
                if np.max(np.abs(c - np.rint(c))) < 1e-6:
                    c = np.rint(c)
                    cur = float(c.max())
            if nc < best_c:
                best_c = nc
                best_h = h.copy()
        T *= cool
    return best_h, best_c

def binary_search(h_in, n, dl, seed, target_bound=0.3808):

    c = np.correlate(h, 1.0 - h).astype(float)

    dx = 2.0 / n
    tc = int(np.floor(target_bound / dx + 1e-9))
    best_h = h.copy()
    best_c = cur
    pos, neg = build_idx(n)
    A = compute_A(h, n)
    zero = [int(x) for x in np.where(h == 0)[0]]
    one = [int(x) for x in np.where(h == 1)[0]]

    for _ in range(100):
        if time.time() > dl - 12 or cur <= tc or not one or not zero:
            break
        oa = np.asarray(one, dtype=int)
        m = oa.size
        rows = np.arange(m)
        center = n - 1
        bnc = None
        bi = -1
        bj = -1
        for ii in zero:
            if time.time() > dl - 12:
                break
            i = int(ii)
            B = c + A[i]
            M = B[None, :] - A[oa]
            M[:, center] -= 2.0
            M[rows, center + oa - i] += 1.0
            M[rows, center + i - oa] += 1.0
            rmax = M.max(axis=1)
            loc = int(np.argmin(rmax))
            nc = float(rmax[loc])
            if bnc is None or nc < bnc - 1e-12:
                bnc = nc
                bi = i
                bj = int(oa[loc])
        if bi >= 0 and bnc < cur - 1e-8:
            i = bi
            j = bj
            v = c + A[i] - A[j]
            v[center] -= 2.0
            v[center + j - i] += 1.0
            v[center + i - j] += 1.0
            nc = float(v.max())
            if nc < cur - 1e-8:
                h[i] = 1.0
                h[j] = 0.0
                c = v
                cur = nc
                if np.max(np.abs(c - np.rint(c))) < 1e-6:
                    c = np.rint(c)
                    cur = float(c.max())
                upd_A(A, i, pos, neg, 1.0)
                upd_A(A, j, pos, neg, -1.0)
                try:
                    zero.remove(i)
                    zero.append(j)
                except Exception:
                    pass
                try:
                    one.remove(j)
                    one.append(i)
                except Exception:
                    pass
                if cur < best_c:
                    best_c = cur
                    best_h = h.copy()
        else:
            break

    if time.time() < dl - 20 and cur > tc:
        ah, ac = anneal_binary(h, n, A, c, cur, dl, seed + 101, pos, neg)
        if ac < best_c:
            best_c = ac
            best_h = ah

    if abs(best_c - np.rint(best_c)) < 1e-5:
        best_c = float(np.rint(best_c))
    return best_h, best_c, dx

def polish(h, n, dl, plan):
    h = project_feasible(h, n)
    bh = h.copy()
    bc = compute_c5(h, n)
    for alpha, iters in plan:
        if time.time() > dl - 8:
            break
        out = slsqp_polish(h, n, alpha, iters)
        c = compute_c5(out, n)
        if np.isfinite(c) and c < bc - 1e-13:
            bc = c
            bh = out.copy()
            h = bh
    return bh, bc, n
