def lse_objective_grad(h, n, beta):
    h = np.clip(np.asarray(h,float).ravel(), 0.0,1.0)
    dx=2.0/n
    v=1.0-h
    c=np.correlate(h,v,mode='full')*dx
    c=np.maximum(c,0.0)
    m=float(c.max())
    y=beta*(c-m)
    e=np.exp(np.minimum(y,0.0))
    s=float(e.sum())
    if s <=0 or not np.isfinite(s):
       p=np.full_like(c,1.0/(2*n-1)); F=m

    B=np.convolve(P,h,mode='full')[n-1:2*n-1]
    g=dx*(A-B)
    if not np.all(np.isfinite(g)):
       g=np.zeros_like(h)
    return float(F), g

import numpy as np
from scipy.optimize import minimize
import time

_stop_time = 0.0

def stop_callback(*args):
    return time.time() > _stop_time

def compute_c5(h, n):
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_feasible(a, n):
    a = np.asarray(a, dtype=float).ravel()
    if a.size != n:
        if a.size < n:
            a = np.concatenate([a, np.zeros(max(0, n - a.size))])
        else:
            a = a[:n]
    if not np.all(np.isfinite(a)):
        a = np.full(n, 0.5)
    target = 0.5 * n
    lo = float(np.min(a)) - 1.0
    hi = float(np.max(a)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(a - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(a - 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(30):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s <= 1e-15:
                break
            h += diff * room / s
        else:
            s = h.sum()
            if s <= 1e-15:
                break
            h += diff * h / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] + diff)
    return np.clip(h, 0.0, 1.0)

def top_n_binary(h, n):
    h = np.asarray(h, dtype=float).ravel()
    target = n // 2
    hb = np.zeros(n)
    if target > 0:
        idx = np.argsort(-h)[:target]
        hb[idx] = 1.0
    return hb

def lse_objective_grad(h, n, beta):
    h = np.clip(np.asarray(h, dtype=float).ravel(), 0.0, 1.0)
    dx = 2.0 / float(n)
    v = 1.0 - h
    c = np.correlate(h, v, mode="full") * dx
    c = np.maximum(c, 0.0)
    m = float(c.max())
    y = beta * (c - m)
    e = np.exp(np.minimum(y, 0.0))
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.full_like(c, 1.0 / (2 * n - 1))
        F = m
    else:
        p = e / s
        F = m + np.log(s) / beta
    P = p
    R = P[::-1]
    A = np.convolve(R, v, mode="full")[n - 1:2 * n - 1]
    B = np.convolve(P, h, mode="full")[n - 1:2 * n - 1]
    g = dx * (A - B)
    if not np.all(np.isfinite(g)):
        g = np.zeros_like(h)
    return float(F), g

def lbfgs_objective(x, n, beta, pen):
    h = np.clip(np.asarray(x, dtype=float).ravel(), 0.0, 1.0)
    F, g = lse_objective_grad(h, n, beta)
    dev = float(h.sum() - 0.5 * n)
    if pen != 0.0:
        F += 0.5 * pen * dev * dev
        g = g + pen * dev
    if not np.isfinite(F):
        return 1e100, np.zeros_like(h)
    return float(F), g

def candidate_variants(h, n):
    h = project_feasible(h, n)
    out = [
        h,
        h[::-1],
        1.0 - h[::-1],
        0.5 * (h + h[::-1]),
        0.5 * (h + 1.0 - h[::-1]),
    ]
    if n % 2 == 0:
        out.append(top_n_binary(h, n))
    res = []
    for x in out:
        res.append(project_feasible(x, n))
    return res

def polish_solution(h, n, betas, pen, deadline, stage_time):
    work = project_feasible(h, n)
    best = work.copy()
    bc = compute_c5(best, n)
    bounds = [(0.0, 1.0)] * n
    for beta in betas:
        now = time.time()
        if now > deadline - 8.0:
            break
        stage_dl = min(deadline - 4.0, now + max(2.0, min(stage_time, 0.5 * max(2.0, deadline - now))))
        _stop_time = stage_dl
        options = {"maxiter": 1000000, "maxls": 40, "ftol": 1e-13, "gtol": 1e-8}
        try:
            res = minimize(
                lbfgs_objective, work, method="L-BFGS-B", jac=True,
                bounds=bounds, args=(n, float(beta), float(pen)),
                callback=stop_callback, options=options
            )
            x = res.x if (res is not None and hasattr(res, "x")) else work
        except Exception:
            x = work
        work = project_feasible(x, n)
        cw = compute_c5(work, n)
        if cw < bc:
            bc = cw
            best = work.copy()

            cc = compute_c5(cand, n)
            if cc < bc:
                bc = cc
                best = cand.copy()
        if bc < cw - 1e-12:
            work = best.copy()
    return best, bc

def compute_base_binary(h, n):
    l = np.arange(-(n - 1), n, dtype=int)
    base = np.zeros((n, 2 * n - 1), dtype=float)
    for p in range(n):
        f = p + l
        mf = (f >= 0) & (f < n)
        base[p, mf] = 1.0 - h[f[mf]]
        b = p - l
        mb = (b >= 0) & (b < n)
        base[p, mb] -= h[b[mb]]
    return base

def binary_greedy(h, n, deadline, max_sweeps=30, chunk=2048):
    if n % 2 != 0:
        hb = top_n_binary(h, n)
        return hb, compute_c5(hb, n)
    h = top_n_binary(h, n)
    corr = np.correlate(h, 1.0 - h, mode="full").astype(float)
    M = int(round(float(corr.max())))
    besth = h.copy()
    bestM = M
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        base = compute_base_binary(h, n)
        zeros = np.nonzero(h == 0)[0]
        ones = np.nonzero(h == 1)[0]
        if zeros.size == 0 or ones.size == 0:
            break
        zi = np.repeat(zeros, ones.size)
        oj = np.tile(ones, zeros.size)
        improved = False
        for start in range(0, zi.size, chunk):
            if time.time() > deadline:
                break
            end = min(start + chunk, zi.size)
            i = zi[start:end]
            j = oj[start:end]
            arr = base[i] - base[j]
            arr += corr[None, :]
            arr[:, n - 1] -= 2.0
            d1 = n - 1 + (j - i)
            arr[:, d1] += 1.0
            d2 = n - 1 + (i - j)
            arr[:, d2] += 1.0
            mx = arr.max(axis=1)
            idx = np.nonzero(mx <= M - 1 + 1e-12)[0]
            if idx.size:
                loc = idx[int(np.argmin(mx[idx]))]
                iv = int(i[loc])
                jv = int(j[loc])
                h[iv] = 1.0
                h[jv] = 0.0
                corr = arr[loc].astype(float, copy=True)
                M = int(round(float(corr.max())))
                if M < bestM:
                    bestM = M
                    besth = h.copy()
                improved = True
                break
        if not improved:
            break
    return besth, bestM * (2.0 / float(n))

        if cc < best_c:
            best_c = cc
            best_h = hh.copy()

    if time.time() < end - 60.0:
        for scale in (0.004, 0.012):
            if time.time() < end - 40.0:
                nh = project_feasible(best_h + scale * rng.randn(best_n), best_n)
                hh, cc = polish_solution(nh, best_n, [5000.0, 20000.0], 1e4, end - 30.0, 25.0)
                if cc < best_c:
                    best_c = cc
                    best_h = hh.copy()

    if time.time() < end - 60.0:
        for nnew in (300, 200):
            if time.time() < end - 40.0:
                x0 = np.linspace(0.0, 2.0, best_n)
                x1 = np.linspace(0.0, 2.0, nnew)
                hr = np.interp(x1, x0, best_h)
                hr = project_feasible(hr, nnew)
                cr = compute_c5(hr, nnew)
                if cr < best_c:
                    best_c = cr
                    best_h = hr.copy()
                    best_n = nnew

                hh, cc = polish_solution(hr, nnew, [500.0, 3000.0], 1e4, end - 25.0, 25.0)
                if cc < best_c:
                    best_c = cc
                    best_h = hh.copy()
                    best_n = nnew

                if nnew % 2 == 0 and time.time() < end - 25.0:
                    hb, _ = binary_greedy(hr, nnew, end - 15.0, max_sweeps=8, chunk=1024)
                    cr = compute_c5(hb, nnew)
                    if cr < best_c:
                        best_c = cr
                        best_h = hb.copy()
                        best_n = nnew

    for cand in candidate_variants(best_h, best_n):
        cc = compute_c5(cand, best_n)
        if cc < best_c:
            best_c = cc
            best_h = cand.copy()

    if best_n % 2 == 0:
        tb = top_n_binary(best_h, best_n)
        ct = compute_c5(tb, best_n)
        if ct < best_c:
            best_c = ct
            best_h = tb

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, int(best_n)
