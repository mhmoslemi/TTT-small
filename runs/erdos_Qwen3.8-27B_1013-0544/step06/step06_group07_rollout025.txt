            ones = np.where(h == 1)[0]
            no_improve = 0
            T = 0.03
    return best_h, best_c * dx

def continuous_refine(h_init, n, deadline, seed=42):
    """Log-sum-exp smoothing with L-BFGS-B style iteration."""
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    h = h_init.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()

    for alpha in [500, 2000, 10000, 50000, 200000]:
        if time.time() > deadline:
            break
        def obj(x):
            g = np.correlate(x, 1-x, mode='full') * dx
            m = g.max()
            sh = alpha * (g - m)
            sm = sh.max()
            return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0}]
        try:
            res = minimize(obj, h, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 800, 'ftol': 1e-14})
            h = project_feasible(res.x, n)
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c; best_h = h.copy()
        except Exception:
            pass
    return best_h, best_c

def greedy_polish(h_bin, n, deadline, seed=42):
    """Greedy: repeatedly find the best single swap that reduces max corr."""
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    h = h_bin.copy().astype(float)
    m_range = np.arange(-(n-1), n)
    corr = np.correlate(h, 1-h, mode='full').astype(np.float64)
    cur_c = corr.max()
    best_c = cur_c
    best_h = h.copy()
    zeros = np.where(h == 0)[0]
    ones = np.where(h == 1)[0]

    improved = True
    while improved and time.time() < deadline:
        improved = False
        max_lag = np.argmax(corr)
        offset = max_lag - (n-1)
        # Try targeted swaps first
        valid_k = np.arange(max(0, -offset), min(n, n-offset))
        for _ in range(2000):
            if time.time() > deadline: break
            i_candidates = valid_k[(h[valid_k] == 1) & (h[valid_k + offset] == 0)]
            if len(i_candidates) == 0:
                i_candidates = ones
            j_candidates = valid_k[(h[valid_k] == 0) & (h[valid_k + offset] == 1)]
            if len(j_candidates) == 0:
                j_candidates = zeros
            i = i_candidates[rng.randint(len(i_candidates))]
            j = j_candidates[rng.randint(len(j_candidates))]
            if h[i] != 1 or h[j] != 0: continue
            delta = np.zeros(2*n-1)
            im = i+m_range; msk=(im>=0)&(im<n); delta[msk] += 1.0-h[im[msk]]
            imn = i-m_range; msk=(imn>=0)&(imn<n); delta[msk] -= h[imn[msk]]
            jm = j+m_range; msk=(jm>=0)&(jm<n); delta[msk] -= 1.0-h[jm[msk]]
            jmn = j-m_range; msk=(jmn>=0)&(jmn<n); delta[msk] += h[jmn[msk]]
            delta[n-1] -= 2.0
            k1=n-1+(j-i)
            if 0<=k1<2*n-1: delta[k1]+=1.0
            k2=n-1+(i-j)
            if 0<=k2<2*n-1: delta[k2]+=1.0
            nc = (corr + delta).max()
            if nc < cur_c:
                h[i]=1.0; h[j]=0.0
                corr = corr + delta
                cur_c = nc
                improved = True
                if nc < best_c:
                    best_c = nc; best_h = h.copy()
                zeros = np.where(h==0)[0]
                ones = np.where(h==1)[0]
    return best_h, best_c * dx

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 8
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)

    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: Round to binary, quick SA
    h_bin = np.zeros(n)
    h_bin[np.argsort(h0)[::-1][:n//2]] = 1.0
    p1 = t0 + 0.15 * (budget_s - 8)
    hb, cb = binary_sa(h_bin, n, p1, seed=seed, t_start=0.05, t_end=0.001, patience=10000)
    if cb < best_c:
        best_c = cb; best_h = hb

    # Phase 2: Targeted SA (main workhorse)
    p2 = t0 + 0.55 * (budget_s - 8)
    hb, cb = targeted_sa(best_h, n, p2, seed=seed+10)
    if cb < best_c:
        best_c = cb; best_h = hb

    # Phase 3: More binary SA with different params
    p3 = t0 + 0.72 * (budget_s - 8)
    h_bin2 = np.zeros(n)
    h_bin2[np.argsort(best_h)[::-1][:n//2]] = 1.0
    hb, cb = binary_sa(h_bin2, n, p3, seed=seed+200, t_start=0.03, t_end=0.0005, patience=12000)
    if cb < best_c:
        best_c = cb; best_h = hb

    # Phase 4: Continuous refinement
    p4 = t0 + 0.88 * (budget_s - 8)
    hc, cc = continuous_refine(best_h, n, p4, seed=seed+300)
    if cc < best_c:
        best_c = cc; best_h = hc

    # Phase 5: Final targeted SA from best
    if time.time() < dl - 5:
        h_bin3 = np.zeros(n)
        h_bin3[np.argsort(best_h)[::-1][:n//2]] = 1.0
        hb, cb = targeted_sa(h_bin3, n, dl - 3, seed=seed+400)
        if cb < best_c:
            best_c = cb; best_h = hb

    # Final polish
    if time.time() < dl - 1:
        hg, cg = greedy_polish(best_h, n, dl, seed=seed+500)
        if cg < best_c:
            best_c = cg; best_h = hg

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
