import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_c(h, n):
    """Return the full correlation vector."""
    return np.correlate(h, 1 - h, mode='full').astype(float)

def compute_grad_c(h, n):
    """Compute grad[m, k] = ∂c[k]/∂h[m] for all m, k."""
    grad = np.zeros((n, 2 * n - 1))
    p = np.arange(n)
    for s in range(-(n - 1), n):
        k = s + n - 1
        if s == 0:
            grad[:, k] = 1.0 - 2.0 * h
        elif s > 0:
            idx = p >= s
            grad[idx, k] = 1.0 - h[p[idx] - s]
            idx2 = p < n - s
            grad[idx2, k] -= h[p[idx2] + s]
        else:
            t = -s
            idx = p < n - t
            grad[idx, k] = 1.0 - h[p[idx] + t]
            idx2 = p >= t
            grad[idx2, k] -= h[p[idx2] - t]
    return grad

def adam_optimize(h0, n, deadline, max_iter=50000):
    """Adam optimizer with soft-max objective."""
    dx = 2.0 / n
    h = h0.astype(float).copy()
    alpha = 500.0  # starting soft-max sharpness
    lr = 0.01
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    while time.time() < deadline and t < max_iter:
        t += 1
        # Periodically increase alpha
        alpha = min(5000.0, 500.0 * (1.0 + t / 5000.0))
        
        c = compute_c(h, n)
        logit = alpha * c * dx
        logit_max = logit.max()
        exp_vals = np.exp(logit - logit_max)
        p = exp_vals / exp_vals.sum()
        
        # Compute gradient
        gc = compute_grad_c(h, n)
        grad = (p * dx) @ gc  # shape (n,)
        
        # Add constraint gradient (sum(h) = n/2)
        constraint_violation = h.sum() - n / 2.0
        grad += 2.0 * constraint_violation * np.ones(n) / n
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = np.clip(h, 0.0, 1.0)
        
        # Project to feasible
        h = project_feasible(h, n)
        
        c_val = compute_c5(h, n)
        if c_val < best_c:
            best_c = c_val
            best_h = h.copy()
    
    return best_h, best_c

import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def round_binary(h, n):
