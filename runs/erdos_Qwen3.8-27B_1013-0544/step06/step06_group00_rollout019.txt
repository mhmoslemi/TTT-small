import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj_and_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = c.max()
    shifted = alpha * (c - max_c)
    shifted_max = shifted.max()
    e = np.exp(shifted - shifted_max)
    sum_e = e.sum()
    p = e / sum_e
    obj = max_c + (shifted_max + np.log(sum_e)) / alpha
    
    # Gradient
    shifts = np.arange(-(n-1), n)
    grad = np.zeros(n)
    for j in range(n):
        idx1 = j - shifts
        v1 = (idx1 >= 0) & (idx1 < n)
        idx2 = j + shifts
        v2 = (idx2 >= 0) & (idx2 < n)
        s1 = p[v1].dot(h[idx1[v1]])
        s2 = p[v2].dot(h[idx2[v2]])
        grad[j] = alpha * dx * (1.0 - s1 - s2)
    
    # Project gradient to feasible subspace (preserve sum)
    grad -= grad.mean()
    
    return obj, grad

def lbfgs_optimize(h, n, alpha, maxiter=5000, deadline=None):
    dx = 2.0 / n
    def obj_fn(x):
        val, _ = smooth_obj_and_grad(x, n, alpha)
        return val
    def grad_fn(x):
        _, g = smooth_obj_and_grad(x, n, alpha)
        return g
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_fn, h, jac=grad_fn, method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-10})
        return project_feasible(res.x, n)
    except Exception:
        return h

    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    m_range = np.arange(-(n - 1), n)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False

            best_nc = cur_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def lse_obj_grad(h, n, alpha, active=None):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    n_corr = 2 * n - 1
    if active is None:
        active = np.arange(n_corr)
    c_act = c[active]
    max_c = c_act.max()
    shifted = alpha * (c_act - max_c)
    shifted_max = shifted.max()
    e = np.exp(shifted - shifted_max)
    Z = e.sum()
    p_act = e / Z
    obj = max_c + (shifted_max + np.log(Z)) / alpha
    p_full = np.zeros(n_corr)
    p_full[active] = p_act
    cum_p = np.concatenate([[0.0], np.cumsum(p_full)])
    j_arr = np.arange(n)
    P1 = cum_p[2*n - 1 - j_arr] - cum_p[n - 1 - j_arr]
    corr_hp = np.correlate(h, p_full, 'full')
    B = corr_hp[n - 1:n + n - 1]
    conv_hp = np.convolve(h, p_full, 'full')
    C = conv_hp[n - 1:n + n - 1]
    grad = alpha * dx * (P1 - B - C)
    grad -= grad.mean()
    return obj, grad, c
