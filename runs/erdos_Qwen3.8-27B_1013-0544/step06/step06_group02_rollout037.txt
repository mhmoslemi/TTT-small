def project_feasible(h,n):
    h=np.clip(np.asarray(h,dtype=float).copy(),0.0,1.0)
    target=0.5*n
    for _ in range(200):
        diff=target-h.sum()
        if abs(diff)<1e-13: break
        if diff>0:
            room=np.clip(1.0-h,0.0,1.0)
            tot=room.sum()
            if tot<1e-15: break
            h += diff*room/tot
        else:
            tot=h.sum()
            if tot<1e-15: break
            h += diff*h/tot
        h=np.clip(h,0.0,1.0)
    diff=target-h.sum()
    if abs(diff)>1e-13:
        idxs=np.where((h>1e-12)&(h<1.0-1e-12))[0]
        for idx in idxs:
            diff=target-h.sum()
            if abs(diff)<1e-15: break
            if diff>0:
                step=min(diff,1.0-h[idx])
            else:
                step=max(diff,-h[idx])
            h[idx]+=step
    return h

import time
import math
import numpy as np
from scipy.optimize import minimize

NEG_INF = -1.0e300


def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    for _ in range(300):
        d = target - h.sum()
        if abs(d) < 1.0e-13:
            break
        if d > 0.0:
            r = np.clip(1.0 - h, 0.0, 1.0)
            s = r.sum()
        else:
            r = h
            s = h.sum()
        if s < 1.0e-15:
            break
        h = np.clip(h + d * r / s, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1.0e-13:
        idx = np.flatnonzero((h > 1.0e-12) & (h < 1.0 - 1.0e-12))
        for k in idx:
            d = target - h.sum()
            if abs(d) < 1.0e-15:
                break
            step = min(d, 1.0 - h[k]) if d > 0.0 else max(d, -h[k])
            h[k] += step
            if h[k] < 0.0:
                h[k] = 0.0
            elif h[k] > 1.0:
                h[k] = 1.0
    return h


def default_h(n):
    x = np.linspace(0.0, 2.0, n)
    h = 0.5 + 0.45 * np.cos(2.0 * np.pi * x) + 0.35 * np.cos(5.0 * np.pi * x)
    return project_feasible(h, n)


def compute_C(h, n):
    return np.correlate(h, 1.0 - h, mode="full")


def c5_from_C(C, n):
    return float(np.max(C) * (2.0 / n))


def slope_diff(H, Hrev, n, i, j, sd):
    sd.fill(0.0)
    sd[i:i + n] += 1.0
    sd[j:j + n] -= 1.0
    sd[i:i + n] -= Hrev
    sd[j:j + n] += Hrev
    si = n - 1 - i
    sj = n - 1 - j
    sd[si:si + n] -= H
    sd[sj:sj + n] += H


def eval_pair(cnd, C, sd, p1, p2, cen):
    cnd = np.asarray(cnd, dtype=float)
    vals = cnd[:, None] * sd[None, :] + C[None, :]
    vals[:, p1] = NEG_INF
    vals[:, p2] = NEG_INF
    vals[:, cen] = NEG_INF
    f = vals.max(axis=1)
    d2 = cnd * cnd
    v1 = C[p1] + sd[p1] * cnd + d2
    v2 = C[p2] + sd[p2] * cnd + d2
    vc = C[cen] + sd[cen] * cnd - 2.0 * d2
    f = np.maximum(f, v1)
    f = np.maximum(f, v2)
    f = np.maximum(f, vc)
    return f


def all_pair_search(H, C, n, deadline, mode, params):
    cen = n - 1
    L = 2 * n - 1
    sd = np.empty(L)
    Hrev = H[::-1]

    if mode == 0:
        M = max(3, int(params))
        u = np.linspace(0.0, 1.0, M)
        cand = np.empty(M + 1)
    else:
        fr = tuple(params)
        cand = np.empty(2 * len(fr))

    cur = float(C.max()) if C.size else 0.0
    sweep = 0
    tol = 1.0e-10

    while sweep < 80 and time.time() < deadline:
        improved = 0
        updates = 0
        stop = False

        for i in range(n - 1):
            if stop:
                break
            for j in range(i + 1, n):
                if time.time() > deadline:
                    stop = True
                    break

                hi = H[i]
                hj = H[j]
                lo = max(-hi, hj - 1.0)
                hi2 = min(1.0 - hi, hj)
                if hi2 - lo < 1.0e-14:
                    continue

                slope_diff(H, Hrev, n, i, j, sd)
                p1 = cen + i - j
                p2 = cen + j - i

                if mode == 0:
                    cand[:M] = lo + (hi2 - lo) * u
                    cand[M] = 0.0
                    fv = eval_pair(cand, C, sd, p1, p2, cen)
                else:
                    cnt = 0
                    for frv in fr:
                        v = hi2 * frv
                        if v > 1.0e-14:
                            cand[cnt] = v
                            cnt += 1
                        v = lo * frv
                        if v < -1.0e-14:
                            cand[cnt] = v
                            cnt += 1
                    if cnt == 0:
                        continue
                    fv = eval_pair(cand[:cnt], C, sd, p1, p2, cen)

                k = int(np.argmin(fv))
                bf = float(fv[k])
                d = float(cand[k])

                if bf < cur - tol and abs(d) > 1.0e-14:
                    if d != 1.0:
                        sd *= d
                    C += sd
                    if d != 1.0:
                        sd /= d
                    dd = d * d
                    C[p1] += dd
                    C[p2] += dd
                    C[cen] -= 2.0 * dd

                    H[i] += d
                    H[j] -= d
                    if H[i] < 0.0:
                        H[i] = 0.0
                    elif H[i] > 1.0:
                        H[i] = 1.0
                    if H[j] < 0.0:
                        H[j] = 0.0
                    elif H[j] > 1.0:
                        H[j] = 1.0

                    cur = bf
                    improved += 1
                    updates += 1
                    if updates % 2000 == 0:
                        C = compute_C(H, n)
                        cur = float(C.max())

        sweep += 1
        if improved == 0:
            break
        C = compute_C(H, n)
        cur = float(C.max())

    return H, C, cur


def grad_weights(H, n, w):
    G = np.zeros(n)
    cen = n - 1
    for p, wm in enumerate(w):
        if wm < 1.0e-13:
            continue
        m = p - cen
        ab = abs(m)
        lg = n - ab
        st = m if m > 0 else 0
        off = -m if m < 0 else 0
        G[st:st + lg] += wm * (1.0 - H[off:off + lg])
        st2 = -m if m < 0 else 0
        off2 = m if m > 0 else 0
        G[st2:st2 + lg] -= wm * H[off2:off2 + lg]
    return G


def smooth_obj(x, n, alpha, mu):
    C = np.correlate(x, 1.0 - x, mode="full")
    m = float(C.max())
    s = alpha * (C - m)
    mx = float(s.max())
    val = m + (mx + math.log(float(np.sum(np.exp(s - mx))))) / alpha
    return val + mu * (float(x.sum()) - 0.5 * n) ** 2


def smooth_grad(x, n, alpha, mu):
    C = np.correlate(x, 1.0 - x, mode="full")
    m = float(C.max())
    s = alpha * (C - m)
    mx = float(s.max())
    w = np.exp(s - mx)
    w /= w.sum()
    G = grad_weights(x, n, w)
    G += 2.0 * mu * (float(x.sum()) - 0.5 * n)
    return G


def lbfgs_refine(H, n, deadline):
    H0 = project_feasible(H, n)
    C0 = compute_C(H0, n)
    best = c5_from_C(C0, n)
    Hw = H0.copy()
    bounds = [(0.0, 1.0)] * n
    configs = ((50.0, 1.0), (300.0, 10.0), (2000.0, 100.0), (10000.0, 1000.0))

    for alpha, mu in configs:
        if time.time() > deadline or deadline - time.time() < 8.0:
            break
        try:
            res = minimize(
                smooth_obj, Hw, jac=smooth_grad, args=(n, alpha, mu),
                method="L-BFGS-B", bounds=bounds,
                options={"maxiter": 300, "ftol": 1e-15,
                         "gtol": 1e-13, "maxfun": 6000, "maxls": 30}
            )
            H2 = project_feasible(res.x, n)
            C2 = compute_C(H2, n)
            c2 = c5_from_C(C2, n)
            if c2 < best - 1.0e-12:
                best = c2
                H0 = H2.copy()
                C0 = C2.copy()
                Hw = H2.copy()
        except Exception:
            pass

    C0 = compute_C(H0, n)
    return H0, C0, c5_from_C(C0, n)


def upd(h, C, n, bh, bC, bc):
    c = c5_from_C(C, n)
    if c < bc - 1.0e-12:
        return h.copy(), C.copy(), c
    return bh, bC, bc

        bh, bC, bc = upd(hp, Cp, n, bh, bC, bc)

    if n <= 350 and time.time() < deadline - 240.0:
        hp = bh.copy()
        Cp = bC.copy()
        hp, Cp, _ = all_pair_search(hp, Cp, n,
                                    min(time.time() + 220.0, deadline - 240.0),
                                    0, 13)
        bh, bC, bc = upd(hp, Cp, n, bh, bC, bc)

    if n <= 350 and time.time() < deadline - 200.0:
        hp = bh.copy()
        Cp = bC.copy()
        hp, Cp, _ = all_pair_search(hp, Cp, n,
                                    min(time.time() + 170.0, deadline - 200.0),
                                    0, 21)
        bh, bC, bc = upd(hp, Cp, n, bh, bC, bc)

    if n <= 350 and time.time() < deadline - 160.0:
        hp = bh.copy()
        Cp = bC.copy()
        hp, Cp, _ = all_pair_search(hp, Cp, n,
                                    min(time.time() + 190.0, deadline - 170.0),
                                    1, (0.1, 0.05, 0.02, 0.005))
        bh, bC, bc = upd(hp, Cp, n, bh, bC, bc)

    if time.time() < deadline - 110.0:
        hp, Cp, _ = lbfgs_refine(bh, n, min(time.time() + 160.0, deadline - 110.0))
        bh, bC, bc = upd(hp, Cp, n, bh, bC, bc)

    if n <= 350 and time.time() < deadline - 60.0:
        hp = bh.copy()
        Cp = bC.copy()
        hp, Cp, _ = all_pair_search(hp, Cp, n,
                                    min(time.time() + 120.0, deadline - 60.0),
                                    1, (0.01, 0.005, 0.002, 0.001))
        bh, bC, bc = upd(hp, Cp, n, bh, bC, bc)

    if n <= 350 and time.time() < deadline - 25.0:
        hp = bh.copy()
        Cp = bC.copy()
        hp, Cp, _ = all_pair_search(hp, Cp, n,
                                    min(time.time() + 70.0, deadline - 12.0),
                                    1, (0.001, 0.0005, 0.0002, 0.0001))
        bh, bC, bc = upd(hp, Cp, n, bh, bC, bc)

    if n <= 350 and time.time() < deadline - 35.0:
        hp = bh.copy()
        Cp = bC.copy()
        hp, Cp, _ = all_pair_search(hp, Cp, n,
                                    min(time.time() + 30.0, deadline - 10.0),
                                    0, 29)
        bh, bC, bc = upd(hp, Cp, n, bh, bC, bc)

    if n <= 350 and time.time() < deadline - 70.0:
        hp = bh + rng.normal(0.0, 0.1, n)
        hp = project_feasible(hp, n)
        Cp = compute_C(hp, n)
        if c5_from_C(Cp, n) < bc + 0.05:
            hp, Cp, _ = all_pair_search(hp, Cp, n,
                                        min(time.time() + 25.0, deadline - 30.0),
                                        0, 13)
            if c5_from_C(Cp, n) < bc + 0.02:
                hp, Cp, _ = all_pair_search(hp, Cp, n,
                                            min(time.time() + 25.0, deadline - 8.0),
                                            1, (0.1, 0.05, 0.02))
            bh, bC, bc = upd(hp, Cp, n, bh, bC, bc)

    out = project_feasible(bh, n)
    C_out = compute_C(out, n)
    c_out = c5_from_C(C_out, n)
    return out, c_out, n
