def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return (m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha) * dx
    return (m + np.log(np.sum(np.exp(sh))) / alpha) * dx

def _con_fun(x, n):
    return x.sum() - n / 2.0

    return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h.copy(), args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except:
        return project_feasible(h, n)

            if tot < 1e-15:

                h[p] = np.clip(h[p] + best_d, 0, 1)
                current_max = max(um, float(c[ks].max()))
                improved = True
        if not improved:
            break
    return h, current_max * dx

def optimize_candidate(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)
    for alpha in [10000, 100000, 1000000]:
        if time.time() > deadline - 40:
            break
        h_s = slsqp(h, n, alpha=alpha, maxiter=600)
        c_s = compute_c5(h_s, n)
        if c_s < best_c:
            h, best_c = h_s, c_s
    if time.time() < deadline - 30:
        cd_dl = min(deadline - 10, time.time() + 100)
        h_cd, c_cd = fast_cd(h, n, max_passes=500, grid_pts=80,
                             deadline=cd_dl, rng=rng)
        h_cd = project_feasible(h_cd, n)
        c_cd = compute_c5(h_cd, n)
        if c_cd < best_c:
            h, best_c = h_cd, c_cd
    if time.time() < deadline - 25:
        h_s = slsqp(h, n, alpha=3000000, maxiter=400)
        c_s = compute_c5(h_s, n)
        if c_s < best_c:
            h, best_c = h_s, c_s
    if time.time() < deadline - 15:
        h_cd, c_cd = fast_cd(h, n, max_passes=300, grid_pts=120,
                             deadline=deadline - 5, rng=rng)
        h_cd = project_feasible(h_cd, n)
        c_cd = compute_c5(h_cd, n)
        if c_cd < best_c:
            h, best_c = h_cd, c_cd
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n_init = 200
    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)
    alloc1 = min(380, (deadline - t0) * 0.38)
    h_opt, c_opt = optimize_candidate(h_init, n_init, t0 + alloc1, rng)
    if c_opt < best_c:
        best_c, best_h = c_opt, h_opt
    for n_try in [200, 250]:
        if time.time() > deadline - 200:
            break
        n_cur = len(best_h)
        x_old = np.linspace(0, 2, n_cur)
        x_new = np.linspace(0, 2, n_try)
        h_interp = project_feasible(np.interp(x_new, x_old, best_h), n_try)
        alloc = min(130, (deadline - time.time()) * 0.2)
        if alloc < 50:
            break
        h_opt, c_opt = optimize_candidate(h_interp, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt
    if time.time() < deadline - 100:
        n = len(best_h)
        remaining = deadline - time.time() - 50
        n_restart = max(2, int(remaining / 70))
        for r in range(n_restart):
            if time.time() > deadline - 60:
                break
            scale = 0.008 + 0.03 * (r / max(1, n_restart - 1))
            hp = project_feasible(best_h + rng.normal(0, scale, n), n)
            alloc = min(55, max(25, (deadline - time.time() - 40) * 0.35))
            if alloc < 20:
                break
            h_opt, c_opt = optimize_candidate(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c, best_h = c_opt, h_opt
    if time.time() < deadline - 25:
        n = len(best_h)
        h_f = slsqp(best_h, n, alpha=5000000, maxiter=800)
        c_f = compute_c5(h_f, n)
        if c_f < best_c:
            best_c, best_h = c_f, h_f
    if time.time() < deadline - 10:
        n = len(best_h)
        h_cd, c_cd = fast_cd(best_h, n, max_passes=400, grid_pts=150,
                             deadline=deadline - 4, rng=rng)
        h_cd = project_feasible(h_cd, n)
        c_cd = compute_c5(h_cd, n)
        if c_cd < best_c:
            best_c, best_h = h_cd, c_cd
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
