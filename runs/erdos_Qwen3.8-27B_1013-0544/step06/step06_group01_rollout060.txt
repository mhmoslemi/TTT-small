import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def _eq_fun(x, m):
    return x.sum() - m / 2.0

def _build_h(x, m):
    """Build full h from symmetric half x of length m."""
    h = np.empty(2 * m)
    h[:m] = x
    h[m:] = x[::-1]
    return h

def obj_sym(x, m, alpha):
    """Smooth max of correlation for symmetric h."""
    h = _build_h(x, m)
    n = 2 * m
    c = np.correlate(h, 1.0 - h, mode='full')
    # By symmetry, c[k] = c[2n-2-k], so just take the first n values
    c_half = c[:n]  # k = 0 to n-1
    scaled = alpha * c_half
    mx = scaled.max()
    if mx > 500:
        lse = mx + np.log(np.sum(np.exp(scaled - mx)))
    else:
        lse = np.log(np.sum(np.exp(scaled)))
    return float((lse / alpha) * (2.0 / n))

def slsqp_sym(h, n, alpha, maxiter=3000):
    """SLSQP on symmetric parameters."""
    m = n // 2
    x0 = h[:m].copy()
    bounds = [(0.0, 1.0)] * m
    cons = [{'type': 'eq', 'fun': _eq_fun, 'args': (m,)}]
    try:
        res = minimize(obj_sym, x0, args=(m, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = _build_h(res.x, m)
        return project_feasible(h_new, n)
    except:
        return project_feasible(h, n)

    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)

        diff = target - h.sum()
        if abs(diff) < 1e-13:

        if diff > 0:
            room = 1.0 - h
            tot = room.sum()

            if tot < 1e-15:

    # Average with mirror

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h = np.minimum(1.0, h + (diff / tot) * room)
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h = np.maximum(0.0, h + (diff / tot) * h)
    return np.clip(h, 0.0, 1.0)

def project_symmetric(h, n):
    m = n // 2
    idx = np.arange(m)
    h_sym = 0.5 * (h[idx] + h[n - 1 - idx])
    h_full = np.empty(n)
    h_full[:m] = h_sym
    h_full[m:] = h_sym[::-1]
    return project_feasible(h_full, n)
