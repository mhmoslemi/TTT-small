import numpy as np
import time


def corr_full(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)


def eval_cmax(h, n):
    return float(np.max(corr_full(h, n)))


def project(h):
    h = np.asarray(h, dtype=float)
    n = len(h)
    target = 0.5 * n
    if not np.all(np.isfinite(h)):
        h = np.nan_to_num(h, nan=0.5, posinf=1.0, neginf=0.0)
    lo = 1.0 - float(np.max(h))
    hi = -float(np.min(h))
    if lo > hi:
        lo, hi = hi, lo
    for _ in range(55):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(h + mid, 0.0, 1.0)))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)


def resample(h, n):
    if len(h) == n:
        return project(h.copy())
    xs = np.linspace(0.0, 2.0, len(h), endpoint=False)
    xn = np.linspace(0.0, 2.0, n, endpoint=False)
    return project(np.interp(xn, xs, h))


def binarize(h, n):
    target = int(n // 2)
    h = np.asarray(h, dtype=float)
    b = np.zeros(n, dtype=float)
    if target > 0:
        idx = np.argsort(h, kind="stable")[::-1]
        b[idx[:target]] = 1.0
    return b


def perturb_h(h, n, rng, strength, k):
    h = np.asarray(h, dtype=float).copy()
    for _ in range(k):
        i = int(rng.randint(0, n))
        j = int(rng.randint(0, n))
        if i == j:
            continue
        s = (rng.random() - 0.5) * strength
        a = h[i] + s
        b = h[j] - s
        if 0.0 <= a <= 1.0 and 0.0 <= b <= 1.0:
            h[i] = a
            h[j] = b
        else:
            h[i], h[j] = h[j], h[i]
    return project(h)


def random_start(n, seed, kind=0):
    rng = np.random.RandomState(seed)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    if kind == 0:
        z = np.zeros(n, dtype=float)
        for m in range(1, 14):
            amp = rng.random() / float(m + 1)
            phi = rng.uniform(0.0, 2.0 * np.pi)
            z += amp * np.cos(2.0 * np.pi * m * x / 2.0 + phi)
        h = 0.5 + 0.5 * np.tanh(4.0 * z)
    elif kind == 1:
        h = np.zeros(n, dtype=float)
        target = int(n // 2)
        if target > 0:
            idx = rng.choice(n, target, replace=False)
            h[idx] = 1.0
    else:
        h = np.clip(0.5 + 0.18 * rng.randn(n), 0.0, 1.0)
    return project(h)


def pair_build_L(h, inv, i, j, n, L):
    C = n - 1
    L.fill(0.0)
    L[C - i:C - i + n] += inv
    L[i:i + n] -= h[::-1]
    L[C - j:C - j + n] -= inv
    L[j:j + n] += h[::-1]


def pair_cost_t(t, cvec, L, idx0, idxp, idxq):
    v = cvec + t * L
    t2 = t * t
    v[idx0] -= 2.0 * t2
    v[idxp] += t2
    v[idxq] += t2
    return float(np.max(v))


def grid_max_vals(cvec, L, idx0, idxp, idxq, ts):
    m = cvec + np.outer(ts, L)
    t2 = ts * ts
    m[:, idx0] -= 2.0 * t2
    m[:, idxp] += t2
    m[:, idxq] += t2
    return m.max(axis=1)


def best_pair_t(cvec, L, idx0, idxp, idxq, lo, hi):
    curmax = float(np.max(cvec))
    span = hi - lo
    if span <= 1e-13:
        return 0.0, curmax
    K = 21
    ts = np.linspace(lo, hi, K)
    m1 = grid_max_vals(cvec, L, idx0, idxp, idxq, ts)
    b1 = int(np.argmin(m1))
    if b1 == 0:
        a, b = ts[0], ts[1]
    elif b1 == K - 1:
        a, b = ts[K - 2], ts[K - 1]
    else:
        a, b = ts[b1 - 1], ts[b1 + 1]
    if b - a <= 1e-13:
        return float(ts[b1]), float(m1[b1])
    ts = np.linspace(a, b, K)
    m2 = grid_max_vals(cvec, L, idx0, idxp, idxq, ts)
    b2 = int(np.argmin(m2))
    best_t = float(ts[b2])
    best_c = float(m2[b2])
    if b2 == 0:
        a2, b2v = ts[0], ts[1]
    elif b2 == K - 1:
        a2, b2v = ts[K - 2], ts[K - 1]
    else:
        a2, b2v = ts[b2 - 1], ts[b2 + 1]
    if b2v - a2 > 1e-12:
        gr = (3.0 - np.sqrt(5.0)) / 2.0
        lo_g, hi_g = a2, b2v
        for _ in range(10):
            m1 = lo_g + (hi_g - lo_g) * gr
            m2 = hi_g - (hi_g - lo_g) * gr
            c1 = pair_cost_t(m1, cvec, L, idx0, idxp, idxq)
            c2 = pair_cost_t(m2, cvec, L, idx0, idxp, idxq)
            if c1 < c2:
                hi_g = m2
            else:
                lo_g = m1
        for cand in (lo_g, hi_g):
            cc = pair_cost_t(cand, cvec, L, idx0, idxp, idxq)
            if cc < best_c:
                best_t, best_c = float(cand), cc
    return best_t, best_c


def try_pair(h, inv, cvec, cur, i, j, n, L):
    ai = float(h[i])
    aj = float(h[j])
    lo = max(-ai, aj - 1.0)
    hi = min(1.0 - ai, aj)
    if hi - lo < 1e-13:
        return False, cur, 0.0, None
    pair_build_L(h, inv, i, j, n, L)
    C = n - 1
    idx0 = C
    idxp = C + j - i
    idxq = C + i - j
    t, bc = best_pair_t(cvec, L, idx0, idxp, idxq, lo, hi)
    if bc >= cur - 1e-11:
        return False, cur, 0.0, None
    cnew = cvec + t * L
    t2 = t * t
    cnew[idx0] -= 2.0 * t2
    cnew[idxp] += t2
    cnew[idxq] += t2
    newc = float(np.max(cnew))
    if newc < cur - 1e-12:
        return True, newc, float(t), cnew
    return False, cur, 0.0, None


def pair_random_search(h, n, deadline, seed, pairs=3500, no_pass_limit=3):
    h = project(h)
    n = len(h)
    inv = 1.0 - h
    cvec = corr_full(h, n)
    cur = float(np.max(cvec))
    best_h = h.copy()
    best_c = cur
    L = np.empty(2 * n - 1, dtype=float)
    rng = np.random.RandomState(seed)
    no_pass = 0
    updates = 0
    while time.time() < deadline and no_pass < no_pass_limit:

                    h[i] = hni

                    inv[j] = 1.0 - hn j
                    cvec = cnew
                    cur = newc

                updates += 1
