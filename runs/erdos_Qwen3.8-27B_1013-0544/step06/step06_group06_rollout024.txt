    """Compute correlation change when h[i]:0->1, h[j]:1->0. O(n)."""
    # lag = m - (n-1), m in 0..2n-2
    # c[m] = sum_p h[p] * (1-h[p+lag])
    # When h[i]: 0->1 adds (1-h[i+lag]) and removes h[i-lag]
    # When h[j]: 1->0 removes (1-h[j+lag]) and adds h[j-lag]

    # + (1-h[i+lag]) where 0<=i+lag<n

    delta[mask_p] += 1 - h[idx_p[mask_p]]

    idx_m = i - lags
    mask_m = (idx_m >= 0) & (idx_m < n)
