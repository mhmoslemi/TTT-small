    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        # Smooth max: m + (1/alpha)*log(sum(exp(alpha*(c-m))))
        shifted = alpha * (c - m)
        logsum = np.log(np.exp(shifted).sum())
        return m + logsum / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h
