def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = float(h.sum())
    if abs(s - target) < 1e-13:
        return h
    if s < target:
        lo, hi = 0.0, 1.0
    else:
        lo, hi = -1.0, 0.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s2 = float(np.clip(h + mid, 0.0, 1.0).sum())
        if s2 < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    s = float(h.sum())
    if abs(s - target) <= 1e-11:
        return h
    # final proportional correction to bound
    diff = target - s
    if diff > 0:
        room = 1.0 - h
        rsum = float(room.sum())
        if rsum > 1e-15:
            h = np.clip(h + diff / rsum * room, 0.0, 1.0)
    else:
        hsum = float(h.sum())
        if hsum > 1e-15:
            h = np.clip(h + diff / hsum * h, 0.0, 1.0)
    return h

        else:

    return np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)

    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

    if G_DL > 0.0 and time.time() > G_DL:

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = float(h.sum())
    if abs(s - target) < 1e-12:
        return h
    if s < target:
        lo, hi = 0.0, 1.0
    else:
        lo, hi = -1.0, 0.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s2 = float(np.clip(h + mid, 0.0, 1.0).sum())
        if s2 < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def smooth_value(g):
    m = float(np.max(g))
    if G_ACTIVE:
        return m
    a = float(G_ALPHA)
    if a <= 0.0:
        return m
    z = a * (g - m)
    zmax = float(np.max(z))
    e = np.exp(z - zmax)
    return float(m + (zmax + float(np.log(e.sum()))) / a)

    time_check()
    x = np.asarray(x, dtype=float)
    g = np.correlate(x, 1.0 - x, mode='full') * G_FACT
    return smooth_value(g)

    time_check()
    n = G_N
    x = np.asarray(x, dtype=float)
    c = np.correlate(x, 1.0 - x, mode='full')
    g = c * G_FACT

        xout = h0
    if xout is None:
        xout = h0
    return project_feasible(xout, n)

def small_obj(x):
    time_check()
    full = G_SM_BASE.copy()
    full[G_SM_IDX] = np.asarray(x, dtype=float)
    full = np.clip(full, 0.0, 1.0)
    g = np.correlate(full, 1.0 - full, mode='full') * G_FACT
    return smooth_value(g)

def small_sum_fun(x):
    return float(np.sum(x) - G_SM_NEED)

def small_sum_jac(x):
    return G_SM_ONES

def small_polish(h, m, maxiter, alpha, exact, deadline):
    global G_N, G_FACT, G_ALPHA, G_ACTIVE, G_TOL
    global G_SM_BASE, G_SM_IDX, G_SM_NEED, G_SM_ONES, G_OPT_X
    n = len(h)
    G_N = n
    G_FACT = 2.0 / n
    G_ALPHA = float(alpha)
    G_ACTIVE = bool(exact)
    G_TOL = 1e-11
    h = project_feasible(h, n)
    if m > n:
        m = n
    if m < 1:
        return h, compute_c5(h, n)
    score = np.minimum(h, 1.0 - h)
    idx = np.argsort(score)[::-1][:m]
    G_SM_BASE = h.copy()
    G_SM_IDX = idx
    G_SM_NEED = n / 2.0 - (float(h.sum()) - float(h[idx].sum()))
    x0 = h[idx].copy()
    G_SM_ONES = np.ones(m)
    G_OPT_X = None
    cons = {'type': 'eq', 'fun': small_sum_fun, 'jac': small_sum_jac}
    bounds = [(0.0, 1.0)] * m
    opts = {'maxiter': int(maxiter), 'ftol': 1e-12, 'maxls': 20, 'gtol': 1e-8}
    try:
        res = minimize(small_obj, x0, method='SLSQP', bounds=bounds,
                       constraints=cons, options=opts, callback=timeout_cb)
        xout = res.x
    except TimeOut:
        xout = G_OPT_X
    except Exception:
        xout = x0
    if xout is None:
        xout = x0
    full = G_SM_BASE.copy()
    full[idx] = xout
    full = project_feasible(full, n)
    return full, compute_c5(full, n)

def line_polish(h, n, alpha, deadline, maxiter=30, mode='smooth', act_tol=1e-9):
    global G_N, G_FACT, G_ALPHA, G_ACTIVE, G_TOL
    G_N = n
    G_FACT = 2.0 / n
    G_ALPHA = float(alpha)
    G_ACTIVE = (mode == 'active')
    G_TOL = float(act_tol)
    cur = project_feasible(h, n)
    curc = compute_c5(cur, n)
    steps = (0.08, 0.04, 0.018, 0.008, 0.003, 0.001, 0.0004)
    stall = 0
    for _ in range(maxiter):
        if time.time() > deadline:
            break
        try:
            grad = grad_full(cur)
        except TimeOut:
            break
        gn = float(np.linalg.norm(grad))
        if gn < 1e-14:
            break
        bestc = curc
        bestx = cur
        for st in steps:
            if time.time() > deadline:
                break
            x = project_feasible(cur - st * (grad / gn), n)
            c = compute_c5(x, n)
            if c < bestc:
                bestc = c
                bestx = x
        if bestc < curc - 1e-15:
            cur = bestx
            curc = bestc
            stall = 0
        else:
            stall += 1
            if stall > 12:
                break
    return cur, curc

def shift_search(h, n, full=True):
    h = project_feasible(h, n)
    besth = h
    bestc = compute_c5(h, n)
    if full:
        for s in range(n):
            r = np.roll(h, s)
            c = compute_c5(r, n)
            if c < bestc:
                bestc = c
                besth = r.copy()
    else:
        for i in range(16):
            s = (i * n) // 16
            r = np.roll(h, s)
            c = compute_c5(r, n)
            if c < bestc:
                bestc = c
                besth = r.copy()
    rev = h[::-1]
    if full:
        for s in range(n):
            r = np.roll(rev, s)
            c = compute_c5(r, n)
            if c < bestc:
                bestc = c
                besth = r.copy()
    else:
        for i in range(16):
            s = (i * n) // 16
            r = np.roll(rev, s)
            c = compute_c5(r, n)
            if c < bestc:
                bestc = c
                besth = r.copy()
    return project_feasible(besth, n), bestc

def perturb(h, n, rng, amp):
    h = h.copy()
    m = min(n, rng.randint(8, 53))
    idx = rng.choice(n, m, replace=False)
    v = rng.normal(0.0, amp, m)
    v -= v.mean()
    h[idx] += v
    return project_feasible(h, n)

def make_starts(h0, n, rng):
    out = []
    out.append(project_feasible(h0, n))
    out.append(project_feasible(np.full(n, 0.5), n))
    t = np.mod((np.arange(n) + 0.5) * 0.61803398875, 1.0)
    out.append(project_feasible((t < 0.5).astype(float), n))
    for _ in range(2):
        g = rng.normal(0.0, 1.0, n)
        out.append(project_feasible(g, n))
        hb = np.zeros(n, dtype=float)
        hb[np.argsort(g)[n // 2:]] = 1.0
        out.append(project_feasible(hb, n))
    x = np.arange(n)
    for f in (0.7, 1.3, 2.7, 4.1):
        ph = rng.uniform(0.0, 2.0 * np.pi)
        h = 0.5 + 0.32 * np.sin(2.0 * np.pi * f * x / n + ph) \
            + 0.16 * np.cos(2.0 * np.pi * 2.0 * f * x / n + 2.0 * ph)
        out.append(project_feasible(h, n))
    for bl in (4, 8):
        period = 2 * bl
        h = ((np.arange(n) % period) < bl).astype(float)
        out.append(project_feasible(h, n))
    return out

def run(seed=42, budget_s=1100, **kwargs):
    global G_DL, G_N, G_FACT
    t0 = time.time()
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = int(len(h0))
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)
    if n < 2:
        return h0, 0.0, n
    G_N = n
    G_FACT = 2.0 / n
    G_DL = t0 + max(10.0, budget_s - 12.0)
    best_h = project_feasible(h0, n)
    best_c = compute_c5(best_h, n)
    if budget_s <= 20.0 or time.time() > G_DL - 5.0:
        return best_h, best_c, n
    rng = np.random.RandomState(seed)
    starts = [best_h.copy()]
    if time.time() < G_DL - 25.0:
        raw = make_starts(best_h, n, rng)
        arrs = []
        cs = []
        for s in raw:
            if time.time() > G_DL - 5.0:
                break
            s = project_feasible(s, n)
            c = compute_c5(s, n)
            arrs.append(s)
            cs.append(c)
            if c < best_c:
                best_c = c
                best_h = s.copy()
        if cs:
            order = np.argsort(cs)
            picks = []
            for ii in order:
                if len(picks) >= 6:
                    break
                picks.append(int(ii))
            starts = []
            for ii in picks:
                starts.append(arrs[ii].copy())
    if not starts:
        starts = [best_h.copy()]

    alphas = (20.0, 100.0, 600.0, 3000.0, 15000.0)
    iters = (60, 60, 80, 80, 60)
    for si in range(len(starts)):
        if time.time() > G_DL - 120:
            break
        h = starts[si].copy()
        for ai in range(len(alphas)):
            if time.time() > G_DL - 60:
                break
            h = slsqp_full(h, alphas[ai], iters[ai], 'smooth', 1e-11)
            if si == 0 or ai >= 2:
                h, c = shift_search(h, n, full=(si == 0 and ai == len(alphas) - 1))
            else:
                c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
        if time.time() > G_DL - 50:
            break
        h, c = line_polish(h, n, 3000.0, G_DL, 25, 'smooth', 1e-9)
        h, c2 = shift_search(h, n, full=True)
        if c2 < best_c:
            best_c = c2
            best_h = h.copy()
        if time.time() < G_DL - 25:
            h, c = small_polish(h, 10, 70, 3000.0, False, G_DL)
            if c < best_c:
                best_c = c
                best_h = h.copy()
        if time.time() < G_DL - 20:
            h, c = small_polish(h, 14, 50, 1.0, True, G_DL)
            if c < best_c:
                best_c = c
                best_h = h.copy()
        if time.time() < G_DL - 15:
            h, c = line_polish(h, n, 15000.0, G_DL, 20, 'smooth', 1e-9)
            h, c2 = shift_search(h, n, full=True)
            if c2 < best_c:
                best_c = c2
                best_h = h.copy()

    trial = 0
    while time.time() < G_DL - 35 and trial < 100:
        trial += 1
        if trial % 8 == 0:
            h = project_feasible(0.5 + rng.normal(0.0, 0.35, n), n)
        else:
            amp = 0.06 + 0.14 * ((trial % 4) / 3.0)
            h = perturb(best_h, n, rng, amp)
        for alpha, mx in ((600.0, 35), (12000.0, 35)):
            if time.time() > G_DL - 10:
                break
            h = slsqp_full(h, alpha, mx, 'smooth', 1e-11)
        if time.time() > G_DL - 12:
            continue
        h, c = shift_search(h, n, full=False)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        if time.time() < G_DL - 8:
            h, c = small_polish(h, 12, 45, 3000.0, False, G_DL)
            if c < best_c:
                best_c = c
                best_h = h.copy()
        if time.time() < G_DL - 6:
            h, c = small_polish(h, 16, 35, 1.0, True, G_DL)
            if c < best_c:
                best_c = c
                best_h = h.copy()
        if time.time() < G_DL - 4:
            h, c = line_polish(h, n, 10000.0, G_DL, 15, 'smooth', 1e-9)
            h, c2 = shift_search(h, n, full=False)
            if c2 < best_c:
                best_c = c2
                best_h = h.copy()

    if time.time() < G_DL - 6:
        best_h, best_c = shift_search(best_h, n, full=True)
    if time.time() < G_DL - 4:
        h, c = small_polish(best_h, 12, 70, 8000.0, False, G_DL)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    if time.time() < G_DL - 3:
        h, c = small_polish(best_h, 20, 50, 1.0, True, G_DL)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    if time.time() < G_DL - 2:
        h, c = line_polish(best_h, n, 50000.0, G_DL, 20, 'smooth', 1e-9)
        h, c2 = shift_search(h, n, full=True)
        if c2 < best_c:
            best_c = c2
            best_h = h.copy()
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
