def binarize(h, n):
    """Project h to nearest binary with exactly n/2 ones."""
    # Start with threshold at 0.5
    h_bin = (h >= 0.5).astype(float)
    # Adjust to get exactly n/2 ones
    n_ones = int(h_bin.sum())
    target = n // 2
    if n_ones > target:
        # Need to flip some 1s to 0s: pick the smallest h values among 1s
        one_idx = np.where(h_bin == 1)[0]
        order = np.argsort(h[one_idx])  # smallest h first
        for idx in order[:n_ones - target]:
            h_bin[one_idx[idx]] = 0
    elif n_ones < target:
        # Need to flip some 0s to 1s: pick the largest h values among 0s
        zero_idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zero_idx])  # largest h first
        for idx in order[:target - n_ones]:
            h_bin[zero_idx[idx]] = 1
    return h_bin

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        order = np.argsort(np.abs(h - 0.5))
        k = int(abs(diff)) + 1
        idx = order[:k]
        h[idx] = np.clip(h[idx] + diff / k, 0, 1)
    return h

def binarize(h, n):
    h_b = (h >= 0.5).astype(float)
    n1 = int(h_b.sum())
    tgt = n // 2
    if n1 > tgt:
        oi = np.where(h_b == 1)[0]
        ord_ = np.argsort(h[oi])
        for k in range(n1 - tgt):
            h_b[oi[ord_[k]]] = 0
    elif n1 < tgt:
        zi = np.where(h_b == 0)[0]
        ord_ = np.argsort(-h[zi])
        for k in range(tgt - n1):
            h_b[zi[ord_[k]]] = 1
    return h_b

def _sum_fun(x):
    return x.sum() - x.shape[0] / 2.0

def _sum_jac(x):
    return np.ones(x.shape[0])

def smooth_obj(x, n, alpha, top_idx=None):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    if top_idx is not None:
        c = c[top_idx]
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_opt(h, n, alpha, top_idx=None, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha, top_idx),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        return res.x
    except Exception:
        return h.copy()

        ones = np.where(h == 1)[0]

        if len(ones) == 0 or len(zeros) == 0:

        if best_flip is not None:
            i, j = best_flip
            h[i] = 0
            h[j] = 1
            stall = 0
        else:
            stall += 1
