For binary: h[i]=0, h'[i]=1, h[j]=1, h'[j]=0:
Term 1: 1*(1-h'[i+t]) - 0 = 1-h'[i+t]
Term 2: 0 - 1*(1-h'[j+t]) = -(1-h'[j+t]) = h'[j+t] - 1
Term 3: h[i-t]*(0-1) = -h[i-t] (if i-t valid and i-t ≠ i, j_swap)
Term 4: h[j_swap-t]*(1-0) = h[j_swap-t] (if j_swap-t valid and j_swap-t ≠ i, j_swap)

So: Δc[n-1+t] = (1-h'[i+t]) + (h'[j+t]-1) - h[i-t] + h[j_swap-t]
= 1 - h'[i+t] + h'[j+t] - 1 - h[i-t] + h[j_swap-t]
= h'[j+t] - h'[i+t] - h[i-t] + h[j_swap-t]

h'[i+t] = h[i+t] except when i+t = i (t=0) → h'[i]=1, or i+t = j (t=j-i) → h'[i+t] = h'[j] = 0

```python
def swap_delta_binary(h, i, j, n):
    """Delta for full correlation when swapping h[i]=0->1, h[j]=1->0 (binary h)."""
    delta = np.zeros(2*n-1)
    # s = shift, m = n-1+s
    # For s in range [-(n-1), n-1]:
    #   delta[n-1+s] = h_new[j+s] - h_new[i+s] - h[i-s] + h[j-s]
    #   (only terms where index is in [0,n-1])
    
    for s in range(-(n-1), n):
        val = 0.0
        # h_new[i+s]: valid if 0 <= i+s < n
        if 0 <= i+s < n:
            if s == 0:
                val -= 1.0  # h_new[i] = 1
            elif i+s == j:
                val -= 0.0  # h_new[j] = 0
            else:
                val -= h[i+s]
        # h_new[j+s]: valid if 0 <= j+s < n
        if 0 <= j+s < n:
            if s == 0:
                val += 0.0  # h_new[j] = 0
            elif j+s == i:
                val += 1.0  # h_new[i] = 1
            else:
                val += h[j+s]
        # h[i-s]: valid if 0 <= i-s < n, and i-s != i (s≠0) and i-s != j (s≠i-j)
        if 0 <= i-s < n and s != 0 and i-s != j:
            val -= h[i-s]
        # h[j-s]: valid if 0 <= j-s < n, and j-s != i (s≠j-i) and j-s != j (s≠0)
        if 0 <= j-s < n and s != 0 and j-s != i:
            val += h[j-s]
        delta[n-1+s] = val
    return delta
```

For the 2-swap move (swap i,k with j,l where i,k are 0s and j,l are 1s):
The delta is NOT simply the sum of two single-swap deltas because the correlation is quadratic in h.

However, for the specific case of binary h, we can compute the 2-swap delta as:
Δ2swap = Δ(i→j) + Δ(k→l) + cross_term

The cross term arises from the interaction between the two swaps. For binary h:
If we apply swap (i→j) first, then (k→l), the delta of the second swap depends on the new state.
Alternatively, we can compute it directly.

Actually, for the correlation c[m] = sum_k h[k]*(1-h[k+s]) where s = m-n+1:

After 2-swap (i,k go 0→1, j,l go 1→0):
c'[m] - c[m] = sum of changes from all 4 position changes.

= [h_new[i]*(1-h_new[i+s]) - h[i]*(1-h[i+s])]  (k=i term)
+ [h_new[k1]*(1-h_new[k1+s]) - h[k1]*(1-h[k1+s])]  (k=k term, let k1=k)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def make_delta(h, i, j, n):
    """Vectorized delta for binary swap h[i]:0->1, h[j]:1->0."""
    s = np.arange(-(n-1), n)
    mask_is = (i + s >= 0) & (i + s < n)
    mask_js = (j + s >= 0) & (j + s < n)
    mask_isu = (i - s >= 0) & (i - s < n)
    mask_jsu = (j - s >= 0) & (j - s < n)
    delta = np.zeros(2 * n - 1)
    if mask_is.any():
        delta[mask_is] -= h[i + s[mask_is]]
    if mask_js.any():
        delta[mask_js] += h[j + s[mask_js]]
    if mask_isu.any():
        delta[mask_isu] -= h[i - s[mask_isu]]
    if mask_jsu.any():
        delta[mask_jsu] += h[j - s[mask_jsu]]
    # Fix special: s=0
    delta[n-1] = 0.0
    # Fix s=j-i (i+s=j, so h_new[i+s]=0 not h[i+s]=1)
    sj = j - i
    if -(n-1) <= sj <= n-1:
        m = n - 1 + sj
        val = 0.0
        if 0 <= j + sj < n and j + sj != j and j + sj != i:
            val += h[j + sj]
        # h_new[i+sj] = h_new[j] = 0, so term is -0 = 0
        # h[i-sj]: valid?
        if 0 <= i - sj < n and i - sj != i and i - sj != j:
            val -= h[i - sj]
        # h[j-sj]: valid?
        if 0 <= j - sj < n and j - sj != i and j - sj != j:
            val += h[j - sj]
        delta[m] = val
    # Fix s=i-j (j+s=i, so h_new[j+s]=1 not h[j+s]=0)
    si = i - j
    if -(n-1) <= si <= n-1 and si != sj:
        m = n - 1 + si
        val = 0.0
        # h_new[i+si]: i+si = 2i-j
        if 0 <= i + si < n and i + si != i and i + si != j:
            val -= h[i + si]
        # h_new[j+si] = h_new[i] = 1
        val += 1.0
        if 0 <= i - si < n and i - si != i and i - si != j:
            val -= h[i - si]
        if 0 <= j - si < n and j - si != i and j - si != j:
            val += h[j - si]
        delta[m] = val
    return delta

def local_search(h, n, deadline, seed=42, max_sweeps=100000):
    h = binarize(h, n)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    cur_c = float(corr.max())
    best_c = cur_c
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(h == 1)[0].copy()
        zeros = np.where(h == 0)[0].copy()
        rng.shuffle(ones)
        rng.shuffle(zeros)
        for i in zeros:
            i = int(i)
            hit = False
            for j in ones:
                j = int(j)
                d = make_delta(h, i, j, n)
                nc = float((corr + d).max())
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr += d
                    cur_c = nc
                    if cur_c < best_c:
                        best_c = cur_c
                        best_h = h.copy()
                    improved = True
                    hit = True
                    break
            if hit or time.time() > deadline:
                if hit:
                    break
            if improved and hit:
                break
        if not improved:
            break
    return best_h, best_c * dx

def two_swap_search(h, n, deadline, seed=42, max_iters=50000):
    h = binarize(h, n)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    cur_c = float(corr.max())
    best_c = cur_c
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        j1, j2 = rng.choice(ones, 2, replace=False)
        i1, i2 = rng.choice(zeros, 2, replace=False)
        j1, j2, i1, i2 = int(j1), int(j2), int(i1), int(i2)
        h[i1], h[j1] = 1.0, 0.0
        h[i2], h[j2] = 1.0, 0.0
        new_corr = np.correlate(h, 1.0 - h, mode='full')
        nc = float(new_corr.max())
        if nc < cur_c - 1e-15:
            corr = new_corr
            cur_c = nc
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            h[j1], h[i1] = 1.0, 0.0
            h[j2], h[i2] = 1.0, 0.0
    return best_h, best_c * dx

def smooth_optimize(h, n, alpha, maxiter=300):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cons_fun}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h
