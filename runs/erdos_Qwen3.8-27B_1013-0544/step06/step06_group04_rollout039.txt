import time

    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c)) * dx

def binarize(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def swap_delta_c(h, i0, j0, n):
    delta = np.zeros(2 * n - 1)
    k_lo = max(-(n-1), -i0, -j0, i0-n+1, j0-n+1)
    k_hi = min(n-1, n-1-i0, n-1-j0, i0, j0)
    for k in range(int(k_lo), int(k_hi) + 1):
        d = 0.0
        for i in (i0, j0, i0-k, j0-k):
            if 0 <= i < n and 0 <= i+k < n:
                hi_o = h[i]; hik_o = h[i+k]
                hi_n = 1.0 if i == i0 else (0.0 if i == j0 else h[i])
                hik_n = 1.0 if (i+k) == i0 else (0.0 if (i+k) == j0 else h[i+k])
                d += hi_n * (1.0 - hik_n) - hi_o * (1.0 - hik_o)
        if d != 0.0:
            delta[k + n - 1] = d
    return delta

def local_search(h, n, deadline, max_sweeps=10000, seed=42):
    h = binarize(h, n).copy()
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_c = float(corr.max()) * dx
    best_c = cur_c
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        rng.shuffle(ones)
        rng.shuffle(zeros)
        improved = False
        for i0 in zeros:
            found = False
            for j0 in ones:
                d = swap_delta_c(h, int(i0), int(j0), n)
                nc = float((corr + d).max()) * dx
                if nc < cur_c - 1e-15:
                    h[i0] = 1.0
                    h[j0] = 0.0
                    corr = corr + d
                    cur_c = nc
                    if nc < best_c:
                        best_c = nc
                        best_h = h.copy()
                    improved = True
                    found = True
                    break
            if found:
                break
        if not improved:
            break
    return best_h, best_c

def slsqp_refine(h, n, alpha, maxiter=200):
    dx = 2.0 / n
    def objective(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        log_sum = sm + np.log(np.exp(shifted - sm).sum())
        return m + log_sum / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(objective, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.success or res.fun < objective(h):
            return np.clip(res.x, 0, 1)
    except:
        pass
    return h

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)
