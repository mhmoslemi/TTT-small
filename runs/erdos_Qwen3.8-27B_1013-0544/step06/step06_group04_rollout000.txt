    # where g = 1-h
    g = 1.0 - h
    # Create a Toeplitz-like matrix: T[i,m] = w[n-1+m] if 0<=i-m<n else 0
    # r = T @ g
    # But building T is O(n²) memory
    # Alternatively, use the FFT approach
    
    # Let me use a simpler vectorized approach:
    # r[i] = sum over valid m of w[n-1+m] * g[i-m]
    # For each m, this is w[n-1+m] * g shifted by m
    # r = sum_m w[n-1+m] * shift(g, m)
    
    r = np.zeros(n)
    for m in range(-(n-1), n):
        idx = n - 1 + m
        if 0 <= idx < 2*n - 1:
            # shift g by m: g_shifted[i] = g[i-m] if 0 <= i-m < n else 0
            lo = max(0, m)
            hi = min(n, n + m)
            if lo < hi:
                r[lo:hi] += w[idx] * g[lo-m:hi-m]
    
    return dx * (r - h * w[n-1])

    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-14:
                h += diff / s * room
        else:
            s = h.sum()
            if s > 1e-14:
                h *= target / s
    return np.clip(h, 0.0, 1.0)

def local_search(h, n, deadline, seed=42, max_sweeps=200000):
    h = binarize(h, n).copy()
    cur_c = compute_c5(h, n)
    best_c = cur_c
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        rng.shuffle(ones)
        rng.shuffle(zeros)
        improved = False
        for i in zeros:
            if time.time() > deadline:
                return best_h, best_c
            for j in ones:
                h[i] = 1.0; h[j] = 0.0
                nc = compute_c5(h, n)
                if nc < cur_c - 1e-15:
                    cur_c = nc
                    if nc < best_c:
                        best_c = nc; best_h = h.copy()
                    improved = True; break
                else:
                    h[i] = 0.0; h[j] = 1.0
            if improved:
                break
        if not improved:
            break
    return best_h, best_c

def local_search_2swap(h, n, deadline, seed=42, max_iters=500000):
    h = binarize(h, n).copy()
    cur_c = compute_c5(h, n)
    best_c = cur_c
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i1, i2 = rng.choice(zeros, 2, replace=False)
        j1, j2 = rng.choice(ones, 2, replace=False)
        h[i1] = 1.0; h[i2] = 1.0
        h[j1] = 0.0; h[j2] = 0.0
        nc = compute_c5(h, n)
        if nc < cur_c - 1e-15:
            cur_c = nc
            if nc < best_c:
                best_c = nc; best_h = h.copy()
        else:
            h[i1] = 0.0; h[i2] = 0.0
            h[j1] = 1.0; h[j2] = 1.0
    return best_h, best_c

def grad_vec(h, w, n, dx):
    g = 1.0 - h
    r = np.zeros(n)
    for m in range(-(n - 1), n):
        idx = n - 1 + m
        lo = max(0, m)
        hi = min(n, n + m)
        if lo < hi:
            r[lo:hi] += w[idx] * g[lo - m:hi - m]
    return dx * (r - h * w[n - 1])

def continuous_opt(h, n, deadline, alpha=50000, lr0=0.0005, max_iter=200000):
    h = h.astype(float).copy()
    h = project_feasible(h, n)
    dx = 2.0 / n
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = lr0
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c = fast_full_corr(h, 1.0 - h)
        cdx = c * dx
        m = cdx.max()
        sh = alpha * (cdx - m)
        sm = sh.max()
        w = np.exp(sh - sm)
        w /= w.sum()
        grad = grad_vec(h, w, n, dx)
        h = h - lr * grad
        h = project_feasible(h, n)
        c_new = compute_c5(h, n)
        if c_new < best_c - 1e-10:
            best_c = c_new; best_h = h.copy(); stall = 0
        else:
            stall += 1
        if stall % 2000 == 1999:
            lr *= 0.9
            if lr < 1e-8:
                break
    return best_h, best_c

def make_pattern(n, freq, phase, seed=0):
    x = np.arange(n) * 2.0 / n
    h = 0.5 + 0.4 * np.sin(2 * np.pi * freq * x + phase)
    h = np.clip(h, 0, 1)
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:n // 2]] = 1.0
    return b

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    best_c = 1.0
    best_h = None
    best_n = 250

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 250
        h0 = np.full(n0, 0.5)

    # Phase 1: Binary local search on initial
    p1 = t0 + 0.18 * (budget_s - 5)
    hb, cb = local_search(h0, n0, p1, seed=seed)
    best_c, best_h, best_n = cb, hb, n0

    # Phase 2: Try different n values
    p2 = t0 + 0.40 * (budget_s - 5)
    for n_try in [200, 280, 320, 400, 500]:
        if time.time() > p2 - 10:
            break
        if n_try == n0:
            continue
        h_init = make_pattern(n_try, 3.5 + rng.rand(), rng.rand() * 6.28)
        t_sub = min(p2 - 5, time.time() + (p2 - time.time()) / 3.0)
        hb2, cb2 = local_search(h_init, n_try, t_sub, seed=seed + n_try)
        if cb2 < best_c:
            best_c, best_h, best_n = cb2, hb2, n_try

    # Phase 3: 2-swap local search on best
    p3 = t0 + 0.55 * (budget_s - 5)
    hb3, cb3 = local_search_2swap(best_h, best_n, p3, seed=seed + 123)
    if cb3 < best_c:
        best_c, best_h = cb3, hb3

    # Phase 4: Continuous optimization
    p4 = t0 + 0.82 * (budget_s - 5)
    h_cont = best_h.astype(float).copy()
    for alpha in [10000, 50000, 200000]:
        if time.time() > p4 - 10:
            break
        t_sub = min(p4 - 5, time.time() + 30)
        hc, cc = continuous_opt(h_cont, best_n, t_sub, alpha=alpha, lr0=0.0003)
        if cc < best_c:
            best_c, best_h = cc, hc
        h_cont = best_h.astype(float)

    # Phase 5: Final binary local search + continuous polish
    if time.time() < deadline - 20:
        hb5, cb5 = local_search(best_h, best_n, deadline - 25, seed=seed + 999)
        if cb5 < best_c:
            best_c, best_h = cb5, hb5
    if time.time() < deadline - 5:
        hc5, cc5 = continuous_opt(best_h, best_n, deadline - 2, alpha=200000, lr0=0.0001)
        if cc5 < best_c:
            best_c, best_h = cc5, hc5

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
