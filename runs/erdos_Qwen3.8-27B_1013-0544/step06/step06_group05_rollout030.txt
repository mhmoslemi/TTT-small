For the output at index k (0-indexed, length 2n-1):
c[k] = sum_i h[i] * (1-h)[i-(k-(n-1))]

Let lag = k - (n-1). Then:
c[k] = sum_i h[i] * (1-h[i-lag]) for valid i and i-lag

When we swap h[a]: 1→0 and h[b]: 0→1:

For each lag, the change in c[k] is:
- Term where i=a: was h[a]*(1-h[a-lag]) = 1*(1-h[a-lag]), now 0*(1-h[a-lag]) = 0
  Change: -(1-h[a-lag])
- Term where i-lag=a, i=a+lag: was h[a+lag]*(1-h[a]) = h[a+lag]*0 = 0, now h[a+lag]*(1-0) = h[a+lag]
  Change: +h[a+lag]
- Term where i=b: was h[b]*(1-h[b-lag]) = 0, now 1*(1-h[b-lag]) = 1-h[b-lag]
  Change: +(1-h[b-lag])
- Term where i-lag=b, i=b+lag: was h[b+lag]*(1-h[b]) = h[b+lag]*1 = h[b+lag], now h[b+lag]*(1-1) = 0
  Change: -h[b+lag]

So the total change for lag k (with lag value ℓ = k-(n-1)):
Δc[k] = -(1-h[a-ℓ]) + h[a+ℓ] + (1-h[b-ℓ]) - h[b+ℓ]
       = -1 + h[a-ℓ] + h[a+ℓ] + 1 - h[b-ℓ] - h[b+ℓ]
       = h[a-ℓ] + h[a+ℓ] - h[b-ℓ] - h[b+ℓ]

(Where h[x] = 0 if x is out of range)

This is a nice formula! It means for a given swap (a,b), the change in correlation at each lag is:
Δc(ℓ) = h[a-ℓ] + h[a+ℓ] - h[b-ℓ] - h[b+ℓ]

This is O(n) to compute for all lags given a swap. And we want to find a swap that reduces the maximum.

This is much more efficient than recomputing the full correlation after each swap!

Actually, let me think about this differently. I'll precompute for each position p the vector v_p[ℓ] = h[p-ℓ] + h[p+ℓ] for all ℓ. This is a 250 × 499 matrix. Then for a swap (a, b), the condition is:
v_a[critical] < v_b[critical] for all critical lags.

And the new correlation at lag ℓ would be: corr[ℓ] + v_a[ℓ] - v_b[ℓ]

So the new max is max_ℓ (corr[ℓ] + v_a[ℓ] - v_b[ℓ]).

We want to find (a, b) that minimizes this new max.

This is a minimax problem over all (a, b) pairs. We can solve it by:
1. For each a, compute the vector d_a = corr + v_a (shape 499)
2. For each b, the new correlation is d_a - v_b
3. We want min_{a,b} max_ℓ (d_a[ℓ] - v_b[ℓ])

For a fixed a, this is equivalent to finding b that minimizes max_ℓ (d_a[ℓ] - v_b[ℓ]).

This is still a minimax, but for a fixed a, we can try all b and pick the best. Total: O(n² * 499) = O(250² * 499) ≈ 31M operations. In numpy with vectorization, this should be fast.

```python
def compute_v_matrix(h, n):
    """Compute v[p, ℓ] = h[p-ℓ] + h[p+ℓ] for all p and ℓ."""
    lags = np.arange(-(n-1), n)
    # For each p, p+lag ranges over... 
    # v[p, idx] = h[p - lags[idx]] (if valid) + h[p + lags[idx]] (if valid)
    # = h[p - lags[idx]] + h[p + lags[idx]]
    
    # For h[p + lags[idx]]: this is h[j] where j = p + lags[idx]
    # For a fixed idx (lag), h[p + lag] for all p is a shifted version of h
    # Specifically, h[p + lag] = h[p] shifted by -lag
    
    # h[p - lag] = h shifted by +lag
    # h[p + lag] = h shifted by -lag
    
    # So v[p, idx] = roll(h, lag)[p] + roll(h, -lag)[p]
    # where roll handles boundaries by wrapping (but we want zero-padding)
    
    # Let's use a different approach:
    # v = zeros(n, 2n-1)
    # for each lag (index idx in lags):
    #   v[:, idx] = zero_padded_shift(h, lag) + zero_padded_shift(h, -lag)
    
    n_lags = 2*n - 1
    v = np.zeros((n, n_lags))
    for idx, lag in enumerate(lags):
        # h[p + lag] for p in 0..n-1:
        # valid when 0 <= p+lag < n, i.e., -lag <= p < n-lag
        term1 = np.zeros(n)
        lo1 = max(0, -lag)
        hi1 = min(n, n - lag)
        if lo1 < hi1:
            term1[lo1:hi1] = h[lo1+lag:hi1+lag]
        
        # h[p - lag] for p in 0..n-1:
        # valid when 0 <= p-lag < n, i.e., lag <= p < n+lag
        term2 = np.zeros(n)
        lo2 = max(0, lag)
        hi2 = min(n, n + lag)
        if lo2 < hi2:
            term2[lo2:hi2] = h[lo2-lag:hi2-lag]
        
        v[:, idx] = term1 + term2
    
    return v
```

This is O(n * (2n-1)) = O(n²) which for n=250 is about 125K operations. Fast.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * room, 0.0, 1.0)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * h, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n // 2
    order = np.argsort(-h)
    hb = np.zeros(n)
    hb[order[:target]] = 1.0
    return hb

def compute_v_matrix(h, n):
    """v[p, idx] = h[p-lag] + h[p+lag] where lag = idx-(n-1)"""
    n_lags = 2 * n - 1
    v = np.zeros((n, n_lags))
    for idx in range(n_lags):
        lag = idx - (n - 1)
        # term1: h[p+lag] valid for p in [max(0,-lag), min(n, n-lag))
        # term2: h[p-lag] valid for p in [max(0,lag), min(n, n+lag))
        term1 = np.zeros(n)
        lo1 = max(0, -lag)
        hi1 = min(n, n - lag)
        if lo1 < hi1:
            term1[lo1:hi1] = h[lo1+lag:hi1+lag]
        term2 = np.zeros(n)
        lo2 = max(0, lag)
        hi2 = min(n, n + lag)
        if lo2 < hi2:
            term2[lo2:hi2] = h[lo2-lag:hi2-lag]
        v[:, idx] = term1 + term2
    return v

def optimal_swap_search(h, n, deadline, max_iter=50000):
    """Find optimal swap at each step using vectorized computation."""
    dx = 2.0 / n
    h = h.copy().astype(float)
    corr = compute_corr(h)
    cur_max = corr.max()
    best_h = h.copy()
    best_max = cur_max
    stall = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        n_ones = len(ones)
        n_zeros = len(zeros)
        
        if n_ones == 0 or n_zeros == 0:
            break
        
        v = compute_v_matrix(h, n)
        
        # c_all[a, lag] = corr[lag] + v[ones[a], lag]
        c_all = corr[np.newaxis, :] + v[ones]  # (n_ones, 2n-1)
        v_zeros = v[zeros]  # (n_zeros, 2n-1)
        
        # new_max(a,b) = max_lag (c_all[a,lag] - v_zeros[b,lag])
        diff = c_all[:, np.newaxis, :] - v_zeros[np.newaxis, :, :]
        new_maxes = diff.max(axis=2)
        
        min_val = new_maxes.min()
```

```python
def compute_v_matrix(h, n):
    """v[p, idx] = h[p-lag] + h[p+lag] where lag = idx-(n-1)"""
    n_lags = 2 * n - 1
    lags = np.arange(-(n-1), n)  # shape (2n-1,)
    
    # For each lag, we need:
    # term1[p] = h[p+lag] if 0 <= p+lag < n, else 0
    # term2[p] = h[p-lag] if 0 <= p-lag < n, else 0
    # v[:, idx] = term1 + term2
    
    # Vectorized approach using broadcasting:
    # p ranges over 0..n-1 (shape n)
    # lag ranges over -(n-1)..n-1 (shape 2n-1)
    # p+lag ranges over -(n-1)..2n-2
    # p-lag ranges over 0-(n-1)..n-1-(-(n-1)) = -(n-1)..2n-2
    
    p = np.arange(n)[:, np.newaxis]  # (n, 1)
    lag_vals = lags[np.newaxis, :]  # (1, 2n-1)
    
    idx1 = p + lag_vals  # (n, 2n-1): p+lag
    idx2 = p - lag_vals  # (n, 2n-1): p-lag
    
    valid1 = (idx1 >= 0) & (idx1 < n)  # (n, 2n-1)
    valid2 = (idx2 >= 0) & (idx2 < n)
    
    # Clip for safe indexing
    idx1_clipped = np.clip(idx1, 0, n-1)
    idx2_clipped = np.clip(idx2, 0, n-1)
    
    v = np.where(valid1, h[idx1_clipped], 0.0) + np.where(valid2, h[idx2_clipped], 0.0)
    return v
```

```python
    idx1_c = np.clip(idx1, 0, n - 1)
    idx2_c = np.clip(idx2, 0, n - 1)
    return np.where(valid1, h[idx1_c], 0.0) + np.where(valid2, h[idx2_c], 0.0)

def optimal_swap_search(h, n, deadline, max_iter=50000):
    """Systematic local search: find the best single swap at each step."""
    dx = 2.0 / n
    h = h.copy().astype(float)
    corr = compute_corr(h)
    cur_max = corr.max()
    best_h = h.copy()
    best_max = cur_max
    stall = 0

    for it in range(max_iter):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        v = compute_v_matrix(h, n)
        c_all = corr[np.newaxis, :] + v[ones]
        v_zeros = v[zeros]
        diff = c_all[:, np.newaxis, :] - v_zeros[np.newaxis, :, :]
        new_maxes = diff.max(axis=2)
        min_val = new_maxes.min()
        if min_val < cur_max - 1e-14:
            a_idx, b_idx = np.unravel_index(new_maxes.argmin(), new_maxes.shape)
            a = ones[a_idx]
            b = zeros[b_idx]
            h[a] = 0.0
            h[b] = 1.0
            corr = compute_corr(h)
            cur_max = corr.max()
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 5:
                break
    return best_h, float(best_max * dx)

def anneal_search(h, n, deadline, T0=1.0, max_iter=50000, seed=777):
    """Simulated annealing to escape local optima."""
    dx = 2.0 / n
    h = h.copy().astype(float)
    corr = compute_corr(h)
    cur_max = corr.max()
    best_h = h.copy()
    best_max = cur_max
    rng = np.random.RandomState(seed)
    T = T0
    cool = (0.005 / T0) ** (1.0 / max(max_iter, 1))

    for it in range(max_iter):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        a = rng.choice(ones)
        b = rng.choice(zeros)
        h[a] = 0.0
        h[b] = 1.0
        new_corr = compute_corr(h)
        new_max = new_corr.max()
        delta = new_max - cur_max
        if delta <= 0 or rng.rand() < np.exp(-delta / (T + 1e-12)):
            cur_max = new_max
            corr = new_corr
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
        else:
            h[a] = 1.0
            h[b] = 0.0
        T *= cool
    return best_h, float(best_max * dx)

def slsqp_refine(h, n, deadline):
    """SLSQP continuous refinement with log-sum-exp smooth max."""
    dx = 2.0 / n
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def objective(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        alpha = 500.0
        shifted = alpha * (c - m)
        return m + np.log(np.exp(shifted).sum()) / alpha

    def constraint_eq(x):
        return x.sum() - target_sum

    try:
        res = minimize(objective, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': constraint_eq}],
                       options={'maxiter': 2000, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        h0 = None
        n0 = 250

    best_h = None
    best_c = float('inf')
    best_n = 0

    # Build n list with n0 first, then neighbors
    n_candidates = sorted(set([n0 - 20, n0 - 10, n0, n0 + 10, n0 + 20, n0 + 40, n0 + 50]))
    n_candidates = [n for n in n_candidates if 100 <= n <= 350 and n % 2 == 0]
    n_candidates.remove(n0)
    n_candidates.insert(0, n0)

    total_time = deadline - t0
    # Allocate 40% to primary n, rest split among others
    primary_time = total_time * 0.45
    other_time = (total_time - primary_time) / max(len(n_candidates) - 1, 1)

    for ni, n in enumerate(n_candidates):
        if time.time() > deadline:
            break
        if ni == 0:
            n_deadline = t0 + primary_time
        else:
            n_deadline = min(deadline, t0 + primary_time + other_time * ni)

        if time.time() > n_deadline:
            continue

        # Create initial point
        if h0 is not None and n == n0:
            h_init = project_feasible(h0.copy(), n)
        elif h0 is not None:
            x_old = np.linspace(0, 2, n0)
            x_new = np.linspace(0, 2, n)
            h_interp = np.interp(x_new, x_old, h0)
            h_init = project_feasible(h_interp, n)
        else:
            rng = np.random.RandomState(seed + n)
            h_init = np.zeros(n)
            h_init[rng.choice(n, n // 2, replace=False)] = 1.0

        c = compute_c5(h_init, n)

        # Phase 1: Optimal swap search (main workhorse)
        h1, c1 = optimal_swap_search(h_init, n, n_deadline - 20)
        if c1 < c:
            h_init, c = h1, c1

        # Phase 2: Annealing for diversity
        if time.time() < n_deadline - 40:
            h2, c2 = anneal_search(h_init, n, n_deadline - 40, T0=0.8, max_iter=30000, seed=seed + 1)
            if c2 < c:
                h_init, c = h2, c2
            # Re-run optimal swap on the annealed result
            h3, c3 = optimal_swap_search(h_init, n, n_deadline - 20)
            if c3 < c:
                h_init, c = h3, c3

        # Phase 3: Second round of annealing from best
        if time.time() < n_deadline - 30:
            h4, c4 = anneal_search(h_init, n, n_deadline - 30, T0=0.3, max_iter=20000, seed=seed + 2)
            if c4 < c:
                h_init, c = h4, c4
            h5, c5 = optimal_swap_search(h_init, n, n_deadline - 10)
            if c5 < c:
                h_init, c = h5, c5

        # Phase 4: SLSQP continuous refinement
        if time.time() < n_deadline - 5:
            h6 = slsqp_refine(h_init, n, n_deadline)
            c6 = compute_c5(h6, n)
            if c6 < c:
                h_init, c = h6, c6
            # Binarize and refine
            h7 = binarize(h_init, n)
            h8, c8 = optimal_swap_search(h7, n, n_deadline)
            if c8 < c:
                h_init, c = h8, c8

        if c < best_c:
            best_c = c
            best_h = h_init.copy()
            best_n = n

    if best_h is None:
        best_h = project_feasible(np.full(n0, 0.5), n0)
        best_c = compute_c5(best_h, n0)
        best_n = n0

    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
```
