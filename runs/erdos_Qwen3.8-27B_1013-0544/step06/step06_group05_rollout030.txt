def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def full_corr(h):
    return np.correlate(h, 1.0 - h, mode='full').astype(np.float64)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5000):
        diff = target - h.sum()
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * room, 0.0, 1.0)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * h, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def greedy_binary_search(h_init, n, deadline, max_iter=50000, seed=42):

    best_max = cur_max
    offset = n - 1
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        # Find critical lags (within 1 of max)

        crit_idx = np.where(crit_mask)[0]
        n_crit = len(crit_idx)
        if n_crit == 0:
            break

        zeros_idx = np.where(h == 0)[0]
        ones_idx = np.where(h == 1)[0]
        if len(zeros_idx) == 0 or len(ones_idx) == 0:
            break
        
        k_vals = crit_idx - offset  # actual lag values
        
        # Compute gradient for zeros: g_i = sum over critical of (1-h[i-k]) - h[i+k]

            # -h[i+k]: valid when 0 <= i+k < n

        # grad_ones[j] = (1-h[j-k]) - h[j+k], and removing j decreases corr by grad_ones[j]
        # So we want large positive grad_ones
        grad_ones = np.zeros(len(ones_idx))
        for t, k in enumerate(k_vals):
            idx_a = ones_idx - k
            valid_a = (idx_a >= 0) & (idx_a < n)

            idx_b = ones_idx + k

            h[best_o] = 1.0

            for _ in range(20):

                oi = rng.randint(len(ones_idx))

                j = ones_idx[oi]

                h[j] = 0.0

                new_max = float(new_corr.max())

                    cur_max = new_max

def slsqp_optimize(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    nc = 2 * n - 1
    
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    
    def obj_grad(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        sm = sh.max()
        w = np.exp(sh - sm)
        w /= w.sum()
        # grad of m + sum(w * (c-m)) ... this is complex
        # Just use numerical gradient via finite differences? No, too slow.
        # Use the fact that grad of logsumexp(alpha*c) = alpha * softmax(alpha*c)
        # And grad of m is just the indicator of argmax
        # Total grad = grad_m + alpha * softmax(alpha*(c-m)) / ... 
        # Actually let's just let scipy compute it numerically
        return None  # Use numerical gradient
    
    def cons_fun(x):
        return x.sum() - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})

        return h
