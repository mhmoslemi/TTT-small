import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:
                break
            h = h + (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h = h + (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def _smo(x, n, alpha):
    dx = 2.0 / n
    g = np.correlate(x, 1 - x, mode='full') * dx
    m = g.max()
    sh = alpha * (g - m)
    sm = sh.max()
    return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

def _con_fun(x, n):
    return x.sum() - n / 2.0

def _con_jac(x, n):
    return np.ones(n)

def do_slsqp(h, n, alpha, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (n,)}]
    try:
        res = minimize(_smo, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.fun <= _smo(h, n, alpha) + 1e-10:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

                    no_imp = 0

                    no_imp += 1
