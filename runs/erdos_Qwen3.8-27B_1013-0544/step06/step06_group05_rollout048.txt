def compute_c5(h, n):

    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_binary(h, n):
    """Project to binary with exactly n/2 ones"""
    h = (h >= 0.5).astype(float)
    n_ones = int(h.sum())
    target = n // 2
    if n_ones > target:
        # Remove excess ones (give them to zeros with highest h value... but h is binary)
        # Just set random excess ones to 0
        ones_idx = np.where(h == 1)[0]
        excess = ones_idx[n_ones - target:]
        h[excess] = 0.0
    elif n_ones < target:
        zeros_idx = np.where(h == 0)[0]
        deficit = zeros_idx[:target - n_ones]
        h[deficit] = 1.0
    return h

def get_delta(h, i, j, n):
    """Compute Δc[k] for all k for swap h[i]:1->0, h[j]:0->1. Returns array of length 2n-1."""
    # Index offset: k = -(n-1) corresponds to index 0
    off = n - 1
    delta = np.zeros(2 * n - 1)
    
    # Second index i (1->0): for k where 0<=i+k<n, delta[k] += h[i+k]
    k_lo = -i
    k_hi = n - 1 - i
    idx_lo = max(k_lo, -(n-1)) + off
    idx_hi = min(k_hi, n-1) + off
    if idx_lo <= idx_hi:
        # h[i+k] for k in [k_lo, k_hi]
        # i+k ranges from i+k_lo to i+k_hi
        delta[idx_lo:idx_hi+1] += h[i+max(k_lo,-(n-1)):i+min(k_hi,n-1)+1]
    
    # Second index j (0->1): for k where 0<=j+k<n, delta[k] -= h[j+k]
    k_lo = -j
    k_hi = n - 1 - j
    idx_lo = max(k_lo, -(n-1)) + off
    idx_hi = min(k_hi, n-1) + off
    if idx_lo <= idx_hi:
        delta[idx_lo:idx_hi+1] -= h[j+max(k_lo,-(n-1)):j+min(k_hi,n-1)+1]
    
    # First index i (1->0): for k where 0<=i-k<n, delta[k] += -(1-h[i-k]) = h[i-k]-1

    k_hi = i

        k_actual_lo = max(k_lo, -(n-1))

        vals = h[i-k_actual_hi:i-k_actual_lo+1][::-1]

    # First index j (0->1): for k where 0<=j-k<n, delta[k] += 1-h[j-k]

    k2 = i - j
    if -(n-1) <= k2 <= n-1:
        delta[k2 + off] += 1.0

    off = n - 1
    delta = np.zeros(2 * n - 1)
    k_min = -(n - 1)
    k_max = n - 1
    
    # Term: h[i+k] for k in [max(-i, k_min), min(n-1-i, k_max)]
    lo = max(-i, k_min)
    hi = min(n - 1 - i, k_max)
    if lo <= hi:
        delta[lo + off : hi + off + 1] += h[i + lo : i + hi + 1]
    
    # Term: -h[j+k] for k in [max(-j, k_min), min(n-1-j, k_max)]
    lo = max(-j, k_min)
    hi = min(n - 1 - j, k_max)
    if lo <= hi:
        delta[lo + off : hi + off + 1] -= h[j + lo : j + hi + 1]
    
    # Term: (h[i-k] - 1) for k in [max(i-(n-1), k_min), min(i, k_max)]
    lo = max(i - (n-1), k_min)
