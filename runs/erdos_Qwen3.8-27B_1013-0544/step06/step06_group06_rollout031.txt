import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def thue_morse_init(n, rng):
    t = np.zeros(n, dtype=int)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    target = n // 2
    num_ones = int(t.sum())
    if num_ones > target:
        ones = np.where(t == 1)[0].copy()
        rng.shuffle(ones)
        t[ones[:num_ones - target]] = 0
    elif num_ones < target:
        zeros = np.where(t == 0)[0].copy()
        rng.shuffle(zeros)
        t[zeros[:target - num_ones]] = 1
    return t.astype(float)

def make_binary(h, n, rng):
    hb = (h > 0.5).astype(float)
    target = n // 2
    n1 = int(hb.sum())
    if n1 > target:
        ones = np.where(hb == 1)[0].copy()
        rng.shuffle(ones)
        hb[ones[:n1 - target]] = 0
    elif n1 < target:
        zeros = np.where(hb == 0)[0].copy()
        rng.shuffle(zeros)
        hb[zeros[:target - n1]] = 1
    return hb

def binary_search(h, n, deadline, seed=42):

    dx = 2.0 / n
    h = h.copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    stall = 0
    no_improve = 0
    temp = 0.004
    total_iterations = 0

    while time.time() < deadline:
        total_iterations += 1
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        improved = False
        # Structured greedy: focus on worst lag
        worst_k = np.argmax(corr)
        q = worst_k - (n - 1)

        # Compute "desirability" of moving each 1 to each 0
        # Score for moving 1 at j to 0 at i: reduction in c[worst_k]
        # = (1-h[j+q]) - h[j-q] - [(1-h[i+q]) - h[i-q]] (with boundary handling)
        # We want this to be negative (reduces c)
        
        # Vectorized: compute cost of each 1 position and each 0 position
        def pos_cost(pos, h_arr, q_val):
            costs = np.zeros(len(pos))
            for a, p in enumerate(pos):
                c = 0.0
                if 0 <= p + q_val < len(h_arr):
                    c += 1.0 - h_arr[p + q_val]
                if 0 <= p - q_val < len(h_arr):
                    c -= h_arr[p - q_val]

                    c -= 1.0 - 2.0 * h_arr[p]
                costs[a] = c
            return costs

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h

def subgradient_step(h, n, lr, iters, deadline):
    h = h.copy()
    for _ in range(iters):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1 - h, mode='full')
        k_star = np.argmax(c)
        q = k_star - (n - 1)
        grad = np.zeros(n)
        for i in range(n):
            g = 0.0
            if 0 <= i + q < n:
                g += 1.0 - h[i + q]
            if 0 <= i - q < n:
                g -= h[i - q]
            if q == 0:
                g -= 1.0 - 2.0 * h[i]
            grad[i] = g
        h = project_feasible(h - lr * grad, n)
    return h
