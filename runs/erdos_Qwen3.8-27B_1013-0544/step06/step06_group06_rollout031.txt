import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def build_g(h, n):
    nc = 2 * n - 1
    g = np.zeros((n, nc))
    for p in range(n):
        ks = max(0, p)
        ke = min(nc - 1, p + n - 1)
        for k in range(ks, ke + 1):
            if k != 2 * p:
                g[p, k] = 1.0 - 2.0 * h[k - p]
    return g

def thue_morse_init(n, rng):
    t = np.zeros(n, dtype=int)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    target = n // 2
    num_ones = int(t.sum())
    if num_ones > target:
        idx = np.where(t == 1)[0]
        rng.shuffle(idx)
        t[idx[:num_ones - target]] = 0
    elif num_ones < target:
        idx = np.where(t == 0)[0]
        rng.shuffle(idx)
        t[idx[:target - num_ones]] = 1
    return t.astype(float)

def beatty_init(n, rng, alpha=None):
    if alpha is None:
        alpha = (np.sqrt(5.0) - 1.0) / 2.0
    h = np.zeros(n)
    for i in range(n):
        frac = (i * alpha) % 1.0
        h[i] = 1.0 if frac < 0.5 else 0.0
    target = n // 2
    diff = target - int(h.sum())
    if diff > 0:
        idx = np.where(h == 0)[0]
        rng.shuffle(idx)
        h[idx[:diff]] = 1.0
    elif diff < 0:
        idx = np.where(h == 1)[0]
        rng.shuffle(idx)
        h[idx[:abs(diff)]] = 0.0
    return h

def block_init(n, rng, base_block=7, variation=3):
    h = np.zeros(n)
    target = n // 2
    pos = 0
    is_one = True
    while pos < n:
        block_len = base_block + rng.randint(-variation, variation + 1)
        block_len = max(1, min(block_len, n - pos))
        if is_one:
            h[pos:pos+block_len] = 1.0
        pos += block_len
        is_one = not is_one
    diff = target - int(h.sum())
    if diff > 0:
        idx = np.where(h == 0)[0]
        rng.shuffle(idx)
        h[idx[:diff]] = 1.0
    elif diff < 0:
        idx = np.where(h == 1)[0]
        rng.shuffle(idx)
        h[idx[:abs(diff)]] = 0.0
    return h

def local_search(h, n, deadline, tabu_size=30, top_k=15, seed=42):

    best_h = h.copy()
    ones = np.where(h == 1)[0].copy()

    no_improve = 0
    pert_count = 0
    g = build_g(h, n)

        nz = len(zeros)
        no = len(ones)
        M = corr[top_idx][None, :] + g[zeros][:, top_idx]
        D = M[:, None, :] - g[ones][None, :, top_idx]
        new_maxes = D.max(axis=2)
        tabu_mask = (tabu[zeros] > iteration)[:, None] | (tabu[ones] > iteration)[None, :]
        new_maxes_tabu = np.where(tabu_mask, 1e18, new_maxes)

                tabu[j] = iteration + tabu_size
                cur_c = actual_max * dx

            zeros = np.where(h == 0)[0].copy()
            ones = np.where(h == 1)[0].copy()

def slsqp_smooth(h, n, alpha, maxiter):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h
