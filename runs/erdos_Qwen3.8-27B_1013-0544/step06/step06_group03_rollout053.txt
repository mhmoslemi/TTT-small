    best_h, best_c = retarget_loop(h_curr, n, p1_end, alphas, top_k=3, max_rounds=25)
    if time.time() < deadline - 40:
        p2_end = min(deadline - 25, time.time() + t_remain * 0.2)
        h_new, c_new = gradient_descent_phase(best_h, n, p2_end, rng)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    if time.time() < deadline - 20:
        alphas2 = [300000, 500000, 1000000, 2000000]
        p3_end = min(deadline - 10, time.time() + t_remain * 0.2)
        best_h, best_c = retarget_loop(best_h, n, p3_end, alphas2, top_k=2, max_rounds=20)
    if time.time() < deadline - 5:
        h_new, c_new = pair_swap_search(best_h, n, deadline - 1)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 4
    rng = np.random.RandomState(seed)
    best_h = None
    best_c = float('inf')

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n_init = 300

    h_start = project_feasible(h_init, n_init)
    alloc1 = min(400, (deadline - t0) * 0.45)
    best_h, best_c = optimize_n(h_start, n_init, t0 + alloc1, rng)

    other_ns = [250, 350, 200, 400]
    for n_try in other_ns:
        if time.time() > deadline - 60:
            break
        if n_try == n_init:
            continue
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_try = np.interp(x_new, x_old, best_h)
        h_try = project_feasible(h_try, n_try)
        rem = deadline - time.time() - 30
        if rem < 50:
            break
        alloc = min(120, rem * 0.4)
        h_opt, c_opt = optimize_n(h_try, n_try, time.time() + alloc, rng)
        if c_opt < best_c - 1e-12:
            best_c = c_opt
            best_h = h_opt

    if best_h is not None and time.time() < deadline - 30:
        n = len(best_h)
        rem = deadline - time.time() - 10
        n_restarts = max(1, int(rem / 60))
        for r in range(n_restarts):
            if time.time() > deadline - 15:
                break
            scale = 0.01 + 0.01 * r
            hp = project_feasible(best_h + scale * rng.randn(n), n)
            alloc = min(50, (deadline - time.time() - 10) * 0.5)
            if alloc < 25:
                break
            h_opt, c_opt = optimize_n(hp, n, time.time() + alloc, rng)
            if c_opt < best_c - 1e-12:
                best_c = c_opt
                best_h = h_opt

    if best_h is not None and time.time() < deadline - 10:
        n = len(best_h)
        h_new = slsqp_opt(best_h, n, 3000000, maxiter=5000, ftol=1e-16)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        h_new, c_new = pair_swap_search(best_h, n, deadline - 1)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
