def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def smooth_max_obj(h, n, alpha, top_idx=None):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    if top_idx is not None:
        c = c[top_idx]
    m = c.max()
    shifted = alpha * (c - m)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

    return smooth_max_obj(h, n, alpha, top_idx)

    """KKT-based local search: alternate between finding active weights and doing projected gradient steps."""
    h = h.copy()
    best_c = compute_c5(h, n)

# Helper functions (top-level, no closures)

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(len(x))

def _smooth_obj(x, n, alpha, top_idx):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    c = c[top_idx]
    m = c.max()
    shifted = alpha * (c - m)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_refine(h, n, alpha, top_m, maxiter, ftol):
    c = compute_all_lags(h, n)
    top_idx = np.argsort(c)[-top_m:]
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(_smooth_obj, h, args=(n, alpha, top_idx),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def lag_gradient(h, n, lag):
    """Gradient of c_lag with respect to h."""
    grad = np.zeros(n)
    m = abs(lag)
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
    elif m >= n:
        return grad
    else:
        if lag > 0:
            grad[:n-m] -= h[m:]
            grad[m:] += (1.0 - h[:n-m])
        else:
            grad[n-m:] -= h[:m]
            grad[:n-m] += (1.0 - h[n-m:])
    return grad

def kkt_polish(h, n, deadline, top_m=10, max_iters=5000):
    """KKT-based polishing using weighted gradient descent."""
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        current_max = c.max()
        
        # Find active lags and weights
        top_idx = np.argsort(c)[-top_m:]
        top_vals = c[top_idx]
        
        # Softmax weights (temperature 1)
        shifted = top_vals - top_vals.max()
        w = np.exp(500 * shifted)
        w /= w.sum()
        
        # Combined gradient
        grad = np.zeros(n)
        for wi, ki in zip(w, top_idx):
            lag = int(ki) - (n - 1)
            grad += wi * lag_gradient(h, n, lag)
        
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Backtracking line search
        direction = -grad / gnorm
        step = 0.01
        found = False
        for _ in range(30):
            h_trial = project_feasible(h + step * direction, n)
            c_trial = compute_c5(h_trial, n)
            if c_trial < current_max - 1e-16:
                if c_trial < best_c:
                    best_c = c_trial
                    best_h = h_trial.copy()
                h = h_trial
                found = True
                break
            step *= 0.7
        
        if not found and step < 1e-10:
            break
    
    return best_h, best_c

def coordinate_polish(h, n, deadline, max_passes=10):
    """Coordinate descent: adjust one element at a time."""
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    deltas = [0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for d in deltas:
        if time.time() > deadline:
            break
        for _ in range(max_passes):
            if time.time() > deadline:
                break
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                for sign in [1, -1]:
                    h_trial = h.copy()
                    h_trial[i] += sign * d
                    h_trial = project_feasible(h_trial, n)
                    c_trial = compute_c5(h_trial, n)
                    if c_trial < best_c - 1e-15:
                        best_c = c_trial
                        best_h = h_trial.copy()
                        h = h_trial
                        improved = True
                        break
            if not improved:
                break
    return best_h, best_c

        direction = -grad / gnorm

        found = False
        for _ in range(40):
            h_trial = project_feasible(h + step * direction, n)
            c_trial = compute_c5(h_trial, n)
            if c_trial < current_max - 1e-16:
                if c_trial < best_c:
                    best_c = c_trial
                    best_h = h_trial.copy()
                h = h_trial
                found = True
                break
            step *= 0.6
        if not found and step < 1e-12:
            break
    return best_h, best_c

def coordinate_polish(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    deltas = [0.003, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    for d in deltas:
        if time.time() > deadline:
            break
        for _ in range(5):
            if time.time() > deadline:
                break
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                for sign in [1, -1]:
                    h_trial = h.copy()
                    h_trial[i] += sign * d
                    h_trial = project_feasible(h_trial, n)
                    c_trial = compute_c5(h_trial, n)
                    if c_trial < best_c - 1e-15:
                        best_c = c_trial
                        best_h = h_trial.copy()
                        h = h_trial
                        improved = True
                        break
            if not improved:
                break
    return best_h, best_c

def optimize_at_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    if time.time() < deadline - 30:
        for alpha in [1000, 5000, 20000, 80000]:
            if time.time() > deadline - 30:
                break
            h_new = slsqp_refine(best_h, n, alpha, 5, 3000, 1e-15)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
                best_h = h_new
    if time.time() < deadline - 25:
        kkt_end = min(deadline - 12, time.time() + 70)
        h_new, c_new = kkt_polish(best_h, n, kkt_end)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    if time.time() < deadline - 20:
        for alpha in [200000, 1000000]:
            if time.time() > deadline - 20:
                break
            h_new = slsqp_refine(best_h, n, alpha, 8, 4000, 1e-16)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
                best_h = h_new
    if time.time() < deadline - 12:
        h_new, c_new = kkt_polish(best_h, n, min(deadline - 5, time.time() + 50), top_m=12)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    if time.time() < deadline - 3:
        h_new, c_new = coordinate_polish(best_h, n, deadline - 1)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 300
    best_h = None
    best_c = float('inf')
    if h_init is not None:
        alloc = min(280, (deadline - t0) * 0.35)
        h_opt, c_opt = optimize_at_n(h_init, n_init, t0 + alloc, rng)
        best_h, best_c = h_opt, c_opt
    n_candidates = [200, 250, 350, 400]
    for n_try in n_candidates:
        if time.time() > deadline - 120:
            break
        if best_h is not None and n_try == len(best_h):
            continue
        remaining = deadline - time.time() - 100
        if remaining < 90:
            break
        alloc = min(130, remaining * 0.3)
        if best_h is not None:
            x_best = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_best, best_h)
        elif h_init is not None:
            x_init = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_init, h_init)
        else:
            h_start = np.full(n_try, 0.5)
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_at_n(h_start, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    if best_h is not None and time.time() < deadline - 100:
        n = len(best_h)
        remaining = deadline - time.time() - 60
        n_restarts = max(2, int(remaining / 75))
        for r in range(n_restarts):
            if time.time() > deadline - 50:
                break
            scale = 0.006 * (1.0 + 0.6 * r)
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(65, (deadline - time.time() - 50) * 0.3)
            if alloc < 30:
                break
            h_opt, c_opt = optimize_at_n(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    if best_h is not None and time.time() < deadline - 15:
        n = len(best_h)
        h_new = slsqp_refine(best_h, n, 5000000, 10, 5000, 1e-18)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        if time.time() < deadline - 8:
            h_new, c_new = kkt_polish(best_h, n, deadline - 3, top_m=15)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        if time.time() < deadline - 2:
            h_new, c_new = coordinate_polish(best_h, n, deadline - 1)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    return best_h, best_c, len(best_h)
