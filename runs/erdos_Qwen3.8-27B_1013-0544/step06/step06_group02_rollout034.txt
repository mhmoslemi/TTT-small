def coord_descent_step(h, c, i, n):
    """Optimize coordinate i given current correlations c.
    Returns new value t and updated correlations."""
    # Compute B_s for all shifts
    # c[m] = sum_j h[j] * (1-h[j+s]) where s = m - (n-1)
    # B_s = (1-h[i+s])*I[0<=i+s<n] - h[i-s]*I[0<=i-s<n]
    # A_s = c_s - h[i]*(1-h[i+s])*I[0<=i+s<n] + h[i]*h[i-s]*I[0<=i-s<n]
    
    B = np.zeros(2*n-1)
    A = c.copy()
    
    for m in range(2*n-1):
        s = m - (n-1)
        b = 0.0
        a_correction = 0.0
        if 0 <= i+s < n:
            b += (1.0 - h[i+s])
            a_correction -= h[i] * (1.0 - h[i+s])
        if 0 <= i-s < n:
            b -= h[i-s]
            a_correction += h[i] * h[i-s]
        B[m] = b
        A[m] = c[m] + a_correction
    
    # Minimize max(A + B*t) over t in [0, 1]
    # Using ternary search
    a, b = 0.0, 1.0
    for _ in range(60):
        m1 = a + (b-a)/3.0
        m2 = b - (b-a)/3.0
        f1 = np.max(A + B*m1)
        f2 = np.max(A + B*m2)
        if f1 < f2:
            b = m2
        else:
            a = m1
    t = (a+b)/2.0
    
    # Update correlations
    delta = t - h[i]
    if abs(delta) > 1e-15:
        c = c + delta * B
    return t, c
