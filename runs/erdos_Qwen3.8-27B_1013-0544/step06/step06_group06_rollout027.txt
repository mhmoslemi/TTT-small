def project_feasible(h, n):

def smooth_obj_and_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    ac = alpha * c * dx
    M = ac.max()
    e = np.exp(ac - M)
    w = e / e.sum()
    smooth_val = (M + np.log(e.sum())) / alpha
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        s = k - n + 1
        wk = w[k] * dx
        lo = max(0, s)
        hi = min(n, s + n)
        if lo < hi:
            grad[lo:hi] += wk * (1.0 - h[lo-s:hi-s])
        lo2 = max(0, -s)
        hi2 = min(n, n - s)
        if lo2 < hi2:
            grad[lo2:hi2] -= wk * h[lo2+s:hi2+s]
    return smooth_val, grad

def smooth_optimize(h, n, alpha, maxiter=5000, deadline=None):

        # Projected gradient descent with adaptive step
        eps = 1e-7
        # Line search (simple backtracking)
        step = 1.0 / alpha
        for _ in range(20):
            h_new = project_feasible(h - step * grad, n)

            if val_new < val:
                break
            step *= 0.5
        else:
            break
        h = h_new
    return h

    best_h = h.copy()
    for it in range(maxiter):
        if deadline and time.time() > deadline:
            break
        val, grad = smooth_grad(h, n, alpha)
        step = 2.0 / (alpha * n)
        for _ in range(30):
            h_new = project_feasible(h - step * grad, n)
            val_new, _ = smooth_grad(h_new, n, alpha)
            if val_new < val:
                break
            step *= 0.5
        else:
            break
        h = h_new
        if val_new < best_val:
            best_val = val_new
            best_h = h.copy()
    return best_h

def thue_morse_init(n, rng):
    t = np.zeros(n, dtype=int)
    mask = 1
    for i in range(1, n):
        v = i
        ct = 0
        while v:
            ct ^= v & 1
            v >>= 1
        t[i] = ct
    target = n // 2
    num_ones = int(t.sum())
    if num_ones > target:
        ones = np.where(t == 1)[0]
        rng.shuffle(ones)
        t[ones[:num_ones - target]] = 0
    elif num_ones < target:
        zeros = np.where(t == 0)[0]
        rng.shuffle(zeros)
        t[zeros[:target - num_ones]] = 1
    return t.astype(float)

def quadratic_residue_init(n, rng):
    # Using a prime p close to n for construction
    h = np.zeros(n)
    p = 4 * n + 1
    while True:
        is_prime = True
        for f in range(2, int(np.sqrt(p)) + 1):
            if p % f == 0:
                is_prime = False
                break
        if is_prime:
            break
        p += 4
    for i in range(n):
        if i == 0:
            h[i] = 1.0
        elif pow(i, (p - 1) // 2, p) == p - 1:
            h[i] = 0.0
        else:
            h[i] = 1.0
    target = n // 2
    num_ones = int(h.sum())
    if num_ones > target:
        idx = np.where(h == 1)[0]
        rng.shuffle(idx)
        h[idx[:num_ones - target]] = 0.0
    elif num_ones < target:
        idx = np.where(h == 0)[0]
        rng.shuffle(idx)
        h[idx[:target - num_ones]] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = 0.5 * np.ones(n)

    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: Binary local search from parent (30% time)
    p1 = t0 + 0.30 * (budget_s - 5)
    hb, cb = binary_search_improved(h0, n, p1, top_k=12, seed=seed)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()

    # Phase 2: Smooth continuous optimization (20% time)
    p2 = t0 + 0.50 * (budget_s - 5)
    if time.time() < p2:
        h_smooth = project_feasible(best_h + 0.05 * rng.randn(n), n)
        for alpha in [2000, 10000, 50000]:
            rem = p2 - time.time() - 5
            if rem < 10:
                break
            h_smooth = smooth_optimize(h_smooth, n, alpha,
                                       maxiter=min(2000, int(rem * 5)), deadline=p2 - 3)
            c = compute_c5(h_smooth, n)
            if c < best_c:
                best_c = c
                best_h = h_smooth.copy()
        # Round smooth solution to binary
        h_bin = np.zeros(n)
        h_bin[np.argsort(h_smooth)[::-1][:n // 2]] = 1.0
        c_bin = compute_c5(h_bin, n)
        if c_bin < best_c:
            best_c = c_bin
            best_h = h_bin.copy()

    # Phase 3: Binary polish on best (25% time)
    p3 = t0 + 0.75 * (budget_s - 5)
    if time.time() < p3:
        hb, cb = binary_search_improved(best_h, n, p3, top_k=15, seed=seed + 500)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()

    # Phase 4: Alternative starts and smooth (15% time)
    p4 = dl - 3
    if time.time() < p4 - 10:
        # Thue-Morse start
        h_tm = thue_morse_init(n, rng)
        rem = min(60, (p4 - time.time()) / 2)
        if rem > 20:
            hb, cb = binary_search_improved(h_tm, n, time.time() + rem, top_k=10, seed=seed + 1000)
            if cb < best_c:
                best_c = cb
                best_h = hb.copy()
            # Quick smooth from Thue-Morse
            if time.time() < p4 - 15:
                hs = project_feasible(h_tm + 0.1 * rng.randn(n), n)
                hs = smooth_optimize(hs, n, 10000, maxiter=800, deadline=p4 - 10)
                c = compute_c5(hs, n)
                if c < best_c:
                    best_c = c
                    best_h = hs.copy()
                hb = np.zeros(n)
                hb[np.argsort(hs)[::-1][:n // 2]] = 1.0
                c = compute_c5(hb, n)
                if c < best_c:
                    best_c = c
                    best_h = hb.copy()

    # Phase 5: Final micro-polish with smooth
    if time.time() < dl - 2:
        hf = project_feasible(best_h + 0.02 * rng.randn(n), n)
        hf = smooth_optimize(hf, n, 100000, maxiter=500, deadline=dl - 1)
        c = compute_c5(hf, n)
        if c < best_c:
            best_c = c
            best_h = hf.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
