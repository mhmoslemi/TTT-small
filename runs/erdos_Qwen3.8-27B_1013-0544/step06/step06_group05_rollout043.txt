        order = np.argsort(-h[zero_indices])

            h_bin[zero_indices[order[k]]] = 1.0

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def precompute_A(h, n):
    mid = n - 1
    p = np.arange(n)
    A = np.zeros((n, 2*n-1), dtype=np.float64)
    for l_idx in range(2*n-1):
        lag = l_idx - mid
        p1 = p + lag
        p2 = p - lag
        v1 = (p1 >= 0) & (p1 < n)
        v2 = (p2 >= 0) & (p2 < n)
        if v1.any():
            A[v1, l_idx] += h[p1[v1]]
        if v2.any():
            A[v2, l_idx] += h[p2[v2]]
    return A

def apply_swap(h, A, corr, n, i, j):
    """Apply swap h[i]:0->1, h[j]:1->0. Updates h, A, corr in-place."""
    mid = n - 1
    old_A_i = A[i].copy()
    old_A_j = A[j].copy()
    
    # Update h
    h[i] = 1.0
    h[j] = 0.0
    
    # Update corr
    delta = old_A_j - old_A_i
    delta[mid] = 0.0
    corr += delta
    
    # Update A incrementally
    # h[i]: 0->1 means A[p, (i-p)+mid] += 1 and A[p, (p-i)+mid] += 1
    p1 = np.arange(max(0, i - n + 1), i + 1)
    A[p1, (i - p1) + mid] += 1.0
    p2 = np.arange(i, n)
    A[p2, (p2 - i) + mid] += 1.0

    p3 = np.arange(max(0, j - n + 1), j + 1)
    A[p3, (j - p3) + mid] -= 1.0
    p4 = np.arange(j, n)
    A[p4, (p4 - j) + mid] -= 1.0

            break

            diffs = C[ii][None, :] + A[ones, :]

                i = zeros[ii]

                cur_max = float(corr.max())
                found = True

def binarize(h, n):
    thresh = np.sort(h)[n // 2]
    h_bin = (h >= thresh).astype(np.float64)
    n_ones = int(h_bin.sum())
    if n_ones > n // 2:
        one_idx = np.where(h_bin == 1)[0]
        order = np.argsort(h[one_idx])
        for k in range(n_ones - n // 2):
            h_bin[one_idx[order[k]]] = 0.0
    elif n_ones < n // 2:
        zero_idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zero_idx])
        for k in range(n // 2 - n_ones):
            h_bin[zero_idx[order[k]]] = 1.0
    return h_bin

def apply_swap(h, A, corr, n, i, j):
    mid = n - 1
    old_A_i = A[i].copy()
    old_A_j = A[j].copy()
    h[i] = 1.0
    h[j] = 0.0
    delta = old_A_j - old_A_i
    delta[mid] = 0.0
    corr += delta
    p1 = np.arange(max(0, i - n + 1), i + 1)
    A[p1, (i - p1) + mid] += 1.0
    p2 = np.arange(i, n)
    A[p2, (p2 - i) + mid] += 1.0
    p3 = np.arange(max(0, j - n + 1), j + 1)
    A[p3, (j - p3) + mid] -= 1.0
    p4 = np.arange(j, n)
    A[p4, (p4 - j) + mid] -= 1.0
    return h, A, corr

def fast_local_search(h, n, deadline):
    dx = 2.0 / n
    h = h.astype(np.float64).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    cur_max = float(corr.max())
    A = precompute_A(h, n)
    mid = n - 1
    while time.time() < deadline:
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        C = corr[None, :] - A[zeros, :]
        C[:, mid] = -np.inf
        found = False
        for ii in range(len(zeros)):
            if time.time() > deadline:
                return h, float(cur_max * dx)
            diffs = C[ii][None, :] + A[ones, :]
            max_vals = diffs.max(axis=1)
            cands = np.where(max_vals < cur_max - 1e-10)[0]
            if len(cands) > 0:
                j = ones[cands[np.argmin(max_vals[cands])]]
                i = zeros[ii]
                h, A, corr = apply_swap(h, A, corr, n, i, j)
                cur_max = float(corr.max())
                found = True
                break
        if not found:
            break
    return h, float(cur_max * dx)

def simulated_annealing(h, n, deadline, seed=42):
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    h = h.astype(np.float64).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    cur_max = float(corr.max())
    A = precompute_A(h, n)
    mid = n - 1
    best_h = h.copy()
    best_max = cur_max
    T = max(cur_max * 0.003, 0.3)
    while time.time() < deadline:
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        for _ in range(100):
            if time.time() > deadline:
                return best_h, float(best_max * dx)
            i = zeros[rng.randint(len(zeros))]
            j = ones[rng.randint(len(ones))]
            delta = A[j] - A[i]
            saved = delta[mid]
            delta[mid] = 0.0
            new_max = float((corr + delta).max())
            delta[mid] = saved
            if new_max <= cur_max or rng.rand() < np.exp(-(new_max - cur_max) / T):
                h, A, corr = apply_swap(h, A, corr, n, i, j)
                cur_max = new_max
                if cur_max < best_max:
                    best_max = cur_max
                    best_h = h.copy()
                zeros = np.where(h == 0)[0]
                ones = np.where(h == 1)[0]
        T *= 0.995
        if T < 0.1:
            T = 0.1
    return best_h, float(best_max * dx)

def slsqp_polish(h, n, deadline, alpha=5000):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 5000, 'ftol': 1e-18})
        x = np.clip(res.x, 0.0, 1.0)
        target = n / 2.0
        for _ in range(2000):
            diff = target - x.sum()
            if abs(diff) < 1e-12:
                break
            if diff > 0:
                room = 1.0 - x
                s = room.sum()
                if s < 1e-14:
                    break
                x = np.clip(x + diff / s * room, 0.0, 1.0)
            else:
                s = x.sum()
                if s < 1e-14:
                    break
                x = np.clip(x + diff / s * x, 0.0, 1.0)
        return x
    except Exception:
        return h

def random_binary(n, rng):
    idx = rng.permutation(n)
    h = np.zeros(n, dtype=np.float64)
    h[idx[:n // 2]] = 1.0
    return h
