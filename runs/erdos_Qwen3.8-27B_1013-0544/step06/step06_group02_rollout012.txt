def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_max = c.max()
    c_shift = alpha * (c - c_max)
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-14:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def smooth_max_obj(x, n, alpha):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def penalty_obj(x, n, alpha, mu):
    val = smooth_max_obj(x, n, alpha)
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def penalty_grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def _slq_con(x, n):
    return float(x.sum() - n / 2.0)

def _slq_jac(x, n):
    return np.ones(n)

def lbfgs_phase(h, n, deadline, max_iters=200):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(100,1.0),(500,5.0),(2000,20.0),(10000,50.0),
               (50000,100.0),(200000,500.0),(1000000,2000.0),
               (5000000,5000.0),(20000000,20000.0),(100000000,50000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(penalty_obj, h, jac=penalty_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': max_iters*8, 'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def nesterov_phase(h, n, deadline, alpha_init=200.0, lr_init=0.003):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    v = np.zeros(n)
    lr = lr_init
    alpha = alpha_init
    beta = 0.85
    stall = 0
    for it in range(100000):
        if time.time() > deadline:
            break
        y = h + beta * v
        y = np.clip(y, 0.0, 1.0)
        g = grad_smooth_max(y, n, alpha)
        v = beta * v - lr * g
        h_new = project_feasible(h + v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            stall = 0
        else:
            v *= 0.5
            lr *= 0.85
            stall += 1
            if stall % 30 == 0:
                alpha *= 1.3
            if lr < 1e-12:
                break
    return best_h, best_c

def slsqp_phase(h, n, deadline, alpha=1000000, maxiter=1500):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slq_con, 'jac': _slq_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_max_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return h.copy(), compute_c5(h, n)

def pair_search(h, n, deadline, radius=None):
    best_c = compute_c5(h, n)
    deltas = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-05, 2e-05]
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return best_h if 'best_h' in dir() else h, best_c
                lo = max(0, i - radius) if radius else 0
                hi = min(n, i + radius + 1) if radius else n
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def resample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new)

def multi_restart(h, n, deadline, rng, n_restarts=3):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    per_restart = (deadline - time.time()) / n_restarts
    for r in range(n_restarts):
        t_end = time.time() + per_restart
        if time.time() > t_end:
            break
        scale = 0.02 * (r + 1)
        h_pert = project_feasible(h + rng.normal(0, scale, n), n)
        h_r, c_r = lbfgs_phase(h_pert, n, min(t_end, deadline - 5), max_iters=100)
        if c_r < best_c:
            best_c = c_r
            best_h = h_r
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n_init = 200

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)

    # Phase 1: Low-res optimization then upsample (find good basin)
    n_low = max(30, n_init // 6)
    t_p1 = min(t0 + 60, deadline - 400)
    h_low = resample(best_h, n_init, n_low)
    h_low, c_low = lbfgs_phase(h_low, n_low, t_p1, max_iters=200)
    h_up = resample(h_low, n_low, n_init)
    t_p1b = min(time.time() + 80, deadline - 350)
    h_up, c_up = lbfgs_phase(h_up, n_init, t_p1b, max_iters=200)
    if c_up < best_c:
        best_c = c_up
        best_h = h_up

    # Phase 2: Nesterov momentum
    t_p2 = min(time.time() + 80, deadline - 250)
    h_ns, c_ns = nesterov_phase(best_h, n_init, t_p2, alpha_init=200.0, lr_init=0.003)
    if c_ns < best_c:
        best_c = c_ns
        best_h = h_ns

    # Phase 3: Higher alpha L-BFGS-B
    t_p3 = min(time.time() + 100, deadline - 180)
    best_h, best_c = lbfgs_phase(best_h, n_init, t_p3, max_iters=300)

    # Phase 4: Try higher resolution (n=500)
    n_high = 500
    if time.time() < deadline - 250:
        h_hi = resample(best_h, n_init, n_high)
        alloc = min(150, deadline - time.time() - 100)
        if alloc > 50:
            h_hi, c_hi = lbfgs_phase(h_hi, n_high, time.time() + alloc * 0.5, max_iters=200)
            if time.time() < deadline - 50:
                h_hi, c_hi = nesterov_phase(h_hi, n_high,
                    min(time.time() + 50, deadline - 40), alpha_init=500.0, lr_init=0.002)
            if c_hi < best_c:
                best_c = c_hi
                best_h = h_hi
                n_init = n_high

    # Phase 5: SLSQP refinement at current resolution
    nf = len(best_h)
    t_p5 = min(time.time() + 60, deadline - 100)
    h_sl, c_sl = slsqp_phase(best_h, nf, t_p5, alpha=5000000, maxiter=2000)
    if c_sl < best_c:
        best_c = c_sl
        best_h = h_sl
        nf = len(best_h)

    # Phase 6: One more Nesterov pass
    t_p6 = min(time.time() + 50, deadline - 80)
    h_n2, c_n2 = nesterov_phase(best_h, nf, t_p6, alpha_init=2000.0, lr_init=0.001)
    if c_n2 < best_c:
        best_c = c_n2
        best_h = h_n2
        nf = len(best_h)

    # Phase 7: Pair search (most impactful for final improvements)
    if time.time() < deadline - 5:
        rad = 40 if nf <= 350 else 25
        best_h, best_c = pair_search(best_h, nf, deadline - 2, radius=rad)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
