```
def project_polytope(x, target):
    x=np.asarray(x,float)
    n=x.size
    if n==0: return x
    if abs(x.sum()-target)<1e-13 and np.all((x>=0)&(x<=1)): return x.copy()
    lo=np.min(x)-1.0; hi=np.max(x)+1.0
    for _ in range(80):
       tau=0.5*(lo+hi)
       y=np.clip(x-tau,0.0,1.0)
       s=y.sum()
       if s > target:
          lo=tau
       else:
          hi=tau
    y=np.clip(x-0.5*(lo+hi),0.0,1.0)
    diff=target-y.sum()
    # distribute
    for _ in range(5):
       if abs(diff) < 1e-13: break
       if diff > 0:
          room=1.0-y
          idx=np.flatnonzero(room > 1e-12)
          if idx.size == 0: break
          add=np.minimum(room[idx], diff/idx.size)
          # if sum add < diff, distribute iteratively
          y[idx]+=add
       else:
          room=y
          idx=np.flatnonzero(room > 1e-12)
          if idx.size == 0: break
          del=np.minimum(room[idx], -diff/idx.size)
          y[idx]-=del
       diff=target-y.sum()
    y=np.clip(y,0,1)
    # final exact-ish adjust one element
    diff=target-y.sum()
    if abs(diff)>1e-14:

          # distribute naively

    return y
```

```python
import time
import numpy as np
from scipy.optimize import minimize


def make_ctx(n):
    dx = 2.0 / float(n)
    L = n - 1
    M = 1
    need = 3 * n + 8
    while M < need:
        M *= 2
    ones = np.zeros(M)
    ones[:n] = 1.0
    ctx = {
        'n': int(n),
        'target': 0.5 * float(n),
        'dx': float(dx),
        'L': int(L),
        'M': int(M),
        'O': np.fft.rfft(ones),
        'deadline': 0.0,
    }
    return ctx


def corr_counts(h, ctx):
    n = ctx['n']
    L = ctx['L']
    M = ctx['M']
    harr = np.zeros(M)
    harr[:n] = h
    H = np.fft.rfft(harr)
    auto = np.fft.irfft(H * np.conj(H), n=M).real

    prefix = np.empty(n + 1)
    prefix[0] = 0.0
    np.cumsum(h, out=prefix[1:])
    total = prefix[n]

    c = np.empty(2 * n - 1)
    s = np.arange(n)
    c[L:L + n] = prefix[n - s] - auto[s]
    if n > 1:
        sneg = np.arange(n - 1, 0, -1)
        c[:n - 1] = total - prefix[sneg] - auto[sneg]
    return c


def true_c(h, ctx):
    c = corr_counts(h, ctx)
    return float(np.max(c) * ctx['dx'])


def logsum_weights(c, alpha):
    m = float(np.max(c))
    if alpha <= 0.0:
        w = np.full(c.shape, 1.0 / c.size)
        return m, m, w
    with np.errstate(over='ignore', under='ignore'):
        e = np.exp(alpha * (c - m))
    z = float(e.sum())
    if z <= 0.0:
        e = np.zeros(c.shape)
        e[np.argmax(c)] = 1.0
        z = 1.0
    w = e / z
    f = m + np.log(z) / alpha
    return m, float(f), w


def weighted_grad(h, w, ctx):
    n = ctx['n']
    L = ctx['L']
    M = ctx['M']

    harr = np.zeros(M)
    harr[:n] = h
    H = np.fft.rfft(harr)

    warr = np.zeros(M)
    warr[:2 * n - 1] = w
    W = np.fft.rfft(warr)

    A = np.fft.irfft(W * ctx['O'], n=M).real[L:L + n]

    wr = np.zeros(M)
    wr[:2 * n - 1] = warr[:2 * n - 1][::-1]
    WR = np.fft.rfft(wr)

    B = np.fft.irfft(WR * H, n=M).real[L:L + n]
    C = np.fft.irfft(W * H, n=M).real[L:L + n]
    return A - B - C


def objective_pen(x, alpha, mu, ctx):
    c = corr_counts(x, ctx)
    m, f, w = logsum_weights(c, alpha)
    s = x.sum()
    t = ctx['target']
    f += 0.5 * mu * (s - t) * (s - t)
    ctx['last_id'] = id(x)
    ctx['last_alpha'] = alpha
    ctx['last_F'] = f
    ctx['last_w'] = w
    return float(f)


def gradient_pen(x, alpha, mu, ctx):
    hit_cache = (ctx.get('last_id') is x and ctx.get('last_alpha') == alpha
                 and 'last_w' in ctx)
    if hit_cache:
        w = ctx['last_w']
    else:
        c = corr_counts(x, ctx)
        m, f, w = logsum_weights(c, alpha)
    g = weighted_grad(x, w, ctx)
    g = g + mu * (x.sum() - ctx['target'])
    return g


def project_polytope_info(x, target):
    x = np.asarray(x, dtype=float)
    if x.size == 1:
        v = float(np.clip(x[0], 0.0, 1.0))
        return np.array([v]), 0.0

    if np.min(x) >= -1e-14 and np.max(x) <= 1.0 + 1e-14 and abs(x.sum() - target) < 1e-13:
        y = np.clip(x, 0.0, 1.0)
        return y, 0.0

    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(80):
        tau = 0.5 * (lo + hi)
        y = np.clip(x - tau, 0.0, 1.0)
        if y.sum() > target:
            lo = tau
        else:
            hi = tau
    tau = 0.5 * (lo + hi)
    y = np.clip(x - tau, 0.0, 1.0)

    diff = target - y.sum()
    free = (y > 1e-12) & (y < 1.0 - 1e-12)
    if free.any():
        y[free] += diff / float(free.sum())
        y = np.clip(y, 0.0, 1.0)

    for _ in range(6):
        diff = target - y.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0.0:
            room = 1.0 - y
            i = int(np.argmax(room))
            if room[i] > 0.0:
                y[i] += min(room[i], diff)
        else:
            room = y
            i = int(np.argmax(room))
            if room[i] > 0.0:
                y[i] -= min(room[i], -diff)
    y = np.clip(y, 0.0, 1.0)
    return y, float(tau)


def project_polytope(x, target):
    y, _ = project_polytope_info(x, target)
    return y


def resample_h(h, n_old, n_new):
    if n_old == n_new:
        return project_polytope(np.clip(h, 0.0, 1.0), 0.5 * float(n_new))
    if n_old <= 0:
        return np.full(n_new, 0.5)
    xo = (np.arange(n_old) + 0.5) * (2.0 / float(n_old))
    xn = (np.arange(n_new) + 0.5) * (2.0 / float(n_new))
    hh = np.interp(xn, xo, h, left=float(h[0]), right=float(h[-1]))
    return project_polytope(hh, 0.5 * float(n_new))


def make_initial(n, seed):
    x = (np.arange(n) + 0.5) * (2.0 / float(n))
    h = (0.5 + 0.55 * np.cos(np.pi * x)
         + 0.18 * np.sin(2.0 * np.pi * x)
         + 0.08 * np.cos(4.0 * np.pi * x))
    rng = np.random.RandomState(seed)
    h += 0.03 * rng.standard_normal(n)
    return project_polytope(h, 0.5 * float(n))


def update_best(h, ctx, state):
    c = true_c(h, ctx)
    if c + 1e-14 < state['best_c']:
        state['best_c'] = c
        state['best_h'] = np.array(h, dtype=float).copy()
        state['best_n'] = int(ctx['n'])


def lbgfs_pass(h, ctx, alpha, mu, deadline, state, max_iter=30):
    n = ctx['n']
    target = ctx['target']
    h = project_polytope(np.clip(h, 0.0, 1.0), target)
    update_best(h, ctx, state)
    bounds = [(0.0, 1.0)] * n

    while time.time() < deadline:
        try:
            res = minimize(
                objective_pen, h, method='L-BFGS-B', jac=gradient_pen,
                bounds=bounds, args=(alpha, mu, ctx),
                options={'maxiter': int(max_iter), 'gtol': 1e-9,
                         'ftol': 1e-15, 'maxls': 40, 'disp': 0}
            )
        except Exception:
            break
        if not hasattr(res, 'x'):
            break
        x = np.clip(res.x, 0.0, 1.0)
        x = project_polytope(x, target)
        if np.linalg.norm(x - h) < 1e-11:
            h = x
            update_best(h, ctx, state)
            break
        h = x
        update_best(h, ctx, state)
    return h


def slsqp_obj(x, alpha, ctx):
    c = corr_counts(x, ctx)
    m, f, w = logsum_weights(c, alpha)
    return float(f)


def slsqp_grad(x, alpha, ctx):
    c = corr_counts(x, ctx)
    m, f, w = logsum_weights(c, alpha)
    return weighted_grad(x, w, ctx)


def slsqp_eq(x, ctx):
    return x.sum() - ctx['target']


def slsqp_eq_jac(x, ctx):
    return np.ones(ctx['n'], dtype=float)


def slsqp_pass(h, ctx, alpha, maxiter, state):
    n = ctx['n']
    target = ctx['target']
    h = project_polytope(np.clip(h, 0.0, 1.0), target)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac, 'args': (ctx,)}]
    try:
        res = minimize(
            slsqp_obj, h, method='SLSQP', bounds=bounds,
            jac=slsqp_grad, constraints=cons, args=(alpha, ctx),
            options={'maxiter': int(maxiter), 'ftol': 1e-15, 'disp': 0}
        )
    except Exception:
        return h
    if hasattr(res, 'x'):
        h = np.clip(res.x, 0.0, 1.0)
        h = project_polytope(h, target)
        update_best(h, ctx, state)
    return h


def exact_c(h, n):
    dx = 2.0 / float(n)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = max(30.0, float(budget_s) - 25.0)
    deadline = t0 + total
    rng = np.random.RandomState(seed)

    try:
        hp = np.array(initial_h_values, dtype=float).copy()
        if hp.size < 20 or hp.size % 2 != 0:
            n = 400
            h = resample_h(hp, int(hp.size), n) if hp.size > 0 else make_initial(n, seed)
        else:
            n = int(hp.size)
            if n > 600:
                h = resample_h(hp, n, 600)
                n = 600
            else:
                h = np.clip(hp, 0.0, 1.0)
                h = project_polytope(h, 0.5 * float(n))
    except Exception:
        n = 400
        h = make_initial(n, seed)

    if n % 2 != 0:
        n2 = n + 1 if n + 1 <= 600 else n - 1
        h = resample_h(h, n, n2)
        n = n2

    ctx = make_ctx(n)
    state = {
        'best_h': np.array(h, dtype=float).copy(),
        'best_c': true_c(h, ctx),
        'best_n': int(n),
    }

    # Primary continuation on the parent resolution.
    phase_a = t0 + 0.40 * total
    work = np.array(state['best_h'], dtype=float).copy()
    alphas = [20.0, 60.0, 200.0, 700.0, 2500.0,
              10000.0, 40000.0, 150000.0, 600000.0]
    for idx, alpha in enumerate(alphas):
        if time.time() >= phase_a:
            break
        remain = phase_a - time.time()
        end = min(phase_a, time.time() + max(8.0, remain / float(len(alphas) - idx)))
        mu = 400.0 if alpha < 1000.0 else 1200.0
        work = lbgfs_pass(work, ctx, alpha, mu, end, state, 32)
        cnow = true_c(work, ctx)
        if state['best_n'] == n and cnow > state['best_c'] + 1e-6:
            work = np.array(state['best_h'], dtype=float).copy()

    # Local diverse restarts at the same resolution.
    phase_b = t0 + 0.68 * total
    for trial in range(10):
        if time.time() >= phase_b:
            break
        base = np.array(state['best_h'], dtype=float).copy()
        if state['best_n'] != n:
            base = resample_h(base, state['best_n'], n)
            base = project_polytope(base, 0.5 * float(n))

        if trial == 0:
            work = 1.0 - base
        elif trial == 1:
            work = base + 0.015 * rng.standard_normal(n)
        elif trial == 2:
            work = base + 0.04 * rng.standard_normal(n)
        elif trial == 3:
            work = base + 0.09 * rng.standard_normal(n)
        elif trial == 4:
            work = base + 0.18 * rng.standard_normal(n)
        elif trial == 5:
            work = 0.5 + 0.6 * (base - 0.5)
        elif trial == 6:
            work = 0.5 + 1.35 * (base - 0.5)
        elif trial == 7:
            work = 0.5 + 2.0 * (base - 0.5)
        elif trial == 8:
            work = base[::-1].copy()
        else:
            work = base + 0.06 * rng.standard_normal(n)

        work = np.clip(work, 0.0, 1.0)
        work = project_polytope(work, 0.5 * float(n))
        end1 = min(phase_b, time.time() + 12.0)
        end2 = min(phase_b, time.time() + 24.0)
        work = lbgfs_pass(work, ctx, 2500.0, 1000.0, end1, state, 28)
        work = lbgfs_pass(work, ctx, 30000.0, 1400.0, end2, state, 28)

    # Multi-resolution refinement.
    if deadline - time.time() > 70.0:
        base_n = int(state['best_n'])
        cand = []
        for nn in (base_n + 100, base_n + 200, base_n - 100, 500, 600, 300):
            if 150 <= nn <= 650 and nn % 2 == 0 and nn != base_n and nn not in cand:
                cand.append(int(nn))
        cand.sort(reverse=True)
        if len(cand) > 3:
            cand = cand[:3]

        rem = deadline - time.time()
        per = max(45.0, rem / float(len(cand))) if cand else 0.0
        for nn in cand:
            if deadline - time.time() <= 25.0:
                break
            base = np.array(state['best_h'], dtype=float).copy()
            work = resample_h(base, int(state['best_n']), nn)
            ctxr = make_ctx(nn)
            block_end = time.time() + per
            block_alphas = [40.0, 400.0, 2500.0, 15000.0, 80000.0]
            chunk = max(8.0, per / float(len(block_alphas)))
            for alpha in block_alphas:
                if time.time() >= block_end:
                    break
                end = min(block_end, time.time() + chunk)
                work = lbgfs_pass(work, ctxr, alpha, 900.0, end, state, 30)

    # Final polish at the currently best resolution.
    if deadline - time.time() > 18.0:
        nbest = int(state['best_n'])
        ctxbest = make_ctx(nbest)
        work = np.array(state['best_h'], dtype=float).copy()
        work = project_polytope(work, 0.5 * float(nbest))
        fin_alphas = [2000.0, 20000.0, 100000.0, 400000.0]
        for alpha in fin_alphas:
            if deadline - time.time() <= 8.0:
                break
            end = min(deadline - 4.0, time.time() + 10.0)
            work = lbgfs_pass(work, ctxbest, alpha, 1200.0, end, state, 24)

    if deadline - time.time() > 25.0:
        nbest = int(state['best_n'])
        ctxbest = make_ctx(nbest)
        work = np.array(state['best_h'], dtype=float).copy()
        work = project_polytope(work, 0.5 * float(nbest))
        work = slsqp_pass(work, ctxbest, 80000.0, 6, state)

    out = np.array(state['best_h'], dtype=float).copy()
    out_n = int(state['best_n'])
    out = project_polytope(np.clip(out, 0.0, 1.0), 0.5 * float(out_n))

    # Make the integral constraint as exact as practical in floating point.
    target = 0.5 * float(out_n)
    for _ in range(8):
        diff = target - out.sum()
        if abs(diff) < 1e-14:
            break
        mask = (out > 1e-14) & (out < 1.0 - 1e-14)
        if mask.any():
            out[mask] += diff / float(mask.sum())
        else:
            out += diff / float(out.size)
        out = np.clip(out, 0.0, 1.0)

    bound = exact_c(out, out_n)
    if not np.isfinite(bound) or bound > 1.0:
        bound = state['best_c']
    return out, float(bound), int(out_n)
```
