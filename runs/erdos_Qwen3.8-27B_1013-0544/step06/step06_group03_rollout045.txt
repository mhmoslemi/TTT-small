import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    return np.correlate(h, 1 - h, mode='full') * (2.0 / n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    n_ones = n // 2
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:n_ones]] = 1.0
    return b

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def sum_con_jac(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def binary_sa(h0, n, deadline, rng, t0=5.0, t_min=0.002, rate=0.99998, max_stall=50000):
    h = to_binary(h0, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    t = t0
    stalls = 0
    while time.time() < deadline:
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        c = compute_c5(h, n)
        if c < best_c - 1e-15:
            best_c = c
            best_h = h.copy()
            stalls = 0
            t = min(t0, t * 1.01)
        else:
            stalls += 1
            accept = np.log(1.0 - rng.rand() + 1e-15) < (best_c - c) / t
            if not accept:
                h[i] = 1.0
                h[j] = 0.0
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
        t = max(t_min, t * rate)
        if stalls > max_stall and t < 0.02:
            h = best_h.copy()
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            t = t0
            stalls = 0
    return best_h, best_c

def targeted_local(h, n, deadline, rng):
    """Targeted local search: focus on reducing the maximum lag."""
    h = to_binary(h, n)
    best_c = compute_c5(h, n)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for _ in range(500000):
        if time.time() > deadline:
            break
        c_arr = compute_all_lags(h, n)
        k_max = int(np.argmax(c_arr))
        lag = k_max - (n - 1)
        m = abs(lag)
        if m == 0 or m >= n:
            break
        # Find pairs contributing to max lag and try to fix them
        if lag > 0:
            bad_i = np.where(h[:n-m] == 1)[0]
            bad_j = np.where(h[m:] == 0)[0]
        else:
            bad_i = np.where(h[:n-m] == 0)[0]
            bad_j = np.where(h[abs(lag):] == 1)[0]
        if len(bad_i) == 0 or len(bad_j) == 0:
            break
        # Try moving a 1 away from a contributing position
        improved = False
        for _ in range(200):
            if time.time() > deadline:
                break
            ri = bad_i[rng.randint(len(bad_i))]
            rj = zeros[rng.randint(len(zeros))]
            if h[ri] != 1.0:
                continue
            h[ri] = 0.0
            h[rj] = 1.0
            c_new = compute_c5(h, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                c_arr = compute_all_lags(h, n)
                k_max = int(np.argmax(c_arr))
                lag = k_max - (n - 1)
                m = abs(lag)
                if m == 0 or m >= n:
                    break
                if lag > 0:
                    bad_i = np.where(h[:n-m] == 1)[0]
                    bad_j = np.where(h[m:] == 0)[0]
                else:
                    bad_i = np.where(h[:n-m] == 0)[0]
                    bad_j = np.where(h[abs(lag):] == 1)[0]
                improved = True
            else:
                h[ri] = 1.0
                h[rj] = 0.0
        if not improved:
            # Try random swaps to escape
            for _ in range(10):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h[i], h[j] = h[j], h[i]
                c_new = compute_c5(h, n)
                if c_new < best_c:
                    best_c = c_new
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                else:
                    h[i], h[j] = h[j], h[i]
    return h, best_c

def optimize_continuous(h_start, n, deadline):
    """Refine a (near-binary) solution with SLSQP."""
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)
    for alpha in [50000, 200000, 1000000]:
        if time.time() > deadline - 2:
            break
        h_new = slsqp_opt(h, n, alpha, maxiter=2000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            h = h_new
    return h, best_c

def interpolate_to_n(h_src, n_src, n_dst):
    x_src = np.linspace(0, 2, n_src)
    x_dst = np.linspace(0, 2, n_dst)
    return np.interp(x_dst, x_src, h_src)

    t0 = time.time()

        remaining = deadline - time.time() - 50
        if remaining < 40:

        h_start = interpolate_to_n(best_h, len(best_h), n_try)

            best_h = h_new
        # Final targeted local

        if c_final < best_c:
            best_c = c_final
            best_h = h_final

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
