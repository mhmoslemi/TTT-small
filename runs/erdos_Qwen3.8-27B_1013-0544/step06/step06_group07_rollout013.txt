def project_feasible(h, n):
    """Project onto {0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Bisection for the projection
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0

        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_soft_gradient(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    max_c = corr.max()
    shifted = alpha * (corr - max_c)
    exp_c = np.exp(shifted)
    p = exp_c / exp_c.sum()
    
    # grad[j] = dx * (sum_i (1-h[i]) * p[n-1+j-i] - sum_i h[i] * p[n-1+i-j])

    conv1 = np.convolve(r, p, 'full')
    grad1 = conv1[n-1 : 2*n-1]
    
    p_rev = p[::-1]
    conv2 = np.convolve(h, p_rev, 'full')
    grad2 = conv2[n-1 : 2*n-1]
    
    return dx * (grad1 - grad2)

def soft_objective(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    max_c = corr.max()
    shifted = alpha * (corr - max_c)
    exp_c = np.exp(shifted)
    return max_c + np.log(exp_c.sum()) / alpha

        # Interpolate alpha

        # Momentum

        if c < best_c:

            best_h = h.copy()

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.0, 2.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_soft_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    max_c = corr.max()
    shifted = alpha * (corr - max_c)
    # Clip to avoid overflow
    shifted = np.clip(shifted, -500, 500)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    if s < 1e-300:
        return np.zeros(n)
    p = exp_c / s
    # Gradient computation
    r = 1.0 - h
    conv1 = np.convolve(r, p, 'full')
    grad1 = conv1[n-1 : 2*n-1]
    p_rev = p[::-1]
    conv2 = np.convolve(h, p_rev, 'full')
    grad2 = conv2[n-1 : 2*n-1]
    return dx * (grad1 - grad2)

def soft_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    max_c = corr.max()
    shifted = alpha * (corr - max_c)
    shifted = np.clip(shifted, -500, 500)
    exp_c = np.exp(shifted)
    return max_c + np.log(exp_c.sum()) / alpha

def gd_phase(h0, n, deadline, momentum=0.9):
    """Gradient descent with adaptive alpha and momentum"""
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    v = np.zeros(n)
    t0 = time.time()
    duration = deadline - t0
    
    # Schedule: alpha goes from 20 to 50000 over the duration
    alpha_start = 20.0
    alpha_end = 50000.0
    # Learning rate schedule: starts at 0.05, decays to 0.001
    lr_start = 0.05
    lr_end = 0.0005
    
    it = 0
    while time.time() < deadline:
        frac = min((time.time() - t0) / max(duration, 1.0), 1.0)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac
        
        grad = compute_soft_grad(h, alpha)
        
        # Momentum update
        v = momentum * v - lr * grad
        
        # Line search: try the step, if it increases the objective, halve
        obj_cur = soft_obj(h, alpha)
        step = v
        accepted = False
        for _ in range(8):
            h_new = project_feasible(h + step, n)
            obj_new = soft_obj(h_new, alpha)
            if obj_new <= obj_cur + 1e-12:
                h = h_new
                accepted = True
                break
            step = step * 0.5
        if not accepted:
            break  # Can't make progress
        
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        it += 1
    
    return best_h, best_c, it

def round_to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    return out

def swap_delta(h, n, i, j, m_range):
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.0, 2.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_soft_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    max_c = corr.max()
    shifted = np.clip(alpha * (corr - max_c), -500, 500)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    if s < 1e-300:
        return np.zeros(n)
    p = exp_c / s
    r = 1.0 - h
    conv1 = np.convolve(r, p, 'full')
    grad1 = conv1[n-1 : 2*n-1]
    p_rev = p[::-1]
    conv2 = np.convolve(h, p_rev, 'full')
    grad2 = conv2[n-1 : 2*n-1]
    return dx * (grad1 - grad2)

def soft_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    max_c = corr.max()
    shifted = np.clip(alpha * (corr - max_c), -500, 500)
    exp_c = np.exp(shifted)
    return max_c + np.log(exp_c.sum()) / alpha

def gd_phase(h0, n, deadline, n_restarts=1):
    best_h = h0.copy()
    best_c = compute_c5(h0, n)
    for restart in range(n_restarts):
        if time.time() > deadline:
            break
        h = h0.copy() if restart == 0 else h0 + 0.05 * np.random.randn(n)
        h = project_feasible(h, n)
        v = np.zeros(n)
        momentum = 0.9
        duration = deadline - time.time()
        
        # Adaptive schedule
        alpha_start = 20.0
        alpha_end = 100000.0
        lr_start = 0.1
        lr_end = 0.001
        
        while time.time() < deadline - 1:
            frac = min((time.time() - (deadline - duration)) / duration, 1.0)
            alpha = alpha_start * (alpha_end / alpha_start) ** frac
            lr = lr_start * (lr_end / lr_start) ** frac
            
            grad = compute_soft_grad(h, alpha)
            v = momentum * v - lr * grad
            
            # Line search
            obj_cur = soft_obj(h, alpha)
            h_new = None
            for _ in range(10):
                cand = project_feasible(h + v, n)
                obj_new = soft_obj(cand, alpha)
                if obj_new <= obj_cur + 1e-10:
                    h_new = cand
                    break
                v = v * 0.5
            if h_new is None:
                break
            h = h_new
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
    return best_h, best_c

def swap_delta(h, n, i, j, m_range):
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    delta[k2] += 1.0
    return delta
