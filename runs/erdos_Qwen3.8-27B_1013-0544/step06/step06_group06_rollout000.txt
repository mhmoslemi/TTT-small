    d = target - h.sum()
    if abs(d) > 1e-14:
        if d > 0:
            idx = np.argmin(1.0 - h)
            h[idx] = min(1.0, h[idx] + d)
        else:
            idx = np.argmax(h)
            h[idx] = max(0.0, h[idx] + d)
    return h

def to_binary(h, n):
    target = int(round(0.5 * n))
    b = np.zeros(n, dtype=float)
    b[np.argsort(-h)[:target]] = 1.0
    return b

def thue_morse_init(n, rng):
    t = np.zeros(n, dtype=int)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    target = n // 2
    ones = int(t.sum())
    if ones > target:
        idx = np.where(t == 1)[0]
        rng.shuffle(idx)
        t[idx[:ones - target]] = 0
    elif ones < target:
        idx = np.where(t == 0)[0]
        rng.shuffle(idx)
        t[idx[:target - ones]] = 1
    return t.astype(float)

def build_neighbors(n):
    m = 2 * n - 1
    m1 = np.zeros((m, n), dtype=float)
    m2 = np.zeros((m, n), dtype=float)
    i1 = np.zeros((m, n), dtype=np.int32)
    i2 = np.zeros((m, n), dtype=np.int32)
    p = np.arange(n)
    for a in range(m):
        s = n - 1 - a
        ps = p + s
        pm = p - s
        v1 = (ps >= 0) & (ps < n)
        v2 = (pm >= 0) & (pm < n)
        m1[a] = v1.astype(float)
        m2[a] = v2.astype(float)
        i1[a] = np.clip(ps, 0, n - 1).astype(np.int32)
        i2[a] = np.clip(pm, 0, n - 1).astype(np.int32)
    return (m1, m2, i1, i2)

def build_Gdx(h, idx, neigh, dx):
    m1, m2, i1, i2 = neigh
    a = m1[idx]
    return dx * (a - a * h[i1[idx]] - m2[idx] * h[i2[idx]])

def top_lags(c, L, exclude_zero, n):
    cc = c.copy()
    if exclude_zero:
        cc[n - 1] = -np.inf
    if L >= cc.size:
        return np.arange(cc.size)
    return np.argsort(cc)[-L:]

def make_corr_dist(n, active, dx):
    abs_s = np.abs((n - 1) - active.astype(np.int64))
    d = np.arange(n)[:, None] == abs_s[None, :]
    return d.astype(float) * dx

def kick_binary(h, rng, k=3):
    h2 = h.copy()
    z = np.where(h2 == 0)[0]
    o = np.where(h2 == 1)[0]
    k = min(k, len(z), len(o))
    if k <= 0:
        return h2
    rz = rng.choice(z, k, replace=False)
    ro = rng.choice(o, k, replace=False)
    h2[ro] = 0.0
    h2[rz] = 1.0
    return h2

def kick_continuous(h, rng, n, scale=0.04, count=15):
    h2 = h.copy()
    idx = rng.choice(n, min(count, n), replace=False)
    h2[idx] += scale * rng.randn(idx.size)
    return project_feasible(h2, n)

def transfer_eval(h, src, snk, delta, hi, n):
    delta = float(np.clip(delta, 0.0, hi))
    if delta <= 1e-9:
        return None
    h2 = h.copy()
    h2[src] = h[src] - delta
    h2[snk] = h[snk] + delta
    h2 = project_feasible(h2, n)
    c2 = corr_h(h2, n)
    cnew = float(np.max(c2) * (2.0 / n))
    return (h2, c2, cnew)

def active_max_d(A, B, Cq, d):
    return float(np.max(A + d * B + d * d * Cq))

def optimize_delta(A, B, Cq, hi):
    if hi <= 1e-12:
        return 0.0
    a = 0.0
    b = float(hi)
    gr = 1.6180339887498949
    x1 = (a + b) / gr
    x2 = a + b - x1
    f1 = active_max_d(A, B, Cq, x1)
    f2 = active_max_d(A, B, Cq, x2)
    for _ in range(50):
        if f1 < f2:
            b = x2
            x2 = x1
            f2 = f1
            x1 = (a + b) / gr
        else:
            a = x1
            x1 = x2
            f1 = f2
            x2 = a + b - x1
        if b - a < 1e-12:
            break
    return 0.5 * (a + b)

def pair_refine(h_start, n, neigh, deadline, seed, L=160, S=90, binary_mode=False):
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    h = project_feasible(h_start, n)
    if binary_mode:
        h = to_binary(h, n)
    c = corr_h(h, n)
    cur = float(np.max(c) * dx)
    best_h = h.copy()
    best_c = cur
    active = top_lags(c, L, True, n)
    A = c[active] * dx
    corr_dist = make_corr_dist(n, active, dx)
    temp = 0.0015
    stall = 0
    no_improve = 0
    topK = 5

    while time.time() < deadline:
        if cur > best_c + 0.04:
            h = best_h.copy()
            c = corr_h(h, n)
            cur = best_c
            active = top_lags(c, L, True, n)
            A = c[active] * dx
            corr_dist = make_corr_dist(n, active, dx)
            stall = 0

        if binary_mode:
            sink = np.where(h == 0)[0]
            source = np.where(h == 1)[0]
            if sink.size == 0 or source.size == 0:
                break
            hd = np.ones((sink.size, source.size), dtype=float)
        else:
            if h.max() - h.min() < 1e-12:
                break
            Su = max(1, min(S, n // 2))
            order = np.argsort(h)
            sink = order[:Su]
            source = order[-Su:]
            hd = np.minimum(h[source][None, :], 1.0 - h[sink][:, None])

        if sink.size == 0 or source.size == 0:
            break
        if sink.size * source.size > 25000:
            take = max(10, 25000 // max(1, source.size))
            take = min(sink.size, take)
            sink = sink[rng.permutation(sink.size)[:take]]
            hd = hd[:sink.size]

        G = build_Gdx(h, active, neigh, dx)
        dist = np.abs(sink[:, None] - source[None, :])
        base = G[sink][:, None, :] - G[source][None, :, :]
        f = hd
        D = base * f[:, :, None] + (f[:, :, None] * f[:, :, None]) * corr_dist[dist]
        score = (A[None, None, :] + D).max(axis=2)

        if not binary_mode:
            f = 0.5 * hd
            D = base * f[:, :, None] + (f[:, :, None] * f[:, :, None]) * corr_dist[dist]
            score = np.minimum(score, (A[None, None, :] + D).max(axis=2))

        ns = source.size
        flat_score = score.ravel()
        if flat_score.size > topK:
            part = np.argpartition(flat_score, topK)[:topK]
        else:
            part = np.arange(flat_score.size)
        part = part[np.argsort(flat_score[part])]

        accepted = False
        for pid in part:
            row = int(pid // ns)
            col = int(pid % ns)
            src = source[col]
            snk = sink[row]
            hi = float(hd[row, col])
            if hi <= 1e-9:
                continue
            if binary_mode:
                delta = hi
            else:
                B = G[snk] - G[src]
                Cq = corr_dist[np.abs(src - snk)]
                delta = optimize_delta(A, B, Cq, hi)
                if delta > hi * 0.98:
                    delta = hi
            got = transfer_eval(h, src, snk, delta, hi, n)
            if got is None:
                continue
            h2, c2, cnew = got
            if cnew < cur - 1e-12:
                h = h2
                c = c2
                cur = cnew
                stall = 0
                temp = max(0.0002, temp * 0.995)
                accepted = True
                if cnew < best_c - 1e-12:
                    best_h = h.copy()
                    best_c = cnew
                    no_improve = 0
                else:
                    no_improve += 1
                break
            if cnew < best_c + max(0.005, 5.0 * temp):
                if cnew < cur:
                    prob = 1.0
                else:
                    r = (cnew - cur) / temp
                    prob = math.exp(-r) if r < 700.0 else 0.0
                if rng.random_sample() < prob:
                    h = h2
                    c = c2
                    cur = cnew
                    stall = 0
                    accepted = True
                    no_improve += 1
                    break

        if accepted:
            active = top_lags(c, L, True, n)
            A = c[active] * dx
            corr_dist = make_corr_dist(n, active, dx)
        else:
            stall += 1
            no_improve += 1
            if stall > 80:
                if binary_mode:
                    hk = kick_binary(h, rng, 3)
                else:
                    hk = kick_continuous(h, rng, n, 0.04, 15)
                ck = corr_h(hk, n)
                ckval = float(np.max(ck) * dx)
                if ckval < best_c + max(0.02, 5.0 * temp):
                    h = hk
                    c = ck
                    cur = ckval
                    active = top_lags(c, L, True, n)
                    A = c[active] * dx
                    corr_dist = make_corr_dist(n, active, dx)
                    temp = max(0.003, temp * 1.5)
                    stall = 0
            if no_improve > 2500 and time.time() < deadline - 5.0:
                h = best_h.copy()
                c = corr_h(h, n)
                cur = best_c
                active = top_lags(c, L, True, n)
                A = c[active] * dx
                corr_dist = make_corr_dist(n, active, dx)
                no_improve = 0
                stall = 0
                temp = 0.002

    return best_h, best_c

def adam_refine(h_start, n, neigh, deadline, seed, active_L=200):
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    h = project_feasible(h_start, n)
    c = corr_h(h, n)
    cur = float(np.max(c) * dx)
    best_h = h.copy()
    best_c = cur
    ema1 = np.zeros(n)
    ema2 = np.zeros(n)
    t = 1
    temp = 0.0012
    no = 0
    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    t0 = time.time()
    span = max(1e-9, deadline - t0)

    while time.time() < deadline:
        prog = min(1.0, max(0.0, (time.time() - t0) / span))
        alpha = 10.0 * (5000.0 ** prog)
        lr = 0.012 * (1.0 - 0.6 * prog) + 0.001

        c = corr_h(h, n)
        R = c * dx
        active = top_lags(c, active_L, False, n)
        Ar = R[active]
        m = float(np.max(Ar))
        e = alpha * (Ar - m)
        e -= np.max(e)
        w = np.exp(e)
        w /= w.sum()

        G = build_Gdx(h, active, neigh, dx)
        grad = G @ w

        ema1 = b1 * ema1 + (1.0 - b1) * grad
        ema2 = b2 * ema2 + (1.0 - b2) * grad * grad
        mh = ema1 / (1.0 - b1 ** t)
        vh = ema2 / (1.0 - b2 ** t)
        h_new = h - lr * mh / (np.sqrt(vh) + eps)
        h_new = project_feasible(h_new, n)

        c_new = corr_h(h_new, n)
        cnew = float(np.max(c_new) * dx)

        accepted = False
        if cnew < cur - 1e-12:
            accepted = True
            temp *= 0.995
            no = 0
            if cnew < best_c - 1e-12:
                best_h = h_new.copy()
                best_c = cnew
        elif cnew < best_c + max(0.01, 5.0 * temp):
            r = (cnew - cur) / temp
            prob = math.exp(-r) if r < 700.0 else 0.0
            if rng.random_sample() < prob:
                accepted = True
                no += 1

        if accepted:
            h = h_new
            c = c_new
            cur = cnew
        else:
            temp = max(0.0002, temp * 0.85)
            no += 1

        if no > 600:
            h = kick_continuous(h, rng, n, 0.03, 20)
            c = corr_h(h, n)
            cur = float(np.max(c) * dx)
            ema1 = np.zeros(n)
            ema2 = np.zeros(n)
            no = 0
            temp = 0.002
        t += 1

    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    usable = max(20.0, float(budget_s) - 10.0)
    end = t0 + float(budget_s) - 2.0
    rs = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = thue_morse_init(n, rs)

    if n % 2 == 1:
        h0 = np.append(h0, 0.5)
        n = len(h0)
    h0 = project_feasible(h0, n)

    neigh = build_neighbors(n)
    best_h = h0.copy()
    best_c = eval_h(best_h, n)

    Lp = min(160, 2 * n - 1)
    Sp = min(90, n // 2)

    d1 = t0 + 0.10 * usable
    if time.time() < d1 - 2.0:
        hb, cb = pair_refine(h0, n, neigh, d1, seed + 11, L=Lp, S=Sp, binary_mode=False)
        if cb < best_c:
            best_h, best_c = hb, cb

    d2 = t0 + 0.42 * usable
    if time.time() < d2 - 2.0:
        hb, cb = pair_refine(h0, n, neigh, d2, seed + 21, L=Lp, S=Sp, binary_mode=True)
        if cb < best_c:
            best_h, best_c = hb, cb

    d3 = t0 + 0.54 * usable
    if time.time() < d3 - 2.0:
        ht = thue_morse_init(n, np.random.RandomState(seed + 37))
        hb, cb = pair_refine(ht, n, neigh, d3, seed + 31, L=Lp, S=Sp, binary_mode=True)
        if cb < best_c:
            best_h, best_c = hb, cb

    d4 = t0 + 0.82 * usable
    if time.time() < d4 - 2.0:
        hb, cb = pair_refine(best_h, n, neigh, d4, seed + 41, L=Lp, S=Sp, binary_mode=False)
        if cb < best_c:
            best_h, best_c = hb, cb

    if time.time() < end - 3.0:
        hb, cb = adam_refine(best_h, n, neigh, end - 1.0, seed + 51, active_L=min(200, 2 * n - 1))
        if cb < best_c:
            best_h, best_c = hb, cb

    best_h = project_feasible(best_h, n)
    best_c = eval_h(best_h, n)
    return best_h, best_c, n
