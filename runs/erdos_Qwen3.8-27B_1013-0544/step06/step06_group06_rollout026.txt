import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def build_g(h, n):
    g = np.zeros((n, 2 * n - 1))
    for p in range(n):
        ks = max(0, p)
        ke = min(2 * n - 2, p + n - 1)
        for k in range(ks, ke + 1):
            if k != 2 * p:
                g[p, k] = 1.0 - 2.0 * h[k - p]
    return g

def apply_swap(g, h, corr, i, j, n):
    g_i = g[i].copy()
    g_j = g[j].copy()
    corr = corr + g_i - g_j
    n2 = 2 * n - 1
    p = np.arange(n)
    mi = (p != i) & (p + i < n2)
    g[p[mi], p[mi] + i] -= 2.0
    mj = (p != j) & (p + j < n2)
    g[p[mj], p[mj] + j] += 2.0
    h[i], h[j] = 1.0, 0.0
    return g, h, corr

def slsqp_polish(h, n, alpha, maxiter):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        m = corr.max()
        s = alpha * (corr - m)
        sm = s.max()
        return m + (sm + np.log(np.exp(s - sm).sum())) / alpha
    def cons(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': cons}]
    try:
        res = minimize(obj, h.copy(), method='SLSQP', bounds=bounds,
                       constraints=constraints, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h
