            break

        zm = zi - shifts_i

        om = oj - shifts_i

        B_val = -np.where(vp_o, 1.0 - h[np.clip(op, 0, n-1)], 0.0) + \
                np.where(vm_o, h[np.clip(om, 0, n-1)], 0.0)
        
        delta_c = A_val + B_val
        new_M = (c + delta_c).max()
        
        if new_M < best_M:
            best_M = new_M
            best_h = (h.copy())
        
        delta = new_M - M
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            # Accept
            h[zi] = 1.0
            h[oj] = 0.0
            c = c + delta_c
            M = new_M
            # Update zeros and ones
            # zi was in zeros, now should be in ones
            # oj was in ones, now should be in zeros
            zeros = np.delete(zeros, np.where(zeros == zi)[0][0])
            zeros = np.append(zeros, oj)
            ones = np.delete(ones, np.where(ones == oj)[0][0])
            ones = np.append(ones, zi)

def vectorized_greedy(h_in, n, deadline, max_iters=50000):
    h = h_in.copy()
    c = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    M = float(c.max())
    best_M = M
    best_h = h.copy()
    shifts_i = np.arange(2 * n - 1) - (n - 1)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if M < best_M:
            best_M = M
            best_h = h.copy()
        
        zeros = np.where(h < 0.5)[0]
        ones = np.where(h >= 0.5)[0]
        nz, no = len(zeros), len(ones)
        if nz == 0 or no == 0:
            break
        
        # Compute A_mat: (nz, 2n-1)
        zp = zeros[:, None] + shifts_i[None, :]
        zm = zeros[:, None] - shifts_i[None, :]
        vp = (zp >= 0) & (zp < n)
        vm = (zm >= 0) & (zm < n)
        A_mat = (np.where(vp, 1.0 - h[np.clip(zp, 0, n-1)], 0.0) -
                 np.where(vm, h[np.clip(zm, 0, n-1)], 0.0))
        
        # Compute B_mat: (no, 2n-1)
        op = ones[:, None] + shifts_i[None, :]
        om = ones[:, None] - shifts_i[None, :]
        vpo = (op >= 0) & (op < n)
        vmo = (om >= 0) & (om < n)
        B_mat = (-np.where(vpo, 1.0 - h[np.clip(op, 0, n-1)], 0.0) +
                 np.where(vmo, h[np.clip(om, 0, n-1)], 0.0))
        
        # D = M - c (shape 2n-1)
        D = M - c  # (2n-1,)

        # Equivalently: rhs_i[k] - B_j[k] >= 0 where rhs_i = D - A_i
        # rhs_mat: (nz, 2n-1)
        rhs_mat = D[None, :] - A_mat  # (nz, 2n-1)

            valid = np.all(diff_mat >= 0, axis=2)  # (nz, no)

                e = min(s + chunk, nz)
                valid[s:e] = np.all(rhs_mat[s:e, None, :] - B_mat[None, :, :] >= 0, axis=2)
        
        if not valid.any():

                new_maxes[s:e] = nm

def sa_fast(h_in, n, deadline, n_iter=30000, T_init=4.0, T_final=0.3, seed=42):
    rng = np.random.RandomState(seed)
    h = h_in.copy()
    c = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    M = float(c.max())
    best_M = M
    best_h = h.copy()
    shifts_i = np.arange(2 * n - 1) - (n - 1)
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        frac = it / n_iter
        T = T_init * (T_final / T_init) ** frac
        
        zeros = np.where(h < 0.5)[0]
        ones = np.where(h >= 0.5)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        
        zi = zeros[rng.randint(len(zeros))]
        oj = ones[rng.randint(len(ones))]
        
        # Compute delta
        zp = zi + shifts_i
        zm = zi - shifts_i
        vp = (zp >= 0) & (zp < n)
        vm = (zm >= 0) & (zm < n)
        Ai = np.where(vp, 1.0 - h[np.clip(zp, 0, n-1)], 0.0) - \
             np.where(vm, h[np.clip(zm, 0, n-1)], 0.0)
        
        op = oj + shifts_i
        om = oj - shifts_i
        vpo = (op >= 0) & (op < n)
        vmo = (om >= 0) & (om < n)
        Bj = (-np.where(vpo, 1.0 - h[np.clip(op, 0, n-1)], 0.0) +
              np.where(vmo, h[np.clip(om, 0, n-1)], 0.0))
        
        delta = Ai + Bj
        new_M = (c + delta).max()
        
        if new_M < best_M:
            best_M = new_M
            best_h = h.copy()
        
        d = new_M - M
        if d < 0 or rng.random() < np.exp(-d / max(T, 1e-10)):
            h[zi] = 1.0
            h[oj] = 0.0
            c = c + delta
            M = new_M
    
    return best_h, float(best_M * 2.0 / n)

def cont_optimize(h, n, deadline, maxiter=3000):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        alpha = 5000.0
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    def cons_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except:
        return h
