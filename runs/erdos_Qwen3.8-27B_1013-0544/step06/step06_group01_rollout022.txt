def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_lse(x, n, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 500:
        return float((m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha) * (2.0 / n))
    return float((m + np.log(np.sum(np.exp(s))) / alpha) * (2.0 / n))

def _feq(x, n):
    return x.sum() - n / 2.0

def _jeq(x, n):
    return np.ones(n)

def slsqp(h, n, alpha=50000, maxiter=800):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _feq, 'jac': _jeq, 'args': (n,)}]
    try:
        res = minimize(obj_lse, h.copy(), args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)
