def D_all(x, n):
    m = 2 * n - 1
    D = np.zeros((m, n))
    dx = 2.0 / n
    t2 = 2.0 * x - 1.0
    for kk in range(-(n - 1), n):
        a = kk + (n - 1)
        lo = max(0, -kk)
        hi = n - 1 - (kk if kk > 0 else 0)
        row = D[a]
        s1 = slice(lo + kk, hi + 1 + kk)
        s2 = slice(lo, hi + 1)
        row[s2] += dx * t2[s2] * x[s1]
        row[s1] += dx * t2[s1] * x[s2]
        row[s1] -= dx * t2[s1]
    return D

def wire_build(n, t_end, rng, keep_box=None):
    s = np.ones(n)
    vlist = []
    v = 1
    vmax = max(16, min(96, n // 6))
    while v <= vmax:
        vlist.append(v)
        v = v * 2
    vlist.append(max(3, vmax // 2))
    vlist.append(max(5, vmax))
    vlist = np.array(sorted(set(vlist)), dtype=float)
    nvs = len(vlist)
    W = np.exp(2j * np.pi * (vlist[:, None] * np.arange(n)[None, :].astype(float) / n))
    S = (W * s[None, :]).sum(axis=1)
    fc = np.zeros(n, dtype=int)
    it = 0
    while time.time() < t_end:
        it += 1
        pv = (S * S.conj()).real
        vv = int(pv.argmax())
        if pv[vv] < 0.08 * n:
            break
        if it % 700 == 0:
            hh = (s + 1.0) * 0.5
            ch = float(c5_vec(hh, n).max())
            if keep_box is not None and ch < keep_box[1]:
                keep_box[:] = [hh.copy(), ch, n]
        q = (S[vv] * np.conj(W[vv])).real
        delt = 4.0 − 4.0 * s * q
        pen = np.where(fc > 30, 0.5, 0.0)
        ii = int(np.argmin(delt + pen))
        if delt[ii] > -1e-7:
            break
        sv = s[ii]
        s[ii] = −sv
        fc[ii] += 1
        S -= 2.0 * sv * W[:, ii]
    c = int((s == -1).sum())
    tgt = n // 2
    if c < tgt:
        idxp = np.flatnonzero(s > 0.0)

import numpy as np
import time


def c5_vec(x, n):
    return np.correlate(x, 1.0 - x, mode="full") * (2.0 / n)


def c5(x, n):
    return float(c5_vec(x, n).max())


def proj(x, n):
    tgt = 0.5 * n
    x = np.clip(np.array(x, dtype=float).copy(), 0.0, 1.0)
    for _ in range(120):
        d = tgt - x.sum()
        if abs(d) <= 1e-12:
            break
        if d > 0:
            fr = x < 1.0 - 1e-13
        else:
            fr = x > 1e-13
        if not fr.any():
            break
        x[fr] += d / float(fr.sum())
        x = np.clip(x, 0.0, 1.0)
    for _ in range(30):
        d = tgt - x.sum()
        if abs(d) < 1e-14:
            break
        fr = (x > 1e-10) & (x < 1.0 - 1e-10)
        if not fr.any():
            break
        x[fr] += d / float(fr.sum())
    return x


def grad_all(x, n):
    m = 2 * n - 1
    F = np.zeros((m, n))
    dx = 2.0 / n
    for k in range(-(n - 1), n):
        a = k + (n - 1)
        lo = max(0, -k)
        hi = n - 1 - (k if k > 0 else 0)
        v = F[a]
        s1 = slice(lo + k, hi + 1 + k)
        s2 = slice(lo, hi + 1)
        v[s1] += dx
        v[s2] -= dx * x[s1]
        v[s1] -= dx * x[s2]
    return F


def D_all(x, n):
    m = 2 * n - 1
    D = np.zeros((m, n))
    dx = 2.0 / n
    t2 = 2.0 * x - 1.0
    for k in range(-(n - 1), n):
        a = k + (n - 1)
        lo = max(0, -k)
        hi = n - 1 - (k if k > 0 else 0)
        row = D[a]
        s1 = slice(lo + k, hi + 1 + k)
        s2 = slice(lo, hi + 1)
        row[s2] += dx * t2[s2] * x[s1]
        row[s1] += dx * t2[s1] * x[s2]
        row[s1] -= dx * t2[s1]
    return D


def pair_sweep_pass(x, n, M, t_end, rng):
    Mx = float(M.max())
    G = grad_all(x, n)
    idxA = np.arange(2 * n - 1)
    ttop = np.argsort(-M, kind="stable")[:12]
    Mparr = M[ttop]
    acc = 0
    ntr = int(min(1400, max(350, int(1.6 * n))))
    for tt in range(ntr):
        if time.time() > t_end:
            break
        u = rng.random_sample()
        if u < 0.30:
            i = int(rng.randint(n))
            j = (i + 1 + int(rng.randint(7))) % n
        else:
            i = int(rng.randint(n))
            j = int(rng.randint(n))
        if i == j:
            continue
        xi = x[i]
        xj = x[j]
        lo = max(-xi, xj - 1.0)
        hi = min(1.0 - xi, xj)
        if hi - lo < 1e-7:
            continue
        a = G[idxA[i]] - G[idxA[j]]
        dset = [lo, hi, 0.5 * (lo + hi), 0.25 * lo + 0.75 * hi, 0.75 * lo + 0.25 * hi]
        pk = int(np.argmax(M))
        ax = a[pk]
        if abs(ax) > 1e-9:
            dl = (Mx - 1e-9 - M[pk]) / ax
            dset.append(min(hi, max(lo, dl)))
        ap = a[ttop]
        ip = int(np.argmax(ap))
        iq = int(np.argmin(ap))
        if ap[ip] > 1e-9 and ap[iq] < -1e-9:
            dc = (Mparr[iq] - Mparr[ip]) / (ap[ip] - ap[iq])
            if lo <= dc <= hi:
                dset.append(dc)
        bestd = dset[0]
        bestv = float(np.max(M + bestd * a))
        for dd in dset[1:]:
            vv = float(np.max(M + dd * a))
            if vv < bestv:
                bestv = vv
                bestd = dd
        if bestv > Mx - 5e-8:
            continue
        blo, bhi = lo, hi
        for _ in range(16):
            if bhi - blo < 7e-5:
                break
            mid = 0.5 * (blo + bhi)
            ep = max(2e-7, 3e-5 * (bhi - blo))
            if np.max(M + (mid + ep) * a) < np.max(M + (mid - ep) * a):
                blo = mid
            else:
                bhi = mid
        cand = 0.5 * (blo + bhi)
        vv = float(np.max(M + cand * a))
        if vv < bestv:
            bestd = cand
        x[i] += bestd
        x[j] -= bestd
        Mnew = c5_vec(x, n)
        mm = float(Mnew.max())
        if mm + 1e-12 < Mx:
            M = Mnew
            Mx = mm
            acc += 1
            if acc % 10 == 0:
                G = grad_all(x, n)
            if acc == 60:
                ttop = np.argsort(-M, kind="stable")[:12]
                Mparr = M[ttop]
        else:
            x[i] -= bestd
            x[j] += bestd
    return x, M, acc


def flip_pass(x, n, M):
    Mx = float(M.max())
    for _ in range(60):
        D = D_all(x, n)
        vals = (M[:, None] + D).max(axis=0)
        ii = int(np.argmin(vals))
        if vals[ii] < Mx - 1e-9:
            x[ii] = 1.0 - x[ii]
            M = c5_vec(x, n)
            Mx = float(M.max())
        else:
            break
    return x, M


def qp_step(x, n, M, G, beta):
    Mx = float(M.max())
    w = np.exp(beta * (M - Mx))
    ws = float(w.sum())
    if ws <= 0.0:
        return x, M, False
    w = w / ws
    keep = np.flatnonzero(w > 1e-4)
    if len(keep) == 0:
        return x, M, False
    g = G[keep].T @ w[keep]
    if n <= 680:
        Q = np.zeros((n, n))
        dq = None
    else:
        Q = None
        dq = np.zeros(n)
    for ii in keep:
        k = int(ii) - (n - 1)
        wm = w[ii]
        lo = max(0, -k)
        hi = n - 1 - (k if k > 0 else 0)
        idx = np.arange(lo, hi + 1)
        jj = idx + k
        if Q is not None:
            if k == 0:
                Q[idx, idx] -= 2.0 * wm
            else:
                Q[idx, jj] -= wm
                Q[jj, idx] -= wm
        else:
            if k == 0:
                dq[idx] -= 2.0 * wm
    rho = 0.02
    if Q is not None:
        Q.flat[::n + 1] += rho
        try:
            delta = np.linalg.solve(Q, -g)
        except np.linalg.LinAlgError:
            delta = -g / (np.diag(Q) + rho)
    else:
        delta = -g / (dq + rho + 1e-9)
    scale = 0.03 * np.sqrt(512.0 / n)
    for lam in (1.0, 0.5, 0.25, 0.125):
        xc = np.clip(x + lam * scale * delta, 0.0, 1.0)
        Mn = c5_vec(xc, n)
        if float(Mn.max()) + 1e-12 < Mx:
            return xc, Mn, True
    return x, M, False


def nudge_x(x, n, rng, M):
    Mx = float(M.max())
    for am in (2e-3, 1e-3, 4e-3, 6e-3):
        dd = rng.randn(n)
        dd = dd - dd.mean()
        xc = np.clip(x + am * dd, 0.0, 1.0)
        Mc = c5_vec(xc, n)
        if float(Mc.max()) < Mx:
            return xc
    return x


def snap_binary(x, n):
    if n % 3 == 1:
        return x
    o = np.argsort(-x, kind="stable")
    xs = np.zeros(n)
    xs[o[:n // 3 * 2]][:n // 2] = 1.0
    xs[o[n // 2:n // 2 * 2]] = 0.0
    return xs


def refine_loop(x, n, t_end, rng):
    M = c5_vec(x, n)
    it = 0
    stall = 0
    bet = 4.0
    while time.time() < t_end:
        M0 = float(M.max())
        x, M, acc = pair_sweep_pass(x, n, M, min(t_end, time.time() + 6.0), rng)
        x, M = flip_pass(x, n, M)
        if it % 5 == 0 and time.time() < t_end - 6.0:
            G0 = grad_all(x, n)
            x, M, okq = qp_step(x, n, M, G0, bet)
            bet = 4.0 if okq else min(40.0, bet * 2.0)
        if it % 7 == 0 and n % 2 == 0 and time.time() < t_end - 3.0:
            xs = snap_binary(x, n)
            ms = float(c5_vec(xs, n).max())
            if ms < float(M.max()) - 1e-9:
                x = xs
                M = c5_vec(x, n)
        if acc == 0 and float(M.max()) >= M0 - 1e-12:
            stall += 1
            if stall >= 4:
                x = nudge_x(x, n, rng, c5_vec(x, n))
                stall = 0
        else:
            stall = 0
        it += 1
    return x, M


def wire_build(n, t_end, rng, box=None):
    s = np.ones(n)
    vmax = max(16, min(96, n // 6))
    vlist = []
    v = 1
    while v <= vmax:
        vlist.append(v)
        v = v * 2
    vlist.append(max(3, vmax // 2))
    vlist.append(vmax)
    vl = np.array(sorted(set(vlist)), dtype=float)
    W = np.exp(2j * np.pi * (vl[:, None] * np.arange(n, dtype=float)[None, :] / n))
    S = (W * s[None, :]).sum(axis=1)
    fc = np.zeros(n, dtype=int)
    it = 0
    while time.time() < t_end:
        it += 1
        pv = (S * S.conj()).real
        vv = int(pv.argmax())
        if pv[vv] < 0.08 * n:
            break
        if it % 700 == 0 and box is not None:
            hh = (s + 1.0) * 0.5
            ch = c5(hh, n)
            if ch < box[1]:
                box[0] = hh.copy()
                box[1] = ch
                box[2] = n
        q = (S[vv] * np.conj(W[vv])).real
        delt = 4.0 - 4.0 * s * q
        ii = int(np.argmin(delt + np.where(fc > 30, 0.5, 0.0)))
        if delt[ii] > -1e-7:
            break
        sv = s[ii]
        s[ii] = -sv
        fc[ii] += 1
        S = S - 2.0 * sv * W[:, ii]
    c = int((s < 0.0).sum())
    tgt = n // 2
    if c < tgt:
        idxp = np.flatnonzero(s > 0.0)
        idxp = idxp[rng.permutation(len(idxp))]
        s[idxp[:tgt - c]] = -1.0
    elif c > tgt:
        idxm = np.flatnonzero(s < 0.0)
        idxm = idxm[rng.permutation(len(idxm))]
        s[idxm[:c - tgt]] = 1.0
    return np.clip((s + 1.0) * 0.5, 0.0, 1.0)


def sturm_best(n, t_end):
    base = np.arange(n) + 0.5
    thes = 0.5 + np.linspace(-0.42, 0.42, 140)
    bestx = None
    bestc = 1e18
    for tt in thes:
        if time.time() > t_end - 0.5:
            break
        hv = np.where(np.mod(base * tt, 1.0) < 0.5, 1.0, 0.0)
        hv = proj(hv, n)
        cc = c5(hv, n)
        if cc < bestc:
            bestc = cc
            bestx = hv
    if bestx is None:
        bestx = np.full(n, 0.5)
    return bestx


def interp_to(h, nf, nt):
    xo = (np.arange(nf) + 0.5) * (2.0 / nf)
    xn = (np.arange(nt) + 0.5) * (2.0 / nt)
    return proj(np.interp(xn, xo, h), nt)


def check_grad():
    n = 42
    xs = np.clip(0.5 + 0.45 * np.cos(np.arange(n) * 0.73) * (1.0 - np.linspace(0.2, 0.6, n)), 0.05, 0.95)
    G = grad_all(xs, n)
    D = D_all(xs, n)
    eps = 1e-6
    err = 0.0
    for j in (0, 5, 20, 35, 41):
        ep = xs.copy()
        ep[j] = min(1.0, xs[j] + eps)
        em = xs.copy()
        em[j] = max(0.0, xs[j] - eps)
        sp = (c5_vec(ep, n) - c5_vec(em, n)) / (2.0 * eps)
        e1 = float(np.abs(sp - G[:, j]).max())
        if e1 > err:
            err = e1
        efl = xs.copy()
        efl[j] = 1.0 - xs[j]
        df = c5_vec(efl, n) - c5_vec(xs, n)
        e2 = float(np.abs(df - D[:, j]).max())
        if e2 > err:
            err = e2
    return err


def upd_box(box, x, n, Dend):
    if box is None:
        return
    cc = c5(x, n)
    if cc < box[1] - 1e-13 and time.time() < Dend:
        box[0] = np.clip(np.array(x, dtype=float), 0.0, 1.0).copy()
        box[1] = cc
        box[2] = int(n)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    Dend = t0 + float(budget_s) - 25.0
    rng = np.random.RandomState(int(seed))
    box = None
    try:
        check_grad()
        paren = None
        try:
            paren = np.array(initial_h_values, dtype=float).copy()
        except Exception:
            paren = None
        if paren is not None and len(paren) >= 24:
            x = proj(paren, len(paren))
            nst = len(paren)
        else:
            nst = 512
            x = wire_build(nst, min(t0 + 35.0, Dend - 300.0), rng)
        box = [x, c5(x, nst), nst]
        if Dend - t0 > 430.0:
            x, M = refine_loop(x, nst, min(t0 + 140.0, Dend - 320.0), rng)
            upd_box(box, x, nst, Dend)
        alloc = {512: 240.0, 640: 100.0, 768: 150.0, 960: 220.0, 1152: 90.0}
        for kk in (512, 640, 768, 960, 1152):
            if kk == nst:
                continue
            if time.time() > Dend - alloc[kk] - 95.0:
                continue
            if kk == 768:
                xh = wire_build(kk, time.time() + 25.0, rng, box)
            elif kk == 640:
                xh = sturm_best(kk, time.time() + 5.0)
            else:
                xh = interp_to(box[0], int(box[2]), kk)
            xh = proj(xh, kk)
            te = min(time.time() + alloc[kk], Dend - 95.0)
            xh, Mh = refine_loop(xh, kk, te, rng)
            upd_box(box, xh, kk, Dend)
        if time.time() < Dend - 10.0:
            xb = proj(box[0].copy(), int(box[2]))
            Mb = c5_vec(xb, int(box[2]))
            nb = int(box[2])
            while time.time() < Dend - 9.0:
                xb, Mb, _ = pair_sweep_pass(xb, nb, Mb, min(Dend - 9.0, time.time() + 4.0), rng)
                xb, Mb = flip_pass(xb, nb, Mb)
            upd_box(box, xb, nb, Dend)
        if time.time() < Dend - 7.0:
            nx2 = int(box[2]) + 32
            x1 = interp_to(box[0], int(box[2]), nx2)
            x1, M1, _ = pair_sweep_pass(x1, nx2, c5_vec(x1, nx2), min(Dend - 4.0, time.time() + 3.0), rng)
            upd_box(box, x1, nx2, Dend)
        hb = box[0]
        nb = int(box[2])
        if nb % 2 == 0:
            hs = snap_binary(hb, nb)
            cs = c5(hs, nb)
            if cs < box[1]:
                hb = hs
        hb = np.clip(proj(hb, nb), 0.0, 1.0)
        return hb, c5(hb, nb), nb
    except Exception:
        if box is not None and len(box) >= 3:
            hb = proj(box[0], int(box[2]))
            return hb, c5(hb, int(box[2])), int(box[2])
        hb = np.full(300, 0.5)
        return hb, 0.5, 300
