def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

    c = compute_correlation(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:

    return m + np.log(np.sum(np.exp(sh))) / alpha

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_correlation(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smo_obj(h, n, alpha):
    c = compute_correlation(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(sh))) / alpha)

def smo_grad(h, n, alpha):
    c = compute_correlation(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-14:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def active_grad(h, n):
    c = compute_correlation(h, n)
    m_idx = int(np.argmax(c))
    k = m_idx - (n - 1)
    dx = 2.0 / n
    grad = np.zeros(n)
    if k >= 0:
        nv = n - k
        grad[0:nv] += dx * (1.0 - h[k:n])
        grad[k:n] += dx * (-h[0:nv])
    else:
        kk = -k
        nv = n - kk
        grad[kk:n] += dx * (1.0 - h[0:nv])
        grad[0:nv] += dx * (-h[kk:n])
    return grad

def active_descent(h, n, deadline):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = 0.01
    for _ in range(100000):
        if time.time() > deadline:
            break
        g = active_grad(h, n)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g = g / gn
        improved = False
        for _ in range(25):
            h_new = project_feasible(h - lr * g, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                h = h_new
                best_c = c_new
                best_h = h.copy()
                improved = True
                break
            lr *= 0.5
        if not improved:
            lr *= 0.3
            if lr < 1e-14:
                break
    return best_h, best_c

def stoch_pairs(h, n, deadline, rng, n_steps=100000, temp_init=0.004):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    cur_c = best_c
    temp = temp_init
    for step in range(n_steps):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(-0.012, 0.012)
        hi = h[i] + d
        hj = h[j] - d
        if hi < 0 or hi > 1 or hj < 0 or hj > 1:
            continue
        h[i] = hi
        h[j] = hj
        c_new = compute_c5(h, n)
        delta = c_new - cur_c
        if delta < 0 or (temp > 1e-13 and rng.random() < np.exp(-delta / temp)):
            cur_c = c_new
            if c_new < best_c - 1e-16:
                best_c = c_new
                best_h = h.copy()
        else:
            h[i] = hi - d
            h[j] = hj + d
        if step > 0 and step % 3000 == 0:
            temp *= 0.94
    return best_h, best_c

def block_search(h, n, deadline, rng):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    for _ in range(60000):
        if time.time() > deadline:
            break
        i = rng.randint(0, max(1, n - 3))

        h2[i:i+bsz] -= 0.015
        h2[j:j+bsz] += 0.015
        if np.all(h2 >= -0.05) and np.all(h2 <= 1.05):
            h2 = project_feasible(h2, n)
            c = compute_c5(h2, n)
            if c < best_c - 1e-16:
                best_c = c
                best_h = h2.copy()
                h = h2
    return best_h, best_c

def lbfgs_optimize(h, n, alpha, deadline):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smo_obj, h, jac=smo_grad, args=(n, alpha),

                      options={'maxiter': 200, 'maxfun': 2000,
                               'ftol': 1e-18, 'gtol': 1e-14})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)
