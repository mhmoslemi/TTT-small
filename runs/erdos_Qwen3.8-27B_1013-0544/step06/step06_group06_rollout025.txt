def active_indices(C,k,band):
    cmax=float(C.max()); m=C.size; k=max(1,min(int(k),m))
    idx=np.flatnonzero(C >= cmax - band)
    if idx.size == 0:
        idx=np.array([int(np.argmax(C))], dtype=int)
    if idx.size >= k:
        if idx.size > k:
            order=np.argsort(-C[idx])[:k]
            idx=idx[order]
        return idx
    topk=np.argpartition(C, m-k)[-k:] if k < m else np.arange(m)
    all_idx=np.union1d(idx, topk)
    if all_idx.size > k:
        order=np.argsort(-C[all_idx])[:k]
        all_idx=all_idx[order]
    return all_idx

def lag_slices(n):

        return cached

        out.append((lo1, hi1, lo1 + d, hi1 + d, lo2, hi2, lo2 - d, hi2 - d))

def corr_C(h, n):

def compute_c5(h, n):

def project_box_sum(v, target):
    v = np.asarray(v, dtype=float)
    n = v.size
    target = float(target)

def binary_from_cont(score, n):

    vals = np.remainder(x * alpha, 1.0)

            hh, ch = lp_polish(hn, n, deadline - 3.0, rng)
            if ch < best_c:
                best, best_c = hh, ch

    best = project_box_sum(best, target)
    best_c = compute_c5(best, n)
    if not np.isfinite(best_c):
        best = project_box_sum(sturmian_init(n, 0.618033988749895), target)
        best_c = compute_c5(best, n)
    return best, float(best_c), int(n)
