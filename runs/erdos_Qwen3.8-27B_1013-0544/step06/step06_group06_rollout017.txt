def slsqp_softmax(h, n, T, maxiter, deadline):
    """SLSQP optimization with soft-max objective."""
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        if T < 1e-10:
            return m
        shifted = (c - m) / T
        logsum = m + T * np.log(np.sum(np.exp(shifted)))
        return logsum

        return x.sum() - n / 2.0

        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    
    try:

                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})

def binarize(h, n):
    """Binarize h while maintaining sum = n/2."""

    idx = np.argsort(h)[::-1]

def thue_morse_init(n, rng):
    t = np.zeros(n, dtype=int)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)

    elif num_ones < target:
        zeros = np.where(t == 0)[0]
        rng.shuffle(zeros)
        t[zeros[:target - num_ones]] = 1
    return t.astype(float)
