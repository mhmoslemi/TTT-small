def fast_cd(h, n, deadline, grid=100, penalty=5.0, rng=None):
    """Fast coordinate descent with approximate line search."""
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    current_max = float(np.max(c))
    sumh = float(h.sum())
    
    while time.time() < deadline:
        improved = False
        indices = np.arange(n)
        if rng is not None:
            rng.shuffle(indices)
        for p in indices:
            d_lo = -h[p]
            d_hi = 1.0 - h[p]
            if d_hi - d_lo < 1e-14:
                continue
            k_lo = min(n - 1, p)
            k_hi = min(2 * n - 2, max(n - 1 + p, 2 * n - 2 - p))
            # Actually the affected range is:
            # c[k] is affected when the index i=p contributes to c[k]
            # c[k] includes h[p] * (1-h[p+k-n+1]) when 0<=p+k-n+1<=n-1
            # i.e., n-1-p <= k <= n-1+p
            # And c[k] includes (1-h[p]) * h[p-k+n-1] when 0<=p-k+n-1<=n-1
            # i.e., p-n+1 <= k <= p+n-1
            # Combined: max(0, n-1-p, p-n+1) <= k <= min(2n-2, n-1+p, p+n-1)
            # = max(0, |n-1-p|) <= k <= min(2n-2, n-1+max(p, n-1-p))
            
            k_start = max(0, n - 1 - p, p - n + 1)
            k_end = min(2 * n - 2, n - 1 + p, p + n - 1)
            
            if k_start > k_end:
                continue
            
            um = 0.0
            if k_start > 0:
                um = float(np.max(c[:k_start]))
            if k_end < 2 * n - 2:
                um = max(um, float(np.max(c[k_end + 1:])))
            
            ks = np.arange(k_start, k_end + 1)
            s = np.zeros(len(ks))
            for idx, k in enumerate(ks):
                # Contribution from h[p] as first factor: (1 - h[p+k-n+1]) if valid
                j1 = p + k - (n - 1)
                if 0 <= j1 <= n - 1:
                    s[idx] += (1.0 - h[j1])
                # Contribution from h[p] as second factor: -h[p-k+n-1] if valid
                j2 = p - k + (n - 1)
                if 0 <= j2 <= n - 1:
                    s[idx] -= h[j2]
            
            deltas = np.linspace(d_lo, d_hi, grid)
            best_d = 0.0
            best_val = current_max + penalty * (sumh - n / 2.0) ** 2
            
            for d in deltas:
                if abs(d) < 1e-14:
                    continue
                new_c = c[ks] + d * s
                # Special handling for k = n-1 (diagonal): c[n-1] += d*(1-2h[p]) - d^2
                ci = (ks == (n - 1))
                if np.any(ci):
                    new_c[ci] = c[n - 1] + d * (1.0 - 2.0 * h[p]) - d * d
                val = max(um, float(np.max(new_c)))
                if penalty > 0:
                    val += penalty * (sumh + d - n / 2.0) ** 2
                if val < best_val - 1e-14:
                    best_val = val
                    best_d = d
            
            if abs(best_d) > 1e-12:
                d = best_d
                for idx, k in enumerate(ks):
                    j1 = p + k - (n - 1)
                    j2 = p - k + (n - 1)
                    if 0 <= j1 <= n - 1:
                        c[k] += d * (1.0 - h[j1])
                    if 0 <= j2 <= n - 1:
                        c[k] -= d * h[j2]
                # Fix the diagonal
                c[n - 1] -= d * d
                h[p] = np.clip(h[p] + d, 0.0, 1.0)
                sumh = float(h.sum())
                current_max = max(um, float(np.max(c[k_start:k_end + 1])))
                improved = True
        
        if not improved:
            break
    
    return h, current_max * (2.0 / n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lse_vals(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    e = np.exp(alpha * (c - m))
    val = (m + np.log(e.sum()) / alpha) * dx
    return val

def lse_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    e = np.exp(alpha * (c - m))
    w = e / e.sum()
    grad = np.zeros(n)
    for r in range(2 * n - 1):
        wr = w[r]
        if wr < 1e-18:
            continue
        s = r - (n - 1)
        p_lo1 = max(0, s)
        p_hi1 = min(n - 1, s + n - 1)
        if p_lo1 <= p_hi1:
            j1 = np.arange(p_lo1, p_hi1 + 1) - s
            grad[p_lo1:p_hi1 + 1] += wr * (1.0 - h[j1])
        p_lo2 = max(0, -s)
        p_hi2 = min(n - 1, n - 1 - s)
        if p_lo2 <= p_hi2:
            j2 = np.arange(p_lo2, p_hi2 + 1) + s
            grad[p_lo2:p_hi2 + 1] -= wr * h[j2]
    return grad * dx

def lse_obj_pen(h, n, alpha, rho):
    val = lse_vals(h, n, alpha)
    val += rho * (h.sum() - n / 2.0) ** 2
    return float(val)

def lse_grad_pen(h, n, alpha, rho):
    g = lse_grad(h, n, alpha)
    g += 2.0 * rho * (h.sum() - n / 2.0)
    return g

def optimize_lse(h, n, alphas, deadline, rho=50.0, maxiter_per=300):
    h = project_feasible(h.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        if time.time() > deadline:
            break
        def obj(x):
            return lse_obj_pen(np.clip(x, 0, 1), n, alpha, rho)
        def grad(x):
            xc = np.clip(x, 0, 1)
            return lse_grad_pen(xc, n, alpha, rho)
        try:
            res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': maxiter_per, 'ftol': 1e-15, 'gtol': 1e-8})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c, best_h = c_new, h_new
            h = h_new
        except Exception:
            pass
    return best_h, best_c

def fast_cd(h, n, deadline, grid=100, penalty=5.0, rng=None):

            best_c, best_h = c4, h4
    # Phase 5: fine coordinate descent
    if time.time() < deadline - 20:
        h5, c5 = fast_cd(best_h, n_init, deadline - 8, grid=200, penalty=0.1, rng=rng)
        h5 = project_feasible(h5, n_init)
        c5 = compute_c5(h5, n_init)
        if c5 < best_c:
            best_c, best_h = c5, h5
    # Phase 6: try smaller n
    for n_try in [200, 150, 100]:
        if time.time() > deadline - 60:
            break
        x_old = np.linspace(0, 2, n_init)
        x_new = np.linspace(0, 2, n_try)
        h_interp = project_feasible(np.interp(x_new, x_old, best_h), n_try)
        c_interp = compute_c5(h_interp, n_try)
        t_sub = min(50, deadline - time.time() - 15)
        if t_sub < 15:
            break
        h_alt, c_alt = optimize_lse(h_interp, n_try, [5000, 50000], time.time() + t_sub, rho=100.0, maxiter_per=200)
        h_alt = project_feasible(h_alt, n_try)
        c_alt = compute_c5(h_alt, n_try)
        if c_alt < c_interp:
            h_interp, c_interp = h_alt, c_alt
        if time.time() < deadline - 10:
            h_cd, c_cd = fast_cd(h_interp, n_try, deadline - 5, grid=120, penalty=1.0, rng=rng)
            h_cd = project_feasible(h_cd, n_try)
            c_cd = compute_c5(h_cd, n_try)
            if c_cd < c_interp:
                h_interp, c_interp = h_cd, c_cd
        if c_interp < best_c:
            best_c, best_h = c_interp, h_interp
    # Final
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
