```python
    """Compute flip_01[p] for all p: the change in correlation when h[p] goes 0->1."""
    # flip_01[p] is a vector of length 2n-1
    # flip_01[p][m] where m = L + n - 1, L is the lag
```

```python
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            # Add proportionally to room, but only to elements not at 1
            mask = room > 1e-10
            if mask.any():
                h[mask] += diff / room[mask].sum() * room[mask]
            h = np.clip(h, 0.0, 1.0)
        else:
            mask = h > 1e-10
            if mask.sum() == 0:
                break
            h[mask] += diff / h[mask].sum() * h[mask]
            h = np.clip(h, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def compute_flip(h, n):
    """Compute flip effects for all positions. flip[p] = change in corr when h[p] goes 0->1."""
    ncorr = 2 * n - 1
    flip = np.zeros((n, ncorr))
    # For each p, term 1: add (1-h[p+L]) at index L+n-1 for L=0..n-1-p
    # term 2: subtract h[p-L] at index L+n-1 for L=p-(n-1)..p
    for p in range(n):
        # Term 1: L from 0 to n-1-p
        if n - 1 - p >= 0:
            Ls = np.arange(0, n - p)
            flip[p, Ls + n - 1] += (1 - h[p: n])
        # Term 2: L from p-(n-1) to p
        Ls = np.arange(p - (n-1), p + 1)
        idx = p - Ls  # = h index
        valid = (idx >= 0) & (idx < n)
        flip[p, Ls[valid] + n - 1] -= h[idx[valid]]
    return flip

def binary_sa(h, n, deadline, seed=42):
    """Simulated annealing on binary sequences."""
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    M = c.max()
    best_h = h.copy()
    best_M = M
    
    T = 10.0  # initial temperature
    T_min = 0.01
    cool_rate = 0.9995
    
    n_steps = 0
    stuck_count = 0
    
    while time.time() < deadline and T > T_min:
        # Recompute flip if needed (after each swap)
        flip = compute_flip(h, n)
        
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        
        if len(zeros) == 0 or len(ones) == 0:
            break
        
        # Pick a random zero and one
        i = rng.choice(zeros)
        j = rng.choice(ones)
        
        # Compute delta = flip[i] - flip[j]
        delta = flip[i] - flip[j]
        
        # New correlation
        new_c = c + delta
        new_M = new_c.max()
        
        # Decide whether to accept
        if new_M <= M:
            accept = True
        else:
            delta_M = new_M - M
            accept = rng.rand() < np.exp(-delta_M / T)
        
        if accept:
            # Make the swap
            h[i] = 1.0
            h[j] = 0.0
            c = new_c
            M = new_M
            n_steps += 1
            stuck_count = 0
            if M < best_M:
                best_M = M
                best_h = h.copy()
        else:
            stuck_count += 1
            if stuck_count > 5000:
                # Restart from best
                h = best_h.copy()
                c = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
                M = c.max()
                stuck_count = 0
        
        T *= cool_rate
    
    return best_h, float(best_M * dx)
```

```python
                # delta_j for j going 1->0 is -delta_i formula applied to j
                # Actually, for j going 1->0:
                # delta_j[L] = -(1-h[j+L]) for valid j+L + h[j-L] for valid j-L
                # This is -flip_01[j]
                delta_j = np.zeros(2*n-1)
                L_lo = max(0, -j)
                L_hi = min(n-1-j, n-1)
                if L_lo <= L_hi:
                    delta_j[L_lo+n-1 : L_hi+n] -= (1 - h[j+L_lo : j+L_hi+1])
                L_lo2 = max(j-(n-1), -(n-1))
                L_hi2 = min(j, n-1)
                if L_lo2 <= L_hi2:
                    delta_j[L_lo2+n-1 : L_hi2+n] += h[j-L_hi2 : j-L_lo2:-1]
                
                new_c = c + delta_i + delta_j
                new_M = new_c.max()
                if new_M < M:
                    h[i] = 1.0
                    h[j] = 0.0
                    c = new_c
                    M = new_M
                    improved = True
                    break
        if not improved:
            break
    return h, M
```

```python
def compute_flip(h, n):
    ncorr = 2 * n - 1
    flip = np.zeros((n, ncorr))
    for p in range(n):
        # 0->1 flip effect
        # Term 1: (1-h[p+L]) for L where 0<=p+L<n
        L1 = np.arange(0, n - p)
        flip[p, L1 + n - 1] += (1 - h[p:p + len(L1)])
        # Term 2: -h[p-L] for L where 0<=p-L<n
        # p-L >= 0 => L <= p; p-L < n => L > p-n
        L2 = np.arange(max(p - (n-1), -(n-1)), p + 1)
        idx = p - L2
        flip[p, L2 + n - 1] -= h[idx]
    return flip
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
            if mask.sum() == 0:
                break
            h[mask] += diff / mask.sum()
        else:
            mask = h > 1e-10
            if mask.sum() == 0:
                break
            h[mask] *= (h[mask].sum() + diff) / h[mask].sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def binarize(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

def compute_flip(h, n):
    nc = 2 * n - 1
    flip = np.zeros((n, nc))
    for p in range(n):
        L1 = np.arange(0, n - p)
        if len(L1) > 0:
            flip[p, L1 + n - 1] += (1.0 - h[p:p + len(L1)])
        L2 = np.arange(max(p - n + 1, -(n - 1)), p + 1)
        if len(L2) > 0:
            flip[p, L2 + n - 1] -= h[p - L2]
    return flip

def greedy_improve(h, n, deadline):
    dx = 2.0 / n
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    M = c.max()
    while time.time() < deadline:
        flip = compute_flip(h, n)
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        threshold_base = c - M + 1.0
        found = False
        for i in zeros:
            if time.time() > deadline:
                break
            thresh = threshold_base + flip[i]
            valid = (flip[ones] >= thresh[None, :]).all(axis=1)
            if valid.any():
                j = ones[valid.argmax()]
                h[i] = 1.0
                h[j] = 0.0
                c = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
                M = c.max()
                found = True
                break
        if not found:
            break
    return h, M, float(M * dx)

def binary_sa(h, n, deadline, seed=42):
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    M = c.max()
    best_h = h.copy()
    best_M = M
    T = 3.0
    T_min = 0.05
    no_improve = 0
    step = 0
    while time.time() < deadline:
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        step += 1
        if step % 5000 == 0:
            flip = compute_flip(h, n)
        else:
            flip = None
        i = rng.choice(zeros)
        j = rng.choice(ones)
        if flip is not None:
            delta = flip[i] - flip[j]
            new_M = (c + delta).max()
        else:
            h[i] = 1.0
            h[j] = 0.0
            new_c = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
            new_M = new_c.max()
            if new_M <= M or rng.rand() < np.exp(-(new_M - M) / max(T, 1e-9)):
                c = new_c
                M = new_M
                if M < best_M:
                    best_M = M
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h[i] = 0.0
                h[j] = 1.0
                no_improve += 1
            T = max(T * 0.9997, T_min)
            if no_improve > 20000:
                h = best_h.copy()
                c = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
                M = c.max()
                T = 2.0
                no_improve = 0
            continue
        if new_M <= M or rng.rand() < np.exp(-(new_M - M) / max(T, 1e-9)):
            h[i] = 1.0
            h[j] = 0.0
            c = c + delta
            M = new_M
            if M < best_M:
                best_M = M
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        T = max(T * 0.9997, T_min)
        if no_improve > 20000:
            h = best_h.copy()
            c = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
            M = c.max()
            T = 2.0
            no_improve = 0
    return best_h, float(best_M * dx)

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h

def slsqp_top(h, n, k, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        topk = np.argsort(g)[-k:]
        sh = 500.0 * (g[topk] - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / 500.0
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h

    t0 = time.time()

        if cb2 < best_c:

            best_h = hb2.copy()

    # Phase 3: More SA with different seed from best
    p3 = t0 + 0.75 * (budget_s - 5)

            best_h = hb3.copy()

    if time.time() < p4:

        for alpha in [5000, 50000, 500000]:
            if time.time() > p4:

            hc = slsqp_smooth(h_cont, n, alpha, 2000, p4)
            cc = compute_c5(hc, n)
            if cc < best_c:

        if time.time() < p4:

        hb5, mb5, cb5 = greedy_improve(hb5, n, dl - 1)

            best_c = cb5
            best_h = hb5.copy()

        hc = slsqp_smooth(best_h.copy(), n, 1000000, 1000, dl)
        cc = compute_c5(hc, n)
        if cc < best_c:
            best_c = cc
            best_h = hc.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
```
