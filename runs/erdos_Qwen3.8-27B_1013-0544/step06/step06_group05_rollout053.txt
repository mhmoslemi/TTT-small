def fast_correlation(h, n):
    """Compute full correlation using FFT for speed."""
    N = 1
    while N < 2 * n - 1:
        N *= 2
    H = fft(h, N)
    H1 = fft(1.0 - h, N)
    c = np.real(ifft(H * np.conj(H1)))

    h[i] should be 1 and h[j] should be 0."""

    v = (ip >= 0) & (ip < n); res[v] += h[ip[v]]
    v = (im >= 0) & (im < n); res[v] += h[im[v]]
    v = (jp >= 0) & (jp < n); res[v] -= h[jp[v]]
    v = (jm >= 0) & (jm < n); res[v] -= h[jm[v]]

def greedy_search(h, n, deadline, n_crit=10, top_cand=15, max_stall=100):

    cur_max = c.max()
    ones_list = np.where(h == 1)[0].copy()
    zeros_list = np.where(h == 0)[0].copy()
    stall = 0

        if stall > max_stall:
            break
        if len(ones_list) == 0 or len(zeros_list) == 0:
            break
        top = np.argsort(c)[-n_crit:]
        sc1 = np.zeros(len(ones_list))
        sc0 = np.zeros(len(zeros_list))
        for li in top:
            lag = int(li) - (n - 1)
            if lag == 0: continue
            for sgn in (1, -1):
                idx1 = ones_list + sgn * lag
                v1 = (idx1 >= 0) & (idx1 < n)
                sc1[v1] += h[idx1[v1]]
                idx0 = zeros_list + sgn * lag
                v0 = (idx0 >= 0) & (idx0 < n)
                sc0[v0] += h[idx0[v0]]
        ti = np.argsort(sc1)[:top_cand]
        tj = np.argsort(sc0)[::-1][:top_cand]
        best = None
        best_m = cur_max
        for ii in ti:
            i = int(ones_list[ii])
            for jj in tj:
                j = int(zeros_list[jj])
                if i == j: continue
                dm = (c + compute_delta(h, i, j, n)).max()
                if dm < best_m - 1e-9:
                    best_m = dm
                    best = (ii, jj, i, j)
        if best is None:
            stall += 1
            continue
        stall = 0
        ii, jj, i, j = best
        c += compute_delta(h, i, j, n)
        cur_max = c.max()
        h[i] = 0.0; h[j] = 1.0
        ones_list = np.delete(ones_list, ii)
        zeros_list = np.delete(zeros_list, jj)
        ones_list = np.append(ones_list, j)
        zeros_list = np.append(zeros_list, i)
    return h, cur_max

def compute_c5(h, n):

    res = np.zeros(2 * n - 1)

    vm = (im >= 0) & (im < n); res[vm] += h[im[vm]]
