def smooth_binary_lp(h0, n, max_iter=100):
    """Use LP to find optimal smoothing of binary h0"""
    import cvxpy as cp
    
    corr0 = np.correlate(h0, 1 - h0, mode='full')
    dx = 2.0 / n
    
    # Build the constraint matrix
    # corr[k] = sum over valid j of h[j] * (1-h[j-d]) where d = n-1-k
    # Linearization: delta_corr[k] = sum_i M[k,i] * delta[i]
    
    m = 2 * n - 1
    M = np.zeros((m, n))
    
    for k in range(m):
        d = n - 1 - k  # lag
        if d >= 0:
            # corr[k] = sum_{j=d}^{n-1} h[j] * (1-h[j-d])
            for j in range(d, n):
                M[k, j] += (1 - h0[j - d])  # coefficient of delta[j]
                if j - d >= 0:
                    M[k, j - d] -= h0[j]  # coefficient of delta[j-d]
        else:
            # d < 0, let e = -d
            # corr[k] = sum_{j=0}^{n-1+e} h[j] * (1-h[j+e])... 
            # Actually for k > n-1, d = n-1-k < 0
            # corr[k] = sum_{j=0}^{n-1} h[j] * (1-h[j-d]) where j-d in [0,n-1]
            # j-d = j+|d|, so need j+|d| <= n-1, i.e., j <= n-1-|d|
            e = -d
            for j in range(0, n - e):
                M[k, j] += (1 - h0[j + e])  # coeff of delta[j]
                if j + e < n:
                    M[k, j + e] -= h0[j]  # coeff of delta[j+e]
    
    # LP: minimize t
    # s.t. corr0[k] + M[k,:] @ delta <= t for all k
    #      sum(delta) = 0
    #      -h0[i] <= delta[i] <= 1-h0[i]
    
    delta = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [
        M @ delta <= t - corr0,  # Wait: corr0 + M@delta <= t => M@delta <= t - corr0
        cp.sum(delta) == 0,
        delta >= -h0,
        delta <= 1 - h0,
        t >= 0
    ]
    
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.GLPK)
        if status == 'optimal':
            return h0 + delta.value
    except:
        pass
    return h0

def equalize_smooth(h0, n, n_active=10, max_iter=50, epsilon=0.1):
    """Iteratively smooth by equalizing the top correlation values"""
    h = h0.copy()
    for iter in range(max_iter):
        corr = np.correlate(h, 1 - h, mode='full')
        # Find active lags
        top_idx = np.argsort(corr)[::-1][:n_active]
        # For each active lag k, compute gradient of corr[k] w.r.t. h
        # grad[k, i] = d(corr[k])/d(h[i])
        # corr[k] = sum_j h[j]*(1-h[j-d]) => d/dh[i] = (1-h[i-d])*1_{i in range1} - h[i+d]*1_{i in range2}
        # We want to find delta that makes corr[top_idx] all equal and minimized
        # This is: minimize max_k corr[k] subject to sum(delta)=0, |delta|<=eps, h+delta in [0,1]
        # Simplified: minimize sum of (corr[k] + grad[k]@delta - target)^2 for k in active
        # with target = mean of active corr values
        
        # Actually, let's use a simpler approach:
        # For each active lag, compute the "deletion vector" (which positions to reduce)
        # and "addition vector" (which positions to increase)
        
        # The gradient of corr[k] w.r.t. h[i]:
        # For d = n-1-k >= 0: corr[k] = sum_{j=d}^{n-1} h[j]*(1-h[j-d])
        # d(corr[k])/dh[i] = (1-h[i-d]) if i>=d and i<=n-1 (from h[j] term, j=i)
        #                    - h[i+d] if i+d<=n-1 and i>=0 (from h[j-d] term, j-d=i, j=i+d)
        # So grad = (1-h[i-d])*1_{i>=d} - h[i+d]*1_{i+d<n}
        
        # We want to find delta minimizing max_k (corr[k] + grad_k @ delta)
        # subject to sum(delta)=0, -h[i]<=delta[i]<=1-h[i], |delta[i]|<=eps
        
        # Use a simple approach: for each active lag, find the best single-pair swap
        # that reduces it, then average.
        
        # Even simpler: use projected gradient descent on the smooth max
        alpha = 0.01
        grad_max = np.zeros(n)
        # softmax-like weighting
        weights = np.exp(corr[top_idx] / 0.01)
        weights /= weights.sum()
        for w, k in zip(weights, top_idx):
            d = n - 1 - k
            g = np.zeros(n)
            if d >= 0:
                g[d:] = (1 - h[:n-d])
                if n-d > 0:
                    g[:n-d] -= h[d:]
            else:
                e = -d
                g[:n-e] = (1 - h[e:])
                g[e:] -= h[:n-e]
            grad_max += w * g
        # Projected gradient step
        h_new = h - alpha * grad_max
        h_new = np.clip(h_new, 0, 1)
        # Restore sum
        h_new = h_new + (h.sum() - h_new.sum()) / n
        h_new = np.clip(h_new, 0, 1)
        # Iterate until convergence
        if np.max(np.abs(h_new - h)) < 1e-8:
            break
        h = h_new
    
    return h

def swap_delta_vec(h, i, j):
    n = len(h)
    delta = np.zeros(2 * n - 1)
    # i: 0->1
    m1 = np.arange(max(0, n-1-i), min(2*n-2, n-1+i) + 1)
    delta[m1] += 1 - h[i + m1 - (n-1)]
    m2 = np.arange(max(0, i), min(2*n-2, i+n-1) + 1)
    delta[m2] -= h[i - m2 + (n-1)]
    # j: 1->0
