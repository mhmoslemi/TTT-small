def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_grad(h, n, alpha):
    """Compute gradient of smoothed max objective."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    # Softmax weights
    c_shifted = c - c.max()
    p = np.exp(alpha * c_shifted)
    p /= p.sum()
    # Gradient: for each lag k, contribution is p[n-1+k] * ((1-h)[j-k] - h[j+k])
    grad = np.zeros(n)
    for k in range(-(n-1), n):
        pk = p[n-1+k]
        if pk < 1e-15:
            continue
        # (1-h)[j-k] for valid j
        js = max(0, k)
        je = min(n, n + k)
        if js < je:
            grad[js:je] += pk * (1.0 - h[js-k:je-k])

        je2 = min(n, n - k)
        if js2 < je2:

    grad *= dx
    return grad
