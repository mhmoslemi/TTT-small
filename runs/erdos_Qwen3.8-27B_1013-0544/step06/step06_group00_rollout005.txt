import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(x, alpha, n):
    dx = 2.0 / n
    h = x
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    z = alpha * c
    m = z.max()
    exp_z = np.exp(z - m)
    return m / alpha + np.log(exp_z.sum()) / alpha

def smooth_max_val_grad(x, alpha, n):
    dx = 2.0 / n
    h = x
    out = np.correlate(h, 1.0 - h, mode='full')
    c = out * dx
    z = alpha * c
    m = z.max()
    exp_z = np.exp(z - m)

    val = m / alpha + np.log(exp_z.sum()) / alpha
    w = p * dx
    w[z.argmax()] += dx
    grad = np.zeros(n)
    grad += w[n - 1] * (1.0 - 2.0 * h)
    for s in range(1, n):
        wp = w[n - 1 + s]
        wm = w[n - 1 - s]

        return x.sum() - n / 2.0

        return np.ones(n)

        best_c, best_h = ca, ha.copy()

    # Phase 4: Re-smooth from best (continuous refinement)
    p4_end = t0 + 0.85 * total
    h = best_h.copy()
    for alpha in [5000, 50000, 500000, 5000000, 50000000, 500000000]:
        if time.time() > p4_end - 5:
            break
        h = slsqp_opt(h, n, alpha, 3000, 1e-18)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 5: Restart with perturbation
    p5_end = t0 + 0.93 * total
    h = best_h + 0.004 * rng.randn(n)
    h = project_feasible(h, n)
    for alpha in [1000000, 10000000, 100000000]:
        if time.time() > p5_end - 3:
            break
        h = slsqp_opt(h, n, alpha, 2000, 1e-18)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 6: Try another resolution
    if time.time() < p5_end + 8:
        nh = n + 100
        if nh % 2 == 0:
            xo = np.linspace(0, 2, n)
            xn = np.linspace(0, 2, nh)
            hh = project_feasible(np.interp(xn, xo, best_h), nh)
            hh = slsqp_opt(hh, nh, 100000, 1000, 1e-16)
            ch = compute_c5(hh, nh)
            if ch < best_c:
                best_c, best_h, n = ch, hh.copy(), nh

    # Phase 7: Final subgradient polish
    if time.time() < dl - 2:
        h = best_h.copy()
        dx = 2.0 / n
        out = np.correlate(h, 1.0 - h, mode='full')
        lr = 0.01
        for step in range(500):
            if time.time() > dl - 2:
                break
            c_full = out * dx
            kmax = c_full.argmax()
            s = kmax - (n - 1)
            if s == 0:
                g = (1.0 - 2.0 * h) * dx
            else:
                g = np.zeros(n)
                if s > 0:
                    g[s:] += (1.0 - h[:n - s]) * dx
                    g[:n - s] -= h[s:] * dx
                else:
                    s2 = -s
                    g[:n - s2] += (1.0 - h[s2:]) * dx
                    g[s2:] -= h[:n - s2] * dx
            h -= lr * g
            h = project_feasible(h, n)
            out = np.correlate(h, 1.0 - h, mode='full')
            c = float(out.max() * dx)
            if c < best_c:
                best_c, best_h = c, h.copy()
            if step > 50:
                lr *= 0.95

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
