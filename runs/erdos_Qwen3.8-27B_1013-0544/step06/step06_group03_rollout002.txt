def compute_c5(h, n):

def compute_all_lags(h, n):

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)

        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()

                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, lag):
    """Gradient of f_lag(h) w.r.t. h"""
    grad = np.zeros(n)
    m = abs(lag)
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
    elif m >= n:
        pass

        grad[m:] += (1.0 - h[:n - m])
    else:
        grad[:n - m] += (1.0 - h[m:])
        grad[m:] -= h[:m]
    return grad

    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        if mx > 500:
            return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(shifted))) / alpha)
    bounds = [(0.0, 1.0)] * n
    ones = np.ones(n)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0, 'jac': lambda x: ones}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})

    # Phase 3: high-alpha SLSQP
    if time.time() < deadline - 40:
        h = slsqp_smooth(h, n, 80000, maxiter=2500)
    # Phase 4: local search
    if time.time() < deadline - 30:
        h, _ = local_search_topk(h, n, min(deadline - 20, time.time() + 50), top_k=3)
    # Phase 5: binary refinement
    if time.time() < deadline - 20:
        h_b, c_b = binary_refinement(h, n, min(deadline - 15, time.time() + 30))
        c_curr = compute_c5(h, n)
        if c_b < c_curr:
            h = h_b
    # Phase 6: final polish
    if time.time() < deadline - 10:
        h = slsqp_smooth(h, n, 500000, maxiter=4000)
    # Phase 7: final local search
    if time.time() < deadline - 5:
        h, _ = local_search_topk(h, n, deadline - 2, top_k=5)
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 200
    best_h = None
    best_c = float('inf')
    n_list = [300, 200, 250, 150, 100]
    if n_init not in n_list:
        n_list.append(n_init)
    # Try each resolution
    time_per_n = min(180, max(40, int((deadline - t0 - 30) / len(n_list))))
    for n_try in n_list:
        if time.time() > deadline - 30:
            break
        alloc = min(time_per_n, deadline - time.time() - 30)
        if alloc < 35:
            break
        if best_h is not None:
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_old, best_h)
        elif h_init is not None and n_try == n_init:
            h_start = h_init.copy()
        elif h_init is not None:
            x_old = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n_try)
            h_start = np.interp(x_new, x_old, h_init)
        else:
            h_start = np.full(n_try, 0.5)
        h_start = project_feasible(h_start, n_try)
        h_opt, c_opt = optimize_single(h_start, n_try, time.time() + alloc)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    # Refinement phase on best
    if best_h is not None and time.time() < deadline - 20:
        n = len(best_h)
        remaining = deadline - time.time() - 10
        n_restarts = max(1, int(remaining / 45))
        for r in range(n_restarts):
            if time.time() > deadline - 20:
                break
            scale = 0.01 * (1.0 + 0.5 * r)
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(40, (deadline - time.time() - 20) * 0.5)
            if alloc < 25:
                break
            h_opt, c_opt = optimize_single(hp, n, time.time() + alloc)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    # Final polish
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        h_new = slsqp_smooth(best_h, n, 1000000, maxiter=5000)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        best_h, _ = local_search_topk(best_h, n, deadline - 1, top_k=5)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    if best_h is None:
        best_h = project_feasible(np.full(n_init, 0.5), n_init)
        best_c = compute_c5(best_h, n_init)
    return best_h, best_c, len(best_h)
