```python
def smooth_gradient(out, w, h, n, dx):
    # w: softmax weights, shape (2n-1,)
    # Returns gradient of smoothed objective
    k = np.arange(2*n-1) - (n-1)  # lags: -(n-1), ..., n-1
    grad = np.ones(n) * np.sum(w)  # the "1" term
    for j in range(n):
        # h[j-k] term: valid when 0 <= j-k < n
        idx1 = j - k  # indices into h for h[j-k_m]
        valid1 = (idx1 >= 0) & (idx1 < n)
        grad[j] -= np.sum(w[valid1] * h[idx1[valid1]])
        # h[j+k] term
        idx2 = j + k
        valid2 = (idx2 >= 0) & (idx2 < n)
        grad[j] -= np.sum(w[valid2] * h[idx2[valid2]])
    return dx * dx * grad
```

```python
import time

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_box_sum(h, n):
    """Project h onto {0 <= h <= 1, sum(h) = n/2}"""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Binary search for tau
    lo, hi = -2.0, 2.0
    for _ in range(60):
        tau = (lo + hi) / 2.0
        h_proj = np.clip(h - tau, 0.0, 1.0)
        s = h_proj.sum()
        if s > target:
            lo = tau
        else:
            hi = tau
    h_proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fine adjustment
    diff = target - h_proj.sum()
    if abs(diff) > 1e-10:
        # Distribute among non-saturated entries
        mask = (h_proj > 1e-10) & (h_proj < 1.0 - 1e-10)
        if mask.sum() > 0:
            h_proj[mask] += diff / mask.sum()
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def smoothed_objective(h, n, alpha):
    dx = 2.0 / n
    out = np.correlate(h, 1.0 - h, mode='full') * dx
    m = out.max()
    return m + np.log(np.exp(out - m).sum()) / alpha

def smoothed_gradient(h, n, alpha):
    dx = 2.0 / n
    out = np.correlate(h, 1.0 - h, mode='full') * dx
    m = out.max()
    w = np.exp(out - m)
    w /= w.sum()
    # grad of smoothed max: dx * sum_m w[m] * grad_out[m]
    # where grad_out[m][j] = (1-h[j-k_m]) - h[j+k_m] with k_m = m-(n-1)
    k = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)
    for j in range(n):
        idx1 = j - k
        valid1 = (idx1 >= 0) & (idx1 < n)
        grad[j] = np.sum(w[valid1] * (1.0 - h[idx1[valid1]]))
        idx2 = j + k
        valid2 = (idx2 >= 0) & (idx2 < n)
        grad[j] -= np.sum(w[valid2] * h[idx2[valid2]])
    return dx * grad

def projected_gradient_solve(h, n, alpha, maxiter, deadline, lr_init=0.5):
    """Minimize smoothed max via projected gradient with backtracking."""
    h = h.copy()
    lr = lr_init
    best_h = h.copy()
    best_f = compute_c5(h, n)
    
    for it in range(maxiter):
        if time.time() > deadline:
            break
        f = smoothed_objective(h, n, alpha)
        g = smoothed_gradient(h, n, alpha)
        
        # Backtracking line search
        while lr > 1e-8:
            h_new = project_box_sum(h - lr * g, n)
            f_new = smoothed_objective(h_new, n, alpha)
            if f_new < f - 1e-12:
                break
            lr *= 0.5
        else:
            break
        
        h = h_new
        f_actual = compute_c5(h, n)
        if f_actual < best_f:
            best_f = f_actual
            best_h = h.copy()
        lr = min(lr * 1.05, lr_init)
    
    return best_h, best_f
```

```python
import time

    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_box_sum(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(50):
        tau = (lo + hi) / 2.0
        hp = np.clip(h - tau, 0.0, 1.0)
        if hp.sum() > target:
            lo = tau
        else:
            hi = tau
    hp = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - hp.sum()

        mask = (hp > 1e-12) & (hp < 1.0 - 1e-12)
        if mask.sum() > 0:
            hp[mask] += diff / mask.sum()
            hp = np.clip(hp, 0.0, 1.0)
    return hp

def smoothed_obj(h, n, alpha):
    dx = 2.0 / n
    out = np.correlate(h, 1.0 - h, mode='full') * dx
    m = out.max()
    return m + np.log(np.exp(out - m).sum()) / alpha

def smoothed_grad(h, n, alpha):
    dx = 2.0 / n
    out = np.correlate(h, 1.0 - h, mode='full') * dx
    m = out.max()
    w = np.exp(out - m)
    w /= w.sum()
    k = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)
    for j in range(n):
        idx1 = j - k
        v1 = (idx1 >= 0) & (idx1 < n)
        grad[j] = np.dot(w[v1], 1.0 - h[idx1[v1]])
        idx2 = j + k
        v2 = (idx2 >= 0) & (idx2 < n)
        grad[j] -= np.dot(w[v2], h[idx2[v2]])

    g = np.zeros((n, 2 * n - 1))
    for p in range(n):
        ks = max(0, p)
        ke = min(2 * n - 2, p + n - 1)
        for kk in range(ks, ke + 1):
            g[p, kk] = 1.0 - 2.0 * h[kk - p]
    return g
```
