   Similarly, for all rows p ≠ j, g[p, p+j] += 2
3. Updates h, zeros, ones

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def build_g(h, n):
    g = np.zeros((n, 2 * n - 1))
    for p in range(n):
        ks = max(0, p)
        ke = min(2 * n - 2, p + n - 1)
        for k in range(ks, ke + 1):
            if k != 2 * p:
                g[p, k] = 1.0 - 2.0 * h[k - p]
    return g

def do_swap(g, h, corr, i, j, n, zeros, ones, ii, jj):
    corr = corr + g[i] - g[j]
    p_idx = np.arange(n)
    mi = (p_idx != i) & (p_idx + i < 2 * n - 1)
    g[p_idx[mi], p_idx[mi] + i] -= 2.0
    mj = (p_idx != j) & (p_idx + j < 2 * n - 1)
    g[p_idx[mj], p_idx[mj] + j] += 2.0
    h[i], h[j] = 1.0, 0.0
    zeros = np.delete(zeros, ii)
    ones = np.delete(ones, jj)
    zeros = np.append(zeros, j)
    ones = np.append(ones, i)
    return g, h, corr, zeros, ones

def local_search(h_start, n, deadline, top_k=40, seed=42):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    target = n // 2
    h = np.zeros(n)
    idx = np.argsort(h_start)[::-1]
    h[idx[:target]] = 1.0
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = float(corr.max() * dx)
    best_c = cur_c
    best_h = h.copy()
    g = build_g(h, n)
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    stall = 0
    temp = 0.004
    no_improve = 0
    total_iter = 0

    while time.time() < deadline:
        top_idx = np.argsort(corr)[-top_k:]
        g_act = g[:, top_idx]
        M = corr[top_idx][None, :] + g_act[zeros]
        D = M[:, None, :] - g_act[ones][None, :, :]
        new_maxes = D.max(axis=2)
        flat = int(new_maxes.argmin())
        ii = flat // len(ones)
        jj = flat % len(ones)
        best_val = float(new_maxes[ii, jj])
        i_pos = zeros[ii]
        j_pos = ones[jj]
        new_c = best_val * dx

        if new_c < cur_c - 1e-14:
            g, h, corr, zeros, ones = do_swap(g, h, corr, i_pos, j_pos, n, zeros, ones, ii, jj)
            cur_c = float(corr.max() * dx)
            stall = 0
            temp = max(0.001, temp * 0.9998)
            if cur_c < best_c - 1e-14:
                best_c = cur_c
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            delta = new_c - cur_c
            if delta < 0 or np.random.rand() < np.exp(-delta / max(temp, 0.001)):
                g, h, corr, zeros, ones = do_swap(g, h, corr, i_pos, j_pos, n, zeros, ones, ii, jj)
                cur_c = float(corr.max() * dx)
                stall = 0
                no_improve += 1
                temp = max(0.001, temp * 0.999)
            else:
                stall += 1
                no_improve += 1

            nk = min(rng.randint(2, 6), len(zeros), len(ones))
            ri = rng.choice(len(zeros), nk, replace=False)

            swap_z = zeros[ri].copy()
            swap_o = ones[rj].copy()

                new_corr += g[swap_z[a]]
                new_corr -= g[swap_o[a]]
            new_c_k = float(new_corr.max() * dx)

                    if len(ii2) > 0 and len(jj2) > 0:

                corr = np.correlate(h, 1 - h, mode='full').astype(float)
                g = build_g(h, n)
                cur_c = float(corr.max() * dx)
                ones = np.where(h == 1)[0].copy()
                zeros = np.where(h == 0)[0].copy()
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                    no_improve = 0
            temp = max(0.002, temp * 0.88)
            stall = 0

        if no_improve > 600:

            if sz > 0:

def thue_morse_init(n, rng):
    t = np.zeros(n, dtype=int)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    target = n // 2
    num_ones = int(t.sum())
    if num_ones > target:
        ones_pos = np.where(t == 1)[0]
        rng.shuffle(ones_pos)
        t[ones_pos[:num_ones - target]] = 0
    elif num_ones < target:
        zeros_pos = np.where(t == 0)[0]
        rng.shuffle(zeros_pos)
        t[zeros_pos[:target - num_ones]] = 1
    return t.astype(float)

def slsqp_polish(h_bin, n, deadline, alpha=3000000):
    dx = 2.0 / n
    x0 = h_bin.astype(float)

    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        sm = sh.max()
        lse = (sm + np.log(np.exp(sh - sm).sum())) / alpha
        return m + lse

    def sum_fun(x):
        return x.sum() - n / 2.0

    try:
        res = minimize(obj, x0, method='SLSQP', bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': sum_fun}],
                       options={'maxiter': 2000, 'ftol': 1e-14})
        x_new = np.clip(res.x, 0.0, 1.0)
        diff = n / 2.0 - x_new.sum()
        if abs(diff) > 1e-12:
            room = 1.0 - x_new
            s = room.sum()
            if s > 1e-14:
                x_new = x_new + (diff / s) * room
        x_new = np.clip(x_new, 0.0, 1.0)
        return x_new
    except Exception:
        return x0
