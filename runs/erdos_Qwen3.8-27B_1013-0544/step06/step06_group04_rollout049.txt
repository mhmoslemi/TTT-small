import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def get_corr(h, n):
    return np.correlate(h, 1 - h, mode='full')

def project_to_sum(h, n):
    """Project h to have sum = n/2, maintaining [0,1] bounds."""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def compute_swap_delta(h, i, j, n):
    """Compute change in correlation vector when h[i] goes 0->1 and h[j] goes 1->0."""
    # Binary swap
    delta = np.zeros(2 * n - 1)
    # Effect of h[i]: 0 -> 1
    # c[n-1+ell] = sum_n h[n+ell]*(1-h[n])
    # h[i] appears as h[n+ell] when n = i-ell, contributing (1-h[i-ell])
    # h[i] appears as h[n] when n = i, contributing -h[i+ell] (since 1-h[i] changes)
    for ell in range(n):
        # h[i] as first position (n+ell = i, n = i-ell)
        if 0 <= i - ell:
            delta[n-1+ell] += 1.0 - h[i-ell]
        # h[i] as second position (n = i)
        if i + ell < n:
            delta[n-1+ell] -= h[i+ell]
    # Effect of h[j]: 1 -> 0 (negative of above)
    for ell in range(n):
        if 0 <= j - ell:
            delta[n-1+ell] -= 1.0 - h[j-ell]
        if j + ell < n:
            delta[n-1+ell] += h[j+ell]

```python
def compute_swap_delta(h, i, j, n):
    delta = np.zeros(2 * n - 1)
    # For positive lags ell = 0..n-1, index in corr is n-1+ell
    # Effect of i (0->1):
    #   As first pos: corr[n-1+ell] += (1-h[i-ell]) for ell <= i
    #   As second pos: corr[n-1+ell] -= h[i+ell] for ell < n-i
    # Effect of j (1->0):
    #   corr[n-1+ell] -= (1-h[j-ell]) for ell <= j
    #   corr[n-1+ell] += h[j+ell] for ell < n-j
    
    ell = np.arange(n)
    idx = n - 1 + ell
    
    # i as first position: valid ell where i-ell >= 0, i.e., ell <= i
    mask1 = ell <= i
    delta[idx[mask1]] += 1.0 - h[i - ell[mask1]]
    
    # i as second position: valid ell where i+ell < n, i.e., ell < n-i
    mask2 = ell < n - i
    delta[idx[mask2]] -= h[i + ell[mask2]]
    
    # j as first position: negative
    mask3 = ell <= j
    delta[idx[mask3]] -= 1.0 - h[j - ell[mask3]]
    
    # j as second position: positive
    mask4 = ell < n - j
    delta[idx[mask4]] += h[j + ell[mask4]]
    
    return delta
```

This is O(n) with vectorized operations. Good.
