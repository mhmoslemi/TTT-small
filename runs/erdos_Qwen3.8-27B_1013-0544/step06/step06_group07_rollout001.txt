```
def smooth_function(h,n, alpha=100.0, tau=None):
    dx=2.0/n
    c = np.correlate(h, 1.0-h, 'full')*dx
    cmax = float(c.max())
    if tau is not None:
       mask = c >= cmax - tau
       if not mask.any(): mask = c >= cmax
       w = np.zeros_like(c)
       w[mask]=1.0
    else:
       z = alpha*(c-cmax)
       # for numerical: if alpha large, z max 0
       ew = np.exp(z)
       w = ew / ew.sum()
    C = n-1
    a = np.convolve(1.0-h, w, mode='full')
    b = np.convolve(h, w[::-1], mode='full')
    grad = (a[C:C+n] - b[C:C+n]) * dx
    return cmax, grad, c
```

```
def project_box_sum(v,n):
```

```python
import numpy as np
import time
from scipy.optimize import minimize


def evaluate_c5(h, n):
    dx = 2.0 / float(n)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))


def project_box_sum(v, n):
    v = np.asarray(v, dtype=float)
    target = 0.5 * float(n)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(55):
        lam = 0.5 * (lo + hi)
        p = np.clip(v - lam, 0.0, 1.0)
        if p.sum() > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)


def smooth_objective(x, n, alpha):
    dx = 2.0 / float(n)
    c = np.correlate(x, 1.0 - x, mode="full") * dx
    cmax = float(np.max(c))
    z = alpha * (c - cmax)
    ew = np.exp(z)
    s = float(ew.sum())
    if not np.isfinite(s) or s <= 0.0:
        return cmax
    return cmax + np.log(s) / alpha


def smooth_gradient(x, n, alpha):
    dx = 2.0 / float(n)
    c = np.correlate(x, 1.0 - x, mode="full") * dx
    cmax = float(np.max(c))
    z = alpha * (c - cmax)
    ew = np.exp(z)
    s = float(ew.sum())
    if not np.isfinite(s) or s <= 0.0:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = ew / s
    C = n - 1
    a = np.convolve(1.0 - x, w, mode="full")
    b = np.convolve(x, w[::-1], mode="full")
    return (a[C:C + n] - b[C:C + n]) * dx


def smooth_function(h, n, alpha, tau=None):
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    cmax = float(np.max(c))
    if tau is None:
        z = alpha * (c - cmax)
        ew = np.exp(z)
        s = float(ew.sum())
        if not np.isfinite(s) or s <= 0.0:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w = ew / s
        F = cmax + np.log(s) / alpha
    else:
        mask = c >= (cmax - float(tau))
        if not np.any(mask):
            mask = c >= cmax
        w = np.zeros_like(c)
        w[mask] = 1.0
        w /= float(w.sum())
        F = cmax
    C = n - 1
    a = np.convolve(1.0 - h, w, mode="full")
    b = np.convolve(h, w[::-1], mode="full")
    grad = (a[C:C + n] - b[C:C + n]) * dx
    return F, cmax, grad, c


def sum_con(x, n):
    return x.sum() - 0.5 * float(n)


def sum_con_jac(x, n):
    return np.ones(len(x), dtype=float)


def slsqp_smooth_polish(h, n, alpha, maxiter):
    h = project_box_sum(h, n)
    c0 = evaluate_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(
            smooth_objective, h, method='SLSQP',
            jac=smooth_gradient, args=(n, alpha),
            bounds=bounds, constraints=cons,
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        h2 = project_box_sum(res.x, n)
    except Exception:
        h2 = h
    c2 = evaluate_c5(h2, n)
    if c2 < c0:
        return h2, c2
    return h, c0


def lag_value_grad(h, n, k):
    C = n - 1
    m = C - int(k)
    L = n - abs(m)
    dx = 2.0 / float(n)
    g = np.zeros(n, dtype=float)
    if m >= 0:
        base = h[:L]
        shift = h[m:m + L]
        f = float(np.dot(base, 1.0 - shift) * dx)
        g[:L] += 1.0 - shift
        g[m:m + L] -= base
    else:
        r = -m
        base = h[r:r + L]
        shift = h[:L]
        f = float(np.dot(base, 1.0 - shift) * dx)
        g[r:r + L] += 1.0 - shift
        g[:L] -= base
    return f, g * dx


def active_obj(y, n, alpha, lam):
    h = y[:-1]
    t = y[-1]
    return t + lam * smooth_objective(h, n, alpha)


def active_obj_jac(y, n, alpha, lam):
    h = y[:-1]
    out = np.zeros(len(y), dtype=float)
    out[:n] = lam * smooth_gradient(h, n, alpha)
    out[-1] = 1.0
    return out


def active_con_fun(y, n, indices):
    h = y[:-1]
    t = y[-1]
    out = np.empty(len(indices), dtype=float)
    for ii, k in enumerate(indices):
        f, _ = lag_value_grad(h, n, k)
        out[ii] = t - f
    return out


def active_con_jac(y, n, indices):
    h = y[:-1]
    K = len(indices)
    J = np.zeros((K, n + 1), dtype=float)
    for ii, k in enumerate(indices):
        _, g = lag_value_grad(h, n, k)
        J[ii, :n] = -g
        J[ii, n] = 1.0
    return J


def sum_y_con(y, n):
    return y[:n].sum() - 0.5 * float(n)


def sum_y_jac(y, n):
    out = np.zeros(len(y), dtype=float)

def active_slsqp_polish(h, n, kmax=10, max_passes=3, maxiter=250):
    h = project_box_sum(h, n)
    best = evaluate_c5(h, n)
    best_h = h.copy()
    alpha = 500.0
    lam = 0.05
    for _pass in range(max_passes):
        c = np.correlate(h, 1.0 - h, mode="full") * (2.0 / float(n))
        cmax = float(np.max(c))
        idx = np.argsort(c)[::-1][:min(kmax, len(c))]
        y0 = np.empty(n + 1, dtype=float)
        y0[:n] = h
        y0[n] = cmax + 1e-5
        bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
        cons = [
            {'type': 'eq', 'fun': sum_y_con, 'jac': sum_y_jac, 'args': (n,)},
            {'type': 'ineq', 'fun': active_con_fun, 'jac': active_con_jac, 'args': (n, idx)}
        ]
        try:
            res = minimize(
                active_obj, y0, method='SLSQP',
                jac=active_obj_jac, args=(n, alpha, lam),
                bounds=bounds, constraints=cons,

    step = step0
    vel = np.zeros(n, dtype=float)
    stall = 0
    tol = 1e-13

    while time.time() < deadline:
        F, cmax, grad, _c = smooth_function(h, n, alpha, None)
        if cmax < best - tol:
            best = cmax
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15 or stall > patience_int:
            h = project_box_sum(h + kick * rng.randn(n), n)
            cur = evaluate_c5(h, n)
            if cur < best - tol:
                best = cur
                best_h = h.copy()
            step = step0
            vel = np.zeros(n, dtype=float)
            stall = 0
            continue

        if momentum > 0.0:
            vel = momentum * vel + grad
            vnorm = float(np.linalg.norm(vel))
            if vnorm < 1e-15:
                direction = grad / gnorm
            else:
                direction = vel / vnorm
        else:
            direction = grad / gnorm

        accepted = False
        local_step = step
        for _ in range(8):
            cand = project_box_sum(h - local_step * direction, n)
            cnew = evaluate_c5(cand, n)
            if cnew < cur - 1e-13:
                accepted = True
                h2 = cand
                break
            local_step *= 0.5
            if local_step < 1e-9:
                break

        if accepted:
            h = h2
            cur = cnew
            if cur < best - tol:
                best = cur
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
            step = min(1.0, local_step * 1.05)
        else:
            h = project_box_sum(h + 0.01 * kick * rng.randn(n), n)
            cur = evaluate_c5(h, n)
            if cur < best - tol:
                best = cur
                best_h = h.copy()
            step = step0
            vel = np.zeros(n, dtype=float)
            stall = 0

    return best_h, best


def resample_h(h, n_old, n_new):
    dx_old = 2.0 / float(n_old)
    x_old = (np.arange(n_old) + 0.5) * dx_old
    dx_new = 2.0 / float(n_new)
    x_new = (np.arange(n_new) + 0.5) * dx_new
    hh = np.interp(x_new, x_old, h)
    return project_box_sum(hh, n_new)


def generate_alternate_starts(n, seed, count):
    rng = np.random.RandomState(seed)
    out = []
    for _ in range(count):
        out.append(project_box_sum(0.5 + 0.15 * rng.randn(n), n))

        a = 0.3 + 0.4 * rng.rand()
        b = rng.rand()
        vals = np.mod(np.arange(n) * a + b, 1.0)
        out.append(project_box_sum(vals, n))

        f = 0.5 + 3.0 * rng.rand()
        ph = 2.0 * np.pi * rng.rand()
        x = np.linspace(0.0, 2.0, n)
        vals = 0.5 + 0.45 * np.sin(np.pi * f * x + ph)
        out.append(project_box_sum(vals, n))
    return out


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    budget_s = max(60.0, float(budget_s))
    deadline = t0 + (budget_s - 18.0)

    try:
        h_in = np.array(initial_h_values, dtype=float).copy()
        n = int(len(h_in))
    except Exception:
        n = 250
        h_in = np.full(n, 0.5, dtype=float)

    h_in = np.clip(h_in, 0.0, 1.0)
    h = project_box_sum(h_in, n)
    best_h = h.copy()
    best_c = evaluate_c5(h, n)
    best_n = n

    if time.time() < deadline - 100.0:
        end = min(deadline - 100.0, t0 + 0.07 * (budget_s - 18.0))
        hb, cb = projected_gradient(best_h, best_n, end, alpha=50.0,
                                    step0=0.03, momentum=0.3, seed=seed,
                                    kick=0.02, patience_int=5000)
        if cb < best_c:
            best_h, best_c = hb, cb

    for alpha in (200.0, 2000.0):
        if time.time() < deadline - 80.0:
            hb, cb = slsqp_smooth_polish(best_h, best_n, alpha, 500)
            if cb < best_c:
                best_h, best_c = hb, cb

    if time.time() < deadline - 120.0:
        end = min(deadline - 120.0, t0 + 0.32 * (budget_s - 18.0))
        hb, cb = projected_gradient(best_h, best_n, end, alpha=3000.0,
                                    step0=0.02, momentum=0.4, seed=seed + 17,
                                    kick=0.012, patience_int=8000)
        if cb < best_c:
            best_h, best_c = hb, cb

    starts = generate_alternate_starts(best_n, seed + 101, 2)
    for idx, hs in enumerate(starts):
        if time.time() > deadline - 90.0:
            break
        hs = project_box_sum(hs, best_n)
        cs = evaluate_c5(hs, best_n)
        if cs > best_c + 0.06:
            continue
        end = min(deadline - 90.0, time.time() + 45.0)
        hb, cb = projected_gradient(hs, best_n, end, alpha=500.0,
                                    step0=0.025, momentum=0.3,
                                    seed=seed + 300 + idx,
                                    kick=0.03, patience_int=4000)
        if cb < best_c:
            best_h, best_c = hb, cb

    if time.time() < deadline - 140.0:
        hb, cb = active_slsqp_polish(best_h, best_n, kmax=10, max_passes=2, maxiter=250)
        if cb < best_c:
            best_h, best_c = hb, cb

    if time.time() < deadline - 100.0:
        hb, cb = active_slsqp_polish(best_h, best_n, kmax=6, max_passes=2, maxiter=200)
        if cb < best_c:
            best_h, best_c = hb, cb

    if time.time() < deadline - 70.0:
        hb, cb = slsqp_smooth_polish(best_h, best_n, 10000.0, 400)
        if cb < best_c:
            best_h, best_c = hb, cb

    for n2 in (200, 300):
        if n2 == best_n or time.time() > deadline - 70.0:
            continue
        h2 = resample_h(best_h, best_n, n2)
        h2, c2 = slsqp_smooth_polish(h2, n2, 3000.0, 300)
        if c2 < best_c:
            best_h, best_c, best_n = h2, c2, n2
            if time.time() < deadline - 60.0:
                h3, c3 = active_slsqp_polish(best_h, best_n, kmax=8, max_passes=1, maxiter=200)
                if c3 < best_c:
                    best_h, best_c = h3, c3
            break

    best_h = np.asarray(best_h, dtype=float)
    if not np.all(np.isfinite(best_h)):
        best_h = np.full(best_n, 0.5, dtype=float)
    best_h = project_box_sum(best_h, best_n)
    best_c = evaluate_c5(best_h, best_n)
    return best_h, best_c, best_n
```
