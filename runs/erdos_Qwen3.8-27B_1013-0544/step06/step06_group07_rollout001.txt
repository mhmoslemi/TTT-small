def project_feasible(h,n):
    h=np.array(h,dtype=float).copy()
    h=np.clip(h,0.0,1.0)
    target=float(n)/2.0
    for _ in range(200):
        diff=target-h.sum()
        if abs(diff)<1e-13: break
        if diff>0:
            room=1.0-h
            s=room.sum()
            if s<=1e-15: break
            h += (diff/s)*room
        else:
            s=h.sum()
            if s<=1e-15: break
            h += (diff/s)*h
    h=np.clip(h,0.0,1.0)
    # final exact-ish
    diff=target-h.sum()
    if abs(diff)>1e-14:
       if diff>0:
          order=np.argsort(1.0-h)[::-1]
       else:
          order=np.argsort(h)[::-1]
       d=diff
       for idx in order:
          if d>0:
             add=min(1.0-h[idx], d)
             h[idx]+=add; d-=add
          else:
             sub=min(h[idx], -d)
             h[idx]-=sub; d+=sub
          if abs(d)<1e-15: break
    return h

import numpy as np
import time
from scipy.optimize import minimize

_GN = 0
_GA = 1.0

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full').astype(float)

def c5_val(h, n):
    return float(np.max(corr_full(h)) * (2.0 / n))

def project(h, n):
    h = np.asarray(h, dtype=float).copy()
    if h.size != n:
        h = np.resize(h, n)
    h[~np.isfinite(h)] = 0.5
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    for _ in range(100):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        if d > 0:
            r = 1.0 - h
            s = r.sum()
            if s < 1e-15:
                break
            h += (d / s) * r
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += (d / s) * h
    h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-14:
        idx = np.argsort(1.0 - h if d > 0 else h)[::-1]
        for i in idx:
            if d > 0:
                a = min(1.0 - h[i], d)
                h[i] += a
                d -= a
            else:
                a = min(h[i], -d)
                h[i] -= a
                d += a
            if abs(d) < 1e-15:
                break
    return h

def rbin(h, n):
    out = np.zeros(n, dtype=float)
    k = n // 2
    if k > 0:
        idx = np.argsort(h)[::-1][:k]
        out[idx] = 1.0
    return out

def resample_to_n(h, n, m):
    if m == n:
        return project(h, n)
    xo = np.linspace(0.0, 2.0, n)
    xm = np.linspace(0.0, 2.0, m)
    return project(np.interp(xm, xo, h), m)

def random_binary(n, seed):
    rng = np.random.RandomState(seed)
    x = rng.rand(n)
    out = np.zeros(n, dtype=float)
    idx = np.argsort(x)[::-1][:n // 2]
    out[idx] = 1.0
    return out

def deriv_one(hrev, omh, n, p):
    d = np.zeros(2 * n - 1, dtype=float)
    d[p:p + n] -= hrev
    d[n - 1 - p:2 * n - 1 - p] += omh[:n - p]
    return d

def eval_pair(g, lv, p0, p1, p2, ds):
    d2 = ds * ds
    b = g[None, :] + ds[:, None] * lv
    b[:, p0] -= 2.0 * d2
    b[:, p1] += d2
    b[:, p2] += d2
    return b.max(axis=1)

def grid_pair_min(g, lv, p0, p1, p2, lo, hi):
    if hi - lo <= 1e-13:
        return 0.0, float(np.max(g))
    m = 33 if hi - lo > 0.3 else 17
    ds = np.linspace(lo, hi, m)
    v = eval_pair(g, lv, p0, p1, p2, ds)
    i = int(np.argmin(v))
    if m > 2:
        if i == 0:
            a, b = lo, ds[1]
        elif i == m - 1:
            a, b = ds[i - 1], hi
        else:
            a, b = ds[i - 1], ds[i + 1]
        if b - a > 1e-12:
            ds2 = np.linspace(a, b, 11)
            v2 = eval_pair(g, lv, p0, p1, p2, ds2)
            j = int(np.argmin(v2))
            return float(ds2[j]), float(v2[j])
    return float(ds[i]), float(v[i])

def pair_cd(h, n, g, c5, deadline, seed=42):
    h = np.array(h, dtype=float).copy()
    g = np.array(g, dtype=float).copy()
    cur_raw = float(g.max())
    c5 = cur_raw * (2.0 / n)
    rng = np.random.RandomState(seed)
    dead_sweeps = 0
    last_imp = time.time()
    updates = 0
    while time.time() < deadline:
        perm = rng.permutation(n)
        improved = 0
        for s in range(0, n - 1, 2):
            if time.time() >= deadline:
                break
            a = int(perm[s])
            b = int(perm[s + 1])
            xa = float(h[a])
            xb = float(h[b])
            lo = max(-xa, xb - 1.0)
            hi = min(1.0 - xa, xb)
            if hi - lo < 1e-12:
                continue
            hrev = h[::-1]
            omh = 1.0 - h
            lv = deriv_one(hrev, omh, n, a) - deriv_one(hrev, omh, n, b)
            p0 = n - 1
            d = b - a
            p1 = p0 + d
            p2 = p0 - d
            delta, best_raw = grid_pair_min(g, lv, p0, p1, p2, lo, hi)
            if best_raw < cur_raw - 1e-8:
                delta = float(min(max(delta, lo), hi))
                h[a] = xa + delta
                h[b] = xb - delta
                d2 = delta * delta
                g = g + delta * lv
                g[p0] -= 2.0 * d2
                g[p1] += d2
                g[p2] += d2
                cur_raw = float(g.max())
                c5 = cur_raw * (2.0 / n)
                improved += 1
                updates += 1
                if updates % 500 == 0:
                    g = corr_full(h)
                    cur_raw = float(g.max())
                    c5 = cur_raw * (2.0 / n)
        if improved == 0:
            dead_sweeps += 1
            if dead_sweeps >= 25 and time.time() - last_imp > 15:
                break
        else:
            dead_sweeps = 0
            last_imp = time.time()
    h = project(h, n)
    g = corr_full(h)
    return h, g, float(g.max() * (2.0 / n))

def smooth_obj(x):
    g = corr_full(x) * (2.0 / _GN)
    m = float(np.max(g))
    a = float(_GA)
    sh = a * (g - m)
    mx = float(np.max(sh))
    return m + (mx + float(np.log(np.exp(sh - mx).sum()))) / a

def smooth_eq(x):
    return float(np.sum(x)) - 0.5 * _GN

def slsqp_polish(h, n, deadline, seed=42):
    global _GN, _GA
    best = np.array(h, dtype=float).copy()
    bc = c5_val(best, n)
    bounds = [(0.0, 1.0)] * n
    alphas = [1000, 4000, 15000]
    for it, al in enumerate(alphas):
        if time.time() > deadline - 6:
            break
        _GN = n
        _GA = float(al)
        x = project(best, n)
        try:
            res = minimize(smooth_obj, x, method='SLSQP', bounds=bounds,
                           constraints=[dict(type='eq', fun=smooth_eq)],
                           options=dict(maxiter=60 + 35 * it, ftol=1e-11))
            rx = res.x
            if rx is None or not np.all(np.isfinite(rx)):
                continue
            cand = project(rx, n)
            val = c5_val(cand, n)
            if val < bc:
                best = cand
                bc = val
        except Exception:
            pass
    return best, bc

def deriv_mat(hrev, omh, n, idx):
    d = np.zeros((len(idx), 2 * n - 1), dtype=float)
    for r, p in enumerate(idx):
        d[r, p:p + n] -= hrev
        d[r, n - 1 - p:2 * n - 1 - p] += omh[:n - p]
    return d

def best_swap_bin(h, n, g):
    if 2 * n - 1 < 4:
        return None
    z = np.nonzero(h == 0.0)[0]
    o = np.nonzero(h == 1.0)[0]
    if len(z) == 0 or len(o) == 0:
        return None
    cur = float(g.max())
    p0 = n - 1
    hrev = h[::-1]
    omh = 1.0 - h
    A = deriv_mat(hrev, omh, n, z)
    B = deriv_mat(hrev, omh, n, o)
    best = cur + 1.0
    bi = -1
    bj = -1
    bg = None
    rows = np.arange(len(o))
    for zi, zv in enumerate(z):
        V = g[None, :] + A[zi][None, :] - B
        part = np.argpartition(-V, 3, axis=1)
        ti = part[:, :4]
        tv = np.take_along_axis(V, ti, axis=1)
        ordi = np.argsort(-tv, axis=1)
        ti = np.take_along_axis(ti, ordi, axis=1)
        tv = np.take_along_axis(tv, ordi, axis=1)
        d = o - zv
        p1 = p0 + d
        p2 = p0 - d
        valid = (ti != p0) & (ti != p1[:, None]) & (ti != p2[:, None])
        fv = np.argmax(valid, axis=1)
        ex = tv[rows, fv]
        a0 = V[:, p0] - 2.0
        a1 = np.take_along_axis(V, p1[:, None], axis=1)[:, 0] + 1.0
        a2 = np.take_along_axis(V, p2[:, None], axis=1)[:, 0] + 1.0
        sc = np.maximum(np.maximum(np.maximum(ex, a0), a1), a2)
        li = int(np.argmin(sc))
        if sc[li] < best - 1e-9:
            best = float(sc[li])
            bi = int(zv)
            bj = int(o[li])
            bg = V[li].copy()
    if bi < 0:
        return None
    ng = bg.copy()
    d = bj - bi
    ng[p0] -= 2.0
    ng[p0 + d] += 1.0
    ng[p0 - d] += 1.0
    return bi, bj, ng, best

def local_bin(h, g, cur_raw, n, deadline):
    bh = h.copy()
    bg = g.copy()
    br = cur_raw
    while time.time() < deadline:
        res = best_swap_bin(h, n, g)
        if res is None:
            break
        i, j, ng, nr = res
        if nr >= cur_raw - 1e-8:
            break
        h[i] = 1.0
        h[j] = 0.0
        g = ng
        cur_raw = nr
        if cur_raw < br - 1e-10:
            br = cur_raw
            bh = h.copy()
            bg = g.copy()
    return bh, bg, br

def random_swaps(h, n, k, rng):
    h = h.copy()
    z = np.nonzero(h == 0.0)[0]
    o = np.nonzero(h == 1.0)[0]
    for _ in range(k):
        if len(z) == 0 or len(o) == 0:
            break
        zi = int(rng.randint(len(z)))
        oj = int(rng.randint(len(o)))
        i = int(z[zi])
        j = int(o[oj])
        h[i] = 1.0
        h[j] = 0.0
        z[zi] = j
        o[oj] = i
    return h

def binary_ils(h_start, n, deadline, seed=42):

    g = corr_full(h)

        h = random_swaps(src, n, k, rng)
