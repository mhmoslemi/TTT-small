def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14: break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14: break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def solve_1d_minimax(a_coeffs, d_vals, b_quadratic, dx):
    """Solve min_{x in [0,1]} max(max_k(a_k*x + d_k), b*x^2 + (a_0)*x + d_0)
    where b = -dx (the quadratic coefficient for k=0)
    Actually let me separate: the quadratic part is -dx*x^2, and the linear part includes a_0*x.
    So we have: max(L(x), Q(x)) where:
    L(x) = max_k (a_k * x + d_k) for k != 0
    Q(x) = -dx * x^2 + a_0 * x + d_0
    
    Parameters:
    - a_coeffs: array of a_k for k != 0 (linear coefficients)
    - d_vals: array of d_k for k != 0 (constant terms)
    - a_0: linear coefficient for k=0
    - d_0: constant term for k=0
    - b_quadratic: the quadratic coefficient (= -dx)
    """
    # Bisection on t
    t_lo = 0.0
    t_hi = 1.0  # upper bound on any overlap value
    
    # Tighten bounds
    # L(x) on [0,1]: min is at least min(d_vals) and at most max(d_vals) or max(a_k+d_k)
    # Q(x) on [0,1]: max is at most max(0, d_0, -dx*0.25 + a_0*0.5 + d_0)
    
    for _ in range(60):
        t_mid = (t_lo + t_hi) / 2.0
        
        # Check feasibility: exists x in [0,1] with L(x) <= t_mid and Q(x) <= t_mid
        # L(x) <= t_mid:
        # For each k: a_k * x + d_k <= t_mid
        # If a_k > 0: x <= (t_mid - d_k) / a_k
        # If a_k < 0: x >= (t_mid - d_k) / a_k
        # If a_k == 0: need d_k <= t_mid
        
        lo = 0.0
        hi = 1.0
        
        if len(a_coeffs) > 0:
            pos = a_coeffs > 1e-15
            neg = a_coeffs < -1e-15
            zero = ~(pos | neg)
            
            if np.any(zero) and np.any(d_vals[zero] > t_mid + 1e-15):
                t_lo = t_mid
                continue
            
            if np.any(pos):
                hi = min(hi, np.min((t_mid - d_vals[pos]) / a_coeffs[pos]))
            if np.any(neg):
                lo = max(lo, np.max((t_mid - d_vals[neg]) / a_coeffs[neg]))
        
        if lo > hi + 1e-15:
            t_lo = t_mid
            continue
        
        # Now check Q(x) = b*x^2 + a_0*x + d_0 <= t_mid (i.e., Q(x) - t_mid <= 0)
        # b*x^2 + a_0*x + (d_0 - t_mid) <= 0
        # b = -dx < 0, so this is a downward parabola
        # Q(x) - t_mid <= 0 means x is between the two roots (if they exist)
        
        b = b_quadratic  # = -dx
        c = d_0 - t_mid
        # b*x^2 + a_0*x + c <= 0
        # discriminant = a_0^2 - 4*b*c
        disc = a_0 * a_0 - 4.0 * b * c
        if disc < 0:
            # No real roots. Since b < 0, the parabola opens downward.
            # b*x^2 + a_0*x + c <= 0 is satisfied everywhere (since max of downward parabola is finite and... 
            # wait, if disc < 0 and b < 0, the parabola is always negative? No.
            # b < 0 means it opens downward. disc < 0 means no real roots.
            # If b < 0 and disc < 0: the quadratic is always negative (since it opens downward and never crosses zero)
            # Wait: b < 0, disc = a_0² - 4bc < 0 means a_0² < 4bc. Since b < 0, we need c > 0 and 4bc > a_0².
            # Hmm, let me just evaluate: if disc < 0 and b < 0, the quadratic b*x²+a_0*x+c is always < 0
            # (since it opens downward and has no real roots, it's always on one side of zero; 
            #  evaluating at x=0 gives c, and if c < 0 it's always negative... this is getting confusing)
            # Let me just check: the max of b*x²+a_0*x+c for x in [0,1] with b<0
            # is at x = clamp(-a_0/(2b), 0, 1) [vertex of the parabola]
            # If the max <= 0, constraint is always satisfied.
            vertex = -a_0 / (2.0 * b) if b != 0 else 0
            vertex = np.clip(vertex, 0, 1)
            q_max = b * vertex * vertex + a_0 * vertex + c
            if q_max <= 0:
                # Q(x) - t_mid <= 0 for all x, so just need [lo,hi] non-empty (already checked)
                t_hi = t_mid
                continue
            else:
                # Q(x) - t_mid > 0 at vertex, so we need x outside some interval
                # But this case shouldn't happen if disc < 0... let me just handle it generally
                t_lo = t_mid
                continue
        else:
            # Two real roots
            sqrt_disc = np.sqrt(disc)
            x1 = (-a_0 - sqrt_disc) / (2.0 * b)
            x2 = (-a_0 + sqrt_disc) / (2.0 * b)
            if x1 > x2:
                x1, x2 = x2, x1
            # b < 0, so b*x²+a_0*x+c <= 0 when x is OUTSIDE [x1, x2]... 
            # Wait no. b < 0 means parabola opens downward.
            # b*x²+a_0*x+c <= 0 is satisfied when x <= x1 OR x >= x2 (outside the roots)
            # Wait, that's for b > 0. For b < 0:
            # The parabola is >= 0 outside the roots and <= 0 between the roots.
            # So b*x²+a_0*x+c <= 0 when x1 <= x <= x2.
            # Let me verify: at x = (x1+x2)/2 = -a_0/(2b), the value is c - a_0²/(4b)
            # = (d_0-t_mid) - a_0²/(-4dx) = (d_0-t_mid) + a_0²/(4dx)
            # This should be >= 0 (since it's the max of the downward parabola)... 
            # Hmm I'm getting confused. Let me just test with a concrete example.
            # b = -1, a_0 = 0, c = -1: -x² - 1 <= 0 → x² >= -1, always true. disc = 0-4(-1)(-1) = -4 < 0.
            # So disc < 0 case: always satisfied. ✓
            # b = -1, a_0 = 0, c = 1: -x² + 1 <= 0 → x² >= 1 → |x| >= 1. disc = 0+4 = 4 > 0.
            # Roots: x = ±1. Between -1 and 1, the expression is > 0. Outside, < 0. ✓
            # So for b < 0: expression <= 0 OUTSIDE [x1, x2].
            # Wait that contradicts what I said. Let me re-derive.
            # b*x² + a_0*x + c with b < 0.
            # The quadratic equals 0 at x1 and x2 (x1 < x2).
            # For x < x1: b*(x-x1)*(x-x2) with b<0, (x-x1)<0, (x-x2)<0, so product is b*neg*neg = b*pos < 0. ✓
            # For x1 < x < x2: b*(x-x1)*(x-x2) with b<0, (x-x1)>0, (x-x2)<0, so product is b*pos*neg = b*neg > 0.
            # For x > x2: b*(x-x1)*(x-x2) with b<0, (x-x1)>0, (x-x2)>0, so product is b*pos*pos = b*pos < 0. ✓
            # So the expression <= 0 when x <= x1 or x >= x2 (OUTSIDE [x1,x2]).
            
            # So Q(x) - t_mid <= 0 when x <= x1 or x >= x2.
            # Combined with [lo, hi]: need [lo, min(hi,x1)] non-empty OR [max(lo,x2), hi] non-empty
            
            feasible = False
            if hi >= x1 and lo <= x1:
                # [lo, hi] overlaps with (-inf, x1]
                feasible = True
            if hi >= x2 and lo <= hi:
                # [lo, hi] overlaps with [x2, inf)
                feasible = True
            # More precisely:
            # Option 1: [lo, min(hi, x1)] non-empty → lo <= min(hi, x1)
            if lo <= min(hi, x1):
                feasible = True
            # Option 2: [max(lo, x2), hi] non-empty → max(lo, x2) <= hi
            if max(lo, x2) <= hi:
                feasible = True
            
            if feasible:
                t_hi = t_mid
            else:
                t_lo = t_mid
    
    # After bisection, t_hi is approximately the optimal value
    # Now find the optimal x
    t_opt = t_hi
    # The optimal x is where L(x) = Q(x) = t_opt (if they cross in [0,1])
    # Or at a boundary, or at a kink of L.
    # For now, just find x that minimizes max(L(x), Q(x)) by scanning a fine grid
    # Actually, let's use the feasibility structure:
    # At t_opt, the feasible set is [lo, hi] ∩ (outside [x1,x2]) ∩ [0,1]
    # The optimal x is any point in this set.
    # Let's find it.
    
    # Recompute at t_opt
    lo = 0.0
    hi = 1.0
    if len(a_coeffs) > 0:
        pos = a_coeffs > 1e-15
        neg = a_coeffs < -1e-15
        if np.any(pos):
            hi = min(hi, np.min((t_opt - d_vals[pos]) / a_coeffs[pos]))
        if np.any(neg):
            lo = max(lo, np.max((t_opt - d_vals[neg]) / a_coeffs[neg]))
    
    b = b_quadratic
    c = d_0 - t_opt
    disc = a_0 * a_0 - 4.0 * b * c
    x_opt = None
    if disc >= 0:
        sqrt_disc = np.sqrt(disc)
        x1 = (-a_0 - sqrt_disc) / (2.0 * b)
        x2 = (-a_0 + sqrt_disc) / (2.0 * b)
        if x1 > x2:
            x1, x2 = x2, x1
        # Feasible: x <= x1 or x >= x2, combined with [lo, hi]
        candidates = []
        # [lo, min(hi, x1)]
        c1_hi = min(hi, x1)
        if lo <= c1_hi:
            candidates.append((lo + c1_hi) / 2)  # midpoint
            candidates.append(lo)
            candidates.append(c1_hi)
        # [max(lo, x2), hi]
        c2_lo = max(lo, x2)
        if c2_lo <= hi:
            candidates.append((c2_lo + hi) / 2)
            candidates.append(c2_lo)
            candidates.append(hi)
        # Also boundary
        candidates.append(0.0)
        candidates.append(1.0)
        candidates.append(np.clip((lo + hi) / 2, 0, 1))
        
        best_x = None
        best_val = np.inf
        for x in candidates:
            x = np.clip(x, 0.0, 1.0)
            L_val = np.max(a_coeffs * x + d_vals) if len(a_coeffs) > 0 else 0
            Q_val = b * x * x + a_0 * x + d_0
            val = max(L_val, Q_val)
            if val < best_val:
                best_val = val
                best_x = x
        x_opt = best_x
    else:
        # Always feasible for Q, so just use midpoint of [lo, hi]
        x_opt = np.clip((lo + hi) / 2, 0, 1)
    
    return np.clip(x_opt, 0.0, 1.0)

def coordinate_descent(h, n, deadline, max_sweeps=200, order='cyclic'):
    """Cyclic coordinate descent with exact 1D minimax."""
    dx = 2.0 / n
    h = h.copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        
        # Get the order of coordinates
        if order == 'cyclic':
            indices = range(n)
        else:
            indices = np.random.permutation(n)
        
        max_change = 0.0
        for i in indices:
            # Compute the coefficients for the 1D subproblem
            # C(k) = sum_n h[n+k]*(1-h[n])*dx for lag k = m - (n-1)
            # Position i contributes to C(k) as described above.
            
            # Get current C values
            C = compute_all_lags(h, n)  # shape (2n-1,)
            # C[m] corresponds to lag k = m - (n-1)
            
            # For position i, find all lags it participates in
            # "a" part: n = i-k, valid if 0 <= i-k <= n-1 → k from i-(n-1) to i
            # "v" part: n = i, valid if 0 <= i+k <= n-1 → k from -(n-1) to n-1-i
            
            # The contribution of h[i] to C[m] (where k = m - (n+1)... let me use m directly)
            # C[m] = sum_n h[n + (m-(n-1))] * (1-h[n]) * dx
            # Let lag = m - (n-1)
            # C[m] = sum_n h[n+lag] * (1-h[n]) * dx
            
            # Position i in "a" part: n+lag = i → n = i - lag = i - (m-(n-1))
            #   valid if 0 <= i-(m-(n-1)) <= n-1 and 0 <= m-(n-1) ... hmm this is getting confusing.
            
            # Let me just directly compute the contribution.
            # For each m in 0..2n-2, the lag is k = m - (n-1).
            # C[m] = dx * sum_{n: 0<=n<=n-1, 0<=n+k<=n-1} h[n+k] * (1-h[n])
            
            # Position i appears as h[n+k] = h[i] when n = i-k:
            #   need 0 <= i-k <= n-1 (for h[n] = h[i-k] to exist)
            #   AND n = i-k must be in [0, n-1] (same condition)
            # Position i appears as h[n] = h[i] (in the 1-h[n] factor) when n = i:
            #   need 0 <= i+k <= n-1 (for h[n+k] = h[i+k] to exist)
            
            # So for each lag k = m - (n-1):
            #   if 0 <= i-k <= n-1: contribution includes h[i]*(1-h[i-k])*dx
            #   if 0 <= i+k <= n-1: contribution includes h[i+k]*(1-h[i])*dx
            
            # Remove h[i]'s contribution and add new:
            # base[m] = C[m] - contribution_of_h[i]_to_C[m]
            
            # Let me compute this efficiently.
            # For m corresponding to lag k = m - (n-1):
            #   contrib[m] = dx * (h[i]*(1-h[i-k]) if 0<=i-k<=n-1 else 0) 
            #              + dx * (h[i+k]*(1-h[i]) if 0<=i+k<=n-1 else 0)
            #   but if k=0, both terms are the same: h[i]*(1-h[i])*dx, so we only count once.
            
            # Actually for k=0: n=i gives h[i+k]*(1-h[n]) = h[i]*(1-h[i]). And n=i-k=i gives the same.
            # But in the sum, n only takes one value (n=i), so it's counted once. No double counting.
            
            # So: contrib for lag k:
            #   if 0<=i-k<=n-1: h[i]*(1-h[i-k])*dx
            #   if 0<=i+k<=n-1 AND k!=0: h[i+k]*(1-h[i])*dx
            #   (for k=0, only the first condition applies, giving h[i]*(1-h[i])*dx)
            #   Wait, for k=0, the second condition is 0<=i<=n-1 which is true, 
            #   and gives h[i]*(1-h[i])*dx which is the same as the first.
            #   But in the sum C[m] = sum_n h[n+k]*(1-h[n])*dx, for k=0, n=i gives ONE term.
            #   So we should only count it once.
            
            # I think the cleanest way is:
            # For each m (0 to 2n-2):
            #   k = m - (n-1)
            #   term1 = 0
            #   if 0 <= i-k <= n-1: term1 = h[i]*(1-h[i-k])*dx
            #   term2 = 0
            #   if 0 <= i+k <= n-1 AND not (k == 0 AND 0 <= i-k <= n-1):
            #       term2 = h[i+k]*(1-h[i])*dx
            #   contrib[m] = term1 + term2
            
            # Hmm wait, for k=0: n ranges over values where 0<=n<=n-1 and 0<=n+0<=n-1, so n in [0,n-1].
            # The term for n=i is h[i+0]*(1-h[i]) = h[i]*(1-h[i]).
            # This is the ONLY term involving h[i] in C[0] (well, C at m=n-1).
            # So contrib at m=n-1 (k=0) is h[i]*(1-h[i])*dx.
            
            # For k≠0: the terms involving h[i] are:
            # - n=i-k (if valid): h[i]*(1-h[i-k])*dx
            # - n=i (if i+k valid): h[i+k]*(1-h[i])*dx
            # These are different n values, so both are present in the sum.
            
            # Great, so:
            # contrib[m] for lag k = m-(n-1):
            #   if k==0: h[i]*(1-h[i])*dx  (if i is in [0,n-1], which it always is)
            #   if k!=0:
            #     c1 = h[i]*(1-h[i-k])*dx if 0<=i-k<=n-1 else 0
            #     c2 = h[i+k]*(1-h[i])*dx if 0<=i+k<=n-1 else 0
            #     contrib = c1 + c2
            
            # Now, base[m] = C[m] - contrib[m]
            # New C[m] = base[m] + new_contrib[m]
            # where new_contrib uses the new h[i] value.
            
            # For the 1D minimax, I need:
            # new_C[m] = base[m] + new_contrib(m, h[i]_new)
            
            # For k=0: new_contrib = h_new*(1-h_new)*dx = dx*h_new - dx*h_new²
            # For k≠0:
            #   new_c1 = h_new*(1-h[i-k])*dx (if valid)
            #   new_c2 = h[i+k]*(1-h_new)*dx = h[i+k]*dx - h[i+k]*h_new*dx (if valid)
            #   new_contrib = new_c1 + new_c2
            #   = h_new*(1-h[i-k])*dx + h[i+k]*dx - h[i+k]*h_new*dx
            #   = h_new*((1-h[i-k]) - h[i+k])*dx + h[i+k]*dx
            
            # So for k≠0: new_contrib = a_k*h_new + c_k
            #   where a_k = ((1-h[i-k]) - h[i+k])*dx (with missing terms as 0)
            #   and c_k = h[i+k]*dx (if i+k valid, else 0)
            
            # For k=0: new_contrib = dx*h_new - dx*h_new²
            #   = a_0*h_new + b_0*h_new² where a_0=dx, b_0=-dx
            
            # So the full expression:
            # new_C[m] = base[m] + a_k*h_new + b_k*h_new² + c_k
            # where b_k = -dx if k==0 else 0
            # and a_k, c_k as above.
            
            # The 1D problem: min_{x in [0,1]} max_m (base[m] + a_m*x + b_m*x² + c_m)
            # = min_{x in [0,1]} max(max_{m: k≠0} (base[m] + c_m + a_m*x), base[k=0] + c_0 + a_0*x - dx*x²)
            
            # Let me define:
            # L(x) = max_{m: lag≠0} (a_m * x + (base[m] + c_m))
            # Q(x) = a_0 * x + (base[k=0] + c_0) - dx * x²
            # where k=0 corresponds to m = n-1.
            
            # Compute a_m and (base[m] + c_m) for each m:
            
            new_h = h[i]  # current value
            
            # I'll compute this in a vectorized way
            lags = np.arange(2*n-1) - (n-1)  # lag values: -(n-1), ..., 0, ..., n-1
            
            # a_m: coefficient of x (= new h[i])
            a = np.zeros(2*n-1)
            c = np.zeros(2*n-1)
            b = np.zeros(2*n-1)  # quadratic coefficient
            
            for m in range(2*n-1):
                k = lags[m]
                if k == 0:
                    a[m] = dx
                    b[m] = -dx
                    c[m] = 0
                else:
                    # c1 = h[i]*(1-h[i-k])*dx → coefficient of h_new is (1-h[i-k])*dx
                    a1 = 0.0
                    if 0 <= i-k <= n-1:
                        a1 = (1.0 - h[i-k]) * dx
                    # c2 = h[i+k]*(1-h_new)*dx = h[i+k]*dx - h[i+k]*h_new*dx
                    a2 = 0.0
                    c2_const = 0.0
                    if 0 <= i+k <= n-1:
                        a2 = -h[i+k] * dx
                        c2_const = h[i+k] * dx
                    a[m] = a1 + a2
                    c[m] = c2_const
            
            # base[m] = C[m] - old_contrib[m]
            # old_contrib[m] = a[m] * h[i] + b[m] * h[i]**2 + c[m]
            old_contrib = a * new_h + b * new_h**2 + c
            base = C - old_contrib
            
            # Now solve: min_x max(max_{m: b[m]≠0} (a[m]*x + base[m] + c[m]), max_{m: b[m]==0} (a[m]*x + base[m] + d_m))
            # where d_m = 0 for all m (since c[m] is already the constant part)
            # Actually: new_C[m] = base[m] + a[m]*x + b[m]*x² + c[m]
            # For m with b[m]=0: new_C[m] = (a[m]*x) + (base[m]+c[m])  → linear in x
            # For m with b[m]=-dx (only m = n-1, i.e., k=0): new_C[m] = base[m] + c[m] + a[m]*x - dx*x² → quadratic
            
            # Separate:
            is_quad = (np.abs(b) > 1e-15)
            quad_idx = np.where(is_quad)[0]
            lin
