```python
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    cur = h.copy()
    cur_c = np.correlate(cur, 1.0 - cur, mode='full').astype(np.float64)
    cur_max = float(cur_c.max())
    
    # Precompute delta arrays
    # delta[pos][k] = change in c[k] when h[pos] goes from 0 to 1
    # = (1 - h[pos + s_k]) * valid(pos + s_k) - h[pos - s_k] * valid(pos - s_k)
    # where s_k = k - (n-1)
    deltas = np.zeros((n, 2*n - 1))
    s = np.arange(2*n - 1) - (n - 1)  # s_k for each k
    for pos in range(n):
        # (1 - h[pos + s_k]) if 0 <= pos + s_k < n
        idx1 = pos + s
        v1 = (idx1 >= 0) & (idx1 < n)
        deltas[pos, v1] += (1.0 - cur[idx1[v1]])
        # - h[pos - s_k] if 0 <= pos - s_k < n
        idx2 = pos - s
        v2 = (idx2 >= 0) & (idx2 < n)
        deltas[pos, v2] -= cur[idx2[v2]]

        if time.time() > deadline:
            break

        ones = np.where(cur == 1)[0]

                cur[j] = 0.0

                # where pos+s_k = i or j, or pos-s_k = i or j.

                    idx2 = pos - s

                    cur_max = new_max

                    break
            if not improved:
                break
```

```python
def grad_descent_smooth(h, n, alpha, deadline, lr=0.01, n_iter=100000):
    dx = 2.0 / n
    cur = h.copy()
    s = np.arange(2*n - 1) - (n - 1)
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        cur_c = np.correlate(cur, 1.0 - cur, mode='full')
        # Log-sum-exp smoothing
        sh = alpha * (cur_c - cur_c.max())
        weights = np.exp(sh)
        weights /= weights.sum()
        
        # Compute gradient: weighted sum of individual gradients
        grad = np.zeros(n)
        for k in range(2*n - 1):
            if weights[k] < 1e-10:
                continue
            s_k = s[k]
            # ∂c(k)/∂h[m] = (1-h[m+s_k]) - h[m-s_k]
            idx1 = np.arange(n) + s_k
            v1 = (idx1 >= 0) & (idx1 < n)
            g1 = np.zeros(n)
            g1[v1] = 1.0 - cur[idx1[v1]]
            idx2 = np.arange(n) - s_k
            v2 = (idx2 >= 0) & (idx2 < n)
            g2 = np.zeros(n)
            g2[v2] = cur[idx2[v2]]
            grad += weights[k] * (g1 - g2)
        
        # Gradient descent step
        cur = cur - lr * grad
        # Project to feasible set
        cur = np.clip(cur, 0.0, 1.0)
        # Fix sum constraint
        target = n / 2.0
        diff = target - cur.sum()
        if abs(diff) > 1e-10:
            # Distribute the difference
            if diff > 0:
                room = 1.0 - cur
                s_room = room.sum()
                if s_room > 1e-14:
                    cur += diff / s_room * room
            else:
                s_cur = cur.sum()
                if s_cur > 1e-14:
                    cur += diff / s_cur * cur
            cur = np.clip(cur, 0.0, 1.0)
    
    return cur
```

def binary_local_search(h, n, deadline, seed=42):
    """Efficient binary local search using delta computation."""
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    cur = h.copy()
    cur_c = np.correlate(cur, 1.0 - cur, mode='full').astype(np.float64)
    cur_max = float(cur_c.max())
    s = np.arange(2*n - 1) - (n - 1)
    
    stuck_count = 0
    while time.time() < deadline:
        zeros = np.where(cur == 0)[0]
        ones = np.where(cur == 1)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        improved = False
        order = rng.permutation(len(zeros))
        for idx in order:
            if time.time() > deadline:
                return cur, float(cur_max * dx)
            i = zeros[idx]
            idx1 = i + s
            v1 = (idx1 >= 0) & (idx1 < n)
            idx2 = i - s
            v2 = (idx2 >= 0) & (idx2 < n)
            delta_i = np.zeros(2*n - 1)
            delta_i[v1] = (1.0 - cur[idx1[v1]])
            delta_i[v2] -= cur[idx2[v2]]
            
            best_j = -1
            best_new_max = cur_max
            for j in ones:
                idx1j = j + s
                v1j = (idx1j >= 0) & (idx1j < n)
                idx2j = j - s
                v2j = (idx2j >= 0) & (idx2j < n)

                break
    return cur, float(cur_max * dx)
