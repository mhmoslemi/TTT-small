```python
def compute_c5(h, n):

    c = np.correlate(h, 1.0 - h, mode='full')

def compute_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full').astype(np.float64)

def project_feasible(h, n):
    """Project h to [0,1]^n with sum = n/2."""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * room, 0.0, 1.0)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * h, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def slsqp_polish(h, n, maxiter, deadline, alpha=10000):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    
    def cons_fun(x):
        return x.sum() - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})

    except Exception:
        return h
```

```python
def compute_delta(h, i, j, N):
    """Compute delta for all lags when flipping h[i]:0->1, h[j]:1->0"""
    lags = np.arange(2*N-1) - (N-1)  # lags from -(N-1) to N-1
    delta = np.zeros(2*N-1)
    
    # Term 1: +(1-h[i-l]) for valid i-l
    idx = i - lags  # = i - (idx_out - (N-1)) 
    valid = (idx >= 0) & (idx < N)
    delta[valid] += (1 - h[idx[valid]])
    
    # Term 2: -h[i+l] for valid i+l
    idx = i + lags
    valid = (idx >= 0) & (idx < N)
    delta[valid] -= h[idx[valid]]
    
    # Term 3: -(1-h[j-l]) for valid j-l
    idx = j - lags
    valid = (idx >= 0) & (idx < N)
    delta[valid] -= (1 - h[idx[valid]])
    
    # Term 4: +h[j+l] for valid j+l
    idx = j + lags
    valid = (idx >= 0) & (idx < N)
    delta[valid] += h[idx[valid]]
    
    return delta
```

```python
def fast_greedy_search(h, n, deadline, n_tries=5000000, seed=42):
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    current_max = corr.max()
    best_h = h.copy()
    best_max = current_max
    
    lags = np.arange(2*n-1) - (n-1)
    rng = np.random.RandomState(seed)
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    no_improve_count = 0
    max_no_improve = 2000
    
    for t in range(n_tries):
        if time.time() > deadline:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        
        # Compute delta
        delta = np.zeros(2*n-1)
        # Term 1: (1-h[i-l]) for valid i-l
        idx = i - lags
        v = (idx >= 0) & (idx < n)
        delta[v] += (1 - h[idx[v]])
        # Term 2: -h[i+l] for valid i+l
        idx = i + lags
        v = (idx >= 0) & (idx < n)
        delta[v] -= h[idx[v]]
        # Term 3: -(1-h[j-l]) for valid j-l
        idx = j - lags
        v = (idx >= 0) & (idx < n)
        delta[v] -= (1 - h[idx[v]])
        # Term 4: +h[j+l] for valid j+l
        idx = j + lags
        v = (idx >= 0) & (idx < n)
        delta[v] += h[idx[v]]
        
        # Corrections
        delta[n-1] -= 2.0  # l=0 correction
        delta[i-j+n-1] += 1.0  # l=i-(q) correction... wait
        # l = i-j (p=i, q=j): output index = i-j+n-1
        # l = j-i: output index = j-i+n-1
        delta[i-j+n-1] += 1.0
        delta[j-i+n-1] += 1.0
        
        new_max = (corr + delta).max()
        
        if new_max < current_max - 1e-12:
            h[i] = 1.0
            h[j] = 0.0
            corr = corr + delta
            current_max = new_max
            # Update ones and zeros
            ones = ones[ones != j]
            zeros = zeros[zeros != i]
            zeros = np.append(zeros, j)
            no_improve_count = 0
            if current_max < best_max:
                best_max = current_max
                best_h = h.copy()
        else:
            no_improve_count += 1
            if no_improve_count > max_no_improve:
                break
    
    return best_h, best_max
```
