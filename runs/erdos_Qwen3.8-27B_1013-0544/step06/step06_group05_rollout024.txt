To speed this up, I can provide an analytical gradient. Let me derive it:

f(h) = soft_max_k (corr_k * dx) where corr_k = ∑_i h[i](1-h[i+k])

Let g_k = corr_k * dx, m = max_k g_k, w_k = exp(α(g_k - m)) / Σ_j exp(α(g_j - m))

f = m + (log Σ_k exp(α(g_k - m))) / α

∂f/∂h[i] = (1/α) * α * Σ_k w_k * ∂g_k/∂h[i] = Σ_k w_k * ∂g_k/∂h[i]

∂g_k/∂h[i] = dx * ∂corr_k/∂h[i]

For corr_k with lag l = k-(n-1):
corr_k = Σ_n h[n] * (1 - h[n+l]) where n+l must be in [0, n)

∂corr_k/∂h[i] = (1 - h[i+l]) * [i+l in [0,n)] + Σ_n [n+l=i] * (-h[n])
= (1 - h[i+l]) * [i+l in [0,n)] - h[i-l] * [i-l in [0,n)]

So:
∂f/∂h[i] = dx * Σ_k w_k * [(1-h[i+l_k]) * 1_{i+l_k ∈ [0,n)} - h[i-l_k] * 1_{i-l_k ∈ [0,n)}]

where l_k = k - (n-1).

This can be computed in O(n * K) where K is the number of significant weights (w_k > threshold). With K ≈ 10-50, this is O(2500-12500) operations, much faster than finite differences.

from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def get_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * room, 0.0, 1.0)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * h, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):

def slsqp_soft(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n

    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

    def obj_grad(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        w = np.exp(sh - sm)
        w /= w.sum()
        grad = np.zeros(n)
        idx = (n - 1)
        for k_idx in range(len(w)):
            if w[k_idx] < 1e-9:
                continue
            lag = k_idx - idx
            wl = w[k_idx] * dx
            for i in range(n):
                c = 0.0
                ip = i + lag
                if 0 <= ip < n:
                    c += (1.0 - x[ip])
                im = i - lag
                if 0 <= im < n:
                    c -= x[im]
                grad[i] += wl * c
        return grad

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        if res.success or res.status in (0, 1, 2):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h
