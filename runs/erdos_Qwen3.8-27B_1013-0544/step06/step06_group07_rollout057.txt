def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def round_to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]

    out[idx[:target]] = 1.0

def swap_delta(h, n, i, j, m_range):

    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range

    delta[msk] -= h[imn[msk]]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def round_to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    return out

def swap_delta(h, n, i, j, m_range):
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    delta[k2] += 1.0
    return delta

def two_swap_delta(h, n, i1, j1, i2, j2, m_range):
    d1 = swap_delta(h, n, i1, j1, m_range)
    h_tmp = h.copy()
    h_tmp[i1] = 1.0
    h_tmp[j1] = 0.0
    d2 = swap_delta(h_tmp, n, i2, j2, m_range)
    return d1 + d2

def binary_sa(h_bin, n, deadline, seed=42, t_start=0.05, t_end=0.00001, patience=60000):
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    h = h_bin.copy()
    m_range = np.arange(-(n - 1), n)
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    zeros_idx = np.where(h == 0)[0]
    ones_idx = np.where(h == 1)[0]
    no_improve = 0
    cool_rate = (t_end / t_start) ** (1.0 / 600000.0)
    T = t_start
    step = 0

    while time.time() < deadline:
        step += 1
        if step % 100000 == 0 and len(zeros_idx) < 5:
            break
        use_two_swap = (rng.random() < 0.05 and len(zeros_idx) > 4 and len(ones_idx) > 4)
        if use_two_swap:
            i1 = zeros_idx[rng.randint(len(zeros_idx))]
            i2 = zeros_idx[rng.randint(len(zeros_idx))]
            if i1 == i2:
                i2 = zeros_idx[rng.randint(len(zeros_idx))]
            j1 = ones_idx[rng.randint(len(ones_idx))]
            j2 = ones_idx[rng.randint(len(ones_idx))]
            if j1 == j2:
                j2 = ones_idx[rng.randint(len(ones_idx))]
            if i1 == i2 or j1 == j2:
                i = zeros_idx[rng.randint(len(zeros_idx))]
                j = ones_idx[rng.randint(len(ones_idx))]
                delta = swap_delta(h, n, i, j, m_range)
            else:
                delta = two_swap_delta(h, n, i1, j1, i2, j2, m_range)
            new_corr = corr + delta
            nc = new_corr.max() * dx
            delta_c = nc - cur_c
            if delta_c < 0 or (T > 1e-12 and rng.random() < np.exp(-delta_c / T)):
                h[i1] = 1.0
                h[j1] = 0.0
                h[i2] = 1.0
                h[j2] = 0.0
                corr = new_corr
                cur_c = nc
                zeros_idx = np.where(h == 0)[0]
                ones_idx = np.where(h == 1)[0]
                if nc < best_c:
                    best_c = nc
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
        else:
            i = zeros_idx[rng.randint(len(zeros_idx))]
            j = ones_idx[rng.randint(len(ones_idx))]
            delta = swap_delta(h, n, i, j, m_range)
            new_corr = corr + delta
            nc = new_corr.max() * dx
            delta_c = nc - cur_c
            if delta_c < 0 or (T > 1e-12 and rng.random() < np.exp(-delta_c / T)):
                h[i] = 1.0
                h[j] = 0.0
                corr = new_corr
                cur_c = nc
                zeros_idx = np.where(h == 0)[0]
                ones_idx = np.where(h == 1)[0]
                if nc < best_c:
                    best_c = nc
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        T *= cool_rate
        if no_improve > patience:
            nk = 4 + rng.randint(10)
            zi = np.where(h == 0)[0]
            oi = np.where(h == 1)[0]
            for _ in range(nk):
                ii = zi[rng.randint(len(zi))]
                jj = oi[rng.randint(len(oi))]
                h[ii] = 1.0
                h[jj] = 0.0
            corr = np.correlate(h, 1 - h, mode='full').astype(float)
            cur_c = corr.max() * dx
            zeros_idx = np.where(h == 0)[0]
            ones_idx = np.where(h == 1)[0]
            no_improve = 0
            T = min(t_start, T * 5.0)
    return best_h, best_c

def greedy_descent(h, n, deadline, seed=42, max_sweeps=100):
    rng = np.random.RandomState(seed)
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        rng.shuffle(zeros)
        rng.shuffle(ones)
        for i in zeros:
            if time.time() > deadline:
                break
            for j in ones:
                delta = swap_delta(h, n, i, j, m_range)
                nc = (corr + delta).max() * dx
                if nc < cur_c - 1e-15:
                    h[i] = 1.0
                    h[j] = 0.0
                    corr = corr + delta
                    cur_c = nc
                    improved = True
                    break
            if improved:
                break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        if not improved:
            break
    return best_h, best_c

def upsample_binary(h, n_old, n_new, rng):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    values = np.interp(x_new, x_old, h)
    return round_to_binary(values, n_new)
