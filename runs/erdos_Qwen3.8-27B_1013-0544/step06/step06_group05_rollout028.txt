def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-15: break
            h = np.clip(h + diff / s * room, 0.0, 1.0)
        else:
            s = h.sum()
            if s < 1e-15: break
            h = np.clip(h + diff / s * h, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def binary_local_search(h_init, n, deadline, max_sweeps=10000, seed=42):
    """Fast binary local search: try all swaps per sweep."""
    rng = np.random.RandomState(seed)
    h = h_init.copy()
    corr = compute_corr(h, n)
    cur_max = corr.max()
    best_c = cur_max
    best_h = h.copy()
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Shuffle for randomness
        zeros = zeros[rng.permutation(len(zeros))]
        ones = ones[rng.permutation(len(ones))]

        for i in zeros:

                break
            for j in ones:
                h[i] = 1.0
