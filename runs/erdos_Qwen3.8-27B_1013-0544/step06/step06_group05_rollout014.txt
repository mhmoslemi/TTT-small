def simulated_annealing(h, n, T_initial, T_final, n_steps, deadline):
    dx = 2.0 / n
    best_h = h.copy()
    best_c = np.max(np.correlate(h, 1-h, mode='full')) * dx
    cur_h = h.copy()
    cur_corr = np.correlate(cur_h, 1-cur_h, mode='full')
    cur_max = cur_corr.max()
    
    ones = np.where(cur_h == 1)[0]
    zeros = np.where(cur_h == 0)[0]
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        T = T_initial * (T_final / T_initial) ** (step / n_steps)
        
        # Pick random swap
        i_idx = np.random.randint(len(ones))
        j_idx = np.random.randint(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]
        if i == j:
            continue
        
        # Compute new correlation
        cur_h[i] = 0
        cur_h[j] = 1
        new_corr = np.correlate(cur_h, 1-cur_h, mode='full')
        new_max = new_corr.max()
        
        if new_max < cur_max or np.random.random() < np.exp((cur_max - new_max) / T):
            cur_max = new_max
            cur_corr = new_corr
            ones = np.where(cur_h == 1)[0]
            zeros = np.where(cur_h == 0)[0]
            if new_max < best_c / dx:  # Actually best_c is already scaled...
                best_h = cur_h.copy()
                best_c = new_max * dx
        else:
            cur_h[i] = 1
            cur_h[j] = 0
    
    return best_h, best_c

        h[best_j] = 1
        new_corr = np.correlate(h, 1-h, mode='full')
        return h, new_corr, best_delta_max
    return h, corr, c.max()

Now let me think about the simulated annealing more carefully. The key is the cooling schedule and the number of steps.

For simulated annealing on binary vectors:
- State: binary vector h with sum = n/2
- Move: pick random i (where h=1) and random j (where h=0), swap
- Energy: E(h) = max_k c(k) (unscaled)
- Temperature schedule: T(t) = T0 * (Tf/T0)^(t/t_max) or T(t) = T0 / (1 + t*cooling_rate)

The acceptance criterion: accept if new_E < old_E, or with probability exp((old_E - new_E)/T).

For the energy difference, I need to compute the new max correlation after a swap. This is O(n) per move (compute delta for all lags, then find max).

Let me estimate the number of SA steps I can do:

- But with Python overhead, maybe 10-50 μs per step

Let me also think about a key optimization: instead of computing the full new correlation and finding its max, I can track the current max and only need to check if the new max exceeds it or is below it.

Actually, for SA, I need the exact new energy to compute the acceptance probability. So I do need to compute the new max.

But I can optimize: the new correlation is c + delta, where delta is O(1) in magnitude (at most 2 or so). So the new max is either:
- The old max (if the lag that achieved the old max isn't affected)
- Or one of the lags where delta is positive (if any)
- Or one of the lags that were close to the old max and got reduced

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * room, 0.0, 1.0)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * h, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    """Convert continuous h to binary with exactly n/2 ones."""
    sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:n//2]] = 1.0
    return h_bin

def delta_for_swap(h, i, j, n):
    """Compute delta c[lag] for all lags when h[i]:1->0, h[j]:0->1."""
    lags = np.arange(-(n-1), n)
    m = 2*n - 1
    
    # v1 = h[i+lag] for all lags (0 out of bounds)
    idx1 = i + lags
    v1 = np.zeros(m)
    mk1 = (idx1 >= 0) & (idx1 < n)
    v1[mk1] = h[idx1[mk1]]
    # h[j] = 0 already, so if i+lag == j, v1 is already 0. Correct.
    
    # v2 = h'[j+lag] for all lags (0 out of bounds)
    idx2 = j + lags
    v2 = np.zeros(m)
    mk2 = (idx2 >= 0) & (idx2 < n)
    v2[mk2] = h[idx2[mk2]]
    # If j+lag == i, h'[i] = 0 (after swap), so set v2 to 0 there
    v2[idx2 == i] = 0
    
    # v3 = h[i-lag] for valid positions, excluding m=i and m=j
    idx3 = i - lags
    v3 = np.zeros(m)
    mk3 = (idx3 >= 0) & (idx3 < n) & (idx3 != i) & (idx3 != j)
    v3[mk3] = h[idx3[mk3]]
    
    # v4 = h[j-lag] for valid positions, excluding m=i and m=j
    idx4 = j - lags
    v4 = np.zeros(m)
    mk4 = (idx4 >= 0) & (idx4 < n) & (idx4 != i) & (idx4 != j)
    v4[mk4] = h[idx4[mk4]]
    
    delta = -(1.0 - v1) + (1.0 - v2) + v3 - v4
    return delta

    """Best-improvement local search for binary h."""
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    cur_max = corr.max()
    dx = 2.0 / n
    best_c = cur_max * dx
    best_h = h.copy()
    
    ones = np.where(h == 1)[0]
    
    while time.time() < deadline:
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        best_new_max = cur_max  # threshold: must be strictly less
        best_i, best_j = -1, -1
        
        for i in ones:
            for j in zeros:
                if i == j:
                    continue
                delta = delta_for_swap(h, i, j, n)
                new_corr = corr + delta
                new_max = new_corr.max()
                if new_max < best_new_max - 1e-12:
                    best_new_max = new_max
                    best_i, best_j = i, j
        
        if best_i >= 0:
            h[best_i] = 0
            h[best_j] = 1
            corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
            cur_max = corr.max()
            improved = True
            if cur_max * dx < best_c:
                best_c = cur_max * dx
                best_h = h.copy()
        else:
            break
    
    return best_h, best_c

def simulated_annealing(h, n, deadline, n_restarts=5, seed=42):
    """Simulated annealing for binary h."""
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    lags = np.arange(-(n-1), n)
    m = 2*n - 1
    
    for restart in range(n_restarts):
        if time.time() > deadline:
            break
        cur = h.copy() if restart == 0 else best_h.copy()
        cur_corr = np.correlate(cur, 1.0 - cur, mode='full').astype(np.float64)
        cur_E = cur_corr.max()
        
        # Precompute index arrays for speed
        ones_list = list(np.where(cur == 1)[0])
        zeros_list = list(np.where(cur == 0)[0])
        
        T_start = 3.0
        T_end = 0.01
        n_steps = 500000
        # Adaptive: run until deadline or n_steps
        
        step = 0
        while time.time() < deadline and step < n_steps:
            step += 1
            T = T_start * (T_end / T_start) ** (step / n_steps)
            
            # Random swap
            i = ones_list[rng.randint(len(ones_list))]
            j = zeros_list[rng.randint(len(zeros_list))]
            if i == j:
                continue
            
            delta = delta_for_swap(cur, i, j, n)
            new_corr = cur_corr + delta
            new_E = new_corr.max()
            
            delta_E = new_E - cur_E
            if delta_E < 0 or rng.random() < np.exp(-delta_E / T):
                # Accept
                cur[i] = 0
                cur[j] = 1
                cur_corr = new_corr
                cur_E = new_E
                # Update lists
                ones_list[ones_list.index(i)] = j
                zeros_list[zeros_list.index(j)] = i
                # Reorganize
                if new_E * dx < best_c:
                    best_c = new_E * dx
                    best_h = cur.copy()
    
    return best_h, best_c
