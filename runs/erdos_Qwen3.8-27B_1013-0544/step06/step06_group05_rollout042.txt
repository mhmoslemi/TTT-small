import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full').astype(np.float64)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * room, 0.0, 1.0)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * h, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def compute_influence(h, n):
    i_idx = np.arange(n)[:, None]
    lag_idx = np.arange(-(n - 1), n)[None, :]
    idx1 = i_idx + lag_idx
    valid1 = (idx1 >= 0) & (idx1 < n)
    term1 = np.where(valid1, 1.0 - h[np.clip(idx1, 0, n - 1)], 0.0)
    idx2 = i_idx - lag_idx
    valid2 = (idx2 >= 0) & (idx2 < n)
    term2 = np.where(valid2, -h[np.clip(idx2, 0, n - 1)], 0.0)
    return term1 + term2

def greedy_binary(h, n, deadline, max_sweeps=30000):
    h = h.copy()
    corr = compute_corr(h, n)
    infl = compute_influence(h, n)
    cur_max = corr.max()
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        critical = np.where(corr >= cur_max - 2.0)[0]
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        d = infl[zeros[:, None, :], critical[None, :]] - infl[ones[None, :, :], critical[None, :]]
        new_vals = corr[critical][None, None, :] + d
        new_max = new_vals.max(axis=2)
        n_try = min(8, new_max.size)
        flat_idx = np.argsort(new_max.ravel())[:n_try]
        improved = False
        for fi in flat_idx:
            r, c = np.unravel_index(fi, new_max.shape)
            if new_max[r, c] >= cur_max:
                break
            bi, bj = zeros[r], ones[c]
            h[bi] = 1.0
            h[bj] = 0.0
            nc = compute_corr(h, n)
            am = nc.max()
            if am < cur_max:
                cur_max = am
                corr = nc
                infl = compute_influence(h, n)
                improved = True
                break
            else:
                h[bi] = 0.0
                h[bj] = 1.0
        if not improved:
            break
    return h, float(cur_max * (2.0 / n))

def iterated_local_search(h, n, deadline, max_iters=80000, rng=None):
    if rng is None:

    h = h.copy()

    infl = compute_influence(h, n)
    cur_max = corr.max()

    best_max = cur_max
    stuck_count = 0
    for _ in range(max_iters):
        if time.time() > deadline:

        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        d = infl[zeros[:, None, :], critical[None, :]] - infl[ones[None, :, :], critical[None, :]]
        new_vals = corr[critical][None, None, :] + d
        new_max = new_vals.max(axis=2)
        n_try = min(5, new_max.size)
        flat_idx = np.argsort(new_max.ravel())[:n_try]
        improved = False
        for fi in flat_idx:
            r, c = np.unravel_index(fi, new_max.shape)
            if new_max[r, c] >= cur_max:
                break
            bi, bj = zeros[r], ones[c]
            h[bi] = 1.0
            h[bj] = 0.0
            nc = compute_corr(h, n)
            am = nc.max()
            if am < cur_max:

                corr = nc
                infl = compute_influence(h, n)
                improved = True
                stuck_count = 0
                if cur_max < best_max:

                break

                    h[rng.choice(z)] = 1.0

def smooth_grad_descent(h, n, deadline, alpha=40.0, max_iter=5000):
    h = h.copy()
    lr = 0.02
    cur_val = compute_c5(h, n)
    patience = 0
    for _ in range(max_iter):
        if time.time() > deadline:
            break
        corr = compute_corr(h, n)
        shifted = alpha * (corr - corr.max())
        p = np.exp(shifted)
        p /= p.sum()
        infl = compute_influence(h, n)
        grad = infl.T @ p
        h_new = project_feasible(h - lr * grad, n)
        new_val = compute_c5(h_new, n)
        if new_val < cur_val:
            h = h_new
            cur_val = new_val
            lr = min(lr * 1.05, 0.15)
            patience = 0
        else:
            lr *= 0.5
            patience += 1
            if patience > 50:
                break
    return h, cur_val
