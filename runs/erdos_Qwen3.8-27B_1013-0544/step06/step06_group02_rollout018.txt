import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_all(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def precompute_coeffs(h, n):
    ncorr = 2 * n - 1
    coeff = np.zeros((n, ncorr))
    for i in range(n):
        lo1 = max(0, i)
        hi1 = min(i + n, ncorr)
        if lo1 < hi1:
            m1 = np.arange(lo1, hi1)
            u1 = i + (n - 1) - m1
            coeff[i, m1] += (1.0 - h[u1])
        lo2 = max(0, n - 1 - i)
        hi2 = min(2 * n - 1 - i, ncorr)
        if lo2 < hi2:
            m2 = np.arange(lo2, hi2)
            v2 = i + m2 - (n - 1)
            coeff[i, m2] -= h[v2]
    return coeff

def smooth_max_obj(x, n, alpha, mu):
    c = compute_c5_all(x, n)
    m = c.max()
    s = alpha * (c - m)
    sm = s.max()
    if sm > 500:
        val = m + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(s))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    c = compute_c5_all(x, n)
    m = c.max()
    s = alpha * (c - m)
    sm = s.max()
    w = np.exp(s - sm)
    w /= w.sum()
    dx = 2.0 / n
    coeff = precompute_coeffs(x, n)
    grad = dx * (w @ coeff)
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad
