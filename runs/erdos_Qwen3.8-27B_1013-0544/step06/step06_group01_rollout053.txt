```python
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def fast_coord_descent(h, n, max_passes=500, grid_pts=60, deadline=None, rng=None):
    h = h.copy()
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    
    for _pass in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        order = np.arange(n)
        if rng is not None:
            rng.shuffle(order)
        improved = False
        for p in order:
            vp = h[p]
            d_lo = -vp
            d_hi = 1.0 - vp
            if d_hi - d_lo < 1e-14:
                continue
            
            S = np.zeros(2*n-1)
            m1_lo = max(0, n-1-p)
            m1_hi = min(2*n-2, n-1+p)
            if m1_lo <= m1_hi:
                idx = p + np.arange(m1_lo, m1_hi+1) - n + 1
                S[m1_lo:m1_hi+1] += (1.0 - h[idx])
            m2_lo = max(0, p)
            m2_hi = min(2*n-2, p+n-1)
            if m2_lo <= m2_hi:
                idx = p - np.arange(m2_lo, m2_hi+1) + n - 1
                S[m2_lo:m2_hi+1] -= h[idx]
            
            deltas = np.linspace(d_lo, d_hi, grid_pts)
            new_c = c[None, :] + deltas[:, None] * S[None, :]
            new_c[:, n-1] = c[n-1] + deltas * S[n-1] - deltas**2
            vals = new_c.max(axis=1)
            best_idx = vals.argmin()
            best_d = deltas[best_idx]
            
            if abs(best_d) > 1e-12 and vals[best_idx] < c.max() - 1e-13:
                h[p] = np.clip(vp + best_d, 0.0, 1.0)
                c = c + best_d * S
                c[n-1] -= best_d**2
                improved = True
        
        if not improved:
            break
    
    return h, float(c.max()) * dx

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    # log-sum-exp for numerical stability
    mx = sh.max()
    if mx > 500:
        lse = (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        lse = np.log(np.sum(np.exp(sh))) / alpha
    return float((m + lse) * dx)

def smooth_obj_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w = w / w.sum()
    # dc/dx[p] for each p
    # c[m] = sum_i x[i+m-n+1]*(1-x[i])
    # dc[m]/dx[p] = (1-x[p+m-n+1]) for valid m (case 1) - x[p-m+n-1] for valid m (case 2) - 2*x[p]*(m==n-1)
    # This is the same sensitivity S[m] plus the -2x[p] at m=n-1
    grad = np.zeros(n)
    for p in range(n):
        # S[p] as before
        s = np.zeros(2*n-1)
        m1_lo = max(0, n-1-p)
        m1_hi = min(2*n-2, n-1+p)
        if m1_lo <= m1_hi:
            idx = p + np.arange(m1_lo, m1_hi+1) - n + 1
            s[m1_lo:m1_hi+1] += (1.0 - x[idx])
        m2_lo = max(0, p)
        m2_hi = min(2*n-2, p+n-1)
        if m2_lo <= m2_hi:
            idx = p - np.arange(m2_lo, m2_hi+1) + n - 1
            s[m2_lo:m2_hi+1] -= x[idx]
        s[n-1] -= 2.0 * x[p]
        grad[p] = w @ s
    return dx * grad

def slsqp_optimize(h, n, alpha=100000, maxiter=2000, deadline=None):
    x0 = h.copy()
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_obj, x0, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        if time.time() < deadline:
            return project_feasible(res.x, n)
    except:
        pass
    return project_feasible(x0, n)

def fourier_perturb(h, n, rng, num_modes=None, scale=0.05):
    """Add structured perturbation in Fourier domain."""
    h_new = h.copy()
    N = len(h)
    if num_modes is None:
        num_modes = max(3, N // 20)
    # Get random low-frequency modes
    modes = rng.choice(range(1, N//2), size=min(num_modes, N//2-1), replace=False)
    for k in modes:
        amp = rng.uniform(-scale, scale)
        phase = rng.uniform(0, 2*np.pi)
        h_new += amp * np.cos(2*np.pi*k*np.arange(N)/N + phase)
    return project_feasible(h_new, N)

def pair_update(h, n, p1, p2, grid_pts=25):
    """Try updating two coordinates simultaneously."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    current_max = float(c.max())
    
    # Sensitivity for p1
    S1 = np.zeros(2*n-1)
    m1_lo = max(0, n-1-p1); m1_hi = min(2*n-2, n-1+p1)
    if m1_lo <= m1_hi:
        S1[m1_lo:m1_hi+1] += (1.0 - h[p1 + np.arange(m1_lo, m1_hi+1) - n + 1])
    m2_lo = max(0, p1); m2_hi = min(2*n-2, p1+n-1)
    if m2_lo <= m2_hi:
        S1[m2_lo:m2_hi+1] -= h[p1 - np.arange(m2_lo, m2_hi+1) + n - 1]
    
    # Sensitivity for p2
    S2 = np.zeros(2*n-1)
    m1_lo = max(0, n-1-p2); m1_hi = min(2*n-2, n-1+p2)
    if m1_lo <= m1_hi:
        S2[m1_lo:m1_hi+1] += (1.0 - h[p2 + np.arange(m1_lo, m1_hi+1) - n + 1])
    m2_lo = max(0, p2); m2_hi = min(2*n-2, p2+n-1)
    if m2_lo <= m2_hi:
        S2[m2_lo:m2_hi+1] -= h[p2 - np.arange(m2_lo, m2_hi+1) + n - 1]
    
    d1_lo = -h[p1]; d1_hi = 1.0 - h[p1]
    d2_lo = -h[p2]; d2_hi = 1.0 - h[p2]
    
    d1s = np.linspace(d1_lo, d1_hi, grid_pts)
    d2s = np.linspace(d2_lo, d2_hi, grid_pts)
    
    best_d1, best_d2 = 0.0, 0.0
    best_val = current_max
    
    # Subsample for speed
    d1s_sub = d1s[::max(1, len(d1s)//12)]
    d2s_sub = d2s[::max(1, len(d2s)//12)]
    
    for d1 in d1s_sub:
        for d2 in d2s_sub:
            new_c = c + d1*S1 + d2*S2
            new_c[n-1] -= d1**2 + d2**2
            val = float(new_c.max())
            if val < best_val - 1e-13:
                best_val = val
                best_d1, best_d2 = d1, d2
    
    if abs(best_d1) > 1e-12 or abs(best_d2) > 1e-12:
        h[p1] = np.clip(h[p1] + best_d1, 0, 1)
        h[p2] = np.clip(h[p2] + best_d2, 0, 1)
    
    return h, best_val * dx
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def build_sensitivity(h, n):
    """Build sensitivity matrix S of shape (n, 2n-1)."""
    S = np.zeros((n, 2*n-1))
    for p in range(n):
        m1_lo = max(0, n-1-p)
        m1_hi = min(2*n-2, n-1+p)
        if m1_lo <= m1_hi:
            idx = p + np.arange(m1_lo, m1_hi+1) - n + 1
            S[p, m1_lo:m1_hi+1] += (1.0 - h[idx])
        m2_lo = max(0, p)
        m2_hi = min(2*n-2, p+n-1)
        if m2_lo <= m2_hi:
            idx = p - np.arange(m2_lo, m2_hi+1) + n - 1
            S[p, m2_lo:m2_hi+1] -= h[idx]
    return S

def fast_cd(h, n, max_passes=500, grid_pts=60, deadline=None, rng=None):
    h = h.copy()
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    S = build_sensitivity(h, n)
    
    for _pass in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        order = np.arange(n)
        if rng is not None:
            rng.shuffle(order)
        improved = False
        for p in order:
            vp = h[p]
            d_lo = -vp
            d_hi = 1.0 - vp
            if d_hi - d_lo < 1e-14:
                continue
            # Update S row for p (since h changed, S changes)
            # Actually, S[p] doesn't change when h[p] changes (it depends on other h values)
            # But S[q] for q≠p might change! Because S[q] depends on h[p] when p appears in the sensitivity of q.
            # Specifically, S[q, m] = (1-h[q+m-n+1]) - h[q-m+n-1] (when valid)
            # So S[q, m] changes when h[q+m-n+1] or h[q-m+n+1] changes.
            # This means we need to update all rows of S after each coordinate change!
            # That's O(n²) per coordinate, making the total O(n³) per pass.
            # For n=300, that's 27M per pass. At 100M ops/sec, that's 0.27s per pass.
            # With 100 passes, that's 27s. Acceptable.
            # 
            # But wait, can we avoid full rebuild? S[p, m] depends on h at indices
            # q+m-n+1 and q-m+n-1 where q=p. So S[p] = (1-h[p+m-n+1]) - h[p-m+n-1]
            # This only depends on h at specific indices, not on h[p] itself (except at m=n-1 where
            # both terms give the same index p, giving (1-h[p])-h[p] = 1-2h[p]).
            # 
            # When we change h[p], which rows of S change?
            # S[q, m] = (1-h[q+m-n+1]) - h[q-m+n-1] (when both valid)
            # S[q, m] changes when q+m-n+1 = p or q-m+n-1 = p
            # i.e., m = p+n-1-q or m = p+n-1-q (same!)
            # So for each q, exactly one m is affected: m = p+n-1-q (if valid)
            # AND m = p-(n-1)-q... wait let me redo this.
            # 
            # S[q, m] = (1-h[q+m-n+1]) [case 1, valid when 0<=q+m-n+1<=n-1]
            #          - h[q-m+n-1] [case 2, valid when 0<=q-m+n-1<=n-1]
            # 
            # Case 1: q+m-n+1 = p → m = p+n-1-q. Valid when 0<=p+n-1-q<=2n-2 and 0<=q<=n-1.
            # Case 2: q-m+n-1 = p → m = q+n-1-p. Valid when 0<=q+n-1-p<=2n-2.
            # 
            # So for each q, at most 2 entries of S[q,:] change.
            # Total: O(n) updates per coordinate change! Much better than O(n²).
            
            # For now, let me just rebuild S (simpler code). I'll optimize later if needed.
            S = build_sensitivity(h, n)
            
            sp = S[p]
            deltas = np.linspace(d_lo, d_hi, grid_pts)
            new_c = c[None, :] + deltas[:, None] * sp[None, :]
            new_c[:, n-1] = c[n-1] + deltas * sp[n-1] - deltas**2
            vals = new_c.max(axis=1)
            best_idx = vals.argmin()
            best_d = deltas[best_idx]
            
            if abs(best_d) > 1e-12 and vals[best_idx] < c.max() - 1e-13:
                h[p] = np.clip(vp + best_d, 0.0, 1.0)
                c = c + best_d * sp
                c[n-1] -= best_d**2
                # Update S (only affected rows)
                S = build_sensitivity(h, n)
                improved = True
        
        if not improved:
            break
    
    return h, float(c.max()) * dx
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def build_sensitivity(h, n):
    S = np.zeros((n, 2 * n - 1))
    for p in range(n):
        m1_lo = max(0, n - 1 - p)
        m1_hi = min(2 * n - 2, n - 1 + p)
        if m1_lo <= m1_hi:
            idx = p + np.arange(m1_lo, m1_hi + 1) - n + 1
            S[p, m1_lo:m1_hi + 1] += (1.0 - h[idx])
        m2_lo = max(0, p)
        m2_hi = min(2 * n - 2, p + n - 1)
        if m2_lo <= m2_hi:
            idx = p - np.arange(m2_lo, m2_hi + 1) + n - 1
            S[p, m2_lo:m2_hi + 1] -= h[idx]
    return S

def update_sens_row(S, p, d, n):
    for q in range(n):
        S[q, p + n - 1 - q] -= d
        S[q, q + n - 1 - p] -= d

def fast_cd(h, n, max_passes=500, grid_pts=60, deadline=None, rng=None):
    h = h.copy()
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    S = build_sensitivity(h, n)
    for _pass in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        order = np.arange(n)
        if rng is not None:
            rng.shuffle(order)
        improved = False
        for p in order:
            vp = h[p]
            d_lo = -vp
            d_hi = 1.0 - vp
            if d_hi - d_lo < 1e-14:
                continue
            sp = S[p]
            deltas = np.linspace(d_lo, d_hi, grid_pts)
            new_c = c[None, :] + deltas[:, None] * sp[None, :]
            new_c[:, n - 1] = c[n - 1] + deltas * sp[n - 1] - deltas ** 2
            vals = new_c.max(axis=1)
            best_idx = int(vals.argmin())
            best_d = deltas[best_idx]
            if abs(best_d) > 1e-12 and vals[best_idx] < c.max() - 1e-13:
                h[p] = np.clip(vp + best_d, 0.0, 1.0)
                c = c + best_d * sp
                c[n - 1] -= best_d ** 2
                update_sens_row(S, p, best_d, n)
                improved = True
        if not improved:
            break
    return h, float(c.max()) * dx

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    sh = alpha * (c - m)
    mx = float(sh.max())
    if mx > 500:
        lse = (mx + float(np.log(np.sum(np.exp(sh - mx))))) / alpha
    else:
        lse = float(np.log(np.sum(np.exp(sh)))) / alpha
    return (m + lse) * dx

def smooth_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = float(c.max())
    sh = alpha * (c - m)
    mx = float(sh.max())
    w = np.exp(sh - mx)
    w /= w.sum()
    S = build_sensitivity(x, n)
    grad = S @ w
    grad -= w[n - 1] * 2.0 * x
    return dx * grad

def _con_fun(x, n):
    return x.sum() - n / 2.0

def _con_jac(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alpha=100000, maxiter=1500, deadline=None):
    try:
        res = minimize(smooth_obj, h.copy(), args=(n, alpha), method='SLSQP',
                       jac=lambda x: smooth_grad(x, n, alpha),
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (n,)}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if time.time() < deadline:
            return project_feasible(res.x, n)
    except Exception:
        pass
    return project_feasible(h.copy(), n)

def fourier_perturb(h, n, rng, num_modes=5, scale=0.04):
    h_new = h.copy()
    modes = rng.choice(range(1, n // 2), size=min(num_modes, n // 2 - 1), replace=False)
    for k in modes:
        amp = rng.uniform(-scale, scale)
        phase = rng.uniform(0, 2 * np.pi)
        h_new += amp * np.cos(2 * np.pi * k * np.arange(n) / n + phase)
    return project_feasible(h_new, n)
```
