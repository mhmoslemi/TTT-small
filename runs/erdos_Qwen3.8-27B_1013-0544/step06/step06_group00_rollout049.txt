def compute_grad(h, n, weights):

    grad = np.zeros(n)
    for s in range(-(n-1), n):
        w = weights[s + n - 1]
        if w < 1e-12:
            continue
        # Term 1: (1-h[j+s]) for valid j
        j_lo = max(0, -s)
        j_hi = min(n-1, n-1-s)
        if j_lo <= j_hi:

        if j_lo2 <= j_hi2:
            grad[j_lo2:j_hi2+1] -= w * dx * h[j_lo2-s:j_hi2-s+1]
    return grad

    return np.clip(h, 0.0, 1.0)

def smooth_max_and_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    m = corr.max()
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -500.0, 500.0)
    sm = shifted.max()
    exps = np.exp(shifted - sm)
    logsum = np.log(exps.sum())
    val = m + logsum / alpha
    weights = exps / exps.sum()
    
    grad = np.zeros(n)
    for ki in range(len(corr)):
        w = weights[ki]
        if w < 1e-13:
            continue
        s = ki - (n - 1)
        j1 = max(0, -s)
        j2 = min(n - 1, n - 1 - s)
        if j1 <= j2:
            grad[j1:j2+1] += w * dx * (1.0 - h[j1+s:j2+s+1])
        j3 = max(0, s)
        j4 = min(n - 1, n - 1 + s)
        if j3 <= j4:
            grad[j3:j4+1] -= w * dx * h[j3-s:j4-s+1]
    return val, grad

def lbbgfs_opt(h_init, n, alpha, mu, maxiter):
    h = np.clip(h_init, 0.01, 0.99)
    u0 = np.log(h / (1.0 - h))
    
    def func(u):
        us = np.clip(u, -40.0, 40.0)
        hs = 1.0 / (1.0 + np.exp(-us))
        val, grad_h = smooth_max_and_grad(hs, n, alpha)
        sig_factor = hs * (1.0 - hs)
        grad_u = grad_h * sig_factor
        pen = mu * (hs.sum() - n / 2.0) ** 2
        gpu = 2.0 * mu * (hs.sum() - n / 2.0) * sig_factor
        return val + pen, grad_u + gpu
    
    try:
        res = minimize(func, u0, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12, 'maxls': 30})
        us = np.clip(res.x, -40.0, 40.0)
        return 1.0 / (1.0 + np.exp(-us))
    except Exception:
        return h_init

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def greed_binary(h_init, n, deadline):
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones = [int(x) for x in np.where(h == 1)[0]]
    zeros = [int(x) for x in np.where(h == 0)[0]]
    m_range = np.arange(-(n - 1), n)
    
    for _ in range(300):
        if time.time() > deadline:
            break
        improved = False
        for i in zeros:
            if time.time() > deadline:
                break
            best_j = -1
            best_nc = cur_c
            for j in ones:
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                delta[k2] += 1.0
                nc = (corr + delta).max() * dx
                if nc < best_nc:
                    best_nc = nc
                    best_j = j
            if best_j >= 0 and best_nc < cur_c - 1e-15:
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                msk = (im >= 0) & (im < n)
                delta[msk] += 1.0 - h[im[msk]]
                imn = i - m_range
                msk = (imn >= 0) & (imn < n)
                delta[msk] -= h[imn[msk]]
                jm = best_j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = best_j - m_range
                msk = (jmn >= 0) & (jmn < n)
                delta[msk] += h[jmn[msk]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (best_j - i)
                delta[k1] += 1.0
                k2 = n - 1 + (i - best_j)
                delta[k2] += 1.0
                h[i], h[best_j] = 1.0, 0.0
                corr += delta
                cur_c = best_nc
                ones.remove(best_j)
                zeros.remove(i)
                ones.append(i)
                zeros.append(best_j)
                improved = True
                break
        if not improved:
            break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 10
    dl = t0 + total
    rng = np.random.RandomState(seed)
    
    try:
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)
    except Exception:
        n = 400
        h = np.full(n, 0.5)
    
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    # Phase 1: L-BFGS-B with sigmoid parameterization, increasing alpha
    p1_end = t0 + 0.48 * total
    alphas = [30, 80, 200, 500, 1500, 5000, 15000, 50000, 150000, 500000]
    for alpha in alphas:
        if time.time() > p1_end - 5:
            break
        h_cur = best_h.copy()
        for mu in [500.0, 5000.0]:
            if time.time() > p1_end - 5:
                break
            h_new = lbbgfs_opt(h_cur, n, alpha, mu, maxiter=500)
            h_new = project_feasible(h_new, n)
            c = compute_c5(h_new, n)
            if c < best_c:
                best_c = c
                best_h = h_new.copy()
                h_cur = h_new.copy()
    
    # Phase 2: Binary greedy refinement
    p2_end = t0 + 0.62 * total
    hb, cb = greed_binary(best_h, n, p2_end)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()
    
    # Phase 3: L-BFGS-B from binary solution
    p3_end = t0 + 0.80 * total
    alphas2 = [5000, 20000, 100000, 500000, 2000000, 8000000]
    for alpha in alphas2:
        if time.time() > p3_end - 4:
            break
        h_cur = best_h.copy()
        h_new = lbbgfs_opt(h_cur, n, alpha, 5000.0, maxiter=400)
        h_new = project_feasible(h_new, n)
        c = compute_c5(h_new, n)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
            h_cur = h_new.copy()
    
    # Phase 4: Noise perturbation + re-optimize
    p4_end = t0 + 0.91 * total
    for trial in range(4):
        if time.time() > p4_end - 5:
            break
        h_noise = best_h + 0.008 * rng.standard_normal(n)
        h_noise = project_feasible(h_noise, n)
        for alpha in [20000, 200000, 2000000]:
            if time.time() > p4_end - 3:
                break
            h_new = lbbgfs_opt(h_noise, n, alpha, 5000.0, maxiter=300)
            h_new = project_feasible(h_new, n)
            c = compute_c5(h_new, n)
            if c < best_c:
                best_c = c
                best_h = h_new.copy()
    
    # Phase 5: Try alternate resolution
    if time.time() < dl - 60:
        for nh in [300, 500]:
            if time.time() > dl - 50:
                break
            xo = np.linspace(0, 2, n)
            xn = np.linspace(0, 2, nh)
            hh = np.interp(xn, xo, best_h)
            hh = project_feasible(hh, nh)
            for alpha in [10000, 100000, 1000000]:
                if time.time() > dl - 45:
                    break
                hh = lbbgfs_opt(hh, nh, alpha, 5000.0, maxiter=300)
                hh = project_feasible(hh, nh)
                ch = compute_c5(hh, nh)
                if ch < best_c:
                    best_c, best_h = ch, hh.copy()
                    n = nh
    
    # Final binary polish
    if time.time() < dl - 15:
        hb2, cb2 = greed_binary(best_h, n, dl - 5)
        if cb2 < best_c:
            best_c, best_h = cb2, hb2.copy()
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
