**Fast projection onto {h: 0≤h≤1, sum(h)=n/2}:**

The projection of a vector x onto this set can be done by:
1. Find λ such that sum(clamp(x - λ, 0, 1)) = n/2
2. Return clamp(x - λ, 0, 1)

This is a 1D root-finding problem. I can use bisection (O(log(1/ε)) iterations, each O(n)) or Newton's method.

With bisection: range of λ is [min(x)-1, max(x)], so about 3 units. To get 1e-10 precision, need ~37 iterations. Each is O(n) = O(300). Total: 11,000 operations per projection. With 1 million iterations, that's 11 billion. Too slow.

Better: use the fact that the function f(λ) = sum(clamp(x-λ, 0, 1)) is piecewise linear with at most n+1 pieces. I can find the exact λ in O(n log n) time:
1. Sort x
2. The breakpoints are at x[i] and x[i]-1
3. Binary search for the interval containing the solution

Actually, there's a simpler O(n) approach:
- f(λ) = sum(clamp(x_i - λ, 0, 1)) = sum of (x_i - λ) for x_i > λ, plus 1 for x_i > λ+1... no wait.
- clamp(x_i - λ, 0, 1) = 0 if x_i ≤ λ, x_i - λ if λ < x_i < λ+1, 1 if x_i ≥ λ+1

So f(λ) = sum_{x_i ≥ λ+1} 1 + sum_{λ < x_i < λ+1} (x_i - λ)

This is a piecewise linear function of λ with breakpoints at x_i and x_i - 1.

The O(n) algorithm:
1. Sort x: x_(1) ≤ x_(2) ≤ ... ≤ x_(n)
2. The breakpoints are at x_(i) - 1 and x_(i)
3. Between consecutive breakpoints, f is linear, so I can solve exactly.

Let me also think about another approach: **proximal gradient with Bregman divergence**.

The mirror descent update for minimizing f(h) subject to h ∈ [0,1]^n, sum(h) = S is:
h_new = argmin_{h' ∈ feasible} <g, h' - h> + (1/(2t)) * KL(h' || h)

where KL(h' || h) = sum h' * log(h'/h) (with the convention 0*log(0/h)=0).

The solution is: h'_i ∝ h_i * exp(-g_i * t), then projected onto {sum = S, 0 ≤ h ≤ 1}.

This is equivalent to:
h'_i = h_i * exp(-t * g_i) * normalizing_factor

The normalizing factor ensures sum = S. This can be found by bisection.

This is the "exponential weights" or "multiplicative weights" update. It naturally keeps h in [0,1] (if h is in (0,1) and t is small enough).

```python
def grad_from_weights(h, n, w):
    """Compute gradient given weight vector w over lags (length 2n-1)."""
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wk = w[idx]
        if wk < 1e-15:
            continue
        k = idx - (n - 1)
        if k == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif k > 0:
            grad[0:n-k] += wk * (1.0 - h[k:n])
            grad[k:n] += wk * (-h[0:n-k])
        else:
            kk = -k
            grad[kk:n] += wk * (1.0 - h[0:n-kk])
            grad[0:n-kk] += wk * (-h[kk:n])
    return grad * dx

    c_shift = alpha * (c - c.max())
    w = np.exp(c_shift - c_shift.max())
    w /= w.sum()
    return grad_from_weights(h, n, w), c

        return smooth_max_obj(x, n, alpha)

        g, _ = smooth_max_grad(x, n, alpha)
        return g
```

```python
def fista(h, n, deadline, alpha=10000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    x = h.copy()
    y = h.copy()
    t_k = 1.0
    L = 1.0  # Lipschitz constant estimate (upper bound on ||Hessian||)
    
    for it in range(100000):
        if time.time() > deadline:
            break
        g, c = smooth_max_grad(y, n, alpha)
        # Gradient step
        x_new = y - g / L
        # Proximal step (project onto feasible set)
        x_new = project_feasible(x_new, n)
        
        c_new = compute_c5(x_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = x_new.copy()
        
        # Backtracking if needed
        if c_new > compute_c5(y, n) + 1e-6:
            L *= 2
            continue
        
        # FISTA update
        t_new = (1 + np.sqrt(1 + 4*t_k*t_k)) / 2
        y = x_new + (t_k - 1) / t_new * (x_new - x)
        y = project_feasible(y, n)
        x = x_new
        t_k = t_new
        L = max(L / 1.05, 0.1)  # slowly decrease L
    
    return best_h, best_c
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    for _ in range(80):
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = mask.sum()
            if cnt == 0:
                break
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def grad_from_weights(h, n, w):
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wk = w[idx]
        if wk < 1e-15:
            continue
        k = idx - (n - 1)
        if k == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif k > 0:
            grad[0:n-k] += wk * (1.0 - h[k:n])
            grad[k:n] += wk * (-h[0:n-k])
        else:
            kk = -k
            grad[kk:n] += wk * (1.0 - h[0:n-kk])
            grad[0:n-kk] += wk * (-h[kk:n])
    return grad * dx

def smooth_max_grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    c_max_shift = c_shift.max()
    w = np.exp(c_shift - c_max_shift)
    w /= w.sum()
    return grad_from_weights(h, n, w)

def smooth_max_obj(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    c_shift = alpha * (c - m)
    mx = c_shift.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(c_shift - mx)))) / alpha
    return m + np.log(np.sum(np.exp(c_shift))) / alpha

def top_p_subgrad(h, n, deadline, p=3, alpha=200.0, lr_init=0.003, momentum=0.85):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    v = np.zeros(n)
    lr = lr_init
    stagnant = 0
    for it in range(300000):
        if time.time() > deadline:
            break
        c = compute_c5_vec(h, n)
        if p < 2 * n - 1:
            top_idx = np.argpartition(c, -p)[-p:]
        else:
            top_idx = np.arange(2 * n - 1)
        c_top = c[top_idx]
        c_shift = alpha * (c_top - c_top.max())
        w_top = np.exp(c_shift - c_shift.max())
        w_top /= w_top.sum()
        w = np.zeros(2 * n - 1)
        w[top_idx] = w_top
        grad = grad_from_weights(h, n, w)
        grad -= grad.mean()
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-14:
            break
        grad = grad / g_norm
        v = momentum * v - lr * grad
        h_new = project_feasible(h + v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            stagnant = 0
            lr = min(lr * 1.01, 0.04)
        else:
            h = h_new
            stagnant += 1
            lr = max(lr * 0.97, 1e-6)
            if stagnant > 300:
                v *= 0.3
                alpha = min(alpha * 2.0, 200000.0)
                stagnant = 0
                if lr < 1e-7:
                    break
    return best_h, best_c

def grad_desc_accel(h, n, deadline, alpha=5000, lr_init=0.008):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    x = h.copy()
    y = h.copy()
    t_k = 1.0
    lr = lr_init
    for it in range(200000):
        if time.time() > deadline:
            break
        g = smooth_max_grad(y, n, alpha)
        g -= g.mean()
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-13:
            break
        g = g / g_norm
        x_new = project_feasible(y - lr * g, n)
        c_new = compute_c5(x_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = x_new.copy()
        c_y = compute_c5(y, n)
        if c_new > c_y + lr * 0.1:
            lr *= 0.7
        else:
            lr = min(lr * 1.02, 0.05)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = x_new + (t_k - 1.0) / t_new * (x_new - x)
        y = project_feasible(y, n)
        x = x_new
        t_k = t_new
    return best_h, best_c

def lbfgs_phase(h, n, deadline, alphas=None, max_iters=80):
    if alphas is None:
        alphas = [100, 1000, 10000, 50000, 200000]
    h_cur = h.copy()
    best_c = compute_c5(h_cur, n)
    bounds = [(0.0, 1.0)] * n
    for a in alphas:
        if time.time() > deadline - 5:
            break
        def obj(x):
            return smooth_max_obj(x, n, a)
        def grad(x):
            return smooth_max_grad(x, n, a)
        try:
            res = minimize(obj, h_cur, jac=grad, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': max_iters, 'ftol': 1e-16,
                                    'gtol': 1e-13, 'maxfun': max_iters * 12,
                                    'maxls': 40})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h_cur = h_new
        except Exception:
            pass
    return h_cur, best_c

def multiplicative_subgrad(h, n, deadline, p=3, alpha=200.0, t_init=0.15):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    t = t_init
    stagnant = 0
    for it in range(200000):
        if time.time() > deadline:
            break
        c = compute_c5_vec(h, n)
        top_idx = np.argpartition(c, -p)[-p:] if p < 2 * n - 1 else np.arange(2 * n - 1)
        c_top = c[top_idx]
        c_shift = alpha * (c_top - c_top.max())
        w_top = np.exp(c_shift - c_shift.max())
        w_top /= w_top.sum()
        w = np.zeros(2 * n - 1)
        w[top_idx] = w_top
        grad = grad_from_weights(h, n, w)
        grad -= grad.mean()
        log_h = np.where(h > 1e-12, np.log(np.maximum(h, 1e-12)), -10.0)
        log_h_new = log_h - t * grad
        log_h_new -= log_h_new.max()
        h_raw = np.exp(log_h_new)
        h_new = project_feasible(h_raw / h_raw.sum() * (n / 2.0), n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            stagnant = 0
            t = min(t * 1.01, 1.0)
        else:
            h = h_new
            stagnant += 1
            if stagnant > 50:
                t *= 0.7
                stagnant = 0
            if t < 1e-4:
                break
    return best_h, best_c
```
