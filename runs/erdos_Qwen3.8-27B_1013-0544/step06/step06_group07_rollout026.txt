import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-12: break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def round_to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    return out

def swap_delta(h, n, i, j, m_range):
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    if 0 <= k1 < 2*n-1: delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    if 0 <= k2 < 2*n-1: delta[k2] += 1.0
    return delta

def slsqp_smooth(h, n, alpha, maxiter=500):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_feasible(res.x, n)
    except Exception:
        return h

def adaptive_sa(h_init, n, deadline, seed=42, t_init=0.05, patience=50000):
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    m_range = np.arange(-(n-1), n)
    h = h_init.copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    T = t_init
    no_improve = 0
    accept_count = 0
    total_count = 0
    kick_count = 0
    
    while time.time() < deadline:
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        delta = swap_delta(h, n, i, j, m_range)
        new_corr = corr + delta
        nc = new_corr.max() * dx
        dc = nc - cur_c
        
        if dc < 0 or (T > 1e-12 and rng.random() < np.exp(-dc / T)):
            h[i] = 1.0
            h[j] = 0.0
            corr = new_corr
            cur_c = nc
            accept_count += 1
            if nc < best_c - 1e-15:
                best_c = nc
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        total_count += 1
        
        # Adaptive temperature
        if total_count % 5000 == 0:
            rate = accept_count / max(total_count, 1)
            if rate > 0.5:
                T *= 0.8
            elif rate < 0.1:
                T *= 1.2
            accept_count = 0
            total_count = 0
        
        # Kick: random perturbation when stuck
        if no_improve > patience:
            nk = 3 + rng.randint(5)
            for _ in range(nk):
                zs = np.where(h == 0)[0]
                os_ = np.where(h == 1)[0]
                ii = zs[rng.randint(len(zs))]
                jj = os_[rng.randint(len(os_))]
                h[ii] = 1.0
                h[jj] = 0.0
            corr = np.correlate(h, 1 - h, mode='full').astype(float)
            cur_c = corr.max() * dx
            no_improve = 0
            T = min(t_init, T * 2.0)
            kick_count += 1
            if kick_count > 30:
                break
    
    return best_h, best_c

def local_search(h_init, n, deadline):
    dx = 2.0 / n
    m_range = np.arange(-(n-1), n)
    h = h_init.copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    
    for _ in range(50):
        if time.time() > deadline:
            break
        improved = False
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        for i in zeros:
            if h[i] != 0: continue  # may have been changed
            if time.time() > deadline:
                corr = np.correlate(h, 1 - h, mode='full').astype(float)
                cur_c = corr.max() * dx
                if cur_c < best_c: best_c, best_h = cur_c, h.copy()
                return best_h, best_c
            best_j = -1
            best_nc = cur_c
            best_delta = None
            for j in ones:

                d = swap_delta(h, n, i, j, m_range)
                nc = (corr + d).max() * dx
                if nc < best_nc - 1e-15:
                    best_nc = nc
                    best_j = j
                    best_delta = d
            if best_j >= 0:
                h[i] = 1.0
                h[best_j] = 0.0
                corr += best_delta
                cur_c = best_nc
                improved = True
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
        if not improved:
            break
    
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    if cur_c < best_c: best_c, best_h = cur_c, h.copy()
    return best_h, best_c

def precompute_swap_components(h, n):
    """Precompute per-position contributions for fast swap evaluation."""
    m_range = np.arange(-(n-1), n)
    A = np.zeros((n, 2*n-1))  # effect of setting h[i] from 0 to 1
    B = np.zeros((n, 2*n-1))  # effect of setting h[j] from 1 to 0
    
    for i in range(n):
        im = i + m_range
        msk = (im >= 0) & (im < n)
        A[i, msk] += 1.0 - h[im[msk]]
        imn = i - m_range
        msk = (imn >= 0) & (imn < n)
        A[i, msk] -= h[imn[msk]]
    
    for j in range(n):
        jm = j + m_range
        msk = (jm >= 0) & (jm < n)
        B[j, msk] -= 1.0 - h[jm[msk]]
        jmn = j - m_range
        msk = (jmn >= 0) & (jmn < n)
        B[j, msk] += h[jmn[msk]]
    
    const = np.zeros(2*n-1)
    const[n-1] = -2.0
    
    return A, B, const

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def round_to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    return out

def swap_delta(h, n, i, j, m_range):
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    if 0 <= k1 < 2 * n - 1:
        delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    if 0 <= k2 < 2 * n - 1:
        delta[k2] += 1.0
    return delta

def slsqp_smooth(h, n, alpha, maxiter=300):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_feasible(res.x, n)
    except Exception:
        return h

def adaptive_sa(h_init, n, deadline, seed=42, t_init=0.04, patience=40000):
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    m_range = np.arange(-(n - 1), n)
    h = h_init.copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    T = t_init
    no_improve = 0
    accept_n = 0
    total_n = 0
    kicks = 0
    while time.time() < deadline:
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        delta = swap_delta(h, n, i, j, m_range)
        new_corr = corr + delta
        nc = new_corr.max() * dx
        dc = nc - cur_c
        if dc < 0 or (T > 1e-12 and rng.random() < np.exp(-dc / T)):
            h[i] = 1.0
            h[j] = 0.0
            corr = new_corr
            cur_c = nc
            accept_n += 1
            if nc < best_c - 1e-15:
                best_c = nc
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        total_n += 1
        if total_n % 4000 == 0:
            rate = accept_n / max(total_n, 1)
            if rate > 0.45:
                T *= 0.75
            elif rate < 0.15:
                T = min(t_init * 0.8, T * 1.3)
            accept_n = 0
            total_n = 0
        if no_improve > patience:
            nk = 3 + rng.randint(5)
            for _ in range(nk):
                zs = np.where(h == 0)[0]
                os_ = np.where(h == 1)[0]
                h[zs[rng.randint(len(zs))]] = 1.0
                h[os_[rng.randint(len(os_))]] = 0.0
            corr = np.correlate(h, 1 - h, mode='full').astype(float)
            cur_c = corr.max() * dx
            no_improve = 0
            T = min(t_init * 0.7, T * 2.5)
            kicks += 1
            if kicks > 50:
                break
    return best_h, best_c

def local_search(h_init, n, deadline):
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    h = h_init.copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    for _ in range(200):
        if time.time() > deadline:
            break
        improved = False
        zeros = list(np.where(h == 0)[0])
        ones = list(np.where(h == 1)[0])
        for i in zeros:
            if h[i] != 0:
                continue
            if time.time() > deadline:
                break
            best_j = -1
            best_nc = cur_c
            best_d = None
            for j in ones:
                if h[j] != 1:
                    continue
                d = swap_delta(h, n, i, j, m_range)
                nc = (corr + d).max() * dx
                if nc < best_nc - 1e-14:
                    best_nc = nc
                    best_j = j
                    best_d = d
            if best_j >= 0:
                h[i] = 1.0
                h[best_j] = 0.0
                corr += best_d
                cur_c = best_nc
                improved = True
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
        if not improved:
            break
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    if cur_c < best_c:
        best_c = cur_c
        best_h = h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
