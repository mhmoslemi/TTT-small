import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def thue_morse_init(n, seed):
    rng = np.random.RandomState(seed)
    t = np.zeros(n, dtype=int)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    target = n // 2
    num_ones = int(t.sum())
    if num_ones > target:
        ones = np.where(t == 1)[0]
        rng.shuffle(ones)
        t[ones[:num_ones - target]] = 0
    elif num_ones < target:
        zeros = np.where(t == 0)[0]
        rng.shuffle(zeros)
        t[zeros[:target - num_ones]] = 1
    return t.astype(float)

def mseq_init(n, seed):
    rng = np.random.RandomState(seed)
    m = int(np.log2(n)) + 1
    length = 2**m - 1
    state = rng.randint(1, 2**m, dtype=int)
    taps = {4: (1,4), 5: (1,5), 6: (1,6), 7: (1,7), 8: (4,8), 9: (1,9), 10: (3,10), 11: (6,11)}
    if m not in taps:
        m = 8
    t1, t2 = taps[m]
    seq = np.zeros(length, dtype=int)
    for i in range(length):
        seq[i] = state & 1
        fb = ((state >> (t1-1)) ^ (state >> (t2-1))) & 1
        state = ((state >> 1) | (fb << (m-1))) & ((1 << m) - 1)
    seq = seq[:n]
    target = n // 2
    num_ones = int(seq.sum())
    if num_ones > target:
        oi = np.where(seq == 1)[0]
        rng.shuffle(oi)
        seq[oi[:num_ones - target]] = 0
    elif num_ones < target:
        zi = np.where(seq == 0)[0]
        rng.shuffle(zi)
        seq[zi[:target - num_ones]] = 1
    return seq.astype(float)

def periodic_init(n, period, duty, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    on = max(1, int(period * duty))
    for i in range(n):
        if (i % period) < on:
            h[i] = 1.0
    target = n // 2
    num_ones = int(h.sum())
    if num_ones > target:
        oi = np.where(h == 1)[0]
        rng.shuffle(oi)
        h[oi[:num_ones - target]] = 0
    elif num_ones < target:
        zi = np.where(h == 0)[0]
        rng.shuffle(zi)
        h[zi[:target - num_ones]] = 1
    return h

def sine_init(n, freq, phase, seed):
    rng = np.random.RandomState(seed)
    x = np.linspace(0, 2 * np.pi * freq, n)
    h = (np.sin(x + phase) > 0).astype(float)
    target = n // 2
    num_ones = int(h.sum())
    if num_ones > target:
        vals = np.abs(np.sin(x + phase))
        oi = np.where(h == 1)[0]
        order = vals[oi].argsort()
        h[oi[order[:num_ones - target]]] = 0
    elif num_ones < target:
        vals = np.abs(np.sin(x + phase))
        zi = np.where(h == 0)[0]
        order = vals[zi].argsort()
        h[zi[order[:target - num_ones]]] = 1
    return h

def fast_local_search(h0, n, deadline, seed, top_k=10):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    target = n // 2
    h = binarize(h0, n)
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = float(corr.max() * dx)
    best_c = cur_c
    best_h = h.copy()
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    stall = 0
    no_improve = 0
    temp = 0.003

    while time.time() < deadline:
        top_idx = np.argsort(corr)[-top_k:]
        na = len(top_idx)
        nz = len(zeros)
        no = len(ones)
        if nz == 0 or no == 0:
            break
        B = np.zeros((nz, na))
        A = np.zeros((no, na))
        for a in range(na):
            k = int(top_idx[a])
            ip = zeros + k
            im = zeros - k
            B[:, a] = (np.where((ip >= 0) & (ip < n), h[np.clip(ip, 0, n-1)], 0.0) +
                        np.where((im >= 0) & (im < n), h[np.clip(im, 0, n-1)], 0.0))
            jp = ones + k
            jm = ones - k
            A[:, a] = (np.where((jp >= 0) & (jp < n), h[np.clip(jp, 0, n-1)], 0.0) +
                        np.where((jm >= 0) & (jm < n), h[np.clip(jm, 0, n-1)], 0.0))
        c_act = corr[top_idx]
        new_val = c_act[None, None, :] + A[None, :, :] - B[:, None, :]
        new_max = new_val.max(axis=2)
        flat = int(new_max.argmin())
        bi = flat // no
        bj = flat % no
        best_val = float(new_max[bi, bj] * dx)
        i_pos = int(zeros[bi])
        j_pos = int(ones[bj])

        if best_val < cur_c - 1e-14:
            h[i_pos] = 1.0
            h[j_pos] = 0.0
            corr = np.correlate(h, 1 - h, mode='full').astype(float)
            cur_c = float(corr.max() * dx)
            ones = np.where(h == 1)[0].copy()
            zeros = np.where(h == 0)[0].copy()
            stall = 0
            temp = max(0.0005, temp * 0.999)
            if cur_c < best_c - 1e-14:
                best_c = cur_c
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            stall += 1
            no_improve += 1
            if stall > 15 or (no_improve > 100 and stall > 5):
                nk = min(2, len(zeros), len(ones))
                ri = rng.choice(len(zeros), nk, replace=False)
                rj = rng.choice(len(ones), nk, replace=False)
                for a in range(nk):
                    h[zeros[ri[a]]] = 1.0
                    h[ones[rj[a]]] = 0.0
                corr = np.correlate(h, 1 - h, mode='full').astype(float)
                cur_c = float(corr.max() * dx)
                ones = np.where(h == 1)[0].copy()
                zeros = np.where(h == 0)[0].copy()
                stall = 0
                temp = max(0.0005, temp * 0.85)

    return best_h, best_c

def slsqp_polish(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h
