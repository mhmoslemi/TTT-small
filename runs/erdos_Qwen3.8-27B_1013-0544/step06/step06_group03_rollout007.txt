import time
import numpy as np
from scipy.optimize import linprog

def project_box_sum(x, target):
    y = np.clip(np.asarray(x, dtype=float), 0.0, 1.0)
    t = float(target)
    for _ in range(200):
        s = float(y.sum())
        d = t - s
        if abs(d) < 1e-15:
            break
        if d > 0:
            r = 1.0 - y
            R = float(r.sum())
            if R <= 1e-15:
                break
            y += (d / R) * r
        elif s > 1e-15:
            y += (d / s) * y
    y = np.clip(y, 0.0, 1.0)
    d = t - float(y.sum())
    if abs(d) > 0:
        if d > 0:
            r = 1.0 - y
            R = float(r.sum())
            if R > 0:
                y += d * r / R
        elif y.sum() > 0:
            y += d * y / y.sum()
    return np.clip(y, 0.0, 1.0)

def compute_c_all(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / float(n))

def compute_c_max(h, n):
    return float(np.max(compute_c_all(h, n)))

def resample_h(h, n2):
    h = np.asarray(h, float).ravel()
    n2 = int(n2)
    if h.size == n2:
        return h.copy()
    xo = np.linspace(0.0, 2.0, h.size)
    xn = np.linspace(0.0, 2.0, n2)
    return np.interp(xn, xo, h)

def _compute_lag_gradient(h, n, lag):
    grad = np.zeros(n)
    m = abs(int(lag))
    if m >= n:
        return grad
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if lag > 0:
        grad[:n - m] -= h[m:]
        grad[m:] += 1.0 - h[:n - m]
    else:
        grad[:n - m] += 1.0 - h[m:]
        grad[m:] -= h[:n - m]
    return grad

def select_indices(c, max_count, slack):
    order = np.argsort(c)[::-1]
    if order.size == 0:
        return order
    cm = float(c[order[0]])
    idx = order[c[order] >= cm - float(slack)]
    if idx.size < 5:
        idx = order[:5]
    if idx.size > max_count:
        idx = order[:max_count]
    return idx

def softmax_weights(vals, alpha):
    v = np.asarray(vals, float)
    if v.size == 0:
        return v
    v = v - float(v.max())
    e = np.exp(float(alpha) * v)
    s = float(e.sum())
    if s <= 0 or not np.isfinite(s):
        return np.full(v.size, 1.0 / v.size)
    return e / s

def best_transfer_exact_all(h, c_all, a, b, n, dmax):
    nlag = 2 * n - 1
    if dmax <= 1e-14:
        L = np.zeros(nlag)
        Q = np.zeros(nlag)
        return 0.0, float(np.max(c_all)), L, Q
    oa = float(h[a])
    ob = float(h[b])
    dx = 2.0 / float(n)
    L = np.zeros(nlag)
    Q = np.zeros(nlag)
    for idx in range(nlag):
        lag = idx - (n - 1)
        q = abs(lag)
        if q == 0:
            L[idx] = dx * (2.0 * oa - 1.0 + 1.0 - 2.0 * ob)
            Q[idx] = -2.0 * dx
            continue
        if not ((a >= q or a + q < n) or (b >= q or b + q < n)):
            continue
        pairs = []
        if lag > 0:
            if a >= q:
                pairs.append((a, a - q))
            if a + q < n:
                pairs.append((a + q, a))
            if b >= q:
                pairs.append((b, b - q))
            if b + q < n:
                pairs.append((b + q, b))
        else:
            if a + q < n:
                pairs.append((a, a + q))
            if a >= q:
                pairs.append((a - q, a))
            if b + q < n:
                pairs.append((b, b + q))
            if b >= q:
                pairs.append((b - q, b))
        ls = 0.0
        qs = 0.0
        seen = []
        for i, j in pairs:
            if (i, j) in seen:
                continue
            seen.append((i, j))
            si = -1.0 if i == a else (1.0 if i == b else 0.0)
            sj = -1.0 if j == a else (1.0 if j == b else 0.0)
            ui = oa if i == a else (ob if i == b else float(h[i]))
            uj = oa if j == a else (ob if j == b else float(h[j]))
            ls += si * (1.0 - uj) - ui * sj
            qs += -si * sj
        L[idx] = dx * ls
        Q[idx] = dx * qs
    Ds = np.linspace(0.0, 1.0, 41) * float(dmax)
    vals = c_all[None, :] + Ds[:, None] * L[None, :] + (Ds[:, None] * Ds[:, None]) * Q[None, :]
    rm = vals.max(axis=1)
    bi = int(np.argmin(rm))
    return float(Ds[bi]), float(rm[bi]), L, Q

def local_two_move_refine(h, n, deadline, rng, max_step=0.25, max_cands=None):
    tgt = n / 2.0
    h = project_box_sum(h, n, tgt)
    c_all = compute_c_all(h, n)
    C = float(c_all.max())
    bh = h.copy()
    bc = C
    if max_cands is None:
        max_cands = 35 if n > 450 else 70
    step = float(max_step)
    stale = 0
    misses = 0
    acc = 0
    while time.time() < deadline and step > 1e-6:
        C = float(c_all.max())
        if C < bc - 1e-14:
            bc = C
            bh = h.copy()
        dx = 2.0 / n
        order = np.argsort(c_all)[::-1]
        tn = min(25, order.size)
        top = order[:tn]
        w = softmax_weights(c_all[top], 40.0)
        G = np.zeros(n)
        for jj in range(tn):
            ww = float(w[jj])
            if ww > 1e-13:
                G += ww * _compute_lag_gradient(h, n, int(top[jj] - (n - 1))) * dx
        hc = 25 if n > 450 else 35
        rc = 12 if n > 450 else 25
        high = np.argsort(G)[::-1][:hc]
        low = np.argsort(G)[:hc]
        cands = []
        for a in high:
            if h[a] <= 1e-9:
                continue
            for b in low:
                cap = 1.0 - h[b]
                if b == a or cap <= 1e-9:
                    continue
                dm = min(float(h[a]), float(cap), step)
                if dm > 1e-9:
                    cands.append((float(G[a] - G[b]), int(a), int(b), float(dm)))
        src = np.where(h > 1e-7)[0]
        dst = np.where(h < 1.0 - 1e-7)[0]
        if src.size and dst.size:
            for _ in range(rc):
                a = int(src[rng.randint(src.size)])
                b = int(dst[rng.randint(dst.size)])
                if a == b:
                    continue
                dm = min(float(h[a]), 1.0 - float(h[b]), step)
                if dm > 1e-9:
                    cands.append((float(G[a] - G[b]), int(a), int(b), float(dm)))
        if not cands:
            break
        cands.sort(reverse=True)
        best = None
        bv = C
        bd = 0.0
        bL = None
        bQ = None
        ev = 0
        for _score, a, b, dm in cands:
            if ev >= max_cands:
                break
            d, val, L, Q = best_transfer_exact_all(h, c_all, a, b, n, dm)
            ev += 1
            if d > 1e-12 and val < bv - 1e-13:
                bv = val
                bd = d
                bL = L
                bQ = Q
                best = (a, b)
        if best is not None:
            a, b = best
            h[a] -= bd
            h[b] += bd
            clamped = False
            if h[a] < 0.0:
                h[a] = 0.0
                clamped = True
            if h[a] > 1.0:
                h[a] = 1.0
                clamped = True
            if h[b] < 0.0:
                h[b] = 0.0
                clamped = True
            if h[b] > 1.0:
                h[b] = 1.0
                clamped = True
            c_all = c_all + bd * bL + bd * bd * bQ
            acc += 1
            if clamped or acc % 12 == 0:
                c_all = compute_c_all(h, n)
            C = float(c_all.max())
            if C > bv + 1e-7:
                c_all = compute_c_all(h, n)
                C = float(c_all.max())
            stale = 0
            misses = 0
            if bd <= 1e-9:
                step *= 0.7
        else:
            misses += 1
            stale += 1
            if misses >= 12:
                break
            if stale >= 3:
                step *= 0.6
                stale = 0
    h = project_box_sum(h, n, tgt)
    C = compute_c_max(h, n)
    if C < bc:
        bc = C
        bh = h.copy()
    return bh, bc

def lp_direction(h, n, sel, c_all, trust):
    dx = 2.0 / float(n)
    r = sel.size
    if r == 0:
        return None
    A = np.zeros((r, n + 1))
    B = np.empty(r)
    for rr, idx in enumerate(sel):
        g = _compute_lag_gradient(h, n, int(idx - (n - 1))) * dx
        B[rr] = float(np.dot(g, h) - c_all[idx])
        A[rr, :n] = g
        A[rr, n] = -1.0
    f = np.zeros(n + 1)
    f[n] = 1.0
    Aeq = np.ones((1, n + 1))
    Aeq[0, n] = 0.0
    lb = np.maximum(0.0, h - float(trust))
    ub = np.minimum(1.0, h + float(trust))
    bounds = list(zip(lb, ub)) + [(None, None)]
    try:
        res = linprog(f, A_ub=A, b_ub=B, A_eq=Aeq, b_eq=[n / 2.0],
                      bounds=bounds, method='highs')
        if res.x is not None and float(res.fun) < 1e60:
            return np.asarray(res.x[:n], float)
    except Exception:
        pass
    return None

def active_lp_refine(h, n, deadline):
    tgt = n / 2.0
    h = project_box_sum(h, n, tgt)
    C = compute_c_max(h, n)
    bh = h.copy()
    bc = C
    trust = 0.15
    stale = 0
    max_count = 40 if n > 450 else 70
    fracs = (1.0, 0.8, 0.6, 0.45, 0.3, 0.2, 0.12, 0.06, 0.03)
    for _ in range(80):
        if time.time() > deadline - 2:
            break
        c_all = compute_c_all(h, n)
        C = float(c_all.max())
        if C < bc:
            bc = C
            bh = h.copy()
        sel = select_indices(c_all, max_count, max(0.0015, 0.01 * C))
        y = lp_direction(h, n, sel, c_all, trust)
        if y is None:
            trust *= 0.7
            stale += 1
            if stale > 6 or trust < 1e-4:
                break
            continue
        y = project_box_sum(y, n, tgt)
        d = y - h
        if np.max(np.abs(d)) <= 1e-10:
            break
        ok = False
        for s in fracs:
            hs = project_box_sum(h + s * d, n, tgt)
            cs = compute_c_max(hs, n)
            if cs < C - 1e-12:
                h = hs
                C = cs
                ok = True
                if C < bc:
                    bc = C
                    bh = h.copy()
                break
        if ok:
            trust = max(0.002, trust * 0.8)
            stale = 0
        else:
            trust = max(0.001, trust * 0.65)
            stale += 1
            if stale > 8 or trust < 1e-4:
                break
    h = project_box_sum(h, n, tgt)
    C = compute_c_max(h, n)
    if C < bc:
        bc = C
        bh = h.copy()
    return bh, bc

def optimize_at_n(h0, n, deadline, rng, use_lp=True):
    tgt = n / 2.0
    h = project_box_sum(h0, n, tgt)
    bh = h.copy()
    bc = compute_c_max(h, n)
    rem = deadline - time.time()
    if rem > 30:
        d1 = min(deadline - 30, time.time() + max(10, rem * 0.45))
        h1, c1 = local_two_move_refine(h, n, d1, rng, max_step=0.25)
        if c1 < bc:
            bc = c1
            bh = h1
    rem = deadline - time.time()
    if use_lp and n <= 520 and rem > 25:
        d2 = min(deadline - 20, time.time() + max(8, rem * 0.35))
        h2, c2 = active_lp_refine(bh, n, d2)
        if c2 < bc:
            bc = c2
            bh = h2
    rem = deadline - time.time()
    if rem > 25:
        d3 = min(deadline - 10, time.time() + max(8, rem * 0.30))
        h3, c3 = local_two_move_refine(bh, n, d3, rng, max_step=0.06)
        if c3 < bc:
            bc = c3
            bh = h3
    h = project_box_sum(bh, n, tgt)
    c = compute_c_max(h, n)
    if c < bc:
        bc = c
        bh = h
    return bh, bc

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + float(budget_s) - 5.0
    rng = np.random.RandomState(seed)
    h_init = None
    try:
        h_init = np.asarray(initial_h_values, dtype=float).ravel().copy()
        if h_init.size < 20:
            raise ValueError
    except Exception:
        h_init = None
    if h_init is None:
        n_init = 300
        h_init = project_box_sum(np.full(n_init, 0.5), n_init, n_init / 2.0)
    else:
        n_init = int(h_init.size)
    bh = project_box_sum(h_init, n_init, n_init / 2.0)
    bc = compute_c_max(bh, n_init)
    rem = deadline - t0
    if rem > 80:
        d1 = t0 + min(600, rem * 0.5)
        h1, c1 = optimize_at_n(bh, n_init, d1, rng, use_lp=True)
        if c1 < bc:
            bc = c1
            bh = h1
    sec_budget = min(420, max(0, (deadline - t0) * 0.35))
    sec_spent = 0
    base = len(bh)
    sug = [int(round(base * f)) for f in (0.5, 0.75, 1.25, 1.5)] + [120, 180, 240, 360, 480, 600]
    secondary = []
    for nt in sug:
        if nt % 2:
            nt += 1
        if 80 <= nt <= 800:
            secondary.append(nt)
    secondary = sorted(set(secondary))
    tried = {base}
    for nt in secondary:
        if nt in tried:
            continue
        if time.time() > deadline - 100:
            break
        avail = sec_budget - sec_spent - 70
        if avail < 40:
            break
        alloc = min(160, avail * 0.35)
        if alloc < 35:
            break
        h0 = resample_h(bh, nt)
        h1, c1 = optimize_at_n(h0, nt, time.time() + alloc, rng, use_lp=(nt <= 520))
        if c1 < bc:
            bc = c1
            bh = h1
        sec_spent += alloc
        tried.add(nt)
    if time.time() < deadline - 40:
        n = len(bh)
        hp = project_box_sum(bh + rng.normal(0, 0.01, n), n, n / 2.0)
        alloc = min(90, max(20, (deadline - time.time() - 30) * 0.4))
        if alloc > 20:
            h1, c1 = optimize_at_n(hp, n, time.time() + alloc, rng, use_lp=False)
            if c1 < bc:
                bc = c1
                bh = h1
    if time.time() < deadline - 3:
        n = len(bh)
        h1, c1 = local_two_move_refine(bh, n, min(deadline - 1, time.time() + 110),
                                       rng, max_step=0.03, max_cands=40)
        if c1 < bc:
            bc = c1
            bh = h1
    bh = project_box_sum(bh, len(bh), len(bh) / 2.0)
    bc = compute_c_max(bh, len(bh))
    return bh, float(bc), int(len(bh))
