import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

    """Coordinate descent directly minimizing max correlation."""
    dx = 2.0 / n
    h = project_feasible(h, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    best_c = corr.max() * dx
    best_h = h.copy()
    
    for sweep in range(n_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for i in range(n):
            t0 = h[i]
            m_lo = max(0, i - n + 1)
            m_hi = min(2 * n - 2, i)
            if m_lo >= m_hi:
                continue
            m_idx = np.arange(m_lo, m_hi + 1)
            cvals = h[m_idx + n - 1 - i]
            slopes = 1.0 - 2.0 * cvals
            intercepts = corr[m_idx] - t0 * slopes
            # Unaffected max
            mask = np.ones(2 * n - 1, dtype=bool)
            mask[m_lo:m_hi + 1] = False
            m_unaff = corr[mask].max()
            
            # Current max of affected
            m_aff = np.max(corr[m_idx])
            if m_unaff >= m_aff - 1e-14:

            for _ in range(30):

                t2 = hi - (hi - lo) / 3.0
                f1 = max(m_unaff, np.max(intercepts + t1 * slopes))
                f2 = max(m_unaff, np.max(intercepts + t2 * slopes))

            f_new = max(m_unaff, np.max(intercepts + t_new * slopes))

                im = i + m_range
                msk = (im >= 0) & (im < n)

                imn = i - m_range

                delta[msk] -= h[imn[msk]]

                delta[msk] += h[jmn[msk]]

def compute_c5(h, n):

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        logsum = sm + np.log(np.exp(shifted - sm).sum())
        return m + logsum / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
