            if s < 1e-14:

    m = c.max()
    sh = alpha * (c - m) - alpha * c.max() + alpha * m

    sh = alpha * (c - m)
    sh = sh - sh.max()
    return m + np.log(np.exp(sh).sum()) / alpha + pen * (h.sum() - n/2.0)**2

def lse_grad(h, n, alpha, pen):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    sh = sh - sh.max()
    w = np.exp(sh)
    w /= w.sum()
    grad = np.zeros(n)
    shifts = np.arange(2*n-1) - (n-1)
    for j in range(n):
        vp = (j + shifts >= 0) & (j + shifts < n)
        vm = (j - shifts >= 0) & (j - shifts < n)
        s1 = 0.0
        if vp.any():

        s2 = 0.0
        if vm.any():

        grad[j] = dx * (s1 - s2)
    grad += 2.0 * pen * (h.sum() - n/2.0)

        h = lbfgs_run(h, n, a, 100.0, 400, d3 - 3)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 4: anneal for better binary base then re-smooth
    d4 = t0 + 0.70 * (dl - t0)
    ha, ca = anneal_bin(best_h, n, d4, seed=seed)
    if ca < best_c:
        best_c, best_h = ca, ha.copy()
    h = best_h.copy()
    for a in [5000, 50000, 500000, 5000000]:
        if time.time() > d4 + 0.10 * (dl - t0) - 3:
            break
        h = lbfgs_run(h, n, a, 100.0, 300, dl - 0.15 * (dl - t0) - 5)
        h, c = mgd(h, n, a, 100.0, 1000, 0.02, 0.9, dl - 0.15 * (dl - t0) - 2)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 5: multi-restart perturbation
    d5 = dl - 5
    for trial in range(4):
        if time.time() > d5 - 5:
            break
        h = project_feasible(best_h + 0.008 * rng.randn(n), n)
        for a in [20000, 200000, 2000000]:
            if time.time() > d5 - 3:
                break
            h = lbfgs_run(h, n, a, 200.0, 200, d5 - 2)
            c = compute_c5(h, n)
            if c < best_c:
                best_c, best_h = c, h.copy()

    # Phase 6: final high-alpha polish
    if time.time() < dl - 3:
        h = best_h.copy()
        h = lbfgs_run(h, n, 10000000, 500.0, 500, dl - 1)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
