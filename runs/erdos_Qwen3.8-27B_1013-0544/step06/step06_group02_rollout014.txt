    sh = alpha * (c - m)

    return m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha

    g = topk_grad(h, n, K, alpha)
    pos_mask = g > 0.01
    neg_mask = g < -0.01
    room_up = np.where(h < 1.0, 1.0 - h, 0.0)
    room_dn = np.where(h > 0.0, h, 0.0)
    total_pos = np.sum(np.maximum(g, 0) * room_dn)
    total_neg = np.sum(np.maximum(-g, 0) * room_up)
    if total_pos < 1e-12 or total_neg < 1e-12:
        return h, False
    scale = min(total_pos, total_neg)
    h_new = h.copy()
    h_new[pos_mask] -= lr * np.maximum(g[pos_mask], 0) * room_dn[pos_mask] / scale * scale
    h_new[neg_mask] += lr * np.maximum(-g[neg_mask], 0) * room_up[neg_mask] / scale * scale
    h_new = project_feasible(h_new, n)
    return h_new, True
