import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:

            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _lag_grad(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        grad[:n-m] += (1.0 - h[m:])
        grad[m:] -= h[:n-m]
    else:
        grad[:n-m] -= h[m:]
        grad[m:] += (1.0 - h[:n-m])
    return grad

def _smooth_obj_and_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        vals = np.exp(shifted - mx)
        log_sum = mx + np.log(vals.sum())
        obj = m + log_sum / alpha
    else:
        vals = np.exp(shifted)
        log_sum = np.log(vals.sum())
        obj = m + log_sum / alpha
    weights = vals / vals.sum()
    grad = np.zeros(n)
    for ki in range(2 * n - 1):
        if weights[ki] < 1e-14:
            continue
        lag = ki - (n - 1)
        grad += weights[ki] * _lag_grad(x, n, lag) * dx
    return obj, grad

def slsqp(h, n, alpha, maxiter=5000, ftol=1e-16):
    def obj(x):
        return _smooth_obj_and_grad(x, n, alpha)[0]
    def jac(x):
        return _smooth_obj_and_grad(x, n, alpha)[1]

    dx = 2.0 / n

        c = compute_all_lags(h, n)
        k_star = int(np.argmax(c))
        lag = k_star - (n - 1)

    return float(np.max(c) * dx)

def compute_all_lags(h, n):

def project_feasible(h, n):
    target = n / 2.0

        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h

def _lag_grad(h, n, lag):

    m = abs(lag)

        grad[m:] -= h[:n - m]
    else:

        grad[m:] += (1.0 - h[:n - m])
    return grad
