            d = a + gr * (b - a)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def corr_update(c, h, p, d, n):
    k1 = max(0, n - 1 - p)
    k2 = min(2 * n - 2, 2 * n - 2 - p)
    if k1 <= k2:
        js = p + np.arange(k1, k2 + 1) - (n - 1)
        c[k1:k2 + 1] += d * (1.0 - h[js])
    k1 = max(0, p)
    k2 = min(2 * n - 2, p + n - 1)
    if k1 <= k2:
        ii = p - np.arange(k1, k2 + 1) + (n - 1)
        c[k1:k2 + 1] -= d * h[ii]
    c[n - 1] -= d * d

def eval_coord(c, h, p, d, n, sumh, target, penalty):
    corr_update(c, h, p, d, n)
    val = float(np.max(c)) + penalty * (sumh + d - target) ** 2
    corr_update(c, h, p, -d, n)
    return val

def cd_phase(h, c, n, deadline, grid, refine, rng, penalty):
    target = n / 2.0
    sumh = float(h.sum())
    cur_max = float(np.max(c))
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for _pass in range(5000):
        if time.time() > deadline:
            break
        idx = np.arange(n)
        rng.shuffle(idx)
        improved = False
        for p in idx:
            d_lo = -h[p]
            d_hi = 1.0 - h[p]
            if d_hi - d_lo < 1e-14:
                continue
            deltas = np.linspace(d_lo, d_hi, grid)
            best_d = 0.0
            best_val = cur_max + penalty * (sumh - target) ** 2
            for d in deltas:
                if abs(d) < 1e-14:
                    continue
                v = eval_coord(c, h, p, d, n, sumh, target, penalty)
                if v < best_val:
                    best_val = v
                    best_d = d
            if refine and abs(best_d) > 1e-10:
                step = (d_hi - d_lo) / (grid - 1)
                a = max(d_lo, best_d - 2 * step)
                b = min(d_hi, best_d + 2 * step)
                if b - a > 1e-10:
                    c1 = b - gr * (b - a)
                    c2 = a + gr * (b - a)
                    f1 = eval_coord(c, h, p, c1, n, sumh, target, penalty)
                    f2 = eval_coord(c, h, p, c2, n, sumh, target, penalty)
                    for _ in range(refine):
                        if f1 < f2:
                            b = c2; c2 = c1; f2 = f1
                            c1 = b - gr * (b - a)
                            f1 = eval_coord(c, h, p, c1, n, sumh, target, penalty)
                        else:
                            a = c1; c1 = c2; f1 = f2
                            c2 = a + gr * (b - a)
                            f2 = eval_coord(c, h, p, c2, n, sumh, target, penalty)
                    d_gold = (a + b) / 2.0
                    v_gold = eval_coord(c, h, p, d_gold, n, sumh, target, penalty)
                    if v_gold < best_val:
                        best_d = d_gold
                        best_val = v_gold
            if abs(best_d) > 1e-12:
                corr_update(c, h, p, best_d, n)
                h[p] = np.clip(h[p] + best_d, 0.0, 1.0)
                sumh += best_d
                cur_max = float(np.max(c))
                improved = True
        if not improved:
            break
    h = project_feasible(h, n)
    c[:] = compute_corr(h, n)
    return h, c, float(np.max(c))

def fast_pair_search(h, c, n, deadline, rng, penalty=3.0):
    target = n / 2.0
    cur_max = float(np.max(c))
    sumh = float(h.sum())
    best_c5 = cur_max * (2.0 / n)
    d_steps = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for d in d_steps:
        improved_round = False
        while time.time() < deadline:
            improved = False
            for _ in range(2000):
                if time.time() > deadline:
                    break
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                for sgn in [1, -1]:
                    di = sgn * d
                    dj = -sgn * d
                    if h[i] + di < -1e-14 or h[i] + di > 1 + 1e-14:
                        continue
                    if h[j] + dj < -1e-14 or h[j] + dj > 1 + 1e-14:
                        continue
                    corr_update(c, h, i, di, n)
                    corr_update(c, h, j, dj, n)
                    val = float(np.max(c)) + penalty * (sumh - target) ** 2
                    cv5 = val * (2.0 / n)
                    if cv5 < best_c5 - 1e-14:
                        h[i] = np.clip(h[i] + di, 0.0, 1.0)
                        h[j] = np.clip(h[j] + dj, 0.0, 1.0)
                        sumh = float(h.sum())
                        cur_max = float(np.max(c))
                        best_c5 = cv5
                        improved = True
                        improved_round = True
                    else:
                        corr_update(c, h, i, -di, n)
                        corr_update(c, h, j, -dj, n)
            if not improved or not improved_round:
                break
    h = project_feasible(h, n)
    c[:] = compute_corr(h, n)
    return h, c, float(np.max(c))

def slsqp_phase(h, n, alpha, maxiter, deadline):
    dims = len(h)
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = float(c.max())
        shifted = alpha * (c - m)
        mx = shifted.max()
        if mx > 500:
            logsum = mx + np.log(np.sum(np.exp(shifted - mx))) / alpha
        else:
            logsum = np.log(np.sum(np.exp(shifted))) / alpha
        return (m + logsum) * (2.0 / dims)
    def obj_jac(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = float(c.max())
        shifted = alpha * (c - m)
        mx = shifted.max()
        if mx > 500:
            w = np.exp(shifted - mx)
            w /= w.sum()
        else:
            w = np.exp(shifted)
            w /= w.sum()
        dc = np.zeros(2 * dims - 1)
        for p in range(dims):
            k1 = max(0, n - 1 - p)
            k2 = min(2 * dims - 2, 2 * dims - 2 - p)
            js = p + np.arange(k1, k2 + 1) - (n - 1)
            dc[k1:k2 + 1] += (1.0 - x[js])
            k1 = max(0, p)
            k2 = min(2 * dims - 2, p + n - 1)
            ii = p - np.arange(k1, k2 + 1) + (n - 1)
            dc[k1:k2 + 1] += -x[ii] + x[p]
        jac = np.zeros(dims)
        for p in range(dims):
            jac[p] = float(np.dot(dc, w)) * (2.0 / dims)
        return jac
    def cons_fun(x):
        return x.sum() - n / 2.0
    def cons_jac(x):
        return np.ones(dims)
    for _ in range(dims):
        pass
    bounds = [(0.0, 1.0)] * dims
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    res = minimize(obj, h, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
    return project_feasible(res.x, n)
