    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):

        if abs(diff) < 1e-12:

        if diff > 0:

            tot = room.sum()

            if tot < 1e-15: break

def fast_coord_descend(h, n, deadline, rng, passes=200):
    """Coordinate descent with incremental correlation update."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full').copy()
    current_max = float(np.max(c))
    sumh = float(h.sum())
    n_full = 2 * n - 1
    
    for p in range(passes):
        if time.time() > deadline:
            break
        indices = np.arange(n) if p % 3 == 0 else rng.permutation(n)
        for idx in indices:
            hp = h[idx]
            if hp <= 1e-14 or hp >= 1 - 1e-14:
                continue
            # Compute affected range
            k_lo = min(n - 1 - idx, idx)
            k_hi = max(2 * n - 2 - idx, idx + n - 1)
            # Maximum of unaffected lags
            um = 0.0
            if k_lo > 0:
                um = max(um, float(np.max(c[:k_lo])))
            if k_hi < n_full - 1:
                um = max(um, float(np.max(c[k_hi + 1:])))
            
            # For each x in [0,1], compute the new max
            # corr[k](x) = c[k] + (x - hp) * s[k] - (x^2 - hp^2) * d[k]
            # where s[k] and d[k] depend on the structure
            
            # Actually, let's just use a grid search on the affected lags
            # For x = hp + delta:
            #   For k in [n-1-idx, 2n-2-idx]: corr[k] += delta * (1 - h[idx + k - n + 1])
            #   For k in [idx, idx+n-1]: corr[k] += -delta * h[idx + n - 1 - k]
            #   For k = n-1 (if in range): corr[k] += -delta^2
            
            # Let's compute the slope s[k] for affected k
            ks = np.arange(k_lo, k_hi + 1)
            s = np.zeros(len(ks))
            # Forward: k where idx + k - (n-1) is valid, i.e., k in [n-1-idx, 2n-2-idx]
            fwd_lo = max(k_lo, n - 1 - idx)
            fwd_hi = min(k_hi, 2 * n - 2 - idx)
            if fwd_lo <= fwd_hi:
                for k in range(fwd_lo, fwd_hi + 1):
                    j = idx + k - (n - 1)
                    s[k - k_lo] += (1.0 - h[j])
            # Backward: k where idx + (n-1) - k is valid, i.e., k in [idx, idx+n-1]
            bwd_lo = max(k_lo, idx)
            bwd_hi = min(k_hi, idx + n - 1)
            if bwd_lo <= bwd_hi:
                for k in range(bwd_lo, bwd_hi + 1):
                    j = idx + (n - 1) - k
                    s[k - k_lo] -= h[j]
            
            # Diagonal: k = n-1 (only if in affected range)
            has_diag = (n - 1 >= k_lo) and (n - 1 <= k_hi)
            
            # Grid search for best x
            grid = np.linspace(0.0, 1.0, 50)
            best_val = current_max
            best_x = hp
            for x in grid:
                d = x - hp
                if abs(d) < 1e-14:
                    continue
                new_c_affected = c[k_lo:k_hi + 1] + d * s
                if has_diag:
                    new_c_affected[n - 1 - k_lo] -= d * d
                val = max(um, float(np.max(new_c_affected)))
                if val < best_val - 1e-15:
                    best_val = val
                    best_x = x
            
            if best_x != hp:
                d = best_x - hp
                h[idx] = best_x
                # Update correlation
                c[k_lo:k_hi + 1] += d * s
                if has_diag:
                    c[n - 1] -= d * d
                sumh += d
                current_max = max(um, float(np.max(c[k_lo:k_hi + 1])))
    
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def frank_wolfe_step(h, n, alpha=1000):
    """One Frank-Wolfe iteration for minimizing max corr."""
    c = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(c)
    # Compute softmax weights
    w = np.exp(alpha * (c - max_val))
    w /= np.sum(w)
    
    # Compute subgradient: w_k * ∂corr[k]/∂h
    # ∂corr[k]/∂h[p] = (1-h[p+k-n+1]) * 1(p+k-n+1 ∈ [0,n-1]) - h[p-k+n-1] * 1(p-k+n+1 ∈ [0,n-1])
    # This is expensive to compute directly. Let's use a different approach.
    
    # Actually, let's compute the gradient of the soft-max approximation:
    # f_alpha(h) = (1/alpha) * log(sum exp(alpha * c_k))
    # ∇f_alpha = (1/alpha) * sum_k w_k * ∇c_k
    
    # ∇c_k is a vector of length n. Computing all of them is O(n^2).
    # For n=500, that's 250,000 per FW step. With 1000 FW steps, that's 250M ops.
    # In numpy, each operation on a 500-element vector is fast, so this should be OK.
    
    g = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-10:
            continue
        # ∂corr[k]/∂h[p]
        # Forward: p + k - (n-1) valid → p in [0, n-1-k] if k < n, else [k-n+1, n-1]
        # Actually: i = p, j = p + k - (n-1), need j in [0, n-1]
        # So p + k - n + 1 in [0, n-1] → p in [n-1-k, n-1-k+n-1] = [n-1-k, 2n-2-k]
        # Intersect with [0, n-1]: p in [max(0, n-1-k), min(n-1, 2n-2-k)]
        f_lo = max(0, n - 1 - k)
        f_hi = min(n - 1, 2 * n - 2 - k)
        if f_lo <= f_hi:
            ps = np.arange(f_lo, f_hi + 1)
            g[ps] += w[k] * (1.0 - h[ps + k - n + 1])
        
        # Backward: i + k - (n-1) = p → i = p - k + n - 1, need i in [0, n-1]
        # p - k + n - 1 in [0, n-1] → p in [k-n+1, k]
        # Intersect with [0, n-1]: p in [max(0, k-n+1), min(n-1, k)]
        b_lo = max(0, k - n + 1)
        b_hi = min(n - 1, k)
        if b_lo <= b_hi:
            ps = np.arange(b_lo, b_hi + 1)
            g[ps] -= w[k] * h[ps - k + n - 1]
    
    # Linear minimization: minimize g·v s.t. 0≤v≤1, sum(v)=n/2
    # Solution: set v[p]=1 where g[p] is smallest, v[p]=0 where largest, one fractional
    order = np.argsort(g)
    v = np.zeros(n)
    # We need sum(v) = n/2
    # Set v[order[0:n/2]] = 1, v[order[n/2:n]] = 0, adjust the boundary
    half = n // 2
    if n % 2 == 0:
        v[order[:half]] = 1.0
    else:
        # This shouldn't happen since n/2 should be integer for our constraint
        v[order[:half]] = 1.0
        v[order[half]] = 1.0  # adjust
    # Actually sum(h) = n/2 exactly, so n must be even or we need a fractional value
    # For even n: set exactly n/2 entries to 1
    # For odd n: set (n-1)/2 entries to 1 and one to 0.5
    # But n/2 must be achievable, so n should be even. Let's assume n is even.
    
    # Step size: use backtracking or fixed
    gamma = 2.0 / (t + 2)  # t is iteration number, use standard FW step size
    h_new = (1 - gamma) * h + gamma * v
    return h_new

        g = np.zeros(n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def fast_cd(h, n, deadline, rng, n_passes=1000, grid_size=80):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full').astype(float).copy()
    n_full = 2 * n - 1
    current_max = float(np.max(c))
    no_improve = 0
    for p in range(n_passes):
        if time.time() > deadline:
            break
        if no_improve > 30:
            break
        if p % 4 == 0:
            indices = np.arange(n)
        elif p % 4 == 1:
            indices = n - 1 - np.arange(n)
        else:
            indices = rng.permutation(n)
        pass_improved = False
        for idx in indices:
            hp = h[idx]
            if hp < 1e-14 or hp > 1.0 - 1e-14:
                continue
            k_lo = min(n - 1 - idx, idx)
            k_hi = max(2 * n - 2 - idx, idx + n - 1)
            um = 0.0
            if k_lo > 0:
                um = max(um, float(c[:k_lo].max()))
            if k_hi < n_full - 1:
                um = max(um, float(c[k_hi + 1:].max()))
            ks = np.arange(k_lo, k_hi + 1)
            s = np.zeros(len(ks))
            f_lo = max(k_lo, n - 1 - idx)
            f_hi = min(k_hi, 2 * n - 2 - idx)
            if f_lo <= f_hi:
                fks = np.arange(f_lo, f_hi + 1)
                s[fks - k_lo] += (1.0 - h[idx + fks - n + 1])
            b_lo = max(k_lo, idx)
            b_hi = min(k_hi, idx + n - 1)
            if b_lo <= b_hi:
                bks = np.arange(b_lo, b_hi + 1)
                s[bks - k_lo] -= h[idx + n - 1 - bks]
            has_diag = (n - 1 >= k_lo) and (n - 1 <= k_hi)
            diag_idx = n - 1 - k_lo if has_diag else -1
            grid = np.linspace(0.0, 1.0, grid_size)
            best_val = current_max
            best_x = hp
            for x in grid:
                d = x - hp
                if abs(d) < 1e-14:
                    continue
                nc = c[k_lo:k_hi + 1] + d * s
                if has_diag:
                    nc[diag_idx] -= d * d
                val = max(um, float(nc.max()))
                if val < best_val - 1e-15:
                    best_val = val
                    best_x = x
            if best_x != hp:
                d = best_x - hp
                h[idx] = best_x
                c[k_lo:k_hi + 1] += d * s
                if has_diag:
                    c[n - 1] -= d * d
                current_max = max(um, float(c[k_lo:k_hi + 1].max()))
                pass_improved = True
        if pass_improved:
            no_improve = 0
        else:
            no_improve += 1
    h = project_feasible(h, n)
    return h, compute_c5(h, n)

def grad_step(h, n, alpha=100.0):
    dx = 2.0 / n
    n_f = 2 * n - 1
    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(c)
    w = np.exp(alpha * (c - c_max))
    w /= w.sum()
    g = np.zeros(n)
    for k in range(n_f):
        wk = w[k]
        if wk < 1e-8:
            continue
        f_lo = max(0, n - 1 - k)
        f_hi = min(n - 1, 2 * n - 2 - k)
        if f_lo <= f_hi:
            ps = np.arange(f_lo, f_hi + 1)
            g[ps] += wk * (1.0 - h[ps + k - n + 1])
        b_lo = max(0, k - n + 1)
        b_hi = min(n - 1, k)
        if b_lo <= b_hi:
            ps = np.arange(b_lo, b_hi + 1)
            g[ps] -= wk * h[ps - k + n - 1]
    return g / len(c)

def grad_polish(h, n, deadline, n_iter=3000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    h = h.copy()
    alpha = 80.0
    for it in range(n_iter):
        if time.time() > deadline:
            break
        if it % 500 == 0 and it > 0:
            alpha = min(alpha * 1.4, 5000.0)
        g = grad_step(h, n, alpha)
        lr = max(0.001, 0.05 * (0.95 ** (it // 200)))
        for t_mult in [1.0, 0.5, 0.2, 0.1]:
            h_new = project_feasible(h - lr * t_mult * g, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new
                break
            if c_new < compute_c5(h, n):
                h = h_new
                break
    return best_h, best_c

def optimize_at_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c
