import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    if s.max() > 500:
        return float(m + (s.max() + np.log(np.sum(np.exp(s - s.max())))) / alpha)
    sh = s - s.max()
    return float(m + s.max() / alpha + np.log(np.sum(np.exp(sh))) / alpha)

def _con_fun(x, n):
    return x.sum() - n / 2.0

def _con_jac(x, n):
    return np.ones(n)

def slsqp_step(h, n, alpha=50000, maxiter=1000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def fast_cd(h, n, max_passes=300, grid=60, deadline=None, rng=None, penalty=5.0):
    h = h.copy()
    dx = 2.0 / n
    c = compute_corr(h, n)
    current_max = float(np.max(c))
    sumh = float(h.sum())
    for _pass in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        order = np.arange(n)
        if rng is not None:
            rng.shuffle(order)
        for p in order:
            d_lo, d_hi = -h[p], 1.0 - h[p]
            if d_hi - d_lo < 1e-14:
                continue
            k_lo = max(0, min(p, n - 1 - p))
            k_hi = min(2 * n - 2, max(p + n - 1, 2 * n - 2 - p))
            um = 0.0
            if k_lo > 0:
                um = max(um, float(np.max(c[:k_lo])))
            if k_hi < 2 * n - 2:
                um = max(um, float(np.max(c[k_hi + 1:])))
            ks = np.arange(k_lo, k_hi + 1)
            s_coef = np.zeros(len(ks))
            fwd = (ks >= p) & (ks <= p + n - 1)
            bwd = (ks >= n - 1 - p) & (ks <= 2 * n - 2 - p)
            if np.any(fwd):
                s_coef[fwd] += 1.0 - h[p - ks[fwd] + n - 1]
            if np.any(bwd):
                s_coef[bwd] -= h[p + ks[bwd] - (n - 1)]
            deltas = np.linspace(d_lo, d_hi, grid)
            best_d = 0.0
            best_val = current_max + penalty * (sumh - n / 2.0) ** 2
            for d in deltas:
                if abs(d) < 1e-14:
                    continue
                new_c = c[ks].copy() + d * s_coef
                ci = (ks == (n - 1))
                if np.any(ci):
                    new_c[ci] = c[n - 1] + d * (1.0 - 2.0 * h[p]) - d * d
                val = max(um, float(np.max(new_c)))
                if penalty > 0:
                    val += penalty * (sumh + d - n / 2.0) ** 2
                if val < best_val - 1e-14:
                    best_val = val
                    best_d = d
            if abs(best_d) > 1e-12:
                d = best_d
                ids_f = np.where(fwd)[0]
                ids_b = np.where(bwd)[0]

                    c[ks[ids_b]] -= d * h[p + ks[ids_b] - (n - 1)]
                c[n - 1] -= d * d
                h[p] = np.clip(h[p] + d, 0.0, 1.0)
                sumh = float(h.sum())
                current_max = max(um, float(np.max(c[ks])))
                improved = True
        if not improved:
            break
    return h, current_max * dx

def pair_search(h, n, deadline, deltas, radius=30):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo, hi = max(0, i - radius), min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        cv = compute_c5(h, n)
                        if cv < best_c - 1e-15:
                            best_c = cv; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        cv = compute_c5(h, n)
                        if cv < best_c - 1e-15:
                            best_c = cv; improved = True
                        else:
                            h[i] += d; h[j] -= d

                break

def fourier_optimize(n, M, deadline, rng):
    """Optimize h parameterized by M cosine modes (symmetric)."""
    x = np.linspace(0, 2, n, endpoint=False)
    basis = np.ones((n, 1))
    for m in range(1, M + 1):
        basis = np.column_stack([basis, np.cos(np.pi * m * x)])
    
    def obj4(a):
        h = 0.5 + basis[:, 1:] @ a
        pen = 20.0 * (np.sum(np.maximum(0, -h)**2) + np.sum(np.maximum(0, h-1)**2))
        h_clipped = np.clip(h, 0, 1)
        return compute_c5(h_clipped, n) + pen
    
    best_val = 1e9
    best_a = np.zeros(M)
    n_starts = max(3, int((deadline - time.time()) / 20))
    for s in range(n_starts):
        if time.time() > deadline:
            break

            a0 = np.array([0.15, -0.1, 0.05] + [0.0]*(M-3)[:M-3])

        else:
            a0 = rng.uniform(-0.25, 0.25, M)
        try:
            res = minimize(obj4, a0, method='Nelder-Mead',
                          options={'maxiter': 8000, 'xatol': 1e-9, 'fatol': 1e-11,

            if res.fun < best_val:

            pass
