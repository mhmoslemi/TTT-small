    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def lse_value(c, alpha):
    m = c.max()
    if alpha < 1e6:
        shifted = alpha * (c - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    return m

def lse_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full').astype(float) * dx
    if alpha < 1e6:
        m = c.max()
        shifted = alpha * (c - m)
        sm = shifted.max()
        exp_v = np.exp(shifted - sm)
        w = exp_v / exp_v.sum()
    else:
        # Use subgradient: weight on the max
        w = np.zeros_like(c)
        w[c == c.max()] = 1.0 / (c == c.max()).sum()
    
    idx = np.arange(n)
    W1 = w[idx[:, None] - idx[None, :] + (n - 1)]
    W2 = w[idx[None, :] - idx[:, None] + (n - 1)]
    grad = W1 @ (1.0 - h) - W2 @ h
    val = lse_value(c, alpha)
    return grad, val, c

def greed_binary(h_init, n, deadline, max_sweeps=100):

    zeros = list(np.where(h == 0)[0])

        if time.time() > deadline:
            break
        found = False
        for i in zeros[:]:
            if time.time() > deadline:
                break
            for j in ones[:]:
                # Compute delta for swap i->1, j->0
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                m1 = (im >= 0) & (im < n)

                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                if 0 <= k1 < 2*n-1: delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                if 0 <= k2 < 2*n-1: delta[k2] += 1.0

                    cur_c = nc
                    ones.remove(j); zeros.remove(i)
                    ones.append(i); zeros.append(j)
                    found = True

            if found:

        s = step
        for ls in range(20):
            h_new = project_feasible(h - s * grad, n)
            c_new = np.correlate(h_new, 1.0 - h_new, mode='full').astype(float)
            cmax = float(c_new.max() * (2.0 / n))
            if cmax < best_c - 1e-15:
                h = h_new
                best_c = cmax
                best_h = h.copy()
                improved = True
                break
            s *= 0.5
        if improved:
            step = min(s * 1.03, n_safe * 0.5)
            stall = 0
        else:
            stall += 1
            if stall > 150:
                if stall % 50 == 0:
                    h = project_feasible(h + 0.05 * rng_local.randn(n), n)
                    cc = compute_c5(h, n)
                    if cc < best_c:
                        best_c, best_h = cc, h.copy()
                if stall > 500:
                    break
    return best_h, best_c

def greed_binary(h_init, n, deadline, max_sweeps=80):
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = float(corr.max() * dx)
    best_h = h.copy()
    best_c = cur_c
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        found = False
        for i in zeros[:]:
            if time.time() > deadline:
                break
            for j in ones[:]:
                delta = np.zeros(2 * n - 1)
                im = i + m_range
                m1 = (im >= 0) & (im < n)
                delta[m1] += 1.0 - h[im[m1]]
                imn = i - m_range
                m2 = (imn >= 0) & (imn < n)
                delta[m2] -= h[imn[m2]]
                jm = j + m_range
                m3 = (jm >= 0) & (jm < n)
                delta[m3] -= 1.0 - h[jm[m3]]
                jmn = j - m_range
                m4 = (jmn >= 0) & (jmn < n)
                delta[m4] += h[jmn[m4]]
                delta[n - 1] -= 2.0
                k1 = n - 1 + (j - i)
                if 0 <= k1 < 2 * n - 1:
                    delta[k1] += 1.0
                k2 = n - 1 + (i - j)
                if 0 <= k2 < 2 * n - 1:
                    delta[k2] += 1.0
                nc = float((corr + delta).max() * dx)
                if nc < cur_c - 1e-14:
                    h[i], h[j] = 1.0, 0.0
                    corr += delta
                    cur_c = nc
                    ones.remove(j)
                    zeros.remove(i)
                    ones.append(i)
                    zeros.append(j)
                    found = True
                    break
            if found:
                break
        if not found:
            break
        if cur_c < best_c:
            best_c, best_h = cur_c, h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 400
        h0 = np.full(n, 0.5)
    best_h = project_feasible(h0, n)
    best_c = compute_c5(best_h, n)
    alphas = [5, 20, 100, 500, 2000, 10000, 50000, 200000, 1000000]
    h = best_h.copy()
    for al in alphas:
        if time.time() > dl - 180:
            break
        h, c = grad_desc(h, n, al, 4000, dl - 180)
        if c < best_c:
            best_c, best_h = c, h
    if time.time() < dl - 100:
        hb, cb = greed_binary(best_h, n, dl - 100, max_sweeps=80)
        if cb < best_c:
            best_c, best_h = cb, hb
        if time.time() < dl - 70:
            h2 = best_h.astype(float)
            for al in [5000, 50000, 500000]:
                if time.time() > dl - 65:
                    break
                h2, c2 = grad_desc(h2, n, al, 3000, dl - 65)
                if c2 < best_c:
                    best_c, best_h = c2, h2
    for nh in [200, 300, 600]:
        if time.time() > dl - 50:
            break
        xo = np.linspace(0, 2, n, endpoint=False)
        xn = np.linspace(0, 2, nh, endpoint=False)
        hh = project_feasible(np.interp(xn, xo, best_h), nh)
        for al in [500, 5000, 50000]:
            if time.time() > dl - 40:
                break
            hh, ch = grad_desc(hh, nh, al, 2000, dl - 40)
            if ch < best_c:
                best_c, best_h, n = ch, hh, nh
    if time.time() < dl - 10:
        for trial in range(8):
            if time.time() > dl - 8:
                break
            h_r = best_h + 0.03 * rng.randn(n)
            h_r = project_feasible(h_r, n)
            for al in [10000, 100000, 1000000]:
                if time.time() > dl - 6:
                    break
                h_r, cr = grad_desc(h_r, n, al, 2000, dl - 6)
                if cr < best_c:
                    best_c, best_h = cr, h_r
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
