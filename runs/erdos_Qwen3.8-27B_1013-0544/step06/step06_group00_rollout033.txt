import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_correlation(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_grad(h, w, alpha):
    n = len(h)
    s = np.arange(-(n - 1), n)
    J = np.arange(n)[:, None]
    S = s[None, :]
    idx1 = J - S
    idx2 = J + S
    m1 = (idx1 >= 0) & (idx1 < n)
    m2 = (idx2 >= 0) & (idx2 < n)
    h1 = np.where(m1, 1.0 - h[np.clip(idx1, 0, n - 1)], 0.0)
    h2 = np.where(m2, h[np.clip(idx2, 0, n - 1)], 0.0)
    return (w[None, :] * (h1 - h2)).sum(axis=1) / alpha

def project_exact(h, n):
    target = n / 2.0
    lo, hi = -1.0, 1.0
    for _ in range(64):
        mid = (lo + hi) * 0.5
        if np.clip(h + mid, 0.0, 1.0).sum() > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_b = np.zeros(n)
    h_b[idx[:n // 2]] = 1.0
    return h_b

def binary_delta(h, i, j, n):
    m_range = np.arange(-(n - 1), n)
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    mk = (im >= 0) & (im < n)
    delta[mk] += 1.0 - h[im[mk]]
    imn = i - m_range
    mk = (imn >= 0) & (imn < n)
    delta[mk] -= h[imn[mk]]
    jm = j + m_range
    mk = (jm >= 0) & (jm < n)
    delta[mk] -= 1.0 - h[jm[mk]]
    jmn = j - m_range
    mk = (jmn >= 0) & (jmn < n)
    delta[mk] += h[jmn[mk]]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    k2 = n - 1 + (i - j)
    if 0 <= k1 < 2 * n - 1:
        delta[k1] += 1.0
    if 0 <= k2 < 2 * n - 1:
        delta[k2] += 1.0
    return delta

def anneal_binary(h_init, n, deadline, seed=42):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    h = to_binary(h_init, n).copy()
    corr = compute_correlation(h).astype(float)
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ones_idx = np.where(h == 1)[0].copy()
    zeros_idx = np.where(h == 0)[0].copy()
    T = 0.008
    cool = 0.99996
    scale = 8000.0
    while time.time() < deadline:
        ai = rng.randint(len(zeros_idx))
        bi = rng.randint(len(ones_idx))
        i, j = zeros_idx[ai], ones_idx[bi]
        delta = binary_delta(h, i, j, n)
        nc = (corr + delta).max() * dx
        if nc <= cur_c or (T > 1e-12 and rng.random() < np.exp(-scale * (nc - cur_c) / T)):
            h[i], h[j] = 1.0, 0.0
            corr += delta
            cur_c = nc
            zeros_idx[ai] = j
            ones_idx[bi] = i
            if cur_c < best_c:
                best_c, best_h = cur_c, h.copy()
        T *= cool
    return best_h, best_c
