The water-filling projection:
To project h onto {0 <= x <= 1, sum(x) = target}:
- Start with h clipped to [0,1]
- If sum > target, subtract a uniform amount from all entries that are at 1, reducing them until sum matches
- If sum < target, add a uniform amount to all entries that are at 0, increasing them until sum matches

This is equivalent to finding a threshold t such that:
sum(clip(h[i] - t, 0, 1)) = target

```python
from scipy.optimize import minimize
from scipy.signal import fftconvolve

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    return h

def lse_grad(h, alpha, n):
    """Compute gradient of log-sum-exp smoothed objective."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = corr.max()
    shifted = alpha * (corr - cmax)
    shifted -= shifted.max()
    exp_s = np.exp(shifted)
    p = exp_s / exp_s.sum()
    # Compute gradient: dx * (1 - S1 - S2)
    # S1[j] = sum_k p[k] * h[j+k]
    # S2[j] = sum_k p[k] * h[j-k]
    shifts = np.arange(-(n-1), n)  # (2n-1,)
    j = np.arange(n)[:, None]  # (n, 1)
    kk = shifts[None, :]  # (1, 2n-1)
    idx1 = j + kk  # (n, 2n-1)
    m1 = (idx1 >= 0) & (idx1 < n)
    S1 = (h[np.clip(idx1, 0, n-1)] * m1).sum(axis=1) * 1  # wait, need to multiply by p
    # S1[j] = sum_k p[k_idx] * h[j + shifts[k_idx]] where k_idx is the index
    # idx1[j, k_idx] = j + shifts[k_idx]
    # S1[j] = sum over k_idx where valid: p[k_idx] * h[idx1[j, k_idx]]
    # = sum over k_idx: p[k_idx] * h[idx1_clamped[j,k_idx]] * m1[j,k_idx]
    h_ext1 = h[np.clip(idx1, 0, n-1)]  # (n, 2n-1)

    S2 = (h_ext2 * m2 * p[None, :]).sum(axis=1)
    
    grad = dx * (1.0 - S1 - S2)
    f = cmax + np.log(exp_s.sum()) / alpha
    return f, grad

def adam_optimize(h, n, alpha, max_iter, lr, deadline):
    """Adam optimizer with log-sum-exp smoothing."""
    h = h.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    t = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        f, grad = lse_grad(h, alpha, n)
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        h -= lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_feasible(h, n)
    return h, f

    """Greedy binary swap local search."""
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)

                jm = j + m_range
                msk = (jm >= 0) & (jm < n)
                delta[msk] -= 1.0 - h[jm[msk]]
                jmn = j - m_range

                delta[k1] += 1.0

                nc = (corr + delta).max() * dx

        imn = i - m_range
```

```python
def compute_S1S2(h, p, n):
    """Compute S1[i] = sum_u p[u] * h[i+u] and S2[i] = sum_u p[u] * h[i-u]"""
    # p is indexed 0..2n-2, where index t corresponds to shift u = t-(n-1)
    shifts = np.arange(-(n-1), n)  # u values, shape (2n-1,)
    i = np.arange(n)[:, None]  # shape (n, 1)
    u = shifts[None, :]  # shape (1, 2n-1)
    
    # S1[i] = sum over u: p[u_idx] * h[i+u] where 0<=i+u<n
    idx1 = i + u  # (n, 2n-1)
    mask1 = (idx1 >= 0) & (idx1 < n)
    h1 = h[np.clip(idx1, 0, n-1)]  # (n, 2n-1)
    S1 = (h1 * mask1 * p[None, :]).sum(axis=1)
    
    # S2[i] = sum over u: p[u_idx] * h[i-u] where 0<=i-u<n
    idx2 = i - u
    mask2 = (idx2 >= 0) & (idx2 < n)
    h2 = h[np.clip(idx2, 0, n-1)]
    S2 = (h2 * mask2 * p[None, :]).sum(axis=1)
    
    return S1, S2
```

```python
def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    return h

def make_masks(n):
    shifts = np.arange(-(n - 1), n)
    i = np.arange(n)[:, None]
    u = shifts[None, :]
    idx1 = i + u
    mask1 = ((idx1 >= 0) & (idx1 < n)).astype(float)
    idx1c = np.clip(idx1, 0, n - 1)
    idx2 = i - u
    mask2 = ((idx2 >= 0) & (idx2 < n)).astype(float)
    idx2c = np.clip(idx2, 0, n - 1)
    return mask1, idx1c, mask2, idx2c

def lse_grad(h, p, alpha, n, masks):
    mask1, idx1c, mask2, idx2c = masks
    S1 = (h[idx1c] * mask1 * p[None, :]).sum(axis=1)
    S2 = (h[idx2c] * mask2 * p[None, :]).sum(axis=1)
    dx = 2.0 / n
    grad = dx * (1.0 - S1 - S2)
    return grad

def lse_objective(h, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = corr.max()
    shifted = alpha * (corr - cmax)
    shifted -= shifted.max()
    return cmax + np.log(np.exp(shifted).sum()) / alpha

def adam_step(h, alpha, n, masks, m_arr, v_arr, t_arr, lr):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = corr.max()
    shifted = alpha * (corr - cmax)
    shifted -= shifted.max()
    exp_s = np.exp(shifted)
    p = exp_s / exp_s.sum()
    grad = lse_grad(h, p, alpha, n, masks)
    t_arr[0] += 1
    bt1, bt2, eps = 0.8, 0.999, 1e-8
    m_arr[:] = bt1 * m_arr + (1.0 - bt1) * grad
    v_arr[:] = bt2 * v_arr + (1.0 - bt2) * grad * grad
    mh = m_arr / (1.0 - bt1 ** t_arr[0])
    vh = v_arr / (1.0 - bt2 ** t_arr[0])
    h_new = h - lr * mh / (np.sqrt(vh) + eps)
    return project_feasible(h_new, n)

        if time.time() > deadline:
            break
        h = adam_step(h, alpha, n, masks, m_arr, v_arr, t_arr, lr)

    return h, c
```
