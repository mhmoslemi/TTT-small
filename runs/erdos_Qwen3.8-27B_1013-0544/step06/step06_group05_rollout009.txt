```python
def fast_binary_search(h, n, deadline):
    dx = 2.0/n
    corr = np.correlate(h, 1.0-h, mode='full').astype(np.float64)
    cur_max = corr.max()
    best_h = h.copy()
    best_max = cur_max
    
    while time.time() < deadline:
        # Find critical lags (within tolerance of max)
        tol = 0.5  # in units of the correlation (before dx)
        critical = np.where(corr >= cur_max - tol)[0]
        
        # Try all pairs (i in zeros, j in ones) - too many!
        # Instead: for each zero position i, find the best one position j to swap with
        zeros_idx = np.where(h == 0)[0]
        ones_idx = np.where(h == 1)[0]
        
        found_improvement = False
        for i in zeros_idx:
            # For this i, find j that minimizes the new max
            best_j = -1
            best_new_max = cur_max + 1  # worse than current
            for j in ones_idx:
                # Compute Δc[k] for all critical k
                # Δc[k] = h[j+k] - h[i+k] - h[i-k] + h[j-k] (with bounds)
                # Check if new max < cur_max
                feasible = True
                new_max = 0
                for k_idx in critical:
                    k = k_idx - (n-1)
                    delta = 0.0
                    # h[j+k]
                    if 0 <= j+k < n: delta += h[j+k]
                    # -h[i+k]
                    if 0 <= i+k < n: delta -= h[i+k]
                    # -h[i-k]
                    if 0 <= i-k < n: delta -= h[i-k]
                    # +h[j-k]
                    if 0 <= j-k < n: delta += h[j-k]
                    new_val = corr[k_idx] + delta
                    if new_val > cur_max:
                        feasible = False
                        break
                    new_max = max(new_max, new_val)
                if feasible and new_max < best_new_max:
                    best_new_max = new_max
                    best_j = j
            if best_j >= 0 and best_new_max < cur_max:
                h[i] = 1.0
                h[best_j] = 0.0
                corr = np.correlate(h, 1.0-h, mode='full').astype(np.float64)
                cur_max = corr.max()
                found_improvement = True
                if cur_max < best_max:
                    best_max = cur_max
                    best_h = h.copy()
                break  # restart the outer loop
        
        if not found_improvement:
            break
    
    return best_h, best_max * dx
```

```python
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h = np.clip(h + diff / s * room, 0.0, 1.0)
        else:
            s = h.sum()
            if s < 1e-14: break
            h = np.clip(h + diff / s * h, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    """Convert to binary with exactly n/2 ones."""
    target = n // 2
    thresh = np.sort(h)[::-1][target - 1]
    h_bin = (h >= thresh).astype(float)
    n_ones = int(h_bin.sum())
    if n_ones > target:
        excess = np.where(h_bin == 1)[0]
        # Remove excess with smallest h values
        order = np.argsort(h[excess])
        for idx in excess[order[:n_ones - target]]:
            h_bin[idx] = 0.0
    elif n_ones < target:
        deficit = np.where(h_bin == 0)[0]
        order = np.argsort(-h[deficit])
        for idx in deficit[order[:target - n_ones]]:
            h_bin[idx] = 1.0
    return h_bin

    best_h = h.copy()

    t_cur = t_start
    cooling = (t_end / t_start) ** (1.0 / max_iter)

        if time.time() > deadline:
            break
        t_cur *= cooling
        
        zeros_idx = np.where(h == 0)[0]
        ones_idx = np.where(h == 1)[0]
        if len(zeros_idx) == 0 or len(ones_idx) == 0:
            break

        j = rng.choice(ones_idx)

        # Δc[k] = h[j+k] - h[i+k] - h[i-k] + h[j-k]
        # For efficiency, compute full new correlation
        h[i] = 1.0
        h[j] = 0.0
        new_corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
        new_max = new_corr.max()
        
        delta = new_max - cur_max
        if delta < 0 or rng.rand() < np.exp(-delta / max(t_cur, 1e-10)):
            cur_max = new_max
            corr = new_corr
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        else:
            h[i] = 0.0
            h[j] = 1.0
    
    return best_h, float(best_max * dx)

        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return float(m + (sm + np.log(np.exp(sh - sm).sum())) / alpha)

        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h
```

```python
def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * room, 0.0, 1.0)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * h, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n // 2
    srt = np.sort(h)[::-1]
    thresh = srt[target - 1] if target > 0 else 1.0
    h_bin = (h >= thresh).astype(float)
    n_ones = int(h_bin.sum())
    if n_ones > target:
        excess = np.where(h_bin == 1)[0]
        order = np.argsort(h[excess])
        for idx in excess[order[:n_ones - target]]:
            h_bin[idx] = 0.0
    elif n_ones < target:
        deficit = np.where(h_bin == 0)[0]
        order = np.argsort(-h[deficit])
        for idx in deficit[order[:target - n_ones]]:
            h_bin[idx] = 1.0
    return h_bin

    p1_end = t0 + 0.25 * (budget_s - 12)
    h_bin = binarize(best_h, n)
    hb, cb = fast_binary_search(h_bin.copy(), n, p1_end, max_iter=30000, seed=seed)
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()

    # Phase 2: SLSQP continuous (25%)
    p2_end = t0 + 0.50 * (budget_s - 12)
    alphas = [200, 1000, 5000, 20000, 100000]
    for a in alphas:
        if time.time() > p2_end:
            break
        h_c = slsqp_optimize(best_h.copy(), n, a, 8000, p2_end)
        c = compute_c5(h_c, n)
        if c < best_c:
            best_c = c
            best_h = h_c.copy()

    # Phase 3: Re-binarize + binary search (15%)
    p3_end = t0 + 0.65 * (budget_s - 12)
    h_bin2 = binarize(best_h, n)
    hb2, cb2 = fast_binary_search(h_bin2.copy(), n, p3_end, max_iter=30000, seed=seed + 7)
    if cb2 < best_c:
        best_c = cb2
        best_h = hb2.copy()

    # Phase 4: SLSQP polish (15%)
    p4_end = t0 + 0.80 * (budget_s - 12)
    for a in [50000, 500000]:
        if time.time() > p4_end:
            break
        h_c = slsqp_optimize(best_h.copy(), n, a, 8000, p4_end)
        c = compute_c5(h_c, n)
        if c < best_c:
            best_c = c
            best_h = h_c.copy()

    # Phase 5: Simulated annealing on binary (10%)
    p5_start = time.time()
    p5_end = t0 + 0.90 * (budget_s - 12)
    if p5_end > p5_start:
        h_sa_init = binarize(best_h, n)
        budget_tracker['sa_start'] = p5_start
        budget_tracker['sa_dur'] = p5_end - p5_start
        hs, cs = sa_binary_search(h_sa_init.copy(), n, p5_end, t_start=4.0, t_end=0.03, seed=seed + 99)
        if cs < best_c:
            best_c = cs
            best_h = hs.copy()
            # Quick SLSQP after SA
            if time.time() < dl - 5:
                h_c = slsqp_optimize(best_h.copy(), n, 100000, 5000, dl - 3)
                c = compute_c5(h_c, n)
                if c < best_c:
                    best_c = c
                    best_h = h_c.copy()

    # Phase 6: Final polish
    if time.time() < dl - 3:
        h_final = slsqp_optimize(best_h.copy(), n, 200000, 5000, dl - 2)
        c = compute_c5(h_final, n)
        if c < best_c:
            best_c = c
            best_h = h_final.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
```
