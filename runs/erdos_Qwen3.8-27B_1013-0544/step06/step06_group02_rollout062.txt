            t *= cool
            continue

        else:

            t *= cool
            if rng.random() < 0.01:

                t = t0 * 0.5
    return best_h, best_c

        wm = w[ki] * dx
        if k >= 0:
            nv = n - k
            g[:nv] += wm * (1.0 - h[k:])
            g[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            g[kk:] += wm * (1.0 - h[:nv])
            g[:nv] += wm * (-h[kk:])
    return g

def lbfgs_opt(h, n, deadline, alphas=(1000, 5000, 20000, 100000, 500000, 2000000), max_iters=400):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    for a in alphas:
        if time.time() > deadline:
            break
        cur_alpha = a
        def obj(x):
            return smooth_max(compute_overlaps(x, n), cur_alpha) + 1e6 * (x.sum() - n/2.0)**2
        def grd(x):
            g = smooth_max_grad(x, n, cur_alpha)
            g += 2e6 * (x.sum() - n/2.0)
            return g
        try:
            res = minimize(obj, h, jac=grd, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-18, 'gtol': 1e-14,
                                   'maxfun': max_iters * 5, 'maxls': 30})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def subgrad_cd(h, n, deadline, top_k=15):
    """Subgradient-guided coordinate descent on the true max objective."""
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    deltas = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    
    for d in deltas:
        n_sweeps = 2000
        for sweep in range(n_sweeps):
            if time.time() > deadline:
                break
            # Compute overlaps and find top k_worst shifts
            ov = compute_overlaps(h, n)
            # Weighted subgradient from top shifts
            top_idx = np.argsort(ov)[::-1][:top_k]
            w = ov[top_idx] - ov[top_idx].min()
            if w.sum() < 1e-15:
                w = np.ones(top_k)
            else:
                w = w / w.sum()
            
            sg = np.zeros(n)
            for ti, ki in enumerate(top_idx):
                k = ki - (n - 1)
                wk = w[ti]
                if k >= 0:
                    nv = n - k
                    sg[:nv] += wk * (1.0 - h[k:]) * dx
                    sg[k:] += wk * (-h[:nv]) * dx
                else:
                    kk = -k
                    nv = n - kk
                    sg[kk:] += wk * (1.0 - h[:nv]) * dx
                    sg[:nv] += wk * (-h[kk:]) * dx
            
            # Find coordinates to increase (sg < 0) and decrease (sg > 0)
            inc = np.where(sg < -1e-12)[0]
            dec = np.where(sg > 1e-12)[0]
            if len(inc) == 0 or len(dec) == 0:
                break
            
            # Sort by |sg| and take top candidates
            inc = inc[np.argsort(-np.abs(sg[inc]))][:(2*top_k)]
            dec = dec[np.argsort(-np.abs(sg[dec]))][:(2*top_k)]
            
            improved = False
            for i in inc:
                if time.time() > deadline:
                    break
                for j in dec:
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-16:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
            if not improved:
                break
    return h, best_c

def full_pair_cd(h, n, deadline, radii=(5, 10, 20)):

                                improved = True
                            else:

                    break

def compute_overlaps(h, n):
    dx = 2.0 / n

    return float(np.max(compute_overlaps(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max(vals, alpha):
    m = vals.max()
    s = alpha * (vals - m)
    mx = s.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha
    return m + np.log(np.sum(np.exp(s))) / alpha
