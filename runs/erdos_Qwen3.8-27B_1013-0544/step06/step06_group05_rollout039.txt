    """Compute delta C from swapping h[a]: 0->1, h[b]: 1->0"""
    # C[k] = sum_i h[i+k] * (1-h[i])
    # For k in range -(n-1) to n-1, indexed as k + (n-1) in the array
    
    delta = np.zeros(2*n - 1)
    # h[a] goes from 0 to 1
    # As first factor h[i+k]: i+k = a => i = a-k, valid when 0<=a-k<n
    #   delta[k] += (1-h[a-k])
    # As second factor (1-h[i]): i = a
    #   delta[k] -= h[a+k]
    
    for k_idx in range(2*n-1):
        k = k_idx - (n-1)
        # a as first factor: i = a - k
        i1 = a - k
        if 0 <= i1 < n:
            delta[k_idx] += (1 - h[i1])
        # a as second factor: i = a, need 0<=a+k<n
        if 0 <= a + k < n:
            delta[k_idx] -= h[a + k]
        # b as first factor: i = b - k
        i2 = b - k
        if 0 <= i2 < n:
            delta[k_idx] -= (1 - h[i2])
        # b as second factor: i = b
        if 0 <= b + k < n:
            delta[k_idx] += h[b + k]
    return delta

def swap_delta(a, b, h, n):
    """Vectorized computation of correlation change from h[a]:0->1, h[b]:1->0."""
    off = n - 1
    delta = np.zeros(2 * n - 1)
    # a: 0->1 as first factor (h[i+k]): k = a-i, valid i in [max(0,a-n+1), min(a,n-1)]
    ilo = max(0, a - n + 1)
    ihi = min(a, n - 1)
    if ilo <= ihi:
        ii = np.arange(ilo, ihi + 1)
        delta[a - ii + off] += (1.0 - h[ii])
    # a: 0->1 as second factor (1-h[i]): i=a, k in [0, n-1-a]
    if a < n:
        ks = np.arange(0, n - a)
        delta[ks + off] -= h[a + ks]
    # b: 1->0 as first factor: k = b-i
    ilo = max(0, b - n + 1)
    ihi = min(b, n - 1)
    if ilo <= ihi:
        ii = np.arange(ilo, ihi + 1)
        delta[b - ii + off] -= (1.0 - h[ii])
    # b: 1->0 as second factor: i=b, k in [0, n-1-b]
    if b < n:
        ks = np.arange(0, n - b)
        delta[ks + off] += h[b + ks]
    return delta

def anneal_binary(h0, n, deadline, seed=42):
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    best_h = h0.copy()
    best_C = np.correlate(h0, 1.0 - h0, mode='full').astype(np.float64)
    best_max = best_C.max()
    
    # Main annealing loop
    cur = h0.copy()
    C = best_C.copy()
    cur_max = best_max
    
    T = 1.0
    T_min = 0.005
    cooling = 0.99997  # per iteration
    no_improve = 0
    
    ones_idx = np.where(cur == 1)[0]
    zeros_idx = np.where(cur == 0)[0]
    
    it = 0
    while time.time() < deadline:
        it += 1
        a = zeros_idx[rng.randint(len(zeros_idx))]
        b = ones_idx[rng.randint(len(ones_idx))]
        d = swap_delta(a, b, cur, n)
        new_max = (C + d).max()
        
        if new_max <= cur_max:
            accept = True
        else:
            delta_E = new_max - cur_max
            if rng.random() < np.exp(-delta_E / T):
                accept = True
            else:
                accept = False
        
        if accept:
            cur[a] = 1.0
            cur[b] = 0.0
            C += d
            cur_max = new_max
            no_improve = 0
            # Update index arrays
            zeros_idx = np.where(cur == 0)[0]
            ones_idx = np.where(cur == 1)[0]
            if cur_max < best_max:
                best_max = cur_max
                best_h = cur.copy()
        else:
            no_improve += 1
            if no_improve > 200000:
                # Restart with perturbation
                for _ in range(rng.randint(5, 15)):
                    ai = zeros_idx[rng.randint(len(zeros_idx))]
                    bi = ones_idx[rng.randint(len(ones_idx))]
                    cur[ai] = 1.0
                    cur[bi] = 0.0
                C = np.correlate(cur, 1.0 - cur, mode='full').astype(np.float64)
                cur_max = C.max()
                zeros_idx = np.where(cur == 0)[0]
                ones_idx = np.where(cur == 1)[0]
                no_improve = 0
                T = 1.0  # Reset temperature
        
        T = max(T_min, T * cooling)
    
    return best_h, best_C, best_max

def gradient_smooth_max(h, n, alpha):
    """
    Compute gradient of (1/alpha)*log(sum exp(alpha * C_k * dx))
    where C_k = sum_i h[i+k]*(1-h[i])
    """
    dx = 2.0 / n
    C = np.correlate(h, 1.0 - h, mode='full') * dx
    # Softmax weights
    C_max = C.max()
    weights = np.exp(alpha * (C - C_max))
    weights /= weights.sum()
    
    # Gradient: sum_k weights[k] * dC_k/dh[j]
    # dC_k/dh[j] = (1-h[j-k])_valid - h[j+k]_valid
    off = n - 1
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        k = k_idx - off
        w = weights[k_idx]
        if w < 1e-10:
            continue
        # (1-h[j-k]) for valid j: j in [max(0,k), min(n-1, n-1+k)] ... 
        # j-k valid means 0<=j-k<n, i.e., k<=j<n+k
        j_lo = max(0, k)
        j_hi = min(n - 1, n - 1 + k)
        if j_lo <= j_hi:
            jj = np.arange(j_lo, j_hi + 1)
            grad[jj] += w * (1.0 - h[jj - k])
        # -h[j+k] for valid j: j+k valid means 0<=j+k<n, i.e., -k<=j<n-k
        j_lo2 = max(0, -k)
        j_hi2 = min(n - 1, n - 1 - k)
        if j_lo2 <= j_hi2:
            jj2 = np.arange(j_lo2, j_hi2 + 1)
            grad[jj2] -= w * h[jj2 + k]
    return grad

def continuous_polish(h0, n, deadline, max_iter=500):
    """Projected gradient on smooth max."""
    h = h0.copy()
    dx = 2.0 / n
    best_h = h.copy()
    best_val = compute_c5(h, n)
    
    alphas = [100, 500, 2000, 10000]
    for alpha in alphas:
        if time.time() > deadline:
            break
        lr = 1.0 / (alpha * dx * (2 * n - 1))  # rough learning rate
        for it in range(max_iter):
            if time.time() > deadline:
                break
            C = np.correlate(h, 1.0 - h, mode='full') * dx
            m = C.max()
            if m < best_val:
                best_val = m
                best_h = h.copy()
            grad = gradient_smooth_max(h, n, alpha)
            h = h - lr * grad
            # Project onto {0<=h<=1, sum(h)=n/2}
            h = project_feasible(h, n)
            lr *= 0.995
    return best_h, best_val
