import time
import numpy as np


def corr_raw(h):
    n = len(h)
    out = np.empty(2 * n - 1)
    dh = 1.0 - h
    m = n - 1
    out[m] = np.dot(h, dh)
    for k in range(1, n):
        out[m + k] = np.dot(h[:n - k], dh[k:])
        out[m - k] = np.dot(h[k:], dh[:n - k])
    return out, float(np.max(out))


def project_h(x, S):
    x = np.asarray(x, dtype=float)
    if x.size == 0:
        return x
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    y = np.clip(x, 0.0, 1.0)
    if abs(float(y.sum()) - S) < 1e-13:
        return y
    lo = -float(np.max(x))
    hi = 1.0 - float(np.min(x))
    if hi < lo:
        lo, hi = hi, lo
    mid = 0.5 * (lo + hi)
    for _ in range(45):
        mid = 0.5 * (lo + hi)
        y = np.clip(x + mid, 0.0, 1.0)
        if y.sum() < S:
            lo = mid
        else:
            hi = mid
    return np.clip(x + mid, 0.0, 1.0)


def fix_sum(h, S):
    h = h.copy()
    diff = S - float(h.sum())
    if abs(diff) > 1e-14:
        for i in range(h.size):
            v = h[i] + diff
            if 0.0 <= v <= 1.0:
                h[i] = v
                break
        else:
            h = project_h(h, S)
    return np.clip(h, 0.0, 1.0)


def weighted_grad_raw(h, idx, weights):
    n = len(h)
    g = np.zeros(n)
    m = n - 1
    ws = np.asarray(weights, dtype=float)
    s = float(ws.sum())
    if s <= 0.0 or len(idx) == 0:
        return g
    ws = ws / s
    for ii in range(len(idx)):
        wi = float(ws[ii])
        if wi <= 1e-16:
            continue
        k = int(idx[ii]) - m
        if k >= 0:
            vv = n - k
            g[:vv] += wi * (1.0 - h[k:k + vv])
            g[k:k + vv] += -wi * h[:vv]
        else:
            kk = -k
            vv = n - kk
            g[kk:kk + vv] += wi * (1.0 - h[:vv])
            g[:vv] += -wi * h[kk:kk + vv]
    return g


def grad_smooth(h, r, temp, max_lags=80):
    L = len(r)
    mx = float(np.max(r))
    temp = max(1e-12, float(temp))
    v = (r - mx) / temp
    v = np.maximum(v, -60.0)
    w = np.exp(v)
    if max_lags < L:
        idx = np.argpartition(w, -max_lags)[-max_lags:]
        ws = w[idx]
        s = float(ws.sum())
        if s <= 0.0:
            idx = np.array([int(np.argmax(r))])
            ws = np.array([1.0])
        else:
            ws = ws / s
    else:
        idx = np.arange(L)
        s = float(w.sum())
        if s <= 0.0:
            idx = np.array([int(np.argmax(r))])
            ws = np.array([1.0])
        else:
            ws = w / s
    return weighted_grad_raw(h, idx, ws)


def projected_gradient(h, n, deadline, step_init=0.03):
    fracs = (0.20, 0.10, 0.05, 0.02, 0.008)
    S = n * 0.5
    h = project_h(h, S)
    r, rmax = corr_raw(h)
    best_h = h.copy()
    best_r = rmax
    left = len(fracs)
    for ph, frac in enumerate(fracs):
        rem = max(0.0, deadline - time.time())
        phase_dl = min(deadline, time.time() + max(2.0, rem / max(1, left)))
        step = step_init * (0.85 ** ph)
        stuck = 0
        while time.time() < phase_dl:
            temp = max(1e-9, frac * max(rmax, 1.0))
            g = grad_smooth(h, r, temp, 80)
            g -= g.mean()
            gn = float(np.linalg.norm(g))
            if gn < 1e-14:
                break
            d = g / gn
            t = step
            acc = False
            for _ in range(4):
                y = project_h(h - t * d, S)
                ry, rymax = corr_raw(y)
                if rymax < rmax - 1e-10:
                    h = y
                    r = ry
                    rmax = rymax
                    acc = True
                    break
                t *= 0.5
            if acc:
                stuck = 0
                if rmax < best_r:
                    best_r = rmax
                    best_h = h.copy()
            else:
                stuck += 1
                step *= 0.65
                if stuck > 25 or step < 1e-5:
                    break
        left -= 1
    return best_h, float(best_r * 2.0 / n)


def pair_delta_raw(h, n, inc, dec, d, out=None):
    if out is None:
        out = np.zeros(2 * n - 1)
    else:
        out[:] = 0.0
    m = n - 1
    hi = h[inc]
    hj = h[dec]
    q = d * d
    out[m] = d * (1.0 - 2.0 * hi) - q - d * (1.0 - 2.0 * hj) - q

    if inc > 0:
        out[m - inc:m] += d * (1.0 - h[:inc])
    if inc + 1 < n:
        out[m + 1:m + n - inc] += d * (1.0 - h[inc + 1:])
    out[m + dec - inc] += d * hi + q

    if inc > 0:
        out[m + 1:m + 1 + inc] += -d * h[inc - 1::-1]
    if inc + 1 < n:
        out[inc:m] += -d * h[n - 1:inc:-1]
    out[m + inc - dec] += d * h[dec]

    if dec > 0:
        out[m - dec:m] += -d * (1.0 - h[:dec])
    if dec + 1 < n:
        out[m + 1:m + n - dec] += -d * (1.0 - h[dec + 1:])
    out[m + inc - dec] += -d * hj + q

    if dec > 0:
        out[m + 1:m + 1 + dec] += d * h[dec - 1::-1]
    if dec + 1 < n:
        out[dec:m] += d * h[n - 1:dec:-1]
    out[m + dec - inc] += -d * h[inc]
    return out


def pair_polish(h, n, deadline, start_scale=0.03, D=None, min_scale=1e-5):
    S = n * 0.5
    h = project_h(h, S)
    r, rmax = corr_raw(h)
    best_h = h.copy()
    best_r = rmax
    if D is None:
        D = int(min(36, max(16, n // 10)))
    scale = start_scale
    pairs = None
    accept = 0
    reduce = 0
    dc = np.empty(2 * n - 1)
    tmp = np.empty(2 * n - 1)

    while time.time() < deadline:
        if rmax < best_r:
            best_r = rmax
            best_h = h.copy()

        if pairs is None or accept % 25 == 0:
            g = grad_smooth(h, r, max(1e-9, 0.03 * max(rmax, 1.0)), 80)
            g -= g.mean()
            order = np.argsort(g)
            recv = order[:D]
            dono = order[-D:][::-1]
            pairs = []
            for a in dono:
                if h[a] <= 1e-15:
                    continue
                for b in recv:
                    if a != b and h[b] < 1.0 - 1e-15:
                        pairs.append((float(g[a] - g[b]), int(a), int(b)))
            pairs.sort(reverse=True)

        improved = False
        for pred, a, b in pairs:
            if pred <= 1e-13:
                break
            d = scale
            if h[a] < d or h[b] > 1.0 - d:
                continue
            dc = pair_delta_raw(h, n, b, a, d, out=dc)
            np.add(r, dc, out=tmp)
            nm = float(np.max(tmp))
            if nm < rmax - 1e-10:
                h[a] -= d
                h[b] += d
                r += dc
                rmax = nm
                improved = True
                accept += 1
                if rmax < best_r:
                    best_r = rmax
                    best_h = h.copy()
                break

        if improved:
            if accept % 100 == 0:
                rr, rrmax = corr_raw(h)
                if abs(rrmax - rmax) > 1e-7 * max(1.0, rmax):
                    r = rr
                    rmax = rrmax
                if rrmax < best_r:
                    best_r = rrmax
                    best_h = h.copy()
            continue

        if scale > min_scale:
            scale *= 0.5
            reduce += 1
            if reduce > 15:
                pairs = None
                scale = max(min_scale * 10.0, start_scale * 0.25)
                reduce = 0
            continue

        done = False
        for tt, fr in ((0.01, 0.05), (0.005, 0.02), (0.002, 0.005)):
            g = grad_smooth(h, r, max(1e-9, fr * max(rmax, 1.0)), 60)
            g -= g.mean()
            gn = float(np.linalg.norm(g))
            if gn < 1e-13:
                continue
            y = project_h(h - tt * g / gn, S)
            ry, rymax = corr_raw(y)
            if rymax < rmax - 1e-10:
                h = y
                r = ry
                rmax = rymax
                accept += 1
                scale = min(start_scale, 0.02)
                pairs = None
                done = True
                if rmax < best_r:
                    best_r = rmax
                    best_h = h.copy()
                break
        if not done:
            break

    best_h = fix_sum(best_h, n * 0.5)
    rr, rrmax = corr_raw(best_h)
    return best_h, float(rrmax * 2.0 / n)


def optimize_candidate(h, n, deadline, step0=0.03, pg_fraction=0.35, pair_scale=0.04):
    h = project_h(h, n * 0.5)
    r, rmax = corr_raw(h)
    best_h = h.copy()
    best_c = float(rmax * 2.0 / n)
    now = time.time()
    if now + 5.0 < deadline:
        pg_dl = min(deadline - 15.0, now + max(5.0, (deadline - now) * pg_fraction))
        h1, c1 = projected_gradient(h, n, pg_dl, step0)
        if c1 < best_c:
            best_h = h1
            best_c = c1
    now = time.time()
    if now + 3.0 < deadline - 2.0:
        h2, c2 = pair_polish(best_h, n, deadline - 2.0, pair_scale)
        if c2 < best_c:
            best_h = h2
            best_c = c2
    rr, rrmax = corr_raw(best_h)
    return best_h, float(rrmax * 2.0 / n)


def even_int(x):
    x = int(round(float(x)))
    if x <= 0:
        x = 2
    if x & 1:
        x += 1
    return x


def resample_h(h, newn):
    n = len(h)
    if newn == n:
        return h.copy()
    dx = 2.0 / n
    x = np.arange(n) * dx + 0.5 * dx
    dxn = 2.0 / newn
    xnew = np.arange(newn) * dxn + 0.5 * dxn
    y = np.interp(xnew, x, h)
    return project_h(y, newn * 0.5)


def make_bump(n, p):
    dx = 2.0 / n
    x = np.arange(n) * dx + 0.5 * dx
    t = np.clip(1.0 - np.abs(x - 1.0), 0.0, 1.0)
    y = np.power(t, p)
    s = float(y.sum())
    if s > 0.0:
        y *= (n * 0.5) / s
    return project_h(y, n * 0.5)


def make_random_start(n, seed, p=1.0):
    h = make_bump(n, p)
    rng = np.random.RandomState(seed)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    noise = np.zeros(n)
    for m in range(1, 6):
        noise += (rng.rand() * 0.06 - 0.03) * np.sin(np.pi * m * x + rng.rand() * 6.28318)
    return project_h(h + noise, n * 0.5)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + max(10.0, float(budget_s)) - 5.0

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 300
        h0 = make_bump(n0, 1.0)

    h0 = project_h(h0, n0 * 0.5)
    best_n = int(n0)
    r, rmax = corr_raw(h0)
    best_c = float(rmax * 2.0 / best_n)
    best_h = h0

    hs = 0.5 * (h0 + h0[::-1])
    hs = project_h(hs, n0 * 0.5)
    rs, rsmax = corr_raw(hs)
    cs = float(rsmax * 2.0 / n0)
    if cs < best_c:
        best_c = cs
        best_h = hs

    total = max(10.0, deadline - t0)
    main_dl = t0 + max(30.0, total * 0.36)
    if main_dl > deadline - 20.0:
        main_dl = deadline - 20.0
    if time.time() + 5.0 < main_dl:
        h1, c1 = optimize_candidate(best_h, best_n, main_dl, 0.03, 0.35, 0.04)
        if c1 < best_c:
            best_c = c1
            best_h = h1
            best_n = len(h1)

    now = time.time()
    if now < deadline - 220.0 and best_n < 800:
        nh = min(800, even_int(max(500, 2 * best_n)))
        if nh != best_n:
            hh = resample_h(best_h, nh)
            hhs = 0.5 * (hh + hh[::-1])
            hhs = project_h(hhs, nh * 0.5)
            r1, rm1 = corr_raw(hh)
            c1r = float(rm1 * 2.0 / nh)
            r2, rm2 = corr_raw(hhs)
            c2r = float(rm2 * 2.0 / nh)
            start_h = hhs if c2r < c1r else hh
            high_dl = min(deadline - 120.0, now + 380.0)
            if high_dl > now + 60.0:
                h2, c2 = optimize_candidate(start_h, nh, high_dl, 0.02, 0.30, 0.03)
                if c2 < best_c:
                    best_c = c2
                    best_h = h2
                    best_n = len(h2)

    now = time.time()
    if now < deadline - 70.0:
        nt = min(800, even_int(max(best_n, 500)))
        for p in (0.55, 0.75, 1.15):
            if time.time() > deadline - 45.0:
                break
            hb = make_bump(nt, p)
            alloc = min(40.0, deadline - time.time() - 30.0)
            if alloc > 20.0:
                h3, c3 = optimize_candidate(hb, nt, time.time() + alloc, 0.02, 0.35, 0.05)
                if c3 < best_c:
                    best_c = c3
                    best_h = h3
                    best_n = len(h3)

        if time.time() < deadline - 60.0:
            hr = make_random_start(best_n, int(seed) + 1)
            hl = make_random_start(best_n, int(seed) + 2)
            r1, rm1 = corr_raw(hr)
            cr1 = float(rm1 * 2.0 / best_n)
            r2, rm2 = corr_raw(hl)
            cr2 = float(rm2 * 2.0 / best_n)
            start_hr = hr if cr1 < cr2 else hl
            rem = deadline - time.time() - 30.0
            if rem > 25.0:
                h4, c4 = optimize_candidate(start_hr, best_n, time.time() + min(70.0, rem), 0.02, 0.35, 0.05)
                if c4 < best_c:
                    best_c = c4
                    best_h = h4
                    best_n = len(h4)

    hs = 0.5 * (best_h + best_h[::-1])
    hs = project_h(hs, best_n * 0.5)
    rs, rsmax = corr_raw(hs)
    cs = float(rsmax * 2.0 / best_n)
    if cs < best_c:
        best_c = cs
        best_h = hs

    if time.time() < deadline - 2.0:
        hf, cf = pair_polish(best_h, best_n, deadline - 2.0, 0.015)
        if cf < best_c:
            best_c = cf
            best_h = hf

    h = fix_sum(best_h, best_n * 0.5)
    r, rmax = corr_raw(h)
    return h, float(rmax * 2.0 / best_n), int(best_n)
