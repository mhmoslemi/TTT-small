def swap_delta_vec(h, i, j, n):
    """Vectorized swap delta computation."""
    f = n - 1
    delta = np.zeros(2 * n - 1)
    
    # h[i]: 0 -> 1
    m1 = np.arange(max(0, f-i), min(2*n-1, f-i+n))
    if len(m1) > 0:
        idx1 = m1 - f + i
        delta[m1] += (1.0 - h[idx1])
    
    m2 = np.arange(max(0, i+f-n+1), min(2*n-1, i+f+1))
    if len(m2) > 0:
        idx2 = i - m2 + f
        delta[m2] -= h[idx2]
    
    # h[j]: 1 -> 0
    m3 = np.arange(max(0, f-j), min(2*n-1, f-j+n))
    if len(m3) > 0:
        idx3 = m3 - f + j
        delta[m3] -= (1.0 - h[idx3])
    
    m4 = np.arange(max(0, j+f-n+1), min(2*n-1, j+f+1))
    if len(m4) > 0:
        idx4 = j - m4 + f
        delta[m4] += h[idx4]
    
    return delta

    h = binarize(h, n)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    cur_c = float(corr.max())
    best_c = cur_c
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break

        zeros = np.where(h == 0)[0]

        best_delta_c = 0
        best_swap = None
        for _ in range(N):

            j = int(rng.choice(ones))
            d = swap_delta_vec(h, i, j, n)

            delta_c = nc - cur_c

                best_swap = (i, j, d)

            h[i], h[j] = 1.0, 0.0
            corr += d
            cur_c += best_delta_c
            no_improve = 0
            if cur_c < best_c:

                best_h = h.copy()

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def get_corr(h):
    return np.correlate(h, 1 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)

def swap_delta(h, i, j, n):
    f = n - 1
    delta = np.zeros(2 * n - 1)
    m1 = np.arange(max(0, f - i), min(2*n-1, f - i + n))

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    def obj(x):
        c = get_corr(x) * dx
        m = c.max()
        sh = alpha * (c - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    try:

                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-13})
        return project_feasible(res.x, n), float(res.fun)
