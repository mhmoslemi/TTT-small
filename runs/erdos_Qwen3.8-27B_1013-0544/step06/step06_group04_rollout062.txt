        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):

    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def local_search_binary(h, n, deadline, max_sweeps=2000, seed=42):
    """Greedy local search with single swaps on binary h."""
    h = binarize(h, n).astype(float)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    cur_c = float(corr.max()) * dx
    best_c = cur_c
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(h == 1)[0].copy()
        zeros = np.where(h == 0)[0].copy()
        rng.shuffle(ones)
        rng.shuffle(zeros)
        for i in zeros:
            i = int(i)
            if time.time() > deadline:
                break
            for j in ones:
                j = int(j)
                # Compute delta for swapping i and j
                # Using the formula from before
                delta = compute_swap_delta(h, i, j, n)
                nc = float((corr + delta).max()) * dx
                if nc < cur_c - 1e-15:
                    h[i], h[j] = 1.0, 0.0
                    corr = corr + delta
                    cur_c = nc
                    if nc < best_c:
                        best_c = nc
                        best_h = h.copy()
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return best_h, best_c

def compute_swap_delta(h, i, j, n):
    """Compute correlation change when swapping positions i (0->1) and j (1->0)."""
    delta = np.zeros(2 * n - 1)
    ni1 = n - 1
    
    # h[i]: 0 -> 1
    # Effect on c[k] = sum_m h[m]*(1-h[m+k-ni1]):
    # As m=i: gain (1-h_new[i+k-ni1]) for valid k
    # As m+k-ni1=i: lose h_new[m] * (old (1-h[i])=1) for valid k
    
    # As m=i (h[i] new=1):
    lo = max(0, ni1 - i)
    hi = min(2 * n - 2, 2 * n - 2 - i)
    if lo <= hi:
        k_range = np.arange(lo, hi + 1)
        idx = i + k_range - ni1  # position of (1-h) factor
        # Use OLD h for idx != i,j; for idx==j use new h[j]=0
        vals = 1.0 - h[idx]
        # Correct for j: if idx==j, old h[j]=1 so vals=0, but new h[j]=0 so should be 1
        mask = (idx == j)
        if mask.any():
            vals[mask] = 1.0  # new (1-h[j]) = 1
        delta[k_range] += vals
    
    # As m+k-ni1=i (i.e., (1-h[i]) old=1, new=0): lose h_new[m] for valid m
    # k = ni1 + i - m, valid when 0 <= k <= 2n-2, i.e., ni1+i-(2n-2) <= m <= ni1+i
    # m range: max(0, ni1+i-(2n-2)) to min(n-1, ni1+i)
    m_lo = max(0, ni1 + i - (2 * n - 2))
    m_hi = min(n - 1, ni1 + i)
    if m_lo <= m_hi:
        m_range = np.arange(m_lo, m_hi + 1)
        k_range = ni1 + i - m_range
        vals = h[m_range].copy()
        # Correct for m==j: old h[j]=1, new h[j]=0
        mask = (m_range == j)
        if mask.any():
            vals[mask] = 0.0  # new h[j] = 0
        delta[k_range] -= vals
    
    # h[j]: 1 -> 0
    # As m=j (h[j] new=0): lose (1-h_new[j+k-ni1]) for valid k
    lo = max(0, ni1 - j)
    hi = min(2 * n - 2, 2 * n - 2 - j)
    if lo <= hi:
        k_range = np.arange(lo, hi + 1)
        idx = j + k_range - ni1
        vals = 1.0 - h[idx]
        # Correct for idx==i: old h[i]=0 so vals=1, but new h[i]=1 so should be 0
        mask = (idx == i)
        if mask.any():
            vals[mask] = 0.0
        delta[k_range] -= vals
    
    # As m+k-ni1=j (i.e., (1-h[j]) old=0, new=1): gain h_new[m] for valid m
    m_lo = max(0, ni1 + j - (2 * n - 2))
    m_hi = min(n - 1, ni1 + j)
    if m_lo <= m_hi:
        m_range = np.arange(m_lo, m_hi + 1)
        k_range = ni1 + j - m_range
        vals = h[m_range].copy()
        # Correct for m==i: old h[i]=0, new h[i]=1
        mask = (m_range == i)
        if mask.any():
            vals[mask] = 1.0
        delta[k_range] += vals
    
    return delta

def smooth_optimize(h, n, alpha, maxiter=300, method='SLSQP'):
    """SLSQP optimization with softmax-regularized max."""
    dx = 2.0 / n
    n_corr = 2 * n - 1
    
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        # Softmax smoothing
        shifted = alpha * (g - m)
        sm = shifted.max()
        softmax_sum = np.exp(shifted - sm).sum()
        return m + np.log(softmax_sum) / alpha
    
    def obj_grad(x):

        m = g.max()

        sm = shifted.max()

        p = exp_vals / exp_vals.sum()  # softmax weights

        # Actually grad of the whole objective:
        # d(m + log(sum(exp(alpha*(g-m)))))/alpha

        # = (1/alpha) * dm/dx + sum_k p_k * dg_k/dx - sum_k p_k * dm/dx
        # = (1/alpha * (1-sum(p_k))) * dm/dx + sum_k p_k * dg_k/dx
        # Since sum(p_k) = 1: = sum_k p_k * dg_k/dx
        # 
        # Wait that's not right. Let me redo:
        # f = m + (1/alpha)*log(sum_k exp(alpha*(g_k - m)))
        # df/dx = dm/dx + (1/alpha) * [sum_k exp(alpha*(g_k-m)) * alpha * (dg_k/dx - dm/dx)] / sum_k exp(alpha*(g_k-m))
        # = dm/dx + sum_k p_k * (dg_k/dx - dm/dx)
        # = dm/dx * (1 - sum_k p_k) + sum_k p_k * dg_k/dx
        # = sum_k p_k * dg_k/dx  (since sum p_k = 1)
        #
        # So the gradient is simply sum_k p_k * dg_k/dx!
        #
        # Now dg_k/dx: g_k = dx * sum_m x[m] * (1-x[m+k-ni1])
        # dg_k/dx[i] = dx * [(1-x[i+k-ni1]) - x[m] where m+k-ni1=i, for valid indices]
        #            = dx * [(1-x[i+k-ni1]) - x[ni1+i-k]] (for valid indices, 0 otherwise)
        
        # This is O(n * (2n)) = O(n²) which is fine for n=250
        grad = np.zeros(n)
        ni1 = n - 1
        for i in range(n):
            # First term: (1-x[i+k-ni1]) summed over valid k with weight p_k
            # Valid k: 0 <= i+k-ni1 <= n-1, i.e., ni1-i <= k <= 2n-2-i
            k_lo = max(0, ni1 - i)
            k_hi = min(2 * n - 2, 2 * n - 2 - i)
            if k_lo <= k_hi:
                idx = i + np.arange(k_lo, k_hi + 1) - ni1
                grad[i] += dx * np.sum(p[k_lo:k_hi+1] * (1 - x[idx]))
            
            # Second term: -x[ni1+i-k] summed over valid k with weight p_k
            # Valid k: 0 <= ni1+i-k <= n-1, i.e., ni1+i-(n-1) <= k <= ni1+i
            # = i <= k <= ni1+i
            k_lo2 = max(0, i)
            k_hi2 = min(2 * n - 2, ni1 + i)
            if k_lo2 <= k_hi2:
                m_idx = ni1 + i - np.arange(k_lo2, k_hi2 + 1)
                grad[i] -= dx * np.sum(p[k_lo2:k_hi2+1] * x[m_idx])
        
        return grad
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 
             'jac': lambda x: np.ones(n)}]
    
    try:

                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})

def smooth_optimize_simple(h, n, alpha, maxiter=300):

    k_min = max(0, ni1-i, i, ni1-j, j)
    k_max = min(2*n-2, 2*n-2-i, ni1+i, 2*n-2-j, ni1+j)
    
    h_new = h.copy()
    h_new[i] = 1.0
    h_new[j] = 0.0
    
    for k in range(int(k_min), int(k_max) + 1):
        d = 0.0
        # Collect all affected m values (with their roles)
        # m=i (first factor for i): valid if i+k-ni1 in [0,n-1]
        # m=ni1+i-k (second factor for i): valid if 0<=ni1+i-k<n and ni1+i-k != i
        # m=j (first factor for j): valid if j+k-ni1 in [0,n-1]
        # m=ni1+j-k (second factor for j): valid if 0<=ni1+j-k<n and ni1+j-k != j
        # Also need: m values should be distinct (avoid double counting)
        
        m1 = i
        m2 = ni1 + i - k
        m3 = j
        m4 = ni1 + j - k
        
        # For each unique m in {m1, m2, m3, m4} that's valid:
        # contribution = h_new[m]*(1-h_new[m+k-ni1]) - h[m]*(1-h[m+k-ni1])
        
        seen = set()
        for m in [m1, m2, m3, m4]:
            if m in seen:
                continue
            seen.add(m)
            if 0 <= m < n:
                idx = m + k - ni1
                if 0 <= idx < n:
                    d += h_new[m] * (1 - h_new[idx]) - h[m] * (1 - h[idx])
        delta[k] = d
    
    return delta
