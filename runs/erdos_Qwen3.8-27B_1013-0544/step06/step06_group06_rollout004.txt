import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def corr_and_weights(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    w = np.exp(sh - sm)
    w /= w.sum()
    return c, m, w

def smooth_obj(h, n, alpha):
    c, m, w = corr_and_weights(h, n, alpha)
    dx = 2.0 / n
    sh = alpha * (c - m)
    sm = sh.max()
    return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

def smooth_grad(h, n, alpha):
    dx = 2.0 / n
    c, m, w = corr_and_weights(h, n, alpha)
    g = np.zeros(n)
    ks = np.arange(2 * n - 1)
    for j in range(n):
        q1 = j + ks - n + 1
        mask1 = (q1 >= 0) & (q1 < n)
        term1 = np.where(mask1, 1.0 - h[np.clip(q1, 0, n-1)], 0.0)
        q2 = j - ks + n - 1
        mask2 = (q2 >= 0) & (q2 < n)
        term2 = np.where(mask2, h[np.clip(q2, 0, n-1)], 0.0)
        g[j] = (w * (term1 - term2)).sum() * dx
    return g

def slsqp_run(h0, n, alpha, maxiter=2000, tol=1e-16):
    dx = 2.0 / n
    def obj(x):
        return smooth_obj(x, n, alpha)
    def jac(x):
        return smooth_grad(x, n, alpha)
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=jac, method='SLSQP', bounds=bounds,
                       constraints=constraints, options={'maxiter': maxiter, 'ftol': tol})
        return project_feasible(res.x, n)
    except Exception:
        return h0

    target = n // 2

    h[idx[:target]] = 1.0
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    zeros = np.where(h == 0)[0]
    ones = np.where(h == 1)[0]
    stall = 0
    temp = 0.003
    no_improve = 0

    while time.time() < deadline:
        top_idx = np.argsort(corr)[-top_k:]
        best_gain = 1e-14
        best_i, best_j = -1, -1

            if i + n - 1 > 2 * n - 2:

    best_h = h0.copy()
    best_c = compute_c5(h0, n)
