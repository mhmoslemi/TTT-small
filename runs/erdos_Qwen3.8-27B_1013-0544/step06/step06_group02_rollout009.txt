```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-13:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-13:
                break
            h *= target / tot
    return np.clip(h, 0.0, 1.0)

def smax_val(c, alpha):
    m = c.max()
    s = alpha * (c - m)
    s -= s.max()
    return float(m + (s.max() + np.log(np.sum(np.exp(s)))) / alpha)

def smax_grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    s = alpha * (c - m)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:
            nv = n - k
            grad[:nv] -= wm * h[k:]
            grad[k:] += wm * (1.0 - h[:nv])
        else:
            r = -k
            nv = n - r
            grad[:nv] -= wm * h[r:]
            grad[r:] += wm * (1.0 - h[:nv])
    return grad

def obj_pen(x, n, alpha, mu):
    return smax_val(compute_c5_vec(x, n), alpha) + mu * (x.sum() - n / 2.0) ** 2

def grad_pen(x, n, alpha, mu):
    return smax_grad(x, n, alpha) + 2.0 * mu * (x.sum() - n / 2.0) * np.ones(len(x))

def obj_sl(x, n, alpha):
    return smax_val(compute_c5_vec(x, n), alpha)

def cons_eq(x, n):
    return x.sum() - n / 2.0

def jac_eq(x, n):
    return np.ones(len(x))

def optimize_lbfgs(h, n, deadline, alphas):
    bounds = [(0.0, 1.0)] * n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha, mu, maxit in alphas:
        rem = deadline - time.time()
        if rem < 3:
            break
        try:
            res = minimize(obj_pen, best_h, jac=grad_pen, args=(n, alpha, mu),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': maxit, 'ftol': 1e-18, 'gtol': 1e-14})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def optimize_slsqp(h, n, deadline, alphas, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_eq, 'jac': jac_eq, 'args': (n,)}]
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alphas:
        rem = deadline - time.time()
        if rem < 5:
            break
        try:
            res = minimize(obj_sl, best_h, args=(n, alpha), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-16})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def gradient_local_search(h, n, deadline, alpha=10000):
    """Use gradient to guide coordinate moves."""
```

Actually, one more important idea: **iterative reweighted least squares (IRLS)** approach.

At each iteration:
1. Compute c_k for all k
2. Find the k* that achieves the maximum
3. Minimize c_{k*} (a single quadratic) subject to constraints
4. Repeat

This is a simple "pursuit of the worst lag" strategy. It's like hot-spotting.

```python
def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_vec(h, n)))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h *= target / tot
    return np.clip(h, 0.0, 1.0)

def smax_val(c, alpha):
    m = c.max()
    s = alpha * (c - m)
    s -= s.max()
    return float(m + (s.max() + np.log(np.sum(np.exp(s)))) / alpha)

def smax_grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    s = alpha * (c - m)
    s -= s.max()
    w = np.exp(s)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        wm = wi * dx
        if k >= 0:

            grad[:nv] -= wm * h[k:]
            grad[k:] += wm * (1.0 - h[:nv])
        else:
            r = -k
            nv = n - r
            grad[:nv] -= wm * h[r:]
            grad[r:] += wm * (1.0 - h[:nv])
    return grad
```
