def project_feasible(h, n):

            break

            tot = room.sum()

            h += (diff / tot) * room

            if tot < 1e-15:

            h += (diff / tot) * h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    cs = alpha * (c - c.max())
    mx = cs.max()
    w = np.exp(cs - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-12:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def _pobj(x, n, alpha, mu):
    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _pggrad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def lbfgs_phase(h, n, deadline, max_iters=200):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(300, 5.0), (1000, 10.0), (5000, 50.0),
               (20000, 200.0), (100000, 1000.0), (500000, 5000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(_pobj, h, jac=_pggrad, args=(n, alpha, mu),
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iters, 'gtol': 1e-14,
                                   'maxfun': max_iters * 5})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def nesterov_phase(h, n, deadline, alpha_init=500.0, lr_init=0.005):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    v = np.zeros(n)
    alpha = alpha_init
    lr = lr_init
    mom = 0.85
    stall = 0
    for it in range(200000):
        if time.time() > deadline:
            break
        g = grad_smooth_max(h, n, alpha)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        mom = min(0.99, mom * 1.0005)
        v = mom * v - lr * g / gn
        h_new = project_feasible(h + v, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            stall = 0
        else:
            h = project_feasible(h - 0.5 * v, n)
            stall += 1
            if stall % 50 == 0:
                lr *= 0.7
                v *= 0.3
            if stall % 200 == 0:
                alpha *= 1.5
        if it % 500 == 0 and it > 0:
            alpha *= 1.2
        if lr < 1e-14:
            break
    return best_h, best_c

def pair_search(h, n, deadline, radius=80, steps=None):
    if steps is None:
        steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    best_c = compute_c5(h, n)
    for d in steps:
        improved_total = False
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    for dd in (d, -d):
                        if h[i] + dd >= 0 and h[i] + dd <= 1 and h[j] - dd >= 0 and h[j] - dd <= 1:
                            h[i] += dd
                            h[j] -= dd
                            c = compute_c5(h, n)
                            if c < best_c - 1e-13:
                                best_c = c
                                improved = True
                                improved_total = True
                            else:
                                h[i] -= dd
                                h[j] += dd
                    if improved:
                        break
            if not improved:
                break
    return h, best_c

def try_res(h, c, n_old, n_new, deadline):
    if time.time() > deadline:
        return h, c, n_old
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    h_new = np.interp(x_new, x_old, h)
    h_new = project_feasible(h_new, n_new)
    c_new = compute_c5(h_new, n_new)
    t_budget = min(180, deadline - time.time() - 40)

    h_new, c_new = lbfgs_phase(h_new, n_new, time.time() + t_budget * 0.5, max_iters=100)

                                       alpha_init=1000, lr_init=0.003)
    if c_new < c:
        return h_new, c_new, n_new
    return h, c, n_old
