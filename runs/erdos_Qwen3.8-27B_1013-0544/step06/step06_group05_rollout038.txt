def fast_swap_update(c, h, i, j, n):
    """Compute delta in correlation after swapping h[i] 0->1 and h[j] 1->0.
    Return new correlation array (modified in place or new)."""
    # i was 0, becomes 1
    # j was 1, becomes 0
    # For lag l (ranging from -(n-1) to n-1):
    # c[l] = sum_p h[p]*(1-h[p+l])
    # Changes at p=i (h goes 0->1): adds (1-h[i+l])
    # Changes at p=j (h goes 1->0): removes (1-h[j+l])
    # Changes at p=i-l (i.e., p+l=i, h[i] goes 0->1): removes h[p]*(1-h[i]) = h[p]*1 -> h[p]*0, so subtract h[p]
    #   Wait, (1-h[i]) goes from 1 to 0, so the term h[p]*(1-h[i]) goes from h[p] to 0. Change: -h[p]
    # Changes at p=j-l (i.e., p+l=j, h[j] goes 1->0): h[j] goes 1->0, (1-h[j]) goes 0->1
    #   term h[p]*(1-h[j]) goes from 0 to h[p]. Change: +h[p]
    
    new_c = c.copy()
    # For each lag l from -(n-1) to n-1:
    # But that's 2n-1 = 499 lags, each O(1) work
    # Total O(n) per swap
    
    for l in range(-(n-1), n):
        delta = 0
        # p=i term: h[i] was 0, now 1. Contribution was 0, now (1-h_new[i+l])
        # i+l must be valid: 0 <= i+l < n
        if 0 <= i+l < n:
            # h_new[i+l] = h[i+l] unless i+l == j
            if i+l == j:
                h_new_val = 0  # h[j] is now 0
            else:
                h_new_val = h[i+l]
            delta += (1 - h_new_val)
        # p=j term: h[j] was 1, now 0. Contribution was (1-h[j+l]), now 0
        if 0 <= j+l < n:
            if j+l == i:
                h_new_val = 1  # h[i] is now 1
            else:
                h_new_val = h[j+l]
            delta -= (1 - h_new_val)
        # p = i-l term: (1-h[i]) goes from 1 to 0
        p = i - l
        if 0 <= p < n:
            delta -= h[p]  # was h[p]*1, now h[p]*0
        # p = j-l term: (1-h[j]) goes from 0 to 1
        p = j - l
        if 0 <= p < n:
            delta += h[p]  # was h[p]*0, now h[p]*1
        new_c[l + n - 1] += delta
    return new_c

def compute_delta_c(h, i, j, n):
    """Compute Δc[l] for swapping h[i]:0->1 and h[j]:1->0. Returns array of length 2n-1."""
    # l ranges from -(n-1) to (n-1), indexed as l + (n-1)
    nc = 2*n - 1
    
    # A[l] for -i <= l <= n-1-i
    A = np.zeros(nc)
    lo = max(-i, -(n-1))
    hi = min(n-1-i, n-1)
    l_range = np.arange(lo, hi+1)
    idx = l_range + (n-1)
    A[idx] = 1.0 - h[i + l_range]
    # Fix special cases
    A[0 + (n-1)] = 0.0  # l=0
    if -i <= (j-i) <= n-1-i:
        A[(j-i) + (n-1)] = 1.0
    
    # B[l] for -j <= l <= n-1-j
    B = np.zeros(nc)
    lo = max(-j, -(n-1))
    hi = min(n-1-j, n-1)
    l_range = np.arange(lo, hi+1)
    idx = l_range + (n-1)
    B[idx] = 1.0 - h[j + l_range]
    
    # D[l] for i-n+1 <= l <= i
    D = np.zeros(nc)
    lo = max(i-(n-1), -(n-1))
    hi = min(i, n-1)
    l_range = np.arange(lo, hi+1)
    idx = l_range + (n-1)
    D[idx] = h[i - l_range]
    
    # E[l] for j-n+1 <= l <= j
    E = np.zeros(nc)
    lo = max(j-(n-1), -(n-1))
    hi = min(j, n-1)
    l_range = np.arange(lo, hi+1)
    idx = l_range + (n-1)
    E[idx] = h[j - l_range]
    
    return A - B - D + E
```

def projected_subgradient(h, n, lr_schedule, max_iter, deadline):
    dx = 2.0/n
    for it in range(max_iter):
        if time.time() > deadline:
            break
        # Compute overlap
        corr = np.correlate(h, 1-h, mode='full') * dx
        k_star = np.argmax(corr)
        # Subgradient
        l = k_star - (n-1)  # lag
        sg = np.zeros(n)
        # For position i: d overlap[k*]/d h[i] = (1-h[i+l])*I(0≤i+l<n) - h[i-l]*I(0≤i-l<n)
        valid1 = (np.arange(n) + l >= 0) & (np.arange(n) + l < n)
        valid2 = (np.arange(n) - l >= 0) & (np.arange(n) - l < n)
        sg[valid1] += (1 - h[np.arange(n)[valid1] + l])
        sg[valid2] -= h[np.arange(n)[valid2] - l]
        # Update
        lr = lr_schedule(it)
        h_new = h - lr * sg
        # Project
        h_new = np.clip(h_new, 0, 1)
        # Fix sum constraint
        h_new = project_feasible(h_new, n)
        # Accept if better
        new_corr = np.correlate(h_new, 1-h_new, mode='full') * dx
        if new_corr.max() <= corr.max() + 1e-10:
            h = h_new
    return h
```

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0-h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h = np.clip(h + diff/s * room, 0, 1)
        else:
            s = h.sum()
            if s < 1e-14: break
            h = np.clip(h + diff/s * h, 0, 1)
    return h

def binarize(h, n):

    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:n//2]] = 1.0
    return b

def greedy_binary(h, n, deadline, max_kicks=100):
    """Greedy local search with random kicks on binary vector."""
    best_h = h.copy()
    best_corr = np.correlate(best_h, 1-best_h, mode='full').astype(np.float64)
    best_max = best_corr.max()
    
    ones = np.where(best_h == 1)[0]
    zeros = np.where(best_h == 0)[0]
    
    kicks = 0
    while time.time() < deadline and kicks < max_kicks:
        improved = False
        best_delta_max = 0  # We want this to be negative (reduce max)
        best_i, best_j = -1, -1
        best_new_corr = None
        
        # Try all swaps
        n_z = len(zeros)
        n_o = len(ones)
        if n_z == 0 or n_o == 0:
            break
        
        for iz in range(n_z):
            i = zeros[iz]
            if best_h[i] != 0:
                continue
            for io in range(n_o):
                j = ones[io]
                if best_h[j] != 1:
                    continue
                dc = delta_c(best_h, i, j, n)
                new_corr = best_corr + dc
                new_max = new_corr.max()
                if new_max < best_max - 1e-10:
                    if not improved or new_max < best_delta_max:
                        best_delta_max = new_max
                        best_i, best_j = i, j
                        best_new_corr = new_corr
                    improved = True
        
        if improved:
            # Apply best swap
            best_h[best_i] = 1.0
            best_h[best_j] = 0.0
            best_corr = best_new_corr
            best_max = best_delta_max
            ones = np.where(best_h == 1)[0]
            zeros = np.where(best_h == 0)[0]
            kicks = 0
        else:
            # No improvement: kick
            kicks += 1
            if kicks > max_kicks:
                break
            # Random swap (accept worse)
            iz = np.random.randint(n_z)
            io = np.random.randint(n_o)

            best_h[i] = 1.0
            best_h[j] = 0.0
            best_corr = np.correlate(best_h, 1-best_h, mode='full').astype(np.float64)
            best_max = best_corr.max()
            ones = np.where(best_h == 1)[0]
            zeros = np.where(best_h == 0)[0]
    
    return best_h, best_corr

def slsqp_polish(h, n, deadline):
    """SLSQP with smooth max objective."""
    dx = 2.0/n
    def obj(x):
        g = np.correlate(x, 1.0-x, mode='full') * dx
        m = g.max()
        alpha = 5000.0
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh-sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n/2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 5000, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except:
        return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0-h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h = np.clip(h + diff/s * room, 0, 1)
        else:
            s = h.sum()
            if s < 1e-14: break
            h = np.clip(h + diff/s * h, 0, 1)
    return h

def binarize(h, n):
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:n//2]] = 1.0
    return b

def delta_c(h, i, j, n):
    nc = 2*n - 1
    off = n - 1
    A = np.zeros(nc)
    lo = max(-i, -(n-1)); hi = min(n-1-i, n-1)
    if lo <= hi:
        lr = np.arange(lo, hi+1)
        A[lr+off] = 1.0 - h[i+lr]
    A[off] = 0.0
    lj = j - i
    if -(n-1) <= lj <= n-1:
        A[lj+off] = 1.0
    B = np.zeros(nc)
    lo = max(-j, -(n-1)); hi = min(n-1-j, n-1)
    if lo <= hi:
        lr = np.arange(lo, hi+1)
        B[lr+off] = 1.0 - h[j+lr]
    D = np.zeros(nc)
    lo = max(i-(n-1), -(n-1)); hi = min(i, n-1)
    if lo <= hi:
        lr = np.arange(lo, hi+1)
        D[lr+off] = h[i-lr]
    E = np.zeros(nc)
    lo = max(j-(n-1), -(n-1)); hi = min(j, n-1)
    if lo <= hi:
        lr = np.arange(lo, hi+1)
        E[lr+off] = h[j-lr]
    return A - B - D + E

def greedy_binary(h, n, deadline, max_kicks=200, seed=42):
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_corr = np.correlate(best_h, 1-best_h, mode='full').astype(np.float64)
    best_max = best_corr.max()
    kicks = 0
    stall = 0
    while time.time() < deadline:
        improved = False
        best_new_max = best_max
        best_i = best_j = -1
        best_new_corr = None
        zeros = np.where(best_h == 0)[0]
        ones = np.where(best_h == 1)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        for i in zeros:
            if best_h[i] != 0: continue
            for j in ones:
                if best_h[j] != 1: continue
                dc = delta_c(best_h, i, j, n)
                nm = (best_corr + dc).max()
                if nm < best_new_max - 1e-9:
                    best_new_max = nm
                    best_i, best_j = i, j
                    best_new_corr = best_corr + dc
                    improved = True
        if improved:
            best_h[best_i] = 1.0
            best_h[best_j] = 0.0
            best_corr = best_new_corr
            best_max = best_new_max
            kicks = 0
            stall = 0
        else:
            kicks += 1
            stall += 1
            if stall > max_kicks:
                break
            # Random kick
            i = zeros[rng.randint(len(zeros))]
            j = ones[rng.randint(len(ones))]
            best_h[i] = 1.0
            best_h[j] = 0.0
            best_corr = np.correlate(best_h, 1-best_h, mode='full').astype(np.float64)
            best_max = best_corr.max()
    return best_h, float(best_max * 2.0/n)

def slsqp_polish(h, n, deadline, alpha=5000):
    dx = 2.0/n
    def obj(x):
        g = np.correlate(x, 1.0-x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh-sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n/2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 8000, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except:
        return h

            if time.time() > dl2: break
            hp = slsqp_polish(best_h.copy(), n, dl2, alpha=alpha)
            cp = compute_c5(hp, n)
            if cp < best_c:

                best_h = hp.copy()
    
    # Phase 3: Re-binarize + greedy (20% of time)

        h_b2, c_b2 = greedy_binary(h_bin2, n, dl3, max_kicks=50, seed=seed+7)
        if c_b2 < best_c:

        if cp < best_c:
            best_c = cp
            best_h = hp.copy()
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * room, 0, 1)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * h, 0, 1)
    return h
