def binarize(h, n):
    """Convert to binary with exactly n/2 ones."""
    target = n // 2
    vals = np.sort(h)[::-1]
    thresh = vals[target - 1]
    h_bin = (h >= thresh).astype(float)
    # Handle ties
    n_ones = int(h_bin.sum())
    if n_ones > target:
        tied = np.where((h_bin == 1) & (np.isclose(h, thresh)))[0]
        excess = n_ones - target
        for idx in tied[:excess]:
            h_bin[idx] = 0
    elif n_ones < target:
        tied = np.where((h_bin == 0) & (np.isclose(h, thresh)))[0]
        deficit = target - n_ones
        for idx in tied[:deficit]:
            h_bin[idx] = 1
    return h_bin

    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def fast_local_search(h_init, n, deadline, max_phases=10000, kick_size=5, seed=42):
    h = h_init.copy().astype(float)
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    cur_max = float(corr.max())
    best_h = h.copy()
    best_max = cur_max
    idx_n = np.arange(n)
    
    for phase in range(max_phases):
        if time.time() > deadline:
            break
        
        if phase > 0:
            h = best_h.copy()
            for _ in range(kick_size):
                oi = np.where(h == 1)[0]
                zi = np.where(h == 0)[0]
                if len(oi) == 0 or len(zi) == 0:
                    break
                a = oi[rng.randint(len(oi))]
                b = zi[rng.randint(len(zi))]
                h[a] = 0.0
                h[b] = 1.0
            corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
            cur_max = float(corr.max())
        
        stall = 0
        while time.time() < deadline and stall < 300:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            
            critical = np.where(corr >= cur_max - 2)[0]
            n_crit = len(critical)
            if n_crit == 0:
                break
            lags = critical - (n - 1)
            corr_crit = corr[critical]
            
            A = np.zeros((n, n_crit))
            B = np.zeros((n, n_crit))
            for k in range(n_crit):
                lag = int(lags[k])
                ip_idx = idx_n + lag
                im_idx = ip_idx - 2 * lag  # = idx_n - lag
                ip_valid = (ip_idx >= 0) & (ip_idx < n)
                im_valid = (im_idx >= 0) & (im_idx < n)
                ip_v = np.where(ip_valid, h[np.clip(ip_idx, 0, n-1)], 0.0)
                im_v = np.where(im_valid, h[np.clip(im_idx, 0, n-1)], 0.0)
                A[:, k] = (1.0 - ip_v) * ip_valid - im_v * im_valid
                B[:, k] = -(1.0 - ip_v) * ip_valid + im_v * im_valid
            
            A_z = A[zeros]
            B_o = B[ones]
            
            best_new_max = cur_max
            best_i_idx = -1
            best_j_idx = -1
            
            for j_idx in range(len(ones)):
                dv = B_o[j_idx, :][None, :] + A_z
                new_maxes = (corr_crit[None, :] + dv).max(axis=1)
                idx = int(new_maxes.argmin())
                if new_maxes[idx] < best_new_max:
                    best_new_max = float(new_maxes[idx])
                    best_i_idx = idx
                    best_j_idx = j_idx
            
            if best_new_max < cur_max - 1e-10 and best_i_idx >= 0:
                i = int(zeros[best_i_idx])
                j = int(ones[best_j_idx])
                h[i] = 1.0
                h[j] = 0.0
                corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
                cur_max = float(corr.max())
                stall = 0
                if cur_max < best_max:
                    best_max = cur_max
                    best_h = h.copy()
            else:
                stall += 1
    
    return best_h, float(best_max * dx)
