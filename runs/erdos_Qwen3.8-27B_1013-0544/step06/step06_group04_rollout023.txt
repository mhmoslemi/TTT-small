    """Compute delta in corr when h[i]: 0->1, h[j]: 1->0"""
    delta = np.zeros(2*n-1)
    g = 1 - h
    
    # Case 1: h[i] 0->1, affects corr[k] where i is in valid range
    # Valid: max(0, n-1-k) <= i <= min(n-1, n-1+k)
    # i.e., n-1-i <= k <= n-1+i, clamped to [0, 2n-2]
    k_lo = max(0, n-1-i)
    k_hi = min(2*n-2, n-1+i)
    for k in range(k_lo, k_hi+1):
        p = i + k - (n-1)  # partner index
        if p == j:
            delta[k] += 1.0  # (1-h')[j] = 1
        elif 0 <= p < n:
            delta[k] += g[p]

    k_lo = max(0, n-1-j)
    k_hi = min(2*n-2, n-1+j)

        if 0 <= p < n:
            delta[k] -= g[p]

    # partner = i' + k - (n-1) = i => k = n-1 + i - i'
    # i' must be valid: max(0, n-1-k) <= i' <= min(n-1, n-1+k)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    sm = shifted.max()
    logsum = sm + np.log(np.exp(shifted - sm).sum())
    return m + logsum / alpha
