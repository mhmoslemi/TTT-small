    log_sum_exp = mv + np.log(np.exp(vals - mv).sum())

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    vals = alpha * corr * dx
    mv = vals.max()
    return (mv + np.log(np.exp(vals - mv).sum())) / alpha

def slsqp_step(h, n, alpha, maxiter=400):
    def _obj(x):
        return smooth_obj(x, n, alpha)
    def _con(x):
        return x.sum() - n / 2.0
    try:

                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': _con}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except:
        return h
