```python
import numpy as np
from scipy.optimize import fmin_l_bfgs_b
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lseobjective(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    mx = c.max()
    val = mx + np.log(np.sum(np.exp(c - mx))) / alpha
    return val * dx

def lse_gradient(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    mx = c.max()
    e = np.exp(c - mx)
    w = e / e.sum()
    
    w_pos = w[n-1:]  # w[n-1], w[n], ..., w[2n-2], length n
    grad = np.zeros(n)
    omh = 1.0 - h
    
    for j in range(n):
        g = 0.0
        # A: sum s=0 to n-1-j of w_pos[s]*(1-h[j+s])
        if n - j > 0:
            g += np.dot(w_pos[:n-j], omh[j:])
        # B: -sum s=0 to j of w_pos[s]*h[j-s]
        g -= np.dot(w_pos[:j+1], h[j::-1])
        # C: sum t=1 to j of w[n-1-t]*(1-h[j-t])
        if j > 0:
            g += np.dot(w[n-2:n-1-j:-1], omh[:j][::-1])
        # D: -sum t=1 to n-1-j of w[n-1-t]*h[j+t]
        if n - 1 - j > 0:
            g -= np.dot(w[n-2:j-1:-1], h[j+1:])
        grad[j] = (dx / alpha) * g
    
    return grad

def lbfgs_phase(h, n, alpha_list, deadline, maxiter_each=2000):
    for alpha in alpha_list:
        if time.time() > deadline:
            break
        def f(x):
            return lseobjective(x, n, alpha)
        def g(x):
            return lse_gradient(x, n, alpha)
        
        res = fmin_l_bfgs_b(f, h, fprime=g, 
                            pgtol=1e-8, maxiter=maxiter_each,
                            bounds=[(0.0, 1.0)] * n)
        h = project_feasible(res, n)
    return h

def coordinate_descent(h, n, deadline, max_passes=200, grid=50, rng=None):
    """Efficient coordinate descent with line search."""
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode='full')
    current_max = c.max()
    sumh = h.sum()
    target = n / 2.0
    
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        indices = np.arange(n)
        if rng is not None:
            rng.shuffle(indices)
        for p in indices:
            if time.time() > deadline:
                return h
            d_lo = -h[p]
            d_hi = 1.0 - h[p]
            if d_hi - d_lo < 1e-14:
                continue
            
            # Determine which correlation entries are affected
            # h[p] appears in c[k] for k where the correlation involves index p
            # c[k] = sum_i h[i]*(1-h[i+k-n+1])
            # p is involved if p is in [0,n-1] and p+k-n+1 is in [0,n-1]
            # i.e., k-n+1-p in [0,n-1] → n-1-p <= k <= 2n-2-p
            # Also p+k-n+1 >= 0 → k >= n-1-p (same)
            # And p >= 0 (always true)
            # So k ranges from n-1-p to 2n-2-p, which is n entries
            
            k_start = n - 1 - p
            k_end = 2 * n - 2 - p
            ks = np.arange(k_start, k_end + 1)
            
            # For each k in ks, compute the coefficient of d in c[k] + d * coeff
            # c[k] = ... + h[p]*(1-h[p+k-n+1]) + h[p+k-n+1-p... ] 
            # Actually, h[p] appears in c[k] in two ways:
            # 1. As h[i] with i=p: contributes (1-h[p+k-n+1]) if p+k-n+1 in [0,n-1]
            # 2. As h[i+k-n+1] with i+k-n+1=p, i.e., i = p-k+n-1: contributes -h[i] = -h[p-k+n-1] if p-k+n-1 in [0,n-1]
            
            # For k in [k_start, k_end] = [n-1-p, 2n-2-p]:
            # p+k-n+1 ranges from p+k_start-n+1 = p+(n-1-p)-n+1 = 0 to p+k_end-n+1 = p+(2n-2-p)-n+1 = n-1
            # So h[p+k-n+1] is always valid → term 1 is always present: (1-h[p+k-n+1])
            
            # p-k+n-1 ranges from p-k_start+n-1 = p-(n-1-p)+n-1 = 2p to p-k_end+n-1 = p-(2n-2-p)+n-1 = 2p-n+1
            # Valid if 0 <= p-k+n-1 <= n-1, i.e., n-1-p <= k <= n-1+p
            # k in [max(k_start, n-1-p), min(k_end, n-1+p)] = [n-1-p, min(2n-2-p, n-1+p)]
            
            coeff = np.zeros(len(ks))
            # Term 1: (1-h[p+k-n+1]) for all k in ks
            idx1 = p + ks - n + 1
            coeff += (1.0 - h[idx1])
            # Term 2: -h[p-k+n-1] for valid k
            valid2 = (ks >= n - 1 - p) & (ks <= n - 1 + p)
            if np.any(valid2):
                idx2 = p - ks[valid2] + n - 1
                coeff[valid2] -= h[idx2]
            # Special case: k = n-1 (s=0), where h[p] appears as both h[i] and h[i]
            # c[n-1] = sum_i h[i](1-h[i]), so ∂c[n-1]/∂h[p] = 1-2h[p]
            # In our formula: term1 gives (1-h[p]), term2 gives -h[p], total = 1-2h[p] ✓
            # But we also need the quadratic term: -d^2 for k=n-1
            
            # The change in c[k] when h[p] → h[p]+d:
            # c_new[k] = c[k] + d*coeff[k] - d^2 * [k==n-1]
            # (the -d^2 is because h[p]*(1-h[p]) → (h[p]+d)*(1-h[p]-d) = h[p](1-h[p]) + d(1-2h[p]) - d^2)
            
            # Unaffected correlations
            mask = np.ones(2*n-1, dtype=bool)
            mask[ks] = False
            um = c[mask].max() if mask.any() else 0.0
            
            # Line search
            deltas = np.linspace(d_lo, d_hi, grid)
            best_d = 0.0
            best_val = current_max
            for d in deltas:
                if abs(d) < 1e-14:
                    continue
                new_ks = c[ks] + d * coeff
                if n-1 in ks:
                    new_ks[ks == n-1] -= d * d
                val = max(um, new_ks.max())
                # Also penalize sum constraint violation
                val += 0.01 * (sumh + d - target) ** 2
                if val < best_val - 1e-12:
                    best_val = val
                    best_d = d
            
            if abs(best_d) > 1e-10:
                d = best_d
                c[ks] += d * coeff
                if n-1 in ks:
                    c[n-1] -= d * d
                h[p] = np.clip(h[p] + d, 0.0, 1.0)
                sumh = h.sum()
                current_max = max(um, c[ks].max())
                improved = True
        
        if not improved:
            break
    
    return h

def simulated_annealing(h, n, deadline, rng, n_iter=50000):
    """Simulated annealing with swap moves."""
    h = h.copy()
    c5 = compute_c5(h, n)
    best_h = h.copy()
    best_c5 = c5
    T = 0.01  # initial temperature
    T_min = 1e-6
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        # Pick two random positions
        i, j = rng.randint(n, size=2)
        if i == j:
            continue
        # Move mass from i to j (or vice versa)
        d = min(h[i], 1.0 - h[j], 0.05)
        if d < 1e-6:
            continue
        direction = 1 if rng.random() < 0.5 else -1
        h[i] -= direction * d
        h[j] += direction * d
        new_c5 = compute_c5(h, n)
        delta = new_c5 - c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, T_min)):
            c5 = new_c5
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[i] += direction * d
            h[j] -= direction * d
        # Cool down
        T = max(T * 0.9995, T_min)
    
    return best_h, best_c5
```
