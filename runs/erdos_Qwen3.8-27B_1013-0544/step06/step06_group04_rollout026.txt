def compute_gradient(h, n, weights):
    # weights has length 2n-1, indexed 0 to 2n-2
    # grad[j] = sum_l weights[n-1+l] * [(1-h[j-l])*1_{0<=j-l<n} - h[j+l]*1_{0<=j+l<n}]
    # where l ranges from -(n-1) to n-1
    
    # Part 1: sum_l weights[n-1+l] * (1-h[j-l]) * 1_{0<=j-l<n}
    # For each l, this contributes weights[n-1+l] * (1-h[j-l]) to positions j where 0<=j-l<n
    # i.e., j in [l, l+n-1] (clipped to [0, n-1])
    
    grad = np.zeros(n)
    lags = np.arange(-(n-1), n)  # length 2n-1
    w = weights  # length 2n-1, w[i] corresponds to lag lags[i] = i - (n-1)
    
    for idx, l in enumerate(lags):
        w_val = w[idx]
        if w_val == 0:
            continue
        # (1-h[j-l]) for valid j: j-l in [0, n-1], so j in [l, l+n-1] ∩ [0, n-1]
        j_start = max(0, l)
        j_end = min(n-1, l + n - 1)
        if j_start <= j_end:
            i_vals = j_vals - l  # where j_vals = arange(j_start, j_end+1)
            grad[j_start:j_end+1] += w_val * (1 - h[i_vals])
        
        # -h[j+l] for valid j: j+l in [0, n-1], so j in [-l, n-1-l] ∩ [0, n-1]
        j_start2 = max(0, -l)
        j_end2 = min(n-1, n - 1 - l)
        if j_start2 <= j_end2:
            i_vals2 = j_vals2 + l  # where j_vals2 = arange(j_start2, j_end2+1)
            grad[j_start2:j_end2+1] -= w_val * h[i_vals2]
    
    return grad
