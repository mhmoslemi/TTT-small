import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(h, n, alpha):
    """Create objective function for SLSQP."""
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    def obj_grad(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        sm = shifted.max()
        w = np.exp(shifted - sm)
        w = w / w.sum()
        # Gradient of c_k = sum_i x[i] * (1-x[i+k]) * dx
        # dc_k/dx[j] = dx * (1 - x[j+k]) - dx * x[j-k] (with appropriate indexing)
        # Actually, c_k = sum_i x[i] * (1 - x[i+k]) * dx
        # dc_k/dx[j] = dx * sum_i [delta(i,j)*(1-x[i+k]) + x[i]*(-delta(i+k,j))]
        #            = dx * [(1-x[j+k]) - x[j-k]]
        # where i+k=j means i=j-k
        # For the full correlation, k ranges from -(n-1) to (n-1)
        # Let's compute this more carefully
        # c[k] for k in 0..2n-2 corresponds to shift s = k-(n-1) in range -(n-1)..(n-1)
        # c_s = sum_i x[i] * (1-x[i+s]) for valid i
        # dc_s/dx[j] = (1-x[j+s]) - x[j-s] (when indices are valid)
        
        # Let's build the gradient matrix
        grad = np.zeros(n)
        for idx, w_val in enumerate(w):
            if w_val < 1e-15:
                continue
            s = idx - (n - 1)  # shift
            # dc_s/dx[j] = (1-x[j+s]) - x[j-s] for valid indices
            # j+s is valid when 0 <= j+s < n, i.e., -s <= j < n-s
            # j-s is valid when 0 <= j-s < n, i.e., s <= j < n+s
            js = np.arange(max(0, -s), min(n, n - s))
            grad[js] += w_val * (1 - x[js + s])
            jm = np.arange(max(0, s), min(n, n + s))
            grad[jm] -= w_val * x[jm - s]
        grad *= dx
        return grad
    return obj, obj_grad

def slsqp_optimize(h, n, alpha, maxiter=2000, ftol=1e-16):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.success or res.fun < compute_c5(h, n):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)

def binary_sa(h_bin, n, deadline, seed=42, t_start=0.03, t_end=0.00002, patience=35000):

    rng = np.random.RandomState(seed)
    h = h_bin.copy()
    m_range = np.arange(-(n - 1), n)
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    zeros_idx = np.where(h == 0)[0]
    ones_idx = np.where(h == 1)[0]
    no_improve = 0
    cool_rate = (t_end / t_start) ** (1.0 / 200000.0)
    T = t_start

    def swap_delta(i, j):
        delta = np.zeros(2 * n - 1)
        im = i + m_range
        msk = (im >= 0) & (im < n)
        delta[msk] += 1.0 - h[im[msk]]
        imn = i - m_range
        msk = (imn >= 0) & (imn < n)
        delta[msk] -= h[imn[msk]]
        jm = j + m_range
        msk = (jm >= 0) & (jm < n)
        delta[msk] -= 1.0 - h[jm[msk]]
        jmn = j - m_range
        msk = (jmn >= 0) & (jmn < n)
        delta[msk] += h[jmn[msk]]
        delta[n - 1] -= 2.0
        k1 = n - 1 + (j - i)
        if 0 <= k1 < 2*n-1:
            delta[k1] += 1.0
        k2 = n - 1 + (i - j)
        if 0 <= k2 < 2*n-1:
            delta[k2] += 1.0
        return delta

    while time.time() < deadline:
        i = zeros_idx[rng.randint(len(zeros_idx))]
        j = ones_idx[rng.randint(len(ones_idx))]
        delta = swap_delta(i, j)
        new_corr = corr + delta
        nc = new_corr.max() * dx
        delta_c = nc - cur_c
        if delta_c < -1e-15 or (T > 1e-12 and rng.random() < np.exp(-delta_c / T)):
            h[i] = 1.0
            h[j] = 0.0
            corr = new_corr
            cur_c = nc
            zeros_idx = np.where(h == 0)[0]
            ones_idx = np.where(h == 1)[0]
            if nc < best_c:
                best_c = nc
                best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        T *= cool_rate
        if no_improve > patience:
            nk = 3 + rng.randint(5)
            for _ in range(nk):
                ii = zeros_idx[rng.randint(len(zeros_idx))]
                jj = ones_idx[rng.randint(len(ones_idx))]
                h[ii] = 1.0
                h[jj] = 0.0
            corr = np.correlate(h, 1 - h, mode='full').astype(float)
            cur_c = corr.max() * dx
            zeros_idx = np.where(h == 0)[0]
            ones_idx = np.where(h == 1)[0]
            no_improve = 0
            T = min(t_start * 0.5, T * 2.0)
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_optimize(h, n, alpha, maxiter=3000, ftol=1e-16):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    ones_vec = np.ones(n)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0, 'jac': lambda x: ones_vec}]

    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        sm = shifted.max()
        return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha

    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        if res.fun < compute_c5(h, n):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def round_to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    return out

def binary_sa(h_bin, n, deadline, seed=42, t_start=0.025, t_end=0.00001, patience=40000):
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    h = h_bin.copy()
    m_range = np.arange(-(n - 1), n)
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    zeros_idx = np.where(h == 0)[0]
    ones_idx = np.where(h == 1)[0]
    no_improve = 0
    cool_rate = (t_end / t_start) ** (1.0 / 200000.0)
    T = t_start

    def swap_delta(h_local, i, j):
        delta = np.zeros(2 * n - 1)
        im = i + m_range
        msk = (im >= 0) & (im < n)
        delta[msk] += 1.0 - h_local[im[msk]]
        imn = i - m_range
        msk = (imn >= 0) & (imn < n)
        delta[msk] -= h_local[imn[msk]]
        jm = j + m_range
        msk = (jm >= 0) & (jm < n)
        delta[msk] -= 1.0 - h_local[jm[msk]]
        jmn = j - m_range
        msk = (jmn >= 0) & (jmn < n)
        delta[msk] += h_local[jmn[msk]]
        delta[n - 1] -= 2.0
        k1 = n - 1 + (j - i)
        if 0 <= k1 < 2 * n - 1:
            delta[k1] += 1.0
        k2 = n - 1 + (i - j)
        if 0 <= k2 < 2 * n - 1:
            delta[k2] += 1.0
        return delta

    while time.time() < deadline:
        i = zeros_idx[rng.randint(len(zeros_idx))]
        j = ones_idx[rng.randint(len(ones_idx))]
        delta = swap_delta(h, i, j)
        new_corr = corr + delta
        nc = new_corr.max() * dx
        delta_c = nc - cur_c
        if delta_c < -1e-15 or (T > 1e-12 and rng.random() < np.exp(-delta_c / T)):
            h[i] = 1.0
            h[j] = 0.0
            corr = new_corr
            cur_c = nc
            zeros_idx = np.where(h == 0)[0]
            ones_idx = np.where(h == 1)[0]
            if nc < best_c:
                best_c = nc
                best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        T *= cool_rate
        if no_improve > patience:
            nk = 3 + rng.randint(5)
            for _ in range(nk):
                ii = zeros_idx[rng.randint(len(zeros_idx))]
                jj = ones_idx[rng.randint(len(ones_idx))]
                h[ii] = 1.0
                h[jj] = 0.0
            corr = np.correlate(h, 1 - h, mode='full').astype(float)
            cur_c = corr.max() * dx
            zeros_idx = np.where(h == 0)[0]
            ones_idx = np.where(h == 1)[0]
            no_improve = 0
            T = min(t_start * 0.6, T * 2.5)
    return best_h, best_c

def slsqp_refine(h_start, n, deadline, rng):
    """Iteratively refine with increasing alpha and small perturbations."""
    h = project_feasible(h_start, n)
    best_local_c = compute_c5(h, n)
    best_local_h = h.copy()
    alphas = [2000, 10000, 50000, 200000, 1000000]
    for alpha in alphas:
        if time.time() > deadline:
            break
        h = slsqp_optimize(h, n, alpha, maxiter=5000, ftol=1e-16)
        c = compute_c5(h, n)
        if c < best_local_c:
            best_local_c = c
            best_local_h = h.copy()
    # Try perturbation restarts
    for _ in range(3):
        if time.time() > deadline:
            break
        h_pert = project_feasible(best_local_h + 0.02 * rng.randn(n), n)
        h_pert = slsqp_optimize(h_pert, n, 500000, maxiter=5000, ftol=1e-16)
        c = compute_c5(h_pert, n)
        if c < best_local_c:
            best_local_c = c
            best_local_h = h_pert.copy()
    return best_local_h, best_local_c

                best_h = h_final.copy()

    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
