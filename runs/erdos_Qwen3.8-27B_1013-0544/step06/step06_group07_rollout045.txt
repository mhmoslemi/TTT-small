import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def round_to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    return out

def swap_delta(h, n, i, j, m_range):
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    delta[k2] += 1.0
    return delta

def structured_init(n, seed=42):
    rng = np.random.RandomState(seed)
    target = n // 2
    candidates = []
    alt = np.zeros(n)
    alt[::2] = 1.0
    if alt.sum() > target:
        idx = np.where(alt == 1)[0]
        alt[idx[:int(alt.sum() - target)]] = 0
    elif alt.sum() < target:
        idx = np.where(alt == 0)[0]
        alt[idx[:int(target - alt.sum())]] = 1
    candidates.append(alt)
    for freq in range(1, 12):
        x = np.arange(n, dtype=float)
        s = np.cos(2 * np.pi * freq * x / n) + 0.4 * np.cos(4 * np.pi * freq * x / n + 1.3)
        thresh = np.sort(s)[target - 1] if target < n else 0
        four = (s >= thresh).astype(float)
        if four.sum() != target:
            if four.sum() > target:
                idx = np.where(four == 1)[0]
                rng.shuffle(idx)
                four[idx[:int(four.sum() - target)]] = 0
            else:
                idx = np.where(four == 0)[0]
                rng.shuffle(idx)
                four[idx[:int(target - four.sum())]] = 1
        candidates.append(four)
    ridx = rng.choice(n, size=target, replace=False)
    rand = np.zeros(n)
    rand[ridx] = 1.0
    candidates.append(rand)
    best = None
    best_c = np.inf
    for c in candidates:
        if c.sum() == target:
            cc = compute_c5(c, n)
            if cc < best_c:
                best_c = cc
                best = c
    return best, best_c

def parallel_tempering_sa(h_init, n, deadline, seed=42, n_reps=5, t_start=0.035, t_end=0.00001, patience=50000):
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    m_range = np.arange(-(n - 1), n)
    replicas = []
    for r in range(n_reps):
        if r == 0:
            h_r = h_init.copy()
        else:
            h_r = h_init.copy()
            n_pert = max(3, rng.randint(n // 15, n // 5))
            ones = np.where(h_r == 1)[0]
            zeros = np.where(h_r == 0)[0]
            si = rng.choice(ones, size=min(n_pert, len(ones)), replace=False)
            st = rng.choice(zeros, size=min(n_pert, len(zeros)), replace=False)
            h_r[si] = 0.0
            h_r[st] = 1.0
        corr_r = np.correlate(h_r, 1 - h_r, mode='full').astype(float)
        replicas.append([h_r, corr_r, corr_r.max() * dx])
    temp_ratio = (t_end / t_start) ** (1.0 / max(1, n_reps - 1))
    temps = np.array([t_start * (temp_ratio ** r) for r in range(n_reps)])
    best_c = min(r[2] for r in replicas)
    best_h = replicas[int(np.argmin([r[2] for r in replicas]))][0].copy()
    no_improve = 0
    step = 0
    while time.time() < deadline:
        for r in range(n_reps):
            h_r = replicas[r][0]
            ones_list = np.where(h_r == 1)[0]
            zeros_list = np.where(h_r == 0)[0]
            i = zeros_list[rng.randint(len(zeros_list))]
            j = ones_list[rng.randint(len(ones_list))]
            delta = swap_delta(h_r, n, i, j, m_range)
            new_corr = replicas[r][1] + delta
            nc = new_corr.max() * dx
            dc = nc - replicas[r][2]
            T = temps[r]
            if dc < 0 or (T > 1e-12 and rng.random() < np.exp(-dc / T)):
                h_r[i] = 1.0
                h_r[j] = 0.0
                replicas[r][1] = new_corr
                replicas[r][2] = nc
        for r in range(n_reps - 1):
            if rng.random() < 0.4:
                e_low = replicas[r][2]
                e_high = replicas[r + 1][2]
                inv_diff = 1.0 / temps[r] - 1.0 / temps[r + 1]
                log_p = -(e_low - e_high) * inv_diff
                if log_p > 0 or rng.random() < np.exp(min(0, log_p)):
                    replicas[r], replicas[r + 1] = replicas[r + 1], replicas[r]
        step += 1
        min_c = min(r[2] for r in replicas)
        if min_c < best_c - 1e-15:
            best_c = min_c
            bi = int(np.argmin([r[2] for r in replicas]))
            best_h = replicas[bi][0].copy()
            no_improve = 0
        else:
            no_improve += 1
        if step % 2000 == 0:
            temps = temps * 0.97
        if no_improve > patience:
            replicas[0] = [best_h.copy(), np.correlate(best_h, 1 - best_h, mode='full').astype(float), best_c]
            for r in range(1, n_reps):
                if rng.random() < 0.5:
                    h_p = best_h.copy()
                    np_ = max(3, rng.randint(n // 10, n // 4))
                    oi = np.where(h_p == 1)[0]
                    zi = np.where(h_p == 0)[0]
                    si2 = rng.choice(oi, size=min(np_, len(oi)), replace=False)
                    st2 = rng.choice(zi, size=min(np_, len(zi)), replace=False)
                    h_p[si2] = 0.0
                    h_p[st2] = 1.0
                    replicas[r] = [h_p, np.correlate(h_p, 1 - h_p, mode='full').astype(float), 0]
                    replicas[r][2] = replicas[r][1].max() * dx
            no_improve = 0
    return best_h, best_c

def greedy_local_search(h_bin, n, deadline, max_rounds=200, seed=42):
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    rng = np.random.RandomState(seed)
    h = h_bin.copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        n_try = min(200, len(zeros) * len(ones))
        for _ in range(n_try):
            i = zeros[rng.randint(len(zeros))]
            j = ones[rng.randint(len(ones))]
            delta = swap_delta(h, n, i, j, m_range)
            nc = (corr + delta).max() * dx
            if nc < cur_c - 1e-15:
                h[i] = 1.0
                h[j] = 0.0
                corr = corr + delta
                cur_c = nc
                improved = True
                if nc < best_c:
                    best_c = nc
                    best_h = h.copy()
                break
        if not improved:
            break
    return best_h, best_c
