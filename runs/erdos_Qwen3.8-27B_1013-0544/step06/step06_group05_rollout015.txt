```python
def fast_corr_update(h, a, b, n):
    """Compute the change in corr array when h[a]: 0->1 and h[b]: 1->0."""
    # a is the zero becoming one, b is the one becoming zero
    corr_change = np.zeros(2*n-1)
    for m in range(2*n-1):
        l = m - (n-1)  # lag
        delta = 0
        # h[a] changes from 0 to 1
        if 0 <= a + l < n:
            delta += (1 - h[a+l])  # was 0*(1-h[a+l]) = 0, now 1*(1-h[a+l])
        if 0 <= a - l < n:
            delta -= h[a-l]  # was h[a-l]*1 = h[a-l], now h[a-l]*0
        # h[b] changes from 1 to 0
        if 0 <= b + l < n:
            delta -= (1 - h[b+l])  # was 1*(1-h[b+l]), now 0
        if 0 <= b - l < n:
            delta += h[b-l]  # was h[b-l]*0 = 0, now h[b-l]*1
        corr_change[m] = delta
    return corr_change
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        mask = h < 1.0
        if diff > 0:
            diff = min(diff, (1.0 - h[mask]).sum() if mask.any() else 0)
        else:
            mask = h > 0.0
            diff = max(diff, -h[mask].sum() if mask.any() else 0)
        if mask.any() and abs(diff) > 1e-14:
            h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def fast_swap_eval(h, n, a, b, corr):
    """Compute new corr max when h[a]:0->1, h[b]:1->0. Returns new corr array."""
    # Change at each lag
    lags = np.arange(-(n-1), n)
    # P1[a, l] = (1-h[a+l]) if valid
    idx_a_plus = a + lags
    valid_ap = (idx_a_plus >= 0) & (idx_a_plus < n)
    P1a = np.where(valid_ap, 1.0 - h[np.where(valid_ap, idx_a_plus, 0)], 0.0)
    # P2[a, l] = h[a-l] if valid
    idx_a_minus = a - lags
    valid_am = (idx_a_minus >= 0) & (idx_a_minus < n)
    P2a = np.where(valid_am, h[np.where(valid_am, idx_a_minus, 0)], 0.0)
    # P1[b, l], P2[b, l]
    idx_b_plus = b + lags
    valid_bp = (idx_b_plus >= 0) & (idx_b_plus < n)
    P1b = np.where(valid_bp, 1.0 - h[np.where(valid_bp, idx_b_plus, 0)], 0.0)
    idx_b_minus = b - lags
    valid_bm = (idx_b_minus >= 0) & (idx_b_minus < n)
    P2b = np.where(valid_bm, h[np.where(valid_bm, idx_b_minus, 0)], 0.0)
    
    delta = P1a - P2a - P1b + P2b
    new_corr = corr + delta
    return new_corr, new_corr.max()
```

```python
def compute_D(h, n):
    """Compute D[x, l] = P1[x,l] - P2[x,l] for all x, l."""
    # D[x, m] where m = l + (n-1), l ranges from -(n-1) to (n-1)
    # P1[x, l] = (1-h[x+l]) if 0<=x+l<n else 0
    # P2[x, l] = h[x-l] if 0<=x-l<n else 0
    
    D = np.zeros((n, 2*n-1))
    lags = np.arange(-(n-1), n)  # shape (2n-1,)
    for x in range(n):
        idx_plus = x + lags  # shape (2n-1,)
        valid_p = (idx_plus >= 0) & (idx_plus < n)
        D[x, :] += np.where(valid_p, 1.0 - h[np.clip(idx_plus, 0, n-1)], 0.0)
        idx_minus = x - lags
        valid_m = (idx_minus >= 0) & (idx_minus < n)
        D[x, :] -= np.where(valid_m, h[np.clip(idx_minus, 0, n-1)], 0.0)
    return D
```

**Continuous approach with proper gradient:**

f(h) = max_k (dx * corr_k(h))

where corr_k(h) = sum_i h[i] * (1-h[i+k])

∂corr_k/∂h[j] = (1-h[j+k]) * [0≤j+k<n] - h[j-k] * [0≤j-k<n]

= dx * sum_k (exp(alpha*corr_k) / sum_m exp(alpha*corr_m)) * ∂corr_k/∂h[j]

This is a weighted average of the gradients of individual corr_k, where weights are softmax over the corr values.

```python
def smooth_c5_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Gradient of corr_k w.r.t. h[j]
    # grad[j, k] = (1-h[j+k])*[valid] - h[j-k]*[valid]
    # For k = 0..2n-2, lag = k-(n-1)
    # grad[j, k] = (1-h[j + k - (n-1)])*[0<=j+k-(n-1)<n] - h[j - k + (n-1)]*[0<=j-k+(n-1)<n]
    
    lags = np.arange(-(n-1), n)  # shape (2n-1,)
    js = np.arange(n)  # shape (n,)
    
    # For each j and each lag:
    # (1-h[j+lag]) if 0<=j+lag<n
    idx_p = js[:, None] + lags[None, :]  # shape (n, 2n-1)
    valid_p = (idx_p >= 0) & (idx_p < n)
    g1 = np.where(valid_p, 1.0 - h[np.clip(idx_p, 0, n-1)], 0.0)
    
    idx_m = js[:, None] - lags[None, :]
    valid_m = (idx_m >= 0) & (idx_m < n)
    g2 = np.where(valid_m, h[np.clip(idx_m, 0, n-1)], 0.0)
    
    grad_corr = g1 - g2  # shape (n, 2n-1)
    
    # Softmax weights
    shifted = alpha * (corr - corr.max())
    weights = np.exp(shifted)
    weights /= weights.sum()  # shape (2n-1,)
    
    # Weighted gradient
    grad_obj = dx * (grad_corr @ weights)  # shape (n,)
    
    # Value
    val = dx * (corr.max() + np.log(weights.sum()) / alpha)
    # Actually: val = dx * (1/alpha) * log(sum exp(alpha*corr))
    # = dx * (corr.max() + (1/alpha)*log(sum exp(alpha*(corr-corr.max()))))
    # = dx * (corr.max() + (1/alpha)*log(weights_unnormalized_sum))
    # where weights_unnormalized = exp(alpha*(corr-corr.max()))
    # weights = weights_unnormalized / sum, so log(sum) = log(weights.sum() * sum_unnorm / sum_unnorm)... 
    # Let me just compute it directly
    unnormalized = np.exp(alpha * (corr - corr.max()))
    val = dx * (corr.max() + np.log(unnormalized.sum()) / alpha)
    
    return val, grad_obj
```

This is O(n²) per evaluation (dominated by the grad_corr computation). For n=250, that's 62500 operations. With L-BFGS-B doing ~100 iterations, that's 6.25M operations total. Very fast.

```python
    target = n / 2.0
```
