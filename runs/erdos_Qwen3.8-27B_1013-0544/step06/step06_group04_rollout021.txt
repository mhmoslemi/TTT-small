import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def smooth_obj_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    f = np.correlate(h, 1 - h, 'full') * dx
    m = f.max()
    shifted = alpha * (f - m)
    e = np.exp(shifted)
    se = e.sum()
    obj = m + np.log(se) / alpha
    p = e / se
    p_pos = p[n - 1:]
    grad = np.zeros(n)
    for j in range(n):
        s1 = 0.0
        for d in range(n - j):
            s1 += p_pos[d] * (1.0 - h[j + d])
        s2 = 0.0
        for d in range(j + 1):
            s2 += p[n - 1 - d] * h[j - d]
        grad[j] = dx * (s1 - s2)
    return obj, grad

def lbfgs_optimize(h, n, alpha, maxiter=200, penalty=1e4):
    def obj_fn(x):
        o, g = smooth_obj_grad(x, alpha)
        c = x.sum() - n / 2.0
        return o + penalty * c * c
    def grad_fn(x):
        o, g = smooth_obj_grad(x, alpha)
        c = x.sum() - n / 2.0
        return g + 2 * penalty * c
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj_fn, h, jac=grad_fn, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-8})
    return project_feasible(res.x, n)

def swap_delta_fast(h, i, j, n):
    delta = np.zeros(2 * n - 1)
    corr_base = np.arange(0, 2 * n - 1)
    lo = max(0, n - 1 - i)
    hi = min(2 * n - 2, 2 * n - 2 - i)
    if lo <= hi:
        m = np.arange(lo, hi + 1)
        delta[m] += 1.0 - h[m - (n - 1) + i]
    m2 = np.arange(max(0, i), min(2 * n - 2, n - 1 + i) + 1)
    jprime = n - 1 + i - m2
    delta[m2] -= h[jprime]
    lo = max(0, n - 1 - j)
    hi = min(2 * n - 2, 2 * n - 2 - j)
    if lo <= hi:
        m = np.arange(lo, hi + 1)
        delta[m] -= 1.0 - h[m - (n - 1) + j]
    m2 = np.arange(max(0, j), min(2 * n - 2, n - 1 + j) + 1)
    jprime = n - 1 + j - m2
    delta[m2] += h[jprime]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    if 0 <= k1 < 2 * n - 1:
        delta[k1] += 1.0
    k2 = n - 1 + (i - j)

def local_search(h, n, deadline, max_sweeps=5000, seed=42, guided_h=None):

    dx = 2.0 / n

        improved = False

        if guided_h is not None:

            rng.shuffle(ones)
            rng.shuffle(zeros)
        found = False
        for idx_i in range(len(zeros)):
            i = int(zeros[idx_i])
            for idx_j in range(len(ones)):
                j = int(ones[idx_j])
                d = swap_delta_fast(h, i, j, n)
