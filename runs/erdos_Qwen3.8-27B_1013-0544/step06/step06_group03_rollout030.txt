def projected_subgradient(h, n, deadline, alpha=5000):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    t = 0
    step_size = 0.01
    momentum = np.zeros(n)
    beta = 0.9  # momentum decay
    
    while time.time() < deadline:
        t += 1
        # Compute gradient of smooth max
        c = np.correlate(h_curr, 1-h_curr, mode='full') * (2.0/n)
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        w = np.exp(shifted - mx)
        w /= w.sum()
        
        # Compute gradient (same as before)
        dx = 2.0 / n
        grad = w[n-1] * dx * (1.0 - 2.0 * h_curr)
        for lag_m in range(1, n):
            wk_pos = w[n-1+lag_m]
            wk_neg = w[n-1-lag_m]
            if wk_pos < 1e-12 and wk_neg < 1e-12:
                continue
            if wk_pos > 1e-12:
                grad[:n-lag_m] += wk_pos * dx * (1.0 - h_curr[lag_m:])
                grad[lag_m:] -= wk_pos * dx * h_curr[:n-lag_m]
            if wk_neg > 1e-12:
                grad[lag_m:] += wk_neg * dx * (1.0 - h_curr[:n-lag_m])
                grad[:n-lag_m] -= wk_neg * dx * h_curr[lag_m:]
        
        # Momentum update
        momentum = beta * momentum - step_size * grad
        h_new = h_curr + momentum
        
        # Project to feasible set
        h_new = project_feasible(h_new, n)
        
        # Check improvement
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        
        h_curr = h_new
        
        # Decay step size
        step_size = 0.01 / np.sqrt(1 + 0.01 * t)
    
    return best_h, best_c

from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def smooth_max_vals(vals, alpha):
    m = vals.max()
    shifted = alpha * (vals - m)
    mx = shifted.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    return m + np.log(np.sum(np.exp(shifted))) / alpha

def compute_smooth_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    
    grad = np.zeros(n)
    # Lag 0 (index n-1 in full mode)
    grad += w[n-1] * dx * (1.0 - 2.0 * h)
    
    # Lags +m and -m
    for m_val in range(1, n):
        wk_pos = w[n - 1 + m_val]
        wk_neg = w[n - 1 - m_val]
        if wk_pos < 1e-13 and wk_neg < 1e-13:
            continue
        nm = n - m_val
        if wk_pos > 1e-13:
            grad[:nm] += wk_pos * dx * (1.0 - h[m_val:])
            grad[m_val:] -= wk_pos * dx * h[:nm]
        if wk_neg > 1e-13:
            grad[m_val:] += wk_neg * dx * (1.0 - h[:nm])
            grad[:nm] -= wk_neg * dx * h[m_val:]
    
    return grad

def slsqp_with_grad(h0, n, alpha, maxiter=5000, ftol=1e-16):
    def obj(x):
        return smooth_max_vals(np.correlate(x, 1-x, mode='full') * (2.0/n), alpha)
    
    def obj_grad(x):
        return compute_smooth_grad(x, n, alpha)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    
    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        if res.success or res.status == 0:
            return project_feasible(res.x, n)
        else:
            return project_feasible(res.x, n)
    except Exception:
        return h0.copy()

def outer_approx_step(h_curr, n, trust):
    dx = 2.0 / n
    c = np.correlate(h_curr, 1 - h_curr, mode='full') * dx
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1,
            cp.abs(h_var - h_curr) <= trust]
    
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        # Linearization of c_k at h_curr
        # c_k(h) ≈ c_k(h_curr) + grad_k · (h - h_curr)
        # grad_k[j] for lag k:
        if lag == 0:
            grad_vec = dx * (1.0 - 2.0 * h_curr)
        elif lag > 0 and lag < n:
            grad_vec = np.zeros(n)
            grad_vec[:n-lag] = dx * (1.0 - h_curr[lag:])
            grad_vec[lag:] = -dx * h_curr[:n-lag]
        elif lag < 0 and abs(lag) < n:
            m_val = -lag
            grad_vec = np.zeros(n)
            grad_vec[m_val:] = dx * (1.0 - h_curr[:n-m_val])
            grad_vec[:n-m_val] = -dx * h_curr[m_val:]
        else:
            continue
        
        cons.append(c[k] + grad_vec @ (h_var - h_curr) <= t_var)
    
    prob = cp.Problem(cp.Minimize(t_var), cons)
    try:
        prob.solve(solver=cp.GLOP)
    except Exception:
        try:
            prob.solve(solver=cp.CBC)
        except Exception:
            return h_curr.copy()
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return h_curr.copy()
    h_new = h_var.value
    if h_new is None:
        return h_curr.copy()
    return project_feasible(h_new, n)

def outer_approx(h, n, deadline, max_iters=80):
    h_curr = h.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    trust = 0.2
    for it in range(max_iters):
        if time.time() > deadline - 2:
            break
        h_new = outer_approx_step(h_curr, n, trust)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
        step = np.max(np.abs(h_new - h_curr))
        if step < 1e-10:
            break
        h_curr = h_new
        trust = max(0.01, trust * 0.85)
    return best_h, best_c

def active_set_slsqp(h, n, top_m, alpha, maxiter=5000):
    c = compute_all_lags(h, n)
    top_idx = np.argsort(c)[-top_m:]
    
    def obj(x):
        cc = np.correlate(x, 1-x, mode='full') * (2.0/n)
        return smooth_max_vals(cc[top_idx], alpha)
    
    def obj_grad(x):
        dx = 2.0 / n
        cc = np.correlate(x, 1-x, mode='full') * dx
        # Softmax over active set only
        active_vals = cc[top_idx]
        m = active_vals.max()
        shifted = alpha * (active_vals - m)
        mx = shifted.max()
        w_active = np.exp(shifted - mx)
        w_active /= w_active.sum()
        
        grad = np.zeros(n)
        for i, k_idx in enumerate(top_idx):
            wk = w_active[i]
            if wk < 1e-13:
                continue
            lag = k_idx - (n - 1)
            if lag == 0:
                grad += wk * dx * (1.0 - 2.0 * x)
            elif 0 < lag < n:
                nm = n - lag
                grad[:nm] += wk * dx * (1.0 - x[lag:])
                grad[lag:] -= wk * dx * x[:nm]
            elif -n < lag < 0:
                m_val = -lag
                nm = n - m_val
                grad[m_val:] += wk * dx * (1.0 - x[:nm])
                grad[:nm] -= wk * dx * x[m_val:]
        return grad
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def local_search_fast(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    c_vec = compute_all_lags(h, n)
    k_star = int(np.argmax(c_vec))
    
    deltas = [0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    for d in deltas:
        if time.time() > deadline:
            break
        improved = True
        while improved:
            if time.time() > deadline:
                break
            improved = False
            # Focus on the worst lag
            c_vec = compute_all_lags(h, n)
            k_star = int(np.argmax(c_vec))
            lag = k_star - (n - 1)
            if lag == 0:
                break
            m = abs(lag)
            if m >= n:
                break
            # Try to reduce the overlap at this lag
            # For lag > 0: c involves h[i]*(1-h[i+m]) for i in [0, n-1-m]
            # To reduce: if h[i] is high and h[i+m] is low, move mass from i to i+m
            # if h[i] is low and h[i+m] is high, that's fine (contributes 0)
            if lag > 0:
                for i in range(n - m):
                    if time.time() > deadline:
                        break
                    # h[i] contributes h[i]*(1-h[i+m])
                    # Reducing h[i] or increasing h[i+m] reduces this term
                    j = i + m
                    if h[i] > 0.01 and h[j] < 0.99:
                        delta = min(d, h[i]*0.3, (1-h[j])*0.3)
                        if delta < 1e-8:
                            continue
                        h[i] -= delta
                        h[j] += delta
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-15:
                            best_c = c_new
                            improved = True
                        else:
                            h[i] += delta
                            h[j] -= delta
            else:
                # lag < 0, m = -lag
                # c involves h[i+m]*(1-h[i]) for i in [0, n-1-m]
                # which is h[j]*(1-h[i]) where j = i+m
                # To reduce: reduce h[j] or increase h[i]
                for i in range(n - m):
                    if time.time() > deadline:
                        break
                    j = i + m
                    if h[j] > 0.01 and h[i] < 0.99:
                        delta = min(d, h[j]*0.3, (1-h[i])*0.3)
                        if delta < 1e-8:
                            continue
                        h[j] -= delta
                        h[i] += delta
                        c_new = compute_c5(h, n)
                        if c_new < best_c - 1e-15:
                            best_c = c_new
                            improved = True
                        else:
                            h[j] += delta
                            h[i] -= delta
    return h, best_c

def optimize_at_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    
    # Phase 1: SLSQP with increasing alpha
    alphas = [500, 2000, 10000, 50000]
    for alpha in alphas:
        if time.time() > deadline - 30:
            break
        alloc = min(60, (deadline - time.time()) * 0.3)
        h_new = slsqp_with_grad(best_h, n, alpha, maxiter=3000, ftol=1e-15)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new
    
    # Phase 2: Outer approximation
    if time.time() < deadline - 30:
        oo_end = min(time.time() + 80, deadline - 30)
        h_oo, c_oo = outer_approx(best_h, n, oo_end)
        if c_oo < best_c - 1e-14:
            best_c = c_oo
            best_h = h_oo
    
    # Phase 3: Active set SLSQP
    if time.time() < deadline - 20:
        for top_m in [5, 10]:
            if time.time() > deadline - 20:
                break
            h_new = active_set_slsqp(best_h, n, top_m, 20000, maxiter=4000)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
                best_h = h_new
    
    # Phase 4: Final SLSQP with very high alpha
    if time.time() < deadline - 15:
        h_new = slsqp_with_grad(best_h, n, 500000, maxiter=5000, ftol=1e-16)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new
    
    # Phase 5: Local search
    if time.time() < deadline - 3:
        best_h, best_c = local_search_fast(best_h, n, deadline - 1)
    
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c
