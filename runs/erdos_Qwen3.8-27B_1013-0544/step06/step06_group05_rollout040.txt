        h_vals = h[i + d_range]  # h[i+d] for d in d_range

    # Contribution from h[i] as second factor: a[n_half+d] += -h[i-d] for 0<=i-d<n
    # 0 <= i-d < n means i-n < d <= i
    # d ranges from max(-(n-1), i-n+1) to min(n-1, i)
    d_lo2 = max(-(n-1), i - (n-1))  # i-n+1 = i-(n-1)
    d_hi2 = min(n-1, i)
    if d_lo2 <= d_hi2:

        idx = n_half + d_range
        h_vals = h[i - d_range]  # h[i-d] for d in d_range
        a[idx] += -h_vals
    
    return a

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(3000):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * room, 0.0, 1.0)
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * h, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def compute_a(i, h, n):
    """Compute coefficient vector a for coordinate i."""
    a = np.zeros(2 * n - 1)
    nh = n - 1
    # First factor contribution: a[nh+d] += (1-h[i+d]) for 0<=i+d<n
    d_lo = max(-(n-1), -i)
    d_hi = min(n-1, n-1-i)
    if d_lo <= d_hi:
        dr = np.arange(d_lo, d_hi + 1)
        a[nh + dr] += (1.0 - h[i + dr])
    # Second factor contribution: a[nh+d] += -h[i-d] for 0<=i-d<n
    d_lo2 = max(-(n-1), i-n+1)
    d_hi2 = min(n-1, i)
    if d_lo2 <= d_hi2:
        dr2 = np.arange(d_lo2, d_hi2 + 1)
        a[nh + dr2] += -h[i - dr2]
    return a

def coord_descent(h, n, max_sweeps=300, deadline=None, tol=1e-12, order='forward'):
    """Continuous coordinate descent."""
    h = h.copy()
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    best_c = corr.max()
    stall_count = 0
    
    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        
        if order == 'forward':
            indices = range(n)
        elif order == 'reverse':
            indices = range(n-1, -1, -1)
        else:
            indices = np.random.permutation(n)
        
        for i in indices:
            x_old = h[i]
            a = compute_a(i, h, n)
            
            # Bisection for optimal x
            lo, hi = 0.0, 1.0
            for _ in range(50):
                mid = 0.5 * (lo + hi)
                delta = mid - x_old
                mx = (corr + a * delta).max()
                # Find active constraints
                vals = corr + a * delta
                active = vals >= mx - 1e-13
                a_act = a[active]
                if a_act.size > 0:
                    amin = float(a_act.min())
                    amax = float(a_act.max())
                    if amin > 1e-14:
                        hi = mid
                    elif amax < -1e-14:
                        lo = mid
                    else:
                        break
            x_opt = 0.5 * (lo + hi)
            
            if abs(x_opt - x_old) > 1e-13:
                delta = x_opt - x_old
                corr += a * delta
                h[i] = x_opt
                improved = True
        
        new_max = float(corr.max())
        if new_max < best_c - tol:
            best_c = new_max
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 5 and not improved:
                break
    
    return h, float(best_c * dx)

def slsqp_polish(h, n, maxiter=2000, deadline=None):
    dx = 2.0 / n
    alpha = 3000.0
    
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = float(g.max())
        sh = alpha * (g - m)
        sm = float(sh.max())
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    
    def cons_fun(x):
        return x.sum() - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    
    try:
        if deadline is not None and time.time() > deadline:
            return h
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h

def resample_to_n(h_old, n_old, n_new):
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    # Nearest-neighbor resampling (preserves step function character)
    old_indices = np.round(x_new * n_old).astype(int) % n_old
    h_new = h_old[old_indices].copy()
    h_new = project_feasible(h_new, n_new)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):

Let's denote the lag as d (so m = n-1+d).

corr[n-1+d] = Σ_{n: valid} h[n] * (1 - h[n+d])

where "valid" means 0 ≤ n < N and 0 ≤ n+d < N.

Now, h[i] appears in this sum in two ways:
1. As h[n] when n = i: this requires 0 ≤ i + d < N.
   Term: h[i] * (1 - h[i+d])

2. As h[n+d] when n+d = i, i.e., n = i-d: this requires 0 ≤ i-d < N.
   Term: h[i-d] * (1 - h[i])

So:
corr[n-1+d] = [sum of terms not involving h[i]] + h[i]*(1-h[i+d]) [if 0≤i+d<N] + h[i-d]*(1-h[i]) [if 0≤i-d<N]

Let me denote:
- C_d = [sum of terms not involving h[i]] (doesn't depend on h[i])

Then:
corr[n-1+d] = C_d + h[i]*(1-h[i+d]) + h[i-d]*(1-h[i])

where the second term is present only if 0≤i+d<N, and the third only if 0≤i-d<N.

Now, as a function of x = h[i]:
corr[n-1+d] = C_d + x*(1-h[i+d]) + h[i-d]*(1-x)
             = C_d + x - x*h[i+d] + h[i-d] - x*h[i-d]
             = (C_d + h[i-d]) + x*(1 - h[i+d] - h[i-d])

So:
b_d = C_d + h[i-d] (if 0≤i-d<N, else C_d)
a_d = 1 - h[i+d] - h[i-d] (with missing terms treated as 0)

And corr[n-1+d] = b_d + x * a_d.

At the current value x = h[i]:
b_d = corr[n-1+d] - h[i] * a_d

Let me verify with the example:
n=4, h=[1,0,1,0], i=0, x=h[0]=1

For d=0: corr[3] = 0
  0≤0+0<4: yes. 0≤0-0<4: yes.
  a_0 = 1 - h[0] - h[0] = 1 - 1 - 1 = -1
  b_0 = 0 - 1*(-1) = 1
  Check: b_0 + x*a_0 = 1 + 1*(-1) = 0 ✓

For d=1: corr[4] = 1
  0≤0+1<4: yes. 0≤0-1<4: no (-1<0).
  a_1 = 1 - h[1] - 0 = 1 - 0 = 1
  b_1 = 1 - 1*1 = 0
  Check: b_1 + x*a_1 = 0 + 1*1 = 1 ✓

For d=2: corr[5] = 0
  0≤0+2<4: yes. 0≤0-2<4: no.
  a_2 = 1 - h[2] = 1 - 1 = 0
  b_2 = 0 - 1*0 = 0
  Check: b_2 + x*a_2 = 0 + 0 = 0 ✓

For d=3: corr[6] = 0
  0≤0+3<4: yes. 0≤0-3<4: no.
  a_3 = 1 - h[3] = 1 - 0 = 1
  b_3 = 0 - 1*1 = -1
  Check: b_3 + x*a_3 = -1 + 1 = 0 ✓

For d=-1: corr[2] = 2
  0≤0+(-1)<4: no. 0≤0-(-1)<4: yes (0≤1<4).
  a_{-1} = 0 - h[1] = -0 = 0
  b_{-1} = 2 - 1*0 = 2
