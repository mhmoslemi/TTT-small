def coord_optimal(h, i, n, lam):
    """Find optimal h[i] for the Lagrangian problem."""
    dx = 2.0 / n
    # Compute current correlation
    corr = np.correlate(h, 1-h, mode='full') * dx  # shape (2n-1,)
    
    # For coordinate i, compute a_k for each lag k
    # Lag k corresponds to index m = k + n - 1 in corr
    # h[i] appears in f_k with coefficient:
    #   a_k = 1 - h[i-k] - h[i+k] (out-of-range h treated as 0)
    
    a = np.zeros(2*n-1)
    for m in range(2*n-1):
        k = m - (n-1)
        # h[i] appears as: h[i]*(1-h[i-k]) and h[i+k]*(1-h[i])
        # Coefficient of h[i] in f_k: (1-h[i-k]) - h[i+k] = 1 - h[i-k] - h[i+k]
        val = 1.0
        if i - k >= 0 and i - k < n:
            val -= h[i-k]
        if i + k >= 0 and i + k < n:
            val -= h[i+k]
        a[m] = val
    
    # Now minimize max_m (corr[m] + a[m]*(t - h[i])) + lam*t over t in [0,1]
    # = max_m (a[m]*t + (corr[m] - a[m]*h[i])) + lam*t
    
    b = corr - a * h[i]  # f_k(0) for each k
    
    # Minimize g(t) = max_m (a[m]*t + b[m]) + lam*t over t in [0,1]
    
    # Ternary search or subgradient binary search
    # Let's use subgradient binary search
    lo, hi = 0.0, 1.0
    
    def subgrad_contains_zero(t):
        """Returns -1 if should go left, 1 if should go right, 0 if at optimum"""
        vals = a * t + b  # shape (2n-1,)
        mx = vals.max()
        active = vals >= mx - 1e-15
        # Subgradient of g(t)+lam*t at t: conv{a[m] : active} + {lam}
        min_sg = a[active].min() + lam
        max_sg = a[active].max() + lam
        if max_sg < 0:
            return -1  # need to increase t
        elif min_sg > 0:
            return 1   # need to decrease t
        else:
            return 0
    
    for _ in range(50):
        mid = (lo + hi) / 2
        sg = subgrad_contains_zero(mid)
        if sg == -1:
            lo = mid
        elif sg == 1:
            hi = mid
        else:
            lo = hi = mid
            break
    
    t_opt = (lo + hi) / 2
    return t_opt

    nc = 2 * n - 1

        if time.time() > deadline:
            break
        max_change = 0.0
        h_sum = h.sum()
        for i in range(n):
            # Compute a[m] for coordinate i
            a = np.ones(nc)
            idx1 = i - k_range
            idx2 = i + k_range
            v1 = (idx1 >= 0) & (idx1 < n)
            v2 = (idx2 >= 0) & (idx2 < n)
            a[v1] -= h[idx1[v1]]
            a[v2] -= h[idx2[v2]]
            
            b = corr - a * h[i]
            c = h_sum - h[i] - n / 2.0
            
            # Binary search for optimal t
            lo, hi = 0.0, 1.0
            for _ in range(45):
                mid = (lo + hi) / 2.0
                vals = a * mid + b
                mx = vals.max()

                    hi = mid
                elif sg_max < 0:
                    lo = mid
                else:
                    break
            t = (lo + hi) / 2.0
            t = max(0.0, min(1.0, t))
            
            delta = t - h[i]
            if abs(delta) > max_change:
                max_change = abs(delta)

                corr += delta * a
                h_sum += delta
                h[i] = t
        
        if max_change < tol:
            break
    return h
