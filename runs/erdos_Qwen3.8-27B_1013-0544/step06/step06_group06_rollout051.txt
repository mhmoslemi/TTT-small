                h[pi1], h[pi2] = 0.0, 0.0
                h[pj1], h[pj2] = 1.0, 1.0
            if best_quad is not None:
                pi1, pi2, pj1, pj2 = best_quad
                h[pi1], h[pi2] = 1.0, 1.0
                h[pj1], h[pj2] = 0.0, 0.0
                corr = best_corr
                cur_max = best_val
                ones = np.where(h > 0.5)[0].copy()
                zeros = np.where(h < 0.5)[0].copy()
                stall = 0
                temp = max(0.01, temp * 0.9999)
                if cur_max < best_max - 1e-15:
                    best_max = cur_max
                    best_h = h.copy()
                found = True

        if not found:
            stall += 1
            if stall > 80:
                nk = rng.randint(3, 7)
                rz = rng.choice(len(zeros), min(nk, len(zeros)), replace=False)
                rj = rng.choice(len(ones), min(nk, len(ones)), replace=False)
                for a in range(min(nk, len(zeros), len(ones))):
                    h[zeros[rz[a]]] = 1.0
                    h[ones[rj[a]]] = 0.0
                nc = get_corr(h)
                nm = float(nc.max())
                delta = nm - cur_max
                if delta < 0 or np.random.rand() < np.exp(-delta / max(temp, 0.01)):
                    corr = nc
                    cur_max = nm
                    ones = np.where(h > 0.5)[0].copy()
                    zeros = np.where(h < 0.5)[0].copy()
                else:
                    for a in range(min(nk, len(zeros), len(ones))):
                        h[zeros[rz[a]]] = 0.0
                        h[ones[rj[a]]] = 1.0
                    corr = get_corr(h)
                    cur_max = float(corr.max())
                    ones = np.where(h > 0.5)[0].copy()
                    zeros = np.where(h < 0.5)[0].copy()
                temp = max(0.01, temp * 0.92)
                stall = 0

    return best_h, best_max * dx

def slsqp_polish(h, n, deadline, seed=42):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    hc = project_feasible(h + 0.003 * rng.randn(n), n)
    best = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cfun}]
    for alpha in [50000, 200000, 800000, 3000000]:
        if time.time() > deadline - 3:
            break
        try:
            res = minimize(soft_obj, hc, args=(alpha, dx), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 1500, 'ftol': 1e-14})
            hc = project_feasible(res.x, n)
            hr = np.zeros(n)
            hr[np.argsort(res.x)[::-1][:n // 2]] = 1.0
            c = compute_c5(hr, n)
            if c < best_c:
                best_c = c
                best = hr.copy()
        except Exception:
            pass
    return best, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 12
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.zeros(n)
        h0[rng.choice(n, n // 2, replace=False)] = 1.0

    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    p1 = t0 + 0.35 * (budget_s - 12)
    hb, cb = local_search(h0, n, p1, seed=seed, mode='mixed')
    if cb < best_c:
        best_c = cb
        best_h = hb.copy()

    p2 = t0 + 0.55 * (budget_s - 12)
    if time.time() < p2:
        h2 = h0.copy()
        nk = n // 8
        fi = rng.choice(n, nk, replace=False)
        h2[fi] = 1 - h2[fi]
        s = h2.sum()
        tgt = n / 2.0
        if s > tgt:
            ex = np.where(h2 == 1)[0]
            h2[rng.choice(ex, int(s - tgt), replace=False)] = 0
        elif s < tgt:
            ms = np.where(h2 == 0)[0]
            h2[rng.choice(ms, int(tgt - s), replace=False)] = 1
        hb, cb = local_search(h2, n, p2, seed=seed + 50, mode='mixed')
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()

    p3 = t0 + 0.70 * (budget_s - 12)
    if time.time() < p3:
        hb, cb = local_search(best_h, n, p3, seed=seed + 150, mode='mixed')
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()

    p4 = t0 + 0.85 * (budget_s - 12)
    if time.time() < p4:
        hb, cb = slsqp_polish(best_h, n, p4, seed=seed + 250)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()

    if time.time() < dl - 3:
        hb, cb = local_search(best_h, n, dl - 2, seed=seed + 350, mode='mixed')
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
