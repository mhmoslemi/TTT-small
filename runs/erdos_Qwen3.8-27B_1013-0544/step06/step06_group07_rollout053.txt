def m_sequence(n):
    """Generate an M-sequence of length n (0/1). Only works for n = 2^k - 1."""
    import math
    k = int(math.log2(n + 1))
    if 2**k - 1 != n:
        # Fall back to a different structured sequence
        return thue_morse(n)
    # Simple LFSR-based M-sequence
    # Use a primitive polynomial
    # For small k, use known tap positions
    taps = {2: [1], 3: [1], 4: [1], 5: [3], 6: [3], 7: [3], 8: [4]}
    if k not in taps:
        return thue_morse(n)
    tap = taps[k] + [0]  # include the feedback tap
    state = np.ones(k, dtype=int)
    seq = np.zeros(n, dtype=int)
    for i in range(n):
        seq[i] = state[-1]
        feedback = 0
        for t in taps:
            feedback ^= state[t]
        state = np.roll(state, 1)
        state[0] = feedback
    return seq.astype(float)

        rng = np.random.RandomState(0)
        add = rng.choice(zeros_idx, int(target - n_ones), replace=False)
        h[add] = 1
    return h

def proj_gradient_step(h, n, alpha, lr):
    """One step of projected gradient descent on the soft-max objective."""
    dx = 2.0 / n
    g = np.correlate(h, 1 - h, mode='full') * dx
    # Soft-max gradient
    sh = alpha * (g - g.max())
    sm = sh.max()
    w = np.exp(sh - sm)
    w = w / w.sum()  # weights
    # Gradient of soft-max wrt h
    # d(max)/dh[i] ≈ sum_k w[k] * d(c[k])/dh[i] * dx
    # d(c[k])/dh[i] = (1-h[i+mr[k]]) - h[i-mr[k]] for valid indices
    # where mr[k] = k - (n-1)
    mr = np.arange(-(n-1), n)
    grad = np.zeros(n)
    for i in range(n):
        # First term: sum_k w[k] * (1-h[i+mr[k]])
        idx1 = i + mr
        m1 = (idx1 >= 0) & (idx1 < n)
        term1 = w[m1] @ (1 - h[idx1[m1]])
        # Second term: sum_k w[k] * h[i-mr[k]]
        idx2 = i - mr
        m2 = (idx2 >= 0) & (idx2 < n)
        term2 = w[m2] @ h[idx2[m2]]
        grad[i] = dx * (term1 - term2)
    # Take gradient step
    h_new = h - lr * grad
    # Project onto feasible set
    h_new = project_feasible(h_new, n)
    return h_new
