def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def round_to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0

    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range  # positions i-m for all lags m (second factor)

def greedy_local(h_bin, n, max_iter=50000):
    """Greedy: always accept the best single swap"""
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    h = h_bin.astype(float).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    cur_c = corr.max() * dx
    zeros = np.where(h == 0)[0]
    ones = np.where(h == 1)[0]
    
    for _ in range(max_iter):
        best_delta_c = 1e10
        best_i, best_j = -1, -1
        best_delta = None
        for i in zeros:
            for j in ones:
                delta = swap_delta(h, n, i, j, m_range)
                nc = (corr + delta).max() * dx
                dc = nc - cur_c
                if dc < best_delta_c:
                    best_delta_c = dc
                    best_i, best_j = i, j
                    best_delta = delta
        if best_delta_c >= -1e-15:
            break
        h[best_i] = 1.0
        h[best_j] = 0.0
        corr = corr + best_delta
        cur_c += best_delta_c
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
    return h, cur_c

import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def round_to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    return out

def swap_delta(h, n, i, j, m_range):
    ny = 2 * n - 1
    delta = np.zeros(ny)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    return delta

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def round_to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    return out

def swap_delta(h, n, i, j, m_range):
    ny = 2 * n - 1
    delta = np.zeros(ny)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    return delta

def sa_fast(h_bin, n, deadline, seed=42, t_start=0.035, t_end=0.000003):
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    h = h_bin.astype(float).copy()
    m_range = np.arange(-(n - 1), n)
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    zeros_idx = np.where(h == 0)[0]
    ones_idx = np.where(h == 1)[0]
    no_improve = 0
    n_cool = 500000
    cool_rate = (t_end / t_start) ** (1.0 / n_cool)
    T = t_start
    patience = 50000
    while time.time() < deadline:
        i = zeros_idx[rng.randint(len(zeros_idx))]
        j = ones_idx[rng.randint(len(ones_idx))]
        delta = swap_delta(h, n, i, j, m_range)
        new_max = (corr + delta).max() * dx
        delta_c = new_max - cur_c
        if delta_c < 0 or (T > 1e-12 and rng.random() < np.exp(-delta_c / T)):
            h[i] = 1.0
            h[j] = 0.0
            corr = corr + delta
            cur_c = new_max
            zeros_idx = np.where(h == 0)[0]
            ones_idx = np.where(h == 1)[0]
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        T *= cool_rate
        if no_improve > patience:
            nk = rng.randint(3, 12)
            zi = zeros_idx.copy()
            oi = ones_idx.copy()
            nz = len(zi)
            no = len(oi)
            for k in range(min(nk, nz, no)):
                ii = zi[rng.randint(nz)]
                jj = oi[rng.randint(no)]
                h[ii] = 1.0
                h[jj] = 0.0
                zi[k], zi[nz-1] = zi[nz-1], zi[k]
                oi[k], oi[no-1] = oi[no-1], oi[k]
                nz -= 1
                no -= 1
            corr = np.correlate(h, 1 - h, mode='full').astype(float)
            cur_c = corr.max() * dx
            zeros_idx = np.where(h == 0)[0]
            ones_idx = np.where(h == 1)[0]
            no_improve = 0
            T = min(t_start * 1.5, T * 8.0)
    return best_h, best_c

def greedy_fast(h_bin, n, deadline):
    dx = 2.0 / n
    m_range = np.arange(-(n - 1), n)
    h = h_bin.astype(float).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    cur_c = corr.max() * dx
    best_h = h.copy()
    best_c = cur_c
    ny = 2 * n - 1
    while time.time() < deadline:
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        nz = len(zeros)
        no = len(ones)
        if nz == 0 or no == 0:
            break
        f = np.zeros((nz, ny))
        for idx in range(nz):
            z = zeros[idx]
            zm = z + m_range
            msk = (zm >= 0) & (zm < n)
            f[idx, msk] += 1.0 - h[zm[msk]]
            zmn = z - m_range
            msk = (zmn >= 0) & (zmn < n)
            f[idx, msk] -= h[zmn[msk]]
        g = np.zeros((no, ny))
        for idx in range(no):
            j = ones[idx]
            jm = j + m_range
            msk = (jm >= 0) & (jm < n)
            g[idx, msk] -= 1.0 - h[jm[msk]]
            jmn = j - m_range
            msk = (jmn >= 0) & (jmn < n)
            g[idx, msk] += h[jmn[msk]]
        combined = f[:, None, :] + g[None, :, :]
        new_maxes = (combined + corr[None, None, :]).max(axis=2)
        min_flat = np.argmin(new_maxes)
        a = min_flat // no
        b = min_flat % no
        new_c = new_maxes[a, b]
        if new_c >= cur_c - 1e-14:
            break
        h[zeros[a]] = 1.0
        h[ones[b]] = 0.0
        corr = corr + f[a] + g[b]
        cur_c = new_c
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def sa_multi(h_bin, n, deadline, seed=42, t_start=0.04, t_end=0.00001):
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    h = h_bin.astype(float).copy()
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    zeros_idx = np.where(h == 0)[0]
    ones_idx = np.where(h == 1)[0]
    no_improve = 0
    n_cool = 400000
    cool_rate = (t_end / t_start) ** (1.0 / n_cool)
    T = t_start
    patience = 40000
    while time.time() < deadline:
        if rng.random() < 0.8:
            i = zeros_idx[rng.randint(len(zeros_idx))]
            j = ones_idx[rng.randint(len(ones_idx))]
            h[i] = 1.0
            h[j] = 0.0
            new_corr = np.correlate(h, 1 - h, mode='full').astype(float)
            nc = new_corr.max() * dx
            delta_c = nc - cur_c
            if delta_c < 0 or (T > 1e-12 and rng.random() < np.exp(-delta_c / T)):
                corr = new_corr
                cur_c = nc
                zeros_idx = np.where(h == 0)[0]
                ones_idx = np.where(h == 1)[0]
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h[i] = 0.0
                h[j] = 1.0
                no_improve += 1
        else:
            nk = rng.randint(2, 5)
            old_h = h.copy()
            h_tmp = h.copy()
            zi = zeros_idx.copy()
            oi = ones_idx.copy()
            zi = zi[rng.choice(len(zi), min(nk, len(zi)), replace=False)]
            oi = oi[rng.choice(len(oi), min(nk, len(oi)), replace=False)]
            for k in range(len(zi)):
                h_tmp[zi[k]] = 1.0
                h_tmp[oi[k]] = 0.0
            new_corr = np.correlate(h_tmp, 1 - h_tmp, mode='full').astype(float)
            nc = new_corr.max() * dx
            delta_c = nc - cur_c
            if delta_c < 0 or (T > 1e-12 and rng.random() < np.exp(-delta_c / T)):
                h = h_tmp
                corr = new_corr
                cur_c = nc
                zeros_idx = np.where(h == 0)[0]
                ones_idx = np.where(h == 1)[0]
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        T *= cool_rate
        if no_improve > patience:
            nk = rng.randint(5, 15)
            h = h.copy()
            zi = np.where(h == 0)[0]
            oi = np.where(h == 1)[0]
            zi = zi[rng.choice(len(zi), min(nk, len(zi)), replace=False)]
            oi = oi[rng.choice(len(oi), min(nk, len(oi)), replace=False)]
            for k in range(len(zi)):
                h[zi[k]] = 1.0
                h[oi[k]] = 0.0
            corr = np.correlate(h, 1 - h, mode='full').astype(float)
            cur_c = corr.max() * dx
            zeros_idx = np.where(h == 0)[0]
            ones_idx = np.where(h == 1)[0]
            no_improve = 0
            T = min(t_start, T * 5.0)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = budget_s - 3
    deadline = t0 + total
    try:

            best_h = hg.copy()
    best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, best_c, n
