    m = corr.max()

    return m + (sh.max() + np.log(np.exp(sh - sh.max()).sum())) / alpha

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    """Project h to {0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_grad_lse(h, alpha, dx):
    """Compute gradient of log-sum-exp objective"""
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full') * dx
    m = corr.max()
    sh = alpha * (corr - m)
    sh -= sh.max()
    p = np.exp(sh)
    p /= p.sum()
    
    # Compute gradient: sum_k p[k] * grad_corr[k]
    # grad_corr[k][i] = dx * [(1-h[i-shift])*1_{0<=i-shift<n} - h[i+shift]*1_{0<=i+shift<n}]
    # where shift = k - (n-1)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-15:
            continue
        shift = k - (n - 1)
        if shift == 0:
            grad += pk * dx * ((1 - h) - h)  # = pk * dx * (1 - 2*h)
        elif shift > 0:

            grad[shift:] += pk * dx * (1 - h[:n-shift])
            # -h[i+shift] for i in [0, n-1-shift]
            grad[:n-shift] -= pk * dx * h[shift:]
        else:
            s = -shift
            # (1-h[i-shift]) = (1-h[i+s]) for i in [0, n-1-s]
            grad[:n-s] += pk * dx * (1 - h[s:])
            # -h[i+shift] = -h[i-s] for i in [s, n-1]

    return grad

def grad_descent_lse(h_start, n, dx, deadline, alpha_start=10, alpha_end=5000, n_iters=5000, seed=42):
    """Log-sum-exp gradient descent with Nesterov momentum"""
    rng = np.random.RandomState(seed)
    h = h_start.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)

    v = np.zeros(n)
    beta = 0.9

        alpha = alpha_start * (alpha_end / alpha_start) ** progress

        sum_err = h_look.sum() - n / 2.0

        for _ in range(20):
            h_new = h - step * g
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c or c_new < compute_c5(h, n):

            step *= 0.5
        
        # Update momentum
        v = beta * v + (1 - beta) * (h_new - h)
        h = h_new
        
        # Track best
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    return best_h, best_c

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def build_g(h, n):
    g = np.zeros((n, 2 * n - 1))
    for p in range(n):
        ks = max(0, p)
        ke = min(2 * n - 2, p + n - 1)
        for k in range(ks, ke + 1):
            if k != 2 * p:
                g[p, k] = 1.0 - 2.0 * h[k - p]
    return g

def lse_grad_descent(h_start, n, deadline, seed=42):
    """Log-sum-exp gradient descent with Nesterov momentum"""
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    h = project_feasible(h_start.astype(float).copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    v = np.zeros(n)
    beta = 0.85
    step = 0.3
    
    it = 0
    total_iters = 5000
    while time.time() < deadline and it < total_iters:
        # Adaptive alpha
        progress = it / total_iters
        alpha = 20.0 * (2000.0 / 20.0) ** progress
        
        # Nesterov lookahead
        h_look = project_feasible(h + beta * v, n)
        
        # Compute gradient
        corr = np.correlate(h_look, 1 - h_look, mode='full') * dx
        m = corr.max()
        sh = alpha * (corr - m)
        sh -= sh.max()
        p = np.exp(sh)
        p /= p.sum()
        
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            pk = p[k]
            if pk < 1e-14:
                continue
            shift = k - (n - 1)
            if shift > 0:
                grad[shift:] += pk * dx * (1.0 - h_look[:n-shift])
                grad[:n-shift] -= pk * dx * h_look[shift:]
            elif shift < 0:
                s = -shift
                grad[:n-s] += pk * dx * (1.0 - h_look[s:])
                grad[s:] -= pk * dx * h_look[:n-s]
            else:
                grad += pk * dx * (1.0 - 2.0 * h_look)
        
        # Step
        h_new = project_feasible(h - step * grad, n)
        
        # Update
        v = beta * v + (1.0 - beta) * (h_new - h)
        h = h_new
        it += 1
        
        if it % 50 == 0:
            c = compute_c5(h, n)
            if c < best_c:
                best_c = c
                best_h = h.copy()
        
        # Decay step size
        step = max(0.01, step * 0.9995)
    
    c = compute_c5(h, n)
    if c < best_c:
        best_c = c
        best_h = h.copy()
    
    return best_h, best_c

def quantize_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:n//2]] = 1.0
    return h_bin

def thue_morse_init(n, rng):
    t = np.zeros(n, dtype=int)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    target = n // 2
    num_ones = int(t.sum())
    if num_ones > target:
        ones = np.where(t == 1)[0]
        rng.shuffle(ones)
        t[ones[:num_ones - target]] = 0
    elif num_ones < target:
        zeros = np.where(t == 0)[0]
        rng.shuffle(zeros)
        t[zeros[:target - num_ones]] = 1
    return t.astype(float)

def slsqp_polish(h, n, deadline, alpha=1000000):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        return m + (sh.max() + np.log(np.exp(sh - sh.max()).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 2000, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h
