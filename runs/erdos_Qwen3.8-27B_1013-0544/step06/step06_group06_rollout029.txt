Let me now think about the **continuous relaxation** approach more carefully.

For continuous h ∈ [0,1]^n with sum = n/2:
f(h) = max_k corr_k(h) where corr_k = sum_i h[i]*(1-h[i+k-(n-1)])

This is a max of quadratic functions. The subgradient at h for the active k* is:
∂corr_{k*}/∂h[i] = (1-h[i+k*-(n-1)]) - h[i-k*+(n-1)] (with out-of-bounds = 0)

Wait, that's not right. Let me recompute:
corr_k = sum_i h[i] * (1-h[i+k-(n-1)])
∂corr_k/∂h[j] = (1-h[j+k-(n-1)]) * [j+k-(n-1) valid] - h[j-k+(n-1)] * [j-k+(n-1) valid]

So the subgradient of f at h (where k* is active) is the vector with components:
g[j] = (1-h[j+k*-(n-1)]) * [valid] - h[j-k*+(n-1)] * [valid]

We want to minimize f, so we take a step in the negative gradient direction:
h_new = h - alpha * g

Then project onto {0 ≤ h ≤ 1, sum(h) = n/2}.

The projection is: clip to [0,1], then adjust to meet the sum constraint. A simple approach:
1. Clip to [0,1]
2. Compute diff = n/2 - sum(h)
3. If diff > 0: add diff * (1-h) / (1-h).sum() to h
4. If diff < 0: add diff * h / h.sum() to h
5. Repeat until converged (or just do a few iterations)

This is basically what `project_feasible` does.

For the step size, I can use a simple decreasing schedule: alpha_t = alpha_0 / (1 + t * decay).

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def build_flip_effect(h, n):
    m = 2 * n - 1
    FE = np.zeros((n, m))
    for p in range(n):
        lo1 = max(0, n - 1 - p)
        hi1 = min(m - 1, n - 1 + p)
        if lo1 <= hi1:
            FE[p, lo1:hi1+1] = 1.0 - h[p + np.arange(lo1, hi1+1) - (n-1)]
        lo2 = max(0, p)
        hi2 = min(m - 1, p + n - 1)
        if lo2 <= hi2:
            FE[p, lo2:hi2+1] -= h[p - np.arange(lo2, hi2+1) + (n-1)]
    return FE

def local_search_binary(h, n, deadline, top_k=30, seed=42):

    dx = 2.0 / n
    h = h.copy()

    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    
    FE = build_flip_effect(h, n)
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    
    temperature = 0.005
    stall = 0
    no_improve = 0
    
    while time.time() < deadline:
        n_z = len(zeros)
        n_on = len(ones)
        if n_z == 0 or n_on == 0:
            break
        
        top_idx = np.argsort(corr)[-top_k:]
        corr_top = corr[top_idx]
        FE_z = FE[zeros][:, top_idx]
        FE_o = FE[ones][:, top_idx]

        new_maxes = D.max(axis=2)

            j = ones[jj]

            h[j] = 0.0

                    else:

def subgradient_continuous(h, n, deadline, max_iter=50000):
    dx = 2.0 / n
    h = project_feasible(h, n).copy()
    best_c = compute_c5(h, n)
    best_h = h.copy()
    
    alpha = 0.1
    decay = 1e-5
    
    for t in range(max_iter):
        if time.time() > deadline:
            break
        
        corr = np.correlate(h, 1-h, mode='full').astype(float)
        k_star = np.argmax(corr)
        
        # Subgradient at h for the active k_star
        g = np.zeros(n)
        for i in range(n):
            idx1 = i + k_star - (n-1)
            if 0 <= idx1 < n:
                g[i] += (1.0 - h[idx1])
            idx2 = i - k_star + (n-1)
            if 0 <= idx2 < n:
                g[i] -= h[idx2]
        
        h = project_feasible(h - alpha * g, n)
        c = compute_c5(h, n)
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
        
        alpha = 0.1 / (1 + t * decay)
    
    return best_h, best_c

def thue_morse_binary(n, rng):
    t = np.zeros(n, dtype=int)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    target = n // 2
    num_ones = t.sum()
    if num_ones > target:
        ones = np.where(t == 1)[0]
        rng.shuffle(ones)
        t[ones[:num_ones - target]] = 0
    elif num_ones < target:
        zeros = np.where(t == 0)[0]
        rng.shuffle(zeros)
        t[zeros[:target - num_ones]] = 1
    return t.astype(float)

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:

            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:

            h = h + diff / s * h

            FE[p, k1] -= 1.0
        k2 = j - p + (n - 1)
        if 0 <= k2 < m:
            FE[p, k2] += 1.0
        k3 = p - (n - 1) - i
        if 0 <= k3 < m:
            FE[p, k3] -= 1.0
        k4 = p - (n - 1) - j
        if 0 <= k4 < m:
            FE[p, k4] += 1.0

def local_search(h, n, deadline, top_k=50, seed=42):

    dx = 2.0 / n
    h = h.copy()

    best_h = h.copy()

    temperature = 0.005
    stall = 0
    no_improve = 0

        zeros = np.where(h == 0)[0]

        n_on = len(ones)

        ii = flat_idx // n_on
        jj = flat_idx % n_on
        best_val = new_maxes[ii, jj] * dx

            accept = True
        else:
            delta = best_val - cur_c
            accept = temperature > 1e-9 and rng.random() < np.exp(-delta / max(temperature, 1e-9))

            j = ones[jj]

            update_fe_swap(FE, n, i, j)
            h[i] = 1.0
            h[j] = 0.0
            cur_c = corr.max() * dx
            stall = 0
            temperature *= 0.999

    best_c = compute_c5(best_h, n)
    return best_h, best_c

def thue_morse_binary(n, rng):
    t = np.zeros(n, dtype=int)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    target = n // 2
    num_ones = int(t.sum())
    if num_ones > target:
        ones = np.where(t == 1)[0]
        rng.shuffle(ones)
        t[ones[:num_ones - target]] = 0
    elif num_ones < target:
        zeros = np.where(t == 0)[0]
        rng.shuffle(zeros)
        t[zeros[:target - num_ones]] = 1
    return t.astype(float)
