import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _active_smooth_obj(x, n, top_idx, alpha):
    dx = 2.0 / n
    cc = np.correlate(x, 1 - x, mode='full') * dx
    active = cc[top_idx]
    m = active.max()
    shifted = alpha * (active - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def _lag_grad(h, n, lag):
    grad = np.zeros(n)
    m = abs(lag)
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if lag > 0:
        grad[:n - m] -= h[m:]
        grad[m:] += (1.0 - h[:n - m])
    else:
        grad[:n - m] += (1.0 - h[m:])
        grad[m:] -= h[:n - m]
    return grad

def slsqp_opt(h, n, alpha, maxiter=4000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(_smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def slsqp_active(h, n, top_idx, alpha, maxiter=4000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(_active_smooth_obj, h, args=(n, top_idx, alpha),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def cutting_plane(h_init, n, deadline, max_iters=40):
    h_curr = h_init.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(h_curr, n)
    dx = 2.0 / n
    cuts = []
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c = compute_all_lags(h_curr, n)
        top_idx = np.argsort(c)[-4:]
        for idx in top_idx:
            lag = idx - (n - 1)
            g = _lag_grad(h_curr, n, lag)
            cuts.append((g, float(c[idx] / dx), h_curr.copy()))
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        cons = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1]
        for g, cv, hp in cuts:
            cons.append(cv + g @ (h_var - hp) <= t_var)
        prob = cp.Problem(cp.Minimize(t_var), cons)
        try:
            prob.solve(solver=cp.GLOP)
        except Exception:
            try:
                prob.solve(solver=cp.CBC)
            except Exception:
                break
        if prob.status not in ['optimal', 'optimal_inaccurate']:
            break
        h_new = h_var.value
        if h_new is None:
            break
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        if np.max(np.abs(h_new - h_curr)) < 1e-9:
            break
        h_curr = h_new
    return best_h, best_c

def grad_local_search(h, n, deadline, max_iters=20000):
    h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        k_star = int(np.argmax(c))
        lag = k_star - (n - 1)
        if lag == 0:
            break
        grad = _lag_grad(h, n, lag)
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if grad[i] - grad[j] < 1e-12:
            break
        max_move = min(h[i], 1.0 - h[j])
        if max_move < 1e-10:
            break
        moved = False
        for frac in [0.5, 0.25, 0.1, 0.05, 0.01, 0.005]:
            delta = frac * max_move
            if delta < 1e-10:
                continue
            h[i] -= delta
            h[j] += delta
            c_new = compute_c5(h, n)
            if c_new < best_c - 1e-14:
                best_c = c_new
                moved = True
                break
            h[i] += delta
            h[j] -= delta
        if not moved:
            break
    return h, best_c

def pair_local_search(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.002, 0.001, 0.0005, 0.0002]

                    if hi > hj + 1e-10:

                        if delta < 1e-9:

                        h[i] -= delta; h[j] += delta
                        cn = compute_c5(h, n)
                        if cn < best_c - 1e-14:
                            best_c = cn; improved = True
                        else:
                            h[i] += delta; h[j] -= delta
                    elif hj > hi + 1e-10:
                        delta = min(d, (hj - hi) * 0.5, hj, 1.0 - hi)
                        if delta < 1e-9:
                            continue
                        h[i] += delta; h[j] -= delta
                        cn = compute_c5(h, n)
                        if cn < best_c - 1e-14:

                        else:
                            h[i] -= delta; h[j] += delta
            else:
                for i in range(m, n):
                    if time.time() > deadline:
                        break
                    j = i - m
                    hi, hj = h[i], h[j]
                    if hi > hj + 1e-10:
