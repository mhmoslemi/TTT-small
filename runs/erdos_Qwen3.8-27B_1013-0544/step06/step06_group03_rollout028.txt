- h[i] appears as h[i+m'] for some other lag... no wait, we're looking at a specific lag m.
- In S_m = sum_{i=0}^{n-1-m} h[i+m]*(1-h[i]):
  - h[i] appears only in term i (as the (1-h[i]) part)
  - h[j] = h[i+m] appears only in term i (as the h[i+m] part)
  
  Wait, that's not right either. h[i] appears as the "first factor" in terms where the index is i-m (if i ≥ m): h[i] = h[(i-m)+m], so it appears in term (i-m) as the first factor.
  
  And h[i] appears as the "second factor" (inside 1-h[...]) in terms where i is the index: h[i+m]*(1-h[i]), so h[i] is in term i.

So h[i] appears in:
- Term i: as (1-h[i]), coefficient h[i+m]
- Term i-m: as h[i] (if i ≥ m), coefficient (1-h[i-m])

And h[j] = h[i+m] appears in:
- Term i: as h[i+m], coefficient (1-h[i])
- Term i+m-m = i: already counted
- Term j-m = i: already counted (same as term i)

Wait, I'm confusing myself. Let me be very explicit.

S_m = sum_{t=0}^{n-1-m} h[t+m] * (1-h[t])

h[i] appears in this sum when:
1. t+m = i, i.e., t = i-m (if 0 ≤ i-m ≤ n-1-m, i.e., m ≤ i ≤ n-1): coefficient is (1-h[i-m])
2. t = i (if 0 ≤ i ≤ n-1-m): coefficient is -h[i+m]

So the total contribution of h[i] to S_m is:
(1-h[i-m]) * 1_{m ≤ i} + (-h[i+m]) * 1_{i ≤ n-1-m}

And similarly for h[j] = h[i+m]:
(1-h[i+m-m]) * 1_{m ≤ i+m} + (-h[i+m+m]) * 1_{i+m ≤ n-1-m}
= (1-h[i]) * 1_{m ≤ i+m} (always true) + (-h[i+2m]) * 1_{i+m ≤ n-1-m}

So if I change h[i] → h[i]+δ and h[i+m] → h[i+m]-δ:
ΔS_m = δ * [(1-h[i-m]) * 1_{i ≥ m} - h[i+m] * 1_{i ≤ n-1-m}]  (from h[i])
       - δ * [(1-h[i]) * 1_{i+m ≥ m} - h[i+2m] * 1_{i+m ≤ n-1-m}]  (from h[i+m])

This is getting complicated because of the boundary effects. For interior points (m ≤ i and i ≤ n-1-m and i+2m < n and i-m ≥ 0):
ΔS_m = δ * [(1-h[i-m]) - h[i+m] - (1-h[i]) + h[i+2m]]
= δ * [h[i] - h[i+m] - h[i-m] + h[i+2m]]

This is the "four-point" interaction. It's not simply 2δ*(h[i]-h[i+m]) as I initially thought.

So the optimal δ for a single lag is:
δ* = -[(1-h[i-m]) - h[i+m] - (1-h[i]) + h[i+2m]] / 2

import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            r = 1.0 - h
            t = r.sum()
            if t < 1e-14:
                break
            h = h + (d / t) * r
        else:
            t = h.sum()
            if t < 1e-14:
                break
            h = h + (d / t) * h
    return np.clip(h, 0.0, 1.0)

def _sc(x, n):
    return float(x.sum() - n / 2.0)

def _scj(x, n):
    return np.ones(n)

def lag_grad(h, n, lag):
    g = np.zeros(n)
    m = abs(lag)
    if m == 0:
        return 1.0 - 2.0 * h
    if m >= n:
        return g
    if lag > 0:
        g[m:] += (1.0 - h[:n - m])
        g[:n - m] -= h[m:]
    else:
        g[:n - m] += (1.0 - h[m:])
        g[m:] -= h[:n - m]
    return g

def lp_step(h0, n, trust):
    dx = 2.0 / n
    c = compute_all_lags(h0, n)
    hv = cp.Variable(n)
    tv = cp.Variable()
    cons = [cp.sum(hv) == n / 2.0, hv >= 0, hv <= 1, cp.abs(hv - h0) <= trust]
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        g = lag_grad(h0, n, lag)
        cons.append(c[k] + dx * float(g @ (hv - h0)) <= tv)
    prob = cp.Problem(cp.Minimize(tv), cons)
    for solver in [cp.MOSEK, cp.GLOP, cp.ECOS, cp.CBC]:
        try:
            prob.solve(solver=solver)
            if prob.status in ['optimal', 'optimal_inaccurate'] and hv.value is not None:
                return project_feasible(np.array(hv.value, dtype=float), n)
        except Exception:
            continue
    return h0.copy()

def smooth_obj(x, n, alpha):
    c = compute_all_lags(x, n)
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def slsqp_run(h, n, alpha, maxiter=3000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sc, 'jac': _scj, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def subgrad_polish(h, n, deadline):
    h_curr = h.copy()
    best_c = compute_c5(h_curr, n)
    best_h = h_curr.copy()
    for t in range(30000):
        if time.time() > deadline:
            break
        c = compute_all_lags(h_curr, n)
        cmax = c.max()
        thresh = cmax - 0.003
        active = np.where(c >= thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        g = np.zeros(n)
        for k in active:
            g += lag_grad(h_curr, n, k - (n - 1))
        g /= len(active)
        g *= 2.0 / n
        ng = np.linalg.norm(g)
        if ng < 1e-15:
            break
        step = min(0.03, 0.5 / (ng * np.sqrt(t + 1)))
        h_new = project_feasible(h_curr - step * g, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
        h_curr = h_new
    return best_h, best_c

def pair_local_search(h, n, deadline):
    h_curr = h.copy()
    best_c = compute_c5(h_curr, n)
    best_h = h_curr.copy()
    for d in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
        if time.time() > deadline:
            break
        improved_any = False
        for _ in range(30):
            if time.time() > deadline:
                break
            c = compute_all_lags(h_curr, n)
            k_star = int(np.argmax(c))
            lag = k_star - (n - 1)
            if lag == 0:
                break
            m = abs(lag)
            count = min(n - m, 300)
            idx = np.random.randint(0, n - m, size=count)
            for i in idx:
                if time.time() > deadline:
                    break
                if lag > 0:
                    a, b = i, i + m
                else:
                    a, b = i - m, i
                da = h_curr[a]
                db = h_curr[b]
                if da > db:
                    delta = min(d, (da - db) * 0.4)
                    if delta < 1e-10:
                        continue
                    h_curr[a] -= delta
                    h_curr[b] += delta
                    cn = compute_c5(h_curr, n)
                    if cn < best_c - 1e-14:
                        best_c = cn
                        best_h = h_curr.copy()
                        improved_any = True
                    else:
                        h_curr[a] += delta
                        h_curr[b] -= delta
                elif db > da:
                    delta = min(d, (db - da) * 0.4)
                    if delta < 1e-10:
                        continue
                    h_curr[b] -= delta
                    h_curr[a] += delta
                    cn = compute_c5(h_curr, n)
                    if cn < best_c - 1e-14:
                        best_c = cn
                        best_h = h_curr.copy()
                        improved_any = True
                    else:
                        h_curr[b] += delta
                        h_curr[a] -= delta
        if not improved_any:
            break
    return best_h, best_c

def optimize_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    t_oo_end = min(deadline, time.time() + (deadline - time.time()) * 0.45)
    trust = 0.12
    for it in range(150):
        if time.time() > t_oo_end - 1:
            break
        h_new = lp_step(best_h, n, trust)
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-13:
            best_c = c_new
            best_h = h_new.copy()
        step = np.max(np.abs(h_new - best_h))
        if step < 1e-10:
            break
        trust = max(0.002, trust * 0.93)
    if time.time() < deadline - 5:
        for alpha in [20000, 100000]:
            if time.time() > deadline - 5:
                break
            h_new = slsqp_run(best_h, n, alpha)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-13:
                best_c = c_new
                best_h = h_new.copy()
    if time.time() < deadline - 5:
        h_new, c_new = subgrad_polish(best_h, n, min(deadline - 2, time.time() + (deadline - time.time()) * 0.3))
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
    if time.time() < deadline - 2:
        best_h, best_c = pair_local_search(best_h, n, deadline - 0.5)
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c
