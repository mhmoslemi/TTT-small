`np.correlate(a, v, mode='full')` gives an array of length `len(a) + len(v) - 1`.
The element at index `k` (0-indexed) is:
`result[k] = sum_i a[i] * v[i + k - (len(v) - 1)]`

Wait no, let me look at the actual definition. From numpy docs:
"Correlate a with v, with the mode given by mode. In 'full' mode, the output is the valid set of correlations, without zero-padding or border effects, of length 2*N-1 where N is the length of the shorter sequence."

Actually the precise definition is:
`correlate(a, v)[k] = sum_n a[n+k] * conj(v[n])` for `k = -(len(v)-1), ..., len(a)-1`

So for `np.correlate(h, 1-h, mode='full')`:
`result[k] = sum_n h[n+k] * (1-h[n])` where k ranges from -(n-1) to n-1.

In the array, `result[0]` corresponds to `k = -(n-1)`, `result[n-1]` corresponds to `k = 0`, etc.

So `result[idx]` corresponds to `k = idx - (n-1)`.

For a given k (the lag), `result[n-1+k] = sum_n h[n+k] * (1-h[n])` where the sum is over all valid n (0 ≤ n < n and 0 ≤ n+k < n).

Now, the derivative with respect to h[j]:
- h[j] appears in the term h[n+k] when j = n+k, i.e., n = j-k
  - This is valid when 0 ≤ j-k < n AND 0 ≤ n = j-k < n (same condition)
  - And we need n+k = j to be in [0, n-1], which is automatic
  - Contribution: (1-h[n]) = (1-h[j-k])
- h[j] appears in the term (1-h[n]) when n = j
  - This is valid when 0 ≤ j < n (automatic) AND 0 ≤ j+k < n
  - Contribution: -h[j+k]

So: `d(result[n-1+k])/dh[j] = (1-h[j-k]) * [0≤j-k<n] - h[j+k] * [0≤j+k<n]`

import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def subgradient_step(h, n, k, lr):
    sg = np.zeros(n)
    if k >= 0:
        sg[k:] += (1.0 - h[:n - k])
        sg[:n - k] -= h[k:n]
    else:
        kk = -k
        sg[kk:] += (1.0 - h[:n - kk])
        sg[kk:] -= h[:n - kk]
    h_new = h - lr * sg
    return project_feasible(h_new, n)

def round_to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    return out

def swap_delta(h, n, i, j, m_range):
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    delta[k2] += 1.0
    return delta

            return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

            s = seed + 100
        elif attempt == 2:
            start_h = round_to_binary(0.5 + 0.5 * rng.randn(n), n)
            s = seed + 200
        else:
            blk = max(1, n // 25)
            pattern = np.zeros(n)
            for b in range(n // blk + 1):
                val = 0.6 if b % 3 != 0 else 0.4
                s_idx = b * blk
                e_idx = min(n, s_idx + blk)
                pattern[s_idx:e_idx] = val
            start_h = round_to_binary(pattern, n)
            s = seed + 300
        hb, cb = binary_sa(start_h, n, p3, seed=s, t_start=0.035, t_end=0.00001, patience=30000)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()

    # Phase 4: Directed local search
    p4 = t0 + 0.70 * (budget_s - 5)
    h_bin2 = round_to_binary(best_h, n)
    hd, cd = directed_local_search(h_bin2, n, p4, seed=seed + 500)
    if cd < best_c:
        best_c = cd
        best_h = hd.copy()

    # Phase 5: SLSQP refinement from best
    p5 = t0 + 0.84 * (budget_s - 5)
    if time.time() < p5 - 5:
        hp = project_feasible(best_h + 0.05 * rng.randn(n), n)
        hp2, cp2 = slsqp_optimize(hp, n, p5, alphas=[10000, 50000, 200000])
        if cp2 < best_c:
            best_c = cp2
            best_h = hp2.copy()

    # Phase 6: Final SA + subgradient
    p6 = dl - 15
    if time.time() < p6 - 5:
        h_bin3 = round_to_binary(best_h, n)
        hb3, cb3 = binary_sa(h_bin3, n, p6 - 5, seed=seed + 700, t_start=0.02, t_end=0.00005, patience=20000)
        if cb3 < best_c:
            best_c = cb3
            best_h = hb3.copy()

    if time.time() < dl - 8:
        h_sg2, c_sg2 = subgradient_optimize(best_h, n, 50000, 0.04, dl - 5, seed=seed + 800)
        if c_sg2 < best_c:
            best_c = c_sg2
            best_h = h_sg2.copy()

    # Try higher resolution if time remains
    if time.time() < dl - 40 and n < 500:
        n2 = min(500, n * 2)
        xo = np.linspace(0, 2, n)
        xn = np.linspace(0, 2, n2)
        hh = np.interp(xn, xo, best_h)
        hh = project_feasible(hh, n2)
        hbin2 = round_to_binary(hh, n2)
        hb4, cb4 = binary_sa(hbin2, n2, dl - 5, seed=seed + 900, t_start=0.02, t_end=0.0001, patience=15000)
        if cb4 < best_c:
            best_c = cb4
            best_h = hb4.copy()
            n = n2

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
