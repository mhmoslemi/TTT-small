    else:

        grad[:n-p] += (1.0 - h[p:])
        grad[p:] += -h[:n-p]
    return grad

    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):

        if abs(diff) < 1e-14:

        if diff > 0:

            tot = room.sum()
            if tot < 1e-14:

            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_grad(h, n, m):
    g = np.zeros(n)
    if m == 0:
        g[:] = 1.0 - 2.0 * h
    elif m > 0:
        g[m:] += (1.0 - h[:n-m])
        g[:n-m] -= h[m:]
    else:
        p = -m
        g[:n-p] += (1.0 - h[p:])
        g[p:] -= h[:n-p]
    return g

def smooth_max_fn(h, n, top_idx, alpha):
    c = np.correlate(h, 1 - h, mode='full') * (2.0/n)
    a = c[top_idx]
    m = a.max()
    s = alpha * (a - m)
    mx = s.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha
    return m + np.log(np.sum(np.exp(s))) / alpha

    return float(np.sum(x) - n / 2.0)

    return np.ones(n)

def slsqp_opt(h, n, top_idx, alpha, maxiter=5000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (n,)}]
    try:

                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

    while time.time() < deadline - 1:
        it += 1
        c = compute_all_lags(hc, n)
        ti = np.argsort(c)[-top_m:]

        sh = alpha * (a - a.max())
        mx = sh.max()
        w = np.exp(sh - mx) if mx > 500 else np.exp(sh)
        w /= w.sum()
        g = np.zeros(n)
        for i in range(len(ti)):
            g += w[i] * lag_grad(hc, n, ti[i] - (n-1)) * (2.0/n)
        lr = lr0 / (1.0 + 0.05 * np.sqrt(it))

        hn = project_feasible(hc + mom, n)

        if cn < best_c - 1e-14:
            best_c = cn
            best_h = hn.copy()
        hc = hn
    return best_h, best_c

            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(30, (deadline - time.time() - 15) * 0.5)
            if alloc < 12:
                break
            h_r, c_r = optimize_n(hp, n, time.time() + alloc)
            if c_r < best_c:
                best_c, best_h = c_r, h_r

    if best_h is not None and time.time() < deadline - 8:
        c = compute_all_lags(best_h, len(best_h))
        ti = np.argsort(c)[-5:]
        hf = slsqp_opt(best_h, len(best_h), ti, 1000000, maxiter=8000)
        cf = compute_c5(hf, len(best_h))
        if cf < best_c:
            best_c, best_h = cf, hf
        best_h = project_feasible(best_h, len(best_h))
        best_c = compute_c5(best_h, len(best_h))

    if best_h is None:
        best_h = project_feasible(np.full(n0, 0.5), n0)
        best_c = compute_c5(best_h, n0)

    return best_h, best_c, len(best_h)
