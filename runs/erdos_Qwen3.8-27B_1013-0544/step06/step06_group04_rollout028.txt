from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c)) * dx

def precompute_A(h, n):
    g = 1.0 - h
    m = 2 * n - 1
    A = np.zeros((n, m))
    for p in range(n):
        # L part: g[p-k+n-1] for valid k
        lo = max(0, p - n + 1)
        hi = min(m - 1, p)
        idx = np.arange(lo, hi + 1)
        A[p, idx] = g[p - idx + n - 1]
        # R part: h[p+k-n+1] for valid k
        lo2 = max(0, n - 1 - p)
        hi2 = min(m - 1, n - 1 + p)
        idx2 = np.arange(lo2, hi2 + 1)
        A[p, idx2] -= h[p + idx2 - n + 1]
    return A

def update_A_swap(A, h, i, j, n):
    """Update A after swapping h[i] (0->1) and h[j] (1->0)."""
    m = 2 * n - 1
    # L[i+k-n+1, k] -= 1 for valid k
    lo = max(0, n - 1 - i)
    hi = min(m - 1, n - 1 + i)
    k_vals = np.arange(lo, hi + 1)
    p_vals = i + k_vals - n + 1
    A[p_vals, k_vals] -= 1.0
    # L[j+k-n+1, k] += 1 for valid k
    lo = max(0, n - 1 - j)
    hi = min(m - 1, n - 1 + j)
    k_vals = np.arange(lo, hi + 1)
    p_vals = j + k_vals - n + 1
    A[p_vals, k_vals] += 1.0
    # R[i-k+n-1, k] += 1 for valid k (A[i-k+n-1, k] -= 1)
    lo = max(0, i - n + 1)
    hi = min(m - 1, i)
    k_vals = np.arange(lo, hi + 1)
    p_vals = i - k_vals + n - 1
    A[p_vals, k_vals] -= 1.0
    # R[j-k+n-1, k] -= 1 for valid k (A[j-k+n-1, k] += 1)
    lo = max(0, j - n + 1)
    hi = min(m - 1, j)
    k_vals = np.arange(lo, hi + 1)
    p_vals = j - k_vals + n - 1
    A[p_vals, k_vals] += 1.0
    # Update h
    h[i] = 1.0
    h[j] = 0.0

def fast_local_search(h, n, deadline, seed=42, max_sweeps=500):
    h = np.array(h, dtype=float).copy()
    # Binarize
    target = n // 2
    idx = np.argsort(h)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    
    dx = 2.0 / n
    m = 2 * n - 1
    c = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur_c = float(c.max()) * dx
    best_c = cur_c
    best_h = h.copy()
    A = precompute_A(h, n)
    e_n1 = np.zeros(m)
    e_n1[n - 1] = 2.0
    rng = np.random.RandomState(seed)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        zeros = np.where(h < 0.5)[0]
        rng.shuffle(zeros)
        for i in zeros:
            if time.time() > deadline:
                break
            ones = np.where(h >= 0.5)[0]
            if len(ones) == 0:
                break
            D = c + A[i] - e_n1
            diffs = D[np.newaxis, :] - A[ones, :]
            new_maxes = diffs.max(axis=1)
            bi = new_maxes.argmin()
            if new_maxes[bi] < cur_c - 1e-10:
                j = ones[bi]
                c = c + A[i] - A[j] - e_n1
                update_A_swap(A, h, i, j, n)
                cur_c = float(c.max()) * dx
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                improved = True
        if not improved:
            break
    return best_h, best_c

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def slsqp_smooth(h, n, alpha, maxiter=300):
    dx = 2.0 / n
