    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

    g = np.correlate(h, 1 - h, mode='full')
    dx = 2.0 / len(h)
    m = g.max()
    sh = alpha * (g - m)
    sm = sh.max()
    soft = m + (np.log(np.exp(sh - sm).sum()) + sm) / alpha
    penalty = mu * (h.sum() - target_sum) ** 2
    return soft * dx + penalty

from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def soft_max_and_weights(g, alpha):
    m = g.max()
    sh = alpha * (g - m)
    sm = sh.max()
    exp_sh = np.exp(sh - sm)
    w = exp_sh / exp_sh.sum()
    val = m + (np.log(exp_sh.sum()) + sm) / alpha
    return val, w

def continuous_obj_and_grad(x, alpha, mu, target_sum):
    n = len(x)
    dx = 2.0 / n
    g = np.correlate(x, 1 - x, mode='full')
    val, w = soft_max_and_weights(g, alpha)
    obj = val * dx + mu * (x.sum() - target_sum) ** 2
    # Gradient
    grad = np.zeros(n)
    for p in range(2 * n - 1):
        wi = w[p]
        if wi < 1e-14:
            continue
        lag = p - (n - 1)
        if lag >= 0:
            # Term 1: j from lag to n-1
            grad[lag:] += wi * (1.0 - x[:n - lag])
            # Term 2: j from 0 to n-1-lag
            grad[:n - lag] += wi * (-x[lag:])
        else:
            lag2 = -lag
            # Term 1: j from 0 to n-1-lag2
            grad[:n - lag2] += wi * (1.0 - x[lag2:])
            # Term 2: j from lag2 to n-1
            grad[lag2:] += wi * (-x[:n - lag2])
    grad *= dx
    grad += 2.0 * mu * (x.sum() - target_sum)
    return obj, grad

    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        lambda x: continuous_obj_and_grad(x, alpha, mu, target_sum)[0],
        h0.copy(),
        jac=lambda x: continuous_obj_and_grad(x, alpha, mu, target_sum)[1],
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10}

                    if 0 <= p < 2 * n - 1:
                        # h[j] was 1, now 0
                        # i'+lag = j → i' = j-lag: if 0 <= j-lag < n: was 1*(1-h[j-lag]), now 0
                        if 0 <= j - lag < n:
                            new_corr[p] -= (1.0 - h[j - lag])
                        # i' = j: if 0 <= j+lag < n: was h[j+lag]*(1-1)=0, now h[j+lag]*(1-0)
                        if 0 <= j + lag < n:

                    cur_c = nc

                        best_c = nc
                        best_h = h.copy()
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return best_h, best_c
