import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-13:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def coord_descent_step(h, i, n, c, s, mu, top_k=15):
    """Find optimal h[i] given all else fixed. Returns (new_t, new_max_val)."""
    dx = 2.0 / n
    old = h[i]
    
    # Compute a[m] = contribution to c[m] from changing h[i]
    # c[m] changes by (t - old) * a[m] where:
    # a[m] = dx * [(1-h[i+lag]) if valid - h[i-lag] if valid]
    lag = np.arange(-(n-1), n)
    a = np.zeros(2*n - 1)
    j1 = i + lag
    v1 = (j1 >= 0) & (j1 < n)
    a[v1] += (1.0 - h[j1[v1]])
    j2 = i - lag
    v2 = (j2 >= 0) & (j2 < n)
    a[v2] -= h[j2[v2]]
    a *= dx
    
    # Objective: max_m(c[m] + (t-old)*a[m]) + mu*(s + (t-old))^2
    # where s = sum(h) - n/2
    # This is convex in t on [0,1]
    
    # Candidate points: 0, 1, and intersections of the top-k lines
    top_idx = np.argsort(c)[-top_k:][::-1]
    candidates = [0.0, 1.0]
    for idx in range(len(top_idx)):
        for jdx in range(idx+1, len(top_idx)):
            m1, m2 = top_idx[idx], top_idx[jdx]
            denom = a[m1] - a[m2]

def nesterov_hedge(h, n, deadline, alpha_init=100.0, start_lr=0.05):
    """Nesterov-accelerated projected gradient with adaptive alpha."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    target = n / 2.0
    mu = 2.0
    lr = start_lr
    alpha = alpha_init
    y = h.copy()
    beta = 0.9
    
    for it in range(50000):
        if time.time() > deadline:
            break
        g = grad_smooth_max(y, n, alpha)
        g += 2.0 * mu * (y.sum() - target)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm
        
        # Nesterov step
        new_y = y - lr * g
        new_y = project_feasible(new_y, n)
        h_new = (1 - beta) * new_y + beta * (new_y - (y - h))  # momentum
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c - 1e-15:
            best_c = c_new
            best_h = h_new.copy()
            y = h_new
        else:
            lr *= 0.7
            alpha *= 1.3
            if lr < 1e-10:
                break
        beta = 0.9 * (0.5 + 0.5 * it / min(it+1, 100))
    
    return best_h, best_c

def multi_start(h, n, deadline, n_starts=5, perturb=0.05):

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_vec(h, n)
    c_shift = alpha * (c - c.max())
    mx = c_shift.max()
    w = np.exp(c_shift - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-13:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def coord_descent_full(h, n, deadline, mu=5.0, max_passes=200, top_k=12):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    target = n / 2.0
    for _pass in range(max_passes):
        if time.time() > deadline:
            break
        c = compute_c5_vec(h, n)
        s = h.sum() - target
        improved = False
        for i in range(n):
            if time.time() > deadline:
                return best_h, best_c
            old = h[i]
            dx = 2.0 / n
            lag = np.arange(-(n - 1), n)
            a = np.zeros(2 * n - 1)
            j1 = i + lag
            v1 = (j1 >= 0) & (j1 < n)
            a[v1] += (1.0 - h[j1[v1]])
            j2 = i - lag
            v2 = (j2 >= 0) & (j2 < n)
            a[v2] -= h[j2[v2]]
            a *= dx
            top_idx = np.argsort(c)[-top_k:][::-1]
            cands = np.array([0.0, 1.0, old - s])
            for ii in range(len(top_idx)):
                for jj in range(ii + 1, len(top_idx)):
                    m1, m2 = top_idx[ii], top_idx[jj]
                    denom = a[m1] - a[m2]
                    if abs(denom) > 1e-15:
                        tc = old + (c[m2] - c[m1]) / denom
                        if -0.01 < tc < 1.01:
                            cands = np.append(cands, tc)
            cands = np.clip(cands, 0.0, 1.0)
            cands = np.unique(cands)
            diff_t = cands - old
            vals = (c[None, :] + diff_t[:, None] * a[None, :]).max(axis=1)
            vals += mu * (s + diff_t) ** 2
            best_idx = np.argmin(vals)
            best_t = cands[best_idx]
            if abs(best_t - old) > 1e-13:
                h[i] = best_t
                s = s - old + best_t
                improved = True
        c = compute_c5_vec(h, n)
        actual_c = c.max()
        if actual_c < best_c + 1e-12:
            best_c = actual_c
            best_h = h.copy()
        if not improved:
            break
    return best_h, best_c

def lbfgs_phase(h, n, deadline, max_iters=300):
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    target = n / 2.0
    configs = [(100, 5.0), (500, 10.0), (2000, 20.0), (10000, 50.0),
               (50000, 100.0), (200000, 500.0), (1000000, 2000.0)]
    for alpha, mu in configs:
        if time.time() > deadline - 2:
            break
        def pobj(x, a=alpha, m=mu):
            c = compute_c5_vec(x, n)
            mx_val = c.max()
            sh = a * (c - mx_val)
            s2 = sh.max()
            if s2 > 500:
                val = mx_val + (s2 + np.log(np.sum(np.exp(sh - s2)))) / a
            else:
                val = mx_val + np.log(np.sum(np.exp(sh))) / a
            val += m * (x.sum() - target) ** 2
            return float(val)
        def pgrad(x, a=alpha, m=mu):
            g = grad_smooth_max(x, n, a)
            g += 2.0 * m * (x.sum() - target)
            return g
        try:
            res = minimize(pobj, h, jac=pgrad, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-16,
                                   'gtol': 1e-13, 'maxfun': max_iters * 10, 'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-15:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def slsqp_phase(h, n, deadline, alpha=500000, maxiter=1500):

    target = n / 2.0
    def sobj(x, a=alpha):

        mx_val = c.max()
        sh = a * (c - mx_val)
        s2 = sh.max()
        if s2 > 500:
            return float(mx_val + (s2 + np.log(np.sum(np.exp(sh - s2)))) / a)
        return float(mx_val + np.log(np.sum(np.exp(sh))) / a)
    cons = [{'type': 'eq',
             'fun': lambda x, n=n: x.sum() - n / 2.0,
             'jac': lambda x, n=n: np.ones(n)}]
    try:
        res = minimize(sobj, h, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_feasible(res.x, n)
        c_new = compute_c5(h_new, n)
        return h_new, c_new
    except Exception:
        return h.copy(), compute_c5(h, n)

def multi_start_refine(h, n, deadline, n_starts=5, perturb=0.03):
    best_c = compute_c5(h, n)
