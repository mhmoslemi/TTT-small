    #   The term involving h[i] from the first factor: h[i] * (1-h[i-k]) → contributes +d * (1-h[i-k])
    #   The term involving h[i] from the (1-h) factor: h[n] * (1-h[i]) where n-k=i, so n=i+k
    #   → h[i+k] * (1-(h[i]-d)) ... no, (1-h[i]) becomes (1-h[i]-d) which is (1-h[i]) - d
    #   Wait, (1-h[n-k]) where n-k = i: (1-h[i]) → (1-(h[i]+d)) = (1-h[i]) - d
    #   So the term h[i+k] * (1-h[i]) becomes h[i+k] * ((1-h[i]) - d) = h[i+k]*(1-h[i]) - d*h[i+k]
    #   Change: -d * h[i+k] (valid when 0 <= i+k < n)
    
    # So total change to c[m] from h[i] += d:
    #   +d * (1-h[i-k]) [if 0 <= i-k < n] - d * h[i+k] [if 0 <= i+k < n]
    
    # Similarly for h[j] -= d:
    #   -d * (1-h[j-k]) [if 0 <= j-k < n] + d * h[j+k] [if 0 <= j+k < n]
    
    # Net coefficient for c[m] (times d):
    #   (1-h[i-k]) - h[i+k] - (1-h[j-k]) + h[j+k]
    #   where each term is included only if the index is valid
    
    # For the active set, we want to minimize max_m (c[m] + d * coef[m])
    # subject to 0 <= h[i]+d <= 1 and 0 <= h[j]-d <= 1
    
    # This is: minimize max_m (c[m] + d * coef[m])
    # = find d that minimizes the maximum of linear functions
    # The optimal d is where the "active" lines cross
    
    # Use bisection on the subgradient:
    # subgradient at d is: sum of coef[m] for m that achieve the max
    # We want subgradient to contain 0, i.e., the max is achieved by both positive and negative coef
    
    # Simple approach: try d values and pick the best
    # Or: the optimal d is the "median" of breakpoints d = (c[m'] - c[m]) / (coef[m] - coef[m'])
    
    # For simplicity, use a grid search over d
    d_min = max(-h[i], h[j] - 1.0)  # from h[i]+d >= 0 and h[j]-d <= 1
    d_max = min(1.0 - h[i], h[j])    # from h[i]+d <= 1 and h[j]-d >= 0
    if d_min >= d_max:
        return 0.0, 0.0
    
    # Compute coefficients for active shifts
    coefs = []
    for m in active_shifts:
        k = m - (n - 1)
        coef = 0.0
        if 0 <= i - k < n:
            coef += (1.0 - h[i-k])
        if 0 <= i + k < n:
            coef -= h[i+k]
        if 0 <= j - k < n:
            coef -= (1.0 - h[j-k])
        if 0 <= j + k < n:
            coef += h[j+k]
        coefs.append(coef)
    coefs = np.array(coefs)
    
    # c values at d=0
    c_vals = compute_c5_all(h, n)[active_shifts]
    
    # Minimize max_m (c_vals[m] + d * coefs[m]) over d in [d_min, d_max]
    # This is a convex 1D problem. Use ternary search or bisection.
    # Or just grid search with enough points.
    d_grid = np.linspace(d_min, d_max, 200)
    obj_vals = np.max(c_vals[np.newaxis, :] + d_grid[:, np.newaxis] * coefs[np.newaxis, :], axis=1)
    best_idx = np.argmin(obj_vals)
    return d_grid[best_idx], obj_vals[best_idx]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_all(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha):
    c = compute_c5_all(h, n)
    c_shift = alpha * (c - c.max())
    c_shift = np.clip(c_shift, -500, 0)
    w = np.exp(c_shift)
    w /= (w.sum() + 1e-30)
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-15:
            continue
        k = m - (n - 1)
        wm = w[m] * dx
        if k >= 0:
            nv = n - k
            grad[0:nv] += wm * (1.0 - h[k:n])
            grad[k:n] += wm * (-h[0:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:n] += wm * (1.0 - h[0:nv])
            grad[0:nv] += wm * (-h[kk:n])
    return grad

def _penalty_obj(x, n, alpha, mu):
    c = compute_c5_all(x, n)
    mx = c.max()
    sh = alpha * (c - mx)
    sh = np.clip(sh, -500, 0)
    w = np.exp(sh)
    val = mx + np.log(w.sum()) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def _penalty_grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

def lbfgs_phase(h, n, deadline, alpha_list, max_iter=200):
    best_c = compute_c5(h, n)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    for alpha in alpha_list:
        if time.time() > deadline - 5:
            break
        mu = max(0.1, alpha * 1e-5)
        try:
            res = minimize(_penalty_obj, best_h, jac=_penalty_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iter, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': max_iter * 12,
                                   'maxls': 30})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c

def active_set_pair_search(h, n, deadline, top_k=20, max_rounds=50):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    c_all = compute_c5_all(best_h, n)
    m = c_all.shape[0]
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        # Find active set (top shifts)
        sorted_idx = np.argsort(-c_all)
        active = sorted_idx[:top_k]
        active_c = c_all[active]
        # Compute coefficients for all pairs
        improved = False
        for i in range(n):
            if time.time() > deadline:
                return best_h, best_c
            for j in range(i + 1, n):
                if time.time() > deadline:
                    return best_h, best_c
                if best_h[i] < 1e-10 or best_h[j] < 1e-10:
                    continue
                # Compute coefficient of d for each active shift
                d_min = max(-best_h[i], best_h[j] - 1.0)
                d_max = min(1.0 - best_h[i], best_h[j])
                if d_min >= d_max - 1e-12:
                    continue
                coefs = np.zeros(top_k)
                for a_idx, am in enumerate(active):
                    k = am - (n - 1)
                    coef = 0.0
                    if 0 <= i - k < n:
                        coef += (1.0 - best_h[i - k])
                    if 0 <= i + k < n:
                        coef -= best_h[i + k]
                    if 0 <= j - k < n:
                        coef -= (1.0 - best_h[j - k])
                    if 0 <= j + k < n:
                        coef += best_h[j + k]
                    coefs[a_idx] = coef
                # Minimize max(active_c + d * coefs) over d in [d_min, d_max]
                # Try a set of candidate d values
                candidates = np.array([0.0, (d_min + d_max) / 2])
                # Add breakpoints where pairs of lines cross
                for a in range(top_k):
                    for b in range(a + 1, top_k):
                        denom = coefs[a] - coefs[b]
                        if abs(denom) > 1e-14:
                            db = (active_c[b] - active_c[a]) / denom
                            if d_min < db < d_max:
                                candidates = np.append(candidates, db)
                # Clip and evaluate
                candidates = np.clip(candidates, d_min + 1e-12, d_max - 1e-12)
                best_d = 0.0
                best_val = best_c
                for d in candidates:
                    obj = np.max(active_c + d * coefs)
                    if obj < best_val - 1e-15:
                        best_val = obj
                        best_d = d
                if best_d != 0.0 and abs(best_d) > 1e-10:
                    best_h[i] += best_d
                    best_h[j] -= best_d
                    best_h = np.clip(best_h, 0.0, 1.0)
                    best_h = project_feasible(best_h, n)
                    c_new = compute_c5(best_h, n)
                    if c_new < best_c - 1e-15:
                        best_c = c_new
                        c_all = compute_c5_all(best_h, n)
                        improved = True
                    else:
                        best_h[i] -= best_d
                        best_h[j] += best_d
                        best_h = project_feasible(best_h, n)
                        c_all = compute_c5_all(best_h, n)
            if not improved:
                break
    return best_h, best_c

def simple_pair_search(h, n, deadline, deltas, radius):
    best_c = compute_c5(h, n)
    for d in deltas:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def gradient_descent_polish(h, n, deadline, alpha=50000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    lr = 0.02
    for _ in range(5000):
        if time.time() > deadline:
            break
        g = grad_smooth_max(best_h, n, alpha)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm
        step = lr
        improved = False
        for _ in range(30):
            h_new = project_feasible(best_h - step * g, n)
            c_new = compute_c5(h_new, n)
            if c_new <= best_c - 1e-15:
                best_h = h_new
                best_c = c_new
                improved = True
                break
            step *= 0.5
        if not improved:
            lr *= 0.5
        if lr < 1e-14:
            break
    return best_h, best_c
