def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + 1.0/alpha * (sm + np.log(np.exp(sh - sm).sum()))
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:

                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:

        active = np.where(corr >= threshold)[0]
        if len(active) > n_active_max:
            active = active[np.argsort(corr[active])[-n_active_max:]]
        
        grad = np.zeros(n)
        for k in active:
            d = k - (n - 1)
            jd = np.arange(n) + d
            mpos = (jd >= 0) & (jd < n)
            grad[mpos] += (1.0 - h[jd[mpos]])
            jn = np.arange(n) - d
            mneg = (jn >= 0) & (jn < n)
            grad[mneg] += (-h[jn[mneg]])
        
        h_new = project_feasible(h - step * grad, n)
        new_corr = np.correlate(h_new, 1 - h_new, mode='full').astype(float)
        new_val = new_corr.max()
        
        if new_val < cur_val:
            h = h_new
            corr = new_corr
            cur_val = new_val
            step = min(step * 1.02, 0.2)
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
        else:
            step *= 0.9
        
        if it % 500 == 0:
            step = max(step, base_step * 0.1)
    
    return best_h, best_val * dx

Wait, the subgradient might be too slow with the loop over active shifts. Let me vectorize it better.

For a given offset d, the contribution to grad is:
- (1-h) shifted by +d (i.e., (1-h)[j+d] for valid j)
- -h shifted by -d (i.e., -h[j-d] for valid j)

I can compute this for all active d's at once using a loop (since n is small) or by precomputing.

Actually for n=250, even a Python loop over 20 active shifts with numpy vectorized inner operations is fine. Each inner iteration does O(n) work with numpy, so 20*250 = 5000 element operations per gradient evaluation. With 3000 iterations, that's 15M operations total. Should be a few seconds at most.

Let me also think about the binary SA more carefully. The key improvement I want:

**Bottleneck-aware move proposal**: Instead of picking i from zeros and j from ones uniformly, I'll:
1. Identify the top K shifts
2. For each top shift, find the "critical" zero positions (zeros that, if set to 1, would most reduce that shift)
3. Sample i from the union of critical positions across all top shifts (with some probability of uniform sampling to avoid getting stuck)

For shift k with offset d, setting h[i]=1 (i currently 0) changes corr[k] by:
(1-h[i+d]) - h[i-d] (with validity)

We want this to be negative, so we want h[i+d]=1 and h[i-d] to be small.

Actually, this is getting complicated. Let me just use a simpler heuristic:
- With probability 0.7, pick i from zeros that are "near" (within some window) positions where h=1 for a bottleneck shift
- With probability 0.3, pick uniformly

Or even simpler: just weight the zero positions by how many bottleneck shifts they're "implicated in".

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-12:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-12:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def round_to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    return out

def swap_delta(h, n, i, j, m_range):
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    delta[k2] += 1.0
    return delta

def slsqp_smooth(h, n, alpha, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + 1.0 / alpha * (sm + np.log(np.exp(sh - sm).sum()))
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h

def subgrad_opt(h, n, max_iter=2000, base_step=0.05):
    corr = np.correlate(h, 1 - h, mode='full').astype(float)
    cur_val = corr.max()
    step = base_step
    best_h = h.copy()
    best_val = cur_val
    for it in range(max_iter):
        thresh = cur_val * 0.98
        active = np.where(corr >= thresh)[0]
        if len(active) > 15:
            active = active[np.argsort(corr[active])[-15:]]
        grad = np.zeros(n)
        for k in active:
            d = k - (n - 1)
            jd = np.arange(n) + d
            mp = (jd >= 0) & (jd < n)
            grad[mp] += (1.0 - h[jd[mp]])
            jn = np.arange(n) - d
            mn = (jn >= 0) & (jn < n)
            grad[mn] += (-h[jn[mn]])
        h_new = project_feasible(h - step * grad, n)
        nc = np.correlate(h_new, 1 - h_new, mode='full').astype(float)
        nv = nc.max()
        if nv < cur_val:
            h, corr, cur_val = h_new, nc, nv
            step = min(step * 1.02, 0.15)
            if nv < best_val:
                best_val, best_h = nv, h.copy()
        else:
            step *= 0.92
        if it % 400 == 0:
            step = max(step, base_step * 0.08)
    return best_h, best_val

def binary_sa(h_bin, n, deadline, seed=42, t_start=0.02, t_end=1e-5, patience=25000):
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    h = h_bin.copy()
    m_range = np.arange(-(n - 1), n)
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    zeros_idx = np.where(h == 0)[0]
    ones_idx = np.where(h == 1)[0]
    no_improve = 0
    cool_rate = (t_end / t_start) ** (1.0 / 400000.0)
    T = t_start
    topK = 5
    while time.time() < deadline:
        top_k = np.argsort(corr)[-topK:]
        use_bottleneck = rng.random() < 0.6 and len(zeros_idx) > 0 and len(ones_idx) > 0
        if use_bottleneck:
            # bias toward zeros near bottleneck shift positions
            d0 = top_k[0] - (n - 1)
            weights = np.zeros(n)
            for kk in top_k:
                d = kk - (n - 1)
                for zi in zeros_idx:
                    zd = zi + d
                    if 0 <= zd < n and h[zd] == 1:
                        weights[zi] += 1.0
                    zn = zi - d
                    if 0 <= zn < n and h[zn] == 1:
                        weights[zi] += 0.5
            wsum = weights.sum()
            if wsum > 1e-10:
                pw = weights / wsum
                i = rng.choice(zeros_idx, p=pw[zeros_idx])
            else:
                i = zeros_idx[rng.randint(len(zeros_idx))]
            j = ones_idx[rng.randint(len(ones_idx))]
        else:
            i = zeros_idx[rng.randint(len(zeros_idx))]
            j = ones_idx[rng.randint(len(ones_idx))]
        delta = swap_delta(h, n, i, j, m_range)
        new_corr = corr + delta
        nc = new_corr.max() * dx
        dc = nc - cur_c
        if dc < 0 or (T > 1e-12 and rng.random() < np.exp(-dc / T)):
            h[i] = 1.0
            h[j] = 0.0
            corr = new_corr
            cur_c = nc
            zeros_idx = np.where(h == 0)[0]
            ones_idx = np.where(h == 1)[0]
            if nc < best_c:
                best_c = nc
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        T *= cool_rate
        if no_improve > patience:
            nk = 4 + rng.randint(8)
            for _ in range(nk):
                ii = zeros_idx[rng.randint(len(zeros_idx))]
                jj = ones_idx[rng.randint(len(ones_idx))]
                h[ii] = 1.0
                h[jj] = 0.0
            corr = np.correlate(h, 1 - h, mode='full').astype(float)
            cur_c = corr.max() * dx
            zeros_idx = np.where(h == 0)[0]
            ones_idx = np.where(h == 1)[0]
            no_improve = 0
            T = min(t_start, T * 4.0)
    return best_h, best_c

def focused_greedy(h_bin, n, deadline, max_sweeps=200, seed=42):
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    h = h_bin.copy()
    m_range = np.arange(-(n - 1), n)
    corr = np.correlate(h, 1 - h, mode='full').astype(float).copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    K = 8
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        top_k = np.argsort(corr)[-K:]
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        # sort zeros by how much they'd help top shifts
        z_scores = np.zeros(len(zeros))
        for zi_idx, zi in enumerate(zeros):
            for kk in top_k:
                d = kk - (n - 1)
                zd = zi + d
                if 0 <= zd < n and h[zd] == 1:
                    z_scores[zi_idx] += 1.0
                zn = zi - d
                if 0 <= zn < n and h[zn] == 1:
                    z_scores[zi_idx] += 0.5
        for zi in zeros[np.argsort(z_scores)[::-1][:min(50, len(zeros))]]:
            best_j = -1
            best_nc = cur_c
            for j in ones:
                delta = swap_delta(h, n, zi, j, m_range)
                nc = (corr + delta).max() * dx
                if nc < best_nc - 1e-15:
                    best_nc = nc
                    best_j = j
            if best_j >= 0:
                h[zi] = 1.0
                h[best_j] = 0.0
                corr = corr + swap_delta(h, n, zi, best_j, m_range) - swap_delta(h, n, zi, best_j, m_range)
                # recompute properly
                h[zi] = 1.0
                h[best_j] = 0.0
                diff = swap_delta(h, n, zi, best_j, m_range)
                # Actually just recompute
                corr = np.correlate(h, 1 - h, mode='full').astype(float)
                cur_c = corr.max() * dx
                improved = True
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                break
        if not improved:
            nk = 5 + rng.randint(5)
            for _ in range(nk):
                ii = zeros[rng.randint(len(zeros))]
                jj = ones[rng.randint(len(ones))]
                h[ii] = 1.0
                h[jj] = 0.0
            corr = np.correlate(h, 1 - h, mode='full').astype(float)
            cur_c = corr.max() * dx
    return best_h, best_c
