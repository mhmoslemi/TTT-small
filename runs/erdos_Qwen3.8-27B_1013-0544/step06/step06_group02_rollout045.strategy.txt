        import numpy as np
        from scipy.optimize import differential_evolution
        import time

        def compute_c5(h, n):
            dx = 2.0 / n
            return float(np.max(np.correlate(h, 1-h, mode='full')) * dx)

        def project_feasible(h, n):
            # Copy and clip to [0,1]
            target = n/2.0
            h = h.copy()
            h = np.clip(h, 0.0, 1.0)
            # 300 iterations to adjust the total
            for i in range(300):
                diff = target - h.sum()
                if abs(diff) < 1e-15:
                    break
                if diff > 0:
                    # There is room to add
                    room = 1.0 - h
                    tot_room = room.sum()
                    if tot_room < 1e-15:
                        break
                    h += (diff / tot_room) * room
                    h = np.clip(h, 0.0, 1.0)
                else:
                    tot = h.sum()
                    if tot < 1e-15:
                        break
                    h += (diff / tot) * h
                # Clip again
                h = np.clip(h, 0.0, 1.0)
            return h.clip(0, 1)

        def run(seed=42, budget_s=1100, **kwargs):
            t0 = time.time()
