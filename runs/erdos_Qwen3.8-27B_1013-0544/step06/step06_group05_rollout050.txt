    m = (jm >= 0) & (jm < n); d[m] -= h[jm[m]]

        improved = False

            # or h[i]=1 and h[i+lag]=0 (for negative lag)

```python
def compute_grad(c, x, n, weights):
    """Compute gradient of sum_k weights[k]*c[k] w.r.t. x."""
    dx = 2.0 / n
    grad = np.zeros(n)
    for d in range(-(n-1), n):
        w = weights[n-1+d] * dx
        if abs(w) < 1e-15:
            continue
        if d >= 0:
            # (1-x[i-d]) for i in [d, n-1]
            idx = np.arange(d, n)
            grad[idx] += w * (1 - x[idx - d])
            # -x[i+d] for i in [0, n-1-d]
            idx2 = np.arange(0, n-d)
            grad[idx2] -= w * x[idx2 + d]
        else:
            e = -d
            # (1-x[i+e]) for i in [0, n-1-e]
            idx = np.arange(0, n - e)
            grad[idx] += w * (1 - x[idx + e])
            # -x[i-e] for i in [e, n-1]
            idx2 = np.arange(e, n)
            grad[idx2] -= w * x[idx2 - e]
    return grad
```
