from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def make_binary(h, n, rng):
    """Convert h to a binary vector with exactly n/2 ones."""
    # Sort by value, take top n/2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:n//2]] = 1.0
    return h_bin

def binary_sa(h_bin, n, deadline, rng, T_init=1.0, T_min=0.01, n_iter_per_T=200):
    """Simulated annealing on binary h."""
    best_h = h_bin.copy()
    best_c = compute_c5(h_bin, n)
    current_c = best_c
    T = T_init
    
    one_pos = np.where(h_bin == 1)[0]
    zero_pos = np.where(h_bin == 0)[0]
    
    while time.time() < deadline:
        # Cooldown
        T = max(T_min, T * 0.99)
        
        for _ in range(n_iter_per_T):
            if time.time() > deadline:
                return best_h, best_c
            i = rng.randint(len(one_pos))
            j = rng.randint(len(zero_pos))
            pi = one_pos[i]
            pj = zero_pos[j]
            
            # Swap
            h_bin[pi] = 0
            h_bin[pj] = 1
            
            c_new = compute_c5(h_bin, n)
            
            if c_new < current_c or rng.rand() < np.exp(-(c_new - current_c) / T):
                current_c = c_new
                one_pos[i] = pj  # pi is now 0, pj is now 1
                zero_pos[j] = pi

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_correlation(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def make_binary(h, n, rng):
    """Round to binary with exactly n//2 ones."""
    target_ones = n // 2
    current_ones = int(np.sum(h > 0.5))
    if current_ones == target_ones:
        return (h > 0.5).astype(float)
    # Need to adjust
    h_sorted = np.sort(h)[::-1]
    threshold = h_sorted[target_ones - 1] if target_ones > 0 else 1.0
    h_bin = (h >= threshold).astype(float)
    # Fix if needed
    ones = int(h_bin.sum())
    if ones > target_ones:
        idx = np.where(h_bin == 1)[0]
        # Remove the ones with lowest h values
        order = idx[np.argsort(h[idx])]
        for i in range(ones - target_ones):
            h_bin[order[i]] = 0
    elif ones < target_ones:
        idx = np.where(h_bin == 0)[0]
        order = idx[np.argsort(-h[idx])]
        for i in range(target_ones - ones):
            h_bin[order[i]] = 1
    return h_bin

    return h

def binary_sa(h_bin, n, deadline, rng, max_evals=500000):
    """Simulated annealing on binary h."""
    h = h_bin.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    current_c = best_c
    T = 0.1
    
    one_pos = np.where(h == 1)[0].copy()
    zero_pos = np.where(h == 0)[0].copy()
    n_evals = 0
    no_improve = 0
    
    while time.time() < deadline and n_evals < max_evals:
        # Generate a random swap
        i = rng.randint(len(one_pos))
        j = rng.randint(len(zero_pos))
        p = one_pos[i]
        q = zero_pos[j]
        
        # Perform swap
        h[p] = 0
        h[q] = 1
        
        c_new = compute_c5(h, n)
        n_evals += 1
        
        delta = c_new - current_c
        if delta < 0 or rng.rand() < np.exp(-delta / T):
            # Accept
            current_c = c_new
            # Update position arrays
            one_pos[i] = q  # p is now 0
            zero_pos[j] = p  # q is now 1
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # Reject
            h[p] = 1
            h[q] = 0
            no_improve += 1
        
        # Cool down
        if no_improve > 1000:
            T = max(0.001, T * 0.95)
    
    return best_h, best_c
