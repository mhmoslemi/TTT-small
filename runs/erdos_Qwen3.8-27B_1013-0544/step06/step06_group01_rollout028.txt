def pair_moves(h, n, deadline, rng):
    """Try moving pairs of coordinates."""
    h = h.copy()
    cur_c = compute_c5(h, n)
    scales = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    for scale in scales:
        improved = False
        for _ in range(5000):
            if time.time() > deadline:
                break
            if rng.uniform() < 0.5:
                p = rng.randint(0, n)
            else:
                p = rng.randint(0, n - 1)
            q = (p + rng.randint(1, min(20, n - p))) % n if p < n - 1 else 0
            if p == q:
                continue
            d = rng.uniform(-scale, scale)
            if h[p] + d < 0 or h[p] + d > 1:
                continue
            if h[q] - d < 0 or h[q] - d > 1:
                continue
            h[p] += d
            h[q] -= d
            new_c = compute_c5(h, n)
            if new_c < cur_c - 1e-15:
                cur_c = new_c
                improved = True
            else:
                h[p] -= d
                h[q] += d
        if not improved:
            break
    return h, cur_c
