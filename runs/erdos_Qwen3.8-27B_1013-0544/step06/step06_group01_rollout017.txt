    dx = 2.0 / n
    n_corr = 2 * n - 1

        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for p in range(n):
            if h[p] < 1e-10 or h[p] > 1.0 - 1e-10:
                continue
            # Compute sensitivity vector s[k] for changing h[p] by delta
            # c_new[k] = c[k] + delta*s[k] - delta^2 * (k==n-1)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    mx = c.max()
    sh = alpha * (c - mx)
    top = sh.max()
    if top > 500:
        return float((mx + (top + np.log(np.sum(np.exp(sh - top)))) / alpha) * dx)
    return float((mx + np.log(np.sum(np.exp(sh))) / alpha) * dx)
