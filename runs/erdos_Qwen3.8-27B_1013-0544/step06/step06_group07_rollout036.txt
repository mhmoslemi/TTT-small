    idx = np.argsort(h)[::-1]
    out = np.zeros(n)

def swap_delta(h, n, i, j, m_range):
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    if 0 <= k1 < 2 * n - 1:
        delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    if 0 <= k2 < 2 * n - 1:
        delta[k2] += 1.0
    return delta

def pt_sa(h_init, n, deadline, seed, n_rep=6, patience=30000):
    """Parallel tempering simulated annealing."""
    rng = np.random.RandomState(seed)
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    
    reps = [h_init.copy() for _ in range(n_rep)]
    corrs = [np.correlate(r, 1 - r, mode='full').astype(float) for r in reps]
    energies = [float(c.max() * dx) for c in corrs]
    
    # Temperatures: geometric spacing
    t_cold = 0.015
    t_hot = 0.15
    temps = [t_cold * (t_hot / t_cold) ** (i / (n_rep - 1)) for i in range(n_rep)]
    
    best_h = reps[0].copy()
    best_c = energies[0]
    no_imp = 0
    total_steps = 0
    
    while time.time() < deadline:
        for r in range(n_rep):
            h = reps[r]
            corr = corrs[r]
            T = temps[r]
            zeros = np.where(h == 0)[0]
            ones = np.where(h == 1)[0]
            i = zeros[rng.randint(len(zeros))]
            j = ones[rng.randint(len(ones))]
            if i == j:
                continue
            delta = swap_delta(h, n, i, j, m_range)
            new_corr = corr + delta
            new_c = float(new_corr.max() * dx)
            dc = new_c - energies[r]
            if dc <= 0 or (T > 1e-12 and rng.random() < np.exp(-dc / T)):
                h[i] = 1.0
                h[j] = 0.0
                corrs[r] = new_corr
                energies[r] = new_c
                if new_c < best_c:
                    best_c = new_c
                    best_h = h.copy()
                    no_imp = 0
                else:
                    no_imp += 1
            else:
                no_imp += 1
            total_steps += 1
        
        # Replica exchange (try adjacent pairs)
        for r in range(n_rep - 1):
            if rng.random() < 0.4:
                dE = energies[r+1] - energies[r]
                dIT = 1.0/temps[r] - 1.0/temps[r+1]
                la = dIT * dE
                if la >= 0 or rng.random() < np.exp(min(0.0, la)):
                    reps[r], reps[r+1] = reps[r+1], reps[r]
                    corrs[r], corrs[r+1] = corrs[r+1], corrs[r]
                    energies[r], energies[r+1] = energies[r+1], energies[r]
        
        # Restart if stuck
        if no_imp > patience:
            r = rng.randint(n_rep)
            h = reps[r]
            nz = np.where(h == 0)[0]
            no_ = np.where(h == 1)[0]
            nk = rng.randint(3, 8)
            for _ in range(nk):
                a = nz[rng.randint(len(nz))]
                b = no_[rng.randint(len(no_))]
                h[a] = 1.0
                h[b] = 0.0
            corrs[r] = np.correlate(h, 1 - h, mode='full').astype(float)
            energies[r] = float(corrs[r].max() * dx)
            no_imp = 0
    
    return best_h, best_c

def greedy_polish(h_init, n, deadline, seed=42, max_sweeps=300):
    """Greedy local search with random restarts."""
    rng = np.random.RandomState(seed)
    target = n // 2
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    dx = 2.0 / n
    m_range = np.arange(-(n-1), n)
    corr = np.correlate(h, 1-h, mode='full').astype(float).copy()
    cur_c = corr.max() * dx
    best_c = cur_c
    best_h = h.copy()
    for round_ in range(5):
        if time.time() > deadline: break
        for _ in range(max_sweeps):
            if time.time() > deadline: break
            improved = False
            zeros = np.where(h == 0)[0]
            ones = np.where(h == 1)[0]
            for i in zeros:
                for j in ones:
                    delta = swap_delta(h, n, i, j, m_range)
                    nc = (corr + delta).max() * dx
                    if nc < cur_c - 1e-15:
                        h[i], h[j] = 1.0, 0.0
                        corr = corr + delta
                        cur_c = nc
                        improved = True
                        break
                if improved: break
            if not improved: break
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        # Random kick
        if len(np.where(h==1)[0]) > 0 and len(np.where(h==0)[0]) > 0:
            oi = np.where(h == 1)[0]
            zi = np.where(h == 0)[0]
            for _ in range(4):
                h[rng.choice(oi)], h[rng.choice(zi)] = 0.0, 1.0
            corr = np.correlate(h, 1-h, mode='full').astype(float)
            cur_c = corr.max() * dx
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def round_to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    return out

def swap_delta(h, n, i, j, m_range):
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    if 0 <= k1 < 2 * n - 1:
        delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    if 0 <= k2 < 2 * n - 1:
        delta[k2] += 1.0
    return delta

def _obj(x, alpha, dx):
    g = np.correlate(x, 1 - x, mode='full') * dx
    m = g.max()
    sh = alpha * (g - m)
    sm = sh.max()
    return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

def _con(x, n):
    return x.sum() - n / 2.0

def pt_sa(h_init, n, deadline, seed, n_rep=6, patience=25000):
    rng = np.random.RandomState(seed)
    m_range = np.arange(-(n - 1), n)
    dx = 2.0 / n
    reps = [h_init.copy() for _ in range(n_rep)]
    corrs = [np.correlate(r, 1 - r, mode='full').astype(float) for r in reps]
    energies = [float(c.max() * dx) for c in corrs]
    t_cold, t_hot = 0.012, 0.25
    temps = [t_cold * (t_hot / t_cold) ** (i / (n_rep - 1)) for i in range(n_rep)]
    best_h = reps[0].copy()
    best_c = energies[0]
    no_imp = 0
    while time.time() < deadline:
        for r in range(n_rep):
            h = reps[r]
            corr = corrs[r]
            T = temps[r]
            z = np.where(h == 0)[0]
            o = np.where(h == 1)[0]
            i = z[rng.randint(len(z))]
            j = o[rng.randint(len(o))]
            if i == j:
                continue
            d = swap_delta(h, n, i, j, m_range)
            nc_arr = corr + d
            nc = float(nc_arr.max() * dx)
            dc = nc - energies[r]
            if dc <= 0 or (T > 1e-12 and rng.random() < np.exp(-dc / T)):
                h[i] = 1.0
                h[j] = 0.0
                corrs[r] = nc_arr
                energies[r] = nc
                if nc < best_c:
                    best_c = nc
                    best_h = h.copy()
                    no_imp = 0
                else:
                    no_imp += 1

                no_imp += 1

    return best_h, best_c
