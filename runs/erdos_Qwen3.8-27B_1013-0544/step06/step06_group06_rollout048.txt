        # We can compute this as: for each i, compute t_i[k] = corr[k] + g[i,k]
        # Then for each j, new_max = max_k (t_i[k] - g[j,k])
        
        # Vectorized:
        # t[i,k] = corr[k] + g[i,k] (for i in zeros, k in top_idx)
        t = corr[top_idx][None, :] + g_act  # shape (len(zeros), top_k)
        # new_max[i,j] = max_k (t[i,k] - g[j,k])
        # = max_k (t[i,k] + (-g[j,k]))
        # Let's compute: for each j, v_j[k] = -g[j,k]
        # new_max[i,j] = max_k (t[i,k] + v_j[k])
        # This is element-wise: (t[i] + v_j).max()
        
        # Shape: t is (nz, tk), g_act is (nz, tk) but we need g[ones, top_idx]
        g_o = g[ones, :]  # wrong, g is (n, 2n-1), we need g[ones, top_idx]
        g_o = g[ones][:, top_idx]  # shape (no, top_k)
        
        # new_max[i,j] = max_k (t[i,k] - g_o[j,k])
        # = max_k (t[i,k] + (-g_o[j,k]))
        # Reshape: t[:, None, :] - g_o[None, :, :] has shape (nz, no, tk)
        new_maxes = (t[:, None, :] - g_o[None, :, :]).max(axis=2)  # (nz, no)
        
        flat = new_maxes.argmin()
        ii = flat // len(ones)
        jj = flat % len(ones)
        best_val = new_maxes[ii, jj]
        i = zeros[ii]
        j = ones[jj]
        new_c = best_val * dx
        
        if new_c < cur_c - 1e-14:
            # Perform swap
            corr = corr + g[i] - g[j]
            # Update g for all other positions
            p_idx = np.arange(n)
            mi = (p_idx != i) & (p_idx + i < 2 * n - 1)
            g[p_idx[mi], p_idx[mi] + i] -= 2.0
            mj = (p_idx != j) & (p_idx + j < 2 * n - 1)
            g[p_idx[mj], p_idx[mj] + j] += 2.0
            h[i], h[j] = 1.0, 0.0
            zeros = np.delete(zeros, ii)
            ones = np.delete(ones, jj)
            zeros = np.append(zeros, j)
            ones = np.append(ones, i)
            cur_c = corr.max() * dx
            if cur_c < best_c - 1e-14:
                best_c = cur_c
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            if no_improve > 50 and kick_count < 30:
                kick_count += 1
                # Random multi-swap
                nk = min(2, len(zeros), len(ones))
                ri = rng.choice(len(zeros), nk, replace=False)
                rj = rng.choice(len(ones), nk, replace=False)
                for a in range(nk):
                    ii2 = np.where(zeros == zeros[ri[a]])[0][0]
                    jj2 = np.where(ones == ones[rj[a]])[0][0]
                    corr = corr + g[zeros[ri[a]]] - g[ones[rj[a]]]
                    p_idx = np.arange(n)
                    mi = (p_idx != zeros[ri[a]]) & (p_idx + zeros[ri[a]] < 2 * n - 1)
                    g[p_idx[mi], p_idx[mi] + zeros[ri[a]]] -= 2.0
                    mj = (p_idx != ones[rj[a]]) & (p_idx + ones[rj[a]] < 2 * n - 1)
                    g[p_idx[mj], p_idx[mj] + ones[rj[a]]] += 2.0
                    h[zeros[ri[a]]], h[ones[rj[a]]] = 1.0, 0.0
                    zeros = np.delete(zeros, ii2)
                    ones = np.delete(ones, jj2)
                    zeros = np.append(zeros, ones[rj[a]])
                    ones = np.append(ones, zeros[ri[a]])
                # Recompute
                corr = np.correlate(h, 1 - h, mode='full').astype(float)
                g = build_g(h, n)
                cur_c = corr.max() * dx
                ones = np.where(h == 1)[0].copy()
                zeros = np.where(h == 0)[0].copy()
                no_improve = 0
    
    return best_h, best_c

def thue_morse_init(n, rng):
    t = np.zeros(n, dtype=np.uint8)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    target = n // 2
    num_ones = int(t.sum())
    if num_ones > target:
        ones = np.where(t == 1)[0]
        rng.shuffle(ones)
        t[ones[:num_ones - target]] = 0
    elif num_ones < target:
        zeros = np.where(t == 0)[0]
        rng.shuffle(zeros)
        t[zeros[:target - num_ones]] = 1
    return t.astype(float)
