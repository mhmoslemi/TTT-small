    c = np.correlate(h, 1 - h, mode='full')

    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    h = h.copy()
    # Bisection on lambda for projection onto {0<=h-threshold<=1, sum=n/2}
    lo, hi = -1.0, 2.0
    for _ in range(100):

        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2, 0.0, 1.0)
    # Fix any small numerical issues

def projected_gradient(h, n, alpha, lr, n_steps, deadline):

    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        c = compute_all_lags(h, n)
        # Compute softmax-weighted gradient
        shifted = alpha * (c - c.max())
        shifted = np.clip(shifted, -500, 500)
        exp_s = np.exp(shifted)
        weights = exp_s / exp_s.sum()
        
        grad = np.zeros(n)
        thresh = 1e-4 / len(c)
        active_idx = np.where(weights > thresh)[0]
        for idx in active_idx:
            lag = idx - (n - 1)
            grad += weights[idx] * lag_gradient(h, n, lag)
        
        # Adaptive step size
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new, n)
        if c_new < best_c:

            h_new = h - lr * grad
            h_new = project_feasible(h_new, n)

import numpy as np
from scipy.optimize import minimize
import time

# === Core computations ===

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_lags(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    h = np.clip(h - t, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def lag_grad(h, n, lag):
    g = np.zeros(n)
    m = abs(lag)
    if m >= n:
        return g
    if m == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        v = n - m
        g[:v] += 1.0 - h[m:]
        g[m:] -= h[:v]
    else:
        v = n - m
        g[m:] += 1.0 - h[:v]
        g[:v] -= h[m:]
    return g

# === Smoothed objective and gradient ===

def _lse_obj(x, n, alpha):
    c = compute_all_lags(x, n)
    m = c.max()
    s = np.clip(alpha * (c - m), -500, 500)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def _lse_grad(x, n, alpha):
    c = compute_all_lags(x, n)
    s = np.clip(alpha * (c - c.max()), -500, 500)
    e = np.exp(s)
    w = e / e.sum()
    g = np.zeros(n)
    thresh = max(1e-8, 1e-5 / len(w))
    idx = np.where(w > thresh)[0]
    for i in idx:
        g += w[i] * lag_grad(x, n, int(i) - (n - 1))
    return g

    return float(x.sum() - n / 2.0)

def _sum_jac(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alpha, maxiter, ftol, deadline):
    h = project_feasible(h.copy(), n)
    c_best = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_jac, 'args': (n,)}]
    try:
        res = minimize(_lse_obj, h, args=(n, alpha), method='SLSQP',
                      jac=_lse_grad, bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None:
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < c_best:
                return h_new, c_new
    except Exception:
        pass
    return h, c_best

def projected_gradient(h, n, alpha_start, alpha_end, lr0, deadline):

    best_h = h.copy()

    lr = lr0
    alpha = alpha_start
    momentum = np.zeros(n)
    mu = 0.9

        c = compute_all_lags(h, n)
        s = np.clip(alpha * (c - c.max()), -500, 500)
        e = np.exp(s)
        w = e / e.sum()
        thresh = max(1e-8, 1e-4 / len(w))
        grad = np.zeros(n)
        idx = np.where(w > thresh)[0]
        for i in idx:
            grad += w[i] * lag_grad(h, n, int(i) - (n - 1))
        
        momentum = mu * momentum - lr * grad
        h_new = project_feasible(h + momentum, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
        else:

            c_new2 = compute_c5(h_new2, n)
            if c_new2 < best_c:
                best_c = c_new2
                best_h = h_new2.copy()
                h = h_new2

                lr *= 0.7
                momentum *= 0.5
