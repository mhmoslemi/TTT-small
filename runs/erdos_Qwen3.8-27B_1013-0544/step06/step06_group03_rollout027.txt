def compute_lags(h, n):

def project_feasible(h, n):
    target = n / 2.0
    h = h.astype(float).copy()

    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    # Fix residual
    diff = target - result.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = result < 0.999999
            count = mask.sum()
            if count > 0:
                result[mask] = np.clip(result[mask] + diff / count, 0, 1)
        else:
            mask = result > 1e-8
            count = mask.sum()
            if count > 0:
                result[mask] = np.clip(result[mask] + diff / count, 0, 1)
    return result

def lag_gradient(h, n, lag):
    dx = 2.0 / n
    m = abs(lag)
    if m >= n:
        return np.zeros(n)
    grad = np.zeros(n)
    if m == 0:
        grad[:] = (1.0 - 2.0 * h) * dx
    elif lag > 0:
        grad[:n-m] += (1.0 - h[m:]) * dx
        grad[m:] -= h[:n-m] * dx
    else:
        grad[:n-m] -= h[m:] * dx
        grad[m:] += (1.0 - h[:n-m]) * dx
    return grad

def obj_penalty(h_flat, n, alpha, lam_pen):
    h = h_flat
    c = compute_lags(h, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 400:
        smooth = m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    else:
        smooth = m + np.log(np.sum(np.exp(shifted))) / alpha
    s = h.sum() - n / 2.0
    return smooth + lam_pen * s * s

        c = compute_lags(x, n)
        if top_k >= len(c):
            c_a = c
        else:
            idx = np.argpartition(c, -top_k)[-top_k:]
            c_a = c[idx]
        m = c_a.max()
        shifted = alpha * (c_a - m)
        mx = shifted.max()
        if mx > 400:
            return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
        return float(m + np.log(np.sum(np.exp(shifted))) / alpha)
    def cf(x):
        return float(x.sum() - n / 2.0)
    def cj(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]

                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

        else:
            lr *= 0.75
            h = best_h.copy()
            if lr < 1e-8:

            p2 = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.003, 0.015)
            pert = amp * (np.sin(2 * np.pi * f1 * xg / n + p1) + 0.5 * np.sin(2 * np.pi * f2 * xg / n + p2))
            h_pert = project_feasible(best_h + pert, n)
            h_new = slsqp_active(h_pert, n, 5000, top_k=3, maxiter=1500, ftol=1e-14)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
            h_new, c_new = grad_local_search(h_new, n, deadline - 5, top_k=2, max_iters=10000)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
