import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(g, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(g - mid, 0.0, 1.0)
        if h.sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            room = (1.0 - h).sum()
            if room > 1e-15:
                h += diff / room * (1.0 - h)
        else:
            tot = h.sum()
            if tot > 1e-15:
                h += diff / tot * h
    return np.clip(h, 0.0, 1.0)

def smax_and_grad(h, n, alpha):
    c = compute_c5_vec(h, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for m_idx in range(2 * n - 1):
        if w[m_idx] < 1e-14:
            continue
        k = m_idx - (n - 1)
        wm = w[m_idx] * dx
        if k >= 0:
            nv = n - k
            grad[:nv] += wm * (1.0 - h[k:])
            grad[k:] += wm * (-h[:nv])
        else:
            kk = -k
            nv = n - kk
            grad[kk:] += wm * (1.0 - h[:nv])
            grad[:nv] += wm * (-h[kk:])
    return val, grad

def fista_opt(h0, n, deadline, alpha, max_iters=5000):
    h = h0.copy()
    z = h0.copy()
    t = 1.0
    best_h = h.copy()
    best_c = compute_c5(h, n)
    val0, g0 = smax_and_grad(h, n, alpha)
    L = max(np.linalg.norm(g0) * 5.0, 0.1)
    step = 1.0 / L
    for it in range(max_iters):
        if time.time() > deadline:
            break
        _, g_z = smax_and_grad(z, n, alpha)
        y = project_feasible(z - step * g_z, n)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        z = y + ((t - 1.0) / t_new) * (y - h)
        z = project_feasible(z, n)
        t = t_new
        h = y
        c = compute_c5(h, n)
        if c < best_c - 1e-15:
            best_c = c
            best_h = h.copy()
        if it % 50 == 0:
            gn = np.linalg.norm(g_z)
            L = max(gn * 5.0, L * 0.85)
            step = 1.0 / L
        if gn < 1e-15:
            break
    return best_h, best_c

def lbfgs_penalty(h, n, deadline, alpha, mu, max_iters=500):
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        v, _ = smax_and_grad(x, n, alpha)
        return float(v + mu * (x.sum() - n / 2.0) ** 2)
    def obj_grad(x):
        _, g = smax_and_grad(x, n, alpha)
        g = g + 2.0 * mu * (x.sum() - n / 2.0) * np.ones(n)
        return g
    try:
        res = minimize(obj, h, jac=obj_grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': max_iters, 'ftol': 1e-18, 'gtol': 1e-12})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def slsqp_opt(h, n, deadline, alpha, max_iters=2000):
    bounds = [(0.0, 1.0)] * n
    def obj(x):
        v, _ = smax_and_grad(x, n, alpha)
        return float(v)
    def obj_grad(x):
        _, g = smax_and_grad(x, n, alpha)
        return g
    def con_fun(x):
        return float(x.sum() - n / 2.0)
    def con_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iters, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()
