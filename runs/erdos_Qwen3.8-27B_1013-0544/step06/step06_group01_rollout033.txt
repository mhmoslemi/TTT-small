def subgradient_opt(h, n, deadline, rng=None):
    """Subgradient method for minimax."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.05
    c = compute_corr(h, n)
    c_max = float(np.max(c))
    
    while time.time() < deadline:
        if eta < 1e-9:
            break
        # Find active set
        eps = 0.01 * c_max + 1e-8
        active = np.where(c >= c_max - eps)[0]
        
        # Average subgradient over active set
        grad = np.zeros(n)
        for m in active:
            k = m - (n - 1)
            # grad_k[p] = (1-h[p+k])*I[0<=p+k<n] - h[p-k]*I[0<=p-k<n]
            # Vectorized:
            # Forward part: for p in 0..n-1-k (if k>=0) or 0..n-1 (if k<0)
            #   I[0<=p+k<n] means p in [-k, n-1-k] intersect [0,n-1]
            # Backward part: I[0<=p-k<n] means p in [k, n-1+k] intersect [0,n-1]
            
            p_fwd_start = max(0, -k)
            p_fwd_end = min(n-1, n-1-k)
            p_bwd_start = max(0, k)
            p_bwd_end = min(n-1, n-1+k)
            
            if p_fwd_start <= p_fwd_end:
                # grad[p] += (1 - h[p+k]) for p in [p_fwd_start, p_fwd_end]
                # h[p+k] for p in [p_fwd_start, p_fwd_end] -> indices [p_fwd_start+k, p_fwd_end+k]
                grad[p_fwd_start:p_fwd_end+1] += (1.0 - h[p_fwd_start+k:p_fwd_end+k+1])
            
            if p_bwd_start <= p_bwd_end:
                # grad[p] -= h[p-k] for p in [p_bwd_start, p_bwd_end]
                # h[p-k] for p in [p_bwd_start, p_bwd_end] -> indices [p_bwd_start-k, p_bwd_end-k]
                grad[p_bwd_start:p_bwd_end+1] -= h[p_bwd_start-k:p_bwd_end-k+1]
        
        n_active = len(active)
        grad /= n_active
        
        # Update
        h_new = h - eta * grad
        h = project_feasible(h_new, n)
        
        c = compute_corr(h, n)
        c_max = float(np.max(c))
        c5 = c_max * (2.0 / n)
        if c5 < best_c:
            best_c = c5
            best_h = h.copy()
        
        eta *= 0.9995
    
    return best_h, best_c

def slsqp_opt(h, n, alpha=100000, maxiter=2000):
    dx = 2.0 / n
    
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = c.max()
        sh = alpha * (c - m)
        mx = sh.max()
        if mx > 500:
            val = (m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha) * dx
        else:
            val = (m + np.log(np.sum(np.exp(sh))) / alpha) * dx
        return float(val)
    
    def con_fun(x):
        return x.sum() - n / 2.0
    
    def con_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': ()}]
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def compute_c5(h, n):

    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def analytical_cd(h, n, max_passes=300, deadline=None, rng=None, penalty=0.0):
    h = h.copy()
    c = compute_corr(h, n)
    c_max = float(np.max(c))
    c_len = 2 * n - 1
    sumh = float(h.sum())
    target = n / 2.0
    
    for _pass in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        order = np.arange(n)
        if rng is not None:
            rng.shuffle(order)
        
        for p in order:
            if deadline is not None and time.time() > deadline:
                break
            d_lo = -h[p]
            d_hi = 1.0 - h[p]
            if d_hi - d_lo < 1e-14:
                continue
            
            s = np.zeros(c_len)
            s[n-1-p : 2*n-1-p] += (1.0 - h[:n-p])
            s[p : n+p] -= h[n-1::-1]
            
            candidates = [d_lo, d_hi]
            mask = np.abs(s) > 1e-15
            if np.any(mask):
                cross = (c_max - c[mask]) / s[mask]
                valid = (cross >= d_lo) & (cross <= d_hi)
                if np.any(valid):
                    xv = cross[valid]
                    if len(xv) > 25:
                        idx = np.linspace(0, len(xv)-1, 25).astype(int)
                        xv = xv[idx]
                    candidates.extend(xv.tolist())
            if d_hi - d_lo > 1e-10:
                candidates.extend(np.linspace(d_lo, d_hi, 8).tolist())
            
            best_d = 0.0
            best_val = c_max + penalty * (sumh - target) ** 2
            for d in candidates:
                if abs(d) < 1e-14:
                    continue
                d = float(np.clip(d, d_lo, d_hi))
                nv = c + d * s
                nv[n-1] -= d * d
                val = float(np.max(nv))
                if penalty > 0:
                    val += penalty * (sumh + d - target) ** 2
                if val < best_val - 1e-13:
                    best_val = val
                    best_d = d
            
            if abs(best_d) > 1e-12:
                c += best_d * s
                c[n-1] -= best_d * best_d
                h[p] = np.clip(h[p] + best_d, 0.0, 1.0)
                sumh = float(h.sum())
                c_max = float(np.max(c))
                improved = True
        
        if not improved:
            break
        # End of pass: project to maintain feasibility
        h = project_feasible(h, n)
        c = compute_corr(h, n)
        c_max = float(np.max(c))

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def analytical_cd(h, n, max_passes=400, deadline=None, rng=None, penalty=0.01):
    h = h.copy()
    c = compute_corr(h, n)
    c_max = float(np.max(c))
    c_len = 2 * n - 1
    sumh = float(h.sum())
    target = n / 2.0
    for _pass in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        order = np.arange(n)
        if rng is not None:
            rng.shuffle(order)
        for p in order:
            if deadline is not None and time.time() > deadline:
                break
            d_lo = -h[p]
            d_hi = 1.0 - h[p]
            if d_hi - d_lo < 1e-14:
                continue
            s = np.zeros(c_len)
            s[n-1-p : 2*n-1-p] += (1.0 - h[:n-p])
            s[p : n+p] -= h[n-1::-1]
            candidates = [d_lo, d_hi]
            mask = np.abs(s) > 1e-15
            if np.any(mask):
                cross = (c_max - c[mask]) / s[mask]
                valid = (cross >= d_lo) & (cross <= d_hi)
                if np.any(valid):
                    xv = cross[valid]
                    if len(xv) > 20:
                        idx = np.linspace(0, len(xv)-1, 20).astype(int)
                        xv = xv[idx]
                    candidates.extend(xv.tolist())
            if d_hi - d_lo > 1e-10:
                candidates.extend(np.linspace(d_lo, d_hi, 8).tolist())
            best_d = 0.0
            best_val = c_max + penalty * (sumh - target) ** 2
            for d in candidates:
                if abs(d) < 1e-14:
                    continue
                d = float(np.clip(d, d_lo, d_hi))
                nv = c + d * s
                nv[n-1] -= d * d
                val = float(np.max(nv))
                if penalty > 0:
                    val += penalty * (sumh + d - target) ** 2
                if val < best_val - 1e-13:
                    best_val = val
                    best_d = d
            if abs(best_d) > 1e-12:
                c += best_d * s
                c[n-1] -= best_d * best_d
                h[p] = np.clip(h[p] + best_d, 0.0, 1.0)
                sumh = float(h.sum())
                c_max = float(np.max(c))
                improved = True
        if not improved:
            break
        h = project_feasible(h, n)
        c = compute_corr(h, n)
        c_max = float(np.max(c))
    return h, c_max * (2.0 / n)

def active_subgradient(h, c, n, frac=0.1):
    c_max = float(np.max(c))
    thresh = c_max - frac * c_max
    active = np.where(c >= thresh)[0]
    grad = np.zeros(n)
    for m in active:
        k = m - (n - 1)
        pfs = max(0, -k)
        pfe = min(n - 1, n - 1 - k)
        pbs = max(0, k)
        pbe = min(n - 1, n - 1 + k)
        if pfs <= pfe:
            grad[pfs:pfe+1] += (1.0 - h[pfs+k:pfe+k+1])
        if pbs <= pbe:
            grad[pbs:pbe+1] -= h[pbs-k:pbe-k+1]
    grad /= len(active)
    return grad

def guided_anneal(h, n, deadline, rng, T0=0.02, cool=0.99995):
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    T = T0
    c = compute_corr(h, n)
    n_iter = 0
    while time.time() < deadline:
        if T < 1e-12:
            break
        grad = active_subgradient(h, c, n, frac=0.1)
        eta = 2.0 * T
        noise_scale = T * 0.3
        h_new = h - eta * grad + noise_scale * rng.standard_normal(n)
        h_new = project_feasible(h_new, n)
        c_new = compute_corr(h_new, n)
        c5_new = float(np.max(c_new) * (2.0 / n))
        c5_cur = float(np.max(c) * (2.0 / n))
        delta = c5_new - c5_cur
        if delta < 0 or rng.random() < np.exp(-delta / T):
            h = h_new
            c = c_new
            if c5_new < best_c:
                best_c = c5_new
                best_h = h.copy()
        T *= cool
        n_iter += 1
        if n_iter % 5000 == 0 and T > 1e-6:
            T = max(T, (best_c - 0.37) * 0.5)
    return best_h, best_c

def slsqp_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = (m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha) * dx
    else:
        val = (m + np.log(np.sum(np.exp(sh))) / alpha) * dx
    return float(val)

def slsqp_con_fun(x, n):
    return x.sum() - n / 2.0

def slsqp_con_jac(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alpha=100000, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_con_fun, 'jac': slsqp_con_jac, 'args': (n,)}]
    try:
        res = minimize(slsqp_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)
