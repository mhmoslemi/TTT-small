    h = h.copy()

    target = n // 2

import time

def binarize(h, n):
    target = n // 2
    idx = np.argsort(-h, kind='stable')
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def swap_delta(h, i, j, n):
    """Compute delta in correlation when swapping 0 at i with 1 at j."""
    delta = np.zeros(2*n-1)
    
    # Pair A: m=i, valid s: 0<=i+s<n → k from n-1-i to 2n-2-i
    lo = n - 1 - i
    hi = 2*n - 2 - i
    k = np.arange(lo, hi + 1)
    delta[k] += 1 - h[k - n + 1 + i]
    
    # Pair B: m+s=i, valid s: 0<=i-s<n → k from i to n-1+i
    lo = i
    hi = n - 1 + i
    k = np.arange(lo, hi + 1)
    delta[k] -= h[n - 1 + i - k]
    
    # Pair C: m=j, valid s: 0<=j+s<n → k from n-1-j to 2n-2-j
    lo = n - 1 - j
    hi = 2*n - 2 - j
    k = np.arange(lo, hi + 1)
    delta[k] -= 1 - h[k - n + 1 + j]
    
    # Pair D: m+s=j, valid s: 0<=j-s<n → k from j to n-1+j
    lo = j
    hi = n - 1 + j
    k = np.arange(lo, hi + 1)
    delta[k] += h[n - 1 + j - k]
    
    # Corrections
    kd1 = n - 1 + j - i
    if 0 <= kd1 < 2*n-1:
        delta[kd1] += 1.0
    kd2 = n - 1 + i - j
    if 0 <= kd2 < 2*n-1:
        delta[kd2] += 1.0
    delta[n-1] = 0.0
    
    return delta

def local_search(h, n, deadline, max_sweeps=5000, seed=42):
    """Best-improvement local search for binary h."""
    h = h.copy()
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_max = float(corr.max())
    best_max = cur_max
    best_h = h.copy()
    
    sweep = 0
    while sweep < max_sweeps and time.time() < deadline:
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        for i in zeros:
            best_j = -1
            best_val = cur_max
            for j in ones:
                d = swap_delta(h, int(i), int(j), n)
                new_max = float((corr + d).max())
                if new_max < best_val:
                    best_val = new_max
                    best_j = int(j)
                    best_d = d
            if best_j >= 0:
                h[i] = 1.0
                h[best_j] = 0.0
                corr = corr + best_d
                cur_max = best_val
                improved = True
                if cur_max < best_max:
                    best_max = cur_max
                    best_h = h.copy()
        
        sweep += 1
        if not improved:
            break
    
    return best_h, best_max * dx
