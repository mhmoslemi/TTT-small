def binary_ls(h, n, deadline, max_sweeps=10000, seed=42, samples_per_zero=50):

        if time.time() > deadline:
            break

        rng.shuffle(zeros)

            # Try up to samples_per_zero random ones

            trial_idx = rng.choice(len(ones), size=n_try, replace=False)
            best_nc = cur_c
            best_j = -1
            for ti in trial_idx:
                j = int(ones[ti])
                d = swap_delta(h, i, j, n)
                nc = float((corr + d).max()) * dx
                if nc < best_nc:
                    best_nc = nc
                    best_j = j

                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                improved = True
        if not improved:
            break
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n // 2
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def swap_delta(h, i, j, n):
    delta = np.zeros(2 * n - 1)
    lo = max(0, i)
    hi = min(2 * n - 2, i + n - 1)
    if lo <= hi:
        m = np.arange(lo, hi + 1)
        delta[m] += 1.0 - h[i + n - 1 - m]
    lo = max(0, n - 1 - i)
    hi = min(2 * n - 2, 2 * n - 2 - i)
    if lo <= hi:
        m = np.arange(lo, hi + 1)
        delta[m] -= h[m - n + 1 + i]
    lo = max(0, j)
    hi = min(2 * n - 2, j + n - 1)
    if lo <= hi:
        m = np.arange(lo, hi + 1)
        delta[m] -= 1.0 - h[j + n - 1 - m]
    lo = max(0, n - 1 - j)
    hi = min(2 * n - 2, 2 * n - 2 - j)
    if lo <= hi:
        m = np.arange(lo, hi + 1)
        delta[m] += h[m - n + 1 + j]
    return delta

def lse_grad(x, alpha, dx, n):
    corr = np.correlate(x, 1 - x, mode='full')
    scaled = alpha * corr * dx
    mx = scaled.max()
    exp_s = np.exp(scaled - mx)
    w = exp_s / exp_s.sum()
    obj = (mx + np.log(exp_s.sum())) / alpha
    W_arr = w[::-1]
    A = np.correlate(1 - x, W_arr, 'full')[n - 1:n - 1 + n]
    B = np.convolve(x, W_arr, 'full')[n - 1:n - 1 + n]
    grad = A - B
    return obj, grad, corr

def smooth_opt(h_init, n, alpha, maxiter=2000):
    dx = 2.0 / n
    n = len(h_init)
    def fun(x):
        return lse_grad(x, alpha, dx, n)[0]
    def jac(x):
        g = lse_grad(x, alpha, dx, n)[1]
        return g
    def cons_fun(x):
        return x.sum() - n / 2.0
    def cons_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(fun, h_init, jac=jac, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h_init

def binary_ls(h, n, deadline, max_sweeps=10000, seed=42, ntry=40):
    h = binarize(h, n)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_c = float(corr.max()) * dx
    best_c = cur_c
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        rng.shuffle(ones)
        rng.shuffle(zeros)
        improved = False
        for i in zeros:
            if time.time() > deadline:
                break
            i = int(i)
            n_t = min(len(ones), ntry)
            trial = rng.choice(len(ones), size=n_t, replace=False)
            best_nc = cur_c
            best_j = -1
            best_d = None
            for ti in trial:
                j = int(ones[ti])
                d = swap_delta(h, i, j, n)
                nc = float((corr + d).max()) * dx
                if nc < best_nc:
                    best_nc = nc
                    best_j = j
                    best_d = d
            if best_j >= 0 and best_nc < cur_c - 1e-15:
                h[i] = 1.0
                h[best_j] = 0.0
                corr = corr + best_d
                cur_c = best_nc
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
                improved = True
        if not improved:
            break
    return best_h, best_c
