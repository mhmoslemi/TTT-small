def smooth_grad(h,n,beta,pen):

    c=np.correlate(h,1.0-h,mode='full')
    cm=float(np.max(c))
    x=beta*dx*(c-cm)
    x=np.minimum(x,0.0)
    ex=np.exp(x)
    s=ex.sum()
    if s <= 0:

    else:

def project_feasible(h,n):
    h=np.asarray(h,dtype=float).copy()
    target=0.5*n
    h=np.clip(h,0.0,1.0)
    for _ in range(100):
        diff=target-h.sum()
        if abs(diff)<1e-14: break
        if diff>0:
            room=1.0-h
            s=room.sum()
            if s < 1e-15: break
            h += (diff/s)*room
        else:
            s=h.sum()
            if s < 1e-15: break
            h += (diff/s)*h
    h=np.clip(h,0.0,1.0)
    diff=target-h.sum()
    for _ in range(50):
        if abs(diff)<1e-15: break
        if diff>0:
            idx=int(np.argmin(h))
            add = min(diff, 1.0-h[idx])
            if add <= 1e-16: break
        else:
            idx=int(np.argmax(h))
            add = max(diff, h[idx]-1.0)
            if add >= -1e-16: break
        h[idx]+=add
        diff=target-h.sum()
    return h

import numpy as np
import time
from scipy.optimize import minimize

_G_N = 0

def make_bounds(n):
    n = int(n)
    return [(0.0, 1.0)] * n

def c5_of(h, n):
    dx = 2.0 / float(n)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    n = int(n)
    target = 0.5 * float(n)
    h = np.clip(h, 0.0, 1.0)

    for _ in range(100):
        diff = target - float(h.sum())
        if abs(diff) < 1e-14:
            break
        if diff > 0.0:
            room = 1.0 - h
            s = float(room.sum())
            if s < 1e-15:
                break
            h += (diff / s) * room
        else:
            s = float(h.sum())
            if s < 1e-15:
                break
            h += (diff / s) * h

    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - float(h.sum())
        if abs(diff) < 1e-15:
            break
        if diff > 0.0:
            idx = int(np.argmin(h))
            add = min(diff, 1.0 - float(h[idx]))
        else:
            idx = int(np.argmax(h))
            add = max(diff, -float(h[idx]))
        if abs(add) < 1e-16:
            break
        h[idx] += add
    return h

def to_binary(h, n):
    n = int(n)
    h = np.asarray(h, dtype=float)
    hb = np.zeros(n, dtype=float)
    target = 0.5 * float(n)
    order = np.argsort(h)[::-1]
    k = int(target)
    if k > 0 and k <= n:
        hb[order[:k]] = 1.0
    rem = target - float(hb.sum())
    if k < n and abs(rem) > 1e-14:
        hb[order[k]] = float(np.clip(rem, 0.0, 1.0))
    return project_feasible(hb, n)

def resample_h(h, n_new):
    n = int(np.size(h))
    n_new = int(n_new)
    if n <= 1 or n_new <= 1 or n_new == n:
        return project_feasible(h, n_new)
    xo = np.linspace(0.0, 2.0, n)
    xn = np.linspace(0.0, 2.0, n_new)
    y = np.interp(xn, xo, np.asarray(h, dtype=float))
    return project_feasible(y, n_new)

def shifted_score(h, n, deadline=None):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = c5_of(h, n)
    n = int(n)
    if n < 2:
        return best_h, best_c

    step = 1
    if n > 450:
        step = 2
    if n > 700:
        step = 3

    best_s = 0
    for s in range(1, n, step):
        if deadline is not None and time.time() > deadline:
            break
        hs = np.roll(h, s)
        cc = c5_of(hs, n)
        if cc < best_c:
            best_c = cc
            best_h = hs.copy()
            best_s = s

    if step > 1:
        lo = max(0, best_s - 2 * step)
        hi = min(n - 1, best_s + 2 * step)
        for s in range(lo, hi + 1):
            if deadline is not None and time.time() > deadline:
                break
            hs = np.roll(h, s)
            cc = c5_of(hs, n)
            if cc < best_c:
                best_c = cc
                best_h = hs.copy()

    return best_h, best_c

def smooth_core(h, n, beta):
    h = np.clip(h, 0.0, 1.0)
    n = int(n)
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode='full')
    cm = float(np.max(c))
    val = cm * dx
    if beta > 1e-9:
        z = beta * dx * (c - cm)
        zm = float(np.max(z))
        z = z - zm
        s = float(np.exp(z).sum())
        if s > 0.0:
            val += (zm + float(np.log(s))) / float(beta)
    return val

def full_smooth_grad(h, n, beta):
    h = np.clip(h, 0.0, 1.0)
    n = int(n)
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, mode='full')
    cm = float(np.max(c))

    if beta > 1e-9:
        z = beta * dx * (c - cm)
        zm = float(np.max(z))
        z = z - zm
        ex = np.exp(z)
        s = float(ex.sum())
        if s <= 1e-300:
            p = np.zeros(2 * n - 1, dtype=float)
            p[int(np.argmax(c))] = 1.0
        else:
            p = ex / s
    else:
        p = np.ones(2 * n - 1, dtype=float) / float(2 * n - 1)

    pref = np.empty(2 * n, dtype=float)
    pref[0] = 0.0
    pref[1:] = np.cumsum(p)
    L = pref[n:n + n] - pref[0:n]

    r = p + p[::-1]
    conv = np.convolve(h, r, mode='full')
    cross = conv[n - 1:2 * n - 1]
    return dx * (L - cross)

def smooth_value(x, n, beta, pen):
    val = smooth_core(x, n, beta)
    if pen:
        x = np.clip(x, 0.0, 1.0)
        d = float(x.sum() - 0.5 * float(n))
        val += float(pen) * d * d
    return val

def smooth_grad(x, n, beta, pen):
    h = np.clip(x, 0.0, 1.0)
    g = full_smooth_grad(h, n, beta)
    if pen:
        d = float(h.sum() - 0.5 * float(n))
        g = g + 2.0 * float(pen) * d
    return g

def constraint_sum(x):
    return float(np.sum(x) - 0.5 * float(_G_N))

def constraint_jac(x):
    return np.ones_like(x, dtype=float)

def run_lbfgs(x, n, beta, pen, maxiter):
    x = project_feasible(x, n)
    n = int(n)
    try:
        res = minimize(
            smooth_value, x,
            method='L-BFGS-B',
            jac=smooth_grad,
            bounds=make_bounds(n),
            args=(n, float(beta), float(pen)),
            options={
                'maxiter': int(maxiters if False else maxiter),
                'ftol': 1e-14,
                'gtol': 1e-9,
                'maxls': 20,
                'disp': False
            }
        )
        return project_feasible(res.x, n)
    except Exception:
        return x

def run_slsqp(x, n, beta, maxiter):
    global _G_N
    _G_N = int(n)
    x = project_feasible(x, n)
    n = int(n)
    try:
        cons = [{'type': 'eq', 'fun': constraint_sum, 'jac': constraint_jac}]
        res = minimize(
            smooth_value, x,
            method='SLSQP',
            bounds=make_bounds(n),
            args=(n, float(beta), 0.0),
            jac=smooth_grad,
            constraints=cons,
            options={'maxiter': int(maxiter), 'ftol': 1e-14, 'disp': False}
        )
        return project_feasible(res.x, n)
    except Exception:
        return x

def active_sum_fun(y, target_free):
    return float(np.sum(y) - float(target_free))

def active_sum_jac(y, target_free):
    return np.ones(np.size(y), dtype=float)

def active_value(y, n, beta, pen, idx, fixed):
    h = fixed.copy()
    if np.size(idx):
        h[idx] = y
    val = smooth_core(h, n, beta)
    if pen:
        d = float(h.sum() - 0.5 * float(n))
        val += float(pen) * d * d
    return val

def active_grad(y, n, beta, pen, idx, fixed):
    h = fixed.copy()
    if np.size(idx):
        h[idx] = y
    g = full_smooth_grad(h, n, beta)
    if pen:
        d = float(h.sum() - 0.5 * float(n))
        g = g + 2.0 * float(pen) * d
    return g[idx]

def run_slsqp_active(h, n, beta, maxiter, free_tol, max_vars):
    h = project_feasible(h, n)
    n = int(n)
    mask = (h > float(free_tol)) & (h < 1.0 - float(free_tol))
    idx_all = np.where(mask)[0]
    m = int(idx_all.size)
    if m == 0:
        return h

    if m > int(max_vars):
        dist = np.minimum(h[idx_all], 1.0 - h[idx_all])
        order = np.argsort(dist)[::-1][:int(max_vars)]
        idx = idx_all[order]
    else:
        idx = idx_all

    fixed = h.copy()
    fixed[idx] = 0.0
    target_free = 0.5 * float(n) - float(fixed.sum())
    target_free = float(np.clip(target_free, 0.0, float(idx.size)))

    y0 = np.clip(h[idx], 0.0, 1.0)
    for _ in range(30):
        y0 = np.clip(y0, 0.0, 1.0)
        diff = target_free - float(y0.sum())
        if abs(diff) < 1e-14:
            break
        if diff > 0.0:
            room = 1.0 - y0
            s = float(room.sum())
            if s < 1e-15:
                break
            y0 += (diff / s) * room
        else:
            s = float(y0.sum())
            if s < 1e-15:
                break
            y0 += (diff / s) * y0

    try:
        cons = [{'type': 'eq', 'fun': active_sum_fun, 'jac': active_sum_jac,
                 'args': (target_free,)}]
        res = minimize(
            active_value, y0,
            method='SLSQP',
            bounds=make_bounds(idx.size),
            args=(n, float(beta), 0.0, idx, fixed),
            jac=active_grad,
            constraints=cons,
            options={'maxiter': int(maxiter), 'ftol': 1e-14, 'disp': False}
        )
        h2 = fixed.copy()
        h2[idx] = res.x
        return project_feasible(h2, n)
    except Exception:
        return h

def optimize_chain(h, n, deadline, use_slsq, liter, beta_list, active):
    n = int(n)
    h = project_feasible(h, n)
    h, c = shifted_score(h, n, deadline)

    pen = 200.0
    for beta in beta_list:
        if time.time() > deadline - 8.0:
            break
        h2 = run_lbfgs(h, n, beta, pen, liter)
        c2 = c5_of(h2, n)
        if c2 < c:
            h, c = h2, c2

    if use_slsq and time.time() < deadline - 60.0:
        sl_betas = (500.0, 5000.0, 50000.0, 200000.0)
        for b in sl_betas:
            if time.time() > deadline - 40.0:
                break
            iters = 12 if b < 20000.0 else 8
            h2 = run_slsqp(h, n, b, iters)
            c2 = c5_of(h2, n)
            if c2 < c:
                h, c = h2, c2

    if active and use_slsq and time.time() < deadline - 25.0:
        h2 = run_slsqp_active(h, n, 30000.0, 40, 1e-3, 300)
        c2 = c5_of(h2, n)
        if c2 < c:
            h, c = h2, c2

    if active and use_slsq and time.time() < deadline - 25.0:
        h2 = run_slsqp_active(h, n, 5000.0, 40, 1e-5, 300)
        c2 = c5_of(h2, n)
        if c2 < c:
            h, c = h2, c2

    if time.time() < deadline - 3.0:
        h2, c2 = shifted_score(h, n, deadline)
        if c2 < c:
            h, c = h2, c2

    h = project_feasible(h, n)
    c = c5_of(h, n)
    return h, c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = max(30.0, float(budget_s) - 10.0)
    deadline = t0 + total

    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = int(len(h0))
    except Exception:
        n0 = 400
        h0 = np.full(n0, 0.5, dtype=float)

    h0 = project_feasible(h0, n0)
    best_h, best_c = shifted_score(h0, n0, deadline)
    best_n = n0

    main_end = t0 + 0.58 * total
    main_betas = (5.0, 10.0, 20.0, 50.0, 100.0, 300.0, 1000.0,
                  3000.0, 10000.0, 30000.0, 100000.0, 300000.0)
    h1, c1 = optimize_chain(best_h, best_n, main_end, True, 320, main_betas, True)
    if c1 < best_c:
        best_h, best_c, best_n = h1, c1, best_n

    alt_end = t0 + 0.84 * total

    if best_n % 2 == 0 and time.time() < alt_end - 45.0:
        hb = to_binary(best_h, best_n)
        h1, c1 = optimize_chain(
            hb, best_n, min(alt_end - 5.0, time.time() + 90.0),
            False, 160, (20.0, 200.0, 2000.0, 20000.0, 100000.0), False
        )
        if c1 < best_c:
            best_h, best_c = h1, c1

    if time.time() < alt_end - 45.0:
        hs = 0.5 * (best_h + 1.0 - best_h[::-1])
        hs = project_feasible(hs, best_n)
        h1, c1 = optimize_chain(
            hs, best_n, min(alt_end - 5.0, time.time() + 80.0),
            False, 140, (50.0, 500.0, 5000.0, 50000.0), False
        )
        if c1 < best_c:
            best_h, best_c = h1, c1

    if time.time() < alt_end - 45.0:
        hn = project_feasible(best_h + 0.02 * rng.randn(best_n), best_n)
        h1, c1 = optimize_chain(
            hn, best_n, min(alt_end - 5.0, time.time() + 80.0),
            False, 130, (50.0, 500.0, 5000.0, 50000.0), False
        )
        if c1 < best_c:
            best_h, best_c = h1, c1

    alt_ns = []
    for f in (0.60, 0.75, 1.25, 1.50, 2.00):
        nn = int(round(float(n0) * f / 2.0) * 2)
        if 128 <= nn <= 960 and abs(nn - best_n) >= 24 and nn not in alt_ns:
            alt_ns.append(nn)

    tried = 0
    for nn in alt_ns:
        if tried >= 4 or time.time() > alt_end - 35.0:
            break
        nn = int(nn)
        if nn % 2 != 0 or nn <= 0:
            continue
        hr = resample_h(best_h, nn)
        sub = min(alt_end - 10.0, time.time() + 80.0)
        use_s = nn <= 520
        h1, c1 = optimize_chain(
            hr, nn, sub, use_s, 140,
            (20.0, 200.0, 2000.0, 20000.0, 100000.0), False
        )
        if c1 < best_c:
            best_h, best_c, best_n = h1, c1, nn
        tried += 1

    final_betas = (1000.0, 5000.0, 20000.0, 100000.0, 500000.0, 2000000.0)
    use_final_slsq = best_n <= 560
    h1, c1 = optimize_chain(best_h, best_n, deadline, use_final_slsq, 220, final_betas, True)
    if c1 < best_c:
        best_h, best_c, best_n = h1, c1, best_n

    best_h = project_feasible(best_h, best_n)
    best_c = c5_of(best_h, best_n)
    return best_h, float(best_c), int(best_n)
