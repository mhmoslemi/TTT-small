    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        q = np.clip(y - mid, 0.0, 1.0)
        if q.sum() > target:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    q = np.clip(y - tau, 0.0, 1.0)
    diff = target - q.sum()
    if abs(diff) > 1e-14:
        free = (q > 1e-12) & (q < 1.0 - 1e-12)
        if free.any():
            q[free] += diff / free.sum()
        elif diff > 0:
            idx = int(np.argmax(1.0 - q))
            q[idx] = min(1.0, q[idx] + diff)
        else:
            idx = int(np.argmax(q))
            q[idx] = max(0.0, q[idx] + diff)
    return np.clip(q, 0.0, 1.0)


def ensure_feasible(h, n):
    h = np.array(h, dtype=float, copy=True)
    h[~np.isfinite(h)] = 0.5
    target = 0.5 * n
    if (np.min(h) < -1e-10 or np.max(h) > 1.0 + 1e-10 or
            abs(float(h.sum()) - target) > 1e-8):
        return project_feasible(h, n)
    return np.clip(h, 0.0, 1.0)


def shift_values(h, n):
    N = 2 * n - 1
    L = next_fast_len(N)
    dx = 2.0 / float(n)
    h = np.ascontiguousarray(h)
    fh = np.fft.rfft(h, n=L)
    fb = np.fft.rfft((1.0 - h)[::-1], n=L)
    conv = np.fft.irfft(fh * fb, n=L)[:N]
    return conv[::-1] * dx


def weighted_grad(h, n, w):
    dx = 2.0 / float(n)
    full = 3 * n - 2
    L = next_fast_len(full)
    h = np.ascontiguousarray(h)
    a = 1.0 - h
    fw = np.fft.rfft(w, n=L)
    fw_rev = np.fft.rfft(w[::-1], n=L)
    fa = np.fft.rfft(a, n=L)
    fh = np.fft.rfft(h, n=L)
    s1 = np.fft.irfft(fa * fw_rev, n=L)
    s2 = np.fft.irfft(fh * fw, n=L)
    return dx * (s1[n - 1:2 * n - 1] - s2[n - 1:2 * n - 1])


def lse_objective(x, alpha, n):
    g = shift_values(x, n)
    m = float(np.max(g))
    if alpha <= 1.0:
        alpha = 1.0
    shifted = alpha * (g - m)
    e = np.exp(np.clip(shifted, -745.0, 0.0))
    s = float(np.sum(e))
    if s <= 0.0 or not math.isfinite(s):
        e = np.zeros_like(g)
        e[int(np.argmax(g))] = 1.0
        s = 1.0
    w = e / s
    F = m + math.log(s) / alpha
    return F, w, m


def fun_pen(x, alpha, n, rho):
    F, _, _ = lse_objective(x, alpha, n)
    if rho:
        s = float(x.sum()) - 0.5 * n
        F += 0.5 * rho * s * s
    return F


def jac_pen(x, alpha, n, rho):
    _, w, _ = lse_objective(x, alpha, n)
    g = weighted_grad(x, n, w)
    if rho:
        g = g + rho * (float(x.sum()) - 0.5 * n)
    return g


def c5_direct(h, n):
    h = np.asarray(h, dtype=float)
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / float(n)))


def refine_h(h, n0, n1):
    if n1 <= 0:
        return np.zeros(0)
    if n0 == n1:
        return ensure_feasible(h, n1)
    if n0 <= 1:
        val = 0.5
        if n0 == 1 and np.isfinite(h[0]):
            val = float(np.clip(h[0], 0.0, 1.0))
        return project_feasible(np.full(n1, val, dtype=float), n1)
    h = np.asarray(h, dtype=float)
    u = np.arange(n0, dtype=float)
    v = np.linspace(0.0, float(n0 - 1), n1, dtype=float)
    return project_feasible(np.interp(v, u, h), n1)


def default_initial(n):
    if n < 2:
        n = 400
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = 0.5 + 0.45 * (x - 1.0) + 0.08 * np.sin(2.0 * np.pi * x)
    return project_feasible(h, n)


def chunk_maxfun(n, alpha):
    if n <= 450:
        mf = 750
    elif n <= 700:
        mf = 480
    else:
        mf = 320
    if alpha > 50000:
        mf = int(0.7 * mf)
    if alpha > 300000:
        mf = int(0.5 * mf)
    return max(120, int(mf))


def optimize_lbfgs(h, n, alpha, rho, maxfun, maxiter=None):
    try:
        if maxiter is None or maxiter <= 0:
            maxiter = max(100, int(maxfun))
        start = np.array(h, dtype=float, copy=True)
        bounds = [(0.0, 1.0)] * n
        options = {
            "maxiter": int(maxiter),
            "maxfun": int(maxfun),
            "ftol": 1e-15,
            "gtol": 1e-10,
            "maxls": 24
        }
        res = minimize(
            fun_pen, start, method="L-BFGS-B", jac=jac_pen,
            args=(alpha, n, rho), bounds=bounds, options=options
        )
        x = np.asarray(res.x, dtype=float)
        if np.all(np.isfinite(x)):
            return project_feasible(x, n)
    except Exception:
        pass
    return np.array(h, dtype=float, copy=True)


def run(seed=42, budget_s=1100, **kwargs):

    total = max(30.0, float(budget_s) - 15.0)
    dl = t0 + total

    try:
        h0 = np.asarray(initial_h_values, dtype=float).ravel().copy()
        n0 = int(len(h0))

    if n0 < 50:
        h0 = default_initial(400)
        n0 = 400

    best_h = h.copy()

    best_c = c5_direct(h, n0)

    frac_reserve = ((0.20, 250.0), (0.55, 150.0), (0.85, 60.0))
    limits = []
    for frac, res in frac_reserve:
        end = min(t0 + total * frac, dl - res)
        if end < t0:
            end = t0
        limits.append(float(end))
    if limits[1] < limits[0]:
        limits[1] = limits[0]
    if limits[2] < limits[1]:
        limits[2] = limits[1]

    def accept_candidate(hc, cn):

            best_h = np.array(hc, dtype=float, copy=True)

            best_c = float(cn)

    # Phase 1: polish the inherited resolution.
    if best_n <= 1024 and time.time() < limits[0] - 15.0:
        n_base = best_n
        work = best_h.copy()
        work_c = best_c
        alphas = (100.0, 1000.0, 10000.0, 100000.0, 500000.0)
        for alpha in alphas:
            chunks = 2 if alpha <= 10000.0 else 1
            for _ in range(chunks):
                if time.time() >= limits[0] - 15.0:
                    break
                mf = chunk_maxfun(n_base, alpha)
                ho = optimize_lbfgs(work, n_base, alpha, 20.0, mf, mf)
                co = c5_direct(ho, n_base)
                if not math.isfinite(co):
                    continue
                if co < best_c:
                    best_h = ho.copy()
                    best_n = n_base
                    best_c = co
                if co < work_c + 1e-5:
                    work = ho.copy()
                    work_c = co
                else:
                    if best_n == n_base:
                        work = best_h.copy()
                        work_c = best_c
                    else:
                        work = refine_h(best_h, best_n, n_base)
                        work_c = c5_direct(work, n_base)

    # Phase 2: first higher resolution available.
    ann = -1
    if best_n < 512:
        ann = 512
    elif best_n < 768:
        ann = 768
    if ann > 0 and time.time() < limits[1] - 15.0:
        work = refine_h(best_h, best_n, ann)
        work_c = c5_direct(work, ann)
        if work_c < best_c:
            best_h = work.copy()
            best_n = ann
            best_c = work_c
        alphas = (100.0, 1000.0, 10000.0, 100000.0, 1000000.0) if ann <= 600 else (
            1000.0, 10000.0, 100000.0, 1000000.0
        )
        for alpha in alphas:
            chunks = 2 if alpha <= 10000.0 else 1
            for _ in range(chunks):
                if time.time() >= limits[1] - 15.0:
                    break
                mf = chunk_maxfun(ann, alpha)
                ho = optimize_lbfgs(work, ann, alpha, 30.0, mf, mf)
                co = c5_direct(ho, ann)
                if not math.isfinite(co):
                    continue
                if co < best_c:
                    best_h = ho.copy()
                    best_n = ann
                    best_c = co
                if co < work_c + 1e-5:
                    work = ho.copy()
                    work_c = co
                else:
                    if best_n == ann:
                        work = best_h.copy()
                        work_c = best_c
                    else:
                        work = refine_h(best_h, best_n, ann)
                        work_c = c5_direct(work, ann)

    # Phase 3: 768 resolution if still useful.
    if best_n < 768 and time.time() < limits[2] - 15.0:
        ann = 768
        work = refine_h(best_h, best_n, ann)
        work_c = c5_direct(work, ann)
        if work_c < best_c:
            best_h = work.copy()
            best_n = ann
            best_c = work_c
        alphas = (1000.0, 10000.0, 100000.0, 1000000.0)
        for alpha in alphas:
            chunks = 2 if alpha <= 10000.0 else 1
            for _ in range(chunks):
                if time.time() >= limits[2] - 15.0:
                    break
                mf = chunk_maxfun(ann, alpha)
                ho = optimize_lbfgs(work, ann, alpha, 30.0, mf, mf)
                co = c5_direct(ho, ann)
                if not math.isfinite(co):
                    continue
                if co < best_c:
                    best_h = ho.copy()
                    best_n = ann
                    best_c = co
                if co < work_c + 1e-5:
                    work = ho.copy()
                    work_c = co
                else:
                    if best_n == ann:
                        work = best_h.copy()
                        work_c = best_c
                    else:
                        work = refine_h(best_h, best_n, ann)
                        work_c = c5_direct(work, ann)

    # Noise restart, only if it looks promising after one high polish.
    if time.time() < dl - 160.0 and best_n <= 1024:
        for amp in (0.01, 0.025):
            if time.time() >= dl - 160.0:
                break
            hn = project_feasible(best_h + amp * rng.standard_normal(best_n), best_n)
            cn = c5_direct(hn, best_n)
            if not math.isfinite(cn) or cn > best_c + 4e-4:
                continue
            work = hn.copy()
            work_c = cn
            if work_c < best_c:
                best_h = work.copy()
                best_n = bn
                best_c = work_c
            for alpha in (10000.0, 100000.0, 1000000.0):
                if time.time() >= dl - 130.0:
                    break
                mf = chunk_maxfun(best_n, alpha)
                ho = optimize_lbfgs(work, best_n, alpha, 50.0, mf, mf)
                co = c5_direct(ho, best_n)
                if not math.isfinite(co):
                    continue
                if co < best_c:
                    best_h = ho.copy()
                    best_n = best_n
                    best_c = co
                if co < work_c + 1e-5:
                    work = ho.copy()
                    work_c = co
                else:
                    work = best_h.copy()
                    work_c = best_c

    # Final cheap refinements.
    for target_n in (768, 998):
        if best_n < target_n and time.time() < dl - 8.0:
            hn = refine_h(best_h, best_n, target_n)
            cn = c5_direct(hn, target_n)
            if math.isfinite(cn) and cn < best_c:
                best_h = hn.copy()
                best_n = target_n

    # Short high-alpha polish at the final resolution.
    if time.time() < dl - 80.0:

            if time.time() >= dl - 60.0:

            co = c5_direct(ho, best_n)
            if math.isfinite(co) and co < best_c:

                best_n = best_n
                best_c = co

    best_h = ensure_feasible(best_h, best_n)
    best_c = c5_direct(best_h, best_n)
    return best_h, best_c, best_n
