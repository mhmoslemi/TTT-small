```python
def vectorized_greedy_alignment(h, n, deadline, max_rounds=200):

    dx = 2.0/n
    corr = np.correlate(cur, 1.0 - cur, mode='full')
    cur_max = corr.max()
    
    s_vals = np.arange(-(n-1), n)  # shape (2n-1,)
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        
        ones = np.where(cur == 1)[0]
        zeros = np.where(cur == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Build Q matrix: Q[j_idx, k] = cur[j + s_k] + cur[j - s_k]
        # j is one of the ones, k is 0..2n-2
        # s_k = s_vals[k]
        # j + s_k valid: 0 <= j + s_k < n
        # j - s_k valid: 0 <= j - s_k < n
        
        # Q shape: (n_ones, 2n-1)
        J = ones[:, None]  # (n_ones, 1)
        S = s_vals[None, :]  # (1, 2n-1)
        
        Js = J + S  # (n_ones, 2n-1)
        Jm = J - S
        
        valid_s = (Js >= 0) & (Js < n)
        valid_m = (Jm >= 0) & (Jm < n)
        
        # For valid indices, get cur value; for invalid, 0
        Js_clipped = np.clip(Js, 0, n-1)
        Jm_clipped = np.clip(Jm, 0, n-1)
        
        Q = cur[Js_clipped] + cur[Jm_clipped]
        Q = Q * valid_s + 0 * (~valid_s)  # Hmm, this doesn't work right
        # Actually: Q[j,k] = (cur[j+s] if valid) + (cur[j-s] if valid)
        Q = np.where(valid_s, cur[Js_clipped], 0.0) + np.where(valid_m, cur[Jm_clipped], 0.0)
        
        best_score = np.inf
        best_i = -1
        best_j = -1
        
        for i in zeros:
            P_i = np.where((i + s_vals >= 0) & (i + s_vals < n), cur[i + np.clip(i+s_vals, 0, n-1)], 0.0) + \
                  np.where((i - s_vals >= 0) & (i - s_vals < n), cur[i - np.clip(i-s_vals, 0, n-1)], 0.0)
            # P_i[k] = cur[i+s_k] (if valid) + cur[i-s_k] (if valid)
            
            R_i = corr - P_i  # (2n-1,)
            
            # Score for each j: max_k (R_i[k] + Q[j,k])
            scores = np.max(R_i[None, :] + Q, axis=1)  # (n_ones,)
            
            min_score = scores.min()
            if min_score < best_score:

                best_i = i
                best_j = ones[scores.argmin()]
        
        if best_score >= cur_max:
            break  # No improving move

        cur[best_i] = 1.0
        cur[best_j] = 0.0
        corr = np.correlate(cur, 1.0 - cur, mode='full')
        cur_max = corr.max()
    
    return cur, cur_max * dx
```

So ∂corr_k/∂h[i] = dx * [(1-h[i+s_k]) - h[i-s_k]] (with boundary conditions)

This is clean! And for the smooth max, the gradient is a weighted sum of the corr_k gradients, weighted by the softmax weights.

```python
def c5_objective_and_grad(h, n, alpha=1000):
    dx = 2.0/n
    corr = np.correlate(h, 1.0-h, mode='full') * dx
    m = corr.max()
    # Smooth max
    shifted = alpha * (corr - m)
    sm = shifted.max()
    log_sum = sm + np.log(np.exp(shifted - sm).sum())
    obj = m + log_sum / alpha
    
    # Gradient
    # Weights for each k
    weights = np.exp(shifted - sm) / np.exp(shifted - sm).sum()
    
    # grad[k, i] = dx * [(1-h[i+s_k]) - h[i-s_k]] (with bounds)
    # Total grad[i] = sum_k weights[k] * grad[k, i]
    
    s_vals = np.arange(-(n-1), n)
    # For each k (shift s_k):
    #   grad_contrib[i] += weights[k] * dx * [(1-h[i+s_k]) - h[i-s_k]]
    # 
    # Let's vectorize:
    # grad[i] = dx * sum_k weights[k] * (1-h[i+s_k]) - dx * sum_k weights[k] * h[i-s_k]
    #         = dx * sum_k weights[k] - dx * sum_k weights[k] * h[i+s_k]
    #           - dx * sum_k weights[k] * h[i-s_k]
    #         = dx - dx * sum_k weights[k] * h[i+s_k] - dx * sum_k weights[k] * h[i-s_k]
    
    # sum_k weights[k] * h[i+s_k]: for each i, sum over k where i+s_k is valid
    # This is a weighted sum of h at shifted positions.
    # Equivalent to: (weights * h) convolved... actually it's a weighted cross-correlation.
    
    # Let's compute it directly:
    # A[i] = sum_k weights[k] * h[i+s_k] (valid i+s_k)
    # B[i] = sum_k weights[k] * h[i-s_k] (valid i-s_k)
    
    # For A: A[i] = sum_k w[k] * h[i+s_k]
    # i+s_k in [0,n): s_k in [-i, n-1-i]
    # Let's use the s_vals array:
    # A[i] = sum_{k: 0<=i+s_k<n} w[k] * h[i+s_k]
    
    # This is like a weighted sum with position-dependent weights.
    # For efficiency, note that:
    # A = sum_k w[k] * shift(h, s_k)  where shift(h, s)[i] = h[i+s] if valid, 0 otherwise
    
    # We can compute this as: for each k, add w[k] * shifted_h to A.
    # But that's (2n-1) iterations. For n=250, that's 499 iterations, each O(n).
    # Total: O(n²) = 62500. Fine.
    
    A = np.zeros(n)  # A[i] = sum_k w[k] * h[i+s_k]
    B = np.zeros(n)  # B[i] = sum_k w[k] * h[i-s_k]
    
    for k in range(2*n-1):
        s = s_vals[k]
        w = weights[k]
        if w < 1e-15:
            continue
        if s >= 0:
            # i+s < n: i < n-s
```

```python
        Jm = J - S
        vs = (Js >= 0) & (Js < n)
        vm = (Jm >= 0) & (Jm < n)
        Js_c = np.clip(Js, 0, n - 1)
        Jm_c = np.clip(Jm, 0, n - 1)
        Q = np.where(vs, cur[Js_c], 0.0) + np.where(vm, cur[Jm_c], 0.0)
        best_score = np.inf
        best_i = -1
        best_j = -1
        for i in zeros:
            ivs = (i + s_vals >= 0) & (i + s_vals < n)
            ivm = (i - s_vals >= 0) & (i - s_vals < n)
            Pi = np.where(ivs, cur[np.clip(i + s_vals, 0, n - 1)], 0.0) + \
                 np.where(ivm, cur[np.clip(i - s_vals, 0, n - 1)], 0.0)
            Ri = corr - Pi
            scores = np.max(Ri[None, :] + Q, axis=1)
            ms = scores.min()
            if ms < best_score:
                best_score = ms
                best_j = ones[scores.argmin()]
                best_i = i
        if best_score >= cur_max - 1e-15:
            break
        cur[best_i] = 1.0
        cur[best_j] = 0.0
        corr = np.correlate(cur, 1.0 - cur, mode='full')
        cur_max = corr.max()
    return cur, float(cur_max * dx)

def sa_binary(h, n, deadline, iters=300000, t_init=0.008, t_final=0.0001, seed=42):
    rng = np.random.RandomState(seed)
    cur = h.astype(float).copy()
    dx = 2.0 / n
    corr = np.correlate(cur, 1.0 - cur, mode='full')
    cur_val = corr.max()
    best = cur.copy()
    best_val = cur_val
    s_vals = np.arange(-(n - 1), n)
    cooling = (t_final / t_init) ** (1.0 / iters)
    t = t_init
    for it in range(iters):
        if time.time() > deadline:
            break
        i = rng.randint(0, n)
        j = rng.randint(0, n)
        if cur[i] != 0 or cur[j] != 1:
            continue
        ivs = (i + s_vals >= 0) & (i + s_vals < n)
        ivm = (i - s_vals >= 0) & (i - s_vals < n)
        Pi = np.where(ivs, cur[np.clip(i + s_vals, 0, n - 1)], 0.0) + \
             np.where(ivm, cur[np.clip(i - s_vals, 0, n - 1)], 0.0)
        jvs = (j + s_vals >= 0) & (j + s_vals < n)
        jvm = (j - s_vals >= 0) & (j - s_vals < n)
        Qj = np.where(jvs, cur[np.clip(j + s_vals, 0, n - 1)], 0.0) + \
             np.where(jvm, cur[np.clip(j - s_vals, 0, n - 1)], 0.0)
        new_max = np.max(corr - Pi + Qj)
        delta = (new_max - cur_val) * dx
        if delta < 0 or rng.random() < np.exp(-delta / max(t, 1e-15)):
            cur[i] = 1.0
            cur[j] = 0.0
            corr = corr - Pi + Qj
            cur_val = new_max
            if cur_val < best_val:
                best_val = cur_val
                best = cur.copy()
        t *= cooling
    return best, float(best_val * dx)

def ils_binary(h, n, deadline, seed=42):
    """Iterated Local Search: destroy + greedy repair."""
    rng = np.random.RandomState(seed)
    cur = h.astype(float).copy()
    dx = 2.0 / n
    corr = np.correlate(cur, 1.0 - cur, mode='full')
    cur_val = corr.max()
    best = cur.copy()
    best_val = cur_val
    s_vals = np.arange(-(n - 1), n)
    n_ones = n // 2
    it = 0
    while time.time() < deadline:
        it += 1
        # Destroy: randomly flip k positions
        k = rng.randint(3, 12)
        cand = cur.copy()
        ones = np.where(cand == 1)[0]
        zeros = np.where(cand == 0)[0]
        for _ in range(k):
            if len(ones) < 2 or len(zeros) < 2:
                break
            oi = rng.choice(ones)
            zi = rng.choice(zeros)
            cand[oi] = 0.0
            cand[zi] = 1.0
            ones = np.where(cand == 1)[0]
            zeros = np.where(cand == 0)[0]
        # Repair with quick greedy (limited rounds)
        sub_dl = min(deadline, time.time() + 5)
        cand, cand_val = greedy_binary_vec(cand, n, sub_dl, max_rounds=30)
        if cand_val < best_val:
            best_val = cand_val
            best = cand.copy()
            cur = cand.copy()
    return best, float(best_val * dx)

def slsqp_polish(h, n, deadline, alpha=10000):
    dx = 2.0 / n
    s_vals = np.arange(-(n - 1), n)

    def _obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        m = corr.max()
        shifted = alpha * (corr - m)
        sm = shifted.max()
        exps = np.exp(shifted - sm)
        return m + (sm + np.log(exps.sum())) / alpha

    def _grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        m = corr.max()
        shifted = alpha * (corr - m)
        sm = shifted.max()
        exps = np.exp(shifted - sm)
        weights = exps / exps.sum()
        A = np.zeros(n)
        B = np.zeros(n)
        for k in range(2 * n - 1):
            w = weights[k]
            if w < 1e-12:
                continue
            s = s_vals[k]
            if s >= 0:
                if n - s > 0:
                    A[:n - s] += w * x[s:]
                    B[s:] += w * x[:n - s]
            else:
                sn = -s
                if n - sn > 0:
                    A[sn:] += w * x[:n - sn]
                    B[:n - sn] += w * x[sn:]
        return dx * (1.0 - A - B)

    def _cons(x):
        return x.sum() - n / 2.0

    def _cons_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons, 'jac': _cons_jac}]
    try:
        res = minimize(_obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, jac=_grad,
                       options={'maxiter': 1500, 'ftol': 1e-18})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        n_init = 250
        h_init = np.full(n_init, 0.5)

    best_h = None
    best_c = np.inf
    best_n = n_init

    def _update(hc, c, n):
        nonlocal best_h, best_c, best_n
        if c < best_c:
            best_c = c
            best_h = hc.copy()
            best_n = n

    # Phase 1: Main optimization at n_init
    h = project_feasible(h_init, n_init)
    h_bin = binarize(h, n_init)

    dl1 = t0 + 0.30 * (budget_s - 3)
    hg, cg = greedy_binary_vec(h_bin, n_init, dl1, max_rounds=3000)
    _update(hg, cg, n_init)

    dl2 = t0 + 0.50 * (budget_s - 3)
    if time.time() < dl2:
        hsa, csa = sa_binary(hg, n_init, dl2, iters=500000, t_init=0.01, t_final=0.0001, seed=seed)
        _update(hsa, csa, n_init)

    dl3 = t0 + 0.60 * (budget_s - 3)
    if time.time() < dl3:
        hg2, cg2 = greedy_binary_vec(hsa, n_init, dl3, max_rounds=2000)
        _update(hg2, cg2, n_init)

    # ILS phase
    dl4 = t0 + 0.75 * (budget_s - 3)
    if time.time() < dl4:
        hil, cil = ils_binary(best_h, best_n, dl4, seed=seed + 7)
        _update(hil, cil, best_n)
        if time.time() < dl4:
            hg3, cg3 = greedy_binary_vec(hil, best_n, dl4, max_rounds=1000)
            _update(hg3, cg3, best_n)

    # Continuous polish
    dl5 = t0 + 0.85 * (budget_s - 3)
    if time.time() < dl5:
        hc = slsqp_polish(best_h, best_n, dl5, alpha=10000)
        cc = compute_c5(hc, best_n)
        _update(hc, cc, best_n)
        # Re-binarize and refine
        if time.time() < dl5:
            hrb = binarize(best_h, best_n)
            hg4, cg4 = greedy_binary_vec(hrb, best_n, dl5, max_rounds=500)
            _update(hg4, cg4, best_n)
            if time.time() < dl5:
                hc2 = slsqp_polish(best_h, best_n, dl5, alpha=50000)
                cc2 = compute_c5(hc2, best_n)
                _update(hc2, cc2, best_n)

    # Phase 2: Try other n values
    if time.time() < deadline - 10:
        for n_alt in [300, 400]:
            if time.time() > deadline - 5:
                break
            h_alt = resample_h(h_init, n_init, n_alt)
            h_alt = project_feasible(h_alt, n_alt)
            h_ab = binarize(h_alt, n_alt)
            dl_alt = min(deadline - 3, time.time() + 0.10 * (budget_s - 3))
            hag, cgag = greedy_binary_vec(h_ab, n_alt, dl_alt, max_rounds=1000)
            _update(hag, cgag, n_alt)
            if time.time() < deadline - 3:
                hca = slsqp_polish(best_h if best_n == n_alt else hag, n_alt, deadline - 3, alpha=20000)
                _update(hca, compute_c5(hca, n_alt), n_alt)

    # Final
    best_h = project_feasible(best_h, best_n)
    best_c = compute_c5(best_h, best_n)
    return best_h, best_c, best_n
```
