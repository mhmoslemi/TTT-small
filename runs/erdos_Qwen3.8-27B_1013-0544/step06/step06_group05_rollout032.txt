    c = np.correlate(h, 1.0 - h, mode='full')

    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break

        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = np.clip(h + diff / s * h, 0.0, 1.0)

def simulated_annealing(h_init, n, deadline, seed=42):

    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    
    # Start from binary
    h = h_init.copy()
    if not np.all((h == 0) | (h == 1)):
        sorted_idx = np.argsort(h)[::-1]
        h = np.zeros(n)
        h[sorted_idx[:n//2]] = 1.0

    M = corr.max()
    best_M = M
    best_h = h.copy()
    
    zeros = np.where(h == 0)[0]
    ones = np.where(h == 1)[0]
    
    total_time = deadline - time.time()
    t_start = time.time()
    
    it = 0
    while time.time() < deadline:
        # Cooling schedule: T goes from 1.5 to 0
        progress = (time.time() - t_start) / max(total_time, 1)
        T = max(0.0, 1.5 * (1.0 - progress) ** 2)
        
        it += 1
        if len(zeros) == 0 or len(ones) == 0:
            break
        
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        
        # Compute delta for the max lag
        # After swap: h[i] 0->1, h[j] 1->0
        # Recompute corr (fast for n=250)
        h[i] = 1.0
        h[j] = 0.0

        new_M = new_corr.max()
        
        delta = new_M - M
        if delta <= 0:
            M = new_M
            corr = new_corr
            if M < best_M:
                best_M = M
                best_h = h.copy()
        else:
            # Accept with probability exp(-delta/T)
            if T > 0 and rng.random() < np.exp(-delta / T):
                M = new_M
                corr = new_corr
            else:
                # Reject
                h[i] = 0.0
                h[j] = 1.0
        
        # Update zero/one sets
        # (Do this less frequently for speed)
        if it % 1000 == 0:
            zeros = np.where(h == 0)[0]
            ones = np.where(h == 1)[0]
    
    return best_h, float(best_M * dx)

def greedy_local_search(h_bin, n, deadline, seed=42):

    ones = np.where(h == 1)[0]
    
    no_improve = 0
    max_no_improve = 200000
    
    while time.time() < deadline and no_improve < max_no_improve:
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        
        h[i] = 1.0
        h[j] = 0.0
        new_corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
        new_M = new_corr.max()
        
        if new_M < M:
            M = new_M
            corr = new_corr
            no_improve = 0
            # Update sets
            zi = np.where(zeros == i)[0]
            if len(zi) > 0:
                zeros = np.delete(zeros, zi[0])
            oi = np.where(ones == j)[0]
            if len(oi) > 0:
                ones = np.delete(ones, oi[0])
            zeros = np.append(zeros, j)
            ones = np.append(ones, i)
            if M < best_M:
                best_M = M
                best_h = h.copy()
        else:
            h[i] = 0.0
            h[j] = 1.0
            no_improve += 1
    
    return best_h, float(best_M * dx)

def slsqp_refine(h, n, deadline, seed=42):
    """SLSQP continuous refinement using soft-max."""
    dx = 2.0 / n
    alpha = 5000.0
    
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    
    def cons_fun(x):
        return x.sum() - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 2000, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

            zeros.append(j)
            ones.remove(j)
            ones.append(i)
            if M < best_M:
                best_M = M
                best_h = h.copy()
        else:
            h[i] = 0.0
            h[j] = 1.0
            no_improve += 1
    return best_h, float(best_M * dx)

def slsqp_refine(h_init, n, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        alpha = 8000.0
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h_init, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 3000, 'ftol': 1e-15})
        return project_feasible(res.x, n)
    except Exception:
        return h_init

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)

    h_bin = binarize(h0, n)
    best_h = h_bin.copy()
    best_c = compute_c5(best_h, n)

    # Also try structured initializations
    structured_inits = []
    for p in [7, 11, 13, 17]:
        idx = (np.arange(n) * p) % n
        h_s = np.zeros(n)
        h_s[idx[:n // 2]] = 1.0
        if h_s.sum() == n // 2:
            structured_inits.append(h_s)
    for trial in range(3):
        h_r = np.zeros(n)
        perm = rng.permutation(n)
        h_r[perm[:n // 2]] = 1.0
        structured_inits.append(h_r)

    # Phase 1: SA from multiple starts (45% budget)
    p1 = t0 + 0.45 * (budget_s - 5)
    all_starts = [h_bin] + structured_inits
    starts_per = max(1, int((p1 - t0) / max(len(all_starts), 1)))
    for si, hs in enumerate(all_starts):
        phase_dl = min(p1, t0 + (si + 1) * starts_per)
        if time.time() >= phase_dl:
            break
        hb, cb = sa_optimize(hs.copy(), n, phase_dl, seed=seed + si * 37, t_start_temp=0.5)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()

    # Phase 2: Greedy local search (20% budget)
    p2 = t0 + 0.65 * (budget_s - 5)
    if time.time() < p2:
        hg, cg = greedy_search(best_h.copy(), n, p2, seed=seed + 200)
        if cg < best_c:
            best_c = cg
            best_h = hg.copy()

    # Phase 3: SLSQP continuous polish (20% budget)
    p3 = t0 + 0.85 * (budget_s - 5)
    if time.time() < p3:
        hc = slsqp_refine(best_h.copy(), n, p3)
        cc = compute_c5(hc, n)
        if cc < best_c:
            best_c = cc
            best_h = hc.copy()

    # Phase 4: Re-binarize + SA + greedy (15% budget)
    if time.time() < dl - 2:
        hb2 = binarize(best_h, n)
        p4a = min(dl - 2, t0 + 0.92 * (budget_s - 5))
        hb3, cb3 = sa_optimize(hb2, n, p4a, seed=seed + 500, t_start_temp=0.3)
        if cb3 < best_c:
            best_c = cb3
            best_h = hb3.copy()
        if time.time() < dl - 2:
            hg2, cg2 = greedy_search(best_h.copy(), n, dl - 1, seed=seed + 600)
            if cg2 < best_c:
                best_c = cg2
                best_h = hg2.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
