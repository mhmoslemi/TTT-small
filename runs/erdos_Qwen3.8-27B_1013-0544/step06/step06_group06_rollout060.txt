def solve_continuous_fista(h0, n, deadline):
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    alpha = 50000  # softmax parameter
    L = 2 * dx * n  # Lipschitz constant estimate (rough)
    step = 1.0 / L
    
    t = 1.0
    y = h.copy()
    
    while time.time() < deadline:
        # Compute p_k = softmax(alpha * C)
        c = np.correlate(y, 1 - y, mode='full') * dx
        ac = alpha * c
        ac_max = ac.max()
        p = np.exp(ac - ac_max)
        p /= p.sum()
        
        # Compute gradient: sum_k p_k * grad C[k]
        # grad C[k] at h[i] = (1-h[i+d_k])*[valid] - h[i-d_k]*[valid]
        # where d_k = k - (n-1)
        grad = np.zeros(n)
        for k in range(2*n-1):
            if p[k] < 1e-15:
                continue
            d = k - (n-1)
            if d == 0:
                grad += p[k] * (1 - 2*y)
            else:
                # (1-y[i+d]) for i+d in [0,n-1]
                if d > 0:
                    # i+d valid when i in [0, n-1-d]
                    grad[:n-d] += p[k] * (1 - y[d:])
                    # -y[i-d] for i-d in [0,n-1], i.e., i in [d, n-1]
                    grad[d:] -= p[k] * y[:n-d]
                else:
                    dd = -d
                    # i+d = i-dd valid when i in [dd, n-1]
                    grad[dd:] += p[k] * (1 - y[:-dd])
                    # i-d = i+dd valid when i in [0, n-1-dd]
                    grad[:n-dd] -= p[k] * y[dd:]
        
        # FISTA update
        z = y - step * grad
        # Project onto {sum = n/2, 0 <= h <= 1}
        z = project_feasible(z, n)
        
        t_new = 0.5 * (1 + np.sqrt(1 + 4*t*t))
        y = z + ((t - 1) / t_new) * (z - h)
        h = z
        t = t_new
        
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        
        # Occasionally decrease step size
        step *= 0.999
    
    return best_h, best_c

import numpy as np

import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def project_simple(z, n):
    """Faster projection: bisection on Lagrange multiplier"""
    target = n / 2.0
    def sum_at(tau):
        return np.minimum(1.0, np.maximum(0.0, z - tau)).sum()
    lo, hi = z.min() - 1, z.max() + 1
    for _ in range(60):
        mid = (lo + hi) / 2
        if sum_at(mid) > target:
            lo = mid
        else:
            hi = mid
    return np.minimum(1.0, np.maximum(0.0, z - (lo+hi)/2))

def softmax_grad(h, n, alpha):
    """Compute gradient of softmax-smoothed max correlation."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    ac = alpha * c
    ac -= ac.max()
    p = np.exp(ac)
    p /= p.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-16:

        d = k - (n - 1)
        if d == 0:

        elif d > 0:
            grad[:n-d] += pk * (1.0 - h[d:])

        else:
            dd = -d
            grad[dd:] += pk * (1.0 - h[:-dd])
            grad[:n-dd] -= pk * h[dd:]
    return grad

def fista_optimize(h0, n, deadline, alpha=50000):
    """FISTA optimization of softmax-smoothed max correlation."""
    h = h0.copy()
    y = h0.copy()
    best_h = h0.copy()
    best_c = compute_c5(h, n)
    
    # Estimate Lipschitz constant
    step = 1.0 / (4.0 * n)  # rough initial guess
    t_mom = 1.0
    
    while time.time() < deadline:
        grad = softmax_grad(y, n, alpha)
        
        z = y - step * grad
        z = project_simple(z, n)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y = z + ((t_mom - 1.0) / t_new) * (z - h)

        t_mom = t_new

        if c < best_c - 1e-15:
            best_c = c
            best_h = h.copy()
        
        # Slowly decrease step

    return best_h, best_c

def build_g(h, n):
    ncorr = 2 * n - 1
    g = np.zeros((n, ncorr))
    for p in range(n):
        ks = max(0, p)
        ke = min(ncorr - 1, p + n - 1)
        for k in range(ks, ke + 1):
            idx = k - p
            if 0 <= idx < n and idx != p:
                g[p, k] = 1.0 - 2.0 * h[idx]

    sh = alpha * (c - m)
    sm = sh.max()
