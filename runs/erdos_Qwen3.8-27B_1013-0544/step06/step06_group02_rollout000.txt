        for _ in range(50):
            h_new = h + step * d
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                h, best_c, best_h = h_new, c_new, h_new.copy()
                improved = True
                break
            step *= 0.5
        if not improved:
            alpha *= 1.8
            if alpha > 1e7:
                break
    return best_h, best_c

def fine_pair_search(h, n, deadline, deltas=None):
    if deltas is None:
        deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    best_c = compute_c5(h, n)
    best_h = h.copy()
    alpha = 5000.0
    radius = 80
    for d in deltas:
        while time.time() < deadline:
            improved = False
            g = grad_smooth_max(h, n, alpha)
            # Sort coordinates by |gradient| to prioritize high-impact moves
            order = np.argsort(-np.abs(g))
            for ii in order:
                i = ii
                if time.time() > deadline:
                    return best_h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    for move in [d, -d]:
                        if h[i] + move >= -1e-12 and h[i] + move <= 1.0 + 1e-12 and \
                           h[j] - move >= -1e-12 and h[j] - move <= 1.0 + 1e-12:
                            h[i] += move
                            h[j] -= move
                            c = compute_c5(h, n)
                            if c < best_c - 1e-15:
                                best_c = c
                                best_h = h.copy()
                                improved = True
                            else:
                                h[i] -= move
                                h[j] += move
            if not improved:
                break
    return best_h, best_c

def coord_descent(h, n, deadline):
    """Exact coordinate descent: for each i, find best h[i] given others fixed."""
    best_c = compute_c5(h, n)
    best_h = h.copy()
    dis = list(range(n))
    for _ in range(2000):
        if time.time() > deadline:
            break
        improved_any = False
        for i in dis:
            if time.time() > deadline:
                break
            # Compute A_k and B_k for each lag k involving h[i]
            # c[k] = A_k * h[i] + B_k for each k
            # We need to find t in [0,1] minimizing max_k(A_k*t + B_k)
            # subject to sum constraint handled by penalty
            a_vals = []
            b_vals = []
            for m in range(2 * n - 1):
                k = m - (n - 1)
                if k >= 0:
                    # c = dx * sum_{j=k}^{n-1} h[j]*(1-h[j-k])
                    if i >= k and i <= n - 1:
                        # h[i] appears as h[j] with j=i: contrib = h[i]*(1-h[i-k])
                        a_vals.append(1.0 - h[i - k])
                    if i + k <= n - 1 and i + k >= k:
                        # h[i] appears as 1-h[j-k] with j-k=i, j=i+k: contrib = -h[i+k]*h[i]
                        a_vals.append(-h[i + k])
                else:
                    j_abs = -k
                    if i <= n - 1 - j_abs:
                        a_vals.append(1.0 - h[i + j_abs])
                    if i >= j_abs:
                        a_vals.append(-h[i - j_abs])
            if not a_vals:
                continue
            a_arr = np.array(a_vals)
            dx = 2.0 / n
            # B_k = dx * (c[k] - a_sum * h[i]) where a_sum = sum of a_vals for that k
            # Actually let's just evaluate c at a few candidate t values
            # Try: current value, 0, 1, and intersection points
            t_candidates = [0.0, 1.0, h[i]]
            # Add some grid points
            n_grid = 10
            t_candidates.extend(np.linspace(0, 1, n_grid + 2)[1:-1].tolist())
            # Evaluate and pick best
            orig = h[i]
            best_t = orig
            best_val = np.inf
            sum_rest = h.sum() - orig
            mu = 10.0
            for t in t_candidates:
                h[i] = t
                c = compute_c5_vec(h, n).max()
                pen = mu * (sum_rest + t - n / 2.0) ** 2
                val = c + pen
                if val < best_val:
                    best_val = val
                    best_t = t
            h[i] = best_t
            h = project_feasible(h, n)
            c_new = compute_c5(h, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                best_h = h.copy()
                improved_any = True
        if not improved_any:
            break
    return best_h, best_c

def lbfgs_phase(h, n, deadline, max_iters=300):
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    configs = [(100, 1.0), (500, 5.0), (1000, 10.0), (5000, 50.0),
               (50000, 200.0), (500000, 1000.0), (2000000, 5000.0)]
    for alpha, mu in configs:
        remaining = deadline - time.time()
        if remaining < 5:
            break
        t_sec = min(80, remaining - 3)
        if t_sec < 5:
            break
        try:
            res = minimize(_obj, h, jac=_obj_grad,
                          args=(n, alpha, mu), method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': max_iters, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': max_iters * 15,
                                   'maxls': 50})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            if c_new < best_c - 1e-16:
                best_c = c_new
                h = h_new
        except Exception:
            pass
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(200, 0.5)
        n_init = 200

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)

    # Phase 1: L-BFGS-B (quick convergence)
    t1 = min(t0 + 200, deadline - 400)
    best_h, best_c = lbfgs_phase(best_h, n_init, t1, max_iters=300)

    # Phase 2: Projected gradient with steepest descent direction
    if time.time() < deadline - 300:
        t2 = min(time.time() + 200, deadline - 300)
        h2, c2 = projected_grad_descent(best_h, n_init, t2, alpha_init=500.0)
        if c2 < best_c:
            best_c = c2
            best_h = h2

    # Phase 3: Fine pair search with gradient guidance
    if time.time() < deadline - 200:
        t3 = min(time.time() + 250, deadline - 200)
        best_h, best_c = fine_pair_search(best_h, n_init, t3)

    # Phase 4: Coordinate descent
    if time.time() < deadline - 100:
        t4 = min(time.time() + 150, deadline - 100)
        h4, c4 = coord_descent(best_h, n_init, t4)
        if c4 < best_c:
            best_c = c4
            best_h = h4

    # Phase 5: Another round of fine pair search (even finer)
    if time.time() < deadline - 30:
        t5 = deadline - 5
        best_h, best_c = fine_pair_search(best_h, n_init, t5,
                                          deltas=[0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001])

    # Phase 6: Try n=200 as alternative
    if time.time() < deadline - 40 and n_init != 200:
        x_old = np.linspace(0, 2, n_init)
        x_new = np.linspace(0, 2, 200)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, 200)
        alloc = max(20, deadline - time.time() - 30)
        h_t, c_t = lbfgs_phase(h_interp, 200, time.time() + alloc * 0.5, max_iters=200)
        if time.time() < deadline - 10:
            h_t, c_t = fine_pair_search(h_t, 200, deadline - 5)
        if c_t < best_c:
            best_c = c_t
            best_h = h_t
            n_init = 200

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
