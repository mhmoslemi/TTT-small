def smooth_obj(h, n, alpha, top_k):
    """Minimize the top-k sum of correlation values"""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    # Use log-sum-exp for top-k
    sorted_c = np.sort(c)[::-1]
    m = sorted_c[0]
    sh = alpha * (sorted_c[:top_k] - m)
    sm = sh.max()
    return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

def slsqp_optimize(h, n, alpha, top_k, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        sorted_c = np.sort(c)[::-1]
        m = sorted_c[0]
        sh = alpha * (sorted_c[:top_k] - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / top_k
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})

Actually, let me try a completely different approach for the smooth part: **iterative water-filling**.

The idea:
1. Start with a binary solution
2. Identify the shift k* where the correlation is maximum
3. The correlation at shift k* is: sum_i h(i) * (1-h(i+k*))
4. To reduce this, we want to either:
   - Where h(i) is high and h(i+k*) is low: reduce h(i) or increase h(i+k*)
   - This is like "moving mass" from positions that contribute to the peak to positions that don't

This is essentially a constrained optimization that can be done iteratively:
- At each step, find the peak shift
- Compute the gradient of that particular correlation w.r.t. h
- Take a small step in the negative gradient direction
- Project back to [0,1]^n with sum constraint

The gradient of c[k] = sum_i h(i)*(1-h(i+k)) w.r.t. h is:
- d c[k] / d h(i) = (1-h(i+k)) - h(i-k) [for valid indices]

This is a subgradient of the max. We can use it for subgradient descent.

        c_vals = corr * dx
        m = c_vals.max()

            best_c = m

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def round_to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:target]] = 1.0
    return out

def get_swap_delta(h, n, i, j, m_range):
    delta = np.zeros(2 * n - 1)
    im = i + m_range
    msk = (im >= 0) & (im < n)
    delta[msk] += 1.0 - h[im[msk]]
    imn = i - m_range
    msk = (imn >= 0) & (imn < n)
    delta[msk] -= h[imn[msk]]
    jm = j + m_range
    msk = (jm >= 0) & (jm < n)
    delta[msk] -= 1.0 - h[jm[msk]]
    jmn = j - m_range
    msk = (jmn >= 0) & (jmn < n)
    delta[msk] += h[jmn[msk]]
    delta[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    if 0 <= k1 < 2 * n - 1:
        delta[k1] += 1.0
    k2 = n - 1 + (i - j)
    if 0 <= k2 < 2 * n - 1:
        delta[k2] += 1.0
    return delta
