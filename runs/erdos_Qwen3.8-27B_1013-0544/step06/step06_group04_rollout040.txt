        max_c = cvals.max()
        
        if max_c < best_c:
            best_c = max_c
            best_h = h.copy()
        
        # Find which shifts achieve (near) the maximum
        threshold = max_c - 0.01 * dx
        max_idx = np.where(cvals >= threshold)[0]
        
        # Compute subgradient (average of gradients at max shifts)
        # g_k(h) = dx * sum_i h[i] * (1-h[i+k]) where k = idx - (n-1)
        # dg_k/dh[m] = dx * [(1-h[m+k]) - h[m-k]] for valid indices
        #   where the first term is from h[m] as first factor
        #   and the second term is from h[m] in (1-h[m-k]) as second factor
        
        grad = np.zeros(n)
        for idx in max_idx:
            k = idx - (n - 1)
            for m in range(n):
                g = 0.0
                if 0 <= m + k < n:
                    g += (1 - h[m + k])
                if 0 <= m - k < n:
                    g -= h[m - k]
                grad[m] += dx * g
        grad /= len(max_idx)
        
        # Projected gradient step
        h = h - step_size * grad
        h = project_feasible(h, n)
        
        # Reduce step size
        step_size *= 0.999
    
    return best_h, best_c

Let me vectorize the gradient computation:

For shift k, the gradient of g_k with respect to h[m] is:

- -dx if m-k is valid (from h[m-k]*(1-h[m])): contributes -h[m-k]

This can be computed as:
- First part: (1-h) shifted by -k, zero-padded

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):

            break

            s = room.sum()
            if s < 1e-14:

        else:
            s = h.sum()
            if s < 1e-14:

            h = h + (diff / s) * h

    b[idx[:target]] = 1.0

    grad = np.zeros(n)
    for idx in max_idxs:
        k = int(idx) - (n - 1)

            if k < n:

    best_c = compute_c5(h, n)
    t_start = time.time()

        max_c = float(corr.max())
        if max_c < best_c:
            best_c = max_c
            best_h = h.copy()

        thresh = max_c - 0.05 * dx
        max_idxs = np.where(corr >= thresh)[0]
        if len(max_idxs) == 0:
            max_idxs = np.array([np.argmax(corr)])

        if len(max_idxs) > 50:
            top50 = np.argsort(corr[max_idxs])[-50:]
            max_idxs = max_idxs[top50]
        
        grad = compute_subgrad(h, n, max_idxs)
        
        # Step size: Polyak-like
        alpha = 0.1 / (1 + it * 0.01)
        h = h - alpha * grad
        h = project_feasible(h, n)
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def swap_delta(h, i, j, n):
    delta = np.zeros(2 * n - 1)
    lo, hi = max(0, n-1-i), min(2*n-2, 2*n-2-i)
    if lo <= hi:
        m = np.arange(lo, hi+1)
        delta[m] += 1.0 - h[m-(n-1)+i]
    lo, hi = max(0, i), min(2*n-2, n-1+i)
    if lo <= hi:
        m = np.arange(lo, hi+1)
        delta[m] -= h[n-1+i-m]
    lo, hi = max(0, n-1-j), min(2*n-2, 2*n-2-j)
    if lo <= hi:
        m = np.arange(lo, hi+1)
        delta[m] -= 1.0 - h[m-(n-1)+j]
    lo, hi = max(0, j), min(2*n-2, n-1+j)
    if lo <= hi:
        m = np.arange(lo, hi+1)
        delta[m] += h[n-1+j-m]
    return delta

def local_search(h, n, deadline, max_sweeps=5000, seed=42):
    h = binarize(h, n)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full').copy()
    cur_c = float(corr.max()) * dx
    best_c = cur_c
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(h == 1)[0].copy()
        zeros = np.where(h == 0)[0].copy()
        rng.shuffle(ones)
        rng.shuffle(zeros)
        for i in zeros:
            if time.time() > deadline:
                break
            i = int(i)
            for j in ones:
                j = int(j)
                d = swap_delta(h, i, j, n)
                nc = float((corr + d).max()) * dx
                if nc < cur_c - 1e-14:
                    h[i] = 1.0
                    h[j] = 0.0
                    corr = corr + d
                    cur_c = nc
                    if cur_c < best_c:
                        best_c = cur_c
                        best_h = h.copy()
                    improved = True
                    break
        if not improved:
            break
    return best_h, best_c

def softmax_optimize(h, n, deadline, alpha=500, max_iter=20000):
    """Smoothed optimization using log-sum-exp surrogate."""
    h = h.astype(float).copy()
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1 - h, mode='full') * dx
        max_c = float(corr.max())
        if max_c < best_c:
            best_c = max_c
            best_h = h.copy()
        
        # Softmax weights for gradient
        shifted = alpha * (corr - max_c)
        exp_vals = np.exp(shifted)
        p = exp_vals / exp_vals.sum()
        
        # Compute weighted gradient
        grad = np.zeros(n)
        for k_idx in range(len(p)):
            if p[k_idx] < 1e-8:
                continue
            k = k_idx - (n - 1)
            w = p[k_idx]
            if k >= 0:
                if n - k > 0:
                    grad[:n-k] += w * (1.0 - h[k:])
                if k < n:
                    grad[k:] -= w * h[:n-k]
            else:
                kp = -k
                if n - kp > 0:
                    grad[kp:] += w * (1.0 - h[:n-kp])
                    grad[:n-kp] -= w * h[kp:]
        grad *= dx
        
        # Adaptive step size
        step = 0.3 / (1 + it * 0.002)
        h = h - step * grad
        h = project_feasible(h, n)
    
    return best_h, best_c

def subgrad_optimize(h, n, deadline, max_iter=20000):
    """Subgradient descent on the max function."""
    h = h.astype(float).copy()
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1 - h, mode='full') * dx
        max_c = float(corr.max())
        if max_c < best_c:
            best_c = max_c
            best_h = h.copy()
        
        # Near-max shifts
        thresh = max_c - 0.03 * dx
        max_idxs = np.where(corr >= thresh)[0]
        if len(max_idxs) == 0:
            max_idxs = np.array([np.argmax(corr)])
        
        grad = np.zeros(n)
        for idx in max_idxs:
            k = int(idx) - (n - 1)
            if k >= 0:
                if n - k > 0:
                    grad[:n-k] += (1.0 - h[k:])
                if k < n:
                    grad[k:] -= h[:n-k]
            else:
                kp = -k
                if n - kp > 0:
                    grad[kp:] += (1.0 - h[:n-kp])
                    grad[:n-kp] -= h[kp:]
        grad /= len(max_idxs)
        grad *= dx
        
        step = 0.5 / (1 + it * 0.003)
        h = h - step * grad
        h = project_feasible(h, n)
    
    return best_h, best_c

def perturb(h, n, rng, strength=5):
    h_p = h.copy()
    ones = np.where(h_p >= 0.5)[0]
    zeros = np.where(h_p < 0.5)[0]
    for _ in range(strength):
        if len(ones) > 0 and len(zeros) > 0:
            a = int(rng.choice(ones))
            b = int(rng.choice(zeros))
            h_p[a], h_p[b] = h_p[b], h_p[a]
    return h_p

def run(seed=42, budget_s=1100, **kwargs):
