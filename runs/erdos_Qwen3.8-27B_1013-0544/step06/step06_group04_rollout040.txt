```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    h = np.clip(x, 0.0, 1.0)
    g = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    m = g.max()
    s = alpha * (g - m)
    val = m + np.log(np.exp(s).sum()) / alpha
    # constraint penalty
    viol = x.sum() - n / 2.0
    val += 1000.0 * viol * viol
    return val

def optimize_smooth(h0, n, deadline, alphas, seed=42):
    best_h = h0.copy()
    best_c = compute_c5(h0, n)
    rng = np.random.RandomState(seed)
    for alpha in alphas:
        if time.time() > deadline - 5:
            break
        # Try multiple starting points
        for restart in range(5):
            if time.time() > deadline - 5:
                break
            if restart == 0:
                x0 = best_h.copy()
            else:
                x0 = best_h + rng.randn(n) * 0.05
                x0 = project(x0, n)
            try:
                res = minimize(obj_smooth, x0, args=(n, alpha),
                             method='L-BFGS-B', bounds=[(0,1)]*n,
                             options={'maxiter': 200, 'ftol': 1e-16})
                h_new = project(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new.copy()
            except Exception:
                pass
    return best_h, best_c
```

```python
def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13: break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def binarize(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def swap_delta(h, i, j, n):
    delta = np.zeros(2 * n - 1)
    lo = n - 1 - i
    hi = 2 * n - 2 - i
    ks = np.arange(lo, hi + 1)
    delta[ks] += (1.0 - h[ks - (n-1) + i])
    lo = n - 1 - j
    hi = 2 * n - 2 - j
    ks = np.arange(lo, hi + 1)
    delta[ks] -= (1.0 - h[ks - (n-1) + j])
    return delta
```

q_k = 1 if k ∈ K* (worst lags), 0 otherwise (or use softmax weights for smoothing)

Then ∂f/∂h ≈ convolve(q, [1,0,...,0,-1]) * dx (something like that)

Actually, let me just compute it directly:
∂g_k/∂h_i = dx * [(1-h[i+k-n+1]) if 0≤i+k-n+1<n] - dx * [h[i-k+n-1] if 0≤i-k+n-1<n]

The first term: for each i, sum over k of q_k * (1-h[i+k-n+1]) * dx
= dx * ∑_k q_k * (1-h[i+k-n+1]) (for valid indices)
= dx * (∑_k q_k - convolve(q, h)[appropriate shift])

The second term: for each i, sum over k of q_k * h[i-k+n-1] * dx
= dx * convolve(q, h)[appropriate shift]

So the gradient is: dx * (∑q_k * 1_vec - 2 * convolve(q, h, 'full')[shift])

This is O(n) to compute! Much faster than numerical gradients.
