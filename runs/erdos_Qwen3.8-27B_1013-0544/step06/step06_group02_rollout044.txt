    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def grad_smooth_max(h, n, alpha, top_k=None):
    c = compute_c5_vec(h, n)
    if top_k is not None and top_k < len(c):
        idx = np.argsort(c)[-top_k:]
        c_use = np.full_like(c, -1e6)
        c_use[idx] = c[idx]

        c_use = c
    m = c_use.max()
    sh = alpha * (c_use - m)
    mx = sh.max()
    w = np.exp(sh - mx)
    w /= w.sum()
    dx = 2.0 / n
    grad = np.zeros(n)
    for mi in range(2 * n - 1):
        if w[mi] < 1e-13:

        wm = w[mi] * dx

            nv = n - k

    c = compute_c5_vec(x, n)
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    if mx > 500:
        val = m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha
    else:
        val = m + np.log(np.sum(np.exp(sh))) / alpha
    val += mu * (x.sum() - n / 2.0) ** 2
    return float(val)

def smooth_max_grad(x, n, alpha, mu):
    g = grad_smooth_max(x, n, alpha)
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return g

        val = m + np.log(np.sum(np.exp(sh))) / alpha
    return float(val)

def slsqp_constraint(x, n):

def slsqp_jac(x, n):
    return np.ones(n)

def lbfgs_phase(h, n, deadline, max_iters=300):

                h2 = best_h.copy()
                h2[start:start+block_size] = np.clip(res.x, 0, 1)
                h2 = project_feasible(h2, N)
                c2 = compute_c5(h2, N)
                if c2 < best_c - 1e-14:
                    best_c = c2
                    best_h = h2
                    improved_any = True
            except Exception:
                pass
        if not improved_any:
            alpha = max(alpha * 2, 100)
            if alpha > 500000:
                break
    return best_h, best_c

def fine_pair_search(h, n, deadline):
    best_c = compute_c5(h, n)
    delta_sequence = [0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    for d in delta_sequence:
        while time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - 80)
                hi = min(n, i + 81)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-14:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
            if not improved:
                break
    return h, best_c

def perturb_and_reoptimize(h, n, deadline, rng, amplitude=0.02):
    best_c = compute_c5(h, n)
    perturbed = h.copy()
    mask = rng.rand(n) < 0.3
    perturbed[mask] += rng.randn(mask.sum()) * amplitude
    perturbed = project_feasible(perturbed, n)
    c_pert = compute_c5(perturbed, n)
    alloc = min(80, deadline - time.time() - 10)
    if alloc < 20:
        return h, best_c
    h_opt, c_opt = lbfgs_phase(perturbed, n, time.time() + alloc * 0.7, max_iters=150)
    if time.time() < deadline - 15:
        h_opt, c_opt = slsqp_phase(h_opt, n, time.time() + min(20, deadline - time.time() - 10),
                                   alpha=1000000, maxiter=1000)
    if c_opt < best_c:
        return h_opt, c_opt
    return h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = np.full(300, 0.5)
        n_init = 300

    best_h = project_feasible(h_init, n_init)
    best_c = compute_c5(best_h, n_init)

    # Phase 1: L-BFGS-B with increasing alpha
    t1 = min(t0 + 200, deadline - 400)
    best_h, best_c = lbfgs_phase(best_h, n_init, t1, max_iters=250)

    # Phase 2: SLSQP refinement
    if time.time() < deadline - 300:
        t2 = min(time.time() + 100, deadline - 300)
        for a in [200000, 1000000, 5000000]:
            if time.time() > t2:
                break
            h2, c2 = slsqp_phase(best_h, n_init, min(time.time() + 40, t2), alpha=a, maxiter=2000)
            if c2 < best_c:
                best_c = c2
                best_h = h2

    # Phase 3: Block descent
    if time.time() < deadline - 200:
        t3 = min(time.time() + 60, deadline - 200)
        best_h, best_c = block_descent(best_h, n_init, t3, block_size=6, alpha=10000)

    # Phase 4: Perturbation + re-optimization (escape local minima)
    if time.time() < deadline - 150:
        for _ in range(3):
            if time.time() > deadline - 60:
                break
            ph, pc = perturb_and_reoptimize(best_h, n_init, time.time() + 80, rng, amplitude=0.015)
            if pc < best_c:
                best_c = pc
                best_h = ph

    # Phase 5: Final L-BFGS
    if time.time() < deadline - 100:
        best_h, best_c = lbfgs_phase(best_h, n_init, min(time.time() + 50, deadline - 60), max_iters=100)

    # Phase 6: Try higher resolution
    if time.time() < deadline - 120:
        n_try = n_init * 2
        if n_try <= 800:
            x_old = np.linspace(0, 2, n_init)
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp, n_try)
            alloc = min(80, deadline - time.time() - 60)
            if alloc > 30:
                h_t, c_t = lbfgs_phase(h_interp, n_try, time.time() + alloc * 0.6, max_iters=200)
                if time.time() < deadline - 30:
                    h_t, c_t = slsqp_phase(h_t, n_try, time.time() + min(30, deadline - time.time() - 20),
                                           alpha=2000000, maxiter=1500)
                if time.time() < deadline - 15:
                    h_t, c_t = fine_pair_search(h_t, n_try, time.time() + min(15, deadline - time.time() - 8))
                if c_t < best_c:
                    best_c = c_t
                    best_h = h_t

    # Phase 7: Fine pair search on best
    if time.time() < deadline - 10:
        best_h, best_c = fine_pair_search(best_h, len(best_h), deadline - 3)

    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
